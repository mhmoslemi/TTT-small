def project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def compute_score(a):
    conv = np.convolve(a, a, mode='full')
    maxc = conv.max()
    s = a.sum()
    return 2.0 * len(a) * maxc / (s*s)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s)-0.05)

    rng = np.random.default_rng(seed)

    if "height_sequence_1" in globals():
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
        parent_raw = np.clip(parent_raw, 0.0, 1000.0)
        if parent_raw.sum() < 0.01:
            parent = None
        else:
            parent = parent_raw / parent_raw.sum()
    else:
        parent = None

    n = len(parent_raw) if parent is not None else 2000  # default if no parent
    # uniform
    uniform = np.full(n, 1.0/n, dtype=np.float64)

    # initial best
    best_vec = uniform.copy()
    best_score = compute_score(best_vec)

    if parent is not None:
        parent_score = compute_score(parent)
        if parent_score < best_score:
            best_vec = parent.copy()
            best_score = parent_score

    # main loop
    a = best_vec.copy()
    lr = 0.1
    no_improve = 0
    iter = 0
    max_iters = 20000

    while time.monotonic() < deadline and iter < max_iters:
        iter += 1

        # compute convolution and max offset
        conv = np.convolve(a, a, mode='full')
        maxc = conv.max()
        # find all indices achieving max (subgradient)
        max_idxs = np.flatnonzero(conv == maxc)
        # compute gradient as average of gradients for each max index
        grad = np.zeros_like(a)
        for k in max_idxs:
            # vectorized assignment for each k
            start_i = max(0, k - (n-1))
            end_i = min(k, n-1)
            idx = np.arange(start_i, end_i+1)
            grad[idx] += 2.0 * a[k - idx]
        grad /= len(max_idxs)

        # gradient descent step
        a_new = a - lr * grad

        # projector onto simplex
        a_new = project_simplex(a_new)

        # occasionally random perturbation
        if rng.random() < 0.02:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = lr * a_new[i]  # move fraction proportional to lr
                a_new[i] -= delta
                a_new[j] += delta
                a_new = project_simplex(a_new)

        # compute score
        score = compute_score(a_new)

        # accept if improves
        if score < best_score:
            best_score = score
            best_vec = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # update a for next iteration: either accept new or keep old
        a = a_new

        # adapt learning rate
        if no_improve >= 200:
            lr *= 0.5
            no_improve = 0
        if lr < 1e-5:

    # final fallback ensure clipping
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    # ensure sum >= 0.01
    if best_vec.sum() < 0.01:
        best_vec = uniform.copy()
    return [float(x) for x in best_vec.tolist()]

import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex {x>=0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    """Return the AC1 upper bound for a non‑negative sequence a."""
    # a is a NumPy 1‑d array
    conv = np.convolve(a, a, mode='full')
    maxc = conv.max()
    s = a.sum()
    # safeguard division (s is never zero after projection)
    return 2.0 * a.shape[0] * maxc / (s * s)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) that tries to minimise
    the AC1 upper bound as measured by ``evaluate_sequence``.
    """
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.05)

    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Determine problem size and initialise candidates
    if "height_sequence_1" in globals():
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
        parent_raw = np.clip(parent_raw, 0.0, 1000.0)
        n = parent_raw.size
        if parent_raw.sum() >= 0.01:
            parent = parent_raw / parent_raw.sum()
        else:
            parent = None
    else:
        parent = None
        n = 2000

    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # start from the better of uniform / parent
    best_vec = uniform.copy()
    best_score = _score(best_vec)
    if parent is not None:
        parent_score = _score(parent)
        if parent_score < best_score:
            best_vec = parent.copy()
            best_score = parent_score

    # ----------------------------------------------------------------------
    # Main optimisation loop (projected gradient descent with occasional
    # random nudges)
    a = best_vec.copy()
    conv = np.convolve(a, a, mode='full')
    cur_score = best_score

    lr = 0.12                 # initial step size
    no_improve = 0
    it = 0
    max_iters = 12000

    while time.monotonic() < deadline and it < max_iters:
        it += 1

        # Identify indices attaining the current maximal convolution value
        max_val = conv.max()
        max_idxs = np.flatnonzero(conv == max_val)

        # Compute sub‑gradient (average over all active indices)
        grad = np.zeros_like(a)
        for k in max_idxs:
            start_i = max(0, k - (n - 1))
            end_i = min(k, n - 1)
            idx = np.arange(start_i, end_i + 1)
            grad[idx] += 2.0 * a[k - idx]
        grad /= float(len(max_idxs))

        # Gradient step and simplex projection
        a_new = a - lr * grad
        a_new = _project_simplex(a_new)

        # Small random perturbation (helps escape flat regions)
        if rng.random() < 0.018:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = lr * a_new[i]
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_simplex(a_new)

        # Evaluate new point
        conv_new = np.convolve(a_new, a_new, mode='full')
        new_score = _score(a_new)

        # Keep the best ever seen
        if new_score < best_score:
            best_score = new_score
            best_vec = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Prepare for next iteration
        a = a_new
        conv = conv_new
        cur_score = new_score

        # Simple step‑size adaptation
        if no_improve >= 300:
            lr *= 0.55
            no_improve = 0
        if lr < 1e-5:
            lr = 1e-5

    # ----------------------------------------------------------------------
    # Final safety clipping (evaluator also clips, but we stay inside limits)
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # Ensure a non‑trivial sum (should already be ~1)
    if best_vec.sum() < 0.01:
        best_vec = uniform.copy()

    return [float(x) for x in best_vec.tolist()]
