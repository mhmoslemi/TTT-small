    if not isinstance(sequence, list):

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _max_convolution_fft(a):
    """Linear convolution a*a using FFT, returning the full (2n‑1) result."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two >= 2n‑1
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    # tiny negative round‑off values can appear; clamp them
    conv[conv < 0] = 0.0
    return conv

# ----------------------------------------------------------------------
def _compute_score(a):
    """Exact score used by the evaluator (avoids a second convolution)."""
    a = np.asarray(a, dtype=np.float64)
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = _max_convolution_fft(a)
    max_b = conv.max()
    n = a.shape[0]
    return float(2.0 * n * max_b / (sum_a * sum_a))

    Return a non‑negative step‑function (list of floats) that seeks to

    """

    # keep a safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.05)

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000
