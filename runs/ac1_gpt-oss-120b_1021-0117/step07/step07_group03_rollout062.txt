def _conv_smooth_max_grad(a, L, T):
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*n-1]
    # compute smooth weights
    # w_i = exp(conv_i / T)
    # subtract max for numerical stability
    m = conv.max()
    exp_vals = np.exp((conv - m) / T)
    w = exp_vals / exp_vals.sum()
    # gradient
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2*n-2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    smooth_max = (w * conv).sum()
    return smooth_max, grad
```

    Tmin = 5e-5

    best_vec = a.copy()

    no_improve = 0
    while True:

            break
        it += 1
        # compute conv and max
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        v = v.ravel()
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1

            break
        it += 1
        # FFT of current a
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        cur_max = conv.max()
        cur_bound = 2.0 * n * cur_max
        # temperature
        T = max(T0 * (0.9995 ** it), Tmin)

        # shift for numerical stability

                best_vec = a.copy()

import time, math, numpy as np

def _project_simplex(v):
    """ Euclidean projection of a vector onto the probability simplex (sum=1, v>=0). """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
