import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    # ensure non-negative
    v = np.maximum(v, 0)
    # project onto simplex sum=1
    if np.abs(v.sum() - 1) < 1e-12:
        return v
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(a):
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _max_conv_and_grad(a):
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)

import time, math, numpy as np


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.maximum(v, 0.0)
    if abs(v.sum() - 1.0) < 1e-12:
        return v
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_vec(a: np.ndarray) -> float:
    """Score for a vector whose sum is 1 (returns 2·n·max(conv))."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _max_conv_and_grad(a: np.ndarray):
    """
    Compute the maximum value of the linear convolution a*a and the
    sub‑gradient of that maximum (exact, using the active index only).
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    k = int(conv.argmax())

    # gradient of c[k] w.r.t a_j : 2*a_{k-j} if the index is in range
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    i_vals = np.arange(i_start, i_end + 1)
    j_vals = k - i_vals
    grad[j_vals] = 2.0 * a[i_vals]
    return max_val, grad


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ----- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        a0 = rng.random(2500, dtype=np.float64)

    cur = _project_simplex(np.maximum(a0, 0.0))
    # start from a symmetric point – many good solutions are symmetric
    cur = _project_simplex(0.5 * (cur + cur[::-1]))
    n = cur.size
    cur_sc = _score_vec(cur)

    best_vec = cur.copy()
    best_sc = cur_sc

    # ----- optimizer state ------------------------------------------------
    momentum = np.zeros_like(cur)
    beta = 0.95                 # momentum coefficient
    step0 = 1.0                 # initial step size (will decay)
    it = 0
    no_improve = 0
    target = 1.50274349

    # parameters for occasional random perturbations
    stagnation_limit = 2000
    perturb_frac = 0.4

    while time.monotonic() < deadline:
        it += 1

        # compute max convolution and its exact sub‑gradient
        max_val, grad = _max_conv_and_grad(cur)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size schedule (1/√t)
        step = step0 / math.sqrt(it)

        # candidate step
        cand = _project_simplex(cur - step * momentum)
        cand_sc = _score_vec(cand)

        # simple back‑tracking if the step was too aggressive
        if cand_sc > cur_sc:
            step2 = step * 0.5
            cand2 = _project_simplex(cur - step2 * momentum)
            cand2_sc = _score_vec(cand2)
            if cand2_sc < cand_sc:
                cand, cand_sc = cand2, cand2_sc

        # accept improvement
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1

        # keep the best ever seen
        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()

        # periodic symmetrisation (helps many optimal patterns)
        if it % 300 == 0:
            sym = _project_simplex(0.5 * (cur + cur[::-1]))
            sym_sc = _score_vec(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
            if sym_sc < best_sc:
                best_sc, best_vec = sym_sc, sym.copy()

        # random local redistribution when the search stalls
        if no_improve > stagnation_limit:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * perturb_frac * rng.random()
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_vec(tmp)
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
            else:
                # occasional full random restart
                if rng.random() < 0.15:
                    rnd = _project_simplex(rng.random(n))
                    rnd_sc = _score_vec(rnd)
                    if rnd_sc < best_sc:
                        best_sc, best_vec = rnd_sc, rnd.copy()
                    cur, cur_sc = rnd, rnd_sc
                    momentum.fill(0.0)
                no_improve = 0

        # early exit if we have already reached the target bound
        if best_sc < target:
            break

        # keep an eye on the deadline
        if it % 1000 == 0 and time.monotonic() >= deadline:
            break

    # ----- final polishing: blend a little with the uniform vector --------
    if best_sc > target:
        uniform = np.full(n, 1.0 / n)
        for alpha in (0.01, 0.02, 0.05, 0.1):
            cand = _project_simplex((1.0 - alpha) * best_vec + alpha * uniform)
            cand_sc = _score_vec(cand)
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                if best_sc < target:
                    break

    return [float(x) for x in best_vec]
