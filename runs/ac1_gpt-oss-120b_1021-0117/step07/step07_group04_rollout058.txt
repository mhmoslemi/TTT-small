import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n+1) > 0
    rho = np.nonzero(rho_arr)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # fix rounding errors
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

def _score_fft(a, n, L):
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

        s = parent.sum()
        if s > 0.0:

            candidates.append(parent)

    candidates.append(np.full(n, 1.0 / n))

    rand = np.maximum(rand, 0.0)
    rand = rand / rand.sum()
    candidates.append(rand)

    xs = np.arange(n, dtype=np.float64)
    sine = np.sin(np.pi * (xs + 1) / (n + 1))
    candidates.append(sine / sine.sum())

    # Exponential decay
    exp = np.exp(-0.004 * xs)
    candidates.append(exp / exp.sum())

    # Choose best start
    best_vec = None
    best_sc = float('inf')
    for vec in candidates:
        sc = _score_fft(vec, n, L)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()
    cur = best_vec.copy()
    cur_sc = best_sc

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.5   # base learning rate factor
    it = 0
    no_improve = 0
    target = 1.50274349

    # Main loop
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # FFT of current vector
        a_fft = np.fft.rfft(cur, n=L)

        # Autoconvolution
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        maxc = conv.max()

        # Smoothed max weights
        alpha = min(60.0, 2.0 + 0.02 * it)   # increase slowly
        exp_arg = alpha * (conv - maxc)    # max exponent = 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum

        # Gradient of smoothed max
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_lr / math.sqrt(it)   # modest decay
        update = step * m_hat / (np.sqrt(v_hat) + eps)
        cand = cur - update
        cand = _project_simplex(cand)

        cand_sc = _score_fft(cand, n, L)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1
            # occasional acceptance of worse move
            if rng.random() < 0.001:
                cur, cur_sc = cand, cand_sc
                no_improve = 0

        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()

        # Symmetrize periodically
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fft(sym, n, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        if no_improve > 2500:
            perturb = rng.normal(scale=0.02, size=n)
            tmp = cur + perturb
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_sc = _score_fft(tmp, n, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                no_improve = 0
            else:
                # also try swapping a small mass between random indices
                i = rng.integers(0, n)
                if cur[i] > 0:
                    delta = cur[i] * rng.random() * 0.3
                    j = rng.integers(0, n-1)
                    if j >= i:
                        j += 1
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score_fft(tmp, n, L)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                        if tmp_sc < best_sc:
                            best_sc, best_vec = tmp_sc, tmp.copy()
                        no_improve = 0

        # Early exit if target reached
        if best_sc < target:
            break

    # Return best found as list of floats
    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # fix tiny rounding errors so the sum is exactly 1
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

def _score_fft(a, n, L):
    """Fast evaluation of the AC1 upper bound (assumes a ≥ 0)."""
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))
