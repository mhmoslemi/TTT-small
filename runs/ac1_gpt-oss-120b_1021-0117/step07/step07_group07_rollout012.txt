def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0., float(budget_s)-0.2)  # margin
    # Initialize
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0)  # ensure nonneg
    else:
        n = rng.integers(2000,3000)
        a = rng.random(n)
    # Project onto simplex (so sum=1)
    a = _project_simplex(a)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    # Learning rate schedule
    lr0 = 0.8  # baseline
    # keep best
    best_a = a.copy()
    best_score = _score(best_a)
    cur_score = best_score
    iteration = 0
    no_improve = 0

    while time.monotonic() < deadline:
        iteration += 1
        # Compute convolution and gradient
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        kmax = conv.argmax()

        # Compute gradient based on max index
        grad = np.zeros_like(a)
        start_idx = max(0, kmax - (n - 1))
        end_idx = min(n - 1, kmax)
        # indices i from start_idx to end_idx
        i_range = np.arange(start_idx, end_idx + 1)
        grad[i_range] = 2.0 * a[kmax - i_range]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        step = lr0 / (np.sqrt(v_hat) + eps)

        cand = a - step * m_hat
        cand = _project_simplex(cand)

        cand_score = _score(cand)

        if cand_score < cur_score:
            a, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best_a, best_score = cand.copy(), cand_score
        else:
            no_improve += 1
            # occasional accept
            if rng.random() < 0.001:
                a, cur_score = cand, cand_score

        # Periodic symmetrisation
        if iteration % 400 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score(sym)
            if sym_score < cur_score:
                a, cur_score = sym, sym_score
                no_improve = 0
                if sym_score < best_score:
                    best_a, best_score = sym.copy(), sym_score

                    j += 1

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score(a):
    """Score for a vector already normalized to sum=1 (2 n * max convolution)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
