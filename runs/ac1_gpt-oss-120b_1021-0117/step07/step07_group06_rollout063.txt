import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w


def _max_conv(a):
    """Maximum entry of the full convolution a*a (a is non‑negative and sums to 1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv.max()


def _score(a):
    """Score used by the evaluator (sum is 1, so the factor 2·n·max)."""
    n = a.shape[0]
    return float(2.0 * n * _max_conv(a))


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Gradient‑based search for a non‑negative step‑function that minimizes the
    AC1 upper bound. Returns a list of coefficients.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # initialise
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        cur = _project_simplex(np.maximum(np.asarray(height_sequence_1, dtype=np.float64), 0.0))
    else:
        n = 2000
        cur = np.full(n, 1.0 / n, dtype=np.float64)  # uniform start

    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # Adam optimiser state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    # soft‑max schedule for the max
    alpha0 = 1.0
    alpha_max = 80.0

    # step‑size scale (initial)
    base_lr = 0.4

    # early‑stop target (the known benchmark)
    target = 1.50274349

    no_improve = 0
    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        t += 1

        # ----- FFT of current vector -----
        a_fft = np.fft.rfft(cur, n=L)

        # ----- convolution (auto‑correlation) -----
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ----- soft‑max weighting of the convolution entries -----
        alpha = min(alpha_max, alpha0 * math.log1p(t))
        w = np.exp(alpha * (conv_full - max_val))
        w_sum = w.sum()
        if w_sum == 0.0:
            # fallback: uniform weight on the maximal entries
            w = (conv_full == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient of the weighted max‑convolution -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)

        cand_sc = _score(cand)

        # ----- greedy acceptance -----
        if cand_sc < cur_sc:

        # ----- early exit if target reached -----
        if best_sc < target:
            break

        # safeguard: keep time in check
        if time.monotonic() >= deadline:
            break
    # ------------------------------------------------------------------
    # return the best vector found (as plain Python floats)
    return [float(x) for x in best_vec]
