def project_simplex(v):
    # v: array, we want x >=0, sum x = 1
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    ind = np.arange(1, n+1)

    if not np.any(cond):
        theta = 0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum==1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_and_info(a, N_fft, n):
    """
    Compute the AC1 upper‑bound score for a, together with the full convolution
    (first 2n‑1 entries) and the index of the maximal entry.
    """
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float("inf"), None, None
    fa = np.fft.rfft(a, n=N_fft)
    conv = np.fft.irfft(fa * fa, n=N_fft)[: 2 * n - 1]
    max_c = np.max(conv)
    k0 = int(np.argmax(conv))
    score = float(2 * n * max_c / (sum_a * sum_a))
    return score, conv, k0

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that attempts to
    minimise the AC1 upper bound as measured by ``evaluate_sequence``.
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # ------------------------------------------------------------------
    # Problem size and initial point
    if "height_sequence_1" in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64)
        n = init.shape[0]
        a = np.clip(init, 0.0, 1000.0)
        s = a.sum()
        a = a / s if s > 1e-12 else np.full(n, 1.0 / n)
    else:
        n = 2000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # Pre‑compute FFT length (next power of two)
    N_fft = 1 << (2 * n - 1).bit_length()

    # Best solution tracking
    cur = a.copy()
    cur_score, _, cur_k0 = _score_and_info(cur, N_fft, n)
    best = cur.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # Optimisation parameters
    rng = np.random.default_rng(seed)
    max_iters = 400_000
    T0 = 1.0            # initial temperature
    T_end = 1e-4        # final temperature
    decay = (T_end / T0) ** (1.0 / max_iters)
    T = T0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        it += 1
        r = rng.random()

        if r < 0.6:                      # gradient descent step
            # gradient based on current maximal convolution index
            grad = np.zeros_like(cur)
            low = max(0, cur_k0 - (n - 1))
            high = min(n - 1, cur_k0)
            idx = np.arange(low, high + 1, dtype=np.int64)
            grad[idx] = 2.0 * cur[cur_k0 - idx]
            eta = 0.1 * T
            cand = cur - eta * grad
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)

        elif r < 0.9:                    # random mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = cur[i] * rng.random() * 0.2   # up to 20 % of source mass
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

            # Dirichlet draws (alpha=0.5 gives sparse vectors)
            cand = rng.dirichlet(np.full(n, 0.5))

        cand_score, _, cand_k0 = _score_and_info(cand, N_fft, n)
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / (T + 1e-12)):
            cur = cand
            cur_score = cand_score
            cur_k0 = cand_k0
            if cand_score < best_score:

                best_score = cand_score

    try:
        final_score = evaluate_sequence(best.tolist())
        if final_score < best_score:
            best_score = final_score
    except Exception:
        pass

    # Return as plain Python list of floats
    return [float(x) for x in best]
