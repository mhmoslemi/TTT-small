import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _objective(x, n, L):
    """Return the AC1 upper bound for x (no scaling assumed)."""
    xf = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(xf * xf, n=L)[: 2 * n - 1]
    maxv = conv.max()
    s = x.sum()
    if s == 0.0:
        return np.inf
    return 2.0 * n * maxv / (s * s)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    n = a.size

        # ---------- sub‑gradient ----------
