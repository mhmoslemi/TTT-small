import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    """Score for a vector already normalized to sum=1 (2*n*max(conv))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        if w_sum == 0 or not np.isfinite(w_sum):
            # fallback to hard max mask
            mask = conv >= max_val - max_val * 1e-12 - 1e-15
            w = mask.astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient of the smoothed max (2 * a * w_rev convolution) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam style update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (math.sqrt(it) + 1.0)
        update = step * m_hat / (np.sqrt(v_hat) + eps)

        # ----- two candidate proposals -----
        cand_add = _project_simplex(a - update)

        exp_factor = np.exp(-step * grad)   # multiplicative move
        cand_mul = a * exp_factor
        s = cand_mul.sum()
        if s > 0:
            cand_mul = cand_mul / s
        else:
            cand_mul = a.copy()

        cand_add_sc = _score(cand_add)
        cand_mul_sc = _score(cand_mul)

        if cand_mul_sc < cand_add_sc:
            cand, cand_sc = cand_mul, cand_mul_sc
        else:
            cand, cand_sc = cand_add, cand_add_sc

        # ----- acceptance logic -----
        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc
            a = cand
            no_improve = 0
        elif cand_sc < _score(a):
            a = cand
            no_improve = 0
        else:
            no_improve += 1
            # occasional probabilistic acceptance
            prob = math.exp(-(cand_sc - best_sc) / (step + 1e-12))
            if rng.random() < prob:
                a = cand

        # ----- symmetrisation every few hundred steps -----
        if it % 400 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best = sym.copy()
                best_sc = sym_sc
                a = sym
                no_improve = 0

        # ----- random perturbation when stuck -----
        if no_improve > 2500:
            # redistribute a small fraction of mass between two random indices
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.35
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                a = tmp
            no_improve = 0

        # ----- restart from a jittered best when really stuck -----
        if no_improve > 5000:
            jitter = rng.normal(scale=0.02, size=n)
            jitter = np.maximum(jitter, -a)          # keep non‑negative
            tmp = a + jitter
            tmp = _project_simplex(tmp)
            a = 0.6 * best + 0.4 * tmp
            a = _project_simplex(a)
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # Return the best sequence found (as plain Python floats)
    return [float(x) for x in best]
