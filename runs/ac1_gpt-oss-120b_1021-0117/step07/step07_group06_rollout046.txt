import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score(a):
    """Score assuming sum(a)=1; returns 2*n*max(conv)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # safety margin

    # ------------------------------------------------------------------
    # initialization
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64) + 1e-3
    cur = _project_simplex(np.maximum(cur, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec)

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05

    # soft‑max temperature schedule
    alpha0 = 10.0
    alpha_max = 2000.0

    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while True:
        now = time.monotonic()
        if now >= deadline:
            break
        it += 1

        # ----- FFT of current vector -----
        a_fft = np.fft.rfft(cur, n=L)

        # ----- autocorrelation -----
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- soft‑max weighting -----
        alpha = min(alpha_max, alpha0 * math.log1p(it))
        diff = conv - max_val                # ≤0
        w = np.exp(alpha * diff)             # stable, values ≤1
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            mask = (conv >= max_val - 1e-12)
            w = mask.astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient (matrix‑free) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        cur_new = cur - step
        cur_new = _project_simplex(cur_new)

        cur_sc = _score(cur_new)

        # ----- keep best -----
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur = cur_new

        # ----- occasional symmetrisation -----
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best_sc = sym_sc
                best_vec = sym.copy()
            cur = sym
            no_improve = 0

            cur = best_vec + delta
            cur = np.maximum(cur, 0.0)
            cur = _project_simplex(cur)
            no_improve = 0
            # reset Adam moments to avoid stale directions
            m.fill(0.0)
            v.fill(0.0)

        # ----- early stop -----
        if best_sc < target:

    # ------------------------------------------------------------------
    # fallback if nothing improved (should not happen)
    if best_sc == float('inf') or np.isnan(best_sc):
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        best_vec = uniform
        best_sc = _score(best_vec)

    return [float(x) for x in best_vec]
