def conv_max(a):
    n = len(a)
    L = 1 << ((2*n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return conv.max()

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _max_conv(a):
    """Maximum value of linear convolution a*a (full length)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv.max()

def _score_vec(a):
    """Score for a vector that already satisfies sum(a)=1."""
    n = a.shape[0]
    return float(2.0 * n * _max_conv(a))

    rng = np.random.default_rng(seed)

    else:
        n = 2500

    cur = _project_simplex(cur)          # ensure exact simplex
    cur_score = _score_vec(cur)

    best_vec = cur.copy()
    best_score = cur_score

    # ----- optimizer state ---------------------------------------------
