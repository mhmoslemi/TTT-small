import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that tries to
    minimise the AC1 upper bound evaluated by `evaluate_sequence`.
    The routine works with any parent sequence ``height_sequence_1`` that
    may already exist; otherwise it starts from the uniform distribution.
    It performs a fast, incremental hill‑climbing search that updates the
    full convolution in O(n) time per move.
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)   # tiny safety margin

    # ----------------------------------------------------------------------
    # Initialise the sequence (sum normalised to 1, which does not affect the score)
    if "height_sequence_1" in globals():
        seq = np.asarray(height_sequence_1, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        total = seq.sum()
        if total < 0.01:                     # degenerate – fall back to uniform
            n = seq.size
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = seq / total                   # normalise to sum == 1
    else:
        n = 2000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size

    # ----------------------------------------------------------------------
    # Full convolution and current maximum
    c = np.convolve(a, a, mode='full')          # length 2*n-1
    cur_max = c.max()
    best_a = a.copy()
    best_max = cur_max

    rng = np.random.default_rng(seed)
    step_factor = 0.3                           # max fraction of source mass to move
    it = 0
    last_improve = 0

    # ----------------------------------------------------------------------
    # Main hill‑climbing loop (incremental convolution updates)
    while time.monotonic() < deadline:
        it += 1

        # pick a source index with some mass
        i = rng.integers(0, n)
        if a[i] <= 1e-15:
            continue

        # pick a distinct destination index
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # amount of mass to move
        delta = a[i] * rng.random() * step_factor
        if delta <= 0.0:
            continue

        # ------------------------------------------------------------------
        # Apply the move – update convolution in O(n) time
        # (use the current a values, modify a only after the slice updates)
        c[i:i + n] -= 2.0 * delta * a
        c[2 * i] += delta * delta

        c[j:j + n] += 2.0 * delta * a
        c[2 * j] += delta * delta

        cross_idx = i + j                         # cross term from d*d
        c[cross_idx] -= 2.0 * delta * delta

        # update the sequence itself (sum stays exactly 1)
        a[i] -= delta
        a[j] += delta

        new_max = c.max()
        if new_max <= cur_max:                    # accept improvement (or equality)
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                last_improve = it
        else:
            # ----------------------------------------------------------------
            # Revert the move (restore a and c)
            a[i] += delta
            a[j] -= delta

            c[i:i + n] += 2.0 * delta * a
            c[2 * i] -= delta * delta

            c[j:j + n] -= 2.0 * delta * a
            c[2 * j] -= delta * delta

            c[cross_idx] += 2.0 * delta * delta
        # --------------------------------------------------------------------
        # Periodically recompute the full convolution to avoid drift
        if it % 5000 == 0:

            cur_max = c.max()
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                last_improve = it

            last_improve = it

    # ----------------------------------------------------------------------

    except Exception:
