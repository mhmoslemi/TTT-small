def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - css / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    M_param = cp.Parameter(nonneg=True)
    conv_expr = cp.conv(f_param, b_var)    # length 2n‑1
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)),
                      [conv_expr <= M_param])

    # ---------- main loop ----------
    mix = 0.6          # initial mixing weight (a ← (1‑mix)·a + mix·b)
    it = 0
    while time.monotonic() < deadline and it < 250:
        it += 1

        # current max‑convolution value M
        L = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        M = float(conv.max())

        # solve LP: max sum(b) s.t. f*b ≤ M, b ≥ 0
        f_param.value = a
        M_param.value = M
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception:
            break
        b = b_var.value
        if b is None:
            break
        b = np.maximum(b, 0.0)

        sum_b = b.sum()
        if sum_b <= 0.0:
            break
        b_norm = b / sum_b                     # enforce sum = 1

        # optional symmetrisation (helps to keep the shape balanced)
        b_norm = (b_norm + b_norm[::-1]) * 0.5
        b_norm = _project_simplex(b_norm)

        # convex combination
        a = (1.0 - mix) * a + mix * b_norm
        a = _project_simplex(a)

        # evaluate
        sc = _score(a)
        if sc < best_sc:
            best = a.copy()
            best_sc = sc
            # accelerate mixing when improvement is strong
            mix = min(0.9, mix * 1.02)

        # early exit if target reached
        if best_sc < target:
            break

        # slowly decay mixing weight to allow fine adjustments
        if it % 20 == 0:
            mix = max(0.05, mix * 0.95)

        # occasional symmetrisation of the current iterate
        if it % 50 == 0:
            a = (a + a[::-1]) * 0.5
            a = _project_simplex(a)

    # ---------- return ----------
    return [float(x) for x in best]
