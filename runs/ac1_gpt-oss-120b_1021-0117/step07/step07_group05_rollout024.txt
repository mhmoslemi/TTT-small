import time
import numpy as np

def _objective_fft(arr: np.ndarray, L: int) -> float:
    """
    Fast computation of 2*n*max(conv(arr,arr))/sum(arr)^2 using FFT.
    L must be a power of two >= 2*len(arr)-1.
    """
    n = arr.shape[0]
    # rfft on zero‑padded array
    f = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    max_b = np.max(conv)
    s = np.sum(arr)
    # s is never zero for the paths we take
    return float(2.0 * n * max_b / (s * s))


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that attempts to
    minimise the AC1 upper‑bound evaluated by ``evaluate_sequence``.
    The routine works with any parent sequence ``height_sequence_1`` that
    may already be present; otherwise it falls back to a uniform distribution.
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)       # tiny safety margin

    # ----------------------------------------------------------------------
    # Determine problem size and initialise a feasible vector (sum == 1)
    if "height_sequence_1" in globals():
        seq = np.asarray(height_sequence_1, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        total = seq.sum()
        if total <= 0.0:
            # fall back to uniform if the parent is all zeros
            n = len(seq)
            cur = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            cur = seq / total                         # normalise to sum == 1
            n = cur.shape[0]
    else:                     # no parent supplied → uniform start
        n = 2000
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    # ----------------------------------------------------------------------
    # Pre‑compute FFT length (next power of two >= 2*n-1)
    L = 1 << (2 * n - 1).bit_length()

    # Initial best values
    best_vec = cur.copy()
    best_score = _objective_fft(best_vec, L)

    # Random number generator
    rng = np.random.default_rng(seed)

    # Simulated‑annealing / stochastic local‑search parameters
    temperature = 1.0
    min_temp = 1e-6
    decay = 0.9995               # geometric cooling
    step_factor = 0.5            # maximal fraction of source mass to move
    it = 0
    max_iters = 500_000          # hard cap; loop will also break on deadline

    while time.monotonic() < deadline and it < max_iters:
        it += 1

        # occasional parameter decay
        if it % 1000 == 0:
            temperature = max(min_temp, temperature * decay)
            step_factor = max(1e-6, step_factor * decay)

        # propose a simple mass‑transfer move that keeps the total sum fixed
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        delta = cur[i] * rng.random() * step_factor   # move at most step_factor of cur[i]
        if delta <= 0.0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # numerical safety – discard moves that make a component slightly negative
        if cand[i] < 0.0:
            continue

        # fast evaluation of the objective
        cand_score = _objective_fft(cand, L)

        # Metropolis acceptance
        if cand_score < best_score or rng.random() < np.exp((best_score - cand_score) / (temperature + 1e-12)):
            cur = cand
            # keep the globally best vector seen so far
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()

    # ----------------------------------------------------------------------
    # Final validation with the official evaluator (protects against tiny
    # numerical drift that could violate the clipping rules).
    try:
        official = evaluate_sequence(best_vec.tolist())
        if not np.isfinite(official):
            # fallback to uniform – guaranteed finite score of 2
            best_vec = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            # keep the vector that gave the official score
            best_vec = np.array(best_vec, copy=True)
    except Exception:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    # ensure strict compliance with the evaluator’s clipping rules
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # return as plain Python list of floats
    return [float(x) for x in best_vec]
