    Returns np.inf if the input is invalid.
    """
    if not isinstance(sequence, list):

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a non‑negative vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    # ensure non‑negativity
    v = np.maximum(v, 0.0)
    # if sum is zero, return uniform
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    v /= s
    # standard simplex projection (might slightly adjust due to rounding)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    return w
