import time, math, numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex (sum == 1, x >= 0)
def _project_simplex(v):
    """Project a vector onto the simplex {x >= 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    # Sort in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    # Find last index where rho_arr > 0
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Helper: compute 2*n * max convolution (a is assumed to sum to 1)
def _score_fft(a):
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative step heights that (approximately) minimize
    the AC1 upper bound as measured by `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # initial vector (use the provided parent if present)
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2000, dtype=np.float64)
    a = np.maximum(a, 0.0)               # non‑negative
    a = _project_simplex(a)               # enforce sum == 1
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    # best solution tracker
    best = a.copy()
    best_sc = _score_fft(best)

    # Adam optimizer state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    step0 = 0.6                # initial step size
    it = 0
    no_improve = 0
    target = 1.50274349        # optional early stop

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- convolution (FFT, full) -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- active set mask (uniform over ties) -----
        eps_mask = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - eps_mask
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt

        # ----- gradient of the active‑max convolution -----
        # w is defined on full convolution index range; reverse for correlation
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        # pick the part that corresponds to gradient entries
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / (math.sqrt(it) + 1.0)
        update = step * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(a - update)

        # ----- evaluate candidate -----
        cand_sc = _score_fft(cand)

        # ----- accept if improvement -----
        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc
            a = cand
            no_improve = 0
        elif cand_sc < _score_fft(a):
            a = cand
            no_improve = 0
        else:
            no_improve += 1
            # occasional probabilistic fallback
            prob = math.exp(-(cand_sc - best_sc) / (step + 1e-12))
            if rng.random() < prob:
                a = cand

        # ----- symmetric averaging every few hundred steps (helps convergence) -----
        if it % 300 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fft(sym)
            if sym_sc < best_sc:
                best = sym.copy()
                best_sc = sym_sc
                a = sym
                no_improve = 0

        # ----- random redistribution if stuck -----
        if no_improve > 1500:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                a = tmp
            no_improve = 0

        # ----- early exit if we beat the target -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return the best vector found (as plain Python floats)
    return [float(x) for x in best]
