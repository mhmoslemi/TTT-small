def grad_submax(a):
    # compute convolution using FFT
    n = len(a)
    L = 1 << ((2*n-1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*n-1]
    max_val = conv.max()
    # mask where conv == max (with tolerance)
    tol = 1e-12 * max_val + 1e-12
    mask = (conv >= max_val - tol).astype(np.float64)
    # compute gradient: 2 * conv of a with reversed mask
    mask_rev = mask[::-1]
    mask_fft = np.fft.rfft(mask_rev, n=L)
    grad_full = np.fft.irfft(a_fft * mask_fft, n=L)
    idx = (2*n-2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad
```

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _conv_max(a, L):
    """Maximum value of the linear convolution a*a using FFT."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv.max()

    """Fast evaluation of the evaluator's objective for a simplex‑normalised a."""
    return float(2.0 * a.size * _conv_max(a, L))

    # linear decreasing / increasing
    lin_dec = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(_project_simplex(lin_dec))
    lin_inc = np.arange(1, n + 1, dtype=np.float64)
    inits.append(_project_simplex(lin_inc))

    # exponential decay
    exp_decay = np.exp(-np.linspace(0, 6, n, dtype=np.float64))
    inits.append(_project_simplex(exp_decay))

    # power‑law families
    for p in (0.5, 1.0, 1.5, 2.0):
        pow_vec = (np.arange(1, n + 1, dtype=np.float64) ** -p)
        inits.append(_project_simplex(pow_vec))

    # random Dirichlet (Gamma) samples
    for _ in range(3):
        gamma = rng.gamma(1.0, size=n)
        inits.append(_project_simplex(gamma))

    # use the provided parent if present
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        inits.append(_project_simplex(parent))

    # ------------------------------------------------------------------
    # pick the best of the initial pool
    best_vec = None
    best_sc = float("inf")
    for vec in inits:
        sc = _score_fast(vec, L)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()
    cur = best_vec.copy()
    cur_sc = best_sc

    # ------------------------------------------------------------------
    # optimisation parameters (momentum / Adam‑like)
    momentum = np.zeros_like(cur)
    beta1 = 0.9          # momentum coefficient
    step0 = 0.7          # initial step magnitude
    it = 0
    no_improve = 0

    # soft‑max schedule for the max‑convolution weighting
    alpha_max = 120.0
    alpha0 = 1.0

    target = 1.50274349   # early‑stop threshold

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- FFT of current vector -----
        a_fft = np.fft.rfft(cur, n=L)

        # ----- full convolution (auto‑correlation) -----
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- soft‑max weighting (focuses on the true maximum) -----
        alpha = min(alpha_max, alpha0 * math.log1p(it))
        w = np.exp(alpha * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            # fallback to a hard mask if numerical issues arise
            mask = conv >= max_val - (max_val * 1e-12 + 1e-15)
            w = mask.astype(np.float64)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.ones_like(conv) / conv.size
                w_sum = 1.0
        w /= w_sum

        # ----- gradient of the weighted max‑convolution (matrix‑free) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)    # map to size‑n gradient
        grad = 2.0 * grad_full[idx]

        # ----- momentum / Adam‑like update -----
        momentum = beta1 * momentum + (1.0 - beta1) * grad

        # ----- step size (decays like 1/√it) -----
        step = step0 / math.sqrt(it)

        # ----- candidate and evaluation -----
        cand = _project_simplex(cur - step * momentum)
        cand_sc = _score_fast(cand, L)

        # ----- accept if improvement, otherwise occasional uphill move -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1
            # occasional Metropolis‑type acceptance
            temp = step * 0.5 + 1e-12
            prob = math.exp(-(cand_sc - cur_sc) / temp)
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # ----- periodic symmetrisation (helps many instances) -----
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fast(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # ----- random local redistribution when stuck -----
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_fast(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # return the best vector found (plain Python floats)
    return [float(x) for x in best_vec]
