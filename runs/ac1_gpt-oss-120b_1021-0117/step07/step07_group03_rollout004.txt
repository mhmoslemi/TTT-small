def conv_max_and_grad(a, L):
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*n-1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15

    if cnt > 0:

    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2*n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad

def score_from_max(max_val, n):
    return 2.0 * n * max_val

def propose_candidate(seed=42, budget_s=1100, **kwargs):

    beta2 = 0.999
    eps = 1e-8
    t_iter = 0
    # parameters
    base_lr = 0.5
    # schedule for restarts
    while time.time() < deadline:

        t_iter += 1

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_max_and_grad(a, L):
    """
    Compute the maximum value of the full convolution a*a and a sub‑gradient.
    Returns (max_value, gradient) where gradient has the same shape as a.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ---- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = (a0 + a0[::-1]) * 0.5          # initial symmetrisation
    else:
        n0 = rng.integers(1500, 3000)
        a0 = rng.random(n0, dtype=np.float64)

    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())    # next power of two >= 2n‑1

    # ---- optimizer state ------------------------------------------------
    m = np.zeros_like(cur)   # first moment
    v = np.zeros_like(cur)   # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.5

    # ---- bookkeeping ----------------------------------------------------
    t_iter = 0
    best_vec = cur.copy()
    max_val, _ = _conv_max_and_grad(cur, L)
    best_score = 2.0 * n * max_val          # = evaluate_sequence when sum=1
    last_improve = 0
    target = 1.50274349

    # ---------------------------------------------------------------------
    while time.time() < deadline:
        t_iter += 1

        # gradient and current score
        max_val, grad = _conv_max_and_grad(cur, L)
        cur_score = 2.0 * n * max_val

        # update best ever
        if cur_score < best_score:
            best_score = cur_score
            best_vec = cur.copy()
            last_improve = t_iter
            if best_score < target:
                break

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        # propose a new point
        step = base_lr / (np.sqrt(v_hat) + eps)
        cand = cur - step * m_hat
        cand = _project_simplex(cand)

        # occasional symmetrisation
        if t_iter % 250 == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        # evaluate candidate (only need max, gradient ignored)
        cand_max, _ = _conv_max_and_grad(cand, L)
        cand_score = 2.0 * n * cand_max

        # acceptance rule (deterministic improvement + occasional uphill)
        if cand_score < cur_score:

        else:
            prob = math.exp(-(cand_score - cur_score) / (base_lr + 1e-12))
            if rng.random() < prob:

        # if stuck for a while, random perturbation around the best found
        if t_iter - last_improve > 3000:
            pert = best_vec + 0.05 * rng.standard_normal(size=n)
            pert = _project_simplex(pert)
            cur = pert
            m.fill(0.0)
            v.fill(0.0)
            last_improve = t_iter

    # safety fallback in the unlikely event of a degenerate result
    if best_vec.sum() < 0.01 or not np.isfinite(best_vec).all():
        best_vec = np.ones(n, dtype=np.float64) / n

    return [float(x) for x in best_vec]
