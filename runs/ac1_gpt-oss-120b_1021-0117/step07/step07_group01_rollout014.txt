```python
import time, math, numpy as np, scipy.optimize

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    return float(2.0 * n * max_val)
```

```python
    target = 1.50274349

    min_t = 0.01

    while True:

            break

        L = 1 << ((2 * n - 1).bit_length())

        eps = max(1e-12, M * 1e-12)
        active_idx = np.where(conv >= M - eps)[0]
        if active_idx.size == 0:
            active_idx = np.array([np.argmax(conv)], dtype=int)
        # limit active size
        max_active = 200
        if active_idx.size > max_active:
            # random sample to keep size
            active_idx = rng.choice(active_idx, size=max_active, replace=False)
        
        # build constraint matrix A (len(active) x n)
        m = active_idx.size
        A = np.zeros((m, n), dtype=np.float64)
        for row, k in enumerate(active_idx):
            # i runs from max(0, k-(n-1)) to min(k, n-1)
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # corresponding j indices: j = k - i
            j_start = k - i_end
            # slice of a reversed
            seg = a[i_start:i_end+1][::-1]  # length i_end-i_start+1
            A[row, j_start:j_start+seg.size] = seg

        c = -np.ones(n, dtype=np.float64)  # minimize -sum(g)

        # Use high-level linprog
        try:

            if res.success:

                s = g.sum()
                if s > 0:
                    g = g / s

                    g = a.copy()
```

import time, math, numpy as np, scipy.optimize

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    """Score = 2 * n * max(convolution of a with itself). Assumes sum(a)=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    return float(2.0 * n * max_val)

def _build_constraint_matrix(a, active_idx):
    """Build matrix A where each row corresponds to (a * g)_k <= M for k in active_idx."""
    n = a.shape[0]
    m = active_idx.size
    A = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(active_idx):

        j_start = k - i_end
        seg = a[i_start:i_end + 1][::-1]          # reversed slice of a
        A[row, j_start:j_start + seg.size] = seg
    return A
