import time, math, numpy as np

def _project_simplex(v):
    # as given
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    # assumes sum(a)=1
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.4)  # reserve a bit
    
    # initialization
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # choose a moderate length if not provided
        n = 2000
        a = rng.random(n, dtype=np.float64)
    # ensure non‑negative and project onto the probability simplex (sum=1)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    
    best = a.copy()
    best_sc = _score(best)
    cur_sc = best_sc
    
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    step0 = 0.8
    
    # temperature for soft‑max approximation of max
    tau = 0.2
    tau_max = 30.0
    tau_mul = 1.001
    
    it = 0
    no_improve = 0
    target = 1.50274349
    
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1
        
        # FFT of current vector
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val
        
        # construct weights (smooth max)
        if tau < tau_max:
            # stable soft‑max
            w = np.exp(tau * (conv - max_val))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
            else:
                w /= w_sum
            tau = min(tau_max, tau * tau_mul)
        else:
            # hard max mask (as in previous code)
            eps_mask = max_val * 1e-12 + 1e-15
            mask = conv >= max_val - eps_mask
            w = mask.astype(np.float64)
            cnt = w.sum()
            if cnt > 0:
                w /= cnt
        
        # gradient of weighted max‑convolution
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (math.sqrt(it) + 1.0)
        update = step * m_hat / (np.sqrt(v_hat) + eps)

        cand_add_sc = _score(cand_add)

        cand_mul = a * exp_factor
        s = cand_mul.sum()
        if s > 0:
            cand_mul = cand_mul / s
        else:
            cand_mul = a.copy()
        cand_mul_sc = _score(cand_mul)
        
        # pick better of the two
        if cand_add_sc < cand_mul_sc:
            cand, cand_sc = cand_add, cand_add_sc
        else:
            cand, cand_sc = cand_mul, cand_mul_sc
        
        # acceptance
        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc
            a = cand
            no_improve = 0
        elif cand_sc < cur_sc:
            a = cand
            no_improve = 0
        else:
            no_improve += 1
            # occasional probabilistic uphill move

            if rng.random() < prob:
                a = cand

                best_sc = sym_sc
                a = sym
                no_improve = 0

        if no_improve > 3000:

            if a[i] > 0.0:

                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                a = tmp
            no_improve = 0

            rand_vec = _project_simplex(rand_vec)
            a = 0.5 * best + 0.5 * rand_vec
            a = _project_simplex(a)
            no_improve = 0
        
        # early exit if target reached
        if best_sc < target:
            break
    
    return [float(x) for x in best]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    """Objective for a simplex‑normalised vector: 2·n·max(a*a)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the final return

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)          # sum = 1, non‑negative
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    best = a.copy()
    best_sc = _score(best)
    cur_sc = best_sc

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    step0 = 0.8

    # soft‑max temperature for approximating max
    tau = 0.02
    tau_max = 30.0
    tau_mul = 1.001

    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- convolution and current score -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val

        # ----- weight vector for (smooth) max -----
        if tau < tau_max:
            # log‑stable softmax centred at the maximum
            w = np.exp(tau * (conv - max_val))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
            else:
                w /= w_sum
            tau = min(tau_max, tau * tau_mul)
        else:

    # ------------------------------------------------------------------
    return [float(x) for x in best]
