import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improve the AC1 bound.
    The routine respects the time budget, uses the supplied `height_sequence_1`
    when available, and runs a projected gradient descent with line‑search.
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise a (sum normalised to 1)
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback length – any reasonable size works
        a = rng.random(2000, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)
    n = a.size
    # length for FFT (next power of two >= 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # helper: max convolution of a with itself (sum(a)=1)
    def _max_conv(x):
        xf = np.fft.rfft(x, n=L)
        conv_full = np.fft.irfft(xf * xf, n=L)
        return conv_full[: 2 * n - 1].max()

    # current score, best so far
    cur_sc = 2.0 * n * _max_conv(a)
    best = a.copy()
    best_sc = cur_sc

    stagnation = 0

    target = 1.50274349

    while True:

        it += 1
