Standard algorithm:

def _project_simplex(v):
    """Project onto simplex {x >= 0, sum = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that (hopefully)
    improves on the previous AC1 upper bound.  The routine works with any
    global `height_sequence_1` that may already be present; otherwise it starts
    from the uniform distribution.  A simple gradient‑guided stochastic search
    is performed under a time budget.
    """
    start = time.monotonic()
    # keep a tiny safety margin
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # ----------------------------------------------------------------------
    # initialise the sequence
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)

        if s < 1e-12:

        else:
            a = a / s                     # enforce sum == 1
    else:
        a = None

    if a is None:
        # fallback – uniform distribution
        n = 2000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.shape[0]

    # ----------------------------------------------------------------------
    # helper: projection onto the probability simplex {x>=0, sum=1}
    def _project_simplex(v):
        """Euclidean projection of v onto the simplex."""
        v = np.asarray(v, dtype=np.float64)
        n = v.shape[0]
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    # ----------------------------------------------------------------------
    # fast objective (sum is always 1 in our search)
    def _score(vec):
        """Returns 2*n*max(conv(vec,vec)) (since sum(vec)=1)."""
        c = np.convolve(vec, vec, mode='full')
        return 2.0 * n * c.max()

    # ----------------------------------------------------------------------
    # initial scores
    cur = a.copy()
    cur_score = _score(cur)
    best = cur.copy()
    best_score = cur_score

    rng = np.random.default_rng(seed)

    # parameters for the guided search
    max_iters = 200000                     # hard cap; time check stops earlier
    factor = 0.5                           # base factor for step size
    max_step = 0.6                         # cap on raw step length
    target_score = 1.50274349 * 2 * n      # corresponding max(conv) target

    it = 0
