import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound evaluated by ``evaluate_sequence``.
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)   # safety margin

    rng = np.random.default_rng(seed)

    # ---------- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if seq0.size == 0:
            seq0 = None
        else:
            seq0 = seq0 / seq0.sum()          # normalise to sum = 1
    else:
        seq0 = None

    # uniform fallback
    n_def = 2000
    uniform = None
    if seq0 is None:
        n = n_def
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = seq0.size
        uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # choose the better of parent / uniform as start
    def _score(arr):
        c = np.convolve(arr, arr)
        return 2.0 * n * c.max() / (arr.sum() ** 2)

    best_seq = uniform
    best_score = _score(best_seq)
    if seq0 is not None:
        sc = _score(seq0)
        if sc < best_score:
            best_seq = seq0
            best_score = sc

    cur_seq = best_seq.copy()
    cur_score = best_score

    # ---------- optimisation loop -----------------------------------------
    step_scale = 0.3                     # initial fraction of moved mass
    no_improve = 0
    it = 0
    max_no_imp = 5000                    # restart after stagnation

    while time.monotonic() < deadline:
        # full convolution and location of the current peak
        conv = np.convolve(cur_seq, cur_seq)
        max_val = conv.max()
        km = int(conv.argmax())

        # contribution of each index to the peak (product a_i * a_{km-i})
        idx = km - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        prod = np.zeros(n, dtype=np.float64)
        prod[valid] = cur_seq[valid] * cur_seq[idx[valid]]

        # pick source (large contribution) and target (small contribution)
        src = int(np.argmax(prod))
        if cur_seq[src] <= 1e-14:                     # fallback if zero mass
            positive = np.where(cur_seq > 1e-14)[0]
            if positive.size == 0:
                break
            src = int(rng.choice(positive))

        tgt = int(np.argmin(prod))
        if tgt == src:
            tgt = (src + 1) % n

        # amount to move
        delta = cur_seq[src] * rng.random() * step_scale
        if delta <= 0.0:
            continue

        # apply tentative move
        cur_seq[src] -= delta
        cur_seq[tgt] += delta

        # compute new score (sum stays =1, so denominator constant)
        new_conv = np.convolve(cur_seq, cur_seq)
        new_score = 2.0 * n * new_conv.max() / (cur_seq.sum() ** 2)

        if new_score < cur_score:                     # accept improvement
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:                                         # reject
            cur_seq[src] += delta
            cur_seq[tgt] -= delta
            no_improve += 1

        it += 1
        # gradual reduction of step size
        if it % 1000 == 0:
            step_scale = max(0.01, step_scale * 0.9)

        # restart from best if stuck
        if no_improve > max_no_imp:
            cur_seq = best_seq.copy()
            cur_score = best_score
            no_improve = 0

    # ---------- final sanity check ----------------------------------------
    # ensure non‑negative and within limits (evaluator also clips)
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    return [float(x) for x in best_seq.tolist()]
