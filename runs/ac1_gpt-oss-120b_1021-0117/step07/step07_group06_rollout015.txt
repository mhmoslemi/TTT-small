def _solve_lp(a, M, time_remaining):
    n = a.shape[0]
    offsets = np.arange(n, dtype=np.int32)
    # Build data arrays as list of same value repeated n times
    # Use np.full for each offset; list comprehension.
    data = [np.full(n, a[i], dtype=np.float64) for i in range(n)]
    T = sp.diags(data, offsets, shape=(2*n-1, n), format='csr')
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    # Provide options time limit
    options = {'presolve': True}
    # if time_remaining is small, we can pass time_limit
    if time_remaining is not None and time_remaining > 0:

    try:
        res = opt.linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs', options=options)
        if res.success:
            return res.x
    except Exception:
        pass
    # fallback: return None
    return None

import time, math, numpy as np, scipy.sparse as sp, scipy.optimize as opt

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else css[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score_vec(a, L):
    """Score for a vector a (assumed sum≈1)."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * a.size - 1]
    return float(2.0 * a.size * conv.max())

def _lp_steering(a, M, time_deadline):
    """Solve LP: maximize sum(g) s.t. conv(a,g) ≤ M, g≥0."""
    n = a.size
    # build lower‑triangular Toeplitz matrix T (size (2n‑1)×n)
    offsets = np.arange(n, dtype=np.int32)
    data = [np.full(n, a[i], dtype=np.float64) for i in range(n)]
    T = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    c = -np.ones(n, dtype=np.float64)                # minimize -∑g  ⇔ maximize ∑g
    bounds = [(0, None)] * n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    opts = {'presolve': True}
    remaining = max(0.0, time_deadline - time.monotonic())
    if remaining > 0:
        opts['time_limit'] = remaining
    try:
        res = opt.linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds,
                          method='highs', options=opts)
        if res.success:
            return res.x
    except Exception:
        pass
    return None

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score_vec(best_vec, L)

    # optimizer state
    momentum = np.zeros_like(cur)
    beta = 0.9
    step0 = 0.8
    no_improve = 0
    target = 1.50274349

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution & max -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        M = conv.max()

        # ----- gradient (top‑k weighting) -----
