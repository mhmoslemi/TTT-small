def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # length n
    if 'height_sequence_1' in globals():
        n = int(len(height_sequence_1))
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0.0)
        if parent.sum() > 0:
            parent = parent / parent.sum()
    else:
        n = 2000
        parent = None

    L = 1 << ((2*n - 1).bit_length())

    # create initial candidates
    xs = np.arange(n, dtype=np.float64)
    candidates = []

    # Uniform
    candidates.append(np.full(n, 1.0 / n))
    # Triangular symmetric
    tri = np.minimum(xs + 1, n - xs)
    candidates.append(tri / tri.sum())
    # Sine bump
    sine = np.sin(np.pi * (xs + 1) / (n + 1))
    candidates.append(sine / sine.sum())
    # Raised cosine
    cosb = (1 - np.cos(2 * np.pi * xs / (n - 1))) / 2
    candidates.append(cosb / cosb.sum())
    # Exponential decays
    for a in (0.001, 0.003, 0.006):
        e = np.exp(-a * xs)
        candidates.append(e / e.sum())
    # Power laws
    for p in (0.5, 1.0, 1.5):
        pw = (xs + 1) ** (-p)
        candidates.append(pw / pw.sum())
    # Random Dirichlet
    rand = rng.random(n)
    candidates.append(rand / rand.sum())
    # Parent if available
    if parent is not None:
        candidates.append(parent)

    # function to compute score using FFT (fast)
    def _score_fft(a):
        s = a.sum()
        if s < 1e-12:
            return float('inf')
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv.max()
        return float(2.0 * n * max_val / (s * s)), conv

    best_sc = float('inf')
    best_vec = None

    for vec in candidates:
        sc, _ = _score_fft(vec)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()
    cur = best_vec.copy()
    cur_sc = best_sc

    # enforce symmetry
    cur = (cur + cur[::-1]) * 0.5
    cur = _project_simplex(cur)  # ensure sum=1 (though unchanged)
    cur_sc, _ = _score_fft(cur)
    best_sc = cur_sc
    best_vec = cur.copy()

    # Adam optimizer state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5   # base learning rate

    it = 0
    no_improve = 0
    target = 1.50274349

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # Compute gradient (subgradient)
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        # tolerance for active set
        tol = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - tol
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        # reverse w for cross-correlation
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr / (np.sqrt(v_hat) + eps)

        cand = cur - step * m_hat
        # project onto simplex and enforce symmetry
        cand = _project_simplex(cand)
        cand = (cand + cand[::-1]) * 0.5

        cand_sc, _ = _score_fft(cand)

        # acceptance
        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
        else:
            no_improve += 1
            # occasional simulated annealing style acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (step.mean() + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_sc = cand_sc
                no_improve = 0

        # update global best
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()

        # random small redistribution if stuck
        if no_improve > 3000:
            # pick two distinct indices
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp = (tmp + tmp[::-1]) * 0.5
            tmp_sc, _ = _score_fft(tmp)
            if tmp_sc < cur_sc:
                cur = tmp
                cur_sc = tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()

        # early exit if target reached
        if best_sc < target:
            break

    # return best found as list
    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # fix tiny numerical drift
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w
