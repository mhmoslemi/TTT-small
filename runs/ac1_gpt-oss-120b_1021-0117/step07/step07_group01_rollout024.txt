def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a using FFT, returns full length (2*len(a)-1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)
    return conv[: 2 * a.shape[0] - 1]

def _max_conv(a: np.ndarray, L: int) -> float:

    return float(_conv_full(a, L).max())

    Gradient is a sub‑gradient of max_k (a*a)[k] w.r.t. a.
