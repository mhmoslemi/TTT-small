import time, math, numpy as np

# ------------------------------------------------------------
def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_and_grad(a, L):
    """
    Compute (max convolution value, subgradient) of the full convolution a*a.
    The subgradient is the usual one for the max-of-convolution functional.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)

    # full convolution via FFT
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]

    max_val = conv.max()
    # mask of entries achieving (or very near) the maximum
    eps = max_val * 1e-9 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    # gradient = 2 * a * mask (convolution with reversed mask)
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad

        lr = lr0 / math.sqrt(it)

        # candidate via Adam step
        cand = cur - lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = _project_simplex(cand)
        # enforce symmetry each iteration
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        cand_max, _ = _conv_max_and_grad(cand, L)
        cand_sc = 2.0 * n * cand_max

        # accept if improvement, otherwise keep current (occasionally accept worse)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            # occasional uphill moves to escape shallow traps
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
                no_improve = 0
            else:
                no_improve += 1

        # update global best
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0

        # random perturbation when stuck
        if no_improve > 3000:
            sigma = 0.05 / math.sqrt(it)
            pert = cur + sigma * rng.standard_normal(size=n)
            pert = _project_simplex(pert)
            pert = (pert + pert[::-1]) * 0.5
            pert = _project_simplex(pert)

            pert_max, _ = _conv_max_and_grad(pert, L)
            pert_sc = 2.0 * n * pert_max

            if pert_sc < cur_sc:
                cur, cur_sc = pert, pert_sc
                if pert_sc < best_sc:
                    best_sc = pert_sc
                    best_vec = pert.copy()
                no_improve = 0
            else:
                # reset moments to avoid stale gradients
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0

        # early exit if target reached
        if best_sc < target:
            break

    # return best found sequence as plain Python floats
    return [float(x) for x in best_vec]
