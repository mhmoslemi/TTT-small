    start = time.time()
    deadline = start + max(0.0, budget_s - 0.2)

    if 'height_sequence_1' in globals():

    n = cur.size

    # parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.15  # initial learning rate
    t = 0
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    best_vec = cur.copy()
    # compute initial score
    a_fft = np.fft.rfft(cur, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    max_conv = conv.max()
    best_sc = 2.0 * n * max_conv
    cur_sc = best_sc
    best_sc = cur_sc
    no_improve = 0

    # schedule parameters
    temp_decay = 0.9995
    T_factor = 0.05  # initial temperature scaling relative to max_conv

    target = 1.50274349

    while True:
        if time.time() >= deadline:
            break
        t += 1
        # compute conv and gradient
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_conv = conv_full.max()

        # temperature
        T = max(max_conv * T_factor * (temp_decay ** t), 1e-8)

        # compute softmax weights
        w = np.exp((conv_full - max_conv) / T)
        w_sum = w.sum()
        if w_sum == 0:
            w = (conv_full == max_conv).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient via convolution with reversed w
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)

        idx = (2*n - 2) - np.arange(n)

        m = beta1 * m + (1 - beta1) * grad

import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast score for a vector already normalized to sum≈1 (sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # initialization
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.35            # base learning rate
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    t = 0

    # temperature schedule
    temp_decay = 0.9995
    temp_factor0 = 0.1    # initial factor of max_conv

    # keep best solution
    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while True:
        if time.time() >= deadline:
            break
        t += 1

        # ----- convolution of current vector -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_conv = conv_full.max()

        # ----- temperature (smooth max) -----
        temp_factor = max(temp_factor0 * (temp_decay ** t), 1e-6)
        T = max_conv * temp_factor

        # ----- softmax weights -----
        w = np.exp((conv_full - max_conv) / T)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv_full == max_conv).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient of weighted max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = lr0 / np.sqrt(t)          # simple decay
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        cand_sc = _score(cand)

        # ----- acceptance -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            # occasional random acceptance for exploration
            if rng.random() < 0.001:
                cur, cur_sc = cand, cand_sc

        # ----- random redistribution when stuck -----
        if no_improve > 2500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.2
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

        # ----- periodic symmetrisation -----
        if t % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return best vector as plain Python floats
    return [float(x) for x in best_vec]
