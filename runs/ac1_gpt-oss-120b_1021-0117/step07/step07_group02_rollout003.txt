import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_simple(a, n, L):
    # compute convolution via FFT
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    max_b = conv_full.max()
    # assuming sum(a) == 1
    return 2.0 * n * max_b

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # init vector
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        # ensure nonnegative, sum to 1
        a0 = np.maximum(a0, 0.0)
    else:
        # length fallback ~2500
        n0 = 2500
        a0 = rng.random(n0, dtype=np.float64)
    # project onto simplex
    cur = _project_simplex(a0)
    n = cur.size
    # pad length for FFT
    L = 1 << ((2*n - 1).bit_length())

    # initial score
    cur_score = _score_simple(cur, n, L)
    best_vec = cur.copy()
    best_score = cur_score

    # optimizer parameters
    beta = 0.9
    momentum = np.zeros_like(cur)
    step0 = 0.5    # initial step size
    it = 0
    stagnation = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # compute convolution and gradient
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_b = conv_full.max()
        # indices where max occurs
        max_idx = np.flatnonzero(conv_full == max_b)
        # compute gradient of max_b w.r.t a: g_i = (2/len) * sum_{k in max_idx} a_{k-i}
        g = np.zeros_like(cur)
        inv_len = 2.0 / max_idx.size
        for k in max_idx:
            # i ranges where 0 <= i < n and 0 <= k-i < n
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # slice
            i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            # compute a[k - i_range]
            g[i_range] += inv_len * cur[k - i_range]
        # scale gradient of objective
        s = cur.sum()  # should be 1
        base = (2.0 * n) / (s * s)
        const = - (4.0 * n * max_b) / (s * s * s)
        grad = base * g + const

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size schedule
        step = step0 / math.sqrt(it)

        cand = cur - step * momentum
        cand = _project_simplex(cand)

        # evaluate candidate quickly (since sum=1)
        cand_score = _score_simple(cand, n, L)

        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            stagnation = 0
            if cand_score < best_score:
                best_score, best_vec = cand_score, cand.copy()
        else:
            stagnation += 1
            # occasional random perturbation if stuck
            if stagnation > 2000:
                # move a small random amount

                if cur[i] > 0:
                    delta = cur[i] * rng.random() * 0.3
                    j = rng.integers(0, n-1)
                    if j >= i:
                        j += 1
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_score = _score_simple(tmp, n, L)
                    if tmp_score < cur_score:
                        cur, cur_score = tmp, tmp_score
                        stagnation = 0
                        if tmp_score < best_score:
                            best_score, best_vec = tmp_score, tmp.copy()
                else:
                    # random restart
                    rnd = _project_simplex(rng.random(n))
                    rnd_score = _score_simple(rnd, n, L)
                    cur, cur_score = rnd, rnd_score
                    momentum.fill(0.0)
                    stagnation = 0

        # occasional symmetrization
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score_simple(sym, n, L)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score

    # return as Python list of floats

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_simple(a, n, L):
    """Score when sum(a)=1 using FFT‑based convolution."""
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_b = conv_full.max()
    return 2.0 * n * max_b
