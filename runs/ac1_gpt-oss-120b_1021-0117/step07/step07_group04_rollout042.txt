import time, math, numpy as np

def _project_simplex(v):
    """ Euclidean projection onto the probability simplex (sum=1, v>=0). """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # correct tiny numerical drift
    excess = w.sum() - 1.0
    if abs(excess) > 1e-12:
        w -= excess / n
        w = np.maximum(w, 0.0)
    return w

def _score_fast(a, n, L):
    """Fast evaluation of the AC1 upper bound (assumes a>=0)."""
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()
    return float(2.0 * n * max_val / (s * s))

    lin_inc = (xs + 1).astype(np.float64)
    cand_list.append(lin_inc / lin_inc.sum())

    # 4) Triangular (peak centre)
    tri = np.minimum(xs + 1, n - xs)
    cand_list.append(tri / tri.sum())

    # 5) Sine bump
    sine = np.sin(np.pi * (xs + 1) / (n + 1))
    cand_list.append(sine / sine.sum())

    # 6) Raised cosine bump
    bump = (1 - np.cos(2 * np.pi * xs / (n - 1))) / 2
    cand_list.append(bump / bump.sum())

    # 7) Exponential decays
    for alpha in np.logspace(-4, -1, 5):
        exp = np.exp(-alpha * xs)
        cand_list.append(exp / exp.sum())

    # 8) Power laws
    for p in np.linspace(0.2, 2.0, 5):
        pw = (xs + 1) ** (-p)
        cand_list.append(pw / pw.sum())

    # 9) Random Dirichlet‑like draws
    for _ in range(3):
        rnd = rng.random(n)
        cand_list.append(rnd / rnd.sum())

    # 10) Parent if available
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0.0)
        if parent.sum() > 0.0:
            parent = parent / parent.sum()
            cand_list.append(parent)

    # ------------------------------------------------------------------
    # Choose the best deterministic start
    best_vec = None
    best_sc = float('inf')
    for vec in cand_list:
        sc = _score_fast(vec, n, L)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()
    cur = best_vec.copy()
    cur_sc = best_sc

    # ------------------------------------------------------------------
    # Optimisation parameters
    momentum = np.zeros_like(cur)
    beta = 0.9               # momentum coefficient
    step0 = 1.0              # base step size
    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- convolution of current vector (FFT) -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ----- active set for the max -----
        eps = max_val * 1e-12 + 1e-15
        active_idx = np.where(conv_full >= max_val - eps)[0]
        cnt = active_idx.size
        if cnt == 0:
            # numerical safety – treat as single max
            active_idx = np.array([conv_full.argmax()], dtype=int)
            cnt = 1
        weight = 1.0 / cnt

        # ----- sub‑gradient of max‑convolution -----
        a_rev = cur[::-1]
        grad = np.zeros_like(cur)
        centre = n - 1
        for k in active_idx:
            d = int(k - centre)         # offset
            if d >= 0:
                grad[d:] += weight * a_rev[: n - d]
            else:
                grad[: n + d] += weight * a_rev[-d:]
        grad *= 2.0

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- step size (inverse sqrt decay) -----
        step = step0 / math.sqrt(it)

        # ----- candidate by projected gradient -----
        cand = _project_simplex(cur - step * momentum)
        cand_sc = _score_fast(cand, n, L)

        # ----- simple acceptance rule -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1
            # occasional simulated‑annealing acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
                no_improve = 0

        # ----- update global best -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()

        # ----- periodic symmetrisation -----
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fast(sym, n, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # ----- random redistribution when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp_sc = _score_fast(tmp, n, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return the best-found sequence as plain floats
    return [float(x) for x in best_vec]
