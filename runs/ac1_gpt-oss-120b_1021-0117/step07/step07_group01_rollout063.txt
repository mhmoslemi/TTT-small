def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)  # small margin

    # Determine base length n
    if "height_sequence_1" in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = init.size
    else:
        n = 2000
        init = None

    # Build candidate from exponent shape
    idx = np.arange(1, n+1, dtype=np.float64)
    # Choose exponent p via simple search?
    best_init = None
    best_sc = float('inf')
    for p in (-0.7, -0.6, -0.5, -0.4, -0.3):
        raw = idx ** p
        a_candidate = raw / raw.sum()
        sc = _score_vec(a_candidate)
        if sc < best_sc:
            best_sc = sc
            best_init = a_candidate.copy()

    # Consider height_sequence_1 if exists
    if init is not None:
        init = np.maximum(init, 0.0)
        init = _project_simplex(init)
        sc = _score_vec(init)
        if sc < best_sc:
            best_sc = sc
            best_init = init.copy()

    # Random initial if still None (shouldn't happen)
    if best_init is None:
        best_init = rng.random(n, dtype=np.float64)
        best_init = np.maximum(best_init, 0.0)
        best_init = _project_simplex(best_init)

    a = best_init.copy()
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    best = a.copy()
    best_sc = best_sc

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    step0 = 0.8  # maybe 1.0?

    it = 0
    no_improve = 0
    target = 1.50274349

    # Main loop
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # FFT of a
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # Subgradient mask
        eps_mask = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - eps_mask
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt

        # Gradient (following previous)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        # select gradient entries corresponding to a positions:
        # In earlier code: idx = (2 * n - 2) - np.arange(n); grad = 2.0 * grad_full[idx]
        idx_arr = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx_arr]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = step0 / (math.sqrt(it) + 1.0)  # slower decay

        update = step * m_hat / (np.sqrt(v_hat) + eps)

        # Additive candidate
        cand_add = _project_simplex(a - update)

        # Multiplicative candidate
        exp_factor = np.exp(-step * grad)
        cand_mul = a * exp_factor
        s = cand_mul.sum()
        if s > 0:
            cand_mul = cand_mul / s
        else:
            cand_mul = a.copy()

        # Scores
        cand_add_sc = _score_vec(cand_add)
        cand_mul_sc = _score_vec(cand_mul)

        if cand_add_sc < cand_mul_sc:
            cand, cand_sc = cand_add, cand_add_sc
        else:
            cand, cand_sc = cand_mul, cand_mul_sc

        # Acceptance criteria
        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc
            a = cand
            no_improve = 0
        elif cand_sc < _score_vec(a):
            a = cand
            no_improve = 0
        else:
            no_improve += 1
            # Simulated annealing acceptance
            prob = math.exp(-(cand_sc - best_sc) / (step + 1e-12))
            if rng.random() < prob:
                a = cand

        # Symmetrization occasional
        if it % 500 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym)
            if sym_sc < best_sc:
                best = sym.copy()
                best_sc = sym_sc
                a = sym
                no_improve = 0

        # Random perturb when stuck
        if no_improve > 2000:
            # small perturbation: pick random index and move some mass
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)

            a = _project_simplex(a)
            no_improve = 0

        if best_sc < target:
            break

    # Return result as list
    return [float(x) for x in best]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(vec):
    """Score 2*n*max(conv) for a vector already summed to 1."""
    a = np.asarray(vec, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
