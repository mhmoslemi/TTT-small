import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _conv_max(a, L):
    """Maximum entry of a*a (full convolution) using FFT."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * a.size - 1].max()

    rng = np.random.default_rng(seed)

        cur = np.maximum(cur, 0.0)
    else:
        n = rng.integers(2000, 3000)
        cur = rng.random(n)
    cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑2 FFT length

    # current score (sum is 1, so score = 2*n*max_conv)
    cur_sc = 2.0 * n * _conv_max(cur, L)
    best_vec = cur.copy()
    best_sc = cur_sc

    # -------------------- Adam state --------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # temperature for soft‑max weighting of the convolution
    temp = 0.1
    temp_factor = 1.002
    temp_max = 200.0

    step0 = 0.6          # base step size
    it = 0

        it += 1

        # convolution and its maximum

        conv = conv_full[: 2 * n - 1]
