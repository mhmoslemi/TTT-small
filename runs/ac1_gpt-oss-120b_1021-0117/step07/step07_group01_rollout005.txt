def _optimize_from(a0, deadline, global_best_score):
    a = a0.copy()
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    best_local = a.copy()
    best_local_score = _score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    base_lr = 0.2
    target = 1.50274349
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1
        # compute conv and max
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()
        score = float(2 * n * max_val)

        if score < best_local_score:
            best_local_score = score
            best_local = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if score < global_best_score:
            # could break early, but we want to continue for better?
            pass

        if score < target:
            break

        # softmax temperature
        temp = max(1e-4, 0.02 / math.sqrt(it))
        # compute w
        # shift conv by max to avoid large exponents
        diff = conv - max_val
        w = np.exp(diff / temp)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.ones_like(conv) / conv.size
        else:
            w = w / w_sum

        # gradient via fft
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = base_lr / (math.sqrt(it) + 1.0)
        update = step * m_hat / (np.sqrt(v_hat) + eps)

        cand1 = _project_simplex(a - update)

        exp_factor = np.exp(-step * grad)
        cand2 = a * exp_factor
        cand2_sum = cand2.sum()
        if cand2_sum > 0:
            cand2 = cand2 / cand2_sum
        else:
            cand2 = cand1

        # evaluate candidates
        sc1 = _score(cand1)
        sc2 = _score(cand2)

        if sc1 < sc2:
            cand = cand1
            cand_sc = sc1
        else:
            cand = cand2
            cand_sc = sc2

        # accept if better than current a
        if cand_sc < score:
            a = cand

        # occasional random perturbation after stagnation
        if no_improve > 2000:
            # reposition: mixture with random vector

            rand = _project_simplex(rand)
            a = 0.5 * best_local + 0.5 * rand
            a = _project_simplex(a)
            no_improve = 0

    return best_local, best_local_score

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _optimize_from(a0, deadline, global_best, rng):
    """Projected Adam on the smooth‑max of the autocorrelation."""
    a = a0.copy()
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    best_local = a.copy()
    best_local_sc = _score(best_local)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    base_lr = 0.2
    target = 1.50274349
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # convolution & score
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()
        sc = float(2.0 * n * max_val)

        if sc < best_local_sc:
            best_local_sc = sc
            best_local = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if sc < target:
            break
        if sc < global_best:
            # allow early break only if enough time remains for safety
            pass

        # smooth max weights (softmax)
        temp = max(1e-4, 0.02 / math.sqrt(it))
        diff = conv - max_val
        w = np.exp(diff / temp)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.ones_like(conv) / conv.size
        else:
            w /= w_sum

        # gradient of weighted sum of autocorrelation
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = base_lr / (math.sqrt(it) + 1.0)
        update = step * m_hat / (np.sqrt(v_hat) + eps)

        # two candidate updates
        cand1 = _project_simplex(a - update)

        exp_factor = np.exp(-step * grad)
        cand2 = a * exp_factor
        s = cand2.sum()
        if s > 0:
            cand2 = cand2 / s
        else:
            cand2 = cand1

        sc1 = _score(cand1)
        sc2 = _score(cand2)

        if sc1 < sc2:
            cand, cand_sc = cand1, sc1
        else:
            cand, cand_sc = cand2, sc2

        # accept if improvement
        if cand_sc < sc:
            a = cand

        # occasional random restart when stalled
        if no_improve > 2500:
            rand = rng.random(n, dtype=np.float64)
            rand = _project_simplex(rand)
            a = 0.5 * best_local + 0.5 * rand
            a = _project_simplex(a)
            no_improve = 0

    return best_local, best_local_sc

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # Build a few diverse initial vectors (no large literals)
    init_vecs = []

    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        a0 = _project_simplex(a0)
        init_vecs.append(a0)

    # length reference
    n = init_vecs[0].size if init_vecs else 2000

    # sinusoidal shape
    sin_vec = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    sin_vec = _project_simplex(sin_vec)
    init_vecs.append(sin_vec)

    # triangular shape
    half = np.arange(1, (n + 1) // 2 + 1, dtype=np.float64)
    if n % 2 == 0:
        tri_vec = np.concatenate([half, half[::-1]])
    else:
        tri_vec = np.concatenate([half, half[-2::-1]])
    tri_vec = _project_simplex(tri_vec)
    init_vecs.append(tri_vec)

    # random uniform
    rand_vec = rng.random(n, dtype=np.float64)
    rand_vec = _project_simplex(rand_vec)
    init_vecs.append(rand_vec)

    # ------------------------------------------------------------------
    best_overall = None
    best_score = float('inf')

    for vec in init_vecs:
        if time.monotonic() >= deadline:
            break
        cand, sc = _optimize_from(vec, deadline, best_score, rng)
        if sc < best_score:
            best_score = sc
            best_overall = cand
        # early exit if target reached
        if best_score < 1.50274349:
            break

    # fallback to a simple normalized vector if all else fails
    if best_overall is None:
        fallback = np.ones(n, dtype=np.float64)
        fallback = _project_simplex(fallback)
        best_overall = fallback

    return [float(x) for x in best_overall]
