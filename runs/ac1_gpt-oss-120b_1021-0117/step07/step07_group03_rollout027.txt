import time, math, numpy as np

def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _conv_max_and_grad(a, L):
    """
    Compute (max of full convolution a*a, gradient of that max w.r.t. a).
    a is non‑negative, but the routine works for any real a.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]                # valid part of full conv
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)            # reverse‑indexing for gradient
    grad = 2.0 * grad_full[idx]
    return max_val, grad

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())      # next power of two ≥ 2n‑1

    # initial symmetrisation (helps reduce peaks)
    cur = (cur + cur[::-1]) * 0.5
    cur = _project_simplex(cur)

    max_val, _ = _conv_max_and_grad(cur, L)
    cur_sc = 2.0 * n * max_val
    best_sc = cur_sc
    best_vec = cur.copy()

    # ---- Adam optimiser ----------------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2
    t = 0

    no_improve = 0
    target = 1.50274349

    while True:
        if time.time() >= deadline:
            break

        max_val, grad = _conv_max_and_grad(cur, L)
        cur_sc = 2.0 * n * max_val

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = _project_simplex(cur - step)

        # occasional symmetrisation
        if t % 100 == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        cand_max, _ = _conv_max_and_grad(cand, L)
        cand_sc = 2.0 * n * cand_max

        # accept if improvement, else occasional downhill move
        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_sc = cand_sc
                no_improve = 0
            else:
                no_improve += 1

        # record best
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0

        # random shake‑up if stuck
        if no_improve > 2000:
            sigma = 0.05 / math.sqrt(t)
            pert = cur + sigma * rng.standard_normal(size=n)
            pert = _project_simplex(pert)
            pert = (pert + pert[::-1]) * 0.5
            pert = _project_simplex(pert)
            pert_max, _ = _conv_max_and_grad(pert, L)
            pert_sc = 2.0 * n * pert_max
            if pert_sc < cur_sc:
                cur = pert
                cur_sc = pert_sc
                if pert_sc < best_sc:
                    best_sc = pert_sc
                    best_vec = pert.copy()
                no_improve = 0
            else:
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0

        if best_sc < target:
            break

    return [float(x) for x in best_vec]
