import time, math, numpy as np

# ------------------------------------------------------------
def _project_simplex(v):
    """ Euclidean projection onto the probability simplex (sum=1, v>=0). """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_and_grad(a, L):
    """
    Compute max value of linear convolution a*a and a subgradient.
    a is a non‑negative vector (not required to sum to 1).
    L is a power‑of‑two >= 2*len(a)-1.
    Returns (max_val, grad) where grad has same shape as a.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    # gradient = 2 * a * mask_rev (convolution)
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad
