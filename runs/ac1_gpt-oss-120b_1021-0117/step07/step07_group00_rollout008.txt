import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

            w[conv == max_val] = 1.0
            w_sum = w.sum()
        else:
            w = np.exp((conv - max_val) / temp)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = (conv == max_val).astype(np.float64)
                w_sum = w.sum()
        w /= w_sum

        # ---- gradient of smoothed max (via FFT)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ---- Adam update
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v2 / (1.0 - beta2 ** t_iter)
        step = 0.4 * m_hat / (np.sqrt(v_hat) + eps)   # base step size 0.4

        cand = _project_simplex(cur - step)

        # ---- candidate score
        a_fft_c = np.fft.rfft(cand, n=L)
        conv_full_c = np.fft.irfft(a_fft_c * a_fft_c, n=L)
        cand_sc = 2.0 * n * conv_full_c[: 2 * n - 1].max()

        # ---- acceptance
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1
            # occasional uphill move (tiny probability)
            if rng.random() < 0.001:
                cur = cand

        # ---- random redistribution when stuck
        if no_improve > 2000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            a_fft_t = np.fft.rfft(tmp, n=L)
            conv_full_t = np.fft.irfft(a_fft_t * a_fft_t, n=L)
            tmp_sc = 2.0 * n * conv_full_t[: 2 * n - 1].max()
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
            else:
                no_improve = 0   # reset to avoid deadlock

        # ---- symmetrisation every few hundred steps
        if t_iter % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            a_fft_s = np.fft.rfft(sym, n=L)
            conv_full_s = np.fft.irfft(a_fft_s * a_fft_s, n=L)
            sym_sc = 2.0 * n * conv_full_s[: 2 * n - 1].max()
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # ---- early termination
        if best_sc < target:
            break

    # Return the best found vector as a plain Python list
    return [float(x) for x in best_vec]
