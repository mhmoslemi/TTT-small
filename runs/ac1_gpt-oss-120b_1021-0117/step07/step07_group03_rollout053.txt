def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # init vector
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)  # default length
    cur = _project_simplex(np.maximum(cur, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())  # power of two for fft

    # optimizer state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.5  # initial
    it = 0
    best_vec = cur.copy()
    max_val, _ = _conv_max_and_grad(cur, L)
    best_sc = 2.0 * n * max_val
    best_score = best_sc
    no_improve = 0

    target = 1.50274349

    while time.time() < deadline:
        it += 1
        max_val, grad = _conv_max_and_grad(cur, L)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = base_lr / math.sqrt(it)  # decaying
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = cur - step
        cand = _project_simplex(cand)

        # occasional symmetrization
        if it % 300 == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        # evaluate candidate
        cand_max, _ = _conv_max_and_grad(cand, L)
        cand_sc = 2.0 * n * cand_max

        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Accept candidate if better else keep current (no acceptance of worse)
        if cand_sc < 2.0 * n * max_val:
            cur = cand
        # else we keep cur unchanged

        # random perturb when stagnant
        if no_improve >= 2000:
            sigma = 0.01 / math.sqrt(it)

            pert_sc = 2.0 * n * pert_max

                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0

        if best_sc < target:

    return [float(x) for x in best_vec]

import time, math, numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v):
    """ Euclidean projection of v onto {x>=0, sum(x)=1}. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _conv_max_and_grad(a, L):
    """
    Returns (max value of a*a, subgradient) for non‑negative a.
    a is assumed to be a 1‑D NumPy array.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a list of non‑negative coefficients that (hopefully) improves
    the AC1 bound.  The implementation runs a projected Adam optimizer

    """

    start = time.time()

    m = np.zeros_like(cur)          # first moment
    v = np.zeros_like(cur)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.5
    it = 0
    no_improve = 0
    target = 1.50274349

    # ---- best‑so‑far tracking -------------------------------------------
    max_val, _ = _conv_max_and_grad(cur, L)
    best_sc = 2.0 * n * max_val
    best_vec = cur.copy()

    # ---- main loop -------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # gradient of current point
        max_val, grad = _conv_max_and_grad(cur, L)
        cur_sc = 2.0 * n * max_val

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
