    temp_factor = None
    decay = 0.995
    min_temp_frac = 0.02

    while True:
        if time.time() >= deadline:
            break

        t += 1
        # FFT of current vector
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # temperature based on max_val
        if temp_factor is None:
            temp_factor = max_val * 0.05
        else:
            temp_factor = max(temp_factor * decay, max_val * min_temp_frac)

        # softmax weights
        w = np.exp((conv - max_val) / (temp_factor + 1e-12))
        w_sum = w.sum()

            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc

            # simulated annealing based on temperature
            prob = math.exp(-(cand_sc - cur_sc) / (step0 + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # keep best

            best_sc, best_vec = cand_sc, cand.copy()

        # periodic operations

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast score for a vector that is (approximately) on the simplex."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
