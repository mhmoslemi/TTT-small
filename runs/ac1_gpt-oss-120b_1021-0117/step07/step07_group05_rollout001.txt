import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that (approximately)
    minimizes the AC1 upper bound measured by ``evaluate_sequence``.
    The routine works with any global ``height_sequence_1`` that may already
    be present; otherwise it falls back to a uniform distribution of the same
    length.
    """
    start = time.monotonic()
    # keep a tiny safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # --------------------------------------------------------------
    # Determine problem size
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000          # a reasonable default if no parent is given

    # --------------------------------------------------------------
    # Simple Euclidean projection onto the probability simplex (sum == 1)
    def _project_simplex(v):
        v = np.asarray(v, dtype=np.float64)
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = css[rho] / (rho + 1)
        w = v - theta
        w[w < 0] = 0.0
        return w

    # --------------------------------------------------------------
    # Candidate 1 – the uniform simplex distribution
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_vec = uniform.copy()
    try:
        best_score = evaluate_sequence(best_vec.tolist())
    except Exception:
        best_score = float("inf")

    # --------------------------------------------------------------
    # Candidate 2 – the supplied parent, if any (clipped to allowed range)
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        try:
            parent_score = evaluate_sequence(parent.tolist())
            if parent_score < best_score:
                best_score = parent_score
                best_vec = parent
        except Exception:
            pass

    # --------------------------------------------------------------
    # Simple stochastic local search with simplex projection
    rng = np.random.default_rng(seed)
    cur = best_vec.copy()
    cur_score = best_score
    it = 0
    max_iters = 5000

    while time.monotonic() < deadline and it < max_iters:

    return [float(x) for x in best_vec]
