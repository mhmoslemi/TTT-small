def _conv_max_and_grad(a, L):
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad

import time, math, numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}. """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _conv_max_and_grad(a, L):
    """
    Compute (max_convolution, gradient) of the full convolution a*a.
    a is a non‑negative vector; L is a power‑of‑two >= 2*len(a)-1.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = (conv >= max_val - eps).astype(np.float64)
    cnt = mask.sum()
    if cnt > 0:
        mask /= cnt
    # reversed mask for gradient
    w_rev = mask[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_val, grad

    # best solution tracking
    max_val, _ = _conv_max_and_grad(cur, L)
    cur_sc = 2.0 * n * max_val
    best_vec = cur.copy()
    best_sc = cur_sc

    # Adam optimizer state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 1.0

    it = 0
    no_improve = 0
    target = 1.50274349   # goal bound

    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # gradient at current point
        max_val, grad = _conv_max_and_grad(cur, L)
        cur_sc = 2.0 * n * max_val

        # Adam momentum updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # adaptive step (decays ~1/sqrt(it))
        step_size = lr0 / math.sqrt(it)
        step = step_size * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur - step)

        # occasional symmetrisation (helps flatten peaks)
        if it % 150 == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        # evaluate candidate (only need max value)
        cand_max, _ = _conv_max_and_grad(cand, L)
        cand_sc = 2.0 * n * cand_max

        # acceptance rule (mostly greedy, occasional annealing)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (step_size + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
                no_improve = 0
            else:
                no_improve += 1

        # update best solution
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0

        # random perturbation when stuck
        if no_improve > 2500:
            sigma = 0.05 / math.sqrt(it)
            pert = cur + sigma * rng.standard_normal(size=n)
            pert = _project_simplex(pert)
            pert_max, _ = _conv_max_and_grad(pert, L)
            pert_sc = 2.0 * n * pert_max
            if pert_sc < cur_sc:
                cur, cur_sc = pert, pert_sc
                if pert_sc < best_sc:
                    best_sc = pert_sc
                    best_vec = pert.copy()
            else:
                # reset Adam moments to escape flat region
                m.fill(0.0)
                v.fill(0.0)
            no_improve = 0

        # early exit if target achieved
        if best_sc < target:
            break

    # Return the best vector found as plain Python floats
    return [float(x) for x in best_vec]
