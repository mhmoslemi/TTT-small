import time, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = rng.integers(2000, 3000)
        a = rng.random(n)

    a = _project_simplex(np.maximum(a, 0.0))
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    best = a.copy()
    best_sc = _score(best)
    cur_sc = best_sc

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    step0 = 0.5
    it = 0
    no_improve = 0
    target = 1.50274349

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        k = int(np.argmax(conv))
        max_val = float(conv[k])

        # gradient of max conv term
        g = np.zeros_like(a)
        if k < n:
            g[:k+1] += a[k::-1]
            if k > 0:
                g[k:] += a[:n - k]
        else:
            start_idx = k - n + 1
            # slice a[n-1:start_idx-1:-1]
            g[start_idx:] += a[n-1:start_idx-1:-1]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (np.sqrt(v_hat) + eps)

        cand = a - step * m_hat
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        if cand_sc < cur_sc:
            a, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            if rng.random() < 0.001:
                a, cur_sc = cand, cand_sc

        if it % 400 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                a, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc

        if no_improve > 2500:
            i = rng.integers(0, n)
            if a[i] > 0:
                delta = a[i] * rng.random() * 0.4
                j = rng.integers(0, n-1)
                if j >= i:

                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)

                if tmp_sc < cur_sc:
                    a, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best, best_sc = tmp.copy(), tmp_sc

        if best_sc < target:
            break

    return [float(x) for x in best]

import time, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = rng.integers(2000, 3000)
        a = rng.random(n)

    a = _project_simplex(np.maximum(a, 0.0))
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    best = a.copy()
    best_sc = _score(best)
    cur_sc = best_sc

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    step0 = 0.5
    it = 0
    no_improve = 0
    target = 1.50274349

    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # convolution and max index
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        k = int(np.argmax(conv))

        # exact gradient of the max-convolution term
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        g[i] += a[j]
        g[j] += a[i]

        # Adam‑style update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (np.sqrt(v_hat) + eps)

        cand = a - step * m_hat
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        if cand_sc < cur_sc:
            a, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            if rng.random() < 0.001:
                a, cur_sc = cand, cand_sc

        # periodic symmetrisation
        if it % 500 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                a, cur_sc = sym, sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc

        # random redistribution when stuck
        if no_improve > 2500:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    a, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best, best_sc = tmp.copy(), tmp_sc

        if best_sc < target:
            break

    return [float(x) for x in best]
