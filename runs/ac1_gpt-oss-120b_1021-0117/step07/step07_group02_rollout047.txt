import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a0 = rng.random(2500, dtype=np.float64)
    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())     # power‑of‑two length for FFT

    # ---- optimizer state ---------------------------------------------------
    m = np.zeros_like(cur)          # first moment
    v = np.zeros_like(cur)          # second moment
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    eps = 1e-8
    lr0 = 0.2                       # base learning rate

    # temperature for soft‑max weighting of the convolution
    temp_factor = 0.01
    temp_decay = 0.9995
    min_temp_factor = 1e-6

    # ---- bookkeeping -------------------------------------------------------
    cur_sc = 2.0 * n * np.fft.irfft(np.fft.rfft(cur, n=L) *
                                   np.fft.rfft(cur, n=L), n=L)[:2 * n - 1].max()
    best_sc, best_vec = cur_sc, cur.copy()
    it, no_improve = 0, 0
    target = 1.50274349

    # ---- main optimisation loop -------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # full self‑convolution via FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # soft‑max weights approximating the max
        T = max_val * max(temp_factor, min_temp_factor)
        w = np.exp((conv - max_val) / T)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(0.0)

        # gradient of the weighted sum (2 * a * w_rev)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = m_hat / (np.sqrt(v_hat) + eps)

        lr = lr0 / math.sqrt(it)
        cand = cur - lr * step
        cand = _project_simplex(cand)

        # enforce symmetry (helps the autocorrelation structure)
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        # evaluate candidate
        a_fft_c = np.fft.rfft(cand, n=L)
        conv_c = np.fft.irfft(a_fft_c * a_fft_c, n=L)[: 2 * n - 1]
        cand_sc = 2.0 * n * conv_c.max()

        # accept if better (deterministic) – otherwise occasional Metropolis step
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
                no_improve = 0

        # update global best
        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()

        # temperature decay
        temp_factor = max(temp_factor * temp_decay, min_temp_factor)

        # stagnation handling
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp = _project_simplex(tmp)
                tmp_sc = 2.0 * n * np.fft.irfft(
                    np.fft.rfft(tmp, n=L) * np.fft.rfft(tmp, n=L), n=L
                )[: 2 * n - 1].max()
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
            else:
                # full random restart with symmetry
                cur = _project_simplex(rng.random(n))
                cur = (cur + cur[::-1]) * 0.5
                cur = _project_simplex(cur)
                m.fill(0.0)
                v.fill(0.0)
                beta1_pow, beta2_pow = 1.0, 1.0
                no_improve = 0
                cur_sc = 2.0 * n * np.fft.irfft(
                    np.fft.rfft(cur, n=L) * np.fft.rfft(cur, n=L), n=L
                )[: 2 * n - 1].max()

        # early exit if target reached
        if best_sc < target:
            break

        # periodic time check
        if it % 1000 == 0 and time.monotonic() >= deadline:
            break

    # return Python list of floats
    return [float(x) for x in best_vec]
