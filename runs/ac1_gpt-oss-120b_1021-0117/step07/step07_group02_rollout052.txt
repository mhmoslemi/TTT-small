    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        return np.full_like(v, 1.0 / v.size)
    return v / s

def _score_vec(a):
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    if s <= 0:
        return float('inf')
    return float(2.0 * n * max_val / (s * s))

import time, math, numpy as np

def _simplex_proj(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (non‑negative, sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _score_vec(a: np.ndarray) -> float:
    """Score 2·n·max(conv) / (sum(a)²) using FFT‑based convolution."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    m = conv.max()
    s = a.sum()
    if s <= 0.0:
        return float('inf')
    return float(2.0 * n * m / (s * s))
