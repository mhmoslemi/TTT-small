def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2) # leaving margin

    rng = np.random.default_rng(seed)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        a = rng.random(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)  # ensures sum=1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    # objective function using normalized sum = 1
    def objective(vec):
        # compute max convolution value
        af = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    best = a.copy()
    best_obj = objective(best)

    # initial step size
    step = 0.8
    min_step = 1e-7
    # parameters
    max_backtrack = 10
    no_improve = 0
    # thresholds for random perturb
    stagnation_limit = 2000

    while time.monotonic() < deadline:
        # compute gradient
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        # mask: positions close to max
        eps_mask = max_val * 1e-12 + 1e-15
        mask = conv >= max_val - eps_mask
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        cur_obj = best_obj  # maybe we need objective of current a?
        # Actually we also want current objective for line search
        cur_obj = objective(a)

        # line search
        improved = False
        step_try = step
        for _ in range(max_backtrack):
            cand = _project_simplex(a - step_try * grad)
            cand_obj = objective(cand)
            if cand_obj < cur_obj - 1e-15:  # improvement
                a = cand
                cur_obj = cand_obj
                improved = True
                # adapt step
                step = step_try * 1.2
                break
            else:
                step_try *= 0.5
        if not improved:
            step = max(step * 0.5, min_step)

        # track best
        if cur_obj < best_obj:
            best = a.copy()
            best_obj = cur_obj
            no_improve = 0
        else:
            no_improve += 1

        # random perturb when stagnate
        if no_improve >= stagnation_limit:
            # perform random redistribution
            i = rng.integers(0, n)
            if a[i] > 0:
                delta = a[i] * rng.random() * 0.5
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                a = _project_simplex(tmp)
            no_improve = 0
            # also maybe reset step
            step = 0.8

        # early exit if meet target bound (converted to objective)
        # target objective = 2 * n * target_score (since sum=1)
        target_score = 1.50274349
        target_obj = 2 * n * target_score
        if best_obj < target_obj:
            break

    # final candidate: need to return list of floats ensuring sum>0.01
    # The evaluator will clip negative, min with 0, max 1000, but we should provide positive.
    # Ensure minimal sum >0.01: due to projection onto simplex sum=1, OK.
    return [float(x) for x in best]

```
def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
```

import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _gradient_and_max(a, L, n):
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    mask = conv >= max_val - eps
    w = mask.astype(np.float64)
    cnt = w.sum()
    if cnt > 0:
        w /= cnt
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, max_val

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0

    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)
