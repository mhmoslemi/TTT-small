def proj_simplex(v):
    # as per previous code
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, len(v)+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time
import numpy as np

def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum == 1, v >= 0)."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def grad_maxconv(v: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv(v, v)) at the index k where the maximum occurs.
    Returns a vector of the same length as v.
    """
    n = v.shape[0]
    g = np.zeros_like(v)

    # first term: v[k - j] for j with 0 <= k-j < n
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    if lo <= hi:
        j = np.arange(lo, hi + 1)
        g[j] += v[k - j]

    # second term: v[j - k] for j with 0 <= j-k < n
    lo2 = max(k, 0)
    hi2 = min(k + n - 1, n - 1)
    if lo2 <= hi2:
        j2 = np.arange(lo2, hi2 + 1)
        g[j2] += v[j2 - k]

    return g

def sequence_score(v: np.ndarray, n: int) -> float:
    """Exact AC1 upper‑bound score for a non‑negative vector v."""
    s = v.sum()
    if s < 0.01:
        return float("inf")
    conv = np.convolve(v, v)
    return 2.0 * n * conv.max() / (s * s)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound measured by ``evaluate_sequence``.
    The routine works with any parent sequence ``height_sequence_1`` that
    may already be present; otherwise it starts from the uniform distribution.
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)

    # ------------------------------------------------------------------
    # Problem size and initial vectors
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    else:
        n = 2000
        parent = None

    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    best_vec = uniform.copy()
    best_score = sequence_score(best_vec, n)

    if parent is not None:
        parent_norm = parent / max(parent.sum(), 1e-12)
        parent_score = sequence_score(parent_norm, n)
        if parent_score < best_score:
            best_vec = parent_norm.copy()
            best_score = parent_score

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Main optimisation loop (projected gradient descent with back‑tracking)
    cur = best_vec.copy()
    cur_score = best_score
    step = 0.12                     # initial step size
    max_iters = 20000               # safety upper bound

    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1

        # current convolution, max index and gradient
        conv_cur = np.convolve(cur, cur)
        k_max = int(conv_cur.argmax())
        grad = grad_maxconv(cur, k_max)

        improved = False
        # try a few back‑tracking factors
        for factor in (1.0, 0.5, 0.2, 0.1, 0.05):
            cand = cur - step * factor * grad
            cand = project_simplex(cand)
            cand_score = sequence_score(cand, n)

            if cand_score < cur_score - 1e-12:

                cur_score = cand_score
                improved = True
                # keep the global best

                    best_score = cand_score
                    best_vec = cand.copy()
                # modestly increase step after success

            if step < 1e-12:

            cand_score = sequence_score(cand, n)
            if cand_score < cur_score:
                cur = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_vec = cand.copy()
                step *= 1.02

    except Exception:
        # fallback to a safe uniform vector if something went wrong
        best_vec = uniform.copy()
