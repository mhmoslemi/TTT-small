def _renorm(v):
    """Clip to non‑negative and renormalise to sum‑one."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-12:        # fall back to uniform if everything vanished
        n = v.shape[0]
        return np.full(n, 1.0 / n, dtype=np.float64)
    return v / s
