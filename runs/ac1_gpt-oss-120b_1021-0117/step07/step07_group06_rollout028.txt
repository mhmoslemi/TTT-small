    from scipy.optimize import linprog
    n = f.size
    # Build matrix A
    diags = []
    offsets = []
    for i in range(n):
        # diag values of length n - i
        diag = np.full(n - i, f[i], dtype=np.float64)
        diags.append(diag)
        offsets.append(i)
    A = sp.diags(diags, offsets, shape=(2*n-1, n), format='csr')
    # Linear program:
    # minimize -sum(b) => c = -1 vector
    c = -np.ones(n, dtype=np.float64)
    b_ub = M * np.ones(2*n-1, dtype=np.float64)
    # bounds: (0, None) for each var
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs', options={'time_limit': 20})
    if not res.success:
        return None
    b = res.x
    # ensure sum>0 (should be)
    return b

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    return w

def _score(a):
    """Score 2*n*max(conv)/(sum(a)^2). a is a numpy array."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    if s == 0:
        return float('inf')
    return float(2.0 * n * max_val / (s * s))

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # initialization
    if "height_sequence_1" in globals():

        a0 = rng.random(2000, dtype=np.float64) + 1e-3
    a = _project_simplex(np.maximum(a0, 0.0))
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = a.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # optimisation state
    momentum = np.zeros_like(a)
    beta = 0.85          # momentum coefficient
    step0 = 0.7          # initial step magnitude
    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- FFT of current vector -----
        a_fft = np.fft.rfft(a, n=L)

        # ----- convolution (autocorrelation) -----
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- exact subgradient (indicator of max entries) -----
        eps = max_val * 1e-12 + 1e-16
        idxs = np.where(conv >= max_val - eps)[0]
        w = np.zeros_like(conv)
        if idxs.size:
            w[idxs] = 1.0 / idxs.size
        else:
            w[conv.argmax()] = 1.0

        # ----- gradient via convolution with reversed weight -----
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_rev_fft, n=L)
        idx_map = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx_map]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- step size (decays like 1/sqrt(it)) -----
        step = step0 / math.sqrt(it)

        # ----- candidate generation and evaluation -----
        cand = _project_simplex(a - step * momentum)
        cand_sc = _score(cand)

        # ----- acceptance -----
        if cand_sc < cur_sc:
            a, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:

            temp = step * 0.4 + 1e-12
            prob = math.exp(-(cand_sc - cur_sc) / temp)

        # ----- periodic symmetrisation -----
        if it % 400 == 0:

            sym_sc = _score(sym)
            if sym_sc < cur_sc:
