def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, non‑negative)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:

        theta = css[rho] / (rho + 1)

        kernel = np.ones(5) / 5.0

        inits.append(_project_simplex(smooth))

    best_vec = None
    best_sc = float("inf")

    for idx, init in enumerate(inits):
        if time.monotonic() >= deadline:
            break
        remaining = len(inits) - idx
        remaining_time = deadline - time.monotonic()
        sub_deadline = time.monotonic() + remaining_time / remaining
        vec, sc = _optimize(init, sub_deadline, rng)
        if sc < best_sc:
            best_sc, best_vec = sc, vec

    if best_vec is None:
        best_vec = inits[0]

    return [float(x) for x in best_vec]
