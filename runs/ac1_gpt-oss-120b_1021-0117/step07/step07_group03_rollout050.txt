import time, math, numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur - step)

        # occasional symmetrisation (helps flatten the convolution)
        if it % 500 == 0:
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

        # ----- evaluate candidate --------------------------------------
        a_fft_c = np.fft.rfft(cand, n=L)
        conv_full_c = np.fft.irfft(a_fft_c * a_fft_c, n=L)
        max_val_c = conv_full_c[: 2 * n - 1].max()
        cand_sc = 2.0 * n * max_val_c

        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur = cand

        # early exit if target reached
        if best_sc < target:
            break

        # random perturbation when stuck
        if no_improve > 3000:
            sigma = 0.1 / math.sqrt(it)
            pert = cur + sigma * rng.standard_normal(size=n)
            pert = _project_simplex(pert)

            a_fft_p = np.fft.rfft(pert, n=L)
            conv_full_p = np.fft.irfft(a_fft_p * a_fft_p, n=L)
            max_val_p = conv_full_p[: 2 * n - 1].max()
            pert_sc = 2.0 * n * max_val_p

            if pert_sc < best_sc:
                best_sc = pert_sc
                best_vec = pert.copy()
                cur = pert
                no_improve = 0
            else:
                # reset optimiser state to escape flat region
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0

    # return the best vector as plain Python floats
    return [float(x) for x in best_vec]
