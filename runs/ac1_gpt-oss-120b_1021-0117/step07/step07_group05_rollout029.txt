import time
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum==1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow_two(x):
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _compute_score(a, n, Nfft, Lc):
    """Exact AC1 score using FFT‑based convolution."""
    s = a.sum()
    if s < 0.01:
        return float("inf")
    A = np.fft.rfft(a, Nfft)
    c = np.fft.irfft(A * A, Nfft)[:Lc]      # a * a
    max_b = c.max()
    return float(2.0 * n * max_b / (s * s))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Generate a non‑negative step‑function (list of floats) that tries to
    minimise the AC1 upper bound.  Uses a projected gradient descent with a
    soft‑max approximation of the max‑operator and FFT‑based convolutions.
    """
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.05)   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size and initial vector
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    else:
        n = 2000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    a = _project_simplex(a)                     # enforce sum==1, non‑negative
    # ------------------------------------------------------------------
    # FFT parameters (fixed for the whole run)
    Lc = 2 * n - 1                               # length of a*a
    Nfft = _next_pow_two(3 * n - 2)              # enough for all convolutions
    idx = Lc - 1 - np.arange(n)                  # indices for gradient extraction

    # ------------------------------------------------------------------
    # Initial score
    best_vec = a.copy()
    best_score = _compute_score(a, n, Nfft, Lc)

    # ------------------------------------------------------------------
    # Optimisation parameters
    rng = np.random.default_rng(seed)
    step_scale = 0.2                               # base step magnitude
    beta = 0.9                                     # momentum coefficient
    momentum = np.zeros_like(a)
    alpha = 5.0                                    # soft‑max temperature
    it = 0

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # FFT of current a
        A = np.fft.rfft(a, Nfft)

        # auto‑convolution a*a
        c = np.fft.irfft(A * A, Nfft)[:Lc]
        max_c = c.max()

        # soft‑max weights approximating the max
        shift = max_c
        exp_vals = np.exp(alpha * (c - shift))
        w = exp_vals / exp_vals.sum()               # sum to 1, all non‑negative

        # gradient of the smooth max (2 * B a where B_{i,j}=w_{i+j})
        w_rev = w[::-1]
        W_rev = np.fft.rfft(w_rev, Nfft)
        grad_full = np.fft.irfft(A * W_rev, Nfft)   # length Nfft
        grad = 2.0 * grad_full[idx]                 # extract needed n components

        else:

    return [float(x) for x in best_vec]
