import numpy as np
import time

# ------------------------------------------------------------
#  Simple stochastic construction of a sparse non‑negative step
#  function.  The convolution of a very sparse vector tends to be
#  flat, giving a small max_b and thus a low AC1 upper bound.
# ------------------------------------------------------------
def _candidate_score(arr):
    """Helper: evaluate a numpy array with the authoritative scorer."""
    return float(evaluate_sequence(arr.tolist()))


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a sequence of non‑negative coefficients that (hopefully)
    beats the target AC1 bound 1.50274349.
    The routine builds sparse binary vectors and keeps the best one.
    """
    rng = np.random.default_rng(seed)
    # length of the problem (uses the provided parent if present)
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256

    # start with the uniform vector as a guaranteed fallback (score = 2.0)
    best_seq = np.ones(n, dtype=np.float64)
    best_score = _candidate_score(best_seq)

    # time management
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # parameters for the random sparse search
    # we try many different sparsities; the range was found
    # empirically to give good chances of a low max_b.
    min_m = max(2, n // 30)        # ~80 for n≈3800
    max_m = min(n, n // 5)         # ~760 for n≈3800 (but we stay well below)

    # main stochastic loop
    while time.monotonic() < deadline:
        # choose a random sparsity level
        m = int(rng.integers(min_m, max_m + 1))
        # random subset of indices, values are all 1.0 (scale does not matter)
        idx = rng.choice(n, size=m, replace=False)
        cand = np.zeros(n, dtype=np.float64)
        cand[idx] = 1.0
        sc = _candidate_score(cand)

        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            # early exit if we have already beaten the target bound
            if best_score <= 1.50274349:
                break

        # ----- a tiny local improvement step -----
        # try a few cheap swaps around the current best vector
        if best_score > 1.6:  # only bother while we are far from the goal
            nonzero = np.flatnonzero(best_seq)
            zero = np.flatnonzero(best_seq == 0.0)
            # a few random swaps
            for _ in range(3):
                if nonzero.size == 0 or zero.size == 0:
                    break
                i = int(rng.choice(nonzero))
                j = int(rng.choice(zero))
                tmp = best_seq.copy()
                tmp[i] = 0.0
                tmp[j] = 1.0
                sc2 = _candidate_score(tmp)
                if sc2 < best_score:
                    best_score = sc2
                    best_seq = tmp
                    nonzero = np.flatnonzero(best_seq)
                    zero = np.flatnonzero(best_seq == 0.0)
                    if best_score <= 1.50274349:
                        break

    # final safeguard: ensure a non‑empty, finite vector is returned
    if best_seq is None or best_seq.size == 0:
        best_seq = np.ones(256, dtype=np.float64)

    return [float(x) for x in best_seq]
