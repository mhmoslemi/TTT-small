def gradient_max_convolution(a, k):
    n = a.shape[0]
    low = max(0, k - (n-1))
    high = min(k, n-1)
    # number of terms = high - low + 1 = t_k

    # indices i = low..high
    # corresponding j = k - i
    i = np.arange(low, high+1)
    j = k - i
    g[i] = 2 * a[j]
    return g

Project onto simplex: typical algorithm:

def project_simplex(v):
    # ensure v sums to 1 and each >=0
    # Implementation from literature
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.nonzero(u - css / np.arange(1, n+1) > 0)[0][-1]
    theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def project_simplex(v):
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css/np.arange(1, len(v)+1) > 0)[0][-1]
    theta = css[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def simplex_proj(v):
    # v: np.ndarray
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.nonzero(u - css / np.arange(1, n+1) > 0)[0][-1]
    theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)   # reserve few seconds

    # Determine n
    if 'height_sequence_1' in globals():
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent_arr.size
    else:
        n = 256

    # Compute sqrt series base
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            # sum of a[1:k] * a[k-1:0:-1]
            s = float(np.dot(a[1:k], a[k-1:0:-1]))
        a[k] = (1.0 - s) / 2.0

    # Normalize to sum=1
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
        a_sum = 1.0
    else:
        a = a / a_sum

    # Evaluate initial score
    conv = np.convolve(a, a)
    best_score = 2.0 * n * conv.max() / (a.sum() ** 2)
    best = a.copy()

    # Helper to compute score
    def compute_score(vec):
        c = np.convolve(vec, vec)
        s = 2.0 * n * c.max() / (vec.sum() ** 2)
        return s

    # Simple hill climbing with random mass transfers
    # Set parameters
    max_iter = 20000
    # We will use a step size factor up to 0.3 (fraction of source amount)
    while time.monotonic() < deadline and max_iter > 0:
        max_iter -= 1
        # pick source index with larger value
        i = rng.integers(0, n)
        if a[i] == 0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        # choose delta up to fraction of source value
        frac = rng.random() * 0.3   # up to 30%
        delta = a[i] * frac
        if delta <= 0:
            continue
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta
        # compute new score
        score = compute_score(a_candidate)
        if score < best_score:
            best_score = score
            best = a_candidate.copy()
            a = a_candidate   # accept move
        # optional: occasionally accept improving move even if not best ?
        # not needed
    # Return best list
    return [float(x) for x in best]

def simplex_proj(v):
    # Project onto simplex {x>=0, sum(x)=1}
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho_arr = u - css / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
```

def compute_score_from_vec(v):
    n = v.shape[0]
    L = 1 << ((2*n-1).bit_length())
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fv*fv, n=L)[:2*n-1]
    max_val = np.max(conv)
    return 2.0 * n * max_val / (v.sum() ** 2)

import time
import numpy as np

# ------------------------------------------------------------
# helper: project a vector onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    # find the last index where the condition holds
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# generate the sqrt-series coefficients (A*A = 1+...+x^{2n-2})
# ------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """Return the first n coefficients of the series a where a*a = 1+…+x^{2n-2}."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        if k == 1:
            s = 0.0
        else:
            # sum_{i=1}^{k-1} a_i * a_{k-i}
            s = float(np.dot(a[1:k], a[k - 1 : 0 : -1]))
        a[k] = (1.0 - s) / 2.0
    return a


# ------------------------------------------------------------
# compute score quickly (exactly the same as evaluate_sequence)
# ------------------------------------------------------------
def _score(vec: np.ndarray) -> float:
    n = vec.size
    # convolution via FFT (fast for n≈2500)
    L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max() / (vec.sum() ** 2)

# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty sequence of non‑negative floats with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # length of the problem (use provided parent if present)
    if "height_sequence_1" in globals():
        n = int(np.asarray(height_sequence_1).size)
    else:
        n = 256

    # start from the analytic sqrt‑series solution
    base = _sqrt_series(n)
    base /= base.sum()                     # normalise to sum = 1
    best_vec = base.copy()
    best_sc = _score(best_vec)

    # --------------------------------------------------------
    # simple hill‑climbing: random mass transfers preserving sum
    # --------------------------------------------------------
    # parameters
    max_iter = 20000
    while time.monotonic() < deadline and max_iter > 0:
        max_iter -= 1

        # pick a source index (with some mass) and a distinct target
        i = rng.integers(0, n)
        if best_vec[i] == 0.0:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction of the source mass to the target
        frac = rng.random() * 0.25            # up to 25 % of the source amount
        delta = best_vec[i] * frac
        if delta <= 0.0:
            continue

        cand = best_vec.copy()
        cand[i] -= delta
        cand[j] += delta
        # keep the vector on the simplex (numeric safety)
        cand = _project_simplex(cand)

        sc = _score(cand)
        if sc < best_sc:
            best_sc = sc
            best_vec = cand

        # occasional aggressive gradient step: reduce the very first coefficient
        if rng.random() < 0.02:   # 2 % chance
            grad = np.zeros_like(best_vec)
            # active index = 0 (conv[0] is maximal for the sqrt series)
            grad[0] = 2.0 * best_vec[0]
            step = rng.random() * 0.5
            trial = _project_simplex(best_vec - step * grad)
            sc2 = _score(trial)
            if sc2 < best_sc:
                best_sc = sc2
                best_vec = trial

    # convert to plain Python list of floats for the evaluator
    return [float(x) for x in best_vec]
