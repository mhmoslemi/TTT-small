    # Determine length n

    best_seq = parent.copy()

import numpy as np
import time

def _sidon_like_sequence(n):
    """Generate the classic a_i = C(2i,i)/4**i sequence (length n)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        # a_i = a_{i-1} * (1 - 1/(2*i))
        a[i] = a[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return a

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a candidate sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Length of the problem – use the provided parent if available

            raise Exception
    except Exception:
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)

    # best solution so far
    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # deterministic good construction (Sidon‑like)
    seq = _sidon_like_sequence(n)
    seq = np.clip(seq, 0.0, 1000.0)
    score = float(evaluate_sequence(seq.tolist()))
    if np.isfinite(score) and score < best_score:
        best_seq, best_score = seq, score

    # simple local‑search: random swaps that try to lower the score
    # we keep the sum unchanged – only the ordering of the values changes.
    # This is cheap (O(n log n) per evaluation) and often reduces the max.
    current = best_seq.copy()
    while time.monotonic() < deadline:
        # propose a swap of two positions
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cand = current.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best_seq, best_score = cand, cand_score
            current = cand
        # occasional random restart to avoid stagnation
        if rng.random() < 0.001:
            current = seq.copy()
            perm = rng.permutation(n)
            current = current[perm]

    return [float(x) for x in best_seq]
