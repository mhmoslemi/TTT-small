def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)
    
    # try get length from height_sequence_1 if exists and valid
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
    except Exception:
        n = 256
    # ensure positive length
    if n <= 0:
        n = 256
    
    # target score (optional)
    target = 1.50274349
    
    # Determine K based on target
    K_target = int(np.ceil(np.sqrt(4.0 * n / target)))
    # choose K <= min(K_target+20, n) but also safe for Sidon
    K = min(K_target + 10, n//2)  # maybe 150
    # But we will attempt to build Sidon set; if fails, reduce K gradually.
    best_seq = None
    best_score = float('inf')
    
    while time.monotonic() < deadline:
        # attempt to construct Sidon set of size K
        positions = []
        used_sums = set()
        attempts = 0
        # limit attempts to avoid infinite loops
        while len(positions) < K and attempts < K * 100:
            cand = int(rng.integers(0, n))
            if cand in positions:
                attempts += 1
                continue
            s_self = cand + cand
            if s_self in used_sums:
                attempts += 1
                continue
            collision = False
            for p in positions:
                if cand + p in used_sums:
                    collision = True
                    break
            if collision:
                attempts += 1
                continue
            # ok, add
            positions.append(cand)
            used_sums.add(s_self)
            for p in positions[:-1]:
                used_sums.add(cand + p)
        if len(positions) < K:
            # reduce K and try again
            K = max(3, K - 5)
            continue
        # build sequence
        seq = np.zeros(n, dtype=np.float64)
        seq[positions] = 1.0
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = seq.copy()
            best_score = score
            # if we beat target, break
            if best_score <= target:
                break
        # Optionally try reduce K further if score not good
        if time.monotonic() > deadline - 0.5:
            break
    if best_seq is None:
        # fallback to uniform vector (as simple as original parent)
        try:
            base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
            if base.ndim != 1 or base.size == 0:
                raise Exception()
        except Exception:
            base = np.ones(256, dtype=np.float64)

    return [float(x) for x in best_seq]

import numpy as np
import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Construct a non‑negative step function whose self‑convolution peak
    is deliberately kept small.  The construction tries to keep the
    number of ordered pairs that fall into any diagonal (i+j = const)
    below a small constant C.  For a binary sequence this bounds the
    maximal convolution entry, yielding a low AC1 score.
    """
    rng = np.random.default_rng(seed)
    # deadline a little before the hard timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # 1.  Use the provided parent as a fallback candidate

                         0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0:
            raise Exception()
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------
    # 2.  Try to build a low‑peak binary sequence
    # ------------------------------------------------------------------
    n = int(parent.shape[0])

    # maximal allowed load per diagonal (ordered pair count)
    C = 4                       # a small integer, 4 works well for n≈6000
    # target number of ones – chosen so that 2·n·C / K² < 1.5
    K_target = int(0.85 * np.sqrt(C * (2 * n - 1)))
    K_target = max(30, min(K_target, n // 2))

    # attempt construction until time runs out or we succeed
    while time.monotonic() < deadline:
        # concentration counts for each possible sum i+j
        sum_counts = np.zeros(2 * n - 1, dtype=np.int16)

        positions = []                     # selected indices
        attempts = 0
        # generous limit to avoid endless loops
        max_attempts = K_target * 25 + 500

        while len(positions) < K_target and attempts < max_attempts and time.monotonic() < deadline:
            cand = int(rng.integers(0, n))

            # check self‑pair capacity
            if sum_counts[2 * cand] + 1 > C:
                attempts += 1
                continue

            # check cross‑pair capacities with already chosen positions
            if positions:
                existing = np.asarray(positions, dtype=np.int32)
                s = cand + existing                     # unordered sums
                if np.any(sum_counts[s] + 2 > C):
                    attempts += 1
                    continue

            # accept this index
            positions.append(cand)
            # update self contribution
            sum_counts[2 * cand] += 1
            # update cross contributions (ordered pairs count twice)

                sum_counts[s] += 2
