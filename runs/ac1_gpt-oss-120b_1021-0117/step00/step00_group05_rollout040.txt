def propose_candidate(...):
    N = len(height_sequence_1) if defined else 1024
    rng = np.random.default_rng(seed)
    # start with random dirichlet
    a = rng.exponential(scale=1.0, size=N)
    a = a / a.sum()
    best = a.copy()
    best_score = compute_score(best)
    # parameters
    start_eta = 0.5  # maybe large
    min_eta = 1e-4
    decay = 0.9995 per iteration

    deadline as before.

    while time < deadline:
        # compute convolution and max offset
        fft = np.fft.rfft(a, n=2*N-1)
        conv = np.fft.irfft(fft*fft, n=2*N-1)
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]
        # compute subgradient
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (N - 1))
        i_end = min(N-1, max_idx)
        idxs = np.arange(i_start, i_end+1)
        partner = max_idx - idxs
        # For each i, contribution is a[partner] (times 2 if partner != i)
        # The subgradient of max in direction of p_i is 2*a[partner]
        # So assign g[i] = 2*a[partner]
        g[idxs] = 2 * a[partner]
        # For symmetric paired indices: both i and partner will have contribution, but that's okay.

        # step size
        eta = start_eta
        # update
        a_new = a - eta * g
        # ensure nonnegativity
        a_new = np.clip(a_new, 0, None)
        # renorm sum
        sum_new = a_new.sum()
        if sum_new == 0:
            # reinitialize
            a_new = rng.exponential(scale=1.0, size=N)
            sum_new = a_new.sum()
        a_new = a_new / sum_new
        # compute new score
        score_new = compute_score(a_new)
        # accept if improvement
        if score_new < best_score:
            best = a_new
            best_score = score_new
        # optionally accept if score_new < current_score (not just best)
        if score_new <= compute_score(a):
            a = a_new
        # schedule step size
        start_eta *= decay
        if start_eta < min_eta:
            start_eta = min_eta

        # occasional random restart
        if rng.random() < 0.001:
            a = rng.exponential(scale=1.0, size=N)
            a = a / a.sum()

    return best.tolist()

import time
import math
import numpy as np


def _conv_max(x: np.ndarray):
    """Return max value of the linear convolution x*x (no scaling)."""
    n = x.shape[0]
    # length of linear convolution is 2*n-1
    fft = np.fft.rfft(x, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    # eliminate tiny negative artefacts
    conv = np.maximum(conv, 0.0)
    return float(conv.max()), conv


def _subgradient(x: np.ndarray, k_max: int):
    """Sub‑gradient of max_k (x*x)_k at the active index k_max."""
    n = x.shape[0]
    g = np.zeros_like(x)
    i0 = max(0, k_max - (n - 1))
    i1 = min(n - 1, k_max)
    idx = np.arange(i0, i1 + 1)
    partner = k_max - idx
    # derivative of sum_i x_i x_{k-i} w.r.t. x_i is 2*x_{k-i}
    g[idx] = 2.0 * x[partner]
    return g
