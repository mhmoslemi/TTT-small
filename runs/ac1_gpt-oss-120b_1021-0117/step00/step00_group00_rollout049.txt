def conv_fft(a, m):
    a_pad = np.zeros(m, dtype=np.float64)
    a_pad[:len(a)] = a

    conv = np.fft.irfft(A * A, n=m)

def _full_convolution(v: np.ndarray) -> np.ndarray:
    """Full convolution v*v using a power‑of‑two FFT (O(n log n))."""
    n = v.shape[0]

    buf = np.zeros(m, dtype=np.float64)
    buf[:n] = v

    conv = np.fft.irfft(F * F, n=m)
    return conv[: 2 * n - 1]                     # length 2n‑1, same as np.convolve
