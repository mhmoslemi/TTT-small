    # main loop
    n = a.size

        b = np.convolve(a, a)
        k0 = int(np.argmax(b))
        max_b = b[k0]

        contrib[mask] = a[i[mask]] * a[j[mask]]
        # select donor index i1 weighted by contrib (higher donor)
        total_contrib = contrib.sum()
        if total_contrib > 0 and rng.random() < 0.9:
            # weighted random

            probs = contrib / total_contrib
            i1 = rng.choice(n, p=probs)
        else:

            # choose random block width
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            candidate = a.copy()
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            candidate[start:stop] *= factor
            if rng.random() < 0.35:
                candidate = 0.5 * (candidate + candidate[::-1])
            candidate = np.clip(candidate, 0.0, 1000.0)
            score = float(evaluate_sequence(candidate.tolist()))
            if np.isfinite(score) and score < best_score:
                best = candidate
                best_score = score
                a = candidate
            continue

        if rng.random() < 0.5:

        it += 1

        # ----- main move: shift mass from high contribution to low risk -----
        b = conv_self(a)
        k0 = int(np.argmax(b))

        idx = np.arange(n)
        j = k0 - idx
        mask = (j >= 0) & (j < n)

        # contribution of each entry to the current maximal convolution bin
        contrib = np.zeros(n, dtype=np.float64)
        contrib[mask] = a[idx[mask]] * a[j[mask]]

        # donor: pick an index proportional to its contribution
        total_c = contrib.sum()
        if total_c > 0 and rng.random() < 0.85:
            probs = contrib / total_c
            i1 = rng.choice(n, p=probs)
        else:
            i1 = -1

        # occasional block‑wise perturbation (fallback for exploration)
        if i1 == -1:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            cand = a.copy()
            cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
            if rng.random() < 0.35:
                cand = 0.5 * (cand + cand[::-1])
            cand = np.clip(cand, 0.0, 1000.0)
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best = cand.copy()
                best_score = sc
                a = cand
                no_improve = 0
                continue
            # if not taken, go to next iteration
            continue

        # risk associated with increasing a particular index: product with its partner
        risk = np.zeros(n, dtype=np.float64)
        risk[mask] = a[j[mask]]

        # receiver: prefer indices with small risk
        max_risk = risk.max()
        low_w = (max_risk - risk) + 1e-12
        if low_w.sum() > 0 and rng.random() < 0.9:
            probs2 = low_w / low_w.sum()
            i2 = rng.choice(n, p=probs2)
        else:
            i2 = rng.integers(0, n)

        if i2 == i1:
            i2 = (i1 + rng.integers(1, n)) % n

        # transfer a small fraction of mass
        if a[i1] <= 0:
            continue
        delta = a[i1] * rng.uniform(0.02, 0.12)   # 2‑12 % of donor
        delta = min(delta, a[i1])
        if a[i2] + delta > 1000.0:
            delta = 1000.0 - a[i2]
        if delta <= 0:
            continue

        a[i1] -= delta
        a[i2] += delta

        # evaluate new candidate
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = a.copy()
            best_score = sc
            no_improve = 0
        else:
            # mostly revert to keep a near‑optimal region
            if rng.random() < 0.8:
                a[i1] += delta
                a[i2] -= delta
            no_improve += 1

        # ----- occasional extra operators -----
        if it % 37 == 0 and rng.random() < 0.2:          # mild symmetrisation
            a = 0.5 * (a + a[::-1])
            a = np.clip(a, 0.0, 1000.0)

        if it % 97 == 0 and rng.random() < 0.03:          # fresh random shape
            rnd = rng.exponential(scale=1.0, size=n)
            rnd.sort()
            cand = np.clip(rnd[::-1], 0.0, 1000.0)        # decreasing exponential
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best = cand.copy()
                best_score = sc
                a = cand.copy()
                no_improve = 0

        # restart if stagnated for a while
        if no_improve > 2500:
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, 0.04, size=n))
            a = np.clip(a * jitter, 0.0, 1000.0)
            no_improve = 0

    return [float(v) for v in best]
```
