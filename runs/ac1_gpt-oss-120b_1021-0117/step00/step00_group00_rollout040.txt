# ------------------------------------------------------------
# Helper: evaluate a sequence using the authoritative evaluator
# ------------------------------------------------------------
def _score(seq):
    """Return the AC1 score for a list/array of coefficients."""
    # The evaluator works with Python lists.
    return float(evaluate_sequence(seq.tolist()))


# ------------------------------------------------------------
# Core: geometric candidate generator
# ------------------------------------------------------------
def _geom_candidate(n, a):
    """Return a decreasing geometric sequence a_i = (1-a)^i for length n."""
    r = 1.0 - a
    # use np.arange for stable exponentiation
    idx = np.arange(n, dtype=np.float64)
    seq = np.power(r, idx)
    # clip to the evaluator limits
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq

        np.linspace(0.00055, 0.00100, 10, endpoint=True),
    ])

    for a in a_candidates:
        if time.monotonic() >= deadline:
            break
        seq = _geom_candidate(n, a)
        sc = _score(seq)
        if sc < best_score:
            best_seq = seq
            best_score = sc

    # -----------------------------------------------------------------
    # Small local refinements around the best geometric decay.
    # Randomly adjust a single contiguous block – the update is cheap
    # because the whole vector is recomputed from the updated decay.
    # -----------------------------------------------------------------
    # Use a modest number of refinement steps; each step evaluates the
    # exact score via the authoritative evaluator.
    refinements = 0
    while time.monotonic() < deadline and refinements < 30:
        # propose a nearby decay parameter (log‑normal perturbation)
        a0 = 1.0 - best_seq[0]  # this is the current decay a
        a_new = a0 * np.exp(rng.normal(0.0, 0.12))
        # Keep the decay within a sensible range
        a_new = float(np.clip(a_new, 1e-6, 0.01))
        seq = _geom_candidate(n, a_new)
        sc = _score(seq)
        if sc < best_score:
            best_seq = seq
            best_score = sc
        refinements += 1

    # -----------------------------------------------------------------
    # Return the best candidate found as a plain Python list.
    # -----------------------------------------------------------------
    return [float(x) for x in best_seq]
