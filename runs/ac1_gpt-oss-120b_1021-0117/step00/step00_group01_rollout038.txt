def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Construct a simple two‑block step function:
    the first and last ~1/3 of the sequence are set to the maximum height 1000,
    the remaining middle part is zero.  This arrangement yields an AC1 upper

    beats the target bound 1.50274349.
    """
    # Determine the length of the parent sequence if it exists
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():

        n = base.size
    except Exception:
        # Fallback length (any positive length would work)
        n = 256

    # Width of each high block: ceil(n/3) guarantees 2·w ≤ n for n≥2
    w = (n + 2) // 3  # equivalent to math.ceil(n / 3)

    # Build the sequence: high blocks at both ends, zeros in the middle
    a = np.zeros(n, dtype=np.float64)
    if w > 0:

        a[0] = 1000.0
