import numpy as np
import time

def _score(seq: np.ndarray) -> float:
    """Fast internal version of evaluate_sequence (same formula)."""
    s = float(np.sum(seq))
    if s < 0.01:
        return float("inf")
    conv = np.convolve(seq, seq)
    maxb = float(np.max(conv))
    n = seq.size
    return 2.0 * n * maxb / (s * s)


def _initial_candidate(rng: np.random.Generator, n: int) -> np.ndarray:
    """Create a diverse start: interleave high/low values."""
    vals = rng.uniform(0.0, 1.0, size=n)
    vals.sort()
    vals = vals[::-1]                 # descending
    seq = np.empty_like(vals)
    half = (n + 1) // 2
    seq[0::2] = vals[:half]          # biggest, third‑biggest, …
    seq[1::2] = vals[half:][::-1]    # smallest, …
    return np.clip(seq, 0.0, 1000.0)

    rng = np.random.default_rng(seed)

    n = best.size
    # ---- optimisation loop -----------------------------------------
    while time.monotonic() < deadline:
        move = rng.random()
        # ----- pair‑transfer move (tries to flatten the peak) -----
        if move < 0.55:
            conv = np.convolve(best, best)
            s_max = int(conv.argmax())
            lo = max(0, s_max - n + 1)

            idx = np.arange(lo, hi + 1, dtype=np.int64)
            partner = s_max - idx
            prods = best[idx] * best[partner]
            i_max = idx[prods.argmax()]
            i_min = idx[prods.argmin()]

            if i_max == i_min:

            # move a fraction of mass from the “heavy” index to the “light” one
            frac = 0.08 + 0.24 * rng.random()          # 0.08 … 0.32
