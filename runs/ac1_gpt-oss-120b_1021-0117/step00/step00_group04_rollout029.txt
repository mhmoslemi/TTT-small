import time
import numpy as np

def generate_power_law_sequence(p: float, delta: float, n: int) -> np.ndarray:
    """Return a non‑negative sequence a_i = 1000 * (delta/(i+delta))**p."""
    i = np.arange(n, dtype=np.float64)
    seq = 1000.0 * (delta / (i + delta)) ** p
    return np.clip(seq, 0.0, 1000.0)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Base (fallback) sequence – the provided parent or a simple uniform one

            raise ValueError
    except Exception:
        parent = np.full(256, 1.0, dtype=np.float64)

    n = int(parent.shape[0])
    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------
    # Parameter grids – coarse then a refined local search
    # ------------------------------------------------------------------
    p_vals = np.linspace(0.45, 0.85, 9)          # exponents around 0.5‑0.8
    delta_vals = np.geomspace(0.5, 2000.0, 12)   # offsets from very sharp to very flat

    best_p = None
    best_delta = None

    # --------------------------------------------------------------

    for p in p_vals:
