import numpy as np, time

def full_conv(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    B = A * A
    conv = np.fft.irfft(B, n=L)[:2*n-1]
    return conv

def compute_score(a, L_cache=None):
    n = a.shape[0]
    # Use conv via fft
    conv = full_conv(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

        a = np.ones(n, dtype=np.float64)

import numpy as np
import time

def _full_conv(a, L):
    """Linear convolution a*a using FFT (length 2*len(a)-1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)
    return conv[: 2 * a.size - 1]

def _score_from_conv(conv, a_sum, n):
    """Compute the AC1 upper‑bound score from convolution result."""
    max_b = float(conv.max())
    if a_sum < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (a_sum * a_sum))

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # Length of the sequence – use the provided parent if possible
    parent = globals().get('height_sequence_1', None)
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        n = parent.shape[0]
    elif isinstance(parent, list) and len(parent) > 0:
        n = len(parent)
    else:
        n = 256                     # fallback length

    # Initialise with a (slightly perturbed) uniform vector.
    a = np.ones(n, dtype=np.float64)
    a *= np.exp(rng.normal(0.0, 0.04, size=n))   # break symmetry a little
    np.clip(a, 0.0, 1000.0, out=a)

    # Pre‑compute FFT length (next power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    # Evaluate the initial candidate
    conv = _full_conv(a, L)
    best = a.copy()
    best_score = _score_from_conv(conv, a.sum(), n)

    # Main optimisation loop – gradient‑like steps that reduce the
    # contribution to the current maximal convolution entry.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    gamma = 0.30                     # step size factor (0 < gamma < 1)

    while time.monotonic() < deadline:
        # current convolution and the index of its maximum
        conv = _full_conv(a, L)
        k_max = int(np.argmax(conv))

        # contributions of each a_i to the maximal entry
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[j[mask]]

        if np.all(contrib == 0):
            break

        # Scale each component down proportionally to its contribution
        rel = contrib / (contrib.max() + 1e-15)          # in [0,1]
        a_new = a * (1.0 - gamma * rel)

        # keep within bounds and renormalise the total sum
        np.clip(a_new, 0.0, 1000.0, out=a_new)
        s_old = a.sum()
        s_new = a_new.sum()
        if s_new > 0:
            a_new *= (s_old / s_new)

        # evaluate
        conv_new = _full_conv(a_new, L)
        score_new = _score_from_conv(conv_new, a_new.sum(), n)

        # accept if we improved
        if score_new < best_score:
            best = a_new.copy()
            best_score = score_new
            a = a_new
        else:
            # occasional random shake‑up to escape plateaus
            if rng.random() < 0.02:
                a = a * np.exp(rng.normal(0.0, 0.06, size=n))
                np.clip(a, 0.0, 1000.0, out=a)

    # Return the best sequence found (as plain Python floats)
    return [float(x) for x in best]
