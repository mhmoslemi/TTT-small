import numpy as np

def make_block_perturbation(base, rng):
    # O(n) memory: copy base (array) to candidate.
    candidate = base.copy()
    n = candidate.size
    # Choose block width between 1% and 20% maybe, ensure at least 1.
    min_width = max(1, n // 200)  # 0.5%?
    max_width = max(2, n // 12)   # ~8%
    width = int(rng.integers(min_width, max_width + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    # Apply multiplicative Gaussian factor
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= factor
    # Occasionally mirror (symmetrize) (as in previous code)
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    # Clip to bounds
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate
