    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 256
    # fallback candidate: uniform ones.
    best = np.ones(n, dtype=np.float64)
    best_score = float(2 * n * np.max(np.convolve(best, best)) / (np.sum(best)**2))
    # But we can just compute using evaluate_sequence maybe.
    while time.monotonic() < deadline:
        # pick density p in [0.2, 0.8]
        p = rng.uniform(0.2, 0.8)
        L = max(1, int(p * n))
        # ensure L not too small or too large for performance
        if L > n:
            L = n
        # random positions unique
        pos = rng.choice(n, size=L, replace=False)
        cand = np.zeros(n, dtype=np.float64)
        cand[pos] = 1.0
        # compute score
        conv = np.convolve(cand, cand)
        max_b = conv.max()
        sum_a = cand.sum()
        if sum_a < 0.01:
            continue
        score = 2.0 * n * max_b / (sum_a * sum_a)
        # if better
        if score < best_score:
            best = cand
            best_score = float(score)
            # optionally break if near optimum
            if best_score <= 1.1:
                break
    # convert best to list, ensure values within bounds (they are)
    return [float(x) for x in best]

import time
import numpy as np

def _random_binary_candidate(n: int, p: float, rng: np.random.Generator) -> np.ndarray:
    """Return a length‑n array of 0.0/1.0 with density ~p."""
    return (rng.random(n) < p).astype(np.float64)


def _candidate_score(a: np.ndarray) -> float:
    """Exact AC1 upper‑bound score for a non‑negative sequence a."""
    n = a.shape[0]
    # np.convolve implements the same operation as the evaluator.
    conv = np.convolve(a, a)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (s * s))


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """
    Search for a step‑function that minimises the AC1 bound.
    The routine works in~O(n) memory and respects the time budget.
    """
    rng = np.random.default_rng(seed)
    # Leave a small safety margin for the harness.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Length of the parent vector (provided by the harness).
    try:
        n = int(len(height_sequence_1))
        parent = np.clip(
            np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0
        )
    except Exception:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    best = parent.copy()
    best_score = float(_candidate_score(best))

    # Main random‑binary search.
    while time.monotonic() < deadline:
        # Random density in a sensible interval.
        p = rng.uniform(0.30, 0.70)
        cand = _random_binary_candidate(n, p, rng)

        # Guard against an all‑zero candidate (unlikely but possible).
        if cand.sum() < 1.0:
            cand[rng.integers(0, n)] = 1.0

        score = _candidate_score(cand)

        if score < best_score:
            best = cand
            best_score = score
            # Very good solution – stop early.
            if best_score <= 1.10:
                break

    # Return a plain Python list as required.
    return [float(x) for x in best]
