def compute_score(a):
    n = a.size
    sum_a = a.sum()
    # compute conv via FFT
    size = 1 << ((2*n-1).bit_length())
    Fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(Fa * Fa, size)[:2*n-1]
    max_b = conv.max()
    return 2 * n * max_b / (sum_a**2)

import numpy as np
import time

# fast convolution using FFT (real inputs)
def _max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Return max of the linear convolution a*a using an FFT of length fft_len."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return float(conv.max())


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a non‑negative step function that gives a low AC1 bound.

    The search starts from a power‑law shape (a_i ∝ (i+1)^{-0.5}) which already
    yields a score ≈ π/2 ≈ 1.57.  Random “mass transfers’’ and small block
    perturbations are applied while the L¹‑norm is kept equal to 1
    (the objective is scale‑invariant).  The best sequence found before the
    deadline is returned.
    """
    rng = np.random.default_rng(seed)
    # length of the problem (use the provided parent if present)
    try:
        n = int(np.asarray(height_sequence_1, dtype=np.intp).size)
    except Exception:  # fallback length if the parent is missing
        n = 5000

    # initialise a power‑law: a_i = (i+1)^{-0.5}, then normalise to sum 1
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** -0.5
    a /= a.sum()                     # L1‑norm = 1  (scale does not affect the score)

    # FFT length: next power of two ≥ 2*n‑1
    fft_len = 1 << ((2 * n - 1).bit_length())

    # current best
    best = a.copy()
    best_score = 2.0 * n * _max_convolution(best, fft_len)

    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # parameters for the stochastic moves
    transfer_frac = 0.02          # fraction of the source value moved per step
    block_sigma = 0.05            # log‑normal perturbation width for block updates
    block_prob = 0.2              # how often to perform a block update

    while time.monotonic() < deadline:
        # ---------- mass‑transfer move ----------
        if rng.random() >= block_prob:
            src = int(rng.integers(0, n))
            dst = int(rng.integers(0, n))
            if src == dst or a[src] <= 0.0:
                continue
            delta = a[src] * transfer_frac
            a[src] -= delta
            a[dst] += delta

            # evaluate
            score = 2.0 * n * _max_convolution(a, fft_len)

            if score < best_score:
                best_score = score
                best = a.copy()
            else:
                # revert the change
                a[src] += delta
                a[dst] -= delta
            continue

        # ---------- block‑scale move ----------
        width = int(rng.integers(1, max(2, n // 50)))        # O(n/50) max width
        start = int(rng.integers(0, n - width + 1))
        factor = float(np.exp(rng.normal(0.0, block_sigma)))
        cand = a.copy()
        cand[start : start + width] *= factor
        # keep non‑negative, renormalise to keep L1‑norm = 1
        cand = np.clip(cand, 0.0, None)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum

        score = 2.0 * n * _max_convolution(cand, fft_len)

        if score < best_score:
            best_score = score
            best = cand.copy()
            a = cand                     # keep the improvement for future moves

    # ensure a valid return (non‑negative, finite, sum>0.01)
    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:
        best[:] = 1.0 / best.size

    return [float(x) for x in best]
