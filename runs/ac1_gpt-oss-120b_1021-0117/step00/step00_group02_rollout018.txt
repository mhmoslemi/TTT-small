    start = time.monotonic()

    best_seq = None

    # initial candidates: uniform of length n (maybe 256)

    while time.monotonic() < deadline:

        # generate Dirichlet

        a = a / a.sum()
        # scale not needed

        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = seq.copy()
            best_score = score
            # maybe print progress (can't)
        # optional small local search from best: perturb a few entries
        # e.g., pick best and try move mass
        if best_seq is not None and time.monotonic() < deadline:
            # subtle local modifications
            cand = best_seq.copy()
            # random block scaling
            width = int(rng.integers(1, max(2, n//10)))
            start_idx = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0, 0.1)))
            cand[start_idx:start_idx+width] *= factor
            # re-normalize to sum 1
            cand = np.clip(cand, 0.0, 1000.0)
            total = cand.sum()
            if total > 0:
                cand = cand / total
            else:
                continue
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best_seq = cand.copy()
                best_score = score

    # final fix: ensure nonnegative and sum>0.01 (should be)

def convolve_full(x, y):
    m = int(1 << ((2*len(x)-1).bit_length()))  # next power of two
    fx = np.fft.rfft(x, m)
    fy = np.fft.rfft(y, m)
    conv = np.fft.irfft(fx * fy, m)[:len(x)+len(y)-1]
    return conv

import time
import numpy as np

# the evaluator and optional parent array are provided by the harness
# evaluate_sequence(sequence) -> float
# height_sequence_1 (optional) -> np.ndarray


def _score(seq):
    """Convenient wrapper for the authoritative evaluator."""
    return float(evaluate_sequence(seq.tolist()))


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a non‑negative step function with a small AC1 upper bound.

    The method combines random Dirichlet draws of moderate size with cheap
    local perturbations.  It never allocates quadratic memory and respects the
    supplied time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 2.0)   # safety margin

    # ---------- fallback construction (uniform) ----------
    n_fallback = 256
    fallback = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
    best = fallback.copy()
    best_score = _score(best)

    # ---------- try the provided parent, if any ----------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            parent_score = _score(parent)
            if parent_score < best_score:
                best = parent.copy()
                best_score = parent_score
    except NameError:
        pass  # no parent supplied

    # ---------- main search loop ----------
    while time.monotonic() < deadline:
        # --- draw a fresh random candidate ---------------------------------
        # length is chosen modestly so that many trials fit in the budget
        n = int(rng.integers(5, 300))                # 5 ≤ n < 300
        alpha = rng.uniform(0.2, 3.0)                # Dirichlet concentration
        a = rng.gamma(alpha, 1.0, size=n)            # gamma → Dirichlet
        if a.sum() == 0.0:
            continue
        a /= a.sum()                                 # normalise to sum = 1

        # occasional mirroring (helps break symmetry)
        if rng.random() < 0.3:
            a = 0.5 * (a + a[::-1])

        cand_score = _score(a)
        if np.isfinite(cand_score) and cand_score < best_score:
            best = a.copy()
            best_score = cand_score

        # --- cheap local tweak of the current best -------------------------
        if rng.random() < 0.07:                      # only occasionally
            cand = best.copy()
            # pick a contiguous block and scale it a little
            width = int(rng.integers(1, max(2, cand.size // 12)))
            start_idx = int(rng.integers(0, cand.size - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            cand[start_idx:start_idx + width] *= factor
            # optional symmetrisation
            if rng.random() < 0.25:
                cand = 0.5 * (cand + cand[::-1])
            # re‑normalise (scale does not affect the objective)
            tot = cand.sum()
            if tot > 0:
                cand /= tot
                cand_score = _score(cand)
                if np.isfinite(cand_score) and cand_score < best_score:
                    best = cand.copy()
                    best_score = cand_score

    # ---------- final sanity checks ----------
    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:
        best = best * (0.01 / best.sum())

    return [float(x) for x in best]
