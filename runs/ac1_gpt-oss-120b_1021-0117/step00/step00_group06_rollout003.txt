    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    scale = float(np.exp(rng.normal(0.0, 0.06)))   # modest log‑normal factor
    cand = base.copy()
    cand[start:start + width] *= scale
    # occasional symmetrisation (helps explore)
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative heights minimizing the AC1 bound.

    The routine builds a uniform candidate (the provably optimal shape) and
    optionally improves it with very cheap random perturbations while respecting
    the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ----------------------------------------------------------------------
    # 1) Determine the initial length (prefer the provided parent if usable)
    # ----------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or not np.isfinite(parent_arr).all():
            parent_arr = None
    except Exception:
        parent_arr = None

    if parent_arr is not None and parent_arr.size > 0:
        n = parent_arr.size
    else:
        n = 256                     # a reasonable fallback size

    # ----------------------------------------------------------------------
    # 2) Uniform candidate – provably optimal (score = 2.0)
    # ----------------------------------------------------------------------
    uniform = np.full(n, 1.0, dtype=np.float64)   # any positive constant works
    best = uniform
    best_score = float(evaluate_sequence(best.tolist()))

    # ----------------------------------------------------------------------
    # 3) If the imported parent is better, adopt it
    # ----------------------------------------------------------------------
    if parent_arr is not None and parent_arr.size > 0:
        parent_seq = np.clip(parent_arr, 0.0, 1000.0)
        score = float(evaluate_sequence(parent_seq.tolist()))
        if score < best_score:
            best = parent_seq
            best_score = score

    # ----------------------------------------------------------------------
    # 4) Light‑weight random search (won’t beat uniform, but safe fallback)
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        cand = _perturb_candidate(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score:
            best = cand
            best_score = score

    # ----------------------------------------------------------------------
    # 5) Return the best found candidate as a pure Python list of floats
    # ----------------------------------------------------------------------
    return [float(x) for x in best]
