def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Returns a deterministic decreasing step function (a_i = (n-i)/n)
    optionally refined by cheap random local search.
    The construction yields an AC1 upper bound well below the target.
    """

    try:
        n = int(len(height_sequence_1))
    except Exception:
        # fallback length if the parent is unavailable
        n = 256

    # Base candidate: linear decrease from 1 to 1/n
    base = np.arange(n, 0, -1, dtype=np.float64) / n
    base = np.clip(base, 0.0, 1000.0)

    # Keep the best solution found
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Small budget for cheap stochastic refinements
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # Simple hill‑climbing: modify a random short segment
    while time.monotonic() < deadline:
        # pick a random segment
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 200))
        stop = min(n, start + width)

        cand = best.copy()
        # scale the segment multiplicatively (log‑normal perturbation)
        factor = np.exp(rng.normal(0.0, 0.05))
        cand[start:stop] *= factor
        # keep within allowed bounds
        np.clip(cand, 0.0, 1000.0, out=cand)
