def conv_max(seq):
    n=len(seq)
    L = 1 << ((2*n-1).bit_length())
    fft_seq = np.fft.rfft(np.pad(seq, (0, L-n), 'constant'))
    # product
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)
    conv = conv[:2*n-1]
    return conv.max()
```

import numpy as np
import time

def _conv_max(seq: np.ndarray) -> float:
    """Return the maximum value of the linear convolution seq*seq."""
    # Using FFT for speed (zero‑padding to next power of two)
    n = seq.size
    L = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)
    return float(conv[: 2 * n - 1].max())


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step function that lowers the AC1 upper bound.
    The routine tries a variety of random families (Dirichlet, exponential,
    power‑law, sparse support) and keeps the best candidate found before the
    time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # length of the problem – use the provided parent if present
    try:
        n = int(len(height_sequence_1))
        base_len = n
    except Exception:
        # fallback length (reasonable size, still fast to evaluate)
        n = 2000
        base_len = n

    # uniform vector is a guaranteed feasible point (score = 2.0)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = float(_conv_max(best_seq) * 2 * n / (best_seq.sum() ** 2))

    # helper factories -------------------------------------------------
    def _dirichlet(alpha):
        w = rng.gamma(alpha, 1.0, size=n)
        if w.sum() == 0:
            w[:] = 1.0 / n
        else:
            w /= w.sum()
        return w

    def _expo(rate):
        x = np.arange(n, dtype=np.float64)
        w = np.exp(-rate * x)
        w /= w.sum()
        return w

    def _power(pwr):
        x = np.arange(n, dtype=np.float64)
        w = (n - x) ** pwr
        w /= w.sum()
        return w

    def _sparse():
        # size roughly sqrt(2n); uniform weight on chosen support
        m = rng.integers(int(np.sqrt(base_len)), int(np.sqrt(2 * base_len)) + 1)
        idx = rng.choice(n, size=m, replace=False)
        w = np.zeros(n, dtype=np.float64)
        w[idx] = 1.0 / m
        return w

    # main search loop -------------------------------------------------
    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.20:                     # Dirichlet family
            cand = _dirichlet(rng.uniform(0.3, 3.0))
        elif r < 0.40:                   # exponential decay
            cand = _expo(rng.uniform(0.0, 0.006))
        elif r < 0.60:                   # power‑law (decreasing)
            cand = _power(rng.uniform(0.5, 4.0))
        elif r < 0.80:                   # sparse support
            cand = _sparse()
        else:                            # perturb the current best
            cand = best_seq.copy()
            # modify a few coordinates (relative changes up to ±45 %)
            for _ in range(rng.integers(1, 4)):
                i = rng.integers(0, n)
                delta = (rng.random() - 0.5) * 0.9 * cand[i]
                cand[i] = max(cand[i] + delta, 0.0)
            s = cand.sum()
            if s > 0:
                cand /= s
            else:
                cand = uniform.copy()

        # compute the score (scale‑invariant, but we keep the exact form)
        max_b = _conv_max(cand)
        score = 2.0 * n * max_b / (cand.sum() ** 2)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = cand

    # final safety – ensure we return a non‑empty, finite list
    if best_seq.ndim != 1 or best_seq.size == 0 or not np.isfinite(best_seq).all():
        best_seq = uniform
    return [float(x) for x in best_seq]
