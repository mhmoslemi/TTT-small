    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)

    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))

def generate_sidon_set(N):
    used = np.zeros(2*N, dtype=bool) # covers sums 0..2N-2
    positions = []
    for i in range(N):
        # check if any sum i+pos already used
        conflict = False
        for pos in positions:
            if used[i+pos]:
                conflict = True
                break
        if not conflict and not used[i+i]:
            # add
            positions.append(i)
            for pos in positions: # after adding, we need update sums with new pos and all existing including itself.
                used[i+pos] = True
                used[pos+i] = True
            used[i+i] = True
    return positions

import numpy as np
import time

def _build_sidon_sequence(n: int, value: float = 1000.0):
    """Greedy construction of a Sidon set (all pairwise sums distinct)."""
    used = np.zeros(2 * n, dtype=bool)          # marks occupied sums i+j
    positions = []                               # indices where we put `value`
    for i in range(n):
        if used[i + i]:                          # i+i already taken -> skip
            continue
        conflict = False
        for p in positions:                      # check i+p against used sums
            if used[i + p]:
                conflict = True
                break
        if conflict:
            continue
        # accept i: mark all new sums i+p (p from existing positions) and i+i
        for p in positions:
            used[i + p] = True
        used[i + i] = True
        positions.append(i)
    seq = np.zeros(n, dtype=np.float64)
    if positions:
        seq[positions] = value
    return seq
