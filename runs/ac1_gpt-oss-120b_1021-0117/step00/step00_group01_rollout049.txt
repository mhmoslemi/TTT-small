import time
import math
import numpy as np

def _sidon_set(n, m, rng, max_tries=5000):
    """Return a list of *m* indices in ``[0, n)`` whose pairwise sums are all
    distinct (a Sidon set).  If a perfect set cannot be found within
    ``max_tries`` attempts an incomplete set is returned.
    """
    positions = []
    sums = set()
    tries = 0
    while len(positions) < m and tries < max_tries:
        cand = int(rng.integers(0, n))
        if cand in positions:
            tries += 1
            continue
        # check new sums
        conflict = False
        cand_self = cand + cand
        if cand_self in sums:
            conflict = True
        else:
            for p in positions:
                s = cand + p
                if s in sums:
                    conflict = True
                    break
        if conflict:
            tries += 1
            continue
        # accept
        positions.append(cand)
        # add new sums (including self‑sum)
        sums.add(cand_self)
        for p in positions:
            sums.add(cand + p)
    return np.array(positions, dtype=np.int64)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that tries to
    minimise the AC1 upper bound evaluated by ``evaluate_sequence``.
    The routine builds a two‑level sequence (large spikes + a tiny
    background) and searches over a small range of parameters using a
    time‑bounded randomised algorithm.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Base sequence (fallback) – use the provided parent if it looks sane
    # ------------------------------------------------------------------
    try:
        parent_arr = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                             0.0, 1000.0)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise ValueError
        n = parent_arr.size
    except Exception:
        n = 256
        parent_arr = np.ones(n, dtype=np.float64)

    best_seq = parent_arr.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # --------------------------------------------------------------
    # Parameters for the two‑level construction
    # --------------------------------------------------------------
    sqrt_n = int(math.isqrt(n))
    # try a few sizes around sqrt(n) – they all respect the Sidon bound
    m_range = list(range(max(2, sqrt_n - 5), min(n, sqrt_n + 6)))
    factor_list = [0.5, 0.8, 1.0, 1.2, 1.5, 2.0]   # controls the tiny background level

    # --------------------------------------------------------------
    # Main time‑bounded search loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # pick a size for the large‑value set
        m = int(rng.choice(m_range))
        # generate a collision‑free Sidon set of that size
        large_idx = _sidon_set(n, m, rng)
        # if the Sidon set is too small, fall back to a random subset
        if large_idx.size < m:
            large_idx = rng.choice(n, size=m, replace=False)

        # evaluate several background magnitudes
        for fac in factor_list:
            # value of the background entries – large entries are fixed at 1.0
            e = fac / m                     #  e ≈ (factor)·1/m
            seq = np.full(n, e, dtype=np.float64)
            seq[large_idx] = 1.0
            # evaluate
            try:
                score = float(evaluate_sequence(seq.tolist()))
            except Exception:
                continue
            if np.isfinite(score) and score < best_score:
                best_seq = seq.copy()
                best_score = score

        # Occasionally try a cheap local perturbation of the current best
        # (block scaling) – it can discover tiny improvements for free.
        if best_score > 1.6 and time.monotonic() < deadline - 0.3:

                if np.isfinite(sc) and sc < best_score:
                    best_seq = cand
                    best_score = sc
            except Exception:
                pass

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
