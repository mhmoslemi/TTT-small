def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    if 'height_sequence_1' in globals():
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        parent_arr = np.ones(256, dtype=np.float64)
    parent_arr = np.clip(parent_arr, 0.0, 1000.0)
    n = parent_arr.size
    best = parent_arr
    best_score = float(evaluate_sequence(best.tolist()))
    # candidate search using exponential shapes
    # define alpha ranges
    alphas = np.concatenate([
        np.geomspace(1e-5, 5e-4, num=100),
        np.linspace(5e-4, 1e-3, num=20)
    ])
    # ensure distinct sorted
    alphas = np.unique(alphas)
    for alpha in alphas:
        if time.monotonic() >= deadline:
            break
        # compute candidate
        log_r = np.log1p(-alpha)
        a = np.exp(log_r * np.arange(n, dtype=np.float64))
        s = a.sum()
        if s == 0:
            continue
        a /= s
        # optional: maybe also try reversed shape but same as original reversed order yields same metric
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a
            best_score = score
    # Additionally try a symmetric exponential shape (peak at centre)
    # shape: a[i] = exp(-beta * |i - mid|)
    for beta in np.geomspace(1e-5, 5e-2, num=60):
        if time.monotonic() >= deadline:
            break
        mid = (n - 1) / 2.0
        a = np.exp(-beta * np.abs(np.arange(n, dtype=np.float64) - mid))
        a /= a.sum()
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a
            best_score = score
    # Additional random local perturbation around best using block perturbation from prior program
    def make_block_perturbation(base):
        n = base.size
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        candidate = base.copy()
        candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
        if rng.random() < 0.35:
            candidate = 0.5 * (candidate + candidate[::-1])
        np.clip(candidate, 0.0, 1000.0, out=candidate)
        return candidate
    # small local search loop
    while time.monotonic() < deadline:
        cand = make_block_perturbation(best)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
    # Return as list
    return [float(x) for x in best]

def make_block_perturbation(base, rng):
    """O(n) non‑negative perturbation of ``base``."""
    n = int(base.size)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))

    if rng.random() < 0.35:
        cand[:] = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
