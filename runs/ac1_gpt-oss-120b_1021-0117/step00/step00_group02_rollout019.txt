    """

    Returns np.inf if the input is invalid.
    """

        parent = None
        has_parent = False

    score = 2.0 * n * max_b / (a.sum() * a.sum())

        best = a.copy()
        best_score = float(score)

    # start iterative refinement from best
    cur = best.copy()
    eps = 0.03  # reduction factor per iteration
    no_improve = 0

import time
import numpy as np

def _conv_max(arr: np.ndarray) -> float:
    """Maximum entry of the linear convolution arr * arr."""
    return np.convolve(arr, arr).max()


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that reduces the AC1 upper bound."""
    rng = np.random.default_rng(seed)
    # reserve a few seconds to guarantee a return before the hard timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ----------------------------------------------------------------------
    # Determine problem size (use the provided parent if it exists)
    # ----------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        have_parent = True
    except Exception:                     # height_sequence_1 may be missing
        n = 1024
        parent = None
        have_parent = False

    best = None
    best_score = float("inf")

    # ----------------------------------------------------------------------
    # 1)  Keep the supplied parent if it is valid and better than the default.
    # ----------------------------------------------------------------------
    if have_parent and parent.size > 0 and np.isfinite(parent).all():
        max_b = _conv_max(parent)
        sum_a = parent.sum()
        if sum_a >= 0.01:
            score = 2.0 * n * max_b / (sum_a * sum_a)
            if np.isfinite(score):
                best = parent.copy()
                best_score = float(score)

    # ----------------------------------------------------------------------
    # 2)  Initialise with the “arcsine” distribution a_i = 1/sqrt(i+1).
    #     It gives a near‑optimal flat additive convolution.
    # ----------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(idx)            # raw weights
    a /= a.sum()                      # normalise to sum = 1
    a = np.clip(a, 0.0, 1000.0)

    max_b = _conv_max(a)
    score = 2.0 * n * max_b / (a.sum() * a.sum())
    if score < best_score:
        best = a.copy()
        best_score = float(score)

    # ----------------------------------------------------------------------
    # 3)  Simple projected sub‑gradient descent.
    #     Reduce the entries that take part in the current maximal
    #     convolution term and redistribute the removed mass uniformly.
    # ----------------------------------------------------------------------
    cur = best.copy()
    eps = 0.03                 # fractional reduction of the active block
    no_improve = 0
    max_no_improve = 200

    while time.monotonic() < deadline:
        # current convolution and its worst index
        conv = np.convolve(cur, cur)
        max_c = conv.max()
        max_idx = int(conv.argmax())

        # index range that contributes to this maximal entry
        lo = max_idx - n + 1
        if lo < 0:
            lo = 0
        hi = max_idx
        if hi >= n:
            hi = n - 1

        # total mass inside the active block
        block_sum = cur[lo:hi + 1].sum()
        if block_sum == 0.0:
            break

        # amount of mass we take away from the block
        removed = eps * block_sum

        # build the next iterate
        nxt = cur.copy()
        nxt[lo:hi + 1] *= (1.0 - eps)          # shrink the active block
        nxt += removed / n                     # redistribute uniformly
        np.clip(nxt, 0.0, 1000.0, out=nxt)    # keep within bounds

        total = nxt.sum()
        if total == 0.0:
            break
        nxt /= total                            # renormalise to keep sum = 1

        # evaluate the new candidate (exactly the same formula as the
        # authoritative evaluator)
        new_max = _conv_max(nxt)
        new_score = 2.0 * n * new_max / (total * total)

            best = nxt.copy()
            best_score = float(new_score)
            cur = nxt
            no_improve = 0
        else:
            cur = nxt
            no_improve += 1

            eps *= 0.5
            no_improve = 0
            if eps < 1e-6:

    # Return the best sequence found as a plain Python list.
    # ----------------------------------------------------------------------
    return [float(v) for v in best]
