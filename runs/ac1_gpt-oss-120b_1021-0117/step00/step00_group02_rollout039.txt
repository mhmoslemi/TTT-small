def propose_candidate(seed=42, budget_s=1100):
    import numpy as np, time, math

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 3.0
    n = len(height_sequence_1) if available else 1024
    # initialize a
    if 'height_sequence_1' in globals():
        init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if init.ndim != 1 or init.size == 0 or not np.isfinite(init).all():
            init = np.ones(256, dtype=np.float64)
        a = init.copy()
    else:
        a = np.ones(256, dtype=np.float64) # fallback
    # maybe random perturbation to break uniform
    a += 0.1 * (rng.random(a.shape) - 0.5)  # small noise
    a = np.clip(a, 0.0, 1000.0)
    # ensure sum > 0
    if a.sum() < 0.01:
        a[0] = 0.01

    # compute initial score
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    S_target = a.sum()  # keep sum; scaling invariance so not needed to fix but we will preserve.

    # precompute FFT size
    n2 = 1
    while n2 < 2 * n - 1:
        n2 <<= 1

    def convolve_full(x):
        # compute linear convolution via FFT
        fx = np.fft.rfft(x, n=n2)
        conv = np.fft.irfft(fx * fx, n=n2)
        # return only first (2*n-1) entries
        return conv[:2*n-1]

    # main loop
    iteration = 0
    while time.monotonic() < deadline:
        # compute convolution
        conv = convolve_full(a)
        # get max value and lag
        k_max = int(np.argmax(conv))
        M = conv[k_max]
        # compute contributions g_i for positions i where partner exists
        g = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)  # inclusive
        # vectorized fill
        i_idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_idx = k_max - i_idx
        g[i_idx] = a[j_idx]

        # gradient
        S = a.sum()
        if S == 0:
            break
        factor = (2.0 * n) / (S ** 3)
        grad = factor * (g * S - 2.0 * M)

        # step size: adapt based on gradient norm
        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0:
            # random perturb
            a = a + 0.05 * (rng.random(a.shape) - 0.5)
            a = np.clip(a, 0.0, 1000.0)
            continue
        # set step etc.
        eta = 0.1 / (1 + iteration * 0.001)  # decaying
        # compute update
        a_new = a - eta * grad
        a_new = np.clip(a_new, 0.0, 1000.0)
        # renormalize to original sum to preserve scale
        sum_new = a_new.sum()
        if sum_new > 0:
            a_new = a_new * (S_target / sum_new)
        else:
            a_new = a

        # evaluate
        score_new = float(evaluate_sequence(a_new.tolist()))
        if np.isfinite(score_new) and score_new < best_score:
            best = a_new.copy()
            best_score = score_new

        a = a_new
        iteration += 1

    return [float(x) for x in best]

import numpy as np, time, math

def _exp_shape(dist, gamma):
    """Return exp(-gamma*dist) vector."""
    return np.exp(-gamma * dist)

def _block_perturb(base, rng):
    """Random multiplicative block perturbation."""
    n = base.size
    w = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    s = int(rng.integers(0, max(1, n - w + 1)))
    e = s + w
    out = base.copy()
    out[s:e] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        out = 0.5 * (out + out[::-1])
    return np.clip(out, 0.0, 1000.0)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----- initial parent (fallback) -----
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ----- distance to nearest edge (for exponential shape) -----
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)

    # ----- search over exponential decay rates -----
    target_ratio = 1.50274349
    gamma_center = 2.0 * math.e * target_ratio / n          # approximate optimum
    gammas = np.linspace(gamma_center * 0.7, gamma_center * 1.3, 40)

    for g in gammas:
        if time.monotonic() >= deadline:
            break
        cand = _exp_shape(dist, g)
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ----- fine‑tune gamma with small random moves -----
    cur_gamma = gamma_center
    sigma = 0.02
    while time.monotonic() < deadline:
        # propose new gamma
        g = cur_gamma * (1.0 + rng.normal(0.0, sigma))
        if g <= 0.0:
            continue
        cand = _exp_shape(dist, g)
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            cur_gamma = g
            sigma *= 0.98
        else:
            sigma *= 0.995
        # occasional block perturbation to escape plateaus
        if rng.random() < 0.07:
            cand2 = _block_perturb(best, rng)
            score2 = float(evaluate_sequence(cand2.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best = cand2
                best_score = score2

    return [float(v) for v in best]
