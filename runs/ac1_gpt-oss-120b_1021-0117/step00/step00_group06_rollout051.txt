    except Exception:
        n = 256  # fallback length

    # Choose an exponent that gives a strong decay.
    # p=3 gives evaluation ≈0.23; higher p improves it further.
    p = 3

    # Build the decreasing power sequence: (n, n-1, ..., 1)^p
    # Use float64 to avoid integer overflow.
    vals = np.arange(n, 0, -1, dtype=np.float64) ** p

    # Scale so that the largest entry equals 1000 (the evaluator clips
    # to this range, and the bound is scale‑invariant).
    scale = 1000.0 / vals[0]
    vals *= scale

    # Clip for safety (max already 1000, min is ≥0).
    vals = np.clip(vals, 0.0, 1000.0)

    # The evaluator also enforces a minimum total sum of 0.01;
    # the construction easily satisfies this.
    # Return a plain Python list of floats as required.
    return [float(x) for x in vals]
