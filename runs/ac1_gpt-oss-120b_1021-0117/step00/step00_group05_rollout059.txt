def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)   # margin
    n = len(height_sequence_1)
    # base parent
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim !=1 or parent.size != n or not np.isfinite(parent).all():
        parent = np.ones(n, dtype=np.float64)
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # Helper to evaluate and update best
    def try_candidate(seq):
        nonlocal best, best_score
        # clip bounds
        seq = np.clip(seq, 0.0, 1000.0)
        # skip if sum < 0.01
        if np.sum(seq) < 0.01:
            return
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best = seq.copy()
            best_score = score

    # First, try geometric families
    # epsilon = c/n; r = 1 - epsilon
    for c in np.linspace(0.5, 3.0, 26):   # 0.5 to 3 step 0.1 approx
        if time.monotonic() >= deadline:
            break
        epsilon = c / n
        r = 1.0 - epsilon
        # Ensure r positive
        if r <= 0.0:
            continue
        seq = np.power(r, np.arange(n, dtype=np.float64))
        try_candidate(seq)

    # Next try quadratic exponent shape: a_i = exp(-c * i/n - gamma * (i/n)^2)
    # Choose grid for c and gamma
    for c in np.linspace(0.5, 3.0, 9):  # coarse
        if time.monotonic() >= deadline:
            break
        for gamma in np.linspace(0.0, 5.0, 11):
            if time.monotonic() >= deadline:
                break
            i = np.arange(n, dtype=np.float64)
            x = i / n
            seq = np.exp(-c * x - gamma * x * x)
            try_candidate(seq)

    # Next try linear decreasing shape: a_i = max(0, a0 - d*i)
    # Choose a0 such that sum close to n (so average 1)
    for d in np.linspace(0.0, 2.0, 21):
        if time.monotonic() >= deadline:
            break
        a0 = 1.0 + d * (n - 1) / 2  # ensure positive at end maybe?
        seq = a0 - d * np.arange(n, dtype=np.float64)
        seq = np.maximum(seq, 0.0)
        try_candidate(seq)

    # Next try "logistic" shape: a_i = 1 / (1 + exp(s*(i - mid)/n))
    mid = (n - 1) / 2.0
    for s in np.linspace(-5, 5, 21):
        if time.monotonic() >= deadline:
            break
        x = (np.arange(n, dtype=np.float64) - mid) / n
        seq = 1.0 / (1.0 + np.exp(s * x))
        # Normalize to have same sum as parent maybe not needed.
        try_candidate(seq)

    # Use random perturbations from best to explore local region
    # Similar to previous code but with some modifications
    while time.monotonic() < deadline:
        candidate = best.copy()
        # Perturb a random block
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        # Multiply block by random factor log-normal
        factor = float(np.exp(rng.normal(0.0, 0.05)))
        candidate[start:stop] *= factor
        # Occasionally reflect
        if rng.random() < 0.3:
            candidate = 0.5 * (candidate + candidate[::-1])
        try_candidate(candidate)

    # return best as list of floats
    return [float(x) for x in best]

import numpy as np

def _maybe_update(seq, best, best_score):
    """Return possibly improved (best, best_score) after evaluating seq."""
    seq = np.clip(seq, 0.0, 1000.0)
    if np.sum(seq) < 0.01:
        return best, best_score
    sc = float(evaluate_sequence(seq.tolist()))
    if np.isfinite(sc) and sc < best_score:
        return seq.copy(), sc
    return best, best_score


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Generate a non‑negative step‑function (list of floats) that
    minimises the AC1 upper bound evaluated by `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # length of the problem
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size > 0 and np.isfinite(base).all():
            n = base.size
        else:
            n = 256
    else:
        n = 256

    # --- 1. linear decreasing sequence (the main theoretical candidate) ---
    if n > 1:
        idx = np.arange(n, dtype=np.float64)
        best = 1000.0 * (1.0 - idx / (n - 1))
    else:
        best = np.full(1, 1000.0, dtype=np.float64)
    best = np.clip(best, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best.tolist()))

    # --- 2. explore simple exponential families -------------------------
    for c in np.linspace(0.5, 5.0, 21):
        if time.monotonic() >= deadline:
            break
        eps = c / n
        r = 1.0 - eps
        if r <= 0.0:
            continue
        seq = 1000.0 * (r ** np.arange(n, dtype=np.float64))
        best, best_score = _maybe_update(seq, best, best_score)

    # --- 3. quadratic‑exponential shapes --------------------------------
    for c in np.linspace(0.2, 4.0, 15):
        if time.monotonic() >= deadline:
            break
        for gamma in np.linspace(0.0, 4.0, 9):
            if time.monotonic() >= deadline:
                break
            x = np.arange(n, dtype=np.float64) / n
            seq = 1000.0 * np.exp(-c * x - gamma * x * x)
            best, best_score = _maybe_update(seq, best, best_score)

    while time.monotonic() < deadline:
