import time
import numpy as np

# Fast convolution‑based score (uses FFT, O(n log n))
def _fast_score(seq, fft_len, n):
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * np.max(conv) / (s * s))


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a non‑negative step‑function that lowers the AC1 bound.

    The method builds an exponential‑type sequence a_i = exp(-u*i/n)
    and refines it with cheap FFT‑based local moves.
    """
    rng = np.random.default_rng(seed)
    # allow a tiny safety margin before the sandbox kills the process
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------ #
    # 1.  Determine length and a reasonable fallback start vector
    # ------------------------------------------------------------------ #
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = int(parent.size)
    idx = np.arange(n, dtype=np.float64)

    # --------------------------------------------------------------- #
    # 2.  Prepare FFT size (next power of two >= 2*n-1)
    # --------------------------------------------------------------- #
    fft_len = 1
    while fft_len < 2 * n - 1:
        fft_len <<= 1

    # --------------------------------------------------------------- #
    # 3.  Scan a small family of exponential shapes a_i = exp(-u*i/n)
    # --------------------------------------------------------------- #
    best_seq = None
    best_score = np.inf

    # explore u in a sensible interval (covers the theoretical optimum ~1)
    for u in np.linspace(0.6, 2.0, 30):
        t = u / n
        seq = np.exp(-t * idx)          # a_i = exp(-t*i)
        sc = _fast_score(seq, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()

    # --------------------------------------------------------------- #
    # 4.  Local refinement using cheap block perturbations
    # --------------------------------------------------------------- #
    # keep a copy that we will mutate
    cur = best_seq.copy()
    cur_score = best_score

    while time.monotonic() < deadline:
        # pick a random contiguous block
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width
        cand = cur.copy()
        # log‑normal scaling of the block
        factor = float(np.exp(rng.normal(0.0, 0.12)))
        cand[start:stop] *= factor
        # occasional symmetrisation (helps a little)
        if rng.random() < 0.35:
            cand = 0.5 * (cand + cand[::-1])
        np.clip(cand, 0.0, 1000.0, out=cand)

        sc = _fast_score(cand, fft_len, n)
        if sc < cur_score:
            cur, cur_score = cand, sc
            if sc < best_score:
                best_seq, best_score = cand.copy(), sc

    # --------------------------------------------------------------- #
    # 5.  Return the best sequence found (as plain Python floats)
    # --------------------------------------------------------------- #
    # final sanity: ensure non‑negative and within bounds
    np.clip(best_seq, 0.0, 1000.0, out=best_seq)
    return [float(x) for x in best_seq]
