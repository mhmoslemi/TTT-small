        cand[source] -= delta
        cand[target] += delta
        # ensure nonnegativity
        if cand[source] < 0:
            cand[source] = 0.0

        # clip to bounds (some values may exceed)
        np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < cur_score:
            cur = cand
            cur_score = score
            if score < best_score:
                best = cand.copy()
                best_score = score
        # else possibly accept with small probability to escape
        else:
            # small probability
            if rng.random() < 0.001:
                cur = cand
                cur_score = score

        # occasionally perform random global perturbation to avoid stagnation
        if rng.random() < 0.005:
            width = rng.integers(1, max(2, n//20))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.2))
            cand2 = cur.copy()

            np.clip(cand2, 0.0, 1000.0, out=cand2)
            score2 = float(evaluate_sequence(cand2.tolist()))
            if np.isfinite(score2) and score2 < cur_score:
                cur = cand2
                cur_score = score2
                if score2 < best_score:
                    best = cand2.copy()
                    best_score = score2

    return best.tolist()

import numpy as np, time, math

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a step‑function that lowers the AC1 upper bound."""
    rng = np.random.default_rng(seed)
    # time limit (reserve a little margin for the final return)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # initialise a feasible sequence (use the supplied parent if possible)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        cur = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    else:
        cur = np.ones(256, dtype=np.float64) * 0.1

    if cur.ndim != 1 or cur.size == 0 or not np.isfinite(cur).all():
        cur = np.ones(256, dtype=np.float64) * 0.1

    if cur.sum() < 0.01:               # enforce the evaluator's lower bound
        cur[:] = 0.01 / cur.size

    n = cur.size
    # ------------------------------------------------------------------
    # helpers for fast convolution (FFT‑based, O(n log n))
    # ------------------------------------------------------------------
    L = int(2 ** np.ceil(np.log2(2 * n - 1)))     # power‑of‑2 size for FFT

    def conv_self(x):
        """Return full convolution of x with itself (FFT, real)."""
        a = np.fft.rfft(x, n=L)
        b = a * a
        return np.fft.irfft(b, n=L)[: 2 * n - 1]

    # ------------------------------------------------------------------
    # keep the best candidate seen so far
    # ------------------------------------------------------------------
    cur_score = float(evaluate_sequence(cur.tolist()))
    best = cur.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # main search loop – move mass from indices that contribute most to the
    # current maximal convolution entry to indices that contribute the least.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # 1) current convolution and the index where it peaks
        b = conv_self(cur)
        k_max = int(np.argmax(b))

        # 2) contributions of each position to that peak
        idx = np.arange(n)
        partner = k_max - idx
        valid = (partner >= 0) & (partner < n)
        contrib = np.zeros(n, dtype=np.float64)
        v = np.where(valid)[0]
        if v.size == 0:          # degenerate (should not happen)
            break
        contrib[v] = cur[v] * cur[partner[v]]

        # 3) choose source (large contribution) and target (small contribution)
        src = int(v[np.argmax(contrib[v])])
        # low‑contribution pool – include also invalid positions (contrib==0)
        low_pool = np.argsort(contrib)[: max(1, n // 100)]   # ~1 % of indices
        # ensure we do not pick the source itself
        low_pool = low_pool[low_pool != src]
        if low_pool.size == 0:
            tgt = rng.integers(0, n)
        else:
            tgt = int(rng.choice(low_pool))

        # 4) move a modest amount of mass from src to tgt
        if cur[src] <= 0.0:
            continue
        frac = rng.uniform(0.01, 0.15)                 # step size
        delta = cur[src] * frac
        # respect the upper bound 1000
        if cur[tgt] + delta > 1000.0:
            delta = 1000.0 - cur[tgt]
        if delta <= 0.0:
            continue
        cand = cur.copy()
        cand[src] -= delta
        cand[tgt] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        # 5) evaluate and accept if better
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < cur_score:
            cur = cand
            cur_score = sc
            if sc < best_score:
                best = cand.copy()
                best_score = sc
        else:
            # occasional uphill move to escape plateaus
            if rng.random() < 0.001:
                cur = cand
                cur_score = sc

        # 6) occasional random block perturbation (helps exploration)
        if rng.random() < 0.005:
            width = rng.integers(1, max(2, n // 20))
            start = rng.integers(0, n - width + 1)
            factor = math.exp(rng.normal(0.0, 0.2))
            alt = cur.copy()
            alt[start:start + width] *= factor
            np.clip(alt, 0.0, 1000.0, out=alt)
            sc2 = float(evaluate_sequence(alt.tolist()))
            if np.isfinite(sc2) and sc2 < cur_score:
                cur = alt
                cur_score = sc2
                if sc2 < best_score:
                    best = alt.copy()
                    best_score = sc2

    # ------------------------------------------------------------------
    # final fallback – return the best sequence discovered
    # ------------------------------------------------------------------
    return [float(v) for v in best]
