def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)
    # initial candidate: use height_sequence_1 if defined
    try:
        base = np.array(height_sequence_1, dtype=np.float64)
    except NameError:
        base = np.ones(256, dtype=np.float64)
    # ensure 1D finite
    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)
    # compute uniform candidate with same sum as base
    total = float(np.sum(base))
    n = base.size
    avg = total / n if n > 0 else 0.0
    uniform = np.full(n, avg, dtype=np.float64)
    # clip
    uniform = np.clip(uniform, 0.0, 1000.0)
    best = uniform
    best_score = float(evaluate_sequence(best.tolist()))
    # Also keep original as fallback in case uniform is not better
    base_score = float(evaluate_sequence(base.tolist()))
    if base_score < best_score:
        best = base.copy()
        best_score = base_score
