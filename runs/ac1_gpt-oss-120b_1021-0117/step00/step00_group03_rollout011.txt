def reduce_peak(a, conv, k_max, rng):
    n = a.size
    # indices that contribute to k_max
    i_start = max(0, k_max - (n-1))
    i_end = min(n-1, k_max)
    # get indices within that window
    contrib_idx = np.arange(i_start, i_end+1)
    # choose i among these with weight proportional to a_i
    weights = a[contrib_idx]
    if weights.sum() <= 0:
        return a.copy()
    i = rng.choice(contrib_idx, p=weights/weights.sum())
    j = k_max - i
    # choose recipient index outside (or any) where we can add mass
    # choose t randomly from all indices
    t = rng.integers(0, n)  # random index
    # compute max transferable from i (or j) to t:
    source_idx = i if a[i] >= a[j] else j
    target_idx = t
    max_transfer = min(a[source_idx], 1000 - a[target_idx])
    if max_transfer <= 0:
        return a.copy()
    delta = rng.random() * max_transfer * 0.5

def reduce_peak_move(a, k_max, rng):
    n = a.size
    i_start = max(0, k_max - (n-1))
    i_end = min(n-1, k_max)
    idx = np.arange(i_start, i_end+1)
    opp = k_max - idx
    # ensure opp in [0,n-1] is guaranteed
    prod = a[idx] * a[opp]
    # For each pair, we count product twice? Actually each unordered pair appears twice, but fine.
    # Choose index with max product.
    max_pos = np.argmax(prod)
    i = idx[max_pos]
    j = opp[max_pos]
    # choose source = i if a[i] >= a[j] else j
    source = i if a[i] >= a[j] else j
    # choose target random not equal source
    target = rng.integers(0, n)
    if target == source:
        target = (target + 1) % n
    max_transfer = min(a[source], 1000.0 - a[target])
    if max_transfer <= 0:
        return a.copy()
    delta = max_transfer * rng.random() * 0.5
    cand = a.copy()
    cand[source] -= delta
    cand[target] += delta
    return cand
