def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # Determine length n
    try:
        base_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = base_arr.size
        # fallback if invalid
        if n == 0 or not np.isfinite(base_arr).all():
            raise Exception
    except Exception:
        n = 256
        base_arr = np.ones(n, dtype=np.float64)

    best_seq = base_arr.copy()
    best_score = float('inf')
    # Evaluate base if possible
    try:
        s = evaluate_sequence(best_seq.tolist())
        if np.isfinite(s) and s < best_score:
            best_score = s
    except Exception:
        pass

    # Candidate: geometric sequences
    # Choose r values log-uniform between 0.1 and 0.9999
    # Use maybe 60 points.
    num_r = 60
    # Use logspace for small r and higher density near 1.
    r_vals = np.concatenate([
        np.linspace(0.1, 0.9, 20, endpoint=False),
        np.linspace(0.9, 0.9999, 40)
    ])
    # Ensure r in (0,1)
    for r in r_vals:
        if time.monotonic() >= deadline:
            break
        # compute geometric sequence
        a = np.empty(n, dtype=np.float64)
        # compute powers efficiently
        # Use np.arange to compute r**i
        a[:] = r ** np.arange(n, dtype=np.float64)
        # clip just in case
        a = np.clip(a, 0.0, 1000.0)
        # Evaluate
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()

    # Optionally include Sidon set (binary values) if time remaining
    if time.monotonic() < deadline:
        # generate Sidon index set (approx)
        used_sums = set()
        indices = []
        for x in range(n):
            # stop if cannot add many
            conflict = False
            if (x + x) in used_sums:
                continue
            for y in indices:
                if (x + y) in used_sums:
                    conflict = True
                    break
            if conflict:
                continue
            # accept x
            indices.append(x)
            used_sums.add(x + x)
            for y in indices[:-1]:
                used_sums.add(x + y)
            # limit size maybe up to 150
            if len(indices) >= 150:
                break
        if indices:
            a = np.zeros(n, dtype=np.float64)
            a[indices] = 1.0
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a.copy()

    # Ensure fallback non-empty
    if best_seq.size == 0:
        best_seq = np.ones(256, dtype=np.float64)

    # Guarantee min sum
    if best_seq.sum() < 0.01:
        # increase first element
        best_seq[0] += 0.01 - best_seq.sum()

    # Clip final bounds
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]

import time
import numpy as np

def _score(seq):
    """Fast wrapper around the authoritative evaluator."""
    try:
        s = evaluate_sequence(seq.tolist())
        return float(s) if np.isfinite(s) else float('inf')
    except Exception:
        return float('inf')


def _geometric_candidate(n, d):
    """
    Return a decreasing geometric sequence a_i = exp(-d * i) of length n.
    d > 0.  The values are already in the allowed range [0,1000].
    """
    # Using the exponent form is numerically stable for very close‑to‑1 ratios.
    a = np.exp(-d * np.arange(n, dtype=np.float64))
    # Clip just in case of tiny round‑off beyond the limits.
    return np.clip(a, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step‑function that minimizes the AC1 upper bound.
    The strategy is a simple sweep over geometric decays a_i = exp(-d*i)
    (theoretical family known to give very low scores) plus a fallback
    to the provided `height_sequence_1`.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # -----------------------------------------------------------------
    # Determine the length of the problem instance.
    # -----------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise Exception
        n = int(base.size)
    except Exception:
        # If the parent is missing or malformed fall back to a modest size.
        n = 256
        base = np.ones(n, dtype=np.float64)

    # -----------------------------------------------------------------
    # Initialise the best solution with the supplied parent (if any).
    # -----------------------------------------------------------------
    best_seq = base.copy()
    best_score = _score(best_seq)

    # -----------------------------------------------------------------
    # Explore geometric sequences of the form a_i = exp(-d*i).
    # The parameter d controls how quickly the heights decay.
    # For d very small the sequence is almost flat (score ≈ 2);
    # for d too large the sum becomes tiny (score blows up).
    # The sweet spot lies around d ≈ 0.0005 for n≈6000.
    # -----------------------------------------------------------------
    # Build a list of d values that is denser near the optimum.
    d_vals = np.concatenate([
        np.linspace(0.0001, 0.0010, 30, endpoint=False),
        np.linspace(0.0010, 0.0040, 40, endpoint=False),
        np.linspace(0.0040, 0.0100, 30)
    ])

    for d in d_vals:
        if time.monotonic() >= deadline:
            break
        cand = _geometric_candidate(n, d)
        # Ensure the sum is not catastrophically small (the evaluator needs ≥0.01).
        if cand.sum() < 1e-2:
            continue
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()

    # -----------------------------------------------------------------
    # As a very cheap extra, try a sparse “Sidon‑like” pattern.
    # It rarely beats the geometric family but costs almost nothing.
    # -----------------------------------------------------------------
    if time.monotonic() < deadline:
        used = set()
        idx = []
        for x in range(n):
            # quick abort if we already have a decent number of points
            if len(idx) >= 120:
                break
            conflict = False
            if (x + x) in used:
                continue
            for y in idx:
                if (x + y) in used:
                    conflict = True
                    break
            if conflict:
                continue
            idx.append(x)
            used.add(x + x)
            for y in idx[:-1]:
                used.add(x + y)

            cand[idx] = 1.0
            sc = _score(cand)
            if sc < best_score:

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return [float(v) for v in best_seq]
