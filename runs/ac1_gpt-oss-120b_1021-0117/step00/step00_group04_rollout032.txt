def fast_score(seq):
    n = seq.size
    # compute convolution via FFT
    # Determine FFT length as next power of two
    m = 1 << ((2*n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)
    conv = conv[:2*n-1]
    M = np.max(conv)
    S = np.sum(seq)
    if S <= 0:
        return np.inf
    return float(2 * n * M / (S * S))

import numpy as np
import time

def _initial_binomial(n: int) -> np.ndarray:
    """Generate a_i = C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        # a[i] = a[i-1] * (2*i-1) / (2*i)
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _fast_score(seq: np.ndarray):
    """Return (score, conv_array, max_val) using FFT‑based convolution."""
    n = seq.shape[0]
    # length for power‑of‑two FFT (covers full 2n‑1 convolution)
    m = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_val = np.max(conv)
    s = np.sum(seq)
    if s <= 0.0:
        return float("inf"), conv, max_val
    score = 2.0 * n * max_val / (s * s)
    return float(score), conv, max_val


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Hill‑climbing search starting from the optimal binomial seed."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Determine problem size and a valid parent (if supplied by the harness)
    # ------------------------------------------------------------------
    n = 3809  # default length
    parent = None
    try:
        # height_sequence_1 is injected by the evaluation harness
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            parent = np.clip(arr, 0.0, 1000.0)
            n = parent.size
    except Exception:
        parent = None

    # ---------------------------------------------------------
    # Two possible starts: the provided parent or the binomial one
    # ---------------------------------------------------------
    best = _initial_binomial(n)
    best_score, best_conv, _ = _fast_score(best)

    if parent is not None:
        p_score, _, _ = _fast_score(parent)
        if p_score < best_score:
            best = parent.copy()
            best_score = p_score
            best_conv, _ = best_conv, _  # recomputed already

    # ---------------------------------------------------------
    # Helper: recompute score & convolution for the current best
    # ---------------------------------------------------------
    def _refresh_best():
        nonlocal best_score, best_conv
        best_score, best_conv, _ = _fast_score(best)

    # ---------------------------------------------------------
    # Main stochastic improvement loop
    # ---------------------------------------------------------
    while time.monotonic() < deadline:
        candidate = best.copy()
        op = rng.integers(0, 4)

        if op == 0:  # block scaling
            width = rng.integers(1, max(2, n // 30))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.08))
            candidate[start : start + width] *= factor

        elif op == 1:  # single index perturbation
            idx = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.30))
            candidate[idx] *= factor

        elif op == 2:  # move mass away from the current maximal pair
            # locate a max‑convolution index
            k_max = int(np.argmax(best_conv))
            lo = max(0, k_max - (n - 1))
            hi = min(k_max, n - 1)
            i = rng.integers(lo, hi + 1)
            j = k_max - i
            if i != j:                     # avoid the diagonal case
                alpha = rng.uniform(0.04, 0.12)   # fraction to move
                removed_i = candidate[i] * alpha
                removed_j = candidate[j] * alpha
                candidate[i] -= removed_i
                candidate[j] -= removed_j
                total = removed_i + removed_j
                # add the mass to a random low‑value position
                tgt = rng.integers(0, n)
                while tgt == i or tgt == j:
                    tgt = rng.integers(0, n)
                candidate[tgt] += total

        else:  # op == 3 : smooth two neighbours
            i = rng.integers(0, n - 1)
            avg = (candidate[i] + candidate[i + 1]) * 0.5
            candidate[i] = avg
            candidate[i + 1] = avg

        # keep within the allowed box
        np.clip(candidate, 0.0, 1000.0, out=candidate)

        # evaluate candidate (FFT based, O(n log n))
        cand_score, cand_conv, _ = _fast_score(candidate)

        if cand_score < best_score:
            best = candidate
            best_score = cand_score
            best_conv = cand_conv

    # ------------------------------------------------------------------
    # Return a plain Python list – evaluator will re‑evaluate it exactly.
    # ------------------------------------------------------------------
    return [float(x) for x in best]
