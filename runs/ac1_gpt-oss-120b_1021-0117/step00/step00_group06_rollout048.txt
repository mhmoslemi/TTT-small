        n = base.size
    best = np.full(n, 1.0)  # initial constant
    best_score = float(evaluate_sequence(best.tolist()))
    # also check baseline parent if available
    if base is not None and base.size > 0:
        parent = np.clip(base, 0.0, 1000.0)
        parent_score = float(evaluate_sequence(parent.tolist()))
        if parent_score < best_score:

        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:

            best_score = score
    return [float(x) for x in best]

import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Determine length
    try:
        base_arr = np.asarray(height_sequence_1, dtype=np.float64)
        valid_base = (base_arr.ndim == 1 and base_arr.size > 0 and np.isfinite(base_arr).all())
    except Exception:
        valid_base = False

    if valid_base:
        n = base_arr.size
        parent = np.clip(base_arr, 0.0, 1000.0)
    else:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # Start with parent as a fallback
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Also test uniform constant equal to mean of parent (if any)
    if valid_base:
        const = float(parent.mean())
        uniform = np.full(n, const, dtype=np.float64)
        score = float(evaluate_sequence(uniform.tolist()))
        if np.isfinite(score) and score < best_score:
            best = uniform
            best_score = score

    # Random constant search
    while time.monotonic() < deadline:
        c = rng.uniform(0.0, 1000.0)
        cand = np.full(n, c, dtype=np.float64)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Return as plain Python list of floats
    return [float(x) for x in best]
