import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Construct a step‑function that places two equal blocks of the maximum
    allowed height at the two ends of the sequence, leaving a gap in the
    middle.  For a length n this yields a convolution peak of size ≈ n/3,
    giving a score around n/(2·⌊n/3⌋) ≈ 1.5 for the given instance.
    """
    try:
        n = len(height_sequence_1)          # parent length if available
    except Exception:                      # fallback length
        n = 256

    # block size – use the largest integer ≤ n/3 to keep the three
    # intervals (left block, gap, right block) disjoint in convolution.
    t = max(1, n // 3)

    seq = np.zeros(n, dtype=np.float64)
    seq[:t] = 1000.0          # left block at maximal height
    seq[-t:] = 1000.0         # right block at maximal height

    # Return a plain Python list as required by the evaluator.
    return [float(v) for v in seq]
