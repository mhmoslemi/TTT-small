def perturb_vector(v, rng):
    n = v.size
    # choose random segment width, as earlier
    width = rng.integers(max(1, n // 200), max(2, n // 12))
    start = rng.integers(0, max(1, n - width + 1))
    # copy and apply scaling factor exp(normal)
    factor = np.exp(rng.normal(0.0, 0.2))
    v2 = v.copy()
    v2[start:start+width] *= factor
    # optional smoothing: average with reversed?
    if rng.random() < 0.3:
        v2 = 0.5 * (v2 + v2[::-1])
    # ensure nonneg
    v2 = np.clip(v2, 0.0, None)
    # renormalize to sum=1
    total = v2.sum()
    if total > 0:
        v2 /= total
    return v2

import time
import numpy as np

# ---------- helper functions (top level) ----------

def _fft_size(n: int) -> int:
    """Smallest power of two >= 2*n-1 (full convolution length)."""
    size = 1
    need = 2 * n - 1
    while size < need:
        size <<= 1
    return size


def _max_convolution(a: np.ndarray, size: int) -> float:
    """Return max_k (a*a)[k] using a power‑of‑two FFT."""
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * a.size - 1]
    return float(conv.max())


def _objective(a: np.ndarray, n: int, size: int) -> float:
    """Exact objective 2*n*max(b)/(sum(a)^2).  a is non‑negative."""
    s = a.sum()
    if s <= 0.0:
        return np.inf
    maxb = _max_convolution(a, size)
    return 2.0 * n * maxb / (s * s)


def _grad_at_shift(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the active convolution term (a*a)[k] :
    g_j = 2*a_{k-j} for indices that stay inside the vector.
    """
    n = a.size
    idx = np.arange(n)
    partner = k - idx
    mask = (partner >= 0) & (partner < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[partner[mask]]
    return g


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x >= 0, sum(x)=1}.
    Standard O(n log n) algorithm (sorting).
    """
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive → uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _central_binomial_sequence(n: int) -> np.ndarray:
    """
    a_i = C(2i,i)/4^i  (central binomial / 4^i)  for i=0..n-1.
    Recurrence: a_0 = 1, a_i = a_{i-1} * (2i-1)/(2i).
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def _perturb(candidate: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative block perturbation followed by simplex re‑normalisation.
    Keeps the vector non‑negative and sum‑normalised.
    """
    n = candidate.size
    # block width roughly between n/200 and n/12
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = np.exp(rng.normal(0.0, 0.12))
    new = candidate.copy()
    new[start : start + width] *= float(factor)
    if rng.random() < 0.35:
        new = 0.5 * (new + new[::-1])
    new = np.clip(new, 0.0, None)
    s = new.sum()
    if s > 0.0:
        new /= s
    return new

        if s == 0.0:
