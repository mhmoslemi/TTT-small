    # but ensure O(n) memory.

        candidate = best.copy()
        # apply random jitter to a random contiguous block
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))

        stop = min(n, start + width)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)-3.0)
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 256
    # initialize with uniform (all ones)
    best = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))
    # generate random binary candidates
    while time.monotonic() < deadline:
        # choose p from a set of probabilities
        p = rng.uniform(0.2, 0.7)  # moderate range to explore
        candidate = (rng.random(n) < p).astype(np.float64)
        # evaluation
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            # optional: if score < target break early
            if best_score < 1.50274349:
                break
        # optionally perturb best slightly to explore local moves
        # use small block perturbation as earlier:
        if rng.random() < 0.1:
            # modify a few entries randomly weighted
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            # multiply selected block by random factor (between 0 and 2)
            factor = float(np.exp(rng.normal(0.0, 0.05)))
            candidate2 = best.copy()
            candidate2[start:stop] *= factor
            candidate2 = np.clip(candidate2, 0.0, 1000.0)
            score2 = float(evaluate_sequence(candidate2.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best = candidate2
                best_score = score2

    return [float(x) for x in best]

import time
import numpy as np

def _random_binary(n, p, rng):
    """Return a binary (float) array of length n with proportion p of ones."""
    return (rng.random(n) < p).astype(np.float64)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimizes the AC1 upper bound used by ``evaluate_sequence``.
    Uses random binary masks with varying densities and a simple
    hill‑climbing refinement.  The search respects the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # length of the sequence – use the provided parent if available
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 256

    # Start from the uniform sequence (score = 2.0) as a safe fallback.
    best = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))

    # Target threshold (provided in the statement) – stop early if reached.
    target = 1.50274349

    # Main stochastic search loop.
    while time.monotonic() < deadline:
        # Choose a density p; keep it away from the extremes where the score
        # is known to be poorer.
        p = rng.uniform(0.25, 0.75)

        # Generate a random binary candidate.
        cand = _random_binary(n, p, rng)

        # Evaluate the candidate.
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            if best_score < target:
                break

        # Occasionally perturb the current best to explore the neighbourhood.
        if rng.random() < 0.07:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)

            # Multiplicative jitter on a block; keep values in the allowed range.
            factor = float(np.exp(rng.normal(0.0, 0.06)))
            neigh = best.copy()
            neigh[start:stop] *= factor
            np.clip(neigh, 0.0, 1000.0, out=neigh)

            score_neigh = float(evaluate_sequence(neigh.tolist()))
            if np.isfinite(score_neigh) and score_neigh < best_score:
                best = neigh
                best_score = score_neigh
                if best_score < target:
                    break

    # Return a plain Python list of floats for the evaluator.
    return [float(x) for x in best]
```
