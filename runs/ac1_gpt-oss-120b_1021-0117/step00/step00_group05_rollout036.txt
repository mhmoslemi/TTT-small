import numpy as np
import time
import math

def _eval_score(seq):
    """Helper to evaluate and return a float score; safely handles list conversion."""
    return float(evaluate_sequence(seq))

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic search for a non‑negative step function whose AC1 upper bound
    (2*n*max(conv)/sum^2) is small.
    The algorithm mixes three simple moves:
      1. Targeted reduction of the dominant convolution entry.
      2. Random sinusoidal shapes.
      3. Local block perturbations (as in the reference solution).
    It keeps the best candidate seen before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    # give a small safety margin before the hard timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ----------------------------------------------------------------------
    # Initialise a starting sequence
    # ----------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    parent = np.clip(parent, 0.0, 1000.0)
    n = parent.size
    total = np.sum(parent)
    if total < 1e-9:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = parent / total                     # normalise to sum == 1

    # current best
    best = a.copy()
    best_score = _eval_score(best.tolist())

    idx = np.arange(n, dtype=np.int64)

    # ----------------------------------------------------------------------
    # Helper to try a new candidate and keep it if it improves the score
    # ----------------------------------------------------------------------
    def _try_update(candidate):
        nonlocal best, best_score, a
        cand = np.clip(candidate, 0.0, 1000.0)
        # renormalise to keep the scale (invariant for the score)
        s = np.sum(cand)
        if s > 0:
            cand = cand / s
        score = _eval_score(cand.tolist())
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score
            a = cand.copy()
            return True
        return False

    # ----------------------------------------------------------------------
    # Main optimisation loop
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        move_type = rng.integers(0, 3)   # 0 = targeted, 1 = sinusoid, 2 = block

        if move_type == 0:
            # --------------------------------------------------------------
            # Targeted move: reduce the largest convolution entry
            # --------------------------------------------------------------
            b = np.convolve(a, a)
            k = int(np.argmax(b))

            i_min = max(0, k - (n - 1))
            i_max = min(k, n - 1)
            i_vals = np.arange(i_min, i_max + 1, dtype=np.int64)
            j_vals = k - i_vals

            contrib = a[i_vals] * a[j_vals]
            if contrib.size == 0:
                continue
            max_idx = int(np.argmax(contrib))
            i_sel = i_vals[max_idx]
            j_sel = j_vals[max_idx]
            if i_sel == j_sel:
                continue

            # step size: a small fraction of the smaller partner
            step_frac = rng.uniform(0.02, 0.2)
            step = step_frac * min(a[i_sel], a[j_sel])
            if step <= 0.0:
                continue

            cand = a.copy()
            cand[i_sel] -= step
            cand[j_sel] -= step

            # redistribute the removed mass to two random locations
            r1 = rng.integers(0, n)
            while r1 == i_sel or r1 == j_sel:
                r1 = rng.integers(0, n)
            r2 = rng.integers(0, n)
            while r2 == i_sel or r2 == j_sel or r2 == r1:
                r2 = rng.integers(0, n)
            cand[r1] += step
            cand[r2] += step

            _try_update(cand)

        elif move_type == 1:
            # --------------------------------------------------------------
            # Sinusoidal candidate: a small sinusoidal modulation of a uniform base
            # --------------------------------------------------------------
            eps = rng.uniform(0.0, 0.9)                     # amplitude
            freq = rng.integers(1, 4)                      # 1,2,3
            phase = rng.random() * 2.0 * math.pi
            cand = 1.0 + eps * np.sin(2.0 * math.pi * freq * idx / n + phase)
            cand = np.clip(cand, 0.0, None)                # keep non‑negative
            if np.sum(cand) == 0.0:
                continue
            _try_update(cand)

        else:
            # --------------------------------------------------------------
            # Simple block perturbation (the baseline method)
            # --------------------------------------------------------------
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            cand = a.copy()
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            cand[start:stop] *= factor
            if rng.random() < 0.35:
                cand = 0.5 * (cand + cand[::-1])
            _try_update(cand)

    # ----------------------------------------------------------------------
    # Return the best sequence found (still normalised; scaling does not matter)
    # ----------------------------------------------------------------------
    final = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in final]
