def propose_candidate(seed=42,budget_s=1100, **kwargs):
   import numpy as np, time
   rng = np.random.default_rng(seed)
   deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)
   if 'height_sequence_1' in globals():
       n = len(height_sequence_1)
   else:
       n = 256
   best = None
   best_score = float('inf')
   # baseline: parent if exists
   try:
       parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
       if parent.ndim != 1 or parent.size != n or not np.isfinite(parent).all():
           raise Exception
   except Exception:
       parent = np.ones(n, dtype=np.float64) / n
   # Evaluate parent
   parent_score = evaluate_sequence(parent.tolist())
   if np.isfinite(parent_score):
       best = parent
       best_score = parent_score
   # test uniform
   uniform = np.full(n, 1.0/n)
   uni_score = evaluate_sequence(uniform.tolist())
   if uni_score < best_score:
       best = uniform; best_score = uni_score
   # test polynomial shapes
   # Choose integer exponents from 1 to max_exp (maybe 7)
   max_exp = 8
   for p in range(1, max_exp+1):
       vals = np.arange(1, n+1)[::-1].astype(np.float64) ** p
       vals = vals / vals.sum()
       # ensure positivity
       # clip not needed but to obey evaluator
       vals = np.clip(vals, 0.0, 1000.0)
       score = evaluate_sequence(vals.tolist())
       if np.isfinite(score) and score < best_score:
           best = vals
           best_score = score
   # Additional random perturbations of best shape
   # Use small block perturbations as earlier (like make_block_perturbation)
   # We'll re-use make_block_perturbation defined earlier maybe
   # Then hill climbing until deadline
   while time.monotonic() < deadline:
       # generate candidate by random scaling of a random block
       width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
       start = int(rng.integers(0, max(1, n - width + 1)))
       candidate = best.copy()
       factor = float(np.exp(rng.normal(0.0, 0.1)))
       candidate[start:start+width] *= factor
       # optionally symmetrize
       if rng.random() < 0.2:
           candidate = 0.5 * (candidate + candidate[::-1])
       candidate = np.clip(candidate, 0.0, 1000.0)
       # renormalize to sum 1
       s = candidate.sum()
       if s > 0:
           candidate = candidate / s
       score = evaluate_sequence(candidate.tolist())
       if np.isfinite(score) and score < best_score:
           best = candidate
           best_score = score
   return [float(x) for x in best]
