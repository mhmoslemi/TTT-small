Implementation of gradient_step:

def gradient_step(a, n, eta, top_frac, rng):
    max_val, k_max, sum_val = compute_metrics(a, n)
    idx = np.arange(n)
    partner = k_max - idx
    mask = (partner >= 0) & (partner < n)
    # g
    g = np.zeros_like(a)
    g[mask] = a[partner[mask]]
    # Determine indices sorted by g (descending)
    num = max(1, int(n * top_frac))
    sorted_idx = np.argsort(g)  # ascending
    bottom_idx = sorted_idx[:num]
    top_idx = sorted_idx[-num:]

    # Decrease top
    # delta_i = eta * a[i]; ensure not exceed a[i]
    delta = eta * a[top_idx]
    new_a = a.copy()
    new_a[top_idx] -= delta
    total_removed = np.sum(delta)
    # add equally to bottom
    if len(bottom_idx) > 0:
        add = total_removed / len(bottom_idx)
        new_a[bottom_idx] += add
    # Clip
    new_a = np.clip(new_a, 0.0, 1000.0)
    return new_a

Now implement compute_metrics:

def compute_metrics(a, n):
    # compute convolution via FFT, return max, index, sum
    L = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, L)
    conv = np.fft.irfft(fft_a * fft_a, L)[:2 * n - 1]
    # ensure real numbers
    max_idx = int(np.argmax(conv))
    max_val = float(conv[max_idx])
    sum_val = float(np.sum(a))
    return max_val, max_idx, sum_val

Now compute_score:

def compute_score(a, n):
    max_val, _, sum_val = compute_metrics(a, n)
    return 2.0 * n * max_val / (sum_val * sum_val)

def make_block_perturbation(base, rng):
    candidate = base.copy()
    n = candidate.size
    # block width random
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= factor
    # optional symm blending
    if rng.random() < 0.3:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper to get the next power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_metrics(a: np.ndarray, n: int):
    """
    Compute the linear convolution a*a via FFT, return max value,
    the index where it occurs, and the total sum of a.
    """
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    max_idx = int(np.argmax(conv))
    max_val = float(conv[max_idx])
    sum_val = float(np.sum(a))
    return max_val, max_idx, sum_val


def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective used by the official evaluator (up to rounding)."""
    max_val, _, sum_val = _conv_metrics(a, n)
    # guard against division by zero (should never happen for valid vectors)
    if sum_val == 0.0:
        return float("inf")
    return 2.0 * n * max_val / (sum_val * sum_val)


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Small multiplicative perturbation of a contiguous block, plus an optional
    symmetrical blending step (kept inexpensive).
    """
    cand = base.copy()
    n = cand.size
    # block length between n/200 and n/12 (inclusive, at least 1)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.3:            # occasional mild symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _gradient_step(a: np.ndarray, n: int, eta: float, top_frac: float,
                   rng: np.random.Generator) -> np.ndarray:
    """
    One gradient‑like redistribution step.
    Reduce the entries that contribute most to the current maximal convolution
    value and transfer the removed mass to the least‑contributing entries.
    """
    max_val, k_max, _ = _conv_metrics(a, n)

    idx = np.arange(n)
    partner = k_max - idx
    mask = (partner >= 0) & (partner < n)

    # contribution of each position to the maximal entry
    contrib = np.zeros_like(a)
    contrib[mask] = a[partner[mask]]

    # select a small fraction of indices with largest / smallest contributions
    num = max(1, int(n * top_frac))
    order = np.argsort(contrib)                     # ascending
    bottom = order[:num]                            # smallest contributions
    top = order[-num:]                              # largest contributions

    # decrease the “top’’ entries a little
    delta = eta * a[top]
    new_a = a.copy()
    new_a[top] -= delta
    removed = float(np.sum(delta))

    # distribute the removed mass uniformly to the “bottom’’ entries
    if removed > 0.0 and bottom.size > 0:
        add = removed / bottom.size
        new_a[bottom] += add

    # keep within the evaluator’s bounds
    return np.clip(new_a, 0.0, 1000.0)


def _sqrt_diff_sequence(n: int) -> np.ndarray:
    """
    a_i = sqrt(i+1) - sqrt(i)  (i = 0 … n‑1).
    This sequence is known to give a near‑constant a*a, yielding a low
    autocorrelation peak (≈π/2 ≈ 1.57 in the objective).
    """
    i = np.arange(n, dtype=np.float64)
    return np.sqrt(i + 1.0) - np.sqrt(i)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step sequence that minimizes the AC1 bound.
    Uses a mix of analytic start points, random block perturbations and a
    simple gradient‑type redistribution to flatten the convolution peak.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------

    try:

    except Exception:

        n = 256
        parent = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Assemble a few cheap candidates and keep the best as a seed
    # ------------------------------------------------------------------
    candidates = []

    # 1. the (possibly provided) parent
    candidates.append(parent)

    # 2. a pure uniform vector (baseline)
    candidates.append(np.ones(n, dtype=np.float64))

    # 3. the analytic sqrt‑difference sequence (excellent starting point)
    candidates.append(_sqrt_diff_sequence(n))

    # 4. a random Dirichlet‑like vector (adds diversity)
    rand_vec = rng.random(n, dtype=np.float64) * 1000.0
    candidates.append(rand_vec)

    best = None
    best_score = float("inf")
    for cand in candidates:
        sc = _objective(cand, n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = cand.copy()

    # use the current working vector
    current = best.copy()

    # Main stochastic optimisation loop
    # ------------------------------------------------------------------
    eta = 0.12                # step size for the gradient‑type move

    # ------------------------------------------------------------------
    return [float(x) for x in best]
