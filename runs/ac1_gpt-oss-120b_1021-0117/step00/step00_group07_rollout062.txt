def gradient_update(a, k_max, step):
    n = len(a)
    low = max(0, k_max - (n - 1))
    high = min(n - 1, k_max)
    indices = np.arange(low, high+1)
    partner_indices = k_max - indices
    a_new = a.copy()
    a_new[indices] = a_new[indices] - step * a[partner_indices]
    a_new = np.clip(a_new, 0, None)
    total = a_new.sum()
    if total > 0:
        a_new /= total
    else:
        a_new = np.ones_like(a) / n
    return a_new

    idx = int(conv.argmax())

    # fallback if something went wrong
    if best is None:
        best = np.ones(n, dtype=np.float64) / n
        best_score = 2.0

    # start the iterative improvement from the current best
    cur = best.copy()
    step = 0.12                     # initial learning rate for the gradient‑like move
    no_improve = 0                  # counter for stagnation

    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # current convolution and the index of its maximum
        # --------------------------------------------------------------
        cur_max, k_max = _conv_max(cur, fft_len)
        cur_score = 2.0 * n * cur_max / (cur.sum() ** 2)

        # keep the globally best found
        if cur_score < best_score:
            best, best_score = cur.copy(), cur_score
            no_improve = 0
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # gradient‑like update aimed at the peak diagonal
        # --------------------------------------------------------------
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        idx = np.arange(low, high + 1, dtype=np.int64)
        partner = k_max - idx

        cand = cur.copy()
        # move a little mass from each involved entry towards its partner
        cand[idx] = cand[idx] - step * cur[partner]
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s > 0:
            cand /= s
        else:
            cand = np.ones_like(cur) / n

        # --------------------------------------------------------------
        # occasional random block perturbation (exploration)
        # --------------------------------------------------------------
        if rng.random() < 0.15:
            blk_w = int(rng.integers(1, max(2, n // 200)))
            start = int(rng.integers(0, n - blk_w + 1))
            factor = np.exp(rng.normal(0.0, 0.1, size=blk_w))
            cand[start:start + blk_w] *= factor
            cand = np.clip(cand, 0.0, None)
            cand /= cand.sum()

        # --------------------------------------------------------------
        # evaluate the candidate (fast via FFT, no extra clipping)
        # --------------------------------------------------------------
        cand_max, _ = _conv_max(cand, fft_len)
        cand_score = 2.0 * n * cand_max / (cand.sum() ** 2)

        # accept if better (or tiny acceptance probability to escape plateaus)
        if cand_score < cur_score or rng.random() < 0.02:
            cur = cand
            cur_score = cand_score

        # decay the step size slowly
        step = max(0.003, step * 0.9995)

        # if we have not improved for a while, re‑initialise with a fresh exponential
        if no_improve > 400:
            r = 1.0 - rng.uniform(0.0004, 0.0030)
            cur = (r ** np.arange(n, dtype=np.float64))
            cur = np.clip(cur, 0.0, None)
            cur /= cur.sum()
            no_improve = 0
            step = 0.12

    # ------------------------------------------------------------------
    # Return the best sequence seen (as plain Python list)
    # ------------------------------------------------------------------
    # The values are already non‑negative and far below 1000,
    # so the evaluator will see exactly the intended numbers.
    return [float(v) for v in best]
