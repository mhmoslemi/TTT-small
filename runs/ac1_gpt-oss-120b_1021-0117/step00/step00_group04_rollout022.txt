import numpy as np
import time

def make_block_perturbation(base, rng):
    """Generate a cheap non‑negative perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # block width between 0.5% and 30% of the length
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= np.exp(rng.normal(0.0, 0.12))
    # occasional symmetrisation (helps for some shapes)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a non‑empty coefficient list with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Determine problem size
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = parent.size
    except Exception:
        # fallback if no parent is supplied
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Use the provided parent as an initial candidate (if it scores)
    best_seq = np.clip(parent, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best_seq.tolist()))
    if not np.isfinite(best_score):
        best_score = np.inf

    # ------------------------------------------------------------------
    # Construct a two‑stage sequence:
    #   - first part (head) : geometric decay r**i
    #   - second part (tail): constant epsilon
    # The parameters are chosen analytically to give a very low bound.
    beta_opt = 1.25643               # solves exp(beta)=1+2*beta
    # try a few head‑length fractions around 0.5
    for frac in (0.5, 0.55, 0.6, 0.45):
        if time.monotonic() >= deadline:
            break
        m = max(1, int(round(n * frac)))          # head length
        alpha = beta_opt / m
        if not (0.0 < alpha < 1.0):
            continue
        r = 1.0 - alpha
        i = np.arange(m, dtype=np.float64)
        head = np.power(r, i)                     # values in (0,1]
        sum_head = head.sum()
        # convolution of the head with itself (O(m^2) but cheap for m≈2000)
        max_head = np.convolve(head, head).max()
        epsilon = max_head / sum_head            # constant tail height
        # build full sequence
        tail_len = n - m
        if tail_len > 0:
            tail = np.full(tail_len, epsilon, dtype=np.float64)
            seq = np.concatenate([head, tail])
        else:
            seq = head
        np.clip(seq, 0.0, 1000.0, out=seq)

        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq

    # ------------------------------------------------------------------
    # Simple local search (block perturbations) if time permits
    while time.monotonic() < deadline:
        cand = make_block_perturbation(best_seq, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = cand
