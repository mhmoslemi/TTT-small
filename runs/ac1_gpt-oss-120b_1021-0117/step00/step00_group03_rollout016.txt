        # fallback length – any positive size works; 256 matches the old code
        return np.ones(256, dtype=np.float64)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Random‑search for a step‑function (non‑negative vector) that minimizes
    the AC1 upper bound used by `evaluate_sequence`.

    The search stays within linear memory, respects the time budget,
    and always returns a feasible fallback candidate.
    """
    rng = np.random.default_rng(seed)
    # Leave a small safety margin before the sandbox timeout.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    parent = _parent_sequence()
    n = parent.shape[0]

    # Initial best = parent (or a random vector if parent is invalid).
    best = parent.copy()
    try:
        best_score = float(evaluate_sequence(best.tolist()))
    except Exception:
        best_score = float("inf")
    if not np.isfinite(best_score):
        best = rng.random(n)
        best_score = float(evaluate_sequence(best.tolist()))

    # Random search loop – generate independent candidates and keep the best.
    while time.monotonic() < deadline:
        # Primary distribution: uniform in [0,1).
        cand = rng.random(n)

        # Occasionally switch to an exponential distribution for broader diversity.
        if rng.random() < 0.05:
            cand = rng.exponential(scale=0.5, size=n)

        # Evaluate using the authoritative function.
        score = float(evaluate_sequence(cand.tolist()))

        # Keep the improvement if it is finite and better.
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Return a plain Python list of floats as required.
    return [float(x) for x in best]
