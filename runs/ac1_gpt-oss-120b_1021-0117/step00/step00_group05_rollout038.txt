import numpy as np
import time

def _score(seq: np.ndarray) -> float:
    """Fast replica of the provided evaluator (without list conversion)."""
    a = np.clip(seq, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    conv = np.convolve(a, a)
    return 2.0 * a.size * conv.max() / (s * s)


def _base_shape(n: int) -> np.ndarray:
    """Monotone 1/sqrt(i+1) shape – the classical near‑optimal start."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return 1.0 / np.sqrt(i)


def _piecewise_shape(n: int, split: int, beta_tail: float) -> np.ndarray:
    """
    Head uses exponent 0.5, tail uses a larger exponent (faster decay).
    split – index where the exponent changes (0 < split < n).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    head = i[:split] ** -0.5
    tail = i[split:] ** -beta_tail
    out = np.empty_like(i)
    out[:split] = head
    out[split:] = tail
    return out
