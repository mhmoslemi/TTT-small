def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    n = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = arr.size
    except Exception:
        n = 256
    L = 1000.0
    q = int(math.isqrt(n))
    if q < 2:
        q = 2
    delta = L / q
    # Build Sidon set positions
    positions = np.empty(q, dtype=np.int64)
    for i in range(q):
        positions[i] = i * q + (i * i) % q
    # Ensure positions < n
    positions = positions[positions < n]
    m = positions.size
    # Build sequence
    seq = np.full(n, delta, dtype=np.float64)
    seq[positions] = L
    # Clip bounds
    seq = np.clip(seq, 0.0, 1000.0)
    # Evaluate
    score = evaluate_sequence(seq.tolist())
    # As fallback, compute parent if available
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            parent_score = evaluate_sequence(parent.tolist())
        else:
            parent = None
            parent_score = float('inf')
    except Exception:
        parent = None
        parent_score = float('inf')
    best_seq = seq
    best_score = float(score)
    if parent_score < best_score:
        best_seq = parent
        best_score = parent_score
    # Optionally run simple local search perturbations for a few iterations to try improve
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)
    # limit number of extra attempts
    while time.monotonic() < deadline:
        # perturb by adding small gaussian noise to a few random indices
        cand = best_seq.copy()
        # choose a few indices to perturb
        k = max(1, int(len(cand) * 0.01))
        idx = rng.choice(len(cand), size=k, replace=False)
        cand[idx] *= np.exp(rng.normal(0.0, 0.05, size=k))
        cand = np.clip(cand, 0.0, 1000.0)
        sc = evaluate_sequence(cand.tolist())
        if np.isfinite(sc) and sc < best_score:
            best_seq = cand
            best_score = sc
    return [float(x) for x in best_seq]

    yielding a score of 8*n*K/(n+K)**2 ≈ 10/9 ≈ 1.111 for any n.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Determine length n from the provided parent if possible
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
    except Exception:
        n = 256  # reasonable fallback length

    # Parameters
    L = 1000.0                 # maximal allowed height
    delta = L / 2.0            # secondary height (half of L)
    K = int(math.ceil(n / 5))  # number of entries at height L, satisfies 5*K >= n

    # Build the candidate sequence
    candidate = np.full(n, delta, dtype=np.float64)
    candidate[:K] = L
    candidate = np.clip(candidate, 0.0, 1000.0)

    # Evaluate the candidate
    best_seq = candidate
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # Also consider the original parent as a fallback
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            parent_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(parent_score) and parent_score < best_score:
                best_seq = parent
                best_score = parent_score
    except Exception:
        pass

    # Simple local perturbations (optional, cheap) until time runs out
    while time.monotonic() < deadline:
        # Small random multiplicative noise on a random 1‑% block
        seq = best_seq.copy()
        block_len = max(1, int(seq.size * 0.01))
        start = rng.integers(0, seq.size - block_len + 1)
        seq[start:start + block_len] *= np.exp(rng.normal(0.0, 0.03))
        seq = np.clip(seq, 0.0, 1000.0)
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = seq
            best_score = score

    return [float(x) for x in best_seq]
