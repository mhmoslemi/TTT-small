import numpy as np
import time
import math

def _conv_max(a, L):
    """Return convolution, its maximum value and the index of that maximum."""
    A = np.fft.rfft(a, L)
    conv = np.fft.irfft(A * A, L)[: 2 * a.size - 1]
    k = int(np.argmax(conv))
    return conv, conv[k], k


def _score_from_vals(a, max_val, sum_a):
    """Exact score as used by the authoritative evaluator."""
    if sum_a < 0.01:
        return np.inf
    return 2.0 * a.size * max_val / (sum_a ** 2)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """

    to minimise the AC1 upper bound evaluated by `evaluate_sequence`.

    theoretical lower bound 2.0) and then performs a cheap
    projected sub‑gradient descent with occasional random perturbations.
    The best sequence found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------

        base = np.clip(base, 0.0, 1000.0)

    except Exception:

        n = 256
        base = np.ones(n, dtype=np.float64)

    # target sum (the evaluator is scale‑invariant; this normalisation is convenient)

    uniform_val = target_sum / n

    best_score = _score_from_vals(best, mx, target_sum)
