    while time.monotonic() < deadline:

        # generate perturbation as small random factor multiply a subset of entries.
        width = max(1, n // 200)

            best_score = score

import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as evaluated by the provided ``evaluate_sequence``.

    A tiny randomized local search is kept as a fallback in case the

    # Base candidate: a uniform constant (score = 2 exactly).
    best = np.full(n, 1.0, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))

    # Small time‑bounded local search – it can only improve due to rounding.
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # Helper to evaluate and maybe accept a candidate.
    def try_candidate(cand):
        nonlocal best, best_score
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Perform cheap block perturbations while time permits.
    while time.monotonic() < deadline:
        # Choose a random block size (at most a few percent of n) and a
        # multiplicative factor close to 1.
        block_len = max(1, rng.integers(n // 200, max(2, n // 20)))
        start = rng.integers(0, max(1, n - block_len + 1))
        factor = float(np.exp(rng.normal(0.0, 0.015)))  # tiny perturbation

        cand = best.copy()
        cand[start:start + block_len] *= factor
        # Ensure all bounds required by the evaluator.
        np.clip(cand, 0.0, 1000.0, out=cand)

        if rng.random() < 0.12:
            cand = 0.5 * (cand + cand[::-1])

        try_candidate(cand)

    return [float(x) for x in best]
