def simplex_projection(v):
    # projects onto {x >=0, sum x = 1}
    # works for any v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import numpy as np
import time


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {x>=0, sum x = 1}."""
    # Implements the algorithm of Wang & Carreira-Perpinan (2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback (should not happen for non‑empty v)
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a non‑negative step‑function that lowers the AC1 bound.

    The algorithm builds a set of simple candidates, picks the best, and
    refines it by projected (simplex) gradient descent on the exact objective.
    All operations stay O(n) in memory; the exact evaluator is used only on
    the final return.
    """
    rng = np.random.default_rng(seed)
    # keep a small safety margin so the harness does not kill us
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # base length and optional parent construction
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            base = None
    except Exception:  # height_sequence_1 may not be defined
        base = None
    n = base.size if base is not None else 256

    # ------------------------------------------------------------------
    # build a few parametric seeds (all normalised to sum 1)
    seeds = []

    # 1. uniform
    seeds.append(np.full(n, 1.0 / n))

    # 2. linear decreasing
    lin = np.arange(n, 0, -1, dtype=np.float64)
    lin = lin / lin.sum()
    seeds.append(lin)

    # 3. exponential families (parameter p = δ·n)
    for p in np.linspace(0.5, 8.0, 15):
        delta = p / n
        r = max(0.0, 1.0 - delta)
        exp_seq = np.power(r, np.arange(n, dtype=np.float64))
        exp_seq = exp_seq / exp_seq.sum()
        seeds.append(exp_seq)

    # 4. power‑law families
    for alpha in (0.3, 0.5, 1.0, 1.5, 2.0):
        pw = 1.0 / np.arange(1, n + 1, dtype=np.float64) ** alpha
        pw = pw / pw.sum()
        seeds.append(pw)

    # 5. a few random Dirichlet draws
    for _ in range(5):
        rnd = rng.random(n)
        rnd = rnd / rnd.sum()
        seeds.append(rnd)

    # 6. parent if available
    if base is not None:
        parent = np.clip(base, 0.0, 1000.0)
        s = parent.sum()
        if s > 0.0:
            seeds.append(parent / s)

    # ------------------------------------------------------------------
    # pick the best initial seed (fast evaluation without calling the
    # heavyweight evaluator – the objective is scale‑invariant)
    best_seq = None
    best_score = np.inf
    for seq in seeds:
        if seq.sum() == 0.0:
            continue
        seq = seq / seq.sum()
        conv = np.convolve(seq, seq)          # full convolution
        cur = 2.0 * n * conv.max()           # because sum(seq)=1
        if cur < best_score:
            best_score = cur
            best_seq = seq.copy()

    if best_seq is None:                     # very unlikely fallback
        best_seq = np.full(n, 1.0 / n)
        best_score = 2.0 * n * np.convolve(best_seq, best_seq).max()

    # ------------------------------------------------------------------
    # projected gradient descent on the simplex
    seq = best_seq.copy()
    while time.monotonic() < deadline:
        conv = np.convolve(seq, seq)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]

        # build a_shift where a_shift[k] = seq[max_idx - k] when valid
        a_shift = np.zeros_like(seq)
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        # indices that contribute
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        a_shift[idx] = seq[max_idx - idx]

        # gradient of the exact objective (see analysis)
        grad = 4.0 * n * (a_shift - max_val * seq)

        # back‑track line search
        step = 1e-3
        improved = False
        for _ in range(12):
            cand = seq - step * grad
            cand = _simplex_projection(cand)          # stay on the simplex
            conv_c = np.convolve(cand, cand)
            cand_score = 2.0 * n * conv_c.max()
            if cand_score + 1e-12 < best_score:       # strict improvement
                best_score = cand_score
                best_seq = cand.copy()
                seq = cand
                improved = True
                break
            step *= 0.5
        if not improved:
            # occasional stochastic escape move
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            pert = seq.copy()
            pert[start:stop] *= np.exp(rng.normal(0.0, 0.2))
            if rng.random() < 0.1:                     # optional symmetry
                pert = 0.5 * (pert + pert[::-1])
            pert = np.clip(pert, 0.0, None)
            if pert.sum() > 0.0:
                pert = pert / pert.sum()
                conv_p = np.convolve(pert, pert)
                pert_score = 2.0 * n * conv_p.max()
                if pert_score < best_score:
                    best_score = pert_score
                    best_seq = pert.copy()
                    seq = pert

    # ------------------------------------------------------------------
    # scale to the range required by the evaluator (values in [0,1000])
    out = np.clip(best_seq * 1000.0, 0.0, 1000.0)
    return out.tolist()
