The projector onto simplex (sum to 1) algorithm: given vector v, we need u = max(v - tau, 0) where tau such that sum_u = 1. Standard algorithm:

```
def proj_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def propose_candidate(seed=42, budget_s=1100, **kwargs):

def local_swap_improve(arr, rng, deadline, max_iter=2000):
    cand = arr.copy()
    best_score = float(evaluate_sequence(cand.tolist()))
    n = len(cand)
    while time.monotonic() < deadline and max_iter > 0:
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        cand[i], cand[j] = cand[j], cand[i]
        # renorm? swapping does not change sum.
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            # keep swap
        else:
            # revert
            cand[i], cand[j] = cand[j], cand[i]
        max_iter -= 1

def generate_candidate(rng, n, target_sum):
    # shape parameter sample
    alpha = rng.uniform(0.5, 2.0)  # adjust
    raw = rng.gamma(shape=alpha, scale=1.0, size=n)
    # Avoid all zeros
    if raw.sum() == 0:
        raw = np.ones(n)
    raw = raw * (target_sum / raw.sum())
    # sort descending
    vals = np.sort(raw)[::-1]
    # interleave extremes
    cand = np.empty(n, dtype=np.float64)
    l = 0
    r = n - 1
    toggle = True
    for v in vals:
        if toggle:
            cand[l] = v
            l += 1
        else:
            cand[r] = v
            r -= 1
        toggle = not toggle
    # clip
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def _interleave_extremes(vals: np.ndarray) -> np.ndarray:
    """Place sorted values alternately from left and right ends."""
    n = vals.shape[0]
    out = np.empty(n, dtype=np.float64)
    left, right = 0, n - 1
    toggle = True
    for v in vals:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


def _make_candidate(rng: np.random.Generator, n: int, target_sum: float) -> np.ndarray:
    """Generate a random non‑negative step function with sum=target_sum."""
    # random concentration for the Dirichlet‑like distribution
