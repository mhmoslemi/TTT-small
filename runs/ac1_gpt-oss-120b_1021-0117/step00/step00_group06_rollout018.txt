        A = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[:2*n-1]
        dmax = int(conv.argmax())
        max_val = conv[dmax]
        # approximate score
        est_score = float(2 * n * max_val)  # because sum=1
        # store best
        if est_score < best_score:
            best = a.copy()
            best_score = est_score
        # gradient (subgradient of max at dmax)
        idx = np.arange(n)
        partner = dmax - idx
        mask = (partner >=0) & (partner < n)

        g[mask] = 2 * a[partner[mask]]
        # multiplicative update
        a = a * np.exp(-eta * g)
        # add some random noise occasionally
        if iteration % 50 == 0:
            noise = rng.normal(0.0, 0.01, size=n)
            a = a + noise
            a = np.clip(a, 0.0, None)
        # renorm
        s = a.sum()
        if s == 0:
            a = rng.exponential(scale=1.0, size=n)
            a = np.clip(a, 0.0, 1000.0)

def _conv_max(a, fft_len):
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:2 * len(a) - 1]
    idx = int(conv.argmax())
    return conv[idx], idx, conv
