    import numpy as np, time, cvxpy as cp
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 1.0
    try:
        n = len(height_sequence_1)
        f = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        n = 256
        f = np.ones(n, dtype=np.float64)
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    sqrt2n = float(np.sqrt(2 * n))
    # parameters
    K = 10
    tau = 0.2
    # Precompute segment boundaries
    seg_bounds = np.linspace(0, n, K+1, dtype=int)
    seg_lengths = np.diff(seg_bounds)
    while time.monotonic() < deadline:
        # compute M_f
        conv_f = np.convolve(f, f)
        M_f = float(np.max(conv_f))
        # compute prefix sum of f
        prefix = np.concatenate(([0.0], np.cumsum(f)))
        # Build C matrix (2n-1) x K
        T = 2 * n - 1
        C = np.empty((T, K), dtype=np.float64)
        for t in range(T):
            # compute lower_i and upper_i for each segment
            # compute contributions vectorized across segments
            # We'll loop over K, small.
            for k in range(K):
                start = seg_bounds[k]
                end = seg_bounds[k+1] - 1
                # if segment empty, coeff =0
                if start > end:
                    C[t, k] = 0.0
                    continue
                lo = max(0, t - end)
                hi = min(n-1, t - start)
                if lo <= hi:
                    C[t, k] = prefix[hi+1] - prefix[lo]
                else:
                    C[t, k] = 0.0
        # solve LP
        v = cp.Variable(K, nonneg=True)
        constraints = [C @ v <= M_f]
        objective = cp.Maximize(seg_lengths @ v)
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception:
            # fallback to ECOS or others
            prob.solve(solver=cp.CVXOPT)
        if v.value is None:
            break
        v_opt = np.maximum(v.value, 0.0)
        # construct b0
        b0 = np.empty_like(f)
        for k in range(K):
            b0[seg_bounds[k]:seg_bounds[k+1]] = v_opt[k]
        # scale b0 to have sum sqrt2n
        sum_b0 = b0.sum()
        if sum_b0 <= 0:
            break
        scale = sqrt2n / sum_b0
        b = b0 * scale
        # convex combine
        f = (1 - tau) * f + tau * b
        # clip
        np.clip(f, 0.0, 1000.0, out=f)
        # evaluate candidate
        score = float(evaluate_sequence(f.tolist()))
        if np.isfinite(score) and score < best_score:
            best = f.copy()
            best_score = score
    # return best
    return [float(x) for x in best]

    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

        f = np.ones(n, dtype=np.float64)
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    K = 10  # number of segments
    seg_bounds = np.linspace(0, n, K+1, dtype=int)
    seg_lengths = np.diff(seg_bounds)
    tau = 0.15
    # Main loop
    while time.monotonic() < deadline:
        # Evaluate convolution and max
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        # compute prefix sum of f
        prefix = np.concatenate(([0.0], np.cumsum(f)))
        T = 2 * n - 1
        # Build C matrix
        C = np.empty((T, K), dtype=np.float64)
        # Precompute for each segment start and end
        for t in range(T):
            for k in range(K):
                start = seg_bounds[k]
                end = seg_bounds[k+1] - 1
                if start > end:
                    C[t, k] = 0.0
                    continue
                lo = max(0, t - end)
                hi = min(n-1, t - start)
                if lo <= hi:
                    C[t, k] = prefix[hi+1] - prefix[lo]
                else:
                    C[t, k] = 0.0
        # Linear program: maximize sum(seg_lengths * v)
        c = -seg_lengths.astype(np.float64)  # minimize -sum
        # bounds: (0, None) for each variable
        bounds = [(0, None)] * K
        # Solve
        res = linprog(c, A_ub=C, b_ub=np.full(T, M), bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            # If LP fails, break loop
            break
        v_opt = np.maximum(res.x, 0.0)
        # construct b0
        b0 = np.empty_like(f)
        for k in range(K):
            b0[seg_bounds[k]:seg_bounds[k+1]] = v_opt[k]
        sum_b0 = b0.sum()
        if sum_b0 <= 0:
            break
        scale = sqrt2n / sum_b0
        b = b0 * scale
        # convex update
        f = (1.0 - tau) * f + tau * b
        # Clip to [0,1000]
        np.clip(f, 0.0, 1000.0, out=f)
        # Evaluate new candidate
        score = float(evaluate_sequence(f.tolist()))
        if np.isfinite(score) and score < best_score:
            best = f.copy()
            best_score = score
        # Early exit if we beat target (optional)

import numpy as np
import time
from scipy.optimize import linprog


def _build_constraint_matrix(f, seg_bounds, prefix):
    """
    Build the (2n-1) x K matrix C where
    C[t, k] = sum_{i in segment k} f[i] for indices i such that t-i lies in segment k.
    """
    n = f.size
    T = 2 * n - 1
    K = len(seg_bounds) - 1
    C = np.empty((T, K), dtype=np.float64)
    for t in range(T):
        for k in range(K):
            start = seg_bounds[k]
            end = seg_bounds[k + 1] - 1          # inclusive
            if start > end:                       # empty segment
                C[t, k] = 0.0
                continue
            lo = max(0, t - end)
            hi = min(n - 1, t - start)
            if lo <= hi:
                C[t, k] = prefix[hi + 1] - prefix[lo]
            else:
                C[t, k] = 0.0
    return C

    tau = 0.12                               # mixing factor (0 < tau ≤ 1)

    # ----------------------------------------------------------------------
    # Main improvement loop – runs until time runs out or target reached.
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        # current peak of the self‑convolution
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))

        # prefix sums of f for fast range queries
        prefix = np.concatenate(([0.0], np.cumsum(f, dtype=np.float64)))

        # build the constraint matrix C (size (2n‑1) × K)
        C = _build_constraint_matrix(f, seg_bounds, prefix)

        # Linear program:
        #   maximize   sum_k seg_lengths[k] * v_k
        #   subject to C @ v <= M   and   v >= 0
        # Using scipy.optimize.linprog (highs solver)
        c = -seg_lengths                      # minimize negative sum => maximize sum
        bounds = [(0, None)] * K
        res = linprog(c, A_ub=C, b_ub=np.full(C.shape[0], M),
                      bounds=bounds, method='highs', options={'presolve': True})

        if not res.success or res.x is None:
            # inability to solve LP – abort further optimisation
            break

        v_opt = np.maximum(res.x, 0.0)

        # expand the segment heights back to a full‑length vector b0
        b0 = np.empty_like(f)
        for k in range(K):
            b0[seg_bounds[k]:seg_bounds[k + 1]] = v_opt[k]
