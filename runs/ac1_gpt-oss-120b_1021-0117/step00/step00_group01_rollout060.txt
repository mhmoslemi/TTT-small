import numpy as np
import time

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Construct a simple two‑block sequence.
    Two equal blocks of height 1000 are placed with a zero gap large enough
    (gap ≥ block length).  This yields a very low AC1 bound:
        score = n / (2 * block_len)   (scale‑invariant)
    The block length is chosen as floor(n/3), the largest possible value
    satisfying the gap condition, giving a score ≈ 1.5 for the default length.
    """
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Determine the length n and (if available) a parent sequence.
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise Exception
        n = parent_arr.size
        parent_valid = True
    except Exception:
        parent_arr = None
        n = 256                     # reasonable fallback length
        parent_valid = False

    # ------------------------------------------------------------------
    # Build the two‑block candidate.
    # ------------------------------------------------------------------
    if n < 3:
        # degenerate case – just use a uniform vector.
        cand = np.ones(n, dtype=np.float64) * 1000.0
    else:
        k = n // 3                         # block length = floor(n/3)
        gap = n - 2 * k                    # remaining positions (gap ≥ k)
        # Create two blocks of height 1000 and a zero gap between them.
        block = np.full(k, 1000.0, dtype=np.float64)
        zeros = np.zeros(gap, dtype=np.float64)
        cand = np.concatenate([block, zeros, block])
        # In the rare case rounding leaves the array shorter/longer, adjust.
        if cand.size != n:
            # truncate or pad with zeros as needed
            if cand.size > n:
                cand = cand[:n]
            else:
                cand = np.concatenate([cand, np.zeros(n - cand.size, dtype=np.float64)])

    # ------------------------------------------------------------------
    # Evaluate candidate (and parent if present) and keep the best.
    # ------------------------------------------------------------------
    best_seq = cand
    best_score = float(evaluate_sequence(best_seq.tolist()))

    if parent_valid:
        parent_score = float(evaluate_sequence(parent_arr.tolist()))
        if parent_score < best_score:
            best_seq = parent_arr.copy()
            best_score = parent_score

    # ------------------------------------------------------------------
    # Optional tiny local search: try moving a single element between the
    # two blocks (changing their lengths by ±1) while preserving the gap≥k
    # condition.  This can capture the +1 remainder case when n%3 != 0.
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    while time.monotonic() < deadline:
        # current block length
        cur_k = (best_seq != 0).sum() // 2
        if cur_k < 1:
            break
        # propose a new block length in {cur_k-1, cur_k, cur_k+1}
        new_k = int(cur_k + rng.integers(-1, 2))
        if new_k < 1:
            continue
        # recompute gap; must satisfy gap >= new_k and total length = n
        new_gap = n - 2 * new_k
        if new_gap < new_k:
            continue
        block = np.full(new_k, 1000.0, dtype=np.float64)
        zeros = np.zeros(new_gap, dtype=np.float64)
        new_seq = np.concatenate([block, zeros, block])
        score = float(evaluate_sequence(new_seq.tolist()))
        if score < best_score:
            best_seq = new_seq
            best_score = score
            # continue searching around the improved length
            continue
        # break if no improvement after a few attempts
        if rng.random() < 0.01:
            break

    return [float(x) for x in best_seq]
