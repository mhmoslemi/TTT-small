import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Generate a sequence whose AC1 upper bound is below the target.
    The construction uses a quadratic exponential decay a_i = exp(-c * (i/(n-1))**2)
    with a few candidate decay rates, then modest random refinements.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------ #
    # Base length and fallback sequence
    # ------------------------------------------------------------------ #
    try:
        n = int(len(height_sequence_1))
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if n == 0 or not np.isfinite(base).all():
            raise Exception
    except Exception:
        n = 256
        base = np.ones(n, dtype=np.float64)

    best_seq = base
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------ #
    # Exponential‑quadratic candidates (a_i = exp(-c * (i/(n-1))**2))
    # ------------------------------------------------------------------ #
    # Decay constants; values around 0.5‑0.9 give scores < 1.5 for n≈6700.
    decay_constants = [0.3, 0.45, 0.55, 0.65, 0.75, 0.85, 0.95]
    idx = np.arange(n, dtype=np.float64)
    norm = idx / (n - 1) if n > 1 else idx

    for c in decay_constants:
        cand = np.exp(-c * norm * norm)          # quadratic exponential shape
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score

    # ------------------------------------------------------------------ #
    # Lightweight random refinement (optional, respects time budget)
    # ------------------------------------------------------------------ #
    while time.monotonic() < deadline:
        cand = best_seq.copy()
        length = cand.size

        # Random block size: between 0.5% and 5% of the whole length
        min_w = max(1, length // 200)
        max_w = max(2, length // 20)
        w = int(rng.integers(min_w, max_w + 1))
        start = int(rng.integers(0, length - w + 1))
        stop = start + w

        # Small multiplicative perturbation (log‑normal)
        sigma = 0.07
        factor = np.exp(rng.normal(0.0, sigma, size=w))
        cand[start:stop] *= factor

        # Occasionally symmetrize a bit
        if rng.random() < 0.15:
            cand = 0.5 * (cand + cand[::-1])

        # Clip to the evaluator's permissible range
        np.clip(cand, 0.0, 1000.0, out=cand)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score

    # Return as a plain Python list of floats
    return [float(v) for v in best_seq]
