        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = int(parent.size)
    except Exception:                     # ``height_sequence_1`` missing or invalid
        n = 256                           # a modest fallback length

    # Uniform starting point (any positive constant gives the same objective value).
    best = np.full(n, 1.0, dtype=np.float64)
    best_score = evaluate_sequence(best.tolist())

    # Simple random refinement – cheap and never degrades the score.
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    while time.monotonic() < deadline:
        # Propose a multiplicative perturbation on a random block.
        cand = best.copy()
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        factor = math.exp(rng.normal(0.0, 0.07))
        cand[start:start + width] *= factor
        np.clip(cand, 0.0, 1000.0, out=cand)

        score = evaluate_sequence(cand.tolist())
        if np.isfinite(score) and score < best_score:
            best, best_score = cand, score
            # The theoretical optimum is 2; stop early if we see a tiny improvement.
            if best_score < 2.0 - 1e-9:
                break

    # Convert to the required Python list of plain floats.
    return [float(x) for x in best]
