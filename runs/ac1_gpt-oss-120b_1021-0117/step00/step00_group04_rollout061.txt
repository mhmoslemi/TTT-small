        active_idx = np.where(conv >= M * (1 - tol))[0]
        # ensure at least one constraint

            col_range = np.arange(j_start, j_end + 1, dtype=np.int32)

            rows.append(np.full(col_range.shape, row, dtype=np.int32))
            cols.append(col_range)
            data.append(f_vals)
            row += 1
        if rows:
            row_arr = np.concatenate(rows)
            col_arr = np.concatenate(cols)
            data_arr = np.concatenate(data)

            A_active = sp.csr_matrix((0, n))
        # Linear program: maximize sum(b) => minimize -sum(b)
        c = -np.ones(n)

        best = best + (0.01 - best.sum()) / n
    return [float(x) for x in best]

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

def _build_active_matrix(f, active_idx):
    """Construct sparse matrix for constraints f * b <= M at rows active_idx."""
    n = f.size
    rows = []
    cols = []
    data = []
    row_id = 0
    for t in active_idx:
        j_start = max(0, t - (n - 1))
        j_end = min(n - 1, t)
        cols_row = np.arange(j_start, j_end + 1, dtype=np.int32)
        rows.append(np.full(cols_row.shape, row_id, dtype=np.int32))
        cols.append(cols_row)
        data.append(f[t - cols_row])
        row_id += 1
    if not rows:
        return sp.csr_matrix((0, n), dtype=np.float64)
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    return sp.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n), dtype=np.float64)

    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------ #

    # ------------------------------------------------------------------ #
    try:

    else:

    # scale to unit ℓ₁‑norm (the objective is scale‑invariant)
    if base.sum() == 0:
        base = np.ones_like(base) / n
    else:
        base = base / base.sum()

    seq = base.copy()
    best_seq = seq.copy()

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    it = 0
    # ------------------------------------------------------------------ #
    #   main improvement loop                                             #
    # ------------------------------------------------------------------ #
    while time.monotonic() < deadline and it < 30:
        it += 1
        conv = np.convolve(seq, seq)
        M = conv.max()

        tol = 0.02                      # 2 % of the max
        active = np.where(conv >= M * (1 - tol))[0]

        A = _build_active_matrix(seq, active)
        # maximise   sum(b)   subject to A·b ≤ M  ,  b ≥ 0
        c = -np.ones(n, dtype=np.float64)          # linprog does minimisation
        ub = np.full(A.shape[0], M, dtype=np.float64)

        # time left for the linear programme
        time_left = max(0.0, deadline - time.monotonic())
        try:

                A_ub=A,
                b_ub=ub,

        except Exception:                         # pragma: no cover

            b *= scale

        new_seq = (1.0 - mix) * seq + mix * b
