    # Determine n

    except NameError:
        n = 256

    best_score = float('inf')
    best_seq = None

    # choose L values - could try few options
    # ensure L between 1 and n//2
    candidates_L = [max(1, int(n * r)) for r in [0.1, 0.2, 0.25, 0.33, 0.4]]
    # also include n//4 exactly

    # unique
    candidates_L = sorted(set(candidates_L))

    for L in candidates_L:
        if L >= n:
            continue
        epsilon = np.sqrt(L / (n - L))
        # Build array
        seq = np.empty(n, dtype=np.float64)
        seq[:L] = 1.0
        seq[L:] = epsilon
        # Clip, just in case
        seq = np.clip(seq, 0.0, 1000.0)
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
            # early exit if meets target
            if best_score <= 1.50274349:
                break
        # Check time
        if time.monotonic() >= deadline:
            break

    # fallback: if no constructed, fallback to parent or uniform
    if best_seq is None:
        # fallback to uniform
        best_seq = np.ones(n, dtype=np.float64)

    return [float(x) for x in best_seq]
