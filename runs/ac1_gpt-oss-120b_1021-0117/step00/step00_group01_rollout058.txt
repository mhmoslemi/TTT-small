def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, random
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0., float(budget_s) - 1.0) # margin
    # initialization: parent fallback
    try:
        parent_arr = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        parent_arr = np.ones(256, dtype=np.float64)
    best_arr = parent_arr.copy()
    best_score = float(evaluate_sequence(best_arr.tolist()))
    n = best_arr.size

    # Precompute range for m and epsilon
    # max possible m less than n
    m_low = max(30, int(np.sqrt(n) * 0.8)) # maybe 60
    m_high = int(np.sqrt(n) * 1.5) + 20   # maybe 120-130
    eps_low = 0.004
    eps_high = 0.02

    # main random search loop
    while time.monotonic() < deadline:
        m = int(rng.integers(m_low, m_high+1))
        eps = float(rng.uniform(eps_low, eps_high))
        candidate = np.full(n, eps, dtype=np.float64)
        # choose m distinct indices
        idx = rng.choice(n, size=m, replace=False)
        candidate[idx] = 1.0
        # optionally shuffle large values maybe 0.9 instead of 1?
        # compute score
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best_arr = candidate
            best_score = score
            # adapt search ranges maybe tighten around current eps,m
            # Not strictly needed
        # occasional small perturbation of best
        if rng.random() < 0.1:
            # modify a few positions of best candidate
            cand2 = best_arr.copy()
            # random block scaling
            width = int(rng.integers(1, max(2, n // 100)))  # small block
            start = int(rng.integers(0, n - width))
            factor = float(np.exp(rng.normal(0., 0.1)))
            cand2[start:start+width] *= factor
            cand2 = np.clip(cand2, 0.0, 1000.0)
            score2 = float(evaluate_sequence(cand2.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best_arr = cand2
                best_score = score2
    return [float(x) for x in best_arr]

import numpy as np
import time

def _build_large_set(n, max_pairs_per_sum=2):
    """Greedy construction of a set where each unordered sum appears at most
    `max_pairs_per_sum` times (i.e. ordered contributions ≤ 2*max_pairs)."""
    large = []
    cnt = {}                       # unordered pair count for each sum
    for i in range(n):
        # quick reject if self‑sum already at limit
        if cnt.get(i + i, 0) >= max_pairs_per_sum:
            continue
        ok = True
        for j in large:
            s = i + j
            if cnt.get(s, 0) >= max_pairs_per_sum:
                ok = False
                break
        if not ok:
            continue
        # accept i
        for j in large:
            s = i + j
            cnt[s] = cnt.get(s, 0) + 1
        cnt[i + i] = cnt.get(i + i, 0) + 1
        large.append(i)
    return np.array(large, dtype=np.int64)


def _max_convolution_of_binary(vec):
    """Return max entry of the full autocorrelation of a 0/1 vector."""
    n = vec.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    ft = np.fft.rfft(vec, n=size)
    conv = np.fft.irfft(ft * ft, n=size)[: 2 * n - 1]
    return float(conv.max())

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------

    try:

    best_seq = parent.copy()

        return [float(x) for x in best_seq]

    # Pre‑compute the contribution from the large entries alone
    binary = np.zeros(n, dtype=np.float64)
    binary[large_idx] = 1.0
    M_large = _max_convolution_of_binary(binary)  # ordered count at the worst sum

    # ------------------------------------------------------------------
    eps_grid = np.linspace(0.005, 0.08, 31)  # 31 candidates, cheap enough
    for eps in eps_grid:

        if np.isfinite(score) and score < best_score:
