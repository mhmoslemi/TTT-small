import time
import numpy as np

def _generate_binomial_seq(n: int) -> np.ndarray:
    """Return a length‑n array a_i = C(2i,i)/4^i using the recurrence
    a_{i+1} = a_i * (2i+1)/(2i+2). All entries are positive and decreasing."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2 * i + 2)
    return a


def _conv_max_fft(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the linear convolution a*a using a single FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return float(conv.max())


def _score_fft(a: np.ndarray, n: int, fft_len: int) -> float:
    """Fast proxy for evaluate_sequence (identical for non‑negative inputs)."""
    max_b = _conv_max_fft(a, fft_len)
    s = a.sum()
    return float(2 * n * max_b / (s * s))


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a sequence of non‑negative heights with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 1.0)

    # Length of the problem (normally supplied via height_sequence_1)
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256                     # harmless fallback length

    # ---- 1. Construct a principled starting point --------------------
    a = _generate_binomial_seq(n)                     # a_i = C(2i,i)/4^i
    a = np.clip(a, 0.0, 1000.0)                       # safety, already satisfied
    total = a.sum()
    if total == 0.0:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = a / total                                 # normalise sum to 1

    # FFT layout (same for the whole run)
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # ---- 2. Evaluate the starter ------------------------------------
    cur = a.copy()
    cur_score = _score_fft(cur, n, fft_len)
    best = cur.copy()
    best_score = cur_score

    # ---- 3. Simple stochastic hill‑climbing -------------------------
    # Move mass between two positions, keep total unchanged.
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)

            if new_score < best_score:

                best_score = new_score
