import numpy as np, time

def _eval(seq):
    return float(evaluate_sequence(seq))

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    # try to get init from parent if present
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        base = None
    if base is None or base.ndim != 1 or base.size == 0:
        n = 256
        base = np.ones(n, dtype=np.float64)
    else:
        n = base.size
        base = np.clip(base, 0.0, 1000.0)

    # constant candidate
    const = np.full(n, 1.0, dtype=np.float64)  # value arbitrary positive
    const = np.clip(const, 0.0, 1000.0)
    best = const.copy()
    best_score = _eval(best.tolist())

    # Compare with parent
    parent_score = _eval(base.tolist())
    if parent_score < best_score:
        best = base.copy()
        best_score = parent_score

    # Simple local search
    while time.monotonic() < deadline:
        # produce perturbed version of best
        cand = best.copy()
        # choose a window size relative to n (1% to 10%)
        w = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - w + 1)))
        stop = min(n, start + w)
        # multiplicative perturbation
        factor = np.exp(rng.normal(0.0, 0.05, size=stop-start))
        cand[start:stop] *= factor
        # optional symmetrization
        if rng.random() < 0.2:
            cand = 0.5 * (cand + cand[::-1])
        cand = np.clip(cand, 0.0, 1000.0)
        score = _eval(cand.tolist())
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            # adapt step size or early exit if near theoretical lower bound?
            if best_score <= 2.000001:
                # can't improve much; break early
                break
