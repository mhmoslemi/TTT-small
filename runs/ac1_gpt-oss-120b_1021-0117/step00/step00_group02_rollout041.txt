    return positions

import numpy as np
import time

def _score(a: np.ndarray) -> float:
    """Fast evaluation matching evaluate_sequence (without clipping)."""
    n = a.shape[0]
    b = np.convolve(a, a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # length of the sequence (use the provided parent if present)
    if 'height_sequence_1' in globals():
        n = int(np.asarray(height_sequence_1, dtype=np.float64).size)
    else:
        n = 2000  # fallback length if parent is unavailable

    # ------------------------------------------------------------
    # 1.  Initialise a sparse random candidate.
    # ------------------------------------------------------------
    # Target total sum – we start with the “natural” normalisation
    # sum = sqrt(2n).  The absolute scale does not affect the ratio,
    # but a larger sum gives more room for later redistributions.
    target_sum = np.sqrt(2.0 * n)

    # Number of non‑zero entries – a bit larger than sqrt(2n) to give
    # us slack for improving the max‑convolution.
    m_initial = max(80, int(2.5 * np.sqrt(2.0 * n)))

    # pick distinct indices uniformly at random
    idx = rng.choice(n, size=m_initial, replace=False)
    vals = rng.random(m_initial)  # uniform (0,1)

    # scale so that the total sum equals target_sum
    vals = vals * (target_sum / vals.sum())
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals

    # enforce evaluator bounds (0 … 1000)
    np.clip(a, 0.0, 1000.0, out=a)

    best = a.copy()
    best_score = _score(best)

    # ------------------------------------------------------------
    # 2.  Simple hill‑climbing / simulated‑annealing loop.
    # ------------------------------------------------------------
    # The move: pick two positions i→j and shift a small amount δ.
    # The total sum stays (almost) constant, so the denominator does
    # not change – we only try to lower the maximal convolution entry.
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # we only move mass from a positive entry
        if a[i] <= 0.0:
            continue

        # propose a transfer amount – at most 20 % of the source value
        delta = a[i] * (rng.random() * 0.2)

        if delta <= 0.0:
            continue

        # build the tentative new vector
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # keep within the global bounds
        if a_new[i] < 0.0 or a_new[j] > 1000.0:
            continue
        np.clip(a_new, 0.0, 1000.0, out=a_new)

        sc = _score(a_new)

        # accept if improvement (or with a tiny probability to escape)
        if sc < best_score or rng.random() < 0.001:
            a = a_new
            if sc < best_score:
                best = a_new
                best_score = sc

    # final safety – run the official evaluator once on the best vector
    # (the official evaluator also clips values, so we mimic that step)
    best_clipped = np.clip(best, 0.0, 1000.0).astype(float).tolist()
    return best_clipped
