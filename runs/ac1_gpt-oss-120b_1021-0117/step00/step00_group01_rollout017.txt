def propose_candidate(seed=42, budget_s=1100, **kwargs):

    The exponent can be raised without changing the scaling invariance of the

    """
    # Basic timing – leave a small safety margin before the sandbox timeout.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # 1️⃣  Determine a base length.
    try:
        n = int(len(height_sequence_1))

    except Exception:

        n = 256

    # Parent must respect the evaluator's clipping rules.
    parent = np.clip(parent, 0.0, 1000.0)
    best_seq = parent
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------
    # 2️⃣  Construct a deterministic power‑law candidate.
    #    a_i = C * i^p   with i = 0,…,n‑1,  C chosen so that max a_i = 1000.
    #    The exponent p≥1 dramatically lowers the constant C in the bound.
    p = 10                     # exponent (larger → smaller score)
    idx = np.arange(n, dtype=np.float64)          # 0 … n‑1
    raw = idx ** p                                 # i^p  (non‑negative)

    max_raw = raw[-1] if raw[-1] != 0 else 1.0
    scale = 1000.0 / max_raw
    cand = raw * scale
    cand = np.clip(cand, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # 3️⃣  Evaluate and keep the better of parent / candidate.
    cand_score = float(evaluate_sequence(cand.tolist()))
