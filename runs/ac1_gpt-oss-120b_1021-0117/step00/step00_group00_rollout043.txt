def conv_max(a):
    n = a.size
    size = 2*n -1
    # Next power of 2 for FFT to avoid circular convolution.
    m = 1 << (size-1).bit_length()
    A = np.fft.rfft(a, n=m)
    B = A * A
    c = np.fft.irfft(B, n=m)[:size]
    return float(c.max())

def _fast_score(a: np.ndarray) -> float:
    """Compute the AC1 upper bound quickly using FFT."""
    n = a.shape[0]
    size = 2 * n - 1
    # next power of two for efficient FFT
    m = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[:size]
    return 2.0 * n * conv.max() / (a.sum() ** 2)
