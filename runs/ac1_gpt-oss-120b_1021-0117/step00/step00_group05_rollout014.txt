                         0.0, 1000.0)
        if parent.ndim == 1 and parent.size == n:
            parent_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(parent_score) and parent_score < best_score:
                best = parent.copy()
                best_score = parent_score

    # --------------------------------------------------------------------
    # Very cheap stochastic local search (block multiplicative perturbations).
    # The search cannot beat the mathematical lower bound of 2, but this
    # framework leaves room for any accidental rounding‑driven improvement.
    # --------------------------------------------------------------------
    while time.monotonic() < deadline:
        # pick a random block (width proportional to n, at least 1)
        width = max(1, n // 200)
        start = rng.integers(0, n - width + 1)
        # multiplicative jitter (log‑normal step)
        factor = math.exp(rng.normal(0.0, 0.07))
        cand = best.copy()
        cand[start:start + width] *= factor
        np.clip(cand, 0.0, 1000.0, out=cand)          # respect bounds
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # --------------------------------------------------------------------
    # Return a plain Python list as required by the evaluator.
    # --------------------------------------------------------------------
    return [float(v) for v in best]
