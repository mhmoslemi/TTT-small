import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimises
    the AC1 upper bound defined by ``evaluate_sequence``.
    The trivial uniform sequence achieves the theoretical optimum
    (score = 2).  If a parent sequence ``height_sequence_1`` is supplied
    we compare it against the uniform candidate and keep the better.
    """
    # -------------------------------------------------------------
    # 1. initialise RNG and deadline
    # -------------------------------------------------------------
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # -------------------------------------------------------------
    # 2. try to obtain a parent sequence (if the harness provides it)
    # -------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        # ensure all entries are finite – otherwise fall back
        if not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        # fallback: simple uniform vector of length 256
        parent = np.full(256, 1.0, dtype=np.float64)

    n = parent.size

    # -------------------------------------------------------------
    # 3. build the canonical uniform candidate (value 1.0)
    # -------------------------------------------------------------
    uniform = np.full(n, 1.0, dtype=np.float64)

    # -------------------------------------------------------------
    # 4. evaluate both candidates using the authoritative evaluator
    # -------------------------------------------------------------
    best = parent
    best_score = float(evaluate_sequence(best.tolist()))
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if np.isfinite(uniform_score) and uniform_score < best_score:
        best = uniform
        best_score = uniform_score

    # -------------------------------------------------------------
    # 5. (optional) quick random local search – exits early if time low
    # -------------------------------------------------------------
    def block_perturb(arr):
        """Multiplicative perturbation of a random contiguous block."""
        a = arr.copy()
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width
        factor = float(np.exp(rng.normal(0.0, 0.12)))
        a[start:stop] *= factor
        if rng.random() < 0.35:
            a = 0.5 * (a + a[::-1])
        # clip to range required by the evaluator
        np.clip(a, 0.0, 1000.0, out=a)
        return a

    # simple hill‑climb, but terminate well before the deadline
    while time.monotonic() < deadline - 0.1:
        cand = block_perturb(best)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # -------------------------------------------------------------
    return [float(x) for x in best]
