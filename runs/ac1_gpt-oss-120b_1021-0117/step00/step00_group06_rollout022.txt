import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Generate a geometric decay sequence that yields a very small AC1 upper bound.
    The length matches the provided `height_sequence_1` if available,
    otherwise falls back to 256.
    """
    # Determine the required length.
    try:
        n = int(np.asarray(height_sequence_1).size)
        if n == 0:
            n = 256
    except Exception:
        n = 256

    # Choose a ratio r that balances the endpoint square and the central
    # convolution value: n * r**(n-1) ≈ 1  ⇒  r = n**(-1/(n-1)).
    if n > 1:
        r = n ** (-1.0 / (n - 1))
    else:
        r = 0.5  # trivial fallback for n==1

    # Scale so the largest entry equals the evaluator's cap (1000).
    scale = 1000.0

    # Build the geometric progression a_i = scale * r**i.
    idx = np.arange(n, dtype=np.float64)
    seq = scale * np.exp(idx * np.log(r))

    # Enforce the evaluator's clipping bounds (mostly a safety net).
    seq = np.clip(seq, 0.0, 1000.0)

    # Return as a plain Python list of floats.
    return seq.tolist()
