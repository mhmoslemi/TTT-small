import numpy as np
import time

# ------------------------------------------------------------
# Helper: build a Sidon set (all unordered sums i+j are distinct)
# ------------------------------------------------------------
def sidon_from_order(order, n):
    """Greedy construction of a Sidon set from a given permutation of 0..n-1.
    Returns the list of chosen positions (ints)."""
    pos = []                 # chosen indices
    pos_set = set()
    used_sums = set()        # sums i+j already taken

    for x in order:
        x = int(x)
        # quick reject if self‑sum already present
        if 2 * x in used_sums:
            continue
        conflict = False
        for y in pos:
            if x + y in used_sums:
                conflict = True
                break
        if conflict:
            continue
        # accept x
        pos.append(x)
        pos_set.add(x)
        # register all new sums (including the square)
        for y in pos:
            used_sums.add(x + y)

    # try to extend greedily after the first pass
    while True:
        added = False
        for x in range(n):
            if x in pos_set:
                continue
            if 2 * x in used_sums:
                continue
            bad = False
            for y in pos:
                if x + y in used_sums:
                    bad = True
                    break
            if bad:
                continue
            # add this element
            pos.append(x)
            pos_set.add(x)
            for y in pos:
                used_sums.add(x + y)
            added = True
            break
        if not added:
            break
    return pos

# ------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Produce a sequence of non‑negative coefficients that (hopefully)
    lowers the AC1 upper bound.  The method builds a large Sidon set
    (distinct unordered sums) and puts unit weight on its positions.
    The resulting convolution has at most one unordered pair per
    index, giving a provably small max value.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # fallback parent (already clipped to the required range)
    if 'height_sequence_1' in globals():
        parent = np.clip(
            np.asarray(height_sequence_1, dtype=np.float64),
            0.0, 1000.0
        )
    else:
        parent = np.ones(256, dtype=np.float64)

    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    n = int(parent.size)

    # minimal size of a Sidon set needed to beat the target bound
    target = 1.50274349
    min_m_needed = int(np.ceil(np.sqrt(4.0 * n / target)))

    # main search loop – try many random orders until time runs out
    while time.monotonic() < deadline:
        order = rng.permutation(n)
        positions = sidon_from_order(order, n)
        m = len(positions)

        # ignore trivially small sets
        if m < min_m_needed:
            continue

        # build candidate: unit weight on the Sidon positions
        cand = np.zeros(n, dtype=np.float64)
        cand[positions] = 1.0

        # exact evaluation (fast for this sparse pattern)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score
            # early exit if we already beat the target
            if best_score <= target:
                break

    # return the best sequence found (as plain Python floats)
    return [float(x) for x in best_seq]
