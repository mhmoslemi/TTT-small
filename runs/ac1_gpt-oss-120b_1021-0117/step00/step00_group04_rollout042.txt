import numpy as np
import time

def _initial_exponential(n, c):
    """Return a simple exponential decay sequence a_i = exp(-c*i/n)."""
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-c * idx / n)
    return np.clip(a, 0.0, 1000.0)


def _score(seq):
    """Convenient wrapper for the authoritative evaluator."""
    return float(evaluate_sequence(seq.tolist()))


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a non‑negative step function with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # length of the problem instance (use the provided parent if present)
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256  # fallback length

    # ----------------------------------------------------------------------
    # 1.  Simple grid search over a few parametrised families (exponential,
    #     exponential with a polynomial prefactor, quadratic exponent).
    # ----------------------------------------------------------------------
    best = np.ones(n, dtype=np.float64)          # uniform baseline (score = 2)
    best_score = _score(best)

    idx = np.arange(n, dtype=np.float64)
    idx_norm = idx / n

    # (a) pure exponential a_i = exp(-c * i / n)
    for c in np.linspace(0.2, 2.0, 10):
        cand = np.exp(-c * idx_norm)
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if sc < best_score:
            best = cand
            best_score = sc

    # (b) exponential with power‑law prefactor a_i = (i+1)^p * exp(-c*i/n)
    p_vals = np.array([-2.0, -1.0, -0.5, 0.0, 0.5, 1.0, 2.0])
    c_vals = np.linspace(0.2, 2.0, 10)
    for c in c_vals:
        for p in p_vals:
            # avoid overflow for large positive p
            with np.errstate(over='ignore'):
                cand = (idx + 1.0) ** p * np.exp(-c * idx_norm)
            cand = np.clip(cand, 0.0, 1000.0)
            sc = _score(cand)
            if sc < best_score:
                best = cand
                best_score = sc

    # (c) quadratic exponent a_i = (i+1)^p * exp(- (c1*i/n + c2*(i/n)^2))
    c1_vals = np.linspace(0.0, 2.0, 5)   # 0,0.5,1,1.5,2
    c2_vals = np.linspace(0.0, 2.0, 5)
    for c1 in c1_vals:
        for c2 in c2_vals:
            for p in (0.0, -0.5, -1.0):
                expo = -(c1 * idx_norm + c2 * (idx_norm ** 2))
                with np.errstate(over='ignore'):
                    cand = (idx + 1.0) ** p * np.exp(expo)
                cand = np.clip(cand, 0.0, 1000.0)
                sc = _score(cand)
                if sc < best_score:
                    best = cand
                    best_score = sc

    # ----------------------------------------------------------------------
    # 2.  Local improvement: move a small amount of mass from the pair
    #     contributing most to the current maximal convolution entry.
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        # current convolution and its maximal index

        k_max = int(np.argmax(conv))

        # range of i that participates in this sum

        i_end = min(k_max, n - 1)

        j_range = k_max - i_range

        if products.size == 0:

        # choose a destination index with relatively small value

        if j_dest == i_best:
            # pick a different index if the minimum happened at the same place

            if candidates.size == 0:
                break
            j_dest = int(rng.choice(candidates))

        # transfer a modest fraction of mass
        delta = best[i_best] * float(np.exp(rng.normal(0.0, 0.05)) - 1.0)
        # keep the move small and keep entries non‑negative
        delta = np.clip(delta, -0.5 * best[i_best], 0.5 * best[i_best])
        if abs(delta) < 1e-12:

        cand = best.copy()
        cand[i_best] -= delta

        cand = np.clip(cand, 0.0, 1000.0)

            best_score = sc
