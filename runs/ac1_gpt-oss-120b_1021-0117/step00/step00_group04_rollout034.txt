def propose_candidate(seed=42, budget_s=1100, **kwargs):

   rng = np.random.default_rng(seed)
   deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

   # If height_sequence_1 exists and valid:

       if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
           n = parent.size
       else:
           n = 256
   else:
       n = 256

   # target sum:
   S = np.sqrt(2.0 * n)

   # baseline uniform:
   uniform_seq = np.full(n, S / n, dtype=np.float64)
   best_seq = uniform_seq.copy()
   best_score = float(evaluate_sequence(best_seq.tolist()))

   # scan exponential
   # Choose range of a (epsilon*n) maybe [0.1, 5]
   a_vals = np.linspace(0.1, 5.0, 200)  # a = λ * n
   for a in a_vals:
        if time.monotonic() >= deadline:
            break
        λ = a / n
        seq = np.exp(-λ * np.arange(n))
        # scaling:
        total = seq.sum()
        seq *= S / total
        # clip (though not needed)
        # compute score
        score = float(evaluate_sequence(seq.tolist()))
        if score < best_score:
            best_seq = seq
            best_score = score

   # scan power law
   alphas = np.linspace(0.0, 5.0, 51) # inclusive
   deltas = np.linspace(0.0, 10.0, 21)
   for α in alphas:
        for δ in deltas:
            if time.monotonic() >= deadline:
                break
            if α == 0.0:
                seq = np.ones(n, dtype=np.float64)
            else:
                base = (np.arange(n, dtype=np.float64) + δ) ** (-α)
                seq = base
            seq_sum = seq.sum()
            if seq_sum == 0:
                continue
            seq = seq * (S / seq_sum)
            score = float(evaluate_sequence(seq.tolist()))
            if score < best_score:
                best_seq = seq
                best_score = score

   # local refinement using random block perturbation
   # using previous make_block_perturbation but with renorm to S
   while time.monotonic() < deadline:
        # random block
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        cand = best_seq.copy()
        # apply multiplicative noise to block:
        noise = np.exp(rng.normal(0.0, 0.1, size=stop-start))
        cand[start:stop] *= noise
        # optional symmetrize
        if rng.random() < 0.3:
            cand = 0.5 * (cand + cand[::-1])
        # renormalize sum
        sum_c = cand.sum()
        if sum_c > 0:
            cand *= (S / sum_c)
        # clip
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score:
            best_seq = cand
            best_score = score

   # return as list of float
   return [float(x) for x in best_seq]

def _exponential_seq(n, a, target_sum):
    lam = a / n
    seq = np.exp(-lam * np.arange(n, dtype=np.float64))
    seq *= target_sum / seq.sum()
    return seq

        seq = np.ones(n, dtype=np.float64)

        base = (np.arange(n, dtype=np.float64) + delta) ** (-alpha)
