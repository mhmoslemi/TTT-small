            raise ValueError
    except Exception:
        # fallback length (any positive length works)
        parent = np.ones(256, dtype=np.float64)

    n = parent.shape[0]

    # ------------------------------------------------------------------ #
    # 2.  Analytic initial shape: a_i = 1 / sqrt(i+1)  (≈ optimal for s=0.5)
    # ------------------------------------------------------------------ #
    base = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    # keep inside the allowed range (already ≤1)
    candidate = base.copy()

    # ------------------------------------------------------------------ #
    # 3.  Choose the better of the analytic shape and the provided parent.
    # ------------------------------------------------------------------ #
    best = candidate.copy()
    best_score = _eval_score(best)
    parent_score = _eval_score(parent)
    if parent_score < best_score:
        best = parent.copy()
        best_score = parent_score

    # the working candidate will be mutated during the search
    candidate = best.copy()
    cur_score = best_score

    # ------------------------------------------------------------------ #
    # 4.  Local optimisation: move a little mass from the entry that
    #     contributes most to the current maximal convolution value to the
    #     entry that contributes the least.
    # ------------------------------------------------------------------ #
    max_step = 0.20   # 20 % of the source entry at start
    min_step = 0.001  # 0.1 % at the end
    start_time = time.monotonic()
    # temperature for occasional uphill moves
    init_temp = best_score * 0.10 if best_score != np.inf else 0.1

    while time.monotonic() < deadline:
        # convolution and location of the current peak
        conv = np.convolve(candidate, candidate)
        max_val = conv.max()
        k_max = int(conv.argmax())

        # contributions a_i * a_{k_max-i} (zero outside the admissible range)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_idx = k_max - i_idx
        contrib = np.zeros_like(candidate)
        contrib[i_idx] = candidate[i_idx] * candidate[j_idx]

        # pick the index that hurts the most and one that helps the most
        i_high = int(np.argmax(contrib))
        i_low = int(np.argmin(contrib))
        if i_low == i_high:
            # second‑best minimum if the absolute minimum happens to be the same
            sorted_idx = np.argsort(contrib)
            for idx in sorted_idx:
                if idx != i_high:
                    i_low = int(idx)
                    break

        # schedule a decreasing step size
        elapsed = time.monotonic() - start_time
        total = deadline - start_time
        if total <= 0:
            break
        prog = elapsed / total
        step_frac = max_step * (1.0 - prog) + min_step * prog

        # amount of mass to shift (preserve total sum)
        delta = candidate[i_high] * step_frac
        delta = min(delta, candidate[i_high])                # stay non‑negative
        delta = min(delta, 1000.0 - candidate[i_low])        # respect the upper bound
        if delta <= 0.0:
            continue

        new_cand = candidate.copy()
        new_cand[i_high] -= delta
        new_cand[i_low] += delta
        np.clip(new_cand, 0.0, 1000.0, out=new_cand)

        # fast recompute of the score
        new_score = _eval_score(new_cand)

        # simulated‑annealing acceptance
        temp = init_temp * (1.0 - prog)          # linear cooling to zero
        accept = False
        if new_score < cur_score:
            accept = True
        elif temp > 0.0:
            # uphill move with probability exp(-(Δ)/T)
            prob = np.exp(-(new_score - cur_score) / temp)
            if rng.random() < prob:
                accept = True

        if accept:
            candidate = new_cand
            cur_score = new_score
            if new_score < best_score:
                best = new_cand.copy()
                best_score = new_score

        # ------------------------------------------------------------------ #
        # 5.  Occasionally perform a random block perturbation to escape plateaus.
        # ------------------------------------------------------------------ #
        if rng.random() < 0.01:          # roughly one in a hundred iterations
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            factor = float(np.exp(rng.normal(0.0, 0.10)))
            candidate[start:stop] *= factor
            if rng.random() < 0.30:
                candidate[:] = 0.5 * (candidate + candidate[::-1])
            np.clip(candidate, 0.0, 1000.0, out=candidate)
            cur_score = _eval_score(candidate)
            if cur_score < best_score:
                best = candidate.copy()
                best_score = cur_score

    # ---------------------------------------------------------------------- #
    # 6.  Return the best sequence we have found (as a plain Python list).
    # ---------------------------------------------------------------------- #
    # final safety check – evaluate via the authoritative function
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        # should never happen; fall back to a simple uniform vector
        fallback = np.full(n, 1.0, dtype=np.float64)
        return fallback.tolist()
    return best.tolist()
