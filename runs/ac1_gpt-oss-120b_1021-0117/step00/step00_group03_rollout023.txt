def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 0.2
    # use initial candidate:
    if 'height_sequence_1' in globals():
        a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if a.ndim != 1 or a.size == 0 or not np.isfinite(a).all():
            a = np.full(256, 1.0, dtype=np.float64)
    else:
        n = 256
        a = np.full(n, 1.0, dtype=np.float64)
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    n = a.size
    # Precompute next power of two for FFT size
    fft_len = 1 << ((2*n-1).bit_length())

    def compute_conv(arr):
        # returns full convolution length 2n-1
        # using FFT for speed
        fa = np.fft.rfft(arr, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)
        return conv[:2*n-1]  # real part

    # gradient descent param schedule
    step = 0.1  # initial step multiplier for gradient
    # Maybe use decreasing step.
    while time.monotonic() < deadline:
        # compute convolution and max
        conv = compute_conv(a)
        # conv may have tiny imaginary rounding errors; use real part
        # get max and index
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]
        # evaluate score
        # Keep best
        cur_score = float(2 * n * max_val / (a.sum()**2))
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        # compute gradient
        S = a.sum()
        if S == 0:
            # handle degenerate
            break
        # compute shifted a for gradient: g_shift[j] = a[k_max - j] if 0 <= k_max - j < n else 0
        # Use slicing:
        start = max(0, max_idx - n + 1)
        end = min(max_idx, n - 1)
        # length = end - start + 1
        # fill zero initially
        a_shift = np.zeros_like(a)
        # positions start..end correspond to a_shift indices
        # indices for a: k_max - np.arange(start, end+1)
        a_shift[start:end+1] = a[max_idx - np.arange(start, end+1)]
        # gradient:
        const = (4.0 * n * max_val) / (S**3)
        grad = (4.0 * n / (S**2)) * a_shift - const
        # take a step
        # adapt step via random scaling maybe for diversity
        eta = step * (0.5 + 0.5 * rng.random())
        a_new = a - eta * grad
        # projection onto [0,1000]
        a_new = np.clip(a_new, 0.0, 1000.0)
        # maybe add small random perturbation to escape flat regions
        if rng.random() < 0.05:
            # random local bump
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            pos = int(rng.integers(0, max(1, n - width + 1)))
            a_new[pos:pos+width] *= np.exp(rng.normal(0.0, 0.2))

def solve_lp(f, max_val, active_idx):
    n = f.size
    g = cp.Variable(n, nonneg=True)
    constraints = []
    for k in active_idx:
        # compute coeff vector
        start = max(0, k - (n-1))
        end = min(k, n-1)
        # create coeff (numpy)
        coeff = np.zeros(n)
        # indices i from start to end inclusive
        # coefficient for g_i is f[k - i]
        idx_range = np.arange(start, end+1)
        coeff[idx_range] = f[k - idx_range]
        constraints.append(cp.sum(cp.multiply(coeff, g)) <= max_val)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    if prob.status in ["optimal", "optimal_inaccurate"]:
        return g.value
    else:
        return None

import time
import numpy as np

# optional linear programming, falls back to random search if unavailable
try:
    import cvxpy as cp
except Exception:  # pragma: no cover
    cp = None


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT (real, O(n log n))."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    # Numerical noise may produce tiny negatives
    conv[conv < 0] = 0.0
    return conv


def _solve_lp(f: np.ndarray, limit: float, active_idx) -> np.ndarray | None:
    """Linear program:
    maximize sum(g)  subject to   g >= 0  and  (f * g)[k] <= limit for k in active_idx.
    Returns a feasible optimal g or None on failure.
    """
    if cp is None:
        return None
    n = f.size
    g = cp.Variable(n, nonneg=True)
    cons = []
    for k in active_idx:
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        # coeff[j] = f[k - j] for j in [start, end]
        coeff = np.zeros(n, dtype=float)
        idx = np.arange(start, end + 1, dtype=int)
        coeff[idx] = f[k - idx]
        cons.append(cp.sum(cp.multiply(coeff, g)) <= limit)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), cons)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:  # pragma: no cover
        return None
    if prob.status in ("optimal", "optimal_inaccurate"):
        val = g.value
        if val is not None and np.all(np.isfinite(val)):
            return val.astype(float)
    return None


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """Search for a non‑negative step sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # initialise from the provided parent if possible
    if "height_sequence_1" in globals():
        a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if a.ndim != 1 or a.size == 0 or not np.isfinite(a).all():
            a = np.full(256, 1.0, dtype=np.float64)
    else:
        a = np.full(256, 1.0, dtype=np.float64)

    # a small random distortion helps escape the uniform stationary point
    a *= (0.8 + 0.4 * rng.random(a.shape))

    n = a.size
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # main loop: occasional LP improvement, otherwise random local tweak
    while time.monotonic() < deadline:
        conv = _full_convolution(a)
        max_val = float(conv.max())
        cur_score = 2.0 * n * max_val / (float(a.sum()) ** 2)

        # keep the best seen so far
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score

        # ----- attempt an LP step -------------------------------------------------
        # choose a handful of indices that are (almost) maximal
        K = 6
        if conv.size <= K:
            top_idx = np.arange(conv.size)
        else:
            top_idx = np.argpartition(-conv, K - 1)[:K]

        g0 = _solve_lp(a, max_val, top_idx)

        if g0 is not None:
            # rescale g0 to preserve the total sum of a (scale‑invariant objective)
            sum_a = a.sum()
            sum_g = g0.sum()
            if sum_g > 0:
                g = g0 * (sum_a / sum_g)
                # convex combination with a modest step size
                t = 0.25
                a = (1.0 - t) * a + t * g
                a = np.clip(a, 0.0, 1000.0)
                continue  # skip the random tweak this round

        # ----- fallback random block perturbation ---------------------------------
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        a[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
        # occasional mirroring to diversify the search
        if rng.random() < 0.3:
            a = 0.5 * (a + a[::-1])
        a = np.clip(a, 0.0, 1000.0)

    # final safety check – if everything went wrong, fall back to a uniform vector
    final = best
    if not np.isfinite(evaluate_sequence(final.tolist())):
        final = np.full_like(a, 1.0)

    return [float(x) for x in final]
