def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # Determine length
    try:
        n = int(len(height_sequence_1))
        if n == 0:
            raise Exception
        base = np.full(n, 1.0, dtype=np.float64)
    except Exception:
        n = 256
        base = np.full(n, 1.0, dtype=np.float64)
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    def perturb(vec):
        cand = vec.copy()
        width = rng.integers(max(1, n // 200), max(2, n // 12))
        start = rng.integers(0, max(1, n - width + 1))
        factor = np.exp(rng.normal(0.0, 0.12))
        cand[start:start+width] *= factor
        # maybe apply symmetry
        if rng.random() < 0.35:
            cand = 0.5 * (cand + cand[::-1])
        # clip
        np.clip(cand, 0.0, 1000.0, out=cand)
        return cand

    while time.monotonic() < deadline:
        cand = perturb(best)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score - 1e-12:
            best = cand
            best_score = score

    return [float(x) for x in best]
