import numpy as np, time

def compute_score(mask):
    n = mask.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(mask, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    max_b = int(np.rint(conv.max()))
    s = mask.sum()
    # s should be > 0
    if s == 0:
        return np.inf, max_b
    score = 2.0 * n * max_b / (s * s)
    return score, max_b

def _conv_max(mask, n, size):
    # compute convolution of mask with itself and return max integer count
    fa = np.fft.rfft(mask, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2 * n - 1]
    # rounding to nearest integer to avoid floating noise
    return int(np.rint(conv.max()))

import numpy as np
import time


def _conv_max(mask: np.ndarray, n: int, size_fft: int) -> int:
    """Return the maximum entry of mask * mask convolution (ordered pair count)."""
    # FFT‑based linear convolution, zero‑padded to at least 2·n‑1 points.
    fa = np.fft.rfft(mask, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
    # Round to nearest integer to avoid tiny floating‑point noise.
    return int(np.rint(conv.max()))


def _score_from_max(n: int, s: int, max_b: int) -> float:
    """Exact AC1 score for a binary mask of support size s."""
    if s == 0:
        return np.inf
    return 2.0 * n * max_b / (s * s)
