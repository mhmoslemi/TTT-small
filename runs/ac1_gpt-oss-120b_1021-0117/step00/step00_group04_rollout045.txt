import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Returns a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as evaluated by ``evaluate_sequence``.
    The algorithm starts from a uniform sequence (the provably optimal shape
    for non‑negative entries) and, if a parent sequence is present, keeps the
    better of the two.  A very cheap local search may improve the result
    within the remaining budget.
    """
    rng = np.random.default_rng(seed)

    # Time budget: leave a small margin for the final return.
    end_time = time.monotonic() + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # Determine problem size.
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            parent_arr = None
    except Exception:                     # height_sequence_1 may be undefined
        parent_arr = None

    n = parent_arr.size if parent_arr is not None else 256

    # ------------------------------------------------------------------
    # Baseline candidates: uniform sequence and (if present) the parent.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0, dtype=np.float64)          # sum = n > 0.01
    uniform = np.clip(uniform, 0.0, 1000.0)

    best_seq = uniform
    best_score = float(evaluate_sequence(best_seq.tolist()))

    if parent_arr is not None:
        parent_score = float(evaluate_sequence(parent_arr.tolist()))
        if parent_score < best_score:
            best_seq = parent_arr.copy()
            best_score = parent_score

    # ------------------------------------------------------------------
    # Very cheap stochastic local refinement (O(n) memory, O(n^2) time per eval).
    # ------------------------------------------------------------------
    while time.monotonic() < end_time:
        # Choose a random contiguous block (size between 1% and 8% of n)
        block_width = max(1, rng.integers(n // 100, max(2, n // 12)))

        factor = np.exp(rng.normal(0.0, 0.07))          # mild multiplicative perturbation

        cand = best_seq.copy()
        cand[start:start + block_width] *= factor

        # Occasionally enforce a mild symmetry (helps avoid pathological spikes)
        if rng.random() < 0.15:

        # Enforce the problem’s box constraints.
        cand = np.clip(cand, 0.0, 1000.0)

        # Evaluate – ``evaluate_sequence`` returns ``np.inf`` for invalid inputs.
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
