def generate_sidon_set(n, m_target, rng, max_trials=20):
    best = []
    for _ in range(max_trials):
        pos=[]
        used=set()

            collide=False
            s = p+p

                pos.append(p)
                if len(pos)>=m_target:
                    return pos
        # keep the best set if not reached target

def generate_sidon_set_greedy(n, rng, max_trials=5):
    best = []
    for _ in range(max_trials):
        pos = []
        used = set()
        order = rng.permutation(n)
        for p in order:
            # quick early out: if we already have many sums maybe we can break? Not needed.
            ok = True

                ok = False
            else:
                for q in pos:
                    if p + q in used:
                        ok = False
                        break
            if ok:
                for q in pos:
                    used.add(p + q)
                used.add(s)

def _make_block_perturbation(base, rng):
    """Small O(n) random modification of ``base``."""
    cand = np.array(base, copy=True)
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _greedy_sidon_set(length, rng, trials=5):
    """Return a list of indices forming a (large) Sidon set for ``length``."""
    best = []
    for _ in range(trials):
        cur = []
        used = set()
        order = rng.permutation(length)
        for p in order:
            # check whether any sum would collide
            collide = False
            s = p + p
            if s in used:
                collide = True
            else:
                for q in cur:
                    if p + q in used:
                        collide = True
                        break
            if collide:
                continue
            # accept p
            for q in cur:
                used.add(p + q)
            used.add(s)
            cur.append(p)
        if len(cur) > len(best):
            best = cur
        # early stop if we already hit a large size
        if len(best) >= int(math.sqrt(length)) + 20:
            break
    return best

        score = float(evaluate_sequence(cand.tolist()))
        if math.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score
            if best_score <= TARGET:
                break

    # ------------------------------------------------------------------
    # Return the best sequence discovered (always non‑empty and finite)
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
