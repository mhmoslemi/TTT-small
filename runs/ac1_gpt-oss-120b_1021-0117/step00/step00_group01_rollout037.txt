import time
import numpy as np

def _conv_max_norm(arr):
    """Return max convolution value of a non‑negative array whose sum is 1."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1
    fft_a = np.fft.rfft(arr, n=size)
    prod = fft_a * fft_a                     # a * a in frequency domain
    conv = np.fft.irfft(prod, n=size)[: 2 * n - 1]
    return float(conv.max())


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a step‑function that lowers the AC1 bound.

    The algorithm keeps a probability vector `a` (sum = 1) and performs a
    simulated‑annealing style random walk, evaluating the maximal entry of
    its self‑convolution using an FFT (O(n log n)).  The best vector found
    is finally scaled into the allowed [0,1000] range and returned as a
    Python list of floats.
    """
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # Length of the problem (use the supplied parent if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
    else:
        n = 256                     # a reasonable fallback length
    if n == 0:
        n = 256

    # ------------------------------------------------------------------
    # Helper to compute the exact evaluator score (used only for the
    # final candidate and for an optional fallback to the provided parent)
    # ------------------------------------------------------------------
    def exact_score(vec):
        return float(evaluate_sequence(vec.tolist()))

    # ------------------------------------------------------------------
    # Initial candidate – uniform distribution (score = 2.0)
    # ------------------------------------------------------------------
    a = np.full(n, 1.0 / n, dtype=np.float64)   # normalized, sum = 1
    cur_max = _conv_max_norm(a)
    cur_score = 2.0 * n * cur_max               # estimated score
    best_a = a.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # Try to use the supplied parent as a fallback (it may already be
    # good).  It is evaluated with the authoritative evaluator.
    # ------------------------------------------------------------------
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                         0.0, 1000.0)
        parent_score = exact_score(parent)
        if np.isfinite(parent_score) and parent_score < best_score:
            best_a = parent / parent.sum()                # normalised
            best_score = parent_score
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Simulated‑annealing search
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 3.0)

    # schedule parameters
    max_step = 0.5          # initial multiplicative log‑step size
    min_step = 0.001        # final step size
    init_temp = 0.08        # initial temperature for Metropolis acceptance
    final_temp = 0.0005     # final temperature

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        progress = elapsed / (deadline - start_time) if deadline > start_time else 1.0

        # linear schedules
        step = max_step - progress * (max_step - min_step)
        temp = init_temp - progress * (init_temp - final_temp)

        # propose a multiplicative perturbation (log‑normal)
        proposal = a * np.exp(step * rng.normal(size=n))
        # keep non‑negative, renormalise to sum = 1
        proposal = np.clip(proposal, 0.0, None)
        s = proposal.sum()
        if s == 0.0:
            continue
        proposal /= s

        # occasional symmetrisation (helps for the even‑length case)
        if rng.random() < 0.12:
            proposal = 0.5 * (proposal + proposal[::-1])

        # evaluate the proposal (fast estimate using FFT)
        prop_max = _conv_max_norm(proposal)
        prop_score = 2.0 * n * prop_max

        # Metropolis acceptance
        accept = False
        if prop_score < cur_score:
            accept = True
        else:
            # probability to accept a worse move
            if temp > 0 and rng.random() < np.exp((cur_score - prop_score) / temp):
                accept = True

        if accept:
            a = proposal
            cur_score = prop_score
            cur_max = prop_max
            if prop_score < best_score:
                best_a = proposal.copy()
                best_score = prop_score

    # ------------------------------------------------------------------
    # Build the final candidate from the best normalised vector.
    # Scaling by 1000 respects the evaluator's clipping limits.
    # ------------------------------------------------------------------
    final_vec = np.clip(best_a * 1000.0, 0.0, 1000.0)
    # Ensure a valid return even if something went wrong
    if final_vec.size == 0 or not np.isfinite(final_vec).all():
        # fallback to a simple uniform vector
        final_vec = np.full(n, 1000.0 / n, dtype=np.float64)

    # Final exact check (optional, does not affect the returned list)
    _ = exact_score(final_vec)

    return [float(x) for x in final_vec]
