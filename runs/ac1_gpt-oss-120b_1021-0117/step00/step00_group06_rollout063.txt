import numpy as np, time

TARGET = 1.50274349

def sidon_backtrack(m, max_val):
    S = []
    sums = set()
    def backtrack(prev):
        # prev is last chosen number; start searching from prev+1
        if len(S) == m:
            return True
        # compute remaining needed
        needed = m - len(S)
        # minimum possible next numbers: start = prev+1
        start = prev + 1
        # quick pruning: if start + needed -1 > max_val: not enough space
        if start + needed - 1 > max_val:
            return False
        for x in range(start, max_val+1):
            # test sums
            conflict = False
            if (x + x) in sums:
                continue
            for s in S:
                if (x + s) in sums:
                    conflict = True
                    break
            if conflict:
                continue
            # add
            added = []
            for s in S:
                sums.add(x + s)
                added.append(x + s)
            sums.add(x + x)
            added.append(x + x)
            S.append(x)
            if backtrack(x):
                return True
            # backtrack
            S.pop()
            for val in added:
                sums.remove(val)
        return False
    if backtrack(-1):
        return sorted(S)
    else:
        return None

import time
import numpy as np

# ------------------------------------------------------------
# Helper: fast bound for a Sidon‐type 0/1 sequence
# ------------------------------------------------------------
def sidon_bound(num_len, set_size):
    """Upper bound 2*n*max_b/(sum^2) for a 0/1 Sidon sequence.
    For a Sidon set with all entries equal to 1 we have:
        max_b = 2, sum = set_size, n = num_len
    Hence bound = 4 * n / set_size**2
    """
    return 4.0 * num_len / (set_size * set_size)


# ------------------------------------------------------------
# Random greedy construction of a Sidon set (B2 set)
# ------------------------------------------------------------
def random_sidon(m, max_val, rng):
    """Try to build a Sidon set of size m inside [0, max_val].
    Returns a sorted list of positions or None on failure."""
    S = []
    sums = set()
    # random ordering of the whole interval
    order = rng.permutation(max_val + 1)
    for x in order:
        # check self‑sum
        if (x + x) in sums:
            continue
        # check sums with already chosen elements
        ok = True
        for s in S:
            if (x + s) in sums:
                ok = False
                break
        if not ok:
            continue
        # add x to the set
        for s in S:
            sums.add(x + s)
        sums.add(x + x)
        S.append(int(x))
        if len(S) == m:
            return sorted(S)
    return None
