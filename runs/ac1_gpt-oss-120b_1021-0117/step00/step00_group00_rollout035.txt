def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # length
    parent = np.asarray(height_sequence_1, dtype=np.float64) if 'height_sequence_1' in globals() else np.ones(256, dtype=np.float64)
    n = parent.size

    # helper: compute score (objective)
    def compute_score(a):
        # sum might not be exactly 1 but we compute objective exact using evaluate_sequence for safety.
        # For speed compute manually.
        m = 1 << ((2*n - 1).bit_length())
        fft = np.fft.rfft(a, n=m)
        conv = np.fft.irfft(fft*fft, n=m)[:2*n-1]
        max_c = np.max(conv)
        s = np.sum(a)
        if s < 0.01:
            return float('inf')
        return float(2 * n * max_c / (s*s))

    # generate multiple initial candidates
    candidates = []
    # uniform
    uniform = np.full(n, 1.0/n, dtype=np.float64)
    candidates.append(uniform)
    # linear decreasing
    lin = np.arange(n, 0, -1, dtype=np.float64) # n, n-1,...,1
    lin = lin / lin.sum()
    candidates.append(lin)
    # quadratic decreasing
    quad = (np.arange(n, 0, -1, dtype=np.float64))**2
    quad = quad / quad.sum()
    candidates.append(quad)
    # exponential decreasing
    exp = np.exp(-np.linspace(0, 5, n))
    exp = exp / exp.sum()
    candidates.append(exp)

    # also some random Dirichlet
    for _ in range(5):
        rand = rng.exponential(scale=1.0, size=n)
        rand = rand / rand.sum()
        candidates.append(rand)

    best = None
    best_score = float('inf')
    for cand in candidates:
        score = compute_score(cand)
        if score < best_score:
            best_score = score
            best = cand.copy()
    # local refinement loop
    while time.monotonic() < deadline:
        # compute convolution and locate max index
        m = 1 << ((2*n - 1).bit_length())
        fft = np.fft.rfft(best, n=m)
        conv = np.fft.irfft(fft*fft, n=m)[:2*n-1]
        max_c = np.max(conv)
        k_max = int(np.argmax(conv))

        # compute contributions vector
        # contributions only for i where 0 <= k_max - i < n
        # we can compute as v[i] = best[i] * best[k_max - i] for valid i.
        # Use slicing.
        i_start = max(0, k_max - (n-1))
        i_end = min(n-1, k_max)  # inclusive
        # create slice of a[i_start:i_end+1] and its counterpart reversed.
        slice_len = i_end - i_start + 1
        a_slice = best[i_start:i_end+1]
        b_slice = best[k_max - i_end : k_max - i_start + 1]  # reversed index?
        # Actually we want a[i] * a[k_max - i]; as i varies from i_start..i_end.
        # So compute product = a_slice * best[k_max - i_slice] ; we can compute via vectorized indexing.
        # Use np.take.
        idx = k_max - np.arange(i_start, i_end+1)
        contrib = a_slice * best[idx]
        # choose i with highest contrib
        argmax_local = np.argmax(contrib)
        i_max = i_start + argmax_local
        j_partner = k_max - i_max

        # Choose a low-index to receive mass: low value index
        # Find index with smallest a (maybe random among low values)
        low_candidates = np.where(best <= best.mean())[0]
        if low_candidates.size == 0:
            low_candidates = np.arange(n)
        j_target = int(rng.choice(low_candidates))

        # Determine epsilon to move: a fraction of a[i_max] and a[j_partner]
        # Ensure we move a small amount.
        eps = min(best[i_max], best[j_partner]) * rng.uniform(0.01, 0.05)
        # If partner equals i_max (i.e., i_max == j_partner), we treat single.
        new = best.copy()
        if i_max == j_partner:
            # reduce by 2*eps? We'll just reduce half epsilon from this index and add to target.
            delta = eps
            new[i_max] = max(0.0, new[i_max] - delta)
            new[j_target] = new[j_target] + delta
        else:
            delta = eps / 2.0
            new[i_max] = max(0.0, new[i_max] - delta)
            new[j_partner] = max(0.0, new[j_partner] - delta)
            # distribute equally to two random low indices (maybe same as j_target)
            # Choose two low indices:
            low2 = rng.choice(low_candidates, size=2, replace=False) if low_candidates.size >= 2 else np.array([j_target, j_target])
            new[low2[0]] += delta
            new[low2[1]] += delta

        # renormalize to keep sum 1 (or preserve sum):
        s = new.sum()
        if s > 0:
            new = new / s
        else:
            continue

        # evaluate
        new_score = compute_score(new)
        if new_score < best_score:
            best = new
            best_score = new_score
        # else maybe random move acceptance? ignore.

    # return best as list of floats
    return [float(x) for x in best]

    # compute convolution
    fft = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fft*fft, n=m)[:2*n-1]
    max_c = conv.max()
    k = int(np.argmax(conv))
    # compute contributions
    i_vals = np.arange(n)
    partner = k - i_vals
    mask = (partner >= 0) & (partner < n)
    contrib = np.zeros(n, dtype=a.dtype)
    contrib[mask] = a[i_vals[mask]] * a[partner[mask]]
    i_max = int(np.argmax(contrib))
    j = k - i_max
    # choose epsilon
    eps = min(a[i_max], a[j]) * 0.5  # maybe 0.5 fraction? but ensure not too large
    if eps <= 0:
        return a, False
    # create new array
    new = a.copy()
    # reduce both
    if i_max == j:
        # same index; reduce by eps
        new[i_max] -= eps
    else:
        new[i_max] -= eps/2
        new[j]    -= eps/2
    # find two low indices to add mass
    # low indices where product with its partner is minimal
    low_mask = (contrib < contrib.mean())

        low_idx = rng.choice(np.where(low_mask)[0], size=2, replace=False)
    new[low_idx[0]] += eps/2
    new[low_idx[1]] += eps/2

    new_sum = new.sum()
    if new_sum <= 0:
        return a, False
    new = new / new_sum
    # compute new score
    new_score = compute_score(new)
    old_score = compute_score(a) # could compute earlier but recall

import numpy as np
import time

def _conv_max(vec: np.ndarray, fft_len: int) -> float:
    """Maximum coefficient of self‑convolution using an FFT of length fft_len."""
    f = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * len(vec) - 1]
    return float(conv.max())


def _score(vec: np.ndarray, fft_len: int) -> float:
    """Exact objective used by the evaluator (avoid the authority call)."""
    s = float(vec.sum())
    if s < 0.01:
        return float("inf")
    m = _conv_max(vec, fft_len)
    return float(2.0 * len(vec) * m / (s * s))


def propose_candidate(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a short non‑negative step function with a low AC1 upper bound.
    The routine keeps a best solution and returns it before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ----- base length -------------------------------------------------
    parent = np.asarray(
        globals().get("height_sequence_1", np.ones(256, dtype=np.float64)),
        dtype=np.float64,
    )
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)
    n = parent.size

    # ----- FFT size (power of two) ------------------------------------
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ----- generate a few deterministic seeds -------------------------
    cand_list = []

    # uniform
    cand_list.append(np.full(n, 1.0 / n, dtype=np.float64))

    # linear decreasing (n ... 1)
    lin = np.arange(n, 0, -1, dtype=np.float64)
    cand_list.append(lin / lin.sum())

    # quadratic decreasing
    quad = (np.arange(n, 0, -1, dtype=np.float64)) ** 2
    cand_list.append(quad / quad.sum())

    # exponential tail
    exp = np.exp(-np.linspace(0.0, 7.0, n))
    cand_list.append(exp / exp.sum())

    # a few random Dirichlet draws
    for _ in range(6):
        tmp = rng.exponential(scale=1.0, size=n)
        cand_list.append(tmp / tmp.sum())

    # ----- pick the best starter ---------------------------------------
    best = None
    best_sc = float("inf")
    for c in cand_list:
        sc = _score(c, fft_len)
        if sc < best_sc:
            best_sc = sc
            best = c.copy()

    # ----- local improvement loop --------------------------------------
    # move mass from the most harmful pair (i, j) to low‑impact positions
    while time.monotonic() < deadline:

        f = np.fft.rfft(best, n=fft_len)
