    """Return (max value, index of max) of the full convolution a*a."""
    b = np.convolve(a, a)          # length 2*n-1, dtype float64

        if rng.random() < 0.6:                     # mostly gradient steps
            max_b, kmax = _conv_max(best)
            rev = _rev_slice(best, kmax)           # a_{kmax-i}
            # gradient of the objective (see analysis)
            grad = 4.0 * n * (rev - max_b)
            norm = np.linalg.norm(grad) + 1e-12
            eta = step_factor / norm
            cand = best - eta * grad
            cand = np.clip(cand, 0.0, None)
            s = cand.sum()
            if s > 0:
                cand /= s
        else:
            # ----- random block perturbation ---------------------------------
            cand = block_perturb(best)

        # occasional random re‑shuffle (helps break structure)
        if rng.random() < 0.02:
            perm = rng.permutation(n)
            cand = cand[perm]

        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc
            no_improve = 0
            # make the gradient steps a little larger after a win
            step_factor = min(step_factor * 1.15, 0.05)
        else:
            no_improve += 1
            # shrink step size if we are stuck
            if no_improve > 40:
                step_factor = max(step_factor * 0.7, 0.005)
                no_improve = 0

    # final sanity – ensure non‑negative and within allowed bounds
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
