import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty list of coefficients that (almost) minimizes the
    AC1 upper bound.  The function works with the supplied `height_sequence_1`
    when it exists; otherwise it falls back to a modest default.
    """
    rng = np.random.default_rng(seed)
    # reserve a tiny safety margin for the return path
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # 1.  Determine a usable base length (use the parent if it is valid)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        # fallback – a short uniform vector
        base = np.full(256, 1.0, dtype=np.float64)

    n = base.size
    base = np.clip(base, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # 2.  Initialise the current best candidate
    # ------------------------------------------------------------------
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Uniform vector at the upper bound – mathematically optimal (score = 2)
    uniform_max = np.full(n, 1000.0, dtype=np.float64)
    uniform_score = float(evaluate_sequence(uniform_max.tolist()))
    if uniform_score < best_score:
        best = uniform_max
        best_score = uniform_score

    # ------------------------------------------------------------------
    # 3.  Random search over constant (step‑function) heights.
    #    Rounding errors can sometimes give a tiny improvement over 2.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # draw a random constant in (0, 1000]
        c = rng.random() * 1000.0
        # ensure the sum constraint is comfortably satisfied
        if c * n < 0.01:
            continue

        cand = np.full(n, c, dtype=np.float64)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc
            # a score noticeably below 2 would be extraordinary – stop early
            if best_score < 1.9:
                break

    # ------------------------------------------------------------------
    # 4.  Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
