    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())  # next power of two >= 2n-1
    A = np.fft.rfft(a, n=size)
    B = A * A

    return b

def _self_convolve(a: np.ndarray, fft_len: int, out_len: int) -> np.ndarray:
    """Return np.convolve(a, a) using a power‑of‑two FFT length."""
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    full = np.fft.irfft(B, n=fft_len)
    return full[:out_len]


# ------------------------------------------------------------
# Main search routine
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative coefficients that (heuristically) lower
    the AC1 upper bound.  The algorithm starts from a random vector,
    repeatedly flattens the peak of its self‑convolution, and keeps the
    best candidate seen before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Obtain a fallback parent sequence (provided as height_sequence_1)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = base.size
        parent = np.clip(base, 0.0, 1000.0)
    except Exception:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Initial random candidate (scaled to a moderate total sum)
    # ------------------------------------------------------------------
    a = rng.random(n) + 1e-9               # strictly positive
    a = a / a.sum() * np.sqrt(2.0 * n)      # keep values O(1)

    # ------------------------------------------------------------------
    # Pre‑compute sizes for the FFT‑based convolution
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len).bit_length())   # next power of two

    # ------------------------------------------------------------------
    # Helper to compute the exact AC1 score from a candidate array
    # ------------------------------------------------------------------
    def _score(vec: np.ndarray) -> float:
        b = _self_convolve(vec, fft_len, conv_len)
        return 2.0 * n * b.max() / (vec.sum() ** 2)

    # ------------------------------------------------------------------
    # Record the best candidate seen so far (fallback is the provided parent)
    # ------------------------------------------------------------------
    best_vec = parent.copy()
    best_score = float(evaluate_sequence(best_vec.tolist()))

    # ------------------------------------------------------------------
    # First evaluate the random start – it might already be better
    # ------------------------------------------------------------------
    cur_score = _score(a)
    if cur_score < best_score:
        best_score = cur_score
        best_vec = a.copy()

    # ------------------------------------------------------------------
    # Main optimisation loop: flatten the highest convolution peak
    # ------------------------------------------------------------------
    step_factor = 0.20          # fraction of a[i] moved each step
    it = 0
    while time.monotonic() < deadline:
        it += 1

        # current convolution and the index of its maximal entry
        b = _self_convolve(a, fft_len, conv_len)
        k_max = int(np.argmax(b))

        # contribution of each index to this peak (double‑count off‑diagonal)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k_max - idx
        coeff = np.where(idx != j, 2.0, 1.0)
        contrib = a[idx] * a[j] * coeff

        # pick a high‑contributing and a low‑contributing index
        high_local = int(np.argmax(contrib))
        low_local = int(np.argmin(contrib))
        i_high = idx[high_local]
        i_low = idx[low_local]
        if i_high == i_low:
            # fallback to any two distinct indices
            i_high = idx[high_local]
            i_low = idx[(low_local + 1) % len(idx)]

        # move a modest amount of mass from the high to the low contributor
        delta = a[i_high] * step_factor
        if delta <= 0.0:
            delta = 1e-6
        a[i_high] -= delta
        a[i_low] += delta

        # enforce the problem bounds
        np.clip(a, 0.0, 1000.0, out=a)

        # re‑evaluate (fast version) and keep the best
        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # occasional random “shake’’ to escape plateaus
        if it % 200 == 0:
            w = rng.integers(1, max(2, n // 50))
            s = rng.integers(0, n - w + 1)
            factor = np.exp(rng.normal(0.0, 0.12))
            a[s:s + w] *= factor
            np.clip(a, 0.0, 1000.0, out=a)
            # recompute after perturbation
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_vec = a.copy()

        # slowly shrink the step size
        if it % 500 == 0:
            step_factor = max(step_factor * 0.90, 0.005)

    # ------------------------------------------------------------------
    # Final safety check: guarantee that the returned vector is accepted
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_vec.tolist())
    if not np.isfinite(final_score):
        best_vec = parent

    return [float(v) for v in best_vec]
