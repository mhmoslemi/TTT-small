        n = 256
    # Uniform sequence
    seq = np.full(n, 1.0, dtype=np.float64)
    # Ensure within limits (clipped already)
    # Optional improvement: if we have the parent, we can compute its score and compare; choose better.
    # Compute parent fallback just in case it's better
    best_seq = seq
    best_score = float(evaluate_sequence(best_seq.tolist()))
    # also evaluate parent if valid
    try:
        parent_seq = np.clip(parent_arr, 0.0, 1000.0)
        parent_score = float(evaluate_sequence(parent_seq.tolist()))
        if parent_score < best_score:
            best_seq = parent_seq
            best_score = parent_score
    except Exception:
        pass
    return [float(x) for x in best_seq]
