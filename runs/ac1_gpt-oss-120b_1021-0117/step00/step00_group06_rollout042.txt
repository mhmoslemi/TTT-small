import numpy as np
import time

def _make_block_perturbation(base, rng):
    """Random multiplicative perturbation of a contiguous block of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # block width between n/200 and n/12 (inclusive)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    # multiplicative log‑normal factor
    factor = np.exp(rng.normal(0.0, 0.12))
    cand[start:stop] *= factor
    # optional mirror‑symmetrisation (helps for some shapes)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a non‑negative sequence that tries to minimise the AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # initialise parent (fallback to a uniform vector if unavailable)
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise Exception
    except Exception:
        parent_arr = np.ones(256, dtype=np.float64)
    parent = np.clip(parent_arr, 0.0, 1000.0)
    n = parent.size

    # ------------------------------------------------------------------
    # generate a few analytical candidates (exponential, linear, power‑law,
    # Gaussian‑like).  These are inexpensive and provide a good starting
    # point that is typically better than the raw parent.
    # ------------------------------------------------------------------
    candidates = []

    # 1) exponential decay a_i = r**i, with r = 1 - c/n    (c≈1.0…2.0)
    for c in (0.8, 1.0, 1.2, 1.4, 1.6, 2.0):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        seq = np.power(r, np.arange(n, dtype=np.float64))
        candidates.append(seq)
        candidates.append(seq[::-1])           # reversed version

    # 2) linear decay from 1 to 0
    lin = np.linspace(1.0, 0.0, n, dtype=np.float64)
    candidates.append(lin)
    candidates.append(lin[::-1])

    # 3) power‑law a_i = (i+1)^{-p}
    for p in (0.5, 0.8, 1.0, 1.2):
        idx = np.arange(1, n + 1, dtype=np.float64)
        seq = np.power(idx, -p)
        candidates.append(seq / seq.max())    # normalise peak to 1
        candidates.append(seq[::-1] / seq.max())

    # 4) truncated Gaussian centred in the middle
    centre = (n - 1) / 2.0
    for sigma_factor in (0.25, 0.33, 0.40):   # sigma = factor * n
        sigma = sigma_factor * n
        dist = np.arange(n, dtype=np.float64) - centre
        seq = np.exp(-0.5 * (dist / sigma) ** 2)
        candidates.append(seq / seq.max())
        candidates.append(seq[::-1] / seq.max())

    # evaluate all deterministic candidates, keep the best
    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    for seq in candidates:
        seq = np.clip(seq, 0.0, 1000.0)
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = seq
            best_score = score

    # ------------------------------------------------------------------
    # Random local search: multiplicative block perturbations.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        cand = _make_block_perturbation(best_seq, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score

    # ------------------------------------------------------------------
    # Return the best sequence found as a plain Python list.
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
