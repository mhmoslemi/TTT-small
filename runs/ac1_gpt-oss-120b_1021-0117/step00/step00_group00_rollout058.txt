def ac1_score(a):
    # a: np.ndarray
    c = np.fft.rfft(a, n=next_pow2(2*n-1))
    conv = np.fft.irfft(c * c, n=next_pow2(2*n-1))
    conv = conv[:2*n-1]  # original length
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2 * len(a) * max_b / (sum_a**2)

import numpy as np
import time, math

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _gen_binomial(n: int) -> np.ndarray:
    """
    Generate the sequence a_i = C(2i,i) / 4**i for i=0..n-1.
    Recurrence: a_0 = 1, a_i = a_{i-1} * (2i-1)/(2i).
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def _ac1_score(arr: np.ndarray, n: int, L: int) -> float:
    """
    Fast AC1 upper bound using FFT convolution.
    Returns 2*n*max(conv)/sum(arr)^2 .
    """
    # FFT of the zero‑padded array
    fa = np.fft.rfft(arr, n=L)
    # convolution of arr with itself (linear, zero‑padded)
    conv = np.fft.irfft(fa * fa, n=L)
    conv = conv[: 2 * n - 1]                # keep the relevant part
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a non‑negative step function that minimizes the AC1 bound.
    Starts from the (near‑optimal) binomial construction and performs
    fast stochastic hill‑climbing using FFT‑based evaluation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # length of the problem (use the provided parent if available)
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        n = 256                     # fallback size
        parent = None

    # FFT size (next power of two for length 2*n-1)
    L = _next_pow2(2 * n - 1)

    # ---- initial candidate -------------------------------------------------
    base = _gen_binomial(n)         # the classic binomial (C(2i,i)/4^i)
    best = base.copy()
    best_score = _ac1_score(best, n, L)

    # if a parent construction is present, keep it as a secondary fallback
    if parent is not None:
        parent_score = _ac1_score(parent, n, L)
        if parent_score < best_score:
            best = parent.copy()
            best_score = parent_score

    # current state for the stochastic walk
    cur = best.copy()
    cur_score = best_score

    # annealing temperature (purely for occasional uphill moves)
    temperature = 1.0
    decay = 0.9995

    # ---- search loop -------------------------------------------------------
    while time.monotonic() < deadline:
        # choose a random move type
        if rng.random() < 0.6:                     # block scaling
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = math.exp(rng.normal(0.0, 0.12))
            cand = cur.copy()
            cand[start:start + width] *= factor
        else:                                      # transfer mass between two positions
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            frac = rng.random() * 0.4               # up to 40 % of source moved
            delta = cur[i] * frac
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

        # enforce value bounds required by the official evaluator
        np.clip(cand, 0.0, 1000.0, out=cand)

        # fast evaluation
        cand_score = _ac1_score(cand, n, L)

        # keep the best ever seen
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            # early exit if we already beat the target bound
            if best_score < 1.50274349:

        # simple simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True

            if rng.random() < prob:
                accept = True
        if accept:
            cur = cand
            cur_score = cand_score

        # temperature schedule
        temperature *= decay

        # occasional restart from the best to avoid drifting too far
        if rng.random() < 0.001:
            cur = best.copy()
            cur_score = best_score
            temperature = 1.0

    # Return the best sequence discovered (as plain Python floats)
    return [float(x) for x in best]
