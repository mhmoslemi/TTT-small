def project_simplex(v):
    if v.sum() == 0:
        return np.ones_like(v)/len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def project_simplex(v):
    # Input v: 1-D array
    # Return projection onto simplex {w >=0, sum w = 1}
    # Using algorithm from https://arxiv.org/abs/1101.6081
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non positive, return uniform
        return np.full_like(v, 1.0/n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import time
import numpy as np

# ------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of v onto {w >= 0, sum w = 1}."""
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative -> uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w
