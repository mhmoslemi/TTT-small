def gradient_step(seq, rng, temperature):
    n = len(seq)
    b = np.convolve(seq, seq)
    k_max = int(np.argmax(b))
    i_start = max(0, k_max - (n-1))
    i_end = min(n-1, k_max)
    indices = np.arange(i_start, i_end+1)
    partner = k_max - indices
    products = seq[indices] * seq[partner]   # maybe same shape
    i_max_rel = int(np.argmax(products))
    i_max = indices[i_max_rel]
    # choose j with low contribution to b: maybe random index not in [i_start,i_end]
    # choose j far from i_max maybe?
    # For simplicity, select j = rng.integers(0, n)
    # If j == i_max, re-select.
    j = int(rng.integers(0, n-1))
    while j == i_max:
        j = int(rng.integers(0, n-1))
    # epsilon relative to seq[i_max] and maybe seq[j]'s capacity
    if seq[i_max] <= 0:
        return seq, None
    eps = seq[i_max] * 0.05   # 5% move
    # also consider not overfill j beyond 1000; but scaling not needed, we can clip after.
    new_seq = seq.copy()

    selected = []

    for i in range(n):

        # build constraint matrix
        rows = []
        for k in top_T:
            coeff = np.zeros(n)
            # i runs where 0<=i<n and 0<=k-i<n -> j = k-i
            # j index for g: j = k - i
            # So for each i in intersect range, coeff[j] = seq[i]
            i_start = max(0, k - (n-1))
            i_end = min(k, n-1)

            i_vals = np.arange(i_start, i_end+1)
            j_vals = k - i_vals
            coeff[j_vals] = seq[i_vals]

        g = cp.Variable(n, nonneg=True)

        objective = cp.Maximize(cp.sum(g))
        prob = cp.Problem(objective, constraints)
        result = prob.solve(solver=cp.ECOS, warm_start=True)
        if g.value is None:
            break
        g_vec = np.array(g.value).flatten()
        # scale g_vec to have same sum as seq

        # convex mix

        new_seq = (1 - t) * seq + t * g_scaled
        # clip

        if new_score < best_score:

            best_score = new_score
            seq = new_seq

import cvxpy as cp

def lp_optimize(f, M, ks):
    n = len(f)
    rows = []
    for k in ks:
        coeff = np.zeros(n)
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        i_vals = np.arange(i_start, i_end+1)
        j_vals = k - i_vals
        coeff[j_vals] = f[i_vals]  # assign
        rows.append(coeff)
    A = np.vstack(rows)  # shape T x n
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500)
    if g.value is None:
        return None
    return np.array(g.value).flatten()

def gradient_step(seq, lr=0.1):
    b = np.convolve(seq, seq)
    k_max = int(np.argmax(b))
    i_start = max(0, k_max-(n-1))
    i_end = min(n-1, k_max)
    # partners:
    i_range = np.arange(i_start, i_end+1)
    partner = k_max - i_range
    grads = seq[i_range] * seq[partner]  # product?
    # Use gradient = seq[partner] (since d product w.r.t a_i = a_partner)
    grad = np.zeros_like(seq)
    grad[i_range] = seq[partner]
    # step
    new_seq = seq - lr * grad
    new_seq = np.maximum(new_seq, 0)
    # renorm to same sum
    s = new_seq.sum()
    if s > 0:
        new_seq *= seq.sum() / s
    return new_seq

def _evaluate(seq):
    """Wrapper for the provided evaluator."""
    return float(evaluate_sequence(seq.tolist()))


def _make_random_block(seq, rng):
    """Random local block perturbation (same idea as before)."""
    cand = np.array(seq, copy=True)
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _gradient_step(seq, lr):
    """One step aiming to reduce the current maximal convolution value."""
    n = seq.size
    b = np.convolve(seq, seq)
    kmax = int(np.argmax(b))
    i_start = max(0, kmax - (n - 1))
    i_end = min(n - 1, kmax)
    i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
    j_vals = kmax - i_vals
    # gradient = partner values (derivative of a_i * a_j w.r.t a_i)
    grad = np.zeros_like(seq)
    grad[i_vals] = seq[j_vals]
    new = seq - lr * grad
    np.clip(new, 0.0, None, out=new)          # keep non‑negative
    s = new.sum()
    if s > 0.0:
        new *= seq.sum() / s                 # renormalise to original sum
    return new


def _lp_step(f_seq, M, ks):
    """
    Solve the tiny LP:
        maximise   sum(g)
        subject to conv(f_seq, g)[k] <= M  for k in ks
                   g >= 0
    Returns the optimal g (numpy array) or None on failure.
    """
    n = f_seq.size
    T = ks.size
    # Build constraint matrix A (T x n)
    A = np.zeros((T, n), dtype=np.float64)
    for row, k in enumerate(ks):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        A[row, j_vals] = f_seq[i_vals]

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500)
    except Exception:               # solver failure guard
        return None
    if g.value is None:
        return None
    return np.asarray(g.value).flatten()
