        start = np.ones(256, dtype=np.float64)
    n = start.size
    # Normalize to sum=1 to keep numbers modest
    start = start.clip(0.0, 1000.0)
    # scale to sum 1 (or any positive)
    total = start.sum()
    if total < 1e-6:
        start = np.full(n, 1.0 / n)
    else:
        start = start / total   # sum=1

    # define convolution helpers
    conv_len = 2 * n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    # preallocate workspace?
    # but we'll compute each time.
    def score_of(x):
        s = x.sum()
        if s < 0.01:
            return float('inf')
        fa = np.fft.rfft(x, fft_size)
        conv = np.fft.irfft(fa * fa, fft_size)[:conv_len]
        maxb = float(conv.max())
        return 2.0 * n * maxb / (s * s)

    best = start.copy()
    best_score = score_of(best)
    # also keep original parent (maybe better)
    # but everything scaling invariant, we can just use best.

    #  simple hill climbing
    # Set parameters
    while time.monotonic() < deadline:
        cand = best.copy()
        # make several random transfers
        num_swaps = int(rng.integers(1, 5))
        for _ in range(num_swaps):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n-1))
            if j >= i:
                j += 1
            # choose delta up to 30% of source value
            if cand[i] <= 0.0:
                continue
            max_delta = cand[i] * rng.uniform(0.0, 0.3)
            delta = rng.uniform(0.0, max_delta)
            cand[i] -= delta
            cand[j] += delta
        # clip boundaries
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = score_of(cand)
        if sc < best_score:
            best = cand
            best_score = sc
            # optional early exit if very good
            if best_score < 1.6:  # target near 1.5
                # continue to try further improvement
                pass

        # occasional random restart

            # generate new random vector with sum=1
            cand = rng.random(n)
            cand = cand / cand.sum()
            sc = score_of(cand)
            if sc < best_score:
                best = cand
                best_score = sc

    return [float(x) for x in best]

import time
import numpy as np


def _conv_max(vec: np.ndarray, fft_size: int, conv_len: int) -> float:
    """Maximum of the full convolution vec * vec using FFT (matrix‑free)."""
    # FFT of the vector padded to fft_size
    fv = np.fft.rfft(vec, fft_size)
    # Inverse FFT of the pointwise product gives the convolution
    conv = np.fft.irfft(fv * fv, fft_size)[:conv_len]
    return float(conv.max())


def _score(vec: np.ndarray, n: int, fft_size: int, conv_len: int) -> float:

    s = vec.sum()
    if s < 0.01:
        return float('inf')
    max_b = _conv_max(vec, fft_size, conv_len)
    return 2.0 * n * max_b / (s * s)
