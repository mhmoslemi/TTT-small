def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0) # leave margin
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 1024
    # initial vector: Dirichlet distribution (Gamma)
    a = rng.gamma(1.0, size=n)
    a = a / a.sum()
    # compute max via FFT
    L = 1 << ((2*n - 1).bit_length())
    def max_conv(v):
        A = np.fft.rfft(v, n=L)
        b = np.fft.irfft(A*A, n=L)[:2*n-1]
        return b.max()
    best_a = a.copy()
    best_max = max_conv(a)
    # main loop
    step_scale = 0.1
    while time.monotonic() < deadline:
        # pick i with large a, j with small a
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # transfer amount
        delta = step_scale * a[i] * rng.random()
        if delta > a[i]:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure nonnegative (already)
        # compute max
        cur_max = max_conv(a_new)
        # accept if improvement
        if cur_max < best_max:
            best_max = cur_max
            best_a = a_new.copy()
        # simple simulated annealing acceptance
        if cur_max < max_conv(a) or rng.random() < 0.001:
            a = a_new
        # occasionally random restart or perturb large step
        if rng.random() < 0.001:
            # random perturb all values a bit
            perturb = rng.normal(0.0, 0.05, size=n)
            a = np.clip(a + perturb, 0, None)
            if a.sum() == 0:
                a = rng.gamma(1.0, size=n)
            a /= a.sum()
    # final evaluation using evaluator
    candidate = best_a / best_a.max() * 1000.0 # maybe scale within limits but not needed
    # ensure sum >0.01
    if candidate.sum() < 0.01:
        candidate = np.ones_like(candidate) * (0.01 / candidate.size)
    # Return list
    return [float(x) for x in candidate.tolist()]

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    # Determine length
    try:
        n = len(height_sequence_1)
    except Exception:
        n = 256
    # Fast convolution length (next power-of-2)
    L = 1 << ((2*n - 1).bit_length())
    def fast_max(v):
        A = np.fft.rfft(v, n=L)
        b = np.fft.irfft(A*A, n=L)[:2*n-1]
        return b.max()
    # Baseline uniform
    a_uniform = np.full(n, 1.0/n, dtype=np.float64)
    best_vec = a_uniform.copy()
    best_score = float(evaluate_sequence(best_vec.tolist()))
    # Keep position of best approximate max for early exit
    best_approx = fast_max(best_vec)
    # Random search
    while time.monotonic() < deadline:
        # Sample Dirichlet-like vector
        alpha = math.exp(rng.uniform(math.log(0.2), math.log(4.0)))  # 0.2~4

                best_score = exact
                best_vec = v.copy()
        # optionally break early if score already below target by a margin
        if best_score < 1.4:
            break
    # final scaling optional but keep as is
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    if best_vec.sum() < 0.01:
        best_vec = np.full_like(best_vec, 0.01 / best_vec.size)
    return [float(x) for x in best_vec.tolist()]

import numpy as np
import time
import math

def _fast_max_conv(vec, fft_len):
    """Return the maximum entry of the full convolution vec*vec using FFT."""
    # real FFT of the zero‑padded vector
    fft_vec = np.fft.rfft(vec, n=fft_len)
    # convolution in frequency domain = element‑wise square
    conv = np.fft.irfft(fft_vec * fft_vec, n=fft_len)
    # first 2*len(vec)-1 entries are the true convolution
    return conv[: 2 * len(vec) - 1].max()


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a non‑negative step function with a low AC1 bound.

    The search works in two phases:
      1. Randomly generate Dirichlet‑like vectors, sort them (small values meet
         large values) and keep the best one seen.
      2. Starting from the best vector, perform a simple projected sub‑gradient
         descent on the maximal convolution entry.

    The heavy evaluation (full np.convolve) is only called when a promising
    candidate is found; all fast evaluations use FFT.
    """
    rng = np.random.default_rng(seed)
    # length of the sequence – use the provided parent if present
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256

    # FFT length: next power of two ≥ 2*n‑1
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 1. Random sampling (Dirichlet style, then sorted)
    # ------------------------------------------------------------------
    # baseline: uniform vector (known score 2.0)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_vec = uniform.copy()
    best_score = float(evaluate_sequence(best_vec.tolist()))
    # fast approx of the maximal convolution entry for the uniform case
    best_approx = _fast_max_conv(best_vec, fft_len)

    # keep looping until the time budget is almost exhausted
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    while time.monotonic() < deadline:
        # draw a Dirichlet‑like vector: Gamma(alpha) i.i.d., then normalise
        # pick alpha log‑uniformly in a reasonable range
        log_alpha = rng.uniform(math.log(0.2), math.log(5.0))
        alpha = math.exp(log_alpha)
        vec = rng.gamma(alpha, size=n)
        if vec.sum() == 0.0:
            continue
        vec = np.sort(vec)                     # ascending → anti‑correlated
        vec /= vec.sum()                       # normalise L1‑norm to 1

        # fast estimate of the score (exact sum is 1, so score ≈ 2*n*max)
        approx_max = _fast_max_conv(vec, fft_len)
        approx_score = 2.0 * n * approx_max

        # only evaluate exactly if we have a genuine improvement chance
        if approx_score < best_score * 0.99:
            exact = float(evaluate_sequence(vec.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best_score = exact
                best_vec = vec.copy()
                best_approx = approx_max
                # early stop if we are already far below the target
                if best_score < 1.4:
                    break

    # ------------------------------------------------------------------
    # 2. Simple projected sub‑gradient descent on the maximal entry
    # ------------------------------------------------------------------
    a = best_vec.copy()
    cur_max = best_approx
    # parameters for the descent
    step_factor = 0.02

    while time.monotonic() < deadline:
        # compute full convolution (FFT) and locate the maximal index
        fft_a = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fft_a * fft_a, n=fft_len)[: 2 * n - 1]
        k = int(conv.argmax())

        # gradient of the selected entry with respect to a
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        diag = mask & (2 * np.arange(n) == k)

        g = np.zeros_like(a)
        g[diag] = a[diag]                     # i == k‑i term appears once
        off = mask & ~diag
        g[off] = 2.0 * a[idx[off]]            # i != k‑i term appears twice

        # step size inversely proportional to the largest gradient entry
        step = step_factor / (np.max(g) + 1e-12)
        a_new = a - step * g
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s == 0.0:
            break
        a_new /= s

        new_max = _fast_max_conv(a_new, fft_len)

        # accept the move if it improves the maximal entry
        if new_max < cur_max:
            a = a_new
            cur_max = new_max
            # occasional exact evaluation to update the global best
            if new_max < best_approx:
                exact = float(evaluate_sequence(a.tolist()))
                if np.isfinite(exact) and exact < best_score:
                    best_score = exact
                    best_vec = a.copy()
                    best_approx = new_max
                    # stop early if we are comfortably under the target
                    if best_score < 1.4:
                        break
        else:
            # occasional random perturbation to escape plateaus
            if rng.random() < 0.001:
                a = a * np.exp(rng.normal(0.0, 0.05, size=n))
                a = np.clip(a, 0.0, None)
                a /= a.sum()
                cur_max = _fast_max_conv(a, fft_len)

    # ------------------------------------------------------------------
    # Final sanitisation and return
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    if best_vec.sum() < 0.01:
        # guarantee a positive L1‑norm (should never happen)
        best_vec = np.full_like(best_vec, 0.01 / best_vec.size)

    return [float(x) for x in best_vec.tolist()]
