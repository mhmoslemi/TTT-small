def build_sequence(delta, block_len, eps, n):
    r = 1.0 - delta
    idx = np.arange(block_len, dtype=np.float64)
    block = np.power(r, idx, dtype=np.float64)
    tail_len = n - block_len
    if tail_len > 0:
        tail = np.full(tail_len, eps, dtype=np.float64)
        seq = np.concatenate((block, tail))
    else:
        seq = block
    return seq
```

import math

# constant epsilon that balances cross terms: eps = exp(-1)/2
_EPS_CONST = math.exp(-1.0) / 2.0  # ≈0.18394


def _make_sequence(delta: float, block_len: int, eps: float, n: int) -> np.ndarray:
    """
    Build a sequence of length n:
    * first `block_len` entries follow a geometric decay a_i = (1-delta)^i
    * remaining entries are constant `eps`
    """
    if delta <= 0.0:
        delta = 1e-12
    if delta >= 1.0:
        delta = 0.999999
    r = 1.0 - delta
    block_len = max(1, min(n, int(block_len)))
    # geometric block
    block = np.power(r, np.arange(block_len, dtype=np.float64))
    tail_len = n - block_len
    if tail_len > 0:
        tail = np.full(tail_len, eps, dtype=np.float64)
        seq = np.concatenate((block, tail))
    else:
        seq = block
    # enforce evaluator bounds
    return np.clip(seq, 0.0, 1000.0)
