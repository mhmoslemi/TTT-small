    n = a.size
    size = 2*n - 1
    # Next power of two
    N = 1 << (size-1).bit_length()
    A = np.fft.rfft(a, N)
    B = A * A

import time
import numpy as np

def _conv_self(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (O(n log n) via FFT)."""
    n = x.shape[0]
    size = 2 * n - 1
    # next power of two for efficient FFT
    N = 1 << (size - 1).bit_length()
    fx = np.fft.rfft(x, N)
    conv = np.fft.irfft(fx * fx, N)[:size]
    # Numerical noise may produce tiny negatives
    conv[conv < 0] = 0.0
    return conv


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Iterative mass‑redistribution search that flattens the self‑convolution.

    Returns a list of non‑negative coefficients whose AC1 upper bound is
    (hopefully) below the baseline of 2.0.
    """
    rng = np.random.default_rng(seed)
    # time limit – leave a small safety margin
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # initialise a starting sequence
    if "height_sequence_1" in globals():
        init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if init.ndim != 1 or init.size == 0 or not np.isfinite(init).all():
            init = None
    else:
        init = None

    if init is not None:
        a = init.copy()
    else:
        # fall back to a modest uniform vector
        a = np.ones(256, dtype=np.float64)

    n = a.shape[0]

    # evaluate the initial candidate
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # keep the best ever seen
    cur = a.copy()

    # parameters for the redistribution step
    reduction_factor = 0.85          # how much mass to move from the dominant pair
    random_move_prob = 0.12          # probability of an exploratory random move
    min_factor = 0.5                 # lower bound for reduction_factor (to keep steps large early)

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 1) compute convolution and locate the worst (largest) entry
        conv = _conv_self(cur)
        k_max = int(np.argmax(conv))

        # 2) identify the pair (i,j) that contributes most to this entry
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k_max - i_vals
        contrib = cur[i_vals] * cur[j_vals]
        idx_local = int(np.argmax(contrib))
        i_star = i_vals[idx_local]
        j_star = j_vals[idx_local]

        # --------------------------------------------------------------
        # 3) propose a new sequence by moving mass away from (i_star,j_star)
        cand = cur.copy()
        lost_i = cand[i_star] * (1.0 - reduction_factor)
        lost_j = cand[j_star] * (1.0 - reduction_factor)
        cand[i_star] *= reduction_factor
        cand[j_star] *= reduction_factor

        # redistribute the removed mass to a randomly chosen third position
        k_target = rng.integers(0, n)
        while k_target == i_star or k_target == j_star:
            k_target = rng.integers(0, n)
        cand[k_target] += lost_i + lost_j

        # enforce bounds
        np.clip(cand, 0.0, 1000.0, out=cand)

        # --------------------------------------------------------------
        # 4) evaluate the proposal
        cand_score = float(evaluate_sequence(cand.tolist()))

        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand
            best_score = cand_score
            cur = cand
            # after a successful step, keep the reduction factor moderate
            reduction_factor = max(min_factor, reduction_factor * 0.995)
            continue  # go to next iteration

        # --------------------------------------------------------------
        # 5) occasional random exploration (helps escape plateaus)
        if rng.random() < random_move_prob:
            # random block rescaling
            block_start = rng.integers(0, n)
            block_len = rng.integers(1, max(2, n // 30))
            block_end = min(n, block_start + block_len)
            scale = np.exp(rng.normal(0.0, 0.15))
            cur[block_start:block_end] *= scale
            np.clip(cur, 0.0, 1000.0, out=cur)
        else:
            # if the move did not improve, gently relax the reduction factor
            reduction_factor = min(0.99, reduction_factor * 1.002)

        # safety: break a little before the hard deadline
        if time.monotonic() > deadline - 0.05:
            break

    return [float(v) for v in best]
