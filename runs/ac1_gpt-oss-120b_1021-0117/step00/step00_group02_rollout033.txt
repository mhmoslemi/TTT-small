    max_val = 0.0
    pow_r = 1.0
    # for k=0..n-1
    for k in range(n):
        val = (k+1) * pow_r
        if val > max_val:

        pow_r *= r
    # continue for k=n..2n-2
    # we need pow_r for r^k; after loop, pow_r = r**n
    pow_r_k = pow_r

        if val > max_val:
            max_val = val
        pow_r_k *= r

def build_C(f):
    n = f.size
    m = 2*n - 1
    B = (n + block_size - 1) // block_size
    block_sizes = np.empty(B, dtype=np.int32)
    starts = np.empty(B, dtype=np.int32)
    for b in range(B):
        start = b*block_size
        end = min(n, start+block_size)
        starts[b] = start
        block_sizes[b] = end - start
    prefixes = np.concatenate(([0.0], np.cumsum(f, dtype=np.float64)))  # length n+1
    C = np.empty((m, B), dtype=np.float64)
    for b in range(B):
        s = starts[b]
        e = s + block_sizes[b]  # exclusive
        # For each shift k: sum_{i} f[i] where i in [max(0, k-(e-1)), min(n-1, k-s)]
        # Equivalent to prefix[max ..] difference
        # We'll compute using vectorized approach:
        ks = np.arange(m)  # shape (m,)
        lo = ks - (e - 1)
        lo = np.maximum(lo, 0)
        hi = ks - s
        hi = np.minimum(hi, n - 1)
        # Need hi >= lo; else sum=0
        mask = hi >= lo
        # When mask true: sum = prefix[hi+1] - prefix[lo]
        col = np.zeros(m, dtype=np.float64)
        col[mask] = prefixes[hi[mask]+1] - prefixes[lo[mask]]
        C[:, b] = col
    return C, block_sizes
```

    f_fft = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f_fft * f_fft, n=L)[:2*n-1]

import time
import numpy as np

# ------------------------------------------------------------
# helper utilities (top‑level, no closures)
# ------------------------------------------------------------
def max_conv_fft(a: np.ndarray) -> float:
    """Maximum of the full convolution a*a using FFT (O(n log n))."""
    n = a.size
    L = 1 << ((2 * n - 2).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return float(conv.max())


def build_constraint_matrix(f: np.ndarray, block_starts, block_sizes):
    """Return C where C[k, b] = sum_{i in block b} f_i that overlaps shift k."""
    n = f.size
    m = 2 * n - 1
    prefix = np.concatenate(([0.0], np.cumsum(f, dtype=np.float64)))
    ks = np.arange(m, dtype=np.int64)

    B = len(block_sizes)
    C = np.empty((m, B), dtype=np.float64)

    for b in range(B):
        s = block_starts[b]
        e = s + block_sizes[b]                     # exclusive
        lo = ks - (e - 1)
        lo = np.maximum(lo, 0)
        hi = ks - s
        hi = np.minimum(hi, n - 1)
        mask = hi >= lo
        col = np.zeros(m, dtype=np.float64)
        col[mask] = prefix[hi[mask] + 1] - prefix[lo[mask]]
        C[:, b] = col
    return C


def solve_lp(C, block_sizes, M, max_iters=500):
    """Linear program: maximise sum(g * block_sizes)  s.t. C @ g <= M, g >= 0."""
    try:
        import cvxpy as cp
    except Exception:                     # cvxpy not available
        return None

    B = len(block_sizes)
    g = cp.Variable(B)
    obj = cp.Maximize(cp.sum(cp.multiply(g, np.array(block_sizes, dtype=np.float64))))
    constraints = [C @ g <= M, g >= 0]
    prob = cp.Problem(obj, constraints)

    # try a fast LP solver, fall back to GLPK if needed
    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                    max_iters=max_iters, feastol=1e-6, abstol=1e-6, reltol=1e-6)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK, warm_start=True, max_iters=max_iters)
        except Exception:
            return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    if g.value is None:
        return None
    return np.array(g.value, dtype=np.float64)


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Small O(n) random perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)
