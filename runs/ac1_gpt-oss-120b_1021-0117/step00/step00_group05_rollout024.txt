import time
import numpy as np

# ------------------------------------------------------------
# Helper: greedy construction of a Sidon (B2) set
# ------------------------------------------------------------
def sidon_set(limit):
    """
    Return a list of indices in [0, limit) forming a Sidon set:
    all pairwise sums i+j (including i+i) are distinct.
    """
    sums = set()
    chosen = []
    for i in range(limit):
        # quick rejection: i+i already taken?
        if (i + i) in sums:
            continue
        conflict = False
        for j in chosen:
            if (i + j) in sums:
                conflict = True
                break
        if conflict:
            continue
        # accept i
        chosen.append(i)
        # record new sums
        sums.add(i + i)
        for j in chosen[:-1]:
            sums.add(i + j)
            # i+j == j+i, no need to add twice
        # stop early if we already have many elements
        if len(chosen) >= int(np.sqrt(limit)) + 5:
            # a safe overshoot; continue a bit longer but not too much
            pass
    return chosen
