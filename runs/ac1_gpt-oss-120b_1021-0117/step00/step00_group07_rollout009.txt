def _build_sidon_set(n: int, rng: np.random.Generator) -> list:
    """Greedy construction of a Sidon set (all pairwise sums distinct)."""
    order = rng.permutation(n)
    s = []
    sums = set()
    for i in order:
        i2 = i + i
        if i2 in sums:
            continue
        conflict = False
        for j in s:
            if i + j in sums:
                conflict = True
                break
        if conflict:
            continue
        # accept i
        s.append(i)
        sums.add(i2)
        for j in s[:-1]:
            sums.add(i + j)
    return s
