def simplex_projection(v):
    # v: 1-d array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

    deadline = time.monotonic() + max(0.0, budget_s - 0.5)

        arr = np.asarray(height_sequence_1, dtype=np.float64)

    if arr.sum() == 0:
        arr = np.ones_like(arr)
    a = arr / arr.sum()
    a += 0.001 * rng.random(a.shape)
    a = np.maximum(a, 0)
    a = a / a.sum()

    n = a.shape[0]
    # compute fft size for longest convolution
    k = 3 * n - 2
    size = 1 << (k - 1).bit_length()   # next pow2 >= k

    best = a.copy()
    # compute best score using self-convolution via FFT
    def compute_maxc(vec):
        fa = np.fft.rfft(vec, n=size)
        c = np.fft.irfft(fa * fa, n=size)
        return np.max(c[:2*n-1])  # only relevant part

    maxc_best = compute_maxc(best)
    best_score = 2 * n * maxc_best / (np.sum(best)**2)

    step = 0.5
    min_step = 1e-6
    # tolerance for mask detection
    eps = 1e-12
    # main loop
    while time.monotonic() < deadline:
        # compute convolution
        fa = np.fft.rfft(a, n=size)
        c_full = np.fft.irfft(fa * fa, n=size)
        c = c_full[:2*n-1]
        maxc = c.max()

def simplex_projection(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non-positive, return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

import numpy as np, time

def simplex_projection(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback uniform
        theta = cssv[-1] / n
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

import numpy as np
import time

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(a: np.ndarray, size: int) -> float:
    """Return max of the linear self‑convolution of a (zero‑padded to size)."""
    fa = np.fft.rfft(a, n=size)
    c = np.fft.ifft(fa * fa, n=size).real
    return float(np.max(c[: 2 * a.size - 1]))


def _perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Small O(n) random perturbation; returns a normalized copy."""
    n = a.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    cand = a.copy()
    cand[start : start + width] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    cand = np.clip(cand, 0.0, 1000.0)
    cand = _simplex_projection(cand)  # keep sum = 1
    return cand
