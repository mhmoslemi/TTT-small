def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a deterministic non‑negative step-function whose AC1 upper bound

    The construction is a linear decreasing profile from the maximum
    (1000) down to zero.
    """
    # Time budget – leave a small safety margin.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # Length of the problem (use the provided parent if available).
    try:
        n = int(np.asarray(height_sequence_1, dtype=np.int64).size)

            raise ValueError

    a = np.clip(a, 0.0, 1000.0)

    rng = np.random.default_rng(seed)
