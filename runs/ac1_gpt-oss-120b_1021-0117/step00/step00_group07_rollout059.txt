def fast_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    # compute convolution via FFT
    # get next power of 2 length at least 2n-1
    m = 1 << ((2*n-1).bit_length())
    fft_a = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fft_a * fft_a, n=m)[:2*n-1]
    max_b = conv.max()
    return 2 * n * max_b  # because sum a = 1

def conv_fft(a):
    n = a.shape[0]

import time
import numpy as np

# ---------- helper functions -------------------------------------------------


def conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT (real input)."""
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv


def fast_score(a: np.ndarray) -> float:
    """Fast surrogate for evaluate_sequence (exact for the bound)."""
    a = np.asarray(a, dtype=np.float64)
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    n = a.shape[0]
    max_b = conv_fft(a).max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


def shape_product_power(n: int, p: float) -> np.ndarray:
    """Symmetric shape a_i ∝ ((i+0.5)*(n-i-0.5))^{-p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    prod = i * (n - i - 0.5)
    # avoid division by zero – prod is never zero for the formula above
    a = prod ** (-p) if p != 0.0 else np.ones_like(i)
    # clip only to obey the evaluator limits (safety)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s == 0.0:
        a = np.ones(n, dtype=np.float64)
        s = n
    return a / s


def mix_with_uniform(a: np.ndarray, alpha: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = a.shape[0]
    u = np.ones(n, dtype=np.float64) / n
    mixed = (1.0 - alpha) * a + alpha * u
    return mixed / mixed.sum()


# ---------- main search ------------------------------------------------------


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a list of non‑negative coefficients minimizing the AC1 upper bound.
    The routine builds a parametrised family of symmetric shapes, refines the
    best via a simple projected gradient descent on the maximal convolution
    entry, and finally returns the best sequence found before the deadline.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the sandbox timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------ length
    try:
        # height_sequence_1 is provided by the harness; fall back if invalid
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        n = parent.size
    except Exception:
        n = 1024  # a reasonable default when no parent is available

    # ----------------------------------------------------------------- initial grid
    best_seq = None
    best_sc = np.inf

    # search a one‑parameter family a_i ∝ ((i+0.5)*(n-i-0.5))^{-p}
    # p = 0 gives uniform, p ≈ 0.5 reproduces the classic arcsine shape.
    for p in np.linspace(0.0, 0.6, 31):
        cand = shape_product_power(n, p)
        sc = fast_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_seq = cand

    # try a few convex mixes with the uniform distribution (helps smoothing)
    uniform = np.ones(n, dtype=np.float64) / n
    for alpha in np.linspace(0.0, 0.5, 11):
        mixed = (1.0 - alpha) * best_seq + alpha * uniform
        mixed /= mixed.sum()
        sc = fast_score(mixed)
        if sc < best_sc:
            best_sc = sc
            best_seq = mixed

    # ---------------------------------------------------------------- gradient descent
    a = best_seq.copy()
    lr = 0.12                     # learning rate for the projected step
    stall = 0                     # counts iterations without improvement

    while time.monotonic() < deadline:
        conv = conv_fft(a)
        k_max = int(conv.argmax())
        # gradient of the maximal convolution entry (ordered pairs)
        idx = np.abs(k_max - np.arange(n, dtype=np.int64))
        mask = idx < n
        g = np.zeros_like(a)
        g[mask] = a[idx[mask]]

        # projected gradient step: subtract, clip, renormalise
        a_new = a - lr * g
        a_new[a_new < 0.0] = 0.0
        tot = a_new.sum()
        if tot < 1e-12:                     # numerical safety net
            a_new = uniform.copy()
            tot = 1.0
        a_new /= tot

        # occasional tiny random multiplicative noise (helps escape plateaux)
        if rng.random() < 0.015:
            noise = np.exp(rng.normal(0.0, 0.012, size=n))
            a_new = np.clip(a_new * noise, 0.0, None)
            a_new /= a_new.sum()

        sc = fast_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_seq = a_new.copy()
            stall = 0
        else:
            stall += 1

        # adapt learning rate if we stall for a while
        if stall > 250:
            lr *= 0.5
            stall = 0
            if lr < 1e-4:
                lr = 1e-4

        a = a_new

        # early exit once we are clearly below the target bound
        if best_sc < 1.5028:
            break

    # ----------------------------------------------------------------- final list
    result = [float(v) for v in best_seq]
    return result
```
