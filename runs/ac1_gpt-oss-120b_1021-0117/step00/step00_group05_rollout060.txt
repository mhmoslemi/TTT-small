import time
import numpy as np

# ------------------------------------------------------------
# Helper: greedy construction with a bound on unordered pair counts
# ------------------------------------------------------------
def _greedy_build(n, C, rng):
    """Return a set of indices and the unordered‑pair count array."""
    selected = set()
    pair_cnt = np.zeros(2 * n - 1, dtype=np.uint8)      # unordered pairs per sum
    order = rng.permutation(n)                         # random insertion order
    for i in order:
        # self‑pair would increase count at 2*i
        if pair_cnt[2 * i] >= C:
            continue
        # check all existing indices
        ok = True
        for j in selected:
            if pair_cnt[i + j] >= C:
                ok = False
                break
        if not ok:
            continue
        # add i
        for j in selected:
            pair_cnt[i + j] += 1
        pair_cnt[2 * i] += 1
        selected.add(i)
    return selected, pair_cnt


# ------------------------------------------------------------
# Helper: eliminate sums that have two cross‑pairs (no self‑pair)
# ------------------------------------------------------------
def _clean_selected(selected, pair_cnt, C, n):
    """Make sure any sum with count==2 also contains a self‑pair."""
    # work directly on the mutable set
    while True:
        bad_sum = None
        # locate a problematic sum
        for s in np.flatnonzero(pair_cnt == 2):
            # self‑pair present?
            if (s % 2 == 0) and ((s // 2) in selected):
                continue
            bad_sum = int(s)
            break
        if bad_sum is None:
            break          # all good

        # find the two unordered pairs that realise this sum
        pairs = []
        for a in selected:
            b = bad_sum - a
            if b < a:          # ensure a <= b to avoid duplicates
                continue
            if b in selected:
                pairs.append((a, b))
                if len(pairs) == 2:
                    break

        if not pairs:          # should not happen, but guard against infinite loop
            break

        # remove one index from the first pair (arbitrary choice)
        rem = pairs[0][0]
        # decrement all counts contributed by this index
        for j in selected:
            if j == rem:
                continue
            pair_cnt[rem + j] -= 1
        pair_cnt[2 * rem] -= 1
        selected.remove(rem)

    return selected, pair_cnt


# ------------------------------------------------------------
# Helper: try to add more indices after cleaning
# ------------------------------------------------------------
def _fill_selected(selected, pair_cnt, C, n, rng):
    """Greedy pass that tries to insert any still admissible indices."""
    remaining = [i for i in rng.permutation(n) if i not in selected]
    for i in remaining:
        if pair_cnt[2 * i] >= C:
            continue
        ok = True
        for j in selected:
            if pair_cnt[i + j] >= C:
                ok = False
                break
        if not ok:
            continue
        # add i
        for j in selected:
            pair_cnt[i + j] += 1
        pair_cnt[2 * i] += 1
        selected.add(i)
    return selected, pair_cnt

# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a short step‑function giving a low AC1 bound."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # initialise best candidate (fallback)
    # ------------------------------------------------------------------
    parent = np.clip(
        np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0
    )
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)
    n = parent.size
    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    C = 2                     # max unordered pairs per convolution index

    # ------------------------------------------------------------------
    # repeated stochastic construction
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # 1) greedy build
        sel, cnt = _greedy_build(n, C, rng)

        # 2) clean sums that would give b_k = 4
        sel, cnt = _clean_selected(sel, cnt, C, n)

        # 3) try to grow a little more
        sel, cnt = _fill_selected(sel, cnt, C, n, rng)

        # 4) build concrete sequence (all heights = 1.0)
        cand = np.zeros(n, dtype=np.float64)
        if sel:

        # 5) evaluate
        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best_seq = cand
            best_score = cand_score

        # early exit if we already beat the target comfortably
        if best_score < 1.55:      # a little cushion for the time remaining
            # keep searching a bit more but allow graceful exit soon
            if time.monotonic() + 0.2 > deadline:
                break

    # ------------------------------------------------------------------
    # return the best found sequence (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
