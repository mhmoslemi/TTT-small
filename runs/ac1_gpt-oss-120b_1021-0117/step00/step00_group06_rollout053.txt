    rng = np.random.default_rng(seed)
    # leave a small safety margin for the sandbox timeout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1. Determine the problem size and a fallback parent sequence
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(256, dtype=np.float64)

    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    best_seq = np.clip(parent, 0.0, 1000.0).copy()

    # fast evaluation (identical to the official one for valid data)
    def _score(seq):
        s = float(seq.sum())
        if s < 0.01:
            return float('inf')
        conv = np.convolve(seq, seq)
        return 2.0 * n * float(conv.max()) / (s * s)

    best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # 2. Helper to build a sparse index set with bounded pair collisions
    # ------------------------------------------------------------------
    def _build_sparse_set(max_pairs, target_sz):
        """Greedy construction of a set S⊂{0,…,n‑1} with
        at most ``max_pairs`` unordered pairs per sum i+j.
        Returns a NumPy array of the chosen indices."""
        sum_counts = {}          # sum → current unordered‑pair count
        indices = []
        # random order of candidate positions
        order = rng.permutation(n)
        for i in order:
            if len(indices) >= target_sz:
                break
            # self‑sum i+i must also respect the bound
            s_self = i + i
            if sum_counts.get(s_self, 0) >= max_pairs:
                continue
            # check all sums with already chosen indices
            ok = True
            for j in indices:
                s = i + j
                if sum_counts.get(s, 0) >= max_pairs:
                    ok = False
                    break
            if not ok:
                continue
            # accept i: update counts
            for j in indices:
                s = i + j
                sum_counts[s] = sum_counts.get(s, 0) + 1
            sum_counts[s_self] = sum_counts.get(s_self, 0) + 1
            indices.append(i)
        return np.array(indices, dtype=int)

    # ------------------------------------------------------------------
    # 3. Main stochastic search loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # random search parameters
        max_pairs = int(rng.integers(1, 4))          # 1, 2 or 3
        want_k   = int(rng.integers(80, 200))       # target support size
        eps_exp  = rng.uniform(-4.0, -0.4)           # log10(epsilon)
        epsilon  = float(10 ** eps_exp)              # tiny background level

        # build a collision‑limited support set
        idx = _build_sparse_set(max_pairs, want_k)
        k = len(idx)
        if k < 20:        # too tiny – unlikely to beat the fallback
            continue

        # construct the candidate sequence
        cand = np.full(n, epsilon, dtype=np.float64)
        cand[idx] += 1.0          # large entries of height 1 on top of the background

        # evaluate (fast, no list conversion)
        sc = _score(cand)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq   = cand.copy()

        # occasional tiny refinement of the current best:
        if rng.random() < 0.05 and time.monotonic() < deadline:
            # pick a random large index and slightly shrink it,
            # moving the removed mass to a random background position
            large_pos = rng.choice(idx)
            delta = cand[large_pos] * 0.15      # 15 % reduction
            cand[large_pos] -= delta
            # redistribute to a random location (could be same as background)
            recv = rng.integers(0, n)
            cand[recv] += delta
            sc2 = _score(cand)
            if math.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq   = cand.copy()

    # ------------------------------------------------------------------
    # 4. Return a valid list (fallback is always present)
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
