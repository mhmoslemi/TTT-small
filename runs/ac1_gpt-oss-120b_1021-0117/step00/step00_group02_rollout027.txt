import numpy as np
import time

def _convolution_max(a: np.ndarray) -> float:
    """Return max entry of the full convolution a*a."""
    return np.max(np.convolve(a, a))

def propose_candidate(seed: int = 42, budget_s: float = 1100.0, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that tries to
    minimise the AC1 upper bound 2*n*max(conv)/sum(a)^2.
    The routine begins from the supplied parent (if present) and
    performs a cheap greedy local‑search that moves mass away from the
    current most‑active convolution index.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # initialise candidate
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise Exception
    except Exception:
        # fallback: a modest uniform vector
        base = np.full(256, 1.0, dtype=np.float64)

    # Clip to the evaluator's admissible interval
    a = np.clip(base, 0.0, 1000.0)
    n = a.size

    # best bookkeeping
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # helper to evaluate a candidate (fast shortcut that mirrors the
    # evaluator but avoids the list conversion overhead)
    # ------------------------------------------------------------------
    def _score(arr: np.ndarray) -> float:
        # quick guard against degenerate sum
        s = arr.sum()
        if s < 0.01:
            return np.inf
        return 2.0 * n * _convolution_max(arr) / (s * s)

    # ------------------------------------------------------------------
    # main loop: move a small amount of mass away from the peak index
    # ------------------------------------------------------------------
    # parameters
    eps = 0.02          # fraction of the source entry to move each step
    improve_patience = 2000  # stop if no improvement after many attempts
    no_improve = 0

    while time.monotonic() < deadline:
        # compute current convolution peak
        conv = np.convolve(a, a)
        k_max = int(np.argmax(conv))
        max_val = float(conv[k_max])

        # contributions to the peak: a[i] * a[k_max - i] for valid i
        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]
        # pick the index that participates most strongly
        i_src = int(np.argmax(contrib))
        j_src = int(k_max - i_src)

        # select a random destination different from the source pair
        # (ensuring we have a place to deposit the mass)
        avoid = {i_src, j_src}
        dst_candidates = np.setdiff1d(np.arange(n), list(avoid), assume_unique=True)
        if dst_candidates.size == 0:
            # nothing to move to – give up this iteration
            break
        i_dst = int(rng.choice(dst_candidates))

        # compute a modest transfer amount
        delta = eps * a[i_src]
        if delta <= 0.0:
            # source already empty; pick another source
            a[i_src] = 0.0
            continue

        # propose the move
        a_new = a.copy()
        a_new[i_src] -= delta
        a_new[i_dst] += delta
        # enforce bounds
        np.clip(a_new, 0.0, 1000.0, out=a_new)

        # Evaluate the new candidate (use fast score then confirm)
        new_score = _score(a_new)
        if not np.isfinite(new_score):

        # occasional full re‑evaluation to stay consistent with the authoritative scorer
        if (no_improve + 1) % 31 == 0:
            new_score = float(evaluate_sequence(a_new.tolist()))
        if new_score < best_score:

            best_score = new_score
            a = a_new

        # early exit if we have already reached the theoretical lower bound ~2
        if best_score <= 2.000001:
