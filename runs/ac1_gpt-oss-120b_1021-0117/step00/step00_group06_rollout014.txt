            factor = float(np.exp(rng.normal(0.0, 0.12)))
            cand[start:start + width] *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)

        elif move == 1:                   # swap two positions
            i, j = rng.choice(n, size=2, replace=False)
            cand[i], cand[j] = cand[j], cand[i]

        elif move == 2:                   # transfer a fraction from i to j
            i, j = rng.choice(n, size=2, replace=False)
            _transfer(cand, i, j, rng.random() * 0.5)

        else:                             # mirror‑symmetrize with some probability
            if rng.random() < 0.2:
                cand = 0.5 * (cand + cand[::-1])
            np.clip(cand, 0.0, 1000.0, out=cand)

        # occasional extra symmetry test (helps when the optimum is symmetric)
        if rng.random() < 0.07:
            sym = cand[::-1]
            sym_score = _ac1_score(sym, n, n_fft)
            if sym_score < best_score:
                best = sym.copy()
                best_score = sym_score
            if sym_score < cur_score:
                cand = sym
                # keep the better score for the acceptance test below
                cur_score = sym_score

        cand_score = _ac1_score(cand, n, n_fft)

        # acceptance: always take improvements, otherwise Metropolis step
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # probability exp(–Δ/Temp) ; guard against overflow
            diff = cand_score - cur_score
            if temp > 0 and diff < 50:          # 50 is a safe upper bound for exp
                if rng.random() < np.exp(-diff / temp):
                    accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score

        # occasionally renormalise the whole vector to avoid drift outside bounds
        if rng.random() < 0.001:
            # scale so that max stays ≤1000 without changing the shape
            scale = min(1.0, 1000.0 / (best.max() + 1e-12))
            if scale < 1.0:
                best *= scale
                cur *= scale
                cur_score = _ac1_score(cur, n, n_fft)
                best_score = cur_score

    # ------------------------------------------------------------------ #
    # 4.  Return the best sequence found (as plain Python floats).
    # ------------------------------------------------------------------ #
    return [float(v) for v in best]
