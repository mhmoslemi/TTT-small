def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0., float(budget_s) - 0.5)

    # get parent initial if exists
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise Exception
    except Exception:
        parent_arr = np.ones(256, dtype=np.float64)
    best = np.clip(parent_arr, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best.tolist()))

    n = best.size
    # generate shape candidates
    indices = np.arange(n, dtype=np.float64)
    t = indices / (n - 1)  # in [0,1]
    # loop over p values
    for p in np.arange(0.5, 6.5, 0.5):
        if time.monotonic() >= deadline:
            break
        seq = (1.0 - t) ** p
        seq = np.clip(seq, 0.0, 1000.0)
        score = float(evaluate_sequence(seq.tolist()))
        if score < best_score:
            best = seq
            best_score = score

    # exponential
    for lam in np.logspace(-2, 2, num=40):
        if time.monotonic() >= deadline:
            break
        seq = np.exp(-lam * t)
        seq = np.clip(seq, 0.0, 1000.0)
        score = float(evaluate_sequence(seq.tolist()))
        if score < best_score:
            best = seq
            best_score = score

    # combined power*exp shape
    for p in np.arange(1.0, 4.1, 1.0):
        for lam in np.logspace(-2, 1, num=20):
            if time.monotonic() >= deadline:
                break
            seq = ((1.0 - t) ** p) * np.exp(-lam * t)
            seq = np.clip(seq, 0.0, 1000.0)
            score = float(evaluate_sequence(seq.tolist()))
            if score < best_score:
                best = seq
                best_score = score

    # local random perturbations around best
    while time.monotonic() < deadline:
        # create perturbed copy by Gaussian scaling in a random block
        candidate = best.copy()
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        # multiplicative perturbation
        factor = np.exp(rng.normal(0.0, 0.1, size=stop-start))
        candidate[start:stop] *= factor
        # optionally symmetric smoothing
        if rng.random() < 0.3:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = np.clip(candidate, 0.0, 1000.0)
        score = float(evaluate_sequence(candidate.tolist()))
        if score < best_score:
            best = candidate
            best_score = score

    return [float(x) for x in best]

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0., float(budget_s) - 2.0)

    # parent fallback
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise Exception
    except Exception:
        parent_arr = np.ones(256, dtype=np.float64)
    best = np.clip(parent_arr, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best.tolist()))

    n = best.size
    idx = np.arange(n, dtype=np.float64)
    t = idx / (n - 1)  # normalized [0,1]

    # Helper to evaluate and update best
    def try_candidate(seq):
        nonlocal best, best_score
        if seq.size != n:
            return
        # quickly compute sum to avoid infinite
        if seq.sum() < 0.01:
            return
        score = float(evaluate_sequence(seq.tolist()))
        if math.isfinite(score) and score < best_score:
            best = seq
            best_score = score

    # families
    # power law (1 - t)^p
    for p in np.arange(0.2, 5.01, 0.2):
        if time.monotonic() >= deadline:
            break
        seq = (1.0 - t) ** p
        try_candidate(seq)

    # exponential
    for lam in np.logspace(-2, 2, num=40):
        if time.monotonic() >= deadline:
            break
        seq = np.exp(-lam * t)
        try_candidate(seq)

    # combined power * exp
    for p in np.arange(0.5, 4.1, 0.5):
        if time.monotonic() >= deadline:
            break
        for lam in np.logspace(-2, 1, num=20):
            if time.monotonic() >= deadline:
                break
            seq = ((1.0 - t) ** p) * np.exp(-lam * t)
            try_candidate(seq)

    # Gaussian-like: exp(-lam * t**2)
    for lam in np.logspace(-2, 2, num=30):
        if time.monotonic() >= deadline:
            break
        seq = np.exp(-lam * (t ** 2))
        try_candidate(seq)

    # logistic decreasing: 1/(1 + np.exp(k*(t - shift)))
    for k in np.linspace(0.5, 10.0, 15):
        if time.monotonic() >= deadline:
            break
        for shift in np.linspace(0.0, 0.5, 5):
            if time.monotonic() >= deadline:
                break
            seq = 1.0 / (1.0 + np.exp(k * (t - shift)))
            try_candidate(seq)

    # local randomized perturbation around best
    while time.monotonic() < deadline:
        if best is None:
            break
        cand = best.copy()
        # random block scaling
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        factor = np.exp(rng.normal(0.0, 0.1, size=stop-start))
        cand[start:stop] *= factor
        if rng.random() < 0.3:
            cand = 0.5 * (cand + cand[::-1])
        cand = np.clip(cand, 0.0, 1000.0)
        try_candidate(cand)

    return [float(x) for x in best]

    idx = np.arange(n, dtype=np.float64)
    t = idx / (n - 1)                     # normalized coordinate in [0, 1]

    def try_candidate(seq):
        """Evaluate seq; keep it if it improves the best score."""
        nonlocal best, best_score
        if seq.shape != (n,):
            return
        if seq.sum() < 0.01:  # evaluator would reject this
            return
        sc = float(evaluate_sequence(seq.tolist()))
        if math.isfinite(sc) and sc < best_score:
            best = seq
            best_score = sc

    # ------------------------------------------------------------------
    # 1) Simple analytic families -------------------------------------------------
    # Power law (1‑t)^p
    for p in np.arange(0.2, 5.01, 0.2):
        if time.monotonic() >= deadline:
            break
        seq = (1.0 - t) ** p
        try_candidate(seq)

    # Pure exponential e^{-λ t}
    for lam in np.logspace(-2, 2, num=40):
        if time.monotonic() >= deadline:
            break
        seq = np.exp(-lam * t)
        try_candidate(seq)

    # Combined power * exponential
    for p in np.arange(0.5, 4.1, 0.5):
        if time.monotonic() >= deadline:
            break
        for lam in np.logspace(-2, 1, num=20):
            if time.monotonic() >= deadline:
                break
            seq = ((1.0 - t) ** p) * np.exp(-lam * t)
            try_candidate(seq)

    # Gaussian‑like decay e^{-λ t²}
    for lam in np.logspace(-2, 2, num=30):
        if time.monotonic() >= deadline:
            break
        seq = np.exp(-lam * (t ** 2))
        try_candidate(seq)

    # Logistic decreasing 1/(1+e^{k(t‑shift)})
    for k in np.linspace(0.5, 10.0, 15):
        if time.monotonic() >= deadline:
            break
        for shift in np.linspace(0.0, 0.5, 5):
            if time.monotonic() >= deadline:
                break
            seq = 1.0 / (1.0 + np.exp(k * (t - shift)))
            try_candidate(seq)

    # ------------------------------------------------------------------
    # 2) Randomised exploration of combined forms -----------------------------
    while time.monotonic() < deadline:
        # Random parameters within generous bounds
        p = rng.uniform(0.0, 5.0)                # power‑law exponent
        lam = rng.uniform(0.0, 5.0)              # exponential rate
        gamma = rng.uniform(0.5, 2.0)            # stretch exponent
        # Build a versatile candidate: (1‑t)^p * exp(-lam * t^γ)
        seq = ((1.0 - t) ** p) * np.exp(-lam * (t ** gamma))
        try_candidate(seq)
        # Occasionally try pure exponential with a stretched argument
        if rng.random() < 0.3:
            seq = np.exp(-lam * (t ** gamma))
            try_candidate(seq)

    # ------------------------------------------------------------------
    # 3) Local perturbations around the current best -------------------------
    while time.monotonic() < deadline:
        cand = best.copy()
        # Choose a random contiguous block and scale it multiplicatively
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        factor = np.exp(rng.normal(0.0, 0.12, size=stop - start))
        cand[start:stop] *= factor
        # Optional symmetric smoothing (helps sometimes)
        if rng.random() < 0.35:
            cand = 0.5 * (cand + cand[::-1])
        cand = np.clip(cand, 0.0, 1000.0)
        try_candidate(cand)

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python floats)
    return [float(x) for x in best]
