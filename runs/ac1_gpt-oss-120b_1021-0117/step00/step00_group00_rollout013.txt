import numpy as np
import time
import math

def _build_sidon_set(n: int, rng: np.random.Generator, deadline: float,
                     target_c: int = 2) -> list:
    """
    Greedy construction of a Sidon (B2) set:
    all unordered sums i+j (i<=j) are distinct, i.e. each sum appears at most once.
    For a binary vector this guarantees max_b <= target_c (2).
    Returns the list of selected indices.
    """
    max_sum = 2 * n - 1               # possible convolution indices
    used = np.zeros(max_sum, dtype=bool)   # marks which unordered sums are taken
    selected = []                     # list of indices already chosen
    # quick‑access for membership test
    is_selected = np.zeros(n, dtype=bool)

    # theoretical upper bound K such that K(K+1)/2 <= 2n-1  (maximal Sidon size)
    bound = int((-1 + math.isqrt(16 * n - 7)) // 2)

    # try random candidates until no progress is possible or time runs out
    attempts_without_success = 0
    # allow many attempts but stop early if we already reached the bound
    while time.monotonic() < deadline and attempts_without_success < 5000:
        x = rng.integers(0, n)
        if is_selected[x]:
            attempts_without_success += 1
            continue
        # self‑sum must be free
        if used[2 * x]:
            attempts_without_success += 1
            continue
        # check sums with existing indices
        if selected:
            s = np.add(selected, x)          # vector of sums i+x
            if used[s].any():
                attempts_without_success += 1
                continue
        # accept x
        if selected:
            used[np.add(selected, x)] = True
        used[2 * x] = True
        selected.append(int(x))
        is_selected[x] = True
        attempts_without_success = 0
        if len(selected) >= bound:
            break

    return selected


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """
    Returns a list of non‑negative step heights that improves the AC1 upper bound.
    The method builds a large Sidon set (B2) of positions and places a value 1.0
    at each selected index, all other entries being 0.0.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)   # keep a safety margin

    # ------------------------------------------------------------------ #
    # Determine problem size and a safe fallback (the provided parent data)
    # ------------------------------------------------------------------ #
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent.size
    except Exception:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # baseline (fallback) – always valid
    base_seq = np.clip(parent, 0.0, 1000.0)
    best_seq = base_seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ----------------------------------------------------------- #
    # Main search: try several independent Sidon‑set constructions
    # ----------------------------------------------------------- #
    best_k = 0
    while time.monotonic() < deadline:
        cand_set = _build_sidon_set(n, rng, deadline, target_c=2)
        k = len(cand_set)
        if k <= best_k:                     # no improvement in size → unlikely better score
            continue

        seq = np.zeros(n, dtype=np.float64)
        if cand_set:

        if np.isfinite(score) and score < best_score:
            best_seq = seq.copy()
            best_score = score
            best_k = k

        # early exit if we have reached the theoretical maximum size
        bound = int((-1 + math.isqrt(16 * n - 7)) // 2)
        if best_k >= bound:
            break

    # ---------------------------------------------------- #
    # Return the best sequence found (as a plain Python list)
    # ---------------------------------------------------- #
    return [float(v) for v in best_seq]
