def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a non‑negative step‑function minimizing the AC1 upper bound.

    The algorithm starts from a uniform sequence (which attains the theoretical
    minimum 2) and performs cheap stochastic transfers between two positions.
    It respects the evaluation budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # length of the problem – use the provided parent if present
    try:
        n = int(height_sequence_1.shape[0])
        if n <= 0:
            raise Exception
    except Exception:
        n = 256

    # uniform base: all entries equal, which gives the theoretical minimum.
    best = np.full(n, 1.0, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))

    # Simple local Monte‑Carlo search: move a random amount from one entry to another.
    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # amount we can take from i (up to 50 % of its current value)
        cur_i = best[i]
        if cur_i == 0.0:
            continue
        delta = cur_i * rng.random() * 0.5
        cand = best.copy()
        cand[i] = cur_i - delta
        cand[j] = cand[j] + delta
        # enforce bounds required by the evaluator
        if cand[i] < 0.0:
            cand[i] = 0.0
        elif cand[i] > 1000.0:
            cand[i] = 1000.0
        if cand[j] < 0.0:
            cand[j] = 0.0
        elif cand[j] > 1000.0:
            cand[j] = 1000.0

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
