def _analytic_initial(n: int) -> np.ndarray:
    """Generate the truncated coefficients of (1‑x)^‑1/2.
    They satisfy sum_i a_i a_{k-i}=1 for all k≤n‑1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    # recurrence: a_{i}=a_{i-1} * (2*i-1)/(2*i)
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    # normalise to unit sum (scale is irrelevant for the objective)
    s = a.sum()
    if s > 0.0:
        a /= s
    return a
