def fast_score(a):
    n = a.shape[0]

    if s < 1e-12:

    # compute next power of two for length >= 2*n-1
    m = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa*fa, m)
    # conv length m, but real convolution's valid part is first 2*n-1 elements. So take conv[:2*n-1]
    max_b = np.max(conv[:2*n-1])

def propose_candidate(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    n = int(np.asarray(height_sequence_1).size) if 'height_sequence_1' in globals() else 256
    # baseline uniform
    best = np.ones(n, dtype=np.float64)
    best_score = fast_score(best)
    # also possibly keep uniform with sum sqrt(2n) scaling? Not needed.
    
    # precompute m for FFT
    conv_len = 2*n - 1
    fft_len = 1 << ((conv_len).bit_length())
    
    def score_vec(a):
        s = a.sum()
        if s < 1e-12:
            return np.inf
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa*fa, fft_len)
        max_b = conv[:conv_len].max()
        return 2.0 * n * max_b / (s*s)
    
    best_score = score_vec(best)
    # main loop
    while time.monotonic() < deadline:
        # choose candidate generation type with probabilities
        typ = rng.integers(0, 4)
        if typ == 0:
            # geometric
            # sample epsilon in [0.0005, 0.05] (range)
            epsilon = rng.uniform(0.0005, 0.05)
            r = np.exp(-epsilon)
            a = r ** np.arange(n, dtype=np.float64)
        elif typ == 1:
            # power law
            p = rng.uniform(0.2, 5.0)
            a = 1.0 / np.power(np.arange(1, n+1, dtype=np.float64), p)
        elif typ == 2:
            # Dirichlet (random spread)
            a = rng.dirichlet(np.ones(n))
        else:
            # perturbation of best
            a = best.copy()
            # sample a random subsegment and scale
            width = int(rng.integers(max(1, n // 200), max(2, n // 20)))
            start = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.3))
            a[start:start+width] *= factor

            if rng.random() < 0.3:

    # Determine length n and initial sequence

        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)

            raise Exception
        parent_arr = np.clip(parent_arr, 0.0, 1000.0)

        best = parent_arr.copy()
    except Exception:
        n = 256
        best = np.ones(n, dtype=np.float64)

    # Pre compute FFT length

    def compute_score(arr):
        s = arr.sum()
        if s == 0.0:
            return np.inf
        fa = np.fft.rfft(arr, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:conv_len]

import numpy as np
import time
import math

def _conv(arr: np.ndarray, n: int, fft_len: int) -> np.ndarray:
    """Full linear convolution of arr with itself (length 2*n-1) using FFT."""
    fa = np.fft.rfft(arr, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv_full


def _score(arr: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective used by the evaluator (scale‑invariant)."""
    s = arr.sum()
    if s == 0.0:
        return np.inf
    conv = _conv(arr, n, fft_len)
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)


def _subgradient_step(arr: np.ndarray, n: int, fft_len: int, factor: float = 0.2) -> np.ndarray:
    """One subgradient descent step aimed at lowering the peak of the convolution."""
    conv = _conv(arr, n, fft_len)
    k = int(conv.argmax())
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    grad = np.zeros_like(arr)
    if i_start <= i_end:
        inds = np.arange(i_start, i_end + 1, dtype=int)
        partner = k - inds
        grad[inds] = arr[partner]
    step = factor * arr.mean()
    new = arr - step * grad
    np.clip(new, 0.0, 1000.0, out=new)
    tot = new.sum()
    if tot > 0.0:
        new *= (arr.sum() / tot)          # keep the total mass (scale‑invariant)
    return new


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a non‑negative step function that reduces the AC1 upper bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise length and a fallback candidate
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        base = np.clip(base, 0.0, 1000.0)
        n = base.size
        best = base.copy()
    except Exception:
        n = 256
        best = np.ones(n, dtype=np.float64)

    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()   # next power of two

    best_score = _score(best, n, fft_len)
    cur = best.copy()                             # current point for subgradient steps

    # ------------------------------------------------------------------
    # main stochastic search loop
    while time.monotonic() < deadline:
        choice = rng.integers(0, 7)

        if choice == 0:  # simple exponential decay
            eps_tot = rng.uniform(0.4, 2.8)          # total exponent across the array
            L = eps_tot / n
