            candidate = best.copy()

            candidate = np.maximum(0.0, candidate + noise)

        np.clip(candidate, 0.0, 1000.0, out=candidate)
        # Ensure sum > 0.01; if sum too small (unlikely), scale.
        if candidate.sum() < 0.01:
            candidate *= 0.01 / (candidate.sum() + 1e-12)
        # Evaluate
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
    return [float(x) for x in best]

import numpy as np
import time

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Fast linear convolution a*a using a real FFT (O(n log n))."""
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=m)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective used by the authoritative evaluator."""
    n = a.size
    b = _conv_self(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _sqrt_series(n: int) -> np.ndarray:
    """Generate the classic a_i = C(2i,i)/4^i sequence (length n)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        # a[i] = a[i-1] * (2*(i-1)+1) / (2*i)
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return a
