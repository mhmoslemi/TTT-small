def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    try:
        parent_arr = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent_arr.ndim != 1 or parent_arr.size == 0 or not np.isfinite(parent_arr).all():
            raise Exception
    except Exception:
        parent_arr = np.ones(256, dtype=np.float64)

    n = parent_arr.size
    # function for fast score
    size = 1 << ((2 * n - 1).bit_length())
    def fast_score(a):
        # a is np.ndarray of float64, length n
        A = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(A * A, n=size)
        maxb = conv[:2 * n - 1].max()
        s = a.sum()
        if s <= 0:
            return float('inf')
        return 2.0 * n * maxb / (s * s)

    # keep best found (fast)
    best_seq = parent_arr.copy()
    best_score_fast = fast_score(best_seq)

    # Evaluate exact for fallback later
    best_score_exact = float(evaluate_sequence(best_seq.tolist()))

    # Search geometric family
    # Use logarithmic L values
    L_vals = np.logspace(-5, -2.5, num=800)  # from 1e-5 to ~0.003162
    indices = np.arange(n, dtype=np.float64)

        if time.monotonic() >= deadline:
            break
        # compute a = exp(-L * i)
        cand = np.exp(-L * indices)
        # Clip not needed; compute fast score
        sc = fast_score(cand)
        if sc < best_score_fast:

    # Search power family: a_i = exp(- (i/tau)^p )
    for _ in range(200):

        sc = fast_score(cand)

def fast_score(arr, size):
    """Return the AC1 upper bound using an FFT‑based convolution (O(n log n))."""
    A = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(A * A, n=size)
    maxb = conv[:2 * arr.shape[0] - 1].max()
    s = arr.sum()
    if s <= 0.0:
        return float("inf")
    return 2.0 * arr.shape[0] * maxb / (s * s)
