def propose_candidate(seed=42, budget_s=1100, **kwargs):

    # small shift to avoid zeros causing issues; add small epsilon
    eps = 1e-6
    a = a + eps
    a = a / a.sum()
    
    # Evaluate baseline
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    
    # Parameters
    it = 0
    step0 = 0.5  # initial step scaling relative to max gradient magnitude?
    step_decay = 1e-4  # per iteration factor
    temp0 = 0.05
    temp_decay = 0.9999
    temp = temp0

    while time.monotonic() < deadline:

        conv = np.convolve(a, a)
        max_val = conv.max()

        nlen = n
        start = max(0, idx - (nlen - 1))

            grad[i_range] = 2.0 * a[j_range]

import numpy as np
import time
import math

def _random_perturb(a, rng, scale):
    """Multiplicative log‑normal perturbation, keeps positivity."""
    a_new = a * np.exp(scale * rng.normal(size=a.shape))
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a, 1.0 / a.size)
    else:
        a_new /= s
    return a_new

def _targeted_transfer(a, rng):
    """Move a fraction of mass from the most harmful index to a random one."""
    n = a.size
    conv = np.convolve(a, a)
    idx = int(conv.argmax())                     # shift giving the current max
    i = np.arange(n)
    j = idx - i
    valid = (j >= 0) & (j < n)
    prod = np.zeros(n, dtype=a.dtype)
    prod[valid] = a[i[valid]] * a[j[valid]]
    i_max = int(np.argmax(prod))
    # pick a recipient that participates little in the offending shift
    candidates = np.where(valid)[0]
    # exclude i_max from candidates
    candidates = candidates[candidates != i_max]
    if candidates.size == 0:                     # fallback: any other index
        i_min = int(rng.integers(0, n, endpoint=False))
        while i_min == i_max:
            i_min = int(rng.integers(0, n, endpoint=False))
    else:
        i_min = int(candidates[np.argmin(prod[candidates])])
    donor = a[i_max]
    if donor <= 0.0:
        return a.copy()
    eps = min(0.5 * donor, 0.02)                 # move up to 50% or a small absolute amount
    a_new = a.copy()
    a_new[i_max] -= eps
    a_new[i_min] += eps
    return a_new

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimiser for the AC1 bound.
    Returns a list of non‑negative coefficients (length inferred from the parent).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ----- initialise candidate -------------------------------------------------
    if 'height_sequence_1' in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            parent = np.ones(256, dtype=np.float64)
    else:
        parent = np.ones(256, dtype=np.float64)

    # normalise to unit sum (scale does not affect the objective)
    a = parent.copy()
    if a.sum() == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / a.sum()

    # also keep a simple U‑shaped start – often better than a flat start
    n = a.size
    i = np.arange(n, dtype=np.float64)
    u_shape = (np.cos(np.pi * i / (n - 1)) + 1.0) / 2.0      # high at the ends, low in the middle
    u_shape = np.maximum(u_shape, 1e-8)                     # avoid exact zeros
    u_shape = u_shape / u_shape.sum()
    # blend the two: 50 % parent, 50 % U‑shape
    a = 0.5 * a + 0.5 * u_shape
    a = np.maximum(a, 0.0)
    a = a / a.sum()

    # ----- bookkeeping ---------------------------------------------------------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    current_score = best_score

    # annealing schedule
    temp = 0.05
    temp_decay = 0.9995
    perturb_scale = 0.05   # for random log‑normal moves, decays over time
    it = 0

    # ----- optimisation loop ---------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # choose a move
        if rng.random() < 0.7:                # most steps are targeted transfers
            cand = _targeted_transfer(a, rng)
        else:                                 # occasional global jitter
            cand = _random_perturb(a, rng, perturb_scale)

        # evaluate candidate
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score):
            # keep the best ever seen
            if score < best_score:
                best, best_score = cand.copy(), score

            # simulated‑annealing acceptance for the current walk
            delta = score - current_score
            if delta < 0.0 or rng.random() < math.exp(-delta / max(temp, 1e-12)):
                a, current_score = cand, score

        # schedule updates
        temp = max(temp * temp_decay, 1e-9)
        perturb_scale *= 0.9999

        # occasional random reshuffle to escape plateaus
        if it % 2500 == 0 and rng.random() < 0.25:
            shuffle = rng.random(a.shape)
            a = a * shuffle
            a = np.clip(a, 0.0, None)
            if a.sum() == 0.0:
                a = np.full_like(a, 1.0 / a.size)
            else:
                a /= a.sum()
            current_score = float(evaluate_sequence(a.tolist()))
            if current_score < best_score:
                best, best_score = a.copy(), current_score

    # final safety clamp (values are already ≤ 1000, non‑negative)
    result = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in result]
