import numpy as np
import time

def _exp_sequence(length: int, a0: float, r: float) -> np.ndarray:
    """Generate a non‑negative step function a_i = a0 * r**i (0 ≤ i < length)."""
    idx = np.arange(length, dtype=np.float64)
    seq = a0 * np.power(r, idx)
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq

def _score(seq: np.ndarray) -> float:
    """Evaluate the AC1 upper bound using the provided authoritative evaluator."""
    # `evaluate_sequence` expects a plain Python list.
    return float(evaluate_sequence(seq.tolist()))
