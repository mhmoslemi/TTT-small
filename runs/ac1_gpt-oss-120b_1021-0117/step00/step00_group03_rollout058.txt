import numpy as np, time, cvxpy as cp

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0) if 'height_sequence_1' in globals() else np.ones(256, dtype=np.float64)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    n = best.size

    # Function to run one LP iteration
    def lp_step(f, eps=1e-6):
        b = np.convolve(f, f)
        M = b.max()
        # Choose active ks
        # take ks where b close to M within tolerance or top K
        # We'll take up to max_active = 7 largest ks
        # get sorted indices descending
        ks = np.argpartition(b, -7)[-7:]  # get 7 largest indices (not sorted)
        # optionally filter to those near M
        # Build constraint coeffs list
        coeffs = []
        for k in ks:
            # build coefficient vector c where c[j] = f[k - j] if in range else 0
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            # length high - low + 1
            # We'll allocate zeros then fill slice
            c = np.zeros(n, dtype=np.float64)
            # j ranges from low to high, where i = k - j is in [0, n-1]
            # Set c[j] = f[i]
            idx_j = np.arange(low, high + 1)
            idx_i = k - idx_j
            c[idx_j] = f[idx_i]
            coeffs.append(c)
        # define cvxpy variable
        g = cp.Variable(n, nonneg=True)
        constraints = [cp.sum(cp.multiply(c, g)) <= M for c in coeffs]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception:
            return None, None
        if g.value is None:
            return None, None
        g_vec = np.asarray(g.value, dtype=np.float64)
        return g_vec, M

    t = 0.3  # mixing fraction
    while time.monotonic() < deadline:
        # try LP step
        g_vec, M = lp_step(best)
        if g_vec is not None:
            # mix and renormalize
            cand = (1 - t) * best + t * g_vec
            # renormalize to keep sum same as best
            sum_best = best.sum()
            sum_cand = cand.sum()
            if sum_cand > 0:
                cand = cand * (sum_best / sum_cand)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score
                continue  # go to next iteration quickly after improvement
        # fallback random block perturbation (similar to previous code)
        # generate random block perturbation centered on best
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        candidate = best.copy()
        candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
        if rng.random() < 0.35:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = np.clip(candidate, 0.0, 1000.0)
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
        # optionally random shuffle of entries
        if rng.random() < 0.1:
            shuffled = best.copy()
            rng.shuffle(shuffled)
            score2 = float(evaluate_sequence(shuffled.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best = shuffled
                best_score = score2
    # return best as list of floats
    return [float(x) for x in best]
