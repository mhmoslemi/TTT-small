    pos_set = []
    sums = set()
    # iterate candidates from 0..n-1, add if its sums with existing positions (including self sum) are unique
    for candidate in range(n):
        conflict = False
        # compute sums candidate+candidate and candidate+existing

import time
import numpy as np

def _conv_score(arr: np.ndarray, n: int) -> float:
    """Return the exact score for a non‑negative array `arr`."""
    s = arr.sum()
    if s < 0.01:
        return float('inf')
    conv = np.convolve(arr, arr)
    return 2.0 * n * conv.max() / (s * s)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step‑function that lowers the AC1 upper bound.

    Uses a geometric‑decay family a_i = r**i (r<1) and a short
    stochastic mass‑transfer hill‑climb.  All work is O(n) memory.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 0.5

    # length of the parent (or a default length)
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise Exception()
        n = base.size
    except Exception:
        n = 256

    # fallback: uniform vector (score = 2)
    best_arr = np.ones(n, dtype=np.float64) / n
    best_score = _conv_score(best_arr, n)

    # ---------- grid search over geometric decay ----------
    # t = 1 - r  ;  r in (0.99, 1)  ->  t in [1e-6, 1e-2]
    t_vals = np.logspace(-6, -2, num=300)  # 300 points gives ~0.2 % granularity
    idx = np.arange(n, dtype=np.float64)

    for t in t_vals:
        if time.monotonic() > deadline:
            break
        r = 1.0 - t
        # a_i = r**i  (vectorised)
        arr = np.power(r, idx, dtype=np.float64)
        score = _conv_score(arr, n)
        if score < best_score:
            best_arr = arr
            best_score = score

    # ---------- optional local refinement ----------
    # simple random mass transfer while time permits
    arr = best_arr.copy()
    while time.monotonic() < deadline:
        # pick donor with some mass and a random receiver
        donor = rng.integers(0, n)
        if arr[donor] <= 1e-12:
            continue
        receiver = rng.integers(0, n)
        if donor == receiver:
            continue
        # move a small fraction of donor's mass
        frac = rng.random() * 0.2 + 0.01          # between 1 % and 21 %
        delta = arr[donor] * frac
        cand = arr.copy()
        cand[donor] -= delta
        cand[receiver] += delta
        # keep within allowed range (0…1000)
        np.clip(cand, 0.0, 1000.0, out=cand)

        score = _conv_score(cand, n)
        if score < best_score:
            arr = cand
            best_arr = cand
            best_score = score

    # return a plain Python list as required by the evaluator
    return [float(v) for v in best_arr]
