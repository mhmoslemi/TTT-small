    # Uniform constant (any positive value works; pick 1.0 for simplicity)
    a = np.full(n, 1.0, dtype=np.float64)

    # Ensure we respect the evaluator's clipping bounds.
    a = np.clip(a, 0.0, 1000.0)

    return a.tolist()
