def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)
    # base initial candidate: exponential shape
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 256
    # Determine t approx 1.256/n to start
    t_opt = 1.256 / n
    r_opt = np.exp(-t_opt)
    # Using start value 1000
    exp_seq = 1000.0 * np.exp(-t_opt * np.arange(n))
    # Clip
    seq = np.clip(exp_seq, 0.0, 1000.0)
    # Ensure sum >0
    if seq.sum() < 0.01:
        seq = np.ones(n) * 0.1
    best = seq.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # Also evaluate parent if exists to have fallback (maybe better)
    if 'height_sequence_1' in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            parent_score = float(evaluate_sequence(parent.tolist()))
            if parent_score < best_score:
                best = parent.copy()
                best_score = parent_score
    # hill climbing
    # parameters for hill climbing
    max_no_improve = 2000
    no_improve_counter = 0
    while time.monotonic() < deadline:
        # pick random pair i, j
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # amount to transfer
        # ensure we move at most a fraction of available mass
        max_transfer = min(best[i], 1000 - best[j])
        if max_transfer <= 0:
            continue
        # choose delta random in (0, max_transfer * 0.5)
        delta = max_transfer * rng.random() * 0.5
        # propose new
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # evaluate
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score - 1e-12:  # improvement
            best = cand
            best_score = score
            no_improve_counter = 0
        else:
            no_improve_counter += 1
        # optional: if many iterations without improvement, try random restart using exponential shape
        if no_improve_counter > max_no_improve:
            # random perturbation of best
            cand = best.copy()
            width = max(1, n // 200)
            start = rng.integers(0, max(1, n - width))
            cand[start:start+width] *= np.exp(rng.normal(0.0, 0.2))
            cand = np.clip(cand,0,1000)
            # optionally renormalize sum to maintain scaling invariance? Not needed
            cand_score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(cand_score) and cand_score < best_score:
                best = cand
                best_score = cand_score
            no_improve_counter = 0
    return [float(x) for x in best]

            if best_score < 1.5028:  # just before target
                break
        # occasional local perturbation of best (like add small noise)

            w = rng.integers(1, max(2, n//50))

import time
import numpy as np

# fast objective that mimics evaluate_sequence but works on np.ndarray
def _bound_from_array(arr: np.ndarray) -> float:
    """Return the AC1 upper bound for a non‑negative array (no clipping needed)."""
    n = arr.shape[0]
    # convolution of the vector with itself (full mode)
    conv = np.convolve(arr, arr)
    max_conv = conv.max()
    s = arr.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_conv / (s * s))


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Generate a non‑negative step‑function that (hopefully) beats the
    target bound 1.50274349.  The routine runs a cheap random search over a few
    parametric families (exponential, power‑law, polynomial decay) and then
    performs a small hill‑climbing phase on the best candidate.
    """
    rng = np.random.default_rng(seed)
    # length of the problem (use the provided parent if present)
    n = (
        int(len(height_sequence_1))
        if "height_sequence_1" in globals()
        else 256
    )
    idx = np.arange(n, dtype=np.float64)

    # start from a very simple fallback – the existing parent if it exists
    best = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if best.ndim != 1 or best.size != n or not np.isfinite(best).all():
        best = np.ones(n, dtype=np.float64)
    best_score = _bound_from_array(best)

    # also try a plain exponential shape as a deterministic seed
    t0 = 1.256 / n                     # near the analytic optimum for a pure exp.
    cand0 = np.exp(-t0 * idx)
    cand0_score = _bound_from_array(cand0)
    if cand0_score < best_score:
        best, best_score = cand0, cand0_score

    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ---------- random parametric search ----------
    while time.monotonic() < deadline:
        # pick a family
        fam = rng.integers(0, 3)
        if fam == 0:                     # exponential decay
            t = rng.uniform(5e-05, 5e-03)   # decay rate
            a = np.exp(-t * idx)
        elif fam == 1:                   # power‑law decay
            alpha = rng.uniform(0.05, 2.5)   # exponent
            a = (idx + 1.0) ** (-alpha)
        else:                            # polynomial (1 - (i/(n‑1))**p)
            p = rng.uniform(0.2, 6.0)
            a = 1.0 - (idx / (n - 1)) ** p
            a = np.clip(a, 0.0, 1.0)

        # quick sanity – sum must be > 0.01 (otherwise evaluator returns inf)
        if a.sum() < 0.01:
            continue

        score = _bound_from_array(a)
        if score < best_score:
            best, best_score = a, score
            # stop early if we already beat the target
            if best_score < 1.5028:
                break

        # occasional local tweak of the current best
        if rng.random() < 0.07:
            tmp = best.copy()
            w = rng.integers(1, max(2, n // 30))
            start = rng.integers(0, n - w + 1)
            factor = np.exp(rng.normal(0.0, 0.12))
            tmp[start:start + w] *= factor
            tmp = np.clip(tmp, 0.0, 1.0)
            if tmp.sum() >= 0.01:
                s2 = _bound_from_array(tmp)
                if s2 < best_score:
                    best, best_score = tmp, s2

    # ---------- final hill‑climbing on the best candidate ----------
    # simple transfer moves that keep the total sum unchanged
    transfers = 0
    while time.monotonic() < deadline and transfers < 3500:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # amount we can move from i to j without violating bounds
        max_move = min(best[i], 1.0 - best[j])
        if max_move <= 0.0:
            continue
        delta = max_move * rng.random() * 0.6
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        s = _bound_from_array(cand)
        if s < best_score:
            best, best_score = cand, s
        transfers += 1

    # Convert to list of Python floats for the official evaluator
    return [float(v) for v in best]
