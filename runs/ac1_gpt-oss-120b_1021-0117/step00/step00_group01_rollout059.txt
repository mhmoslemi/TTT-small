def _make_initial_sequence(n):
    """Power‑law (α=0.5) decreasing sequence a_i = 1/√(i+1)."""
    idx = np.arange(n, dtype=np.float64)
    # avoid division by zero: i+1 >= 1

    """Convenient wrapper around the provided evaluator."""
    # evaluate_sequence is supplied in the execution environment
    return float(evaluate_sequence(seq.tolist()))


def _perturb(seq, rng):
    """Return a small multiplicative block perturbation of ``seq``."""
    seq = seq.copy()
    n = seq.size
    # random block width (at least 1, at most n//10)
    width = int(rng.integers(1, max(2, n // 10)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = math.exp(rng.normal(0.0, 0.06))
    seq[start:stop] *= factor
    # occasional global jitter
    if rng.random() < 0.35:
        seq += rng.normal(scale=0.015, size=n)
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound computed by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Determine problem size
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256                     # fallback size
    # ------------------------------------------------------------------
    # Initial candidates
    best_seq = None
    best_score = float("inf")

    # baseline uniform (always feasible)
    uniform = np.ones(n, dtype=np.float64)
    best_score = _eval(uniform)
    best_seq = uniform.copy()

    # ------------------------------------------------------------------
    # Grid search over simple parametric families
    idx = np.arange(n, dtype=np.float64)

    # (i+offset)^(-alpha) * r**i  families
    alphas = [0.45, 0.48, 0.50, 0.52, 0.55]
    offsets = [0.9, 1.0, 2.0, 5.0]          # offset > 0 to avoid singularity
    r_vals = np.linspace(0.990, 0.9999, 20)  # exponential decay factors

    for alpha in alphas:
        for offset in offsets:
            base_pow = (idx + offset) ** (-alpha)   # power‑law part
            for r in r_vals:
                if time.monotonic() > deadline:
                    break
                if abs(r - 1.0) < 1e-12:
                    cand = base_pow
                else:
                    cand = base_pow * (r ** idx)
                np.clip(cand, 0.0, 1000.0, out=cand)
                score = _eval(cand)
                if score < best_score:
                    best_score = score
                    best_seq = cand.copy()
            if time.monotonic() > deadline:
                break
        if time.monotonic() > deadline:
            break

    # Also try the pure α=0.5 power‑law (no exponential factor)
    if time.monotonic() < deadline:
        cand = _make_initial_sequence(n)
        score = _eval(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()

    # ------------------------------------------------------------------
    # Local random search (hill climbing) starting from the best found
    while time.monotonic() < deadline:
        candidate = _perturb(best_seq, rng)
        score = _eval(candidate)
        if score < best_score:
            best_score = score
            best_seq = candidate

    # Ensure a valid non‑empty return
    if best_seq is None or best_seq.size == 0:
        best_seq = np.ones(n, dtype=np.float64)

    return [float(x) for x in best_seq]
