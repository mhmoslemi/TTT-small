    indices = list(rng.permutation(n)) # random order
    selected = []
    sum_counts = {} dict mapping sum-> count (int)
    for i in indices:
        # compute new sums if i added
        ok = True
        if max_mult == 0: maybe treat no collisions at all? but we allow up to max_mult per sum.
        # Check for each j in selected:
        for j in selected:
            s = i + j
            cnt = sum_counts.get(s,0)
            if cnt >= max_mult:
                ok = False
                break
        if not ok: continue
        # also check self sum (i+i)
        s2 = i + i
        cnt2 = sum_counts.get(s2,0)
        if cnt2 >= max_mult:
            continue
        # accept
        for j in selected:
            s = i + j
            sum_counts[s] = sum_counts.get(s,0) + 1
        sum_counts[s2] = sum_counts.get(s2,0) + 1
        selected.append(i)
        if len(selected) >= target_m:
            break
    return selected

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: greedy construction of a sparse index set with a bound on the
# ordered convolution count for each possible sum (i+j).
# ----------------------------------------------------------------------
def greedy_set(n: int, L: int, rng: np.random.Generator):
    """
    Build a set I ⊂ {0,…,n‑1} such that for any k the ordered count
    of pairs (i,j)∈I×I with i+j=k never exceeds L.
    Returns (list_of_indices, max_ordered_count).
    """
    selected = []                     # chosen indices
    ordered = {}                      # sum → ordered pair count
    # random permutation of all positions
    perm = rng.permutation(n)
    for i in perm:
        # check self‑pair contribution
        s0 = i + i
        if ordered.get(s0, 0) + 1 > L:
            continue
        # check contributions with the already chosen indices
        ok = True
        for j in selected:
            s = i + j
            if ordered.get(s, 0) + 2 > L:
                ok = False
                break
        if not ok:
            continue
        # accept i : update the counters
        ordered[s0] = ordered.get(s0, 0) + 1
        for j in selected:
            s = i + j
            ordered[s] = ordered.get(s, 0) + 2
        selected.append(i)
    max_ord = max(ordered.values()) if ordered else 0
    return selected, max_ord

    except Exception:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # baseline (uniform over all positions) – always valid
    best_arr = parent.copy()
    best_score = float(evaluate_sequence(best_arr.tolist()))

    # ------------------------------------------------------------------
    # Search: try several capacity limits L and random greedy constructions.
    # ------------------------------------------------------------------
    # We aim for a sparse index set; the bound on the ordered convolution
    # count (L) directly controls the objective:
    #   score = 2 * n * L / m^2   (with equal weights 1/m)
    # where m = |I|.
    L_values = [4, 5, 6, 7, 8, 9, 10]
    attempts_per_L = 30

    for L in L_values:
        for _ in range(attempts_per_L):
            if time.monotonic() >= deadline:
                break

            indices, max_ord = greedy_set(n, L, rng)
            m = len(indices)
            if m < 2:                     # need at least two points to beat uniform
                continue

            # quick theoretical score (exact because weights are equal)
            theo_score = 2.0 * n * max_ord / (m * m)

            if theo_score < best_score:
                # construct the actual sequence (all non‑zero entries equal 1/m)
                arr = np.zeros(n, dtype=np.float64)
                val = 1.0 / m
                arr[indices] = val
                # final authoritative check
                real_score = float(evaluate_sequence(arr.tolist()))
                if np.isfinite(real_score) and real_score < best_score:
                    best_arr = arr
                    best_score = real_score

        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python floats).
    # ------------------------------------------------------------------
    return [float(x) for x in best_arr]
```
