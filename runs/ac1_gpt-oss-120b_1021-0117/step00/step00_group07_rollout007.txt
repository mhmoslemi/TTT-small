def build_sparse_set(N, target_M, max_mult=4, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    S = []
    counts = {}  # sum -> ordered count
    attempts = 0
    while len(S) < target_M and attempts < 10 * target_M:
        idx = rng.integers(0, N)
        # compute contributions:
        ok = True
        local_counts = {}
        # contribution from self-pair
        s_self = 2*idx
        new_self = counts.get(s_self, 0) + 1
        if new_self > max_mult: ok = False
        else: local_counts[s_self] = new_self
        # contributions from existing indices
        for i in S:
            s = i + idx
            new_val = counts.get(s, 0) + 2
            if new_val > max_mult: ok = False; break
            else: local_counts[s] = new_val
        if not ok:
            attempts += 1
            continue
        # accept
        S.append(idx)
        for s, v in local_counts.items():
            counts[s] = v
        attempts = 0 # reset attempts after success (optional)

import numpy as np
import time
import math

def _build_sparse_set(N, M_target, max_mult, rng):
    """Greedy construction of a set S⊂{0,…,N‑1} of size M_target.
    For every sum s = i+j (ordered pairs) the total count of contributions
    (1 for i=j, 2 for i≠j) never exceeds max_mult.
    Returns the list of indices and the actual maximal count encountered.
    """
    S = []
    S_set = set()
    cnt = {}               # sum → ordered count
    attempts = 0
    # allow many tries before giving up on this target size
    max_attempts = M_target * 30
    while len(S) < M_target and attempts < max_attempts:
        idx = int(rng.integers(0, N))
        if idx in S_set:
            attempts += 1
            continue

        # prospective updates
        new_counts = {}
        s_self = idx + idx
        v_self = cnt.get(s_self, 0) + 1
        if v_self > max_mult:
            attempts += 1
            continue
        new_counts[s_self] = v_self

        ok = True
        for i in S:
            s = i + idx
            v = cnt.get(s, 0) + 2       # (i,idx) and (idx,i)
            if v > max_mult:
                ok = False
                break
            new_counts[s] = v
        if not ok:
            attempts += 1
            continue

        # accept this index
        S.append(idx)
        S_set.add(idx)
        cnt.update(new_counts)
        attempts = 0                     # reset counter after a success

    if len(S) < M_target:
        return None, None
    maxcnt = max(cnt.values()) if cnt else 0
    return S, maxcnt

    # Length of the problem (fallback if the parent is missing)

    except Exception:
        N = 256

    # Uniform vector as a safe fallback (score = 2.0)
