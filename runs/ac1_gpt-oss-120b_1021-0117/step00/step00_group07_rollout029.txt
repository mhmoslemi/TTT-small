def _build_sparse_conv_matrix(f_vec):
    """
    Build sparse matrix A where (A @ g)[k] = (f * g)[k] (full convolution).
    Only non‑zero entries of f are used.
    """
    n = f_vec.size
    nz_idx = np.nonzero(f_vec)[0]
    if nz_idx.size == 0:
        # empty matrix (will never be used)
        return csr_matrix((2 * n - 1, n), dtype=np.float64)

    # each non‑zero f[i] contributes a shifted copy of g across rows i … i+n-1
    rows = (nz_idx[:, None] + np.arange(n, dtype=np.int64)[None, :]).ravel()
    cols = np.tile(np.arange(n, dtype=np.int64), nz_idx.size)
    data = np.repeat(f_vec[nz_idx], n)

    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n), dtype=np.float64)

    # ----------------------------------------------------------------------
    # Base problem size and simple decreasing step‑function (the “root” candidate)
    # ----------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = int(np.asarray(height_sequence_1).size)
    else:
        n = 256

    max_val = 1000.0
    base = max_val / np.sqrt(np.arange(n, dtype=np.float64) + 1.0)
    base = np.clip(base, 0.0, max_val)

    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ----------------------------------------------------------------------
    # One LP‑based improvement (sparse parent → dense improvement)
    # ----------------------------------------------------------------------
    if _has_cvxpy and time.monotonic() < deadline - 30.0:
        # keep only a modest number of the largest coefficients (they are
        # already the first entries because of the monotone shape)
        m = min(150, max(5, n // 30))
        idx = np.arange(m, dtype=np.int64)                     # first m indices
        f_sparse = np.zeros(n, dtype=np.float64)
        f_sparse[idx] = base[idx]

        # bound on the convolution of the sparse vector with itself
        conv_f = np.convolve(f_sparse, f_sparse)
        M = float(np.max(conv_f))

        if M > 0.0:
            A = _build_sparse_conv_matrix(f_sparse)

            # LP: maximise sum(g)  s.t.  g >= 0  and  A @ g <= M
            g = cp.Variable(n, nonneg=True)
            constraints = [A @ g <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

            try:
                prob.solve(solver='ECOS', warm_start=True,
                           max_iters=2500, abstol=1e-5, reltol=1e-5)
            except Exception:  # pragma: no cover
                prob = None

            if prob is not None and g.value is not None:
                g_val = np.maximum(0.0, np.asarray(g.value).ravel())
                # convex combination (t = 0.5 gives a good trade‑off)
                t_mix = 0.5
                cand = (1.0 - t_mix) * f_sparse + t_mix * g_val
                cand = np.clip(cand, 0.0, max_val)
                cand_score = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(cand_score) and cand_score < best_score:
                    best = cand
                    best_score = cand_score

    # ----------------------------------------------------------------------
    # Local stochastic improvement (block scalings + tiny mass transfers)
    # ----------------------------------------------------------------------
    iter_cnt = 0
    while time.monotonic() < deadline:
        iter_cnt += 1
        # slowly shrink perturbation magnitude as time goes on
        progress = (time.monotonic() - start_time) / max(1e-12, deadline - start_time)
        sigma = 0.22 * (1.0 - progress) + 0.03 * progress   # from ~0.22 → 0.03
        candidate = best.copy()

        if rng.random() < 0.73:                     # block scaling
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = start + width
            factor = math.exp(rng.normal(0.0, sigma))
            candidate[start:stop] *= factor
            if rng.random() < 0.33:                 # occasional mirroring
                candidate = 0.5 * (candidate + candidate[::-1])
        else:                                        # tiny mass transfer
            i, j = rng.choice(n, size=2, replace=False)
            if candidate[i] > candidate[j]:
                high, low = i, j
            else:
                high, low = j, i
            if candidate[high] > 0.0:
                delta = rng.uniform(0.0, 0.1 * candidate[high])
                candidate[high] -= delta
                candidate[low] += delta

        candidate = np.clip(candidate, 0.0, max_val)
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score

    return [float(x) for x in best]
