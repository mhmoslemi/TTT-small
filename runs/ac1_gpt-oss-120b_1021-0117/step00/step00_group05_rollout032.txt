def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    start_time = time.monotonic()
    rng = np.random.default_rng(seed)

        if (
            parent_arr.ndim == 1
            and parent_arr.size > 0
            and np.isfinite(parent_arr).all()

        else:
            n = None
    except Exception:  # height_sequence_1 may be absent
        n = None

    if n is None:
        # fallback length when no parent is available
        n = 256

    # ----- arithmetic decreasing sequence ----------------------------------
    idx = np.arange(n, dtype=np.float64)                # 0 … n‑1
    seq = 2.0 * (n - idx) / (n * (n + 1))               # a_i = 2·(n‑i)/(n·(n+1))
    seq = np.clip(seq, 0.0, None)                      # ensure non‑negative
    tot = seq.sum()
    if tot > 0.0:
        seq /= tot                                      # normalise to sum = 1
    else:                                               # never happens, but keep safe
        seq = np.ones(n, dtype=np.float64) / n

    if 'parent_arr' in locals():
