def make_block_perturbation(base, rng):
    candidate = np.array(base, copy=True)
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))

import time
import numpy as np

def make_block_perturbation(base, rng):
    """Return a copy of ``base`` with a random contiguous block scaled."""
    cand = np.array(base, copy=True)
    n = cand.size
    # block width between n/200 and n/12 (at least 1)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    # random multiplicative factor
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional symmetrisation (helps a bit on some instances)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound defined by ``evaluate_sequence``.
    """

    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    #  initialise a parent (fallback) sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.clip(
            np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0
        )
        n = parent.size

    best = parent.copy()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
