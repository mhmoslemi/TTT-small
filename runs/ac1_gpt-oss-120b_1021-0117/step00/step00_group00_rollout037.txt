def conv_max(a):
    n = a.size
    L = 1 << ((2*n -1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    max_val = conv.max()
    argmax = int(conv.argmax())
    return max_val, argmax, conv

import time
import numpy as np

def _conv_max(arr: np.ndarray):
    """Return (max_convolution_value, argmax_index) for a real sequence."""
    n = arr.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n‑1
    fa = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_val = conv.max()
    max_idx = int(conv.argmax())
    return max_val, max_idx


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Generate a non‑negative step‑function (list of floats) whose AC1 upper bound
    is (hopefully) below the reference value 1.50274349.

    The routine works with a simple hill‑climbing strategy that repeatedly
    moves a small amount of mass from indices that contribute most to the
    currently maximal convolution entry to indices that contribute least.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # --------------------------------------------------------------
    # initialise a sequence of the same length as the provided parent
    # (height_sequence_1) – if unavailable we fall back to length 256.
    # --------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent.size
    except Exception:                     # height_sequence_1 may be missing
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # start from a random positive vector and normalise its L1‑norm to n
    a = rng.random(n) + 0.01               # avoid exact zeros
    a = a * (n / np.sum(a))                # average value ≈ 1

    # interleave large and small values – a cheap “anti‑clustering”
    # heuristic that already reduces the initial peak.
    vals = np.sort(a)[::-1]                # descending
    a = np.empty_like(vals)
    l, r = 0, n - 1
    for idx, v in enumerate(vals):
        if idx & 1:
            a[r] = v
            r -= 1
        else:
            a[l] = v
            l += 1

    # --------------------------------------------------------------
    # main hill‑climbing loop
    # --------------------------------------------------------------
    best_max, best_k = _conv_max(a)
    best_a = a.copy()
    target_score = 1.50274349

    while time.monotonic() < deadline:
        N = best_a.shape[0]
        k = best_k

        # indices that actually appear in the maximal convolution term
        i0 = max(0, k - (N - 1))
        i1 = min(k, N - 1)
        i_idx = np.arange(i0, i1 + 1, dtype=np.int64)
        j_idx = k - i_idx
        contrib = best_a[i_idx] * best_a[j_idx]

        # indices with largest / smallest contribution to this peak
        imax = i_idx[contrib.argmax()]
        imin = i_idx[contrib.argmin()]

        # move a modest fraction of mass from imax to imin
        delta = best_a[imax] * 0.20                 # 20 % transfer
        if delta <= 0:

        cand = best_a.copy()
        cand[imax] -= delta
        cand[imin] += delta

        new_max, new_k = _conv_max(cand)

        if new_max < best_max:
            best_a = cand
            best_max = new_max
            best_k = new_k
            # early exit if we already beat the target
            if 2.0 * N * best_max / (np.sum(best_a) ** 2) < target_score:
