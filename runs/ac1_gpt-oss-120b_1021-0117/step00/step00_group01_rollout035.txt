def generate_sidon_set(n, rng, max_trials=10):
    best_set = []
    for t in range(max_trials):
        candidates = list(range(n))
        rng.shuffle(candidates)
        S = []
        sum_counts = np.zeros(2*n-1, dtype=np.int8)
        for i in candidates:
            # compute sums with S
            ok = True
            # check self sum
            if sum_counts[i*2] >= 2: # but Sidon self sum 1 allowed
                ok = False
            else:
                # check with each s in S
                for j in S:
                    idx = i + j
                    if sum_counts[idx] >= 2:
                        ok = False
                        break
                # if still ok, also need to check that new sums will not exceed 2
            if ok:
                # update

def build_set(n, c, rng):
    indices = np.arange(n)
    rng.shuffle(indices)
    selected = []
    sum_counts = np.zeros(2*n-1, dtype=np.int8)
    for i in indices:
        # check self sum
        idx_self = i*2
        if sum_counts[idx_self] + 1 > c:
            continue
        ok = True
        for j in selected:
            s = i + j
            if sum_counts[s] + 2 > c:
                ok = False
                break
        if not ok:
            continue
        # add
        selected.append(i)
        sum_counts[idx_self] += 1
        for j in selected[:-1]:  # all previous existing j before addition
            sum_counts[i + j] += 2
    return selected

import numpy as np
import time

# fast local score (same formula as the evaluator, but without the safety checks)
def _local_score(seq: np.ndarray) -> float:
    n = seq.size
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = np.convolve(seq, seq)
    return float(2.0 * n * conv.max() / (s * s))


# greedy construction of a set where each sum i+j (ordered) appears at most `c` times
def _build_limited_sum_set(n: int, c: int, rng: np.random.Generator) -> list:
    order = np.arange(n, dtype=np.int32)
    rng.shuffle(order)
    chosen = []
    # ordered‑pair count for each possible sum (0 .. 2n-2)
    sum_cnt = np.zeros(2 * n - 1, dtype=np.int8)

    for i in order:
        # self‑pair contributes 1 ordered pair
        self_idx = i + i
        if sum_cnt[self_idx] + 1 > c:
            continue
        # check all pairs with already chosen elements (adds 2 ordered pairs each)
        good = True
        for j in chosen:
            s = i + j
            if sum_cnt[s] + 2 > c:
                good = False
                break
        if not good:
            continue

        # accept i
        chosen.append(i)
        sum_cnt[self_idx] += 1
        for j in chosen[:-1]:          # only with the older elements
            sum_cnt[i + j] += 2
    return chosen
