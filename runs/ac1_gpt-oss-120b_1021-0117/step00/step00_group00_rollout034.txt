def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 4806
    # base candidate: power law p=0.5, beta=0.5
    p = 0.5
    beta = 0.5
    a = (np.arange(n, dtype=np.float64) + beta) ** (-p)
    a = np.clip(a, 0.0, 1000.0)
    # optional scaling: none
    def conv_max(x):
        # compute convolution via FFT
        m = 1 << ((2*x.size - 1).bit_length())
        X = np.fft.rfft(x, m)
        y = np.fft.irfft(X * X, m)[:2*x.size-1]
        return y.max()
    def fast_score_vec(x):
        maxb = conv_max(x)
        s = x.sum()
        return 2 * n * maxb / (s * s)
    best_vec = a.copy()
    best_score = fast_score_vec(best_vec)
    # also evaluate with exact for fallback reference
    best_exact = float(evaluate_sequence(best_vec.tolist()))
    # search loops
    while time.monotonic() < deadline:
        # random choose modification type
        if rng.random() < 0.7:
            # perturb parameters p, beta
            p_candidate = np.clip(p + rng.normal(scale=0.05), 0.3, 0.7)
            beta_candidate = max(1e-6, beta * np.exp(rng.normal(scale=0.1)))

        score = fast_score_vec(cand)

            best_vec = cand

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 4806

    # base shape
    beta = 1.0
    p = 0.5
    a = (np.arange(n, dtype=np.float64) + beta) ** (-p)
    a = np.clip(a, 0.0, 1000.0)

    # compute fast score
    def compute_max_conv(x):
        m = 1 << ((2*x.size - 1).bit_length())
        X = np.fft.rfft(x, m)
        conv = np.fft.irfft(X * X, m)[:2*x.size-1]
        return conv.max()
    def fast_score(x):
        maxb = compute_max_conv(x)
        s = x.sum()
        return 2.0 * n * maxb / (s * s)

    best = a.copy()
    best_score = fast_score(best)
    # optionally evaluate exact candidate for fallback
    best_exact_score = float(evaluate_sequence(best.tolist()))

    # helper for block perturb
    def perturb(x):
        y = x.copy()
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        factor = math.exp(rng.normal(scale=0.12))
        y[start:start+width] *= factor
        # occasional symmetrization
        if rng.random() < 0.35:
            y = 0.5 * (y + y[::-1])
        np.clip(y, 0.0, 1000.0, out=y)
        return y

    # search loop
    while time.monotonic() < deadline:
        # randomly either adjust parameters or perturb vector
        if rng.random() < 0.3:
            # modify parameters p, beta
            p_candidate = np.clip(p + rng.normal(scale=0.02), 0.3, 0.7)
            beta_candidate = max(1e-6, beta * math.exp(rng.normal(scale=0.05)))
            cand = (np.arange(n, dtype=np.float64) + beta_candidate) ** (-p_candidate)
            cand = np.clip(cand, 0.0, 1000.0)
        else:
            cand = perturb(best)
        score = fast_score(cand)
        if score < best_score:
            best = cand
            best_score = score
            # optionally keep exact
            if time.monotonic() + 0.5 < deadline:
                exact = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(exact):
                    best_exact_score = exact
    # final evaluation with exact evaluator
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        # fallback
        fallback = np.ones(n, dtype=np.float64)
        fallback = np.clip(fallback, 0.0, 1000.0)
        return [float(x) for x in fallback]
    return [float(x) for x in best]

import numpy as np
import time
import math

def _next_pow2(x: int) -> int:
    """Small helper: next power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_max_fft(a: np.ndarray, mfft: int) -> float:
    """Return max value of the linear convolution a*a using an FFT of size mfft."""
    # real FFT, square, inverse FFT
    A = np.fft.rfft(a, mfft)
    conv = np.fft.irfft(A * A, mfft)[: 2 * a.size - 1]
    return float(conv.max())


def _fast_score(a: np.ndarray, n: int, mfft: int) -> float:
    """Fast surrogate for evaluate_sequence (no clipping, no safety checks)."""
    maxb = _conv_max_fft(a, mfft)
    s = float(a.sum())
    return 2.0 * n * maxb / (s * s)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step‑function whose AC1 upper bound is low.
    The routine respects the time budget, keeps memory O(n) and returns a
    concrete list of coefficients.
    """
    rng = np.random.default_rng(seed)

    # length of the problem (use the provided parent if present)
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 4806

    # FFT size (next power of two) – reused for all fast evaluations
    mfft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------ #
    # initialise with a power‑law vector a_i = (i+beta)^-p (p≈0.5)
    p = 0.5
    beta = 1.0
    a_base = (np.arange(n, dtype=np.float64) + beta) ** (-p)
    a_base = np.clip(a_base, 0.0, 1000.0)

    best_vec = a_base.copy()
    best_score = _fast_score(best_vec, n, mfft)

    # keep a safe fallback evaluated with the authoritative evaluator
    try:
        best_exact = float(evaluate_sequence(best_vec.tolist()))
    except Exception:                       # should never happen
        best_exact = float('inf')

    # ------------------------------------------------------------------ #
    # helper to generate a candidate from current (p,beta)
    def _make_powlaw(pv, betav):
        vec = (np.arange(n, dtype=np.float64) + betav) ** (-pv)
        return np.clip(vec, 0.0, 1000.0)

    # helper: random block perturbation of an existing vector
    def _perturb(vec):
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        out = vec.copy()
        factor = math.exp(rng.normal(scale=0.12))
        out[start : start + width] *= factor
        if rng.random() < 0.35:               # occasional symmetrisation
            out = 0.5 * (out + out[::-1])
        np.clip(out, 0.0, 1000.0, out=out)
        return out

    # ------------------------------------------------------------------ #
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    while time.monotonic() < deadline:

        if rng.random() < 0.7:

            p_cand = np.clip(p + rng.normal(scale=0.015), 0.30, 0.70)

            cand = _make_powlaw(p_cand, beta_cand)
