def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Construct a non‑negative step function whose AC1 upper bound is
    (much) better than the uniform construction.
    The idea is to place unit heights at positions forming an almost‑Sidon
    set: each possible sum of two indices appears at most a few times.
    With at most `c` ordered pairs per sum the bound is
        2·n·c / m²   (n = length, m = number of non‑zero entries).
    For n≈6700 a set with m≈200 and c≤3 yields a score ≈1.0,
    well below the target 1.5027.
    """
    rng = np.random.default_rng(seed)
    # length of the parent sequence (if present) otherwise a reasonable default
    try:
        n = int(len(height_sequence_1))  # type: ignore[name-defined]
    except Exception:
        n = 256

    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # fallback: uniform sequence (score = 2.0)
    best_seq = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best_seq.tolist()))  # evaluator is pre‑imported
    target_score = 1.50274349

    # we try to keep at most this many ordered pairs per convolution entry
    max_allowed = 3

    # helper: greedy construction of a set with limited sum multiplicities
    def greedy_build():
        order = rng.permutation(n)
        sums = {}          # map sum -> ordered‑pair count
        pos = []           # selected indices
        for p in order:
            # self‑pair (p,p) contributes 1 to sum 2p
            s_self = p + p
            if sums.get(s_self, 0) + 1 > max_allowed:
                continue

            good = True

                s = p + q
                if sums.get(s, 0) + 2 > max_allowed:
                    good = False
                    break
            if not good:
                continue
            # accept p
            pos.append(p)
            sums[s_self] = sums.get(s_self, 0) + 1
            for q in pos[:-1]:
                s = p + q
                sums[s] = sums.get(s, 0) + 2
        return pos, sums

    # main search loop
    while time.monotonic() < deadline:
        positions, sum_counts = greedy_build()
        m = len(positions)
        if m == 0:
            continue

        max_conv = max(sum_counts.values()) if sum_counts else 0
        # theoretical score for unit heights
        score = 2.0 * n * max_conv / (m * m)

        # keep the best we have seen
        if score < best_score:
            seq = np.zeros(n, dtype=np.float64)
            seq[positions] = 1.0
            # evaluate once with the authoritative evaluator (costly but infrequent)
            eval_score = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(eval_score) and eval_score <= best_score:
                best_score = eval_score
                best_seq = seq.copy()
                if best_score <= target_score:
                    break

        # if we haven't reached a size capable of beating the target, try again
        # (the greedy routine is fast; many random orders are tried until deadline)

    # Return the best sequence discovered (as plain Python floats)
    return [float(x) for x in best_seq]
