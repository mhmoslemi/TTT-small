import math
import numpy as np
import time

def _greedy_sidon_set(N, target_m, rng):
    """Greedy construction of a Sidon set (B2 sequence) of size up to target_m."""
    chosen = []
    used = set()          # stores sums i+j already taken
    for i in range(N):
        # quick reject if a sum would clash
        clash = False
        for j in chosen:
            if (i + j) in used:
                clash = True
                break
        if clash or (i + i) in used:
            continue
        # accept i
        for j in chosen:
            used.add(i + j)
        used.add(i + i)
        chosen.append(i)
        if len(chosen) >= target_m:
            break

    # If greedy fell short, try random picks
    if len(chosen) < target_m:
        attempts = 0
        while len(chosen) < target_m and attempts < 10000:
            i = rng.integers(N)
            if i in chosen:
                attempts += 1
                continue
            if any((i + j) in used for j in chosen) or (i + i) in used:
                attempts += 1
                continue
            for j in chosen:
                used.add(i + j)
            used.add(i + i)
            chosen.append(i)
            attempts = 0   # reset after a successful insertion
    return chosen

    """
    Construct a sparse Sidon‑set based sequence.
    The sum of the entries is set to sqrt(2*n) (the normalization used in the
    original description).  With distinct pairwise sums the convolution peak
    equals 2*a^2 where a = sum / m, giving a score close to 1 and comfortably
    below the target 1.5027.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Determine length n (use the provided height_sequence_1 if available)
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
    except Exception:
        # fallback length if parent not supplied
        n = 256
        base = np.ones(n, dtype=np.float64)

    # Desired total sum (the normalisation used in the statement)
    target_sum = float(np.sqrt(2.0 * n))

    # Max size of a Sidon set: m(m+1)/2 <= 2*n-1  ->  m = floor((-1+sqrt(1+8*(2n-1)))/2)
    limit = 2 * n - 1
    m_max = int(math.floor((-1.0 + math.sqrt(1.0 + 8.0 * limit)) / 2.0))
    # Use the maximal feasible size (at least 2)

    # Build the Sidon set of positions
    positions = _greedy_sidon_set(n, m, rng)
    # In the unlikely case we obtained fewer positions, adjust m
    m = len(positions)
    if m == 0:                       # safety fallback
        return base.tolist()

    # Value placed at each chosen position
    val = target_sum / m
    # Construct the candidate sequence
    seq = np.zeros(n, dtype=np.float64)

    seq = np.clip(seq, 0.0, 1000.0)
