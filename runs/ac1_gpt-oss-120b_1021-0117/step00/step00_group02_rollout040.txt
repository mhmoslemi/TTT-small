import numpy as np
import time

# ------------------------------------------------------------
# Helper: fast convolution (linear, not circular) and objective
# ------------------------------------------------------------
def max_convolution(a: np.ndarray) -> float:
    """Return the maximum value of the linear convolution a*a."""
    n = a.shape[0]
    # next power of two for efficient FFT
    fft_len = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # only the first 2*n-1 entries belong to the linear convolution
    return float(conv[: 2 * n - 1].max())


def objective(a: np.ndarray) -> float:
    """Exact objective used by the official evaluator."""
    s = a.sum()
    if s == 0.0:
        return float("inf")
    max_c = max_convolution(a)
    n = a.shape[0]
    return 2.0 * n * max_c / (s * s)

# Main candidate generation
# ------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Produce a non‑negative step‑function sequence that aims to minimise
    the AC1 upper bound (the official objective).  The routine first
    builds a simple exponential profile, refines it with a few
    gradient‑type updates and falls back to the best sequence seen
    before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Base length (use the supplied parent if it exists, otherwise a tidy
    # default length)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = base.size
    except Exception:
        n = 256
        base = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------
    # 1.  Initialise with a single‑parameter exponential (geometric) law.
    # ------------------------------------------------------------
    # We sweep a few eps = 1-r values and keep the best start.
    def build_exp(eps):
        r = 1.0 - eps
        # r**i may under‑flow for very small r, but eps is tiny here.
        powers = np.power(r, np.arange(n, dtype=np.float64))
        # normalisation (sum of a geometric series, safe for r≈1)
        if eps == 0.0:
            total = float(n)
        else:
            total = (1.0 - r ** n) / eps
        return powers / total

    eps_candidates = np.array([1.0 / n, 0.9 / n, 0.7 / n, 0.5 / n,
                               0.35 / n, 0.25 / n, 0.18 / n, 0.12 / n,
                               0.08 / n, 0.05 / n, 0.03 / n])
    best_vec = None
    best_score = float("inf")
    for eps in eps_candidates:
        vec = build_exp(eps)
        score = objective(vec)
        if score < best_score:
            best_score = score
            best_vec = vec

    # If something went wrong, fall back to uniform.
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = objective(best_vec)

    # ------------------------------------------------------------
    # 2.  Gradient‑type refinement (projected sub‑gradient on the simplex)
    # ------------------------------------------------------------
    a = best_vec.copy()
    best = a.copy()
    # parameters for the simple sub‑gradient loop
    step = 0.5          # initial additive step size
    no_improve = 0
    max_no_improve = 40   # after this many stalls we sprinkle random noise

    while time.monotonic() < deadline:
        # compute convolution and the index where it attains its maximum
        n = a.shape[0]
        fft_len = 1 << ((2 * n - 1).bit_length())
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k_max = int(conv.argmax())
        max_c = float(conv.max())

        # sub‑gradient of the maximal convolution term
        idx = np.arange(n, dtype=np.int64)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[partner[mask]]      # derivative of a_i*a_{k_max-i}

        # additive update then simplex projection
        cand = a - step * grad
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s > 0.0:
            cand /= s
        else:
            # all zeros – should never happen, reset to uniform
            cand = np.full_like(a, 1.0 / n)

        cand_score = 2.0 * n * max_c / (s * s)  # same as objective(cand) but cheap
        # (we already know max_c for the *old* a; for the candidate we recompute)
        cand_score = objective(cand)

        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            a = cand
            no_improve = 0
            # slowly grow the step if we are making progress
            step = min(1.0, step * 1.03)
        else:
            no_improve += 1
            # shrink step on stagnation
            step *= 0.90
            # occasional random perturbation to escape plateaus
            if no_improve >= max_no_improve:
                # perturb a small block (similar to the baseline program)
                width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
                start = int(rng.integers(0, max(1, n - width + 1)))
                stop = start + width
                perturb = np.exp(rng.normal(0.0, 0.12, size=stop - start))
                a[start:stop] *= perturb
                a = np.clip(a, 0.0, 1000.0)
                s = a.sum()
                if s > 0.0:
                    a /= s
                else:
                    a = np.full_like(a, 1.0 / n)
                no_improve = 0

        # early exit if we get under the target bound
        if best_score < 1.50274349:
            break

    # ------------------------------------------------------------
    # 3.  Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------
    return [float(x) for x in best]
