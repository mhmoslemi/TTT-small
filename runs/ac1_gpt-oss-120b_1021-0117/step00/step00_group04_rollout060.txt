import numpy as np, time, cvxpy as cp

def solve_lp_for_g(f, M):
    n = len(f)
    g = cp.Variable(n)
    # The convolution can be expressed as sum of weighted g's: use cp.convolve(f, g)
    conv_fg = cp.convolve(f, g)  # returns (2n-1)-dim expression
    constraints = [conv_fg <= M, g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, abstol=1e-7, reltol=1e-6)
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        # fallback: return None
        return None
    return np.maximum(g.value, 0.0)
```

import math

# ----------------------------------------------------------------------
# Helper: build a decreasing exponential shape (max value = 1000)
def _exp_shape(n: int, t: float = 1.255) -> np.ndarray:
    """
    Exponential decreasing sequence a_i = 1000 * exp(-t * i / n).
    Parameter t roughly minimises the AC1 bound for a pure exponential.
    """
    # use exp to avoid loss of precision for very small epsilons
    decay = math.exp(-t / n)
    return 1000.0 * decay ** np.arange(n, dtype=np.float64)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a sequence with a low AC1 upper bound.
    The routine tries many random permutations of a decreasing exponential
    sequence (which already gives a score ≈ 1.8) and keeps the best one.
    Simple hill‑climbing swaps are also performed to polish the best
    candidate.  The parent ``height_sequence_1`` is used as a safe fallback.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    limit = start + max(0.0, float(budget_s) - 2.0)   # keep a tiny margin

    # ------------------------------------------------------------------
    # Base length (use the provided parent if present)
    if 'height_sequence_1' in globals():
        parent = np.clip(
            np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0
        )
        n = parent.size
        # ensure parent is a valid fallback
        if n == 0 or not np.isfinite(parent).all():
            parent = np.ones(256, dtype=np.float64) * 1000.0 / 256
            n = parent.size
    else:
        n = 256
        parent = np.ones(n, dtype=np.float64) * 1000.0 / n

    # ------------------------------------------------------------------
    # Initialise best candidate with the parent
    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
