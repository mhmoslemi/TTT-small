def approx_ac1(seq):
    # compute conv via FFT using precomputed length
    n = len(seq)
    L = fft_len   # global computed
    # zero pad
    a = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(a * a, n=L)
    # conv length L maybe > 2n-1; we only need up to 2n-1.
    max_val = conv[:2*n-1].max()
    sum_val = seq.sum()
    if sum_val < 1e-12:
        return np.inf
    return 2.0 * n * max_val / (sum_val**2)

import time
import numpy as np

# ----------------------- helper functions ---------------------------------
def _make_exponential(n, r):
    """a_i = r**i, 0 < r < 1."""
    return np.power(r, np.arange(n, dtype=np.float64))


def _make_powerlaw(n, p):
    """a_i = (i+1)**-p, p > 0."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def _make_half_sine(n):
    """a_i = sin(π*(i+1)/(2n)).  Monotone, non‑negative."""
    return np.sin(np.pi * (np.arange(1, n + 1, dtype=np.float64) / (2.0 * n)))


def _approx_ac1(seq, fft_len, n):
    """Fast approximation of the AC1 bound using FFT."""
    a_fft = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    maxv = conv.max()
    sumv = seq.sum()
    if sumv == 0.0:
        return np.inf
    return 2.0 * n * maxv / (sumv * sumv)


def _gradient_step(seq, step):
    """One projected sub‑gradient descent step on the max‑convolution."""
    n = seq.size
    # FFT based convolution (full length)
    fft_len = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

    k = int(conv.argmax())                     # index of the current max
    start = max(0, k - (n - 1))
    end = min(n - 1, k)
    i = np.arange(start, end + 1, dtype=np.int64)
    j = k - i                                   # partner indices
    grad = np.zeros_like(seq)
    grad[i] = seq[j]                            # sub‑gradient of the max term

    # descent
    new_seq = seq - step * grad
    np.clip(new_seq, 0.0, 1000.0, out=new_seq)   # feasibility
    # avoid degenerate zero vector (evaluate_sequence would reject it)
    if new_seq.sum() < 1e-6:
        new_seq += 1e-6
    return new_seq


# ----------------------- main entry point ---------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that seeks to
    minimise the AC1 upper bound evaluated by `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------------
    # 1) obtain a fallback parent sequence (provided as `height_sequence_1`)
    # --------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    fft_len = 1 << ((2 * n - 1).bit_length())   # power‑of‑two length for FFT

    # --------------------------------------------------------------------
    # 2) generate a modest collection of parametrised seeds and keep the best
    # --------------------------------------------------------------------
    seeds = []

    # uniform
    seeds.append(np.ones(n, dtype=np.float64))

    # one exponential very close to uniform (r = exp(-1/n))
    r0 = np.exp(-1.0 / n)
    seeds.append(_make_exponential(n, r0))

    # random exponentials
    for _ in range(5):
        r = rng.uniform(0.9, 0.99999)
        seeds.append(_make_exponential(n, r))

    # power‑law candidates
    for _ in range(3):
        p = rng.uniform(0.2, 4.0)
        seeds.append(_make_powerlaw(n, p))

    # half‑sine
    seeds.append(_make_half_sine(n))

    # clip to the evaluator's admissible range
    seeds = [np.clip(s, 0.0, 1000.0) for s in seeds]

    best_seq = None
    best_score = np.inf
    for s in seeds:
        sc = float(evaluate_sequence(s.tolist()))
        if sc < best_score:
            best_score = sc
            best_seq = s.copy()

    # --------------------------------------------------------------------
    # 3) iterative optimisation – projected sub‑gradient descent + occasional
    #    random perturbations
    # --------------------------------------------------------------------
    a = best_seq.copy()
    step = max(1e-3, 0.02 * a.mean())            # initial step size

    iteration = 0
    while time.monotonic() < deadline:
        # gradient descent step
        a = _gradient_step(a, step)

        # occasionally try a random local perturbation (helps escape plateaus)
        if iteration % 30 == 0:
            width = int(rng.integers(max(1, n // 200), max(2, n // 20)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.07)))
            a[start:start + width] *= factor
            np.clip(a, 0.0, 1000.0, out=a)

        # every few iterations evaluate the exact score (cheap proxy otherwise)
        if iteration % 7 == 0:
            approx = _approx_ac1(a, fft_len, n)
            if approx < best_score * 0.99:          # promising candidate
                exact = float(evaluate_sequence(a.tolist()))
                if exact < best_score:
                    best_score = exact
                    best_seq = a.copy()
                    # if we already beat the target, stop early
                    if best_score < 1.50274349:
                        break

        # step‑size schedule
        if iteration % 40 == 0 and iteration > 0:
            step *= 0.80
            if step < 1e-5:
                step = 1e-5

        iteration += 1

    # --------------------------------------------------------------------
    # 4) return the best sequence found (fallback to parent if something went wrong)
    # --------------------------------------------------------------------
    if best_seq is None:
        best_seq = parent
    return [float(x) for x in best_seq]
