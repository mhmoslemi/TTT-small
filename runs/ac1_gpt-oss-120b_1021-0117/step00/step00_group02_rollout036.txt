def simplex_projection(v, s):
    # v: array, target sum s
    # returns projection onto { w >= 0, sum w = s }
    if s < 0:
        raise ...
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    # find rho = max i s.t. u_i - cssv_i / i > 0
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```
