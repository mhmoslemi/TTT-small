Implementation details for LFSR:

def lfsr_seq(length, seed=0xACE, taps=(12,11,10,5)):
    # 12-bit LFSR; taps as positions (1-indexed from LSB)
    reg = seed & ((1 << 12) - 1)
    seq = np.empty(length, dtype=np.float64)
    for i in range(length):
        bit = reg & 1
        seq[i] = float(bit)
        # compute feedback bit = XOR of tapped bits
        feedback = 0
        for t in taps:
            feedback ^= (reg >> (t-1)) & 1
        reg = (reg >> 1) | (feedback << (12 - 1))
    return seq
```

import numpy as np
import time

def _random_binary(rng, n, ones):
    """Return a binary array of length n with exactly `ones` entries equal to 1."""
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=ones, replace=False)
    a[idx] = 1.0
    return a

def _score(seq):
    """Call the authoritative evaluator and return a float."""
    return float(evaluate_sequence(seq.tolist()))

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a low‑AC1 bound sequence by swapping bits to flatten the convolution."""
    rng = np.random.default_rng(seed)

    # length of the problem
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 256
    # we keep roughly half the entries equal to 1 (any density >0 works, 0.5 is a good trade‑off)
    target_ones = n // 2

    # start from a random binary vector
    cur = _random_binary(rng, n, target_ones)
    cur_score = _score(cur)

    # keep the best found so far (fallback to parent if anything goes wrong)
    best = cur.copy()
    best_score = cur_score

    # time budget – keep a few seconds margin for returning the result
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # main improvement loop
    while time.monotonic() < deadline:
        # full convolution and the shift with the largest value
        conv = np.convolve(cur, cur)
        k_max = int(np.argmax(conv))

        # collect indices that take part in the maximal shift
        ones_candidates = []
        zeros_candidates = []
        # iterate over possible i where partner=j=k_max-i stays inside the array
        for i in range(n):
            j = k_max - i
            if j < 0 or j >= n:
                continue
            if cur[i] == 1.0 and cur[j] == 1.0:
                ones_candidates.append(i)
            elif cur[i] == 0.0 and cur[j] == 0.0:
                zeros_candidates.append(i)

        # if we found a pair to break, try to swap a “bad” 1 with a “good” 0
        made_swap = False
        if ones_candidates and zeros_candidates:
            i_swap = int(rng.choice(ones_candidates))
            j_swap = int(rng.choice(zeros_candidates))
            cand = cur.copy()
            cand[i_swap] = 0.0
            cand[j_swap] = 1.0
            cand_score = _score(cand)
            made_swap = True
        else:
            # fallback: random exchange of a 1 and a 0 (helps escape dead‑ends)
            i_one = int(rng.choice(np.flatnonzero(cur == 1.0)))
            i_zero = int(rng.choice(np.flatnonzero(cur == 0.0)))
            cand = cur.copy()
            cand[i_one] = 0.0
            cand[i_zero] = 1.0
            cand_score = _score(cand)

        # accept only improving moves (strict descent – fast and safe)
        if np.isfinite(cand_score) and cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
            # early stop when we are clearly under the target bound
            if best_score < 1.50274349:
                break
        elif not made_swap:
            # if even the random move was not improving, try a small perturbation
            # (multiply a random block by a factor close to 1)
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            factor = float(np.exp(rng.normal(0.0, 0.06)))
            cand = cur.copy()
            cand[start:stop] *= factor
            cand = np.clip(cand, 0.0, 1000.0)
            cand_score = _score(cand)

                cur_score = cand_score
                if cand_score < best_score:
                    best = cand.copy()
                    best_score = cand_score

    # final safety: guarantee a non‑empty valid sequence
    if best_score == float('inf') or not np.isfinite(best_score):
        # fall back to a simple uniform vector (score = 2) – never happens in practice
        best = np.full(n, 1.0, dtype=np.float64)
    return [float(x) for x in best]
