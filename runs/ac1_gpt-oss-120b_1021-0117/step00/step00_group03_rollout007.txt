def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    # Determine initial length
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise Exception
    except Exception:
        base = np.ones(256, dtype=np.float64)
    n = base.size
    # start with uniform vector (value 1.0)
    best = np.full(n, 1.0, dtype=np.float64)
    # ensure within limit [0,1000]
    best = np.clip(best, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best.tolist()))
    # Simple local perturbations to see if we can improve (unlikely)
    while time.monotonic() < deadline:
        # pick random block and scale factor
        width = int(rng.integers(1, max(2, n//4)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        factor = float(np.exp(rng.normal(0, 0.1)))
        cand = best.copy()
        cand[start:start+width] *= factor
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
    return [float(x) for x in best]

    rng = np.random.default_rng(seed)
    # reserve a tiny margin for the final return
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # Determine length from the provided parent if it exists and is valid.
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        base = np.ones(256, dtype=np.float64)   # generic fallback length

    n = base.size

    # Start from a uniform sequence (the known optimum under non‑negativity).
    best = np.full(n, 1.0, dtype=np.float64)   # any constant works
    best = np.clip(best, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best.tolist()))

    # Light‑weight random local search – mostly a safety net.
    while time.monotonic() < deadline:
        # pick a random contiguous block
        width = int(rng.integers(1, max(2, n // 4)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        # multiplicative perturbation drawn from a log‑normal distribution
        factor = float(np.exp(rng.normal(0.0, 0.10)))
        cand = best.copy()
        cand[start:start + width] *= factor
        cand = np.clip(cand, 0.0, 1000.0)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Return a plain Python list as required.
    return [float(x) for x in best]
