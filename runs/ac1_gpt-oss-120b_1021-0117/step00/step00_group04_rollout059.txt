def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    # Determine n
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.shape[0]
    else:
        n = 256
    # start with uniform
    best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    best_score = float(evaluate_sequence(best_seq.tolist()))
    # Search exponential parameters
    for sigma in np.linspace(1e-5, 0.001, 30):  # maybe up to
        # generate exponential decreasing
        seq = np.exp(-sigma * np.arange(n, dtype=np.float64))
        seq_sum = seq.sum()
        if seq_sum == 0:
            continue
        seq /= seq_sum
        # evaluate
        score = float(evaluate_sequence(seq.tolist()))
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    # Search power law exponents
    for p in np.linspace(0.1, 0.9, 81):
        idx = np.arange(1, n+1, dtype=np.float64)
        seq = idx ** (-p)
        seq_sum = seq.sum()
        seq = seq / seq_sum
        score = float(evaluate_sequence(seq.tolist()))
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    # optional random refinement
    while time.monotonic() < deadline:
        # perturb best
        candidate = best_seq.copy()
        # apply small random multiplicative noise to a random block
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 100))
        end = min(n, start + width)
        factor = np.exp(rng.normal(0.0, 0.05))
        candidate[start:end] *= factor
        # renormalize
        sumc = candidate.sum()
        if sumc > 0:
            candidate /= sumc
        # evaluate
        score = float(evaluate_sequence(candidate.tolist()))
        if score < best_score:
            best_score = score
            best_seq = candidate.copy()
    # Return list of floats (ensuring within bounds)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # Ensure sum >=0.01 (already)
    return [float(x) for x in best_seq]

import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a non‑negative step function that lowers the AC1 bound.

    The algorithm tests simple polynomially decreasing shapes
    a_i ∝ (n‑i)^p (p=1…10) which analytically give very low
    max‑convolution values, and then refines the best found by cheap
    random block perturbations.  All work is O(n) memory and runs well
    within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Length of the problem instance (use the provided parent if present)
    n = int(np.asarray(globals().get("height_sequence_1", np.ones(256)),
                       dtype=np.float64).size)

    # Uniform sequence as a safe fallback (score = 2.0)
    best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------
    # 1.  Try decreasing polynomial shapes a_i ∝ (n-i)^p  (p = 1 … 10)
    # ------------------------------------------------------------------
    base_idx = np.arange(n, 0, -1, dtype=np.float64)   # n,…,1
    for p in range(1, 11):
        seq = base_idx ** p
        total = seq.sum()
        if total == 0.0:
            continue
        seq /= total                     # normalise to sum‑1

        # evaluate the shape and its reverse (mirror image)
        for cand in (seq, seq[::-1]):
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = cand.copy()

    # ------------------------------------------------------------------
    # 2.  Light stochastic refinement of the current best candidate
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        cand = best_seq.copy()
        # Random contiguous block scaling
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 20))
        end = min(n, start + width)
        factor = np.exp(rng.normal(0.0, 0.1))
        cand[start:end] *= factor

        # Re‑normalise (scale does not affect the objective)
        s = cand.sum()
        if s > 0.0:
            cand /= s

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = cand.copy()

    # Final safety: clip to the allowed range and return as a list
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
