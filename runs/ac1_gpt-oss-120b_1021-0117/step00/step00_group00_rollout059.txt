import time
import numpy as np

def _build_sparse_set(n, max_pair, rng):
    """Greedy construction of a set S⊂{0,…,n-1}
    such that each unordered sum i+j occurs at most max_pair times.
    Returns the list of selected indices."""
    sum_counts = np.zeros(2 * n - 1, dtype=np.int16)
    chosen = []
    order = rng.permutation(n)
    for idx in order:
        # quick abort if we already have a large set
        # (the max achievable size for given max_pair is roughly sqrt(4*max_pair*n))
        if len(chosen) * len(chosen) > 4 * max_pair * n * 1.2:
            # already close to theoretical limit, stop early
            break
        conflict = False
        # check sums with already chosen indices
        for j in chosen:
            if sum_counts[idx + j] >= max_pair:
                conflict = True
                break
        if conflict:
            continue
        # check the diagonal sum (i+i)
        if sum_counts[idx + idx] >= max_pair:
            continue
        # add the new index
        for j in chosen:
            sum_counts[idx + j] += 1
        sum_counts[idx + idx] += 1
        chosen.append(int(idx))
    return chosen

    best_score = float(evaluate_sequence(best_seq.tolist()))

    # main stochastic search
    while time.monotonic() < deadline:
        # randomly vary the allowed multiplicity of unordered sums
        max_pair = int(rng.integers(2, 5))  # try 2,3 or 4
        selected = _build_sparse_set(n, max_pair, rng)
        m = len(selected)
        if m < 2:
            continue

        # construct candidate: unit weight on the selected positions
        cand = np.zeros(n, dtype=np.float64)
        cand[selected] = 1.0   # any positive constant works (scale‑invariant)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = cand
            # stop early if we already beat the target comfortably
            if best_score < 1.51:
                break

    # safety net: ensure we return a valid non‑empty list
    if best_seq.size == 0:
        best_seq = np.ones(n, dtype=np.float64)

    return [float(x) for x in best_seq]
