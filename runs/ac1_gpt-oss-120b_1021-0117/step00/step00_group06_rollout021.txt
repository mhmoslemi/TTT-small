import numpy as np
import time

def _make_decreasing_seq(length: int) -> np.ndarray:
    """Return a non‑negative linearly decreasing sequence of given length.
    The maximum value is capped at 1000, preserving the optimal ratio."""
    # Original values would be length, length‑1, … , 1.
    max_val = float(length)
    scale = 1.0
    if max_val > 1000.0:
        scale = 1000.0 / max_val
    # descending arithmetic progression
    seq = (max_val - np.arange(length, dtype=np.float64)) * scale
    # safety clipping (should already be within bounds)
    return np.clip(seq, 0.0, 1000.0)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Produce a step‑function (non‑negative) sequence that attains a low
    AC1 upper bound.  The construction is a linearly decreasing sequence,
    scaled so the largest entry does not exceed 1000.
    """
    # Determine the target length: use the provided parent if sensible,
    # otherwise fall back to a modest default.
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim == 1 and parent_arr.size > 0 and np.isfinite(parent_arr).all():
            n = int(parent_arr.size)
        else:
            n = 256
    except NameError:
        n = 256

    # Build the decreasing sequence once – O(n) memory and time.
    seq = _make_decreasing_seq(n)

    # Return a plain Python list of floats, as required by the evaluator.
    return [float(v) for v in seq]
