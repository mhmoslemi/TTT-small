def gradient_step(a, eta):
    b = np.convolve(a, a)
    k = int(np.argmax(b))
    # compute reversed slice
    grad = np.zeros_like(a)
    lo = max(0, k - (len(a)-1))
    hi = min(k, len(a)-1)
    # for i from lo to hi:
    # j = k - i
    # grad[i] = 2*a[j]
    idx_i = np.arange(lo, hi+1)
    idx_j = k - idx_i
    grad[idx_i] = 2 * a[idx_j]
    new = a - eta * grad
    new = np.clip(new, 0.0, None)
    if new.sum() > 0:
        new /= new.sum()

def _catalan_weights(n: int) -> np.ndarray:
    """Return the first n coefficients w_i = C(2i,i)/4**i."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for i in range(1, n):
        # recurrence: w[i] = w[i-1] * (2i)*(2i-1) / (i*i*4)
        w[i] = w[i - 1] * ((2 * i) * (2 * i - 1)) / (i * i * 4.0)
    return w

    a = w / w.sum()                     # sum(a) = 1   ⇒  sum is normalised
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # also try the provided parent (normalised) as a possible better start
    parent_norm = parent / parent.sum()
    parent_score = float(evaluate_sequence(parent_norm.tolist()))
    if parent_score < best_score:
        best = parent_norm.copy()
        best_score = parent_score

    # ------------------------------------------------------------------
    # 3.  Simple stochastic optimisation
    # ------------------------------------------------------------------
    # step for the gradient‑style move (fraction of the maximal gradient entry)
    grad_frac = 0.05
    # minimal allowed fraction – if the algorithm stalls we reset it
    min_grad_frac = 1e-4

    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # a)  Gradient‑inspired move (push mass towards the right)
        # --------------------------------------------------------------
        if rng.random() < 0.55:
            # g_i = 2 * sum_{j=0}^{n-1-i} a_j  = 2 * cumulative sum reversed
            cum = np.cumsum(best)                      # forward cumulative sum
            grad = 2.0 * cum[::-1]                     # reversed

            max_g = grad.max()
            if max_g > 0.0:
                # move a fraction of the maximal gradient component
                step = grad_frac * (best.max() / max_g)
                cand = best - step * grad
                np.clip(cand, 0.0, None, out=cand)
                s = cand.sum()
                if s > 0.0:
                    cand /= s
                    score = float(evaluate_sequence(cand.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best = cand
                        best_score = score
                        grad_frac = min(grad_frac * 1.2, 1.0)   # accelerate if we improve
                        if best_score <= 1.50274349:          # early exit on target
                            break
                    else:
                        grad_frac = max(grad_frac * 0.5, min_grad_frac)

        # --------------------------------------------------------------
        # b)  Random pairwise transfer (mass‑conserving)
        # --------------------------------------------------------------
        else:
            i, j = rng.choice(n, size=2, replace=False)
            if best[i] == 0.0:
                continue
            # transfer a random fraction of a[i] to a[j]
            eps = rng.random() * best[i] * 0.5
            cand = best.copy()
            cand[i] -= eps
            cand[j] += eps
            # normalise (sum stays 1 but numerical noise may appear)
            s = cand.sum()
            if s > 0.0:
                cand /= s
                score = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(score) and score < best_score:
                    best = cand
                    best_score = score
                    if best_score <= 1.50274349:
                        break

        # --------------------------------------------------------------
        # c)  Occasional global jitter – helps escape flat plateaux
        # --------------------------------------------------------------
        if rng.random() < 0.01:
            jitter = np.exp(rng.normal(0.0, 0.018, size=n))
            cand = best * jitter
            cand_sum = cand.sum()
            if cand_sum > 0.0:
                cand /= cand_sum
                score = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(score) and score < best_score:
                    best = cand
                    best_score = score
                    if best_score <= 1.50274349:
                        break

    # ------------------------------------------------------------------
    # 4.  Return the best found sequence as a plain Python list
    # ------------------------------------------------------------------
    return [float(v) for v in best]
