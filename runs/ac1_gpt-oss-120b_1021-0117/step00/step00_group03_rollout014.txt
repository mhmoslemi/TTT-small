import time
import numpy as np

def _make_seq(length: int, power: int) -> np.ndarray:
    """Return a monotone sequence a_i = (i/length)^power scaled to max 1000."""
    # indices 1..length, scaled to [0,1]
    idx = (np.arange(1, length + 1, dtype=np.float64) / length)
    seq = (idx ** power) * 1000.0
    return np.clip(seq, 0.0, 1000.0)

    The shape a_i = ((i+1)/n)^p (scaled to 1000) yields the score
    2 · (p+1)²·(p!²)/(2p+1)!, which for p≥3 is well below the target 1.5027.
    """
