import numpy as np
import time
import math

# fast convolution (full linear) using FFT, returns the maximum coefficient
def _max_convolution(seq: np.ndarray) -> float:
    n = seq.size
    # next power of two for efficient FFT
    size = 1 << ((2 * n - 1).bit_length())
    # real FFT because the sequence is real
    freq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(freq * freq, n=size)[: 2 * n - 1]
    return float(conv.max())


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a non‑negative step‑function (list of floats) with a low
    AC1 upper‑bound.  The method builds an exponential‑decay profile,
    refines the decay rate by a cheap grid search and then performs a
    time‑bounded stochastic local search.
    """
    rng = np.random.default_rng(seed)
    # length of the problem (use the provided parent if present)
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        # fallback: a modest size if the parent is unavailable
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # clamp the fallback to legal range – it will only be used if we
    # never find a better candidate.
    parent = np.clip(parent, 0.0, 1000.0)

    # the evaluator normalises by sqrt(2*n); using the same total makes
    # the final score equal to the maximum convolution value.
    target_sum = math.sqrt(2.0 * n)

    # -----------------------------------------------------------------
    # 1) construct the basic exponential family   a_i = exp(-eps*i)
    # -----------------------------------------------------------------
    i_vec = np.arange(n, dtype=np.float64)

    def _make_exp(eps: float) -> np.ndarray:
        raw = np.exp(-eps * i_vec)          # positive, decreasing
        scale = target_sum / raw.sum()
        return raw * scale

    # initial guess: eps = 1/n (theoretical optimum for a pure exponential)
    eps_best = 1.0 / n
    best_seq = _make_exp(eps_best)
    best_score = _max_convolution(best_seq)

    # -----------------------------------------------------------------
    # 2) quick grid search on epsilon (linear sweep, cheap)
    # -----------------------------------------------------------------
    # search a reasonable interval around the theoretical point
    eps_grid = np.linspace(0.2 / n, 5.0 / n, 40)   # 40 candidates
    for eps in eps_grid:
        if time.monotonic() > time.monotonic() + (budget_s - 3):
            break
        seq = _make_exp(eps)
        score = _max_convolution(seq)
        if score < best_score:
            best_score = score
            best_seq = seq
            eps_best = eps

    # -----------------------------------------------------------------
    # 3) stochastic local optimisation (block perturbations)
    # -----------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 4.0)

    while time.monotonic() < deadline:
        # pick a random contiguous block and perturb it multiplicatively
        width = int(rng.integers(1, max(2, n // 40)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width

        cand = best_seq.copy()
        factor = float(np.exp(rng.normal(0.0, 0.07)))   # modest log‑normal tweak
        cand[start:stop] *= factor

        # re‑normalise to keep the prescribed sum (scale invariance)
        cand *= (target_sum / cand.sum())
        np.clip(cand, 0.0, 1000.0, out=cand)

        score = _max_convolution(cand)
        if score < best_score:
            best_score = score
            best_seq = cand

    # -----------------------------------------------------------------
    # 4) final safety fallback (original parent) – never worse than 2
    # -----------------------------------------------------------------
    # evaluate the fallback just in case all generated sequences become
    # invalid (should not happen).
    try:
        parent_score = _max_convolution(parent)
        if parent_score < best_score:
            best_seq = parent
    except Exception:
        pass

    # Return as a plain Python list of floats (the evaluator expects this)
    return [float(v) for v in best_seq]
