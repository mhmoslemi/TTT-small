# Helper: a cheap block perturbation (kept from the reference solution)
# ----------------------------------------------------------------------
def _make_block_perturbation(base, rng):
    """Return a non‑negative O(n) perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    # block width between n/200 and n/12
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # optional mirroring
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Linear‑programming sub‑routine (maximise total mass under a fixed AC1 bound)
# ----------------------------------------------------------------------
def _solve_lp_max_sum(f_vec, M):
    """
    Solve   maximise sum(g)   subject to   f_vec * g  <=  M  (full convolution)
    and g >= 0.  Returns the optimiser g as a NumPy array, or None on failure.
    """
    try:
        import cvxpy as cp
    except Exception:          # cvxpy not available – give up on the LP
        return None

    n = int(f_vec.shape[0])
    g = cp.Variable(n, nonneg=True)
    # convolution of constant f_vec with variable g
    conv = cp.conv(cp.Constant(f_vec), g)   # length 2n‑1
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [conv <= float(M)])
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                   abstol=1e-7, reltol=1e-6, feastol=1e-7, verbose=False)
    except Exception:
        return None
    if g.value is None:
        return None
    return np.array(g.value, dtype=np.float64)
