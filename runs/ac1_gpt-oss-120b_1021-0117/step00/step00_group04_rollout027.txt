import time
import numpy as np

# Helper: construct a shifted geometric (exponential decay) sequence.
def shifted_geometric_sequence(length, block_len, decay):
    """
    length    – total length of the sequence (n)
    block_len – number of non‑zero entries placed at the right end
    decay     – positive float t; the values are exp(-t * k) for k = 0..block_len‑1
    """
    seq = np.zeros(length, dtype=np.float64)
    start = length - block_len
    # a_i = exp(-decay * (i-start)) for i >= start
    seq[start:] = np.exp(-decay * np.arange(block_len, dtype=np.float64))
    return seq
