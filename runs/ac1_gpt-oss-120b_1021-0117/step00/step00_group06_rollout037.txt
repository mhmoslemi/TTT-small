def generate_sidon_set(N, target_m, rng):
    sums_set = set()
    positions = []
    attempts = 0
    while len(positions) < target_m and attempts < max_attempts:
        x = rng.integers(0, N)
        if x in positions:
            attempts += 1
            continue
        # compute new sums: x+x, x+y for y in positions
        new_sums = {x+x}
        for y in positions:
            new_sums.add(x+y)
        # If any of new_sums intersect sums_set, reject
        if new_sums.isdisjoint(sums_set):
            # Accept
            positions.append(x)
            sums_set.update(new_sums)
            attempts = 0
        else:
            attempts += 1
    return positions

import time
import math
import numpy as np

def _generate_sidon_set(n, rng, max_size=None):
    """
    Greedy construction of a Sidon set (all unordered sums i+j are distinct)
    using a random permutation.  Returns a list of indices.
    """
    if max_size is None:
        max_size = int(math.sqrt(n)) * 2  # generous upper bound
    perm = rng.permutation(n)
    sums = set()
    pos = []
    for x in perm:
        # quick abort if we already have enough elements
        if len(pos) >= max_size:
            break
        # test collisions: self‑sum and sum with existing elements
        s_self = x + x
        if s_self in sums:
            continue
        collide = False
        for y in pos:
            if x + y in sums:
                collide = True
                break
        if collide:
            continue
        # accept x
        pos.append(x)
        sums.add(s_self)
        for y in pos[:-1]:
            sums.add(x + y)
    return pos

def _build_sequence(n, sidon_pos, w=1.0):
    """
    Constructs the candidate sequence.
    All entries start with epsilon = w / m (where m = |sidon_pos|),
    then the Sidon positions are set to w.
    """
    m = max(1, len(sidon_pos))
    eps = w / m
    seq = np.full(n, eps, dtype=np.float64)
    if sidon_pos:
        seq[sidon_pos] = w
    return seq
