import time
import math
import numpy as np


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that tries to minimise
    the AC1 upper bound evaluated by ``evaluate_sequence``.
    The search is deliberately lightweight: a constant fallback (score = 2)
    is kept as a safe best candidate, while a few cheap random block
    perturbations are tried until the time budget expires.
    """
    rng = np.random.default_rng(seed)
    # reserve a small safety margin for the return path
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ---------- safe constant fallback (always yields score = 2) ----------
    const_len = 256                     # modest size, constant value any >0
    const_val = 1.0
    best = np.full(const_len, const_val, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))

    # ---------- try the provided parent (if any) ----------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            parent = np.clip(parent, 0.0, 1000.0)
            parent_score = float(evaluate_sequence(parent.tolist()))
            if parent_score < best_score:
                best = parent.copy()
                best_score = parent_score
    except NameError:          # height_sequence_1 not supplied
        pass

    # ---------- cheap block‑wise perturbation (matrix‑free, O(n) memory) ----------
    def perturb(arr):
        """Return a copy of ``arr`` with a random block scaled."""
        n = arr.size
        # choose a block width between n/200 and n/12 (at least 1)
        low = max(1, n // 200)
        high = max(2, n // 12)
        width = int(rng.integers(low, high + 1))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        out = arr.copy()
        out[start:stop] *= float(math.exp(rng.normal(0.0, 0.12)))
        # occasional symmetrisation (helps avoid pathological spikes)
        if rng.random() < 0.35:
            out = 0.5 * (out + out[::-1])
        return np.clip(out, 0.0, 1000.0)

    # Main loop – keep trying until the deadline.
    while time.monotonic() < deadline:
        cand = perturb(best)
        cand_score = float(evaluate_sequence(cand.tolist()))
        if math.isfinite(cand_score) and cand_score < best_score:
            best = cand
            best_score = cand_score

    # Return a plain Python list of floats as required.
    return [float(v) for v in best]
