def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    total_time = max(0.0, float(budget_s) - 1.0)  # keep safety margin
    deadline = start_time + total_time

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
    except Exception:
        parent = np.ones(256, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.sum() < 0.01:
        parent = np.full_like(parent, 0.01/parent.size)

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    cur = best.copy()
    cur_score = best_score

    n = len(cur)
    # Precompute original sum for normalizing
    target_sum = cur.sum()

    # Temperature schedule: start with some factor relative to best_score
    start_temp = max(best_score - 1.0, 0.5)  # ensure >0
    end_temp = 1e-6

    # iteration counter
    it = 0

    # probabilities
    probs = [0.2, 0.2, 0.5, 0.1]  # swap, transfer, scale_block, symmetrize

    while time.monotonic() < deadline:
        # compute temperature
        elapsed = time.monotonic() - start_time
        fraction = elapsed / total_time
        temperature = start_temp * (1 - fraction) + end_temp
        # pick move
        r = rng.random()
        if r < probs[0]:
            # swap
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            cand = cur.copy()
            cand[i], cand[j] = cand[j], cand[i]
        elif r < probs[0] + probs[1]:
            # transfer
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            amount = cur[i] * rng.random()
            cand = cur.copy()
            cand[i] -= amount
            cand[j] += amount
        elif r < probs[0] + probs[1] + probs[2]:
            # scale block
            width_min = max(1, n // 200)
            width_max = max(2, n // 12)
            width = int(rng.integers(width_min, width_max + 1))
            start_idx = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            cand = cur.copy()
            cand[start_idx:start_idx+width] *= factor
        else:
            # symmetrize
            cand = 0.5 * (cur + cur[::-1])

        # clip and renorm to target sum
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s > 0:
            cand = cand * (target_sum / s)
        else:
            # avoid division by zero, skip this candidate
            continue

        # evaluate
        cand_score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(cand_score):
            # invalid candidate, discard
            continue

        # acceptance
        if cand_score < cur_score or (temperature > 0 and rng.random() < math.exp(-(cand_score - cur_score) / temperature)):
            cur = cand
            cur_score = cand_score

            if cand_score < best_score:
                best = cand
                best_score = cand_score

        # occasionally re-normalize cur to target sum after many steps?
        # Not needed.

        it += 1

        # Optional early break if best_score already below target threshold (maybe ~1.5)
        if best_score < 1.49:
            break

    # Return best candidate as list of floats
    return [float(x) for x in best]

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper functions – all are top‑level, no closures or lambdas.
# ----------------------------------------------------------------------


def _clip_and_renorm(arr: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1000] and rescale to the given total sum."""
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if s > 0.0:
        arr = arr * (target_sum / s)
    arr = np.clip(arr, 0.0, 1000.0)  # safety after renorm
    return arr


def _score_from_conv(conv: np.ndarray, total: float, n: int) -> float:
    """Exact objective value from a convolution array."""
    if total < 0.01:
        return float('inf')
    return 2.0 * n * conv.max() / (total * total)


def _make_swap(x: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = x.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    y = x.copy()
    y[i], y[j] = y[j], y[i]
    return y


def _make_transfer(x: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = x.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    amount = x[i] * rng.random() * 0.2  # up to 20 % of source
    y = x.copy()
    y[i] -= amount
    y[j] += amount
    return y


def _make_scale_block(x: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = x.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.5)))  # broader log‑normal steps
    y = x.copy()
    y[start:start + width] *= factor
    return y


def _make_symmetrize(x: np.ndarray) -> np.ndarray:
    return 0.5 * (x + x[::-1])


def _make_reduce_pair(x: np.ndarray,
                      cur_conv: np.ndarray,
                      rng: np.random.Generator) -> np.ndarray:
    """Move a small amount from the index that contributes most
    to the current maximal autocorrelation to the least‑contributing index."""
    n = x.size
    kmax = int(cur_conv.argmax())
    idx = np.arange(n)
    valid = (kmax - idx >= 0) & (kmax - idx < n)
    partner = np.zeros_like(x)
    partner[valid] = x[kmax - idx[valid]]
    contrib = x * partner
    i_max = int(contrib.argmax())
    i_min = int(contrib.argmin())
    if i_max == i_min:
        return x.copy()
    delta = x[i_max] * rng.random() * 0.1   # up to 10 % shift
    y = x.copy()
    y[i_max] -= delta
    y[i_min] += delta
    return y
