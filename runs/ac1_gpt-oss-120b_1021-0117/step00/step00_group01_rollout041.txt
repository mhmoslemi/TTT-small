def max_convolution(seq):
    n = len(seq)
    L = 2*n-1
    size = 1<< (L-1).bit_length()
    fa = np.fft.rfft(seq, size)
    fb = fa
    conv = np.fft.irfft(fa*fb, size)[:L]
    return conv.max()

import numpy as np
import time

def _build_symmetric_exponential(n: int) -> np.ndarray:
    """
    Construct a non‑negative sequence a[0..n‑1] with a[0]=1/1000,
    a[-1]=1000 and a[i]*a[n‑1‑i]=1 for all i.  This yields a very
    small AC1 upper bound.
    """
    if n <= 1:
        return np.ones(n, dtype=np.float64)

    # log(1000) appears because we want a[0]=1/1000 and a[-1]=1000.
    log1000 = np.log(1000.0)
    # exponent step so that a[-1] = exp(kappa*(n-1)/2) = 1000
    kappa = 2.0 * log1000 / (n - 1)
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(kappa * (i - mid))
    # Guard against tiny numerical overshoot
    np.clip(a, 0.0, 1000.0, out=a)
    return a
