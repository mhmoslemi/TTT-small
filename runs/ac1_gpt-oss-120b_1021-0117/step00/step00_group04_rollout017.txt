def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a uniform non‑negative step function of the appropriate length.
    For non‑negative sequences the AC1 upper bound 2·n·max(b)/(∑a)²
    is minimized (value = 2) exactly by a uniform distribution.
    """
    # Determine the length from the provided parent if possible.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = parent.size
    except Exception:
        # Fallback length if the parent is missing or malformed.
        n = 256

    # Construct a constant sequence (any constant works; we use 1.0).
    seq = np.full(n, 1.0, dtype=np.float64)

    # Clip to the allowed range – redundant for 1.0 but kept for safety.
    seq = np.clip(seq, 0.0, 1000.0)

    # Quick sanity check with the authoritative evaluator.
    # If something went wrong we fall back to a plain ones‑vector.
    try:
        _ = float(evaluate_sequence(seq.tolist()))
    except Exception:
        seq = np.ones(n, dtype=np.float64)

    return [float(x) for x in seq]
