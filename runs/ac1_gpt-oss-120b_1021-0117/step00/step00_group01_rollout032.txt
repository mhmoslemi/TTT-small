def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Generate a step‑function that minimizes the AC1 upper bound.

    The strategy evaluates a few simple parametric families (polynomial
    growth, decreasing variants, etc.) that are known to give very low
    max‑convolution values, then performs a light local search.
    """
    rng = np.random.default_rng(seed)
    # Reserve a few seconds for clean‑up before the sandbox timeout.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Length of the problem instance.
    if "height_sequence_1" in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = parent.size
    else:

    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------
    # 1.  Test simple polynomial shapes a_i = (i+1)^p  (and the decreasing version).
    # ------------------------------------------------------------------
    exponents = (0.5, 1.0, 1.5, 2.0, 3.0, 4.0)   # a few representative powers
    idx = np.arange(1, n + 1, dtype=np.float64)  # 1‑based indices

    for p in exponents:
        if time.monotonic() >= deadline:
            break
        # increasing power law
        cand = idx ** p
        cand *= 1000.0 / cand.max()            # scale to respect the 0‑1000 bound
        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score:
            best_score, best_seq = score, cand.copy()

        # decreasing power law (mirror of the above)

        cand *= 1000.0 / cand.max()
        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
