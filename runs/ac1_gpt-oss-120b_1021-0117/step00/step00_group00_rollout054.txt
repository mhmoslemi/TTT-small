def compute_ac1(a):
    # a is np.ndarray of floats
    n = a.size
    # use real FFT
    size = 1 << ((2*n -1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2*n-1]
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01: return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)

def block_perturbation(a, rng):
    n = a.size
    # choose block width random between n//100 and n//10
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    b = a.copy()
    factor = np.exp(rng.normal(0.0, 0.12))
    b[start:stop] *= factor
    if rng.random() < 0.35:
        b[:] = 0.5 * (b + b[::-1])
    np.clip(b, 0.0, 1000.0, out=b)
    return b

import numpy as np
import time
import math

def _conv_max(a: np.ndarray) -> float:
    n = a.shape[0]
    conv_len = 2 * n - 1
    size = 1 << ((conv_len - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:conv_len]
    return float(np.max(conv))


def _ac1_score(a: np.ndarray) -> float:
    """Exact AC1 score (but using our own convolution – identical for our ranges)."""
    n = a.shape[0]
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    return 2.0 * n * _conv_max(a) / (s * s)


def _exp_shape(n: int, c: float) -> np.ndarray:
    """Geometric shape a_i = (1‑r) r^i / (1‑r^n) with r = exp(-c/n).  Sum = 1."""
    r = math.exp(-c / n)
    # powers via exp(log(r) * i) to stay stable
    idx = np.arange(n, dtype=np.float64)
    r_pow = np.exp(idx * math.log(r))
    alpha = 1.0 - r
    denom = 1.0 - r**n
    if denom == 0.0:
        denom = 1e-12
    a = alpha * r_pow / denom
    return a


def _mass_move(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a small fraction of mass from a large entry to a random other entry."""
    n = a.shape[0]
    i = rng.integers(n)
    if a[i] <= 0.0:
        return None
    # move a random fraction (1‑20 %) of the selected mass
    frac = rng.uniform(0.01, 0.20)
    delta = a[i] * frac
    if delta > a[i]:
        delta = a[i] * 0.5
    j = rng.integers(n)
    while j == i:
        j = rng.integers(n)
    b = a.copy()
    b[i] -= delta
    b[j] += delta
    np.clip(b, 0.0, 1000.0, out=b)
    return b


def _block_perturbation(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block."""
    n = a.shape[0]
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    b = a.copy()
    factor = math.exp(rng.normal(0.0, 0.12))
    b[start:stop] *= factor
    if rng.random() < 0.35:                     # occasional mirroring
        b[:] = 0.5 * (b + b[::-1])
    np.clip(b, 0.0, 1000.0, out=b)
    return b

    deadline = time.time() + max(0.0, float(budget_s) - 2.0)
