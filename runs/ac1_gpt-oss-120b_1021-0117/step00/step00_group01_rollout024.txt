import time

def _sidon_set(n: int, target: int, rng: np.random.Generator):
    """
    Greedy construction of a Sidon set (all unordered sums are distinct)
    of size up to ``target`` inside {0,…,n-1}.
    """
    chosen = []
    sums = set()
    indices = np.arange(n, dtype=np.int64)
    rng.shuffle(indices)
    for i in indices:
        if len(chosen) >= target:
            break
        i2 = int(i)
        # self‑sum must be unused
        if i2 + i2 in sums:
            continue
        # check sums with already chosen elements
        conflict = False
        for j in chosen:
            if i2 + j in sums:
                conflict = True
                break
        if conflict:
            continue
        # no conflict → add the element
        for j in chosen:
            sums.add(i2 + j)          # unordered sum, same as j+i2
        sums.add(i2 + i2)
        chosen.append(i2)
    return chosen

    sel = _sidon_set(n, target_L, rng)
    L = len(sel)
    if L == 0:                                       # degenerate safety net
        candidate = np.full(n, 1.0, dtype=np.float64)
        return [float(v) for v in candidate]

    # ------------------------------------------------------------------ #
    # 3. Compute the optimal uniform “background” value m.
    #    It solves  n·m² = 2·H² + 2·L·H·m    (balance large‑large and background)
    # ------------------------------------------------------------------ #
    H = max_large
    # formula: m = (L·H + sqrt((L·H)² + 2·n·H²)) / n
    m = (L * H + math.sqrt((L * H) ** 2 + 2.0 * n * H * H)) / n
    m = max(0.0, min(1000.0, m))

    # ------------------------------------------------------------------ #
    # 4. Assemble the initial candidate
    # ------------------------------------------------------------------ #
    candidate = np.full(n, m, dtype=np.float64)
    candidate[sel] = H

    best = candidate.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------ #
    # 5. Small local search on the background value (fast, few evaluations)
    # ------------------------------------------------------------------ #
    for _ in range(30):
        if time.monotonic() > deadline:
            break
        # propose a perturbed background (log‑normal step)
        factor = 1.0 + rng.normal(0.0, 0.2)      # ~20 % variation
        m2 = m * factor
        m2 = max(0.0, min(1000.0, m2))

        cand = np.full(n, m2, dtype=np.float64)
        cand[sel] = H
        sc = float(evaluate_sequence(cand.tolist()))
        if sc < best_score:
            best = cand
            best_score = sc
            m = m2                               # keep the improvement

    # ------------------------------------------------------------------ #
    # 6. One final safety check: if a uniform vector happens to be better
    # ------------------------------------------------------------------ #
    uniform = np.full(n, 1.0, dtype=np.float64)
    sc_uni = float(evaluate_sequence(uniform.tolist()))
    if sc_uni < best_score:
        best = uniform
        best_score = sc_uni

    # Return as a plain Python list of finite floats
    return [float(v) for v in best]
```
