import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast score for a vector already normalized to sum≈1."""
    a = np.asarray(a, dtype=np.float64)

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

            a_fft = np.fft.rfft(a, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[: 2 * n - 1]
            max_val = conv.max()

            # temperature schedule
            if temp is None:
                temp = max_val * 0.1
            else:
                temp = max(temp * decay, max_val * min_temp_factor)

            # soft‑max weighting
            w = np.exp((conv - max_val) / (temp + 1e-20))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = (conv == max_val).astype(np.float64)
                w_sum = w.sum()
            w /= w_sum

            # gradient (FFT‑based)
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            grad = 2.0 * grad_full[idx]
            grad = (grad + grad[::-1]) * 0.5      # enforce symmetry

            # momentum & step
            momentum = beta * momentum + (1.0 - beta) * grad
            step = step0 / math.sqrt(it)

            # candidate & projection
            cand = a - step * momentum
            cand = _project_simplex(cand)
            cand = (cand + cand[::-1]) * 0.5
            cand = _project_simplex(cand)

            cand_sc = _score(cand)

            if cand_sc < a_sc:
                a, a_sc = cand, cand_sc
                no_improve = 0
                if a_sc < best_sc:
                    best_sc, best_vec = a_sc, a.copy()
            else:
                no_improve += 1
                prob = math.exp(-(cand_sc - a_sc) / (step + 1e-12))
                if rng.random() < prob:
                    a, a_sc = cand, cand_sc

            # random redistribution when stuck
            if no_improve > 2000:
                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = a[i] * rng.random() * 0.3
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < a_sc:
                    a, a_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()

            # occasional explicit symmetrisation
            if it % 500 == 0:
                sym = (a + a[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_sc = _score(sym)
                if sym_sc < a_sc:
                    a, a_sc = sym, sym_sc
                    if sym_sc < best_sc:
                        best_sc, best_vec = sym_sc, sym.copy()

            if best_sc < target:
                break
            if time.time() >= deadline:
                break

        restart += 1
        if best_sc < target:
            break

    # ------------------------------------------------------------------ #
    # Final local refinement (hill‑climbing)
    a = best_vec.copy()
    a_sc = best_sc
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        eps = a[i] * rng.random() * 0.015
        if eps <= 0.0:
            continue
        tmp = a.copy()
        tmp[i] -= eps
        tmp[j] += eps
        tmp = _project_simplex(tmp)
        tmp = (tmp + tmp[::-1]) * 0.5
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp)
        if tmp_sc < a_sc:
            a, a_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
        else:
            if rng.random() < 0.001:
                a, a_sc = tmp, tmp_sc

    # ------------------------------------------------------------------ #
    # Return the best vector found (as plain Python floats)
    return [float(x) for x in best_vec]
