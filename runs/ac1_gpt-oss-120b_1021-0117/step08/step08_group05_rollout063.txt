import time, math, numpy as np

def _project_simplex(v):
    """Project onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---------- initialization ----------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)                     # sum = 1

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())            # power‑of‑two FFT length
    idx = (2 * n - 2) - np.arange(n)               # index map for gradient

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_alpha = 0.03

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    # initial score (sum is 1, so score = 2·n·max(conv))
    a_fft = np.fft.rfft(cur, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_conv = conv.max()
    cur_sc = 2.0 * n * max_conv
    best_vec = cur.copy()
    best_sc = cur_sc

    T = None               # temperature for soft‑max
    it = 0
    no_improve = 0
    target = 1.50274349

    # ---------- main optimisation loop ----------
    while time.time() < deadline:
        it += 1

        # ----- convolution & temperature -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_conv = conv.max()
        if T is None:
            T = max_conv * 0.07
        else:
            T = max(T * 0.9995, max_conv * 1e-5)

        # ----- soft‑max weighting (smooth max) -----
        w = np.exp((conv - max_conv) / T)
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = (conv == max_conv).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient of smoothed max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (base_alpha / (1.0 + it * 1e-4)) * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        a_fft_c = np.fft.rfft(cand, n=L)
        conv_c = np.fft.irfft(a_fft_c * a_fft_c, n=L)[: 2 * n - 1]
        cand_sc = 2.0 * n * conv_c.max()

        # ----- accept or reject (annealing) -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (T + 1e-12)):
                cur, cur_sc = cand, cand_sc

        # ----- random redistribution when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            a_fft_t = np.fft.rfft(tmp, n=L)
            conv_t = np.fft.irfft(a_fft_t * a_fft_t, n=L)[: 2 * n - 1]
            tmp_sc = 2.0 * n * conv_t.max()
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()

        # ----- symmetrisation -----
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            a_fft_s = np.fft.rfft(sym, n=L)
            conv_s = np.fft.irfft(a_fft_s * a_fft_s, n=L)[: 2 * n - 1]
            sym_sc = 2.0 * n * conv_s.max()
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ---------- final polishing (tiny steps) ----------
    cur = best_vec.copy()
    cur_sc = best_sc
    for _ in range(200):
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_conv = conv.max()
        T = max_conv * 1e-6
        w = np.exp((conv - max_conv) / T)
        w /= w.sum()
        grad = 2.0 * np.fft.irfft(a_fft * np.fft.rfft(w[::-1], n=L), n=L)[idx]
        step = 0.005 * grad
        cand = cur - step
        cand = _project_simplex(cand)
        a_fft_c = np.fft.rfft(cand, n=L)
        conv_c = np.fft.irfft(a_fft_c * a_fft_c, n=L)[: 2 * n - 1]
        cand_sc = 2.0 * n * conv_c.max()
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            break

    # Return the best vector found
    return [float(x) for x in best_vec]
