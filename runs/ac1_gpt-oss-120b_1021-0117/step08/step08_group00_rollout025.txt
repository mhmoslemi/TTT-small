def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    start = time.time()
    deadline = start + max(0, budget_s - 0.5) # safety

    # Initialize
    if 'height_sequence_1' in globals():
        cur = np.array(height_sequence_1, dtype=np.float64)
    else:
        # create a cosine window
        n_default = 2000
        xs = np.arange(n_default)
        cur = 0.5 * (1 - np.cos(2 * math.pi * xs / (n_default - 1)))
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur) # sum to 1

    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of 2 >= 2n-1

    # functions
    def compute_conv_and_max(a):
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]
        return conv, conv.max()
    def compute_score(a):
        _, mx = compute_conv_and_max(a)
        return 2.0 * n * mx

    # initial
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score

    # momentum
    momentum = np.zeros_like(cur)

    # hyperparameters
    beta = 0.95
    step0 = 1.0

    it = 0
    no_improve = 0

    # main loop
    while time.time() < deadline:
        it += 1
        # compute convolution and gradient
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]
        max_val = conv_full.max()
        # locate max indices
        # use tolerance to capture near-equal values
        tol = max_val * 1e-12 + 1e-14
        mask = np.abs(conv_full - max_val) <= tol
        # indicator vector w (binary)
        w = mask.astype(np.float64)
        # compute gradient via convolution with reversed w
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        # indices for gradient (size n)
        idx = (2 * n - 2) - np.arange(n)  # because convolution result length = 2n-1
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # adaptive step size
        step = step0 / math.sqrt(it)

        # candidate
        cand = cur - step * momentum
        cand = _project_simplex(cand)
        cand_score = compute_score(cand)

        # Accept if better
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            no_improve += 1
            # occasional stochastic acceptance
            if np.random.rand() < math.exp(-(cand_score - cur_score) / (step + 1e-12)):
                cur = cand
                cur_score = cand_score

        # occasional random redistribution when stuck
        if no_improve > 2000:
            # random move: pick two indices i, j, transfer some mass
            i = np.random.randint(0, n)
            j = np.random.randint(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * np.random.rand() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_score = compute_score(tmp)
            if tmp_score < cur_score:
                cur = tmp
                cur_score = tmp_score
                no_improve = 0
                if tmp_score < best_score:
                    best = tmp.copy()
                    best_score = tmp_score

        # periodic symmetrisation (every 500 iters)
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = compute_score(sym)
            if sym_score < cur_score:
                cur = sym
                cur_score = sym_score
                if sym_score < best_score:
                    best = sym.copy()
                    best_score = sym_score

        # occasional sparsify: keep top proportion
        if it % 1200 == 0:
            thresh = np.quantile(cur, 0.1)  # keep top 90%
            tmp = cur.copy()
            tmp[tmp < thresh] = 0.0
            tmp = _project_simplex(tmp)
            tmp_score = compute_score(tmp)
            if tmp_score < cur_score:
                cur = tmp
                cur_score = tmp_score
                if tmp_score < best_score:
                    best = tmp.copy()
                    best_score = tmp_score

        # early exit
        if best_score < 1.50274349:
            break

    # final return
    return [float(x) for x in best]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of floats) with low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ---- initialise ----------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # a smooth cosine window as a generic start
        n0 = 2000
        xs = np.arange(n0, dtype=np.float64)
        cur = 0.5 * (1.0 - np.cos(2.0 * math.pi * xs / (n0 - 1.0)))
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)               # enforce sum = 1
    n = cur.shape[0]

    # FFT length (next power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    # ---- helpers -------------------------------------------------------
    def _conv_max(a_fft):
        """Return full convolution (len=2n‑1) and its maximum."""
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        return conv, conv.max()

    def _score(a):
        """Score = 2·n·max(conv).  Assumes sum(a)=1."""
        a_fft = np.fft.rfft(a, n=L)
        _, mx = _conv_max(a_fft)
        return 2.0 * n * mx

    # ---- optimisation state --------------------------------------------
    cur_score = _score(cur)
    best_vec = cur.copy()
    best_score = cur_score

    momentum = np.zeros_like(cur)

    # hyper‑parameters
    beta = 0.90                # momentum decay
    step0 = 0.8                # base step magnitude
    it = 0
    no_improve = 0
    target = 1.50274349

    # ---- main loop -----------------------------------------------------
    while time.time() < deadline:
        it += 1

        # FFT of current vector
        a_fft = np.fft.rfft(cur, n=L)

        # full convolution and its max
        conv_full, max_val = _conv_max(a_fft)

        # ---- sub‑gradient of the max (binary mask) ----
        # tolerance to collect all (numerically) maximal entries
        tol = max_val * 1e-12 + 1e-14
        mask = np.abs(conv_full - max_val) <= tol
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 1.0:
            w /= cnt                         # average over active constraints

        # gradient via convolution with reversed mask
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)     # map to length‑n gradient
        grad = 2.0 * grad_full[idx]          # sub‑gradient of max(conv)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # adaptive step size
