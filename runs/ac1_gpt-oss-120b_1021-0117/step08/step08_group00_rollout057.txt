    # log‑it for a soft‑max parametrisation (sum = 1 enforced automatically)
    y = np.log(cur + eps)
    y = y - np.max(y)                               # any shift is fine

    n = y.shape[0]
    L = 1 << ((2 * n - 1).bit_length())            # power‑of‑two FFT size

    # ---- Adam optimiser --------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    alpha0 = 0.05                                   # base learning rate
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1_pow, beta2_pow = beta1, beta2

    # ---- tracking ---------------------------------------------------------
    best_a = None
    best_score = np.inf
    last_best_iter = 0
    temp = None                                     # temperature for soft‑max weighting
    it = 0

    # ----------------------------------------------------------------------
    while True:
        it += 1
        if time.time() >= deadline:
            break

        # current probability vector (sum = 1)
        a = _softmax_logits(y)

        # convolution a * a via FFT
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val                # exact AC1 bound (sum = 1)

        # record best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_best_iter = it
            if best_score < 1.50274349:              # target reached
                break

        # ----- temperature for smooth max ---------------------------------
        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * 0.9995, max_val * 1e-5)

        # soft‑max weights over the convolution entries
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:                             # fallback to hard max
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of the weighted max w.r.t. a (FFT‑based)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * grad_full[idx]

        # chain rule through soft‑max: grad_y = a * (grad_a - ⟨a,grad_a⟩)
        dot = np.dot(a, grad_a)
        grad_y = a * (grad_a - dot)

        # Adam update with a decaying step size
        lr = alpha0 / math.sqrt(it)
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- occasional operations ---------------------------------------
        if it % 800 == 0:                             # symmetrisation
            y = (y + y[::-1]) / 2.0
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow, beta2_pow = beta1, beta2

        if it % 1500 == 0:                            # sparsify low entries
            a = _softmax_logits(y)
            thresh = np.quantile(a, 0.2)               # prune bottom 20 %
            a_s = a.copy()
            a_s[a_s < thresh] = 0.0
            y = np.log(a_s + eps)
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow, beta2_pow = beta1, beta2

        if it - last_best_iter > 3000:                # escape stagnation
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = a[i] * rng.random() * 0.4
            a_pert = a.copy()
            a_pert[i] -= delta
            a_pert[j] += delta
            y = np.log(a_pert + eps)
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow, beta2_pow = beta1, beta2
            last_best_iter = it

    # ----------------------------------------------------------------------
    if best_a is None:                               # safety fallback
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    # return plain Python floats as required by the harness
    return [float(x) for x in best_a]
