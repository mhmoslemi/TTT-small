def _gradient_max_conv(a, conv):
    n = a.shape[0]
    max_val = conv.max()
    # mask positions within tolerance
    tol = max(1e-12, max_val * 1e-12)
    ks = np.where(conv >= max_val - tol)[0]
    g = np.zeros_like(a)
    for k in ks:
        # compute idx bounds
        idx_low = max(0, k - (n - 1))
        idx_high = min(n - 1, k)
        # corresponding j bounds
        j_low = k - idx_high
        j_high = k - idx_low
        # slice
        a_slice = a[idx_low: idx_high + 1]
        # reversed slice
        g[j_low: j_high + 1] += 2.0 * a_slice[::-1]
    return g

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0.0] = 0.0
    return w

def _score(a):
    """Exact AC1 bound for vector a."""
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_conv = conv.max()
    return float(2.0 * n * max_conv / (sum_a * sum_a))

def _gradient_max(a, conv):
    """Sub‑gradient of max(conv) w.r.t a (exact, using mask of maximal entries)."""
    n = a.shape[0]
    max_val = conv.max()
    # treat values within a tiny relative tolerance as maximal
    tol = max(1e-12, max_val * 1e-12)

    grad = np.zeros_like(a)

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        j_lo = k - hi
        j_hi = k - lo
        # reversed slice of a contributes to grad
        grad[j_lo:j_hi + 1] += 2.0 * a[lo:hi + 1][::-1]
    return grad
