def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # init
    if 'height_sequence_1' in globals():
        cur = np.array(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)  # sum=1

    n = cur.shape[0]
    L = 1 << ((2*n-1).bit_length())

    def score_vec(a):
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())

    best_vec = cur.copy()
    best_sc = score_vec(best_vec)

    # Adam variables
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha0 = 0.2  # starting learning rate

    it = 0
    no_improve = 0
    warmup = 5000

    while time.time() < deadline:
        it += 1

        # FFT
        cur_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(cur_fft * cur_fft, n=L)[:2*n-1]
        max_val = conv_full.max()

        # choose weight
        if it <= warmup:
            temp = max(max_val * 0.07, 1e-12)
            w = np.exp((conv_full - max_val) / temp)
            w_sum = w.sum()
            if w_sum == 0:
                w = (conv_full == max_val).astype(np.float64)
                w_sum = w.sum()
            w /= w_sum
        else:
            w = (conv_full == max_val).astype(np.float64)
            if w.sum() == 0:  # shouldn't happen
                w[0] = 1.0
            w = w / w.sum()

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(cur_fft * w_fft, n=L)[:2*n-1]
        idx = (2*n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)

        step = alpha0 / (np.sqrt(it) + 1.0)
        direction = m_hat / (np.sqrt(v_hat) + eps)
        cand = cur - step * direction
        cand = _project_simplex(cand)

        cand_sc = score_vec(cand)

        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
            cur = cand
            # reset moments for fresh start after improvement
            m = np.zeros_like(cur)
            v = np.zeros_like(cur)
            no_improve = 0
        else:
            no_improve += 1
            # optional acceptance for exploration
            if rng.random() < 0.001:
                cur = cand

        # periodic symmetry enforcement
        if it % 2000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = score_vec(sym)
            if sym_sc < best_sc:
                best_sc = sym_sc
                best_vec = sym.copy()
                cur = sym
                m = np.zeros_like(cur)
                v = np.zeros_like(cur)

        # occasional random redistribution when stuck
        if no_improve > 8000:
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = score_vec(tmp)
            if tmp_sc < best_sc:
                best_sc = tmp_sc
                best_vec = tmp.copy()
                cur = tmp
            no_improve = 0

        # early exit if target reached
        if best_sc < 1.50274349:
            break

    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(vec):
    """Return the AC1 upper bound for a vector (sum is assumed =1)."""
    a = np.asarray(vec, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    n = cur.shape[0]

    m = np.zeros_like(cur)     # first moment (Adam)
    v = np.zeros_like(cur)     # second moment (Adam)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha0 = 0.25               # initial learning rate
    it = 0

        cur_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(cur_fft * cur_fft, n=L)[: 2 * n - 1]
