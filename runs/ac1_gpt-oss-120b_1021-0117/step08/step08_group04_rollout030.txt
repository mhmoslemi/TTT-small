import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0.0] = 0.0
    return w

def _score_norm1(a):
    """Score for a vector normalized to sum=1: 2*n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---------------------- initialization ----------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = np.clip(cur, 0.0, 1000.0)
    cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best = cur.copy()
    best_score = _score_norm1(best)
    cur_score = best_score

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.02
    max_step = 0.2

    it = 0
    no_improve = 0
    target = 1.50274349
    temp = None

    # ------------------------ main loop ------------------------
    while time.time() < deadline:
        it += 1

        # --- convolution and temperature ---
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        if temp is None:
            temp = max_val * 0.08
        else:
            temp = max(temp * 0.9995, max_val * 1e-5)

        # --- softmax weights for a smooth sub‑gradient ---
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # --- weighted gradient via FFT (2 * (w * a_rev)) ---
        a_rev = cur[::-1]
        w_fft = np.fft.rfft(w, n=L)
        a_rev_fft = np.fft.rfft(a_rev, n=L)
        grad_full = np.fft.irfft(w_fft * a_rev_fft, n=L)
        g = 2.0 * grad_full[n - 1 : n - 1 + n]  # length n

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (base_lr / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps)

        # limit step size
        step_norm = np.linalg.norm(step)
        if step_norm > max_step:
            step = step * (max_step / step_norm)

        cand = _project_simplex(cur - step)
        cand_score = _score_norm1(cand)

        # Metropolis‑style acceptance
        if cand_score < cur_score or rng.random() < math.exp(
            -(cand_score - cur_score) / (temp + 1e-12)
        ):
            cur = cand
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            no_improve += 1

        # --- occasional symmetrisation ---
        if it % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score_norm1(sym)
            if sym_score < cur_score:
                cur = sym
                cur_score = sym_score
                no_improve = 0
                if sym_score < best_score:
                    best = sym.copy()
                    best_score = sym_score

        # --- random redistribution if stuck ---
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_score = _score_norm1(tmp)
            if tmp_score < cur_score:
                cur = tmp
                cur_score = tmp_score
                no_improve = 0
                if tmp_score < best_score:
                    best = tmp.copy()
                    best_score = tmp_score

        if best_score <= target:
            break

    # -------------------- final polishing --------------------
    # small random transfers
    while time.time() < deadline - 0.05:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = best[i] * rng.random() * 0.1
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_score = _score_norm1(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        # occasional smoothing step
        if rng.random() < 0.02:
            smooth = best.copy()
            smooth[0] = (best[0] + best[1]) * 0.5
            smooth[-1] = (best[-2] + best[-1]) * 0.5
            smooth[1:-1] = (best[:-2] + best[1:-1] + best[2:]) / 3.0
            smooth = _project_simplex(smooth)
            smooth_score = _score_norm1(smooth)
            if smooth_score < best_score:
                best = smooth
                best_score = smooth_score

    return [float(x) for x in best]
