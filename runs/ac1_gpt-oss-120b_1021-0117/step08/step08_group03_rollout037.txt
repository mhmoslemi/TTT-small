import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v):
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees non‑negative entries that sum to 1.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def propose_candidate(seed: int = 42,
                      budget_s: float = 1100,
                      **kwargs) -> list:
    """
    Returns a non‑negative step‑function (list of floats) that reduces
    the AC1 upper bound.  The routine respects the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # tiny safety margin

    # ---- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: uniform simplex
        n = 2500
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    cur = _project_simplex(np.maximum(cur, 0.0))
    n = cur.shape[0]

    # ---- FFT parameters -------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())          # power of two >= 2n‑1
    idx = (2 * n - 2) - np.arange(n)             # slice for gradient

    # ---- Adam optimiser state -------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.12                               # base learning rate
    soft_alpha = 5.0                             # softness of the max‑approx

    # ---- bookkeeping ----------------------------------------------------
    # current score (2 * n * max(conv)) – sum is 1 by construction
    af_cur = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(af_cur * af_cur, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    best_vec = cur.copy()
    best_score = 2.0 * n * max_val
    cur_score = best_score

    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- forward FFT & convolution --------------------------------
        af = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(af * af, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- smooth approximation of the max -------------------------
        shift = conv - max_val                     # ≤ 0
        exp_shift = np.exp(soft_alpha * shift)    # under‑flow safe
        w = exp_shift / exp_shift.sum()           # softmax weights

        # ----- gradient of the weighted sum (matrix‑free) ---------------
        w_rev = w[::-1]
        wf = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(af * wf, n=L)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
