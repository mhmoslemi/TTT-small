import time, math, numpy as np

def _project_simplex(v):
    """Project non‑negative vector v onto the probability simplex (sum = 1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0.0] = 0.0
    return w

def _score_from_conv(conv, n):
    """Given full convolution (length 2n‑1) and length n, return the AC1 score."""
    return float(2.0 * n * conv.max())
