def grad_from_argmax(a, argmax_positions):
    n = len(a)
    grad = np.zeros_like(a)
    for k in argmax_positions:
        d = k - (n - 1)   # shift offset
        if d >= 0:
            # a shifted right by d: a[d:] corresponds to indices i where i-d >=0
            # gradient[i] += 2 * a[i - d] for i >= d
            grad[d:] += 2.0 * a[:n-d]
        else:
            # d < 0: shift left
            # gradient[i] += 2 * a[i - d] = a[i + |d|]
            grad[:n+d] += 2.0 * a[-d:]
    return grad

import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

    n = cur.shape[0]
