import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a, n, L):
    """Fast surrogate of the true score (exact when sum(a)=1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

                best_sc, best_vec = tmp_sc, tmp.copy()

        # early exit if target already reached
        if best_sc < 1.5027:
            break

    # ------------------------------------------------------------------
    # final sanity check – ensure the returned vector yields a finite score
    try:
        final_sc = evaluate_sequence(best_vec.tolist())
        if not (math.isfinite(final_sc) and final_sc < 1e9):
            raise Exception
    except Exception:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(x) for x in best_vec]
