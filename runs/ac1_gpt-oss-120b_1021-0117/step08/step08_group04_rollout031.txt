def _project_simplex(v):
    """Project onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_norm1(a):
    """Score assuming sum(a)=1: 2*n*max(convolution(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ----- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = (cur + cur[::-1]) * 0.5               # enforce symmetry
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best = cur.copy()
    best_score = _score_norm1(best)
    cur_score = best_score

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.1          # base learning rate
    max_step = 1.0

    temp = None
    temp_decay = 0.9995
    min_temp_factor = 1e-6

    it = 0
    no_improve = 0
    target = 1.50274349

    # ----- main optimization loop -----------------------------------------
    while time.time() < deadline:
        it += 1

        # convolution of current vector with itself
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # temperature for soft‑max weighting
        if temp is None:
            temp = max_val * 0.08
        else:
            temp = max(temp * temp_decay, max_val * min_temp_factor)

        # soft‑max weights over all lags
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of the smoothed max‑convolution
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 / math.sqrt(it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # limit step length
        step_norm = np.linalg.norm(step)
        if step_norm > max_step:
            step = step * (max_step / step_norm)

        cand = cur - step
        cand = (cand + cand[::-1]) * 0.5               # keep symmetry
        cand = _project_simplex(np.maximum(cand, 0.0))
        cand_score = _score_norm1(cand)

        # acceptance test (Metropolis)
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / (temp + 1e-12)):
            cur, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
        else:
            no_improve += 1

        # ----- exploration when stuck ------------------------------------
        if no_improve > 2000:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = cur[i] * rng.random() * 0.1
                tmp = cur.copy()
                tmp[i] = max(0.0, tmp[i] - delta)
                tmp[j] += delta
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp = _project_simplex(np.maximum(tmp, 0.0))
                tmp_score = _score_norm1(tmp)
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                    no_improve = 0
                    if tmp_score < best_score:
                        best, best_score = tmp.copy(), tmp_score

        # ----- periodic symmetrisation ----------------------------------
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(np.maximum(sym, 0.0))
            sym_score = _score_norm1(sym)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                no_improve = 0
                if sym_score < best_score:
                    best, best_score = sym.copy(), sym_score

        # ----- hard restart if really stuck -------------------------------
        if no_improve > 8000:
            cur = best + rng.normal(scale=0.01, size=n)
            cur = (cur + cur[::-1]) * 0.5
            cur = _project_simplex(np.maximum(cur, 0.0))
            cur_score = _score_norm1(cur)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        if best_score <= target:
            break

    # ----- final local refinement -----------------------------------------
    ft_deadline = deadline - 0.1
    while time.time() < ft_deadline:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.05
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(np.maximum(cand, 0.0))
        cand_score = _score_norm1(cand)
        if cand_score < best_score:
            best, best_score = cand, cand_score

    return [float(x) for x in best]
