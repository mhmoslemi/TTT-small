def run_opt(initial, time_budget, target=1.50274349):
    cur = initial.copy()
    cur = np.maximum(cur, 0)
    cur = _project_simplex(cur)
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())
    idx = (2 * n - 2) - np.arange(n)
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    base_alpha = 0.2
    best_vec = cur.copy()
    best_sc = _score_simplex(best_vec, n)
    cur_sc = best_sc
    it = 0
    stagnation = 0
    while True:
        if time.time() >= deadline:
            break
        it += 1

        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        # compute subgradient indicator
        tol = max(1e-12 * max_val, 1e-14)
        mask = conv >= max_val - tol
        cnt = mask.sum()
        if cnt == 0:
            mask = conv == max_val
            cnt = mask.sum()
        w = mask.astype(np.float64) / cnt
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = base_alpha / np.sqrt(it) * m_hat / (np.sqrt(v_hat) + epsilon)

        cand = cur - step
        cand = _project_simplex(cand)
        # compute Score
        cand_sc = _score_simplex(cand, n)
        # accept if better
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            stagnation = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            stagnation += 1
            # occasional uphill based on random acceptance with probability
            # use temperature based on max_val
            T = max_val * 0.02  # small
            if np.random.rand() < math.exp(-(cand_sc - cur_sc) / (T + 1e-12)):
                cur, cur_sc = cand, cand_sc

        # occasional symmetrization
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_simplex(sym, n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # occasional random perturbation when stuck
        if stagnation > 1500:
            # pick two indices and transfer some mass
            i = np.random.randint(0, n)
            j = np.random.randint(0, n-1)
            if j >= i:

            delta = cur[i] * np.random.rand() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)

                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
        # early exit
        if best_sc < target:
            break

import time, math, numpy as np

# ---------- helpers ----------
def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_simplex(a, n):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ---------- core optimizer ----------
def _optimize_one(init_vec, deadline, target):
    """Run projected Adam on a single start vector until deadline."""
    cur = _project_simplex(np.maximum(init_vec, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())
    idx = (2 * n - 2) - np.arange(n)               # indices for gradient extraction

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_alpha = 0.2

    best_vec = cur.copy()
    best_sc = _score_simplex(best_vec, n)
    cur_sc = best_sc

    it = 0
    stagn = 0
    rng = np.random.default_rng()

    while True:
        if time.time() >= deadline:
            break
        it += 1

        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # subgradient: indicator of (almost) maximal entries
        tol = max(1e-12 * max_val, 1e-14)
        mask = conv >= max_val - tol
        cnt = int(mask.sum())
        if cnt == 0:
            mask = conv == max_val
            cnt = int(mask.sum())
        w = mask.astype(np.float64) / cnt
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (base_alpha / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score_simplex(cand, n)

        # acceptance
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            stagn = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            stagn += 1
            # occasional uphill move (simulated annealing style)
            T = max_val * 0.02
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (T + 1e-12)):
                cur, cur_sc = cand, cand_sc

        # periodic symmetrisation
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_simplex(sym, n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # random mass transfer when stuck
        if stagn > 1500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_simplex(tmp, n)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                stagn = 0
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()

        # early exit if target reached
        if best_sc < target:
            break

    return best_vec, best_sc

# ---------- main entry ----------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # length determination
    if "height_sequence_1" in globals():
        n = int(np.asarray(height_sequence_1).size)
    else:
        n = 2500

    # build a set of diverse initial vectors
    inits = []

    # 1) parent if present
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        inits.append(parent)

    # 2) random simplex vector
    rand_vec = rng.random(n, dtype=np.float64)
    inits.append(rand_vec)

    # 3) symmetric cosine shape
    x = np.arange(n, dtype=np.float64)
    cos_shape = (np.cos(np.pi * (x - (n - 1) / 2) / (n - 1)) + 1.0) * 0.5
    inits.append(cos_shape)

    # 4) triangular shape
    tri_shape = 1.0 - np.abs(x - (n - 1) / 2) / ((n - 1) / 2)
    inits.append(tri_shape)

    # 5) Gaussian bump
    sigma = 0.2
    gauss_shape = np.exp(-0.5 * ((x - (n - 1) / 2) / (sigma * n)) ** 2)
    inits.append(gauss_shape)

    # keep the best across runs
    global_best_vec = None
    global_best_sc = float('inf')
    target = 1.50274349

    for init in inits:
        if time.time() >= deadline:
            break
        # allocate a per‑run deadline (leave a small buffer for later runs)
        run_deadline = min(deadline, time.time() + 30.0)  # at most 30 s per start
        vec, sc = _optimize_one(init, run_deadline, target)
        if sc < global_best_sc:
            global_best_sc, global_best_vec = sc, vec
        if global_best_sc < target:
            break

    # fallback – if everything failed, use a simple normalized random vector
    if global_best_vec is None:
        fallback = rng.random(n, dtype=np.float64)
        fallback = _project_simplex(fallback)
        global_best_vec = fallback

    # return as plain Python list
    return [float(x) for x in global_best_vec]
