    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # Determine n and initial a
    if "height_sequence_1" in globals():

    # uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_vec = cur0.copy()
    # compute objective via conv max
    L = 1 << ((2 * n - 1).bit_length())
    # function to compute objective (score)
    def _obj(a):
        max_val = _conv_max(a, L)
        return 2.0 * n * max_val
    best_obj = _obj(best_vec)
    uni_obj = _obj(uniform)
    if uni_obj < best_obj:
        best_obj = uni_obj
        best_vec = uniform.copy()
    target = 1.50274349

    # hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr0 = 0.5
    restart = 0

    while time.time() < deadline:
        # initialise current vector
        if restart == 0:
            cur = best_vec.copy()
        else:
            cur = rng.exponential(scale=1.0, size=n)
            cur = _project_simplex(cur)
        cur_obj = _obj(cur)

        # Adam moments
        m = np.zeros_like(cur)
        v = np.zeros_like(cur)
        t_iter = 0

        # inner loop
        while time.time() < deadline:
            t_iter += 1
            # compute FFT of cur
            af = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
            max_val = conv.max()
            # hardmax weighting
            mask = (conv == max_val)
            w = mask.astype(np.float64)
            w_sum = w.sum()
            if w_sum > 0:
                w /= w_sum
            else:
                w[0] = 1.0  # fallback
                w_sum = 1.0

            # gradient via convolution with w_rev
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(af * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            grad = 2.0 * grad_full[idx]

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)
            step = lr0 * m_hat / (np.sqrt(v_hat) + epsilon)

            cand = _project_simplex(cur - step)
            cand_obj = _obj(cand)

            if cand_obj < cur_obj:
                cur, cur_obj = cand, cand_obj
                # update best

                    best_obj, best_vec = cand_obj, cand.copy()
                    if best_obj < target:
                        break

                    cur, cur_obj = cand, cand_obj

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _conv_max(a, L):
    """Maximum value of the full convolution a*a using an FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.shape[0] - 1]
    return conv.max()

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------ #
    # Problem size and initial vectors
    if "height_sequence_1" in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64)
        n = init.shape[0]
    else:
        n = 2000
        init = rng.random(n, dtype=np.float64)

    # Project onto simplex (sum = 1) and set a uniform baseline
    cur0 = _project_simplex(np.maximum(init, 0.0))
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # Length for FFT (next power of two >= 2n-1)
    L = 1 << ((2 * n - 1).bit_length())

    # Helper: objective for a vector whose sum is 1
    def _obj(a):
        return 2.0 * n * _conv_max(a, L)

    best_vec = cur0.copy()
    best_obj = _obj(best_vec)

    uni_obj = _obj(uniform)
    if uni_obj < best_obj:
        best_vec, best_obj = uniform.copy(), uni_obj

    target = 1.50274349          # desired upper bound

    # ------------------------------------------------------------------ #

    beta1, beta2 = 0.9, 0.999    # Adam moments

    while time.time() < deadline:
        # ------------------------------------------------------------------

        if restart == 0:
            cur = best_vec.copy()
        else:

            cur = _project_simplex(cur)

        cur_obj = _obj(cur)

        m = np.zeros_like(cur)   # first‑moment accumulator
        v = np.zeros_like(cur)   # second‑moment accumulator
        it = 0

        # --------------------------- inner optimisation -------------------
        while time.time() < deadline:
            it += 1

            # FFT of current vector and its full convolution
            af = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
            max_val = conv.max()

            # Hard‑max weighting (sub‑gradient of the max)
            mask = (conv == max_val)
            w = mask.astype(np.float64)
            w_sum = w.sum()
            if w_sum > 0:

            else:
                w[0] = 1.0

            # Gradient of the weighted max (matrix‑free via FFT)

            idx = (2 * n - 2) - np.arange(n)

            # Adam update

            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)

            step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

                if sym_obj < best_obj:
                    best_vec, best_obj = sym.copy(), sym_obj

    # Return the best vector found (as plain Python floats)
    return [float(x) for x in best_vec]
