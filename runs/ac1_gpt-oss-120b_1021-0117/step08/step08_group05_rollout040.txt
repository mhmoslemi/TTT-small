import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_simplex(a, n):
    """Score for a vector a with sum(a)=1: 2*n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)  # margin for finalisation

    # ---------- initial vector ----------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = (cur + cur[::-1]) * 0.5                    # enforce symmetry
    cur = _project_simplex(cur)                     # sum = 1

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())
    idx = (2 * n - 2) - np.arange(n)                # indices to extract gradient

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    alpha = 0.04
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    best_vec = cur.copy()
    best_sc = _score_simplex(best_vec, n)
    cur_sc = best_sc

    T = None
    it = 0
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1

        # ---- convolution of current vector ----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()

        # ---- temperature schedule (soft‑max smoothing) ----
        if T is None:
            T = max_conv * 0.10
        else:
            T = max(T * 0.9995, max_conv * 5e-5)

        # ---- soft‑max weights (or exact indicator when T tiny) ----
        if T < max_conv * 1e-8:
            w = (conv == max_conv).astype(np.float64)
            w = w / w.sum()
        else:
            w = np.exp((conv - max_conv) / T)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = (conv == max_conv).astype(np.float64)
                w_sum = w.sum()
            w = w / w_sum

        # ---- gradient of the smoothed max ----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (alpha / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + epsilon)

        cand = cur - step
        cand = _project_simplex(cand)
        cand = (cand + cand[::-1]) * 0.5               # keep symmetry

        cand_sc = _score_simplex(cand, n)

        # ---- accept / reject (Simulated Annealing) ----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (T + 1e-12)):
                cur, cur_sc = cand, cand_sc

        # ---- occasional random redistribution to escape plateaus ----
        if no_improve > 2000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp = (tmp + tmp[::-1]) * 0.5
            tmp_sc = _score_simplex(tmp, n)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()

        # ---- periodic forced symmetrisation ----
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_simplex(sym, n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # ---- early stop if target reached ----
        if best_sc < target:
            break

    # ---------- fine‑tuning with exact max (indicator) ----------
    while time.time() + 0.5 < deadline:
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()
        w = (conv == max_conv).astype(np.float64)
        w = w / w.sum()
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # tiny normalized step
        step = 0.01 * grad / (np.linalg.norm(grad) + 1e-12)
        cand = cur - step
        cand = _project_simplex(cand)
        cand = (cand + cand[::-1]) * 0.5
        cand_sc = _score_simplex(cand, n)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cur_sc < best_sc:
                best_sc, best_vec = cur_sc, cur.copy()
        else:
            # occasional uphill move
            if rng.random() < 0.01:
                cur, cur_sc = cand, cand_sc

    return [float(x) for x in best_vec]
