import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _max_conv(a, L=None):

    af = np.fft.rfft(a, n=L)

        # ---- learning‑rate decay (gentle) ----
        lr *= 0.9999

    # final optional symmetrisation
    sym = (best_vec + best_vec[::-1]) * 0.5
    sym = _project_simplex(sym)
    sym_sc = 2.0 * n * _max_conv(sym, L)
    if sym_sc < best_sc:
        best_vec = sym

    return [float(x) for x in best_vec]
```
