import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        theta = css[rho[-1]] / (rho[-1] + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a, n, L):
    """Fast surrogate for the AC1 upper bound (uses FFT, assumes a sum≈1)."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    if s <= 0:
        return float('inf')
    return float(2.0 * n * max_val / (s * s))

def _max_contrib_index(a, k):
    """Return index i that maximizes a[i] * a[k-i] for a given lag k."""
    n = a.shape[0]
    i0 = max(0, k - (n - 1))
    i1 = min(n - 1, k)
    i = np.arange(i0, i1 + 1)
    prod = a[i] * a[k - i]
    idx = i[prod.argmax()]
    return int(idx)
