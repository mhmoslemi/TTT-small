import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # candidate step
        cand = cur - lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = _project_simplex(cand)

        cand_score = _score_vec(cand)

        # acceptance
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best_vec, best_score = cand.copy(), cand_score
        else:
            no_improve += 1
            prob = math.exp(-(cand_score - cur_score) / (temp + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score

        # ----- diversification when stuck ---------------------------------
        if no_improve > 2500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_vec(tmp)
            if tmp_sc < cur_score:
                cur, cur_score = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_score:
                    best_vec, best_score = tmp.copy(), tmp_sc

        # periodic symmetrisation
        if it % 700 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym)
            if sym_sc < cur_score:
                cur, cur_score = sym, sym_sc
                if sym_sc < best_score:
                    best_vec, best_score = sym.copy(), sym_sc

        # periodic sparsification
        if it % 1500 == 0:
            thresh = np.quantile(cur, 0.2)       # drop bottom 20 %
            tmp = cur.copy()
            tmp[tmp < thresh] = 0.0
            tmp = _project_simplex(tmp)
            tmp_sc = _score_vec(tmp)
            if tmp_sc < cur_score:
                cur, cur_score = tmp, tmp_sc
                if tmp_sc < best_score:
                    best_vec, best_score = tmp.copy(), tmp_sc

        # early exit
        if best_score < target:
            break

    # Return the best sequence found (as plain Python floats)
    return [float(x) for x in best_vec]
