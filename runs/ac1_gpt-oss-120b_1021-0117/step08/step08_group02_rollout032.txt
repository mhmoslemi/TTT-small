import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Return 2*n*max(conv(a,a)) assuming sum(a)=1 (the evaluator's score)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    lr, eps = 0.2, 1e-8
    min_lr = 0.01

    # temperature for soft‑max weighting
    temp = None
    decay = 0.998
    min_temp_factor = 1e-6

    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    # main optimisation loop
    while True:
        if time.time() >= deadline:
            break
        it += 1

        # full convolution via FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # update temperature
        if temp is None:
            temp = max_val * 0.1
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # soft‑max weights
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:                     # fallback to exact max mask
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of weighted max
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # candidate from Adam
        cand = _project_simplex(cur - step)
        cand_sc = _score(cand)

        # mirror‑descent (exponential) candidate
        eta = lr * 0.5
        exp_factor = np.exp(np.clip(-eta * grad, -50, 50))
        exp_vec = cur * exp_factor
        s = exp_vec.sum()
        if s > 0:
            exp_vec = exp_vec / s
        else:
            exp_vec = cur.copy()
        exp_sc = _score(exp_vec)

        # pick the best among cur, cand, exp_vec
        if cand_sc < cur_sc or exp_sc < cur_sc:
            if cand_sc <= exp_sc:
                cur, cur_sc = cand, cand_sc
            else:
                cur, cur_sc = exp_vec, exp_sc
            no_improve = 0
            if cur_sc < best_sc:
                best_vec, best_sc = cur.copy(), cur_sc
        else:
            # probabilistic acceptance of worse Adam step
            no_improve += 1
            prob = math.exp(-(cand_sc - cur_sc) / (temp + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # ------------------------------------------------------------------
        # occasional random redistribution when stuck
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

        # ------------------------------------------------------------------
        # occasional symmetrisation (helps for many instances)
        if it % 1500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # ------------------------------------------------------------------
        # periodic mild restart towards the current best
        if no_improve > 2000:
            rand_vec = rng.random(n)
            rand_vec = _project_simplex(rand_vec)
            cur = 0.9 * best_vec + 0.1 * rand_vec
            cur = _project_simplex(cur)
            cur_sc = _score(cur)
            no_improve = 0

        # ------------------------------------------------------------------
        # slowly decay learning rate
        if it % 500 == 0:
            lr = max(min_lr, lr * 0.95)

        # early exit if target reached
        if best_sc < target:
            break

    # ----------------------------------------------------------------------
    # final local refinement with tiny random moves
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = best_vec[i] * rng.random() * 0.05
        tmp = best_vec.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp)
        if tmp_sc < best_sc:
            best_vec, best_sc = tmp, tmp_sc

    return [float(x) for x in best_vec]
