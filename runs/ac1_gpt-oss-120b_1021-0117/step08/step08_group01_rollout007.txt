def propose_candidate(seed=42, budget_s=1100, **kwargs):

    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.1) # leave 0.1 sec margin

    # Determine n and initial vector
    if 'height_sequence_1' in globals():
        init_full = np.asarray(height_sequence_1, dtype=np.float64)
        n = init_full.shape[0]
    else:
        n = 2000
        init_full = rng.random(n, dtype=np.float64)

    # Normalize to sum 1 (project onto simplex)
    cur0 = _project_simplex(np.maximum(init_full, 0.0))
    best_vec = cur0.copy()
    best_score = _score(best_vec)

    # also try uniform as fallback:
    uniform = np.full(n, 1.0/n, dtype=np.float64)
    uniform_score = _score(uniform)
    if uniform_score < best_score:
        best_vec = uniform.copy()
        best_score = uniform_score

    # Prepare parameters
    L = 1 << ((2*n - 1).bit_length())
    beta = 0.95
    step0 = 0.5
    decay = 0.9995
    temp_factor_initial = 0.2
    min_temp_factor = 1e-5
    max_restarts = 1000 # approximate
    # track number of restarts, but we may break early.

    restart = 0

    while time.time() < deadline:
        # Choose start vector
        if restart == 0:
            cur = best_vec.copy()
        else:
            # Random Dirichlet (exponential) and optionally symmetric
            cur = rng.exponential(scale=1.0, size=n)
            cur = _project_simplex(cur)
            # with probability, symmetrize
            if rng.random() < 0.3:
                cur = (cur + cur[::-1]) * 0.5
                cur = _project_simplex(cur)
        cur_score = _score(cur)

        # momentum init
        momentum = np.zeros_like(cur)
        temp = None
        it = 0
        no_improve = 0

        while time.time() < deadline:
            it += 1

            # compute convolution via FFT
            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[: 2*n - 1]
            max_val = conv.max()

            # temperature schedule
            if temp is None:

            else:

            # compute softmax weights
            w = np.exp((conv - max_val) / (temp + 1e-20))
            w_sum = w.sum()
            if w_sum == 0.0:

            if cand_score < cur_score:

def _project_simplex(v):
    """Project onto probability simplex (sum=1, nonnegative)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n+1) > 0
    if not np.any(rho_arr):
        theta = 0.0
    else:
        rho = np.where(rho_arr)[0][-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L=None):
    """Score 2·n·max(a*a) assuming sum(a)=1 (fast via FFT)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
