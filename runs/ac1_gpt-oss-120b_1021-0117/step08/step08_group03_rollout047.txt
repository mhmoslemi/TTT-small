import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # shift to guarantee sum(v) == 1 after projection
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_from_maxconv(max_conv, n, sum_a):
    """Compute the exact evaluator score from max convolution value."""
    return float(2.0 * n * max_conv / (sum_a * sum_a))

def _max_conv_fft(a, L, n):
    """Return max of the linear convolution a*a (full length 2n-1)."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    return conv.max(), conv

def _grad_softmax_max(a, conv, L, n, alpha):
    """
    Gradient of the softmax-approximated max of conv w.r.t. a.
    Returns a vector of length n.
    """
    # softmax weights on conv (size 2n-1)
    max_val = conv.max()
    # shift for numerical stability
    w = np.exp(alpha * (conv - max_val))
    w /= w.sum()
    # reverse weights
    w_rev = w[::-1]

    # convolution a * w_rev (linear) via FFT
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    conv2_full = np.fft.irfft(af * wf, n=L)

    # gradient = 2 * (a * w_rev) evaluated on indices n-1 .. 2n-2, reversed
    grad = 2.0 * conv2_full[n - 1: 2 * n - 1][::-1]
    return grad
