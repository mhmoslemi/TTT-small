import time, math, numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v, sum_target=1.0):
    """Euclidean projection of v onto the simplex {x>=0, sum x = sum_target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if sum_target <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - sum_target
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast exact score (uses FFT, assumes a is a 1‑D np.ndarray)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    return float(2.0 * n * max_b / (s * s))

def _grad_softmax(a, temp, L):
    """Gradient of the soft‑max weighted max‑convolution."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    w = np.exp((conv - max_val) / (temp + 1e-20))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = (conv == max_val).astype(np.float64)
        w_sum = w.sum()
    w /= w_sum
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

def _symmetrize(a):
    """Force symmetry and re‑project onto the simplex."""
    return _project_simplex((a + a[::-1]) * 0.5)

            if best_sc < target:
                break

            if time.time() >= deadline:
                break

        restart += 1
        if best_sc < target:
            break

    # ----- final intensive hill‑climbing ---------------------------------
    cur = best_vec.copy()
    cur_sc = best_sc
    eps_factor = 0.05
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        eps = cur[i] * rng.random() * eps_factor
        if eps <= 0.0:
            continue
        tmp = cur.copy()
        tmp[i] -= eps
        tmp[j] += eps
        tmp = _project_simplex(tmp)
        tmp = _symmetrize(tmp)
        tmp_sc = _score(tmp)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
            # slowly tighten perturbations
            eps_factor = max(0.005, eps_factor * 0.9995)
        else:
            # occasionally accept a worsening move to escape plateaus
            if rng.random() < 0.001:
                cur, cur_sc = tmp, tmp_sc
            # decay eps_factor when stuck
            eps_factor = max(0.005, eps_factor * 0.9999)

    # ----------------------------------------------------------------------
    # Return the best found sequence as plain Python floats
    return [float(x) for x in best_vec]
