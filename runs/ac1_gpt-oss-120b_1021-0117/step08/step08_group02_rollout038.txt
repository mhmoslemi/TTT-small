def geometric_score(r):
    # vector lengths
    ks = np.arange(2*n-1, dtype=np.float64)
    # compute overlap length:
    overlap = np.where(ks < n, ks+1, 2*n-1-ks)
    # compute r**k
    r_powers = np.power(r, ks)
    max_val = np.max(overlap * r_powers)
    s = (1 - r**n) / (1 - r)
    score = 2.0 * n * max_val / (s * s)
    return float(score)

def golden_section_minimize(f, a, b, tol=1e-9, maxiter=60):
    phi = (1 + 5**0.5) / 2
    invphi = 1/phi
    invphi2 = invphi**2
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(maxiter):
        if abs(b - a) < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = f(d)
    # return best point
    if fc < fd:
        return c, fc
    else:
        return d, fd
```

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast evaluation of the AC1 bound for a vector with sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
