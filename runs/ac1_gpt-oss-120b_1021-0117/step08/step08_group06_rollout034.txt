import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a, n, L):
    """2*n*max(conv) for a vector whose sum is 1 (the denominator of the true score is 1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    cur_sc = _fast_score(cur, n, L)

    # optimisation state

    beta = 0.92
