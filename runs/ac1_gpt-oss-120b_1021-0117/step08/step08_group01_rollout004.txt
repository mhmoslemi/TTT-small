def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(vec, L):
    """Score for a vector with sum≈1 using FFT convolution."""
    a_fft = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * vec.size - 1]
    return float(2.0 * vec.size * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------ #
    # Determine size and initial vector
    if "height_sequence_1" in globals():
        init = np.maximum(np.asarray(height_sequence_1, dtype=np.float64), 0.0)
        n = init.shape[0]
    else:
        n = 2000
        init = rng.random(n, dtype=np.float64)
    cur = _project_simplex(init)
    L = 1 << ((2 * n - 1).bit_length())

    # initial best
    best_vec = cur.copy()
    best_sc = _score(best_vec, L)

    # compare with uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uni_sc = _score(uniform, L)
    if uni_sc < best_sc:
        best_sc, best_vec = uni_sc, uniform.copy()

    # ------------------------------------------------------------------ #
    # Hyper‑parameters
    step0 = 0.35          # initial step magnitude
    beta = 0.78           # momentum factor
    temp_target = 1.5e-4  # minimal temperature for mask tolerance
    max_no_improve = 2500
    target_score = 1.50274349

    momentum = np.zeros_like(cur)
    no_improve = 0
    it = 0

    # ------------------------------------------------------------------ #
    # Main optimisation loop
    while time.time() < deadline:
        it += 1

        # FFT of current vector
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # active set mask (subgradient)
        eps_mask = max_val * 1e-12 + temp_target
        mask = conv >= max_val - eps_mask
        w = mask.astype(np.float64)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(conv) / conv.size
        else:
            w /= w_sum

        # gradient of weighted max (subgradient when w uniform on active set)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # momentum / step update
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it)
        cand = _project_simplex(cur - step * momentum)
        cand_sc = _score(cand, L)

        # accept improvements
        if cand_sc < cur_sc if (cur_sc := _score(cur, L)) else float('inf'):
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1
            # occasional uphill move
            if rng.random() < 0.001:
                cur, cur_sc = cand, cand_sc

        # random redistribution when stuck
        if no_improve > max_no_improve:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.35
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
            no_improve = 0

        # periodic symmetrisation
        if it % 900 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # occasional restart from best with small noise
        if it % 4200 == 0:
            cur = best_vec.copy()
            momentum = np.zeros_like(cur)
            noise = rng.exponential(scale=0.015, size=n)
            cur = _project_simplex(cur + noise)
            cur_sc = _score(cur, L)

        if best_sc < target_score:
            break

    # ------------------------------------------------------------------ #
    # Final local refinement (tiny moves)
    cur = best_vec.copy()
    cur_sc = best_sc
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        eps = cur[i] * rng.random() * 0.012
        if eps <= 0.0:
            continue
        tmp = cur.copy()
        tmp[i] -= eps
        tmp[j] += eps
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp, L)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
        # occasional symmetry push
        if rng.random() < 0.04:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

    # Return best vector as plain Python floats
    return [float(x) for x in best_vec]
