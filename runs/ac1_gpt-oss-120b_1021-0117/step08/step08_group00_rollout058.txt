import time
import math
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

    # ---- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)               # enforce sum=1, non‑negative

    n = cur.shape[0]
    # power‑of‑two length for FFTs (>= 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    # ---- helpers --------------------------------------------------------
    def _conv_max(a):
        """Maximum entry of the linear convolution a*a."""
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        return float(conv.max())

    # initial evaluation
    cur_max = _conv_max(cur)
    cur_score = 2.0 * n * cur_max
    best_vec = cur.copy()
    best_score = cur_score

    # momentum vector for smoother updates
    momentum = np.zeros_like(cur)

    # hyper‑parameters
    beta = 0.90           # momentum decay
    step0 = 0.6           # initial step magnitude
    no_improve = 0
    target = 1.50274349
    it = 0

    # ---- optimisation loop ---------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- convolution and identification of active max indices -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        # all indices achieving the max (numeric tolerance)
        active = np.where(conv_full >= max_val - 1e-12)[0]
        # weight equally among active constraints
        w = np.zeros_like(conv_full)
        w[active] = 1.0 / max(1, active.size)

        # ----- sub‑gradient of the weighted max ----------------------
        # gradient = 2 * (a * w_rev) evaluated on the first n positions
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)          # mapping to the first n entries
        grad = 2.0 * grad_full[idx]

        # ----- momentum update ----------------------------------------
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- adaptive step ------------------------------------------
        step = step0 / math.sqrt(it)

        # ----- candidate generation ------------------------------------
        cand = cur - step * momentum
        cand = _project_simplex(cand)

        # ----- evaluate candidate ---------------------------------------
        cand_max = _conv_max(cand)
        cand_score = 2.0 * n * cand_max

        # ----- acceptance test -----------------------------------------
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best_vec, best_score = cand.copy(), cand_score
        else:
            no_improve += 1
            # occasional simulated‑annealing acceptance
            prob = math.exp(-(cand_score - cur_score) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score

        # ----- random mass redistribution when stuck -------------------
        if no_improve > 2500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_score = 2.0 * n * _conv_max(tmp)
            if tmp_score < cur_score:
                cur, cur_score = tmp, tmp_score
                no_improve = 0
                if tmp_score < best_score:
                    best_vec, best_score = tmp.copy(), tmp_score

        # ----- occasional symmetrisation (helps flatten peaks) -------
        if it % 700 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = 2.0 * n * _conv_max(sym)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                if sym_score < best_score:
                    best_vec, best_score = sym.copy(), sym_score

        # ----- early exit if target reached ----------------------------
        if best_score < target:
            break

    # Return the best sequence found as plain Python floats
    return [float(x) for x in best_vec]
