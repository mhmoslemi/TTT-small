# -*- coding: utf-8 -*-
import time
import math
import numpy as np

# -------------------- helper utilities --------------------

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Full (zero‑padded) autocorrelation a*a via FFT, length 2·len(a)‑1."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return conv


def _gradient(a: np.ndarray, conv: np.ndarray, max_val: float,
              temp: float, L: int) -> np.ndarray:
    """
    Gradient of the soft‑max of the autocorrelation.
    Returns a vector of length len(a).
    """
    if temp > 0:
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w = w / w_sum
    else:                                   # pure max – uniform over all maxima
        w = (conv == max_val).astype(np.float64)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(conv) / conv.size
        else:
            w = w / w_sum

    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    a_fft = np.fft.rfft(a, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)[: 2 * a.size - 1]
    idx = (2 * a.size - 2) - np.arange(a.size)
    return 2.0 * grad_full[idx]

    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())          # next power of two

    while time.time() < deadline:
        it += 1

        # recompute convolution and max (needed after every accepted move)
        conv = _conv_full(cur, L)
        max_val = conv.max()

        # temperature schedule
        if it == 1:
            temp = max_val * 0.15
        else:
            temp = max(temp * temp_decay, max_val * min_temp_factor)

        # gradient of the (soft) max‑autocorrelation
        grad = _gradient(cur, conv, max_val, temp, L)

        # descent direction (we minimise)
        direction = -grad

        # ----- simple back‑tracking line search ---------------------------
        alpha = init_alpha
        improved = False
        for _ in range(max_line_steps):
            cand = cur + alpha * direction
            cand = _project_simplex(cand)
            cand_conv = _conv_full(cand, L)
            cand_max = cand_conv.max()
            cand_score = 2.0 * n * cand_max
            if cand_score + 1e-12 < cur_score:
                improved = True
                break
            alpha *= alpha_decay

        if improved:
            cur = cand
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                if best_score <= target:
                    break
        else:
            no_improve += 1

        # ----- escape stagnation -----------------------------------------
        if no_improve >= 4000:
            # attempt a small random mass transfer
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_score = 2.0 * n * _conv_full(tmp, L).max()
            if tmp_score < cur_score:
                cur = tmp
                cur_score = tmp_score
                no_improve = 0
                if tmp_score < best_score:
                    best = tmp.copy()
                    best_score = tmp_score
                continue
            # otherwise re‑inject noise around the best known point
            cur = best + rng.normal(scale=0.008, size=n)
            cur = _project_simplex(np.maximum(cur, 0.0))
            cur_score = 2.0 * n * _conv_full(cur, L).max()
            no_improve = 0

        # ----- occasional symmetrisation -----------------------------------
        if it % 2000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = 2.0 * n * _conv_full(sym, L).max()
            if sym_score < cur_score:
                cur = sym
                cur_score = sym_score
                no_improve = 0
                if sym_score < best_score:
                    best = sym.copy()
                    best_score = sym_score

    # Return the best vector found (as plain Python floats)
    return [float(x) for x in best]
