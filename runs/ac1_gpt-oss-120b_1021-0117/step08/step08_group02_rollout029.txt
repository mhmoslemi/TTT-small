def _subgradient(a, tol=1e-12):
    n = len(a)
    L = 1 << ((2*n-1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*n-1]
    max_val = conv.max()
    # active indices where conv >= max_val - tol * max_val (or absolute)
    active = (conv >= max_val - max(1e-12, tol*max_val))
    w = active.astype(np.float64)
    # compute gradient similar to weighted case
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2*n-2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, max_val

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

        cur = rng.random(2000, dtype=np.float64)
        cur = _project_simplex(cur)
    else:
        cur = best_vec.copy()
    cur_sc = _score(cur)

    # ------------------------------------------------------------------ #
    # pre‑compute FFT size
    L = 1 << ((2 * n - 1).bit_length())

    # state for an Adam‑like update
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    lr = 0.25
    eps = 1e-8

    temp = None
    decay = 0.997
    min_temp_factor = 1e-7

    it = 0
    target = 1.50274349

    # ------------------------------------------------------------------ #
    # main gradient‑based optimisation loop
    while time.time() < deadline:
        it += 1

        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # temperature for a smooth‑max (gradually cools)
        if temp is None:
            temp = max_val * 0.2
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # soft‑max weights; fall back to hard max when temperature tiny
        diff = conv_full - max_val
        w = np.exp(diff / temp)
        w_sum = w.sum()
        if w_sum == 0.0 or temp < 1e-12:
            w = (conv_full == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of the weighted max via FFT (see derivation in comments)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur - step)
        cand_sc = _score(cand)

        # accept / reject with simple adaptive learning‑rate handling
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            lr = min(lr * 1.02, 2.0)
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (temp + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            lr = max(lr * 0.5, 1e-6)

        # occasional random mass redistribution to escape plateaux
        if it % 3000 == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()

        # occasional symmetrisation (often beneficial)
        if it % 1200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # early exit if we have already beaten the target
        if best_sc < target:
            break

    # ------------------------------------------------------------------ #
    # final local pairwise exchange phase (simple hill‑climbing)
    step_factor = 0.25
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = cur[i] * step_factor * rng.random()
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
        # slowly shrink the step factor
        if rng.random() < 0.01:
            step_factor = max(step_factor * 0.5, 1e-6)

    # ------------------------------------------------------------------ #
    # return the best vector found (as a plain Python list)
    return [float(x) for x in best_vec]
