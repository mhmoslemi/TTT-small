    else:
        n = 2000
        init_full = rng.random(n, dtype=np.float64)

    # Normalise to the simplex (sum = 1)
    cur = _project_simplex(np.maximum(init_full, 0.0))

    # ------------------------------------------------------------------ #
    # Helper: score for vectors with sum=1 (the objective of the evaluator)
    Lfft = 1 << ((2 * n - 1).bit_length())   # power‑of‑two >= 2n‑1
    def _score(a):
        af = np.fft.rfft(a, n=Lfft)
        conv = np.fft.irfft(af * af, n=Lfft)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    # ------------------------------------------------------------------ #
    # Keep the best found so far
    best_vec = cur.copy()
    best_sc = _score(best_vec)

    # Uniform baseline (often worse, but cheap to check)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    uni_sc = _score(uniform)
    if uni_sc < best_sc:
        best_sc, best_vec = uni_sc, uniform.copy()

    # ------------------------------------------------------------------ #
    # Adam optimiser parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.5
    it = 0

    # ------------------------------------------------------------------ #
    while time.time() < deadline:
        it += 1

        # convolution of the current vector
        af = np.fft.rfft(cur, n=Lfft)
        conv_full = np.fft.irfft(af * af, n=Lfft)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val

        # build sub‑gradient using the exact max set
        max_idx = np.where(conv == max_val)[0]
        w = np.zeros_like(conv)
        w[max_idx] = 1.0 / max_idx.size   # average over all maximal positions

        # gradient via FFT (same trick as the reference implementation)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=Lfft)
        grad_full = np.fft.irfft(af * w_fft, n=Lfft)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur - step)
        cand_sc = _score(cand)

        # keep the overall best
        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()

        # accept only improvements (no uphill moves)
        if cand_sc < cur_sc:
            cur = cand
        else:
            # occasional random perturbation to escape plateaus
            if rng.random() < 0.01:
                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.2
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                cur = _project_simplex(tmp)

        # periodic symmetrisation (helps because the optimum is symmetric)
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best_sc, best_vec = sym_sc, sym.copy()
            if sym_sc < cur_sc:
                cur = sym

        # early exit if we already beat the target
        if best_sc < 1.50274349:
            break

    # ------------------------------------------------------------------ #
    # Fine‑grained local refinement (tiny mass shifts)
    cur = best_vec.copy()
    cur_sc = best_sc
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        eps_move = cur[i] * rng.random() * 0.01
        if eps_move <= 0.0:
            continue
        tmp = cur.copy()
        tmp[i] -= eps_move
        tmp[j] += eps_move
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
        # occasional extra symmetrisation
        if rng.random() < 0.05:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

    # Return the best candidate as a plain Python list of floats
    return [float(x) for x in best_vec]
