import time, math, numpy as np

# ---------- helper utilities ----------
def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L):
    """Fast objective for a vector with sum(a)=1: 2*n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _grad_weighted(a, w, L):
    """Gradient of sum_k w_k * (a*a)_k w.r.t a (w length 2n-1)."""
    n = a.shape[0]
    w_rev = w[::-1]
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    full = np.fft.irfft(af * wf, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * full[idx]

def _softmax_weights(x, alpha):
    """Stable softmax of alpha * x."""
    mx = np.max(alpha * x)
    e = np.exp(alpha * x - mx)
    return e / e.sum()

# ---------- main candidate proposal ----------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # safety margin

    # ---- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    # initial score (fast)
    cur_score = _score(cur, L)
    best_vec = cur.copy()
    best_score = cur_score

    # ---- optimizer hyper‑parameters ------------------------------------
    lr = 0.05                     # base learning rate for Adam
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1_pow, beta2_pow = 1.0, 1.0

    alpha = 0.5                  # start smoothing factor
    max_alpha = 30.0
    alpha_inc_factor = 1.2

    it = 0
    no_improve = 0
    target = 1.50274349

    # ---- main loop -----------------------------------------------------
    while time.time() < deadline:
        it += 1

        # convolution of current vector with itself
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]

        # softmax weights (smooth approximation of max)
        w = _softmax_weights(conv, alpha)

        grad = _grad_weighted(cur, w, L)
