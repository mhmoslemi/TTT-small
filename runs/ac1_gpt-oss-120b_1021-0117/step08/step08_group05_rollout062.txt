    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_simplex(a, n):
    """Score for a vector a with sum(a)=1: 2*n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ----- initialization -----
    if "height_sequence_1" in globals():
        cur = np.array(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # enforce sum = 1

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())  # power‑of‑two FFT length
    idx = (2 * n - 2) - np.arange(n)      # index map for gradient extraction

    # ----- Adam parameters -----
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha = 0.12                         # base learning rate
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    best_vec = cur.copy()
    best_sc = _score_simplex(best_vec, n)
    cur_sc = best_sc

    T = None
    it = 0
    no_improve = 0
    target = 1.50274349

    # schedule parameters
    lp_interval = 2500          # how often to run the LP refinement
    lp_rel_eps = 1e-3           # relative tolerance for active constraints
    mix_t = 0.15                # mixing weight with LP solution
    sym_interval = 800          # how often to symmetrise
    random_interval = 3000     # when to perform a random mass move

    while time.time() < deadline:
        it += 1

        # ----- convolution and max -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()

        # ----- temperature schedule -----
        if T is None:
            T = max_conv * 0.20
        else:
            T = max(T * 0.9995, max_conv * 1e-5)

        # ----- soft‑max weights -----
        w = np.exp((conv - max_conv) / T)
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = (conv == max_conv).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient of smoothed max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score_simplex(cand, n)

        # ----- accept or reject -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (T + 1e-12)):
                cur, cur_sc = cand, cand_sc

        # ----- random redistribution when stuck -----
        if no_improve >= random_interval:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_simplex(tmp, n)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()

        # ----- periodic symmetrisation -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_simplex(sym, n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # ----- LP refinement step -----
        if it % lp_interval == 0:
            active = np.where(conv >= max_conv - lp_rel_eps * max_conv)[0]
            if active.size > 0:
                rows = []
                for s in active:
                    row = np.zeros(n, dtype=np.float64)
                    start = max(0, s - (n - 1))
                    end = min(s, n - 1)
                    row[start:end + 1] = cur[s - np.arange(start, end + 1)]
                    rows.append(row)
                A_ub = np.vstack(rows)
                b_ub = np.full(active.size, max_conv, dtype=np.float64)
                c = -np.ones(n, dtype=np.float64)          # maximise sum(g)
                bounds = [(0, None)] * n
                try:
                    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                                  bounds=bounds, method='highs',
                                  options={'presolve': True})
                except Exception:
                    res = None
                if res is not None and res.success and res.x is not None:
                    g = res.x
                    tot = g.sum()
                    if tot > 0:
                        g = g / tot
                        cur = (1 - mix_t) * cur + mix_t * g
                        cur = _project_simplex(cur)
                        cur_sc = _score_simplex(cur, n)
                        if cur_sc < best_sc:
                            best_sc, best_vec = cur_sc, cur.copy()

        # ----- early stop if target reached -----
        if best_sc < target:
            break

        # ----- final deadline safeguard -----
        if time.time() > deadline - 0.05:
            break

    return [float(x) for x in best_vec]
