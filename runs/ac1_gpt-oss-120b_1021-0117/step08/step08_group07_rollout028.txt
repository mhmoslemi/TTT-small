import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection onto the probability simplex (sum = 1, v >= 0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L=None):
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # initialise
    if "height_sequence_1" in globals():
        cur0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        cur0 = rng.random(2500, dtype=np.float64)
    cur0 = _project_simplex(np.maximum(cur0, 0.0))
    n = cur0.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur0.copy()
    best_sc = _score(best_vec, L)

    # ------------------------------------------------------------------
    # hyper‑parameters
    step0 = 0.9            # base step size
    beta = 0.95            # momentum decay
    max_iter_restart = 1800
    target = 1.50274349
    alpha_soft = 0.6       # blend factor for soft‑max gradient
    decay_temp = 0.999
    min_temp_fac = 1e-4

    # ------------------------------------------------------------------
    while time.time() < deadline:
        # start a new local search around the current best
        cur = best_vec.copy()
        cur += rng.normal(scale=0.02, size=n)          # jitter
        cur = np.clip(cur, 0.0, None)
        cur = _project_simplex(cur)

        momentum = np.zeros_like(cur)
        temp = None                                   # temperature for soft‑max
        cur_sc = _score(cur, L)
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()

        # -------- inner optimisation loop --------------------------------
        for it in range(max_iter_restart):
            if time.time() >= deadline:
                break

            # convolution of current iterate
            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
            max_val = conv_full.max()
            max_idx = int(np.argmax(conv_full))

            # -------- soft‑max gradient (smooth approximation) ----------
            if temp is None:
                temp = max_val * 0.07
            else:
                temp = max(temp * decay_temp, max_val * min_temp_fac)
            w = np.exp((conv_full - max_val) / temp)
            w_sum = w.sum()
            if w_sum == 0.0 or np.isnan(w_sum):
                w = (conv_full == max_val).astype(np.float64)
                w_sum = w.sum()
            w /= w_sum
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_soft_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx_rev = (2 * n - 2) - np.arange(n)
            grad_soft = 2.0 * grad_soft_full[idx_rev]

            # -------- exact sub‑gradient at the maximal lag -------------
            idx = max_idx - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad_exact = np.zeros_like(cur)
            grad_exact[mask] = 2.0 * cur[idx[mask]]

            # blend the two gradients
            grad = alpha_soft * grad_soft + (1.0 - alpha_soft) * grad_exact

            # -------- momentum & step ----------------------------------
            momentum = beta * momentum + (1.0 - beta) * grad
            step = step0 / math.sqrt(it + 1.0)

            # candidate update
            cand = cur - step * momentum
            cand = _project_simplex(cand)
            cand_sc = _score(cand, L)

            # acceptance
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                if cand_sc < best_sc:
                    best_sc, best_vec = cand_sc, cand.copy()
            else:
                # occasional Metropolis move
                prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
                if rng.random() < prob:
                    cur, cur_sc = cand, cand_sc

            # ---------- occasional symmetrisation ----------------------
            if (it + 1) % 300 == 0:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_sc = _score(sym, L)
                if sym_sc < cur_sc:
                    cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

            # ---------- directed mass transfer when stuck -------------
            if (it + 1) % 500 == 0:

                i_min = np.argmin(cur)
                if i_max != i_min:

                    tmp = cur.copy()
                    tmp[i_max] -= delta
                    tmp[i_min] += delta

                        cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()

    while time.time() < deadline:

        pert = np.clip(pert, 0.0, None)
        pert = _project_simplex(pert)

        if sc < best_sc:
            best_sc, best_vec = sc, pert.copy()
            if best_sc < target:
                break

    # Return as plain Python floats
    return [float(x) for x in best_vec]
