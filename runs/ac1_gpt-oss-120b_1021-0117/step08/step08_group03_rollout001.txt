def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np, cvxpy as cp
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety

    # Initialization
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1
    t = 0

    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:
        t += 1

        # compute conv and grad
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        grad = _grad_max(cur, conv, max_val, L)

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # line search scaling
        cand = None
        cand_sc = None
        scale = 1.0
        for _ in range(5):
            cand_try = _project_simplex(cur - scale * step)
            cand_try_sc = _score(cand_try, L)
            if cand_try_sc < cur_sc:
                cand, cand_sc = cand_try, cand_try_sc
                break
            scale *= 0.5
        if cand is not None:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc - 1e-12:
                best_vec, best_sc = cand.copy(), cand_sc
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional LP refinement
        if (t % 1000 == 0) or (no_improve > 200):
            lp_vec = _solve_lp(cur, max_val, t_mix=0.2)
            if lp_vec is not None:
                lp_sc = _score(lp_vec, L)
                if lp_sc < cur_sc:
                    cur, cur_sc = lp_vec, lp_sc
                    if lp_sc < best_sc:
                        best_vec, best_sc = lp_vec.copy(), lp_sc
                    no_improve = 0

        # random perturbation if stuck
        if no_improve > 800:
            # random transfer of mass between indices
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1

import time, math, numpy as np
import cvxpy as cp

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L=None):
    """Return 2*n*max(conv(a,a)) where conv is full convolution."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    max_val = conv_full[: 2 * n - 1].max()
    return float(2.0 * n * max_val)

def _grad_max(a, conv, max_val, L):
    """Subgradient of max(conv) w.r.t. a (binary mask of all maxima)."""
    mask = np.isclose(conv, max_val, atol=1e-12).astype(np.float64)
    w_rev = mask[::-1]
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    n = a.shape[0]
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

def _solve_lp(cur, bound, t_mix=0.2):
    """
    Solve   max sum(g)  s.t.  conv(cur,g) <= bound, g >= 0.
    Return a convex combination (1-t_mix)*cur + t_mix*g_norm on the simplex,
    or None on failure.
    """
    n = cur.shape[0]
    g = cp.Variable(n, nonneg=True)
    constraints = [cp.conv(cp.Constant(cur), g) <= bound]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000,
                   abstol=1e-8, reltol=1e-6, feastol=1e-8)
    except Exception:
        return None
    if g.value is None:
        return None
    g0 = np.maximum(g.value, 0.0)
    s = g0.sum()
    if s <= 0:
        return None
    g_norm = g0 / s
    return (1.0 - t_mix) * cur + t_mix * g_norm
