import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_norm1(a, L):
    """Exact score when sum(a)=1: 2*n*max(convolution(a,a))."""
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    return float(2.0 * a.shape[0] * conv_full[: 2 * a.shape[0] - 1].max())

        if best_local_score < cur_score:
            cur, cur_score = best_local, best_local_score
            no_improve = 0
            if cur_score < best_score:
                best, best_score = cur.copy(), cur_score
        else:
            # Metropolis acceptance with current temperature
            if rng.random() < math.exp(-(best_local_score - cur_score) / (temp + 1e-12)):
                cur, cur_score = best_local, best_local_score
                no_improve = 0
                if cur_score < best_score:
                    best, best_score = cur.copy(), cur_score
            else:
                no_improve += 1

        # ----- stagnation handling -----------------------------------------
        if no_improve > 3000:
            # inject noise around the best known solution
            cur = best + rng.normal(scale=0.02, size=n)
            cur = _project_simplex(np.maximum(cur, 0.0))
            cur_score = _score_norm1(cur, L)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        # ----- early exit --------------------------------------------------
        if best_score <= target:
            break

    return [float(x) for x in best]
