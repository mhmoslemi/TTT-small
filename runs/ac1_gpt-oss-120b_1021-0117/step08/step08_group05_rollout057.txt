def propose_candidate(seed=42,...):
    rng = np.random.default_rng(seed)
    start=time.time()
    deadline=start+budget-0.2

    n = len(height_sequence_1) if "height_sequence_1" in globals() else 2500

    L = 1 << ((2*n -1).bit_length())
    idx = (2*n -2) - np.arange(n)

    projection = _project_simplex (reuse from earlier)

    def score_vec(v):
        a = np.asarray(v, dtype=np.float64)
        # ensuring sum=1
        # but may be not exactly due to rounding; compute conv and max.
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af*af, n=L)[:2*n-1]
        max_conv = conv.max()
        return float(2.0*n*max_conv)

    best_vec = None
    best_sc = np.inf

    # set parameters
    base_lr = 0.05
    beta1=0.9
    beta2=0.999
    eps=1e-8
    max_iters_without_improve = 5000

    # number of restarts limited by time
    restart = 0

    while time.time() < deadline:
        # initialization for each restart
        if restart == 0 and "height_sequence_1" in globals():
            cur = np.copy(height_sequence_1)
        else:
            # random Dirichlet
            cur = rng.exponential(1.0, size=n)
        cur = projection(cur)  # sum=1

        m = np.zeros_like(cur)
        v = np.zeros_like(cur)
        T = None
        cur_sc = score_vec(cur)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()

        it = 0
        no_improve = 0

        while time.time() < deadline:
            it += 1

            # compute conv and gradient weights
            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[:2*n-1]
            max_conv = conv.max()

            if T is None:
                T = max_conv * 0.1
            else:
                T = max(T * 0.999, max_conv * 1e-5)

            diff = conv - max_conv

            w = np.exp(diff / T)

def _project_simplex(v):
    """Project a vector onto the probability simplex (sum=1, v>=0)."""
