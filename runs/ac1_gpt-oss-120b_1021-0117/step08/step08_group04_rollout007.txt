import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_norm1(a):
    """Score for sum(a)=1: 2*n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

            no_improve += 1

        # occasional symmetrization
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score_norm1(sym)
            if sym_score < best_score:
                best, best_score = sym.copy(), sym_score
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                no_improve = 0

        # random mass redistribution when stuck
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_score = _score_norm1(tmp)
            if tmp_score < best_score:
                best, best_score = tmp.copy(), tmp_score
            cur, cur_score = tmp, tmp_score
            no_improve = 0
            m.fill(0.0)
            v.fill(0.0)

        # full restart if really stuck
        if no_improve > 6000:
            cur = best + rng.normal(scale=0.01, size=n)
            cur = _project_simplex(np.maximum(cur, 0.0))
            cur_score = _score_norm1(cur)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        if best_score <= target:
            break

    # --- final local refinement ----------------------------------------
    while time.time() < deadline - 0.1:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = cur[i] * rng.random() * 0.02
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_score = _score_norm1(cand)
        if cand_score < best_score:
            best, best_score = cand.copy(), cand_score
            cur, cur_score = cand, cand_score
            if best_score <= target:
                break

    return [float(x) for x in best]
