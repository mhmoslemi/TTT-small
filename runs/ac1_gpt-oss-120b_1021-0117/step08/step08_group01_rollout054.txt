# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex (sum = 1, x >= 0)
def _project_simplex(v: np.ndarray) -> np.ndarray:
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

# Helper: fast evaluation of the AC1 bound for a vector whose sum ≈ 1
def _score(a: np.ndarray, L: int) -> float:
    a = np.asarray(a, dtype=np.float64)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(2.0 * a.size * conv.max() / (a.sum() ** 2))

                    cur, cur_sc = sym, sym_sc
                    if sym_sc < best_sc:
                        best_sc, best_vec = sym_sc, sym.copy()
        else:
            # rare uphill to escape plateaus
            if rng.random() < 0.001:
                cur, cur_sc = tmp, tmp_sc

    # Return the best vector found as plain Python floats
    return [float(x) for x in best_vec]
