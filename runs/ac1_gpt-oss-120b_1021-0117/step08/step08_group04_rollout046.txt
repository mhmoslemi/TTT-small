import time, math, numpy as np

def _project_simplex(v):
    """Project vector onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

        if it % 1000 == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.1
            cur[i] -= delta
            cur[j] += delta
            cur = _project_simplex(cur)
            a_fft = np.fft.rfft(cur, n=L)
            conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score

        # count non‑improving steps
        if cand_score >= cur_score:
            no_improve += 1

    # return the best sequence discovered
    return [float(x) for x in best]
