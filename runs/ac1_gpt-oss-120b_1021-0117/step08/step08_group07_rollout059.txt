import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    # sine window (positive)
    sin_vals = np.sin(np.pi * np.arange(1, n + 1) / (n + 1))
    starts.append(_project_simplex(sin_vals))

    # Hann window (zero at ends, positive inside)
    hann_vals = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    starts.append(_project_simplex(hann_vals))

    # random start
    starts.append(_project_simplex(rng.random(n)))

    best_vec = None
    best_sc = float('inf')
    for vec in starts:
        sc = _score(vec)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec.copy()

    cur = best_vec.copy()
    cur_sc = best_sc
    best_vec = cur.copy()
    best_sc = cur_sc

    # ------------------------------------------------------------------
    # optimisation state
    momentum = np.zeros_like(cur)
    beta = 0.95          # momentum decay
    step0 = 1.0          # base step size
    temp = None
    decay = 0.9995
    min_temp_fac = 1e-5
    no_improve = 0
    it = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while True:
        if time.time() >= deadline:
            break
        it += 1

        # ----- convolution & temperature ---------------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        if temp is None:
            temp = max_val * 0.1
        else:
            temp = max(temp * decay, max_val * min_temp_fac)

        # ----- smooth max (soft‑max) ------------------------------------
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient of the smooth max -------------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- momentum & step -----------------------------------------
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1.0)

        # ----- candidate ------------------------------------------------
        cand = cur - step * momentum
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()

        # ----- accept ---------------------------------------------------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # ----- directed mass transfer when stuck -----------------------
        if (it % 100 == 0) and (no_improve > 200):
            i_max = np.argmax(grad)
            i_min = np.argmin(grad)
            if i_max != i_min:
                delta = min(cur[i_max] * 0.2, 0.001)
                tmp = cur.copy()
                tmp[i_max] -= delta
                tmp[i_min] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()

        # ----- random pair move when very stuck ------------------------
        if no_improve > 800:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()

        # ----- occasional symmetrisation --------------------------------
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()

        # ----- heavy restart if hopelessly stuck -----------------------
        if no_improve > 3000:
            cur = best_vec + rng.normal(scale=0.01, size=n)
            cur = np.clip(cur, 0.0, None)
            cur = _project_simplex(cur)
            cur_sc = _score(cur)
            no_improve = 0

        # ----- early exit if target reached -----------------------------
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # final local‑search polish (small Gaussian perturbations)
    while time.time() < deadline:
        pert = best_vec + rng.normal(scale=0.003, size=n)
        pert = np.clip(pert, 0.0, None)
        pert = _project_simplex(pert)
        sc = _score(pert)
        if sc < best_sc:
            best_sc = sc
            best_vec = pert.copy()
            if best_sc < target:
                break

    # Return the best vector as plain Python floats
    return [float(x) for x in best_vec]
