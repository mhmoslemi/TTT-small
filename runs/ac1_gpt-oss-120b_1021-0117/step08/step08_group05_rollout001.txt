def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0,budget_s-0.2)

    # initialization
    if 'height_sequence_1' in globals():
        cur = np.array(height_sequence_1, dtype=np.float64).copy()
    else:
        # random vector length 2000 (choose moderately)
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur,0)
    cur = _project_simplex(cur) # sum=1

    n = cur.size
    L = 1 << ((2*n-1).bit_length())
    idx = (2*n-2) - np.arange(n)   # for gradient extraction

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.07   # maybe larger?
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    best_vec = cur.copy()
    best_score = _score_simplex(best_vec, n)
    cur_score = best_score

    T = None
    it = 0
    no_improve = 0
    target = 1.50274349

    # main loop
    while time.time() < deadline:
        it += 1
        # compute convolution via FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_conv = conv.max()

        # temperature for softmax
        if T is None:
            T = max_conv * 0.08
        else:
            T = max(T * 0.997, max_conv * 1e-4)

        w = np.exp((conv - max_conv) / T)
        w_sum = w.sum()
        if w_sum == 0:
            w = (conv == max_conv).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of smoothed max
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1-beta1**it)
        v_hat = v / (1-beta2**it)
        step = base_lr / math.sqrt(it) * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        cand = _project_simplex(cand)
        cand_score = _score_simplex(cand,n)

        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best_score, best_vec = cand_score, cand.copy()
        else:
            no_improve += 1
            # occasional uphill acceptance
            if rng.random() < math.exp(-(cand_score - cur_score) / (T + 1e-12)):
                cur, cur_score = cand, cand_score

        # occasional random redistribution (same as before)
        if no_improve > 2000:
            i = rng.integers(0,n)
            j = rng.integers(0,n-1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random()*0.3
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_score = _score_simplex(tmp,n)
            if tmp_score < cur_score:
                cur, cur_score = tmp, tmp_score
                no_improve = 0
                if tmp_score < best_score:
                    best_score, best_vec = tmp_score, tmp.copy()

        # symmetrization every 800 iterations
        if it % 800 == 0:
            sym = (cur + cur[::-1]) * 0.5

import time, math, numpy as np

def _project_simplex(v):
    """Project v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_simplex(a, n):
    """Fast score for a vector a with sum(a)=1: 2*n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
