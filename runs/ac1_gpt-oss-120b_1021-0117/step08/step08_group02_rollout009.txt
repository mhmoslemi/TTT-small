import time, numpy as np, math

def _project_simplex(v):
    # Euclidean projection onto simplex sum=1, v>=0
    # Implementation from original
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_fft(a, L=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2*n-1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    max_val = conv.max()
    return 2.0 * n * max_val / (np.sum(a)**2)   # but denominator sum==1 (simplex)

        if no_improve > 2500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_fft(tmp, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

        # --------------------------------------------------------------
        # occasional symmetrisation
        if it % 1000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fft(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # --------------------------------------------------------------
        # lightweight pairwise mass transfers
        if it % 500 == 0:
            for _ in range(4):
                a_idx = rng.integers(0, n)
                b_idx = rng.integers(0, n - 1)
                if b_idx >= a_idx:
                    b_idx += 1
                delta = cur[b_idx] * 0.1
                tmp = cur.copy()
                tmp[b_idx] -= delta
                tmp[a_idx] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_fft(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc

        # --------------------------------------------------------------
        # early exit if target reached
        if best_sc < target:
            break

    # Return best found vector as plain Python list
    return [float(x) for x in best_vec]
