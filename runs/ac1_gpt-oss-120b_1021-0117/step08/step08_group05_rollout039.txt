def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    else:

    idx = (2 * n - 2) - np.arange(n)
    # hyperparameters
    alpha0 = 0.08
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    # best overall
    best_vec = None
    best_score = float('inf')
    # define a function to evaluate with fast method
    def fast_score(a):
        # a is numpy array sum=1
        n_local = a.shape[0]
        L_local = 1 << ((2*n_local - 1).bit_length())
        af = np.fft.rfft(a, n=L_local)
        conv = np.fft.irfft(af*af, n=L_local)[:2*n_local-1]
        return float(2.0*n_local*conv.max())
    # but we will reuse L and n for speed, avoid recalculating bit length each time.
    # main loop
    restarts = 0
    while time.time() < deadline:
        # initialization for this restart
        # try random simplex vector
        # sample exponential
        y = rng.exponential(scale=1.0, size=n)
        cur = y / y.sum()
        cur_score = fast_score(cur)
        # track current best in this restart
        # Adam state
        m = np.zeros_like(cur)
        v = np.zeros_like(cur)
        # temperature schedule
        T = cur_score * 0.05 + 1e-12
        it = 0
        no_improve = 0
        while time.time() < deadline:
            it += 1
            # compute FFT of cur
            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[:2*n-1]
            max_conv = conv.max()
            # update temperature
            T = max(T * 0.9995, max_conv * 1e-4 + 1e-12)
            # compute softmax weights
            w = np.exp((conv - max_conv) / T)
            w_sum = w.sum()
            if w_sum == 0:
                w = (conv == max_conv).astype(np.float64)
                w_sum = w.sum()
            w /= w_sum
            # gradient of smoothed max
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            grad = 2.0 * grad_full[idx]
            # Adam update
            m = beta1 * m + (1 - beta1) * grad

            m_hat = m / (1 - beta1**it)

            cand = cur - step

            cand_score = fast_score(cand)
            # accept

                cur, cur_score = cand, cand_score
                no_improve = 0
                if cand_score < best_score:

            else:

import time
import math
import numpy as np

def _project_simplex(v):
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # leave a small safety margin
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # length of the problem
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2500

    # FFT length (next power of two) and index map for gradient extraction
    L = 1 << ((2 * n - 1).bit_length())
    idx = (2 * n - 2) - np.arange(n)          # maps full convolution back to length‑n gradient

    # Adam‑style hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    base_alpha = 0.07                         # base learning rate
    # temperature schedule parameters
    temp_factor = 0.06
    temp_decay = 0.9995

    # ------------------------------------------------------------------
    # helper to compute the objective 2·n·max(conv)  (sum of entries assumed = 1)
    def _score(vec):
        af = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    # ------------------------------------------------------------------
    # initialise a collection of candidates and keep the best one
    best_vec = None
    best_score = float("inf")

    # few deterministic starters
    starters = []

    # 1) the supplied parent, re‑normalised onto the simplex
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)
        starters.append(a)

    # 2) a simple triangular shape
    tri = np.arange(n, dtype=np.float64)
    tri = np.minimum(tri, n - 1 - tri)          # peak in the middle
    tri = _project_simplex(tri)
    starters.append(tri)

    # 3) a cosine‑shaped (non‑negative) vector
    x = np.linspace(0, math.pi, n, dtype=np.float64)
    cos = np.cos(x)
    cos = np.maximum(cos, 0.0)
    cos = _project_simplex(cos)
    starters.append(cos)

    # 4) a random simplex point
    rand_simplex = rng.exponential(scale=1.0, size=n)
    rand_simplex = _project_simplex(rand_simplex)
    starters.append(rand_simplex)

    # pick the best of the quick starters
    for cand in starters:
        sc = _score(cand)
        if sc < best_score:
            best_score, best_vec = sc, cand.copy()

    # ------------------------------------------------------------------
    # main multi‑restart optimisation loop
    while time.time() < deadline:
        # ----- initialise a fresh random simplex point -----
        cur = rng.exponential(scale=1.0, size=n)
        cur = _project_simplex(cur)
        cur_score = _score(cur)

        # Adam state
        m = np.zeros_like(cur)
        v = np.zeros_like(cur)

        # temperature (used for the smooth max)
        T = max(cur_score * temp_factor, 1e-12)

        it = 0
        stagnation = 0
        # per‑restart iteration budget – generous but still bounded by overall deadline
        while time.time() < deadline and it < 15000:
            it += 1

            # ----- convolution of the current vector -----
            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[: 2 * n - 1]
            max_conv = conv.max()

            # ----- update temperature (slow exponential cooling) -----
            T = max(T * temp_decay, max_conv * 1e-4 + 1e-12)

            # ----- soft‑max weights for a smooth approximation of the max -----
            w = np.exp((conv - max_conv) / T)
            w_sum = w.sum()
            if w_sum == 0.0:                     # fall back to a hard max if under‑flow occurs
                w = (conv == max_conv).astype(np.float64)
                w_sum = w.sum()
            w /= w_sum

            # ----- gradient of the smoothed max (see original description) -----
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            grad = 2.0 * grad_full[idx]

            # ----- Adam update -----
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            step = base_alpha / math.sqrt(it) * m / (np.sqrt(v) + epsilon)

            cand = cur - step
            cand = _project_simplex(cand)

            cand_score = _score(cand)

            # ----- accept/reject (basic simulated annealing) -----
            if cand_score < cur_score:
                cur, cur_score = cand, cand_score
                stagnation = 0
                if cand_score < best_score:
                    best_score, best_vec = cand_score, cand.copy()
            else:
                stagnation += 1
                if rng.random() < math.exp(-(cand_score - cur_score) / (T + 1e-12)):
                    cur, cur_score = cand, cand_score

            # ----- occasional symmetrisation (helps for many instances) -----
            if it % 1000 == 0:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_score = _score(sym)
                if sym_score < cur_score:
                    cur, cur_score = sym, sym_score
                    if sym_score < best_score:
                        best_score, best_vec = sym_score, sym.copy()

            # ----- restart if stagnation is prolonged -----
            if stagnation > 3000:
                break

        # early exit if we have already beaten the reference target
        if best_score < 1.50274349:
            break

    # ------------------------------------------------------------------
    # safety net: fall back to a uniform vector if nothing was found
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    # Return plain Python floats
    return [float(x) for x in best_vec]
