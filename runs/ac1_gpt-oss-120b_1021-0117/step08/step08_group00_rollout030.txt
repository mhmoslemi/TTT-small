def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)

        cand = _project_simplex(cur - step * momentum)
        cand_sc = _score_vec(cand, L, n)

        # ---- accept / occasional simulated‑annealing move --------------------
        if cand_sc < cur_score:
            cur, cur_score = cand, cand_sc
            stalls = 0
        else:
            prob = math.exp(-(cand_sc - cur_score) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_sc

        # ---- periodic symmetry enforcement ----------------------------------
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym, L, n)
            if sym_sc < cur_score:
                cur, cur_score = sym, sym_sc
                stalls = 0
            if sym_sc < best_sc:
                best_sc = sym_sc
                best_vec = sym.copy()

        # ---- occasional sparsification --------------------------------------
        if it % 1500 == 0:
            thresh = np.quantile(cur, 0.3)
            tmp = cur.copy()
            tmp[tmp < thresh] = 0.0
            tmp = _project_simplex(tmp)
            tmp_sc = _score_vec(tmp, L, n)
            if tmp_sc < cur_score:
                cur, cur_score = tmp, tmp_sc
                stalls = 0
            if tmp_sc < best_sc:
                best_sc = tmp_sc
                best_vec = tmp.copy()

        # ---- random redistribution when stuck --------------------------------
        if stalls > 5000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.35
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_vec(tmp, L, n)
            if tmp_sc < cur_score:
                cur, cur_score = tmp, tmp_sc
                stalls = 0
            if tmp_sc < best_sc:
                best_sc = tmp_sc
                best_vec = tmp.copy()

        # ---- deadline guard --------------------------------------------------
        if time.time() > deadline - 0.05:
            break

    # return the best sequence found
    return [float(x) for x in best_vec]
