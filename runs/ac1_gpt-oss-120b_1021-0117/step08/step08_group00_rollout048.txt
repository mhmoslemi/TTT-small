import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_vec(a, L, idx):
    """Fast AC1 upper bound for a vector already normalized to sum≈1."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2 * a.size - 1]
    return float(2.0 * a.size * conv.max())

        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)        # enforce sum = 1

    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())    # power‑of‑two FFT length
    idx = (2 * n - 2) - np.arange(n)       # indices to extract gradient

    # ---- Adam optimiser -------------------------------------------------
    lr = 0.08
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    t = 0

    best_vec = cur.copy()
    best_sc = _score_vec(best_vec, L, idx)
    cur_sc = best_sc

    target = 1.50274349
    no_improve = 0

    while time.time() < deadline:
        # compute convolution and max
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]
        max_val = conv.max()

        # weight vector: uniform over all maximal positions (subgradient)
        mask = (conv == max_val)
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
        else:
            w = mask.astype(np.float64) / float(cnt)

        # gradient via convolution with reversed weight vector
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur - step_vec)
        cand_sc = _score_vec(cand, L, idx)

        # acceptance
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # periodic symmetrisation (helps flatten peaks)
        if t % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym, L, idx)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # sparsification step (encourage zero entries)
        if t % 1500 == 0:
            thresh = np.quantile(cur, 0.1)   # prune bottom 10 %
            tmp = cur.copy()
            tmp[tmp < thresh] = 0.0
            tmp = _project_simplex(tmp)
            tmp_sc = _score_vec(tmp, L, idx)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

        # random redistribution when stuck
        if no_improve > 5000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_vec(tmp, L, idx)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

        if best_sc < target:
            break

    # Return as plain Python list
    return [float(x) for x in best_vec]
