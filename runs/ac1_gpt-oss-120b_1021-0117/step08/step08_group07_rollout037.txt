import time, math, numpy as np, scipy.sparse as sp
import cvxpy as cp

# ----------------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex (sum=1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast evaluation of 2·n·max(conv(a,a)) for a vector with sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
def _build_conv_matrix(f, active_idx):
    """Return a sparse (len(active_idx) × len(f)) matrix representing
    (f * g)[k] for k in active_idx, i.e. rows = active convolutions."""
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for r, k in enumerate(active_idx):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i_vals = np.arange(i_min, i_max + 1, dtype=np.int32)
        j_vals = k - i_vals
        rows.extend([r] * len(i_vals))
        cols.extend(j_vals.tolist())
        data.extend(f[i_vals].tolist())
    return sp.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))

def _solve_lp(f, M, active_idx, warm_g=None):
    """Solve max sum(g) s.t. (f*g)[k] ≤ M for k in active_idx, g ≥ 0."""
    n = f.shape[0]
    A = _build_conv_matrix(f, active_idx)
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M * np.ones(len(active_idx))]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    # a modest iteration limit keeps the budget safe
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000,
                   abstol=1e-9, reltol=1e-9)
    except Exception:
        return None
    if g.value is None:
        return None
    return np.asarray(g.value, dtype=np.float64)
