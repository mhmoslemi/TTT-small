def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # Init vector
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # create length 2000 random
        n0 = 2000
        a = rng.random(n0, dtype=np.float64)
    # Ensure nonnegative
    a = np.maximum(a, 0.0)
    # Project onto simplex (sum=1)
    a = _project_simplex(a)

    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Helper for score
    def _score_from_max(max_val):
        # sum assumed 1
        return 2.0 * n * max_val

    # compute initial score
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()
    best_a = a.copy()
    best_score = _score_from_max(max_val)
    cur_score = best_score

    # hyperparameters for temperature schedule
    temp = None
    decay_temp = 0.9996
    min_temp_factor = 1e-5

    # counters
    it = 0
    no_improve = 0
    target = 1.50274349

    # Precompute index mapping for gradient
    idx = (2 * n - 2) - np.arange(n)

    while time.time() < deadline:
        it += 1

        # recompute FFT only if needed
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()
        cur_score = _score_from_max(max_val)

        # temperature
        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay_temp, max_val * min_temp_factor)

        # softmax weights
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            # fallback to hard max
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient via FFT
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        cand = a - step

        cand = (cand + cand[::-1]) * 0.5
        # project onto simplex
        cand = _project_simplex(cand)

        cand_fft = np.fft.rfft(cand, n=L)

        cand_max = cand_conv.max()
        cand_score = _score_from_max(cand_max)

        if cand_score < best_score:
            best_a = cand.copy()
            best_score = cand_score
            no_improve = 0

            no_improve += 1

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0

        theta = 0.0
