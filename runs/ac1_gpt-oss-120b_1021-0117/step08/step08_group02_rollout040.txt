import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """2 * n * max(convolution) for a vector with sum≈1."""
    a = np.asarray(a, dtype=np.float64)

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0

        # --------------------------------------------------------------
        # random redistribution when stuck
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur = tmp
                cur_sc = tmp_sc
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                no_improve = 0

        # --------------------------------------------------------------
        # occasional symmetrisation (helps many instances)
        if it % 1200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()

        # --------------------------------------------------------------
        # early exit if target reached
        if best_sc < target:
            break

        # --------------------------------------------------------------
        # decay learning rate slightly
        lr = max(lr * 0.9999, 1e-4)

    # ------------------------------------------------------------------
    # return the best candidate found
    return [float(x) for x in best_vec]
```
