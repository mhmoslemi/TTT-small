def _grad_max(a):
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    max_idxs = np.flatnonzero(conv == max_val)
    # compute subgradient as average of contributions from each max_idx
    grad = np.zeros_like(a)
    for k in max_idxs:
        # shift index
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # we have i from lo to hi, j = k - i.
        # grad[i] += 2 * a[j]
        # This is vectorized:
        i_range = np.arange(lo, hi + 1)
        j_range = k - i_range
        grad[lo:hi+1] += 2.0 * a[j_range]
    grad /= len(max_idxs)
    return grad, max_val

        j_range = k - i_range

    grad /= len(max_idxs)
    score = 2.0 * n * max_val
    return grad, score

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast score 2·n·max(conv) for a simplex‑normalised vector."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _grad_max(a):
    """
    Sub‑gradient of max(conv) and its corresponding score.
    Returns (grad, score) where grad has the same shape as a.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    max_idxs = np.flatnonzero(conv == max_val)
    grad = np.zeros_like(a)
    for k in max_idxs:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        j = k - i
        grad[lo:hi + 1] += 2.0 * a[j]
    grad /= len(max_idxs)
    return grad, float(2.0 * n * max_val)
