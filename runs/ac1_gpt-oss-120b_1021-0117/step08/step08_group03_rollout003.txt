import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    max_val = conv.max()
    sum_a = a.sum()
    if sum_a == 0:
        return np.inf
    return float(2.0 * n * max_val / (sum_a*sum_a))

def _grad_max(a, conv, max_val, L):
    mask = np.isclose(conv, max_val, atol=1e-12)
    w = mask.astype(np.float64)
    w_rev = w[::-1]
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    n = a.shape[0]
    idx = (2*n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

    n = cur.shape[0]

    best_vec = cur.copy()

    beta = 0.96          # momentum decay

    while time.time() < deadline:
        it += 1

        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v_i>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L):
    """Return the AC1 upper bound for vector a (sum may be arbitrary)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    if s == 0:
        return np.inf
    return float(2.0 * n * max_val / (s * s))

def _grad_max(a, conv, max_val, L):
    """Subgradient of max(conv) w.r.t. a (using binary mask of maximal entries)."""
    mask = np.isclose(conv, max_val, atol=1e-12)
    w = mask.astype(np.float64)
    w_rev = w[::-1]
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    n = a.shape[0]
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

            step *= 0.5

        if cand is not None:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- rescue when stuck ------------------------------------
        if no_improve > 2000:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = cur[i] * rng.random() * 0.2
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                    no_improve = 0
                else:
                    # small smoothing attempt
                    ker = np.array([0.25, 0.5, 0.25], dtype=np.float64)
                    smooth = np.convolve(cur, ker, mode='same')
                    smooth = _project_simplex(smooth)
                    smooth_sc = _score(smooth, L)
                    if smooth_sc < cur_sc:
                        cur, cur_sc = smooth, smooth_sc
                        if smooth_sc < best_sc:
                            best_sc, best_vec = smooth_sc, smooth.copy()
                        no_improve = 0
            no_improve = 0   # reset regardless

        # ----- periodic symmetrisation -------------------------------
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # ----- early exit -------------------------------------------
        if best_sc < target:
            break

    # ----- return best found vector ---------------------------------
    return [float(x) for x in best_vec]
