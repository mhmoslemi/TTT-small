def _grad_softmax(a, conv, alpha, L):
    max_val = conv.max()
    w = np.exp(alpha * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.zeros_like(conv)
    else:
        w = w / w_sum
    w_rev = w[::-1]
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    idx = (2 * len(a) - 2) - np.arange(len(a))
    return 2.0 * grad_full[idx]
```

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


def _score(a, L=None):
    """Fast objective for a vector with sum≈1 (returns 2·n·max(conv))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _grad_softmax(a, conv, alpha, L):
    """Gradient of a soft‑max approximation to max(conv) w.r.t. a."""
    max_val = conv.max()
    w = np.exp(alpha * (conv - max_val))
    w /= w.sum()
    w_rev = w[::-1]
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    n = a.shape[0]
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]


def _solve_lp(cur, bound, t_mix=0.2):

    if cp is None:

    n = cur.shape[0]

    constraints = [cp.conv(cp.Constant(cur), g) <= bound]
