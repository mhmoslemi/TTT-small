def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, budget_s - 0.2)

    # init vector
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.shape[0]
    L = 1 << ((2*n - 1).bit_length())

    # evaluate best
    best_vec = cur.copy()
    best_sc = evaluate_sequence(best_vec.tolist())
    # compute surrogate
    cur_sc = _fast_score(cur, n, L)

    # optimizer state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    lr = 0.8   # base learning rate
    it=0
    # temperature schedule
    temp = None
    decay = 0.9995
    min_temp_factor = 5e-5

    # evaluation schedule
    exact_interval = 1000
    # stuck detection
    no_improve = 0
    stuck_limit = 4000

    while time.time() < deadline:
        it += 1

        # compute convolution via FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2*n - 1]
        max_val = conv.max()

        # temperature update
        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # softmax weights
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient computation
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1**it)
        v_hat = v / (1.0 - beta2**it)
        direction = m_hat / (np.sqrt(v_hat) + eps)

        # candidate steps with scaling factors
        best_cand = None
        best_cand_sc = float('inf')
        for factor in (0.5, 1.0, 1.5):
            cand = _project_simplex(cur - lr * factor * direction)
            cand_sc = _fast_score(cand, n, L)
            if cand_sc < best_cand_sc:
                best_cand_sc = cand_sc
                best_cand = cand

        # acceptance criteria
        accept = False
        if best_cand_sc < cur_sc:
            accept = True
        else:
            # Metropolis criterion with temperature scaled by iteration count
            prob = math.exp(-(best_cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = best_cand
            cur_sc = best_cand_sc
            no_improve = 0
            if it % exact_interval == 0:
                true_sc = evaluate_sequence(cur.tolist())
                if true_sc < best_sc:
                    best_sc = true_sc
                    best_vec = cur.copy()
        else:
            no_improve += 1

        # random redistribution if stuck
        if no_improve > stuck_limit:
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _fast_score(tmp, n, L)
            if tmp_sc < cur_sc:
                cur = tmp
                cur_sc = tmp_sc
                no_improve = 0
                true_sc = evaluate_sequence(tmp.tolist())
                if true_sc < best_sc:
                    best_sc = true_sc
                    best_vec = tmp.copy()

        # symmetrization occasionally
        if it % 600 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _fast_score(sym, n, L)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                true_sc = evaluate_sequence(sym.tolist())
                if true_sc < best_sc:
                    best_sc = true_sc
                    best_vec = sym.copy()
        # early stopping if good enough
        if best_sc < 1.45:
            break

    # final sanity check
    try:
        final_sc = evaluate_sequence(best_vec.tolist())
        if not (math.isfinite(final_sc) and final_sc < 1e9):
            best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    except Exception:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

    # analytic decreasing power shape

    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function vector with a low AC1 upper bound."""
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # Determine the problem size.
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000                     # fallback length if no parent is present

    best_vec = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Parent candidate (projected onto the simplex).
    if "height_sequence_1" in globals():
        try:
            cand1 = np.asarray(height_sequence_1, dtype=np.float64).copy()
            cand1 = np.maximum(cand1, 0.0)
            cand1 = _project_simplex(cand1)
            sc1 = evaluate_sequence(cand1.tolist())
            if math.isfinite(sc1):
                best_vec = cand1
                best_score = sc1
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2) Analytic decreasing‑power shape.
    #    a_i ∝ (n‑i)^p, projected to sum = 1.  p=5 gives a very low bound.
    p = 5
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    cand2 = idx ** p
    s = cand2.sum()
    if s > 0:
        cand2 = cand2 / s
    else:
        cand2 = np.full(n, 1.0 / n, dtype=np.float64)
    cand2 = np.clip(cand2, 0.0, 1000.0)

    try:
        sc2 = evaluate_sequence(cand2.tolist())
        if math.isfinite(sc2) and sc2 < best_score:
            best_vec = cand2
            best_score = sc2
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3) Fallback to uniform distribution if nothing else succeeded.
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Final sanity check before returning (ensures a finite score).
    try:
        final_sc = evaluate_sequence(best_vec.tolist())
        if not (math.isfinite(final_sc) and final_sc < 1e9):
            best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    except Exception:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(x) for x in best_vec]
