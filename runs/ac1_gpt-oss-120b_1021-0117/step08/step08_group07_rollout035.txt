def compute_score(epsilon, n):
    if epsilon <= 0 or epsilon >= 1:
        return np.inf
    log_r = np.log1p(-epsilon)  # log(1 - epsilon) stable
    # r_n = exp(n * log_r)
    r_n = np.exp(n * log_r)
    one_minus_r_n = 1 - r_n
    if one_minus_r_n <= 0:
        return np.inf
    c = epsilon / one_minus_r_n
    # r_n1 = exp((n-1) * log_r)
    r_n1 = np.exp((n-1) * log_r)
    max_conv = n * c * c * r_n1
    score = 2.0 * n * max_conv
    return score
```

We might have numeric issues for extremely small epsilon leading to one_minus_r_n ≈ n*epsilon (floating), but fine.

import time, math, numpy as np

# ------------------------------------------------------------
# Helper functions (simplex projection and fast score)
# ------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else css[rho[-1]] / (rho[-1] + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Exact AC1 bound (2·n·max(conv(a,a))) for a vector with sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
# Analytic bound for a geometric (exponential) step function
# ------------------------------------------------------------
def _geom_score(eps, n):
    """Score of a geometric sequence a_i = c·r^i  (r=1‑eps)."""
    if eps <= 0.0 or eps >= 1.0:
        return np.inf
    log_r = math.log1p(-eps)          # log(1‑eps) – stable for tiny eps
    r_n = math.exp(n * log_r)         # r**n
    one_minus_r_n = 1.0 - r_n
    if one_minus_r_n <= 0.0:
        return np.inf
    c = eps / one_minus_r_n
    r_n1 = math.exp((n - 1) * log_r)  # r**(n‑1)
    max_conv = n * c * c * r_n1
    return 2.0 * n * max_conv
