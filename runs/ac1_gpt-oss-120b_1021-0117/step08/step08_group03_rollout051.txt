import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L):
    """Score for a vector with sum≈1: 2·n·max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _grad_max(a, conv, max_val, L):
    """Subgradient of max(conv) w.r.t. a (binary mask of maximal entries)."""
    mask = np.isclose(conv, max_val, atol=1e-12).astype(np.float64)
    w_rev = mask[::-1]
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    n = a.shape[0]
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

        # convolution & max
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # gradient of the binary max
        grad = _grad_max(cur, conv, max_val, L)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size schedule
        base_step = step0 / math.sqrt(it)
        step = base_step

        # backtracking line search
        cand = None
        cand_sc = None
        for _ in range(7):
            cand_try = _project_simplex(cur - step * momentum)
            cand_try_sc = _score(cand_try, L)
            if cand_try_sc < best_sc:  # accept any improvement over global best
                cand, cand_sc = cand_try, cand_try_sc
                break
            if cand_try_sc < _score(cur, L):  # improvement over current
                cand, cand_sc = cand_try, cand_try_sc
                break
            step *= 0.5

        if cand is not None:
            cur = cand
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
            no_improve = 0
        else:
            no_improve += 1

        # ----- random rescue when stuck --------------------------------
        if no_improve > 400:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.2
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp, L)
            if tmp_sc < best_sc:
                cur = tmp
                best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0
            else:
                # small random reshuffle of a few coordinates
                idxs = rng.choice(n, size=min(12, n), replace=False)
                total = cur[idxs].sum()
                if total > 0:
                    new_vals = rng.random(len(idxs))
                    new_vals = new_vals / new_vals.sum() * total
                    tmp = cur.copy()
                    tmp[idxs] = new_vals
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score(tmp, L)
                    if tmp_sc < best_sc:
                        cur = tmp
                        best_vec, best_sc = tmp.copy(), tmp_sc
                        no_improve = 0
                else:
                    no_improve = 0  # reset to avoid deadlock

        # ----- occasional symmetrisation --------------------------------
        if it % 2000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < best_sc:
                cur = sym
                best_vec, best_sc = sym.copy(), sym_sc

        # early exit if target reached
        if best_sc < target:
            break

    return [float(x) for x in best_vec]
