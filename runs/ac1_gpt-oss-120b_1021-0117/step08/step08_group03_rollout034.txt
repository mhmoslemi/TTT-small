import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _grad_of_max(a, conv, max_val, L):
    """Subgradient of max(conv) w.r.t a (binary mask version)."""
    mask = np.isclose(conv, max_val, atol=1e-14)
    w = mask.astype(np.float64)          # length 2n-1
    w_rev = w[::-1]
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    n = a.shape[0]
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.1)   # safety margin

    # ---------- initial vector ----------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        if cur.sum() == 0:
            cur = np.full_like(cur, 1.0 / cur.size)
        else:
            cur = cur / cur.sum()
    else:
        n = 2000
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    # initial full convolution and max
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1].copy()
    max_val = conv.max()

    best_vec = cur.copy()
    best_max = max_val

    it = 0
    recalc_interval = 4000          # recompute conv from scratch periodically
    target = 1.50274349 / (2 * n)   # target max value (objective / (2n))

    while time.time() < deadline:
        it += 1

        # recompute convolution occasionally to avoid drift
        if it % recalc_interval == 0:
            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[: 2 * n - 1].copy()
            max_val = conv.max()

        # gradient of the (binary) max
        grad = _grad_of_max(cur, conv, max_val, L)
        i_src = int(np.argmax(grad))
        i_tgt = int(np.argmin(grad))
        if i_src == i_tgt:
            # fallback to a random move
            i_src = rng.integers(0, n)
            i_tgt = rng.integers(0, n - 1)
            if i_tgt >= i_src:
                i_tgt += 1

        # step size shrinks as time passes
        progress = (time.time() - start) / (deadline - start)
        step_frac = 0.3 * (1.0 - progress) + 0.01
        delta = cur[i_src] * step_frac
        if delta <= 1e-15:
            continue

        # store old values for rollback
        a_i = cur[i_src]
        a_j = cur[i_tgt]
        idx_2i = 2 * i_src
        idx_2j = 2 * i_tgt
        idx_ij = i_src + i_tgt
        old_2i = conv[idx_2i]
        old_2j = conv[idx_2j]
        old_ij = conv[idx_ij]

        # incremental update of convolution (exact)
        delta2 = delta * delta
        change_2i = -2.0 * delta * a_i + delta2
        change_2j =  2.0 * delta * a_j + delta2
        change_ij =  2.0 * delta * (a_i - a_j) - 2.0 * delta2
        conv[idx_2i] = old_2i + change_2i
        conv[idx_2j] = old_2j + change_2j
        conv[idx_ij] = old_ij + change_ij
        cur[i_src] = a_i - delta
        cur[i_tgt] = a_j + delta

        new_max = conv.max()

        # simulated‑annealing acceptance
        T0 = max_val * 0.1
        T = max(T0 * (1.0 - progress), 1e-12)
        accept = False
        if new_max <= max_val:
            accept = True
        else:
            # probability to accept a worsening move
            if rng.random() < math.exp(-(new_max - max_val) / T):
                accept = True

        if accept:
            max_val = new_max
            if new_max < best_max:
                best_max = new_max
                best_vec = cur.copy()
            # early exit if we have already beaten the target
            if best_max < target:
                break
        else:
            # rollback
            conv[idx_2i] = old_2i
            conv[idx_2j] = old_2j
            conv[idx_ij] = old_ij
            cur[i_src] = a_i
            cur[i_tgt] = a_j

    # final safeguard: ensure the returned vector is feasible
    best_vec = np.maximum(best_vec, 0.0)
    if best_vec.sum() == 0:
        best_vec = np.full_like(best_vec, 1.0 / best_vec.size)
    else:
        best_vec = best_vec / best_vec.sum()
    return [float(x) for x in best_vec]
