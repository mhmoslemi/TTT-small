import time, math, numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex (sum=1, non‑negative)
def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

# ----------------------------------------------------------------------
# Fast surrogate score: 2 * n * max(convolution)  (sum of a is 1)
def _fast_score(a, n, L):
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

            break

    # ------------------------------------------------------------------
    # final sanity check
    try:
        final_sc = evaluate_sequence(best_vec.tolist())
        if not math.isfinite(final_sc):
            best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    except Exception:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(x) for x in best_vec]
