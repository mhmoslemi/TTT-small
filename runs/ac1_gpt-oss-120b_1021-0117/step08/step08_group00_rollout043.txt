import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else css[rho[-1]] / (rho[-1] + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_vec(a, L):
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return float(2.0 * a.size * conv.max())

def _optimize(init_vec, deadline, rng):
    cur = init_vec.copy()
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())
    cur_sc = _score_vec(cur, L)
    best_vec, best_sc = cur.copy(), cur_sc

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step0 = 1.5
    temp = None
    decay_temp = 0.9995
    min_tmp = 1e-5
    no_improve = 0
    it = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        cur_sc = 2.0 * n * max_val

        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay_temp, max_val * min_tmp)

        w = np.exp((conv_full - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv_full == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = step0 / math.sqrt(it)
        direction = step * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur - direction)
        cand_sc = _score_vec(cand, L)
