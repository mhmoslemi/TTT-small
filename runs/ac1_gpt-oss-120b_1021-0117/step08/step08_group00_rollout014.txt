import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_vec(a, L):
    """Return the AC1 upper bound for a (assumes a sum≈1)."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_val / (sum_a * sum_a))

    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # ---- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # enforce sum=1

    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (power of two)

    cur_sc = _score_vec(cur, L)
    best_vec = cur.copy()
    best_sc = cur_sc
    momentum = np.zeros_like(cur)

    # ---- hyper‑parameters ------------------------------------------------
    beta = 0.9                # momentum decay
    step0 = 0.8               # initial step size factor
    target = 1.50274349
    it = 0
    no_improve = 0

    # ---- main loop -------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # convolution and max
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # sub‑gradient of max(conv) using binary mask of maximal entries
        mask = np.isclose(conv, max_val, atol=1e-12)
        w = mask.astype(np.float64)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)          # extract linear‑convolution part
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # candidate via projected gradient step
        step = step0 / math.sqrt(it)
        cand = _project_simplex(cur - step * momentum)
        cand_sc = _score_vec(cand, L)

        # acceptance (simulated‑annealing flavour)
        if cand_sc < cur_sc or rng.random() < math.exp(-(cand_sc - cur_sc) / (step + 1e-12)):
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1

        # update global best
        if cur_sc < best_sc:
            best_vec, best_sc = cur.copy(), cur_sc

        # ----- occasional perturbations ----------------------------------
        if no_improve > 2000:
            # small random redistribution between two coordinates
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_vec(tmp, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
            if tmp_sc < best_sc:
                best_vec, best_sc = tmp.copy(), tmp_sc

        if it % 600 == 0:
            # symmetrisation
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
            if sym_sc < best_sc:
                best_vec, best_sc = sym.copy(), sym_sc

        if it % 1200 == 0:
            # sparsify (drop bottom 20 %)
            thresh = np.quantile(cur, 0.2)
            tmp = cur.copy()
            tmp[tmp < thresh] = 0.0
            tmp = _project_simplex(tmp)
            tmp_sc = _score_vec(tmp, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
            if tmp_sc < best_sc:
                best_vec, best_sc = tmp.copy(), tmp_sc

        if it % 2500 == 0:
            # mix best with a fresh random vector
            alpha = 0.02
            rand_vec = rng.random(n, dtype=np.float64)
            rand_vec = _project_simplex(rand_vec)
            mixed = (1 - alpha) * best_vec + alpha * rand_vec
            mixed = _project_simplex(mixed)
            mixed_sc = _score_vec(mixed, L)
            if mixed_sc < best_sc:
                best_vec, best_sc = mixed.copy(), mixed_sc
                cur, cur_sc = mixed.copy(), mixed_sc

        if best_sc < target:
            break

    # return as plain Python list
    return [float(x) for x in best_vec]
