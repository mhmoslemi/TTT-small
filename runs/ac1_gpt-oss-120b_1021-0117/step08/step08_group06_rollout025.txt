import time, math, numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a, n, L):
    """Fast surrogate for the exact evaluator: 2*n*max(convolution)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---- initialise ----------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two >= 2n-1

    best_vec = cur.copy()
    best_sc = _fast_score(best_vec, n, L)

    # optimisation state
    momentum = np.zeros_like(cur)
    beta = 0.90          # momentum decay
    step0 = 0.70         # base additive step size
    mult_lr0 = 0.45      # base multiplicative learning rate
    temp = None
    temp_decay = 0.999
    min_temp_factor = 5e-5
    it = 0
    no_improve = 0
    exact_interval = 3000   # not used for heavy exact evals, kept as fallback

    while time.time() < deadline:
        it += 1

        # ---- convolution & temperature ---------------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = float(2.0 * n * max_val)

        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * temp_decay, max_val * min_temp_factor)

        # ---- soft‑max weighting ----------------------------------------
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ---- gradient of smooth max ------------------------------------
        w_rev = w[::-1]

        momentum = beta * momentum + (1.0 - beta) * grad
