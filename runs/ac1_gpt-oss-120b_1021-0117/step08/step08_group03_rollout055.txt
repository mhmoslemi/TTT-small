def _grad_weighted(a, L, w):
    w_rev = w[::-1]
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    n = a.shape[0]
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

def _solve_lp_cur(cur, bound):
    n = cur.shape[0]
    g = cp.Variable(n, nonneg=True)
    constraints = [cp.conv(cp.Constant(cur), g) <= bound]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=3000, abstol=1e-8, reltol=1e-6, feastol=1e-8)
    except Exception:
        return None
    if g.value is None:
        return None
    g0 = np.maximum(g.value, 0.0)
    s = g0.sum()
    if s <= 0:
        return None
    g_norm = g0 / s
    return g_norm

import time, math, numpy as np, cvxpy as cp

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _grad_weighted(a, L, w):
    """Gradient of weighted sum of autocorrelations (weights w, sum(w)=1)."""
    # w is length 2n-1
    w_rev = w[::-1]
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    n = a.shape[0]
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

def _solve_lp_cur(cur, bound):
    """Linear program: max sum(g) s.t. conv(cur,g) <= bound, g>=0.
    Returns the normalized solution (sum=1) or None on failure."""
    n = cur.shape[0]
    g = cp.Variable(n, nonneg=True)
    constraints = [cp.conv(cp.Constant(cur), g) <= bound]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=3000,
                   abstol=1e-8, reltol=1e-6, feastol=1e-8)
    except Exception:
        return None
    if g.value is None:
        return None
    g0 = np.maximum(g.value, 0.0)
    s = g0.sum()
    if s <= 0:
        return None
    return g0 / s

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # reserve a safety margin

    # ---- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: random initialization of length ~2500
        cur = rng.random(2500, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑2 for FFT

    # ---- bookkeeping ----------------------------------------------------
    best_vec = cur.copy()
    # score for sum=1 vectors: 2*n*max(conv)
    a_fft = np.fft.rfft(best_vec, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    best_sc = 2.0 * n * conv.max()

    momentum = np.zeros_like(cur)
    beta_mom = 0.9          # momentum decay
    step0 = 1.0
    it = 0
    no_imp = 0

    # parameters
    smooth_beta = 10.0          # temperature for soft‑max weighting
    max_no_imp_random = 800     # when to try random rescue
    max_no_imp_lp = 1200        # when to try LP rescue
    sym_interval = 1500         # symmetrisation frequency
    target = 1.50274349

    while time.time() < deadline:
        it += 1

        # ----- convolution & score ---------------------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

                    no_imp = 0

        # ----- occasional symmetrisation ---------------------------------
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            af = np.fft.rfft(sym, n=L)
            conv_sym = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
            sym_sc = 2.0 * n * conv_sym.max()
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_imp = 0

        # ----- early exit -------------------------------------------------
        if best_sc < target:
            break

    # Return best-known feasible vector as plain Python floats
    return [float(x) for x in best_vec]
