import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex {x>=0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(vec, n, L):
    """Return 2*n*max(conv(vec,vec)).  Uses FFT of length L (power of two)."""
    a_fft = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a tiny safety margin
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ---------- initial vector ----------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # sum = 1

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())  # power‑of‑two FFT length
    idx = (2 * n - 2) - np.arange(n)      # for gradient extraction

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    alpha = 0.12                         # base learning rate
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    best_vec = cur.copy()
    best_sc = _score(best_vec, n, L)
    cur_sc = best_sc

    T = None                               # temperature for soft‑max
    it = 0
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1

        # ----- convolution of current vector (FFT) -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()

        # ----- temperature schedule (aggressive cooling) -----
        if T is None:
            T = max_conv * 0.12
        else:
            T = max(T * 0.994, max_conv * 5e-5)   # never let it drop to zero

        # ----- soft‑max weights over convolution entries -----
        w = np.exp((conv - max_conv) / T)
        w_sum = w.sum()
        if w_sum == 0.0:                     # fallback to hard max
            w = (conv == max_conv).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient of the smoothed max ----- (matrix‑free)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = alpha * m_hat / (np.sqrt(v_hat) + epsilon)

        cand = cur - step
        cand = _project_simplex(cand)

        # enforce symmetry each iteration
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        cand_sc = _score(cand, n, L)

        # ----- accept if improvement, otherwise occasional uphill move -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (T + 1e-12)):
                cur, cur_sc = cand, cand_sc

        # ----- random redistribution when stuck -----
        if no_improve > 1500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp = (tmp + tmp[::-1]) * 0.5
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp, n, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()

        # ----- early termination if target reached -----
        if best_sc < target:
            break

    # Return the best vector found (as plain Python floats)
    return [float(x) for x in best_vec]
