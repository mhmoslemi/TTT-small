import time, math, numpy as np
import cvxpy as cp
from scipy.sparse import csr_matrix

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_norm1(a):
    """Fast score for a probability‑simplex vector: 2·n·max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))
    n = cur.size

    best = cur.copy()
    best_score = _score_norm1(best)
    target = 1.50274349

    it = 0
    # ---- main optimisation loop -----------------------------------------
    while time.time() < deadline:
        it += 1

        # convolution and current worst value
        conv = np.convolve(cur, cur)
        M = conv.max()

        # early exit if target already reached
        if best_score <= target:
            break

        # indices of the largest convolution entries (active constraints)
        K = min(120, conv.size)                       # number of constraints
        idx_top = np.argpartition(conv, -K)[-K:]      # K largest indices

        # build sparse matrix A where A[k,i] = cur[k-i] (zero if out of range)
        rows = []
        cols = []
        data = []
        for row_id, k in enumerate(idx_top):
            i_start = max(0, k - (n - 1))
            i_end = min(n, k + 1)
            cols_range = np.arange(i_start, i_end, dtype=int)
            rows.extend([row_id] * len(cols_range))
            cols.extend(cols_range.tolist())
            vals = cur[k - cols_range]               # cur[k-i]
            data.extend(vals.tolist())
        A = csr_matrix((data, (rows, cols)), shape=(len(idx_top), n), dtype=float)

        # linear program: maximise sum(g) subject to A g <= M, 0 <= g <= 1
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M, g <= 1.0]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500, verbose=False)
        except Exception:
            break
        if g.value is None:
            break
        g_opt = np.asarray(g.value, dtype=float).reshape(-1)
        sum_g = g_opt.sum()
        if sum_g == 0:
            break

        # rescale to keep the simplex normalisation (sum == 1)
        g_scaled = g_opt * (cur.sum() / sum_g)

        # convex combination with a modest step size
        t = 0.2
        cur = (1 - t) * cur + t * g_scaled
        cur = _project_simplex(cur)   # eliminate tiny numerical drift

        # evaluate new candidate
        cur_score = _score_norm1(cur)
        if cur_score < best_score:
            best = cur.copy()
            best_score = cur_score

        # occasional tiny random perturbation to avoid stagnation
        if it % 30 == 0:
            perturb = rng.normal(scale=0.003, size=n)
            cur = best + perturb
            cur = np.maximum(cur, 0.0)
            cur = _project_simplex(cur)

    return [float(x) for x in best]
