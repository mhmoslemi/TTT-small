import time, math, numpy as np

def _project_simplex(v):
    """Project v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max())

                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.25
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score_simplex(tmp, n, L, idx)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_score:
                        best_score, best_vec = tmp_sc, tmp.copy()
                elif rng.random() < 0.05:
                    cur, cur_sc = tmp, tmp_sc

            # ----- symmetrisation occasional -----
            if it % 800 == 0:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_sc = _score_simplex(sym, n, L, idx)
                if sym_sc < cur_sc:
                    cur, cur_sc = sym, sym_sc
                    if sym_sc < best_score:
                        best_score, best_vec = sym_sc, sym.copy()

            # ----- early exit if target reached -----
            if best_score < 1.50274349:
                break

    # Run optimizer starting from the given parent (if any)
    _optimize(init_arr)

    # If time permits, try a few random restarts
    while time.time() < deadline - 5 and best_score > 1.50274349:
        rnd = rng.random(n, dtype=np.float64)
        _optimize(rnd)

    # Fallback to uniform distribution if nothing was found
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    # Ensure compliance with evaluator limits
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    return [float(x) for x in best_vec]
