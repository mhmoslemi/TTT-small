import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_vec(a, L):
    """Score 2*n*max(conv) for a vector a (assumed sum(a)=1)."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a list of non‑negative step heights with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ----- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # enforce sum = 1

    n = cur.shape[0]
    # next power of two >= 2*n-1 for FFT efficiency
    L = 1 << ((2 * n - 2).bit_length())

    # ----- optimizer state ------------------------------------------------
    m = np.zeros_like(cur)   # first moment
    v = np.zeros_like(cur)   # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    step0 = 0.6
    best_vec = cur.copy()
    best_sc = _score_vec(best_vec, L)

    it = 0
    # ----- main loop ------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- convolution and soft‑max weights -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # temperature decays slowly; prevents vanishing weights
        temp = max_val * max(0.02, 0.7 ** (it / 1500))
        w = np.exp((conv_full - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:

            w_sum = w.sum()
        w /= w_sum
