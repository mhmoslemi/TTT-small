        cur = np.array(height_sequence_1, dtype=float)

        n = 2500
        cur = rng.random(n)
        cur = np.maximum(cur, 0.0)
    # project onto simplex sum=1
    cur = _project_simplex(cur)
    n = cur.shape[0]
    L = 1 << ((2*n-1).bit_length())
    # precompute index for gradient mapping
    idx = (2*n - 2) - np.arange(n)  # will be used for gradient extraction
    best = cur.copy()
    best_score = _score_norm1(best)
    # adaptation parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.1
    max_step = 1.0
    # temperature schedule
    temp = None
    temp_decay = 0.9993
    min_temp_factor = 5e-6  # maybe
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)
    it = 0
    while time.time() < deadline:
        it += 1
        # compute conv and max
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        # update temperature
        if temp is None:
            temp = max_val * 0.05
        else:
            # lower bound based on max_val * factor
            temp = max(temp * temp_decay, max_val * min_temp_factor)
        # compute weights (softmax)
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0:
            w = (conv == max_val).astype(float)
            w_sum = w.sum()
        w /= w_sum
        # gradient via convolution with reversed weights
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_rev_fft, n=L)
        # extract gradient for cur entries
        grad = 2.0 * grad_full[idx]
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1**it)
        v_hat = v / (1.0 - beta2**it)
        step = lr0 / math.sqrt(it) * m_hat / (np.sqrt(v_hat) + eps)
        # clip step length
        step_norm = np.linalg.norm(step)
        if step_norm > max_step:
            step = step * (max_step / step_norm)
        # update

        if cand_score < best_score:

        # deterministic descent: only accept if improves cur

    return [float(x) for x in best]

def _project_simplex(v):
    # from previous code
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w
```

import time, math, numpy as np

# ------------------------------------------------------------
# Helper functions (simple, self‑contained, no external data)
# ------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


def _score_norm1(a):
    """Score for a vector whose entries sum to 1:
    2 * n * max(convolution(a, a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # leave a short safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ----- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fall back to a random start if the parent is missing
        cur = rng.random(2500, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)
    # enforce symmetry (helps flatten the autocorrelation)
    cur = (cur + cur[::-1]) * 0.5
    cur = _project_simplex(cur)

    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    idx = (2 * n - 2) - np.arange(n)          # index map for gradient extraction

    # ----- optimisation state --------------------------------------------
    best = cur.copy()
    best_score = _score_norm1(best)           # = 2*n*max_conv, sum(best)=1
    cur_score = best_score

    # Adam‑style moments
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.08
    max_step = 1.0

    # temperature for the soft‑max weighting of the convolution
    temp = None
    temp_decay = 0.998          # aggressive cooling
    min_temp_factor = 1e-5

    no_improve = 0
    target = 1.50274349

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- convolution, max, and current score -----------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # ----- temperature update ----------------------------------------
        if temp is None:
            temp = max_val * 0.1
        else:
            temp = max(temp * temp_decay, max_val * min_temp_factor)

        # ----- soft‑max weights over the convolution -----------------------
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient of the weighted max (soft‑max) --------------------
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_rev_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (base_lr / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps)

        # clip the step length
        step_norm = np.linalg.norm(step)
        if step_norm > max_step:
            step = step * (max_step / step_norm)

        # ----- candidate generation ---------------------------------------
        cand = cur - step
        cand = np.maximum(cand, 0.0)

        # enforce symmetry and project back onto the simplex
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        cand_score = _score_norm1(cand)

        # ----- accept if improvement --------------------------------------
        if cand_score < cur_score:

            cur_score = cand_score
            no_improve = 0
        else:

            best_score = cand_score
            no_improve = 0

            cur_score = _score_norm1(cur)

            v.fill(0.0)
            no_improve = 0

    return [float(x) for x in best]
