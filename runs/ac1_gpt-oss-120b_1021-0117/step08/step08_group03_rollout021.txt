    n = len(a)
    L = 1 << ((3*n - 2).bit_length())
    A = np.fft.rfft(a, n=L)
    # compute conv a*a to get conv

def _solve_lp_sparse(cur, bound, t_mix=0.2, tol=0.01):
    n = cur.shape[0]
    L = 1 << ((2*n-1).bit_length())
    # compute conv and max
    af = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[:2*n-1]
    # mask
    mask = conv >= bound * (1 - tol)
    idx = np.where(mask)[0]
    m = len(idx)
    if m == 0:
        return None
    # Build constraint matrix: each row k corresponds to conv entry at offset idx[k]
    # For j from 0..n-1, coefficient = cur[ idx[k] - j ] if 0 <= idx[k] - j < n else 0
    # This is like shifted cur.
    # Build using sparse
    from scipy.sparse import lil_matrix
    A = lil_matrix((m, n), dtype=np.float64)
    for row, k in enumerate(idx):
        start = max(0, k - (n-1))
        end = min(n-1, k)
        # j runs from start..end: coefficient cur[k - j]
        # we can fill columns j = start..end with cur[k - j]
        # Use vectorized assignment
        cols = np.arange(start, end+1)
        vals = cur[k - cols]
        A[row, cols] = vals
    # Solve LP: maximize sum(g) => minimize -sum(g)
    from scipy.optimize import linprog
    c = -np.ones(n)
    # bounds (0, None)
    bounds = [(0, None)] * n
    # convert A to csr for linprog
    A_csr = A.tocsr()
    res = linprog(c, A_ub=A_csr, b_ub=np.full(m, bound), bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 5})
    if not res.success:
        return None
    g0 = np.maximum(res.x, 0.0)
    s = g0.sum()
    if s <= 0:
        return None
    g_norm = g0 / s
    new_vec = (1 - t_mix) * cur + t_mix * g_norm
    return new_vec

def _score(a, L=None):
    a = np.asarray(a, dtype=np.float64)

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

def _solve_lp_sparse(cur, bound, t_mix=0.2, tol=0.02):
    n = cur.shape[0]
    # compute full conv to get indices near bound
    L = 1 << ((2*n - 1).bit_length())
    af = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[:2*n-1]
    # select indices with conv >= bound * (1 - tol)
    idx = np.where(conv_full >= bound * (1 - tol))[0]
    if idx.size == 0:
        return None
    # Build sparse matrix A of shape (len(idx), n)
    from scipy.sparse import lil_matrix
    A = lil_matrix((idx.size, n), dtype=np.float64)
    for row, k in enumerate(idx):
        # valid j range:
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        # column indices j_min..j_max
        cols = np.arange(j_min, j_max + 1)
        # values: cur[k - cols]
        vals = cur[k - cols]
        A[row, cols] = vals
    # Solve LP: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    # bounds for variables >=0
    var_bounds = [(0, None)] * n
    from scipy.optimize import linprog
    try:
        res = linprog(c, A_ub=A.tocsr(), b_ub=np.full(idx.size, bound), bounds=var_bounds, method='highs')
    except Exception:
        return None
    if not res.success or res.x is None:
        return None
    g = np.maximum(res.x, 0.0)
    s = g.sum()
    if s <= 0:
        return None
    g_norm = g / s
    new_vec = (1.0 - t_mix) * cur + t_mix * g_norm
    return new_vec

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]

    w = v - theta
    w[w < 0] = 0.0
    return w


def _score(a, L=None):
    """Fast evaluation of the AC1 upper bound for a vector with sum≈1."""

    n = a.shape[0]

def _grad_smooth_max(a, epsilon):
    """
    Gradient of a smooth approximation of max(conv(a,a)).
    The smooth max is epsilon * logsumexp(conv/epsilon).
    """
    n = a.shape[0]
    Lg = 1 << ((3 * n - 2).bit_length())          # enough for a * a and a * w
    A = np.fft.rfft(a, n=Lg)

    # full auto‑convolution
    conv_full = np.fft.irfft(A * A, n=Lg)[: 2 * n - 1]

    # soft‑max weights on the convolution entries
    w = np.exp(conv_full / epsilon)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:

def _solve_lp_sparse(cur, bound, t_mix=0.2, tol=0.02):
    """
    Solve a small linear program:
        max   sum(g)
        s.t.  (cur * g)[k] <= bound  for all k with conv_k >= bound*(1-tol)
              g >= 0
    The constraints are built sparsely, using only the near‑binding indices.
    """
    n = cur.shape[0]
    # compute full auto‑convolution once
    Lc = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(cur, n=Lc)
    conv = np.fft.irfft(Af * Af, n=Lc)[: 2 * n - 1]

    idx = np.where(conv >= bound * (1 - tol))[0]
    if idx.size == 0:
        return None

    A = lil_matrix((idx.size, n), dtype=np.float64)
    for row, k in enumerate(idx):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        cols = np.arange(j_min, j_max + 1)
        vals = cur[k - cols]
        A[row, cols] = vals

    c = -np.ones(n)                     # maximize sum(g)  → minimize -sum(g)
    var_bounds = [(0, None)] * n
    try:

    s = g.sum()
    if s <= 0:
        return None
    g_norm = g / s

    return new_vec
