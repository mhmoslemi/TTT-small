import time, math, numpy as np

def _project_simplex(v):
    """Project v onto probability simplex (nonnegative, sum=1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_simplex(a, n):
    """Score for a vector with sum a_i = 1 (ignores denominator)."""
    a = np.asarray(a, dtype=np.float64)
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---- initialization ----
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    s = cur.sum()
    if s == 0.0:
        cur.fill(1.0 / cur.size)
    else:
        cur /= s  # scale to sum = 1

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    base_lr = 0.08  # base learning rate
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    best_vec = cur.copy()
    best_sc = _score_simplex(best_vec, n)
    cur_sc = best_sc

    it = 0
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1

        # ----- convolution & max -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- gradient of current max entry -----
        max_idxs = np.where(conv >= max_val - 1e-12)[0]
        k_max = rng.choice(max_idxs)

        g = np.zeros_like(cur)
        j_start = max(0, k_max - (n - 1))
        j_end = min(n, k_max + 1)  # exclusive
        if j_start < j_end:
            idxs = np.arange(j_start, j_end, dtype=np.int64)
            g[idxs] = 2.0 * cur[k_max - idxs]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (base_lr / math.sqrt(it + 1.0)) * m_hat / (np.sqrt(v_hat) + epsilon)

        cand = cur - step
        cand = _project_simplex(cand)

        cand_sc = _score_simplex(cand, n)

        # ----- accept / reject -----
        if cand_sc < cur_sc:

            no_improve = 0
            if cand_sc < best_sc:

                no_improve = 0
                if tmp_sc < best_sc:

        if it % 1000 == 0:

            sym = _project_simplex(sym)

            break

        # ----- time guard -----
        if time.time() > deadline - 0.05:
            break

    # Return best vector as plain Python floats
    return [float(x) for x in best_vec]
