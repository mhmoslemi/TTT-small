def _project_simplex(v):
    """Project v onto probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

            if tmp.sum() > 0:

                tmp_sc = _score_vec(tmp, L, n)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc

        # ----- early exit --------------------------------------------------
        if best_sc < target:
            break

    # fallback sanity check
    if not np.isfinite(best_sc):
        best_vec = np.full_like(cur, 1.0 / n)

    return [float(x) for x in best_vec]
