def _optimize_from(cur, deadline):
    n = cur.size
    L = 1 << ((2*n - 1).bit_length())
    idx = (2*n - 2) - np.arange(n)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    alpha = 0.08 # step factor
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    it = 0
    T = None
    cur_sc = _score_simplex(cur, n)
    best_vec = cur.copy()
    best_sc = cur_sc
    no_improve = 0
    while time.time() < deadline:
        it += 1
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2*n - 1]
        max_conv = conv.max()
        if T is None:
            T = max_conv * 0.07
        else:
            T = max(T * 0.9995, max_conv * 1e-5)
        w = np.exp((conv - max_conv) / T)
        w_sum = w.sum()
        if w_sum == 0:
            w = (conv == max_conv).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = (alpha / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + epsilon)
        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score_simplex(cand, n)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
        else:
            no_improve += 1
            # occasional uphill via SA
            if np.random.rand() < math.exp(-(cand_sc - cur_sc) / (T + 1e-12)):
                cur, cur_sc = cand, cand_sc
        # random redistribution if stuck
        if no_improve > 1500:
            i = np.random.randint(0, n)
            j = np.random.randint(0, n-1)
            if j >= i:
                j += 1
            delta = cur[i] * np.random.rand() * 0.3
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_simplex(tmp, n)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
        # symmetrize occasionally

                    best_sc, best_vec = sym_sc, sym.copy()

    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_simplex(a, n):
    """Fast score when sum(a)=1: 2*n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _optimize_from(start_vec, rng, deadline, target=None):
    """Adam‑based AC1 minimisation starting from start_vec."""
    cur = start_vec.copy()
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())
    idx = (2 * n - 2) - np.arange(n)

    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    alpha = 0.08                     # base learning rate
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    cur_sc = _score_simplex(cur, n)
    best_vec = cur.copy()
    best_sc = cur_sc
    T = None
    it = 0
    no_improve = 0

    while time.time() < deadline:
        it += 1

        # convolution and max
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()

        # temperature schedule
        if T is None:
            T = max_conv * 0.07
        else:
            T = max(T * 0.9995, max_conv * 1e-5)

        # soft‑max weights for smoothed max
        w = np.exp((conv - max_conv) / T)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_conv).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of smoothed max
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (alpha / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + epsilon)

        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score_simplex(cand, n)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1
            # occasional uphill move (simulated annealing)
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (T + 1e-12)):
                cur, cur_sc = cand, cand_sc

        # random redistribution when stuck
        if no_improve > 1500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_simplex(tmp, n)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()

        # symmetrisation periodically
        if it % 800 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_simplex(sym, n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # early exit if target reached
        if target is not None and best_sc < target:
            break

    return best_vec, best_sc

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # safety margin

    # ---------- initialization ----------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)                # enforce sum = 1
    n = cur.size

    best_vec = cur.copy()
    best_sc = _score_simplex(best_vec, n)

    target = 1.50274349

    # main multi‑restart loop
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        # allocate a slice of remaining time for a local run
        sub_deadline = time.time() + min(remaining * 0.5, 30.0)

        # perturb current best slightly to escape possible local minima
        pert = best_vec + rng.normal(scale=0.02, size=n)
        pert = np.clip(pert, 0.0, None)
        pert = _project_simplex(pert)

        vec, sc = _optimize_from(pert, rng, sub_deadline, target=target)
        if sc < best_sc:
            best_sc, best_vec = sc, vec.copy()
        if best_sc < target:
            break

    # return as plain Python list
    return [float(x) for x in best_vec]
