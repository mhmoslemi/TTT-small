def _grad_softmax(a, L, temp):
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2*n-1]
    max_c = conv.max()
    # compute softmax weights
    if temp <= 0: temp = 1e-12
    w = np.exp((conv - max_c)/temp)
    w_sum = w.sum()
    if w_sum == 0 or np.isnan(w_sum):
        w = (conv == max_c).astype(np.float64)
        w_sum = w.sum()
    w = w / w_sum
    # gradient
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2*n-2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, max_c

def _score(a):
    n = a.shape[0]; L = 1 << ((2*n-1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af*af, n=L)[:2*n-1]

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start=time.time()
    deadline=start+max(0.0,float(budget_s)-0.2)
    if "height_sequence_1" in globals():
        cur=np.asarray(height_sequence_1,dtype=np.float64).copy()
    else:
        cur=rng.random(2000,dtype=np.float64)
    cur=_project_simplex(np.maximum(cur,0.0))
    n=cur.size
    L=1 << ((2*n-1).bit_length())
    # initial scores
    cur_score=_score(cur)
    best_vec=cur.copy()
    best_score=cur_score
    # optimizer states
    m=np.zeros_like(cur)
    v=np.zeros_like(cur)
    beta1=0.9; beta2=0.999; eps=1e-8
    lr=0.1
    t=0
    no_improve=0
    temp=None
    target=1.50274349

    while time.time()<deadline:
        t+=1
        # compute gradient and max_c
        a_fft=np.fft.rfft(cur, n=L)
        conv_full=np.fft.irfft(a_fft*a_fft, n=L)
        conv=conv_full[:2*n-1]
        max_c=conv.max()
        # temperature schedule
        if temp is None:
            temp = max_c*0.07
        else:
            temp = max(temp*0.999, max_c*1e-4)
        # softmax weights
        w=np.exp((conv - max_c)/temp)
        w_sum=w.sum()
        if w_sum==0 or np.isnan(w_sum):
            w=(conv==max_c).astype(np.float64)
            w_sum=w.sum()
        w=w/w_sum
        # gradient
        w_rev=w[::-1]
        w_fft=np.fft.rfft(w_rev, n=L)
        grad_full=np.fft.irfft(a_fft*w_fft, n=L)
        idx=(2*n-2) - np.arange(n)
        grad=2.0*grad_full[idx]
        # Adam update
        m=beta1*m + (1-beta1)*grad
        v=beta2*v + (1-beta2)*(grad*grad)
        m_hat=m/(1-beta1**t)
        v_hat=v/(1-beta2**t)
        step=lr*m_hat/(np.sqrt(v_hat)+eps)
        # limit step size
        max_step=0.1
        step=np.clip(step,-max_step,max_step)
        cand=cur - step
        cand=_project_simplex(cand)
        cand_score=_score(cand)
        # acceptance
        if cand_score < cur_score:
            cur=cand; cur_score=cand_score; no_improve=0
        else:
            no_improve+=1
            prob=math.exp(-(cand_score-cur_score)/(temp*0.5+1e-12))
            if rng.random()<prob:
                cur=cand; cur_score=cand_score; no_improve=0
        # update best
        if cand_score < best_score:
            best_score=cand_score; best_vec=cand.copy()
        # early exit if target reached
        if best_score < target:
            break
        # periodic symmetrisation
        if t % 500 == 0:
            sym=(cur+cur[::-1])*0.5
            sym=_project_simplex(sym)
            sym_score=_score(sym)
            if sym_score < cur_score:
                cur=sym; cur_score=sym_score; no_improve=0
                if sym_score < best_score:
                    best_score=sym_score; best_vec=sym.copy()
        # mass transfer if stuck
        if no_improve>2000:
            # move small mass from max to min
            i_max=np.argmax(cur)
            i_min=np.argmin(cur)
            if i_max!=i_min:
                delta=min(cur[i_max]*0.2, 0.001)
                tmp=cur.copy()
                tmp[i_max]-=delta
                tmp[i_min]+=delta
                tmp=_project_simplex(tmp)
                tmp_score=_score(tmp)
                if tmp_score < cur_score:
                    cur=tmp; cur_score=tmp_score; no_improve=0
                    if tmp_score < best_score:
                        best_score=tmp_score; best_vec=tmp.copy()
        # random redistribution if very stuck
        if no_improve>5000:
            i=rng.integers(0,n)
            j=rng.integers(0,n-1)
            if j>=i:
                j+=1
            delta=cur[i]*rng.random()*0.4
            tmp=cur.copy()
            tmp[i]-=delta
            tmp[j]+=delta
            tmp=_project_simplex(tmp)
            tmp_score=_score(tmp)
            if tmp_score < cur_score:
                cur=tmp; cur_score=tmp_score; no_improve=0
                if tmp_score < best_score:
                    best_score=tmp_score; best_vec=tmp.copy()
        # update learning rate schedule
        lr *= 0.9999

    # final local search: gaussian perturbations
    while time.time() < deadline:
        pert=best_vec + rng.normal(scale=0.005, size=n)
        pert=np.clip(pert,0.0,None)
        pert=_project_simplex(pert)
        sc=_score(pert)
        if sc < best_score:
            best_score=sc
            best_vec=pert.copy()
            if best_score < target:
                break

    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())
    cur_score = _score(cur)
    best_vec = cur.copy()
    best_score = cur_score

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    beta1_pow, beta2_pow = 1.0, 1.0
    lr = 0.1
    temp = None
    no_improve = 0
    it = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1
        # gradient and max autocorrelation
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_c = conv.max()

        if temp is None:
            temp = max_c * 0.07
        else:
            temp = max(temp * 0.999, max_c * 1e-4)

        w = np.exp((conv - max_c) / temp)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = (conv == max_c).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        step = np.clip(step, -0.1, 0.1)

        cand = _project_simplex(cur - step)
        cand_score = _score(cand)

        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
        else:

            prob = math.exp(-(cand_score - cur_score) / (temp * 0.5 + 1e-12))
            if rng.random() < prob:

                no_improve = 0

        if cand_score < best_score:
            best_score, best_vec = cand_score, cand.copy()

        if best_score < target:

        # periodic symmetrisation
        if it % 500 == 0:

            sym_score = _score(sym)

                tmp = cur.copy()
                tmp[i_max] -= delta
                tmp[i_min] += delta

            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1

            tmp[i] -= delta

            tmp = _project_simplex(tmp)

    # final polishing with Gaussian noise

        pert = np.clip(pert, 0.0, None)
        pert = _project_simplex(pert)
        sc = _score(pert)
        if sc < best_score:
            best_score, best_vec = sc, pert.copy()
            if best_score < target:
                break

    return [float(x) for x in best_vec]
