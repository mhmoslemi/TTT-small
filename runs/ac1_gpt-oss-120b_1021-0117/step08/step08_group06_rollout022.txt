import time, math, numpy as np

def _project_simplex(v):
    """ Euclidean projection of a vector onto the probability simplex (sum=1, v>=0). """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a, n, L):
    """Fast surrogate score 2*n*max(conv) for vectors with sum≈1."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _conv_and_grad(a, n, L):
    """
    Return (conv, max_val, mask, grad) where
    conv = a*a (linear convolution),
    mask = indicator of entries attaining the maximum (within eps),
    grad = sub‑gradient of max(conv) w.r.t. a (using the mask).
    """
    # convolution a*a
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()
    # mask of max positions (tolerance avoids missing numerically)
    eps = 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    # gradient: 2 * (a_rev * mask) restricted to first n entries
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=L)
    mask_fft = np.fft.rfft(mask, n=L)
    grad_full = np.fft.irfft(a_rev_fft * mask_fft, n=L)
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # slice length n
    return conv, max_val, mask, grad

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # -------------------- initialise --------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback length if no parent is supplied
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    # FFT length sufficient for a*a (size 2n-1) and mask (size 2n-1)
    L = 1 << ((3 * n - 2).bit_length())

    # best‑so‑far (true score via evaluate_sequence)
    best_vec = cur.copy()
    best_sc = evaluate_sequence(best_vec.tolist())

    # fast surrogate score for the current iterate
    cur_sc = _fast_score(cur, n, L)

    # optimisation state
    momentum = np.zeros_like(cur)
    beta = 0.92               # momentum decay
    step0 = 0.8               # base step size
    it = 0
    no_improve = 0
    exact_interval = 500      # how often to call the exact evaluator
    target_score = 1.48       # ambitious early‑stop target

    while time.time() < deadline:
        it += 1

        # ----- convolution, mask and gradient -----
        conv, max_val, mask, grad = _conv_and_grad(cur, n, L)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # adaptive step (sqrt decay)
        step = step0 / math.sqrt(it)

        # try a few scalings of the step and keep the best candidate (fast score)
        cand_best = None
        cand_best_sc = float('inf')
        for factor in (0.5, 1.0, 1.5):
            cand = _project_simplex(cur - step * factor * momentum)
            cand_sc = _fast_score(cand, n, L)
            if cand_sc < cand_best_sc:
                cand_best_sc = cand_sc
                cand_best = cand

        # acceptance rule (always accept improvement, otherwise small probability)
        accept = False
        if cand_best_sc < cur_sc:
            accept = True
        else:
            prob = math.exp(-(cand_best_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob * 0.01:   # damped random walk
                accept = True

        if accept:
            cur, cur_sc = cand_best, cand_best_sc
            no_improve = 0
        else:
            no_improve += 1

        # ----- occasional exact evaluation -----
        if it % exact_interval == 0:
            true_sc = evaluate_sequence(cur.tolist())
            if true_sc < best_sc:
                best_sc, best_vec = true_sc, cur.copy()

        # ----- random perturbation when stuck -----
        if no_improve > 4000:
            # move a random fraction of mass from one index to another
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _fast_score(tmp, n, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                true_sc = evaluate_sequence(tmp.tolist())
                if true_sc < best_sc:

            else:

        # ----- symmetrisation (often beneficial) -----
        if it % 1000 == 0:

            sym = _project_simplex(sym)
            sym_sc = _fast_score(sym, n, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                true_sc = evaluate_sequence(sym.tolist())
                if true_sc < best_sc:
                    best_sc, best_vec = true_sc, sym.copy()

        # ----- early exit if we beat the target -----
        if best_sc < target_score:
            break

    # final safety: ensure the returned vector yields a finite score
    try:
        final_sc = evaluate_sequence(best_vec.tolist())
        if not math.isfinite(final_sc) or final_sc >= 1e9:
            raise RuntimeError
    except Exception:
        #fallback to a simple uniform simplex vector
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(x) for x in best_vec]
