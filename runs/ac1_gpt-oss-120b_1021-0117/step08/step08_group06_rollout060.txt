    """Euclidean projection of a vector onto the probability simplex (sum=1, v>=0)."""

    # main optimisation loops (one after another, respecting the deadline)
    for init in seeds:
        cur = _project_simplex(np.maximum(init, 0.0))
        cur_fast = _max_conv_fft(cur, L)          # surrogate: max convolution (sum=1)

        # Adam‑style state
        m = np.zeros_like(cur)
        v = np.zeros_like(cur)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        base_lr = 0.5

        it = 0
        no_improve = 0
        tol_rel = 1e-12

        while time.time() < deadline:
            it += 1

            # ----- convolution and active set -----
            af = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(af * af, n=L)
            conv = conv_full[: 2 * n - 1]
            max_val = conv.max()

            # active constraints (those attaining the maximum, up to numerical tolerance)
            if max_val == 0:
                active = np.ones_like(conv, dtype=bool)
            else:
                active = conv >= max_val - max(tol_rel, max_val * 1e-12)

            w = active.astype(np.float64)
            cnt = w.sum()
            if cnt == 0:
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
                cnt = 1
            w /= cnt                                 # uniform weights over active set

            # ----- gradient of the (soft) max -----
            grad = _grad_weighted(cur, w, L)

            # ----- Adam update -----
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = base_lr / (np.sqrt(v_hat) + eps)   # per‑coordinate step

            # ----- try a few scaling factors and keep the best -----
            best_local = None
            best_local_val = float("inf")
            for factor in (0.5, 1.0, 1.5):
                cand = _project_simplex(cur - step * factor * m_hat)
                cand_val = _max_conv_fft(cand, L)
                if cand_val < best_local_val:
                    best_local_val = cand_val
                    best_local = cand

            # ----- acceptance test (surrogate) -----
            accepted = False
            if best_local_val < cur_fast:
                accepted = True
            else:
                # tiny chance to escape flat regions
                if rng.random() < 1e-3:
                    accepted = True

            if accepted:
                cur, cur_fast = best_local, best_local_val
                no_improve = 0
                # occasional exact evaluation
                if it % 1000 == 0:
                    true_sc = evaluate_sequence(cur.tolist())
                    if true_sc < best_score:
                        best_score, best_vec = true_sc, cur.copy()
            else:
                no_improve += 1

            # ----- random redistribution when stuck -----
            if no_improve > 2000:
                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.5
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_val = _max_conv_fft(tmp, L)
                if tmp_val < cur_fast:
                    cur, cur_fast = tmp, tmp_val
                no_improve = 0

            # ----- enforce symmetry every few iterations -----
            if it % 400 == 0:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_val = _max_conv_fft(sym, L)
                if sym_val < cur_fast:
                    cur, cur_fast = sym, sym_val

            # ----- early exit if we already passed a very good target -----
            if best_score < 1.45:
                break

        # final exact check for this seed
        if time.time() < deadline:
            final_sc = evaluate_sequence(cur.tolist())
            if final_sc < best_score:
                best_score, best_vec = final_sc, cur.copy()

        # stop early if overall time is almost exhausted
        if time.time() >= deadline:
            break

    # -----------------------------------------------------------------
    # fallback: uniform simplex if nothing succeeded
    if best_vec is None or not np.isfinite(best_score):
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    # return as plain Python list of floats
    return [float(x) for x in best_vec]
