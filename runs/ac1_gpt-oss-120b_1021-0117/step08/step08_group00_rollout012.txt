import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w = w_raw / w_sum

        # gradient of the (soft) maximum term
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]          # sub‑gradient (scale irrelevant)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # candidate and projection back to simplex
        cand = _project_simplex(cur - step)
        cand_sc = _score_vec(cand, L, n)

        # accept if improvement
        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc

        # update global best
        if cand_sc < best_sc:
            best_vec, best_sc = cand.copy(), cand_sc
            no_improve = 0
        else:
            no_improve += 1

        # symmetrisation every few hundred steps
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym, L, n)
            if sym_sc < best_sc:
                best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc

        # sparsify (zero out the lower 20 % of entries) periodically
        if it % 2000 == 0:
            thresh = np.quantile(cur, 0.2)
            tmp = cur.copy()
            tmp[tmp < thresh] = 0.0
            tmp = _project_simplex(tmp)
            tmp_sc = _score_vec(tmp, L, n)
            if tmp_sc < best_sc:
                best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc

        # random redistribution when stuck
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_vec(tmp, L, n)
            if tmp_sc < best_sc:
                best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc

        # early exit if target achieved
        if best_sc < target:
            break

        # gradual learning‑rate decay
        lr *= lr_decay

    # ---------- final local refinement (time‑aware) ----------
    remaining = deadline - time.time()
    if remaining > 0.5:                     # give a tiny budget for fine‑tuning
        fine_iters = int(min(2000, remaining * 40))   # ~40 it/s
        for _ in range(fine_iters):
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.1
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_vec(tmp, L, n)
            if tmp_sc < best_sc:
                best_vec, best_sc = tmp.copy(), tmp_sc
                cur, cur_sc = tmp, tmp_sc

    # return best sequence as plain Python floats
    return [float(x) for x in best_vec]
