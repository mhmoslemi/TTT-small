import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        theta = css[rho[-1]] / (rho[-1] + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a, n, L):
    """Fast surrogate for 2*n*max(conv) when sum(a)=1."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT size (power of two)

    best_vec = cur.copy()
    best_sc = evaluate_sequence(best_vec.tolist())
    cur_sc = _fast_score(cur, n, L)

    # ---- optimisation state -----------------------------------------
    momentum = np.zeros_like(cur)
    beta = 0.85                # momentum decay
    step0 = 0.6                # base step size
    temp_decay = 0.99
    min_temp_factor = 1e-4
    temp = None
    it = 0
    no_improve = 0
    exact_interval = 1500
    sym_interval = 400
    random_restart_thresh = 8000

    # ---- main loop --------------------------------------------------
    while time.time() < deadline:
        it += 1

        # convolution and temperature
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        if temp is None:
            temp = max_val * 0.5
        else:
            temp = max(temp * temp_decay, max_val * min_temp_factor)

        # soft‑max weights
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of weighted max
        w_rev = w[::-1]

        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # momentum & step size
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1)

        best_cand = None
        best_cand_sc = float('inf')
