import time, math, numpy as np

# -------------------------------------------------------------
# Helper functions (all at top level, no closures)
# -------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L=None):
    """Fast evaluation of the AC1 upper bound for a vector a (assumed summed to 1)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    max_val = conv_full[: 2 * n - 1].max()
    return float(2.0 * n * max_val)

def _grad_max(a, conv, max_val, L):
    """Sub‑gradient of max(conv) w.r.t. a."""
    mask = np.isclose(conv, max_val, atol=1e-12)
    w = mask.astype(np.float64)
    w_rev = w[::-1]
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    n = a.shape[0]
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # small safety margin

    # ----- problem size -------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: random vector of length ~2000
        base = rng.random(2000, dtype=np.float64)

    n = base.size
    L = 1 << ((2 * n - 1).bit_length())

    # ----- initialise current vector on the simplex --------------------
    cur = _project_simplex(np.maximum(base, 0.0))
    cur_sc = _score(cur, L)

    best_vec = cur.copy()
    best_sc = cur_sc

    # ----- optimiser hyper‑parameters ----------------------------------
    momentum = np.zeros_like(cur)
    beta = 0.85          # momentum decay
    step0 = 1.8          # initial step magnitude
    it = 0
    no_improve = 0
    target = 1.50274349

    # ----- main optimisation loop --------------------------------------
    while time.time() < deadline:
        it += 1

        # compute convolution and its maximal entry
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # gradient of the (binary) max
        grad = _grad_max(cur, conv, max_val, L)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step‑size schedule
        step = step0 / math.sqrt(it)

        # back‑tracking line‑search (up to 5 reductions)
        cand = None
        cand_sc = None
        for _ in range(5):
            trial = _project_simplex(cur - step * momentum)
            trial_sc = _score(trial, L)
            if trial_sc < cur_sc:
                cand, cand_sc = trial, trial_sc
                break
            step *= 0.5

        if cand is not None:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
            no_improve = 0
        else:
            no_improve += 1

        # ----- rescue when stuck ---------------------------------------
        if no_improve > 300:
            # random mass transfer between two coordinates
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0
            else:

                pert = cur[idx] * rng.random(5) * 0.2
                tmp = cur.copy()
                tmp[idx] -= pert
                tmp[idx] += rng.random(5) * 0.2 * tmp[idx].mean()
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc
                    no_improve = 0

            sym = _project_simplex(sym)

            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # ----- early exit ------------------------------------------------
        if best_sc < target:
            break

    # Return the best vector found as plain Python floats
    return [float(x) for x in best_vec]
