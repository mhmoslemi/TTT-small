import time, math, numpy as np
from numpy.fft import rfft, irfft

# ------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x >= 0, sum(x)=1}
def _project_simplex(v):
    """Project v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    idx = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if idx.size == 0:
        theta = 0.0
    else:
        rho = idx[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

# ------------------------------------------------------------
# Fast surrogate score: 2*n*max(conv) assuming sum(a)=1
def _fast_score(a, n, L):
    af = rfft(a, n=L)
    conv = irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)  # safety margin

    # ---------- initialise ----------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two for FFT

    # best-so-far (true score via evaluate_sequence)
    best_vec = cur.copy()
    best_sc = evaluate_sequence(best_vec.tolist())

    # surrogate score for current iterate
    cur_sc = _fast_score(cur, n, L)

    # Adam optimizer state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.25          # base learning rate (will be scaled)

    # temperature for soft‑max smoothing
    temp = None
    decay = 0.997
    min_temp_factor = 1e-4

    it = 0
    no_improve = 0
    exact_eval_interval = 2000   # exact evaluation every this many iters

    # ------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- convolution & temperature -----
        a_fft = rfft(cur, n=L)
        conv_full = irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        if temp is None:
            temp = max_val * 0.08
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # ----- smooth weighting of the maximum -----
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient of the smoothed max -----
        w_rev = w[::-1]
        w_fft = rfft(w_rev, n=L)
        grad_full = irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]          # shape (n,)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step_vec = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- try a few step scalings, keep best surrogate -----
        best_cand = None
        best_cand_sc = float('inf')
        for factor in (0.5, 1.0, 1.5):
            cand = _project_simplex(cur - factor * step_vec)
            cand_sc = _fast_score(cand, n, L)
            if cand_sc < best_cand_sc:
                best_cand_sc = cand_sc
                best_cand = cand

        # ----- acceptance -----
        if best_cand_sc < cur_sc:
            cur, cur_sc = best_cand, best_cand_sc
            no_improve = 0
        else:
            # occasional stochastic acceptance
            prob = math.exp(-(best_cand_sc - cur_sc) / (base_lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = best_cand, best_cand_sc
                no_improve = 0
            else:
                no_improve += 1

        # ----- occasional exact evaluation -----
        if it % exact_eval_interval == 0:
            true_sc = evaluate_sequence(cur.tolist())
            if true_sc < best_sc:
                best_sc, best_vec = true_sc, cur.copy()

        # ----- random redistribution when stuck -----
        if no_improve > 4000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _fast_score(tmp, n, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                true_sc = evaluate_sequence(tmp.tolist())
                if true_sc < best_sc:
                    best_sc, best_vec = true_sc, tmp.copy()

        # ----- symmetrisation every few hundred iters -----
        if it % 600 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _fast_score(sym, n, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                true_sc = evaluate_sequence(sym.tolist())
                if true_sc < best_sc:
                    best_sc, best_vec = true_sc, sym.copy()

        # ----- early termination if time is almost up -----
        if time.time() > deadline - 0.5:
            break

    # ------------------------------------------------------------
    # final sanity check – ensure the returned vector yields a finite score
    try:
        final_sc = evaluate_sequence(best_vec.tolist())
        if not math.isfinite(final_sc) or final_sc >= 1e9:
            best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    except Exception:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(x) for x in best_vec]
