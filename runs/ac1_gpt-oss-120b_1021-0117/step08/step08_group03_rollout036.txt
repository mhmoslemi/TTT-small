def _grad_smooth(a, conv, L, alpha):
    # compute weight vector w of length 2n-1
    max_val = conv.max()
    # subtract max to avoid overflow
    w = np.exp(alpha * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(conv) / conv.size
    else:
        w /= w_sum

    # compute gradient using convolution trick
    w_rev = w[::-1]
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    n = a.shape[0]
    idx = (2*n - 2) - np.arange(n)  # same as before
    grad = 2.0 * grad_full[idx]
    return grad
```

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L=None):
    """Score = 2 * n * max(conv(a,a)).  a is assumed non‑negative and summed to 1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

            cur, cur_score = cand, cand_sc
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
            no_improve = 0
        else:
            no_improve += 1

        # ----- stochastic rescue when stuck -------------------------
        if no_improve > 1200:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = cur[i] * rng.random() * 0.3
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_score:
                    cur, cur_score = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc
                    no_improve = 0

        # ----- occasional symmetrisation ----------------------------
        if it % 2000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_score:
                cur, cur_score = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # early exit if target reached
        if best_sc < target:
            break

    return [float(x) for x in best_vec]
