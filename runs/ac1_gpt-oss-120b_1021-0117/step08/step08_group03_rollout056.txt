def propose_candidate(seed=42,budget_s=1100,**kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0,float(budget_s)-0.5)
    # init
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        # If any element >1000, clip
        cur = np.clip(cur, 0, 1000)
    else:
        cur = rng.random(2500)
    # project onto simplex sum=1
    cur = _project_simplex(np.maximum(cur,0))
    n = cur.shape[0]
    L = 1 << ((2*n-1).bit_length())
    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc
    # parameters for Adam
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step_alpha = 0.5  # base step size

    it = 0
    while time.time() < deadline:
        it += 1
        # compute conv and max
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        # compute subgradient
        grad = _grad_max(cur, conv, max_val, L)
        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = step_alpha * m_hat / (np.sqrt(v_hat) + eps)
        cand = _project_simplex(cur - step)
        cand_sc = _score(cand, L)
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        # occasional random perturbation
        if it % 200 == 0:
            i = rng.integers(0,n)
            j = rng.integers(0,n)
            if i != j:
                delta = cur[i] * rng.random() * 0.2
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc
        # early exit
        if best_sc < 1.50274349:
            break
