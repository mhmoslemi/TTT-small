import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# The simplex is the set {x >= 0, sum(x)=1}.
def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

# ------------------------------------------------------------
# Fast evaluation for vectors whose sum is (or has been projected to) 1.
def _score_simplex(a, L):
    """Return 2*n*max(conv(a,a)) for a already on the simplex."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    start = time.time()
    # keep a small safety margin for finalisation
    deadline = start + max(0.0, float(budget_s) - 0.2)

    best_vec = cur.copy()
