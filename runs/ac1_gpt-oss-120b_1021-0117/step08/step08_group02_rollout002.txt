import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Exact evaluator needed for final comparison (uses dense convolution)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    b = np.convolve(a, a)
    max_b = np.max(b)
    sum_a = np.sum(a)
    return float(2.0 * n * max_b / (sum_a * sum_a))

        # ---- soft‑max weights (sharpened by top‑k) ----
        w = np.exp((conv - max_val) / temp)
        # keep only a small fraction of largest weights to sharpen the gradient
        top_k = max(1, int(0.02 * w.size))  # about 2 %
        if top_k < w.size:
            idx = np.argpartition(w, -top_k)[-top_k:]
            mask = np.zeros_like(w, dtype=bool)
            mask[idx] = True
            w = np.where(mask, w, 0.0)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ---- gradient of the smooth max ----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur - step)
        cand_sc = _score(cand)

        # ---- acceptance ----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            stagnation = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            stagnation += 1
            prob = math.exp(-(cand_sc - cur_sc) / (temp + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
                stagnation = 0

        # ---- occasional symmetrisation ----
        if it % 1500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best_sc, best_vec = sym_sc, sym.copy()
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                stagnation = 0

        # ---- local random redistribution when stuck ----
        if stagnation > stagnation_limit:
            # perform a short local search
            local_best = cur.copy()
            local_best_sc = cur_sc
            for _ in range(120):
                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = local_best.copy()
                delta = tmp[i] * rng.random() * 0.4
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                sc = _score(tmp)
                if sc < local_best_sc:
                    local_best, local_best_sc = tmp, sc
            cur, cur_sc = local_best, local_best_sc
            stagnation = 0
            if local_best_sc < best_sc:
                best_sc, best_vec = local_best_sc, local_best.copy()

        # ---- early stop if target reached ----
        if best_sc < target:
            break

    return [float(x) for x in best_vec]
