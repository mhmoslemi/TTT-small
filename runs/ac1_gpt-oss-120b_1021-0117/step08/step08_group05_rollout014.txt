def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time, math, numpy as np

def _project_simplex(v):
    """Project vector v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_simplex(a, n):
    """Fast objective for a vector a with sum(a)=1: 2*n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)            # enforce sum = 1

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (power of two)
    idx = (2 * n - 2) - np.arange(n)      # indices for gradient extraction

    # ---- Adam parameters ------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    alpha = 0.12                          # base learning rate
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    best_vec = cur.copy()
    best_sc = _score_simplex(best_vec, n)
    cur_sc = best_sc

    T = None
    it = 0
    no_improve = 0
    target = 1.50274349

    # ---- main optimisation loop -----------------------------------------
    while time.time() < deadline:
        it += 1

        # convolution of current vector
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()

        # temperature schedule (soft‑max smoothing)
        if T is None:
            T = max_conv * 0.08
        else:
            T = max(T * 0.997, max_conv * 1e-5)

        # soft‑max weights over convolution entries
        w = np.exp((conv - max_conv) / T)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_conv).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of the smoothed max (matrix‑free convolution)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # optional gradient clipping for stability
        gn = np.linalg.norm(grad)
        if gn > 0.5:
            grad = grad * (0.5 / gn)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (alpha / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + epsilon)

        cand = cur - step
        cand = _project_simplex(cand)

        cand_sc = _score_simplex(cand, n)

        # accept if improvement, otherwise occasional uphill move
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (T + 1e-12)):
                cur, cur_sc = cand, cand_sc

        # random redistribution when stuck
        if no_improve > 2000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_simplex(tmp, n)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()

        # symmetrisation every few hundred iterations
        if it % 800 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_simplex(sym, n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # early exit if target reached
        if best_sc < target:
            break

        # safeguard against overrunning the budget
        if time.time() > deadline - 0.1:
            break

    # ---- final local refinement (tiny random moves) --------------------
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        delta = cur[i] * rng.random() * 0.1
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score_simplex(tmp, n)
        if tmp_sc < best_sc:
            best_sc, best_vec = tmp_sc, tmp.copy()
            cur, cur_sc = tmp, tmp_sc

        if best_sc < target:
            break

    return [float(x) for x in best_vec]
