def _score(a, L=None):
    # compute objective = 2 * n * max(conv) / (sum_a**2)
    n = len(a)
    if L is None:
        # compute next power of two >= 2n-1
        L = 1 << ((2*n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af*af, n=L)[:2*n-1]
    max_val = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_val / (sum_a**2))

def _grad_max(a, conv, max_val, L, eps_rel=1e-8):
    # near max mask
    eps = max(eps_rel * max_val, 1e-12)
    mask = (conv >= max_val - eps).astype(np.float64)  # shape (2n-1,)
    rev = a[::-1]
    rev_fft = np.fft.rfft(rev, n=L)
    mask_fft = np.fft.rfft(mask, n=L)
    conv_rev = np.fft.irfft(rev_fft * mask_fft, n=L)
    n = a.shape[0]
    grad = 2.0 * conv_rev[n-1:n-1+n]
    return grad

import time, math, numpy as np, cvxpy as cp

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L=None):
    """Exact AC1 score (no normalization assumed)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_val / (sum_a ** 2))

def _grad_max(a, conv, max_val, L, eps_rel=1e-8):
    """Sub‑gradient of max(conv) w.r.t. a using a mask of near‑max entries."""
    eps = max(eps_rel * max_val, 1e-12)
    mask = (conv >= max_val - eps).astype(np.float64)          # length 2n‑1
    rev = a[::-1]
    rev_fft = np.fft.rfft(rev, n=L)
    mask_fft = np.fft.rfft(mask, n=L)
    conv_rev = np.fft.irfft(rev_fft * mask_fft, n=L)
    n = a.shape[0]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad

def _solve_lp(cur, bound, t_mix=0.2):
    """
    Linear‑program refinement:
        max   sum(g)
        s.t.  conv(cur, g) <= bound, g >= 0
    Returns a convex combination of cur and the optimal g (on the simplex)
    or None on failure.
    """
    n = cur.shape[0]
    g = cp.Variable(n, nonneg=True)
    constraints = [cp.conv(cp.Constant(cur), g) <= bound]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=3000,
                    abstol=1e-8, reltol=1e-6, feastol=1e-8)
    except Exception:
        return None
    if g.value is None:
        return None
    g0 = np.maximum(g.value, 0.0)
    s = g0.sum()
    if s <= 0:
        return None
    g_norm = g0 / s
    return (1.0 - t_mix) * cur + t_mix * g_norm
