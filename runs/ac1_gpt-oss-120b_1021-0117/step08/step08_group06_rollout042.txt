def evaluate_sequence(sequence: list) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))

import time, math, numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function minimizing the AC1 upper bound.

    The routine builds a few analytically‑motivated candidates (linear,
    power‑law, exponential, mixtures) and selects the one with the smallest
    exact score reported by ``evaluate_sequence``.
    """
    start = time.time()
    # keep a tiny safety margin for the final return
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # length of the problem instance
    n = len(height_sequence_1) if "height_sequence_1" in globals() else 2000
    if n < 2:                     # degenerate safety fallback
        return [1.0]

    i = np.arange(n, dtype=np.float64)          # indices 0 … n‑1

    # ----- basic shapes -------------------------------------------------
    # linear decreasing (sum = 1, last entry = 0)
    a_lin = 2.0 * (n - 1 - i) / (n * (n - 1))

    # uniform distribution (sum = 1)
    a_uni = np.full(n, 1.0 / n, dtype=np.float64)

    candidates = [a_lin, a_uni]

    # power‑law shapes  a_i ∝ (n‑1‑i)^p
    for p in (0.5, 1.5, 2.0):
        v = (n - 1 - i) ** p
        candidates.append(v / v.sum())

    # exponential shapes  a_i ∝ r^i
    for r in (0.98, 0.99, 0.995):
        v = r ** i
        candidates.append(v / v.sum())

    # convex mixtures of linear and uniform
    for t in (0.25, 0.5, 0.75):
        mix = t * a_lin + (1.0 - t) * a_uni
        candidates.append(mix / mix.sum())

    best_vec = None
    best_sc = float('inf')

    # ----- evaluation ----------------------------------------------------
    for a in candidates:
        if time.time() >= deadline:
            break
        a = np.clip(a, 0.0, 1000.0)          # respects evaluator’s bounds
        sc = evaluate_sequence(a.tolist())
        if sc < best_sc:
            best_sc, best_vec = sc, a

    # ----- fallback if nothing succeeded ---------------------------------
    if best_vec is None:
        best_vec = a_uni
        best_sc = evaluate_sequence(best_vec.tolist())

    # final sanity check – guarantee a finite score
    try:
        final_sc = evaluate_sequence(best_vec.tolist())
        if not math.isfinite(final_sc):
            best_vec = a_uni
    except Exception:
        best_vec = a_uni

    return [float(x) for x in best_vec]
