    else:

    # Project onto simplex (sum=1)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.shape[0]
    best_vec = cur.copy()
    best_sc = _score(best_vec)

    # Precompute row and col indices for convolution matrix
    i = np.arange(n, dtype=np.int64)[:, None]
    j = np.arange(n, dtype=np.int64)[None, :]
    rows = (i + j).ravel()
    cols = np.tile(np.arange(n, dtype=np.int64), n)

    # Main iteration loop
    iteration = 0

import time, math, numpy as np, scipy.optimize, scipy.sparse

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex (sum=1, non‑negative)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast evaluation of 2·n·max(conv(a,a)) for a vector with sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
