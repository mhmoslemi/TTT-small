import time, math, numpy as np

# ---------- helpers ----------
def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (non‑negative, sum=1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _max_conv(a, n, L):
    """Maximum entry of the full convolution a*a using an FFT of length L."""
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    return conv.max()

def _gradient_mask(a, n, L, conv, max_val):
    """Sub‑gradient of max(conv) at a using the exact active mask."""
    # mask of (near) maximal entries
    tol = 1e-12
    mask = (conv >= max_val - tol).astype(np.float64)
    s = mask.sum()
    if s == 0:
        mask = (conv == max_val).astype(np.float64)
        s = mask.sum()
    mask /= s

    # reverse mask for convolution
    w_rev = mask[::-1]
    a_fft = np.fft.rfft(a, n=L)
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad
