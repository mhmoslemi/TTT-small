def propose_candidate(seed=42,budget_s=1100,**kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s)-0.5)
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur,0.0)
    cur = _project_simplex(cur)
    n = cur.size
    L = 1 << ((2*n-1).bit_length())
    # compute initial best
    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc
    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    beta1_pow = beta2_pow = 1.0
    lr = 0.05
    eps = 1e-8
    # schedule parameters
    iteration = 0
    no_improve = 0
    # main loop
    while True:
        if time.time() >= deadline:
            break
        iteration += 1
        # conv
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2*n - 1]
        max_val = conv.max()
        # indices where max holds
        max_idx = np.flatnonzero(conv == max_val)
        # weight vector
        w = np.zeros_like(conv)
        w[max_idx] = 1.0 / max_idx.size
        # gradient: convolution of cur with reversed w
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = _project_simplex(cur - step)
        cand_sc = _score(cand)
        if cand_sc < cur_sc:
            cur = cand; cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec = cand.copy()
                best_sc = cand_sc
        else:
            no_improve += 1
            # occasional accept with probability based on temperature
            # define temp as decreasing function of iteration
            temp = max_val * 0.01 * (0.998 ** iteration)
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (temp + 1e-12)):
                cur = cand; cur_sc = cand_sc
        # random perturb when stuck
        if no_improve > 2000:
            # make a random small transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
            delta = cur[i] * rng.random() * 0.2
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur = tmp; cur_sc = tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_vec = tmp.copy()
                    best_sc = tmp_sc
        # symmetrization occasional
        if iteration % 1000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym; cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_vec = sym.copy()
                    best_sc = sym_sc
        # optional early exit if target reached
        if best_sc < 1.50274349:
            break
    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0.0] = 0.0
    return w

def _score(a):
    """Fast evaluation of the AC1 bound for a vector with sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        max_idx = np.flatnonzero(conv == max_val)

        # --------------------------------------------------------------

        grad = 2.0 * grad_full[idx]
