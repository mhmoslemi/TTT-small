    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:

    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)
    # init sequence:
    if "height_sequence_1" in globals():
        a = np.array(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        a = rng.random(n)
    a = np.maximum(a, 0)
    a = a / a.sum()   # simplex
    n = a.shape[0]
    # pre-alloc arrays for padded
    pad_len = n - 1
    a_padded = np.empty(3*n-2, dtype=np.float64) # will fill each iteration
    # function to compute obj
    L = 1 << ((2*n - 1).bit_length())
    def compute_obj(x):
        fft_x = np.fft.rfft(x, n=L)
        conv = np.fft.irfft(fft_x * fft_x, n=L)[:2*n-1]
        max_conv = conv.max()
        return max_conv, conv

    # initial best
    max_conv, conv = compute_obj(a)
    best_a = a.copy()
    best_obj = max_conv
    # track iteration
    it = 0
    # hyperparameters
    base_lr = 0.5  # tuning
    decay = 0.999
    min_lr_factor = 1e-4
    # main loop
    while time.time() < deadline:
        it += 1
        # compute conv, max, gradient
        max_conv, conv = compute_obj(a)
        # update best
        if max_conv < best_obj:
            best_obj = max_conv
            best_a = a.copy()
            # optional early exit target
            if best_obj * 2 * n < 1.50274349:
                break
        # find max index
        k0 = int(conv.argmax())
        # compute gradient via padded indexing
        # a_padded = [0]*(pad_len) + a + [0]*pad_len
        a_padded[:pad_len] = 0.0
        a_padded[pad_len:pad_len+n] = a
        a_padded[pad_len+n:] = 0.0
        grad = 2.0 * a_padded[(pad_len + k0) - np.arange(n)]
        # compute step size
        lr = base_lr * (decay ** it)
        lr = max(lr, min_lr_factor)
        # candidate update using exponentiated gradient
        # a_new = a * np.exp(-lr * grad)
        # Avoid overflow: clip grad*lr
        exp_arg = -lr * grad

        exp_arg = np.clip(exp_arg, -50, 0)  # negative only
        a_new = a * np.exp(exp_arg)
        a_new_sum = a_new.sum()
        if a_new_sum == 0.0:

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --- initial vector -------------------------------------------------

        a = rng.random(2000, dtype=np.float64)

    n = a.shape[0]
    # length for FFT (next power of two >= 2n-1)

    def _conv_max(x):
        fft = np.fft.rfft(x, n=L)
        conv = np.fft.irfft(fft * fft, n=L)[:2*n-1]
        maxc = conv.max()
        return maxc, conv

    max_conv, conv = _conv_max(a)
    best_a = a.copy()

    m = np.zeros_like(a)

        max_conv, conv = _conv_max(a)

    if "height_sequence_1" in globals():
        a = np.array(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2000, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = a / a.sum()                       # place on the simplex (sum = 1)

    n = a.shape[0]
    # FFT length – next power of two ≥ 2n‑1
    L = 1 << ((2 * n - 1).bit_length())

    def _conv_max(x):
        """Return max of the autocorrelation of x (sum(x)=1)."""
        fft = np.fft.rfft(x, n=L)
        conv = np.fft.irfft(fft * fft, n=L)[: 2 * n - 1]
        return conv.max(), conv

    max_conv, conv = _conv_max(a)
    best_a = a.copy()
    best_obj = max_conv                     # with sum=1, score = 2·n·best_obj

    # ---- optimisation parameters ----------------------------------------
    m = np.zeros_like(a)                   # momentum
    beta = 0.9
    base_lr = 1.0
    decay = 0.9995
    min_lr = 1e-5
    it = 0
    target_score = 1.50274349

    # ---- main loop ------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # current autocorrelation
        max_conv, conv = _conv_max(a)

        # keep best seen so far
        if max_conv < best_obj:
            best_obj = max_conv
            best_a = a.copy()
            if 2 * n * best_obj < target_score:   # early exit
                break

        # --- gradient from the top few peaks ---------------------------
        M = 5                                 # number of peaks to consider
        if M >= conv.size:
            top_idxs = np.arange(conv.size)
        else:
            thr = np.partition(conv, -M)[-M]
            top_idxs = np.where(conv >= thr)[0]

        pad = n - 1
        a_padded = np.concatenate((np.zeros(pad, dtype=np.float64), a,
                                   np.zeros(pad, dtype=np.float64)))
        idx = np.arange(n)
        grad = np.zeros_like(a)
        for k in top_idxs:
            grad += 2.0 * a_padded[(pad + k) - idx]

        # momentum update
        m = beta * m + (1.0 - beta) * grad

        # step size (decreasing)
        lr = base_lr / math.sqrt(it)
        lr = max(lr, min_lr)

        # candidate by additive update + projection
        cand = a - lr * m
        cand = _project_simplex(cand)
        # enforce symmetry (helps to flatten peaks)
        cand = (cand + cand[::-1]) * 0.5
        cand_sum = cand.sum()
        if cand_sum > 0:
            cand = cand / cand_sum

        cand_obj, _ = _conv_max(cand)

        # accept if improvement; otherwise occasional random shake‑up
        if cand_obj < max_conv:
            a = cand
        elif it % 1000 == 0 and rng.random() < 0.2:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                delta = min(a[i], rng.random() * 0.1)
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()

        # occasional line‑search on the same direction
        if it % 50 == 0:
            best_local = a
            best_local_obj = max_conv
            for factor in (0.5, 1.5):
                tmp = a - lr * factor * m
                tmp = _project_simplex(tmp)
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp = tmp / tmp.sum()
                tmp_obj, _ = _conv_max(tmp)
                if tmp_obj < best_local_obj:
                    best_local_obj = tmp_obj
                    best_local = tmp
            a = best_local

        # sparsify every few hundred iterations
        if it % 500 == 0:
            thresh = np.quantile(a, 0.1)
            a = np.where(a < thresh, 0.0, a)
            a = _project_simplex(a)
            a = (a + a[::-1]) * 0.5
            a = a / a.sum()

    # ---- final random tweaks (if time permits) -------------------------
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        delta = min(best_a[i], rng.random() * 0.02)
        tmp = best_a.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp = (tmp + tmp[::-1]) * 0.5
        tmp = tmp / tmp.sum()
        obj, _ = _conv_max(tmp)
        if obj < best_obj:
            best_obj = obj
            best_a = tmp

    # return the best sequence found as plain Python floats
    return [float(x) for x in best_a]
