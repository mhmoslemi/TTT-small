import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


def _score(a):
    """Fast AC1 upper‑bound for a vector whose entries sum to 1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _build_conv_matrix(f):
    """Return sparse matrix A (shape (2n‑1, n)) with A @ g = f * g."""
    n = f.shape[0]
    # each diagonal offset d (0 ≤ d < n) has constant value f[d]
    diag_data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(0, n, dtype=int)
    return _sp.diags(diag_data, offsets, shape=(2 * n - 1, n), format='csr')
