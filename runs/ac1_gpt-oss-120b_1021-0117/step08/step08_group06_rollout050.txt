import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a, n, L):
    """Fast surrogate for 2*n*max(conv) when sum(a)=1."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---------- initial vector ----------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    # force symmetry (optimal solutions are symmetric)
    cur = (cur + cur[::-1]) * 0.5
    # project to probability simplex (sum=1, non‑negative)
    cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())          # FFT size (next power of two)

    # best exact solution seen so far
    best_vec = cur.copy()
    best_sc = evaluate_sequence(best_vec.tolist())

    # surrogate score for current iterate
    cur_sc = _fast_score(cur, n, L)

    # optimisation state
    momentum = np.zeros_like(cur)
    beta = 0.95            # momentum decay
    base_step = 0.8        # initial step magnitude
    temp_factor = 0.07     # initial temperature as fraction of max(conv)
    min_temp_factor = 1e-5
    it = 0
    no_improve = 0

    # control frequencies
    exact_interval = 1500
    sym_interval = 500
    random_restart_interval = 6000

    while time.time() < deadline:
        it += 1

        # ----- convolution and temperature -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        if it == 1:
            temp = max_val * temp_factor
        else:
            temp = max(temp * 0.997, max_val * min_temp_factor)

        # ----- soft‑max weights -----
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient of weighted max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- momentum & step size -----
        momentum = beta * momentum + (1.0 - beta) * grad
        step = base_step / math.sqrt(it)

        # ----- try several step scalings -----
        cand_best = None
        cand_best_sc = float('inf')
        for factor in (0.4, 0.8, 1.0, 1.2):
            cand = cur - step * factor * momentum
            cand = _project_simplex(cand)
            cand = (cand + cand[::-1]) * 0.5          # enforce symmetry
            cand_sc = _fast_score(cand, n, L)
            if cand_sc < cand_best_sc:
                cand_best_sc = cand_sc
                cand_best = cand

        # ----- acceptance test -----
        accept = False
        if cand_best_sc < cur_sc:
            accept = True
        else:
            prob = math.exp(-(cand_best_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand_best
            cur_sc = cand_best_sc
            no_improve = 0
        else:
            no_improve += 1

        # ----- exact evaluation occasionally -----
        if it % exact_interval == 0:
            true_sc = evaluate_sequence(cur.tolist())
            if true_sc < best_sc:
                best_sc, best_vec = true_sc, cur.copy()

        # ----- random redistribution when stuck -----
        if no_improve > random_restart_interval:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp = (tmp + tmp[::-1]) * 0.5
            tmp_sc = _fast_score(tmp, n, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                true_sc = evaluate_sequence(tmp.tolist())
                if true_sc < best_sc:
                    best_sc, best_vec = true_sc, tmp.copy()
            else:
                # reset momentum to escape flat region
                momentum.fill(0.0)
                no_improve = 0

        # ----- periodic symmetrisation -----
        if it % sym_interval == 0:
            cur = (cur + cur[::-1]) * 0.5
            cur_sc = _fast_score(cur, n, L)

        # ----- early stop if target reached -----
        if best_sc < 1.45:          # comfortably below historic target
            break

    # ----- final sanity check -----
    try:
        final_sc = evaluate_sequence(best_vec.tolist())
        if not math.isfinite(final_sc) or final_sc >= 1e9:
            raise Exception
    except Exception:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(x) for x in best_vec]
