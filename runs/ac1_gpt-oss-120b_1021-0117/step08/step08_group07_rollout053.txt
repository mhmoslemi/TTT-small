import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast evaluation of 2·n·max(conv(a,a)) for a vector with sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ---- initialise -------------------------------------------------

        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        cur = _project_simplex(cur)
    else:
        # fallback: uniform distribution on simplex
        n = 2000
        cur = rng.random(n, dtype=np.float64)
        cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())
    idx = (2 * n - 2) - np.arange(n)          # for gradient extraction

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # ---- Adam optimiser state ---------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.12                                 # base learning rate

    # ---- temperature (soft‑max) ------------------------------------
    temp = None
    decay = 0.999
    min_temp_fac = 1e-4

    it = 0
    no_improve = 0
    target = 1.50274349

    # -----------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # convolution and max
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # temperature schedule
        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay, max_val * min_temp_fac)

        # smooth max weights
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of smooth max
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = -lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur + step
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()

        # acceptance
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1
            # occasional Metropolis acceptance
            prob = math.exp(-(cand_sc - cur_sc) * 0.05)
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # ---------- occasional symmetry enforcement -----------------
        if it % 800 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()

        # ---------- directed mass transfer when stalled ------------
        if (it % 200 == 0) and (no_improve > 500):
            i_max = np.argmax(grad)
            i_min = np.argmin(grad)
            if i_max != i_min:
                delta = min(cur[i_max] * 0.2, 0.001)
                tmp = cur.copy()
                tmp[i_max] -= delta
                tmp[i_min] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()

        # ---------- random redistribution when very stuck ----------
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()

        # early exit if target reached
        if best_sc < target:
            break

    # ---- final polishing with tiny Gaussian perturbations ----------
    while time.time() < deadline:
        pert = best_vec + rng.normal(scale=0.004, size=n)
        pert = np.clip(pert, 0.0, None)
        pert = _project_simplex(pert)
        sc = _score(pert)
        if sc < best_sc:
            best_sc = sc
            best_vec = pert
            if best_sc < target:
                break

    return [float(x) for x in best_vec]
