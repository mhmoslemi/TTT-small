import numpy as np, time, math

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # shift to make sum 1 and nonnegative
    # algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w
```

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # candidate step and projection
        cand = _project_simplex(cur - step)

        # evaluate candidate (cheap: reuse FFT)
        a_fft_c = np.fft.rfft(cand, n=L)
        conv_c = np.fft.irfft(a_fft_c * a_fft_c, n=L)[: 2 * n - 1]
        cand_sc = float(2.0 * n * conv_c.max())

        # accept if improvement; else Metropolis with current temperature
        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (temp + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_sc = cand_sc

        # ---------- diversification heuristics ----------
        if no_improve > 2500:
            # random mass transfer between two positions
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            a_fft_tmp = np.fft.rfft(tmp, n=L)
            conv_tmp = np.fft.irfft(a_fft_tmp * a_fft_tmp, n=L)[: 2 * n - 1]
            tmp_sc = float(2.0 * n * conv_tmp.max())
            if tmp_sc < cur_sc:
                cur = tmp
                cur_sc = tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
            else:
                no_improve = 0

        if it % 1100 == 0:
            # symmetrisation often reduces the peak
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            a_fft_sym = np.fft.rfft(sym, n=L)
            conv_sym = np.fft.irfft(a_fft_sym * a_fft_sym, n=L)[: 2 * n - 1]
            sym_sc = float(2.0 * n * conv_sym.max())
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # learning‑rate decay when stuck for a long time
        if no_improve > 5000:
            lr = max(lr * 0.5, 1e-4)
            no_improve = 0

        # early exit if target attained
        if best_sc < target:
            break

    # ---------- return ----------
    return [float(x) for x in best_vec]
