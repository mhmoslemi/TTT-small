    deadline = start + max(0.0, budget_s - 0.2)

    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    alpha = 0.05  # base learning rate

    temp = None
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:

        # compute convolution etc.
        a_fft = np.fft.rfft(cur, n=L)

        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * 0.9995, max_val * 1e-5)

        w = np.exp((conv_full - max_val) / temp)
        w_sum = w.sum()

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_seq(a, L):
    """Return the AC1 upper‑bound score for a non‑negative vector a (sum≈1)."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.shape[0] - 1]
    return float(2.0 * a.shape[0] * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function minimizing the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # ---- initial sequence -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # enforce sum = 1, non‑negative

    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # power‑of‑two FFT length

    # ---- optimisation state -----------------------------------------------
    best = cur.copy()
    best_score = _score_seq(best, L)
    cur_score = best_score

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.2

    temp = None
    no_improve = 0
    target = 1.50274349
    it = 0

    # ---- main loop --------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # convolution of current vector
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        cur_score = 2.0 * n * max_val

        # temperature for soft‑max (approaches true max)
        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * 0.9995, max_val * 1e-5)

        # soft‑max weights over convolution entries
        w = np.exp((conv_full - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv_full == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of weighted max (via FFT)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ---- Adam update ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = base_lr / math.sqrt(it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        cand = _project_simplex(cand)

        # evaluate candidate
        cand_score = _score_seq(cand, L)

        # accept if better, otherwise occasional SA step
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
        else:
            no_improve += 1
            prob = math.exp(-(cand_score - cur_score) / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score
                no_improve = 0

        # keep global best
        if cur_score < best_score:
            best, best_score = cur.copy(), cur_score

        # ---- heuristic perturbations ---------------------------------------
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_score = _score_seq(tmp, L)
            if tmp_score < cur_score:
                cur, cur_score = tmp, tmp_score
                no_improve = 0
                if tmp_score < best_score:
                    best, best_score = tmp.copy(), tmp_score

        # symmetrisation every ~800 iterations
        if it % 800 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score_seq(sym, L)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                if sym_score < best_score:
                    best, best_score = sym.copy(), sym_score

        # sparsification every ~1600 iterations
        if it % 1600 == 0:
            thresh = np.quantile(cur, 0.15)   # prune ~15 % smallest entries
            tmp = cur.copy()
            tmp[tmp < thresh] = 0.0
            tmp = _project_simplex(tmp)
            tmp_score = _score_seq(tmp, L)
            if tmp_score < cur_score:
                cur, cur_score = tmp, tmp_score
                if tmp_score < best_score:
                    best, best_score = tmp.copy(), tmp_score

        # early exit if target reached
        if best_score < target:
            break

        # cheap time check near the deadline
        if time.time() > deadline - 0.5:
            break

    # return plain Python list
    return [float(x) for x in best]
