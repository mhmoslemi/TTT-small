import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L):
    """Fast AC1 score for a vector already normalized to sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    grad_every = 100                      # gradient step frequency
    sym_every = 2000                      # symmetrisation frequency

    # ---- main optimisation loop -----------------------------------------
    while time.time() < deadline and it < max_iters:
        it += 1
        # temperature (linear cooling)
        T = T0 * (1.0 - it / max_iters)
        if T < T_end:
            T = T_end

        # ----- stochastic mass transfer ---------------------------------
        # choose a source index weighted by current mass
        i = int(rng.choice(n, p=cur))
        # decide whether to move far or near
        if rng.random() < 0.7:
            # far: offset at least n/4 away
            offset = rng.integers(n // 4, n // 2 + 1)
            j = (i + offset) % n
        else:
            # uniform random destination distinct from i
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
        # move a fraction of the mass from i to j
        delta = cur[i] * rng.random() * 0.5   # up to 50 % of source
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # renormalise (sum stays 1 up to floating error)
        cand /= cand.sum()
        cand_sc = _score(cand, L)

        # acceptance (simulated annealing)
        if cand_sc < cur_sc or rng.random() < math.exp(-(cand_sc - cur_sc) / max(T, 1e-12)):
            cur = cand
            cur_sc = cand_sc

        # ----- deterministic gradient descent step -----------------------
        if it % grad_every == 0:
            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[: 2 * n - 1]
            max_val = conv.max()
            mask = np.isclose(conv, max_val, atol=1e-12)
            w = mask.astype(np.float64)
            w_rev = w[::-1]
            wf_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * wf_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            grad = 2.0 * grad_full[idx]
            step = 0.5 / math.sqrt(it)   # decreasing step size
            cand2 = _project_simplex(cur - step * grad)
            cand2_sc = _score(cand2, L)
            if cand2_sc < cur_sc:
                cur = cand2
                cur_sc = cand2_sc
            if cand2_sc < best_sc:
                best_vec = cand2.copy()
                best_sc = cand2_sc

        # ----- occasional symmetrisation --------------------------------
        if it % sym_every == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
            if sym_sc < best_sc:
                best_vec = sym.copy()
                best_sc = sym_sc

        # ----- keep global best -----------------------------------------
        if cur_sc < best_sc:
            best_vec = cur.copy()
            best_sc = cur_sc

        # ----- early exit -------------------------------------------------
        if best_sc < 1.50274349:
            break

    return [float(x) for x in best_vec]
