import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    cond = u - css / np.arange(1, n + 1) > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


def _bound_fast(a: np.ndarray, L: int) -> float:
    """Fast evaluation of the AC1 bound when sum(a)=1:
    returns 2 * n * max(conv(a,a))."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.15)   # small safety margin

    max_step = 1.0

    min_temp_factor = 1e-5

    it = 0
    target = 1.50274349       # known target bound

    # ----- main loop ----------------------------------------------------
    while time.time() < deadline:
        it += 1

        a_fft = np.fft.rfft(cur, n=L)

        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        if w_sum == 0.0:
