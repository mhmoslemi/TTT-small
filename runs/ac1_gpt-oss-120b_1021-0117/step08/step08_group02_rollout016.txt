import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex {x>=0, sum=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(vec):
    """Fast score 2*n*max(conv)/(sum^2) for a non‑negative vector."""
    a = np.asarray(vec, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return float(2.0 * n * max_val / (s * s))
