def _project_simplex(v):
    # as before
    v = np.asarray(v)
    n = v.shape[0]
    # get sorted descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0
    return w

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)

    eps = 1e-8
    temp_factor = 0.08   # initial temperature as fraction of max conv
    temp_decay = 0.9995
    target = 1.50274349

    # ----- state ---------------------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    best_vec = cur.copy()
    best_score = evaluate_sequence(best_vec.tolist())
    cur_score = best_score
    temp = None
    it = 0
    no_improve = 0

    # ----- main optimisation loop ----------------------------------------
    while time.time() < deadline:
        it += 1

        # full convolution of current iterate
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # temperature schedule (soft‑max smoothing)
        if temp is None:
            temp = max_val * temp_factor
        else:
            temp = max(temp * temp_decay, max_val * 1e-5)

        # soft‑max weights (focus on near‑max entries)
        w = np.exp((conv_full - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv_full == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of weighted max (via convolution)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # exponentiated gradient step
        step = step0 / math.sqrt(it)
        multiplier = np.exp(-step * m_hat / (np.sqrt(v_hat) + eps))
        cand = cur * multiplier
        cand = _project_simplex(cand)

        # approximate score (sum≈1 ⇒ 2·n·max_conv)
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_max = cand_conv.max()
        cand_score_approx = 2.0 * n * cand_max

        # acceptance
        if cand_score_approx < cur_score:
            cur, cur_score = cand, cand_score_approx
            no_improve = 0
            exact = evaluate_sequence(cand.tolist())
            if exact < best_score:
                best_score, best_vec = exact, cand.copy()
        else:
            no_improve += 1
            prob = math.exp(-(cand_score_approx - cur_score) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score_approx

        # ----- diversification when stagnating ---------------------------
        if no_improve > 2000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_fft = np.fft.rfft(tmp, n=L)
            tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
            tmp_score_approx = 2.0 * n * tmp_conv.max()
            if tmp_score_approx < cur_score:
                cur, cur_score = tmp, tmp_score_approx
                no_improve = 0
                exact = evaluate_sequence(tmp.tolist())
                if exact < best_score:
                    best_score, best_vec = exact, tmp.copy()

        # ----- symmetrisation ------------------------------------------------
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
            sym_score_approx = 2.0 * n * sym_conv.max()
            if sym_score_approx < cur_score:
                cur, cur_score = sym, sym_score_approx
                exact = evaluate_sequence(sym.tolist())
                if exact < best_score:
                    best_score, best_vec = exact, sym.copy()

        # ----- sparsification ------------------------------------------------
        if it % 1200 == 0:
            thresh = np.quantile(cur, 0.2)   # keep top 80 %
            tmp = cur.copy()
            tmp[tmp < thresh] = 0.0
            tmp = _project_simplex(tmp)
            tmp_fft = np.fft.rfft(tmp, n=L)
            tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
            tmp_score_approx = 2.0 * n * tmp_conv.max()
            if tmp_score_approx < cur_score:
                cur, cur_score = tmp, tmp_score_approx
                exact = evaluate_sequence(tmp.tolist())
                if exact < best_score:
                    best_score, best_vec = exact, tmp.copy()

        # ----- early exit ---------------------------------------------------
        if best_score < target:
            break

    # fallback safety
    if not np.isfinite(best_score):
        best_vec = cur.copy()
        best_score = evaluate_sequence(best_vec.tolist())

    # return plain Python floats
    return [float(x) for x in best_vec]
