def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy, time, math
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s)-0.2)

    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # random length maybe 2500
        n = 2500
        cur = rng.random(n, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    # project onto simplex (sum=1)
    cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2*n-1).bit_length())
    idx = (2*n-2) - np.arange(n)

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    base_alpha = 0.1   # larger base
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    # temperature
    T = None

    best_vec = cur.copy()
    best_sc = _score_simplex(best_vec, n)
    cur_sc = best_sc

    it = 0
    no_improve_iters = 0
    # thresholds
    max_no_improve = 3000
    sym_interval = 500

    while time.time() < deadline:
        it += 1

        # compute conv and max
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_conv = conv.max()

        # update temperature
        if T is None:
            T = max_conv * 0.05
        else:
            T = max(T * 0.995, max_conv * 5e-4)  # lower bound relative

        # softmax weights
        w = np.exp((conv - max_conv) / T)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_conv).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = (base_alpha / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + epsilon)

        # candidate with backtracking
        cand = cur - step
        cand = _project_simplex(cand)

        cand_sc = _score_simplex(cand, n)
        # backtrack if not improved
        bt = 0
        while cand_sc > cur_sc and bt < 5:
            step *= 0.5
            cand = cur - step
            cand = _project_simplex(cand)
            cand_sc = _score_simplex(cand, n)
            bt += 1

        # accept if improved
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve_iters = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve_iters += 1

        # random redistribution if stuck
        if no_improve_iters > max_no_improve:
            # pick random i, j
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_simplex(tmp, n)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve_iters = 0
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
            else:
                # maybe also reset cur to best?
                cur = best_vec.copy()
                cur_sc = best_sc
                no_improve_iters = 0

        # symmetrize periodically (or every iteration)
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_simplex(sym, n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # early stop if target reached
        if best_sc < 1.50274349:
            break

    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_simplex(a, n):
    """Score for a vector a that satisfies sum(a)=1: 2*n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
