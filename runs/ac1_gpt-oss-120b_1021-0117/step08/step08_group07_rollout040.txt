import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex (sum=1, non‑negative)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a: np.ndarray) -> float:
    """Fast evaluation of 2·n·max(conv(a,a)) for a vector with sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        t += 1

        # ---- convolution & active set ---------------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val

        # ---- sub‑gradient of the max (exact) -------------------------
        # active indices where the convolution attains its maximum
        tol = max_val * 1e-12 + 1e-14
        active = (conv >= max_val - tol)
        if not np.any(active):
            active = np.ones_like(conv, dtype=bool)
        w = active.astype(np.float64)
        w /= w.sum()                     # uniform weight on the active set
        w_rev = w[::-1]

        # gradient = 2 * (a * w_rev)  (convolution in the frequency domain)
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ---- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = step0 * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        if cand_sc < best_sc:
            best_sc = cand_sc
            best = cand.copy()

        # ---- acceptance ------------------------------------------------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1
            # occasional Metropolis move (temperature decays with iterations)
            prob = math.exp(-(cand_sc - cur_sc) / (step0 + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # ---- when progress stalls, perform a directed mass transfer ----
        if no_improve > 800:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < best_sc:
                best_sc = tmp_sc
                best = tmp.copy()
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0

        # ---- early exit if target achieved ----------------------------
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # final local‑search polishing (tiny Gaussian perturbations)
    while time.time() < deadline:
        pert = best + rng.normal(scale=0.001, size=n)
        pert = np.clip(pert, 0.0, None)
        pert = _project_simplex(pert)
        sc = _score(pert)
        if sc < best_sc:
            best_sc = sc
            best = pert.copy()
            if best_sc < target:
                break

    # return plain Python floats
    return [float(x) for x in best]
