def grad_softmax(a, conv, max_val, L, beta=50.0):

    w = np.exp(beta * (conv - max_val))

    w_sum = w.sum()

    if w_sum == 0: w = np.zeros_like(w); w[conv.argmax()] = 1.0; w_sum = 1.0

    w = w / w_sum

    # reverse w

    w_rev = w[::-1]

    a_fft = np.fft.rfft(a, n=L)

    w_fft = np.fft.rfft(w_rev, n=L)

    grad_full = np.fft.irfft(a_fft * w_fft, n=L)

    n = a.shape[0]

    idx = (2 * n - 2) - np.arange(n)

    grad = 2.0 * grad_full[idx]

    return grad

    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_fft(a, L):
    """Fast proxy for the true score: 2*n*max(conv(a,a)) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _grad_softmax(a, conv, max_val, L, gamma=30.0):
    """
    Approximate subgradient of max(conv) using a soft‑max weighting.
    Returns a vector of length len(a).
    """
    # soft‑max weights (sum to 1)
    w = np.exp(gamma * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(conv)
        w[conv.argmax()] = 1.0
        w_sum = 1.0
    w = w / w_sum
    w_rev = w[::-1]

    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)

    n = a.shape[0]
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

def _random_perturb(x, rng, eps=0.1):
    """Move a random fraction of mass from one coordinate to another."""
    n = x.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    delta = eps * x[i] * rng.random()
    y = x.copy()
    y[i] -= delta
    y[j] += delta
    return _project_simplex(y)

# ---------- main entry point ------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # ---- initialise --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: generate a random simplex vector of length ~2000
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # next power‑of‑two for FFT

    best_vec = cur.copy()
    best_sc = _score_fft(best_vec, L)
    cur_sc = best_sc

    # momentum parameters
    momentum = np.zeros_like(cur)
    beta_mom = 0.92
    step0 = 1.0
    stagnate = 0
    target = 1.50274349
    it = 0

    # ---- main optimisation loop -------------------------------------------
    while time.time() < deadline:
        it += 1
        # convolution via FFT
        af = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(af * af, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # gradient (soft‑max approximation of the max‑operator)
        grad = _grad_softmax(cur, conv, max_val, L, gamma=30.0)

        # momentum update
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad

        # step‑size schedule
        step = step0 / math.sqrt(it)

        # candidate by a projected gradient step
        cand = _project_simplex(cur - step * momentum)
        cand_sc = _score_fft(cand, L)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
            stagnate = 0
        else:
            stagnate += 1

        # ----- occasional random rescue ------------------------------------
        if stagnate > 2000:
            # random perturbation of the current best
            pert = _random_perturb(cur, rng, eps=0.2)
            pert_sc = _score_fft(pert, L)
            if pert_sc < cur_sc:
                cur, cur_sc = pert, pert_sc
                if pert_sc < best_sc:
                    best_vec, best_sc = pert.copy(), pert_sc
                stagnate = 0
            else:
                # mix with a fresh random simplex vector
                fresh = _project_simplex(rng.random(n))
                mix = 0.5 * cur + 0.5 * fresh
                mix = _project_simplex(mix)
                mix_sc = _score_fft(mix, L)
                if mix_sc < cur_sc:
                    cur, cur_sc = mix, mix_sc
                    if mix_sc < best_sc:
                        best_vec, best_sc = mix.copy(), mix_sc
                    stagnate = 0

        # ----- occasional symmetrisation ------------------------------------
        if it % 1200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_fft(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # early exit if target reached
        if best_sc < target:
            break

    # Return the best vector found (as plain Python floats)
    return [float(v) for v in best_vec]
