    n = cur.shape[0]; L = ...

    best = cur.copy(); best_sc = _score(best)

    cur_sc = best_sc

    momentum = np.zeros_like(cur)

    beta = 0.97

    step0 = 0.5

    temp = None; decay = 0.998; min_temp_fac = 1e-5

    no_improve = 0; it = 0

    while time.time() < deadline:

        it += 1

        # compute FFT and conv
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()

        # update temperature
        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay, max_val * min_temp_fac)

        # compute softmax weights
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient from softmax
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """2·n·max(conv(a,a)) for a vector with sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _optimize(cur, deadline, rng, target_score):
    """Projected gradient descent with soft‑max smoothing + exact max step."""
    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    best = cur.copy()
    best_sc = _score(best)

    cur_sc = best_sc
    momentum = np.zeros_like(cur)
    beta = 0.97          # momentum decay
    step0 = 0.5          # base step size
    temp = None
    decay = 0.998
    min_temp_fac = 1e-5
    no_improve = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- convolution -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- temperature for smooth max -----
        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay, max_val * min_temp_fac)

        # ----- soft‑max weights -----
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient of smooth max (FFT‑based) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- momentum & step -----
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1.0)

        # ----- candidate from smooth gradient -----
        cand = cur - step * momentum
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        # ----- exact max‑gradient candidate (hard max) -----
        max_inds = np.flatnonzero(conv == max_val)
        grad_max = np.zeros_like(cur)
        for k in max_inds:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1)
            grad_max[idx_i] += 2.0 * cur[k - idx_i]
        grad_max /= max(1, len(max_inds))
        cand2 = cur - step * grad_max
        cand2 = _project_simplex(cand2)
        cand2_sc = _score(cand2)

        # keep the better of the two candidates
        if cand2_sc < cand_sc:
            cand, cand_sc = cand2, cand2_sc

        # ----- update best overall -----
        if cand_sc < best_sc:
            best_sc = cand_sc
            best = cand.copy()

        # ----- accept / probabilistic move -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # ----- directed mass transfer when progress stalls -----
        if (it % 150 == 0) and (no_improve > 300):
            i_max = np.argmax(grad)
            i_min = np.argmin(grad)
            if i_max != i_min:
                delta = min(cur[i_max] * 0.3, 0.0005)
                tmp = cur.copy()
                tmp[i_max] -= delta
                tmp[i_min] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best = tmp.copy()

        # ----- random redistribution when very stuck -----
        if no_improve > 2000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best = tmp.copy()

        # ----- occasional symmetrisation -----
        if it % 1000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best = sym.copy()

        # ----- early exit if target reached -----
        if best_sc < target_score:
            break

    return best, best_sc

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a small safety margin before the sandbox timeout
    deadline = start + max(0.0, float(budget_s) - 0.4)

    # ---- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        cur0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur0 = rng.random(2000, dtype=np.float64)
    cur0 = _project_simplex(np.maximum(cur0, 0.0))
    n = cur0.shape[0]

    best_vec = cur0.copy()
    best_sc = _score(best_vec)

    target = 1.50274349

    # ---- multi‑restart optimisation ---------------------------------
    while time.time() < deadline - 5.0:
        # decide on a start vector for this run
        if best_sc > 1.6:
            init = _project_simplex(rng.random(n, dtype=np.float64))
        else:
            init = best_vec.copy()

        # give each run at most ~30 s (or the remaining time)
        sub_deadline = min(deadline - 2.0, time.time() + 30.0)
        if sub_deadline - time.time() < 1.0:
            break

        cand_vec, cand_sc = _optimize(init, sub_deadline, rng, target)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand_vec.copy()
        if best_sc < target:
            break

    # ---- final polishing with small Gaussian perturbations -----
    while time.time() < deadline:
        pert = best_vec + rng.normal(scale=0.003, size=n)
        pert = np.clip(pert, 0.0, None)
        pert = _project_simplex(pert)
        sc = _score(pert)
        if sc < best_sc:
            best_sc = sc
            best_vec = pert.copy()
            if best_sc < target:
                break

    # return plain Python floats
    return [float(x) for x in best_vec]
