def solve_lp_max_sum(f, M):
    n = f.shape[0]
    A = build_conv_matrix(f)
    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M * np.ones(2*n-1)]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, abstol=1e-8, reltol=1e-8)
    except cp.error.SolverError:
        return None
    if b.value is None:
        return None
    return np.array(b.value, dtype=np.float64)

import time, math, numpy as np
import scipy.sparse as sp
import cvxpy as cp

# ----------------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0.0] = 0.0
    return w

# ----------------------------------------------------------------------
def _score(a):
    """Exact AC1 upper bound for a (no assumption on sum)."""
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

# ----------------------------------------------------------------------
def _gradient_and_max(a):
    """Sub‑gradient of the max‑convolution objective and the current max value."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv_full.max()
    # mask of positions attaining (or nearly attaining) the maximum
    mask = conv_full >= max_val - 1e-12
    w = mask.astype(np.float64)
    w_sum = w.sum()
    if w_sum == 0.0:                     # safety fallback
        w = (conv_full == max_val).astype(np.float64)
        w_sum = w.sum()
    w /= w_sum
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, max_val

def _build_conv_matrix(f):
    """Sparse matrix A (size (2n‑1)×n) such that A @ b = f * b (linear convolution)."""
    n = f.shape[0]
    m = 2 * n - 1
    col = np.tile(np.arange(n, dtype=np.int32), n)
    row_off = np.repeat(np.arange(n, dtype=np.int32), n)
    row = row_off + col
    data = np.repeat(f, n)
    return sp.csr_matrix((data, (row, col)), shape=(m, n))

# ----------------------------------------------------------------------
def _solve_lp_max_sum(f, M):
    """Linear programme: maximise Σ b_j  subject to  b≥0,  f*b ≤ M."""
    n = f.shape[0]
    A = _build_conv_matrix(f)
    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M * np.ones(2 * n - 1)]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=2000, abstol=1e-9, reltol=1e-9)
    except (cp.error.SolverError, Exception):
        return None
    if b.value is None:
        return None
    return np.asarray(b.value, dtype=np.float64)

    """Short deterministic hill‑climb using the sub‑gradient."""
    for i in range(steps):

        step = 0.5 / math.sqrt(i + 1)
        cand = _project_simplex(a - step * grad)

        if it % 1000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # occasional sparsification (keep stronger entries)
        if it % 1500 == 0:
            thresh = np.quantile(cur, 0.2)          # prune bottom 20 %
            tmp = cur.copy()
            tmp[tmp < thresh] = 0.0
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

        # lightweight smoothing (3‑point moving average)
        if it % 2000 == 0:
            smooth = np.convolve(cur, np.array([0.25, 0.5, 0.25]), mode='same')
            smooth = _project_simplex(smooth)
            sm_sc = _score(smooth)
            if sm_sc < cur_sc:
                cur, cur_sc = smooth, sm_sc
                if sm_sc < best_sc:
                    best_vec, best_sc = smooth.copy(), sm_sc

        # ------------------------------------------------------------------
        # LP‑based boost every lp_interval iterations
        if it % lp_interval == 0 and time.time() + 5.0 < deadline:
            b = _solve_lp_max_sum(cur, max_val)
            if b is not None and np.all(np.isfinite(b)):
                sb = b.sum()
                if sb > 0:
                    b_norm = b / sb
                    b_norm = _project_simplex(b_norm)
                    cur = b_norm
                    cur_sc = _score(cur)
                    if cur_sc < best_sc:
                        best_vec, best_sc = cur.copy(), cur_sc

        # early exit if we have already crossed the target
        if best_sc < target:
            break

    # final short deterministic refinement if time permits
    if time.time() + 1.0 < deadline:
        refined = _local_refine(best_vec, steps=120)
        if _score(refined) < best_sc:
            best_vec = refined
            best_sc = _score(refined)

    # return as plain Python list
    return [float(x) for x in best_vec]
