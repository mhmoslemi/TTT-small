def _solve_lp(f, M, active_idx):
    n = f.size
    rows = []
    cols = []
    data = []
    for row_idx, k in enumerate(active_idx):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals
        rows.append(np.full_like(i_vals, row_idx))
        cols.append(j_vals)
        data.append(f[i_vals])
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    A = sp.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver='ECOS', warm_start=True, max_iters=2000, verbose=False)
    except cp.error.SolverError:
        return None
    if g.value is None:
        return None
    return np.array(g.value).reshape(-1)

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else css[rho[-1]] / (rho[-1] + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_norm1(a, L=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _conv_fft(a, L=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv, L

def _solve_lp(f, M, active_idx):
    try:
        import cvxpy as cp
    except Exception:
        return None
    n = f.size
    rows = []
    cols = []
    data = []
    for r, k in enumerate(active_idx):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int32)
        j_vals = k - i_vals
        rows.append(np.full(i_vals.shape, r, dtype=np.int32))
        cols.append(j_vals)
        data.append(f[i_vals])
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    A = sp.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver='ECOS', warm_start=True, max_iters=2000, verbose=False)
    except Exception:
        return None
    if g.value is None:
        return None
    return np.array(g.value, dtype=np.float64)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # initialization
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best = cur.copy()
    best_score = _score_norm1(best, L)
    cur_score = best_score

    # --- LP warm‑start -------------------------------------------------
    conv_full, _ = _conv_fft(cur, L)
    M = conv_full.max()
    num_active = min(200, conv_full.size)
    active_idx = np.argpartition(-conv_full, num_active)[:num_active]
    g = _solve_lp(cur, M, active_idx)
    if g is not None and np.isfinite(g).all():
        if g.sum() == 0:
            g = np.full_like(g, 1.0 / n)
        else:
            g = g / g.sum()
        t = 0.25
        cur = _project_simplex((1 - t) * cur + t * g)
        cur_score = _score_norm1(cur, L)
        if cur_score < best_score:
            best, best_score = cur.copy(), cur_score

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.06
    max_step = 0.55

    temp = None
    temp_decay = 0.9994
    min_temp_factor = 1e-5

    it = 0
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1

        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        if temp is None:
            temp = max_val * 0.08
        else:
            temp = max(temp * temp_decay, max_val * min_temp_factor)

        w = np.exp((conv_full - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv_full == max_val).astype(np.float64)
            w_sum = w.sum()
        w = w / w_sum

        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (lr / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps)

        step_norm = np.linalg.norm(step)
        if step_norm > max_step:
            step = step * (max_step / step_norm)

        cand = _project_simplex(cur - step)
        cand_score = _score_norm1(cand, L)

        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / (temp + 1e-12)):
            cur, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
        else:
            no_improve += 1

        # random redistribution if stuck
        if no_improve > 3500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_score = _score_norm1(tmp, L)
            if tmp_score < cur_score:
                cur, cur_score = tmp, tmp_score
                no_improve = 0
                if tmp_score < best_score:
                    best, best_score = tmp.copy(), tmp_score

        # occasional symmetrisation
        if it % 1200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score_norm1(sym, L)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                no_improve = 0
                if sym_score < best_score:
                    best, best_score = sym.copy(), sym_score

        # restart with noise if really stuck
        if no_improve > 7000:
            cur = best + rng.normal(scale=0.01, size=n)
            cur = _project_simplex(np.maximum(cur, 0.0))
            cur_score = _score_norm1(cur, L)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        if best_score <= target:
            break

        # keep deadline safety
        if time.time() > deadline - 0.5:
            break

    return [float(x) for x in best]
