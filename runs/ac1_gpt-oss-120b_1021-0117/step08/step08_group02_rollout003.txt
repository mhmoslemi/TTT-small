import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _objective(a, n, L):
    """Return 2*n*max(conv)/(sum(a)**2) for a non‑negative vector."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s <= 0:
        return np.inf
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max() / (s * s)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # leave a little head‑room for the final return
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # --------------------------------------------------------------
    # problem size & initial vector
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        cur = _project_simplex(cur)          # sum = 1, non‑negative
    else:
        n = 2000
        cur = rng.random(n, dtype=np.float64)
        cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())       # FFT size ≥ 2n‑1

    # --------------------------------------------------------------
    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    lr = 0.2
    eps = 1e-8

    # --------------------------------------------------------------
    # temperature for soft‑max
    temp = None
    decay = 0.998
    min_temp_factor = 1e-5

    # --------------------------------------------------------------
    best_vec = cur.copy()
    best_sc = _objective(best_vec, n, L)

    cur_sc = best_sc
    no_improve = 0
    it = 0
    target = 1.50274349

    # --------------------------------------------------------------

    while time.time() < deadline:
        it += 1

        # ---- convolution and max
        a_fft = np.fft.rfft(cur, n=L)

        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ---- temperature update (soft‑max smoothing)
        if temp is None:
            temp = max_val * 0.15
        else:

        # ---- soft‑max weights
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:

        w /= w_sum

        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        temp = max_val * 5e-5          # very tight smoothing
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
