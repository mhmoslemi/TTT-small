import time, math, numpy as np

def _project_simplex(v):
    """Project onto the probability simplex (non-negative, sum=1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---- initialization ----
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        if s == 0.0:
            cur = np.full_like(cur, 1.0 / cur.size)
        else:
            cur /= s
    else:
        n0 = 2500
        cur = np.full(n0, 1.0 / n0, dtype=np.float64)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())      # power‑of‑two FFT length
    idx = (2 * n - 2) - np.arange(n)          # index map for gradient extraction

    # ---- optimizer state ----
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    base_lr = 0.12          # base learning rate (larger than previous)

def _project_simplex(v):
    """Project vector v onto the probability simplex (non‑negative, sum = 1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w
