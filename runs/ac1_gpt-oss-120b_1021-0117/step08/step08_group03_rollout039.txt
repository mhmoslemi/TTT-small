import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (project onto simplex, score, gradient of max‑conv)
# ----------------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L):
    """Fast objective for a simplex‑normalised vector: 2·n·max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _grad_max(a, conv, max_val, L):
    """Sub‑gradient of the (binary) max of the convolution w.r.t. a."""
    mask = np.isclose(conv, max_val, atol=1e-12)
    w = mask.astype(np.float64)                 # weights on the convolution output
    w_rev = w[::-1]                             # reverse for the convolution trick
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)      # linear convolution gradient (full)
    n = a.shape[0]
    idx = (2 * n - 2) - np.arange(n)            # pick the relevant slice
    return 2.0 * grad_full[idx]

# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.1)   # tiny safety margin

    # ----- initialise ----------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)

    cur = _project_simplex(np.maximum(cur, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two for FFT

    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.15          # base learning rate
    it = 0
    no_improve = 0
    # occasional uniform mixing strength
    uniform_mix = 0.02

    while time.time() < deadline:
        it += 1

        # convolution and its maximum
        af = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(af * af, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # gradient of max‑conv
        grad = _grad_max(cur, conv, max_val, L)

        # Adam update (move opposite to grad)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = cur - step
        cand = _project_simplex(cand)

        cand_sc = _score(cand, L)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1

        # ---- random rescue when stuck ------------------------------------
        if no_improve > 800:
            # tiny random redistribution between two indices
            i, j = rng.integers(0, n, size=2)
            if i == j:
                j = (j + 1) % n
            delta = cur[i] * rng.random() * 0.25
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
            else:
                # blend a bit towards the uniform vector
                uni = np.full_like(cur, 1.0 / n)
                cur = (1.0 - uniform_mix) * cur + uniform_mix * uni
                cur = _project_simplex(cur)
                cur_sc = _score(cur, L)
                no_improve = 0

        # ---- occasional symmetrisation ------------------------------------
        if it % 1200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # early exit if we hit a very low score (well below any known bound)
        if best_sc < 1.4:
            break

    # Return the best vector found as plain Python floats
    return [float(x) for x in best_vec]
