import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):

    a = np.asarray(a, dtype=np.float64)

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

            no_improve = 0
            temp = None

        # ----- early exit if target beaten -----
        if best_sc <= 1.50274349 * 0.999:
            break

    # ------------------------------------------------------------------ #
    # Final fine‑grained hill climbing (cheap local moves)
    cur = best_vec.copy()
    cur_sc = best_sc
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        eps_move = cur[i] * rng.random() * 0.02
        if eps_move <= 0.0:
            continue
        tmp = cur.copy()
        tmp[i] -= eps_move
        tmp[j] += eps_move
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()

    # ------------------------------------------------------------------ #
    # Return the best sequence found
    return [float(x) for x in best_vec]
