def _score(a, L=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    max_val = conv.max()
    sum_a = a.sum()
    return float(2.0 * n * max_val / (sum_a * sum_a))

import time, math, numpy as np

# ------------------------------------------------------------
# Helper functions (all top‑level as required)
# ------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


def _score(a, L=None):
    """Exact score used by the evaluator (scale‑invariant)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    # safeguard against division by zero – the caller guarantees s>0
    return float(2.0 * n * max_val / (s * s))


def _grad_max(a, conv, max_val, L):
    """Sub‑gradient of max(conv) w.r.t. a (binary mask for all maximal entries)."""
    mask = np.isclose(conv, max_val, atol=1e-12).astype(np.float64)
    w_rev = mask[::-1]
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    n = a.shape[0]
    idx = (2 * n - 2) - np.arange(n)
    # factor 2 because derivative of a*a is 2*a (convolution form)
    return 2.0 * grad_full[idx]
