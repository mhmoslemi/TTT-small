import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Score = 2·n·max(conv(a,a)) assuming sum(a)=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------
    # initialise
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    if cur.sum() == 0:
        cur[:] = 1.0
    cur = cur / cur.sum()
    cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    momentum = np.zeros_like(cur)
    beta = 0.95          # momentum decay
    step0 = 0.8          # initial step magnitude
    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while time.time() < deadline:
        # convolution
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        i_max = int(conv.argmax())

        # gradient from the (unique) maximal autocorrelation entry
        idx = np.arange(n)
        j = i_max - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(cur)
        grad[mask] = cur[mask] + cur[j[mask]]
        grad *= 2.0

        # momentum & step
        momentum = beta * momentum + (1.0 - beta) * grad
        step = step0 / math.sqrt(it + 1.0)

        # candidate update
        cand = cur - step * momentum
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        # keep best ever
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()

        # accept / probabilistic move
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
