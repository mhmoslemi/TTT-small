import time, math, numpy as np

def _project_simplex(v):
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0.0] = 0.0
    return w

def _score_norm1(a):
    """Score for a vector with sum = 1: 2 * n * max(convolution(a, a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # sum = 1, non‑negative

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best = cur.copy()
    best_score = _score_norm1(best)
    cur_score = best_score

    no_improve = 0
    it = 0
    target = 1.50274349

    # ---- main search loop -----------------------------------------------
    while time.time() < deadline:
        it += 1

        # convolution and max index
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        idx_max = int(conv.argmax())

        # gradient of the (currently) maximal convolution entry
        g = np.zeros_like(cur)
        for j in range(n):
            k1 = idx_max - j
            if 0 <= k1 < n:
                g[j] += cur[k1]
            k2 = j - idx_max
            if 0 <= k2 < n:
                g[j] += cur[k2]

        i_high = int(g.argmax())
        i_low = int(g.argmin())
        if i_high == i_low:
            # nothing to move – perform a tiny random perturbation
            cur += rng.normal(scale=1e-4, size=n)
            cur = np.maximum(cur, 0.0)
            cur = _project_simplex(cur)
            cur_score = _score_norm1(cur)
            continue

        # try moving all mass from i_high to i_low
        move_full = cur[i_high]
        cand_full = cur.copy()
        cand_full[i_high] -= move_full
        cand_full[i_low] += move_full
        cand_full_score = _score_norm1(cand_full)

        # also try half‑mass move (more conservative)
        move_half = move_full * 0.5
        cand_half = cur.copy()
        cand_half[i_high] -= move_half
        cand_half[i_low] += move_half
        cand_half_score = _score_norm1(cand_half)

        # select best among current, full, half moves
        if cand_full_score < cur_score or cand_half_score < cur_score:
            if cand_full_score <= cand_half_score:
                cur, cur_score = cand_full, cand_full_score
            else:
                cur, cur_score = cand_half, cand_half_score
            no_improve = 0
            if cur_score < best_score:

        else:

        # -------- occasional diversification --------------------------------
        if no_improve > 2500:

                if sym_score < best_score:
                    best, best_score = sym.copy(), sym_score

        if best_score <= target:
            break

    # Return the best vector found (as plain Python floats)
    return [float(x) for x in best]
