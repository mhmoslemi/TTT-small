def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.nonzero(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    w = v - theta
    w[w < 0] = 0.0
    return w
```

def _fast_score(a, n, L):
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * np.max(conv))

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a, n, L):
    """Fast surrogate: 2*n*max(full convolution) for vectors with sum≈1."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * np.max(conv))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # margin for return

    # ------------------------------------------------------------------
    # initialise
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        cur = _project_simplex(cur)
    else:
        n0 = 2000
        cur = rng.random(n0, dtype=np.float64)
        cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two for FFT

    # best-so-far (true score via evaluate_sequence)
    try:
        best_sc = evaluate_sequence(cur.tolist())
    except Exception:
        best_sc = float('inf')
    best_vec = cur.copy()

    cur_sc = _fast_score(cur, n, L)

    # optimisation state
    momentum = np.zeros_like(cur)
    beta = 0.9            # momentum decay
    step0 = 0.2           # base step size
    it = 0
    no_improve = 0

    exact_eval_interval = 5000
    sym_interval = 1000
    restart_thresh = 3000

    while time.time() < deadline:
        it += 1

        # ---------- convolution ----------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = float(conv.max())
        k_max = int(conv.argmax())

        # ---------- gradient of the (hard) max ----------
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(cur)
        g[mask] = 2.0 * cur[idx[mask]]

        # ---------- momentum ----------
        momentum = beta * momentum + (1.0 - beta) * g

        # ---------- step size ----------
        step = step0 / math.sqrt(it)

        # ---------- try a few step scalings ----------
        best_cand = None
        best_cand_sc = float('inf')
        for factor in (0.5, 1.0, 1.5):
            cand = _project_simplex(cur - step * factor * momentum)
            cand_sc = _fast_score(cand, n, L)
            if cand_sc < best_cand_sc:
                best_cand_sc = cand_sc
                best_cand = cand

        # ---------- acceptance ----------
        if best_cand_sc < cur_sc:
            cur, cur_sc = best_cand, best_cand_sc
            no_improve = 0
            if it % exact_eval_interval == 0:
                true_sc = evaluate_sequence(cur.tolist())
                if true_sc < best_sc:
                    best_sc, best_vec = true_sc, cur.copy()
        else:
            no_improve += 1

        # ---------- random redistribution when stuck ----------
        if no_improve > restart_thresh:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _fast_score(tmp, n, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                true_sc = evaluate_sequence(tmp.tolist())
                if true_sc < best_sc:
                    best_sc, best_vec = true_sc, tmp.copy()
            else:
                # occasional full random restart (cheap)
                cur = _project_simplex(rng.random(n))
                cur_sc = _fast_score(cur, n, L)

        # ---------- symmetrisation ----------
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _fast_score(sym, n, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                true_sc = evaluate_sequence(sym.tolist())
                if true_sc < best_sc:
                    best_sc, best_vec = true_sc, sym.copy()

        # ---------- early stop ----------
        if best_sc < 1.45:   # well below historic target
            break

    # final sanity check – ensure the returned vector yields a finite score
    try:
        final_sc = evaluate_sequence(best_vec.tolist())
        if not (math.isfinite(final_sc) and final_sc < 1e9):
            raise ValueError
    except Exception:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(x) for x in best_vec]
