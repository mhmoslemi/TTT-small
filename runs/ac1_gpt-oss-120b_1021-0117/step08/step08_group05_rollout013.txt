import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex {x >=0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _max_conv_and_grad(a, conv_full, n):
    """Return (max value, gradient of max convolution w.r.t. a)."""
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    max_idxs = np.where(conv == max_val)[0]
    grad = np.zeros_like(a)
    ar = np.arange(n)
    for k in max_idxs:
        idx = k - ar
        valid = (idx >= 0) & (idx < n)
        grad[valid] += a[idx[valid]]
    grad /= max(len(max_idxs), 1)
    return max_val, grad

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)  # tiny safety margin

    # ----- initialization -----
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # enforce sum = 1

    n = cur.size
    # FFT length: next power of two >= 2n-1
    L = 1 << ((2 * n - 1).bit_length())

    # Adam‑like moments
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # bookkeeping
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    max_val, _ = _max_conv_and_grad(cur, conv_full, n)
    best_score = 2.0 * n * max_val
    best_vec = cur.copy()
    cur_score = best_score

    it = 0
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1

        # ----- convolution & gradient of current max -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        max_val, grad = _max_conv_and_grad(cur, conv_full, n)

        # ----- Adam update (projected) -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # base learning rate, decays with sqrt(iter)
        lr = 0.08 / math.sqrt(it)
