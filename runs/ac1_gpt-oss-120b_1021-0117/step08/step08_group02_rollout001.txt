import time
import math
import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a, n, L):
    """Score assuming sum(a)=1, i.e. 2*n*max(conv)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # reserve a small head‑room for return and cleanup
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------ #
    # Initialization
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two for FFT speed

    best_sc = _fast_score(best_vec, n, L)
    cur_sc = best_sc

    # algorithm parameters

    min_step = 1e-6
    max_step = 2.0
    no_improve = 0
    sym_interval = 500
    perturb_after = 3000
    target = 1.50274349

    it = 0
    while True:
        if time.time() >= deadline:
            break
        it += 1

        # -------------------------------------------------------------- #
