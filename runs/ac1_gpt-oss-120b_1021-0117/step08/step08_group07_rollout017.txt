import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else css[rho[-1]] / (rho[-1] + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast computation of 2·n·max(conv(a,a)) for a vector with sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    cur_sc = 2.0 * n * conv.max()

    it = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- convolution & max index -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        k_max = int(conv.argmax())
        cur_sc = 2.0 * n * max_val

        # ----- temperature schedule -----
        if temp is None:
            temp = max_val * 0.2
        else:
            temp = max(temp * decay, max_val * min_temp_fac)

        # ----- gradient -----
        if temp > max_val * 0.01:                     # smooth approximation
            w = np.exp((conv - max_val) / temp)
            w_sum = w.sum()
            if w_sum == 0.0 or np.isnan(w_sum):
                w = (conv == max_val).astype(np.float64)
                w_sum = w.sum()
            w /= w_sum
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            grad = 2.0 * grad_full[idx]
        else:                                         # exact sub‑gradient
            shift = n - 1 - k_max
            a_rev = cur[::-1]
            if shift >= 0:
                grad = np.zeros_like(cur)
                grad[: n - shift] = 2.0 * a_rev[shift:]
            else:
                grad = np.zeros_like(cur)
                s = -shift
                grad[s:] = 2.0 * a_rev[: n - s]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (math.sqrt(it) + 1e-8)
        cand = cur - step * (m_hat / (np.sqrt(v_hat) + eps))

        # ----- feasibility projection -----
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()

        # ----- acceptance -----
        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
        else:
            # occasional stochastic move to escape plateaus
            if it % 200 == 0:
                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.2
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()

        # ----- periodic symmetrisation -----
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                m.fill(0.0)
                v.fill(0.0)

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Final local polishing with small Gaussian perturbations
    while time.time() < deadline:
        pert = best_vec + rng.normal(scale=0.003, size=n)
        pert = np.clip(pert, 0.0, None)
        pert = _project_simplex(pert)
        sc = _score(pert)
        if sc < best_sc:
            best_sc = sc
            best_vec = pert.copy()
            if best_sc < target:
                break

    # Return as plain Python floats
    return [float(x) for x in best_vec]
