    start = time.time()

    v = np.zeros_like(cur)

    lr = init_lr
    eps = 1e-8

    temp_factor = 0.5

    min_temp_factor = 1e-6

    it = 0
    while True:

            break
        it += 1

        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        # update temp
        if temp is None:
            temp = max_val * temp_factor
        else:
            temp = max(temp * decay, max_val * min_temp_factor)
        # compute softmax weights

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the simplex {x >= 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_fast(a):
    """Score 2*n*max(conv) for a vector a already on the simplex (sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
