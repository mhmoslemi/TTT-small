def _simplex_project(v):
    """Project onto unit simplex with sum normalization"""
    v = np.asarray(v)
    n = v.size
    sorted_v = np.sort(-v)[::-1]
    css = np.cumsum(sorted_v) - sorted_v[0] * n
    ct = np.minimum(range(n), np.nonzero(css < 1)[0] + 1).min()
