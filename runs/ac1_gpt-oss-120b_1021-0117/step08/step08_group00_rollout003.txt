def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc

        # random mass transfer when stuck
        if no_improve > 2000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] = max(0.0, tmp[i] - delta)
            tmp[j] = tmp[j] + delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_vec(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc

        # early exit if target achieved
        if best_sc < target:
            break

    # ---------- final fine‑tuning (tiny local moves) ----------
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        epsilon = best[i] * rng.random() * 0.05
        cand = best.copy()
        cand[i] = max(0.0, cand[i] - epsilon)
        cand[j] = cand[j] + epsilon
        cand = _project_simplex(cand)
        cand_sc = _score_vec(cand)
        if cand_sc < best_sc:
            best, best_sc = cand, cand_sc
            if best_sc < target:
                break

    # return plain Python list
    return [float(x) for x in best]
