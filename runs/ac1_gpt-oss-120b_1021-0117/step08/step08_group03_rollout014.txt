import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort v descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

def _grad_max(a, conv, max_val, L):
    mask = np.isclose(conv, max_val, atol=1e-12)
    w = mask.astype(np.float64)
    w_rev = w[::-1]
    af = np.fft.rfft(a, n=L)
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    n = a.shape[0]
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # initialization
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    momentum = np.zeros_like(cur)
    beta = 0.9
    step0 = 1.0
    it = 0
    no_improve = 0
    target = 1.50274349
    # thresholds
    perturb_after = 400   # iterations without improvement before perturb
    max_backtrack = 5

    while time.time() < deadline:
        it += 1

        # convolution & max
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # gradient
        grad = _grad_max(cur, conv, max_val, L)

        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size
        base_step = step0 / math.sqrt(it)

        # backtracking line search
        step = base_step
        improved = False
        for _ in range(max_backtrack):
            cand = _project_simplex(cur - step * momentum)
            cand_sc = _score(cand, L)
            if cand_sc < cur_sc:
                cur = cand
                cur_sc = cand_sc
                improved = True
                break
            step *= 0.5

        if improved:
            if cur_sc < best_sc:
                best_vec = cur.copy()
                best_sc = cur_sc
            no_improve = 0
        else:
            no_improve += 1

        # random perturbation when stagnated
        if no_improve >= perturb_after:
            # generate a few perturbations and keep the best
            best_pert_sc = cur_sc
            best_pert = cur
            for _ in range(3):
                # small Gaussian noise then projection
                pert = cur + rng.normal(scale=0.02, size=n)
                pert = np.maximum(pert, 0.0)
                pert = _project_simplex(pert)
                pert_sc = _score(pert, L)
                if pert_sc < best_pert_sc:
                    best_pert_sc = pert_sc
                    best_pert = pert
            if best_pert_sc < cur_sc:
                cur = best_pert
                cur_sc = best_pert_sc
                if cur_sc < best_sc:
                    best_vec = cur.copy()
                    best_sc = cur_sc
                no_improve = 0
            else:
                # reset momentum to explore fresh direction
                momentum = np.zeros_like(cur)
                no_improve = 0

        # occasional symmetrisation
        if it % 1500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if cur_sc < best_sc:
                    best_vec = cur.copy()
                    best_sc = cur_sc

        if best_sc < target:
            break

    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ----- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()

    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    momentum = np.zeros_like(cur)
    beta = 0.9               # momentum decay
    step0 = 0.5              # initial step magnitude
    alpha = 15.0             # soft‑max smoothing for the subgradient
    it = 0
    no_improve = 0
    target = 1.50274349
    perturb_after = 300      # iterations without improvement before a rescue
    max_backtrack = 3

    # ----- main optimisation loop ----------------------------------------
    while time.time() < deadline:
        it += 1

        # FFT of current vector
        a_fft = np.fft.rfft(cur, n=L)

        # convolution and its maximum
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # --- smooth subgradient (soft‑max approximation) -----------------
        exp_w = np.exp(alpha * (conv - max_val))
        w = exp_w / exp_w.sum()

        grad_full = np.fft.irfft(a_fft * w_rev_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
