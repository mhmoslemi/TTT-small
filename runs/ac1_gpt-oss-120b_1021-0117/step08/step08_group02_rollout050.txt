import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(vec):
    """Exact score used by the evaluator (sum normalization handled)."""
    n = vec.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_val / (s * s))
