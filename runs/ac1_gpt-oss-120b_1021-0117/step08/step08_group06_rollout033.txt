    deadline = start + budget_s - 0.2

    # evaluate exact (should equal cur_score)
    best_score = evaluate_sequence(best_vec.tolist())

    # parameters
    step = 0.5   # initial step size
    momentum = np.zeros_like(cur)
    beta = 0.7
    it = 0
    no_improve = 0

    while time.time() < deadline:
        it += 1
        # compute gradient (subgradient of max conv)
        conv = compute_full_conv(cur)  # length 2n-1
        max_val = conv.max()
        max_idx = np.where(conv == max_val)[0]
        grad = np.zeros_like(cur)
        for k in max_idx:

            start_j = max(0, k - (n - 1))
            end_j = min(n - 1, k)
            # segment length = end_j - start_j + 1
            idx_range = slice(start_j, end_j + 1)

            f_indices = k - np.arange(start_j, end_j + 1)

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _conv_max(a, L):
    """Return max of (a*a) convolution and the full convolution array."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max()), conv

def _grad_max(a, conv, max_val):
    """Sub‑gradient of max(conv) w.r.t. a (sum of contributions from all max locations)."""
    n = a.size
    # indices where conv attains the maximal value (with tolerance)
    idx = np.where(np.isclose(conv, max_val, atol=1e-12))[0]
    grad = np.zeros_like(a)
    for k in idx:
        start_j = max(0, k - (n - 1))
        end_j = min(n - 1, k)
        js = np.arange(start_j, end_j + 1)
        grad[start_j:end_j + 1] += 2.0 * a[k - js]
    return grad
