def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    start = time.time()
    deadline = start + max(0., float(budget_s) - 0.5)   # safety margin
    rng = np.random.default_rng(seed)
    # get length
    if "height_sequence_1" in globals():
        n = int(height_sequence_1.shape[0])
    else:
        n = 2500  # default
    # function to initialize vector
    def init_vec():
        if "height_sequence_1" in globals():
            vec = np.asarray(height_sequence_1, dtype=np.float64).copy()
        else:
            vec = rng.random(n, dtype=np.float64)
        vec = np.maximum(vec, 0.0)
        vec = _project_simplex(vec)
        return vec
    best_vec = None
    best_score = np.inf

    # precompute FFT length
    L = 1 << ((2 * n - 1).bit_length())
    # optimizer hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5   # base learning rate
    decay_temp = 0.997
    min_temp_factor = 1e-4

    # outer restarts
    restart = 0
    while time.time() < deadline:
        # maybe limit number of restarts to avoid infinite
        if restart >= 8:
            break
        cur = init_vec()
        # ensure sum not too small
        cur_sum = cur.sum()
        if cur_sum < 0.01:
            cur = cur + (0.01 - cur_sum)/n
        # compute initial score/evaluate
        cur_score = _score(cur)
        # store best
        if cur_score < best_score:
            best_score = cur_score
            best_vec = cur.copy()
        # optimizer state
        m = np.zeros_like(cur)
        v = np.zeros_like(cur)
        temp = None
        t_iter = 0
        no_improve = 0
        # inner loop
        while time.time() < deadline:
            t_iter += 1
            # compute convolution
            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[: 2 * n - 1]
            max_val = conv.max()
            # temperature
            if temp is None:
                temp = max_val * 0.07
            else:
                temp = max(temp * decay_temp, max_val * min_temp_factor)
            # compute softmax weights
            # subtract max for stability
            diff = conv - max_val
            # to avoid overflow, scale by temp, but temp positive
            w_exp = np.exp(diff / temp)
            w_sum = w_exp.sum()
            if w_sum == 0 or np.isnan(w_sum):
                # fallback: uniform over max locations
                mask = (conv == max_val)
                w_exp = mask.astype(np.float64)
                w_sum = w_exp.sum()
            w = w_exp / w_sum
            # gradient
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            grad = 2.0 * grad_full[idx]
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t_iter)
            v_hat = v / (1 - beta2 ** t_iter)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            cand = cur - step
            # projection onto simplex
            cand = _project_simplex(cand)
            cand_score = _score(cand)
            # record best
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
            # accept if improvement else maybe accept with prob
            if cand_score < cur_score:
                cur, cur_score = cand, cand_score
                no_improve = 0
            else:
                no_improve += 1
                # probability decreasing with iteration
                prob = math.exp(-(cand_score - cur_score) / (lr + 1e-12))
                if rng.random() < prob:
                    cur, cur_score = cand, cand_score
                    no_improve = 0

            # mass transfer when stuck
            if no_improve > 800:
                # move mass from highest to lowest gradient indices
                i_max = np.argmax(grad)
                i_min = np.argmin(grad)
                if i_max != i_min:
                    delta = min(cur[i_max] * 0.2, 0.001)
                    tmp = cur.copy()
                    tmp[i_max] -= delta
                    tmp[i_min] += delta
                    tmp = _project_simplex(tmp)
                    tmp_score = _score(tmp)
                    if tmp_score < cur_score:
                        cur, cur_score = tmp, tmp_score
                        no_improve = 0
                        if tmp_score < best_score:
                            best_score = tmp_score
                            best_vec = tmp.copy()
                else:
                    # random perturb
                    i, j = rng.integers(0, n, size=2)
                    if i == j:
                        j = (j + 1) % n
                    delta = cur[i] * rng.random() * 0.4
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_score = _score(tmp)
                    if tmp_score < cur_score:
                        cur, cur_score = tmp, tmp_score
                        no_improve = 0
                        if tmp_score < best_score:
                            best_score = tmp_score
                            best_vec = tmp.copy()
            # occasional random perturbation (smooth)
            if t_iter % 200 == 0:
                pert = cur + rng.normal(scale=0.005, size=n)
                pert = np.clip(pert, 0.0, None)
                pert = _project_simplex(pert)
                pert_score = _score(pert)
                if pert_score < best_score:
                    best_score = pert_score
                    best_vec = pert.copy()
                    if pert_score < cur_score:
                        cur, cur_score = pert, pert_score
                        no_improve = 0

            # early stop if best_score already below target
            # target = 1.50274349
            if best_score < 1.50274349:
                break

            # also check total iteration limit maybe 20000 loops.
            if t_iter >= 20000:
                break

        restart += 1
        # check if best achieved target, break outer
        if best_score < 1.50274349:
            break

    # final fallback: if best_vec still None, create uniform
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    # return as list of floats
    return [float(x) for x in best_vec]

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    # ensure nonnegative? Not required; algorithm automatically handles negatives.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0
    return w

def _score(a):

    max_b = conv.max()
    sum_a = a.sum()

import time, math, numpy as np

def _project_simplex(v):
    """ Euclidean projection of a vector onto the probability simplex (sum=1, non‑negative). """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """ Exact AC1 upper‑bound score for a coefficient vector. """
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = a.sum()
    return float(2.0 * n * max_b / (sum_a * sum_a))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a safety margin before the sandbox timeout
    deadline = start + max(0.0, float(budget_s) - 0.4)

    # determine problem size
    if "height_sequence_1" in globals():
        n = int(height_sequence_1.shape[0])
    else:
        n = 2500                     # fallback length

    L = 1 << ((2 * n - 1).bit_length())   # FFT length (power of two)

    best_vec = None
    best_sc = np.inf

    # ------------------------------------------------------------------
    # 1) Quick random search over a two‑exponential family
    # ------------------------------------------------------------------
    # reserve some time for the more expensive gradient phase
    reserve = 200.0
    param_deadline = deadline - reserve
    if param_deadline > time.time():
        i_arr = np.arange(n)
        trials = 8000
        for _ in range(trials):
            if time.time() >= param_deadline:
                break
            alpha = rng.random()
            # sample (1‑r) on a log‑scale to favour values close to 1
            log_low, log_high = -5, -1          # 1‑r in [1e‑5, 1e‑1]
            one_minus_r1 = 10 ** rng.uniform(log_low, log_high)
            one_minus_r2 = 10 ** rng.uniform(log_low, log_high)
            r1 = 1.0 - one_minus_r1
            r2 = 1.0 - one_minus_r2
            # build raw vector and project onto the simplex
            b = alpha * (r1 ** i_arr) + (1.0 - alpha) * (r2 ** i_arr)
            b = _project_simplex(b)
            sc = _score(b)
            if sc < best_sc:
                best_sc = sc
                best_vec = b.copy()

    # ------------------------------------------------------------------
    # 2) If the random search produced nothing, fall back to a simple init
    # ------------------------------------------------------------------
    if best_vec is None:
        if "height_sequence_1" in globals():
            cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        else:
            cur = rng.random(n, dtype=np.float64)
        cur = _project_simplex(np.maximum(cur, 0.0))
        best_vec = cur.copy()
        best_sc = _score(best_vec)

    # ------------------------------------------------------------------
    # 3) Refine the best candidate with an Adam‑style projected gradient descent
    # ------------------------------------------------------------------
    cur = best_vec.copy()
    cur_sc = _score(cur)

    # Adam hyper‑parameters (chosen for stability on this problem)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr_base = 0.3

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    temp = None                       # temperature for soft‑max smoothing
    decay_temp = 0.996
    min_temp_fac = 1e-4

    no_improve = 0
    it = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1

        # ----- convolution and smooth‑max weight computation -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay_temp, max_val * min_temp_fac)

        diff = conv - max_val
        w_exp = np.exp(diff / temp)
        w_sum = w_exp.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            mask = (conv == max_val)
            w_exp = mask.astype(np.float64)
            w_sum = w_exp.sum()
        w = w_exp / w_sum

        w_rev = w[::-1]
