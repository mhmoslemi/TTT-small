import time, math, numpy as np
from numpy.fft import rfft, irfft

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_vec(v, L):
    """Return the AC1 upper bound for a vector already normalised to sum≈1."""
    a_fft = rfft(v, n=L)
    conv = irfft(a_fft * a_fft, n=L)[: 2 * v.shape[0] - 1]
    return float(2 * v.shape[0] * conv.max())

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v_arr = beta2 * v_arr + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v_arr / (1 - beta2 ** it)
        step = base_lr / (np.sqrt(v_hat) + eps)

        cand = cur - step * m_hat

        # ---- enforce feasibility -------------------------------------------
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)
        # enforce symmetry (helps reduce peaks)
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        cand_sc = _score_vec(cand, L)

        # ---- accept or reject ------------------------------------------------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            # occasional uphill move (simulated‑annealing flavour)
            prob = math.exp(-(cand_sc - cur_sc) / (base_lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # ---- stagnation handling --------------------------------------------
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp = (tmp + tmp[::-1]) * 0.5
            tmp = _project_simplex(tmp)
            tmp_sc = _score_vec(tmp, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

        # ---- occasional sparsification ---------------------------------------
        if it % 2000 == 0:
            thresh = np.quantile(cur, 0.25)     # drop bottom 25 %
            tmp = cur.copy()
            tmp[tmp < thresh] = 0.0
            tmp = _project_simplex(tmp)
            tmp = (tmp + tmp[::-1]) * 0.5
            tmp = _project_simplex(tmp)
            tmp_sc = _score_vec(tmp, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

        # ---- periodic explicit symmetrisation --------------------------------
        if it % 1500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # ---- early exit ------------------------------------------------------
        if best_sc < target:
            break

    # return the best vector found (as plain Python floats)
    return [float(x) for x in best_vec]
