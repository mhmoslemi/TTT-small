import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_vec(a, n, L):
    """Exact AC1 upper bound for a non‑negative vector a (sum may be arbitrary)."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_val / (sum_a * sum_a))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a list of non‑negative step heights with low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # enforce sum = 1, non‑negative

    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # power‑of‑two FFT length

    # ---- Adam optimizer parameters ---------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.02

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    best_vec = cur.copy()
    best_sc = _score_vec(best_vec, n, L)
    cur_sc = best_sc

    # ---- auxiliary settings ---------------------------------------------
    no_improve = 0
    target = 1.50274349
    it = 0
    temp = None
    decay = 0.9995
    min_temp_factor = 1e-5

    # ---- main optimisation loop -----------------------------------------
    while time.time() < deadline:
        it += 1

        # current convolution via FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # temperature schedule for soft‑max smoothing
        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # soft‑max weights (approaches exact max as temp ↓)
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of weighted max (via FFT)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)

        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]
