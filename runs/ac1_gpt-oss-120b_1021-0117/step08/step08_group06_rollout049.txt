import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _obj_and_grad(a, L=None):
    n = a.shape[0]
    if L is None:
        L = 1 << ((2*n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[:2*n-1]
    M = conv_full.max()
    # compute gradient of M
    max_idxs = np.where(np.isclose(conv_full, M, atol=1e-12))[0]
    gradM = np.zeros_like(a)
    for k in max_idxs:
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        i = np.arange(i_start, i_end+1)
        j = k - i
        gradM[i] += 2.0 * a[j]
    gradM /= len(max_idxs)  # average if multiple maxima
    S = a.sum()
    factor = 2.0 * n / (S**2)
    grad = factor * gradM - (4.0 * n * M / (S**3)) * np.ones_like(a)
    obj = 2.0 * n * M / (S**2)
    return obj, grad

    v = np.asarray(v, dtype=np.float64)

    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    M = conv_full.max()
    # indices where the maximum is attained (within a tiny tolerance)
    max_idxs = np.where(np.isclose(conv_full, M, atol=1e-12))[0]
    gradM = np.zeros_like(a)
    for k in max_idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        gradM[i] += 2.0 * a[j]
    if max_idxs.size > 1:
        gradM /= max_idxs.size
    S = a.sum()
    obj = 2.0 * n * M / (S ** 2)
    factor = 2.0 * n / (S ** 2)
    grad = factor * gradM - (4.0 * n * M / (S ** 3)) * np.ones_like(a)
    return obj, grad

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # --- initialise ---------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())          # FFT size (next power of two)

    # best-so-far tracking (exact evaluation)
    best_vec = cur.copy()
    best_sc = evaluate_sequence(best_vec.tolist())

    # Adam optimizer state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.05                     # base learning rate
    it = 0
    no_improve = 0
    exact_interval = 300           # how often to call evaluate_sequence

    while time.time() < deadline:
        it += 1

        # objective and gradient (exact max, not a soft‑max surrogate)
        obj, grad = _obj_and_grad(cur, L)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        cand = cur - step

        # feasibility: simplex projection
        cand = _project_simplex(cand)

        # enforce symmetry (helps the convolution structure)
        cand = (cand + cand[::-1]) * 0.5
        cand = _project_simplex(cand)

        # accept the candidate
        cur = cand

        # occasional exact evaluation for global best
        if it % exact_interval == 0:
            sc = evaluate_sequence(cur.tolist())
            if sc < best_sc:
                best_sc, best_vec = sc, cur.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # inexpensive surrogate improvement check (optional)
            if obj < best_sc:
                best_sc, best_vec = obj, cur.copy()
                no_improve = 0
            else:
                no_improve += 1

        # random perturbation if stuck
        if no_improve > 4000:
            noise = rng.normal(scale=0.03, size=n)
            pert = cur + noise
            pert = np.maximum(pert, 0.0)
            pert = _project_simplex(pert)
            cur = pert
            no_improve = 0

        # early exit if we already beat a comfortable margin
        if best_sc < 1.48:
            break

    # final safety check
    try:
        final_sc = evaluate_sequence(best_vec.tolist())
        if not (math.isfinite(final_sc) and final_sc < 1e9):
            raise ValueError
    except Exception:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(x) for x in best_vec]
