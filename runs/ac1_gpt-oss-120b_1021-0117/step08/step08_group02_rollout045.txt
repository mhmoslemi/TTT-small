import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast 2*n*max(convolution) for a vector with sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        else:
            no_improve += 1
            prob = math.exp(-(cand_sc - cur_sc) / (temp + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # --------------------------------------------------------------
        # random redistribution when stuck
        if no_improve > 2500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp = (tmp + tmp[::-1]) * 0.5
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

        # --------------------------------------------------------------
        # occasional full restart around the best point
        if it % 4000 == 0:
            new = best_vec + rng.normal(scale=0.02, size=n)
            new = np.maximum(new, 0.0)
            new = _project_simplex(new)
            new = (new + new[::-1]) * 0.5
            new = _project_simplex(new)
            cur, cur_sc = new, _score(new)
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = beta2_pow = 1.0
            temp = None
            no_improve = 0

        # --------------------------------------------------------------
        if best_sc < target:
            break

    return [float(x) for x in best_vec]
