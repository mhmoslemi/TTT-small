import time, math, numpy as np

# --------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex (sum = 1)
def _project_simplex(v):
    """Project a vector onto the simplex {x >= 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - css / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative coefficients whose AC1 upper bound
    (as computed by the hidden evaluator) is as small as possible.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ----- problem size & initial vector ---------------------------------
    if "height_sequence_1" in globals():
        init_full = np.asarray(height_sequence_1, dtype=np.float64)
        n = init_full.shape[0]
    else:
        n = 2000
        init_full = rng.random(n, dtype=np.float64)

    # project to simplex (sum = 1) – the scale does not affect the score
    cur = _project_simplex(np.maximum(init_full, 0.0))
    best = cur.copy()

    # Pre‑compute FFT length (next power of two >= 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    # ----- scoring utilities ---------------------------------------------
    def _score_from_max(max_conv):
        """Score = 2 * n * max(conv)  (sum is 1)."""
        return float(2.0 * n * max_conv)

    # initial score
    Af = np.fft.rfft(best, n=L)
    conv_full = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    best_sc = _score_from_max(conv_full.max())

    # also keep the uniform vector as a cheap baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    Au = np.fft.rfft(uniform, n=L)
    conv_u = np.fft.irfft(Au * Au, n=L)[: 2 * n - 1]
    uni_sc = _score_from_max(conv_u.max())
    if uni_sc < best_sc:
        best_sc, best = uni_sc, uniform.copy()

    # ----- optimisation hyper‑parameters ---------------------------------
    target = 1.50274349          # desired benchmark
    beta1, beta2 = 0.9, 0.999    # Adam moments
    eps = 1e-12
    lr0 = 0.12                   # base learning‑rate
    temp_decay = 0.996
    min_temp_factor = 1e-4

    restart = 0
    # --------------------------------------------------------------------
    while time.time() < deadline:
        # ----- start a new run (use best or a fresh random start) -----
        if restart == 0:
            cur = best.copy()
        else:
            cur = rng.exponential(scale=1.0, size=n)
            cur = _project_simplex(cur)

        # Adam state
        m = np.zeros_like(cur)
        v = np.zeros_like(cur)
        it = 0
        no_improve = 0
        temp = None

        # ----- inner optimisation loop ---------------------------------
        while time.time() < deadline:
            it += 1

            # current convolution
            Af = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
            max_conv = conv_full.max()
            cur_sc = _score_from_max(max_conv)

            # keep global best
            if cur_sc < best_sc:
                best_sc, best = cur_sc, cur.copy()
                if best_sc < target:
                    break

            # temperature for smoothed max (soft‑max weighting)
            if temp is None:
                temp = max_conv * 0.07
            else:
                temp = max(temp * temp_decay, max_conv * min_temp_factor)

            # soft‑max weights over all lags
            w = np.exp((conv_full - max_conv) / (temp + 1e-20))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = (conv_full == max_conv).astype(np.float64)
                w_sum = w.sum()
            w /= w_sum

            # gradient of the smoothed max (FFT‑based)
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(Af * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            grad = 2.0 * grad_full[idx]

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

            # candidate step and projection
            cand = _project_simplex(cur - step)

            # evaluate candidate (reuse FFT)
            Af_c = np.fft.rfft(cand, n=L)
            conv_c_full = np.fft.irfft(Af_c * Af_c, n=L)[: 2 * n - 1]
            cand_sc = _score_from_max(conv_c_full.max())

            # acceptance rule (strict descent with occasional uphill moves)
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                no_improve = 0
            else:
                no_improve += 1
                prob = math.exp(-(cand_sc - cur_sc) / (lr0 + 1e-12))
                if rng.random() < prob:
                    cur, cur_sc = cand, cand_sc

            # ----- occasional random mass redistribution when stuck -----
            if no_improve > 3000:
                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.4
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                Af_t = np.fft.rfft(tmp, n=L)
                conv_t = np.fft.irfft(Af_t * Af_t, n=L)[: 2 * n - 1]
                tmp_sc = _score_from_max(conv_t.max())
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best = tmp_sc, tmp.copy()

            # ----- occasional symmetrisation --------------------------------
            if it % 800 == 0:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                Af_s = np.fft.rfft(sym, n=L)
                conv_s = np.fft.irfft(Af_s * Af_s, n=L)[: 2 * n - 1]
                sym_sc = _score_from_max(conv_s.max())
                if sym_sc < cur_sc:
                    cur, cur_sc = sym, sym_sc
                    if sym_sc < best_sc:
                        best_sc, best = sym_sc, sym.copy()

            # ----- termination checks ---------------------------------------
            if best_sc < target:
                break
            if time.time() >= deadline:
                break

        # end of a restart
        restart += 1
        if best_sc < target:
            break

    # --------------------------------------------------------------------
    # Final small‑step hill climbing (cheap)
    cur = best.copy()
    cur_sc = best_sc
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        eps = cur[i] * rng.random() * 0.02
        if eps <= 0.0:
            continue
        tmp = cur.copy()
        tmp[i] -= eps
        tmp[j] += eps
        tmp = _project_simplex(tmp)
        Af_tmp = np.fft.rfft(tmp, n=L)
        conv_tmp = np.fft.irfft(Af_tmp * Af_tmp, n=L)[: 2 * n - 1]
        tmp_sc = _score_from_max(conv_tmp.max())
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_sc, best = tmp_sc, tmp.copy()
            if rng.random() < 0.1:          # occasional extra symmetrisation
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                Af_sym = np.fft.rfft(sym, n=L)
                conv_sym = np.fft.irfft(Af_sym * Af_sym, n=L)[: 2 * n - 1]
                sym_sc = _score_from_max(conv_sym.max())
                if sym_sc < cur_sc:
                    cur, cur_sc = sym, sym_sc
                    if sym_sc < best_sc:
                        best_sc, best = sym_sc, sym.copy()
        else:
            if rng.random() < 0.001:        # tiny chance to escape flat regions
                cur, cur_sc = tmp, tmp_sc

    # --------------------------------------------------------------------
    # Return the best found coefficient vector as plain Python floats
    return [float(x) for x in best]
