import time, math, numpy as np

# ------------------------------------------------------------
# basic utilities
def _project_simplex(v):
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_and_max(a):
    """Return (score, max_convolution) for a non‑negative vector a."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()
    sum_a = a.sum()
    if sum_a <= 0:
        return float("inf"), float("inf")
    score = float(2.0 * n * max_val / (sum_a ** 2))
    return score, float(max_val)

    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    lr, eps = 0.4, 1e-8                # a bit larger lr than the reference

    # -------------------- temperature for soft‑max --------------------
    temp = None
    decay = 0.9995                     # slower decay → smoother gradients longer
    min_temp_factor = 1e-10

    no_improve = 0
    it = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1

        # ---- convolution and max ----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ---- temperature handling ----
        if temp is None:
            temp = max_val * 0.1
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # ---- weight vector (soft‑max or exact mask) ----
        if temp < 1e-12 * max_val:                     # switch to exact sub‑gradient
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.ones_like(conv)
                w_sum = w.size
            w = w / w_sum
        else:
            w = np.exp((conv - max_val) / temp)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = (conv == max_val).astype(np.float64)
                w_sum = w.sum()
            w = w / w_sum

        # ---- gradient (via FFT) ----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ---- Adam step ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur - step_vec)
        cand_score, _ = _score_and_max(cand)

        # ---- accept if better, else occasional simulated‑annealing move ----
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
        else:
            no_improve += 1
            prob = math.exp(-(cand_score - cur_score) / (temp + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score
                no_improve = 0

        # ---- keep global best ----
        if cand_score < best_score:
            best_vec, best_score = cand.copy(), cand_score

        # ---- random redistribution when stuck ----
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_score, _ = _score_and_max(tmp)
            if tmp_score < cur_score:
                cur, cur_score = tmp, tmp_score
                no_improve = 0
                if tmp_score < best_score:
                    best_vec, best_score = tmp.copy(), tmp_score

        # ---- symmetrisation periodically ----
        if it % 1200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score, _ = _score_and_max(sym)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                no_improve = 0
                if sym_score < best_score:
                    best_vec, best_score = sym.copy(), sym_score

        # ---- early exit if target achieved ----
        if best_score < target:
            break

    # ------------------------------------------------------------------------
    # final safeguard: if time permits, perform a tiny LP‑based polishing step
    try:
        import cvxpy as cp

        # tiny LP loop, at most 5 iterations and respecting the remaining time
        lp_deadline = deadline - 0.05
        t = 0.2
        while time.time() < lp_deadline:
            # current bound M
            M = np.fft.irfft(np.fft.rfft(cur, n=L) ** 2, n=L)[: 2 * n - 1].max()

            # LP: maximise sum(g)  s.t. conv(cur,g) <= M , g >= 0
            f_param = cp.Parameter(n, value=cur)
            g = cp.Variable(n, nonneg=True)
            conv_expr = cp.conv(f_param, g)          # length 2n‑1
            prob = cp.Problem(cp.Maximize(cp.sum(g)), [conv_expr <= M])
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
            if g.value is None:
                break
            g0 = np.maximum(g.value, 0.0)
            s = g0.sum()
            if s == 0.0:
                break
            g_scaled = g0 / s                         # keep sum = 1
            cur = (1.0 - t) * cur + t * g_scaled
            cur = _project_simplex(cur)
            cur_score, _ = _score_and_max(cur)
            if cur_score < best_score:
                best_vec, best_score = cur.copy(), cur_score
            if best_score < target:
                break
    except Exception:
        # CVXPY unavailable or failed – ignore and return current best
        pass

    # ------------------------------------------------------------------------
    return [float(x) for x in best_vec]
