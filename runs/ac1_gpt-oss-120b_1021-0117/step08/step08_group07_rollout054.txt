def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # initialize
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    # enforce nonnegativity
    cur = np.maximum(cur, 0.0)
    # project to simplex sum=1
    cur = _project_simplex(cur)

    n = cur.shape[0]
    L = 1 << ((2*n - 1).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5  # base learning rate

    it = 0
    no_improve = 0

    # main loop
    while time.time() < deadline:
        it += 1

        # compute convolution and max index
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        k_star = int(conv.argmax())
        # gradient of max conv
        idx = k_star - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(cur)
        grad[mask] = 2.0 * cur[idx[mask]]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr0 / (np.sqrt(v_hat) + eps) * m_hat

        # move opposite direction (descend) as we are minimizing max conv => gradient points direction of increase
        cand = cur - step
        # ensure nonnegative & project onto simplex
        cand = _project_simplex(np.maximum(cand, 0.0))

        cand_sc = _score(cand)

        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand.copy()
        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
        else:
            no_improve += 1

        # occasional random perturbation if stuck
        if no_improve > 1000:
            # random shuffle mass between two indices
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur = tmp
                cur_sc = tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
            # also reset Adam moments to avoid stale direction
            m.fill(0)
            v.fill(0)

        # occasional symmetrization half-way
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()

        # early exit if target reached

            break

    # final small random perturbation local search

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_vec(a):
    """Exact score used by the evaluator (2·n·max(conv)/sum(a)^2)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return float(2.0 * n * max_val / (s * s))
