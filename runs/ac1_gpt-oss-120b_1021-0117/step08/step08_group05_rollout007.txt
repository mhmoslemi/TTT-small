def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, budget_s - 0.3)

    # initialize
    if 'height_sequence_1' in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        init = rng.random(2000, dtype=np.float64)
    init = np.maximum(init, 0.0)
    cur = _project_simplex(init)

    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    idx = (2 * n - 2) - np.arange(n)

    # Adam hyperparameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.03  # smaller
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    t = 0

    # track best
    best = cur.copy()
    best_sc = _score(cur, n)
    # separate from cur_sc
    cur_sc = best_sc

    no_improve = 0
    # Temperature schedule variables
    T = None

    # loop
    while time.time() < deadline:
        t += 1
        # compute conv via FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()

        # update temperature
        if T is None:
            T = max_conv * 0.08
        else:
            T = max(T * 0.997, max_conv * 1e-4)

        # compute softmax weights
        # For numerical stability, subtract max first
        w = np.exp((conv - max_conv) / T)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_conv).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr0 / (np.sqrt(t)) * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        cand = _project_simplex(cand)

        cand_sc = _score(cand, n)

        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc = cand_sc
                best = cand.copy()
        else:
            no_improve += 1
            # accept uphill with probability based on T (annealing)
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (T + 1e-12)):
                cur = cand
                cur_sc = cand_sc

        # periodic random perturb when stuck
        if no_improve >= 3000:
            # reinitialize from best + noise
            noise = rng.normal(scale=0.02, size=n)
            cur = best + noise
            cur = np.maximum(cur, 0.0)
            cur = _project_simplex(cur)
            cur_sc = _score(cur, n)
            no_improve = 0

        # occasional symmetrization
        if t % 2000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, n)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best = sym.copy()

        # early exit
        if best_sc < 1.50274349:
            break

    return [float(x) for x in best]

import time, math, numpy as np

# ---------- helpers ----------
def _project_simplex(v):
    """Euclidean projection of vector v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_vec(a, n, L):
    """Exact score 2*n*max(conv(a,a)) when sum(a)=1 (i.e. denominator =1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
