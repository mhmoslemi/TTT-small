def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a non‑negative vector onto the simplex (sum=1)."""
    v = np.asarray(v, dtype=np.float64)

    css = np.cumsum(u) - 1.0

    return w

                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()

            # ---- time check -------------------------------------------------
            if time.time() >= deadline - 0.2:
                break

        # end of inner loop
        restart_cnt += 1
        if best_sc < target_score:
            break

    # -----------------------------------------------------------------------
    # Fallback: uniform distribution if everything failed
    # -----------------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    # Convert to plain Python floats for the harness
    return [float(x) for x in best_vec]
```
