import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast evaluation of 2·n·max(conv(a,a)) when sum(a)=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    # ------------------------------------------------------------------
    # initialise – use the provided construction if present
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()

        cur = rng.random(2500, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)        # forces sum = 1

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = cur.copy()

    momentum = np.zeros_like(cur)
    beta = 0.92                # momentum decay

    min_temp_fac = 1e-5
    target = 1.50274349

    # ------------------------------------------------------------------

        # ----- convolution and temperature --------------------------------

        max_val = conv.max()
