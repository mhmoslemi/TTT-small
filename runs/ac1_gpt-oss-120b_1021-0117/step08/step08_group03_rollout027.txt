import time, math, numpy as np
import cvxpy as cp

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L=None):
    """Score = 2*n*max(conv(a,a)) for a already on the simplex."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _solve_lp(cur, bound, t_mix=0.2):
    """Solve  max sum(g)  s.t. conv(cur,g) <= bound, g>=0. Returns a mixed
    vector (already simplex‑normalised) or None."""
    n = cur.shape[0]
    g = cp.Variable(n, nonneg=True)
    constraints = [cp.conv(cp.Constant(cur), g) <= bound]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=3000,
                   abstol=1e-8, reltol=1e-6, feastol=1e-8)
    except Exception:
        return None
    if g.value is None:
        return None
    g0 = np.maximum(g.value, 0.0)
    s = g0.sum()
    if s <= 0:
        return None
    g_norm = g0 / s
    return (1.0 - t_mix) * cur + t_mix * g_norm

    lam = 5.0          # start lambda
    lam_max = 200.0    # eventual lambda
    lam_factor = 1.001  # multiplicative growth per iteration

    it = 0
    no_improve = 0
    target = 1.50274349

    # ---- main optimisation loop ----------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- convolution & smooth‑max weights -------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        conv_max = conv.max()

        # smooth approximation of max
        scaled = lam * (conv - conv_max)          # shift for stability
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            # fallback to binary mask
            w = (np.isclose(conv, conv_max, atol=1e-12)).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient of smooth‑max ----------------------------------
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_rev_fft, n=L)
        idx_sel = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx_sel]

        # ----- Adam update -------------------------------------------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = - lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur + step)
        cand_sc = _score(cand, L)

        # accept if improvement
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
                no_improve = 0
        else:
            # one backtrack attempt
            cand2 = _project_simplex(cur + 0.5 * step)
            cand2_sc = _score(cand2, L)
            if cand2_sc < cur_sc:
                cur, cur_sc = cand2, cand2_sc
                if cand2_sc < best_sc:
                    best_vec, best_sc = cand2.copy(), cand2_sc
                no_improve = 0
            else:
                no_improve += 1

        # ----- increase smoothing parameter ----------------------------
        if lam < lam_max:
            lam = min(lam_max, lam * lam_factor)

        # ----- stagnation handling ------------------------------------
        if no_improve > 1500:
            # random local perturbation
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.2
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0
            else:
                # occasional LP refinement
                lp_vec = _solve_lp(cur, conv_max, t_mix=0.2)
                if lp_vec is not None:
                    lp_sc = _score(lp_vec, L)
                    if lp_sc < cur_sc:
                        cur, cur_sc = lp_vec, lp_sc
                        if lp_sc < best_sc:
                            best_vec, best_sc = lp_vec.copy(), lp_sc
                        no_improve = 0

        # ----- occasional symmetrisation -------------------------------
        if it % 2000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # ----- early exit ------------------------------------------------
        if best_sc < target:
            break

    # Return the best vector found as plain Python floats
    return [float(x) for x in best_vec]
