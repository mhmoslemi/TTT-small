def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]

# ---------- Helper: projection onto the probability simplex ----------
def _project_simplex(v):
    """Return the Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

# ---------- Helper: fast evaluation of max(conv(a,a)) for sum(a)=1 ----------
def _max_convolution_fft(a, L=None):
    """Maximum value of the linear convolution a*a (sum(a)=1)."""
    n = a.size
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(conv.max())

# ---------- Main entry point ----------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a small safety margin for finalisation
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # random start if no parent is present
        cur = rng.random(2500, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)               # enforce sum = 1

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())       # power‑of‑two FFT length
    idx = (2 * n - 2) - np.arange(n)          # index map for gradient

    # ---- Adam state -------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    alpha0 = 0.08                               # base learning rate
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    # ---- bookkeeping -------------------------------------------------
    best = cur.copy()
    max_conv = _max_convolution_fft(best, L)
    best_score = 2.0 * n * max_conv               # current objective value
    cur_score = best_score

    it = 0
    no_improve = 0
    T = None                                      # temperature for soft‑max
    target = 1.50274349

    # ---- optimisation loop -----------------------------------------
    while time.time() < deadline:
        it += 1

        # convolution of current vector
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()

        # temperature schedule (starts relatively high, then decays)
        if T is None:
            T = max_conv * 0.07
        else:
            T = max(T * 0.998, max_conv * 1e-4)

        # soft‑max weights approximating the max‑operator
        w = np.exp((conv - max_conv) / T)
        w_sum = w.sum()
        if w_sum == 0.0:                     # fall back to exact indicator
            w = (conv == max_conv).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of the smoothed max (via convolution with reversed weights)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (alpha0 / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + epsilon)

        cand = cur - step
        cand = _project_simplex(cand)

        # evaluate candidate
        cand_max = _max_convolution_fft(cand, L)
        cand_score = 2.0 * n * cand_max

        # acceptance logic
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
        else:
            no_improve += 1
            # occasional uphill move (simulated annealing)
            if rng.random() < math.exp(-(cand_score - cur_score) / (T + 1e-12)):
                cur, cur_score = cand, cand_score

        # random redistribution when stuck
        if no_improve > 2500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_max = _max_convolution_fft(tmp, L)
            tmp_score = 2.0 * n * tmp_max
            if tmp_score < cur_score:
                cur, cur_score = tmp, tmp_score
                no_improve = 0
                if tmp_score < best_score:
                    best, best_score = tmp.copy(), tmp_score

        # periodic symmetrisation (helps for many problems)
        if it % 1000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_max = _max_convolution_fft(sym, L)
            sym_score = 2.0 * n * sym_max
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                if sym_score < best_score:
                    best, best_score = sym.copy(), sym_score

        # early exit if target reached
        if best_score < target:
            break

    # ---- return best found -------------------------------------------------
    return [float(x) for x in best]
