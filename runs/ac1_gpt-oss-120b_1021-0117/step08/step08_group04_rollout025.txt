    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0.0] = 0.0
    return w

def _score_norm1(a):
    """Score for a vector with sum = 1: 2 * n * max(convolution(a, a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a small safety margin for returning the result
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())
    idx = (2 * n - 2) - np.arange(n)

    # Adam optimizer state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.12
    max_step = 1.0

    # temperature for the weighted‑gradient trick
    temp = None
    temp_decay = 0.999
    min_temp_factor = 1e-6

    best = cur.copy()
    best_score = _score_norm1(best)
    cur_score = best_score

    it = 0
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1

        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # --- temperature handling ---------------------------------------
        if temp is None:
            temp = max_val * 0.1
        else:
            temp = max(temp * temp_decay, max_val * min_temp_factor)

        # --- weighted gradient -----------------------------------------
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w = w / w_sum

        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # --- Adam update -----------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / math.sqrt(it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        step_norm = np.linalg.norm(step)
        if step_norm > max_step:
            step = step * (max_step / step_norm)

        cand = _project_simplex(cur - step)
        cand_score = _score_norm1(cand)

        # --- acceptance -------------------------------------------------
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / (temp + 1e-12)):
            cur = cand
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            no_improve += 1

        # --- random redistribution when stagnating --------------------
        if no_improve > 2000:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = cur[i] * rng.random() * 0.5
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score = _score_norm1(tmp)
                if tmp_score < cur_score:
                    cur = tmp
                    cur_score = tmp_score
                    no_improve = 0
                    if tmp_score < best_score:
                        best = tmp.copy()
                        best_score = tmp_score

        # --- occasional symmetrisation ---------------------------------
        if it % 800 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score_norm1(sym)
            if sym_score < cur_score:
                cur = sym
                cur_score = sym_score
                no_improve = 0
                if sym_score < best_score:
                    best = sym.copy()
                    best_score = sym_score

        # --- restart with noise if really stuck -------------------------
        if no_improve > 6000:
            cur = best + rng.normal(scale=0.01, size=n)
            cur = _project_simplex(np.maximum(cur, 0.0))
            cur_score = _score_norm1(cur)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        if best_score <= target:
            break

    return [float(x) for x in best]
