def evaluate_sequence(sequence: list) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))

import time, math, numpy as np

# ----------------------------------------------------------------------
# helper utilities (simple, fast, no large allocations)
# ----------------------------------------------------------------------
def _project_simplex(v):
    """Project a vector onto the probability simplex (sum == 1, v >= 0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0.0] = 0.0
    return w

def _score_norm1(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    Return a non‑negative step‑function height vector whose AC1 upper bound
    (as evaluated by `evaluate_sequence`) is very small.
    The construction is a decreasing geometric sequence whose parameters are
    tuned analytically for the given length.
    """
    start = time.time()
    # a small safety margin – we must return before the sandbox timeout
    deadline = start + max(0.0, float(budget_s) - 0.15)

    # ------------------------------------------------------------------
    # determine length from the provided parent (if any)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

        n = 2500

    #   r = exp(-log(n)/n)  ->  r**n = 1/n
    # This balances the peak at index 0 with the central overlap,
    # yielding a very small maximal convolution value.
    # ------------------------------------------------------------------
    log_r = -math.log(n) / n
    r = math.exp(log_r)

    # generate the geometric vector a_i = r**i, then normalise to sum = 1
    idx = np.arange(n, dtype=np.float64)
    powers = np.exp(idx * log_r)            # equivalent to r**idx but more stable
    a = powers / powers.sum()               # now sum(a) == 1 and a >= 0

    # ------------------------------------------------------------------
    # (optional) a very short refinement using projected gradient descent.
    # This step is cheap and may shave a few percent off the already tiny
    # bound without ever risking a timeout.
    # ------------------------------------------------------------------
    max_refine_iters = 500
    it = 0
    # Adam‑style state for a few iterations – cheap and memory‑light
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-9
    lr = 0.1
    while it < max_refine_iters and time.time() < deadline:
        it += 1
        # convolution via FFT
        L = 1 << ((2 * n - 1).bit_length())
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # soft‑max weighting of the convolution values (temperature = a fraction)
        temp = max_val * 0.05
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of the weighted max‑convolution
        w_rev = w[::-1]
        grad_full = np.fft.irfft(a_fft * np.fft.rfft(w_rev, n=L), n=L)
        idx_back = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx_back]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        max_step = 0.5

    # ------------------------------------------------------------------
    a = np.maximum(a, 0.0)

    return [float(x) for x in a]
