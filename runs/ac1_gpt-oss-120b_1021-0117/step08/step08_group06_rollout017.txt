def _gradient_submax(a, conv, n):
    max_val = conv.max()
    # tolerance: relative difference very small; maybe using np.isclose
    # but conv is float; we use absolute tolerance.
    idxs = np.where(conv >= max_val - 1e-12)[0]
    offsets = {abs(idx - (n-1)) for idx in idxs}
    grad = np.zeros_like(a)
    for off in offsets:
        if off == 0:
            grad += 2.0 * a
        else:
            grad[off:] += a[:-off]
            grad[:-off] += a[off:]
    return grad

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    rho = rho_arr[-1] if rho_arr.size else 0
    theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _fast_score(a, n, L):
    """Fast surrogate for the true objective (sum of a is 1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _gradient_submax(a, conv, n):
    """Subgradient of max_conv at a using the active set of maxima."""
    max_val = conv.max()
    # indices where conv is essentially maximal
    idxs = np.where(conv >= max_val - 1e-12)[0]
    offsets = {abs(idx - (n - 1)) for idx in idxs}
    grad = np.zeros_like(a)
    for off in offsets:
        if off == 0:
            grad += 2.0 * a
        else:
            grad[off:] += a[:-off]      # a_{i-off}
            grad[:-off] += a[off:]      # a_{i+off}
    return grad
