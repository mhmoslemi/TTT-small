import time
import math
import numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Fast 2*n*max(conv) assuming sum(a)=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # power of two >= 2n‑1 (also >= 3n‑2)

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    lr, eps = 0.2, 1e-8

    # temperature for soft‑max (used only while max‑mask is empty)
    temp = None
    decay = 0.998
    min_temp_factor = 1e-7

    target = 1.50274349
    it = 0

    while True:
        if time.time() >= deadline:
            break
        it += 1

        # --------------------------------------------------------------
        # convolution a*a (FFT)
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # --------------------------------------------------------------
        # temperature for smooth max (fallback)
        if temp is None:
            temp = max_val * 0.05
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # --------------------------------------------------------------
        # weight vector w: exact max mask, fallback to soft‑max
        eps_mask = 1e-12 * max_val
        mask = conv >= max_val - eps_mask
        cnt = mask.sum()
        if cnt == 0:
            w = np.exp((conv - max_val) / temp)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= w_sum
        else:
            w = mask.astype(np.float64) / cnt

        # --------------------------------------------------------------
        # gradient of weighted max (cross‑correlation)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # --------------------------------------------------------------
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur - step_vec)
        cand_sc = _score(cand)

        # --------------------------------------------------------------
        # accept if improvement, else simulated‑annealing step
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (temp + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # --------------------------------------------------------------
        # occasional random local redistribution when progress stalls
        if it % 3000 == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.25
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc

        # --------------------------------------------------------------
        # occasional symmetrisation (helps many instances)
        if it % 1500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < best_sc:
                best_sc, best_vec = sym_sc, sym.copy()
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc

        # --------------------------------------------------------------
        # early exit if target reached
        if best_sc < target:
            break

    # ensure a valid list is returned
    return [float(x) for x in best_vec]
