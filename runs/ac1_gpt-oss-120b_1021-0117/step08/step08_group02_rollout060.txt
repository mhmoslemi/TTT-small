def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        v = v.ravel()
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2) # leave headroom.

    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64) # fallback
    cur = np.maximum(cur, 0.0)

    # project onto simplex (sum=1)
    cur = _project_simplex(cur)

    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    # Pre-allocate momentum
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    beta1_pow = 1.0
    beta2_pow = 1.0
    eps = 1e-8
    lr = 0.2  # initial learning rate

    # For schedule
    decay_lr = 0.9995

    # compute initial score
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    cur_sc = 2.0 * n * max_val
    best_vec = cur.copy()
    best_sc = cur_sc

    it = 0
    no_improve = 0
    # target bound
    target = 1.50274349

    while time.time() < deadline:
        it += 1

        # recompute conv and gradient
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        # subgradient mask based on tolerance
        tol = max_val * 1e-12 + 1e-15
        mask = (conv >= max_val - tol).astype(np.float64)
        w_sum = mask.sum()
        if w_sum == 0:
            # just in case
            mask = (conv == max_val).astype(np.float64)
            w_sum = mask.sum()
        w = mask / w_sum

        # gradient via FFT
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur - step)
        # evaluate candidate
        a_fft_c = np.fft.rfft(cand, n=L)
        conv_c = np.fft.irfft(a_fft_c * a_fft_c, n=L)[: 2 * n - 1]
        cand_sc = 2.0 * n * conv_c.max()

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            # Simulated annealing acceptance (optional)
            no_improve += 1
            # annealing temperature based on current max value
            temp = max_val * 0.01 + 1e-12
            prob = math.exp(-(cand_sc - cur_sc) / temp)
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # Periodic random shake when stuck
        if no_improve > 5000:
            # random transfer of mass between two indices
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            a_fft_tmp = np.fft.rfft(tmp, n=L)
            cand_sc2 = 2.0 * n * np.fft.irfft(a_fft_tmp * a_fft_tmp, n=L)[:2*n-1].max()
            if cand_sc2 < cur_sc:
                cur, cur_sc = tmp, cand_sc2
                no_improve = 0
                if cand_sc2 < best_sc:
                    best_sc, best_vec = cand_sc2, tmp.copy()

        # Periodic symmetrization
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            a_fft_sym = np.fft.rfft(sym, n=L)

            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # Learning rate decay
        lr *= decay_lr

        # Early exit if target reached
        if best_sc < target:
            break

    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v):
    """Simple non‑negative projection onto the probability simplex (sum = 1)."""
    v = np.asarray(v, dtype=np.float64)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v /= s
    else:
        v = np.full_like(v, 1.0 / v.size)
    return v
