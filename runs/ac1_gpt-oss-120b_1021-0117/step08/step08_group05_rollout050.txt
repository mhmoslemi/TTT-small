def _score_simplex(a,n):
    L = 1 << ((2*n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, budget_s - 0.2)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2500
        a = rng.random(n, dtype=np.float64)
    a = np.maximum(a, 0.0)
    # project onto simplex sum=1
    a = _project_simplex(a)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    idx = (2 * n - 2) - np.arange(n)

    # Functions defined earlier etc.

    # Evaluate initial score
    cur = a.copy()
    cur_sc = _score_simplex(cur, n)
    best_vec = cur.copy()
    best_sc = cur_sc

    # Adam states
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    alpha = 0.05 # base learning rate
    T = None

    it = 0
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1
        # compute conv and gradient effectively
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_conv = conv.max()

        # improve best if needed
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0

        # temperature schedule
        if T is None:
            T = max_conv * 0.07
        else:
            T = max(T * 0.998, max_conv * 1e-4)

        # softmax weights
        w = np.exp((conv - max_conv) / T)
        ws = w.sum()
        if ws == 0.0:
            w = (conv == max_conv).astype(np.float64)
            ws = w.sum()
        w /= ws

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = (alpha / math.sqrt(it)) * m_hat / (np.sqrt(v_hat) + epsilon)

        cand = cur - step
        cand = _project_simplex(cand)

        cand_sc = _score_simplex(cand, n)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            # occasional uphill acceptance
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (T + 1e-12)):
                cur, cur_sc = cand, cand_sc

        # random redistribution if stuck
        if no_improve > 2000:
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.2
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_simplex(tmp, n)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc = tmp_sc
                    best_vec = tmp.copy()
        # symmetrisation every 500 iter
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_simplex(sym, n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        if best_sc < target:
            break

    return [float(x) for x in best_vec]

import time, math, numpy as np
from scipy.signal.windows import dpss  # may be unavailable, handled later

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_simplex(a, n):
    """Fast objective for a vector with sum(a)=1: 2*n*max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---- obtain problem size -------------------------------------------------
    if "height_sequence_1" in globals():
        n = int(height_sequence_1.shape[0])
        base_vec = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2500
        base_vec = rng.random(n, dtype=np.float64)

    # ---- build a pool of initial candidates ----------------------------------
    candidates = []

    # parent vector (projected)
    candidates.append(_project_simplex(np.maximum(base_vec, 0.0)))

    i = np.arange(n, dtype=np.float64)

    # analytic windows
    candidates.append(np.full(n, 1.0 / n))                                   # uniform
    candidates.append(0.5 * (1 - np.cos(2 * np.pi * i / (n - 1))))          # Hann
    candidates.append(0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1)))       # Hamming
    a0, a1, a2 = 0.42, 0.5, 0.08
    candidates.append(a0 - a1 * np.cos(2 * np.pi * i / (n - 1)) +
                     a2 * np.cos(4 * np.pi * i / (n - 1)))                # Blackman
    candidates.append(np.sin(np.pi * (i + 1) / (n + 1)))                    # sine
    sigma = 0.2 * n
    candidates.append(np.exp(-0.5 * ((i - (n - 1) / 2) / sigma) ** 2))     # Gaussian

    # DPSS windows (if SciPy supports)

        pass

    # random Dirichlet‑like vectors
    for _ in range(6):
        rnd = rng.random(n, dtype=np.float64)
        candidates.append(_project_simplex(rnd))

    # ---- evaluate pool, keep best -------------------------------------------
    best_vec = None
    best_score = float("inf")
    for cand in candidates:

        sc = _score_simplex(cand, n)

            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_simplex(tmp, n)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_score:
                    best_score = tmp_sc
                    best_vec = tmp.copy()

        # ----- symmetrisation periodically -----
        if it % 500 == 0:

            sym = _project_simplex(sym)
            sym_sc = _score_simplex(sym, n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_score:
                    best_score = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

    # ----- final local jitter (tiny random moves) ----------------------------
    while time.time() < deadline - 0.05:

            best_vec = tmp.copy()
            cur, cur_sc = tmp, tmp_sc

    return [float(x) for x in best_vec]
