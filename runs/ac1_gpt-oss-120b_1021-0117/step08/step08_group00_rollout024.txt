import time, math, numpy as np

def _project_simplex(v):
    """Project vector v onto the probability simplex (non‑negative, sum=1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_vec(a, n, L):
    """Fast AC1 upper‑bound score for a vector a already normalised (sum≈1)."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function achieving a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)  # safety margin

    # ------------------------------------------------------------------ #
    # Initialise a feasible sequence (sum=1, non‑negative).
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)

    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two

    # pre‑compute index mapping for gradient extraction
    idx = (2 * n - 2) - np.arange(n)

    # Adam optimiser state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.5

    # temperature schedule for soft‑max approximation of max
    temp = None
    decay_temp = 0.998
    min_temp_factor = 5e-5

    # bookkeeping
    cur_sc = _score_vec(cur, n, L)
    best_vec = cur.copy()
    best_sc = cur_sc
    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------ #
    # Main optimisation loop
    while time.time() < deadline:
        it += 1

        # convolution of cur with itself
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        # update temperature
        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay_temp, max_val * min_temp_factor)

        # soft‑max weights over convolution values
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of the weighted max (via FFT)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        if tmp_sc < best_sc:
            best_vec, best_sc = tmp.copy(), tmp_sc
            cur, cur_sc = tmp, tmp_sc

    # ------------------------------------------------------------------ #
    # Return best sequence found as plain Python floats
    return [float(x) for x in best_vec]
