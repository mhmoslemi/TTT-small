import time, math, numpy as np

def _make_block_sequence(n: int, p: int):
    """
    Build a sequence of length n consisting of p equally‑sized blocks,
    each separated by a gap of the same size.
    The total mass is 1.0 (so the bound is scale‑invariant).
    Returns None if the parameters are infeasible.
    """
    # block length (including the following gap)
    block_len = n // (2 * p - 1)
    if block_len < 1:
        return None

    block_sum = 1.0 / p                # each block carries the same share
    height = block_sum / block_len      # uniform height inside a block

    seq = np.zeros(n, dtype=np.float64)
    for i in range(p):
        start = i * (2 * block_len)    # block, then a gap of the same size
        end = start + block_len
        if end > n:                     # safety – should not happen
            break
        seq[start:end] = height

    # renormalise (tiny adjustment because the last gap may be truncated)
    total = seq.sum()
    if total == 0.0:
        return None
    seq *= 1.0 / total
    return seq
