def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # initialize
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # generate random vector of length ~2000 maybe; but we need length at least 1. Use a default length?
        n = 2500
        cur = rng.random(n, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))
    n = cur.shape[0]
    L = 1 << ((2*n - 1).bit_length())

    # Scoring and best
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha = 0.2  # base learning rate

    # temperature schedule
    temp_factor = 0.08
    temp_decay = 0.995
    min_temp_factor = 1e-4

    temp = None
    it = 0
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1
        # FFT and conv
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2*n - 1]
        max_val = conv.max()

        # temperature
        if temp is None:
            temp = max_val * temp_factor
        else:
            temp = max(temp * temp_decay, max_val * min_temp_factor)

        # softmax weights
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        # apply step
        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        # Accept
        if cand_sc < cur_sc or rng.random() < math.exp(-(cand_sc - cur_sc) / (temp + 1e-12)):
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1

        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()

        # stagnation moves
        if no_improve > 2000:
            # random displacement move
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()

        # occasional mass transfer towards extremes
        if it % 500 == 0:
            # move mass from max gradient to min gradient
            i_max = int(np.argmax(grad))
            i_min = int(np.argmin(grad))
            if i_max != i_min:
                delta = min(cur[i_max] * 0.2, 0.001)
                tmp = cur.copy()
                tmp[i_max] -= delta
                tmp[i_min] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()

        # early exit
        if best_sc < target:
            break

    # final polishing: small Gaussian perturbations
    while time.time() < deadline:
        pert = best_vec + rng.normal(scale=0.003, size=n)
        pert = np.clip(pert, 0.0, None)
        pert = _project_simplex(pert)
        sc = _score(pert)
        if sc < best_sc:
            best_sc, best_vec = sc, pert.copy()
            if best_sc < target:
                break

    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Exact score (2·n·max(conv(a,a))) for a vector with sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    # Adam optimizer state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha = 0.20          # base learning rate

    # temperature (soft‑max) schedule
    temp_factor = 0.08
    temp_decay = 0.995
    min_temp_factor = 1e-4
    temp = None

    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # convolution of current point
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # temperature for smooth max
        if temp is None:
            temp = max_val * temp_factor
        else:
            temp = max(temp * temp_decay, max_val * min_temp_factor)

        # soft‑max weights
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # gradient of smooth max
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # --------------------------------------------------------------
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        # candidate point
        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        # --------------------------------------------------------------
        # acceptance (Metropolis)
        if cand_sc < cur_sc or rng.random() < math.exp(-(cand_sc - cur_sc) / (temp + 1e-12)):
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            no_improve += 1

        if cand_sc < best_sc:
            best_sc, best_vec = cand_sc, cand.copy()

        # --------------------------------------------------------------
        # stagnation handling
        if no_improve > 2000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.4
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()

        if it % 500 == 0:
            # directed mass transfer using gradient extremes
            i_max = int(np.argmax(grad))
            i_min = int(np.argmin(grad))
            if i_max != i_min:
                delta = min(cur[i_max] * 0.2, 0.001)
                tmp = cur.copy()
                tmp[i_max] -= delta
                tmp[i_min] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()

        # early exit if target reached
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # final polishing: small Gaussian perturbations
    while time.time() < deadline:
        pert = best_vec + rng.normal(scale=0.003, size=n)
        pert = np.clip(pert, 0.0, None)
        pert = _project_simplex(pert)
        sc = _score(pert)
        if sc < best_sc:
            best_sc, best_vec = sc, pert.copy()
            if best_sc < target:
                break

    # Return plain Python list
    return [float(x) for x in best_vec]
