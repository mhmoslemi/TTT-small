        max_val = conv.max()
        # compute temperature for softmax
        if temp is None:

        if temp < 1e-8:
            w = (conv == max_val).astype(np.float64)
            cnt = w.sum()
            if cnt == 0:

                cnt = conv.size

                    best_vec = sym.copy()

import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _max_conv(x: np.ndarray, L: int) -> float:
    """Maximum of the linear convolution x*x using FFT of length L."""
    a_fft = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * x.size - 1]
    return float(conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a safety margin for final return
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------ init
    if "height_sequence_1" in globals():
        cur = np.array(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two FFT length

    best_vec = cur.copy()
    best_sc = 2.0 * n * _max_conv(best_vec, L)

    # Adam‑style state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    lr0 = 0.3
    eps = 1e-8

    temp = None
    decay = 0.998
    min_temp_factor = 5e-7
    it = 0
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1

        # ------------------------------------------------------------ conv
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = float(conv.max())
        cur_sc = 2.0 * n * max_val

        # ----------------------------------------------------------- temp
        if temp is None:
            temp = max_val * 0.1
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # ------------------------------------------------------ weights
        if temp < 1e-8:                     # hard max
            w = (conv == max_val).astype(np.float64)
            cnt = w.sum()
            if cnt == 0:
                w = np.ones_like(conv) / conv.size
                cnt = conv.size
            w = w / cnt
        else:                               # soft max
            w = np.exp((conv - max_val) / temp)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = (conv == max_val).astype(np.float64)
                w_sum = w.sum()
            w = w / w_sum

        # ---------------------------------------------------------- grad
        w_rev = w[::-1]

        idx = (2 * n - 2) - np.arange(n)

        tmp_sc = 2.0 * n * _max_conv(tmp, L)
        if tmp_sc < best_sc:
            best_sc = tmp_sc
            best_vec = tmp.copy()
            cur = tmp

    return [float(x) for x in best_vec]
