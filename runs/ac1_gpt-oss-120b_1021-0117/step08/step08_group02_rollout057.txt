import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(v, L):
    """Score = 2*n*max(conv(v,v)) for vectors whose sum≈1."""
    n = v.shape[0]
    a_fft = np.fft.rfft(v, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    max_val = conv_full[: 2 * n - 1].max()
    return float(2.0 * n * max_val)

    decay = 0.9995          # slower decay than the reference
    min_temp_factor = 1e-6

    it = 0
    no_improve = 0
    target = 1.50274349

    while True:
        if time.time() >= deadline:
            break
        it += 1

        # ----- convolution and max -----------------------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- temperature update -------------------------------------------
        if temp is None:
            temp = max_val * 0.1
        else:
            temp = max(temp * decay, max_val * min_temp_factor)

        # ----- soft‑max weights ---------------------------------------------
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient of weighted max ------------------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam step ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_simplex(cur - step)
        cand_sc = _score(cand, L)

        accepted = False
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            accepted = True
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (temp + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
                accepted = True

        if accepted:
            no_improve = 0
        else:
            no_improve += 1

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()

        # ----- occasional random redistribution -------------------------------
        if rng.random() < 0.05:                      # ~5 % of iterations
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.3
            if delta > 0.0:
                cand2 = cur.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2_sc = _score(cand2, L)
                if cand2_sc < cur_sc:
                    cur, cur_sc = cand2, cand2_sc
                    no_improve = 0
                else:
                    prob2 = math.exp(-(cand2_sc - cur_sc) / (temp + 1e-12))
                    if rng.random() < prob2:
                        cur, cur_sc = cand2, cand2_sc
                        no_improve = 0
                if cur_sc < best_sc:
                    best_sc = cur_sc
                    best_vec = cur.copy()

        # ----- symmetrisation ------------------------------------------------
        if it % 2500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
                if cur_sc < best_sc:
                    best_sc = cur_sc
                    best_vec = cur.copy()

        # ----- learning‑rate schedule ----------------------------------------
        if it % 20000 == 0:
            lr = max(lr * 0.5, 1e-4)

        # ----- restart when stuck --------------------------------------------
        if no_improve > 20000:
            rand_vec = rng.random(n)
            rand_vec = _project_simplex(rand_vec)
            cur = _project_simplex(best_vec * 0.9 + rand_vec * 0.1)
            cur_sc = _score(cur, L)
            no_improve = 0

        # ----- early exit ----------------------------------------------------
        if best_sc < target:
            break

    return [float(x) for x in best_vec]
