def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np, scipy.sparse as sp, scipy.optimize as opt
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)
    
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    # project onto simplex sum=1
    cur = _project_simplex(cur)
    
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())
    
    # scoring function using sum=1 normalization
    def score(a):
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())
    
    best = cur.copy()
    best_score = score(best)
    
    # parameters
    t = 0.2
    max_iter = 1000  # will break earlier via time
    stagnation = 0
    target = 1.50274349
    
    while time.time() < deadline:
        cur_score = score(cur)
        # compute M
        af = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(af * af, n=L)
        conv = conv_full[:2*n-1]
        M = conv.max()
        # Build LP constraints A @ g <= M
        # A is Toeplitz matrix built from cur
        # Use sp.diags
        diags = [np.full(n, cur[i], dtype=np.float64) for i in range(n)]
        offsets = -np.arange(n, dtype=int)
        A = sp.diags(diags, offsets, shape=(2*n-1, n), format='csr')
        # Solve LP
        c = -np.ones(n, dtype=np.float64)  # minimize -sum(g)
        b_ub = np.full(2*n-1, M, dtype=np.float64)
        bounds = [(0, None)] * n
        
        # optional: limit iterations for linprog using options
        # SciPy's linprog options: time limit not directly, but we can set max_iter low if needed
        # default likely okay
        
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': float(deadline - time.time())})
        # SciPy's 'highs' does not accept time_limit; maybe we can just break if our time after LP > deadline
        # We'll check after.
        if not res.success:
            # fallback: random local perturbation
            # break or do gradient step
            # Let's just break
            break
        g = res.x
        sum_g = g.sum()
        if sum_g == 0:
            # skip
            break
        # Normalize g to sum=1
        g = g / sum_g
        # update cur via convex combination
        cur = (1 - t) * cur + t * g
        # Ensure nonnegative and normalization (should sum to 1)
        cur = np.maximum(cur, 0.0)
        cur = _project_simplex(cur)  # cleaning
        # eval score
        cur_score = score(cur)
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        # reduce t gradually
        t = max(0.01, t * 0.98)
        # If stagnation large, maybe random perturbation
        if stagnation > 20:
            # random small redistribution

            cur = _project_simplex(cur)
            stagnation = 0

def _project_simplex(v):
    """Project a vector onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

            cols.extend(j_vals.tolist())
            data.extend(cur[i_vals].tolist())
        A_sub = sp.coo_matrix((data, (rows, cols)), shape=(len(active_idx), n)).tocsr()

        # ----- linear program: maximize sum(g)  (minimize -sum(g)) ------------
        c = -np.ones(n, dtype=np.float64)        # minimize -∑g
        b_ub = np.full(len(active_idx), M, dtype=np.float64)
        bounds = [(0, None)] * n
        try:
            res = opt.linprog(c, A_ub=A_sub, b_ub=b_ub, bounds=bounds,
                              method='highs', options={'presolve': True})
        except Exception:
            # any unexpected failure -> break out
            break

        if not res.success:
            # If LP fails, abort the refinement phase
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0.0:
            break
        g = g / sum_g                     # normalize to simplex

        # ----- candidate via convex combination -------------------------------
        cand = (1.0 - t) * cur + t * g
        cand = _project_simplex(cand)

        cand_sc = _score_simplex(cand, L)

        # ----- accept only if non‑increasing --------------------------------
        if cand_sc <= cur_sc:
            cur, cur_sc = cand, cand_sc
            stagnation = 0
            if cand_sc < best_sc:
                best_sc, best = cand_sc, cand.copy()
        else:
            stagnation += 1
            # reduce step size on rejection
            t = max(0.01, t * 0.7)

        # ----- occasional symmetrisation ------------------------------------
        if it % 1000 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_simplex(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best = sym_sc, sym.copy()

        # ----- random reshuffle when stuck ----------------------------------
        if stagnation > 30:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.2
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_simplex(tmp, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                stagnation = 0
                if tmp_sc < best_sc:
                    best_sc, best = tmp_sc, tmp.copy()
            else:
                stagnation = 0  # give up and continue

        # ----- cool t slowly -------------------------------------------------
        t = max(0.01, t * 0.995)

        # ----- early exit ----------------------------------------------------
        if best_sc < target:
            break

        # safety time check
        if time.time() >= deadline:
            break

    # Return the best candidate found (as plain Python floats)
    return [float(x) for x in best]
