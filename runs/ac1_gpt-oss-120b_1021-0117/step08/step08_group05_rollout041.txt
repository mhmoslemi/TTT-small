import time, math, numpy as np

# ---------- utilities ----------
def _project_simplex(v):
    """Project onto the probability simplex (sum = 1, v >= 0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_fast(a, L, idx, n):
    """2*n*max(conv(a,a)) using pre‑allocated FFT length."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ---------- main ----------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ---- problem size ----
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2500                     # fallback length
    L = 1 << ((2 * n - 1).bit_length())    # power‑of‑two FFT length
    idx = (2 * n - 2) - np.arange(n)       # indices for gradient extraction

    # ---- candidate generators ----
    def gen_uniform():
        a = np.full(n, 1.0 / n, dtype=np.float64)
        return a

    def gen_from_parent():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
        return _project_simplex(cur)

    def gen_exponential(p):
        x = np.arange(1, n + 1, dtype=np.float64)
        a = x ** (-p)
        a /= a.sum()
        return a

    def gen_geometric(r):
        i = np.arange(n, dtype=np.float64)
        a = r ** i
        a /= a.sum()
        return a

    # ---- initial exploration (grid search) ----
    best_vec = gen_uniform()
    best_sc = _score_fast(best_vec, L, idx, n)

    # include parent if available
    if "height_sequence_1" in globals():
        cand = gen_from_parent()
        sc = _score_fast(cand, L, idx, n)
        if sc < best_sc:
            best_sc, best_vec = sc, cand

    # exponential family
    for p in np.linspace(0.0, 2.0, 21):
        cand = gen_exponential(p)
        sc = _score_fast(cand, L, idx, n)
        if sc < best_sc:
            best_sc, best_vec = sc, cand

    # geometric family (r close to 1)
    for r in np.linspace(0.90, 0.9999, 25):
        cand = gen_geometric(r)
        sc = _score_fast(cand, L, idx, n)
        if sc < best_sc:
            best_sc, best_vec = sc, cand

    # ---- gradient descent with Adam (soft‑max smoothing) ----
    cur = best_vec.copy()
    cur_sc = best_sc

    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    alpha = 0.08                     # base learning rate

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    best_overall_vec = cur.copy()
    best_overall_sc = cur_sc

    T = None                         # temperature for soft‑max
    it = 0
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1

        # ----- convolution of current vector -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()

        # ----- temperature schedule -----
        if T is None:

        else:

        w_sum = w.sum()
        if w_sum == 0.0:

            w_sum = w.sum()
        w /= w_sum

        w_rev = w[::-1]

        grad = 2.0 * grad_full[idx]

        m = beta1 * m + (1.0 - beta1) * grad

                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                    if tmp_sc < best_overall_sc:
                        best_overall_sc, best_overall_vec = tmp_sc, tmp.copy()

        # ----- early exit if target reached -----
        if best_overall_sc < target:
            break

    # Return the best candidate found as plain Python floats
    return [float(x) for x in best_overall_vec]
