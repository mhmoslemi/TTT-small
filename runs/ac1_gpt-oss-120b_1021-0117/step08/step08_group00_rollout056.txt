            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_score:
                    best_seq, best_score = sym.copy(), sym_sc

        if best_score < target:
            break

    # ---- final return -------------------------------------------------------
    return [float(x) for x in best_seq]
