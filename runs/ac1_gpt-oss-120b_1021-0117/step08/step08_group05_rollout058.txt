def project_simplex(v):
    # returns projection onto simplex sum=1, v_i>=0
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # algorithm similar to previous
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    w = v - theta
    w[w < 0] = 0
    return w

import time, math, numpy as np

def _project_simplex(v):
    """Project v onto the probability simplex (sum = 1, v >= 0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score_simplex(a, n, L):
    """Fast score for a vector a with sum(a)=1: 2*n*max(conv(a,a))."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    # indices to extract gradient components (as in the reference implementation)
    idx = (2 * n - 2) - np.arange(n)

    # ----- Adam bookkeeping -----
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha = 0.12          # base learning rate
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    best_vec = cur.copy()
    best_sc = _score_simplex(best_vec, n, L)
    cur_sc = best_sc

    T = None               # temperature for soft‑max smoothing
    it = 0
    no_improve = 0
    target = 1.50274349

    while time.time() < deadline:
        it += 1

        # ----- convolution of current vector -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()

        # ----- temperature schedule -----
        if T is None:
            T = max_conv * 0.08
        else:
            T = max(T * 0.9995, max_conv * 1e-5)

        # ----- weight vector for (smoothed) max -----
        if T < 1e-7:  # near convergence: use exact active set
            w = (conv == max_conv).astype(np.float64)
        else:
            w = np.exp((conv - max_conv) / T)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = (conv == max_conv).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient of the smoothed max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        cand = _project_simplex(cand)

        # ----- evaluate candidate -----
        cand_sc = _score_simplex(cand, n, L)

        # ----- accept/reject -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1
            # occasional uphill move (simulated annealing)
            if rng.random() < math.exp(-(cand_sc - cur_sc) / (T + 1e-12)):
                cur, cur_sc = cand, cand_sc

        # ----- random redistribution when stuck -----
        if no_improve > 4000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.2
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score_simplex(tmp, n, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()

        # ----- symmetrisation periodically -----
        if it % 800 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_simplex(sym, n, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # Return the best vector found (as plain Python floats)
    return [float(x) for x in best_vec]
