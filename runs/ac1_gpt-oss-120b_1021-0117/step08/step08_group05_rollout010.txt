import time
import math
import numpy as np


def _project_simplex(v):
    """Project vector v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w


def _conv_max(v, n, L):
    """Return (max of convolution v*v, the full convolution array)."""
    a_fft = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    return conv.max(), conv


def _softmax_weights(conv, max_conv, T):
    """Soft‑max weights over convolution entries with temperature T."""
    w = np.exp((conv - max_conv) / T)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = (conv == max_conv).astype(np.float64)
        w_sum = w.sum()
    return w / w_sum


def _gradient(v, w, n, L, idx):
    """Gradient of the smoothed max w.r.t. v."""
    w_rev = w[::-1]
    v_fft = np.fft.rfft(v, n=L)
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(v_fft * w_fft, n=L)
    return 2.0 * grad_full[idx]


def _ac1_score(v, n, L):
    """Exact AC1 upper bound for a non‑negative sequence with sum=1."""
    max_conv, _ = _conv_max(v, n, L)
    return float(2.0 * n * max_conv)
