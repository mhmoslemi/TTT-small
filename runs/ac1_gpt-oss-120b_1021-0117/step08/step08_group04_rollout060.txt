        w_sum = w.sum()

            cur = cur / cur.sum()
        else:
            cur = cur / cur_sum

        cur_score = _score_norm1(cur)
        if cur_score < best_score:
            best = cur.copy()
            best_score = cur_score
            no_improve = 0
        else:
            no_improve += 1

        # occasional random perturbation when stuck

            # redistribute small mass randomly
            i = rng.integers(0, n)

import time, math, numpy as np

def _project_simplex(v):
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    # re‑normalise (numerical safety)
    s = w.sum()
    if s == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _score_norm1(a):
    """Score assuming sum(a)=1: 2·n·max(conv(a,a))."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
