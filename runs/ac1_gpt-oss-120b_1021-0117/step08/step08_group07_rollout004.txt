import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Exact objective for a vector with sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # safety margin

    # ------------------------------------------------------------------
    # Initialise ---------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2500, dtype=np.float64)
    cur = _project_simplex(np.maximum(cur, 0.0))
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    # Adam hyper‑parameters
    lr = 0.12          # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    t = 0

    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # soft‑max temperature handling
    temp = None
    decay = 0.995
    min_temp_fac = 1e-4

    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    while time.time() < deadline:
        t += 1

        # ----- convolution & smooth‑max ---------------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        if temp is None:
            temp = max_val * 0.07
        else:
            temp = max(temp * decay, max_val * min_temp_fac)

        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = (conv == max_val).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ----- gradient of smooth max ------------------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = -lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur + step
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        if cand_sc < best_sc:

            best_vec = cand.copy()

            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
