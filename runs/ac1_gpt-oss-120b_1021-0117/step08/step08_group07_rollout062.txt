import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum = 1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a):
    """Exact score 2·n·max(conv(a,a)) for a vector with sum≈1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _conv(a, b):
    """Linear convolution via FFT (fast, memory‑linear)."""
    n = len(a)
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    bf = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(af * bf, n=L)[: 2 * n - 1]
    return conv

def _active_constraints(a, conv, M, max_active=30, rel_tol=0.995):
    """Indices where conv is close to its maximum, limited to max_active."""
    idx = np.where(conv >= rel_tol * M)[0]
    if idx.size > max_active:
        order = np.argsort(-conv[idx])[:max_active]
        idx = idx[order]
    return idx

def _build_A(a, active_idx):
    """Build matrix A (|active| × n) for constraints (a * b)[k] ≤ M."""
    n = len(a)
    rows = []
    for k in active_idx:
        row = np.zeros(n, dtype=np.float64)
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        j = np.arange(start, end + 1, dtype=int)
        row[j] = a[k - j]
        rows.append(row)
    return np.vstack(rows) if rows else np.zeros((0, n), dtype=np.float64)
