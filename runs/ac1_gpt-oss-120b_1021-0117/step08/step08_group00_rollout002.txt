import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else css[rho[-1]] / (rho[-1] + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

        tmp[i] -= delta
        tmp[j] += delta
        if tmp[i] < 0:
            continue
        tmp_sc = _score_vec(tmp)
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if cur_sc < best_sc:
                best_vec, best_sc = cur.copy(), cur_sc

    # ----- return best found ------------------------------------------------
    return [float(x) for x in best_vec]
