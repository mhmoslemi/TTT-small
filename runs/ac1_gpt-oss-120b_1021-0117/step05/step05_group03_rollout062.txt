import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex.
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast bound for a vector with sum = 1 (returns 2 * n * max(conv))
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * a.shape[0] - 1]
    return float(2.0 * a.shape[0] * conv.max())
