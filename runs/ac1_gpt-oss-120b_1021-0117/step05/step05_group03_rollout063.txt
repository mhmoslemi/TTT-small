import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection – Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound for a vector with sum = 1 (returns 2 * n * max(conv))
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Compute convolution, its maximum and a sub‑gradient of the max
# ------------------------------------------------------------
def _conv_and_subgrad(a: np.ndarray, L: int):
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)

    # full linear convolution via zero‑padded FFT
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]

    max_val = conv.max()
    # hard‑max weight vector (sub‑gradient of max)
    mask = np.isclose(conv, max_val, atol=1e-12)
    w = mask.astype(np.float64)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / w.size
    else:
        w /= w_sum

    # gradient of max(conv) w.r.t. a:
    #   grad = 2 * conv(w_rev, a) evaluated at appropriate indices
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)

    idx = (2 * n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]
    return max_val, g


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # keep a safety margin

    # ------------------------------------------------------------------
    # Initialise – use the provided parent if available, otherwise uniform
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = np.full(1024, 1.0 / 1024, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur /= cur.sum()                     # enforce sum = 1
    n = cur.size

    # ------------------------------------------------------------
    # FFT length – power of two safely larger than 3*n-2
    # ------------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------
    # Optimisation bookkeeping
    # ------------------------------------------------------------
    best_vec = cur.copy()
    best_sc = _score(best_vec, L)

    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.5

    no_improve = 0
    sym_period = 300
    restart_after = 6000

    target = 1.50274349  # target bound – stop early if reached

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution, max and sub‑gradient -----
        max_val, grad = _conv_and_subgrad(cur, L)

        # ----- Adam step -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = base_lr / (math.sqrt(it) + 1.0)    # modest decay
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score(cand, L)

        # ----- accept if improvement -----
        if cand_sc < best_sc:
            best_vec, best_sc = cand.copy(), cand_sc
            no_improve = 0
        else:
            no_improve += 1

        cur = cand

        # ----- periodic symmetrisation (helps with balanced ends) -----
        if it % sym_period == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < best_sc:
                best_vec, best_sc = sym.copy(), sym_sc
                cur = sym
                no_improve = 0

        # ----- random restart when stuck -----
        if no_improve > restart_after:
            # generate a fresh random vector, keep a tiny fraction of the best
            mix = rng.random(n)
            mix = _project_simplex(mix)
            cur = 0.8 * best_vec + 0.2 * mix
            cur = _project_simplex(cur)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        # ----- early stop if we beat the target -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
