import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum(x)=z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex."""
    n = v.size
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # Initial vector – use the parent if present, otherwise a random start
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2048, dtype=np.float64)

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                      # enforce sum = 1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())  # FFT length (next power of two)

    # ------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    best_sc = 2.0 * n * conv_full.max()
    best_vec = a.copy()

    # optimisation parameters
    step0 = 1.2          # base step magnitude
    beta = 0.90          # momentum factor
    momentum = np.zeros_like(a)
    sym_interval = 250   # symmetrisation frequency
    target = 1.50274349
    it = 0
    no_improve = 0

    # ------------------------------------------------------------
    # Optimisation loop (matrix‑free, O(n log n) per iteration)
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution & its maximum -----
        af = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        cur_max = conv_full.max()
        cur_sc = 2.0 * n * cur_max

        # ----- keep the best seen so far -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- early exit if target reached -----
        if best_sc < target:
            break

        # ----- indices where the convolution is (almost) maximal -----
        eps = max(1e-14, cur_max * 1e-8)
        idx_max = np.where(conv_full >= cur_max - eps)[0]

        # ----- sub‑gradient of the max‑convolution -----
        g = np.zeros_like(a)
        for k in idx_max:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i = np.arange(i_start, i_end + 1, dtype=int)
            g[i] += 2.0 * a[k - i]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- back‑tracking line search -----
        base_alpha = step0 / math.sqrt(it)
        alpha = base_alpha
        cand = None
        cand_sc = None
        for _ in range(6):                     # up to 6 attempts
            a_tmp = a - alpha * momentum
            a_tmp = _project_simplex(a_tmp)     # stay on simplex
            af_tmp = np.fft.rfft(a_tmp, n=L)
            conv_tmp = np.fft.irfft(af_tmp * af_tmp, n=L)[: 2 * n - 1]
            cand_sc = 2.0 * n * conv_tmp.max()
            if cand_sc <= cur_sc:
                cand, cand_sc = a_tmp, cand_sc
                break
            alpha *= 0.5

        # if no descent found, keep the last trial (might be uphill)
        if cand is None:
            cand, cand_sc = a_tmp, cand_sc

        # ----- accept / reject (Metropolis style) -----
        if cand_sc < cur_sc:
            a, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                a, cur_sc = cand, cand_sc

        # ----- periodic symmetrisation (helps with Toeplitz structure) -----
        if it % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            af_sym = np.fft.rfft(a_sym, n=L)
            conv_sym = np.fft.irfft(af_sym * af_sym, n=L)[: 2 * n - 1]
            sym_sc = 2.0 * n * conv_sym.max()
            if sym_sc < cur_sc:
                a, cur_sc = a_sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, a_sym.copy()
                    no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.4
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
                no_improve = 0

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
