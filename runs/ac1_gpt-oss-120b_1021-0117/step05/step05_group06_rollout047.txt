import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else css[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _max_conv(a: np.ndarray, L: int) -> float:
    """Maximum value of the linear convolution a*a (full length)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())


def _anti_identity(seq: np.ndarray) -> np.ndarray:
    """Place sorted values alternately from the ends toward the centre."""
    idx = np.argsort(-seq)
    out = np.empty_like(seq)
    l, r = 0, seq.size - 1
    take_left = True
    for i in idx:
        if take_left:
            out[l] = seq[i]
            l += 1
        else:
            out[r] = seq[i]
            r -= 1
        take_left = not take_left
    return out
