                    cur_max = tmp_max
                    if tmp_max < best_max:
                        best_max = tmp_max
                        best_vec = tmp.copy()
                        best_score = 2.0 * n * best_max
                    no_improve = 0
                else:
                    no_improve = 0

        # ---- early exit if target achieved ----
        if best_score < target_score:
            break

    # --------------------------------------------------------
    # Final short stochastic refinement (optional)
    # --------------------------------------------------------
    while time.monotonic() < deadline - 0.05:
        pert = best_vec + rng.normal(scale=0.005, size=n)
        pert = np.clip(pert, 0.0, None)
        pert = _normalize(pert)
        p_max, _, _ = _self_conv_max(pert, L)
        if p_max < best_max:
            best_max = p_max
            best_vec = pert.copy()

    # --------------------------------------------------------
    # Return the best sequence found (plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
