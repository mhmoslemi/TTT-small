import time, math, numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 bound (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        if eps is None:
            eps = max(max_val * 0.1, 1e-9)
        else:
            eps = max(eps * eps_decay, 1e-9)

        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of smoothed max (via FFT) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)      # map to needed n components
        g = 2.0 * grad_full[idx]              # gradient of smoothed max

        # ----- Adam step -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (0.5 / math.sqrt(it + 1.0)) * (m_hat / (np.sqrt(v_hat) + eps_adam))

        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        # ----- simple line‑search if needed -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            improved = False
            alpha = 0.5
            for _ in range(5):
                cand2 = cur - alpha * step
                cand2 = _project_simplex(cand2)
                cand2_sc = _score(cand2)
                if cand2_sc < cur_sc:
                    cur, cur_sc = cand2, cand2_sc
                    improved = True
                    no_improve = 0
                    break
                alpha *= 0.5
            if not improved:
                no_improve += 1
                # occasional random mass‑transfer when stalled
                if no_improve > 2000:
                    i = rng.integers(0, n)
                    if cur[i] > 0.0:
                        delta = cur[i] * rng.random() * 0.45
                        j = rng.integers(0, n - 1)
                        if j >= i:
                            j += 1
                        tmp = cur.copy()
                        tmp[i] -= delta
                        tmp[j] += delta
                        tmp = _project_simplex(tmp)
                        tmp_sc = _score(tmp)
                        if tmp_sc < cur_sc:
                            cur, cur_sc = tmp, tmp_sc
                            no_improve = 0

        # ----- keep global best -----
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0

        # ----- periodic symmetrisation -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # ----- early exit -----
        if best_sc < improve_target:
            break

    # ----- final local refinement (tiny random nudges) -----
    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if best_vec[i] <= 0:
            continue
        delta = best_vec[i] * rng.random() * 0.02
        cand = best_vec.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_sc = _score(cand)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_vec = cand

    return [float(x) for x in best_vec]
