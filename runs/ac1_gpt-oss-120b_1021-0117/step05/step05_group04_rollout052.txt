import time

def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale vector to have sum = 1 (non‑negative)."""
    s = v.sum()
    if s <= 0.0:
        v = np.full_like(v, 1.0 / v.size, dtype=np.float64)
        s = v.sum()
    return v / s

def _conv_max_fft(a: np.ndarray, L: int):
    """
    Compute linear convolution a*a via FFT (length L),
    return its maximum value, the index of the maximum and the full
    convolution (first 2*len(a)-1 entries) together with the FFT of a.
    """
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    max_idx = int(np.argmax(conv))
    return float(conv[max_idx]), max_idx, conv, a_fft

def _grad_log(a: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient of the (unsmoothed) max‑convolution w.r.t. log‑weights y=log a.
    grad_y[i] = 2 * a[i] * a[k-i]  (zero when k-i is out of bounds).
    """
    n = a.shape[0]
    idx = k - np.arange(n)
    b = np.zeros_like(a)
    mask = (idx >= 0) & (idx < n)
    b[mask] = a[idx[mask]]
    return 2.0 * a * b
