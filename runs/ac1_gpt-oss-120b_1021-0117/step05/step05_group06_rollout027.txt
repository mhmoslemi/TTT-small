def project_simplex(v):
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho+1)
    return np.maximum(v - theta, 0.0)

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum = 1})
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Maximum value of a*a (full convolution) via zero‑padded FFT
# ----------------------------------------------------------------------
def _max_conv_fft(a: np.ndarray, L: int) -> float:
    """Return max(conv_full) for a*a using an FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return float(conv_full[: 2 * a.size - 1].max())


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the final return
    deadline = start + max(0.0, float(budget_s) - 1.0)

    # ------------------------------------------------------------------
    # Base length (use parent if it exists)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
    else:
        n = 2048
        base = None

    # ------------------------------------------------------------------
    # FFT length (next power of two >= 2n‑1)
    # ------------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Helper to compute the bound (sum is kept equal to 1)
    # ------------------------------------------------------------------
    def _bound(a: np.ndarray) -> float:
        return 2.0 * n * _max_conv_fft(a, L)

    # ------------------------------------------------------------------
    # Initialise best candidate
    # ------------------------------------------------------------------
    if base is not None:
        a0 = np.maximum(base, 0.0)
        a0 = _project_simplex(a0)
    else:
        a0 = rng.random(n, dtype=np.float64)
        a0 = _project_simplex(a0)

    best_seq = a0.copy()
    best_bound = _bound(best_seq)

    # ------------------------------------------------------------------
    # Main optimisation loop – multiple short restarts
    # ------------------------------------------------------------------
    max_restarts = 6                # enough restarts for exploration
    restart_cnt = 0
    while time.monotonic() < deadline - 5.0 and restart_cnt < max_restarts:
        # ----------------------------------------------------------------
        # Starting point for this restart
        # ----------------------------------------------------------------
        if restart_cnt == 0:
            a = best_seq.copy()
        else:
            a = rng.random(n, dtype=np.float64)
            a = _project_simplex(a)

        # ----------------------------------------------------------------
        # Hyper‑parameters (very similar to the reference implementation)
        # ----------------------------------------------------------------
        step0 = 0.8
        beta = 0.9                     # momentum factor
        momentum = np.zeros_like(a)
        eps0 = 1.0
        eps_decay = 0.9997
        eps_min = 1e-9
        it = 0
        cur_sc = _max_conv_fft(a, L)
        stagnation = 0
        max_iter = 70000               # generous iteration budget per restart

        while time.monotonic() < deadline - 5.0 and it < max_iter:
            it += 1

            # ------------------------------------------------------------
            # Convolution and its maximum (full, length 2n‑1)
            # ------------------------------------------------------------
            a_fft = np.fft.rfft(a, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[: 2 * n - 1]
            max_val = conv.max()

            # ------------------------------------------------------------
            # Soft‑max weights (log‑sum‑exp smoothing of the max)
            # ------------------------------------------------------------
            eps = max(eps_min, eps0 * (eps_decay ** it))
            if eps > 1e-7:
                w = np.exp((conv - max_val) / eps)
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(conv, 1.0 / conv.size)
                else:
                    w /= w_sum
            else:
                mask = conv >= max_val - 1e-12
                cnt = int(mask.sum())
                if cnt == 0:
                    w = np.full_like(conv, 1.0 / conv.size)
                else:
                    w = np.where(mask, 1.0 / cnt, 0.0)

            # ------------------------------------------------------------
            # Gradient of the smoothed max (2 * a * w_rev convolution)
            # ------------------------------------------------------------
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            g = 2.0 * grad_full[idx]

            # ------------------------------------------------------------
            # Momentum update
            # ------------------------------------------------------------
            momentum = beta * momentum + (1.0 - beta) * g

            # ------------------------------------------------------------
            # Simple back‑tracking line search
            # ------------------------------------------------------------
            alpha = step0 / math.sqrt(it + 1.0)
            improved = False
            for _ in range(8):
                cand = a - alpha * momentum
                cand = _project_simplex(cand)
                cand_sc = _max_conv_fft(cand, L)
                if cand_sc <= cur_sc:
                    a = cand
                    cur_sc = cand_sc
                    improved = True
                    break
                alpha *= 0.5

            # ------------------------------------------------------------
            # Book‑keeping
            # ------------------------------------------------------------
            if improved:
                stagnation = 0
                cand_bound = 2.0 * n * cur_sc
                if cand_bound < best_bound:
                    best_bound = cand_bound
                    best_seq = a.copy()
            else:
                stagnation += 1
                if stagnation >= 2500:
                    break                     # give up this restart

            # ------------------------------------------------------------
            # Occasional symmetrisation (helps due to Toeplitz symmetry)
            # ------------------------------------------------------------
            if it % 300 == 0:
                sym = (a + a[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_sc = _max_conv_fft(sym, L)
                if sym_sc < cur_sc:
                    a = sym
                    cur_sc = sym_sc
                    sym_bound = 2.0 * n * sym_sc
                    if sym_bound < best_bound:
                        best_bound = sym_bound
                        best_seq = a.copy()

            # ------------------------------------------------------------
            # Early exit if we have already reached the target
            # ------------------------------------------------------------
            if best_bound < 1.50274349:
                break

        # ----------------------------------------------------------------
        # Permutation phase – try a few deterministic patterns
        # ----------------------------------------------------------------
        vals = np.sort(best_seq)[::-1]          # descending values
        candidates = []

        # (1) pure descending order
        candidates.append(vals.copy())
        # (2) pure ascending order
        candidates.append(vals[::-1].copy())

        # (3) anti‑identity (alternating high / low)
        anti = np.empty_like(vals)
        l, r = 0, n - 1
        k = 0
        while l <= r:
            anti[l] = vals[k]; k += 1
            if l < r:
                anti[r] = vals[k]; k += 1
            l += 1
            r -= 1
        candidates.append(anti)

        # evaluate all candidates
        for cand in candidates:
            sc = _max_conv_fft(cand, L)
            bound = 2.0 * n * sc
            if bound < best_bound:
                best_bound = bound
                best_seq = cand.copy()

        # ----------------------------------------------------------------
        # Random hill‑climbing on the best permutation found so far
        # ----------------------------------------------------------------
        cur = best_seq.copy()
        cur_sc = _max_conv_fft(cur, L)
        cur_bound = 2.0 * n * cur_sc

        swap_attempts = 0
        while time.monotonic() < deadline - 0.5:
            # simple swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cur[i], cur[j] = cur[j], cur[i]

            new_sc = _max_conv_fft(cur, L)
            new_bound = 2.0 * n * new_sc
            if new_bound < cur_bound:
                cur_bound = new_bound
                cur_sc = new_sc
                if new_bound < best_bound:
                    best_bound = new_bound
                    best_seq = cur.copy()
                    # stop early if we hit the target
                    if best_bound < 1.50274349:
                        break
            else:
                # revert the swap
                cur[i], cur[j] = cur[j], cur[i]

            swap_attempts += 1
            # occasional random mass‑transfer move (helps fine‑tuning)
            if swap_attempts % 2000 == 0:
                i, j = rng.integers(0, n, size=2)
                if i != j and cur[i] > 0:
                    delta = rng.random() * cur[i] * 0.3
                    cur[i] -= delta
                    cur[j] += delta
                    # keep feasibility (still non‑negative, sum unchanged)
                    new_sc = _max_conv_fft(cur, L)
                    new_bound = 2.0 * n * new_sc
                    if new_bound < cur_bound:
                        cur_bound = new_bound
                        cur_sc = new_sc
                        if new_bound < best_bound:
                            best_bound = new_bound
                            best_seq = cur.copy()
                    else:
                        # revert mass transfer
                        cur[i] += delta
                        cur[j] -= delta

        restart_cnt += 1

        # ----------------------------------------------------------------
        # If a sufficiently good bound has been found we can quit early
        # ----------------------------------------------------------------
        if best_bound < 1.50274349:
            break

    # --------------------------------------------------------------------
    # Fallback safety net (should never be needed)
    # --------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        best_seq = np.array([1.0])

    # Return as plain Python list of floats
    return [float(x) for x in best_seq]
