import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 bound (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(conv) for a sequence with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------
    # initialise a feasible point on the simplex
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:                                               # fallback
        n0 = 2000
        a0 = np.exp(-np.arange(n0, dtype=np.float64) / 100.0)

    a0 = np.maximum(a0, 0.0)
    s0 = a0.sum()
    if s0 <= 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
        s0 = 1.0
    a0 = a0 / s0                         # now sum=1

    n = a0.size
    idx = (2 * n - 2) - np.arange(n)     # map full convolution back to n entries
    L = 1 << ((2 * n - 1).bit_length())  # FFT length (power of two)

    cur = a0.copy()
    best_vec = cur.copy()
    cur_sc = best_sc = _score(cur)

    # --------------------------------------------------------
    # optimisation parameters
    # --------------------------------------------------------
    it = 0
    no_improve = 0
    eps = None
    eps_decay = 0.998
    beta1, beta2 = 0.9, 0.999            # Adam moments
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    lr0 = 0.8

    max_random_transfer = 500            # start random mass moves after this many fails

    while time.monotonic() < deadline:

        # ----- convolution & smoothed weight -----
        a_fft = np.fft.rfft(cur, n=L)

        max_val = conv_full.max()

        if eps is None:
            eps = max(max_val * 0.15, 1e-12)

            eps = max(eps * eps_decay, 1e-12)

        w_sum = w.sum()
        if w_sum == 0.0:

        # ----- gradient of the smoothed max (via FFT) -----

            m.fill(0.0)

            eps = None
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < improve_target:
            break

    # return as plain list of Python floats
    return [float(x) for x in best_vec]
