import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x)=1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of 2*n*max(conv(a,a)) for sum(a)=1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(convolution(a,a)). Assumes sum(a)=1 (or close)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # power‑of‑two >= 2n‑1
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

        lr = lr0 * (lr_decay ** it)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand1 = cur - step
        cand1 = _project_simplex(cand1)
        cand1_sc = _score(cand1)

        # ---- Mirror‑descent style candidate (entropic) ----
        eta = 0.5 / math.sqrt(it)                # analogous to previous work
        exp_arg = -eta * grad
        np.clip(exp_arg, -50, 50, out=exp_arg)   # avoid overflow
        cand2 = cur * np.exp(exp_arg)
        s = cand2.sum()
        if s > 0.0:
            cand2 /= s
        else:
            cand2 = cur.copy()
        cand2_sc = _score(cand2)

        # ---- pick the better of the two ----
        if cand1_sc <= cand2_sc:
            cand, cand_sc = cand1, cand1_sc
        else:
            cand, cand_sc = cand2, cand2_sc

        # ---- accept / reject ----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            # occasional Metropolis‑like acceptance to escape plateaus
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # ---- periodic symmetrisation (helps with the Toeplitz structure) ----
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # ---- random perturbation when stuck ----
        if no_improve > 3000:
            # Gaussian jitter around the current best
            pert = best_vec + 0.02 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            pert = _project_simplex(pert)
            cur, cur_sc = pert, _score(pert)
            no_improve = 0

        # ---- early exit if target reached ----
        if best_sc < target:
            break

    # --------------------------------------------------------
    # Return the best sequence discovered (as plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
