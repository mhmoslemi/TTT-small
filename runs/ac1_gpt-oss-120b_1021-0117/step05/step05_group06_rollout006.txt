import time
import math

# ------------------------------------------------------------
# Helper: projection onto the probability simplex (sum = 1, x>=0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

                    best_bound = 2.0 * n * best_sc

        if best_bound < target_bound:
            break

    # ------------------- permutation heuristics --------------------
    # sorted values descending
    sorted_vals = np.sort(best)[::-1]

    # anti‑identity (alternating high/low) – already useful
    vals = sorted_vals
    perm = np.empty_like(vals)
    left, right = 0, n - 1
    k = 0
    while left <= right:
        perm[left] = vals[k]; k += 1
        if left < right:
            perm[right] = vals[k]; k += 1
        left += 1
        right -= 1
    perm_sc = _max_conv(perm, L)
    perm_bound = 2.0 * n * perm_sc
    if perm_bound < best_bound:
        best = perm.copy()
        best_sc = perm_sc
        best_bound = perm_bound

    # cyclic step permutations (gcd(step,n)=1)
    def _perm_by_step(vals, step):
        idx = (np.arange(vals.size) * step) % vals.size
        out = np.empty_like(vals)
        out[idx] = vals
        return out

    for step in (2, 3, 5, 7, 11, 13, 17, 19, 23):
        if math.gcd(step, n) != 1:
            continue
        cand = _perm_by_step(sorted_vals, step)
        cand_sc = _max_conv(cand, L)
        cand_bound = 2.0 * n * cand_sc
        if cand_bound < best_bound:
            best = cand.copy()
            best_sc = cand_sc
            best_bound = cand_bound
            if best_bound < target_bound:
                break

    # --------------------------- local hill‑climb -----------------
    cur = best.copy()
    cur_bound = best_bound
    while time.monotonic() < deadline - 0.3:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_sc = _max_conv(cur, L)
        new_bound = 2.0 * n * new_sc
        if new_bound < cur_bound:
            cur_bound = new_bound
            best = cur.copy()
            best_sc = new_sc
            best_bound = new_bound
            if best_bound < target_bound:
                break
        else:
            # revert the swap
            cur[i], cur[j] = cur[j], cur[i]

        # occasional tiny random perturbation (simulated‑annealing style)
        if rng.random() < 0.001:
            eps = 0.02
            pert = cur + rng.normal(scale=eps, size=n)
            pert = _project_simplex(pert)
            pert_sc = _max_conv(pert, L)
            pert_bound = 2.0 * n * pert_sc
            if pert_bound < cur_bound or rng.random() < 0.01:
                cur = pert
                cur_bound = pert_bound
                if pert_bound < best_bound:
                    best = pert.copy()
                    best_sc = pert_sc
                    best_bound = pert_bound

    # ----------------------------------------------------------------
    # Return the best candidate found (plain Python list)
    # ----------------------------------------------------------------
    if best.size == 0:
        best = np.array([1.0])
    return [float(x) for x in best]
