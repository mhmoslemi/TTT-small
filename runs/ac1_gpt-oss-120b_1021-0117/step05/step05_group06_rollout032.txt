    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*n-1]

    if eps > 1e-7:
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0/w.size)
        else:
            w /= w_sum
    else:
        mask = conv >= max_val - 1e-12
        cnt = int(mask.sum())
        if cnt == 0:
            w = np.full_like(conv, 1.0/conv.size)
        else:
            w = np.where(mask, 1.0/cnt, 0.0)

    w_rev = w[::-1]  # length 2*n-1
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2*n-2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, max_val, conv

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper: project onto probability simplex (sum = 1, x >= 0)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# --------------------------------------------------------------
# Helper: maximum value of a*a (full convolution) via FFT
# --------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return float(conv_full[: 2 * a.size - 1].max())


# --------------------------------------------------------------
# Helper: gradient of the smoothed max‑convolution objective
# --------------------------------------------------------------
def _grad_max_conv(a: np.ndarray, L: int, n: int, eps: float):
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()

    if eps > 1e-7:
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
    else:
        mask = conv >= max_val - 1e-12
        cnt = int(mask.sum())
        if cnt == 0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = np.where(mask, 1.0 / cnt, 0.0)

    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad, max_val, conv

# Main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a non‑empty list of non‑negative coefficients aiming to
    minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ----- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2048, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)        # enforce sum = 1, non‑negative
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (power of two)

    # ----- bookkeeping ------------------------------------------------
    cur = a.copy()
    cur_sc = _max_conv(cur, L)
    best = cur.copy()
    best_sc = cur_sc
    best_bound = 2.0 * n * best_sc

    # ----- hyper‑parameters -----------------------------------------
    step0 = 1.0
    beta = 0.9
    eps0 = 1.0
    eps_decay = 0.9995
    eps_min = 1e-9
    momentum = np.zeros_like(cur)
    it = 0
    target_bound = 1.50274349

    # ----- gradient‑descent phase ------------------------------------
    while time.monotonic() < deadline - 2.0:

                    best_sc = cand_sc

            best_sc = sc
            best_bound = 2.0 * n * sc

    # ----- Random permutations (global shuffling) -------------------
    while time.monotonic() < deadline - 1.0:
        perm = best[rng.permutation(n)]
        sc = _max_conv(perm, L)
        if sc < best_sc:
            best = perm.copy()
            best_sc = sc
            best_bound = 2.0 * n * sc
            if best_bound < target_bound:
                break

    # ----- Intensive hill‑climbing swaps -----------------------------
    cur = best.copy()
    cur_sc = best_sc
    while time.monotonic() < deadline - 0.2:

    # ----- final return ------------------------------------------------
