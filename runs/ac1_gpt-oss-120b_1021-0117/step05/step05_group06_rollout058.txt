def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np, random
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # setup length and initial arrays
    if 'height_sequence_1' in globals():
        init_seq = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = init_seq.size
        # Normalize to sum=1 on simplex
        init_seq = np.maximum(init_seq, 0.0)
        init_seq_sum = init_seq.sum()
        if init_seq_sum == 0:
            init_seq = np.full(n, 1.0 / n)
        else:
            init_seq = init_seq / init_seq_sum
    else:
        n = 2000
        init_seq = rng.random(n)
        init_seq = np.maximum(init_seq, 0)
        init_seq = init_seq / init_seq.sum()
    # parameters
    L = 1 << ((2*n-1).bit_length())
    best_seq = init_seq.copy()
    best_sc = max_conv(best_seq, L)
    best_bound = 2.0 * n * best_sc

    # define simple projection
    def proj_simplex(v):
        v = np.maximum(v, 0)
        s = v.sum()
        if s == 0:
            return np.full_like(v, 1.0 / v.size)
        return v / s

    # gradient function
    def grad_max(a):
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        mask = conv >= max_val - 1e-12
        w = mask.astype(float)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        # reversed w
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        return g, max_val

    # optimizer loop
    cur = init_seq.copy()
    cur_sc = best_sc
    momentum = np.zeros_like(cur)
    step0 = 2.0
    beta = 0.9
    eps_decrease = 0.9999 # not used for subgrad now maybe
    # Number of restarts
    # We'll run at most maybe 5 restarts
    restarts = 5
    for r in range(restarts):
        # optionally random perturbation for each restart
        if r > 0:
            # random start: perturb parent
            cur = np.maximum(init_seq + 0.1 * rng.standard_normal(n), 0)
            cur = proj_simplex(cur)
            cur_sc = max_conv(cur, L)

        it = 0
        while time.monotonic() < deadline - 5.0: # keep some time for final hill climbing
            it += 1
            # compute gradient and max conv
            g, max_val = grad_max(cur)
            # momentum
            momentum = beta * momentum + (1.0 - beta) * g
            # step size schedule
            alpha = step0 / math.sqrt(it)
            # try step

            cand = proj_simplex(cand)
            cand_sc = max_conv(cand, L)
            if cand_sc <= cur_sc:
                cur = cand
                cur_sc = cand_sc
                # update best

                    best_seq = cand.copy()
                    best_sc = cand_sc
                    best_bound = 2.0 * n * best_sc
                    # early stop if below target
                    if best_bound < 1.50274349:
                        break
                # continue
                continue
            # if not improvement, try smaller step sizes
            alpha *= 0.5
            improved = False
            for _ in range(4):
                cand = cur - alpha * momentum
                cand = proj_simplex(cand)
                cand_sc = max_conv(cand, L)
                if cand_sc <= cur_sc:
                    cur = cand
                    cur_sc = cand_sc
                    if cand_sc < best_sc:

            if not improved:

                if rng.random() < 0.01:
                    cur = proj_simplex(cur + 0.01 * rng.standard_normal(n))
                    cur_sc = max_conv(cur, L)
                # else continue

    # After gradient descent, try symmetrization
    # (if time)
    if time.monotonic() < deadline - 2.0:

        sym = proj_simplex(sym)
        sym_sc = max_conv(sym, L)
        if sym_sc < best_sc:
            best_seq = sym
            best_sc = sym_sc
            best_bound = 2.0 * n * best_sc

    # Permute using anti-identity pattern (alternating extremes)

        vals = np.sort(best_seq)[::-1]
        perm = np.empty_like(vals)
        left = 0
        right = n - 1
        k = 0
        while left <= right:
            perm[left] = vals[k]; k += 1
            if left < right:
                perm[right] = vals[k]; k += 1
            left += 1
            right -= 1
        perm_sc = max_conv(perm, L)
        if perm_sc < best_sc:
            best_seq = perm
            best_sc = perm_sc
            best_bound = 2.0 * n * best_sc

    cur = best_seq.copy()
    cur_sc = best_sc
    while time.monotonic() < deadline - 0.2:

        new_sc = max_conv(cur, L)
        if new_sc < cur_sc:
            cur_sc = new_sc
            best_seq = cur.copy()
            best_sc = new_sc
            best_bound = 2.0 * n * best_sc
            if best_bound < 1.50274349:
                break
        else:
            # revert
            cur[i], cur[j] = cur[j], cur[i]

    # final fallback if best_seq empty
    if best_seq.size == 0:
        best_seq = np.array([1.0])
    return [float(x) for x in best_seq]

import time

# --------------------------------------------------------------
# Helper: project onto the probability simplex (sum = 1, x >= 0)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


# --------------------------------------------------------------
# Helper: maximum entry of the full (linear) convolution a*a
# --------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * a.size - 1]
    return float(conv.max())


# --------------------------------------------------------------
# Helper: gradient of a smoothed max (soft‑max) of the convolution
# --------------------------------------------------------------
def _grad(a: np.ndarray, eps: float, L: int):
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)

    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()

    # soft‑max weights (clipped to avoid overflow)
    delta = (conv - max_val) / eps
    delta = np.clip(delta, -50, 50)

    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_rev = w[::-1]
    wf = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(af * wf, n=L)
    idx = (2 * n - 2) - np.arange(n)          # pick the correct slice
    g = 2.0 * grad_full[idx]
