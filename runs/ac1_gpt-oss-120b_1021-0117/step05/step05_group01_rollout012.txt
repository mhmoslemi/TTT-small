import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x>=0, sum(x)=z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum=z."""
    # Works for any real vector v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    else:

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                     # enforce Σa_i = 1

    n = a.size

    # --------------------------------------------------------
    # FFT lengths (power of two) – one length works for both conv & grad
    # --------------------------------------------------------

    # --------------------------------------------------------
