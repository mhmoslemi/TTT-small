    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 1.0)  # leave 1 s safety

    # get length n (use parent if provided)
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2000  # fallback
    # compute FFT length
    L = 1 << ((3 * n - 2).bit_length())
    # helper functions
    def max_conv(a):
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return float(conv.max()), conv  # also return full conv for gradient
    def project_simplex(v):
        # same as earlier
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - 1.0
        rho_arr = np.where(u - css / np.arange(1, v.size+1) > 0)[0]
        if rho_arr.size == 0:
            theta = 0.0
        else:
            rho = rho_arr[-1]
            theta = css[rho] / (rho + 1)
        return np.maximum(v - theta, 0.0)
    def gradient(a, conv, mask):
        # conv length = 2n-1
        # mask array same length, sum to 1
        rev = mask[::-1]
        a_fft = np.fft.rfft(a, n=L)
        rev_fft = np.fft.rfft(rev, n=L)
        grad_full = np.fft.irfft(a_fft * rev_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        return g
    # keep best solution
    best_seq = None
    best_bound = float('inf')

    # function to evaluate bound fast
    def bound_from_maxconv(maxc):
        return 2.0 * n * maxc
    # main loop: random restarts

            if left < right:

            left += 1
            right -= 1
        a = perm.copy()

        cur_bound = bound_from_maxconv(cur_max)
        # track best for this restart
        if cur_bound < best_bound:
            best_seq = a.copy()
            best_bound = cur_bound
        # subgradient loop

                cur_max = cand_max
                conv = cand_conv

                    best_bound = cur_bound

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (L2 onto {x>=0, sum=1})
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (sum=1, v>=0)."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Full linear convolution a*a (zero‑padded) via FFT, length 2n‑1
# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, L: int) -> np.ndarray:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)
    return conv[: 2 * a.size - 1]


def _max_conv(a: np.ndarray, L: int) -> float:
    """Maximum of the full convolution a*a."""
    return float(_full_conv(a, L).max())


# ----------------------------------------------------------------------
# Gradient of the (hard) max‑convolution functional
# ----------------------------------------------------------------------
def _gradient(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """Sub‑gradient of max_{k} (a*a)[k] w.r.t a, for weight vector w."""
    rev = w[::-1]
    a_fft = np.fft.rfft(a, n=L)
    rev_fft = np.fft.rfft(rev, n=L)
    grad_full = np.fft.irfft(a_fft * rev_fft, n=L)
    idx = (2 * a.size - 2) - np.arange(a.size)
    return 2.0 * grad_full[idx]


# ----------------------------------------------------------------------
# Deterministic anti‑identity permutation (large opposite small)
# ----------------------------------------------------------------------
def _anti_identity(arr: np.ndarray) -> np.ndarray:
    vals = np.sort(arr)[::-1]
    out = np.empty_like(vals)
    left, right = 0, vals.size - 1
    k = 0
    while left <= right:
        out[left] = vals[k]; k += 1
        if left < right:
            out[right] = vals[k]; k += 1
        left += 1
        right -= 1
    return out

    start = time.monotonic()

    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

    L = 1 << ((3 * n - 2).bit_length())   # power of two >= 3n‑2

    # Initialise candidate (use parent if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

    else:

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # enforce sum = 1
    cur = _anti_identity(cur)            # spread large values

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    cur_max = _max_conv(cur, L)
    best_seq = cur.copy()
    best_bound = 2.0 * n * cur_max

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
