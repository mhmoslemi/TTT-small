    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]
    return g

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # margin
    # Determine n from parent if present else use default maybe 2000
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
        # Clip values to [0,1000] and project to simplex sum=1
        a0 = np.clip(base, 0, 1000)
        a0 = _project_simplex(a0)
    else:
        n = 2000
        a0 = np.full(n, 1.0/n, dtype=np.float64)
    # Prepare second initial: random
    a_rand = rng.random(n, dtype=np.float64)
    a_rand = _project_simplex(a_rand)

    # choose which to start: we can start with one and later switch if stagnates.

    cur = a0.copy()
    # Precompute FFT length L >= 3n-2
    L = 1 << ((3 * n - 2).bit_length())   # next power of 2

    # Optimization hyperparameters
    step0 = 1.0
    momentum_factor = 0.85
    momentum = np.zeros_like(cur)
    alpha0 = 1.0   # softmax temperature inverse (higher => sharper)
    alpha_rate = 1.005   # increase slowly each iteration
    max_alpha = 30.0

    best_vec = cur.copy()
    best_score = _score(best_vec, L)
    cur_score = best_score

    it = 0
    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1
        # compute convolution and max
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2 * n - 1]
        max_val = conv.max()

        # compute softmax weights
        alpha = min(max_alpha, alpha0 * (alpha_rate ** it))
        # shift to avoid overflow
        shifted = conv - max_val  # negative or zero
        exp_vals = np.exp(alpha * shifted)  # exp(alpha*(conv-max))
        w = exp_vals / exp_vals.sum()

        # gradient of weighted sum
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum update and step size
        momentum = momentum_factor * momentum + (1.0 - momentum_factor) * g
        step = step0 / math.sqrt(it)

        # candidate update
        cand = cur - step * momentum
        cand = _project_simplex(cand)
        cand_score = _score(cand, L)

        # accept if better else maybe accept uphill probabilistically
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best_vec, best_score = cand.copy(), cand_score
        else:
            # Simulated annealing style acceptance
            prob = math.exp(-(cand_score - cur_score) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score
            no_improve += 1

        # periodic symmetrization
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score(sym, L)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
            if sym_score < best_score:
                best_vec, best_score = sym.copy(), sym_score
                no_improve = 0

        # occasional random jump if stuck
        if no_improve > 3000:
            # try random shift of mass between two indices
            i = rng.integers(0, n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score = _score(tmp, L)
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                if tmp_score < best_score:
                    best_vec, best_score = tmp.copy(), tmp_score
                no_improve = 0

        # early termination if reached target
        if best_score < target:
            break

        # optional restart from random if no improvement for long
        if no_improve > 8000:
            cur = a_rand.copy()
            cur_score = _score(cur, L)
            momentum = np.zeros_like(cur)
            no_improve = 0

    # after loop, return best_vec as list
    return [float(x) for x in best_vec]

def propose_candidate(seed=42,budget_s=1100,**kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0,float(budget_s)-0.5)
    # length n
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
        # initial candidate from base, clip, project
        a0 = np.clip(base,0,1000)
        a0 = _project_simplex(a0)
    else:
        n = 2000
        a0 = rng.random(n)
        a0 = _project_simplex(a0)
    # second random init
    a_rand = rng.random(n)
    a_rand = _project_simplex(a_rand)
    # choose best among runs
    best_vec = None
    best_sc = float('inf')
    # run two phases: first start with a0, then with a_rand
    for init_vec in [a0, a_rand]:
        cur = init_vec.copy()
        # compute FFT length L >= 3n-2
        L = 1 << ((3 * n - 2).bit_length())
        momentum = np.zeros_like(cur)
        step0 = 1.0
        beta = 0.85
        alpha0 = 1.0
        alpha_mul = 1.005
        max_alpha = 30.0
        it = 0
        no_improve = 0
        cur_sc = _score(cur, L)
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()
        while time.monotonic() < deadline:
            it += 1
            # compute conv and max
            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[:2*n-1]
            max_val = conv.max()
            # compute softmax weights
            alpha = min(max_alpha, alpha0 * (alpha_mul ** it))
            # Avoid overflow: shift by max_val
            shifted = conv - max_val
            w = np.exp(alpha * shifted)
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.size)
            else:
                w /= w_sum
            # gradient
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=L)
            grad_full = np.fft.irfft(a_fft * w_fft, n=L)
            idx = (2 * n - 2) - np.arange(n)
            g = 2.0 * grad_full[idx]
            # momentum
            momentum = beta * momentum + (1.0 - beta) * g
            # step size
            step = step0 / math.sqrt(it)
            # candidate update
            cand = cur - step * momentum
            cand = _project_simplex(cand)
            cand_sc = _score(cand, L)
            if cand_sc < cur_sc:
                cur, cur_sc = cand, cand_sc
                no_improve = 0
                if cand_sc < best_sc:
                    best_sc, best_vec = cand_sc, cand.copy()
            else:
                # accept with probability
                prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
                if rng.random() < prob:
                    cur, cur_sc = cand, cand_sc
                no_improve += 1
            # periodic symmetrize
            if it % 200 == 0:
                sym = (cur + cur[::-1]) * 0.5
                sym = _project_simplex(sym)
                sym_sc = _score(sym, L)
                if sym_sc < cur_sc:
                    cur, cur_sc = sym, sym_sc
                    if sym_sc < best_sc:
                        best_sc, best_vec = sym_sc, sym.copy()
                    no_improve = 0
            # restart if stuck
            if no_improve > 5000:
                # random jitter
                cur = init_vec.copy()
                cur = _project_simplex(cur + 0.01 * rng.standard_normal(n))
                cur = _project_simplex(cur)
                cur_sc = _score(cur, L)
                momentum = np.zeros_like(cur)
                no_improve = 0
            # early exit

                break
        # after finishing with this init, continue next init if time remains
        if time.monotonic() >= deadline:
            break
    # return best vector
    return [float(x) for x in (best_vec if best_vec is not None else cur)]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast internal score (assumes sum = 1): 2 * n * max(conv)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------
    # Initialise – use the parent if present, otherwise a random start
    # --------------------------------------------------------

        a0 = np.clip(base, 0.0, 1000.0)

    else:
        n = 2000
        a0 = rng.random(n, dtype=np.float64)
        a0 = _project_simplex(a0)

    # extra random initialisation for a possible restart
    a_rand = rng.random(n, dtype=np.float64)
    a_rand = _project_simplex(a_rand)

    # --------------------------------------------------------
    # Common hyper‑parameters
    # --------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())   # FFT length ≥ 3n‑2
    step0 = 1.0
    beta = 0.85                # momentum factor
    alpha0 = 1.0               # initial soft‑max sharpness
    alpha_mul = 1.005          # growth per iteration
    max_alpha = 30.0
    best_vec = None
    best_sc = float('inf')
    target = 1.50274349

    # --------------------------------------------------------
    # Run the optimiser from two different starts
    # --------------------------------------------------------
    for init in (a0, a_rand):
        cur = init.copy()
        momentum = np.zeros_like(cur)
        cur_sc = _score(cur, L)
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()
        it = 0
        no_improve = 0

        while time.monotonic() < deadline:
            it += 1

            # ----- convolution of cur with itself -----
            a_fft = np.fft.rfft(cur, n=L)
            conv_full = np.fft.irfft(a_fft * a_fft, n=L)
            conv = conv_full[: 2 * n - 1]
            max_val = conv.max()

            # ----- soft‑max weights over the convolution -----
            alpha = min(max_alpha, alpha0 * (alpha_mul ** it))
            shifted = conv - max_val                 # numerical stability
            w = np.exp(alpha * shifted)

            if w_sum == 0.0:

    # ------------------------------------------------------------------

        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
