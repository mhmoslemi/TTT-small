    import time, math, numpy as np

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # initialization vector
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total <= 0:
            a = rng.random(a.shape[0])
        else:
            a = a / total
        # symmetrize
        a = (a + a[::-1]) * 0.5
        # project to simplex to ensure exact sum=1
        a = _project_simplex(a)
    else:
        n = 1024
        a = rng.random(n)
        a = _project_simplex(a)

    n = a.size
    # FFT length power of two >= 2 * n
    L = 1 << ((2 * n - 1).bit_length())

    # Adam hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.2   # initial learning rate
    # Softmax temperature (beta)
    temp_beta = 20.0
    # For scaling conv differences
    scale_factor = n

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_seq = a.copy()
    # initial score
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    max_conv = conv_full.max()
    sum_a = a.sum()
    best_score = 2.0 * n * max_conv / (sum_a * sum_a)

    no_improve = 0
    iteration = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        iteration += 1

        # compute convolution and max
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]

import time, math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function minimizing the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small head‑room for final conversion
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----------------------- initialise -----------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = rng.random(a.shape[0])
        else:
            a = a / s
        a = (a + a[::-1]) * 0.5            # enforce symmetry
        a = _project_simplex(a)            # sum = 1
    else:
        n = 1024
        a = _project_simplex(rng.random(n))
    n = a.size

    # FFT length – power of two ≥ 2n‑1 (covers a * a and a * w)
    L = 1 << ((2 * n - 1).bit_length())

    # -------------------- optimiser state --------------------
    # Adam parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max “temperature” (will be increased gradually)
    beta_max = 30.0          # final sharpness
    beta_inc = 0.01          # per‑iteration increase factor

    # learning‑rate schedule
    base_lr = 0.4

    # bookkeeping
    best_seq = a.copy()
    # initial score (objective with sum=1)
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_conv = conv.max()
    best_score = 2.0 * n * max_conv / (a.sum() ** 2)

    no_improve = 0
    it = 0
    target = 1.50274349

    # ----------------------- optimisation loop -----------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and objective -----
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_conv = conv.max()
        sum_a = a.sum()
        cur_score = 2.0 * n * max_conv / (sum_a * sum_a)

        # ----- record best -----
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- early exit if target reached -----
        if best_score < target:
            break

        # ----- soft‑max weights for the (smoothed) max -----
        beta = min(beta_max, beta_inc * it)          # growing sharpness
        diff = conv - max_conv                       # ≤ 0
        # scaling by n puts the argument of exp in O(1)
        w = np.exp(beta * diff * n)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]                     # ∇ objective (up to constant)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = base_lr / math.sqrt(it)                # simple decay
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- take step, enforce constraints -----
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)               # non‑negativity
        a_new = _project_simplex(a_new)              # sum = 1
        a_new = (a_new + a_new[::-1]) * 0.5          # symmetry
        a_new = _project_simplex(a_new)              # re‑normalise

        a = a_new

        # ----- occasional random perturbation when stuck -----
        if no_improve > 4000:
            # perturb around the best known point
            a = best_seq.copy()
            a += rng.normal(scale=0.02, size=n)
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            a = (a + a[::-1]) * 0.5
            a = _project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

    # --------------------- final safety checks ---------------------
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    if best_seq.sum() < 0.01:                       # should never happen
        best_seq = np.full(n, 1.0 / n)

    # return as plain Python list
    return [float(x) for x in best_seq]
