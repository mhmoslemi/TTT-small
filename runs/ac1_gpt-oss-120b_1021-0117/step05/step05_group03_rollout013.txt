def propose_candidate(...):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0, budget_s - 0.5)

    if 'height_sequence_1' in globals(): a0 = np.asarray(height_sequence_1, dtype=float).copy()
    else: a0 = rng.random(1024)

    a0 = np.maximum(a0, 0)
    a0 = _project_simplex(a0)
    cur = a0.copy()
    n = cur.size
    L = 1 << ((3*n - 2).bit_length())

    def _score(arr):
        # compute conv via FFT, but we may reuse some computed conv? We'll compute from scratch each time maybe okay.
        arr_fft = np.fft.rfft(arr, n=L)
        conv_full = np.fft.irfft(arr_fft * arr_fft, n=L)
        conv = conv_full[:2*n-1]
        return float(2.0 * n * conv.max())

    # compute first score
    best = cur.copy()
    best_sc = _score(best)
    cur_sc = best_sc

    # parameters
    step0 = 0.5
    beta = 0.9
    momentum = np.zeros_like(cur)
    tau0 = 0.5
    tau_decay = 0.9995
    min_tau = 1e-4
    iteration = 0
    no_improve = 0
    target = 1.50274349
    sym_period = 300
    backtrack_factors = [1., 0.7, 0.5, 0.3, 0.1]
    # random mass transfer strategy after stagnation

    while time.monotonic() < deadline:
        iteration += 1
        # compute convolution and gradient weight w
        cur_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(cur_fft * cur_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        # softmax weights
        tau = max(min_tau, tau0 * (tau_decay ** iteration))
        # compute exps
        exps = np.exp((conv - max_val) / tau)
        w = exps / exps.sum()
        # compute gradient
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(cur_fft * w_rev_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1 - beta) * g

        # step size
        alpha = step0 / math.sqrt(iteration)

        # line search
        cand = None
        cand_sc = None
        for factor in backtrack_factors:
            step = alpha * factor
            tmp = cur - step * momentum
            tmp = _project_simplex(tmp)
            sc = _score(tmp)
            if cand_sc is None or sc < cand_sc:
                cand, cand_sc = tmp, sc
            if sc < cur_sc:
                break

        # accept
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
        else:
            # uphill acceptance with probability
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # periodic symmetrization
        if iteration % sym_period == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # random mass transfer if stuck
        if no_improve > 3000:
            # random small redistribution
            i = rng.integers(0, n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # early exit
        if best_sc < target:
            break

    # return best list
    return [float(x) for x in best]

import time, math, numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1) > 0
    if not rho_arr.any():
        theta = 0.0
    else:
        rho = np.where(rho_arr)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(v, 1.0 / n)
    else:
        w /= s
    return w


# ------------------------------------------------------------
# Fast AC1 bound for a vector whose sum is 1
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * a.shape[0] - 1]
    return float(2.0 * a.shape[0] * conv.max())

    while time.monotonic() < deadline:

        conv = conv_full[: 2 * n - 1]

        if sc < best_sc:
