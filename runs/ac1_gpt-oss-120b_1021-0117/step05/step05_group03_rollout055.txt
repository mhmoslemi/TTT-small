import time
import math

# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        conv = conv_full[: 2 * n - 1]

        max_val = conv.max()

        # ----- smooth (log‑sum‑exp) weights -----
        exp_term = np.exp(beta_smooth * (conv - max_val))
        w = exp_term / exp_term.sum()

        # ----- gradient for smooth max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        g_smooth = 2.0 * grad_full[idx]

        # ----- hard‑max (sub‑gradient) weights -----
        mask = np.isclose(conv, max_val, atol=1e-12)
        w_mask = mask.astype(np.float64)
        if w_mask.sum() == 0.0:
            w_mask[:] = 1.0 / w_mask.size
        else:
            w_mask /= w_mask.sum()
        w_mask_rev = w_mask[::-1]
        w_mask_fft = np.fft.rfft(w_mask_rev, n=L)
        grad_full_hard = np.fft.irfft(a_fft * w_mask_fft, n=L)
        g_hard = 2.0 * grad_full_hard[idx]

        # ----- Adam step (same for both directions) -----
        m = beta1 * m + (1.0 - beta1) * g_smooth
        v = beta2 * v + (1.0 - beta2) * (g_smooth * g_smooth)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        step = step / math.sqrt(it)                     # simple decay

        # ----- candidate from smooth gradient -----
        cand_s = cur - step
        cand_s = np.maximum(cand_s, 0.0)
        cand_s = _project_simplex(cand_s)
        cand_s_sc = _score(cand_s, L)

        # ----- candidate from hard sub‑gradient -----
        cand_h = cur - step
        cand_h = np.maximum(cand_h, 0.0)
        cand_h = _project_simplex(cand_h)
        cand_h_sc = _score(cand_h, L)

        # ----- pick best of the three (current, smooth, hard) -----
        cand, cand_sc = cur, cur_sc
        if cand_s_sc < cand_sc:
            cand, cand_sc = cand_s, cand_s_sc
        if cand_h_sc < cand_sc:
            cand, cand_sc = cand_h, cand_h_sc

        # ----- accept with occasional uphill move -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            no_improve += 1
            if rng.random() < 0.001:          # tiny chance to move uphill
                cur, cur_sc = cand, cand_sc

        # ----- increase smoothing temperature -----
        if beta_smooth < beta_max:
            beta_smooth = min(beta_max, beta_smooth * smooth_inc)

        # ----- periodic symmetrisation (helps with Toeplitz structure) -----
        if it % sym_period == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # ----- random perturbation when stuck -----
        if no_improve > max_no_improve:
            pert = cur.copy()
            k = max(1, n // 100)                 # perturb up to 1 % of entries
            idxs = rng.choice(n, size=k, replace=False)
            noise = rng.standard_normal(k) * 0.01
            pert[idxs] = np.maximum(pert[idxs] + noise, 0.0)
            pert = _project_simplex(pert)
            pert_sc = _score(pert, L)
            if pert_sc < cur_sc:
                cur, cur_sc = pert, pert_sc
                if pert_sc < best_sc:
                    best_vec, best_sc = pert.copy(), pert_sc
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
