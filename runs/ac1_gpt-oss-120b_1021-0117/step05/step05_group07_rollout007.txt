def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    # Normalize to sum=1
    s = cur.sum()
    if s == 0:
        cur[:] = 1.0 / cur.size
    else:
        cur /= s
    # enforce symmetry
    cur = (cur + cur[::-1]) * 0.5
    cur = _project_simplex(cur)   # ensure sum=1

    n = cur.size
    L = 1 << ((3 * n - 2).bit_length())
    idx = (2 * n - 2) - np.arange(n)

    # evaluate initial score
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]
    max_val = conv_full.max()
    cur_sc = 2.0 * n * max_val
    best_sc = cur_sc
    best_vec = cur.copy()

    # Hyperparameters
    step0 = 0.6
    beta = 0.9
    momentum = np.zeros_like(cur)
    eps0 = 0.5
    eps_decay = 0.997
    eps_min = 1e-9
    it = 0
    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # convolution and max
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]
        max_val = conv_full.max()

        # temperature
        eps = max(eps_min, eps0 * (eps_decay ** it))

        if eps > 1e-12:
            w = np.exp((conv_full - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.size)
            else:
                w /= w_sum
        else:
            w = np.zeros_like(conv_full)
            mask = conv_full == max_val
            cnt = mask.sum()
            if cnt > 0:
                w[mask] = 1.0 / cnt

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # momentum and symmetrize
        momentum = beta * momentum + (1.0 - beta) * grad
        momentum = (momentum + momentum[::-1]) * 0.5

        # step size, backtracking
        base_alpha = step0 / math.sqrt(it)
        alpha = base_alpha
        cand = None
        cand_sc = None
        for _ in range(8):

            cand_tmp = (cand_tmp + cand_tmp[::-1]) * 0.5
            cand_tmp = _project_simplex(cand_tmp)

            cand_fft = np.fft.rfft(cand_tmp, n=L)
            cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[:2 * n - 1]
            cand_max = cand_conv.max()
            cand_sc = 2.0 * n * cand_max

                break

import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ---- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    s = cur.sum()
    cur = cur / s if s > 0 else np.full_like(cur, 1.0 / cur.size)

    # enforce symmetry from the start
    cur = (cur + cur[::-1]) * 0.5
    cur = _project_simplex(cur)

    n = cur.size
    L = 1 << ((3 * n - 2).bit_length())          # FFT length >= 3n‑2
    idx = (2 * n - 2) - np.arange(n)             # index map for gradient

    # ---- initial score ----------------------------------------------
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    cur_sc = 2.0 * n * conv_full.max()
    best_sc, best_vec = cur_sc, cur.copy()

    # ---- hyper‑parameters --------------------------------------------
    step0 = 0.8
    beta = 0.95
    momentum = np.zeros_like(cur)
    eps0 = 0.5
    eps_decay = 0.997
    eps_min = 1e-9
    it = 0
    no_improve = 0
    target = 1.50274349

    # ---- optimisation loop -------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # convolution of current point
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # temperature for soft‑max weighting
        eps = max(eps_min, eps0 * (eps_decay ** it))
        if eps > 1e-12:
            w = np.exp((conv_full - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.size)
            else:
                w /= w_sum
        else:
            w = np.zeros_like(conv_full)
            mask = conv_full == max_val
            cnt = mask.sum()
            if cnt:
                w[mask] = 1.0 / cnt

        # gradient of (soft) max
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # momentum update and enforce symmetry on the direction
        momentum = beta * momentum + (1.0 - beta) * grad
        momentum = (momentum + momentum[::-1]) * 0.5

        # backtracking line search
        base_alpha = step0 / math.sqrt(it)
        alpha = base_alpha
        cand = None
        cand_sc = None
        for _ in range(8):
            cand_tmp = cur - alpha * momentum
            cand_tmp = (cand_tmp + cand_tmp[::-1]) * 0.5
            cand_tmp = _project_simplex(cand_tmp)

            cand_fft = np.fft.rfft(cand_tmp, n=L)
            cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
            cand_sc = 2.0 * n * cand_conv.max()

                cand, cand_sc = cand_tmp, cand_sc
                break
            alpha *= 0.5

        if cand is None:               # no improvement in line search
            cand, cand_sc = cand_tmp, cand_sc

        # accept/reject

            cur, cur_sc = cand, cand_sc
            no_improve = 0

        else:

            if rng.random() < prob:

            no_improve += 1

        if no_improve > 3000:

        if no_improve > 5000:
            momentum.fill(0.0)

        # early exit
        if best_sc < target:
            break

        if time.monotonic() >= deadline:
            break

    # return plain Python list
    return [float(x) for x in best_vec]
