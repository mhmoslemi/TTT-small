def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)
    # init vector
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
    else:
        n = 2000  # arbitrary
        cur = rng.random(n, dtype=np.float64)
    cur = _normalize(cur)
    n = cur.size
    L = _next_pow2(2*n - 1)
    # optimizer state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    eta0 = 0.5  # base learning rate for exponentiated update
    it = 0
    best_vec = cur.copy()
    max_val, conv, max_idx = _self_conv_max(cur, L)
    best_score = 2.0 * n * max_val
    target_score = 1.50274349
    # cooling schedule for epsilon in softmax (if we use)
    eps_soft = 0.5
    eps_decay = 0.9995
    # record no improvement count
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1
        # compute conv and max
        max_val, conv, max_idx = _self_conv_max(cur, L)
        # shape weight w: either smoothed or one-hot
        # Use softmax when epsilon hasn't decayed too much (say eps_soft > 1e-3)
        eps = max(1e-12, eps_soft * (eps_decay ** it))
        if eps > 1e-4:
            # softmax weight
            # compute w as exp((conv - max_val)/eps)
            w = np.exp((conv - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w = w / w_sum
        else:
            # one-hot at max index
            w = np.zeros_like(conv)
            w[max_idx] = 1.0
        # gradient via FFT
        a_fft = np.fft.rfft(cur, n=L)
        grad = _gradient_fft(a_fft, w, n, L)
        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        # bias correction
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        # step size
        eta = eta0 / np.sqrt(it)  # decays
        # exponentiated gradient factor
        exp_factor = - eta * m_hat / (np.sqrt(v_hat) + eps_adam)
        # clip
        exp_factor = np.clip(exp_factor, -50.0, 50.0)
        cand = cur * np.exp(exp_factor)
        cand = _normalize(cand)
        cand_max, _, _ = _self_conv_max(cand, L)
        # acceptance: accept if improves else occasionally accept
        if cand_max < max_val - 1e-15:
            cur = cand
            max_val = cand_max
            no_improve = 0
            if cand_max < best_score / (2.0 * n):
                best_vec = cand.copy()
                best_score = 2.0 * n * cand_max
        else:
            no_improve += 1
            # Metropolis acceptance
            prob = math.exp(-(cand_max - max_val) / (eta + 1e-12))
            if rng.random() < prob:
                cur = cand
                max_val = cand_max
        # periodic symmetrization can help
        if it % 300 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _normalize(sym)
            sym_max, _, _ = _self_conv_max(sym, L)
            if sym_max < max_val:
                cur = sym
                max_val = sym_max
                if sym_max < best_score / (2.0 * n):
                    best_vec = sym.copy()
                    best_score = 2.0 * n * sym_max
                no_improve = 0
        # random mass transfer if stuck
        if no_improve > 2000:
            # pick two indices random
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _normalize(tmp)
                tmp_max, _, _ = _self_conv_max(tmp, L)
                if tmp_max < max_val:
                    cur = tmp
                    max_val = tmp_max
                    if tmp_max < best_score / (2.0 * n):
                        best_vec = tmp.copy()
                        best_score = 2.0 * n * tmp_max
                    no_improve = 0
                else:
                    no_improve = 0
        # early exit if target reached
        if best_score < target_score:
            break

    # return best found
    return [float(x) for x in best_vec]

import time, math, numpy as np

# ------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ------------------------------------------------------------
def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale non‑negative vector to have sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_and_fft(v: np.ndarray, L: int):
    """
    Compute linear self‑convolution v*v via FFT.
    Returns (max_val, full_conv, argmax, rfft_of_v).
    """
    a_fft = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * v.size - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), conv, idx, a_fft


def _gradient_fft(a_fft: np.ndarray, w: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of Σ_i w_i·(v*v)[i] w.r.t. v.
    Implements the matrix‑free formula using FFT.
    """
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)          # positions for gradient components
    return 2.0 * grad_full[idx]


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {x≥0, Σx=1} (O(n log n))."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
