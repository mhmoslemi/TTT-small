import time
import math
import numpy as np

# ------------------------------------------------------------
# Basic utilities
# ------------------------------------------------------------
def _normalize(v: np.ndarray) -> np.ndarray:
    """Return a non‑negative vector scaled to have sum = 1."""
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _conv_max_fft(a: np.ndarray, L: int):
    """
    Compute linear convolution a*a via FFT, return
    (max value, full convolution slice, index of max, FFT of a).
    """
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    idx = int(conv.argmax())
    return float(conv.max()), conv, idx, a_fft


def _subgradient_max(a: np.ndarray, k_max: int):
    """
    Sub‑gradient of the maximum convolution term.
    For each i: g_i = 2*a_{k_max-i} if that index exists,
    and g_i = a_i when i == k_max-i (diagonal term counted once).
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k_max - i
    mask = (j >= 0) & (j < n)
    grad[mask] = 2.0 * a[j[mask]]
    diag = mask & (i == j)
    grad[diag] = a[i[diag]]          # replace 2*a_i by a_i on the diagonal
    return grad
