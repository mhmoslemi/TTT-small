# Convolution utilities (FFT‑based, matrix‑free)

def _conv_and_fft(a: np.ndarray, L: int):
    """Return linear convolution a*a (length 2·len(a)-1) and its FFT."""
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * a.size - 1]
    return conv, a_fft


def _gradient_from_weights(a_fft: np.ndarray, w_rev: np.ndarray, n: int, L: int):
    """Gradient of weighted sum Σ w_i (a*a)_i w.r.t. a (matrix‑free)."""
    w_rev_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_rev_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)        # selects the n entries matching a
    return 2.0 * grad_full[idx]

        a = rng.random(2048, dtype=np.float64)

    # Normalise to a simplex (sum = 1)
    s = a.sum()
    if s == 0.0:
        a.fill(1.0 / a.size)
    else:
        a /= s

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # next power‑of‑two ≥ 2n‑1

    # --------------------------------------------------------------
    # Adam optimiser state
    # --------------------------------------------------------------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.05

    beta1_pow = 1.0
    beta2_pow = 1.0

    # --------------------------------------------------------------
    # Smoothing (log‑sum‑exp) schedule
    # --------------------------------------------------------------
    alpha = 10.0            # start temperature (low = smooth)
    alpha_max = 20000.0
    alpha_mul = 1.001

    # --------------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------------
    best_a = a.copy()
    conv, _ = _conv_and_fft(a, L)
    best_max = conv.max()
    best_score = 2.0 * n * best_max
    cur_max = best_max
    no_improve = 0
    it = 0

    # --------------------------------------------------------------
    # Optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- current convolution & smoothed weights ----
        conv, a_fft = _conv_and_fft(a, L)
        cur_max = conv.max()

        # log‑sum‑exp smoothing (stable)
        max_c = cur_max
        exp_vals = np.exp(alpha * (conv - max_c))
        sum_exp = exp_vals.sum()
        w = exp_vals / sum_exp                       # softmax weights

        # ---- gradient (matrix‑free) ----
        w_rev = w[::-1]
        grad = _gradient_from_weights(a_fft, w_rev, n, L)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = lr0 * math.sqrt(1.0 - beta2_pow) / (1.0 - beta1_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new == 0.0:
            a_new.fill(1.0 / n)
        else:
            a_new /= s_new

        # ---- occasional symmetrisation (helps Toeplitz structure) ----
        if it % 250 == 0:
            a_new = (a_new + a_new[::-1]) * 0.5
            a_new /= a_new.sum()

        # ---- evaluate new candidate ----
        conv_new, _ = _conv_and_fft(a_new, L)
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        # ---- acceptance test ----
        if new_max < cur_max - 1e-14:
            a = a_new
            cur_max = new_max
            no_improve = 0
        else:
            no_improve += 1
            # occasional uphill move (tiny probability)
            if rng.random() < 0.001:
                a = a_new
                cur_max = new_max

        # ---- record best ever ----
        if new_max < best_max:
            best_max = new_max
            best_score = new_score
            best_a = a_new.copy()
            no_improve = 0

        # ---- stagnation escape ----
        if no_improve > 4000:
            i = rng.integers(0, n)
            if a[i] > 1e-12:
                delta = a[i] * rng.random() * 0.5
                j = rng.integers(0, n)
                a_pert = a.copy()
                a_pert[i] -= delta
                a_pert[j] += delta
                a_pert = np.maximum(a_pert, 0.0)
                s_pert = a_pert.sum()
                a_pert /= s_pert
                a = a_pert
                # reset counters
                no_improve = 0
                # recompute current max after perturb
                conv, _ = _conv_and_fft(a, L)
                cur_max = conv.max()
                if cur_max < best_max:
                    best_max = cur_max
                    best_score = 2.0 * n * best_max
                    best_a = a.copy()

        # ---- anneal smoothing parameter ----
        alpha = min(alpha_max, alpha * alpha_mul)

        # ---- early exit if target reached ----
        if best_score < 1.50274349:
            break

    # Return the best vector found (plain Python list)
    return [float(x) for x in best_a]
