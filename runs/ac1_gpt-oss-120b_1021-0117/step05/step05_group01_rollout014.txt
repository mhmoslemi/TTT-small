def compute_subgradient(a, conv, eps=1e-12):
    n = a.size
    inds = np.where(conv >= conv.max() - eps)[0]
    g = np.zeros_like(a)
    for k in inds:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # for i from i_start to i_end:
        # contribution 2*a[k - i]
        # can vectorize:
        i = np.arange(i_start, i_end + 1)
        g[i] += 2.0 * a[k - i]
    return g / max(1, len(inds))
```

import time, math, numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x) = z}."""
    if z < 0:
        raise ValueError("z must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Convolution (full) maximum via FFT
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    """Return max_k (a*a)[k] using an FFT of length L (>=2n‑1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())


# ------------------------------------------------------------
# Sub‑gradient of the max‑convolution (all active indices)
# ------------------------------------------------------------
def _grad_for_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. a (averaged over all maximal indices)."""
    n = a.size
    max_val = conv.max()
    # treat values within 1e-12 of the maximum as active
    eps = 1e-12
    idxs = np.where(conv >= max_val - eps)[0]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        i = np.arange(i_start, i_end + 1, dtype=int)
        g[i] += 2.0 * a[k - i]
    if idxs.size > 0:
        g /= float(idxs.size)
    return g
