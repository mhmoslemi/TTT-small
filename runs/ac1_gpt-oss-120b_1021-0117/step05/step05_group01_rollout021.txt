import time, math, numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x) = z}."""
    n = v.size
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    n = a.size
    # FFT lengths
    L = 1 << ((2 * n - 1).bit_length())          # for a*a
    Lg = 1 << ((3 * n - 2).bit_length())         # for weighted gradient

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best_vec = a.copy()
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    best_sc = float(2.0 * n * conv_full.max())

    # optimisation parameters
    step0 = 0.8
    beta = 0.9
    momentum = np.zeros_like(a)
    tau = 0.1            # temperature for smooth max
    tau_inc = 0.0006
    target = 1.50274349
    no_improve = 0
    sym_interval = 250
    it = 0

    # ------------------------------------------------------------
    # Optimisation loop (matrix‑free, O(n log n) per iteration)
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and its score -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        cur_max = conv_full.max()
        cur_sc = 2.0 * n * cur_max

        # ----- keep the best seen so far -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- early exit if target reached -----
        if best_sc < target:
            break

        # ----- smooth‑max weights -----
        w = np.exp(tau * (conv_full - cur_max))   # max weight = 1, others <= 1

        # ----- gradient of weighted sum of convolutions -----
        a_rev = a[::-1]
        w_fft = np.fft.rfft(w, n=Lg)
        a_rev_fft = np.fft.rfft(a_rev, n=Lg)
        grad_full = np.fft.irfft(w_fft * a_rev_fft, n=Lg)
        start_idx = n - 1
        grad = 2.0 * grad_full[start_idx:start_idx + n]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- back‑tracking line search -----
        base_alpha = step0 / math.sqrt(it)
        alpha = base_alpha
        cand = None
        cand_sc = None
        for _ in range(6):
            a_tmp = a - alpha * momentum
            a_tmp = _project_simplex(a_tmp)          # stay on simplex
            a_tmp_fft = np.fft.rfft(a_tmp, n=L)
            tmp_full = np.fft.irfft(a_tmp_fft * a_tmp_fft, n=L)[: 2 * n - 1]
            cand_sc_tmp = 2.0 * n * tmp_full.max()
            if cand_sc_tmp <= cur_sc:
                cand, cand_sc = a_tmp, cand_sc_tmp
                break
            alpha *= 0.5

        if cand is None:          # no descent found – keep last trial
            cand, cand_sc = a_tmp, cand_sc_tmp

        # ----- accept / reject -----
        if cand_sc < cur_sc:
            a, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            # occasional uphill move (Metropolis style)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                a, cur_sc = cand, cand_sc

        # ----- periodic symmetrisation -----
        if it % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            a_sym_fft = np.fft.rfft(a_sym, n=L)
            sym_full = np.fft.irfft(a_sym_fft * a_sym_fft, n=L)[: 2 * n - 1]
            sym_sc = 2.0 * n * sym_full.max()
            if sym_sc < cur_sc:
                a, cur_sc = a_sym, sym_sc

        # ----- random mass‑transfer when stuck -----
        if no_improve > 2500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.3
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
            no_improve = 0

        # ----- temperature schedule -----
        tau = min(5.0, tau + tau_inc)

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    if best_vec.sum() < 0.01:
        best_vec = best_vec + 0.01 / best_vec.size
    best_vec /= best_vec.sum()
    return [float(x) for x in best_vec]
