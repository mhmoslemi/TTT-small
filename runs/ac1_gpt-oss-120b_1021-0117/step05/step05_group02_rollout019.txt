def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # initialize uniform length 1024 or maybe 2048 for speed?
        cur = np.full(1024, 1.0/1024, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)
    n = cur.size
    L = 1 << ((3*n - 2).bit_length())

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # Adam parameters
    lr = 0.2
    beta1 = 0.9
    beta2 = 0.999
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    pow_beta1 = 1.0
    pow_beta2 = 1.0

    # softmax temperature
    eps0 = 5.0
    eps_decay = 0.9995
    eps_min = 1e-8

    no_improve = 0
    it = 0
    target = 1.50274349

    # main loop
    while time.monotonic() < deadline:
        it += 1

        # FFT of current vector
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n - 1]
        max_val = conv.max()

        eps = max(eps_min, eps0 * (eps_decay ** it))
        # compute softmax weights
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        pow_beta1 *= beta1
        pow_beta2 *= beta2
        m_hat = m / (1.0 - pow_beta1)
        v_hat = v / (1.0 - pow_beta2)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        # multiplicative (mirror) update
        eta = lr * 0.5 / math.sqrt(it)  # decreasing
        exp_arg = -eta * g
        np.clip(exp_arg, -50, 50, out=exp_arg)
        cand2 = cur * np.exp(exp_arg)
        s = cand2.sum()
        if s > 0:
            cand2 /= s
        else:
            cand2 = cur.copy()
        cand2_sc = _score(cand2)

        # choose better
        if cand_sc <= cand2_sc:
            cand_best, cand_best_sc = cand, cand_sc
        else:
            cand_best, cand_best_sc = cand2, cand2_sc

        # acceptance
        if cand_best_sc < cur_sc:
            cur, cur_sc = cand_best, cand_best_sc
            no_improve = 0
            if cur_sc < best_sc:
                best_vec, best_sc = cur.copy(), cur_sc
        else:
            # occasional uphill acceptance
            prob = math.exp(-(cand_best_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand_best, cand_best_sc
            no_improve += 1

        # symmetrisation every 500

            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # random perturb when stuck
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n-1)
                if j >= i:

                tmp = cur.copy()

                tmp[j] += delta

                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        if best_sc < target:
            break

    return [float(x) for x in best_vec]

import time, math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – enforces a >= 0, sum(a) = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Scoring used by the evaluator (sum is assumed to be 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(convolution(a, a))."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
