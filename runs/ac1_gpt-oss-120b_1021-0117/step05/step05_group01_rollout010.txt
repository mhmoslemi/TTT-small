import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Convolution (via FFT) and its maximum
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int):
    """Return (max value, full convolution) of a with itself using length L FFT."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv_full.max()), conv_full


# ------------------------------------------------------------
# Sub‑gradient of the max‑convolution (averaged over active indices)
# ------------------------------------------------------------
def _grad_active(a: np.ndarray, conv: np.ndarray, act_idx: np.ndarray) -> np.ndarray:
    n = a.size
    g = np.zeros_like(a)
    for k in act_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        g[i] += 2.0 * a[k - i]          # derivative of a_i * a_{k-i}
    g /= float(len(act_idx))
    return g


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------------------
    # Initialise a
    # --------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-12:
            a[:] = 1.0
        a = a / a.sum()
    else:
        # fall‑back: exponential decay
        n = 2048
        beta = 3.0 / n
        idx = np.arange(n)
        a = np.exp(-beta * idx)
        a = a / a.sum()
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n‑1

    # --------------------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------------------
    best_vec = a.copy()
    max_b, _ = _conv_max(a, L)
    best_sc = 2.0 * n * max_b                         # sum(a) = 1
    momentum = np.zeros_like(a)

    # optimisation hyper‑parameters
    step0 = 0.8
    beta_m = 0.85
    no_improve = 0
    it = 0
    target = 1.50274349
    eps_active = 1e-12

    # --------------------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # --------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        max_b, conv = _conv_max(a, L)
        cur_sc = 2.0 * n * max_b

        # keep the best solution seen so far
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # early exit if we already beat the target
        if best_sc < target:
            break

        # active set: indices where the convolution attains (near) the maximum
        act = np.where(conv >= max_b - eps_active)[0]
        if act.size == 0:                     # extremely unlikely
            act = np.arange(conv.size)

        # sub‑gradient of the max‑convolution (average over active set)
        g = _grad_active(a, conv, act)

        # momentum update
        momentum = beta_m * momentum + (1.0 - beta_m) * g

        alpha = step0 / math.sqrt(it)
        cand = None

            a_tmp = _project_simplex(a_tmp)      # stay on simplex

            cand_sc = 2.0 * n * max_b_tmp
            if cand_sc <= cur_sc:

            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:

        else:

            idx = np.arange(n)
            a = np.exp(-beta_restart * idx)
            a = a / a.sum()
            momentum = np.zeros_like(a)
            no_improve = 0

    # --------------------------------------------------------------------
    # Return the best candidate found (plain Python list)
    # --------------------------------------------------------------------
    return [float(x) for x in best_vec]
