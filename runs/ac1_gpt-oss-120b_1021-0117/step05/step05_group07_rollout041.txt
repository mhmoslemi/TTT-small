import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence that (hopefully) lowers the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # sum = 1
    n = cur.size

    # ----- FFT helpers ------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())          # power of two >= 3n‑2
    idx = (2 * n - 2) - np.arange(n)             # indices for gradient extraction

    # ----- bookkeeping ------------------------------------------------
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv_full.max()
    cur_sc = best_sc = 2.0 * n * max_val
    best_vec = cur.copy()

    # ----- Adam parameters -------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05                       # base learning rate for Adam

    # ----- temperature schedule ---------------------------------------
    eps0, eps_decay, eps_min = 1.0, 0.9995, 1e-12

    # ----- loop control -----------------------------------------------
    it = 0
    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution ------------------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        cur_sc = 2.0 * n * max_val

        # ----- soft‑max weights ---------------------------------------
        eps = max(eps_min, eps0 * (eps_decay ** it))
        if eps > 1e-12:
            w = np.exp((conv_full - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.size)
            else:
                w /= w_sum
        else:
            w = np.zeros_like(conv_full)
            mask = conv_full == max_val
            cnt = mask.sum()
            if cnt > 0:
                w[mask] = 1.0 / cnt

        # ----- gradient of (soft) max ---------------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]                     # size n

        # ----- Adam update --------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        direction = lr * m_hat / (np.sqrt(v_hat) + eps_adam)   # descent direction

        # ----- line‑search (multiply step by 1, 1/2, …) ---------------
        cand = None
        cand_sc = None
        alpha = 1.0
        for _ in range(8):
            tmp = cur - alpha * direction
            tmp = _project_simplex(tmp)
            tmp_fft = np.fft.rfft(tmp, n=L)
            tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
            tmp_sc = 2.0 * n * tmp_conv.max()
            if tmp_sc <= cur_sc:
                cand, cand_sc = tmp, tmp_sc
                break
            alpha *= 0.5
        if cand is None:                     # no improvement after back‑track
            cand, cand_sc = tmp, tmp_sc

        # ----- accept / reject ----------------------------------------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            # occasional uphill move (Metropolis style)
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- random mass transfer when stuck ------------------------
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
                tmp_sc = 2.0 * n * tmp_conv.max()
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                no_improve = 0

        # ----- periodic symmetrisation (helps reduce peaks) ----------
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
            sym_sc = 2.0 * n * sym_conv.max()
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # ----- early exit ---------------------------------------------
        if best_sc < target:
            break

        # ----- keep time safety ---------------------------------------
        if time.monotonic() >= deadline:
            break

    return [float(x) for x in best_vec]
