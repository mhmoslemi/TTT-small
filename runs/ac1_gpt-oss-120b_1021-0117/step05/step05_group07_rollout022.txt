import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Produce a non‑negative step‑function (probability distribution) that
    analytically minimizes the AC1 upper bound.
    The optimal shape for a sequence of length n is a linearly increasing
    vector a_i ∝ i+1, normalised to sum = 1.
    """
    # Respect the time budget – the computation is trivial, but keep the pattern.
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1)

    # Length of the problem (use the parent if present).
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)

    # Linear increasing distribution: a_i = 2*(i+1) / (n*(n+1))
