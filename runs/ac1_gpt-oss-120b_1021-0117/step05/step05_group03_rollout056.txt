import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection – Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Score (exact evaluator) – 2 * n * max(conv) / (sum^2)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    max_b = np.max(conv_full[: 2 * n - 1])
    s = np.sum(a)
    if s == 0.0:
        return np.inf
    return 2.0 * n * max_b / (s * s)


# ------------------------------------------------------------
# Gradient of the smoothed max (soft‑max) of the convolution
# Returns gradient vector and the current max(convolution)
# ------------------------------------------------------------
def _grad(a: np.ndarray, T: float, L: int):
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]

    # soft‑max weights
    if T <= 1e-12:
        mask = (conv == conv.max())
        w = mask.astype(np.float64)
        ws = w.sum()
        if ws > 0:
            w /= ws
        else:
            w[:] = 1.0 / w.size
    else:
        maxc = conv.max()
        exps = np.exp((conv - maxc) / T)
        w = exps / exps.sum()

    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)

    idx = (2 * n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]
    return g, conv.max()

        # ----- accept / reject (with occasional uphill move) -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            # Metropolis‑style acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- periodic symmetrisation (helps with Toeplitz structure) -----
        if it % 1200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # ----- random restart when stuck for a long stretch -----
        if no_improve >= 4000:
            fn = init_funs[rng.integers(len(init_funs))]
            cur = fn(rng, n) if fn.__code__.co_argcount == 2 else fn()
            cur_sc = _score(cur, L)
            momentum.fill(0.0)
            no_improve = 0

        # early exit if target reached
        if best_sc < target:
            break

    # ----------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ----------------------------------------------------------------------
    return [float(x) for x in best_vec]
```
