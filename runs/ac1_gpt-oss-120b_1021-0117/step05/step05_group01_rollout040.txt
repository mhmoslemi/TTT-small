import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x) = z}."""
    n = v.size
    if z < 0:
        raise ValueError("z must be non‑negative")
    # sort v descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Fast max‑convolution via FFT (linear convolution)
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

# ------------------------------------------------------------
# Sub‑gradient of max‑convolution (average over all tying indices)
# ------------------------------------------------------------
def _subgrad_max(a: np.ndarray, conv: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    n = a.size
    max_val = conv.max()
    idxs = np.where(conv >= max_val - eps)[0]
    if idxs.size == 0:
        return np.zeros_like(a)
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        g[i] += 2.0 * a[k - i]
    g /= float(idxs.size)
    return g

# ------------------------------------------------------------
# Score (the quantity the evaluator returns, up to the denominator)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """2 * n * max(conv) (sum of a is assumed to be 1)."""
    n = a.size
    return 2.0 * n * _conv_max(a, L)

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 1.0)

    # ------------------------------------------------------------------
    # Initialise – use the parent if present, otherwise a uniform start
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2048, dtype=np.float64)

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                     # enforce sum = 1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())  # power‑of‑two FFT length

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best_vec = a.copy()
    best_sc = _score(a, L)

    # optimisation parameters
    beta = 0.85                     # momentum factor
    step0 = 0.4                     # initial step magnitude
    momentum = np.zeros_like(a)
    sym_interval = 250             # symmetrisation frequency
    no_improve = 0
    it = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    # Optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and score -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        cur_sc = 2.0 * n * conv.max()

        # ----- store the best seen so far -----
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- early exit if we already beat the benchmark -----
        if best_sc < target:
            break

        # ----- sub‑gradient of the max‑convolution -----
        g = _subgrad_max(a, conv)

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- back‑tracking line search -----
        base_alpha = step0 / math.sqrt(it)
        cand = None
        cand_sc = None
        for _ in range(9):                     # up to 9 attempts
            alpha = base_alpha * (0.5 ** _)
            a_tmp = a - alpha * momentum
            a_tmp = _project_simplex(a_tmp)     # stay on simplex
            cand_sc = _score(a_tmp, L)
            if cand_sc <= cur_sc:
                cand = a_tmp
                break

        # if no descent found, keep the last trial (might be uphill)
        if cand is None:
            cand = a_tmp
            cand_sc = cand_sc

        # ----- accept / reject -----
        if cand_sc < cur_sc:
            a = cand
            cur_sc = cand_sc
            no_improve = 0
        else:
            # occasional uphill move (Metropolis‑style)
            prob = math.exp(-(cand_sc - cur_sc) / (base_alpha + 1e-12))
            if rng.random() < prob:
                a = cand
                cur_sc = cand_sc

        # ----- periodic symmetrisation (helps with the Toeplitz structure) -----
        if it % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            sym_sc = _score(a_sym, L)

                a = a_sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = a_sym.copy()
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.35
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
                no_improve = 0

        # ----- occasional full restart if we have been stuck too long -----
        if no_improve > 9000:
            a = rng.random(n, dtype=np.float64)
            a = _project_simplex(a)
            momentum[:] = 0.0
            no_improve = 0

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
