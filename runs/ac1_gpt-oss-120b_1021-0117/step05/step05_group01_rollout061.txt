import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x) = z}."""
    n = v.size
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Convolution via FFT (returns full length and max index)
# ------------------------------------------------------------
def _conv_full(a: np.ndarray, L: int):
    """Linear convolution a*a using FFT length L (>=2n‑1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)
    return conv[: 2 * a.size - 1]

# ------------------------------------------------------------
def _grad_at_max(a: np.ndarray, max_idx: int) -> np.ndarray:
    """Gradient of the convolution entry attaining the maximum."""
    n = a.size
    idx = np.arange(n, dtype=np.int64)
    j = max_idx - idx

    g = np.zeros_like(a)
    # 2 * a[j] because each product appears twice (unless j==i)
    g[mask] = 2.0 * a[j[mask]]
    return g

        if cur_score < best_score:
            best_score, best_vec = cur_score, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if best_score < target:
            break

        # ----- exact gradient at the max entry -----
        g = _grad_at_max(a, max_idx)

        # ----- momentum -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- simple back‑tracking line search -----
        base_alpha = step0 / math.sqrt(it)
        alpha = base_alpha
        cand = None
        cand_score = None
        for _ in range(8):
            a_tmp = a - alpha * momentum
            a_tmp = _project_simplex(a_tmp)   # stay on simplex
            conv_tmp = _conv_full(a_tmp, L)
            cand_score = 2.0 * n * conv_tmp.max()
            if cand_score <= cur_score:
                cand = a_tmp
                break
            alpha *= 0.5
        if cand is None:                     # no descent found
            cand = a_tmp
            cand_score = cand_score if cand_score is not None else cur_score

        # ----- accept / reject (Metropolis style) -----
        if cand_score < cur_score:
            a = cand
        else:
            prob = math.exp(-(cand_score - cur_score) / (alpha + 1e-12))
            if rng.random() < prob:
                a = cand

        # ----- occasional symmetrisation (helps with the Toeplitz structure) -----
        if it % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            conv_sym = _conv_full(a_sym, L)
            sym_score = 2.0 * n * conv_sym.max()
            if sym_score < cur_score:
                a = a_sym

        # ----- random mass‑transfer when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.4
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
                no_improve = 0

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
```
