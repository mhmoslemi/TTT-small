# ------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum = 1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # re‑normalize to avoid tiny drift
    s = w.sum()
    if s == 0:
        return np.full_like(v, 1.0 / v.size)
    return w / s

# Max of full convolution a * a (computed via zero‑padded FFT)
# ------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    """Return max(conv_full) for a*a using FFT length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)
    return float(conv[: 2 * a.size - 1].max())
