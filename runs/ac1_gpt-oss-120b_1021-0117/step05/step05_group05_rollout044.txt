    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _conv_max(a, L):
    """Maximum value of the autocorrelation of a (via FFT)."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ---- initialization -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: random positive vector of moderate size
        a0 = rng.random(2048, dtype=np.float64) + 1e-3
    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.size
    # FFT length a bit larger than needed for the gradient step
    L = 1 << ((3 * n - 2).bit_length())

    best = cur.copy()
    best_sc = 2.0 * n * _conv_max(best, L)
    cur_sc = best_sc

    # ---- optimizer state (Adam) -----------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0
    no_improve = 0

    # hyper‑parameters
    base_lr = 0.5
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution & active mask ----------------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = float(conv.max())

        # active indices where the max is attained (within tolerance)
        tol = 1e-12
        active = conv >= max_val - tol
        w = active.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt

        # ----- gradient of the max (subgradient) -------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]                     # subgradient

        # ----- Adam update -----------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = base_lr / (math.sqrt(it) + 1.0)
        delta = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - delta
        cand = _project_simplex(cand)
        cand_sc = 2.0 * n * _conv_max(cand, L)

        # ----- accept / reject --------------------------------------------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
        else:
            # occasional stochastic acceptance to escape plateaus
            if rng.random() < 1e-4:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- random mass exchange on long stagnation --------------------
        if no_improve > 2500 and time.monotonic() + 1.0 < deadline:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta_mass = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n)
                tmp = cur.copy()
                tmp[i] -= delta_mass
                tmp[j] += delta_mass
                tmp = _project_simplex(tmp)
                tmp_sc = 2.0 * n * _conv_max(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- occasional LP refinement ------------------------------------
        if it % 6000 == 0 and time.monotonic() + 5.0 < deadline:
            try:
                import cvxpy as cp
                g = cp.Variable(n, nonneg=True)
                a_par = cp.Parameter(n, value=cur, constant=True)
                conv_expr = cp.conv(a_par, g)          # length 2n‑1
                prob = cp.Problem(cp.Maximize(cp.sum(g)), [conv_expr <= max_val])
                prob.solve(solver=cp.ECOS, warm_start=True,
                           max_iters=2000, verbose=False)
                if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                    gv = np.asarray(g.value, dtype=np.float64)
                    if gv is not None and gv.sum() > 0.0:
                        gv = gv / gv.sum()
                        gv = _project_simplex(gv)
                        gv_sc = 2.0 * n * _conv_max(gv, L)
                        if gv_sc < best_sc:
                            best, best_sc = gv.copy(), gv_sc
                            cur, cur_sc = gv.copy(), gv_sc
                            no_improve = 0
            except Exception:
                pass

        # ----- early exit if target reached --------------------------------
        if best_sc < target:
            break

    # Return as a plain Python list of floats
    return [float(x) for x in best]
