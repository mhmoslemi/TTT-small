import time, math, numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Simple (non‑negative) simplex projection: clip negative entries,
    then renormalise to make the sum equal to 1.  Falls back to uniform
    distribution if the sum is zero."""
    w = np.maximum(v, 0.0)
    s = w.sum()
    if s <= 0.0:
        # uniform fallback – never happens for a reasonable start vector
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return w / s


def _score(a: np.ndarray) -> float:
    """Exact objective value used by the evaluator (sum of a is assumed 1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
