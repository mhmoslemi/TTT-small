import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (L2 onto {x>=0, sum=1})
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Maximum value of the (linear) convolution a * a (full, length 2n‑1)
# ----------------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    """Return max(conv_full) for a*a using zero‑padded FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv_full.max())


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty, non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------------
    # Initialise – use provided parent if available, otherwise random
    # --------------------------------------------------------------

        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2048, dtype=np.float64)
    a = np.maximum(a, 0.0)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())          # power‑of‑two FFT length

    # normalise to the simplex (sum = 1) – optimisation works on this scale
    a = _project_simplex(a)

    # --------------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------------
    best = a.copy()
    best_sc = _max_conv(best, L)
    best_bound = 2.0 * n * best_sc                 # sum == 1 ⇒ denominator = 1

    # --------------------------------------------------------------
    # Adam‑style gradient descent on the smoothed max‑conv objective
    # --------------------------------------------------------------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.5
    eps_opt = 1e-8          # for Adam denominator
    eps_soft = 0.2          # soft‑max temperature
    decay = 0.9995
    it = 0
    target = 1.50274349

    while time.monotonic() < deadline - 2.0:
        it += 1

        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # soft‑max over the convolution entries
        eps = max(eps_opt, eps_soft * (decay ** it))
        shift = conv_full - max_val
        w = np.exp(shift / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** it)

        if s > 0.0:

                best_sc = sym_sc

    vals = np.sort(best)[::-1]
    perm = np.empty_like(vals)
    left, right = 0, n - 1
    k = 0
    while left <= right:

        if left < right:

        best_sc = perm_sc

        if new_bound < cur_bound:

            best = cur.copy()
            best_sc = new_sc
            best_bound = new_bound
        else:
            cur[i], cur[j] = cur[j], cur[i]

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    if best.size == 0:
        best = np.array([1.0])
    return [float(x) for x in best]
