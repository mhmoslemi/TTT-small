import time, math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Gradient of the max‑convolution entry (exact sub‑gradient)
# ------------------------------------------------------------
def _grad_max(a: np.ndarray, k: int) -> np.ndarray:
    """Gradient of (a*a)[k] w.r.t. a."""
    n = a.size
    g = np.zeros_like(a)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    i = np.arange(start, end + 1, dtype=int)
    j = k - i
    g[i] += a[j]
    g[j] += a[i]
    return g
