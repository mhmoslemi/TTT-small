import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Evaluate the AC1 bound (same formula as the external evaluator)
# ------------------------------------------------------------
def _objective(a: np.ndarray, L: int, n: int) -> float:
    """Return 2 * n * max(conv(a, a)) / (sum(a)**2)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_conv = conv.max()
    s = a.sum()
    # s is never zero because we keep sum = 1
    return 2.0 * n * max_conv / (s * s)

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence with a low AC1 bound.
    Runs for up to `budget_s` seconds and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the harness timeout
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initialise – use the parent if present, otherwise a uniform start

    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)               # enforce sum = 1
    n = cur.shape[0]

    # --------------------------------------------------------
    # FFT length (power of two) for full convolution
    # --------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------
    best_vec = cur.copy()
    best_sc = _objective(best_vec, L, n)
    cur_sc = best_sc

    # --------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------
    eta0 = 0.2               # base step size (decays as 1/sqrt(t))
    beta = 0.9               # momentum coefficient
    momentum = np.zeros_like(cur)
    no_improve = 0
    it = 0
    target = 1.50274349

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        af = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
        max_val = conv.max()
        k_max = int(conv.argmax())
        cur_sc = 2.0 * n * max_val / (cur.sum() * cur.sum())

        # ----- keep global best -----
        if cur_sc < best_sc:

            best_vec = cur.copy()
            if best_sc < target:

            no_improve = 0

        idx = np.arange(n)

            cur = rng.gamma(0.5, 1.0, size=n)
            cur = _project_simplex(cur)
            no_improve = 0
            # recompute current score
            cur_sc = _objective(cur, L, n)

    # ------------------------------------------------------------------
    # Return the best candidate found (as a plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
