import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a: np.ndarray) -> float:
    """Fast evaluation of the AC1 bound (sum of a assumed to be 1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    base_lr = 0.20          # base learning rate for Adam
    base_eta = 0.50         # base step for exponentiated (mirror) update
    eps0 = 0.50             # initial smoothing for the soft‑max
    eps_decay = 0.9995
    eps_min = 1e-8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    target = 1.50274349

    m = np.zeros_like(cur)   # first moment (Adam)
    v = np.zeros_like(cur)   # second moment (Adam)
    it = 0
    no_improve = 0

    # --------------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and max -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val

        # ----- smooth approximation of the max -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]              # ∇ max(conv)

        # ----- Adam additive update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step_factor = base_lr / math.sqrt(it)
        step_vec = step_factor * m_hat / (np.sqrt(v_hat) + eps_adam)
        cand_add = cur - step_vec
        cand_add = _project_simplex(cand_add)
        cand_add = (cand_add + cand_add[::-1]) * 0.5
        cand_add = _project_simplex(cand_add)

        # ----- mirror (multiplicative) update -----
        eta = base_eta / math.sqrt(it)
        exp_arg = -eta * g
        np.clip(exp_arg, -50, 50, out=exp_arg)
        cand_mul = cur * np.exp(exp_arg)
        s = cand_mul.sum()
        if s > 0.0:
            cand_mul /= s
        else:
            cand_mul = cur.copy()
        cand_mul = (cand_mul + cand_mul[::-1]) * 0.5
        cand_mul = _project_simplex(cand_mul)

        # ----- evaluate both candidates (FFT‑based) -----
        cand_fft = np.fft.rfft(cand_add, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_add_sc = 2.0 * n * cand_conv.max()

        cand_fft = np.fft.rfft(cand_mul, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_mul_sc = 2.0 * n * cand_conv.max()

        if cand_add_sc <= cand_mul_sc:
            cand, cand_sc = cand_add, cand_add_sc
        else:
            cand, cand_sc = cand_mul, cand_mul_sc

        # ----- accept / reject -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
                if best_sc < target:
                    break
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (step_factor + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- occasional random perturbation when stuck -----
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp = (tmp + tmp[::-1]) * 0.5
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
                tmp_sc = 2.0 * n * tmp_conv.max()
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                    if best_sc < target:
                        break

    # --------------------------------------------------------------------
    # Return the best sequence discovered
    # --------------------------------------------------------------------
    return [float(x) for x in best_vec]
