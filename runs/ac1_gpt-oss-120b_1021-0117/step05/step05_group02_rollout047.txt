import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – sum = 1, non‑negative
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Score used by the evaluator – assumes sum = 1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return the bound 2 * n * max(convolution(a, a))."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence with a low AC1 bound.
    Runs up to `budget_s` seconds and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initialise – use the provided parent if present, else uniform
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
    else:
        n = 2048
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    cur = _project_simplex(cur)
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------
    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # --------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------
    step0 = 1.5                # base step size (scaled by 1/√t)
    eps0 = 1e-3                # initial smoothing for soft‑max
    eps_decay = 0.9995
    eps_min = 1e-12
    top_k = 5                  # fallback: number of top autocorr entries
    stagnate = 0
    stagnate_limit = 500
    sym_interval = 300
    target = 1.50274349
    it = 0

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1
        # ----------- compute autocorrelation & smoothing weights ----------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # smoothing parameter (decreases)
        eps = max(eps_min, eps0 * (eps_decay ** it))

        if eps > 1e-8:
            # soft‑max weights (smooth approximation of max)
            w = np.exp((conv - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.size)
            else:
                w /= w_sum
        else:
            # indicator on the top‑k values (sub‑gradient)
            w = np.zeros_like(conv)
            if top_k >= conv.size:
                w[:] = 1.0 / conv.size
            else:
                idx_top = np.argpartition(conv, -top_k)[-top_k:]
                w[idx_top] = 1.0 / top_k

        # ----------- gradient of the smoothed objective ---------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----------- line‑search step ---------------------------------
        step = step0 / math.sqrt(it)
        cand = cur - step * g
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        # backtrack if no improvement
        backtrack_cnt = 0
        while cand_sc > cur_sc and backtrack_cnt < 8 and step > 1e-12:
            step *= 0.5
            cand = cur - step * g
            cand = _project_simplex(cand)
            cand_sc = _score(cand)
            backtrack_cnt += 1

        # ----------- accept / reject ---------------------------------
        improved = False
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            improved = True
        else:
            # occasional Metropolis‑like uphill acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
                improved = True

        # ----------- keep global best ---------------------------------
        if cur_sc < best_sc:
            best_vec, best_sc = cur.copy(), cur_sc
            stagnate = 0
        else:
            stagnate += 0 if improved else 1

        # ----------- periodic symmetrisation -------------------------
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                stagnate = 0

        # ----------- random mass‑transfer when stuck -----------------
        if stagnate > stagnate_limit:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                stagnate = 0

        # ----------- early exit ---------------------------------------
        if best_sc < target:
            break

    # ------------------------------------------------------------
    # Return best candidate as plain Python list
    # ------------------------------------------------------------
    return [float(x) for x in best_vec]
