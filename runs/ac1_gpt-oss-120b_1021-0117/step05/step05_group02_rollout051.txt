import time
import math

# ----------------------------------------------------------------------
# Simplex projection (onto {x>=0, sum(x)=1})
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    max_val = conv_full[: 2 * n - 1].max()
    return float(2.0 * n * max_val)

                best_vec, best_score = new_vec.copy(), new_sc
        else:
            prob = math.exp(-(new_sc - cur_score) / (T + 1e-12))
            if rng.random() < prob:
                cur, cur_score = new_vec, new_sc

        T *= 0.9999  # cool down

        # occasional extra symmetrisation
        if rng.random() < 0.001:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score_vec(sym, n, L)
            if sym_sc < cur_score:
                cur, cur_score = sym, sym_sc
                if sym_sc < best_score:
                    best_vec, best_score = sym.copy(), sym_sc

    # --------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # --------------------------------------------------------------
    return [float(x) for x in best_vec]
```
