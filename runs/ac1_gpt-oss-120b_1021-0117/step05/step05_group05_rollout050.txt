    v = np.asarray(v, dtype=np.float64)

    while time.monotonic() < deadline:
        # ----- gradient of max‑convolution -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # weights for the active (max) lags
        mask = conv >= max_val - 1e-12
        w = mask.astype(np.float64)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]               # sub‑gradient of max(conv)

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- Euclidean candidate (backtracking) -----
        step = 0.9 / math.sqrt(it + 1)
        alpha = step
        cand = None
        cand_sc = None
        for _ in range(4):
            cand_tmp = cur - alpha * momentum
            cand_tmp = _project_simplex(cand_tmp)
            cand_tmp_sc = _score(cand_tmp)
            if cand_tmp_sc <= cur_sc:
                cand, cand_sc = cand_tmp, cand_tmp_sc
                break
            alpha *= 0.5
        if cand is None:
            cand, cand_sc = cand_tmp, cand_tmp_sc

        # ----- Mirror‑descent style candidate -----
        eta = 0.6 / math.sqrt(it + 1)
        factor = np.exp(-eta * g)
        mir = cur * factor
        s = mir.sum()
        if s > 0:
            mir = mir / s
        mir_sc = _score(mir)

        # pick the better of the two proposals
        if mir_sc < cand_sc:
            cand, cand_sc = mir, mir_sc

        # ----- acceptance -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- periodic symmetrisation -----
        if it % 50 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # ----- random mass‑exchange when stuck -----
        if no_improve > 2000 and time.monotonic() + 1.0 < deadline:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            tmp = cur.copy()
            delta = tmp[i] * rng.random() * 0.5
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- early stop if target reached -----
        if best_sc < target:
            break

        it += 1

    # ------------------------------------------------------------------
    return [float(x) for x in best]
