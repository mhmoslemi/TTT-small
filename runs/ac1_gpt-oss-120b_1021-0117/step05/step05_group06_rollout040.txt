def _apply_move(conv, a, i, j, d):
    """Return new conv after moving mass d from i to j (a_i -= d, a_j += d). conv and a are original arrays."""
    n = a.size
    new = conv.copy()
    # delta_i = -d
    coeff_i = -2.0 * d
    # shift for i
    start = i
    end = i + n
    s0 = max(start, 0)
    e0 = min(end, 2*n-1)
    if s0 < e0:
        src_start = s0 - start
        new[s0:e0] += coeff_i * a[src_start:src_start + (e0 - s0)]
    # self term for i
    new[2*i] += d * d  # delta_i^2 = d^2 (since (-d)^2 = d^2)
    # delta_j = +d
    coeff_j = 2.0 * d
    start = j
    end = j + n
    s0 = max(start, 0)
    e0 = min(end, 2*n-1)
    if s0 < e0:
        src_start = s0 - start
        new[s0:e0] += coeff_j * a[src_start:src_start + (e0 - s0)]
    # self term for j
    new[2*j] += d * d

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper: projection onto the probability simplex (sum = 1, x>=0)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project the vector v onto the simplex {x>=0, sum(x)=1}."""
    if v.ndim != 1:
        v = v.ravel()
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# --------------------------------------------------------------
# Helper: maximal value of the full self‑convolution using FFT
# --------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    """Return max(conv_full) for a*a using zero‑padded FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv_full.max())


# --------------------------------------------------------------
# Helper: apply a mass transfer d from index i to index j
# --------------------------------------------------------------
def _apply_move(conv: np.ndarray, a: np.ndarray, i: int, j: int, d: float) -> np.ndarray:
    """
    Return a new convolution array after moving amount d from a[i] to a[j]
    (i.e. a[i]←a[i]-d, a[j]←a[j]+d).  The input `conv` must be the full
    convolution of the *old* vector `a`.  The returned array is a copy.
    """
    n = a.size
    new = conv.copy()

    # delta for i: -d (mass removed)
    coeff_i = -2.0 * d                     # factor 2 from a*δ + δ*a
    start = i
    end = i + n
    s0 = max(start, 0)
    e0 = min(end, 2 * n - 1)
    if s0 < e0:
        src_start = s0 - start
        new[s0:e0] += coeff_i * a[src_start:src_start + (e0 - s0)]

    new[2 * i] += d * d                    # self‑term (δ_i)^2

    # delta for j: +d (mass added)
    coeff_j = 2.0 * d
    start = j
    end = j + n
    s0 = max(start, 0)
    e0 = min(end, 2 * n - 1)
    if s0 < e0:
        src_start = s0 - start
        new[s0:e0] += coeff_j * a[src_start:src_start + (e0 - s0)]

    new[2 * j] += d * d                    # self‑term (δ_j)^2
    new[i + j] += -2.0 * d * d             # cross term 2·δ_i·δ_j = -2 d²
    return new
