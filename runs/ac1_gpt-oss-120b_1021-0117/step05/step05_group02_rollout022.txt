def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math, random
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1)

    # Initialize cur
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # choose length maybe 2048? but use 1024 as fallback
        n0 = 1024
        cur = np.full(n0, 1.0 / n0, dtype=np.float64)
    # Ensure nonnegative and simplex
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)   # sum=1
    # enforce symmetry
    cur = (cur + cur[::-1]) * 0.5
    cur = _project_simplex(cur)

    n = cur.size
    # FFT length
    L = 1 << ((3 * n - 2).bit_length())

    # Helper score using conv via FFT
    def compute_score(vec):
        # conv via FFT
        a_fft = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    cur_sc = compute_score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    # hyperparameters
    base_eta = 0.7
    no_improve = 0
    it = 0
    # tolerance for conv equality
    eq_tol = 1e-12

    while time.monotonic() < deadline:
        it += 1
        # Compute convolution and gradient
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()
        # Identify active indices within tolerance
        # use np.isclose with relative tolerance maybe 1e-12
        idx_max = np.where(np.isclose(conv, max_val, atol=1e-12, rtol=0.0))[0]
        # weight vector
        w = np.zeros_like(conv)
        if idx_max.size:
            w[idx_max] = 1.0 / idx_max.size
        else:
            # should not happen, but fallback to full uniform
            w[:] = 1.0 / conv.size

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # step size
        eta = base_eta / math.sqrt(it)

        # gradient descent step
        cand = cur - eta * g

        cand = _project_simplex(cand)

        cand_sc = compute_score(cand)

        # Accept better candidate

            no_improve = 0
            if cand_sc < best_sc:

        else:

def _project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
