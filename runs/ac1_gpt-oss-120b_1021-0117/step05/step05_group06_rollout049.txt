def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + float(budget_s) - 0.2

    # initial sequence
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # generate random uniform vector length 2000 maybe...
        base = rng.random(2048, dtype=np.float64)

    # ensure nonneg, clip
    base = np.clip(base, 0.0, 1000.0)

    # Create initial candidate: sort descending and anti-identity interleaving
    vals = np.sort(base)[::-1]
    n = vals.size
    seq = np.empty_like(vals)
    left, right = 0, n-1
    toggle = True
    idx = 0
    while left <= right:
        seq[left] = vals[idx]; idx+=1
        if left < right:
            seq[right] = vals[idx]; idx+=1
        left += 1
        right -= 1
    cur = seq.copy()
    # compute bound
    L = 1 << ((2 * n - 1).bit_length())  # ensure enough length; maybe 3n for safety
    # Actually using 1 << ((3 * n - 2).bit_length()) in prior code; we can similar.
    L = 1 << ((3 * n - 2).bit_length())
    def compute_bound(arr):
        max_conv = _max_conv(arr, L)
        s = arr.sum()
        if s < 1e-12:
            return np.inf
        return 2.0 * n * max_conv / (s * s)
    cur_bound = compute_bound(cur)
    best = cur.copy()
    best_bound = cur_bound

    # parameters for annealing
    T0 = 0.1 * best_bound
    decay = 0.9995
    T = T0
    it = 0

    # main loop
    while time.monotonic() < deadline:
        it += 1
        # randomly pick two distinct positions
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # swap
        cur[i], cur[j] = cur[j], cur[i]
        new_bound = compute_bound(cur)
        delta = new_bound - cur_bound
        accept = False
        if delta < 0:
            accept = True
        else:
            # accept with probability exp(-delta/T)
            if T > 0:
                if rng.random() < math.exp(-delta / T):
                    accept = True
        if accept:
            cur_bound = new_bound
            if new_bound < best_bound:
                best_bound = new_bound
                best = cur.copy()
                # early exit if reached target
                if best_bound < 1.50274349:
                    break
        else:
            # revert swap
            cur[i], cur[j] = cur[j], cur[i]
        T *= decay

        if it % 5000 == 0:

                best = cur.copy()

    return [float(x) for x in best]

# Helper: max of linear convolution a*a (full convolution)
# -------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    """Return the maximum entry of the full convolution a*a using FFT."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    # keep only the valid 2*len(a)-1 entries
    return float(conv_full[: 2 * a.size - 1].max())
