import time
import math
import numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound for a sequence with sum≈1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) for a sequence whose sum is (or is forced to) 1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_score:
                    cur = tmp
                    cur_score = tmp_sc
                    if tmp_sc < best_score:
                        best = tmp.copy()
                        best_score = tmp_sc
                    no_improve = 0
                else:
                    no_improve = 0

        # ----- restart from the current best if we are stuck too long -----
        if no_improve > 5000:
            cur = best + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            no_improve = 0
            # reset step‑size schedule
            # (the next iteration will recompute eta from it)

        # ----- early exit if the target is already reached -----
        if best_score < 1.50274349:
            break

    # ----- final cheap local refinement (if time permits) -----
    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.05
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()
        cand_sc = _score(cand)
        if cand_sc < best_score:
            best = cand
            best_score = cand_sc

    # return as plain list of Python floats
    return [float(x) for x in best]
