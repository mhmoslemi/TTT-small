    all entries equal, sum = 1.  If a parent construction `height_sequence_1`
    is available, we use its length; otherwise a default length of 1024 is
    chosen.  The routine respects the time budget and falls back to the
    uniform vector if any unexpected error occurs.
    """
    start = time.monotonic()
    # leave a small safety margin before the hard timeout
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Determine the problem size (length of the sequence)
    # ------------------------------------------------------------------
    try:
        n = int(len(globals()["height_sequence_1"]))
    except Exception:
        n = 1024  # arbitrary fallback size

    # ------------------------------------------------------------------
    # Simple uniform candidate (optimal for the given objective)
    # ------------------------------------------------------------------
    best_vec = _uniform_sequence(n)
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional cheap random search (few iterations) – never exceeds budget
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    # we keep the number of evaluations tiny; each evaluation costs O(n²)
    max_iters = 15
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        # Dirichlet‑like random point on the simplex
        rnd = rng.exponential(scale=1.0, size=n)
        cand = rnd / rnd.sum()
        # use the provided evaluator (matrix‑free, reliable)
        try:
            sc = evaluate_sequence(cand.tolist())
        except Exception:
            # if the evaluator misbehaves, skip this candidate
            continue
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
        # early exit if we accidentally beat the (unattainable) target
        if best_score < 1.50274349:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
