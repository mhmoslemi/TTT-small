import time, math, numpy as np

# ----------------------------------------------------------------------
# Simplex projection (L2 onto {x>=0, sum=1})
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else css[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Maximum value of the (linear) convolution a * a (full, length 2n‑1)
# using zero‑padded FFT of length L (power of two >= 2n)
# ----------------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return float(conv_full[: 2 * a.size - 1].max())


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the final return
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise – use provided parent if available, otherwise random.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        seq = rng.random(2048, dtype=np.float64)
    seq = np.maximum(seq, 0.0)
    if seq.sum() == 0.0:
        seq[0] = 1.0
    seq = seq / seq.sum()                     # normalise to sum = 1
    n = seq.size

    # ------------------------------------------------------------------
    # FFT length (next power of two ≥ 2n)
    # ------------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Gradient‑descent phase (soft‑max smoothing)
    # ------------------------------------------------------------------
    cur = seq.copy()
    best = cur.copy()
    cur_max = _max_conv(cur, L)
    best_max = cur_max
    best_bound = 2.0 * n * best_max

    momentum = np.zeros_like(cur)
    step0 = 1.0
    beta = 0.85
    eps = 1.0
    eps_decay = 0.9995
    eps_min = 1e-10
    it = 0
    target = 1.50274349

    while time.monotonic() < deadline - 2.0:
        it += 1
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        eps = max(eps_min, eps * eps_decay) if it > 1 else eps
        if eps > 1e-7:
            w = np.exp((conv - max_val) / eps)
            w_sum = w.sum()
            w = w / w_sum if w_sum != 0.0 else np.full_like(w, 1.0 / w.size)
        else:
            mask = conv >= max_val - 1e-12
            cnt = int(mask.sum())
            w = np.where(mask, 1.0 / cnt, 0.0) if cnt else np.full_like(conv, 1.0 / conv.size)

        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

                if new_bound < best_bound:
                    best, best_bound = cur.copy(), new_bound
                    if best_bound < target:

            else:
                cur[i:j] = cur[i:j][::-1]          # revert
        temp = max(temp * decay, 1e-12)

    # ------------------------------------------------------------------
    # Return best candidate as plain Python list
    # ------------------------------------------------------------------
    if best.size == 0:
        best = np.array([1.0])
    return [float(x) for x in best]
