    it = 0

    step = 0.6
    beta = 0.9
    momentum = np.zeros_like(cur)
    eps = 1.0

    eps_min = 1e-8

    target = 1.50274349

    no_improve = 0
    
    while time.monotonic() < deadline:

        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        # softmax
        eps_cur = max(eps_min, eps * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps_cur)

        if w_sum == 0:

        else:

import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Score used by the evaluator – assumes sum(a)=1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(convolution(a, a)).  Sum of a must be 1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence with a low AC1 bound.
    The function runs for up to `budget_s` seconds and returns the best
    sequence discovered.
    """

    start = time.monotonic()
    # keep a small safety margin for the outer harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initialise – use the parent if present, otherwise a uniform start
    # --------------------------------------------------------
    if "height_sequence_1" in globals():

    else:
        cur = np.full(2048, 1.0 / 2048, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)               # enforce sum = 1
    n = cur.size

    # --------------------------------------------------------
    # Mix with a half‑sine shape to diversify the starting point
    # --------------------------------------------------------
    x = np.arange(n) + 0.5
    sine = np.sin(np.pi * x / n)              # positive on (0, n)
    sine = _project_simplex(sine)
    cur = 0.5 * cur + 0.5 * sine
    cur = _project_simplex(cur)

    # --------------------------------------------------------

    # --------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())

    no_improve = 0

    # --------------------------------------------------------
    while time.monotonic() < deadline:

        # ----- current convolution (for gradient) -----
        a_fft = np.fft.rfft(cur, n=L)

        conv = conv_full[: 2 * n - 1]
