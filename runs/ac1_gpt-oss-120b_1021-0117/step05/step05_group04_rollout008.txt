def _normalize(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (non‑negative, sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # very unlikely – fall back to uniform
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _self_conv_max(a: np.ndarray, L: int):
    """
    Compute max of a*a (linear convolution) together with its FFT.
    Returns (max_value, rfft_of_a, full_convolution_array_of_length_2n‑1).
    """
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * a.size - 1]
    return conv.max(), a_fft, conv


def _softmax_weights(conv: np.ndarray, max_val: float, eps: float) -> np.ndarray:
    """Soft‑max over convolution values; temperature = eps (eps>0)."""
    # (conv - max_val) ≤ 0, so exponentials stay ≤1
    w = np.exp((conv - max_val) / eps)
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size, dtype=np.float64)
    return w / s


def _gradient(a_fft: np.ndarray, w: np.ndarray, n: int, L: int) -> np.ndarray:

    Gradient of Σ_k w_k (a*a)_k w.r.t. a.

    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
