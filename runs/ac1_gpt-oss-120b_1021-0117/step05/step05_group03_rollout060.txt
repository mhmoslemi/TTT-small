import time
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum x = 1}."""
    n = v.size
    if n == 0:
        return v
    # shift by mean then use the standard algorithm
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast evaluation of 2*n*max(conv(a,a))  – the numerator of the
# AC1 bound when sum(a)=1
# ------------------------------------------------------------
def _score_vec(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(2.0 * a.size * conv_full.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence whose AC1 bound is low."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for conversion & return
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Determine length – use the parent if present
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        n = int(len(height_sequence_1))
    else:
        n = 1024

    # --------------------------------------------------------
    # Initialise with the uniform distribution (sum = 1)
    # --------------------------------------------------------
    cur = np.full(n, 1.0 / n, dtype=np.float64)
    best = cur.copy()
    best_score = _score_vec(best, 1 << ((3 * n - 2).bit_length()))

    # --------------------------------------------------------
    # Parameters for the projected gradient scheme
    # --------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())   # FFT length (power of two)
    momentum = np.zeros_like(cur)
    beta = 0.9                # momentum factor
    base_step = 0.5           # initial step magnitude
    it = 0

    # --------------------------------------------------------
    # Optimisation loop – matrix‑free, O(n log n) per iteration
    # --------------------------------------------------------
    while time.monotonic() < deadline and it < 6000:
        it += 1

        # ----- current convolution and its maximal value -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # ----- sub‑gradient of the max (hard‑max) -----
        mask = np.isclose(conv_full, max_val, atol=1e-14)
        w = mask.astype(np.float64)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # ----- gradient of max(conv) w.r.t. a -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)          # map to original indices
        g = 2.0 * grad_full[idx]

        # ----- momentum update and step -----
        momentum = beta * momentum + (1.0 - beta) * g
        step = base_step / np.sqrt(it)
        cand = cur - step * momentum

        # ----- simplex projection (enforces non‑negativity & sum=1) -----
        cand = _project_simplex(cand)

        cand_score = _score_vec(cand, L)

        # ----- accept if better -----
        if cand_score < best_score:
            best, best_score = cand.copy(), cand_score
            cur = cand
        else:

            prob = np.exp(-(cand_score - best_score) / (step + 1e-12))
            if rng.random() < prob:
                cur = cand

        # ----- early stop when close to the theoretical optimum (2) -----
        if best_score <= 2.0 + 1e-12:
            break

    # --------------------------------------------------------
    # Return the best vector found as a plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best]
