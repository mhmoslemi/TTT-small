def propose_candidate(seed=42, budget_s=1100):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # keep margin

    # Initialize a
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = rng.integers(1500, 3500)   # some size in thousands
        a = rng.random(n, dtype=np.float64)
    a = np.maximum(a,0)
    # Normalize sum to 1 via simplex projection
    a = _project_simplex(a)
    n = a.size

    # compute optimal length for FFT: at least 3n-2
    L = 1 << ((3 * n - 2).bit_length())

    # compute initial conv max
    max_conv = _max_conv(a, L)
    best = a.copy()
    best_sc = max_conv
    best_bound = 2.0 * n * max_conv

    # gradient descent loop
    beta = 0.9      # momentum factor
    momentum = np.zeros_like(a)
    lr0 = 0.5       # initial learning rate (since sum=1)
    iter = 0
    patience = 0
    while time.monotonic() < deadline - 1.0:
        iter += 1
        # compute conv & max
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2*n-1]
        max_val = conv.max()
        # mask for max indices (within tolerance)
        # tolerance relative maybe 1e-12
        mask = conv >= max_val - 1e-12
        # Create weight vector w as normalized (so gradient magnitude similar)
        # Actually we just need integer weights
        w = mask.astype(np.float64)
        # w_rev
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]   # gradient of max conv (subgradient)
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # step schedule
        lr = lr0 / np.sqrt(iter)

        # candidate update
        cand = a - lr * momentum
        # project onto simplex
        cand = _project_simplex(cand)   # ensures sum=1, nonneg

        # evaluate
        cand_sc = _max_conv(cand, L)   # max conv
        if cand_sc <= max_val:  # improvement
            a = cand
            max_conv = cand_sc
            patience = 0
        else:
            # try halving step up to few times
            improved = False
            step = lr
            for _ in range(4):
                step *= 0.5
                cand = a - step * momentum
                cand = _project_simplex(cand)
                cand_sc = _max_conv(cand, L)
                if cand_sc <= max_val:
                    a = cand
                    max_conv = cand_sc
                    improved = True
                    break
            if not improved:
                # maybe perturb a bit to try escape
                patience += 1
                if patience > 200:
                    # random perturbation of a small subset
                    # randomly redistribute small mass
                    perturb = rng.random(n)
                    perturb = np.maximum(perturb, 0)
                    # mix with current a
                    a = _project_simplex(0.9 * a + 0.1 * perturb)
                    patience = 0
                continue
        # update best
        if max_conv < best_sc:
            best = a.copy()
            best_sc = max_conv
            best_bound = 2.0 * n * max_conv
            # early exit if bound good
            if best_bound < 1.50274349:
                break

        # occasional symmetrization
        if iter % 250 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _max_conv(sym, L)
            if sym_sc < max_conv:
                a = sym
                max_conv = sym_sc
                if max_conv < best_sc:
                    best = a.copy()
                    best_sc = max_conv
                    best_bound = 2.0 * n * max_conv

    # After gradient descent, try deterministic anti-identity permutation of best.
    vals = np.sort(best)[::-1]
    perm = np.empty_like(vals)
    left, right = 0, n-1
    k = 0
    while left <= right:
        perm[left] = vals[k]; k+=1
        if left < right:
            perm[right] = vals[k]; k+=1
        left += 1
        right -= 1
    perm_sc = _max_conv(perm, L)
    perm_bound = 2.0 * n * perm_sc
    if perm_bound < best_bound:

        best_sc = perm_sc
        best_bound = perm_bound

    # final random swaps hill climbing
    cur = best.copy()
    cur_sc = best_sc
    cur_bound = best_bound
    while time.monotonic() < deadline - 0.05:
        i, j = rng.integers(0, n, size=2)
        if i == j: continue

        if new_sc < cur_sc:

            cur_bound = 2.0 * n * new_sc
            best = cur.copy()
            best_sc = new_sc
            best_bound = cur_bound
            if best_bound < 1.50274349:

        else:

            cur[i], cur[j] = cur[j], cur[i]

    # Return best as list
    if best.size == 0:
        best = np.array([1.0])
    return [float(x) for x in best]

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, x >= 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑dimensional")
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # tiny negative due to fp errors
    w[w < 0] = 0.0
    # re‑scale to exactly sum 1 (numerical safety)
    s = w.sum()
    if s == 0:
        w[0] = 1.0
    else:
        w /= s
    return w


# ----------------------------------------------------------------------
# Maximum of the full convolution a*a using FFT of length L
# ----------------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    """Return max(conv_full) for a*a using zero‑padded FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv_full.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    Return a non‑empty non‑negative sequence whose AC1 bound is as low as possible.
    The function respects the given time budget and always returns the best
    candidate found before the deadline.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for finalisation
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------------
    # Initialise – use the provided parent if present, else random
    # --------------------------------------------------------------

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)          # enforce sum = 1, non‑negative
    n = a.size

    # --------------------------------------------------------------
    # FFT length (need at least 3n‑2 for gradient computations)
    # --------------------------------------------------------------

    best_sc = _max_conv(best, L)          # max convolution (sum = 1)
    best_bound = 2.0 * n * best_sc

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    beta = 0.9                 # momentum factor
    momentum = np.zeros_like(a)
    lr0 = 0.6                  # initial learning rate (adjustable)
    it = 0

    target_bound = 1.50274349

    # --------------------------------------------------------------
    # Main optimisation loop (matrix‑free sub‑gradient descent)
    # --------------------------------------------------------------
    while time.monotonic() < deadline - 1.0:
        it += 1

        # current convolution and its maximum
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # mask for indices attaining the maximum (allow tiny tolerance)
        mask = conv >= max_val - 1e-12
        w = mask.astype(np.float64)                 # weight = 1 on max positions

        # gradient of the max‑convolution (sub‑gradient)
        w_rev = w[::-1]

        idx = (2 * n - 2) - np.arange(n)

    vals_asc = vals_desc[::-1]               # ascending
    half_big = vals_desc[: (n + 1) // 2]      # larger half (descending)
    half_small = vals_asc[: n // 2]          # smaller half (ascending)
    perm2 = np.empty_like(best)
    perm2[::2] = half_big[: len(perm2[::2])]
    perm2[1::2] = half_small[: len(perm2[1::2])]

    # 3) simple reversal
    perm3 = best[::-1].copy()

    candidates = [best, perm1, perm2, perm3]
    for cand in candidates:
        sc = _max_conv(cand, L)
        bound = 2.0 * n * sc
        if bound < best_bound:
            best = cand.copy()
            best_sc = sc
            best_bound = bound

    # Final local search: random pairwise swaps
    # --------------------------------------------------------------

    # --------------------------------------------------------------
    if best.size == 0:
        best = np.array([1.0])
    return [float(x) for x in best]
