def _project_simplex(v):
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
```

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum == 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Main optimisation routine
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of floats) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a safety margin for final conversion/return
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initialise – use the parent if present, else a uniform start
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            a = np.full_like(a, 1.0)
        a = a / a.sum()
    else:
        n = 2048
        a = rng.random(n, dtype=np.float64)
        a = np.maximum(a, 0.0)
        a = a / a.sum()

    n = a.shape[0]

    # --------------------------------------------------------
    # FFT length – must contain a full convolution of length 2n‑1
    # --------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------
    # Optimisation hyper‑parameters
    # --------------------------------------------------------
    # Adam‑style moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # learning‑rate schedule (base step)
    eta0 = 0.2

    # soft‑max temperature schedule (smooth max)
    tau0 = 0.2
    tau_max = 12.0
    tau_inc = (tau_max - tau0) / 3000.0  # reach tau_max after ~3000 iters

    # bookkeeping
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    best_score = float(2.0 * n * conv.max())
    best_vec = a.copy()
    best_iter = 0
    latest_improve = 0
    target = 1.50274349

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # ----------------------------------------------------
        # Compute convolution and its (soft) max‑weights
        # ----------------------------------------------------
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()

        # soft‑max weights (temperature τ)
        tau = min(tau_max, tau0 + it * tau_inc)
        # shift by max to keep numerics stable
        shifted = conv - max_conv
        exps = np.exp(tau * shifted)
        w = exps / exps.sum()

        # ----------------------------------------------------
        # Gradient of the smoothed / max objective
        # ----------------------------------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----------------------------------------------------
        # Adam update (with projection onto simplex)
        # ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = eta0 / (np.sqrt(v_hat) + eps) * m_hat
        a_candidate = a - step
        # enforce non‑negativity before projection
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = _project_simplex(a_candidate)

        # evaluate candidate (reuse fft for speed)
        cand_fft = np.fft.rfft(a_candidate, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
        cand_score = float(2.0 * n * cand_conv.max())

        # accept if better; otherwise keep current (no Metropolis step here)
        if cand_score < best_score:
            best_score = cand_score
            best_vec = a_candidate.copy()
            latest_improve = it
            best_iter = it

        cur_score = float(2.0 * n * max_conv)
        if cand_score <= cur_score:
            a = a_candidate
        else:
            # occasional uphill move with decreasing probability
            prob = math.exp(-(cand_score - cur_score) / (eta0 + 1e-12))
            if rng.random() < prob:
                a = a_candidate

        # ----------------------------------------------------
        if best_score < target:
            break

    # --------------------------------------------------------
    # Return the best sequence discovered (plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
