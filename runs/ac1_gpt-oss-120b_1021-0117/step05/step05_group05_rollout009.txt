def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # Initialization
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fallback random start length 2000 maybe
        n = 2000
        a0 = rng.random(n, dtype=np.float64)
    # symmetrize
    a0 = (a0 + a0[::-1]) / 2.0
    cur = _project_simplex(np.maximum(a0, 0.0))
    n = cur.size
    L = 1 << ((3 * n - 2).bit_length())
    # define best:
    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    momentum = np.zeros_like(cur)
    beta = 0.9
    step0 = 1.2 # maybe larger step for early descent
    it = 0
    no_improve = 0
    target = 1.50274349

    # main loop
    while time.monotonic() < deadline:
        it += 1
        # compute a_fft and conv
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        # gradient subgradient
        tol = max_val * 1e-8  # tolerance for equality
        # maybe compute mask for values close to max
        mask = conv >= max_val - tol
        # At least one element; compute weight distribution
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 1:
            w /= cnt
        # convolution with reversed w to get gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g
        step = step0 / math.sqrt(it)
        # candidate with backtracking line search
        cand = None
        cand_sc = None
        # use up to 10 reductions
        alpha = step
        for _ in range(10):
            cand_tmp = cur - alpha * momentum
            # ensure nonnegativity
            cand_tmp = _project_simplex(cand_tmp)

            cand_tmp_sc = _score(cand_tmp)
            if cand_tmp_sc <= cur_sc:
                cand, cand_sc = cand_tmp, cand_tmp_sc
                break
            alpha *= 0.5
        if cand is None:
            # take the last tried
            cand_tmp = cur - alpha * momentum
            cand_tmp = _project_simplex(cand_tmp)

            no_improve = 0
            if cand_sc < best_sc:

import time, math, numpy as np

def _project_simplex(v):
    """Projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score(a):
    """Exact AC1 upper bound for a vector a."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (s * s))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ---- initial vector ----------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fallback: random positive vector of length ~2000
        n0 = 2000
        a0 = rng.random(n0, dtype=np.float64)

    # enforce symmetry, nonnegativity and simplex constraint
    a0 = (a0 + a0[::-1]) * 0.5
    cur = _project_simplex(np.maximum(a0, 0.0))

    n = cur.size
    L = 1 << ((3 * n - 2).bit_length())   # FFT length for gradient step

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    momentum = np.zeros_like(cur)
    beta = 0.9               # momentum factor
    step0 = 1.2              # initial step size
    it = 0
    no_improve = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # ---------- convolution and max value ----------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ---------- sub‑gradient of max ----------
        # use a tiny tolerance to collect near‑max entries
        tol = max_val * 1e-8
        mask = conv >= max_val - tol
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 1:
            w /= cnt
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ---------- momentum update ----------
        momentum = beta * momentum + (1.0 - beta) * g

        # ---------- backtracking line search ----------
        step = step0 / math.sqrt(it)
        alpha = step
        cand = None
