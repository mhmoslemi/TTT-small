def geometric_vec(r):
    n = N
    if r == 1.0:
        return np.full(N, 1.0 / N)
    geom = np.empty(N, dtype=np.float64)
    # compute r^i iteratively for stability
    val = 1.0
    for i in range(N):
        geom[i] = val
        val *= r
    # Normalize to sum = 1
    total = geom.sum()
    geom /= total
    # ensure nonnegative
    return geom
```

def _gradient_descent_refine(a_start, rng, deadline):
    a = a_start.copy()
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())
    momentum = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5
    cur_sc = _full_score(a)
    best_sc = cur_sc
    best_a = a.copy()

        if time.monotonic() >= deadline:
            break
        it += 1
        # compute convolution
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        eps = 1e-12
        mask = conv >= max_val - eps
        w = mask.astype(np.float64)
        if w.sum() > 1:
            w /= w.sum()

        w_fft = np.fft.rfft(w_rev, n=L)

        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        momentum = beta * momentum + (1.0 - beta) * g

        # try decreasing alphas
        cand = None

        for _ in range(10):

            a_temp = _project_simplex(a_temp)

                break

            cand, cand_sc = a_temp, sc_temp

        a, cur_sc = cand, cand_sc
        if cur_sc < best_sc:
            best_sc, best_a = cur_sc, a.copy()
            if best_sc < 1.0:
                break
        # early termination if no improvement after many steps
        if it > 2000:

    return best_a, best_sc

import time, math, numpy as np

# ---------- helper utilities ----------
def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _full_score(a):
    """Exact score used by the evaluator (sum is not assumed to be 1)."""
    a = np.asarray(a, dtype=np.float64)

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return float(2.0 * n * max_val / (s * s))

    target = 1.50274349
    it = 0
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1
        # linear cooling
        elapsed = time.monotonic() - start
        total = deadline - start
        T = T0 * max(0.0, 1.0 - elapsed / total)

        cand = _perturb(current, rng)
        cand_sc = _full_score(cand)

        if cand_sc < cur_sc or rng.random() < math.exp(-(cand_sc - cur_sc) / (T + 1e-12)):
            current, cur_sc = cand, cand_sc
            if cur_sc < best_sc:
                best_sc, best_vec = cur_sc, current.copy()
                if best_sc < target:
                    break

        # every few thousand steps try a short gradient descent from current
        if it % 3000 == 0:
            refined, refined_sc = _gradient_refine(current, rng, deadline)
            if refined_sc < best_sc:
                best_sc, best_vec = refined_sc, refined.copy()
                current, cur_sc = refined, refined_sc
                if best_sc < target:
                    break

    # final safety projection (ensures feasibility for the evaluator)
    best_vec = _project_simplex(best_vec)
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    return [float(x) for x in best_vec]
