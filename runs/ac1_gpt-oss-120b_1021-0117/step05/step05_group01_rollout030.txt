import time
import math
import numpy as np

# ------------------------------------------------------------
#  Project onto the probability simplex (non‑negative, sum = z)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=z}."""
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # modest length when no parent is supplied
        n0 = 2048
        a = rng.random(n0, dtype=np.float64)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                     # enforce sum = 1

    n = a.size
    # FFT length – at least 3n to accommodate convolution with weight vector
    L = 1 << ((3 * n - 1).bit_length())

    # --------------------------------------------------------
    #  Book‑keeping
    # --------------------------------------------------------
    # current score (sum=1 -> score = 2*n*max(conv))
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max

    best_vec = a.copy()
    best_score = cur_score

    # optimisation hyper‑parameters
    base_step = 0.6
    beta = 0.85                # momentum factor
    momentum = np.zeros_like(a)
    alpha = 12.0               # smoothing parameter for max‑approximation
    sym_interval = 250         # symmetrisation frequency
    no_improve = 0
    it = 0
    target_score = 1.50274349

    # --------------------------------------------------------
    #  Main loop (matrix‑free, O(n log n) per iteration)
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- recompute convolution and smooth weights -----
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # ----- keep best seen -----
        if cur_score < best_score:
            best_score, best_vec = cur_score, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- early exit if target achieved -----
        if best_score < target_score:
            break

        # ----- smooth approximation of the max -----
        w = np.exp(alpha * (conv - cur_max))  # max entry gets weight 1
        w_sum = w.sum()
        if w_sum == 0.0:                     # numerical safety
            w = np.zeros_like(w)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # ----- gradient of the smoothed max -----
        a_rev = a[::-1]
        rev_fft = np.fft.rfft(a_rev, n=L)
        w_fft = np.fft.rfft(w, n=L)
        grad_rev = np.fft.irfft(rev_fft * w_fft, n=L)
        grad = 2.0 * grad_rev[(n - 1):(n - 1 + n)]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- back‑tracking line search -----
        step = base_step / math.sqrt(it)
        cand = None
        cand_score = None
        for _ in range(8):
            a_tmp = a - step * momentum
            a_tmp = _project_simplex(a_tmp)          # keep on simplex
            # optional symmetrisation (helps when close to optimum)
            if it % sym_interval == 0:
                a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
                a_tmp = _project_simplex(a_tmp)
            # evaluate candidate
            tmp_fft = np.fft.rfft(a_tmp, n=L)
            conv_tmp = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
            cand_score = 2.0 * n * conv_tmp.max()
            if cand_score <= cur_score:
                cand = a_tmp
                break
            step *= 0.5                     # shrink step and retry

        # if line search failed, keep current point
        if cand is None:
            cand = a
            cand_score = cur_score

        # ----- accept if improvement -----
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best_score, best_vec = cand_score, cand.copy()
        else:
            # occasional probabilistic uphill move
            prob = math.exp(-(cand_score - cur_score) / (step + 1e-12))
            if rng.random() < prob:
                a = cand
                cur_score = cand_score

        # ----- periodic full symmetrisation -----
        if it % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            sym_fft = np.fft.rfft(a_sym, n=L)
            conv_sym = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
            sym_score = 2.0 * n * conv_sym.max()
            if sym_score < cur_score:
                a = a_sym
                cur_score = sym_score
                if sym_score < best_score:
                    best_score, best_vec = sym_score, a_sym.copy()
                no_improve = 0

        # ----- random mass transfer when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.4
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
                no_improve = 0

    # --------------------------------------------------------
    #  Return the best candidate found as a plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
