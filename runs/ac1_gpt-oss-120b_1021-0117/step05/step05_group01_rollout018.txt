# ------------------------------------------------------------
# Simplex projection (non‑negative, sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=z}."""
    if z < 0:
        raise ValueError("z must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

                a, cur_sc = cand, cand_sc

        # ----- periodic symmetrisation (helps Toeplitz structure) ----
        if it % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            a_sym_fft = np.fft.rfft(a_sym, n=L)
            conv_sym = np.fft.irfft(a_sym_fft * a_sym_fft, n=L)[: 2 * n - 1]
            sym_sc = 2.0 * n * conv_sym.max()
            if sym_sc < cur_sc:
                a, cur_sc = a_sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, a_sym.copy()
                stagnation = 0

        # ----- random mass‑transfer when stuck -------------------------
        if stagnation > 5000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.5
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a)   # restore feasibility
            stagnation = 0

    # ---------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ---------------------------------------------------------------
    return [float(x) for x in best_vec]
