def evaluate_sequence(sequence: list) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf

    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + float(budget_s) - 0.5 # small margin.

    # init
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n0 = 2000
        a = rng.random(n0, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    a = _project_simplex(a)
    n = a.size
    L = 1 << ((3*n - 2).bit_length())

    # keep best
    best = a.copy()
    best_sc = _max_conv(best, L)
    best_bound = 2.0 * n * best_sc

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_opt = 1e-8
    alpha0 = 0.2  # base learning rate
    # epsilon for softmax (smoothness)
    eps_smooth = 0.1
    eps_decay = 0.9995
    it = 0

    while time.monotonic() < deadline - 2.0:
        it += 1
        # compute convolution and its max
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        # compute softmax
        eps_cur = max(eps_opt, eps_smooth * (eps_decay ** it))
        if eps_cur > 1e-8:
            w = np.exp((conv - max_val) / eps_cur)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
        else:
            # exact subgradient: uniform over maximal locations
            mask = np.abs(conv - max_val) < 1e-12
            cnt = mask.sum()
            if cnt == 0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = np.where(mask, 1.0 / cnt, 0.0)
        # compute gradient (2 * a * w reversed convolution)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n-2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = alpha0 * m_hat / (np.sqrt(v_hat) + eps_opt)

        cand = _project_simplex(cand)

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum=1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _max_conv(a: np.ndarray, L: int) -> float:
    """Maximum of the full convolution a*a using a zero‑padded FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return float(conv_full[: 2 * a.size - 1].max())


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of floats) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # Initial sequence (use the parent if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2048, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    a = _project_simplex(a)               # enforce sum = 1
    n = a.size
    L = 1 << ((3 * n - 2).bit_length())   # FFT length (power of two)

    # ------------------------------------------------------------------
    # Book‑keeping of the best solution found so far
    # ------------------------------------------------------------------
    best = a.copy()
    best_sc = _max_conv(best, L)
    best_bound = 2.0 * n * best_sc

    # ------------------------------------------------------------------
    # Adam optimiser parameters (matrix‑free)
    # ------------------------------------------------------------------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_opt = 1e-8
    lr0 = 0.12               # base learning rate
    max_step = 0.15          # clip for stability

    # soft‑max smoothing for the max‑conv objective
    eps_smooth = 0.12
    eps_decay = 0.9994

    it = 0
    while time.monotonic() < deadline - 2.0:
        it += 1

        # --------------------------------------------------------------
        # Current convolution and soft‑max weights
        # --------------------------------------------------------------
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        eps_cur = max(eps_opt, eps_smooth * (eps_decay ** it))
        if eps_cur > 1e-8:
            w = np.exp((conv - max_val) / eps_cur)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
        else:
            mask = np.abs(conv - max_val) < 1e-12
            cnt = mask.sum()
            if cnt == 0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = np.where(mask, 1.0 / cnt, 0.0)

        # --------------------------------------------------------------
        # Gradient of the smoothed max‑convolution (matrix‑free)
        # --------------------------------------------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # --------------------------------------------------------------
        # Adam update with projection onto the simplex
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_opt)
        step = np.clip(step, -max_step, max_step)

        cand = a - step
        cand = _project_simplex(cand)

        cand_sc = _max_conv(cand, L)

        # keep the globally best
        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc
            best_bound = 2.0 * n * best_sc

        # accept if we made progress (otherwise stay)
        if cand_sc <= max_val:
            a = cand

        # occasional symmetrisation (helps with the Toeplitz structure)
        if it % 500 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _max_conv(sym, L)
            if sym_sc < best_sc:
                best = sym.copy()
                best_sc = sym_sc

            if sym_sc < max_val:
                a = sym

    # ------------------------------------------------------------------
    # Deterministic anti‑identity permutation (large values opposite small)
    # ------------------------------------------------------------------
    vals = np.sort(best)[::-1]
    perm = np.empty_like(vals)
    left, right = 0, n - 1
    k = 0
    while left <= right:
        perm[left] = vals[k]; k += 1
        if left < right:
            perm[right] = vals[k]; k += 1
        left += 1
        right -= 1
    perm_sc = _max_conv(perm, L)
    if perm_sc < best_sc:
        best = perm.copy()
        best_sc = perm_sc
        best_bound = 2.0 * n * best_sc

    # ------------------------------------------------------------------
    # Simple hill‑climbing swaps to polish the permutation
    # ------------------------------------------------------------------
    cur = best.copy()
    cur_sc = best_sc
    while time.monotonic() < deadline - 0.2:
        i, j = rng.integers(0, n, size=2)
        if i == j:

        new_sc = _max_conv(cur, L)
        if new_sc < cur_sc:
            cur_sc = new_sc
            if new_sc < best_sc:
                best = cur.copy()
                best_sc = new_sc
                best_bound = 2.0 * n * new_sc
        else:
            cur[i], cur[j] = cur[j], cur[i]

    # ------------------------------------------------------------------
    # Return the best candidate (plain Python list)
    # ------------------------------------------------------------------
    if best.size == 0:
        best = np.array([1.0])
    return [float(x) for x in best]
