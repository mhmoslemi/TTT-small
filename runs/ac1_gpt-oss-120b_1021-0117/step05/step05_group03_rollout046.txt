def propose_candidate(seed=42,budget_s=1100,...):

    rng = np.random.default_rng(seed)

    start=time.monotonic()
    deadline = start + max(0.0,budget_s-0.5)

    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n=1024
        a = rng.random(n)
    a = np.maximum(a,0)
    a = _project_simplex(a) # ensure sum=1

    n = a.size
    L = 1 << ((3*n-2).bit_length()) # power of two >= 3n-2

    # Adam parameters
    beta1=0.9 ; beta2=0.999 ; eps=1e-8 ; alpha=0.05 (maybe)
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_vec = a.copy()
    # compute initial conv and score
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*n-1]
    best_sc = 2.0*n*conv.max()
    cur_sc = best_sc

    # scheduling
    lambda_start = 0.5
    lambda_max = 15.0
    lambda_inc = (lambda_max - lambda_start) / (deadline - start) # per second maybe

    it =0

    while time.monotonic() < deadline:
        it+=1

        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()

        # compute λ (maybe linearly increase)
        elapsed = time.monotonic() - start
        λ = lambda_start + (lambda_max - lambda_start) * (elapsed / (budget_s-0.5))
        if λ > lambda_max:
            λ = lambda_max

        # compute softmax weights
        # subtract max_val for stability
        exp_term = np.exp(λ * (conv - max_val))
        w = exp_term / exp_term.sum()
        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1-beta1) * g
        v = beta2 * v + (1-beta2) * (g * g)
        mhat = m / (1 - beta1**it)
        vhat = v / (1 - beta2**it)
        step = alpha * mhat / (np.sqrt(vhat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new,0.0)
        a_new = _project_simplex(a_new)

        # compute new score quickly (need convolution)
        a_fft_new = np.fft.rfft(a_new, n=L)
        conv_full_new = np.fft.irfft(a_fft_new * a_fft_new, n=L)
        conv_new = conv_full_new[:2*n-1]
        cur_sc_new = 2.0*n*conv_new.max()

        # Accept if better else maybe keep old or random acceptance? Could implement Metropolis.
        # We'll adopt simple acceptance: if new_score < cur_sc: accept; else accept with probability p.
        if cur_sc_new < cur_sc:
            a = a_new
            cur_sc = cur_sc_new
        else:
            # maybe accept with small prob to escape

            if rng.random() < prob:
                a = a_new
                cur_sc = cur_sc_new
        # update best
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = a.copy()

                    best_sc = sym_sc
                    best_vec = sym.copy()
        # early exit if < target
        if best_sc < 1.50274349:
            break

    return [float(x) for x in best_vec]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast score (sum = 1) -> 2 * n * max(conv(a,a))
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max())
