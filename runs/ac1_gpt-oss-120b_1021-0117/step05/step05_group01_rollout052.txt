import numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x) = z}."""
    n = v.size
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Compute max of convolution and its exact sub‑gradient
# ------------------------------------------------------------
def _max_and_gradient(a: np.ndarray, L: int):
    """Return (max_a*a, gradient) for full convolution a*a."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv_full.max()
    # tolerance to collect all (near‑)max entries
    tol = max_val * 1e-12 + 1e-16
    max_idx = np.where(conv_full >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in max_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        partner = k - i
        grad[i] += 2.0 * a[partner]
    return max_val, grad

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------
    # Initial vector – parent if present, else a random start
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2048, dtype=np.float64)

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                     # enforce sum = 1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())  # power‑of‑two for FFT

    # --------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------
    max_val, _ = _max_and_gradient(a, L)
    best_sc = 2.0 * n * max_val
    best_vec = a.copy()

    # optimisation parameters
    step0 = 0.7                # base step magnitude
    sym_interval = 250         # how often to symmetrise
    no_improve = 0
    it = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # ---------- current max and sub‑gradient ----------
        cur_max, grad = _max_and_gradient(a, L)
        cur_sc = 2.0 * n * cur_max

        # keep best
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # early exit
        if best_sc < target:
            break

        # ---------- descent direction ----------
        maxg = grad.max()
        if maxg == 0:                     # reached a stationary point
            break
        direction = grad / maxg           # scale to unit max entry

        # ---------- back‑tracking line search ----------
        base_alpha = step0 / math.sqrt(it)
        alpha = base_alpha
        cand = None
        cand_sc = None
        for _ in range(8):
            a_tmp = a - alpha * direction
            a_tmp = _project_simplex(a_tmp)        # stay on simplex
            tmp_max, _ = _max_and_gradient(a_tmp, L)
            cand_sc = 2.0 * n * tmp_max
            if cand_sc <= cur_sc:
                cand = a_tmp
                break
            alpha *= 0.5

        # if no descent found, keep current a (occasionally try a random move)
        if cand is not None:
            a = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
                no_improve = 0
        else:
            # occasional uphill acceptance (simulated‑annealing style)
            if rng.random() < 0.02:
                a = a_tmp
                cur_sc = cand_sc

        # ---------- periodic symmetrisation ----------
        if it % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            sym_max, _ = _max_and_gradient(a_sym, L)
            sym_sc = 2.0 * n * sym_max
            if sym_sc < cur_sc:
                a = a_sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, a_sym.copy()
                    no_improve = 0

        # ---------- random mass‑transfer when stuck ----------
        if no_improve > 2500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.35
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
                no_improve = 0

        # ---------- occasional perturbation of the best ----------
        if it % 1500 == 0:
            # add tiny Gaussian noise to the current best and continue
            pert = best_vec + rng.normal(scale=1e-4, size=n)
            pert = np.maximum(pert, 0.0)
            pert = pert / pert.sum()
            a = pert

    # return the best candidate found
    return [float(x) for x in best_vec]
