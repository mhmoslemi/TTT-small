import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Helper: score (2·n·max(conv)), full convolution and its FFT
# ------------------------------------------------------------
def _score_and_fft(x: np.ndarray, n: int, L: int, idx: np.ndarray):
    a_fft = np.fft.rfft(x, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv_full.max()
    return 2.0 * n * max_val, conv_full, a_fft

    cur_sc, cur_conv, cur_fft = _score_and_fft(cur, n, L, idx)
    best_vec = cur.copy()
    best_sc = cur_sc

    # --------------------------------------------------------
    # Optimisation hyper‑parameters
    # --------------------------------------------------------
    step0 = 0.5                # base learning‑rate factor
    beta1, beta2 = 0.9, 0.999  # Adam coefficients
    m = np.zeros_like(cur)    # first moment
    v = np.zeros_like(cur)    # second moment
    eps_adam = 1e-8
    eps0 = 0.5                 # initial soft‑max temperature
    eps_decay = 0.9995
    eps_min = 1e-12
    it = 0
    no_improve = 0
    target = 1.50274349

    # --------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- soft‑max weighting for the max ----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        max_val = cur_conv.max()
        w = np.exp((cur_conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the soft max ----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(cur_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        base_alpha = step0 / math.sqrt(it)
        update = base_alpha * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- back‑tracking line search ----
        cand = None
        cand_sc = None
        alpha = 1.0
        for _ in range(8):
            cand_tmp = cur - alpha * update
            cand_tmp = _project_simplex(cand_tmp)
            cand_sc_tmp, cand_conv, cand_fft = _score_and_fft(cand_tmp, n, L, idx)
            if cand_sc_tmp <= cur_sc:
                cand = cand_tmp
                cand_sc = cand_sc_tmp
                cur_conv, cur_fft = cand_conv, cand_fft
                break
            alpha *= 0.5
        if cand is None:
            cand = cand_tmp
            cand_sc = cand_sc_tmp
            cur_conv, cur_fft = cand_conv, cand_fft

        # ---- accept / reject ----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (base_alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
                # cur_conv and cur_fft already updated above
            no_improve += 1

        # ---- occasional symmetrisation (helps with Toeplitz structure) ----
        if it % 100 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc, sym_conv, sym_fft = _score_and_fft(sym, n, L, idx)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                cur_conv, cur_fft = sym_conv, sym_fft
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # ---- random mass‑transfer when stuck ----
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc, tmp_conv, tmp_fft = _score_and_fft(tmp, n, L, idx)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    cur_conv, cur_fft = tmp_conv, tmp_fft
                    no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                    no_improve = 0

        # ---- early exit if target reached ----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
