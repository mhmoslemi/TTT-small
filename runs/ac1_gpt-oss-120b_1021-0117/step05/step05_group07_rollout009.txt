def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s)-0.5)

    # initial vector cur
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        # ensure nonnegative
        cur = np.maximum(cur, 0.0)
        # normalise to sum=1
        total = cur.sum()
        if total <= 0:
            cur = np.ones_like(cur) / cur.size
        else:
            cur = cur / total
    else:
        n = 2000 # default length maybe
        cur = rng.random(n).astype(np.float64)
        cur = cur / cur.sum()

    n = cur.size
    L = 1 << ((3 * n - 2).bit_length())
    idx = (2 * n - 2) - np.arange(n)

    # compute initial best
    def compute_obj(a):
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2*n - 1]
        return 2.0 * n * conv.max(), conv.max()

    best_obj, _ = compute_obj(cur)
    best_vec = cur.copy()
    cur_obj = best_obj

    # parameters
    eps0 = 2.0           # smoothing parameter start
    eps_decay = 0.995    # very slow cooling (makes gradient smoother longer)
    eta0 = 0.5           # step size for EG
    eta_decay = 0.995
    eps_min = 1e-12
    it = 0

    # main loop
    while time.monotonic() < deadline:
        it += 1
        # compute current convolution and objective
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2*n - 1]
        max_val = conv_full.max()

        # compute temperature
        eps = max(eps_min, eps0 * (eps_decay ** it))

        # compute softmax weights
        # To avoid overflow when eps very small, shift by max val
        w_raw = np.exp((conv_full - max_val) / eps)
        w_sum = w_raw.sum()
        if w_sum == 0.0 or np.isinf(w_sum) or np.isnan(w_sum):
            w = np.full_like(conv_full, 1.0 / conv_full.size)
        else:
            w = w_raw / w_sum

        # gradient via convolution with reversed weights
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # step size schedule
        eta = eta0 * (eta_decay ** it)

        # exponentiated gradient update
        # to avoid overflow, clip grad
        grad_clip = np.clip(grad, -1e3, 1e3)
        # exponentiated update
        cur_new = cur * np.exp(-eta * grad_clip)
        # project onto simplex (normalize)
        sum_new = cur_new.sum()
        if sum_new == 0:
            cur_new = np.full_like(cur, 1.0 / cur.size)
        else:
            cur_new = cur_new / sum_new

        # evaluate new candidate
        obj_new, _ = compute_obj(cur_new)

        if obj_new < cur_obj:
            cur, cur_obj = cur_new, obj_new
            if obj_new < best_obj:
                best_obj, best_vec = obj_new, cur_new.copy()
        else:
            # acceptance via Metropolis with probability exp(-(obj_new-cur_obj)/temp2)
            # Use simple probability
            prob = math.exp(-(obj_new - cur_obj) / (eta + 1e-12))
            if rng.random() < prob:
                cur, cur_obj = cur_new, obj_new

        # occasional random perturbation if stagnated
        # track stagnation via iteration count since best improved
        # We'll use simple rule: every 500 iterations, do a random mass transfer
        if it % 400 == 0:
            # pick two indices random
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i: j += 1
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.2
                cur[i] -= delta
                cur[j] += delta
                # renormalize
                cur_sum = cur.sum()
                cur = cur / cur_sum
                # check improvement
                cand_obj, _ = compute_obj(cur)
                if cand_obj < cur_obj:
                    cur_obj = cand_obj
                    if cand_obj < best_obj:
                        best_obj, best_vec = cand_obj, cur.copy()
                # else maybe revert? Not necessary

        # early exit if target reached
        if best_obj < 1.50274349:
            break

    # final conversion to list
    return [float(x) for x in best_vec]

def local_swap(cur, cur_obj, num_trials=30):
    for _ in range(num_trials):
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i: j += 1
        # choose delta based on cur[i]
        if cur[i] <= 1e-12:
            continue
        delta = cur[i] * rng.random() * 0.5
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = cand / cand.sum()
        obj, _ = compute_obj(cand)
        if obj < cur_obj:
            cur = cand
            cur_obj = obj
    return cur, cur_obj

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    rng = np.random.default_rng(seed)

        cur = cur / cur.sum()

    n = cur.size
    # FFT length: power of two ≥ 3n‑2 (avoid circular wrap‑around)
    L = 1 << ((3 * n - 2).bit_length())
    # indices that map the full convolution gradient back to the original vector
    idx = (2 * n - 2) - np.arange(n)

    # ---------- helpers ----------
    def evaluate(a):
        """Return (objective, max convolution) for vector a."""
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        mx = conv.max()
        return 2.0 * n * mx, mx, a_fft

    # initial best
    best_obj, _, best_fft = evaluate(cur)
    best_vec = cur.copy()
    cur_obj = best_obj

    # ---------- optimisation parameters ----------
    eps0 = 2.0            # initial soft‑max temperature
    eps_decay = 0.995
    eps_min = 1e-12
    eta0 = 0.5            # initial EG step size
