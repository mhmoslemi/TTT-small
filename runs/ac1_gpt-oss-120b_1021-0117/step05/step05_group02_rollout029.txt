        max_val = conv.max()
        eps = max(eps_min, eps0 * (eps_decay ** it))

        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        # momentum
        momentum = beta * momentum + (1 - beta) * g
        # candidate 1: momentum step
        alpha = step0 / math.sqrt(it)
        cand1 = cur - alpha * momentum
        cand1 = _project_simplex(cand1)
        cand1_sc = _score(cand1, n, L)
        # candidate 2: mirror descent
        eta = step0_mirror / math.sqrt(it)

        cand2 = cur * np.exp(exp_arg)
        s = cand2.sum()
        if s > 0:

        else:

        cand2_sc = _score(cand2, n, L)
        # choose best among cand1,cand2,cur
        cand = cur
        cand_sc = cur_sc

        if cand2_sc < cand_sc:

        if cand_sc < cur_sc - 1e-12:

            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:

            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))

        # no early exit for target

    return [float(x) for x in best_vec]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (L2 onto {x >= 0, sum(x) = 1})
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast maximum of self‑convolution via FFT (full length)
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    """Return max(conv(a, a)) using power‑of‑two FFT length L."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    # convolution length = 2*len(a)-1, take the relevant prefix
    conv = conv_full[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _score(a: np.ndarray, n: int, L: int) -> float:
    """Exact score used by the evaluator when sum(a)=1."""
    return 2.0 * n * _conv_max(a, L)
