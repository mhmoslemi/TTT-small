import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Fast AC1 bound for sum≈1 (returns 2*n*max(conv))
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a short safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # -----------------------------------------------------------------
    # initialise a feasible point on the simplex
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
            s = 1.0
        a0 = a0 / s
    else:   # fallback initialization
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 100.0)
        a0 = a0 / a0.sum()

    # a quick symmetrisation of the start (helps a lot)
    a0 = (a0 + a0[::-1]) * 0.5
    a0 = _project_simplex(a0)

    n = a0.size
    # Length for convolutions involving the weight vector (size 2n‑1)
    L = 1 << ((3 * n - 2).bit_length())

    cur = a0.copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    it = 0

    beta = 0.9                # momentum coefficient

                else:

            momentum = np.zeros_like(cur)
            eps = None
