import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# the maximum value and the indices where it is attained.
# ------------------------------------------------------------
def _conv_max_and_idx(a: np.ndarray, L: int):
    """Return (max value, array of indices attaining the max)."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    max_val = conv_full.max()
    # use a relative tolerance to catch multiple maximizers
    tol = 1e-14 * max_val
    idx = np.where(np.abs(conv_full - max_val) <= tol)[0]
    return float(max_val), idx


# ------------------------------------------------------------
# Sub‑gradient of the max‑convolution at the given indices.
# ------------------------------------------------------------
def _max_conv_subgrad(a: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    """Average sub‑gradient over all indices in idxs."""
    n = a.size
    g = np.zeros_like(a)
    if idxs.size == 0:
        return g
    for k in idxs:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo > hi:
            continue
        i = np.arange(lo, hi + 1, dtype=int)
        # contribution from a_{k-i}
        g[i] += a[k - i]
        # contribution from a_i (the diagonal term)
        g[i] += a[i]
    g /= float(idxs.size)
    return g


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # Initial vector – parent if supplied, otherwise a random start
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # modest random start
        a = rng.random(2048, dtype=np.float64)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a /= a.sum()                     # enforce sum = 1
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two for FFT

    # ------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------
    cur_max, cur_idx = _conv_max_and_idx(a, L)
    best_score = 2.0 * n * cur_max
    best_vec = a.copy()

    step0 = 0.5                     # base step magnitude
    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------
    # Optimisation loop (matrix‑free, O(n log n) per iteration)
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and its maximum -----
        cur_max, cur_idx = _conv_max_and_idx(a, L)
        cur_score = 2.0 * n * cur_max

        # ----- keep the best seen so far -----
        if cur_score < best_score:
            best_score, best_vec = cur_score, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- early exit if target reached -----
        if best_score < target:
            break

        # ----- sub‑gradient of the maximal convolution entry -----
        g = _max_conv_subgrad(a, cur_idx)

        # ----- step size (diminishing) -----
        alpha = step0 / math.sqrt(it + 1.0)

        # ----- projected gradient step -----
        cand = a - alpha * g
        cand = np.maximum(cand, 0.0)
        if cand.sum() == 0.0:
            cand = a.copy()
        else:
            cand /= cand.sum()

        # ----- evaluate candidate -----
        cand_max, _ = _conv_max_and_idx(cand, L)
        cand_score = 2.0 * n * cand_max

        # ----- accept if improvement, otherwise Metropolis style -----
        if cand_score < cur_score:
            a = cand
            no_improve = 0
        else:
            # occasional uphill move
            prob = math.exp(-(cand_score - cur_score) / (alpha + 1e-12))
            if rng.random() < prob:
                a = cand

        # ----- periodic tiny jitter to break possible symmetry -----
        if it % 500 == 0:
            jitter = rng.standard_normal(n) * 1e-6
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()

        # ----- random mass‑transfer when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.4
                a[i] -= delta
                a[j] += delta
                a /= a.sum()
                no_improve = 0

        # ----- occasional pure random restart (exploration) -----
        if rng.random() < 0.001:
            a_rand = rng.random(n, dtype=np.float64)
            a_rand /= a_rand.sum()
            rand_max, _ = _conv_max_and_idx(a_rand, L)
            rand_score = 2.0 * n * rand_max
            if rand_score < best_score:
                best_score, best_vec = rand_score, a_rand.copy()
            # adopt the random point if it is not dramatically worse
            if rand_score < cur_score * 1.05:
                a = a_rand

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
