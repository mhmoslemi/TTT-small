import time, math, numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum=1, non‑neg)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound for a vector with sum=1
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Return 2 * n * max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    """Return a non‑negative sequence with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)

    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------- initialise ---------------------------
    if "height_sequence_1" in globals():

    else:
        a = rng.random(1024, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)            # enforce sum = 1
    n = a.size

    # FFT length large enough for a (len n) * w (len 2n‑1)
    L = 1 << ((3 * n - 2).bit_length())

    # -------------------------- bookkeeping -------------------------
    best_vec = a.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    # optimisation hyper‑parameters
    alpha0 = 1.0                       # base step magnitude
    mom_beta = 0.9                     # momentum decay
    momentum = np.zeros_like(a)

    # smoothing (soft‑max) schedule
    beta_start = 0.5
    beta_max = 8.0

    # control loops
    it = 0
    no_improve = 0
    target = 1.50274349
    sym_period = 350

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        maxc = conv.max()

        # ----- soft‑max weighting of convolution entries -----
        beta = beta_start + (beta_max - beta_start) * min(1.0, it / 5000.0)
        # shift by max to avoid overflow
        exp_vals = np.exp(beta * (conv - maxc))
        w = exp_vals / exp_vals.sum()
