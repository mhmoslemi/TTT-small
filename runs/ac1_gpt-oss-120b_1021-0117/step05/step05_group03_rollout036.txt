def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        # Normalize to sum 1
        s = a0.sum()
        if s <= 0:
            a0 = rng.random(a0.shape, dtype=np.float64)
            s = a0.sum()
        a0 = a0 / s
    else:
        n0 = 2048
        a0 = rng.random(n0, dtype=np.float64)
        a0 = a0 / a0.sum()

    a = _project_simplex(a0)   # ensure sum 1
    n = a.size
    L = 1 << ((2 * n - 1).bit_length()) # power of 2 for FFT
    
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # temperature for Metropolis
    temp0 = 0.01
    # softmax beta
    beta_soft = 1.0
    beta_max = 40.0
    beta_factor = 1.001  # increase slowly

    # keep best
    best = a.copy()
    best_sc = _score_simplex(best, L)
    cur_sc = best_sc

    it = 0
    no_imp = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1
        # Update softmax beta gradually:
        if beta_soft < beta_max:
            beta_soft = min(beta_max, beta_soft * beta_factor)

        # Forward FFT of current a
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()

        # Score for current
        cur_sc = float(2.0 * n * max_val)

        # Update best
        if cur_sc < best_sc:
            best_sc = cur_sc
            best = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Softmax weights
        # subtract max to avoid overflow
        w_exp = np.exp(beta_soft * (conv - max_val))
        w = w_exp / w_exp.sum()

        # Gradient via convolution with w_rev
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n-2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # Learning rate schedule
        # base step size maybe 0.5 / sqrt(it) ?
        lr = 0.5 / math.sqrt(it)

        # Proposed step
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = a - step
        cand = _project_simplex(cand)

        # Score candidate
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[:2*n-1]
        cand_sc = float(2.0 * n * cand_conv.max())

        # Accept if better, else maybe accept uphill with probability
        if cand_sc < cur_sc:
            a = cand
            cur_sc = cand_sc
            no_imp = 0
        else:
            # Metropolis accept
            temperature = temp0 / math.sqrt(it)
            prob = math.exp(-(cand_sc - cur_sc) / (temperature + 1e-12))
            if rng.random() < prob:
                a = cand
                cur_sc = cand_sc
                no_imp = 0

        # Occasional random perturb when stuck
        if no_imp > 5000:
            # random mass transfer
            i = rng.integers(0, n)
            if a[i] > 0:
                delta = a[i] * rng.random() * 0.5
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                # compute score
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[:2*n-1]
                tmp_sc = float(2.0 * n * tmp_conv.max())
                if tmp_sc < cur_sc:
                    a = tmp
                    cur_sc = tmp_sc
                if tmp_sc < best_sc:
                    best = tmp.copy()
                    best_sc = tmp_sc
                no_imp = 0

        # Periodic symmetrization (helps reduce max perhaps)
        if it % 500 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L)[:2*n-1]
            sym_sc = float(2.0 * n * sym_conv.max())
            if sym_sc < cur_sc:
                a = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best = sym.copy()
                    best_sc = sym_sc
                no_imp = 0

        if best_sc < target:
            break

    # final fallback ensure sum >= 0.01 (projected to simplex ensures sum=1)
    # Multiply by any factor? Not needed.
    return [float(x) for x in best]

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection – Euclidean projection onto {x >= 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast AC1‑style score for vectors normalised to sum = 1
# ----------------------------------------------------------------------
def _score_simplex(a: np.ndarray, L: int) -> float:
    """
    Return 2 * n * max(convolution(a, a)) assuming sum(a) = 1.
    L must be a power‑of‑two >= 3*len(a)-2.
    """
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    max_val = conv_full[: 2 * n - 1].max()
    return float(2.0 * n * max_val)


# ----------------------------------------------------------------------
# Propose a candidate sequence – gradient descent with soft‑max weighting
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function that minimises the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # small safety margin

    # ------------------------------------------------------------------
    # Initialise – use the provided parent if available, otherwise random
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = rng.random(a0.shape, dtype=np.float64)
            s = a0.sum()
        a0 = a0 / s
    else:
        n0 = 2048
        a0 = rng.random(n0, dtype=np.float64)
        a0 = a0 / a0.sum()

    a = _project_simplex(a0)          # enforce sum = 1
    n = a.size
    # FFT length – must accommodate a (len n) * w (len 2n‑1)
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Optimisation state
    # ------------------------------------------------------------------
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-12

    # Soft‑max temperature for the max‑approximation
    beta_soft = 1.0          # start fairly smooth
    beta_max = 40.0

    # Metropolis temperature (for occasional uphill moves)
    temp0 = 0.01

    # Book‑keeping
    best = a.copy()
    best_sc = _score_simplex(best, L)
    cur_sc = best_sc
    it = 0
    no_imp = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- increase soft‑max sharpness gradually ----
        if beta_soft < beta_max:
            beta_soft = min(beta_max, beta_soft * beta_factor)

        # ---- current convolution and score ----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = float(2.0 * n * max_val)

        lr = 0.5 / math.sqrt(it)

    # ----------------------------------------------------------------------
