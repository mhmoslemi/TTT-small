import time
import math
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # Initialise from the parent if available, otherwise random.
    # ------------------------------------------------------------------
    parent = globals().get('height_sequence_1')
    if parent is not None:
        a0 = np.asarray(parent, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = rng.random(a0.shape, dtype=np.float64)
    else:
        a0 = rng.random(1024, dtype=np.float64)

    # Normalise to a probability vector (sum = 1)
    a0 = a0 / max(a0.sum(), 1e-12)

    n = a0.size
    # Parameters for the soft‑max parametrisation
    u = np.log(a0 + 1e-12)                     # log‑probabilities

    # FFT length (next power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    # Adam optimiser state
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    eps = 1e-8
    lr = 0.1                                   # base learning rate

    # Book‑keeping
    best_a = a0.copy()
    # initial score (objective = 2 * n * max(conv) because sum(a)=1)
    a_fft = np.fft.rfft(best_a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    best_score = 2.0 * n * conv.max()
    best_iter = 0
    no_improve = 0
    sym_period = 20                              # symmetrise every few steps

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # --------------------------------------------------------------
        # Symmetrise the log‑weights periodically (helps with the Toeplitz
        # structure of the problem).
        # --------------------------------------------------------------
        if it % sym_period == 0:
            u = 0.5 * (u + u[::-1])

        # --------------------------------------------------------------
        # Soft‑max to obtain a feasible a (non‑negative, sum = 1)
        # --------------------------------------------------------------
        u_max = np.max(u)
        exp_u = np.exp(u - u_max)               # stabilised exponentiation
        a = exp_u / np.sum(exp_u)

        # --------------------------------------------------------------
        # Convolution and objective value
        # --------------------------------------------------------------
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        M = conv.max()
        score = 2.0 * n * M

        # --------------------------------------------------------------
        # Keep the best vector seen so far
        # --------------------------------------------------------------
        if score < best_score:
            best_score = score
            best_a = a.copy()
            best_iter = it
            no_improve = 0
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # Early exit if target reached
        # --------------------------------------------------------------
        if best_score < 1.50274349:
            break

        # --------------------------------------------------------------
        # Sub‑gradient of the max‑convolution (active‑set weight vector)
        # --------------------------------------------------------------
        mask = np.isclose(conv, M, atol=1e-12, rtol=0.0)
        w = mask.astype(np.float64)
        w_sum = w.sum()

        # --------------------------------------------------------------

        # --------------------------------------------------------------

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
