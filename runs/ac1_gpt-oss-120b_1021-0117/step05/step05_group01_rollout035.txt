import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    # --------------------------------------------------------

        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:

    if s < 1e-12:
        a[:] = 1.0
        s = a.sum()
    a = a / s                         # enforce sum = 1

    n = a.size
