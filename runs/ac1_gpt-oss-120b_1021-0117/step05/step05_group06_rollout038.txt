    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 1.0)
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 2000
    # baseline uniform
    best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    best_score = evaluate_sequence(best_seq.tolist())
    best_m = n
    # track best positions set
    best_positions = None

    # experiences: run attempts within time
    attempt = 0
    while time.monotonic() < deadline:
        attempt += 1
        # random order of indices
        perm = rng.permutation(n)
        positions = []
        sums_set = set()
        for idx in perm:
            # quick skip if idx*2 already taken
            s2 = idx * 2
            if s2 in sums_set:
                continue
            conflict = False
            for p in positions:
                if idx + p in sums_set:
                    conflict = True
                    break
            if conflict:
                continue
            # add idx
            positions.append(idx)
            # update sums_set
            for p in positions:
                sums_set.add(idx + p)  # includes 2*idx
        m = len(positions)
        if m == 0:
            continue
        # compute theoretical bound (fast)
        score = 2.0 * n / (m * m)
        if score < best_score:
            best_m = m
            best_positions = positions.copy()
            # construct sequence
            seq = np.zeros(n, dtype=np.float64)
            seq[best_positions] = 1.0 / best_m
            best_score = evaluate_sequence(seq.tolist())
            best_seq = seq

            if best_score < 1.0:
                break

    # ensure non-empty
    if best_seq.size == 0:

    return [float(x) for x in best_seq]

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Project onto the probability simplex (sum = 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Maximum value of the (full) convolution a*a using FFT
# ----------------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    """Return max(conv_full) for a*a using zero‑padded FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv_full.max())


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------------
    # initialise – use the supplied parent if present, otherwise random
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2048, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)                     # enforce sum = 1
    n = a.size

    # --------------------------------------------------------------
    # choose FFT length (power of two ≥ 2n‑1)
    # --------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------------
    best = a.copy()
    best_sc = _max_conv(best, L)
    best_bound = 2.0 * n * best_sc
    cur = a.copy()
    cur_sc = best_sc

    # --------------------------------------------------------------
    # hyper‑parameters
    # --------------------------------------------------------------
    step0 = 1.0
    beta = 0.9
    momentum = np.zeros_like(a)
    eps0 = 5e-3
    eps_decay = 0.9994
    eps_min = 1e-9
    it = 0
    target_bound = 1.50274349

    # --------------------------------------------------------------
    # main optimisation loop (matrix‑free gradient descent)
    # --------------------------------------------------------------
    while time.monotonic() < deadline - 2.0:
        it += 1

        # full convolution and its maximal entry
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()

        # soft‑max weights (smooth approximation of the max)
        eps = max(eps_min, eps0 * (eps_decay ** it))
        if eps > 1e-7:
            w = np.exp((conv_full - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
        else:
            mask = conv_full >= max_val - 1e-12
            cnt = int(mask.sum())
            w = np.where(mask, 1.0 / cnt, 0.0)

        # gradient of the smoothed objective
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # back‑tracking line search
        alpha = step0 / math.sqrt(it + 1.0)
        improved = False
        for _ in range(8):
            cand = cur - alpha * momentum
            cand = _project_simplex(cand)
            cand_sc = _max_conv(cand, L)
            if cand_sc <= cur_sc:
                cur = cand
                cur_sc = cand_sc
                improved = True
                if cand_sc < best_sc:
                    best = cand.copy()
                    best_sc = cand_sc
                    best_bound = 2.0 * n * best_sc
                break
            alpha *= 0.5

        if not improved:

            if it % 200 == 0:
                delta = np.zeros_like(cur)
                i, j = rng.integers(0, n, size=2)
                delta[i] = 0.02
                delta[j] = -0.02
                cand = _project_simplex(cur + delta)
                cand_sc = _max_conv(cand, L)
                if cand_sc < cur_sc:

                    cur_sc = cand_sc
                    if cand_sc < best_sc:

                        best_sc = cand_sc

        # periodic symmetrisation (helps with Toeplitz structure)
        if it % 300 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _max_conv(sym, L)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best = sym.copy()
                    best_sc = sym_sc
                    best_bound = 2.0 * n * best_sc

        # early exit if we have already passed the target
        if best_bound < target_bound:
            break

    # ------------------------------------------------------------------
    # Simple stochastic hill‑climbing on the best candidate found above
    # ------------------------------------------------------------------
    cur = best.copy()
    cur_bound = best_bound
    while time.monotonic() < deadline - 0.2:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_sc = _max_conv(cur, L)
        new_bound = 2.0 * n * new_sc
        if new_bound < cur_bound:
            cur_bound = new_bound
            best = cur.copy()
            best_sc = new_sc
            best_bound = new_bound
            if best_bound < target_bound:
                break
        else:
            # revert the swap
            cur[i], cur[j] = cur[j], cur[i]

    # ------------------------------------------------------------------
    # Return the best candidate (as a plain Python list)
    # ------------------------------------------------------------------
    if best.size == 0:
        best = np.array([1.0])
    return [float(x) for x in best]
