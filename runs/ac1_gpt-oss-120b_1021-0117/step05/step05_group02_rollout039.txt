import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex (sum = 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Score used by the evaluator – assumes sum(a)=1
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())          # power of two ≥ 3n‑2
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Greedy construction of a large Sidon set (no repeated unordered sums)
# ------------------------------------------------------------
def _sidon_greedy(n: int, rng: np.random.Generator, deadline: float) -> list:
    # theoretical upper bound for size of a Sidon set in [0,n‑1]
    max_possible = int((math.isqrt(8 * n + 1) - 1) // 2)
    best_inds = []
    best_len = 0
    # try several random permutations until time runs out
    while time.monotonic() < deadline:
        order = rng.permutation(n)
        sums = set()
        inds = []
        add = inds.append
        for idx in order:
            # quick reject if self‑sum already used
            s0 = idx + idx
            if s0 in sums:
                continue
            # check sums with existing members
            conflict = False
            for j in inds:
                if idx + j in sums:
                    conflict = True
                    break
            if conflict:
                continue
            # accept idx
            add(idx)
            sums.add(s0)
            for j in inds[:-1]:
                sums.add(idx + j)
            # stop early if we already hit the theoretical bound
            if len(inds) >= max_possible:
                break
        if len(inds) > best_len:
            best_len = len(inds)
            best_inds = inds
            if best_len >= max_possible:
                break
    return best_inds

                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # --------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # --------------------------------------------------------------------
    return [float(x) for x in best_vec]
