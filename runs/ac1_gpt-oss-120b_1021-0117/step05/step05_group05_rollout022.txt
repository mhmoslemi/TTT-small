def _score(a, n=None):
    a = np.asarray(a, dtype=np.float64)
    if n is None:
        n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    max_val = conv.max()
    # sum of a may be slightly off 1 due to numeric errors; compute denominator
    s = a.sum()
    if s <= 0:
        return float('inf')
    return float(2.0 * n * max_val / (s * s))

import time, math, numpy as np

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    # ensure sum close to 1 (might lose a few eps)
    # renormalize tiny leftover due to numerical error
    s = w.sum()
    if s > 0:
        w *= 1.0 / s
    return w

def _score(a, n=None):
    a = np.asarray(a, dtype=np.float64)
    if n is None:
        n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    if s <= 0:
        return float('inf')
    return float(2.0 * n * max_val / (s * s))
