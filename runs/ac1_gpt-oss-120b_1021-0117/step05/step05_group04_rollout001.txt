import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ------------------------------------------------------------
def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale vector to have sum = 1 (non‑negative)."""
    total = v.sum()
    if total == 0.0:
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / total


def _self_conv_max(a: np.ndarray, L: int):
    """
    Compute max of a*a (linear convolution) via FFT.
    Returns (max_value, a_fft, full_convolution_vector).
    """
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv.max(), a_fft, conv


def _softmax_weights(conv: np.ndarray, max_val: float, eps: float) -> np.ndarray:
    """
    Soft‑max over convolution values with temperature eps.
    """
    # (conv - max_val)/eps <= 0, safe exponentials
    w = np.exp((conv - max_val) / eps)
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size, dtype=np.float64)
    return w / s


def _gradient(a_fft: np.ndarray, w: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of the smoothed max‑conv objective.
    Returns gradient vector of length n.
    """
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    # indices mapping for length‑n gradient
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty list of non‑negative coefficients that (heuristically)
    minimizes the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------
    # Initialise vector (use parent if provided)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
    else:
        # fallback: random positive vector
        cur = rng.random(2000, dtype=np.float64)

    cur = _normalize(cur)
    n = cur.size

    # FFT length ≥ 2n‑1 (full linear convolution)
    L = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------
    eps0 = 0.4                 # initial soft‑max temperature
    eps_decay = 0.9993         # exponential decay per iteration
    eps_min = 1e-12
    step0 = 0.38               # base step size
    momentum_factor = 0.94
    momentum = np.zeros_like(cur)
    best_vec = cur.copy()
    best_max, cur_fft, cur_conv = _self_conv_max(cur, L)
    best_score = 2.0 * n * best_max
    target_score = 1.50274349
    no_improve = 0
    it = 0

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution & max -----
        cur_max, cur_fft, cur_conv = _self_conv_max(cur, L)

        # ----- soft‑max weights -----
        eps = max(eps_min, eps0 * (eps_decay ** it))

        # ----- gradient (matrix‑free) -----
