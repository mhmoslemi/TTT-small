def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1)
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0)
    cur = _project_simplex(cur)
    n = cur.shape[0]

    L_pow = (2*n - 1).bit_length()
    L = 1 << L_pow
    idx = (2*n - 2) - np.arange(n)

    # optimizer params
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.1
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1_pow = 1.0
    beta2_pow = 1.0

    eps0 = 1.0
    eps_decay = 0.995
    eps_min = 1e-9

    best_vec = cur.copy()
    a_fft = np.fft.rfft(cur, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    best_sc = 2.0 * n * conv.max()
    cur_sc = best_sc

    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1
        # compute fft and conv (already maybe computed but we recompute for each iteration after updating cur)
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv_full.max()
        cur_sc = 2.0 * n * max_val

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            no_improve = 0

        # temperature
        eps = eps0 * (eps_decay ** it)
        if eps < eps_min:
            eps = eps_min

        # compute softmax weights
        if eps > 1e-12:
            w = np.exp((conv_full - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.size)
            else:
                w /= w_sum
        else:
            w = np.zeros_like(conv_full)
            mask = conv_full == max_val
            cnt = np.count_nonzero(mask)
            if cnt > 0:
                w[mask] = 1.0 / cnt

        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # Adam update
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step
        cand = _project_simplex(cand)

        # compute candidate score
        cand_fft = np.fft.rfft(cand, n=L)
        cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[:2*n-1]
        cand_sc = 2.0 * n * cand_conv.max()

        # accept criteria
        if cand_sc <= cur_sc or rng.random() < math.exp(-(cand_sc - cur_sc) / (lr0 + 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
        else:

        # occasional random perturb

                    best_sc = tmp_sc
                    best_vec = tmp.copy()
                    no_improve = 0

        # early stop if best_sc already below target (maybe)
        # Not necessary but we can break if time.

    # return best solution as list
    return [float(x) for x in best_vec]

import time, math, numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin before the hard sandbox timeout
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # Initial vector (use the supplied parent if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fall back to a random dense start of comparable length
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)            # enforce sum = 1
    n = cur.size

    # ------------------------------------------------------------------
    # FFT planning (zero‑padded length, power of two)
    # ------------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())   # >= 2n‑1
    # indices that pick the gradient slice (see derivation in original code)
    idx = (2 * n - 2) - np.arange(n)

    # ------------------------------------------------------------------
    # Optimiser hyper‑parameters
    # ------------------------------------------------------------------
    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1_pow, beta2_pow = 1.0, 1.0

    # temperature for soft‑max approximation of the max
    eps0 = 0.5
    eps_decay = 0.999
    eps_min = 1e-4

    # bookkeeping
    a_fft = np.fft.rfft(cur, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    best_sc = 2.0 * n * conv_full.max()
    best_vec = cur.copy()
    cur_sc = best_sc

    it = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and its max -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        cur_sc = 2.0 * n * max_val

        # keep best so far -------------------------------------------------
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
            if best_sc < target:
                break

        # ----- soft‑max weights (temperature) -----
        eps = eps0 * (eps_decay ** it)
        if eps < eps_min:
            eps = eps_min
        if eps > 1e-12:
            w = np.exp((conv_full - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.size)
            else:
                w /= w_sum
        else:
            # hard max – equal weight on all maximal positions
            w = np.zeros_like(conv_full)
            mask = conv_full == max_val

        w_rev = w[::-1]

        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----

        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

            if cur[i] > 0.0:

    # ------------------------------------------------------------------
