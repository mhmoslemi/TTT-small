    # sequence: list of numbers
    if not isinstance(sequence, list): return inf
    if not sequence: return inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return inf

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score(a):
    """Score = 2*n*max(conv(a,a)) for a that already satisfies sum(a)=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    lr0 = 0.2                         # base learning rate

    # ------------------------------------------------------------------
    # Tracking best solution
    # ------------------------------------------------------------------
    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    it = 0
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while True:
        if time.monotonic() >= deadline:
            break
        it += 1

        # ----- convolution & sub‑gradient of max(conv) -----------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # weight over all (almost) active shifts
        tol = max_val * 1e-8 + 1e-12
        mask = conv >= max_val - tol
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 1:
            w /= cnt

        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]               # sub‑gradient of max(conv)

        # ----- Adam update ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        beta1_pow *= beta1
        beta2_pow *= beta2

        step = lr0 / math.sqrt(it)
        cand = cur - step * m_hat / (np.sqrt(v_hat) + eps)
        cand = _project_simplex(cand)

        cand_sc = _score(cand)

        # ----- accept if improvement --------------------------------------
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            # tiny chance to move anyway – helps escape plateaus
            if rng.random() < 0.001:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- occasional symmetrisation ----------------------------------
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # ----- random redistribution when stuck ---------------------------
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = cur[i] * rng.random() * 0.5
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc
                    no_improve = 0

        # ----- early exit if target reached --------------------------------
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return the best found vector as a plain Python list of floats
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
```
