import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score used by the evaluator – assumes sum(a)=1
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Return 2 * n * max(convolution(a, a)) for sum(a)=1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Symmetrisation / unimodal projection helpers
# ------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry."""
    return (v + v[::-1]) * 0.5


def _sym_unimodal(v: np.ndarray) -> np.ndarray:
    """Arrange mass decreasingly with distance from the centre, then symmetrise."""
    n = v.size
    # order indices by distance to centre (closer first)
    order = np.argsort(np.abs(np.arange(n) - (n - 1) / 2))
    sorted_vals = np.sort(v)[::-1]  # descending
    new = np.empty_like(v)
    new[order] = sorted_vals
    new = _symmetrize(new)
    # re‑project onto simplex to keep sum = 1
    new = _project_simplex(new)
    return new

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence with a low AC1 bound.
    The function runs for up to `budget_s` seconds and returns the best
    sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the harness timeout
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initialise – use the parent if present, otherwise uniform
    # --------------------------------------------------------
    if "height_sequence_1" in globals():

    else:

        cur = np.full(n0, 1.0 / n0, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)            # enforce sum = 1
    n = cur.size

    # --------------------------------------------------------

    # --------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------

    # --------------------------------------------------------
    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    # --------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------
    step0 = 0.6               # base step size for momentum updates
    step0_mirror = 0.35       # base step for mirror‑descent updates
    beta = 0.9                # momentum coefficient
    momentum = np.zeros_like(cur)

    it = 0

    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of cur with itself -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        max_idx = int(conv.argmax())

        # ----- exact gradient of the max‑convolution -----
        g = np.zeros_like(cur)
        low_i = max(0, max_idx - (n - 1))

                no_improve = 0

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
