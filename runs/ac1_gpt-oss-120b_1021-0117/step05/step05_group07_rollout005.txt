import time, math, numpy as np
from numpy.fft import rfft, irfft

# ------------------------------------------------------------
# Project onto the probability simplex {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

    idx = (2 * n - 2) - np.arange(n)      # index map for gradient

    # --------------------------------------------------------
    # Helper to compute score, full convolution and its FFT
    # --------------------------------------------------------
    def _score_conv_fft(a):
        a_fft = rfft(a, n=L)
        conv_full = irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        # sum(a) == 1 after projection, so score = 2 * n * max_val
        return 2.0 * n * max_val, conv_full, a_fft

    cur_sc, cur_conv, cur_fft = _score_conv_fft(cur)

    best_sc = cur_sc
    best_vec = cur.copy()

    # --------------------------------------------------------
    # Adam optimiser parameters
    # --------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    lr0 = 0.2                     # base learning rate
    m = np.zeros_like(cur)       # first moment
    v = np.zeros_like(cur)       # second moment

    # temperature schedule for soft‑max
    temp0 = 0.5
    temp_decay = 0.995
    temp_min = 1e-6

    it = 0
    no_improve = 0
    target = 1.50274349

    # --------------------------------------------------------
    # Optimisation loop (matrix‑free, O(n log n) per iteration)
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- temperature and soft‑max weights ----
        temp = max(temp_min, temp0 * (temp_decay ** it))
        max_val = cur_conv.max()
        w = np.exp((cur_conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of soft‑max max(conv) ----
        w_rev = w[::-1]
        w_fft = rfft(w_rev, n=L)
        grad_full = irfft(cur_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]               # size n

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + adam_eps)

        # ---- candidate generation ----
        cand = cur - step
        cand = _project_simplex(cand)

        # ---- evaluate candidate ----
        cand_sc, cand_conv, cand_fft = _score_conv_fft(cand)

        # ---- accept if improvement ----
        if cand_sc < cur_sc:
            cur, cur_sc, cur_conv, cur_fft = cand, cand_sc, cand_conv, cand_fft
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            no_improve += 1

        # ---- occasional symmetrisation ----
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc, sym_conv, sym_fft = _score_conv_fft(sym)
            if sym_sc < cur_sc:
                cur, cur_sc, cur_conv, cur_fft = sym, sym_sc, sym_conv, sym_fft
                no_improve = 0
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # ---- random mass‑transfer when stuck ----
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc, tmp_conv, tmp_fft = _score_conv_fft(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc, cur_conv, cur_fft = tmp, tmp_sc, tmp_conv, tmp_fft
                    no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                    no_improve = 0

        # ---- early exit if target reached ----
        if best_sc < target:
            break

    # --------------------------------------------------------
    # Return best candidate as plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
