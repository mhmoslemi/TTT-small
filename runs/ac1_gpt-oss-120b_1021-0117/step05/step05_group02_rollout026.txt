import numpy as np

def _score(arr: np.ndarray) -> float:
    """Score = 2 * n * max(conv(arr,arr)) / (sum(arr)**2),
    but for a normalized vector (sum=1) it's simply 2*n*max(conv)."""
    n = arr.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power‑of‑two
    af = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence that (approximately) minimises
    the AC1 upper bound.  The implementation builds a uniform step‑function
    of the appropriate length (the length of the parent if provided) and,
    as a fall‑back, also evaluates the provided parent after normalisation.
    The best of the two candidates is returned.
    """
    # Length of the problem – use the parent if it exists.
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 1024                     # a reasonable default when no parent is given

    # ----- candidate 1: uniform distribution (theoretical optimum) -----
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # ----- candidate 2: the supplied parent, normalised to sum = 1 ----------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.maximum(parent, 0.0)            # enforce non‑negativity
        s = parent.sum()
        if s < 0.01:                               # avoid invalid (rare) case
            parent = uniform.copy()
        else:
            parent = parent / s                     # simple normalisation
    else:
        parent = uniform.copy()

    # Evaluate both candidates (fast FFT‑based evaluation)
    sc_uniform = _score(uniform)
    sc_parent  = _score(parent)

    best = uniform if sc_uniform <= sc_parent else parent

    # Return as plain Python list of floats
    return [float(x) for x in best]
