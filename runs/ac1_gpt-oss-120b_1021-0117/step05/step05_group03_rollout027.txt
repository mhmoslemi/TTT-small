import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto probability simplex."""
    n = v.size
    if n == 0: 
        return v
    # sort v descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _max_convolution(a, L):
    """Return max of a*a convolution using FFT length L."""
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*a.size-1]
    return conv.max()

    a0 = np.maximum(a0, 0.0)

    beta2 = 0.999

    mask_tol = 1e-12

        max_val = conv.max()

        w = mask.astype(np.float64)

        else:

        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a0 = rng.random(1024, dtype=np.float64)
    a0 = np.maximum(a0, 0.0)
    cur = _project_simplex(a0)
    n = cur.size

    # FFT length sufficient for convolutions up to length 3n‑2
    L = 1 << ((3 * n - 2).bit_length())

    # ---- bookkeeping ----------------------------------------------------
    cur_score = 2.0 * n * _max_convolution(cur, L)
    best = cur.copy()
    best_score = cur_score

    # ---- Adam optimiser state -------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha0 = 1.0                     # base learning rate
    it = 0
    no_improve = 0
    sym_period = 300
    mask_tol = 1e-12

    # --------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- autocorrelation and active mask -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        mask = np.isclose(conv, max_val, atol=mask_tol)
        w = mask.astype(np.float64)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # ----- sub‑gradient of the max‑convolution -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = (alpha0 / math.sqrt(it)) * (m_hat / (np.sqrt(v_hat) + eps))

        cand = cur - step
        cand = np.maximum(cand, 0.0)
        cand = _project_simplex(cand)
        cand_score = 2.0 * n * _max_convolution(cand, L)

        # ----- accept/reject (with occasional uphill) -----
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            no_improve = 0
        else:
            prob = math.exp(-(cand_score - cur_score) / (alpha0 + 1e-12))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score
            no_improve += 1

        if cand_score < best_score:
            best, best_score = cand.copy(), cand_score

        # ----- periodic symmetrisation -----
        if it % sym_period == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = 2.0 * n * _max_convolution(sym, L)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                no_improve = 0
                if sym_score < best_score:
                    best, best_score = sym.copy(), sym_score

        # ----- random mass‑transfer when stuck -----
        if no_improve > 4000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score = 2.0 * n * _max_convolution(tmp, L)
                if tmp_score < cur_score:
                    cur, cur_score = tmp, tmp_score
                    no_improve = 0
                if tmp_score < best_score:
                    best, best_score = tmp.copy(), tmp_score

        # ----- early exit if target reached -----
        if best_score < 1.50274349:
            break

    # --------------------------------------------------------------------
    return [float(x) for x in best]
