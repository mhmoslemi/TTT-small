import time, math, numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum=1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Core optimisation routine
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin
    deadline = start + max(0.0, float(budget_s) - 0.25)

    # --------------------------------------------------------
    # Initialise – use the parent if present, otherwise random
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)                 # sum = 1
    n = cur.size

    # --------------------------------------------------------
    # FFT set‑up (length >= 3n‑2, power of two)
    # --------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())
    idx = (2 * n - 2) - np.arange(n)            # indices for gradient extraction

    # --------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------
    def _conv_and_score(a_vec):
        a_fft = np.fft.rfft(a_vec, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        return conv_full, 2.0 * n * conv_full.max()

    conv_full, cur_sc = _conv_and_score(cur)
    best_sc = cur_sc
    best_vec = cur.copy()

    # --------------------------------------------------------
    # Optimisation parameters
    # --------------------------------------------------------
    eps0 = 0.5                # initial temperature for soft‑max
    eps_decay = 0.996
    eps_min = 1e-12

    beta1 = 0.9               # Adam first‑moment decay
    beta2 = 0.999             # Adam second‑moment decay
    eta = 0.08                # base learning rate
    eps_adam = 1e-8

    m = np.zeros_like(cur)    # first moment
    v = np.zeros_like(cur)    # second moment

    no_improve = 0
    target = 1.50274349

    it = 0
    # --------------------------------------------------------

    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- temperature for soft‑max -----
        eps = max(eps_min, eps0 * (eps_decay ** it))

        # ----- soft‑max weights over the convolution -----
        max_val = conv_full.max()
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed maximum -----
        a_fft = np.fft.rfft(cur, n=L)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]                     # size n

        # ----- Adam moment updates -----
