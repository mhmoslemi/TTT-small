    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Scoring – same definition used by the evaluator (sum=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a, a))."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # power‑of‑two >= 2n‑1
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Core optimizer – matrix‑free, FFT‑based Adam + mirror step
# ------------------------------------------------------------
def _optimize(initial: np.ndarray, rng: np.random.Generator,
              deadline: float, target: float) -> tuple[np.ndarray, float]:
    a = _project_simplex(np.maximum(initial, 0.0))
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.18                     # base learning rate

    # Soft‑max smoothing for the max
    eps0, eps_decay, eps_min = 0.05, 0.9995, 1e-7

    mt = np.zeros_like(a)
    vt = np.zeros_like(a)

    best = a.copy()
    best_sc = _score(best)

    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1
        # ----------------------------------------------------
        # Convolution & soft‑max weights
        # ----------------------------------------------------
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----------------------------------------------------
        # Gradient of the smoothed objective
        # ----------------------------------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----------------------------------------------------
        # Adam step
        # ----------------------------------------------------
        mt = beta1 * mt + (1.0 - beta1) * g
        vt = beta2 * vt + (1.0 - beta2) * (g * g)
        mt_hat = mt / (1.0 - beta1 ** it)
        vt_hat = vt / (1.0 - beta2 ** it)

        lr = lr0 / math.sqrt(it)
        step = lr * mt_hat / (np.sqrt(vt_hat) + eps_adam)
        cand1 = _project_simplex(a - step)

        # ----------------------------------------------------
        # Mirror‑descent (multiplicative) step
        # ----------------------------------------------------
        eta = 0.5 * lr
        exp_arg = -eta * g
        np.clip(exp_arg, -50, 50, out=exp_arg)
        cand2 = a * np.exp(exp_arg)
        s = cand2.sum()
        if s > 0.0:
            cand2 /= s
        else:
            cand2 = a.copy()

        # ----------------------------------------------------
        # Choose better of the two candidates
        # ----------------------------------------------------
        sc1 = _score(cand1)
        sc2 = _score(cand2)
        if sc1 <= sc2:
            cand, cand_sc = cand1, sc1
        else:
            cand, cand_sc = cand2, sc2

        cur_sc = _score(a)

        # ----------------------------------------------------
        # Acceptance (accept improvements, occasional uphill)
        # ----------------------------------------------------
        if cand_sc < cur_sc:
            a, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                a, cur_sc = cand, cand_sc
            no_improve += 1

        # ----------------------------------------------------
        # Periodic symmetrisation (helps with Toeplitz structure)
        # ----------------------------------------------------
        if it % 600 == 0:
            sym = (a + a[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                a, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # ----------------------------------------------------
        # Random mass transfer when stuck
        # ----------------------------------------------------
        if no_improve > 2500:
            i = rng.integers(0, n)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    a, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----------------------------------------------------
        # Early exit if target reached
        # ----------------------------------------------------
        if best_sc < target:
            break

    return best, best_sc


# ------------------------------------------------------------
# Public entry point – respects budget and returns best list
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence with a low AC1 bound.
    The function runs for up to `budget_s` seconds and returns the best
    sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # keep a safety margin

    # ----------------------------------------------------
    # Determine problem size and some initial candidates
    # ----------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2048                     # fallback length if parent missing

    # Build a list of diverse starts (parent, uniform, random)
    inits = []
    if "height_sequence_1" in globals():
        inits.append(np.asarray(height_sequence_1, dtype=np.float64))
    # uniform start
    inits.append(np.full(n, 1.0 / n, dtype=np.float64))
    # random start (scaled to sum 1)
    rand_vec = rng.random(n)
    rand_vec = np.maximum(rand_vec, 0.0)
    rand_vec /= rand_vec.sum()
    inits.append(rand_vec)

    target = 1.50274349
    best_overall = None
    best_score = np.inf

    # ----------------------------------------------------
    # Run optimizer on each start while time permits
    # ----------------------------------------------------
    for init in inits:
        if time.monotonic() >= deadline:
            break
        cand, sc = _optimize(init, rng, deadline, target)
        if sc < best_score:
            best_overall, best_score = cand, sc
        if best_score < target:
            break

    # ----------------------------------------------------
    # Fallback: if nothing succeeded, return a uniform vector
    # ----------------------------------------------------
    if best_overall is None or best_score == np.inf:
        best_overall = np.full(n, 1.0 / n, dtype=np.float64)

    # Convert to plain Python list of floats as required
    return [float(x) for x in best_overall]
