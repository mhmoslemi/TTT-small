    rng = np.random.default_rng(seed)
    start=time.monotonic()
    deadline=start+budget_s-0.5

    # Precompute index mapping for gradient extraction
    idx = (2*n-2) - np.arange(n)

    momentum=np.zeros_like(cur)
    # Adam
    beta1=0.9
    beta2=0.999

    v=np.zeros_like(cur)
    adam_eps=1e-8
    # bookkeeping
    best_vec=cur.copy()
    best_sc=_score(cur)
    cur_sc=best_sc
    no_improve=0
    it=0
    target=1.50274349
    while time.monotonic()<deadline:
        it+=1
        # compute conv and gradient
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[:2*n-1]
        max_val = conv.max()
        # softmax weights
        eps = max(eps_min, eps0 * (eps_decay**it))
        w = np.exp((conv - max_val)/eps)
        w_sum = w.sum()
        if w_sum == 0:
            w.fill(1.0/w.size)
        else:
            w /= w_sum

        w_fft = np.fft.rfft(w_rev, n=L)

        cand_mom_sc = _score(cand_mom)

        s = cand_mir.sum()

        v = beta2 * v + (1-beta2) * (g**2)

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Scoring function (matches the evaluator's internal metric)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(convolution(a, a))."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
