import time
import math
import numpy as np

def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x) = z}."""
    n = v.size
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        a = np.maximum(a, 0.0)

            a[:] = 1.0
    else:
        a = rng.random(2048, dtype=np.float64)

    a = a / a.sum()               # enforce sum = 1
    n = a.size

    L = 1 << ((3 * n - 2).bit_length())

    # Book‑keeping
    # ------------------------------------------------------------------

    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    best_sc = 2.0 * n * conv.max()
    best_vec = a.copy()

    # optimisation parameters
    step0 = 0.6

    sym_interval = 300
    no_improve = 0
    it = 0
