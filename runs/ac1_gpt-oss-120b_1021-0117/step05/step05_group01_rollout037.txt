import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolution (linear) via FFT, returning full result
# ----------------------------------------------------------------------
def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * a.size - 1]

# ----------------------------------------------------------------------
# Gradient of the max‑convolution entry (sub‑gradient)
# ----------------------------------------------------------------------
def _grad_max_conv(a: np.ndarray, k_max: int) -> np.ndarray:
    """Sub‑gradient of the convolution entry attaining the maximum."""
    n = a.size
    g = np.zeros_like(a)
    if k_max < n:
        # overlap region = [0, k_max]
        g[: k_max + 1] = 2.0 * a[k_max::-1]
    else:
        offset = k_max - (n - 1)          # start index of overlap on the left side
        length = 2 * n - 1 - k_max        # size of the overlapping segment
        g[offset:] = 2.0 * a[:length][::-1]
    return g

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------------
    # Initial vector – use the parent if present, otherwise random
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2048, dtype=np.float64)

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                     # enforce sum = 1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1

    # --------------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------------
    best_vec = a.copy()
    conv = _conv_full(a, L)
    best_sc = 2.0 * n * conv.max() / (a.sum() ** 2)

    # optimisation parameters
    step0 = 1.0
    beta = 0.9
    momentum = np.zeros_like(a)
    sym_interval = 150
    no_improve = 0
    it = 0
    target = 1.50274349

    # --------------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and its maximum -----
        conv = _conv_full(a, L)
        max_val = conv.max()
        k_max = int(conv.argmax())
        cur_sc = 2.0 * n * max_val / (a.sum() ** 2)

        # ----- keep the best seen so far -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- early exit if target reached -----
        if best_sc < target:
            break

        # ----- sub‑gradient at the maximal entry -----
        g = _grad_max_conv(a, k_max)

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- back‑tracking line search -----
        base_alpha = step0 / math.sqrt(it)
        alpha = base_alpha
        cand = None
        cand_sc = None
        for _ in range(8):

            a_tmp = _project_simplex(a_tmp)          # keep on simplex

                break

            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            conv_sym = _conv_full(a_sym, L)
            sc_sym = 2.0 * n * conv_sym.max() / (a_sym.sum() ** 2)
            if sc_sym < cur_sc:
                a, cur_sc = a_sym, sc_sym
                if sc_sym < best_sc:
                    best_sc, best_vec = sc_sym, a_sym.copy()
                    no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 1500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.35
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a)
                no_improve = 0

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
