import time, math, numpy as np

# --------------------------------------------------------------
# Simplex projection (L2 onto {x>=0, sum=1})
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # renormalise to avoid tiny numerical drift
    s = w.sum()
    if s > 0:
        w /= s
    return w


# --------------------------------------------------------------
# Max of full convolution a*a using zero‑padded FFT
# --------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    """Return max(conv_full) for a*a using FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())


# --------------------------------------------------------------
# Deterministic anti‑identity permutation (large values opposite small)
# --------------------------------------------------------------
def _anti_identity(arr: np.ndarray) -> np.ndarray:
    """Return a copy of arr with largest, smallest, ... alternating at the ends."""
    vals = np.sort(arr)[::-1]
    out = np.empty_like(vals)
    l, r = 0, len(vals) - 1
    k = 0
    while l <= r:
        out[l] = vals[k]
        k += 1
        if l < r:
            out[r] = vals[k]
            k += 1
        l += 1
        r -= 1
    return out

        best_a = np.array([1.0], dtype=np.float64)

    return [float(x) for x in best_a]
