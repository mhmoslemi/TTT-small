# Helper functions (top‑level, no closures)

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x (x>0)."""
    return 1 << (x - 1).bit_length()


def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to have sum = 1 (protect against zero)."""
    s = v.sum()
    if s == 0.0:
        v[:] = 1.0 / v.size
        return v
    v /= s
    return v


def _conv_fft(a: np.ndarray, L: int) -> tuple:
    """Full linear convolution of a with itself, using FFT length L.
    Returns (conv_full, a_fft) where conv_full is length 2·len(a)-1."""
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv_full, a_fft


def _softmax_weights(conv: np.ndarray, cur_max: float, eps: float) -> np.ndarray:
    """Soft‑max over convolution values with temperature eps (eps>0)."""
    # the exponent is non‑positive, overflow unlikely; still clamp for safety
    exp_arg = (conv - cur_max) / eps
    exp_arg = np.clip(exp_arg, -50.0, 0.0)
    w = np.exp(exp_arg)
    s = w.sum()
    if s == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= s
    return w


def _gradient(a_fft: np.ndarray, w: np.ndarray, n: int, L: int) -> np.ndarray:
    """Gradient of the smoothed max‑conv objective.
    Implements: g = 2·(a * rev(w)) in the appropriate slice."""
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
    else:
        # modest random start
        cur = rng.random(1024, dtype=np.float64)

    cur = _normalize(cur)
    n = cur.size
    L = _next_pow2(2 * n - 1)                     # FFT length ≥ 2n‑1

    # ------------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------------
    eps0 = 0.5
    eps_decay = 0.9995
    eps_min = 1e-12

    step0 = 0.4
    beta = 0.9                      # momentum coefficient

    target_score = 1.50274349

    momentum = np.zeros_like(cur)

    # best‑so‑far bookkeeping
    conv_cur, a_fft_cur = _conv_fft(cur, L)
    best_max = conv_cur.max()
    best_vec = cur.copy()
    best_score = 2.0 * n * best_max

    no_improve = 0
    it = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution & max -----
        conv_cur, a_fft_cur = _conv_fft(cur, L)
        cur_max = conv_cur.max()

        # ----- soft‑max weights (smooth approximation of max) -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = _softmax_weights(conv_cur, cur_max, eps)

        # ----- gradient (matrix‑free via FFT) -----
        grad = _gradient(a_fft_cur, w, n, L)

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- exponentiated‑gradient candidate -----
        step = step0 / math.sqrt(it)
        exponent = -step * momentum
        exponent = np.clip(exponent, -50.0, 50.0)
        cand = cur * np.exp(exponent)
        cand = _normalize(cand)

        # ----- evaluate candidate -----
        conv_cand, _ = _conv_fft(cand, L)
        cand_max = conv_cand.max()

        # ----- accept if improvement, otherwise Metropolis move -----
        if cand_max < cur_max - 1e-15:
            cur = cand
            cur_max = cand_max
            no_improve = 0
        else:
            no_improve += 1
            prob = math.exp(-(cand_max - cur_max) / (step + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_max = cand_max
                no_improve = 0

        # ----- record best -----
        if cur_max < best_max - 1e-13:
            best_max = cur_max
            best_vec = cur.copy()
            best_score = 2.0 * n * best_max
            no_improve = 0

        # ----- occasional symmetrisation (helps for Toeplitz structure) -----
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _normalize(sym)
            conv_sym, _ = _conv_fft(sym, L)
            sym_max = conv_sym.max()
            if sym_max < cur_max:
                cur = sym
                cur_max = sym_max
                if sym_max < best_max:
                    best_max = sym_max
                    best_vec = sym.copy()
                    best_score = 2.0 * n * best_max
                no_improve = 0

        # ----- random mass‑transfer when stuck for a while -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.35
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _normalize(tmp)
                conv_tmp, _ = _conv_fft(tmp, L)
                tmp_max = conv_tmp.max()
                if tmp_max < cur_max:
                    cur = tmp
                    cur_max = tmp_max
                    if tmp_max < best_max:
                        best_max = tmp_max
                        best_vec = tmp.copy()
                        best_score = 2.0 * n * best_max
                    no_improve = 0
                else:
                    no_improve = 0

        # ----- hard restart if stuck far too long -----
        if no_improve > 20000:
            cur = rng.random(n, dtype=np.float64)
            cur = _normalize(cur)
            momentum.fill(0.0)
            no_improve = 0

        # ----- early exit if target already reached -----
        if best_score < target_score:
            break

    # ------------------------------------------------------------------
    # Return best sequence found (plain Python list)
    # ------------------------------------------------------------------
    return [float(v) for v in best_vec]
