import time, math
import numpy as np

# ------------------------------------------------------------
# Fast AC1 bound (sum of coefficients ≈ 1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """2 * n * max (linear convolution of a with itself)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Simulated‑annealing / hill‑climbing search
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # safety margin – leave a bit of time for the final return
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # -------------------- initialise --------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
    else:                                 # fallback: exponential decay
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 100.0)

    a0 = a0 / a0.sum()                    # normalise to sum = 1
    n = a0.shape[0]

    cur = a0.copy()
    cur_sc = _score(cur)
    best = cur.copy()
    best_sc = cur_sc

    # -------------------- SA parameters --------------------
    temperature = 0.05                     # initial temperature
    decay = 0.99996                        # exponential cooling
    max_delta_frac = 0.12                  # max fraction of mass moved per step
    sym_interval = 300                     # how often to symmetrise
    no_improve = 0
    last_sym = 0

    # pre‑compute FFT length once
    L = 1 << ((2 * n - 1).bit_length())

    # -------------------- main loop --------------------
    it = 0
    while time.monotonic() < deadline:
        it += 1

        # ---- random mass transfer between two positions ----
        i = rng.integers(0, n)
        if cur[i] == 0.0:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        # move a small fraction of mass from i to j
        delta = cur[i] * rng.random() * max_delta_frac
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # ---- evaluate ----
        cand_sc = _score(cand)

        # ---- Metropolis acceptance ----
        if cand_sc < cur_sc or rng.random() < math.exp(-(cand_sc - cur_sc) / (temperature + 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best = cand.copy()
                best_sc = cand_sc
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ---- periodic symmetrisation (helps to lower the peak) ----
        if it - last_sym >= sym_interval:
            sym = (cur + cur[::-1]) * 0.5
            # projection onto the simplex is just normalising sum to 1
            sym = np.maximum(sym, 0.0)
            s = sym.sum()
            if s > 0.0:
                sym /= s
                sym_sc = _score(sym)
                if sym_sc < cur_sc:
                    cur = sym
                    cur_sc = sym_sc
                    if sym_sc < best_sc:
                        best = sym.copy()
                        best_sc = sym_sc
                    no_improve = 0
            last_sym = it

        # ---- if stuck, restart from best + tiny noise ----
        if no_improve > 5000:
            noise = rng.random(n) * 0.005
            cur = best + noise
            cur = np.maximum(cur, 0.0)
            cur /= cur.sum()
            cur_sc = _score(cur)
            temperature = 0.05
            no_improve = 0

        # ---- cool down ----
        temperature = max(temperature * decay, 5e-7)

        # ---- early exit if we already beat the target ----
        if best_sc < 1.50274349:
            break

    return [float(x) for x in best]
