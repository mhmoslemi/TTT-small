def _smooth_max_and_grad(a, L, alpha):
    """Given a vector a (sum=1), compute smooth max of conv and gradient."""
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*n-1]  # length 2n-1
    # compute softmax weights
    maxval = conv.max()
    exp_vals = np.exp(alpha * (conv - maxval))  # avoid overflow
    w = exp_vals / exp_vals.sum()
    # gradient
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    # Map to gradient indices size n: grad_i = 2 * grad_full[(2*n-2)-i]
    idx = (2*n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    smooth_max = (np.log(exp_vals.sum()) / alpha) + maxval
    return smooth_max, grad

import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection – Euclidean projection onto {x>=0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# Exact AC1 score (assumes sum(a)=1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return float(2.0 * n * conv_full[: 2 * n - 1].max())

# ------------------------------------------------------------
def _soft_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    m = conv.max()
    exps = np.exp(alpha * (conv - m))
    return exps / exps.sum()
