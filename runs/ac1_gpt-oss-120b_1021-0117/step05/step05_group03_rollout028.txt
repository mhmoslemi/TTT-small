import time

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation for sum=1 vectors (objective = 2*n*max(conv))
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, L: int, temp: float) -> np.ndarray:
    """
    Compute a gradient of 2*n*max(conv) using a soft‑max weighting.
    temp controls smoothing; smaller temp → sharper max.
    """
    n = a.shape[0]

        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / math.sqrt(it) * m_hat / (np.sqrt(v_hat) + eps)

        # ----- line‑search over a few scales -----
        factors = [1.0, 0.7, 0.5, 0.3, 0.1]
        cand = None
        cand_sc = None
        for f in factors:
            tmp = cur - f * step
            tmp = _project_simplex(tmp)
            sc = _score(tmp, L)
            if cand_sc is None or sc < cand_sc:
                cand, cand_sc = tmp, sc
        # ----- accept if improvement, otherwise occasional uphill -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            # simple Metropolis style uphill move
            prob = math.exp(-(cand_sc - cur_sc) / (step0 + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- periodic symmetrisation (helps for Toeplitz structure) -----
        if it % sym_period == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.35
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- early exit if target achieved -----
        if best_sc < target:
            break

    # ------------------------------------------------------------
    # Return best sequence found (plain Python list)
    # ------------------------------------------------------------
    return [float(x) for x in best_vec]
