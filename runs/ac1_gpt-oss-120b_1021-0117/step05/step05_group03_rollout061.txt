def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = rng.integers(500, 3000)
        a = rng.random(n, dtype=np.float64)
    a = np.maximum(a, 0.0)
    # Normalize sum to 1
    s = a.sum()
    if s < 1e-12:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a /= s
    n = a.size
    # precompute FFT length
    L = 1 << ((2*n - 2).bit_length() + 1)  # ensure >=2n-1
    # compute initial score
    def score_vec(v):
        af = np.fft.rfft(v, n=L)
        conv = np.fft.irfft(af * af, n=L)[:2*n-1]
        return float(2.0 * n * conv.max())
    best = a.copy()
    best_sc = score_vec(best)
    cur = best.copy()
    cur_sc = best_sc

    # parameters
    temp = 1.0
    temp_decay = 0.995
    min_temp = 0.001
    attempts = 0
    while time.monotonic() < deadline:
        # propose move: pick i!=j, transfer small delta
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        if cur[i] == 0:
            continue
        delta = cur[i] * rng.random() * temp
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        # maintain sum =1 automatically
        # compute score
        sc = score_vec(new)
        if sc < cur_sc or rng.random() < math.exp((cur_sc - sc) / max(temp, 1e-12)):
            cur = new
            cur_sc = sc
            if sc < best_sc:
                best = new.copy()
                best_sc = sc
        temp = max(min_temp, temp * temp_decay)
        attempts += 1
        # occasional reset: if no improvement for many steps, random perturbation
        if attempts % 5000 == 0:
            # reinitialize from best with random noise
            noise = rng.normal(scale=0.01, size=n)
            cand = best + noise
            cand = np.maximum(cand, 0)
            cand_sum = cand.sum()
            if cand_sum > 0:
                cand /= cand_sum
                sc = score_vec(cand)
                if sc < best_sc:
                    best = cand.copy()
                    best_sc = sc
                    cur = cand.copy()
                    cur_sc = sc
    return [float(x) for x in best]
```

import time
import math
import numpy as np

def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale non‑negative vector to have sum = 1 (leaving zeros untouched)."""
    s = v.sum()
    if s <= 0:
        # fallback to uniform if all zeros (should not happen)
        v[:] = 1.0 / v.size
    else:
        v /= s
    return v


def _score(v: np.ndarray, fft_len: int) -> float:
    """2 · n · max(conv(v,v)) – assumes sum(v)==1."""
    n = v.shape[0]
    fv = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(fv * fv, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
