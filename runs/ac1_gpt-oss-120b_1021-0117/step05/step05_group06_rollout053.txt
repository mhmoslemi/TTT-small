import time
import math
import numpy as np

# --------------------------------------------------------------
# Simplex projection (L2 onto {x>=0, sum=1})
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    if v.ndim != 1:
        v = v.ravel()
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# --------------------------------------------------------------
# Maximum value of the (linear) convolution a * a (full length 2n‑1)
# --------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    """Return max of full convolution a*a using zero‑padded FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    # keep only the first 2n-1 entries (the non‑zero part)
    conv = conv_full[: 2 * a.size - 1]
    return float(conv.max())
