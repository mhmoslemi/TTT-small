# ------------------------------------------------------------
# Simplex projection onto {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # fallback for the all‑zero case
    if w.sum() == 0.0:
        w = np.full_like(v, 1.0 / n)
    return w

            no_improve += 1
            # occasional stochastic jump when stuck
            if no_improve > 800:
                delta = rng.normal(loc=0.0, scale=0.02, size=n)
                tmp = cur + delta
                tmp = np.maximum(tmp, 0.0)
                tmp = _project_simplex(tmp)
                tmp_sc = float(evaluate_sequence(tmp.tolist()))
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- periodic symmetrisation (helps with the Toeplitz structure) -----
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = float(evaluate_sequence(sym.tolist()))
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # ----- early exit if we have already beaten the target -----
        if best_sc < target:
            break

    # --------------------------------------------------------
    # Return the best sequence discovered as a plain Python list
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
