Pseudo-code:

```
def _proj_simplex(v):
    # Project v onto simplex { w >= 0, sum w = 1 }
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # Find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _self_conv_max(a: np.ndarray, L: int):
    """
    Compute linear convolution a*a (full) via FFT and return its maximum value.
    Also returns a's rfft (for possible reuse) and the full convolution array.
    """
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return float(conv_full.max()), a_fft, conv_full


def _gradient_submax(a: np.ndarray, conv: np.ndarray, L: int) -> np.ndarray:
    """
    Subgradient of the max‑convolution functional.
    a – current vector (sum = 1)
    conv – full convolution a*a (length 2n‑1)
    Returns a length‑n gradient vector.
    """
    n = a.size
    max_val = conv.max()
    # mask of positions attaining the maximum (within a tiny tolerance)
    tol = max(1e-12, max_val * 1e-12)
    mask = (np.abs(conv - max_val) <= tol).astype(np.float64)
    # reversed a for the convolution that yields the subgradient
    a_rev = a[::-1]
    # FFT‑based convolution mask * a_rev
    mask_fft = np.fft.rfft(mask, n=L)
    rev_fft = np.fft.rfft(a_rev, n=L)
    grad_full = np.fft.irfft(mask_fft * rev_fft, n=L)
    # slice to obtain length‑n vector (indices n‑1 … n‑1+n‑1)
    return grad_full[(n - 1) : (n - 1 + n)]
