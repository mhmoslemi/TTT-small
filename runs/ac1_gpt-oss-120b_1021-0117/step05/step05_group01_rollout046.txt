import time

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x) = z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    n = v.size
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Gradient of the max entry of the (full) autocorrelation a*a
# ------------------------------------------------------------
def _grad_max_conv(a: np.ndarray, conv: np.ndarray, max_val: float) -> np.ndarray:
    n = a.size
    # indices where conv == max (tolerance for floating errors)
    idxs = np.where(np.abs(conv - max_val) < 1e-12)[0]
    if idxs.size == 0:  # fallback (should not happen)
        idxs = np.array([conv.argmax()], dtype=int)
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        g[i] += 2.0 * a[k - i]
    if idxs.size > 1:
        g /= idxs.size
    return g

    a = a / a.sum()                     # normalize to sum = 1

    # enforce (optional) symmetry at start – often beneficial
    a = (a + a[::-1]) * 0.5
    a = _project_simplex(a)

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length – next power of two

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best_vec = a.copy()
    # initial score (sum = 1, so denominator = 1)
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    best_sc = float(2.0 * n * conv_full.max())

    # optimisation parameters
    step0 = 3.0            # base stepsize (will be divided by sqrt(it))
    beta = 0.9             # momentum factor
    momentum = np.zeros_like(a)
    sym_interval = 120     # symmetrise every this many iterations
    no_improve = 0
    it = 0
    target = 1.50274349

    # ------------------------------------------------------------------
    # Optimisation loop (matrix‑free, O(n log n) per iteration)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # current convolution & max
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        cur_max = conv.max()
        cur_sc = 2.0 * n * cur_max

        # keep best seen so far
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # early exit if target reached
        if best_sc < target:
            break

        # gradient of the maximal autocorrelation entry
        g = _grad_max_conv(a, conv, cur_max)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # back‑tracking line search
        base_alpha = step0 / math.sqrt(it)
        alpha = base_alpha
        cand = None
        cand_sc = None
        for _ in range(5):                     # up to 5 attempts
            a_tmp = a - alpha * momentum
            a_tmp = _project_simplex(a_tmp)     # stay on simplex
            a_tmp_fft = np.fft.rfft(a_tmp, n=L)
            conv_tmp = np.fft.irfft(a_tmp_fft * a_tmp_fft, n=L)[: 2 * n - 1]
            cand_sc = 2.0 * n * conv_tmp.max()
            if cand_sc <= cur_sc:
                cand, cand_sc = a_tmp, cand_sc
                break
            alpha *= 0.5

        if cand is None:   # no descent found – accept the last trial
            cand, cand_sc = a_tmp, cand_sc

        # accept / possibly occasional uphill move
        if cand_sc < cur_sc:
            a = cand
            cur_sc = cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                a = cand
                cur_sc = cand_sc

        # periodic symmetrisation (helps due to Toeplitz symmetry)
        if it % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            a_sym_fft = np.fft.rfft(a_sym, n=L)
            conv_sym = np.fft.irfft(a_sym_fft * a_sym_fft, n=L)[: 2 * n - 1]
            sym_sc = 2.0 * n * conv_sym.max()
            if sym_sc < cur_sc:
                a = a_sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, a_sym.copy()
                    no_improve = 0

        # random mass transfer when stuck for a long time
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.4
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
                no_improve = 0

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
