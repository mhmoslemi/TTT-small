import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex of sum z."""
    n = v.size
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence whose AC1 bound is as low as
    possible.  Uses a soft‑max approximation of the max‑convolution and an
    Adam‑style update with simplex projection.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # Initialise
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # modest length if no parent is supplied
        a = rng.random(2048, dtype=np.float64)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    # make symmetric initially (helps with Toeplitz structure)
    a = (a + a[::-1]) * 0.5
    a = _project_simplex(a)               # enforce sum = 1
    n = a.size
    # FFT length (next power of two >= 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------
    # Optimisation state
    # ------------------------------------------------------------
    # Adam parameters
    lr0 = 0.06
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = beta1
    beta2_pow = beta2

    # temperature schedule for soft‑max
    T0 = 0.05
    Tmin = 0.001

    # bookkeeping
    best_a = a.copy()
    conv_full = np.fft.irfft(np.fft.rfft(a, n=L) * np.fft.rfft(a, n=L), n=L)[: 2 * n - 1]
    best_sc = 2.0 * n * conv_full.max()
    no_improve = 0
    target = 1.50274349

    # ------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------
    it = 0
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and current score -----
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        cur_max = conv.max()
        cur_sc = 2.0 * n * cur_max

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            no_improve = 0
            if best_sc < target:
                break
        else:
            no_improve += 1

        # ----- soft‑max weights (smooth max) -----
        T = max(Tmin, T0 * (0.999 ** it))
        shift = cur_max
        exps = np.exp((conv - shift) / T)
        p = exps / exps.sum()                     # length 2n‑1, sums to 1

        # ----- gradient of smooth max (via correlation) -----
        # gradient = 2 * correlate(p, a, mode='valid')
        grad = 2.0 * np.correlate(p, a, mode='valid')

        # ----- Adam update (with decaying step) -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # decaying learning‑rate factor
        lr = lr0 / math.sqrt(it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_simplex(a_new)

        a = a_new

        # ----- update the bias‑correction powers -----
        beta1_pow *= beta1
        beta2_pow *= beta2

        # ----- periodic symmetrisation -----
        if it % 250 == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            a_sym_fft = np.fft.rfft(a_sym, n=L)
            conv_sym = np.fft.irfft(a_sym_fft * a_sym_fft, n=L)[: 2 * n - 1]
            sc_sym = 2.0 * n * conv_sym.max()
            if sc_sym < cur_sc:
                a = a_sym
                cur_sc = sc_sym
                if sc_sym < best_sc:
                    best_sc = sc_sym
                    best_a = a_sym.copy()
                    no_improve = 0

        # ----- random jitter when stuck -----
        if no_improve > 4000:
            # transfer a random chunk of mass
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.4
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a)       # re‑normalise
            no_improve = 0

    # ------------------------------------------------------------------
    # Return the best candidate found
    # ------------------------------------------------------------------
    return [float(x) for x in best_a]
