        delta = best_vec[i] * rng.random() * 0.2
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        cand = best_vec.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        sc = _score(cand)
        if sc < best_sc:
            best_sc = sc
            best_vec = cand

    # return as plain list of Python floats
    return [float(x) for x in best_vec]
```
