def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale vector to have sum = 1 (non‑negative entries assumed)."""
    s = v.sum()
    if s == 0.0:
        # fallback uniform
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _self_conv_max(v: np.ndarray, L: int):
    """Return (max convolution, FFT of v, full convolution) for v*v."""
    v_fft = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(v_fft * v_fft, n=L)[: 2 * v.size - 1]
    return conv.max(), v_fft, conv


def _softmax_weights(conv: np.ndarray, cur_max: float, eps: float) -> np.ndarray:
    """
    Softmax weights for smoothing the max.
    We compute exp((conv‑cur_max)/eps) which is never >1, then normalise.
    """
    ex = (conv - cur_max) / eps
    w = np.exp(ex)                     # values in (0,1]
    s = w.sum()
    if s == 0.0:                       # all under‑flowed -> uniform fallback
        w.fill(1.0 / w.size)
    else:
        w /= s
    return w


def _gradient(v_fft: np.ndarray, w: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of the smoothed max‐conv objective.
    g_i = 2 * (v * w_rev)_((2n‑2)‑i) .
    """
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(v_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)   # slice that yields length‑n vector
    return 2.0 * grad_full[idx]


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce mirror symmetry and renormalise to sum = 1."""
    s = (v + v[::-1]) * 0.5
    return _normalize(s)

    cur = _normalize(cur)
    n = cur.size

    # FFT length (next power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 2).bit_length())

    # --------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------
    eps0 = 1.0               # initial soft‑max temperature
    eps_decay = 0.9995       # exponential decay per iteration
    eps_min = 1e-8
    step0 = 0.7              # base step size for exponentiated update
    beta = 0.8               # momentum factor
    momentum = np.zeros_like(cur)
    no_improve = 0
    best_vec = cur.copy()
    best_max, _, _ = _self_conv_max(cur, L)
    best_score = 2.0 * n * best_max
    target_score = 1.50274349
    it = 0

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and max -----
        cur_max, cur_fft, conv = _self_conv_max(cur, L)

        # ----- soft‑max weights -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = _softmax_weights(conv, cur_max, eps)

        # ----- gradient (matrix‑free via FFT) -----
        grad = _gradient(cur_fft, w, n, L)

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- candidate generation (exponentiated gradient) -----
        step = step0 / math.sqrt(it)
        exponent = -step * momentum
        exponent = np.clip(exponent, -50.0, 50.0)   # avoid overflow
        cand = cur * np.exp(exponent)
        cand = _normalize(cand)

        cand_max, _, _ = _self_conv_max(cand, L)

        # ----- acceptance test -----
        if cand_max < cur_max - 1e-15:
            cur, cur_max = cand, cand_max
            no_improve = 0
            if cand_max < best_max:
                best_max = cand_max
                best_vec = cand.copy()
                best_score = 2.0 * n * best_max
        else:
            # occasional uphill move (Metropolis style)
            no_improve += 1
            prob = math.exp(-(cand_max - cur_max) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_max = cand, cand_max

        # ----- periodic symmetrisation -----
        if it % 250 == 0:
            sym = _symmetrize(cur)
            sym_max, _, _ = _self_conv_max(sym, L)
            if sym_max <= cur_max:
                cur, cur_max = sym, sym_max
                if sym_max < best_max:
                    best_max = sym_max
                    best_vec = sym.copy()
                    best_score = 2.0 * n * best_max
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 5000:
            # small transfer between two random indices
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3   # move up to 30 % of mass
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _normalize(tmp)
                tmp_max, _, _ = _self_conv_max(tmp, L)
                if tmp_max < cur_max:
                    cur, cur_max = tmp, tmp_max
                    if tmp_max < best_max:
                        best_max = tmp_max
                        best_vec = tmp.copy()
                        best_score = 2.0 * n * best_max
                    no_improve = 0
                else:
                    # if no improvement, try a total random restart after a while
                    if no_improve > 20000:
                        cur = rng.random(n, dtype=np.float64)
                        cur = _normalize(cur)
                        momentum.fill(0.0)
                        no_improve = 0

        # ----- early exit if target reached -----
        if best_score < target_score:
            break

    # --------------------------------------------------------
    # Return the best sequence found (plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
