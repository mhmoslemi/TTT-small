import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection of a vector onto the probability simplex
# (non‑negative entries that sum to 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound for a vector with sum = 1
# Returns 2 * n * max(conv)  (the evaluator multiplies this by 1)
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int, n: int) -> float:
    """Compute 2 * n * max(convolution(a, a)).  Assumes sum(a)=1."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence with a low AC1 bound.
    The returned list is a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin for the final conversion
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise – use the parent if present, otherwise a random start
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # a modest default length; the exact length does not matter
        a0 = rng.random(1024, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)          # enforce sum = 1
    cur = a0.copy()
    n = cur.size

    # ------------------------------------------------------------
    # FFT length – must be power of two ≥ 3*n‑2
    # ------------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------
    # ADAM state for gradient‑based optimisation
    # ------------------------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.5

    # ------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------
    best_vec = cur.copy()
    best_sc = _score(best_vec, L, n)
    cur_sc = best_sc
    it = 0
    no_improve = 0

    # smoothing schedule for the soft‑max approximation of max
    beta_s_init = 1.0
    beta_s_final = 30.0
    # target bound (stop early if achieved)
    target = 1.50274349

    # ------------------------------------------------------------
    # Optimisation loop – matrix‑free, O(n log n) per iteration
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of current vector with itself -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        conv_max = conv.max()

        # ----- soft‑max weights (smooth approximation of the max) -----
        beta_s = beta_s_init + (beta_s_final - beta_s_init) * min(1.0, it / 20000.0)
        # shift for numerical stability
        shifted = conv - conv_max
        exps = np.exp(beta_s * shifted)
        w = exps / exps.sum()

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- ADAM update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = base_lr / math.sqrt(it)            # simple decay
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        cand = _project_simplex(cand)           # keep feasibility
        cand_sc = _score(cand, L, n)

        # ----- accept if improvement, otherwise occasional uphill move -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:

            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

            sym = _project_simplex(sym)
            sym_sc = _score(sym, L, n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # ----- early exit if target achieved -----
        if best_sc < target:
            break

    # ------------------------------------------------------------
    # Return the best candidate found (as a plain Python list)
    # ------------------------------------------------------------
    return [float(x) for x in best_vec]
