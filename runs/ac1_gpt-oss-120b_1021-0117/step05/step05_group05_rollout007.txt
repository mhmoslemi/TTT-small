import time
import math
import numpy as np


def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _score(a):
    """Score = 2 * n * max(conv(a,a)) for a that already sums to 1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _local_optimize(init_vec, n, L, deadline, rng, target_score, max_iter=20000):
    """Gradient‑based projected sub‑gradient descent with occasional tricks."""
    cur = init_vec.copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    momentum = np.zeros_like(cur)
    beta = 0.9
    step0 = 0.8
    it = 0
    no_improve = 0

    while it < max_iter and time.monotonic() < deadline:
        it += 1

        # convolution and max
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # weights for sub‑gradient (uniform over near‑max entries)
        tol = max_val * 1e-12 + 1e-14
        w = (conv >= max_val - tol).astype(np.float64)
        ws = w.sum()
        if ws > 1:
            w /= ws

        # sub‑gradient of max‑convolution
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # step size schedule
        step = step0 / math.sqrt(it)

        # try a projected step (backtracking line‑search)
        cand = None
        cand_sc = None
        step_tmp = step
        for _ in range(8):
            cand_tmp = cur - step_tmp * momentum
            cand_tmp = _project_simplex(cand_tmp)
            cand_tmp_sc = _score(cand_tmp)
            if cand_tmp_sc <= cur_sc:
                cand, cand_sc = cand_tmp, cand_tmp_sc
                break
            step_tmp *= 0.5
        if cand is None:
            cand, cand_sc = cand_tmp, cand_tmp_sc

        # accept/reject
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # periodic symmetrisation
        if it % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # occasional random mass transfer
        if it % 100 == 0:
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.2
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_vec, best_sc = tmp.copy(), tmp_sc
                    no_improve = 0

        # occasional smoothing
        if it % 500 == 0:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smooth = np.convolve(cur, kernel, mode='same')
            smooth = _project_simplex(smooth)
            smooth_sc = _score(smooth)
            if smooth_sc < cur_sc:
                cur, cur_sc = smooth, smooth_sc
                if smooth_sc < best_sc:
                    best_vec, best_sc = smooth.copy(), smooth_sc
                no_improve = 0

        if no_improve > 4000:
            break
        if best_sc <= target_score:
            break

    return best_vec, best_sc
