import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ------------------------------------------------------------
def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale non‑negative vector to have sum = 1 (or uniform if zero)."""
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_conv_max(v: np.ndarray, L: int):
    """Return (max convolution, FFT of v, full convolution) for linear conv."""
    a_fft = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * v.size - 1]
    return conv.max(), a_fft, conv


def _softmax_weights(conv: np.ndarray, cur_max: float, eps: float) -> np.ndarray:
    """Soft‑max weights w_i ∝ exp((conv_i - cur_max)/eps)."""
    w = np.exp((conv - cur_max) / eps)
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size, dtype=np.float64)
    return w / s


def _gradient(a_fft: np.ndarray, w: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of smoothed max‑conv w.r.t. the vector.
    g_i = 2 * (a * w_rev)_{ (2n‑2)‑i } .
    """
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce even symmetry v[i] = v[-i‑1]."""
    return (v + v[::-1]) * 0.5

    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # -------------------- initial vector --------------------

    cur = _normalize(cur)

    L = _next_pow2(3 * n - 2)

    # -------------------- algorithmic parameters --------------------
    eps0 = 0.5               # initial soft‑max temperature
