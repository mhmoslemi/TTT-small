import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score(a):
    """Exact AC1 upper‑bound for a vector with sum ≈ 1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((3 * n - 2).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

    while time.monotonic() < deadline:
        it += 1

        # ------- convolution and current score ---------------------------------
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_c = conv.max()
        cur_sc = 2.0 * n * max_c          # same as _score(cur)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = cur.copy()
        if best_sc < target:
            break

        # ------- smooth weighting of the max (softmax) ------------------------
        conv_range = max_c - conv.min()
        if conv_range <= 0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = np.exp(soft_beta * (conv - max_c) / (conv_range + 1e-12))
            s = w.sum()
            w = w / s if s > 0 else np.full_like(conv, 1.0 / conv.size)

        # ------- gradient of the smooth surrogate -----------------------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ------- Adam update ---------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        # ------- acceptance ----------------------------------------------------
        if cand_sc <= cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
        else:
            # simple simulated‑annealing style acceptance
            temp = max(0.1, 1.0 / math.sqrt(it))
            if rng.random() < math.exp((cur_sc - cand_sc) / temp):
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ------- occasional local random perturbation -------------------------
        if it % 300 == 0:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n)
                if j != i:
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    tmp_sc = _score(tmp)
                    if tmp_sc < cur_sc:
                        cur, cur_sc = tmp, tmp_sc
                    if tmp_sc < best_sc:
                        best_sc, best_vec = tmp_sc, tmp.copy()
                    no_improve = 0

        # ------- restart when stuck -------------------------------------------
        if no_improve > 5000 and time.monotonic() + 5.0 < deadline:
            pert = best_vec.copy()
            pert += rng.normal(scale=0.01, size=n)
            pert = np.maximum(pert, 0.0)
            pert = _project_simplex(pert)
            cur, cur_sc = pert, _score(pert)
            no_improve = 0

    # ----------------------------------------------------------------------
    final = np.maximum(best_vec, 0.0)
    final = np.minimum(final, 1000.0)   # evaluator clips anyway
    return [float(x) for x in final]
