import time
import math
import numpy as np

# ------------------------------------------------------------
# Basic utilities (no closures)
# ------------------------------------------------------------
def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale to non‑negative simplex (sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


def _conv_and_fft(a: np.ndarray, L: int):
    """Return full linear convolution a*a (length 2·len(a)-1) and its FFT."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv, a_fft


def _subgradient(a: np.ndarray, conv: np.ndarray, max_val: float):
    """
    Sub‑gradient of the (non‑smoothed) max‑convolution.
    For each shift attaining the maximum add the reversed slice.
    """
    n = a.size
    # indices where conv is (almost) maximal
    tol = max_val * 1e-12 + 1e-12
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i = np.arange(start_i, end_i + 1)
        j = k - i
        # vectorised accumulation
        np.add.at(grad, i, a[j])
    grad *= 2.0
    return grad

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty list of non‑negative coefficients with low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ---- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2048, dtype=np.float64)
    cur = _normalize(cur)
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length ≥ 2n‑1

    # ---- optimisation state -----------------------------------------
    momentum = np.zeros_like(cur)
    beta = 0.9
    step0 = 0.4          # initial step for exponentiated update
    no_improve = 0
    best_vec = cur.copy()

    conv, a_fft = _conv_and_fft(cur, L)
    cur_max = conv.max()
    best_max = cur_max
    best_score = 2.0 * n * best_max
    target_score = 1.50274349
    it = 0

    # ---- main loop --------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # recompute convolution (FFT already available from previous step)
        conv, a_fft = _conv_and_fft(cur, L)
        cur_max = conv.max()

        # sub‑gradient of the (soft) max
        grad = _subgradient(cur, conv, cur_max)

        # momentum (mirror of Adam’s first moment)
        momentum = beta * momentum + (1.0 - beta) * grad

        # adaptive step size (decays like 1/√it)
        step = step0 / math.sqrt(it)

        # ---- candidate generation (exponentiated update) -------------
        cand = None
        cand_max = None
        for _ in range(8):          # simple back‑tracking
            exponent = -step * momentum
            exponent = np.clip(exponent, -50.0, 50.0)   # avoid overflow
            cand_tmp = cur * np.exp(exponent)
            cand_tmp = _normalize(cand_tmp)

            conv_tmp, _ = _conv_and_fft(cand_tmp, L)
            cand_tmp_max = conv_tmp.max()

            if cand_tmp_max <= cur_max + 1e-15:
                cand, cand_max = cand_tmp, cand_tmp_max
                break
            step *= 0.5

        if cand is None:            # no descent found – keep last trial
            cand, cand_max = cand_tmp, cand_tmp_max

        # ---- acceptance ------------------------------------------------
        if cand_max < cur_max - 1e-15:
            cur = cand
            cur_max = cand_max
            no_improve = 0
            if cand_max < best_max:
                best_max = cand_max
                best_vec = cand.copy()
                best_score = 2.0 * n * best_max
        else:
            no_improve += 1
            # occasional uphill move (Metropolis style) to escape plateaus
            prob = math.exp(-(cand_max - cur_max) / (step + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_max = cand_max

        # ---- random mass‑transfer when stuck ---------------------------
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta

                        best_vec = tmp.copy()
                        best_score = 2.0 * n * best_max
                no_improve = 0

        # ---- early exit if target achieved -----------------------------
        if best_score < target_score:
            break

    return [float(x) for x in best_vec]
