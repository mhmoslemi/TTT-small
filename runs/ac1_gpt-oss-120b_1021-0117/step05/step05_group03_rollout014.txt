    # candidate
    cand = cur - alpha * momentum
    cand = _project_simplex(cand)
    # compute score
    cand_score = _score(cand, L)

    # acceptance
    if cand_score < cur_score:
        cur, cur_score = cand, cand_score
        no_improve = 0
        if cand_score < best_score:

        if rng.random() < prob:

        no_improve += 1

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast objective for sum(a)=1: 2 * n * max(conv(a,a))
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty list of non‑negative coefficients whose AC1 upper bound
    is (hopefully) below the target.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # keep safety margin

    # ------------------------------------------------------------------
    # Initialise – use the provided parent if available, otherwise random.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(1024, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)                # enforce sum = 1
    n = cur.size

    # ------------------------------------------------------------
    # FFT length – must accommodate convolution of a (len n) with a weight
    # vector of length 2n‑1 (the full convolution needed for gradient).
    # ------------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------
    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    # algorithmic parameters
    step0 = 1.0                # base step magnitude
    mom_factor = 0.90          # momentum coefficient
    momentum = np.zeros_like(cur)
    no_improve = 0
    it = 0
    target = 1.50274349
    sym_interval = 400
    stuck_thresh = 3500

    # soft‑max (smoothed max) schedule
    beta_init = 0.20
    beta_inc = 0.0005
    beta_max = 6.0

    # line‑search scale factors
    factors = [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]

    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- soft‑max weighting of convolution entries -----
        beta = min(beta_max, beta_init + beta_inc * it)
        w = np.exp(beta * (conv - max_val))   # stable softmax
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- momentum update -----
        momentum = mom_factor * momentum + (1.0 - mom_factor) * g

        # ----- step size (inverse‑sqrt decay) -----
        base_alpha = step0 / math.sqrt(it)

        # ----- line‑search over a few scales -----
        cand = None
        cand_sc = None
        for factor in factors:
            alpha = base_alpha * factor
            tmp = cur - alpha * momentum
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp, L)

            if cand_sc is None or tmp_sc < cand_sc:
                cand, cand_sc = tmp, tmp_sc

            if tmp_sc < cur_sc:
                break

        # ----- accept / reject (with occasional uphill move) -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (base_alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- periodic symmetrisation (helps with Toeplitz structure) -----
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > stuck_thresh:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- early exit if the target is reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
