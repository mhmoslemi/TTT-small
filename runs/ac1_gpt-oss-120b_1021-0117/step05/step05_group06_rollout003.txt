import time

# ----------------------------------------------------------------------
# Simplex projection (onto non‑negative vectors with sum = 1)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Maximum value of full linear convolution a*a using zero‑padded FFT
# ----------------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    """Return max of a*a (full convolution) using FFT length L."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv_full.max())


# ----------------------------------------------------------------------
# Reverse a contiguous block of length `length` starting at `start`
# (wrap‑around allowed) in‑place.
# ----------------------------------------------------------------------
def _reverse_block(arr: np.ndarray, start: int, length: int) -> None:
    n = arr.size
    if length <= 1 or length >= n:
        return
    end = (start + length) % n
    if start < end:
        arr[start:end] = arr[start:end][::-1]
    else:
        # Wrapped block: concatenate, reverse, write back
        block = np.concatenate((arr[start:], arr[:end]))
        block = block[::-1]
        arr[start:] = block[: n - start]
        arr[:end] = block[n - start :]

    n = vals.size
    out = np.empty_like(vals)
    l, r = 0, n - 1
    i = 0
    while l <= r:
        out[l] = vals[i]; i += 1
        if l < r:
            out[r] = vals[i]; i += 1
        l += 1
        r -= 1
    return out

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------------
    # Initial vector (use parent if present)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2048, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)          # enforce sum = 1, non‑negative
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (next power of two)

    # --------------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------------
    cur = a.copy()
    cur_sc = _max_conv(cur, L)
    best = cur.copy()
    best_sc = cur_sc
    best_bound = 2.0 * n * best_sc
    target_bound = 1.50274349

    # --------------------------------------------------------------
    # Adam‑style optimisation (gradient of smoothed max)
    # --------------------------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.12                     # base learning rate
    temp_start = 0.02                  # soft‑max temperature start
    temp_decay = 0.9995                # geometric decay per iteration
    it = 0

    while time.monotonic() < deadline - 5.0:
        it += 1
        # current convolution
        cur_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(cur_fft * cur_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # soft‑max weights
        temp = max(1e-9, temp_start * (temp_decay ** it))
        w = np.exp((conv - max_val) / temp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective:
        # g = 2 * (a * w_rev)   (where * denotes linear convolution)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(cur_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = base_lr / (np.sqrt(v_hat) + eps)
        cand = cur - step * m_hat
        cand = _project_simplex(cand)

        cand_sc = _max_conv(cand, L)
        if cand_sc <= cur_sc:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best = cand.copy()
                best_sc = cand_sc
                best_bound = 2.0 * n * best_sc
                if best_bound < target_bound:
                    break

        # occasional symmetrisation (helps with Toeplitz structure)
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _max_conv(sym, L)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best = sym.copy()
                    best_sc = sym_sc
                    best_bound = 2.0 * n * best_sc
                    if best_bound < target_bound:
                        break

    # --------------------------------------------------------------
    # Deterministic anti‑identity permutation (large opposite small)
    # --------------------------------------------------------------
    perm = _anti_identity(best)
    perm_sc = _max_conv(perm, L)
    perm_bound = 2.0 * n * perm_sc
    if perm_bound < best_bound:
        best = perm.copy()
        best_sc = perm_sc
        best_bound = perm_bound

    # --------------------------------------------------------------
    # Local hill‑climbing (swap / block‑reverse / small perturb)
    # --------------------------------------------------------------
    cur = best.copy()
    cur_sc = best_sc
    while time.monotonic() < deadline - 0.2:
        op = rng.integers(0, 3)          # 0: swap, 1: reverse block, 2: perturb
        if op == 0:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cur[i], cur[j] = cur[j], cur[i]
            new_sc = _max_conv(cur, L)
            if new_sc < cur_sc:
                cur_sc = new_sc
                if new_sc < best_sc:
                    best = cur.copy()
                    best_sc = new_sc
                    best_bound = 2.0 * n * best_sc
                    if best_bound < target_bound:
                        break
            else:
                cur[i], cur[j] = cur[j], cur[i]   # revert
        elif op == 1:
            start_idx = rng.integers(0, n)
            max_len = min(80, n // 2)
            length = rng.integers(2, max_len + 1)
            _reverse_block(cur, start_idx, length)
            new_sc = _max_conv(cur, L)
            if new_sc < cur_sc:
                cur_sc = new_sc
                if new_sc < best_sc:
                    best = cur.copy()
                    best_sc = new_sc
                    best_bound = 2.0 * n * best_sc
                    if best_bound < target_bound:
                        break
            else:
                _reverse_block(cur, start_idx, length)   # revert
        else:
            # small Gaussian perturbation followed by projection
            cand = cur + rng.normal(scale=0.01, size=n)
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)
            cand_sc = _max_conv(cand, L)
            if cand_sc < cur_sc:
                cur = cand
                cur_sc = cand_sc
                if cand_sc < best_sc:
                    best = cand.copy()
                    best_sc = cand_sc
                    best_bound = 2.0 * n * best_sc
                    if best_bound < target_bound:
                        break

    # ------------------------------------------------------------------
    # Return the best candidate found as a plain Python list
    # ------------------------------------------------------------------
    if best.size == 0:                     # safety fallback
        best = np.array([1.0])
    return [float(x) for x in best]
