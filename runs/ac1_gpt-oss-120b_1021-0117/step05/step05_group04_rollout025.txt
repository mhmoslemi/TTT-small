def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum=1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive after shift -> uniform distribution
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safety: renormalise to exact sum 1
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    return w / w_sum


def _self_conv_max_fft(a: np.ndarray, L: int):
    """
    Compute self‑convolution a*a via FFT, return (conv, max(conv), rfft(a)).
    conv is a length‑(2n‑1) real array (full linear convolution).
    """
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv_full, float(conv_full.max()), a_fft


def _softmax_weights(conv: np.ndarray, max_val: float, eps: float) -> np.ndarray:
    """Soft‑max weights over convolution entries with temperature eps."""
    # clip exponent to avoid overflow/underflow
    exps = np.exp(np.clip((conv - max_val) / eps, -50.0, 50.0))
    s = exps.sum()
    if s == 0.0:
        return np.full_like(exps, 1.0 / exps.size, dtype=np.float64)
    return exps / s

    Returns a length‑n vector.

    idx = (2 * n - 2) - np.arange(n)  # map to length‑n gradient slice
    return 2.0 * grad_full[idx]

        else:
            no_improve += 1
            # occasional stochastic escape when stuck
            if no_improve > 3000:
                i = rng.integers(0, n)
                if cur[i] > 0.0:
                    delta = cur[i] * rng.random() * 0.4
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _project_simplex(tmp)
                    _, tmp_max, _ = _self_conv_max_fft(tmp, L)
                    if tmp_max < cur_max:
                        cur = tmp
                        cur_max = tmp_max
                        no_improve = 0
                        if tmp_max < best_max:
                            best_max = tmp_max
                            best_vec = tmp.copy()
                            best_score = 2.0 * n * best_max

        # ---- periodic symmetrisation (helps due to convolution symmetry) ----
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            _, sym_max, _ = _self_conv_max_fft(sym, L)
            if sym_max < cur_max:
                cur = sym
                cur_max = sym_max
                if sym_max < best_max:
                    best_max = sym_max
                    best_vec = sym.copy()
                    best_score = 2.0 * n * best_max
                no_improve = 0

        # ---- early exit if target already reached ----
        if best_score < target_score:
            break

    # --------------------------------------------------------
    # Return the best sequence discovered (plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
