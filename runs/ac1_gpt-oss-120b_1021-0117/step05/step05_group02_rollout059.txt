def _conv_and_grad(a, a_fft=None, eps=1.0):
    # compute conv via FFT
    # a_fft provided maybe to avoid recompute? But we need a_fft anyway.
    n = a.shape[0]
    L = 1 << ((3*n - 2).bit_length())
    if a_fft is None:
        a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[:2*n-1]
    max_val = conv.max()
    # softmax weights
    w = np.exp((conv - max_val) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0/w.size)
    else:
        w /= w_sum
    # gradient
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2*n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]
    return conv, g

    g = 2.0 * grad_full[idx]
    return conv, g

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_vec(a: np.ndarray) -> float:
    """Exact score used by the evaluator (sum of a is irrelevant)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _conv_grad(a: np.ndarray, L: int, eps: float):
    """
    Returns the full linear convolution (first 2n‑1 entries) and the
    smoothed gradient of the max‑convolution objective.
    """
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    n = a.shape[0]
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    w = np.exp((conv - max_val) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= w_sum
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]
    return conv, g

    """
    Returns a non‑empty list of non‑negative coefficients that (hopefully)
    yields a low AC1 upper bound.  The routine runs for at most
    `budget_s` seconds and returns the best vector discovered.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # -----------------------------------------------------------------
    # Initialise – use the parent sequence if present, otherwise uniform.
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = np.full(1024, 1.0 / 1024, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # enforce sum = 1
    n = cur.size

    # ------------------------------------------------------------
    # Pre‑compute FFT length for gradient (needs >= 3n‑2).
    # ------------------------------------------------------------
    L_grad = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------
    best_vec = cur.copy()
    best_sc = _score_vec(best_vec)
    cur_sc = best_sc

    # ------------------------------------------------------------
    # Hyper‑parameters (tuned empirically)
    # ------------------------------------------------------------
    eps0 = 1.0
    eps_decay = 0.9994
    eps_min = 1e-12
    step0 = 1.2
    step0_mirror = 0.7
    beta = 0.98
    momentum = np.zeros_like(cur)
    no_improve = 0
    it = 0
    target = 1.50274349
    T0 = 0.08
    T_decay = 0.9995

    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # ------------------------------------------------------------
    while time.monotonic() < deadline:

        # temperature for occasional uphill moves
        T = max(1e-12, T0 * (T_decay ** it))

        conv, g = _conv_grad(cur, L_grad, eps)

        alpha = step0 / math.sqrt(it)

        cand1 = _project_simplex(cand1)

                tmp[i] -= delta

                no_improve = 0

        # ---- early exit if target is reached ------------------------------------
        if best_sc < target:
            break

    # ------------------------------------------------------------------------
    # Return best vector found (as plain Python list)
    # ------------------------------------------------------------------------
    return [float(x) for x in best_vec]
