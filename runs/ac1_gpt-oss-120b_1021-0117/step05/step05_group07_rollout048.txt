# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    # shift by min to ensure non‑negativity before projection
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    s = v.sum()
    if s <= 0.0:
        return float('inf')
    return float(2.0 * n * max_val / (s * s))

    # hyper‑parameters (tuned empirically)
    step0 = 0.45
    beta = 0.88
    eps0 = 0.6
    eps_decay = 0.998
    eps_min = 1e-10
    no_improve = 0
    target = 1.50274349
    it = 0

    # --------------------------------------------------------
    # Optimisation loop – matrix‑free, O(n log n) per iteration
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution & soft‑max weights -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the (soft) max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]                     # size n

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- back‑tracking line search -----
        base_alpha = step0 / math.sqrt(it)
        alpha = base_alpha
        cand = None
        cand_sc = None
        for _ in range(12):
            cand_tmp = cur - alpha * momentum
            cand_tmp = _project_simplex(cand_tmp)
            cand_sc = _score(cand_tmp, L, n)
            if cand_sc <= cur_sc:
                cand = cand_tmp
                break
            alpha *= 0.5
        if cand is None:               # no improvement found
            cand = cand_tmp

        # ----- accept / reject (Metropolis) -----
        if cand_sc < cur_sc - 1e-12:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- periodic symmetrisation (helps with Toeplitz) -----
        if it % 150 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L, n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L, n)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                no_improve = 0

        # ----- occasional soft restart from best with jitter -----
        if no_improve > 8000:
            cur = best_vec + rng.normal(scale=0.01, size=n)
            cur = np.clip(cur, 0.0, None)
            cur = _project_simplex(cur)
            cur_sc = _score(cur, L, n)
            momentum.fill(0.0)
            no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # Return the best candidate found (as plain Python list)
    return [float(x) for x in best_vec]
