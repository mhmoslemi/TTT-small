def _grad_maxsubgrad(a):
    n = a.size
    # compute convolution via FFT (since O(n log n)).
    L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2*n-1]
    max_val = conv.max()
    # mask for values within tolerance (like relative)
    # using absolute tolerance maybe 1e-12
    mask = (conv >= max_val - 1e-12)  # boolean array
    # compute gradient
    g = np.zeros_like(a)
    # iterate over mask indices where True
    ks = np.flatnonzero(mask)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # slice indices
        i = np.arange(i_start, i_end + 1, dtype=int)
        g[i] += 2.0 * a[k - i]
    return g, max_val

import time, math, numpy as np

# ----------------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Compute convolution max and its exact sub‑gradient (w.r.t. a)
# ----------------------------------------------------------------------
def _conv_max_and_grad(a: np.ndarray, L: int, eps: float = 1e-12):
    """Return (max_convolution, gradient) for a."""
    n = a.size
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_val = conv.max()
    # mask of entries attaining the max (allow tiny tolerance)
    mask = conv >= max_val - eps
    ks = np.flatnonzero(mask)
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        g[i] += 2.0 * a[k - i]
    return max_val, g

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------
    # Initial vector (use parent if present)
    # --------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # modest random start if no parent is supplied
        a = rng.random(2000, dtype=np.float64)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                     # enforce sum = 1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (next power of two)

    # --------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------
    max_val, _ = _conv_max_and_grad(a, L)
    best_sc = 2.0 * n * max_val
    best_vec = a.copy()

    # optimisation parameters
    step0 = 0.7
    beta = 0.9
    momentum = np.zeros_like(a)
    sym_interval = 250
    no_improve = 0
    it = 0
    target = 1.50274349

    # --------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # --------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # current objective and gradient
        cur_max, grad = _conv_max_and_grad(a, L)
        cur_sc = 2.0 * n * cur_max

        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if best_sc < target:
            break

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # back‑tracking line search
        base_alpha = step0 / math.sqrt(it)
        alpha = base_alpha
        cand = None
        cand_sc = None
        for _ in range(6):
            a_tmp = a - alpha * momentum
            a_tmp = _project_simplex(a_tmp)
            tmp_max, _ = _conv_max_and_grad(a_tmp, L)
            cand_sc = 2.0 * n * tmp_max
            if cand_sc <= cur_sc:
