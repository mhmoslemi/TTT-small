import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative sequence of heights with a low AC1 bound.
    The routine uses the supplied parent (height_sequence_1) if available,
    otherwise falls back to a uniform distribution and performs a tiny
    random local search while respecting the time budget.
    """
    # ----------------------------------------------------------------------
    # 1.  Basic timing
    # ----------------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----------------------------------------------------------------------
    # 2.  Initialise from parent or uniform fallback
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # a modest default length; the exact size is unimportant
        cur = np.full(1024, 1.0 / 1024, dtype=np.float64)

    # enforce the evaluator's hard clipping
    cur = np.clip(cur, 0.0, 1000.0)

    # If the sum is accidentally too small (should not happen), use uniform.
    if cur.sum() < 0.01:
        cur = np.full_like(cur, 1.0 / cur.size, dtype=np.float64)

    # ----------------------------------------------------------------------
    # 3.  Evaluate the starting point
    # ----------------------------------------------------------------------
    best_vec = cur.copy()
    best_sc = evaluate_sequence(best_vec.tolist())
    # If the parent somehow gave an infinite score, replace with uniform.
    if not np.isfinite(best_sc):
        best_vec = np.full_like(cur, 1.0 / cur.size, dtype=np.float64)
        best_sc = evaluate_sequence(best_vec.tolist())

    cur_sc = best_sc
    n = cur.size
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # 4.  Very cheap random local search (mass‑transfer moves)
    # ----------------------------------------------------------------------
    # The search never increases the objective, but it may find tiny
    # numerical improvements caused by rounding in the evaluator.
    while time.monotonic() < deadline:
        # choose two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] <= 0.0:
            continue

        # move a random fraction (up to 10 %) of mass from i to j
        delta = cur[i] * rng.random() * 0.10
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # evaluate – the evaluator handles clipping automatically
        cand_sc = evaluate_sequence(cand.tolist())
        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_vec = cand.copy()
                best_sc = cand_sc

        # optional early exit if we already beat a reasonable target
        if best_sc < 1.6:          # well below the uniform score of 2
            break

    # ----------------------------------------------------------------------
    # 5.  Return the best candidate found
    # ----------------------------------------------------------------------
    return [float(x) for x in best_vec]
