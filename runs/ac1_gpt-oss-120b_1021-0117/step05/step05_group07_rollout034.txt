import time, math, numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex (sum == 1, non‑negative)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Main optimisation routine
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function (list of floats) with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small head‑room before the hard timeout
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise – use the parent if present, otherwise a random start
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2000, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # enforce sum == 1   (scale‑invariant)
    n = cur.size

    # ------------------------------------------------------------
    # FFT planning – length at least 2n‑1 (next power of two)
    # ------------------------------------------------------------
    L = 1 << ((2 * n - 2).bit_length())
    idx = (2 * n - 2) - np.arange(n)    # index mapping for gradient

    # ------------------------------------------------------------
    # Adam optimiser state
    # ------------------------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # ------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------
    step0 = 0.5           # base step magnitude (decays with √t)
    eps0 = 1.0            # temperature for soft‑max weighting
    eps_decay = 0.995
    eps_min = 1e-12
    target = 1.50274349   # desired upper bound
    best_sc = float("inf")
    best_vec = cur.copy()
    no_improve = 0
    it = 0

    # ------------------------------------------------------------
    # Helper: compute bound for a given vector (sum == 1)
    # ------------------------------------------------------------
    def _score(vec):
        a_fft = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        return 2.0 * n * conv.max()

    # initial score
    cur_sc = _score(cur)
    best_sc, best_vec = cur_sc, cur.copy()

    # ------------------------------------------------------------
    # Optimisation loop (matrix‑free, O(n log n) per iteration)
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution (needed for gradient) -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()
        cur_sc = 2.0 * n * max_val

        # ----- keep global best -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- soft‑max weighting of the max -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        if eps > 1e-12:
            w = np.exp((conv - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.size)
            else:
                w /= w_sum
        else:
            w = np.zeros_like(conv)
            mask = conv == max_val
            cnt = mask.sum()
            if cnt:
                w[mask] = 1.0 / cnt

        # ----- gradient of (soft) max -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        grad = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        direction = m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- step size (inverse‑sqrt decay) with back‑tracking -----
        base_alpha = step0 / math.sqrt(it)
        alpha = base_alpha
        cand = None
        cand_sc = None
        for _ in range(8):
            cand_tmp = cur - alpha * direction
            cand_tmp = _project_simplex(cand_tmp)
            cand_fft = np.fft.rfft(cand_tmp, n=L)
            cand_conv = np.fft.irfft(cand_fft * cand_fft, n=L)[: 2 * n - 1]
            cand_sc_val = 2.0 * n * cand_conv.max()
            if cand_sc_val <= cur_sc:
                cand, cand_sc = cand_tmp, cand_sc_val
                break
            alpha *= 0.5
        if cand is None:               # no improvement after back‑track
            cand, cand_sc = cand_tmp, cand_sc_val

        # ----- accept / reject -----
        if cand_sc < cur_sc - 1e-12:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # ----- periodic symmetrisation (helps with the Toeplitz structure) -----
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_fft = np.fft.rfft(sym, n=L)
            sym_conv = np.fft.irfft(sym_fft * sym_fft, n=L)[: 2 * n - 1]
            sym_sc = 2.0 * n * sym_conv.max()
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()
                no_improve = 0

        # ----- random mass‑transfer when progress stalls -----
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n)
                if j == i:
                    j = (i + 1) % n
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_fft = np.fft.rfft(tmp, n=L)
                tmp_conv = np.fft.irfft(tmp_fft * tmp_fft, n=L)[: 2 * n - 1]
                tmp_sc = 2.0 * n * tmp_conv.max()
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                if tmp_sc < best_sc:
                    best_sc, best_vec = tmp_sc, tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

        # ----- keep respecting the overall deadline -----
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
