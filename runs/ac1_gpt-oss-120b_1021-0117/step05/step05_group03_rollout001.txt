import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a simple uniform non‑negative sequence.
    
    The AC1 upper bound 2·n·max(conv)/sum² is minimized (value = 2) by any
    constant non‑negative vector, because max(conv) = sum(a_i²) and for a given
    sum the L₂ norm is minimized when all entries are equal.
    """
    start = time.monotonic()
    # leave a tiny margin for the final conversion/return
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # choose a length – reuse the parent length if it exists, otherwise a modest default
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 1024

    # uniform vector (value 1.0 stays safely below the 1000 cap)
    vec = np.full(n, 1.0, dtype=np.float64)

    # ensure we exit before the hard timeout
    if time.monotonic() > deadline:
        # fallback to a very short safe vector
        vec = np.array([1.0])

    # convert to plain Python list of floats as required
    return [float(x) for x in vec]
