import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection – Euclidean projection onto {x >= 0, sum x = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast AC1 score for a vector with sum = 1
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """2 * n * max(convolve(a, a))  (assumes sum(a)=1)."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise – use the parent if present, otherwise a random start
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # a modest default length if no parent is supplied
        cur = rng.random(1024, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # enforce sum = 1
    n = cur.shape[0]

    # ------------------------------------------------------------
    # FFT length – power of two >= 2n‑1
    # ------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------
    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    momentum = np.zeros_like(cur)

    # algorithmic parameters
    beta = 0.90                # momentum factor
    step0 = 1.0                # base step magnitude
    eps_factor = 0.6           # initial soft‑max scaling factor
    eps_decay = 0.9995
    eps_min_factor = 1e-4
    target = 1.50274349
    sym_period = 400
    no_improve = 0
    it = 0

    # ------------------------------------------------------------
    # Optimisation loop – matrix‑free, O(n log n) per iteration
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and its maximum -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- soft‑max weight vector (smooth sub‑gradient of max) -----
        epsilon = max(eps_factor * max_val, 1e-12)
        diff = conv - max_val                     # non‑positive
        w = np.exp(diff / epsilon)
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.ones_like(conv) / conv.size
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- step size (inverse‑sqrt decay) -----
        base_alpha = step0 / math.sqrt(it)

        # ----- two candidate updates: additive (projected) and exponentiated -----
        # additive update
        cand_add = cur - base_alpha * momentum
        cand_add = _project_simplex(cand_add)
        cand_add_sc = _score(cand_add, L)

        # exponentiated update (mirror descent with KL)
        exp_factor = np.exp(-base_alpha * g)
        cand_exp = cur * exp_factor
        s = cand_exp.sum()
        if s > 0.0:
            cand_exp = cand_exp / s
            cand_exp_sc = _score(cand_exp, L)
        else:
            cand_exp_sc = float('inf')

        # select the better of the two
        if cand_add_sc <= cand_exp_sc:
            cand_vec, cand_sc = cand_add, cand_add_sc
        else:
            cand_vec, cand_sc = cand_exp, cand_exp_sc

        # ----- accept / reject (with occasional uphill move) -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand_vec, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand_vec.copy(), cand_sc
        else:
            # Metropolis‑style uphill acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (base_alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand_vec, cand_sc
            no_improve += 1

        # ----- periodic symmetrisation (helps with the Toeplitz structure) -----
        if it % sym_period == 0:

            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 5000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- occasional restart from best with small noise -----
        if no_improve > 12000:
            cur = best_vec + 0.01 * rng.standard_normal(n)
            cur = np.maximum(cur, 0.0)
            cur = _project_simplex(cur)
            cur_sc = _score(cur, L)
            momentum = np.zeros_like(cur)
            no_improve = 0

        # ----- decay the soft‑max scaling factor -----
        eps_factor = max(eps_factor * eps_decay, eps_min_factor)

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
