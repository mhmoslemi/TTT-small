import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x) = z}."""
    n = v.size
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Compute max of the (linear) convolution a*a (full length) via FFT
# ------------------------------------------------------------
def _conv_max_fft(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return float(conv_full[: 2 * a.size - 1].max())

# ------------------------------------------------------------
# Gradient of the (exact) max‑convolution entry (or all tying entries)
# ------------------------------------------------------------
def _grad_max_entry(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the max entry of a*a (average over all ties)."""
    n = a.size
    max_val = conv.max()
    idxs = np.flatnonzero(conv == max_val)
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        g[i] += 2.0 * a[k - i]
    if idxs.size:
        g /= float(idxs.size)
    return g

# ------------------------------------------------------------
# Optional extra gradient using a few largest entries (helps smoothing)
# ------------------------------------------------------------
def _grad_topk(a: np.ndarray, conv: np.ndarray, top_k: int) -> np.ndarray:
    n = a.size
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        g[i] += 2.0 * a[k - i]
    g /= max(1, top_k)
    return g

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence with a low AC1 bound.
    The algorithm is a matrix‑free projected (Nesterov‑style) gradient descent
    that focuses on the exact maximal autocorrelation entry.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # Initial vector – use the provided parent if available,
    # otherwise a random start.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = rng.random(2048, dtype=np.float64)

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                     # enforce sum = 1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length – next power of two

    # ------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------
    best_vec = a.copy()
    best_sc = 2.0 * n * _conv_max_fft(a, L)

    # optimisation parameters
    step0 = 1.0                # initial step magnitude
    beta = 0.9                 # momentum factor
    momentum = np.zeros_like(a)
    top_k_extra = 5            # for the secondary smoother gradient
    sym_interval = 150         # how often to symmetrise
    no_improve = 0
    it = 0
    target = 1.50274349

    # ------------------------------------------------------------
    # Optimisation loop (matrix‑free, O(n log n) per iteration)
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and its maximum -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        cur_max = conv.max()
        cur_sc = 2.0 * n * cur_max

        # ----- keep the best seen so far -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- early exit if target reached -----
        if best_sc < target:
            break

        # ----- gradient: primary (max entry) + secondary (top‑k) -----
        g_max = _grad_max_entry(a, conv)
        g_extra = _grad_topk(a, conv, top_k_extra)
        g = 0.7 * g_max + 0.3 * g_extra   # blend for stability

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- back‑tracking line search -----
        base_alpha = step0 / math.sqrt(it)
        alpha = base_alpha
        cand = None
        cand_sc = None

        for _ in range(12):          # up to 12 attempts
            a_tmp = a - alpha * momentum
            a_tmp = _project_simplex(a_tmp)   # stay on simplex
            # score of the candidate
            a_tmp_fft = np.fft.rfft(a_tmp, n=L)
            conv_tmp_full = np.fft.irfft(a_tmp_fft * a_tmp_fft, n=L)
            conv_tmp = conv_tmp_full[: 2 * n - 1]
            cand_sc = 2.0 * n * conv_tmp.max()
            if cand_sc <= cur_sc:
                cand, cand_sc = a_tmp, cand_sc
                break
            alpha *= 0.5

        # if no descent found, keep the last trial (may be uphill)
        if cand is None:
            cand, cand_sc = a_tmp, cand_sc

        # ----- accept / reject -----
        if cand_sc < cur_sc:
            a, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            # occasional uphill move (Metropolis‑style)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                a, cur_sc = cand, cand_sc

        # ----- periodic symmetrisation (helps with the Toeplitz structure) -----
        if it % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            a_sym_fft = np.fft.rfft(a_sym, n=L)
            conv_sym_full = np.fft.irfft(a_sym_fft * a_sym_fft, n=L)
            conv_sym = conv_sym_full[: 2 * n - 1]
            sym_sc = 2.0 * n * conv_sym.max()
            if sym_sc < cur_sc:
                a, cur_sc = a_sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, a_sym.copy()
                    no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 1500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.5
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
                no_improve = 0

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
