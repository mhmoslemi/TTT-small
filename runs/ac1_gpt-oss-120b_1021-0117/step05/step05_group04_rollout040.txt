def _max_conv_and_grad(a, L):
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*a.size-1]
    max_val = conv.max()
    # find indices of max (within tolerance)
    idxs = np.where(np.abs(conv - max_val) <= 1e-12)[0]
    g = np.zeros_like(a)
    n = a.size
    for k in idxs:
        # Contribution for shift k: conv[k] = sum_i a[i] * a[i+k]
        # gradient w.r.t a[i] is a[i+k] (if i+k < n) + a[i-k] (if i-k >= 0)
        # we can vectorize:
        # shift forward
        forward = np.zeros_like(a)
        if k < n:
            forward[:n-k] = a[k:]  # a[i+k] for i=0..n-k-1
        # shift backward
        backward = np.zeros_like(a)
        if k > 0:
            backward[k:] = a[:n-k]  # a[i-k] for i=k..n-1
        g += forward + backward
    return max_val, a_fft, conv, g
```

def _max_and_subgrad(a):
    n = a.shape[0]
    L = 1 << ((3*n-2).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    max_val = conv.max()
    # find positions where conv close to max
    idxs = np.where(conv >= max_val - 1e-15)[0]
    grad = np.zeros_like(a)
    for idx in idxs:
        d = idx - (n-1)
        # shift +d
        if d >= 0:
            # for i such that i+d < n
            grad[:n-d] += a[d:]
            # and i-d >=0 => i>=d => gradient addition a[i-d]?
            grad[d:] += a[:n-d]
        else:
            dd = -d
            # shift -dd
            grad[dd:] += a[:n-dd]
            grad[:n-dd] += a[dd:]

    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv.max(), a_fft, conv


def _softmax_weights(conv: np.ndarray, max_val: float, eps: float) -> np.ndarray:
    """Soft‑max weights over convolution values (temperature = eps)."""
    w = np.exp((conv - max_val) / eps)
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / s

                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.35
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = tmp / tmp.sum()
                tmp_max, _, _ = _conv_max(tmp, L)
                if tmp_max < cur_max:
                    cur = tmp
                    cur_max = tmp_max
                no_improve = 0

        # ----- early exit if target reached -----
        if best_score < target_score:
            break

    # --------------------------------------------------------
    # Return best sequence found (as plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
