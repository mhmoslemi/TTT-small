def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1)

    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)
    else:
        n = 2048
        a = rng.random(n, dtype=np.float64)
    # Normalize to sum 1
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full_like(a, 1.0/a.size)
    else:
        a = a / a_sum

    n = a.size
    # FFT length
    L = 1 << ((2*n - 1).bit_length())  # Might be enough for conv length 2n-1 and w_sym length.
    # track best
    best_a = a.copy()
    best_sc = _max_conv(best_a, L)
    best_bound = 2.0 * n * best_sc

    eps0 = 0.05  # small temperature for softmax (initial)
    eps_decay = 0.9995
    eta0 = 0.5   # initial learning rate for EG
    eta_decay = 0.9998

    iteration = 0
    while time.monotonic() < deadline - 0.5:
        iteration += 1
        # compute conv
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()
        # compute softmax weights
        eps = eps0 * (eps_decay ** iteration)
        if eps < 1e-8:
            # use exact max mask
            mask = conv >= max_val - 1e-12
            cnt = int(mask.sum())
            if cnt == 0:
                w = np.full_like(conv, 1.0/conv.size)
            else:
                w = np.where(mask, 1.0/cnt, 0.0)
        else:
            w = np.exp((conv - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
        # gradient
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]
        # exponentiated gradient update
        eta = eta0 * (eta_decay ** iteration)
        # update a
        a = a * np.exp(-eta * g)
        # renorm to sum 1
        a_sum = a.sum()
        if a_sum == 0 or not np.isfinite(a_sum):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum
        # evaluate bound
        cur_sc = _max_conv(a, L)
        cur_bound = 2.0 * n * cur_sc
        if cur_bound < best_bound:
            best_a = a.copy()
            best_sc = cur_sc
