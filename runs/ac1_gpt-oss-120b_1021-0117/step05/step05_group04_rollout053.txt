import time
import math
import numpy as np

def _self_conv_max(a: np.ndarray, L: int):
    """Return (max of a*a, FFT of a, full convolution slice)."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv.max(), a_fft, conv

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative step‑function minimizing the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
    else:
        cur = rng.random(1024, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur_sum = cur.sum()
    if cur_sum == 0.0:
        cur[:] = 1.0 / cur.size
    else:
        cur /= cur_sum                     # sum = 1
    n = cur.size
    L = 1 << ((3 * n - 2).bit_length())   # pow‑2 ≥ 3n‑2 for all FFTs

    # ----- bookkeeping ------------------------------------------------
    cur_max, _, _ = _self_conv_max(cur, L)
    best_vec = cur.copy()
    best_max = cur_max
    target_score = 1.50274349
    best_score = 2.0 * n * best_max

    # ----- phase 1: exact‑gradient descent ---------------------------
    base_step = 0.5
    it = 0
    while time.monotonic() < deadline - 5.0:
        it += 1
        cur_max, a_fft, conv = _self_conv_max(cur, L)

        # exact sub‑gradient of the max‑convolution
        mask = (conv >= cur_max - 1e-12).astype(np.float64)
        mask_rev = mask[::-1]
        mask_fft = np.fft.rfft(mask_rev, n=L)
        grad_full = np.fft.irfft(a_fft * mask_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # multiplicative (exponentiated) update with back‑tracking
        step = base_step / math.sqrt(it)
        cand = None
        cand_max = None
        for _ in range(8):
            expo = -step * grad
            expo = np.clip(expo, -50.0, 50.0)
            cand_tmp = cur * np.exp(expo)
            s = cand_tmp.sum()
            if s == 0.0:
                cand_tmp[:] = 1.0 / n
            else:
                cand_tmp /= s
            cand_tmp_max, _, _ = _self_conv_max(cand_tmp, L)
            if cand_tmp_max < cur_max - 1e-15:
                cand, cand_max = cand_tmp, cand_tmp_max
                break
            step *= 0.5
        if cand is None:                     # no descent found
            cand, cand_max = cand_tmp, cand_tmp_max

        # accept if improvement; occasional uphill move
        if cand_max < cur_max - 1e-15:
            cur, cur_max = cand, cand_max
            if cand_max < best_max:
                best_max, best_vec = cand_max, cand.copy()
                best_score = 2.0 * n * best_max
        else:

                cur, cur_max = cand, cand_max

        if it % 1000 == 0:

            sym /= sym.sum()

        if cur[i] <= 0.0:

        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
