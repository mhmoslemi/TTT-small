import time, math, numpy as np

def _score(a):
    """Fast AC1 score for a sequence with sum=1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _exp_sequence(n, tau):
    """Return normalized exponential decreasing sequence of length n."""
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-idx / tau)
    return a / a.sum()

def _powerlaw_sequence(n, p):
    """Return normalized power‑law decreasing sequence a_i = (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return a / a.sum()

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # length of the problem
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 2000

    best_a = None
    best_sc = float('inf')

    # consider the provided parent if present
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s > 0.0:
            a0 = a0 / s
        else:
            a0 = np.full_like(a0, 1.0 / n)
        sc0 = _score(a0)
        best_a = a0
        best_sc = sc0

    # ------------------------------------------------------------------
    # exponential family – pick tau ≈ n / x_opt where x_opt solves
    # (2 - x) + (3x - 2) e^{-x} = 0   →   x_opt ≈ 2.459821
    # ------------------------------------------------------------------
    X_OPT = 2.459821  # pre‑computed root of the optimality equation
    tau_opt = n / X_OPT

    # explore a modest neighbourhood of the analytical optimum
    for factor in np.linspace(0.6, 1.4, 17):
        if time.monotonic() > deadline:
            break
        tau = tau_opt * factor
        if tau <= 0:
            continue
        a = _exp_sequence(n, tau)
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a

    # ------------------------------------------------------------------

def _score(a: np.ndarray) -> float:
    """Fast AC1 upper‑bound for a sequence with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _exp_sequence(n: int, tau: float) -> np.ndarray:
    """Normalized exponentially decreasing sequence of length n."""
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-idx / tau)
    return a / a.sum()


def _powerlaw_sequence(n: int, p: float) -> np.ndarray:
    """Normalized power‑law decreasing sequence a_i = (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return a / a.sum()
