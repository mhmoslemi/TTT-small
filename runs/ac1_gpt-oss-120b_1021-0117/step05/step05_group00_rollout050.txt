def propose_candidate(...):
    rng = np.random.default_rng(seed)
    start=time.monotonic()
    deadline = start + max(0.0,float(budget_s)-0.5)

    # init a0
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0:
            a0 = np.full_like(a0, 1.0/a0.size)
            s = 1.0
        a0 = a0 / s   # sum=1
    else:
        n_init=2000
        t = np.arange(n_init, dtype=np.float64)
        a0 = np.exp(-t/100.0)
        a0 = a0 / a0.sum()
    n=a0.size
    L=1 << ((2*n-1).bit_length())
    cur=a0.copy()
    cur_score = _score(cur)   # compute 2*n*max conv

    best_vec = cur.copy()
    best_sc = cur_score

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1=0.9
    beta2=0.999
    adam_eps=1e-8
    base_lr=0.8   # adjust
    eps_factor=0.2  # initial smoothing factor
    eps_decay=0.9995
    # "eps" variable will be epsilon for softmax: max_val * eps_factor
    eps_fac = eps_factor
    # counts
    it=0
    no_improve=0
    sym_interval=250
    improve_target=1.50274349

    while time.monotonic() < deadline:
        it+=1

        # FFT conv
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv_full.max()

        # update epsilon factor
        eps_fac = max(eps_fac * eps_decay, 1e-5)  # never too small?
        eps = max(max_val * eps_fac, 1e-12)

        # compute weights
        w = np.exp((conv_full - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0/w.size)
            w_sum = w.size
        w = w / w_sum

        # grad
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2*n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = base_lr / np.sqrt(it) * m_hat / (np.sqrt(v_hat) + adam_eps)

        cand = cur - step
        cand = _project_simplex(cand)
        cand_score = _score(cand)

        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_sc:
                best_sc = cand_score
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # accept with prob
            prob = math.exp(-(cand_score - cur_score) / (base_lr + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_score = cand_score
                no_improve += 1
            else:
                no_improve += 1

        # symmetrize
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_score = _score(sym)
            if sym_score < cur_score:
                cur = sym
                cur_score = sym_score
            if sym_score < best_sc:
                best_sc = sym_score
                best_vec = sym.copy()
                no_improve = 0

        # occasional random mass transfer if stuck
        if no_improve > 1500:
            # random mass shift
            i = rng.integers(0, n)
            if cur[i] > 1e-12:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_score = _score(tmp)
                if tmp_score < cur_score:
                    cur = tmp
                    cur_score = tmp_score
                if tmp_score < best_sc:
                    best_sc = tmp_score
                    best_vec = tmp.copy()
                    no_improve = 0
                else:
                    no_improve = 0

        # restart if hopeless
        if no_improve > 4000:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            # reset Adam
            m = np.zeros_like(cur)
            v = np.zeros_like(cur)
            eps_fac = eps_factor
            no_improve = 0
            cur_score = _score(cur)

        # early exit
        if best_sc < improve_target:
            break

    return [float(x) for x in best_vec]

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast AC1 bound (assumes sum ≈ 1)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(convolution) for a sequence with sum≈1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------
    # Initialise a feasible point on the simplex
    # ------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:

            s = 1.0
        a0 = a0 / s                                   # now sum=1
    else:                                               # fallback
        n_fallback = 2000
        t = np.arange(n_fallback, dtype=np.float64)
        a0 = np.exp(-t / 100.0)
        a0 = a0 / a0.sum()

    n = a0.size
    L = 1 << ((2 * n - 1).bit_length())                # FFT length for autocorrelation

    cur = a0.copy()
    cur_score = _score(cur)          # current AC1 bound (sum(cur)=1)

    best_vec = cur.copy()

    # ------------------------------------------------------------
