def adam_step(m, v, g, t, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    m = beta1*m + (1-beta1)*g
    v = beta2*v + (1-beta2)*(g*g)
    m_hat = m/(1-beta1**t)
    v_hat = v/(1-beta2**t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    return step, m, v

import time, math
import numpy as np

# --------------------------------------------------------------
# Helper: projection onto the probability simplex (sum = 1, x >= 0)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# --------------------------------------------------------------
# Helper: maximum value of the full convolution a*a using FFT
# --------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv_full.max())

    rng = np.random.default_rng(seed)
    start = time.monotonic()

    # ------------------- bookkeeping -------------------

    # ------------------- Adam optimiser parameters -------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.04

    # ------------------- gradient‑descent phase -------------------
    while time.monotonic() < deadline - 2.0:
        it += 1

        # current full convolution
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        eps_temp = max(1e-8, 0.2 * (0.995 ** it))

                break

    # final fallback – never return an empty sequence
    if best.size == 0:
        best = np.array([1.0])

    return [float(x) for x in best]
