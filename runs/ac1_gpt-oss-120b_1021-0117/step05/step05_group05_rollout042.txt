import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _score(a, L=None):
    """Exact score used by the evaluator (scale‑invariant)."""
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    # s will be 1 after simplex projection, but keep the formula for safety
    return 2.0 * n * max_val / (s * s)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a small safety margin
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # initial vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a0 = rng.random(1024, dtype=np.float64)

    n = a0.shape[0]
    cur = _project_simplex(np.maximum(a0, 0.0))
    L = 1 << ((2 * n - 1).bit_length())

    cur_sc = _score(cur, L)
    best_vec = cur.copy()
    best_sc = cur_sc

    # ------------------------------------------------------------------
    # Simulated‑annealing style search
    # ------------------------------------------------------------------
    # temperature initialised from the current score (scale‑invariant)
    T = cur_sc
    T_decay = 0.9998           # exponential cooling
    step_factor = 0.1          # max fraction of a source entry moved each step
    step_decay = 0.99999       # slowly shrink step_factor
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # ----------- a random mass transfer between two coordinates -----------
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)

        # -------------- occasional large random perturbation -----------------
        if it % 2000 == 0:
            rnd = _project_simplex(rng.random(n))
            alpha = 0.1
            cand = (1.0 - alpha) * cur + alpha * rnd
            cand = _project_simplex(cand)
            cand_sc = _score(cand, L)
            if cand_sc < cur_sc or rng.random() < math.exp(-(cand_sc - cur_sc) / max(T, 1e-12)):
                cur, cur_sc = cand, cand_sc
                if cand_sc < best_sc:
                    best_sc, best_vec = cand_sc, cand.copy()

        # -------------------- temperature and step decay --------------------
        T *= T_decay
        step_factor = max(step_factor * step_decay, 1e-4)

        # ----------------------- early success check -------------------------
        if best_sc < 1.50274349:
            break

    # ------------------------------------------------------------------
    # Final polishing using the sub‑gradient of the maximal convolution term
    # ------------------------------------------------------------------
    # try a few gradient steps if time still permits
    grad_iters = 0
    while time.monotonic() < deadline and grad_iters < 250:
        grad_iters += 1

        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        # mask for all positions achieving the maximum (within tolerance)
        mask = conv_full >= max_val - 1e-12
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt

        # sub‑gradient of max(conv) w.r.t. a
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # simple line‑search step
        step = 0.8 / math.sqrt(grad_iters)
        cand = cur - step * grad
        cand = _project_simplex(cand)
        cand_sc = _score(cand, L)

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()

    # ------------------------------------------------------------------
    # Return the best vector found (as plain Python floats)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
