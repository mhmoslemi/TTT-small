def _conv_max(a):
    n = a.shape[0]
    L = 1 << ((2*n-1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    return conv.max()
def _score(a):
    n = a.shape[0]
    maxb = _conv_max(a)
    s = a.sum()
    return 2.0 * n * maxb / (s*s)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, math, random
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s)-0.5)

    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=np.float64)
    else:
        n = 2000
        a = rng.random(n)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        a = a * (0.01 / max(s, 1e-12))
    # Normalize sum to 1 for stable range (optional)
    a = a / a.sum()
    n = a.shape[0]

    def score_vec(vec):
        maxb = _conv_max(vec)
        s = vec.sum()
        return 2.0 * n * maxb / (s*s)

    cur = a.copy()
    cur_score = score_vec(cur)
    best = cur.copy()
    best_score = cur_score

    # temperature schedule
    T = cur_score * 0.1  # start temperature
    iter = 0
    while time.monotonic() < deadline:
        iter += 1
        # Random move: select i != j
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # delta as a proportion of a[i]
        if cur[i] == 0:
            continue
        delta = cur[i] * rng.random() * 0.1  # up to 10% transfer
        new_vec = cur.copy()
        new_vec[i] -= delta
        new_vec[j] += delta
        # enforce nonneg
        # not needed since delta <= cur[i]
        new_score = score_vec(new_vec)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = new_vec
            cur_score = new_score
            if new_score < best_score:
                best = new_vec.copy()
                best_score = new_score
                # optionally reduce T for better acceptance
        # update temperature

        # Periodically perform smoothing step (optional)
        if iter % 20000 == 0:
            # apply averaging filter to reduce peaks
            kernel = np.ones(5) / 5.0
            smooth = np.convolve(cur, kernel, mode='same')

import time, math, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

def _score(a, L, n):
    """Score = 2*n*max(conv), sum(a) is 1 by construction."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

            cand_tmp = _project_simplex(cand_tmp)
            cand_tmp_sc = _score(cand_tmp, L, n)
            if cand_tmp_sc <= cur_sc:
                cand, cand_sc = cand_tmp, cand_tmp_sc
                break
            alpha *= 0.5
        if cand is None:               # last trial kept even if worse
            cand, cand_sc = cand_tmp, cand_tmp_sc

        # accept / reject
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            prob = math.exp(-(cand_sc - cur_sc) / max(alpha, 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # periodic symmetrisation (helps for the symmetric optimum)
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L, n)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # if stuck, perform a small redistribution move
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.4
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L, n)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # occasional stochastic perturbation of the best solution
        if no_improve > 5000 and time.monotonic() + 2.0 < deadline:
            sigma = 0.02 / math.sqrt(it + 1)
            pert = best_vec + sigma * rng.standard_normal(n)
            pert = np.maximum(pert, 0.0)
            pert = _project_simplex(pert)
            pert_sc = _score(pert, L, n)
            cur, cur_sc = pert, pert_sc
            no_improve = 0

        # early exit if target reached
        if best_sc < target:
            break

    return [float(x) for x in best_vec]
