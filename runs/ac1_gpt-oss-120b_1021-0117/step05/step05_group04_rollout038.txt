import time
import math
import numpy as np

# ------------------------------------------------------------
# Simple utilities (top‑level, no closures)
# ------------------------------------------------------------
def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to have sum = 1 (fallback to uniform)."""
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _self_conv_max(a: np.ndarray, L: int):
    """Return (max(conv), fft of a, full convolution) for a*a."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv.max(), a_fft, conv


def _softmax_weights(conv: np.ndarray, cur_max: float, eps: float) -> np.ndarray:
    """Soft‑max over convolution values with temperature = eps."""
    w = np.exp((conv - cur_max) / eps)
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size, dtype=np.float64)
    return w / s


def _gradient(a_fft: np.ndarray, w: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution objective.
    Implements  g_i = 2 * (a * w_rev)_{(2n‑2)-i}.
    """
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]
