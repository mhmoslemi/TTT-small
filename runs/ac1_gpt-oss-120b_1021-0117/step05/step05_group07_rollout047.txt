def _softmax_weights(conv, eps):
    # compute w
    max_val = conv.max()
    if eps > 1e-12:
        w = np.exp((conv - max_val) / eps)
        sum_w = w.sum()
        if sum_w == 0:
            w = np.ones_like(conv) / conv.size
        else:
            w /= sum_w
    else:
        w = np.zeros_like(conv)
        mask = conv == max_val
        cnt = mask.sum()
        if cnt > 0:
            w[mask] = 1.0 / cnt
    return w

import time, math, numpy as np

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
