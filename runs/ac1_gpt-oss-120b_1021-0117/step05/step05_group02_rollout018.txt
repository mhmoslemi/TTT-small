import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (non‑negative, sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast score (2 * n * max a*a convolution) – assumes sum(a)=1
# ------------------------------------------------------------
def _score_fft(a: np.ndarray, L: int) -> float:
    """Return 2 * n * max(conv(a,a)) using FFT of length L."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

                cur, cur_sc = pert, pert_sc
                a_fft, conv_full, conv_vals, max_cur = (
                    pert_fft, pert_conv_full, pert_conv_full[: 2 * n - 1], pert_max)
                if pert_sc < best_sc:
                    best_vec, best_sc = pert.copy(), pert_sc
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------
    # Return the best discovered candidate as a plain Python list
    # ------------------------------------------------------------
    return [float(x) for x in best_vec]
