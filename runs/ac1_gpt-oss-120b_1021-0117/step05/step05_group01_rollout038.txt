def _grad_max(a, max_idxs):
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in max_idxs:
        # compute range of j where both a_j and a_{k-j} in [0,n)
        j_start = max(0, k - (n - 1))
        j_end = min(k, n-1)  # inclusive
        # slice of a_j
        j_range = np.arange(j_start, j_end + 1)
        # corresponding partner indices:
        partner = k - j_range
        # now add a[partner] + a[j_range] to gradient
        g[j_range] += a[partner] + a[j_range]
    # average if multiple indices
    g /= len(max_idxs)
    return g

# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = z}."""
    n = v.size
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Compute full linear convolution via FFT and return it
# ------------------------------------------------------------
def _full_conv(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length L (>=2n‑1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)
    return conv[: 2 * a.size - 1]


# ------------------------------------------------------------
# Gradient of the (sub‑)gradient of the maximal convolution entry.
# Uses the top‑k largest convolution values (averaged).
# ------------------------------------------------------------
def _grad_topk(a: np.ndarray, conv: np.ndarray, top_k: int) -> np.ndarray:
    n = a.size
    # indices of the `top_k` largest entries
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    g = np.zeros_like(a)
    for k in idxs:
        if k < n:                     # left part of the convolution
            sub = a[: k + 1]
            g[: k + 1] += sub + sub[::-1]
        else:                         # right part
            start = k - n + 1
            sub = a[start : n]
            g[start : n] += sub + sub[::-1]
    if idxs.size:
        g /= float(idxs.size)
    return g


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # -------------------- initial vector --------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # a modest random start if no parent is supplied
        a = rng.random(2048, dtype=np.float64)

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:               # avoid division by zero
        a[:] = 1.0
    a = a / a.sum()                   # enforce sum = 1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1

    # -------------------- bookkeeping --------------------
    best_vec = a.copy()
    conv = _full_conv(a, L)
    best_sc = 2.0 * n * conv.max()

    # optimisation parameters
    step0 = 1.0               # base step magnitude
    beta = 0.95               # momentum factor
    momentum = np.zeros_like(a)
    top_k = min(30, max(5, n // 100))    # a few dozen biggest entries
    sym_interval = 120                     # symmetrise every 120 iters
    no_improve = 0
    target = 1.50274349
    it = 0

    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and its max -----
        conv = _full_conv(a, L)
        cur_max = conv.max()
        cur_sc = 2.0 * n * cur_max

        # ----- keep the best seen so far -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- early exit if target reached -----
        if best_sc < target:
            break

        # ----- gradient from top‑k largest conv entries -----
        g = _grad_topk(a, conv, top_k)

        # ----- momentum update (descent direction) -----
        momentum = beta * momentum - (1.0 - beta) * g

        # ----- back‑tracking line search -----
        base_alpha = step0 / math.sqrt(it)
        cand = None
        cand_sc = None
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125):
            alpha = base_alpha * factor
            a_tmp = a + alpha * momentum
            a_tmp = _project_simplex(a_tmp)      # stay on simplex
            tmp_sc = 2.0 * n * _full_conv(a_tmp, L).max()
            if tmp_sc <= cur_sc:
                cand, cand_sc = a_tmp, tmp_sc
                break
        # if no descent was found, keep the best trial
        if cand is None:
            # pick the smallest score among the tried alphas
            best_trial_sc = float("inf")
            best_trial_vec = None
            for factor in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125):
                alpha = base_alpha * factor
                a_tmp = a + alpha * momentum
                a_tmp = _project_simplex(a_tmp)
                tmp_sc = 2.0 * n * _full_conv(a_tmp, L).max()
                if tmp_sc < best_trial_sc:
                    best_trial_sc, best_trial_vec = tmp_sc, a_tmp.copy()
            cand, cand_sc = best_trial_vec, best_trial_sc

        # ----- accept if improvement -----
        if cand_sc < cur_sc:
            a, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            # occasional uphill acceptance (simulated‑annealing style)
            prob = math.exp(-(cand_sc - cur_sc) / (base_alpha + 1e-12))
            if rng.random() < prob:
                a, cur_sc = cand, cand_sc

        # ----- periodic symmetrisation (helps with Toeplitz structure) -----
        if it % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            sym_sc = 2.0 * n * _full_conv(a_sym, L).max()
            if sym_sc < cur_sc:
                a, cur_sc = a_sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, a_sym.copy()
                    no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 2500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.4
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
            no_improve = 0

    # --------------------------------------------------------
    # Return the best candidate found (as a plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
