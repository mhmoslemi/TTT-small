def project_simplex(v):
    # ensure sum(v)=1 and v>=0
    # based on algorithm by Wang & Carreira-Perpinan (2013)
    # v can be any real vector.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

import time, math, numpy as np

def _fft_len(n):
    return 1 << ((2 * n - 1).bit_length())

def _conv_max(vec, L):
    fft = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fft * fft, n=L)[:2*vec.size - 1]
    return conv.max(), conv

def _subgrad_from_idx(a, idx):
    n = a.shape[0]
    g = np.zeros_like(a)
    start = max(0, idx - (n - 1))
    end = min(n - 1, idx)
    if start <= end:
        i = np.arange(start, end + 1)
        g[i] = a[idx - i]
    return 2.0 * g

def _subgrad_topk(a, conv, k):
    # compute gradient as sum of subgradients of top k conv entries
    n = a.shape[0]
    flat = conv
    if flat.size <= k:
        idxs = np.arange(flat.size)
    else:
        idxs = np.argpartition(-flat, k-1)[:k]  # unsorted top k indices
    grad = np.zeros_like(a)
    for idx in idxs:
        grad += _subgrad_from_idx(a, int(idx))
    return grad / float(k)  # average

import time
import math
import numpy as np

def _fft_len(n):
    """next power of two ≥ 2·n‑1 for convolution via FFT."""
    return 1 << ((2 * n - 1).bit_length())

def _conv_max(vec, L):
    """Return (max(conv), full convolution) of vec with itself using FFT."""
    fft = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fft * fft, n=L)[: 2 * vec.size - 1]
    return conv.max(), conv

def _subgrad_from_idx(a, idx):
    """Sub‑gradient of the max‑entry of a*a at index idx."""
    n = a.shape[0]
    g = np.zeros_like(a)
    start = max(0, idx - (n - 1))
    end = min(n - 1, idx)
    if start <= end:
        i = np.arange(start, end + 1)
        g[i] = a[idx - i]
    return 2.0 * g

def _subgrad_topk(a, conv, k):
    """Average sub‑gradient over the k largest convolution entries."""
    if conv.size <= k:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, k - 1)[:k]   # unsorted top‑k
    grad = np.zeros_like(a)
    for idx in idxs:
        grad += _subgrad_from_idx(a, int(idx))
    return grad / float(k)
