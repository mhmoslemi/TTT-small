# --------------------------------------------------------------
# Simplex projection (L2 onto {x >= 0, sum = 1})
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex. """
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv_full.max())

            break

    # ------------------------------------------------------------------
    # Deterministic anti‑identity permutation (large opposite small)
    # ------------------------------------------------------------------
    vals = np.sort(best)[::-1]                 # descending
    perm = np.empty_like(vals)
    left, right, k = 0, n - 1, 0
    while left <= right:
        perm[left] = vals[k]
        k += 1
        if left < right:
            perm[right] = vals[k]
            k += 1
        left += 1
        right -= 1
    perm_sc = _max_conv(perm, L)
    perm_bound = 2.0 * n * perm_sc
    if perm_bound < best_bound:
        best = perm.copy()
        best_bound = perm_bound

    # ------------------------------------------------------------------
    # Simple hill‑climbing swaps to polish the permutation
    # ------------------------------------------------------------------
    cur = best.copy()
    cur_bound = best_bound
    while time.monotonic() < deadline - 0.2:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        cur_max = _max_conv(cur, L)
        new_bound = 2.0 * n * cur_max
        if new_bound < cur_bound:
            cur_bound = new_bound
            if new_bound < best_bound:
                best = cur.copy()
                best_bound = new_bound
            if best_bound < target_bound:
                break
        else:
            # revert swap
            cur[i], cur[j] = cur[j], cur[i]

    # ------------------------------------------------------------------
    # Return the best candidate found (as a plain Python list)
    # ------------------------------------------------------------------
    if best.size == 0:
        best = np.array([1.0])
    return [float(x) for x in best]
