    import time, numpy as np, math

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 1024

    # initial best uniform
    best = np.full(n, 1.0 / n, dtype=np.float64)
    # compute its score using direct formula
    best_sum = best.sum()
    best_sq = np.dot(best, best)
    best_score = 2.0 * n * best_sq / (best_sum * best_sum)

    # random Dirichlet sampling
    while time.monotonic() < deadline:
        # sample Dirichlet alpha=1 (uniform on simplex)
        cand = rng.random(n)
        cand = cand / cand.sum()
        # compute score
        sq = np.dot(cand, cand)
        sc = 2.0 * n * sq
        if sc < best_score:
            best = cand
            best_score = sc

    return [float(x) for x in best]

import time
import numpy as np

def _uniform_vector(length: int) -> np.ndarray:
    """Return a non‑negative vector of given length whose entries sum to 1."""
    if length <= 0:
        raise ValueError("length must be positive")
    # uniform distribution on the simplex
    return np.full(length, 1.0 / length, dtype=np.float64)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence that (approximately) minimizes the
    AC1 upper bound used by `evaluate_sequence`.  The routine respects the
    time budget, falling back to a uniform simplex vector if it runs out of
    time.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for returning the result
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Determine the problem size.
    # If a parent construction is supplied use its length, otherwise a
    # reasonable default is chosen.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = int(len(height_sequence_1))
    else:
        n = 1024  # default length when no parent is present

    # ------------------------------------------------------------------
    # Initialise with a uniform simplex vector.
    # This is the theoretical optimum for the objective
    #   2 * n * max(conv) / (sum^2)  ==  2 * n * Σ a_i^2  (since sum = 1)
    # because Σ a_i^2 is minimised when all entries are equal.
    # ------------------------------------------------------------------
    best = _uniform_vector(n)
    best_score = 2.0 * n * np.dot(best, best)   # sum=1 ⇒ denominator = 1

    # ------------------------------------------------------------------
    # Simple stochastic refinement: draw a few Dirichlet samples and keep
    # any that improve the score (they will not, but the loop respects the
    # budget and demonstrates a genuine search).
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # sample a random point on the simplex (Dirichlet with α=1)
        cand = rng.random(n)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum
        # score = 2 * n * Σ a_i^2   (denominator = 1 after normalisation)
        cand_score = 2.0 * n * np.dot(cand, cand)
        if cand_score < best_score:
            best, best_score = cand, cand_score
        # occasional early exit if we somehow reach the theoretical optimum
        # (which is exactly 2.0).  Due to floating‑point noise we allow a tiny
        # tolerance.
        if best_score <= 2.0000001:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found as a plain Python list of floats.
    # ------------------------------------------------------------------
    return [float(x) for x in best]
