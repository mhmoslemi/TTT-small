# Project onto the probability simplex (sum = 1, x_i >= 0)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if v.size == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    return w

    vals = np.sort(best_a)[::-1]
    perm = np.empty_like(vals)
    left, right = 0, n - 1
    k = 0
    while left <= right:
        perm[left] = vals[k]
        k += 1
        if left < right:
            perm[right] = vals[k]
            k += 1
        left += 1
        right -= 1
    perm_max = np.convolve(perm, perm).max()
    if perm_max < best_max:
        best_a = perm.copy()
        best_max = perm_max
        best_bound = 2.0 * n * best_max

    # --------------------------------------------------------------
    # Light hill‑climbing swaps to polish the permutation
    # --------------------------------------------------------------
    cur = best_a.copy()
    cur_max = best_max
    while time.monotonic() < deadline - 0.1:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # perform a pure swap (delta = a[i] - a[j])
        delta = cur[i] - cur[j]
        if delta == 0.0:
            continue
        # treat as moving delta from i to j
        old_ai = cur[i]
        old_aj = cur[j]

        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j

        old_c2i = conv[idx2i] if cur is best_a else np.convolve(cur, cur)[idx2i]
        old_cij = conv[idxij] if cur is best_a else np.convolve(cur, cur)[idxij]
        old_c2j = conv[idx2j] if cur is best_a else np.convolve(cur, cur)[idx2j]

        # new convolution entries after the swap
        new_c2i = old_c2i + (-2.0 * delta * old_ai + delta * delta)
        new_cij = old_cij + (2.0 * delta * (old_ai - old_aj - delta))
        new_c2j = old_c2j + (2.0 * delta * old_aj + delta * delta)

        # compute new maximum
        if idx_max not in (idx2i, idxij, idx2j):
            cand_max = max(cur_max, new_c2i, new_cij, new_c2j)
        else:
            tmp = np.convolve(cur, cur)
            tmp[idx2i] = new_c2i
            tmp[idxij] = new_cij
            tmp[idx2j] = new_c2j
            cand_max = tmp.max()

        if cand_max < cur_max:
            cur[i], cur[j] = cur[j], cur[i]
            conv[idx2i] = new_c2i
            conv[idxij] = new_cij
            conv[idx2j] = new_c2j
            cur_max = cand_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = cur.copy()
                best_bound = 2.0 * n * best_max
                if best_bound < 1.50274349:
                    break

    # ------------------------------------------------------------------
    # Return the best sequence discovered (plain Python list)
    # ------------------------------------------------------------------
    if best_a.size == 0:
        best_a = np.array([1.0])
    return [float(x) for x in best_a]
