    start = time.monotonic()
    deadline = start + max(0.0,float(budget_s)-0.5)

    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 1024

    # Helper functions
    project_simplex = ...

    def max_conv(a):
        L = 1 << ((2*n - 1).bit_length())
        af = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(af*af, n=L)[:2*n-1]
        return conv.max()

    # Evaluate function: score = 2*n*max_conv
    def evaluate(a):
        return 2.0 * n * max_conv(a)

    # 1. Initialize candidates
    best_vec = None
    best_score = np.inf

    # Starting candidates:
    # a) uniform
    uniform = np.full(n, 1.0/n)
    best_vec = uniform
    best_score = evaluate(uniform)

    # b) exponential decays across range of r.
    for r in np.linspace(0.5, 0.999, 30):  # exclude extremes
        a = np.power(r, np.arange(n))
        a = a / a.sum()
        sc = evaluate(a)
        if sc < best_score:
            best_score, best_vec = sc, a

    # c) reversed exponential (i.e., decreasing from right): same as above because shape same after normalization but reversed may affect auto? Actually exponent decreasing from 0 to n-1 same as r^i; reverse is r^(n-1-i). That is same distribution reversed, but autocorrelation is symmetric; but shift correlation with reverse is same?     # d) power law
    for p in np.linspace(0.0, 3.0, 30):

        a = a / a.sum()
        sc = evaluate(a)
        if sc < best_score:
            best_score, best_vec = sc, a

    # e) linear decreasing shape: a_i = (n-i) (or maybe (i+1)).
    a = np.arange(n, 0, -1, dtype=np.float64)
    a = a / a.sum()
    sc = evaluate(a)
    if sc < best_score:
        best_score, best_vec = sc, a

    # f) quadratic shape: a_i = (n-i)^2
    a = np.square(np.arange(n, 0, -1))
    a = a / a.sum()
    sc = evaluate(a)
    if sc < best_score:
        best_score, best_vec = sc, a

    # g) more exotic logistic shape
    for k in [0.05, 0.1, 0.2, 0.4]:
        x = np.arange(n) - n/2
        a = 1.0 / (1.0 + np.exp(k * x))
        a = a / a.sum()
        sc = evaluate(a)
        if sc < best_score:
            best_score, best_vec = sc, a

    # Now we have a decent candidate.

    # 2. Local search/hill climbing using random swaps.
    # We'll run loop until deadline. 
    cur = best_vec.copy()
    cur_score = best_score

    temperature = 0.1  # initial "step magnitude" for delta ratio
    # We'll adapt temperature decreasing.

    while time.monotonic() < deadline:
        # propose move: pick two indices i,j and move delta
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # Choose delta as scaling of cur[i]
        max_delta = cur[i] * 0.5
        if max_delta <= 0:
            continue
        delta = max_delta * (rng.random() - 0.5) * 2.0 * temperature  # within +/- max_delta*temperature
        # ensure new values nonnegative
        new_i = cur[i] - delta
        new_j = cur[j] + delta
        if new_i < 0 or new_j < 0:
            continue
        cand = cur.copy()
        cand[i] = new_i
        cand[j] = new_j

        cand_score = evaluate(cand)

        # acceptance rule
        if cand_score < cur_score or rng.random() < math.exp((cur_score - cand_score) / (temperature+1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score, best_vec = cand_score, cand.copy()
        # cooling schedule
        temperature *= 0.9999  # slow cooling
        if temperature < 1e-4:
            temperature = 0.1  # reset occasionally

        # optionally occasional symmetrization
        if rng.random() < 0.01:
            sym = (cur + cur[::-1]) * 0.5
            sym = project_simplex(sym)
            sym_score = evaluate(sym)
            if sym_score < cur_score:
                cur, cur_score = sym, sym_score
                if sym_score < best_score:
                    best_score, best_vec = sym_score, sym.copy()

    # Return best vector as list.
    return [float(x) for x in best_vec]

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    # find rho: last index where u_i - css_i/(i+1) > 0
    cond = u - css / np.arange(1, n+1)
    rho = np.where(cond > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0
    return w

import time, math, numpy as np

# -------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x >= 0, sum(x)=1}
# -------------------------------------------------------------
def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    cond = u - css / np.arange(1, n + 1)
    rho = np.where(cond > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = v - theta
    w[w < 0.0] = 0.0
    return w

    return float(2.0 * a.size * conv.max())

                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, sym.copy()

        # -------------------------------------------------
        # occasional random mass‑transfer move (exploration)
        # -------------------------------------------------
        if it % 2000 == 0:
            i1 = rng.integers(0, n)
            i2 = rng.integers(0, n - 1)
            if i2 >= i1:
                i2 += 1
            delta = cur[i1] * rng.random() * 0.5
            tmp = cur.copy()
            tmp[i1] -= delta
            tmp[i2] += delta
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp, L)
            if tmp_sc < best_sc:
                best_sc, best_vec = tmp_sc, tmp.copy()
                cur, cur_sc = tmp, tmp_sc

        if best_sc < target:
            break

    # ---------------------------------------------------------
    # 3️⃣  Safety fallback (should never be needed)
    # ---------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)

    return [float(x) for x in best_vec]
