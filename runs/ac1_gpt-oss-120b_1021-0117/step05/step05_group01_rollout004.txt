import time, math, numpy as np

def _project_simplex(v, z=1.0):
    # same as previous
    n=v.size
    u=np.sort(v)[::-1]
    css=np.cumsum(u)-z
    rho=np.where(u - css/np.arange(1,n+1) > 0)[0]
    if rho.size==0:
        theta=0.0
    else:
        rho=rho[-1]
        theta=css[rho]/(rho+1)
    w=np.maximum(v-theta,0.0)
    return w

def _softmax_weights(conv, tau):
    # stable softmax
    cmax = np.max(conv)
    ex = np.exp((conv - cmax)/tau)
    sum_ex = np.sum(ex)
    if sum_ex == 0:
        # fallback uniform
        return np.full_like(conv, 1.0/conv.size)
    return ex / sum_ex

def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=z}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)
