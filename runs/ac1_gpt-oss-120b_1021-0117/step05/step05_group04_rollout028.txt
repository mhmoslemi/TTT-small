import time

import numpy as np

# ------------------------------------------------------------
# Helper functions (all top‑level)
# ------------------------------------------------------------
def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector so its sum is 1 (or fallback to uniform)."""
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _max_convolution(v: np.ndarray, L: int) -> float:
    """Return max value of the linear self‑convolution of v using FFT."""
    fft = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fft * fft, n=L)[: 2 * v.size - 1]
    return float(conv.max())


def _softmax_weights(conv: np.ndarray, cur_max: float, eps: float) -> np.ndarray:
    """Soft‑max over convolution values, stabilised by temperature eps."""
    w = np.exp((conv - cur_max) / eps)          # ≤ 1, never overflows
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size, dtype=np.float64)
    return w / s


def _gradient(a_fft: np.ndarray, w: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution objective.
    Returns a length‑n vector.
    """
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]


def _alternating_order(n: int) -> np.ndarray:
    """
    Produce indices 0, n‑1, 1, n‑2, 2, n‑3, … .
    Used to place large values far apart.
    """
    order = np.empty(n, dtype=int)
    left, right = 0, n - 1
    pos = 0
    while left <= right:
        order[pos] = left
        pos += 1
        if left != right:
            order[pos] = right
            pos += 1
        left += 1
        right -= 1
    return order

    """
    Return a non‑empty list of non‑negative coefficients whose AC1 bound
    (2 n·max(conv)/sum²) is as low as possible.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin before the sandbox timeout
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initial vector – use the parent if it exists
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)
    else:
        cur = rng.random(2048, dtype=np.float64)

    cur = _normalize(cur)
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())      # next power of two ≥ 2n‑1

    # --------------------------------------------------------
    # Best‑so‑far bookkeeping
    # --------------------------------------------------------
    cur_max = _max_convolution(cur, L)
    best_vec = cur.copy()
    best_max = cur_max
    best_score = 2.0 * n * best_max
    target_score = 1.50274349

    # --------------------------------------------------------
    # One cheap structural improvement: alternating order
    # --------------------------------------------------------
    order = _alternating_order(n)
    sorted_vals = np.sort(cur)[::-1]          # descending
    alt = np.empty_like(cur)
    alt[order] = sorted_vals
    alt = _normalize(alt)                     # permutation keeps sum 1
    alt_max = _max_convolution(alt, L)
    if alt_max < best_max:
        best_max = alt_max
        best_vec = alt.copy()
        cur = alt.copy()
        cur_max = alt_max
        best_score = 2.0 * n * best_max

    # --------------------------------------------------------
    # Parameters for the soft‑max gradient descent
    # --------------------------------------------------------
    eps0 = 0.4
    eps_decay = 0.9995
    eps_min = 1e-12
    step0 = 0.3
    beta = 0.9
    momentum = np.zeros_like(cur)
    it = 0
    no_improve = 0

    # --------------------------------------------------------
    # Main optimisation loop (matrix‑free, FFT‑based)
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution & max (FFT) -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        cur_max = float(conv.max())

        # ----- soft‑max weights -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = _softmax_weights(conv, cur_max, eps)

        # ----- gradient (matrix‑free) -----
        grad = _gradient(a_fft, w, n, L)

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- multiplicative (exponentiated) update -----
        step = step0 / math.sqrt(it)
        exponent = -step * momentum
        exponent = np.clip(exponent, -50.0, 50.0)   # stability
        cand = cur * np.exp(exponent)
        cand = _normalize(cand)

        cand_max = _max_convolution(cand, L)

        # ----- acceptance test -----
        if cand_max < cur_max - 1e-15:
            cur = cand
            cur_max = cand_max
            no_improve = 0
            if cand_max < best_max:
                best_max = cand_max
                best_vec = cand.copy()
                best_score = 2.0 * n * best_max
        else:
            no_improve += 1
            # occasional stochastic move to avoid plateaus
            if no_improve % 500 == 0:
                i = rng.integers(0, n)
                if cur[i] > 0.0:
                    delta = cur[i] * rng.random() * 0.2
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1
                    tmp = cur.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _normalize(tmp)
                    tmp_max = _max_convolution(tmp, L)
                    if tmp_max < cur_max:
                        cur = tmp
                        cur_max = tmp_max
                        no_improve = 0
                        if tmp_max < best_max:
                            best_max = tmp_max
                            best_vec = tmp.copy()
                            best_score = 2.0 * n * best_max

        # ----- periodic symmetrisation (helps because convolution is symmetric) -----
        if it % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _normalize(sym)
            sym_max = _max_convolution(sym, L)
            if sym_max < cur_max:
                cur = sym
                cur_max = sym_max
                if sym_max < best_max:
                    best_max = sym_max
                    best_vec = sym.copy()
                    best_score = 2.0 * n * best_max
                no_improve = 0

        # ----- re‑apply alternating order every few thousand steps -----
        if it % 1000 == 0:
            vals = np.sort(cur)[::-1]
            alt2 = np.empty_like(cur)
            alt2[order] = vals
            alt2 = _normalize(alt2)
            alt2_max = _max_convolution(alt2, L)
            if alt2_max < cur_max:
                cur = alt2
                cur_max = alt2_max
                if alt2_max < best_max:
                    best_max = alt2_max
                    best_vec = alt2.copy()
                    best_score = 2.0 * n * best_max
                no_improve = 0

        # ----- early exit if already under the target -----
        if best_score < target_score:
            break

    # --------------------------------------------------------
    # Return the best sequence found (plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
