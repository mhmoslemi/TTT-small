def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to have sum = 1 (handles zero‑vector)."""
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _self_conv_max(a: np.ndarray, L: int):
    """Return max of a*a (linear convolution) together with its FFT."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv.max(), a_fft, conv


def _softmax_weights(conv: np.ndarray, cur_max: float, eps: float) -> np.ndarray:
    """Soft‑max over convolution values; temperature = eps."""
    w = np.exp((conv - cur_max) / eps)
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size, dtype=np.float64)
    return w / s


def _gradient(a_fft: np.ndarray, w: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of the smoothed‑max objective.
    Implements 2·(a * w_rev)[(2n‑2)‑i] using FFTs.

    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]

                best_vec = cand.copy()
                best_score = 2.0 * n * best_max
        else:
            no_improve += 1
            # occasional uphill move (Metropolis style)
            prob = math.exp(-(cand_max - cur_max) / (base_lr + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_max = cand_max

        # ---- periodic symmetrisation (helps with Toeplitz structure) ----
        if it % 200 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _normalize(sym)
            sym_max, _, _ = _self_conv_max(sym, L)
            if sym_max < cur_max:
                cur = sym
                cur_max = sym_max
                if sym_max < best_max:
                    best_max = sym_max
                    best_vec = sym.copy()
                    best_score = 2.0 * n * best_max
                no_improve = 0

        # ---- random mass‑transfer when the search stalls ----
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _normalize(tmp)
                tmp_max, _, _ = _self_conv_max(tmp, L)
                if tmp_max < cur_max:
                    cur = tmp
                    cur_max = tmp_max
                    no_improve = 0
                    if tmp_max < best_max:
                        best_max = tmp_max
                        best_vec = tmp.copy()
                        best_score = 2.0 * n * best_max
                else:
                    no_improve = 0

        # ---- early exit if target reached ----
        if best_score < target_score:
            break

    # ------------------------------------------------------------
    # Return the best sequence found (plain Python list)
    # ------------------------------------------------------------
    return [float(x) for x in best_vec]
