    n = len(a)
    L = 1 << ((2*n-2).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
    return conv.max()

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ------------------------------------------------------------
def _normalize(v: np.ndarray) -> np.ndarray:
    """Return a non‑negative vector with sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _conv_max(a: np.ndarray, L: int):
    """Return (max value, index of max) of the linear convolution a*a."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _gradient_at_max(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Gradient of the (unsmoothed) max‑convolution at index k_max.
    For each i:  g_i = a_{k_max-i} (if in range) + a_{i-k_max} (if in range).
    """
    n = a.size
    grad = np.zeros_like(a)

    # first term: a_{k_max-i}
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    if i_end >= i_start:
        i = np.arange(i_start, i_end + 1)
        j = k_max - i
        grad[i] += a[j]

    # second term: a_{i-k_max}
    i_start2 = max(k_max, 0)
    i_end2 = n - 1
    if i_end2 >= i_start2:
        i2 = np.arange(i_start2, i_end2 + 1)
        j2 = i2 - k_max
        grad[i2] += a[j2]

    return grad

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative list of coefficients whose AC1 bound
    (2 n·max(a*a)/sum(a)^2) is as low as possible.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # keep a safety margin

    # --------------------------------------------------------
    # Initialise vector (parent if present)
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # uniform fallback of length ≈2000
        cur = np.full(2048, 1.0 / 2048, dtype=np.float64)

    cur = _normalize(cur)
    n = cur.size

    # FFT length – next power of two ≥ 2n‑1
    L = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------
    # Optimisation parameters
    # --------------------------------------------------------
    step0 = 0.35                # base step size for exponentiated update
    eps0 = 0.6                  # initial soft‑max temperature (unused in final version)
    beta = 0.9                  # momentum factor
    momentum = np.zeros_like(cur)

    best_vec = cur.copy()
    best_max, _ = _conv_max(best_vec, L)
    best_score = 2.0 * n * best_max
    target_score = 1.50274349

    it = 0
    no_improve = 0

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- current convolution and maximum index ----
        cur_max, k_max = _conv_max(cur, L)

        # ---- gradient at the active (max) index ----
        grad = _gradient_at_max(cur, k_max)

        # ---- momentum update ----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ---- exponentiated‑gradient candidate ----
        step = step0 / math.sqrt(it)
        exponent = -step * momentum
        exponent = np.clip(exponent, -50.0, 50.0)
        cand = cur * np.exp(exponent)
        cand = _normalize(cand)

        cand_max, _ = _conv_max(cand, L)

        # ---- acceptance test ----
        if cand_max < cur_max:
            cur = cand
            cur_max = cand_max
            no_improve = 0
            if cand_max < best_max:
                best_max = cand_max
                best_vec = cand.copy()
                best_score = 2.0 * n * best_max
        else:
            no_improve += 1
            # occasional uphill move (Metropolis style)
            prob = math.exp(-(cand_max - cur_max) / (step + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_max = cand_max

        # ---- periodic random mass‑transfer when stuck ----
        if no_improve > 2000:

    # --------------------------------------------------------
    return [float(x) for x in best_vec]
