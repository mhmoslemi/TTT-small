def _softmax_gradient(a, eps, L):
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    # compute softmax weights
    w = np.exp((conv - max_val) / eps)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]
    return g, max_val

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: projection onto the probability simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Helper: compute the AC1 bound (2 * n * max(conv)) for a vector
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(convolution(a, a)).  Assumes sum(a)=1."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Helper: gradient of the smoothed max (soft‑max) objective
# ------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, eps: float, L: int):
    """
    Compute gradient of  f(a)=log-sum-exp(conv(a,a)/eps)  w.r.t. a.
    Returns (gradient, max_conv).
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L)

    # full convolution a*a
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()

    # soft‑max weights
    w = np.exp((conv - max_val) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= w_sum

    # gradient via convolution a * w_rev
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)

    idx = (2 * n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]
    return g, max_val

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence with a low AC1 bound.
    The routine runs for up to `budget_s` seconds and returns the best
    sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----------------------------------------------------------------
    # Initialise – use the parent if present, otherwise a uniform start
    # ----------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = np.full(1024, 1.0 / 1024, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # enforce sum = 1
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())  # FFT length (power of two)

    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc

    # ------------------------------------------------------------
    # Adam‑style optimisation parameters
    # ------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.2                     # base learning rate
    eps0, eps_decay = 0.5, 0.9995
    eps_min = 1e-8

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)

    it = 0
    target = 1.50274349

    # ----------------------------------------------------------------
    # Phase 1 – gradient‑based optimisation (≈70 % of the budget)
    # ----------------------------------------------------------------
    grad_time = start + 0.7 * (deadline - start)

    while True:
        now = time.monotonic()
        if now >= deadline:
            break
        it += 1

        eps = max(eps_min, eps0 * (eps_decay ** it))
        g, _ = _softmax_gradient(cur, eps, L)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 / math.sqrt(it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step
        cand = _project_simplex(cand)
        cand_sc = _score(cand)

        # simple accept‑reject with a tiny Metropolis kick
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc

        # occasional symmetrisation (helps with the Toeplitz structure)
        if it % 300 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        # small random mass‑transfer with probability 0.02
        if rng.random() < 0.02:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

        # exit gradient phase when its allocated time expires
        if now >= grad_time:
            break

        # early stop if we have already reached the target
        if best_sc < target:
            break

    # ------------------------------------------------------------
    # Phase 2 – intensive random local search (remaining budget)
    # ------------------------------------------------------------
    cur = best_vec.copy()
    cur_sc = best_sc

    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        if cur[i] <= 0.0:
            continue
        delta = cur[i] * rng.random() * 0.5
        j = rng.integers(0, n)
        if j == i:
            continue
        tmp = cur.copy()
        tmp[i] -= delta
        tmp[j] += delta
        tmp = _project_simplex(tmp)
        tmp_sc = _score(tmp)

        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            if tmp_sc < best_sc:
                best_vec, best_sc = tmp.copy(), tmp_sc
        else:
            # simulated‑annealing acceptance
            remaining = (deadline - time.monotonic()) / (deadline - start)
            T = remaining * 0.1          # temperature ≤ 0.1
            prob = math.exp(-(tmp_sc - cur_sc) / (T + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = tmp, tmp_sc

        # occasional symmetrisation during the annealing phase
        if rng.random() < 0.01:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc

        if best_sc < target:
            break

    # ------------------------------------------------------------
    # Return the best candidate as a plain Python list
    # ------------------------------------------------------------
    return [float(x) for x in best_vec]
