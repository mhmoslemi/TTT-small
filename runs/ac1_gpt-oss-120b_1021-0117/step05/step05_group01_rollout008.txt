def _project_simplex(v, z=1.0):

    # Works for v length n, returns w s.t. w>=0, sum w = z

    # Based on algorithm by Duchi et al.

    n = v.shape[0]

    if n == 0:

        theta = 0.0

        rho = rho[-1]

        theta = css[rho] / (rho + 1)

    w = np.maximum(v - theta, 0.0)

    return w

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=z}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)
