import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score used by the evaluator – works for any positive sum
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return the evaluator's objective for a given sequence."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))

        g = 2.0 * grad_full[idx]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- candidate 1 : momentum step -----
        alpha = step0 / math.sqrt(it)
        cand1 = cur - alpha * momentum
        cand1 = _project_simplex(cand1)
        cand1_sc = _score(cand1)

        # ----- candidate 2 : mirror‑descent step -----
        eta = step0_md / math.sqrt(it)
        exp_arg = -eta * g
        np.clip(exp_arg, -50, 50, out=exp_arg)
        cand2 = cur * np.exp(exp_arg)
        s = cand2.sum()
        if s > 0:
            cand2 /= s
        else:
            cand2 = cur.copy()
        cand2_sc = _score(cand2)

        # ----- pick the better candidate -----
        if cand1_sc <= cand2_sc:
            cand, cand_sc = cand1, cand1_sc
        else:
            cand, cand_sc = cand2, cand2_sc

        # ----- accept / reject (occasional uphill moves) -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- periodic symmetrisation (helps with Toeplitz structure) -----
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # --------------------------------------------------------
    # Final local refinement (simple hill‑climbing)
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        if best_vec[i] <= 0.0:
            continue
        delta = best_vec[i] * rng.random() * 0.1
        j = rng.integers(0, n - 1)
        if j == i:
            continue
        cand = best_vec.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        cand_sc = _score(cand)
        if cand_sc < best_sc:
            best_vec, best_sc = cand, cand_sc

    # --------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
