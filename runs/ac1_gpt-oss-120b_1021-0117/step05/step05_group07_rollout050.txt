def _project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]

        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (non‑negative, sum = 1)."""

        rho = rho[-1]
        theta = css[rho] / (rho + 1)
