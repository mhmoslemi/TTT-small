def _project_simplex(v):
    v = np.maximum(v, 0.0)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
```

def _score(a, L=None):
    n = a.shape[0]

    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2000
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)
    # Ensure sum exactly 1
    cur /= cur.sum()
    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    # adam states
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_lr = 0.5
    eps0 = 0.5   # smoothing epsilon init
    eps_decay = 0.9995
    eps_min = 1e-8

    best = cur.copy()
    best_sc = _score(best, L)

    cur_sc = best_sc

    it = 0
    no_improve = 0
    target_sc = 1.50274349

    while time.monotonic() < deadline:
        it += 1
        # Compute convolution via FFT
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()

        # softmax weights
        eps_cur = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_conv) / eps_cur)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient computation
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = base_lr / np.sqrt(it) * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = cur - step
        cand = _project_simplex(cand)
        cand /= cand.sum()
        cand_sc = _score(cand, L)

        # Accept if improvement
        if cand_sc < cur_sc:
            cur = cand
            cur_sc = cand_sc
            no_improve = 0
        else:
            # occasional stochastic acceptance
            prob = math.exp(-(cand_sc - cur_sc) / (base_lr + 1e-12))
            if rng.random() < prob:
                cur = cand
                cur_sc = cand_sc
            no_improve += 1

        if cand_sc < best_sc:
            best = cand.copy()
            best_sc = cand_sc

        # Random mass transfer if stuck
        if no_improve > 2000:
            i = rng.integers(0, n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp /= tmp.sum()
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    no_improve = 0
                if tmp_sc < best_sc:
                    best = tmp.copy()
                    best_sc = tmp_sc

        # periodic symmetrisation
        if it % 500 == 0:

            sym_sc = _score(sym, L)

                no_improve = 0
            if sym_sc < best_sc:
                best = sym.copy()
                best_sc = sym_sc

        if best_sc < target_sc:
            break

    # Return best as list of floats
    return [float(x) for x in best]

import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection – ensures non‑negative entries with sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    # make a copy to avoid side‑effects
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, u.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # numerical safety – re‑normalise exactly to 1
    s = w.sum()
    if s > 0:
        w /= s
    else:  # fallback – uniform distribution
        w[:] = 1.0 / w.size
    return w


# ------------------------------------------------------------
# Score (objective) for a vector with sum == 1
# ------------------------------------------------------------
def _score(a: np.ndarray, fft_len: int) -> float:
    """Return 2 * n * max(convolution(a, a)) assuming sum(a)=1."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence with a low AC1 bound.
    The function runs for up to `budget_s` seconds and returns the best
    sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initialise – use the parent if present, otherwise a uniform start
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # a modest default length – will be overwritten after reading n
        cur = np.full(2048, 1.0 / 2048, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)          # enforce sum = 1 (exact)
    n = cur.size

    # --------------------------------------------------------
    # FFT planning – power of two, at least 2n‑1 points
    # --------------------------------------------------------
    fft_len = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------
    # Optimiser state (Adam) and hyper‑parameters
    # --------------------------------------------------------
    m = np.zeros_like(cur)               # first moment
    v = np.zeros_like(cur)               # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    base_lr = 0.5                        # base learning rate for Adam
    base_lr_mirror = 0.5                 # base rate for mirror‑step
    eps0 = 0.5                           # initial smoothing for soft‑max
    eps_decay = 0.9995
    eps_min = 1e-8

    best_vec = cur.copy()
    best_sc = _score(best_vec, fft_len)
    cur_sc = best_sc

    it = 0
    no_improve = 0
    target_sc = 1.50274349

    # --------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution and soft‑max weights -----
        a_fft = np.fft.rfft(cur, n=fft_len)
        conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()

        eps_cur = max(eps_min, eps0 * (eps_decay ** it))
        exp_arg = (conv - max_conv) / eps_cur
        # Clamp to avoid under‑/overflow in exp
        np.clip(exp_arg, -50, 0, out=exp_arg)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=fft_len)
        grad_full = np.fft.irfft(a_fft * w_fft, n=fft_len)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # ----- Adam (first) candidate -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_lr / math.sqrt(it) * m_hat / (np.sqrt(v_hat) + eps_adam)
        cand1 = cur - step
        cand1 = _project_simplex(cand1)
        cand1_sc = _score(cand1, fft_len)

        # ----- Mirror‑descent (exponential) candidate -----
        eta = base_lr_mirror / math.sqrt(it)
        exp_arg_m = -eta * grad
        np.clip(exp_arg_m, -50, 50, out=exp_arg_m)
        cand2 = cur * np.exp(exp_arg_m)
        cand2 = _project_simplex(cand2)
        cand2_sc = _score(cand2, fft_len)

        # ----- Choose the better of the two -----
        if cand1_sc <= cand2_sc:
            cand, cand_sc = cand1, cand1_sc
        else:
            cand, cand_sc = cand2, cand2_sc

        # ----- Acceptance policy (always accept improvement) -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            # occasional uphill move (Metropolis‑style)
            prob = math.exp(-(cand_sc - cur_sc) / (base_lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- Random mass‑transfer when stuck -----
        if no_improve > 2500:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, fft_len)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                    no_improve = 0
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc

        # ----- Periodic symmetrisation (helps with the Toeplitz structure) -----
        if it % 400 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, fft_len)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                no_improve = 0
            if sym_sc < best_sc:
                best_vec, best_sc = sym.copy(), sym_sc

        # ----- Early exit if target reached -----
        if best_sc < target_sc:
            break

    # --------------------------------------------------
    # Return the best candidate found (plain Python list)
    # --------------------------------------------------
    return [float(x) for x in best_vec]
