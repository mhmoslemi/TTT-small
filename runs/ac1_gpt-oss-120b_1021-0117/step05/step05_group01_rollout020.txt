def _smooth_max_grad(a, alpha, L1, L2):
    n = a.size
    # Compute convolution a*a
    a_fft = np.fft.rfft(a, n=L1)
    conv = np.fft.irfft(a_fft * a_fft, n=L1)[:2*n-1]

# Euclidean projection onto the simplex {x >= 0, sum(x)=z}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex."""
    if z < 0:
        raise ValueError("z must be non‑negative")
    n = v.size
    # sort v descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Maximum of the (linear) convolution a*a (full length)
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    """Return max_k (a*a)[k] using an FFT of length L (>=2n‑1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())


# ------------------------------------------------------------
# Gradient of a smooth approximation of max(conv) via softmax.
# ------------------------------------------------------------
def _smooth_max_grad(a: np.ndarray, alpha: float, L1: int, L2: int):
    """
    Returns
        grad  – ∇_a  (softmax‑approximated max of a*a)
        mmax – the true max(a*a) (for scoring)
    """
    n = a.size

    # ----- a*a ------------------------------------------------
    a_fft = np.fft.rfft(a, n=L1)
    conv = np.fft.irfft(a_fft * a_fft, n=L1)[: 2 * n - 1]

    # ----- softmax weights over conv -------------------------
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_vals = np.exp(scaled - max_scaled)
    w = exp_vals / np.sum(exp_vals)            # size 2n‑1, sums to 1

    # ----- weighted term  sum_i w_i a_{i-j} ------------------
    a_rev = a[::-1]                            # reversed a
    w_fft = np.fft.rfft(w, n=L2)
    a_rev_fft = np.fft.rfft(a_rev, n=L2)
    wa = np.fft.irfft(w_fft * a_rev_fft, n=L2)
    # keep only the needed part (length 3n‑2)
    needed_len = (2 * n - 1) + n - 1          # = 3n‑2
    wa = wa[:needed_len]

    start = n - 1
    weighted = wa[start:start + n]            # size n

    # ----- gradient of smooth max -----------------------------
    grad = weighted + a                       # (a appears from d conv_i / d a_j)

    return grad, float(conv.max())


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ---- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # modest random start if no parent is supplied
        a = rng.random(2000, dtype=np.float64)

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                                   # enforce sum = 1

    n = a.size
    L1 = 1 << ((2 * n - 1).bit_length())               # for a*a
    L2 = 1 << ((3 * n - 2).bit_length())               # for w * a_rev

    # ---- bookkeeping ------------------------------------------------
    best_vec = a.copy()
    cur_max = _conv_max(a, L1)
    best_sc = 2.0 * n * cur_max

    momentum = np.zeros_like(a)
    beta = 0.9
    base_step = 0.7
    no_improve = 0
    it = 0
    target = 1.50274349

    alpha0 = 5.0          # start smoothing
    alpha_max = 30.0
    alpha_inc = 0.018    # per iteration

    # ---- optimisation loop -----------------------------------------
    while time.monotonic() < deadline:
        it += 1
        alpha = min(alpha_max, alpha0 + alpha_inc * it)

        # gradient of smooth max and current true max
        grad_smooth, true_max = _smooth_max_grad(a, alpha, L1, L2)

        # current score (sum is 1 → denominator = 1)
        cur_sc = 2.0 * n * true_max

        # keep global best
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = a.copy()
            no_improve = 0

        # stop if target reached
        if best_sc < target:
            break

        # momentum update (using smooth‑max gradient)
        momentum = beta * momentum + (1.0 - beta) * grad_smooth

        # back‑tracking step
        step = base_step / math.sqrt(it)
        cand = None
        cand_sc = None

        for _ in range(7):
            a_tmp = a - step * momentum
            a_tmp = _project_simplex(a_tmp)
            cand_max = _conv_max(a_tmp, L1)
            cand_sc = 2.0 * n * cand_max
            if cand_sc <= cur_sc:
                cand, cand_sc = a_tmp, cand_sc
                break
            step *= 0.5

        if cand is None:                     # no descent found
            cand, cand_sc = a_tmp, cand_sc

        # accept / reject
        if cand_sc < cur_sc:
            a = cand
            cur_sc = cand_sc
            no_improve = 0
        else:
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                a = cand
                cur_sc = cand_sc
                no_improve = 0
            else:
                no_improve += 1

        # ----- periodic symmetrisation ----------
        if it % 250 == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            sym_max = _conv_max(a_sym, L1)
            sym_sc = 2.0 * n * sym_max
            if sym_sc < cur_sc:
                a = a_sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = a_sym.copy()
                no_improve = 0

        # ----- random mass‑transfer when stuck ----------
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.35
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
                # recompute score after the jump
                cur_max = _conv_max(a, L1)
                cur_sc = 2.0 * n * cur_max
            no_improve = 0

    # -----------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # -----------------------------------------------------------------
    return [float(x) for x in best_vec]
