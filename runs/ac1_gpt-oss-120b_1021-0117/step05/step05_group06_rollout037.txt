import time
import math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex (sum = 1, x_i >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Maximum value of the (linear) convolution a * a (full)
# ------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv_full.max())


# ------------------------------------------------------------
# Geometric (exponential) sequence, sum normalized to 1
# ------------------------------------------------------------
def _geo_seq(n: int, delta: float) -> np.ndarray:
    """delta = 1 - q, 0 < delta < 1.  Returns a non‑negative simplex vector."""
    if delta <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    q = 1.0 - delta
    # compute q**i safely
    i = np.arange(n, dtype=np.float64)
    log_q = math.log(q)
    q_pow = np.exp(i * log_q)
    q_n = math.exp(n * log_q)
    denom = 1.0 - q_n
    if denom == 0.0:                     # fallback – uniform
        return np.full(n, 1.0 / n, dtype=np.float64)
    a = delta * q_pow / denom
    return a
