import time, math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    cond = u - css / np.arange(1, n + 1) > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Fast score (same as evaluate_sequence for sum=1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2 * n * max(convolution(a, a))  (a already normalized)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence with a low AC1 bound.
    Runs for up to `budget_s` seconds and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # --------------------------------------------------------
    # Initialise – use the parent if present, otherwise uniform
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = np.full(1024, 1.0 / 1024, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)            # enforce sum = 1
    n = cur.size

    # --------------------------------------------------------
    # FFT planning – length ≥ 3n‑2, power of two (for gradient step)
    # --------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------
    # Adam optimizer state
    # --------------------------------------------------------
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps_opt = 1e-8
    base_lr = 0.5   # base learning rate for Adam

    # --------------------------------------------------------
    # Soft‑max smoothing for max‑approximation
    # --------------------------------------------------------
    eps0 = 1.0
    eps_decay = 0.9995
    eps_min = 1e-8

    # --------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------
    best_vec = cur.copy()
    best_sc = _score(best_vec)
    cur_sc = best_sc
    it = 0
    no_improve = 0
    target = 1.50274349

    # --------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution (for gradient) -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- soft‑max weights approximating the max -----
        eps = max(eps_min, eps0 * (eps_decay ** it))
        w = np.exp((conv - max_val) / eps)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr_t = base_lr / math.sqrt(it)
        step = lr_t * m_hat / (np.sqrt(v_hat) + eps_opt)
        cand_adam = cur - step
        cand_adam = _project_simplex(cand_adam)
        cand_adam_sc = _score(cand_adam)

        # ----- mirror‑descent candidate -----
        eta = 0.6 / math.sqrt(it)
        exp_arg = -eta * g
        np.clip(exp_arg, -50, 50, out=exp_arg)
        cand_mirror = cur * np.exp(exp_arg)
        s = cand_mirror.sum()
        if s > 0:
            cand_mirror /= s
        cand_mirror_sc = _score(cand_mirror)

        # ----- random‑perturbation candidate (simulated‑annealing style) -----
        sigma = max(0.05, 0.5 * (0.999 ** it))
        cand_rand = cur + rng.normal(scale=sigma, size=n)
        cand_rand = _project_simplex(cand_rand)
        cand_rand_sc = _score(cand_rand)

        # ----- select the best of the three candidates -----
        cand, cand_sc = min(
            [(cand_adam, cand_adam_sc),
             (cand_mirror, cand_mirror_sc),
             (cand_rand, cand_rand_sc)],
            key=lambda x: x[1]
        )

        # ----- accept / reject (with occasional uphill moves) -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            # Metropolis‑like acceptance using the current temperature
            prob = math.exp(-(cand_sc - cur_sc) / (sigma + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # ----- periodic symmetrisation (helps with Toeplitz structure) -----
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 3000:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
