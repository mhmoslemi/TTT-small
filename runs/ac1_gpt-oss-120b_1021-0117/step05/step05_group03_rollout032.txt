        af = np.fft.rfft(v, n=L)
        conv_full = np.fft.irfft(af * af, n=L)

        a_fft = np.fft.rfft(a, n=L)

        # convolution and max
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)

        shifted = conv - max_conv

import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho_arr = u - css / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score for a vector with sum = 1 (2 * n * max(conv))
# ------------------------------------------------------------
def _score_vec(v: np.ndarray, L: int) -> float:
    """Return 2 * n * max(convolution(v, v)) assuming sum(v) = 1."""
    n = v.shape[0]
    af = np.fft.rfft(v, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin so we can return before the hard timeout
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise – use the provided parent if available, otherwise random
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: a modest random start
        a = rng.random(2048, dtype=np.float64)

    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a[0] = 1.0
    a = _project_simplex(a)          # enforce sum = 1
    n = a.shape[0]

    # ------------------------------------------------------------
    # FFT length – must accommodate a * (softmax weight) convolution
    # ------------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())   # smallest power of two ≥ 3n‑2

    # ------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------
    best_vec = a.copy()
    best_sc = _score_vec(best_vec, L)

    # Adam optimiser state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2

    # temperature (relative to current max) for soft‑max weighting
    tau = 1.0          # start with a fairly smooth weight
    tau_decay = 0.9995

    it = 0
    # early‑exit target (the challenge goal)
    target = 1.50274349

    # ------------------------------------------------------------
    # Optimisation loop – matrix‑free, O(n log n) per iteration
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of a with itself -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()

        # ----- soft‑max weighting of convolution entries -----
        # Temperature proportional to the current peak, with decaying factor
        T = max(max_conv * max(tau, 1e-12), 1e-12)
        shifted = conv - max_conv          # for numerical stability
        exp_vals = np.exp(shifted / T)
        w = exp_vals / exp_vals.sum()      # length = 2n‑1, sums to 1

        # ----- gradient of the smoothed max (2 * a * w_rev) -----
        w_rev = w[::-1]                    # reverse for convolution
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)   # pick the right slice
        g = 2.0 * grad_full[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)     # stay non‑negative
        a_new = _project_simplex(a_new)    # enforce sum = 1

        cur_sc = _score_vec(a_new, L)

        # ----- keep the best solution seen so far -----
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = a_new.copy()

        a = a_new

        # ----- temperature decay -----
        tau *= tau_decay

        # ----- early exit if we have already beaten the target -----

    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
