import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x) = z}."""
    n = v.size
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Full convolution (via FFT) and gradient of the max entry
# ------------------------------------------------------------
def _conv_full_fft(a: np.ndarray, L: int) -> np.ndarray:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return conv


def _max_and_grad(a: np.ndarray, L: int):
    """Return (max_conv, gradient_of_max) for the current vector."""
    n = a.size
    conv = _conv_full_fft(a, L)
    cur_max = conv.max()
    # indices where the maximum is attained (allow tiny tolerance)
    eps = 1e-14
    max_idxs = np.where(np.abs(conv - cur_max) <= eps)[0]

    grad = np.zeros_like(a)
    for k in max_idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        grad[i] += 2.0 * a[k - i]   # sub‑gradient of (a*a)[k]
    return cur_max, grad
