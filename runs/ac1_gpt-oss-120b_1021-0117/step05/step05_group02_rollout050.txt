import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative sum = 1
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Score used by the evaluator – assumes sum(a)=1
# ------------------------------------------------------------
def _score(a: np.ndarray, L: int) -> float:
    """Return 2 * n * max(convolution(a, a))."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty non‑negative sequence with a low AC1 bound.
    The function runs for up to `budget_s` seconds and returns the best
    sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initialise – use the parent if present, otherwise a uniform start
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = np.full(1024, 1.0 / 1024, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)            # enforce sum = 1
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())    # FFT length (power of two)

    # --------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------
    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    # --------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------
    step0 = 0.45           # base step for sub‑gradient descent
    step0_mirror = 0.30    # base step for mirror‑descent update
    target = 1.50274349
    no_improve = 0
    it = 0

    # --------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- current convolution -----
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # ----- active shifts (sub‑gradient of max) -----
        # tolerance guards against floating‑point noise
        tol = 1e-12
        active = np.where(conv >= max_val - tol)[0]

        # ----- sub‑gradient of the max convolution -----
        g = np.zeros_like(cur)
        for k in active:
            if k == 0:
                # derivative of a_i^2 term → 2*a_i
                g += 2.0 * cur
            else:
                # contributions from a_{j-k} and a_{j+k}, each counted twice
                contrib = np.zeros_like(cur)
                # a_{j-k} part (shift right by k)
                contrib[k:] = cur[: n - k]
                # a_{j+k} part (shift left by k)
                contrib[: n - k] += cur[k:]
                g += 2.0 * contrib

        # ----------------------------------------------------
        # Candidate 1 : sub‑gradient descent with back‑tracking
        # ----------------------------------------------------
        alpha = step0 / math.sqrt(it)
        cand1 = cur - alpha * g
        cand1 = _project_simplex(cand1)
        cand1_sc = _score(cand1, L)

        # back‑track if no improvement
        if cand1_sc > cur_sc:

                alpha *= 0.5
                tmp = cur - alpha * g
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L)
                if tmp_sc <= cur_sc:
                    cand1, cand1_sc = tmp, tmp_sc
                    break
            else:
                cand1, cand1_sc = tmp, tmp_sc

        # ----------------------------------------------------
        # Candidate 2 : mirror‑descent (multiplicative) step
        # ----------------------------------------------------
        eta = step0_mirror / math.sqrt(it)
        exp_arg = -eta * g

        cand2 = cur * np.exp(exp_arg)
        s = cand2.sum()
        if s > 0:
            cand2 /= s
        else:
            cand2 = cur.copy()
        cand2_sc = _score(cand2, L)

        # ----- pick the best of the two candidates -----
        if cand1_sc <= cand2_sc:

        else:
            cand, cand_sc = cand2, cand2_sc

        if cand_sc < cur_sc:

            no_improve = 0

            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:

            no_improve += 1

        if it % 300 == 0:

            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)

                no_improve = 0

        # ----- random mass‑transfer when stuck -----
        if no_improve > 2500:

            if cur[i] > 0.0:

                j = rng.integers(0, n - 1)

                    j += 1
                tmp = cur.copy()

                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp, L)
                if tmp_sc < cur_sc:
                    cur, cur_sc = tmp, tmp_sc
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
                no_improve = 0

        # ----- early exit if target reached -----
        if best_sc < target:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
