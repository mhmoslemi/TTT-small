import time
import math
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex {x >= 0, sum(x)=z}."""
    if z < 0:
        raise ValueError("z must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Full convolution a*a via FFT (returns length 2n‑1)
# ------------------------------------------------------------
def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return conv_full[: 2 * a.size - 1]

# ------------------------------------------------------------
# Exact sub‑gradient of max{a*a}
# ------------------------------------------------------------
def _grad_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    n = a.size
    max_val = conv.max()
    # tolerance to capture all essentially‑maximal entries
    eps = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - eps)[0]
    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        seg = a[i_start : i_end + 1]
        g[i_start : i_end + 1] += 2.0 * seg[::-1]
    if active.size:
        g /= active.size
    return g
