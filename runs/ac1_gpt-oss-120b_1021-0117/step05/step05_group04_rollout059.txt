def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale to have sum = 1 (preserves non‑negativity)."""
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _conv_max(a: np.ndarray, L: int):
    """Return max of a*a (full linear convolution) together with its FFT."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv.max(), a_fft, conv


def _subgrad_max(a: np.ndarray, conv: np.ndarray, max_val: float, tol: float = 1e-12):
    """
    Exact sub‑gradient of the max‑convolution functional.
    For each index k where conv[k] is (near) maximal we add
        2 * a_{k‑j}  to the j‑th component.
    """
    n = a.size
    # indices where the maximum is attained (within tolerance)
    idxs = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in idxs:
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        j = np.arange(start, end + 1, dtype=np.int64)
        grad[j] += 2.0 * a[k - j]
    return grad
