import time
import math
import numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the simplex {x >= 0, sum(x)=1}
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise exactly to 1
    s = w.sum()
    if s == 0:
        return np.full_like(v, 1.0 / n)
    return w / s


# ------------------------------------------------------------
# Full convolution via FFT (real inputs)
# ------------------------------------------------------------
def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)
    return conv[: 2 * a.size - 1]


# ------------------------------------------------------------
# Gradient of the convolution entry at given index (or indices)
# ------------------------------------------------------------
def _grad_at_idxs(a: np.ndarray, idxs) -> np.ndarray:
    """Return the average gradient of conv[idx] for idx in idxs."""
    n = a.size
    g = np.zeros_like(a)
    for idx in idxs:
        d = idx - (n - 1)                # lag value
        if d >= 0:
            # a[i+d] term
            g[: n - d] += a[d:]
            # a[i-d] term
            g[d:] += a[: n - d]
        else:
            off = -d
            # a[i+d] term (i+d >=0)
            g[off:] += a[: n - off]
            # a[i-d] term (i-d < n)
            g[: n - off] += a[off:]
    g /= len(idxs)
    return g
