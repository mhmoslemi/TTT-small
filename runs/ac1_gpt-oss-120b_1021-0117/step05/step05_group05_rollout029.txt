    deadline = start + max(0.0, budget_s - 0.5)  # keep 0.5 sec headroom

    n = cur.shape[0]
    L = 1 << ((2 * n - 1).bit_length())

    def score_vec(v):
        # compute max convolution
        fv = np.fft.rfft(v, n=L)
        conv_full = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
        return float(conv_full.max())

    best = cur.copy()
    best_sc = score_vec(best)
    cur_sc = best_sc

    momentum = np.zeros_like(cur)
    beta = 0.9
    step0 = 0.1  # smaller initial step maybe
    it = 0
    no_improve = 0

    eps = 1e-12

    while time.monotonic() < deadline:
        it += 1

        # compute convolution and gradient of max
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        # mask of max positions
        mask = conv_full >= max_val - eps
        w = mask.astype(np.float64)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt

        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1 - beta) * grad

        # step size schedule
        step = step0 / math.sqrt(it)

        # try decreasing step if not improving
        cand = None
        cand_sc = None
        for _ in range(4):
            cand_tmp = cur - step * momentum
            cand_tmp = _project_simplex(cand_tmp)
            cand_tmp_sc = score_vec(cand_tmp)
            if cand_tmp_sc < cur_sc:
                cand = cand_tmp
                cand_sc = cand_tmp_sc
                break
            step *= 0.5

        if cand is None:
            # no improvement from gradient, try random perturb
            i = rng.integers(n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.5
                j = rng.integers(n)
                if j == i:
                    j = (j + 1) % n
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                cand = tmp
                cand_sc = score_vec(cand)
            else:
                cand = cur
                cand_sc = cur_sc

        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best, best_sc = cand.copy(), cand_sc
        else:
            # occasional acceptance of worse solution (simulated annealing)
            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:

            no_improve = 0

    return [float(x) for x in best]

import time

import numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _score(vec, L=None):
    """Return 2*n*max(conv(vec,vec)).  The vector is assumed to sum to 1."""
    n = vec.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    fv = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fv * fv, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

            prob = math.exp(-(cand_sc - cur_sc) / (step + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            no_improve += 1

        # if stuck for a while, randomise a bit to escape local minima
        if no_improve > 2000 and time.monotonic() + 1.0 < deadline:
            perm = rng.permutation(n)
            cur = cur[perm]
            cur += 0.001 * rng.random(n)
            cur = _project_simplex(cur)
            cur_sc = _score(cur, L)
            no_improve = 0

        # early termination when the target is reached
        if best_sc <= target:
            break

    # return the best candidate found
    return [float(v) for v in best]
```
