import time, math, numpy as np

def _project_simplex(v):
    """Project onto the probability simplex (non‑negative with sum = 1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _score(a):
    """Fast evaluation of the AC1 bound (same formula as evaluate_sequence)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = a.sum()
    if s == 0:
        return np.inf
    return float(2.0 * n * max_val / (s * s))

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a small safety margin for the return path
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ---------------------------------------------------------------------

    if "height_sequence_1" in globals():

    cur = _project_simplex(a0)
    n = cur.size
