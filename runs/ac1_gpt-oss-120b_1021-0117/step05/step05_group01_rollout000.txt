import time, math, numpy as np

# ------------------------------------------------------------
# Simplex projection (sum = z, here z = 1)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=z}."""
    if z < 0:
        raise ValueError("z must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Max of a*a (full convolution) using FFT
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

# ------------------------------------------------------------
# Sub‑gradient using the top‑k largest convolution entries
# ------------------------------------------------------------
def _grad_topk(a: np.ndarray, conv: np.ndarray, top_k: int) -> np.ndarray:
    n = a.size
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    g = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            i = np.arange(i_start, i_end + 1, dtype=int)
            g[i] += 2.0 * a[k - i]
    return g / max(1, top_k)

                break
        if cand is None:          # no descent found, keep the smallest trial
            cand, cand_sc = a_tmp, cand_sc

        # accept if better
        if cand_sc < cur_sc:
            a, cur_sc = cand, cand_sc
            no_improve = 0
            if cand_sc < best_sc:
                best_sc, best_vec = cand_sc, cand.copy()
        else:
            # occasional uphill acceptance (small probability)
            prob = math.exp(-(cand_sc - cur_sc) / (alpha + 1e-12))
            if rng.random() < prob:
                a, cur_sc = cand, cand_sc

        # ----- periodic symmetrisation + smoothing -----
        if t % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            a_sym_fft = np.fft.rfft(a_sym, n=L)
            conv_sym = np.fft.irfft(a_sym_fft * a_sym_fft, n=L)[: 2 * n - 1]
            sym_sc = 2.0 * n * conv_sym.max()
            if sym_sc < cur_sc:
                a, cur_sc = a_sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, a_sym.copy()
                    no_improve = 0
            # simple smoothing (averaging neighbours)
            a_smooth = np.convolve(a, np.array([0.5, 0.5]), mode='same')
            a_smooth = _project_simplex(a_smooth)
            a_smooth_fft = np.fft.rfft(a_smooth, n=L)
            conv_smooth = np.fft.irfft(a_smooth_fft * a_smooth_fft, n=L)[: 2 * n - 1]
            smooth_sc = 2.0 * n * conv_smooth.max()
            if smooth_sc < cur_sc:
                a, cur_sc = a_smooth, smooth_sc
                if smooth_sc < best_sc:
                    best_sc, best_vec = smooth_sc, a_smooth.copy()
                    no_improve = 0

        # ----- random mass‑transfer when stagnated -----
        if no_improve > 2500:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.4
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
                no_improve = 0

        # ----- tiny local perturbations nearer deadline -----
        if deadline - time.monotonic() < 15.0:
            # try a few cheap random moves
            for _ in range(200):
                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                if a[i] <= 0.0:
                    continue
                delta = a[i] * rng.random() * 0.05
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()
                # evaluate
                a_fft = np.fft.rfft(a, n=L)
                conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
                sc = 2.0 * n * conv.max()
                if sc < best_sc:
                    best_sc, best_vec = sc, a.copy()
                # keep the move if it improves current
                if sc < cur_sc:
                    cur_sc = sc
                else:
                    # revert
                    a[j] -= delta
                    a[i] += delta
                    a = a / a.sum()
            break   # exit main loop after final sweep

    # ------------------------------------------------------------
    # Return best observed candidate (as plain Python list)
    # ------------------------------------------------------------
    return [float(x) for x in best_vec]
