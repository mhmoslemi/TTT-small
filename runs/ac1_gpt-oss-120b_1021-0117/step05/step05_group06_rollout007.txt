def _anti_identity(arr):
    vals = np.sort(arr)[::-1]
    n = vals.shape[0]
    perm = np.empty_like(vals)
    left = 0
    right = n-1
    k = 0
    while left <= right:
        perm[left] = vals[k]; k+=1
        if left < right:
            perm[right] = vals[k]; k+=1
        left+=1
        right-=1
    return perm

# ----------------------------------------------------------------------
# Project onto the probability simplex (sum = 1, x_i >= 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=1}."""
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# ----------------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    """Return max value of the full convolution a*a using an FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv_full.max())

def _anti_identity(arr: np.ndarray) -> np.ndarray:
    """Reorder arr so that sorted values are placed alternating at the ends."""

    while left <= right:
        out[left] = vals[k]; k += 1
        if left < right:
            out[right] = vals[k]; k += 1
        left += 1
        right -= 1
    return out
