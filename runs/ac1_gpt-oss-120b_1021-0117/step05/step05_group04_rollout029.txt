import time
import math
import numpy as np

def _max_conv(v: np.ndarray, L: int):
    """Return (max convolution value, FFT of v, full convolution array)."""
    v_fft = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(v_fft * v_fft, n=L)[: 2 * v.size - 1]
    return float(conv.max()), v_fft, conv

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty list of non‑negative coefficients that (approximately)
    minimize 2·n·max(a*a)/sum(a)**2.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --- initial vector -------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        cur = rng.random(2048, dtype=np.float64)
    cur = np.maximum(cur, 0.0)
    s = cur.sum()
    if s == 0.0:
        cur[:] = 1.0
        s = cur.sum()
    cur = cur / s

    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())       # power‑of‑two ≥ 2n‑1

    cur_max, _, cur_conv = _max_conv(cur, L)
    best_max = cur_max
    best_vec = cur.copy()

    # sub‑gradient descent parameters
    momentum = np.zeros_like(cur)
    beta = 0.9
    step0 = 0.2
    it = 0
    target_score = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # ---- sub‑gradient of the current maximal convolution entry ----
        kmax = int(np.argmax(cur_conv))
        lo = max(0, kmax - (n - 1))
        hi = min(n - 1, kmax)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(cur)
        grad[idx] = 2.0 * cur[kmax - idx]

        # ---- momentum update -----------------------------------------
        momentum = beta * momentum + (1.0 - beta) * grad

        # ---- exponentiated‑gradient candidate -------------------------
        step = step0 / math.sqrt(it)
        exponent = -step * momentum
        exponent = np.clip(exponent, -50.0, 50.0)
        cand = cur * np.exp(exponent)
        cs = cand.sum()
        cand = cand / cs if cs != 0.0 else np.full_like(cand, 1.0 / n)

        cand_max, _, _ = _max_conv(cand, L)

        # ---- acceptance -----------------------------------------------
        if cand_max < cur_max:
            cur = cand
            cur_max = cand_max
            cur_max, _, cur_conv = _max_conv(cur, L)
            if cur_max < best_max:
                best_max = cur_max
                best_vec = cur.copy()
        else:
            # occasional random jump to escape flat regions
            if rng.random() < 0.001:
                cur = cand
                cur_max = cand_max
                cur_max, _, cur_conv = _max_conv(cur, L)

        # ---- periodic symmetrisation ----------------------------------
        if it % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = sym / sym.sum()
            sym_max, _, _ = _max_conv(sym, L)
            if sym_max < cur_max:
                cur = sym
                cur_max = sym_max
                cur_max, _, cur_conv = _max_conv(cur, L)
                if cur_max < best_max:
                    best_max = cur_max
                    best_vec = cur.copy()

        # ---- occasional small random mass transfer --------------------
        if it % 2000 == 0:
            i = rng.integers(0, n)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.05
                j = rng.integers(0, n)
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = tmp / tmp.sum()
                tmp_max, _, _ = _max_conv(tmp, L)

                    cur_max = tmp_max

                        best_max = cur_max
                        best_vec = cur.copy()

        if 2.0 * n * best_max < target_score:
            break

    return [float(x) for x in best_vec]
