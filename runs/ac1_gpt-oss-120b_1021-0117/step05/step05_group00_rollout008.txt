def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.1) # margin

    # init a
    if 'height_sequence_1' in globals():
         a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
         n_small=2000; a0 = np.exp(-np.arange(n_small)/100.0)
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s <= 0:
         a0 = np.full_like(a0, 1.0/a0.size); s=1
    a0 = a0 / s

    n = a0.size
    L = 1 << ((2*n-1).bit_length())
    cur = a0.copy()
    cur_sc = _score(cur)
    best_vec = cur.copy()
    best_sc = cur_sc

    momentum = np.zeros_like(cur)
    beta = 0.9
    base_alpha = 0.5
    eps = None
    eps_decay = 0.97
    no_improve = 0
    sym_interval = 300
    max_trials = 7
    # Additional thresholds.
    target = 1.50274349
    it = 0

    while time.monotonic() < deadline:
        it += 1
        # Convolution
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]
        max_val = conv_full.max()
        # epsilon
        if eps is None:
            eps = max_val * 0.15 + 1e-12
        else:
            eps = max(eps * eps_decay, max_val * 1e-12)
        # determine weighting
        if eps < max_val * 1e-6 or it > 3000:
            # hard max: indicator
            mask = (conv_full == max_val)
            w = mask.astype(np.float64)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(conv_full, 1.0/conv_full.size)
            else:
                w /= w_sum
        else:
            w = np.exp((conv_full - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(conv_full, 1.0/w.size)
            else:
                w /= w_sum
        # gradient compute
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size
        norm_m = np.linalg.norm(momentum) + 1e-12
        alpha = base_alpha / (np.sqrt(it) * norm_m)

        # line search
        improved = False
        cand = None
        cand_sc = None
        a_alpha = alpha
        for _ in range(max_trials):
            cand_try = cur - a_alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            if cand_sc < cur_sc:
                cand = cand_try
                improved = True
                break
            a_alpha *= 0.5
        # if still not improved, allow uphill with prob
        if not improved:
            cand_try = cur - a_alpha * momentum
            cand_try = _project_simplex(cand_try)
            cand_sc = _score(cand_try)
            # compute acceptance probability based on energy difference and iteration
            temp = max(1e-3, 1.0 / np.sqrt(it))
            prob = math.exp(-(cand_sc - cur_sc) / temp)
            if rng.random() < prob:
                cand = cand_try
                improved = True

        if improved:
            cur = cand
            cur_sc = cand_sc
            if cand_sc < best_sc:
                best_sc = cand_sc
                best_vec = cand.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # symmetrize
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc = sym_sc
                    best_vec = sym.copy()
                no_improve = 0

        # random mass transfer
        if no_improve > 1000:
            i = rng.integers(0, n)
            if cur[i] > 0:
                delta = cur[i] * rng.random() * 0.3
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)
                tmp_sc = _score(tmp)
                if tmp_sc < cur_sc:
                    cur = tmp
                    cur_sc = tmp_sc
                    if tmp_sc < best_sc:
                        best_sc = tmp_sc
                        best_vec = tmp.copy()
                    no_improve = 0
                else:
                    # maybe accept small uphill less often
                    if rng.random() < 0.01:
                        cur = tmp
                        cur_sc = tmp_sc
                        no_improve = 0

        # restart
        if no_improve > 3000:
            cur = best_vec + 0.01 * rng.random(n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            momentum = np.zeros_like(cur)
            eps = None
            no_improve = 0

        if best_sc < target:
            break

        # Check time occasionally
        if time.monotonic() > deadline:
            break

    return [float(x) for x in best_vec]

import time, math
import numpy as np

# ------------------------------------------------------------
# Project onto the probability simplex (sum = 1, x >= 0)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Fast AC1 bound (assumes sum(a)≈1)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())
