def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # init vector
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = rng.integers(2000, 3000) # maybe 2500 to be similar
        a = rng.random(n, dtype=np.float64)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()  # sum =1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())
    # Pre-allocate gradient moving averages
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_vec = a.copy()
    best_sc = 2.0 * n * _conv_max(a, L)  # compute conv_max only once

    # Optimization parameters
    base_alpha = 0.2   # base learning rate
    top_k = 10
    stagnation = 0
    stagnation_limit = 2000
    sym_interval = 500
    # Multi-restart: we can store best over restarts.

    while time.monotonic() < deadline:
        t += 1
        # compute convolution
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        cur_max = conv.max()
        cur_sc = 2.0 * n * cur_max

        # update best
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            stagnation = 0
        else:
            stagnation += 1

        # early stopping
        if best_sc < 1.50274349:
            break

        # compute gradient from top_k largest entries
        # find top_k indices
        k_idxs = np.argpartition(-conv, top_k-1)[:top_k]
        # compute gradient per index
        g = np.zeros_like(a)
        for k in k_idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # a values for j = k-i
            # Use vectorized slice
            i_range = np.arange(i_start, i_end + 1, dtype=int)
            g[i_range] += 2.0 * a[k - i_range]
        g = g / top_k  # average

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = base_alpha / (np.sqrt(v_hat) + eps) * m_hat

        # update and project onto simplex
        a_new = a - step
        # ensure nonnegative and sum to one via projection
        a_new = _project_simplex(a_new)  # call function defined earlier that projects onto simplex with sum=1

        # compute new score for acceptance - maybe not needed
        a_fft_new = np.fft.rfft(a_new, n=L)
        conv_tmp = np.fft.irfft(a_fft_new * a_fft_new, n=L)[: 2 * n - 1]
        new_sc = 2.0 * n * conv_tmp.max()

        # accept if better, otherwise possibly accept with small probability:
        if new_sc <= cur_sc:
            a = a_new
        else:
            # Metropolis acceptance
            prob = math.exp(-(new_sc - cur_sc) / (base_alpha + 1e-12))
            if rng.random() < prob:
                a = a_new

        # periodic symmetrisation (optional)
        if t % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            # compute its score:
            a_fft_sym = np.fft.rfft(a_sym, n=L)
            conv_sym = np.fft.irfft(a_fft_sym * a_fft_sym, n=L)[: 2 * n - 1]
            sym_sc = 2.0 * n * conv_sym.max()
            if sym_sc < cur_sc:
                a = a_sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, a_sym.copy()
                    stagnation = 0

        # restart if stagnation high
        if stagnation > stagnation_limit:
            # reinitialize with random values but keep best fraction from best_vec to preserve progress.
            # Choose random mask
            mask = rng.random(n) < 0.1
            a = best_vec.copy()
            a[mask] = rng.random(mask.sum())
            a = np.maximum(a, 0.0)
            a /= a.sum()
            # reset momentums
            m[:] = 0.0
            v[:] = 0.0
            stagnation = 0

    # ensure best_sum > 0.01 (should be)
    # clip to [0,1000] final but evaluation will clip anyway.
    result = [float(x) for x in best_vec]
    # optionally ensure result has nonzero sum
    # evaluation expects sum != 0 (should be 1)
    return result

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum z."""
    n = v.size
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho_arr = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Maximum entry of a*a using an FFT (full convolution)
# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    """Return max_k (a*a)[k] using FFT of length L (L >= 2n‑1)."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # leave a safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------------
    # Initial vector – use provided parent if present, else random
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # modest size if no parent; pick a length in the 2‑3k range
        n0 = rng.integers(2000, 3000)
        a = rng.random(n0, dtype=np.float64)

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                       # enforce sum = 1

    n = a.size
    # FFT length – next power of two >= 2n‑1
    L = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # Optimisation bookkeeping
    # --------------------------------------------------------------
    best_vec = a.copy()
    best_sc = 2.0 * n * _conv_max(a, L)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    base_alpha = 0.15          # base learning rate
    t_iter = 0

    # other parameters
    top_k = 10                 # number of top convolution entries used for gradient
    stagnation = 0
    stagnation_limit = 2500
    sym_interval = 400
    target = 1.50274349

    # --------------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        t_iter += 1

        # ----- current convolution and score -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        cur_max = conv_full.max()
        cur_sc = 2.0 * n * cur_max

        # ----- keep best so far -----
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            stagnation = 0
        else:
            stagnation += 1

        # ----- early exit if target reached -----
        if best_sc < target:
            break

        # ----- gradient from the top_k largest autocorrelation entries -----
        # indices of the top_k entries (unsorted)
        idxs = np.argpartition(-conv_full, top_k - 1)[:top_k]
        g = np.zeros_like(a)
        for k in idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=int)
            g[i_range] += 2.0 * a[k - i_range]
        g /= top_k

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        step_vec = base_alpha * m_hat / (np.sqrt(v_hat) + eps)

        # ----- back‑tracking line search (up to 4 attempts) -----
        factor = 1.0
        cand = None
        cand_sc = None
        for _ in range(4):
            a_new = a - factor * step_vec
            a_new = _project_simplex(a_new)          # keep on simplex
            # score of the candidate
            a_new_fft = np.fft.rfft(a_new, n=L)
            conv_new = np.fft.irfft(a_new_fft * a_new_fft, n=L)[: 2 * n - 1]
            cand_sc = 2.0 * n * conv_new.max()
            if cand_sc <= cur_sc:
                cand = a_new
                break
            factor *= 0.5

        # if no improvement after line search, keep the last trial
        if cand is None:
            cand = a_new
            cand_sc = 2.0 * n * np.fft.irfft(np.fft.rfft(cand, n=L) *

        # ----- accept / possibly accept uphill move -----
        if cand_sc < cur_sc:
            a, cur_sc = cand, cand_sc
        else:

            prob = math.exp(-(cand_sc - cur_sc) / (base_alpha + 1e-12))
            if rng.random() < prob:
                a, cur_sc = cand, cand_sc

        # ----- periodic symmetrisation (helps with Toeplitz structure) -----
        if t_iter % sym_interval == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = _project_simplex(a_sym)
            a_sym_fft = np.fft.rfft(a_sym, n=L)

            sym_sc = 2.0 * n * conv_sym.max()
            if sym_sc < cur_sc:
                a, cur_sc = a_sym, sym_sc
                if sym_sc < best_sc:
                    best_sc, best_vec = sym_sc, a_sym.copy()
                    stagnation = 0

        # ----- restart if stagnation persists -----
        if stagnation > stagnation_limit:
            # Keep the best seen so far, perturb a small fraction of entries
            a = best_vec.copy()
            mask = rng.random(n) < 0.08               # ~8 % randomised entries
            a[mask] = rng.random(mask.sum())
            a = _project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            stagnation = 0

    # --------------------------------------------------------------
    # Return the best candidate discovered (as a plain Python list)
    # --------------------------------------------------------------
    return [float(x) for x in best_vec]
