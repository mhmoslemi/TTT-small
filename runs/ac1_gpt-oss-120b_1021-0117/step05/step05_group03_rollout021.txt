import time

# Optional import of cvxpy for a final LP refinement step.

    _CVXPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVXPY_AVAILABLE = False


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a: np.ndarray, L: int) -> float:
    """2 * n * max(conv(a, a))  (assumes sum(a)=1)."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    conv = conv_full[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _max_convolution(a: np.ndarray, L: int) -> float:
    """Maximum value of the full linear convolution a * a."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return float(conv_full[: 2 * a.size - 1].max())


def _lp_refine(a: np.ndarray, cur_score: float, L: int,
               deadline: float, t: float = 0.2, max_iter: int = 5) -> (np.ndarray, float):
    """Run a few LP refinements (if cvxpy is present) and return improved a and its score."""
    if not _CVXPY_AVAILABLE:
        return a, cur_score

    n = a.size
    a_const = cp.Constant(a)  # make a constant expression once
    best_a = a.copy()
    best_sc = cur_score

    it = 0
    while time.monotonic() < deadline and it < max_iter:
        it += 1
        # current bound M
        M = _max_convolution(best_a, L)

        # Linear program: max sum(g) s.t. conv(a, g) <= M, g >= 0
        g = cp.Variable(n, nonneg=True)
        conv_ag = cp.conv(a_const, g)  # length 2*n-1
        prob = cp.Problem(cp.Maximize(cp.sum(g)), [conv_ag <= M])
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception:  # pragma: no cover
            break
        if g.value is None:
            break
        g_opt = np.asarray(g.value, dtype=np.float64)
        s = g_opt.sum()
        if s < 1e-12:
            break
        g_norm = g_opt / s

        # convex combination and projection
        cand = (1.0 - t) * best_a + t * g_norm
        cand = _project_simplex(cand)
        cand_sc = _score(cand, L)

        if cand_sc < best_sc:
            best_a, best_sc = cand, cand_sc
        else:
            # if not improving, halve the step size and try again
            t *= 0.5
            if t < 1e-4:
                break
    return best_a, best_sc

                                       deadline, t=0.2, max_iter=5)

    # ------------------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
