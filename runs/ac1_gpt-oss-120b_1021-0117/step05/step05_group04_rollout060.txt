import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ------------------------------------------------------------
def _score_fft(vec: np.ndarray, L: int):
    """Return (score, max_conv, fft) for a non‑negative vector.
    The vector is assumed to have sum ≈ 1 (so denominator ≈1).
    """
    n = vec.size
    a_fft = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    m = conv.max()
    # sum is 1 → denominator drops out; keep generic form for safety
    s = vec.sum()
    if s == 0.0:
        s = 1.0
    score = 2.0 * n * m / (s * s)
    return score, m, a_fft


def _random_transfer(v: np.ndarray, rng: np.random.Generator, max_frac: float = 0.1):
    """Move a random fraction of mass from one index to another."""
    n = v.size
    i = rng.integers(n)
    # pick a different index j
    j = rng.integers(n - 1)
    if j >= i:
        j += 1
    if v[i] == 0.0:
        return v  # nothing to move
    delta = v[i] * rng.random() * max_frac
    out = v.copy()
    out[i] -= delta
    out[j] += delta
    return out

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑empty list of non‑negative coefficients that (hopefully)
    improves the AC1 bound.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # reserve a small margin for the sandbox timeout
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------
    # Initial vector – use the parent if present, else a random one
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        cur = np.maximum(cur, 0.0)            # enforce non‑negativity
    else:
        cur = rng.random(1024, dtype=np.float64)   # modest fallback size

    # Normalise to unit sum – scaling does not affect the score
    tot = cur.sum()
    if tot == 0.0:
        cur.fill(1.0 / cur.size)
    else:
        cur /= tot

    n = cur.size
    # FFT length big enough for linear convolution (2n‑1 points)
    L = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------
    # Initial evaluation
    # --------------------------------------------------------
    cur_score, cur_max, cur_fft = _score_fft(cur, L)
    best_vec = cur.copy()
    best_score = cur_score

    # --------------------------------------------------------
    # Simple stochastic local search
    # --------------------------------------------------------
    it = 0
    stagnation = 0
    TARGET = 1.50274349

    while time.time() < deadline:
        it += 1

        # Propose a neighbour by transferring a random mass fraction
        cand = _random_transfer(cur, rng, max_frac=0.07)
        cand_score, cand_max, cand_fft = _score_fft(cand, L)

        if cand_score < cur_score:
            cur, cur_score, cur_max, cur_fft = cand, cand_score, cand_max, cand_fft
            stagnation = 0
            if cand_score < best_score:
                best_score, best_vec = cand_score, cand.copy()
        else:
            # occasional uphill move to escape plateaus
            stagnation += 1
            if rng.random() < 0.001:
                cur, cur_score, cur_max, cur_fft = cand, cand_score, cand_max, cand_fft

        # Global reshuffle if stuck for a while
        if stagnation > 5000:
            # Dirichlet draw gives a fresh probability vector (sum = 1)
            cur = rng.dirichlet(np.ones(n))
            cur_score, cur_max, cur_fft = _score_fft(cur, L)
            stagnation = 0

        # Early exit once we beat the target
        if best_score < TARGET:

    # --------------------------------------------------------
    # Return the best vector found (as plain Python list)
    # --------------------------------------------------------
