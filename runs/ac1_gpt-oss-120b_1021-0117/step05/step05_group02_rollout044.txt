def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    # Ensure nonnegative before projection? We'll handle outside.
    # Use standard algorithm for projection onto simplex sum=1.
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    n = a.shape[0]

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a: np.ndarray, L: int = None) -> float:
    """Exact evaluator score (without the final 2*n factor)."""
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    return 2.0 * n * max_b / (s * s)

def _gradient_max(a: np.ndarray, L: int) -> np.ndarray:
    """
    Sub‑gradient of max(convolution(a,a)) w.r.t. a using the exact max mask.

    start = time.monotonic()
    # keep a small safety margin before the sandbox kills the process
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialisation – use the provided parent if available, otherwise uniform
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2048
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _project_simplex(cur)
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length (power of two)

    best_vec = cur.copy()
    best_sc = _score(best_vec, L)
    cur_sc = best_sc

    # ------------------------------------------------------------------
    # Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    step0 = 0.6          # base step size for gradient descent
    beta = 0.8           # momentum coefficient
    momentum = np.zeros_like(cur)
    patience = 0
    max_patience = 1500  # trigger random restart after this many failures
    iteration = 0
    T0 = 0.01            # temperature scale for Metropolis acceptance
    target = 1.50274349  # known target bound

    # ------------------------------------------------------------------
    # Main optimisation loop (matrix‑free, O(n log n) per iteration)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1

        # ----- sub‑gradient of the max‑convolution -----
        g = _gradient_max(cur, L)

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- tentative gradient‑descent step (projected) -----
        alpha = step0 / math.sqrt(iteration + 1e-12)
        cand = cur - alpha * momentum
        cand = _project_simplex(cand)
        cand_sc = _score(cand, L)

        # ----- acceptance test (deterministic improvement + Metropolis) -----
        if cand_sc < cur_sc:
            cur, cur_sc = cand, cand_sc
            patience = 0
            if cand_sc < best_sc:
                best_vec, best_sc = cand.copy(), cand_sc
        else:
            T = T0 / math.sqrt(iteration)
            prob = math.exp(-(cand_sc - cur_sc) / (T + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = cand, cand_sc
            patience += 1

        # ----- random perturbation when progress stalls -----
        if patience > max_patience:
            noise = rng.normal(scale=0.01, size=n)
            tmp = cur + noise
            tmp = np.maximum(tmp, 0.0)
            tmp = _project_simplex(tmp)
            tmp_sc = _score(tmp, L)
            if tmp_sc < cur_sc:
                cur, cur_sc = tmp, tmp_sc
                patience = 0
                if tmp_sc < best_sc:
                    best_vec, best_sc = tmp.copy(), tmp_sc
            else:
                # Reset momentum to avoid being stuck in a bad direction
                momentum.fill(0.0)
                patience = 0

        # ----- occasional symmetrisation (helps due to Toeplitz structure) -----
        if iteration % 500 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best_vec, best_sc = sym.copy(), sym_sc
                patience = 0

        # ----- early exit if we already beat the known target -----
        if best_sc < target:
            break

    # Return the best candidate as a plain Python list of floats
    return [float(x) for x in best_vec]
