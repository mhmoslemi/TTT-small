def _topk_weights(conv, frac):
    m = conv.size
    k = max(1, int(m * frac))
    if k >= m:
        w = np.ones_like(conv) / m
        return w
    # find threshold
    # np.partition returns k-th order stat (0-index). For top k values, partition at -k.
    # Example:
    # thresh = np.partition(conv, -k)[-k]
    # but if there are ties, we may include more.
    thresh = np.partition(conv, -k)[-k]
    w = np.where(conv >= thresh, 1.0, 0.0)
    # if more than k due to equality, adjust:
    cnt = w.sum()
    if cnt > 0:
        w = w / cnt
    else:
        w = np.full_like(conv, 1.0 / m)
    return w
```

def gaussian_seq(n, sigma):
    idx = np.arange(n)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
    # ensure non-negative
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a

# Basic utilities (must stay linear memory)

def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale non‑negative vector to have sum = 1 (returns a new array)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _self_conv_max(a: np.ndarray, L: int):
    """
    Compute linear convolution a*a via FFT of length L,
    return (max value, a_fft, full convolution of length 2*len(a)-1).
    """
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * a.size - 1]
    return conv.max(), a_fft, conv


def _gradient(a_fft: np.ndarray, w: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of the (soft) max‑convolution objective.
    Returns vector of length n.
    """
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)          # map to length‑n gradient
    return 2.0 * grad_full[idx]


def _topk_weights(conv: np.ndarray, frac: float) -> np.ndarray:
    """
    Uniform weights over the top `frac` fraction of convolution entries.
    """
    m = conv.size
    k = max(1, int(m * frac))
    if k >= m:
        return np.full_like(conv, 1.0 / m)
    # threshold for top‑k (numpy's partition works in‑place)
    thresh = np.partition(conv, -k)[-k]
    w = np.where(conv >= thresh, 1.0, 0.0)
    cnt = w.sum()
    if cnt == 0:
        return np.full_like(conv, 1.0 / m)
    return w / cnt

    """Discrete Gaussian centred on the middle of the array."""
    idx = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
    return _normalize(a)
