import time, math, numpy as np

def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale non-negative vector to have L1 sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback uniform
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s

def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a using FFT of length L, returns full conv."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)
    # keep only linear part (length 2*len(a)-1)
    return conv[: 2 * a.size - 1]

    while time.monotonic() < deadline:

        conv = _conv_fft(cur, L)
        cur_max = conv.max()
