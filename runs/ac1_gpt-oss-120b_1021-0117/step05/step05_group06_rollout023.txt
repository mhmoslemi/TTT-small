import time

# --------------------------------------------------------------
# Simplex projection (L1 onto {x>=0, sum=1})
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (non‑negative, sum=1)."""
    if v.ndim != 1:
        v = v.ravel()
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# --------------------------------------------------------------

    """Return max(conv_full) for a*a using zero‑padded FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv_full.max())

            w = np.exp((conv_full - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
        else:
            mask = conv_full >= max_val - 1e-12
            cnt = int(mask.sum())
            if cnt == 0:
                w = np.full_like(conv_full, 1.0 / conv_full.size)
            else:
                w = np.where(mask, 1.0 / cnt, 0.0)

        # gradient of the (smoothed) objective
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * g

        # back‑tracking line search
        alpha = step0 / math.sqrt(it)
        improved = False
        for _ in range(6):
            cand = cur - alpha * momentum
            cand = _project_simplex(cand)
            cand_sc, cand_bound = _evaluate(cand)
            if cand_bound < cur_bound:
                cur, cur_sc, cur_bound = cand, cand_sc, cand_bound
                improved = True
                if cand_bound < best_bound:
                    best_seq = cand.copy()
                    best_sc, best_bound = cand_sc, cand_bound
                break
            alpha *= 0.5

        if not improved:
            # optionally try a tiny step without momentum
            cand = cur - (alpha * 0.1) * g
            cand = _project_simplex(cand)
            cand_sc, cand_bound = _evaluate(cand)
            if cand_bound < cur_bound:
                cur, cur_sc, cur_bound = cand, cand_sc, cand_bound
                if cand_bound < best_bound:
                    best_seq = cand.copy()
                    best_sc, best_bound = cand_sc, cand_bound

        # occasional symmetrisation (helps for Toeplitz‑like structure)
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc, sym_bound = _evaluate(sym)
            if sym_bound < cur_bound:
                cur, cur_sc, cur_bound = sym, sym_sc, sym_bound
                if sym_bound < best_bound:
                    best_seq = sym.copy()
                    best_sc, best_bound = sym_sc, sym_bound

        # early stop if we are already at the theoretical minimum (2)
        if best_bound <= 2.0001:
            break

    # ------------------------------------------------------------------
    # Random permutation exploration (spreads large entries)
    # ------------------------------------------------------------------
    sorted_vals = np.sort(best_seq)[::-1]               # descending
    perm_trials = 0
    while time.monotonic() < deadline - 1.0 and perm_trials < 3000:
        step = rng.integers(1, n)                       # random stride
        perm = sorted_vals[np.mod(np.arange(n) * step, n)]
        perm_sc, perm_bound = _evaluate(perm)
        if perm_bound < best_bound:
            best_seq = perm.copy()
            best_sc, best_bound = perm_sc, perm_bound
        perm_trials += 1

    # ------------------------------------------------------------------
    # Simple hill‑climbing swaps to polish the best candidate
    # ------------------------------------------------------------------
    cur = best_seq.copy()
    cur_bound = best_bound
    while time.monotonic() < deadline - 0.2:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_sc, new_bound = _evaluate(cur)
        if new_bound < cur_bound:
            cur_bound = new_bound
            best_seq = cur.copy()
            best_sc, best_bound = new_sc, new_bound
            if best_bound < 2.0001:
                break
        else:
            # revert
            cur[i], cur[j] = cur[j], cur[i]

    # ------------------------------------------------------------------
    # Fallback – uniform vector guarantees bound = 2
    # ------------------------------------------------------------------
    if best_seq.size == 0:
        best_seq = np.full(1, 1.0, dtype=np.float64)

    return [float(x) for x in best_seq]
