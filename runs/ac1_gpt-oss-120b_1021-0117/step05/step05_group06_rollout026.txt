import time
import math
import numpy as np

# --------------------------------------------------------------
# Projection onto the probability simplex (sum=1, non‑negative)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum x = 1}."""
    # works for any real vector v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# --------------------------------------------------------------
# Maximum value of the (non‑reversed) self‑convolution via FFT
# --------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    """Return max(full convolution a*a) using zero‑padded FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    return float(conv_full[: 2 * a.size - 1].max())


# --------------------------------------------------------------
# Core optimisation routine (projected gradient descent)
# --------------------------------------------------------------
def _optimize(
    a_start: np.ndarray,
    n: int,
    L: int,
    deadline: float,
    target_bound: float,
) -> tuple[np.ndarray, float]:
    """
    Return (best_sequence, best_max_conv) after a gradient‑descent run.
    The sequence is kept on the probability simplex (sum=1).
    """
    # hyper‑parameters – tuned for fast early progress and fine later tuning
    step0 = 1.0
    beta = 0.9
    eps0 = 0.5
    eps_decay = 0.9995
    eps_min = 1e-9

    cur = _project_simplex(a_start.copy())
    cur_sc = _max_conv(cur, L)
    best = cur.copy()
    best_sc = cur_sc
    momentum = np.zeros_like(cur)

    it = 0
    while time.monotonic() < deadline - 0.5:
        it += 1

        # ----- convolution & soft‑max weights (smooth approximation of max) -----
        a_fft = np.fft.rfft(cur, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_val = conv.max()

        eps = max(eps_min, eps0 * (eps_decay ** it))
        if eps > 1e-7:
            w = np.exp((conv - max_val) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.size)
            else:
                w /= w_sum
        else:
            # exact sub‑gradient when temperature is tiny
            mask = conv >= max_val - 1e-12
            cnt = int(mask.sum())
            if cnt == 0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = np.where(mask, 1.0 / cnt, 0.0)

        # ----- gradient of the smoothed objective (matrix‑free) -----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)          # alignment for derivative
        g = 2.0 * grad_full[idx]

        # ----- momentum update -----
        momentum = beta * momentum + (1.0 - beta) * g

        # ----- back‑tracking line search -----
        alpha = step0 / math.sqrt(it)
        improved = False
        for _ in range(10):
            cand = cur - alpha * momentum
            cand = _project_simplex(cand)
            cand_sc = _max_conv(cand, L)
            if cand_sc <= cur_sc:
                cur = cand
                cur_sc = cand_sc
                improved = True
                if cand_sc < best_sc:
                    best = cand.copy()
                    best_sc = cand_sc
                break
            alpha *= 0.5

        # occasional random jitter when stuck
        if not improved and np.random.rand() < 0.01:
            jitter = np.random.normal(scale=0.01, size=n)
            cand = cur + jitter
            cand = _project_simplex(cand)
            cand_sc = _max_conv(cand, L)
            if cand_sc < cur_sc:
                cur = cand
                cur_sc = cand_sc
                if cand_sc < best_sc:
                    best = cand.copy()
                    best_sc = cand_sc

        # occasional symmetrisation (helps exploit the Toeplitz structure)
        if it % 250 == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _max_conv(sym, L)
            if sym_sc < cur_sc:
                cur = sym
                cur_sc = sym_sc
                if sym_sc < best_sc:
                    best = sym.copy()
                    best_sc = sym_sc

        # early exit if target already reached
        if 2.0 * n * best_sc < target_bound:
            break

    return best, best_sc

    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # ----------------------------------------------------------
    # Initialise – use the provided parent if present, otherwise a random start
    # ----------------------------------------------------------
    if "height_sequence_1" in globals():

    else:

    n = a0.size

    # ----------------------------------------------------------
    # FFT length (next power of two >= 2n‑1)
    # ----------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())

    # ----------------------------------------------------------
    # Run several optimisation attempts while time permits
    # ----------------------------------------------------------
    target_bound = 1.50274349
    best_overall = None
    best_sc_overall = np.inf

    # first attempt using the parent (or the random start)
    attempts = 0
    while time.monotonic() < deadline - 1.0:
        attempts += 1
        # occasionally re‑initialise with a fresh random vector to escape local minima
        if attempts == 1:
            init_vec = a0
        else:
            init_vec = rng.random(n, dtype=np.float64)
        # optimise from this start
        cand_seq, cand_sc = _optimize(init_vec, n, L, deadline, target_bound)
        # keep the best across attempts
        if cand_sc < best_sc_overall:
            best_overall = cand_seq
            best_sc_overall = cand_sc
            if 2.0 * n * best_sc_overall < target_bound:
                break

    # ----------------------------------------------------------
    # Simple hill‑climbing swaps to polish the best found sequence
    # ----------------------------------------------------------
    if best_overall is None:
        # fallback – uniform distribution (always feasible)
        best_overall = np.full(n, 1.0 / n, dtype=np.float64)
        best_sc_overall = _max_conv(best_overall, L)

    cur = best_overall.copy()
    cur_sc = best_sc_overall
    while time.monotonic() < deadline - 0.1:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cur[i], cur[j] = cur[j], cur[i]            # swap
        new_sc = _max_conv(cur, L)
        if new_sc < cur_sc:
            cur_sc = new_sc
            best_overall = cur.copy()
            best_sc_overall = new_sc
            if 2.0 * n * best_sc_overall < target_bound:
                break
        else:
            # revert
            cur[i], cur[j] = cur[j], cur[i]

    # ----------------------------------------------------------
    # Return the best candidate as a plain Python list of floats
    # ----------------------------------------------------------
    return [float(x) for x in best_overall]
