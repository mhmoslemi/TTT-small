def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # arbitrary length 2048
        n0 = 2048
        a = rng.random(n0, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a[:] = 1.0
        s = a.sum()
    a = a / s  # sum = 1

    n = a.size
    # choose FFT length as power of two >= 2n-1
    L = 1 << ((2 * n - 1).bit_length())

    best_vec = a.copy()
    best_sc = 2.0 * n * np.max(np.convolve(a, a))

    # Adam parameters
    lr0 = 0.5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # smoothing parameter alpha start low then increase
    alpha = 5.0
    # schedule for alpha
    alpha_inc = 0.01  # per iteration

    # symmetrization
    sym_interval = 200

    it = 0
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # compute convolution via FFT
        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]

        M = conv.max()
        # score = 2*n*M/(sum^2) but sum=1
        score = 2.0 * n * M
        if score < best_sc:
            best_sc = score
            best_vec = a.copy()
        if best_sc < target:
            break

        # compute softmax weights
        # shift to avoid overflow
        # w_k = exp(alpha * (conv_k - M))
        # w sum normalized
        # else if alpha too large cause all zeros except max; that's okay
        exp_arg = alpha * (conv - M)  # <= 0
        # to avoid underflow, we can compute max of exp_arg which is 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):

            w = np.zeros_like(conv)
            w[conv == M] = 1.0
            w_sum = w.sum()

        g = 2.0 * g_full[n-1:n-1+n]

        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        v_hat = v / (1 - beta2 ** it)

def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=z}."""
    if z < 0:
        raise ValueError("z must be non‑negative")
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)
