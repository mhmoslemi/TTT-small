def _score(a):
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:2*n-1]
    max_conv = conv.max()
    s = a.sum()
    if s <= 0:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Fast evaluation of the AC1 bound (exact, not approximated)
# ------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return the exact AC1 bound for a non‑negative vector."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s <= 0.0:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))
