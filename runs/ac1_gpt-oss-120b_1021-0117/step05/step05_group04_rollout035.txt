def _convolve_max(a, L):
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2*a.size-1]
    max_idx = int(conv.argmax())
    max_val = float(conv[max_idx])
    return max_val, max_idx, a_fft, conv
```

def _max_gradient(a, k):
    n = a.size
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    grad = np.zeros_like(a)
    # double contribution
    grad[mask] = 2.0 * a[j[mask]]
    # correct for diagonal where idx == j
    diag = mask & (j == idx)
    grad[diag] = a[idx[diag]]
    return grad
```

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ------------------------------------------------------------
def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to have sum = 1 (avoid division by zero)."""
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


def _conv_max_fft(a: np.ndarray, L: int):
    """
    Full linear convolution a*a via FFT.
    Returns (max_value, argmax_index, rfft_of_a, full_conv_array).
    """
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, a_fft, conv


def _softmax_weights(conv: np.ndarray, cur_max: float, eps: float) -> np.ndarray:
    """Soft‑max over convolution entries, temperature = eps."""
    w = np.exp((conv - cur_max) / eps)
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size, dtype=np.float64)
    return w / s


def _gradient(a_fft: np.ndarray, w: np.ndarray, n: int, L: int) -> np.ndarray:
    """
    Gradient of the soft‑max smoothed max‑conv objective.
    Implements g_i = 2·(a * w_rev)_{(2n‑2)‑i}.
    """
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=L)
    grad_full = np.fft.irfft(a_fft * w_fft, n=L)
    idx = (2 * n - 2) - np.arange(n)
    return 2.0 * grad_full[idx]


def _max_gradient(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of conv[k] = Σ_i a_i·a_{k‑i} with respect to a.
    Handles boundary conditions and the i=j case.
    """
    n = a.size
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad = np.zeros_like(a)
    # contribution a_j appears twice unless i == j
    grad[mask] = 2.0 * a[j[mask]]
    diag = mask & (j == i)
    grad[diag] = a[i[diag]]          # correct double‑count on the diagonal
    return grad

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a list of non‑negative coefficients whose AC1 bound
    (2·n·max(a*a)/sum(a)^2) is as small as possible.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------
    # Initial vector – parent if available, else random
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 2500
        cur = rng.random(n, dtype=np.float64)

    cur = np.maximum(cur, 0.0)
    cur = _normalize(cur)
    n = cur.size
    L = 1 << ((2 * n - 1).bit_length())   # FFT length ≥ 2n‑1

    # --------------------------------------------------------
    # Optimisation state
    # --------------------------------------------------------
    cur_max, cur_idx, cur_fft, cur_conv = _conv_max_fft(cur, L)
    best_vec = cur.copy()
    best_max = cur_max
    best_score = 2.0 * n * best_max
    target_score = 1.50274349

    # parameters
    eps0 = 0.5
    eps_decay = 0.9985          # faster cooling than the reference
    step0 = 0.4
    beta = 0.9                  # momentum coefficient
    momentum = np.zeros_like(cur)
    no_improve = 0
    it = 0

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- temperature for soft‑max ----
        eps = max(1e-12, eps0 * (eps_decay ** it))

        # ---- soft‑max weights and gradient ----
        w = _softmax_weights(cur_conv, cur_max, eps)
        grad = _gradient(cur_fft, w, n, L)

        # ---- optional switch to true‑max gradient when very cold ----
        if eps < 1e-8:
            grad = _max_gradient(cur, cur_idx)

        # ---- momentum update ----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ---- candidate generation (exponentiated update) ----
        step = step0 / math.sqrt(it)
        exponent = -step * momentum
        exponent = np.clip(exponent, -50.0, 50.0)      # protect exp()
        cand = cur * np.exp(exponent)

                    best_score = 2.0 * n * best_max

        if no_improve > 3000:

                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _normalize(tmp)
                tmp_max, tmp_idx, _, _ = _conv_max_fft(tmp, L)

                    cur_idx = tmp_idx
                    best_max = tmp_max
                    best_vec = tmp.copy()
                    best_score = 2.0 * n * best_max
                no_improve = 0

        if best_score < target_score:
            break

    # --------------------------------------------------------
    # Return best vector found (plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
