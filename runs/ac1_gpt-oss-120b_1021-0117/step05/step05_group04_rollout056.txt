import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int):
    """Linear convolution via FFT and its maximum value."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * a.size - 1]
    return conv.max(), a_fft, conv


def _grad_at_max(a: np.ndarray, idx_max: int):
    """
    Exact sub‑gradient of the max‑convolution at the active index.
    For each i, g_i = 2 * a_{idx_max - i} if the index is inside the vector,
    otherwise 0.
    """
    n = a.size
    js = idx_max - np.arange(n)
    mask = (js >= 0) & (js < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[js[mask]]
    return g
