def _softmax_weights(conv, beta):
    if beta <= 0: # uniform
        w = np.full_like(conv, 1.0/conv.size)
    else:
        maxc = conv.max()
        # subtract maxc for stability
        exp_vals = np.exp(beta * (conv - maxc))
        sum_exp = exp_vals.sum()
        if sum_exp == 0.0:
            w = np.full_like(conv, 1.0/conv.size)
        else:
            w = exp_vals / sum_exp
    return w

import time, math, numpy as np

# ------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum = 1, non‑neg.)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum x = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Main optimisation routine
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a safety margin for final conversion/return
    deadline = start + max(0.0, float(budget_s) - 0.8)

    # --------------------------------------------------------
    # Initial vector – use the provided parent if available
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: uniform vector of length 1024
        a = np.ones(1024, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)          # enforce sum = 1
    n = a.size

    # --------------------------------------------------------
    # FFT length – must accommodate convolution of length 3n‑2
    # --------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())   # next power of two ≥ 3n‑2
    idx = (2 * n - 2) - np.arange(n)      # indices for gradient extraction

    # --------------------------------------------------------
    # Adam hyper‑parameters
    # --------------------------------------------------------
    base_lr = 0.25
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # --------------------------------------------------------
    # Soft‑max (smooth max) schedule
    # --------------------------------------------------------
    beta_soft = 0.0          # start with uniform weights
    beta_inc = 0.004         # increment per iteration
    beta_max = 45.0

    # --------------------------------------------------------
    # Misc. book‑keeping
    # --------------------------------------------------------
    target = 1.50274349
    sym_period = 350
    noimp_limit = 3000

    # initial score
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)
    conv = conv_full[: 2 * n - 1]
    max_val = conv.max()
    cur_sc = 2.0 * n * max_val
    best_vec = a.copy()
    best_sc = cur_sc

    it = 0
    no_improve = 0

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---- current FFT & convolution (already have a_fft, conv) ----
        # compute smooth‑max weights
        beta_soft = min(beta_max, beta_soft + beta_inc)
        if beta_soft <= 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            # soft‑max over (conv - max)
            shifted = conv - max_val
            exp_vals = np.exp(beta_soft * shifted)
            s = exp_vals.sum()
            if s == 0.0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = exp_vals / s

        # ---- gradient of the (smooth) max ----
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        g = 2.0 * grad_full[idx]

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = base_lr * (math.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_simplex(a_new)

        # ---- occasional symmetrisation ----
        if it % sym_period == 0:
            a_new = (a_new + a_new[::-1]) * 0.5
            a_new = _project_simplex(a_new)

        # ---- evaluate new candidate ----
        a_fft_new = np.fft.rfft(a_new, n=L)
        conv_full_new = np.fft.irfft(a_fft_new * a_fft_new, n=L)
        conv_new = conv_full_new[: 2 * n - 1]
        max_val_new = conv_new.max()
        sc_new = 2.0 * n * max_val_new

        if sc_new < best_sc:
            best_sc = sc_new
            best_vec = a_new.copy()

        # ---- accept or reject (simple greedy + occasional uphill) ----
        if sc_new <= cur_sc:
            a, a_fft, conv, max_val, cur_sc = a_new, a_fft_new, conv_new, max_val_new, sc_new
            no_improve = 0
        else:

            prob = math.exp(-(sc_new - cur_sc) / (lr + 1e-12))

                a, a_fft, conv, max_val, cur_sc = a_new, a_fft_new, conv_new, max_val_new, sc_new
                no_improve = 0
            else:

        # ---- random mass‑transfer when stuck ----
        if no_improve > noimp_limit:

                j = rng.integers(0, n - 1)
                if j >= i:

                tmp = a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_simplex(tmp)

                if tmp_sc < best_sc:

                if tmp_sc < cur_sc:
                    a, a_fft, conv, max_val, cur_sc = tmp, tmp_fft, tmp_conv, tmp_conv.max(), tmp_sc
                no_improve = 0

        # ---- early exit if target reached ----
        if best_sc < target:
            break

    # --------------------------------------------------------
    # Return the best candidate found (as plain Python list)
    # --------------------------------------------------------
    return [float(x) for x in best_vec]
