import time, numpy as np

def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable softmax, returns a probability vector (sum=1, non‑negative)."""
    m = np.max(x)
    e = np.exp(x - m)
    s = e.sum()
    if s == 0.0:
        # fallback to uniform distribution
        return np.full_like(x, 1.0 / x.size)
    return e / s

    else:

    if s0 <= 0.0:
