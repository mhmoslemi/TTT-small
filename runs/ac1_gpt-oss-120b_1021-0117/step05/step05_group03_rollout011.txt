def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # initialization
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a0 = rng.random(2048, dtype=np.float64)  # fallback length
    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)
    cur = a0.copy()
    n = cur.size
    L = 1 << ((3 * n - 2).bit_length())  # FFT length

    # book-keeping
    best = cur.copy()
    cur_sc = _score(cur, L)
    best_sc = cur_sc

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.05  # base learning rate
    eps_adam = 1e-8
    it = 0
    no_improve = 0
    sym_interval = 500
    perturb_interval = 3000
    target = 1.50274349

    while time.monotonic() < deadline:
        it += 1

        # Compute convolution
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # compute softmax weights for smooth max
        eps_s = max_val * 0.1 + 1e-12
        w_raw = np.exp((conv - max_val) / eps_s)
        w = w_raw / w_raw.sum()

        # Gradient via FFT
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        tmp = cur - step
        tmp = _project_simplex(tmp)

        tmp_sc = _score(tmp, L)

        # Accept if better, otherwise maybe keep current
        if tmp_sc < cur_sc:
            cur, cur_sc = tmp, tmp_sc
            no_improve = 0
            if tmp_sc < best_sc:
                best, best_sc = tmp.copy(), tmp_sc
        else:
            # occasional uphill acceptance
            prob = math.exp(-(tmp_sc - cur_sc) / (lr + 1e-12))
            if rng.random() < prob:
                cur, cur_sc = tmp, tmp_sc
            no_improve += 1

        # Symmetrization
        if it % sym_interval == 0:
            sym = (cur + cur[::-1]) * 0.5
            sym = _project_simplex(sym)
            sym_sc = _score(sym, L)
            if sym_sc < cur_sc:
                cur, cur_sc = sym, sym_sc
                if sym_sc < best_sc:
                    best, best_sc = sym.copy(), sym_sc
                no_improve = 0

        # Random perturbation if stuck
        if no_improve >= perturb_interval:
            # random exchange of mass between two coordinates
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            delta = cur[i] * rng.random() * 0.5
            tmp2 = cur.copy()
            tmp2[i] -= delta
            tmp2[j] += delta
            tmp2 = _project_simplex(tmp2)
            tmp2_sc = _score(tmp2, L)
            if tmp2_sc < cur_sc:
                cur, cur_sc = tmp2, tmp2_sc
                if tmp2_sc < best_sc:
                    best, best_sc = tmp2.copy(), tmp2_sc
                no_improve = 0

        # early stop target
        if best_sc < target:
            break

    return [float(x) for x in best]

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score_vec(a: np.ndarray, L: int) -> float:
    """Exact score 2*n*max(conv) for a vector a (assumes sum(a)=1)."""
    n = a.shape[0]
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)
    max_conv = conv_full[: 2 * n - 1].max()
    return float(2.0 * n * max_conv)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a non‑empty non‑negative sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ----- initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a0 = rng.random(2048, dtype=np.float64)
    a0 = np.maximum(a0, 0.0)
    a0 = _project_simplex(a0)          # enforce sum = 1
    cur = a0.copy()
    n = cur.size

    # FFT length >= 3n-2 (covers a * w where w length = 2n-1)
    L = 1 << ((3 * n - 2).bit_length())

    # ----- bookkeeping ------------------------------------------------
    best_vec = cur.copy()
    cur_sc = _score_vec(cur, L)
    best_sc = cur_sc

    # Adam state
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    lr = 0.3                         # base learning rate
    eps_adam = 1e-8

    it = 0
    no_improve = 0
    sym_interval = 500
    perturb_interval = 3000
    target = 1.50274349

    # ----- optimisation loop ----------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # convolution of current vector with itself
        a_fft = np.fft.rfft(cur, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        max_val = conv.max()

        # hard‑max weights (sub‑gradient of max)
        mask = np.isclose(conv, max_val, atol=1e-12)
        w = mask.astype(np.float64)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # gradient via FFT (2 * (a * w_rev) on appropriate indices)
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(a_fft * w_fft, n=L)
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * grad_full[idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
