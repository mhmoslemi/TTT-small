import time
import math
import numpy as np

def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = z}."""
    n = v.size
    if z < 0:
        raise ValueError("z must be non‑negative")
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - z
    rho = np.where(u - css / np.arange(1, n + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_max_and_idx(a: np.ndarray, L: int):
    """Return (max value, index) of the full convolution a*a (linear, not circular)."""
    af = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    idx = int(np.argmax(conv_full))
    return float(conv_full[idx]), idx

def _grad_at_max(a: np.ndarray, idx: int) -> np.ndarray:
    """Sub‑gradient of max‑convolution at the given index."""
    n = a.size
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
    j_range = idx - i_range
    g[i_range] = 2.0 * a[j_range]
    return g

    """Return a non‑negative step‑function that (approximately) minimizes the AC1 bound."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ---- initialise ---------------------------------------------------------

        a = rng.random(2048, dtype=np.float64)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a[:] = 1.0
    a = a / a.sum()                     # enforce sum = 1

    n = a.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two >= 2n‑1

    # ---- bookkeeping --------------------------------------------------------
    best_vec = a.copy()
    best_max, _ = _conv_max_and_idx(a, L)
    best_sc = 2.0 * n * best_max

    momentum = np.zeros_like(a)
    step0 = 0.6            # initial step magnitude
    beta = 0.85            # momentum factor
    sym_interval = 200    # how often to enforce symmetry
    no_improve = 0
    it = 0
    target = 1.50274349   # stop if we beat this

    # ---- main optimisation loop ---------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # current convolution and its worst index
        cur_max, cur_idx = _conv_max_and_idx(a, L)
        cur_sc = 2.0 * n * cur_max

        # keep best seen so far
        if cur_sc < best_sc:
            best_sc, best_vec = cur_sc, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # early exit if target reached
        if best_sc < target:
            break

        # gradient of the maximal entry
        grad = _grad_at_max(a, cur_idx)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # back‑tracking line search
        base_alpha = step0 / math.sqrt(it)
        alpha = base_alpha
        cand = None
        cand_sc = None
        for _ in range(8):

            a_tmp = _project_simplex(a_tmp)        # simplex projection
