import time, math, numpy as np

def _max_conv(a: np.ndarray, L: int) -> float:
    """Maximum value of full convolution a*a using zero‑padded FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.size - 1]
    return float(conv.max())

def _anti_identity(arr: np.ndarray) -> np.ndarray:
    """Place sorted values in anti‑identity order (largest opposite smallest)."""
    vals = np.sort(arr)[::-1]
    out = np.empty_like(vals)
    left, right = 0, vals.size - 1
    k = 0
    while left <= right:
        out[left] = vals[k]; k += 1
        if left < right:
            out[right] = vals[k]; k += 1
        left += 1
        right -= 1
    return out
