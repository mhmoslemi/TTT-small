import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate multiple initial configurations with random offsets
    np.random.seed(42)
    centers = np.random.rand(n, 2)
    # Normalize to unit square
    centers = (centers - np.min(centers, axis=0)) / (np.max(centers, axis=0) - np.min(centers, axis=0))

    def objective(r):
        return -np.sum(r)

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636

    # Initial guess based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': check_overlap}, {'type': 'eq', 'fun': constraint}], method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
