import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.ones(n) * 0.1

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636

    def boundary_constraint(r):
        for i in range(n):
            x, y = centers[i]
            r_i = r[i]
            if x - r_i < -1e-12 or x + r_i > 1 + 1e-12 or y - r_i < -1e-12 or y + r_i > 1 + 1e-12:
                return -np.inf
        return np.sum(r)

    bounds = [(0, 1) for _ in range(n)]

    result = minimize(objective, radii, constraints=[
        {'type': 'ineq', 'fun': constraint},
        {'type': 'ineq', 'fun': boundary_constraint}
    ], method='SLSQP', bounds=bounds)

    return centers, result['x'], np.sum(result['x'])
