from scipy.optimize import differential_evolution
from itertools import combinations

def run_packing():

    # Initial hexagonal grid with boundary-aware adjustment

        return -np.sum(r)

                return -np.inf
        return np.sum(r)

    dists = np.array([[np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1]) for j in range(n)] for i in range(n)])
    initial_r = np.min(dists) * np.ones(n)
    
    result = differential_evolution(objective, bounds=[(0, 1)] * n, strategy='best1bin', popsize=20, maxiter=100, tol=1e-8, mutation=(0.5, 1), recombination=0.9, polish=False)
    
    # Final adjustment for boundary and overlap
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
