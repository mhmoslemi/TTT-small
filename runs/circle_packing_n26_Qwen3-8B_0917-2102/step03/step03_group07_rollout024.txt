import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal grid with boundary-aware placement
    centers = np.zeros((n, 2))
    col = 0
    row = 0
    for i in range(n):
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        col += 1
        if col >= 5:
            col = 0
            row += 1

    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)

    def constraint_dist(x):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < x[i] + x[j] - 1e-8:
                    return -np.inf
        return 0

    def constraint_boundary(x):
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = x[i]
            if (x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12
                    or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12):
                return -np.inf
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_dist},
            {'type': 'ineq', 'fun': constraint_boundary}]

    # Initial guess
    initial_guess = np.full(n, 0.1)

    # Optimization
    result = minimize(objective, initial_guess, constraints=cons, method='SLSQP', bounds=Bounds(0, 1))

    radii = result.x

    # Final check and adjustment for boundary
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
