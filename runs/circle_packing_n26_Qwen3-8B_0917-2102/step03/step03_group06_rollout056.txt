import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Create a hexagonal grid with 5 columns and 6 rows for 26 circles
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]

    # Scale centers to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(radii):
        return -np.sum(radii)

    def constraint(radii):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return -np.inf
        return 0

    bounds = [(0, 1) for _ in range(n)]
    cons = {'type': 'ineq', 'fun': constraint}

    # Initial guess: scale based on grid spacing
    grid_spacing = np.min(np.sqrt(np.sum((centers[1] - centers[0])**2)))
    radii_initial = np.full(n, grid_spacing * 0.5)
    result = minimize(objective, radii_initial, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-8, options={'maxiter': 500})

    radii = result.x

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
