import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial layout with more optimal corner and edge placements
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        if i < 5:
            centers[i] = [i / 4, 0.1]
            radii[i] = 0.1
        elif i < 10:
            centers[i] = [0.1, (i - 5) / 5 + 0.2]
            radii[i] = 0.1
        elif i < 15:
            centers[i] = [(i - 10) / 5 + 0.1, 0.8]
            radii[i] = 0.1
        elif i < 20:
            centers[i] = [1 - (i - 15) / 5 - 0.1, 0.8]
            radii[i] = 0.1
        else:
            centers[i] = [0.9, (i - 20) / 6 + 0.1]
            radii[i] = 0.1

    def objective(r):
        return -np.sum(r)

    def constraint_sum(r):
        return np.sum(r) - 2.636

    cons = [{'type': 'ineq', 'fun': lambda r: np.array([r[i] for i in range(n)])}]
    cons.append({'type': 'eq', 'fun': constraint_sum})

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons.append({'type': 'ineq', 'fun': check_overlap})

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n, tol=1e-10)

    # Adjust radii to correct overlaps
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
