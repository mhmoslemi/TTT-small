import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal lattice with adaptive spacing
    centers = np.zeros((n, 2))
    row = 0
    col = 0
    for i in range(n):
        if i % 5 == 0:
            row += 1
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        col += 1
        if col >= 5:
            col = 0
            row += 1

    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    # Initial radii based on minimum spacing
    def initial_radius(i):
        x, y = centers[i]
        r = 0.1
        for j in range(n):
            if i == j:
                continue
            dx = np.abs(x - centers[j][0])
            dy = np.abs(y - centers[j][1])
            r = min(r, np.sqrt(dx*dx + dy*dy) / 2)
        return r

    radii = np.array([initial_radius(i) for i in range(n)])

    def objective(r):
        return -np.sum(r)

    def constraint1(x):
        return np.sum(x)

    def constraint2(x):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < x[i] + x[j] - 1e-8:
                    return -np.inf
        return 0

    cons = [{'type': 'ineq', 'fun': constraint1, 'lb': 2.636 - 1e-8, 'name': 'min_radius'},
            {'type': 'ineq', 'fun': constraint2, 'name': 'non_overlapping'}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})

    # Final adjustment to ensure boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result.x[i] = 0.0

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
