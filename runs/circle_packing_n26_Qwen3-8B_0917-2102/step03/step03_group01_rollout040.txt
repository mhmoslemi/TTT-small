import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)

    def objective(radii):
        return -np.sum(radii)

    def constraint(x):
        return np.sum(x) - 0.01

    cons = [{'type': 'ineq', 'fun': constraint}]

    def update_radii(centers, radii):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j]:
                    ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                    radii[i] *= ratio
                    radii[j] *= ratio
        return radii

    def boundary_constraint(x):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or
                y - r < -1e-12 or y + r > 1 + 1e-12):
                return -1
        return 0

    def update_centers(centers, radii):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12:
                centers[i, 0] = r
            elif x + r > 1 + 1e-12:
                centers[i, 0] = 1 - r
            if y - r < -1e-12:
                centers[i, 1] = r
            elif y + r > 1 + 1e-12:
                centers[i, 1] = 1 - r
        return centers

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1) for _ in range(n)])
    radii = result['x']
    radii = update_radii(centers, radii)
    centers = update_centers(centers, radii)

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
