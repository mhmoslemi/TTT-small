from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Initial guess with hexagonal packing
    centers = np.zeros((n, 2))
    col, row = 0, 0
    for i in range(n):
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        col += 1
        if col >= 5:
            col = 0
            row += 1

    # Normalize centers to [0,1]^2
    centers -= centers.min(axis=0)
    centers /= centers.max(axis=0) - centers.min(axis=0)

    def objective(r):
        return -np.sum(r)

    def constraint1(r):
        return np.sum(r)

    def constraint2(r, i, j):
        dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
        return dist - r[i] - r[j] + 1e-8

    cons = [{'type': 'ineq', 'fun': constraint1, 'lb': 0, 'ub': np.inf}]
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: constraint2(r, i, j)})

    result = minimize(objective, np.full(n, 0.1), constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})

    radii = result.x

    # Final check for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
