from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import product

    centers = np.zeros((n, 2))
    radii = np.zeros(n)

        col = i % 5

            x = col + 0.5

            x = col + 0.5 - 0.5
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = 0.1

    centers -= centers.min(axis=0)

        return -np.sum(radii)
