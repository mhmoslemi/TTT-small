import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal grid positions in [0, 1]^2
    centers = []
    r = np.sqrt(3) / 3  # Hexagonal grid spacing
    for i in range(n):
        col = i // 5
        row = i % 5
        x = col + (row % 2) * r
        y = row + (row + 1) * r / 2
        # Scale to [0,1]^2
        x = x / (n // 2 + 1)
        y = y / (n // 2 + 1)
        centers.append([x, y])
    centers = np.array(centers)

    # Initial radii based on minimal spacing
    dists = squareform(pdist(centers))
    min_dist = np.min(dists)
    radii = np.full(n, min_dist / 2)

    def objective(r):
        return -np.sum(r)

    # Constraints to ensure circles stay within unit square
    def constraint1(x):
        return np.sum(x) - 0.01

    # Constraints to ensure minimal spacing between circles
    def constraint2(x):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < x[i] + x[j] - 1e-8:
                    return -np.inf
        return 0

    cons = [{'type': 'ineq', 'fun': constraint1}, {'type': 'ineq', 'fun': constraint2}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})

    # Final adjustment
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result.fun / 2:
                ratio = (dists[i, j] - 1e-8) / (result.fun / 2)
                result.x[i] *= ratio
                result.x[j] *= ratio

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result.x[i] = 0.0

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
