import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initialize with hexagonal lattice and adjust for square boundary
    centers = np.zeros((n, 2))
    col, row = 0, 0
    for i in range(n):
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        col += 1
        if col >= 5:
            col = 0
            row += 1

    # Scale to [0,1]^2 with boundary-aware adjustment
    min_x, min_y = centers.min(axis=0)
    max_x, max_y = centers.max(axis=0)
    centers = (centers - min_x) / (max_x - min_x)

    # Initial radii based on minimal spacing
    radii = np.full(n, 0.1)
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            radii[i] = min(radii[i], (dists[i, j] - 1e-8) / 2)
            radii[j] = min(radii[j], (dists[i, j] - 1e-8) / 2)

    def objective(r):
        return -np.sum(r)

    def constraint1(x):
        return np.sum(x)

    def constraint2(x):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < x[i] + x[j] - 1e-8:
                    return -np.inf
        return 0

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})

            result.x[i] = 0.0

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
