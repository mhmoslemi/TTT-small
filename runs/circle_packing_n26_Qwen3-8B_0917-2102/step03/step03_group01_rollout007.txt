import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Create a hexagonal grid with 5 rows and 5 columns, adjusting for boundary alignment
    for i in range(n):
        row = i // 5
        col = i % 5
        if row % 2 == 0:
            x = col + 0.5
        else:
            x = col + 0.5 - 0.5
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = 0.1
    
    # Normalize centers to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(radii):
        return -np.sum(radii)

    def constraint(radii):
        return np.sum(radii)

    cons = [{'type': 'ineq', 'fun': constraint, 'jac': lambda r: np.ones_like(r)}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})

    radii = result.x

    # Ensure radii are non-negative
    radii[radii < 0] = 0

    # Compute pairwise distances and adjust radii to maintain non-overlapping
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            d = dists[i, j]
            if d < radii[i] + radii[j] - 1e-8:
                ratio = (d - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
