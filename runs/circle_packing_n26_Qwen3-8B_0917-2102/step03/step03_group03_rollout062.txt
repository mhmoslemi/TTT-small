import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Create a hybrid arrangement: 5 rows with 5 circles each, and 1 extra circle
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]

    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

        return -np.sum(r)

                    return -np.inf

            if x - r[i] < -1e-12 or x + r[i] > 1 + 1e-12 or y - r[i] < -1e-12 or y + r[i] > 1 + 1e-12:
                return -np.inf
        return 0

    cons = [{'type': 'ineq', 'fun': constraint}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'gtol': 1e-10, 'maxiter': 500})

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
