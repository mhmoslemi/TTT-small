import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal lattice with boundary adaptation
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
    
    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)

    def constraint1(x):
        return np.sum(x) - 2.636 + 1e-8

        for i in range(n):

                if dists[i,j] < x[i] + x[j] - 1e-8:
                    return -np.inf
        return 0

    cons = [{'type': 'ineq', 'fun': constraint1, 'name': 'min_radius'},
            {'type': 'ineq', 'fun': constraint2, 'name': 'non_overlapping'}]

    # Initial guess with boundary-aware radii
    radii = np.full(n, 0.1)

        x, y = centers[i]

            radii[i] = 0.05

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})

    for i in range(n):

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
