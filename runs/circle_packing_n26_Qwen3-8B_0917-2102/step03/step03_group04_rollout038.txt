import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create a staggered hexagonal grid with boundary-aware placement
    centers = np.zeros((n, 2))
    col = 0
    row = 0
    for i in range(n):
        if i % 5 == 0:
            col = 0
            row += 1
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        col += 1

    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    # Initial guess for radii
    radii = np.full(n, 0.1)

    def objective(r):
        return -np.sum(r)

    # Constraint: minimum total radius
    def constraint_min_radius(x):
        return np.sum(x) - 2.636 + 1e-8

    # Constraint: non-overlapping circles
    def constraint_non_overlapping(x):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i+1, n):
                if dists[i,j] < x[i] + x[j] - 1e-8:
                    return -np.inf
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_min_radius, 'name': 'min_radius'},
            {'type': 'ineq', 'fun': constraint_non_overlapping, 'name': 'non_overlapping'}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 1000})

    # Adjust radii to stay within bounds
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result.x[i] = 0.0

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
