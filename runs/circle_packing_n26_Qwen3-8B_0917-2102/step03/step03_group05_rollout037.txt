import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create clusters of circles with hexagonal packing
    clusters = []
    for i in range(2):
        for j in range(13):
            if i == 0:
                # First cluster: hexagonal arrangement
                row = j // 4
                col = j % 4
                x = col + 0.5 * (1 - 2 * (row % 2))
                y = row + 0.5
                clusters.append((x, y))
            else:
                # Second cluster: offset to avoid overlap
                row = j // 4
                col = j % 4
                x = col + 0.5 * (1 - 2 * (row % 2)) + 0.5
                y = row + 0.5 + 0.5
                clusters.append((x, y))
    
    # Convert to numpy array
    centers = np.array(clusters)
    radii = np.zeros(n)

    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)

    def constraint1(r):
        return np.sum(r) - 0.01  # Lower bound on total radii

    def constraint2(r):
        return 1 - np.sum(r) - 0.01  # Upper bound on total radii

    cons = [{'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2}]

    def constraint3(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.inf

    result = minimize(objective, radii, constraints=cons + [{'type': 'ineq', 'fun': constraint3}])

    radii = result.x

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
