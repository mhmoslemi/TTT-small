import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Generate a hexagonal grid with more circles near boundaries
    row_idx = np.arange(n)
    col_idx = np.arange(n)
    row = (row_idx // 5).astype(int)
    col = (row_idx % 5).astype(int)
    row = np.where(row > 2, 2, row)
    col = np.where(col > 2, 2, col)
    
    for i in range(n):
        if row[i] % 2 == 0:
            x = col[i] + 0.5
        else:
            x = col[i] + 0.5 - 0.5
        y = row[i] + 0.5
        centers[i] = [x, y]
        radii[i] = 0.1
    
    # Convert to [0,1]^2
    centers -= centers.min(axis=0)
    centers /= centers.max(axis=0) - centers.min(axis=0)

    def objective(radii):
        return -np.sum(radii)

    def constraint(radii):
        return np.sum(radii) - 0.01

    cons = [{'type': 'ineq', 'fun': constraint}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP')

    # Adjust radii to maintain spacing
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
