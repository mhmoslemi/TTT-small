import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create initial hexagonal grid with boundary-aligned circles
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Grid parameters
    rows = 5
    cols = 6
    row_height = 0.8
    col_width = 0.8
    offset = 0.5
    
    # Place circles in a staggered hexagonal grid
    for i in range(n):
        row = i // cols
        col = i % cols
        if row % 2 == 0:
            x = col * col_width + offset
        else:
            x = col * col_width + offset - 0.5 * col_width
        y = row * row_height + offset
        centers[i] = [x, y]
        radii[i] = 0.05
    
    # Normalize to [0,1]^2
    centers -= centers.min(axis=0)
    centers /= centers.max(axis=0) - centers.min(axis=0)

    def objective(radii):
        return -np.sum(radii)

    def constraint(radii):
        return np.sum(radii)

    cons = [{'type': 'ineq', 'fun': constraint}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 1000})

    # Update radii from optimization
    radii = result.x

    # Enforce minimum distance between circles
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final boundary check and adjustment
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
