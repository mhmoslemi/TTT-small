from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initialize centers in a hexagonal grid with boundary-aware placement
    centers = []
    col = 0
    row = 0
    for i in range(n):
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers.append([x, y])
        col += 1
        if col >= 5:
            col = 0
            row += 1
    centers = np.array(centers)

    # Normalize to [0,1]^2

                return -np.inf
        return 0

    cons = [{'type': 'ineq', 'fun': constraint1, 'lb': 0, 'ub': np.inf},
            {'type': 'ineq', 'fun': constraint2}]

    result = minimize(objective, np.full(n, 0.1), constraints=cons, method='SLSQP')

    radii = result.x

    dists = squareform(pdist(centers))

        r = radii[i]

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
