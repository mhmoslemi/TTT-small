import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)

    # Boundary-aligned circles
    radii[0] = 0.15
    radii[1] = 0.15
    radii[2] = 0.15
    radii[3] = 0.15
    radii[4] = 0.15
    radii[5] = 0.15
    radii[25] = 0.15
    radii[24] = 0.15
    radii[23] = 0.15
    radii[22] = 0.15
    radii[21] = 0.15

    def objective(radii):
        return -np.sum(radii)

    def constraint(radii):
        return np.sum(radii) - 0.01

    cons = [{'type': 'ineq', 'fun': constraint}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP')

    # Update centers to reduce overlap
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
