import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create a more compact initial configuration with boundary-aware placement
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.1)
    col = 0
    row = 0
    for i in range(n):
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        col += 1
        if col >= 5:
            col = 0
            row += 1

    # Normalize to [0,1]^2 while preserving aspect ratio
    max_x = np.max(centers[:, 0])
    max_y = np.max(centers[:, 1])
    centers = (centers - centers.min(axis=0)) / np.array([max_x, max_y])

    def objective(r):
        return -np.sum(r)

    def constraint1(x):
        return np.sum(x)

    def constraint2(x):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i+1, n):
                if dists[i,j] < x[i] + x[j] - 1e-8:
                    return -np.inf
        return 0

    cons = [{'type': 'ineq', 'fun': constraint1, 'lb': 2.636 - 1e-8, 'name': 'min_radius'},
            {'type': 'ineq', 'fun': constraint2, 'name': 'non_overlapping'}]

    # Add boundary constraints as soft constraints
    def boundary_constraint(x):
        total = 0
        for i in range(n):
            x_, y_ = centers[i]
            r = x[i]
            if (x_ - r < -1e-12 or x_ + r > 1 + 1e-12 or
                y_ - r < -1e-12 or y_ + r > 1 + 1e-12):
                total += 1
        return total

    cons.append({'type': 'ineq', 'fun': boundary_constraint, 'name': 'boundary'})

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})

    # Final adjustment to ensure boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            result.x[i] = 0.0

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
