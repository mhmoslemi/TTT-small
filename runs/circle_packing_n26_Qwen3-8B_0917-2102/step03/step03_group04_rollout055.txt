import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate boundary-aware hexagonal packing
    centers = np.zeros((n, 2))
    col, row = 0, 0
    for i in range(n):
        if col >= 5:
            col = 0
            row += 1
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
    
    # Normalize to [0,1]^2
    centers -= centers.min(axis=0)
    centers /= centers.max(axis=0) - centers.min(axis=0)

    # Initial radii based on minimal spacing
    dists = squareform(pdist(centers))
    min_dist = np.min(dists[np.triu_indices(n, 1)])
    radii = np.full(n, min_dist / np.sqrt(3) / 2)

    def objective(r):
        return -np.sum(r)

    def non_overlap_cons(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return 0

    def boundary_cons(r):
        for i in range(n):
            x, y = centers[i]
            r_val = r[i]
            if (x - r_val < -1e-12 or x + r_val > 1 + 1e-12
                    or y - r_val < -1e-12 or y + r_val > 1 + 1e-12):
                return -np.inf
        return 0

    cons = [{'type': 'ineq', 'fun': non_overlap_cons, 'name': 'non_overlap'},
            {'type': 'ineq', 'fun': boundary_cons, 'name': 'boundary'}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 300})

    # Final boundary adjustment
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result.x[i] = 0.0

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
