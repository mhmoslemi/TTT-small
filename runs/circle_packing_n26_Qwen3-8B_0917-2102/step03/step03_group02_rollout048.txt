import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Hexagonal packing with staggered rows
    centers = []
    row = 0
    col = 0
    for i in range(n):
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers.append([x, y])
        col += 1
        if col >= 5:
            col = 0
            row += 1
    centers = np.array(centers)
    
    # Scale to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    # Initial radii based on minimal spacing
    dists = squareform(pdist(centers))
    min_dist = np.min(dists)
    radii = np.full(n, min_dist / 2)

    def objective(r):
        return -np.sum(r)

    # Constraints to ensure circles stay within unit square
    def constraint1(x):
        return np.sum(x) - 0.01

    # Constraints to ensure minimal spacing between circles
    def constraint2(x):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < x[i] + x[j] - 1e-8:
                    return -np.inf
        return 0

    cons = [{'type': 'ineq', 'fun': constraint1}, {'type': 'ineq', 'fun': constraint2}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})

    # Final adjustment to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result.x[i] = 0.0

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
