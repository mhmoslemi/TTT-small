import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate initial hexagonal grid with boundary-aware placement
    centers = np.zeros((n, 2))
    col = 0
    row = 0
    for i in range(n):
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        col += 1
        if col >= 5:
            col = 0
            row += 1

    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)

    # Initial guess for radii
    radii = np.full(n, 0.1)

    # Constraints: non-overlap and boundary
    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        for i in range(n):
            x, y = centers[i]
            r_val = r[i]
            if x - r_val < -1e-12 or x + r_val > 1 + 1e-12 or y - r_val < -1e-12 or y + r_val > 1 + 1e-12:
                return -np.inf
        return 0

    cons = [{'type': 'ineq', 'fun': constraint}]

    # Use a tighter tolerance and more iterations for better convergence
    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-12, 'gtol': 1e-12, 'maxiter': 1000})

    radii = result.x

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
