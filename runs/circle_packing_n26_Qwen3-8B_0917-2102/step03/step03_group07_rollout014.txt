import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Create a near-hexagonal grid with staggered rows
    centers = np.zeros((n, 2))
    col = 0
    row = 0
    for i in range(n):
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        col += 1
        if col >= 5:
            col = 0
            row += 1

    min_x, min_y = np.min(centers, axis=0)
    max_x, max_y = np.max(centers, axis=0)

        return -np.sum(r)

    def constraint1(x):
        return np.sum(x)

        for i in range(n):
            x_i, y_i = centers[i]
            r_i = x[i]
            if (x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12
                    or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12):
                return -np.inf
        return 0

    cons = [{'type': 'ineq', 'fun': constraint1, 'lb': 0, 'ub': np.inf},
            {'type': 'ineq', 'fun': constraint2}]

    result = minimize(objective, np.full(n, 0.1), constraints=cons, method='SLSQP', tol=1e-8)

    radii = result.x

                radii[j] *= ratio

    # Final check to ensure all circles are within [0,1]^2

        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
