import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create hexagonal lattice with boundary-aware initial placement
    centers = np.zeros((n, 2))
    col, row = 0, 0
    for i in range(n):
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        col += 1
        if col >= 5:
            col = 0
            row += 1

    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)

    def non_overlapping_constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return 0

    def boundary_constraint(r):
        for i in range(n):
            x, y = centers[i]
            r_i = r[i]
            if x - r_i < -1e-12 or x + r_i > 1 + 1e-12 or y - r_i < -1e-12 or y + r_i > 1 + 1e-12:
                return -np.inf
        return 0

    # Initial guess with uniform radii
    initial_guess = np.full(n, 0.1)

    # Optimization with constraints
    cons = (
        {'type': 'ineq', 'fun': non_overlapping_constraint, 'name': 'non_overlapping'},
        {'type': 'ineq', 'fun': boundary_constraint, 'name': 'boundary'},
    )

    result = minimize(objective, initial_guess, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})

    # Final adjustment to ensure boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12):
            result.x[i] = 0.0

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
