import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal packing in unit square
    centers = []
    for row in range(int(np.sqrt(n)) + 1):
        for col in range(int(np.sqrt(n)) + 1):
            if row % 2 == 0:
                x = col / (np.sqrt(n) + 1)
            else:
                x = (col + 0.5) / (np.sqrt(n) + 1)
            y = row / (np.sqrt(n) + 1)
            if x <= 1 and y <= 1:
                centers.append([x, y])
    centers = np.array(centers)
    
    # Initial radii based on minimal spacing
    dists = squareform(pdist(centers))
    min_dist = np.min(dists)
    radii = np.full(n, min_dist / 2)

    def objective(r):
        return -np.sum(r)

    # Constraint function for keeping circles within square
    def constraint1(x):
        return np.sum(x) - 0.01

    # Constraint function for minimal spacing between circles
    def constraint2(x):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < x[i] + x[j] - 1e-8:
                    return -np.inf
        return 0

    cons = [{'type': 'ineq', 'fun': constraint1}, {'type': 'ineq', 'fun': constraint2}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})

    # Final check and adjustment
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result.x[i] + result.x[j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result.x[i] + result.x[j])
                result.x[i] *= ratio
                result.x[j] *= ratio

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result.x[i] = 0.0

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
