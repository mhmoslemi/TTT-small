import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Define initial centers with a hybrid pattern: hexagonal grid and cluster near corners
    centers = []
    radii = []
    # Hexagonal grid with boundary-aware placement
    for i in range(n):
        col = i % 5
        row = i // 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        # Adjust to be within [0, 1]^2
        x = np.clip(x, 0, 1)
        y = np.clip(y, 0, 1)
        centers.append([x, y])
    centers = np.array(centers)
    
    # Initial radius estimate based on grid spacing
    initial_radius = 0.15
    radii = np.full(n, initial_radius)

    def objective(r):
        return -np.sum(r)

    def constraint1(x):
        return np.sum(x)

    def constraint2(x):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i+1, n):
                if dists[i,j] < x[i] + x[j] - 1e-8:
                    return -np.inf
        return 0

    cons = [{'type': 'ineq', 'fun': constraint1, 'lb': 2.636 - 1e-8, 'name': 'min_radius'},
            {'type': 'ineq', 'fun': constraint2, 'name': 'non_overlapping'}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})

    # Final adjustment to ensure boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result.x[i] = 0.0

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
