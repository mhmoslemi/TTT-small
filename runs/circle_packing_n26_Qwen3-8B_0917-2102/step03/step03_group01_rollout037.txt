import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Initial hexagonal packing with boundary alignment
    for i in range(n):
        row = i // 5
        col = i % 5
        if row % 2 == 0:
            x = col + 0.5
        else:
            x = col + 0.5 - 0.5
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = 0.1
    
    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)
    
    def constraint(r):
        return np.sum(r)
    
    cons = [{'type': 'ineq', 'fun': constraint}]

    def packing_constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] + 1e-8:
                    return -1
        return 0

    cons.append({'type': 'ineq', 'fun': packing_constraint})

    def boundary_constraint(r):
        for i in range(n):
            x, y = centers[i]
            if x - r[i] < -1e-8 or x + r[i] > 1 + 1e-8 or y - r[i] < -1e-8 or y + r[i] > 1 + 1e-8:
                return -1
        return 0

    cons.append({'type': 'ineq', 'fun': boundary_constraint})

    result = minimize(objective, radii, constraints=cons, method='SLSQP', tol=1e-8)

    if not result.success:
        return np.zeros((n, 2)), np.zeros(n), 0.0

    radii = result.x

    # Final check and adjustments
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j] + 1e-8:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
