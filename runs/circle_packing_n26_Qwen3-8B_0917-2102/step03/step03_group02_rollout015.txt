import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal grid with boundary adjustments
    centers = []
    for row in range(5):
        for col in range(5):
            x = col + 0.5 * (row % 2)
            y = row + 0.5
            if row % 2 == 0:
                x = col + 0.5
            centers.append([x, y])
    centers = np.array(centers)

    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    # Initial radii based on minimal spacing
    dists = squareform(pdist(centers))
    min_dist = np.min(dists)
    radii = np.full(n, min_dist / 2)

    def objective(r):
        return -np.sum(r)

    # Constraints to ensure circles stay within unit square and don't overlap
    def constraint(x):
        dists = squareform(pdist(centers))
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = x[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return -np.inf
            for j in range(i + 1, n):
                if dists[i, j] < x[i] + x[j] - 1e-8:
                    return -np.inf
        return 0

    cons = [{'type': 'ineq', 'fun': constraint}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})

    # Final adjustment to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12):
            result.x[i] = 0.0

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
