from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

    # Create a non-uniform hexagonal grid with rows and columns adjusted for better boundary utilization
    centers = []

    min_dist = np.min(dists)

        return -np.sum(r)

        {'type': 'eq', 'fun': lambda r: np.sum(r) - 2.636}
    ]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)
