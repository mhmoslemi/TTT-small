import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    for i in range(n):
        row = i // 7
        col = i % 7
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = 0.1

    scale_y = (1 - 1e-12) / (y_max - y_min)
    centers[:, 0] = (centers[:, 0] - x_min) * scale_x
    centers[:, 1] = (centers[:, 1] - y_min) * scale_y

    def objective(radii):
