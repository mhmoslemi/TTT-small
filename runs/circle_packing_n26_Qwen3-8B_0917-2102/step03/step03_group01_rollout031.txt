import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initialize with a more efficient layout, such as a hexagonal grid with boundary alignment
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Create a hexagonal grid with balanced rows and adjusted for boundary
    row = 0
    col = 0
    for i in range(n):
        if i % 5 == 0:
            row += 1
            col = 0
        if row % 2 == 0:
            x = col + 0.5
        else:
            x = col + 0.5 - 0
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = 0.1
        col += 1
    
    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(radii):
        return -np.sum(radii)
    
    def constraint(radii):
        return np.sum(radii) - 0.01
    
    cons = [{'type': 'ineq', 'fun': constraint}]
    
    result = minimize(objective, radii, constraints=cons, method='SLSQP')
    radii = result.x

    # Adjust radii to maintain spacing
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final boundary check and clipping
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
