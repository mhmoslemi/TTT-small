from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

    cols = 6
    centers = np.zeros((n, 2))

        col = i % cols
        x = col + 0.5 * (1 - 2 * (row % 2))

        centers[i] = [x, y]
