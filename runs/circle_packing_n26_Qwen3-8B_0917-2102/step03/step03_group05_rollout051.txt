import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal packing in unit square with optimized spacing
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Generate a hexagonal grid in a compressed square
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = 0.1
    
    # Normalize to fit within [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)

    def constraint_min_total(r):
        return np.sum(r) - 0.01  # Lower bound on total radii

    def constraint_max_total(r):
        return 2.636 - np.sum(r)  # Upper bound on total radii

    cons = [{'type': 'ineq', 'fun': constraint_min_total},
            {'type': 'ineq', 'fun': constraint_max_total}]

    def constraint_distances(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.inf

    result = minimize(objective, radii, constraints=cons + [{'type': 'ineq', 'fun': constraint_distances}])

    radii = result.x

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
