import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate non-uniform hexagonal grid with 5 columns and 6 rows
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]

    # Initial radii based on minimal spacing
    initial_radii = np.full(n, 0.1)

    def objective(radii):
        return -np.sum(radii)

    def constraint(radii):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return -np.inf
        return 0

    bounds = [(0, 1) for _ in range(n)]
    cons = {'type': 'ineq', 'fun': constraint}

    # First optimization with uniform grid
    result = minimize(objective, initial_radii, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-8)

    # Refine by adjusting boundary circles
    for i in range(n):
        x, y = centers[i]
        if x < 0.2 or x > 0.8 or y < 0.2 or y > 0.8:
            result.x[i] *= 1.2

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result.x[i] = 0.0

    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
