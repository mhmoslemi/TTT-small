import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial configuration: 5 columns, 6 rows with offset rows
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        if row % 2 == 0:
            x = col + 0.5
        else:
            x = col + 0.5 - 0.5 * (1 if col == 0 else 0)
        y = row + 0.5
        centers[i] = [x, y]
    # Scale centers to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    radii = np.full(n, 0.1)

    def objective(radii):
        return -np.sum(radii)

    def constraint(radii):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12 or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return -np.inf
        return 0

    bounds = [(0, 1) for _ in range(n)]
    cons = {'type': 'ineq', 'fun': constraint}

    # Initial guess with better scaling
    radii_guess = np.linspace(0.05, 0.2, n)

    result = minimize(objective, radii_guess, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})

    radii = result.x

    for i in range(n):

        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
