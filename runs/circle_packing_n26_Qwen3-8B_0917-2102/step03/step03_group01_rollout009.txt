import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Hybrid grid: 5 rows with staggered columns
    row_height = 1.0 / (n ** 0.5)
    col_width = 1.0 / (n ** 0.5)
    for i in range(n):
        row = i // 5
        col = i % 5
        if row % 2 == 0:
            x = col * col_width + 0.5 * col_width
        else:
            x = (col + 0.5) * col_width
        y = row * row_height + 0.5 * row_height
        centers[i] = [x, y]
        radii[i] = 0.1
    
    def objective(radii):
        return -np.sum(radii)
    
    # Constraints: non-overlap and boundary
    def constraint(radii):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8 or y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8:
                return -np.inf
        return np.sum(radii) - 0.01
    
    cons = [{'type': 'ineq', 'fun': constraint}]
    
    result = minimize(objective, radii, method='SLSQP', constraints=cons, bounds=[(0, 1) for _ in range(n)])
    
    # Final adjustment for spacing and boundary constraints
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result.x[i] + result.x[j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result.x[i] + result.x[j])
                result.x[i] *= ratio
                result.x[j] *= ratio
    
    # Final boundary check
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
            result.x[i] = 0.0
    
    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
