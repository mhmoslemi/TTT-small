import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initialize centers in a hybrid hexagonal and boundary-aligned configuration
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Place circles in a staggered hexagonal grid with boundary alignment
    for i in range(n):
        row = i // 6
        col = i % 6
        if row % 2 == 0:
            x = col + 0.5
        else:
            x = col + 0.5 - 0.5
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = 0.1

    # Convert to [0,1]^2 with some boundary alignment
    centers[:, 0] = (centers[:, 0] - centers.min(axis=0)[0]) / (centers.max(axis=0)[0] - centers.min(axis=0)[0])
    centers[:, 1] = (centers[:, 1] - centers.min(axis=0)[1]) / (centers.max(axis=0)[1] - centers.min(axis=0)[1])

    def objective(radii):
        return -np.sum(radii)

    # Constraints: non-overlapping and within square
    def constraint(radii):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        return np.sum(radii) - 0.01

    cons = [{'type': 'ineq', 'fun': constraint}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP')

    radii = result.x

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
