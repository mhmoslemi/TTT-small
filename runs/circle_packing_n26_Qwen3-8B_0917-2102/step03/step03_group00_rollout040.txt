import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Initial cluster-based arrangement with 5 clusters of 5 circles
    cluster_centers = np.array([
        [0.2, 0.2], [0.5, 0.2], [0.8, 0.2], [0.3, 0.5], [0.7, 0.5],
        [0.2, 0.8], [0.5, 0.8], [0.8, 0.8], [0.3, 0.5], [0.7, 0.5],
        [0.2, 0.5], [0.5, 0.5], [0.8, 0.5], [0.3, 0.5], [0.7, 0.5],
        [0.2, 0.7], [0.5, 0.7], [0.8, 0.7], [0.3, 0.5], [0.7, 0.5],
        [0.2, 0.4], [0.5, 0.4], [0.8, 0.4], [0.3, 0.5], [0.7, 0.5]
    ])
    for i in range(n):
        centers[i] = cluster_centers[i]

    def objective(r):
        return -np.sum(r)

    def constraint_sum(r):
        return np.sum(r) - 2.636

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap}]
    cons.append({'type': 'eq', 'fun': constraint_sum})

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final check and adjustment
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
