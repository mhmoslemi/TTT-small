import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial non-hexagonal layout with boundary-aware spacing
    centers = np.zeros((n, 2))
    for i in range(n):
        if i < 13:
            centers[i] = [i / 12, 0.5]
        else:
            centers[i] = [(i - 13) / 12, 0.75]

    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)

    # Initial guess for radii
    radii = np.full(n, 0.1)

    # Constraints: non-overlap and boundary
    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        for i in range(n):
            x, y = centers[i]
            if x - r[i] < -1e-12 or x + r[i] > 1 + 1e-12 or y - r[i] < -1e-12 or y + r[i] > 1 + 1e-12:
                return -np.inf
        return 0

    cons = [{'type': 'ineq', 'fun': constraint}]

    # Optimizer parameters
    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'gtol': 1e-10, 'maxiter': 500})

    radii = result.x

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
