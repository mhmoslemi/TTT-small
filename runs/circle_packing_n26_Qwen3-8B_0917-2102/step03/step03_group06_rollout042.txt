import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Generate adaptive hexagonal grid with variable spacing
    centers = []
    for row in range(4):

            x = col / 5.0
            y = row / 3.0 + 0.5
            centers.append([x, y])

            x = col / 4.0 + 0.5
            y = row / 3.0 + 0.5
            centers.append([x, y])
    centers = np.array(centers)
    
    radii = np.full(n, 0.1)

    def objective(radii):
        return -np.sum(radii)

                    return -np.inf

                return -np.inf
        return 0

    bounds = [(0, 1) for _ in range(n)]
    cons = {'type': 'ineq', 'fun': constraint}

    result = minimize(objective, radii, bounds=bounds, constraints=cons, method='SLSQP')

    radii = result.x

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
