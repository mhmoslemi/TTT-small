from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

    # Create an optimal hexagonal arrangement with boundary-aware scaling

    row = 0

        y = row + 0.5

            row += 1

    scale_y = (1 - 1e-8) / (max_y - min_y)
    centers = np.column_stack([
        (centers[:, 0] - min_x) * scale_x,
        (centers[:, 1] - min_y) * scale_y

        return -np.sum(r)

        for i in range(n):
            for j in range(i+1, n):

    return centers, result.x, sum_radii
