import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Define a staggered hexagonal arrangement with varying row lengths
    centers = []
    for row in range(5):
        for col in range(5 - (row % 2)):
            x = col + 0.5 * (1 - 2 * (row % 2))
            y = row + 0.5
            centers.append([x, y])
    centers = np.array(centers)

    # Initial radii based on minimum pairwise distance
    dists = squareform(pdist(centers))
    min_dist = np.min(dists)
    radii = np.full(n, min_dist / 2.5)

    def objective(r):
        return -np.sum(r)

    def constraint_sum(r):
        return np.sum(r) - 2.636

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap}]
    cons.append({'type': 'eq', 'fun': constraint_sum})

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment to ensure boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    # Ensure no overlaps after adjustment
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):

                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
