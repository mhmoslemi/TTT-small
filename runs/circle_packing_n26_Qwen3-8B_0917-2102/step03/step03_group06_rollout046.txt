import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Initial hexagonal grid with variable column counts
    col_counts = [5, 5, 5, 5, 5]
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = 0.1

    # Scale to unit square while preserving spacing
    min_x, max_x = centers[:, 0].min(), centers[:, 0].max()
    min_y, max_y = centers[:, 1].min(), centers[:, 1].max()
    centers[:, 0] = (centers[:, 0] - min_x) / (max_x - min_x)
    centers[:, 1] = (centers[:, 1] - min_y) / (max_y - min_y)

    def objective(radii):
        return -np.sum(radii)

    def constraint(radii):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        for i in range(n):
            x, y = centers[i]
            if (x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12 or
                y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12):
                return -np.inf
        return 0

    bounds = [(0, 1) for _ in range(n)]
    cons = {'type': 'ineq', 'fun': constraint}

    result = minimize(objective, radii, bounds=bounds, constraints=cons, method='SLSQP')

    radii = result.x

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
