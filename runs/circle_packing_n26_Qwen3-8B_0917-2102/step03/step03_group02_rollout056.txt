import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Hexagonal packing grid with 5 columns and 6 rows
    centers = []
    for row in range(6):
        for col in range(5):
            x = col + 0.5 * (row % 2)
            y = row + 0.5 * (row % 2)
            centers.append([x, y])
    centers = np.array(centers)

    dists = squareform(pdist(centers))

            if (x - r_i < -1e-12 or x + r_i > 1 + 1e-12
                or y - r_i < -1e-12 or y + r_i > 1 + 1e-12):
                return -np.inf
        return 0

    def constraint2(r):
        # Ensure circles do not overlap
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return 0

    cons = [{'type': 'ineq', 'fun': constraint1}, {'type': 'ineq', 'fun': constraint2}]

    return centers, result.x, sum_radii
