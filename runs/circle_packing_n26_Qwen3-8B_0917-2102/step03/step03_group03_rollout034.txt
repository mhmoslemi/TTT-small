from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Generate hexagonal grid with boundary-aware placement
    centers = np.zeros((n, 2))
    col = 0
    row = 0
    for i in range(n):
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        col += 1
        if col >= 5:
            col = 0
            row += 1

    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

        return -np.sum(r)

                    return -np.inf
        for i in range(n):

            if x - r[i] < -1e-12 or x + r[i] > 1 + 1e-12 or y - r[i] < -1e-12 or y + r[i] > 1 + 1e-12:
                return -np.inf
        return 0

    cons = [{'type': 'ineq', 'fun': constraint}]

    def perturb_centers(centers, perturb=0.05):
        return centers + np.random.normal(0, perturb, centers.shape)

    best_sum = 0
    best_centers = centers
    best_radii = radii

        radii = np.full(n, 0.1)

        result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'gtol': 1e-10, 'maxiter': 500})
