from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    centers = np.zeros((n, 2))
    # Generate a hybrid hexagonal and compact packing
    for i in range(n):
        if i % 2 == 0:
            row = i // 2
            col = i % 2
            x = col + 0.1
            y = row + 0.1
        else:
            row = (i - 1) // 2
            col = (i - 1) % 2
            x = 0.9 - col * 0.1
            y = row + 0.1
        centers[i] = [x, y]

        return -np.sum(r)

                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

        dists = squareform(pdist(centers))

                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636
    
    # Initial guess based on minimum inter-circle distance

    initial_r = np.min(dists) * np.ones(n)

    result = minimize(objective, initial_r, constraints=[
        {'type': 'ineq', 'fun': check_overlap},
