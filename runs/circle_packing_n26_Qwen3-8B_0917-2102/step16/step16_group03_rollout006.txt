from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

    initial_r = np.min(dists) * np.ones(n)
    
    # Optimization with SLSQP
    result = minimize(
        objective, 
        initial_r,
        constraints=[{'type': 'ineq', 'fun': constraint_func}],
