import numpy as np
from scipy.optimize import minimize, LinearConstraint
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create hexagonal grid with boundary-aware placement
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (row % 2)
        y = row + 0.5
        centers[i] = [x, y]
    
    # Apply boundary-aware scaling
    min_coords = np.min(centers, axis=0)
    max_coords = np.max(centers, axis=0)
    scale = (1 - 2 * 0.1) / (max_coords - min_coords)
    centers = (centers - min_coords) * scale + 0.1
    centers = np.clip(centers, 0.1, 0.9)

    def objective(r):
        return -np.sum(r)

    dists = squareform(pdist(centers))
    min_dist = np.min(dists)
    initial_r = np.ones(n) * min_dist

    # Create constraints
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append((dists[i, j] - (initial_r[i] + initial_r[j])))

    A = np.zeros((len(constraints), n))
    for idx, (i, j) in enumerate([(a, b) for a in range(n) for b in range(a + 1, n)]):
        A[idx, i] = 1
        A[idx, j] = 1

    constraints = LinearConstraint(A, -np.inf, 0)

    result = minimize(objective, initial_r, constraints=constraints, bounds=[(0, 1)] * n, method='SLSQP')

    # Final adjustment for overlap with boundary-aware scaling
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2 with boundary padding
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
