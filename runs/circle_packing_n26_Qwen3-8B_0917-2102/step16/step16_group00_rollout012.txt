import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal grid with adaptive row spacing
    centers = []
    for row in range(4):
        for col in range(6):
            x = col + 0.5 * (row % 2)
            y = row + 0.5
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
    centers = np.array(centers)
    
    # Add corner circles if space allows
    if len(centers) < n:
        corners = np.array([[0, 0], [1, 0], [1, 1], [0, 1]])
        centers = np.vstack((centers, corners))
    
    # Optimize radii with boundary and overlap constraints
    def objective(r):
        return -np.sum(r)
    
    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636
    
    initial_r = np.min(squareform(pdist(centers)), axis=1) * np.ones(n)
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': constraint}], method='SLSQP', bounds=[(0, 1)] * n)
    
    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0
    
    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
