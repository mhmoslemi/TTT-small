import numpy as np
from scipy.optimize import differential_evolution
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Initial hexagonal grid with boundary-aware placement
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    def objective(r):
        return -np.sum(r)

    def check_constraints(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1

            r_i = r[i]
            if x - r_i < -1e-12 or x + r_i > 1 + 1e-12 or y - r_i < -1e-12 or y + r_i > 1 + 1e-12:
                return -1
        return 0

    bounds = [(0, 1)] * n

    result = differential_evolution(objective, bounds, constraints=[check_constraints], popsize=20, maxiter=100, tol=1e-6)

    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
