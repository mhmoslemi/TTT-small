from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    def objective(r):
        return -np.sum(r)
    
    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):

                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf

    def boundary_constraint(r):
        for i in range(n):
            x, y = centers[i]
            r_val = r[i]
            if x - r_val < -1e-12 or x + r_val > 1 + 1e-12 or y - r_val < -1e-12 or y + r_val > 1 + 1e-12:
                return -np.inf
        return np.sum(r)
    
    initial_r = np.min(squareform(pdist(centers))) * np.ones(n)
    result = minimize(objective, initial_r, constraints=[
        {'type': 'ineq', 'fun': constraint},
        {'type': 'ineq', 'fun': boundary_constraint}
    ], method='SLSQP', bounds=[(0, 1)] * n)
    
    # Final adjustment for boundary constraints
    for i in range(n):

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
