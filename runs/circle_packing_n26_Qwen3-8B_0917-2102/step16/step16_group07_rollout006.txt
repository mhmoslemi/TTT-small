import numpy as np
from scipy.optimize import differential_evolution
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal packing with boundary-aware adjustment
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    # Apply boundary-aware scaling
    min_coords = np.min(centers, axis=0)
    max_coords = np.max(centers, axis=0)
    scale = (1 - 2 * 0.1) / (max_coords - min_coords)
    centers = (centers - min_coords) * scale + 0.1
    centers = np.clip(centers, 0.1, 0.9)

    def objective(r):
        return -np.sum(r)

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    bounds = [(0, 1)] * n

    result = differential_evolution(
        fun=objective,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': check_overlap}
        ],
        popsize=15,
        maxiter=500,
        strategy='best1bin',
        tol=1e-10,
        mutation=(0.5, 1),
        recombination=0.8
    )

    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
