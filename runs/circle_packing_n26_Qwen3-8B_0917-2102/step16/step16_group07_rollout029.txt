import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate initial grid with boundary offset
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col * 0.25 + 0.1
        y = row * 0.25 + 0.1
        centers[i] = [x, y]
    
    # Define objective to maximize sum of radii
    def objective(r):
        return -np.sum(r)
    
    # Define constraints for non-overlapping and boundary
    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):

                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        for i in range(n):
            x, y = centers[i]
            r_i = r[i]
            if (x - r_i < -1e-12 or x + r_i > 1 + 1e-12
                    or y - r_i < -1e-12 or y + r_i > 1 + 1e-12):
                return -np.inf
        return np.sum(r) - 2.636

    # Initial guess based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)

    # Optimization with bounds and constraints
    result = minimize(objective, initial_r, method='SLSQP', bounds=[(0, 1)] * n,
                      constraints=[{'type': 'ineq', 'fun': constraint}])
    
    # Final adjustment for boundary and overlap
    for i in range(n):
        x, y = centers[i]

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
