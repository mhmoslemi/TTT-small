from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Generate hybrid hexagonal grid with variable row spacing and boundary-aware offsets
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row + 0.5
        # Apply boundary-aware offset to keep circles near edges
        if row % 2 == 0:
            x += 0.1 * (col / 4)
        else:
            x -= 0.1 * (col / 4)
        centers[i] = [x, y]

        return 0

    def constraint(r):
        dists = squareform(pdist(centers))

            for j in range(i + 1, n):

                    return -np.inf

        for i in range(n):
            x, y = centers[i]
            r = r[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12

                return -np.inf

    for i in range(n):
