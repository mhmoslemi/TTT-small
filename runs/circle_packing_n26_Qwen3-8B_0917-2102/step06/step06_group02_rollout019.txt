from scipy.optimize import minimize

    centers = []

            y = row + 0.5
            centers.append([x, y])
    
    # Convert to numpy array
    centers = np.array(centers)
    
    # Scale to unit square
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))
    
    def objective(r):
        return -np.sum(r)
    
    def constraint1(r):
        return np.sum(r) - 2.636

                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    cons = [{'type': 'ineq', 'fun': constraint2}]
    cons.append({'type': 'eq', 'fun': lambda r: np.sum(r) - 2.636})
    
    result = minimize(objective, np.full(n, 0.1), constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
