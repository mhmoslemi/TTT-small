import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create staggered hexagonal grid with 5 columns and 6 rows
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        if row % 2 == 0:
            x = col + 0.5
        else:
            x = col + 0.5 - 0.5
        y = row + 0.5
        centers[i] = [x, y]
    
    # Scale centers to [0,1]^2
    min_vals = np.min(centers, axis=0)
    max_vals = np.max(centers, axis=0)
    centers = (centers - min_vals) / (max_vals - min_vals)

    # Initial radii based on minimal spacing
    dists = squareform(pdist(centers))
    initial_radii = np.min(dists) / 2.5
    radii = np.full(n, initial_radii)

    def objective(r):
        return -np.sum(r)

    def constraint_sum(r):
        return np.sum(r) - 2.636

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    # Optimization bounds
    bounds = [(0, 1)] * n

    # Constraints
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    cons.append({'type': 'ineq', 'fun': constraint_overlap})

    # Optimize
    result = minimize(objective, radii, bounds=bounds, constraints=cons, method='SLSQP')

    # Final check and adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
