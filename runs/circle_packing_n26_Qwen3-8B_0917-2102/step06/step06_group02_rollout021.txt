import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    n = 26
    # Initialize with a staggered hexagonal grid inside [0,1]^2
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        if row % 2 == 0:
            x = col + 0.5
        else:
            x = col + 0.5 - 0.5
        y = row + 0.5
        centers[i] = [x, y]
    
    def objective(r):
        return -np.sum(r)
    
    def constraint(r):
        return np.sum(r) - 2.636
    
    def overlap_constraint(r):
        dists = squareform(pdist(centers))
        for i, j in combinations(range(n), 2):
            if dists[i, j] < r[i] + r[j] - 1e-8:
                return -1
        return 0
    
    # Use a modified SLSQP with bounds and constraints
    bounds = [(0, 1)] * n
    cons = [{'type': 'ineq', 'fun': constraint}, 
            {'type': 'ineq', 'fun': overlap_constraint}]
    
    result = minimize(objective, np.full(n, 0.1), bounds=bounds, constraints=cons, method='SLSQP')
    
    # Post-processing to ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0
    
    return centers, result['x'], np.sum(result['x'])
