import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate a more compact hexagonal grid with boundary-aware placement
    centers = np.zeros((n, 2))
    row = 0
    col = 0
    for i in range(n):
        if i % 5 == 0:
            row += 1
            col = 0
        x = col + 0.5 * (1 - 2 * (row % 2)) * 0.8
        y = row + 0.5
        centers[i] = [x, y]
        col += 1
    
    # Scale to [0,1]^2 with boundary constraints
    def scale_to_unit(bounds, centers):
        min_x, max_x = bounds[0][0], bounds[1][0]
        min_y, max_y = bounds[0][1], bounds[1][1]
        scaled_centers = np.zeros_like(centers)
        for i in range(n):
            scaled_centers[i, 0] = (centers[i, 0] - min_x) / (max_x - min_x)
            scaled_centers[i, 1] = (centers[i, 1] - min_y) / (max_y - min_y)
        return scaled_centers
    
    bounds = np.array([[0, 1], [0, 1]])
    centers = scale_to_unit(bounds, centers)

    def objective(r):
        return -np.sum(r)

    def constraint_sum(r):
        return np.sum(r) - 2.636

    def constraint_nonneg(r):
        return np.array([r[i] for i in range(n)])

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_nonneg}]
    cons.append({'type': 'eq', 'fun': constraint_sum})
    cons.append({'type': 'ineq', 'fun': check_overlap})

    result = minimize(objective, np.full(n, 0.1), constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for overlap with boundary-aware scaling
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
