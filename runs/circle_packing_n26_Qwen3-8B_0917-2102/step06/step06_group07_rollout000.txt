import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create hexagonal grid with 5 columns and 6 rows, adding corner circles
    centers = np.zeros((n, 2))
    idx = 0
    for row in range(6):
        for col in range(5):
            x = col + 0.5
            y = row + 0.5
            if row % 2 == 0:
                centers[idx] = [x, y]
            else:
                centers[idx] = [x - 0.5, y]
            idx += 1
    # Add corner circles
    centers[23] = [0.1, 0.1]
    centers[24] = [0.9, 0.1]
    centers[25] = [0.9, 0.9]
    
    # Scale centers to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)

    def constraint_sum(r):
        return np.sum(r) - 2.636

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    cons.append({'type': 'ineq', 'fun': constraint_overlap})

    result = minimize(objective, np.full(n, 0.1), constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final check and adjustment for boundary constraints
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
