import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal packing with spacing
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5
        y = row + 0.5
        centers[i] = [x, y]

    # Refine initial positions for better packing
    def refine_positions(centers, n):
        refined = np.copy(centers)
        for i in range(n):
            for j in range(n):
                if i != j:
                    dx = refined[i, 0] - refined[j, 0]
                    dy = refined[i, 1] - refined[j, 1]
                    dist = np.hypot(dx, dy)
                    if dist < 1:
                        refined[i] += (dx / dist) * (1 - dist)
                        refined[j] -= (dx / dist) * (1 - dist)
        return refined

    centers = refine_positions(centers, n)

    def objective(r):
        return -np.sum(r)

    def constraint_sum(r):
        return np.sum(r) - 2.636

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap}]
    cons.append({'type': 'eq', 'fun': constraint_sum})

    result = minimize(objective, np.full(n, 0.1), constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Adjust radii to fix overlaps
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
