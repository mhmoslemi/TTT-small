import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal grid with 5 columns and 6 rows, offset rows
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        if row % 2 == 0:
            x = col + 0.5
        else:
            x = col + 0.5 - 0
        y = row + 0.5
        centers[i] = [x, y]
    
    # Stretch grid to fit within unit square while preserving aspect ratio
    x_min, x_max = centers[:, 0].min(), centers[:, 0].max()
    y_min, y_max = centers[:, 1].min(), centers[:, 1].max()
    centers[:, 0] = (centers[:, 0] - x_min) / (x_max - x_min)
    centers[:, 1] = (centers[:, 1] - y_min) / (y_max - y_min)

    def objective(r):
        return -np.sum(r)

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    def constraint_boundary(r):
        for i in range(n):
            x, y = centers[i]
            r_i = r[i]
            if (x - r_i < -1e-12 or x + r_i > 1 + 1e-12 or
                y - r_i < -1e-12 or y + r_i > 1 + 1e-12):
                return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap},
            {'type': 'ineq', 'fun': constraint_boundary}]

    result = minimize(objective, np.full(n, 0.1), constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final check and adjustment for boundary constraints
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
