import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate a hybrid hexagonal and nested arrangement
    centers = np.zeros((n, 2))
    row, col = 0, 0
    for i in range(n):
        if i % 5 == 0 and i != 0:
            row += 1
            col = 0
        if row % 2 == 0:
            x = col + 0.5
        else:
            x = col + 0.5 - 0
        y = row + 0.5
        centers[i] = [x, y]
        col += 1
        if col >= 5:
            col = 0

    # Scale centers to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    # Objective function to maximize sum of radii
    def objective(r):
        return -np.sum(r)

    # Constraint to ensure sum of radii is at least 2.636
    def constraint_sum(r):
        return np.sum(r) - 2.636

    # Constraint to ensure non-overlapping
    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    # Optimization bounds
    bounds = [(0, 1) for _ in range(n)]

    # Initial guess based on uniform distribution
    initial_guess = np.full(n, 0.1)

    # Optimization with constraints
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    cons.append({'type': 'ineq', 'fun': constraint_overlap})

    result = minimize(objective, initial_guess, constraints=cons, method='SLSQP', bounds=bounds)

    # Post-processing to adjust radii for boundary constraints
    dists = squareform(pdist(centers))
    for i in range(n):

                result['x'][i] *= ratio
                result['x'][j] *= ratio

    for i in range(n):
        x, y = centers[i]

        if (x - r < -1e-12 or x + r > 1 + 1e-12 or

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
