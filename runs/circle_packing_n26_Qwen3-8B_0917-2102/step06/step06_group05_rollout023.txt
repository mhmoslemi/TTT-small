import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initialize with a more refined staggered hexagonal grid
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        if row % 2 == 0:
            x = col + 0.25
        else:
            x = col + 0.25 - 0.5
        y = row + 0.25
        centers[i] = [x, y]
    
    # Scale centers to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    # Initial radii based on minimal spacing
    dists = squareform(pdist(centers))
    initial_radii = np.min(dists) / 2 * np.ones(n)

    def objective(r):
        return -np.sum(r)

    def constraint_sum(r):
        return np.sum(r) - 2.636

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap}]
    cons.append({'type': 'eq', 'fun': constraint_sum})

    result = minimize(objective, initial_radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment to ensure boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
