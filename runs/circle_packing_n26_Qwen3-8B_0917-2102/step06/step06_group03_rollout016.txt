import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Define hexagonal lattice coordinates
    centers = []
    for row in range(4):
        for col in range(6):
            x = col + 0.5
            y = row + 0.5
            if row % 2 == 1:
                x += 0.5
            centers.append([x, y])
    centers = np.array(centers)
    
    # Scale to unit square
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)
    
    def constraint_total(r):
        return np.sum(r) - 2.636
    
    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    cons = [{'type': 'ineq', 'fun': constraint_overlap}]
    cons.append({'type': 'eq', 'fun': lambda r: np.sum(r) - 2.636})
    
    result = minimize(objective, np.full(n, 0.1), constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Post-processing for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    return centers, result['x'], np.sum(result['x'])
