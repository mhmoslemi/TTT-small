import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate a more optimal hexagonal packing with rows
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5
        y = row + 0.5
        centers[i] = [x, y]

    # Scale to unit square
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 2.636

    cons = [{'type': 'ineq', 'fun': lambda r: np.array([r[i] for i in range(n)])}]
    cons.append({'type': 'eq', 'fun': constraint})

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    def constraint_overlap(r):
        return check_overlap(r)

    cons.append({'type': 'ineq', 'fun': constraint_overlap})

    result = minimize(objective, np.full(n, 0.1), constraints=cons, method='SLSQP', bounds=Bounds(0, 1), options={'maxiter': 500})

    # Refine with simulated annealing
    def objective_sa(r):
        return -np.sum(r)

    def constraint_sa(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    result_sa = minimize(objective_sa, result['x'], method='SLSQP', bounds=Bounds(0, 1), constraints=[{'type': 'ineq', 'fun': constraint_sa}], options={'maxiter': 200})

    # Ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result_sa['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result_sa['x'][i] = 0

    sum_radii = np.sum(result_sa['x'])
    return centers, result_sa['x'], sum_radii
