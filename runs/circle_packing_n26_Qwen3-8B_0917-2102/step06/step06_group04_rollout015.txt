import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create hexagonal lattice with optimal spacing
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5
        y = row + 0.5
        centers[i] = [x, y]

    # Normalize to unit square
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 2.636

    def overlap_constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    # Initial guess with uniform radii
    r0 = np.full(n, 0.1)

    # Bounds for radii
    bounds = Bounds(0, 1)

    # Optimization with adaptive constraints
    result = minimize(objective, r0, bounds=bounds, constraints=[
        {'type': 'ineq', 'fun': constraint},
        {'type': 'ineq', 'fun': overlap_constraint}
    ], method='SLSQP', options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-10})

    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
