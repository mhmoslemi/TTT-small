import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate initial hexagonal-like grid with variable row spacing
    centers = np.zeros((n, 2))
    row_height = 0.9
    col_width = 0.9
    row_space = 0.5
    for i in range(n):
        row = i // 5
        col = i % 5
        y = row * row_space + 0.5
        if row % 2 == 0:
            x = col * col_width + 0.5
        else:
            x = col * col_width + 0.5 - 0.5
        centers[i] = [x, y]

    # Scale centers to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 2.636

    cons = [{'type': 'ineq', 'fun': lambda r: np.array([r[i] for i in range(n)])}]
    cons.append({'type': 'eq', 'fun': constraint})

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    def constraint_overlap(r):
        return check_overlap(r)

    cons.append({'type': 'ineq', 'fun': constraint_overlap})

    # Initial guess with uniform radii
    initial_guess = np.full(n, 0.1)

    # Use simulated annealing with bounds
    bounds = [(0, 1)] * n
    result = minimize(objective, initial_guess, bounds=bounds, constraints=cons, method='SLSQP')

    # Adjust radii to fix overlaps
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
