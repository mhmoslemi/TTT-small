from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Generate hexagonal grid with staggered rows
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    def objective(r):
        return -np.sum(r)
    
    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf

    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': constraint}], method='SLSQP', bounds=[(0, 1)] * n)
    
    # Boundary-aware scaling
    def scale_to_unit(r):

        scale = (1 - 2 * 0.1) / (max_coords - min_coords)

        scaled_centers = np.clip(scaled_centers, 0.1, 0.9)
        return scaled_centers
    
    scaled_centers = scale_to_unit(result['x'])
