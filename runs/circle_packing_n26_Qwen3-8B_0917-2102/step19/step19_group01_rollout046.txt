import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.ones(n) * 0.05

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636

    bounds = [(0, 1)] * n

    result = minimize(objective, radii, constraints=[{'type': 'ineq', 'fun': constraint}], method='SLSQP', bounds=bounds)

    # Boundary-aware adjustment
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if x - r < -1e-12:
            result['x'][i] = x - 1e-12
        if x + r > 1 + 1e-12:
            result['x'][i] = 1 - 1e-12
        if y - r < -1e-12:
            result['x'][i] = 1e-12
        if y + r > 1 + 1e-12:
            result['x'][i] = 1 - 1e-12

    return centers, result['x'], np.sum(result['x'])
