import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    for i in range(n):
        if i < n // 2:
            centers[i, 0] = i / (n // 2 - 1)
        else:
            centers[i, 0] = 1 - (n - i - 1) / (n // 2 - 1)
        centers[i, 1] = i / (n - 1)

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636

    def bounds_func(r):
        for i in range(n):
            if (centers[i, 0] - r[i] < -1e-12 or centers[i, 0] + r[i] > 1 + 1e-12
                or centers[i, 1] - r[i] < -1e-12 or centers[i, 1] + r[i] > 1 + 1e-12):
                return -np.inf
        return 0

    initial_r = np.min(squareform(pdist(centers))) * np.ones(n)
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': constraint}, {'type': 'ineq', 'fun': bounds_func}], method='SLSQP', bounds=[(0, 1)] * n)

    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
