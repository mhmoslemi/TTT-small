from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    centers = np.zeros((n, 2))

        col = i % 5
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row + 0.5

            x, y = 0.15, 0.15

            x, y = 0.85, 0.15
        elif row == 4 and col == 0:
            x, y = 0.15, 0.85

            x, y = 0.85, 0.85

        return -np.sum(r)

                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636
