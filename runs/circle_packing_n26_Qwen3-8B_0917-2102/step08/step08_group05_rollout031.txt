import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    # Hexagonal grid with boundary-aware placement
    for i in range(n):
        row = i // 2
        col = i % 2
        x = col + 0.5 if row % 2 else col
        y = row + 0.5
        centers[i] = [x, y]
    
    # Normalize to [0,1]^2 with boundary padding
    padding = 0.1
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))
    centers = np.clip(centers, padding, 1 - padding)

    def objective(r):
        return -np.sum(r)

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    # Initial guess based on distance to boundaries and neighbors
    initial_guess = np.full(n, 0.1)
    for i in range(n):
        x, y = centers[i]
        dist_to_boundary = np.min([x, 1 - x, y, 1 - y])
        initial_guess[i] = np.clip(0.1 * dist_to_boundary, 0, 0.2)

    # Optimization constraints
    cons = []
    cons.append({'type': 'ineq', 'fun': check_overlap})
    cons.append({'type': 'ineq', 'fun': lambda r: np.array([r[i] for i in range(n)])})

    result = minimize(objective, initial_guess, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for overlap with boundary-aware scaling
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):

                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2 with boundary padding
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
