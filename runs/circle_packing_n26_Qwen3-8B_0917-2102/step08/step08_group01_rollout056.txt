import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Hexagonal packing with 5 rows: 6, 6, 6, 6, 2 circles
    centers = np.zeros((n, 2))
    idx = 0
    for row in range(5):
        for col in range(6 if row % 2 == 0 else 5):
            x = col + 0.5 if row % 2 == 0 else col + 0.5
            y = row + 0.5
            centers[idx] = [x, y]
            idx += 1

    # Normalize to [0,1]^2 with boundary padding
    padding = 0.05
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))
    centers = np.clip(centers, padding, 1 - padding)

    def objective(r):
        return -np.sum(r)

    # Constraints: non-negative radii, boundary, and non-overlap
    cons = [{'type': 'ineq', 'fun': lambda r: r}]

    def constraint_sum(r):
        return np.sum(r) - 2.636

    cons.append({'type': 'eq', 'fun': constraint_sum})

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons.append({'type': 'ineq', 'fun': check_overlap})

    # Bounds for radii
    bounds = [(0, 1)] * n

    # Optimization with more accurate method and scaling
    result = minimize(objective, np.full(n, 0.1), bounds=bounds, constraints=cons, method='SLSQP', tol=1e-10)

    # Final adjustment for overlap with boundary-aware scaling
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2 with boundary padding
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
