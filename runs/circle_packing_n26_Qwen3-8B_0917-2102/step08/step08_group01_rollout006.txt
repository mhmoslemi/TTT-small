import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Hexagonal packing with staggered rows
    centers = np.zeros((n, 2))
    row_height = np.sqrt(3)/2
    col_width = 1 - 2 * 0.1  # leave 10% padding on sides

    for i in range(n):
        row = i // 5
        col = i % 5
        x = col * (col_width / (5 - 1)) + 0.1
        y = row * row_height + 0.1 if row % 2 == 0 else row * row_height + 0.1 + row_height / 2
        centers[i] = [x, y]
    
    # Normalize to [0,1]^2 with boundary padding
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))
    centers = np.clip(centers, 0.1, 0.9)

    def objective(r):
        return -np.sum(r)

    # Constraints: non-negative radii, boundary, and non-overlap
    cons = [{'type': 'ineq', 'fun': lambda r: r}]

    def constraint_sum(r):
        return np.sum(r) - 2.636

    cons.append({'type': 'eq', 'fun': constraint_sum})

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons.append({'type': 'ineq', 'fun': check_overlap})

    # Bounds for radii
    bounds = [(0, 1)] * n

    # Optimization with more accurate method and scaling
    result = minimize(objective, np.full(n, 0.1), bounds=bounds, constraints=cons, method='SLSQP', tol=1e-10)

        for j in range(i + 1, n):

                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2 with boundary padding
    for i in range(n):

        if (x - r < -1e-12 or x + r > 1 + 1e-12

    sum_radii = np.sum(result['x'])
