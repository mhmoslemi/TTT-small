import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    cols = 5
    rows = 6
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // cols
        col = i % cols
        offset = 0.5 if row % 2 else 0
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    def objective(r):
        return -np.sum(r)
    
    def constraint(r):
        return np.sum(r) - 2.636
    
    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    def boundary_constraint(r):
        for i in range(n):
            x, y = centers[i]
            r_i = r[i]
            if (x - r_i < -1e-12 or x + r_i > 1 + 1e-12
                    or y - r_i < -1e-12 or y + r_i > 1 + 1e-12):
                return -1
        return 0
    
    cons = [{'type': 'ineq', 'fun': check_overlap},
            {'type': 'ineq', 'fun': boundary_constraint}]
    
    initial_r = np.min(pdist(centers)) * np.ones(n)
    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)
    
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio
    
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0
    
    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
