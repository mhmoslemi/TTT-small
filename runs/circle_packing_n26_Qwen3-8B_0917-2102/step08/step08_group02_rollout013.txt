import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create boundary-aligned hexagonal lattice with adaptive spacing
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        if row % 2 == 1:
            offset = 0.5
        else:
            offset = 0
        x = col + offset + np.random.uniform(-0.05, 0.05)
        y = row + 0.5 + np.random.uniform(-0.05, 0.05)
        centers[i] = [x, y]

    # Normalize to [0,1]^2 with minimal padding
    min_vals = centers.min(axis=0)
    max_vals = centers.max(axis=0)
    scale = (max_vals - min_vals)
    centers = (centers - min_vals) / scale
    padding = 0.05
    centers = np.clip(centers, padding, 1 - padding)

    def objective(r):
        return -np.sum(r)

    def boundary_constraint(r):
        return np.sum(r) - 2.636

    def overlap_constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': overlap_constraint}]
    cons.append({'type': 'eq', 'fun': lambda r: np.sum(r) - 2.636})

    result = minimize(objective, np.full(n, 0.1), constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for overlap with boundary-aware scaling
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):

                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2 with boundary padding
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
