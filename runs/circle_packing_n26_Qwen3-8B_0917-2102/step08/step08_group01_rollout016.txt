import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal packing with boundary-aware spacing
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 6
        col = i % 6
        if row % 2 == 1:
            col = 5 - col
        x = col * (1 + 1e-3) + 0.5 / (1 + 1e-3)
        y = row * (1 + 1e-3) + 0.5 / (1 + 1e-3)
        centers[i] = [x, y]
    
    def objective(r):
        return -np.sum(r)

    def constraint_sum(r):
        return np.sum(r) - 2.636

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    bounds = [(0, 1)] * n
    cons = [{'type': 'ineq', 'fun': check_overlap}]
    cons.append({'type': 'eq', 'fun': constraint_sum})

    result = minimize(objective, np.full(n, 0.1), bounds=bounds, constraints=cons, method='SLSQP', tol=1e-10)

    # Final adjustment for boundary and overlap
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2 with boundary padding
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
