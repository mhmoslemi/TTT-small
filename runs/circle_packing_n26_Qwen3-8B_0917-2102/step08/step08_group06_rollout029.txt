import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial staggered hexagonal lattice with adaptive spacing
    cols = 5
    rows = 6
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // cols
        col = i % cols
        offset = 0.5 if row % 2 else 0
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    def objective(r):
        return -np.sum(r)
    
    def constraint_sum(r):
        return np.sum(r) - 2.636
    
    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    # Initial guess with adaptive radius based on distances
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    
    # Optimization bounds
    bounds = [(0, 1)] * n
    
    # Constraints
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    cons.append({'type': 'ineq', 'fun': constraint_overlap})
    
    result = minimize(objective, initial_r, bounds=bounds, constraints=cons, method='SLSQP')
    
    # Adjust radii for boundary padding
    padding = 0.1
    adjusted_centers = np.copy(centers)
    for i in range(n):
        x, y = adjusted_centers[i]
        r = result['x'][i]
        if x - r < padding or x + r > 1 - padding or y - r < padding or y + r > 1 - padding:
            result['x'][i] = 0
    
    return adjusted_centers, result['x'], np.sum(result['x'])
