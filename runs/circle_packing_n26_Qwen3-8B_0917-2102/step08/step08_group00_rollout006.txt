import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Hexagonal close packing with boundary-aware scaling
    centers = np.zeros((n, 2))
    row = 0
    col = 0
    for i in range(n):
        if i % 2 == 0:
            centers[i] = [col, row]
        else:
            centers[i] = [col + 0.5, row + 0.5]
        col += 1
        if col >= 5:
            col = 0
            row += 1
    
    # Initial radius based on minimum distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    
    def objective(r):
        return -np.sum(r)
    
    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    # Optimization with boundary constraints
    cons = [{'type': 'ineq', 'fun': check_overlap}]
    bounds = [(0, 1)] * n
    
    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=bounds)
    
    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0
    
    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
