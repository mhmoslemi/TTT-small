import numpy as np
from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial staggered hexagonal lattice
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 if row % 2 else 0
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 2.636

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    def update_centers(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    ratio = (dists[i, j] - 1e-8) / (r[i] + r[j])
                    r[i] *= ratio
                    r[j] *= ratio
        return r

    def boundary_constraint(r):
        for i in range(n):
            x, y = centers[i]
            r = r[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return -1
        return 0

    cons = [
        {'type': 'ineq', 'fun': lambda r: np.sum(r)},
        {'type': 'eq', 'fun': lambda r: np.sum(r) - 2.636},
        {'type': 'ineq', 'fun': check_overlap},
        {'type': 'ineq', 'fun': boundary_constraint}
    ]

    initial_r = np.min(squareform(pdist(centers))) * np.ones(n)
    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    result['x'] = update_centers(result['x'])

    return centers, result['x'], np.sum(result['x'])
