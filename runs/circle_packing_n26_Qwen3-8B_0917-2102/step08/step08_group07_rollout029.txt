import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.01)

    def objective(r):
        return -np.sum(r)

    def constraint1(r):
        return np.sum(r)

    def constraint2(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [
        {'type': 'ineq', 'fun': constraint1},
        {'type': 'ineq', 'fun': constraint2}
    ]

    def boundary_constraint(r):
        for i in range(n):
            x, y = centers[i]
            r_val = r[i]
            if x - r_val < -1e-12 or x + r_val > 1 + 1e-12 or y - r_val < -1e-12 or y + r_val > 1 + 1e-12:
                return -1
        return 0

    cons.append({'type': 'ineq', 'fun': boundary_constraint})

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for boundary and overlap
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            result['x'][i] = 0

    return centers, result['x'], np.sum(result['x'])
