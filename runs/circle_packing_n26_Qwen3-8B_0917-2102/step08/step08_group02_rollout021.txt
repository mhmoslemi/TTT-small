import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Generate hexagonal packing with adaptive spacing
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 if row % 2 else 0
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    # Scale to [0,1]^2 with boundary padding
    def scale_to_unit(centers, padding=0.1):
        min_vals = centers.min(axis=0)
        max_vals = centers.max(axis=0)
        scale = (1 - 2 * padding) / (max_vals - min_vals)
        scaled = (centers - min_vals) * scale + padding
        return scaled

    centers = scale_to_unit(centers)

    def objective(r):
        return -np.sum(r)

    def constraint_sum(r):
        return np.sum(r) - 2.636

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap}]
    cons.append({'type': 'eq', 'fun': constraint_sum})

    # Initial guess based on hexagonal packing
    initial_radii = np.full(n, 0.1)
    result = minimize(objective, initial_radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for boundary and overlap
    for i in range(n):

        r = result['x'][i]
