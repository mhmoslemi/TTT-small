import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.ones(n) * 0.05

    def objective(r):
        return -np.sum(r)

    def constraint_sum(r):
        return np.sum(r) - 2.636

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [
        {'type': 'ineq', 'fun': constraint_overlap},
        {'type': 'eq', 'fun': constraint_sum}
    ]

    def bounds_func(x):
        return [(0, 1)] * n

    result = minimize(objective, radii, method='SLSQP', bounds=bounds_func, constraints=cons)

    # Refine positions to avoid boundary collisions
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if x - r < 0:
            centers[i, 0] = r
        elif x + r > 1:
            centers[i, 0] = 1 - r
        if y - r < 0:
            centers[i, 1] = r
        elif y + r > 1:
            centers[i, 1] = 1 - r

    return centers, result['x'], np.sum(result['x'])
