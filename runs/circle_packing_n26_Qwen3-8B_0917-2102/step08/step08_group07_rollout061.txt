import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal lattice with staggered rows
    centers = []
    for row in range(3):
        for col in range(5):
            x = col + 0.5 * (row % 2)
            y = row + 0.5
            if row == 0:
                y += 0.25
            elif row == 2:
                y -= 0.25
            centers.append([x, y])
    
    # Add remaining circles in a compact cluster
    for i in range(3, 5):
        centers.append([0.5, 0.5 + 0.2 * i])
    
    centers = np.array(centers)
    
    # Ensure centers are within [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)

    def constraint_sum(r):
        return np.sum(r) - 2.636

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    # Initial guess based on minimal distance between circles
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)

    # Set up optimization problem
    cons = [
        {'type': 'ineq', 'fun': constraint_overlap},
        {'type': 'eq', 'fun': constraint_sum}
    ]

    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
