import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Custom non-uniform layout to better utilize corners and edges
    centers = np.zeros((n, 2))
    positions = np.random.rand(n, 2)
    positions[:, 0] = np.sort(positions[:, 0])
    for i in range(n):
        if i < n // 2:
            centers[i] = [positions[i, 0], 0.2 + positions[i, 1] * 0.6]
        else:
            centers[i] = [0.2 + positions[i, 0] * 0.6, positions[i, 1]]
    
    # Normalize to [0,1]^2 with boundary padding
    padding = 0.05
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))
    centers = np.clip(centers, padding, 1 - padding)

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 2.636

    cons = [{'type': 'ineq', 'fun': lambda r: np.array([r[i] for i in range(n)])}]
    cons.append({'type': 'eq', 'fun': constraint})

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons.append({'type': 'ineq', 'fun': check_overlap})

    # Initial guess based on average nearest distance
    dists = squareform(pdist(centers))
    avg_dist = np.mean(dists[np.triu_indices(n, 1)])
    r0 = np.full(n, avg_dist * 0.4)

    result = minimize(objective, r0, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for overlap with boundary-aware scaling
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2 with boundary padding
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
