import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Hexagonal close packing with 5 rows
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        if row % 2 == 0:
            x = col + 0.5
        else:
            x = col + 0.5 - 0.5 / (5 - 1)
        y = row + 0.5
        centers[i] = [x, y]
    
    # Normalize to [0,1]^2 with boundary padding
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))
    centers = np.clip(centers, 0.05, 0.95)

    def objective(r):
        return -np.sum(r)

    def constraint_sum(r):
        return np.sum(r) - 2.636

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    bounds = [(0, 1)] * n
    cons = [{'type': 'ineq', 'fun': check_overlap}]
    cons.append({'type': 'eq', 'fun': constraint_sum})

    result = minimize(objective, np.full(n, 0.1), bounds=bounds, constraints=cons, method='SLSQP', tol=1e-10)

    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
