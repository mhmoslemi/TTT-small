import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Hexagonal packing with boundary-aware adjustment
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 2
        col = i % 2
        x = col + 0.5 * row
        y = row + 0.5
        centers[i] = [x, y]
    
    # Adjust for boundary constraints
    def adjust_boundary(centers):
        min_x, max_x = np.min(centers[:, 0]), np.max(centers[:, 0])
        min_y, max_y = np.min(centers[:, 1]), np.max(centers[:, 1])
        scale_x = (1 - 2 * 0.1) / (max_x - min_x)
        scale_y = (1 - 2 * 0.1) / (max_y - min_y)
        centers[:, 0] = (centers[:, 0] - min_x) * scale_x + 0.1
        centers[:, 1] = (centers[:, 1] - min_y) * scale_y + 0.1
        return np.clip(centers, 0.1, 0.9)

    centers = adjust_boundary(centers)

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 2.636

    # Constraints
    cons = []
    cons.append({'type': 'ineq', 'fun': lambda r: np.sum(r)})
    cons.append({'type': 'eq', 'fun': constraint})

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons.append({'type': 'ineq', 'fun': check_overlap})

    # Initial guess based on minimal distances
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for overlap with boundary-aware scaling
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:

                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2
    for i in range(n):

        r = result['x'][i]

            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
