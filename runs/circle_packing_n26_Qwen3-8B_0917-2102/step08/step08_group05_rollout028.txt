from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Generate hexagonal grid with boundary-aware coordinates
    centers = []

            x = col + 0.5 * (row % 2)
            y = row + 0.5
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
    centers = np.array(centers)

    radii = np.zeros(n)

        return -np.sum(r)
    
    def constraint(r):
        return np.sum(r) - 2.636
    
    cons = [{'type': 'ineq', 'fun': lambda r: np.array([r[i] for i in range(n)])}]

        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    cons.append({'type': 'ineq', 'fun': check_overlap})
    
    # Initial guess with varying radii based on position
    initial_guess = np.full(n, 0.1)

        dist_to_boundary = np.min([x, 1 - x, y, 1 - y])
        initial_guess[i] = np.clip(0.1 * dist_to_boundary, 0, 0.2)
    
    result = minimize(objective, initial_guess, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)
