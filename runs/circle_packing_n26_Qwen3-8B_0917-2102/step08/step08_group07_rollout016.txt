from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

    # Hexagonal packing with alternating row offsets

    row_indices = np.floor(np.linspace(0, n - 1, n)).astype(int)

        row = row_indices[i]
        col = col_indices[i]
        x = col / 5.0 + 0.1
        y = row / 5.0 + 0.1
        centers[i] = [x, y]

        return 0

    cons.append({'type': 'ineq', 'fun': check_overlap})

    # Initial guess based on minimum pairwise distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
