import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create a hybrid arrangement: 5 rows with 5 circles, 1 row with 6 circles
    centers = []
    for i in range(n):
        if i < 20:
            row = i // 5
            col = i % 5
            x = col + 0.5 * (row % 2)
            y = row + 0.5
        else:
            row = 4
            col = i - 20
            x = col + 0.5 * (row % 2)
            y = row + 0.5
        centers.append([x, y])
    centers = np.array(centers)
    
    # Adjust for boundary-aware spacing
    def boundary_adjustment(centers, padding=0.1):
        min_coords = np.min(centers, axis=0)
        max_coords = np.max(centers, axis=0)
        scale = (1 - 2 * padding) / (max_coords - min_coords)
        centers = (centers - min_coords) * scale + padding
        centers = np.clip(centers, padding, 1 - padding)
        return centers
    
    centers = boundary_adjustment(centers)

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636

    # Initial guess based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': lambda r: 0}], method='SLSQP', bounds=[(0, 1)] * n)

    # Refine with constraint on total radii
    result = minimize(objective, result['x'], constraints=[{'type': 'eq', 'fun': lambda r: np.sum(r) - 2.636}], method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for overlap with boundary-aware scaling
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2 with boundary padding
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
