from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

        scale_y = (1 - 2 * padding) / dy if dy > 0 else 1
        scale = min(scale_x, scale_y)

        return np.clip(scaled_centers, padding, 1 - padding)

        return -np.sum(r)
