import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal packing with adjusted row spacing
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (row % 2)
        y = row + 0.5
        centers[i] = [x, y]
    
    # Adjust for boundary by shifting towards corners
    for i in range(n):
        x, y = centers[i]
        if x < 0.5:
            x = 0.5 - (0.5 - x) * 0.7
        elif x > 0.5:
            x = 0.5 + (x - 0.5) * 0.7
        if y < 0.5:
            y = 0.5 - (0.5 - y) * 0.7
        elif y > 0.5:
            y = 0.5 + (y - 0.5) * 0.7
        centers[i] = [x, y]
    
    # Compute initial radii based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    min_dist = np.min(dists)
    initial_r = np.full(n, min_dist * 0.9)
    
    def objective(r):
        return -np.sum(r)
    
    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636
    
    # Optimization with boundary constraints
    bounds = [(0, 1)] * n
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': constraint}], method='SLSQP', bounds=bounds)
    
    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            result['x'][i] = 0
    
    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
