import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Hexagonal packing with 5 rows: 5, 6, 7, 6, 2
    centers = np.zeros((n, 2))
    row_height = np.sqrt(3)/2 * 0.9
    for i in range(n):
        if i < 5:
            row = 0
            col = i
        elif i < 11:
            row = 1
            col = i - 5
        elif i < 18:
            row = 2
            col = i - 11
        elif i < 24:
            row = 3
            col = i - 18
        else:
            row = 4
            col = i - 24
        x = col / 4.5 + 0.1
        y = row * row_height + 0.1
        centers[i] = [x, y]
    
    def objective(r):
        return -np.sum(r)
    
    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636
    
    # Initial guess based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    
    # Optimization with bounds and constraints
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': check_overlap}, {'type': 'eq', 'fun': constraint}], method='SLSQP', bounds=[(0, 1)] * n)
    
    # Final adjustment for boundary overlap
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0
    
    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
