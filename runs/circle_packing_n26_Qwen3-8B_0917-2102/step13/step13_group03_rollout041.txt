import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal grid with optimized spacing
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    # Compute initial distances and padding
    dists = squareform(pdist(centers))
    min_dist = np.min(dists)
    padding = 0.1
    initial_r = np.ones(n) * (min_dist - 2 * padding) / 2
    
    def objective(r):
        return -np.sum(r)
    
    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636
    
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': constraint}], method='SLSQP', bounds=[(0, 1)] * n)
    
    # Final boundary check and adjustment
    final_r = result['x'].copy()
    for i in range(n):
        x, y = centers[i]
        r = final_r[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            final_r[i] = 0
    
    return centers, final_r, np.sum(final_r)
