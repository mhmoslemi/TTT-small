import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Initialize with hexagonal packing but random perturbations
    centers = np.zeros((n, 2))

        row = i // 5
        col = i % 5
        x = col + 0.5 * (row % 2)
        y = row + 0.5
        centers[i] = [x, y]
    centers += np.random.uniform(-0.1, 0.1, centers.shape)

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636

    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)

    # Use L-BFGS-B with bounds and constraints
    result = minimize(objective, initial_r, bounds=[(0, 1)] * n, constraints=[{'type': 'ineq', 'fun': constraint}], method='L-BFGS-B')

    # Ensure circles are within [0,1]^2 with boundary padding
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
