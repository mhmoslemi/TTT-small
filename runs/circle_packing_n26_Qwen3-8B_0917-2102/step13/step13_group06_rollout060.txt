import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create initial hexagonal grid with varying row heights
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    # Create boundary-aware perturbation
    for i in range(n):
        x, y = centers[i]
        if x < 0.5:
            centers[i, 0] = np.random.uniform(0.2, 0.5)
        elif x > 0.5:
            centers[i, 0] = np.random.uniform(0.5, 0.8)
        if y < 0.5:
            centers[i, 1] = np.random.uniform(0.2, 0.5)
        elif y > 0.5:
            centers[i, 1] = np.random.uniform(0.5, 0.8)
    
    def objective(r):
        return -np.sum(r)
    
    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636
    
    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    # Initial guess based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    
    # Optimization with multi-start
    best_sum = 0
    best_r = np.zeros(n)
    best_centers = centers.copy()
    
    for _ in range(3):
        perturb = np.random.uniform(-0.1, 0.1, centers.shape)
        centers_perturb = np.clip(centers + perturb, 0, 1)
        
        dists = squareform(pdist(centers_perturb))
        initial_r = np.min(dists) * np.ones(n)
        
        result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': check_overlap}, {'type': 'eq', 'fun': constraint}], method='SLSQP', bounds=[(0, 1)] * n)

            best_centers = centers_perturb

        r = best_r[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            best_r[i] = 0
    
    return best_centers, best_r, best_sum
