import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate a hexagonal grid with variable row heights
    centers = np.zeros((n, 2))
    row_heights = np.linspace(0.4, 0.6, 5)
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5
        y = row * row_heights[row] + 0.2
        centers[i] = [x, y]
    
    # Define objective and constraints
    def objective(r):
        return -np.sum(r)
    
    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636
    
    # Initial guess based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    
    # Optimize radii with constraints
    result = minimize(
        objective,
        initial_r,
        constraints=[{'type': 'ineq', 'fun': check_overlap}, {'type': 'eq', 'fun': constraint}],
        method='SLSQP',
        bounds=[(0, 1)] * n
    )
    
    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0
    
    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
