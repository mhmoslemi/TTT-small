import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create staggered hexagonal grid with variable row spacing
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 6
        col = i % 6
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row + 0.5
        if x > 1 - 1e-12:
            x = 1 - 1e-12
        if y > 1 - 1e-12:
            y = 1 - 1e-12
        centers[i] = [x, y]
    
    # Initial radii based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)

    def objective(r):
        return -np.sum(r)

    # Constraint to ensure no overlap
    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636

    # Optimization with boundary constraints
    bounds = [(0, 1)] * n
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': constraint_overlap}], method='SLSQP', bounds=bounds)

    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
