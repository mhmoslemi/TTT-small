import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal-like grid with variable spacing
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    # Define objective: maximize sum of radii
    def objective(r):
        return -np.sum(r)
    
    # Define constraint: no overlap between circles
    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r)
    
    # Define boundary constraint: circles within [0,1]^2
    def boundary_constraint(r):
        for i in range(n):
            x, y = centers[i]
            r_i = r[i]
            if (x - r_i < -1e-12 or x + r_i > 1 + 1e-12
                    or y - r_i < -1e-12 or y + r_i > 1 + 1e-12):
                return -np.inf
        return np.sum(r)
    
    # Initial guess: based on min inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    
    # Optimization using SLSQP with both constraints
    result = minimize(objective, initial_r, 
                     constraints=[
                         {'type': 'ineq', 'fun': constraint},
                         {'type': 'ineq', 'fun': boundary_constraint}
                     ],
                     method='SLSQP',
                     bounds=[(0, 1)] * n)
    
    # Final check and adjustment
    centers = np.array(centers)
    radii = result['x']
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0
    
    return centers, radii, np.sum(radii)
