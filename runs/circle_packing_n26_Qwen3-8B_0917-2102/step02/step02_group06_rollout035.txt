import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate initial hexagonal packing with varying row heights
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    row_height = 0.8
    col_spacing = 0.4
    row_spacing = row_height * np.sqrt(3)/2
    
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col * col_spacing
        y = row * row_spacing
        if row % 2 == 1:
            x += col_spacing / 2
        centers[i] = [x, y]
        radii[i] = 0.1
    
    # Normalize to [0,1]^2
    centers -= centers.min(axis=0)
    centers /= centers.max(axis=0) - centers.min(axis=0)
    
    # Objective: maximize sum of radii
    def objective(x):
        return -np.sum(x)
    
    # Constraints: ensure spacing between circles and boundary
    def constraint(x):
        return np.sum(x) - 1.0
    
    cons = [{'type': 'ineq', 'fun': constraint}]
    
    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)
    radii = result.x
    
    # Update radii to ensure spacing
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio
    
    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
