from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import product
import random

def run_packing():

    # Initial random placement with small radii
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n) * 0.05

        return -np.sum(x)
    
    def constraint(x):
        return np.sum(x) - 1.0
    
    cons = [{'type': 'ineq', 'fun': constraint}]
    
    # Optimization with random restarts
    best_sum = 0
    best_centers = centers.copy()
    best_radii = radii.copy()
    
    for _ in range(5):
        # Perturb initial positions
        centers = np.random.rand(n, 2)
        radii = np.random.rand(n) * 0.1
        
        # Scale to [0,1]^2
        centers -= centers.min(axis=0)
        centers /= centers.max(axis=0) - centers.min(axis=0)
        
        result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)
        radii = result.x
        
        # Adjust radii to maintain spacing
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j]:
                    ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                    radii[i] *= ratio
                    radii[j] *= ratio
        
        # Final check to ensure all circles are within [0,1]^2
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12

                radii[i] = 0.0
        
        current_sum = np.sum(radii)
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
    
    return best_centers, best_radii, best_sum
