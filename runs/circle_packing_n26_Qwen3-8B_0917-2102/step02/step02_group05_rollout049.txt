import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Hybrid hexagonal packing with boundary-aware positioning
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    base_r = 0.15
    for i in range(n):
        row = i // 5
        col = i % 5
        if row % 2 == 0:
            x = col + 0.5
        else:
            x = col + 0.5 + 0.5
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = base_r

    # Convert to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(x):
        return -np.sum(x)

    # Constraints for radii bounds
    cons = [{'type': 'ineq', 'fun': lambda x: x - 0.05},
            {'type': 'ineq', 'fun': lambda x: 1 - x - 0.05}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})

    # Update radii based on optimization
    radii = result.x

    # Compute pairwise distances
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
