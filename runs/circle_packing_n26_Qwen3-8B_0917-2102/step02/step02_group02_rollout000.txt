import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal grid with boundary-aware placement
    centers = []
    for row in range(4):
        for col in range(6):
            x = col / 5.0
            y = row / 3.0
            if row % 2 == 1:
                x += 0.5 / 5.0
            centers.append([x, y])
    centers = np.array(centers)
    
    # Initial radii based on minimum spacing
    initial_radii = np.full(n, 0.1)
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraint(radii):
        return np.sum(radii) - 0.01
    
    cons = [{'type': 'ineq', 'fun': constraint}]
    
    result = minimize(objective, initial_radii, constraints=cons, method='SLSQP')
    
    # Reshape to maintain spacing
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j]:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio
    
    # Final check and boundary adjustment
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            result['x'][i] = 0.0
    
    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
