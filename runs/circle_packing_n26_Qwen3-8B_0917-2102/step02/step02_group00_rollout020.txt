import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.ones(n) * 0.05

    def objective(x):
        return -np.sum(x)

    def constraint(x, i, j):
        return np.sqrt(np.sum((centers[i] - centers[j])**2)) - (radii[i] + radii[j])

    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: constraint(x, i, j)})

    def boundary_constraint(x, i):
        x_i, y_i = centers[i]
        return (x_i - radii[i]) >= -1e-12 and (x_i + radii[i]) <= 1 + 1e-12 and (y_i - radii[i]) >= -1e-12 and (y_i + radii[i]) <= 1 + 1e-12

    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: boundary_constraint(x, i)})

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})

    radii = result.x
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    return centers, radii, np.sum(radii)
