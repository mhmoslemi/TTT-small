import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Generate hexagonal grid points within [0,1]^2
    centers = []
    r = 0.1
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers.append([x, y])
    centers = np.array(centers)
    
    # Normalize centers to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    radii = np.full(n, 0.1)

        return -np.sum(x)

        return x - 0.01

        min_dist = np.min(dists)
        return 1 - (np.sum(x) / (n * np.min(x))) - (min_dist / np.min(x))
    
    cons = [{'type': 'ineq', 'fun': constraint_min}]
    
    # Optimize radii
    result = minimize(objective, radii, constraints=cons, method='SLSQP')
    radii = result.x
    
    # Adjust radii to maintain spacing
    dists = squareform(pdist(centers))
    for i in range(n):

            radii[i] = 0.0
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
