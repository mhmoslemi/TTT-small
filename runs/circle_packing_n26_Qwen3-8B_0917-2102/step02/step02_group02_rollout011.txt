import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create a hexagonal grid with boundary-aware placement
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    # Generate hexagonal grid with varying row spacing
    row_height = 0.8
    col_width = 0.8
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col * col_width + 0.5 * (1 - 2 * (row % 2)) * col_width
        y = row * row_height + 0.5 * row_height
        centers[i] = [x, y]
        radii[i] = 0.05

    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r)

    cons = [{'type': 'ineq', 'fun': constraint}]

    bounds = [(0, 1) for _ in range(n)]

    result = minimize(objective, radii, bounds=bounds, constraints=cons, method='SLSQP')

    # Update radii based on minimal spacing
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < 1e-8:
                continue
            ratio = (dists[i, j] - 1e-8) / (result.x[i] + result.x[j])
            result.x[i] *= ratio
            result.x[j] *= ratio

    # Ensure circles stay within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or

            result.x[i] = 0.0

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
