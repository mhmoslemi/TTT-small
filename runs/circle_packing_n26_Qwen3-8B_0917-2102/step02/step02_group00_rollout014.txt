from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

    # Initial hexagonal grid positions
    positions = []
    for row in range(4):
        for col in range(6):
            x = col + 0.5 * (row % 2)
            y = row + 0.5
            positions.append([x, y])
    centers = np.array(positions)
    
    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    radii = np.full(n, 0.05)

    def objective(radii):
        return -np.sum(radii)

    def constraint1(radii):
        return np.sum(radii) - 2.636

    cons = [{'type': 'ineq', 'fun': constraint1}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 2.636)] * n)

    # Adjust radii to maintain spacing

    for i in range(n):

            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
