import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
import random

def run_packing():
    n = 26
    # Initial random positions in [0,1]^2
    centers = np.random.rand(n, 2)
    radii = np.ones(n) * 0.1

    def objective(x):
        return -np.sum(x)
    
    def constraint(x):
        return np.sum(x) - 1.0

    cons = [{'type': 'ineq', 'fun': constraint}]
    bounds = Bounds(0, 1)

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=bounds)

    # Adjust radii to maintain spacing
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
