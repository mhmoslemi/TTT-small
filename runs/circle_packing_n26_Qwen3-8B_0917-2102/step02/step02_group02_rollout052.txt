import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create a staggered hexagonal grid with more refined spacing
    centers = np.zeros((n, 2))
    row = 0
    col = 0
    for i in range(n):
        if i % 2 == 0:
            centers[i] = [col / 4, row / 4]
        else:
            centers[i] = [(col + 0.5) / 4, row / 4]
        col += 1
        if col >= 5:
            col = 0
            row += 1

    # Convert to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    # Initial radii
    radii = np.full(n, 0.1)

    def objective(radii):
        return -np.sum(radii)

    # Constraints: minimum and maximum sum of radii
    def constraint_min(radii):
        return np.sum(radii) - 0.01

    def constraint_max(radii):
        return 1 - np.sum(radii) - 0.01

    cons = [{'type': 'ineq', 'fun': constraint_min},
            {'type': 'ineq', 'fun': constraint_max}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP')

    radii = result['x']

    # Adjust radii to maintain spacing
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j] + 1e-8:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
