import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate initial hexagonal grid
    centers = []
    for row in range(4):
        for col in range(6):
            x = col + 0.5 * (row % 2)
            y = row + 0.5
            centers.append([x, y])
    centers = np.array(centers)
    
    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    # Initial radii based on minimum spacing
    min_dist = np.min(pdist(centers))
    radii = np.full(n, min_dist / 2)

    def objective(x):
        return -np.sum(x)

    def constraint(x):
        return np.min(pdist(centers)) - np.sum((np.min(x) + np.max(x))/2)

    cons = [{'type': 'ineq', 'fun': constraint}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10})

    # Final adjustment for boundary and spacing
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radi[i]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
