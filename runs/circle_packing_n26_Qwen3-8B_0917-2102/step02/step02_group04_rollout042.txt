import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal lattice with optimized spacing
    centers = np.zeros((n, 2))
    row_count = 5
    col_count = 5
    for i in range(n):
        row = i // col_count
        col = i % col_count
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]

    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    # Initial radii guess
    radii = np.ones(n) * 0.1

    # Objective function to maximize sum of radii
    def objective(r):
        return -np.sum(r)

    # Constraint: circles must stay within the square
    def constraint_boundary(r):
        return np.array([1.0 - r[i] - 1e-8 for i in range(n)])

    # Constraint: circles must not overlap
    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        return np.array([dists[i, j] - r[i] - r[j] + 1e-8 for i in range(n) for j in range(i + 1, n)])

    # Optimization constraints
    cons = [{'type': 'ineq', 'fun': constraint_boundary},
            {'type': 'ineq', 'fun': constraint_overlap}]

    # Optimization
    result = minimize(objective, radii, constraints=cons, method='SLSQP', tol=1e-8)

    # Update radii based on inter-circle distances
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result.x[i] + result.x[j]:
                ratio = (dists[i, j] - 1e-8) / (result.x[i] + result.x[j])
                result.x[i] *= ratio
                result.x[j] *= ratio

    # Final check to ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result.x[i] = 0.0

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
