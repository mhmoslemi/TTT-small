import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create initial hexagonal grid with spacing
    centers = []
    for row in range(5):
        for col in range(5):
            x = col + 0.5 * (1 - 2 * (row % 2))
            y = row + 0.5
            centers.append([x, y])
    for row in range(5, 7):
        for col in range(5):
            x = col + 0.5 * (1 - 2 * (row % 2))
            y = row + 0.5
            centers.append([x, y])
    centers = np.array(centers)
    radii = np.full(n, 0.1)

    # Convert to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)

    def constraint1(r):
        return r - 0.01

    def constraint2(r):
        return 1 - r - 0.01

    cons = [{'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP')

    # Adjust radii to maintain spacing
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
