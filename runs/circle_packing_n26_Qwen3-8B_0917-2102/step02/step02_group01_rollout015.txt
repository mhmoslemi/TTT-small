import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.stats import truncnorm

def run_packing():
    n = 26
    # Initial guess: hexagonal grid
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]

    # Scale to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    # Initial radii guess
    radii = np.full(n, 0.1)

    def objective(r):
        return -np.sum(r)

    def constraint1(r):
        return r - 0.01

    def constraint2(r):
        return 1 - r - 0.01

    cons = [{'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2}]

    # Bounds for radii
    bounds = [(0, 1) for _ in range(n)]

    # Optimization with bounds
    result = minimize(objective, radii, bounds=bounds, constraints=cons, method='SLSQP')

    radii = result.x

    # Compute pairwise distances
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Ensure circles are within unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
