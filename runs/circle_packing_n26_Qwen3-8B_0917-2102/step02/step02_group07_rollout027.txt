import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal grid with optimal spacing
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        for i in range(n):
            x, y = centers[i]
            if x - r[i] < -1e-8 or x + r[i] > 1 + 1e8 or y - r[i] < -1e-8 or y + r[i] > 1 + 1e-8:
                return -1
        return 0

    # Initial guess with uniform radius
    r0 = np.full(n, 0.1)
    bounds = [(0, 1) for _ in range(n)]
    cons = {'type': 'ineq', 'fun': constraint}
    result = minimize(objective, r0, bounds=bounds, constraints=cons, method='SLSQP')

    radii = result.x
    # Final check and adjust radii to ensure no overlaps and stay within square
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio
    # Final boundary check
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8):
            radii[i] = 0.0
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
