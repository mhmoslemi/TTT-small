import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal grid with boundary-aware placement
    centers = np.zeros((n, 2))
    col = 0
    row = 0
    for i in range(n):
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        col += 1
        if col >= 5:
            col = 0
            row += 1

    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    # Initial radii based on average distance between circles
    dists = squareform(pdist(centers))
    avg_dist = np.mean(dists[dists > 0])
    radii = np.full(n, avg_dist / 2)

    def objective(r):
        return -np.sum(r)

    def constraint(x):
        return np.sum(x)

    cons = [{'type': 'ineq', 'fun': constraint, 'args': ()}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-8})

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result.x[i] = 0.0

    # Ensure non-negative radii
    result.x[result.x < 0] = 0.0

    # Adjust radii to maintain spacing
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result.x[i] + result.x[j]:
                ratio = (dists[i, j] - 1e-8) / (result.x[i] + result.x[j])
                result.x[i] *= ratio
                result.x[j] *= ratio

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
