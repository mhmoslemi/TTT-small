import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.1)
    
    def objective(x):
        return -np.sum(x)
    
    def constraint(x, i):
        return x[i] - 0.01
    
    cons = [{'type': 'ineq', 'fun': lambda x: x[i] - 0.01, 'args': (i,)} for i in range(n)]
    
    result = minimize(objective, radii, constraints=cons, method='SLSQP')
    radii = result.x
    
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0
    
    return centers, radii, np.sum(radii)
