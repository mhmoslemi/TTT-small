import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create a hybrid packing with some circles near boundaries
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    # Define boundary positions
    boundary_positions = np.array([
        [0.1, 0.1], [0.1, 0.5], [0.1, 0.9],
        [0.5, 0.1], [0.5, 0.5], [0.5, 0.9],
        [0.9, 0.1], [0.9, 0.5], [0.9, 0.9],
        [0.1, 0.1], [0.1, 0.5], [0.1, 0.9],
        [0.5, 0.1], [0.5, 0.5], [0.5, 0.9],
        [0.9, 0.1], [0.9, 0.5], [0.9, 0.9],
        [0.1, 0.1], [0.1, 0.5], [0.1, 0.9],
        [0.5, 0.1], [0.5, 0.5], [0.5, 0.9]
    ])
    centers[:n//2] = boundary_positions[:n//2]
    # Create hexagonal grid for remaining circles
    hex_centers = []
    for row in range(4):
        for col in range(5):
            x = col + 0.5 * (1 - 2 * (row % 2))
            y = row + 0.5
            hex_centers.append([x, y])
    centers[n//2:] = hex_centers[:n - n//2]
    # Initial radii
    radii = np.full(n, 0.1)
    
    def objective(r):
        return -np.sum(r)
    
    # Constraints to prevent radii from being too small or too large
    def constraint_min(r):
        return r - 0.01
    
    def constraint_max(r):
        return 1 - r - 0.01
    
    cons = [{'type': 'ineq', 'fun': constraint_min},
            {'type': 'ineq', 'fun': constraint_max}]
    
    # Optimization
    result = minimize(objective, radii, constraints=cons, method='SLSQP')
    radii = result.x
    
    # Update radii based on inter-circle distances
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio
    
    # Final check to ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
