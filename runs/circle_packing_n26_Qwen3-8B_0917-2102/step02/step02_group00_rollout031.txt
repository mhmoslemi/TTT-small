import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create hexagonal grid with varying row offsets
    centers = []
    for row in range(4):
        for col in range(6):
            x = col + 0.5 * (1 - 2 * (row % 2))
            y = row + 0.5 * (1 - 2 * (row % 2))
            centers.append([x, y])
    centers = np.array(centers)
    
    # Normalize centers to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    # Initial radii guess based on spacing
    initial_radii = np.full(n, 0.1)
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraint_min(radii):
        return radii - 0.01
    
    def constraint_max(radii):
        return 1 - radii - 0.01
    
    cons = [{'type': 'ineq', 'fun': constraint_min},
            {'type': 'ineq', 'fun': constraint_max}]
    
    result = minimize(objective, initial_radii, constraints=cons, method='SLSQP')
    radii = result.x
    
    # Enforce non-overlapping constraint
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio
    
    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
