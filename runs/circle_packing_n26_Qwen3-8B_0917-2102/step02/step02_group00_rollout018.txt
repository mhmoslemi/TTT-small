from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Hexagonal packing with boundary adjustments

    radii = np.zeros(n)
    r = 0.1
    row = 0
    col = 0
    for i in range(n):

            row += 1
            col = 0

        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = r
        col += 1

    # Normalize centers to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

        return -np.sum(x)

        return x - 0.01

    def constraint_max(x):
        return 1 - x - 0.01

    cons = [{'type': 'ineq', 'fun': constraint_min},
            {'type': 'ineq', 'fun': constraint_max}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP')

    # Adjust radii to maintain spacing
    dists = squareform(pdist(centers))
    for i in range(n):

            if dists[i, j] < radii[i] + radii[j]:

                radii[j] *= ratio

        if (x - r < -1e-12 or x + r > 1 + 1e-12

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
