import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal grid with offset rows
    centers = []
    for row in range(4):
        for col in range(6):
            x = col + 0.5 * (row % 2)
            y = row + 0.5
            centers.append([x, y])
    centers = np.array(centers)
    
    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))
    
    # Initial radii
    radii = np.full(n, 0.05)
    
    def objective(x):
        return -np.sum(x)
    
    # Constraints for minimum and maximum radius
    def constraint_min(x):
        return x - 0.01
    
    def constraint_max(x):
        return 1 - x - 0.01
    
    cons = [{'type': 'ineq', 'fun': constraint_min},
            {'type': 'ineq', 'fun': constraint_max}]
    
    result = minimize(objective, radii, constraints=cons, method='SLSQP')
    
    # Update radii with optimized values
    radii = result.x
    
    # Calculate minimal distances between circles
    dists = squareform(pdist(centers))
    
    # Adjust radii based on distance constraints
    for i in range(n):
        for j in range(i + 1, n):
            d = dists[i, j]
            if d < radii[i] + radii[j] - 1e-8:
                ratio = (d - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio
    
    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
