import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Create a hexagonal grid with 5 columns and 6 rows
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]

    # Scale centers to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.array([r[i] + r[j] <= np.sqrt(np.sum((centers[i] - centers[j])**2)) for i in range(n) for j in range(i+1, n)] + \
                        [r[i] <= 1 - 1e-8 for i in range(n)])

    # Initial guess with uniform radius
    r0 = np.full(n, 0.1)

    # Optimize radii with constraints
    result = minimize(objective, r0, method='SLSQP', constraints={'type': 'ineq', 'fun': constraint})

    radii = result.x

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
