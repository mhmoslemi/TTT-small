import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Hexagonal packing with boundary adjustment
    positions = []
    for row in range(4):
        for col in range(6):
            x = col + 0.5 * (row % 2)
            y = row + 0.5
            if row % 2 == 0:
                x = col + 0.5
            positions.append([x, y])
    centers = np.array(positions)
    
    # Normalize to [0,1]^2
    centers -= centers.min(axis=0)
    centers /= centers.max(axis=0)
    
    # Initial radii (based on spacing)
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) / 2
    radii = np.full(n, initial_r)
    
    # Optimization objective: maximize sum of radii
    def objective(r):
        return -np.sum(r)
    
    # Constraints: circles must stay within unit square and not overlap
    def constraint(r):
        for i in range(n):
            for j in range(i + 1, n):
                if np.sqrt(np.sum((centers[i] - centers[j])**2)) < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    cons = [{'type': 'ineq', 'fun': constraint}]
    
    result = minimize(objective, radii, constraints=cons, method='SLSQP')
    radii = result.x
    
    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
