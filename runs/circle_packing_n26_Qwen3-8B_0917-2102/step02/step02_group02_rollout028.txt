import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal grid with varying row spacing
    centers = np.zeros((n, 2))
    row_height = 0.8
    col_width = 0.4
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col * col_width + 0.5 * (1 - 2 * (row % 2)) * col_width
        y = row * row_height + 0.5 * row_height
        centers[i] = [x, y]

    # Convert to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(radii):
        return -np.sum(radii)

    def constraint(radii):
        return np.sum(radii) - 0.01

    cons = [{'type': 'ineq', 'fun': constraint}]

    result = minimize(objective, np.full(n, 0.1), constraints=cons, method='SLSQP', options={'ftol': 1e-10})

    radii = result.x
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final boundary check and adjustment
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
