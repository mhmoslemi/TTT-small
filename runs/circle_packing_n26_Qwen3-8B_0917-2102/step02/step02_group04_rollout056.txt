import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate initial hexagonal packing with adjustable spacing
    centers = np.zeros((n, 2))
    row_count = 5
    col_count = 5
    for i in range(n):
        row = i // col_count
        col = i % col_count
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]

    # Normalize centers to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    # Initial radii
    radii = np.full(n, 0.1)

    # Objective function to maximize sum of radii
    def objective(r):
        return -np.sum(r)

    # Constraints: radii must be non-negative and not exceed spacing between circles
    def constraint_min(r):
        return r

    def constraint_max(r):
        dists = squareform(pdist(centers))
        return np.min(dists[np.triu_indices(n, 1)]) - np.sum(r) + 1e-8

    cons = [{'type': 'ineq', 'fun': constraint_min},
            {'type': 'ineq', 'fun': constraint_max}]

    # Optimization
    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final check for boundary and overlap constraints
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result.x[i] = 0.0

    # Ensure no overlaps
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result.x[i] + result.x[j] - 1e-12:
                ratio = (dists[i, j] - 1e-8) / (result.x[i] + result.x[j])
                result.x[i] *= ratio
                result.x[j] *= ratio

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
