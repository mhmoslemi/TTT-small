import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal grid with boundary-aware placement
    centers = np.zeros((n, 2))
    col = 0
    row = 0
    for i in range(n):
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        col += 1
        if col >= 5:
            col = 0
            row += 1

    # Convert to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)

    def constraint1(x):
        return np.sum(x)

    def constraint2(x, i):
        return x[i] - 0.5

    cons = [{'type': 'ineq', 'fun': constraint1, 'bounds': [(0, None) for _ in range(n)]}]
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[i] - 0.5})

    result = minimize(objective, np.full(n, 0.1), constraints=cons, method='SLSQP')

    radii = result.x

    # Adjust radii to maintain spacing
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
