import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Hexagonal packing with adaptive row spacing
    centers = []
    for row in range(5):
        for col in range(5):
            x = col + 0.5 * (1 - 2 * (row % 2))
            y = row + 0.5
            centers.append([x, y])
    centers = np.array(centers)
    
    # Normalize to [0,1]^2
    centers -= centers.min(axis=0)
    centers /= centers.max(axis=0) - centers.min(axis=0)
    
    radii = np.full(n, 0.1)
    
    def objective(r):
        return -np.sum(r)
    
    # Constraints to ensure non-negative radii and boundary fit
    def constraint_min(r):
        return r
    
    def constraint_max(r):
        return 1 - r
    
    cons = [{'type': 'ineq', 'fun': constraint_min}]
    
    # Optimization
    result = minimize(objective, radii, constraints=cons, method='SLSQP')
    radii = result.x
    
    # Update radii to ensure non-overlapping
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio
    
    # Ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
