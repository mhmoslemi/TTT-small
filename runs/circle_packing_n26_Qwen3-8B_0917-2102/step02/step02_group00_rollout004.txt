import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Initial guess with hexagonal packing
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    r = 0.1
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = r

    # Normalize centers to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(x):
        return -np.sum(x)

    def constraint(x, i, j):
        return np.sqrt(np.sum((centers[i] - centers[j])**2)) - (x[i] + x[j])

    # Set up constraints for non-overlapping circles
    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: constraint(x, i, j)})

    # Constraints for circles within [0,1]^2
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[i]})

    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[i]})

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 500})

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):

                result.x[i] *= ratio
                result.x[j] *= ratio

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
