import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)

    def objective(r):
        return -np.sum(r)

    def non_overlap(r, i, j):
        return r[i] + r[j] - np.sqrt(np.sum((centers[i] - centers[j])**2))

    def boundary_constraint(r, i):
        return r[i] - (1 - np.max(np.abs(centers[i])))

    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda r, i=i: boundary_constraint(r, i)})
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: non_overlap(r, i, j)})

    bounds = Bounds(np.zeros(n), np.inf)

    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)

    radii = result.x
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
