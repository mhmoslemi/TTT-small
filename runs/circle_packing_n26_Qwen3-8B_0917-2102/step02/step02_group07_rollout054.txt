from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

        col = i % 5

        y = row + 0.5
        centers[i] = [x, y]

                if dists[i, j] < x[i] + x[j] - 1e-8:

        return 1

    cons = [{'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2},
            {'type': 'ineq', 'fun': constraint3}]
