import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Hexagonal packing with boundary adaptation
    positions = []
    for row in range(3):
        for col in range(5):
            x = col + 0.5 * (row % 2)
            y = row + 0.5
            positions.append([x, y])
    centers = np.array(positions)
    # Scale to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))
    
    # Initial radii based on minimum spacing
    dists = squareform(pdist(centers))
    initial_r = np.min(dists[np.triu_indices(n, 1)]) / 2
    radii = np.full(n, initial_r)
    
    def objective(radii):
        return -np.sum(radii)
    
    # Constraints: radii >= 0 and non-overlapping
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - 1e-8})
    
    for i in range(n):
        for j in range(i + 1, n):
            def constraint_ineq(r, i=i, j=j):
                return np.sqrt(np.sum((centers[i] - centers[j])**2)) - (r[i] + r[j]) - 1e-8
            cons.append({'type': 'ineq', 'fun': constraint_ineq})
    
    result = minimize(objective, radii, constraints=cons, method='SLSQP', tol=1e-10)
    radii = result.x
    
    # Final check to ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
