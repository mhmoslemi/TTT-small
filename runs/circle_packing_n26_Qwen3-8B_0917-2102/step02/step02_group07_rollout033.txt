import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal grid with spacing
    centers = []
    for row in range(5):
        for col in range(5):
            if row % 2 == 0:
                x = col + 0.5
            else:
                x = col + 0.0
            y = row + 0.5
            centers.append([x, y])
    centers = np.array(centers)
    
    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))
    
    # Initial radii guess
    radii = np.full(n, 0.1)
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraint1(radii):
        return np.sum(radii) - 2.636
    
    cons = [{'type': 'ineq', 'fun': constraint1}]
    
    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 2.636)] * n)
    
    # Final radii from optimization
    radii = result.x
    
    # Compute pairwise distances and adjust radii to ensure non-overlapping
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio
    
    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
