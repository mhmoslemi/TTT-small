import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)

    def objective(x):
        return -np.sum(x)

    def constraint(x, i, j):
        return np.sqrt(np.sum((centers[i] - centers[j])**2)) - (radii[i] + radii[j])

    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: constraint(x, i, j)})

    def boundary_constraint(x, i):
        x_i, y_i = centers[i]
        r_i = radii[i]
        return x_i - r_i >= -1e-12 and x_i + r_i <= 1 + 1e-12 and y_i - r_i >= -1e-12 and y_i + r_i <= 1 + 1e-12

    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: boundary_constraint(x, i)})

    bounds = Bounds(lb=-1e-12, ub=1 + 1e-12)

    result = minimize(objective, radii, constraints=cons, bounds=bounds, method='SLSQP')

    if not result.success:
        return np.zeros((n, 2)), np.zeros(n), 0.0

    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
