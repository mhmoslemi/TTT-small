import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal grid with adaptive spacing
    centers = []
    for row in range(4):
        for col in range(6):
            x = col + 0.5 * (1 - 2 * (row % 2))
            y = row + 0.5
            centers.append([x, y])
    centers = np.array(centers)
    
    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    dists = squareform(pdist(centers))
    r_initial = np.min(dists) / 2
    radii = np.full(n, r_initial)

        return -np.sum(radii)

                    return -np.inf
        return np.sum(radii)

    cons = [{'type': 'ineq', 'fun': constraint}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})

    dists = squareform(pdist(centers))

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
