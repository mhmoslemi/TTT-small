import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.1)

    def objective(x):
        return -np.sum(x)

    def constraint1(x):
        return np.minimum(x, 1 - 1e-12) - np.maximum(x, 1e-12)

    cons = [{'type': 'ineq', 'fun': constraint1}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10})

    # Refine positions and radii with adaptive packing
    for _ in range(5):
        dists = squareform(pdist(centers))
        for i in range(n):

                    ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                    radii[i] *= ratio
                    radii[j] *= ratio

        # Update centers to maintain spacing
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j]:
                    dir_vec = (centers[j] - centers[i]) / dist
                    update = 0.1 * (radii[i] + radii[j] - dist) * dir_vec
                    centers[i] -= update
                    centers[j] += update

        # Scale centers to [0,1]^2
        centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
