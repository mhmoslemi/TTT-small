import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create a hexagonal grid with spacing
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    row, col = 0, 0
    for i in range(n):
        centers[i] = [col + 0.5 * (1 - 2 * (row % 2)), row + 0.5]
        col += 1
        if col >= 5:
            col = 0
            row += 1

    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(radii):
        return -np.sum(radii)

    def constraint(radii):
        return np.sum(radii)

    cons = [{'type': 'ineq', 'fun': constraint, 'jac': lambda r: np.ones_like(r)}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-12, 'maxiter': 500})

    # Adjust radii to maintain spacing
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j] + 1e-8:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
