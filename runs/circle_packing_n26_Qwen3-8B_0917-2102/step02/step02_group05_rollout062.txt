import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Hexagonal packing with boundary-aware positioning
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 * (1 - 2 * (row % 2))
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = 0.1
    
    # Normalize to [0,1]^2
    centers -= centers.min(axis=0)
    centers /= centers.max(axis=0) - centers.min(axis=0)

        return -np.sum(radii)
    
    def constraint1(radii):
        return radii - 0.01
    
    def constraint2(radii):
        return 1 - radii - 0.01
    
    cons = [{'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2}]
    
    result = minimize(objective, radii, constraints=cons, method='SLSQP')
    radii = result.x
    
    # Ensure circles are within bounds
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0
    
    # Compute pairwise distances and adjust radii for non-overlapping
    dists = squareform(pdist(centers))
    for i in range(n):

            dist = dists[i, j]
            if dist < radii[i] + radii[j]:
                ratio = (dist - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio
    
    # Final boundary check
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
