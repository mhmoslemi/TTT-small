import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Generate hexagonal grid with 5 columns and 6 rows
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]

    # Scale to unit square
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)

    def constraint(r, i, j):
        return r[i] + r[j] - np.sqrt(np.sum((centers[i] - centers[j])**2))

    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: constraint(r, i, j)})

    # Add boundary constraints
    for i in range(n):
        def boundary_constraint(r, i=i):
            x, y = centers[i]
            return r[i] - (1 - np.max([x, y]))  # Left/Bottom boundary
        cons.append({'type': 'ineq', 'fun': boundary_constraint})

        def boundary_constraint_right(r, i=i):
            x, y = centers[i]
            return r[i] - (np.min([1 - x, 1 - y]))  # Right/Top boundary
        cons.append({'type': 'ineq', 'fun': boundary_constraint_right})

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})

    radii = result.x

    # Ensure non-negative radii
    radii[radii < 0] = 0

    # Final check for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
