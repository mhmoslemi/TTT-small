import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.1)

    def objective(r):
        return -np.sum(r)
    
    def constraint(r):
        return np.sum(r)
    
    cons = [{'type': 'ineq', 'fun': constraint, 'jac': lambda r: np.ones(n)}]
    bounds = Bounds(np.zeros(n), np.inf * np.ones(n))
    
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)
    radii = result.x

    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            radii[i] = 0.0

    return centers, radii, np.sum(radii)
