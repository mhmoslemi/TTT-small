import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create initial staggered hexagonal grid
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = 0.1

    # Convert to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)

    # Constraints: radii must be non-negative
    def constraint1(r):
        return r - 0.01

    cons = [{'type': 'ineq', 'fun': constraint1}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)]*n)

    # Update radii based on minimal distances
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            d = dists[i, j]
            if d < 1e-8:
                continue
            ratio = (d - 1e-8) / (result.x[i] + result.x[j])
            result.x[i] *= ratio
            result.x[j] *= ratio

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result.x[i] = 0.0

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
