import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate initial hexagonal grid with varying row spacing
    centers = np.zeros((n, 2))
    row_count = 5
    col_count = 5
    for i in range(n):
        row = i // col_count
        col = i % col_count
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]

    def objective(r):
        return -np.sum(r)

    # Constraints to ensure radii are positive and reasonable
    def constraint_min(r):
        return r - 0.01

    def constraint_max(r):
        return 1 - r - 0.01

    cons = [{'type': 'ineq', 'fun': constraint_min},
            {'type': 'ineq', 'fun': constraint_max}]

    # Optimization
    result = minimize(objective, centers.flatten(), method='SLSQP', bounds=[(0, 1)]*2*n, constraints=cons)

    # Reconstruct centers and radii from optimization result
    optimized = result.x
    centers_opt = optimized[:n*2].reshape(n, 2)
    radii = optimized[n*2:]

    # Ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers_opt[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    # Ensure non-overlapping and adjust radii based on inter-circle distances
    dists = squareform(pdist(centers_opt))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final check to ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers_opt[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers_opt, radii, sum_radii
