import numpy as np
from scipy.optimize import minimize, minimize_slsqp
from scipy.spatial.distance import pdist, squareform

def run_packing():

    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    r = 0.1

        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = r
    
    # Normalize centers to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

        return -np.sum(x)
    
    # Constraints: circles must be within the unit square and non-overlapping

        return x - 1e-8
    
    def constraint_max(x):
        return 1 - x - 1e-8
    
    cons = [{'type': 'ineq', 'fun': constraint_min},
            {'type': 'ineq', 'fun': constraint_max}]
    
    # Optimization
    result = minimize_slsqp(objective, radii, constraints=cons, method='SLSQP')
    radii = result.x

    dists = squareform(pdist(centers))
    for i in range(n):

                radii[i] *= ratio

    return centers, radii, sum_radii
