import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Generate staggered hexagonal grid with boundary-aware placement
    centers = np.zeros((n, 2))
    col = 0
    row = 0
    for i in range(n):
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        col += 1
        if col >= 6:
            col = 0
            row += 1

    # Convert to [0,1]^2 and scale to fit square
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

        return -np.sum(r)

    def constraint(x):
        return np.sum(x)

    cons = [{'type': 'ineq', 'fun': constraint}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 1000})

    # Final adjustment for boundary and spacing
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        # Ensure circle stays within [0,1]^2
        if x - r < 0:
            r = x
        elif x + r > 1:
            r = 1 - x
        if y - r < 0:
            r = y
        elif y + r > 1:
            r = 1 - y
        result.x[i] = r

    # Final spacing adjustment
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result.x[i] + result.x[j]:

                result.x[j] *= ratio

        if (x - r < -1e-12 or x + r > 1 + 1e-12

            result.x[i] = 0.0

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
