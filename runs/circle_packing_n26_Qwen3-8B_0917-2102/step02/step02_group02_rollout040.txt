import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate a hexagonal lattice with adaptive spacing
    centers = []
    for row in range(5):
        for col in range(5):
            if row % 2 == 0:
                x = col + 0.5
            else:
                x = col + 0.5
            y = row + 0.5
            centers.append([x, y])
    for row in range(5):
        for col in range(5):
            if row % 2 == 0:
                x = col + 0.5 + 0.5
            else:
                x = col + 0.5
            y = row + 1.5
            centers.append([x, y])
    centers = np.array(centers)
    radii = np.full(n, 0.1)

    # Normalize centers to [0,1]^2
    centers -= centers.min(axis=0)
    centers /= centers.max(axis=0) - centers.min(axis=0)

    def objective(radii):
        return -np.sum(radii)

    def constraint(radii):
        return np.sum(radii) - 0.01  # Initial constraint for optimization

    cons = [{'type': 'ineq', 'fun': constraint}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP')

    # Adjust radii to maintain spacing
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j] + 1e-8:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Check boundary constraints and adjust radii
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
