from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Generate initial hexagonal-like arrangement
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
    
    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

        return -np.sum(radii)

        return radii - 0.01
    
    # Constraint to ensure circles are within square

            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or
                y - r < -1e-12 or y + r > 1 + 1e-12):
                return -np.inf
        return 0
    
    # Optimization with bounds and constraints
    cons = [{'type': 'ineq', 'fun': constraint1}]
    bounds = [(0, 1) for _ in range(n)]
    
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)
    radii = result.x
    
    # Ensure circles within square
