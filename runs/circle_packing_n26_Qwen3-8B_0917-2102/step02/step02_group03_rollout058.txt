import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create a hexagonal grid with dynamic boundary-aware placement
    centers = np.zeros((n, 2))
    col = 0
    row = 0
    for i in range(n):
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        col += 1
        if col >= 5:
            col = 0
            row += 1

    # Scale to [0,1]^2 with boundary-aware adjustment
    min_x, min_y = np.min(centers, axis=0)
    max_x, max_y = np.max(centers, axis=0)
    centers = (centers - min_x) / (max_x - min_x)
    centers = np.clip(centers, 0, 1)

    # Initial radii guess based on minimum spacing
    dists = squareform(pdist(centers))
    min_dist = np.min(dists)
    radii = np.full(n, min_dist / 2)

    def objective(r):
        return -np.sum(r)

    # Constraints: each circle must fit in the square and not overlap
    def constraint(r):
        for i in range(n):
            x, y = centers[i]
            if x - r[i] < -1e-12 or x + r[i] > 1 + 1e-12 or y - r[i] < -1e-12 or y + r[i] > 1 + 1e-12:
                return -np.inf
        for i in range(n):
            for j in range(i + 1, n):
                if np.sqrt(np.sum((centers[i] - centers[j])**2)) < r[i] + r[j] - 1e-12:
                    return -np.inf
        return np.sum(r)

    cons = [{'type': 'ineq', 'fun': lambda r: constraint(r)}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', tol=1e-8)

    # Extract optimized radii
    radii = result.x

    # Final check to ensure all circles are within [0,1]^2 and non-overlapping
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
