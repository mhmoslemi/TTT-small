import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Hexagonal packing with boundary adjustments
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Generate hexagonal grid with boundary adjustments
    row_height = 1.0 / np.sqrt(3)
    col_width = 1.0 / 4.0
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col * col_width + 0.5 * (1 - 2 * (row % 2)) * col_width
        y = row * row_height
        centers[i] = [x, y]
        radii[i] = 0.1

    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(x):
        return -np.sum(x)

    def constraint(x):
        return np.array([x[0] - 0.01, 1 - x[1] - 0.01])

    cons = [{'type': 'ineq', 'fun': constraint}]

    # Initial guess
    initial_guess = np.full(n, 0.1)

    # Optimize
    result = minimize(objective, initial_guess, constraints=cons, method='SLSQP', tol=1e-10)

    # Update radii
    radii = result.x

    # Enforce spacing constraints
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final check for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
