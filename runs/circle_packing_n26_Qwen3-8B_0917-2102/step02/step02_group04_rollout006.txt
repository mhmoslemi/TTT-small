import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Generate a hexagonal grid with adaptive spacing
    rows = []
    for r in range(5):
        if r % 2 == 0:
            cols = [0.1 + i * 0.2 for i in range(13)]
        else:
            cols = [0.15 + i * 0.2 for i in range(12)]
        for c in cols:
            centers.append([c, r + 0.5])
    centers = np.array(centers)
    radii = np.full(n, 0.1)
    
    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))
    
    def objective(r):
        return -np.sum(r)
    
    def constraint_min(r):
        return r - 0.05
    
    def constraint_max(r):
        return 1 - r - 0.05
    
    cons = [{'type': 'ineq', 'fun': constraint_min},
            {'type': 'ineq', 'fun': constraint_max}]
    
    result = minimize(objective, radii, constraints=cons, method='SLSQP')
    radii = result.x
    
    # Update radii based on inter-circle distances
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio
    
    # Final check to ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
