import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Generate hexagonal grid with boundary-aware placement
    centers = np.zeros((n, 2))
    col = 0
    row = 0
    for i in range(n):
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        col += 1
        if col >= 5:
            col = 0
            row += 1

    # Normalize centers to [0,1]^2
    centers -= centers.min(axis=0)
    centers /= centers.max(axis=0) - centers.min(axis=0)

    # Initial radii based on minimal spacing
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) / 2
    radii = np.full(n, initial_r)

        return -np.sum(r)

    def constraint(x):
        return np.sum(x)

    cons = [{'type': 'ineq', 'fun': constraint}]

    # Optimization with bounds and constraints
    result = minimize(objective, radii, bounds=[(0, 1)]*n, constraints=cons, method='SLSQP')

    radii = result.x

    # Adjust radii based on distances
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):

        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
