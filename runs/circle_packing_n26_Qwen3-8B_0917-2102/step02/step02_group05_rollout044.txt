import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Initial hexagonal arrangement with boundary-aware spacing
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = 0.1
    
    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))
    
    # Objective: maximize sum of radii
    def objective(x):
        return -np.sum(x)
    
    # Constraints: radii must be positive and within reasonable bounds
    def constraint_min(x):
        return x - 0.01
    
    def constraint_max(x):
        return 1 - x - 0.01
    
    cons = [{'type': 'ineq', 'fun': constraint_min},
            {'type': 'ineq', 'fun': constraint_max}]
    
    result = minimize(objective, radii, constraints=cons, method='SLSQP')
    radii = result.x
    
    # Adjust radii to maintain spacing
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            min_dist = dists[i, j]
            required_dist = radii[i] + radii[j]
            if min_dist < required_dist:
                ratio = (min_dist - 1e-8) / required_dist
                radii[i] *= ratio
                radii[j] *= ratio
    
    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
