from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    radii = np.zeros(n)

        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))

        centers[i] = [x, y]
        radii[i] = r

    centers /= centers.max(axis=0) - centers.min(axis=0)

        return -np.sum(x)

        return np.sum(x) - 1.0

    for i in range(n):
        x, y = centers[i]
        r = radii[i]

            centers[i, 1] = r

            centers[i, 1] = 1 - r
