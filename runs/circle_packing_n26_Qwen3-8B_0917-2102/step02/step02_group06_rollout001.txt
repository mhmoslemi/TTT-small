import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance as sp_dist
from scipy.optimize import Bounds, LinearConstraint

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Hexagonal initial placement with some variation
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = 0.1
    
    # Normalize to [0,1]^2
    centers -= centers.min(axis=0)
    centers /= centers.max(axis=0) - centers.min(axis=0)

    def objective(radii):
        return -np.sum(radii)
    
    def constraint1(radii):
        return np.sum(radii)
    
    def constraint2(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dist = sp_dist.euclidean(centers[i], centers[j])
                if dist < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        return 1.0
    
    bounds = Bounds(0, 1)
    cons = [
        {'type': 'eq', 'fun': constraint1},
        {'type': 'ineq', 'fun': constraint2}
    ]
    
    result = minimize(objective, radii, bounds=bounds, constraints=cons, method='SLSQP', options={'maxiter': 500})
    radii = result.x
    
    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
