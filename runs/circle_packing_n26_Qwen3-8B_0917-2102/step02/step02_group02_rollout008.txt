import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create a refined hexagonal grid with variable row spacing
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    row_height = 1.0 / (np.sqrt(3) * (n**0.5))
    col_spacing = 1.0 / (n**0.5)
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col * col_spacing + 0.5 * (1 - 2 * (row % 2))
        y = row * row_height + 0.5
        centers[i] = [x, y]
        radii[i] = 0.1

    # Normalize centers to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(radii):
        return -np.sum(radii)

    def constraint(radii):
        return np.sum(radii)

    cons = [{'type': 'ineq', 'fun': constraint}]

    def inequality_constraint(radii):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] + 1e-8:
                    return 0.0
        return 1.0

    cons.append({'type': 'ineq', 'fun': inequality_constraint})

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1.0) for _ in range(n)])

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result.x[i] = 0.0

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
