import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial placement with some circles at corners and edges
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Place 4 circles at corners
    centers[0] = [0.0, 0.0]
    centers[1] = [1.0, 0.0]
    centers[2] = [1.0, 1.0]
    centers[3] = [0.0, 1.0]
    radii[0] = radii[1] = radii[2] = radii[3] = 0.1

    # Place remaining circles in hexagonal packing along edges
    for i in range(4, n):
        if i - 4 < 12:
            row = (i - 4) // 5
            col = (i - 4) % 5
            x = col / 5.0 + 0.05
            y = row / 5.0 + 0.05
        else:
            row = (i - 4 - 12) // 5
            col = (i - 4 - 12) % 5
            x = 1.0 - col / 5.0 - 0.05
            y = row / 5.0 + 0.05
        centers[i] = [x, y]
        radii[i] = 0.1

    # Convert to [0,1]^2 (already in [0,1]^2)

    def objective(x):
        return -np.sum(x)

    def constraint1(x):
        return x - 0.1

    def constraint2(x):
        return 1 - x - 0.1

    cons = [{'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP')

    # Adjust radii to maintain spacing
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
