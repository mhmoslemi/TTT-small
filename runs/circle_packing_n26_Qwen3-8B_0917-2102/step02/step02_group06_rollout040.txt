import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial guess with hexagonal packing
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    r = 0.1
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = r
    
    # Convert to [0,1]^2
    centers -= centers.min(axis=0)
    centers /= centers.max(axis=0) - centers.min(axis=0)
    
    # Objective function to minimize negative sum of radii
    def objective(x):
        return -np.sum(x)
    
    # Constraints: sum of radii must be at least 1.0
    def constraint(x):
        return np.sum(x) - 1.0
    
    cons = [{'type': 'ineq', 'fun': constraint}]
    
    # Bounds for radii
    bounds = [(0, 1)] * n
    
    # Initial guess
    x0 = radii
    
    # Optimization with SLSQP
    result = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP')
    radii = result.x
    
    # Compute pairwise distances and adjust radii to maintain spacing
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio
    
    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
