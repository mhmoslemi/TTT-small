import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Create initial hexagonal grid with spacing
    centers = np.zeros((n, 2))

        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
    
    # Normalize to [0,1]^2
    centers -= centers.min(axis=0)
    centers /= centers.max(axis=0) - centers.min(axis=0)

        return -np.sum(radii)

    def constraint1(radii):

    radii = np.full(n, 0.1)

    result = minimize(objective, radii, constraints=cons, method='SLSQP', tol=1e-8)

    dists = squareform(pdist(centers))

            if dists[i, j] < result.x[i] + result.x[j]:
                ratio = (dists[i, j] - 1e-8) / (result.x[i] + result.x[j])
                result.x[i] *= ratio
                result.x[j] *= ratio

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):

        if (x - r < -1e-12 or x + r > 1 + 1e-12

    sum_radii = np.sum(result.x)
