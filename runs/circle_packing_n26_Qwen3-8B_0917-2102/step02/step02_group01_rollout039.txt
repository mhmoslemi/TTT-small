import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal grid with 5 columns and 6 rows for 26 circles
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
    
    # Scale to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))
    
    # Initial radii
    radii = np.full(n, 0.1)
    
    # Objective function to maximize sum of radii
    def objective(r):
        return -np.sum(r)
    
    # Constraint: minimum radius
    def constraint_min_r(r):
        return r - 0.01
    
    # Constraint: maximum radius
    def constraint_max_r(r):
        return 1 - r - 0.01
    
    cons = [{'type': 'ineq', 'fun': constraint_min_r},
            {'type': 'ineq', 'fun': constraint_max_r}]
    
    result = minimize(objective, radii, constraints=cons, method='SLSQP')
    radii = result.x
    
    # Compute pairwise distances
    dists = squareform(pdist(centers))
    
    # Adjust radii to maintain spacing
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j] + 1e-8:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio
    
    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
