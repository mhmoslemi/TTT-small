from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

    # Initial staggered square packing in [0,1]^2
    centers = np.zeros((n, 2))

        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]

    centers -= centers.min(axis=0)
    centers /= centers.max(axis=0) - centers.min(axis=0)

        return -np.sum(x)

        cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[i]})
    
    result = minimize(objective, np.full(n, 1.0), constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

                radii[j] *= ratio

    return centers, radii, sum_radii
