import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal grid with spacing
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
    
    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    # Initial radius guess
    radii = np.full(n, 0.1)

    def objective(r):
        return -np.sum(r)

    def constraints(r):
        return np.sum(r) - 0.01  # Lower bound on total sum

    cons = [{'type': 'ineq', 'fun': constraints}]

    def gradient(r):
        return -np.ones_like(r)

    result = minimize(objective, radii, method='SLSQP', constraints=cons, jac=gradient)

    # Final radii
    radii = result.x

    # Compute pairwise distances
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j] + 1e-8:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
