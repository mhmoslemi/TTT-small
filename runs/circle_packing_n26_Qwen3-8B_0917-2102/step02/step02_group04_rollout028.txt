import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create a hexagonal grid with adaptive row spacing
    centers = np.zeros((n, 2))
    row_count = 5
    col_count = 5
    for i in range(n):
        row = i // col_count
        col = i % col_count
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
    
    # Normalize centers to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    # Initial radii based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    min_dist = np.min(dists)
    radii = np.full(n, min_dist / np.sqrt(2))  # Initial radii based on hexagonal packing

    def objective(r):
        return -np.sum(r)

    # Constraint to ensure circles stay within the square
    def constraint(r):
        for i in range(n):
            x, y = centers[i]
            if (x - r[i] < -1e-12 or x + r[i] > 1 + 1e-12
                    or y - r[i] < -1e-12 or y + r[i] > 1 + 1e-12):
                return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint}]

    # Optimization with bounds and constraints
    bounds = [(0, 1) for _ in range(n)]
    result = minimize(objective, radii, bounds=bounds, constraints=cons, method='SLSQP')

    radii = result.x

    # Final check to ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    # Ensure non-overlapping
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j] - 1e-12:
                factor = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= factor
                radii[j] *= factor

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
