import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n) * 0.1

    def objective(r):
        return -np.sum(r)

    def constraint1(r):
        return r - 0.01

    def constraint2(r):
        return 1 - r - 0.01

    cons = [{'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2}]

    def non_overlap_constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i+1, n):
                if dists[i,j] < r[i] + r[j] - 1e-8:
                    return -dists[i,j] + r[i] + r[j] - 1e-8
        return 0

    cons.append({'type': 'ineq', 'fun': non_overlap_constraint})

    result = minimize(objective, radii, constraints=cons, method='SLSQP')

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result.x[i] = 0.0

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
