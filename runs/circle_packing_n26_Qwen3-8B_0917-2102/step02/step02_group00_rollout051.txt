from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)

    def objective(radii):
        return -np.sum(radii)

    def constraint1(radii):
        return np.sum(radii) - 2.636

    cons = [{'type': 'ineq', 'fun': constraint1}]

    def bounds_func(radii):
        return [(0, 1) for _ in range(n)]

    result = minimize(objective, radii, method='SLSQP', bounds=bounds_func, constraints=cons)

    # Get optimal radii
    radii = result.x

    # Adjust centers to maintain spacing
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
