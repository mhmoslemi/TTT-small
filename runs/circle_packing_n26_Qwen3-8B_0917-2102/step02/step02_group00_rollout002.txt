import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Initialize with hexagonal packing
    for i in range(n):
        row = i // 5
        col = i % 5
        angle = 2 * np.pi / 6 * (row % 2)
        r = 0.1
        x = col + 0.5 * np.cos(angle) * r
        y = row + 0.5 * np.sin(angle) * r
        centers[i] = [x, y]
        radii[i] = r

    # Normalize centers to [0,1]^2
    min_coords = centers.min(axis=0)
    max_coords = centers.max(axis=0)
    centers = (centers - min_coords) / (max_coords - min_coords)

    def objective(x):
        return -np.sum(x)

    # Constraints to ensure radii are non-negative and spacing is maintained
    def constraint1(x):
        return x - 0.01

    def constraint2(x):
        return 1 - x - 0.01

    cons = [{'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP')

    # Adjust radii to maintain spacing
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
