import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Generate hexagonal grid points within unit square
    centers = []
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers.append([x, y])

    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    radii = np.ones(n) * 0.1

                    return -1

            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8 or y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8:
                return -1
        return 0

    # Use SLSQP with bounds and constraints
    cons = [{'type': 'ineq', 'fun': constraint}]
    bounds = [(0, 1) for _ in range(n)]

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
