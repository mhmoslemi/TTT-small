import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial guess for circle centers using a hexagonal packing in a square
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
    
    # Initial radii guess
    radii = np.full(n, 0.1)
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraint(radii):
        return np.sum(radii) - 2.636
    
    def bounds(radii):
        return [(0, 1) for _ in range(n)]
    
    def update_centers(radii):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] + 1e-8:
                    # Move circles to prevent overlap
                    delta = (radii[i] + radii[j] - dists[i, j]) / 2
                    direction = (centers[j] - centers[i]) / dists[i, j]
                    centers[i] += direction * delta
                    centers[j] -= direction * delta
        # Normalize centers to [0,1]^2
        centers -= centers.min(axis=0)
        centers /= centers.max(axis=0) - centers.min(axis=0)
    
    # Optimization
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        constraints=[{'type': 'ineq', 'fun': constraint}],
        bounds=bounds
    )
    
    radii = result.x
    update_centers(radii)
    
    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
