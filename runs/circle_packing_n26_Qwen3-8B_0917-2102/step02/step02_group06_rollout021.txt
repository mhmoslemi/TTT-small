import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create a more optimal initial hexagonal packing with boundary-aware placement
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    r = 0.1
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = r
    
    # Normalize to [0,1]^2
    centers -= centers.min(axis=0)
    centers /= centers.max(axis=0) - centers.min(axis=0)

    def objective(x):
        return -np.sum(x)
    
    # Define constraints for maximum total radius and non-overlapping
    def constraint_total(x):
        return np.sum(x) - 1.0

    def constraint_overlap(x):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < x[i] + x[j] - 1e-8:
                    return -np.inf
        return 0.0
    
    cons = [{'type': 'ineq', 'fun': constraint_total},
            {'type': 'ineq', 'fun': constraint_overlap}]
    
    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)
    radii = result.x

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
