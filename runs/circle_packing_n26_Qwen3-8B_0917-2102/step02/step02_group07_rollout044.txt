from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Define positions for circles: 4 corners, 4 edges, and 18 inner

    radii = np.zeros(n)

    positions[0] = [0, 0]
    positions[1] = [1, 0]
    positions[2] = [1, 1]
    positions[3] = [0, 1]

        row = (i - 8) // 6

        x = 0.5 + 0.25 * (1 - 2 * (row % 2)) + 0.25 * (col - 2.5)
        y = 0.5 + 0.5 * row
        positions[i] = [x, y]
        radii[i] = 0.1

    # Normalize positions to [0,1]^2
    positions -= positions.min(axis=0)
    positions /= positions.max(axis=0) - positions.min(axis=0)

    def objective(r):
        return -np.sum(r)

    def constraint(x):
        return np.sum(x) - 0.01

    cons = [{'type': 'ineq', 'fun': constraint}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 1000})

    # Adjust radii to maintain spacing
    dists = squareform(pdist(positions))

                radii[j] *= ratio

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = positions[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            radii[i] = 0.0
