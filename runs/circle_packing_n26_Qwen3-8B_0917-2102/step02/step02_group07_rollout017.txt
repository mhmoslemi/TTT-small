import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Create a hexagonal packing with variable spacing
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    row_height = np.sqrt(3) / 2
    col_width = 1 - 1e-12
    for i in range(n):
        row = i // 5
        col = i % 5
        y = row * row_height
        if row % 2 == 1:
            x = col + 0.5 * col_width
        else:
            x = col * col_width
        centers[i] = [x, y]
        radii[i] = 0.1

    # Normalize centers to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(x):
        return -np.sum(x)

    def constraint(x):
        return np.sum(x)

    cons = [{'type': 'ineq', 'fun': constraint, 'lb': 0, 'ub': 10}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', tol=1e-8)

    # Adjust radii to maintain spacing
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radi[i]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
