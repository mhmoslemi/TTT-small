from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Create initial staggered hexagonal grid with spacing
    centers = np.zeros((n, 2))

    spacing = 0.4

        col = i % 5
        offset = 0.5 * (1 - 2 * (row % 2))
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = 0.05

    # Convert to [0,1]^2
    centers -= centers.min(axis=0)
    centers /= centers.max(axis=0) - centers.min(axis=0)

    def objective(x):
        return -np.sum(x)

    def constraint(x):
        return np.concatenate([x - 0.01, 1 - x - 0.01])

    cons = [{'type': 'ineq', 'fun': constraint}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})

    # Adjust radii to maintain spacing
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
