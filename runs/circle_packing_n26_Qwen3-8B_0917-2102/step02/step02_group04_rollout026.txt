import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal grid with adjusted spacing
    centers = np.zeros((n, 2))
    row_count = 6
    col_count = 5
    for i in range(n):
        row = i // col_count
        col = i % col_count
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]

    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    # Initial radii guess
    radii = np.full(n, 0.1)

    # Objective function to maximize sum of radii
    def objective(r):
        return -np.sum(r)

    # Constraints: radii must be non-negative and circles must not overlap
    def constraint(r):
        return r

    cons = [{'type': 'ineq', 'fun': constraint}]

    # Optimization
    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Update radii to ensure non-overlapping
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            dist = dists[i, j]
            if dist < radii[i] + radii[j] - 1e-8:
                ratio = (dist - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final check to ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
