import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Initial guess with hexagonal packing in a grid
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.1)

        row = i // 5
        col = i % 5

        centers[i] = [x, y]

        return -np.sum(radii)

        return np.sum(radii) - 2.636

    cons = [{'type': 'ineq', 'fun': constraint1}]

                    return -1
        return 0

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 2.636)] * n, options={'ftol': 1e-10, 'maxiter': 500})
    radii = result.x

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):

        r = radii[i]
