    # Optimization
    cons = [{'type': 'ineq', 'fun': constraint_1}]

        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: constraint_2(i, j)})

    result = minimize(objective, radii, constraints=cons, method='SLSQP')

    radii = result.x
