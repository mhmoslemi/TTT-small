from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

    row = 0
    for i in range(n):
        x = col + 0.5 * (1 - 2 * (row % 2))

        centers[i] = [x, y]

            col = 0
            row += 1

        return -np.sum(r)

    def constraint_total(r):

        return x - r[i] >= -1e-12 and x + r[i] <= 1 + 1e-12 and y - r[i] >= -1e-12 and y + r[i] <= 1 + 1e-12

    def constraint_spacing(r, i, j):
        dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
        return dist >= r[i] + r[j] - 1e-12

    # Setup optimization with constraints
    cons_total = [{'type': 'ineq', 'fun': constraint_total}]
    cons_boundary = [{'type': 'ineq', 'fun': lambda r: constraint_boundary(r, i)} for i in range(n)]
    cons_spacing = [{'type': 'ineq', 'fun': lambda r, i, j: constraint_spacing(r, i, j)} for i in range(n) for j in range(i+1, n)]

    # Combine constraints
    cons = cons_total + cons_boundary + cons_spacing

    # Initial guess
    initial_guess = np.full(n, 0.1)

    # Optimization
    result = minimize(objective, initial_guess, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})

                radii[j] *= ratio

    for i in range(n):

        r = radii[i]
