import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create a refined hexagonal grid with alternating row offsets
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.1
    idx = 0
    for row in range(5):
        for col in range(5):
            if row % 2 == 0:
                x = col + 0.5
            else:
                x = col + 0.5 - 0.5
            y = row + 0.5
            if idx < n:
                centers[idx] = [x, y]
                idx += 1

    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(radii):
        return -np.sum(radii)

    def constraint(radii):
        return np.sum(radii)

    cons = [{'type': 'ineq', 'fun': constraint}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP')

    # Adjust radii to maintain spacing with boundary-aware repulsion
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            dist = dists[i, j]
            if dist < radii[i] + radii[j]:
                ratio = (dist - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
