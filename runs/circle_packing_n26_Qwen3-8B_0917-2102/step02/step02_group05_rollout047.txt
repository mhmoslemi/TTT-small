import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    initial_centers = np.zeros((n, 2))
    initial_radii = np.zeros(n)

    row = 0
    col = 0
    for i in range(n):
        initial_centers[i] = [col + 0.5 * (1 - 2 * (row % 2)), row + 0.5]
        if col == 4:
            row += 1
            col = 0
        else:
            col += 1
    
    # Convert to [0,1]^2
    initial_centers = (initial_centers - initial_centers.min(axis=0)) / (initial_centers.max(axis=0) - initial_centers.min(axis=0))

        return -np.sum(radii)

    cons = [{'type': 'ineq', 'fun': constraint_min},

    result = minimize(objective, initial_radii, constraints=cons, method='SLSQP', bounds=[(0, 1)]*n)

    for i in range(n):
        x, y = initial_centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            radii[i] = 0.0

    dists = squareform(pdist(initial_centers))

    sum_radii = np.sum(radii)
    return initial_centers, radii, sum_radii
