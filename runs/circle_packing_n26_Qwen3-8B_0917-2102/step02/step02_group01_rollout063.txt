import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Generate a hexagonal grid with 5 columns and 6 rows for 26 circles
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]

    # Normalize centers to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)

    def constraints(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] + 1e-8:
                    return np.inf
        return 0

    # Initial guess with uniform radius
    initial_radius = 0.15
    radii_initial = initial_radius * np.ones(n)

    # Use differential evolution for global optimization
    result = minimize(
        objective,
        radii_initial,
        constraints={'type': 'ineq', 'fun': constraints},
        method='SLSQP',
        options={'ftol': 1e-10, 'max_iter': 500}
    )

    radii = result.x

    # Final check to ensure all circles are within [0,1]^2 and non-overlapping
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
