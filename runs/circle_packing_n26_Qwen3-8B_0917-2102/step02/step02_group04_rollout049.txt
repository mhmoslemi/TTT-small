import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal grid with dynamic row spacing
    centers = np.zeros((n, 2))
    row_count = 5
    col_count = 5
    for i in range(n):
        row = i // col_count
        col = i % col_count
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
    
    # Normalize centers to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))
    
    # Initial radii based on minimal spacing
    dists = squareform(pdist(centers))
    min_dist = np.min(dists[np.triu_indices(n, 1)])
    radii = np.full(n, min_dist / 2)
    
    # Objective function to maximize sum of radii
    def objective(r):
        return -np.sum(r)
    
    # Constraints to ensure circles stay within [0,1]^2 and do not overlap
    def constraint_boundary(r):
        return np.array([np.min(centers[:, 0] - r[i]) for i in range(n)]) + np.array([np.min(centers[:, 1] - r[i]) for i in range(n)])
    
    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        return np.array([dists[i, j] - r[i] - r[j] for i in range(n) for j in range(i + 1, n)])
    
    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]
    
    # Optimization
    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 1000})
    
    radii = result.x
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
