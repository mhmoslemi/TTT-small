import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial layout: hexagonal packing with staggered rows
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    row_spacing = 1.0 / np.sqrt(3)
    col_spacing = 1.0 / (np.sqrt(3) + 1)
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col * col_spacing + 0.5 * (1 - 2 * (row % 2)) * col_spacing
        y = row * row_spacing
        centers[i] = [x, y]
        radii[i] = 0.1

    # Normalize centers to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(x):
        return -np.sum(x)

    def constraint(x):
        return np.array([x[0] - 0.01, 1 - x[0] - 0.01])

    cons = [{'type': 'ineq', 'fun': constraint}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 1000})

    # Update radii from optimization
    radii = result.x

    # Adjust radii to maintain spacing
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
