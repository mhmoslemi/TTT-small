import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)
    
    def objective(r):
        return -np.sum(r)
    
    def constraint(r):
        return np.sum(r) - 1.5
    
    cons = [{'type': 'ineq', 'fun': constraint}]
    
    def in_bounds(c, r):
        for i in range(n):
            x, y = c[i]
            if (x - r[i] < -1e-12 or x + r[i] > 1 + 1e-12
                or y - r[i] < -1e-12 or y + r[i] > 1 + 1e-12):
                return False
        return True
    
    def non_overlap(c, r):
        dists = squareform(pdist(c))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-12:
                    return False
        return True
    
    def fitness(c, r):
        return -np.sum(r) if in_bounds(c, r) and non_overlap(c, r) else np.inf
    
    result = minimize(objective, radii, method='SLSQP', constraints=cons)
    radii = result.x
    
    centers = np.random.rand(n, 2)
    for _ in range(100):
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < radii[i] + radii[j]:
                    dir_vec = (centers[i] - centers[j]) / dist
                    centers[i] += dir_vec * (radii[i] + radii[j] - dist) * 0.5
                    centers[j] -= dir_vec * (radii[i] + radii[j] - dist) * 0.5
    
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0
    
    return centers, radii, np.sum(radii)
