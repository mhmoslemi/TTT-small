import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Create hexagonal grid with boundary-aware coordinates
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        if row % 2 == 0:
            x = col + 0.5
        else:
            x = col + 0.5 - 0.5
        y = row + 0.5
        centers[i] = [x, y]
    
    # Normalize to [0,1]^2
    centers -= centers.min(axis=0)
    centers /= centers.max(axis=0) - centers.min(axis=0)

    # Initial estimate for radii

        return -np.sum(x)

                    return -np.inf
        return 0

    cons = [{'type': 'ineq', 'fun': constraint}]

    # Optimization
    result = minimize(objective, radii, constraints=cons, method='SLSQP', tol=1e-8)

    # Final adjustment to maintain spacing and boundary constraints
    dists = squareform(pdist(centers))

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
