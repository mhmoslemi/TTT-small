import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Define hexagonal grid with spacing adjusted for boundary constraints
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.1)

    # Generate initial hexagonal grid
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]

    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(radii):
        return -np.sum(radii)

    # Constraints: sum_radii >= 0.01 and sum_radii <= 1.0 - 0.01
    def constraint_min(radii):
        return np.sum(radii) - 0.01

    def constraint_max(radii):
        return 1.0 - np.sum(radii) - 0.01

    cons = [{'type': 'ineq', 'fun': constraint_min},
            {'type': 'ineq', 'fun': constraint_max}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', tol=1e-8)

    # Adjust radii to maintain spacing
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
