import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)
    
    def objective(x):
        return -np.sum(x)
    
    def constraint(x):
        return np.sum(x) - 1.0
    
    cons = [{'type': 'ineq', 'fun': constraint}]
    
    def update_centers(radii):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] + 1e-8:
                    dir_vec = centers[i] - centers[j]
                    dist = np.sqrt(np.sum(dir_vec**2))
                    dir_vec /= dist
                    shift = (radii[i] + radii[j] - dists[i, j]) * dir_vec * 0.5
                    centers[i] -= shift
                    centers[j] += shift
    
    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)
    radii = result.x
    
    for _ in range(10):
        update_centers(radii)
    
    dists = squareform(pdist(centers))
    for i in range(n):

            if dists[i, j] < radii[i] + radii[j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
