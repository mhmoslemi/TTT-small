import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.ones(n) * 0.1

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 1.5

    cons = [{'type': 'ineq', 'fun': constraint}]

    def dist_constraint(i, j, r):
        def cons_func(r):
            return np.sqrt(np.sum((centers[i] - centers[j])**2)) - r[i] - r[j]
        return cons_func

    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: dist_constraint(i, j, r)})

    def boundary_constraint(i, r):
        x, y = centers[i]
        return np.array([x - r[i], y - r[i], x + r[i] - 1, y + r[i] - 1])

    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda r, i=i: boundary_constraint(i, r)})

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 1000})

    radii = result.x

    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
