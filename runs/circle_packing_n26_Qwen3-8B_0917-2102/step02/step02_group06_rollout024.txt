import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Create a more efficient hexagonal-like layout with spacing
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = 0.1
    
    # Rescale to unit square
    min_coords = centers.min(axis=0)
    max_coords = centers.max(axis=0)
    centers = (centers - min_coords) / (max_coords - min_coords)
    
    def objective(r):
        return -np.sum(r)
    
    def constraint(r):
        return np.sum(r)
    
    cons = [{'type': 'ineq', 'fun': constraint, 'jac': lambda r: np.ones(n)}]
    
    bounds = [(0, 1.0) for _ in range(n)]
    
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)
    radii = result.x
    
    # Compute pairwise distances
    dists = squareform(pdist(centers))
    
    # Update radii to maintain spacing
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio
    
    # Adjust radii to ensure full containment in unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
