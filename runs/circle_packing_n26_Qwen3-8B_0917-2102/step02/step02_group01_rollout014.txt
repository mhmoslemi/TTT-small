import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Start with a hexagonal grid configuration
    centers = []
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers.append([x, y])
    centers = np.array(centers)
    
    # Scale centers to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))
    
    # Initial radii based on minimum spacing
    r = 0.1
    radii = np.full(n, r)
    
    def objective(x):
        return -np.sum(x)
    
    # Constraints: radii must be at least 0.01
    cons = [{'type': 'ineq', 'fun': lambda x: x - 0.01}]
    
    # Optimize radii
    result = minimize(objective, radii, constraints=cons, method='SLSQP')
    radii = result.x
    
    # Compute pairwise distances
    dists = squareform(pdist(centers))
    
    # Adjust radii to maintain spacing
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio
    
    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
