import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.1)
    col, row = 0, 0
    for i in range(n):
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        col += 1
        if col >= 5:
            col = 0
            row += 1

    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)

    def constraint(x):
        return np.sum(x)

    cons = [{'type': 'ineq', 'fun': constraint, 'jac': lambda x: np.ones(n)}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'gtol': 1e-10})

    radii = result.x

    # Adjust radii to maintain spacing
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
