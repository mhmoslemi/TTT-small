import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    row_count = 5
    col_count = 5
    for i in range(n):
        row = i // col_count
        col = i % col_count
        if row % 2 == 0:
            x = col + 0.5 * (1 - 2 * (row % 2))
        else:
            x = col + 0.5
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = 0.1

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 1.5

    cons = [{'type': 'ineq', 'fun': constraint}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP')

    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
