import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal grid with boundary-aware placement
    centers = np.zeros((n, 2))
    col = 0
    row = 0
    for i in range(n):
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        col += 1
        if col >= 5:
            col = 0
            row += 1

    # Convert to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    # Initial radii
    radii = np.full(n, 0.1)

    def objective(r):
        return -np.sum(r)

    def constraint(x):
        return np.sum(x)

    cons = [{'type': 'ineq', 'fun': constraint, 'args': ()}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', tol=1e-10)

    # Adjust radii to maintain spacing
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            min_dist = dists[i, j]
            if min_dist < 1e-8:
                continue
            r1, r2 = result.x[i], result.x[j]
            if min_dist < r1 + r2:
                alpha = (min_dist - 1e-8) / (r1 + r2)
                result.x[i] *= alpha
                result.x[j] *= alpha

    # Final check to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result.x[i] = 0.0

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
