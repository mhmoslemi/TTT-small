import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Define a hexagonal packing with 5 rows and 6 columns for 26 circles
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 6
        col = i % 6
        x = col / 5.0
        y = row / 5.0 + (0.5 if row % 2 == 0 else 0)
        centers[i] = [x, y]

    def objective(r):
        return -np.sum(r)

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    bounds = [(0, 1) for _ in range(n)]
    cons = [{'type': 'ineq', 'fun': constraint_overlap}]
    result = minimize(objective, np.full(n, 0.1), constraints=cons, method='SLSQP', bounds=bounds)

    # Adjust radii to ensure circles are within the square
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
