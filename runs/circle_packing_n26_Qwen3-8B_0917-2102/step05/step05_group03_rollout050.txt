import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal grid with 5 rows and 6 columns
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Create a hexagonal grid with 5 rows and 6 columns
    for i in range(n):
        row = i // 6
        col = i % 6
        # Offset for hexagonal packing
        if row % 2 == 0:
            x = col * (1 + 1e-8)  # Slight spacing for boundary alignment
        else:
            x = (col + 0.5) * (1 + 1e-8)
        y = row * (1 + 1e-8)
        centers[i] = [x, y]
        radii[i] = 0.1

    # Scale to unit square with boundary alignment
    min_x, max_x = centers[:, 0].min(), centers[:, 0].max()
    min_y, max_y = centers[:, 1].min(), centers[:, 1].max()
    centers[:, 0] = (centers[:, 0] - min_x) / (max_x - min_x)
    centers[:, 1] = (centers[:, 1] - min_y) / (max_y - min_y)

    def objective(r):
        return -np.sum(r)

    def constraint_sum(r):
        return np.sum(r) - 2.636

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap}]
    cons.append({'type': 'eq', 'fun': lambda r: np.sum(r) - 2.636})

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n, tol=1e-10, options={'ftol': 1e-10, 'maxiter': 500})

    # Final check and adjustment
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
