import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initialize centers in a hexagonal-like grid with varying row spacing
    centers = np.zeros((n, 2))
    row = 0
    col = 0
    for i in range(n):
        if i % 5 == 0:
            row += 1
            col = 0
        x = col / 4.5 + 0.5 / 4.5
        y = row / 5.0
        centers[i] = [x, y]
        col += 1

    def objective(r):
        return -np.sum(r)

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    def constraint_boundary(r):
        for i in range(n):
            x, y = centers[i]
            r_i = r[i]
            if (x - r_i < -1e-12 or x + r_i > 1 + 1e-12
                    or y - r_i < -1e-12 or y + r_i > 1 + 1e-12):
                return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap},
            {'type': 'ineq', 'fun': constraint_boundary}]
    bounds = [(0, 1) for _ in range(n)]

    result = minimize(objective, np.full(n, 0.1), bounds=bounds, constraints=cons, method='SLSQP')

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
