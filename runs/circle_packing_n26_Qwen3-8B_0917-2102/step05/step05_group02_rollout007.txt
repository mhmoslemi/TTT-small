from scipy.optimize import minimize

    centers = np.random.rand(n, 2) * 0.9 + 0.05  # Place circles with some spacing

        return -np.sum(r)

    def constraint1(r):
        return np.sum(r)  # For equality constraint

        dists = np.sqrt(np.sum((centers[:, np.newaxis] - centers[np.newaxis, :]) ** 2, axis=2))

            for j in range(i + 1, n):

        r = result['x'][i]

            result['x'][i] = 0

    return centers, result['x'], np.sum(result['x'])
