<strategy>Reposition circles in a hexagonal grid with variable spacing to exploit corner and boundary space more efficiently.</strategy>

```python
import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal grid with 5 rows and 5 columns, with some circles near corners
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        row = i // 5
        col = i % 5
        if row == 0:
            if col == 0:
                x = 0.1
                y = 0.1
            elif col == 4:
                x = 0.9
                y = 0.1
            else:
                x = col * 0.2 + 0.1
                y = 0.1
        elif row == 1:
            if col == 0:
                x = 0.1
                y = 0.3
            elif col == 4:
                x = 0.9
                y = 0.3
            else:
                x = col * 0.2 + 0.1
                y = 0.3
        elif row == 2:
            if col == 0:
                x = 0.1
                y = 0.5
            elif col == 4:
                x = 0.9
                y = 0.5
            else:
                x = col * 0.2 + 0.1
                y = 0.5
        elif row == 3:
            if col == 0:
                x = 0.1
                y = 0.7
            elif col == 4:
                x = 0.9
                y = 0.7
            else:
                x = col * 0.2 + 0.1
                y = 0.7
        elif row == 4:
            if col == 0:
                x = 0.1
                y = 0.9
            elif col == 4:
                x = 0.9
                y = 0.9
            else:
                x = col * 0.2 + 0.1
                y = 0.9
        centers[i] = [x, y]
        radii[i] = 0.1

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 2.636

    cons = [{'type': 'ineq', 'fun': lambda r: np.array([r[i] for i in range(n)])}]
    cons.append({'type': 'eq', 'fun': constraint})

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    def constraint_overlap(r):
        return check_overlap(r)

    cons.append({'type': 'ineq', 'fun': constraint_overlap})

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n, tol=1e-8, options={'ftol': 1e-10, 'maxiter': 500})

    # Final check and adjustment
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
```
