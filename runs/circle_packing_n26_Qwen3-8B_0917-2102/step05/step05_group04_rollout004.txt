import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Hexagonal grid with variable row spacing
    centers = np.zeros((n, 2))
    row_height = 0.75
    col_width = 0.75
    for i in range(n):
        row = i // 5
        col = i % 5
        y = row * row_height + 0.5 * row_height
        x = col * col_width + 0.5 * (1 - 2 * (row % 2)) * col_width / 2
        centers[i] = [x, y]
    
    # Normalize to unit square
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    # Initial radii
    radii = np.full(n, 0.1)

    def objective(r):
        return -np.sum(r)

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint} for _ in range(n)]
    for i in range(n):
        def constraint(x, i=i):
            r = x[i]
            x_i, y_i = centers[i]
            if (x_i - r < -1e-12 or x_i + r > 1 + 1e-12 or
                y_i - r < -1e-12 or y_i + r > 1 + 1e-12):
                return -1
            return 0
        cons.append({'type': 'ineq', 'fun': constraint})

    cons.append({'type': 'eq', 'fun': lambda r: np.sum(r) - 2.636})

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final check and adjustment for overlaps
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
