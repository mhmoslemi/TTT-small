import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal grid with 5 rows of 5 circles and 1 row of 6 circles
    centers = np.zeros((n, 2))
    row_height = np.sqrt(3)/2
    col_width = 1 - 1/np.sqrt(3)
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col * col_width + 0.5 * (1 - 2 * (row % 2)) * col_width / 2
        y = row * row_height + 0.5 * row_height
        centers[i] = [x, y]

    # Normalize to unit square
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    # Initial radii
    radii = np.full(n, 0.1)

    def objective(r):
        return -np.sum(r)

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_boundaries}]
    cons.append({'type': 'ineq', 'fun': constraint_overlap})

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final check and adjustment for boundaries
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    # Final check and adjustment for overlaps
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii

def constraint_boundaries(r):
    for i in range(len(r)):
        x, y = centers[i]
        r_val = r[i]
        if x - r_val < -1e-12 or x + r_val > 1 + 1e-12 or y - r_val < -1e-12 or y + r_val > 1 + 1e-12:
            return -1
    return 0
