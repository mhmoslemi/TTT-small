import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate non-uniform hexagonal packing with boundary-aware placement
    centers = np.zeros((n, 2))
    col_offsets = np.linspace(0, 1, n + 1)[1:-1]
    row_heights = np.linspace(0, 1, n + 1)[1:-1]
    for i in range(n):
        col = i // 5
        row = i % 5
        if row % 2 == 0:
            x = col_offsets[i % 5] + 0.5 * (col_offsets[i % 5 + 1] - col_offsets[i % 5])
        else:
            x = col_offsets[i % 5]
        y = row_heights[row]
        centers[i] = [x, y]

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 2.636

    def check_overlap(r):

        for i in range(n):

                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': check_overlap}]
    cons.append({'type': 'eq', 'fun': lambda r: np.sum(r) - 2.636})

    result = minimize(objective, np.full(n, 0.1), constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for overlap
    dists = squareform(pdist(centers))

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
