import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    idx = 0
    for row in range(5):
        for col in range(5):
            if row % 2 == 0:
                x = col / 4.5
            else:
                x = (col + 0.5) / 4.5
            y = row / 5.0
            centers[idx] = [x, y]
            idx += 1
        if row < 4:
            for col in range(5):
                if row % 2 == 1:
                    x = col / 4.5
                else:
                    x = (col + 0.5) / 4.5
                y = row / 5.0 + 0.5 / 5.0
                centers[idx] = [x, y]
                idx += 1

    def objective(r):
        return -np.sum(r)

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap}]
    cons.append({'type': 'eq', 'fun': lambda r: np.sum(r) - 2.636})

    result = minimize(objective, np.full(n, 0.1), constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
