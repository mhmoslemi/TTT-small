import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Improved initial placement: cluster some circles near boundaries and use hexagonal grid for others
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Place 10 circles near the boundary
    for i in range(10):
        if i < 5:
            centers[i] = [0.1 + i * 0.1, 0.5]
        else:
            centers[i] = [0.5, 0.1 + (i - 5) * 0.1]
        radii[i] = 0.1
    
    # Place remaining 16 circles in staggered hexagonal grid
    for i in range(10, n):
        row = (i - 10) // 5
        col = (i - 10) % 5
        x = col + 0.5
        if row % 2 == 1:
            x -= 0.5
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = 0.08

    # Scale centers to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)

    def constraint_sum(r):
        return np.sum(r) - 2.636

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap}]
    cons.append({'type': 'eq', 'fun': constraint_sum})

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final check and adjustment
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
