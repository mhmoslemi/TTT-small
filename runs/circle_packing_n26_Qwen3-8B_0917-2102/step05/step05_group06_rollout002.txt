import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    initial_centers = np.zeros((n, 2))
    for i in range(n):
        initial_centers[i] = [i % 5 / 4, i // 5 + 0.5]
    
    def objective(r):
        return -np.sum(r)
    
    def constraint_sum(r):
        return np.sum(r) - 2.636
    
    def constraint_overlap(r):
        dists = squareform(pdist(initial_centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    cons = [{'type': 'ineq', 'fun': constraint_overlap}]
    cons.append({'type': 'eq', 'fun': constraint_sum})
    
    result = minimize(objective, np.full(n, 0.1), constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)
    
    # Adjust for boundary constraints
    final_centers = initial_centers.copy()
    final_radii = result['x'].copy()
    for i in range(n):
        x, y = final_centers[i]
        r = final_radii[i]
        if x - r < 0:
            final_radii[i] = x
        elif x + r > 1:
            final_radii[i] = 1 - x
        if y - r < 0:
            final_radii[i] = y
        elif y + r > 1:
            final_radii[i] = 1 - y
    
    # Final overlap check and adjustment
    dists = squareform(pdist(final_centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < final_radii[i] + final_radii[j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (final_radii[i] + final_radii[j])
                final_radii[i] *= ratio
                final_radii[j] *= ratio
    
    sum_radii = np.sum(final_radii)
    return final_centers, final_radii, sum_radii
