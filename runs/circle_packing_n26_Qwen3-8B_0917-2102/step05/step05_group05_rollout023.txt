import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initialize with a staggered grid with variable row heights
    centers = np.random.rand(n, 2)
    for i in range(n):
        if i % 2 == 0:
            centers[i, 1] = i / 10.0
        else:
            centers[i, 1] = (i + 0.5) / 10.0
        centers[i, 0] = i / 10.0

    def objective(r):
        return -np.sum(r)

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    def constraint_boundary(r):
        for i in range(n):
            x, y = centers[i]
            r_val = r[i]
            if (x - r_val < -1e-12 or x + r_val > 1 + 1e-12
                or y - r_val < -1e-12 or y + r_val > 1 + 1e-12):
                return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap},
            {'type': 'ineq', 'fun': constraint_boundary}]

    result = minimize(objective, np.full(n, 0.1), constraints=cons, method='SLSQP', bounds=Bounds(0, 1))

    # Refine with adaptive adjustments for boundary circles
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
