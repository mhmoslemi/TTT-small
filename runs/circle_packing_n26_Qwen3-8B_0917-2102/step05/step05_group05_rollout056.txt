import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate a hexagonal grid with rows and columns adjusted for unit square
    centers = np.zeros((n, 2))
    row = 0
    col = 0
    for i in range(n):
        centers[i] = [
            0.5 + (i % 5 - 2.5) / 5.0,
            (i // 5) / 5.0 + (i % 5) / 10.0
        ]

    def objective(r):
        return -np.sum(r)

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    def constraint_boundary(r):
        for i in range(n):
            x, y = centers[i]
            r_val = r[i]
            if (x - r_val < -1e-12 or x + r_val > 1 + 1e-12
                    or y - r_val < -1e-12 or y + r_val > 1 + 1e-12):
                return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap},
            {'type': 'ineq', 'fun': constraint_boundary}]

    result = minimize(objective, np.full(n, 0.1), constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
