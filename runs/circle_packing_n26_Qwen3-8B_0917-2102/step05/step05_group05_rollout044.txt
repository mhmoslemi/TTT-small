```python
import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Define a hexagonal grid with staggered rows
    centers = np.zeros((n, 2))
    row_height = 1.0 / np.sqrt(3)
    col_width = 1.0 / (np.ceil(np.sqrt(n))) 

    for i in range(n):
        row = i // np.ceil(np.sqrt(n))
        col = i % np.ceil(np.sqrt(n))
        y = row * row_height
        x = col * col_width + (row % 2) * col_width / 2
        centers[i] = [x, y]

    def objective(r):
        return -np.sum(r)

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap}]
    cons.append({'type': 'ineq', 'fun': lambda r: np.sum(r) - 2.636})

    result = minimize(objective, np.full(n, 0.1), constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
```
