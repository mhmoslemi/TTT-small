from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

    centers = []

                y = row + 0.5
            else:
                x = col + 0.5
                y = row + 0.5 + 0.25
            if len(centers) < n:
                centers.append([x, y])
    
    # Convert to numpy array
    centers = np.array(centers)
    
    def objective(r):
        return -np.sum(r)

        return np.sum(r) - 2.636
    
    # Initial guess based on lattice spacing

    # Optimization constraints
    cons = [{'type': 'ineq', 'fun': lambda r: np.array([r[i] for i in range(n)])}]
    cons.append({'type': 'eq', 'fun': lambda r: np.sum(r) - 2.636})
    
    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    cons.append({'type': 'ineq', 'fun': check_overlap})
    
    # Bounds for radii
    bounds = [(0, 1) for _ in range(n)]
    
    result = minimize(objective, initial_guess, constraints=cons, method='SLSQP', bounds=bounds)
