import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    # Create a hexagonal packing with 5 columns and 6 rows
    centers = np.zeros((n, 2))
    row = 0
    col = 0
    for i in range(n):
        if i % 5 == 0:
            row += 1
            col = 0
        x = col + 0.5
        y = row + 0.5
        centers[i] = [x, y]
        col += 1

    # Scale to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 2.636

    cons = [{'type': 'ineq', 'fun': lambda r: np.array([r[i] for i in range(n)])}]
    cons.append({'type': 'eq', 'fun': constraint})

    def check_overlap(r):
        dists = np.sqrt(np.sum((centers[:, np.newaxis] - centers[np.newaxis, :])**2, axis=2))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons.append({'type': 'ineq', 'fun': check_overlap})

    result = minimize(objective, np.full(n, 0.1), constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Adjust radii to fix overlaps
    dists = np.sqrt(np.sum((centers[:, np.newaxis] - centers[np.newaxis, :])**2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
