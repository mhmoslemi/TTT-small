import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Hexagonal packing with 5 rows and 6 columns (adjusted for boundary)
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 6
        col = i % 6
        x = col + 0.5 * (row % 2)
        y = row + 0.5 * (row % 2)
        # Scale to fit within [0,1]
        x = x / (np.max([col + 0.5 * (row % 2) for i in range(n)]) + 1)
        y = y / (np.max([row + 0.5 * (row % 2) for i in range(n)]) + 1)
        centers[i] = [x, y]

    def objective(r):
        return -np.sum(r)

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap}]

    result = minimize(objective, np.full(n, 0.1), constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
