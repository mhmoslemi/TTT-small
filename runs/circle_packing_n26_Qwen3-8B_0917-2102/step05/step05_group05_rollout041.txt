import numpy as np
from scipy.optimize import minimize, Bounds

def run_packing():
    n = 26
    # Initial non-uniform grid with adaptive spacing
    centers = np.zeros((n, 2))
    for i in range(n):
        col = i % 5
        row = i // 5
        if row % 2 == 0:
            x = col / 4.5
        else:
            x = (col + 0.5) / 4.5
        y = row / 5.0
        centers[i] = [x, y]
    
    def objective(r):
        return -np.sum(r)
    
    def constraint_overlap(r):
        dists = np.sqrt(((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2).sum(axis=2))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    def constraint_boundary(r):
        for i in range(n):
            x, y = centers[i]
            r_val = r[i]
            if x - r_val < -1e-12 or x + r_val > 1 + 1e-12 or y - r_val < -1e-12 or y + r_val > 1 + 1e-12:
                return -1
        return 0
    
    cons = [{'type': 'ineq', 'fun': constraint_overlap}, {'type': 'ineq', 'fun': constraint_boundary}]
    
    result = minimize(objective, np.full(n, 0.1), method='SLSQP', bounds=Bounds(np.zeros(n), np.full(n, 1)), constraints=cons)
    
    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
