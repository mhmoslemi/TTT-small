import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate adaptive hexagonal grid with variable row heights and column spacing
    centers = np.zeros((n, 2))
    row_heights = np.linspace(0.6, 0.8, 5)
    col_widths = np.linspace(0.6, 0.8, 6)
    
    idx = 0
    for r in range(5):
        for c in range(6):
            if idx >= n:
                break
            x = c * col_widths[c] + 0.5 * (1 - 2 * (r % 2)) * col_widths[c] / 2
            y = r * row_heights[r] + 0.5 * row_heights[r]
            centers[idx] = [x, y]
            idx += 1
    
    # Normalize to unit square
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    # Initial radii
    radii = np.full(n, 0.1)

    def objective(r):
        return -np.sum(r)

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap}]
    cons.append({'type': 'eq', 'fun': lambda r: np.sum(r) - 2.636})

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    # Final check and adjustment for overlaps
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
