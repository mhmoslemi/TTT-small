import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints_func(centers, radii):
        n = len(radii)
        cons = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            cons.append((x - r, -1e-12))
            cons.append((x + r, 1 + 1e-12))
            cons.append((y - r, -1e-12))
            cons.append((y + r, 1 + 1e-12))
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                cons.append((dist - radii[i] - radii[j], -1e-12))
        return cons

    initial_centers = np.random.rand(26, 2)
    initial_radii = np.random.rand(26) * 0.1

    bounds = []
    for _ in range(26):
        bounds.append((0, 1))
    bounds += [(0, 1) for _ in range(26)]

    result = minimize(
        objective,
        np.concatenate((initial_centers.reshape(-1), initial_radii)),
        method='SLSQP',
        bounds=bounds,
        constraints={
            'type': 'ineq',
            'fun': lambda x: np.array(constraints_func(
                x[:52].reshape(26, 2),
                x[52:]
            )),
            'name': 'constraints'
        }
    )

    centers = result.x[:52].reshape(26, 2)
    radii = result.x[52:]
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
