import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import LinearConstraint, Bounds

def run_packing():
    # Initial hexagonal packing with 26 circles
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    radius = 0.1
    row = 0
    col = 0
    for i in range(n):
        centers[i, 0] = col / 2 + 0.5
        centers[i, 1] = row / np.sqrt(3) + 0.5
        col += 1
        if col % 2 == 0:
            row += 1
        radii[i] = radius

    # Distance threshold for repulsion
    dist_threshold = 1e-6

    # Optimization variables
    x = np.concatenate((centers.flatten(), radii))
    bounds = Bounds(np.zeros_like(x), np.inf * np.ones_like(x))

    def objective(x):
        centers_flat = x[:n*2].reshape((n, 2))
        radii = x[n*2:]
        return -np.sum(radii)  # Maximize sum of radii

    def constraints(x):
        centers_flat = x[:n*2].reshape((n, 2))
        radii = x[n*2:]
        constraints = []
        # Boundary constraints
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12:
                constraints.append(0)
            if y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                constraints.append(0)
        # Circle-circle distance constraints
        for i in range(n):
            for j in range(i+1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - dist_threshold:
                    constraints.append(0)
        return np.array(constraints)

    res = minimize(objective, x, bounds=bounds, constraints=constraints, method='SLSQP', tol=1e-8, options={'maxiter': 500})
    x_opt = res.x
    centers_opt = x_opt[:n*2].reshape((n, 2))
    radii_opt = x_opt[n*2:]
    return centers_opt, radii_opt, np.sum(radii_opt)
