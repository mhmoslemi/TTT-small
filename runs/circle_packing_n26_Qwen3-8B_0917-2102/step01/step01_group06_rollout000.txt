import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    np.random.seed(42)
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraint1(radii):
        return np.sum(radii)
    
    def constraint2(radii):
        return np.sum(radii) - 2.636
    
    bounds = Bounds(0, 1)
    
    def dist_matrix(centers):
        return squareform(pdist(centers))
    
    def constraint3(radii):
        dists = dist_matrix(centers)
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        return 0
    
    res = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint2},
            {'type': 'eq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint3}
        ]
    )
    
    return (centers, res.x, np.sum(res.x))
