from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create initial hexagonal grid with spacing adjusted for boundary constraints
    centers = np.zeros((n, 2))

    row_indices = np.sort(row_indices)

    centers = centers[:n]

        return np.maximum(0, 1 - np.max(x) - 1e-8)
    
    cons = [{'type': 'ineq', 'fun': boundary_constraint}]
    
    # Optimization
    result = minimize(objective, radii, constraints=cons, method='SLSQP')
    
    radii = result.x

    dists = squareform(pdist(centers))
    for i in range(n):

    sum_radii = np.sum(radii)
