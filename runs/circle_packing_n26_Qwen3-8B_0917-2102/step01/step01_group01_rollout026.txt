import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.cluster.hierarchy import linkage, fcluster
from itertools import combinations

def run_packing():
    n = 26
    centers = []
    radii = []

    # Initial hexagonal packing
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col / 4.0 + 0.05
        y = row / 4.0 + 0.05
        r = 0.05
        centers.append([x, y])
        radii.append(r)

    centers = np.array(centers)
    radii = np.array(radii)

    def objective(radii):
        return -np.sum(radii)

    def constraint(x):
        centers_flat = x.reshape(-1, 2)
        dist_matrix = squareform(pdist(centers_flat))
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return 1
        return 0

    bounds = [(0, 1) for _ in range(2 * n)]
    result = minimize(objective, centers.reshape(-1), bounds=bounds, constraints={"type": "ineq", "fun": constraint}, method="SLSQP")

    optimized_centers = result.x.reshape(-1, 2)
    optimized_radii = np.clip(radii, 0, 1)

    # Boundary refinement
    for i in range(n):
        x, y = optimized_centers[i]
        r = optimized_radii[i]
        if x - r < 0:
            optimized_radii[i] = max(0, x)
        elif x + r > 1:
            optimized_radii[i] = max(0, 1 - x)
        if y - r < 0:
            optimized_radii[i] = max(0, y)
        elif y + r > 1:
            optimized_radii[i] = max(0, 1 - y)

    # Final validation
    for i in range(n):
        x, y = optimized_centers[i]
        r = optimized_radii[i]
        if (x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1):
            optimized_radii[i] = max(0, min(r, 1 - max(x, 1 - x, y, 1 - y)))

    dist_matrix = squareform(pdist(optimized_centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dist_matrix[i, j] < optimized_radii[i] + optimized_radii[j] - 1e-8:
                optimized_radii[i] = max(0, (dist_matrix[i, j] - optimized_radii[j]) * 0.5)
                optimized_radii[j] = max(0, (dist_matrix[i, j] - optimized_radii[i]) * 0.5)

    return optimized_centers, optimized_radii, np.sum(optimized_radii)
