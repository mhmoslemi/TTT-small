import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import math

def run_packing():
    n = 26
    # Initial guess: hexagonal packing with 5 rows
    centers = np.zeros((n, 2))
    r = np.zeros(n)
    row = 0
    col = 0
    for i in range(n):
        centers[i, 0] = col / 4.0 + 0.25
        centers[i, 1] = row / 3.0 + 0.33
        if i % 5 == 4:
            row += 1
            col = 0
        else:
            col += 1
    r = np.full(n, 0.1)

    # Optimization
    def objective(x):
        r = x[:n]
        return -np.sum(r)

    def constraints(x):
        r = x[:n]
        centers_flat = x[n:]
        centers = centers_flat.reshape((n, 2))
        for i in range(n):
            x_i, y_i = centers[i]
            if x_i - r[i] < -1e-12 or x_i + r[i] > 1 + 1e-12 or y_i - r[i] < -1e-12 or y_i + r[i] > 1 + 1e-12:
                return 1
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < r[i] + r[j] - 1e-12:
                    return 1
        return 0

    x0 = np.concatenate((r, centers.flatten()))
    res = minimize(objective, x0, constraints={"type": "ineq", "fun": constraints}, method='SLSQP', options={'maxiter': 500})
    x = res.x
    radii = x[:n]
    centers = x[n:].reshape((n, 2))
    return centers, radii, np.sum(radii)
