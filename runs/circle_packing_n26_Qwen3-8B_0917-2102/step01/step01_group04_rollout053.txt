import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance

def run_packing():

    n = 26
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n) * 0.1
    best_sum = 0.0

    def cost(radii):
        nonlocal best_sum
        sum_r = np.sum(radii)
        if sum_r > best_sum:
            best_sum = sum_r
        return -sum_r

    def constraint_func(x, centers, radii):
        constraints = []
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                constraints.append(dist - (radii[i] + radii[j]) >= 1e-8)
        return constraints

    def boundary_constraints(x, centers, radii):
        for i in range(n):
            if centers[i, 0] - radii[i] < -1e-8 or centers[i, 0] + radii[i] > 1 + 1e-8:
                return False
            if centers[i, 1] - radii[i] < -1e-8 or centers[i, 1] + radii[i] > 1 + 1e-8:
                return False
        return True

            radii[i] = np.random.rand() * 0.2

        )

            best_centers = result.x[:2*n].reshape(n, 2)

            return (best_centers, best_radii, np.sum(best_radii))

            radii = np.random.rand(n) * 0.2
