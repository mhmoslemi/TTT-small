import numpy as np
from scipy.optimize import minimize

def run_packing():
    np.random.seed(42)
    n = 26
    # Hexagonal packing with staggered rows
    rows = 5
    cols = 6
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (col % 2)
        y = row + 0.5 * (row % 2)
        centers[i] = [x, y]
        radii[i] = 0.1

    # Initial sum of radii
    sum_radii = np.sum(radii)

    # Define function to compute sum of radii given centers
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Constraints: non-overlap and boundary
    def constraints(radii):
        res = []
        for i in range(n):
            res.append(radii[i] >= 0)
            x, y = centers[i]
            res.append(radii[i] <= 1 - max(x, 1 - x, y, 1 - y))
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                res.append(radii[i] + radii[j] <= dist)
        return res

    # Optimization
    bounds = [(0, 1) for _ in range(n)]
    result = minimize(objective, radii, bounds=bounds, constraints=constraints)

    return (centers, result.x, np.sum(result.x))
