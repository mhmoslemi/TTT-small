import numpy as np
from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d

def run_packing():
    np.random.seed(42)
    
    def objective(r):
        return -np.sum(r)
    
    def constraints(r, centers):
        constraints = []
        for i in range(len(r)):
            constraints.append((centers[i, 0] - r[i] >= 0))
            constraints.append((centers[i, 1] - r[i] >= 0))
            constraints.append((1 - centers[i, 0] - r[i] >= 0))
            constraints.append((1 - centers[i, 1] - r[i] >= 0))
        for i in range(len(r)):
            for j in range(i + 1, len(r)):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                constraints.append((dist - r[i] - r[j] >= 1e-8))
        return constraints
    
    def pack_hexagonal(n):
        centers = []
        r = []
        for i in range(n):
            r_i = np.random.uniform(0.05, 0.1)
            r.append(r_i)
            col = i // 7
            row = i % 7
            x = 0.5 + (col - 3) * (0.5 + r_i)
            y = 0.5 + (row - 3) * (0.5 + r_i)
            if row % 2 == 1:
                x += (0.5 + r_i)
            centers.append([x, y])
        return np.array(centers), np.array(r)
    
    centers, radii = pack_hexagonal(26)
    
    def constraint_func(r):
        return constraints(r, centers)
    
    bounds = [(0, 1) for _ in range(26)]
    
    result = minimize(objective, radii, bounds=bounds, constraints=constraint_func)
    
    if not result.success:
        raise Exception("Optimization failed")
    
    return centers, result.x, np.sum(result.x)
