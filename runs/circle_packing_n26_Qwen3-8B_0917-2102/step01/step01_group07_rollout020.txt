import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial guess for centers using hexagonal packing
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    rows = int(np.ceil(np.sqrt(num_circles)))
    cols = int(np.ceil(num_circles / rows))
    
    for i in range(num_circles):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (1 - np.cos(2 * np.pi / cols)) * (1 - row % 2)
        y = row + 0.5
        centers[i] = [x, y]
    
    # Optimization parameters
    bounds = [(0, 1) for _ in range(2 * num_circles)]
    constraints = []
    
    # Non-overlap constraints
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            def dist_constraint(x):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                r_i = x[2 * i]
                r_j = x[2 * j + 1]
                return dist - r_i - r_j + 1e-6
            constraints.append({'type': 'ineq', 'fun': dist_constraint})
    
    # Boundary constraints
    for i in range(num_circles):
        def x_left_constraint(x):
            return x[2 * i] - centers[i][0] + 1e-6
        constraints.append({'type': 'ineq', 'fun': x_left_constraint})
        
        def x_right_constraint(x):
            return (1 + 1e-6) - x[2 * i] - centers[i][0]
        constraints.append({'type': 'ineq', 'fun': x_right_constraint})
        
        def y_bottom_constraint(x):
            return x[2 * i + 1] - centers[i][1] + 1e-6
        constraints.append({'type': 'ineq', 'fun': y_bottom_constraint})
        
        def y_top_constraint(x):
            return (1 + 1e-6) - x[2 * i + 1] - centers[i][1]
        constraints.append({'type': 'ineq', 'fun': y_top_constraint})
    
    # Objective function: maximize sum of radii
    def objective(x):
        return -np.sum(x[1::2])  # Negative for minimization
    
    # Initial guess for radii
    initial_guess = np.zeros(2 * num_circles)
    initial_guess[1::2] = 0.05
    
    result = minimize(objective, initial_guess, method='SLSQP', bounds=bounds, constraints=constraints)
    
    if not result.success:
        raise ValueError("Optimization failed")
    
    radii = result.x[1::2]
    centers = np.array([[col + 0.5 * (1 - np.cos(2 * np.pi / cols)) * (1 - row % 2), 
                         row + 0.5] for row in range(rows) for col in range(cols)])
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
