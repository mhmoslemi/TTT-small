import numpy as np
from scipy.optimize import minimize

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        constraints = []
        for i in range(26):
            constraints.append(radii[i] >= 0)
        return constraints

    def packing_func(centers, radii):
        for i in range(26):
            x, y = centers[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12 or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return False
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    # Generate a hexagonal grid pattern with 26 circles
    rows = 6
    cols = 5
    centers = np.zeros((26, 2))
    for i in range(rows):
        for j in range(cols):
            x = j + 0.5 * (i % 2)
            y = i * np.sqrt(3) / 2
            centers[i * cols + j] = [x, y]

    # Scale centers to fit in unit square
    max_x = np.max(centers[:, 0])
    max_y = np.max(centers[:, 1])
    centers /= np.max([max_x, max_y])

    # Initial guess for radii
    radii = np.ones(26) * 0.1

    # Optimization
    bounds = [(0, 1) for _ in range(26)]
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': lambda r: np.array([packing_func(centers, r)])}]
    )

    return (centers, result.x, np.sum(result.x))
