import numpy as np
from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraint_centers(centers, radii):
        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return 1.0
        return 0.0

    def constraint_overlap(centers, radii):
        for i, j in combinations(range(26), 2):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < radii[i] + radii[j] - 1e-12:
                return 1.0
        return 0.0

    def bounds_func(x):
        return [(0, 1) for _ in range(52)]

    def initial_guess():
        grid = np.linspace(0, 1, 5)
        points = np.array([[x, y] for x in grid for y in grid])
        return points[:26]

    centers = initial_guess()
    radii = np.ones(26) * 0.1

    cons = [
        {'type': 'ineq', 'fun': lambda x: -constraint_centers(x[:52], x[52:])},
        {'type': 'ineq', 'fun': lambda x: -constraint_overlap(x[:52], x[52:])}
    ]

    bounds = bounds_func

    result = minimize(
        fun=objective,
        x0=np.concatenate([centers.reshape(-1), radii]),
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-8,
        options={'maxiter': 500, 'disp': False}
    )

    if not result.success:
        raise Exception("Optimization failed")

    centers = result.x[:52].reshape(26, 2)
    radii = result.x[52:]
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
