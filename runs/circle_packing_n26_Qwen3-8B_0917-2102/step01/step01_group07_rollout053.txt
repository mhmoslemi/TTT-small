import numpy as np
from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d

def run_packing():
    # Initial hexagonal packing with 26 circles
    n = 26
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        col = i % cols
        row = i // cols
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        centers[i, 0] = x / (cols - 1) if cols > 1 else 0.5
        centers[i, 1] = y / (rows - 1) if rows > 1 else 0.5
        radii[i] = 0.05

    # Function to compute objective and constraints
    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii

    def constraint_overlap(i, j):
        def cons(x):
            r_i, r_j = x[i], x[j]
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            distance = np.sqrt(dx*dx + dy*dy)
            return distance - (r_i + r_j)
        return cons

    def constraint_boundary(i):
        def cons(x):
            r = x[i]
            x_c, y_c = centers[i]
            return (x_c - r) - 1e-8, (x_c + r) - 1, (y_c - r) - 1e-8, (y_c + r) - 1

    # Optimization
    bounds = [(0, 0.1) for _ in range(n)]
    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({"type": "ineq", "fun": constraint_overlap(i, j)})
        cons += [constraint_boundary(i)]

    result = minimize(objective, radii, bounds=bounds, constraints=cons)
    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
