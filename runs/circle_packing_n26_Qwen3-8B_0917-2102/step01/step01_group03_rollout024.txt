import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial hexagonal grid with 5 rows (26 circles)
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    
    # Create hexagonal grid with 5 rows
    for i in range(5):
        for j in range(5 + i):
            x = j + 0.5 * i
            y = i * np.sqrt(3) / 2
            centers[i * 6 + j] = [x, y]
    
    # Adjust for unit square and initial radius based on spacing
    max_dist = np.min(np.sqrt(np.sum((centers[1] - centers[0])**2)))
    initial_radius = max_dist / 2
    radii[:] = initial_radius
    
    # Define optimization variables and constraints
    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii
    
    def constraints(radii):
        # Boundary constraints
        for i in range(num_circles):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return False
        # Circle-circle constraints
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True
    
    # Optimization
    bounds = [(0, 1) for _ in range(num_circles)]
    result = minimize(objective, radii, bounds=bounds, constraints={"type": "callback", "fun": constraints})
    
    # Return the result
    return (centers, result.x, np.sum(result.x))
