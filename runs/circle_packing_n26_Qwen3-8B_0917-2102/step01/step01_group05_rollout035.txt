import numpy as np
import math

def run_packing():
    # Hexagonal packing with 25 small circles and 1 large circle in the center
    # Small circles radius: r = 0.13
    # Large circle radius: R = 0.5 - 25 * r * math.sqrt(3) / 2

    r_small = 0.13
    r_large = 0.5 - 25 * r_small * math.sqrt(3) / 2

    centers = []
    radii = []

    # Place 25 small circles in hexagonal grid
    for i in range(5):
        for j in range(5):
            x = 0.5 + (i - 2) * (2 * r_small * math.sqrt(3))
            y = 0.5 + (j - 2) * (2 * r_small)
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
                radii.append(r_small)

    # Place the large circle in the center
    centers.append([0.5, 0.5])
    radii.append(r_large)

    centers = np.array(centers)
    radii = np.array(radii)

    # Validate packing
    assert np.all(np.isfinite(centers)), "NaN in centers"
    assert np.all(np.isfinite(radii)), "NaN in radii"
    assert np.all(radii >= 0), "Negative radius"

    # Calculate sum of radii
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
