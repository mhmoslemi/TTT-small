import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.cluster.hierarchy import linkage, fcluster

def run_packing():

    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)

    def objective(radii):
        return -np.sum(radii)

    def constraint1(radii):
        return np.sum(radii) + 1e-12

    def constraint2(radii):
        dist_matrix = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return 1
        return 0

    def update_centers(radii):
        dist_matrix = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    centers[i] = [0.5, 0.5]
                    centers[j] = [0.5, 0.5]
                    radii[i] = max(0, (dist_matrix[i, j] - radii[j]) * 0.5)
                    radii[j] = max(0, (dist_matrix[i, j] - radii[i]) * 0.5)
        return centers, radii

    for _ in range(100):
        radii = np.clip(radii, 0, 1)
        centers, radii = update_centers(radii)
        result = minimize(objective, radii, constraints={"type": "ineq", "fun": constraint2})
        radii = result.x
        radii = np.clip(radii, 0, 1)

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:

    dist_matrix = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                radii[i] = max(0, (dist_matrix[i, j] - radii[j]) * 0.5)
                radii[j] = max(0, (dist_matrix[i, j] - radii[i]) * 0.5)

    return centers, radii, np.sum(radii)
