import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    # Initial hexagonal packing with 5 rows (adjustable)
    n = 26
    rows = 5
    cols = 5
    spacing = 1.0 / (rows + 1)
    
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * spacing
            centers[i * cols + j] = [x, y]
    
    # Adjust to fit in [0,1]^2 and add some randomness to explore better configuration
    centers += np.random.uniform(-0.05, 0.05, centers.shape)
    
    # Initial guess for radii: uniform
    radii = np.full(n, 0.04)
    
    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum
    
    # Constraints: non-overlapping and within square
    def constraint(radii, centers):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return False
        return True
    
    # Optimization
    bounds = [(0, 0.1) for _ in range(n)]
    result = minimize(objective, radii, bounds=bounds, constraints={"type": "ineq", "fun": lambda r: constraint(r, centers)})
    
    # Correct the radii based on the result
    radii = result.x
    
    # Final validation and adjustment
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = max(0, min(r, 1 - max(x, y)))
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
