import numpy as np
from scipy.optimize import minimize

def run_packing():
    def objective(r):
        return -np.sum(r)  # Maximize sum of radii

    def constraints(r, centers):
        n = len(r)
        cons = []
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] * 2 - 1e-12})
        for i in range(n):
            for j in range(i + 1, n):
                cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: np.sqrt(np.sum((centers[i] - centers[j])**2)) - r[i] - r[j] - 1e-12})
        return cons

    def bounds():
        return [(0, 1) for _ in range(26)]

    # Initial guess with hexagonal packing
    centers = []
    r = []
    for i in range(26):
        row = i // 5
        col = i % 5
        x = col / 4.5
        y = row / 4.5
        centers.append([x, y])
        r.append(0.1)

    centers = np.array(centers)
    radii = np.array(r)

    result = minimize(objective, radii, bounds=bounds(), constraints=constraints(radii, centers), method='SLSQP')

    if not result.success:
        raise RuntimeError("Optimization failed")

    return centers, result.x, np.sum(result.x)
