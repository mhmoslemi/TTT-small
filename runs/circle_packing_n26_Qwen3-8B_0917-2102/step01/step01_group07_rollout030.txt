import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    # Hexagonal packing with 5 rows (1, 2, 3, 4, 5 circles)
    # We will use a staggered hexagonal arrangement with optimized radii
    
    # Initial guess for circle centers
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # First row (top)
    for i in range(5):
        centers[i, 0] = i / 4
        centers[i, 1] = 0.1
    
    # Second row (offset)
    for i in range(5, 10):
        centers[i, 0] = (i - 5) / 4 + 0.25
        centers[i, 1] = 0.2
    
    # Third row
    for i in range(10, 15):
        centers[i, 0] = (i - 10) / 4
        centers[i, 1] = 0.3
    
    # Fourth row (offset)
    for i in range(15, 20):
        centers[i, 0] = (i - 15) / 4 + 0.25
        centers[i, 1] = 0.4
    
    # Fifth row
    for i in range(20, 26):
        centers[i, 0] = (i - 20) / 4
        centers[i, 1] = 0.5
    
    # Initial guess for radii: same for all
    initial_radius = 0.1
    radii[:] = initial_radius
    
    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)
    
    # Constraints:
    # 1. Each circle is within the unit square
    # 2. Non-overlapping between circles
    # 3. Non-negative radii
    
    def constraint1(radii):
        return np.sum(radii)
    
    def constraint2(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        return np.inf
    
    # Optimization bounds
    bounds = [(0, 1) for _ in range(n)]
    
    # Initial guess
    x0 = radii
    
    # Optimization
    result = minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2}
        ],
        method='SLSQP'
    )
    
    radii = result.x
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
