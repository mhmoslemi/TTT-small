import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance as sp_dist
from scipy.optimize import Bounds

def run_packing():
    # Initial hexagonal packing with 26 circles
    n = 26
    col = int(np.sqrt(n))
    row = n // col + (1 if n % col else 0)
    
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.05)
    
    for i in range(n):
        col_idx = i % col
        row_idx = i // col
        x = col_idx / (col - 1) if col > 1 else 0.5
        y = row_idx / (row - 1) if row > 1 else 0.5
        centers[i] = np.array([x, y])
    
    # Initial guess for radii
    radii = np.full(n, 0.05)
    
    # Objective function to maximize total radius sum
    def objective(r):
        return -np.sum(r)  # Negative for minimization
    
    # Constraint: circle must fit in unit square
    def constraint1(r, centers):
        return np.array([r[i] <= 1 - centers[i, 0] for i in range(n)]) + np.array([r[i] <= centers[i, 0] for i in range(n)])
    
    # Constraint: circles must not overlap
    def constraint2(r, centers):
        dist = sp_dist.cdist(centers, centers)
        return np.array([dist[i, j] >= r[i] + r[j] - 1e-8 for i in range(n) for j in range(i + 1, n)])
    
    # Optimization
    cons1 = {'type': 'ineq', 'fun': lambda r: constraint1(r, centers)}
    cons2 = {'type': 'ineq', 'fun': lambda r: constraint2(r, centers)}
    bounds = Bounds(np.zeros(n), np.inf)
    
    result = minimize(objective, radii, bounds=bounds, constraints=[cons1, cons2], method='SLSQP')
    
    if not result.success:
        raise ValueError("Optimization failed")
    
    centers = np.array(centers)
    radii = result.x
    total = np.sum(radii)
    
    # Final validation
    def validate_packing(centers, radii):
        n = centers.shape[0]
        
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
        
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"
        
        return True, "ok"
    
    is_valid, msg = validate_packing(centers, radii)
    if not is_valid:
        raise ValueError(msg)
    
    return centers, radii, total
