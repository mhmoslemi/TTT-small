from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    centers = []
    radii = []

        row = i // 5
        col = i % 5
        x = col / 4.0 + 0.05
        y = row / 4.0 + 0.05
        r = 0.05
        centers.append([x, y])
        radii.append(r)

    radii = np.array(radii)

        return -np.sum(radii)

            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:

        return 0

    result = minimize(objective, x0, constraints=[
        {"type": "ineq", "fun": constraint_func},
