    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (row % 2)

        centers[i] = [x / (cols - 1), y / (rows - 1)]

    def compute_radii(centers):
        n = centers.shape[0]
        radii = np.ones(n) * 0.05
        for i in range(n):
            min_dist = np.inf
            for j in range(n):
                if i != j:
                    dist = np.linalg.norm(centers[i] - centers[j])
                    if dist < min_dist:
                        min_dist = dist
            radii[i] = min_dist / 2 - 1e-5
        return radii

    radii = compute_radii(centers)

    # Optimization function to maximize sum of radii
    def optimize_func(x):
        # x contains flattened [centers, radii]
        centers = x[:n * 2].reshape(n, 2)
        radii = x[n * 2:].reshape(n)
        # Compute all pairwise distances
        dists = np.zeros((n, n))
        for i in range(n):
            for j in range(n):
                if i != j:
                    dists[i, j] = np.linalg.norm(centers[i] - centers[j])
        # Check boundary constraints
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return -np.inf
        # Check circle-circle constraints
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] - 1e-6:
                    return -np.inf
        # Objective function: maximize sum of radii
        return np.sum(radii)

    # Initial guess for optimization
    initial_guess = np.concatenate([centers.flatten(), radii])

    # Bounds: centers must be in [0,1]^2, radii must be nonnegative
    bounds = []

        bounds.append((0, 1))
    for _ in range(n):

    result = optimize.minimize(lambda x: -optimize_func(x), initial_guess, bounds=bounds, method='L-BFGS-B', options={'maxiter': 1000})
