import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Initial hexagonal packing with 26 circles
    n = 26
    centers = []
    radii = []
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        # Adjust for boundary constraints
        x = np.clip(x, 0, 1)
        y = np.clip(y, 0, 1)
        centers.append([x, y])
    centers = np.array(centers)
    radii = np.full(n, 0.05)

    # Optimization function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraints: non-overlapping and within square
    def constraint(radii):
        dists = cdist(centers, centers)
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    # Initial guess
    bounds = [(0, 0.1) for _ in range(n)]

    # Run optimization
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraint},
        method="SLSQP"
    )

    return (centers, result.x, np.sum(result.x))
