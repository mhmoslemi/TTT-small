from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d
from itertools import combinations

def run_packing():
    np.random.seed(42)
    n = 26
    initial_centers = np.random.rand(n, 2)
    initial_radii = np.random.rand(n) * 0.1

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        constraints = []
        for i in range(n):
            constraints.append((radii[i] <= 1 - initial_centers[i, 0] - 1e-8))
            constraints.append((radii[i] <= 1 - initial_centers[i, 1] - 1e-8))
            constraints.append((radii[i] <= initial_centers[i, 0] + 1e-8))
            constraints.append((radii[i] <= initial_centers[i, 1] + 1e-8))
        for i in range(n):
            for j in range(i+1, n):
                dist = np.linalg.norm(initial_centers[i] - initial_centers[j])
                constraints.append((radii[i] + radii[j] <= dist - 1e-8))
        return constraints

    bounds = [(0, 1) for _ in range(n)]
    res = minimize(objective, initial_radii, method='SLSQP', bounds=bounds, constraints=constraints)

    centers = initial_centers
    radii = res.x
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
