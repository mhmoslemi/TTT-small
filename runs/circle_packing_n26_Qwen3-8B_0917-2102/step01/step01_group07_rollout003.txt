import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Define centers for a hexagonal packing with boundary circles
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Create a hexagonal lattice pattern with some circles at the corners
    # and others on the edges to maximize the use of space
    idx = 0
    
    # Bottom row (y=0)
    for i in range(5):
        x = i * 0.25
        centers[idx] = [x, 0.0]
        radii[idx] = 0.1
        idx += 1
    
    # Right edge (x=1)
    for i in range(5):
        y = i * 0.25
        centers[idx] = [1.0, y]
        radii[idx] = 0.1
        idx += 1
    
    # Top row (y=1)
    for i in range(5):
        x = i * 0.25
        centers[idx] = [x, 1.0]
        radii[idx] = 0.1
        idx += 1
    
    # Left edge (x=0)
    for i in range(5):
        y = i * 0.25
        centers[idx] = [0.0, y]
        radii[idx] = 0.1
        idx += 1
    
    # Fill in the center with a hexagonal arrangement
    for i in range(4):
        for j in range(5 - i):
            x = 0.25 + i * 0.25
            y = 0.25 + j * 0.25
            centers[idx] = [x, y]
            radii[idx] = 0.1
            idx += 1
    
    # Additional circles in the center for densification
    for i in range(2):
        x = 0.5 + i * 0.1
        y = 0.5 + i * 0.1
        centers[idx] = [x, y]
        radii[idx] = 0.1
        idx += 1
    
    # Optimization to maximize radii while ensuring non-overlap and boundary constraints
    def objective(radii):
        return -np.sum(radii)
    
    def constraints(radii):
        constraints = []
        for i in range(n):
            constraints.append((radii[i] >= 0))
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                constraints.append((radii[i] + radii[j] <= dist))
        for i in range(n):
            x, y = centers[i]
            constraints.append((radii[i] <= 1.0 - x))
            constraints.append((radii[i] <= x))
            constraints.append((radii[i] <= 1.0 - y))
            constraints.append((radii[i] <= y))
        return constraints
    
    bounds = [(0, 1) for _ in range(n)]
    
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=constraints)
    
    return (centers, result.x, np.sum(result.x))
