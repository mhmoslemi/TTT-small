import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r)

    cons = ({
        'type': 'ineq',
        'fun': constraint,
        'jac': lambda r: np.ones_like(r)
    })

    bounds = Bounds(np.zeros(n), np.full(n, 1.0))

    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)

    if not result.success:
        raise RuntimeError("Optimization failed")

    radii = result.x

    # Hexagonal packing with boundary-aware placement
    row = []
    for i in range(n):
        col = i // 5
        row_idx = i % 5
        x = row_idx / 5.0
        y = col / 5.0
        if col % 2 == 1:
            x += 0.5 / 5.0
        row.append([x, y])
    centers = np.array(row)

    # Add small perturbation to avoid overlap
    centers += np.random.normal(0, 0.05, centers.shape)

    # Ensure centers are within [0,1]^2
    centers = np.clip(centers, 0, 1)

    # Final validation and adjustment
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = np.max([0, 1 - np.max([x, 1 - x, y, 1 - y])])

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                radii[i] = (dist - 1e-12) / 2
                radii[j] = (dist - 1e-12) / 2

    return centers, radii, np.sum(radii)
