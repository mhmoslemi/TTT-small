from scipy.optimize import minimize

def run_packing():

        return -np.sum(radii)

        return np.sum(radii)

    centers = np.random.uniform(0, 1, (n, 2))
    radii = np.ones(n) * 0.1
    
    def packing_func(radii):
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
                return np.inf
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return np.inf
        return -np.sum(radii)
    
    result = minimize(lambda r: packing_func(r), radii, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n)])
    
    return (centers, result.x, -result.fun)
