from scipy.optimize import minimize, Bounds

        return np.sum(r)

    })

    bounds = Bounds(np.zeros(n), np.full(n, 1.0))

    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)

    if not result.success:
        raise RuntimeError("Optimization failed")

    radii = result.x

    # Hexagonal packing
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (row % 2)
        y = row * (np.sqrt(3)/2)
        centers[i] = [x / 5, y / 8]

    # Small perturbation
    centers += np.random.normal(0, 0.05, centers.shape)

    # Clip centers to [0,1]^2
    centers = np.clip(centers, 0, 1)

    # Adjust radii to fit boundary
    for i in range(n):
        x, y = centers[i]
        max_dist = np.max([x, 1 - x, y, 1 - y])
        radii[i] = np.min([radii[i], max_dist])

    # Non-overlapping constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                radii[i] = (dist - 1e-12) / 2
                radii[j] = (dist - 1e-12) / 2

    return centers, radii, np.sum(radii)
