def run_packing():

    # Calculate positions for small circles in a hexagonal grid

    n_rows = int(np.ceil(np.sqrt(n_small)))
    n_cols = int(np.ceil(n_small / n_rows))
    
    # Define positions for small circles in a hexagonal lattice
    small_radii = np.full(n_small, 0.04)
    small_centers = []
    
    for i in range(n_rows):
        for j in range(n_cols):
            if i % 2 == 0:
                x = j * (1 + small_radii[0])
                y = i * (np.sqrt(3) * small_radii[0])
            else:
                x = (j + 0.5) * (1 + small_radii[0])
                y = i * (np.sqrt(3) * small_radii[0])
            small_centers.append([x, y])
    
    small_centers = np.array(small_centers)
    small_radii = np.full(len(small_centers), 0.04)
    
    # Add a large circle in the center
    large_radius = 0.1
    large_center = [0.5, 0.5]
    
    # Check if large circle overlaps with small circles
    for i in range(len(small_centers)):
        dist = np.sqrt((small_centers[i, 0] - large_center[0])**2 + 
                      (small_centers[i, 1] - large_center[1])**2)
        if dist < large_radius + small_radii[i] - 1e-12:
            large_radius -= (dist - (large_radius + small_radii[i])) / 2
    
    centers = np.vstack([large_center, small_centers])
    radii = np.concatenate([np.array([large_radius]), small_radii])
    
    return centers, radii, np.sum(radii)
