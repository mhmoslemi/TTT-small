from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Initial hexagonal grid with 26 circles
    n = 26
    centers = []
    radii = []
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers.append([x, y])
    centers = np.array(centers)
    radii = np.ones(n) * 0.1

    def objective(r):
        return -np.sum(r)

    def constraints(r):
        dists = cdist(centers, centers)
        for i in range(n):

                    return False

                return False
        return True

    result = minimize(
        objective,
        radii,
        method='SLSQP',
        constraints={'type': 'constraint', 'fun': constraints, 'args': ()},
        bounds=[(0, 1) for _ in range(n)],
        options={'disp': False}
    )

    if not result.success:
        raise ValueError("Optimization failed")

    return (centers, result.x, np.sum(result.x))
