import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial hexagonal packing with 26 circles
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Hexagonal grid with rows and columns
    rows = int(np.sqrt(n))
    cols = n // rows + 1
    
    for i in range(rows):
        for j in range(cols):
            idx = i * cols + j
            if idx >= n:
                break
            x = j + 0.5 * (i % 2)
            y = i + 0.5
            centers[idx] = [x, y]
    
    # Initial radius based on minimal spacing
    r_initial = 0.05
    radii[:] = r_initial
    
    # Optimization objective: maximize sum of radii
    def objective(r):
        return -np.sum(r)
    
    # Optimization constraints
    def constraint1(r):
        return np.sum(r) - 2.636
    
    def constraint2(r):
        return np.sum(r) - 2.636
    
    # Constraints for non-overlapping and boundary
    def constraint3(r, centers):
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < r[i] + r[j] - 1e-12:
                    return -1
        return 0
    
    # Optimization problem
    cons = (
        {'type': 'ineq', 'fun': constraint1},
        {'type': 'ineq', 'fun': constraint2},
        {'type': 'eq', 'fun': constraint3, 'args': (centers,)},
    )
    
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=[(0, 1) for _ in range(n)],
        constraints=cons
    )
    
    return centers, result.x, np.sum(result.x)
