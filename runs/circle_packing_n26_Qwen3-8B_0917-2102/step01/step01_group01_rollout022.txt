import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.cluster.hierarchy import linkage, fcluster
from itertools import combinations

def run_packing():
    n = 26
    centers = []
    radii = []

    # Hexagonal grid with staggered rows and varying radii
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col / 4.5 + 0.05
        y = row / 4.5 + 0.05
        r = 0.07
        centers.append([x, y])
        radii.append(r)

    def objective(radii):
        return -np.sum(radii)

    def constraint(x):
        centers_flat = x.reshape(-1, 2)
        dist_matrix = squareform(pdist(centers_flat))
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return 1
        return 0

    def boundary_constraint(x):
        centers_flat = x.reshape(-1, 2)
        for i in range(len(radii)):
            x, y = centers_flat[i]
            r = radii[i]
            if not (0 - r <= x <= 1 + r and 0 - r <= y <= 1 + r):
                return 1
        return 0

    x0 = np.array(centers).reshape(-1)
    result = minimize(objective, x0, constraints={"type": "ineq", "fun": constraint, "method": "SLSQP"},
                      bounds=[(0, 1) for _ in range(2 * n)], options={"maxiter": 500})
    centers_opt = result.x.reshape(-1, 2)
    radii_opt = np.array(radii)

    # Refinement to fit boundary
    for i in range(n):
        x, y = centers_opt[i]
        r = radii_opt[i]
        if x - r < 0:
            radii_opt[i] = max(0, x)
        elif x + r > 1:
            radii_opt[i] = max(0, 1 - x)
        if y - r < 0:
            radii_opt[i] = max(0, y)
        elif y + r > 1:
            radii_opt[i] = max(0, 1 - y)

    # Final refinement to ensure non-overlapping
    dist_matrix = squareform(pdist(centers_opt))
    for i in range(n):
        for j in range(i + 1, n):
            if dist_matrix[i, j] < radii_opt[i] + radii_opt[j] - 1e-8:
                radii_opt[i] = (dist_matrix[i, j] - radii_opt[j]) * 0.5
                radii_opt[j] = (dist_matrix[i, j] - radii_opt[i]) * 0.5

    radii_opt = np.clip(radii_opt, 0, 1)
    return np.array(centers_opt), radii_opt, np.sum(radii_opt)
