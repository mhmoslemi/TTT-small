import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initialize centers in hexagonal packing
    centers = []
    for i in range(n):
        row = i // 2
        col = i % 2
        x = 0.5 + (col - 0.5) * (1 / np.sqrt(3))
        y = row / np.sqrt(3)
        centers.append([x, y])
    centers = np.array(centers)

    # Adjust centers to fit in [0,1]^2
    scale = 1 / np.max(centers, axis=0)
    centers = centers / scale

    # Initial radii based on minimal spacing
    radii = np.full(n, 0.05)

    # Objective function to maximize sum of radii
    def objective(r):
        return -np.sum(r)

    # Constraints: non-overlapping and within square
    def constraint_func(r):
        dists = squareform(pdist(centers))
        return np.array([dists[i][j] - r[i] - r[j] for i in range(n) for j in range(i+1)]) - 1e-8

    cons = {'type': 'ineq', 'fun': constraint_func}

    # Bounds for radii
    bounds = Bounds(np.zeros(n), np.full(n, 1.0))

    # Optimization
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)

    if not result.success:
        raise RuntimeError("Optimization failed")

    radii = result.x

    # Final validation and boundary adjustment
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = np.max([0, 1 - np.max([x, 1 - x, y, 1 - y])])

    return centers, radii, np.sum(radii)
