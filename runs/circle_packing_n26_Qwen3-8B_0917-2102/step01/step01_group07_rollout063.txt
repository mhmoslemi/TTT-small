import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Initial hexagonal packing with 25 circles
    n = 26
    centers = []
    radii = []
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 1 / (2 * row + 1))
        y = row + 0.5 / (2 * row + 1)
        centers.append([x, y])
    centers = np.array(centers)
    radii = np.array([0.05] * n)
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraints(radii):
        constraints = []
        for i in range(n):
            constraints.append((radii[i] >= 0))
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                constraints.append((radii[i] + radii[j] <= dist))
        for i in range(n):
            x, y = centers[i]
            constraints.append((radii[i] <= x))
            constraints.append((radii[i] <= 1 - x))
            constraints.append((radii[i] <= y))
            constraints.append((radii[i] <= 1 - y))
        return constraints
    
    def optimize():
        bounds = [(0, 1)] * n
        res = minimize(objective, radii, bounds=bounds, constraints=constraints)
        return res.x
    
    radii = optimize()
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
