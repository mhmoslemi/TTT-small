import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal packing
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 2
        col = i % 2
        x = (row + 0.5 * col) / (np.sqrt(3) * (n ** 0.5))
        y = (row + 0.5 * col) / (n ** 0.5)
        centers[i] = [x, y]
    centers += np.random.normal(0, 0.01, centers.shape)
    centers = np.clip(centers, 0, 1)

    radii = np.full(n, 0.05)

    def objective(r):
        return -np.sum(r)

    def constraint_func(r):
        return np.sum(r)

    cons = ({
        'type': 'ineq',
        'fun': constraint_func,
        'jac': lambda r: np.ones(n)
    })

    bounds = Bounds(np.zeros(n), np.full(n, 1.0))

    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)

    if not result.success:
        raise RuntimeError("Optimization failed")

    radii = result.x

    # Boundary adjustment
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            max_dist = np.max([x, 1 - x, y, 1 - y])
            radii[i] = np.max([0, np.min([radii[i], max_dist - 1e-12])])

    # Non-overlapping constraint enforcement
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                radii[i] = (dist - 1e-12) / 2
                radii[j] = (dist - 1e-12) / 2

    return centers, radii, np.sum(radii)
