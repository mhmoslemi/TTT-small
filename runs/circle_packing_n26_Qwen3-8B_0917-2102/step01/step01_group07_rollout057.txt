import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from itertools import combinations

def run_packing():
    # Initial hexagonal packing with 5 rows
    centers = []
    radii = []
    for row in range(5):
        for col in range(13):
            x = col + 0.5 * row / 2
            y = row + 0.5 * row
            if row % 2 == 0:
                x = col + 0.5 * row / 2
            else:
                x = col + 0.5 * (row + 1) / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
                radii.append(0.05)
    
    centers = np.array(centers)
    radii = np.array(radii)
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraint_overlap(radii):
        n = len(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0
    
    cons = {'type': 'ineq', 'fun': constraint_overlap}
    
    result = minimize(objective, radii, method='SLSQP', constraints=cons)
    
    # Final check and return
    final_radii = result.x
    final_centers = centers
    
    # Ensure circles are within bounds
    for i in range(len(final_centers)):
        x, y = final_centers[i]
        r = final_radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            final_radii[i] = 0
    
    return final_centers, final_radii, np.sum(final_radii)
