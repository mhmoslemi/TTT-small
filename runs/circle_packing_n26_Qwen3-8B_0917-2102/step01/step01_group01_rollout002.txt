import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.cluster.hierarchy import linkage, fcluster

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraint(x):
        centers = x.reshape(-1, 2)
        dist = squareform(pdist(centers))
        for i in range(26):
            for j in range(i + 1, 26):
                if dist[i, j] < radii[i] + radii[j] - 1e-8:
                    return 1
        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return 1
        return 0

    def run_hexagonal_packing():
        n = 26
        centers = []
        radii = []
        for i in range(n):
            row = i // 5
            col = i % 5
            x = col / 4.0 + 0.05
            y = row / 4.0 + 0.05
            r = 0.05
            centers.append([x, y])
            radii.append(r)
        return np.array(centers), np.array(radii)

    def validate_and_refine(centers, radii):
        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                radii[i] = max(0, min(r, 1 - max(x, 1 - x, y, 1 - y)))
        dist_matrix = squareform(pdist(centers))
        for i in range(26):
            for j in range(i + 1, 26):
                d = dist_matrix[i, j]
                if d < radii[i] + radii[j] - 1e-8:
                    radii[i] = max(0, (d - radii[j]) * 0.5)
                    radii[j] = max(0, (d - radii[i]) * 0.5)
        return centers, radii

    centers, radii = run_hexagonal_packing()
    x0 = centers.reshape(-1)
    result = minimize(objective, x0, constraints={"type": "ineq", "fun": constraint}, method="SLSQP", options={"maxiter": 200, "ftol": 1e-10})

    radii = np.clip(radii, 0, 1)
    centers, radii = validate_and_refine(centers, radii)
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
