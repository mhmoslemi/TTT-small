import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    def objective(r):
        return -np.sum(r)  # Maximize sum of radii

    def constraints(r, centers):
        n = len(r)
        cons = []
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - 1e-12})
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: dist - r[i] - r[j] - 1e-12})
        for i in range(n):
            x, y = centers[i]
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - x + 1e-12})
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: x + r[i] - 1e-12})
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - y + 1e-12})
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: y + r[i] - 1e-12})
        return cons

    n = 26
    # Hexagonal packing: 1st row 1 circle, then 2, 3, ..., 7, 6, 5, 4, 3, 2, 1
    rows = []
    for i in range(1, 8):
        rows.append(i)
    for i in range(6, 0, -1):
        rows.append(i)
    n_rows = len(rows)
    centers = []
    for i in range(n_rows):
        row = []
        for j in range(rows[i]):
            x = j * (1 / (rows[i] - 0.5)) + 0.5 / (rows[i] - 0.5)
            y = i * (1 / (n_rows - 0.5)) + 0.5 / (n_rows - 0.5)
            row.append([x, y])
        centers.extend(row)
    centers = np.array(centers)
    r0 = np.ones(n) * 0.05

    res = minimize(objective, r0, method='SLSQP', constraints=constraints(r0, centers))

    centers = centers[:n]
    radii = res.x
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
