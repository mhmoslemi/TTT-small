from scipy.optimize import LinearConstraint
from scipy.spatial.distance import pdist, squareform

def run_packing():

            x = 1 - 1e-6

            y = 1 - 1e-6
        centers.append([x, y])

    initial_radii = np.full(n, 0.05)

            constraints.append(y + r[i] <= 1 + 1e-12)
        for i in range(n):

                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
