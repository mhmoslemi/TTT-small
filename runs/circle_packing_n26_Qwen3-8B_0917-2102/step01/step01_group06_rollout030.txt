import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    np.random.seed(42)
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n)

    def obj_func(x):
        centers[:] = x[:2*n].reshape(n, 2)
        radii[:] = x[2*n:]
        sum_r = np.sum(radii)
        return -sum_r

    def constraints(x):
        centers[:] = x[:2*n].reshape(n, 2)
        radii[:] = x[2*n:]
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return -np.inf
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        return np.sum(radii)

    x0 = np.random.rand(2*n + n)
    result = minimize(obj_func, x0, method='L-BFGS-B', bounds=[(0, 1) for _ in range(2*n)] + [(0, 1) for _ in range(n)])
    centers = result.x[:2*n].reshape(n, 2)
    radii = result.x[2*n:]
    sum_r = np.sum(radii)
    return centers, radii, sum_r
