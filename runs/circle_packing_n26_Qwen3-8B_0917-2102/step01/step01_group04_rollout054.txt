import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.cluster.hierarchy import linkage, fcluster
from itertools import combinations

def run_packing():
    # Initial hexagonal packing with 26 circles
    n = 26
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    
    # Hexagonal grid with spacing
    spacing = 0.3
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col * spacing + (spacing / 2) * (1 if row % 2 else 0)
        y = row * spacing * np.sqrt(3) / 2
        centers[i] = [x, y]
    
    # Refine with optimization
    def objective(radii):
        return -np.sum(radii)
    
    def constraints(radii):
        constraints = []
        for i in range(n):
            constraints.append((radii[i], 0, np.inf))
        for i, j in combinations(range(n), 2):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            constraints.append((radii[i] + radii[j], 0, dist))
        for i in range(n):
            x, y = centers[i]
            constraints.append((radii[i], 0, 1 - x))
            constraints.append((radii[i], 0, x))
            constraints.append((radii[i], 0, 1 - y))
            constraints.append((radii[i], 0, y))
        return constraints
    
    bounds = [(0, np.inf) for _ in range(n)]
    result = minimize(objective, radii, bounds=bounds, constraints=constraints)
    
    return centers, result.x, np.sum(result.x)
