import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal packing with boundary-aligned circles
    centers = []
    for i in range(n):
        row = i // 6
        col = i % 6
        x = col / 6.0 + 0.5 / 6.0
        y = row / 6.0 + 0.5 / 6.0
        if row % 2 == 1:
            y += 1 / np.sqrt(3) / 6.0
        centers.append([x, y])
    centers = np.array(centers)
    
    # Initial radii estimate based on hexagonal packing
    radii = np.full(n, 1 / np.sqrt(3) / 6.0)
    
    # Objective: maximize sum of radii
    def objective(r):
        return -np.sum(r)
    
    # Constraints: non-overlapping and boundary
    def constraint_func(r):
        dist_matrix = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        for i in range(n):
            if centers[i, 0] - r[i] < -1e-8 or centers[i, 0] + r[i] > 1 + 1e-8:
                return -np.inf
            if centers[i, 1] - r[i] < -1e-8 or centers[i, 1] + r[i] > 1 + 1e-8:
                return -np.inf
        return np.sum(r)
    
    # Optimization with bounds
    bounds = Bounds(np.zeros(n), np.full(n, 1.0))
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraint_func})
    
    if not result.success:
