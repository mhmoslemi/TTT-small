import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize

def run_packing():
    # Hexagonal packing with 5 rows (26 circles)
    # Start with a base hexagonal grid
    def hex_grid(n):
        rows = int(np.ceil(n / 6))
        cols = int(np.ceil(n / rows))
        centers = []
        for i in range(rows):
            for j in range(cols):
                x = j + i / 2
                y = i * np.sqrt(3) / 2
                if i % 2 == 1:
                    x += 0.5
                centers.append((x, y))
        return np.array(centers)
    
    initial_centers = hex_grid(26)
    initial_radii = np.ones(26) * 0.1
    
    # Optimization objective: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum
    
    # Constraints: non-overlapping and within square
    def constraint(radii):
        for i in range(26):
            for j in range(i + 1, 26):
                dx = initial_centers[i, 0] - initial_centers[j, 0]
                dy = initial_centers[i, 1] - initial_centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0
    
    # Optimization
    bounds = [(0, 1) for _ in range(26)]
    cons = [{'type': 'ineq', 'fun': constraint}]
    result = minimize(objective, initial_radii, bounds=bounds, constraints=cons)
    
    final_centers = initial_centers
    final_radii = result.x
    final_sum = np.sum(final_radii)
    
    return final_centers, final_radii, final_sum
