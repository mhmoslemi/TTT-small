import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    np.random.seed(42)
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraints(radii):
        constraints = []
        for i in range(26):
            constraints.append((radii[i] >= 0))
        return constraints
    
    def pack_hexagonal(n):
        rows = int(np.ceil(n / 2))
        cols = int(np.ceil(n / rows))
        centers = []
        for i in range(rows):
            for j in range(cols):
                x = j + 0.5 * (i % 2)
                y = i + 0.5
                if len(centers) < n:
                    centers.append(np.array([x, y]))
        return np.array(centers)
    
    initial_centers = pack_hexagonal(26)
    initial_radii = np.full(26, 0.05)
    
    def optimize():
        bounds = [(0, 1)] * 26
        result = minimize(
            objective,
            initial_radii,
            bounds=bounds,
            constraints=[
                {'type': 'ineq', 'fun': lambda r: 1 - cdist(initial_centers, initial_centers, 'euclidean').min(axis=1) + np.sum(r) / 26}
            ],
            method='SLSQP'
        )
        return result.x
    
    radii = optimize()
    centers = initial_centers.copy()
    
    for i in range(26):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            centers[i, 0] = r
        if x + r > 1:
            centers[i, 0] = 1 - r
        if y - r < 0:
            centers[i, 1] = r
        if y + r > 1:
            centers[i, 1] = 1 - r
    
    return centers, radii, np.sum(radii)
