import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial hexagonal packing with 26 circles
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Hexagonal packing parameters
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    spacing = 1.0 / (cols - 1)
    offset = 0.5 / (rows - 1)
    
    for i in range(n):
        r = 0.1  # Initial radius
        row = i // cols
        col = i % cols
        x = col * spacing
        y = row * offset
        centers[i] = [x, y]
        radii[i] = r

    # Objective function to maximize sum of radii
    def obj_func(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Constraints: circles must not overlap and stay within the unit square
    def constraints(radii, centers):
        constraints = []
        for i in range(n):
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - 1e-12})
        for i in range(n):
            for j in range(i + 1, n):
                def dist_func(r, i=i, j=j):
                    dx = centers[i][0] - centers[j][0]
                    dy = centers[i][1] - centers[j][1]
                    dist = np.sqrt(dx*dx + dy*dy)
                    return dist - (r[i] + r[j]) + 1e-12
                constraints.append({'type': 'ineq', 'fun': dist_func})
        return constraints

    # Optimization
    result = minimize(obj_func, radii, method='SLSQP', constraints=constraints(radii, centers))
    radii = result.x
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
