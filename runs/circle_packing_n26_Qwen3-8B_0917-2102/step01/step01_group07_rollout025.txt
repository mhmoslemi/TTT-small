import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    def objective(r):
        return -np.sum(r)
    
    def constraints(r, centers):
        n = len(r)
        cons = []
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - 1e-8})
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j, dist=dist: dist - r[i] - r[j] - 1e-8})
        return cons
    
    def boundary_constraints(r, centers):
        n = len(r)
        cons = []
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i, c=centers[i]: r[i] - (1 - c[0]) - 1e-8})
            cons.append({'type': 'ineq', 'fun': lambda r, i=i, c=centers[i]: r[i] - (1 - c[1]) - 1e8})
            cons.append({'type': 'ineq', 'fun': lambda r, i=i, c=centers[i]: r[i] - (c[0]) - 1e-8})
            cons.append({'type': 'ineq', 'fun': lambda r, i=i, c=centers[i]: r[i] - (c[1]) - 1e-8})
        return cons
    
    # Hexagonal grid in unit square
    grid = []
    r = []
    for row in range(5):
        for col in range(5):
            x = col / 5.0
            y = row / 5.0 + (row % 2) * 0.5 / 5.0
            grid.append([x, y])
            r.append(0.05)
    
    centers = np.array(grid)
    radii = np.array(r)
    
    bounds = [(0, 1) for _ in range(26)]
    
    res = minimize(objective, radii, method='SLSQP', bounds=bounds,
                   constraints=constraints(radii, centers) + boundary_constraints(radii, centers))
    
    radii = res.x
    centers = np.array(grid)
    
    return (centers, radii, np.sum(radii))
