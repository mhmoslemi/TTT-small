from scipy.optimize import minimize

    n = 26
    centers = []
    radii = []

        r = 0.1
        centers.append([x, 1 - r])
        radii.append(r)

            for i in range(4):
                x = i / 4 + 0.2
                y = 0.5 - row * (np.sqrt(3)/2 * 0.1)
                r = 0.1
                centers.append([x, y])
                radii.append(r)

    # Final adjustment to maximize sum of radii
    def objective(radii_param):
        return -np.sum(radii_param)

    def constraint_func(radii_param):
        centers_param = []
        for i in range(n):
            x = centers[i][0]
            y = centers[i][1]
            r = radii_param[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:

                dx = centers[i][0] - centers[j][0]
