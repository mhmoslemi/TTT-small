import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.cluster.vq import kmeans

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii, centers):
        n = len(radii)
        cons = []
        for i in range(n):
            cons.append(radii[i] >= 0)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                cons.append(radii[i] + radii[j] <= dist + 1e-8)
        for i in range(n):
            x, y = centers[i]
            cons.append(radii[i] <= x)
            cons.append(radii[i] <= 1 - x)
            cons.append(radii[i] <= y)
            cons.append(radii[i] <= 1 - y)
        return cons

    def initial_centers():
        rows = 5
        cols = 5
        spacing = 1 / (rows + 1)
        centers = []
        for r in range(rows):
            for c in range(cols):
                x = c * spacing + spacing / 2
                y = r * spacing + spacing / 2
                centers.append([x, y])
        return np.array(centers)

    centers = initial_centers()
    n = len(centers)
    radii_initial = np.full(n, 0.1)

    bounds = [(0, 1) for _ in range(2 * n)]
    result = minimize(objective, radii_initial, bounds=bounds, constraints=constraints)

    return (centers, result.x, np.sum(result.x))
