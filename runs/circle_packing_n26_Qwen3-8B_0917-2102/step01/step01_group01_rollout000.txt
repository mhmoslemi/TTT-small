import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    n = 26
    centers = []
    radii = []

    # Initial hexagonal grid with staggered rows
    for i in range(n):
        row = i // 4
        col = i % 4
        x = col / 4.0 + 0.2
        y = row / 4.0 + 0.2
        r = 0.05
        centers.append([x, y])
        radii.append(r)

    centers = np.array(centers)
    radii = np.array(radii)

    def objective(radii):
        return -np.sum(radii)

    def constraint(x):
        centers_flat = x.reshape(-1, 2)
        dist_matrix = squareform(pdist(centers_flat))
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return 1
        return 0

    def boundary_constraint(x):
        centers_flat = x.reshape(-1, 2)
        for i in range(len(radii)):
            x, y = centers_flat[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return 1
        return 0

    bounds = [(0, 1) for _ in range(2 * n)]

    x0 = centers.reshape(-1)
    result = minimize(objective, x0, method='SLSQP', constraints=[
        {'type': 'ineq', 'fun': constraint},
        {'type': 'ineq', 'fun': boundary_constraint}
    ])

    centers = result.x.reshape(-1, 2)
    radii = np.clip(radii, 0, 1)

    # Post-refinement to adjust radii based on distances
    dist_matrix = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            d = dist_matrix[i, j]
            if d < radii[i] + radii[j] - 1e-8:
                radii[i] = max(0, (d - radii[j]) * 0.5)
                radii[j] = max(0, (d - radii[i]) * 0.5)

    # Ensure radii do not cause overflow
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            radii[i] = max(0, 1 - max(x, 1 - x, y, 1 - y))

    return centers, radii, np.sum(radii)
