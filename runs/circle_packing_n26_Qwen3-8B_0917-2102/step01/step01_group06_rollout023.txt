import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance
from scipy.optimize import Bounds

def run_packing():
    np.random.seed(42)
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.1)

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii, centers):
        cons = []
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - 1e-12})
        for i in range(n):
            for j in range(i + 1, n):
                def dist_constraint(r, i=i, j=j):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    return r[i] + r[j] - np.sqrt(dx**2 + dy**2) - 1e-12
                cons.append({'type': 'ineq', 'fun': dist_constraint})
        for i in range(n):
            def boundary_constraint(r, i=i):
                x, y = centers[i]
                return r[i] - (1 - np.maximum(x, 1 - x) - np.maximum(y, 1 - y)) - 1e-12
            cons.append({'type': 'ineq', 'fun': boundary_constraint})
        return cons

    bounds = Bounds(np.zeros(n), np.inf)

    result = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints(radii, centers),
        options={'ftol': 1e-12, 'maxiter': 1000}
    )

    if not result.success:
        raise RuntimeError("Optimization failed")

    return centers, result.x, -result.fun
