from scipy.optimize import minimize

                x = j + 0.5 * i
                y = i + 0.5 * (1 - i)  # Slight adjustment to fit within square

    # Constraints: non-overlapping and within square
    def constraint(radii, centers):

        for i in range(len(r)):
            for j in range(i + 1, len(r)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]

    # Set up optimization
    bounds = [(0, 1) for _ in range(26)]
    cons = ({'type': 'ineq', 'fun': lambda r: -constraint(r, centers), 'args': (centers,)})

    result = minimize(objective, radii, bounds=bounds, constraints=cons, method='SLSQP')
