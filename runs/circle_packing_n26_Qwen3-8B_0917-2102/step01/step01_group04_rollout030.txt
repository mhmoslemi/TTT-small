from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)

    def objective(r):
        return -np.sum(r)

    def constraints(r):
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < r[i] + r[j] - 1e-8:
                    return False
        for i in range(n):
            x, y = centers[i]
            if x - r[i] < -1e-8 or x + r[i] > 1 + 1e-8 or y - r[i] < -1e-8 or y + r[i] > 1 + 1e-8:
                return False
        return True

    def pack():
        bounds = [(0, 1) for _ in range(n)]
        result = minimize(lambda r: -np.sum(r), x0=radii, bounds=bounds, constraints=[{"type": "ineq", "fun": constraints}])
        return result.x

    radii = pack()
    centers = np.array([np.random.rand(2) for _ in range(n)])
    for i in range(n):
        x, y = centers[i]
        radii[i] = min(1 - max(x, 1 - x), 1 - max(y, 1 - y)) / 2
    return centers, radii, np.sum(radii)
