        centers[i] = [x, y]

        r_boundary = boundary_dist
        r_overall = np.minimum(r, r_boundary)
        return r_overall

        dist = pdist(centers)
        return np.sum(r[np.newaxis, :] + r[:, np.newaxis] - dist - 1e-12)

    bounds = Bounds(np.zeros(n), np.full(n, 1.0))

    radii_initial = np.full(n, 0.05)
