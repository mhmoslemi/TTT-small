import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate initial hexagonal grid with boundary-aligned rows
    rows = []
    for i in range(n):
        if i % 2 == 0:
            x = i / (np.sqrt(3) * 1)
            y = (i // 2) * (2 / np.sqrt(3))
        else:
            x = (i - 1) / (np.sqrt(3) * 1) + 0.5 / np.sqrt(3)
            y = (i // 2) * (2 / np.sqrt(3))
        rows.append([x, y])
    centers = np.array(rows)
    centers = np.clip(centers, 0, 1)
    
    # Initial radii based on minimum distance to boundary
    radii = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        radii[i] = np.min([x, 1 - x, y, 1 - y])
    
    # Optimization objective: maximize sum of radii
    def objective(r):
        return -np.sum(r)
    
    # Constraints: non-overlapping circles
    def constraint_func(r):
        dists = squareform(pdist(centers))
        return np.array([dists[i][j] - r[i] - r[j] for i in range(n) for j in range(i+1, n)])
    
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    # Bounds for radii
    bounds = Bounds(np.zeros(n), np.full(n, 1.0))
    
    # Optimization
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)
    
    if not result.success:
        raise RuntimeError("Optimization failed")
    
    radii = result.x
    
    # Final validation and adjustment
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = np.max([0, 1 - np.max([x, 1 - x, y, 1 - y])])
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                radii[i] = (dist - 1e-12) / 2
                radii[j] = (dist - 1e-12) / 2
    
    return centers, radii, np.sum(radii)
