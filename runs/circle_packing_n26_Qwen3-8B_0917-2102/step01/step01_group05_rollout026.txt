import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    # Initial guess for centers using hexagonal packing
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    r = np.full(num_circles, 0.01)

    # Initial hexagonal grid layout
    for i in range(num_circles):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        centers[i] = [x / 5, y / 5]

    def objective(radii):
        return -np.sum(radii)

    def constraint1(radii):
        for i in range(num_circles):
            x, y = centers[i]
            if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
                return -np.inf
        return 0

    def constraint2(radii):
        dist = squareform(pdist(centers))
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                if dist[i, j] < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        return 0

    bounds = Bounds([0.0] * num_circles, [1.0] * num_circles)

    result = minimize(
        objective,
        r,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2}
        ],
        method='SLSQP'
    )

    if not result.success:
        raise ValueError("Optimization failed")

    return centers, result.x, np.sum(result.x)
