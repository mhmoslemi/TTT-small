<strategy>Use a hexagonal packing with optimized circle sizes near boundaries to maximize total radius sum.</strategy>

import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    np.random.seed(42)
    n = 26

    # Initial guess: hexagonal packing with spacing
    centers = np.zeros((n, 2))
    r = np.ones(n) * 0.1
    for i in range(n):
        row = i // 5
        col = i % 5
        centers[i, 0] = col * 0.3 + 0.15
        centers[i, 1] = row * 0.3 + 0.15

    # Objective function: maximize sum of radii
    def obj(x):
        r_new = x[:n]
        return -np.sum(r_new)

    # Constraints: non-overlapping and within square
    def cons(x):
        r_new = x[:n]
        centers_new = np.reshape(x[n:], (n, 2))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_new[i, 0] - centers_new[j, 0]
                dy = centers_new[i, 1] - centers_new[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < r_new[i] + r_new[j] - 1e-8:
                    return -1
        for i in range(n):
            x, y = centers_new[i]
            if not (0 <= x - r_new[i] and x + r_new[i] <= 1 and 0 <= y - r_new[i] and y + r_new[i] <= 1):
                return -1
        return 0

    # Optimization
    x0 = np.concatenate((r, centers.reshape(-1)))
    cons_dict = {'type': 'ineq', 'fun': cons}
    res = minimize(obj, x0, constraints=[cons_dict], method='SLSQP', options={'ftol': 1e-10, 'maxiter': 1000})
    x_opt = res.x
    r_opt = x_opt[:n]
    centers_opt = x_opt[n:].reshape(n, 2)
    sum_radii = np.sum(r_opt)
    return centers_opt, r_opt, sum_radii
