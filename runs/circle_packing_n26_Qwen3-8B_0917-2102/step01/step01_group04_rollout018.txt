import numpy as np
from scipy.optimize import minimize

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        return np.sum(radii) - 2.636

    def ineq_constraints(radii):
        return [radii[i] >= 0 for i in range(26)]

    def boundary_constraints(radii):
        return [radii[i] <= 1 - 1e-12 for i in range(26)]

    def non_overlap_constraints(radii):
        return [radii[i] + radii[j] <= 1e12 for i in range(26) for j in range(i + 1, 26)]

    def pack_circles():
        centers = np.zeros((26, 2))
        radii = np.full(26, 1 / 26)

        for i in range(26):
            if i % 2 == 0:
                row = i // 2
                col = i % 2
                centers[i] = [col / (2 * np.sqrt(2)), row / (2 * np.sqrt(2))]
            else:
                row = i // 2
                col = i % 2
                centers[i] = [(col + 0.5) / (2 * np.sqrt(2)), row / (2 * np.sqrt(2))]

        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j]:
                    radii[i] = (dist - radii[j]) / 2
                    radii[j] = (dist - radii[i]) / 2

        for i in range(26):
            x, y = centers[i]
            radii[i] = min(radii[i], 1 - max(x, 1 - x, y, 1 - y))

        return centers, radii

    centers, radii = pack_circles()
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
