import numpy as np
from scipy.optimize import minimize

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(centers, radii):
        n = len(radii)
        cons = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            cons.append((x - r >= -1e-12) & (x + r <= 1 + 1e-12) & (y - r >= -1e-12) & (y + r <= 1 + 1e-12))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                cons.append(dist >= radii[i] + radii[j] - 1e-12)
        return cons

    initial_centers = np.random.rand(26, 2)
    initial_radii = np.random.rand(26) * 0.1

    result = minimize(
        fun=objective,
        x0=initial_radii,
        constraints=constraints,
        method='SLSQP',
        options={'ftol': 1e-10, 'maxiter': 1000}
    )

    centers = np.random.rand(26, 2)
    radii = np.abs(result.x)

    return centers, radii, np.sum(radii)
