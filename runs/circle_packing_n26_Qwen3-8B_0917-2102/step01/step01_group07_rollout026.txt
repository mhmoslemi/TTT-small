import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial guess for centers in a hexagonal packing
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    
    # Generate a hexagonal packing pattern
    rows = int(np.ceil(np.sqrt(num_circles)))
    cols = int(np.ceil(num_circles / rows))
    
    for i in range(rows):
        for j in range(cols):
            idx = i * cols + j
            if idx >= num_circles:
                break
            x = j + 0.5 * i
            y = i
            centers[idx] = [x, y]
    
    # Adjust to fit in unit square and optimize radii
    def objective(radii):
        return -np.sum(radii)
    
    def constraints(radii):
        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append(dist - (radii[i] + radii[j]) >= -1e-8)
        for i in range(num_circles):
            x, y = centers[i]
            constraints.append(x - radii[i] >= -1e-8)
            constraints.append(x + radii[i] <= 1 + 1e-8)
            constraints.append(y - radii[i] >= -1e-8)
            constraints.append(y + radii[i] <= 1 + 1e-8)
        return constraints
    
    # Optimize radii
    bounds = [(0, 1) for _ in range(num_circles)]
    result = minimize(objective, radii, bounds=bounds, constraints=constraints)
    radii = result.x
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
