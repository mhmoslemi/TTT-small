import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    # Initial guess: hexagonal packing with 5 rows
    centers = []
    radii = []
    for row in range(5):
        for col in range(5 + (row < 4)):
            x = col + (row % 2) / 2
            y = row + (col % 2) / 2
            if row < 3:
                centers.append([x, y])
                radii.append(0.1)
    centers = np.array(centers)
    radii = np.array(radii)
    n = len(radii)

    def objective(x):
        r = x[:n]
        dx = x[n:]
        centers_new = centers + dx
        radii_new = r
        sum_radii = np.sum(r)
        return -sum_radii  # Minimize negative to maximize

    def constraints(x):
        r = x[:n]
        dx = x[n:]
        centers_new = centers + dx
        dists = squareform(pdist(centers_new))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return 1.0
        for i in range(n):
            x, y = centers_new[i]
            if x - r[i] < -1e-8 or x + r[i] > 1 + 1e-8 or y - r[i] < -1e-8 or y + r[i] > 1 + 1e8:
                return 1.0
        return 0.0

    x0 = np.concatenate([radii, np.zeros(2 * n)])
    bounds = Bounds(np.zeros(3 * n), np.inf * np.ones(3 * n))

    result = minimize(objective, x0, bounds=bounds, constraints={"type": "ineq", "fun": constraints, "jac": None})
    x = result.x
    radii = x[:n]
    dx = x[n:]
    centers = centers + dx

    # Final validation
    centers = np.clip(centers, 0, 1)
    radii = np.clip(radii, 0, 1)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            raise ValueError("Final validation failed: circle outside unit square")
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError("Final validation failed: circles overlap")
    return centers, radii, np.sum(radii)
