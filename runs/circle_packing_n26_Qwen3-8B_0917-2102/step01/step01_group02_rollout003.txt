import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal packing with boundary constraints
    centers = []
    for i in range(n):
        row = i // 2
        col = i % 2
        x = col + (row % 2) * 0.5
        y = row * (np.sqrt(3) / 2)
        if y > 1:
            y = 1 - (y - 1) * 0.5
            x = np.clip(x, 0, 1)
        centers.append([x, y])
    centers = np.array(centers)
    
    # Initial radii based on minimum distance to boundaries
    radii = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        radii[i] = np.min([x, 1 - x, y, 1 - y])
    
    def objective(r):
        return -np.sum(r)
    
    def constraint_dist(r):
        dist_matrix = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < r[i] + r[j] - 1e-12:
                    return dist_matrix[i, j] - r[i] - r[j] + 1e-12
        return 0
    
    cons = ({'type': 'ineq', 'fun': constraint_dist})
    bounds = Bounds(np.zeros(n), np.full(n, 1.0))
    
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)
    
    if not result.success:
        raise RuntimeError("Optimization failed")
    
    radii = result.x
    
    # Final validation and adjustment
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = np.max([0, 1 - np.max([x, 1 - x, y, 1 - y])])
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                radii[i] = (dist - 1e-12) / 2
                radii[j] = (dist - 1e-12) / 2
    
    return centers, radii, np.sum(radii)
