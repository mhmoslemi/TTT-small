import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    # Initial estimate of circle positions using hexagonal packing
    num_circles = 26
    rows = int(np.ceil(np.sqrt(num_circles)))
    cols = int(np.ceil(num_circles / rows))
    spacing = 1.0 / (cols + 1)
    centers = np.empty((num_circles, 2))
    idx = 0
    for r in range(rows):
        for c in range(cols):
            x = c * spacing + spacing / 2
            y = r * spacing + spacing / 2
            if r % 2 == 0:
                x += spacing / 2
            if idx < num_circles:
                centers[idx] = [x, y]
                idx += 1

    # Initial guess for radii
    radii = np.full(num_circles, 0.05)

    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraints: circle must be within the square and not overlap with others
    def constraint_func(radii):
        for i in range(num_circles):
            x, y = centers[i]
            if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
                return -1
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    # Optimization
    bounds = [(0, 0.1) for _ in range(num_circles)]
    constraints = [{'type': 'ineq', 'fun': constraint_func}]
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=constraints)

    # Final centers and radii
    return (centers, result.x, np.sum(result.x))
