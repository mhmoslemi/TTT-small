import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import matplotlib.pyplot as plt

def run_packing():
    # Initial hexagonal grid with 26 circles
    n = 26
    rows = int(np.sqrt(n))
    cols = n // rows + (1 if n % rows else 0)
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.05)
    
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col / (cols - 1) if cols > 1 else 0.5
        y = row / (rows - 1) if rows > 1 else 0.5
        centers[i] = np.array([x, y])
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraints(radii):
        cons = []
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] * 2 - 1e-12})
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: (dist - (r[i] + r[j]))})
        return cons
    
    result = minimize(objective, radii, constraints=constraints)
    
    if not result.success:
        raise ValueError("Optimization failed")
    
    return centers, result.x, np.sum(result.x)
