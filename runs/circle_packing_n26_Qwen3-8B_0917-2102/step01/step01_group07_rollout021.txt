import numpy as np
from scipy.optimize import minimize

def run_packing():

    n = 26
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n) * 0.1

    def objective(r):
        return -np.sum(r)

    def constraint(r, i, j):
        return np.sqrt(np.sum((centers[i] - centers[j])**2)) - (r[i] + r[j])

    bounds = [(0, 1) for _ in range(n)]
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: constraint(r, i, j)})
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            r = x
        if x + r > 1:
            r = 1 - x
        if y - r < 0:
            r = y
        if y + r > 1:
            r = 1 - y
        radii[i] = r

    result = minimize(objective, radii, bounds=bounds, constraints=constraints)
    radii = result.x

        x, y = centers[i]
        r = radii[i]

            r = x
        if x + r > 1:
            r = 1 - x

            r = y
