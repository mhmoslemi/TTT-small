<strategy>Arrange circles in a hexagonal grid near boundaries with staggered rows to fill gaps efficiently.</strategy>

```python
import numpy as np
import scipy.optimize as opt

def run_packing():
    def fitness(radii):
        centers = np.zeros((26, 2))
        for i in range(26):
            r = radii[i]
            if i % 2 == 0:
                centers[i, 0] = (i // 2 + 0.5) * (1 - 2 * r) + r
                centers[i, 1] = (i // 2) * (1 - 2 * r) + r
            else:
                centers[i, 0] = (i // 2 + 0.5) * (1 - 2 * r) + r
                centers[i, 1] = (i // 2) * (1 - 2 * r) + r
        return centers

    def objective(radii):
        centers = fitness(radii)
        dists = np.zeros((26, 26))
        for i in range(26):
            for j in range(i + 1, 26):
                dists[i, j] = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
        return -np.sum(radii)

    def constraint(radii):
        centers = fitness(radii)
        for i in range(26):
            r = radii[i]
            if centers[i, 0] - r < 0 or centers[i, 0] + r > 1 or centers[i, 1] - r < 0 or centers[i, 1] + r > 1:
                return -1
        for i in range(26):
            for j in range(i + 1, 26):
                if np.sqrt(np.sum((centers[i] - centers[j]) ** 2)) < radii[i] + radii[j]:
                    return -1
        return 0

    bounds = [(0, 0.5) for _ in range(26)]
    x0 = np.full(26, 0.05)
    result = opt.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraint}],
        method='SLSQP'
    )
    radii = result.x
    centers = fitness(radii)
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
```
