import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial guess for centers in hexagonal grid
    centers = []
    radii = []
    for i in range(26):
        row = i // 6
        col = i % 6
        x = col / 6 + 0.1
        y = row / 4 + 0.1
        centers.append([x, y])
    
    centers = np.array(centers)
    radii = np.full(26, 0.1)

    def objective(radii):
        return -np.sum(radii)

    def constraint(radii):
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    res = minimize(
        objective,
        radii,
        constraints=[{'type': 'ineq', 'fun': constraint}],
        bounds=[(0, 1) for _ in range(26)],
        method='SLSQP'
    )

    # Final optimization to improve spacing
    for _ in range(5):
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    radii[i] = dist - radii[j] - 1e-8
                    radii[j] = dist - radii[i] - 1e-8

    # Final check for boundary constraints
    for i in range(26):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            radii[i] = min(r, 1 - np.min([x, 1 - x, y, 1 - y]))

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
