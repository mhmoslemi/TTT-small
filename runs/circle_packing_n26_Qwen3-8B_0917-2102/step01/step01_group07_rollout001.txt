import numpy as np
import scipy.optimize as optimize
import math

def run_packing():
    # Define hexagonal grid with clusters
    def generate_hex_grid(n=26, side=0.2, cluster_size=4):
        centers = []
        radii = []
        for i in range(n):
            row = i // cluster_size
            col = i % cluster_size
            x = col * (1 + side)
            y = row * (1 + side)
            # Adjust for boundary and cluster interactions
            if row % 2 == 1:
                x += side / 2
            x = np.clip(x, 0, 1)
            y = np.clip(y, 0, 1)
            centers.append([x, y])
            radii.append(side)
        return np.array(centers), np.array(radii)

    centers, radii = generate_hex_grid()

    # Optimization to maximize sum of radii with non-overlapping constraints
    def objective(r):
        return -np.sum(r)

    def constraint_func(r):
        n = len(r)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - 1e-8:
                    return 1
        return 0

    bounds = [(0, 1) for _ in range(26)]

    result = optimize.minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraint_func}]
    )

    centers = np.array(centers)
    radii = result.x

    # Final validation and adjustment
    for i in range(26):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            radii[i] = min(r, 1 - np.min([x, y]))
        for j in range(i + 1, 26):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < radii[i] + radii[j] - 1e-12:
                radii[i] = (dist - radii[j]) * 0.5
                radii[j] = (dist - radii[i]) * 0.5

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
