from scipy.optimize import Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Adjusting for optimal spacing and boundary conditions
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    r = 0.1  # initial guess for radius
    for i in range(n):

        col = i % 5
        x = col + 0.5 * (row % 2)
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = r

    # Objective: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

        return 0

    # Optimization bounds for radii
    bounds = [(0, 1) for _ in range(n)]

    # Run optimization
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=Bounds(0, 1),
        constraints=[{'type': 'ineq', 'fun': constraint_non_overlap}]
    )

    # Return best configuration
    return centers, result.x, np.sum(result.x)
