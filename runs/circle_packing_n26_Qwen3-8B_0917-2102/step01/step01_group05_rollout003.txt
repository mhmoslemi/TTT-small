import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance as dist

def run_packing():
    # Hexagonal grid parameters
    num_rows = 5
    num_cols = 5
    spacing = 1.0 / (num_rows + 1)
    
    # Initial circle centers in hexagonal grid
    centers = []
    radii = []
    
    for i in range(num_rows):
        for j in range(num_cols):
            x = j * spacing
            y = i * spacing
            if i % 2 == 0:
                x += spacing / 2
            centers.append([x, y])
    
    # Add boundary circles
    for i in range(4):
        x = i * spacing
        y = 0
        centers.append([x, y])
        radii.append(0.05)
    
    for i in range(4):
        x = i * spacing
        y = 1
        centers.append([x, y])
        radii.append(0.05)
    
    for i in range(4):
        x = 0
        y = i * spacing
        centers.append([x, y])
        radii.append(0.05)
    
    for i in range(4):
        x = 1
        y = i * spacing
        centers.append([x, y])
        radii.append(0.05)
    
    # Flatten and convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)
    
    # Define objective function to maximize total radius
    def objective(radii):
        return -np.sum(radii)  # Minimize negative total
    
    # Define constraints for non-overlapping and boundary
    def constraint_func(radii):
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0
    
    # Add constraints for boundary
    def boundary_constraint(radii):
        for i in range(len(centers)):
            x, y = centers[i]
            if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
                return -1
        return 0
    
    # Optimization
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        jac=None,
        bounds=[(0, 0.5) for _ in range(len(radii))],
        constraints=[
            {'type': 'ineq', 'fun': constraint_func},
            {'type': 'ineq', 'fun': boundary_constraint}
        ]
    )
    
    centers = np.array(centers)
    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
