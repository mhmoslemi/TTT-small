import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import matplotlib.pyplot as plt

def run_packing():
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraints(centers, radii):
        cons = []
        # Boundary constraints
        for i in range(len(radii)):
            x, y = centers[i]
            r = radii[i]
            cons.append((x - r >= -1e-12) & (x + r <= 1 + 1e-12) & (y - r >= -1e-12) & (y + r <= 1 + 1e-12))
        # Pairwise distance constraints
        dists = squareform(pdist(centers))
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                cons.append(dists[i, j] >= radii[i] + radii[j] - 1e-12)
        return cons

    # Initialize with hexagonal packing
    n = 26
    centers = []
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col / 5 + 0.1
        y = row / 5 + 0.1
        centers.append([x, y])
    centers = np.array(centers)

    # Initial radii based on packing density
    initial_radii = np.full(n, 0.05)

    # Optimization
    result = minimize(
        objective,
        initial_radii,
        method='SLSQP',
        bounds=[(0, 1) for _ in range(n)],
        constraints=[{'type': 'ineq', 'fun': lambda x: x[i] for i in range(n)} for _ in range(n)],
        options={'ftol': 1e-8, 'maxiter': 1000}
    )

    centers = np.array(centers)
    radii = result.x
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
