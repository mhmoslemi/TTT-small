import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from sklearn.cluster import KMeans

def run_packing():
    np.random.seed(42)
    n = 26
    initial_centers = np.random.rand(n, 2)
    initial_radii = np.random.rand(n) * 0.1

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((initial_centers[i] - initial_centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        for i in range(n):
            x, y = initial_centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8 or y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8:
                return False
        return True

    bounds = [(0, 1) for _ in range(n)]
    result = minimize(objective, initial_radii, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraints})

    radii = result.x
    centers = initial_centers.copy()

    # Optimize cluster positions using KMeans
    cluster_centers = KMeans(n_clusters=4, random_state=42).fit(centers).cluster_centers_
    for i in range(n):
        for j in range(4):
            dist = np.sqrt(np.sum((centers[i] - cluster_centers[j])**2))
            if dist < 0.1:
                centers[i] = cluster_centers[j] + np.random.rand(2) * 0.05 - 0.025

    # Final validation
    def validate(radii, centers):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8):
                return False
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    if not validate(radii, centers):
        raise ValueError("Final packing invalid")

    return centers, radii, np.sum(radii)
