import numpy as np
import scipy.optimize as opt

def run_packing():
    # Hexagonal grid parameters
    n = 26
    rows = int(np.ceil(n / 3))
    cols = 3
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    total_radius = 0
    
    # Place circles in hexagonal packing
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        centers[i] = [x, y]
        radii[i] = 0.1
        total_radius += 0.1
    
    # Adjust radii for boundary constraints
    boundary_radii = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        r = 0.1
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            r = min(x, 1 - x, y, 1 - y)
        radii[i] = r
        boundary_radii[i] = r
        total_radius = np.sum(boundary_radii)
    
    # Improve by adjusting for spacing
    def objective(radii):
        return -np.sum(radii)
    
    def constraint(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0
    
    bounds = [(0, 1) for _ in range(n)]
    result = opt.minimize(
        objective,
        x0=boundary_radii,
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraint}]
    )
    
    radii = result.x
    total_radius = np.sum(radii)
    return (centers, radii, total_radius)
