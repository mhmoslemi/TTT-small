import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial guess: Hexagonal packing with 5 rows (26 circles)
    # First row: 5 circles, next 4 rows with 5, 5, 5, 7 (adjusting for 26 circles)
    # Starting from bottom (y=0) to top (y=1)
    centers = []
    radii = []
    for row in range(5):
        for col in range(5):
            if row % 2 == 0:
                x = col + 0.5
            else:
                x = col + 0.25
            y = row + 0.5
            if x > 1 or y > 1:
                break
            centers.append([x, y])
        if row % 2 == 1:
            for col in range(5):
                if col % 2 == 1:
                    x = col + 0.5
                    y = row + 0.5
                    if x > 1 or y > 1:
                        break
                    centers.append([x, y])
    # Adjust for 26 circles
    centers = centers[:26]
    # Initial radius for all circles
    r_initial = 0.1
    radii = [r_initial] * len(centers)
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraint_center(radii):
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if (x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1):
                return -1
        return 0
    
    def constraint_overlap(radii):
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0
    
    constraints = [
        {'type': 'ineq', 'fun': constraint_center},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]
    
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        constraints=constraints,
        bounds=[(0, 1)] * len(radii),
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    return np.array(centers), np.array(result.x), np.sum(result.x)
