from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():

    radii = np.full(n, 0.05)

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r)

    cons = ({
        'type': 'ineq',
        'fun': constraint,
        'jac': lambda r: np.full_like(r, 1)
    })

    bounds = Bounds(np.zeros(n), np.full(n, 1.0))

    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)

    if not result.success:
        raise RuntimeError("Optimization failed")

    radii = result.x

    def initial_centers_hex(n):
        rows = []
        for i in range(n):
            row = i // 2
            col = i % 2
            x = (col + 0.5) / (np.sqrt(3) * 1)
            y = row / (np.sqrt(3) * 1)
            rows.append([x, y])
        return np.array(rows)

    centers = initial_centers_hex(n)
    centers += np.random.normal(0, 0.01, centers.shape)
    centers = np.clip(centers, 0, 1)

    def adjust_radii(r, c):
        for i in range(len(r)):
            x, y = c[i]
            r[i] = np.min([r[i], 1 - np.max([x, 1 - x, y, 1 - y])])
        for i in range(len(r)):
            for j in range(i + 1, len(r)):
                dx, dy = c[i] - c[j]
                dist = np.hypot(dx, dy)
                if dist < r[i] + r[j] - 1e-12:
                    r[i] = (dist - 1e-12) / 2
                    r[j] = (dist - 1e-12) / 2
        return r

    radii = adjust_radii(radii, centers)

    return centers, radii, np.sum(radii)
