import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n) * 0.1

    def objective(radii):
        return -np.sum(radii)

    def constraint(radii):
        dist_matrix = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return 1
        return 0

    def boundary_constraint(radii):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return 1
        return 0

    result = minimize(objective, radii, constraints=[
        {"type": "ineq", "fun": constraint},
        {"type": "ineq", "fun": boundary_constraint}
    ])

    radii = result.x
    radii[radii < 0] = 0
    radii = np.clip(radii, 0, 1)

    # Adjust centers to fit within boundaries
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            centers[i, 0] = r
        elif x + r > 1:
            centers[i, 0] = 1 - r
        if y - r < 0:
            centers[i, 1] = r
        elif y + r > 1:
            centers[i, 1] = 1 - r

    # Final validation and adjustment
    dist_matrix = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                radii[i] = max(0, (dist_matrix[i, j] - radii[j]) * 0.5)
                radii[j] = max(0, (dist_matrix[i, j] - radii[i]) * 0.5)

    return centers, radii, np.sum(radii)
