from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

    radii = []

        return -np.sum(radii)

        centers_flat = x.reshape(-1, 2)
        dist_matrix = squareform(pdist(centers_flat))

                    return 1
        return 0

    result = minimize(objective, x0, constraints={"type": "ineq", "fun": constraint_func})

    centers = result.x.reshape(-1, 2)
    radii = np.array([0.06] * n)

    # Adjust radii to fit in square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
