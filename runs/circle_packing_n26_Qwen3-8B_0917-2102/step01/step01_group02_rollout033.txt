import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal packing with spacing and scaling
    rows = []
    for i in range(n):
        row_idx = i // 5
        col_idx = i % 5
        x = col_idx / (np.sqrt(3) * 1.2)
        y = row_idx / 1.5
        rows.append([x, y])
    centers = np.array(rows)
    centers += np.random.normal(0, 0.05, centers.shape)
    centers = np.clip(centers, 0, 1)

    # Initial radii based on spacing
    dists = squareform(pdist(centers))
    radii = np.min(dists) / 2 * np.ones(n)

    def objective(r):
        return -np.sum(r)

    def constraint_func(r):
        return np.sum(r)

    # Optimization bounds
    bounds = Bounds(np.zeros(n), np.full(n, 1.0))

    # Optimization with soft constraints
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, 
                     constraints=[{'type': 'ineq', 'fun': lambda r: np.sum(r) - 2.636}])

    if not result.success:
        raise RuntimeError("Optimization failed")

    radii = result.x

    # Final adjustment for boundary and non-overlapping constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):

                radii[i] = (dist - 1e-12) / 2
                radii[j] = (dist - 1e-12) / 2

    return centers, radii, np.sum(radii)
