import numpy as np
from scipy.spatial import Voronoi, voronoi_plot_2d
from scipy.optimize import minimize
import matplotlib.pyplot as plt

def run_packing():
    # Initial hexagonal packing with 26 circles
    n = 26
    centers = []
    radii = []
    for i in range(n):
        row = i // 7
        col = i % 7
        x = col + 0.5 * (1 - 2 * (col % 2))
        y = row + 0.5 * (1 - 2 * (row % 2))
        centers.append([x, y])
    centers = np.array(centers)
    radii = np.full(n, 0.1)

    # Function to compute total sum of radii
    def total_sum(radii):
        return np.sum(radii)

    # Objective function to maximize sum of radii
    def objective(radii):
        return -total_sum(radii)  # Minimize negative sum

    # Constraints: circles must not overlap and stay within unit square
    def constraint(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        for i in range(n):
            if centers[i, 0] - radii[i] < -1e-8 or centers[i, 0] + radii[i] > 1 + 1e-8:
                return False
            if centers[i, 1] - radii[i] < -1e-8 or centers[i, 1] + radii[i] > 1 + 1e-8:
                return False
        return True

    # Optimization
    result = minimize(objective, radii, bounds=[(0, 1)] * n, constraints={"type": "constraints", "fun": constraint})
    radii = result.x

    # Final check and adjust to ensure centers are within bounds
    for i in range(n):
        r = radii[i]
        if centers[i, 0] - r < -1e-8:
            radii[i] = centers[i, 0] - 1e-8
        if centers[i, 0] + r > 1 + 1e-8:
            radii[i] = 1 - centers[i, 0] - 1e-8
        if centers[i, 1] - r < -1e-8:
            radii[i] = centers[i, 1] - 1e-8
        if centers[i, 1] + r > 1 + 1e-8:
            radii[i] = 1 - centers[i, 1] - 1e-8

    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
