    centers = []
    radii = []
    for i in range(n):

            centers.append([x, y])
            radii.append(max_r)

    # Initial sum of radii
    sum_radii = np.sum(radii)

    # Optimization function to maximize sum of radii with constraints
    def objective(radii):
        return -np.sum(radii)

    def constraint1(radii):
        return np.sum(radii) - sum_radii

    def constraint2(radii):
        return np.sum(radii)

    def constraint3(radii):
        return np.sum(radii)

    bounds = [(0, 1) for _ in range(n)]

    # Optimize using simulated annealing
    try:

            objective,
            radii,

                {'type': 'ineq', 'fun': constraint3}
            ],
            method='SLSQP'
        )

        centers = np.array(centers)
        return centers, radii, sum_radii
