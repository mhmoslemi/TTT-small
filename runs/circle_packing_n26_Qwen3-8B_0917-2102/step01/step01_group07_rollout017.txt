from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

        return -np.sum(r)

    def constraint1(r):
        return np.sum(r) - 2.636

    def constraint2(r):
        return np.sum(r) - 2.636

    def bounds_func(r):
        return [(0, 1) for _ in r]

    n = 26

    def constraints(r):
        cons = []
        cons.append({'type': 'ineq', 'fun': constraint1, 'jac': lambda r: np.ones_like(r)})
        cons.append({'type': 'ineq', 'fun': constraint2, 'jac': lambda r: np.zeros_like(r)})
        return cons

    result = minimize(objective, radii, method='SLSQP', bounds=bounds_func(n), constraints=constraints(radii))
    radii = result.x

    # Simple hexagonal packing
    centers = np.zeros((n, 2))
    r = radii[0]
    for i in range(n):
        row = i // 7
        col = i % 7
        centers[i, 0] = col * (1 + r) / 7
        centers[i, 1] = row * (1 + r) / 7
    for i in range(n):
        centers[i, 0] += 0.5 * (1 - (1 + r) / 7) * (1 + r) ** 2
        centers[i, 1] += 0.5 * (1 - (1 + r) / 7) * (1 + r) ** 2

    for i in range(n):

        r_i = radii[i]
        if x - r_i < 0 or x + r_i > 1 or y - r_i < 0 or y + r_i > 1:
            r_i = np.min([r_i, np.min([x, 1 - x, y, 1 - y])])
        radii[i] = r_i
        centers[i] = [np.clip(x, r_i, 1 - r_i), np.clip(y, r_i, 1 - r_i)]

    return centers, radii, np.sum(radii)
