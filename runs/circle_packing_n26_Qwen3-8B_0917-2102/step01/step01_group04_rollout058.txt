import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import cdist

def run_packing():
    # Initial guess for circle centers using hexagonal packing
    def hexagonal_centers(n):
        rows = int(np.ceil(n / 6))
        cols = int(np.ceil(n / rows))
        centers = []
        for i in range(rows):
            for j in range(cols):
                x = j + 0.5
                y = i + 0.5
                centers.append((x, y))
        return np.array(centers)
    
    n = 26
    centers = hexagonal_centers(n)
    radii = np.full(n, 0.1)

    # Optimization variables
    bounds = Bounds(np.zeros(n * 2), np.ones(n * 2))

    def objective(x):
        radii = x[n:]
        centers = x[:n*2].reshape(n, 2)
        return -np.sum(radii)

    def constraints(x):
        radii = x[n:]
        centers = x[:n*2].reshape(n, 2)
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    result = minimize(
        objective,
        np.concatenate([centers.flatten(), np.full(n, 0.1)]),
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraints},
        method='SLSQP'
    )

    centers = result.x[:n*2].reshape(n, 2)
    radii = result.x[n:]
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
