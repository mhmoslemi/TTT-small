import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import LinearConstraint
import random

def run_packing():
    n = 26
    # Initialize centers in a hexagonal pattern
    centers = []
    for i in range(n):
        row = i // 2
        col = i % 2
        x = col + row / (np.sqrt(3) + 1)
        y = row / np.sqrt(3)
        centers.append([x, y])
    centers = np.array(centers)
    # Clip to [0, 1]^2
    centers = np.clip(centers, 0, 1)
    # Small perturbation
    centers += np.random.normal(0, 0.01, centers.shape)
    centers = np.clip(centers, 0, 1)
    
    # Initial radii as uniform
    radii = np.full(n, 0.05)
    
    # Define objective function
    def objective(r):
        return -np.sum(r)
    
    # Define constraints for non-overlapping and boundary
    def constraint_func(r):
        r = r.reshape(n)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < r[i] + r[j] - 1e-12:
                    return -np.inf
        for i in range(n):
            x, y = centers[i]
            if x - r[i] < -1e-12 or x + r[i] > 1 + 1e-12 or y - r[i] < -1e-12 or y + r[i] > 1 + 1e-12:
                return -np.inf
        return np.sum(r) - 2.636
    
    # Optimization with bounds and constraints
    bounds = Bounds(np.zeros(n), np.full(n, 1.0))
    cons = ({'type': 'ineq', 'fun': constraint_func})
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)
    
    if not result.success:
        raise RuntimeError("Optimization failed")
    
    radii = result.x
    
    # Final validation and adjustment
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = np.max([0, 1 - np.max([x, 1 - x, y, 1 - y])])
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist < radii[i] + radii[j] - 1e-12:
                radii[i] = (dist - 1e-12) / 2
                radii[j] = (dist - 1e-12) / 2
    
    return centers, radii, np.sum(radii)
