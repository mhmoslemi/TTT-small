import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial guess: hexagonal packing with 6 rows
    centers = np.array([
        [0.5, 0.5],
        [0.25, 0.25], [0.75, 0.25],
        [0.125, 0.375], [0.375, 0.375], [0.625, 0.375], [0.875, 0.375],
        [0.0625, 0.5], [0.25, 0.5], [0.5, 0.5], [0.75, 0.5], [0.9375, 0.5],
        [0.03125, 0.625], [0.125, 0.625], [0.25, 0.625], [0.375, 0.625], [0.5, 0.625], [0.625, 0.625], [0.75, 0.625], [0.875, 0.625], [0.96875, 0.625],
        [0.015625, 0.75], [0.0625, 0.75], [0.125, 0.75], [0.1875, 0.75], [0.25, 0.75]
    ])

    # Initial radii based on spacing
    radii = np.full(n, 0.04)

    def objective(x):
        radii[:] = x[:n]
        return -np.sum(radii)  # Maximize sum by minimizing negative sum

    def constraint1(x):
        radii[:] = x[:n]
        for i in range(n):
            x_i, y_i = centers[i]
            r = radii[i]
            if x_i - r < 0 or x_i + r > 1 or y_i - r < 0 or y_i + r > 1:
                return -1
        return 0

    def constraint2(x):
        radii[:] = x[:n]
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    # Initial guess
    x0 = radii.copy()

    # Optimization
    constraints = [
        {'type': 'ineq', 'fun': constraint1},
        {'type': 'ineq', 'fun': constraint2}
    ]

    result = minimize(objective, x0, method='SLSQP', constraints=constraints, bounds=[(0, 1)] * n)

    if not result.success:
        raise Exception("Optimization failed")

    centers = np.array(centers)
    radii = result.x
    sum_radii = np.sum(radii)

    return (centers, radii, sum_radii)
