import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal packing with adaptive spacing
    centers = []
    r = 0
    for i in range(n):
        col = i // 5
        row = i % 5
        x = (col + 0.5 * (row % 2)) / (np.sqrt(3) * 1)
        y = (row + 0.5 * (col % 2)) / (np.sqrt(3) * 1)
        centers.append([x, y])
    centers = np.array(centers)
    centers += np.random.normal(0, 0.01, centers.shape)
    centers = np.clip(centers, 0, 1)

    # Initial radii based on spacing
    dist = squareform(pdist(centers))
    max_dist = np.max(dist)
    radii = np.min(np.abs(dist - max_dist / 2), axis=1)

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 2.636

    cons = ({
        'type': 'ineq',
        'fun': constraint
    })

    bounds = Bounds(np.zeros(n), np.full(n, 1.0))

    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)

    if not result.success:
        raise RuntimeError("Optimization failed")

    radii = result.x

    # Final validation and adjustment
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = np.max([0, 1 - np.max([x, 1 - x, y, 1 - y])])

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                radii[i] = (dist - 1e-12) / 2
                radii[j] = (dist - 1e-12) / 2

    return centers, radii, np.sum(radii)
