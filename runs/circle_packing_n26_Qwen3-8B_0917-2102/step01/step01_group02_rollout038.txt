import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Initial hexagonal packing with boundary-aware adjustment
    centers = []
    r = 0.5
    for i in range(n):
        col = i // 4
        row = i % 4
        x = col + (row % 2) * 0.5
        y = row * (np.sqrt(3) / 2) + 0.25
        centers.append([x, y])

    centers /= np.max(centers)

    # Initial uniform radii based on packing density
    radii = np.full(n, 0.15)

    def objective(r):
        return -np.sum(r)

                if dists[i, j] < r[i] + r[j] + 1e-8:
                    return -1
        return 0

    cons = (
        {'type': 'ineq', 'fun': lambda r: np.sum(r) - 2.636},
        {'type': 'eq', 'fun': lambda r: constraint_dist(r, centers)}
    )

    bounds = Bounds(np.zeros(n), np.full(n, 1.0))

    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)

    if not result.success:
        raise RuntimeError("Optimization failed")

    radii = result.x

    # Final boundary adjustment
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            radii[i] = np.min([r, x])
        elif x + r > 1:
            radii[i] = np.min([r, 1 - x])

            radii[i] = np.min([radii[i], y])

            radii[i] = np.min([radii[i], 1 - y])

    return centers, radii, np.sum(radii)
