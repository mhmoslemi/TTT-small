import numpy as np
from scipy.optimize import minimize

def run_packing():
    np.random.seed(42)
    n = 26

    # Initial guess: hexagonal packing with clusters near boundaries
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        # Boundary constraints
        for i in range(n):
            if radii[i] < 0:
                return False
            x, y = centers[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12 or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return False
        # Circle-circle constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    # Optimization
    bounds = [(0, 1) for _ in range(n)]
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints={'type': 'callback', 'fun': constraints},
        method='SLSQP'
    )

    if not result.success:
        raise ValueError("Optimization failed")

    return centers, result.x, np.sum(result.x)
