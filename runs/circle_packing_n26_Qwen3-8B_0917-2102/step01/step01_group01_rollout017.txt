import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        return np.sum(radii) + 1e-12

    def boundary_circle_constraints(radii):
        return np.sum(radii) + 1e-12

    def circle_overlap_constraints(radii):
        return np.sum(radii) + 1e-12

    def run_hybrid_packing():
        n = 26
        centers = []
        radii = []

        # Hexagonal grid with varying spacing for denser packing
        for i in range(n):
            row = i // 5
            col = i % 5
            x = col / 5.0 + (0.05 if i % 2 == 0 else 0.07)
            y = row / 5.0 + (0.05 if i % 2 == 0 else 0.07)
            r = 0.08 if i % 2 == 0 else 0.06
            centers.append([x, y])
            radii.append(r)

        centers = np.array(centers)
        radii = np.array(radii)

        # Add extra circles near boundaries
        for i in range(5):
            x, y = centers[i]
            if x < 0.25:
                centers[i] = [0.25, y]
            elif x > 0.75:
                centers[i] = [0.75, y]
            if y < 0.25:
                centers[i] = [x, 0.25]
            elif y > 0.75:
                centers[i] = [x, 0.75]
            radii[i] = min(radii[i], 0.15)

        return centers, radii

    def validate_and_refine(centers, radii):
        for i in range(len(radii)):
            x, y = centers[i]
            r = radii[i]
            if (x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1):
                radii[i] = max(0, min(r, 1 - max(x, 1 - x, y, 1 - y)))

        dist_matrix = squareform(pdist(centers))
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    radii[i] = max(0, (dist_matrix[i, j] - radii[j]) * 0.5)
                    radii[j] = max(0, (dist_matrix[i, j] - radii[i]) * 0.5)

        return centers, radii

    def optimize_centers(centers, radii):
        def objective_func(x):
            centers_flat = x.reshape(-1, 2)
            return -np.sum(radii)

        def constraint_func(x):
            centers_flat = x.reshape(-1, 2)
            dist_matrix = squareform(pdist(centers_flat))
            for i in range(len(radii)):
                for j in range(i + 1, len(radii)):
                    if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                        return 1
            return 0

        x0 = centers.reshape(-1)
        result = minimize(objective_func, x0, constraints={"type": "ineq", "fun": constraint_func})
        return result.x.reshape(-1, 2)

    centers, radii = run_hybrid_packing()
    centers = optimize_centers(centers, radii)
    radii = np.clip(radii, 0, 1)
    centers, radii = validate_and_refine(centers, radii)
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
