import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Define hexagonal packing with 5 rows (approximate for 26 circles)
    rows = 5
    cols = 5  # 5 rows with 5, 5, 5, 5, 6 circles respectively (total 26)
    centers = []

        for c in range(cols):
            if r % 2 == 0:
                x = c * (1 + 1e-6)
                y = r * (np.sqrt(3)/2) * (1 + 1e-6)
            else:
                x = (c + 0.5) * (1 + 1e-6)
                y = r * (np.sqrt(3)/2) * (1 + 1e-6)
            # Adjust to fit in unit square with some padding
            x = np.clip(x, 0, 1 - 1e-6)
            y = np.clip(y, 0, 1 - 1e-6)
            centers.append([x, y])
    
    n = len(centers)
    # Initial radius guess based on packing density
    initial_radii = np.full(n, 1 / (2 * np.sqrt(3)) * 0.85)
    radii = initial_radii
    
    # Optimization to maximize sum of radii with non-overlap and boundary constraints
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum
    
    def constraint_func(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0
    
    constraints = [{'type': 'ineq', 'fun': constraint_func}]
    
    result = minimize(
        objective, 
        initial_radii, 
        method='SLSQP', 
        bounds=[(0, 1)] * n, 
        constraints=constraints
    )
    
    if not result.success:
        raise ValueError("Optimization failed")
    
    return (np.array(centers), result.x, np.sum(result.x))
