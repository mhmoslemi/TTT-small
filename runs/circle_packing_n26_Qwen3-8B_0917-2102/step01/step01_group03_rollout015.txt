import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    n = 26
    # Initial hexagonal packing with boundary adjustment
    def initial_positions(n):
        rows = int(np.ceil(n / 2))
        cols = int(np.ceil(n / rows))
        centers = np.zeros((n, 2))
        for i in range(n):
            row = i // cols
            col = i % cols
            x = col + 0.5 * (row % 2)
            y = row + 0.5
            centers[i] = [x, y]
        return centers

    centers = initial_positions(n)
    radii = np.zeros(n)
    # Initial guess for radii based on packing density
    radii = np.full(n, 0.1)  # Will be refined

    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii

    def constraints(radii):
        cons = []
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] * 2 - 1e-6})
        for i, j in combinations(range(n)):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: dist - (r[i] + r[j]) + 1e-6})
        return cons

    res = minimize(objective, radii, method='SLSQP', constraints=constraints, bounds=[(0, 1)]*n)
    radii = res.x

    # Final validation
    for i in range(n):
        x, y = centers[i]
        if (x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1):
            radii[i] = np.minimum(radii[i], 1 - np.min([x, 1 - x, y, 1 - y]))

    return (centers, radii, np.sum(radii))
