import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    # Initial guess for hexagonal packing
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 0.5 * (row % 2))
        y = row * (np.sqrt(3) / 2) * (1 - 0.5 * (row % 2))
        centers[i] = [x, y]
        radii[i] = 0.05

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8 or y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8:
                return False
        return True

    bounds = [(0, 0.1) for _ in range(n)]
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints={'type': 'callback', 'fun': constraints})
    radii = result.x
    return centers, radii, np.sum(radii)
