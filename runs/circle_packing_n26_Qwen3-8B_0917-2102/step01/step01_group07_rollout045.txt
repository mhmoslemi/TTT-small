import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    # Initial guess for centers with hexagonal packing
    n = 26
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        centers[i, 0] = col / 4.5
        centers[i, 1] = row / 4.5

    def objective(radii):
        return -np.sum(radii)

    def constraint1(radii):
        for i in range(n):
            if centers[i, 0] - radii[i] < 0 or centers[i, 0] + radii[i] > 1:
                return -1
            if centers[i, 1] - radii[i] < 0 or centers[i, 1] + radii[i] > 1:
                return -1
        return 0

    def constraint2(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    bounds = [(0, 1) for _ in range(n)]
    cons = [{'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2}]

    result = minimize(objective, np.ones(n) * 0.05, bounds=bounds, constraints=cons, method='SLSQP')
    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
