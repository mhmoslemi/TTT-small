from scipy.optimize import minimize

def run_packing():

    radii[0] = 0.1

    # Last row (bottom boundary)
    centers[-1] = [0.5, 0.0]
    radii[-1] = 0.1

    for i in range(1, n - 1):

            centers[i] = [0.25, 0.25 + i * 0.2]

            centers[i] = [0.75, 0.25 + i * 0.2]

                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:

    # Optimization
    bounds = [(0, 1.0) for _ in range(n)]
    res = minimize(objective, radii, bounds=bounds, constraints={"type": "custom", "fun": constraints, "args": ()})
