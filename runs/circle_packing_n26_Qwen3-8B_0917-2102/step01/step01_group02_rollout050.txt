import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import LinearConstraint

def run_packing():
    n = 26
    # Initial hexagonal packing with boundary alignment
    centers = []
    for i in range(n):
        row = i // 2
        col = i % 2
        x = 0.5 + (col - 0.5) / np.sqrt(3)
        y = row / np.sqrt(3)
        centers.append([x, y])
    centers = np.array(centers)
    # Adjust to fit within [0,1]^2
    scale = np.sqrt(3)
    centers /= scale
    # Add small perturbation
    centers += np.random.normal(0, 0.05, centers.shape)
    centers = np.clip(centers, 0, 1)

    # Initial radii based on minimum spacing
    radii = np.full(n, 0.05)

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r)

    cons = ({
        'type': 'ineq',
        'fun': constraint,
        'jac': lambda r: np.ones_like(r)
    })

    bounds = Bounds(np.zeros(n), np.full(n, 1.0))

    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)

    if not result.success:
        raise RuntimeError("Optimization failed")

    radii = result.x

    # Final validation and adjustment
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = np.max([0, 1 - np.max([x, 1 - x, y, 1 - y])])

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                radii[i] = (dist - 1e-12) / 2
                radii[j] = (dist - 1e-12) / 2

    return centers, radii, np.sum(radii)
