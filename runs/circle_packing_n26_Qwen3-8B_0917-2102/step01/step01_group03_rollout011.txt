import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    # Initial guess based on hexagonal packing
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    num_circles = 26
    # First row
    centers[0] = [0.5, 1.0]
    radii[0] = 0.15
    # Subsequent rows with hexagonal spacing
    for i in range(1, num_circles):
        row = i // 5
        col = i % 5
        x = 0.5 + (col - 2.5) * 0.2
        y = 1.0 - (row + 1) * 0.2
        centers[i] = [x, y]
        radii[i] = 0.1

    def objective(x):
        return -np.sum(x)

    def constraints(x):
        x = x.reshape((num_circles, 2))
        r = x[:, 2:]
        # Boundary constraints
        for i in range(num_circles):
            xi, yi = x[i, :2]
            ri = r[i]
            if xi - ri < -1e-12 or xi + ri > 1 + 1e-12 or yi - ri < -1e-12 or yi + ri > 1 + 1e-12:
                return False
        # Pairwise distance constraints
        dists = squareform(pdist(x[:, :2]))
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                if dists[i, j] < r[i] + r[j] - 1e-12:
                    return False
        return True

    x0 = np.concatenate([centers.reshape(-1), radii])
    bounds = Bounds(np.concatenate([np.repeat(0, num_circles * 2), np.zeros(num_circles)]),
                    np.concatenate([np.repeat(1, num_circles * 2), np.inf * np.ones(num_circles)]))

    result = minimize(objective, x0, bounds=bounds, constraints=lambda x: constraints(x), method='SLSQP')

    x = result.x
    centers = x[:num_circles * 2].reshape((num_circles, 2))
    radii = x[num_circles * 2:]
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
