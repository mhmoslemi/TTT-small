from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():

    centers = []
    radii = []

    # Start with a hexagonal packing
    for i in range(n):
        row = i // 5
        col = i % 5
        x = (col + 0.5) / 5.0
        y = (row + 0.5) / 5.0
        r = 0.05
        centers.append([x, y])
        radii.append(r)

    centers = np.array(centers)
    radii = np.array(radii)

    # Add boundary circles near edges
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x < r + 0.1:
            centers[i] = [r + 0.1, y]
            radii[i] = min(r, 0.1)
        elif x > 1 - r - 0.1:
            centers[i] = [1 - r - 0.1, y]
            radii[i] = min(r, 0.1)
        elif y < r + 0.1:
            centers[i] = [x, r + 0.1]
            radii[i] = min(r, 0.1)
        elif y > 1 - r - 0.1:
            centers[i] = [x, 1 - r - 0.1]
            radii[i] = min(r, 0.1)

    def objective(x):
        centers_flat = x.reshape(-1, 2)
        return -np.sum(radii)

    def constraint(x):
        centers_flat = x.reshape(-1, 2)
        dist_matrix = squareform(pdist(centers_flat))
        for i in range(n):
            for j in range(i + 1, n):
                dist = dist_matrix[i, j]
                if dist < radii[i] + radii[j] - 1e-8:
                    return dist - (radii[i] + radii[j] - 1e-8)
        return 0

    x0 = centers.reshape(-1)
    result = minimize(objective, x0, method='SLSQP', constraints={"type": "ineq", "fun": constraint})
    centers = result.x.reshape(-1, 2)
    radii = np.clip(radii, 0, 1)

    # Final boundary check and adjustment
    for i in range(n):
        x, y = centers[i]

        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:

    dist_matrix = squareform(pdist(centers))
    for i in range(n):

            if dist < radii[i] + radii[j] - 1e-8:

                radii[j] = max(0, (dist - radii[i]) * 0.5)

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
