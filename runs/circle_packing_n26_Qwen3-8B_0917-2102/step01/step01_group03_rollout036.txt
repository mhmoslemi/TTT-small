import numpy as np
from scipy.optimize import minimize

def run_packing():
    np.random.seed(42)
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n) * 0.1

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        nonlocal centers
        radii = radii.reshape(-1)
        for i in range(len(radii)):
            x, y = centers[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12 or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return False
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    bounds = [(0, 0.5) for _ in range(n)]
    result = minimize(lambda r: -np.sum(r), x0=radii, bounds=bounds, constraints={"type": "ineq", "fun": constraints})
    radii = result.x
    return centers, radii, np.sum(radii)
