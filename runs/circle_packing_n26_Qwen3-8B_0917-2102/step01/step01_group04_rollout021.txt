import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.cluster.hierarchy import linkage, fcluster

def run_packing():
    # Initial hexagonal grid layout
    rows = 5
    cols = 5
    spacing = 0.3
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * spacing * np.sqrt(3)
            centers.append([x, y])
    centers = np.array(centers)
    radii = np.full(len(centers), 0.05)

    # Optimization parameters
    bounds = [(0, 1) for _ in range(2 * len(centers))]
    constraints = []

    # Add constraints for boundary and circle-circle distances
    def objective(r):
        return -np.sum(r)

    def constraint_func(x):
        r = x[len(centers):]
        c = x[:len(centers)*2].reshape(-1, 2)
        for i in range(len(c)):
            x_i, y_i = c[i]
            r_i = r[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return -1
        for i in range(len(c)):
            for j in range(i + 1, len(c)):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - 1e-12:
                    return -1
        return 0

    constraints.append({
        'type': 'ineq',
        'fun': constraint_func
    })

    x0 = np.concatenate([centers.flatten(), radii])
    result = minimize(objective, x0, bounds=bounds, constraints=constraints, method='SLSQP')

    centers_opt = result.x[:len(centers)*2].reshape(-1, 2)
    radii_opt = result.x[len(centers)*2:]
    sum_radii = np.sum(radii_opt)

    return centers_opt, radii_opt, sum_radii
