import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.cluster.vq import kmeans
from itertools import combinations

def run_packing():
    np.random.seed(42)
    n = 26

    # Initial hexagonal packing
    def hexagonal_layout():
        rows = int(np.ceil(np.sqrt(n)))
        cols = int(np.ceil(n / rows))
        centers = np.zeros((n, 2))
        for i in range(n):
            row = i // cols
            col = i % cols
            x = col + 0.5 * (row % 2)
            y = row * np.sqrt(3) / 2
            centers[i] = [x / (cols - 1) if cols > 1 else 0, y / (rows - 1) if rows > 1 else 0]
        return centers

    centers = hexagonal_layout()
    radii = np.full(n, 0.05)

    # Optimize radii with constraints
    def objective(r):
        return -np.sum(r)  # Maximize sum of radii

    def constraint_dist(i, j):
        return lambda r: np.sqrt(np.sum((centers[i] - centers[j])**2)) - (r[i] + r[j])

    cons = []
    for i, j in combinations(range(n), 2):
        cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: constraint_dist(i, j)(r)})

    result = minimize(objective, radii, method='SLSQP', constraints=cons, bounds=[(0, 1) for _ in range(n)])

    radii = result.x
    sum_radii = np.sum(radii)

    # Final check for boundary
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = np.clip(r, 0, np.min([x, 1 - x, y, 1 - y]))

    return centers, radii, sum_radii
