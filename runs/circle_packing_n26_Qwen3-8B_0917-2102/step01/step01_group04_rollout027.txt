import scipy.optimize as optimize

def run_packing():

    initial_radii = np.full(n, 0.05)

        return -np.sum(r)

        return r
    
    def bounds(r):
        return [(0, 1) for _ in range(n)]

        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': lambda r: np.full(n, 0.01), 'jac': None}]
    )
    
    centers = initial_centers
    radii = result.x

                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                energy += max(0, (radii[i] + radii[j] - dist) / 100)
        return energy
    
    def gradient(centers, radii):
        grad = np.zeros_like(radii)

                dist = np.sqrt(dx*dx + dy*dy)

        return grad

        )
        radii = result.x
    
    return (centers, radii, np.sum(radii))
