import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    np.random.seed(42)
    n = 26

    # Initial hexagonal grid with some randomness
    def hex_grid(n):
        rows = int(np.ceil(np.sqrt(n)))
        cols = int(np.ceil(n / rows))
        grid = []
        for i in range(rows):
            for j in range(cols):
                x = j + 0.5 * i
                y = i
                if len(grid) < n:
                    grid.append((x, y))
        return np.array(grid)

    centers = hex_grid(n)
    centers -= np.min(centers, axis=0)
    centers /= np.max(centers, axis=0)
    centers = np.random.permutation(centers)

    radii = np.full(n, 0.04)

    def objective(r):
        return -np.sum(r)

    def constraint(r, i, j):
        return np.sqrt(np.sum((centers[i] - centers[j])**2)) - r[i] - r[j]

    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: constraint(r, i, j)})

    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda r, i=i: np.sqrt(centers[i, 0]**2 + centers[i, 1]**2) - r[i]})

    bounds = [(0, 0.2) for _ in range(n)]

    result = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-10}
    )

    radii = result.x
    sum_radii = np.sum(radii)

    # Adjust radii to fit within the square
    for i in range(n):
        dist_to_boundary = np.array([centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1]])
        radii[i] = np.min([radii[i], np.min(dist_to_boundary)])

    # Final check to ensure no overlap
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist < radii[i] + radii[j] - 1e-12:
                radii[i] = dist - radii[j] + 1e-12

    return centers, radii, sum_radii
