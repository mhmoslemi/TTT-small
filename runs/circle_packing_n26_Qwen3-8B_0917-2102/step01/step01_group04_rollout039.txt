import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial hexagonal packing with 5 rows
    # Generated using hexagonal grid with decreasing row lengths
    centers = []
    radii = []
    for row in range(5):
        for col in range(5 - row):
            x = col + 0.5 * row / 2
            y = row * np.sqrt(3) / 2
            centers.append([x, y])
    for row in range(5, 11):
        for col in range(11 - row):
            x = col + 0.5 * (row - 5) / 2
            y = (row - 5) * np.sqrt(3) / 2
            centers.append([x, y])
    
    # Optimize radii for boundary and non-overlapping constraints
    n = len(centers)
    radii = np.array([0.05] * n)
    
    def objective(r):
        return -np.sum(r)  # Maximize sum of radii
    
    def constraint1(r):
        for i in range(n):
            if centers[i][0] - r[i] < -1e-12 or centers[i][0] + r[i] > 1 + 1e-12:
                return -1
            if centers[i][1] - r[i] < -1e-12 or centers[i][1] + r[i] > 1 + 1e-12:
                return -1
        return 0
    
    def constraint2(r):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - 1e-12:
                    return -1
        return 0
    
    cons = [{'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2}]
    
    result = minimize(objective, radii, constraints=cons, method='SLSQP')
    
    if not result.success:
        raise ValueError("Optimization failed")
    
    return np.array(centers), result.x, np.sum(result.x)
