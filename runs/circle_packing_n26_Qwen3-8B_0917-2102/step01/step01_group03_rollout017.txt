from scipy.optimize import minimize
from scipy.stats import qmc

def run_packing():

    n = 26
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n) * 0.1

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        for i in range(n):
            if radii[i] < 0:
                return False
            x, y = centers[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12 or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return False
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    bounds = [(0, 0.5) for _ in range(n)]
    res = minimize(objective, radii, bounds=bounds, constraints={"type": "success", "fun": constraints})

    if not res.success:
        raise RuntimeError("Optimization failed")

    radii = res.x
    return centers, radii, np.sum(radii)
