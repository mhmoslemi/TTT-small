from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

        col = i % 5

        y = row + 0.5
        centers.append([x, y])
    centers = np.array(centers)
    
    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

            if centers[i, 1] - r[i] < -1e-8 or centers[i, 1] + r[i] > 1 + 1e-8:
                return -1
        return 0
    
    # Optimization
    result = minimize(objective, radii, constraints={"type": "ineq", "fun": constraint}, method='SLSQP')
