from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d
from scipy.spatial.distance import pdist, squareform

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii, centers):
        n = len(radii)
        cons = []
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] + centers[i, 0] - 1e-12})
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: 1 - r[i] - centers[i, 0] - 1e-12})
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] + centers[i, 1] - 1e-12})
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: 1 - r[i] - centers[i, 1] - 1e-12})
        for i in range(n):
            for j in range(i + 1, n):
                cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: np.sqrt(np.sum((centers[i] - centers[j])**2)) - r[i] - r[j] - 1e-12})
        return cons

    def bounds(radii):
        return [(0, 1) for _ in radii]

    # Initial hexagonal packing with 26 circles
    n = 26
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        centers[i] = [x / (cols - 1), y / (rows - 1)]

    # Optimization
    radius_guess = np.ones(n) * 0.1
    cons = constraints(radius_guess, centers)
    res = minimize(
        objective,
        radius_guess,
        method='SLSQP',
        bounds=bounds(radius_guess),
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 1000}
    )

    radii = res.x
    return centers, radii, np.sum(radii)
