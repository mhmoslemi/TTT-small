import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial grid-based hexagonal packing with 5 rows
    rows = 5
    cols = 5
    centers = []
    radii = []

    # Base radius for hexagonal packing
    base_r = 0.15
    gap = 0.15

    # Initial hexagonal grid
    for i in range(rows):
        for j in range(cols):
            x = j * (2 * base_r + gap)
            y = i * (np.sqrt(3) * base_r + gap)
            if i % 2 == 1:
                x += base_r
            if x < 1 - 1e-8 and y < 1 - 1e-8:
                centers.append([x, y])
                radii.append(base_r)

    # Optimized boundary circles
    boundary_r = 0.15
    boundary_centers = [
        [0, 0.5], [1, 0.5], [0.5, 0], [0.5, 1],
        [0.25, 0.25], [0.75, 0.25], [0.25, 0.75], [0.75, 0.75],
        [0.125, 0.125], [0.875, 0.125], [0.125, 0.875], [0.875, 0.875]
    ]
    for x, y in boundary_centers:
        if x < 1 - 1e-8 and y < 1 - 1e-8 and x > 1e-8 and y > 1e-8:
            centers.append([x, y])
            radii.append(boundary_r)

    # Add small circles in remaining gaps
    gap_r = 0.05
    gap_centers = [
        [0.3, 0.8], [0.7, 0.8], [0.3, 0.2], [0.7, 0.2],
        [0.2, 0.6], [0.8, 0.6], [0.5, 0.3], [0.5, 0.7],
        [0.4, 0.4], [0.6, 0.4], [0.4, 0.6], [0.6, 0.6]

            centers.append([x, y])
            radii.append(gap_r)

    # Convert lists to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Check total number of circles
    if len(centers) != 26:
        raise ValueError("Packing must contain exactly 26 circles")

    # Initial optimization to refine positions and radii
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        constraints = []
        for i in range(len(radii)):
            constraints.append((radii[i], 0, np.inf))
        return constraints

    result = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=[(0, np.inf) for _ in range(len(radii))],
        constraints=[{"type": "ineq", "fun": lambda r, i=i: r[i] if i < len(r) else 0} for i in range(len(radii))]
    )

    radii = result.x

    # Final check to ensure no overlaps and within bounds
    for i in range(len(radii)):
        x, y = centers[i]
        if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12 or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
            raise ValueError("Circle out of bounds after optimization")

    for i in range(len(radii)):
        for j in range(i + 1, len(radii)):
            dist = np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2)
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError("Overlap detected after optimization")

    return centers, radii, np.sum(radii)
