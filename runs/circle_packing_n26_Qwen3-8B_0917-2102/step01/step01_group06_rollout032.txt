import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.cluster.vq import kmeans

def run_packing():
    np.random.seed(42)
    n = 26
    # Initial hexagonal packing with spacing
    rows = int(np.ceil(n / 2))
    cols = int(np.ceil(n / rows))
    centers = np.zeros((n, 2))
    r = 0.1
    for i in range(n):
        col = i % cols
        row = i // cols
        x = col + (row % 2) * (0.5 * r)
        y = row * np.sqrt(3) * r
        centers[i] = [x, y]
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraints(radii):
        constraints = []
        for i in range(n):
            constraints.append((radii[i] <= 1 - centers[i][0]))
            constraints.append((radii[i] <= 1 - centers[i][1]))
            constraints.append((radii[i] <= centers[i][0]))
            constraints.append((radii[i] <= centers[i][1]))
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                constraints.append((radii[i] + radii[j] <= dist))
        return constraints
    
    result = minimize(
        objective,
        x0=np.full(n, r),
        constraints=[{'type': 'ineq', 'fun': lambda x, i=i: x[i] for i in range(n)} for _ in range(n)] +
                   [({'type': 'ineq', 'fun': lambda x, i=j, j=i: x[i] + x[j] - np.sqrt(np.sum((centers[i] - centers[j])**2))} 
                     for i in range(n) for j in range(i + 1, n))],
        method='SLSQP'
    )
    
    radii = result.x
    sum_radii = np.sum(radii)
    
    return (centers, radii, sum_radii)
