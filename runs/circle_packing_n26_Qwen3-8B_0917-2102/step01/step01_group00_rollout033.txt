import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Initial hexagonal packing
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = 0.1
    
    # Normalize centers to [0,1]^2
    min_coords = np.min(centers, axis=0)
    max_coords = np.max(centers, axis=0)
    centers = (centers - min_coords) / (max_coords - min_coords)
    
    def objective(r):
        return -np.sum(r)
    
    # Constraints to ensure radii are positive and within bounds
    cons = [{'type': 'ineq', 'fun': lambda r: r - 1e-8}]
    
    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10})
    radii = result.x
    
    # Ensure circles are within the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0
    
    # Enforce non-overlapping constraint
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            dist = dists[i, j]
            if dist < radii[i] + radii[j] - 1e-8:
                # Adjust radii to maintain spacing
                ratio = (dist - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
