import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)

    def objective(r):
        return -np.sum(r)

    def constraint_func(r):
        return np.sum(r)

    cons = ({
        'type': 'ineq',
        'fun': constraint_func,
        'args': (),
        'jac': lambda r: np.ones(n)
    })

    bounds = Bounds(np.zeros(n), np.full(n, 1.0))

    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)

    if not result.success:
        raise RuntimeError("Optimization failed")

    radii = result.x

    # Hexagonal packing with boundary-aware positioning
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + (row % 2) * 0.5
        y = row * (1 / 2.5)
        centers[i] = [x, y]

    # Ensure centers are within [0,1]^2
    centers = np.clip(centers, 0, 1)

    # Adjust radii to fit boundaries
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        max_dist = np.max([x, 1 - x, y, 1 - y])
        radii[i] = np.min([radii[i], max_dist])

    # Iterative non-overlapping adjustment
    for _ in range(10):
        dist_matrix = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-12:
                    delta = (dist_matrix[i, j] - 1e-12) / 2
                    radii[i] = np.max([0, radii[i] - delta])
                    radii[j] = np.max([0, radii[j] - delta])

    return centers, radii, np.sum(radii)
