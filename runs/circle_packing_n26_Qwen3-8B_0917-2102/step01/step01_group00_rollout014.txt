import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.empty((n, 2))
    radii = np.zeros(n)
    initial_r = 0.15
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = initial_r

    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(x):
        return -np.sum(x)

    def constraint(x):
        return np.sum(x) - 1.5

    cons = [{'type': 'ineq', 'fun': constraint}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', tol=1e-8)

    # Update radii based on distances
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            d = dists[i, j]
            r1 = result.x[i]
            r2 = result.x[j]
            if d < r1 + r2:
                ratio = (d - 1e-8) / (r1 + r2)
                result.x[i] *= ratio
                result.x[j] *= ratio

    # Final distance-based adjustment
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            d = dists[i, j]
            r1 = result.x[i]
            r2 = result.x[j]
            if d < r1 + r2:
                ratio = (d - 1e-8) / (r1 + r2)
                result.x[i] *= ratio
                result.x[j] *= ratio

    # Ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            result.x[i] = 0.0

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
