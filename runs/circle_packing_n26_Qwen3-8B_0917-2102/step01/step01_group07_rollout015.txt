import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints_func(radii):
        constraints = []
        for i in range(26):
            constraints.append((radii[i], 0, np.inf))
        return constraints

    centers = []
    radii = []

    # Hexagonal grid with 5 rows
    for row in range(5):
        for col in range(5):
            x = col + 0.5 * row
            y = row + 0.5 * row
            if x < 1 and y < 1:
                centers.append([x, y])
                radii.append(0.05)

    # Add boundary circles
    for i in range(len(centers)):
        x, y = centers[i]
        if x < 0.1:
            centers.append([0.1, y])
            radii.append(0.1)
        if x > 0.9:
            centers.append([0.9, y])
            radii.append(0.1)
        if y < 0.1:
            centers.append([x, 0.1])
            radii.append(0.1)
        if y > 0.9:
            centers.append([x, 0.9])
            radii.append(0.1)

    # Optimize
    centers = np.array(centers)
    radii = np.array(radii)

    def constraint_func(radii):
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        for i in range(len(radii)):
            x, y = centers[i]
            if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
                return False
        return True

    result = minimize(objective, radii, constraints=[{'type': 'ineq', 'fun': constraint_func}])

    return (centers, result.x, np.sum(result.x))
