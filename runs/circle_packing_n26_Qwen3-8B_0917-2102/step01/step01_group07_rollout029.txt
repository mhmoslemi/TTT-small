import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    
    # Initial guess for centers using hexagonal packing
    centers = np.zeros((n, 2))
    r = np.zeros(n)
    
    # Hexagonal grid with decreasing radii for tighter packing
    for i in range(n):
        row = i // 6
        col = i % 6
        x = 0.5 + 0.5 * (col - 2.5)
        y = 0.5 + 0.5 * (row - 2.5)
        centers[i] = [x, y]
        r[i] = 0.1 + 0.05 * (n - i - 1)
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraint_func(radii):
        dist_matrix = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return 1.0
        return 0.0
    
    # Optimization with bounds and constraints
    bounds = [(0, 0.5) for _ in range(n)]
    
    res = minimize(
        objective,
        r,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraint_func}]
    )
    
    radii = res.x
    sum_radii = np.sum(radii)
    
    # Final validation
    centers = np.array(centers)
    for i in range(n):
        x, y = centers[i]
        if (x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12
                or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12):
            raise ValueError("Circle outside unit square")
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError("Circles overlap")
    
    return centers, radii, sum_radii
