        sum_radii = np.sum(radii)
        return -sum_radii  # Minimize negative sum

    def constraints_func(params):
        centers_flat = params[:2 * n].reshape((n, 2))
        radii_flat = params[2 * n:]
        radii = np.clip(radii_flat, 0, 1)

        # Boundary constraints
        for i in range(n):
            x, y = centers_flat[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return 1.0

        # Pairwise distance constraints
        for (i, j) in combinations(range(n), 2):
            dx = centers_flat[i, 0] - centers_flat[j, 0]

    initial_params = centers.reshape(-1).tolist() + radii.tolist()

        initial_params,
        method='SLSQP',
        bounds=[
            (0, 1) for _ in range(2 * n)

        ],
        constraints=[
            {
                'type': 'ineq',
                'fun': constraints_func,
                'jac': None
            }
        ],
        options={'ftol': 1e-8, 'disp': False}
    )
