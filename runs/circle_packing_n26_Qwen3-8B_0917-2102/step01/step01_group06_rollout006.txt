from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    n = 26
    initial_centers = np.random.rand(n, 2)
    initial_radii = np.random.rand(n) * 0.1

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        return np.array([radii[i] for i in range(n)]) - np.random.rand(n) * 0.1

    def non_overlap_constraints(radii):
        centers = np.random.rand(n, 2)
        dists = squareform(pdist(centers))
        return np.array([dists[i, j] - (radii[i] + radii[j]) for i in range(n) for j in range(i + 1)])

    bounds = [(0, 0.5) for _ in range(n)]
    result = minimize(objective, initial_radii, bounds=bounds, constraints=non_overlap_constraints)

    optimal_radii = result.x
    optimal_centers = np.random.rand(n, 2)

    return optimal_centers, optimal_radii, np.sum(optimal_radii)
