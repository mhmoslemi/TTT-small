def run_packing():

    N = 26

    # Initial hexagonal grid parameters
    rows = int(np.sqrt(N))
    cols = (N + rows - 1) // rows
    r = 0.1  # Initial radius

    # Generate initial hexagonal packing
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j + 0.5 * (i % 2)
            y = i + 0.5
            if i % 2 == 0:
                x += 0.5
            centers.append([x, y])
    centers = np.array(centers)
    radii = np.full(N, r)

    # Adjust centers to fit in [0,1]^2
    scale = 1.0 / (np.max(centers, axis=0) - np.min(centers, axis=0))
    centers = (centers - np.min(centers, axis=0)) * scale
    radii *= np.min(scale)

    # Refine with optimization
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        for i in range(N):
            for j in range(i+1, N):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                if np.sqrt(dx*dx + dy*dy) < radii[i] + radii[j] - 1e-8:
                    return False
        for i in range(N):
            if centers[i, 0] - radii[i] < -1e-8 or centers[i, 0] + radii[i] > 1 + 1e-8:
                return False
            if centers[i, 1] - radii[i] < -1e-8 or centers[i, 1] + radii[i] > 1 + 1e-8:
                return False
        return True

    result = minimize(objective, radii, bounds=[(0, 1)]*N, constraints=[{"type": "ineq", "fun": constraints}])

    centers = np.clip(centers, 0, 1)

    for i in range(N):

            raise ValueError("Circle out of bounds")

    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
