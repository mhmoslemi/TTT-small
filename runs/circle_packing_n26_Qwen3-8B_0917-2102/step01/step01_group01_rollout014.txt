import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraint_func(x, centers, radii):
        centers_flat = x.reshape(-1, 2)
        dist_matrix = squareform(pdist(centers_flat))
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return 1
        return 0

    n = 26
    centers = []
    radii = []

    # Initial hexagonal packing with variable radii
    for i in range(n):
        row = i // 4
        col = i % 4
        x = col / 4.0 + 0.05
        y = row / 4.0 + 0.05
        r = 0.05
        centers.append([x, y])
        radii.append(r)

    centers = np.array(centers)
    radii = np.array(radii)

    # Add some circles near boundaries
    for i in range(n):
        if i % 5 == 0:
            x = np.random.uniform(0.1, 0.9)
            y = np.random.uniform(0.1, 0.9)
            r = 0.05
            centers = np.vstack((centers, [x, y]))
            radii = np.append(radii, r)

    centers = centers[:26]
    radii = radii[:26]

    # Initial optimization to refine positions
    def objective_func(x):
        centers_flat = x.reshape(-1, 2)
        return -np.sum(radii)

    x0 = centers.reshape(-1)
    result = minimize(objective_func, x0, constraints={"type": "ineq", "fun": lambda x: constraint_func(x, centers, radii)})
    centers = result.x.reshape(-1, 2)

    # Adjust radii to fit within the square and avoid overlaps
    def validate_and_refine(centers, radii):
        for i in range(len(radii)):
            x, y = centers[i]
            r = radii[i]
            if (x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1):
                radii[i] = max(0, min(r, 1 - max(x, 1 - x, y, 1 - y)))
        dist_matrix = squareform(pdist(centers))
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    radii[i] = max(0, (dist_matrix[i, j] - radii[j]) * 0.5)
                    radii[j] = max(0, (dist_matrix[i, j] - radii[i]) * 0.5)
        return centers, radii

    centers, radii = validate_and_refine(centers, radii)

    # Final optimization with boundary constraints
    def boundary_constraint(x):
        centers_flat = x.reshape(-1, 2)
        for i in range(len(radii)):
            x, y = centers_flat[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return 1
        return 0

    result = minimize(objective_func, centers.reshape(-1), constraints={"type": "ineq", "fun": boundary_constraint})
    centers = result.x.reshape(-1, 2)

    # Final validation
    def final_validate(centers, radii):
        for i in range(len(radii)):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                raise ValueError(f"Circle {i} at ({x},{y}) r={r} outside unit square")
        dist_matrix = squareform(pdist(centers))
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-12:
                    raise ValueError(f"Circles {i} and {j} overlap")
        return True

    final_validate(centers, radii)

    return centers, radii, np.sum(radii)
