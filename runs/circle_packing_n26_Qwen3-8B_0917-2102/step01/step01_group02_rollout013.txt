import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import LinearConstraint

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Initial hexagonal packing
    row = []
    for i in range(n):
        col = i // 5
        row_idx = i % 5
        x = (row_idx + 0.5) / 5
        y = col / (np.sqrt(3) * 1)
        row.append([x, y])
    centers = np.array(row)

    # Initial radius guess based on spacing
    def initial_radius(i):
        if i < 5:
            return 0.1
        return 0.08

    radii = np.array([initial_radius(i) for i in range(n)])

    def objective(r):
        return -np.sum(r)

    def constraint_func(r):
        dists = squareform(pdist(centers))
        min_dist = np.min(dists)
        return np.sum(r) - (min_dist - 1e-12)

    cons = ({
        'type': 'ineq',
        'fun': constraint_func
    })

    bounds = Bounds(np.zeros(n), np.full(n, 1.0))

    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)

    if not result.success:
        raise RuntimeError("Optimization failed")

    radii = result.x

    # Final validation and adjustment
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = np.max([0, 1 - np.max([x, 1 - x, y, 1 - y])])

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                radii[i] = (dist - 1e-12) / 2
                radii[j] = (dist - 1e-12) / 2

    return centers, radii, np.sum(radii)
