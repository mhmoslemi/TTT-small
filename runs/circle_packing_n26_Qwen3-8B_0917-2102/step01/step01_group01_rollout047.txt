from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

    radii = []

        col = i % 5
        x = col / 4.0 + 0.05
        y = row / 4.0 + 0.05
        r = 0.05
        centers.append([x, y])
        radii.append(r)

    centers = np.array(centers)
    radii = np.array(radii)

    # Add extra circles near boundaries for optimization
    max_iter = 100

        # Move closest pair to boundary

        x2, y2 = centers[j]
        dx = x2 - x1
        dy = y2 - y1
        dist = np.sqrt(dx*dx + dy*dy)
        if dist < 1e-8:
            continue
        dir_x = dx / dist
        dir_y = dy / dist
        new_x = x1 + dir_x * 0.1
        new_y = y1 + dir_y * 0.1
        if new_x < 0:
            new_x = 0
        elif new_x > 1:
            new_x = 1
        if new_y < 0:
            new_y = 0
        elif new_y > 1:
            new_y = 1

        dist_matrix = squareform(pdist(centers_flat))

                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:

        return 0

                radii[i] = (dist + radii[j]) * 0.5
                radii[j] = (dist + radii[i]) * 0.5
