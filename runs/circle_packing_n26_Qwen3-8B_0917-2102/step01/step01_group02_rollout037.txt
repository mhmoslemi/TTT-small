import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import LinearConstraint
import random

def run_packing():
    n = 26
    # Initial hexagonal packing with small perturbation
    centers = []
    for i in range(n):
        row = i // 2
        col = i % 2
        x = 0.5 + (col - 0.5) * (1 / np.sqrt(3))
        y = row / np.sqrt(3)
        centers.append([x, y])
    centers = np.array(centers)
    centers += np.random.normal(0, 0.01, centers.shape)
    centers = np.clip(centers, 0, 1)

    radii = np.array([np.min([x, 1 - x, y, 1 - y]) for x, y in centers])

    def objective(r):
        return -np.sum(r)
    
    # Constraint to ensure no overlap between circles
    def constraint_func(r):
        dist_matrix = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return 0
    
    # Optimization bounds
    bounds = Bounds(np.zeros(n), np.full(n, 1.0))
    
    # Initial guess with some radius adjustment
    initial_guess = radii.copy()
    
    # Run optimization
    result = minimize(objective, initial_guess, method='SLSQP', bounds=bounds, constraints=({'type': 'ineq', 'fun': constraint_func}))
    
    if not result.success:
        raise RuntimeError("Optimization failed")
    
    radii = result.x
    
    # Final validation and adjustment
    for i in range(n):
        x, y = centers[i]

            if dist < radii[i] + radii[j] - 1e-12:
                radii[i] = (dist - 1e-12) / 2
                radii[j] = (dist - 1e-12) / 2
    
    return centers, radii, np.sum(radii)
