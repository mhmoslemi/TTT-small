def run_packing():

    initial_centers = np.random.rand(n, 2)
    initial_radii = np.random.rand(n) * 0.05

    def objective(r):
        return -np.sum(r)

    def constraints(r, centers):
        cons = []
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] * 2 - 1e-12})
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: dist - r[i] - r[j] - 1e-12})
        return cons

    def bounds_func(r):
        return [(0, 1) for _ in range(n)]

    result = minimize(objective, initial_radii, method='SLSQP', jac=None, bounds=bounds_func, constraints=constraints(initial_radii, initial_centers))
    radii = result.x
    centers = initial_centers

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            dist = np.linalg.norm(centers[i] - np.array([0, 0]))
            radii[i] = min(r, dist - 1e-12)
            dist = np.linalg.norm(centers[i] - np.array([1, 1]))
            radii[i] = min(radii[i], dist - 1e-12)

    return centers, radii, np.sum(radii)
