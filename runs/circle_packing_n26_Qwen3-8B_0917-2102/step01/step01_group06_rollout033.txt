import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    # Initial hexagonal packing with boundary circles
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    
    # First layer: 12 circles in hexagonal grid along the bottom
    for i in range(12):
        x = i * 0.5 + 0.25
        y = 0.1
        centers[i] = [x, y]
        radii[i] = 0.1
    
    # Second layer: 12 circles above the first layer
    for i in range(12):
        x = i * 0.5 + 0.25
        y = 0.3
        centers[12 + i] = [x, y]
        radii[12 + i] = 0.1
    
    # Third layer: 2 circles near the top
    centers[24] = [0.5, 0.85]
    radii[24] = 0.1
    centers[25] = [0.5, 0.95]
    radii[25] = 0.05
    
    # Optimize radii and positions
    def objective(r):
        return -np.sum(r)
    
    def constraints(r):
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < r[i] + r[j] - 1e-8:
                    return False
        for i in range(26):
            x, y = centers[i]
            if x - r[i] < 0 or x + r[i] > 1 or y - r[i] < 0 or y + r[i] > 1:
                return False
        return True
    
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=[(0, 0.5) for _ in range(26)],
        constraints=[{'type': 'ineq', 'fun': constraints}]
    )
    
    return (centers, result.x, np.sum(result.x))
