import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Hexagonal packing configuration with 26 circles
    rows = 6
    cols = 5
    spacing = 1 / (rows + 1)
    offsets = np.array([0, spacing / 2, spacing, spacing / 2, 0, spacing / 2])
    offsets = offsets.reshape(-1, 1)

    centers = np.empty((26, 2))
    radii = np.empty(26)
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + spacing / 2
            y = i * spacing + spacing / 2
            # Adjust y for odd rows to create staggered pattern
            if i % 2 == 1:
                y += spacing / 2
            centers[i * cols + j] = [x, y]
            radii[i * cols + j] = 0.1

    # Initial sum of radii
    sum_radii = np.sum(radii)

    # Optimization function
    def objective(radii):
        nonlocal sum_radii
        sum_radii = np.sum(radii)
        return -sum_radii

    def constraint1(radii):
        for i in range(26):
            r = radii[i]
            if r < 0:
                return -1
        return 0

    def constraint2(radii):
        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
                return -1
        return 0

    def constraint3(radii):
        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0

    # Optimization bounds
    bounds = [(0, 0.5) for _ in range(26)]

    # Optimization
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2},
            {'type': 'ineq', 'fun': constraint3}
        ],
        method='SLSQP'
    )

    # Return the result
    return (centers, result.x, sum_radii)
