from scipy.optimize import minimize

def run_packing():
    def objective(r):
        return -np.sum(r)  # Minimize negative sum to maximize sum of radii

    def constraints(r):
        return np.sum(r) - 2.636  # Ensure sum does not exceed target

    def bounds():
        return [(0, 1) for _ in range(26)]  # All radii in [0, 1]

    # Initial guess for radii (equal radii for symmetry)
    r0 = np.full(26, 0.1)

    # Optimization
    result = minimize(
        objective,
        r0,
        bounds=bounds(),
        constraints={"type": "ineq", "fun": constraints},
        method="SLSQP"
    )

    # Generate initial circle centers using hexagonal packing
    centers = []
    for i in range(26):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (i % 2))
        y = row + 0.5 * (1 - 2 * (i // 2 % 2))
        centers.append([x / 10, y / 10])

    centers = np.array(centers)
    radii = result.x

    # Check if we need to adjust radii due to boundary constraints
    for i in range(26):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            radii[i] = min(r, 1 - max(x, 1 - x, y, 1 - y))

    return centers, radii, np.sum(radii)
