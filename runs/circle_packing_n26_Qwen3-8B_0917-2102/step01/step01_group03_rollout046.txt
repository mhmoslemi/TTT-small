<strategy>Use hexagonal packing with adaptive resizing to fill gaps and maximize total radius sum.</strategy>

```python
import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    np.random.seed(42)
    n = 26

    # Initial hexagonal lattice
    def hex_grid(n):
        r = 0.1
        angles = np.linspace(0, 2 * np.pi, n, endpoint=False)
        centers = np.zeros((n, 2))
        for i in range(n):
            theta = angles[i]
            centers[i] = [r * (1 + np.cos(theta)), r * np.sin(theta)]
        return centers

    centers = hex_grid(n)
    radii = np.full(n, 0.1)

    # Objective function to maximize sum of radii
    def objective(r):
        return -np.sum(r)  # Minimize negative sum

    # Constraints: circles don't overlap and stay within square
    def constraint_overlap(r, c):
        dists = squareform(pdist(c))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] + 1e-8:
                    return -1
        return 0

    def constraint_bounds(r, c):
        for i in range(n):
            if c[i, 0] - r[i] < 0 or c[i, 0] + r[i] > 1 or c[i, 1] - r[i] < 0 or c[i, 1] + r[i] > 1:
                return -1
        return 0

    cons = (
        {'type': 'ineq', 'fun': lambda r: constraint_overlap(r, centers)},
        {'type': 'ineq', 'fun': lambda r: constraint_bounds(r, centers)},
    )

    # Optimize
    res = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=[(0, 1)] * n,
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 1000}
    )

    # Final result
    return (centers, res.x, np.sum(res.x))
```
