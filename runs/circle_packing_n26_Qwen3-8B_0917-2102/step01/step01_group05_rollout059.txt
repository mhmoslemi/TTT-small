import numpy as np
from scipy.optimize import minimize

def run_packing():
    def objective(radii):
        return -np.sum(radii)
    
    def constraint_overlap(i, j):
        def cons(x):
            dx = x[2*i] - x[2*j]
            dy = x[2*i+1] - x[2*j+1]
            dist = np.sqrt(dx*dx + dy*dy)
            return radii[i] + radii[j] - dist
        return cons
    
    def constraint_boundary(i):
        def cons(x):
            x_i, y_i = x[2*i], x[2*i+1]
            r_i = radii[i]
            return np.array([x_i - r_i, x_i + r_i - 1, y_i - r_i, 1 - y_i - r_i])
        return cons
    
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.1)
    
    bounds = [(0, 1) for _ in range(2*n)]
    constraints = []
    
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': constraint_boundary(i)})
    
    for i in range(n):
        for j in range(i+1, n):
            constraints.append({'type': 'ineq', 'fun': constraint_overlap(i, j)})
    
    result = minimize(objective, np.concatenate((centers.reshape(-1), radii)),
                     bounds=bounds,
                     constraints=constraints)
    
    centers = result.x[:2*n].reshape(n, 2)
    radii = result.x[2*n:]
    
    return centers, radii, np.sum(radii)
