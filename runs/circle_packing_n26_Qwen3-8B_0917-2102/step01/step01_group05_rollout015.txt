import numpy as np
from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d

def run_packing():
    np.random.seed(42)
    n = 26

    # Initial hexagonal grid with spacing
    grid = np.linspace(0.1, 0.9, 5)
    x = np.repeat(grid, len(grid))
    y = np.tile(grid, len(grid))

    initial_centers = np.column_stack((x, y))

    # Initial radii based on spacing
    r = np.full(n, 0.05)

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        return np.array([np.sum(radii) - 2.636])

    def non_overlap(radii):
        dist = np.sum((initial_centers[:, np.newaxis, :] - initial_centers[np.newaxis, :])**2, axis=-1)**0.5
        return np.sum(np.abs(dist - (radii[:, np.newaxis] + radii[np.newaxis, :])))

    bounds = [(0, 0.1)] * n

    result = minimize(objective, r, bounds=bounds, method='SLSQP', constraints=[{'type': 'ineq', 'fun': lambda x: np.sum(x) - 2.636}])

    radii = result.x
    centers = initial_centers

    # Adjust for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            radii[i] = x
        elif x + r > 1:
            radii[i] = 1 - x
        if y - r < 0:
            radii[i] = y
        elif y + r > 1:
            radii[i] = 1 - y

    return centers, radii, np.sum(radii)
