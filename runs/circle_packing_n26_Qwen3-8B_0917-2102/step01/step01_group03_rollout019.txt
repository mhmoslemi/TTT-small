import numpy as np
from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d

def run_packing():
    # Initial guess: hexagonal grid with 5 rows
    centers = []
    radii = []
    for row in range(5):
        for col in range(5):
            x = col + 0.5 * (row % 2)
            y = row + 0.1
            if x < 0 or x > 1 or y < 0 or y > 1:
                continue
            centers.append([x, y])
            radii.append(0.05)
    centers = np.array(centers)
    radii = np.array(radii)
    
    def objective(r):
        return -np.sum(r)
    
    def constraint(r, i, j):
        return np.sqrt(np.sum((centers[i] - centers[j])**2)) - (r[i] + r[j])
    
    cons = []
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: constraint(r, i, j)})
    
    result = minimize(objective, radii, constraints=cons, bounds=[(0, 0.5)]*len(radii))
    return centers, result.x, np.sum(result.x)
