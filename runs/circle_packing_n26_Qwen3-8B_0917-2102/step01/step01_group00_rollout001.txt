import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Generate hexagonal grid with spacing adjusted for boundary constraints
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
    
    # Normalize centers to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))
    
    # Initial radii based on spacing
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) / 2
    radii = initial_r * np.ones(n)
    
    # Objective function to maximize sum of radii
    def objective(x):
        return -np.sum(x)
    
    # Constraints: radii must be non-negative
    def constraint(x):
        return np.sum(x) - 1e-8
    
    cons = [{'type': 'ineq', 'fun': constraint}]
    
    # Optimization with bounds and constraints
    bounds = [(0, 1) for _ in range(n)]
    result = minimize(objective, radii, bounds=bounds, constraints=cons, method='SLSQP')
    
    radii = result.x
    
    # Check and adjust for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0
    
    # Ensure no overlap between circles
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
