import numpy as np
from scipy.optimize import minimize

def run_packing():
    np.random.seed(42)
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)
    max_iter = 100
    tol = 1e-6

    def objective(x):
        nonlocal centers, radii
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        return -np.sum(radii)

    def constraint(x):
        nonlocal centers, radii
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return -1
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0

    bounds = []
    for _ in range(n):
        bounds.extend([(0, 1)])
        bounds.extend([(0, 1)])
    for _ in range(n):
        bounds.extend([(0, 0.1)])  # Radii upper bound

    result = minimize(objective, x0=np.concatenate([centers.flatten(), np.full(n, 0.1)]),
                      method='SLSQP',
                      bounds=bounds,
                      constraints={'type': 'ineq', 'fun': constraint, 'jac': None},
                      options={'maxiter': max_iter, 'ftol': tol})

    if result.success:
        centers = result.x[:2*n].reshape(n, 2)
        radii = result.x[2*n:]
        sum_radii = np.sum(radii)
        return (centers, radii, sum_radii)
    else:
        raise ValueError("Optimization failed")
