from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.cluster.hierarchy import linkage, fcluster

def run_packing():

    centers = []
    radii = []

    # Create a hexagonal packing with 26 circles
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col / 5.0 + 0.1
        y = row / 5.0 + 0.1
        r = 0.05
        centers.append([x, y])
        radii.append(r)

    centers = np.array(centers)
    radii = np.array(radii)

    # Initial refinement to bring circles closer to boundaries
    for _ in range(5):
        dist_matrix = squareform(pdist(centers))
        idx = np.argmin(dist_matrix)

    def objective(radii):
        return -np.sum(radii)

    def constraint(x):
        centers_flat = x.reshape(-1, 2)
        dist_matrix = squareform(pdist(centers_flat))
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):

                    return 1
        return 0

    result = minimize(objective, x0, constraints={"type": "ineq", "fun": constraint}, method='SLSQP')

    # Post-optimization refinement

        x, y = optimized_centers[i]
