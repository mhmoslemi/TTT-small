import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(centers, radii):
        cons = []
        for i in range(len(radii)):
            x, y = centers[i]
            r = radii[i]
            cons.append((x - r, -1e-12))
            cons.append((x + r, 1 + 1e-12))
            cons.append((y - r, -1e-12))
            cons.append((y + r, 1 + 1e-12))
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                cons.append((dist - radii[i] - radii[j], -1e-12))
        return cons

    def pack():
        n = 26
        centers = np.random.rand(n, 2)
        radii = np.random.rand(n) * 0.1

        def cost(r):
            return -np.sum(r)

        def ineq_cons(r):
            cons = []
            for i in range(n):
                cons.append((centers[i, 0] - r[i], -1e-12))
                cons.append((centers[i, 0] + r[i], 1 + 1e-12))
                cons.append((centers[i, 1] - r[i], -1e-12))
                cons.append((centers[i, 1] + r[i], 1 + 1e-12))
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                    cons.append((dist - r[i] - r[j], -1e-12))
            return cons

        res = minimize(cost, radii, method='SLSQP', bounds=[(0, 1) for _ in range(n)], constraints={'type': 'ineq', 'fun': ineq_cons})

    best_sum = 0
    best_centers = np.zeros((26, 2))
    best_radii = np.zeros(26)

    for _ in range(10):
        centers, radii = pack()
        sum_r = np.sum(radii)
        if sum_r > best_sum:
            best_sum = sum_r
            best_centers = centers
            best_radii = radii

    return best_centers, best_radii, best_sum
