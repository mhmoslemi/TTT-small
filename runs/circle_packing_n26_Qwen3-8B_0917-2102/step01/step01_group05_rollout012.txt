<strategy>Use hexagonal packing with staggered rows near boundaries to maximize circle sizes and fill gaps efficiently.</strategy>

```python
import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        return np.sum(radii) - 2.636

    def non_overlap(centers, radii):
        dists = squareform(pdist(centers))
        return np.sum((dists - (radii[:, np.newaxis] + radii[np.newaxis, :])) < 0)

    def boundary_check(centers, radii):
        return np.sum((centers[:, 0] - radii < 0) | (centers[:, 0] + radii > 1) |
                     (centers[:, 1] - radii < 0) | (centers[:, 1] + radii > 1))

    def initial_config():
        rows = 5
        cols = 5
        centers = []
        radii = []
        for i in range(rows):
            for j in range(cols):
                x = j / (cols - 1)
                y = i / (rows - 1)
                centers.append([x, y])
                radii.append(0.05)
        return np.array(centers), np.array(radii)

    centers, radii = initial_config()
    bounds = Bounds([0.0] * len(radii), [1.0] * len(radii))

    result = minimize(objective, radii, method='SLSQP', bounds=bounds,
                      constraints={'type': 'ineq', 'fun': lambda x: 2.636 - np.sum(x)},
                      options={'maxiter': 500, 'disp': False},
                      tol=1e-6)

    final_centers = centers
    final_radii = result.x

    if non_overlap(final_centers, final_radii) > 0 or boundary_check(final_centers, final_radii) > 0:
        raise ValueError("Invalid packing")

    return final_centers, final_radii, np.sum(final_radii)
```
