import numpy as np
import scipy.optimize as opt
from scipy.spatial import Voronoi, voronoi_plot_2d
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    # Initial hexagonal packing with 26 circles
    num_circles = 26
    centers = []
    radii = []

    # Initial hexagonal grid with spacing
    spacing = 0.55
    rows = int(np.ceil(np.sqrt(num_circles)))
    cols = int(np.ceil(num_circles / rows))
    
    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * spacing * np.sqrt(3) / 2
            if x < 1 and y < 1:
                centers.append([x, y])
    
    centers = np.array(centers)
    radii = np.full(len(centers), 0.05)

    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum

    # Constraints: circles within square, non-overlapping
    def constraint_within_square(radii):
        return np.sum((centers[:, 0] - radii) >= 0) + np.sum((1 - centers[:, 0] - radii) >= 0) + np.sum((centers[:, 1] - radii) >= 0) + np.sum((1 - centers[:, 1] - radii) >= 0)

    def constraint_non_overlapping(radii):
        dists = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :])**2, axis=2))
        return np.sum(dists < radii[:, np.newaxis] + radii[np.newaxis, :] - 1e-8)

    # Optimization
    bounds = [(0, 0.5) for _ in range(num_circles)]
    constraints = [{'type': 'ineq', 'fun': lambda r: constraint_within_square(r)},
                  {'type': 'ineq', 'fun': lambda r: constraint_non_overlapping(r)}]

    result = minimize(objective, radii, bounds=bounds, constraints=constraints, method='SLSQP')

    if not result.success:
        raise ValueError("Optimization failed")

    best_radii = result.x
    best_centers = centers

    return best_centers, best_radii, np.sum(best_radii)
