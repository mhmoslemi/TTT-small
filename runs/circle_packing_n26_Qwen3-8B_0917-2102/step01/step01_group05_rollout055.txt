from scipy.optimize import minimize

def run_packing():

                y = row - 0.5

            centers.append([x, y])

    radii = np.full(len(centers), 0.1)

    def objective(r):
        return -np.sum(r)

        return np.sum(r)

    # Optimization

    # Check and adjust for boundary conditions
