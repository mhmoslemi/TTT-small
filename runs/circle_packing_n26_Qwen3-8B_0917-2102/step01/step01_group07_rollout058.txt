from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():

        return -np.sum(radii)

    def constraints(centers, radii):
        cons = []
        for i in range(26):
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[i] - radii[i]})
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[i]})
        for i in range(26):

                cons.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: np.sqrt(np.sum((centers[i] - centers[j])**2)) - (radii[i] + radii[j])})

    initial_radii = np.random.rand(26) * 0.1

    result = minimize(
        objective,
        np.concatenate((initial_centers.flatten(), initial_radii)),
        bounds=Bounds(np.concatenate((np.zeros(52), np.zeros(26))), np.concatenate((np.ones(52), np.ones(26)))),
        constraints=constraints(initial_centers, initial_radii),
        method='SLSQP'
    )

    centers = result.x[:52].reshape(26, 2)
    radii = result.x[52:]
    return centers, radii, np.sum(radii)
