import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial import distance
from itertools import combinations

def run_packing():
    # Initial hexagonal packing with 26 circles
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 1/5)
        y = row + 0.5 * (1 - 1/5)
        centers[i] = [x, y]
        radii[i] = 0.1

    # Optimization parameters
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        c = []
        for i in range(n):
            c.append((radii[i], 0, np.inf))
        for i, j in combinations(range(n), 2):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            c.append((radii[i] + radii[j] - dist, -np.inf, 0))
        for i in range(n):
            x, y = centers[i]
            c.append((radii[i], 0, 1 - x))
            c.append((radii[i], 0, 1 - y))
            c.append((radii[i], 0, x))
            c.append((radii[i], 0, y))
        return c

    bounds = Bounds(np.zeros(n), np.inf * np.ones(n))
    res = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=constraints)
    radii = res.x
    sum_radii = np.sum(radii)
    centers = np.array([centers[i] for i in range(n)])
    return (centers, radii, sum_radii)
