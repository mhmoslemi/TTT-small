def run_packing():

    initial_radii = np.full(n, 0.05)

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        constraints = []
        for i in range(n):
            constraints.append((initial_centers[i][0] - radii[i], 0, 1, 0))
            constraints.append((initial_centers[i][1] - radii[i], 0, 1, 0))
            constraints.append((initial_centers[i][0] + radii[i], 1, 1, 0))
            constraints.append((initial_centers[i][1] + radii[i], 1, 1, 0))
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((initial_centers[i] - initial_centers[j]) ** 2))
                constraints.append((dist - radii[i] - radii[j], 0, 1, 0))
        return constraints

    bounds = [(0, 1) for _ in range(n)]

    result = minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP'
    )

    radii = result.x
    centers = np.zeros((n, 2))
    for i in range(n):
        if radii[i] > 1e-6:
            centers[i] = initial_centers[i]

    return (centers, radii, np.sum(radii))
