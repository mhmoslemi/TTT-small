import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    n = 26
    centers = []
    radii = []

    # Initial hexagonal grid with row-wise spacing
    rows = 5
    cols = 6
    for r in range(rows):
        for c in range(cols):
            x = c / (cols - 1) if cols > 1 else 0.5
            y = r / (rows - 1) if rows > 1 else 0.5
            r_val = 0.05
            centers.append([x, y])
            radii.append(r_val)

    # Add extra circles for boundary optimization
    for i in range(n - len(centers)):
        centers.append([0.5, 0.5])
        radii.append(0.05)

    # Function to check if a circle is within bounds
    def is_in_bounds(x, y, r):
        return (x - r >= 0 and x + r <= 1 and
                y - r >= 0 and y + r <= 1)

    # Function to check if two circles overlap
    def circles_overlap(c1, c2, r1, r2):
        dx = c1[0] - c2[0]
        dy = c1[1] - c2[1]
        dist_sq = dx**2 + dy**2
        return np.sqrt(dist_sq) < r1 + r2 - 1e-8

    # Objective function for optimization
    def objective(x):
        centers_flat = x.reshape(-1, 2)
        return -np.sum(np.clip(radii, 0, 1))

    # Constraint function for non-overlapping circles
    def constraint(x):
        centers_flat = x.reshape(-1, 2)
        for i in range(len(centers_flat)):
            for j in range(i + 1, len(centers_flat)):
                if circles_overlap(centers_flat[i], centers_flat[j], radii[i], radii[j]):
                    return 1
        return 0

    # Optimize using scipy.optimize.minimize with bounds
    bounds = [(0, 1) for _ in range(2 * n)]
    result = minimize(objective, np.array(centers).flatten(), bounds=bounds, constraints={"type": "ineq", "fun": constraint}, method='SLSQP', options={'maxiter': 1000, 'ftol': 1e-10, 'eps': 1e-8})

    # Extract optimized centers and radii
    centers_opt = result.x.reshape(-1, 2)
    radii_opt = np.clip(radii, 0, 1)

    # Final validation and refinement
    for i in range(n):
        x, y = centers_opt[i]
        r = radii_opt[i]
        if not is_in_bounds(x, y, r):
            radii_opt[i] = max(0, min(r, 1 - max(x, 1 - x, y, 1 - y)))

    dist_matrix = squareform(pdist(centers_opt))
    for i in range(n):
        for j in range(i + 1, n):
            if dist_matrix[i, j] < radii_opt[i] + radii_opt[j] - 1e-8:
                radii_opt[i] = max(0, (dist_matrix[i, j] - radii_opt[j]) * 0.5)
                radii_opt[j] = max(0, (dist_matrix[i, j] - radii_opt[i]) * 0.5)

    return centers_opt, radii_opt, np.sum(radii_opt)
