import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist

def run_packing():
    # Initial guess: 25 small circles in hexagonal pattern and 1 large circle
    centers = np.random.rand(26, 2)
    radii = np.random.rand(26) * 0.05 + 0.01

    def objective(r):
        return -np.sum(r)

    def constraints(r):
        return np.array([r[i] for i in range(26)])

    def bounds():
        return [(0, 0.1) for _ in range(26)]

    def packing_func(x):
        centers[:, 0] = x[::2]
        centers[:, 1] = x[1::2]
        radii = np.clip(x[26:], 0, 0.1)
        return centers, radii

    def constraint_func(x):
        centers, radii = packing_func(x)
        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        for i in range(26):
            x, y = centers[i]
            if (x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8 or
                y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8):
                return -1
        return 0

    x0 = np.concatenate([centers.flatten(), radii])
    res = minimize(objective, x0, method='SLSQPR', bounds=bounds(), constraints={'type': 'ineq', 'fun': constraint_func})
    centers, radii = packing_func(res.x)
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
