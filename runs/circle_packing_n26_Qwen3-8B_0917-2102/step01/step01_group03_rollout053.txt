import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize

def run_packing():
    # Initial hexagonal packing with 26 circles
    n = 26
    centers = []
    radii = []
    for i in range(n):
        row = i // 7
        col = i % 7
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        centers.append([x, y])
        radii.append(0.05)
    
    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)
    
    # Optimization to maximize sum of radii while satisfying constraints
    def objective(r):
        return -np.sum(r)
    
    def constraints(r):
        # Boundary and overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - 1e-8:
                    return False
        for i in range(n):
            if centers[i, 0] - radii[i] < -1e-8 or centers[i, 0] + radii[i] > 1 + 1e-8:
                return False
            if centers[i, 1] - radii[i] < -1e-8 or centers[i, 1] + radii[i] > 1 + 1e-8:
                return False
        return True
    
    result = minimize(objective, radii, constraints=lambda r: constraints(r), method='SLSQP', bounds=[(0, 1) for _ in range(n)])
    
    if not result.success:
        raise RuntimeError("Optimization failed")
    
    radii = result.x
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
