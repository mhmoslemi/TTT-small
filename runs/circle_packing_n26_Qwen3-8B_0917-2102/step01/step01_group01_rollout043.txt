from scipy.spatial.distance import pdist, squareform

def run_packing():

    centers = []
    radii = []

        x = col / 4.0 + 0.05
        y = row / 4.0 + 0.05
        r = 0.05
        centers.append([x, y])

        centers = centers[:n]
        radii = radii[:n]

        return -np.sum(radii)

        dist_matrix = squareform(pdist(centers_flat))
        for i in range(len(radii)):

                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:

        return 0

    result = minimize(objective, x0, constraints=[{'type': 'ineq', 'fun': constraint}, {'type': 'ineq', 'fun': boundary_constraint}])
