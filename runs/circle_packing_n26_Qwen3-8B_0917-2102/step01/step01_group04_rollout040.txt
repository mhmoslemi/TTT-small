import scipy.optimize as opt

    cols = 5

            y = i + 0.25

                r *= 1.2  # Adjust top and bottom rows

                r *= 1.15  # Adjust left and right columns
            circles.append([x, y])
            radii.append(r)

        y = np.random.uniform(0, 1)
        # Start with a radius of 0.05 and adjust
        r = 0.05
        # Check for overlap with existing circles
        overlap = False
        for i in range(len(circles)):

            dy = circles[i][1] - y
            dist = np.sqrt(dx**2 + dy**2)
            if dist < radii[i] + r - 1e-12:
                overlap = True
                break
        if not overlap:
            circles.append([x, y])
            radii.append(r)

    def objective(r):
        return -np.sum(r)

    def constraint1(r):
        return np.sum(r)

    bounds = [(0, 1.0) for _ in range(len(radii))]

    res = opt.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000}

                radii[i] = max(0, radii[i] - overlap / 2)

    return (centers, np.array(radii), np.sum(radii))
