from scipy.spatial.distance import pdist, squareform

        radii.append(0.08)

    centers = np.array(centers)
    radii = np.array(radii)

    # Define objective to maximize sum of radii
    def objective(x):
        centers_flat = x.reshape(-1, 2)
        return -np.sum(radii)

    # Define constraints to ensure circle non-overlap and boundary conditions
    def constraint(x):
        centers_flat = x.reshape(-1, 2)
        dist_matrix = squareform(pdist(centers_flat))
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return 1
        for i in range(len(radii)):

            r = radii[i]

                return 1
        return 0

    # Optimize centers
    x0 = centers.reshape(-1)
    result = minimize(objective, x0, constraints={"type": "ineq", "fun": constraint}, method='SLSQP')

    # Final validation and adjustment
    for i in range(len(radii)):

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
