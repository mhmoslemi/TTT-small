    def objective(x):
        # x is a flat array of [x0, y0, r0, x1, y1, r1, ...]
        x_vec = x[:n*2].reshape(n, 2)
        r_vec = x[n*2:]
        # Objective: maximize sum of radii
        return -np.sum(r_vec)
    
    def constraint_overlap(x):
        x_vec = x[:n*2].reshape(n, 2)
        r_vec = x[n*2:]
        for i in range(n):
            for j in range(i+1, n):
                dx = x_vec[i, 0] - x_vec[j, 0]
                dy = x_vec[i, 1] - x_vec[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)

        return 0

        x_vec = x[:n*2].reshape(n, 2)
