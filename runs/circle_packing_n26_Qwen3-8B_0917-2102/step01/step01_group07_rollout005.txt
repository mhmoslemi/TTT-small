import numpy as np
import scipy.optimize as opt

def run_packing():
    # Initial guess: hexagonal packing with rows
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Hexagonal packing with 5 rows (1, 2, 3, 4, 5 circles)
    row_height = np.sqrt(3)/2
    col_width = 1 / (np.ceil(n/2) + 1)
    
    for i in range(n):
        row = i // 2
        col = i % 2
        x = col * col_width
        y = row * row_height
        centers[i] = [x, y]
    
    # Optimize radii and positions
    def objective(r):
        return -np.sum(r)
    
    def constraints(r, centers):
        cons = []
        for i in range(n):
            cons.append((r[i], 0, np.inf))
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                cons.append((r[i] + r[j] - dist, 0, np.inf))
        
        for i in range(n):
            x, y = centers[i]
            cons.append((r[i], 0, 1 - x - r[i]))
            cons.append((r[i], 0, x - r[i]))
            cons.append((r[i], 0, 1 - y - r[i]))
            cons.append((r[i], 0, y - r[i]))
        
        return cons
    
    bounds = [(0, np.inf) for _ in range(n)]
    cons = constraints(radii, centers)
    
    result = opt.minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)
    
    radii = result.x
    sum_radii = np.sum(radii)
    
    return (centers, radii, sum_radii)
