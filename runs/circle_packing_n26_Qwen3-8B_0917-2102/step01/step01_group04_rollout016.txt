import numpy as np

def run_packing():
    # Define hexagonal packing with 26 circles
    # Using a 5-row hexagonal grid with 5, 5, 5, 5, 6 circles respectively
    # Optimizing circle positions and radii with constraints
    
    # Initial guess for centers (hexagonal grid)
    rows = 5
    cols = [5, 5, 5, 5, 6]
    centers = np.zeros((26, 2))
    for i in range(rows):
        for j in range(cols[i]):
            x = j + 0.5 if i % 2 else j
            y = i + 0.25
            centers[i * cols[i] + j] = [x, y]
    
    # Initial guess for radii (equal for all)
    radii = np.full(26, 0.05)
    
    # Optimization using scipy.optimize
    from scipy.optimize import minimize
    
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum
    
    def constraints(radii):
        # Ensure all circles are within unit square
        for i in range(26):
            if radii[i] > 1 - 0.001:
                return False
        return True
    
    bounds = [(0, 1 - 0.001) for _ in range(26)]
    
    result = minimize(objective, radii, bounds=bounds, constraints=constraints)
    
    radii = result.x
    sum_radii = np.sum(radii)
    
    # Final centers adjustment for boundary conditions
    for i in range(26):
        x, y = centers[i]
        r = radii[i]
        # Ensure circle is within [0, 1] x [0, 1]
        if x - r < 0:
            centers[i, 0] = r
        elif x + r > 1:
            centers[i, 0] = 1 - r
        if y - r < 0:
            centers[i, 1] = r
        elif y + r > 1:
            centers[i, 1] = 1 - r
    
    return centers, radii, sum_radii
