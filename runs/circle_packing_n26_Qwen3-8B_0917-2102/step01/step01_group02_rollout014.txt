import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r)

    cons = ({
        'type': 'ineq',
        'fun': constraint,
        'args': (),
        'jac': None
    })

    bounds = Bounds(np.zeros(n), np.full(n, 1.0))

    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)

    if not result.success:
        raise RuntimeError("Optimization failed")

    radii = result.x

    # Hexagonal packing with boundary adjustments
    def generate_hexagonal_centers(n):
        centers = []
        cols = int(np.ceil(np.sqrt(n)))
        rows = int(np.ceil(n / cols))
        for i in range(rows):
            for j in range(cols):
                if i % 2 == 0:
                    x = j / (cols - 1) if cols > 1 else 0.5
                else:
                    x = (j + 0.5) / (cols - 1) if cols > 1 else 0.5
                y = i / (rows - 1) if rows > 1 else 0.5
                centers.append([x, y])
        return np.array(centers)

    centers = generate_hexagonal_centers(n)
    centers += np.random.normal(0, 0.01, centers.shape)
    centers = np.clip(centers, 0, 1)

    # Initial radius estimate based on spacing
    distances = pdist(centers)
    min_dist = np.min(distances)
    radii = np.full(n, min_dist / 2.5)

    # Optimization with constraints
    def objective_func(r):
        return -np.sum(r)

    def constraint_func(r):
        return np.sum(r)

    cons = ({
        'type': 'ineq',
        'fun': constraint_func,
        'args': (),
        'jac': None
    })

    bounds = Bounds(np.zeros(n), np.full(n, 1.0))

    result = minimize(objective_func, radii, method='SLSQP', bounds=bounds, constraints=cons)

    if not result.success:
        raise RuntimeError("Optimization failed")

    radii = result.x

    # Boundary adjustment
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12:
            radii[i] = np.max([0, x])
        elif x + r > 1 + 1e-12:
            radii[i] = np.max([0, 1 - x])
        if y - r < -1e-12:
            radii[i] = np.max([0, y])
        elif y + r > 1 + 1e-12:
            radii[i] = np.max([0, 1 - y])

    # Non-overlap adjustment
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                radii[i] = (dist - 1e-12) / 2
                radii[j] = (dist - 1e-12) / 2

    return centers, radii, np.sum(radii)
