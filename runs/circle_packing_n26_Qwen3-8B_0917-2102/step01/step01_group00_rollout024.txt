import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Start with hexagonal grid offset for better boundary utilization
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    row, col = 0, 0
    for i in range(n):
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = 0.1
        col += 1
        if col >= 5:
            col = 0
            row += 1

    # Normalize centers to unit square
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(radii):
        return -np.sum(radii)

    def constraint(radii):
        return np.sum(radii) - 2.636

    cons = [{'type': 'ineq', 'fun': constraint}]

    def ineq_constraint_circle(i):
        def cons_func(radii):
            return radii[i] - 1e-8
        return cons_func

    for i in range(n):
        cons.append({'type': 'ineq', 'fun': ineq_constraint_circle(i)})

    def dist_constraint(i, j):
        def cons_func(radii):
            dist = np.linalg.norm(centers[i] - centers[j])
            return dist - (radii[i] + radii[j] - 1e-8)
        return cons_func

    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': dist_constraint(i, j)})

    result = minimize(objective, radii, constraints=cons, method='SLSQP', tol=1e-8)

    radii = result.x
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    # Final check and adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
