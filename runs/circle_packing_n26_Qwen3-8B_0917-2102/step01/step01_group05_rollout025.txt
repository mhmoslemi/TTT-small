import numpy as np
from scipy.optimize import minimize
import matplotlib.pyplot as plt

def run_packing():
    n = 26
    # Define a hexagonal grid with 5 rows (starting from top)
    rows = 5
    cols = 5
    # Adjust for 26 circles (5 + 4 + 5 + 4 + 8)
    # First 5 rows with 5 circles, next 4 rows with 4 circles, last row with 8 circles
    centers = []
    radii = []
    for i in range(rows):
        if i < 2 or i > 3:
            col_count = 5
        else:
            col_count = 4
        for j in range(col_count):
            x = j + 0.5
            y = i + 0.5
            # Scale to unit square
            x = x / (cols - 1)
            y = y / (rows - 1)
            centers.append([x, y])
    centers = np.array(centers)
    # Initial guess for radii: based on hexagonal packing
    initial_radii = np.full(n, 0.1)
    # Optimization function to maximize total radius sum
    def objective(radii):
        return -np.sum(radii)
    
    # Constraints: circles must not overlap and stay within [0,1]^2
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            def constraint_func(radii):
                dist = np.linalg.norm(centers[i] - centers[j])
                return dist - (radii[i] + radii[j])
            constraints.append({'type': 'ineq', 'fun': constraint_func})
    
    for i in range(n):
        def boundary_func(radii):
            x, y = centers[i]
            return x - radii[i] + 1e-12
        constraints.append({'type': 'ineq', 'fun': boundary_func})
        def boundary_func2(radii):
            x, y = centers[i]
            return y - radii[i] + 1e-12
        constraints.append({'type': 'ineq', 'fun': boundary_func2})

            x, y = centers[i]
            return 1 - x - radii[i] + 1e-12
        constraints.append({'type': 'ineq', 'fun': boundary_func3})

            x, y = centers[i]
            return 1 - y - radii[i] + 1e-12
        constraints.append({'type': 'ineq', 'fun': boundary_func4})
    
    # Bounds for radii
    bounds = [(0, 1) for _ in range(n)]
    
    # Run optimization
    result = minimize(objective, initial_radii, method='SLSQP', bounds=bounds, constraints=constraints)
    radii = result.x
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
