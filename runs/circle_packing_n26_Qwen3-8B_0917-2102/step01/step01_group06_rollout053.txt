import numpy as np
import random
import math

def run_packing():
    # Initial configuration: 4 rows of hexagonal packing with 7, 6, 6, 7 circles
    # Near boundaries with adjusted positions for optimal packing
    centers = np.array([
        [0.1, 0.1], [0.35, 0.1], [0.6, 0.1], [0.85, 0.1], [0.97, 0.1],
        [0.15, 0.35], [0.4, 0.35], [0.65, 0.35], [0.9, 0.35],
        [0.1, 0.6], [0.35, 0.6], [0.6, 0.6], [0.85, 0.6],
        [0.15, 0.85], [0.4, 0.85], [0.65, 0.85], [0.9, 0.85],
        [0.3, 0.5], [0.55, 0.5], [0.8, 0.5],
        [0.25, 0.7], [0.5, 0.7], [0.75, 0.7],
        [0.4, 0.4], [0.6, 0.6]
    ])
    radii = np.array([0.05] * 26)
    
    # Optimize via simulated annealing
    temp = 1.0
    cooling_rate = 0.995
    iterations = 2000
    
    for _ in range(iterations):
        for i in range(26):
            # Small random perturbation
            dx = random.uniform(-0.01, 0.01)
            dy = random.uniform(-0.01, 0.01)
            new_centers = centers.copy()
            new_centers[i] = [new_centers[i][0] + dx, new_centers[i][1] + dy]
            
            # Check boundary constraints
            valid = True
            for j in range(26):
                x, y = new_centers[j]
                r = radii[j]
                if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                    valid = False
                    break
            if not valid:
                continue
            
            # Check circle-circle collisions
            collision = False
            for j in range(i + 1, 26):
                dist = math.hypot(new_centers[i][0] - new_centers[j][0], new_centers[i][1] - new_centers[j][1])
                if dist < radii[i] + radii[j] - 1e-12:
                    collision = True
                    break
            if collision:
                continue
            
            # Acceptance probability
            old_sum = sum(radii)
            new_sum = old_sum + random.uniform(-0.001, 0.001)
            if new_sum <= 0:
                continue
            new_radii = radii.copy()
            new_radii[i] = new_sum
            new_sum = sum(new_radii)
            
            if random.random() < math.exp((new_sum - old_sum) / temp):
                centers = new_centers
                radii = new_radii
        
        temp *= cooling_rate
    
    return centers, radii, sum(radii)
