from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import product

def run_packing():

    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    total_sum = 0

    # Hexagonal packing with varying radii
    rows = 5
    cols = 5 + 1  # 5 full rows with 6 circles, 1 row with 5 circles

    for i in range(rows):
        for j in range(cols):
            if i % 2 == 0:
                x = j / (cols - 1)
                y = i / (rows - 1)
            else:
                x = (j + 0.5) / (cols - 1)
                y = i / (rows - 1)
            centers[i * cols + j] = [x, y]
            radii[i * cols + j] = 0.15

    # Optimize radii with constraints
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        dist_matrix = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    bounds = [(0, 1) for _ in range(n)]

    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraints}],
        method='SLSQP'
    )

    radii = result.x
    total_sum = np.sum(radii)

    # Ensure circles are within the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            radii[i] = min(r, 1 - np.min([x, 1 - x, y, 1 - y]))

    total_sum = np.sum(radii)
    return centers, radii, total_sum
