import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    n = 26
    # Initial hexagonal packing with 5 rows
    rows = 5
    cols = 6
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Hexagonal grid with vertical spacing
    for i in range(rows):
        for j in range(cols):
            x = j + 0.5
            y = i + 0.5
            if i % 2 == 1:
                x += 0.5
            centers[i * cols + j] = [x, y]
    
    # Initial guess for radii
    radii = np.ones(n) * 0.05
    
    # Optimization
    def objective(r):
        return -np.sum(r)
    
    def constraints(r):
        dists = cdist(centers, centers)
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] + 1e-8:
                    return False
        return True
    
    res = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=[(0, 0.1)] * n,
        constraints=[{'type': 'ineq', 'fun': constraints}]
    )
    
    # Apply boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = res.x[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            res.x[i] = min(max(r, 1 - max(x, 1 - x)), 0.1)
    
    centers = np.clip(centers, 0, 1)
    radii = res.x
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
