import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():

    # Initial hexagonal packing with boundary adjustments
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    r = 0.1  # Start with small radii and adjust later

    # Initial hexagonal grid
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = r

    # Adjust radii based on distance to boundaries
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1 - x, y, 1 - y)
        radii[i] = min(radii[i], max_r)

    # Optimization function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraints(radii):
        # Distance between circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    # Bounds for radii
    bounds = [(0, 1) for _ in range(n)]

    # Optimization
    result = opt.minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=Bounds([0]*n, [1]*n),
        constraints=[{'type': 'ineq', 'fun': constraints}]
    )

    # Use the optimized radii
    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
