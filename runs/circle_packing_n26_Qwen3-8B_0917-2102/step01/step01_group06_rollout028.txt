import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Initial configuration: hexagonal packing with 26 circles
    # First, define a grid for hexagonal packing
    num_circles = 26
    rows = int(np.ceil(np.sqrt(num_circles)))
    cols = int(np.ceil(num_circles / rows))
    
    # Generate initial positions for hexagonal grid
    centers = np.zeros((num_circles, 2))
    for i in range(num_circles):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        centers[i, 0] = x / (cols - 1) if cols > 1 else 0
        centers[i, 1] = y / (rows - 1) if rows > 1 else 0

    # Normalize centers to unit square
    centers = (centers - np.min(centers)) / (np.max(centers) - np.min(centers))

    # Initial guess for radii: uniform based on minimal distance between circles
    distances = cdist(centers, centers)
    np.fill_diagonal(distances, np.inf)
    min_dist = np.min(distances)
    radii = np.full(num_circles, min_dist / 2.5)

    # Define objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Define constraints: circles must not overlap
    def constraint_overlap(radii):
        dists = cdist(centers, centers)
        np.fill_diagonal(dists, np.inf)
        return np.array([dists[i, j] - radii[i] - radii[j] for i in range(num_circles) for j in range(i + 1, num_circles)]).flatten()

    # Define constraints: circles must stay within unit square
    def constraint_boundary(radii):
        return np.array([centers[i, 0] - radii[i], centers[i, 0] + radii[i], centers[i, 1] - radii[i], centers[i, 1] + radii[i]] for i in range(num_circles)).flatten()

    # Bounds: radii must be non-negative
    bounds = [(0, np.inf) for _ in range(num_circles)]

    # Optimization: maximize sum of radii with constraints
    cons = [{'type': 'ineq', 'fun': constraint_overlap},
            {'type': 'ineq', 'fun': constraint_boundary}]
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)

    # Extract result
    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
