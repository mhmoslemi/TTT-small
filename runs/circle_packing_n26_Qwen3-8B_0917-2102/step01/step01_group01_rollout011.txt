import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.cluster.hierarchy import linkage, fcluster
from itertools import combinations

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)

    def objective(radii):
        return -np.sum(radii)

    def constraint_func(x):
        centers_flat = x.reshape(-1, 2)
        dist_matrix = squareform(pdist(centers_flat))
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return 1
        return 0

    def boundary_constraint(x):
        centers_flat = x.reshape(-1, 2)
        for i in range(len(radii)):
            x, y = centers_flat[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return 1
        return 0

    def refine_centers(centers, radii):
        dist_matrix = squareform(pdist(centers))
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    mid = (centers[i] + centers[j]) / 2
                    new_r = (dist_matrix[i, j] - radii[i] - radii[j]) / 2
                    radii[i] = max(0, radii[i] - new_r)
                    radii[j] = max(0, radii[j] - new_r)
                    centers[i] = mid
                    centers[j] = mid
        return centers, radii

    def boundary_refinement(centers, radii):
        for i in range(len(radii)):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0:
                centers[i, 0] = r
                radii[i] = max(0, x - r)
            elif x + r > 1:
                centers[i, 0] = 1 - r
                radii[i] = max(0, r - (1 - x))
            if y - r < 0:
                centers[i, 1] = r
                radii[i] = max(0, y - r)
            elif y + r > 1:
                centers[i, 1] = 1 - r
                radii[i] = max(0, r - (1 - y))
        return centers, radii

    def optimize(x0):
        x = minimize(objective, x0, constraints=[
            {"type": "ineq", "fun": constraint_func},
            {"type": "ineq", "fun": boundary_constraint}
        ]).x
        return x

    centers = optimize(centers.reshape(-1))
    centers = centers.reshape(-1, 2)
    radii = np.clip(radii, 0, 1)
    centers, radii = refine_centers(centers, radii)
    centers, radii = boundary_refinement(centers, radii)
    return centers, radii, np.sum(radii)
