import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal packing
    centers = []
    for i in range(n):
        row = i // 2
        col = i % 2
        x = (col + 0.5) / np.sqrt(3)
        y = row / np.sqrt(3)
        centers.append([x, y])
    centers = np.array(centers)

    # Perturb to avoid boundary issues
    centers += np.random.normal(0, 0.05, centers.shape)
    centers = np.clip(centers, 0, 1)

    # Initial radius guess based on minimum distance to boundary
    radii = np.zeros(n)
    for i in range(n):
        dist_to_boundary = np.min([centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1]])
        radii[i] = dist_to_boundary * 0.9

    # Objective function: maximize sum of radii
    def objective(r):
        return -np.sum(r)

    # Constraints: non-overlapping circles
    def constraint_func(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    # Bounds for radii
    bounds = Bounds(np.zeros(n), np.full(n, 1.0))

    # Optimization
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraint_func})

    if not result.success:
        raise RuntimeError("Optimization failed")

    radii = result.x

    # Final validation and adjustment
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = np.max([0, 1 - np.max([x, 1 - x, y, 1 - y])])

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                radii[i] = (dist - 1e-12) / 2
                radii[j] = (dist - 1e-12) / 2

    return centers, radii, np.sum(radii)
