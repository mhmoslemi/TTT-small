import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)

    def objective(radii):
        return -np.sum(radii)

    def constraint_func(radii):
        dist_matrix = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return 1
        return 0

    bounds = Bounds(0, 1)
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints={"type": "ineq", "fun": constraint_func})

    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if (x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1):
            result.x[i] = max(0, min(r, 1 - max(x, 1 - x, y, 1 - y)))

    return centers, result.x, np.sum(result.x)
