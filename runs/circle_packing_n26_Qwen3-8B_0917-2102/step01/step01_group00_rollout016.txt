import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)

    def objective(x):
        return -np.sum(x)

    def constraint(x, i, j):
        return np.sqrt(np.sum((centers[i] - centers[j]) ** 2)) - (radii[i] + radii[j])

    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: constraint(x, i, j)})

    for i in range(n):
        def constraint_left(x, i=i):
            return centers[i, 0] - radii[i]
        constraints.append({'type': 'ineq', 'fun': constraint_left})

        def constraint_right(x, i=i):
            return 1 - centers[i, 0] - radii[i]
        constraints.append({'type': 'ineq', 'fun': constraint_right})

        def constraint_bottom(x, i=i):
            return centers[i, 1] - radii[i]
        constraints.append({'Type': 'ineq', 'fun': constraint_bottom})

        def constraint_top(x, i=i):
            return 1 - centers[i, 1] - radii[i]
        constraints.append({'type': 'ineq', 'fun': constraint_top})

    result = minimize(objective, np.concatenate((centers.reshape(-1), radii)), constraints=constraints, method='SLSQP')

    optimized = np.split(result.x, [2 * n, 2 * n + n])
    centers = optimized[0].reshape(n, 2)
    radii = optimized[1]

    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j]:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0.0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
