import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n) * 0.1

    def objective(r):
        return -np.sum(r)

    def constraint(r, centers):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] + 1e-8:
                    return float('-inf')
        for i in range(n):
            x, y = centers[i]
            if x - r[i] < -1e-12 or x + r[i] > 1 + 1e-12 or y - r[i] < -1e-12 or y + r[i] > 1 + 1e-12:
                return float('-inf')
        return 0

    cons = {'type': 'ineq', 'fun': lambda r: constraint(r, centers)}

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})

    # Final adjustment for boundary constraints
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result.x[i] + result.x[j]:
                ratio = (dists[i, j] - 1e-8) / (result.x[i] + result.x[j])
                result.x[i] *= ratio
                result.x[j] *= ratio

    # Ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12):
            result.x[i] = 0.0

    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
