import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist

def run_packing():
    # Initial hexagonal packing with 26 circles
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Hexagonal grid parameters
    rows = int(np.sqrt(n))
    cols = (n + rows - 1) // rows
    radius = 0.15
    
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        centers[i] = [x / (cols - 0.5), y / (rows * np.sqrt(3) / 2)]
        radii[i] = radius
    
    # Define function to calculate total radius sum
    def objective(radii):
        return -np.sum(radii)
    
    # Define constraints for boundary and circle overlaps
    def constraints(radii):
        constraints = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return False

                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True
    
    # Optimization
    bounds = [(0, 1) for _ in range(n)]
    res = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'constraint', 'fun': constraints},

    )
