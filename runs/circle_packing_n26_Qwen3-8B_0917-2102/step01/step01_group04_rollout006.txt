import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist

def run_packing():
    # Initial guess for centers in hexagonal packing
    centers = np.zeros((26, 2))
    for i in range(26):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row * np.sqrt(3) / 2
        centers[i] = [x, y]
    
    # Initial radii as a fraction of the distance to the nearest boundary
    radii = np.zeros(26)
    for i in range(26):
        x, y = centers[i]
        dist_to_boundary = np.min([x, 1 - x, y, 1 - y])
        radii[i] = dist_to_boundary * 0.8
    
    # Optimization function
    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii
    
    def constraints(radii):
        for i in range(26):
            if radii[i] < 0:
                return False
        for i in range(26):
            x, y = centers[i]
            if (x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12
                or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12):
                return False
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True
    
    # Optimization
    bounds = [(0, 1) for _ in range(26)]
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[{'type': 'callback', 'fun': constraints}],
        method='SLSQP'
    )
    
    return (centers, result.x, np.sum(result.x))
