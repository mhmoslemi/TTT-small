import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal packing with adjusted spacing
    centers = []
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + (row % 2) * 0.5
        y = row * (np.sqrt(3) / 2)
        centers.append([x, y])
    centers = np.array(centers) / np.max(centers, axis=0)

    # Initial radii based on minimum distance between circles
    dists = squareform(pdist(centers))
    min_dist = np.min(dists)
    radii = np.full(n, min_dist / 2)

    # Objective function to maximize sum of radii
    def objective(r):
        return -np.sum(r)

    # Constraints: non-overlapping and within unit square
    def non_overlapping(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r)

    def boundary(r):
        for i in range(n):
            x, y = centers[i]
            r_i = r[i]
            if x - r_i < -1e-8 or x + r_i > 1 + 1e-8 or y - r_i < -1e-8 or y + r_i > 1 + 1e-8:
                return -np.inf
        return np.sum(r)

    # Bounds for radii
    bounds = Bounds(np.zeros(n), np.full(n, 1))

    # Optimization with non-linear constraints
    cons = [{'type': 'ineq', 'fun': non_overlapping}, {'type': 'ineq', 'fun': boundary}]
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)

    if not result.success:
        raise RuntimeError("Optimization failed")

    return centers, result.x, np.sum(result.x)
