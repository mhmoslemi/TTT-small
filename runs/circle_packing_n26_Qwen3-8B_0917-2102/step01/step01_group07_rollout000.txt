import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial hexagonal packing with 5 rows
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    row_height = np.sqrt(3) / 2
    col_width = 1.0 - 2 * 0.1  # Leave 0.1 margin on both sides

    for i in range(5):
        for j in range(13):
            if i % 2 == 0:
                x = 0.1 + j * (col_width / 12)
            else:
                x = 0.1 + (j + 0.5) * (col_width / 12)
            y = 0.1 + i * row_height
            if i == 4:
                y = 0.1 + (i - 0.5) * row_height
            centers[i * 13 + j] = [x, y]

    # Adjust radii to fit within unit square
    for i in range(26):
        x, y = centers[i]
        radii[i] = min(0.1, (0.1 - x) if x < 0.1 else (1 - 0.1 - x), (0.1 - y) if y < 0.1 else (1 - 0.1 - y))

    # Optimization to maximize sum of radii with non-overlap
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-10:
                    return 1
        return 0

    res = minimize(objective, radii, method='SLSQP', constraints={'type': 'ineq', 'fun': constraints})
    radii = res.x

    # Final check and return
    return centers, radii, np.sum(radii)
