        row = i // 5
        col = i % 5
        centers[i, 0] = col * r + 0.5 * r

        for i in range(n):

                return False

    bounds = [(0, 1) for _ in range(n)]
    result = minimize(
        objective,
        radii,
        method='SLSQP',

    return (centers, radii, sum_radii)
