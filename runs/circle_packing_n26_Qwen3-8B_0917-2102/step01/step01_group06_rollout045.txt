import numpy as np
import math

def run_packing():
    # Hexagonal packing with 25 small circles and 1 large circle at the center
    # Calculate radius of small circles based on spacing
    small_radius = (1 - 2 * math.sqrt(3) / 3) / 2
    big_radius = 1 - 2 * small_radius

    # Generate centers for the hexagonal packing
    centers = []
    for i in range(5):
        for j in range(5):
            x = j + 0.5
            y = i + 0.5
            if i % 2 == 0:
                x += math.sqrt(3)/2 * (j % 2)
            centers.append([x, y])

    # Add the big circle at the center
    centers.append([0.5, 0.5])

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array([small_radius] * len(centers) - 1 + [big_radius])

    # Ensure the big circle is not overlapping with the small ones
    for i in range(len(centers) - 1):
        dist = np.sqrt((centers[i, 0] - centers[-1, 0])**2 + (centers[i, 1] - centers[-1, 1])**2)
        if dist < radii[i] + radii[-1] - 1e-12:
            radii[-1] = dist - radii[i] - 1e-12

    return centers, radii, np.sum(radii)
