from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import matplotlib.pyplot as plt

def run_packing():

    # Positions are adjusted to fit in unit square with boundary circles
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # First row (top boundary)
    centers[0] = [0.25, 1.0]
    centers[1] = [0.75, 1.0]
    radii[0] = radii[1] = 0.15

    # Second row (above boundary circles)
    centers[2] = [0.5, 0.75]
    radii[2] = 0.25

    # Third row
    centers[3] = [0.25, 0.5]
    centers[4] = [0.75, 0.5]
    radii[3] = radii[4] = 0.2

    # Fourth row
    centers[5] = [0.5, 0.25]
    radii[5] = 0.2

    # Fifth row (bottom boundary)
    centers[6] = [0.25, 0.0]
    centers[7] = [0.75, 0.0]
    radii[6] = radii[7] = 0.15

    # Remaining circles placed in hexagonal pattern
    for i in range(8, num_circles):
        row = (i - 8) // 2
        col = (i - 8) % 2
        x = 0.5 - 0.25 + (col * 0.5)
        y = 0.5 - row * 0.5
        centers[i] = [x, y]
        radii[i] = 0.15

    # Optimization to maximize total radius while ensuring non-overlapping and fitting within square
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        # Ensure all circles fit within the unit square
        for i in range(num_circles):
            if centers[i][0] - radii[i] < 0 or centers[i][0] + radii[i] > 1:
                return False
            if centers[i][1] - radii[i] < 0 or centers[i][1] + radii[i] > 1:
                return False
        # Ensure no overlapping between circles
        dists = squareform(pdist(centers))
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    # Initial guess with all radii set to 0.15
    initial_radii = np.full(num_circles, 0.15)
    result = minimize(
        objective,
        initial_radii,
        constraints=lambda x: constraints(x),
        bounds=[(0, 1)] * num_circles,
        method='SLSQP'
    )

    return (centers, result.x, np.sum(result.x))
