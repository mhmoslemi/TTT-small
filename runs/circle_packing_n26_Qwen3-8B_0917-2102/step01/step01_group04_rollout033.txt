from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():

        return -np.sum(radii)

                constraints.append(lambda r, i=i: r[i] - (1 - centers[i, 0]))
                constraints.append(lambda r, i=i: r[i] - centers[i, 0])

        return constraints
