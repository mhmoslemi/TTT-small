import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)

    def objective(radii_vec):
        return -np.sum(radii_vec)

    def constraint_overlap(radii_vec):
        dist_matrix = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii_vec[i] + radii_vec[j] - 1e-8:
                    return 1
        return 0

    def constraint_boundary(radii_vec):
        for i in range(n):
            x, y = centers[i]
            r = radii_vec[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return 1
        return 0

    bounds = Bounds(0, 1)

    result = minimize(objective, radii, method='SLSQP',
                     constraints=[{'type': 'ineq', 'fun': constraint_overlap},
                                 {'type': 'ineq', 'fun': constraint_boundary}],
                     bounds=bounds)

    radii = result.x
    centers = np.array(centers)

    # Refine boundary circles
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            radii[i] = max(0, 1 - max(x, 1 - x, y, 1 - y))

    return centers, radii, np.sum(radii)
