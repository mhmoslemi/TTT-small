import numpy as np

def run_packing():
    # Hexagonal packing of 25 circles with radius r1, and one large circle near (0,0)
    # Calculate radius for hexagonal grid in unit square
    r1 = (1 - 1/np.sqrt(3)) / 3  # Sufficient to fit 25 circles in hex grid
    
    # Positions for 25 circles in hexagonal grid
    centers_hex = []
    for i in range(5):
        for j in range(5):
            x = j + i / 2
            y = i * np.sqrt(3) / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers_hex.append([x, y])
    
    # Add one large circle near (0,0)
    r2 = 1 - 1/np.sqrt(3) - 1e-3  # Leave space for 25 small circles
    centers = np.array(centers_hex + [[0.1, 0.1]])
    radii = np.array([r1]*25 + [r2])
    
    # Validate packing
    assert np.all(radii >= 0)
    assert np.all(centers >= 0)
    assert np.all(centers <= 1)
    
    return centers, radii, np.sum(radii)
