import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    np.random.seed(42)
    n = 26
    initial_centers = np.random.rand(n, 2)
    initial_radii = np.full(n, 0.05)

    def obj_func(radii):
        total = np.sum(radii)
        return -total  # Minimize negative to maximize

    def constraints(radii):
        dist_matrix = squareform(pdist(initial_centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return False
        for i in range(n):
            if initial_centers[i, 0] - radii[i] < -1e-8 or initial_centers[i, 0] + radii[i] > 1 + 1e-8:
                return False
            if initial_centers[i, 1] - radii[i] < -1e-8 or initial_centers[i, 1] + radii[i] > 1 + 1e-8:
                return False
        return True

    bounds = Bounds(0, 1)
    result = minimize(obj_func, initial_radii, method='SLSQP', bounds=bounds, constraints=constraints)
    radii = result.x
    centers = initial_centers.copy()

    # Apply boundary adjustment to maximize radii
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            centers[i, 0] = r
        elif x + r > 1:
            centers[i, 0] = 1 - r
        if y - r < 0:
            centers[i, 1] = r
        elif y + r > 1:
            centers[i, 1] = 1 - r

    # Final check for boundary
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-8 or x + r > 1 + 1e-8 or
            y - r < -1e-8 or y + r > 1 + 1e-8):
            radii[i] = 1 - np.min([x, 1 - x, y, 1 - y])

    return centers, radii, np.sum(radii)
