import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial guess: hexagonal packing with 5 rows
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    row_height = np.sqrt(3) / 2
    col_width = 1
    num_rows = 5
    num_cols = 6

    for i in range(num_rows):
        for j in range(num_cols):
            x = j * col_width / (num_cols - 1)
            y = i * row_height
            centers[i * num_cols + j] = [x, y]

    # Adjust for boundary and optimize radii
    def objective(radii):
        return -np.sum(radii)

    def constraint1(radii):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return False
        return True

    def constraint2(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    bounds = [(0, 1) for _ in range(n)]

    res = minimize(
        objective,
        x0=np.ones(n),
        method='SLSQP',
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2}
        ]
    )

    return (centers, res.x, np.sum(res.x))
