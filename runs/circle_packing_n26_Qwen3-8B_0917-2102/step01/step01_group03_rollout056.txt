import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance

def run_packing():
    # Hexagonal packing with 26 circles
    # Rows: 5 rows with 5, 5, 5, 5, 6 circles
    # Starting from bottom-left corner

    centers = []
    radii = []

    # Row 0 (bottom row)
    for i in range(5):
        x = i / 4.0
        y = 0.05
        centers.append([x, y])
        radii.append(0.05)

    # Row 1
    for i in range(5):
        x = i / 4.0 + 0.5 / 4.0
        y = 0.15
        centers.append([x, y])
        radii.append(0.05)

    # Row 2
    for i in range(5):
        x = i / 4.0
        y = 0.25
        centers.append([x, y])
        radii.append(0.05)

    # Row 3
    for i in range(5):
        x = i / 4.0 + 0.5 / 4.0
        y = 0.35
        centers.append([x, y])
        radii.append(0.05)

    # Row 4 (top row)
    for i in range(6):
        x = i / 5.0
        y = 0.45
        centers.append([x, y])
        radii.append(0.05)

    # Optimize for maximum sum of radii with non-overlapping and boundary constraints
    def objective(radii):
        return -sum(radii)  # Maximize sum of radii

    def constraints(radii):
        # Boundary constraints
        for i in range(len(radii)):
            if centers[i][0] - radii[i] < -1e-12 or centers[i][0] + radii[i] > 1 + 1e-12:
                return False
            if centers[i][1] - radii[i] < -1e-12 or centers[i][1] + radii[i] > 1 + 1e-12:
                return False

        # Pairwise distance constraints
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dist = np.sqrt((centers[i][0] - centers[j][0])**2 + (centers[i][1] - centers[j][1])**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    # Initial guess
    initial_guess = [0.05] * len(radii)

    # Bounds for radii (non-negative)
    bounds = [(0, 1)] * len(radii)

    # Optimization
    result = minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        constraints=[{"type": "ineq", "fun": constraints}],
        method="SLSQP"
    )

    if not result.success:
        raise ValueError("Optimization failed")

    optimized_radii = result.x
    sum_radii = sum(optimized_radii)

    return np.array(centers), np.array(optimized_radii), sum_radii
