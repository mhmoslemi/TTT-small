import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    # Initial guess for centers using hexagonal packing
    num_circles = 26
    rows = 6
    cols = 5
    spacing = 1.0 / (rows + 1)
    centers = np.zeros((num_circles, 2))
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + spacing / 2
            y = i * spacing + spacing / 2
            centers[i * cols + j] = [x, y]

    # Initial guess for radii
    radii = np.full(num_circles, 0.1)

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraints(radii):
        # Boundary constraint: circle must be fully inside [0,1]^2
        for i in range(num_circles):
            if (centers[i, 0] - radii[i] < -1e-8 or
                centers[i, 0] + radii[i] > 1 + 1e-8 or
                centers[i, 1] - radii[i] < -1e-8 or
                centers[i, 1] + radii[i] > 1 + 1e-8):
                return False
        # Pairwise non-overlapping constraint
        dists = squareform(pdist(centers))
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    # Optimization bounds
    bounds = Bounds(np.zeros(num_circles), np.full(num_circles, 1.0))

    # Optimization with constraints
    res = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'soft_bound', 'fun': constraints}
    )

    # Extract results
    centers_final = centers
    radii_final = res.x
    sum_radii = np.sum(radii_final)
    return centers_final, radii_final, sum_radii
