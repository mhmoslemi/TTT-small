import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    np.random.seed(42)
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.ones(n) * 0.1

    def objective(r):
        return -np.sum(r)

    def constraints(r):
        dist = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist[i, j] < r[i] + r[j] + 1e-8:
                    return False
        for i in range(n):
            x, y = centers[i]
            if x - r[i] < -1e-8 or x + r[i] > 1 + 1e-8 or y - r[i] < -1e-8 or y + r[i] > 1 + 1e8:
                return False
        return True

    bounds = Bounds(np.zeros(n), np.full(n, 1.0))

    result = minimize(objective, radii, bounds=bounds, constraints={"type": "custom", "fun": constraints, "args": ()})

    if not result.success:
        raise ValueError("Optimization failed")

    return centers, result.x, -result.fun
