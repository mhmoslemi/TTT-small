import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import matplotlib.pyplot as plt

def run_packing():
    np.random.seed(42)
    n = 26

    # Initial hexagonal packing
    rows = 5
    cols = 6
    centers = np.zeros((n, 2))
    r = np.zeros(n)
    for i in range(rows):
        for j in range(cols):
            idx = i * cols + j
            if idx >= n:
                break
            x = j + 0.5 * (i % 2)
            y = i * np.sqrt(3) / 2
            centers[idx] = [x, y]
            r[idx] = 0.05

    # Initial radius guess with boundary alignment
    for i in range(n):
        x, y = centers[i]
        dist_to_boundary = np.array([
            x - 1e-12, 1 - x - 1e-12,
            y - 1e-12, 1 - y - 1e-12
        ])
        r[i] = np.min(dist_to_boundary) / 2

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    bounds = [(0, 1) for _ in range(n)]

    result = minimize(
        objective,
        r,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'success', 'fun': constraints}
    )

    return (centers, result.x, np.sum(result.x))
