import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Hexagonal packing with adjusted rows
    for i in range(n):
        row_idx = i // 6
        col_idx = i % 6
        x = col_idx * (1 + 1 / np.sqrt(3)) / (n / 2)
        y = row_idx * (1 + 1 / np.sqrt(3)) / (n / 2)
        centers[i] = [x, y]

    # Initial radius based on minimal spacing
    radii = np.full(n, 0.05)

    def objective(r):
        return -np.sum(r)

    def constraint_func(r):
        dist_matrix = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < r[i] + r[j] + 1e-12:
                    return -np.inf
        return np.sum(r) - 2.636

    cons = ({
        'type': 'ineq',
        'fun': constraint_func
    })

    bounds = Bounds(np.zeros(n), np.full(n, 1.0))

    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)

    if not result.success:
        raise RuntimeError("Optimization failed")

    radii = result.x

    # Final validation and boundary adjustment
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = np.max([0, 1 - np.max([x, 1 - x, y, 1 - y])])

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                radii[i] = (dist - 1e-12) / 2
                radii[j] = (dist - 1e-12) / 2

    return centers, radii, np.sum(radii)
