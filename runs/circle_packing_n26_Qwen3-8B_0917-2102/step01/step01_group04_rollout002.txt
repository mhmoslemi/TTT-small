def run_packing():

    n = 26
    rows = 5
    cols = 5
    offset = 0.5 / np.sqrt(3)
    spacing = 0.5
    
    # Generate centers
    centers = []
    for i in range(rows):
        if i % 2 == 0:
            for j in range(cols):
                x = j * spacing
                y = i * spacing + offset * i
                centers.append([x, y])
        else:
            for j in range(cols - 1):
                x = j * spacing + spacing / 2
                y = i * spacing + offset * i
                centers.append([x, y])
    
    centers = np.array(centers)
    
    # Calculate radii based on minimal spacing between circles
    # We'll use a constant radius for all circles to maintain uniform packing
    # The maximum radius is determined by the minimal spacing between circles and boundaries
    
    # Compute minimal distances to boundaries and between circles
    min_dist_to_boundary = np.min(np.array([
        centers[:, 0] - 1e-12,
        1 - centers[:, 0] - 1e-12,
        centers[:, 1] - 1e-12,
        1 - centers[:, 1] - 1e-12
    ]))
    
    # Minimal distance between circles
    min_circle_dist = np.inf
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist < min_circle_dist:
                min_circle_dist = dist
    
    # Compute maximal radius
    max_radius = min(min_dist_to_boundary, min_circle_dist / 2)
    
    # Return centers, radii, and total sum
    radii = np.full(n, max_radius)
    return centers, radii, np.sum(radii)
