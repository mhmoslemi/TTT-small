from scipy.optimize import minimize, Bounds
from scipy.optimize import LinearConstraint
from scipy.spatial.distance import pdist, squareform

def run_packing():

    radii = np.full(n, 0.05)

    def objective(r):
        return -np.sum(r)

    def constraint_func(r):
        return np.sum(r)

    cons = ({
        'type': 'ineq',
        'fun': constraint_func,
        'jac': lambda r: np.ones(n)
    })

    bounds = Bounds(np.zeros(n), np.full(n, 1.0))

    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)

    if not result.success:
        raise RuntimeError("Optimization failed")

    radii = result.x

    # Hexagonal packing with boundary-aware placement
    centers = []
    r = 0.1
    for i in range(n):
        row = i // 2
        col = i % 2
        x = row + col * (1 / (np.sqrt(3) * 1))
        y = (i % 2) * (1 / np.sqrt(3))
        centers.append([x, y])
    centers = np.array(centers)

    # Boundary alignment
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            centers[i][0] = r
        elif x + r > 1:
            centers[i][0] = 1 - r
        if y - r < 0:
            centers[i][1] = r
        elif y + r > 1:
            centers[i][1] = 1 - r

    # Final validation and adjustment
    for i in range(n):
        x, y = centers[i]

        for j in range(i + 1, n):

                radii[i] = (dist - 1e-12) / 2
                radii[j] = (dist - 1e-12) / 2
