import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance_matrix
from scipy.stats import qmc

def run_packing():
    n = 26
    # Initial guess using QMC for uniform distribution
    sampler = qmc.MultivariateUniformSobol(d=2)
    centers = sampler.random(n)
    radii = np.full(n, 0.1)

    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii

    def constraint_dist(x, i, j):
        return np.sqrt(np.sum((x[i] - x[j])**2)) - (radii[i] + radii[j])

    bounds = [(0, 1) for _ in range(2 * n)]
    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: constraint_dist(x, i, j)})
    for i in range(n):
        x, y = centers[i]
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i] - x[2*i + 1] - 1e-12})
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i + 1] - x[2*i] - 1e-12})

    x0 = np.concatenate([centers.flatten(), radii])
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    if not res.success:
        raise Exception("Optimization failed")
    centers = res.x[::2].reshape(-1, 2)
    radii = res.x[1::2]
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
