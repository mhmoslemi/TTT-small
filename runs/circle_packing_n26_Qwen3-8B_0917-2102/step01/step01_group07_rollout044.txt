import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial hexagonal packing with 26 circles
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Hexagonal packing for even rows
    row_count = 6
    col_count = 5
    for i in range(row_count):
        for j in range(col_count):
            row = i
            col = j
            if i % 2 == 1:
                col += 0.5
            x = col / (col_count - 0.5)
            y = i / (row_count - 0.5)
            centers[i * col_count + j] = [x, y]
    
    # Initialize radii based on minimal spacing
    radius = 0.1
    radii[:] = radius
    
    # Define optimization variables
    bounds = [(0, 1) for _ in range(2 * n)]
    x = np.concatenate((centers.flatten(), radii))
    
    # Objective function to maximize sum of radii
    def objective(x):
        centers_flat = x[:2 * n].reshape(n, 2)
        radii_flat = x[2 * n:]
        return -np.sum(radii_flat)
    
    # Constraints: non-overlapping and within square
    constraints = []
    
    # Boundary constraint
    def boundary_cons(x):
        centers_flat = x[:2 * n].reshape(n, 2)
        radii_flat = x[2 * n:]
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return 1
        return 0
    
    constraints.append({'type': 'ineq', 'fun': boundary_cons})
    
    # Pairwise non-overlapping constraint
    def non_overlap_cons(x):
        centers_flat = x[:2 * n].reshape(n, 2)
        radii_flat = x[2 * n:]
        for i in range(n):
            for j in range(i + 1, n):
                x1, y1 = centers_flat[i]
                x2, y2 = centers_flat[j]
                r1 = radii_flat[i]
                r2 = radii_flat[j]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                if dist < r1 + r2 - 1e-12:
                    return 1
        return 0
    
    constraints.append({'type': 'ineq', 'fun': non_overlap_cons})
    
    # Optimize with bounds and constraints
    result = optimize.minimize(
        objective,
        x,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'disp': False}
    )
    
    # Extract result
    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = result.x[2 * n:]
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii
