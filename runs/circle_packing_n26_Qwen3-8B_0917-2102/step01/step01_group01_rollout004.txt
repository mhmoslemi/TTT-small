import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.cluster.hierarchy import linkage, fcluster
from itertools import combinations

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraint_func(x):
        centers_flat = x.reshape(-1, 2)
        dist_matrix = squareform(pdist(centers_flat))
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return 1
        return 0

    def run_hybrid_packing():
        n = 26
        centers = []
        radii = []

        # Initial hexagonal grid with rows and columns
        for i in range(n):
            row = i // 5
            col = i % 5
            x = col / 4.0 + 0.1
            y = row / 4.0 + 0.1
            r = 0.05
            centers.append([x, y])
            radii.append(r)

        centers = np.array(centers)
        radii = np.array(radii)

        # Add extra circles near boundaries for optimization
        max_iter = 100
        for _ in range(max_iter):
            dist_matrix = squareform(pdist(centers))
            idx = np.argmin(dist_matrix)
            i, j = np.unravel_index(idx, dist_matrix.shape)
            centers[i] = [0.5, 0.5]
            radii[i] = 0.1

        return centers, radii

    centers, radii = run_hybrid_packing()
    x0 = centers.reshape(-1)
    result = minimize(objective, x0, constraints={"type": "ineq", "fun": constraint_func}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
    centers = result.x.reshape(-1, 2)
    radii = np.clip(radii, 0, 1)

    return centers, radii, np.sum(radii)
