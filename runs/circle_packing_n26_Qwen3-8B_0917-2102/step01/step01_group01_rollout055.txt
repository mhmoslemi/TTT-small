import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.cluster.hierarchy import linkage, fcluster

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)

    # Initial hexagonal packing with boundary adjustments
    for i in range(n):
        if i < 8:
            row = i // 4
            col = i % 4
            x = col / 4.0 + 0.05
            y = row / 4.0 + 0.05
            r = 0.05
            centers[i] = [x, y]
            radii[i] = r
        elif i >= 8 and i < 16:
            row = (i - 8) // 4
            col = (i - 8) % 4
            x = col / 4.0 + 0.15
            y = row / 4.0 + 0.15
            r = 0.05
            centers[i] = [x, y]
            radii[i] = r
        else:
            row = (i - 16) // 4
            col = (i - 16) % 4
            x = col / 4.0 + 0.25
            y = row / 4.0 + 0.25
            r = 0.05
            centers[i] = [x, y]
            radii[i] = r

    def objective(radii):
        return -np.sum(radii)

    def constraint(x):
        centers_flat = x.reshape(-1, 2)
        dist_matrix = squareform(pdist(centers_flat))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return 1
        return 0

    def boundary_constraint(x):
        centers_flat = x.reshape(-1, 2)
        for i in range(n):
            x, y = centers_flat[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return 1
        return 0

    bounds = [(0, 1) for _ in range(n * 2)]
    x0 = centers.reshape(-1)
    result = minimize(objective, x0, constraints=[
        {"type": "ineq", "fun": constraint},
        {"type": "ineq", "fun": boundary_constraint}
    ], bounds=bounds, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})

    centers = result.x.reshape(-1, 2)
    radii = np.clip(radii, 0, 1)
    return centers, radii, np.sum(radii)
