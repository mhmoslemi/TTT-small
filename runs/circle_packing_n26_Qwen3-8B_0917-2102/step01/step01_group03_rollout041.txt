import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    # Initial hexagonal packing with 26 circles
    # First row: 1 circle
    # Second row: 2 circles
    # Third row: 3 circles
    # Fourth row: 4 circles
    # Fifth row: 5 circles
    # Sixth row: 5 circles (to reach 26)
    
    # Generate initial guess for centers in hexagonal packing
    rows = [1, 2, 3, 4, 5, 5]
    cols = [1, 2, 3, 4, 5, 5]
    centers = []
    for i, r in enumerate(rows):
        for j in range(r):
            x = (j + 0.5) / (np.sum(rows) + 1)
            y = (i + 0.5) / (np.sum(rows) + 1)
            centers.append((x, y))
    centers = np.array(centers)
    
    # Initial radii guess: same for all circles
    radii = np.full(len(centers), 0.05)
    
    # Optimization bounds for centers
    bounds_centers = [(0, 1) for _ in range(2 * len(centers))]
    
    # Optimization bounds for radii
    bounds_radii = [(0, 0.2) for _ in range(len(centers))]
    
    # Objective function: maximize sum of radii
    def obj(radii_vec):
        return -np.sum(radii_vec)
    
    # Constraints: non-overlapping circles and within unit square
    def constraints(x):
        centers_flat = x[:2 * len(centers)].reshape(len(centers), 2)
        radii_flat = x[2 * len(centers):]
        
        # Check boundaries
        for i in range(len(centers_flat)):
            x, y = centers_flat[i]
            r = radii_flat[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return -np.inf
        # Check circle overlaps
        for i in range(len(centers_flat)):
            for j in range(i + 1, len(centers_flat)):
                dist = np.sqrt(np.sum((centers_flat[i] - centers_flat[j])**2))
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return -np.inf
        return np.sum(radii_flat)
    
    # Combine centers and radii into a single vector for optimization
    x0 = np.concatenate([centers.ravel(), radii])
    result = minimize(obj, x0, bounds=Bounds(bounds_centers + bounds_radii), constraints=constraints)
    
    centers_opt = result.x[:2 * len(centers)].reshape(len(centers), 2)
    radii_opt = result.x[2 * len(centers):]
    
    return centers_opt, radii_opt, np.sum(radii_opt)
