import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial hexagonal packing with 26 circles
    # First row: 13 circles
    # Second row: 13 circles, offset by 0.5 in x
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    row_height = np.sqrt(3)/2  # Height of equilateral triangle
    col_width = 1.0 / (np.ceil(np.sqrt(n)))  # Column spacing for hexagonal packing

    # First row with 13 circles
    for i in range(13):
        centers[i, 0] = i * col_width
        centers[i, 1] = 0.5
        radii[i] = 0.1

    # Second row with 13 circles, offset
    for i in range(13):
        centers[13 + i, 0] = i * col_width + 0.5 * col_width
        centers[13 + i, 1] = 0.5 + row_height
        radii[13 + i] = 0.1

    # Adjust positions to fit within [0, 1]^2 and optimize radii
    def objective_func(radii):
        return -np.sum(radii)

    def constraint_func(radii):
        # Boundary constraints
        for i in range(n):
            if centers[i, 0] - radii[i] < 0 or centers[i, 0] + radii[i] > 1:
                return False
            if centers[i, 1] - radii[i] < 0 or centers[i, 1] + radii[i] > 1:
                return False
        # Circle-circle constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    constraints = [{'type': 'ineq', 'fun': constraint_func}]
    bounds = [(0, 1) for _ in range(n)]

    result = minimize(objective_func, radii, bounds=bounds, constraints=constraints, method='SLSQP')

    if not result.success:
        raise RuntimeError("Optimization failed")

    # Final packing
    return (centers, result.x, np.sum(result.x))
