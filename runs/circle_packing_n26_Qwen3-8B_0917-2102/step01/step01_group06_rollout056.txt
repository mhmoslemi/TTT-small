import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    n = 26
    # Initial hexagonal packing with spacing
    centers = []
    radii = []
    for i in range(n):
        row = i // 7
        col = i % 7
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers.append([x, y])
    centers = np.array(centers)
    radii = np.ones(n) * 0.1  # Initial guess

    def obj_func(r):
        return -np.sum(r)  # Maximize sum of radii

    def constraints(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return False
        for i in range(n):
            if centers[i, 0] - r[i] < -1e-8 or centers[i, 0] + r[i] > 1 + 1e-8:
                return False
            if centers[i, 1] - r[i] < -1e-8 or centers[i, 1] + r[i] > 1 + 1e-8:
                return False
        return True

    # Optimization
    bounds = Bounds(lb=0, ub=np.inf)
    result = minimize(obj_func, radii, method='SLSQP', bounds=bounds, constraints={'type': 'success', 'fun': constraints})
    radii = result.x

    # Final check and return
    for i in range(n):
        x, y = centers[i]
        if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8:
            radii[i] = max(0, 1 - x)
        if y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8:
            radii[i] = max(0, 1 - y)

    return centers, radii, np.sum(radii)
