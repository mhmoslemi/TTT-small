import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance_matrix

def run_packing():
    # Initial hexagonal grid with 26 circles
    n = 26
    centers = []
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (i % 5 == 0)
        y = row + 0.5 * (i % 5 == 0)
        centers.append([x, y])
    centers = np.array(centers)
    
    # Initial radii based on inter-circle distance
    dist_matrix = distance_matrix(centers, centers)
    diag = np.diag_indices(n)
    dist_matrix[diag] = np.inf
    r0 = np.min(dist_matrix) / 2
    radii = np.full(n, r0)
    
    # Optimization function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)
    
    # Constraints: circles must not overlap and stay within unit square
    def constraint_overlap(radii):
        dist_matrix = distance_matrix(centers, centers)
        diag = np.diag_indices(n)
        dist_matrix[diag] = np.inf
        return np.sum(dist_matrix < radii[:, np.newaxis] + radii[np.newaxis, :])

    def constraint_boundary(radii):
        return np.sum(centers[:, 0] - radii < 0) + np.sum(centers[:, 1] - radii < 0)
    
    # Optimization
    res = minimize(
        objective,
        radii,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n)],
        constraints=[
            {'type': 'ineq', 'fun': constraint_overlap},
            {'type': 'ineq', 'fun': constraint_boundary}
        ]
    )
    
    centers = np.clip(centers, 0, 1)
    radii = res.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
