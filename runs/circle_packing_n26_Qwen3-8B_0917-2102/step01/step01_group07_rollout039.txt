import numpy as np
import math

def run_packing():
    # Hexagonal packing for 26 circles in a unit square

    # Define positions for hexagonal packing
    # First row: 1 circle
    # Second row: 2 circles
    # Third row: 3 circles
    # Fourth row: 4 circles
    # Fifth row: 6 circles
    # Sixth row: 5 circles
    # Seventh row: 2 circles
    # Total: 1+2+3+4+6+5+2 = 23, add 3 more to reach 26

    def generate_hexagonal_positions(rows):
        positions = []
        for r in range(rows):
            for c in range(r + 1):
                x = c + 0.5 * r
                y = r * math.sqrt(3) / 2
                positions.append((x, y))
        return positions

    base_positions = generate_hexagonal_positions(6)
    # Add 3 more circles in the last row
    added_positions = [(6.5, 0.1), (6.5, 0.3), (6.5, 0.5)]
    all_positions = base_positions + added_positions

    # Normalize coordinates to fit within [0,1]^2
    max_x = max(p[0] for p in all_positions)
    max_y = max(p[1] for p in all_positions)
    centers = np.array([[p[0] / max_x, p[1] / max_y] for p in all_positions])

    # Calculate radii based on minimum inter-circle distance and boundary distance
    n = len(centers)
    min_dist = np.inf
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist < min_dist:
                min_dist = dist
    min_dist = min(min_dist, 1 - np.max(centers, axis=0))

    radii = np.full(n, min_dist)

    # Ensure all circles are within the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1):
            radii[i] = min(r, 1 - max(x, 1 - x, y, 1 - y))

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
