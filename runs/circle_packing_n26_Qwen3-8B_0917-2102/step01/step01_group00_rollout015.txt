import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal grid with boundary-aware placement
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = 0.1

    # Convert to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(x):
        return -np.sum(x)

    def constraint(x):
        return np.sum(x) - 0.01

    cons = [{'type': 'ineq', 'fun': constraint}]

    def perturb_centers(centers, factor=0.1):
        return centers + np.random.normal(0, factor, centers.shape)

    best_sum = 0
    best_centers = np.zeros((n, 2))
    best_radii = np.zeros(n)

    for _ in range(5):
        perturbed = perturb_centers(centers)
        result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1) for _ in range(n)])
        if not result.success:
            continue
        radii = result.x
        dists = squareform(pdist(perturbed))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j]:
                    ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                    radii[i] *= ratio
                    radii[j] *= ratio

        # Final check to ensure all circles are within [0,1]^2
        for i in range(n):
            x, y = perturbed[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                radii[i] = 0.0

        sum_radii = np.sum(radii)
        if sum_radii > best_sum:
            best_sum = sum_radii
            best_centers[:] = perturbed[:]
            best_radii[:] = radii[:]

    return best_centers, best_radii, best_sum
