<strategy>Use hexagonal packing with adjusted circle sizes near boundaries for optimal space utilization.</strategy>

```python
import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial guess for centers using hexagonal packing
    def hexagonal_centers(n):
        centers = []
        for i in range(n):
            row = i // 5
            col = i % 5
            x = col + 0.5 * (i % 2)
            y = row + (np.sqrt(3)/2) * (i % 2)
            centers.append([x, y])
        return np.array(centers)

    centers = hexagonal_centers(26)
    radii = np.full(26, 0.1)

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        constraints = []
        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return False
            constraints.append((x - r) >= 0)
            constraints.append((x + r) <= 1)
            constraints.append((y - r) >= 0)
            constraints.append((y + r) <= 1)
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    result = minimize(objective, radii, method='SLSQP', constraints={'type': 'ineq', 'fun': constraints})

    final_radii = result.x
    final_centers = centers

    sum_radii = np.sum(final_radii)
    return final_centers, final_radii, sum_radii
```
