import numpy as np
from scipy.optimize import Bounds, minimize
import random

    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    
    # First row
    for i in range(13):
        centers[i, 0] = i / 12
        centers[i, 1] = 0.1
        radii[i] = 0.1

        centers[13 + i, 0] = i / 12 + 0.5 / 12

        radii[13 + i] = 0.15

    bounds_centers = Bounds(np.zeros((26, 2)), np.ones((26, 2)))
    bounds_radii = Bounds(np.zeros(26), np.inf)

        radii_flat = x[52:]

    # Constraints: boundary and non-overlapping
    def constraint(x):
        centers_flat = x[:52].reshape((26, 2))
        radii_flat = x[52:]
        for i in range(26):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
