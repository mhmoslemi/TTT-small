def run_packing():

    # Use a hybrid approach: 1 row of 1, 2 rows of 2, 3 rows of 3, 4 rows of 4, and 5 rows of 5 with extra circles

                dx = centers[i, 0] - centers[j, 0]

                dist = np.sqrt(dx**2 + dy**2)

    final_radii = np.maximum(final_radii, 0)
