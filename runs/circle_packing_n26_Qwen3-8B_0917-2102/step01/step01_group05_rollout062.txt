from scipy.optimize import minimize

def run_packing():

    rows = 5
    cols = 5
    centers = []
    radii = []
    for i in range(rows):
        for j in range(cols):
            x = j + 0.5 * (i % 2)
            y = i * np.sqrt(3) / 2
            centers.append([x, y])
            radii.append(0.05)
    # Adjust for boundary and overlap
    def objective(radii):
        return -sum(radii)
    def constraints(radii):
        cons = []
        for i in range(len(centers)):
            cons.append((radii[i], 0, 1))
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                cons.append((radii[i] + radii[j], dist, 1))
        return cons
    res = minimize(lambda r: -sum(r), np.array([0.05]*len(radii)), method='SLSQP', constraints=constraints)
    radii = res.x
    return np.array(centers), radii, sum(radii)
