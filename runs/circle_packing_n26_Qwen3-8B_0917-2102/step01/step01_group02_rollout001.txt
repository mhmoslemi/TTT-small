import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal packing with boundary alignment
    centers = []
    r = 0.0
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col / 4.5
        y = row / (np.sqrt(3) * 1.5)
        if i % 2 == 0:
            y += 0.5 / np.sqrt(3)
        centers.append([x, y])
    centers = np.array(centers)
    
    # Initial radii based on minimum spacing
    radii = np.full(n, 0.05)
    
    def objective(r):
        return -np.sum(r)
    
    def constraint_min_spacing(r):
        dists = squareform(pdist(centers))
        return np.min(dists - (r[:, np.newaxis] + r[np.newaxis, :])) - 1e-12
    
    def constraint_boundary(r):
        return np.sum(r) - 2.636
    
    cons = ({
        'type': 'ineq',
        'fun': constraint_min_spacing
    }, {
        'type': 'eq',
        'fun': constraint_boundary
    })
    
    bounds = Bounds(np.zeros(n), np.full(n, 1.0))
    
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)
    
    if not result.success:
        raise RuntimeError("Optimization failed")
    
    radii = result.x
    
    # Final validation and adjustment
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = np.max([0, 1 - np.max([x, 1 - x, y, 1 - y])])
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                radii[i] = (dist - 1e-12) / 2
                radii[j] = (dist - 1e-12) / 2
    
    return centers, radii, np.sum(radii)
