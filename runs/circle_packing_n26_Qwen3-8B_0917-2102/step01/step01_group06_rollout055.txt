import numpy as np
from scipy.optimize import minimize

def run_packing():
    np.random.seed(42)
    
    # Hexagonal grid with 5 rows for 26 circles
    num_rows = 5
    num_cols = 5
    total_circles = num_rows * num_cols + 1  # 5x5 grid + 1 center
    
    # Generate hexagonal grid points and one central circle
    centers = []
    for i in range(num_rows):
        for j in range(num_cols):
            x = (j + 0.5) / num_cols
            y = (i + 0.5) / num_rows
            centers.append([x, y])
    centers.append([0.5, 0.5])  # Central circle
    centers = np.array(centers)
    
    # Initial guess for radii: uniform
    radii = np.full(len(centers), 0.1)
    
    # Optimization objective: maximize sum of radii
    def objective(radii_flat):
        return -np.sum(radii_flat)  # Minimize negative sum
    
    # Constraints: circles must not overlap and stay within unit square
    def constraint_overlap(radii_flat):
        radii = radii_flat.reshape(-1)
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return 1  # Violation
        return 0
    
    def constraint_boundary(radii_flat):
        radii = radii_flat.reshape(-1)
        for i in range(len(radii)):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8:
                return 1
            if y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e8:
                return 1
        return 0
    
    # Optimization problem
    bounds = [(0, 1)] * len(radii)
    cons = [{'type': 'ineq', 'fun': lambda r: -constraint_overlap(r)},
            {'type': 'ineq', 'fun': lambda r: -constraint_boundary(r)}]
    
    result = minimize(objective, radii, bounds=bounds, constraints=cons, method='SLSQP')
    
    return (centers, result.x, -result.fun)
