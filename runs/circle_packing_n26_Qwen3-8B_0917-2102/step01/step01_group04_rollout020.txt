import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial hexagonal grid packing with 26 circles
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Generate hexagonal lattice with adjusted rows
    rows = 6
    cols = 5
    for i in range(rows):
        for j in range(cols):
            x = j + 0.5 * (i % 2)
            y = i * np.sqrt(3) / 2
            if i % 2 == 1:
                x += 0.5
            centers[i * cols + j] = [x, y]
    
    # Scale to unit square and adjust radii
    max_dist = np.max(np.linalg.norm(centers[1:] - centers[0], axis=1))
    radii = np.full(n, 0.1)  # Initial guess
    radii[0] = max_dist * 0.5
    
    # Define objective function
    def objective(radii):
        return -np.sum(radii)
    
    # Define constraints: non-overlapping and within bounds
    def constraint(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
                return -1
        return 0
    
    # Optimization
    bounds = [(0, 1) for _ in range(n)]
    
    res = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraint})
    
    # Final check and return
    return centers, res.x, np.sum(res.x)
