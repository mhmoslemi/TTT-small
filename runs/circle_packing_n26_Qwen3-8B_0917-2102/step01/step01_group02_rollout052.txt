import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.optimize import LinearConstraint

def run_packing():

    # Initial hexagonal packing with alternating row offsets
    centers = []
    for i in range(n):
        row = i // 13
        col = i % 13
        x = col / (np.sqrt(3) * 1)
        y = row * (2 / np.sqrt(3)) + (col % 2) * (1 / np.sqrt(3))
        centers.append([x, y])

    radii = np.full(n, 0.05)

        return -np.sum(r)

        raise RuntimeError("Optimization failed")
    
    radii = result.x
    
    # Adjust centers to fit within [0,1]^2
    centers -= np.min(centers, axis=0)
    centers /= np.max(centers, axis=0)
    
    # Final validation and adjustment
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

                radii[i] = (dist - 1e-12) / 2
                radii[j] = (dist - 1e-12) / 2
    
    return centers, radii, np.sum(radii)
