def run_packing():

    n = 26

    radii = np.full(n, 0.05)

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        constraints = []
        for i in range(n):
            constraints.append((radii[i] >= 0))
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                constraints.append(dist >= radii[i] + radii[j] - 1e-8)
        for i in range(n):
            x, y = centers[i]
            constraints.append(radii[i] <= 1 - x)
            constraints.append(radii[i] <= x)
            constraints.append(radii[i] <= 1 - y)
            constraints.append(radii[i] <= y)
        return constraints

    bounds = [(0, 1) for _ in range(n)]
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'ftol': 1e-10, 'maxiter': 1000}
    )

    centers = np.random.rand(n, 2)
    radii = result.x
    total = np.sum(radii)
    return centers, radii, total
