import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.cluster.hierarchy import linkage, fcluster

def run_packing():
    n = 26
    centers = []
    radii = []

    # Initialize with a hexagonal arrangement
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col / 4.5 + 0.05
        y = row / 4.5 + 0.05
        r = 0.05
        centers.append([x, y])
        radii.append(r)

    centers = np.array(centers)
    radii = np.array(radii)

    # Define objective function
    def objective(radii):
        return -np.sum(radii)

    # Define constraints for boundaries and non-overlapping
    def constraint(x):
        centers_flat = x.reshape(-1, 2)
        dist_matrix = squareform(pdist(centers_flat))
        for i in range(len(radii)):
            x_i, y_i = centers_flat[i]
            r_i = radii[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return 1
            for j in range(i + 1, len(radii)):
                dist = dist_matrix[i, j]
                if dist < radii[i] + radii[j] - 1e-8:
                    return 1
        return 0

    # Optimization
    x0 = centers.reshape(-1)
    result = minimize(objective, x0, constraints={"type": "ineq", "fun": constraint}, method="SLSQP")

    centers = result.x.reshape(-1, 2)
    radii = np.clip(radii, 0, 1)

    # Final refinement to ensure boundaries and non-overlapping
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            radii[i] = max(0, x)
        elif x + r > 1:
            radii[i] = max(0, 1 - x)
        if y - r < 0:
            radii[i] = max(0, y)
        elif y + r > 1:
            radii[i] = max(0, 1 - y)

    dist_matrix = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                radii[i] = max(0, (dist_matrix[i, j] - radii[j]) * 0.5)
                radii[j] = max(0, (dist_matrix[i, j] - radii[i]) * 0.5)

    return centers, radii, np.sum(radii)
