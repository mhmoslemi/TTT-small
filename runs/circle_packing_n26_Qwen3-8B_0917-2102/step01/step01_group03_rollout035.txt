import scipy.optimize as opt

    cols = 5
    spacing = 1.0 / (np.sqrt(3) * (cols - 1))

            radii[i] = y
        if y + r > 1:
            radii[i] = 1 - y

        return 0
    
    bounds = [(0, 1) for _ in range(num_circles)]
    
    result = opt.minimize(
        objective,

            radii[i] = y
