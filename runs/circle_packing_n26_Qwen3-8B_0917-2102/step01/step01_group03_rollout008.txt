    initial_sum = np.sum(radii)

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

        n = len(radii)
        for i in range(n):
            for j in range(i + 1, n):

                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
