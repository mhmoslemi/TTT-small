import numpy as np
from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d

def run_packing():
    np.random.seed(42)
    n = 26
    max_iter = 100
    tol = 1e-6

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii, centers):
        cons = []
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - 1e-12})
        for i in range(n):
            for j in range(i + 1, n):
                cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: np.sqrt(np.sum((centers[i] - centers[j])**2)) - r[i] - r[j] + 1e-12})
        return cons

    def place_centers():
        x = np.random.rand(n)
        y = np.random.rand(n)
        return np.column_stack((x, y))

    centers = place_centers()
    radii = np.full(n, 0.1)

    for _ in range(max_iter):
        cons = constraints(radii, centers)
        res = minimize(objective, radii, method='SLSQP', constraints=cons, bounds=[(0, 1) for _ in range(n)])
        radii = res.x
        if res.success:
            # Improve placement with Voronoi refinement
            vor = Voronoi(centers)
            new_centers = np.array([vor.vertices[v] for v in vor.point_region])
            new_centers = np.vstack((new_centers, centers))
            new_centers = np.vstack((new_centers, np.random.rand(n, 2)))
            centers = np.unique(new_centers, axis=0)[:n]
        else:
            break

    return centers, radii, np.sum(radii)
