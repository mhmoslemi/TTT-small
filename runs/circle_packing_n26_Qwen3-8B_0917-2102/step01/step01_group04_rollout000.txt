import numpy as np
import math
from scipy.optimize import minimize

def run_packing():
    # Hexagonal packing with staggered rows
    num_circles = 26
    max_radius = 0.15  # Initial guess
    centers = []
    radii = []
    
    # First row: 13 circles
    for i in range(13):
        x = i * (2 * max_radius) + max_radius
        y = max_radius
        centers.append([x, y])
        radii.append(max_radius)
    
    # Second row: 13 circles, offset and smaller
    for i in range(13):
        x = i * (2 * max_radius) + max_radius
        y = 1 - max_radius
        centers.append([x, y])
        radii.append(max_radius * 0.9)
    
    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)
    
    # Optimization to maximize sum of radii under constraints
    def objective(radii):
        return -np.sum(radii)
    
    def constraint1(radii):
        return np.sum(radii)
    
    def constraint2(radii):
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return -np.inf
        return 1.0
    
    # Initial guess for radii
    initial_radii = np.array([max_radius] * 13 + [max_radius * 0.9] * 13)
    
    # Optimization
    result = minimize(
        objective,
        initial_radii,
        constraints=[
            {'type': 'ineq', 'fun': constraint2},
            {'type': 'eq', 'fun': lambda x: np.sum(x) - np.sum(initial_radii)}
        ],
        bounds=[(0, 1.0) for _ in range(num_circles)]
    )
    
    radii = result.x
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
