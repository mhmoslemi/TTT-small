from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():

    r_initial = 0.1
    radii = np.full(num_circles, r_initial)
    
    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization
    
    # Constraints: non-overlapping and inside unit square
    def constraint_overlap(radii, centers):
        n = len(radii)
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0
    
    def constraint_boundary(radii, centers):
        n = len(radii)
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8 or y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8:

        return 0
    
    # Optimization
    def optimize():
        bounds = Bounds([0.0] * num_circles, [1.0] * num_circles)

            radii,

            bounds=bounds,
            constraints=[

            ]
        )
