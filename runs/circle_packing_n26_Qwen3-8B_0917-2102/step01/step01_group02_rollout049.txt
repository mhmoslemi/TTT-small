import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r)

    cons = ({'type': 'ineq', 'fun': constraint, 'args': ()})
    bounds = Bounds(np.zeros(n), np.full(n, 1.0))

    def constraint_dist(r, centers):
        dists = squareform(pdist(centers))
        return np.array([dists[i, j] - (r[i] + r[j]) for i in range(n) for j in range(i + 1)])

    def constraint_boundary(r, centers):
        return np.array([1 - np.max([centers[i, 0] + r[i], 1 - centers[i, 0] + r[i],
                                   centers[i, 1] + r[i], 1 - centers[i, 1] + r[i]]) for i in range(n)])

    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)

    if not result.success:
        raise RuntimeError("Optimization failed")

    radii = result.x
    centers = np.random.rand(n, 2)

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            radii[i] = np.max([0, 1 - np.max([x, 1 - x, y, 1 - y])])

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                radii[i] = (dist - 1e-12) / 2
                radii[j] = (dist - 1e-12) / 2

    return centers, radii, np.sum(radii)
