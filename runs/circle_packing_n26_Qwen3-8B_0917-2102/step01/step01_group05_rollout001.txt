import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from itertools import combinations

def run_packing():
    np.random.seed(42)
    num_circles = 26
    centers = np.random.rand(num_circles, 2)
    radii = np.random.rand(num_circles) * 0.1 + 0.05
    initial_sum = np.sum(radii)

    def objective(r):
        return -np.sum(r)

    def constraints(r):
        return np.sum(r)

    bounds = [(0, 1) for _ in range(num_circles)]

    def packing_problem(r):
        radii = r
        for i in range(num_circles):
            x, y = centers[i]
            if (x - radii[i] < 0 or x + radii[i] > 1 or
                y - radii[i] < 0 or y + radii[i] > 1):
                return -np.inf
        for i, j in combinations(range(num_circles), 2):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist < radii[i] + radii[j] - 1e-8:
                return -np.inf
        return np.sum(radii)

    result = minimize(lambda r: -packing_problem(r), x0=radii, bounds=bounds, method='SLSQP')
    radii = result.x
    sum_radii = np.sum(radii)
    centers = np.array([np.array([np.clip(c[0] + (radii[i] / 2), 0, 1), np.clip(c[1] + (radii[i] / 2), 0, 1)]) for i, c in enumerate(zip(range(num_circles), centers))])

    return centers, radii, sum_radii
