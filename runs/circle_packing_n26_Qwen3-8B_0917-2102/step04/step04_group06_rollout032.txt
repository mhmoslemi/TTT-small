import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create a hybrid grid with staggered rows and adjusted spacing
    centers = []
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2)) * 0.9
        y = row + 0.5 * 0.9
        centers.append([x, y])
    centers = np.array(centers)

    # Initial radii guess
    radii = np.full(n, 0.1)

    def objective(r):
        return -np.sum(r)

    def constraint_sum(r):
        return np.sum(r) - 2.636

    def constraint_overlap(r):

        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap}]
    cons.append({'type': 'eq', 'fun': lambda r: np.sum(r) - 2.636})

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final check and adjustment
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):

                result['x'][i] *= ratio
                result['x'][j] *= ratio

        if (x - r < -1e-12 or x + r > 1 + 1e-12

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
