import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate a hybrid hexagonal grid with 5 columns and 6 rows, including some corner adjustments
    centers = []
    for i in range(n):
        row = i // 5
        col = i % 5
        if row % 2 == 0:
            x = col + 0.5 * (1 - 2 * (row % 2))
        else:
            x = col + 0.5
        y = row + 0.5
        if col == 0 and row % 2 == 0:
            x = 0.1
        elif col == 4 and row % 2 == 1:
            x = 0.9
        elif row == 5:
            y = 0.9
        centers.append([x, y])
    centers = np.array(centers)

    # Initial radii based on minimum distance between circles
    dists = squareform(pdist(centers))
    initial_radii = np.min(dists) / 2 * np.ones(n)

    def objective(r):
        return -np.sum(r)

    def constraint_sum(r):
        return np.sum(r) - 2.636

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap}]
    cons.append({'type': 'eq', 'fun': lambda r: np.sum(r) - 2.636})

    result = minimize(objective, initial_radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final check and adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
