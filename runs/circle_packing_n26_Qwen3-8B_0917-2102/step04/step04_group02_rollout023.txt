import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate initial staggered hexagonal grid with 5 columns and 6 rows
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
    
    # Scale to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    # Initial radii based on minimal distances
    dists = squareform(pdist(centers))

        return -np.sum(r)

    def constraint_total(r):
        return np.sum(r) - 2.636

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    # Optimization bounds and constraints
    bounds = [(0, 1)] * n
    cons = [{'type': 'eq', 'fun': constraint_total}]
    cons.append({'type': 'ineq', 'fun': constraint_overlap})

    # Run optimization
    result = minimize(objective, radii, bounds=bounds, constraints=cons, method='SLSQP')

    # Final adjustment to prevent boundary violations
    for i in range(n):

        r = result['x'][i]

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
