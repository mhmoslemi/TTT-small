import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate a more refined hexagonal grid with staggered rows
    centers = []
    for row in range(6):
        for col in range(5):
            x = col + 0.5 * (1 - 2 * (row % 2))
            y = row + 0.5
            centers.append([x, y])
    
    # Add one more circle to reach 26
    centers.append([2.5, 2.5])  # This will be adjusted later
    centers = np.array(centers)
    
    # Normalize to unit square
    centers -= centers.min(axis=0)
    centers /= centers.max(axis=0)
    
    # Initial radii based on grid spacing
    initial_radii = np.zeros(n)
    for i in range(n):
        dists = np.linalg.norm(centers[i] - centers, axis=1)
        initial_radii[i] = np.min(dists[dists > 1e-8]) / 2
    
    def objective(r):
        return -np.sum(r)
    
    def constraint_total(r):
        return np.sum(r) - 2.636
    
    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i+1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    cons = [{'type': 'ineq', 'fun': constraint_overlap}]
    cons.append({'type': 'eq', 'fun': constraint_total})
    
    result = minimize(objective, initial_radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)
    
    # Adjust circles near boundaries
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if x - r < -1e-12:
            result['x'][i] = x - 1e-12
        if x + r > 1 + 1e-12:
            result['x'][i] = 1 - 1e-12
        if y - r < -1e-12:
            result['x'][i] = 1e-12
        if y + r > 1 + 1e-12:
            result['x'][i] = 1 - 1e-12
    
    # Final check and adjustment
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio
    
    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
