import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create a near-optimal hexagonal grid with 5 columns and 6 rows with boundary-aware adjustment
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
    
    # Adjust centers to be within [0,1]^2
    min_x, max_x = np.min(centers[:, 0]), np.max(centers[:, 0])
    min_y, max_y = np.min(centers[:, 1]), np.max(centers[:, 1])
    centers[:, 0] = (centers[:, 0] - min_x) / (max_x - min_x)
    centers[:, 1] = (centers[:, 1] - min_y) / (max_y - min_y)

    # Initial radii based on minimum distance between centers
    dists = squareform(pdist(centers))
    min_dist = np.min(dists[np.triu_indices(n, 1)])
    radii = np.full(n, min_dist / 2.5)

    def objective(r):
        return -np.sum(r)

    def constraint_sum(r):
        return np.sum(r) - 2.636

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap}]
    cons.append({'type': 'eq', 'fun': constraint_sum})

    bounds = [(0, 1)] * n

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=bounds)

    # Final check and adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
