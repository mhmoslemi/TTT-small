import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import squareform, pdist

def run_packing():

    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Create a more efficient hexagonal packing with staggered rows
    row = 0
    col = 0
    for i in range(n):
        if i % 5 == 0:
            row += 1
            col = 0
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = 0.1
        col += 1

    # Scale to unit square while preserving aspect ratio
    def scale_to_unit(centers):
        min_x, max_x = centers[:, 0].min(), centers[:, 0].max()
        min_y, max_y = centers[:, 1].min(), centers[:, 1].max()
        scale_x = (1 - 1e-8) / (max_x - min_x)
        scale_y = (1 - 1e-8) / (max_y - min_y)
        centers[:, 0] = (centers[:, 0] - min_x) * scale_x
        centers[:, 1] = (centers[:, 1] - min_y) * scale_y
        return centers

    centers = scale_to_unit(centers)

    def objective(r):
        return -np.sum(r)

    def constraint_volume(r):
        return np.sum(r) - 2.636

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [
        {'type': 'ineq', 'fun': constraint_overlap},
        {'type': 'eq', 'fun': lambda r: np.sum(r) - 2.636}
    ]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment to ensure boundary constraints
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
