import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Define a more optimized initial configuration with corner and edge circles
    centers = np.zeros((n, 2))
    idx = 0
    # Place circles in corners and edges
    for i in range(4):
        if i == 0:
            centers[idx] = [0, 0]
        elif i == 1:
            centers[idx] = [1, 0]
        elif i == 2:
            centers[idx] = [0, 1]
        else:
            centers[idx] = [1, 1]
        idx += 1

    for i in range(4):
        if i == 0:
            centers[idx] = [0, 0.5]
        elif i == 1:
            centers[idx] = [0.5, 0]
        elif i == 2:
            centers[idx] = [1, 0.5]
        else:
            centers[idx] = [0.5, 1]
        idx += 1

    # Fill remaining with hexagonal packing in the middle
    for row in range(2):
        for col in range(5):
            if row % 2 == 0:
                x = col + 0.5
            else:
                x = col + 0.5 - 0.5
            y = row + 0.5
            centers[idx] = [x, y]
            idx += 1

    # Normalize centers to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 2.636

    cons = [{'type': 'ineq', 'fun': lambda r: np.array([r[i] for i in range(n)])}]
    cons.append({'type': 'eq', 'fun': constraint})

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    def constraint_overlap(r):
        return check_overlap(r)

    cons.append({'type': 'ineq', 'fun': constraint_overlap})

    # Initial radii estimate based on average distance
    avg_dist = np.mean(squareform(pdist(centers)))
    result = minimize(objective, np.full(n, avg_dist / 2), constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final check and adjustment
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
