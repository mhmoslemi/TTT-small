import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create a more compact hexagonal grid with 5 columns and 6 rows
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        # Adjust vertical offset for staggered rows
        y_offset = 0.5 if (row % 2) else 0
        x = col + 0.5
        y = row + 0.5 + y_offset
        centers[i] = [x, y]
    
    # Initial radii based on inter-circle distances
    dists = squareform(pdist(centers))
    min_dist = np.min(dists)
    initial_radii = np.full(n, min_dist / 2)

    def objective(r):
        return -np.sum(r)

    def constraint_total(r):
        return np.sum(r) - 2.636

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap}]
    cons.append({'type': 'eq', 'fun': constraint_total})

    result = minimize(objective, initial_radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final refinement to ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
