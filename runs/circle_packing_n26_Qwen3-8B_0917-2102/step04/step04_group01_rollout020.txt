import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate initial hexagonal grid with 5 columns and 6 rows
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]

    # Scale centers to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    # Initial radii with boundary-aware adjustment
    radii = np.full(n, 0.1)
    for i in range(n):
        x, y = centers[i]
        min_dist_to_boundary = min(x, 1 - x, y, 1 - y)
        radii[i] *= min_dist_to_boundary / 0.5

    def objective(r):
        return -np.sum(r)

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    def constraint_sum(r):
        return np.sum(r) - 2.636

    cons = [{'type': 'ineq', 'fun': constraint_overlap}]
    cons.append({'type': 'eq', 'fun': constraint_sum})

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for boundary constraints
    dists = squareform(pdist(centers))
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        min_dist_to_boundary = min(x - r, 1 - x - r, y - r, 1 - y - r)
        if min_dist_to_boundary < -1e-8:
            result['x'][i] = max(0, result['x'][i] + min_dist_to_boundary + 1e-8)

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
