import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import random

def run_packing():
    n = 26
    # Initial staggered hexagonal grid with 5 columns and 6 rows
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]

    def objective(r):
        return -np.sum(r)

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    def constraint_boundary(r):
        for i in range(n):
            x, y = centers[i]
            r_i = r[i]
            if (x - r_i < -1e-12 or x + r_i > 1 + 1e-12
                or y - r_i < -1e-12 or y + r_i > 1 + 1e-12):
                return -1
        return 0

    def constraint_sum(r):
        return np.sum(r) - 2.636

    cons = [{'type': 'ineq', 'fun': constraint_overlap},
            {'type': 'ineq', 'fun': constraint_boundary},
            {'type': 'eq', 'fun': lambda r: constraint_sum(r)}]

    # Initial guess with uniform radii
    radii = np.full(n, 0.1)

    # Use simulated annealing with bounds
    result = minimize(objective, radii, bounds=[(0, 1)] * n, constraints=cons, method='SLSQP')

    # Final check and adjustment
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
            or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
