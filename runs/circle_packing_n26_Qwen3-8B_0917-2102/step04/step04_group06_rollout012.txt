import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial staggered hexagonal grid with variable column counts
    cols = [5, 5, 5, 5, 5]  # 5 columns for 25 circles, 1 extra for 26th
    rows = [5, 5, 5, 5, 5]  # 5 rows to fit 25 circles
    centers = np.zeros((n, 2))
    for i in range(n):
        if i < 25:
            col = i % 5
            row = i // 5
            x = col + 0.5 * (1 - 2 * (row % 2))
            y = row + 0.5
        else:
            # Place the 26th circle in a corner with maximum margin
            x, y = 0.1, 0.1
        centers[i] = [x, y]
    radii = np.zeros(n)

    # Scale centers to [0,1]^2
    min_coords = centers.min(axis=0)
    max_coords = centers.max(axis=0)
    centers = (centers - min_coords) / (max_coords - min_coords)

    def objective(r):
        return -np.sum(r)

    def constraint_sum(r):
        return np.sum(r) - 2.636

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap}]
    cons.append({'type': 'eq', 'fun': constraint_sum})

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final check and adjustment
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
