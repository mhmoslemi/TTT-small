import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create a hybrid hexagonal and nested arrangement
    centers = []
    for i in range(n):
        if i < 13:
            row = i // 5
            col = i % 5
            x = col + 0.5 * (1 - 2 * (row % 2))
            y = row + 0.5
        else:
            row = (i - 13) // 5
            col = (i - 13) % 5
            x = 0.5 + col + 0.5 * (1 - 2 * (row % 2))
            y = row + 1.5
        centers.append([x, y])
    centers = np.array(centers)
    
    # Normalize to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    radii = np.full(n, 0.1)

    def objective(r):
        return -np.sum(r)

    def constraint_total(r):
        return np.sum(r) - 2.636

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap}]
    cons.append({'type': 'eq', 'fun': constraint_total})

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Post-processing to ensure circles are within bounds
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
