import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate a non-uniform hexagonal grid with adaptive spacing
    centers = np.zeros((n, 2))
    col_indices = np.array([0, 1, 2, 3, 4, 5])
    row_indices = np.array([0, 1, 2, 3, 4, 5, 6])
    idx = 0
    for row in row_indices:
        for col in col_indices:
            if row % 2 == 0:
                x = col / 5.0
                y = row / 6.0
            else:
                x = (col + 0.5) / 5.0
                y = (row + 0.5) / 6.0
            centers[idx] = [x, y]
            idx += 1
    # Remove excess circles to get exactly 26
    centers = centers[:26]

    # Initial guess for radii
    radii = np.full(n, 0.1)

    def objective(r):
        return -np.sum(r)

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap}]
    cons.append({'type': 'eq', 'fun': lambda r: np.sum(r) - 2.636})

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    # Final check and adjustment for overlaps
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
