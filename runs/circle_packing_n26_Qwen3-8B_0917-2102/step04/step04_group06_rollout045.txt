import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Define a more efficient hybrid packing with staggered rows and adjusted spacing
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Define a staggered hexagonal grid with rows and columns optimized for 26 circles
    row_height = 1.0 / (np.sqrt(3) / 2)
    col_width = 1.0 / (np.sqrt(3) / 2)
    
    # Define a better staggered layout for 26 circles
    row_indices = [0, 1, 2, 3, 4, 5]
    col_indices = [0, 1, 2, 3, 4]
    for i, row in enumerate(row_indices):
        for j, col in enumerate(col_indices):
            if i % 2 == 0:
                x = col * col_width
            else:
                x = (col + 0.5) * col_width
            y = row * row_height
            centers[i * len(col_indices) + j] = [x, y]
            radii[i * len(col_indices) + j] = 0.1
    
    # Scale centers to [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    def objective(r):
        return -np.sum(r)

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    def constraint_sum(r):
        return np.sum(r) - 2.636

    cons = [{'type': 'ineq', 'fun': constraint_overlap}]
    cons.append({'type': 'eq', 'fun': constraint_sum})

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final check and adjustment
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
