import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Define a non-uniform hexagonal grid with dynamic row and column spacing
    centers = np.zeros((n, 2))
    row_heights = np.linspace(0.5, 1, 6)
    col_widths = np.linspace(0.2, 0.8, 5)
    
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col * col_widths[col] + 0.5 * (1 - 2 * (row % 2))
        y = row_heights[row]
        centers[i] = [x, y]
    
    # Scale to unit square
    min_vals = centers.min(axis=0)
    max_vals = centers.max(axis=0)
    centers = (centers - min_vals) / (max_vals - min_vals)

    def objective(r):
        return -np.sum(r)

    def constraint_sum(r):
        return np.sum(r) - 2.636

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap}]
    cons.append({'type': 'eq', 'fun': lambda r: np.sum(r) - 2.636})

    result = minimize(objective, np.full(n, 0.1), constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final check and adjustment
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
