import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

    # Hybrid layout: 5 columns with staggered rows
    centers = []

            y = row + 0.5
            if col < 5 and row < 6:
                centers.append([x, y])
    centers = np.array(centers)

    initial_radii = np.full(n, 0.1)

        return -np.sum(r)

    cons.append({'type': 'eq', 'fun': lambda r: np.sum(r) - 2.636})

    result = minimize(objective, initial_radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)
