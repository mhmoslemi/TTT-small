import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Define a more compact hexagonal packing with 5 columns and 5 rows
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
    
    # Initial radii based on spacing between circles
    initial_radii = np.full(n, 0.1)
    
    def objective(r):
        return -np.sum(r)
    
    def constraint_sum(r):
        return np.sum(r) - 2.636
    
    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    cons = [
        {'type': 'ineq', 'fun': constraint_overlap},
        {'type': 'eq', 'fun': lambda r: np.sum(r) - 2.636}
    ]
    
    result = minimize(objective, initial_radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)
    
    # Final check and adjustment for boundary constraints
    final_radii = result['x'].copy()
    for i in range(n):
        x, y = centers[i]
        r = final_radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12):
            final_radii[i] = 0
    
    # Final check for overlap
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < final_radii[i] + final_radii[j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (final_radii[i] + final_radii[j])
                final_radii[i] *= ratio
                final_radii[j] *= ratio
    
    sum_radii = np.sum(final_radii)
    return centers, final_radii, sum_radii
