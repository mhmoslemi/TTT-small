import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Create non-uniform hexagonal grid with 5 columns and 6 rows

        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5

        min_x, max_x = np.min(centers[:, 0]), np.max(centers[:, 0])
        min_y, max_y = np.min(centers[:, 1]), np.max(centers[:, 1])
        scale_x = (1 - 2 * 1e-12) / (max_x - min_x)
        scale_y = (1 - 2 * 1e-12) / (max_y - min_y)
        scaled = np.zeros_like(centers)
        scaled[:, 0] = (centers[:, 0] - min_x) * scale_x
        scaled[:, 1] = (centers[:, 1] - min_y) * scale_y
        return scaled
    
    centers = boundary_aware_scale(centers)

    radii = np.full(n, 0.1)

    def objective(r):
        return -np.sum(r)

    # Constraint to ensure circles stay within bounds
    def constraint_bounds(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            x, y = centers[i]
            r_i = r[i]
            if (x - r_i < -1e-12 or x + r_i > 1 + 1e-12 or
                y - r_i < -1e-12 or y + r_i > 1 + 1e-12):
                return -1
        return 0

    # Constraint for circle overlaps
    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

            {'type': 'eq', 'fun': lambda r: np.sum(r) - 2.636}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]

                result['x'][i] *= ratio
                result['x'][j] *= ratio

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
