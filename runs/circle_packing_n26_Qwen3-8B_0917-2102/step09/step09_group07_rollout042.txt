import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create a staggered hybrid layout with varying row lengths
    hex_centers = []
    for row in range(4):
        for col in range(5):
            if row % 2 == 0:
                x = col + 0.25
            else:
                x = col + 0.75
            y = row + 0.5
            if row == 0 and col == 4:
                continue  # Skip to make space for corner circle
            hex_centers.append([x, y])
    for row in range(4):
        if row % 2 == 1:
            for col in range(4):
                x = col + 0.6
                y = row + 0.5
                hex_centers.append([x, y])
    centers = np.array(hex_centers)

    min_coords = np.min(centers, axis=0)
    max_coords = np.max(centers, axis=0)
    scale = (1 - 2 * 0.1) / (max_coords - min_coords)
    centers = (centers - min_coords) * scale + 0.1
    centers = np.clip(centers, 0.1, 0.9)

    def objective(r):
        return -np.sum(r)

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

        for i in range(n):
            x, y = centers[i]
            r_val = r[i]
            if (x - r_val < -1e-12 or x + r_val > 1 + 1e-12
                    or y - r_val < -1e-12 or y + r_val > 1 + 1e-12):
                return -1
        return 0

    # Initial guess with adaptive radius based on distances
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    result = minimize(objective, initial_r, method='SLSQP', bounds=[(0, 1)] * n,
                      constraints=[
                          {'type': 'ineq', 'fun': check_overlap},
                          {'type': 'ineq', 'fun': check_boundary}
                      ])

    # Final adjustment for overlap with boundary-aware scaling
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):

    # Ensure circles are within [0,1]^2 with boundary padding

        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
