import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate a staggered packing with boundary-aware placement
    centers = []
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 if row % 2 else 0
        x = col + offset
        y = row + 0.5
        centers.append([x, y])
    centers = np.array(centers)

    # Adjust for boundaries by compressing outer circles
    boundary_padding = 0.1
    min_coords = np.min(centers, axis=0)
    max_coords = np.max(centers, axis=0)
    scale = (1 - 2 * boundary_padding) / (max_coords - min_coords)
    centers = (centers - min_coords) * scale + boundary_padding
    centers = np.clip(centers, boundary_padding, 1 - boundary_padding)

    def objective(r):
        return -np.sum(r)

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    def check_boundary(r):
        for i in range(n):
            x, y = centers[i]
            r_val = r[i]
            if (x - r_val < -1e-12 or x + r_val > 1 + 1e-12
                    or y - r_val < -1e-12 or y + r_val > 1 + 1e-12):
                return -1
        return 0

    cons1 = {'type': 'ineq', 'fun': check_overlap}
    cons2 = {'type': 'ineq', 'fun': check_boundary}
    cons = [cons1, cons2]

    # Initial guess based on min distances
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)

    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for boundary
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
