import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal packing with boundary-aware offset
    hex_centers = []
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 if row % 2 else 0
        x = col + offset
        y = row + 0.5
        hex_centers.append([x, y])
    centers = np.array(hex_centers)

    # Initial scaling to fit within unit square
    min_coords = np.min(centers, axis=0)
    max_coords = np.max(centers, axis=0)
    scale = (1 - 2 * 0.1) / (max_coords - min_coords)
    centers = (centers - min_coords) * scale + 0.1
    centers = np.clip(centers, 0.1, 0.9)

    def objective(r):
        return -np.sum(r)

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    # Initial guess based on minimal distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)

    # Optimization with boundary constraints
    bounds = [(0, 1)] * n
    cons = [{'type': 'ineq', 'fun': check_overlap}]
    result = minimize(objective, initial_r, bounds=bounds, constraints=cons, method='SLSQP')

    # Final adjustment for boundary proximity
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
