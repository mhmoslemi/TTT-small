import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    # Initial hexagonal packing
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 if row % 2 else 0
        x = col + offset
        y = row
        centers[i] = [x, y]
    
    # Scale to fit in unit square with boundary padding
    max_x = np.max(centers[:, 0])
    min_x = np.min(centers[:, 0])
    max_y = np.max(centers[:, 1])
    min_y = np.min(centers[:, 1])
    scale_x = (1 - 2 * 0.1) / (max_x - min_x)
    scale_y = (1 - 2 * 0.1) / (max_y - min_y)
    centers[:, 0] = (centers[:, 0] - min_x) * scale_x + 0.1
    centers[:, 1] = (centers[:, 1] - min_y) * scale_y + 0.1

    def objective(r):
        return -np.sum(r)

    def constraint_sum(r):
        return np.sum(r) - 2.636

    def constraint_overlap(r):
        dists = np.sqrt(((centers[:, np.newaxis, 0] - centers[np.newaxis, :, 0])**2 + 
                        (centers[:, np.newaxis, 1] - centers[np.newaxis, :, 1])**2))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap}, 
            {'type': 'eq', 'fun': constraint_sum}]

    initial_r = np.min(np.sqrt(((centers[:, np.newaxis, 0] - centers[np.newaxis, :, 0])**2 + 
                               (centers[:, np.newaxis, 1] - centers[np.newaxis, :, 1])**2)), axis=1) * 0.9

    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or 
            y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
