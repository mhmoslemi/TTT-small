import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal packing with staggered rows
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (row % 2)
        y = row + 0.5
        centers[i] = [x, y]
    
    # Adjust for unit square with dynamic padding
    def adjust_padding(centers):
        min_coords = np.min(centers, axis=0)
        max_coords = np.max(centers, axis=0)
        padding = 0.05
        scale = (1 - 2 * padding) / (max_coords - min_coords)
        centers = (centers - min_coords) * scale + padding
        centers = np.clip(centers, padding, 1 - padding)
        return centers

    centers = adjust_padding(centers)

    def objective(r):
        return -np.sum(r)

    def constraint_sum(r):
        return np.sum(r) - 2.636

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_sum}]
    cons.append({'type': 'ineq', 'fun': constraint_overlap})

    # Initial guess based on min pairwise distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)

    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Post-optimization adjustment to avoid boundary violations
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
