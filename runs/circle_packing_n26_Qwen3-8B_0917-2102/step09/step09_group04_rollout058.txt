import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Hexagonal packing with boundary-aware spacing
    centers = np.zeros((n, 2))
    for i in range(n):
        col = i // 2
        row = i % 2
        x = col + 0.5 * row
        y = row + 0.5 * (i // 2)
        centers[i] = [x, y]
    
    # Initial radius estimate based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    
    # Objective function to maximize sum of radii
    def objective(r):
        return -np.sum(r)
    
    # Constraints
    cons = []
    cons.append({'type': 'ineq', 'fun': lambda r: np.sum(r)})
    cons.append({'type': 'ineq', 'fun': lambda r: np.sum(r) - 2.636})
    
    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    cons.append({'type': 'ineq', 'fun': check_overlap})
    
    # Optimization
    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)
    
    # Boundary-aware scaling
    min_x, max_x = np.min(centers, axis=0)
    min_y, max_y = np.min(centers, axis=0)
    scale_x = (1 - 2 * 0.1) / (max_x - min_x)
    scale_y = (1 - 2 * 0.1) / (max_y - min_y)
    centers = (centers - np.array([min_x, min_y])) * np.array([scale_x, scale_y]) + 0.1
    centers = np.clip(centers, 0.1, 0.9)
    
    # Final adjustment for boundary and overlap
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio
    
    # Ensure circles are within [0,1]^2 with boundary padding
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0
    
    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
