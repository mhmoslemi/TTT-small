import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Hexagonal packing with boundary-aware offsets
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        col = i // 2
        row = i % 2
        x = col + 0.5 if row == 0 else col + 0.5 - 0.25
        y = row + 0.5
        centers[i] = [x, y]

    cons = []
    cons.append({'type': 'ineq', 'fun': lambda r: np.sum(r)})

                    return -1
        return 0

    cons.append({'type': 'ineq', 'fun': check_overlap})

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for overlap with boundary-aware scaling
    dists = squareform(pdist(centers))

    sum_radii = np.sum(result['x'])
