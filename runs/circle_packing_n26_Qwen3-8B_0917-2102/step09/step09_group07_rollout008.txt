import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create staggered hybrid layout with boundary-aware spacing
    centers = []
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 if row % 2 else 0
        x = col + offset
        y = row + 0.5
        # Apply boundary padding
        x = np.clip(x, 0.1, 0.9)
        y = np.clip(y, 0.1, 0.9)
        centers.append([x, y])
    centers = np.array(centers)

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 2.636

        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    # Initial guess with adaptive radius based on distances
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    result = minimize(objective, initial_r, method='SLSQP', bounds=[(0, 1)] * n,
                      constraints=[
                          {'type': 'ineq', 'fun': lambda r: np.sum(r)},
                          {'type': 'eq', 'fun': constraint},
                          {'type': 'ineq', 'fun': check_overlap}
                      ])

    # Final adjustment for overlap with boundary-aware scaling
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
