from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Create staggered hexagonal packing with boundary-aware spacing
    centers = np.zeros((n, 2))

        row = i // 5
        col = i % 5
        offset = 0.5 if row % 2 else 0
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]

    # Normalize to unit square with padding
    min_coords = np.min(centers, axis=0)
    max_coords = np.max(centers, axis=0)
    scale = (1 - 2 * 0.1) / (max_coords - min_coords)
    centers = (centers - min_coords) * scale + 0.1
    centers = np.clip(centers, 0.1, 0.9)

    def objective(r):
        return -np.sum(r)

        return np.sum(r) - 2.636

    cons = []
    cons.append({'type': 'ineq', 'fun': lambda r: np.sum(r)})
    cons.append({'type': 'eq', 'fun': constraint})

                if dists[i, j] < r[i] + r[j] - 1e-8:

        return 0

    cons.append({'type': 'ineq', 'fun': check_overlap})

    # Initial guess based on minimal distances
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)

        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
