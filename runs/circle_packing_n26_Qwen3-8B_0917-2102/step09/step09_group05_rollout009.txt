from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

        col = i % 5
        offset = 0.5 if row % 2 else 0
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]

        [0.2, 0.2], [0.8, 0.2], [0.2, 0.8], [0.8, 0.8],

    ]

        centers = np.vstack((centers, pos))

        return -np.sum(r)

        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    # Constraints: non-overlapping, boundary within [0,1], and target sum
    cons = [
        {'type': 'ineq', 'fun': check_overlap},
        {'type': 'ineq', 'fun': lambda r: np.sum(r)},
        {'type': 'eq', 'fun': lambda r: np.sum(r) - 2.636}
