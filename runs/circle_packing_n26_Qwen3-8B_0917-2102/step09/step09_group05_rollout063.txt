from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Create a hybrid layout: 5 rows of 5 circles with alternating offsets + 1 corner circle
    centers = np.zeros((n, 2))

        if i == 0:
            centers[i] = [0.1, 0.1]
        elif i == 25:
            centers[i] = [0.9, 0.9]

            row = i // 5
            col = i % 5
            offset = 0.5 if row % 2 else 0
            x = col + offset
            y = row + 0.5
            centers[i] = [x, y]

    initial_r = np.min(dists) * np.ones(n)
    
    def objective(r):
        return -np.sum(r)

        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    # First optimization stage: maximize sum with overlap and boundary constraints
    cons = [
        {'type': 'ineq', 'fun': check_overlap},
        {'type': 'ineq', 'fun': lambda r: np.sum(r)}
    ]
    
    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
