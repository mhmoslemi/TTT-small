import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.ones(n) * 0.05

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 2.636

    cons = [{'type': 'ineq', 'fun': lambda r: np.sum(r)}]
    cons.append({'type': 'eq', 'fun': constraint})

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons.append({'type': 'ineq', 'fun': check_overlap})

    def update_centers(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    direction = (centers[i] - centers[j]) / dists[i, j]
                    centers[i] += direction * (r[i] + r[j] - dists[i, j]) * 0.5
                    centers[j] -= direction * (r[i] + r[j] - dists[i, j]) * 0.5

    for _ in range(5):
        result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)
        radii = result['x']
        update_centers(radii)

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0

    return centers, radii, np.sum(radii)
