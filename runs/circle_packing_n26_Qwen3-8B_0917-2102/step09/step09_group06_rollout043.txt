from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

    centers = np.zeros((n, 2))

        for j in range(row_len):

            y = i + 0.5
            centers[i * row_len + j] = [x, y]

    min_coords = np.min(centers, axis=0)
    max_coords = np.max(centers, axis=0)
    scale = (1 - 2 * 0.05) / (max_coords - min_coords)
    centers = (centers - min_coords) * scale + 0.05

        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 2.636

    cons = [{'type': 'ineq', 'fun': lambda r: np.sum(r)}]
    cons.append({'type': 'eq', 'fun': constraint})

    def check_overlap(r):

        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons.append({'type': 'ineq', 'fun': check_overlap})

    # Initial guess based on distances
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for overlap

                result['x'][i] *= ratio
                result['x'][j] *= ratio

    for i in range(n):
