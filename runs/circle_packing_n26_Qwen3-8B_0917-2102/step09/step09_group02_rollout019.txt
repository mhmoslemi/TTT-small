import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Custom hexagonal-like arrangement with staggered rows
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 if row % 2 else 0
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    # Initial scaling to fit square with padding
    padding = 0.05
    min_coords = np.min(centers, axis=0)
    max_coords = np.max(centers, axis=0)
    scale = (1 - 2 * padding) / (max_coords - min_coords)

    def objective(r):
        return -np.sum(r)

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    # Initial guess based on min pairwise distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)

    # Optimization with bounds and constraints
    cons = [{'type': 'ineq', 'fun': check_overlap}]
    result = minimize(objective, initial_r, bounds=[(0, 1)] * n, constraints=cons, method='SLSQP')

    # Final adjustment for boundary constraints and ensure non-overlap
    dists = squareform(pdist(centers))
    for i in range(n):

    sum_radii = np.sum(result['x'])
