import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    cols = 5
    rows = 6
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // cols
        col = i % cols
        offset = 0.5 if row % 2 else 0
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    # Adjust for unit square
    min_coords = np.min(centers, axis=0)
    max_coords = np.max(centers, axis=0)
    scale = 1 / (max_coords - min_coords)
    centers = (centers - min_coords) * scale
    centers = np.clip(centers, 0, 1)

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 2.636

    cons = [{'type': 'ineq', 'fun': lambda r: np.sum(r)}]
    cons.append({'type': 'eq', 'fun': constraint})

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons.append({'type': 'ineq', 'fun': check_overlap})

    initial_r = np.min(squareform(pdist(centers))) * np.ones(n)
    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for overlap
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    return centers, result['x'], np.sum(result['x'])
