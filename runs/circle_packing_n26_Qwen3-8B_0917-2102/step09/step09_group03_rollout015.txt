import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal lattice with variable row spacing
    centers = []
    row_height = 0.5
    col_width = 0.5
    for row in range(5):
        for col in range(5):
            if row % 2 == 0:
                x = col + 0.5 * col_width
            else:
                x = col + 0.5 * col_width - 0.25 * col_width
            y = row * row_height + 0.5 * row_height
            centers.append([x, y])
    
    # Convert to numpy array
    centers = np.array(centers)
    
    # Initial radius based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    
    def objective(r):
        return -np.sum(r)
    
    def constraint_sum(r):
        return np.sum(r) - 2.636
    
    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    cons = [
        {'type': 'ineq', 'fun': constraint_overlap},
        {'type': 'eq', 'fun': constraint_sum}
    ]
    
    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)
    
    # Boundary check and adjustment
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            result['x'][i] = 0
    
    return centers, result['x'], np.sum(result['x'])
