from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

        row = i // cols
        col = i % cols
        x = col + 0.5 * (row % 2)
        y = row + 0.5
        centers[i] = [x, y]
    
    def objective(r):
        return -np.sum(r)
    
    def constraint(r):
        return np.sum(r) - 2.636
    
    def check_overlap(r):
        dists = squareform(pdist(centers))

                    return -1
        return 0
    
    # Initial guess based on minimal distances
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    
    # Optimization
    cons = [{'type': 'ineq', 'fun': lambda r: np.sum(r)}, 
            {'type': 'eq', 'fun': constraint},
            {'type': 'ineq', 'fun': check_overlap}]
    
    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
