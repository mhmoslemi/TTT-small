from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Hexagonal grid with boundary-aware scaling
    centers = np.zeros((n, 2))
    col = 0
    row = 0
    for i in range(n):

            row += 1
            col = 0
        centers[i] = [col + 0.1, row + 0.1]
        col += 0.3

        return 0

    cons.append({'type': 'ineq', 'fun': check_overlap})
