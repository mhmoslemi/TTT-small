import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Hexagonal packing with staggered rows
    centers = np.zeros((n, 2))
    row = 0
    col = 0
    for i in range(n):
        if i % 2 == 0:
            centers[i] = [col / (n // 2 - 1), row / (n // 2 - 1)]
        else:
            centers[i] = [(col + 0.5) / (n // 2 - 1), row / (n // 2 - 1)]
        col += 1
        if col >= n // 2:
            col = 0
            row += 1

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 2.636

    cons.append({'type': 'ineq', 'fun': lambda r: np.sum(r)})
    cons.append({'type': 'eq', 'fun': constraint})

            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons.append({'type': 'ineq', 'fun': check_overlap})

    # Initial guess with adaptive radius based on distances
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)
