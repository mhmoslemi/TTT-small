from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import random

def run_packing():

    centers = np.random.rand(n, 2)
    # Initial radii based on average nearest neighbor distance
    dists = squareform(pdist(centers))
    initial_r = np.mean(dists, axis=1) * 0.5
    initial_r = np.clip(initial_r, 0, 1)

    def objective(r):
        return -np.sum(r)

    def constraint_sum(r):
        return np.sum(r) - 2.636

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_sum}]
    cons.append({'type': 'ineq', 'fun': constraint_overlap})

    def bounds_func(x):
        return [(0, 1)] * n

    result = minimize(objective, initial_r, bounds=bounds_func, constraints=cons, method='SLSQP')

    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

    sum_radii = np.sum(result['x'])
