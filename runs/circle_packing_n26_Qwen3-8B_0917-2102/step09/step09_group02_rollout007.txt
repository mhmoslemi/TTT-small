from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

    # Create a hexagonal packing with adjusted spacing
    centers = np.zeros((n, 2))

        col = i % 5
        row = i // 5
        offset = 0.5 if row % 2 else 0
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]

    padding = 0.1
    scale = (1 - 2 * padding) / np.max(np.abs(centers))
    centers = centers * scale + padding

        return 0

    cons.append({'type': 'ineq', 'fun': check_overlap})

    # Initial guess with adaptive radius based on distances
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
