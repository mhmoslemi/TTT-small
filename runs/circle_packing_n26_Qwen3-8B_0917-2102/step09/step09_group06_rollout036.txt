import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Initial hexagonal packing with 5 rows, 5 or 6 columns
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (row % 2)
        y = row + 0.5
        centers[i] = [x, y]

    padding = 0.05
    min_coords = np.min(centers, axis=0)
    max_coords = np.max(centers, axis=0)
    scale = (1 - 2 * padding) / (max_coords - min_coords)
    centers = (centers - min_coords) * scale + padding

        return -np.sum(r)

    def constraint_sum(r):
        return np.sum(r) - 2.636

        for i in range(n):

                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap},
            {'type': 'eq', 'fun': constraint_sum}]

    # Initial guess based on distances
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    for i in range(n):

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
