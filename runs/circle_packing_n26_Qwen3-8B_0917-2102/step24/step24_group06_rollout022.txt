import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

    # Generate hexagonal grid with boundary-aware placement
    centers = np.zeros((n, 2))

            row += 1
            col = 0

        centers[i] = [x, y]

    max_coords = np.max(centers, axis=0)

        return -np.sum(r)

    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)

    # Optimization with bounds and constraints
    result = minimize(objective, initial_r, method='SLSQP', bounds=[(0, 1)] * n, constraints=[{'type': 'ineq', 'fun': constraint}])

    return centers, result['x'], sum_radii
