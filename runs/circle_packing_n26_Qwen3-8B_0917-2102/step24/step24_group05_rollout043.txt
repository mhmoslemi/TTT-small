from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Initial guess with hexagonal packing

    row, col = 0, 0
    for i in range(n):
        if i % 5 == 0:
            row += 1
            col = 0
        x = col + 0.5 * (1 - row % 2)
        y = row
        centers[i] = [x, y]
        col += 1

        return -np.sum(r)

        dists = squareform(pdist(centers))
        for i in range(n):

    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)

    def annealing_objective(r):
        return -np.sum(r)

    def annealing_constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):

                    return -np.inf
        return np.sum(r) - 2.636

        annealing_objective,
        initial_r,
        constraints=[{'type': 'ineq', 'fun': lambda r: 0}, {'type': 'eq', 'fun': annealing_constraint}],
