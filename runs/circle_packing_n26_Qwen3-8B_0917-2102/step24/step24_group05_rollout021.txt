import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal grid
    centers = np.zeros((n, 2))
    row, col = 0, 0
    for i in range(n):
        if i % 5 == 0:
            row += 1
            col = 0
        x = col + 0.5 * (1 - row % 2)
        y = row
        centers[i] = [x, y]
        col += 1

    # Compute initial radii based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)

    # Define objective to maximize sum of radii
    def objective(r):
        return -np.sum(r)

    # Define constraints for non-overlapping and boundary

        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        for i in range(n):
            x, y = centers[i]
            r_val = r[i]
            if x - r_val < -1e-12 or x + r_val > 1 + 1e-12 or y - r_val < -1e-12 or y + r_val > 1 + 1e-12:
                return -np.inf
        return np.sum(r) - 2.636

    # Optimize with bounds and constraints
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': constraint}], method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for boundary overlap
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2
    for i in range(n):

        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
