import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create a hexagonal grid with varying row densities
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    # Adjust for boundary-aware placement
    def adjust_for_boundary(c):
        min_x, max_x = np.min(c[:, 0]), np.max(c[:, 0])
        min_y, max_y = np.min(c[:, 1]), np.max(c[:, 1])
        scale_x = (1 - 2 * 0.1) / (max_x - min_x)
        scale_y = (1 - 2 * 0.1) / (max_y - min_y)
        scaled = np.column_stack([(c[:, 0] - min_x) * scale_x + 0.1, (c[:, 1] - min_y) * scale_y + 0.1])
        return np.clip(scaled, 0.1, 0.9)
    
    centers = adjust_for_boundary(centers)

    def objective(r):
        return -np.sum(r)

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636

    # Initial guess with tighter minimum distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': check_overlap}, {'type': 'eq', 'fun': constraint}], method='SLSQP', bounds=[(0, 1)] * n, options={'ftol': 1e-8, 'maxiter': 500})

    # Final adjustment for overlap with boundary-aware scaling
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2 with boundary padding
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
