import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal grid with staggered rows
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    # Compute initial distances
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    
    def objective(r):
        return -np.sum(r)
    
    def check_overlap(r):
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    def constraint(r):
        return np.sum(r) - 2.636
    
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': check_overlap}, {'type': 'eq', 'fun': constraint}], method='SLSQP', bounds=[(0, 1)] * n, options={'ftol': 1e-8, 'maxiter': 500})
    
    # Final adjustment for boundary and overlap
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0
    
    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
