import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal grid with boundary-aware placement
    centers = np.zeros((n, 2))
    row, col = 0, 0
    for i in range(n):
        if i % 5 == 0:
            row += 1
            col = 0
        x = col + 0.5 * (1 - row % 2)
        y = row
        centers[i] = [x, y]
        col += 1
    
    # Boundary-aware scaling
    min_coords = np.min(centers, axis=0)
    max_coords = np.max(centers, axis=0)
    scale = (1 - 2 * 0.1) / (max_coords - min_coords)
    centers = (centers - min_coords) * scale + 0.1
    centers = np.clip(centers, 0.1, 0.9)

    def objective(r):
        return -np.sum(r)

    # Initial guess based on inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)

    # Optimization with constraints
    def constraint(r):
        for i in range(n):
            for j in range(i + 1, n):
                if np.sqrt(np.sum((centers[i] - centers[j]) ** 2)) < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636

    result = minimize(objective, initial_r, method='SLSQP', bounds=[(0, 1)] * n, constraints=[{'type': 'ineq', 'fun': lambda r: 0}, {'type': 'eq', 'fun': constraint}])

    # Post-processing for boundary and overlap
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Final boundary check and adjustment
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
