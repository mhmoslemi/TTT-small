import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Generate adaptive hexagonal grid with boundary-aware placement
    centers = np.zeros((n, 2))
    row, col = 0, 0
    for i in range(n):
        if i % 5 == 0:
            row += 1
            col = 0
        x = col + 0.5 * (1 - row % 2)
        y = row
        centers[i] = [x, y]

    max_coords = np.max(centers, axis=0)

    scale = (1 - 2 * padding) / (max_coords - min_coords)

        return -np.sum(r)

                      constraints=[{'type': 'ineq', 'fun': lambda r: np.min(squareform(pdist(centers)) - (r[:, np.newaxis] + r[np.newaxis, :]) + 1e-8)}])

    # Final refinement for boundary and overlap

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
