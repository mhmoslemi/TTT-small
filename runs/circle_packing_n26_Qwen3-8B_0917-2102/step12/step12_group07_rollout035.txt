import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Adaptive hexagonal grid with variable row and column spacing
    centers = np.zeros((n, 2))
    row_spacing = 0.8
    col_spacing = 0.8
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col * col_spacing + 0.5 * (1 - row % 2) * (col_spacing / 2)
        y = row * row_spacing + 0.5
        centers[i] = [x, y]
    
    # Normalize to [0,1]^2
    centers = (centers - np.min(centers, axis=0)) / (np.max(centers, axis=0) - np.min(centers, axis=0))

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636

    # Initial guess based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': constraint}], method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
