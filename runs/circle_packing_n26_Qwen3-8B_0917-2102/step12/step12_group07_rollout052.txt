import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    # Ensure all centers are within [0,1]^2
    centers = np.clip(centers, 0, 1)

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r)

    initial_r = np.min(pdist(centers)) * np.ones(n)
    bounds = [(0, 1)] * n

    # Use a more robust optimization method with tighter tolerances
    result = minimize(objective, initial_r, bounds=bounds, method='SLSQP', constraints=[{'type': 'ineq', 'fun': lambda r: np.sum(r)}], options={'ftol': 1e-10, 'maxiter': 1000})

    # Final adjustment for boundary and overlap
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
