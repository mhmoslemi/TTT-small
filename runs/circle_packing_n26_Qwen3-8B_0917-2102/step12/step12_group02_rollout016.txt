import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636

    bounds = [(0, 1)] * n

    result = minimize(objective, radii, constraints=[{'type': 'ineq', 'fun': constraint}], method='SLSQP', bounds=bounds)

    # Iteratively refine positions to reduce boundary conflicts
    for _ in range(3):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                    dir_vec = centers[i] - centers[j]
                    dir_vec /= np.linalg.norm(dir_vec)
                    centers[i] += dir_vec * (dists[i, j] - result['x'][i] - result['x'][j]) * 0.5
                    centers[j] -= dir_vec * (dists[i, j] - result['x'][i] - result['x'][j]) * 0.5

    # Final boundary check and adjustment
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    return centers, result['x'], np.sum(result['x'])
