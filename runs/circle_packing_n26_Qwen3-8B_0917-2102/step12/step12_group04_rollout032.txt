from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

        else:
            x = col + 0.25
        y = row + 0.5
        centers[i] = [x, y]

    centers -= np.min(centers, axis=0)
    centers /= np.max(centers, axis=0)

    initial_r = np.min(dists) * np.ones(n)
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': constraint_func}], method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
