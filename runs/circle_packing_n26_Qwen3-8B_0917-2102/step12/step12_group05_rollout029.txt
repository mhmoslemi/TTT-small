from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

    # Create a denser staggered hexagonal grid with variable row spacing

        col = i % 5
        offset = 0.5 * (1 - row % 2)

        y = row + 0.5
        centers[i] = [x, y]

        return np.sum((pos - np.array([0.5, 0.5])) ** 2)

    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    
    def objective_func(r):
        return -np.sum(r)

        dists = squareform(pdist(centers))
        for i in range(n):

                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636
    
    result = minimize(objective_func, initial_r, constraints=[{'type': 'ineq', 'fun': lambda r: 0}, {'type': 'eq', 'fun': constraint_func}],

    for i in range(n):
