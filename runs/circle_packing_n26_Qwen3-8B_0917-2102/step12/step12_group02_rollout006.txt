from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

        offset = 0.5 * (1 - row % 2)

        centers[i] = [x, y]
    
    # Normalize to [0,1]^2
    centers -= np.min(centers, axis=0)
    centers /= np.max(centers, axis=0)

        return -np.sum(r)

        dists = squareform(pdist(centers))
        for i in range(n):

    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    
    # Optimization with bounds and constraints
    result = minimize(objective, initial_r, method='SLSQP', bounds=[(0, 1)] * n,
                      constraints=[{'type': 'ineq', 'fun': constraint_func}])
    
    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
