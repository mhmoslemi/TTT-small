import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate adaptive hexagonal layout with variable row heights
    centers = np.zeros((n, 2))
    row_height = 1.0 / np.sqrt(3)
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row * row_height + 0.5
        centers[i] = [x, y]
    
    # Initial radii based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    
    def objective(r):
        return -np.sum(r)
    
    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636
    
    # Optimization bounds
    bounds = Bounds(0, 1)
    
    # Optimization with adaptive center adjustment
    def adjust_centers(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    delta = (dists[i, j] - r[i] - r[j]) / (r[i] + r[j])
                    centers[i] += delta * (centers[i] - centers[j]) / dists[i, j]
                    centers[j] -= delta * (centers[i] - centers[j]) / dists[i, j]
    
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': constraint}], method='SLSQP', bounds=bounds)
    
    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0
    
    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
