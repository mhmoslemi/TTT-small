import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    row = 0
    col = 0
    for i in range(n):
        if i % 5 == 0:
            row += 1
            col = 0
        x = col + 0.5 * (1 - (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        col += 1

    def objective(r):
        return -np.sum(r)

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636

    initial_r = np.min(squareform(pdist(centers))) * np.ones(n)
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': check_overlap}, {'type': 'eq', 'fun': constraint}], method='SLSQP', bounds=[(0, 1)] * n)

    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):

                result['x'][i] *= ratio
                result['x'][j] *= ratio

    for i in range(n):

        r = result['x'][i]

            result['x'][i] = 0

    return centers, result['x'], np.sum(result['x'])
