import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate staggered hexagonal grid with variable row spacing
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 * (row % 2)
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    # Boundary-aware scaling
    def scale_centers(centers, boundary_pad=0.1):
        min_coords = np.min(centers, axis=0)
        max_coords = np.max(centers, axis=0)
        scale = (1 - 2 * boundary_pad) / (max_coords - min_coords)
        scaled_centers = (centers - min_coords) * scale + boundary_pad
        return np.clip(scaled_centers, boundary_pad, 1 - boundary_pad)
    
    centers = scale_centers(centers)
    
    def objective(r):
        return -np.sum(r)
    
    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636
    
    # Initial guess based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    
    # Optimization with SLSQP
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': lambda r: np.sum(r)}], method='SLSQP', bounds=[(0, 1)] * n)
    
    # Ensure circles are within [0,1]^2 with boundary padding
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0
    
    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
