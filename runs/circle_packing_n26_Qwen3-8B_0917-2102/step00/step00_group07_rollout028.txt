import numpy as np
from scipy.optimize import minimize

def run_packing():
    np.random.seed(42)
    n = 26
    
    # Initial guess: hexagonal packing with boundary correction
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Hexagonal packing with spacing and boundary adjustment
    spacing = 0.15
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col * spacing + spacing / 2
        y = row * spacing * np.sqrt(3) + spacing / 2
        centers[i] = [x, y]
        radii[i] = min(0.05, 0.1 - (x + y) / 2)
    
    # Boundary correction
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            centers[i, 0] = r
        elif x + r > 1:
            centers[i, 0] = 1 - r
        if y - r < 0:
            centers[i, 1] = r
        elif y + r > 1:
            centers[i, 1] = 1 - r
        radii[i] = min(0.1, 0.1 - (x + y) / 2)
    
    # Optimization to increase radii while maintaining non-overlap
    def objective(r):
        return -np.sum(r)
    
    def constraint(r, i, j):
        return np.sqrt(np.sum((centers[i] - centers[j])**2)) - (r[i] + r[j]) >= 0
    
    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: constraint(r, i, j)})
    
    bounds = [(0, 0.2) for _ in range(n)]
    
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=cons,
        method='SLSQP'
    )
    
    radii = result.x
    # Final boundary check and adjust
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            centers[i, 0] = r
        elif x + r > 1:
            centers[i, 0] = 1 - r
        if y - r < 0:
            centers[i, 1] = r
        elif y + r > 1:
            centers[i, 1] = 1 - r
    
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
