import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from itertools import combinations

def run_packing():
    # Hexagonal grid with 5 rows
    rows = 5
    cols = 5
    centers = []
    radii = []

    # Generate hexagonal grid
    for r in range(rows):
        for c in range(cols):
            x = c / (cols - 1)
            y = r / (rows - 1)
            centers.append([x, y])
    
    # Initial radii based on grid spacing
    spacing = 1 / (cols - 1)
    initial_radii = [spacing / 2] * len(centers)
    
    # Define optimization variables
    x = np.array(centers)
    r = np.array(initial_radii)
    
    # Objective function to maximize sum of radii
    def objective(params):
        return -np.sum(params[1:])
    
    # Constraints: circle within square and non-overlapping
    def constraint(params):
        x = params[0]
        r = params[1:]
        for i in range(len(x)):
            xi, yi = x[i]
            ri = r[i]
            if xi - ri < -1e-12 or xi + ri > 1 + 1e-12 or yi - ri < -1e-12 or yi + ri > 1 + 1e-12:
                return -1
        for i in range(len(x)):
            for j in range(i + 1, len(x)):
                dx = x[i, 0] - x[j, 0]
                dy = x[i, 1] - x[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - 1e-12:
                    return -1
        return 0
    
    # Combine variables
    params = np.concatenate([x.flatten(), r])
    
    # Optimize
    result = minimize(objective, params, constraints={"type": "ineq", "fun": constraint})
    
    # Extract results
    optimized_params = result.x
    centers_opt = optimized_params[:2 * len(centers)].reshape((len(centers), 2))
    radii_opt = optimized_params[2 * len(centers):]
    
    # Check if optimization was successful
    if result.success:
        return (centers_opt, radii_opt, np.sum(radii_opt))
    else:
        raise Exception("Optimization failed")
