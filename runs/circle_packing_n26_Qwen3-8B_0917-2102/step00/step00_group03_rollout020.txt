import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.cluster.hierarchy import linkage, fcluster
from itertools import combinations

def run_packing():
    # Initial hexagonal grid with 26 circles
    n = 26
    rows = int(np.sqrt(n))
    cols = n // rows + (1 if n % rows else 0)
    
    # Initial positions for a hexagonal grid
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        centers[i] = [x, y]
    
    # Scale to unit square
    scale = 1.0 / (np.max(centers[:, 0]) - np.min(centers[:, 0]))
    centers = centers * scale
    radii = np.ones(n) * 0.05
    
    # Optimization parameters
    bounds = [(0, 1) for _ in range(2 * n)]
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i] - x[2*i + 1]})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i + 1] - x[2*i]})
    
    def objective(x):
        r = x[2*n:]
        return -np.sum(r)
    
    def constraint_overlap(x):
        centers_flat = x[:2*n].reshape(-1, 2)
        radii_flat = x[2*n:]
        for i, j in combinations(range(n), 2):
            dist = np.linalg.norm(centers_flat[i] - centers_flat[j])
            if dist < radii_flat[i] + radii_flat[j] - 1e-8:
                return dist - radii_flat[i] - radii_flat[j] + 1e-8
        return 0
    
    constraints.append({'type': 'ineq', 'fun': constraint_overlap})
    
    result = minimize(objective, x0=np.concatenate([centers.reshape(-1), np.ones(n)*0.05]),
                      bounds=bounds, constraints=constraints, method='SLSQP', tol=1e-6)
    
    if not result.success:
        raise ValueError("Optimization failed")
    
    centers = result.x[:2*n].reshape(-1, 2)
    radii = result.x[2*n:]
    
    # Final validation
    for i in range(n):
        x, y = centers[i]
        if x < 0 or x > 1 or y < 0 or y > 1:
            raise ValueError("Circle outside unit square")
        if radii[i] < 0:
            raise ValueError("Negative radius")
    
    for i in range(n):
        for j in range(i+1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist < radii[i] + radii[j] - 1e-8:
                raise ValueError("Overlap detected")
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
