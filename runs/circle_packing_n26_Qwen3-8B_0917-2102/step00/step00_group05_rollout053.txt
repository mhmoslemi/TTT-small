import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize

def run_packing():
    # Hexagonal packing base with 5 rows
    rows = 5
    cols = 5
    centers = []
    radii = []
    
    # First row: 5 circles horizontally
    for i in range(cols):
        x = i / (cols - 1)
        y = 0.05
        r = 0.05
        centers.append([x, y])
        radii.append(r)
    
    # Second to fifth rows: hexagonal packing
    for row in range(1, rows):
        for col in range(cols):
            x = (col + 0.5) / (cols - 1)
            y = 0.1 + row * (0.25)
            r = 0.05
            centers.append([x, y])
            radii.append(r)
    
    # Add 21 more circles in the remaining space
    additional_circles = []
    for i in range(21):
        x = np.random.rand()
        y = np.random.rand()
        r = 0.03
        additional_circles.append([x, y, r])
    
    centers += [[x, y] for x, y, r in additional_circles]
    radii += [r for x, y, r in additional_circles]
    
    # Optimize radii and positions
    def objective(x):
        return -sum(x[::2] - x[1::2])
    
    def constraint(x):
        for i in range(len(x)//2):
            for j in range(len(x)//2):
                if i != j:
                    dx = x[2*i] - x[2*j + 1]
                    dy = x[2*i + 1] - x[2*j + 2]
                    dist = np.sqrt(dx**2 + dy**2)
                    if dist < x[2*i + 2] + x[2*j + 3] - 1e-12:
                        return 1
        return 0
    
    constraints = [{'type': 'ineq', 'fun': constraint}]
    bounds = [(0, 1) for _ in range(len(x))]
    
    x0 = np.concatenate([centers, radii])
    result = minimize(objective, x0, constraints=constraints, bounds=bounds)
    
    optimized_centers = result.x[::2]
    optimized_radii = result.x[1::2]
    
    # Ensure centers within [0,1]^2 and radii non-negative
    optimized_centers = np.clip(optimized_centers, 0, 1)
    optimized_radii = np.clip(optimized_radii, 0, 1)
    
    sum_radii = np.sum(optimized_radii)
    return (optimized_centers.reshape(-1, 2), optimized_radii, sum_radii)
