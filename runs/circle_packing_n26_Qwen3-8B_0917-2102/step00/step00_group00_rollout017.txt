from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():

        return -np.sum(radii)

    def constraints(radii):
        constraints = []
        for i in range(n):
            constraints.append(radii[i] >= 0)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                constraints.append(radii[i] + radii[j] <= dist + 1e-8)
        for i in range(n):

            constraints.append(radii[i] <= (1 - x) + 1e-8)

            constraints.append(radii[i] <= x + 1e-8)
            constraints.append(radii[i] <= y + 1e-8)
        return constraints

    bounds = [(0, 1) for _ in range(n)]
    res = minimize(objective, radii, bounds=bounds, constraints=constraints)

    centers = np.array(centers)
    radii = res.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
