import numpy as np
from scipy.optimize import minimize
import random

def run_packing():
    def objective(radii, centers, bounds, constraints):
        return -np.sum(radii)

    def non_overlap_constraints(centers, radii):
        n = len(centers)
        return [(np.sqrt(np.sum((centers[i] - centers[j])**2)) - radii[i] - radii[j]) for i in range(n) for j in range(i+1, n)]

    def boundary_constraints(centers, radii):
        n = len(centers)
        return [(centers[i, 0] - radii[i]) for i in range(n)] + [(1 - centers[i, 0] - radii[i]) for i in range(n)] + \
               [(centers[i, 1] - radii[i]) for i in range(n)] + [(1 - centers[i, 1] - radii[i]) for i in range(n)]

    def initial_centers():
        n = 26
        centers = np.random.rand(n, 2)
        return centers

    centers = initial_centers()
    radii = np.random.rand(26) * 0.1

    bounds = [(0, 1) for _ in range(26 * 2)]
    constraints = non_overlap_constraints(centers, radii) + boundary_constraints(centers, radii)

    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=constraints)

    if not result.success:
        raise ValueError("Optimization failed")

    return centers, result.x, np.sum(result.x)
