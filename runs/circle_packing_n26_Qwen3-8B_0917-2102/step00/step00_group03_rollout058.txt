import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial placement: 5 rows of 5 circles (with 1 spare) + 1 row of 6 circles
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # First 5 rows: 5 circles each
    for i in range(5):
        for j in range(5):
            centers[i * 5 + j] = [j / 4.5, i / 4.5]
    
    # Last row: 6 circles (with 1 circle moved to the end)
    for j in range(6):
        centers[25] = [j / 5.0, 0.5]
    
    # Initial guess for radii
    radii[:25] = 0.1
    radii[25] = 0.1
    
    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization
    
    # Constraint: no overlap between circles
    def constraint_func(radii):
        dist_matrix = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return 1
        return 0
    
    # Optimization
    bounds = [(0, 1) for _ in range(n)]
    cons = {'type': 'ineq', 'fun': constraint_func}
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)
    
    # Final centers and radii
    final_centers = centers
    final_radii = result.x
    
    # Check boundary constraints
    for i in range(n):
        x, y = final_centers[i]
        r = final_radii[i]
        if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
            raise ValueError("Circle out of bounds")
    
    sum_radii = np.sum(final_radii)
    return final_centers, final_radii, sum_radii
