import numpy as np
import scipy.optimize as opt

def run_packing():
    # Hexagonal packing with 5 rows, 5 columns, and 12 boundary circles
    # First, generate hexagonal grid in the center
    rows = 5
    cols = 5
    spacing = 0.25
    centers_hex = []
    radii_hex = []
    
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + 0.5 * spacing
            y = i * spacing * np.sqrt(3) / 2 + 0.5 * spacing
            if i % 2 == 0:
                x = j * spacing + 0.5 * spacing
            else:
                x = (j + 0.5) * spacing
            centers_hex.append([x, y])
            radii_hex.append(0.05)
    
    # Add boundary circles for optimal packing
    boundary_circles = []
    boundary_radii = []
    # Top boundary
    for x in np.linspace(0, 1, 13):
        boundary_circles.append([x, 1 - 0.05])
        boundary_radii.append(0.05)
    # Bottom boundary
    for x in np.linspace(0, 1, 13):
        boundary_circles.append([x, 0.05])
        boundary_radii.append(0.05)
    # Left and right boundaries
    for y in np.linspace(0.05, 1 - 0.05, 12):
        boundary_circles.append([0.05, y])
        boundary_radii.append(0.05)
        boundary_circles.append([1 - 0.05, y])
        boundary_radii.append(0.05)
    
    # Combine centers and radii
    centers = np.array(centers_hex + boundary_circles)
    radii = np.array(radii_hex + boundary_radii)
    
    # Local optimization to improve packing
    def objective(radii):
        return -np.sum(radii)
    
    def constraints(radii):
        constraints = []
        for i in range(len(radii)):
            constraints.append((radii[i], 0, np.inf))
        return constraints
    
    # Optimization
    opt_result = opt.minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=[(0, 1)] * len(radii),
        constraints=[
            {'type': 'ineq', 'fun': lambda r, i: r[i] + r[j] - np.sqrt(np.sum((centers[i] - centers[j])**2)), 'args': (i, j)} for i in range(len(radii)) for j in range(i+1, len(radii))
        ]
    )
    
    # Apply optimization
    radii_opt = opt_result.x
    sum_radii = np.sum(radii_opt)
    
    return (centers, radii_opt, sum_radii)
