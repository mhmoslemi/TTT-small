import numpy as np
from scipy.optimize import minimize

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        return np.sum(radii) - 2.636

    def bounds(radii):
        return np.zeros_like(radii)

    def packing_constraints(radii):
        return np.sum(radii) - 2.636

    def cluster_centers(n_clusters):
        clusters = []
        for i in range(n_clusters):
            x = np.random.uniform(0.1, 0.9)
            y = np.random.uniform(0.1, 0.9)
            clusters.append([x, y])
        return np.array(clusters)

    def refine_centers(centers, radii):
        for i in range(len(centers)):
            for j in range(len(centers)):
                if i != j:
                    dist = np.linalg.norm(centers[i] - centers[j])
                    if dist < radii[i] + radii[j] - 1e-6:
                        if dist < 0.1:
                            centers[i] += 0.05 * (centers[j] - centers[i]) / dist
                            centers[j] += 0.05 * (centers[i] - centers[j]) / dist
        return centers

    def run_optimization():
        np.random.seed(42)
        initial_centers = np.random.rand(26, 2)
        initial_radii = np.random.rand(26) * 0.1

        result = minimize(objective, initial_radii, method='SLSQP',
                          bounds=[(0, 0.1) for _ in range(26)],
                          constraints=[{'type': 'ineq', 'fun': lambda x: packing_constraints(x)}],
                          options={'ftol': 1e-9, 'maxiter': 1000})
        if not result.success:
            raise ValueError("Optimization failed")
        radii = result.x
        centers = initial_centers
        return centers, radii

    centers, radii = run_optimization()
    centers = refine_centers(centers, radii)
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
