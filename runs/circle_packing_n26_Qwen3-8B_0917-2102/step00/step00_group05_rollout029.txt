import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    np.random.seed(42)
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraints(radii):
        constraints = []
        for i in range(n):
            constraints.append((centers[i, 0] - radii[i]) >= 0)
            constraints.append((centers[i, 0] + radii[i]) <= 1)
            constraints.append((centers[i, 1] - radii[i]) >= 0)
            constraints.append((centers[i, 1] + radii[i]) <= 1)
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                constraints.append((dist - radii[i] - radii[j]) >= 0)
        return constraints
    
    def bounds():
        return [(0, 1) for _ in range(n)]
    
    result = minimize(objective, radii, bounds=bounds, constraints=constraints)
    radii = result.x
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
