def run_packing():
    n = 26
    # Initial guess: hexagonal grid with adjusted spacing
    spacing = 1.0 / np.sqrt(3)
    centers = []
    for i in range(n):
        row = i // 4
        col = i % 4
        x = col * spacing
        y = row * spacing + (col % 2) * spacing / 2
        if x > 1 or y > 1:
            x = 1 - (1 - x) % 1
            y = 1 - (1 - y) % 1
        centers.append([x, y])
    centers = np.array(centers)
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraints(radii):
        nonlocal centers
        dists = squareform(pdist(centers))
        for i, j in combinations(range(n), 2):

                return False
            if centers[i, 1] - radii[i] < 0 or centers[i, 1] + radii[i] > 1:
                return False
        return True
    
    bounds = [(0, 1) for _ in range(n)]
    result = minimize(objective, np.ones(n) * 0.05, method='SLSQP', bounds=bounds, constraints={"type": "callback", "fun": constraints})
