import numpy as np
from scipy.optimize import minimize

def run_packing():
    np.random.seed(42)
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.1)
    
    def objective(r):
        return -np.sum(r)
    
    def constraints(r):
        for i in range(n):
            if r[i] < 0:
                return False
        for i in range(n):
            x, y = centers[i]
            if (x - r[i] < -1e-12 or x + r[i] > 1 + 1e-12
                    or y - r[i] < -1e-12 or y + r[i] > 1 + 1e-12):
                return False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < r[i] + r[j] - 1e-12:
                    return False
        return True
    
    def constraint_func(r):
        return constraints(r)
    
    result = minimize(
        fun=objective,
        x0=radii,
        method='SLSQP',
        constraints={"type": "ineq", "fun": constraint_func},
        bounds=[(0, 1) for _ in range(n)],
        options={'ftol': 1e-12, 'maxiter': 1000}
    )
    
    if not result.success:
        raise ValueError("Optimization failed")
    
    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
