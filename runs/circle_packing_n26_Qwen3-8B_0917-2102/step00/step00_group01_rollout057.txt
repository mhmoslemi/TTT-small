import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    # Initial guess: Hexagonal packing with 5 rows
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    x = np.linspace(0.1, 0.9, 5)
    y = np.linspace(0.1, 0.9, 5)
    for i in range(5):
        for j in range(5):
            centers[i * 5 + j] = [x[j], y[i]]
    # Initial radii as equal for simplicity
    radii[:] = 0.1

    # Optimization function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Constraints: non-overlapping and within square
    def constraint(radii):
        dist_matrix = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        return np.sum(radii)

    result = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=[(0, 0.5) for _ in range(n)],
        constraints=[
            {'type': 'ineq', 'fun': constraint}
        ]
    )

    centers = np.clip(centers, 0, 1)
    radii = result.x

    # Final validation
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            raise ValueError(f"Circle {i} outside unit square")

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError(f"Circles {i} and {j} overlap")

    return centers, radii, np.sum(radii)
