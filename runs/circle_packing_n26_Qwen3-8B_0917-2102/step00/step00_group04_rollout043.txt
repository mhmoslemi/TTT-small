import numpy as np
from scipy.optimize import minimize

from functools import partial

def run_packing():

    n = 26

    # Initial hexagonal packing with boundary circles
    boundary_circles = []
    for i in range(4):
        if i == 0:
            boundary_circles.append((0, 0.5))
        elif i == 1:
            boundary_circles.append((0.5, 1))
        elif i == 2:
            boundary_circles.append((1, 0.5))
        else:
            boundary_circles.append((0.5, 0))

    # Generate hexagonal grid for interior circles
    interior_circles = []
    for row in range(3):
        for col in range(5):
            x = col / 5
            y = row / 3
            interior_circles.append((x, y))

    centers = np.array(boundary_circles + interior_circles)
    radii = np.full(n, 0.05)

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        for i in range(n):
            x, y = centers[i]
            if (x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12
                or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12):
                return False, "Circle outside unit square"
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, "Circle overlap"
        return True, "valid"

    def optimize():
        bounds = [(0, 0.1) for _ in range(n)]
        res = minimize(objective, radii, bounds=bounds, method='SLSQP',
                      constraints=[{'type': 'ineq', 'fun': partial(constraints, radii)}])
        return res.x

    radii_opt = optimize()

    # Final refinement with local adjustments
    for _ in range(10):

            max_radius = 0

                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx*dx + dy*dy)
                    max_radius = max(max_radius, (dist - radii[j]) * (1 - 1e-3))
            radii_opt[i] = min(radii_opt[i] + 1e-4, max_radius)

            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < radii_opt[i] + radii_opt[j] - 1e-12:
                raise ValueError("Circle overlap")

    return centers, radii_opt, np.sum(radii_opt)
