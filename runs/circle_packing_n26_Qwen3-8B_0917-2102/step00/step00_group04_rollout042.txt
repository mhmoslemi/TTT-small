from scipy.optimize import minimize

def run_packing():

    n = 26
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n) * 0.1

    def objective(r):
        return -np.sum(r)

    def constraint1(r):
        return np.sum(r) - 2.636

    def constraint2(r):
        return np.sum(r) - np.sum(r)

    cons = [{'type': 'ineq', 'fun': constraint1}]
    bounds = [(0, 1) for _ in range(n)]

    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)

    radii = result.x
    radii = np.clip(radii, 0, 1)

        return np.linalg.norm(c1 - c2)

        for j in range(i + 1, n):
            if compute_distances(centers[i], centers[j]) < radii[i] + radii[j] - 1e-8:
                radii[i] *= 0.9
                radii[j] *= 0.9

    for i in range(n):
        dx = centers[i, 0]
        dy = centers[i, 1]

        if dx - r < -1e-8 or dx + r > 1 + 1e-8 or dy - r < -1e-8 or dy + r > 1 + 1e8:
            radii[i] *= 0.5

    return centers, radii, np.sum(radii)
