from scipy.spatial.distance import cdist

    # Convert to numpy arrays
    centers = np.array(centers)

    # Optimization to maximize sum of radii with non-overlapping and boundary constraints
    def objective(radii):
        return -np.sum(radii)
    
    def constraint1(radii):
        for i in range(n):
            if radii[i] < 0:
                return -1
        return 0
    
    def constraint2(radii):
        for i in range(n):
            x, y = centers[i]
            if (x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12

    cons = [
        {'type': 'ineq', 'fun': constraint1},
        {'type': 'ineq', 'fun': constraint2},
        {'type': 'ineq', 'fun': constraint3}
    ]
    
    result = minimize(objective, radii, constraints=cons, method='SLSQP')
    if not result.success:
        raise ValueError("Optimization failed")
    
    radii = result.x
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
