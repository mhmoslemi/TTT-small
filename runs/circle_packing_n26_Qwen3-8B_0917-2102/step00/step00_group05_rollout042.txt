import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    n = 26
    initial_centers = np.random.rand(n, 2)
    initial_radii = np.random.rand(n) * 0.1

    def objective(radii):
        return -np.sum(radii)

    def constraints_func(radii):
        constraints = []
        for i in range(n):
            if radii[i] < 0:
                constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i]})
        for i in range(n):
            x, y = initial_centers[i]
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] + 1e-8 - x})
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i: x + 1e-8 - r[i]})
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] + 1e-8 - y})
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i: y + 1e-8 - r[i]})
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((initial_centers[i] - initial_centers[j])**2))
                constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: r[i] + r[j] - dist - 1e-8})
        return constraints

    bounds = Bounds(0, 100)

    result = minimize(objective, initial_radii, method='SLSQP', bounds=bounds, constraints=constraints_func)

    radii = result.x
    centers = np.copy(initial_centers)

    return centers, radii, np.sum(radii)
