import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial guess for centers in a hexagonal packing
    def initial_hex_packing(n):
        centers = []
        for i in range(n):
            row = i // 5
            col = i % 5
            x = 0.2 + col * 0.21
            y = 0.2 + row * (0.21 * np.sqrt(3))
            centers.append([x, y])
        return np.array(centers)

    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraints: circle within unit square and non-overlapping
    def constraints(radii, centers):
        cons = []
        for i in range(len(radii)):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] + centers[i, 0] - 1e-12})
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: 1e-12 + 1 - r[i] - centers[i, 0]})
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] + centers[i, 1] - 1e-12})
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: 1e-12 + 1 - r[i] - centers[i, 1]})
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: np.sqrt(np.sum((centers[i] - centers[j])**2)) - r[i] - r[j]})
        return cons

    # Initial centers in hexagonal packing
    centers = initial_hex_packing(26)
    n = centers.shape[0]
    radii = np.ones(n) * 0.1

    # Optimization
    res = minimize(objective, radii, constraints=constraints(radii, centers), method='SLSQP', bounds=[(0, 1)] * n)

    # Return centers, radii, and sum
    return centers, res.x, np.sum(res.x)
