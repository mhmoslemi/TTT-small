import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    n = 26
    # Initial hexagonal packing in unit square
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        row = i // 5
        col = i % 5
        centers[i, 0] = col + 0.5 * (1 - (1 - 0.5 * np.sqrt(3)) * (row % 2))
        centers[i, 1] = row * (1 + 0.5 * np.sqrt(3)) / 3
        radii[i] = 0.05

    # Optimization parameters
    bounds = Bounds([0, 0], [1, 1])
    constraints = []

    def objective(radii):
        return -np.sum(radii)

    def constraint_func(radii):
        dist_matrix = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return 1
        return 0

    cons = {'type': 'ineq', 'fun': constraint_func}

    result = minimize(objective, radii, bounds=bounds, constraints=cons, method='SLSQP')

    # Adjust radii to ensure boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            r = min(r, 1 - np.min([x, 1 - x]), np.min([y, 1 - y]))
            result.x[i] = r

    return centers, result.x, np.sum(result.x)
