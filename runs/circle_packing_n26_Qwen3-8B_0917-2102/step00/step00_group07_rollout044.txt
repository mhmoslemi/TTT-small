import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial hexagonal packing with 5 rows
    centers = []
    radii = []
    for row in range(5):
        for col in range(13):
            x = col + 0.5 * row / 2
            y = row / 2
            if x > 1 - 1e-10 or y > 1 - 1e-10:
                continue
            centers.append([x, y])
            radii.append(0.05)
    centers = np.array(centers)
    radii = np.array(radii)
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraints(radii):
        n = len(radii)
        cons = []
        for i in range(n):
            cons.append((radii[i], 0, np.inf))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                cons.append((radii[i] + radii[j], dist, np.inf))
        return cons
    
    bounds = [(0, None) for _ in range(len(radii))]
    res = minimize(objective, radii, bounds=bounds, constraints=constraints)
    
    return (centers, res.x, np.sum(res.x))
