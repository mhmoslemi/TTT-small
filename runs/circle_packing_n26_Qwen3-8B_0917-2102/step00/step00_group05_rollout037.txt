import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        return np.sum(radii) - 2.636

    def packing_func(centers, radii):
        n = centers.shape[0]
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, "Circle out of bounds"
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, "Circles overlap"
        return True, "Valid packing"

    n = 26
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n) * 0.1

    bounds = [(0, 1) for _ in range(2 * n)]
    for i in range(n):
        bounds[2 * i] = (0, 1)
        bounds[2 * i + 1] = (0, 1)

    def f(x):
        centers = x[:2 * n].reshape(n, 2)
        radii = x[2 * n:]
        valid, _ = packing_func(centers, radii)
        if not valid:
            return np.inf
        return -np.sum(radii)

    res = minimize(f, x0=np.concatenate([centers.flatten(), radii]), method='L-BFGS-B', bounds=bounds, options={'maxiter': 1000})
    centers = res.x[:2 * n].reshape(n, 2)
    radii = res.x[2 * n:]
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
