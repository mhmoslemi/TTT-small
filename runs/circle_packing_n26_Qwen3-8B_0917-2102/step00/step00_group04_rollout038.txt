import numpy as np
import scipy.optimize as opt

def run_packing():
    # Hexagonal grid parameters
    num_rows = 5
    num_cols = 5
    rows = np.linspace(0, 1, num_rows)
    cols = np.linspace(0, 1, num_cols)
    centers = np.array([[x, y] for y in rows for x in cols])
    radii = np.full(26, 0.05)
    
    # First pass: optimize radii with constraints
    def obj_func(r):
        return -np.sum(r)
    
    def constraint1(r):
        return np.sum(r) - 2.636
    
    def constraint2(r):
        return np.sum(r) - np.sum(r)
    
    cons = [{'type': 'ineq', 'fun': constraint1}]
    bounds = [(0, 1) for _ in range(26)]
    
    result = opt.minimize(obj_func, radii, method='SLSQP', bounds=bounds, constraints=cons)
    
    # Ensure non-overlapping and boundary constraints
    for i in range(26):
        for j in range(i + 1, 26):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < result['x'][i] + result['x'][j] - 1e-8:
                result['x'][i] = dist - result['x'][j] + 1e-8
    
    for i in range(26):
        x, y = centers[i]
        if x - result['x'][i] < 0 or x + result['x'][i] > 1:
            result['x'][i] = np.min([x, 1 - x]) * 0.9
        if y - result['x'][i] < 0 or y + result['x'][i] > 1:
            result['x'][i] = np.min([y, 1 - y]) * 0.9
    
    return centers, result['x'], np.sum(result['x'])
