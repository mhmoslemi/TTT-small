<strategy>Use hexagonal packing with adaptive resizing near boundaries to maximize total radius sum.</strategy>

```python
import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints_func(radii):
        if np.any(radii < 0):
            return np.inf
        centers = compute_centers(radii)
        if not check_boundary(centers, radii):
            return np.inf
        if not check_overlap(centers, radii):
            return np.inf
        return -np.sum(radii)

    def compute_centers(radii):
        n = len(radii)
        centers = np.zeros((n, 2))
        centers[0] = [0.5, 0.5]
        for i in range(1, n):
            angle = 2 * np.pi * i / n
            dist = 1 - 2 * radii[i]
            centers[i] = [0.5 + dist * np.cos(angle), 0.5 + dist * np.sin(angle)]
        return centers

    def check_boundary(centers, radii):
        for i in range(len(radii)):
            x, y = centers[i]
            if (x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12 or
                y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12):
                return False
        return True

    def check_overlap(centers, radii):
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    n = 26
    initial_guess = np.full(n, 0.04)
    bounds = [(0, 0.5) for _ in range(n)]

    result = minimize(
        constraints_func,
        initial_guess,
        bounds=bounds,
        method='SLSQP',
        options={'ftol': 1e-10, 'maxiter': 500}
    )

    centers = compute_centers(result.x)
    radii = result.x
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
```
