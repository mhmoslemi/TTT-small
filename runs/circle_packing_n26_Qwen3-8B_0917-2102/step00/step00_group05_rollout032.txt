import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import matplotlib.pyplot as plt

def run_packing():
    # Hexagonal packing: 5 rows with 5, 5, 5, 5, 6 circles
    # Adjusted for unit square boundary constraints
    N = 26
    centers = []
    radii = []

    # First row (top)
    row1 = np.linspace(0.5, 0.5, 5)
    for x in row1:
        centers.append([x, 1 - 0.1])
        radii.append(0.1)
    
    # Second row
    row2 = np.linspace(0.25, 0.75, 5)
    for x in row2:
        centers.append([x, 0.8])
        radii.append(0.1)
    
    # Third row
    row3 = np.linspace(0.125, 0.875, 5)
    for x in row3:
        centers.append([x, 0.6])
        radii.append(0.1)
    
    # Fourth row
    row4 = np.linspace(0.0625, 0.9375, 5)
    for x in row4:
        centers.append([x, 0.4])
        radii.append(0.1)
    
    # Fifth row
    row5 = np.linspace(0.03125, 0.96875, 6)
    for x in row5:
        centers.append([x, 0.2])
        radii.append(0.1)
    
    # Sixth row (bottom)
    row6 = np.linspace(0.015625, 0.984375, 6)
    for x in row6:
        centers.append([x, 0.05])
        radii.append(0.05)

    centers = np.array(centers)
    radii = np.array(radii)
    
    # Optimization to maximize sum of radii
    def objective(r):
        return -np.sum(r)
    
    def constraints(r):
        dist = squareform(pdist(centers))
        for i in range(N):
            for j in range(i + 1, N):
                if dist[i, j] < r[i] + r[j] + 1e-8:
                    return False
        for i in range(N):
            if centers[i, 0] - r[i] < 0 or centers[i, 0] + r[i] > 1:
                return False
            if centers[i, 1] - r[i] < 0 or centers[i, 1] + r[i] > 1:
                return False
        return True
    
    bounds = [(0, 1) for _ in range(N)]
    result = minimize(objective, radii, bounds=bounds, constraints={"type": "constraint", "fun": constraints})
    
    radii = result.x
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
