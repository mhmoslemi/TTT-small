from scipy.optimize import minimize

def run_packing():

    # 5 rows: 5, 5, 5, 6, 5 circles
    # Hexagonal packing with horizontal spacing and vertical spacing
    # Target sum of radii: 2.636

        return -np.sum(radii)

    def constraints(radii):
        # Boundary constraints
        for i in range(26):
            if radii[i] < 0:
                return False
        return True

    centers = []
    radii = []
    # Initial placement based on hexagonal packing
    for row in range(5):
        for col in range(5 + (row == 3)):
            x = col + 0.5 * (row % 2)
            y = row * np.sqrt(3) / 2
            centers.append([x, y])
            radii.append(0.1)

    centers = np.array(centers)
    radii = np.array(radii)

    # Optimization
    bounds = [(0, 1) for _ in range(26)]
    result = minimize(

        bounds=bounds,

            }
        ]
    )

    if not result.success:
        raise ValueError("Optimization failed")

    return centers, result.x, np.sum(result.x)
