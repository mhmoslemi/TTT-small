import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        constraints = []
        for i in range(26):
            constraints.append((radii[i], 0, np.inf))
        return constraints

    def bounds(radii):
        return [(0, np.inf) for _ in range(26)]

    def calc_centers(radii):
        num_layers = 5
        layer_radii = [radii[i] for i in range(num_layers * 4 + 1)]

        centers = []
        for i in range(num_layers):
            layer = []
            for j in range(i + 1):
                x = 0.5 - (i + j) * (radii[0] + radii[1])
                y = 0.5 - j * (radii[0] + radii[1])
                layer.append([x, y])
            for j in range(i, -1, -1):
                x = 0.5 + j * (radii[0] + radii[1])
                y = 0.5 - (i + j) * (radii[0] + radii[1])
                layer.append([x, y])
            centers.append(layer)
        return np.array(centers)

    def calc_radii(centers):
        n = len(centers)
        radii = np.zeros(n)
        for i in range(n):
            dists = np.linalg.norm(centers[i] - centers, axis=1)
            radii[i] = np.min(dists) / 2
        return radii

    centers = calc_centers(np.ones(26))
    radii = calc_radii(centers)

    result = minimize(
        objective,
        radii,
        bounds=bounds(radii),
        constraints=constraints(radii),
        method="SLSQP"
    )

    if not result.success:
        raise RuntimeError("Optimization failed")

    return (centers, result.x, -result.fun)
