import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize

def run_packing():
    n = 26

    # Define positions for hexagonal packing
    rows = 6
    cols = 5
    spacing = 1.0 / (rows + 1)
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    for i in range(rows):
        for j in range(cols):
            x = j * spacing + spacing
            y = i * spacing + spacing
            if i % 2 == 1:
                x += spacing / 2
            centers[i * cols + j] = [x, y]

    # Initial radii based on minimum distance between circles
    min_dist = np.min(np.sqrt(np.sum((centers[i] - centers[j])**2) for i in range(n) for j in range(i+1, n)))
    radii = np.full(n, min_dist / 2)

    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraints: circles must not overlap, stay within unit square
    def constraint1(radii):
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    def constraint2(radii):
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8 or y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8:
                return -1
        return 0

    # Optimization
    bounds = [(0, 1) for _ in range(n)]
    cons = [
        {'type': 'ineq', 'fun': constraint1},
        {'type': 'ineq', 'fun': constraint2}
    ]
    result = minimize(objective, radii, bounds=bounds, constraints=cons, method='SLSQP')

    return centers, result.x, np.sum(result.x)
