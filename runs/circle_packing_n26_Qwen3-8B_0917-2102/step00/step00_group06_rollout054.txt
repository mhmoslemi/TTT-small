import numpy as np
from scipy.optimize import minimize

def run_packing():

        return -np.sum(radii)  # Maximize sum by minimizing negative sum

    def constraints(radii):
        constraints = []
        for i in range(26):
            constraints.append((radii[i], 0, np.inf))  # Non-negative radii
        return constraints

    # Initial guess for radii: uniform distribution
    radii_initial = np.full(26, 0.1)

    # Optimization
    result = minimize(obj_func, radii_initial, method='SLSQP', bounds=[(0, np.inf) for _ in range(26)], constraints=constraints)

    # Construct centers using hexagonal packing
    centers = []
    for i in range(26):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (col % 2))
        y = row + 0.5 * (1 - 2 * (row % 2))
        centers.append([x, y])
    centers = np.array(centers)

    # Adjust radii to minimize overlap
    radii = np.zeros(26)
    for i in range(26):
        for j in range(i + 1, 26):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            radii[i] = min(radii[i], dist / 2)
            radii[j] = min(radii[j], dist / 2)

    # Final adjustment based on optimization result
    radii = np.clip(result.x, 0, np.inf)

    # Check and adjust to fit in unit square
    for i in range(26):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            radii[i] = min(r, 1 - np.min([x, 1 - x, y, 1 - y]))

    sum_radii = np.sum(radii)
    return np.array(centers), radii, sum_radii
