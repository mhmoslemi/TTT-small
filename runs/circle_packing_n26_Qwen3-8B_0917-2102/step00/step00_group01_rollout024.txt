from scipy.optimize import minimize
from scipy.spatial import distance

            col = j
            x = col + offset * (i % 2)

            centers[i * (5 + i % 2) + j] = [x, y]

    # Optimization variables: radii

        return -np.sum(radii)

        for i in range(26):
            x, y = centers[i]
            if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
                return False

                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:

    # Optimization
    result = minimize(

        x0=radii,
        method='SLSQP',
        bounds=[(0, 1) for _ in range(26)],
