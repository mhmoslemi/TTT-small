<strategy>Use a hexagonal packing with optimized circle sizes near boundaries to maximize total radii.</strategy>

import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial guess for centers using hexagonal packing
    n = 26
    centers = []
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 0.5 * (row % 2))
        y = row * (np.sqrt(3)/2)
        centers.append([x, y])
    centers = np.array(centers)

    # Initial radii based on minimum distance between circles
    def obj_func(radii, centers):
        return -np.sum(radii)

    def constraints(radii, centers):
        n = len(radii)
        cons = []
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - 1e-8})
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: (r[i] + r[j]) - dist + 1e-8})
        return cons

    initial_radii = np.ones(n) * 0.1
    bounds = [(0, 1) for _ in range(n)]

    result = optimize.minimize(
        fun=obj_func,
        x0=initial_radii,
        args=(centers,),
        method='SLSQP',
        bounds=bounds,
        constraints=constraints(initial_radii, centers)
    )

    radii = result.x
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
