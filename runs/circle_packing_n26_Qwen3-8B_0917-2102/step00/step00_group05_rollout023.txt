import numpy as np
import scipy.optimize as opt

def run_packing():
    # Define hexagonal packing coordinates
    def hex_coords(n, rows):
        coords = []
        for r in range(rows):
            for c in range(n):
                x = c + r / 2
                y = r * np.sqrt(3) / 2
                coords.append([x, y])
        return np.array(coords)
    
    # Initial hexagonal packing with 5 rows
    rows = 5
    n_per_row = 5
    initial_centers = hex_coords(n_per_row, rows)
    
    # Clip coordinates to [0, 1]
    initial_centers[:, 0] = np.clip(initial_centers[:, 0], 0, 1)
    initial_centers[:, 1] = np.clip(initial_centers[:, 1], 0, 1)
    
    # Simplified radii initialization based on spacing
    radii = np.full(len(initial_centers), 0.05)
    
    # Optimization function to maximize sum of radii with constraints
    def obj(r):
        return -np.sum(r)
    
    def constraint_func(r):
        constraints = []
        for i in range(len(r)):
            constraints.append((r[i], 0, np.inf))
        for i in range(len(r)):
            for j in range(i + 1, len(r)):
                dist = np.sqrt(np.sum((initial_centers[i] - initial_centers[j])**2))
                constraints.append((dist - r[i] - r[j], 0, np.inf))
        for i in range(len(r)):
            dist_to_boundary = np.min([initial_centers[i, 0] - 0, 1 - initial_centers[i, 0],
                                       initial_centers[i, 1] - 0, 1 - initial_centers[i, 1]])
            constraints.append((dist_to_boundary - r[i], 0, np.inf))
        return constraints
    
    # Optimization setup
    bounds = [(0, np.inf) for _ in range(len(radii))]
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    # Run optimization
    result = opt.minimize(obj, radii, method='SLSQP', bounds=bounds, constraints=cons)
    
    centers = initial_centers
    radii = result.x
    sum_radii = np.sum(radii)
    
    return (centers, radii, sum_radii)
