import numpy as np
import scipy.optimize as opt

def run_packing():
    # Hexagonal packing with 2 rows of 13 circles
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Row 1: centered horizontally
    row1_y = 0.5
    row1_x = np.linspace(0.5 - 0.5, 0.5 + 0.5, 13)
    centers[:13, 0] = row1_x
    centers[:13, 1] = row1_y
    
    # Row 2: offset and lower
    row2_y = 0.5 - np.sqrt(3)/2 * 0.5
    row2_x = np.linspace(0.5 - 0.5, 0.5 + 0.5, 13)
    centers[13:, 0] = row2_x
    centers[13:, 1] = row2_y
    
    # Initial radius guess
    r0 = 0.1
    
    # Optimization with constraints
    def objective(r):
        return -np.sum(r)  # Maximize sum of radii
    
    def constraint1(r):
        return r - 0.1  # r >= 0.1
    
    def constraint2(r):
        # Check distances between centers
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - 1e-8:
                    return -np.inf  # Violation
        return 0
    
    cons = [{'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2}]
    
    bounds = [(0.1, 1.0) for _ in range(n)]
    
    result = opt.minimize(objective, r0 * np.ones(n), bounds=bounds, constraints=cons)
    
    radii = result.x
    sum_radii = np.sum(radii)
    
    # Final validation
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < radii[i] + radii[j] - 1e-8:
                raise ValueError("Overlap between circles")
    
    return centers, radii, sum_radii
