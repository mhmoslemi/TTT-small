import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    # Initial guess for centers: hexagonal packing with 5 rows
    centers = np.zeros((n, 2))
    r = 0.1  # Initial radius for all circles
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - (row % 2))
        y = row + 0.5 * (1 - (col % 2))
        centers[i] = [x, y]
    
    # Define objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum

    # Define constraints: non-overlapping and within unit square
    def constraints(radii):
        cons = []
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - 1e-12})
        for i in range(n):
            for j in range(i + 1, n):
                def dist_constraint(r):
                    dx = centers[i][0] - centers[j][0]
                    dy = centers[i][1] - centers[j][1]
                    dist = np.sqrt(dx**2 + dy**2)
                    return dist - (r[i] + r[j]) + 1e-12
                cons.append({'type': 'ineq', 'fun': dist_constraint})
        return cons

    # Initial guess for radii
    radii = np.full(n, r)

    # Optimization
    result = minimize(objective, radii, method='SLSQP', constraints=constraints(radii))
    radii = result.x

    # Final validation
    centers = np.array(centers)
    radii = np.array(radii)
    return centers, radii, np.sum(radii)
