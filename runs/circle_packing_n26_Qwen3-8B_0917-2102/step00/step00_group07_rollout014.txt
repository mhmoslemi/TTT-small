import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    # Hexagonal packing with 26 circles
    # Define grid of centers
    rows = 6
    cols = 5
    centers = []
    radii = []

    # Create hexagonal grid with varying spacing
    for i in range(rows):
        for j in range(cols):
            x = j / (cols - 1) if cols > 1 else 0.5
            y = i / (rows - 1) if rows > 1 else 0.5
            x_offset = (j - (cols - 1)/2) / (cols - 1)
            y_offset = (i - (rows - 1)/2) / (rows - 1)
            x = 0.5 + x_offset
            y = 0.5 + y_offset
            centers.append([x, y])

    # Add corner circles
    centers.append([0.1, 0.1])
    centers.append([0.9, 0.1])
    centers.append([0.1, 0.9])
    centers.append([0.9, 0.9])
    
    # Initial radii as equal for all circles
    r = 0.1
    radii = [r] * len(centers)

    # Define objective function to maximize sum of radii
    def objective(radii):
        return -sum(radii)  # Minimize negative sum to maximize sum

    # Define constraint for circle boundaries
    def boundary_constraint(radii):
        for i, (x, y) in enumerate(centers):
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12 or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return -1
        return 0

    # Define constraint for circle-circle non-overlapping
    def non_overlapping_constraint(radii):
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0

    # Combine all constraints
    def constraints(radii):
        bc = boundary_constraint(radii)
        nc = non_overlapping_constraint(radii)
        return bc + nc

    # Optimize with bounds and constraints
    bounds = Bounds([0.0] * len(radii), [1.0] * len(radii))
    result = minimize(objective, radii, bounds=bounds, constraints=constraints)

    # Extract final centers and radii
    centers = np.array(centers)
    radii = result.x
    sum_radii = sum(radii)
    return centers, radii, sum_radii
