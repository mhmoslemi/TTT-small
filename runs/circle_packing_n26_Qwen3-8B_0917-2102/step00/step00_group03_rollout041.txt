from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import matplotlib.pyplot as plt

def run_packing():

    radii = np.zeros(num_circles)

    for i in range(5):
        centers[i, 0] = i / 4
        centers[i, 1] = 0.1
        radii[i] = 0.05

    # Second row: 5 circles
    for i in range(5, 10):
        centers[i, 0] = (i - 5) / 4
        centers[i, 1] = 0.25
        radii[i] = 0.05

    # Third row: 5 circles
    for i in range(10, 15):
        centers[i, 0] = (i - 10) / 4
        centers[i, 1] = 0.4
        radii[i] = 0.05

    # Extra circles in the bottom row
    for i in range(15, 26):
        centers[i, 0] = (i - 15) / 4
        centers[i, 1] = 0.55
        radii[i] = 0.05

    # Initial sum of radii
    sum_radii = np.sum(radii)

    # Optimization parameters
    bounds = [((0, 1), (0, 1)) for _ in range(num_circles * 2)]
    constraints = []

    # Constraint for non-overlapping circles
    def constraint_nonoverlap(x):
        centers_flat = x[:num_circles * 2].reshape((num_circles, 2))
        radii_flat = x[num_circles * 2:]
        dist_matrix = squareform(pdist(centers_flat))
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                if dist_matrix[i, j] < radii_flat[i] + radii_flat[j] - 1e-8:
                    return dist_matrix[i, j] - (radii_flat[i] + radii_flat[j] - 1e-8) + 1
        return 0

    # Constraint for staying within the unit square
    def constraint_boundary(x):
        centers_flat = x[:num_circles * 2].reshape((num_circles, 2))
        radii_flat = x[num_circles * 2:]
        for i in range(num_circles):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < -1e-8 or x_i + r_i > 1 + 1e-8:
                return x_i - r_i + 1
            if y_i - r_i < -1e-8 or y_i + r_i > 1 + 1e-8:
                return y_i - r_i + 1
        return 0

    constraints.append({'type': 'ineq', 'fun': constraint_nonoverlap})
    constraints.append({'type': 'ineq', 'fun': constraint_boundary})

    # Optimization
    x0 = np.concatenate([centers.flatten(), radii])
    result = minimize(lambda x: -np.sum(x[num_circles * 2:]), x0, bounds=bounds, constraints=constraints, method='SLSQP')

    centers = result.x[:num_circles * 2].reshape((num_circles, 2))
    radii = result.x[num_circles * 2:]
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
