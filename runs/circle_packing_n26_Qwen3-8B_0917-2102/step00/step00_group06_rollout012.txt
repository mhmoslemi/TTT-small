from scipy.optimize import minimize

    n = 26

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        return np.sum(radii)

    def bounds(radii):
        return [(0, 1) for _ in range(n)]

        centers = []

            x = np.random.uniform(0, 1)

        fun=objective_positions,

        bounds=bounds_positions(),
