import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():

    # Initialize positions with hexagonal packing
    rows = 5
    cols = 6
    offsets = np.linspace(0, 1, cols + 1)
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    for i in range(rows):
        for j in range(cols):
            x = offsets[j] if i % 2 == 0 else offsets[j + 1]
            y = i * np.sqrt(3) / 2 / (rows - 1) * 0.9
            centers[i * cols + j] = [x, y]
    # Initial radius based on spacing
    initial_radius = 0.1
    radii[:] = initial_radius
    # Objective function to maximize sum of radii
    def objective(x):
        radii[:] = x[:num_circles]
        return -np.sum(x[:num_circles])  # Minimize negative sum

    # Constraints: circles must be within [0,1]^2 and non-overlapping
    def constraint(x):
        radii[:] = x[:num_circles]
        for i in range(num_circles):
            x_i, y_i = centers[i]
            if x_i - radii[i] < 0 or x_i + radii[i] > 1 or y_i - radii[i] < 0 or y_i + radii[i] > 1:
                return -1
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    # Optimization
    bounds = [(0, 1) for _ in range(num_circles)]
    result = minimize(objective, np.full(num_circles, initial_radius), bounds=bounds, constraints={"type": "ineq", "fun": constraint})
