import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    # Initial guess: hexagonal packing with 5 rows of 5, 5, 5, 5, 6 circles
    centers = []
    radii = []
    for i in range(5):
        for j in range(5 if i % 2 == 0 else 6):
            x = j + 0.5 if i % 2 == 1 else j
            y = i * (np.sqrt(3)/2)
            centers.append([x, y])
    centers = np.array(centers)
    
    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum
    
    # Constraint: circles must be inside the square
    def constraint_inside(radii):
        return np.sum((centers[:, 0] - radii[:, np.newaxis]) ** 2 + (centers[:, 1] - radii[:, np.newaxis]) ** 2 <= 1)
    
    # Constraint: circles must not overlap
    def constraint_no_overlap(radii):
        dists = squareform(pdist(centers))
        return np.sum(dists <= radii[:, np.newaxis] + radii[np.newaxis, :])
    
    # Optimization
    result = minimize(
        objective,
        x0=np.ones(26),
        bounds=[(0, 1) for _ in range(26)],
        constraints=[{'type': 'ineq', 'fun': constraint_inside},
                    {'type': 'ineq', 'fun': constraint_no_overlap}]
    )
    
    return (centers, result.x, np.sum(result.x))
