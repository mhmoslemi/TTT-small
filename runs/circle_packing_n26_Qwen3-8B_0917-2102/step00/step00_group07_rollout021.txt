import numpy as np
from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d
from itertools import combinations

def run_packing():
    np.random.seed(42)
    n = 26

    # Initial hexagonal grid with spacing
    spacing = 0.3
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col * spacing + spacing / 2
        y = row * spacing * np.sqrt(3) / 2
        centers[i] = [x, y]

    # Optimize radii and positions with constraints
    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii

    def constraints(radii):
        constraints = []
        for i in range(n):
            constraints.append((radii[i] >= 0))
        for i, j in combinations(range(n), 2):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            constraints.append((radii[i] + radii[j] <= dist))
        for i in range(n):
            x, y = centers[i]
            constraints.append((radii[i] <= 1 - max(x, 1 - x, y, 1 - y)))
        return constraints

    bounds = [(0, 1) for _ in range(n)]
    result = minimize(objective, x0=np.ones(n) * 0.1, bounds=bounds, constraints=constraints)
    radii = result.x

    # Final validation
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = max(0, 1 - max(x, 1 - x, y, 1 - y))
    return centers, radii, np.sum(radii)
