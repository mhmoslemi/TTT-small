from scipy.optimize import minimize

    # Row 3: 4 circles
    # Row 4: 5 circles
    # Row 5: 4 circles
    # Row 6: 3 circles
    # Row 7: 2 circles
    # Row 8: 1 circle

    def objective(radii):

    def constraints(radii):
        # Boundary constraints
        for i in range(len(centers)):
            x, y = centers[i]
            if radii[i] > 1 - max(x, 1 - x, y, 1 - y):
                return False

                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)

    # Use a numerical optimization with constraints
    result = minimize(
        fun=objective,
        x0=initial_radii,
        bounds=[(0, 1) for _ in range(81)],

    return (centers_opt, radii_opt, np.sum(radii_opt))
