import numpy as np
from scipy.optimize import minimize

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        return np.sum(radii)

    def bounds(radii):
        return [(0, 1) for _ in radii]

    def pack_circles():
        centers = np.random.rand(26, 2)
        radii = np.random.rand(26)
        return centers, radii

    centers, radii = pack_circles()
    result = minimize(objective, radii, method='SLSQP', bounds=bounds(radii), constraints=[{'type': 'eq', 'fun': constraints}])
    return result.x, np.random.rand(26), np.sum(result.x)
