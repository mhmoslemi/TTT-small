import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize

def run_packing():
    # Hexagonal packing with 26 circles
    # First row: 5 circles
    # Second row: 5 circles
    # Third row: 5 circles
    # Fourth row: 5 circles
    # Fifth row: 6 circles (as last row)
    
    # Initial approximate radius estimate
    approx_radius = 0.1

    # Generate hexagonal grid points
    centers = []
    radii = []
    for row in range(5):
        for col in range(5 + (row == 4)):
            x = col + 0.5 * (row % 2)
            y = row * np.sqrt(3) / 2
            centers.append([x, y])
            radii.append(approx_radius)
    
    centers = np.array(centers)
    radii = np.array(radii)
    
    # Objective function to maximize total radius
    def obj(x):
        return -sum(x)
    
    # Constraints: circle within unit square
    def constraint1(x):
        return np.array([x[i] <= 1 - radii[i] for i in range(len(radii))])
    
    def constraint2(x):
        return np.array([x[i] >= radii[i] for i in range(len(radii))])
    
    constraints = [{'type': 'ineq', 'fun': constraint1},
                   {'type': 'ineq', 'fun': constraint2}]
    
    # Initial guess: radii = approx_radius * np.ones(len(radii))
    initial_guess = approx_radius * np.ones(len(radii))
    
    # Optimize radii
    result = minimize(obj, initial_guess, method='SLSQP', bounds=[(0, 1) for _ in range(len(radii))], constraints=constraints)
    
    optimized_radii = result.x
    sum_radii = sum(optimized_radii)
    
    return centers, optimized_radii, sum_radii
