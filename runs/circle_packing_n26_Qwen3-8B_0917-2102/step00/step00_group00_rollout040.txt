import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import matplotlib.pyplot as plt

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.ones(n) * 0.1

    def objective(r):
        return -np.sum(r)
    
    def constraints(r):
        constraints = []
        for i in range(n):
            constraints.append((centers[i, 0] - r[i] >= 0) & (centers[i, 0] + r[i] <= 1) &
                               (centers[i, 1] - r[i] >= 0) & (centers[i, 1] + r[i] <= 1))
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                constraints.append(dist >= r[i] + r[j] - 1e-8)
        return np.array([constraint for constraint in constraints])
    
    result = minimize(
        fun=objective,
        x0=radii,
        method='SLSQP',
        constraints={'type': 'ineq', 'fun': constraints},
        bounds=[(0, 1) for _ in range(n)]
    )
    
    if not result.success:
        raise ValueError("Optimization failed")
    
    return centers, result.x, np.sum(result.x)
