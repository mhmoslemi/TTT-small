import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    # Initial guess for centers using hexagonal packing
    n = 26
    centers = []
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (i % 5 == 0)
        y = row + 0.5 * (i % 5 == 0)
        centers.append([x, y])
    centers = np.array(centers)
    
    # Initial guess for radii
    radii = np.full(n, 0.1)
    
    # Define bounds for centers and radii
    bounds_centers = [(0, 1) for _ in range(n * 2)]
    bounds_radii = [(0, 1) for _ in range(n)]
    
    # Objective function to maximize sum of radii
    def objective(x):
        flat_x = x.reshape((n, 2))
        flat_r = x[n*2:]
        return -np.sum(flat_r)
    
    # Constraints: circles must not overlap and must be inside [0,1]^2
    def constraint(x):
        flat_x = x.reshape((n, 2))
        flat_r = x[n*2:]
        for i in range(n):
            x_i, y_i = flat_x[i]
            r_i = flat_r[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return -np.inf

                dy = flat_x[i, 1] - flat_x[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < flat_r[i] + flat_r[j] - 1e-12:

    bounds = Bounds(bounds_centers + bounds_radii)
