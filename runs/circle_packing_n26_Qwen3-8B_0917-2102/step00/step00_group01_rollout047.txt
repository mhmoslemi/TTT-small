<strategy>Use a hexagonal packing with adjusted circles near boundaries and corners to maximize total radius sum.</strategy>

```python
import numpy as np
from scipy.optimize import minimize

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        return np.sum(radii) - 2.636

    def bounds():
        return [(0, 1.0) for _ in range(26)]

    def packing_function():
        centers = np.random.rand(26, 2)
        radii = np.random.rand(26) * 0.1
        return centers, radii

    centers, radii = packing_function()

    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraints})

    if not result.success:
        raise RuntimeError("Optimization failed")

    return centers, result.x, np.sum(result.x)
```
