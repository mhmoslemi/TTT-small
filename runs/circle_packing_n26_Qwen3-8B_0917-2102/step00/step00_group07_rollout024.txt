import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial hexagonal packing with 26 circles
    centers = []
    radii = []
    for row in range(4):
        for col in range(6):
            x = col + 0.5 * (row % 2)
            y = row * np.sqrt(3) / 2
            if row == 0:
                x = col + 0.5
                y = 0.25
            elif row == 3:
                x = col + 0.5
                y = 0.75
            centers.append([x, y])
            radii.append(0.1)
    
    centers = np.array(centers)
    radii = np.array(radii)
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraints(radii):
        n = len(radii)
        cons = []
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] * 2 - 1e-12})
        for i in range(n):
            for j in range(i+1, n):
                def dist(r, i=i, j=j):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    return np.sqrt(dx**2 + dy**2) - (r[i] + r[j]) + 1e-12
                cons.append({'type': 'ineq', 'fun': dist})
        return cons
    
    res = minimize(objective, radii, method='SLSQP', constraints=constraints(radii))
    
    if not res.success:
        raise ValueError("Optimization failed")
    
    sum_radii = np.sum(res.x)
    return centers, res.x, sum_radii
