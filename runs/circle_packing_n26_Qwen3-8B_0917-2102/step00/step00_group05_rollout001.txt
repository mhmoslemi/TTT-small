import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance_matrix

def run_packing():
    n = 26
    # Initial hexagonal grid with 5 rows and 6 columns
    centers = []
    radii = []
    for i in range(5):
        for j in range(6):
            x = j + 0.5 * (i % 2)
            y = i * np.sqrt(3) / 2
            centers.append([x / 5, y / 5])
    centers = np.array(centers)
    radii = np.full(n, 0.05)

    # Optimization parameters
    bounds = [(0, 1) for _ in range(2 * n)] + [(0, 0.5) for _ in range(n)]
    res = minimize(lambda x: -np.sum(x[n:]), x=np.concatenate([centers.flatten(), radii]), bounds=bounds,
                   constraints=[{'type': 'ineq', 'fun': lambda x: np.sqrt((centers[i] - centers[j])**2) - (x[n + i] + x[n + j]) for i in range(n) for j in range(i + 1)] + [lambda x: x[n + i] for i in range(n)] + [lambda x: 1 - np.sqrt((centers[i] - [0, 0])**2) - x[n + i] for i in range(n)]])
    centers = res.x[:2 * n].reshape((n, 2))
    radii = res.x[n:]
    sum_radii = np.sum(radii)

    # Validate and adjust for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1):
            radii[i] = min(r, 1 - np.min([x, 1 - x, y, 1 - y]))
    return centers, radii, sum_radii
