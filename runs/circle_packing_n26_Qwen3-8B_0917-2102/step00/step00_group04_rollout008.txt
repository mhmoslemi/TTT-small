import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraint_overlap(i, j):
        def cons(x):
            r = x[0]
            c = x[1:]
            return np.sqrt(np.sum((c[i] - c[j])**2)) - (r[i] + r[j])
        return cons

    def constraint_boundary(i):
        def cons(x):
            r = x[0]
            c = x[1:]
            return np.abs(c[i][0] - r[i]) - 1e-8
        return cons

    def constraint_radius(r):
        return r >= 0

    n = 26
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n)

    bounds = [(0, 1) for _ in range(n)]
    constraints = []

    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': constraint_boundary(i), 'args': (i,)})

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': constraint_overlap(i, j), 'args': (i, j)})

    result = minimize(objective, np.concatenate((radii, centers.flatten())), bounds=bounds, constraints=constraints)

    final_radii = result.x[:n]
    final_centers = result.x[n:].reshape(n, 2)

    return final_centers, final_radii, np.sum(final_radii)
