import numpy as np
from scipy.optimize import minimize

def run_packing():

    centers = np.zeros((26, 2))
    radii = np.zeros(26)

    # Place 25 small circles in a hexagonal grid
    for i in range(25):
        row = i // 5
        col = i % 5
        x = 0.1 + col * 0.2
        y = 0.1 + row * 0.2
        centers[i] = [x, y]
        radii[i] = 0.05

    # Place one large circle near the top right corner
    centers[25] = [0.8, 0.8]
    radii[25] = 0.3

    # Optimization function to maximize sum of radii with constraints
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        # Boundary constraints
        for i in range(26):
            if centers[i][0] - radii[i] < -1e-12 or centers[i][0] + radii[i] > 1 + 1e-12:
                return False
            if centers[i][1] - radii[i] < -1e-12 or centers[i][1] + radii[i] > 1 + 1e-12:
                return False
        # Pairwise distance constraints
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    # Convert to optimization variables
    x0 = radii.copy()

    # Optimization
    result = minimize(
        objective,
        x0,
        constraints={"type": "constraint", "fun": constraints},
        bounds=[(0, 1) for _ in range(26)],
        method="SLSQP"
    )

    # Extract results
    radii = result.x
    return centers, radii, np.sum(radii)
