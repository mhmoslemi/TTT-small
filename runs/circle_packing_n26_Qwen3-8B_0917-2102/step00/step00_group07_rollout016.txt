import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance

def run_packing():
    n = 26
    # Initial guess: hexagonal packing with adjusted radii
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Hexagonal grid pattern with 5 rows
    for i in range(5):
        for j in range(5 + i % 2):
            row = i / 2
            col = j
            x = col / 4.5 + 0.05
            y = row / 3.0 + 0.1
            centers[i * 5 + j] = [x, y]

    # Initial radii based on spacing
    radii = np.full(n, 0.1)

    # Optimization objective: maximize sum of radii
    def obj(r):
        return -np.sum(r)

    # Constraints: circles must not overlap and stay within square
    def cons(r):
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append(dist >= r[i] + r[j] - 1e-12)
        return constraints

    # Optimization with bounds
    bounds = [(0, 1) for _ in range(n)]

    # Initial guess for radii
    initial_guess = radii

    # Run optimization
    result = minimize(obj, initial_guess, bounds=bounds, constraints=cons)

    # Extract centers and radii
    centers = centers
    radii = result.x

    # Ensure circles are within square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            radii[i] = min(r, 1 - np.min([x, 1 - x, y, 1 - y]))

    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
