import numpy as np
import scipy.optimize as opt

def run_packing():
    np.random.seed(42)
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n) * 0.1

    def objective(r):
        return -np.sum(r)

    def constraints(r):
        return np.sum(r) - 0.0

    bounds = [(0, 1) for _ in range(n)]

    for _ in range(10):
        res = opt.minimize(
            objective,
            radii,
            bounds=bounds,
            constraints=[{'type': 'ineq', 'fun': constraints}]
        )
        radii = res.x
        centers = np.random.rand(n, 2)

    radii = np.clip(radii, 0, 1)
    return centers, radii, np.sum(radii)
