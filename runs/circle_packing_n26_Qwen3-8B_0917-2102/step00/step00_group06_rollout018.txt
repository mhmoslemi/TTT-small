from scipy.spatial.distance import pdist, squareform

        return np.sum(radii)
    
    def distance_matrix(centers):
        return squareform(pdist(centers))
    
    def circle_overlap(centers, radii):

        return np.any(dists < radii[:, np.newaxis] + radii[np.newaxis, :] - 1e-8)

        return np.any(centers[:, 0] - radii < -1e-8) or np.any(centers[:, 0] + radii > 1 + 1e-8) or \

                return np.inf

                return np.inf
            return 0
        
        bounds = [(0, 1) for _ in range(n*2)] + [(0, 0.5) for _ in range(n)]
