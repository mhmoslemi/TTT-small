import numpy as np
from scipy.optimize import minimize, Bounds

def run_packing():
    # Initial hexagonal packing with 26 circles
    num_circles = 26
    centers = []
    for row in range(5):
        for col in range(5 + (row % 2)):
            x = col + 0.5 * (row % 2)
            y = row + 0.5
            if row % 2 == 1:
                x -= 0.5
            centers.append([x, y])
    centers = np.array(centers)
    
    # Initial radii based on spacing
    r = np.full(num_circles, 0.1)
    
    # Objective function to maximize sum of radii
    def objective(x):
        return -np.sum(x)  # Negative for minimization

    # Constraints: non-overlapping and boundary
    def constraint(x, idx):
        r = x
        for i in range(num_circles):
            if i == idx:
                continue
            dist = np.sqrt(np.sum((centers[i] - centers[idx])**2))
            if dist < r[i] + r[idx] - 1e-8:
                return -np.inf
        return 0

    # Prepare constraints
    cons = []
    for i in range(num_circles):
        def cons_i(x):
            r = x
            for j in range(num_circles):
                if i == j:
                    continue
                dist = np.sqrt(np.sum((centers[j] - centers[i])**2))
                if dist < r[j] + r[i] - 1e-8:
                    return -np.inf
            return 0
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: cons_i(x)})

    # Optimization
    x0 = np.full(num_circles, 0.1)
    result = minimize(objective, x0, method='SLSQP', constraints=cons, bounds=Bounds(0, 1))
    
    # Final check and return
    radii = result.x
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
