import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance
import random

def run_packing():
    np.random.seed(42)
    n = 26
    max_iter = 100
    alpha = 0.95
    eps = 1e-6

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        return np.sum(radii)

    def update_centers(centers, radii):
        new_centers = np.copy(centers)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0:
                new_centers[i, 0] = r
            elif x + r > 1:
                new_centers[i, 0] = 1 - r
            if y - r < 0:
                new_centers[i, 1] = r
            elif y + r > 1:
                new_centers[i, 1] = 1 - r
        return new_centers

    def initial_guess():
        centers = np.zeros((n, 2))
        for i in range(n):
            if i % 5 == 0:
                centers[i, 0] = 0.2 + i * 0.1
                centers[i, 1] = 0.2
            else:
                centers[i, 0] = 0.2 + i * 0.1
                centers[i, 1] = 0.8
        return centers

    centers = initial_guess()
    radii = np.full(n, 0.01)

    for _ in range(max_iter):
        params = np.concatenate((centers.flatten(), radii))
        bounds = [(0, 1) for _ in range(2 * n)] + [(0, 1) for _ in range(n)]
        cons = []
        for i in range(n):
            for j in range(i + 1, n):
                cons.append((lambda x, i=i, j=j: distance.euclidean(x[2 * i:2 * i + 2], x[2 * j:2 * j + 2]) - (x[-n + i] + x[-n + j]) + 1e-8))
        res = minimize(objective, params, bounds=bounds, constraints=cons)
        params = res.x
        centers = params[:2 * n].reshape(n, 2)
        radii = params[2 * n:]
        centers = update_centers(centers, radii)

    return centers, radii, np.sum(radii)
