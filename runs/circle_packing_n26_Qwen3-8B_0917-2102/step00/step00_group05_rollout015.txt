import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance

def run_packing():
    np.random.seed(42)
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.1)
    max_iter = 100
    tol = 1e-5

    def objective(r):
        return -np.sum(r)

    def constraints(r):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - 1e-8:
                    return False
        for i in range(n):
            if centers[i, 0] - radii[i] < -1e-8 or centers[i, 0] + radii[i] > 1 + 1e-8:
                return False
            if centers[i, 1] - radii[i] < -1e-8 or centers[i, 1] + radii[i] > 1 + 1e-8:
                return False
        return True

    def bounds(r):
        return [(0, 1) for _ in range(n)]

    for _ in range(max_iter):
        res = minimize(
            objective,
            radii,
            bounds=bounds,
            constraints={"type": "callback", "fun": constraints},
            method="SLSQP"
        )
        radii = res.x
        if res.success and np.all(radii >= 0):
            break

    return centers, radii, np.sum(radii)
