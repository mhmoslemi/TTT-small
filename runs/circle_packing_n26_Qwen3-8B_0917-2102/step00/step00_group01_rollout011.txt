        return -np.sum(radii)

    def constraints(radii, centers):
        cons = []
        for i in range(len(radii)):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] * 2 - 1e-6})
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                def dist_func(r):
                    return np.sqrt(np.sum((centers[i] - centers[j]) ** 2)) - (r[i] + r[j]) + 1e-6
                cons.append({'type': 'ineq', 'fun': dist_func})
        return cons

    def bounds(radii):
        return [(0, 1) for _ in range(len(radii))]

    result = minimize(
        objective,
        initial_radii,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints(initial_radii, initial_centers),
        options={'ftol': 1e-8, 'maxiter': 1000}
    )

    centers = initial_centers
    radii = result.x
    sum_radii = np.sum(radii)

    return (centers, radii, sum_radii)
