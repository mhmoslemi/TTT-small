import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance
from itertools import combinations

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraint_dist(x, i, j):
        dx = x[2*i] - x[2*j]
        dy = x[2*i+1] - x[2*j+1]
        dist = np.sqrt(dx**2 + dy**2)
        return dist - (radii[i] + radii[j]) + 1e-8

    def constraint_boundary(x, i):
        x_i = x[2*i]
        y_i = x[2*i+1]
        r_i = radii[i]
        return (x_i - r_i) - 0 + 1e-8

    def constraint_boundary_right(x, i):
        x_i = x[2*i]
        y_i = x[2*i+1]
        r_i = radii[i]
        return (1 + 1e-8) - (x_i + r_i)

    def constraint_boundary_top(x, i):
        x_i = x[2*i]
        y_i = x[2*i+1]
        r_i = radii[i]
        return (1 + 1e-8) - (y_i + r_i)

    def constraint_boundary_bottom(x, i):
        x_i = x[2*i]
        y_i = x[2*i+1]
        r_i = radii[i]
        return (y_i - r_i) - 0 + 1e-8

    def initial_guess():
        grid = np.linspace(0.1, 0.9, 5)
        grid = np.sort(grid)
        centers = []
        for i in range(5):
            for j in range(5):
                if i % 2 == 0:
                    x = grid[i] + (grid[1] - grid[0]) / 2
                else:
                    x = grid[i]
                y = grid[j]
                centers.append([x, y])
        return np.array(centers)

    n = 26
    initial_centers = initial_guess()
    initial_radii = np.full(n, 0.05)
    x0 = np.concatenate([initial_centers.flatten(), initial_radii])

    bounds = []
    for i in range(n):
        bounds += [(0, 1)] * 2
        bounds += [(0, 1)]

    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: constraint_boundary(x, i)})
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: constraint_boundary_right(x, i)})
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: constraint_boundary_top(x, i)})
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: constraint_boundary_bottom(x, i)})

    for i in range(n):
        for j in range(i+1, n):
            cons.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: constraint_dist(x, i, j)})

    result = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6, options={'maxiter': 1000})

    x = result.x
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:]
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
