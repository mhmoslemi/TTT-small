import numpy as np
import math

def run_packing():

    # Generate centers in hexagonal grid
    centers = []

            # Adjust for boundary

        centers.append([x, y])

    for i in range(len(centers)):
        centers[i][0] += np.random.uniform(-0.01, 0.01)
        centers[i][1] += np.random.uniform(-0.01, 0.01)
    
    # Compute radii based on minimal distance between circles
    min_dist = np.inf
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = math.sqrt(dx*dx + dy*dy)
            if dist < min_dist:
                min_dist = dist

    # Set radius as min_dist / 2
    radius = min_dist / 2
    radii = [radius] * len(centers)
    
    # Adjust radii to ensure all circles are within the square

            radii[i] = x
        elif x + r > 1:
            radii[i] = 1 - x
        if y - r < 0:
            radii[i] = y
        elif y + r > 1:
            radii[i] = 1 - y

    # Compute total sum
    sum_radii = sum(radii)
    return (np.array(centers), np.array(radii), sum_radii)
