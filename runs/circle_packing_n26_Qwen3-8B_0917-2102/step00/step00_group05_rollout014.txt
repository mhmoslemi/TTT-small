from scipy.optimize import minimize

    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 0.5 * (row % 2))  # Offset for hexagonal packing
        y = row + 0.5
        centers[i] = [x, y]
    
    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)
    
    # Constraints: non-overlapping and within unit square
    def constraint(radii, centers):
        dist_matrix = np.sqrt(((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2).sum(axis=2))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        return np.sum(radii)
    
    # Optimization
    result = minimize(lambda x: -constraint(x, centers), x0=radii, method='SLSQP')
    radii = result.x
    sum_radii = np.sum(radii)
    
    # Final validation
    for i in range(n):
        x, y = centers[i]
        if (x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12
                or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12):
            raise ValueError("Circle outside unit square")
    
    return centers, radii, sum_radii
