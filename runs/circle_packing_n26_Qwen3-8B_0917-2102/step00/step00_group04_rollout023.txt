from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import linprog

def run_packing():

    n = 26

    radii = np.zeros(n)

    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))

    radii = np.full(n, 1 / (2 * (rows + cols)))
    
    # Place circles in hexagonal grid
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (row % 2)
        y = row
        centers[i] = [x, y]
    
    # Define optimization problem
    def objective(radii):
        return -np.sum(radii)
    
    def constraint_func(radii):
        dists = squareform(pdist(centers))

    initial_guess = np.full(n, 1 / (2 * (rows + cols)))
    
    # Optimization
    result = minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=[(0, 1) for _ in range(n)],
        constraints=[

                'type': 'ineq',
                'fun': lambda r: -constraint_func(r),

                'name': 'non_overlap'
            }
        ],
        options={'ftol': 1e-8, 'disp': False}
