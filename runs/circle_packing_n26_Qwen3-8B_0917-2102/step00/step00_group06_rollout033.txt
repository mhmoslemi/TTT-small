import numpy as np
import scipy.optimize as optimize

def run_packing():

    centers = []
    radii = []
    for r in range(5):
        for c in range(5 - r % 2):
            x = c + 0.5 + (r % 2) * 0.5
            y = r + 0.5
            centers.append([x, y])

        x = c + 0.5
        y = 5 + 0.5
        centers.append([x, y])
    
    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.full(26, 0.1)

    # Optimize radii to maximize sum under constraints
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraints_func(radii):
        n = len(radii)
        constraints = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return False
            for j in range(i + 1, n):
                dx = centers[j, 0] - centers[i, 0]
                dy = centers[j, 1] - centers[i, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True
    
    # Initial guess
    initial_guess = np.full(26, 0.1)

    # Bounds for radii
    bounds = [(0, 1) for _ in range(26)]

    # Optimization
    result = optimize.minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=optimize.Constraint(lambda x: constraints_func(x), {"type": "ineq"}),
        method='SLSQP'
    )

    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
