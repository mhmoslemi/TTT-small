import numpy as np
import scipy.optimize as opt

def run_packing():
    # Hexagonal packing parameters
    n = 26
    rows = 5
    cols = 5
    radius = 0.1

    # Base hexagonal grid
    centers = []
    radii = []

    # First row (top)
    for i in range(cols):
        x = i * (1 + radius) + radius
        y = radius
        centers.append([x, y])
        radii.append(radius)

    # Middle rows
    for row in range(1, rows - 1):
        for i in range(cols):
            x = i * (1 + radius) + radius
            y = row * (np.sqrt(3) * radius) + radius
            centers.append([x, y])
            radii.append(radius)

    # Bottom row
    for i in range(cols):
        x = i * (1 + radius) + radius
        y = (rows - 1) * (np.sqrt(3) * radius) + radius
        centers.append([x, y])
        radii.append(radius)

    # Resize radii to meet the target sum
    target_sum = 2.636
    current_sum = np.sum(radii)
    scale_factor = target_sum / current_sum
    radii = [r * scale_factor for r in radii]

    # Fine-tune positions to improve packing
    def objective(x):
        return -np.sum(x)

    def constraints(x):
        return np.sum(x) - target_sum

    bounds = [(0, 1) for _ in range(2 * n)]
    result = opt.optimize.minimize(
        objective,
        x0=np.concatenate(centers + radii),
        bounds=bounds,
        constraints={"type": "eq", "fun": constraints}
    )

    centers = np.array(result.x[:2 * n]).reshape(n, 2)
    radii = np.array(result.x[2 * n:])

    # Ensure all circles are within the square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            raise ValueError(f"Circle {i} out of bounds after optimization")

    # Ensure no overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError(f"Circles {i} and {j} overlap after optimization")

    return centers, radii, np.sum(radii)
