import numpy as np

from scipy.optimize import minimize

from scipy.optimize import Bounds, LinearConstraint

def run_packing():
    n = 26
    # Initial guess for centers - hexagonal packing with 4 rows
    centers = np.zeros((n, 2))
    r = np.zeros(n)
    for i in range(n):
        row = i // 6
        col = i % 6
        x = col + 0.5 * (1 - 0.5 * (row % 2))
        y = row * (np.sqrt(3) / 2)
        centers[i] = [x, y]
        r[i] = 0.1

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        constraints = []
        for i in range(n):
            constraints.append(radii[i] >= 0)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                constr = dist - (radii[i] + radii[j]) >= 0
                constraints.append(constr)
        for i in range(n):
            x, y = centers[i]
            constr = (x - radii[i] >= 0) + (x + radii[i] <= 1) + (y - radii[i] >= 0) + (y + radii[i] <= 1)
            constraints.append(constr)
        return constraints

    bounds = Bounds(lb=0, ub=np.inf)
    cons = [LinearConstraint(np.ones(n), -np.inf, 0) for _ in range(n)]
    for i in range(n):

            cons.append(LinearConstraint(np.concatenate([np.zeros(n), -np.eye(n)[i], -np.eye(n)[j]]), -np.inf, 0))

    result = minimize(objective, r, bounds=bounds, constraints=cons)
    radii = result.x
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
