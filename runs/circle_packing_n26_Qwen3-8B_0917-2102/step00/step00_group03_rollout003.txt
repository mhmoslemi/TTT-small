import numpy as np
import scipy.optimize as opt
import matplotlib.pyplot as plt

def run_packing():
    np.random.seed(42)
    n = 26
    
    # Initial guess: hexagonal packing with boundary circles
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Place boundary circles
    centers[0] = [0.1, 0.5]
    centers[1] = [0.9, 0.5]
    centers[2] = [0.5, 0.1]
    centers[3] = [0.5, 0.9]
    radii[0] = radii[1] = radii[2] = radii[3] = 0.15
    
    # Place inner circles in hexagonal lattice
    for i in range(4, n):
        row = (i - 4) // 6
        col = (i - 4) % 6
        x = 0.5 + (0.5 - 0.3) * (col - 2.5) / 3
        y = 0.5 + (0.5 - 0.3) * (row - 2.5) / 3
        centers[i] = [x, y]
        radii[i] = 0.1
    
    # Define constraints for optimization
    def objective(radii):
        return -np.sum(radii)
    
    def constraint_overlap(i, j):
        def cons(r):
            return np.sqrt(np.sum((centers[i] - centers[j]) ** 2)) - (radii[i] + radii[j])
        return cons
    
    bounds = [(0, 1)] * n
    
    # Optimization with simulated annealing
    result = opt.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[constraint_overlap(i, j) for i in range(n) for j in range(i+1, n)],
        method='SLSQP'
    )
    
    return (centers, result.x, np.sum(result.x))
