import numpy as np
import scipy.optimize as opt

def run_packing():
    # Define initial centers for hexagonal packing
    centers = np.array([
        [0.5, 0.5],
        [0.25, 0.25], [0.75, 0.25],
        [0.125, 0.125], [0.375, 0.125], [0.625, 0.125], [0.875, 0.125],
        [0.0625, 0.3125], [0.28125, 0.3125], [0.5, 0.3125], [0.71875, 0.3125], [0.9375, 0.3125],
        [0.15625, 0.5], [0.390625, 0.5], [0.609375, 0.5], [0.84375, 0.5],
        [0.21875, 0.6875], [0.4375, 0.6875], [0.65625, 0.6875], [0.875, 0.6875],
        [0.375, 0.875], [0.625, 0.875],
    ])
    
    # Initial radii based on spacing
    radii = np.ones(26) * 0.1

    # Optimization: maximize sum of radii with constraints
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Constraints: circles must not overlap and stay within square
    def constraint_overlap(radii):
        n = len(radii)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    def constraint_boundary(radii):
        for i in range(len(radii)):
            x, y = centers[i]
            if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
                return -1
        return 0

    # Optimization bounds
    bounds = [(0, 1) for _ in range(26)]

    # Optimization with constraints
    res = opt.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_overlap},
            {'type': 'ineq', 'fun': constraint_boundary}
        ],
        method='SLSQP'
    )

    return (centers, res.x, np.sum(res.x))
