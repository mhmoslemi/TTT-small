import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    np.random.seed(42)
    n = 26
    num_cols = 6
    num_rows = 5
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    max_radius = 0.1

    # Hexagonal packing with varying radii
    for i in range(n):
        row = i // num_cols
        col = i % num_cols
        x = col + 0.5 * (1 - (1 / (num_cols + 1)))
        y = row + 0.5 * (1 - (1 / (num_rows + 1)))
        centers[i] = [x, y]

    def objective(r):
        return -np.sum(r)  # Maximize sum of radii

    def constraint(r, i, j):
        dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
        return dist - r[i] - r[j]

    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({
                'type': 'ineq',
                'fun': lambda r, i=i, j=j: constraint(r, i, j),
                'jac': lambda r, i=i, j=j: np.concatenate([[-1 if k == i else 0 for k in range(n)], [-1 if k == j else 0 for k in range(n)]]),
                'args': (i, j)
            })

    for i in range(n):
        cons.append({
            'type': 'ineq',
            'fun': lambda r, i=i: np.sqrt(centers[i][0] ** 2 + centers[i][1] ** 2) - r[i],
            'jac': lambda r, i=i: np.concatenate([[-1 if k == i else 0 for k in range(n)]]),
            'args': (i,)
        })
        cons.append({
            'type': 'ineq',
            'fun': lambda r, i=i: np.sqrt((1 - centers[i][0]) ** 2 + (1 - centers[i][1]) ** 2) - r[i],
            'jac': lambda r, i=i: np.concatenate([[-1 if k == i else 0 for k in range(n)]]),
            'args': (i,)
        })

    bounds = [(0, 1) for _ in range(n)]

    result = minimize(objective, x0=max_radius * np.ones(n), constraints=cons, bounds=bounds)

    radii = result.x
    sum_radii = np.sum(radii)

    return (centers, radii, sum_radii)
