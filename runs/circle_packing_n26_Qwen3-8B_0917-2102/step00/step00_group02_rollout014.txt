import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    n = 26

    # Initial guess: hexagonal packing with varying radii
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = 0.1

    def objective(r):
        return -np.sum(r)

    def constraints(r):
        violated = False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                d = np.sqrt(dx**2 + dy**2)
                if d < r[i] + r[j] - 1e-8:
                    violated = True
        for i in range(n):
            x, y = centers[i]
            if x - r[i] < -1e-8 or x + r[i] > 1 + 1e-8 or y - r[i] < -1e-8 or y + r[i] > 1 + 1e8:
                violated = True
        return violated

    bounds = [(0, 0.5) for _ in range(n)]
    result = minimize(objective, radii, bounds=bounds, constraints={"type": "callback", "fun": constraints})
    radii = result.x

    # Post-optimization boundary adjustments
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            radii[i] = x
        elif x + r > 1:
            radii[i] = 1 - x
        if y - r < 0:
            radii[i] = y
        elif y + r > 1:
            radii[i] = 1 - y

    # Final check for overlaps
    for i in range(n):

            dx = centers[i, 0] - centers[j, 0]

            if d < radii[i] + radii[j] - 1e-8:
                radii[i] = (d - radii[j]) * 0.5
                radii[j] = (d - radii[i]) * 0.5

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
