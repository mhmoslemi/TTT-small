import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial hexagonal packing with 5 rows (max 26 circles)
    centers = []
    radii = []
    for row in range(5):
        for col in range(5 - row):
            x = col + 0.5 * row / (1 + 0.5)
            y = 0.5 + row * (1 + 0.5) / 2
            centers.append([x, y])
    
    # Add boundary circles
    for i in range(25):
        if i % 5 == 0:
            centers.append([0.5, i / 5 + 0.5])
        elif i % 5 == 4:
            centers.append([0.5, i / 5 + 0.5])
    
    centers = np.array(centers)
    radii = np.array([0.1] * len(centers))
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraints(radii):
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        for i in range(len(radii)):
            x, y = centers[i]
            if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
                return False
        return True
    
    res = minimize(objective, radii, method='SLSQP', constraints={'type': 'ineq', 'fun': constraints})
    return centers, res.x, np.sum(res.x)
