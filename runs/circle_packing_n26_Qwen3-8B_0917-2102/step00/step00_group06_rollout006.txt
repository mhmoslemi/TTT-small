import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance

def run_packing():
    np.random.seed(42)
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n) * 0.1
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraints(radii):
        constraints = []
        for i in range(n):
            constraints.append((radii[i] <= 1 - centers[i, 0] - 1e-8))
            constraints.append((radii[i] <= 1 - centers[i, 1] - 1e-8))
            constraints.append((radii[i] <= centers[i, 0] + 1e-8))
            constraints.append((radii[i] <= centers[i, 1] + 1e-8))
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                constraints.append((radii[i] + radii[j] <= dist + 1e-8))
        return constraints
    
    bounds = [(0, 1) for _ in range(n)]
    res = minimize(objective, radii, bounds=bounds, constraints=constraints)
    
    if not res.success:
        raise ValueError("Optimization failed")
    
    radii = res.x
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
