import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    # Initial hexagonal packing with 5 rows
    centers = []
    radii = []
    for row in range(5):
        for col in range(5):
            x = col + 0.5 * row / 2
            y = row * np.sqrt(3) / 2
            if row % 2 == 0:
                x = col + 0.5 * row / 2
            else:
                x = col + 0.5 * (row + 1) / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
                radii.append(0.05)
    centers = np.array(centers)
    radii = np.array(radii)
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraints(radii):
        cons = []
        for i in range(len(radii)):
            cons.append((radii[i], 0, 1))
        return cons
    
    def evaluate(x):
        radii = x
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        for i in range(len(radii)):
            if centers[i, 0] - radii[i] < 0 or centers[i, 0] + radii[i] > 1:
                return -np.inf
            if centers[i, 1] - radii[i] < 0 or centers[i, 1] + radii[i] > 1:
                return -np.inf
        return np.sum(radii)
    
    result = minimize(lambda x: -evaluate(x), radii, method='SLSQP', bounds=[(0, 1)] * n, options={'maxiter': 200})
    radii = result.x
    centers = np.array([centers[i] for i in range(len(radii))])
    
    return (centers, radii, np.sum(radii))
