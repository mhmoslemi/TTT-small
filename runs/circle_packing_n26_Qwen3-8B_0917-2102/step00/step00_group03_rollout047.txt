import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance
from itertools import combinations

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        n = len(radii)
        cons = []
        for i in range(n):
            cons.append((radii[i], 1.0))
        for i in range(n):
            for j in range(i+1, n):
                cons.append((radii[i] + radii[j], distance.euclidean(centers[i], centers[j])))
        return cons

    def pack_hexagonal(n):
        centers = []
        for i in range(n):
            row = i // 6
            col = i % 6
            x = col / 6.0 + 0.5 / 6.0
            y = row / 6.0 + 0.5 / 6.0
            centers.append([x, y])
        return np.array(centers)

    centers = pack_hexagonal(26)
    initial_radii = np.ones(26) * 0.1

    bounds = [(0, 1.0) for _ in range(26)]
    constraints = []
    for i in range(26):
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] * 2 <= 1.0})
        for j in range(i + 1, 26):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: r[i] + r[j] <= dist})

    result = minimize(objective, initial_radii, bounds=bounds, constraints=constraints)

    radii = result.x
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
