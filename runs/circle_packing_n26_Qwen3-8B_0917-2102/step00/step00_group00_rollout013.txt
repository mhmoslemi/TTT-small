import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial hexagonal packing with 5 rows, each with 5 circles
    rows = 5
    cols = 5
    # Generate initial positions with hexagonal packing
    centers = []
    radii = []
    for i in range(rows):
        for j in range(cols):
            x = j + 0.5 * (i % 2)
            y = i * np.sqrt(3)/2
            centers.append([x, y])
            radii.append(0.05)

    # Adjust for unit square
    scale = 1 / (np.sqrt(3)/2 * (rows - 1) + 0.5)
    centers = np.array(centers) * scale
    radii = np.array(radii) * scale

    # Add additional circles in the remaining space
    # Add 21 circles (total of 26)
    for _ in range(21):
        # Randomly place test points in the unit square
        x = np.random.uniform(0, 1)
        y = np.random.uniform(0, 1)
        centers.append([x, y])
        radii.append(0.01)

    # Optimize to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        # Ensure all circles are non-overlapping
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        # Ensure circles are within the unit square
        for i in range(len(radii)):
            x, y = centers[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12:
                return False
            if y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return False
        return True

    # Optimize with constraints
    bounds = [(0, 1) for _ in range(26)]
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[
            {
                'type': 'ineq',
                'fun': lambda x: constraints(x)
            }
        ],
        method='SLSQP'
    )

    # Return the result
    return (centers, result.x, np.sum(result.x))
