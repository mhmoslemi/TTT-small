from scipy.optimize import minimize

def run_packing():

    num_circles = 26
    row_count = 5
    col_count = 5
    centers = []
    radii = []

    # Top row with 5 circles
    for i in range(col_count):
        x = i + 0.5
        y = 1 - 0.5
        centers.append([x, y])
        radii.append(0.1)

    # Middle rows with decreasing number of circles
    for i in range(1, row_count - 1):
        row = []
        for j in range(col_count - i):
            x = j + 0.5
            y = 1 - (0.5 + i * (1.0 / (row_count - 1)))
            row.append([x, y])
        centers.extend(row)
        radii.extend([0.1] * len(row))

    # Bottom row with 5 circles
    for i in range(col_count):
        x = i + 0.5
        y = 0.5
        centers.append([x, y])
        radii.append(0.1)

    # Initial packing
    centers = np.array(centers)
    radii = np.array(radii)

    # Objective: maximize total radius
    def objective(radii):
        return -np.sum(radii)

    # Constraints: non-overlapping and within bounds
    def constraint(x):
        centers_flat = x[:26]
        radii_flat = x[26:]
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers_flat[i*2] - centers_flat[j*2]
                dy = centers_flat[i*2+1] - centers_flat[j*2+1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii_flat[i] + radii_flat[j] - 1e-8:
                    return -1
        for i in range(26):
            x, y = centers_flat[i*2], centers_flat[i*2+1]
            r = radii_flat[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:

        return 0

    # Optimization
    x0 = np.concatenate((centers.flatten(), radii))
    res = minimize(objective, x0, constraints={"type": "ineq", "fun": constraint}, method="SLSQP")

    # Extract results
    centers_opt = res.x[:52].reshape(26, 2)
    radii_opt = res.x[52:]

    return centers_opt, radii_opt, np.sum(radii_opt)
