import numpy as np
from scipy.optimize import minimize

def run_packing():

    # Start with 5 rows of hexagonal packing
    centers = []
    radii = []
    for row in range(5):
        for col in range(5 - row):
            x = col + 0.5 * row
            y = row
            centers.append([x, y])
            radii.append(0.05)
    
    # Add 11 more circles in gaps
    add_centers = [
        [0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9],
        [0.4, 0.45], [0.6, 0.45], [0.4, 0.55], [0.6, 0.55],
        [0.25, 0.7], [0.75, 0.7], [0.5, 0.8]
    ]
    add_radii = [0.05] * 11
    centers += add_centers
    radii += add_radii

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Initial sum of radii
    sum_radii = np.sum(radii)

    # Define objective function to maximize sum of radii
    def objective(x, centers, radii):
        # x is the vector of parameters to optimize
        # For this simple case, we'll manually adjust radii and positions
        return -np.sum(radii)  # We minimize -sum for maximization

    # Constraints function
    def constraints(x, centers, radii):
        # Check boundary and non-overlapping constraints
        for i in range(len(centers)):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12:
                return False
            if y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return False
            for j in range(i + 1, len(centers)):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    # Optimization
    result = minimize(objective, np.concatenate((centers.flatten(), radii)), 
                      method='SLSQP', constraints={'type': 'ineq', 'fun': constraints, 'args': (centers, radii)},

    # Extract results
    centers_opt = result.x[:2 * len(centers)].reshape(-1, 2)
    radii_opt = result.x[2 * len(centers):]

    # Ensure non-negative radii
    radii_opt = np.maximum(radii_opt, 0)

    # Final sum
    sum_radii = np.sum(radii_opt)

    return centers_opt, radii_opt, sum_radii
