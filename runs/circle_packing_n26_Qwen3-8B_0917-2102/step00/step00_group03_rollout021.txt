import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    # Initial hexagonal packing with 26 circles
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        row = i // 6
        col = i % 6
        centers[i, 0] = col + 0.5
        centers[i, 1] = row + 0.5
    # Adjust for unit square
    centers[:, 0] *= 1 / 6
    centers[:, 1] *= 1 / 6
    # Initial radii based on spacing
    initial_r = 0.1
    radii = initial_r * np.ones(n)
    
    # Objective function: maximize sum of radii
    def objective(r):
        return -np.sum(r)  # Negative for minimization

    # Constraints: non-overlapping and within square
    def constraint_func(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return 1
        for i in range(n):
            if centers[i, 0] - r[i] < 0 or centers[i, 0] + r[i] > 1:
                return 1
            if centers[i, 1] - r[i] < 0 or centers[i, 1] + r[i] > 1:
                return 1
        return 0

    # Optimization
    res = minimize(
        objective,
        radii,
        bounds=[(0, 1)] * n,
        constraints=[{'type': 'ineq', 'fun': constraint_func}],
        method='SLSQP'
    )
    
    centers = np.clip(centers, 0, 1)
    radii = res.x
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
