import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial guess: hexagonal packing with spacing based on circle radii
    initial_centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        initial_centers[i, 0] = col * 0.4 + 0.2
        initial_centers[i, 1] = row * 0.4 + 0.2
    # Optimization parameters
    bounds = [(0, 1) for _ in range(2 * n)]
    constraints = []
    # Objective: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)
    # Constraint: circles must be within the square
    def square_constraint(radii, centers):
        return np.sum((centers[:, 0] - radii) >= 0) + np.sum((1 - centers[:, 0] - radii) >= 0) + np.sum((centers[:, 1] - radii) >= 0) + np.sum((1 - centers[:, 1] - radii) >= 0)
    # Constraint: circles must not overlap
    def no_overlap_constraint(radii, centers):
        dist = squareform(pdist(centers))
        return np.sum(np.sum(dist < radii[:, np.newaxis] + radii[np.newaxis, :], axis=1) == 0)
    # Run optimization
    result = minimize(objective, x0=np.ones(n), bounds=bounds, constraints=[{'type': 'ineq', 'fun': lambda x: square_constraint(x, initial_centers)}, {'type': 'ineq', 'fun': lambda x: no_overlap_constraint(x, initial_centers)}])
    radii = result.x
    centers = initial_centers
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
