from scipy.optimize import minimize

def run_packing():

    num_circles = 26
    grid_cols = 6
    grid_rows = 5
    spacing = 0.2
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    for i in range(grid_rows):
        for j in range(grid_cols):
            col = j * spacing
            row = i * spacing * np.sqrt(3) / 2
            idx = i * grid_cols + j
            centers[idx] = [col, row]
    # Adjust for boundary and corner placements
    for i in range(num_circles):
        x, y = centers[i]
        radius = 0.1
        if i % grid_cols == 0:
            radius = 0.15
        elif i % grid_cols == grid_cols - 1:
            radius = 0.15
        elif i < grid_cols:
            radius = 0.12
        elif i >= grid_rows * grid_cols - grid_cols:
            radius = 0.12
        radii[i] = radius

    # Optimization to maximize sum of radii while maintaining non-overlapping and boundary constraints
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        dists = np.zeros((num_circles, num_circles))
        for i in range(num_circles):
            for j in range(num_circles):
                if i != j:
                    dists[i, j] = np.linalg.norm(centers[i] - centers[j])
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    def bounds(radii):
        for i in range(num_circles):
            if radii[i] < 0:
                return False
        return True

    # Initial guess for radii
    initial_radii = np.copy(radii)
    result = minimize(objective, initial_radii, method='SLSQP',
                      bounds=[(0, 1)] * num_circles,
                      constraints=[{'type': 'ineq', 'fun': constraints}],
                      options={'eps': 1e-6})

    # Final centers and radii
    return (centers, result.x, np.sum(result.x))
