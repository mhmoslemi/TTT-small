from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    num_circles = 26
    centers = []
    radii = []

    # First row (top)

        x = i / 12
        y = 0.95
        centers.append([x, y])
        radii.append(0.05)

    # Second row (bottom)
    for i in range(13):
        x = i / 12
        y = 0.05
        centers.append([x, y])
        radii.append(0.05)

    centers = np.array(centers)
    radii = np.array(radii)

    def objective(r):
        return -np.sum(r)

    def constraints(r):
        n = len(r)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < r[i] + r[j] - 1e-8:
                    return False
        return True

    def bounds():
        return [(0, 1) for _ in range(num_circles)]

    res = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds(),
        constraints={'type': 'ineq', 'fun': constraints, 'jac': None},
        options={'maxiter': 500, 'eps': 1e-8}
    )

    if not res.success:
        raise ValueError("Optimization failed")

    centers = np.array(centers)
    radii = res.x

    # Final check for boundary constraints
    for i in range(num_circles):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            raise ValueError(f"Circle {i} is out of bounds")

    return centers, radii, np.sum(radii)
