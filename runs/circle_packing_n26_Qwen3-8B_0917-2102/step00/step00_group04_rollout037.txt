import numpy as np
import scipy.optimize as opt

def run_packing():
    # Base hexagonal packing with 6 rows
    rows = 5
    cols = 5
    centers = []
    radii = []
    for i in range(rows):
        for j in range(cols):
            x = j + 0.5 * (i % 2)
            y = i * np.sqrt(3) / 2
            x = x / (cols - 1) if cols > 1 else 0
            y = y / (rows - 1) if rows > 1 else 0
            centers.append([x, y])
            radii.append(0.05)
    
    # Add extra circles in corners
    extra = [
        [0.1, 0.1], [0.9, 0.1], [0.1, 0.9], [0.9, 0.9]
    ]
    for x, y in extra:
        centers.append([x, y])
        radii.append(0.03)
    
    # Optimize radii using scipy.optimize
    def objective(r):
        return -sum(r)
    
    def constraints(r):
        n = len(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - 1e-8:
                    return False
        for i in range(n):
            x, y = centers[i]
            if x - r[i] < 0 or x + r[i] > 1 or y - r[i] < 0 or y + r[i] > 1:
                return False
        return True
    
    bounds = [(0, 0.1) for _ in range(len(radii))]
    
    result = opt.minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraints}]
    )
    
    return (np.array(centers), np.array(result.x), sum(result.x))
