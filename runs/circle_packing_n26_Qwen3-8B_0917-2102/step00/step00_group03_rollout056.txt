import numpy as np
from scipy.optimize import minimize

def run_packing():

    # 26 circles arranged in 5 rows with 5, 5, 5, 5, 6 circles respectively
    centers = np.zeros((26, 2))
    radius_guess = 0.15

    for i in range(5):

            row = i
            col = j
            x = col + 0.5 * (i % 2)

            centers[i * 5 + j] = [x, y]

    # Ensure all centers are within the square and adjust for boundary
    centers[:, 0] = np.clip(centers[:, 0], 0, 1)
    centers[:, 1] = np.clip(centers[:, 1], 0, 1)

    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Constraint: circles must not overlap and stay within square
    def constraint_func(radii):

        for i in range(26):
            x, y = centers[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12 or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return -np.inf
        # Check overlaps
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -np.inf
        return np.sum(radii)

    # Optimization
    bounds = [(0, 1) for _ in range(26)]
    result = minimize(objective, x0=np.full(26, radius_guess), bounds=bounds, constraints={"type": "ineq", "fun": constraint_func})

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
