from scipy.optimize import Bounds

    centers = []
    radii = []

    for i in range(13):

    # Middle rows

            x = i / 12 + 0.5 / 12

        x = i / 12
        y = 0.9

        radii.append(0.05)

    radii = np.array(radii)
    
    def objective(radii):

    def constraint_func(radii):

        dist_matrix = squareform(pdist(centers))
        for i in range(n):
            for j in range(i+1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        return np.sum(radii)
    
    bounds = [(0, 0.1) for _ in range(26)]
    
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        constraints=[
            {'type': 'ineq', 'fun': constraint_func}
        ],
        bounds=Bounds([0]*26, [0.1]*26)
    )
    
    return centers, result.x, np.sum(result.x)
