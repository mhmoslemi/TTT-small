import numpy as np
from scipy.optimize import minimize

def run_packing():

    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.1)
    initial_sum = np.sum(radii)

        return -np.sum(r)

        return np.sum(r)

    radii = result.x

    for _ in range(100):
        for i in range(n):

                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-5:
                    alpha = (radii[i] + radii[j] - dist) / (dist - 1e-5)
                    centers[i] += alpha * (dx, dy)
                    centers[j] -= alpha * (dx, dy)

    # Final validation
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-12:
                radii[i] = (dist - radii[j]) * 0.5
                radii[j] = (dist - radii[i]) * 0.5

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
