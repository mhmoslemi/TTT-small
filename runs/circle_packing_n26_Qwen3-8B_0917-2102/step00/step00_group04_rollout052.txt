from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    np.random.seed(42)
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.1)
    best_sum = 0

    def objective(r):
        nonlocal best_sum
        if sum(r) > best_sum:
            best_sum = sum(r)
        return -sum(r)

    def constraint(r, idx):
        if r[idx] < 0:
            return -np.inf
        for i in range(n):
            if i == idx:
                continue
            dist = np.sqrt(np.sum((centers[idx] - centers[i]) ** 2))
            if dist < r[idx] + r[i] - 1e-8:
                return -np.inf
        return r[idx]

    for i in range(n):
        cons = []
        for j in range(n):
            if i == j:
                continue
            cons.append({'type': 'ineq', 'fun': lambda r, idx=i, j=j: constraint(r, j)})
        result = minimize(lambda r: -sum(r), radii, constraints=cons)
        if result.success:
            radii = result.x

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            radii[i] = min(r, 1 - np.minimum(x, 1 - x))

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-8:
                radii[i] = (dist - radii[j] + 1e-8) / 2
                radii[j] = (dist - radii[i] + 1e-8) / 2

    return centers, radii, sum(radii)
