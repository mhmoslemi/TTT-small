import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial guess: clustered hexagonal packing with 7 rows
    centers = np.zeros((n, 2))
    r = np.full(n, 0.05)
    row = 0
    col = 0
    for i in range(n):
        if i % 7 == 0:
            row += 1
            col = 0
        centers[i, 0] = col * (1 + 0.5) / (n // 2)
        centers[i, 1] = row * (1 + 0.5) / (n // 2)
        col += 1

    def obj_func(radii):
        return -np.sum(radii)

    def constraints(radii):
        dist_matrix = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return False
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8 or y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8:
                return False
        return True

    result = minimize(obj_func, r, method='SLSQP', constraints={'type': 'ineq', 'fun': constraints})
    radii = result.x
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
