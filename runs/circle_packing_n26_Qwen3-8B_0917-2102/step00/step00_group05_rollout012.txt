import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial hexagonal packing with 26 circles
    n = 26
    rows = 6
    cols = 5
    spacing = 1.0 / (np.sqrt(3) * (cols - 1))
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Create a hexagonal grid with adaptive spacing
    for i in range(rows):
        for j in range(cols):
            col = j
            row = i
            x = col + 0.5 * (row % 2)
            y = row * np.sqrt(3) / 2
            if i % 2 == 0:
                x = j + 0.5
            else:
                x = j + 0.5 + 0.5
            x = x / (cols + 1)
            y = y / (rows + 1)
            centers[i * cols + j] = [x, y]
            radii[i * cols + j] = 0.1

    # Define objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Define constraints: circles must fit in unit square and not overlap
    def constraint_center(radii):
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12 or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return -1
        return 0

    def constraint_overlap(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0

    # Optimization
    bounds = [(0, 1) for _ in range(n)]
    cons = [{'type': 'ineq', 'fun': constraint_center},
            {'type': 'ineq', 'fun': constraint_overlap}]
    
    result = optimize.minimize(objective, radii, bounds=bounds, constraints=cons, method='SLSQP')
    centers = result.x
    radii = result.x
    sum_radii = np.sum(radii)
    
    return (centers.reshape(-1, 2), radii, sum_radii)
