import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        return np.sum(radii) - 2.636

    def bounds(radii):
        return np.zeros_like(radii)

    def inequality_constraints(radii):
        return np.array([radii[i] >= 0 for i in range(26)])

    def circle_positions():
        hex_grid = []
        radius = 0.1
        for i in range(26):
            row = i // 5
            col = i % 5
            x = col * (1 + radius)
            y = row * (1 + radius)
            if i == 25:
                x = 0.5
                y = 0.5
            hex_grid.append([x, y])
        return np.array(hex_grid)

    centers = circle_positions()
    radii = np.full(26, 0.1)

    def update_centers(radii):
        new_centers = []
        for i in range(26):
            if i == 25:
                new_centers.append([0.5, 0.5])
            else:
                row = i // 5
                col = i % 5
                x = col * (1 + radii[i])
                y = row * (1 + radii[i])
                new_centers.append([x, y])
        return np.array(new_centers)

    def distance_matrix(centers):
        dists = squareform(pdist(centers))
        return dists

    def constraints_matrix(centers, radii):
        dists = distance_matrix(centers)
        return dists - radii[:, np.newaxis] - radii[np.newaxis, :]

    def optimize_packing():
        bounds = [(0, 1) for _ in range(26)]
        res = minimize(
            objective,
            x0=radii,
            bounds=bounds,
            constraints=[
                {'type': 'ineq', 'fun': lambda x: np.min(np.abs(np.array(x) - 0.1))},
                {'type': 'ineq', 'fun': lambda x: np.sum(x) - 2.636}
            ],
            method='SLSQP'
        )
        return res.x

    radii = optimize_packing()
    centers = update_centers(radii)
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
