import numpy as np
import scipy.optimize as opt

def run_packing():
    # Initial hexagonal arrangement with 26 circles
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Hexagonal packing with staggered rows
    rows = 6
    cols = 5
    spacing = 1.0 / (rows + 1)
    hex_row = 0.5 * np.sqrt(3)
    
    for i in range(rows):
        row_offset = i * spacing
        for j in range(cols):
            col_offset = j * spacing
            x = col_offset
            y = row_offset + i * hex_row
            centers[i * cols + j] = [x, y]
    
    # Optimize radii to maximize sum while ensuring non-overlap and boundary constraints
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum
    
    def constraints(radii):
        constraints = []
        for i in range(n):
            # Boundary constraint
            if i == 0:
                constraints.append(radii[i] <= centers[i, 0] - 1e-12)
            elif i == n - 1:
                constraints.append(radii[i] <= 1 - centers[i, 0] - 1e-12)
            if i == n // 2:
                constraints.append(radii[i] <= centers[i, 1] - 1e-12)
            elif i == n // 2 + 1:
                constraints.append(radii[i] <= 1 - centers[i, 1] - 1e-12)
            
            # Pairwise distance constraint
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constr = radii[i] + radii[j] <= dist + 1e-12
                constraints.append(constr)
        return constraints
    
    # Initial guess for radii
    radii_guess = np.full(n, 0.05)
    
    # Optimization
    opt_result = opt.minimize(
        objective,
        radii_guess,
        constraints=[{"type": "ineq", "fun": lambda r: constraint} for constraint in constraints(radii_guess)]
    )
    
    radii = opt_result.x
    sum_radii = np.sum(radii)
    
    return (centers, radii, sum_radii)
