import numpy as np
from scipy.optimize import minimize

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraint1(radii):
        return np.sum(radii) - 2.636

    def constraint2(radii):
        return np.sum(radii) - 2.636

    def bounds():
        return [(0, 1) for _ in range(26)]

    def initial_guess():
        return np.full(26, 0.05)

    result = minimize(
        objective,
        initial_guess(),
        method='SLSQP',
        bounds=bounds(),
        constraints=[
            {'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2}
        ]
    )

    radii = result.x
    centers = np.random.rand(26, 2)
    centers = np.clip(centers, 0, 1)

    return centers, radii, np.sum(radii)
