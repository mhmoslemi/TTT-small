from scipy.optimize import minimize

def run_packing():

    centers = []
    radii = []

            x = j / 12 + 0.05
            y = i / 5 + 0.05
            centers.append([x, y])

    def obj_func(radii):
        return -np.sum(radii)

    def constraints(radii):
        n = len(radii)
        cons = []
        for i in range(n):
            cons.append((radii[i], 0, np.inf))
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                cons.append((radii[i] + radii[j], dist, np.inf))
        return cons

    radii_opt = minimize(obj_func, radii, method='SLSQP', bounds=[(0, np.inf)] * len(radii),
                         constraints=[{'type': 'ineq', 'fun': lambda x, i=i: x[i] if i < len(x) else 0} for i in range(len(radii))])
    
    radii = radii_opt.x
    sum_radii = np.sum(radii)
    
    # Final validation
    for i in range(len(radii)):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:

    for i in range(len(radii)):

                raise ValueError(f"Circles {i} and {j} overlap")
    
    return (centers, radii, sum_radii)
