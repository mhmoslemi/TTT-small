import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        constraints = []
        for i in range(26):
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] * 2 - 1e-12})
        return constraints

    def pack_hexagonal():
        centers = []
        radii = []
        for row in range(4):
            for col in range(6):
                x = col / 6.0 + 0.5 / 6.0
                y = row / 4.0 + 0.5 / 4.0
                if row % 2 == 0:
                    x += 0.5 / 6.0
                centers.append([x, y])
                radii.append(0.05)
        return np.array(centers), np.array(radii)

    centers, radii = pack_hexagonal()
    result = minimize(objective, radii, constraints=constraints(radii), method='SLSQP')
    return (centers, result.x, np.sum(result.x))
