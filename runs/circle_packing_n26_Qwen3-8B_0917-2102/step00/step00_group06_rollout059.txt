import numpy as np
import scipy.optimize as optimize

def run_packing():

    # First layer: 10 circles in a hexagonal arrangement
    # Second layer: 16 circles in a hexagonal arrangement, spaced to avoid overlap

    layer1 = []
    for i in range(5):
        x = 0.1 + i * 0.2
        y = 0.4
        layer1.append([x, y])
    
    # Second layer: 16 circles
    layer2 = []

        x = 0.1 + i * 0.15
        y = 0.7
        layer2.append([x, y])
    
    centers = np.array(layer1 + layer2)
    radii = np.full(26, 0.1)
    
    # Optimization to maximize sum of radii under constraints
    def objective(r):
        return -np.sum(r)
    
    def constraints(r):
        # Boundary constraints
        for i in range(26):
            if (centers[i, 0] - r[i] < 0 or centers[i, 0] + r[i] > 1 or
                centers[i, 1] - r[i] < 0 or centers[i, 1] + r[i] > 1):
                return False
        # Pairwise distance constraints
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < r[i] + r[j] - 1e-8:
                    return False
        return True
    
    bounds = [(0, 1) for _ in range(26)]
    
    result = optimize.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints={'type': 'successive', 'fun': constraints, 'method': 'SLSQP'}
    )
    
    if not result.success:
        raise Exception("Optimization failed")
    
    return centers, result.x, np.sum(result.x)
