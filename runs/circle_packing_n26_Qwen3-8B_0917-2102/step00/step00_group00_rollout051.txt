import numpy as np
import scipy.optimize as opt

def run_packing():
    # Initial guess for circle centers using hexagonal packing
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Hexagonal packing with dense rows
    row_count = 6
    col_count = 5
    spacing = 0.5 / (np.sqrt(3) * (col_count - 1))
    
    for i in range(row_count):
        for j in range(col_count):
            x = j * spacing
            y = i * (2 * spacing / np.sqrt(3))
            if i % 2 == 0:
                x += spacing / 2
            centers[i * col_count + j] = [x, y]
    
    # Adjust positions near boundaries for better packing
    for i in range(n):
        x, y = centers[i]
        if x < 0.1:
            centers[i, 0] = 0.1
        elif x > 0.9:
            centers[i, 0] = 0.9
        if y < 0.1:
            centers[i, 1] = 0.1
        elif y > 0.9:
            centers[i, 1] = 0.9
    
    # Initial radii based on minimum distance to neighbors
    def initial_radii():
        r = np.zeros(n)
        for i in range(n):
            min_dist = np.inf
            for j in range(n):
                if i != j:
                    dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                    if dist < min_dist:
                        min_dist = dist
            r[i] = min_dist / 2
        return r
    
    radii = initial_radii()
    
    # Optimization to maximize sum of radii with constraints
    def objective(r):
        return -np.sum(r)
    
    def constraint_func(r):
        dist_matrix = np.zeros((n, n))
        for i in range(n):
            for j in range(n):
                if i != j:
                    dist_matrix[i, j] = np.sqrt(np.sum((centers[i] - centers[j])**2))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < r[i] + r[j] - 1e-8:
                    return 1
        return 0
    
    constraints = [{'type': 'ineq', 'fun': constraint_func}]
    bounds = [(0, 1) for _ in range(n)]
    
    result = opt.minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=constraints)
    
    if not result.success:
        raise ValueError("Optimization failed")
    
    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
