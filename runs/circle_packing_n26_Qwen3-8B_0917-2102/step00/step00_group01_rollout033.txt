<strategy>Use hexagonal packing with optimized boundary circles to maximize total radius sum.</strategy>

```python
import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    n = 26
    # Initial guess: 5 rows of 5 circles with hexagonal packing
    centers = np.zeros((n, 2))
    r = np.full(n, 0.1)
    row_height = np.sqrt(3)/2
    col_width = 1
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col * col_width / 4 + 0.5 * (col % 2)
        y = row * row_height
        centers[i] = [x, y]
    
    # Optimization parameters
    bounds = Bounds([0, 0], [1, 1])
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraint(radii):
        dist_matrix = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0
    
    result = minimize(
        objective,
        r,
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraint},
        method="SLSQP"
    )
    
    radii = result.x
    # Check boundary constraints
    for i in range(n):
        x, y = centers[i]
        if (x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8
            or y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8):
            radii[i] = 0
    
    return centers, radii, np.sum(radii)
```
