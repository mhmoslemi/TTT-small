import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    n = 26
    # Start with a hexagonal packing with 5 rows
    centers = []
    radii = []
    for row in range(5):
        for col in range(6):
            x = col + 0.5 * row / 2
            y = row + 0.5
            if row % 2 == 0:
                x = col * (1 + 1 / 6) + 0.5 * row / 2
            else:
                x = (col + 0.5) * (1 + 1 / 6) - 0.5 * row / 2
            if row < 2:
                x = col * (1 + 1 / 5) + 0.5 * row / 2
            if row < 4:
                y = row + 0.5
            centers.append([x, y])
    centers = np.array(centers)
    radii = np.full(n, 0.05)

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        constraints = []
        for i in range(n):
            constraints.append((centers[i, 0] - radii[i] >= 0))
            constraints.append((centers[i, 1] - radii[i] >= 0))
            constraints.append((1 - centers[i, 0] - radii[i] >= 0))
            constraints.append((1 - centers[i, 1] - radii[i] >= 0))
        for i, j in combinations(range(n)):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            constraints.append((dist - radii[i] - radii[j] >= 0))
        return constraints

    bounds = [(0, 1) for _ in range(n)]

    result = minimize(objective, radii, bounds=bounds, constraints=constraints)

    return (centers, result.x, np.sum(result.x))
