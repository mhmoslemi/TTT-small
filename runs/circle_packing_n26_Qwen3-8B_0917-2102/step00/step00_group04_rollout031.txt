import numpy as np
import matplotlib.pyplot as plt
import scipy.optimize as opt

def run_packing():
    # Hexagonal packing of 25 circles with radius r
    # 1 large circle at center with radius R
    # Total circles: 25 + 1 = 26

    # Parameters for hexagonal grid
    grid_size = 5
    grid_step = 1 / (grid_size + 1)
    hex_radius = 0.2

    # Create hexagonal grid of 25 circles
    centers_hex = []
    radii_hex = []

    for i in range(grid_size):
        for j in range(grid_size):
            x = i * grid_step + (j % 2) * grid_step / 2
            y = j * grid_step * np.sqrt(3) / 2
            centers_hex.append([x, y])
            radii_hex.append(hex_radius)

    # Add a large circle at the center
    center_x = 0.5
    center_y = 0.5
    R = 0.35  # Radius of central circle

    centers_hex.append([center_x, center_y])
    radii_hex.append(R)

    # Ensure all circles are within unit square
    for i in range(len(centers_hex)):
        x, y = centers_hex[i]
        r = radii_hex[i]
        if not (0 <= x - r and x + r <= 1 and 0 <= y - r and y + r <= 1):
            # Adjust radius to fit within square
            max_possible_r = min(x, 1 - x, y, 1 - y)
            radii_hex[i] = max_possible_r

    # Optimization to slightly adjust radii for better packing
    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii

    def constraints(radii):
        # Ensure non-overlapping
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dx = centers_hex[i][0] - centers_hex[j][0]
                dy = centers_hex[i][1] - centers_hex[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    # Initial guess
    initial_guess = radii_hex.copy()

    # Optimization
    result = opt.minimize(
        objective,
        initial_guess,
        constraints={"type": "constraint", "fun": constraints},
        method="SLSQP"
    )

    # Extract results
    centers = np.array(centers_hex)
    radii = np.array(result.x)
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
