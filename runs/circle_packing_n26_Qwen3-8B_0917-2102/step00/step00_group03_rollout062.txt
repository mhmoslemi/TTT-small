import numpy as np
from scipy.optimize import minimize

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        return np.sum(radii) - 2.636

    def bounds(radii):
        return [(0, 1) for _ in radii]

    def feasible(radii):
        return np.sum(radii) <= 2.636

    def initial_guess():
        return np.array([0.1] * 26)

    def refine_packing():
        centers = np.random.rand(26, 2)
        radii = np.full(26, 0.05)
        for i in range(26):
            x, y = centers[i]
            radii[i] = min(0.5, 0.5 * (1 - x) * (1 - y) * (x) * (y))
        return centers, radii

    centers, radii = refine_packing()
    result = minimize(objective, radii, method='SLSQP', constraints={'type': 'ineq', 'fun': constraints}, bounds=bounds(centers))
    return (centers, result.x, np.sum(result.x))
