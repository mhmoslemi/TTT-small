import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Hexagonal packing with 26 circles
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Layered hexagonal packing
    layer = 0
    col = 0
    row = 0
    for i in range(n):
        if i % 6 == 0:
            layer += 1
            col = 0
            row = layer
        else:
            col += 1
            row = layer - col // 2
        
        # Positioning circles in hexagonal grid
        x = col + 0.5 * (col % 2)
        y = row
        centers[i] = [x / 10, y / 10]
    
    # Initial radius based on spacing
    r0 = 0.02
    radii = np.full(n, r0)
    
    # Optimization function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)
    
    # Constraints: non-overlapping and within bounds
    def constraint(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0
    
    # Bounds for radii
    bounds = [(0, 1) for _ in range(n)]
    
    # Optimization
    result = minimize(objective, radii, bounds=bounds, constraints={"type": "ineq", "fun": constraint})
    
    # Final check and return
    centers = np.round(centers, 8)
    radii = np.round(result.x, 8)
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
