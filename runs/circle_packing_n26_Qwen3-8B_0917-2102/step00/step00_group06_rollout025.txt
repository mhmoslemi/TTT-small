from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.linalg import toeplitz

def run_packing():

    n = 26

    # Initial hexagonal grid with adjusted spacing
    cols = 6
    rows = 5
    spacing = 0.7
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        col = i % cols
        row = i // cols
        x = col * spacing + (0.5 if col % 3 == 0 else 0)
        y = row * (spacing * np.sqrt(3)/2)
        centers[i] = [x, y]
        radii[i] = 0.05

    def objective(radii):
        return -np.sum(radii)

    def constraint1(radii):
        return np.sum(radii) - 2.636

    def constraint2(radii, centers):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return np.array([1.0])
        return np.array([0.0])

    bounds = [(0, 1.0) for _ in range(n)]
    cons = (
        {'type': 'ineq', 'fun': lambda r: constraint1(r)},
        {'type': 'eq', 'fun': lambda r: -np.sum(r) + 2.636},
        {'type': 'ineq', 'fun': constraint2, 'args': (centers,)}
    )

    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)

    return (centers, result.x, np.sum(result.x))
