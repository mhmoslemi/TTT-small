import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial configuration: 5 rows of 5 circles, with 1 extra circle at bottom right
    centers = np.array([
        [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
        [0.1, 0.3], [0.3, 0.3], [0.5, 0.3], [0.7, 0.3], [0.9, 0.3],
        [0.1, 0.5], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5], [0.9, 0.5],
        [0.1, 0.7], [0.3, 0.7], [0.5, 0.7], [0.7, 0.7], [0.9, 0.7],
        [0.1, 0.9], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9], [0.9, 0.9]
    ])
    radii = np.full(26, 0.05)
    
    # Initial try with uniform radii
    def objective(x):
        radii[:] = x[:26]
        return -np.sum(radii)
    
    def constraints(x):
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < x[i] + x[j] - 1e-8:
                    return False
        for i in range(26):
            x_center, y_center = centers[i]
            if x_center - radii[i] < -1e-8 or x_center + radii[i] > 1 + 1e-8:
                return False
            if y_center - radii[i] < -1e-8 or y_center + radii[i] > 1 + 1e-8:
                return False
        return True
    
    result = minimize(objective, radii, method='SLSQP', constraints={'type': 'ineq', 'fun': constraints})
    
    return (centers, result.x, -result.fun)
