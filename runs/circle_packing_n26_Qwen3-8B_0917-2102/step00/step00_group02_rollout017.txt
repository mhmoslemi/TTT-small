<strategy>Use hexagonal packing with dense clusters near corners and edges to maximize total radius sum.</strategy>

import numpy as np

def run_packing():
    np.random.seed(42)
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n) * 0.1

    # Initial heuristic: place circles near corners and edges
    corner_centers = np.array([
        [0, 0], [1, 0], [1, 1], [0, 1]
    ])
    edge_centers = np.array([
        [0.25, 0.5], [0.75, 0.5], [0.5, 0.25], [0.5, 0.75]
    ])
    centers[:4] = corner_centers
    centers[4:8] = edge_centers

    # Set initial radii based on distance to boundaries
    radii[:4] = np.minimum(centers[:4, 0], 1 - centers[:4, 0])
    radii[4:8] = np.minimum(centers[4:8, 0], 1 - centers[4:8, 0])

    # Add remaining circles in grid-like fashion
    for i in range(8, n):
        x, y = centers[i]
        if x < 0.5:
            x = 0.5 - np.random.uniform(0, 0.5)
        else:
            x = 0.5 + np.random.uniform(0, 0.5)
        if y < 0.5:
            y = 0.5 - np.random.uniform(0, 0.5)
        else:
            y = 0.5 + np.random.uniform(0, 0.5)
        centers[i] = [x, y]
        radii[i] = 0.05

    # Optimize using greedy approach
    for _ in range(100):
        best = None
        best_dist = -1
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                if dist > radii[i] + radii[j]:
                    if best is None or dist > best_dist:
                        best = (i, j)
                        best_dist = dist
        if best is None:
            break
        i, j = best
        # Expand smaller circle to fill gap
        if radii[i] < radii[j]:
            radii[i] += (best_dist - radii[i] - radii[j]) / 2
        else:
            radii[j] += (best_dist - radii[i] - radii[j]) / 2

    # Final validation
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = max(0, min(r, 1 - max(x, 1 - x, y, 1 - y)))

    return centers, radii, np.sum(radii)
