import numpy as np
import scipy.optimize as opt

def run_packing():
    # Initial guess for centers in a hexagonal packing
    centers = []
    radii = []

    # Place 10 rows of circles in hexagonal packing
    for row in range(10):
        for col in range(13):
            x = col + row / 2
            y = row * np.sqrt(3) / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
    
    # Adjust centers to fit in unit square
    centers = np.array(centers)
    scale = 1.0 / max(np.max(centers, axis=0))
    centers *= scale

    # Initial radius guess
    initial_radius = 0.1
    radii = np.full(len(centers), initial_radius)

    # Optimization objective: maximize sum of radii
    def objective(x):
        return -np.sum(x)

    # Optimization constraints
    def constraint(x, i, j):
        return np.sqrt(np.sum((centers[i] - centers[j])**2)) - (x[i] + x[j])

    cons = []
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            cons.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: constraint(x, i, j)})

    # Bounds for radii
    bounds = [(0, 1) for _ in range(len(centers))]

    # Run optimization
    result = opt.minimize(objective, radii, bounds=bounds, constraints=cons)

    # Return result
    return (centers, result.x, np.sum(result.x))
