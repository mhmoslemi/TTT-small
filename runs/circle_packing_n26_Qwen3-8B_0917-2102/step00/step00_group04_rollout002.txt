import matplotlib.pyplot as plt
from scipy.optimize import minimize

def run_packing():

    corners = np.array([

        [1, 0],
        [1, 1],
        [0, 1]
    ])
    
    # Edge circles (avoiding corners)
    edge_centers = np.array([
        [0.2, 0.5],
        [0.8, 0.5],
        [0.5, 0.2],
        [0.5, 0.8]
    ])
    
    # Inner circles (hexagonal grid)
    inner_centers = []
    for i in range(5):
        for j in range(5 if i % 2 == 0 else 6):
            x = 0.2 + i * 0.25 + (j % 2) * 0.125
            y = 0.2 + (i % 2) * 0.25
            if 0.2 <= x <= 0.8 and 0.2 <= y <= 0.8:
                inner_centers.append([x, y])
    inner_centers = np.array(inner_centers)
    
    # Combine all centers
    centers = np.vstack([corners, edge_centers, inner_centers])
    
    # Initial radius guess (based on hexagonal packing)
    r = 0.12
    radii = np.full(centers.shape[0], r)
    
    # Optimization function to maximize sum of radii
    def objective(x):
        return -np.sum(x)  # Minimize negative sum to maximize sum
    
    # Optimization parameters
    bounds = [(0, 1) for _ in range(centers.shape[0])]
    
    # Optimization loop
    for _ in range(10):
        result = minimize(
            objective,
            radii,
            bounds=bounds,
            method='L-BFGS-B',
            jac=lambda x: -np.ones_like(x)
        )
        radii = result.x
    
    # Final check for boundary and non-overlap
    for i in range(centers.shape[0]):
        x, y = centers[i]
        if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
            radii[i] = min(radii[i], 1 - np.min([x, 1 - x, y, 1 - y]))
    
    for i in range(centers.shape[0]):
        for j in range(i + 1, centers.shape[0]):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-8:
                radii[i] = (dist - radii[j]) / 2
                radii[j] = (dist - radii[i]) / 2
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
