import numpy as np

def run_packing():
    # Hexagonal packing with 25 small circles and 1 large central circle
    # Small circles arranged in 5 rows with decreasing counts
    # Central circle at the origin

    centers = []
    radii = []

    # Central circle
    centers.append([0.5, 0.5])
    r_center = 0.2  # Max radius without overlapping small circles
    radii.append(r_center)

    # Small circles in hexagonal arrangement
    for row in range(5):
        for col in range(5 - row):
            x = 0.5 + (col - (4 - row) / 2) * (2 * r_center) * np.cos(np.pi / 3)
            y = 0.5 + (row - 2) * (2 * r_center) * np.sin(np.pi / 3)
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
                radii.append(r_center)

    # Adjust radii to fit in unit square
    max_r = 0.25
    radii = np.array(radii) * max_r / r_center
    centers = np.array(centers)

    # Ensure all circles are within the square
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            raise ValueError("Circle out of bounds")

    # Check for overlaps
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError("Circle overlap detected")

    return centers, radii, np.sum(radii)
