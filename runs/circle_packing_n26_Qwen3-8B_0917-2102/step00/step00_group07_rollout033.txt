from scipy.optimize import minimize
from scipy.spatial.distance import cdist

    radii = []

    col_count = 5

            x = j + 0.5 * (1 - 2 * i) / (row_count - 1)
            y = i / (row_count - 1)
            centers.append([x, y])

        return -np.sum(radii)  # Negative for minimization

    def constraint_func(radii):

        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):

            constraints.append(x - radii[i] >= -1e-8)
            constraints.append(x + radii[i] <= 1 + 1e-8)
            constraints.append(y - radii[i] >= -1e-8)
            constraints.append(y + radii[i] <= 1 + 1e-8)
        return constraints

    # Optimization
    bounds = [(0, 1) for _ in range(len(centers))]
    res = minimize(objective, radii, bounds=bounds, constraints=constraint_func)

    # Ensure non-negative radii
    radii = np.clip(res.x, 0, 1)
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
