import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist

def run_packing():
    np.random.seed(42)
    n = 26

    # Initial guess: hexagonal packing with boundary-aligned circles
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    row = 0
    col = 0
    for i in range(n):
        if i % 2 == 0:
            centers[i] = [col / (n - 1), row / (n - 1)]
        else:
            centers[i] = [(col + 0.5) / (n - 1), row / (n - 1)]
        col += 1
        if col >= n - 1:
            row += 1
            col = 0

    # Initial radii as uniform
    radii = np.full(n, 0.05)

    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum

    # Constraints: non-overlapping and within square
    def constraint(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8 or y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8:
                return -1
        return 0

    # Optimization
    res = minimize(objective, radii, method='SLSQP', constraints={'type': 'eq', 'fun': constraint})
    radii = res.x

    # Final check to ensure no negative radii
    radii[radii < 0] = 0

    # Compute sum of radii
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
