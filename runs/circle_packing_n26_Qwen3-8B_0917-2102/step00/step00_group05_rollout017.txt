import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Initial hexagonal packing with 26 circles
    n = 26
    centers = []
    radii = []
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5
        y = row + 0.5
        if row % 2 == 1:
            x += 0.5
        centers.append([x, y])
    centers = np.array(centers)
    radii = np.ones(n) * 0.05

    # Optimization to maximize sum of radii
    def objective(r):
        return -np.sum(r)

    def constraints(r):
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < r[i] + r[j] - 1e-12:
                    return False
        for i in range(n):
            x, y = centers[i]
            if x - r[i] < 0 or x + r[i] > 1 or y - r[i] < 0 or y + r[i] > 1:
                return False
        return True

    result = minimize(
        objective,
        radii,
        constraints={"type": "constraint", "fun": constraints},
        method="SLSQP"
    )

    return (centers, result.x, np.sum(result.x))
