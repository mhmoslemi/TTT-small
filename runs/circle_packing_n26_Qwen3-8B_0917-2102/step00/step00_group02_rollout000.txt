import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        return np.sum(radii)

    def bounds(radii):
        return [(0, 1) for _ in radii]

    def helper_centers():
        n = 26
        centers = []
        radius = 1 / (2 * np.sqrt(3))
        for i in range(n):
            if i % 2 == 0:
                x = i * (1 + radius) / n
                y = 0.5
                centers.append((x, y))
            else:
                x = (i - 0.5) * (1 + radius) / n
                y = 0.5
                centers.append((x, y))
        return np.array(centers)

    centers = helper_centers()
    n = len(centers)
    radii = np.ones(n) * 0.05

    def constraint_func(radii):
        constraints = []
        for i in range(n):
            r_i = radii[i]
            for j in range(i + 1, n):
                r_j = radii[j]
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                constraints.append(dist - r_i - r_j)
            if centers[i][0] - radii[i] < 0:
                constraints.append(centers[i][0] - radii[i])
            if centers[i][1] - radii[i] < 0:
                constraints.append(centers[i][1] - radii[i])
            if centers[i][0] + radii[i] > 1:
                constraints.append(1 - centers[i][0] - radii[i])
            if centers[i][1] + radii[i] > 1:
                constraints.append(1 - centers[i][1] - radii[i])
        return np.array(constraints)

    res = minimize(objective, radii, constraints=constraint_func, bounds=bounds)
    return (centers, res.x, np.sum(res.x))
