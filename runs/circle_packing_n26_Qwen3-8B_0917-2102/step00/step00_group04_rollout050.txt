def run_packing():

        return -np.sum(radii)

    def constraints(radii):
        return np.sum(radii)

    def boundary_constraint(radii, centers):
        for i in range(len(radii)):
            x, y = centers[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12 or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return np.inf
        return 0

    def overlap_constraint(radii, centers):
        for i in range(len(radii)):
            for j in range(i+1, len(radii)):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return np.inf
        return 0

    def initial_guess():
        n = 26
        x = np.linspace(0, 1, int(np.sqrt(n)) + 1)
        y = np.linspace(0, 1, int(np.sqrt(n)) + 1)
        grid = np.array(np.meshgrid(x, y)).T
        return grid[:n]

    centers = initial_guess()
    radii = np.full(26, 0.05)

    bounds = Bounds([0.0]*len(radii), [1.0]*len(radii))

    result = minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': boundary_constraint, 'args': (centers,)},
            {'type': 'ineq', 'fun': overlap_constraint, 'args': (radii, centers)}
        ]
    )

    return (centers, result.x, np.sum(result.x))
