import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():

        return -np.sum(radii)

    def constraints(radii):
        return np.sum(radii)

    def bounds(radii):
        return np.zeros_like(radii)

    def circle_positions(n):
        if n == 0:
            return np.empty((0, 2)), np.empty(0)
        cols = int(np.floor(np.sqrt(n)))
        rows = int(np.ceil(n / cols))
        centers = []
        for i in range(rows):
            for j in range(cols):
                if i * cols + j < n:
                    x = j / (cols - 1) if cols > 1 else 0.5
                    y = i / (rows - 1) if rows > 1 else 0.5
                    centers.append([x, y])
        return np.array(centers), np.zeros(len(centers))

    centers, _ = circle_positions(26)
    radii = np.full(26, 0.01)

    def packing_objective(x):
        radii = x
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -np.inf
        for i in range(26):
            if centers[i, 0] - radii[i] < 0 or centers[i, 0] + radii[i] > 1:
                return -np.inf
            if centers[i, 1] - radii[i] < 0 or centers[i, 1] + radii[i] > 1:
                return -np.inf
        return -np.sum(radii)

    result = minimize(lambda x: packing_objective(x), radii, method='L-BFGS-B', bounds=[(0, 1)] * 26, options={'maxiter': 1000})

    best_sum = np.sum(best_radii)
    return centers, best_radii, best_sum
