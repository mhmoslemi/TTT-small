import numpy as np

def run_packing():
    # Define a hexagonal packing with 26 circles
    # First, determine the number of rows
    # Using formula for hexagonal packing in a square: n = m*(m+1)/2
    # Solving m*(m+1)/2 ≈ 26 → m ≈ 7 (7*8/2=28), so use 7 rows with 7,6,5,4,3,2,1 circles
    
    # Generate centers for hexagonal packing
    centers = []
    radii = []
    
    # Parameters
    spacing = 0.5  # spacing between circle centers
    margin = 0.1  # margin from the boundary to place circles
    
    # Place circles in rows
    for row in range(7):
        # Number of circles in this row
        num_circles = 7 - row
        # Horizontal shift for even/odd rows
        shift = (row % 2) * spacing / 2
        # Vertical position
        y = row * spacing + margin
        # Horizontal positions
        for col in range(num_circles):
            x = col * spacing + shift + margin
            centers.append([x, y])
            # Radius is determined by spacing
            # To fit in square of size 1, we compute radius as spacing / 2
            radii.append(spacing / 2)
    
    # Additional circles in corners and edges if needed
    # Add a circle at top-left corner
    centers.append([margin, 1 - margin])
    radii.append(0.1)
    # Add a circle at top-right corner
    centers.append([1 - margin, 1 - margin])
    radii.append(0.1)
    # Add a circle at bottom-left corner
    centers.append([margin, margin])
    radii.append(0.1)
    # Add a circle at bottom-right corner
    centers.append([1 - margin, margin])
    radii.append(0.1)
    
    # Convert lists to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)
    
    # Compute sum of radii
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
