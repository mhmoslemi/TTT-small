import numpy as np

def run_packing():
    # Hexagonal packing with 5 rows and 5 columns, plus 1 extra in last row
    centers = []
    radii = []
    row_height = np.sqrt(3)/2  # Height of equilateral triangle
    spacing = 0.2  # Spacing between circles

    # First 5 rows with 5 circles each
    for i in range(5):
        for j in range(5):
            x = j * (1 + spacing)
            y = i * row_height
            if i % 2 == 0:
                x = j * (1 + spacing) + spacing / 2
            centers.append([x, y])
            radii.append(0.05)

    # Extra row with 6 circles to reach 26
    for j in range(6):
        x = j * (1 + spacing)
        y = 5 * row_height + spacing
        if 5 % 2 == 0:
            x = j * (1 + spacing) + spacing / 2
        centers.append([x, y])
        radii.append(0.05)

    # Adjust radii to fit in unit square
    max_r = 0.1
    for i in range(len(radii)):
        x, y = centers[i]
        if x - max_r < 0 or x + max_r > 1 or y - max_r < 0 or y + max_r > 1:
            radii[i] = max(0, min(max_r, (1 - max(0, x)) * 2, (1 - max(0, y)) * 2))

    # Ensure circles don't overlap and adjust for boundary
    for i in range(len(radii)):
        for j in range(i + 1, len(radii)):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < radii[i] + radii[j] - 1e-8:
                radii[i] = (dist - radii[j] + 1e-8) / 2
                radii[j] = (dist - radii[i] + 1e-8) / 2

    # Final boundary clipping
    for i in range(len(radii)):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            radii[i] = max(0, x)
        if x + r > 1:
            radii[i] = max(0, 1 - x)
        if y - r < 0:
            radii[i] = max(0, y)
        if y + r > 1:
            radii[i] = max(0, 1 - y)

    return np.array(centers), np.array(radii), np.sum(radii)
