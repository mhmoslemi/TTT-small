import numpy as np
from scipy.optimize import minimize

def run_packing():
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize total radius

    def constraints(centers, radii):
        # Boundary constraints
        for i in range(len(radii)):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8 or y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8:
                return False
        # Pairwise distance constraints
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    # Initial hexagonal packing with 26 circles
    n = 26
    rows = int(np.ceil(n / 2))
    cols = int(np.ceil(n / rows))
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        row = i // cols
        col = i % cols
        centers[i, 0] = 0.5 + (col - (cols - 1) / 2) * (1 / (cols - 1))
        centers[i, 1] = 0.5 + (row - (rows - 1) / 2) * (1 / (rows - 1))
        radii[i] = 0.05

    # Optimization
    bounds = [(0, 0.1) for _ in range(n)]
    res = minimize(objective, radii, bounds=bounds, constraints=constraints, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 1000})
    return (centers, res['x'], np.sum(res['x']))
