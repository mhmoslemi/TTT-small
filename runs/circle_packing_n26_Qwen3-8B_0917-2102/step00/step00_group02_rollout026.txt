    radii = np.array(radii)

    # Optimization function to maximize sum of radii
    def objective(r):
        return -np.sum(r)

    def constraint_func(r):
        # Ensure no overlap
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
