from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        return np.sum(radii)

    def non_overlap(centers, radii):
        n = len(centers)
        constraints = []
        for i in range(n):
            for j in range(i+1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                constraints.append(dist - (radii[i] + radii[j]) >= 1e-8)
        return constraints

    def boundary(centers, radii):
        n = len(centers)
        constraints = []
        for i in range(n):
            x, y = centers[i]
            constraints.append(x - radii[i] >= 1e-8)
            constraints.append(x + radii[i] <= 1 - 1e-8)
            constraints.append(y - radii[i] >= 1e-8)
            constraints.append(y + radii[i] <= 1 - 1e8)
        return constraints

    def pack():
        n = 26
        centers = np.random.rand(n, 2)
        radii = np.random.rand(n) * 0.1

        def get_constraints(radii):
            return non_overlap(centers, radii) + boundary(centers, radii)

        bounds = [(0, 1) for _ in range(2*n)]
        res = minimize(objective, np.concatenate((centers.reshape(-1), radii)), bounds=bounds, constraints=get_constraints)
        return res.x[:2*n].reshape(n, 2), res.x[2*n:], np.sum(res.x[2*n:])

    centers, radii, sum_rad = pack()
    return centers, radii, sum_rad
