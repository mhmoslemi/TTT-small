import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance

def run_packing():
    def objective(r):
        return -np.sum(r)  # Maximize sum of radii

    def constraints(r, centers):
        n = len(r)
        cons = []
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - 1e-12})
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j, dist=dist: dist - r[i] - r[j] - 1e-12})
        return cons

    # Define initial hexagonal packing configuration
    rows = 5
    cols = 6
    centers = np.zeros((rows * cols, 2))
    for i in range(rows):
        for j in range(cols):
            x = j / (cols - 1) if cols > 1 else 0.5
            y = i / (rows - 1) if rows > 1 else 0.5
            centers[i * cols + j] = [x, y]

    # Optimize radii
    r0 = np.full(len(centers), 0.05)
    result = minimize(
        objective,
        r0,
        constraints=constraints(r0, centers),
        method='SLSQP',
        bounds=[(0, 1)] * len(r0)
    )

    radii = result.x
    centers = np.array(centers)
    return centers, radii, -result.fun
