    cols = 5

    radii = []

        y = 1 - 1e-3

        radii.append(0.01)

            x = i / (cols - 1)
            y = 1 - (r + 0.5) / (rows + 1)
            centers.append([x, y])
            radii.append(0.01)

    centers = np.array(centers)
    radii = np.array(radii)

    # Optimization parameters
    bounds = [(0, 1) for _ in range(2 * len(centers))]
    constraints = []

    # Define objective function
    def objective(x):
        centers_flat = x[:2 * len(centers)].reshape((len(centers), 2))
        radii_flat = x[2 * len(centers):]
        return -np.sum(radii_flat)

    # Define constraints for boundaries
    def boundary(x):
        centers_flat = x[:2 * len(centers)].reshape((len(centers), 2))
        radii_flat = x[2 * len(centers):]
        for i in range(len(centers_flat)):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12:
                return 1
            if y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return 1
        return 0

        centers_flat = x[:2 * len(centers)].reshape((len(centers), 2))

                dx = centers_flat[i, 0] - centers_flat[j, 0]

                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:

        return 0

    def constraints_func(x):
        return boundary(x) + circle_dist(x)
