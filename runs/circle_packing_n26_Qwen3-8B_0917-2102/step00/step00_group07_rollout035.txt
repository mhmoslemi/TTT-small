import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial guess for centers in hexagonal packing
    def hexagonal_centers(n):
        centers = []
        for i in range(n):
            row = i // 5
            col = i % 5
            x = col + 0.5 * (row % 2)
            y = row * np.sqrt(3) / 2
            centers.append([x, y])
        return np.array(centers)

    centers = hexagonal_centers(26)
    radii = np.full(26, 0.05)

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        # Boundary constraints
        for i in range(26):
            r = radii[i]
            if (centers[i, 0] - r < 0 or centers[i, 0] + r > 1 or
                centers[i, 1] - r < 0 or centers[i, 1] + r > 1):
                return False
        # Circle-circle constraints
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    bounds = [(0, 0.2) for _ in range(26)]

    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP'
    )

    return (centers, result.x, np.sum(result.x))
