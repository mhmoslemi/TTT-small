import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    np.random.seed(42)
    n = 26

    # Initial hexagonal packing with 5 rows
    rows = 5
    cols = 5
    circles = []
    for i in range(rows):
        for j in range(cols):
            x = j + 0.5
            y = i + 0.5
            circles.append((x, y))
    centers = np.array(circles)
    radii = np.full(n, 0.1)

    # Initial guess for radii
    r_guess = np.full(n, 0.1)

    def objective(r):
        return -np.sum(r)

    def constraints(r):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - 1e-8:
                    return False
        for i in range(n):
            x, y = centers[i]
            if x - r[i] < 0 or x + r[i] > 1 or y - r[i] < 0 or y + r[i] > 1:
                return False
        return True

    res = minimize(objective, r_guess, constraints=lambda r: constraints(r), method='SLSQP')

    radii = res.x
    centers = np.array([[j + 0.5, i + 0.5] for i in range(rows) for j in range(cols)])
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
