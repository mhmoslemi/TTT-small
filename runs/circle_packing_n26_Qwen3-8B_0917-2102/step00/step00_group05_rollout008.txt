import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal packing with adjusted spacing
    angles = np.linspace(0, 2 * np.pi, n, endpoint=False)
    radii_initial = np.array([0.05] * n)
    centers = np.array([(np.cos(a), np.sin(a)) for a in angles]) * 0.5 + 0.5

    def objective(r):
        return -np.sum(r)

    def constraint1(r):
        return np.sum(r)

    def constraint2(r, centers):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] + 1e-8:
                    return False
        return True

    bounds = [(0, 1.0) for _ in range(n)]

    result = minimize(
        objective,
        radii_initial,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint2, 'args': (centers,)}
        ]
    )

    if not result.success:
        raise RuntimeError("Optimization failed")

    radii = result.x
    centers = np.array([(np.cos(a), np.sin(a)) for a in angles]) * 0.5 + 0.5
    # Perturb centers to fit within square and avoid boundary overlaps
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        while x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            angle = np.random.uniform(0, 2 * np.pi)
            centers[i] = (np.cos(angle), np.sin(angle)) * (1 - r) + r

    # Final validation step
    def validate_packing(centers, radii):
        n = centers.shape[0]
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"
        return True, "ok"

    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(msg)

    return centers, radii, np.sum(radii)
