    def non_overlap(centers, radii):
        n = len(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < radii[i] + radii[j] - 1e-6:
                    return False
        return True

    def boundary(centers, radii):
        for i in range(len(centers)):
            x, y = centers[i]
            if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
                return False
        return True

    def hexagonal_grid(n):
        rows = int(np.ceil(n / 6))
        cols = int(np.ceil(n / rows))
        centers = []
        for i in range(rows):
            for j in range(cols):
                x = j + 0.5 * i
                y = i * np.sqrt(3) / 2
                if x < 1 and y < 1:
                    centers.append([x, y])
        return np.array(centers)
