from scipy.spatial.distance import pdist, squareform

def run_packing():

    rows = 5
    cols = 5
    centers = []
    radii = []

            x = j + 0.5 * (1 - 2 * (i % 2))

            centers.append([x, y])

    radii = np.full(25, 0.1)

    # Function to compute distance matrix
    def dist_matrix(c):
        return squareform(pdist(c))
    
    # Objective function to maximize sum of radii
    def obj(r):
        nonlocal centers
        dist = dist_matrix(centers)
        for i in range(len(r)):
            for j in range(i + 1, len(r)):
                if dist[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return -np.sum(r)
    
    # Optimization
    bounds = [(0, 0.5) for _ in range(25)]
    result = minimize(lambda r: -obj(r), np.full(25, 0.1), bounds=bounds, method='L-BFGS-B')
    radii = result.x
    sum_radii = np.sum(radii)
    
    # Ensure all circles are inside the square
    for i in range(25):
        x, y = centers[i]

        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            radii[i] = min(r, 1 - np.min([x, 1 - x, y, 1 - y]))

                min_rad = (radii[i] + radii[j] - dist) / 2

    return centers, radii, sum_radii
