import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import matplotlib.pyplot as plt

def run_packing():
    np.random.seed(42)
    
    # Initial hexagonal grid with boundary circles
    def hex_grid(n):
        r = np.sqrt(1 / (n * np.sqrt(3)))
        rows = int(np.ceil(n / 6))
        cols = int(np.ceil(n / rows))
        centers = []
        for i in range(rows):
            for j in range(cols):
                if i % 2 == 0:
                    x = j * (2 * r * np.sqrt(3))
                    y = i * (2 * r)
                else:
                    x = j * (2 * r * np.sqrt(3)) + r * np.sqrt(3)
                    y = i * (2 * r)
                if 0 <= x <= 1 and 0 <= y <= 1:
                    centers.append([x, y])
        return np.array(centers)
    
    centers = hex_grid(26)
    radii = np.full(centers.shape[0], 0.05)
    
    # Optimization function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)
    
    # Constraints: non-overlapping and within unit square
    def constraint(radii):
        dists = squareform(pdist(centers))
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return -1
        for i in range(len(radii)):
            if centers[i, 0] - radii[i] < -1e-8 or centers[i, 0] + radii[i] > 1 + 1e-8:
                return -1
            if centers[i, 1] - radii[i] < -1e-8 or centers[i, 1] + radii[i] > 1 + 1e-8:
                return -1
        return 0
    
    # Optimization with bounds and constraints
    bounds = [(0, 1) for _ in range(len(radii))]
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraint},
        options={'ftol': 1e-10, 'maxiter': 1000}
    )
    
    centers = np.array(centers)
    radii = result.x
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
