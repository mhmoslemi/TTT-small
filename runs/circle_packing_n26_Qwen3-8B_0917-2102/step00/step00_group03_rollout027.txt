import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial guess for centers in hexagonal packing
    def hex_packing(n):
        centers = []
        for i in range(n):
            row = i // 5
            col = i % 5
            x = col + 0.5 * (i % 2)
            y = row + 0.5
            centers.append([x, y])
        return np.array(centers)

    centers = hex_packing(26)
    radii = np.full(26, 0.1)

    # Define objective function to maximize sum of radii
    def objective(r):
        return -np.sum(r)

    # Define constraints for boundary and non-overlapping
    def constraints(r):
        constraints = []
        # Boundary constraint: x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1
        def boundary_constraint(r_idx):
            def cons(x):
                r = x[r_idx]
                return np.array([centers[i, 0] - r, centers[i, 1] - r, 1 - centers[i, 0] - r, 1 - centers[i, 1] - r])
            return cons
        for i in range(26):
            constraints.append(boundary_constraint(i))
        # Non-overlapping constraint: distance between centers >= r_i + r_j
        def overlap_constraint(i, j):
            def cons(x):
                r_i = x[i]
                r_j = x[j]
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                return np.array([dist - r_i - r_j])
            return cons
        for i in range(26):
            for j in range(i+1, 26):
                constraints.append(overlap_constraint(i, j))
        return constraints

    # Optimization
    bounds = [(0, 1) for _ in range(26)]
    res = minimize(objective, radii, bounds=bounds, constraints=constraints(radii))
    radii = res.x
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
