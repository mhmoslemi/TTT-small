import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, cdist

def run_packing():
    np.random.seed(42)
    n = 26

    # Initial hexagonal packing with 5 rows (max 5 for unit square)
    centers = []
    radii = []

    # Bottom row
    for i in range(5):
        x = i / 4.0
        y = 0.1
        centers.append([x, y])
        radii.append(0.1)

    # Middle rows
    for row in range(1, 4):
        for i in range(5 - row):
            x = i + row / 4.0
            y = row * (np.sqrt(3)/2) * 0.1 + 0.1
            centers.append([x, y])
            radii.append(0.1)

    # Top row
    for i in range(5):
        x = i / 4.0
        y = 1 - 0.1
        centers.append([x, y])
        radii.append(0.1)

    centers = np.array(centers)
    radii = np.array(radii)

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        constraints = []
        for i in range(n):
            constraints.append((radii[i], 0, None, 'ineq'))
        for i in range(n):
            for j in range(i + 1, n):
                def dist_con(radii):
                    return np.sqrt(np.sum((centers[i] - centers[j]) ** 2)) - (radii[i] + radii[j])
                constraints.append((dist_con(radii), 0, None, 'ineq'))
        return constraints

    result = minimize(objective, radii, method='SLSQP', constraints=constraints)

    return (centers, result.x, np.sum(result.x))
