import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    bounds = [(0, 1) for _ in range(2 * n)]
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraints(centers, radii):
        constraints = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            constraints.append((x - r >= -1e-12) & (x + r <= 1 + 1e-12) & (y - r >= -1e-12) & (y + r <= 1 + 1e-12))
        
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                constraints.append(dist >= radii[i] + radii[j] - 1e-12)
        
        return constraints
    
    initial_centers = np.random.rand(n, 2)
    initial_radii = np.random.rand(n) * 0.1
    
    result = minimize(
        fun=objective,
        x0=np.concatenate((initial_centers.flatten(), initial_radii)),
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': lambda x: np.sum((x[:2*n] - np.roll(x[:2*n], 1))**2) - (x[2*n:]**2 + x[2*n+1:])**2}
        ]
    )
    
    centers = result.x[:2*n].reshape(n, 2)
    radii = result.x[2*n:]
    
    return centers, radii, np.sum(radii)
