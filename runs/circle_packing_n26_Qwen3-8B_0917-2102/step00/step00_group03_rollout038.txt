import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n) * 0.1

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        constraints = []
        for i in range(n):
            constraints.append((radii[i] <= 1 - centers[i, 0] - 1e-6))
            constraints.append((radii[i] <= 1 - centers[i, 1] - 1e-6))
            constraints.append((radii[i] <= centers[i, 0] + 1e-6))
            constraints.append((radii[i] <= centers[i, 1] + 1e-6))
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                constraints.append((radii[i] + radii[j] <= dist + 1e-6))
        return constraints

    bounds = [(0, 1) for _ in range(n)]
    res = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=constraints)

    if not res.success:
        raise Exception("Optimization failed")

    return centers, res.x, np.sum(res.x)
