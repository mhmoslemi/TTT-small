import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import LinearConstraint, Bounds

def run_packing():

    n = 26
    max_iter = 100
    tol = 1e-5

    def objective(radii):
        return -np.sum(radii)

    def gradient(radii):
        return -np.ones_like(radii)

    def constraints(radii):
        return []

    def non_negative(radii):
        return np.all(radii >= 0)

    def boundary(x, y, r):
        return (x - r >= 0) & (x + r <= 1) & (y - r >= 0) & (y + r <= 1)

    def dist_matrix(centers):
        return squareform(pdist(centers))

    def inequalities(centers, radii):
        return [(dist_matrix(centers)[i, j] >= radii[i] + radii[j] - 1e-8) for i in range(n) for j in range(i + 1, n)]

    def initial_centers():
        grid = np.linspace(0, 1, 6)
        centers = []
        for i in range(6):
            for j in range(6):
                if i % 2 == 0:
                    x = grid[i]
                    y = grid[j] + (grid[1] - grid[0]) * (0.5 if i < 5 else 0)
                else:
                    x = grid[i] + (grid[1] - grid[0]) * 0.5
                    y = grid[j]
                centers.append([x, y])
        return np.array(centers[:n])

    centers = initial_centers()
    radii = np.full(n, 0.05)

    for _ in range(max_iter):

        constraints = []

            constraints.append(centers[i, 0] - radii[i] >= 0)
            constraints.append(centers[i, 0] + radii[i] <= 1)
            constraints.append(centers[i, 1] - radii[i] >= 0)
            constraints.append(centers[i, 1] + radii[i] <= 1)

        res = minimize(
            objective,
            radii,
            method='SLSQP',
            jac=gradient,
            constraints=constraints,
            bounds=[(0, 1)] * n,
            options={'maxiter': 100, 'ftol': 1e-8}
        )

            break

        radii = res.x
        centers = np.array([centers[i] for i in range(n) if radii[i] > 1e-8])

                            centers[i] = [centers[i, 0] + np.random.uniform(-0.1, 0.1), centers[i, 1] + np.random.uniform(-0.1, 0.1)]
                            break

        if np.sum(radii) > 2.636:
            break

    return centers, radii, np.sum(radii)
