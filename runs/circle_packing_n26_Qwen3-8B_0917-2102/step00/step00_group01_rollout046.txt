import numpy as np
from scipy.optimize import minimize

def run_packing():
    np.random.seed(42)
    n = 26

    # Initial hexagonal grid with 5 rows
    rows = 5
    cols = 5
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j + 0.5 * (i % 2)
            y = i + 0.5
            centers.append([x, y])

    centers = np.array(centers)
    radii = np.ones(n) * 0.1

    def objective(r):
        return -np.sum(r)

    def constraints(r):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - 1e-8:
                    return False
        for i in range(n):
            x, y = centers[i]
            if x - r[i] < -1e-8 or x + r[i] > 1 + 1e-8 or y - r[i] < -1e-8 or y + r[i] > 1 + 1e-8:
                return False
        return True

    bounds = [(0, 1) for _ in range(n)]

    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints={"type": "success", "fun": constraints},
        method="SLSQP"
    )

    if not result.success:
        raise RuntimeError("Optimization failed")

    return centers, result.x, np.sum(result.x)
