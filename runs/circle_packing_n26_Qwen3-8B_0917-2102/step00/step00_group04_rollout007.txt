import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Hexagonal packing with 26 circles
    # Approximate coordinates for hexagonal grid
    centers = []
    radii = []
    for i in range(26):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        # Scale to fit in [0,1]^2
        x = x / (np.sqrt(3) / 2 * 4)
        y = y / (np.sqrt(3) / 2 * 4)
        centers.append([x, y])
        radii.append(0.04)
    
    # Optimize radii with boundary constraints
    def objective(r):
        return -sum(r)
    
    def constraints(r):
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - 1e-8:
                    return False
            if centers[i][0] - r[i] < 0 or centers[i][0] + r[i] > 1 or centers[i][1] - r[i] < 0 or centers[i][1] + r[i] > 1:
                return False
        return True
    
    r0 = np.full(26, 0.04)
    result = minimize(objective, r0, constraints=lambda x: constraints(x), method='SLSQP')
    
    centers = np.array(centers)
    radii = result.x
    sum_radii = sum(radii)
    
    return centers, radii, sum_radii
