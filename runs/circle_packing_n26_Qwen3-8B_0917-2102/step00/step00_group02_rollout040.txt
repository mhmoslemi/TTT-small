import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def run_packing():
    n = 26

    # Initial hexagonal packing with spacing
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        row = i // 5
        col = i % 5
        x = 0.1 + col * 0.2
        y = 0.1 + row * 0.2
        if row % 2 == 1:
            x += 0.1
        centers[i] = [x, y]

    # Initial guess for radii
    radii = np.full(n, 0.05)

    # Objective function
    def objective(radii):
        return -np.sum(radii)

    # Constraints
    constraints = []
    for i in range(n):
        # Boundary constraint
        def boundary_constraint(r):
            return [centers[i, 0] - r[i], centers[i, 1] - r[i], 1 - centers[i, 0] - r[i], 1 - centers[i, 1] - r[i]]
        constraints.append(LinearConstraint(np.eye(n), lb=np.array([-1]*n), ub=np.array([0]*n)))

    # Pairwise distance constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            def dist_constraint(r):
                return [r[i] + r[j] - dist]
            constraints.append(LinearConstraint(np.zeros(n), lb=[-np.inf], ub=[dist]))

    # Optimization bounds
    bounds = Bounds(np.zeros(n), np.full(n, 1))

    # Run optimization
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=constraints)

    centers = np.array(centers)
    radii = result.x
    sum_radii = np.sum(radii)

    return (centers, radii, sum_radii)
