import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    n = 26

    # Initial guess: hexagonal packing with 5 rows
    centers = np.zeros((n, 2))
    x = np.linspace(0, 1, 5)
    y = np.linspace(0, 1, 5)
    for i in range(5):
        for j in range(5):
            centers[5*i + j] = [x[j], y[i]]
    centers = centers[:n]

    # Initial radii guess: uniform
    radii = np.full(n, 0.1)

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        constraints = []
        for i in range(n):
            constraints.append(radii[i] >= 0)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                constraints.append(dist >= radii[i] + radii[j] - 1e-8)
        return constraints

    def bounds():
        return [(0, None) for _ in range(n)]

    res = minimize(objective, radii, bounds=bounds(), constraints=constraints)

    if not res.success:
        raise Exception("Optimization failed")

    radii = res.x
    sum_radii = np.sum(radii)

    # Check boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            raise Exception("Boundary constraint violated")

    return centers, radii, sum_radii
