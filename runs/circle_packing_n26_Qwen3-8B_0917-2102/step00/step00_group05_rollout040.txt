import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    np.random.seed(42)
    n = 26

    # Initial guess: hexagonal packing with 5 rows
    centers = []
    radii = []
    for i in range(5):
        for j in range(5 - abs(i - 2)):
            x = j + 0.5 * (i % 2)
            y = i * np.sqrt(3) / 2
            centers.append([x, y])
    centers = np.array(centers)
    centers = (centers - centers.mean(axis=0)) / np.max(np.abs(centers))

    # Adjust to fit in unit square
    max_dist = np.max(np.sqrt(np.sum((centers[1:] - centers[:-1])**2, axis=1)))
    radii = np.full(n, 1.0 / (4 * max_dist))

    # Function to compute total radius
    def total_radius(radii):
        return np.sum(radii)

    # Optimization bounds
    bounds = Bounds(0, 1.0)

    # Objective function
    def objective(x):
        return -total_radius(x)

    # Constraints: non-overlapping and within square
    def constraints(x):
        r = x
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < r[i] + r[j] - 1e-12:
                    return -np.inf
        for i in range(n):
            if centers[i, 0] - r[i] < -1e-12 or centers[i, 0] + r[i] > 1 + 1e-12 or centers[i, 1] - r[i] < -1e-12 or centers[i, 1] + r[i] > 1 + 1e-12:
                return -np.inf
        return 0

    # Run optimization
    result = minimize(objective, radii, bounds=bounds, constraints={"type": "ineq", "fun": constraints})

    return (centers, result.x, total_radius(result.x))
