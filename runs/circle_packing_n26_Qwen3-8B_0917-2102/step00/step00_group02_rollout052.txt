import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds as ScipyBounds
import itertools

def run_packing():
    n = 26

    # Initial guess with staggered hexagonal packing
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    row_count = 5
    col_count = 5
    max_radius = 0.15

    for i in range(n):
        row = i // col_count
        col = i % col_count
        x = col * (1.0 / (col_count - 1)) + 0.5 / (col_count - 1)
        y = row * (1.0 / (row_count - 1)) + 0.5 / (row_count - 1)
        centers[i, 0] = x
        centers[i, 1] = y
        radii[i] = max_radius

    def objective(x):
        return -np.sum(x)

    def constraints(x):
        centers_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:]
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return False
        for i in range(n):
            for j in range(i+1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-8:
                    return False
        return True

    bounds = []
    for i in range(n):
        bounds.append((0, 1))
        bounds.append((0, 1))
        bounds.append((0, 1))  # Radius

    bounds = ScipyBounds(bounds)

    x0 = np.array([centers.flatten(), radii]).flatten()

    result = minimize(objective, x0, bounds=bounds, constraints={"type": "ineq", "fun": constraints}, method='SLSQP', options={"ftol": 1e-10, "maxiter": 1000})

    centers = result.x[:n*2].reshape(n, 2)
    radii = result.x[n*2:]
    sum_radii = np.sum(radii)

    return (centers, radii, sum_radii)
