import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import time

def run_packing():
    np.random.seed(42)
    
    # Initial guess: grid-based hexagonal packing with 5 rows
    centers = []
    radii = []
    for row in range(5):
        for col in range(5 + (row % 2)):
            x = col + (row % 2) / 2
            y = row * np.sqrt(3)/2
            centers.append([x, y])
    
    # Normalize to unit square
    centers = np.array(centers)
    centers /= np.max(centers, axis=0)
    
    # Initial radii based on min distance between circles
    dists = squareform(pdist(centers))
    min_dist = np.min(dists)
    radii = np.full(len(centers), min_dist / 2)
    
    # Optimization objective
    def objective(r):
        return -np.sum(r)
    
    # Constraints: no overlap, boundary
    def constraint_overlap(x):
        r = x[1:]
        c = x[:-1]
        dists = squareform(pdist(c))
        return np.min(dists - r[:-1][:, np.newaxis] - r[np.newaxis, 1:]) - 1e-8
    
    def constraint_boundary(x):
        r = x[1:]
        c = x[:-1]
        return np.min(c - r[:, np.newaxis]) - 1e-8
    
    # Initial guess (radii + centers)
    x0 = np.concatenate((centers.flatten(), radii))
    
    # Optimization
    cons = [{'type': 'ineq', 'fun': constraint_overlap},
            {'type': 'ineq', 'fun': constraint_boundary}]
    
    result = minimize(objective, x0, constraints=cons, method='SLSQP')
    
    # Extract final centers and radii
    centers = result.x[:-len(radii)].reshape(-1, 2)
    radii = result.x[-len(radii):]
    
    # Further refine by adjusting radii to fit boundaries
    boundary_distances = np.min(np.abs(centers - np.array([[0, 0], [1, 0], [0, 1], [1, 1]])), axis=1)
    radii = np.minimum(radii, boundary_distances - 1e-8)
    
    # Final validation
    def validate_packing(centers, radii):
        n = centers.shape[0]
        
        if np.isnan(centers).any() or np.isnan(radii).any():
            return False, "NaN values present"
        
        for i in range(n):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"
        
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
        
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"
        
        return True, "ok"
    
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(msg)
    
    return centers, radii, np.sum(radii)
