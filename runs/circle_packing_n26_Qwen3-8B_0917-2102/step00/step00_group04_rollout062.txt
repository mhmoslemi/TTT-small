import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import LinearConstraint, Bounds

def run_packing():
    np.random.seed(42)
    n = 26
    
    # Initial hexagonal packing with boundary circles
    def hexagonal_grid(m, n):
        centers = []
        for i in range(n):
            for j in range(m):
                x = j + 0.5
                y = i + 0.5
                centers.append([x, y])
        return np.array(centers)
    
    # Create a hexagonal grid with 5 rows and 5 columns
    centers = hexagonal_grid(5, 5)
    radii = np.full(n, 0.1)
    
    # Define objective function to maximize total radius
    def objective(radii):
        return -np.sum(radii)
    
    # Define constraints
    def constraint1(radii):
        return np.sum(radii)
    
    def constraint2(radii, centers):
        dists = squareform(pdist(centers))
        for i in range(len(radii)):
            for j in range(i+1, len(radii)):
                if dists[i,j] < radii[i] + radii[j] - 1e-12:
                    return False
        return True
    
    # Optimize
    cons = ({'type': 'ineq', 'fun': lambda r: constraint2(r, centers)})
    bounds = Bounds(0, 1)
    
    result = minimize(objective, radii, bounds=bounds, constraints=cons)
    
    # Adjust circles to fit within unit square
    for i in range(len(centers)):
        x, y = centers[i]
        r = result.x[i]
        if x - r < 0:
            result.x[i] = x - r
        if x + r > 1:
            result.x[i] = 1 - r
        if y - r < 0:
            result.x[i] = y - r
        if y + r > 1:
            result.x[i] = 1 - r
    
    # Final check
    for i in range(len(centers)):
        x, y = centers[i]
        r = result.x[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:

    return centers, result.x, np.sum(result.x)
