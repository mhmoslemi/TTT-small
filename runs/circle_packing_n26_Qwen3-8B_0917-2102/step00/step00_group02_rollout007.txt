import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial hexagonal packing with 5 rows
    n = 26
    centers = []
    radii = []

    # Generate hexagonal grid
    for row in range(5):
        for col in range(5 + (row % 2)):
            x = col + (row % 2) * 0.5
            y = row * np.sqrt(3)/2
            x = np.clip(x, 0, 1)
            y = np.clip(y, 0, 1)
            centers.append([x, y])
    
    # Adjust for tight packing
    for i in range(n):
        if i < len(centers):
            # Initial radii based on distance to nearest neighbors
            x, y = centers[i]
            r = 0.05
            if i < len(centers) - 1:
                dx = centers[i+1][0] - x
                dy = centers[i+1][1] - y
                r = np.sqrt(dx**2 + dy**2) / 3
            if i > 0:
                dx = centers[i-1][0] - x
                dy = centers[i-1][1] - y
                r = np.sqrt(dx**2 + dy**2) / 3
            radii.append(r)
    
    # Refine with optimization
    def objective(radii_vec):
        return -np.sum(radii_vec)
    
    def constraints(radii_vec):
        res = []
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                res.append(dist - (radii_vec[i] + radii_vec[j]) >= 1e-8)
        return res
    
    # Convert centers to numpy array
    centers = np.array(centers)
    
    # Optimization
    bounds = [(0, 0.2) for _ in range(n)]
    cons = [{'type': 'ineq', 'fun': lambda x, i=i: (np.sqrt((centers[i][0] - 0)**2 + (centers[i][1] - 0)**2) - x[i]) >= 1e-8} for i in range(n)]
    cons += [{'type': 'ineq', 'fun': lambda x, i=i: (np.sqrt((1 - centers[i][0])**2 + (1 - centers[i][1])**2) - x[i]) >= 1e-8} for i in range(n)]
    
    res = minimize(objective, np.array(radii), bounds=bounds, constraints=cons)
    
    return (centers, res.x, np.sum(res.x))
