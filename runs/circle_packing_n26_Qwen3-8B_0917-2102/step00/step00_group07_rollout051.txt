import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Initialize centers in hexagonal grid
    centers = []
    radii = []
    for row in range(13):
        for col in range(26):
            x = col / (25)
            y = row / (12)
            if row % 2 == 1:
                x += 0.5 / 25
            centers.append([x, y])
    
    # Initial radius for all circles
    initial_radius = 0.05
    
    # Objective function to maximize sum of radii
    def objective(radii_vec):
        return -np.sum(radii_vec)  # Minimize negative sum to maximize sum
    
    # Constraints: circles must be within square and not overlapping
    def constraint_func(radii_vec):
        constraints = []
        for i in range(26):
            constraints.append((centers[i][0] - radii_vec[i] >= 0, centers[i][0] + radii_vec[i] <= 1))
            constraints.append((centers[i][1] - radii_vec[i] >= 0, centers[i][1] + radii_vec[i] <= 1))
        
        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.sqrt((centers[i][0] - centers[j][0])**2 + (centers[i][1] - centers[j][1])**2)
                constraints.append((dist >= radii_vec[i] + radii_vec[j],))
        return constraints
    
    # Optimization
    bounds = [(0, 1) for _ in range(26)]
    initial_guess = [initial_radius] * 26
    cons = []
    for i in range(26):
        for j in range(i + 1, 26):
            cons.append((lambda r, i=i, j=j: np.sqrt((centers[i][0] - centers[j][0])**2 + (centers[i][1] - centers[j][1])**2) - (r[i] + r[j])) <= 0)
        for i in range(26):
            cons.append((lambda r, i=i: centers[i][0] - r[i]) >= 0)
            cons.append((lambda r, i=i: 1 - centers[i][0] - r[i]) >= 0)
            cons.append((lambda r, i=i: centers[i][1] - r[i]) >= 0)
            cons.append((lambda r, i=i: 1 - centers[i][1] - r[i]) >= 0)
    
    result = minimize(objective, initial_guess, bounds=bounds, constraints=cons)
    
    centers = np.array(centers)
    radii = result.x
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
