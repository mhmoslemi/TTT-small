import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Hexagonal packing with 5 rows (26 circles)
    # Rows will be staggered with alternating x positions
    # Calculate optimal radii and positions
    
    # Initial guess for radii (symmetric distribution)
    radii = np.full(26, 0.05)
    centers = np.zeros((26, 2))
    
    # Generate hexagonal grid with 5 rows
    for i in range(5):
        for j in range(5 + (i % 2)):
            row = i
            col = j
            x = col + 0.5 * (i % 2)
            y = row * np.sqrt(3) / 2
            centers[i * 5 + j] = [x, y]
    
    # Optimization to maximize sum of radii with constraints
    def objective(r):
        return -np.sum(r)  # Minimize negative sum to maximize sum
    
    def constraints(r):
        # Boundary constraints
        for i in range(26):
            x, y = centers[i]
            if x - r[i] < -1e-12 or x + r[i] > 1 + 1e-12 or y - r[i] < -1e-12 or y + r[i] > 1 + 1e-12:
                return False
        # Non-overlapping constraints
        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < r[i] + r[j] - 1e-12:
                    return False
        return True
    
    bounds = [(0, 0.5) for _ in range(26)]
    
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraints}]
    )
    
    return (centers, result.x, np.sum(result.x))
