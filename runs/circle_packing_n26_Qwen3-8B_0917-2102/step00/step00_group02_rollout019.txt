import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import matplotlib.pyplot as plt

def run_packing():
    # Hexagonal packing with 5 rows of 5, 5, 5, 5, 6 circles
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    # Row 1: 6 circles
    for i in range(6):
        centers[i] = [i / 5, 0.1]
    # Row 2: 5 circles offset
    for i in range(5):
        centers[6 + i] = [i / 4 + 0.5, 0.3]
    # Row 3: 5 circles
    for i in range(5):
        centers[11 + i] = [i / 4 + 0.25, 0.5]
    # Row 4: 5 circles offset
    for i in range(5):
        centers[16 + i] = [i / 4 + 0.75, 0.7]
    # Row 5: 6 circles
    for i in range(6):
        centers[21 + i] = [i / 5, 0.9]

    # Initial radii based on spacing
    r = 0.15
    for i in range(n):
        x, y = centers[i]
        if x < r or x > 1 - r or y < r or y > 1 - r:
            r = min(r, min(x, 1 - x, y, 1 - y))

    radii = r * np.ones(n)

    # Define objective and constraints
    def objective(radii):
        return -np.sum(radii)

    def constraint_overlap(radii):
        dist = pdist(centers)
        dist = squareform(dist)
        for i in range(n):
            for j in range(i + 1, n):
                if dist[i, j] < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    def constraint_boundary(radii):
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8 or y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8:
                return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap},
            {'type': 'ineq', 'fun': constraint_boundary}]

    result = minimize(objective, radii, method='SLSQP', constraints=cons, bounds=[(0, 1) for _ in range(n)])
    radii = result.x
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
