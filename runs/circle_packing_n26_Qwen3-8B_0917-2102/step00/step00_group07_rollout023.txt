import numpy as np
from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d

        return -np.sum(radii)  # Maximize sum of radii

    def constraints(radii, centers, k=10):
        def circle_boundary(centers, radii):
            for i in range(len(centers)):
                x, y = centers[i]
                if not (0 <= x - radii[i] and x + radii[i] <= 1 and 0 <= y - radii[i] and y + radii[i] <= 1):
                    return False
            return True

        def circle_distances(centers, radii):
            n = len(centers)
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx**2 + dy**2)
                    if dist < radii[i] + radii[j] - 1e-8:
                        return False
            return True

        return circle_boundary(centers, radii) and circle_distances(centers, radii)

    # Initial hexagonal packing
    def hexagonal_centers(n):
        rows = int(np.ceil(n / 3))
        cols = int(np.ceil(n / rows))
        centers = []
        for i in range(rows):
            for j in range(cols):
                x = j + i / 2
                y = i
                if x < 1 and y < 1:
                    centers.append([x, y])
        return np.array(centers)

    centers = hexagonal_centers(26)
    radii = np.full(26, 0.01)

    # Optimization
    def f(r):
        return -np.sum(r)

    def cons(r):
        return constraints(r, centers)

    bounds = [(0, 0.1) for _ in range(26)]
    result = minimize(
        fun=f,
        x0=radii,
        bounds=bounds,
        constraints={"type": "ineq", "fun": cons},
        method="SLSQP"
    )

    final_radii = result.x
    return centers, final_radii, np.sum(final_radii)
