import numpy as np
from scipy.optimize import minimize
import random

def run_packing():
    # Hexagonal packing with 26 circles
    # First row: 1 circle
    # Second row: 2 circles
    # Third row: 3 circles
    # Fourth row: 4 circles
    # Fifth row: 5 circles
    # Sixth row: 5 circles (due to total 26)
    # Adjust positions to fit in unit square

    rows = [1, 2, 3, 4, 5, 5]
    total_circles = sum(rows)
    centers = np.zeros((total_circles, 2))
    radii = np.zeros(total_circles)
    max_radius = 0.1  # Initial guess for radius

    # Place circles in hexagonal packing
    for i, row in enumerate(rows):
        for j in range(row):
            col = j + i * 0.5
            row_pos = i * np.sqrt(3) / 2
            x = col / (sum(rows) - 1) if sum(rows) > 1 else 0.5
            y = row_pos / (len(rows) - 1) if len(rows) > 1 else 0.5
            centers[i * row + j] = [x, y]
            radii[i * row + j] = max_radius

    # Optimization to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    def constraint_func(radii):
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                x1, y1 = centers[i]
                x2, y2 = centers[j]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        for i in range(len(radii)):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8 or y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8:
                return False
        return True

    initial_guess = radii.copy()
    bounds = [(0, 0.5) for _ in range(total_circles)]
    cons = {'type': 'callback', 'fun': constraint_func}
    result = minimize(objective, initial_guess, bounds=bounds, constraints=cons, method='SLSQP')

    radii = result.x
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
