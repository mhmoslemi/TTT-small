import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import itertools

def run_packing():
    n = 26

    # Initial guess: hexagonal grid with 5 rows
    centers = []
    radii = []
    for row in range(5):
        for col in range(5):
            x = col + 0.5 * (row % 2)
            y = row + 0.5
            if row < 5 and col < 5:
                centers.append([x, y])
                radii.append(0.05)
    centers = np.array(centers)
    radii = np.array(radii)

    def objective(r):
        return -np.sum(r)  # maximize sum of radii

    def constraints(r):
        for i in range(n):
            if r[i] < 0:
                return False
        for i in range(n):
            x, y = centers[i]
            if (x - r[i] < -1e-12 or x + r[i] > 1 + 1e-12
                    or y - r[i] < -1e-12 or y + r[i] > 1 + 1e-12):
                return False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - 1e-12:
                    return False
        return True

    bounds = [(0, 0.5) for _ in range(n)]

    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints={"type": "callback", "fun": constraints},
        options={"disp": False}
    )

    return (centers, result.x, -result.fun)
