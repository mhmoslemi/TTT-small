import numpy as np
from scipy.optimize import minimize
from itertools import combinations

def run_packing():
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraints(radii, centers):
        constraints = []
        # Boundary constraint: radius must not exceed distance to nearest boundary
        for i in range(len(radii)):
            r = radii[i]
            x, y = centers[i]
            dist_to_left = x
            dist_to_right = 1 - x
            dist_to_bottom = y
            dist_to_top = 1 - y
            constraints.append((dist_to_left - r) * 1000)
            constraints.append((dist_to_right - r) * 1000)
            constraints.append((dist_to_bottom - r) * 1000)
            constraints.append((dist_to_top - r) * 1000)
        # Pairwise distance constraint: distance between centers >= sum of radii
        for i, j in combinations(range(len(radii)), 2):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            constraints.append((dist - radii[i] - radii[j]) * 1000)
        return constraints

    # Initial guess: hexagonal packing with adjusted radii
    n = 26
    centers = []
    for row in range(5):
        for col in range(5 + (row % 2)):
            x = col + (row % 2) / 2
            y = row + 0.25
            centers.append([x / 4, y / 4])
    centers = np.array(centers)
    radii = np.ones(n) * 0.05

    # Optimization
    bounds = [(0, 1) for _ in range(2 * n)]
    res = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': lambda r: r[i] for i in range(n)},
            {'type': 'ineq', 'fun': lambda r: np.sum(np.sqrt(np.sum((centers[i] - centers[j])**2) - r[i] - r[j]) for i, j in combinations(range(n), 2)))}
        ]
    )

    # Extract result
    radii = res.x
    centers = np.array(centers)
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
