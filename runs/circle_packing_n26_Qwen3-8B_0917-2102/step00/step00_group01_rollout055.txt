from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():

    rows = 5
    cols = 5
    centers = []
    radii = []
    
    # First row (top)
    for i in range(cols):
        x = i / (cols - 1)
        y = 1 - 1e-4  # Slight offset to avoid boundary issues
        centers.append([x, y])
    
    # Middle rows
    for row in range(1, rows - 1):
        for i in range(cols):
            x = i / (cols - 1)
            y = 1 - (row + 0.5) / (rows - 1) - 1e-4
            centers.append([x, y])
    
    # Bottom row (bottom)
    for i in range(cols):
        x = i / (cols - 1)
        y = 1e-4  # Slight offset to avoid boundary issues
        centers.append([x, y])
    
    # Initial radii estimation

        if row == 0 or row == rows - 1:
            return 0.1
        else:
            return 0.15
    
    radii = [initial_radius(x, y, row) for row, (x, y) in enumerate(centers)]
    
    # Optimization to maximize sum of radii while maintaining non-overlap and boundary constraints
    def objective(radii):
        return -np.sum(radii)
    
    def constraint1(radii):
        for i in range(len(radii)):
            x, y = centers[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12 or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:

        dist_matrix = squareform(pdist(centers))

        return 0
    
    cons = [{'type': 'ineq', 'fun': constraint1},

    bounds = Bounds(lb=np.zeros(len(radii)), ub=np.inf)
    
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)
    
    return (np.array(centers), np.array(result.x), np.sum(result.x))
