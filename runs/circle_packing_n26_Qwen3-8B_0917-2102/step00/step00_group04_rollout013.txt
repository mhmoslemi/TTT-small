import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraint1(radii):
        return np.sum(radii) - 2.636

    def constraint2(radii):
        return np.sum(radii) - 2.636

    # Initial guess for radii
    initial_r = np.random.uniform(0.01, 0.1, 26)

    # Bounds for radii
    bounds = [(0, 1) for _ in range(26)]

    # Initial guess for centers
    centers = np.random.rand(26, 2)

    def constraint3(x):
        # x[0:26] are the radii
        # x[26:] are the centers, flattened
        r = x[:26]
        centers = x[26:].reshape(26, 2)
        # Check boundaries
        for i in range(26):
            if centers[i, 0] - r[i] < -1e-12 or centers[i, 0] + r[i] > 1 + 1e-12:
                return 1.0
            if centers[i, 1] - r[i] < -1e-12 or centers[i, 1] + r[i] > 1 + 1e-12:
                return 1.0
        # Check overlaps
        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < r[i] + r[j] - 1e-12:
                    return 1.0
        return 0.0

    # Optimize
    result = minimize(
        objective,
        x0=np.concatenate((initial_r, centers.flatten())),
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2},
            {'type': 'eq', 'fun': constraint3}
        ],
        method='SLSQP'
    )

    radii = result.x[:26]
    centers = result.x[26:].reshape(26, 2)
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
