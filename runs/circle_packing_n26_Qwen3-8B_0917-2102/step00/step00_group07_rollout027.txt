import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    # Initial guess: hexagonal packing with 5 rows
    n = 26
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.05)

    # Hexagonal packing with 5 rows
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row * np.sqrt(3) / 2
        centers[i, 0] = x / 5
        centers[i, 1] = y / 5

    def objective(radii):
        return -np.sum(radii)

    def constraint_circle(i, radii):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            return -np.inf
        return r

    def constraint_pair(i, j, radii):
        x1, y1 = centers[i]
        x2, y2 = centers[j]
        dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
        return dist - (radii[i] + radii[j]) + 1e-8

    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda r, i=i: constraint_circle(i, r)})
    for i, j in combinations(range(n), 2):
        cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: constraint_pair(i, j, r)})

    bounds = [(0, 1) for _ in range(n)]

    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)
    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
