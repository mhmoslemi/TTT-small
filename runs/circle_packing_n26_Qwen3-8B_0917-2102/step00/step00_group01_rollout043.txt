from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():

    radii = np.full(n, 0.05)

    def objective(r):
        return -np.sum(r)

    def constraints(r, centers):
        return (
            np.sum((pdist(centers) < r[:, np.newaxis] + r[np.newaxis, :]) * (pdist(centers) < r[:, np.newaxis] + r[np.newaxis, :]), axis=1) - n + 1
        )

    def bounds(r):
        return Bounds(0, 1.0)

    result = minimize(
        objective,
        radii,
        args=(initial_guess,),
        method='SLSQP',
        constraints={'type': 'ineq', 'fun': constraints, 'args': (initial_guess,)},
        bounds=bounds(radii)
    )

    centers = initial_guess
    radii = result.x
    sum_radii = np.sum(radii)

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            raise ValueError(f"Circle {i} outside unit square")

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError(f"Circles {i} and {j} overlap")

    return centers, radii, sum_radii
