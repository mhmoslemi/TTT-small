<strategy>Use a hexagonal packing with adjusted circles near boundaries and corners to maximize total radius sum.</strategy>

import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import lsqr
import time

def run_packing():
    np.random.seed(42)
    n = 26
    
    # Initial guess: hexagonal packing
    rows, cols = np.indices((n, n))
    mask = (rows + cols) % 2 == 0
    idx = np.where(mask)[0]
    centers = np.column_stack((idx % n / n, idx // n / n))
    radii = np.full(n, 0.05)
    
    def objective(r):
        return -np.sum(r)
    
    def constraints(r, centers):
        return np.concatenate([
            np.sum((centers[i] - centers[j])**2) - (r[i] + r[j])**2 for i in range(n) for j in range(i+1, n)
        ])
    
    def bounds(r):
        return [(0, 0.1) for _ in range(n)]
    
    def pack(r):
        obj = -np.sum(r)
        cons = []
        for i in range(n):
            for j in range(i+1, n):
                d = np.linalg.norm(centers[i] - centers[j])
                cons.append(d - (r[i] + r[j]) >= 0)
        return obj, cons
    
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=pack, options={'ftol': 1e-10, 'maxiter': 500})
    
    if not result.success:
        raise ValueError("Optimization failed")
    
    final_radii = result.x
    return centers, final_radii, np.sum(final_radii)
