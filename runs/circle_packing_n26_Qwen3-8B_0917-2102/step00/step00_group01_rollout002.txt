import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial guess: hexagonal packing with 5 rows
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Generate hexagonal grid
    for i in range(5):
        for j in range(6):
            row = i * (np.sqrt(3) / 2) + 0.5
            col = j + 0.5
            x = col - 3.5
            y = row - 2.5
            if i % 2 == 0:
                x = col - 3.5
            else:
                x = col - 3.5 + 0.5
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers[i * 6 + j] = [x, y]
                radii[i * 6 + j] = 0.05

    # Add boundary circles
    for i in range(26):
        if centers[i, 0] < 0.1:
            centers[i, 0] = 0.1
            radii[i] = 0.05
        elif centers[i, 0] > 0.9:
            centers[i, 0] = 0.9
            radii[i] = 0.05
        if centers[i, 1] < 0.1:
            centers[i, 1] = 0.1
            radii[i] = 0.05
        elif centers[i, 1] > 0.9:
            centers[i, 1] = 0.9
            radii[i] = 0.05

    def objective(radii):
        return -np.sum(radii)

    def constraint_func(radii):
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return 1
        return 0

    cons = {'type': 'ineq', 'fun': constraint_func}
    bounds = [(0, 0.5) for _ in range(num_circles)]

    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)

    return (centers, result.x, np.sum(result.x))
