import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Hexagonal packing with 26 circles
    # First row: 1 circle
    # Second row: 2 circles
    # Third row: 3 circles
    # Fourth row: 4 circles
    # Fifth row: 5 circles
    # Sixth row: 5 circles
    # Seventh row: 4 circles
    # Eighth row: 3 circles
    # Ninth row: 2 circles
    # Tenth row: 1 circle
    # Total: 1+2+3+4+5+5+4+3+2+1 = 26

    # Positions for hexagonal packing
    pos = []
    for row in range(10):
        for col in range(10 - row):
            x = col + 0.5 * row
            y = 0.5 * row
            if row % 2 == 0:
                x += 0.5
            pos.append((x, y))

    # Convert to numpy array
    centers = np.array(pos)
    n = len(centers)

    # Initial radii as uniform value
    r0 = 0.1
    radii = np.full(n, r0)

    # Define optimization variables
    bounds = [(0, 1) for _ in range(2 * n)]  # for centers
    bounds += [(0, 1)] * n  # for radii

    # Objective function to maximize sum of radii
    def objective(x):
        centers_flat = x[:2 * n].reshape(n, 2)
        radii_flat = x[2 * n:].reshape(n)
        return -np.sum(radii_flat)  # minimize negative sum

    # Constraints for non-overlapping and boundary conditions
    def constraint(x):
        centers_flat = x[:2 * n].reshape(n, 2)
        radii_flat = x[2 * n:].reshape(n)
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return -1
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_flat[i] + radii_flat[j]:
                    return -1
        return 0

    # Optimization
    x0 = np.concatenate([centers.ravel(), radii])
    cons = {'type': 'ineq', 'fun': constraint}
    result = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP')

    # Extract optimal centers and radii
    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = result.x[2 * n:].reshape(n)
    sum_radii = np.sum(radii_opt)

    return centers_opt, radii_opt, sum_radii
