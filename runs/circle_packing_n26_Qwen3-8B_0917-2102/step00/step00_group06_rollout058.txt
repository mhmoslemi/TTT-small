    centers = []

    radii.append(r_large)

    for i in range(5):

        x = 0.5 + (r_small + r_large) * np.cos(angle)

        centers.append([x, y])
        radii.append(r_small)
    
    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)
