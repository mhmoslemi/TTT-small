import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    np.random.seed(42)
    
    n = 26
    initial_centers = np.random.rand(n, 2)
    initial_radii = np.full(n, 0.1)
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraints(radii):
        constraints = []
        for i in range(n):
            constraints.append((radii[i] <= 1 - initial_centers[i, 0]))
            constraints.append((radii[i] <= 1 - initial_centers[i, 1]))
            constraints.append((radii[i] <= initial_centers[i, 0]))
            constraints.append((radii[i] <= initial_centers[i, 1]))
        return constraints
    
    def dist_matrix(centers):
        return squareform(pdist(centers))
    
    def non_overlap_constraints(centers, radii):
        return [radii[i] + radii[j] <= dist_matrix(centers)[i, j] for i in range(n) for j in range(i + 1)]
    
    def optimization():
        bounds = Bounds(0, 1)
        cons = non_overlap_constraints(initial_centers, initial_radii)
        res = minimize(objective, initial_radii, method='SLSQP', bounds=bounds, constraints=cons)
        return res.x
    
    radii = optimization()
    
    # Final refinement
    centers = np.zeros((n, 2))
    for i in range(n):
        dx = np.random.rand() * 0.5
        dy = np.random.rand() * 0.5
        centers[i] = np.clip([initial_centers[i, 0] + dx, initial_centers[i, 1] + dy], 0, 1)
    
    # Re-optimization with fixed centers
    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: dist - (r[i] + r[j])})
    
    bounds = Bounds(0, 1 - np.min(np.abs(centers), axis=1))
    res = minimize(lambda r: -np.sum(r), radii, method='SLSQP', bounds=bounds, constraints=cons)
    
    final_radii = res.x
    final_centers = centers
    
    return final_centers, final_radii, np.sum(final_radii)
