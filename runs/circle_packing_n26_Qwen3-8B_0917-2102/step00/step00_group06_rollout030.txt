import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    n = 26
    # Initial guess for centers in hexagonal packing
    centers = np.zeros((n, 2))
    r = 0.1  # Initial radius guess
    radii = np.full(n, r)
    # Hexagonal packing pattern for 26 circles
    for i in range(n):
        row = i // 5
        col = i % 5
        centers[i, 0] = col / 4.0 + 0.1
        centers[i, 1] = row / 4.0 + 0.1

    # Optimization parameters
    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii

    def constraints(radii):
        dist_matrix = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return False
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8 or y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e8:
                return False
        return True

    res = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=[(0, 1)] * n,
        constraints=[{'type': 'ineq', 'fun': constraints}]
    )
    radii = res.x
    # Final check and adjust for boundary condition
    for i in range(n):
        x, y = centers[i]
        if x - radii[i] < -1e-8:
            radii[i] = x
        if x + radii[i] > 1 + 1e-8:
            radii[i] = 1 - x
        if y - radii[i] < -1e-8:
            radii[i] = y
        if y + radii[i] > 1 + 1e-8:
            radii[i] = 1 - y
    return centers, radii, np.sum(radii)
