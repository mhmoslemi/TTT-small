import numpy as np
from scipy.optimize import minimize

def run_packing():
    def objective_function(radii):
        return -np.sum(radii)

    def constraints(radii):
        return np.sum(radii) - 2.636

    def boundary_constraints(radii):
        return np.sum(radii) - 2.636

    def generate_boundary_circles():
        boundary_radii = []
        boundary_centers = []
        for i in range(4):
            radius = 0.2
            if i == 0:
                boundary_centers.append([0.2, 0.5])
            elif i == 1:
                boundary_centers.append([0.5, 0.2])
            elif i == 2:
                boundary_centers.append([0.8, 0.5])
            elif i == 3:
                boundary_centers.append([0.5, 0.8])
            boundary_radii.append(radius)
        return np.array(boundary_centers), np.array(boundary_radii)

    def generate_inner_circles():
        inner_centers = []
        inner_radii = []
        for i in range(22):
            if i < 11:
                x = 0.25 + i * 0.1
                y = 0.25
                r = 0.15
                inner_centers.append([x, y])
                inner_radii.append(r)
            else:
                x = 0.75 - (i - 11) * 0.1
                y = 0.75
                r = 0.15
                inner_centers.append([x, y])
                inner_radii.append(r)
        return np.array(inner_centers), np.array(inner_radii)

    boundary_centers, boundary_radii = generate_boundary_circles()
    inner_centers, inner_radii = generate_inner_circles()

    all_centers = np.vstack((boundary_centers, inner_centers))
    all_radii = np.concatenate((boundary_radii, inner_radii))

    def compute_distances(centers, radii):
        n = centers.shape[0]
        dists = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dists[i, j] = np.sqrt(dx**2 + dy**2)
                dists[j, i] = dists[i, j]
        return dists

    def constraints_function(radii):
        dists = compute_distances(all_centers, radii)
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    def optimize_packing():
        bounds = [(0, 1.0) for _ in range(26)]
        result = minimize(
            fun=objective_function,
            x0=all_radii,
            method='SLSQP',
            bounds=bounds,
            constraints=[
                {'type': 'ineq', 'fun': constraints_function},
                {'type': 'eq', 'fun': lambda x: np.sum(x) - 2.636}
            ]
        )
        return result.x

    radii = optimize_packing()
    centers = all_centers

    return centers, radii, np.sum(radii)
