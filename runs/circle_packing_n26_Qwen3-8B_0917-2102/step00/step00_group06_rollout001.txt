import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial guess for centers using hexagonal grid
    centers = []
    radii = []

    # Place 13 horizontal rows with 2 circles per row
    for row in range(13):
        y = row / 12
        for col in range(2):
            x = col + 0.5
            centers.append([x, y])
            radii.append(0.05)

    # Place 13 vertical columns with 2 circles per column
    for col in range(13):
        x = col / 12
        for row in range(2):
            y = row + 0.5
            centers.append([x, y])
            radii.append(0.05)

    centers = np.array(centers)
    radii = np.array(radii)

    def objective(r):
        return -np.sum(r)

    def constraint_func(r):
        n = len(r)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < r[i] + r[j] - 1e-12:
                    return False
        return True

    # Optimization with bounds and constraints
    bounds = [(0, 0.5) for _ in range(len(radii))]
    cons = {'type': 'callback', 'fun': constraint_func}

    result = minimize(objective, radii, bounds=bounds, constraints=cons)

    if not result.success:
        raise RuntimeError("Optimization failed")

    return centers, result.x, np.sum(result.x)
