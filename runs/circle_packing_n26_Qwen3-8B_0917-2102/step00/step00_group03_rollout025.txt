import numpy as np
import scipy.optimize as opt

def run_packing():
    n = 26
    # Initial hexagonal packing in a square
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Hexagonal packing in a square, first row
    r = 1.0 / (2 * np.sqrt(3))
    for i in range(n):
        if i % 2 == 0:
            x = i * (1.5 * r)
            y = 0.5 * r
        else:
            x = (i - 1) * (1.5 * r) + 0.75 * r
            y = 0.5 * r
        centers[i] = [x, y]
        radii[i] = r

    # Adjust for boundary conditions using optimization
    bounds = [(0, 1) for _ in range(2 * n)]
    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            cons.append((centers[i] - centers[j]) ** 2 - (radii[i] + radii[j]) ** 2)

    # Update boundaries and radii via optimization
    def objective(x):
        centers_flat = x[:2 * n].reshape(n, 2)
        radii_flat = x[2 * n:].reshape(n)
        return -np.sum(radii_flat)

    result = opt.minimize(objective, np.concatenate([centers.flatten(), radii]), bounds=bounds, constraints=cons)
    centers = result.x[:2 * n].reshape(n, 2)
    radii = result.x[2 * n:].reshape(n)

    # Final validation
    for i in range(n):
        x, y = centers[i]
        if not (0 <= x <= 1 and 0 <= y <= 1):
            raise ValueError("Center out of bounds")
        if radii[i] < 0:
            raise ValueError("Negative radius")

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            if np.sqrt(dx ** 2 + dy ** 2) < radii[i] + radii[j] - 1e-8:
                raise ValueError("Overlap detected")

    return centers, radii, np.sum(radii)
