import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initialize centers with hexagonal packing
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    max_radius = 0.0

    # Place circles in hexagonal grid with decreasing spacing
    for i in range(26):
        row = i // 5
        col = i % 5
        x = 0.2 + col * 0.25
        y = 0.2 + row * 0.25
        if row % 2 == 1:
            x += 0.125
        centers[i] = [x, y]
        # Compute radius based on distance to nearest boundary
        dist_to_boundary = np.min([x, 1 - x, y, 1 - y])
        radii[i] = dist_to_boundary * 0.8
        max_radius += radii[i]

    # Optimize radii with constraints
    def objective(radii):
        return -np.sum(radii)

    def constraint_func(radii):
        dists = np.zeros(26)
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dists[i, j] = np.sqrt(dx**2 + dy**2)
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return 1
        return 0

    cons = {'type': 'ineq', 'fun': constraint_func}
    bounds = [(0, 1.0) for _ in range(26)]

    result = optimize.minimize(objective, radii, bounds=bounds, constraints=cons)
    radii = result.x
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
