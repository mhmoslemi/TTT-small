    centers = []

    radii = np.full(len(centers), 0.1)

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        constraints = []
        for i in range(len(radii)):
            r = radii[i]
            if r < 0:
                return False
            if centers[i, 0] - r < 0 or centers[i, 0] + r > 1 or centers[i, 1] - r < 0 or centers[i, 1] + r > 1:

                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    # Convert radii to flat array
    flat_radii = radii.flatten()
    result = minimize(objective, flat_radii, constraints=lambda x: constraints(x), method='SLSQP')

    if not result.success:
        raise ValueError("Optimization failed")

    centers = np.array(centers)
    sum_radii = np.sum(radii)
