import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    # Initial hexagonal packing with 5 rows
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    
    # Initial hexagonal packing with spacing
    row_height = np.sqrt(3) / 2
    spacing = 1.0 / (np.sqrt(3) * (num_circles ** 0.5))
    
    for i in range(num_circles):
        row = i // 5
        col = i % 5
        x = col * spacing
        y = row * row_height
        centers[i] = [x, y]
        radii[i] = spacing / 2
    
    # Define objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)
    
    # Define constraints: non-overlapping and boundary
    def constraint_non_overlap(radii):
        dists = squareform(pdist(centers))
        return np.sum(np.maximum(0, radii[:, np.newaxis] + radii[np.newaxis, :] - dists - 1e-8))
    
    def constraint_boundary(radii):
        return np.sum(np.maximum(0, radii - (1 - centers[:, 0]) - 1e-8) + 
                     np.sum(np.maximum(0, radii - (1 - centers[:, 1]) - 1e-8)))
    
    # Optimization
    res = minimize(
        objective,
        radii,
        constraints=[
            {'type': 'ineq', 'fun': constraint_non_overlap},
            {'type': 'ineq', 'fun': constraint_boundary}
        ],
        method='SLSQP'
    )
    
    return centers, res.x, np.sum(res.x)
