from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import random

    # Initial hexagonal packing with boundary adjustment
    centers = []
    radii = []
    
    # First row of circles aligned to bottom
    for i in range(13):
        x = i * (1 - 0.05) / 12
        y = 0.05
        centers.append([x, y])
        radii.append(0.05)
    
    # Second row of circles aligned to top
    for i in range(13):
        x = i * (1 - 0.05) / 12
        y = 1 - 0.05
        centers.append([x, y])
        radii.append(0.05)
    
    centers = np.array(centers)
    radii = np.array(radii)
    
    # Optimization setup
    def objective(r):
        return -sum(r)
    
    def constraints(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return False

            x, y = centers[i]
            if x - r[i] < 0 or x + r[i] > 1 or y - r[i] < 0 or y + r[i] > 1:
                return False
        return True
    
    result = minimize(objective, radii, method='SLSQP', constraints={"type": "constraint", "fun": constraints})
    
    return (centers, result.x, sum(result.x))
