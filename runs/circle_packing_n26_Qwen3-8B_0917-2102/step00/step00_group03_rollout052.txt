                centers.append([x, y])
                radii.append(0.01)

    radii = np.array(radii)

        return -np.sum(r)  # Maximize sum of radii

        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))

    result = minimize(lambda r: -np.sum(r), radii, method='SLSQP', bounds=bounds, constraints={'type': 'custom', 'fun': constraints})
