import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraints(radii):
        return np.sum(radii) - 2.636  # Ensure sum is at least target

    def non_overlapping(c):
        dists = squareform(pdist(c))
        for i in range(len(c)):
            for j in range(i + 1, len(c)):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    def boundary(c, r):
        for i in range(len(c)):
            x, y = c[i]
            if x - r[i] < -1e-8 or x + r[i] > 1 + 1e-8 or y - r[i] < -1e-8 or y + r[i] > 1 + 1e-8:
                return False
        return True

    # Initial hexagonal packing with 26 circles
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    rows = int(np.sqrt(n))
    cols = int(np.ceil(n / rows))
    for i in range(rows):
        for j in range(cols):
            idx = i * cols + j
            if idx >= n:
                break
            x = j + 0.5 * (i % 2)
            y = i + 0.5
            centers[idx] = [x, y]
            radii[idx] = 0.05

    # Optimization
    bounds = [(0, 0.1) for _ in range(n)]
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': lambda r: constraints(r)},
            {'type': 'ineq', 'fun': lambda r: non_overlapping(centers)},
            {'type': 'ineq', 'fun': lambda r: boundary(centers, r)}
        ],
        method='SLSQP'
    )

    return (centers, result.x, np.sum(result.x))
