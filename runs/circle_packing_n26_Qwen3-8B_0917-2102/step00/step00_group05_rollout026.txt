from scipy.optimize import minimize

def run_packing():
    np.random.seed(42)
    n = 26
    # Initial guess: hexagonal packing
    centers = np.zeros((n, 2))
    r = 0.1
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
    # Scale to fit in [0,1]^2
    centers -= centers.min(axis=0)
    centers /= centers.max(axis=0)
    radii = np.full(n, r)
    
    # Objective function: maximize sum of radii
    def objective(x):
        # x[0:2n] = centers flattened, x[2n:2n+n] = radii
        centers_flat = x[:2*n].reshape(n, 2)
        radii_flat = x[2*n:]
        sum_radii = np.sum(radii_flat)
        return -sum_radii  # Minimize negative to maximize
    
    # Constraints: non-overlapping and within bounds
    def constraint(x):
        centers_flat = x[:2*n].reshape(n, 2)
        radii_flat = x[2*n:]
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return -1
        for i in range(n):
            for j in range(i+1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return -1
        return 0
    
    # Initial guess
    x0 = np.r_[centers.reshape(-1), radii]
    
    # Optimization
    result = minimize(objective, x0, constraints={"type": "ineq", "fun": constraint}, bounds=[(-1, 1+1e-12) for _ in range(2*n + n)])
    
    centers_opt = result.x[:2*n].reshape(n, 2)
    radii_opt = result.x[2*n:]
    sum_radii = np.sum(radii_opt)
    
    # Validate packing
    def validate_packing(centers, radii):
        n = centers.shape[0]
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True
    
    if not validate_packing(centers_opt, radii_opt):
        raise ValueError("Invalid packing")
    
    return (centers_opt, radii_opt, sum_radii)
