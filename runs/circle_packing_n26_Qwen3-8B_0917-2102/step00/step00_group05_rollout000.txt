import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance_matrix

def run_packing():
    np.random.seed(42)
    n = 26
    # Initial hexagonal packing
    centers = np.zeros((n, 2))
    r = np.zeros(n)
    for i in range(n):
        row = i // 5
        col = i % 5
        centers[i, 0] = 0.2 + col * 0.22
        centers[i, 1] = 0.2 + row * 0.22
        r[i] = 0.1

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        constraints = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return False
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    result = minimize(objective, r, constraints=lambda x: constraints(x), method='SLSQP', bounds=[(0, 1)] * n)
    radii = result.x
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
