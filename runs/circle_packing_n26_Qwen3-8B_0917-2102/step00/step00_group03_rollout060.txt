from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    n = 26
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        centers[i, 0] = col + 0.5 * (1 - 2 * (row % 2))
        centers[i, 1] = row + 0.5

    # Optimization variables: radii
    radii0 = np.full(n, 0.1)

    # Objective: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraints: non-overlapping and boundary
    def constraint(radii):

        for i in range(n):

                if dists[i, j] < radii[i] + radii[j] + 1e-8:

                return -1
        return 0

    # Optimization
    bounds = [(0, 1) for _ in range(n)]
    result = minimize(objective, radii0, bounds=bounds, constraints={"type": "ineq", "fun": constraint})

    # Extract results
    radii = result.x
    centers = np.array(centers)
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
