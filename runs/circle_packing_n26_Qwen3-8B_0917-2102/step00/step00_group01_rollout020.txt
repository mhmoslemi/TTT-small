import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    # Initial hexagonal grid with 26 circles
    n = 26
    rows = 5
    cols = 5 + 1  # 5 full rows, 6th row starting with 1 circle
    grid = []
    radii = []

    # Create hexagonal grid
    for i in range(rows):
        for j in range(cols):
            if i % 2 == 0:
                x = j / (cols - 1) if j < cols - 1 else 0.5
                y = i / (rows - 1)
            else:
                x = (j - 0.5) / (cols - 1) if j > 0 else 0.5
                y = i / (rows - 1)
            grid.append([x, y])
    
    centers = np.array(grid)
    radii = np.array([0.1] * n)
    
    # Optimization
    def objective(r):
        return -np.sum(r)
    
    def constraints(r):
        constraints = []
        for i in range(n):
            constraints.append((centers[i][0] - r[i] >= 0, "x_low_" + str(i)))
            constraints.append((centers[i][0] + r[i] <= 1, "x_high_" + str(i)))
            constraints.append((centers[i][1] - r[i] >= 0, "y_low_" + str(i)))
            constraints.append((centers[i][1] + r[i] <= 1, "y_high_" + str(i)))
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                constraints.append((dist >= r[i] + r[j] - 1e-8, "dist_" + str(i) + "_" + str(j)))
        return constraints
    
    bounds = Bounds(0, 1)
    
    result = minimize(objective, radii, bounds=bounds, constraints=constraints)
    
    centers = np.array(grid)
    radii = result.x
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
