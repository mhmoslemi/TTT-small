import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    # Define initial hexagonal grid with 5 rows (13 circles) and 2 rows of 12 circles
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    
    # Layer 1: 13 circles in one row (aligned to bottom)
    for i in range(13):
        centers[i] = [i / 12, 0.0]
        radii[i] = 0.05
    
    # Layer 2: 12 circles in a hexagonal row
    for i in range(12):
        centers[13 + i] = [i / 11, 0.25]
        radii[13 + i] = 0.05
    
    # Layer 3: 12 circles in a hexagonal row
    for i in range(12):

        radii[25 + i] = 0.05
    
    # Optimize radii and positions with constraints
    def objective(x):
        return -np.sum(x[13:])  # maximize sum of radii
    
    def constraint1(x):
        # Ensure all circles are within the unit square
        for i in range(num_circles):
            x_, y_ = x[2*i], x[2*i + 1]
            r = x[num_circles + i]
            if x_ - r < -1e-12 or x_ + r > 1 + 1e-12 or y_ - r < -1e-12 or y_ + r > 1 + 1e-12:
                return -1
        return 0
    
    def constraint2(x):
        # Ensure no overlap between circles
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = x[2*i] - x[2*j]
                dy = x[2*i + 1] - x[2*j + 1]
                dist = np.sqrt(dx*dx + dy*dy)

                if dist < r1 + r2 - 1e-12:

        return 0
    
    x0 = np.concatenate([centers.flatten(), radii])
    cons = [{'type': 'ineq', 'fun': constraint1},

    res = minimize(objective, x0, constraints=cons, method='SLSQP')
    
    centers = res.x[:2*num_circles].reshape(-1, 2)

    return centers, radii, sum_radii
