import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import matplotlib.pyplot as plt

def run_packing():
    # Initial hexagonal packing with 26 circles
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Generate hexagonal packing
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 1/(5))
        y = row + 0.5 * (1 - 1/(5))
        centers[i] = [x, y]
        radii[i] = 0.1
    
    # Optimization function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)
    
    # Constraint: circles do not overlap
    def constraint(radii):
        dist = squareform(pdist(centers))
        return np.array([dist[i, j] - (radii[i] + radii[j]) for i in range(n) for j in range(i+1)])
    
    # Optimization
    bounds = [(0, 1) for _ in range(n)]
    result = minimize(objective, radii, bounds=bounds, constraints={'type': 'ineq', 'fun': constraint})
    
    # Final radii and centers
    radii = result.x
    sum_radii = np.sum(radii)
    
    # Check boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            radii[i] = max(0, min(r, 1 - max(x, 1 - x)))
    
    return centers, radii, sum_radii
