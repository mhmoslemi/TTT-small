import numpy as np
import math

def run_packing():
    # Define hexagonal grid parameters
    num_rows = 5
    num_cols = 5
    side_length = 1 / (num_rows + 1)
    radius = side_length * math.sqrt(3) / 2
    
    centers = []
    radii = []
    
    # Create hexagonal grid with boundary circles
    for row in range(num_rows):
        for col in range(num_cols):
            x = col * side_length + side_length / 2
            y = row * (side_length * math.sqrt(3)) + (side_length / 2)
            centers.append([x, y])
            radii.append(radius)
    
    # Add boundary circles
    # Top-left
    centers.append([0.1, 1 - 0.1])
    radii.append(0.1)
    # Top-right
    centers.append([1 - 0.1, 1 - 0.1])
    radii.append(0.1)
    # Bottom-left
    centers.append([0.1, 0.1])
    radii.append(0.1)
    # Bottom-right
    centers.append([1 - 0.1, 0.1])
    radii.append(0.1)
    
    # Adjust radii for boundary circles to prevent overlap
    for i in range(4):
        radii[i] = 0.08
    
    centers = np.array(centers)
    radii = np.array(radii)
    
    return centers, radii, np.sum(radii)
