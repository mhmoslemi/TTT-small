import scipy.optimize as opt
from scipy.spatial import distance

    col_count = 5

    # Create a hexagonal grid with staggered rows
    centers = []
    radii = []

            centers.append([x, y])

    max_x = max(c[0] for c in centers)
    max_y = max(c[1] for c in centers)
    scale = 1 / max(max_x, max_y)
    centers = np.array(centers) * scale

    radii = np.full(n, spacing / 2)
    
    # Optimization problem to maximize sum of radii with constraints
    def objective(x):
        return -np.sum(x)  # Minimize negative sum to maximize sum
    
    def constraint1(x):
        return np.sum(x)  # Ensure all radii are non-negative
    
    def constraint2(x):
        # Check if any circle exceeds square boundary
        for i in range(n):
            if centers[i][0] - x[i] < -1e-12 or centers[i][0] + x[i] > 1 + 1e-12:
                return -1
            if centers[i][1] - x[i] < -1e-12 or centers[i][1] + x[i] > 1 + 1e-12:
                return -1
        return 0

        # Check if any pair of circles overlap

        return 0
    
    # Set up optimization problem
    bounds = [(0, 1) for _ in range(n)]

        {'type': 'ineq', 'fun': constraint2},

    ]

        bounds=bounds,
        constraints=cons,

    sum_radii = np.sum(radii)
    return np.array(centers), radii, sum_radii
