from scipy.optimize import minimize
from scipy.spatial import distance
import matplotlib.pyplot as plt

def run_packing():
    n = 26
    initial_centers = np.random.rand(n, 2)
    initial_radii = np.random.rand(n) * 0.1

    def objective(r, centers):
        return -np.sum(r)

    def constraints(r, centers):
        constraints = []
        for i in range(n):
            constraints.append((centers[i, 0] - r[i] >= 0) * 1e12)
            constraints.append((centers[i, 1] - r[i] >= 0) * 1e12)
            constraints.append((1 - centers[i, 0] - r[i] >= 0) * 1e12)
            constraints.append((1 - centers[i, 1] - r[i] >= 0) * 1e12)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                constraints.append((dist - r[i] - r[j] >= 0) * 1e12)
        return constraints

    bounds = [(0, 1) for _ in range(2 * n)]
    result = minimize(objective, initial_radii, bounds=bounds, constraints=constraints, method='SLSQP')
    radii = result.x
    centers = initial_centers

    return (centers, radii, np.sum(radii))
