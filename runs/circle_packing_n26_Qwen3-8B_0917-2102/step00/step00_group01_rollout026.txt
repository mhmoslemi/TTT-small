import numpy as np
from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d
from scipy.spatial.distance import pdist, squareform

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        return np.sum(radii) - 2.636

    def bounds(radii):
        return [(0, 1) for _ in radii]

    def packing_func(centers, radii):
        n = len(radii)
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
                return False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    n = 26
    initial_centers = np.random.rand(n, 2)
    initial_radii = np.random.rand(n) * 0.1

    res = minimize(
        objective,
        initial_radii,
        method='SLSQP',
        bounds=bounds(initial_radii),
        constraints=[
            {'type': 'ineq', 'fun': lambda x: constraints(x)}
        ],
        options={'ftol': 1e-10, 'maxiter': 500},
        jac=None
    )

    centers = initial_centers
    radii = res.x

    if packing_func(centers, radii):
        total = np.sum(radii)
        return (centers, radii, total)
    else:
        raise ValueError("Packing failed to satisfy constraints")
