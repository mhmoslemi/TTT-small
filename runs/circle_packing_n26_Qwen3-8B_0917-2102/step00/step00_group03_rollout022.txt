import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Hexagonal packing with staggered rows
    rows = 5
    cols = 6
    spacing = 1 / (rows + 1)
    stagger = spacing / 2

    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * spacing + (i % 2) * stagger
            centers[i * cols + j] = [x, y]
    
    # Calculate radii based on minimal distance between circles and boundaries
    min_dist = 1 / (rows + 1)
    radii = np.full(n, min_dist / 2)

    # Optimize radii to maximize sum while ensuring no overlaps
    def objective(r):
        return -np.sum(r)

    def constraint1(r):
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint1}]
    bounds = [(0, 1) for _ in range(n)]

    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)

    return (centers, result.x, np.sum(result.x))
