import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance
from scipy.optimize import Bounds

def run_packing():
    np.random.seed(42)
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Initial hexagonal grid with some boundary circles
    rows = 5
    cols = 5
    spacing = 0.5

    # Place central circles in hexagonal grid
    for i in range(rows):
        for j in range(cols):
            if i % 2 == 0:
                x = j * spacing
                y = i * spacing * np.sqrt(3)/2
            else:
                x = (j + 0.5) * spacing
                y = i * spacing * np.sqrt(3)/2
            centers[i * cols + j] = [x, y]

    # Resize circles to fit in unit square
    def objective(r):
        return -np.sum(r)

    def constraints(r, centers):
        cons = []
        for i in range(n):
            r_i = r[i]
            cons.append((r_i, 'ineq', 0.5 - np.sqrt((centers[i, 0] - centers[i, 0])**2 + (centers[i, 1] - centers[i, 1])**2)))
        return cons

    bounds = [(0, 0.5) for _ in range(n)]
    r0 = np.ones(n) * 0.1

    result = minimize(objective, r0, bounds=bounds, constraints=constraints, method='SLSQP')
    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
