import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Hexagonal packing with 26 circles
    # First row: 1 circle
    # Second row: 2 circles
    # Third row: 3 circles
    # Fourth row: 4 circles
    # Fifth row: 4 circles
    # Sixth row: 2 circles
    # Seventh row: 1 circle
    # Total: 1+2+3+4+4+2+1 = 17, plus 9 more for a total of 26
    
    # Initial estimate for radii based on hexagonal packing
    # Assume maximum radius such that circles fit with spacing
    def calc_radii(n_circles):
        return np.full(n_circles, 0.15)
    
    # Define centers based on hexagonal packing
    def calc_centers(n_circles):
        centers = []
        for i in range(n_circles):
            row = i // 4
            col = i % 4
            x = col + 0.5 * (row % 2)
            y = 0.5 * row
            centers.append((x, y))
        return np.array(centers)
    
    # Initial packing
    initial_centers = calc_centers(26)
    initial_radii = calc_radii(26)
    
    # Optimization objective: maximize sum of radii
    def objective(x):
        return -np.sum(x)
    
    # Optimization constraints
    def constraint(x, i):
        return x[i]
    
    # Bounds: radii >= 0
    bounds = [(0, 1) for _ in range(26)]
    
    # Constraints for non-overlapping and boundary
    def constraint_func(x, centers):
        radii = x
        for i in range(26):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return -1
            for j in range(i + 1, 26):

                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0
    
    # Run optimization
    result = minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints={"type": "ineq", "fun": lambda x: constraint_func(x, initial_centers)},
        method="SLSQP"
    )
    
    centers = initial_centers
    radii = result.x
    sum_radii = np.sum(radii)
    
    return (centers, radii, sum_radii)
