import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    # Hexagonal packing with 26 circles
    # First row: 1 circle
    # Second row: 2 circles
    # Third row: 3 circles
    # Fourth row: 4 circles
    # Fifth row: 5 circles
    # Sixth row: 5 circles (max 5 in a row for unit square)
    
    # Initial positions in hexagonal grid
    def initial_hex_grid(num_rows):
        rows = []
        for i in range(num_rows):
            y = i * np.sqrt(3) / 2
            if i % 2 == 0:
                x_start = 0.5
                step = 1.0
            else:
                x_start = 0.0
                step = 1.0
            x_positions = np.arange(x_start, 1.0, step)
            for x in x_positions:
                rows.append((x, y))
        return np.array(rows)

    centers = initial_hex_grid(6)
    # Scale to fit in unit square
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))
    n = len(centers)
    # Initial uniform radii
    radii = np.ones(n) * 0.1
    
    # Optimization
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum
    
    def constraints(radii):
        constraints = []
        for i in range(n):
            constraints.append((radii[i] >= 0))
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                constraints.append((radii[i] + radii[j] <= dist + 1e-6))
        return constraints
    
    bounds = [(0, 1) for _ in range(n)]
    
    result = minimize(objective, radii, bounds=bounds, constraints=constraints)
    radii = result.x
    
    # Compute sum of radii
    sum_radii = np.sum(radii)
    
    # Ensure all circles are within the square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-6 or x + r > 1 + 1e-6 or
            y - r < -1e-6 or y + r > 1 + 1e6):
            radii[i] = 0
    
    return (centers, radii, sum_radii)
