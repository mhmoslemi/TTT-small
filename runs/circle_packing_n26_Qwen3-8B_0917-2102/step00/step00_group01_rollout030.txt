import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize

def run_packing():
    # Define initial staggered grid with 5 rows and 5 columns with spacing
    num_circles = 26
    rows = 5
    cols = 5 + 1  # Add one extra circle for better utilization

    # Initial grid positions
    centers = np.zeros((num_circles, 2))
    for i in range(rows):
        for j in range(cols):
            row = i
            col = j
            x = col / (cols - 1)
            y = i / (rows - 1)
            centers[i * cols + j] = [x, y]

    # Precompute distances between all pairs
    dists = np.zeros((num_circles, num_circles))
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            dists[i, j] = np.sqrt(np.sum((centers[i] - centers[j])**2))
            dists[j, i] = dists[i, j]

    # Initial guess for radii: uniform size based on minimum distance
    min_dist = np.min(dists[np.triu_indices(num_circles, 1)])
    radii = np.full(num_circles, min_dist / 2)

    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize

    # Constraints: non-overlapping, within square, radii >= 0
    def constraint(radii):
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                if np.sqrt(np.sum((centers[i] - centers[j])**2)) < radii[i] + radii[j] - 1e-8:
                    return -1
        for i in range(num_circles):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8 or y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8:
                return -1
        return 0

    # Optimization
    bounds = [(0, 1) for _ in range(num_circles)]
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraint})

    # Check result
    if not result.success:
        raise RuntimeError("Optimization failed")

    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
