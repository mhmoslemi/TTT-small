import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import matplotlib.pyplot as plt

def run_packing():
    n = 26
    # Initial hexagonal packing with 6 layers
    layers = 6
    centers = []
    radii = []

    # First layer: 1 circle in center
    centers.append([0.5, 0.5])
    radii.append(0.05)

    # Second to sixth layers: hexagonal rings
    for layer in range(1, layers + 1):
        count = 6 * layer
        for i in range(count):
            angle = 2 * np.pi / count * i
            r = layer * 0.5 * (1 - 0.5 / (layer + 1))
            x = 0.5 + r * np.cos(angle)
            y = 0.5 + r * np.sin(angle)
            centers.append([x, y])
            radii.append(0.05)

    # Final optimization
    def objective(radii_vec):
        return -np.sum(radii_vec)

    def constraints(radii_vec):
        return np.array(radii_vec) >= 0

    def bounds(radii_vec):
        return [(0, 1) for _ in radii_vec]

    def non_overlap_constr(x):
        r = np.array(x)
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    res = minimize(
        objective,
        np.array(radii),
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': non_overlap_constr}
        ],
        method='SLSQP'
    )

    return np.array(centers), np.array(res.x), np.sum(res.x)
