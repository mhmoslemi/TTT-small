import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds

def run_packing():
    # Initial estimate for circle positions using hexagonal packing
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Hexagonal grid with vertical spacing ~sqrt(3)/2 of horizontal spacing
    horizontal_spacing = 0.4
    vertical_spacing = np.sqrt(3)/2 * horizontal_spacing
    
    # Distribute circles in hexagonal grid with 5 rows and 6 columns
    for i in range(5):
        for j in range(6):
            x = j * horizontal_spacing
            y = i * vertical_spacing if i % 2 == 0 else (i + 0.5) * vertical_spacing
            if x + 0.5 * horizontal_spacing <= 1 and y + 0.5 * vertical_spacing <= 1:
                centers[i * 6 + j] = [x, y]
    
    # Optimize radii using constrained optimization
    def objective(r):
        return -np.sum(r)  # Maximize sum of radii
    
    def constraint1(r):
        return np.sum(r)  # Ensure radii are non-negative
    
    def constraint2(r):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < r[i] + r[j] - 1e-8:
                    return -np.inf
        return 0
    
    cons = (
        {'type': 'ineq', 'fun': constraint2},
        {'type': 'ineq', 'fun': constraint1}
    )
    
    bounds = Bounds(0, 1)
    
    result = optimize.minimize(
        objective, 
        np.ones(n), 
        method='SLSQP', 
        bounds=bounds,
        constraints=cons
    )
    
    radii = result.x
    sum_radii = np.sum(radii)
    
    # Adjust positions to stay within unit square
    for i in range(n):
        r = radii[i]
        x, y = centers[i]
        if x - r < 0:
            centers[i, 0] = r
        elif x + r > 1:
            centers[i, 0] = 1 - r
        if y - r < 0:
            centers[i, 1] = r
        elif y + r > 1:
            centers[i, 1] = 1 - r
    
    return centers, radii, sum_radii
