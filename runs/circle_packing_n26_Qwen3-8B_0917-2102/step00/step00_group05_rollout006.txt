import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    # Define number of circles
    n = 26

    # Initial guess: rows of circles with spacing
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # First row of circles
    for i in range(13):
        centers[i, 0] = i / 12
        centers[i, 1] = 0.05
        radii[i] = 0.05

    # Second row of circles
    for i in range(13):
        centers[13 + i, 0] = i / 12
        centers[13 + i, 1] = 0.5
        radii[13 + i] = 0.15

    # Third row of circles
    for i in range(13):
        centers[26 + i, 0] = i / 12
        centers[26 + i, 1] = 0.95
        radii[26 + i] = 0.05

    # Define objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization

    # Define constraints for non-overlapping
    def constraint(radii):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return 1
        return 0

    # Bounds for radii
    bounds = [(0, 1) for _ in range(n)]

    # Optimization
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=[{'type': 'ineq', 'fun': constraint}])

    # Check if the optimization succeeded
    if not result.success:
        raise RuntimeError("Optimization failed")

    # Final centers and radii
    final_radii = result.x
    final_centers = centers.copy()

    # Ensure circles are within square
    for i in range(n):
        x, y = final_centers[i]

        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            final_radii[i] = min(r, 1 - np.min([x, 1 - x]))

    return (final_centers, final_radii, np.sum(final_radii))
