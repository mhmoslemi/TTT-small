import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial guess for centers in a hexagonal grid
    centers = []
    radii = []
    for row in range(5):
        for col in range(5 - row % 2):
            x = col + 0.5 * row / 2
            y = row + 0.5 * row / 2
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
    centers = np.array(centers)
    n = len(centers)
    radii = np.full(n, 0.1)

    # Objective function
    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii

    # Constraints: non-overlapping and boundary
    def constraint(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
                return -1
        return 0

    # Optimization
    bounds = [(0, 0.5) for _ in range(n)]
    result = minimize(objective, radii, bounds=bounds, constraints={"type": "soft", "fun": constraint, "jac": None})
    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
