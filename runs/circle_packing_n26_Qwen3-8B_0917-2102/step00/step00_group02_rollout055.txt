import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        return np.sum(radii)

    def inequality_constraints(radii):
        return np.array([radii[i] >= 0 for i in range(26)])

    def non_overlap_constraints(radii, centers):
        dist_matrix = squareform(pdist(centers))
        return np.array([dist_matrix[i, j] >= radii[i] + radii[j] - 1e-8 for i in range(26) for j in range(i+1, 26)])

    def boundary_constraints(radii, centers):
        return np.array([centers[i, 0] - radii[i] >= 0 for i in range(26)] + 
                       [centers[i, 0] + radii[i] <= 1 for i in range(26)] +
                       [centers[i, 1] - radii[i] >= 0 for i in range(26)] +
                       [centers[i, 1] + radii[i] <= 1 for i in range(26)])

    # Initial guess for centers in hexagonal packing
    def hexagonal_centers(n):
        rows = int(np.ceil(n / 6))
        cols = int(np.ceil(n / rows))
        centers = np.zeros((n, 2))
        idx = 0
        for row in range(rows):
            for col in range(cols):
                x = col + (row % 2) * 0.5
                y = row + 0.5
                if idx < n:
                    centers[idx] = [x, y]
                    idx += 1
        return centers

    centers = hexagonal_centers(26)
    radii = np.full(26, 0.1)

    # Optimization
    bounds = [(0, 1) for _ in range(26 * 2)]
    constraints_list = [
        {'type': 'ineq', 'fun': lambda r: np.sum(r)},
        {'type': 'ineq', 'fun': lambda r: np.array([r[i] >= 0 for i in range(26)])}
    ]

    result = minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints=constraints_list,
        method='SLSQP'
    )

    if not result.success:
        raise RuntimeError("Optimization failed")

    radii = result.x
    centers = hexagonal_centers(26)
    for i in range(26):
        if centers[i, 0] - radii[i] < 0:
            radii[i] = centers[i, 0]
        if centers[i, 0] + radii[i] > 1:
            radii[i] = 1 - centers[i, 0]
        if centers[i, 1] - radii[i] < 0:
            radii[i] = centers[i, 1]
        if centers[i, 1] + radii[i] > 1:
            radii[i] = 1 - centers[i, 1]

    return (centers, radii, np.sum(radii))
