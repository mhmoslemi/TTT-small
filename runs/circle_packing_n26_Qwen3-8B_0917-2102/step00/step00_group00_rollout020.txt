import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal packing with 5 rows
    centers = []
    radii = []
    for row in range(5):
        for col in range(5 if row % 2 == 0 else 6):
            x = col + 0.5 * (row % 2)
            y = row + 0.5
            centers.append([x, y])
    centers = np.array(centers)
    radii = np.array([0.1] * n)
    
    def objective(r):
        return -np.sum(r)
    
    def constraint(r):
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < r[i] + r[j] - 1e-12:
                    return -np.inf
        for i in range(n):
            x, y = centers[i]
            if x - r[i] < -1e-12 or x + r[i] > 1 + 1e-12 or y - r[i] < -1e-12 or y + r[i] > 1 + 1e-12:
                return -np.inf
        return 0
    
    res = minimize(objective, radii, method='SLSQP', constraints=[{'type': 'ineq', 'fun': constraint}])
    
    return (centers, res.x, np.sum(res.x))
