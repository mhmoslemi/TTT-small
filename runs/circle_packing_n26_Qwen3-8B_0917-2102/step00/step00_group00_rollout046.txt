import numpy as np
import scipy.optimize as opt
import scipy.spatial as spatial

def run_packing():
    # Initial grid-based centers for hexagonal packing
    n = 26
    centers = []
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        centers.append([x, y])
    centers = np.array(centers)
    
    # Initial radii based on minimum distance to nearest neighbors
    dists = spatial.distance.cdist(centers, centers)
    min_dist = np.min(dists[dists > 1e-12])
    radii = np.full(n, min_dist / 2)
    
    # Optimize radii to maximize sum while maintaining non-overlap and boundary constraints
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum
    
    def constraints(radii):
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12 or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True
    
    bounds = [(0, 1) for _ in range(n)]
    
    result = opt.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=opt.Constraint(lambda r: constraints(r), {}),
        method='SLSQP'
    )
    
    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
