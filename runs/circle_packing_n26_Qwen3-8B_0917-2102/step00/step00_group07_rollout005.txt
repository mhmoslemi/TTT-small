import numpy as np
from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d

def run_packing():
    np.random.seed(42)
    n = 26
    
    # Initial guess: hexagonal grid with small circles
    base_radius = 0.05
    grid = []
    for i in range(2):
        for j in range(2):
            grid.append((i*0.5, j*0.5))
    grid += [(x, y) for x in np.linspace(0.25, 0.75, 5) for y in [0.25, 0.75]]
    centers = np.array(grid[:n])
    radii = np.full(n, base_radius)
    
    # Optimization
    def objective(radii):
        return -np.sum(radii)
    
    def constraints(radii):
        cons = []
        for i in range(n):
            cons.append((radii[i], 0, 100))
        return cons
    
    bounds = [(0, 100) for _ in range(n)]
    
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=constraints)
    
    # Final refinement by adjusting for boundary and overlap constraints
    final_centers = centers.copy()
    final_radii = result.x
    
    # Boundary constraint enforcement
    for i in range(n):
        x, y = final_centers[i]
        r = final_radii[i]
        if x - r < 0:
            final_radii[i] = x
        elif x + r > 1:
            final_radii[i] = 1 - x
        if y - r < 0:
            final_radii[i] = y
        elif y + r > 1:
            final_radii[i] = 1 - y
    
    # Final overlap check
    for i in range(n):
        for j in range(i + 1, n):
            dx = final_centers[i, 0] - final_centers[j, 0]
            dy = final_centers[i, 1] - final_centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < final_radii[i] + final_radii[j] - 1e-6:
                final_radii[i] = (dist - final_radii[j]) / 2
                final_radii[j] = (dist - final_radii[i]) / 2
    
    sum_radii = np.sum(final_radii)
    return (final_centers, final_radii, sum_radii)
