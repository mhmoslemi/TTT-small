import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial guess for centers using hexagonal packing
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    row = 0
    col = 0
    for i in range(n):
        if i % 2 == 0:
            centers[i] = [col / (n - 1), row / (n - 1)]
        else:
            centers[i] = [(col + 0.5) / (n - 1), row / (n - 1)]
        col += 1
        if col > (n - 1) / 2:
            row += 1
            col = 0

    # Optimization parameters
    bounds = [(0, 1) for _ in range(2 * n)]
    constraints = []

    # Boundary constraint: circle must be within [0,1]^2
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i] - x[2*i + n]})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i + n] - x[2*i]})

    # Non-overlap constraints between circles
    for i in range(n):
        for j in range(i + 1, n):
            def constraint(x, i=i, j=j):
                dx = x[2*i] - x[2*j]
                dy = x[2*i + n] - x[2*j + n]
                dist = np.sqrt(dx**2 + dy**2)
                r_i = x[2*i + n]
                r_j = x[2*j + n]
                return dist - r_i - r_j
            constraints.append({'type': 'ineq', 'fun': constraint})

    # Objective function: maximize sum of radii
    def objective(x):
        return -sum(x[2*i + n] for i in range(n))

    result = minimize(objective, x0=np.concatenate((centers.flatten(), np.ones(n))), bounds=bounds, constraints=constraints)

    centers_opt = result.x[:2*n].reshape(n, 2)
    radii_opt = result.x[2*n:]

    # Ensure non-negative radii and boundary constraints
    radii_opt = np.clip(radii_opt, 0, 1)
    for i in range(n):
        x, y = centers_opt[i]
        r = radii_opt[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            radii_opt[i] = 0

    return centers_opt, radii_opt, np.sum(radii_opt)
