import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    n = 26

    # Initial guess for circle centers using hexagonal packing
    centers = np.zeros((n, 2))
    r = np.zeros(n)
    for i in range(n):
        row = i // 5
        col = i % 5
        centers[i, 0] = 0.1 + col * 0.2
        centers[i, 1] = 0.1 + row * 0.25

    # Initial radii guess (will be optimized)
    r = np.ones(n) * 0.1

    # Objective function to maximize sum of radii
    def objective(x):
        r = x[:n]
        return -np.sum(r)  # Minimize negative sum to maximize

    # Constraint: circles must not overlap
    def constraint(x):
        r = x[:n]
        dist_matrix = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    # Optimization bounds: radii >= 0
    bounds = Bounds(np.zeros(n), np.inf)

    # Run optimization
    result = minimize(objective, x0=r, bounds=bounds, constraints={"type": "ineq", "fun": constraint})

    # Extract results
    if result.success:
        best_r = result.x
        sum_radii = np.sum(best_r)
        return (centers, best_r, sum_radii)
    else:
        raise ValueError("Optimization failed: " + result.message)
