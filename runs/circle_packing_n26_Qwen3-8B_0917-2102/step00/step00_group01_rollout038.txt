import numpy as np
from scipy.optimize import minimize

def run_packing():
    np.random.seed(42)
    n = 26

    # Initial hexagonal packing with spacing and boundary adjustments
    def hex_grid(x_start, y_start, spacing, rows, cols):
        centers = []
        for i in range(rows):
            for j in range(cols):
                x = x_start + j * spacing
                y = y_start + i * spacing * np.sqrt(3) / 2
                centers.append([x, y])
        return np.array(centers)

    # Initial grid with spacing and boundaries
    rows = 5
    cols = 5
    spacing = 0.25
    grid_centers = hex_grid(0.25, 0.25, spacing, rows, cols)

    # Add circles near corners
    corner_radius = 0.1
    corners = np.array([
        [0.1, 0.1],
        [0.9, 0.1],
        [0.1, 0.9],
        [0.9, 0.9]
    ])
    grid_centers = np.vstack((grid_centers, corners))

    # Initial guess for radii
    radii = np.full(n, 0.1)

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        constraints = []
        for i in range(n):
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] + 1e-12})
        for i in range(n):
            for j in range(i+1, n):
                def dist_func(r):
                    return np.sqrt(np.sum((grid_centers[i] - grid_centers[j])**2)) - (r[i] + r[j] - 1e-12)
                constraints.append({'type': 'ineq', 'fun': dist_func})
        return constraints

    result = minimize(objective, radii, method='SLSQP', constraints=constraints)

    centers = grid_centers
    radii = result.x
    sum_radii = np.sum(radii)

    # Final check and adjustment
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            radii[i] = min(r, 1 - np.min([x, 1 - x, y, 1 - y]))

    return (centers, radii, np.sum(radii))
