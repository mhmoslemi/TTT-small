import numpy as np
from scipy.optimize import minimize

def run_packing():
    def objective(x):
        return -np.sum(x[26:])  # Maximize sum of radii

    def constraints(x):
        centers = x[:52].reshape(26, 2)
        radii = x[52:]
        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        for i in range(26):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < -1e-8 or x_i + r_i > 1 + 1e-8:
                return False
            if y_i - r_i < -1e-8 or y_i + r_i > 1 + 1e-8:
                return False
        return True

    def bounds(x):
        centers = x[:52].reshape(26, 2)
        radii = x[52:]
        for i in range(26):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i < 0 - r_i or x_i > 1 + r_i:
                return False
            if y_i < 0 - r_i or y_i > 1 + r_i:
                return False
        return True

    initial_centers = np.random.rand(26, 2)
    initial_radii = np.random.rand(26) * 0.1
    x_initial = np.concatenate((initial_centers.ravel(), initial_radii))

    res = minimize(objective, x_initial, method='SLSQP', bounds=[(-1, 1) for _ in range(52 + 26)], constraints={"type": "ineq", "fun": constraints}, options={'ftol': 1e-10, 'maxiter': 1000})

    centers = res.x[:52].reshape(26, 2)
    radii = res.x[52:]
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
