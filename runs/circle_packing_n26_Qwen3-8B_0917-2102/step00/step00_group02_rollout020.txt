import numpy as np
import scipy.optimize as opt

def run_packing():

    # 1 row of 1 circle, 2 rows of 2 circles, 3 rows of 3 circles, 4 rows of 4 circles
    # Total: 1 + 2 + 3 + 4 + 5 + 6 = 21, need 5 more to reach 26
    # Add 5 more circles in denser clusters

    # Define initial positions
    centers = []
    radii = []

    # First 21 circles in hexagonal grid

            y = (2 * np.sqrt(3) * j) / (i + 1)
            centers.append([x, y])

    # Add 5 more circles in dense clusters

    centers.append([0.2, 0.3])
    radii.append(0.07)
    # Cluster 2
    centers.append([0.8, 0.3])
    radii.append(0.07)
    # Cluster 3
    centers.append([0.5, 0.7])
    radii.append(0.07)
    # Cluster 4
    centers.append([0.1, 0.6])
    radii.append(0.07)
    # Cluster 5
    centers.append([0.9, 0.6])
    radii.append(0.07)

    centers = np.array(centers)
    radii = np.array(radii)

    # Optimization function to maximize sum of radii with non-overlapping constraints
    def objective(r):
        return -np.sum(r)

    def constraints(r):
        # Boundary constraint
        for i in range(len(r)):
            x, y = centers[i]
            if x - r[i] < -1e-12 or x + r[i] > 1 + 1e-12 or y - r[i] < -1e-12 or y + r[i] > 1 + 1e-12:
                return False
        # Pairwise distance constraint
        for i in range(len(r)):
            for j in range(i + 1, len(r)):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - 1e-12:
                    return False
        return True

    # Optimize with bounds and constraints
    bounds = [(0, 1) for _ in range(len(radii))]
    result = opt.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[
            {'type': 'constraint', 'fun': constraints, 'jac': None}
        ]
    )

    # Return the optimized centers, radii, and sum
    return (centers, result.x, np.sum(result.x))
