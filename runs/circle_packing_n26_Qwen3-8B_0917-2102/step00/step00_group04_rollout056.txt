import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Initial hexagonal packing with 26 circles
    n = 26
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    
    # Generate centers in hexagonal grid
    centers = np.zeros((n, 2))
    r = 0.1  # Initial radius
    for i in range(n):
        row = i // cols
        col = i % cols
        centers[i, 0] = 0.5 + (col - (cols - 1)/2) * (1 / (cols - 1))
        centers[i, 1] = 0.5 + (row - (rows - 1)/2) * (1 / (rows - 1))
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraints(radii):
        # Boundary constraint
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return False
        # Pairwise distance constraint
        dists = cdist(centers, centers)
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return False
        return True
    
    # Optimize radii
    bounds = [(0, 1) for _ in range(n)]
    result = minimize(
        objective,
        x0=np.full(n, r),
        bounds=bounds,
        constraints=[{'type': 'custom', 'fun': constraints}]
    )
    
    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
