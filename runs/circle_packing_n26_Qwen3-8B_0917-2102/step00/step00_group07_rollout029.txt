```python
import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    # Initial guess for centers in a hexagonal packing
    def initial_hex_grid(n):
        rows = int(np.ceil(n / 6))
        cols = int(np.ceil(n / rows))
        centers = []
        for i in range(rows):
            for j in range(cols):
                x = j + 0.5 * (i % 2)
                y = i * np.sqrt(3) / 2
                centers.append([x, y])
        return np.array(centers[:n])

    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraint function for non-overlapping circles
    def constraint(radii, centers):
        n = len(radii)
        dists = squareform(pdist(centers))
        return np.sum((dists - (radii[:, np.newaxis] + radii[np.newaxis, :])) < 1e-8)

    # Boundary constraint: circles must be inside unit square
    def boundary(radii, centers):
        n = len(radii)
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
                return -1
        return 0

    # Initial centers with hexagonal packing
    centers = initial_hex_grid(26)
    radii = np.full(26, 0.1)

    # Optimization with constraints
    def constraint_func(x):
        return constraint(x, centers)

    def boundary_func(x):
        return boundary(x, centers)

    bounds = [(0, 2) for _ in range(26)]

    result = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_func},
            {'type': 'ineq', 'fun': boundary_func}
        ]
    )

    if not result.success:
        raise ValueError("Optimization failed")

    return centers, result.x, np.sum(result.x)
```
