import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    # Initial hexagonal packing with adjusted positions to fit unit square
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Hexagonal packing grid with adjusted spacing
    rows = 5
    cols = 6
    spacing = 1.0 / np.sqrt(3)  # Spacing between centers in hexagonal packing
    scale = 1.0 / (np.sqrt(3) * (cols - 1))

    for i in range(rows):
        for j in range(cols):
            x = j * spacing * scale
            y = i * spacing * scale
            if i % 2 == 1:
                x += spacing * scale / 2
            centers[i * cols + j] = np.array([x, y])

    # Adjust outer circles to fit within unit square
    for i in range(num_circles):
        x, y = centers[i]
        if x > 1 - 1e-8:
            centers[i, 0] = 1 - 1e-8
        if y > 1 - 1e-8:
            centers[i, 1] = 1 - 1e-8
        if x < 1e-8:
            centers[i, 0] = 1e-8
        if y < 1e-8:
            centers[i, 1] = 1e-8

    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraints: circles must not overlap and stay within square
    def constraint_func(radii):
        dist = squareform(pdist(centers))
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                if dist[i, j] < radii[i] + radii[j] - 1e-8:
                    return -1
        for i in range(num_circles):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-8 or x + r > 1 + 1e-8 or
                y - r < -1e-8 or y + r > 1 + 1e-8):
                return -1
        return 0

    # Initial guess for radii
    initial_r = np.min([np.min(centers, axis=0) * 0.5, np.max(centers, axis=0) * 0.5])
    initial_radii = np.full(num_circles, initial_r)

    # Optimization
    cons = ({'type': 'ineq', 'fun': constraint_func, 'args': ()})
    result = minimize(objective, initial_radii, method='SLSQP', constraints=cons)

    radii = result.x
    sum_radii = np.sum(radii)

    # Final validation
    assert np.all(radii >= 0), "Negative radius found"
    assert np.all(np.array([np.min(centers, axis=0) - radii, np.max(centers, axis=0) - radii]) >= -1e-8), "Radius exceeds square boundaries"
    assert np.all(np.array([pdist(centers) - (radii.reshape(-1, 1) + radii.reshape(1, -1))]) >= -1e-8), "Circle overlap detected"

    return centers, radii, sum_radii
