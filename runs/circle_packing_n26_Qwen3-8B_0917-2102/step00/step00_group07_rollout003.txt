import numpy as np
import scipy.optimize as opt

def run_packing():
    np.random.seed(42)
    
    # Initial hexagonal packing with 26 circles
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Create a hexagonal grid with 5 rows of circles
    cols = 5
    rows = 6
    spacing = 1.0 / (cols - 1)
    
    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * (np.sqrt(3)/2) * spacing
            centers[i * cols + j] = [x, y]
            radii[i * cols + j] = 0.1
    
    # Refine positions using gradient descent to maximize sum of radii
    def objective(radii_vec):
        return -np.sum(radii_vec)
    
    def constraints(radii_vec):
        r = radii_vec.reshape(-1)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < r[i] + r[j] - 1e-12:
                    return False
        for i in range(n):
            x, y = centers[i]
            if x - r[i] < -1e-12 or x + r[i] > 1 + 1e-12 or y - r[i] < -1e-12 or y + r[i] > 1 + 1e-12:
                return False
        return True
    
    bounds = [(0, 1) for _ in range(n)]
    
    result = opt.minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraints, 'jac': None}
    )
    
    radii = result.x
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
