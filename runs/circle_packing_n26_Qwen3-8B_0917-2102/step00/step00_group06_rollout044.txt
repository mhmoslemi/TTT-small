from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d

def run_packing():
    def objective(radii):

    def constraints(radii):
        return np.sum(radii) - 2.636

    def bounds(radii):
        return [(0, 1) for _ in radii]

    def packing_func(centers, radii):
        for i in range(len(centers)):
            x, y = centers[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12:
                return False
            if y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return False
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    def initial_centers():
        grid = np.linspace(0.1, 0.9, 5)
        centers = []
        for i in range(5):
            for j in range(5):
                x = grid[i]
                y = grid[j]
                centers.append([x, y])
        return np.array(centers)

    centers = initial_centers()
    radii = np.array([0.05] * len(centers))

    bounds = [(0, 1) for _ in radii]
    constraints = [{'type': 'ineq', 'fun': lambda r: np.sum(r) - 2.636}]
    result = minimize(objective, radii, bounds=bounds, constraints=constraints, method='SLSQP')

    if not result.success:
        raise ValueError("Optimization failed")

    radii = result.x
    return centers, radii, np.sum(radii)
