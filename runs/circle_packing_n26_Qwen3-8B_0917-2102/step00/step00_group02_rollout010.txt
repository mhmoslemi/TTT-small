import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.cluster.vq import kmeans

def run_packing():
    np.random.seed(42)
    num_circles = 26
    initial_centers = np.random.rand(num_circles, 2)
    initial_radii = np.random.rand(num_circles) * 0.1

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        dist_matrix = squareform(pdist(initial_centers))
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                if dist_matrix[i, j] < radii[i] + radii[j] + 1e-8:
                    return False
        for i in range(num_circles):
            if initial_centers[i, 0] - radii[i] < 0 or initial_centers[i, 0] + radii[i] > 1:
                return False
            if initial_centers[i, 1] - radii[i] < 0 or initial_centers[i, 1] + radii[i] > 1:
                return False
        return True

    result = minimize(
        objective,
        initial_radii,
        method='SLSQP',
        jac=None,
        bounds=[(0, 1)] * num_circles,
        constraints=[{'type': 'ineq', 'fun': constraints}]
    )

    centers = initial_centers.copy()
    radii = result.x
    sum_radii = np.sum(radii)

    return (centers, radii, sum_radii)
