from scipy.optimize import minimize

    cols = 5

        x_vec = x[:26]
        r_vec = x[26:]
        return -np.sum(r_vec)

        r_vec = x[26:]
        for i in range(26):

            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return 1

                dist = np.sqrt(dx**2 + dy**2)
                if dist < r_vec[i] + r_vec[j] - 1e-12:
                    return 1
        return 0
    
    # Initial guess with equal radii
    x0 = np.concatenate((centers.flatten(), np.ones(26) * 0.1))
    
    res = minimize(objective, x0, constraints={"type": "ineq", "fun": constraints}, method="SLSQP")
