from scipy.optimize import Bounds

def run_packing():

    # Initial guess for centers using hexagonal packing
    centers = np.zeros((n, 2))
    r = 0.1
    for i in range(n):
        row = i // 7
        col = i % 7
        centers[i, 0] = col * (1 + r) / 7 + r
        centers[i, 1] = row * (1 + r) / 4 + r

            constraints.append((centers[i, 1] - radii[i] >= 0))

        return constraints

    x0 = radii
    bounds = Bounds(0, 1)
    cons = [{'type': 'ineq', 'fun': lambda r, i=i: r[i] - 0} for i in range(n)]

    for i in range(n):
