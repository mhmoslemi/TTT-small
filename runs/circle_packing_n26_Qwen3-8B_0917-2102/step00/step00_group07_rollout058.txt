import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Define initial hexagonal packing with 26 circles
    # Rows: 5 rows with 5, 6, 5, 6, 5 circles
    # Vertical spacing = sqrt(3)/2 * diameter
    # Horizontal spacing = diameter
    # Initial radius guess: 0.08
    # Adjust radii to fit within unit square with non-overlapping constraint

    n = 26
    centers = np.zeros((n, 2))
    radii = np.full(n, 0.08)

    # Hexagonal packing with 5 rows
    row_height = np.sqrt(3)/2
    row_offset = row_height / 2
    horizontal_spacing = 1.0
    vertical_spacing = row_height

    for i in range(5):
        row = i
        col_start = i % 2
        max_cols = 5 if i % 2 == 0 else 6
        for j in range(max_cols):
            x = j * horizontal_spacing + (col_start + 0.5) * horizontal_spacing
            y = row * vertical_spacing + row_offset
            centers[i * max_cols + j] = [x, y]

    # Define objective function to maximize total radius
    def objective(radii):
        return -np.sum(radii)

    # Define constraints: circles stay within square, no overlap
    def constraint1(radii):
        for i in range(n):
            r = radii[i]
            x, y = centers[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return -1
        return 0

    def constraint2(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0

    # Use SLSQP to optimize radii
    bounds = [(0.0, 0.1) for _ in range(n)]
    cons = [{'type': 'ineq', 'fun': lambda r: constraint1(r)}, {'type': 'ineq', 'fun': lambda r: constraint2(r)}]
    result = optimize.minimize(objective, radii, bounds=bounds, constraints=cons, method='SLSQP')

    return (centers, result.x, np.sum(result.x))
