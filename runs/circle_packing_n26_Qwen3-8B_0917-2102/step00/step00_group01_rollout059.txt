import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance

def run_packing():
    # Initial guess: hexagonal packing with boundary circles
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Place 6 boundary circles
    for i in range(6):
        angle = 2 * np.pi / 6 * i
        r = 0.3
        centers[i] = np.array([np.cos(angle), np.sin(angle)]) * (1 - 2 * r) + r
        radii[i] = r

    # Place inner circles in hexagonal grid
    for i in range(6, n):
        row = (i - 6) // 7
        col = (i - 6) % 7
        r = 0.15
        centers[i] = np.array([0.5 + col * (2 * r), 0.5 + row * (np.sqrt(3) * r)])
        radii[i] = r

    # Optimization
    def objective(radii_opt):
        return -np.sum(radii_opt)

    def constraint(radii_opt):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_opt[i] + radii_opt[j] - 1e-8:
                    return -1
        return 0

    # Initialize optimization
    bounds = [(0, 0.5) for _ in range(n)]
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[{"type": "ineq", "fun": constraint}]
    )

    # Final centers and radii
    return (centers, result.x, np.sum(result.x))
