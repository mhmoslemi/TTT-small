import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance_matrix

def run_packing():
    np.random.seed(42)
    n = 26
    # Initial hexagonal grid with some boundary adjustments
    centers = np.array([
        [0.1, 0.1],
        [0.3, 0.1],
        [0.5, 0.1],
        [0.7, 0.1],
        [0.9, 0.1],
        [0.1, 0.3],
        [0.3, 0.3],
        [0.5, 0.3],
        [0.7, 0.3],
        [0.9, 0.3],
        [0.1, 0.5],
        [0.3, 0.5],
        [0.5, 0.5],
        [0.7, 0.5],
        [0.9, 0.5],
        [0.1, 0.7],
        [0.3, 0.7],
        [0.5, 0.7],
        [0.7, 0.7],
        [0.9, 0.7],
        [0.1, 0.9],
        [0.3, 0.9],
        [0.5, 0.9],
        [0.7, 0.9],
        [0.9, 0.9]
    ])
    radii = np.full(n, 0.05)

    def objective(r):
        return -np.sum(r)

    def constraints(r):
        r = r.reshape(n)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < r[i] + r[j] - 1e-8:
                    return False
        for i in range(n):
            x, y = centers[i]
            if x - r[i] < -1e-8 or x + r[i] > 1 + 1e-8 or y - r[i] < -1e-8 or y + r[i] > 1 + 1e-8:
                return False
        return True

    result = minimize(
        objective,
        radii,
        method='SLSQP',
        constraints={'type': 'constraint', 'fun': constraints},
        options={'maxiter': 1000}
    )

    final_radii = result.x
    final_centers = centers

    sum_radii = np.sum(final_radii)
    return (final_centers, final_radii, sum_radii)
