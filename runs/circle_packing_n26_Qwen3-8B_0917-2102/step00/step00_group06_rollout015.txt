import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal grid configuration
    rows = 5
    cols = 6
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Generate initial hexagonal grid
    for i in range(rows):
        for j in range(cols):
            idx = i * cols + j
            if i % 2 == 0:
                x = j / (cols - 1)
                y = i / (rows - 1)
            else:
                x = (j + 0.5) / (cols - 1)
                y = i / (rows - 1)
            centers[idx] = [x, y]
    
    # Initial guess for radii
    radii = np.full(n, 0.1)
    
    # Define objective function to maximize sum of radii
    def objective(r):
        return -np.sum(r)  # Negative for minimization
    
    # Define constraints: non-overlapping and within square
    def constraint(r, c):
        dists = squareform(pdist(c))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r)
    
    # Optimization
    res = minimize(
        objective,
        radii,
        args=(),
        method='SLSQP',
        constraints=[
            {
                'type': 'ineq',
                'fun': lambda r: np.sum(r),
                'jacobian': lambda r: np.ones(n)
            },
            {
                'type': 'ineq',
                'fun': lambda r: np.sum([np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2) - (r[i] + r[j]) for i in range(n) for j in range(i + 1, n)]),
                'jacobian': lambda r: np.zeros(n)
            }
        ],
        bounds=[(0, 1) for _ in range(n)]
    )
    
    radii = res.x
    sum_radii = np.sum(radii)
    
    # Check if centers are within square
    for i in range(n):
        if centers[i, 0] < -1e-8 or centers[i, 0] > 1 + 1e-8 or centers[i, 1] < -1e-8 or centers[i, 1] > 1 + 1e-8:
            raise ValueError("Some centers are outside unit square")
    
    return centers, radii, sum_radii
