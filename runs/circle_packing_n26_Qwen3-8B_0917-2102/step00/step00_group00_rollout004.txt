import numpy as np

def run_packing():
    # Hexagonal packing with 5 rows (alternating even and odd)
    # First row: 5 circles, next 4, then 5, then 4, then 5
    # Adjusted to fit in unit square with spacing and radii
    centers = []
    radii = []
    # Define base spacing and radius
    spacing = 0.35
    base_radius = 0.15

    # Row 1: 5 circles
    for i in range(5):
        x = i * spacing
        y = base_radius
        centers.append([x, y])
        radii.append(base_radius)

    # Row 2: 4 circles, offset by half spacing
    for i in range(4):
        x = i * spacing + spacing / 2
        y = base_radius + spacing * np.sqrt(3) / 2
        centers.append([x, y])
        radii.append(base_radius)

    # Row 3: 5 circles
    for i in range(5):
        x = i * spacing
        y = base_radius + spacing * np.sqrt(3)
        centers.append([x, y])
        radii.append(base_radius)

    # Row 4: 4 circles, offset
    for i in range(4):
        x = i * spacing + spacing / 2
        y = base_radius + spacing * np.sqrt(3) * 2
        centers.append([x, y])
        radii.append(base_radius)

    # Row 5: 5 circles
    for i in range(5):
        x = i * spacing
        y = base_radius + spacing * np.sqrt(3) * 3
        centers.append([x, y])
        radii.append(base_radius)

    # Adjusted radius for last row to fit in unit square
    last_row_radius = min(base_radius, 1 - y)
    for i in range(5):
        centers[-5 + i][1] = 1 - last_row_radius
        radii[-5 + i] = last_row_radius

    centers = np.array(centers)
    radii = np.array(radii)
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
