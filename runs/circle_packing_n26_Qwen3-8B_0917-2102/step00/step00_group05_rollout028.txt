import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    np.random.seed(42)
    n = 26

    # Initial hexagonal grid with 5 rows (5, 4, 5, 4, 3) for 26 circles
    rows = [5, 4, 5, 4, 3]
    cols = []
    for i, r in enumerate(rows):
        cols.append([0.5 * (j + 0.5) for j in range(r)])
    cols = np.array(cols)

    # Generate initial positions
    centers = []
    for i in range(len(rows)):
        for j in range(len(cols[i])):
            x = cols[i][j]
            y = 0.5 + i * (np.sqrt(3)/2)
            centers.append([x, y])
    centers = np.array(centers)

    # Initial radii estimate based on spacing
    def initial_radii(centers):
        dists = pdist(centers)
        min_dist = np.min(dists)
        max_rad = min_dist / 2
        return np.full(n, max_rad)

    radii = initial_radii(centers)

    # Objective function to maximize sum of radii
    def objective(x):
        r = x[0]
        dx = x[1:]
        radii = np.full(n, r)
        radii[0] = r + dx[0]
        for i in range(1, n):
            radii[i] = r + dx[i]
        return -np.sum(radii)

    # Constraints: circles do not overlap or go out of bounds
    def constraint(x, centers, radii):
        r = x[0]
        dx = x[1:]
        radii = np.full(n, r)
        radii[0] = r + dx[0]
        for i in range(1, n):
            radii[i] = r + dx[i]
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return 1
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8 or y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8:
                return 1
        return 0

    # Optimization
    bounds = [(0, 1)] * n
    result = minimize(lambda x: -objective(x), np.concatenate((np.array([0.5]), np.zeros(n-1))),
                      bounds=bounds, constraints=[{'type': 'ineq', 'fun': lambda x: -constraint(x, centers, radii)}])

    # Final centers and radii
    final_radii = np.full(n, result.x[0])
    final_radii[0] = result.x[0] + result.x[1]
    for i in range(1, n):
        final_radii[i] = result.x[0] + result.x[i+1]

    sum_radii = np.sum(final_radii)
    return centers, final_radii, sum_radii
