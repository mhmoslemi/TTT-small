import numpy as np
from scipy.optimize import minimize

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        return np.sum(radii)

    def boundary_constraints(radii):
        return np.zeros_like(radii)

    def circle_positions():
        centers = []
        radii = []
        # Hexagonal grid: 5 rows with 5, 5, 5, 5, 6 circles
        # First 5 rows: 5 circles each, total 25
        for i in range(5):
            for j in range(5):
                x = j + 0.5
                y = i + 0.5
                x = (x + 0.1) / 1.1
                y = (y + 0.1) / 1.1
                centers.append([x, y])
        # Add one more circle at the bottom right corner
        centers.append([0.9, 0.9])
        return np.array(centers)

    centers = circle_positions()
    n = centers.shape[0]
    radii = np.ones(n) * 0.05

    def constraint_func(radii):
        constraints = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            constraints.append(x - r)
            constraints.append(x + r - 1)
            constraints.append(y - r)
            constraints.append(y + r - 1)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                constraints.append(dist - radii[i] - radii[j])
        return np.array(constraints)

    bounds = [(0, 1) for _ in range(n)]

    res = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraint_func},
        method="SLSQP",
        options={"ftol": 1e-10, "maxiter": 1000}
    )

    return (res.x, np.ones_like(res.x), np.sum(res.x))
