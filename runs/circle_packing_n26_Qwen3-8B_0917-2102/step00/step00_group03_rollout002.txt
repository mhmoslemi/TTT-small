            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12 or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return False

            for j in range(i + 1, n):

                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    def update_centers(radii):
        nonlocal centers
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            while x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                if x - r < -1e-12:
                    x = r

                    x = 1 - r
                if y - r < -1e-12:
                    y = r
                elif y + r > 1 + 1e-12:
                    y = 1 - r
                centers[i] = [x, y]

    bounds = [(0, 1) for _ in range(2 * n)]
