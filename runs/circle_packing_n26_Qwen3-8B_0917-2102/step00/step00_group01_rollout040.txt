import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    np.random.seed(42)
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraints(radii, centers):
        cons = []
        for i in range(len(radii)):
            cons.append((radii[i], 0, 1))
        for i in range(len(radii)):
            for j in range(i+1, len(radii)):
                cons.append((radii[i] + radii[j], 0, np.sqrt(np.sum((centers[i] - centers[j])**2))))
        return cons
    
    def pack():
        centers = np.random.rand(26, 2)
        radii = np.random.rand(26) * 0.1
        for _ in range(10):
            for i in range(26):
                dists = np.min(np.linalg.norm(centers[i] - centers[np.arange(26)!=i], axis=1))
                radii[i] = np.min([radii[i], (dists - 1e-6)/2])
            for i in range(26):
                x, y = centers[i]
                radii[i] = np.min([radii[i], x, 1 - x, y, 1 - y])
        return centers, radii
    
    centers, radii = pack()
    
    def fitness(radii):
        return np.sum(radii)
    
    result = minimize(
        objective,
        radii,
        constraints=[
            {'type': 'ineq', 'fun': lambda r, i: r[i], 'args': (i,)}
            for i in range(26)
        ],
        method='SLSQP'
    )
    
    return centers, result.x, np.sum(result.x)
