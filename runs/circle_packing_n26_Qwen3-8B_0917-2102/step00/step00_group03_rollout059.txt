import numpy as np
from scipy.optimize import minimize

def run_packing():
    np.random.seed(42)
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n) * 0.1

    def objective(x):
        return -np.sum(x[1:])  # Maximize sum of radii

    def constraint(x):
        centers_flat = x[:2*n].reshape(n, 2)
        radii_flat = x[2*n:]
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return -np.inf
            for j in range(i + 1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return -np.inf
        return 0

    bounds = [(0, 1) for _ in range(2*n)] + [(0, 1) for _ in range(n)]
    result = minimize(objective, x0=np.concatenate([centers.flatten(), radii]), bounds=bounds, constraints={"type": "soft_bound", "fun": constraint})
    centers_opt = result.x[:2*n].reshape(n, 2)
    radii_opt = result.x[2*n:]
    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii
