import numpy as np
from scipy.optimize import minimize

def run_packing():

        return -np.sum(radii)  # Maximize sum of radii

    def constraints(centers, radii):
        constraints = []
        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            constraints.append((x - r >= 0) & (x + r <= 1) & (y - r >= 0) & (y + r <= 1))
        for i in range(26):
            for j in range(i+1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                constraints.append(dist >= radii[i] + radii[j])
        return constraints

    # Initial guess: hexagonal packing with 5 rows (1, 2, 3, 4, 5 circles)
    centers = []
    for row in range(5):
        for col in range(5 - row):
            x = col + 0.5 * row
            y = row + 0.5
            centers.append([x, y])
    for row in range(5):
        for col in range(5 - row - 1):
            x = col + 0.5 * (row + 1)
            y = row + 1.5
            centers.append([x, y])
    centers = np.array(centers)
    radii = np.full(26, 0.1)

    # Optimize
    bounds = [(0, 1) for _ in range(52)]
    res = minimize(objective, np.concatenate((centers.flatten(), radii)), bounds=bounds, constraints=constraints)

    centers_opt = res.x[:52].reshape(26, 2)
    radii_opt = res.x[52:]
    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii
