import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import random

def run_packing():
    np.random.seed(42)
    
    # Initial hexagonal grid with 5 rows
    rows = 5
    cols = 5
    centers = []
    radii = []
    
    # First row at top
    for i in range(cols):
        x = i / (cols - 1)
        y = 0.05
        centers.append([x, y])
        radii.append(0.05)
    
    # Middle rows
    for row in range(1, rows - 1):
        for i in range(cols):
            x = i / (cols - 1)
            y = 0.1 + row * (0.8 / (rows - 1))
            centers.append([x, y])
            radii.append(0.05)
    
    # Bottom row
    for i in range(cols):
        x = i / (cols - 1)
        y = 0.95
        centers.append([x, y])
        radii.append(0.05)
    
    # Add extra circles in gaps
    extra_circles = 26 - len(centers)
    for _ in range(extra_circles):
        x = np.random.uniform(0, 1)
        y = np.random.uniform(0, 1)
        centers.append([x, y])
        radii.append(0.01)
    
    # Optimization parameters
    bounds = [(0, 1) for _ in range(2 * len(centers))]
    radii_bounds = [(0, 0.1) for _ in range(len(centers))]
    
    def objective(x):
        centers = x[:2 * len(centers)].reshape(-1, 2)
        radii = x[2 * len(centers):].clip(0, 0.1)
        return -np.sum(radii)
    
    def constraint(x):
        centers = x[:2 * len(centers)].reshape(-1, 2)
        radii = x[2 * len(centers):].clip(0, 0.1)
        dist_matrix = squareform(pdist(centers))
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0
    
    result = minimize(objective, np.concatenate([np.random.uniform(0, 1, 2 * len(centers)), np.random.uniform(0, 0.1, len(centers))]),
                      bounds=bounds + radii_bounds,
                      constraints=[{'type': 'ineq', 'fun': constraint}],
                      method='SLSQP',
                      options={'ftol': 1e-10, 'maxiter': 200})
    
    centers = result.x[:2 * len(centers)].reshape(-1, 2)
    radii = result.x[2 * len(centers):].clip(0, 0.1)
    
    # Final validation
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            raise ValueError(f"Circle {i} outside unit square")
    
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError(f"Circles {i} and {j} overlap")
    
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
