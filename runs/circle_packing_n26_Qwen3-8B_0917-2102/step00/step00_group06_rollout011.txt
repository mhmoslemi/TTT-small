import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import LinearConstraint

def run_packing():
    n = 26
    # Initialize centers in a hexagonal grid
    centers = np.zeros((n, 2))
    r = np.zeros(n)
    max_r = 0.1
    r.fill(max_r)
    centers[0] = [0.5, 0.5]
    centers[1] = [0.5 - np.sqrt(3)/2 * max_r, 0.5 + max_r/2]
    centers[2] = [0.5 + np.sqrt(3)/2 * max_r, 0.5 + max_r/2]
    
    # Try to optimize the radii and positions
    def objective(x):
        return -np.sum(x[:n])
    
    def constraints(x):
        r = x[:n]
        centers_opt = x[n:]
        constraints = []
        for i in range(n):
            x_i, y_i = centers_opt[i*2], centers_opt[i*2+1]
            r_i = r[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return False
            for j in range(i+1, n):
                x_j, y_j = centers_opt[j*2], centers_opt[j*2+1]
                r_j = r[j]
                dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
                if dist < r_i + r_j - 1e-12:
                    return False
        return True
    
    x0 = np.concatenate((r, centers.flatten()))
    bounds = Bounds([0]*n + [0]*2*n, [np.inf]*n + [1]*2*n)
    cons = {'type': 'success', 'fun': constraints}
    result = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 1000})
    
    centers_opt = result.x[n:]
    r_opt = result.x[:n]
    sum_radii = np.sum(r_opt)
    return (centers_opt.reshape(n, 2), r_opt, sum_radii)
