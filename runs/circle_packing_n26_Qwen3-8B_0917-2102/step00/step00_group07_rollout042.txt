import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.sparse import csr_matrix

def run_packing():
    np.random.seed(42)
    n = 26

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        return np.sum(radii)

    def bounds():
        return [(0, 1.0) for _ in range(n)]

    def packing_function(centers, radii):
        dist_matrix = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    def initial_guess():
        grid = np.linspace(0, 1, 7)
        x, y = np.meshgrid(grid, grid)
        centers = np.stack((x.ravel(), y.ravel()), axis=-1)
        centers = centers[centers[:, 0] < 1 - 1e-8]
        centers = centers[centers[:, 1] < 1 - 1e-8]
        return centers[:n]

    centers = initial_guess()
    radii = np.full(n, 0.05)

    def pack_update(centers, radii):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                radii[i] = min(r, min(x, 1 - x, y, 1 - y))

    for _ in range(10):
        dist_matrix = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    factor = (dist_matrix[i, j] / (radii[i] + radii[j])) ** 2
                    radii[i] *= np.sqrt(factor)
                    radii[j] *= np.sqrt(factor)

        pack_update(centers, radii)

    for _ in range(5):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            max_r = min(x, 1 - x, y, 1 - y)
            if r < max_r:
                radii[i] = max_r

    return centers, radii, np.sum(radii)
