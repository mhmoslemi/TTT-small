from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    # Initial guess for centers in hexagonal packing
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    rows = 6
    cols = 5
    spacing = 1.0 / (rows + 1)

            x = j * spacing + spacing / 2

            centers[i * cols + j] = [x, y]
    
    # Optimization: maximize sum of radii with non-overlapping constraint

        return -np.sum(radii)  # Negative for minimization

    result = minimize(objective, np.ones(num_circles) * 0.1, method='SLSQP', bounds=bounds, constraints=cons)

    # Adjust radii to fit within square (max radius based on distance to nearest edge)
    for i in range(num_circles):
        x, y = centers[i]

            y - 1e-8,
            1 - y - 1e-8
        ])
