import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist
from scipy.optimize import Bounds

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        return np.sum(radii)

    n = 26
    base_centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 7
        col = i % 7
        base_centers[i, 0] = 0.1 + col * 0.15
        base_centers[i, 1] = 0.1 + row * 0.15

    base_centers = np.vstack((base_centers, np.array([0.5, 0.5])))
    base_radii = np.full(n, 0.05)

    def compute_distances(centers):
        dists = pdist(centers)
        return dists

    def pack(centers, radii):
        for i in range(n):
            if (centers[i, 0] - radii[i] < -1e-12 or centers[i, 0] + radii[i] > 1 + 1e-12 or
                centers[i, 1] - radii[i] < -1e-12 or centers[i, 1] + radii[i] > 1 + 1e-12):
                return False
        for i in range(n):
            for j in range(i + 1, n):
                if np.sqrt(np.sum((centers[i] - centers[j])**2)) < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    def fitness(x):
        centers = np.array([x[2 * i:2 * i + 2] for i in range(n)])
        radii = np.array(x[2 * n:])
        if not pack(centers, radii):
            return -np.inf
        return -np.sum(radii)

    x0 = np.hstack((base_centers.flatten(), base_radii))

    bounds = Bounds([], [])
    result = minimize(fitness, x0, method='L-BFGS-B', bounds=bounds, tol=1e-8)

    x_opt = result.x
    centers = np.array([x_opt[2 * i:2 * i + 2] for i in range(n)])
    radii = x_opt[2 * n:]
    return centers, radii, -result.fun
