from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():

    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Hexagonal packing with boundary circles
    # First row: left to right
    for i in range(13):
        centers[i, 0] = i / 12
        centers[i, 1] = 0.25
    # Second row: offset and lower
    for i in range(13):
        centers[13 + i, 0] = (i + 0.5) / 12
        centers[13 + i, 1] = 0.75

    initial_radius = 0.15

                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:

        return 0.0

    x0 = np.r_[centers.reshape(-1), radii]

    result = minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=[

    # Extract final centers and radii
