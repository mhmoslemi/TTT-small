import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance

def run_packing():

    # Each row has 13 circles with decreasing spacing
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # First row (top) has 13 circles

        centers[i] = [x, y]
        radii[i] = 0.15
    
    # Optimization objective: maximize sum of radii
    def objective(r):
        return -np.sum(r)  # negative for minimization
    
    # Constraints: circles must not overlap and stay within [0,1]^2
    def constraint_overlap(r, centers):
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < r[i] + r[j] - 1e-8:
                    return False
        return True
    
    # Initial guess
    initial_r = radii.copy()
    
    # Optimization
    res = minimize(objective, initial_r, method='SLSQP',
                   constraints={'type': 'ineq', 'fun': lambda r: constraint_overlap(r, centers)},

    # Final radii and centers
    radii = res.x

    return centers, radii, sum_radii
