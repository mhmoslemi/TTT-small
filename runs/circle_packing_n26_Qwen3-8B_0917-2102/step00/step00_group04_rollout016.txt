import numpy as np
from scipy.optimize import minimize

def run_packing():
    n = 26
    # Define initial positions with hexagonal packing
    rows = 5
    cols = 6
    spacing = 1.0 / (cols - 1)
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(rows):
        for j in range(cols):
            if i % 2 == 0:
                x = j * spacing
            else:
                x = j * spacing + spacing / 2
            y = i * (np.sqrt(3) / 2) * spacing
            centers[i * cols + j] = [x, y]
    
    # Optimize radii to maximize sum while maintaining non-overlap and boundary constraints
    def objective(radii):
        return -np.sum(radii)
    
    def constraints(radii):
        cons = []
        for i in range(n):
            cons.append((radii[i], 0, np.inf))  # Non-negative radii
            x, y = centers[i]
            cons.append((radii[i], 0, 1 - x))  # Left boundary
            cons.append((radii[i], 0, x - 0))  # Right boundary
            cons.append((radii[i], 0, 1 - y))  # Bottom boundary
            cons.append((radii[i], 0, y - 0))  # Top boundary
        
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                cons.append((radii[i] + radii[j], 0, dist - 1e-12))  # Non-overlap constraint
        
        return cons
    
    # Initial guess for radii
    initial_guess = np.full(n, 0.05)
    bounds = [(0, 1) for _ in range(n)]
    
    # Set up and run optimization
    result = minimize(objective, initial_guess, bounds=bounds, method='SLSQP', constraints=constraints)
    
    if not result.success:
        raise ValueError("Optimization failed")
    
    return centers, result.x, np.sum(result.x)
