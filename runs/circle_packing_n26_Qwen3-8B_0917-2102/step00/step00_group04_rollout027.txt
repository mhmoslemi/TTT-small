import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    n = 26
    # Initial guess: hexagonal packing
    centers = []
    radii = []
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (row % 2))
        y = row + 0.5
        centers.append([x, y])
    centers = np.array(centers)
    radii = np.full(n, 0.05)

    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Constraints: non-overlapping and within unit square
    def constraint_func(radii):
        constraints = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return False, f"Circle {i} outside unit square"
            constraints.append((x - r, x + r))
            constraints.append((y - r, y + r))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"
        return True, "ok"

    # Optimize with constraints
    radii_bounds = Bounds([0.01] * n, [0.5] * n)
    result = minimize(
        fun=objective,
        x0=radii,
        bounds=radii_bounds,
        method='Nelder-Mead',
        options={'xatol': 1e-6, 'maxiter': 1000}
    )
    radii = result.x
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
