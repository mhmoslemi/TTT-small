import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial guess: 4 clusters near corners with hexagonal packing
    # Each cluster will have 6 circles
    centers = np.array([
        [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
        [0.1, 0.3], [0.3, 0.3], [0.5, 0.3], [0.7, 0.3], [0.9, 0.3],
        [0.1, 0.5], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5], [0.9, 0.5],
        [0.1, 0.7], [0.3, 0.7], [0.5, 0.7], [0.7, 0.7], [0.9, 0.7],

    ])
    radii = np.full(26, 0.05)

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        # Boundary constraint
        for i in range(26):
            if centers[i, 0] - radii[i] < -1e-12 or centers[i, 0] + radii[i] > 1 + 1e-12:
                return False
            if centers[i, 1] - radii[i] < -1e-12 or centers[i, 1] + radii[i] > 1 + 1e-12:
                return False
        # Pairwise distance constraint
        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    bounds = [(0, 0.5) for _ in range(26)]

    result = minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints={"type": "exterior", "fun": constraints},
        method="SLSQP"
    )

    return (centers, result.x, -result.fun)
