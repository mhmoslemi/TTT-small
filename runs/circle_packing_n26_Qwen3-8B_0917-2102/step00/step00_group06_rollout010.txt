import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():

    num_circles = 26

    rows = 5
    cols = 5
    # Positions in hexagonal grid
    centers = []
    for i in range(rows):

            x = j + 0.5 * (i % 2)
            y = i * np.sqrt(3) / 2
            centers.append([x, y])
    centers = np.array(centers)

    shift = np.mean(centers, axis=0)
    centers -= shift
    # Initial radii estimate based on hex grid spacing
    spacing = np.min(cdist(centers, centers))
    radii = np.full(num_circles, spacing / 2)

    def objective(radii):
        return -np.sum(radii)

    # Constraints: circles must not overlap and stay within [0, 1]^2

                if dist < radii[i] + radii[j] - 1e-8:
                    return dist - radii[i] - radii[j] + 1e-8
        return 0

    def constraint_boundary(radii):
        for i in range(num_circles):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8 or y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8:

        return 0

    # Optimization
    bounds = [(0, 1) for _ in range(num_circles)]
    cons = [{'type': 'ineq', 'fun': constraint_overlap},

    res = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)

    # Return the result
