import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():
    np.random.seed(42)
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraints(radii):
        constraints = []
        for i in range(n):
            constraints.append((radii[i] >= 0))
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                constraints.append((radii[i] + radii[j] <= dist))
        for i in range(n):
            x, y = centers[i]
            constraints.append((radii[i] <= x))
            constraints.append((radii[i] <= 1 - x))
            constraints.append((radii[i] <= y))
            constraints.append((radii[i] <= 1 - y))
        return constraints
    
    bounds = [(0, 1) for _ in range(n)]
    result = minimize(objective, radii, bounds=bounds, constraints=constraints)
    radii = result.x
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
