from scipy.spatial import distance
from itertools import combinations

        for i, j in combinations(range(26), 2):

            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < radii[i] + radii[j] - 1e-8:
                return -total
        return total

    # Initial hexagonal packing with boundary constraints
    centers = np.zeros((26, 2))
    row = 0
    col = 0
    for i in range(26):
        row = i // 5
        col = i % 5
        x = col / 5 + 0.1
        y = row / 5 + 0.1
        centers[i] = [x, y]

    def boundary_constraints(x):
        return np.array([x - 0.01, 1 - x - 0.01, x - 0.01, 1 - x - 0.01])

    def update_centers(radii):

    def objective(radii):
        return -fitness(radii)

    result = minimize(objective, x0=np.full(26, 0.05), method='L-BFGS-B', bounds=(0, 0.5))
