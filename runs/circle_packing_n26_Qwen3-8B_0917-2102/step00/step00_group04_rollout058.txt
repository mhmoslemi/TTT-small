import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial hexagonal packing with 26 circles
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Rows in hexagonal packing
    rows = 6
    cols = 5
    for i in range(rows):
        for j in range(cols):
            x = j + 0.5
            y = i + 0.5
            centers[i * cols + j] = [x, y]
    
    # Initial guess for radii - uniform
    radii = np.full(n, 0.1)
    
    # Optimization objective: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum
    
    # Constraints: circles must not overlap and stay within square
    def constraint(radii):
        nonlocal centers
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
                return -1
        return 0
    
    # Optimization
    res = minimize(objective, radii, method='SLSQP', constraints={'type': 'ineq', 'fun': constraint})
    
    # Return result
    return (centers, res.x, np.sum(res.x))
