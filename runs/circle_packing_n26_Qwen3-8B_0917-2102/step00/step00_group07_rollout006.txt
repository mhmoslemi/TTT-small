import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Hexagonal grid parameters
    cols = 6
    rows = 5
    spacing = 1.0 / np.sqrt(3)
    
    # Generate hexagonal grid
    centers = []
    radii = []
    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * spacing * np.sqrt(3) / 2
            if i % 2 == 0:
                x += spacing / 2
            centers.append([x, y])
    
    # Adjust for unit square
    centers = np.array(centers)
    scale = 1.0 / np.max(centers, axis=0)
    centers = centers * scale
    
    # Initial radii based on minimum distance between circles
    def min_distance(i, j):
        return np.sqrt(np.sum((centers[i] - centers[j])**2))
    
    min_dist = np.inf
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            d = min_distance(i, j)
            if d < min_dist:
                min_dist = d
    
    radii = np.full(len(centers), min_dist / 2)
    
    # Optimization to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)
    
    def constraint(radii):
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                if np.sqrt(np.sum((centers[i] - centers[j])**2)) < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0
    
    result = optimize.minimize(
        objective,
        radii,
        method='SLSQP',
        constraints=[{'type': 'ineq', 'fun': constraint}]
    )
    
    return (centers, result.x, np.sum(result.x))
