import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Initial guess for centers in hexagonal grid
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Create a hexagonal grid with 5 rows
    rows = 5
    cols = 5
    for i in range(rows):
        for j in range(cols):
            x = j / (cols - 1) if cols > 1 else 0.5
            y = i / (rows - 1) if rows > 1 else 0.5
            centers[i * cols + j] = [x, y]
    
    # Initial guess for radii
    radii = np.ones(n) * 0.1

    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization

    # Constraints: non-overlapping and within square
    def constraint(radii):
        dist = cdist(centers, centers)
        for i in range(n):
            for j in range(i + 1, n):
                if dist[i, j] < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        for i in range(n):
            if centers[i, 0] - radii[i] < -1e-8 or centers[i, 0] + radii[i] > 1 + 1e-8:
                return -np.inf
            if centers[i, 1] - radii[i] < -1e-8 or centers[i, 1] + radii[i] > 1 + 1e-8:
                return -np.inf
        return np.sum(radii)

    # Optimization
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=[(0, 1)] * n,
        constraints=[{'type': 'ineq', 'fun': constraint}]
    )

    centers = np.array(centers)
    radii = np.array(result.x)
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
