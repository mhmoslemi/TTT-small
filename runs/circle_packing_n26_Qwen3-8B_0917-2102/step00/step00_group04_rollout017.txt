<strategy>Use staggered rows with alternating circle sizes to optimize space near boundaries and corners.</strategy>

```python
import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial guess: 5 rows of 5 circles and 1 row of 6 circles, staggered
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    for i in range(5):
        for j in range(5):
            centers[i * 5 + j] = [j / 4.5, i / 4.5]
            radii[i * 5 + j] = 0.1
    for i in range(6):
        centers[25 - i] = [i / 5.0, 0.5]
        radii[25 - i] = 0.1
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraint1(radii):
        for i in range(26):
            x, y = centers[i]
            if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
                return -1
        return 0
    
    def constraint2(radii):
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0
    
    cons = [{'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2}]
    
    bounds = [(0, 0.5) for _ in range(26)]
    
    result = minimize(objective, radii, bounds=bounds, constraints=cons)
    
    return (centers, result.x, np.sum(result.x))
```
