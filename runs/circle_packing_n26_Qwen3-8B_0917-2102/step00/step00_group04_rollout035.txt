import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        cons = []
        for i in range(len(radii)):
            cons.append((radii[i], 1.0 - 1e-12))
        return cons

    def packing_func(centers, radii):
        n = len(radii)
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12:
                return False, "Circle out of bounds"
            if y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return False, "Circle out of bounds"
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, "Overlap detected"
        return True, "Valid packing"

    def fitness_func(centers, radii):
        if not packing_func(centers, radii):
            return -np.inf
        return np.sum(radii)

    n = 26
    initial_centers = np.random.rand(n, 2)
    initial_radii = np.random.rand(n)

    bounds = [(0, 1 - 1e-12) for _ in range(n)]

    result = minimize(lambda x: -fitness_func(initial_centers, x), x0=initial_radii, bounds=bounds, method='L-BFGS-B')

    best_radii = result.x
    best_centers = initial_centers

    for i in range(n):
        best_centers[i] = np.array([np.clip(best_centers[i][0], 0, 1), np.clip(best_centers[i][1], 0, 1)])

    if not packing_func(best_centers, best_radii):
        raise ValueError("Final packing is invalid")

    sum_radii = np.sum(best_radii)
    return best_centers, best_radii, sum_radii
