import numpy as np
from scipy.optimize import minimize

def run_packing():

    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Hexagonal grid layout
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - 2 * (i % 5 == 0))
        y = row + 0.5 * (1 - 2 * (i // 5 == 0))
        centers[i] = [x, y]
        radii[i] = 0.05
    
    # Define objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)
    
    # Define constraints
    def constraint_func(radii):
        return np.sum(radii)
    
    # Optimization using scipy
    bounds = [(0, 1) for _ in range(n)]
    
    # Initial guess for radii
    initial_guess = np.full(n, 0.06)
    
    # Optimization with constraints
    result = minimize(objective, initial_guess, bounds=bounds, method='SLSQP')
    
    # Adjust radii to fit in the square
    max_radius = 0.15
    radii = np.clip(result.x, 0, max_radius)
    
    # Final validation
    def validate_packing(centers, radii):
        n = centers.shape[0]
        if np.isnan(centers).any() or np.isnan(radii).any():
            return False, "NaN values present"
        for i in range(n):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"

            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or

                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"
        return True, "ok"
    
    # Run validation
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(msg)
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
