import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Define initial hexagonal packing with 5 rows (max 26 circles)
    n = 26
    centers = []
    radii = []

    # Define initial positions in staggered rows
    row_count = 5
    col_count = 5
    for row in range(row_count):
        for col in range(col_count):
            x = col / (col_count - 1) if col_count > 1 else 0.5
            y = row / (row_count - 1) if row_count > 1 else 0.5
            centers.append([x, y])
    centers = np.array(centers)
    radii = np.full(n, 0.1)

    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum for maximization

    # Constraints: non-overlapping and within unit square
    def constraint(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    # Optimization
    res = minimize(
        objective,
        radii,
        method='SLSQP',
        constraints=[
            {'type': 'ineq', 'fun': constraint},
        ],
        bounds=[(0, 1) for _ in range(n)]
    )

    # Final values
    centers = np.array(centers)
    radii = res.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
