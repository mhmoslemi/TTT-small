<strategy>Use hexagonal packing with optimized boundary circles to maximize total radius sum.</strategy>

```python
import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii, centers):
        n = len(radii)
        cons = []
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - 1e-12})
        for i in range(n):
            for j in range(i + 1, n):
                cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: np.sqrt(np.sum((centers[i] - centers[j])**2)) - r[i] - r[j] - 1e-12})
        for i in range(n):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - 1e-12})
        return cons

    def boundary_circles():
        row = np.linspace(0, 1, 6)
        col = np.linspace(0, 1, 5)
        centers = []
        for x in row:
            for y in col:
                if x == 0 or x == 1 or y == 0 or y == 1:
                    centers.append([x, y])
        return np.array(centers)

    def inner_circles():
        inner = []
        for x in np.linspace(0.1, 0.9, 5):
            for y in np.linspace(0.1, 0.9, 5):
                if (x - 0.1 > 0.1) and (x + 0.1 < 0.9) and (y - 0.1 > 0.1) and (y + 0.1 < 0.9):
                    inner.append([x, y])
        return np.array(inner)

    centers = np.vstack((boundary_circles(), inner_circles()))
    n = len(centers)

    initial_radii = np.full(n, 0.05)
    bounds = [(0, 1) for _ in range(n)]

    result = minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints=constraints(initial_radii, centers),
        method='SLSQP'
    )

    return (centers, result.x, np.sum(result.x))
```
