import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    # Initial guess: hexagonal packing with boundary circles
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Place circles in a hexagonal grid with boundary circles
    for i in range(n):
        if i < 4:
            # Top boundary circles
            centers[i] = [0.2 + i * 0.2, 0.95]
            radii[i] = 0.05
        elif i < 8:
            # Left boundary circles
            centers[i] = [0.05, 0.1 + (i - 4) * 0.2]
            radii[i] = 0.05
        elif i < 12:
            # Right boundary circles
            centers[i] = [0.95, 0.1 + (i - 8) * 0.2]
            radii[i] = 0.05
        else:
            # Main body of circles
            row = (i - 12) // 4
            col = (i - 12) % 4
            centers[i] = [0.2 + col * 0.25, 0.3 + row * 0.25]
            radii[i] = 0.05

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraint_overlap(radii):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return -1.0
        return 0.0

    def constraint_boundary(radii):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e8:
                return -1.0
        return 0.0

    bounds = Bounds(0, 1)

    # Initial guess
    initial_radii = radii.copy()

    # Optimization
    result = minimize(
        fun=objective,
        x0=initial_radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_overlap},
            {'type': 'ineq', 'fun': constraint_boundary}
        ],
        method='SLSQP'
    )

    centers = np.array(centers)
    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
