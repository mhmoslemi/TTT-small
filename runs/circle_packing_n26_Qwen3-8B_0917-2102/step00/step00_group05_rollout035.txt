import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    # Initial guess for centers in a hexagonal packing
    rows = 6
    cols = 5
    spacing = 0.2
    centers = []
    radii = []
    
    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * spacing * np.sqrt(3)
            centers.append([x, y])
    
    # Define objective function to maximize sum of radii
    def objective(params):
        radii = params[:cols * rows]
        radii = radii.reshape((rows, cols))
        sum_radii = np.sum(radii)
        return -sum_radii  # Minimize negative to maximize
    
    # Define constraints for non-overlapping and boundary conditions
    def constraint(params):
        radii = params[:cols * rows]
        radii = radii.reshape((rows, cols))
        centers = np.array(centers)
        for i in range(rows):
            for j in range(cols):
                r_i = radii[i, j]
                for k in range(rows):
                    for l in range(cols):
                        if i != k or j != l:
                            dx = centers[i, j, 0] - centers[k, l, 0]
                            dy = centers[i, j, 1] - centers[k, l, 1]
                            dist = np.sqrt(dx**2 + dy**2)
                            if dist < r_i + radii[k, l] - 1e-6:
                                return -1
        for i in range(rows):
            for j in range(cols):
                r_i = radii[i, j]
                if centers[i, j, 0] - r_i < 0 or centers[i, j, 0] + r_i > 1:
                    return -1
                if centers[i, j, 1] - r_i < 0 or centers[i, j, 1] + r_i > 1:
                    return -1

    # Flatten centers for optimization

    bounds = Bounds(0, 0.5)
    
    # Optimization
    result = minimize(
        objective,
        params,
        method='SLSQP',

        constraints={'type': 'ineq', 'fun': constraint},

    )

    sum_radii = np.sum(radii)
