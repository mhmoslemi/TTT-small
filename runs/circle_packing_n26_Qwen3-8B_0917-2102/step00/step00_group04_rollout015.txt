from scipy.optimize import minimize, Bounds

def run_packing():
    np.random.seed(42)
    n = 26

    # Initial guess: hexagonal packing
    sides = int(np.ceil(np.sqrt(n)))
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // sides
        col = i % sides
        x = col + 0.5 * (1 - 1 / sides) * (row % 2)
        y = row + 0.5 * (1 - 1 / sides) * (row % 2)
        centers[i] = [x, y]

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        constraints = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            constraints.append((x - r >= -1e-12) * 1)
            constraints.append((x + r <= 1 + 1e-12) * 1)
            constraints.append((y - r >= -1e-12) * 1)
            constraints.append((y + r <= 1 + 1e-12) * 1)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append((dist >= radii[i] + radii[j] - 1e-12) * 1)
        return constraints

    bounds = [(0, 1) for _ in range(n)]

    result = minimize(
        objective,
        np.ones(n),
        bounds=bounds,
        constraints=constraints,
        method='SLSQP'
    )

    if not result.success:
        raise RuntimeError("Optimization failed")

    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
