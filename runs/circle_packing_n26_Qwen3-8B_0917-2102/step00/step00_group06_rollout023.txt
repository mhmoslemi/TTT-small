import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Hexagonal grid setup with boundary circles
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    
    # First row of circles near top
    for i in range(6):
        centers[i, 0] = i * 0.5
        centers[i, 1] = 1 - 0.2
        radii[i] = 0.18
    
    # Second row of circles near bottom
    for i in range(6):
        centers[6+i, 0] = i * 0.5 + 0.25
        centers[6+i, 1] = 0.2
        radii[6+i] = 0.18
    
    # Middle circles
    for i in range(10):
        centers[12+i, 0] = 0.5 + i * 0.1
        centers[12+i, 1] = 0.5
        radii[12+i] = 0.15
    
    # Top and bottom boundary circles
    centers[12+10, 0] = 0.5
    centers[12+10, 1] = 1 - 0.15
    radii[12+10] = 0.15
    
    centers[12+11, 0] = 0.5
    centers[12+11, 1] = 0.15
    radii[12+11] = 0.15
    
    # Optimization to maximize total radius
    def objective(radii):
        return -np.sum(radii)
    
    def constraint_func(radii):
        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                constraints.append(dist >= radii[i] + radii[j] - 1e-8)
        return constraints
    
    bounds = [(0, 1)] * num_circles
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[{"type": "ineq", "fun": lambda x: np.min(constraint_func(x))}]
    )
    
    return (centers, result.x, np.sum(result.x))
