import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    # Initial hexagonal packing
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Hexagonal grid parameters
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    
    # Place circles in hexagonal grid
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        centers[i] = [x / (cols - 1) if cols > 1 else 0, y / (rows - 1) if rows > 1 else 0]
    
    # Initial radius estimate based on spacing
    initial_radius = np.min(np.linalg.norm(centers[1:] - centers[0], axis=1)) / 2
    radii[:] = initial_radius
    
    # Function to calculate objective: sum of radii
    def objective(x):
        radii[:] = x
        return -np.sum(radii)  # Minimize negative sum to maximize sum
    
    # Constraints: circles must not overlap and stay within the unit square
    def constraint(x, i):
        radii[:] = x
        for j in range(n):
            if j == i:
                continue
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx**2 + dy**2
            if dist_sq < (x[i] + x[j])**2 - 1e-12:
                return 1
        return 0
    
    # Create constraints for all pairs
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: constraint(x, i)})
    
    # Boundary constraints
    def boundary_constraint(x):
        for i in range(n):
            dx = centers[i, 0]
            dy = centers[i, 1]
            if dx - x[i] < -1e-12 or dx + x[i] > 1 + 1e-12 or dy - x[i] < -1e-12 or dy + x[i] > 1 + 1e-12:
                return 1
        return 0
    
    cons.append({'type': 'ineq', 'fun': boundary_constraint})
    
    # Optimization
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        constraints=cons,
        bounds=[(0, 1)] * n,
        options={'ftol': 1e-8, 'maxiter': 1000}
    )
    
    if not result.success:
        raise ValueError("Optimization failed")
    
    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
