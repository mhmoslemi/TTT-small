import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n)

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        constraints = []
        for i in range(n):
            constraints.append((radii[i], 0, 1))
        return constraints

    bounds = [(0, 1) for _ in range(n)]
    result = minimize(objective, radii, bounds=bounds, method='SLSQP')

    radii = result.x
    centers = np.random.rand(n, 2)

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            radii[i] = min(r, 1 - np.min([x, 1 - x, y, 1 - y]))

    dist_matrix = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dist_matrix[i, j] < radii[i] + radii[j] - 1e-12:
                radii[i] = (dist_matrix[i, j] + radii[j]) / 2
                radii[j] = (dist_matrix[i, j] + radii[i]) / 2

    return centers, radii, np.sum(radii)
