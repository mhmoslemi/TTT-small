import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance

def run_packing():
    # Initial guess for centers using hexagonal packing
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    radius = 0.1  # Start with a small radius for optimization
    for i in range(num_circles):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (1 - (row % 2))
        y = row + 0.5
        centers[i] = [x, y]
        radii[i] = radius

    def objective(radii):
        return -np.sum(radii)  # Maximize sum by minimizing negative sum

    def constraints(radii):
        # Boundary constraints
        for i in range(num_circles):
            if centers[i, 0] - radii[i] < 0:
                return False
            if centers[i, 0] + radii[i] > 1:
                return False
            if centers[i, 1] - radii[i] < 0:
                return False
            if centers[i, 1] + radii[i] > 1:
                return False
        # Circle-circle constraints
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    # Optimization
    bounds = [(0, 0.5) for _ in range(num_circles)]
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints={"type": "soft_bound", "fun": constraints, "jac": None},
        method="SLSQP"
    )

    # Return centers, radii, and sum
    return centers, result.x, np.sum(result.x)
