import numpy as np
from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d

def run_packing():
    def objective(r):
        return -np.sum(r)  # maximize sum of radii

    def constraints(r, centers):
        cons = []
        for i in range(len(r)):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i]})
        for i in range(len(r)):
            for j in range(i + 1, len(r)):
                cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: np.sqrt(np.sum((centers[i] - centers[j])**2)) - (r[i] + r[j])})
        return cons

    def get_initial_centers():
        n = 26
        cols = int(np.sqrt(n))
        rows = (n + cols - 1) // cols
        centers = np.zeros((n, 2))
        for i in range(n):
            row = i // cols
            col = i % cols
            centers[i, 0] = col / (cols - 1)
            centers[i, 1] = row / (rows - 1)
        return centers

    centers = get_initial_centers()
    r0 = np.ones(26) * 0.05

    bounds = [(0, 1) for _ in range(26 * 2)]

    result = minimize(objective, r0, method='SLSQP', constraints=constraints(r0, centers), bounds=bounds)

    if not result.success:
        raise RuntimeError("Optimization failed")

    radii = result.x
    return centers, radii, np.sum(radii)
