from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    n = 26
    centers = []
    radii = []

        for j in range(6):
            x = j + 0.5
            y = i + 0.5
            centers.append([x, y])

    for j in range(2):
        x = j + 0.5
        y = 4 + 0.5
        centers.append([x, y])

    centers = np.array(centers)
    radii = np.full(n, 0.0)

    # Initial guess for radii
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        # Distance between centers minus radii sum
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    # Use optimization to maximize sum of radii
    result = minimize(lambda r: -np.sum(r), radii, constraints=lambda r: constraints(r))

    # Final radii
    radii = result.x

    # Ensure circles are within the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8):
            radii[i] = min(r, 1 - min(x, 1 - x, y, 1 - y))

    return centers, radii, np.sum(radii)
