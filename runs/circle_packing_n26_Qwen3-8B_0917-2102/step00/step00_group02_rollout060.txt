from scipy.optimize import minimize

def run_packing():

    n = 26
    # Initial hexagonal packing with adaptive scaling

    radii = np.zeros(n)
    total_sum = 0.0

    centers[0] = [0.5, 0.5]
    radii[0] = 0.1
    total_sum += radii[0]

    for i in range(1, 7):
        angle = 2 * np.pi / 6 * i
        r = 0.1
        centers[i] = [0.5 + r * np.cos(angle), 0.5 + r * np.sin(angle)]
        radii[i] = r
        total_sum += r
    
    # Third layer: 12 circles around second layer
    for i in range(7, 19):
        angle = 2 * np.pi / 12 * (i - 6)
        r = 0.08
        centers[i] = [0.5 + r * np.cos(angle), 0.5 + r * np.sin(angle)]
        radii[i] = r
        total_sum += r
    
    # Fourth layer: 8 circles around third layer
    for i in range(19, 27):
        angle = 2 * np.pi / 8 * (i - 18)
        r = 0.06
        centers[i] = [0.5 + r * np.cos(angle), 0.5 + r * np.sin(angle)]
        radii[i] = r
        total_sum += r
    
    # Optimization to refine positions and radii
    def objective(x):
        # x is a flat array of [x0, y0, r0, x1, y1, r1, ...]
        centers_flat = x[:2*n].reshape(n, 2)
        radii_flat = x[2*n:].reshape(n)
        # Check boundaries
        for i in range(n):
            x, y = centers_flat[i]
            r = radii_flat[i]
            if not (0 - r <= x <= 1 + r and 0 - r <= y <= 1 + r):
                return -np.inf
        # Check overlaps
        for i in range(n):
            for j in range(i+1, n):
                dx = centers_flat[i][0] - centers_flat[j][0]
                dy = centers_flat[i][1] - centers_flat[j][1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii_flat[i] + radii_flat[j] - 1e-8:
                    return -np.inf
        # Maximize sum of radii
        return np.sum(radii_flat)
    
    # Initial guess
    x0 = np.concatenate([centers.flatten(), radii])
    result = minimize(lambda x: -objective(x), x0, method='L-BFGS-B', bounds=[(0, 1) for _ in range(2*n)] + [(0, 1) for _ in range(n)])
    
    # Extract final centers and radii
    centers = result.x[:2*n].reshape(n, 2)
    radii = result.x[2*n:].reshape(n)
    total_sum = objective(result.x)
    
    return centers, radii, total_sum
