import numpy as np
from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d

def run_packing():
    # Initial guess: hexagonal packing with 5 rows
    centers = []
    radii = []
    for i in range(5):
        for j in range(6 - i % 2):
            x = j + 0.5 * i / 5
            y = i / 5 + 0.5
            centers.append([x, y])
    centers = np.array(centers)
    radii = np.full(len(centers), 0.1)

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        n = len(radii)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8 or y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e8:
                return False
        return True

    res = minimize(
        objective,
        radii,
        constraints=lambda r: constraints(r),
        method='SLSQP',
        bounds=[(0, 0.5) for _ in range(len(radii))]
    )
    return centers, res.x, np.sum(res.x)
