import numpy as np
from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d

def run_packing():
    np.random.seed(42)
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraint_circle(x, y, r):
        return lambda r: (x - r < 0 or x + r > 1) or (y - r < 0 or y + r > 1)
    
    def constraint_overlap(i, j):
        return lambda r: np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2) < r[i] + r[j]
    
    def generate_initial_centers():
        grid = np.linspace(0, 1, 5)
        x, y = np.meshgrid(grid, grid)
        x = x.flatten()
        y = y.flatten()
        centers = np.column_stack((x, y))
        return centers
    
    def refine_centers(centers):
        vor = Voronoi(centers)
        regions = vor.regions
        refined = []
        for region in regions:
            if not -1 in region:
                refined.append(centers[region])
        return np.vstack(refined)
    
    def optimize():
        initial_centers = generate_initial_centers()
        n = initial_centers.shape[0]
        centers = refine_centers(initial_centers)
        bounds = [(0, 1)] * 2 * n
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                constraints.append({'type': 'ineq', 'fun': constraint_overlap(i, j)})
        for i in range(n):
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i]})
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i: 1 - r[i]})
        result = minimize(objective, x0=np.ones(n), bounds=bounds, constraints=constraints)
        return result.x
    
    radii = optimize()
    centers = generate_initial_centers()
    return (centers, radii, np.sum(radii))
