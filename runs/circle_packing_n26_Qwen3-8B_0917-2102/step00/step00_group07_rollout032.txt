def run_packing():

    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    grid = np.zeros((n, 2))
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (row % 2)
        y = row
        grid[i] = [x / (cols - 1) if cols > 1 else 0, y / (rows - 1) if rows > 1 else 0]

    # Perform k-means clustering for better distribution
    centers = np.random.rand(n, 2)
    centroids, _ = kmeans(centers, n)
    grid = centroids

    # Optimize radii and positions
    def objective(x):
        centers = x[:n*2].reshape(n, 2)
        radii = x[n*2:]
        return -np.sum(radii)

    def constraints(x):
        centers = x[:n*2].reshape(n, 2)
        radii = x[n*2:]
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return np.inf
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)

                    return np.inf
        return 0

    x0 = np.concatenate((grid.ravel(), np.random.rand(n)))
    result = minimize(objective, x0, constraints=lambda x: constraints(x), method='SLSQP', tol=1e-8, options={'maxiter': 500})
