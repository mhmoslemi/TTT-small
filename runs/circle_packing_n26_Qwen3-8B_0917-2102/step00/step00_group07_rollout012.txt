import numpy as np
from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d
from itertools import combinations

def run_packing():
    np.random.seed(42)
    n = 26

    # Initial guess: hexagonal packing with 5 rows
    centers = []
    radii = []
    for row in range(5):
        for col in range(5):
            x = col + 0.5 * (row % 2)
            y = row + 0.5
            centers.append([x, y])
    centers = np.array(centers)
    radii = np.full(len(centers), 0.1)

    # Optimize radii with fixed positions
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        constraints = []
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append(dist - (radii[i] + radii[j]) >= 1e-8)
        for i in range(len(radii)):
            x, y = centers[i]
            constraints.append(x - radii[i] >= -1e-8)
            constraints.append(x + radii[i] <= 1 + 1e-8)
            constraints.append(y - radii[i] >= -1e-8)
            constraints.append(y + radii[i] <= 1 + 1e-8)
        return constraints

    bounds = [(0, 1) for _ in range(n)]
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': lambda r: np.array(constraints(r))}]
    )

    if not result.success:
        raise ValueError("Optimization failed")

    return centers, result.x, np.sum(result.x)
