import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
import random

def run_packing():
    n = 26
    # Initial random positions within [0,1]^2
    centers = np.random.rand(n, 2)
    
    def objective(r):
        return -np.sum(r)
    
    def constraint(r):
        return np.sum(r) - 2.636
    
    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    def boundary_constraint(r):
        for i in range(n):
            x, y = centers[i]
            r_i = r[i]
            if (x - r_i < -1e-12 or x + r_i > 1 + 1e-12
                    or y - r_i < -1e-12 or y + r_i > 1 + 1e-12):
                return -1
        return 0
    
    cons = [{'type': 'ineq', 'fun': lambda r: np.sum(r)}]
    cons.append({'type': 'eq', 'fun': constraint})
    cons.append({'type': 'ineq', 'fun': check_overlap})
    cons.append({'type': 'ineq', 'fun': boundary_constraint})
    
    # Initial guess based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    
    # Use bounds for radii
    bounds = [(0, 1)] * n
    
    # Use SLSQP with bounds and constraints
    result = minimize(objective, initial_r, bounds=bounds, constraints=cons, method='SLSQP')
    
    # Post-processing to ensure boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0
    
    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
