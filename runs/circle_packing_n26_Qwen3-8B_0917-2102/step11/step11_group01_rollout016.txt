import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create a hexagonal lattice with 5 rows and 6 columns, adjusted for boundary constraints
    initial_centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 6
        col = i % 6
        x = col + 0.5 * (1 - row % 2)
        y = row + 0.5
        initial_centers[i] = [x, y]
    
    def objective(r):
        return -np.sum(r)
    
    def constraint_sum(r):
        return np.sum(r) - 2.636
    
    def constraint_overlap(r):
        dists = squareform(pdist(initial_centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    cons = [{'type': 'ineq', 'fun': constraint_overlap}]
    cons.append({'type': 'eq', 'fun': constraint_sum})
    
    initial_r = np.min(squareform(pdist(initial_centers))) * np.ones(n)
    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)
    
    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = initial_centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0
    
    return initial_centers, result['x'], np.sum(result['x'])
