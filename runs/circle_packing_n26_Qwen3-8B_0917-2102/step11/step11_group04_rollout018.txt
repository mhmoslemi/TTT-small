import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal grid with variable rows and columns
    rows = 5
    cols = 6
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // cols
        col = i % cols
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    def scale_to_unit_square(centers, padding=0.1):
        min_coords = np.min(centers, axis=0)
        max_coords = np.max(centers, axis=0)
        scale = (1 - 2 * padding) / (max_coords - min_coords)
        centers = (centers - min_coords) * scale + padding
        centers = np.clip(centers, padding, 1 - padding)
        return centers
    
    centers = scale_to_unit_square(centers)

    def objective(r):
        return -np.sum(r)
    
    def constraint(r):
        return np.sum(r) - 2.636
    
    cons = [{'type': 'ineq', 'fun': lambda r: np.sum(r)}]
    cons.append({'type': 'eq', 'fun': constraint})

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    cons.append({'type': 'ineq', 'fun': check_overlap})

    # Initial guess based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)

    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Post-processing to fix boundary overlaps
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2 with boundary padding
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
