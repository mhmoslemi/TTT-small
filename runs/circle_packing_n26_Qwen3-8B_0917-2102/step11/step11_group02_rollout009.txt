import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate staggered grid with variable row spacing
    centers = np.zeros((n, 2))
    row_height = 1.0 / (np.sqrt(3) * 2)
    col_width = 1.0 / (np.sqrt(3) * 2)
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col * col_width + 0.5 * (1 - row % 2) * col_width
        y = row * row_height + 0.5 * row_height
        centers[i] = [x, y]
    
    # Initial radii based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 2.636

    cons = [{'type': 'ineq', 'fun': lambda r: np.sum(r)}]
    cons.append({'type': 'eq', 'fun': constraint})

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons.append({'type': 'ineq', 'fun': check_overlap})

    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Post-processing to ensure boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
