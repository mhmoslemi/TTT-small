import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    # Hexagonal grid with variable row spacing
    rows = 5
    cols = 6
    for i in range(n):
        row = i // cols
        col = i % cols
        y = row + 0.5
        x = col + 0.5 * (1 - row % 2)
        centers[i] = [x, y]

    # Enforce boundary alignment
    min_dist = np.min(pdist(centers))
    initial_r = np.full(n, min_dist / 2)
    bounds = [(0, 1)] * n

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint}]
    result = minimize(objective, initial_r, bounds=bounds, constraints=cons, method='SLSQP')

    # Final adjustment to ensure boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
