import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Generate a hexagonal grid with variable spacing
    centers = []
    for row in range(4):

            x = col + 0.5
            y = row + 0.5
            if row % 2 == 1:
                x += 0.5
            centers.append([x, y])
    centers = np.array(centers)

    # Scale and shift to fit within [0,1] with padding
    centers -= np.min(centers, axis=0)
    centers /= np.max(centers, axis=0)
    centers *= 0.8
    centers += 0.1

        return -np.sum(r)

                if dists[i, j] < r[i] + r[j] - 1e-8:

        return 0

        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    # Initial guess based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)

    # Optimization with tighter constraints
    cons = [{'type': 'ineq', 'fun': lambda r: np.sum(r)}]
    cons.append({'type': 'ineq', 'fun': check_overlap})

    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    return centers, result['x'], sum_radii
