import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    initial_centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row + 0.5
        initial_centers[i] = [x, y]
    
    def objective(r):
        return -np.sum(r)
    
    def constraint(r):
        return np.sum(r) - 2.636
    
    cons = [{'type': 'ineq', 'fun': lambda r: np.sum(r)}]
    cons.append({'type': 'eq', 'fun': constraint})
    
    def check_overlap(r):
        dists = squareform(pdist(initial_centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    cons.append({'type': 'ineq', 'fun': check_overlap})
    
    initial_r = np.min(squareform(pdist(initial_centers))) * np.ones(n)
    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)
    
    dists = squareform(pdist(initial_centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio
    
    # Boundary-aware scaling
    min_coords = np.min(initial_centers, axis=0)
    max_coords = np.max(initial_centers, axis=0)
    scale = (1 - 2 * 0.1) / (max_coords - min_coords)
    scaled_centers = (initial_centers - min_coords) * scale + 0.1
    scaled_centers = np.clip(scaled_centers, 0.1, 0.9)
    
    # Final adjustment based on scaled centers
    dists = squareform(pdist(scaled_centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio
    
    # Ensure circles are within [0,1]^2
    for i in range(n):

        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0
    
    return scaled_centers, result['x'], np.sum(result['x'])
