import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial hexagonal grid with optimized spacing
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    # Define objective to maximize sum of radii
    def objective(r):
        return -np.sum(r)
    
    # Define constraints
    def constraint_sum(r):
        return np.sum(r) - 2.636
    
    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    # Bounds for radii
    bounds = [(0, 1)] * n
    
    # Initial guess based on inter-circle distances
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    
    # Define constraints for optimization
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    cons.append({'type': 'ineq', 'fun': constraint_overlap})
    
    # Run optimization with SLSQP
    result = minimize(objective, initial_r, bounds=bounds, constraints=cons, method='SLSQP')
    
    # Final adjustment to ensure circles are within [0,1]^2
    radii = result['x']
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
