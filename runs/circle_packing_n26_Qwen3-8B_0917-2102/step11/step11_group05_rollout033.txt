import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create a refined hexagonal grid
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        if row % 2 == 0:
            x = col + 0.3
        else:
            x = col + 0.3 - 0
        y = row + 0.25
        centers[i] = [x, y]
    
    # Define bounds and constraints
    bounds = [(0, 1)] * (2 * n)
    constraints = []

    def objective(r):
        return -np.sum(r)
    
    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    def constraint_sum(r):
        return np.sum(r) - 2.636
    
    def constraint_bounds(r):
        for i in range(n):
            if r[i] < 0:
                return -1
        return 0
    
    constraints.append({'type': 'ineq', 'fun': constraint_bounds})
    constraints.append({'type': 'eq', 'fun': constraint_sum})
    constraints.append({'type': 'ineq', 'fun': check_overlap})
    
    initial_r = np.min(squareform(pdist(centers))) * np.ones(n)
    result = minimize(objective, initial_r, bounds=bounds, constraints=constraints, method='SLSQP')
    
    # Post-processing to ensure boundaries
    for i in range(n):
        r = result['x'][i]
        x, y = centers[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0
    
    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
