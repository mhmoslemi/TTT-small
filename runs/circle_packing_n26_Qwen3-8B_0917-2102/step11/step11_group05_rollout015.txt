import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal grid with variable spacing
    centers = np.zeros((n, 2))
    row, col = 0, 0
    for i in range(n):
        if i % 5 == 0:
            row += 1
            col = 0
        centers[i] = [0.2 + col * 0.25, 0.2 + row * 0.3]
        col += 1
    
    def objective(r):
        return -np.sum(r)
    
    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    def boundary_check(r):
        for i in range(n):
            x, y = centers[i]
            r_i = r[i]
            if (x - r_i < -1e-12 or x + r_i > 1 + 1e-12
                    or y - r_i < -1e-12 or y + r_i > 1 + 1e-12):
                return -1
        return 0
    
    initial_r = np.min(squareform(pdist(centers))) * np.ones(n)
    bounds = [(0, 1)] * n
    
    # Initial optimization with constraints
    cons = [{'type': 'ineq', 'fun': check_overlap},
            {'type': 'ineq', 'fun': boundary_check}]
    
    result = minimize(objective, initial_r, bounds=bounds, constraints=cons, method='SLSQP')
    
    # Post-processing to ensure boundary constraints and radius allocation
    for i in range(n):
        x, y = centers[i]
        r_i = result['x'][i]
        if (x - r_i < -1e-12 or x + r_i > 1 + 1e-12
                or y - r_i < -1e-12 or y + r_i > 1 + 1e-12):
            result['x'][i] = 0
    
    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
