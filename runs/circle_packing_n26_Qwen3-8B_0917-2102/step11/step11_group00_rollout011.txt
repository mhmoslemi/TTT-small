import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from genetic import GeneticAlgorithm

def run_packing():
    n = 26
    initial_centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row + 0.5
        initial_centers[i] = [x, y]
    
    ga = GeneticAlgorithm(population_size=50, generations=100, mutation_rate=0.15, crossover_rate=0.8)
    def fitness_func(centers):
        dists = squareform(pdist(centers))
        radii = np.min(dists) * np.ones(n)
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return -np.inf
        return -np.sum(radii)
    
    best_centers = ga.optimize(initial_centers, fitness_func)
    
    dists = squareform(pdist(best_centers))
    radii = np.min(dists) * np.ones(n)
    
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < radii[i] + radii[j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (radii[i] + radii[j])
                radii[i] *= ratio
                radii[j] *= ratio
    
    for i in range(n):
        x, y = best_centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0
    
    sum_radii = np.sum(radii)
    return best_centers, radii, sum_radii
