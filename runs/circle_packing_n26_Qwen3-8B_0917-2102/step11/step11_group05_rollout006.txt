import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate staggered hexagonal grid with more rows and optimized column spacing
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        if row % 2 == 0:
            x = col + 0.5
        else:
            x = col + 0.5 - 0
        y = row + 0.5
        centers[i] = [x, y]
    
    # Scale to unit square with boundary padding
    min_coords = np.min(centers, axis=0)
    max_coords = np.max(centers, axis=0)
    scale = (1 - 2 * 0.1) / (max_coords - min_coords)
    centers = (centers - min_coords) * scale + 0.1
    centers = np.clip(centers, 0.1, 0.9)
    
    def objective(r):
        return -np.sum(r)
    
    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    def constraint(r):
        return np.sum(r) - 2.636
    
    cons = [{'type': 'ineq', 'fun': lambda r: np.sum(r)}]
    cons.append({'type': 'eq', 'fun': constraint})
    cons.append({'type': 'ineq', 'fun': check_overlap})
    
    # Initial guess based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)
    
    # Post-processing for boundary constraints and overlap adjustments
    for i in range(n):
        for j in range(i + 1, n):
            if result['x'][i] + result['x'][j] > dists[i, j] + 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio
    
    # Ensure circles are within [0,1]^2 with boundary padding
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0
    
    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
