from scipy.optimize import differential_evolution
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Generate initial hexagonal lattice points within unit square
    initial_centers = np.zeros((n, 2))
    row = 0
    col = 0

            row += 1
            col = 0

        return -np.sum(r)
    
    # Constraint: circles must not overlap and stay within unit square

            if (x - r_i < -1e-12 or x + r_i > 1 + 1e-12

        return 0
    
    # Bounds for radii
    bounds = [(0, 1)] * n
    
    # Use differential evolution for global optimization
    result = differential_evolution(objective, bounds, constraints=[constraint], strategy='best1bin', popsize=15, maxiter=200, tol=1e-8)

    return centers, radii, np.sum(radii)
