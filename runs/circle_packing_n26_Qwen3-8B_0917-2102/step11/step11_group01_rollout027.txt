import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create a more optimal initial arrangement with staggered rows
    rows = 5
    cols = 6
    initial_centers = np.zeros((n, 2))
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (1 - (row % 2))
        y = row + 0.5
        initial_centers[i] = [x, y]
    
    def objective(r):
        return -np.sum(r)
    
    def constraint1(r):
        return np.sum(r)
    
    def constraint2(r):
        dists = squareform(pdist(initial_centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    cons = [{'type': 'ineq', 'fun': constraint1}]
    cons.append({'type': 'ineq', 'fun': constraint2})
    
    initial_r = np.min(squareform(pdist(initial_centers))) * np.ones(n)
    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)
    
    # Final adjustment to ensure circles are within [0,1]^2
    for i in range(n):
        x, y = initial_centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0
    
    return initial_centers, result['x'], np.sum(result['x'])
