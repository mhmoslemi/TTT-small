from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Generate hexagonal grid with adaptive row spacing

    col_coords = np.linspace(0.1, 0.9, 5)
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col_coords[col]
        y = row_coords[row]
        centers[i] = [x, y]
    
    def objective(r):
        return -np.sum(r)
    
    def constraint_sum(r):
        return np.sum(r) - 2.636

        for i in range(n):

                    return -1
        return 0
    
    cons = [{'type': 'ineq', 'fun': constraint_overlap},
            {'type': 'eq', 'fun': constraint_sum}]
    
    initial_r = np.min(squareform(pdist(centers))) * np.ones(n)
    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]

            result['x'][i] = 0
