import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Hexagonal lattice with 5 rows and 5 or 6 columns
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        if row % 2 == 0:
            x = col + 0.5
        else:
            x = col + 0.5 - 0
        y = row + 0.5
        centers[i] = [x, y]
    
    # Initial scaling to fit within [0,1]^2
    min_coords = np.min(centers, axis=0)
    max_coords = np.max(centers, axis=0)
    scale = (1 - 2 * 0.1) / (max_coords - min_coords)
    centers = (centers - min_coords) * scale + 0.1
    centers = np.clip(centers, 0.1, 0.9)
    
    def objective(r):
        return -np.sum(r)
    
    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    def constraint1(r):
        return np.sum(r)
    
    def constraint2(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    # Initial guess based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    
    # Optimization
    cons = [{'type': 'ineq', 'fun': constraint1}]

    cons.append({'type': 'ineq', 'fun': constraint2})
    
    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)
    
    # Post-processing to ensure boundaries
    radii = result['x']
    for i in range(n):
        x, y = centers[i]
        r = radii[i]

                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0
    
    return centers, radii, np.sum(radii)
