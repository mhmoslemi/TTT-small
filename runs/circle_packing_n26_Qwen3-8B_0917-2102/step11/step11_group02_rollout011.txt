import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    row_height = 1.0 / (np.sqrt(3) * 2)
    col_width = 1.0 / 5
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col * col_width + 0.5 * (1 - row % 2) * col_width
        y = row * row_height + 0.5 * row_height
        centers[i] = [x, y]
    
    # Adaptive boundary-aware scaling
    def scale_centers(centers):
        min_coords = np.min(centers, axis=0)
        max_coords = np.max(centers, axis=0)
        scale = (1.0 - 1e-3) / (max_coords - min_coords)
        scaled_centers = (centers - min_coords) * scale + 1e-3
        return np.clip(scaled_centers, 1e-3, 1.0 - 1e-3)
    centers = scale_centers(centers)

    def objective(r):
        return -np.sum(r)

    def constraint_sum(r):
        return np.sum(r) - 2.636

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_sum}]
    cons.append({'type': 'ineq', 'fun': constraint_overlap})

    initial_r = np.min(squareform(pdist(centers))) * np.ones(n)
    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for overlap with boundary-aware scaling
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2 with boundary padding
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
