import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Define a more compact staggered hexagonal grid with variable row heights
    rows = 5
    cols = 6
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // cols
        col = i % cols
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row + 0.5 * (1 - row % 2)
        centers[i] = [x, y]
    
    # Compute initial minimum distances and scale to fit within square
    dists = squareform(pdist(centers))
    min_dist = np.min(dists)
    scale = 1 / (np.max(centers, axis=0) - np.min(centers, axis=0))
    centers = centers * scale - np.min(centers, axis=0) * scale + 0.1
    centers = np.clip(centers, 0.1, 0.9)

    def objective(r):
        return -np.sum(r)

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    # Initial guess based on initial minimum inter-circle distance
    initial_r = np.min(dists) * np.ones(n)

    # Optimization with boundary constraints and overlap check
    cons = [{'type': 'ineq', 'fun': check_overlap}]
    bounds = [(0, 1)] * n

    result = minimize(objective, initial_r, constraints=cons, bounds=bounds, method='SLSQP')

    # Final adjustment for overlap with boundary-aware scaling
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2 with boundary padding
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
