import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate a hexagonal grid with variable row heights
    rows = 6
    cols = 5
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // cols
        col = i % cols
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    # Adjust for boundary-aware packing
    def adjust_boundary(r):
        for i in range(n):
            x, y = centers[i]
            r_val = r[i]
            if x - r_val < -1e-12:
                r[i] = x - 1e-12
            elif x + r_val > 1 + 1e-12:
                r[i] = 1 - x - 1e-12
            if y - r_val < -1e-12:
                r[i] = y - 1e-12
            elif y + r_val > 1 + 1e-12:
                r[i] = 1 - y - 1e-12
        return r

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 2.636

    cons = [{'type': 'ineq', 'fun': lambda r: np.sum(r)}]
    cons.append({'type': 'eq', 'fun': constraint})

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons.append({'type': 'ineq', 'fun': check_overlap})

    # Initial guess based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for overlap with boundary-aware scaling
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):

                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    for i in range(n):

        r = result['x'][i]

            result['x'][i] = 0

    # Boundary-aware adjustment
    result['x'] = adjust_boundary(result['x'].copy())

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
