import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create staggered hexagonal grid
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    # Calculate initial minimum distances
    dists = squareform(pdist(centers))
    min_dist = np.min(dists)
    initial_r = np.min(dists) * np.ones(n)
    
    # Objective function to maximize sum of radii
    def objective(r):
        return -np.sum(r)
    
    # Constraints: non-overlapping and within unit square
    def constraint1(r):
        return np.sum(r) - 2.636
    
    def constraint2(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    cons = [{'type': 'ineq', 'fun': constraint1}]
    cons.append({'type': 'ineq', 'fun': constraint2})
    
    # Bounds for radii
    bounds = [(0, 1)] * n
    
    # Optimization
    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=bounds)
    
    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0
    
    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
