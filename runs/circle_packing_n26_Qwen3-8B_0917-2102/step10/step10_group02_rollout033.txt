import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Hexagonal grid with staggered rows
    centers = np.empty((n, 2))
    row_height = 1.0 / np.sqrt(3)
    col_width = 1.0 / np.sqrt(2)
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col * col_width
        y = row * row_height
        if row % 2 == 1:
            x += col_width / 2
        centers[i] = [x, y]
    
    # Boundary-aware scaling
    min_coords = np.min(centers, axis=0)
    max_coords = np.max(centers, axis=0)
    scale = (1 - 2 * 0.1) / (max_coords - min_coords)
    centers = (centers - min_coords) * scale + 0.1
    centers = np.clip(centers, 0.1, 0.9)

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 2.636

    cons = [{'type': 'ineq', 'fun': lambda r: np.sum(r)}]
    cons.append({'type': 'eq', 'fun': constraint})

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons.append({'type': 'ineq', 'fun': check_overlap})

    # Initial guess based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Refine via simulated annealing
    def simulated_annealing():
        temp = 1.0
        cooling = 0.995
        while temp > 1e-6:
            for i in range(n):
                delta = np.random.uniform(-0.01, 0.01)
                new_r = result['x'][i] + delta
                if new_r < 0:
                    continue
                new_rads = result['x'].copy()
                new_rads[i] = new_r
                dists = squareform(pdist(centers))
                overlap = False
                for j in range(n):
                    for k in range(j + 1, n):
                        if dists[j, k] < new_rads[j] + new_rads[k] - 1e-8:
                            overlap = True
                            break
                    if overlap:
                        break
                if not overlap:
                    result['x'] = new_rads
            temp *= cooling

    simulated_annealing()

    # Ensure circles are within [0,1]^2 with boundary padding
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
