import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate initial staggered hexagonal grid
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    # Define objective and constraints
    def objective(r):
        return -np.sum(r)
    
    def constraint_sum(r):
        return np.sum(r) - 2.636
    
    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    cons.append({'type': 'ineq', 'fun': constraint_overlap})
    
    # Initial guess based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    
    # Optimize with bounds and constraints
    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)
    
    # Final adjustment for boundary and overlap
    final_r = result['x'].copy()
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < final_r[i] + final_r[j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (final_r[i] + final_r[j])
                final_r[i] *= ratio
                final_r[j] *= ratio
    
    # Ensure circles are within [0,1]^2 with boundary padding
    for i in range(n):
        x, y = centers[i]
        r = final_r[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            final_r[i] = 0
    
    sum_radii = np.sum(final_r)
    return centers, final_r, sum_radii
