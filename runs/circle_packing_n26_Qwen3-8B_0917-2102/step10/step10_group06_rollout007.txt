import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n) * 0.1

    def objective(r):
        return -np.sum(r)

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    def boundary_check(r):
        for i in range(n):
            x, y = centers[i]
            r_i = r[i]
            if (x - r_i < -1e-12 or x + r_i > 1 + 1e-12
                    or y - r_i < -1e-12 or y + r_i > 1 + 1e-12):
                return -1
        return 0

    cons = [{'type': 'ineq', 'fun': check_overlap},
            {'type': 'ineq', 'fun': boundary_check}]

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for boundary proximity
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if x - r < -1e-12:
            result['x'][i] = x - 1e-12
        elif x + r > 1 + 1e-12:
            result['x'][i] = 1 - 1e-12
        if y - r < -1e-12:
            result['x'][i] = y - 1e-12
        elif y + r > 1 + 1e-12:
            result['x'][i] = 1 - 1e-12

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
