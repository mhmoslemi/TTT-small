import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Define centers with more optimal boundary placement
    centers = np.zeros((n, 2))
    for i in range(n):
        if i < 13:
            centers[i, 0] = i / 12
            centers[i, 1] = 0.5
        else:
            centers[i, 0] = (n - i - 1) / 12
            centers[i, 1] = 0.5

    # Set up initial radii
    initial_radii = np.full(n, 0.1)

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 2.636

    cons = [{'type': 'ineq', 'fun': lambda r: np.array([r[i] for i in range(n)])}]
    cons.append({'type': 'eq', 'fun': constraint})

    def check_overlap(r):

        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons.append({'type': 'ineq', 'fun': check_overlap})

    result = minimize(objective, initial_radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for overlap with boundary-aware scaling
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:

                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2 with boundary padding

        if (x - r < -1e-12 or x + r > 1 + 1e-12

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
