import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Hexagonal packing with adaptive scaling
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 2
        col = i % 2
        x = col + 0.5 * (1 - 1e-6)
        y = row + 0.5 * (1 - 1e-6)
        centers[i] = [x, y]
    
    # Normalize to [0,1]^2 with boundary padding
    padding = 0.1
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))
    centers = np.clip(centers, padding, 1 - padding)

    def objective(r):
        return -np.sum(r)

    def constraint_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': constraint_overlap}]

    result = minimize(objective, np.full(n, 0.1), constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Ensure circles are within [0,1]^2 with boundary padding
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
