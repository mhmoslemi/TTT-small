from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

        col = i % cols
        offset = 0.5 if row % 2 else 0

        y = row + 0.5

    padding = 0.1

        return -np.sum(r)

        return np.sum(r) - 2.636

    cons.append({'type': 'eq', 'fun': constraint})

        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons.append({'type': 'ineq', 'fun': check_overlap})

    # Initial guess with adaptive scaling based on distances
    dists = squareform(pdist(centers))
    initial_radius = 0.1
    r0 = np.min(dists) * initial_radius
