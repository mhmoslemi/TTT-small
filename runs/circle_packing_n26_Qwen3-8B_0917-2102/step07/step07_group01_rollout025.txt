import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate initial staggered hexagonal packing with 5 columns and 6 rows
    cols = 5
    rows = 6
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // cols
        col = i % cols
        offset = 0.5 if row % 2 else 0
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    # Normalize to [0,1]^2 with boundary padding
    padding = 0.1
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))
    centers = np.clip(centers, padding, 1 - padding)

    def objective(r):
        return -np.sum(r)
    
    def constraint(r):
        return np.sum(r) - 2.636
    
    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    # Initial guess based on minimum distance between circles
    dists = squareform(pdist(centers))
    initial_radius = 0.1
    r0 = np.min(dists) * initial_radius
    
    # Optimization with bounds and constraints
    cons = [{'type': 'ineq', 'fun': check_overlap}]
    cons.append({'type': 'eq', 'fun': constraint})
    
    result = minimize(objective, np.full(n, r0), bounds=[(0, 1)] * n, constraints=cons, method='SLSQP')
    
    # Post-processing to ensure circles are within bounds
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0
    
    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
