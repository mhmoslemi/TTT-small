from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    boundary_circles = 12

    padding = 0.1

        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 2.636

    cons.append({'type': 'eq', 'fun': constraint})
