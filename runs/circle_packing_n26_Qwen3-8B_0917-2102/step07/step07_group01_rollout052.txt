from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

    min_vals = centers.min(axis=0)
    max_vals = centers.max(axis=0)
    centers = (centers - min_vals) / (max_vals - min_vals)
    padding = 0.1
    centers = np.clip(centers, padding, 1 - padding)

        return -np.sum(r)

        return np.sum(r) - 2.636

    cons.append({'type': 'eq', 'fun': constraint})
