import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Custom arrangement with denser clusters near boundaries
    centers = np.zeros((n, 2))
    for i in range(n):
        if i < 13:
            # Left cluster
            x = 0.2 + i * 0.07
            y = 0.5 + np.sin(i * 0.5) * 0.2
        else:
            # Right cluster
            x = 0.8 - (i - 13) * 0.07
            y = 0.5 + np.sin((i - 13) * 0.5) * 0.2
        centers[i] = [x, y]

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 2.636

    # Constraints
    cons = []
    cons.append({'type': 'ineq', 'fun': lambda r: np.sum(r)})
    cons.append({'type': 'eq', 'fun': constraint})

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons.append({'type': 'ineq', 'fun': check_overlap})

    # Initial guess with adaptive radius based on distances
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    result = minimize(objective, initial_r, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Adjust radii to ensure all circles are within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
