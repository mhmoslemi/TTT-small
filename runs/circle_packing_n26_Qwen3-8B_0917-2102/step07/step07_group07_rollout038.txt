from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Generate hexagonal grid with adjusted boundary padding
    cols = 5
    rows = 6
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // cols
        col = i % cols
        offset = 0.5 if row % 2 else 0
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    # Normalize to [0,1]^2 with boundary padding
    padding = 0.1
    min_vals = centers.min(axis=0)
    max_vals = centers.max(axis=0)
    centers = (centers - min_vals) / (max_vals - min_vals)
    centers = np.clip(centers, padding, 1 - padding)

    def objective(r):
        return -np.sum(r)

    # Define constraints for non-negative radii and target sum
    cons = [{'type': 'ineq', 'fun': lambda r: r}]

    def constraint_sum(r):
        return np.sum(r) - 2.636

    cons.append({'type': 'eq', 'fun': constraint_sum})

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons.append({'type': 'ineq', 'fun': check_overlap})

    # Initial guess with adaptive radius based on distances
    initial_guess = np.full(n, 0.1)
    dists = squareform(pdist(centers))
    for i in range(n):
        initial_guess[i] = 0.1 * np.min(dists[i, :i] + dists[i, i+1:])

    result = minimize(objective, initial_guess, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for overlap with boundary-aware scaling
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):

                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2 with boundary padding
    for i in range(n):

        if (x - r < -1e-12 or x + r > 1 + 1e-12
