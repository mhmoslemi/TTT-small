from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    centers = np.random.uniform(size=(n, 2))
    radii = np.full(n, 0.1)

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 2.636

    def overlap_constraint(r):

        for i in range(n):

        return 0

    cons.append({'type': 'eq', 'fun': lambda r: np.sum(r) - 2.636})

    return centers, result['x'], np.sum(result['x'])
