import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 2.636

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [{'type': 'ineq', 'fun': check_overlap}]
    cons.append({'type': 'eq', 'fun': lambda r: np.sum(r) - 2.636})

    def update_centers(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    dx, dy = centers[j] - centers[i]
                    dist = np.hypot(dx, dy)
                    if dist == 0:
                        continue
                    shift = (r[i] + r[j] - dist) / dist
                    centers[i] += shift * dx / 2
                    centers[j] -= shift * dx / 2
                    shift = (r[i] + r[j] - dist) / dist
                    centers[i] += shift * dy / 2
                    centers[j] -= shift * dy / 2

    result = minimize(objective, radii, constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    update_centers(result['x'])

    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            result['x'][i] = 0

    return centers, result['x'], np.sum(result['x'])
