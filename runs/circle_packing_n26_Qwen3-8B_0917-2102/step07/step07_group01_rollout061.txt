import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance_matrix

def run_packing():
    n = 26
    # Generate hexagonal lattice with 5 columns and 6 rows
    centers = np.zeros((n, 2))
    for i in range(n):
        col = i % 5
        row = i // 5
        offset = 0.5 if row % 2 else 0
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    # Normalize to unit square with padding
    padding = 0.1
    min_vals = centers.min(axis=0)
    max_vals = centers.max(axis=0)
    centers = (centers - min_vals) / (max_vals - min_vals)
    centers = np.clip(centers, padding, 1 - padding)

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        return np.sum(r) - 2.636

    cons = [{'type': 'ineq', 'fun': lambda r: np.array([r[i] for i in range(n)])}]
    cons.append({'type': 'eq', 'fun': constraint})

    def check_overlap(r):
        dists = distance_matrix(centers, centers)
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons.append({'type': 'ineq', 'fun': check_overlap})

    # Initial guess based on minimal pairwise distance
    dists = distance_matrix(centers, centers)
    initial_radius = 0.1
    r0 = np.min(dists[np.triu_indices(n, 1)]) * initial_radius
    result = minimize(objective, np.full(n, r0), constraints=cons, method='SLSQP', bounds=[(0, 1)] * n)

    # Final boundary-aware adjustment
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
