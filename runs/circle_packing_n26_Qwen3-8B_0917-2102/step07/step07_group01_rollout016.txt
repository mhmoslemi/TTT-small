from scipy.optimize import dual_annealing
from scipy.spatial.distance import pdist, squareform

    cols = 5

        col = i % cols
        offset = 0.5 if row % 2 else 0

                    return -1
        return 0

        return np.array([0.0 for _ in range(n)])

            r_val = r[i]
            if (x - r_val < -1e-12 or x + r_val > 1 + 1e-12
