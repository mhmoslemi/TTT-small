import numpy as np
from scipy.optimize import differential_evolution
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial randomized centers within unit square
    centers = np.random.rand(n, 2)
    
    def objective(r):
        return -np.sum(r)
    
    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636
    
    # Bounds for radii
    bounds = [(0, 1)] * n
    
    # Genetic algorithm with adaptive mutation
    result = differential_evolution(
        fun=objective,
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': check_overlap}, {'type': 'eq', 'fun': constraint}],
        maxiter=100,
        popsize=20,
        mutation=(0.5, 1),
        recombination='arithmetic',
        tol=1e-8,
        polish=True
    )
    
    # Ensure circles are within [0,1]^2 with boundary padding
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result.x[i] = 0
    
    sum_radii = np.sum(result.x)
    return centers, result.x, sum_radii
