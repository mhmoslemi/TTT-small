from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Generate staggered hexagonal grid with boundary-aware spacing
    centers = []

        col = i % 5
        x = col + 0.5 * (row % 2)
        y = row + 0.5
        centers.append([x, y])
    centers = np.array(centers)
    
    # Normalize to unit square with dynamic boundary padding
    def compute_boundary_padding(centers):
        min_x, max_x = np.min(centers[:, 0]), np.max(centers[:, 0])
        min_y, max_y = np.min(centers[:, 1]), np.max(centers[:, 1])
        pad_x = (1 - max_x) / (max_x - min_x) if (max_x - min_x) > 1e-12 else 0.1
        pad_y = (1 - max_y) / (max_y - min_y) if (max_y - min_y) > 1e-12 else 0.1
        return pad_x, pad_y

    centers[:, 0] = (centers[:, 0] - min_x) / (max_x - min_x) * (1 - pad_x)
    centers[:, 1] = (centers[:, 1] - min_y) / (max_y - min_y) * (1 - pad_y)
    centers[:, 0] += pad_x / 2
    centers[:, 1] += pad_y / 2

    def objective(r):
        return -np.sum(r)

                    return -1
        return 0

                    return -np.inf

    # Initial guess based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': check_overlap}, {'type': 'eq', 'fun': constraint}], method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for overlap with boundary-aware scaling
    dists = squareform(pdist(centers))
    for i in range(n):

                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2 with boundary padding
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
