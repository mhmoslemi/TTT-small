from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

        centers[i] = [x, y]
    
    def objective(r):
        return -np.sum(r)

        for i in range(n):

    dists = squareform(pdist(centers))

    for i in range(n):
