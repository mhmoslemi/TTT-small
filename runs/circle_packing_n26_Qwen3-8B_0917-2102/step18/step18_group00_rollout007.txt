import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Create adaptive staggered grid with boundary-aware spacing
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (row % 2) * 0.8
        y = row + 0.5
        centers[i] = [x, y]
    
    # Initial radius guess based on minimum distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    
    def objective(r):
        return -np.sum(r)
    
    def constraint(r):
        dists = squareform(pdist(centers))

                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636

    bounds = [(0, 1)] * n
    constraints = [{'type': 'ineq', 'fun': lambda r: np.sum(r) - 2.636}]
    
    result = minimize(objective, initial_r, bounds=bounds, constraints=constraints, method='SLSQP')
    
    # Final adjustment for boundary constraints
    for i in range(n):

        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12):

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
