import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Generate a more compact staggered hexagonal grid
    centers = []
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (row % 2)
        y = row + 0.5
        centers.append([x, y])
    centers = np.array(centers)
    
    # Adjust grid to be more compact and closer to boundaries
    for i in range(n):
        if i % 5 == 0:
            centers[i, 0] = 0.1
        elif i % 5 == 4:
            centers[i, 0] = 0.9
        if i // 5 % 2 == 0:
            centers[i, 1] = 0.1
        else:
            centers[i, 1] = 0.9

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636

    # Initial guess based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': lambda r: np.sum(r)}], method='SLSQP', bounds=[(0, 1)] * n)

    # Adjust radii to ensure all circles are within the unit square
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            result['x'][i] = 0

    # Final adjustment for overlap with boundary-aware scaling
    dists = squareform(pdist(centers))

        for j in range(i + 1, n):

                result['x'][i] *= ratio
                result['x'][j] *= ratio

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
