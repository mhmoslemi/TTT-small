import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create optimized hexagonal packing with boundary-aware placement
    centers = np.array([
        [0.1, 0.1],
        [0.3, 0.1],
        [0.5, 0.1],
        [0.7, 0.1],
        [0.9, 0.1],
        [0.1, 0.3],
        [0.3, 0.3],
        [0.5, 0.3],
        [0.7, 0.3],
        [0.9, 0.3],
        [0.1, 0.5],
        [0.3, 0.5],
        [0.5, 0.5],
        [0.7, 0.5],
        [0.9, 0.5],
        [0.1, 0.7],
        [0.3, 0.7],
        [0.5, 0.7],
        [0.7, 0.7],
        [0.9, 0.7],
        [0.1, 0.9],
        [0.3, 0.9],
        [0.5, 0.9],
        [0.7, 0.9],
        [0.9, 0.9]
    ])

    # Initial radii based on minimum distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    result = minimize(lambda r: -np.sum(r), initial_r, method='SLSQP',
                      bounds=[(0, 1)] * n,
                      constraints=[
                          {'type': 'ineq', 'fun': lambda r: np.min(dists) - np.sum(np.minimum(r, np.min(dists) - 1e-8))},
                          {'type': 'ineq', 'fun': lambda r: np.min(np.array([np.min(centers[:, 0] - r[np.newaxis, :], axis=1)), 
                                                                       np.min(centers[:, 1] - r[np.newaxis, :], axis=1)), axis=0)}
                      ])

    # Final adjustment for boundary and overlaps
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2 with boundary padding
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
