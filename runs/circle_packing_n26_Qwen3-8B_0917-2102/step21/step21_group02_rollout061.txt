from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():

    centers = []

            y = row + 0.5
            centers.append([x, y])
    centers = np.array(centers)
    centers = (centers - np.min(centers, axis=0)) / (np.max(centers, axis=0) - np.min(centers, axis=0))

    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)

        return -np.sum(r)
    
    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):

                    return -np.inf
        return np.sum(r)
    
    # Optimization with bounds and constraints
    bounds = Bounds(0, 1)
    result = minimize(objective, initial_r, method='SLSQP', bounds=bounds, constraints=[{'type': 'ineq', 'fun': constraint}])
    
    # Final adjustment for boundary constraints
    final_r = result['x'].copy()
    for i in range(n):
        x, y = centers[i]
        r = final_r[i]

    sum_radii = np.sum(final_r)
    return centers, final_r, sum_radii
