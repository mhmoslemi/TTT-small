from scipy.optimize import minimize, LinearConstraint, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Generate hexagonal lattice with staggered rows
    centers = []
    for row in range(5):
        for col in range(5 + (row % 2)):
            x = col + 0.5 * (row % 2)
            y = row + 0.5
            centers.append([x, y])

    max_coord = np.max(centers, axis=0)

    # Initial radius estimate
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)

        return -np.sum(r)

                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf

    cons = []
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            cons.append(LinearConstraint(ranges=(0, np.inf),

    cons.append(LinearConstraint(np.ones(n), lb=0, ub=2.636, name='sum_radius'))

    # Optimization bounds
    bounds = Bounds(np.zeros(n), np.full(n, 1))
