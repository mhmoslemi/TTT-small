from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Hexagonal grid with boundary-aware placement
    centers = []
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (row % 2)
        y = row + 0.5
        centers.append([x, y])
    centers = np.array(centers)

    max_coords = np.max(centers, axis=0)

        return -np.sum(r)

        dists = squareform(pdist(centers))
        for i in range(n):

                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636

    # Initial guess based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': lambda r: -np.sum(np.maximum(dists - r[:, np.newaxis] - r[np.newaxis, :], 0))}], method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for boundary and overlaps
    final_centers = centers.copy()
    final_radii = result['x'].copy()

        x, y = final_centers[i]

            final_radii[i] = 0

                ratio = (dist - 1e-8) / (final_radii[i] + final_radii[j])

    sum_radii = np.sum(final_radii)
