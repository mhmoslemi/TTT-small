from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Generate staggered hexagonal grid with adjustable spacing
    centers = []
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (row % 2)
        y = row + 0.5
        centers.append([x, y])
    centers = np.array(centers)
    
    # Initial radius guess based on minimum distance
    dists = squareform(pdist(centers))
    min_dist = np.min(dists)
    initial_r = np.ones(n) * min_dist * 0.8
    
    def objective(r):
        return -np.sum(r)
    
    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636
    
    # Add boundary constraints
    bounds = [(0, 1) for _ in range(n)]
    
    # Optimization with SLSQP
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': constraint}], method='SLSQP', bounds=bounds)
    
    # Final adjustment to ensure circles within [0,1]^2
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
