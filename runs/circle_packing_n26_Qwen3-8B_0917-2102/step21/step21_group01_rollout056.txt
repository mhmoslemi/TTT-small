import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal lattice with variable row spacing
    centers = []
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (row % 2)
        y = row + 0.5
        centers.append([x, y])
    centers = np.array(centers)
    
    # Normalize to unit square with boundary padding
    min_coords = np.min(centers, axis=0)
    max_coords = np.max(centers, axis=0)
    scale = (1 - 2 * 0.1) / (max_coords - min_coords)
    centers = (centers - min_coords) * scale + 0.1
    centers = np.clip(centers, 0.1, 0.9)

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r)

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    def bounds_func(x):
        return np.clip(x, 0, 1)

    initial_r = np.min(squareform(pdist(centers))) * np.ones(n)
    result = minimize(objective, initial_r, bounds=[(0, 1)] * n, constraints=[{'type': 'ineq', 'fun': check_overlap}], method='L-BFGS-B')

    # Final adjustment for boundary constraints
    for i in range(n):

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
