import numpy as np
from scipy.optimize import differential_evolution

def run_packing():
    n = 26
    # Initial guess with hexagonal grid
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    def fitness(params):
        r = params[:n]
        if np.any(r < 0):
            return -np.inf
        # Check boundary constraints
        for i in range(n):
            x, y = centers[i]
            if (x - r[i] < -1e-12 or x + r[i] > 1 + 1e-12
                or y - r[i] < -1e-12 or y + r[i] > 1 + 1e-12):
                return -np.inf
        # Check circle overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < r[i] + r[j] - 1e-12:
                    return -np.inf
        return -np.sum(r)
    
    result = differential_evolution(
        fitness,
        bounds=[(0, 1)] * n,
        strategy='best1bin',
        popsize=20,
        maxiter=100,
        tol=1e-10,
        mutation=(0.5, 1),
        recombination=0.8
    )
    
    radii = result.x
    # Ensure circles are within [0,1]^2 with boundary padding
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
