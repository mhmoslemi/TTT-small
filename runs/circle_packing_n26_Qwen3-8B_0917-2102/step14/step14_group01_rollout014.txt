import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create staggered hexagonal grid with variable row spacing
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (row % 2)
        y = row + 0.5
        centers[i] = [x, y]
    
    # Calculate initial distances between all pairs
    dists = squareform(pdist(centers))
    min_dist = np.min(dists)
    
    # Initial guess for radii based on minimum distance
    initial_r = np.full(n, min_dist / 2)
    
    # Optimization objective: maximize sum of radii
    def objective(r):
        return -np.sum(r)
    
    # Constraint: ensure circles do not overlap
    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636
    
    # Optimization bounds
    bounds = [(0, 1)] * n
    
    # Run optimization with SLSQP
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': constraint}], method='SLSQP', bounds=bounds)
    
    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0
    
    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
