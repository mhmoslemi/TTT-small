import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.01)

    # First, optimize boundary circles to maximize radius while staying within bounds
    def boundary_objective(r):
        return -np.sum(r)
    
    def boundary_constraint(r):
        for i in range(n):
            x, y = centers[i]
            if (x - r[i] < -1e-12 or x + r[i] > 1 + 1e-12
                or y - r[i] < -1e-12 or y + r[i] > 1 + 1e-12):
                return -np.inf
        return 0
    
    bounds = Bounds(0, 1)
    result = minimize(boundary_objective, radii, bounds=bounds, constraints=[{'type': 'ineq', 'fun': boundary_constraint}])
    radii = result['x']

    # Second, use adaptive hexagonal packing with repulsion forces to optimize inter-circle spacing
    def objective(r):
        return -np.sum(r)
    
    def inter_circle_constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return 0
    
    bounds = Bounds(0, 1)
    result = minimize(objective, radii, bounds=bounds, constraints=[{'type': 'ineq', 'fun': inter_circle_constraint}])
    radii = result['x']

    # Final check to ensure boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            radii[i] = 0

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
