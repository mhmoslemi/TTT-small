from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

    centers = np.zeros((n, 2))

    col = 0

        if i % 5 == 0:
            row += 1
            col = 0
        centers[i] = [col + 0.25, row + 0.5]
        col += 1
    
    def objective(r):
        return -np.sum(r)

        dists = squareform(pdist(centers))

                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf

    # Initial guess based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    
    # Optimization setup
    bounds = [(0, 1) for _ in range(n)]
    constraints = [{'type': 'ineq', 'fun': lambda r: np.sum(r) - 2.636}]
    
    result = minimize(objective, initial_r, constraints=constraints, bounds=bounds, method='SLSQP')
    
    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
