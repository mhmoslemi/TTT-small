import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import matplotlib.pyplot as plt

def run_packing():
    n = 26
    # Create hexagonal grid with optimal row-column distribution
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (1 - row % 2)
        y = row + 0.5
        centers[i] = [x, y]
    
    # Apply boundary-aware scaling with padding
    min_coords = np.min(centers, axis=0)
    max_coords = np.max(centers, axis=0)
    scale = (1 - 2 * 0.1) / (max_coords - min_coords)
    centers = (centers - min_coords) * scale + 0.1
    centers = np.clip(centers, 0.1, 0.9)

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636

    # Initial guess based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': lambda r: np.sum(r) - 2.636}], method='SLSQP', bounds=[(0, 1)] * n)

    # Refine with simulated annealing
    def simulated_annealing(r):
        T = 1.0
        Tmin = 1e-6
        iter = 1000
        for step in range(iter):
            T *= 0.995
            if T < Tmin:
                break
            idx = np.random.randint(0, n)
            delta = np.random.uniform(-0.01, 0.01)
            r_new = r.copy()
            r_new[idx] += delta
            r_new = np.clip(r_new, 0, 1)
            dists = squareform(pdist(centers))
            overlap = False
            for i in range(n):
                for j in range(i + 1, n):
                    if dists[i, j] < r_new[i] + r_new[j] - 1e-8:
                        overlap = True
                        break
                if overlap:
                    break
            if not overlap:
                if np.sum(r_new) > np.sum(r):
                    r = r_new
        return r

    result['x'] = simulated_annealing(result['x'])

    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
