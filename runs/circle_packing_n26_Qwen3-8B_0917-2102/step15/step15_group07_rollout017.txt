import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate staggered hexagonal grid with variable row spacing
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    # Compute initial minimum distances for radius scaling
    dists = squareform(pdist(centers))
    min_dist = np.min(dists)
    initial_r = np.ones(n) * min_dist / 2

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636

    # Add boundary constraints
    def boundary_constraint(r):
        for i in range(n):
            x, y = centers[i]
            r_i = r[i]
            if x - r_i < -1e-12 or x + r_i > 1 + 1e-12 or y - r_i < -1e-12 or y + r_i > 1 + 1e-12:
                return -np.inf
        return np.sum(r)

    # Optimization with boundary constraints
    result = minimize(objective, initial_r, constraints=[
        {'type': 'ineq', 'fun': constraint},
        {'type': 'ineq', 'fun': boundary_constraint}
    ], method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for boundary and overlap
    dists = squareform(pdist(centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < result['x'][i] + result['x'][j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (result['x'][i] + result['x'][j])
                result['x'][i] *= ratio
                result['x'][j] *= ratio

    # Ensure circles are within [0,1]^2 with boundary padding
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
