from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Generate a hexagonal grid with variable row spacing
    centers = np.zeros((n, 2))

    col = 0
    for i in range(n):
        if i % 5 == 0 and i != 0:
            row += 1
            col = 0
        centers[i] = [0.1 + col * 0.2, 0.1 + row * 0.3]
        col += 1
    
    def objective(r):
        return -np.sum(r)
    
    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):

                    return -np.inf
        return np.sum(r) - 2.636
    
    # Initial guess based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': lambda r: np.sum(r)}], method='SLSQP', bounds=[(0, 1)] * n, tol=1e-8, options={'ftol': 1e-8, 'maxiter': 500})
    
    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            result['x'][i] = 0
    
    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
