import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import differential_evolution

def run_packing():
    n = 26
    # Generate staggered hexagonal grid with boundary-aware placement
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row
        centers[i] = [x, y]
    
    def objective(r):
        return -np.sum(r)
    
    def overlap_constraints(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r)
    
    def boundary_constraints(r):
        for i in range(n):
            x, y = centers[i]
            r_i = r[i]
            if x - r_i < -1e-12 or x + r_i > 1 + 1e-12 or y - r_i < -1e-12 or y + r_i > 1 + 1e-12:
                return -np.inf
        return np.sum(r)
    
    bounds = [(0, 1)] * n
    
    result = differential_evolution(objective, bounds, constraints=[
        {'type': 'ineq', 'fun': overlap_constraints},
        {'type': 'ineq', 'fun': boundary_constraints}
    ])
    
    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
