import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate staggered hexagonal grid with boundary-aware placement
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row
        centers[i] = [x, y]
    
    # Apply boundary-aware scaling and adjust for corners
    min_coords = np.min(centers, axis=0)
    max_coords = np.max(centers, axis=0)
    scale = (1 - 2 * 0.1) / (max_coords - min_coords)
    centers = (centers - min_coords) * scale + 0.1
    centers = np.clip(centers, 0.1, 0.9)

    def objective(r):
        return -np.sum(r)

    def check_overlap(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r)

    # Initial guess based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)

    # First optimization pass
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': check_overlap}], method='SLSQP', bounds=[(0, 1)] * n)

    # Second optimization pass with boundary-aware constraints
    def boundary_constraint(r):
        for i in range(n):

            if (x - r[i] < -1e-12 or x + r[i] > 1 + 1e-12
                    or y - r[i] < -1e-12 or y + r[i] > 1 + 1e-12):
                return -np.inf
        return np.sum(r)

    result = minimize(objective, result['x'], constraints=[{'type': 'ineq', 'fun': check_overlap}, {'type': 'ineq', 'fun': boundary_constraint}], method='SLSQP', bounds=[(0, 1)] * n)

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
