from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # Hexagonal grid with variable row spacing
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    # Bound circles to unit square with padding
    bounds = np.array([[0.1, 0.9]] * 2)
    centers = np.clip(centers, bounds[0], bounds[1])

    def objective(r):
        return -np.sum(r)

        for i in range(n):

                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf

    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': lambda r: np.inf if np.any(pdist(centers) < r[np.newaxis, :] + r[:, np.newaxis] - 1e-8) else 0}], method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for boundary constraints
    for i in range(n):

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
