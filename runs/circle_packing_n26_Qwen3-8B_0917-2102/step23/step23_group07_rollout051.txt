import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate hexagonal grid with staggered rows
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (row % 2)
        y = row + 0.5
        centers[i] = [x, y]
    
    # Compute initial minimum distances
    dists = squareform(pdist(centers))
    min_dist = np.min(dists)
    
    # Initial radius guess based on minimum distance
    initial_r = np.full(n, min_dist / 2)
    
    # Objective function to maximize sum of radii
    def objective(r):
        return -np.sum(r)
    
    # Constraint to ensure non-overlapping circles
    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r)
    
    # Bounds for radii
    bounds = [(0, 1) for _ in range(n)]
    
    # Optimization using SLSQP
    result = minimize(objective, initial_r, bounds=bounds, constraints=[{'type': 'ineq', 'fun': constraint}], method='SLSQP')
    
    # Final adjustment to ensure circles stay within unit square
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0
    
    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
