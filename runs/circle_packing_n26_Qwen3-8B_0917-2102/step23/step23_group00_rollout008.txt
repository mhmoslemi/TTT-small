import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Define cluster-based layout with adaptive spacing
    clusters = []
    for i in range(2):
        for j in range(13):
            if i == 0:
                x = 0.2 + j * 0.1
                y = 0.2 + 0.1 * i
            else:
                x = 0.7 + j * 0.1
                y = 0.7 + 0.1 * i
            clusters.append([x, y])
    centers = np.array(clusters)
    
    # Initial radius guess based on cluster spacing
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    
    def objective(r):
        return -np.sum(r)
    
    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636
    
    # Optimization with boundary-aware constraints
    result = minimize(
        objective,
        initial_r,
        constraints=[{'type': 'ineq', 'fun': lambda r: np.inf if np.any(r < 0) else 0},
                    {'type': 'eq', 'fun': constraint}],
        method='SLSQP',
        bounds=[(0, 1)] * n
    )
    
    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0
    
    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
