import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create a hexagonal grid with staggered rows
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (row % 2)
        y = row + 0.5
        centers[i] = [x, y]
    
    # Initial radii based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    
    def objective(r):
        return -np.sum(r)
    
    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636
    
    # Optimization with boundary-aware constraints
    bounds = [(0, 1)] * n
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': lambda r: np.sum(r) - 2.636}], method='SLSQP', bounds=bounds)
    
    # Final adjustment for boundary and overlap
    final_centers = np.copy(centers)
    final_radii = np.copy(result['x'])
    
    # Boundary padding
    for i in range(n):
        x, y = final_centers[i]
        r = final_radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            final_radii[i] = 0
    
    # Ensure no overlap
    dists = squareform(pdist(final_centers))
    for i in range(n):
        for j in range(i + 1, n):
            if dists[i, j] < final_radii[i] + final_radii[j] - 1e-8:
                ratio = (dists[i, j] - 1e-8) / (final_radii[i] + final_radii[j])
                final_radii[i] *= ratio
                final_radii[j] *= ratio
    
    return final_centers, final_radii, np.sum(final_radii)
