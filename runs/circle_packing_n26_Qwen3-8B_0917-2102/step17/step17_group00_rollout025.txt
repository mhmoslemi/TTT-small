from scipy.spatial.distance import pdist, squareform

                col += 1
    centers[15] = [0.5, 0.5]  # Center circle

            for j in range(i + 1, n):

    initial_r = np.min(dists) * np.ones(n)

    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': constraint}], method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for boundary and overlap
    for i in range(n):
        x, y = centers[i]
