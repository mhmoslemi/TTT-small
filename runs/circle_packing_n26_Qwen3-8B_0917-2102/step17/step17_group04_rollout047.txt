import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial grid with staggered rows
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    # Initial radius guess based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)

    def objective(r):
        return -np.sum(r)

    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636

    # Optimization with boundary-aware constraints
    bounds = [(0, 1)] * n
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': constraint}], method='SLSQP', bounds=bounds)

    # Final adjustment for boundary and overlap
    final_centers = centers.copy()
    final_radii = result['x'].copy()

    # Boundary padding
    for i in range(n):
        x, y = final_centers[i]
        r = final_radii[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            final_radii[i] = min(r, 1 - np.min([x, 1 - x, y, 1 - y]))

    # Final check and return
    return final_centers, final_radii, np.sum(final_radii)
