import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Generate staggered hexagonal grid with adaptive row spacing
    centers = np.zeros((n, 2))
    row_height = 0.7
    col_width = 0.7
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col * col_width + 0.5 * (1 - row % 2) * col_width
        y = row * row_height + 0.5
        centers[i] = [x, y]
    
    # Calculate initial minimum distances for radius estimation
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    
    # Define objective function to maximize sum of radii
    def objective(r):
        return -np.sum(r)
    
    # Define constraint to ensure no overlap
    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636
    
    # Set bounds for optimization
    bounds = [(0, 1)] * n
    
    # Run optimization with SLSQP
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': constraint}], method='SLSQP', bounds=bounds)
    
    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0
    
    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
