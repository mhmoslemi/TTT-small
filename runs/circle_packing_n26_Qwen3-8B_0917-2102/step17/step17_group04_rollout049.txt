import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Define positions with corner clusters and staggered rows
    centers = np.zeros((n, 2))
    for i in range(n):
        if i < 8:
            row = 0
            col = i
        elif i < 16:
            row = 1
            col = i - 8
        else:
            row = 2
            col = i - 16
        x = col / 4
        y = row / 2
        centers[i] = [x, y]
    
    def objective(r):
        return -np.sum(r)

    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r) - 2.636

    # Initial guess based on minimum inter-circle distance
    dists = squareform(pdist(centers))
    initial_r = np.min(dists) * np.ones(n)
    result = minimize(objective, initial_r, constraints=[{'type': 'ineq', 'fun': lambda r: -np.sum(np.abs(pdist(centers)) - np.sum(np.minimum(r, np.abs(pdist(centers)))))}], method='SLSQP', bounds=[(0, 1)] * n)

    # Final adjustment for boundary and overlap
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            result['x'][i] = 0

    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
