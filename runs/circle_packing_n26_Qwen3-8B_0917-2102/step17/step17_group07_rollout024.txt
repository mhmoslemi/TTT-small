import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Create a hexagonal grid with adjustable spacing
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        offset = 0.5 * (1 - row % 2)
        x = col + offset
        y = row + 0.5
        centers[i] = [x, y]
    
    # Add boundary-aware circles
    for i in range(n):
        if i % 5 == 0 and i != 0:  # Add boundary circles
            x = 0.1 if i % 10 == 0 else 0.9
            y = 0.1 if i % 10 == 1 else 0.9
            centers = np.vstack((centers, [x, y]))
    n = len(centers)
    
    # Compute initial inter-circle distances
    dists = squareform(pdist(centers))
    min_dist = np.min(dists)
    
    # Initial guess based on minimum inter-circle distance
    initial_r = np.min(dists) * np.ones(n)
    
    # Objective function to maximize sum of radii
    def objective(r):
        return -np.sum(r)
    
    # Constraint function to ensure non-overlap
    def constraint(r):
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return -np.inf
        return np.sum(r)
    
    # Bounds for radii
    bounds = [(0, 1)] * n
    
    # Optimization using SLSQP
    result = minimize(objective, initial_r, method='SLSQP', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}])
    
    # Final adjustment for boundary constraints
    for i in range(n):
        x, y = centers[i]
        r = result['x'][i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            result['x'][i] = 0
    
    sum_radii = np.sum(result['x'])
    return centers, result['x'], sum_radii
