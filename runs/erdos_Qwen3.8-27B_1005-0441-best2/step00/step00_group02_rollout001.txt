import numpy as np

def project_sum(v, target):
    """Project v onto {h: sum(h)=target, 0<=h<=1} via bisection on shift."""
    n = len(v)
    lo, hi = v.min() - target, v.max()
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2, 0, 1)

import numpy as np

def project_sum(v, target):
    """Project v onto {h: sum(h)=target, 0<=h<=1} via bisection."""
    lo = v.min() - target
    hi = v.max()
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def compute_grad_softmax(h, lam):
    """Gradient of soft-max smoothing of max correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = float(np.max(corr))
    shifted = corr - c_max
    w = np.exp(lam * shifted)
    w = w / np.sum(w)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s > 0:
            grad[:n - s] += wk * dx * (1.0 - h[s:])
            grad[s:] += wk * dx * (-h[:n - s])
        elif s < 0:
            sp = -s
            grad[sp:] += wk * dx * (1.0 - h[:n - sp])
            grad[:n - sp] += wk * dx * (-h[sp:])
        else:
            grad += wk * dx * (1.0 - 2.0 * h)
    return grad, c_max

def optimize_h(h0, target_sum, time_limit):

    t0 = time.time()
    h = project_sum(np.clip(h0, 0.0, 1.0), target_sum)
    best_h = h.copy()
    best_c = eval_c5(h)
    lam = 20.0
    lr = 0.03
    stuck = 0
    for it in range(5000):
        if time.time() - t0 > time_limit:
            break
        grad, c_max = compute_grad_softmax(h, lam)
        h_new = project_sum(h - lr * grad, target_sum)
        new_c = eval_c5(h_new)
        if new_c < best_c - 1e-10:
            best_c = new_c
            best_h = h_new.copy()
            h = h_new
            stuck = 0
            lr = min(lr * 1.03, 0.4)
        else:
            h = h_new
            lr *= 0.85
            stuck += 1
            if stuck > 50:
                lr = max(lr, 0.01)
                lam = min(lam * 1.4, 120)
                stuck = 0
        if it % 30 == 0:
            lam = min(lam * 1.03, 120)
    # Fine-tune phase
    lam = min(lam, 80.0)
    lr = 0.008
    for it in range(1000):
        if time.time() - t0 > time_limit:

        grad, c_max = compute_grad_softmax(h, lam)
        h_new = project_sum(h - lr * grad, target_sum)
        new_c = eval_c5(h_new)

    return best_h, best_c
