def compute_c5(h, dx):

    return float(np.max(corr) * dx)

def project_feasible(y, n):
    """Project y onto {0<=h<=1, sum(h)=n/2}"""
    # Find alpha such that sum(clip(y+alpha, 0, 1)) = n/2
    lo, hi = -max(np.abs(y)) - 1, max(np.abs(y)) + 1
    
    def f(alpha):
        return np.sum(np.clip(y + alpha, 0, 1)) - n / 2
    
    for _ in range(60):
        mid = (lo + hi) / 2
        if f(mid) > 0:
            hi = mid
        else:
            lo = mid
    alpha = (lo + hi) / 2
    h = np.clip(y + alpha, 0, 1)
    # Fix any remaining discrepancy
    h = h + (n / 2 - np.sum(h)) / n
    h = np.clip(h, 0, 1)
    return h

    m = np.max(scaled)
