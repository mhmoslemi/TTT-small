import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def fix_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not free.any():
            if diff > 0:
                free = h < 1.0 - 1e-12
            else:
                free = h > 1e-12
        if not free.any():
            break
        h[free] += diff / float(free.sum())
        h = np.clip(h, 0.0, 1.0)
    return h

def perturb_h(h, rng, n_pert, scale):
    n = len(h)
    target_sum = n / 2.0
    h_new = h.copy()
    idx = rng.choice(n, size=n_pert, replace=False)
    h_new[idx] += rng.normal(0, scale, size=n_pert)
    return fix_sum(h_new, target_sum)

def crossover_h(h1, h2, rng):
    n = len(h1)
    target_sum = n / 2.0
    p = rng.randint(1, n - 1)
    h_new = np.empty(n)
    h_new[:p] = h1[:p]
    h_new[p:] = h2[p:]
    return fix_sum(h_new, target_sum)

def move_mass_search(h, c5, rng, max_time):
    t_start = time.time()
    n = len(h)
    target_sum = n / 2.0
    d_vals = np.array([0.02, 0.05, 0.08, 0.12, 0.18, 0.25, 0.4])
    stall = 0
    while time.time() - t_start < max_time:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() - t_start > max_time:
                break
            js = rng.choice([j for j in range(n) if j != i], size=min(8, n - 1), replace=False)
            for j in js:
                for d in d_vals:
                    ok1 = h[i] - d >= 0 and h[j] + d <= 1
                    ok2 = h[i] + d <= 1 and h[j] - d >= 0
                    for delta, cond in [(d, ok1), (-d, ok2)]:
                        if not cond:
                            continue
                        h_try = h.copy()
                        h_try[i] += delta
                        h_try[j] -= delta
                        c5_try = compute_c5(h_try)
                        if c5_try < c5 - 1e-10:
                            h = h_try
                            c5 = c5_try
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        if not improved:
            stall += 1
            d_vals = d_vals * 0.7
            if d_vals[0] < 0.003 or stall > 8:
                break
        else:
            stall = 0
    return h, c5

def random_walk_search(h, c5, rng, max_time):
    t_start = time.time()
    n = len(h)
    scale = 0.12
    while time.time() - t_start < max_time:
        h_try = perturb_h(h, rng, rng.randint(1, 5), scale)
        c5_try = compute_c5(h_try)
        if c5_try < c5 - 1e-10:
            h = h_try
            c5 = c5_try
        elif (time.time() - t_start) % 5.0 > 4.5:
            scale *= 0.85
            if scale < 0.002:
                break
    return h, c5

def genetic_algorithm(h_init, c5_init, rng, max_time):
    t_start = time.time()
    n = len(h_init)
    pop_size = 35
    pop = [h_init.copy()]
    for _ in range(pop_size - 1):
        pop.append(perturb_h(h_init, rng, rng.randint(2, 8), 0.15))
    fits = [c5_init] + [compute_c5(p) for p in pop[1:]]
    best_ga_h = h_init.copy()
    best_ga_c5 = c5_init
    n_elite = 8
    while time.time() - t_start < max_time:
        order = np.argsort(fits)
        pop = [pop[i] for i in order]
        fits = [fits[i] for i in order]
        if fits[0] < best_ga_c5 - 1e-10:
            best_ga_c5 = fits[0]
            best_ga_h = pop[0].copy()
        children = [p.copy() for p in pop[:n_elite]]
        while len(children) < pop_size:
            p1 = pop[rng.randint(0, max(1, n_elite))]
            p2 = pop[rng.randint(0, max(1, min(n_elite, pop_size // 2)))]
            if rng.random() < 0.6:
                child = crossover_h(p1, p2, rng)
            else:
                child = p1.copy()
            if rng.random() < 0.85:
                child = perturb_h(child, rng, rng.randint(1, 5), 0.08)
            children.append(child)
        pop = children
        fits = [compute_c5(p) for p in pop]
    return best_ga_h, best_ga_c5

def upscale_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return fix_sum(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = np.array(initial_h_values, dtype=float)
    best_c5 = compute_c5(best_h)
    n = len(best_h)

    def time_left():
        return max(0, budget_s - (time.time() - t0))

    # Phase 1: Move-mass local search
    if time_left() > 50:
        best_h, best_c5 = move_mass_search(best_h, best_c5, rng, min(budget_s * 0.18, time_left() - 5))

    # Phase 2: Random walk polish
    if time_left() > 40:
        best_h, best_c5 = random_walk_search(best_h, best_c5, rng, min(budget_s * 0.10, time_left() - 5))

    # Phase 3: Coordinate/move-mass again on improved
    if time_left() > 50:
        best_h, best_c5 = move_mass_search(best_h, best_c5, rng, min(budget_s * 0.15, time_left() - 5))

    # Phase 4: Genetic algorithm
    if time_left() > 80:
        ga_h, ga_c5 = genetic_algorithm(best_h, best_c5, rng, min(budget_s * 0.30, time_left() - 10))
        if ga_c5 < best_c5:
            best_h, best_c5 = ga_h, ga_c5

    # Phase 5: Local search after GA
    if time_left() > 40:
        best_h, best_c5 = move_mass_search(best_h, best_c5, rng, min(budget_s * 0.12, time_left() - 5))

    # Phase 6: Try finer discretization
    if time_left() > 80:
        n_try = n * 2
        h_up = upscale_h(best_h, n_try)
        c5_up = compute_c5(h_up)
        if c5_up <= best_c5 + 0.01:
            h_up, c5_up = move_mass_search(h_up, c5_up, rng, min(120, time_left() - 30))
            h_up, c5_up = random_walk_search(h_up, c5_up, rng, min(40, time_left() - 20))
            if c5_up < best_c5:
                best_h, best_c5, n = h_up, c5_up, n_try

    # Phase 7: Final polish
    if time_left() > 20:
        best_h, best_c5 = move_mass_search(best_h, best_c5, rng, min(60, time_left() - 5))
    if time_left() > 10:
        best_h, best_c5 = random_walk_search(best_h, best_c5, rng, min(15, time_left() - 3))

    # Ensure validity
    best_h = fix_sum(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
