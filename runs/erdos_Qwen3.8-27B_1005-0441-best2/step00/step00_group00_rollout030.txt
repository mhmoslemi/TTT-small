Let me also verify the gradient computation one more time.

Given c = np.correlate(h, 1-h, mode='full'), element p corresponds to:
c[p] = Σ_i h[i + k] · (1 - h[i])  where k = p - (n-1)

The sum is over i such that 0 ≤ i < n and 0 ≤ i+k < n.

∂c[p]/∂h[j]:
- h[j] appears as h[i+k] when i+k=j, i.e., i=j-k. Valid if 0≤j-k<n. Contribution: (1-h[j-k])
- h[j] appears as h[i] in (1-h[i]) when i=j. Valid if 0≤j<n (always) AND 0≤j+k<n. Contribution: -h[j+k]

So ∂c/∂h[j] = (1-h[j-k])·[0≤j-k<n] - h[j+k]·[0≤j+k<n]

import numpy as np
import time

def project_feasible(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    return np.clip(h - t, 0.0, 1.0)

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def grad_at_lag(h, k):
    n = len(h)
    js = np.arange(n)
    g = np.zeros(n)
    v1 = (js - k >= 0) & (js - k < n)
    g[v1] += 1.0 - h[js[v1] - k]
    v2 = (js + k >= 0) & (js + k < n)
    g[v2] -= h[js[v2] + k]
    return g

def subgrad_optimize(h, target, lr_init, max_iters, deadline, top_m=3):
    n = len(h)
    h_cur = h.copy()
    c5_cur = c5_value(h_cur)
    best = h_cur.copy()
    best_c5 = c5_cur
    lr = lr_init
    stall = 0
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        T = min(3.0 + stall * 0.03, 25.0)
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        idx = np.argsort(corr)[::-1][:top_m]
        vals = corr[idx]
        w = np.exp(T * (vals - vals[0]))
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(top_m):
            k = idx[ii] - (n - 1)
            g += w[ii] * grad_at_lag(h_cur, k)
        h_new = project_feasible(h_cur - lr * g, target)
        c5_new = c5_value(h_new)
        if c5_new < c5_cur - 1e-14:
            h_cur = h_new
            c5_cur = c5_new
            if c5_new < best_c5:
                best = h_cur.copy()
                best_c5 = c5_new
            stall = 0
            lr = min(lr * 1.005, 0.12)
        else:
            stall += 1
            if stall % 20 == 0:
                lr *= 0.88
        if lr < 0.0002 and stall > 200:
            break
    return best, best_c5
