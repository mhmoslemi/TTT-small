```python
def project_onto_simplex_box(h, target_sum):
    """Project h onto {0 ≤ x ≤ 1, sum(x) = target_sum}"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-12:
        return h
    if current_sum > target_sum:
        # Need to reduce: find λ > 0 such that sum(min(h, h-λ>0... )) 
        # Actually: h_new[i] = max(0, h[i] - λ) summed = target_sum
        # h_new[i] = clip(h[i] - λ, 0, 1)
        # Since h[i] ≤ 1, clip(h[i]-λ, 0, 1) = max(0, h[i]-λ) for λ ≥ 0
        lo, hi = 0.0, np.max(h)
        for _ in range(50):
            mid = (lo + hi) / 2
            s = np.sum(np.maximum(0, h - mid))
            if s > target_sum:
                lo = mid
            else:
                hi = mid
        lam = (lo + hi) / 2
        return np.maximum(0, h - lam)
    else:
        # Need to increase: h_new[i] = min(1, h[i] + λ)
        lo, hi = 0.0, 1.0 - np.min(h)
        for _ in range(50):
            mid = (lo + hi) / 2
            s = np.sum(np.minimum(1, h + mid))
            if s < target_sum:
                lo = mid
            else:
                hi = mid
        lam = (lo + hi) / 2
        return np.minimum(1, h + lam)
```

```python
def project_feasible(h, target_sum):
    """Project h onto {0 ≤ x ≤ 1, sum(x) = target_sum}"""

    s = np.sum(h)
    if abs(s - target_sum) < 1e-10:
        return h
    if s > target_sum:
        lo, hi = 0.0, np.max(h) + 1e-8
        for _ in range(60):
            mid = (lo + hi) / 2
            if np.sum(np.maximum(0.0, h - mid)) > target_sum:
                lo = mid
            else:
                hi = mid
        return np.maximum(0.0, h - (lo + hi) / 2)

    return np.max(corr) * dx

def smooth_min_grad(h, epsilon):
    """Compute smooth approximation of min_k R(k) and its gradient.
    R(k) = autocorrelation of h at lag k-(n-1)."""
    n = len(h)
    R = np.correlate(h, h, mode='full')
    R_min = np.min(R)
    # For numerical stability
    exp_terms = np.exp(-(R - R_min) / epsilon)
    sum_exp = np.sum(exp_terms)
    value = R_min - epsilon * np.log(sum_exp)
    weights = exp_terms / sum_exp
    
    # Gradient: weighted sum of ∇R(k) for each k
    n_full = len(R)

    for k in range(n_full):
        m = k - (n - 1)
        p = abs(m)
        if p == 0:
            g = 2.0 * h
        else:
            g = np.zeros(n)
            g[:n-p] += h[p:]
            g[p:] += h[:n-p]
```
