```python
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)

def compute_c5_gradient(h, beta=100):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    exps = np.exp(beta * (corr - max_c))
    weights = exps / np.sum(exps)
    
    # Gradient of sum_k w_k * corr[k] * dx
    # corr has 2n-1 elements corresponding to lags d = -(n-1), ..., -(1), 0, 1, ..., n-1
    # In numpy correlate(h, 1-h, 'full'), index k corresponds to lag = k - (n-1)
    # i.e., for k=0: lag = -(n-1), k=n-1: lag=0, k=2n-2: lag=n-1
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        d = k - (n - 1)  # lag from -(n-1) to (n-1)
        # d/dh[j] of corr[k] = (1-h[j+d]) * I(0<=j+d<n) - h[j-d] * I(0<=j-d<n)
        # For j in [0, n):
        #   j+d valid: 0 <= j+d < n => -d <= j < n-d (if d>0) or 0 <= j < n-d (if d<0)
        #   j-d valid: 0 <= j-d < n => d <= j < n+d (if d>0) or d <= j < n (if d<0)
        
        # Term 1: (1-h[j+d]) for j where 0<=j<n and 0<=j+d<n
        if d >= 0:
            j_min = 0
            j_max = n - d  # j+d < n => j < n-d
        else:
            j_min = -d  # j+d >= 0 => j >= -d
            j_max = n
        
        if j_min < j_max:
            grad[j_min:j_max] += w * (1 - h[j_min + d : j_max + d])
        
        # Term 2: -h[j-d] for j where 0<=j<n and 0<=j-d<n
        if d >= 0:

            j_max = n + d  # j-d < n => j < n+d (but d<0 so n+d < n)
```

```python
def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        d = k - (n - 1)
        if d >= 0:
            a, b = 0, n - d
        else:
            a, b = -d, n
        if a < b:
            grad[a:b] += w[k] * (1.0 - h[a+d:b+d])
        if d >= 0:
            c, e = d, n
        else:
            c, e = 0, n + d
        if c < e:
            grad[c:e] -= w[k] * h[c-d:e-d]
    return grad * dx

    s = h.sum()
    if s != t:

        diff = (t - s) / n
        h += diff
        h = np.clip(h, 0, 1)
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _smooth_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    return float((mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta) * dx)

def _smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-14:
            continue
        d = (n - 1) - i
        lo1 = max(0, -d)
        hi1 = min(n, n - d)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + d:hi1 + d])
        lo2 = max(0, d)
        hi2 = min(n, d + n)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2 - d:hi2 - d]
    return grad * dx

def _project(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        h = np.clip(h + (target - s) / n, 0.0, 1.0)
    return h

def _sqsqp_phase(h0, beta, maxiter=300):
    n = len(h0)
    target = n / 2.0
    def obj(x):
        return _smooth_val(x, beta)
    def obj_jac(x):
        return _smooth_grad(x, beta)
    cons = [
        {'type': 'eq', 'fun': lambda x: x.sum() - target,
         'jac': lambda x: np.ones(n)}
    ]
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=obj_jac, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return _project(res.x)

def _optimize(h0, t_budget=60, n_betas=5):
    h = h0.copy().astype(float)
    h = _project(h)
    best = compute_c5(h)
    hb = h.copy()
    t0 = time.time()
    for i in range(n_betas):
        if time.time() - t0 > t_budget:
            break
        beta = 10.0 * (i + 1) ** 2
        h = _sqsqp_phase(h, beta, maxiter=200)
        c = compute_c5(h)
        if c < best:
            best = c
            hb = h.copy()
    # Nelder-Mead polish
    if time.time() - t0 < t_budget - 5:
        h_nm = hb.copy()
        np.random.seed(0)
        nm_res = minimize(lambda x: compute_c5(_project(x)), h_nm,
                         method='Nelder-Mead',
                         options={'maxiter': 300, 'xatol': 1e-8, 'fatol': 1e-10})
        h_nm = _project(nm_res.x)
        c_nm = compute_c5(h_nm)
        if c_nm < best:
            best = c_nm
            hb = h_nm.copy()
    return hb, best

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None
    
    if h_init is None:
        n = 70
        h_init = np.zeros(n)
        h_init[:n // 2] = 1.0
    
    n = len(h_init)
    best_h = _project(h_init.copy())
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Main optimization from initial
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 60:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _smooth_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    return float((mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta) * dx)

def _smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-14:
            continue
        d = (n - 1) - i
        lo1 = max(0, -d)
        hi1 = min(n, n - d)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + d:hi1 + d])
        lo2 = max(0, d)
        hi2 = min(n, d + n)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2 - d:hi2 - d]
    return grad * dx

def _project(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        h = np.clip(h + (target - s) / n, 0.0, 1.0)
    return h

def _slsqp_phase(h0, beta, maxiter=300):
    n = len(h0)
    target = n / 2.0
    def obj(x):
        return _smooth_val(x, beta)
    def obj_jac(x):
        return _smooth_grad(x, beta)
    def con_fun(x):
        return x.sum() - target
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return _project(res.x)
    except Exception:
        return _project(h0.copy())

def _optimize(h0, t_budget=60, n_betas=5, nm_iter=400):
    h = _project(h0.copy().astype(float))
    best = compute_c5(h)
    hb = h.copy()
    t0 = time.time()
    for i in range(n_betas):
        if time.time() - t0 > t_budget - 10:
            break
        beta = 15.0 * (i + 1) ** 1.5
        h = _slsqp_phase(h, beta, maxiter=250)
        c = compute_c5(h)
        if c < best:
            best = c
            hb = h.copy()
    if time.time() - t0 < t_budget - 5:
        def nm_obj(x):
            return compute_c5(_project(x))
        try:
            nm_res = minimize(nm_obj, hb.copy(), method='Nelder-Mead',
                             options={'maxiter': min(nm_iter, int(2*(t_budget - (time.time()-t0))))
                                      if t_budget > (time.time()-t0)+5 else 50,
                                      'xatol': 1e-9, 'fatol': 1e-11})
            h_nm = _project(nm_res.x)
            c_nm = compute_c5(h_nm)
            if c_nm < best:
                best = c_nm
                hb = h_nm.copy()
        except Exception:
            pass
    return hb, best
```
