def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

    p = r.permutation(n)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def rand_bin(n, s):
    r = np.random.RandomState(s)
    p = r.permutation(n)
    h = np.zeros(n)
    h[p[:n // 2]] = 1.0
    return h

def thue_morse_bin(n, s):
    r = np.random.RandomState(s)
    t = np.zeros(n)
    for i in range(n):
        t[i] = bin(i).count('1') % 2
    n1 = int(t.sum())
    target = n // 2
    if n1 > target:
        ones = np.where(t == 1)[0]
        for k in ones[r.permutation(n1)][:n1 - target]:
            t[k] = 0
    elif n1 < target:
        z = np.where(t == 0)[0]
        for k in z[r.permutation(len(z))][:target - n1]:
            t[k] = 1
    return t

def fix_bin(h, n, s):
    r = np.random.RandomState(s)
    b = (h >= 0.5).astype(float)
    n1 = int(b.sum())
    target = n // 2
    if n1 > target:
        ones = np.where(b == 1)[0]
        for k in ones[r.permutation(n1)][:n1 - target]:
            b[k] = 0
    elif n1 < target:
        z = np.where(b == 0)[0]
        for k in z[r.permutation(len(z))][:target - n1]:
            b[k] = 1
    return b

def local_search(h0, time_limit, seed):
    n = len(h0)
    r = np.random.RandomState(seed)
    h = h0.copy().astype(float)
    bh = h.copy()
    bc = compute_c5(h)
    cc = bc
    T = 0.08
    st = 0
    t0 = time.time()
    while time.time() - t0 < time_limit:
        oi = np.where(h == 1.0)[0]
        zi = np.where(h == 0.0)[0]
        if len(oi) < 1 or len(zi) < 1:
            break
        i = oi[r.randint(len(oi))]
        j = zi[r.randint(len(zi))]
        h[i] = 0.0
        h[j] = 1.0
        nc = compute_c5(h)
        if nc - cc < 0 or r.random() < np.exp(-(nc - cc) / max(T, 1e-10)):
            cc = nc
            if nc < bc:
                bc = nc
                bh = h.copy()
                st = 0
            else:
                st += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            st += 1
        T *= 0.9997
        if T < 1e-5 or st > 20000:
            T = 0.06
            h = bh.copy()
            cc = bc
            st = 0
    return bh, bc

def greedy_sweep(h, time_limit):
    n = len(h)
    bh = h.copy()
    bc = compute_c5(bh)
    t0 = time.time()
    while time.time() - t0 < time_limit:
        ones = list(np.where(bh == 1)[0])
        zeros = list(np.where(bh == 0)[0])
        bp = None
        best_c = bc
        for i in ones:
            bh[i] = 0
            for j in zeros:
                bh[j] = 1
                c = compute_c5(bh)
                if c < best_c:
                    best_c = c
                    bp = (i, j)
                bh[j] = 0
            bh[i] = 1
        if bp is None:
            break
        bh[bp[0]] = 0
        bh[bp[1]] = 1
        bc = best_c
    return bh, bc

    bc = compute_c5(h)
    cc = bc
    T = 0.08
    st = 0
    t0 = time.time()
    while time.time() - t0 < time_limit:
        oi = np.where(h == 1.0)[0]
        zi = np.where(h == 0.0)[0]
        if len(oi) < 1 or len(zi) < 1:
            break
        i = oi[r.randint(len(oi))]
        j = zi[r.randint(len(zi))]
        h[i] = 0.0
        h[j] = 1.0
        nc = compute_c5(h)
        if nc - cc < 0 or r.random() < np.exp(-(nc - cc) / max(T, 1e-10)):
            cc = nc
            if nc < bc:
                bc = nc
                bh = h.copy()
                st = 0
            else:
                st += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            st += 1
        T *= 0.9997
        if T < 1e-5 or st > 20000:
            T = 0.06
            h = bh.copy()
            cc = bc
            st = 0
    return bh, bc

def greedy_sweep(h, time_limit):
    n = len(h)
    bh = h.copy()
    bc = compute_c5(bh)
    t0 = time.time()
    while time.time() - t0 < time_limit:
        ones = list(np.where(bh == 1)[0])
        zeros = list(np.where(bh == 0)[0])
        bp = None
        best_c = bc
        for i in ones:
            bh[i] = 0
            for j in zeros:
                bh[j] = 1
                c = compute_c5(bh)
                if c < best_c:
                    best_c = c
                    bp = (i, j)
                bh[j] = 0
            bh[i] = 1
        if bp is None:
            break
        bh[bp[0]] = 0
        bh[bp[1]] = 1
        bc = best_c
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t_end = time.time() + budget_s
    best_h = None
    best_c5 = 1e9
    cands = []
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float)
        if len(h0) % 2 == 0:
            cands.append(h0.copy())
    for nm in [64, 70, 80, 96, 100, 120, 128, 160]:
        cands.append(thue_morse_bin(nm, seed))
    for nm in [70, 96, 120, 160]:
        cands.append(rand_bin(nm, seed))
        cands.append(rand_bin(nm, seed + 13))

    results = []
    for ci in range(len(cands)):
        if time.time() > t_end - 200:
            break
        hc = cands[ci]
        nn = len(hc)
        hb = fix_bin(hc, nn, seed + ci)
        bh, bc = local_search(hb, 6.0, seed + ci)
        results.append([bc, bh, nn])

    m = len(results)
    for i in range(m):
        for j in range(i + 1, m):
            if results[j][0] < results[i][0]:
                results[i], results[j] = results[j], results[i]

    if m > 0:
        best_c5 = results[0][0]
        best_h = results[0][1].copy()

    for ri in range(min(4, m)):
        if time.time() > t_end - 80:
            break
        t_left = t_end - time.time() - 80
        if t_left < 15:
            break
        bh, bc = local_search(results[ri][1], min(180.0, t_left * 0.65), seed + 500 + ri)
        t2 = t_end - time.time() - 40
        if t2 > 3:
            bh2, bc2 = greedy_sweep(bh, min(15.0, t2))
            if bc2 < bc:
                bh, bc = bh2, bc2
        if bc < best_c5:
            best_c5 = bc
            best_h = bh.copy()

    if best_h is not None:
        for rd in range(4):
            if time.time() > t_end - 10:
                break
            t_left = min(40.0, t_end - time.time() - 5)
            if t_left < 3:
                break
            bh, bc = local_search(best_h, t_left * 0.75, seed + 900 + rd)
            t2 = t_end - time.time() - 5
            if t2 > 2:
                bh2, bc2 = greedy_sweep(bh, min(8.0, t2))
                if bc2 < bc:
                    bh, bc = bh2, bc2
            if bc < best_c5:
                best_c5 = bc
                best_h = bh.copy()

    if best_h is None:
        best_h = rand_bin(70, 0)
        best_c5 = compute_c5(best_h)

    nf = len(best_h)
    ss = int(np.round(best_h.sum()))
    tt = nf // 2
    if ss > tt:
        for _ in range(ss - tt):
            idx = np.where(best_h == 1)[0][0]
            best_h[idx] = 0
    elif ss < tt:
        for _ in range(tt - ss):
            idx = np.where(best_h == 0)[0][0]
            best_h[idx] = 1

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
