import time


def _compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def _fix_sum(h, target_sum):
    h = h.astype(float).copy()
    s = np.sum(h)
    if s < 1e-12:
        h[:] = target_sum / len(h)
        return np.clip(h, 0, 1)
    h *= target_sum / s
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        nm = np.sum(mask)
        if nm > 0:
            h[mask] += diff / nm
    return np.clip(h, 0, 1)


def _solve_lp(h, N, target_sum):
    h_var = cp.Variable(N)
    t = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
    for s in range(-(N - 1), N):
        if s >= 0:
            idx = np.arange(N - s)
            shifted = idx + s
        else:
            idx = np.arange(-s, N)
            shifted = idx + s
        coeffs = 1.0 - h[shifted]
        constraints.append(cp.sum(h_var[idx] * coeffs) <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    for solver in ['CBC', 'GLPK', 'ECOS']:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                val = h_var.value
                if val is not None:

            continue
    return None
