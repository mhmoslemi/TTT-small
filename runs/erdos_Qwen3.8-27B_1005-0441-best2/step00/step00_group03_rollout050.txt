import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def project_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def pair_search(h, target_sum, rng, eps, n_iter, t0, budget):
    n = len(h)
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    for _ in range(n_iter):
        if time.time() - t0 > budget - 5:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(0.003, eps)
        # Try transfer i -> j
        if h[i] >= d and h[j] <= 1.0 - d:
            h[i] -= d
            h[j] += d
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            else:
                h[i] += d
                h[j] -= d
        # Try transfer j -> i
        if h[j] >= d and h[i] <= 1.0 - d:
            h[j] -= d
            h[i] += d
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            else:
                h[j] += d
                h[i] -= d
    return best_h, best_c5
