def evaluate_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    k_max = np.argmax(corr)
    # The correlation at shift k_max (in full mode, index 0 corresponds to shift -(n-1))

    # d/dh[i] of this = (1-h[i + k_max - (n-1)]) - h[i - k_max + (n-1)]
    # Let shift = k_max - (n-1)
    shift = k_max - (n - 1)
    # d_corr/dh[i] = (1-h)[i+shift] - h[i-shift] (with 0 outside bounds)

        im = i - shift
        if 0 <= ip < n:
            grad[i] += (1.0 - h[ip])
        if 0 <= im < n:
            grad[i] -= h[im]
