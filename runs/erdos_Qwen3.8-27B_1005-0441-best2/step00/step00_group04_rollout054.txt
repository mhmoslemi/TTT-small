    """Project h onto {0≤h≤1, sum(h)=n/2}"""

    corr = compute_corr(h)
    return float(np.max(corr) * dx)

        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])[:excess]
        h_bin[one_idx[order]] = 0
    elif s < target:
        missing = int(target - s)
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])[:missing]
        h_bin[zero_idx[order]] = 1
    h_bin = project_feasible(h_bin, n)
    
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_h = h_bin
        best_c5 = c5_bin
    
    return best_h, best_c5
