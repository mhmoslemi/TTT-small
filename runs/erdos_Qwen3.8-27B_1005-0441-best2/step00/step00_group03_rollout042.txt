import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=np.float64)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(proj)
    if abs(diff) > 1e-12:
        mask = (proj > 1e-9) & (proj < 1.0 - 1e-9)
        cnt = float(np.sum(mask))
        if cnt > 0:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def eval_coord_val(h, i, n, s_other, t):
    h[i] = t
    h[n - 1] = n / 2.0 - s_other - t
    return compute_c5(h)


def golden_coord(h, i, n, iters=35):
    s_other = np.sum(h[:n - 1]) - h[i]
    lo = max(0.0, n / 2.0 - s_other - 1.0)
    hi = min(1.0, n / 2.0 - s_other)
    if hi - lo < 1e-14:
        return
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = eval_coord_val(h, i, n, s_other, c)
    fd = eval_coord_val(h, i, n, s_other, d)
    for _ in range(iters):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = eval_coord_val(h, i, n, s_other, c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = eval_coord_val(h, i, n, s_other, d)
    h[i] = (a + b) / 2.0
    h[n - 1] = n / 2.0 - s_other - h[i]


def coord_descent(h, n_sweeps=50, iters=35):
    h = h.astype(np.float64).copy()
    n = len(h)
    prev_c = compute_c5(h)
    for sweep in range(n_sweeps):
        old = h.copy()
        for i in range(n - 1):
            golden_coord(h, i, n, iters)
        cur_c = compute_c5(h)
        if cur_c < prev_c - 1e-13:
            prev_c = cur_c
        else:
            if np.max(np.abs(h - old)) < 1e-12:
                break
    return h


def nm_objective(x):
    return compute_c5(x)


def nm_opt(h0, max_iter=2000):
    h0 = h0.astype(np.float64).copy()
    n = len(h0)
    res = minimize(nm_objective, h0, method='Nelder-Mead',
                   options={'maxiter': max_iter, 'xatol': 1e-10,
                            'fatol': 1e-12, 'adaptive': True})
    return project_feasible(res.x, n)


def gen_binary(n, rng):
    idx = rng.permutation(n)
    h = np.zeros(n, dtype=np.float64)
    h[idx[:n // 2]] = 1.0
    h += rng.random(n) * 0.04
    return project_feasible(h, n)


def gen_smooth(n, rng):
    h = rng.random(n).astype(np.float64)
    return project_feasible(h, n)


def gen_block_perturb(n, rng):
    h = np.zeros(n, dtype=np.float64)
    h[:n // 2] = 1.0
    rng.shuffle(h)
    h += rng.random(n) * 0.06
    return project_feasible(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s

    if 'initial_h_values' in dir() and initial_h_values is not None:

    if time.time() < deadline - 5:
        h = coord_descent(best_h, 50, 40)
        c = compute_c5(h)
