from scipy.optimize import minimize, brentq
import time

def project_onto_simplex(h, target_sum):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=target_sum}"""
    # Use bisection on lambda
    def f(lam):
        return np.sum(np.clip(h - lam, 0, 1)) - target_sum
    # f is monotonically decreasing in lam
    # Find lam such that f(lam) = 0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    lam = brentq(f, lo, hi, xtol=1e-12)
    return np.clip(h - lam, 0, 1)

def smooth_max(corr, beta):
    """Compute smooth max and its gradient weights."""
    shifted = beta * corr - np.max(beta * corr)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    val = (np.log(np.sum(np.exp(shifted))) / beta) + np.max(corr)
    return val, weights

import numpy as np
from scipy.optimize import minimize, brentq
import time


def project_feasible(h, target_sum):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=target_sum}."""
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-14:
        return h
    def f(lam):
        return np.sum(np.clip(h - lam, 0.0, 1.0)) - target_sum
    lo, hi = -3.0, 3.0
    lam = brentq(f, lo, hi, xtol=1e-15)
    return np.clip(h - lam, 0.0, 1.0)


def compute_c5(h, dx):
    """Compute C5 = max_k correlate(h, 1-h)[k] * dx."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def c5_smooth(h, dx, beta):
    """Compute smooth C5 objective value and gradient."""
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full") * dx
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    exps = np.exp(shifted)
    s = np.sum(exps)
    val = max_c + np.log(s) / beta
    weights = exps / s

    grad = np.zeros(n)
    half_n = n - 1
    for idx in range(len(corr)):
        w = weights[idx]
        if w < 1e-12:
            continue
        k = idx - half_n
        if k >= 0:
            g = np.zeros(n)
            g[:n - k] += omh[k:]
            g[k:] -= h[:n - k]
        else:
            ks = -k
            g = np.zeros(n)
            g[:n - ks] += omh[ks:]
            g[ks:] -= h[:n - ks]
        grad += w * g
    return val, grad


def _obj_func(h, dx, beta):
    return c5_smooth(h, dx, beta)[0]


def _grad_func(h, dx, beta):
    return c5_smooth(h, dx, beta)[1]


def _cons_fun(h):
    return np.sum(h) - len(h) / 2.0


def _cons_jac(h):
    return np.ones(len(h))
