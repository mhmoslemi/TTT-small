    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            m = h < 1.0
            cnt = int(m.sum())
            if cnt == 0:
                break
            h[m] += diff / cnt
        else:
            m = h > 0.0
            cnt = int(m.sum())
            if cnt == 0:
                break
            h[m] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_at_lag(h, k, n):
    n1 = n - 1
    j = np.arange(n)
    i1 = j - k + n1
    i2 = j + k - n1
    g = np.zeros(n)
    v1 = (i1 >= 0) & (i1 < n)
    v2 = (i2 >= 0) & (i2 < n)
    if np.any(v1):
        g[v1] += h[i1[v1]]
    if np.any(v2):
        g[v2] += h[i2[v2]]
    return g

def subgradient_opt(h0, max_time, lr_init=0.4, n_lags=7):
    n = len(h0)
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    lr = lr_init
    stall = 0
    while time.time() - t0 < max_time:
        ac = compute_ac(h)
        ac_min = ac.min()
        thresh = ac_min + max(3.0, 0.05 * ac.mean())
        active = np.where(ac <= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmin(ac)])
        active = active[np.argsort(ac[active])][:n_lags]
        g = np.zeros(n)
        for k in active:
            g += grad_at_lag(h, k, n)
        g /= len(active)
        h = h + lr * g / n
        h = project(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
            lr = min(lr * 1.02, 1.5)
        else:
            stall += 1
            if stall > 80:
                lr *= 0.7
                stall = 0
                if lr < 0.03:
                    h = best_h + np.random.randn(n) * 0.15
                    h = project(h, n)
                    lr = 0.3
        if lr < 1e-10:
            break
    return best_h, best_c5

def binary_local_search(h0, max_time, n_trials=200):
    n = len(h0)
    h = np.where(h0 >= 0.5, 1.0, 0.0)
    # Ensure sum = n/2
    s = int(h.sum())
    target = n // 2
    if s > target:
        ones = np.where(h == 1)[0]
        to_flip = np.random.choice(ones, s - target, replace=False)
        h[to_flip] = 0.0
    elif s < target:
        zeros = np.where(h == 0)[0]
        to_flip = np.random.choice(zeros, target - s, replace=False)
        h[to_flip] = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    while time.time() - t0 < max_time:
        improved = False
        zero_idx = np.where(h == 0)[0]
        one_idx = np.where(h == 1)[0]
        if len(zero_idx) == 0 or len(one_idx) == 0:
            break
        for _ in range(n_trials):
            i = zero_idx[np.random.randint(len(zero_idx))]
            j = one_idx[np.random.randint(len(one_idx))]
            h_new = best_h.copy()
            h_new[i] = 1.0
            h_new[j] = 0.0
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new
                h = h_new
                improved = True
                break
        if not improved and np.random.rand() < 0.1:
            # Double swap
            if len(zero_idx) >= 2 and len(one_idx) >= 2:
                i1 = zero_idx[np.random.randint(len(zero_idx))]
                i2 = zero_idx[np.random.randint(len(zero_idx))]
                j1 = one_idx[np.random.randint(len(one_idx))]
                j2 = one_idx[np.random.randint(len(one_idx))]
                if i1 != i2 and j1 != j2:
                    h_new = best_h.copy()
                    h_new[i1] = 1.0
                    h_new[i2] = 1.0
                    h_new[j1] = 0.0
                    h_new[j2] = 0.0
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5 - 1e-12:
                        best_c5 = c5_new
                        best_h = h_new
                        h = h_new
    return best_h, best_c5

def upsample(h, n_new):
    old_n = len(h)
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        n_fallback = 100
        h_init = np.zeros(n_fallback)
        ones = np.random.choice(n_fallback, n_fallback // 2, replace=False)
        h_init[ones] = 1.0

    n_init = len(h_init)
    h_init = project(h_init, n_init)

    # Evaluate initial
    try:
        if 'evaluate_erdos_solution' in globals():
            res = evaluate_erdos_solution(h_init)
            if isinstance(res, tuple) and len(res) >= 2:
                init_c5 = float(res[1])
            else:
                init_c5 = compute_c5(h_init)
        else:
            init_c5 = compute_c5(h_init)
    except Exception:
        init_c5 = compute_c5(h_init)

    best_h = h_init.copy()
    best_c5 = init_c5

    # Phase 1: Optimize at initial n
    elapsed = time.time() - t_start
    if budget_s - elapsed > 100:
        alloc = min(0.35 * budget_s, budget_s - elapsed - 60)
        h_opt, c5 = subgradient_opt(h_init, alloc, lr_init=0.4)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt

    # Phase 1b: Binary local search at initial n
    elapsed = time.time() - t_start
    if budget_s - elapsed > 60:
        alloc = min(0.15 * budget_s, budget_s - elapsed - 40)
        h_bin, c5_bin = binary_local_search(best_h, alloc)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin

    # Phase 2: Try larger n_points
    elapsed = time.time() - t_start
    n_targets = [100, 140, 200, 300]
    for n_try in n_targets:
        elapsed = time.time() - t_start
        if budget_s - elapsed < 80:
            break
        if n_try <= n_init:
            continue
        h_up = upsample(best_h, n_try)
        c5_up = compute_c5(h_up)
        alloc_sub = min(0.2 * (budget_s - elapsed - 40), 250)
        if alloc_sub < 30:
            break
        h_opt, c5 = subgradient_opt(h_up, alloc_sub, lr_init=0.25)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
        # Quick binary search
        elapsed = time.time() - t_start
        if budget_s - elapsed > 40 and c5 < best_c5 + 0.01:
            alloc_bin = min(40, budget_s - elapsed - 30)
            h_b, c5_b = binary_local_search(h_opt, alloc_bin, n_trials=100)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b

    # Phase 3: Final polish on best solution
    elapsed = time.time() - t_start
    if budget_s - elapsed > 30:
        alloc = budget_s - elapsed - 5
        n_best = len(best_h)
        h_opt, c5 = subgradient_opt(best_h, alloc, lr_init=0.15, n_lags=10)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt

    # Final projection
    n_final = len(best_h)
    best_h = project(best_h, n_final)

    # Final evaluation
    try:
        if 'evaluate_erdos_solution' in globals():
            res = evaluate_erdos_solution(best_h)
            if isinstance(res, tuple) and len(res) >= 2:
                final_c5 = float(res[1])
            else:
                final_c5 = compute_c5(best_h)
        else:
            final_c5 = compute_c5(best_h)
    except Exception:
        final_c5 = compute_c5(best_h)

    return best_h, final_c5, n_final
