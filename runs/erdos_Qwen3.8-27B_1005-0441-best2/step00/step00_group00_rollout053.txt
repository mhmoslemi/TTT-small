            contrib1 = np.zeros(n)

            contrib2 = np.zeros(n)
            contrib2[l:] = -h[:n-l]

            # l < 0, let m = -l
            m = -l

            contrib1[m:] = 1.0 - h[:n-m]

            contrib2 = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
import time


def proj_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s > target:
        lo, hi = 0.0, 1.0
        for _ in range(70):
            mid = (lo + hi) / 2.0
            if np.clip(h - mid, 0.0, 1.0).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    else:
        lo, hi = 0.0, 1.0
        for _ in range(70):
            mid = (lo + hi) / 2.0
            if np.clip(h + mid, 0.0, 1.0).sum() < target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def subgradient(h, n, top_k=3):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for ki in idx:
        l = ki - (n - 1)
        c1 = np.zeros(n)
        c2 = np.zeros(n)
        if l >= 0:
            c1[:n - l] = 1.0 - h[l:n]
            if l > 0:
                c2[l:] = -h[:n - l]
        else:
            m = -l
            c1[m:] = 1.0 - h[:n - m]
            c2[:n - m] = -h[m:n]
        grad += (c1 + c2) * dx
    return grad / top_k


def gd_optimize(h0, n, lr0, max_iters, timeout, top_k=3, momentum=0.65):
    h = proj_feasible(h0.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr0
    vel = np.zeros(n)
    stall = 0
    t0 = time.time()
    for _ in range(max_iters):
        if time.time() - t0 > timeout:
            break
        g = subgradient(h, n, top_k)
        vel = momentum * vel + (1.0 - momentum) * g
        h = proj_feasible(h - lr * vel, n)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall % 300 == 0 and lr > 1e-8:
                lr *= 0.4
                vel *= 0.3
    return best_h, best_c5
