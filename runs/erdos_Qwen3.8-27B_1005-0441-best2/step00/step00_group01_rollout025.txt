    return np.clip(y - 0.5*(lo+hi), 0.0, 1.0)

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def _sog(h, n, alpha):
    """Smooth max objective and gradient."""
    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode='full')
    c = corr * dx
    m = c.max()
    ec = np.exp(alpha*(c-m))
    s = ec.sum()
    f = m + np.log(s)/alpha
    w = ec / s
    gr = np.zeros(n)
    for l in range(2*n-1):
        if w[l] < 1e-13:
            continue
        k = l - (n-1)
        wl = w[l] * dx
        a, b = max(0,k), min(n, n+k)
        if a < b:
            gr[a:b] += wl * (1.0 - h[a-k:b-k])
        a2, b2 = max(0,-k), min(n, n-k)
        if a2 < b2:
            gr[a2:b2] -= wl * h[a2+k:b2+k]
    return f, gr

def _pg(h0, n, tlim):
    """Projected gradient with Adam."""
    h = h0.copy()
    tgt = n/2.0
    bh, bf = h.copy(), _c5(h)
    mv = np.zeros(n); vv = np.zeros(n)
    t = 0; al = 30.0; lr = 0.05
    t0 = time.time()
    while time.time()-t0 < tlim:
        t += 1
        f, g = _sog(h, n, al)
        mv = 0.9*mv + 0.1*g
        vv = 0.999*vv + 0.001*g*g
        mh = mv/(1-0.9**t)
        vh = vv/(1-0.999**t)
        h = _project(h - lr*mh/(np.sqrt(vh)+1e-8), tgt)
        cf = _c5(h)
        if cf < bf:
            bf, bh = cf, h.copy()
        if t % 300 == 0:
            al = min(al*1.4, 500.0)
        if t % 1000 == 0:
            lr *= 0.7
    return bh, bf

def _ls(h0, n, tlim):
    """Local search via random swaps."""
    h = h0.copy()
    bf = _c5(h)
    bh = h.copy()
    rng = np.random.RandomState(42)
    t0 = time.time()
    step = 0.1
    while time.time()-t0 < tlim:
        i, j = rng.randint(0, n), rng.randint(0, n)
        if i == j:
            continue
        di = step * (1.0 - h[i]) * rng.choice([-1,1])
        dj = -di
        h[i] += di; h[j] += dj
        cf = _c5(h)
        if cf < bf - 1e-12:
            bf, bh = cf, h.copy()
            step = min(step*1.2, 0.5)
        else:
            h[i] -= di; h[j] -= dj
            if step > 0.01:
                step *= 0.8
    return bh, bf

def _slsqp(h0, n):
    def obj(x):
        return _c5(x)
    cons = [{'type':'eq','fun':lambda x: np.sum(x)-n/2.0,'jac':lambda x: np.ones(n)}]
    res = minimize(obj, h0, method='SLSQP',
                   bounds=[(0,1)]*n, constraints=cons,
                   options={'maxiter':300,'ftol':1e-14})
    return res.x, _c5(res.x)

def _mk_h(n, rng):
    h = rng.uniform(0.15, 0.85, n)
    return _project(h, n/2.0)

def run(seed=42, budget_s=1100, **kwargs):
