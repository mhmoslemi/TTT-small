import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def c5_grad_analytic(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(corr) * dx
    max_idx = np.argmax(corr)
    p = max_idx - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        if p == 0:
            g = 1.0 - 2.0 * h[i]
        else:
            ip = i - p
            iq = i + p
            if 0 <= ip < n:
                g += 1.0 - h[ip]
            if 0 <= iq < n:
                g -= h[iq]
        grad[i] = g * dx
    return c5, grad

def project_polytope(v, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - mid, 0.0, 1.0)

def projected_subgradient(h0, budget_s):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    h = project_polytope(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    alpha = 0.05
    for it in range(50000):
        if time.time() - t0 > budget_s:
            break
        c5, grad = c5_grad_analytic(h)
        h_new = h - alpha * grad
        h_new = project_polytope(h_new, target)
        h = h_new
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        alpha *= 0.9999
    return best_h, best_c5

def smooth_max_func(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = corr.max()
    return m + (1.0 / T) * np.log(np.sum(np.exp(T * (corr - m))))

def smooth_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = corr.max()
    shifts = T * (corr - m)
    shifts = shifts - shifts.max()  # extra safety
    weights = np.exp(shifts) / np.sum(np.exp(shifts))
    # Gradient of smoothed max = weighted sum of gradients of each f_k
    # f_k = corr[k] * dx
    # For the weight-weighted gradient:
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-10:
            continue
        p = k - (n - 1)
        for i in range(n):
            g = 0.0
            if p == 0:
                g = 1.0 - 2.0 * h[i]
            else:
                ip = i - p
                iq = i + p
                if 0 <= ip < n:
                    g += 1.0 - h[ip]
                if 0 <= iq < n:
                    g -= h[iq]
            grad[i] += weights[k] * g * dx
    return grad

def slsqp_optimize(h0, budget_s, T=20.0):
    n = len(h0)
    target = n / 2.0
    h0_p = project_polytope(h0, target)
    
    def objective(h):
        return smooth_max_func(h, T)
    
    def gradient(h):
        return smooth_max_grad(h, T)
    
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,
                    'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    t0 = time.time()
    res = minimize(objective, h0_p, jac=gradient, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': 200, 'ftol': 1e-10})
    if res.success and res.fun < compute_c5(h0_p):
        return res.x, compute_c5(res.x)
    return h0_p, compute_c5(h0_p)

def upscale(h, factor):
    """Upscale h by interpolation."""
    n_old = len(h)
    n_new = n_old * factor
    x_old = np.arange(n_old)
    x_new = np.linspace(0, n_old - 1, n_new)
    h_new = np.interp(x_new, x_old, h)
    # Adjust to satisfy sum constraint
    target = n_new / 2.0
    h_new = project_polytope(h_new, target)
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def single_lag_grad(h, p):
    """Gradient of f_p = sum_j h[j+p]*(1-h[j]) w.r.t. h, times dx."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if p == 0:
        g[:] = (1.0 - 2.0 * h)
    else:
        for i in range(n):
            v = 0.0
            ip = i - p
            iq = i + p
            if 0 <= ip < n:
                v += 1.0 - h[ip]
            if 0 <= iq < n:
                v -= h[iq]
            g[i] = v * dx
    return g

def smooth_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = corr.max()
    shifted = T * (corr - m)
    shifted -= shifted.max()
    return m + np.log(np.sum(np.exp(shifted))) / T

def smooth_max_grad_vec(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = corr.max()
    shifted = T * (corr - m)
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    grad = np.zeros(n)
    active = w > 1e-8
    for k in np.where(active)[0]:
        p = k - (n - 1)
        grad += w[k] * single_lag_grad(h, p)
    return grad

def project_polytope(v, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - mid, 0.0, 1.0)

def slsqp_opt(h0, budget_s, T=30.0, maxiter=300):
    n = len(h0)
    target = n / 2.0
    h0 = project_polytope(h0.copy(), target)
    elapsed0 = time.time()

    def _elapsed():
        return time.time() - elapsed0

    def obj(h):
        return smooth_max_obj(h, T)

    def jac(h):
        return smooth_max_grad_vec(h, T)

    cons = [
        {"type": "eq", "fun": lambda h: np.sum(h) - target, "jac": lambda h: np.ones(n)}
    ]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=jac, method="SLSQP", bounds=bnds,
                       constraints=cons,
                       options={"maxiter": maxiter, "ftol": 1e-12})
        if res.x is not None:
            res.x = project_polytope(res.x, target)
            return res.x
    except Exception:
        pass
    return h0

def subgradient_opt(h0, budget_s, max_iter=80000):
    n = len(h0)
    target = n / 2.0
    h = project_polytope(h0.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    alpha = 0.02
    decay = 0.99995
    patience = 0
    for it in range(max_iter):
        if time.time() - t0 > budget_s:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        p = max_idx - (n - 1)
        g = single_lag_grad(h, p)
        h = h - alpha * g
        h = project_polytope(h, target)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        if patience > 2000:
            alpha *= 0.5
            patience = 0
        alpha *= decay
    return best_h, best_c5

def upscale(h, factor):
    n_old = len(h)
    n_new = n_old * factor
    if n_new % 2 != 0:
        n_new += 1
    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0.0, n_old - 1, n_new)
    h_new = np.interp(x_new, x_old, h)
    target = n_new / 2.0
    return project_polytope(h_new, target)

def random_restart(h0, n, scale=0.05):
    h = h0.copy()
    h += np.random.randn(n) * scale
    target = n / 2.0
    return project_polytope(h, target)

                break

        h3 = slsqp_opt(h_up, rem * 0.3, T=30, maxiter=200)
