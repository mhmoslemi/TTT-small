                    res2 = minimize(penalized_obj, h_cur, method='Nelder-Mead',
                                   options={'maxiter': 2000, 'xatol': 1e-9, 'fatol': 1e-9})
                    h_cur = project_feasible(res2.x, n)
                    try_update(h_cur, n)
            except Exception:
                pass

            # Local coordinate search
            if time.time() - t0 < budget_s - 50:
                h_cur, _ = local_search(h_cur, n, rng, max_iters=300)
                try_update(h_cur, n)

    # Final polish on best solution
    if best_h is not None and time.time() - t0 < budget_s - 5:
        best_h, _ = local_search(best_h, best_n, rng, max_iters=500)
        best_h = project_feasible(best_h, best_n)
        best_c5 = eval_c5(best_h)

    if best_h is None:
        n = n_init if n_init > 0 else 47
        best_h = np.ones(n) * 0.5
        best_n = n
        best_c5 = eval_c5(best_h)

    return best_h, best_c5, best_n
