    # Project h onto {x : 0<=x<=1, sum(x)=target_sum}
    # Binary search on the Lagrange multiplier
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

    m = np.max(val)
    to = T * (val - m)
    to = np.clip(to, -500, 500)
    return m + np.log(np.sum(np.exp(to))) / T

def soft_max_grad(val, T, h):
    n = len(h)
    dx = 2.0 / n
    m = np.max(val)
    to = T * (val - m)
    to = np.clip(to, -500, 500)
    exps = np.exp(to)

    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        ms = idx - (n - 1)
        if ms >= 0:
            j1 = np.arange(ms, n)
            if len(j1) > 0:
                grad[j1] += wi * (1.0 - h[j1 - ms])
            j2 = np.arange(0, n - ms)
            if len(j2) > 0:
                grad[j2] -= wi * h[j2 + ms]
        else:
            p = -ms
            j1 = np.arange(0, n - p)
            if len(j1) > 0:
                grad[j1] += wi * (1.0 - h[j1 + p])
            j2 = np.arange(p, n)
            if len(j2) > 0:
                grad[j2] -= wi * h[j2 - p]
    return grad * dx

def projected_grad_step(h, grad, target_sum, alpha=0.05):
    h_new = h - alpha * grad
    return project_feasible(h_new, target_sum)
