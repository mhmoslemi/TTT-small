def compute_c5_and_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr), corr

def project_feasible(h, target_sum):
    """Project h onto {0 ≤ x ≤ 1, sum(x) = target_sum}"""
    lo, hi = -h.max(), 1.0 - h.min()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        p = np.clip(h + mid, 0.0, 1.0)
        if p.sum() > target_sum:
            hi = mid
        else:
            lo = mid
    p = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune
    diff = target_sum - p.sum()
    if abs(diff) > 1e-12:
        mask = (p > 1e-8) & (p < 1.0 - 1e-8)
        if mask.any():
            p[mask] += diff / mask.sum()
            p = np.clip(p, 0.0, 1.0)
    return p

def grad_at_lag(h, lag):

    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
