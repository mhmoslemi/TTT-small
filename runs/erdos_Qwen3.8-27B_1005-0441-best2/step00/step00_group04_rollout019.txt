Actually for m < 0 (s = -m > 0):
result = sum_j h[j] * (1-h[j-s]) for j from s to n-1

h[j] appears:
- As the first factor: in term i=j, contributes (1-h[j-s]) for j >= s
- As the second factor: (1-h[i-s]) where i-s = j, so i = j+s. Contributes -h[j+s] for j+s < n (and j+s >= s automatically since j >= 0, s > 0)

So for s > 0:
df/dh[j] = (1-h[j-s]) * [j >= s, j < n] - h[j+s] * [j >= 0, j+s < n]
= (1-h[j-s]) * [s <= j < n] - h[j+s] * [0 <= j < n-s]

def compute_grad(h, shift):
    n = len(h)
    grad = np.zeros(n)
    if shift >= 0:
        # m = shift
        # df/dh[j] = (1-h[j+m]) for j in [0, n-m)
        #           - h[j-m] for j in [m, n)
        if n - shift > 0:
            grad[:n-shift] += (1.0 - h[shift:])
        if shift < n:
            grad[shift:] -= h[:-shift] if shift > 0 else 0
    else:
        s = -shift
        # df/dh[j] = (1-h[j-s]) for j in [s, n)
        #           - h[j+s] for j in [0, n-s)
        grad[s:] += (1.0 - h[:-s])
        if n - s > 0:
            grad[:n-s] -= h[s:]
    return grad

    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fix any numerical issues
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > 0) & (proj < 1)
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0, 1)
    return proj

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    n = len(h)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > 1e-9) & (proj < 1.0 - 1e-9)
        if mask.sum() > 0:
            adj = diff / mask.sum()
            proj[mask] = np.clip(proj[mask] + adj, 0.0, 1.0)
    return proj

def compute_grad(h, m):
    n = len(h)
    grad = np.zeros(n)
    if m >= 0:
        if n - m > 0:
            grad[:n - m] += (1.0 - h[m:])
        if 0 < m < n:
            grad[m:] -= h[:n - m]
    else:
        s = -m
        if n - s > 0:
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] -= h[s:]
    return grad

def local_optimize(h, n_pts, max_iter=3000, time_limit=None):
    start = time.time()
    target_sum = n_pts / 2.0
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    step = 0.15
    for it in range(max_iter):
        if time_limit is not None and time.time() - start > time_limit:
            break
        dmh = 1.0 - h
        corr = np.correlate(h, dmh, mode="full")
        k_star = int(np.argmax(corr))
        cur_val = corr[k_star] * (2.0 / n_pts)
        if cur_val < best_c5 - 1e-12:
            best_c5 = cur_val
            best_h = h.copy()
        m = k_star - n_pts + 1
        grad = compute_grad(h, m)
        h_new = h - step * grad
        h_new = project_feasible(h_new, target_sum)
        new_c5 = evaluate_c5(h_new)
        if new_c5 <= best_c5:
            h = h_new
            step = min(step * 1.02, 0.5)
        else:
            h = h_new
            step *= 0.9
        if step < 1e-6:
            break
    return best_h, best_c5

def targeted_local(h, n_pts, max_iter=5000, time_limit=None):
    start = time.time()
    target_sum = n_pts / 2.0
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    step = 0.08
    for it in range(max_iter):
        if time_limit is not None and time.time() - start > time_limit:
            break
        dmh = 1.0 - h
        corr = np.correlate(h, dmh, mode="full") * (2.0 / n_pts)
        k_star = int(np.argmax(corr))
        cur_val = corr[k_star]
        if cur_val < best_c5 - 1e-12:
            best_c5 = cur_val
            best_h = h.copy()
        # Use top-3 shifts for averaged gradient
        top_k = np.argsort(corr)[-3:]
        m_list = [int(k) - n_pts + 1 for k in top_k]
        grad = np.zeros(n_pts)
        for m in m_list:
            grad += compute_grad(h, m)
        grad /= len(m_list)
        h_new = h - step * grad
        h_new = project_feasible(h_new, target_sum)
        new_c5 = evaluate_c5(h_new)
        if new_c5 <= best_c5:
            h = h_new
        else:
            h = h_new
            step *= 0.92
        if step < 1e-7:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
