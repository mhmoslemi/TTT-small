```python
    best_h, best_c5 = sa_phase(best_h, n, rng, t_end, 0.0005, 0.004)

    # Final tiny greedy
    best_h, best_c5 = greedy_phase(best_h, n, rng, t_end, 0.002)

    # Ensure exact sum constraint
    best_h = fix_sum(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, n)
```
