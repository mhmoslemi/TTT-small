import numpy as np

def project_feasible(h, n):
    """Euclidean projection onto {0<=h[i]<=1, sum(h)=n/2} via bisection on shift."""
    target = n / 2.0
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def c5_of(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))
