import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def compute_correlation(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return c * dx

def gradient_single_c(h, j):
    n = len(h)
    grad = np.zeros(n)
    if j >= 0:
        grad[:n-j] += (1.0 - h[j:n])
        if j > 0:
            grad[j:] -= h[:n-j]
    else:
        jj = -j
        grad[jj:] += (1.0 - h[:n-jj])
        if jj > 0:
            grad[:n-jj] -= h[jj:]
    return grad

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-12:
        h = h * (target_sum / s)
    h = np.clip(h, 0.0, 1.0)
    return h

def softmax_gradient(c, h, lam):
    """Compute gradient of (1/lam)*log(sum(exp(lam*c_k)))"""
    n = len(h)
    shifted = c - np.max(c)  # for numerical stability
    weights = np.exp(lam * shifted)
    weights /= weights.sum()
    grad = np.zeros(n)
    for k in range(len(c)):
        j = k - n + 1
        grad += weights[k] * gradient_single_c(h, j)
    return grad

def project_feasible(h, target_sum):

    s = h.sum()

        h = h * (target_sum / s)

        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
