import numpy as np
import time

def project_feasible(v, target_sum):
    n = len(v)
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        h = np.clip(v - mid, 0.0, 1.0)
        if np.sum(h) > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def obj(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def grad_fd(h, dx, eps=1e-5):
    n = len(h)
    c0 = obj(h, dx)
    g = np.zeros(n)
    for i in range(n):
        hp = h.copy()
        hp[i] += eps
        g[i] = (obj(hp, dx) - c0) / eps
    return g

def pgd(h0, dx, tgt, tmax, lr0=0.02, seed=0):

    h = h0.copy()
    bh = h.copy()

    t = time.time()
    while time.time() - t < tmax:
        g = grad_fd(h, dx)
        h2 = project_feasible(h - lr * g, tgt)
        c2 = obj(h2, dx)
        if c2 < bc:
            bc = c2
            bh = h2.copy()
            h = h2
            ni = 0

                lr *= 0.5
