def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target, rng):
    s = np.sum(h)
    if s > target:
        excess = s - target
        for idx in rng.permutation(len(h)):
            r = min(excess, h[idx])
            h[idx] -= r
            excess -= r
            if excess < 1e-12:
                break
    elif s < target:
        deficit = target - s
        for idx in rng.permutation(len(h)):
            a = min(deficit, 1.0 - h[idx])
            h[idx] += a
            deficit -= a
            if deficit < 1e-12:
                break
    return h
