def project_feasible(h, target_sum):

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(c), c

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(c), c

def weighted_subgradient(c, h, n, T=40.0):
    dx = 2.0 / n
    full_len = 2 * n - 1
    c_shifted = T * (c - c.max())
    w = np.exp(c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        idx = np.argmax(c)
        w = np.zeros(full_len)
        w[idx] = 1.0
    else:

    j = np.arange(n)

        lag = li - (n - 1)
