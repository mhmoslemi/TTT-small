import numpy as np
import time

def normalize_h(h, target_sum):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = h.sum() - target_sum
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            m = h > 1e-12
            if not m.any():
                break
            h[m] -= diff * h[m] / h[m].sum()
        else:
            m = h < 1.0 - 1e-12
            if not m.any():
                break
            h[m] += (-diff) * (1.0 - h[m]) / (1.0 - h[m]).sum()
        h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target_sum
    if abs(diff) > 1e-13:
        m = (h > 0) if diff > 0 else (h < 1)
        if m.any():
            h[m] -= diff / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def compute_grad(h, dx, n, n_worst):
    corr = np.correlate(h, 1.0 - h, mode="full")
    overlaps = corr * dx
    top_idx = np.argsort(overlaps)[-n_worst:]
    grad = np.zeros(n)
    for idx in top_idx:
        s = int(idx - (n - 1))
        if s == 0:
            grad += (1.0 - 2.0 * h) * dx
        elif s > 0:
            a1 = np.arange(s, n)
            grad[a1] += (1.0 - h[a1 - s]) * dx
            a2 = np.arange(0, n - s)
            grad[a2] -= h[a2 + s] * dx
        else:
            sp = -s
            a1 = np.arange(0, n - sp)
            grad[a1] += (1.0 - h[a1 + sp]) * dx
            a2 = np.arange(sp, n)
            grad[a2] -= h[a2 - sp] * dx
    return grad, overlaps

def gradient_optimize(h, target, dx, n, budget_s, t0, rng, lr_init=0.06, n_worst=5):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    stale = 0
    for it in range(500000):
        if budget_s - (time.time() - t0) < 20:
            break
        grad, overlaps = compute_grad(h, dx, n, n_worst)
        cur = float(overlaps.max())
        if cur < best_c5:
            best_c5 = cur
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        moved = False
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_t = h - lr * frac * grad
            h_t = np.clip(h_t, 0.0, 1.0)
            h_t = normalize_h(h_t, target)
            c5_t = compute_c5(h_t)
            if c5_t < cur:
                h = h_t
                moved = True
                break
        if not moved:
            lr = max(lr * 0.93, 0.0003)
        if stale > 800:
            lr = max(lr * 0.7, 0.0003)
            stale = 0
            # perturbation
            hp = h + rng.randn(n) * 0.015
            hp = np.clip(hp, 0, 1)
            hp = normalize_h(hp, target)
            if compute_c5(hp) < best_c5:
                h = hp
                best_c5 = compute_c5(h)
                best_h = h.copy()
    return best_h, best_c5

def binary_local_search(h_b, budget_s, t0, rng):

    h_b = h_b.copy()
    best_c5 = compute_c5(h_b)
    best_h = h_b.copy()
    for sweep in range(50000):

            break
        for i in ones:

                    break
    return best_h, best_c5
