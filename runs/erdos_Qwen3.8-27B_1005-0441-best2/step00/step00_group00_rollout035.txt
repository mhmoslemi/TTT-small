import numpy as np
import time

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = h.sum() - target
    for _ in range(500):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-15
            cnt = int(mask.sum())
            if cnt == 0:
                break
            step = min(diff / cnt, h[mask].min())
            h[mask] -= step
            diff -= step * cnt
        else:
            mask = h < 1.0 - 1e-15
            cnt = int(mask.sum())
            if cnt == 0:
                break
            step = min(-diff / cnt, (1.0 - h[mask]).min())
            h[mask] += step
            diff += step * cnt
    return np.clip(h, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)
