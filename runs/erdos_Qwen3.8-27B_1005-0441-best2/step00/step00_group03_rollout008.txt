import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_detail(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    return vals, int(np.argmax(vals))

def project_feasible(h, n, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 0.001) & (h < 0.999)
        nm = mask.sum()
        if nm > 0:
            h[mask] += diff / nm
        else:
            h += diff / n
    h = np.clip(h, 0.0, 1.0)
    # Final correction on free entries
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.where((h > 0) & (h < 1))[0]
        if len(idx) > 0:
            h[idx[0]] += diff
        else:
            h[np.argmin(h)] += max(0, diff)
            h[np.argmax(h)] += min(0, diff)
        h = np.clip(h, 0.0, 1.0)
    return h

def perturb_pair(h, n, rng):
    """Move small mass between two positions."""
    h_new = h.copy()
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    if rng.random() < 0.5:
        d = min(h_new[i], 1.0 - h_new[j])
    else:
        i, j = j, i
        d = min(h_new[i], 1.0 - h_new[j])
    if d <= 1e-6:
        return h
    delta = rng.uniform(0.005, 0.3) * d
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def perturb_block(h, n, rng):
    """Shift a small block by 1-3 positions."""
    h_new = h.copy()
    block_len = rng.integers(2, max(3, n // 10))
    start = rng.integers(0, max(1, n - block_len))
    shift = rng.integers(-3, 4)
    if shift == 0:
        return h
    new_start = max(0, min(n - block_len, start + shift))
    if new_start == start:
        return h
    # Copy block to new position (overwrite)
    block = h_new[start:start + block_len].copy()
    h_new[start:start + block_len] = 0.0
    h_new[new_start:new_start + block_len] = np.clip(block, 0, 1)
    return h_new

def perturb_gaussian(h, n, rng, scale=0.05):
    """Small Gaussian noise then project."""
    h_new = h + rng.normal(0, scale, n)
    return project_feasible(h_new, n, n / 2.0)

def perturb_lag_focus(h, n, dx, rng, iters=50):

    h_new = h.copy()

        vals = corr * dx
