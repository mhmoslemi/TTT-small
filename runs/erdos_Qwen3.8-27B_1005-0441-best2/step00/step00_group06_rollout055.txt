```python
def project_feasible(h, target_sum):
    h = h.copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = s - target_sum
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h > 1e-12
            if mask.sum() == 0: break
            step = min(diff / mask.sum(), h[mask].min())
            h[mask] -= step
        else:
            mask = h < 1.0 - 1e-12
            if mask.sum() == 0: break
            step = min(-diff / mask.sum(), 1.0 - h[mask].max())
            h[mask] += step
    h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_c5_max(h, margin_frac=0.05):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = c.max()
    # Active set
    active = c >= cmax - margin_frac * max(cmax, 1.0)
    # Finite difference gradient for active set
    grad = np.zeros(n)

        h2[j] += eps

    weights = np.exp(tau * c_shifted)
```

```python
def analytic_subgrad(h, margin_frac=0.03):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = c.max()
    active = c >= cmax - margin_frac * cmax
    active_lags = np.arange(len(c))[active] - (n - 1)  # L values
    n_active = len(active_lags)
    
    grad = np.zeros(n)
    for t in range(n):
        # First term: sum over active L where t+L is valid
        s1 = 0.0
        s2 = 0.0
        for L in active_lags:
            idx1 = t + L
            if 0 <= idx1 < n:
                s1 += (1.0 - h[idx1])
            idx2 = t - L
            if 0 <= idx2 < n:
                s2 += h[idx2]
        grad[t] = dx * (s1 - s2) / n_active
    return grad, cmax
```

```python
import numpy as np
import time

def eval_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        d = h.sum() - target
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h > 1e-14
            if not m.any():
                break
            h[m] -= min(d / m.sum(), h[m].min())
        else:
            m = h < 1.0 - 1e-14
            if not m.any():
                break
            h[m] += min(-d / m.sum(), 1.0 - h[m].max())
    return np.clip(h, 0.0, 1.0)

def compute_grad(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = c.max()
    active = c >= cmax - 0.015
    has_act = active.any()
    ai = np.where(active)[0] if has_act else None
    grad = np.zeros(n)
    eps = 1e-3
    for j in range(n):
        hp = h.copy()
        hp[j] += eps
        cp = np.correlate(hp, 1.0 - hp, mode='full') * dx
        if has_act:
            grad[j] = (cp[ai].max() - cmax) / eps
        else:
            grad[j] = (cp.max() - cmax) / eps
    return grad

def optimize(h0, deadline):
    n = len(h0)
    tgt = n / 2.0
    h = project_feasible(h0, tgt)
    bh, bc = h.copy(), eval_c5(h)
    lr, stag = 0.5, 0
    while time.time() < deadline:
        g = compute_grad(h)
        h = project_feasible(h - lr * g, tgt)
        cn = eval_c5(h)
        if cn < bc - 1e-9:
            bh, bc, stag = h.copy(), cn, 0
            lr = min(lr * 1.02, 2.0)
        else:
            stag += 1
            if stag > 40:
                lr *= 0.65
                stag = 0
                if lr < 0.005:
                    break
    return bh, bc

def make_alt_starts(n, rng, x):
    alts = []
    alts.append(np.tile([1.0, 1.0, 0.5, 0.0, 0.0], n // 5 + 1)[:n])
    alts.append(np.tile([1.0, 0.0, 1.0, 0.5, 0.0], n // 5 + 1)[:n])
    alts.append(np.clip(0.5 + 0.35 * np.sin(2.0 * np.pi * x), 0, 1))
    alts.append(np.clip(0.5 + 0.3 * np.cos(4.0 * np.pi * x), 0, 1))
    alts.append(rng.random(n))
    alts.append(np.where(x < 1.0, 1.0, 0.0))
    return alts

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    bh, bc, bn = None, 1.0, 0

    iv = globals().get('initial_h_values')
    if iv is not None:
        h0 = np.array(iv, dtype=float)
        n0 = len(h0)
        h0 = project_feasible(h0, n0 / 2.0)
        c0 = eval_c5(h0)
        bh, bc, bn = h0.copy(), c0, n0
        dl = t0 + min(350, budget_s - 100)
        ho, co = optimize(h0, dl)
        if co < bc:
            bh, bc, bn = ho, co, n0

    n = 95
    x = np.linspace(0.0, 2.0, n)
    alts = make_alt_starts(n, rng, x)
    n_alts = len(alts)
    for idx in range(n_alts):
        if time.time() - t0 > budget_s - 20:
            break
        a = project_feasible(alts[idx].copy(), n / 2.0)
        ca = eval_c5(a)
        if ca < bc:
            bh, bc, bn = a.copy(), ca, n
        sb = min(35.0, max(5.0, (budget_s - 20.0 - (time.time() - t0)) / max(1, n_alts - idx)))
        if sb > 5:
            ho, co = optimize(a, time.time() + sb)
            if co < bc:
                bh, bc, bn = ho, co, n

    if bh is not None and time.time() - t0 < budget_s - 15:
        nn = len(bh)
        for trial in range(10):
            if time.time() - t0 > budget_s - 8:
                break
            pert = 0.03 + 0.02 * (trial % 3)
            hp = project_feasible(bh + pert * rng.randn(nn), nn / 2.0)
            sb = min(12.0, max(3.0, (budget_s - 8.0 - (time.time() - t0)) / max(1, 10 - trial)))
            if sb > 3:
                ho, co = optimize(hp, time.time() + sb)
                if co < bc:
                    bh, bc, bn = ho, co, nn

    if bh is not None and time.time() - t0 < budget_s - 3:
        nn = len(bh)
        for th in [0.3, 0.4, 0.5, 0.6, 0.7]:
            hs = np.where(bh > th, 1.0, 0.0)
            hs = project_feasible(hs, nn / 2.0)
            cs = eval_c5(hs)
            if cs < bc:
                bh, bc = hs, cs
        hs2 = bh.copy()
        hs2[np.argsort(hs2)[:nn // 4]] = np.clip(hs2[np.argsort(hs2)[:nn // 4]] - 0.05, 0, 1)
        hs2[np.argsort(hs2)[-nn // 4:]] = np.clip(hs2[np.argsort(hs2)[-nn // 4:]] + 0.05, 0, 1)
        hs2 = project_feasible(hs2, nn / 2.0)
        if eval_c5(hs2) < bc:
            bh, bc = hs2, eval_c5(hs2)

    if bh is None:
        bh = np.full(95, 0.5)
        bc = eval_c5(bh)
        bn = 95

    return bh, bc, bn
```
