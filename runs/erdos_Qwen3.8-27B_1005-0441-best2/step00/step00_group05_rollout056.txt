import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def smooth_max(corr, temp):
    m = np.max(corr)
    if temp < 1e-9:
        return m
    return m + temp * np.log(np.mean(np.exp((corr - m) / temp)))

def project_constraint(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff * room / s
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff * h / s
    return np.clip(h, 0.0, 1.0)

def objective(x, dx, target, temp, pen_weight):
    c = np.correlate(x, 1.0 - x, 'full') * dx
    sm = smooth_max(c, temp)
    return sm + pen_weight * (x.sum() - target) ** 2

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bnds = [(0.0, 1.0)] * n

    h0 = np.array(initial_h_values, dtype=float).copy()
    h0 = project_constraint(h0, target)
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)

    # Phase 1: Multi-temperature annealing from initial
    temps = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
             0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for t in temps:
        if (time.time() - t0) > budget_s * 0.3:
            break
        try:
            res = minimize(objective, best_h, args=(dx, target, t, 100.0),
                          method='L-BFGS-B', bounds=bnds,
                          options={'maxiter': 300, 'ftol': 1e-16})
            h_new = project_constraint(res.x, target)
            c5 = compute_c5(h_new, dx)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5
        except Exception:
            pass

    # Phase 2: Aggressive random restarts
    for i in range(400):
        if (time.time() - t0) > budget_s * 0.85:
            break
        sig = 0.02 + 0.3 * rng.random()
        hp = project_constraint(best_h + rng.normal(0, sig, n), target)
        for t in [0.001, 0.0003, 0.0001]:
            if (time.time() - t0) > budget_s * 0.85:
                break
            try:
                res = minimize(objective, hp, args=(dx, target, t, 200.0),
                              method='L-BFGS-B', bounds=bnds,
                              options={'maxiter': 150, 'ftol': 1e-15})
                h_new = project_constraint(res.x, target)
                c5 = compute_c5(h_new, dx)
                if c5 < best_c5:
                    best_h, best_c5 = h_new, c5
                    break
            except Exception:
                pass

    # Phase 3: Structured perturbations (block noise)
    for i in range(100):
        if (time.time() - t0) > budget_s * 0.90:
            break
        bp = rng.randint(2, 10)
        n_blocks = n // bp
        noise = np.zeros(n)
        for b in range(n_blocks):
            noise[b * bp:(b + 1) * bp] = rng.normal(0, 0.1)
        hp = project_constraint(best_h + noise, target)
        try:
            res = minimize(objective, hp, args=(dx, target, 0.0005, 200.0),
                          method='L-BFGS-B', bounds=bnds,
                          options={'maxiter': 200, 'ftol': 1e-15})
            h_new = project_constraint(res.x, target)
            c5 = compute_c5(h_new, dx)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5
        except Exception:
            pass

    # Phase 4: Fine polish with very low temperature
    if (time.time() - t0) < budget_s * 0.95:
        for t in [0.00001, 0.000001, 1e-8]:
            if (time.time() - t0) > budget_s * 0.96:
                break
            try:
                res = minimize(objective, best_h, args=(dx, target, t, 500.0),
                              method='L-BFGS-B', bounds=bnds,
                              options={'maxiter': 500, 'ftol': 1e-18})
                h_new = project_constraint(res.x, target)
                c5 = compute_c5(h_new, dx)
                if c5 < best_c5:
                    best_h, best_c5 = h_new, c5
            except Exception:
                pass

    # Phase 5: Try coarser/finer grids by interpolation
    if (time.time() - t0) < budget_s * 0.7:
        for n_trial in [50, 80, 120, 160]:
            if (time.time() - t0) > budget_s * 0.88:
                break
            if n_trial == n:
                continue
            dx_t = 2.0 / n_trial
            target_t = n_trial / 2.0
            bnds_t = [(0.0, 1.0)] * n_trial
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_trial, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = project_constraint(h_up, target_t)
            for t in [0.005, 0.001, 0.0003]:
                if (time.time() - t0) > budget_s * 0.88:
                    break
                try:
                    res = minimize(objective, h_up,
                                  args=(dx_t, target_t, t, 200.0),
                                  method='L-BFGS-B', bounds=bnds_t,
                                  options={'maxiter': 200, 'ftol': 1e-15})
                    h_new = project_constraint(res.x, target_t)
                    c5 = compute_c5(h_new, dx_t)
                    if c5 < best_c5:
                        best_h, best_c5, n = h_new, c5, n_trial
                        dx, target, bnds = dx_t, target_t, bnds_t
                        break
                except Exception:
                    pass

    # Final: verify constraint
    best_h = project_constraint(best_h, target)
    best_c5 = compute_c5(best_h, dx)

    return best_h, best_c5, n
