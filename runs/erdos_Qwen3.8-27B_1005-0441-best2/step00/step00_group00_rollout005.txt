import numpy as np
import time

def project_to_box_sum(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-12:
        mask = (hp > 1e-9) & (hp < 1.0 - 1e-9)
        if np.any(mask):
            hp[mask] += diff / np.count_nonzero(mask)
            hp = np.clip(hp, 0, 1)
        else:
            idx = np.argmin(np.abs(hp - 0.5))
            hp[idx] = np.clip(hp[idx] + diff, 0, 1)
    return hp

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    return np.max(compute_corr(h)) * (2.0 / len(h))

def compute_subgrad(h, topk=5):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_val = np.max(corr)
    sorted_idx = np.argsort(corr)[::-1]
    active = sorted_idx[:topk]
    tol = 1e-8
    active = active[corr[active] >= max_val - tol * max(1, n) * dx]
    if len(active) == 0:
        active = sorted_idx[:1]
    grad = np.zeros(n)
    for k in active:
        offset = k - (n - 1)
        s1 = max(0, -offset)
        e1 = min(n, n - offset)
        if s1 < e1:
            grad[s1:e1] += 1.0 - h[s1 + offset:e1 + offset]
        s2 = max(0, offset)
        e2 = min(n, n + offset)
        if s2 < e2:
            grad[s2:e2] += -h[s2 - offset:e2 - offset]
    grad *= dx
    return grad, corr, max_val

def optimize_projected_subgrad(h0, n, rng, time_limit, max_iter=200000):
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    avg_h = np.zeros(n)
    count = 0
    patience = 0
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        grad, corr, max_val = compute_subgrad(h, topk=min(5, 3 + it // 5000))
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        alpha = 2.0 / (gnorm * np.sqrt(it + 1.0))
        alpha = min(alpha, 0.5)
        h_new = h - alpha * grad
        h_new = project_to_box_sum(h_new, n)
        c5_new = compute_c5(h_new)
        avg_h += h_new / (count + 1.0)
        avg_h -= h_new / (count + 1.0)
        avg_h = (count * avg_h + h_new) / (count + 1.0)
        count += 1
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            patience = 0
        else:
            h = h_new
            patience += 1
            if patience % 2000 == 0:
                c5_avg = compute_c5(avg_h)
                if c5_avg < best_c5:
                    best_c5 = c5_avg
                    best_h = avg_h.copy()
                if patience > 10000:
                    h = project_to_box_sum(h + rng.normal(0, 0.003, n), n)
                    patience = 0
    c5_final = compute_c5(best_h)
    if c5_avg < best_c5 if count > 0 else False:
        c5_final = compute_c5(avg_h)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = avg_h.copy()
    return best_h, best_c5

def make_structured_init(n, rng):
    x = np.linspace(0, 2, n)
    h = 0.5 + 0.3 * np.sin(2 * np.pi * x / 1.0 + rng.uniform(0, 6.28))
    h = np.clip(h, 0, 1)
    h = project_to_box_sum(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):

    n0 = len(h0)

    # Phase 1: optimize from initial
    t_rem = budget_s - (time.time() - start_time) - 10

        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n0

    elapsed = time.time() - start_time
    t_left = budget_s - elapsed - 10

    # Phase 2: try different n values
    if t_left > 30:

        per_trial = t_left / (len(n_trials) + 1)
        for nt in n_trials:

            tl = min(per_trial, budget_s - (time.time() - start_time) - 10)
            if tl < 3:

            # Interpolate from best
            h_init = np.interp(np.linspace(0, 2, nt), np.linspace(0, 2, best_n), best_h)
            h_init = project_to_box_sum(h_init, nt)

    # Phase 3: restarts from best with perturbation
    t_left = budget_s - (time.time() - start_time) - 5
    if t_left > 10:
        n_restarts = min(5, int(t_left // 8) + 1)
        for r in range(n_restarts):
            if time.time() - start_time > budget_s - 5:
                break
            tl = t_left / (n_restarts - r) * 0.7
            if tl < 3:
                break
            h_pert = best_h + rng.normal(0, 0.02, best_n)
            h_pert = project_to_box_sum(h_pert, best_n)
            h_r, c5_r = optimize_projected_subgrad(h_pert, best_n, rng, tl, max_iter=50000)
            if c5_r < best_c5:
                best_h, best_c5 = h_r, c5_r

    # Phase 4: final polish
    t_left = budget_s - (time.time() - start_time) - 2
    if t_left > 3:
        h_f, c5_f = optimize_projected_subgrad(best_h, best_n, rng, t_left, max_iter=100000)

    # Final verification

    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1.0 + 1e-10)
    best_h = np.clip(best_h, 0, 1)
    return best_h, float(best_c5), int(best_n)
