    bounds = [(0.0, 1.0)] * n
    res = minimize(_smooth_obj, h_init.copy(), args=(alpha,),
                   jac=_smooth_grad, method='SLSQP', constraints=cons,
                   bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-14})
    return res

def _optimize_lfbgsp(h_init, n, alpha, max_iter=3000):
    bounds = [(0.0, 1.0)] * n
    res = minimize(_smooth_obj, h_init.copy(), args=(alpha,),
                   jac=_smooth_grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10})
    return res

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except (NameError, UnboundLocalError):
        h0 = np.full(45, 0.5)
    n0 = len(h0)
    h0 = _project(h0, n0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - start_time

    def time_ok(margin=40):
        return elapsed() < budget_s - margin

    sizes = sorted(set([n0, 2*n0, 3*n0, 4*n0, 5*n0]))
    sizes = [s for s in sizes if s <= 400]
    alphas = [60, 120, 250, 500, 800]

    for n in sizes:
        if not time_ok(80):
            break
        if n == n0:
            h_init = best_h.copy()
        else:
            x_old = np.arange(n0, dtype=float) + 0.5
            x_new = np.arange(n, dtype=float) * (n0 / n) + 0.5 * (n0 / n)
            h_init = np.interp(x_new, x_old, best_h)
            h_init = _project(h_init, n)
        c5_init = compute_c5(h_init)

        for alpha in alphas:
            if not time_ok(25):
                break
            try:
                if n <= 120:
                    res = _optimize_slsqp(h_init, n, alpha, max_iter=1500)
                else:
                    res = _optimize_lfbgsp(h_init, n, alpha, max_iter=2000)
                    h_init = _project(res.x, n)
                if res.x is not None:
                    h_opt = _project(res.x, n)
                    c5 = compute_c5(h_opt)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_opt.copy()
                        h_init = h_opt.copy()
                        print(f"n={n} alpha={alpha} C5={best_c5:.8f} t={elapsed():.0f}s")
            except Exception:
                pass

    # Random restarts at best size
    n_best = len(best_h)
    for r in range(30):
        if not time_ok(20):
            break
        scale = 0.15 / (1.0 + 0.2 * r)
        h_p = best_h + rng.uniform(-scale, scale, n_best)
        h_p = _project(h_p, n_best)
        for alpha in [400, 800]:
            if not time_ok(10):
                break
            try:
                res = _optimize_slsqp(h_p, n_best, alpha, max_iter=800)
                h_opt = _project(res.x, n_best)
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
                    print(f"Restart{r} a={alpha} C5={best_c5:.8f}")
            except Exception:
                pass

    # Binary-like polish: try rounding to {0,1} neighborhood
    if time_ok(10):
        h_bin = (best_h > 0.5).astype(float)
        h_bin = _project(h_bin, n_best)
        c5b = compute_c5(h_bin)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = h_bin.copy()

    best_h = _project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
