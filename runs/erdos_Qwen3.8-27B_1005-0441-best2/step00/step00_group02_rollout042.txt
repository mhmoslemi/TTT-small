    """Compute all overlap values. overlaps[s] for s in range(-(n-1), n)."""
    overlaps = np.zeros(2*n - 1)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr.max()

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.zeros(2 * n - 1)
    omh = 1.0 - h
    for k in range(2 * n - 1):
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            overlaps[k] = np.dot(h[:m], omh[s:s+m]) * dx
        else:
            m = n + s
            overlaps[k] = np.dot(h[-s:], omh[:m]) * dx
    return overlaps

def compute_gradient(h, weights):
    n = len(h)
    grad = np.zeros(n)
    omh = 1.0 - h
    for k in range(len(weights)):
        w = weights[k]
        if abs(w) < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[:n-s] += w * omh[s:]
            grad[s:] -= w * h[:n-s]
        else:
            grad[-s:] += w * omh[:n+s]
            grad[:n+s] -= w * h[-s:]
    return grad

    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

    for it in range(1000000):
        if time.time() - t0 > max_time:
            break
        
        overlaps = compute_overlaps(h)
        c5 = overlaps.max()
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Use top 5% of lags for stability
        n_lags = len(overlaps)
        top_k = max(1, n_lags // 10)
        top_indices = np.argsort(overlaps)[-top_k:]
        
        # Normalize weights (softmax-like for smoothness)
        w_raw = overlaps[top_indices]
        w_max = w_raw.max()
        weights = np.zeros(2 * n - 1)
        # Use softmax weighting
        temp = 0.01  # temperature for softmax
        exps = np.exp((w_raw - w_max) / temp)
        weights[top_indices] = exps / exps.sum()

        grad = compute_gradient(h, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        h_new = project_feasible(h - alpha * grad, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < c5:
            h = h_new
            # Optionally increase alpha slightly
            alpha = min(alpha * 1.05, 2.0 / n)

                h = project_feasible(h + 0.01 * rng.randn(n), n)
                alpha = 0.5 / n
    
    # Final check
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5

def compute_overlaps_raw(h):
    n = len(h)
    omh = 1.0 - h
    overlaps = np.empty(2 * n - 1)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            overlaps[k] = np.dot(h[s:], omh[:m])
        else:
            t = -s
            m = n - t
            overlaps[k] = np.dot(h[:m], omh[t:])
    return overlaps

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode='full') * dx).max()

def compute_gradient(h, weights):
    n = len(h)
    grad = np.zeros(n)
    omh = 1.0 - h
    for k in range(len(weights)):
        w = weights[k]
        if w < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += w * omh[:m]
            grad[:m] -= w * h[s:]
        else:
            t = -s
            m = n - t
            grad[:m] += w * omh[t:]
            grad[t:] -= w * h[:m]
    return grad

        h_opt, c5_val = optimize_h(h_osc, time_alloc, seed=seed + 9000)
        results.append((h_opt, c5_val, n_pts))

    if not results:
        # Fallback: uniform
        n = 45
        h = np.full(n, 0.5)
        return h, 0.5, n

    results.sort(key=lambda x: x[1])
    best_h, best_c5, best_n = results[0]

    # Final coordinate descent polish on best
    t2 = time.time()
    remaining = budget_s - (t2 - t_start) - 5
    if remaining > 10:
        h_work = best_h.copy()
        rng = np.random.RandomState(seed + 12345)
        dx = 2.0 / best_n
        for sweep in range(200):
            if time.time() - t2 > remaining - 5:
                break
            h_cd = coord_descent_sweep(h_work.copy(), best_n)
            c5_cd = (compute_overlaps_raw(h_cd) * dx).max()
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd
            else:
                h_work = project_feasible(h_cd + 0.02 * rng.randn(best_n), best_n)

    # Verify
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, best_n
