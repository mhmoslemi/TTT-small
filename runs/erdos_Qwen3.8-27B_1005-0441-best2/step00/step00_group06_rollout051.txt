import numpy as np

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5_val(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h)) * dx)

def compute_smoothed_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = compute_overlaps(h)
    vals = corr * dx
    vmax = np.max(vals)
    exps = np.exp((vals - vmax) / T)
    sw = np.sum(exps)
    weights = exps / sw
    n1 = n - 1
    grad = np.zeros(n)
    for s in range(-n1, n1 + 1):
        w = weights[s + n1]
        if w < 1e-12:
            continue
        j1_lo = max(0, -s)
        j1_hi = min(n - 1, n1 - s)
        if j1_lo <= j1_hi:
            grad[j1_lo:j1_hi + 1] += w * dx * (1.0 - h[j1_lo + s:j1_hi + s + 1])
        j2_lo = max(0, s)
        j2_hi = min(n - 1, n1 + s)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi + 1] -= w * dx * h[j2_lo - s:j2_hi - s + 1]
    return grad

def optimize_single(h_init, t_deadline, max_budget_frac=1.0):
