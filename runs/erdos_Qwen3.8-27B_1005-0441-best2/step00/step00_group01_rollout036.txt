import numpy as np
from scipy.optimize import minimize
import time

def project_simplex(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    lo, hi = h.min() - target_sum - 1.0, h.max() + target_sum + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def overlaps_of(h, n):
    """All 2n-1 overlap values (multiplied by dx)"""
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_shift(h, m, dx, n):
    """Gradient of overlap at shift m"""
    g = np.zeros(n)
    if m >= 0:
        cnt = n - m
        if cnt > 0:
            g[m:] += (1.0 - h[:cnt]) * dx
            g[:cnt] -= h[m:] * dx
    else:
        p = -m
        cnt = n - p
        if cnt > 0:
            g[:cnt] += (1.0 - h[p:]) * dx
            g[p:] -= h[:cnt] * dx
    return g

def smooth_max(overlaps, beta):
    """Smooth maximum via log-sum-exp"""
    m = overlaps.max()
    if m * beta > 700:
        return m + np.log(np.exp(beta * (overlaps - m)).sum()) / beta
    else:
        return np.log(np.exp(beta * (overlaps - m)).sum()) / beta + m

def smooth_grad(overlaps, h, beta, n, dx):
    """Gradient of smooth max"""
    shifts = np.arange(-(n-1), n)
    m = overlaps.max()
    shifted = beta * (overlaps - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for idx in range(len(overlaps)):
        if weights[idx] > 1e-10:
            grad += weights[idx] * grad_shift(h, shifts[idx], dx, n)
    return grad

            if elapsed() > budget_s - 60:

            def obj(x):

                ov = overlaps_of(hv, n)
                return smooth_max(ov, beta)
            
            def obj_grad(x):
                hv = project_simplex(x, target_sum)
                ov = overlaps_of(hv, n)
                return smooth_grad(ov, hv, beta, n, dx)
            
            res = minimize(obj_grad, h, method='L-BFGS-B',
                          jac=True, options={'maxiter': 100, 'ftol': 1e-12})
            h = project_simplex(res.x, target_sum)
            c5 = overlaps_of(h, n).max()

def project_simplex(h, target_sum):

        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:

        else:
            hi = mid

    return np.correlate(h_val, 1.0 - h_val, mode="full") * dx

            g[:cnt] -= h_val[m:n_pts] * dx
