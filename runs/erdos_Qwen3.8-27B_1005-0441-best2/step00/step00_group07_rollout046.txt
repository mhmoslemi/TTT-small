    return float(np.max(corr)) * dx

def c5_smoothed_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(tau * (corr - m)))) / tau

def c5_smoothed_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    weights = np.exp(tau * (corr - m))
    weights /= np.sum(weights)
    
    grad = np.zeros(n)

            s = -shift
            n1 = n - s

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def c5_smoothed_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    vals = np.exp(tau * (corr - m))
    return m + np.log(np.sum(vals)) / tau

def c5_smoothed_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    weights = np.exp(tau * (corr - m))
    sw = np.sum(weights)
    weights /= sw
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-14:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            n1 = n - shift
            grad[:n1] += w * (1.0 - h[shift:shift + n1])
            grad[shift:] += w * (-h[:n1])
        else:
            s = -shift
            n1 = n - s
            grad[s:] += w * (1.0 - h[:n1])
            grad[:n1] += w * (-h[s:s + n1])
    return grad

def project_bounded_simplex(h, target_sum):
    n = len(h)

        if s > target_sum:

        else:
