def compute_correlation(h):
    one_minus_h = 1.0 - h
    return np.correlate(h, one_minus_h, mode="full")

def compute_c5(h, dx):
    corr = compute_correlation(h)
    return np.max(corr) * dx

def compute_gradient(h, k, n):
    """Compute gradient of C(k) with respect to h, where k is the lag (0-indexed, 0..n-1 for positive lags)."""
    grad = np.zeros(n)
    if k >= 0:
        # C(k) = sum_{i=k}^{n-1} h[i] * (1-h[i-k])
        grad[k:n] += (1.0 - h[:n-k])
        grad[:n-k] -= h[k:n]
    else:
        j = -k
        # C(-j) = sum_{i=0}^{n-1-j} h[i] * (1-h[i+j])
        grad[:n-j] += (1.0 - h[j:n])
        grad[j:n] -= h[:n-j]
    return grad

def project_feasible(h, n):
    """Project h onto {0<=h[i]<=1, sum(h)=n/2}."""
    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-12:
            break
        # Water-filling
        h = np.clip(h + diff / n, 0.0, 1.0)
    return h

    corr = compute_correlation(h)
