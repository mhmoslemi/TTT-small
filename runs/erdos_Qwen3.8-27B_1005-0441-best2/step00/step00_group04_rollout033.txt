def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_binary(h, target):
    h = np.where(h > 0.5, 1.0, 0.0)
    s = int(np.sum(h))
    if s > target:
        ones = np.flatnonzero(h)
        idx = np.random.choice(ones, s - target, replace=False)
        h[idx] = 0.0
    elif s < target:
        zeros = np.flatnonzero(1 - h)
        idx = np.random.choice(zeros, target - s, replace=False)
        h[idx] = 1.0
    return h

def local_search(h, max_rounds=50):
    n = len(h)
    target = n // 2
    c5 = compute_c5(h)
    for r in range(max_rounds):
        improved = False
        ones = list(np.flatnonzero(h))
        zeros = list(np.flatnonzero(1 - h))
        best_delta = 0.0
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                nc = compute_c5(h)
                delta = c5 - nc
                h[i] = 1.0
                h[j] = 0.0

                    best_pair = (i, j)

            h[i] = 0.0
            h[j] = 1.0
