import numpy as np
import time

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def c5_value(h):
    return float(np.max(compute_overlaps(h)))

def project_feasible(h, n_half):
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > n_half:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def compute_gradient(h, k_prime, n):
    lag = k_prime - n + 1
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            m1 = np.arange(lag, n)
            grad[m1] += 1.0 - h[m1 - lag]
            m2 = np.arange(0, n - lag)
            grad[m2] -= h[m2 + lag]
    else:
        d = -lag
        if d < n:
            m1 = np.arange(0, n - d)
            grad[m1] += 1.0 - h[m1 + d]
            m2 = np.arange(d, n)
            grad[m2] -= h[m2 - d]
    return grad

def optimize_h(h_init, n, n_half, budget_s, rng, base_lr=0.008, temp=0.003):

    h = project_feasible(np.clip(h_init, 0.0, 1.0), n_half)

    best_c5 = c5_value(h)
    m_s = np.zeros(n)
    v_s = np.zeros(n)
    t = 0
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    n_full = 2 * n - 1
    stagnant = 0
    cur_lr = base_lr
    cur_temp = temp

    while time.time() - t_start < budget_s - 5:
        t += 1
        overlaps = compute_overlaps(h)

        shifted = overlaps / cur_temp
        shifted = shifted - np.max(shifted)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum > 0:
            w = w / w_sum

        grad = np.zeros(n)
        for k in range(n_full):
            if w[k] > 1e-5:
                grad += w[k] * compute_gradient(h, k, n)
        grad /= w_sum if w_sum > 0 else 1.0

        m_s = beta1 * m_s + (1.0 - beta1) * grad
        v_s = beta2 * v_s + (1.0 - beta2) * grad * grad
        m_hat = m_s / (1.0 - beta1 ** t)
        v_hat = v_s / (1.0 - beta2 ** t)

        h = h - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h, n_half)

        new_c5 = c5_value(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1

            cur_lr *= 0.95
        if stagnant > 2000:
            h = best_h + 0.015 * rng.standard_normal(n)
            h = project_feasible(h, n_half)
            cur_lr = base_lr
            cur_temp = temp * 1.5
            stagnant = 0
