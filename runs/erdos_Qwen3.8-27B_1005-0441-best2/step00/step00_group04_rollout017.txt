    for i in sig:
        g += w[i] * grad_c_vec(h, kvals[i])
    return g * dx, omax


def optimize(h0, time_limit, seed=42):
    n = len(h0)
    h = project_h(h0.copy(), n)
    best_h = h.copy()
    best_c5 = compute_overlaps(h).max()

    lr = 0.06
    mu = 0.85
    alpha = 40.0
    v = np.zeros(n)
    no_imp = 0
    rng = np.random.RandomState(seed)
    iters = 0

    t0 = time.time()
    while time.time() - t0 < time_limit:
        iters += 1
        grad, _ = soft_grad(h, alpha)
        v = mu * v - lr * grad
        h_new = project_h(h + v, n)
        c5_new = compute_overlaps(h_new).max()

        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        h = h_new

        if no_imp > 40:
            h = best_h.copy() + 0.015 * rng.randn(n)
            h = project_h(h, n)
            v *= 0.3
            no_imp = 0
            lr = max(lr * 0.85, 0.003)
            alpha = min(alpha * 1.4, 400.0)

        if lr < 0.002 and no_imp > 150:
            break

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(42, 0.5)

    n0 = len(h0)
    best_h = project_h(h0.copy(), n0)
    best_c5 = compute_overlaps(best_h).max()

    # Phase 1: optimize at n0 (60% budget)
    elapsed = time.time() - t_start
    phase1 = min(600, (budget_s - elapsed) * 0.6)
    if phase1 > 20:
        h_opt, c5_opt = optimize(h0, phase1, seed)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 2: try other sizes and starts (35% budget)
    n_candidates = []
    for m in [2, 3, 4]:
        n_candidates.append(n0 * m)
    n_candidates.extend([50, 64, 80, 100])
    n_candidates = list(dict.fromkeys(n_candidates))  # dedupe, keep order

    elapsed = time.time() - t_start
    remaining = (budget_s - elapsed) * 0.35
    per_try = max(remaining / max(len(n_candidates), 1), 10.0)

    for i, n_try in enumerate(n_candidates):
        elapsed = time.time() - t_start
        if budget_s - elapsed < 30:
            break

        if n_try == n0:
            h_init = h0.copy()
        else:
            x0 = (np.arange(n0) + 0.5) * 2.0 / n0
            x1 = (np.arange(n_try) + 0.5) * 2.0 / n_try
            h_interp = np.interp(x1, x0, h0)
            h_rand = np.clip(0.5 + 0.35 * rng.randn(n_try), 0.0, 1.0)
            mix = 0.4 + 0.2 * rng.rand()
            h_init = mix * h_interp + (1.0 - mix) * h_rand

        h_init = project_h(h_init, n_try)
        alloc = min(per_try, budget_s - (time.time() - t_start) - 20)
        if alloc < 10:
            break
        h_opt, c5_opt = optimize(h_init, alloc, seed + 200 + i * 13)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Final polish on best solution
    elapsed = time.time() - t_start
    polish_time = max(budget_s - elapsed - 2, 5.0)
    if polish_time > 5:
        h_opt, c5_opt = optimize(best_h.copy(), polish_time, seed + 999)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Ensure feasibility
    best_h = project_h(best_h, len(best_h))
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, len(best_h)
