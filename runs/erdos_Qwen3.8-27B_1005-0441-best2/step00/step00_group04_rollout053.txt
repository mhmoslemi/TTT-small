    return np.max(c) * dx

        cnt = int(np.sum(interior))
        if cnt > 0:
            result[interior] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
    return result

def _subgradient_descent(h, budget):
    n = len(h)
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    step = 0.02
    start = time.time()
    no_improve = 0
    
    while time.time() - start < budget:
        overlaps = _get_overlaps(h)
        c5 = np.max(overlaps)
        
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve % 300 == 0:
            step *= 0.94
        
        top_k = min(5, 2 * n - 1)
        idx_top = np.argsort(overlaps)[-top_k:]
        ov_top = overlaps[idx_top]
        weights = np.exp(12.0 * (ov_top - ov_top.max()))
        sw = np.sum(weights)
        if sw < 1e-15:
            weights = np.ones(top_k) / top_k
        else:
            weights /= sw
        
        grad = np.zeros(n)
        for i in range(top_k):
            shift = idx_top[i] - (n - 1)
            grad += weights[i] * _get_subgradient(h, shift)
        
        h = _project_feasible(h - step * grad)
    
    return best_h, best_c5

def _smooth_obj(x, lam, penalty):
    h_loc = np.clip(x, 0.0, 1.0)
    ov = _get_overlaps(h_loc)
    m = np.max(ov)
    scaled = lam * (ov - m)
    scaled = np.clip(scaled, -50, 50)
    val = m + np.log(np.sum(np.exp(scaled))) / lam
    s = np.sum(h_loc) - len(h_loc) / 2.0
    val += penalty * s * s
    return val

def _smooth_grad(x, lam, penalty):
    h_loc = np.clip(x, 0.0, 1.0)
    ov = _get_overlaps(h_loc)
    m = np.max(ov)
    scaled = lam * (ov - m)
    scaled = np.clip(scaled, -50, 50)
    exp_s = np.exp(scaled)
    softmax = exp_s / np.sum(exp_s)
    n = len(h_loc)
    grad = np.zeros(n)
    for i in range(len(softmax)):
        if softmax[i] > 1e-8:
            shift = i - (n - 1)
            grad += softmax[i] * _get_subgradient(h_loc, shift)
    s = np.sum(h_loc) - n / 2.0
    grad += 2.0 * penalty * s - 2.0 * penalty * s
    grad += 2.0 * penalty * (np.sum(h_loc) - n / 2.0)
    return grad

def _smooth_optimize(h, lam, budget):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    start = time.time()
    penalty = 50.0
    maxiter = 1500
    try:
        res = minimize(_smooth_obj, h, args=(lam, penalty),
                      jac=_smooth_grad, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        h_res = _project_feasible(res.x)
        c5_res = _compute_c5(h_res)
        return h_res, c5_res
    except Exception:
        return h, _compute_c5(h)

def _local_search(h, budget, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    start = time.time()
    
    while time.time() - start < budget:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(-0.04, 0.04)
        h_trial = best_h.copy()
        h_trial[i] += delta
        h_trial[j] -= delta
        if h_trial[i] < 0.0 or h_trial[i] > 1.0 or h_trial[j] < 0.0 or h_trial[j] > 1.0:
            continue
        c5_trial = _compute_c5(h_trial)
        if c5_trial < best_c5 - 1e-12:
            best_h = h_trial
            best_c5 = c5_trial
    
    return best_h, best_c5

def _perturb_and_refine(h, budget, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    start = time.time()
    
    while time.time() - start < budget:
        h_trial = best_h + rng.uniform(-0.05, 0.05, n)
        h_trial = _project_feasible(h_trial)
        c5_trial = _compute_c5(h_trial)
        if c5_trial < best_c5:
            best_h, best_c5 = h_trial, c5_trial
            h2, c5_2 = _subgradient_descent(h_trial, min(5, budget - (time.time() - start)))
            if c5_2 < best_c5:
                best_h, best_c5 = h2, c5_2
        elif time.time() - start > budget * 0.7:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n = 42
        h0 = np.ones(n) * 0.5
    
    n = len(h0)
    h0 = _project_feasible(h0)
    
    best_h = h0.copy()
    best_c5 = _compute_c5(h0)
    
    # Phase 1: Subgradient descent
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        h1, c5_1 = _subgradient_descent(h0, rem * 0.2)
        if c5_1 < best_c5:
            best_h, best_c5 = h1, c5_1
    
    # Phase 2: Smooth optimization
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        for lam in [25.0, 12.0, 6.0]:
            rem = budget_s - (time.time() - t0)
            if rem < 25:
                break
            h2, c5_2 = _smooth_optimize(best_h, lam, rem * 0.12)
            if c5_2 < best_c5:
                best_h, best_c5 = h2, c5_2
    
    # Phase 3: Refine with subgradient
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        h3, c5_3 = _subgradient_descent(best_h, rem * 0.3)
        if c5_3 < best_c5:
            best_h, best_c5 = h3, c5_3
    
    # Phase 4: Local search
    rem = budget_s - (time.time() - t0)
    if rem > 10:
        h4, c5_4 = _local_search(best_h, rem * 0.3, rng)
        if c5_4 < best_c5:
            best_h, best_c5 = h4, c5_4
    
    # Phase 5: Perturbation to escape local optima
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        h5, c5_5 = _perturb_and_refine(best_h, rem * 0.5, rng)
        if c5_5 < best_c5:
            best_h, best_c5 = h5, c5_5
    
    # Final: ensure feasibility and recompute
    best_h = _project_feasible(best_h)
    best_c5 = _compute_c5(best_h)
    
    return best_h, best_c5, n
