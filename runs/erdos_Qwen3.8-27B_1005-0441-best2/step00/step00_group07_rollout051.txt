The correlation: co[idx] where idx = k + (n-1) gives:
co[idx] = Σ_i h[i+k] * (1-h[i]) for valid i

where "valid" means 0 ≤ i < n and 0 ≤ i+k < n.

d(co[idx])/d(h[j]):
- h[j] appears in h[i+k] when i+k = j, i.e., i = j-k. Valid when 0 ≤ j-k < n, i.e., k ≤ j < n+k.
  Contributes: (1-h[j-k])
- h[j] appears in (1-h[i]) when i = j. Valid when 0 ≤ j < n (always) and 0 ≤ j+k < n.
  Contributes: -h[j+k]

So: d(co[idx])/d(h[j]) = (1-h[j-k])·I(0≤j-k<n) - h[j+k]·I(0≤j+k<n)

With dx factor: d(co[idx]*dx)/d(h[j]) = dx · [(1-h[j-k])·I(0≤j-k<n) - h[j+k]·I(0≤j+k<n)]

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_h = None
    best_c = 1.0

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float)
    else:
        h_start = np.zeros(80)
        h_start[0::2] = 1.0
    n80 = len(h_start)
    h_start = proj_box_sum(h_start, n80 / 2.0)
    best_h = h_start.copy()
    best_c = c5_of(h_start)

    elapsed = time.time() - t0
    tlim = max(10.0, budget_s - elapsed - 300)
    h_opt, c_opt = pgd_optimize(h_start, 8000, 60, 2.0, tlim)
    if c_opt < best_c:
        best_c, best_h = c_opt, h_opt

    elapsed = time.time() - t0
    tlim = max(5.0, budget_s - elapsed - 200)
    h_lb = lbb_optimize(best_h, 80, 50.0, 200)
    h_lb = proj_box_sum(h_lb, len(h_lb) / 2.0)
    c_lb = c5_of(h_lb)
    if c_lb < best_c:
        best_c, best_h = c_lb, h_lb

    elapsed = time.time() - t0
    tlim = max(5.0, budget_s - elapsed - 150)
    h_cr, c_cr = coordinate_refine(best_h, 30, tlim)
    if c_cr < best_c:
        best_c, best_h = c_cr, h_cr

    for attempt in range(10):
        elapsed = time.time() - t0
        if elapsed > budget_s - 50:
            break
        h_rand = np.zeros(80)
        perm = np.random.permutation(80)
        h_rand[perm[:40]] = 1.0
        tlim = min(20.0, (budget_s - elapsed - 50) * 0.15)
        h_opt, c_opt = pgd_optimize(h_rand, 3000, 60, 2.0, tlim)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt
            h_lb = lbb_optimize(best_h, 100, 100.0, 150)
            h_lb = proj_box_sum(h_lb, 40.0)
            c_lb = c5_of(h_lb)
            if c_lb < best_c:
                best_c, best_h = c_lb, h_lb

    elapsed = time.time() - t0
    if elapsed < budget_s - 120:
        n160 = 160
        x_old = (np.arange(len(best_h)) + 0.5) * (2.0 / len(best_h))
        x_new = (np.arange(n160) + 0.5) * (2.0 / n160)
        h160 = np.interp(x_new, x_old, best_h)
        h160 = proj_box_sum(h160, 80.0)
        tlim = max(10.0, budget_s - (time.time() - t0) - 60)
        h_opt, c_opt = pgd_optimize(h160, 4000, 60, 2.0, tlim * 0.6)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt
        elapsed = time.time() - t0
        if elapsed < budget_s - 40:
            h_lb = lbb_optimize(best_h, 80, 50.0, 150)
            h_lb = proj_box_sum(h_lb, len(h_lb) / 2.0)
            c_lb = c5_of(h_lb)
            if c_lb < best_c:
                best_c, best_h = c_lb, h_lb

    elapsed = time.time() - t0
    if elapsed < budget_s - 20:
        h_lb = lbb_optimize(best_h, 150, 200.0, 200)
        h_lb = proj_box_sum(h_lb, len(h_lb) / 2.0)
        c_lb = c5_of(h_lb)
        if c_lb < best_c:
            best_c, best_h = c_lb, h_lb

    n_final = len(best_h)
    best_h = proj_box_sum(best_h, n_final / 2.0)
    best_c = c5_of(best_h)
    return best_h, best_c, n_final
