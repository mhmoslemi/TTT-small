import numpy as np
import time

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = 0.5 * n
    for _ in range(500):
        d = tgt - h.sum()
        if abs(d) < 1e-11:
            break
        m = (h < 1.0 - 1e-10) if d > 0 else (h > 1e-10)
        c = int(m.sum())
        if c == 0:
            break
        h[m] += d / c
        np.clip(h, 0.0, 1.0, out=h)
    return h

def c5_val(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def upsample(h, nn):
    no = len(h)
    xo = (np.arange(no) + 0.5) * 2.0 / no
    xn = (np.arange(nn) + 0.5) * 2.0 / nn
    return project_h(np.interp(xn, xo, h), nn)
