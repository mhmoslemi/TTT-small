import numpy as np
from scipy.optimize import minimize
import time

def compute_correlate(h):
    return np.correlate(h, 1.0 - h, mode="full")

def c5_value(h):
    n = len(h)
    return compute_correlate(h).max() * (2.0 / n)

def project_sum(h, target_sum):
    if abs(h.sum() - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def coord_optimize(h, n, rng, sweeps=50):
    target_sum = n * 0.5
    best_h = h.copy()
    best_c5 = c5_value(h)
    k0 = n - 1
    
    for sweep in range(sweeps):
        order = np.arange(n)
        rng.shuffle(order)
        
        for i in order:
            h[i] = 0.0
            O0 = compute_correlate(h)
            h[i] = 1.0
            O1 = compute_correlate(h)
            
            slope = O1 - O0
            intercept = O0
            
            xg = np.linspace(0.0, 1.0, 65)
            vals = intercept[None, :] + xg[:, None] * slope[None, :]
            vals[:, k0] = O0[k0] + xg - xg * xg
            maxv = vals.max(axis=1)
            h[i] = xg[np.argmin(maxv)]
        
        h = project_sum(h, target_sum)
        c5 = c5_value(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def smooth_obj(h, beta=40.0):
    n = len(h)
    dx = 2.0 / n
    c = compute_correlate(h) * dx
    m = c.max()
    sh = np.clip(beta * (c - m), -50.0, 0.0)
    return m + np.log(np.sum(np.exp(sh))) / beta

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.ones(95) * 0.5
    
    n = len(h0)
    ts = n * 0.5
    h = project_sum(np.clip(h0, 0, 1), ts)
    
    best_h = h.copy()
    best_c5 = c5_value(h)
    
    def elapsed():
        return time.time() - t0
    
    # Phase 1: Coordinate descent
    h, c5 = coord_optimize(h, n, rng, sweeps=100)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()
    
    if elapsed() > budget_s * 0.5:
        return (best_h, best_c5, n)
    
    # Phase 2: scipy SLSQP
    try:
        cons = [{'type': 'eq', 'fun': lambda x: float(np.sum(x) - ts)}]
        bnds = [(0.0, 1.0)] * n
        res = minimize(lambda x: smooth_obj(x, 40.0), best_h,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-12})
        h2 = project_sum(np.clip(res.x, 0, 1), ts)
        c5_2 = c5_value(h2)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2
    except Exception:
        pass
    
    if elapsed() > budget_s * 0.7:
        return (best_h, best_c5, n)
    
    # Phase 3: Perturbation + local opt
    for _ in range(200):
        if elapsed() > budget_s * 0.95:
            break
        pert = rng.randn(n) * 0.08
        h_trial = project_sum(np.clip(best_h + pert, 0, 1), ts)
        h_opt, c5_opt = coord_optimize(h_trial, n, rng, sweeps=15)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt.copy()
    
    best_h = project_sum(best_h, ts)
    best_c5 = c5_value(best_h)
    return (best_h, best_c5, n)

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlate(h):
    return np.correlate(h, 1.0 - h, mode="full")


def c5_value(h):
    n = len(h)
    return compute_correlate(h).max() * (2.0 / n)


def project_sum(h, target_sum):
    if abs(h.sum() - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = compute_correlate(h) * dx
    m = c.max()
    sh = np.clip(beta * (c - m), -50.0, 0.0)
    return m + np.log(np.sum(np.exp(sh))) / beta


def coord_optimize(h, n, rng, sweeps, grid_size=101):
    target_sum = n * 0.5
    best_h = h.copy()
    best_c5 = c5_value(h)
    k0 = n - 1
    xg = np.linspace(0.0, 1.0, grid_size)
    xg2 = xg * xg

    for _ in range(sweeps):
        order = np.arange(n)
        rng.shuffle(order)
        for i in order:
            h[i] = 0.0
            O0 = compute_correlate(h)
            h[i] = 1.0
            O1 = compute_correlate(h)
            slope = O1 - O0
            vals = O0[None, :] + xg[:, None] * slope[None, :]
            vals[:, k0] = O0[k0] + xg - xg2
            h[i] = xg[np.argmin(vals.max(axis=1))]
        h = project_sum(h, target_sum)
        c5 = c5_value(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:

    n = len(h0)

    best_h = h.copy()
    best_c5 = c5_value(h)

    # Phase 1: Extensive coordinate descent
    h, c5 = coord_optimize(h, n, rng, 120, grid_size=101)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    elapsed = time.time() - t0
    if elapsed > budget_s * 0.35:
        return (best_h, best_c5, n)

    # Phase 2: SLSQP on smoothed objective
    try:
        bnds = [(0.0, 1.0)] * n
        res = minimize(smooth_obj, best_h, args=(60.0,),
                       method='SLSQP', bounds=bnds,
                       options={'maxiter': 400, 'ftol': 1e-14})
        h2 = project_sum(np.clip(res.x, 0.0, 1.0), ts)
        h3, c5_3 = coord_optimize(h2, n, rng, 40, grid_size=151)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()
    except Exception:
        pass

    elapsed = time.time() - t0
    if elapsed > budget_s * 0.55:
