        idx = int(np.argmin(h)) if res > 0 else int(np.argmax(h))

import numpy as np

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def fix_feasibility(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            space = 1.0 - h
            s = space.sum()
            if s > 1e-15:
                h += (diff / s) * space
        else:
            s = h.sum()
            if s > 1e-15:
                h += (diff / s) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-14:
        idx = int(np.argmin(h)) if res > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + res, 0.0, 1.0)
    return h
