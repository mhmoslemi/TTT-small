import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def fix_sum(h, target_sum):
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_optimize(h, target_sum, n_sweeps=200, tol=1e-10):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    xs = np.linspace(0.0, 1.0, 121)
    c5_prev = np.inf
    for sweep in range(n_sweeps):
        max_change = 0.0
        for t in range(n):
            old = h[t]
            h[t] = 0.0
            corr0 = np.correlate(h, 1.0 - h, mode='full')
            h[t] = 1.0
            corr1 = np.correlate(h, 1.0 - h, mode='full')
            h[t] = old
            a = corr0.astype(float)
            b = (corr1 - corr0).astype(float)
            vals = a[None, :] + b[None, :] * xs[:, None]
            max_vals = np.max(vals, axis=1)
            x_opt = xs[np.argmin(max_vals)]
            max_change = max(max_change, abs(x_opt - old))
            h[t] = x_opt
        h = fix_sum(h, target_sum)
        c5_cur = compute_c5(h)
        if abs(c5_cur - c5_prev) < tol and max_change < 1e-6:
            break
        c5_prev = c5_cur
    return h, compute_c5(h)
