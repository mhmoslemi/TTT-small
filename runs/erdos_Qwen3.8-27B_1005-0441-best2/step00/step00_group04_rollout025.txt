import time
from scipy import correlate

def compute_c5(h, dx):
    c1 = 1.0 - h
    corr = np.correlate(h, c1, mode="full")
    return np.max(corr) * dx

def subgradient(h, k_star, n):
    """Compute subgradient of overlap at lag k_star w.r.t. h"""
    g = np.zeros(n)
    for j in range(n):
        idx1 = j - k_star
        idx2 = j + k_star
        if 0 <= idx1 < n:
            g[j] += (1.0 - h[idx1])
        if 0 <= idx2 < n:
            g[j] -= h[idx2]
    return g

def project_feasible(h, n):
    """Project h onto {0 ≤ h ≤ 1, sum(h) = n/2}"""
    target = n / 2.0
    # Water-filling
    lo, hi = 0.0, 1.0
    for _ in range(100):

        s = np.sum(clipped)
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix rounding
    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        # Distribute the difference
        result += diff / n
        result = np.clip(result, 0.0, 1.0)
        result[np.sum(result) > target] = min(result[np.sum(result) > target], 1.0)
    return result

    """Subgradient optimization"""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for it in range(max_iters):

def all_overlaps(h, dx):

    g = np.zeros(n)
    if k >= 0:
        nk = n - k
        if nk > 0:
            g[k:] += (1.0 - h[:nk])
            g[:nk] -= h[k:]
    else:
        kk = -k
        nk = n - kk
        if nk > 0:
            g[:nk] += (1.0 - h[kk:])
            g[kk:] -= h[:nk]
    return g
