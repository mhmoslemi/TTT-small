def project(h, target_sum):

def compute_subgrad(h, dx):
    """Subgradient of max_l corr[l]*dx, using top-k active lags."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c_max = np.max(corr)
    spread = c_max - np.min(corr)

            if c5_sg < best_c5:
                best_h, best_c5 = h_sg, c5_sg

    # Phase 3: Try alternative n values if time permits
    t_phase3_end = start + budget_s * 0.92
    for n_alt in [48, 64, 80, 128, 100]:
        if time.time() > t_phase3_end:
            break
        dx_alt = 2.0 / n_alt
        target_alt = n_alt / 2.0
        max_t = min(15.0, budget_s * 0.05)
        h_alt, c5_alt, _ = try_other_n(n_alt, dx_alt, target_alt, rng, max_t)
        if c5_alt < best_c5:
            best_h, best_c5, n, dx = h_alt, c5_alt, n_alt, dx_alt

    # Phase 4: Final fine-tuning
    best_h, best_c5 = subgrad_optimize(best_h, dx, n, target, 5.0, rng, lr_init=0.02)
    h_ls, c5_ls = local_search(best_h, dx, n, target, 400, rng)
    if c5_ls < best_c5:
        best_h, best_c5 = h_ls, c5_ls

    # Final projection and evaluation
    best_h = project(best_h, target)
    best_c5 = c5_val(best_h, dx)

    # Verify with pre-imported evaluator if available
    try:
        verified = evaluate_erdos_solution(best_h, dx)
        if verified < best_c5:
            best_c5 = verified
    except Exception:
        pass

    return best_h, best_c5, n
