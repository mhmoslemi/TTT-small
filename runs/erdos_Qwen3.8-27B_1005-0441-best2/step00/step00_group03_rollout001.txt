import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _smoobj(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    shifted = c - cmax
    s = np.exp(shifted / eps)
    return (cmax + eps * np.log(s.sum())) * dx

def _smograd(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    shifted = c - cmax
    s = np.exp(shifted / eps)
    ssum = s.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = s[k] / ssum
        if wk < 1e-14:
            continue
        m = k - (n - 1)
        if m >= 0:
            g[m:] += wk * (1.0 - h[:n - m])
            g[:n - m] -= wk * h[m:]
        else:
            p = -m
            g[:n - p] += wk * (1.0 - h[p:])
            g[p:] -= wk * h[:n - p]
    return g * dx

def _proj(h, n):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        d = n * 0.5 - h.sum()
        if abs(d) < 1e-13:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        if free.sum() == 0:
            h = np.clip(h + d / n, 0, 1)
            break
        h[free] += d / free.sum()
        h = np.clip(h, 0, 1)
    return h

    return x.sum() - n * 0.5

def _eq_jac(x, n):
    return np.ones(n)
