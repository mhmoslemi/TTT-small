    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        n_init = 95
        h = np.zeros(n_init)
        mid = n_init // 2
        h[mid - 12:mid + 13] = 1.0
        h = project_onto_constraint(h)

    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = float(np.max(compute_overlaps(h))) * dx

    # Cutting plane method with growing active set
    active_set = []
    for it in range(20000):
        elapsed = time.time() - t0
        if elapsed > budget_s - 30:
            break

        ov = compute_overlaps(h)
        c5_curr = float(np.max(ov)) * dx
        if c5_curr < best_c5 - 1e-12:
            best_c5 = c5_curr
            best_h = h.copy()

        # Build active set: top shifts + random
        k_max = int(np.argmax(ov))
        top_k = np.argsort(ov)[-min(20, len(ov)):]
        top_k = list(set(list(top_k) + [k_max]))
        if len(top_k) < 8:
            top_k += list(rng.choice(len(ov), 8 - len(top_k), replace=False))
        top_k = sorted(top_k)[:40]  # cap at 40 constraints

        h_new, t_val, _ = solve_cutting_plane_lp(h, n_top=len(top_k),
                                                   all_indices=np.array(top_k))
        if h_new is None:
            h_new = h

        new_ov = compute_overlaps(h_new)
        new_c5 = float(np.max(new_ov)) * dx

        if new_c5 < c5_curr - 1e-10:
            h = h_new
        else:
            # Subgradient escape
            if it % 50 == 49:
                grad = np.zeros(n)
                for idx in top_k[:5]:
                    grad += 0.2 * compute_gradient(h, int(idx))
                h_try = project_onto_constraint(h - 0.05 * grad)
                tc = float(np.max(compute_overlaps(h_try))) * dx
                if tc < c5_curr:
                    h = h_try

        # Periodic perturbation
        if it % 200 == 199:
            h_pert = project_onto_constraint(h + 0.03 * rng.randn(n))
            if float(np.max(compute_overlaps(h_pert))) * dx < c5_curr:
                h = h_pert

    # Phase 2: Fine subgradient polish
    elapsed = time.time() - t0
    if elapsed < budget_s - 20:
        h = best_h.copy()
        momentum = np.zeros(n)
        beta = 0.85
        step = 0.05
        for it in range(50000):
            if time.time() - t0 > budget_s - 10:
                break
            ov = compute_overlaps(h)
            c5_curr = float(np.max(ov)) * dx
            if c5_curr < best_c5:
                best_c5 = c5_curr
                best_h = h.copy()
            k_act = int(np.argmax(ov))
            top_idx = np.argsort(ov)[-5:]
            logits = 3.0 * (ov[top_idx] - ov[k_act])
            w = np.exp(logits)
            w /= w.sum()
            grad = np.zeros(n)
            for idx, wi in zip(top_idx, w):
                if wi > 0.05:
                    grad += wi * compute_gradient(h, int(idx))
            momentum = beta * momentum + grad
            h_try = project_onto_constraint(h - step * momentum)
            if float(np.max(compute_overlaps(h_try))) * dx < c5_curr:
                h = h_try
                step = min(step * 1.01, 0.3)
            else:
                step *= 0.8
            step *= 0.9998

    # Phase 3: Try binary rounding
    elapsed = time.time() - t0
    if elapsed < budget_s - 5 and n % 2 == 0:
        h_bin = np.zeros(n)
        h_bin[np.argsort(best_h)[-n // 2:]] = 1.0
        c5_bin = float(np.max(compute_overlaps(h_bin))) * dx
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin

    # Final verification
    best_h = project_onto_constraint(best_h)
    best_c5 = float(np.max(compute_overlaps(best_h))) * dx
    return best_h, best_c5, n
