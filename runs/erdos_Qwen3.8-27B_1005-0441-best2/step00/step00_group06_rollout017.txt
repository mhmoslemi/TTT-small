def subgradient_descent(h, deadline, learning_rate=0.01):
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    
    while time.time() < deadline:
        # Compute subgradient
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_max = int(np.argmax(corr))
        shift = k_max - (n - 1)
        
        grad = np.zeros(n)
        for i in range(n):
            g = 0.0
            if 0 <= i + shift < n:
                g += (1.0 - h[i + shift])
            if 0 <= i - shift < n:
                g -= h[i - shift]
            grad[i] = g
        grad *= dx
        
        # Update
        h = h - learning_rate * grad
        
        # Project: 0 <= h <= 1
        h = np.clip(h, 0, 1)
        
        # Fix sum
        h = fix_sum(h, n / 2.0)
    
    return h, compute_c5(h)

    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * 2.0 / n

def fix_sum(h, target):
    h = h.astype(float).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    order = np.argsort(h) if diff > 0 else np.argsort(-h)

            diff -= d
        else:
            d = min(-diff, h[i])
            h[i] -= d
            diff += d
        if abs(diff) < 1e-12:
