        expr = 0
        for j in range(jlo, jhi):
            a0 = h_init[j]
            b0 = h_init[j + s]
            expr += hv[j + s] - hv[j + s] * a0 - hv[j] * b0 + a0 * b0
        cons.append(tv >= dx * expr)
    prob = cp.Problem(cp.Minimize(tv), cons)
    for solver in [cp.CBC, cp.GLOP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, timeout=timeout)
            if hv.value is not None and not np.any(np.isnan(hv.value)):
                return np.clip(hv.value, 0.0, 1.0)
        except Exception:
            continue
    return h_init.copy()

def coord_refine(h, time_limit):
    n = len(h)
    target = n / 2.0
    h = project(h.copy(), target)
    cur = compute_c5(h)
    t0 = time.time()
    deltas = [0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    for _ in range(300):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            for d in deltas:
                val = h[i] + d
                if 0.0 <= val <= 1.0:
                    h_t = h.copy()
                    h_t[i] = val
                    h_t = project(h_t, target)
                    c5_t = compute_c5(h_t)
                    if c5_t < cur - 1e-12:
                        h = h_t
                        cur = c5_t
                        improved = True
                        break
        if not improved:
            break
    return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    global evaluate_erdos_solution
    t_start = time.time()

    def rem():
        return budget_s - time.time() - 10.0

    rng = np.random.default_rng(seed)
    best_h, best_c5 = None, float('inf')

    # Phase 0: Initial point
    has_parent = 'initial_h_values' in globals() and initial_h_values is not None
    if has_parent:
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        h0 = project(rng.random(95), 47.5)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: LP successive linearization from initial
    h_cur = best_h.copy()
    for it in range(15):
        if rem() < 40:
            break
        n = len(h_cur)
        h_lp = lp_solve(h_cur, timeout=25)
        h_lp = project(h_lp, n / 2.0)
        c5_lp = compute_c5(h_lp)
        if c5_lp < best_c5 - 1e-8:
            best_h, best_c5 = h_lp.copy(), c5_lp
            h_cur = h_lp.copy()
        else:
            break

    # Phase 2: Subgradient polish on best
    if rem() > 60:
        t_lim = min(rem() * 0.35, 200)
        h_pg, c5_pg = subgrad_opt(best_h, t_lim)
        if c5_pg < best_c5:
            best_h, best_c5 = h_pg, c5_pg

    # Phase 3: Multi-start with different n and random inits
    n_opts = [95, 100, 120, 150, 200]
    for n_try in n_opts:
        if rem() < 50:
            break
        for rst in range(4):
            if rem() < 30:
                break
            if n_try == 95 and rst == 0 and has_parent:
                h_s = np.asarray(initial_h_values, dtype=float).copy()
            else:
                h_s = project(rng.random(n_try), n_try / 2.0)
            # Quick LP
            h_s = lp_solve(h_s, timeout=15)
            h_s = project(h_s, n_try / 2.0)
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_h, best_c5 = h_s.copy(), c5_s
            # Subgrad polish
            if rem() > 40:
                h_s, c5_s = subgrad_opt(h_s, min(50, rem() * 0.25))
                if c5_s < best_c5:
                    best_h, best_c5 = h_s, c5_s

    # Phase 4: LP refinement on best solution
    if rem() > 30:
        h_cur = best_h.copy()
        for it in range(8):
            if rem() < 40:
                break
            h_lp = lp_solve(h_cur, timeout=20)
            h_lp = project(h_lp, len(h_cur) / 2.0)
            c5_lp = compute_c5(h_lp)
            if c5_lp < best_c5 - 1e-9:
                best_h, best_c5 = h_lp.copy(), c5_lp
                h_cur = h_lp.copy()
            else:
                break

    # Phase 5: Coordinate refinement
    if rem() > 15:
        best_h, best_c5 = coord_refine(best_h, rem() - 10)

    # Final: ensure feasibility
    n = len(best_h)
    best_h = project(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
