```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(c)

def project_onto_simplex(h, target_sum):
    """Project h onto {0 <= h_i <= 1, sum(h) = target_sum}"""
    h = np.clip(h, 0, 1)
    # Bisection on threshold
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fine-tune
    diff = target_sum - np.sum(projected)
    projected += diff / len(projected)
    projected = np.clip(projected, 0, 1)
    return projected

def gradient_c5(h, dx):
    """Compute gradient of max_p correlate(h, 1-h, p) * dx"""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(c)
    # Find all active (max) lags
    active = np.where(np.abs(c - max_val) < 1e-10)[0]
    
    grad = np.zeros(n)
    for p in active:
        a = p - (n - 1)
        i_min = max(0, -a)
        i_max = min(n, n - a)
        for j in range(n):
            g = 0.0
            if i_min <= j - a < i_max:
                g += (1.0 - h[j - a])
            if i_min <= j < i_max:
                g -= h[j + a]
```
