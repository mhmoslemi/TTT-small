import numpy as np
from scipy.optimize import minimize
from numpy import correlate
import time

_gp = {}

def compute_c5(h, dx):
    c = correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def smax_obj(x, dx, tau):
    c = correlate(x, 1.0 - x, mode='full') * dx
    s = tau * c
    m = np.max(s)
    e = np.exp(s - m)
    return (m + np.log(np.sum(e))) / tau

def smax_grad(x, dx, tau):
    c = correlate(x, 1.0 - x, mode='full') * dx
    s = tau * c
    m = np.max(s)
    e = np.exp(s - m)
    w = e / np.sum(e)
    return corr_grad(x, w, dx)

def corr_grad(h, w, dx):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        wk = w[k] * dx
        if wk == 0.0:
            continue
        a = max(0, lag)
        b = min(n - 1, n - 1 + lag)
        if a <= b:
            i = np.arange(a, b + 1)
            g[i] += wk * (1.0 - h[i - lag])
        a = max(0, -lag)
        b = min(n - 1, n - 1 - lag)
        if a <= b:
            i = np.arange(a, b + 1)
            g[i] -= wk * h[i + lag]
    return g

def _f(x):
    return smax_obj(x, _gp['dx'], _gp['tau'])

def _g(x):
    return smax_grad(x, _gp['dx'], _gp['tau'])

def _c(x):
    return np.sum(x) - _gp['target']

def _cj(x):
    return np.ones(_gp['n'])

def project(h, n):
    target = n * 0.5
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        p = np.clip(h - mid, 0.0, 1.0)
        if np.sum(p) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def snap_01(h, n):
    t = n * 0.5
    sorted_idx = np.argsort(-h)
    out = np.zeros(n)
    rem = t
    for i in sorted_idx:
        if rem >= 1.0:
            out[i] = 1.0
            rem -= 1.0
        elif rem > 0:
            out[i] = rem
            rem = 0.0
    return out

def swap_search(h, dx, n, rng, max_iter=4000):
    best = h.copy()
    best_c5 = compute_c5(best, dx)
    for it in range(max_iter):
        if it % 200 == 0 and compute_c5(best, dx) < best_c5:
            best_c5 = compute_c5(best, dx)
        ones = np.where(best > 0.4)[0]
        zeros = np.where(best < 0.6)[0]
        ones_pure = np.where(best > 0.9)[0]
        zeros_pure = np.where(best < 0.1)[0]
        if len(ones_pure) == 0:
            ones_pure = ones
        if len(zeros_pure) == 0:
            zeros_pure = zeros
        if len(ones_pure) == 0 or len(zeros_pure) == 0:
            break
        i = zeros_pure[rng.randint(len(zeros_pure))]
        j = ones_pure[rng.randint(len(ones_pure))]
        if i == j:
            continue
        oi, oj = best[i].copy(), best[j].copy()
        best[i] = oj
        best[j] = oi
        c5 = compute_c5(best, dx)
        if c5 < best_c5:
            best_c5 = c5
        else:
            best[i] = oi
            best[j] = oj
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h0 = np.zeros(45)
        h0[:22] = 1.0
        h0[22] = 0.5

    n = len(h0)
    dx = 2.0 / n
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    best_n = n

    def try_slsqp(x0, n_pts, dx_pts, tau_list, max_iter, ns=1):
        nonlocal best_h, best_c5, best_n
        for tau in tau_list:
            if time.time() > deadline:
                return
            _gp['dx'] = dx_pts
            _gp['tau'] = tau
            _gp['n'] = n_pts
            _gp['target'] = n_pts * 0.5
            for r in range(ns):
                if time.time() > deadline:
                    return
                if r == 0:
                    x0_r = x0.copy()
                else:
                    x0_r = x0 + rng.randn(n_pts) * 0.08
                    x0_r = project(x0_r, n_pts)
                try:
                    res = minimize(_f, x0_r, jac=_g, method='SLSQP',
                                   bounds=[(0.0, 1.0)] * n_pts,
                                   constraints=[{'type': 'eq', 'fun': _c, 'jac': _cj}],
                                   options={'maxiter': max_iter, 'ftol': 1e-14})
                    if res.x is not None:
                        hn = np.clip(res.x, 0.0, 1.0)
                        hn = project(hn, n_pts)
                        c5 = compute_c5(hn, dx_pts)
                        if c5 < best_c5:
                            best_h, best_c5, best_n = hn, c5, n_pts
                except Exception:
                    pass

    # Phase 1: SLSQP at original n with decreasing tau
    try_slsqp(best_h, n, dx, [400, 200, 100, 50, 30, 20, 10, 5], 250, ns=2)

    # Phase 2: Local swap search
    if time.time() < deadline:
        h_snap = snap_01(best_h, n)
        if compute_c5(h_snap, dx) <= best_c5 + 0.01:
            _, sc5 = swap_search(h_snap, dx, n, rng, 3000)
            if sc5 < best_c5:
                best_h, best_c5 = h_snap, sc5
        h_cont = project(best_h, n)
        hs, cs = swap_search(h_cont, dx, n, rng, 3000)
        if cs < best_c5:
            best_h, best_c5 = hs, cs

    # Phase 3: Projected gradient polish
    if time.time() < deadline:
        h = best_h.copy()
        for tau_pg in [20, 10, 5, 3, 2]:
            for it in range(3000):
                if time.time() > deadline:
                    break
                c = correlate(h, 1 - h, mode='full') * dx
                s = tau_pg * c
                m = np.max(s)
                e = np.exp(s - m)
                w = e / np.sum(e)
                g = corr_grad(h, w, dx)
                for lr in [0.02, 0.01, 0.005]:
                    h_try = project(h - lr * g, n)
                    c5_try = compute_c5(h_try, dx)
                    if c5_try < best_c5:
                        best_h, best_c5 = h_try, c5_try
                    h = h_try
                    break

    # Phase 4: Try different n
    for n_try in [n + 1, n + 2, 2 * n, n * 3 // 2]:
        if time.time() > deadline:
            break
        if n_try < 16 or n_try > 400:
            continue
        dx_t = 2.0 / n_try
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        x_old_ext = np.concatenate([x_old - 2, x_old, x_old + 2])
        h_ext = np.concatenate([best_h, best_h, best_h])
        h_interp = np.interp(x_new, x_old_ext, h_ext)
        h_interp = project(h_interp, n_try)
        try_slsqp(h_interp, n_try, dx_t, [100, 50, 25], 150, ns=1)

    # Final swap search at best_n
    if time.time() < deadline - 5:
        dx_f = 2.0 / best_n
        h_f = snap_01(best_h, best_n)
        if compute_c5(h_f, dx_f) <= best_c5 + 0.005:
            _, sc5 = swap_search(h_f, dx_f, best_n, rng, 5000)
            if sc5 < best_c5:
                best_h, best_c5 = h_f, sc5

    return (best_h, best_c5, best_n)
