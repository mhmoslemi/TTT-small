import numpy as np
from scipy.optimize import minimize
import time

N_POINTS = 95
TARGET_SUM = N_POINTS / 2.0
DX = 2.0 / N_POINTS


def compute_c5(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * DX)


def _grad_shift(h, k):
    n = N_POINTS
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[k:] += (1.0 - h[:n - k])
            g[:n - k] -= h[k:]
    else:
        kk = -k
        if kk < n:
            g[:n - kk] += (1.0 - h[kk:])
            g[kk:] -= h[:n - kk]
    return g


def obj_grad(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * DX
    max_val = float(np.max(vals))
    active = np.where(vals >= max_val - 1e-10)[0]
    grad = np.zeros(N_POINTS)
    for m in active:
        k = int(m - (N_POINTS - 1))
        grad += _grad_shift(h, k)
    grad *= DX / len(active)
    return max_val, grad


def obj_func(h):
    return compute_c5(h)


def obj_jac(h):
    return obj_grad(h)[1]


def _constraint_fun(x):
    return np.sum(x) - TARGET_SUM


def _constraint_jac(x):
    return np.ones(N_POINTS)


def project_sum(h, target):
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)


def _slsqp_opt(h_start, maxiter=300, ftol=1e-12):
    res = minimize(
        func=obj_func,
        x0=h_start,
        jac=obj_jac,
        method='SLSQP',
        bounds=[(0.0, 1.0)] * N_POINTS,
        constraints=[{'type': 'eq', 'fun': _constraint_fun, 'jac': _constraint_jac}],
        options={'maxiter': maxiter, 'ftol': ftol}
    )
    return res


def _local_search(h_best, c5_best, rng, max_iters=3000, time_fn=None):
    step = 0.03
    no_improve = 0
    for _ in range(max_iters):
        if time_fn is not None and time_fn() < 30:
            break
        n_pert = int(rng.integers(1, 6))
        idx = rng.choice(N_POINTS, size=n_pert, replace=False)
        perturb = rng.uniform(-step, step, size=n_pert)
        h_try = h_best.copy()
        h_try[idx] = np.clip(h_try[idx] + perturb, 0.0, 1.0)
        h_try = project_sum(h_try, TARGET_SUM)
        c5_try = compute_c5(h_try)
        if c5_try < c5_best - 1e-13:
            h_best = h_try
            c5_best = c5_try
            no_improve = 0
        else:
            no_improve += 1
        if no_improve % 300 == 299:
            step *= 0.7
            if step < 0.001:
                break
    return h_best, c5_best
