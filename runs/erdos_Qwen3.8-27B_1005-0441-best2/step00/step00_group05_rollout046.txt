def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * (2.0 / n)

    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-15:
        h = h * (0.5 * n) / s
        h = np.clip(h, 0.0, 1.0)
    return h

    """Compute smooth max and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Numerically stable log-sum-exp
    corr_max = corr.max()
    exps = np.exp(alpha * (corr - corr_max))
    smax = (corr_max + np.log(exps.sum()) / alpha) * (2.0 / n)
    
    # Gradient: sum_k exp(alpha*corr_k) / sum_k exp(alpha*corr_k) * grad(corr_k)
    weights = exps / exps.sum()
    grad = np.zeros(n)

            js = np.arange(jlo, jhi)
            grad[js] -= weights[k] * h[js + off]
            grad[js + off] += weights[k] * (1.0 - h[js])
    grad *= (2.0 / n)
    return smax, grad

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")  # length 2n-1
    corr_max = corr.max()

    w = exps / exps.sum()  # weights for each lag
    smax = (corr_max + np.log(exps.sum()) / alpha) * dx
    
    # For each lag k, offset = k - (n-1)
    offsets = np.arange(len(corr)) - (n - 1)  # -(n-1) to (n-1)

    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:

            grad[idx] -= wk * h[idx + off]

import numpy as np
import time

def eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * (2.0 / n)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-15:
        h = h * (0.5 * n) / s
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = corr.max()
    exps = np.exp(alpha * (corr - cmax))
    w = exps / exps.sum()
    smax = (cmax + np.log(exps.sum()) / alpha) * dx
    offsets = np.arange(len(corr)) - (n - 1)
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        off = offsets[k]
        i_lo = max(0, off)
        i_hi = min(n, n + off)
        if i_lo < i_hi:
            idx = np.arange(i_lo, i_hi)
            grad[idx] += wk * (1.0 - h[idx - off])
        j_lo = max(0, -off)
        j_hi = min(n, n - off)
        if j_lo < j_hi:
            idx = np.arange(j_lo, j_hi)
            grad[idx] -= wk * h[idx + off]
    grad *= dx
    return smax, grad

def hard_subgrad(h, n, k_star):
    off = k_star - (n - 1)
    jlo = max(0, -off)
    jhi = min(n, n - off)
    if jlo >= jhi:
        return np.zeros(n)
    js = np.arange(jlo, jhi)
    g = np.zeros(n)
    g[js] -= h[js + off]
    g[js + off] += (1.0 - h[js])
    return g
