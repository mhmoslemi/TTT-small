import numpy as np
import time

def _corr_max(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx

def _subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    m = int(np.argmax(corr_raw))
    shift = m - (n - 1)
    g = np.zeros(n)
    t_lo1 = max(0, shift)
    t_hi1 = min(n, shift + n)
    g[t_lo1:t_hi1] += 1.0 - h[t_lo1 - shift:t_hi1 - shift]
    t_lo2 = max(0, -shift)
    t_hi2 = min(n, n - shift)
    g[t_lo2:t_hi2] -= h[t_lo2 + shift:t_hi2 + shift]
    return g * dx

def _proj(h, target):
    lo = h.min() - 2.0
    hi = 2.0 - h.min()
    for _ in range(80):
        t = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - t, 0.0, 1.0))
        if s > target:
            lo = t
        else:
            hi = t
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _round_binary(h, rng, target):
    n = len(h)
    hb = (h > 0.5).astype(np.float64)
    diff = int(hb.sum()) - target
    if diff > 0:
        idx = np.where(hb == 1.0)[0]
        perm = rng.permutation(idx)
        for i in perm[:diff]:
            hb[i] = 0.0
    elif diff < 0:
        idx = np.where(hb == 0.0)[0]
        perm = rng.permutation(idx)
        for i in perm[:(-diff)]:
            hb[i] = 1.0
    return hb

def _binary_local(h, rng, time_limit):
    n = len(h)
    target = n // 2
    hb = _round_binary(h, rng, target)
    c5 = _corr_max(hb)
    best_hb = hb.copy()
    best_c5 = c5
    t0 = time.time()
    stall = 0
    temp = 0.03
    while time.time() - t0 < time_limit:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or hb[i] == hb[j]:
            continue
        hb[i], hb[j] = hb[j], hb[i]
        new_c5 = _corr_max(hb)
        if new_c5 < c5 or (new_c5 - c5) < temp * rng.exponential():
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_hb = hb.copy()
            stall = 0
        else:
            hb[i], hb[j] = hb[j], hb[i]
            stall += 1
        if stall > 4000:
            temp *= 0.93
            for _ in range(4):
                a, b = rng.randint(n), rng.randint(n)
                if a != b and hb[a] != hb[b]:
                    hb[a], hb[b] = hb[b], hb[a]
            c5 = _corr_max(hb)
            stall = 0
        if temp < 0.0005:
            break
    return best_hb, best_c5

def _subgrad_descent(h, rng, time_limit):
    n = len(h)
    best_h = h.copy()
    best_c5 = _corr_max(h)
    alpha = 0.4
    v = np.zeros(n)
    beta = 0.85
    t0 = time.time()
    while time.time() - t0 < time_limit:
        g = _subgrad(h)
        v = beta * v + g
        h = _proj(h - alpha * v, 0.5 * n)
        c5 = _corr_max(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        alpha *= 0.999
        if alpha < 0.005:
            alpha = 0.005
            beta *= 0.99
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    T0 = time.time()

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        h = np.full(100, 0.5)
    n = len(h)
    best_h = h.copy()
    best_c5 = _corr_max(h)

    # Phase 1: Continuous subgradient descent
    t1 = budget_s * 0.12
    h1, c5_1 = _subgrad_descent(h, rng, t1)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    # Phase 2: Binary local search from best continuous
    elapsed = time.time() - T0
    t2 = budget_s * 0.5 - elapsed
    if t2 > 5:
        hb, c5b = _binary_local(best_h, rng, t2)
        if c5b < best_c5:
            best_c5, best_h = c5b, hb

    # Phase 3: Multi-start binary optimization
    elapsed = time.time() - T0
    remaining = budget_s - 12 - elapsed
    if remaining > 20:
        n_starts = min(int(remaining / 25), 8)
        for s in range(n_starts):
            t_rem = budget_s - 10 - (time.time() - T0)
            if t_rem < 12:
                break
            h_rand = np.zeros(n)
            perm = rng.permutation(n)
            h_rand[perm[:n // 2]] = 1.0
            hb2, c5r = _binary_local(h_rand, rng, t_rem * 0.85)
            if c5r < best_c5:
                best_c5, best_h = c5r, hb2

    # Final refinement: a few more subgradient steps on the best solution
    elapsed = time.time() - T0
    if elapsed < budget_s - 15 and 0 < best_h[0] < 1:
        hf = best_h.copy()
        c5f = _subgrad_descent(hf, rng, min(20, budget_s - 12 - elapsed))
        if c5f < best_c5:
            best_c5, best_h = c5f, hf

    best_c5 = _corr_max(best_h)
    return best_h, best_c5, n
