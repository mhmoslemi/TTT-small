def compute_c5(h):
    """Compute C5 value for given h."""
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def compute_all_overlaps(h):
    """Compute all overlap values."""
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return corr

def compute_gradient(h, L):
    """Compute gradient of overlap at lag L w.r.t. h."""
    n = len(h)
    # L is in range [-(n-1), n-1]
    grad = np.zeros(n)
    # h[i] * (1-h)[i-L]: derivative w.r.t. h[j]
    # From h[j] term: (1-h)[j-L] if j-L in [0,n-1]
    # From (1-h) term: -h[j+L] if j+L in [0,n-1]
    for j in range(n):
        val = 0.0
        if 0 <= j - L < n:
            val += (1.0 - h[j - L])
        if 0 <= j + L < n:
            val -= h[j + L]
        grad[j] = val
    return grad
