import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum}"""
    n = len(h)
    # Bisection for lambda
    lo, hi = h.min() - target_sum/n - 1, h.max() + target_sum/n + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        clamped = np.clip(h - mid, 0, 1)
        s = clamped.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    clamped = np.clip(h - (lo+hi)/2, 0, 1)
    # Fine-tune
    diff = target_sum - clamped.sum()
    # Distribute difference to elements not at boundary
    interior = (clamped > 1e-10) & (clamped < 1 - 1e-10)
    if interior.sum() > 0:
        adj = diff / interior.sum()
        clamped[interior] += adj
        clamped = np.clip(clamped, 0, 1)
    return clamped

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0

        mid = (lo + hi) / 2
        clamped = np.clip(h - mid, 0, 1)

    return c.max() * dx, c

    g = np.zeros(n)
    for j in range(n):
        # (1-h[j-L]) if 0<=j-L<n
        if 0 <= j-L < n:
            g[j] += (1 - h[j-L])
        # -h[j+L] if 0<=j+L<n
        if 0 <= j+L < n:
            g[j] -= h[j+L]
    return g
