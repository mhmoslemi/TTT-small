import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(40):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        if not m.any():
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def optimize_fixed_n(h, n, dx, target, rng, t_deadline, init_c5):
