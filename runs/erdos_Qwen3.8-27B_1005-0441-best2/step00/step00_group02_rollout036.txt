def compute_corr_vals(h):
    """Return all correlation values c_k."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def compute_grad(h, alpha):
    """Gradient of smooth max of correlation values."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    
    # Softmax weights
    max_c = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_c))
    w = exp_vals / np.sum(exp_vals)
    
    # Gradient
    grad = np.zeros(n)
    for j in range(n):
        for k in range(2*n - 1):
            p = k - (n - 1)
            # h[j] in first position: i = j - p, valid if 0 <= j-p < n
            val1 = 0.0
            if 0 <= j - p < n:
                val1 = 1.0 - h[j - p]
            # h[j] in second position: i = j, j+p valid if 0 <= j+p < n
            val2 = 0.0
            if 0 <= j + p < n:
                val2 = h[j + p]
            grad[j] += w[k] * dx * (val1 - val2)
    
    return grad

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    obj = m + np.log(s) / alpha
    
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        p = k - (n - 1)
        wk_dx = w[k] * dx
        idx1 = j - p
        v1 = np.zeros(n)
        valid1 = (idx1 >= 0) & (idx1 < n)
        v1[valid1] = 1.0 - h[idx1[valid1]]
        idx2 = j + p
        v2 = np.zeros(n)
        valid2 = (idx2 >= 0) & (idx2 < n)
        v2[valid2] = h[idx2[valid2]]
        grad += wk_dx * (v1 - v2)

    """Projected gradient descent with smooth max."""
    n = len(h)
    h = h.copy()
    best_local = h.copy()
    best_c5_local = compute_c5(best_local)
    
    for alpha in alphas:
        lr = 5.0 / (alpha * n)
        stale = 0
        for it in range(max_iters_per_alpha):
            if time.time() > t_deadline:
                return best_local, best_c5_local
            obj, grad = smooth_obj_and_grad(h, alpha)
            # Backtracking line search
            for backtrack in range(10):
                step = lr * (0.5 ** backtrack)
                h_trial = h - step * grad

                stale = 0

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    s = np.sum(exp_vals)
    w = exp_vals / s
    obj = m + np.log(s) / alpha
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        p = k - (n - 1)
        wk_dx = w[k] * dx
        idx1 = j - p
        v1 = np.zeros(n)
        valid1 = (idx1 >= 0) & (idx1 < n)
        v1[valid1] = 1.0 - h[idx1[valid1]]
        idx2 = j + p
        v2 = np.zeros(n)
        valid2 = (idx2 >= 0) & (idx2 < n)
        v2[valid2] = h[idx2[valid2]]
        grad += wk_dx * (v1 - v2)
    return obj, grad
