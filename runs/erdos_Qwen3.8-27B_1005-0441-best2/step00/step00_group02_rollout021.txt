                       options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.x is not None:
            h_new = _normalize(res.x, ts)
            c5_new = _c5(h_new, n)
            c5_old = _c5(h, n)
            if c5_new < c5_old:
                return h_new, c5_new
    except Exception:
        pass
    return h, _c5(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(45, 0.5)
    n = len(h_init)
    ts = n / 2.0
    h_init = _normalize(h_init, ts)

    best_h = h_init.copy()
    best_c5 = _c5(h_init, n)

    # Phase 1: subgradient on initial
    h1, c5_1 = _subgrad_iter(h_init, n, lr=0.08, alpha_w=20.0, n_iter=5000, t_end=t_end)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # SLSQP polish
    hp, c5p = _slsqp_polish(best_h, n, max_iter=150)
    if c5p < best_c5:
        best_h, best_c5 = hp, c5p

    # Phase 2: multiple restarts
    for restart in range(12):
        if time.time() > t_end - 60:
            break
        amp = 0.03 + 0.02 * (restart % 4)
        pert = rng.normal(0, amp, n)
        h_r = _normalize(best_h + pert, ts)

        hr, cr = _subgrad_iter(h_r, n, lr=0.04, alpha_w=30.0, n_iter=3000, t_end=t_end)
        if cr < best_c5:
            pass  # hr and cr are already best
        h_r = hr if cr < best_c5 else h_r

        hr3, cr3 = _slsqp_polish(h_r, n, max_iter=120)
        if cr3 < best_c5:
            best_h, best_c5 = hr3, cr3

    # Phase 3: fine-tune on best
    if time.time() < t_end - 30:
        h3, c5_3 = _subgrad_iter(best_h, n, lr=0.01, alpha_w=80.0, n_iter=5000, t_end=t_end)
        if c5_3 < best_c5:
            best_h, best_c5 = h3, c5_3
        h3p, c5p3 = _slsqp_polish(best_h, n, max_iter=200)
        if c5p3 < best_c5:
            best_h, best_c5 = h3p, c5p3

    # Final normalization and evaluation
    best_h = _normalize(best_h, ts)
    best_c5 = _c5(best_h, n)
    return best_h, best_c5, n
