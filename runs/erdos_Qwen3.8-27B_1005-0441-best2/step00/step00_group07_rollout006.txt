import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nfree = np.sum(free)
        if nfree == 0:
            # All at bounds, distribute to nearest
            h = h.copy()
            if diff > 0:
                # Need to increase; find values closest to 1
                idx = np.argsort(h)[-min(int(abs(diff))+1, n):]
                for i in idx:
                    step = min(1.0 - h[i], diff)
                    h[i] += step
                    diff -= step
                    if abs(diff) < 1e-14:
                        break
            else:
                idx = np.argsort(h)[:min(int(abs(diff))+1, n)]
                for i in idx:
                    step = min(h[i], -diff)
                    h[i] -= step
                    diff += step
                    if abs(diff) < 1e-14:
                        break
            break
        h[free] += diff / nfree
        h = np.clip(h, 0.0, 1.0)
    return h

    t0 = time.time()
    rng = np.random.RandomState(seed)

    if budget_s > 400:
