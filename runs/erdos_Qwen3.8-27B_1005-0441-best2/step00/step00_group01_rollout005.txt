    dx = 2.0 / n

def evaluate_c5(h):

    dx = 2.0 / n
    overlap = np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        val = np.clip(h - mid, 0.0, 1.0)
        if np.sum(val) > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def overlap_grad(h, n, lag):
    dx = 2.0 / n
    g = np.zeros(n)
    if lag == 0:
        g[:] = (1.0 - 2.0 * h)
    elif 0 < lag < n:
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    elif -n < lag < 0:
        p = -lag
        g[:n - p] += (1.0 - h[p:])
        g[p:] -= h[:n - p]
    return g * dx

                lr *= 0.85
            if stale > 1500:
                h = project_feasible(best_h + np.random.randn(n) * 0.02, n)
                c5 = evaluate_c5(h)
                lr = 0.1
                stale = 0
                tau = 0.008
    return best_h, best_c5

def scipy_local(h0, n, deadline):
    from scipy.optimize import minimize, NonlinearConstraint
    dx = 2.0 / n
    m_tot = 2 * n - 1

    def obj(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def make_constraint(k_idx):
        lag = k_idx - (n - 1)
        def fun(x):
            h = x[:n]
            ov = np.dot(_overlap_vec(h, n, lag), np.ones(1)) * dx
            return x[-1] - ov
        def jac(x):
            h = x[:n]
            g = overlap_grad(h, n, lag)
            out = np.zeros(n + 1)
            out[:n] = -g
            out[-1] = 1.0
            return out
        return fun, jac

    def _overlap_vec(h, n, lag):
        # Compute overlap at single lag
        if lag == 0:
            return np.sum(h * (1.0 - h))
        elif 0 < lag < n:
            i = np.arange(n - lag)
            return np.sum(h[i + lag] * (1.0 - h[i]))
        elif -n < lag < 0:
            p = -lag
            i = np.arange(n - p)
            return np.sum(h[i + p] * (1.0 - h[i]))
        return 0.0

    c5_now = evaluate_c5(h0)
    c5_now = max(c5_now, 0.01)
    x0 = np.concatenate([h0, [c5_now + 0.05]])

    # Use only top constraints for speed
    ov_all = np.correlate(h0, 1.0 - h0, mode="full") * dx
    top_idx = np.argsort(ov_all)[-20:]

    cons = []
    for ki in top_idx:
        fun, jac = make_constraint(int(ki))
        cons.append(NonlinearConstraint(fun, 0, np.inf, jac=jac))

    eq = NonlinearConstraint(lambda x: x[:n].sum(), n / 2.0, n / 2.0,
                             jac=lambda x: np.concatenate([np.ones(n), [0.0]]))
    cons.append(eq)

    bounds = [(0, 1)] * n + [(0, 1.0)]
    res = minimize(obj, x0, jac=obj_jac, bounds=bounds, constraints=cons,
                   method='SLSQP', options={'maxiter': 500, 'ftol': 1e-10})
    if time.time() > deadline:
        return h0, c5_now
    if res.success and res.fun < c5_now:
        h_new = project_feasible(res.x[:n], n)
        c5_new = evaluate_c5(h_new)
        if c5_new < c5_now:
            return h_new, c5_new
    return h0, c5_now

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s * 0.93

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = None

    best_h = None
    best_c5 = float('inf')

    def update_best(h, c5):
        nonlocal best_h, best_c5
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 1: Projected gradient from initial
    if h_init is not None:
        bh, bc = pg_optimize(h_init, len(h_init), deadline)
        update_best(bh, bc)
        # Refine with scipy
        if len(h_init) <= 60 and time.time() < deadline:
            bh2, bc2 = scipy_local(bh, len(bh), deadline)
            update_best(bh2, bc2)
            if bc2 < best_c5:
                bh3, bc3 = pg_optimize(bh2, len(bh2), deadline)
                update_best(bh3, bc3)

    # Phase 2: Try other starting points
    np.random.seed(seed + 100)
    for n_try in [47, 94, 188]:
        if time.time() > deadline:
            break
        # Random perturbation
        h_r = project_feasible(np.clip(0.5 + 0.4 * np.random.randn(n_try), 0, 1), n_try)
        bh, bc = pg_optimize(h_r, n_try, deadline)
        update_best(bh, bc)

        # Structured: two high blocks separated by low
        h_s = np.zeros(n_try)
        third = n_try // 3
        h_s[:third] = 1.2
        h_s[third:2*third] = 0.2
        h_s[2*third:] = 1.2
        h_s = project_feasible(h_s, n_try)
        bh, bc = pg_optimize(h_s, n_try, deadline)
        update_best(bh, bc)

    if best_h is None:
        n = 47
        best_h = project_feasible(np.ones(n) * 0.5, n)
        best_c5 = evaluate_c5(best_h)

    # Final verification
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
