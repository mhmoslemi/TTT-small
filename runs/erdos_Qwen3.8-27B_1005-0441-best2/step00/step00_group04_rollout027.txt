import numpy as np

def project_to_feasible(h, n):
    """Project h onto {0≤h≤1, sum=h/2}"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Find non-bound elements
        if diff > 0:
            # Need to increase; can only increase where h < 1
            mask = h < 1.0 - 1e-10
            if not mask.any():
                break
            adjust = diff / mask.sum()
            h[mask] = np.minimum(h[mask] + adjust, 1.0)
        else:
            # Need to decrease; can only decrease where h > 0
            mask = h > 1e-10
            if not mask.any():
                break
            adjust = diff / mask.sum()
            h[mask] = np.maximum(h[mask] + adjust, 0.0)
    return h

def compute_c5(h):

    dx = 2.0 / n
    overlap = np.correlate(h, 1-h, mode="full") * dx
    return np.max(overlap), overlap

def subgradient(h, k_star, n):
    """Compute subgradient of c[k_star] with respect to h"""
    s = k_star - (n - 1)  # the shift
    grad = np.zeros(n)
    for i in range(n):
        # Contribution from h[n+s] term: (1-h[i-s]) if 0<=i-s<n
        if 0 <= i - s < n:
            grad[i] += (1 - h[i - s])
        # Contribution from (1-h[n]) term: -h[i+s] if 0<=i+s<n
        if 0 <= i + s < n:
            grad[i] -= h[i + s]
    return grad
