    grad = np.zeros(n)
    # l ranges from -(n-1) to n-1, stored in weights as index l+(n-1)

            if cnt == 0:
                break
            h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)

def num_gradient(h, weights, dx, eps=1e-6):
    n = len(h)
    grad = np.zeros(n)
    for j in range(n):

        hp[j] = min(1.0, h[j] + eps)
        hm = h.copy()

        fp = np.dot(weights, compute_overlaps(hp, dx))
        fm = np.dot(weights, compute_overlaps(hm, dx))
        grad[j] = (fp - fm) / (hp[j] - hm[j])
    return grad

        h = project_h(h - lr * grad, n)

import numpy as np
import time


def compute_overlaps(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx


def compute_c5(h, dx):
    return float(np.max(compute_overlaps(h, dx)))


def project_h(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(5):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    return h


def num_gradient(h, weights, dx, eps=1e-5):
    n = len(h)
    grad = np.zeros(n)
    for j in range(n):
        hp = h.copy()
        hp[j] = min(1.0, h[j] + eps)
        hm = h.copy()
        hm[j] = max(0.0, h[j] - eps)
        denom = hp[j] - hm[j]
        if denom < 1e-12:
            grad[j] = 0.0
            continue
        fp = np.dot(weights, compute_overlaps(hp, dx))
        fm = np.dot(weights, compute_overlaps(hm, dx))
        grad[j] = (fp - fm) / denom
    return grad


def optimize_starts(h_init, n, dx, deadline, rng):
