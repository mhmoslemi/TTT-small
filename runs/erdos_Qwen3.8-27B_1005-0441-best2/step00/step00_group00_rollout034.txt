import numpy as np
from scipy.optimize import minimize as scipy_minimize
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_sum(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(h - lam, 0.0, 1.0)


def smooth_obj(x, dx, target_sum, alpha):
    h = project_sum(x, target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    if alpha <= 1.0:
        return float(np.max(corr))
    shifted = alpha * corr
    shifted = shifted - np.max(shifted)
    return (float(np.max(shifted)) + np.log(np.sum(np.exp(shifted)))) / alpha

    vel = np.zeros((n_particles, n))
    pb = np.zeros((n_particles, n))

    t_now = time.time()
    t_pso2_end = t_now + min(budget_s * 0.1, t_start + budget_s * 0.8 - t_now)
    if t_pso2_end > t_now + 30:
        gb2, gb_c5_2 = pso_run(best_h, dx, target_sum, n, 60, t_now, min(t_pso2_end, t_start + budget_s * 0.8), seed + 1000)
        if gb_c5_2 < best_c5:
            best_c5 = gb_c5_2
            best_h = gb2.copy()

    # Phase 2: Local refinement (remaining budget)
    t_ref_end = t_start + budget_s - 5.0
    if time.time() < t_ref_end - 20:
        best_h, best_c5 = local_refine(best_h, dx, target_sum, n, t_start, t_ref_end, seed + 777)

    # Final validation
    best_h = project_sum(best_h, target_sum)
    final_c5 = compute_c5(best_h, dx)

    return (best_h, final_c5, n)
