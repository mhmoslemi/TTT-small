            if c5r < best_c5:
                best_h = hr
                best_c5 = c5r

    # Final exact computation
    best_c5 = compute_c5(best_h, dx)

    # Validate sum constraint
    s = np.sum(best_h)
    if abs(s - n * 0.5) > 1e-6:
        best_h = np.clip(best_h + (n * 0.5 - s) / n, 0.0, 1.0)
        best_c5 = compute_c5(best_h, dx)

    # Try external evaluator
    try:
        ec = evaluate_erdos_solution(best_h)
        if ec is not None and np.isfinite(ec):
            best_c5 = float(ec)
    except Exception:
        pass

    return (best_h, best_c5, n)
