    return np.max(corr) * dx, corr * dx

            lo = mid
        else:
            hi = mid

    total_ops = 1.0  # for step size normalization

        r = corr * (2.0 / n)
        r_shifted = r - r.max()
        weights = np.exp(beta * r_shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        grad = np.zeros(n)

            js2 = max(0, -k)

                grad[js2:je2] -= weights[m] * h[js2 + k:je2 + k]

        step = 1.0 / max(gnorm, 0.01)
