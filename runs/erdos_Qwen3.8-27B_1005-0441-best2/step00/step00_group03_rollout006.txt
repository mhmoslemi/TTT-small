def project_feasible(h):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0, 1)
    s = h.sum()
    if s > 0 and abs(s - target) > 1e-10:
        # Simple rescaling
        h = h * (target / s)
        h = np.clip(h, 0, 1)
        # If clipping changed the sum, do iterative adjustment
        for _ in range(100):
            s = h.sum()
            if abs(s - target) < 1e-10:
                break
            if s > target:
                # Reduce elements with h close to 1
                mask = h > 0.5
                if mask.any():
                    h[mask] -= (s - target) / mask.sum()
                else:
                    h -= (s - target) / n
            else:
                mask = h < 0.5
                if mask.any():
                    h[mask] += (target - s) / mask.sum()
                else:
                    h += (target - s) / n
            h = np.clip(h, 0, 1)
    return h

    """Compute subgradient of max_k corr[k] with respect to h."""
    n = len(h)
    max_val = np.max(corr)
    # Find all indices achieving (near) the max
    threshold = max_val - 1e-10
    max_indices = np.where(corr >= threshold)[0]
    
    grad = np.zeros(n)
    for k_idx in max_indices:
        lag = k_idx - (n - 1)
        g = np.zeros(n)
        for i in range(n):
            if 0 <= i - lag < n:
                g[i] += (1 - h[i - lag])
            if 0 <= i + lag < n:
                g[i] -= h[i + lag]
        grad += g
    grad /= len(max_indices)
    return grad

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def grad_for_lag_vectorized(h, L):
    n = len(h)
    g = np.zeros(n)
    om = 1.0 - h
    if L >= 0:
        if L < n:
            g[L:] += om[:n-L]
            g[:n-L] -= h[L:]
    else:
        La = -L
        if La < n:
            g[:n-La] += om[La:]
            g[La:] -= h[:n-La]
    return g

def smooth_max_gradient(h, lam=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Log-sum-exp for numerical stability
    mc = np.max(corr)
    exps = np.exp(lam * dx * (corr - mc))
    weights = exps / exps.sum()
    
    grad = np.zeros(n)
    # Only compute gradient for lags with significant weight
    threshold = 1e-6
    significant = weights > threshold
    for k_idx in np.where(significant)[0]:
        lag = k_idx - (n - 1)
        grad += weights[k_idx] * grad_for_lag_vectorized(h, lag)
    return grad * dx, np.max(corr) * dx

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def optimize(h_init, budget_s, start_time):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Log-sum-exp parameter

    lr = 0.01
    patience = 0
    max_patience = 50

            best_h = h.copy()
            patience = 0

        h = project_feasible(h_new)

    n = len(h)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    zeta = lam * dx * (corr - mc)
    exp_zeta = np.exp(zeta)
    Z = exp_zeta.sum()

    obj = mc * dx + np.log(Z) / lam
    grad = np.zeros(n)

        w = weights[m]
        if w > 1e-10:
            lo = max(0, m - n + 1)
            hi = min(n - 1, m)
            if lo <= hi:
                idx = np.arange(lo, hi + 1)
                grad[idx] += w * (1.0 - 2.0 * h[m - idx])
    grad *= dx
    return obj, grad, mc * dx
