import time

def compute_c5(h):
    m = len(h)
    dx = 2.0 / m
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr * dx)

def compute_overlaps(h):
    m = len(h)
    dx = 2.0 / m
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def grad_overlap(h, lag):
    """Gradient of overlap at given lag (actual shift, can be negative)."""
    m = len(h)
    grad = np.zeros(m)
    d = abs(lag)
    if d >= m:
        return grad
    if lag >= 0:
        # O(d) = sum_{n=0}^{m-1-d} h[n+d]*(1-h[n])

            grad[n + d] += (1.0 - h[n])
            grad[n] += (-h[n + d])

            grad[n] += (1.0 - h[n + d])
            grad[n + d] += (-h[n])

    target = n_points / 2.0
    current = h.sum()
    if abs(current - target) > 1e-10:
        # Proportional adjustment
        h = h + (target - current) / n_points
        h = np.clip(h, 0.0, 1.0)
        # If still not exact, adjust
        diff = target - h.sum()
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
        elif diff < 0:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

    zeros = set(np.where(h == 0)[0].tolist())

def subgradient_optimize(h, n_iters, rng, time_limit, t0):
    n = len(h)
    h = h.copy()
    momentum = np.zeros(n)
    lr = 0.003
    mu = 0.85

    best_h = h.copy()
    no_improve = 0
    
    for it in range(n_iters):
        if time.time() - t0 > time_limit:
            break
        
        overlaps = compute_overlaps(h)
        c5 = np.max(overlaps)
        
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 3000:
            # Restart with smaller lr
            momentum *= 0.5
            lr *= 0.5
            no_improve = 0
            if lr < 1e-6:
                break

        lag = d_star_idx - (n - 1)
        grad = grad_overlap(h, lag)
        
        momentum = mu * momentum - lr * grad
        h = h + momentum
        h = enforce_constraints(h, n)
        
        if it % 500 == 0:
            lr *= 0.92
    
    return best_h, best_c5
