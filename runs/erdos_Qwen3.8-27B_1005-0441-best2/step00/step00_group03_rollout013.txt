        rng2 = np.random.RandomState(seed + 100)
        h2, e2 = sa_optimize(best_h, remain, rng2, T0=0.004, cool=0.99995)
        if e2 < best_e:
            best_e = e2
            best_h = h2

    # Polish: very fine local search
    elapsed = time.time() - t_start
    remain = budget_s - elapsed - 2
    if remain > 5:
        rng3 = np.random.RandomState(seed + 200)
        h3, e3 = sa_optimize(best_h, remain, rng3, T0=0.001, cool=0.99998)
        if e3 < best_e:
            best_e = e3
            best_h = h3

    # Final feasibility check
    n = len(best_h)
    best_h = make_feasible(best_h, n / 2.0)
    try:
        c5 = evaluate_erdos_solution(best_h)
    except Exception:
        c5 = fast_c5(best_h)

    return (best_h, c5, n)
