import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return (1.0/alpha) * (m + np.log(np.sum(np.exp(alpha * (corr - m)))))

def smooth_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    ev = np.exp(alpha * (corr - m))
    w = ev / np.sum(ev)
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        shift = k - (n - 1)
        for j in range(n):
            g = 0.0
            js = j - shift
            if 0 <= js < n:
                g += 1.0 - h[js]
            jps = j + shift
            if 0 <= jps < n:
                g -= h[jps]
            grad[j] += w[k] * dx * g
    return smooth_obj(h, alpha), grad

def project_feasible(h, n):
    """Project h onto [0,1]^n with sum(h) = n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Simple: if sum is off, scale or adjust
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    # Scale
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    # Fine-tune
    diff = target - np.sum(h)
    if diff > 0:
        # Need to add diff total; spread over elements with room to grow
        order = np.argsort(h)  # ascending
        for idx in order:
            room = 1.0 - h[idx]
            add = min(room, diff)
            h[idx] += add
            diff -= add
            if diff < 1e-12:
                break
    else:
        # Need to remove |diff| total
        order = np.argsort(h)[::-1]  # descending
        for idx in order:
            rem = min(h[idx], -diff)
            h[idx] -= rem
            diff += rem
            if diff > -1e-12:
                break
    return h
