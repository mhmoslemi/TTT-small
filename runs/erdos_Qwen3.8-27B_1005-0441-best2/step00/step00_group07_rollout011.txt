def constraint_func(h):
    return np.sum(h) - _N / 2.0

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(3000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_old, n_new):
    t_old = np.arange(n_old) * 2.0 / n_old
    t_new = np.arange(n_new) * 2.0 / n_new
    return np.interp(t_new, t_old, h)

def grad_for_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        grad[:n - lag] += (1.0 - h[lag:])
        if lag > 0:
            grad[lag:] -= h[:n - lag]
    else:
        la = -lag
        grad[:n - la] += (1.0 - h[la:])
        if la > 0:
            grad[la:] -= h[:n - la]
    return grad

def subgradient_polish(h, n, steps=300, lr_init=0.008):
    h = h.copy()
    dx = 2.0 / n
    for s in range(steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_max = int(np.argmax(corr))
        lag = k_max - (n - 1)
        lr = lr_init / (np.sqrt(s + 1))
        grad = grad_for_lag(h, lag)
        h = h - lr * grad * dx
        h = project_feasible(h, n)
    return h

def coordinate_polish(h, n, n_sweeps=5):
    h = h.copy()
    dx = 2.0 / n
    for _ in range(n_sweeps):
        improved = False
        for j in range(n):
            best_val = h[j]
            best_c5 = compute_c5(h)
            for delta in [-0.1, -0.05, -0.02, 0.02, 0.05, 0.1]:
                cand = h.copy()
                cand[j] = np.clip(cand[j] + delta, 0.0, 1.0)
                cand = project_feasible(cand, n)
                val = compute_c5(cand)
                if val < best_c5:
                    best_c5 = val
                    best_val = cand[j]
            if best_val != h[j]:
                h[j] = best_val
                h = project_feasible(h, n)
                improved = True
        if not improved:
            break
    return h

def slsqp_optimize(h0, n, max_iter):
    global _N
    _N = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_func}]
    res = minimize(obj_func, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14})
    return project_feasible(res.x, n)

def run(seed=42, budget_s=1100, **kwargs):
    global _N
    start = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.ones(80) * 0.5
    n_best = len(h_best)
    h_best = project_feasible(h_best, n_best)
    c5_best = compute_c5(h_best)

    n_list = sorted(set([n_best]))
    for nr in [120, 160, 200, 240, 320, 400]:
        if nr > n_best:
            n_list.append(nr)

    for n in n_list:
        if time.time() - start > budget_s * 0.82:
            break

        if n == n_best:
            h_start = h_best.copy()
        else:
            h_start = upsample(h_best, n_best, n)
            h_start = project_feasible(h_start, n)

        n_restarts = 4 if n <= 160 else (3 if n <= 320 else 2)
        max_iter = 2000 if n <= 160 else (1200 if n <= 320 else 600)

        for r in range(n_restarts):
            if time.time() - start > budget_s * 0.88:
                break

            if r == 0:
                h0 = h_start.copy()
            else:
                scale = 0.015 * (r + 1)
                h0 = project_feasible(h_start + rng.randn(n) * scale, n)

            try:
                h_cand = slsqp_optimize(h0, n, max_iter)
            except Exception:
                continue

            c5_cand = compute_c5(h_cand)

            if c5_cand < c5_best:
                h_cand = subgradient_polish(h_cand, n, steps=200, lr_init=0.005)
                c5_cand2 = compute_c5(h_cand)
                if c5_cand2 < c5_cand:
                    h_cand, c5_cand = h_cand, c5_cand2
                h_cand = coordinate_polish(h_cand, n, n_sweeps=3)
                c5_cand3 = compute_c5(h_cand)
                if c5_cand3 < c5_cand:
                    h_cand, c5_cand = h_cand, c5_cand3

            if c5_cand < c5_best:
                h_best = h_cand
                n_best = n
                c5_best = c5_cand
                h_start = h_best.copy()

    # Final stretch: if time remains, do extra polishing at best n
    if time.time() - start < budget_s * 0.92:
        for _ in range(3):
            if time.time() - start > budget_s * 0.95:
                break
            h_pol = subgradient_polish(h_best, n_best, steps=300, lr_init=0.003)
            cp = compute_c5(h_pol)
            if cp < c5_best:
                h_best, c5_best = h_pol, cp
            h_pol2 = coordinate_polish(h_best, n_best, n_sweeps=2)
            cp2 = compute_c5(h_pol2)
            if cp2 < c5_best:
                h_best, c5_best = h_pol2, cp2

    h_best = project_feasible(h_best, n_best)
    c5_best = compute_c5(h_best)
    return (h_best, c5_best, n_best)
