def smooth_max_overlaps(h, temperature=50.0):
    """Smooth approximation of max of overlaps using log-sum-exp."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    overlaps = corr * dx
    # log-sum-exp for numerical stability
    max_o = np.max(overlaps)
    if temperature < 1e-10:
        return max_o
    val = max_o + (1.0/temperature) * np.log(np.sum(np.exp(temperature * (overlaps - max_o))))
    return val

def smooth_max_grad(h, temperature=50.0):
    """Gradient of smooth max of overlaps."""
    dx = 2.0 / len(h)
    n = len(h)
    one_minus_h = 1 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    overlaps = corr * dx
    
    max_o = np.max(overlaps)
    weights = np.exp(temperature * (overlaps - max_o))
    weights /= np.sum(weights)
    
    # Gradient is weighted average of individual gradients
    grad = np.zeros(n)

        for j in range(n):
            val = 0.0
            if 0 <= j + k < n:
                val += 1 - h[j + k]
            if 0 <= j - k < n:
                val -= h[j - k]
            grad[j] += w * val * dx
    return grad

import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _grad_one_shift(h, k):
    """Gradient of overlap at shift k (in units of index)."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for j in range(n):
        v = 0.0
        jp = j + k
        jm = j - k
        if 0 <= jp < n:
            v += 1.0 - h[jp]
        if 0 <= jm < n:
            v -= h[jm]
        g[j] = v * dx
    return g

def soft_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    ov = corr * dx
    m = np.max(ov)
    if temp < 0.5:
        return m
    return m + np.log(np.sum(np.exp(temp * (ov - m)))) / temp

def soft_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    ov = corr * dx
    m = np.max(ov)
    if temp < 0.5:
        idx = np.argmax(ov)
        return _grad_one_shift(h, idx - (n - 1))
    w = np.exp(temp * (ov - m))
    sw = np.sum(w)
    g = np.zeros(n)
    for idx in range(len(ov)):
        wi = w[idx] / sw
        if wi < 1e-12:
            continue
        g += wi * _grad_one_shift(h, idx - (n - 1))
    return g

    return np.sum(h) - n / 2.0

    return np.ones(n)

        else:

    """Gradient of overlap at index-shift k."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for j in range(n):
        v = 0.0
        jp = j + k
        jm = j - k
        if 0 <= jp < n:
            v += 1.0 - h[jp]
        if 0 <= jm < n:
            v -= h[jm]
        g[j] = v * dx
    return g
