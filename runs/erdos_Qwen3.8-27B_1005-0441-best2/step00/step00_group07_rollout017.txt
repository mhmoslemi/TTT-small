from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt == 0:
            mask = (h > 1e-6) & (h < 1.0 - 1e-6)
            cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_eq(x):
    return float(np.sum(x) - len(x) / 2.0)

def sum_eq_jac(x):
    return np.ones(len(x))

def obj_func(x):
    return compute_c5(x)

def interp_h(h_src, n_dst):
    ns = len(h_src)
    if ns == n_dst:
        return h_src.copy()
    xo = np.linspace(0.0, 1.0, ns)
    xn = np.linspace(0.0, 1.0, n_dst)
    hd = np.interp(xn, xo, h_src)
    return project_feasible(hd, n_dst)

def opt_slsqp(h0, maxiter=1000, ftol=1e-10):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac}]
    try:
        res = minimize(obj_func, h0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_out = project_feasible(res.x, n)
    except Exception:
        h_out = project_feasible(h0, n)
    return h_out, compute_c5(h_out)

def to_binary(h, n):

def binary_local_search(h_bin, max_rounds=30):

        best_c5 = c5_curr
        best_swap = None

            for j in zeros:
                h_bin[i] = 0.0

                h_bin[j] = 0.0
                if c5_new < best_c5 - 1e-14:

            break
        h_bin[best_swap[0]] = 0.0
        h_bin[best_swap[1]] = 1.0
        c5_curr = best_c5
    return h_bin, c5_curr
