def evaluate_c5(h, dx=None):

    return np.correlate(h, 1.0-h, mode="full") * dx

def make_binary(n, h_start=None, rng=None):
    if h_start is not None:

        h[idx] = 1.0
    # Ensure exact count

    count = int(h.sum())

        h[np.sort(ones[:count-target])] = 0
    elif count < target:
        zeros = np.where(h == 0)[0]
        h[np.sort(zeros[:target-count])] = 1
    return h

def swap_local_search(h, n, dx, max_iters=100000, rng=None, time_budget=60):

            break
        i = list(ones)[rng.randint(len(ones))]
        j = list(zeros)[rng.randint(len(zeros))]

        new_c5 = evaluate_c5(h, dx)
