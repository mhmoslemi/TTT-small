    n = len(h)
    grad = np.zeros(n)
    js = max(0, d)
    je = min(n - 1, n - 1 + d)
    if js <= je:
        jv = np.arange(js, je + 1)
        grad[jv] += (1 - h[jv - d])

import numpy as np
import time

def project_onto_box_sum(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    # Bisection on shift
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def c5_value(h):
    """Compute C5 = max(corr) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def grad_c_d(h, d):
    """Gradient of c(d) = sum_i h[i+d]*(1-h[i]) w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    # (1-h[j-d]) for j where 0<=j<n and 0<=j-d<n
    js = max(0, d)
    je = min(n - 1, n - 1 + d)
    if js <= je:
        jv = np.arange(js, je + 1)
        g[jv] += 1.0 - h[jv - d]
    # -h[j+d] for j where 0<=j<n and 0<=j+d<n
    js2 = max(0, -d)
    je2 = min(n - 1, n - 1 - d)
    if js2 <= je2:
        jv2 = np.arange(js2, je2 + 1)
        g[jv2] -= h[jv2 + d]
    return g

def smooth_grad(h, T):
    """Gradient of (1/T)log(sum exp(T*c_d))"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = corr.max()
    e = np.exp(T * (corr - mx))
    w = e / e.sum()
    g = np.zeros(n)
    # Only use significant weights
    thresh = max(w.max() * 0.001, 1e-10)
    sig = np.where(w >= thresh)[0]
    for k in sig:
        d = int(k - (n - 1))
        g += w[k] * grad_c_d(h, d)
    return g

def interpolate_h(h, new_n):
    """Interpolate h from old_n to new_n points"""
    old_n = len(h)
    x_old = np.linspace(0.0, 2.0, old_n)
    x_new = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(x_new, x_old, h, left=0.0, right=0.0)
    return project_onto_box_sum(h_new, new_n / 2.0)

def optimize(h, t_start, t_end, verbose=False):
    """Subgradient descent with adaptive step"""
    n = len(h)
    target = n / 2.0
    h = project_onto_box_sum(h.copy(), target)
    best_h = h.copy()
    best_c5 = c5_value(h)
    alpha = 0.005
    stagnation = 0
    
    while time.time() < t_end:
        T = 50.0 + 50.0 * min(1.0, stagnation / 500.0)
        g = smooth_grad(h, T)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-10:
            break
        step = alpha * g / gnorm
        h_new = project_onto_box_sum(h - step, target)
        c5_new = c5_value(h_new)
        
        if c5_new < best_c5 - 1e-8:
            best_c5 = c5_new
            best_h = h_new.copy()
            alpha = min(alpha * 1.08, 0.05)
            stagnation = 0
        elif c5_new < best_c5:

            h = project_onto_box_sum(h, target)

    return best_h, best_c5

import numpy as np
import time

def project_onto_box_sum(h, target_sum):
    lo, hi = -4.0, 4.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def grad_c_d(h, d):
    n = len(h)
    g = np.zeros(n)
    js = max(0, d)
    je = min(n - 1, n - 1 + d)
    if js <= je:
        jv = np.arange(js, je + 1)
        g[jv] += 1.0 - h[jv - d]
    js2 = max(0, -d)
    je2 = min(n - 1, n - 1 - d)
    if js2 <= je2:
        jv2 = np.arange(js2, je2 + 1)
        g[jv2] -= h[jv2 + d]
    return g

def smooth_grad(h, T, topk=20):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = corr.max()
    e = np.exp(T * (corr - mx))
    w = e / e.sum()
    # Use top-k weights for efficiency
    idx = np.argsort(w)[::-1][:topk]
    wsum = w[idx].sum()
    w_norm = w[idx] / wsum
    g = np.zeros(n)
    for i in range(len(idx)):
        d = int(idx[i] - (n - 1))
        g += w_norm[i] * grad_c_d(h, d)
    return g

def interpolate_h(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0.0, 2.0, old_n)
    x_new = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(x_new, x_old, h, left=0.0, right=0.0)
    return project_onto_box_sum(h_new, new_n / 2.0)

def optimize(h_init, t_limit, rng):
    n = len(h_init)
    target = n / 2.0
    h = project_onto_box_sum(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = c5_value(h)
    alpha = 0.003
    stagnation = 0
    it = 0
    while time.time() < t_limit:
        it += 1
        T = 40.0 + 60.0 * min(1.0, stagnation / 1000.0)
        g = smooth_grad(h, T, topk=15)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        h_new = project_onto_box_sum(h - alpha * g, target)
        c5_new = c5_value(h_new)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
            alpha = min(alpha * 1.06, 0.04)
            stagnation = 0
        elif c5_new < best_c5:
            alpha = min(alpha * 1.005, 0.04)
            stagnation = max(0, stagnation - 2)
        else:
            alpha *= 0.93
            stagnation += 1
        h = h_new
        if stagnation > 400:
            kick = 0.01 * (0.5 + 0.5 * stagnation / 200.0)
            h = project_onto_box_sum(h + kick * rng.randn(n), target)
            stagnation = 0
        if it % 200 == 0:
            alpha = max(alpha, 0.001)
    return best_h, best_c5
