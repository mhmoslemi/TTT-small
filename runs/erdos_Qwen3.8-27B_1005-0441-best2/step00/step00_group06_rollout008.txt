```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def smooth_max_obj(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return temp * logsumexp(corr / temp)

def sum_constraint(h, target_sum):
    return np.sum(h) - target_sum

def make_bounds(n):
    return [(0.0, 1.0)] * n

def fix_feasibility(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-10:
        return np.ones(len(h)) * (target_sum / len(h))
    h = h / s * target_sum
    h = np.clip(h, 0.0, 1.0)
    # Distribute residual
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        # Add/subtract uniformly
        h += residual / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    n = 95
    dx = 2.0 / n
    target_sum = n / 2.0
    
    # Get initial values
    h0 = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
        dx = 2.0 / n
        target_sum = n / 2.0
    if h0 is None:
        h0 = np.ones(n) * 0.5
    
    h0 = fix_feasibility(h0, target_sum)
    h_best = h0.copy()
    c5_best = compute_c5(h_best, dx)
    
    bounds = make_bounds(n)
    cons = [{'type': 'eq', 'fun': sum_constraint, 'args': (target_sum,)}]
    
    # Multi-temperature SLSQP
    temps = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    for temp in temps:
        if time.time() - start_time > budget_s * 0.7:
            break
        res = minimize(smooth_max_obj, h_best, args=(dx, temp),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': 400, 'ftol': 1e-12})
        if res.x is not None:
            h_try = fix_feasibility(res.x, target_sum)
            c5_try = compute_c5(h_try, dx)
            if c5_try < c5_best:
                h_best = h_try
                c5_best = c5_try
    
    # Random restarts
    if time.time() - start_time < budget_s * 0.4:
        n_restarts = 20
        for r in range(n_restarts):
            if time.time() - start_time > budget_s * 0.85:
                break
            h_try = h_best + np.random.randn(n) * 0.05
            h_try = fix_feasibility(h_try, target_sum)
            res = minimize(smooth_max_obj, h_try, args=(dx, 0.05),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-10})
            if res.x is not None:
                h_try = fix_feasibility(res.x, target_sum)
                c5_try = compute_c5(h_try, dx)
                if c5_try < c5_best:
                    h_best = h_try
                    c5_best = c5_try
                    # Refine
                    for temp in [0.01, 0.005]:
                        if time.time() - start_time > budget_s * 0.85:
                            break
                        res2 = minimize(smooth_max_obj, h_best, args=(dx, temp),
                                       method='SLSQP', bounds=bounds, constraints=cons,
                                       options={'maxiter': 200, 'ftol': 1e-12})
                        if res2.x is not None:
                            h_ref = fix_feasibility(res2.x, target_sum)
                            c5_ref = compute_c5(h_ref, dx)
                            if c5_ref < c5_best:
                                h_best = h_ref
                                c5_best = c5_ref
    
    # Local refinement with very low temperature
    if time.time() - start_time < budget_s * 0.8:
        res = minimize(smooth_max_obj, h_best, args=(dx, 0.001),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': 500, 'ftol': 1e-14})
        if res.x is not None:
            h_try = fix_feasibility(res.x, target_sum)
            c5_try = compute_c5(h_try, dx)
            if c5_try < c5_best:
                h_best = h_try
                c5_best = c5_try
    
    return h_best, c5_best, n
```

```python
from scipy.special import logsumexp

def compute_c5(h, dx):

    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    c = corr / temp
    c = c - c.max()
    return temp * (c.max() + np.log(np.exp(c).sum()))

    return np.sum(h) - target

    h = h * target / s
    h = np.clip(h, 0.0, 1.0)
```
