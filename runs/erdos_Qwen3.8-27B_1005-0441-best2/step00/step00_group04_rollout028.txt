import numpy as np

def _c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.argsort(1.0 - h)[::-1]
    else:
        idx = np.argsort(h)[::-1]
    step = diff / len(idx)
    for i in idx:
        if diff > 0:
            d = min(step, 1.0 - h[i])
            h[i] += d
            diff -= d
        else:
            d = min(-step, h[i])
            h[i] -= d
            diff += d
        if abs(diff) < 1e-14:
            break
    r = target - h.sum()
    if abs(r) > 1e-15:
        j = np.argmin(np.abs(h - 0.5))
        h[j] = max(0.0, min(1.0, h[j] + r))
