    m = np.max(corr)

A simple approach: ternary search won't work for piecewise linear. But bisection on the subgradient does:
- At the optimum, 0 is in the subgradient
- Subgradient at x: the set of active b_k (those achieving the max)
- If all active b_k > 0, f is increasing, go left
- If all active b_k < 0, f is decreasing, go right
- If there's a mix (some > 0, some < 0), we're at the optimum

- Or: the minimum value t* is the smallest t such that {x : a_k + b_k*x ≤ t for all k, 0≤x≤1} is non-empty
- For each k: x ≥ (t - a_k)/b_k (if b_k > 0) or x ≤ (t - a_k)/b_k (if b_k < 0)
- So t* = min over feasible intervals of t

This is a 1D LP. But for simplicity, let me just use a fine grid or bisection.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(corr))

def project_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def corr_and_grads_k(h, n, dx, j):
    """Return current corr array and d(corr[k])/d(h[j]) for all k."""
    corr = np.correlate(h, 1 - h, mode='full') * dx
    grads = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        g = 0.0
        if 0 <= j - lag < n:
            g += (1.0 - h[j - lag])
        if 0 <= j + lag < n:
            g += (-h[j + lag])
        grads[k] = dx * g
    return corr, grads

def min_1d_convex(corr, grads, old_val, n_iter=50):
    """Minimize max_k(corr[k] + grads[k]*(x - old_val)) over x in [0,1]."""
    a = corr - grads * old_val
    b = grads
    lo, hi = 0.0, 1.0
    for _ in range(n_iter):
        mid = (lo + hi) / 2.0
        vals = a + b * mid
        val_max = vals.max()
        active = vals >= val_max - 1e-14
        if active.any():
            if np.all(b[active] > 1e-14):
                hi = mid
            elif np.all(b[active] < -1e-14):
                lo = mid
            else:
                break
        else:
            idx = int(np.argmax(vals))
            if b[idx] > 0:
                hi = mid
            else:
                lo = mid
    return (lo + hi) / 2.0

def coordinate_descent(h, n, n_sweeps=50, t_start=None, time_limit=None):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    best_max = float(corr.max())
    for sweep in range(n_sweeps):
        if t_start is not None and time_limit is not None:
            if time.time() - t_start > time_limit:
                break
        improved = False
        for j in range(n):
            grads = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                lag = k - (n - 1)
                g = 0.0
                if 0 <= j - lag < n:
                    g += (1.0 - h[j - lag])
                if 0 <= j + lag < n:
                    g += (-h[j + lag])
                grads[k] = dx * g
            x_new = min_1d_convex(corr, grads, h[j])
            new_corr = corr + grads * (x_new - h[j])
            new_max = float(new_corr.max())
            if new_max < best_max - 1e-13:
                h[j] = x_new
                corr = new_corr
                best_max = new_max
                improved = True
        if not improved:
            break
    return h, best_max

def smooth_obj_and_close(z, n, t, lam):
    dx = 2.0 / n
    corr = np.correlate(z, 1 - z, mode='full') * dx
    m = corr.max()
    obj = m + np.log(np.sum(np.exp(t * (corr - m)))) / t
    obj += lam * (z.sum() - n / 2.0) ** 2
    return obj

def lbfbsb_polish(h, n, t_start, time_limit):
    dx = 2.0 / n
    for t_val in [20, 50, 100, 200, 500, 1000]:
        if time.time() - t_start > time_limit * 0.95:
            break
        lam = 500.0
        def obj(z):
            return smooth_obj_and_close(z, n, t_val, lam)
        res = minimize(obj, h, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-10})
        h_new = project_sum(res.x, n)
        c5_new = compute_c5(h_new)
        c5_old = compute_c5(h)
        if c5_new < c5_old:
            h = h_new
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    # Get initial
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        init_h = np.full(47, 0.5)
    n0 = len(init_h)
    init_h = project_sum(init_h, n0)
    c5_init = compute_c5(init_h)
    best_c5 = c5_init
    best_h = init_h.copy()

    def update_best(h, n):
        nonlocal best_h, best_c5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    # Phase 1: Coordinate descent from initial
    h1, _ = coordinate_descent(init_h.copy(), n0, n_sweeps=200, t_start=t_start, time_limit=budget_s * 0.5)

    update_best(h1, n0)

    # Phase 2: Multiple random restarts at various n

            h_try = project_sum(h_try, n_try)

            h_try = project_sum(h_try, n_try)
            update_best(h_try, n_try)

    # Phase 3: L-BFGS-B polish on best
    if time.time() - t_start < budget_s * 0.9:

        h_pol2 = lbfbsb_polish(best_h.copy(), n_best, t_start, budget_s)
        update_best(h_pol2, n_best)

    return best_h, best_c5, len(best_h)
