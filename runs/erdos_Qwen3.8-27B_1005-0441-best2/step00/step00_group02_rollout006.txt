import numpy as np
import time
import math

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(1000):
        d = tgt - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = h < 1.0 - 1e-12
            if m.sum() == 0:
                break
            s = d / m.sum()
            h[m] = np.minimum(h[m] + s, 1.0)
        else:
            m = h > 1e-12
            if m.sum() == 0:
                break
            s = -d / m.sum()
            h[m] = np.maximum(h[m] - s, 0.0)
    return h

def num_gradient(h, n, p):
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode='full') * dx
    gc = np.clip(g, 1e-15, None)
    S = np.sum(gc ** p)
    if S < 1e-30:
        return np.zeros(n)
    f0 = S ** (1.0 / p)
    grad = np.zeros(n)
    eps = 1e-6
    for i in range(n):
        hp = h.copy()
        hp[i] = min(1.0, h[i] + eps)
        hm = h.copy()
        hm[i] = max(0.0, h[i] - eps)
        hp = proj(hp, n)
        hm = proj(hm, n)
        gp = np.correlate(hp, 1.0 - hp, mode='full') * dx
        gm = np.correlate(hm, 1.0 - hm, mode='full') * dx
        fp = np.sum(np.clip(gp, 1e-15, None) ** p) ** (1.0 / p)
        fm = np.sum(np.clip(gm, 1e-15, None) ** p) ** (1.0 / p)
        denom = hp[i] - hm[i]
        if abs(denom) > 1e-14:
            grad[i] = (fp - fm) / denom
        else:
            grad[i] = 0.0
    return grad

def optimize_h(h, n, p, lr, max_iters, t0, budget_s):
    best_h = h.copy()
    best_c5 = eval_c5(h)
    patience = 0
    for it in range(max_iters):
        if time.time() - t0 > budget_s - 40:
            break
        h_new = h - lr * num_gradient(h, n, p)
        h_new = proj(h_new, n)
        c5_new = eval_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience = 0
            lr = min(lr * 1.03, 0.15)
        else:
            patience += 1
            if c5_new < eval_c5(h) - 1e-12:
                h = h_new.copy()
            lr *= 0.75
        h = proj(h, n)
        if not np.isfinite(h).all():
            h = best_h.copy()
            break
        if patience > 300:
            break
    return best_h, best_c5

def perturb(h, n, rng, sigma):
    h_p = h + rng.normal(0, sigma, n)
    return proj(h_p, n)
