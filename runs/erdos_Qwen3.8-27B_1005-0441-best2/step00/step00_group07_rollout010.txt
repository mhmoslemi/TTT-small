import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def soft_max_obj(h, alpha):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    return (cmax + np.log(np.sum(np.exp(alpha * (corr - cmax)))) / alpha) * dx

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    # Binary search for lambda to fix the sum
    s = h.sum()
    if abs(s - target_sum) < 1e-10:
        return h
    # Need s + delta = target_sum where delta = target_sum - s
    # Adjust by adding/subtracting lambda to interior points
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h_new = np.clip(h + mid, 0.0, 1.0)
        if h_new.sum() > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def soft_max_obj(h, alpha):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    return (cmax + np.log(np.sum(np.exp(alpha * (corr - cmax)))) / alpha) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def constraint_sum(x, n):
    return np.sum(x) - n / 2.0

def constraint_jac(x, n):
    return np.ones(n)

def make_binary_init(n, rng):
    vals = rng.random(n * 2)
    sorted_vals = np.sort(vals)[:n]
    threshold = sorted_vals[n // 2]
    h = (vals[:n] > threshold).astype(float)
    # Ensure exactly n//2 ones
    n_ones = int(h.sum())
    idx = np.where(h == 1.0)[0]
    idx0 = np.where(h == 0.0)[0]
    if n_ones > n // 2:
        to_flip = rng.choice(idx, n_ones - n // 2, replace=False)
        h[to_flip] = 0.0
    elif n_ones < n // 2:
        to_flip = rng.choice(idx0, n // 2 - n_ones, replace=False)
        h[to_flip] = 1.0
    return h
