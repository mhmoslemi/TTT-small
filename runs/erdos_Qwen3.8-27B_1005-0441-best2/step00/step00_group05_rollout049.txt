        else:

    h_bin = (h >= 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones > target_sum:
        # Remove excess ones (those closest to 0.5)
        one_indices = np.where(h_bin == 1)[0]
        # Sort by h value (ascending), remove smallest
        order = np.argsort(h[one_indices])
        for i in range(n_ones - target_sum):
            h_bin[one_indices[order[i]]] = 0

        zero_indices = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_indices])
        for i in range(target_sum - n_ones):
            h_bin[zero_indices[order[i]]] = 1
    return h_bin

    zeros = np.where(h == 0)[0]

        if len(ones) == 0 or len(zeros) == 0:
