    w = np.exp(c / T - c.max())
    w /= w.sum()

import numpy as np
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))

def _normalize_h(h):
    h = np.clip(h, 0.0, 1.0).copy()
    target = len(h) * 0.5
    for _ in range(1000):
        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] -= diff / cnt
        else:
            mask = h < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] -= diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _soft_gradient(h, T=0.02):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    w = np.exp(c / T - c.max())
    w /= w.sum()
    g = np.zeros(n)
    for j in range(2 * n - 1):
        tau = j - (n - 1)
        t = abs(tau)
        if t >= n:
            continue
        coeff = w[j]
        if coeff < 1e-8:
            continue
        if tau >= 0:
            g[:n - t] += coeff * (1.0 - h[t:])
            g[t:] -= coeff * h[:n - t]
        else:
            g[t:] += coeff * (1.0 - h[:n - t])
            g[:n - t] -= coeff * h[t:]
    return g

def _subgrad_single(h, tau):
    n = len(h)
    g = np.zeros(n)
    t = abs(tau)
    if t >= n:
        return g
    if tau >= 0:
        g[:n - t] += 1.0 - h[t:]
        g[t:] -= h[:n - t]
    else:
        g[t:] += 1.0 - h[:n - t]
        g[:n - t] -= h[t:]
    return g

def _optimize_continuous(h0, n_iter, lr0, decay, mom, T, t_start, time_limit):
    h = _normalize_h(h0.astype(np.float64).copy())
    n = len(h)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    for i in range(n_iter):
        if time.time() - t_start > time_limit:
            break
        g = _soft_gradient(h, T)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        v = mom * v + g / gnorm
        lr = lr0 * (decay ** i)
        h = _normalize_h(h - lr * v)
        c5 = _compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def _binary_local_search(h_cont, n_iters, t_start, time_limit, rng):
    n = len(h_cont)
    idx = np.argsort(h_cont)[::-1]
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    temp = 0.01
    for i in range(n_iters):
        if time.time() - t_start > time_limit:
            break
        if i % 500 == 499:
            temp = max(0.001, temp * 0.95)
        oi = int(rng.choice(ones))
        zi = int(rng.choice(zeros))
        h[oi] = 0
        h[zi] = 1
        c5_new = _compute_c5(h)
        delta = c5_new - best_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(temp, 1e-6)):
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            ones.remove(oi)
            ones.append(zi)
            zeros.remove(zi)
            zeros.append(oi)
        else:
            h[oi] = 1
            h[zi] = 0
    return best_h, best_c5

def _try_n_values(h_parent, n_val, rng, t_start, time_limit):
    n_old = len(h_parent)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 2.0 / (2 * n_old)
    x_new = np.linspace(0, 2, n_val, endpoint=False) + 2.0 / (2 * n_val)
    h_try = np.interp(x_new, x_old, h_parent)
    h_try = _normalize_h(h_try)
    h_opt, c5_opt = _optimize_continuous(h_try, 2000, 0.03, 0.9995, 0.9, 0.02, t_start, time_limit)
    h_bin, c5_bin = _binary_local_search(h_opt, 3000, t_start, time_limit, rng)
    if c5_bin < c5_opt:
        return h_bin, c5_bin
    return h_opt, c5_opt

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    remaining = lambda: max(0.0, budget_s - (time.time() - t_start))

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=np.float64)
    else:
        h_parent = np.full(70, 0.5)

    best_h = _normalize_h(h_parent.copy())
    best_c5 = _compute_c5(best_h)
    n = len(best_h)

    # Phase 1: continuous subgradient from parent
    for lr0, T in [(0.08, 0.03), (0.04, 0.02), (0.02, 0.01), (0.01, 0.005)]:
        if remaining() < 60:
            break
        h_opt, c5_opt = _optimize_continuous(best_h, 3000, lr0, 0.9997, 0.92, T,
                                             t_start, t_start + remaining() * 0.25)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: binary local search from current best
    if remaining() > 30:
        h_bin, c5_bin = _binary_local_search(best_h, int(remaining() * 80),
                                             t_start, t_start + remaining() * 0.2, rng)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 3: perturbation restarts
    for attempt in range(15):
        if remaining() < 30:
            break
        frac = min(0.15, remaining() * 0.05)
        h_try = best_h + 0.08 * rng.randn(n)
        h_try = _normalize_h(h_try)
        h_opt, c5_opt = _optimize_continuous(h_try, 2000, 0.03, 0.9997, 0.9, 0.015,
                                             t_start, t_start + frac)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        if remaining() < 40:
            break
        h_b, c5_b = _binary_local_search(h_opt, 2000, t_start, t_start + remaining() * 0.05, rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 4: try different n
    for n_try in [98, 140, 200]:
        if remaining() < 80:
            break
        h_opt, c5_opt = _try_n_values(best_h, n_try, rng, t_start, t_start + remaining() * 0.15)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final polish: a few more subgradient steps
    if remaining() > 20:
        h_opt, c5_opt = _optimize_continuous(best_h, 1000, 0.005, 0.9999, 0.95, 0.01,
                                             t_start, t_start + remaining() * 0.4)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    best_h = _normalize_h(best_h)
    final_c5 = _compute_c5(best_h)
    return best_h, final_c5, len(best_h)
