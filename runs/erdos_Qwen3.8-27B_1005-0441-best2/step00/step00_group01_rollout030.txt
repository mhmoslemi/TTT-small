    return float(np.max(corr) * dx)

def num_grad(h, eps=1e-6):
    n = len(h)
    g = np.zeros(n)
    base = compute_c5(h)
    for j in range(n):
        h[j] += eps
        g[j] = (compute_c5(h) - base) / eps
        h[j] -= eps
    return g

def fix_sum(h, target):
    """Adjust h so sum == target, keeping in [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    for _ in range(500):
        if abs(diff) < 1e-14:
            break
        free = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
        if len(free) == 0:
            break
        step = diff / len(free)
        h[free] += step
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.asarray(initial_h_values, dtype=float).copy()
    n = len(h0)
    ts = n / 2.0

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    def tl():
        return budget_s - (time.time() - t0)

    # Evaluate a few alternative starts
    alternates = [np.full(n, 0.5)]
    for frac in [0.35, 0.55, 0.65]:
        m = max(1, int(n * frac))
        hb = np.zeros(n)
        val = ts / m
        if val <= 1.0:
            hb[:m] = val
            alternates.append(fix_sum(hb, ts))
    # Sine pattern
    xs = np.linspace(0, 2, n, endpoint=False)
    hs = 0.5 + 0.25 * np.sin(np.pi * xs)
    alternates.append(fix_sum(hs, ts))

    for s in alternates:
        if len(s) == n and abs(np.sum(s) - ts) < 1e-8:
            c = compute_c5(s)
            if c < best_c5:
                best_h, best_c5 = s.copy(), c

    # Main local search loop
    stall = 0
    iter_count = 0
    while tl() > 180:
        improved = False
        iter_count += 1
        h = best_h.copy()

        # Gradient-informed step
        g = num_grad(h)
        gc = g - np.mean(g)
        ng = np.linalg.norm(gc)
        if ng > 1e-15:
            d = -gc / ng
            for alpha in [0.04, 0.02, 0.01, 0.005, 0.002]:
                hn = h + alpha * d
                hn = fix_sum(np.clip(hn, 0, 1), ts)
                c5n = compute_c5(hn)
                if c5n < best_c5 - 1e-12:
                    best_h, best_c5 = hn.copy(), c5n
                    improved = True
                    break

        # Gradient-directed pairwise swaps
        order = np.argsort(g)
        top_k = min(25, n)
        for _ in range(300):
            i = order[rng.randint(top_k)]
            j = order[n - 1 - rng.randint(top_k)]
            if i == j:
                continue
            if g[i] < g[j]:
                dmax = min(1.0 - best_h[i], best_h[j])
                if dmax < 1e-9:
                    continue
                dd = rng.uniform(0.002, min(0.06, dmax * 0.7))
                oi, oj = best_h[i], best_h[j]
                best_h[i], best_h[j] = oi + dd, oj - dd
            else:
                continue
            c5n = compute_c5(best_h)
            if c5n < best_c5 - 1e-12:
                best_c5 = c5n
                improved = True
            else:
                best_h[i], best_h[j] = oi, oj

        # Random swaps
        for _ in range(200):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            dd = rng.uniform(-0.04, 0.04)
            oi, oj = best_h[i], best_h[j]
            ni, nj = oi + dd, oj - dd
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                best_h[i], best_h[j] = ni, nj
                c5n = compute_c5(best_h)
                if c5n < best_c5 - 1e-12:
                    best_c5 = c5n
                    improved = True
                else:
                    best_h[i], best_h[j] = oi, oj

        # 4-swap for more flexibility (every few iterations)
        if iter_count % 3 == 0:
            for _ in range(50):
                sel = rng.choice(n, size=4, replace=False)
                i1, j1, i2, j2 = sel
                d1 = rng.uniform(0.005, 0.03)
                d2 = rng.uniform(0.005, 0.03)
                vals = [best_h[i1] + d1, best_h[j1] - d1,
                        best_h[i2] + d2, best_h[j2] - d2]
                if all(0.0 <= v <= 1.0 for v in vals):
                    old = [best_h[i1], best_h[j1], best_h[i2], best_h[j2]]
                    best_h[i1], best_h[j1], best_h[i2], best_h[j2] = vals
                    c5n = compute_c5(best_h)
                    if c5n < best_c5 - 1e-12:
                        best_c5 = c5n
                        improved = True
                    else:
                        best_h[i1], best_h[j1], best_h[i2], best_h[j2] = old

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 60:
                # Perturb to escape local optimum
                h_p = best_h + rng.normal(0, 0.015, n)
                h_p = fix_sum(h_p, ts)
                c5p = compute_c5(h_p)
                if c5p < best_c5:
                    best_h, best_c5 = h_p, c5p
                stall = 0

    # SLSQP polishing
    if tl() > 40:
        try:
            def _obj(x):
                return compute_c5(x)
            def _jac(x):
                return num_grad(x, 1e-7)
            cons = [{'type': 'eq',
                     'fun': lambda x: np.sum(x) - ts,
                     'jac': lambda x: np.ones(n)}]
            bnds = [(0.0, 1.0)] * n
            res = minimize(_obj, best_h.copy(), jac=_jac,
                          method='SLSQP', bounds=bnds,
                          constraints=cons,
                          options={'maxiter': 400, 'ftol': 1e-14})
            if res.x is not None and np.all(res.x >= -1e-9) and np.all(res.x <= 1 + 1e-9):
                hx = np.clip(res.x, 0, 1)
                if abs(np.sum(hx) - ts) < 1e-8:
                    cx = compute_c5(hx)
                    if cx < best_c5:
                        best_h, best_c5 = hx, cx
        except Exception:
            pass

    # Final neighborhood polish (exhaustive 2-swap passes)
    if tl() > 10:
        for _ in range(3):
            if tl() < 5:
                break
            for i in range(n):
                if tl() < 3:
                    break
                for j in range(n):
                    if i == j:
                        continue
                    for sign in (1, -1):
                        dd = 0.01
                        ni, nj = best_h[i] + sign * dd, best_h[j] - sign * dd
                        if 0 <= ni <= 1 and 0 <= nj <= 1:
                            oi, oj = best_h[i], best_h[j]
                            best_h[i], best_h[j] = ni, nj
                            c5n = compute_c5(best_h)
                            if c5n < best_c5 - 1e-12:
                                best_c5 = c5n
                            else:
                                best_h[i], best_h[j] = oi, oj

    # Ensure exact sum
    diff = ts - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h = fix_sum(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
