import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    cc = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(cc), cc

def grad_for_shift(h, s):
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        if m > 0:
            grad[:m] += (1.0 - h[s:])
            grad[s:] -= h[:m]
    else:
        t = -s
        m = n - t
        if m > 0:
            grad[t:] += (1.0 - h[:m])
            grad[:m] -= h[t:]
    return grad

def project_feasible(h, target):
    n = len(h)
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def soft_max_grad(h, T=40.0):
    n = len(h)
    dx = 2.0 / n
    cc = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = np.max(cc)
    shifted = (cc - mx) * T
    shifted -= np.max(shifted)
    ew = np.exp(shifted)
    zw = ew / ew.sum()
    c_soft = mx + (np.max((cc - mx) * T) + np.log(ew.sum()) - shifted.max()) / T
    # c_soft = mx + (mx*T + log(sum(exp(T*(cc-mx))))) / T ... let me just compute directly
    c_soft = mx + np.log(ew.sum()) / T
    grad = np.zeros(n)
    for i in range(len(cc)):
        w = zw[i]
        if w < 1e-7:
            continue
        s = i - n + 1
        if abs(s) >= n:
            continue
        grad += w * grad_for_shift(h, s)
    return c_soft, grad * dx

def optimize_h(h, time_limit, start_time):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)[0]
    lr = 0.03
    T = 40.0
    momentum = np.zeros(n)
    beta = 0.7
    no_improve = 0

    while time.time() - start_time < time_limit:
        c_soft, grad = soft_max_grad(h, T)
        gn = np.linalg.norm(grad)
        if gn > 1e-12:
            momentum = beta * momentum + (1 - beta) * (grad / gn)
            h_new = h - lr * momentum
        else:
            break
        h_new = project_feasible(h_new, target)
        c_new = compute_c5(h_new)[0]

        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            lr = min(lr * 1.01, 0.15)
        else:
            no_improve += 1
            lr *= 0.9
            if no_improve > 40:
                # Switch to exact subgradient
                _, cc = compute_c5(h)
                k_star = np.argmax(cc)
                s = k_star - n + 1
                eg = grad_for_shift(h, s)
                en = np.linalg.norm(eg)
                if en > 1e-12:
                    h_new2 = h - lr * 0.5 * eg / en

                        best_h = h_new2.copy()
                        h = h_new2
                no_improve = 0
                lr = max(lr * 1.2, 0.02)
