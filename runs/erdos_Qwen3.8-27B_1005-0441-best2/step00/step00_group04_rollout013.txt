import numpy as np
from scipy.optimize import minimize
import time


def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))


def _smooth_max(c, beta):
    m = np.max(c)
    s = c - m
    return m + np.log(np.sum(np.exp(beta * s))) / beta


def _fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        h = h + diff / n
    return np.clip(h, 0.0, 1.0)


def _local_opt(h, n, beta=20.0, max_iter=600):
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        sm = _smooth_max(c, beta)
        pen = 800.0 * (x.sum() - target) ** 2
        return sm + pen

    res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-13, 'maxls': 40})
    out = np.clip(res.x, 0.0, 1.0)
    out = _fix_sum(out, n)
    return out


def _nelder_mead_refine(h, n, max_iter=200):
    """Local refinement with Nelder-Mead on a small neighborhood."""
    dx = 2.0 / n
    target = n / 2.0

        c = np.correlate(x, 1.0 - x, mode='full') * dx
        sm = _smooth_max(c, 50.0)
        pen = 1000.0 * (x.sum() - target) ** 2
        return sm + pen

    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='Nelder-Mead',
                   options={'maxiter': max_iter, 'xatol': 1e-9, 'fatol': 1e-12})
    return _fix_sum(res.x, n)
