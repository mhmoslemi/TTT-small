    for k_idx in active_idx:

        # First term: (1-h[m-s]) for valid 0<=m-s<n
        if s >= 0:
            # m-s >= 0 means m >= s; m-s < n means m < n+s (always true since m < n)
            grad[s:] += (1.0 - h[:n-s])  # for m = s..n-1, h[m-s] = h[0..n-1-s]
        else:
            # m-s >= 0 means m >= 0 (always); m-s < n means m < n+s
            grad[:n+s] += (1.0 - h[-s:])  # for m = 0..n-1+s, h[m-s] = h[-s..n-1]
        
        # Second term: -h[m+s] for valid 0<=m+s<n
        if s >= 0:
            # m+s < n means m < n-s
            grad[:n-s] -= h[s:]  # for m = 0..n-1-s, h[m+s] = h[s..n-1]
        else:
            # m+s >= 0 means m >= -s
            grad[-s:] -= h[:s]  # for m = n-s..n-1, h[m+s] = h[0..s-1]
    return grad

import numpy as np
import time

def adjust_sum(h, target):
    """Adjust h in-place to meet sum constraint while staying in [0,1]."""
    h = h.copy()
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        h[mask] = np.clip(h[mask] + diff / np.sum(mask), 0.0, 1.0)
    return h

def compute_c5(h):
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / len(h))

def compute_gradient(h, active_idx):
    """Gradient of sum_{k in active} c[k] w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    for k_idx in active_idx:
        s = k_idx - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += (1.0 - h[:n - s])
            if s > 0:
                grad[:n - s] -= h[s:]
        else:
            if -s < n:
                grad[:n + s] += (1.0 - h[-s:])
            if s > -n:
                grad[-s:] -= h[:s]
    return grad

def do_gradient_step(h, c, step_size, target_sum, rng):
