import numpy as np
import time


def project_to_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(h)
            for i in range(n):
                space = 1.0 - h[order[i]]
                add = min(diff, space)
                h[order[i]] += add
                diff -= add
                if abs(diff) < 1e-14:
                    break
        else:
            order = np.argsort(-h)
            for i in range(n):
                space = h[order[i]]
                sub = min(-diff, space)
                h[order[i]] -= sub
                diff += sub
                if abs(diff) < 1e-14:
                    break
    return h


def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_soft_gradient(h, T):
    """Gradient of soft-max of correlate values. Returns (grad, max_corr)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    w = np.exp((corr - mx) / T)
    w /= np.sum(w)
    grad = np.zeros(n)
    for idx in range(len(corr)):
        wi = w[idx]
        if wi < 1e-12:
            continue
        s = idx - (n - 1)
        if s == 0:
            grad -= wi * 2.0 * h
        elif s > 0:
            grad[s:] -= wi * h[:n - s]
            grad[:n - s] -= wi * h[s:]
        else:
            p = -s
            grad[:n - p] -= wi * h[p:]
            grad[p:] -= wi * h[:n - p]
    return grad, mx


def local_optimize(h, n, budget_s, start_time, rng, max_iters=20000):

    dx = 2.0 / n
    T = 0.02
    lr = 0.08
    no_improve = 0

    best_local = evaluate_c5(h)

    for it in range(max_iters):
        if time.time() - start_time > budget_s - 15:
            break
        grad, mx_corr = compute_soft_gradient(h, T)
        curr_c5 = mx_corr * dx

        h_new = project_to_feasible(h_new, n)
        new_c5 = evaluate_c5(h_new)

        if new_c5 < curr_c5 * dx + 1e-10:
            h = h_new
        else:
            lr *= 0.88

            best_local = new_c5
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                T *= 0.85
                lr = 0.04
                no_improve = 0
                if T < 0.001:
                    break

    return h, evaluate_c5(h)
