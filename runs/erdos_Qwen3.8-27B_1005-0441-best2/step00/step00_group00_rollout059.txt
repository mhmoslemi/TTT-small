import numpy as np
import time


def project_to_feasible(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        m = (h > 0) & (h < 1)
        idx = np.where(m)[0]
        if idx.size > 0:
            h[idx] = np.clip(h[idx] + diff / idx.size, 0.0, 1.0)
    return h


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def subgrad_dir(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    ki = int(np.argmax(corr))
    k = ki - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        m = n - k
        grad[:m] += (1.0 - h[k:])
        grad[k:] -= h[:m]
    else:
        m = -k
        grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] -= h[m:]
    gn = np.linalg.norm(grad)
    if gn < 1e-14:
        return None
    return grad / gn


def try_binary_snap(h, n):
    idx = np.argsort(h)
    snap = np.zeros(n)
    nh = n // 2
    snap[idx[:nh]] = 0.0
    snap[idx[nh + 1:]] = 1.0
    if n % 2 == 1:
        snap[idx[nh]] = 0.5
    else:
        snap[idx[nh:]] = 1.0
    return snap


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    limit = budget_s - 4.0

    n = 45
    if 'initial_h_values' in globals() and initial_h_values is not None:

    best_h = project_to_feasible(h0, n).copy()
    best_c5 = compute_c5(best_h, n)

    rng = np.random.RandomState(seed)

    # Phase 1: Simulated Annealing
    T = 0.005
    alpha = 0.99999
    T_min = 1e-9
    cur_h = best_h.copy()
    cur_c5 = best_c5
    sa_end = t0 + limit * 0.65

    while time.time() < sa_end:
        r = rng.rand()
        if r < 0.50:
            new_h = project_to_feasible(cur_h + rng.randn(n) * 0.05, n)
        elif r < 0.80:
            new_h = cur_h.copy()
            i, j = rng.randint(0, n, 2)
            if i != j:
                new_h[i], new_h[j] = new_h[j], new_h[i]
            else:

        else:
            new_h = cur_h.copy()
            i, j = rng.randint(0, n, 2)
            if i != j:

                new_h[i] = np.clip(cur_h[i] + delta, 0.0, 1.0)

                new_h = project_to_feasible(new_h, n)
            else:
