        zeros_idx = np.where(h < 0.5)[0]
        to_one = rng.choice(zeros_idx, half - s, replace=False)
        h[to_one] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial if available
    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        pass
    
    best_h = None
    best_c5 = float('inf')
    
    def try_record(h, c5):
        nonlocal best_h, best_c5
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Candidate n values
    n_list = [48, 50, 52, 54, 56, 60]
    if h_init is not None:
        n_list.append(len(h_init))
        n_list = sorted(set(n_list))
    
    n_patterns = {47: [0], 48: [0, 2, 3], 50: [0, 2], 52: [1, 2], 54: [2, 3], 56: [2], 60: [2]}
    
    for n in n_list:
        if time.time() - t0 > budget_s - 20:
            break
        elapsed = time.time() - t0
        remaining = budget_s - elapsed - 10
        if remaining < 5:
            break
        
        patterns = n_patterns.get(n, [0, 2, 3])
        patterns = patterns[:3]
        
        for pat in patterns:
            if time.time() - t0 > budget_s - 20:
                break
            remaining = budget_s - (time.time() - t0) - 10
            if remaining < 3:
                break
            
            # Initial guess
            if h_init is not None and n == len(h_init) and pat == patterns[0]:
                h0 = h_init.copy()
                # Ensure sum constraint
                h0 = project_feasible(h0, n / 2.0)
            else:
                h0 = make_initial(n, pat, rng)
            
            # Continuous phase (short)
            b_cont = min(20, remaining * 0.25)
            if remaining > 10:
                h_cont, c5_cont = optimize_continuous(h0, b_cont, rng)
                try_record(h_cont, c5_cont)
            
            # Binary + local search
            h_bin = to_binary(h0, n)
            b_ls = min(80, remaining * 0.55)
            if b_ls > 2:
                h_ls, c5_ls = local_search_binary(h_bin, b_ls, rng)
                try_record(h_ls, c5_ls)
            
            # Second round of local search from best binary with different seed
            if time.time() - t0 < budget_s - 30:
                h_bin2 = to_binary(h_ls if 'h_ls' in dir() else h0, n)
                b_ls2 = min(40, (budget_s - (time.time() - t0)) * 0.2 - 5)
                if b_ls2 > 2:
                    h_ls2, c5_ls2 = local_search_binary(h_bin2, b_ls2, rng)
                    try_record(h_ls2, c5_ls2)
    
    # Final: try one more continuous refinement from best binary
    if best_h is not None and time.time() - t0 < budget_s - 10:
        n_best = len(best_h)
        h_refine = project_feasible(best_h.astype(float), n_best / 2.0)
        b_final = min(30, budget_s - (time.time() - t0) - 2)
        if b_final > 2:
            h_f, c5_f = optimize_continuous(h_refine, b_final, rng)
            try_record(h_f, c5_f)
    
    if best_h is None:
        n_fb = 48
        best_h = np.zeros(n_fb)
        best_h[:n_fb // 2] = 1.0
        best_c5 = c5_value(best_h)
    
    final_c5 = c5_value(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
