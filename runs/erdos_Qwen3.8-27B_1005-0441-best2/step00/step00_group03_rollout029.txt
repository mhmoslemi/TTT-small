def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return np.max(c) * dx

def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    target = n / 2.0
    if s > 1e-10:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient_vec(h, t, n):
    """Gradient of c[n-1+t] w.r.t. h, vectorized."""
    grad = np.zeros(n)
    # d/dh[m] of sum_j h[j]*(1-h[j-t])
    # = (1-h[m-t])*1[0<=m-t<n] - h[m+t]*1[0<=m+t<n]
    # Term 1: m ranges from max(0,t) to min(n-1, n-1+t)
    lo1, hi1 = max(0, t), min(n, n + t)

def compute_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return np.max(c) * (2.0 / n)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-12:
        h *= (n / 2.0) / s
    return np.clip(h, 0.0, 1.0)

def grad_lag(h, t, n):
    g = np.zeros(n)
    # Term 1: 1_{j range}
    jlo = max(0, t)
    jhi = min(n - 1, n - 1 + t)
    if jlo <= jhi:
        g[jlo:jhi+1] = 1.0
    # Term 2: -h[j-t]
    a = max(0, t)
    b = min(n, n + t)
    if a < b:
        g[a:b] -= h[a-t:b-t]
    # Term 3: -h[j+t]
    a = max(0, -t)
    b = min(n, n - t)
    if a < b:
        g[a:b] -= h[a+t:b+t]
    return g

    h = project_feasible(h_init.astype(float).copy(), n)
    best = compute_c5(h)
    best_h = h.copy()
    stale = 0
