```python
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return np.max(c) * dx

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return np.max(c) * dx, c

def project_feasible(h, n):
    """Project h onto {x : 0<=x[i]<=1, sum(x)=n/2} using bisection on shift."""
    target = n / 2.0
    s0 = np.sum(h)
    if abs(s0 - target) < 1e-12:
        return np.clip(h, 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + 0.5*(lo+hi), 0.0, 1.0)

def corr_grad(h, k_idx, n):
    """Gradient of the k_idx-th component of correlate(h, 1-h, 'full') wrt h."""
    m = k_idx - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        jm = j - m
        jp = j + m
        if 0 <= jm < n:
            g += 1.0 - h[jm]
        if 0 <= jp < n:
            g -= h[jp]
        grad[j] = g
    return grad
```

```python
def corr_grad(h, k_idx, n):
    m = k_idx - (n - 1)
    grad = np.zeros(n)
    # For each j, grad[j] = (1-h[j-m]) * [0<=j-m<n] - h[j+m] * [0<=j+m<n]
    # Split into valid ranges
    # h[j-m] is valid when 0 <= j-m < n, i.e., m <= j < n+m
    lo1 = max(0, m)
    hi1 = min(n, n + m)
    if lo1 < hi1:
        grad[lo1:hi1] += 1.0 - h[lo1-m:hi1-m]
    # h[j+m] is valid when 0 <= j+m < n, i.e., -m <= j < n-m
    lo2 = max(0, -m)
    hi2 = min(n, n - m)
    if lo2 < hi2:
        grad[lo2:hi2] -= h[lo2+m:hi2+m]
    return grad
```

```python
import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def compute_c5_arr(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx), c


def project_feasible(h, n):
    target = n * 0.5
    hclip = np.clip(h, 0.0, 1.0)
    if abs(float(np.sum(hclip)) - target) < 1e-13:
        return hclip
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)


def corr_grad(h, k_idx, n):
    m = k_idx - (n - 1)
    grad = np.zeros(n)
    lo1 = max(0, m)
    hi1 = min(n, n + m)
    if lo1 < hi1:
        grad[lo1:hi1] += 1.0 - h[lo1 - m:hi1 - m]
    lo2 = max(0, -m)
    hi2 = min(n, n - m)
    if lo2 < hi2:
        grad[lo2:hi2] -= h[lo2 + m:hi2 + m]
    return grad


def subgrad_opt(h0, n, max_it=5000, lr0=0.04, tl=None, t0=None):
    h = h0.copy()
    h_prev = h.copy()
    _, c = compute_c5_arr(h)
    best = float(np.max(c) * 2.0 / n)
    best_h = h.copy()
    lr = lr0
    mom = 0.0
    for it in range(max_it):
        if tl is not None and t0 is not None and time.time() - t0 > tl:
            break
        top3 = np.argsort(c)[-3:]
        g = np.zeros(n)
        for k in top3:
            g += corr_grad(h, int(k), n)
        g *= 1.0 / 3.0
        h_new = h - lr * g + mom * (h - h_prev)
        h_prev = h.copy()
        h = project_feasible(h_new, n)
        c5v, c = compute_c5_arr(h)
        if c5v < best - 1e-13:
            best = c5v
            best_h = h.copy()
            lr = min(lr * 1.03, 0.15)
            mom = min(mom + 0.01, 0.12)
        else:
            lr = max(lr * 0.97, 0.002)
            mom = max(mom * 0.92, 0.0)
    return best_h, best


def coord_descent(h0, n, sweeps=100, tl=None, t0=None):
    h = h0.copy()
    c5c = compute_c5(h)
    best = c5c
    best_h = h.copy()
    dlist = [0.04, 0.02, 0.01, 0.005, -0.04, -0.02, -0.01, -0.005]
    for sw in range(sweeps):
        if tl is not None and t0 is not None and time.time() - t0 > tl:
            break
        imp = False
        for j in range(n):
            for d in dlist:
                ht = h.copy()
                ht[j] += d
                ht = project_feasible(ht, n)
                c5t = compute_c5(ht)
                if c5t < c5c - 1e-11:
                    h = ht
                    c5c = c5t
                    imp = True
                    if c5c < best:
                        best = c5c
                        best_h = h.copy()
                    break
        if not imp:
            break
    return best_h, best


def multi_start(h_best, c5_best, n, n_start=12, tl=None, t0=None, seed=0):
    rng = np.random.RandomState(seed)
    for i in range(n_start):
        if tl is not None and t0 is not None and time.time() - t0 > tl:
            break
        sig = 0.1 + 0.2 * rng.rand()
        hp = h_best + rng.normal(0.0, sig, n)
        hp = project_feasible(hp, n)
        h1, c5_1 = subgrad_opt(hp, n, max_it=1500, lr0=0.03, tl=tl, t0=t0)
        h2, c5_2 = coord_descent(h1, n, sweeps=30, tl=tl, t0=t0)
        if c5_2 < c5_best:
            h_best, c5_best = h2.copy(), c5_2
    return h_best, c5_best


def upsample_h(h, factor):
    no = len(h)
    nn = no * factor
    xo = np.linspace(0.0, 2.0, no, endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    hn = np.interp(xn, xo, h)
    return project_feasible(hn, nn)


def run(seed=42, budget_s=1100, **kwargs):

    np.random.seed(seed)

    h3, c5_3 = multi_start(h_best, c5_best, n, n_start=15,
                           tl=budget_s * 0.72, t0=t0, seed=seed + 1)
    h_best, c5_best = h3, c5_3

    if budget_s - (time.time() - t0) > 80:
        h_up = upsample_h(h_best, 2)
        n2 = len(h_up)
        h4, c5_4 = subgrad_opt(h_up, n2, max_it=6000, lr0=0.03,
                               tl=budget_s * 0.86, t0=t0)
        h5, c5_5 = coord_descent(h4, n2, sweeps=60,
                                 tl=budget_s * 0.92, t0=t0)
        if c5_5 < c5_best:
            h_best, c5_best = h5, c5_5

    nfin = len(h_best)
    hf, c5f = subgrad_opt(h_best, nfin, max_it=3000, lr0=0.02,
                          tl=budget_s * 0.97, t0=t0)
    if c5f < c5_best:

    hf2, c5f2 = coord_descent(h_best, nfin, sweeps=40,
                              tl=budget_s * 0.99, t0=t0)
    if c5f2 < c5_best:
        h_best, c5_best = hf2, c5f2

    h_best = np.clip(h_best, 0.0, 1.0)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
```

```python
def corr_grad(h, k_idx, n):
    m = k_idx - (n - 1)
    grad = np.zeros(n)
    # First term: (1-h[j+m]) for valid j
    j1_lo = max(0, -m)
    j1_hi = min(n, n - m)
    if j1_lo < j1_hi:
        grad[j1_lo:j1_hi] += 1.0 - h[j1_lo + m:j1_hi + m]
    # Second term: -h[j-m] for valid j
    j2_lo = max(0, m)
    j2_hi = min(n, n + m)
    if j2_lo < j2_hi:
        grad[j2_lo:j2_hi] -= h[j2_lo - m:j2_hi - m]
    return grad
```
