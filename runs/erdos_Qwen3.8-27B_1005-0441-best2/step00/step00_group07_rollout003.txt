import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_obj(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.minimum(shifted, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        step = diff / n
        h_new = np.clip(h + step, 0.0, 1.0)
        # Water-filling: only adjust non-clipped entries
        free = (h_new > 1e-12) & (h_new < 1.0 - 1e-12)
        if np.sum(free) > 0:
            residual = target - np.sum(h_new)
            h_new[free] += residual / np.sum(free)
            h_new = np.clip(h_new, 0.0, 1.0)
        h = h_new
    return h

def make_constraint(n):
    target = n / 2.0
    def constr(h):
        return np.sum(h) - target
    return constr
