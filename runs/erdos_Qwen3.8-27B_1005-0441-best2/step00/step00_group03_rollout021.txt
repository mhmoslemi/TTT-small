    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to add diff total; add to elements with most room
        order = np.argsort(h)  # smallest first (most room to grow)
        remaining = diff
        for i in order:
            add = min(remaining, 1.0 - h[i])

            remaining -= add

        for i in order:
            sub = min(remaining, h[i])
            h[i] -= sub
            remaining -= sub
            if remaining < 1e-14:
                break
    return h

    if abs(diff) < 1e-14:
        return h

            h[i] += add
            remaining -= add

                break

            sub = min(remaining, h[i])
            h[i] -= sub
            remaining -= sub

def compute_soft_gradient(h, alpha):
