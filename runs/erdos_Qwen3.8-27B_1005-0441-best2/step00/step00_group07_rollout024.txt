import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_objective(n, alpha, nu):
    dx = 2.0 / n
    target = n / 2.0
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted = np.clip(shifted, -500, 0)
        smooth = (m + np.log(np.exp(shifted).sum()) / alpha) * dx
        penalty = nu * (np.sum(h) - target) ** 2
        return smooth + penalty
    return obj

def optimize_h(h0, n, max_iters, time_limit):
    dx = 2.0 / n
    target = n / 2.0
    h0 = project_feasible(h0.copy(), n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    bounds = [(0.0, 1.0)] * n
    t0 = time.time()
    
    # Progressive smoothing
    alphas = [10, 50, 100, 200, 500, 1000, 2000, 5000]
    for alpha in alphas:
        if time.time() - t0 > time_limit:
            break
        nu = 1.0
        for nu in [1.0, 10.0, 100.0]:
            if time.time() - t0 > time_limit:
                break
            obj = make_objective(n, alpha, nu)
            res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-10, 'gtol': 1e-8})
            h_new = project_feasible(res.x, n)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
    return best_h, best_c5

    h_cur = project_feasible(h_cur, n)
    best_h = h_cur.copy()
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t_start

    # Stage 1: Progressive L-BFGS-B smoothing
    alphas = [5, 15, 40, 100, 300, 800, 2000, 5000]
    for alpha in alphas:
        if elapsed() > budget_s * 0.45:
            break
        for nu in [1.0, 5.0, 20.0, 100.0]:
            if elapsed() > budget_s * 0.45:
                break
            h_out = optimize_lbfgsb(best_h, n, alpha, nu, maxiter=200,
                                    t_limit=budget_s * 0.45)
            c5 = compute_c5(h_out)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_out.copy()

    # Stage 2: SLSQP refinement
    if elapsed() < budget_s * 0.6:
        h_out = optimize_slqp(best_h, n, maxiter=400)
        c5 = compute_c5(h_out)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_out

    # Stage 3: More L-BFGS-B after SLSQP
    if elapsed() < budget_s * 0.7:
        for alpha in [200, 1000, 5000]:
            if elapsed() > budget_s * 0.7:
                break
            h_out = optimize_lbfgsb(best_h, n, alpha, 50.0, maxiter=200,
                                    t_limit=budget_s * 0.7)
            c5 = compute_c5(h_out)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_out

    # Stage 4: Perturbation restarts
    if elapsed() < budget_s * 0.82:
        for trial in range(15):
            if elapsed() > budget_s * 0.82:
                break
            scale = 0.01 + 0.015 * trial
            h_pert = best_h + np.random.randn(n) * scale
            h_pert = project_feasible(h_pert, n)
            h_out = optimize_lbfgsb(h_pert, n, 200, 20.0, maxiter=100,
                                    t_limit=budget_s * 0.82)
            c5 = compute_c5(h_out)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_out.copy()
            h_out2 = optimize_slqp(h_out, n, maxiter=200)
            c52 = compute_c5(h_out2)
            if c52 < best_c5:
                best_c5 = c52
                best_h = h_out2.copy()

    # Stage 5: Try upscaling
    if elapsed() < budget_s * 0.92:
        for factor in [2, 3, 4]:
            n_new = n * factor
            if n_new > 800 or elapsed() > budget_s * 0.92:
                break
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_new, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = project_feasible(h_up, n_new)
            done = False
            for alpha in [50, 500, 3000]:
                if elapsed() > budget_s * 0.92 or done:
                    break
                h_out = optimize_lbfgsb(h_up, n_new, alpha, 20.0,
                                        maxiter=min(200, n_new // 2),
                                        t_limit=budget_s * 0.92)
                c5 = compute_c5(h_out)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_out.copy()
                    n = n_new
            if n_new > n:
                n = n_new

    # Final SLSQP polish
    if elapsed() < budget_s - 10:
        h_out = optimize_slqp(best_h, len(best_h), maxiter=300)
        c5 = compute_c5(h_out)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_out

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
