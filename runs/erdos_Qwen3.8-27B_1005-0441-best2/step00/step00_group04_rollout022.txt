    n = len(v)
    # Find lambda such that sum(clip(v - lambda, 0, 1)) = target_sum
    lo, hi = v.min() - 1, v.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(v - mid, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) / 2, 0, 1)
    # Fine-tune: adjust the nearest-to-boundary elements
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-10:
        # Adjust elements not at boundary
        interior = (proj > 0.01) & (proj < 0.99)
        if interior.sum() > 0:
            proj[interior] += diff / interior.sum()
            proj = np.clip(proj, 0, 1)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def project_feasible(v, target_sum):
    n = len(v)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(v - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-12:
        interior = (proj > 1e-6) & (proj < 1.0 - 1e-6)
        cnt = int(interior.sum())

def compute_gradient(h, top_m=20):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
