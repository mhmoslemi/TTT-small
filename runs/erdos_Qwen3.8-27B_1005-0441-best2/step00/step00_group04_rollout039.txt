def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(np.sum(h))
    if abs(s - target) < 1e-12:
        return h
    if s > target:
        excess = s - target
        mask = h > 1e-12
        tot = float(np.sum(h[mask]))
        if tot > 0:
            h[mask] -= excess * h[mask] / tot
    else:
        deficit = target - s
        mask = h < 1.0 - 1e-12
        tot = float(np.sum(1.0 - h[mask]))
        if tot > 0:
            h[mask] += deficit * (1.0 - h[mask]) / tot
    return np.clip(h, 0.0, 1.0)

    best_c5 = compute_c5(best_h, n)
    for _ in range(max_rounds):
        improved = False

            best_local = best_c5
            best_dv = 0.0
