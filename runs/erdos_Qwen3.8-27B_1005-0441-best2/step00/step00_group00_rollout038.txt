        x = np.linspace(0.0, 2.0, n_alt)
        for freq in [1, 2, 3, 4, 6]:
            if time.time() > d2:
                break
            phase = rng.rand() * 2.0 * np.pi
            amp = 0.35 + 0.1 * rng.rand()
            h0 = 0.5 + amp * np.sin(freq * np.pi * x * 0.5 + phase)
            h0 = project_feasible(h0, n_alt)
            d_sub = min(d2, time.time() + 25.0)
            h, c5 = optimize_cont(h0, d_sub, rng)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
        # Also try binary for even n
        if n_alt % 2 == 0 and time.time() < d2:
            h_rand = project_feasible(0.5 + 0.49 * rng.randn(n_alt), n_alt)
            order = np.argsort(h_rand)[::-1]
            hb = np.zeros(n_alt)
            hb[order[:n_alt // 2]] = 1.0
            d_sub = min(d2, time.time() + 30.0)
            h2, c52 = optimize_bin(hb, d_sub, rng)
            if c52 < best_c5:
                best_c5, best_h = c52, h2.copy()

    # Phase 3: Polish the best solution found
    if best_h is not None and time.time() < deadline:
        np_best = len(best_h)
        # Continuous polish
        h_pol, c5_pol = optimize_cont(best_h, min(deadline, time.time() + 300.0), rng)
        if c5_pol < best_c5:
            best_c5, best_h = c5_pol, h_pol.copy()
        # Binary polish if even
        if np_best % 2 == 0 and time.time() < deadline:
            order = np.argsort(best_h)[::-1]
            hb = np.zeros(np_best)
            hb[order[:np_best // 2]] = 1.0
            h3, c53 = optimize_bin(hb, deadline, rng)
            if c53 < best_c5:
                best_c5, best_h = c53, h3.copy()

    # Final validation
    if best_h is not None:
        final_c5 = compute_c5(best_h)
        if final_c5 < best_c5:
            best_c5 = final_c5

    if best_h is None:
        # Fallback: constant 0.5
        n_fb = 50
        best_h = np.full(n_fb, 0.5)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
