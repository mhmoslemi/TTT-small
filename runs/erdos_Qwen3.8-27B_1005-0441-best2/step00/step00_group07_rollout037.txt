import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def all_c_values(h):
    n = len(h)
    c = np.empty(n)
    for m in range(n):
        c[m] = np.dot(h[:n - m], 1.0 - h[m:])
    return c

def smooth_max_grad_weights(c, s):
    c_shift = c - np.max(c)
    e = np.exp(s * c_shift)
    return e / np.sum(e)

def compute_gradient(x, s):
    n = len(x)
    c = all_c_values(x)
    w = smooth_max_grad_weights(c, s)
    grad = np.zeros(n)
    for m in range(n):
        wm = w[m]
        if wm < 1e-14:
            continue
        grad[:n - m] += wm * (1.0 - x[m:])
        grad[m:] += wm * (-x[:n - m])
    return grad * (2.0 / n)

def project_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h
