        h_new = h.copy()
        r = rng.random()
        if r < 0.40:
            sigma = 0.004 + 0.025 * min(1.0, T / 0.03)
            h_new += rng.normal(0, sigma, n)
        elif r < 0.70:
            nf = max(1, int(n * rng.uniform(0.02, 0.12)))
            idx = rng.choice(n, nf, replace=False)
            h_new[idx] = 1.0 - h_new[idx]
        elif r < 0.88:
            i, j = rng.choice(n, 2, replace=False)
            a = rng.uniform(0.03, 0.35)
            h_new[i] -= a
            h_new[j] += a
        else:
            i, j = rng.choice(n, 2, replace=False)
            h_new[i] = 1.0 - h_new[i]
            h_new[j] = 1.0 - h_new[j]
        h_new = repair(h_new, n)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best = h_new.copy()
            best_c5 = c5_new
            no_imp = 0
        else:
            no_imp += 1
        if c5_new <= c5 or (T > T_min and rng.random() < np.exp(-(c5_new - c5) / T)):
            h = h_new
            c5 = c5_new
        T = max(T_min, T * alpha)
        if no_imp > 4000 and T > 0.005:
            T = 0.012
            no_imp = 0
    return best, best_c5

def local_2opt(h, n, dx, t_end, rng):
    h = h.copy()
    c5 = compute_c5(h, dx)
    best = h.copy()
    best_c5 = c5
    total_imp = 0
    while time.time() < t_end and total_imp < 500:
        round_imp = 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for _ in range(8000):
            if time.time() >= t_end:
                break
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            c5_new = compute_c5(h, dx)
            if c5_new < c5 - 1e-12:
                c5 = c5_new
                ones = ones[np.where(ones != i)]
                zeros = zeros[np.where(zeros != j)]
                round_imp += 1
                if c5 < best_c5:
                    best = h.copy()
                    best_c5 = c5
            else:
                h[i] = 1.0
                h[j] = 0.0
        total_imp += round_imp
        if round_imp == 0:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    t_end = t0 + budget_s - 4

    try:
        h0 = np.asarray(initial_h_values, dtype=float)
    except (NameError, AttributeError):
        h0 = np.zeros(42)
        h0[::2] = 1.0

    n = len(h0)
    dx = 2.0 / n
    h_start = repair(h0.copy(), n)

    # Phase 1: SA from initial
    t_phase1 = min(t_end, t0 + budget_s * 0.45)
    best_h, best_c5 = sa_loop(h_start, n, dx, t_phase1, rng)
    h_b = binarize(best_h, n)
    c5_b = compute_c5(h_b, dx)
    if c5_b < best_c5:
        best_h, best_c5 = h_b, c5_b
    t_local1 = min(t_end, time.time() + 25)
    best_h, best_c5 = local_2opt(best_h, n, dx, t_local1, rng)

    # Phase 2: Random restarts
    n_restarts = 12
    for ri in range(n_restarts):
        if time.time() >= t_end:
            break
        h_rand = np.zeros(n)
        idx = rng.permutation(n)[:n // 2]
        h_rand[idx] = 1.0
        c5_r = compute_c5(h_rand, dx)
        if c5_r < best_c5:
            best_h, best_c5 = h_rand, c5_r
        frac = max(0.05, (t_end - time.time()) / max(1, n_restarts - ri))
        t_sub = min(t_end, time.time() + budget_s * frac * 0.5)
        h_opt, c5_opt = sa_loop(h_rand, n, dx, t_sub, rng)
        h_b2 = binarize(h_opt, n)
        c5_b2 = compute_c5(h_b2, dx)
        if c5_b2 < c5_opt:
            h_opt, c5_opt = h_b2, c5_b2
        t_loc = min(t_end, time.time() + 8)
        h_opt, c5_opt = local_2opt(h_opt, n, dx, t_loc, rng)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Final polish
    if time.time() < t_end:
        t_fin = min(t_end, time.time() + 15)
        best_h, best_c5 = local_2opt(best_h, n, dx, t_fin, rng)

    best_h = repair(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n
