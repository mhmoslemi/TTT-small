            if time.time() > end_t:
                break
            _, g = smooth_max_obj_grad(h, tau)
            h_new = project_h(h - lr * g, target)
            nc5 = compute_c5(h_new)
            if nc5 < c5 - 1e-13:
                h = h_new
                c5 = nc5
                lr = min(lr * 1.04, 12.0 / n)
                stuck = 0
            else:
                lr *= 0.6
                stuck += 1
                if stuck > 100 or lr < 1e-10:
                    break
    return h, c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    budget_end = t0 + budget_s - 25

    cands = []
    try:
        h0 = np.asarray(initial_h_values, dtype=float)
        if len(h0) > 0:
            cands.append(project_h(h0, len(h0) / 2.0))
    except Exception:
        pass

    for ns in (40, 60, 70, 90):
        cands.append(project_h(rng.uniform(0, 1, ns), ns / 2.0))

    best_h = None
    best_c5 = np.inf
    for hc in cands:
        c = compute_c5(hc)
        if c < best_c5:
            best_c5 = c
            best_h = hc.copy()

    cands.sort(key=compute_c5)
    n_cands = len(cands)

    for ci, hc in enumerate(cands):
        if time.time() > budget_end:
            break
        remaining = budget_end - time.time()
        alloc = max(15.0, remaining / (n_cands - ci + 2))
        end_phase = min(budget_end, time.time() + alloc)

        h_o, c5_o = optimize_phase(hc, len(hc) / 2.0, end_phase, [20.0, 60.0, 150.0])
        if c5_o < best_c5:
            best_c5 = c5_o
            best_h = h_o.copy()

        for _ in range(3):
            if time.time() > budget_end:
                break
            rem = budget_end - time.time()
            if rem < 10:
                break
            hp = project_h(h_o + 0.12 * rng.standard_normal(len(h_o)), len(h_o) / 2.0)
            h_p, c5_p = optimize_phase(hp, len(h_p) / 2.0, min(budget_end, time.time() + rem * 0.5),
                                        [50.0, 120.0])
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()
                h_o = h_p.copy()

    h_f = project_h(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(h_f)
    return h_f, final_c5, len(h_f)
