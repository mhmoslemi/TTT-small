from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(vals, alpha):
    m = np.max(vals)
    return float(m + (1.0/alpha) * np.log(np.sum(np.exp(alpha * (vals - m)))))

def project_feasible(h, target_sum):
    # Project onto {0<=h<=1, sum(h)=target_sum}
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute among non-bound variables
        free = (h > 1e-6) & (h < 1.0 - 1e-6)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def eval_objective(h, alpha):
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')
    vals = corr * dx
    return smooth_max(vals, alpha)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(vals, alpha):
    m = np.max(vals)
    return float(m + (1.0 / alpha) * np.log(np.sum(np.exp(alpha * (vals - m)))))

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        nfree = int(np.sum(free))
        if nfree > 0:
            h[free] += diff / nfree
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_smo(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    return smooth_max(vals, alpha)

def sum_con(h):
    return float(np.sum(h) - len(h) / 2.0)

def sum_con_jac(h):
    return np.ones(len(h))

def make_kronecker(n, alpha, rng):
    phase = rng.rand()
    x = (np.arange(n) + phase) / n * alpha
    h = np.where(x % 1.0 < 0.5, 1.0, 0.0)
    return project_feasible(h, n / 2.0)
