    def c5_eval(h):
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode='full')
        return float(np.max(corr) * dx)

    def project(h, n):
        h = h.copy()
        for _ in range(300):
            diff = n / 2.0 - np.sum(h)
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                mask = h < 1.0 - 1e-15
            else:
                mask = h > 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        return h

    def smooth_grad(h, p):
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode='full')
        cs = np.maximum(corr, 1e-12)
        sp = np.sum(cs ** p)
        factor = dx * (sp ** (1.0 / p - 1.0))

        for m in range(n):

            i2 = m - shifts

            v2 = (i2 >= 0) & (i2 < n)

            t2[v2] = h[i2[v2]]
