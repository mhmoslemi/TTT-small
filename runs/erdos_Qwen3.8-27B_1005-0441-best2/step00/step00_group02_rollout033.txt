def compute_grad_weighted(h, beta, n_pts):
    corr = np.correlate(h, 1-h, mode='full')
    # Softmax weights
    corr_shifted = corr - corr.max()
    weights = np.exp(beta * corr_shifted)
    weights /= weights.sum()
    
    grad = np.zeros(n_pts)
    for k, w in enumerate(weights):
        if w < 1e-10:
            continue
        shift = k - (n_pts - 1)
        lo = max(0, -shift)
        hi = min(n_pts, n_pts - shift)
        if lo < hi:

        hi2 = min(n_pts, n_pts + shift)
        if lo2 < hi2:

    return grad

grad = corr_of_w_with_(1-h) - corr_of_w_with_h (approximately, with appropriate alignment)

Actually, more precisely:
grad[i] = Σ_{s: i+s∈[0,n)} w[s] * (1-h[i+s]) - Σ_{s: i-s∈[0,n)} w[s] * h[i-s]

The first sum: let j = i+s, so s = j-i. Sum over j where j∈[0,n) and s=j-i is a valid lag:
= Σ_j (1-h[j]) * w[j-i] for valid (j-i)

This is a convolution of (1-h) with w, but with a flip! So it's actually a correlation.

Specifically, if we define w_ext[s] for s in [-(n-1), n-1], then:
grad[i] = Σ_j (1-h[j]) * w_ext[j-i] - Σ_j h[j] * w_ext[i-j]

The first is (1-h) correlated with w_ext, the second is h correlated with w_ext (with sign flip).

Hmm, this is getting complicated to vectorize cleanly. For n up to a few hundred, the O(n²) loop is fine in Python with numpy operations inside.

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    n = len(h)
    h = h.copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        h = h + (target_sum - s) / n
    return np.clip(h, 0.0, 1.0)

def compute_grad(h, beta, n_pts):
    """Compute weighted subgradient of max overlap"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights for emphasis on max
    corr_norm = corr - corr.max()
    weights = np.exp(beta * corr_norm)
    wsum = weights.sum()
    if wsum > 0:
        weights = weights / wsum
    else:
        weights = np.zeros_like(corr)
        weights[np.argmax(corr)] = 1.0
    
    grad = np.zeros(n_pts)
    for k in range(len(weights)):
        w = weights[k]
        if w < 1e-12:
            continue
        shift = k - (n_pts - 1)
        # First part: (1-h[i+shift]) for i where both valid
        lo = max(0, -shift)
        hi = min(n_pts, n_pts - shift)
        if lo < hi:
            grad[lo:hi] += w * (1.0 - h[lo+shift:hi+shift])
        # Second part: -h[i-shift] for i where both valid
        lo2 = max(0, shift)
        hi2 = min(n_pts, n_pts + shift)
        if lo2 < hi2:
            grad[lo2:hi2] += w * (-h[lo2-shift:hi2-shift])
    
    return grad

    n_pts = len(h_start)

    best_c5 = compute_c5(h)

            h_new = project_h(h_new, target_sum)
            new_c5 = compute_c5(h_new)
            
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
                h = h_new
                patience = 0
                alpha = min(alpha * 1.01, 0.2)
            else:

                    h = project_h(h, target_sum)

                    alpha = 0.05

                h = project_h(h, target_sum)
