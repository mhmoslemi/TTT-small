```python
def compute_c5_vector(h):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(overlaps), overlaps

def compute_grad(h):
    """Compute weighted gradient of max overlap using softmax."""
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
    # Softmax weights
    t = 0.3  # temperature
    shifted = overlaps - np.max(overlaps)
    w = np.exp(shifted / t)
    w /= np.sum(w)
    # Gradient: for each shift s, grad_s[j] = (1-h[j+s])*I - h[j-s]*I
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-12:
            continue
        s = idx - (n - 1)
        # valid j: 0<=j<n and 0<=j+s<n
        j_min = max(0, -s)
        j_max = min(n, n - s)
        for j in range(j_min, j_max):
            grad[j] += w[idx] * (1.0 - h[j + s])
            grad[j] -= w[idx] * h[j - s]
    grad *= dx
    return grad
```

```python
import numpy as np
import time

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_grad_softmax(h, temp=0.4):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
    shifted = overlaps - np.max(overlaps)
    w = np.exp(shifted / temp)
    w /= np.sum(w)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-12:
            continue
        s = idx - (n - 1)
        if s >= 0:
            grad[:n-s] += wi * (1.0 - h[s:])
            grad[s:] -= wi * h[:n-s]
        else:
            ss = -s
            grad[ss:] += wi * (1.0 - h[:n-ss])
            grad[:n-ss] -= wi * h[ss:]
    return grad * dx

def compute_grad_worst(h, top_k=5):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
    top_idx = np.argsort(overlaps)[::-1][:top_k]
    grad = np.zeros(n)
    for idx in top_idx:
        s = idx - (n - 1)
        if s >= 0:
            grad[:n-s] += (1.0 - h[s:])
            grad[s:] -= h[:n-s]
        else:
            ss = -s
            grad[ss:] += (1.0 - h[:n-ss])
            grad[:n-ss] -= h[ss:]
    return grad * dx

    if remaining > 20 and best_overall_h is not None:
        n_best = best_overall_n
        h_polished, c5_polished = optimize_one(best_overall_h, n_best, time_limit=remaining, mode="softmax", seed=seed + 9999)
        if c5_polished < best_overall_c5:
            best_overall_c5 = c5_polished
            best_overall_h = h_polished

    # Final verification
    if best_overall_h is not None:
        n = best_overall_n
        c5 = np.max(compute_overlaps(best_overall_h))
        target = n / 2.0
        best_overall_h = project_onto(best_overall_h, target)
        c5 = np.max(compute_overlaps(best_overall_h))
    else:
        n = 95
        best_overall_h = make_initial(n, rng)
        c5 = np.max(compute_overlaps(best_overall_h))
        best_overall_n = n

    return (best_overall_h, c5, best_overall_n)
```
