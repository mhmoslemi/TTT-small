d/dh[q] of first sum: 1 if q = n+p for some valid n, i.e., n = q-p, valid if 0 <= q-p < n.
d/dh[q] of second sum:

  - from h[n] term (q = n, n = q): -h[q+p] if 0 <= q+p < n

= (1-h[q-p])*I[0<=q-p<n] - h[q+p]*I[0<=q+p<n]

Where p = j - (n-1) and j ranges from 0 to 2n-2, so p ranges from -(n-1) to (n-1).

For L-BFGS-B, I need a smooth objective. I'll use the log-sum-exp:
f_β(h) = (1/β) * log(sum_j exp(β * out[j] * dx))

    """Return all overlap values (not just max)"""
    return np.correlate(h, 1-h, mode="full") * dx

def compute_c5(h, dx):
    """Return the C5 value (max overlap)"""
    return compute_c5_values(h, dx).max()

def fix_sum(h, target, n):
    """Fix sum of h to target, keeping 0<=h<=1"""
    h = h.copy()
    h = np.clip(h, 0, 1)
    diff = target - np.sum(h)
    if diff > 0:
        # Need to increase some entries
        mask = h < 1
        if mask.any():
            add = diff / mask.sum()
            h[mask] = np.minimum(h[mask] + add, 1.0)
            # If still need more, distribute to all
            diff2 = target - np.sum(h)
            if diff2 > 0:
                h += diff2 / n
            h = np.clip(h, 0, 1)
    elif diff < 0:
        mask = h > 0
        if mask.any():
            sub = -diff / mask.sum()
            h[mask] = np.maximum(h[mask] - sub, 0.0)
            diff2 = target - np.sum(h)
            if diff2 < 0:
                h += diff2 / n
            h = np.clip(h, 0, 1)
    # Final exact fix
    h += (target - np.sum(h)) / n
    h = np.clip(h, 0, 1)
    return h

def grad_smoothed(h, dx, beta):
    """Analytical gradient of log-sum-exp smoothed C5"""
    n = len(h)
    c = np.correlate(h, 1-h, mode="full") * dx
    # Weights
    beta_c = beta * c
    beta_c -= beta_c.max()
    w = np.exp(beta_c)
    w /= w.sum()
    # For each lag j (0 to 2n-2), p = j - (n-1)
    # d(out[j])/dh[q] = (1-h[q-p])*I[0<=q-p<n] - h[q+p]*I[0<=q+p<n]
    # Sum over j with weights w[j]
    grad = np.zeros(n)
    for j in range(2*n-1):
        p = j - (n-1)
        if abs(w[j]) < 1e-15:
            continue
        # (1-h[q-p]) for q such that 0<=q-p<n, i.e., p<=q<p+n
        q_lo = max(0, p)
        q_hi = min(n, p+n)
        if q_lo < q_hi:
            grad[q_lo:q_hi] += w[j] * (1 - h[q_lo-p:q_hi-p])
        # -h[q+p] for q such that 0<=q+p<n, i.e., -p<=q<n-p
        q_lo2 = max(0, -p)
        q_hi2 = min(n, n-p)
        if q_lo2 < q_hi2:
            grad[q_lo2:q_hi2] -= w[j] * h[q_lo2+p:q_hi2+p]
    return dx * grad

        beta_c = beta * c
        beta_c -= beta_c.max()
        lse = (c.max() + (1.0/beta) * math.log(np.sum(np.exp(beta_c))))
        penalty = penalty_weight * (np.sum(h) - target) ** 2
        return lse + penalty
    return obj

        g = grad_smoothed(h, dx, beta)
        g += 2 * penalty_weight * (np.sum(h) - target)
        return g
    return grad

            best_h = h.copy()
        else:

import numpy as np
import scipy.optimize as so
import time
import math


def compute_c5_values(h, dx):
    return np.correlate(h, 1 - h, mode="full") * dx


def compute_c5(h, dx):
    return compute_c5_values(h, dx).max()


def fix_sum(h, target, n):
    h = np.clip(h, 0, 1).copy()
    h += (target - np.sum(h)) / n
    h = np.clip(h, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    return h


def _obj(h, dx, target, beta, pw):
    c = np.correlate(h, 1 - h, mode="full") * dx
    bc = beta * c
    bc_max = bc.max()
    lse = c.max() + (1.0 / beta) * math.log(np.sum(np.exp(bc - bc_max)))
    pen = pw * (np.sum(h) - target) ** 2
    return lse + pen


def _grad(h, dx, target, beta, pw):
    n = len(h)
    c = np.correlate(h, 1 - h, mode="full") * dx
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        if w[j] < 1e-14:
            continue
        p = j - (n - 1)
        q_lo = max(0, p)
        q_hi = min(n, p + n)
        if q_lo < q_hi:
            grad[q_lo:q_hi] += w[j] * (1.0 - h[q_lo - p:q_hi - p])
        q_lo2 = max(0, -p)
        q_hi2 = min(n, n - p)
        if q_lo2 < q_hi2:
            grad[q_lo2:q_hi2] -= w[j] * h[q_lo2 + p:q_hi2 + p]
    grad *= dx
    grad += 2.0 * pw * (np.sum(h) - target)
    return grad


def _local_search(h, dx, target, n, rng, n_iter):
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    for i in range(n_iter):
        step = 0.06 * max(0.1, 1.0 - i / max(1, n_iter))
        j = rng.randint(n)
        h[j] += rng.uniform(-step, step)
        h = fix_sum(h, target, n)
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        else:
            h = best_h.copy()
    return best_h, best_c5


def _flip_search(h0, dx, target, n, rng, n_iter):
    """0-1 local search: flip pairs to maintain sum"""
    h = h0.copy()
    h = (h > 0.5).astype(float)
    h = fix_sum(h, target, n)
    # Ensure exactly n/2 ones if possible
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) != n // 2:
        h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    for i in range(n_iter):
        if len(ones) == 0 or len(zeros) == 0:
            break
        j = ones[rng.randint(len(ones))]
        k = zeros[rng.randint(len(zeros))]
        h[j], h[k] = h[k], h[j]
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[j], h[k] = h[k], h[j]
    return best_h, best_c5
