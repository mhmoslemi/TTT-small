from scipy.optimize import minimize

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_h(h, n):
    """Project to {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Binary search for offset
    lo, hi = -1.0, 1.0
    if diff > 0:
        lo, hi = 0.0, 1.0
    else:
        lo, hi = -1.0, 0.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_test = np.clip(h + mid, 0.0, 1.0)
        if np.sum(h_test) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Final adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        n_free = np.sum((h > 0) & (h < 1))
        if n_free > 0:
            mask = (h > 0) & (h < 1)
            h[mask] += diff / n_free
            h = np.clip(h, 0.0, 1.0)
    return h

    m = np.max(values)
    if T * m > 500:
        return m
    return m + np.log(np.sum(np.exp(T * (values - m)))) / T

    n = len(h)
    A = np.zeros(n)
    one_minus_h = 1.0 - h
    for k in range(len(p)):
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                # Term 1: (1-h[m-s]) for m in [s, n-1]
                A[s:] += p[k] * one_minus_h[:n-s]
                # Term 2: -h[m+s] for m in [0, n-1-s]
                A[:n-s] -= p[k] * h[s:]
        else:
            s2 = -s
            if s2 < n:
                # Term 1: (1-h[m+s2]) for m in [0, n-1-s2]
                A[:n-s2] += p[k] * one_minus_h[s2:]
                # Term 2: -h[m-s2] for m in [s2, n-1]
                A[s2:] -= p[k] * h[:n-s2]
