def grad_for_lag(h, lag, n):
    """Compute gradient of the lag-th correlation value w.r.t. h."""
    grad = np.zeros(n)
    if lag == 0:
        # f = sum h[i](1-h[i]) = sum h[i] - h[i]^2
        # df/dh[j] = 1 - 2*h[j]
        grad[:] = 1.0 - 2.0 * h

        # df/dh[j]:
        #   from h[j] in the first slot: j = i+lag, so i = j-lag, valid when 0<=j-lag and j-lag+lag<n => 0<=j<n, j>=lag
        #     contribution: (1-h[j-lag])

import numpy as np
import time

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_c5(h):
    return float(np.max(compute_corr(h)))

def grad_lag(h, lag, n):
    g = np.zeros(n)
    if lag == 0:
        g = 1.0 - 2.0 * h
    elif 0 < lag < n:
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    elif -n < lag < 0:
        lag = -lag
        g[:n - lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n - lag]
    return g

def proj(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        d = t - np.sum(h)
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    d = t - np.sum(h)
    if abs(d) > 1e-10:
        if d > 0:
            m = h < 1.0
            if m.any():
                r = 1.0 - h[m]
                h[m] += r * d / r.sum()
        else:
            m = h > 0.0
            if m.any():
                r = h[m]
                h[m] -= r * (-d) / r.sum()
    return np.clip(h, 0.0, 1.0)

    m = corr.max()
    e = np.exp(T * (corr - m))
    s = e.sum()
    if s < 1e-300:

        e[np.argmax(corr)] = 1.0
        return e
    return e / s
