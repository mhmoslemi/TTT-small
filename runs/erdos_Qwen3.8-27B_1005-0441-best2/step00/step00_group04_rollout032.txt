import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-15
    else:
        mask = h > 1e-15
    count = mask.sum()
    if count == 0:
        return h
    step = diff / count
    h[mask] += step
    h = np.clip(h, 0.0, 1.0)
    diff2 = target_sum - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(1.0, h[idx] + diff2)
        else:
            idx = np.argmin(h)
            h[idx] = max(0.0, h[idx] + diff2)
    return h

def sa_loop(h, best_h, best_c5, n, target_sum, T_init, cooling, max_stall,
            eps_lo, eps_hi, max_moves, time_limit_fn, np_random):

    stall = 0
    min_T = 1e-10
    while time_limit_fn():
        h_new = h.copy()
        n_moves = np_random.randint(1, max_moves + 1)
        for _ in range(n_moves):
            i = np_random.randint(n)
            j = np_random.randint(n)
            if i == j:
                continue
            eps = np_random.uniform(eps_lo, eps_hi)
            if np_random.random() > 0.5:
                move = min(eps, h_new[i], 1.0 - h_new[j])
            else:
                move = min(eps, h_new[j], 1.0 - h_new[i])
                i, j = j, i
            if move > 1e-12:
                h_new[i] -= move
                h_new[j] += move
        c5_new = compute_c5(h_new)
        c5_cur = compute_c5(h)
        delta = c5_new - c5_cur
        if delta <= 0 or (T > min_T and np_random.random() < np.exp(-delta / T)):

                best_c5 = c5_new
                stall = 0

            h = repair_sum(h, target_sum)
            T = T_init * np_random.uniform(0.4, 1.0)
            stall = 0
