import numpy as np
import time

def project(h, S):
    """Project onto {x: 0<=x[i]<=1, sum(x)=S}"""
    h = np.clip(h, 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        p = np.clip(h - mid, 0.0, 1.0)
        if np.sum(p) > S:
            lo = mid
        else:
            hi = mid
    p = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = S - np.sum(p)
    if abs(diff) > 1e-14:
        idx = (p > 1e-10) & (p < 1.0 - 1e-10)
        if idx.any():
            p[idx] += diff / idx.sum()
            p = np.clip(p, 0.0, 1.0)
    return p

def c5val(h):
    return np.correlate(h, 1.0 - h, mode="full").max() * 2.0 / len(h)

def grad_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    if T < 1e-10:
        k_star = np.argmax(c)
        s = k_star - (n - 1)
        g = np.zeros(n)
        j = np.arange(n)
        vs = j - s
        m1 = (vs >= 0) & (vs < n)
        g[m1] = (1.0 - h[vs[m1]])
        vp = j + s
        m2 = (vp >= 0) & (vp < n)
        g[m2] -= h[vp[m2]]
        return g * dx
    else:
        mc = c.max()
        w = np.exp((c - mc) / T)
        w /= w.sum()
        g = np.zeros(n)
        j = np.arange(n)
        for k in range(len(c)):
            if w[k] < 5e-4:
                continue
            s = k - (n - 1)
            vs = j - s
            m1 = (vs >= 0) & (vs < n)
            g[m1] += w[k] * (1.0 - h[vs[m1]])
            vp = j + s
            m2 = (vp >= 0) & (vp < n)
            g[m2] -= w[k] * h[vp[m2]]
        return g * dx

    n = len(h)
    h = project(h.copy(), S)
    best = h.copy()
    best_f = c5val(best)
    lr = lr_init
    mom = np.zeros(n)
    beta = 0.85
    T = T_start
    it = 0
    patience = 0
    t0 = time.time()
    
    while time.time() - t0 < time_limit:
        it += 1
        # Anneal T
        frac = min(1.0, it / 10000)
        T = T_start * (T_end / T_start) ** frac
        if T < T_end:
            T = T_end
        
        g = grad_c5(h, T)
        mom = beta * mom + (1 - beta) * g
        
        # Backtracking line search
        step = lr
        improved = False
        for attempt in range(8):
            h_new = project(h - step * mom, S)
            f_new = c5val(h_new)
            if f_new < best_f or f_new < c5val(h) * 0.999:
                h = h_new
                if f_new < best_f:
                    best_f = f_new
                    best = h_new.copy()
                    patience = 0
                else:
                    patience += 1
                step = min(step * 1.05, lr_init)
                improved = True
                break
            step *= 0.5

            patience += 5
            step = lr * 0.5

        if patience > 200 and patience % 100 == 0:

            perturb = np.random.randn(n_perturb) * 0.05
            h[idx] += perturb
            h = project(h, S)
            patience = 0

</parameter>"""
