import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr * dx

def project_feasible(h, target_sum):
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_corr_k(h, k, n):
    lo = max(0, n - 1 - k)
    hi = min(n - 1, 2 * n - 2 - k)
    m = np.arange(n)
    grad = np.zeros(n)
    i1 = m - k + n - 1
    mask1 = (i1 >= lo) & (i1 <= hi)
    grad[mask1] += 1.0 - h[i1[mask1]]
    mask2 = (m >= lo) & (m <= hi)
    j2 = m + k - n + 1
    mask2 &= (j2 >= 0) & (j2 <= n - 1)
    grad[mask2] -= h[j2[mask2]]
    return grad

def _pgd_step(h, target_sum, eps, step_init):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    w = np.exp((corr - cmax) / eps)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-10:
            g += w[k] * grad_corr_k(h, k, n)
    g *= dx
    a = step_init
    for _ in range(35):
        h_new = project_feasible(h - a * g, target_sum)
        c5_new, _ = compute_c5(h_new)
        if c5_new < np.max(corr) * dx - 1e-12:
            return h_new, c5_new, True
        a *= 0.6
    return h, np.max(corr) * dx, False

def _slsqp_obj(x):
    c, _ = compute_c5(x)
    return c

def _slsqp_con(x):
    return x.sum() - len(x) / 2.0

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(80, 0.5)
    n = len(h)
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target_sum) > 1e-9:
        h = project_feasible(h, target_sum)

    best_h = h.copy()
    best_c5, _ = compute_c5(h)

    # Phase 1: Softmax PGD
    eps = 0.12
    step_init = 0.3
    stall = 0
    n_restarts = 0

    while time.time() - t0 < budget_s * 0.65:
        h, c5_now, ok = _pgd_step(h, target_sum, eps, step_init)
        if ok:
            if c5_now < best_c5:
                best_h = h.copy()
                best_c5 = c5_now
            stall = 0
        else:
            stall += 1
            if stall > 80:
                # Explore: perturb from best
                n_restarts += 1
                h = project_feasible(best_h + np.random.randn(n) * 0.08, target_sum)
                stall = 0
                eps = max(eps * 0.8, 0.015)
                step_init = max(step_init * 0.8, 0.02)
                if step_init < 0.03:
                    eps = 0.06
                    step_init = 0.15

    # Phase 2: Target worst lags directly
    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        for _ in range(30):
            if time.time() - t0 > budget_s - 15:
                break
            _, corr_dx = compute_c5(best_h)
            top_k = np.argsort(corr_dx)[::-1][:6]
            improved = False
            for k in top_k:
                gk = grad_corr_k(best_h, k, n)
                for scale in [0.06, 0.03, 0.015, 0.008]:
                    h_t = project_feasible(best_h - scale * gk, target_sum)
                    c5_t, _ = compute_c5(h_t)
                    if c5_t < best_c5 - 1e-13:
                        best_h = h_t
                        best_c5 = c5_t
                        improved = True
                        break
            if not improved:
                break

    # Phase 3: SLSQP polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 20 and n <= 200:
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _slsqp_con}]
        try:
            res = minimize(_slsqp_obj, best_h.copy(), method='SLSQP',
                           bounds=bnds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-10})
            if res.fun < best_c5:
                best_h = project_feasible(res.x, target_sum)
                best_c5, _ = compute_c5(best_h)
        except Exception:
            pass

    # Phase 4: Fine-tune PGD
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        eps = 0.025
        for _ in range(300):
            if time.time() - t0 > budget_s - 4:
                break

            if ok and c5x < best_c5:
                best_h = hx
                best_c5 = c5x
            else:
                break

    best_h = project_feasible(best_h, target_sum)
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, n
