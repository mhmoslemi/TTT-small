import numpy as np
from scipy.optimize import minimize
import time

_g_n = 80
_g_temp = 0.01

def _obj(h):
    dx = 2.0 / _g_n
    corr = np.correlate(h, 1.0 - h, mode="full")
    t = max(_g_temp, 1e-10)
    m = np.max(corr)
    return (m + t * np.log(np.sum(np.exp((corr - m) / t)))) * dx

def _cons_eq(h):
    return np.sum(h) - _g_n / 2.0

def _cons_eq_jac(h):
    return np.ones(_g_n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _upsample(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old, dtype=float) * 2.0 / n_old
    x_new = np.arange(n_new, dtype=float) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    s = np.sum(h_new)
    if s > 1e-12:
        h_new = h_new * (float(n_new) / 2.0) / s
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new
