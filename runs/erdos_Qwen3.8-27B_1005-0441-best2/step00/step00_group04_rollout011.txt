import numpy as np
from scipy import optimize
import time

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return corr * dx

def grad_overlap(h, shift):
    n = len(h)
    grad = np.zeros(n)
    if shift >= 0:
        s = shift
        if s < n:
            grad[s:] += (1.0 - h[:n-s])
            grad[:n-s] -= h[s:]
    else:
        s = -shift
        if s < n:
            grad[:n-s] += (1.0 - h[s:])
            grad[s:] -= h[:n-s]
    return grad

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        excess = h.sum() - target_sum
        if abs(excess) < 1e-12:
            break
        if excess > 0:
            mask = h > 1e-15
            if mask.sum() == 0:
                break
            h[mask] -= excess / mask.sum()
        else:
            mask = h < 1.0 - 1e-15
            if mask.sum() == 0:
                break
            h[mask] += (-excess) / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    return np.max(compute_overlaps(h))

            no_improve = 0

    best_h = h0.copy()
    best_c5 = c5_of(best_h)
    best_n = n0

    # Phase 1: subgradient at initial resolution
    h_o, c5_o = subgrad_optimize(h0, n0 / 2.0, deadline, max_iter=40000, top_k=5)
    if c5_o < best_c5:
        best_h, best_c5, best_n = h_o.copy(), c5_o, n0

    # Coordinate descent polish
    h_cd, c5_cd = coordinate_descent(best_h, best_n / 2.0, deadline, max_sweeps=80)
    if c5_cd < best_c5:
        best_h, best_c5 = h_cd.copy(), c5_cd

    # Phase 2: increase resolution
    for n_try in [84, 140, 200, 350, 500]:
        if time.time() > deadline - 50:
            break
        h_try = interp_h(best_h, n_try)
        h_try, c5_try = subgrad_optimize(h_try, n_try / 2.0, deadline, max_iter=15000, top_k=7)
        h_try, c5_try2 = coordinate_descent(h_try, n_try / 2.0, deadline, max_sweeps=30)
        c5_try = min(c5_try, c5_try2)
        if c5_try < best_c5:
            best_h, best_c5, best_n = h_try.copy(), c5_try, n_try

    # Phase 3: randomized restarts
    rng = np.random.RandomState(seed + 500)
    n_restarts = min(6, max(1, int((deadline - time.time()) / 40)))
    for r in range(n_restarts):
        if time.time() > deadline - 20:
            break
        h_r = best_h.copy()
        mask = rng.rand(len(h_r)) < 0.25
        h_r[mask] += rng.uniform(-0.6, 0.6, mask.sum())
        h_r = project_feasible(h_r, len(h_r) / 2.0)
        h_r, c5_r = subgrad_optimize(h_r, len(h_r) / 2.0, deadline, max_iter=12000, top_k=5)
        h_r, c5_r2 = coordinate_descent(h_r, len(h_r) / 2.0, deadline, max_sweeps=20)
        c5_r = min(c5_r, c5_r2)
        if c5_r < best_c5:
            best_h, best_c5 = h_r.copy(), c5_r

    # Final: also try a fully random binary at best_n
    if time.time() < deadline - 30:
        h_rb = random_binary(best_n, seed + 7777)
        h_rb, c5_rb = subgrad_optimize(h_rb, best_n / 2.0, deadline, max_iter=10000, top_k=5)
        if c5_rb < best_c5:
            best_h, best_c5 = h_rb.copy(), c5_rb

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_c5 = c5_of(best_h)
    return best_h, final_c5, len(best_h)
