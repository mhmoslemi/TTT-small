                r = minimize(compute_c5, h_alt, method='SLSQP', bounds=bounds_alt,
                             constraints=cons_alt, options={'maxiter': 1000, 'ftol': 1e-13})
                h_alt = project_feasible(r.x, ts_alt)
                c_alt = compute_c5(h_alt)
                if c_alt < best_c5:
                    best_h, best_c5, n = h_alt, c_alt, n_alt
            except Exception:
                pass
                n = n  # keep original n

    return best_h, best_c5, n
