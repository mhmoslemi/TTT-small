import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c * dx))


def smooth_max(values, alpha):
    av = alpha * values
    m = np.max(av)
    return (m + np.log(np.sum(np.exp(av - m)))) / alpha


def project_h(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h = np.clip(h + diff / n, 0.0, 1.0)
    return h


def obj_func(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(c * dx, alpha)


def cons_eq_func(h, n):
    return np.sum(h) - n / 2.0
