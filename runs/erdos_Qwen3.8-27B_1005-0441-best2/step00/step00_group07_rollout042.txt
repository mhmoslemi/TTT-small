    lo1 = max(0, k)

    if lo1 < hi1:
        cnt = hi1 - lo1
        g[lo1:hi1] += 1.0 - h[max(0, -k):max(0, -k) + cnt]
    lo2 = max(0, -k)
    hi2 = min(n, n - k)

    return g
