```python
def evaluate_binary(h):
    """Compute C5 for binary h."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1-h, mode='full')) * dx

def local_search(h, time_limit=10.0):
    n = len(h)
    h = h.copy()
    best = evaluate_binary(h)
    best_h = h.copy()
    
    start = time.time()
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1-h, mode='full')
        cur_max = corr.max()
        
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        
        improved = False
        # Greedy: find best single swap
        best_improvement = 0
        best_i, best_j = -1, -1
        
        for i in ones:
            for j in zeros:
                h[i], h[j] = h[j], h[i]
                new_max = np.max(np.correlate(h, 1-h, mode='full'))
                h[i], h[j] = h[j], h[i]
                if cur_max - new_max > best_improvement:
                    best_improvement = cur_max - new_max
                    best_i, best_j = i, j
        
        if best_i >= 0:
            h[best_i], h[best_j] = h[best_j], h[best_i]
            improved = True
        
        if not improved:
            # Random perturbation
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
        
        cur = evaluate_binary(h)
        if cur < best:
            best = cur
            best_h = h.copy()
    
    return best_h, best
```

```python
def local_search_anti(h, time_limit=10.0):
    """Local search maintaining anti-symmetry."""
    n = len(h)
    half = n // 2
    h = h.copy()
    
    # Current correlation
    corr = np.correlate(h.astype(float), (1-h).astype(float), mode='full')
    current_max = corr.max()
    
    best_max = current_max
    best_h = h.copy()
    
    start = time.time()
    stall = 0
    
    while time.time() - start < time_limit:
        improved = False
        
        # Try all single pair flips
        best_delta_max = current_max
        best_p = -1
        
        corr = np.correlate(h.astype(float), (1-h).astype(float), mode='full')
        current_max = corr.max()
        
        for p in range(half):
            i, j = p, n-1-p
            # Swap i and j
            h[i], h[j] = h[j], h[i]
            new_corr = np.correlate(h.astype(float), (1-h).astype(float), mode='full')
            new_max = new_corr.max()
            h[i], h[j] = h[j], h[i]
            
            if new_max < best_delta_max:
                best_delta_max = new_max
                best_p = p

            i, j = best_p, n-1-best_p
            h[i], h[j] = h[j], h[i]
            current_max = best_delta_max
            stall = 0
            improved = True
        else:
            stall += 1
            # Random flip
            p = np.random.randint(half)
            h[p], h[n-1-p] = h[n-1-p], h[p]
            current_max = current_max  # approximate
        
        cm = np.max(np.correlate(h.astype(float), (1-h).astype(float), mode='full'))

                h[p], h[n-1-p] = h[n-1-p], h[p]
            stall = 0
```
