import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def grad_at_lag(h, lag):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for j in range(n):
        jp = j + lag
        jm = j - lag
        if 0 <= jp < n:
            g[j] += 1.0 - h[jp]
        if 0 <= jm < n:
            g[j] -= h[jm]
    return g * dx

def project_feasible(h, n):
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h
