import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def initial_binary(n, seed):
    rng = np.random.RandomState(seed)
    indices = rng.choice(n, size=n//2, replace=False)
    h = np.zeros(n)
    h[indices] = 1.0
    return h

def binary_steepest(h, max_time):
    n = len(h)
    t0 = time.time()
    best_c5 = compute_c5(h)
    
    while True:
        if time.time() - t0 > max_time:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_improve = 0
        best_pair = None
        
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                c5 = compute_c5(h)
                h[i] = 1
                h[j] = 0
                improve = best_c5 - c5
                if improve > best_improve:
                    best_improve = improve
                    best_pair = (i, j)
        
        if best_pair is None:
            break
        
        h[best_pair[0]] = 0
        h[best_pair[1]] = 1
        best_c5 = compute_c5(h)
    
    return h, best_c5

def simulate_anneal(h, max_time):
    n = len(h)
    t0 = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    T = 0.1
    
    while time.time() - t0 < max_time:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        h[i] = 0
        h[j] = 1
        c5_new = compute_c5(h)
        
        delta = c5_new - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
        
        T *= 0.999
    
    # Final local search
    h, c5 = binary_steepest(best_h, 0.1)
    if c5 < best_c5:
