def project_to_feasible(h, target_sum):
    """Project h onto the feasible set: 0<=h[i]<=1, sum(h)=target_sum"""
    h = np.clip(h, 0, 1)
    for _ in range(100):
        excess = np.sum(h) - target_sum
        if abs(excess) < 1e-12:
            break
        if excess > 0:
            mask = h > 0
            count = np.sum(mask)
            if count == 0:
                break
            step = excess / count
            h[mask] = np.maximum(h[mask] - step, 0)
        else:
            mask = h < 1
            count = np.sum(mask)
            if count == 0:
                break
            step = -excess / count
            h[mask] = np.minimum(h[mask] + step, 1)

    h = np.clip(h, 0, 1)
    # Tiny adjustment for floating point
    diff = target_sum - np.sum(h)

        for i in range(len(h)):
            if h[i] + diff >= 0 and h[i] + diff <= 1:

import numpy as np
import time


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        excess = np.sum(h) - target
        if abs(excess) < 1e-13:
            break
        if excess > 0:
            mask = h > 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] - excess / cnt, 0.0)
        else:
            mask = h < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] - excess / cnt, 1.0)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def c5_val(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def psocore(rng, h0, n, best, best_c5, NP, w, c1, c2, vmax, t_end):
    pos = np.zeros((NP, n))
    vel = np.zeros((NP, n))
    pb = np.zeros((NP, n))
    pb_c5 = np.full(NP, np.inf)

    pos[0] = h0
    pb[0] = h0
    pb_c5[0] = best_c5

    for i in range(1, NP):
        mag = 0.08 + 0.5 * (i / max(NP - 1, 1))
        h = h0 + rng.uniform(-mag, mag, n)
        h = project_feasible(h, n)
        pos[i] = h
        pb[i] = h
        pb_c5[i] = c5_val(h)
        if pb_c5[i] < best_c5:
            best_c5 = pb_c5[i]
            best = h.copy()

    while time.time() < t_end:
        for i in range(NP):
            r1 = rng.random(n)
            r2 = rng.random(n)
            vel[i] = w * vel[i] + c1 * r1 * (pb[i] - pos[i]) + c2 * r2 * (best - pos[i])
            np.clip(vel[i], -vmax, vmax, out=vel[i])
            pos[i] += vel[i]
            pos[i] = project_feasible(pos[i], n)
            c5i = c5_val(pos[i])
            if c5i < pb_c5[i]:
                pb_c5[i] = c5i
                pb[i] = pos[i].copy()
                if c5i < best_c5:
                    best_c5 = c5i
                    best = pos[i].copy()
    return best, best_c5
