import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, N, temp):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = corr / temp
    mx = s.max()
    e = np.exp(s - mx)
    ss = e.sum()
    vals = e / ss
    obj = (mx + np.log(ss)) * temp * dx

    nfull = 2 * N - 1
    grad = np.zeros(N)
    m = np.arange(N)
    for k in range(nfull):
        wk = vals[k]
        if wk < 1e-12:
            continue
        i_lo = max(0, k - N + 1)
        i_hi = min(N - 1, k)
        ir = (m >= i_lo) & (m <= i_hi)
        g = ir.astype(np.float64)
        jm = m - k + N - 1
        v1 = ir & (jm >= 0) & (jm < N)
        g[v1] -= h[jm[v1]]
        im2 = m + k - N + 1
        v2 = (im2 >= 0) & (im2 < N) & (im2 >= i_lo) & (im2 <= i_hi)
        g[v2] -= h[im2[v2]]
        grad += wk * g
    return obj, grad * dx

def hard_obj_grad(h, N):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    obj = float(np.max(corr) * dx)
    k = int(np.argmax(corr))
    i_lo = max(0, k - N + 1)
    i_hi = min(N - 1, k)
    m = np.arange(N)
    ir = (m >= i_lo) & (m <= i_hi)
    g = ir.astype(np.float64)
    jm = m - k + N - 1
    v1 = ir & (jm >= 0) & (jm < N)
    g[v1] -= h[jm[v1]]
    im2 = m + k - N + 1
    v2 = (im2 >= 0) & (im2 < N) & (im2 >= i_lo) & (im2 <= i_hi)
    g[v2] -= h[im2[v2]]
    return obj, g * dx

def optimize_at_n(h_start, N, t0, budget):
    h_cur = project_feasible(h_start.copy(), N)
    temps = [0.1, 0.05, 0.02, 0.01, 0.005]
    for temp in temps:
        if time.time() - t0 > budget * 0.85:
            break
        for _ in range(3):
            if time.time() - t0 > budget * 0.85:
                break
            try:
                res = minimize(
                    lambda x: smooth_obj_grad(x, N, temp),
                    h_cur, method='L-BFGS-B',
                    bounds=[(0.0, 1.0)] * N,
                    options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-10,
                             'eps': 1e-8, 'maxfun': 50000}
                )
                h_cur = project_feasible(res.x, N)
            except Exception:
                break
    return h_cur

def interp_h(h_from, n_to):
    n_from = len(h_from)
    if n_from == n_to:
        return h_from.copy()
    xf = np.linspace(0.0, 2.0, n_from, endpoint=False)
    xt = np.linspace(0.0, 2.0, n_to, endpoint=False)
    return np.interp(xt, xf, h_from)

def local_search(h, N, rng, n_iter=200):
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for _ in range(n_iter):
        idx = rng.randint(N)
        delta = rng.uniform(-0.05, 0.05)
        h_new = h_cur.copy()
        h_new[idx] = np.clip(h_new[idx] + delta, 0.0, 1.0)
        h_new = project_feasible(h_new, N)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h_cur = h_new
            c5_cur = c5_new
    return h_cur, c5_cur
