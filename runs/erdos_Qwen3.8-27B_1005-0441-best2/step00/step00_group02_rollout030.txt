            else:

def c5_of_h(h):

def to_binary(h, n):

        h_sorted = np.sort(h)[::-1]
        threshold = h_sorted[n // 2 - 1]
        h_bin = (h >= threshold).astype(float)
        # Fix count
        ones = h_bin.sum()

            # Set some borderline to 0

            h_border = h[border]
            order = np.argsort(h_border)

            border = np.where(h_bin == 0)[0]
            h_border = h[border]
            order = np.argsort(h_border)[::-1]
            for i in range(int(n / 2 - ones)):
                h_bin[border[order[i]]] = 1
        return h_bin
    else:
        # Need sum = n/2, e.g., (n-1)/2 ones + one 0.5

        h_sorted_idx = np.argsort(h)[::-1]
        h_bin = np.zeros(n)
        h_bin[h_sorted_idx[:k]] = 1.0
        h_bin[h_sorted_idx[k]] = 0.5
        return h_bin

def c5_of_h(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(np.float64).copy()
    target = n / 2.0
    diff = target - h.sum()
    for _ in range(80):
        if abs(diff) < 1e-13:
            break
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = m.sum()
        if cnt == 0:
            break
        h[m] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(h) if diff > 0 else np.argsort(-h)
        for i in order:
            if diff > 0:
                h[i] = min(1.0, h[i] + diff)
            else:
                h[i] = max(0.0, h[i] + diff)
            if abs(diff) < 1e-12:
                break
    return h

def to_binary(h, n):
    if n % 2 == 0:
        k = n // 2
        idx = np.argsort(h)[::-1]
        h_b = np.zeros(n)
        h_b[idx[:k]] = 1.0
        return h_b
    else:
        k = n // 2
        idx = np.argsort(h)[::-1]
        h_b = np.zeros(n)
        h_b[idx[:k]] = 1.0
        h_b[idx[k]] = 0.5
        return h_b

def grad_shift(h, s, n):
    g = np.zeros(n)
    if s == 0:
        return 1.0 - 2.0 * h
    if s > 0:
        g[:n - s] += 1.0 - h[s:]
        g[s:] += -h[:n - s]
    else:
        m = -s
        g[m:] += 1.0 - h[:n - m]
        g[:n - m] += -h[m:]
    return g

def subgradient_step(h, n, top_k=5, alpha=0.05):
    overlaps = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    cur = np.max(overlaps)
    top_idx = np.argsort(overlaps)[-top_k:]
    top_shifts = top_idx - (n - 1)
    g = np.zeros(n)
    for s in top_shifts:
        g += grad_shift(h, s, n)
    g /= top_k
    h_new = project_feasible(h - alpha * g, n)
    return h_new, cur

        # Try best swap

        best_move = None
        if len(ones_idx) > 0 and len(zeros_idx) > 0:
            for i in ones_idx:
                for j in zeros_idx:
                    h_try = h_b.copy()
                    h_try[i] = 0.0
                    h_try[j] = 1.0
                    c5_try = c5_of_h(h_try)
                    d = c5_try - best_c5
                    if d < best_delta:
                        best_delta = d
                        best_move = (i, j)
        # For odd n, also try swapping with half
        if half_idx and len(ones_idx) > 0:
            i = ones_idx[rng.integers(0, len(ones_idx))]
            j = half_idx[0]
            h_try = h_b.copy()
            h_try[i] = 1.0
            h_try[j] = 1.0
            h_try[j] = 0.5  # keep as half, move the 1 elsewhere
            # Actually: swap 1 and 0.5

            h_try[i] = 0.5

            c5_try = c5_of_h(h_try)

def subgradient_optimize(h, n, rng, time_limit, t0):
    h = project_feasible(h, n)
    best = h.copy()
    best_c5 = c5_of_h(h)
    alpha = 0.08
    vel = np.zeros(n)
    patience = 300
    no_imp = 0
    while time.time() - t0 < time_limit:
        h, cur_c5 = subgradient_step(h, n, 5, alpha)
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best = h.copy()
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > patience:
            break
        # Adaptive step
        vel = 0.7 * vel - alpha * (grad_shift(h, 0, n) * 0.1)
        if no_imp % 50 == 0:
            alpha *= 0.7
        elif no_imp == 0:
            alpha = min(alpha * 1.05, 0.3)
    return best, best_c5
