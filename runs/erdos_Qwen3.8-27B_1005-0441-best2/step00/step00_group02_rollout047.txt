import numpy as np
import time

def compute_C(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(50, 0.5)
    n = len(h)
    h = project_feasible(h, n)
    best_C = compute_C(h)
    best_h = h.copy()
    
    def elapsed():
        return time.time() - t0
    
    # Phase 1: Local search with pair transfers
    cur_C = best_C
    stagnation = 0
    
    while elapsed() < budget_s - 100:
        # Pick two indices and a transfer amount
        i = rng.randint(0, n)
        j = rng.randint(0, n)
        if i == j:
            continue
        d = rng.choice([0.005, 0.01, 0.02, 0.05, -0.005, -0.01, -0.02, -0.05])
        # Greedy direction: try to reduce the max correlation
        h_try = h.copy()
        new_vi = h_try[i] + d
        new_vj = h_try[j] - d
        if 0 <= new_vi <= 1 and 0 <= new_vj <= 1:
            h_try[i] = new_vi
            h_try[j] = new_vj
            c_try = compute_C(h_try)
            if c_try < cur_C - 1e-14:
                h = h_try
                cur_C = c_try
                stagnation = 0
                if c_try < best_C:
                    best_C = c_try
                    best_h = h.copy()
            else:
                stagnation += 1
        else:
            stagnation += 1
        
        # Periodic restart with perturbation
        if stagnation > 2000:
            # Random restart around current best
            h = best_h + rng.normal(0, 0.02, n)
            h = project_feasible(h, n)
            cur_C = compute_C(h)
            stagnation = 0
    
    # Phase 2: Try to increase n (interpolation + re-optimize)
    for n_try in [50, 60, 80, 100]:
        if elapsed() > budget_s - 100:
            break
        
        # Interpolate best_h to n_try
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_new = np.interp(x_new, x_old, best_h)
        h_new = project_feasible(h_new, n_try)
        c_new = compute_C(h_new)
        
        if c_new < best_C * 1.01:  # close enough to try improving
            # Quick local search at new n
            h_cur = h_new.copy()
            cur_c = c_new
            stag = 0
            steps = 0
            while stag < 1000 and steps < 200000 and elapsed() < budget_s - 100:
                i = rng.randint(0, n_try)
                j = rng.randint(0, n_try)
                if i == j:
                    continue
                d = rng.choice([0.01, 0.02, 0.05, -0.01, -0.02, -0.05])
                h_t = h_cur.copy()
                nvi = h_t[i] + d
                nvj = h_t[j] - d
                if 0 <= nvi <= 1 and 0 <= nvj <= 1:
                    h_t[i] = nvi
                    h_t[j] = nvj
                    ct = compute_C(h_t)
                    if ct < cur_c - 1e-14:
                        h_cur = h_t
                        cur_c = ct
                        stag = 0
                        if cur_c < best_C:
                            best_C = cur_c
                            best_h = h_cur.copy()
                    else:
                        stag += 1
                steps += 1
        
        # Single-point refinement
        if elapsed() < budget_s - 100:
            h_cur = best_h.copy()
            cur_c = best_C
            for _ in range(100):
                if elapsed() > budget_s - 101:
                    break
                improved = False
                for i in range(n):
                    for d in [0.01, -0.01, 0.03, -0.03]:
                        h_t = h_cur.copy()
                        h_t[i] += d
                        if 0 <= h_t[i] <= 1:
                            h_t = project_feasible(h_t, n)
                            ct = compute_C(h_t)
                            if ct < cur_c - 1e-14:
                                h_cur = h_t
                                cur_c = ct
                                improved = True
                                if cur_c < best_C:
                                    best_C = cur_c
                                    best_h = h_cur.copy()
                                break
                    if improved:
                        break
                if not improved:
                    break
    
    # Final
    best_h = project_feasible(best_h, len(best_h))
    final_C = compute_C(best_h)
    if final_C < best_C:
        best_C = final_C
    
    return best_h, best_C, len(best_h)
