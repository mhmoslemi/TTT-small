def gradient_for_lag(h, n, m):
    """Gradient of overlap at lag m (m can be negative, 0, or positive)."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if m == 0:
        grad = dx * (1.0 - 2.0 * h)
    elif m > 0:
        # c(m) = sum_{n=0}^{n-1-m} a[n+m] * (1-a[n])

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def lag_gradient(h, n, m):
    dx = 2.0 / n
    grad = np.zeros(n)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m > 0 and m < n:
        grad[m:] += (1.0 - h[:n-m])
        grad[:n-m] += -h[m:]
    elif m < 0 and -m < n:
        p = -m
        grad[:n-p] += (1.0 - h[p:])
        grad[p:] += -h[:n-p]
    return grad * dx

def full_subgradient(h, n, top_k=5):
    lags = compute_lags(h)
    max_val = np.max(lags)
    # Get top-k indices
    idx = np.argsort(lags)[-top_k:]
    grad = np.zeros(n)
    for i in idx:
        m = i - (n - 1)
        grad += lag_gradient(h, n, m)
    grad /= top_k
    return grad

    best = h.copy()
    improved = True
    max_rounds = 3
    for _ in range(max_rounds):
        if not improved:
            break
        improved = False
        c5_cur = compute_c5(best)
        for i in range(n):
            old = best[i]
            candidates = [0.0, 1.0, old * 0.5, old + (1.0 - old) * 0.5]
            if 0.0 < old < 1.0:
                candidates.append(old - 0.15)
                candidates.append(old + 0.15)
            best_c5 = c5_cur
            best_val = old
            for c in candidates:
                if c < -0.01 or c > 1.01:
                    continue
                best[i] = np.clip(c, 0.0, 1.0)
                tmp = project_h(best, n)
                cv = compute_c5(tmp)
                if cv < best_c5 - 1e-9:
                    best_c5 = cv
                    best_val = tmp[i]
            if best_val != old:
                best = project_h(best, n)
                improved = True
        best = project_h(best, n)
    return best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf
    best_n = 0

    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass
    if h_init is None:
        n = 95
        h_init = np.zeros(n)
        h_init[:n // 2] = 1.0

    n0 = len(h_init)
    h_init = project_h(h_init, n0)
    c5 = compute_c5(h_init)
    best_h = h_init.copy()
    best_c5 = c5
    best_n = n0

    # Phase 1: Subgradient descent
    h = h_init.copy()
    for it in range(40000):
        if time.time() - t0 > budget_s - 350:
            break
        lr = 0.5 / np.sqrt(float(it + 1))
        g = subgradient(h, n0, top_k=5)
        h = project_h(h - lr * g, n0)
        if it % 500 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n = n0
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5, best_h, best_n = c5, h.copy(), n0

    # Phase 2: Smooth (log-sum-exp) gradient descent
    h = best_h.copy()
    temp = 0.01
    for it in range(20000):
        if time.time() - t0 > budget_s - 180:
            break
        obj, g = smooth_obj_grad(h, temp)
        lr = 0.025 * (1.0 + it) ** (-0.5)
        h = project_h(h - lr * g, n0)
        if it % 200 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n = n0
            temp = max(0.002, temp * 0.9995)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5, best_h, best_n = c5, h.copy(), n0

    # Phase 3: Multi-start smooth optimization
    starts = []
    for trial in range(5):
        h_t = np.zeros(n0)
        if trial == 0:
            for i in range(n0):
                h_t[i] = 1.0 if (i * 7) % 13 < 6 else 0.0
        elif trial == 1:
            idx = rng.choice(n0, size=n0 // 2, replace=False)
            h_t[idx] = 1.0
        elif trial == 2:
            h_t = 0.5 + 0.4 * np.cos(2.0 * np.pi * 3.0 * np.arange(n0) / float(n0))
            h_t = np.clip(h_t, 0.0, 1.0)
        elif trial == 3:
            h_t = 0.5 * (1.0 + np.sin(np.pi * np.arange(n0) / float(n0)))
        else:
            h_t = 0.5 + 0.3 * np.sin(2.0 * np.pi * 5.0 * np.arange(n0) / float(n0) + float(seed))
            h_t = np.clip(h_t, 0.0, 1.0)
        starts.append(h_t)

    for si in range(len(starts)):
        if time.time() - t0 > budget_s - 40:
            break
        h_s = project_h(starts[si].copy(), n0)
        temp = 0.005
        for it in range(6000):
            if time.time() - t0 > budget_s - 10:
                break
            obj, g = smooth_obj_grad(h_s, temp)
            lr = 0.02 * (1.0 + it) ** (-0.4)
            h_s = project_h(h_s - lr * g, n0)
            if it % 500 == 0:
                c5 = compute_c5(h_s)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_s.copy()
                    best_n = n0
            temp = max(0.001, temp * 0.9995)
        c5 = compute_c5(h_s)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_s.copy()
            best_n = n0

    # Phase 4: Coordinate polish on best solution
    if time.time() - t0 < budget_s - 20:
        polished = coordinate_polish(best_h.copy(), best_n)
        c5_pol = compute_c5(polished)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = polished
            best_n = best_n

    if best_h is None:
        best_h = h_init
        best_c5 = compute_c5(h_init)
        best_n = len(h_init)

    return (best_h, best_c5, best_n)
