    return np.correlate(h, 1.0 - h, mode="full") * dx

def weighted_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    ov = np.correlate(h, 1.0 - h, mode="full")
    max_ov = np.max(ov)
    w = np.exp((ov - max_ov) / tau)
    w /= w.sum()
    w = w * dx
    
    grad = np.zeros(n)
    grad += w[n-1] * (1 - 2*h)

        wn = w[n-1-j]
        nj = n - j
        if wp > 1e-15:
            grad[:nj] += wp * (1 - h[j:])
            grad[j:] -= wp * h[:nj]
        if wn > 1e-15:
            grad[j:] += wn * (1 - h[:nj])
            grad[:nj] -= wn * h[j:]
    return grad

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        # Distribute correction among non-saturated elements
        free = (h > 1e-8) & (h < 1 - 1e-8)
        if free.any():
            step = diff / free.sum()
            h[free] = np.clip(h[free] + step, 0, 1)

            diff2 = target_sum - h.sum()
            if abs(diff2) > 1e-10:
                h += diff2 / len(h)
                h = np.clip(h, 0, 1)
    return h

import numpy as np
import time

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def weighted_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    ov = np.correlate(h, 1.0 - h, mode="full")
    max_ov = np.max(ov)
    w = np.exp((ov - max_ov) / tau)
    w /= w.sum()
    w = w * dx
    grad = np.zeros(n)
    grad += w[n - 1] * (1.0 - 2.0 * h)
    for j in range(1, n):
        wp = w[n - 1 + j]
        wn = w[n - 1 - j]
        nj = n - j
        if wp > 1e-15:
            grad[:nj] += wp * (1.0 - h[j:])
            grad[j:] -= wp * h[:nj]
        if wn > 1e-15:
            grad[j:] += wn * (1.0 - h[:nj])
            grad[:nj] -= wn * h[j:]
    return grad

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        frac = diff / len(h)
        h = np.clip(h + frac, 0.0, 1.0)
    return h

def make_balanced(h_raw, rng, n):
    h = (h_raw >= 0.5).astype(float)
    target = n // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > target:
        idx = rng.choice(ones, len(ones) - target, replace=False)
        h[idx] = 0
    elif len(ones) < target:
        idx = rng.choice(zeros, target - len(ones), replace=False)
        h[idx] = 1
    return h

def sa_search(h, rng, n, t_end, T_init, restart_every=50000):
    cur_ov = compute_overlaps(h)
    cur_c5 = np.max(cur_ov)
    best_h = h.copy()
    best_c5 = cur_c5
    T = T_init
    it = 0
    no_improve = 0
    while time.time() < t_end:
        it += 1
        if it % restart_every == 0:
            T = max(T, T_init * 0.5)
        a = rng.integers(n)
        b = rng.integers(n)
        if a == b or h[a] == h[b]:
            continue
        h[a], h[b] = h[b], h[a]
        new_ov = compute_overlaps(h)
        new_c5 = np.max(new_ov)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            cur_ov = new_ov
            no_improve = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        elif rng.random() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-10)):
            cur_c5 = new_c5
            cur_ov = new_ov
            no_improve += 1
        else:
            h[a], h[b] = h[b], h[a]
            no_improve += 1
        T = max(1e-6, T * 0.99999)
        if no_improve > 50000:
            no_improve = 0
            T = T_init * 0.7
    return best_h, best_c5

def gd_refine(h, n, t_end):
    target = n / 2.0
    h = h.astype(float).copy()
    h = project_sum(h, target)
    lr = 0.3
    tau = 0.08
    cur_c5 = np.max(compute_overlaps(h))
    best_h = h.copy()
    best_c5 = cur_c5
    stall = 0
    while time.time() < t_end:
        grad = weighted_gradient(h, tau)
        h_new = project_sum(h - lr * grad, target)
        new_c5 = np.max(compute_overlaps(h_new))
        if new_c5 < cur_c5 - 1e-10:
            h = h_new
            cur_c5 = new_c5
            lr = min(lr * 1.1, 3.0)
            stall = 0

                best_h = h.copy()

            stall += 1

                tau = max(tau * 0.9, 0.005)
                if tau <= 0.005:

                lr = 0.5
                stall = 0
    return best_h, best_c5
