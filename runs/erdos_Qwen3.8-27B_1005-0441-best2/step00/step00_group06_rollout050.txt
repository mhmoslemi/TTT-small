def compute_c5(h):
    """Compute C5 for a given step function."""
    n = len(h)
    dx = 2.0 / n
    result = np.correlate(h, 1 - h, mode="full")
    return np.max(result) * dx

def compute_all_c5(h):
    """Compute all shift values for C5."""
    n = len(h)
    dx = 2.0 / n
    result = np.correlate(h, 1 - h, mode="full")
    return result * dx

def project_to_constraint(h, target_sum):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum} using bisection."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        vals = np.clip(h - mid, 0, 1)
        s = np.sum(vals)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2
    return np.clip(h - mid, 0, 1)

    """Smooth max approximation."""
    scaled = values / tau
    max_val = np.max(scaled)

from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def project_simplex(h, target):
    """Project h onto {0<=h<=1, sum(h)=target} via bisection on shift."""
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)


def smooth_max(vals, tau):
    """Log-sum-exp approximation of max."""
    mx = np.max(vals)
    if mx < 1e-14:
        return 0.0
    return mx + tau * np.log(np.sum(np.exp((vals - mx) / tau)))

import numpy as np
from scipy.optimize import minimize
import time
from functools import partial


def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def project_simplex(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)


def smooth_max(vals, tau):
    mx = np.max(vals)
    if mx < 1e-14:
        return 0.0
    return mx + tau * np.log(np.sum(np.exp((vals - mx) / tau)))


def smooth_obj(x, n, target, tau):
    h = project_simplex(x, target)
    vals = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    return smooth_max(vals, tau)


def binary_local_search(h_start, max_rounds, t0, budget_s):
    h = h_start.astype(np.float64).copy()
    cur = compute_c5(h)
    ones = np.flatnonzero(h > 0.5)
    zeros = np.flatnonzero(h <= 0.5)
    for _ in range(max_rounds):
        if budget_s - (time.time() - t0) < 20:
            break
        best_d = 0.0
        best_p = None
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                c = compute_c5(h)
                h[i], h[j] = 1.0, 0.0
