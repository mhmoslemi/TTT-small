def project(h, n, target_sum):
    """Project h onto {0 <= h[i] <= 1, sum(h) = target_sum}"""
    h = np.clip(h, 0, 1)
    # Water filling to adjust sum
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Need to add diff total
    if diff > 0:
        # Can only add to elements not at 1
        mask = h < 1
        count = np.sum(mask)
        if count == 0:
            return h
        step = diff / count
        h[mask] = np.clip(h[mask] + step, 0, 1)
    else:
        # Need to remove -diff total
        mask = h > 0
        count = np.sum(mask)
        if count == 0:
            return h
        step = -diff / count
        h[mask] = np.clip(h[mask] - step, 0, 1)
    # May need iteration for exact sum
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1 - 1e-10
            count = np.sum(mask)
            if count == 0: break
            h[mask] = np.clip(h[mask] + diff/count, 0, 1)
        else:
            mask = h > 1e-10
            count = np.sum(mask)
            if count == 0: break
            h[mask] = np.clip(h[mask] + diff/count, 0, 1)
    return h

    dx = 2.0 / n

    hi = min(n-1, m)
    if lo > hi:
        return np.zeros(n)
    grad = np.zeros(n)
    # Term 1: grad[lo:hi+1] += (1 - h[p(k)]) for k in [lo,hi], p(k)=k-m+(n-1)
    p_lo = lo - m + (n-1)
    p_len = hi - lo + 1
    grad[lo:hi+1] += (1.0 - h[p_lo:p_lo+p_len])
    # Term 2: for j in [max(0, lo-m+(n-1)), min(n-1, hi-m+(n-1))], grad[j] -= h[j+m-(n-1)]
    j_lo = max(0, lo - m + (n-1))
