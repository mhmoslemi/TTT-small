import numpy as np
import time

def compute_c5_corrections(h):
    """Compute all correlation values."""
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(y, target_sum):
    """Project y onto {x: 0<=x_i<=1, sum(x)=target_sum}."""
    n = len(y)
    # Binary search for tau
    lo = y.min() - target_sum
    hi = y.max()
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        s = np.clip(y - tau, 0.0, 1.0).sum()
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    tau = 0.5 * (lo + hi)
    return np.clip(y - tau, 0.0, 1.0)

def weighted_gradient(h, weights):
    """Compute weighted gradient sum over lags."""
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        w = weights[k]
        if w < 1e-12:
            continue
        if s >= 0:
            i_min = 0
            i_max = n - 1 - s
        else:
            i_min = -s
            i_max = n - 1
        for i in range(i_min, i_max + 1):
            grad[i] += w * (1.0 - h[i + s])
            grad[i + s] -= w * h[i]
    return grad

def optimize_h(h, n_iterations, beta_schedule, step_schedule, start_time, time_budget):
    """Optimize h using smoothed-max gradient descent with projection."""
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_c5 = compute_c5_corrections(h).max() * dx
    best_h = h.copy()
    
    for it in range(n_iterations):
        if time.time() - start_time > time_budget - 10:
            break
        
        # Compute correlation
        corr = compute_c5_corrections(h)
        current_c5 = corr.max() * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute softmax weights
        beta = 50.0  # temperature for smoothed max
        shifted = corr - corr.max()
        weights = np.exp(beta * shifted)
        weights /= weights.sum()
        
        # Compute weighted gradient
        grad = weighted_gradient(h, weights)
        
        # Step size (adaptive)
        step = 0.5 / (beta * dx * n)
        # Actually let's use a simpler step
        step = 0.1 / max(1.0, np.max(np.abs(grad)))
        
        # Update and project

        h_new = project_feasible(h_new, target_sum)
        
        # Check if it improved
        corr_new = compute_c5_corrections(h_new)
        c5_new = corr_new.max() * dx

    return best_h, best_c5

import numpy as np
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(y, target_sum):
    n = len(y)
    lo = y.min() - float(n)
    hi = y.max()
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        s = np.clip(y - tau, 0.0, 1.0).sum()
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    tau = 0.5 * (lo + hi)
    return np.clip(y - tau, 0.0, 1.0)

def compute_gradient(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        w = weights[k]
        if abs(w) < 1e-14:
            continue
        if s >= 0:
            m = n - s
            idx = np.arange(m)
            grad[idx] += w * (1.0 - h[idx + s])
            grad[idx + s] -= w * h[idx]
        else:
            t = -s
            idx = np.arange(t, n)
            grad[idx] += w * (1.0 - h[idx - t])
            grad[idx - t] -= w * h[idx]
    return grad

def optimize_at_n(h, max_iters, deadline, beta_start=25.0):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    beta = beta_start
    stuck_count = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr) * dx)

        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            stuck_count = 0
        else:
            stuck_count += 1
            if stuck_count > 500:
                beta = max(beta * 0.7, 5.0)
                stuck_count = 0

        max_corr = float(corr.max())
        weights = np.exp(beta * (corr - max_corr))
        tw = weights.sum()
        if tw < 1e-300:
            weights = np.zeros_like(corr)
            weights[np.argmax(corr)] = 1.0
        else:
            weights /= tw

        grad = compute_gradient(h, weights)
        grad -= grad.mean()

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break

        step = 0.8 / max(gnorm, 1e-10)
        step = min(step, 1.0)
        improved = False
        for _ in range(8):
            h_new = project_feasible(h - step * grad, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5_val - 1e-14:
                h = h_new
                improved = True
                break
            step *= 0.5

        if not improved:
            if stuck_count > 2000:
                break

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    deadline = start_time + budget_s - 10

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.ones(42) * 0.5

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Main optimization at initial n
    h_opt, c5_opt = optimize_at_n(h0, 100000, deadline, beta_start=25.0)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt

    # Phase 2: Try multiple random perturbations from best
    remaining = deadline - time.time()
    n = len(best_h)
    target_sum = n / 2.0
    perturb_sizes = [0.1, 0.05, 0.02, 0.15, 0.2]
    for ps in perturb_sizes:
        if time.time() > deadline or remaining < 20:
            break
        remaining = deadline - time.time()
        h_pert = best_h + rng.randn(n) * ps
        h_pert = project_feasible(h_pert, target_sum)
        h_try, c5_try = optimize_at_n(h_pert, 20000, deadline, beta_start=15.0)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 3: Try larger n via interpolation
    if time.time() < deadline - 40:
        for n_try in [84, 168]:
            if time.time() > deadline - 20:
                break
            n0 = len(best_h)
            x_old = (np.arange(n0) + 0.5) * 2.0 / n0
            x_new = (np.arange(n_try) + 0.5) * 2.0 / n_try
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try / 2.0)
            h_try, c5_try = optimize_at_n(h_interp, 8000, deadline, beta_start=20.0)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    # Phase 4: Fine-tune with higher beta
    if time.time() < deadline - 15:
        h_try, c5_try = optimize_at_n(best_h, 30000, deadline, beta_start=80.0)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Final validation
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
