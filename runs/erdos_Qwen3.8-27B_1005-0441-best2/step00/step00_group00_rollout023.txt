def get_c5_and_grad(h, n):
    dx = 2.0 / n
    one_minus_h = 1 - h
    corr = np.correlate(h, one_minus_h, mode='full') * dx
    c5 = corr.max()

    n_active = int(active.sum())

    for k_idx in range(2*n - 1):
        if not active[k_idx]:
            continue
        j = k_idx - (n - 1)  # lag
        # dcorr[k_idx]/dh[m] for each m
        # = dx * (1 - h[m+j]) if 0 <= m+j < n
        #   - dx * h[m-j] if 0 <= m-j < n
        m = np.arange(n)
        mask1 = (m + j >= 0) & (m + j < n)
        mask2 = (m - j >= 0) & (m - j < n)
        g = np.zeros(n)
        g[mask1] += (1 - h[m[mask1] + j]) * dx
        g[mask2] -= h[m[mask2] - j] * dx
        grad += g / n_active
    
    return c5, grad

    beta = 30.0
    start = time.time()
    while time.time() - start < budget_s:
        it += 1
        corr = get_corr(h)
        if use_smooth and beta < 50:
            grad = smooth_grad(h, corr, n, beta)
            beta = min(beta * 1.01, 80.0)
        else:
            grad = subgradient(h, corr, n)
        h_new = h - lr * grad
        h_new = project(h_new, n)
        c5_new = compute_c5(h_new)
        h = h_new
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 80:
            lr *= 0.75
            no_improve = 0
            if lr < 1e-9:
                break
        if it % 800 == 0:
            h = best_h + rng.normal(0, 0.015, n)
            h = project(h, n)
            lr = max(lr, 0.03)
            no_improve = 0
            beta = 30.0
    return best_h, best_c5

def make_initial(n, seed, mode=0):
    rng = np.random.RandomState(seed)
    if mode == 0:
        h = np.zeros(n)
        h[:n // 2] = 1.0
        if n % 2 == 1:
            h[n // 2] = 0.0
        h += rng.normal(0, 0.03, n)
    elif mode == 1:
        freq = 3 + rng.randint(0, 4)
        x = np.arange(n)
        h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + rng.uniform(0, 2 * np.pi))
        h = np.clip(h, 0, 1)
    elif mode == 2:
        h = np.zeros(n)
        block = max(2, n // 10)
        for i in range(0, n, 2 * block):
            h[i:i + block] = 1.0
        h += rng.normal(0, 0.02, n)
    else:
        h = (rng.rand(n) > 0.5).astype(float)
        h += rng.normal(0, 0.02, n)
    return project(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Start from initial_h_values
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    h_init = project(h_init, n_init)
    c5_init = compute_c5(h_init)
    
    best_h = h_init.copy()
    best_c5 = c5_init
    best_n = n_init
    
    # Strategy: try multiple (n, init) combinations
    n_list = [n_init, n_init + 1, n_init + 3, n_init + 5, n_init + 10, 100, 200]
    n_list = sorted(set(n_list))
    
    # Allocate time: first n gets most time
    total_n = len(n_list)
    for idx, n in enumerate(n_list):
        remaining = budget_s - (time.time() - t0) - 15
        if remaining < 3:
            break
        # Allocate time proportionally, first gets more
        if idx == 0:
            alloc = remaining * 0.4
        else:
            alloc = remaining * 0.15
        alloc = max(alloc, 5.0)
        
        # Multiple initializations for each n
        n_inits = 4 if idx == 0 else 2
        for mi in range(n_inits):
            if time.time() - t0 > budget_s - 10:
                break
            if n == n_init and mi == 0:
                h = h_init.copy()
            else:
                h = make_initial(n, seed + idx * 100 + mi, mode=mi % 4)
            
            sub_budget = alloc / n_inits
            h_opt, c5_opt = optimize_h(h, n, sub_budget, seed + idx * 200 + mi * 7)
            
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n
    
    # Final polish on best solution with more iterations
    remaining = budget_s - (time.time() - t0) - 5
    if remaining > 10:
        h_pol, c5_pol = optimize_h(best_h, best_n, remaining, seed + 999)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
    
    # Verify
    best_h = project(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    return best_h, final_c5, best_n
