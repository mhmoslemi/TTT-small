import scipy.optimize as optimize

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        deficit = n / 2.0 - np.sum(h)
        if abs(deficit) < 1e-13:
            break
        if deficit > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        if not m.any():
            break
        h[m] += deficit / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _interpolate(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)

            return h_out, _c5(h_out, dx)

        pass
