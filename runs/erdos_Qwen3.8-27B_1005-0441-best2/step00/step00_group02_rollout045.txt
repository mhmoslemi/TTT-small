    def project(y, S):

        # Bisection on lambda
        lo, hi = y.max() - S, y.min() + (S - n) + 1  # rough bounds
        lo = -1.0

            proj = np.clip(y - mid, 0, 1)

import numpy as np
import time
from scipy.optimize import minimize, LinearConstraint

def _project_box_sum(y, S, n):
    """Project y onto {h: 0<=h_i<=1, sum(h)=S} via bisection on lambda."""
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def _corr_vals(h, dx):
    """Return the full correlation vector times dx."""
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _smooth_max(c, eps):
    """Compute smooth max and its gradient weights."""
    cmax = c.max()
    shifted = (c - cmax) * eps
    log_sum = cmax + np.log(np.sum(np.exp(shifted))) / eps
    w = np.exp(shifted)
    w /= w.sum()
    return log_sum, w

def _corr_grad(h, n):
    """Return list of per-lag gradients dc_k/dh."""
    n1 = n - 1
    grad_list = []
    for k in range(2 * n - 1):
        s = k - n1
        g = np.zeros(n)
        j1 = np.arange(n) + s
        m1 = (j1 >= 0) & (j1 < n)
        j2 = np.arange(n) - s
        m2 = (j2 >= 0) & (j2 < n)
        g[m1] += (1.0 - h[j1[m1]])
        g[m2] -= h[j2[m2]]
        grad_list.append(g)
    return grad_list

    lin_con = LinearConstraint(A, S, S)

    h = _project_box_sum(h, S, n)

    best_h = h.copy()
    best_C = _corr_vals(h, dx).max()

    # Phase 1: Multi-scale smooth optimization via L-BFGS-B

            return v

            g = np.zeros(n)
