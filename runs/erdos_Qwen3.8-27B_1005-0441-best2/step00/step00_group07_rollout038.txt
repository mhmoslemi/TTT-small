def project_onto_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if s < 1e-15:
            h = np.ones_like(h) * (target_sum / len(h))
            break
        if abs(s - target_sum) < 1e-12:
            break
        h = h * (target_sum / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx

def compute_smooth_max_grad(h, alpha):

    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    c5 = np.max(corr)
    c_max = np.max(corr)
    exp_c = np.exp(alpha * (corr - c_max))
    w = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):

        if wk < 1e-7:
