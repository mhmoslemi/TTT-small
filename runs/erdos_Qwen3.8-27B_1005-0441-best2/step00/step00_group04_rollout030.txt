                best_val, best_h = v, cand
        except Exception:
            pass

    # Phase 4: Coordinate descent / local search polish
    if time.time() - t_start < time_limit:
        for sweep in range(80):
            if time.time() - t_start >= time_limit:
                break
            improved = False
            for i in range(n):
                # Try moving h[i] and compensating in h[j] for a random j
                for j in [(i + 1) % n, (i + n // 2) % n, (i + 7) % n]:
                    if j == i:
                        j = (i + 3) % n
                deltas = [0.02, -0.02, 0.05, -0.05, 0.1, -0.1, 0.01, -0.01]
                for d in deltas:
                    trial = best_h.copy()
                    trial[i] += d
                    trial[j] -= d
                    trial = repair_h(trial, n)
                    v = c5_value(trial)
                    if v < best_val - 1e-14:
                        best_h = trial
                        best_val = v
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                # Try single-coordinate moves (adjust all others proportionally)
                for i in range(n):
                    for d in [0.01, -0.01, 0.02, -0.02, 0.05, -0.05]:
                        trial = best_h.copy()
                        val = np.clip(trial[i] + d, 0, 1)
                        trial[i] = val
                        trial = repair_h(trial, n)
                        v = c5_value(trial)
                        if v < best_val - 1e-14:
                            best_h = trial
                            best_val = v
                            improved = True
                            break
                    if improved:
                        break
                if not improved:
                    break

    # Final repair to ensure exact constraint
    best_h = repair_h(best_h, n)
    final_val = c5_value(best_h)

    return (best_h, final_val, n)
