import numpy as np
import time


def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(5):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def grad_shift(h, m):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        jm = j - m
        jp = j + m
        if 0 <= jm < n:
            g += 1.0 - h[jm]
        if 0 <= jp < n:
            g -= h[jp]
        grad[j] = g
    return grad * dx


def smooth_grad(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = np.max(corr)
    w = np.exp((corr - max_c) / temp)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        w /= wsum
    grad = np.zeros(n)
    for idx in range(len(corr)):
        if w[idx] < 1e-6:
            continue
        m = idx - (n - 1)
        grad += w[idx] * grad_shift(h, m)
    return grad, max_c
