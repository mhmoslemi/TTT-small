    diff = n_target_sum - np.sum(h)

            if np.sum(mask) > 0:

                h[mask] = np.minimum(h[mask] + add_per, 1.0)

            if np.sum(mask) > 0:

    c = np.correlate(x, 1.0 - x, mode='full') * dx
    mx = np.max(c)
    if tt < 1e-8:
        return mx
    return mx + tt * np.log(np.sum(np.exp((c - mx) / tt)) + 1e-30)

def run(seed=42, budget_s=1100, **kwargs):
    rs = np.random.RandomState(seed)
    t0 = time.time()

    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h = np.full(45, 0.5)

    n = len(h)
    ts = n * 0.5
    h = proj(h, ts)
    bh = h.copy()
    bc = eval_c5(bh)

    # Phase 1: Coarse simulated annealing (60% budget)
    T = 0.025
    Tlo = 0.0004
    cl = 0.99997
    stag = 0
    deadline1 = t0 + budget_s * 0.60

    while time.time() < deadline1:
        h2 = bh.copy()
        r = rs.random()
        if r < 0.40:
            i, j = rs.randint(0, n, 2)
            if i != j:
                d = rs.uniform(-1.0, 1.0) * (0.02 + T * 0.6)
                h2[i] += d
                h2[j] -= d
        elif r < 0.70:
            k = rs.randint(2, 8)
            ix = rs.choice(n, k, replace=False)
            d = rs.randn(k) * (T * 0.3 + 0.004)
            d -= d.mean()
            h2[ix] += d
        elif r < 0.90:
            w = rs.randint(4, max(5, n // 3))
            s = rs.randint(0, n - w)
            d = rs.randn(w) * (T * 0.15 + 0.003)
            d -= d.mean()
            h2[s:s+w] += d
        else:
            i, j = rs.randint(0, n, 2)
            if i != j:
                d = rs.uniform(-0.12, 0.12)
                h2[i] += d
                h2[j] -= d

        h2 = proj(h2, ts)
        c2 = eval_c5(h2)
        if c2 < bc:
            bh[:] = h2
            bc = c2
            stag = 0
        else:
            g = (c2 - bc) / (T + 1e-15)
            if g < 50.0 and rs.rand() < np.exp(-g):
                bh[:] = h2
            stag += 1

        T = max(T * cl, Tlo)
        if stag > 30000:
            T = max(T, 0.01)
            stag = 0

    # Phase 2: Fine pair-swap annealing (25% budget)
    T = 0.0012
    Tlo2 = 0.00004
    cl2 = 0.99999
    deadline2 = t0 + budget_s * 0.85

    while time.time() < deadline2:
        h2 = bh.copy()
        i, j = rs.randint(0, n, 2)
        if i != j:
            d = rs.uniform(-1.0, 1.0) * (T + 0.00015)
            h2[i] += d
            h2[j] -= d
            h2 = proj(h2, ts)
            c2 = eval_c5(h2)
            if c2 < bc:
                bh[:] = h2
                bc = c2
            elif rs.rand() < np.exp(-(c2 - bc) / (T + 1e-20)):
                bh[:] = h2
        T = max(T * cl2, Tlo2)

    # Phase 3: Multi-pair refinement (5% budget)
    T = 0.0008
    deadline3 = t0 + budget_s - 10
    while time.time() < deadline3:
        h2 = bh.copy()
        k = rs.randint(2, 5)
        ix = rs.choice(n, k, replace=False)
        d = rs.randn(k) * (T + 0.0001)
        d -= d.mean()
        h2[ix] += d
        h2 = proj(h2, ts)
        c2 = eval_c5(h2)
        if c2 < bc:
            bh[:] = h2
            bc = c2
        T = max(T * 0.9999, 0.00002)

    # Phase 4: scipy polish with soft-max
    rem = budget_s - (time.time() - t0)
    if rem > 4.0:
        try:
            def obj(x):
                return soft_max_obj(x, n, 0.003)
            cons = [{'type': 'eq', 'fun': lambda x: x.sum() - ts}]
            bnds = [(0.0, 1.0)] * n
            res = minimize(obj, bh, method='SLSQP', bounds=bnds,
                          constraints=cons, options={'maxiter': 300, 'ftol': 1e-14})
            hx = proj(res.x, ts)
            cx = eval_c5(hx)
            if cx < bc:
                bh[:] = hx
                bc = cx
        except Exception:
            pass

    # Final check and repair
    bh = proj(bh, ts)
    bc = eval_c5(bh)
    return bh, bc, n
