from scipy.optimize import minimize

def _sum_eq_fun(hv):
    return np.sum(hv) - _target_sum

def _sum_eq_jac(hv):
    return np.ones(len(hv))

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def softmax_obj(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip((corr - m) / temp, -40.0, 40.0)
    log_sum_exp = np.log(np.sum(np.exp(shifted)))
    return (m + temp * log_sum_exp) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-10
            if not np.any(mask):
                break
            frac = diff / np.sum(room[mask])
            h[mask] += frac * room[mask]
        else:
            mask = h > 1e-10
            if not np.any(mask):
                break
            frac = diff / np.sum(h[mask])
            h[mask] += frac * h[mask]
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return project_feasible(h_new, n_new)
