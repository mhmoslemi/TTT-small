def linearized_step(h, n):
    """One step of the linearized (Benders) method."""
    overlaps = np.correlate(h, 1.0 - h, mode="full")
    t_current = np.max(overlaps)
    
    # Build LP: min t
    # s.t. O_k(h_new) ≈ overlaps[k] + grad_k · (h_new - h) <= t
    #      0 <= h_new_i <= 1
    #      sum h_new = n/2
    #
    # Variables: [h_new_0, ..., h_new_{n-1}, t]
    # Objective: minimize t → c = [0,...,0, 1]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = []
    b_ub = []
    
    for k in range(2*n - 1):
        s = k - (n - 1)
        grad = subgrad_overlap(h, s, n)
        # Constraint: overlaps[k] + grad·(h_new - h) <= t
        # → grad·h_new - t <= t_current... no
        # → grad·h_new <= t - overlaps[k] + grad·h
        # Let's write: grad[0]*h_new[0] + ... + grad[n-1]*h_new[n-1] - t <= -overlaps[k] + grad·h
        
        row = np.zeros(n + 1)
        row[:n] = grad
        row[-1] = -1.0
        A_ub.append(row)
        b_ub.append(-overlaps[k] + np.dot(grad, h))
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Equality: sum(h_new) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [n / 2.0]
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        h_new = res.x[:n]
        t_new = res.x[-1]
    else:
        h_new = h
        t_new = t_current
    
    return h_new, t_new

def project_constraint(h, n_points):
    target = n_points / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(70):

        if s > target:

        else:
            lo = mid
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-10:
        non_sat = (result > 1e-8) & (result < 1.0 - 1e-8)
        if non_sat.sum() > 0:
            result[non_sat] += diff / non_sat.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def subgrad_overlap(h, s, n):
    grad = np.zeros(n)
    j1_lo = max(0, s)
    j1_hi = min(n, n + s)
    if j1_lo < j1_hi:
        idx = np.arange(j1_lo, j1_hi)
        grad[idx] += (1.0 - h[idx - s])
    j2_lo = max(0, -s)
    j2_hi = min(n, n - s)
    if j2_lo < j2_hi:
        idx = np.arange(j2_lo, j2_hi)
        grad[idx] -= h[idx + s]
    return grad
