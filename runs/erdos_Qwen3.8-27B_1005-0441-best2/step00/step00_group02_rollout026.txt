    # Project h onto {0 <= h <= 1, sum(h) = target_sum}
    # Find lambda such that sum(clamp(h - lambda, 0, 1)) = target_sum
    lo, hi = -1.0, 2.0

import numpy as np
import time

def compute_C5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(h, target_sum):
    lo, hi = -3.0, 4.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_subgradient(h, top_m=5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_m:]
    grad = np.zeros(n)
    for idx in top_idx:
        k = int(idx - (n - 1))
        if k >= 0:
            grad[k:] += (1.0 - h[:n - k])
            grad[:n - k] -= h[k:]
        else:
            p = -k
            grad[:n - p] += (1.0 - h[p:])
            grad[p:] -= h[:n - p]
    c5 = float(np.max(corr) * dx)
    grad *= dx / max(1, top_m)
    return grad, c5

def optimize_subgradient(h, max_iter, alpha0, time_limit, top_m=5):

        h = project_box_sum(h_new, n * 0.5)
