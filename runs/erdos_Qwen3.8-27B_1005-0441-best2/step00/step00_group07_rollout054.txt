import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_binary(h, n):
    """Round to binary with exactly n//2 ones"""
    target = n // 2
    idx = np.argsort(-h)  # largest first
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def make_binary(h, n, rng):
    h_noisy = h + rng.random(n) * 1e-6
    return project_binary(h_noisy, n)

def binary_local_search(h, n, timeout=400, kick_timeout=50):
    dx = 2.0 / n
    best = h.copy()
    h1 = 1.0 - best
    corr = np.correlate(best, h1, mode='full')
    best_c5 = float(np.max(corr) * dx)
    
    t0 = time.time()
    stall = 0
    last_kick = t0
    
    while time.time() - t0 < timeout:
        if time.time() - last_kick > kick_timeout and stall > 3:
            # Kick: random perturbation
            ones = np.where(best == 1)[0]
            zeros = np.where(best == 0)[0]
            for _ in range(5):
                a = rng.choice(ones)
                b = rng.choice(zeros)
                best[a], best[b] = 0, 1
            h1 = 1.0 - best
            corr = np.correlate(best, h1, mode='full')
            best_c5 = float(np.max(corr) * dx)
            last_kick = time.time()
            stall = 0

        zeros = np.where(best == 0)[0]

        best_move_c5 = best_c5
        best_pair = None
        best_new_corr = None

def continuous_optimize(h, n, maxiter=200):
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return float(np.max(corr)) * dx
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}]
    bounds = [(0, 1)] * n
    
    result = minimize(obj, h, method='SLSQP', bounds=bounds, 
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})
    if result.success:
        return result.x, compute_c5(result.x)
    return h, compute_c5(h)

    return max_val + np.log(np.sum(np.exp(beta * shifted))) / beta

Let me also think about whether there's a smarter structure to exploit. The Erdős problem's constant C₅ is related to the "minimum overlap problem" where you want to pack two copies of a set in [0,2] to minimize overlap.

For a 0-1 function h with h(x)=1 on set A (|A|=1), the overlap at shift k is |A ∩ (A+k)^c| (roughly). We want to minimize the max over all k.

A natural candidate: make A "as uniform as possible" while being concentrated. For example, in [0,2], take A = [0, 1] (h=1 on [0,1], 0 on [1,2]). Then:
- Shift 0: overlap = 0 (h is 0-1)
- Shift k ∈ (0,1): overlap = k (the part of [0,1] that, when shifted by k, lands in [1,2])
- Shift k = 1: overlap = 1 (all of [0,1] shifted by 1 is in [1,2])
- Shift k ∈ (1,2): overlap = 2-k

So max overlap = 1, giving C5 = 1. That's terrible.

What about A = [0, 0.5] ∪ [1.5, 2]? |A| = 1.
- Shift 0.5: [0,0.5]+0.5 = [0.5,1.0]. Overlap with A^c: all of it. Overlap = 0.5.
- Shift 1.0: [0,0.5]+1 = [1,1.5]. In A^c. Overlap = 0.5. Also [1.5,2]+1 goes out of range.
- Shift 1.5: [0,0.5]+1.5 = [1.5,2.0]. In A! Overlap = 0. Also [1.5,2]+1.5 goes out.
