        else:
            stagnation += 1
            h = h_new
            if stagnation > 200:
                sigma = 0.03 + 0.02 * min(stagnation // 300, 5)
                h = project_onto_feasible(h + rng.normal(0, sigma, n), n_half)
                h_prev = h.copy()
                stagnation = 0
        it += 1
    return best_h, best_c5

def refine_local(h, time_limit, seed=0):
    n = len(h)
    rng = np.random.default_rng(seed)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_half = n / 2.0
    t0 = time.time()
    
    while time.time() - t0 < time_limit:
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        for delta in rng.uniform(-0.05, 0.05, 5):
            h_try = h.copy()
            h_try[i] = np.clip(h_try[i] + delta, 0, 1)
            h_try[j] = np.clip(h_try[j] - delta, 0, 1)
            h_try = project_onto_feasible(h_try, n_half)
            c5 = compute_c5(h_try)
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                best_h = h_try.copy()
                h = h_try
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + budget_s
    rng = np.random.default_rng(seed)
    best_h = None
    best_c5 = np.inf
    best_n = 0
    
    n_values = [70, 100, 140, 200]
    
    for n in n_values:
        remaining = deadline - time.time()
        if remaining < 30:
            break
        time_per_start = min(remaining * 0.18, 120.0)
        if time_per_start < 15:
            time_per_start = 15
        
        starts = []
        if n == 70:
            try:
                iv = np.array(initial_h_values, dtype=float).copy()
                if len(iv) == n:
                    starts.append(iv)
            except (NameError, UnboundLocalError, Exception):
                pass
        
        for _ in range(3):
            h_rand = rng.uniform(0, 1, n)
            h_rand = project_onto_feasible(h_rand, n / 2.0)
            starts.append(h_rand)
        
        # Thue-Morse inspired
        tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
        h_tm = 0.5 + 0.25 * (1.0 - 2.0 * tm)
        h_tm = project_onto_feasible(h_tm, n / 2.0)
        starts.append(h_tm)
        
        # Sine wave structure
        h_sin = 0.5 + 0.25 * np.sin(2.0 * np.pi * np.arange(n) * 3.0 / n)
        h_sin = project_onto_feasible(h_sin, n / 2.0)
        starts.append(h_sin)
        
        for si, h0 in enumerate(starts):
            remaining = deadline - time.time()
            if remaining < 15:
                break
            alloc = min(remaining * 0.35, time_per_start)
            h_opt, c5 = optimize_one(n, h0, alloc, seed=seed + si * 7919 + n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                best_n = n
        
        # Refine best for this n if time allows
        remaining = deadline - time.time()
        if best_n == n and remaining > 40:
            h_ref, c5_ref = refine_local(best_h, min(remaining * 0.3, 60.0), seed=seed)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
    
    if best_h is None:
        n = 70
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n
    
    return best_h, best_c5, best_n
