import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1-h, mode='full') * dx
    return np.max(corr), corr

def project(h, n):
    """Project h to [0,1]^n with sum = n/2"""
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if s > 0:
        h = h / s * (n / 2.0)
    h = np.clip(h, 0, 1)
    # Fix sum constraint more carefully
    s = np.sum(h)
    target = n / 2.0
    if abs(s - target) < 1e-10:
        return h
    # Distribute the difference
    diff = target - s
    # Add/subtract to elements that have room
    if diff > 0:
        room = 1 - h
        room_sum = np.sum(room)
        if room_sum > 0:
            h += room / room_sum * diff
    else:
        room = h
        room_sum = np.sum(room)
        if room_sum > 0:
            h += room / room_sum * diff
    h = np.clip(h, 0, 1)
    return h

def finite_diff_gradient(h, n, dx, eps=1e-7):
    """Compute gradient of max(corr) using finite differences"""
    c5, corr = compute_c5(h, dx)
    grad = np.zeros(n)
    for i in range(n):
        h_p = h.copy()
        h_p[i] += eps
        h_p = np.clip(h_p, 0, 1)
        c5_p, _ = compute_c5(h_p, dx)
        h_m = h.copy()
        h_m[i] -= eps
        h_m = np.clip(h_m, 0, 1)
        c5_m, _ = compute_c5(h_m, dx)
        grad[i] = (c5_p - c5_m) / (2*eps)
    return grad, c5

    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr), corr

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = h.sum()
    if s < 1e-15:
        h[:] = 0.5
        h[:int(target)] = 1.0
        return h
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-10:
        if residual > 0:
            slack = 1.0 - h
            if slack.sum() > 1e-15:
                h += slack * (residual / slack.sum())
        else:
            if h.sum() > 1e-15:
                h -= h * (abs(residual) / h.sum())
    h = np.clip(h, 0.0, 1.0)
    # Final correction for exact sum
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.argmax(h) if diff < 0 else np.argmin(h)
        h[idx] += diff
        h[idx] = np.clip(h[idx], 0.0, 1.0)
    return h

    c5 = np.max(corr)

        weights[np.argmax(corr)] = 1.0

The autocorrelation has a peak at shift 0 (period n/f) and then decays. The overlap integral at shift k will be related to the autocorrelation.

Let me just go with the numerical optimization approach and let the algorithm find the good solutions.

Final code structure:
1. Helper functions (project, compute_c5, gradient, optimize)
2. run() function that:
   a. Tries initial_h_values
   b. Tries various n values with random/sinusoidal starts
   c. Polishes the best
   d. Returns the best (h, c5, n)

import numpy as np
import time


def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr), corr


def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = h.sum()
    if s < 1e-15:
        h[:] = 0.5
        return np.clip(h, 0.0, 1.0)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-10:
        if residual > 0:
            slack = 1.0 - h
            if slack.sum() > 1e-15:
                h += slack * (residual / slack.sum())
        else:
            if h.sum() > 1e-15:
                h -= h * (abs(residual) / h.sum())
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def _grad_c5(h, n, dx, temperature=0.03):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c5 = np.max(corr)
    weights = np.exp((corr - c5) / temperature)
    tw = weights.sum()
    if tw < 1e-15:
        weights = np.zeros_like(corr)
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= tw
    grad = np.zeros(n)
    n1 = n - 1
    j = np.arange(n)
    for idx in range(len(corr)):
        w = weights[idx]
        if w < 1e-10:
            continue
        s = idx - n1
        mask1 = (j - s >= 0) & (j - s < n)
        if np.any(mask1):
            grad[mask1] += w * (1.0 - h[j[mask1] - s])
        mask2 = (j + s >= 0) & (j + s < n)
        if np.any(mask2):
            grad[mask2] -= w * h[j[mask2] + s]
    return grad * dx, c5


def _optimize(h_init, n, dx, time_limit, lr_init=0.05, seed=42):
    start = time.time()
    rng = np.random.RandomState(seed)
    h = _project(h_init.copy(), n)
    best_h = h.copy()
    best_c5 = _compute_c5(h)[0]
    lr = lr_init
    vel = np.zeros(n)
    stall = 0
    beta = 0.85
    temp = 0.03

    while time.time() - start < time_limit:
        grad, c5_cur = _grad_c5(h, n, dx, temperature=temp)
        vel = beta * vel - lr * grad
        h_new = _project(h + vel, n)
        c5_new = _compute_c5(h_new)[0]

        if c5_new < best_c5 - 1e-9:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            stall = 0
            temp = max(0.005, temp * 0.995)
        else:
            h = h_new
            stall += 1
            if stall % 20 == 0:
                lr *= 0.8
            if stall > 150:
                h = h + 0.05 * rng.randn(n)
                h = _project(h, n)
                vel = np.zeros(n)
                lr = lr_init * 0.4
                temp = 0.05
                stall = 0

    return best_h, best_c5


def _random_01(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h


def _sinusoidal(n, freq, amp, phase=0.0):
    x = np.arange(n, dtype=float)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _project(h, n)


def _multi_sinusoidal(n, freqs, amps, phases):
    x = np.arange(n, dtype=float)
    h = np.full(n, 0.5)
    for f, a, p in zip(freqs, amps, phases):
        h += a * np.sin(2.0 * np.pi * f * x / n + p)
    if h.max() > h.min():
        h = (h - h.min()) / (h.max() - h.min())
    return _project(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = np.inf
    best_n = None

    def track(h, n):
        nonlocal best_h, best_c5, best_n
        h = _project(h, n)
        c5, _ = _compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_n = n

    # Phase 1: initial_h_values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        n0 = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=float)
        track(h0, n0)
        remaining = budget_s - (time.time() - t0)
        if remaining > 15:
            tl = min(remaining * 0.2, 50)
            h_opt, c5_opt = _optimize(h0, n0, 2.0 / n0, tl, lr_init=0.03, seed=seed)
            track(h_opt, n0)

    # Phase 2: Multiple n values with diverse starts
    remaining = budget_s - (time.time() - t0)
    n_list = [47, 49, 53, 59, 61, 67, 71, 79, 83, 89, 95, 101, 107]
    per_trial = max(3.0, remaining / (len(n_list) * 3))

    for n in n_list:
        remaining = budget_s - (time.time() - t0)
        if remaining < 3:
            break
        tl = min(per_trial, remaining * 0.06)
        if tl < 2:
            break
        dx = 2.0 / n

        h_rand = _random_01(n, rng)
        track(h_rand, n)
        h_opt, c5_opt = _optimize(h_rand, n, dx, tl, lr_init=0.05, seed=seed + n)
        track(h_opt, n)

        remaining = budget_s - (time.time() - t0)
        if remaining < 3:
            break
        tl2 = min(per_trial * 0.5, remaining * 0.04)
        if tl2 > 2:
            h_sin = _sinusoidal(n, 3, 0.45)
            h_opt2, _ = _optimize(h_sin, n, dx, tl2, lr_init=0.04, seed=seed + n + 100)
            track(h_opt2, n)

    # Phase 3: Multi-sinusoidal starts
    remaining = budget_s - (time.time() - t0)
    for n in [47, 59, 71, 95]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 3:
            break
        tl = min(5.0, remaining * 0.04)
        if tl < 2:
            break
        dx = 2.0 / n
        for f1 in [2, 3]:
            for f2 in [5, 7, 9]:
                h_ms = _multi_sinusoidal(n, [f1, f2], [0.35, 0.2], [0.0, np.pi / 3])
                track(h_ms, n)
                if remaining > 5:
                    h_opt, _ = _optimize(h_ms, n, dx, min(tl, 4.0), lr_init=0.04,
                                         seed=seed + n + f1 * 10 + f2)
                    track(h_opt, n)
                remaining = budget_s - (time.time() - t0)
                if remaining < 3:
                    break

    # Phase 4: Polish best solution with smaller lr and longer time
    remaining = budget_s - (time.time() - t0)
    if remaining > 10 and best_h is not None:
        polish_tl = min(remaining * 0.4, 80)
        h_pol, c5_pol = _optimize(best_h, best_n, 2.0 / best_n, polish_tl,
                                  lr_init=0.01, seed=seed + 7777)
        track(h_pol, best_n)

    # Phase 5: Final fine polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 5 and best_h is not None:
        fine_tl = min(remaining * 0.3, 30)
        h_fin, c5_fin = _optimize(best_h, best_n, 2.0 / best_n, fine_tl,
                                  lr_init=0.003, seed=seed + 4321)
        track(h_fin, best_n)

    # Fallback
    if best_h is None:
        best_n = 47
        best_h = _project(np.full(best_n, 0.5), best_n)
        best_c5, _ = _compute_c5(best_h)

    best_h = _project(best_h, best_n)
    best_c5, _ = _compute_c5(best_h)

    return (best_h, best_c5, best_n)
