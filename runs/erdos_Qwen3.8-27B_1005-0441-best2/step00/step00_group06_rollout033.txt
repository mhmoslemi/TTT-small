def project_onto_simplex_bounds(h, target_sum):
    """Project h onto {0 <= h[i] <= 1, sum(h) = target_sum}"""

            g[kp:] -= w * h[:n-kp] * dx
    return g

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: Smooth gradient descent
    h = h0.copy()
    temp = 20.0
    lr = 0.06
    for it in range(8000):
        if rem() < 500:
            break
        g = smooth_grad(h, dx, temp)
        h_new = project(h - lr * g, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        if it % 400 == 399:
            lr = max(0.008, lr * 0.9)
            temp = min(80.0, temp * 1.06)

    # Phase 2: Multi-lag subgradient
    h = best_h.copy()
    lr = 0.025
    for it in range(5000):
        if rem() < 250:
            break
        g = multi_subgrad(h, dx, 0.9)
        h_new = project(h - lr * g, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        if it % 400 == 399:
            lr = max(0.004, lr * 0.93)

    # Phase 3: Coordinate descent
    h = best_h.copy()
    deltas = [0.03, -0.03, 0.07, -0.07, 0.15, -0.15]
    for rnd in range(150):
        if rem() < 100:
            break
        improved = False
        for i in range(n):
            if rem() < 100:
                break
            old = h[i]
            for d in deltas:
                h[i] = np.clip(old + d, 0.0, 1.0)
                h_try = project(h, target_sum)
                c5_try = compute_c5(h_try, dx)
                if c5_try < best_c5 - 1e-13:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    improved = True
                    break
            if not improved:
                h[i] = old
        if not improved:
            break

    # Phase 4: Random restarts
    if rem() > 300:
        for rst in range(4):
            if rem() < 300:
                break
            h = best_h + rng.randn(n) * 0.06
            h = project(np.clip(h, 0.0, 1.0), target_sum)
            temp_r = 30.0
            lr_r = 0.05
            for it in range(2000):
                if rem() < 300:
                    break
                g = smooth_grad(h, dx, temp_r)
                h_new = project(h - lr_r * g, target_sum)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
                if it % 200 == 199:
                    lr_r = max(0.01, lr_r * 0.92)

    # Phase 5: Push toward binary extremes
    if rem() > 30:
        for strength in [0.04, 0.08, 0.15, 0.25]:
            if rem() < 30:
                break
            h_p = np.where(best_h > 0.5,
                           np.minimum(1.0, best_h + strength),
                           np.maximum(0.0, best_h - strength))
            h_p = project(h_p, target_sum)
            c5_try = compute_c5(h_p, dx)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_p.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project(best_h, target_sum)
    final_c5 = compute_c5(best_h, dx)
    return (best_h, final_c5, n)
