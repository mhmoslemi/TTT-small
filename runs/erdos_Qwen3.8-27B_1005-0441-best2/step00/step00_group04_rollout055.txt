        new_c5 = float(get_corr(h_new).max() * dx)

        if new_c5 <= c5:
            h = h_new
            lr = min(lr * 1.004, 0.25)
        else:
            h = h_new
            lr *= 0.93
            if lr < 0.002:
                lr = 0.06
                if stall > 200:
                    h = project_feasible(best_h + rng.randn(n) * 0.08, n)
                    stall = 0
        # Gradually increase tau for sharper focus
        tau = min(tau * 1.0001, 12.0)

    return best_c5, best_h

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    np.random.seed(seed)

    n = len(initial_h_values)
    dx = 2.0 / n
    h0 = np.array(initial_h_values, dtype=float)

    # Phase 1: optimize the given starting point
    t_alloc = budget_s * 0.45
    best_c5, best_h = optimize_one(h0, n, dx, seed, t_alloc)

    # Phase 2: explore alternative starts
    remaining = budget_s - (time.time() - t_start) - 5
    for i in range(8):
        if remaining < 15:
            break
        alloc = min(remaining * 0.25, 40.0)
        s2 = seed + 7919 * (i + 1)
        r = np.random.RandomState(s2)

        if i % 3 == 0:
            h_init = best_h + r.randn(len(best_h)) * 0.1
            n_try = len(best_h)
        elif i % 3 == 1:
            n_try = max(14, n - 10 + i * 4)
            h_init = (r.rand(n_try) > 0.5).astype(float)
        else:
            n_try = n + i * 5
            x = np.linspace(0.3, 1.7, n_try)
            h_init = np.exp(-1.5 * (x - 1.0) ** 2) / (sigma_fn(n_try))

        n_try = len(h_init)
        c5_try, h_try = optimize_one(h_init, n_try, 2.0 / n_try, s2, alloc)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try
        remaining = budget_s - (time.time() - t_start) - 5

    # Final feasibility check
    best_h = project_feasible(best_h, len(best_h))
    final_c5 = float(get_corr(best_h).max() * (2.0 / len(best_h)))
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)

def sigma_fn(n):
    return float(np.sqrt(2.0 * np.pi) / n)
