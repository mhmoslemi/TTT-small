            grad[i] = (_smooth_obj(hp1, dx, T) - _smooth_obj(hp2, dx, T)) / d
    return grad

def _local_search(h_init, dx, target_sum, t0, budget_s, max_iters=30000):
    n = len(h_init)
    h = h_init.copy()
    cur_c5 = _get_c5(h, dx)
    best_h = h.copy()
    best_c5 = cur_c5
    lr = 0.04
    T = 40.0
    stagnant = 0

    for it in range(max_iters):
        if time.time() - t0 > budget_s - 5:
            break
        if stagnant > 800:
            break

        grad = _smooth_grad(h, dx, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            stagnant += 100
            T = min(T * 1.5, 300.0)
            continue

        # Backtracking line search
        step = lr
        improved = False
        h_new = h
        new_c5 = cur_c5
        for _ls in range(25):
            candidate = h - step * grad
            candidate = _project_h(candidate, target_sum)
            c5_candidate = _get_c5(candidate, dx)
            if c5_candidate < cur_c5 - 1e-12:
                h_new = candidate
                new_c5 = c5_candidate
                improved = True
                break
            step *= 0.5

        if improved:
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
            h = h_new
            cur_c5 = new_c5
            stagnant = 0
            T = min(T * 1.03, 250.0)
        else:
            stagnant += 1
            if stagnant % 30 == 0:
                T = 40.0
            if stagnant > 300:
                # Perturb to escape
                h = h + 0.01 * np.random.randn(n)
                h = _project_h(h, target_sum)
                cur_c5 = _get_c5(h, dx)
                stagnant = 0

    return best_h, best_c5

def _pairwise_refine(h, dx, target_sum, t0, budget_s, max_sweeps=200):
    n = len(h)
    h = h.copy()
    cur_c5 = _get_c5(h, dx)
    best_h = h.copy()
    best_c5 = cur_c5
    rng = np.random.default_rng(hash(tuple(h)) % (2**31))

    for sweep in range(max_sweeps):
        if time.time() - t0 > budget_s - 5:
            break
        any_improve = False
        for i in range(n - 1):
            if time.time() - t0 > budget_s - 5:
                break
            for j in range(i + 1, n):
                for delta in (0.05, 0.1, 0.2, -0.05, -0.1, -0.2):
                    h_try = h.copy()
                    h_try[i] += delta
                    h_try[j] -= delta
                    h_try = _project_h(h_try, target_sum)
                    if np.sum(np.abs(h_try - h)) < 1e-12:
                        continue
                    c5_try = _get_c5(h_try, dx)
                    if c5_try < cur_c5 - 1e-10:
                        h = h_try
                        cur_c5 = c5_try
                        any_improve = True
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        break
        if not any_improve:
            break

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(95, 0.5)

    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0

    h0 = _project_h(h0, target_sum)
    best_h = h0.copy()
    best_c5 = _get_c5(h0, dx)

    # Phase 1: Smooth gradient descent from initial
    bh, bc = _local_search(h0, dx, target_sum, t0, budget_s, max_iters=40000)
    if bc < best_c5:
        best_h, best_c5 = bh, bc

    # Phase 2: Pairwise refinement on best
    if time.time() - t0 < budget_s - 120:
        bh2, bc2 = _pairwise_refine(best_h, dx, target_sum, t0, budget_s, max_sweeps=100)
        if bc2 < best_c5:
            best_h, best_c5 = bh2, bc2

    # Phase 3: Thresholded initialization
    if time.time() - t0 < budget_s - 180:
        h_t = (best_h >= 0.5).astype(float)
        h_t = _project_h(h_t, target_sum)
        ct = _get_c5(h_t, dx)
        if ct < best_c5:
            best_h, best_c5 = h_t, ct
        bh3, bc3 = _local_search(h_t, dx, target_sum, t0, budget_s, max_iters=15000)
        if bc3 < best_c5:
            best_h, best_c5 = bh3, bc3

    # Phase 4: Random restarts with smoothing
    while time.time() - t0 < budget_s - 40:
        h_r = rng.random(n)
        kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
        h_s = np.convolve(h_r, kernel, mode='same')
        h_s = _project_h(h_s, target_sum)
        bh4, bc4 = _local_search(h_s, dx, target_sum, t0, budget_s, max_iters=8000)
        if bc4 < best_c5:
            best_h, best_c5 = bh4, bc4
        if time.time() - t0 < budget_s - 60:
            bh5, bc5 = _pairwise_refine(best_h, dx, target_sum, t0, budget_s, max_sweeps=50)
            if bc5 < best_c5:
                best_h, best_c5 = bh5, bc5

    # Phase 5: Final gradient polish
    if time.time() - t0 < budget_s - 10:
        bh6, bc6 = _local_search(best_h, dx, target_sum, t0, budget_s, max_iters=10000)
        if bc6 < best_c5:
            best_h, best_c5 = bh6, bc6

    # Final projection and evaluation
    best_h = _project_h(best_h, target_sum)
    final_c5 = _get_c5(best_h, dx)

    return best_h, final_c5, n
