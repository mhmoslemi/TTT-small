        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h > 0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] -= diff / cnt
        else:
            mask = h < 1
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += (-diff) / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient(h, n, t=5.0):
    """Compute softmax-weighted gradient of max C_k."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Softmax weights (numerically stable)
    max_c = np.max(corr)
    exp_vals = np.exp(t * (corr - max_c))
    weights = exp_vals / np.sum(exp_vals)
    
    n_corr = len(corr)
    grad = np.zeros(n)
    
    # For each lag m, k = m - (n-1)
    # dC_m/dh[j] = dx * [(1-h[j-k]) - h[j+k]] for valid j
    # We only need to compute for significant weights
    
    for m in range(n_corr):
        if weights[m] < 1e-8:
            continue
        k = m - (n - 1)
        if k >= 0:
            # (1-h[j-k]): j-k >= 0 => j >= k, and j-k < n => j < n+k (always true)
            # h[j+k]: j+k < n => j < n-k
            # Valid j for (1-h[j-k]): j in [k, n-1]
            # Valid j for h[j+k]: j in [0, n-1-k]
            # Combined valid: j in [0, n-1]
            # grad[j] += weights[m] * [(1-h[j-k]) if j>=k else 0] - [h[j+k] if j+k<n else 0]
            if k == 0:
                grad += weights[m] * ((1.0 - h) - h)  # (1-h[j]) - h[j] = 1-2h[j]
            else:
                # (1-h[j-k]) for j >= k
                term1 = np.zeros(n)
                term1[k:] = (1.0 - h[:n-k])
                # h[j+k] for j <= n-1-k
                term2 = np.zeros(n)
                term2[:n-k] = h[k:]
                grad += weights[m] * (term1 - term2)
        else:
            kk = -k  # positive
            # (1-h[j-k]) = (1-h[j+kk]): j+kk < n => j < n-kk
            # h[j+k] = h[j-kk]: j-kk >= 0 => j >= kk
            term1 = np.zeros(n)
            term1[:n-kk] = (1.0 - h[kk:])
            term2 = np.zeros(n)
            term2[kk:] = h[:n-kk]
            grad += weights[m] * (term1 - term2)
    
    return dx * grad

def optimize_subgrad(h, n, seed, time_limit):
    rng = np.random.RandomState(seed)
    h = project_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Momentum
    velocity = np.zeros(n)
    beta = 0.9

    start = time.time()

        t_eff = max(1.0, t_smooth - iteration * 0.001)

        # Momentum update
        velocity = beta * velocity - alpha * grad
        h_new = h + velocity
        
        # Project
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new

            no_improve = 0
