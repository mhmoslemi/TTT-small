        tr = np.sum(room)

            h = h + diff * room / tr
        else:
            h = h + diff / n
    else:
        space = np.maximum(h, 0.0)
        ts = np.sum(space)
        if ts > 1e-14:
            h = h + diff * space / ts
        else:
            h = h + diff / n
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(h_flat, alpha):
    h_curr = np.clip(h_flat, 0.0, 1.0)
    n = len(h_curr)
    dx = 2.0 / n
    corr = np.correlate(h_curr, 1.0 - h_curr, mode="full") * dx
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500, 500)
    return c_max + (1.0 / alpha) * np.log(np.sum(np.exp(shifted)))

def slsqp_phase(h, n, time_limit):
    t_end = time.time() + time_limit
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for alpha in [30.0, 50.0, 80.0]:
        if time.time() > t_end:
            break
        try:
            cons = [{'type': 'eq', 'fun': lambda x, n=n: np.sum(x) - n / 2.0}]
            bnds = [(0.0, 1.0)] * n
            res = optimize.minimize(
                lambda x: soft_max_obj(x, alpha),
                best_h, method='SLSQP',
                bounds=bnds, constraints=cons,
                options={'maxiter': 80, 'ftol': 1e-12}
            )
            h_new = project_sum(np.clip(res.x, 0.0, 1.0), n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

def local_search(h, rng, t_end, best_c5, best_h):
    n = len(h)
    epoch = 0
    step_size = 0.08
    while time.time() < t_end:
        # Adaptive step: shrink if no improvement, expand for diversification
        if epoch % 500 == 0:
            step_size = max(0.01, step_size * 0.85)
            if epoch % 2000 == 0 and epoch > 0:
                h = best_h + rng.uniform(-0.04, 0.04, n)
                h = project_sum(np.clip(h, 0, 1), n)
                step_size = 0.08
        # Two-index transfer
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            epoch += 1
            continue
        delta = rng.uniform(-step_size, step_size)
        hv = h[i] + delta
        uv = h[j] - delta
        if hv < -1e-10 or hv > 1.0 + 1e-10 or uv < -1e-10 or uv > 1.0 + 1e-10:
            epoch += 1
            continue
        h[i] = np.clip(hv, 0, 1)
        h[j] = np.clip(uv, 0, 1)
        c5_trial = compute_c5(h)
        if c5_trial < best_c5 - 1e-13:
            best_c5 = c5_trial
            best_h = h.copy()
        else:
            h[i] -= h[i] - hv  # revert if clipped changed things
            h[j] -= h[j] - uv
            h[i] = best_h[i] if h[i] > 1 or h[i] < 0 else hv
            h[j] = best_h[j] if h[j] > 1 or h[j] < 0 else uv
            # Simpler: just reset h from best if rejected
            pass
        # Actually simpler: keep a working copy
        epoch += 1
    return best_h, best_c5

def local_search2(h, rng, t_end, best_c5, best_h):
    n = len(h)
    step_size = 0.06
    no_improve = 0
    while time.time() < t_end:
        if no_improve > 1500:
            # Diversification
            frac = 0.15
            n_flip = max(2, int(n * frac))
            idx = rng.choice(n, n_flip, replace=False)
            h = best_h.copy()
            for k in idx:
                h[k] = rng.uniform(0, 1)
            h = project_sum(h, n)
            no_improve = 0
            step_size = 0.08
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            no_improve += 1
            continue
        delta = rng.uniform(-step_size, step_size)
        hv = h[i] + delta
        uv = h[j] - delta
        if hv < 0.0 or hv > 1.0 or uv < 0.0 or uv > 1.0:
            no_improve += 1
            continue
        h[i] = hv
        h[j] = uv
        c5_trial = compute_c5(h)
        if c5_trial < best_c5 - 1e-13:
            best_c5 = c5_trial
            best_h = h.copy()
            no_improve = 0
            step_size = min(0.12, step_size * 1.02)
        else:
            h[i] -= hv - h[i]
            h[j] -= uv - h[j]
            h[i] = hv - delta
            h[j] = uv + delta
            no_improve += 1
            if no_improve > 400:
                step_size = max(0.02, step_size * 0.9)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    t_end = t_start + budget_s

    # Get initial values
    h_init = np.array(globals().get('initial_h_values', np.concatenate([np.ones(35), np.zeros(35)])), dtype=float)
    n = len(h_init)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP optimization (~20% budget)
    t_phase1 = t_start + budget_s * 0.20
    if time.time() < t_phase1:
        best_h, best_c5 = slsqp_phase(best_h, n, t_phase1 - time.time())

    # Phase 2: Main local search (~55% budget)
    t_phase2 = t_start + budget_s * 0.75
    if time.time() < t_phase2:
        best_h, best_c5 = local_search2(best_h.copy(), rng, t_phase2, best_c5, best_h)

    # Phase 3: Second local search round with fresh perturbation
    t_phase3 = t_start + budget_s * 0.90
    if time.time() < t_phase3:
        h_pert = best_h + rng.uniform(-0.02, 0.02, n)
        h_pert = project_sum(np.clip(h_pert, 0, 1), n)
        best_h, best_c5 = local_search2(h_pert, rng, t_phase3, best_c5, best_h)

    # Phase 4: Final fine-tuning
    t_final = t_end - 5
    if time.time() < t_final:
        h_fine = best_h + rng.uniform(-0.005, 0.005, n)
        h_fine = project_sum(np.clip(h_fine, 0, 1), n)
        best_h, best_c5 = local_search2(h_fine, rng, t_final, best_c5, best_h)

    # Ensure exact feasibility
    best_h = project_sum(best_h, n)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
