        return h
    lo, hi = -1.0, 2.0

        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any residual
    diff = target_sum - np.sum(projected)
    if abs(diff) > 1e-14:
