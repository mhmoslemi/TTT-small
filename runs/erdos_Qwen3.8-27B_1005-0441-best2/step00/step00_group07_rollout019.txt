                h_opt2 = slsqp_optimize(best_h, n, max_iter=400)
                c5v2 = compute_c5(h_opt2)
                if c5v2 < best_c5:
                    best_c5, best_h = c5v2, h_opt2

    # Phase 4: Upsample to higher resolution
    for n_high in [160, 320, 640]:
        if time_left() < 60:
            break
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_high, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_feasible(h_up, n_high)

        h_opt = slsqp_optimize(h_up, n_high, max_iter=400)
        c5v = compute_c5(h_opt)
        if c5v < best_c5:
            best_c5, best_h = c5v, h_opt
            n = n_high
            # Refine further
            if time_left() > 60:
                h_sub, c5_sub = subgrad_refine(best_h, n, iters=3000, time_limit=10.0)
                if c5_sub < best_c5:
                    best_c5, best_h = c5_sub, h_sub
                h_opt2 = slsqp_optimize(best_h, n, max_iter=400)
                c5v2 = compute_c5(h_opt2)
                if c5v2 < best_c5:
                    best_c5, best_h = c5v2, h_opt2
                # A few more restarts at this resolution
                for _ in range(3):
                    if time_left() < 60:
                        break
                    h_start = best_h + rng.randn(n) * 0.03
                    h_start = project_feasible(h_start, n)
                    h_opt3 = slsqp_optimize(h_start, n, max_iter=400)
                    c5v3 = compute_c5(h_opt3)
                    if c5v3 < best_c5:
                        best_c5, best_h = c5v3, h_opt3

    # Final polish
    if time_left() > 10:
        h_opt = slsqp_optimize(best_h, len(best_h), max_iter=500)
        c5v = compute_c5(h_opt)
        if c5v < best_c5:
            best_c5, best_h = c5v, h_opt

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
