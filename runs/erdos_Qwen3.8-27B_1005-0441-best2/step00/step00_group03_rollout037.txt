import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def make_feasible(h):
    h = np.array(h, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    target = len(h) / 2.0
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            m = h < 1.0
            cnt = np.sum(m)
            if cnt == 0:
                break
            h[m] += diff / cnt
        else:
            m = h > 0.0
            cnt = np.sum(m)
            if cnt == 0:
                break
            h[m] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def sa_optimize(h_init, rng, n_inner, T_start, T_end, cool_rate):
    n = len(h_init)
    target = n * 0.5
    h = h_init.copy()
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    T = T_start
    g = 1.0 - h
    dx = 2.0 / n

    for it in range(n_inner):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(-0.06, 0.06)
        h_save = h.copy()
        g_save = g.copy()

        hi = h[i] + delta
        hj = h[j] - delta
        if hi < 0.0:
            hi = 0.0
        elif hi > 1.0:
            hi = 1.0
        if hj < 0.0:
            hj = 0.0
        elif hj > 1.0:
            hj = 1.0
        h[i] = hi
        h[j] = hj
        g[i] = 1.0 - hi
        g[j] = 1.0 - hj

        s_diff = (hi - h_save[i]) - (hj - h_save[j])
        if abs(s_diff) > 1e-13:
            diff = -s_diff
            m = (h > 1e-14) & (h < 1.0 - 1e-14)
            cnt = np.sum(m)

                    h[:] = h_save
                    g[:] = g_save

                g[:] = g_save

                best_c5 = new_c5
                best_h = h.copy()

                    cur_c5 = new_c5
                else:
                    h[:] = h_save
                    g[:] = g_save
        T = max(T * cool_rate, T_end)
    return best_h, best_c5
