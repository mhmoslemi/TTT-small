import numpy as np
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-12:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
        # Fine-tune
        s2 = np.sum(h)
        if abs(target - s2) > 1e-10:
            for _ in range(3):
                idx = np.argmax((h < 1.0 - 1e-10) * (target - s2 > 0) + (h > 1e-10) * (target - s2 < 0))
                if target > s2:
                    h[idx] = min(1.0, h[idx] + target - s2)
                else:
                    h[idx] = max(0.0, h[idx] + target - s2)
                s2 = np.sum(h)
                if abs(target - s2) < 1e-10:
                    break
    return h

def local_search(h, rng, max_iters=50000):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    no_improve = 0
    delta = 0.05
    
    for it in range(max_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        direction = 1 if rng.random() < 0.5 else -1
        dh = direction * delta
        h[i] += dh
        h[j] -= dh
        new_h = project_feasible(h, n)
        new_c5 = compute_c5(new_h)
        if new_c5 < c5:
            h = new_h
            c5 = new_c5
            no_improve = 0
        else:
            # Revert
            h[i] -= dh
            h[j] += dh
            no_improve += 1
            if no_improve > 2000:
                # Random restart perturbation
                n_pert = max(3, n // 10)
                idxs = rng.choice(n, n_pert, replace=False)
                perturb = rng.uniform(-0.2, 0.2, n_pert)
                h[idxs] += perturb
                h = project_feasible(h, n)
                c5 = compute_c5(h)
                no_improve = 0
                delta = max(0.01, delta * 0.5)
        if no_improve == 0 and it % 5000 == 0:
            delta = max(0.01, delta * 0.99)
    
    return h, c5

    t0 = time.time()

        h_init = np.array(initial_h_values, dtype=float).copy()

        c5_init = compute_c5(h_init)
        results.append((h_init.copy(), c5_init, len(h_init)))

    for n_try in [50, 60, 70, 100]:
        if time.time() - t0 > budget_s * 0.8:
            break
        h_rand = rng.uniform(0, 1, n_try)
        h_rand = project_feasible(h_rand, n_try)
        c5_rand = compute_c5(h_rand)
        results.append((h_rand.copy(), c5_rand, n_try))
    
    # Optimize each
    best_h = None
    best_c5 = float('inf')
    
    for h_start, c5_start, n in results:

        remaining = budget_s * 0.85 - (time.time() - t0)
        # Scale iterations by time

        h_opt, c5_opt = local_search(h_start, rng, max_iters=max_iter)

import numpy as np
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-12:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.any(free):
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if abs(target - s) > 1e-12:
            for _ in range(8):
                if abs(target - s) < 5e-13:
                    break
                if s < target:
                    idx = int(np.argmax(1.0 - h))
                    amt = min(1.0 - h[idx], target - s)
                    h[idx] += amt
                else:
                    idx = int(np.argmax(h))
                    amt = min(h[idx], s - target)
                    h[idx] -= amt
                s = np.sum(h)
    return h

def sa_optimize(h, rng, t_end):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    T = 0.004
    T_min = 1e-7
    delta = 0.06
    no_improve = 0

    while time.time() < t_end:
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        d = delta * rng.uniform(0.3, 1.7)
        sgn = 1.0 if rng.random() < 0.5 else -1.0
        h[i] += sgn * d
        h[j] -= sgn * d
        new_h = project_feasible(h, n)
        new_c5 = compute_c5(new_h)

        if new_c5 < c5:
            h = new_h
            c5 = new_c5
            no_improve = 0
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            if T > T_min and rng.random() < np.exp(-(new_c5 - c5) / T):
                h = new_h
                c5 = new_c5
            else:
                h[i] = best_h[i] if False else h[i]  # just revert
                h[i] = h[i] - sgn * d
                h[j] = h[j] + sgn * d
                no_improve += 1

        T = max(T_min, T * 0.9997)
        delta = max(0.01, delta * 0.99999)

        if no_improve > 3000:
            npert = max(5, n // 6)
            idxs = rng.choice(n, npert, replace=False)
            h[idxs] += rng.uniform(-0.25, 0.25, npert)
            h = project_feasible(h, n)
            c5 = compute_c5(h)
            T = 0.004
            delta = 0.06
            no_improve = 0

    return best_h, best_c5

def steepest_descent(h, n, t_end):
    delta = 0.03
    h = h.copy()
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()

    while time.time() < t_end:
        moved = False
        for i in range(n):
            if time.time() > t_end:
                break
            for j in range(i + 1, n):
                for sgn in (1.0, -1.0):
                    h[i] += sgn * delta
                    h[j] -= sgn * delta
                    nh = project_feasible(h, n)
                    nc5 = compute_c5(nh)
                    h[i] -= sgn * delta
                    h[j] += sgn * delta
                    if nc5 < c5 - 1e-12:
                        h = nh
                        c5 = nc5
                        if c5 < best_c5:
                            best_c5 = c5
                            best_h = h.copy()
                        moved = True
                        break
                if moved:
                    break
            if moved:
                break
        if not moved:
            delta *= 0.7
            if delta < 0.003:
                break

    return best_h, best_c5

def make_smooth_start(n, rng):
    x = np.linspace(0, 2, n)
    mode = int(rng.randint(3))
    if mode == 0:
        h = np.maximum(0.0, np.sin(np.pi * x / 2.0))
    elif mode == 1:
        h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    else:
        center = rng.uniform(0.5, 1.5)
        width = rng.uniform(0.4, 0.9)
        h = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    h = h / (h.sum() / (n / 2.0) if h.sum() > 0 else 1.0)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n)

            best_c5 = c5_sd

        for _ in range(n_restarts):
            if time.time() > t3 - t_per:

            h_s = make_smooth_start(n, rng)
            t_sa_end = time.time() + t_per * 0.7
            if t_sa_end > t3:
                t_sa_end = t3
            h_r, c5_r = sa_optimize(h_s, rng, t_sa_end)

                best_h = h_r.copy()

                if c5_sd2 < best_c5:
                    best_c5 = c5_sd2
                    best_h = h_sd2.copy()

    if time.time() < t_end and best_c5 > 0.37:

        if c5_f < best_c5:

            best_h = h_f.copy()
