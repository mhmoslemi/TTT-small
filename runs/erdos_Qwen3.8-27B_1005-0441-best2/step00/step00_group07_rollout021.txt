def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def normalize_h(h, n):
    """Clip to [0,1] and adjust sum to n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute the difference
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        # Re-adjust (the clip might have changed the sum)
        for _ in range(10):
            diff = target - np.sum(h)
            if abs(diff) < 1e-12:
                break
            mask_up = h < 1.0
            mask_dn = h > 0.0
            if diff > 0:
                h[mask_up] += diff / mask_up.sum()
            else:
                h[mask_dn] += diff / mask_dn.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def objective_and_grad(h, n):
    """Compute C5 and its subgradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    val = np.max(corr) * dx
    # Subgradient: gradient of the active term(s)
    max_val = np.max(corr)
    active = (corr == max_val)
    # For each active lag k, compute gradient
    # corr[k] corresponds to lag m = k - (n-1)
    # c_m = sum_n h_{n+m}*(1-h_n) for valid n
    # d/dh_j: if j=n: -(h_{n+m}) for the active term where n is valid
    #          if j=n+m: (1-h_n) for the active term where n is valid
    grad = np.zeros(n)

        m = k - (n - 1)  # the actual lag

                # d/dh[i2] = (1-h[i])
                grad[i] -= h[i2]
                grad[i2] += (1 - h[i])
    grad *= dx
    return val, grad

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def local_search(h, n, rng, max_iters, check_time):
    h = h.copy()
    cur = compute_c5(h)
    step = 0.5
    for it in range(max_iters):
        if check_time():
            break
        i, j = rng.choice(n, size=2, replace=False)
        d = rng.uniform(-step, step)
        d = max(d, -h[i], h[i] - 1.0)
        d = min(d, h[j], 1.0 - h[j])
        if abs(d) < 1e-12:
            continue
        h[i] += d
        h[j] -= d
        v = compute_c5(h)
        if v < cur:
            cur = v
            if it % 50 == 0:
                step = min(0.8, step * 1.02)
        else:
            h[i] -= d
            h[j] += d
            if it % 80 == 0:
                step = max(0.03, step * 0.96)
    return h, cur

def make_starts(n, rng, best_h, best_n):

    if n == best_n:

        starts.append(fix_sum(h_r, n))
    # Smooth random
    noise = rng.normal(0, 0.3, n)
    h_sm = 0.5 + noise / (1.0 + np.exp(-np.abs(noise * 5)))
    starts.append(fix_sum(h_sm, n))
    return starts

    t0 = time.time()
    rng = np.random.default_rng(seed)
