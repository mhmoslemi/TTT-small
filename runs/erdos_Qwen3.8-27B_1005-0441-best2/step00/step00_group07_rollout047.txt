def c5_with_grad(h, n_points):
    """Return (c5_value, gradient) for the objective."""
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    k_max = np.argmax(corr)
    l = k_max - (n_points - 1)  # the actual lag
    # gradient: (1-h[j-l]) - h[j+l] for all j, out of range = 0
    grad = np.zeros(n_points)

        if 0 <= idx2 < n_points:

import numpy as np
from scipy.optimize import minimize
import time

_gN = 80

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def c5_val_grad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = int(np.argmax(corr))
    l = k - (n - 1)
    grad = np.zeros(n)
    if l >= 0:
        grad[l:n] += 1.0 - h[:n - l]
        grad[:n - l] -= h[l:n]
    else:
        grad[:n + l] += 1.0 - h[-l:]
        grad[-l:] -= h[:n + l]
    return float(corr[k] * dx), grad * dx

def _obj(x):
    return c5_val_grad(x, _gN)

def _con_fun(x):
    return np.sum(x) - _gN / 2.0

def _con_jac(x):
    return np.ones(_gN)

def make_feasible(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s > 1e-15:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h):
    n = len(h)
    half = n // 2
    idx = np.argsort(-h)[:half]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def slsqp(h_init, max_iter=2000):
    global _gN
    n = len(h_init)
    _gN = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    res = minimize(_obj, h_init.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    return res.x

def upsample(h, new_n):
    x_old = np.linspace(0, 1, len(h))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, h)

            best_h = h.copy()
