        projected = np.clip(h - mid, 0, 1)
        s = projected.sum()

            lo = mid

    diff = target_sum - projected.sum()

        else:
            mask = (projected > 0) & (projected < 1)

    grad *= dx / len(active)
    return mv, grad

def coordinate_refine(h, dx, target_sum, n_sweeps=50):
    n = len(h)
    best = h.copy()
    best_val = compute_c5(best, dx)
    for sweep in range(n_sweeps):
        changed = False
        deltas = [0.01, -0.01, 0.005, -0.005, 0.02, -0.02, 0.002, -0.002]
        for i in range(n):
            for d in deltas:
                h_try = best.copy()
                h_try[i] += d
                if 0.0 <= h_try[i] <= 1.0:
                    h_try = project_feasible(h_try, target_sum)
                    c5_try = compute_c5(h_try, dx)
                    if c5_try < best_val - 1e-13:
                        best, best_val = h_try, c5_try
                        changed = True
                        break
        if not changed:
            break
    return best, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    np.random.seed(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except (NameError, UnboundLocalError):
        h0 = np.zeros(95)
        h0[:47] = 1.0
        h0[47] = 0.5

    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_feasible(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    print(f"Initial: C5 = {best_c5:.8f}, n = {n_points}")

    bounds = [(0.0, 1.0)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n_points)}]

    # Phase 1: SLSQP
    elapsed = time.time() - start
    if elapsed < budget_s * 0.25:
        try:
            res = minimize(compute_c5, best_h, args=(dx,), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 400, 'ftol': 1e-14, 'maxls': 50})
            h_new = np.clip(res.x, 0.0, 1.0)
            h_new = project_feasible(h_new, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
                print(f"SLSQP: C5 = {best_c5:.8f}")
        except Exception as e:
            print(f"SLSQP: {e}")

    # Phase 2: Projected subgradient with adaptive step
    elapsed = time.time() - start
    if elapsed < budget_s * 0.6:
        h_curr = best_h.copy()
        for it in range(5000):
            if time.time() - start > budget_s * 0.55:
                break
            val, grad = subgradient_c5(h_curr, dx)
            lr = 0.03 / (1.0 + 0.003 * it)
            h_next = h_curr - lr * grad
            h_next = project_feasible(h_next, target_sum)
            c5_next = compute_c5(h_next, dx)
            if c5_next < best_c5 - 1e-13:
                best_h, best_c5 = h_next.copy(), c5_next
                h_curr = h_next
            elif c5_next < val + 1e-8:
                h_curr = h_next
            if it % 1000 == 0 and it > 0:
                print(f"  Subgrad it{it}: best={best_c5:.8f}")

    # Phase 3: Coordinate refinement
    elapsed = time.time() - start
    if elapsed < budget_s * 0.8:
        best_h, best_c5 = coordinate_refine(best_h, dx, target_sum, n_sweeps=30)
        print(f"Coord: C5 = {best_c5:.8f}")

    # Phase 4: SLSQP from refined point
    elapsed = time.time() - start
    if elapsed < budget_s * 0.92:
        try:
            res = minimize(compute_c5, best_h, args=(dx,), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-14, 'maxls': 50})
            h_new = np.clip(res.x, 0.0, 1.0)
            h_new = project_feasible(h_new, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
                print(f"Final SLSQP: C5 = {best_c5:.8f}")
        except Exception:
            pass

    # Phase 5: Final coordinate polish
    elapsed = time.time() - start
    if elapsed < budget_s * 0.96:
        best_h, best_c5 = coordinate_refine(best_h, dx, target_sum, n_sweeps=15)

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, target_sum)
    final_c5 = compute_c5(best_h, dx)
    print(f"Result: C5 = {final_c5:.8f}, n = {n_points}, time = {time.time()-start:.1f}s")
    return (best_h, final_c5, n_points)
