    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode="full")
    return np.max(c) * dx

def project_feasible(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    # Bisection on lambda: sum(clamp(h - lambda, 0, 1)) = target_sum
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1)
        if s.sum() > target_sum:

        else:
            hi = mid

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def _c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(c))
    L = k_star - (n - 1)
    grad = np.zeros(n)
    if L >= 0:
        if L < n:
            grad[L:] += (1.0 - h[:n - L])
            grad[:n - L] -= h[L:]
    else:
        Lp = -L
        if Lp < n:
            grad[:n - Lp] += (1.0 - h[Lp:])
            grad[Lp:] -= h[:n - Lp]
    return grad * dx

def _proj(h, ts):
