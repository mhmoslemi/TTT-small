import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def active_gradient(x, n):
    """Gradient of the max correlation term w.r.t. x."""
    corr = np.correlate(x, 1.0 - x, mode="full")
    mi = int(np.argmax(corr))
    k = mi - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        for i in range(n - k):
            grad[i + k] += (1.0 - x[i])
            grad[i] -= x[i + k]
    else:
        m = -k
        for i in range(n - m):
            grad[i] += (1.0 - x[i + m])
            grad[i + m] -= x[i]
    return corr * dx, mi, grad


def binary_sa(h, n, dx, best_c5, best_h_ref, rng, max_time, T_init=0.15):
    """Simulated annealing on binary sequence. Returns (best_c5, best_h, h_curr)."""
    t_start = time.time()
    n_ones = n // 2
    c5 = compute_c5(h, dx)
    best_c5_local = c5
    T = T_init
    T_min = 0.002
    no_improve = 0
    restarts = 0
    max_restart_count = 8

    while time.time() - t_start < max_time:
        if no_improve > 60000 and restarts < max_restart_count:
            h = np.zeros(n)
            perm = rng.permutation(n)
            h[perm[:n_ones]] = 1.0
            c5 = compute_c5(h, dx)
            T = T_init
            no_improve = 0
            restarts += 1
            continue

        T = max(T * 0.9997, T_min)

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break

        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])

        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h, dx)
        delta = c5_new - c5

        if delta <= 0 or rng.random() < np.exp(-delta / T):
            c5 = c5_new
            if c5 < best_c5_local:
                best_c5_local = c5
                best_h_ref[:] = h
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1

    return best_c5_local, h


def continuous_refine(x, n, dx, best_c5, best_h, rng, max_iter=5000):
    """Projected gradient on continuous relaxation."""
    t_start = time.time()
    lr = 0.02
    x = x.astype(float).copy()
    for it in range(max_iter):
        if time.time() - t_start > max_iter * 0.001 * 60:
            break
        if time.time() - t_start > 300:
            break
        c5_curr, mi, grad = active_gradient(x, n)
        # Adaptive step
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-12:
            x -= lr * grad / (gnorm + 1e-8)
        x = np.clip(x, 0.0, 1.0)
        # Enforce sum constraint
        s = x.sum()
        target = n / 2.0
        if abs(s - target) > 1e-12:
            x += (target - s) / n
            x = np.clip(x, 0.0, 1.0)
        if it % 200 == 0:
            c5_check = compute_c5(x, dx)
            if c5_check < best_c5:
                best_c5 = c5_check
                best_h[:] = x
    c5_final = compute_c5(x, dx)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h[:] = x
    return best_c5, best_h


def local_continuous_search(h, n, dx, best_c5, best_h):
    """Try setting each coordinate to various values."""
    x = h.astype(float).copy()
    candidates = [0.0, 0.25, 0.5, 0.75, 1.0]
    improved = True
    while improved:
        improved = False
        for i in range(n):
            old = x[i]
            for c in candidates:
                if c == old:
                    continue
                x[i] = c
                s = x.sum()
                if abs(s - n / 2.0) > 0.01:
                    x[i] = old
                    continue
                c5_val = compute_c5(x, dx)
                if c5_val < best_c5 - 1e-10:
                    best_c5 = c5_val
                    best_h[:] = x
                    improved = True
                    break
                else:
                    x[i] = old
    return best_c5, best_h
