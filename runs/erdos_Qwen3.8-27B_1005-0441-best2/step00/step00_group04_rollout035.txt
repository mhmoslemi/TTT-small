def repair_sum(h, n, rng):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = h.sum() - target
        if abs(diff) < 1e-12:
            break

            break

        step = diff / len(idxs)

        h = np.clip(h, 0.0, 1.0)

            if 0 <= new_start and new_start + blen <= n:

                h_new[bstart:bstart+blen] = h_new[new_start:new_start+blen]
                pos = max(bstart, new_start)

                h_new = repair_sum(h_new, n, rng)
        
        new_c5 = compute_c5(h_new, n)

            cur_c5 = new_c5
        else:

                cur_c5 = new_c5

            best_c5 = cur_c5
            best_h = h.copy()
            best_n = n

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_sum(h, n, rng):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = h.sum() - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cands = np.where(h > 1e-8)[0]
        else:
            cands = np.where(h < 1.0 - 1e-8)[0]
        if len(cands) == 0:
            break
        k = min(len(cands), max(1, int(np.ceil(abs(diff) * 4))))
        idxs = rng.choice(cands, size=k, replace=False)
        step = diff / len(idxs)
        h[idxs] -= step
        h = np.clip(h, 0.0, 1.0)
    return h

def make_initial(n, rng):
    h = np.zeros(n)
    h[::2] = 1.0
    if n % 2 == 1:
        h[-1] = 0.5
    h += rng.normal(0, 0.05, n)
    return repair_sum(h, n, rng)

def local_refine(h, n, rng, deadline):
    cur_c5 = compute_c5(h, n)
    while time.time() < deadline:
        improved = False
        for rep in range(3):
            i = int(rng.integers(n))
            best_v = h[i]
            best_trial = cur_c5
            for v in (0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0,
                      h[i] + 0.05, h[i] - 0.05, h[i] + 0.1, h[i] - 0.1):
                v = float(np.clip(v, 0.0, 1.0))
                if abs(v - h[i]) < 1e-9:
                    continue
                h[i] = v
                h = repair_sum(h, n, rng)
                trial = compute_c5(h, n)
                if trial < best_trial:
                    best_trial = trial
                    best_v = v
            if best_v != h[i]:
                h[i] = best_v
                h = repair_sum(h, n, rng)
                cur_c5 = best_trial
                improved = True
        if not improved:
            break
    return h, cur_c5

    best_h = None
    best_n = 0

    start_configs = []
    if h0 is not None and n0 > 0:
        h_init = repair_sum(h0, n0, rng)
        start_configs.append((h_init, n0))
        # Upscale versions
        for mult in (2, 3, 5):
            n_up = n0 * mult
            x_orig = np.linspace(0, n0 - 1, n0)
            x_new = np.linspace(0, n0 - 1, n_up)
            h_up = np.interp(x_new, x_orig, h0)
            h_up = repair_sum(h_up, n_up, rng)

    else:
        h_init = make_initial(n0, rng)

    return best_h, final_c5, best_n
