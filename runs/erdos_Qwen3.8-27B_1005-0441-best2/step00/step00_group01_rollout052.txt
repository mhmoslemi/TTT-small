    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def _corr_all(h):
    dx = _G['dx']
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _gen_grad(h):
    n = _G['n']
    dx = _G['dx']
    corr = _corr_all(h)
    lag = int(np.argmax(corr)) - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        v = 1.0
        if 0 <= j + lag < n:
            v -= h[j + lag]
        if 0 <= j - lag < n:
            v -= h[j - lag]
        grad[j] = dx * v
    return grad

def _gen_grad_multi(h, m=3):
    n = _G['n']
    dx = _G['dx']
    corr = _corr_all(h)
    top_idx = np.argsort(corr)[-m:]
    grad = np.zeros(n)
    for k_idx in top_idx:
        lag = k_idx - (n - 1)
        for j in range(n):
            v = 1.0
            if 0 <= j + lag < n:
                v -= h[j + lag]
            if 0 <= j - lag < n:
                v -= h[j - lag]
            grad[j] += dx * v
    return grad / float(m)

def _make_feasible(h):
    n = _G['n']
    ts = _G['target_sum']
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) > 1e-14:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = ts - h.sum()
        if abs(diff) > 1e-14:
            if diff > 0:
                idx = np.where(h < 1.0 - 1e-12)[0]
            else:
                idx = np.where(h > 1e-12)[0]
            if len(idx) > 0:
                h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def _obj(x):
    return _c5(x)

def _obj_grad(x):
    return _gen_grad(x)

def _obj_smooth(x):
    corr = _corr_all(x)
    top = np.sort(corr)[-3:]
    return float(np.mean(top))

def _obj_smooth_grad(x):
    return _gen_grad_multi(x, m=3)

def _sum_con(x):
    return np.sum(x) - _G['target_sum']

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h_start, bnds, cons, maxiter, ftol):
    res = minimize(_obj, h_start, jac=_obj_grad, method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0

    _G['n'] = n
    _G['dx'] = dx
    _G['target_sum'] = target_sum

    best_h = _make_feasible(h_init.copy())
    best_c5 = _c5(best_h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac}]

    def rem():
        return budget_s - (time.time() - t0)

    def update(h_cand):
        nonlocal best_h, best_c5
        h_cand = _make_feasible(h_cand)
        c = _c5(h_cand)
        if c < best_c5:
            best_h, best_c5 = h_cand, c
        return c

    # Phase 1: Smooth SLSQP (multi-lag gradient)
    if rem() > 30:
        cons_s = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac}]
        try:
            res = minimize(_obj_smooth, best_h, jac=_obj_smooth_grad, method='SLSQP',
                           bounds=bnds, constraints=cons_s,
                           options={'maxiter': 500, 'ftol': 1e-13})
            update(res.x)
        except Exception:
            pass

    # Phase 2: SLSQP with random restarts
    for it in range(40):
        if rem() < 12:
            break
        if it == 0:
            h_s = best_h.copy()
        else:
            s = it % 6
            if s == 0:
                h_s = _make_feasible(best_h + rng.normal(0, 0.05, n))
            elif s == 1:
                h_s = best_h.copy()
                flips = rng.choice(n, size=3, replace=False)
                h_s[flips] = 1.0 - h_s[flips]
                h_s = _make_feasible(h_s)
            elif s == 2:
                h_s = _make_feasible(best_h + rng.normal(0, 0.15, n))
            elif s == 3:
                h_s = (best_h > 0.4).astype(float)
                h_s = _make_feasible(h_s)
            elif s == 4:
                h_s = _make_feasible(rng.permutation(best_h))
            else:
                h_s = _make_feasible(best_h + 0.1 * (rng.random(n) - 0.5))
            try:
                h_new = _slsqp(h_s, bnds, cons, 300, 1e-13)
                update(h_new)
            except Exception:
                pass

    # Phase 3: Nelder-Mead (derivative-free, handles non-smoothness)
    if rem() > 25:
        try:
            res = minimize(_obj, best_h, method='Nelder-Mead',
                           options={'maxiter': 5000, 'xatol': 1e-12,
                                    'fatol': 1e-14, 'adaptive': True})
            update(res.x)
        except Exception:
            pass

    # Phase 4: Subgradient descent with line search
    if rem() > 20:
        h = best_h.copy()
        step = 0.04
        for _ in range(5000):
            if rem() < 5:
                break
            g = _gen_grad(h)
            h_new = _make_feasible(h - step * g)
            if _c5(h_new) < _c5(h):
                h = h_new
                step = min(step * 1.03, 0.15)
            else:
                step *= 0.82
            if step < 1e-8:
                break
        update(h)

    # Phase 5: Binary structure exploration
    if rem() > 30:
        for trial in range(10):
            if rem() < 10:
                break
            # Generate random binary pattern with correct count
            n_ones = int(target_sum)
            n_fracs = target_sum - n_ones
            if n_fracs == 0:
                h_s = np.zeros(n)
                ones = rng.choice(n, size=n_ones, replace=False)
                h_s[ones] = 1.0
            else:
                h_s = np.zeros(n)
                ones = rng.choice(n, size=int(n_ones), replace=False)
                h_s[ones] = 1.0
                frac_pos = rng.choice(n, size=1, replace=False)
                while frac_pos[0] in ones:
                    frac_pos = rng.choice(n, size=1, replace=False)
                h_s[frac_pos[0]] = n_fracs
            h_s = _make_feasible(h_s)
            try:
                h_new = _slsqp(h_s, bnds, cons, 400, 1e-13)
                update(h_new)
            except Exception:
                pass

    # Phase 6: Final polish
    if rem() > 8:
        try:
            h_new = _slsqp(best_h, bnds, cons, 2000, 1e-15)
            update(h_new)
        except Exception:
            pass

    # Ensure final feasibility
    best_h = _make_feasible(best_h)
    best_c5 = _c5(best_h)
    return best_h, best_c5, n
