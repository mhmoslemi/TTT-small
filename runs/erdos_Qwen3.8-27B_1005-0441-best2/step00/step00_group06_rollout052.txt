```python
def compute_gradient(h, N, tau):
    """Compute gradient of smoothed max overlap objective."""
    ones_h = 1 - h
    corr = np.correlate(h, ones_h, mode='full')  # 2N-1 values
    # corr[m] corresponds to lag k = m - (N-1)
    
    # Softmax weights
    tau_corr = tau * corr
    max_tau_corr = np.max(tau_corr)
    exp_vals = np.exp(tau_corr - max_tau_corr)
    p = exp_vals / np.sum(exp_vals)  # shape (2N-1,)
    
    dx = 2.0 / N
    grad = np.zeros(N)
    
    for j in range(N):
        g = 0.0
        for m in range(2*N-1):
            k = m - (N-1)
            # First term: (1-h(j-k)) if 0<=j-k<=N-1
            if 0 <= j-k <= N-1:
                g += p[m] * (1 - h[j-k])
            # Second term: -h(j+k) if 0<=j+k<=N-1
            if 0 <= j+k <= N-1:
                g -= p[m] * h[j+k]
        grad[j] = dx * g
    
    return grad, np.max(corr) * dx, p
```

```python
def evaluate_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    lo = -h.max() - 1.0
    hi = 1.0 - h.min() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0, 1).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0, 1)

def compute_grad(h, tau):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    tc = tau * corr
    tc -= tc.max()
    exp_v = np.exp(tc)
    p = exp_v / exp_v.sum()

        s = 0.0
        for m in range(2 * N - 1):

                s += p[m] * (1 - h[j - k])
            if 0 <= j + k <= N - 1:
                s -= p[m] * h[j + k]
```

```python
import numpy as np
import time

def evaluate_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    lo = -float(h.max()) - 1.0
    hi = 1.0 - float(h.min()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(h + mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def compute_grad(h, tau):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    tc = tau * (corr - corr.max())
    e = np.exp(tc)
    p = e / e.sum()
    ones_rev = (1.0 - h)[::-1]
    grad = np.zeros(N)
    for j in range(N):
        grad[j] = dx * (float(np.dot(p[j:j+N], ones_rev)) - float(np.dot(p[N-1-j:2*N-1-j], h)))
    return grad, float(np.max(corr) * dx)

def optimize_h(h_init, N, time_limit=300, seed=42):

    best_h = h.copy()

    stall_count = 0

            break
```
