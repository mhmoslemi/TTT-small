        n = len(best_h)
        polish_deadline = deadline(0.93)

        # Extended SLSQP polish with very small eps
        h_cur = best_h.copy()
        for eps in [0.006, 0.003, 0.0015, 0.001, 0.0005]:
            if time.time() > polish_deadline:
                break
            h_res, v_res = _slsqp_run(h_cur, n, eps, 1200, polish_deadline)
            if v_res < best_val:
                best_val = v_res
                best_h = h_res.copy()
            h_cur = h_res

        # Subgradient targeted polish
        if time.time() < deadline(0.97):
            h_res, v_res = _subgrad_polish(best_h, n, 5000, deadline(0.97))
            if v_res < best_val:
                best_val = v_res
                best_h = h_res

        # Local perturbation search
        if time.time() < deadline(0.99):
            h_res, v_res = _local_perturb(best_h, n, 200, rng, deadline(0.99))
            if v_res < best_val:
                best_val = v_res
                best_h = h_res

    if best_h is None:
        n = 70
        best_h = np.full(n, 0.5)
        best_val = _c5(best_h)

    best_h = _proj(best_h, len(best_h))
    best_val = _c5(best_h)
    return (best_h, best_val, len(best_h))
