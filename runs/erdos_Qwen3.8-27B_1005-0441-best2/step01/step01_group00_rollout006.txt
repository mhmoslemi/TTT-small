    rng2 = np.random.RandomState(seed + 12345)
    if best_h is not None and remaining() > total_budget * 0.08:
        n = len(best_h)
        for attempt in range(4):
            if remaining() < total_budget * 0.09:
                break
            h_pert = best_h + 0.015 * rng2.randn(n)
            h_pert = _proj(h_pert, n)
            h_res = _slsqp_solve(h_pert, 0.004, 6, 400, 1e-13)
            if h_res is not None:
                v = _c5(h_res)
                if v < best_val:
                    best_val = v
                    best_h = h_res.copy()
            h_sg, v_sg = _subgrad_polish(h_res if h_res is not None else h_pert, remaining, 3, 0.002)
            if v_sg < best_val:
                best_val = v_sg
                best_h = h_sg.copy()

    # Phase 5: Final micro-polish
    if best_h is not None and remaining() > 10:
        h_sg, v_sg = _subgrad_polish(best_h, remaining, 2, 0.001)
        if v_sg < best_val:
            best_val = v_sg
            best_h = h_sg.copy()
        # One more SLSQP at tightest eps
        h_res = _slsqp_solve(best_h, 0.0005, 4, 300, 1e-15)
        if h_res is not None:
            v = _c5(h_res)
            if v < best_val:
                best_val = v
                best_h = h_res

    if best_h is None:
        n = 100
        best_h = np.full(n, 0.5)
        best_val = _c5(best_h)

    best_h = _proj(best_h, len(best_h))
    best_val = _c5(best_h)
    return (best_h, best_val, len(best_h))
