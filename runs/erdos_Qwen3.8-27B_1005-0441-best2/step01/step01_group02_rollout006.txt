```python
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        h = np.clip(h + diff / n, 0.0, 1.0)
    return h

def smooth_max_val(vals, beta):
    m = vals.max()
    z = beta * (vals - m)
    z = np.clip(z, -50, 50)
    return m + np.log(np.exp(z).sum()) / beta

def slsqp_opt(h0, beta, active, maxiter=300):
    n = len(h0)
    target = n / 2.0
    
    if active is not None:
        def f(x):
            sv = all_shifts(x)[active]
            return smooth_max_val(sv, beta)
        def g(x):
            sv = all_shifts(x)[active]
            m = sv.max()
            w = np.exp(beta * (sv - m))
            ws = w.sum()
            if ws < 1e-300: return np.zeros(n)
            w /= ws
            gr = np.zeros(n)
            for j, k in enumerate(active):
                if w[j] > 1e-10:
                    gr += w[j] * grad_shift(x, k)
            return gr
    else:
        def f(x):
            return smooth_max_val(all_shifts(x), beta)
        def g(x):
            sv = all_shifts(x)
            m = sv.max()
            w = np.exp(beta * (sv - m))
            ws = w.sum()
            if ws < 1e-300: return np.zeros(n)
            w /= ws
            gr = np.zeros(n)
            for k in range(2*n-1):
                if w[k] > 1e-10:
                    gr += w[k] * grad_shift(x, k)
            return gr
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project(res.x)
    except Exception:
        return project(h0.copy())

def activate_set(h, m=4):
    sv = all_shifts(h)
    return np.argsort(sv)[-m:]

def nm_polish(h0, maxiter=500):
    n = len(h0)
    def f(x):
        return compute_c5(project(x))
    try:
        res = minimize(f, h0.copy(), method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-11, 'fatol': 1e-13})
        return project(res.x)
    except Exception:
        return project(h0.copy())

def coord_descent(h0, n_passes=10, steps=5):
    n = len(h0)
    h = h0.copy()
    best = compute_c5(h)
    deltas = np.array([0.001, 0.005, 0.01, 0.02, -0.001, -0.005, -0.01, -0.02])
    for _ in range(n_passes):
        improved = True
        while improved:
            improved = False
            for j in range(n):
                for d in deltas:
                    h_try = h.copy()
                    h_try[j] += d
                    if h_try[j] < 0 or h_try[j] > 1:
                        continue
                    # Adjust another coordinate to maintain sum
                    diff = -d
                    # Spread across all other coordinates
                    h_try[(j+1):] += diff / (n-1)
                    h_try = np.clip(h_try, 0, 1)
                    # Re-project
                    h_try = project(h_try)
                    c = compute_c5(h_try)
                    if c < best - 1e-14:
                        h = h_try
                        best = c
                        improved = True
                        break
                if improved:
                    break
    return h, best
```

```python
import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def all_shifts(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_shift(h, s):
    n = len(h)
    dx = 2.0 / n
    d = s - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        r = n - d
        g[:r] += (1.0 - h[d:d+r]) * dx
        g[d:] -= h[:r] * dx
    else:
        dd = -d
        r = n - dd
        g[dd:] += (1.0 - h[:r]) * dx
        g[:r] -= h[dd:] * dx
    return g

def project(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        h = np.clip(h + diff / n, 0.0, 1.0)
    return h

def smooth_max_val(vals, beta):
    m = vals.max()
    if beta < 0.5:
        return m
    z = np.clip(beta * (vals - m), -50, 50)
    return m + np.log(np.exp(z).sum()) / beta

class _St:
    __slots__ = ('beta', 'active', 'n', 'target')
    def __init__(self, beta, active, n):
        self.beta = beta
        self.active = active
        self.n = n
        self.target = n / 2.0

def _obj(x, st):
    sv = all_shifts(x)
    if st.active is not None:
        sv = sv[st.active]
    return smooth_max_val(sv, st.beta)

def _grad(x, st):
    n = st.n
    sv = all_shifts(x)
    sv_use = sv[st.active] if st.active is not None else sv
    m = sv_use.max()
    z = np.clip(st.beta * (sv_use - m), -50, 50)
    w = np.exp(z)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    if st.active is not None:
        for j, k in enumerate(st.active):
            if w[j] > 1e-10:
                g += w[j] * grad_shift(x, k)
    else:
        for k in range(2 * n - 1):
            if w[k] > 1e-10:
                g += w[k] * grad_shift(x, k)
    return g

def _con_fun(x, st):
    return x.sum() - st.target

def _con_jac(x, st):
    return np.ones(st.n)

class _NmFunc:
    def __call__(self, x):
        return compute_c5(project(x))

def slsqp(h0, beta, active, maxiter=250):
    st = _St(beta, active, len(h0))
    cons = [{'type': 'eq', 'fun': partial(_con_fun, st=st),
             'jac': partial(_con_jac, st=st)}]
    bounds = [(0.0, 1.0)] * st.n
    try:
        res = minimize(partial(_obj, st=st), h0, jac=partial(_grad, st=st),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project(res.x)
    except Exception:
        return project(h0.copy())

def top_active(h, m=4):
    sv = all_shifts(h)
    return np.sort(np.argsort(sv)[-m:])

def nm_polish(h0, maxiter=400):
    try:
        res = minimize(_NmFunc(), h0.copy(), method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-11, 'fatol': 1e-13})
        return project(res.x)
    except Exception:
        return project(h0.copy())

def coord_descent(h0, best_c, n_passes=5):
    n = len(h0)
    h = h0.copy()
    best = best_c
    deltas = np.array([0.002, 0.005, 0.01, 0.02, -0.002, -0.005, -0.01, -0.02])
    for _ in range(n_passes):
        improved = True
        iters = 0
        while improved and iters < 60:
            improved = False
            iters += 1
            for j in range(n):
                for d in deltas:
                    h_try = h.copy()
                    h_try[j] += d
                    if h_try[j] < 0 or h_try[j] > 1:
                        continue
                    h_try[(j + 1):] += -d / (n - 1)
                    h_try = project(h_try)
                    c = compute_c5(h_try)
                    if c < best - 1e-14:
                        h = h_try
                        best = c
                        improved = True
                        break
                if improved:
                    break
    return h, best

def grad_line_search(h0, best_c, n_steps=50, lr0=0.01):
    n = len(h0)
    h = h0.copy()
    best = best_c
    for it in range(n_steps):
        lr = lr0 * (0.95 ** it)
        sv = all_shifts(h)
        idx_max = np.argmax(sv)
        g = grad_shift(h, idx_max)
        for shrink in [1.0, 0.5, 0.25, 0.1]:
            h_try = project(h - lr * shrink * g)
            c = compute_c5(h_try)
            if c < best - 1e-14:
                h = h_try
                best = c
                break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    def rem():
        return budget_s - (time.time() - t_start)

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = np.zeros(70)
        h_init[:35] = 1.0

    n = len(h_init)
    h = project(h_init.copy())
    best_h = h.copy()
    best_c = compute_c5(h)

    # Phase 1: Continuation SLSQP with active sets
    betas = [8, 15, 30, 60, 120, 250, 500, 800]
    for b in betas:
        if rem() < 70:
            break
        act = top_active(h, 4)
        h = slsqp(h, b, act, maxiter=250)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()
        if rem() > 50:
            act = top_active(h, 5)
            h = slsqp(h, b, act, maxiter=200)
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()

    # Phase 2: Full-space SLSQP (no active restriction)
    if rem() > 60:
        h = slsqp(best_h, 500, None, maxiter=300)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 3: Gradient line search on worst shift
    if rem() > 40:
        best_h, best_c = grad_line_search(best_h, best_c, n_steps=60, lr0=0.02)

    # Phase 4: NM polish
    if rem() > 50:
        h = nm_polish(best_h, maxiter=min(400, int(rem() - 10)))
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 5: Multi-restart with structured and random perturbations
    n_restarts = min(int(rem() // 42), 10) if rem() > 50 else 0
    for i in range(n_restarts):
        if rem() < 45:
            break
        # Structured sinusoidal perturbation
        freq = rng.randint(1, 7)
        ph = rng.uniform(0, 6.283)
        amp = 0.06 + 0.02 * rng.random()
        pert = amp * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
        hp = project(best_h + pert)
        hp = slsqp(hp, 50, top_active(hp, 4), maxiter=150)
        hp = slsqp(hp, 150, top_active(hp, 3), maxiter=120)
        c = compute_c5(hp)
        if c < best_c:
            best_c, best_h = c, hp.copy()
        # Random perturbation
        if rem() > 50:
            hp = project(best_h + 0.08 * rng.randn(n))
            hp = slsqp(hp, 50, top_active(hp, 4), maxiter=120)
            c = compute_c5(hp)
            if c < best_c:
                best_c, best_h = c, hp.copy()

    # Phase 6: Coordinate descent
    if rem() > 25:
        best_h, best_c = coord_descent(best_h, best_c, n_passes=4)

    # Phase 7: Final gradient + NM polish
    if rem() > 30:
        best_h, best_c = grad_line_search(best_h, best_c, n_steps=40, lr0=0.01)
    if rem() > 20:
        h = nm_polish(best_h, maxiter=250)
        c = compute_c5(h)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 8: Tiny coordinate nudges
    if rem() > 10:
        best_h, best_c = coord_descent(best_h, best_c, n_passes=2)

    best_h = project(best_h)
    final_c = compute_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return (best_h, best_c, n)
```
