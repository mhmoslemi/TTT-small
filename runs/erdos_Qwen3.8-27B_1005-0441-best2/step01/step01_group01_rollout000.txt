from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += resid / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def objective(x, alpha, n, active_idx):

    corr = np.correlate(x, 1.0 - x, mode='full')

        c = corr[active_idx]
    else:
        c = corr
    m = np.max(c)
    shifted = alpha * (c - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    val = m + np.log(np.sum(np.exp(shifted))) / alpha
    return val * dx

    return np.sum(x) - n * 0.5

    return np.ones(n)
