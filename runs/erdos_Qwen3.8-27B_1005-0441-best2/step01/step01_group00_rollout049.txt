def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _smax_obj(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    s = np.exp((c - cmax) / eps)
    return (cmax + eps * np.log(s.sum())) * dx

def _smax_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    s = np.exp((c - cmax) / eps)
    sw = s / s.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = sw[k]
        if wk < 1e-14:
            continue
        m = k - (n - 1)
        if m >= 0:
            g[m:] += wk * (1.0 - h[:n - m])
            g[:n - m] -= wk * h[m:]
        else:
            p = -m
            g[:n - p] += wk * (1.0 - h[p:])
            g[p:] -= wk * h[:n - p]

                res = minimize(_smax_obj, cur.copy(), args=(eps,),
                               jac=_smax_grad, method='SLSQP',
                               bounds=bnds, constraints=cons,
                               options={'maxiter': mi, 'ftol': 1e-13})
                cur = _proj(res.x, n)
            except Exception:
                continue
            v = _c5(cur)
            if v < best_val:
                best_val = v
                best_h = cur.copy()

    if best_h is not None:
        n = len(best_h)
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _eq, 'jac': _eqj, 'args': (n,)}]
        sigma = 0.06
        for bh in range(8):
            if tfrac() > 0.62:
                break
            cur = _proj(best_h + sigma * rng.randn(n), n)
            for eps in [0.03, 0.008]:
                if tfrac() > 0.62:
                    break
                try:
                    res = minimize(_smax_obj, cur.copy(), args=(eps,),
                                   jac=_smax_grad, method='SLSQP',
                                   bounds=bnds, constraints=cons,
                                   options={'maxiter': 300, 'ftol': 1e-13})
                    cur = _proj(res.x, n)
                except Exception:
                    break
                v = _c5(cur)
                if v < best_val:
                    best_val = v
                    best_h = cur.copy()
            sigma *= 0.75

    if best_h is not None and tfrac() < 0.64:
        n = len(best_h)
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _eq, 'jac': _eqj, 'args': (n,)}]
        for eps in [0.003, 0.001]:
            if tfrac() > 0.72:
                break
            try:
                res = minimize(_smax_obj, best_h.copy(), args=(eps,),
                               jac=_smax_grad, method='SLSQP',
                               bounds=bnds, constraints=cons,
                               options={'maxiter': 500, 'ftol': 1e-14})
                hr = _proj(res.x, n)
                v = _c5(hr)
                if v < best_val:
                    best_val = v
                    best_h = hr
            except Exception:
                pass

    if best_h is not None:
        h = best_h.copy()
        n = len(h)
        lr = 0.004
        for it in range(12000):
            if tfrac() > 0.90:
                break
            c = _corr(h)
            cm = c.max()
            w = np.exp(-(cm - c) / 0.005)
            w /= w.sum()
            g = _topgrad(h, w)
            hn = _proj(h - lr * g, n)
            v = _c5(hn)
            if v < best_val - 1e-13:
                best_val = v
                best_h = hn.copy()
                h = hn
                lr = min(lr * 1.004, 0.02)
            else:
                lr *= 0.997
                if lr < 1e-7:
                    break

    if best_h is not None and tfrac() < 0.92:
        h = best_h.copy()
        n = len(h)
        lr = 0.0008
        for it in range(15000):
            if tfrac() > 0.985:
                break
            c = _corr(h)
            ks = int(np.argmax(c))
            m = ks - (n - 1)
            g = np.zeros(n)
            if m >= 0:
                g[m:] += (1.0 - h[:n - m])
                g[:n - m] -= h[m:]
            else:
                p = -m
                g[:n - p] += (1.0 - h[p:])
                g[p:] -= h[:n - p]
            hn = _proj(h - lr * g, n)
            v = _c5(hn)
            if v < best_val - 1e-14:
                best_val = v
                best_h = hn.copy()
                h = hn
            else:
                lr *= 0.998
                if lr < 1e-9:
                    break

    if best_h is None:
        n = 70
        best_h = _proj(np.full(n, 0.5), n)
        best_val = _c5(best_h)

    best_h = _proj(best_h, len(best_h))
    best_val = _c5(best_h)
    return (best_h, best_val, len(best_h))
