        improved = False
        for _ in range(25):
            h_trial = project_feasible(h_work - alpha_s * grad, n)
            c5t = compute_c5(h_trial)
            if c5t < c5 - 1e-12:
                h_work = h_trial
                c5 = c5t
                improved = True
                break
            alpha_s *= 0.5
        if not improved:
            if active_frac > 0.002:
                active_frac *= 0.4
            else:
                break
    return h_best, best_c5


def swap_search(h, n_swaps, t0, t_limit):
    n = len(h)
    h_work = h.copy()
    best_c5 = compute_c5(h_work)
    gamma = 0.015
    for it in range(n_swaps):
        if time.time() - t0 > t_limit:
            break
        i = it % n
        j = (it * 7 + 3) % n
        if i == j:
            j = (j + 1) % n
        direction = 1.0 if h_work[i] < 0.5 else -1.0
        delta = gamma * direction
        hi_new = h_work[i] + delta
        hj_new = h_work[j] - delta
        if hi_new < 0 or hi_new > 1 or hj_new < 0 or hj_new > 1:
            continue
        h_work[i] = hi_new
        h_work[j] = hj_new
        c5n = compute_c5(h_work)
        if c5n < best_c5 - 1e-10:
            best_c5 = c5n
        else:
            h_work[i] -= delta
            h_work[j] += delta
    return h_work, best_c5


def make_blocks(n, n_blocks, block_frac):
    h = np.zeros(n)
    bs = max(1, int(n * block_frac))
    sp = n // n_blocks
    for b in range(n_blocks):
        s = b * sp
        e = min(s + bs, n)
        h[s:e] = 1.0
    return project_feasible(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        h0 = make_blocks(80, 4, 0.25)
    n = len(h0)
    h_best = project_feasible(h0.copy(), n)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: L-BFGS-B smooth max + penalty (quick)
    for alpha in [60.0, 150.0, 300.0]:
        if time.time() - t0 > budget_s * 0.15:
            break
        res = minimize(smooth_obj_pen, h_best.copy(), args=(alpha, 5e4),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 200, 'ftol': 1e-14})
        hc = project_feasible(res.x, n)
        if compute_c5(hc) < c5_best:
            c5_best = compute_c5(hc)
            h_best = hc

    # Phase 2: Subgradient with annealed active set
    if time.time() - t0 < budget_s * 0.45:
        for frac in [0.10, 0.05, 0.02, 0.01, 0.005]:
            if time.time() - t0 > budget_s * 0.45:
                break
            h_sub, c5_sub = subgrad_refine(h_best, 400, frac, t0, budget_s * 0.45)
            if c5_sub < c5_best:
                c5_best = c5_sub
                h_best = h_sub

    # Phase 3: Swap local search (exact, no approximation)
    if time.time() - t0 < budget_s * 0.60:
        h_ls, c5_ls = swap_search(h_best, 15000, t0, budget_s * 0.60)
        if c5_ls < c5_best:
            c5_best = c5_ls
            h_best = h_ls

    # Phase 4: Multiple restarts with diverse structures
    if time.time() - t0 < budget_s - 150:
        starts = [h_best.copy()]
        for nb, bf in [(3, 0.20), (5, 0.15), (6, 0.12)]:
            starts.append(make_blocks(n, nb, bf))
        starts.append(project_feasible(h_best + 0.04 * rng.standard_normal(n), n))
        starts.append(project_feasible(h_best + 0.10 * rng.standard_normal(n), n))
        # Antisymmetric: h'(x) = 1 - h(x+1) structure
        ha = np.zeros(n)
        half = n // 2
        ha[:half] = 1.0 - h_best[half:]
        ha[half:] = 1.0 - h_best[:half]
        starts.append(project_feasible(ha, n))

        for sp in starts:
            if time.time() - t0 > budget_s - 150:
                break
            res = minimize(smooth_obj_pen, sp.copy(), args=(250.0, 5e4),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 150, 'ftol': 1e-13})
            hc = project_feasible(res.x, n)
            c5c = compute_c5(hc)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc
            if time.time() - t0 < budget_s - 130:
                h_sub, c5_sub = subgrad_refine(h_best, 200, 0.02, t0, budget_s - 130)
                if c5_sub < c5_best:
                    c5_best = c5_sub
                    h_best = h_sub
                h_ls, c5_ls = swap_search(h_best, 5000, t0, budget_s - 120)
                if c5_ls < c5_best:
                    c5_best = c5_ls
                    h_best = h_ls

    # Phase 5: Try smaller grids
    for n_try in [120, 100, 80]:
        if time.time() - t0 > budget_s - 60:
            break
        if n_try >= n:
            continue
        x_old = np.arange(n, dtype=float)
        x_new = np.linspace(0.0, float(n - 1), n_try)
        h_sm = np.interp(x_new, x_old, h_best)
        h_sm = project_feasible(h_sm, n_try)
        bm = [(0.0, 1.0)] * n_try
        res = minimize(smooth_obj_pen, h_sm.copy(), args=(300.0, 5e4),
                       method='L-BFGS-B', bounds=bm,
                       options={'maxiter': 150, 'ftol': 1e-13})
        hc = project_feasible(res.x, n_try)
        c5c = compute_c5(hc)
        h_sub, c5_sub = subgrad_refine(hc, 200, 0.02, t0, budget_s - 50)
        c5c2 = min(c5c, c5_sub)
        pick = h_sub if c5_sub < c5c else hc
        if c5c2 < c5_best:
            c5_best = c5c2
            h_best = project_feasible(pick, n_try)
            n = n_try
            h_best = project_feasible(h_best, n)

    # Phase 6: Final polish
    if time.time() - t0 < budget_s - 10:
        h_sub, c5_sub = subgrad_refine(h_best, 150, 0.01, t0, budget_s - 10)
        if c5_sub < c5_best:
            c5_best = c5_sub
            h_best = h_sub
        h_ls, c5_ls = swap_search(h_best, 3000, t0, budget_s - 5)
        if c5_ls < c5_best:
            c5_best = c5_ls
            h_best = h_ls

    h_best = project_feasible(h_best, len(h_best))
    c5_best = compute_c5(h_best)
    return h_best, c5_best, len(h_best)
