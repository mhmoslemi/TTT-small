def _slsqp_jac(h_v):
    return np.ones(len(h_v))

def _slsqp_fun(h_v, n_pts):
    return np.sum(h_v) - n_pts / 2.0

def _make_binary_init(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def _make_pulse_init(n, rng, n_pulses):
    h = np.zeros(n)
    seg = n // n_pulses
    per_pulse = (n // 2) // n_pulses
    for p in range(n_pulses):
        start = p * seg + rng.randint(0, max(1, seg - per_pulse))
        for i in range(per_pulse):
            h[(start + i) % n] = 1.0
    s = h.sum()
    if s < n // 2:
        zeros = np.where(h == 0.0)[0]
        rng.shuffle(zeros)
        for idx in zeros[:int(n // 2 - s)]:
            h[idx] = 1.0
    elif s > n // 2:
        ones = np.where(h == 1.0)[0]
        rng.shuffle(ones)
        for idx in ones[:int(s - n // 2)]:
            h[idx] = 0.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(160, 0.5)

    n = len(h0)
    target_sum = float(n) / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    def _rem():
        return budget_s - (time.time() - t0)

    # Phase 1: PGD annealing from initial (most important)
    if _rem() > 200:
        schedules = [
            (0.08, 0.5, 150),
            (0.03, 0.3, 150),
            (0.01, 0.2, 150),
            (0.003, 0.1, 150),
            (0.001, 0.05, 200),
        ]
        for temp, lr, iters in schedules:
            if _rem() < 40:
                break
            h_new, v_new = _pgd(best_h.copy(), temp, lr, iters, t0, _rem() - 25)
            if v_new < best_c5:
                best_h, best_c5 = h_new, v_new

    # Phase 2: Binary nudging at various thresholds + re-optimize
    if _rem() > 120:
        for thresh in [0.35, 0.45, 0.5, 0.55, 0.65]:
            if _rem() < 25:
                break
            h_bin = _binary_nudge(best_h, thresh)
            for temp, lr, iters in [(0.03, 0.2, 120), (0.005, 0.08, 120)]:
                if _rem() < 15:
                    break
                h_new, v_new = _pgd(h_bin.copy(), temp, lr, iters, t0, _rem() - 10)
                if v_new < best_c5:
                    best_h, best_c5 = h_new, v_new

    # Phase 3: Diverse structured initializations
    if _rem() > 100:
        inits = []
        inits.append(_make_binary_init(n, rng))
        inits.append(_make_pulse_init(n, rng, 2))
        inits.append(_make_pulse_init(n, rng, 3))
        inits.append(_make_pulse_init(n, rng, 4))
        h_pert = best_h + 0.15 * rng.randn(n)
        inits.append(_project(h_pert, target_sum))
        # Antipodal
        inits.append(_project(1.0 - best_h, target_sum))

        for h_init in inits:
            if _rem() < 25:
                break
            h_init = _project(h_init.copy(), target_sum)
            v_init = compute_c5(h_init)
            if v_init >= best_c5 + 0.05:
                continue
            for temp, lr, iters in [(0.05, 0.4, 100), (0.01, 0.1, 150), (0.003, 0.05, 100)]:
                if _rem() < 12:
                    break
                h_new, v_new = _pgd(h_init.copy(), temp, lr, iters, t0, _rem() - 8)
                if v_new < best_c5:
                    best_h, best_c5 = h_new, v_new

    # Phase 4: SLSQP polish (use scipy's constraint handling)
    if _rem() > 40:
        n_pts = n
        bounds = [(0.0, 1.0)] * n
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                 'jac': _slsqp_jac},)
        try:
            res = minimize(lambda x: _slsqp_obj(x, n_pts), best_h.copy(),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-14})
            h_fin = _project(np.clip(res.x, 0.0, 1.0), target_sum)
            v_fin = compute_c5(h_fin)
            if v_fin < best_c5:
                best_h, best_c5 = h_fin, v_fin
        except Exception:
            pass

    # Phase 5: Final fine PGD
    if _rem() > 15:
        h_new, v_new = _pgd(best_h.copy(), 0.0005, 0.03, 400, t0, _rem() - 5)
        if v_new < best_c5:
            best_h, best_c5 = h_new, v_new

    # Phase 6: Try binary snap of final
    if _rem() > 5:
        h_snap = _binary_nudge(best_h, 0.5)
        v_snap = compute_c5(h_snap)
        if v_snap < best_c5:
            best_h, best_c5 = h_snap, v_snap
        # Slight perturbations of binary
        for eps in [0.02, 0.05]:
            h_t = best_h * (1.0 - eps) + 0.5 * eps
            h_t = _project(h_t, target_sum)
            v_t = compute_c5(h_t)
            if v_t < best_c5:
                best_h, best_c5 = h_t, v_t

    return best_h, best_c5, n
