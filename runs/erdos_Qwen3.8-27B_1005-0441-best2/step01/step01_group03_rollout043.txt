def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-10:
        mask = (hp > 1e-8) & (hp < 1.0 - 1e-8)
        cnt = mask.sum()
        if cnt > 0:
            hp[mask] = np.clip(hp[mask] + diff / cnt, 0.0, 1.0)
    return hp

def get_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def get_c5(h, n):
    return float(np.max(get_overlaps(h, n)))

def grad_lag(h, k, n):
    """Vectorized gradient of correlation at lag k."""
    g = np.zeros(n)
    if k >= 0:
        if k > 0:
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] -= h[k:]
    else:
        l = -k
        g[:n-l] += 1.0
        g[:n-l] -= h[l:]
        g[l:] -= h[:n-l]
    return g

def multi_grad(h, n, temp=0.01, top_k=5):
    """Weighted sum of gradients over top lags."""
    ov = get_overlaps(h, n)
    c5 = np.max(ov)
    # Wait, we need the overlap values BEFORE dx multiplication for the gradient
    # Actually no - the gradient of the overlap (after *dx) is dx * gradient of correlate
    # But since dx is constant, we can just work with the correlate values
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Weights: proportional to how close each lag is to the max
    # Use softmax for smooth max
    weights = np.exp((corr - np.max(corr)) / (temp * n / 2))  # scale temp to problem size
    weights /= weights.sum()
    
    g = np.zeros(n)
    # Only process top_k lags for efficiency
    top_indices = np.argsort(corr)[-top_k:]
    for idx in top_indices:
        w = weights[idx]
        if w < 1e-4:
            continue
        k = int(idx) - (n - 1)
        g += w * grad_lag(h, k, n)
    return g, c5 * dx

            h_new = project_feasible(h_new, n)
            c_new = get_c5(h_new, n)
            if c_new < best_c5 - 1e-13:
                best_h = h_new
                best_c5 = c_new
                stalled = 0
                found = True

            step *= 0.5

                c_new = get_c5(h_new, n)
                if c_new < best_c5:
                    best_h = h_new
                    best_c5 = c_new
                stalled = 0
                lr = 0.4

                h_bin = project_feasible(h_bin, n)
                c_bin = get_c5(h_bin, n)
                if c_bin < best_c5:
                    best_h = h_bin
                    best_c5 = c_bin
        
        it += 1
        # Time check within loop
        if time.time() > deadline:
            break
    
    return best_h, best_c5

def binary_local_search(h0, n, deadline, rng):
    """Local search on binary sequences via swap moves."""
    h = np.where(h0 > 0.5, 1.0, 0.0)
    # Fix to exactly n/2 ones
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    target = n // 2
    if len(ones) > target:
        # Remove excess ones
        excess = len(ones) - target
        for _ in range(excess):
            i = ones[rng.randint(len(ones))]
            h[i] = 0.0
            ones = np.where(h == 1.0)[0]
    elif len(ones) < target:
        deficit = target - len(ones)
        for _ in range(deficit):
            i = zeros[rng.randint(len(zeros))]
            h[i] = 1.0
            zeros = np.where(h == 0.0)[0]
    
    best_h = h.copy()
    best_c5 = get_c5(h, n)
    
    temp = 0.05  # annealing temperature
    it = 0
    while time.time() < deadline:
        it += 1
        if it % 200 == 0:
            temp *= 0.95
        
        # Pick a random swap
        a = rng.randint(n)
        b = rng.randint(n)
        if a == b:
            continue
        if h[a] == h[b]:
            continue  # no change
        
        # Swap
        h[a], h[b] = h[b], h[a]
        c_new = get_c5(h, n)
        
        if c_new < best_c5 - 1e-12:
            best_h = h.copy()
            best_c5 = c_new
        else:
            # Accept with probability exp(-(c_new - best_c5)/temp) - simulated annealing
            if np.log(rng.random()) > (c_new - best_c5) / max(temp, 1e-8):
                best_h = h.copy()
                best_c5 = c_new
            else:
                h[a], h[b] = h[b], h[a]  # revert
    
    return best_h, best_c5
