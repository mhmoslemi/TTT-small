    best_h = h.copy()

    taus = [5, 10, 20, 50, 100, 200, 500]
    for tau in taus:
        if time.time() > t_end:
            break
        try:
            res = minimize(smooth_obj, h, args=(n, tau),
                          jac=smooth_grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 200, 'maxfun': 1500, 'ftol': 1e-16})
            h_new = project_feasible(res.x, n)
        except Exception:
            h_new = project_feasible(h + rng.randn(n) * 0.01, n)
        c5_new = get_c5(h_new, n)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    best_h = None
    best_c5 = 1.0
    best_n = 0

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = None

    # Phase 1: Binary ILS on initial (35% budget)
    if h0 is not None:
        n0 = len(h0)
        h_bin = binarize(h0, n0)
        t1 = t0 + budget_s * 0.35
        bh, bc5 = binary_ilc(h_bin, n0, t1, rng)
        if bc5 < best_c5:
            best_c5, best_h, best_n = bc5, bh.copy(), n0
        # Continuous refine from initial
        t2 = t0 + budget_s * 0.50
        bh2, bc52 = continuous_refine(h0, n0, t2, rng)
        if bc52 < best_c5:
            best_c5, best_h, best_n = bc52, bh2.copy(), n0
        # Continuous refine from binary result
        bh3, bc53 = continuous_refine(bh, n0, t2, rng)
        if bc53 < best_c5:
            best_c5, best_h, best_n = bc53, bh3.copy(), n0

    # Phase 2: Try other n values with binary ILS + SA
    n_cands = [48, 50, 52, 54, 56, 58, 60, 62, 64, 68, 70, 72, 74, 76]
    if h0 is not None:
        n_cands = [x for x in n_cands if x != len(h0)]
    per_n = budget_s * 0.10
    for n_try in n_cands:
        if time.time() > t0 + budget_s - 40:
            break
        t_try_end = min(time.time() + per_n, t0 + budget_s - 40)
        # Multiple starts
        for st in range(2):
            if time.time() > t_try_end:
                break
            if st == 0:
                h_s = np.zeros(n_try)
                h_s[:n_try // 2] = 1.0
            else:
                h_s = (rng.rand(n_try) < 0.5).astype(float)
                no = int(np.sum(h_s))
                tg = n_try // 2
                if no > tg:
                    idx = np.where(h_s == 1)[0]
                    h_s[idx[rng.permutation(no)[:no - tg]]] = 0.0
                elif no < tg:
                    idx = np.where(h_s == 0)[0]
                    h_s[idx[rng.permutation(n_try - no)[:tg - no]]] = 1.0
            # Brief SA then ILS
            t_half = (time.time() + t_try_end) * 0.5
            hh, c1 = binary_sa(h_s, n_try, t_half, rng, n_iter=300000)
            hh, c2 = binary_ilc(hh, n_try, t_try_end, rng)
            if c2 < best_c5:
                best_c5, best_h, best_n = c2, hh.copy(), n_try

    # Phase 3: Final continuous refinement
    if best_h is not None and time.time() < t0 + budget_s - 10:
        t_fin = t0 + budget_s - 5
        bh, bc5 = continuous_refine(best_h, best_n, t_fin, rng)
        if bc5 < best_c5:
            best_c5, best_h = bc5, bh

    h_final = project_feasible(best_h, best_n)
    c5_final = get_c5(h_final, best_n)
    return h_final, float(c5_final), best_n
