def get_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-12:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            hp[mask] = np.clip(hp[mask] + diff / np.sum(mask), 0.0, 1.0)
    return hp

def to_binary(h, n):
    n_half = n // 2
    idx = np.argsort(h, kind='stable')[::-1][:n_half]
    b = np.zeros(n)
    b[idx] = 1.0
    return b
