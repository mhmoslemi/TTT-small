import numpy as np
import time

def get_c5_binary(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def get_overlaps_binary(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_binary(h, n, rng):
    """Convert to binary with exactly n/2 ones."""
    h = np.clip(h, 0, 1)
    ones_needed = n // 2
    # Threshold
    thresh = np.sort(h)[ones_needed]
    b = (h >= thresh).astype(float)
    # Fix count
    n_ones = int(np.sum(b))
    if n_ones > ones_needed:
        # Remove excess ones
        one_idx = np.where(b == 1)[0]
        # Remove the ones with smallest h values
        order = np.argsort(h[one_idx])
        for idx in order[:n_ones - ones_needed]:
            b[one_idx[idx]] = 0
    elif n_ones < ones_needed:
        # Add more ones
        zero_idx = np.where(b == 0)[0]
        order = np.argsort(-h[zero_idx])
        for idx in order[:ones_needed - n_ones]:
            b[zero_idx[idx]] = 1
    return b

def sa_optimize(h, n, deadline, rng, T_init=0.003, T_min=0.0001):
    dx = 2.0 / n
    ov = get_overlaps_binary(h, n)
    c5 = float(np.max(ov))
    best_c5 = c5
    best_h = h.copy()
    
    t_start = time.time()
    total_time = deadline - t_start
    n_ones = n // 2
    
    it = 0
    no_improve = 0
    
    while time.time() < deadline:
        # Get current ones and zeros
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        
        # Random swap
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        
        h[i] = 0
        h[j] = 1
        
        ov_new = np.correlate(h, 1.0 - h, mode='full') * dx
        c5_new = float(np.max(ov_new))
        
        # Temperature schedule
        frac = (time.time() - t_start) / max(total_time, 1.0)
        T = T_init * (T_min / T_init) ** frac
        T = max(T, T_min)
        
        # Accept
        accept = False
        if c5_new <= c5:
            accept = True
        elif rng.random() < np.exp(-(c5_new - c5) / T):
            accept = True
        
        if accept:
            c5 = c5_new
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1
            h[j] = 0
            no_improve += 1
        
        it += 1
        
        # Restart if stuck
        if no_improve > 5000:
            # Perturb: do 3-5 random swaps
            for _ in range(3 + rng.randint(3)):

                zi = np.where(h == 0)[0]
                a = oi[rng.randint(len(oi))]
                b = zi[rng.randint(len(zi))]
                h[a], h[b] = 0, 1
            c5 = float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))
            no_improve = 0
            T = T_init * 0.5
    
    return best_h, best_c5

def targeted_local_search(h, n, deadline, rng):

    dx = 2.0 / n
    ov = get_overlaps_binary(h, n)
    c5 = float(np.max(ov))
    best_c5 = c5
    best_h = h.copy()
    
    t_start = time.time()
    
    while time.time() < deadline:
        ov = np.correlate(h, 1.0 - h, mode='full') * dx
        k_max = int(np.argmax(ov))
        lag = k_max - (n - 1)
        
        # Find contributing pairs at this lag
        if lag > 0:
            # h[j]=1, h[j+lag]=0 for j in 0..n-1-lag
            js = np.arange(n - lag)
            contrib = (h[js] == 1) & (h[js + lag] == 0)
        elif lag < 0:
            # h[j]=1, h[j+lag]=0 for j in -lag..n-1
            js = np.arange(-lag, n)
            contrib = (h[js] == 1) & (h[js + lag] == 0)
        else:
            break  # lag 0 is always 0 for binary
        
        if not np.any(contrib):
            break
        
        # Pick a contributing pair and try to fix it
        contrib_js = js[contrib]
        j_pick = contrib_js[rng.randint(len(contrib_js))]
        partner = j_pick + lag
        
        # We want to either: set h[j_pick]=0 (swap with some zero) or h[partner]=1 (swap with some one)
        # Try both
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        
        made_move = False
        for _ in range(10):
            # Try swapping h[j_pick] with a random zero
            if h[j_pick] == 1:
                candidates = zeros_idx[zeros_idx != partner]
                if len(candidates) == 0:
                    continue
                z = candidates[rng.randint(len(candidates))]
                h[j_pick], h[z] = 0, 1
                c5_test = float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))
                if c5_test < c5 - 1e-12:
                    c5 = c5_test
                    made_move = True
                    break
                else:
                    h[j_pick], h[z] = 1, 0
            
            # Try swapping h[partner] with a random one
            if h[partner] == 0:
                candidates = ones_idx[ones_idx != j_pick]
                if len(candidates) == 0:
                    continue
                o = candidates[rng.randint(len(candidates))]
                h[partner], h[o] = 1, 0
                c5_test = float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))
                if c5_test < c5 - 1e-12:
                    c5 = c5_test
                    made_move = True
                    break
                else:
                    h[partner], h[o] = 0, 1
        
        if made_move and c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def get_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def make_binary(h, n, rng):

                    break
                h[i] = 0
                h[j] = 1
                ov_new = np.correlate(h, 1.0 - h, mode='full') * dx
                c5_new = float(np.max(ov_new))
                if c5_new < c5 - 1e-12:
                    c5 = c5_new
                    improved = True
                    oi.remove(i)
                    oi.append(j)
                    zi.remove(j)
                    zi.append(i)
                    break
                else:
                    h[i] = 1
                    h[j] = 0
        if not improved:
            break
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def sa_binary(h, n, deadline, rng, T_init=0.002):
    dx = 2.0 / n
    ov = np.correlate(h, 1.0 - h, mode='full') * dx
    c5 = float(np.max(ov))
    best_c5 = c5
    best_h = h.copy()
    t_start = time.time()
    total_budget = max(deadline - t_start, 1.0)
    T = T_init
    cool = (1e-2 / T_init) ** (1.0 / max(total_budget, 1.0))
    no_improve = 0
    while time.time() < deadline:
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        i = oi[rng.randint(len(oi))]
        j = zi[rng.randint(len(zi))]
        h[i] = 0
        h[j] = 1
        ov_new = np.correlate(h, 1.0 - h, mode='full') * dx
        c5_new = float(np.max(ov_new))
        T = max(T * cool, 1e-7)
        if c5_new <= c5 or rng.random() < np.exp(-(c5_new - c5) / T):
            c5 = c5_new
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1
            h[j] = 0
            no_improve += 1
        if no_improve > 4000:
            for _ in range(rng.randint(2, 6)):
                oi2 = np.where(h == 1)[0]
                zi2 = np.where(h == 0)[0]
                a = oi2[rng.randint(len(oi2))]
                b2 = zi2[rng.randint(len(zi2))]
                h[a] = 0
                h[b2] = 1
            c5 = float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))
            no_improve = 0
            T = max(T_init * 0.4, 0.0004)
    return best_h, best_c5

def ils_binary(h, n, deadline, rng):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = get_c5(h, n)
    t_start = time.time()
    perturb = 3
    while time.time() < deadline:
        sub_dl = min(time.time() + 5.0, deadline)
        h_cur, c5_cur = greedy_hill_climb(h.copy(), n, sub_dl, rng)
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        if time.time() >= deadline:
            break
        oi = np.where(best_h == 1)[0]
        zi = np.where(best_h == 0)[0]
        h = best_h.copy()
        for _ in range(perturb):
            a = oi[rng.randint(len(oi))]
            b_i = zi[rng.randint(len(zi))]
            h[a] = 0
            h[b_i] = 1
            oi = np.where(h == 1)[0]
            zi = np.where(h == 0)[0]
        perturb = max(2, perturb + rng.randint(-1, 2))
        if get_c5(best_h, n) < 0.365:
            perturb = 2
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    best_h = None
    best_c5 = 1.0
    best_n = 0

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except Exception:
        h0 = None
        n0 = 46

    if h0 is not None:
        b0 = make_binary(h0.copy(), n0, rng)
        dl1 = t0 + budget_s * 0.20
        bh, bc5 = ils_binary(b0, n0, dl1, rng)
        best_h, best_c5, best_n = bh, bc5, n0

    n_priority = [44, 50, 52, 48, 54, 46, 42, 56, 38, 40, 58, 60, 64, 68, 72, 76, 80, 36, 34, 32]
    n_list = [x for x in n_priority if (h0 is None or x != n0)]
    if h0 is None:
        n_list.append(46)

    phase2_budget = budget_s * 0.55
    per_n = phase2_budget / max(len(n_list), 1)

    for n_try in n_list:
        if time.time() > t0 + budget_s - 30:
            break
        n_starts = 3 if per_n > 20 else 2
        per_start = per_n / n_starts
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        for s in range(n_starts):
            dl = min(time.time() + per_start, t0 + budget_s - 30)
            if time.time() >= dl:
                break
            if s == 0:
                pos = rng.permutation(n_try)[:n_try // 2]
                hs = np.zeros(n_try)
                hs[pos] = 1.0
            elif s == 1:
                hs = np.zeros(n_try)
                for i in range(n_try):
                    if (i * phi) % 1.0 < 0.5:
                        hs[i] = 1.0
                cnt = int(np.sum(hs))
                if cnt > n_try // 2:
                    oip = np.where(hs == 1)[0]
                    for p in oip[cnt - n_try // 2:]:
                        hs[p] = 0
                elif cnt < n_try // 2:
                    zip_ = np.where(hs == 0)[0]
                    for p in zip_[:n_try // 2 - cnt]:
                        hs[p] = 1
            else:
                hs = np.zeros(n_try)
                hs[::2] = 1.0
                if n_try % 2 == 1:
                    hs[-1] = 0
                cnt = int(np.sum(hs))
                if cnt > n_try // 2:
                    oip = np.where(hs == 1)[0]
                    hs[oip[-1]] = 0
                elif cnt < n_try // 2:
                    zip_ = np.where(hs == 0)[0]
                    hs[zip_[-1]] = 1
            bh, bc5 = ils_binary(hs, n_try, dl, rng)
            if bc5 < best_c5:
                best_h, best_c5, best_n = bh.copy(), bc5, n_try

    if best_h is not None and time.time() < t0 + budget_s - 15:
        dl3 = t0 + budget_s - 5
        bh, bc5 = ils_binary(best_h.copy(), best_n, dl3, rng)
        if bc5 < best_c5:
            best_c5 = bc5
            best_h = bh

    if best_h is not None and time.time() < t0 + budget_s - 3:
        bh, bc5 = greedy_hill_climb(best_h.copy(), best_n, t0 + budget_s - 2, rng)
        if bc5 < best_c5:
            best_c5 = bc5
            best_h = bh

    dx_f = 2.0 / best_n
    best_h = np.clip(best_h, 0.0, 1.0)
    assert abs(np.sum(best_h) * dx_f - 1.0) < 1e-8
    return best_h, float(best_c5), best_n
