        for ei, eps in enumerate(eps_schedule):
            if time.time() - t0 > budget_s * 0.78:
                break
            frac_left = (budget_s * 0.78 - (time.time() - t0)) / budget_s
            maxit_l = max(500, int(4000 * frac_left * 1.5))
            maxit_s = max(200, int(1500 * frac_left))

            try:
                h_new = _lbgfbs_round(cur, eps, 2000.0, n, maxit_l)
                v = _c5(h_new)
                if v < best_val:
                    best_val = v
                    best_h = h_new.copy()
                cur = h_new
            except Exception:
                pass

            if time.time() - t0 < budget_s * 0.78:
                try:
                    h_new = _slsqp_round(cur, eps, n, maxit_s)
                    v = _c5(h_new)
                    if v < best_val:
                        best_val = v
                        best_h = h_new.copy()
                    cur = h_new
                except Exception:
                    pass

            # Adaptive perturbation to escape local optima
            if ei < len(eps_schedule) - 2 and time.time() - t0 < budget_s * 0.72:
                amp = 0.015 * (len(eps_schedule) - ei) / len(eps_schedule)
                cur = _proj(cur + amp * rng.randn(n), n)

        v = _c5(cur)
        if v < best_val:
            best_val = v
            best_h = cur.copy()

    # Multi-level subgradient polish
    if best_h is not None and time.time() - t0 < budget_s * 0.90:
        n = len(best_h)
        best_h, best_val = _subgrad_polish(best_h, n, t0, budget_s, 0.93, 8000)

    # Final fine SLSQP polish
    if best_h is not None and time.time() - t0 < budget_s * 0.96:
        n = len(best_h)
        for eps_fin in [0.0015, 0.0008]:
            if time.time() - t0 > budget_s * 0.975:
                break
            try:
                h_new = _slsqp_round(best_h, eps_fin, n, 700)
                v = _c5(h_new)
                if v < best_val:
                    best_val = v
                    best_h = h_new
            except Exception:
                pass

    # Last subgradient sweep
    if best_h is not None and time.time() - t0 < budget_s * 0.985:
        n = len(best_h)
        best_h, best_val = _subgrad_polish(best_h, n, t0, budget_s, 0.995, 5000)

    if best_h is None:
        n = 70
        best_h = np.full(n, 0.5)
        best_val = _c5(best_h)

    best_h = _proj(best_h, len(best_h))
    best_val = _c5(best_h)
    return (best_h, best_val, len(best_h))
