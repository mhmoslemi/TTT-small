```python
        if wp != 0:
            grad[k:] += wp * (1.0 - h[:n-k])
            grad[:n-k] += wp * (-h[k:])

            grad[:n-k] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:n-k])
    
    return grad
```

```python
def project(h, n):
    # Project onto {0<=x<=1, sum(x)=n/2}
    target = n / 2.0
    # Binary search for lambda
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    # Fine-adjust: distribute remaining
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        # Add/subtract uniformly from non-saturated entries
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        if mask.sum() > 0:
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x
```

Let me try yet another approach: **alternating minimization**.

The objective is max_k sum_i h[i+k](1-h[i]). Each term is bilinear in h. If I fix all of h except one element h[j], the problem becomes:
minimize max_k (a_k + h[j]*b_k)
where a_k = corr_k - h[j]*d(corr_k)/dh[j] (the "base" without h[j]'s contribution)
and b_k = d(corr_k)/dh[j]

This is a 1D problem: find h[j] to minimize max_k (a_k + h[j]*b_k) subject to 0<=h[j]<=1.

The optimal h[j] is at the intersection point where the max is minimized. This is a "minimax" 1D problem that can be solved by:
- For each pair of active constraints k1, k2, find h[j] where a_k1 + h[j]*b_k1 = a_k2 + h[j]*b_k2
- Check which intersection gives the minimum max

This is essentially the "Gilbert algorithm" or "Chebyshev approximation" approach.

```python
def coord_descend_1d(base_corrs, b, h_lo, h_hi):
    """Find h_j in [h_lo, h_hi] to minimize max_k (base_corrs[k] + h_j * b[k])"""
    # The function is max of linear functions of h_j
    # The minimum of the max is at the intersection of the two "active" lines
    # that cross at the lowest point
    
    # Simple approach: ternary search on h_j (the max of linear functions is convex in h_j)
    lo, hi = h_lo, h_hi
    for _ in range(50):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f1 = np.max(base_corrs + m1 * b)
        f2 = np.max(base_corrs + m2 * b)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    h_opt = (lo + hi) / 2
    return h_opt
```

```python
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        t = 0.001
        return (m + t * np.log(np.sum(np.exp((corr - m) / t)))) * dx

        corr = np.correlate(h, 1.0 - h, mode="full")

        t = 0.001
        exps = np.exp((corr - m) / t)

        return g

    def con_eq_jac(h):
        return np.ones(n)

    cons = ({'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac},)

    s = np.sum(h_new)
    if s > 1e-12:
        h_new = h_new * (n / 2.0) / s
        h_new = np.clip(h_new, 0.0, 1.0)
    return h_new, compute_c5(h_new)

def _binary_init(n, seed):
    np.random.seed(seed)
    h = np.zeros(n)
    idx = np.random.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def _two_block_init(n, gap_frac, seed):
    np.random.seed(seed)
    h = np.zeros(n)
    n1 = n // 2
    half = n1 // 2
    gap = int(n * gap_frac)
    start = np.random.randint(0, max(1, n - 2 * half - gap))
    for i in range(half):
        h[(start + i) % n] = 1.0
        h[(start + half + gap + i) % n] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(160, 0.5)
    n = len(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent (strong, exact 1D minimax)
    remaining = budget_s - (time.time() - t0)
    if remaining > 120:
        h1, c5_1 = coordinate_descent(best_h.copy(), max_sweeps=200,
                                      t0=t0, time_limit=time.time() + min(350, remaining - 50))
        if c5_1 < best_c5:
            best_h, best_c5 = h1, c5_1

    # Phase 2: PGD with soft-max from best
    remaining = budget_s - (time.time() - t0)
    if remaining > 80:
        h2, c5_2 = pgd_optimize(best_h.copy(), max_iters=3000,
                                 t0=t0, time_limit=time.time() + min(200, remaining - 40))
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2

    # Phase 3: More coordinate descent
    remaining = budget_s - (time.time() - t0)
    if remaining > 80:
        h3, c5_3 = coordinate_descent(best_h.copy(), max_sweeps=150,
                                      t0=t0, time_limit=time.time() + min(250, remaining - 40))
        if c5_3 < best_c5:
            best_h, best_c5 = h3, c5_3

    # Phase 4: SLSQP polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 60:
        h4, c5_4 = slsqp_polish(best_h.copy(), t0, time.time() + min(150, remaining - 30))
        if c5_4 < best_c5:
            best_h, best_c5 = h4, c5_4

    # Phase 5: Final coordinate descent
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        h5, c5_5 = coordinate_descent(best_h.copy(), max_sweeps=100,
                                      t0=t0, time_limit=time.time() + min(remaining - 10, 200))
        if c5_5 < best_c5:
            best_h, best_c5 = h5, c5_5

    # Phase 6: Try different initializations if time remains
    remaining = budget_s - (time.time() - t0)
    n_inits = min(4, int(remaining // 80))
    for i in range(n_inits):
        remaining = budget_s - (time.time() - t0)
        if remaining < 80:
            break
        if i == 0:
            h_init = _binary_init(n, seed + 700)
        elif i == 1:
            h_init = _two_block_init(n, 0.25, seed + 800)
        elif i == 2:
            h_init = _two_block_init(n, 0.4, seed + 900)
        else:
            h_init = best_h + 0.1 * np.random.randn(n)
            h_init = project_feasible(h_init)
        hc, c5c = coordinate_descent(h_init, max_sweeps=80,
                                      t0=t0, time_limit=time.time() + min(70, remaining - 20))
        if c5c < best_c5:
            best_h, best_c5 = hc, c5c

    # Final: verify and return
    best_h = np.clip(best_h, 0.0, 1.0)
    s = np.sum(best_h)
    if s > 1e-12:
        best_h = best_h * (n / 2.0) / s
        best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
