def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()

                break

        h3, c3 = slsqp_opt(h2, n0, deadline3, beta=1000)
        update_best(h3, c3)
        deadline3b = t0 + budget_s * 0.65
        h3b, c3b = coord_descent(h3, n0, deadline3b)
        update_best(h3b, c3b)

    # Phase 2: Try Thue-Morse and other starts at n=192
    n_alt = 192
    deadline4 = t0 + budget_s * 0.75
    h_tm = project_feasible(thue_morse(n_alt) + rng.randn(n_alt) * 0.1, n_alt)
    h_tm2, c_tm2 = subgradient_opt(h_tm, n_alt, min(time.time() + 30, deadline4))
    update_best(h_tm2, c_tm2)
    h_tm3, c_tm3 = coord_descent(h_tm2, n_alt, deadline4)
    update_best(h_tm3, c_tm3)

    # Phase 3: Perturb best and refine
    if best_h is not None:
        nb = len(best_h)
        deadline5 = t0 + budget_s * 0.88
        for sigma in [0.05, 0.15, 0.3]:
            if time.time() > deadline5:
                break
            h_p = project_feasible(best_h + rng.randn(nb) * sigma, nb)
            hp1, cp1 = subgradient_opt(h_p, nb, min(time.time() + 40, deadline5), lr0=0.03)
            update_best(hp1, cp1)
            hp2, cp2 = coord_descent(hp1, nb, min(time.time() + 50, deadline5))
            update_best(hp2, cp2)

    # Phase 4: Final SLSQP + coord descent polish
    if best_h is not None and time.time() < t0 + budget_s * 0.92:
        nb = len(best_h)
        deadline6 = t0 + budget_s * 0.95
        h_f1, c_f1 = slsqp_opt(best_h, nb, deadline6, beta=2000)
        update_best(h_f1, c_f1)
        h_f2, c_f2 = coord_descent(h_f1, nb, t0 + budget_s * 0.97)
        update_best(h_f2, c_f2)

    # Phase 5: Try smaller n from best (downsample + optimize)
    if best_h is not None and time.time() < t0 + budget_s * 0.95:
        nb = len(best_h)
        if nb >= 256:
            n_small = nb // 2
            x_old = np.linspace(0, 2, nb, endpoint=False)
            x_new = np.linspace(0, 2, n_small, endpoint=False)
            h_ds = np.interp(x_new, x_old, best_h)
            h_ds = project_feasible(h_ds, n_small)
            h_d1, c_d1 = subgradient_opt(h_ds, n_small, t0 + budget_s * 0.97, lr0=0.02)
            update_best(h_d1, c_d1)
            h_d2, c_d2 = coord_descent(h_d1, n_small, t0 + budget_s * 0.99)
            update_best(h_d2, c_d2)

    # Final validation
    if best_h is None:
        n_fb = 128
        best_h = project_feasible(thue_morse(n_fb) + rng.randn(n_fb) * 0.05, n_fb)
        best_c5 = compute_c5(best_h)

    nf = len(best_h)
    best_h = project_feasible(best_h, nf)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    try:
        val = evaluate_erdos_solution(best_h)
        if val < best_c5:
            best_c5 = val
    except Exception:
        pass

    return (best_h, best_c5, nf)
