    c = _corr(h)
    cmax = c.max()
    shifted = c - cmax
    s = np.exp(shifted / eps)

            g[:n - m] -= wk * h[m:]
        else:
            p = -m
            g[:n - p] += wk * (1.0 - h[p:])
            g[p:] -= wk * h[:n - p]
    return g * dx

def _proj(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(500):
        d = n * 0.5 - h.sum()
        if abs(d) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h = np.clip(h + d / n, 0, 1)
            break
        h[free] += d / free.sum()
        h = np.clip(h, 0, 1)
    return h

def _eq_fun(x, n):
    return x.sum() - n * 0.5

def _eq_jac(x, n):
    return np.ones(n)

def _subgrad_topk(h, n, k):
    c = _corr(h)
    idx = np.argsort(c)[-k:]
    g = np.zeros(n)
    for ki in idx:
        m = ki - (n - 1)
        if m >= 0:
            g[m:] += (1.0 - h[:n - m])
            g[:n - m] -= h[m:]
        else:
            p = -m
            g[:n - p] += (1.0 - h[p:])
            g[p:] -= h[:n - p]
    return g / k * (2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _smoobj(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    shifted = c - cmax
    s = np.exp(shifted / eps)
    return (cmax + eps * np.log(s.sum())) * dx

def _smograd(h, eps):
