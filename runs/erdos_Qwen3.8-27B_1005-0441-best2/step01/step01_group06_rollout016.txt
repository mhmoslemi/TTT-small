import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum_box(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2} via bisection on constant shift."""
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        val = np.sum(np.clip(h + mid, 0.0, 1.0))
        if val > target:
            hi = mid
        else:
            lo = mid
    lam = (lo + hi) / 2.0
    return np.clip(h + lam, 0.0, 1.0)

def lse_objective(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    m = c.max()
    shifted = beta * (c - m)
    shifted -= shifted.max()
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def sum_fun(x, n):
    return np.sum(x) - n / 2.0

def sum_jac(x, n):
    return np.ones(n)

def do_slsqp(h_init, n, beta, maxiter, t0, deadline):
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac, 'args': (n,)}]
    try:
        if time.time() > deadline:
            return h_init, compute_c5(h_init)
        res = minimize(lse_objective, h_init, args=(n, beta),
                       method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_sum_box(res.x, n)
            return h_new, compute_c5(h_new)
    except Exception:
        pass
    return h_init, compute_c5(h_init)

def upsamp(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def downsamp(h, n_new):
    """Average-stride downsample."""
    step = len(h) // n_new
    h_d = np.array([np.mean(h[i*step:(i+1)*step]) for i in range(n_new)])
    return h_d

def optimize_at(n, h_start, t0, deadline):
    """Run multi-beta SLSQP at fixed resolution."""
    h = project_sum_box(h_start, n)
    best = h.copy()
    best_c5 = compute_c5(h)
    for beta in [2, 8, 25, 80, 250, 800, 2500, 8000, 25000]:
        if time.time() > deadline:
            break
        h_new, c5_new = do_slsqp(h, n, beta, 200, t0, deadline)
        if c5_new < best_c5:
            best = h_new.copy()
            best_c5 = c5_new
        h = h_new.copy()
    return best, best_c5
