    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def build_M(h_old, n):
    k_idx = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    idx = j + (n - 1) - k_idx
    W = np.zeros((2 * n - 1, n))
    valid = (idx >= 0) & (idx < n)
    W[valid] = 1.0 - h_old[idx[valid]]
    return W

def solve_lp(h_old, n, solver=None):
    dx = 2.0 / n
    M = build_M(h_old, n)
    h_var = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0,
        M @ h_var * dx <= t
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    kwargs = {}
    if solver:
        kwargs['solver'] = solver
    try:
        prob.solve(**kwargs)
    except Exception:
        return None
    if h_var.value is None:
        return None
    h_new = np.clip(h_var.value, 0.0, 1.0)
    s = np.sum(h_new)
    target = n / 2.0
    if s > 1e-12:
        h_new *= target / s
        h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def slp_optimize(h_init, max_iters=80, time_budget=300, solver=None):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    t0 = time.time()
    staller = 0
    for it in range(max_iters):
        if time.time() - t0 > time_budget:
            break
        h_new = solve_lp(h, n, solver)
        if h_new is None:
            break
        c5_full = compute_c5(h_new)
        improved = False
        if c5_full < best_c5 - 1e-12:
            best_h = h_new.copy()
            best_c5 = c5_full
            h = h_new.copy()
            improved = True
        else:
            for alpha in [0.5, 0.3, 0.15]:
                h_mix = alpha * h_new + (1.0 - alpha) * h
                h_mix = np.clip(h_mix, 0.0, 1.0)
                s = np.sum(h_mix)
                if s > 1e-12:
                    h_mix *= (n / 2.0) / s
                    h_mix = np.clip(h_mix, 0.0, 1.0)
                c5_mix = compute_c5(h_mix)
                if c5_mix < best_c5 - 1e-12:
                    best_h = h_mix.copy()
                    best_c5 = c5_mix
                    h = h_mix.copy()
                    improved = True
                    break
        if not improved:
            staller += 1
            if staller >= 8:
                break
        else:
            staller = 0
    return best_h, best_c5

def _slsqp_obj(h):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    t = max(_g_temp, 1e-10)
    return (m + t * np.log(np.sum(np.exp((corr - m) / t)))) * dx

def _ceq(h):
    return np.sum(h) - _g_n / 2.0

def _ceq_jac(h):
    return np.ones(_g_n)

def slsqp_polish(h_init, max_time=80):
    global _g_n, _g_temp
    n = len(h_init)
    _g_n = n
    t0 = time.time()
    best_h = h_init.copy()
    best_val = compute_c5(best_h)
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': _ceq, 'jac': _ceq_jac},)
    temps = [0.01, 0.003, 0.001, 0.0003, 0.0001]
    for temp in temps:
        if time.time() - t0 > max_time - 5:
            break
        _g_temp = temp
        h = best_h.copy()
        for it in range(4):
            if time.time() - t0 > max_time - 2:
                break
            try:
                res = minimize(_slsqp_obj, h, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 300, 'ftol': 1e-13})
                h_new = np.clip(res.x, 0.0, 1.0)
                s = np.sum(h_new)
                if s > 1e-12:
                    h_new *= (float(n) / 2.0) / s
                    h_new = np.clip(h_new, 0.0, 1.0)
                v = compute_c5(h_new)
                if v < best_val - 1e-12:
                    best_h = h_new.copy()
                    best_val = v
                h = h_new.copy()
            except Exception:
                break
    return best_h, best_val

def _upsample(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old, dtype=float) * 2.0 / n_old
    x_new = np.arange(n_new, dtype=float) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    s = np.sum(h_new)
    if s > 1e-12:
        h_new *= (float(n_new) / 2.0) / s
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def _perturb(h, scale, rng):
    h_p = h + scale * rng.standard_normal(len(h))
    h_p = np.clip(h_p, 0.0, 1.0)
    s = np.sum(h_p)
    target = float(len(h)) / 2.0
    if s > 1e-12:
        h_p *= target / s
        h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def _try_solvers(n):
    """Find a working LP solver for the problem size."""
    h_test = np.full(n, 0.5)
    for sol in [None, cp.ECOS, cp.SCS, cp.PDLP, cp.CBC, cp.GLPK]:
        try:
            t_s = time.time()
            res = solve_lp(h_test, n, solver=sol)
            if res is not None and time.time() - t_s < 10:
                return sol
        except Exception:
            continue
    return None

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t_start = time.time()

    def _remaining():
        return budget_s - (time.time() - t_start)

    # Get initial construction
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(160, 0.5)
    n = len(best_h)
    best_c5 = compute_c5(best_h)

    # Find a good LP solver
    solver = _try_solvers(min(n, 100))

    # Phase 1: SLP optimization at starting n
    if _remaining() > 100:
        h_slp, c5_slp = slp_optimize(best_h.copy(), max_iters=80,
                                     time_budget=min(250, _remaining() - 60), solver=solver)
        if c5_slp < best_c5:
            best_h = h_slp
            best_c5 = c5_slp
            n = len(best_h)

    # Phase 2: SLSQP polish
    if _remaining() > 60:
        h_pol, c5_pol = slsqp_polish(best_h.copy(), max_time=min(100, _remaining() - 30))
        if c5_pol < best_c5:
            best_h = h_pol
            best_c5 = c5_pol
            n = len(best_h)

    # Phase 3: SLP at higher resolution
    for target_n in [n + 80, min(n * 2, 400)]:
        if _remaining() < 80 or target_n == n or target_n > 500:
            continue
        h_up = _upsample(best_h, target_n)
        h_slp2, c5_slp2 = slp_optimize(h_up, max_iters=50,
                                       time_budget=min(120, _remaining() - 40), solver=solver)
        if c5_slp2 < best_c5 - 1e-6:
            best_h = h_slp2
            best_c5 = c5_slp2
            n = len(best_h)
        if _remaining() < 40:
            break

    # Phase 4: SLSQP polish at best resolution
    if _remaining() > 40:
        n = len(best_h)
        h_pol2, c5_pol2 = slsqp_polish(best_h.copy(), max_time=min(80, _remaining() - 20))
        if c5_pol2 < best_c5:
            best_h = h_pol2
            best_c5 = c5_pol2
            n = len(best_h)

    # Phase 5: Random restarts with SLP
    restart = 0
    while _remaining() > 40 and restart < 30:
        scale = 0.02 + 0.05 * (restart % 5) / 4.0
        h_pert = _perturb(best_h, scale, rng)
        max_t = min(40, _remaining() - 15)
        if max_t < 10:
            break
        h_r, c5_r = slp_optimize(h_pert, max_iters=30, time_budget=max_t, solver=solver)
        if c5_r < best_c5 - 1e-10:
            best_h = h_r
            best_c5 = c5_r
            n = len(best_h)
        restart += 1

    # Phase 6: Final SLSQP polish
    if _remaining() > 15:
        n = len(best_h)
        h_final, c5_final = slsqp_polish(best_h.copy(), max_time=max(10, _remaining() - 3))
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

    # Ensure feasibility
    n = len(best_h)
    s = np.sum(best_h)
    if s > 1e-12:
        best_h *= (float(n) / 2.0) / s
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
