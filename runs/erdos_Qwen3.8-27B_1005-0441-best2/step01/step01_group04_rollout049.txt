def _fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0

        else:

def _subgrad_h(h, s):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        js = j + s
        jm = j - s
        if 0 <= js < n:
            g += 1.0 - h[js]
        if 0 <= jm < n:
            g -= h[jm]
        grad[j] = dx * g
    return grad

def _fit_fourier(h, M):
    n = len(h)
    x = np.linspace(0.0, 2.0, n, endpoint=[True, False][1])  # endpoint=False
    A = np.zeros((n, 2*M))

def _sin_init(n, freq):
    x = np.linspace(0, 2.0 * np.pi * freq, n, endpoint=False)
    h = 0.5 + 0.3 * np.sin(x) + 0.15 * np.sin(2.0 * x + 1.0)
    return _fix_sum(h, n)

def _block_init(n):
    period = max(3, n // 20)
    h = np.array([0.8 if (i % period) < int(period * 0.6) else 0.2 for i in range(n)])
    return _fix_sum(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    best_c5 = np.inf
    best_h = None

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = np.full(80, 0.5)

    h_init = _fix_sum(h_init, len(h_init))
    best_h = h_init.copy()
    best_c5 = _eval_c5(h_init)

    # Phase 1: Subgradient with momentum
    if time.time() - t0 < budget_s - 400:
        h1, c51 = _subgrad_optimize(best_h, 6000, 0.06, 3, 0.85)
        if c51 < best_c5:
            best_h, best_c5 = h1, c51

    # Phase 2: Fourier refinement
    n = len(best_h)
    if time.time() - t0 < budget_s - 350:
        for M in [4, 6, 8, 12]:
            if time.time() - t0 > budget_s - 350:
                break
            coeffs = _fit_fourier(best_h, M)
            try:
                res = minimize(_FObj(M, n), coeffs, method='Nelder-Mead',
                               options={'maxiter': 600, 'xatol': 1e-9, 'fatol': 1e-11})
                h_f = np.clip(_fourier_h(res.x, M, n), 0.0, 1.0)
                h_f = _fix_sum(h_f, n)
                c5_f = _eval_c5(h_f)
                if c5_f < best_c5:
                    best_h, best_c5 = h_f, c5_f
            except Exception:
                pass

    # Phase 3: SLSQP with increasing alpha
    n = len(best_h)
    if time.time() - t0 < budget_s - 250:
        for alpha in [60, 120, 250, 500]:
            if time.time() - t0 > budget_s - 250:
                break
            h_s = _slsqp_optimize(best_h, alpha, n, 350)
            h_s = _fix_sum(h_s, n)
            c5_s = _eval_c5(h_s)
            if c5_s < best_c5:
                best_h, best_c5 = h_s, c5_s

    # Phase 4: More subgradient with higher top_k
    if time.time() - t0 < budget_s - 200:
        for tk in [5, 7, 10]:
            if time.time() - t0 > budget_s - 200:
                break
            h4, c54 = _subgrad_optimize(best_h, 4000, 0.04, tk, 0.9)
            if c54 < best_c5:
                best_h, best_c5 = h4, c54

    # Phase 5: Perturbation exploration
    remaining = budget_s - (time.time() - t0)
    n_tests = min(int(remaining / 50), 10)
    for i in range(n_tests):
        if time.time() - t0 > budget_s - 60:
            break
        n = len(best_h)
        scale = 0.015 + 0.01 * i
        h_p = _fix_sum(best_h + rng.normal(0, scale, n), n)
        hp, c5p = _subgrad_optimize(h_p, 2000, 0.05, 3, 0.85)
        if c5p < best_c5:
            best_h, best_c5 = hp, c5p
        if time.time() - t0 < budget_s - 80:
            hs = _slsqp_optimize(hp, 80, n, 200)
            hs = _fix_sum(hs, n)
            c5s = _eval_c5(hs)
            if c5s < best_c5:
                best_h, best_c5 = hs, c5s

    # Phase 6: Alternative starting points and smaller n
    if time.time() - t0 < budget_s - 120:
        starts = [_sin_init(80, 2), _sin_init(80, 3), _block_init(80)]
        for h0 in starts:
            if time.time() - t0 > budget_s - 120:
                break
            h6, c56 = _subgrad_optimize(h0, 3000, 0.06, 3, 0.85)
            if c56 < best_c5:
                best_h, best_c5 = h6, c56

    # Phase 7: Try different resolutions
    if time.time() - t0 < budget_s - 100:
        for n_try in [40, 60, 100, 120]:
            if time.time() - t0 > budget_s - 100:
                break
            n_cur = len(best_h)
            if n_try == n_cur:
                continue
            fr = np.linspace(0, n_cur - 1, n_try)
            h0 = np.interp(fr, np.arange(n_cur), best_h)
            h0 = _fix_sum(h0, n_try)
            h7, c57 = _subgrad_optimize(h0, 3000, 0.06, 3, 0.85)
            if c57 < best_c5:
                best_h, best_c5 = h7, c57
            if time.time() - t0 < budget_s - 60:
                hs7 = _slsqp_optimize(h7, 100, n_try, 250)
                hs7 = _fix_sum(hs7, n_try)
                c5s7 = _eval_c5(hs7)
                if c5s7 < best_c5:
                    best_h, best_c5 = hs7, c5s7

    # Final polish
    if time.time() - t0 < budget_s - 30:
        hf, c5f = _subgrad_optimize(best_h, 3000, 0.02, 5, 0.92)
        if c5f < best_c5:
            best_h, best_c5 = hf, c5f

    best_h = _fix_sum(best_h, len(best_h))
    best_c5 = _eval_c5(best_h)
    return best_h, best_c5, len(best_h)
