    m = np.max(corr)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_obj(h, alpha, mu):
    n = len(h)
    dx = 2.0 / n
    cvals = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(cvals)
    shifted = alpha * (cvals - m)
    shifted = np.clip(shifted, -500, 500)
    obj = (m + np.log(np.sum(np.exp(shifted))) / alpha)
    obj += mu * (np.sum(h) - 0.5 * n) ** 2
    return obj

def smooth_obj_grad(h, alpha, mu):
    n = len(h)
    dx = 2.0 / n
    cvals = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(cvals)
    shifted = alpha * (cvals - m)
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    w /= np.sum(w)
    v = 1.0 - 2.0 * h
    corr = np.correlate(w, v, mode='full')
    grad = dx * corr[n-1 : 2*n-1]
    grad += 2.0 * mu * (np.sum(h) - 0.5 * n) * np.ones(n)
    return grad

def subgrad_step(h, eta, n_worst=5):
    n = len(h)
    cvals = compute_c5_vals(h)
    idx = np.argsort(cvals)[-n_worst:]
    # Subgradient is average gradient of worst constraints
    # For exact max, any weighted avg of active constraint gradients is a subgradient
    grads = []
    for k in idx:
        m = k  # index in the full correlate
        L = max(0, m - n + 1)
        R = min(n - 1, m)
        g = np.zeros(n)
        for j in range(n):
            mpj = m - j
            if L <= j <= R and 0 <= mpj <= n-1:
                g[j] += (1.0 - h[mpj])
            if L <= mpj <= R and 0 <= j <= n-1:
                g[j] -= h[mpj]
        grads.append(g)
    grad = np.mean(grads, axis=0)

- Term 2 range: $j \in [m-n+1, n-1]$ (since $\max(0, m-n+1) = m-n+1$ and $\min(n-1, m) = n-1$)

$g_m[j] = (1 - 2h[m-j]) \cdot \mathbf{1}[j \in J_m]$

$g_m[j] = u[m-j]$ for $j \in J_m$, and 0 otherwise.

This is a Toeplitz-like structure. For the subgradient (average of worst m's), the gradient is a sum of shifted copies of u, which can be computed efficiently.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target n * 0.5
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_obj(h, alpha, mu):
    n = len(h)
    dx = 2.0 / n
    cvals = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(cvals)
    shifted = alpha * (cvals - m)
    shifted = np.clip(shifted, -500, 500)
    val = m + np.log(np.sum(np.exp(shifted))) / alpha
    val += mu * (np.sum(h) - 0.5 * n) ** 2
    return val

def smooth_obj_grad(h, alpha, mu):
    n = len(h)
    dx = 2.0 / n
    cvals = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(cvals)
    shifted = alpha * (cvals - m)
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    w /= np.sum(w)
    v = 1.0 - 2.0 * h
    corr = np.correlate(w, v, mode='full')
    grad = dx * corr[n-1:2*n-1]
    grad += 2.0 * mu * (np.sum(h) - 0.5 * n) * np.ones(n)
    return grad

def subgradient_repair(h, n, n_steps=200, n_worst=8):
    best = h.copy()
    best_c5 = compute_c5(h)
    for step in range(n_steps):
        cvals = compute_c5_vals(h) * 1.0
        dx = 2.0 / n
        cvals = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = np.argsort(cvals)[-n_worst:]
        wts = np.zeros(len(cvals))
        wts[idx] = 1.0 / len(idx)
        v = 1.0 - 2.0 * h
        corr = np.correlate(wts, v, mode='full')
        grad = dx * corr[n-1:2*n-1]
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        eta = 0.01 * best_c5 / (gn + 1e-15)
        eta = min(eta, 0.05)
        h_new = h - eta * grad + (0.5 * (np.sum(h) - 0.5 * n) / n) * np.ones(n)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = h_new.copy()
            h = h_new
        else:
            h = best + 0.005 * np.random.standard_normal(n)
            h = project_feasible(h, n)
    return best, best_c5

def make_two_block(n):
    h = np.zeros(n)
    b = n // 4
    h[:b] = 1.0
    h[-b:] = 1.0
    return project_feasible(h, n)

def make_periodic(n, period):
    h = np.zeros(n)
    for i in range(n):
        m = i % period
        if m < period // 2:
            h[i] = 1.0
    return project_feasible(h, n)

        if c5_sub < c5_best:

    # Phase 3: More L-BFGS-B passes from best

def compute_c5_vals(h):
