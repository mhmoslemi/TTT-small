def _sqp_step(h0, n):
    """One SQP iteration: linearize and solve LP"""
    from scipy.optimize import linprog
    # c_k(h) ≈ c_k(h0) + g_k^T (h - h0) where g_k = ∂c_k/∂h at h0
    # Minimize t s.t. g_k^T h + (c_k(h0) - g_k^T h0) <= t, 0<=h<=1, sum(h)=n/2
    # Variables: [h(0), ..., h(n-1), t]
    # Minimize: t (index n)
    # Constraints:
    #   sum(h) = n/2
    #   g_k^T h - t <= -(c_k(h0) - g_k^T h0) for each k  [inequality]
    #   0 <= h_i <= 1 [bounds]
    
    c0 = _corr(h0)  # length 2n-1
    n_corr = len(c0)
    
    # Compute gradients g_k for each k
    # ... this is 2n-1 vectors of length n
    # This might be expensive for large n, but for n=70, 2n-1=139
    
    # Actually, let me compute the gradient matrix G where G[k, j] = ∂c_k/∂h_j
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):

            G[k, :n-m] += (1.0 - h0[m:])  # from h[i] term
            G[k, m:] -= h0[:n-m]           # from -h[i]*h[i+m] term, h[i+m] part
        else:
            p = -m
            G[k, :n-p] -= h0[p:]           # from -h[i]*h[i+p] term  
            G[k, p:] += (1.0 - h0[:n-p])   # from h[i+p] term
    
    # LP: min t
    # vars: h[0..n-1], t (index n)
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0  # minimize t
    
    # Inequality constraints: G[k] @ h - t <= -(c0[k] - G[k] @ h0)
    # i.e., G[k] @ h - t + G[k] @ h0 - c0[k] <= 0
    # A_ub @ vars <= b_ub
    A_ub = np.zeros((2*n-1, n+1))
    for k in range(2*n-1):

        A_ub[k, n] = -1.0
    b_ub = np.zeros(2*n-1)
    for k in range(2*n-1):
        b_ub[k] = -(c0[k] - G[k] @ h0)
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n*0.5])
    
    bounds = [(0,1)]*n + [(None, None)]  # h bounded, t unbounded
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.success:
        h_new = _proj(res.x[:n], n)
        return h_new, res.x[n]
    return h0, float('inf')

        h_new = _proj(h - lr * g, n)
        v = _c5(h_new)
        if v < best - 1e-15:
            best = v
            h = h_new
            lr = min(lr * 1.03, 0.04)
        else:
            lr *= 0.85
            if lr < 1e-10:
                break
    return h, best

def _coord(h, n, t0, budget, frac):
    best = _c5(h)
    h = h.copy()
    step = 0.015
    for _ in range(30):
        if time.time() - t0 > budget * frac:
            break
        imp = False
        for i in range(n):
            for d in [step, -step]:
                h2 = h.copy()
                h2[i] = np.clip(h2[i] + d, 0, 1)
                h2 = _proj(h2, n)
                if _c5(h2) < best - 1e-14:
                    h = h2
                    best = _c5(h2)
                    imp = True
                    break
        if not imp:
            step *= 0.5
            if step < 1e-5:
                break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_val = np.inf
    n_use = 70

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        h_init = _proj(h_init, len(h_init))
    except Exception:
        h_init = None

    cands = []
    if h_init is not None:
        cands.append(h_init)
        ni = len(h_init)
        for sc in [0.01, 0.02, 0.04, 0.07, 0.12]:
            for _ in range(2):
                hp = _proj(h_init + rng.randn(ni) * sc, ni)
                if _c5(hp) < 0.42:
                    cands.append(hp)

    for a in [0.25, 0.35, 0.45, 0.55]:
        for f1 in [1, 2, 3]:
            for f2 in [2, 3, 5]:
                x = np.arange(n_use)
                hh = (0.5 + a * np.sin(2 * np.pi * f1 * x / n_use)
                      + 0.4 * a * np.cos(2 * np.pi * f2 * x / n_use + 0.5))
                hh = _proj(hh, n_use)
                if _c5(hh) < 0.44:
                    cands.append(hh)

    for _ in range(4):
        cands.append(_proj(rng.rand(n_use), n_use))

    cands.sort(key=lambda h: _c5(h))
    uniq = [cands[0]]
    for c in cands[1:]:
        if len(uniq) >= 12:
            break
        if all(np.linalg.norm(c - u) > 0.01 or len(c) != len(u) for u in uniq):
            uniq.append(c)
    cands = uniq

    eps_list = [0.1, 0.05, 0.025, 0.012, 0.006, 0.003, 0.0015]
    for h0 in cands:
        if time.time() - t0 > budget_s * 0.50:
            break
        nn = len(h0)
        hl = _lbfgs_h(h0, nn, eps_list, t0, budget_s, 0.48)
        v = _c5(hl)
        if v < best_val:
            best_val = v
            best_h = hl.copy()

    if best_h is not None:
        nn = len(best_h)
        for eps in [0.004, 0.002, 0.001, 0.0005]:
            if time.time() - t0 > budget_s * 0.80:
                break
            hn = _slsqp_h(best_h, nn, eps, 500)
            if _c5(hn) < best_val:
                best_val = _c5(hn)
                best_h = hn

    if best_h is not None:
        best_h, best_val = _subgrad(best_h, len(best_h), t0, budget_s, 0.93)
        if time.time() - t0 < budget_s * 0.96:
            best_h, best_val = _coord(best_h, len(best_h), t0, budget_s, 0.97)

    if best_h is None:
        best_h = np.full(n_use, 0.5)
        best_val = _c5(best_h)

    best_h = _proj(best_h, len(best_h))
    best_val = _c5(best_h)
    return (best_h, best_val, len(best_h))
