import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _proj_bisect(z, n):
    target = n * 0.5
    lo, hi = z.min() - 1.0, z.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        ht = np.clip(z - mid, 0.0, 1.0)
        if ht.sum() > target:
            lo = mid
        else:
            hi = mid
    h_out = np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)
    d = target - h_out.sum()
    if abs(d) > 1e-14:
        h_out += d / n
        h_out = np.clip(h_out, 0.0, 1.0)
    return h_out

def _subgrad(h, n, tol):
    dx = 2.0 / n
    c = _corr(h) * dx
    cmax = c.max()
    w = np.zeros(2 * n - 1)
    active = c >= cmax - tol
    if active.sum() == 0:
        active[np.argmax(c)] = True
    if tol > 0:
        w[active] = np.exp((c[active] - cmax) / tol)
        w /= w.sum()
    else:
        w[active] = 1.0 / active.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        m = k - (n - 1)
        if m >= 0:
            g[m:] += wk * (1.0 - h[:n - m])
            g[:n - m] += -wk * h[m:]
        else:
            p = -m
            g[:n - p] += wk * (1.0 - h[p:])
            g[p:] += -wk * h[:n - p]
    return g

def _grad_corr_lag(h, n, m):
    """Grad of c_m = sum h[i]*(1-h[i+m]) for m>=0."""
    g = np.zeros(n)
    if m >= 0:
        g[m:] += (1.0 - h[:n - m])
        g[:n - m] -= h[m:]
    else:
        p = -m
        g[:n - p] += (1.0 - h[p:])
        g[p:] -= h[:n - p]
    return g

def _smooth_obj(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h) * dx
    cmax = c.max()
    shifted = c - cmax
    return cmax + eps * np.log(np.exp(shifted / eps).sum())

def _smooth_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h) * dx
    cmax = c.max()
    shifted = c - cmax
    s = np.exp(shifted / eps)
    ssum = s.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = s[k] / ssum
        if wk < 1e-14:
            continue
        m = k - (n - 1)
        if m >= 0:
            g[m:] += wk * (1.0 - h[:n - m])
            g[:n - m] += -wk * h[m:]
        else:
            p = -m
            g[:n - p] += wk * (1.0 - h[p:])
            g[p:] += -wk * h[:n - p]
    return g

def _subgrad_loop(h0, t_deadline, lr0, decay, tol_start, tol_end, momentum=0.9):
    n = len(h0)
    h = h0.copy().astype(float)
    v = np.zeros(n)
    best = h.copy()
    best_cv = _c5(h)
    lr = lr0
    stuck = 0
    total = max(1, int((t_deadline - time.time()) / 0.02))
    for it in range(total):
        if time.time() > t_deadline:
            break
        tol = tol_start + (tol_end - tol_start) * (it / total)
        g = _subgrad(h, n, tol)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        v = momentum * v + (1.0 - momentum) * g / gn
        step = lr / max(np.linalg.norm(v), 0.5)
        h_new = _proj_bisect(h - step * v, n)
        cv = _c5(h_new)
        if cv < best_cv - 1e-10:
            best_cv = cv
            best = h_new.copy()
            h = h_new
            stuck = 0
            lr = min(lr * 1.01, lr0)
        else:
            h = h_new
            stuck += 1
            lr *= decay
            if stuck > 200:
                pert = np.random.randn(n) * 0.002
                h = _proj_bisect(h + pert, n)
                lr = max(lr, 0.005)
                stuck = 0
            if lr < 1e-9:
                break
    return best, best_cv

def _lbfgs_smooth(h0, eps, maxit=300):
    n = len(h0)
    try:
        res = minimize(_smooth_obj, h0.copy(), args=(eps,),
                       jac=_smooth_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxit, 'ftol': 1e-15, 'gtol': 1e-8})
        return _proj_bisect(res.x, n)
    except Exception:
        return h0.copy()

def _upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_ext = np.concatenate([h, h])
    h_interp = np.interp(x_new, np.concatenate([x_old, x_old + 2.0]), h_ext)
    return _proj_bisect(h_interp, n_new)
