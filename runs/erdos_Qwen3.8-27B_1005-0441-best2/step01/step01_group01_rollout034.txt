            h_up = slsqp_opt(h_up, n2, 150.0, 800)
            h_up = slsqp_opt(h_up, n2, 400.0, 800)
            c5_up = compute_c5(h_up)
            if c5_up < c5_best:
                h_best = h_up
                c5_best = c5_up
                n = n2

    # Final feasibility check
    h_best = project_feasible(h_best, len(h_best))
    c5_best = compute_c5(h_best)
    return h_best, c5_best, len(h_best)
