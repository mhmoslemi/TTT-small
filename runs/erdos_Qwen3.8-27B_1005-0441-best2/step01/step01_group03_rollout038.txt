def lp_subgradient(h, n, max_step=1.0):
    """Solve LP subproblem to get next iterate."""
    dx = 2.0 / n
    ov = get_overlaps(h, n)
    A = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        lag = m - (n-1)
        for j in range(n):
            gm = 0.0
            jr = j + lag
            if 0 <= jr < n:
                gm += (1.0 - h[jr])
            jl = j - lag
            if 0 <= jl < n:
                gm -= h[jl]
            A[m, j] = gm * dx
    
    b = ov - A @ h  # b[m] = ov[m] - A[m,:] @ h
    
    var_h = cp.Variable(n)
    var_t = cp.Variable()
    constraints = [
        A @ var_h - var_t <= -b,
        var_h >= 0,
        var_h <= 1,
        cp.sum(var_h) == n/2
    ]
    obj = cp.Minimize(var_t)
    prob = cp.Problem(obj, constraints)
    prob.solve()
    
    if var_t.value is None or var_h.value is None:
        return h, ov
    
    h_new = np.clip(var_h.value, 0, 1)
    h_new = project_feasible(h_new, n)
    
    # Trust region: limit step size
    step = np.linalg.norm(h_new - h)
    if step > max_step:
        h_new = h + max_step / step * (h_new - h)
        h_new = project_feasible(h_new, n)

def binary_local_search(h, n, max_passes=30):
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    s = int(h_bin.sum())
    if s < target:
        zeros = np.where(h_bin < 0.5)[0]
        for i in range(target - s):
            h_bin[zeros[i]] = 1.0
    elif s > target:
        ones = np.where(h_bin > 0.5)[0]
        for i in range(s - target):
            h_bin[ones[i]] = 0.0
    best_c5 = get_c5(h_bin, n)
    best_h = h_bin.copy()
    for _ in range(max_passes):
        improved = False
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        for i in zeros:
            for j in ones:
                test = best_h.copy()
                test[i] = 1.0
                test[j] = 0.0
                c = get_c5(test, n)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    best_h = test
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False) + 1.0 / n_src
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False) + 1.0 / n_dst
    h_interp = np.interp(x_dst, x_src, h_src)
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return project_feasible(h_interp, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    best_h = None
    best_c5 = 1.0
    best_n = 0

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = None

    # Phase 1: LP optimization on initial construction (35% budget)
    if h0 is not None:
        n0 = len(h0)
        h_work = project_feasible(h0.copy(), n0)
        c_init = get_c5(h_work, n0)
        best_h, best_c5, best_n = h_work.copy(), c_init, n0
        d1 = t0 + budget_s * 0.35
        while time.time() < d1:
            h_new, ok = lp_step(h_work, n0, max_step=0.35)
            if not ok:
                break
            c_new = get_c5(h_new, n0)
            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
                best_h = h_new.copy()
            h_work = h_new
        # Subgradient polish
        d1b = t0 + budget_s * 0.42
        if time.time() < d1b:
            bh, bc5 = subgrad_optimize(best_h, best_n, d1b, rng)
            if bc5 < best_c5:
                best_c5 = bc5
                best_h = bh

    # Phase 2: Binary local search (13% budget)
    d2 = t0 + budget_s * 0.55
    if best_h is not None and time.time() < d2:
        bh, bc5 = binary_local_search(best_h, best_n, max_passes=40)
        if bc5 < best_c5:
            best_c5 = bc5
            best_h = bh.copy()
        # Continue LP from binary solution
        d2b = t0 + budget_s * 0.60
        h_work = best_h.copy()
        while time.time() < d2b:
            h_new, ok = lp_step(h_work, best_n, max_step=0.25)
            if not ok:
                break
            c_new = get_c5(h_new, best_n)
            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
                best_h = h_new.copy()
            h_work = h_new

    # Phase 3: Try other n values (25% budget)
    n_candidates = [44, 48, 50, 52, 54, 56, 60, 64]
    if h0 is not None:
        n_candidates = [x for x in n_candidates if x != len(h0)]
    d3_end = t0 + budget_s * 0.85
    if n_candidates:
        time_per = (d3_end - time.time()) / len(n_candidates)
        time_per = max(time_per, 5.0)
        for n_try in n_candidates:
            if time.time() > d3_end:
                break
            d_try = min(time.time() + time_per, d3_end)
            if best_h is not None and best_n != n_try:
                h_init = interpolate_h(best_h, best_n, n_try)
            else:
                h_init = np.zeros(n_try)
                h_init[:n_try // 2] = 1.0
                h_init = project_feasible(h_init, n_try)
            # LP iterations
            d_lp = min(time.time() + time_per * 0.5, d_try - 3.0)
            h_work = h_init.copy()
            while time.time() < d_lp:
                h_new, ok = lp_step(h_work, n_try, max_step=0.3)
                if not ok:
                    break
                h_work = h_new
            # Subgradient + binary search
            bh, bc5 = subgrad_optimize(h_work, n_try, min(time.time() + 3.0, d_try - 2.0), rng)
            bh2, bc52 = binary_local_search(bh, n_try, max_passes=10)
            if bc52 < bc5:
                bh, bc5 = bh2, bc52
            if bc5 < best_c5:
                best_c5 = bc5
                best_h = bh.copy()
                best_n = n_try

    # Phase 4: Final refinement (remaining budget)
    d4 = t0 + budget_s - 5
    if best_h is not None and time.time() < d4:
        h_work = best_h.copy()
        while time.time() < d4 - 10:
            h_new, ok = lp_step(h_work, best_n, max_step=0.2)
            if not ok:
                break
            c_new = get_c5(h_new, best_n)
            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
                best_h = h_new.copy()
            h_work = h_new
        if time.time() < d4 - 5:
            bh, bc5 = binary_local_search(best_h, best_n, max_passes=20)
            if bc5 < best_c5:
                best_c5 = bc5
                best_h = bh.copy()

    h_final = project_feasible(best_h, best_n)
    c5_final = get_c5(h_final, best_n)
    dx_f = 2.0 / best_n
    if abs(np.sum(h_final) * dx_f - 1.0) > 1e-6:
        h_final = project_feasible(best_h, best_n)
        c5_final = get_c5(h_final, best_n)
    return h_final, float(c5_final), best_n
