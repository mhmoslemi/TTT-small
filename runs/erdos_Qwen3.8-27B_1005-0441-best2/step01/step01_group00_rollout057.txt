import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _smoobj(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    shifted = c - cmax
    s = np.exp(shifted / eps)
    return (cmax + eps * np.log(s.sum())) * dx

def _smograd(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    shifted = c - cmax
    s = np.exp(shifted / eps)
    ssum = s.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = s[k] / ssum
        if wk < 1e-14:
            continue
        m = k - (n - 1)
        if m >= 0:
            g[m:] += wk * (1.0 - h[:n - m])
            g[:n - m] -= wk * h[m:]
        else:
            p = -m
            g[:n - p] += wk * (1.0 - h[p:])
            g[p:] -= wk * h[:n - p]
    return g * dx

def _proj(h, n):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        d = n * 0.5 - h.sum()
        if abs(d) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            h = np.clip(h + d / n, 0, 1)
            break
        h[free] += d / free.sum()
        h = np.clip(h, 0, 1)
    return h

    h_prev = h0.copy()
    t = 1.0

    L = 8.0 * np.max(_smograd(h0, eps)) ** 2 + 1.0
    alpha = 1.0 / L

            break
        # FISTA extrapolation
        if it > 0:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = h + ((t - 1.0) / t_new) * (h - h_prev)
            t = t_new
        else:
            y = h.copy()
        
        g = _smograd(y, eps)
        h_new = _proj(y - alpha * g, n)

        f_new = _smoobj(h_new, eps)
        f_y = _smoobj(y, eps)
        if f_new > f_y + 0.5 * alpha * np.dot(g, h_new - y):
            alpha *= 0.5
            h_new = _proj(y - alpha * g, n)
        
        h_prev = h
        h = h_new
    
    return h

def _coord_descent(h, n, iters, t_max):
    """Coordinate descent: for each coordinate, find optimal value."""
    h = h.copy()
    dx = 2.0 / n
    
    for it in range(iters):

        improved = False

            h[j] = best_j_val
            if abs(best_j_val - old_val) > 1e-8:

        if not improved:

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _smoobj(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    s = np.exp((c - cmax) / eps)
    return (cmax + eps * np.log(s.sum())) * dx

def _smograd(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    s = np.exp((c - cmax) / eps)
    sw = s / s.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = sw[k]
        if wk < 1e-15:
            continue
        m = k - (n - 1)
        if m >= 0:
            nm = n - m
            g[:nm] += wk * (1.0 - h[m:])
            g[m:] -= wk * h[:nm]
        else:
            p = -m
            g[:n - p] += wk * (1.0 - h[p:])
            g[p:] -= wk * h[:n - p]
    return g * dx

def _proj(h, n):
    h = h.astype(float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(20000):
        d = n * 0.5 - h.sum()
        if abs(d) < 1e-15:
            break
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        nf = free.sum()
        if nf == 0:
            h += d / n
            np.clip(h, 0.0, 1.0, out=h)
            break
        h[free] += d / nf
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _fista_solve(h0, n, eps, niter, deadline):
    h = h0.copy()
    hp = h0.copy()
    tc = 1.0
    g0 = _smograd(h, eps)
    alpha = 1.0 / (np.linalg.norm(g0) + 1.0)
    for it in range(niter):
        if time.time() > deadline:
            break
        if it > 0:
            tn = (1.0 + np.sqrt(1.0 + 4.0 * tc * tc)) / 2.0
            y = h + ((tc - 1.0) / tn) * (h - hp)
            tc = tn
            y = np.clip(y, 0.0, 1.0)
        else:
            y = h.copy()
        gy = _smograd(y, eps)
        for _ in range(50):
            xn = _proj(y - alpha * gy, n)
            if (_smoobj(xn, eps) <= _smoobj(y, eps)
                    + np.dot(gy, xn - y)
                    + 1.5 * alpha * np.dot(xn - y, xn - y)):
                break
            alpha *= 0.5
        hp, h = h, xn
    return h

def _cd_update(c, h, j, old, new, n, sign):
    d = sign * (new - old)
    c[n - 1] += d * (1.0 - new - old)
    if n - 1 - j > 0:
        c[n:2*n-1-j] += d * (1.0 - h[j+1:n])
    if j > 0:
        c[n-1-j:n-1] += d * (1.0 - h[0:j])

def _coord_cd(h, n, nsw, deadline, npts=41):
    h = h.copy()
    c = _corr(h).copy()
    grid = np.linspace(0.0, 1.0, npts)
    for sw in range(nsw):
        if time.time() > deadline:
            break
        imp = False
        for j in range(n):
            if time.time() > deadline:
                break
            old = h[j]
            bv = c.max()
            jv = old
            for cv in grid:
                if abs(cv - old) < 1e-10:
                    continue
                _cd_update(c, h, j, old, cv, n, 1.0)
                nm = c.max()
                if nm < bv - 1e-14:
                    bv = nm
                    jv = cv
                _cd_update(c, h, j, old, cv, n, -1.0)
            if jv != old:
                h[j] = jv
                _cd_update(c, h, j, old, jv, n, 1.0)
                imp = True
        if not imp:
            break
    h = _proj(h, n)
    return h

def _local_cd(h, n, nsw, deadline, npts=13):
    h = h.copy()
    c = _corr(h).copy()
    for sw in range(nsw):
        if time.time() > deadline:
            break
        imp = False
        for j in range(n):
            if time.time() > deadline:
                break
            old = h[j]
            lo = max(0.0, old - 0.05)
            hi = min(1.0, old + 0.05)

            jv = old

                h[j] = jv
                _cd_update(c, h, j, old, jv, n, 1.0)
                imp = True
        if not imp:
            break
    h = _proj(h, n)
    return h

    return x.sum() - n * 0.5

def _eq_j(x, n):
    return np.ones(n)
