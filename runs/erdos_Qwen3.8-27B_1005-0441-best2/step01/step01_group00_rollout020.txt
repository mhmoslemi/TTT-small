    g = g + 2.0 * mu * (h.sum() - n * 0.5) * np.ones(n)
    return g

def _subgrad_step(h, lr, thresh):
    """One subgradient step focusing on active lags."""
    n = len(h)
    dx = 2.0 / n
    c = _corr(h) * dx
    cmax = c.max()
    active = c >= cmax - thresh
    g = np.zeros(n)
    count = 0
    for k in range(2 * n - 1):
        if not active[k]:
            continue
        count += 1
        m = k - (n - 1)
        if m >= 0:
            g[m:] += (1.0 - h[:n - m])
            g[:n - m] -= h[m:]
        else:
            p = -m
            g[:n - p] += (1.0 - h[p:])
            g[p:] -= h[:n - p]
    if count > 0:
        g /= count
    return _proj(h - lr * g * dx, n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _proj(h, n):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        d = n * 0.5 - h.sum()
        if abs(d) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            h = np.clip(h + d / n, 0, 1)
            break
        h[free] += d / free.sum()
        h = np.clip(h, 0, 1)
    return h

def _smax_and_exp(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h) * dx
    cmax = c.max()
    shifted = (c - cmax) / eps
    s = np.exp(shifted)
    return cmax + eps * np.log(s.sum()), s

    val, _ = _smax_and_exp(h, eps)
    n = len(h)
    s = h.sum() - n * 0.5
