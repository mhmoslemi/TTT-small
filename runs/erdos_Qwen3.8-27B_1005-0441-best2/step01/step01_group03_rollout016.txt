import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_binary(h, n):
    idx = np.argsort(h)[::-1][:n // 2]
    result = np.zeros(n)
    result[idx] = 1.0
    return result

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-14:
        mask = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            hp[mask] = np.clip(hp[mask] + diff / cnt, 0.0, 1.0)
    return hp

def iterated_sa(h, n, deadline, rng):
    current = h.copy().astype(float)
    best = current.copy()
    best_c5 = compute_c5(current, n)
    n_ones = n // 2
    round_num = 0
    t_base = 0.06
    while time.time() < deadline:
        current = best.copy()
        cur_c5 = best_c5
        ones_list = list(np.where(current == 1)[0])
        zeros_list = list(np.where(current == 0)[0])
        temp = t_base / (1.0 + 0.4 * round_num)
        t_min = 0.0008
        cool = 0.9994
        stall = 0
        while time.time() < deadline:
            if stall > 5000:
                break
            oi = rng.randint(n_ones)
            zi = rng.randint(n_ones)
            i1 = ones_list[oi]
            i0 = zeros_list[zi]
            current[i1] = 0.0
            current[i0] = 1.0
            new_c5 = compute_c5(current, n)
            delta = new_c5 - cur_c5
            if delta <= 0 or (temp > t_min and rng.random() < np.exp(-delta / temp)):
                cur_c5 = new_c5
                ones_list[oi] = i0

                    stall = 0

                    stall += 1

                current[i1] = 1.0
                current[i0] = 0.0
                stall += 1
            temp *= cool
        round_num += 1
    return best, best_c5

def continuous_refine(h_bin, n, deadline, rng):

    best = h.copy()

    lr = 0.1
    stalled = 0
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = set(np.argsort(corr)[-6:].tolist())
        lag_center = n - 1
        g = np.zeros(n)
        for kk in top_k:
            d = kk - lag_center
            for i in range(n):
                il = i - d
                if 0 <= il < n:
                    g[i] += (1.0 - h[il])
                ir = i + d
                if 0 <= ir < n:
                    g[i] -= h[ir]
        dx = 2.0 / n
        h_new = h - lr * g * dx
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c5 - 1e-12:

            stalled = 0

        h[::2] = 1.0
        if n % 2 != 0:
            idx1 = np.where(h == 1)[0]
            if len(idx1) > n // 2:
                h[idx1[-1]] = 0
    elif mode == 2:
        idx = rng.permutation(n)
        h[idx[:n // 2]] = 1.0
    elif mode == 3:
        h[:n // 4] = 1.0
        h[3 * n // 4:] = 1.0
        ones = int(np.sum(h))
        idx1 = np.where(h == 1)[0]
        idx0 = np.where(h == 0)[0]
        if ones > n // 2:
            h[idx1[:ones - n // 2]] = 0
        else:
            h[idx0[:n // 2 - ones]] = 1
    return h
