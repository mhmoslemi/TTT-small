So both A(d) and B(d) are related to the difference set S - S (or S and S shifted).

Specifically:
A(d) = |S ∩ {0,...,n-1-d}| - |S ∩ (S-d) ∩ {0,...,n-1-d}|
     = |{i ∈ S : i < n-d}| - |{i ∈ S : i < n-d and i+d ∈ S}|

B(d) = |{i ∈ S : i ≥ d}| - |{i ∈ S : i ≥ d and i-d ∈ S}|
     = |{i ∈ S : i ≥ d}| - |{i-d ∈ S : i ∈ S, i ≥ d}|
     = |S ∩ {d,...,n-1}| - |{(i-d, i) ∈ S×S : i ≥ d}|

The second term in both is |S ∩ (S+d)| ∩ appropriate range, which is the same.

So A(d) + B(d) involves the "boundary" terms, and A(d) - B(d) relates to the boundary structure.

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr_max(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))

def make_binary(h, n):
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n_ones]] = 1.0
    return h_b

def local_search(h, t_end, rng):
    n = len(h)
    dx = 2.0 / n
    best = compute_corr_max(h)
    h_cur = h.copy()
    stalled = 0

    while time.time() < t_end:
        improved = False
        ones = np.where(h_cur == 1.0)[0]
        zeros = np.where(h_cur == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_c = best + 1e-12
        best_i, best_j = -1, -1
        for i in ones:
            h_cur[i] = 0.0
            for j in zeros:
                h_cur[j] = 1.0
                c = compute_corr_max(h_cur)
                if c < best_c:
                    best_c = c
                    best_i, best_j = int(i), int(j)
                h_cur[j] = 0.0
            h_cur[i] = 1.0

        if best_i >= 0:
            h_cur[best_i] = 0.0
            h_cur[best_j] = 1.0
            best = best_c
            improved = True
            stalled = 0
        else:
            stalled += 1
            if stalled >= 1:
                break

    return h_cur, best

def sa_local_search(h, t_end, rng, n_iter=5000):
    n = len(h)
    dx = 2.0 / n
    cur = h.copy()
    cur_c = compute_corr_max(cur)
    best = cur_c
    best_h = cur.copy()
    T = 2.0 / n
    for it in range(n_iter):
        if time.time() > t_end:
            break
        T *= 0.998
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        cur[i] = 0.0
        cur[j] = 1.0
        new_c = compute_corr_max(cur)
        delta = new_c - cur_c
        if delta < 0 or rng.randn() < -delta * T * 500:
            cur_c = new_c
            if new_c < best:
                best = new_c
                best_h = cur.copy()
        else:
            cur[i] = 1.0
            cur[j] = 0.0
    return best_h, best

def focused_perturb(h, rng, n_flips=4):
    n = len(h)
    h_p = h.copy()
    corr = np.correlate(h_p, 1.0 - h_p, mode="full")
    center = n - 1
    max_idx = np.argmax(corr)
    for _ in range(n_flips):
        ones = np.where(h_p == 1.0)[0]
        zeros = np.where(h_p == 0.0)[0]
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h_p[i] = 0.0
        h_p[j] = 1.0
    return h_p

def try_nonbinary(h, t_end):
    n = len(h)
    best = compute_c5(h)
    best_h = h.copy()
    fracs = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9]
    for i in range(n):
        if time.time() > t_end:
            break
        for j in range(n):
            if i == j:
                continue
            for f in fracs:
                h_try = best_h.copy()
                if best_h[i] >= f:
                    h_try[i] = best_h[i] - f
                    h_try[j] = best_h[j] + f
                elif best_h[j] >= f:
                    h_try[i] = best_h[i] + f
                    h_try[j] = best_h[j] - f
                if h_try[i] < -1e-9 or h_try[i] > 1+1e-9 or h_try[j] < -1e-9 or h_try[j] > 1+1e-9:
                    continue
                h_try = np.clip(h_try, 0, 1)
                if abs(h_try.sum() - n/2) > 1e-9:
                    continue
                c = compute_c5(h_try)
                if c < best - 1e-12:
                    best = c
                    best_h = h_try.copy()
    return best_h, best

    rng = np.random.RandomState(seed)
    t_start = time.time()
    t_hard = t_start + budget_s - 3

    if cs * (2.0/n) < best_c5:

    # Local search from SA result
    h2, c2 = local_search(hs, t_phase2, rng)

    for r in range(30):
        if time.time() > t_phase3:
            break
        h_p = focused_perturb(best_h, rng, rng.randint(2, 7))
        h_r, c_r = local_search(h_p, min(time.time() + 12, t_phase3), rng)
        if c_r * (2.0/n) < best_c5 - 1e-10:

            best_c5 = c_s2 * (2.0/n)

        h_rand[rng.choice(n, n_ones, replace=False)] = 1.0
        h_r, c_r = local_search(h_rand, min(time.time() + 15, t_phase4), rng)
        if c_r * (2.0/n) < best_c5 - 1e-10:
            best_c5 = c_r * (2.0/n)
            best_h = h_r.copy()

    nonb_end = min(time.time() + 20, t_hard - 5)
    h_nb, c_nb = try_nonbinary(best_h, nonb_end)
    if c_nb < best_c5 - 1e-12:
        best_c5 = c_nb
        best_h = h_nb

    # Final local polish on binary version of best
    h_fb = make_binary(best_h, n)
    t_final = min(time.time() + 15, t_hard)
    h_f, c_f = local_search(h_fb, t_final, rng)
    if c_f * (2.0/n) < best_c5 - 1e-12:
        best_c5 = c_f * (2.0/n)
        best_h = h_f.copy()

    # One more non-binary try
    if time.time() < t_hard - 3:
        h_nb2, c_nb2 = try_nonbinary(best_h, t_hard)
        if c_nb2 < best_c5 - 1e-12:
            best_c5 = c_nb2
            best_h = h_nb2

    # Ensure feasibility
    best_h = np.clip(best_h, 0, 1)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, n)
