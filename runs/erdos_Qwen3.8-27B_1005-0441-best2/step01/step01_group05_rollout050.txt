    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = float(n) / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def softmax_corr_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    w = np.exp((corr - m) / temp)
    w /= np.sum(w)
    grad = np.zeros(n)
    for j in range(len(corr)):
        if w[j] < 1e-10:
            continue
        s = j - (n - 1)
        if s == 0:
            grad += w[j] * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += w[j] * (1.0 - h[:n-s])
            grad[:n-s] -= w[j] * h[s:]
        else:
            s2 = -s
            grad[:n-s2] += w[j] * (1.0 - h[s2:])
            grad[s2:] -= w[j] * h[:n-s2]
    return dx * grad

    best_val = compute_c5(best_h)
    h = h.copy()
