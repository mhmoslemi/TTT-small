    tau = 30.0
    t0 = time.time()
    dl = t0 + t_budget
    ni = 0
    while time.time() < dl - 0.3:
        t += 1
        ni += 1
        # Alternate gradient styles
        if ni % 200 < 100:
            w = softmax_weights(compute_corr(h), tau)
        else:
            w = topm_weights(compute_corr(h), 3)
        g = corr_grad(h, dx, w)
        h = adam_update(h, g, m, v, t, lr, ts)
        c5 = compute_c5(h)
        if c5 < bc:
            bh, bc = h.copy(), c5
            ni = 0
        else:
            ni += 1  # double count
        tau = min(tau * 1.006, 400.0)
        if ni > 300:
            lr *= 0.5
            ni = 0
        if ni > 800:
            h = project_feasible(bh + rng.randn(n) * 0.06, ts)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            lr = lr0
            tau = 30.0
            ni = 0
    return bh, bc

def binary_sa(h_frac, t_budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h_frac)
    ts = int(n // 2)
    h = (h_frac >= 0.5).astype(float)
    d = ts - int(h.sum())
    if d > 0:
        h[np.argsort(h_frac)[::-1][:d]] = 1.0
    elif d < 0:
        h[np.argsort(h_frac)[:(-d)]] = 0.0
    bh = h.copy()
    bc = compute_c5(h)
    t0 = time.time()
    dl = t0 + t_budget
    T = 0.8
    it = total = 0
    while time.time() < dl - 0.5:
        total += 1
        if total % 400 == 0:
            T = max(T * 0.93, 0.005)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if rng.rand() < 0.7:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            c5 = compute_c5(h)
            if c5 < bc:
                bh, bc = h.copy(), c5
            elif T > 0.01 and rng.rand() < math.exp((bc - c5) / T):
                pass
            else:
                h[i], h[j] = 1.0, 0.0
        else:
            # 2-swap: move two 1s to two 0 positions
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = sorted(rng.choice(ones, 2, replace=False))
                j1, j2 = sorted(rng.choice(zeros, 2, replace=False))
                h[i1] = h[i2] = 0.0
                h[j1] = h[j2] = 1.0
                c5 = compute_c5(h)
                if c5 < bc:
                    bh, bc = h.copy(), c5
                elif T > 0.02 and rng.rand() < math.exp((bc - c5) / (2 * T)):
                    pass
                else:
                    h[i1] = h[i2] = 1.0
                    h[j1] = h[j2] = 0.0
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    n = 70
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.asarray(initial_h_values, dtype=float)
        n = len(h_init)
    else:
        h_init = np.full(n, 0.5)
    ts = n / 2.0
    h_init = project_feasible(h_init, ts)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: Main Adam fractional optimization
    t_rem = deadline - time.time() - 250
    if t_rem > 30:
        h1, c1 = optimize_frac(h_init, min(t_rem, 400), seed, lr0=0.06)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    # Phase 2: Second Adam pass from best with different params
    t_rem = deadline - time.time() - 180
    if t_rem > 20:
        h2, c2 = optimize_frac(best_h, min(t_rem, 200), seed + 11, lr0=0.03)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: Binary SA from best fractional
    t_rem = deadline - time.time() - 120
    if t_rem > 15:
        hb, cb = binary_sa(best_h, min(t_rem, 250), seed + 100)
        if cb < best_c5:
            best_h, best_c5 = hb, cb

    # Phase 4: Structured restarts with fractional optimization
    structured_starts = [
        0.5 + 0.25 * np.sin(2 * np.pi * 3.0 * np.arange(n) / n),
        0.5 + 0.25 * np.cos(2 * np.pi * 5.0 * np.arange(n) / n),
        np.where(np.arange(n) % 5 < 2, 0.8, 0.2),
        best_h + np.random.RandomState(seed + 200).randn(n) * 0.07,
    ]
    for si, hs in enumerate(structured_starts):
        t_rem = deadline - time.time() - 60
        if t_rem < 25:
            break
        h_try = project_feasible(hs.copy(), ts)
        t_alloc = min(t_rem / (len(structured_starts) - si), 80)
        ho, co = optimize_frac(h_try, t_alloc, seed + 300 + si * 10, lr0=0.04)
        if co < best_c5:
            best_h, best_c5 = ho, co
            t_rem2 = deadline - time.time() - 40
            if t_rem2 > 10:
                hb2, cb2 = binary_sa(ho, min(t_rem2, 30), seed + 500 + si)
                if cb2 < best_c5:
                    best_h, best_c5 = hb2, cb2

    # Phase 5: Try different n values
    for n2 in [100, 140]:
        t_rem = deadline - time.time() - 30
        if t_rem < 25:
            break
        # Interpolate best_h to n2
        h_interp = np.interp(
            np.linspace(0, 1, n2, endpoint=False),
            np.linspace(0, 1, n, endpoint=False),
            np.linspace(0, 1, n, endpoint=False) * 0 + best_h,
        )
        # Actually just resample
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n2 / 2.0)
        t_alloc = min(t_rem, 60)
        ho, co = optimize_frac(h_interp, t_alloc, seed + 900 + n2, lr0=0.04)
        c5_new = compute_c5(ho)
        if c5_new < best_c5:
            best_h = ho
            best_c5 = c5_new
            n = n2
            ts = n2 / 2.0

    # Phase 6: Final polish - short Adam + binary
    t_rem = deadline - time.time() - 2
    if t_rem > 8:
        hp, cp = optimize_frac(best_h, min(t_rem - 5, 25), seed + 7777, lr0=0.015)
        if cp < best_c5:
            best_h, best_c5 = hp, cp
    t_rem = deadline - time.time() - 1
    if t_rem > 8:
        hb, cb = binary_sa(best_h, min(t_rem, 20), seed + 8888)
        if cb < best_c5:
            best_h, best_c5 = hb, cb

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
