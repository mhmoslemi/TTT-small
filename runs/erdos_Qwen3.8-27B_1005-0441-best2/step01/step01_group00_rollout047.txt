    n = len(h)

    s = np.exp((c - cmax) / eps)

    n = len(h)
    dx = 2.0 / n
    c = _corr(h)

                        lr *= 0.5
                        stuck = 0
                    else:
                        lr *= 0.99
                    if lr < 1e-9:
                        h = _proj(h + 0.002 * rng.randn(n), n)
                        lr = 0.003

    # Phase 4: Fine SLSQP polish
    if time.time() - t0 < budget_s * 0.85:
        for eps, mt in [(0.0004, 3000), (0.0002, 4000)]:
            if time.time() - t0 > budget_s * 0.88:
                break
            h_new = _slsqp(best_h, eps, mt)
            v = _c5(h_new)
            if v < best_val:
                best_val = v
                best_h = h_new.copy()

    # Phase 5: Perturbation restarts with re-optimization
    if time.time() - t0 < budget_s * 0.94:
        for pert in [0.004, 0.002, 0.001, 0.0005, 0.003]:
            if time.time() - t0 > budget_s * 0.96:
                break
            hp = _proj(best_h + pert * rng.randn(n), n)
            h_new = _slsqp(hp, 0.001, 1500)
            v = _c5(h_new)
            if v < best_val:
                best_val = v
                best_h = h_new.copy()
                h_new2 = _slsqp(h_new, 0.0003, 2500)
                v2 = _c5(h_new2)
                if v2 < best_val:
                    best_val = v2
                    best_h = h_new2.copy()
            else:
                h_new2 = _slsqp(hp, 0.0005, 2000)
                v2 = _c5(h_new2)
                if v2 < best_val:
                    best_val = v2
                    best_h = h_new2.copy()

    # Phase 6: Final subgradient cleanup
    if time.time() - t0 < budget_s * 0.98:
        h = best_h.copy()
        lr = 0.001
        for it in range(3000):
            if time.time() - t0 > budget_s * 0.99:
                break
            g = _topk_grad(h, n, 1)
            h_new = _proj(h - lr * g, n)
            v = _c5(h_new)
            if v < best_val:
                best_val = v
                best_h = h_new.copy()
                h = h_new
            else:
                lr *= 0.98
                if lr < 1e-10:
                    break

    best_h = _proj(best_h, n)
    best_val = _c5(best_h)
    return (best_h, best_val, n)
