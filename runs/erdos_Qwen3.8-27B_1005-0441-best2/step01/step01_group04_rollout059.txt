import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def _all_corr(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _fix_sum(h):
    n = len(h)
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = np.clip(h + diff * room / s, 0, 1)
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = np.clip(h + diff * h / s, 0, 1)
    return h

def _project(h):
    n = len(h)
    target = n / 2.0
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _grad_h(h, w):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    nm1 = n - 1
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-12:
            continue
        d = m - nm1
        if d >= 0:
            n1 = n - d
            grad[:n1] += wm * (1.0 - h[d:n])
            grad[d:] += wm * (-h[:n1])
        else:
            dp = -d
            n2 = n - dp
            grad[dp:] += wm * (1.0 - h[:n2])
            grad[:n2] += wm * (-h[dp:])
    return grad * dx

def _thue_morse(n):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def _thue_h(n, amp=0.3):
    t = _thue_morse(n)
    h = 0.5 + amp * (1 - 2 * t)
    return _fix_sum(h)

def _sin_h(n, freq=1.0):
    x = np.arange(n)
    h = 0.5 + 0.3 * np.sin(2 * np.pi * freq * x / n)
    return _fix_sum(h)

def _rand_h(n, rng):
    h = rng.random(n)
    h = 0.5 + 0.3 * (2 * h - 1)  # map to [0.2, 0.8]
    return _fix_sum(h)

def _pgd(h0, alpha, n_iter, lg, t0, budget_s):
    n = len(h0)
    h = h0.copy()
    best_c5 = _eval_c5(h)
    best_h = h.copy()
    lr = 1.0
    stall = 0
    
    for it in range(n_iter):
        if time.time() - t0 > budget_s - 25:
            break
        c = _all_corr(h)
        cmax = c.max()
        w = np.exp(alpha * (c - cmax))
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= wsum
        
        grad = _grad_h(h, w)
        gnorm = np.linalg.norm(grad, np.inf) + 1e-12
        
        # Step
        h_new = _project(h - lr * grad / gnorm)
        c5 = _eval_c5(h_new)
        
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            h = h_new  # still move, but track stall
            stall += 1
            if stall > 100:
                lr *= 0.7
                stall = 0
            if lr < 0.01:
                # Perturb and restart
                h = _project(h + 0.05 * lg.normal(0, 1, n))
                lr = 1.0
                stall = 0
    
    return best_h, best_c5

def _refine_step(h, lg, n_trials=20):
    """Perturbation-based refinement"""
    n = len(h)
    best = _eval_c5(h)
    best_h = h.copy()

    for _ in range(n_trials):
        h_p = _project(h + scale * lg.normal(0, 1, n))
        c5 = _eval_c5(h_p)
        if c5 < best:
            best = c5
            best_h = h_p
    return best_h, best
