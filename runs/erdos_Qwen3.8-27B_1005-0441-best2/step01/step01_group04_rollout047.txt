import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(h, target):
    """Project h to {0<=h<=1, sum(h)=target} via bisection."""
    h = np.clip(h.astype(float), 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo = -1.0
    hi = 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _lse_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    corr = raw * dx
    cmax = np.max(corr)
    shifted = corr - cmax
    p = np.exp(alpha * shifted)
    p_sum = np.sum(p)
    p = p / p_sum
    obj = cmax + np.log(p_sum) / alpha
    
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        s = j - (n - 1)
        pj = p[j] * dx
        # (1-h[i-s]) for 0<=i-s<n
        i_lo, i_hi = max(0, s), min(n, s + n)
        if i_lo < i_hi:
            idx = np.arange(i_lo, i_hi)
            grad[idx] += pj * (1.0 - h[idx - s])
        # -h[i+s] for 0<=i+s<n
        i_lo2, i_hi2 = max(0, -s), min(n, n - s)
        if i_lo2 < i_hi2:
            idx2 = np.arange(i_lo2, i_hi2)
            grad[idx2] -= pj * h[idx2 + s]
    return obj, grad

def _adam_opt(h0, alphas, steps_per_alpha, lr0=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(h0)
    target = n / 2.0
    h = _project(h0, target).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    for alpha in alphas:
        for step in range(steps_per_alpha):
            t += 1
            obj, grad = _lse_obj_grad(h, alpha)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / np.sqrt(1 + 0.001 * t)
            h = _project(h - lr * m_hat / (np.sqrt(v_hat) + eps), target)
    return h

def _slsqp_minimax(h0, max_iter=500):
    n = len(h0)
    target = n / 2.0
    def obj(x):
        return x[-1]
    def ineq(x):
        h = x[:-1]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return t - corr
    def eq(x):
        return np.sum(x[:-1]) - target
    
    x0 = np.concatenate([h0, [_eval_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    try:
        res = minimize(obj, x0, method='SLSQP',
                       constraints=[{'type': 'ineq', 'fun': ineq},
                                    {'type': 'eq', 'fun': eq}],
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        return _project(res.x[:-1], target)
    except Exception:
        return h0.copy()

def run(seed=42, budget_s=1100, **kwargs):

        if time.time() - t0 > budget_s - 120:

        steps = max(200, 500 - i * 50)

def _subgradient_opt(h0, n_iters=3000, lr0=0.01, frac=0.1):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    h = _project(h0, target).copy()
    for t in range(n_iters):
        raw = np.correlate(h, 1.0 - h, mode='full')
        corr = raw * dx
        cmax = np.max(corr)
        # Select shifts within 1% of max
        active = np.where(corr >= cmax * (1 - frac))[0]
        grad = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            i_lo, i_hi = max(0, s), min(n, s + n)
            if i_lo < i_hi:

                grad[idx] += (1.0 - h[idx - s]) / len(active)

            if i_lo2 < i_hi2:

                grad[idx2] -= h[idx2 + s] / len(active)
        grad *= dx
        lr = lr0 / (1 + 0.005 * t)
        h = _project(h - lr * grad, target)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _lse_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    corr = raw * dx
    cmax = np.max(corr)
    shifted = corr - cmax
    p = np.exp(alpha * shifted)
    p_sum = p.sum()
    p = p / p_sum
    obj = cmax + np.log(p_sum) / alpha
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        s = j - (n - 1)
        pj = p[j] * dx
        i1, i2 = max(0, s), min(n, s + n)
        if i1 < i2:
            idx = np.arange(i1, i2)
            grad[idx] += pj * (1.0 - h[idx - s])
        i3, i4 = max(0, -s), min(n, n - s)
        if i3 < i4:
            idx2 = np.arange(i3, i4)
            grad[idx2] -= pj * h[idx2 + s]
    return obj, grad

def _adam(h0, alpha, steps, lr0=0.03, b1=0.9, b2=0.999, eps=1e-8):
    n = len(h0)
    tgt = n / 2.0
    h = _project(h0, tgt)
    m = np.zeros(n)
    v = np.zeros(n)
    for t in range(1, steps + 1):
        _, g = _lse_obj_grad(h, alpha)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        lr = lr0 / (1.0 + 0.005 * t)
        h = _project(h - lr * mh / (np.sqrt(vh) + eps), tgt)
    return h

def _subgrad(h0, steps=3000, lr0=0.015, frac=0.05):
    n = len(h0)
    tgt = n / 2.0
    dx = 2.0 / n
    h = _project(h0, tgt)
    for t in range(steps):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        cmax = corr.max()
        active = np.where(corr >= cmax * (1.0 - frac))[0]
        g = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            i1, i2 = max(0, s), min(n, s + n)
            if i1 < i2:
                g[i1:i2] += (1.0 - h[i1 + s:i2 + s]) / len(active)
            i3, i4 = max(0, -s), min(n, n - s)
            if i3 < i4:
                g[i3:i4] -= h[i3 + s:i4 + s] / len(active)
        g *= dx
        lr = lr0 / (1.0 + 0.003 * t)
        h = _project(h - lr * g, tgt)
    return h

def _slsqp_mm(h0, max_iter=400):
    n = len(h0)
    tgt = n / 2.0
    dx = 2.0 / n
    def obj(x):
        return x[-1]
    def ineq(x):
        a = x[:-1]
        return x[-1] - np.correlate(a, 1.0 - a, mode='full') * dx
    def eq(x):
        return float(np.sum(x[:-1]) - tgt)
    x0 = np.concatenate([h0, [_eval_c5(h0)]]).copy()
    bnd = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    try:
        res = minimize(obj, x0, method='SLSQP',
                       constraints=[{'type': 'ineq', 'fun': ineq},
                                    {'type': 'eq', 'fun': eq}],
                       bounds=bnd,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        return _project(res.x[:-1], tgt)
    except Exception:
        return h0.copy()

def _try_update(h_cand, best_h, best_c5):
    h_cand = _project(h_cand, len(h_cand) / 2.0)
    c5 = _eval_c5(h_cand)
    if c5 < best_c5 - 1e-10:
        return h_cand.copy(), c5
    return best_h, best_c5
