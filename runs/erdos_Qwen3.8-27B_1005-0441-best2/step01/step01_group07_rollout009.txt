    lo, hi = -2.0, 2.0

        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

import numpy as np
import time, math

def project_feasible(h, target_sum):
    # Projection onto {0<=h<=1, sum(h)=target_sum} via bisection
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def compute_grad_smooth(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    shifted = np.clip(tau * (corr - mc), -500, 500)
    exps = np.exp(shifted)
    weights = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            L = -lag
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
    return grad * dx

def smooth_optimize(h_init, time_budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.8, 0.999
    eps = 1e-8
    t = 0
    tau = 40.0
    lr = 0.06
    t0 = time.time()

        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = project_feasible(h - lr * m_hat / (np.sqrt(v_hat) + eps), target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1

                tau = 40.0; lr = 0.06
                no_improve = 0
            else:
                tau = max(tau * 0.9, 20.0)
                lr = max(lr * 0.9, 0.005)
    
    return best_h, best_c5

def to_binary(h_frac, n_target_ones):
    """Convert fractional h to binary with exact sum."""
    n = len(h_frac)
    h = (h_frac >= 0.5).astype(float)
    n_ones = int(h.sum())
    if n_ones > n_target_ones:
        ones_idx = np.where(h == 1.0)[0]
        # Remove (n_ones - target) ones with lowest h_frac values
        order = np.argsort(h_frac[ones_idx])[::-1]

        order = np.argsort(h_frac[zeros_idx])[::-1]
        for i in range(n_target_ones - n_ones):
            h[zeros_idx[order[i]]] = 1.0
    return h

def binary_sa(h_bin, time_budget, seed, T_init=0.12, T_min=0.005):

        # Periodic reheat
        if T <= T_min * 1.1 and rng.random() < 0.01:
            T = T_init * 0.3
    
    return best_h, best_c5

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.asarray(initial_h_values, dtype=float)
    else:
        h_init = np.full(70, 0.5)
    n = len(h_init)
    target_sum = n / 2.0
    h_init = project_feasible(h_init, target_sum)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: Smooth Adam optimization
    t_rem = deadline - time.time() - 280
    if t_rem > 30:
        h1, c5_1 = smooth_optimize(h_init, min(t_rem, 300), seed)
        if c5_1 < best_c5:
            best_h, best_c5 = h1, c5_1

    # Phase 2: Binary SA from smooth solution
    h_bin = to_binary(best_h, n // 2)
    t_rem = deadline - time.time() - 220
    if t_rem > 15:
        hb, c5b = binary_sa(h_bin, min(t_rem, 140), seed + 100)
        if c5b < best_c5:
            best_h, best_c5 = hb, c5b

    # Phase 3: More smooth opt + SA with different inits
    for trial in range(3):
        t_rem = deadline - time.time() - 55
        if t_rem < 20:
            break
        rng_t = np.random.RandomState(seed + 500 + trial * 47)
        h_try = best_h + rng_t.randn(n) * 0.07
        h_try = project_feasible(h_try, target_sum)
        t_alloc = min(t_rem * 0.45, 90)
        h_opt, c5_opt = smooth_optimize(h_try, t_alloc, seed + 2000 + trial * 7)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        t_rem2 = deadline - time.time() - 15
        if t_rem2 > 10:
            hb2 = to_binary(h_opt, n // 2)
            hb2, c5_2 = binary_sa(hb2, min(t_rem2, 40), seed + 3000 + trial * 13)
            if c5_2 < best_c5:
                best_h, best_c5 = hb2, c5_2

    # Phase 4: Try larger n values with structured init
    for n_try in [100, 140]:
        t_rem = deadline - time.time() - 15
        if t_rem < 35:
            break
        rng_n = np.random.RandomState(seed + n_try * 7)
        h_n = np.zeros(n_try)
        h_n[::2] = 1.0
        h_n += rng_n.randn(n_try) * 0.15
        h_n = project_feasible(h_n, float(n_try // 2))
        t1 = min(t_rem * 0.55, 70)
        t2 = min(t_rem * 0.35, 40)
        h_n_opt, c5_n = smooth_optimize(h_n, t1, seed + n_try)
        t_rem3 = deadline - time.time() - 5
        if t_rem3 > 10:
            hn_bin = to_binary(h_n_opt, n_try // 2)
            hnb, c5_nb = binary_sa(hnb_bin if False else hn_bin, min(t_rem3, t2), seed + n_try + 5)
            if c5_nb < c5_n:
                h_n_opt, c5_n = hnb, c5_nb
        if c5_n < best_c5:
            best_h, best_c5 = h_n_opt, c5_n

    # Phase 5: Final random binary SA restart from best
    t_rem = deadline - time.time() - 2
    if t_rem > 15:
        rng_f = np.random.RandomState(seed + 9999)
        h_f = (best_h >= 0.5).astype(float)
        # Perturb a few positions
        ones_f = np.where(h_f == 1.0)[0]
        zeros_f = np.where(h_f == 0.0)[0]
        for _ in range(min(3, len(ones_f), len(zeros_f))):
            i = ones_f[rng_f.randint(len(ones_f))]
            j = zeros_f[rng_f.randint(len(zeros_f))]
            h_f[i], h_f[j] = 0.0, 1.0
        hf, c5f = binary_sa(h_f, min(t_rem, 30), seed + 7777, T_init=0.2)
        if c5f < best_c5:
            best_h, best_c5 = hf, c5f

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
