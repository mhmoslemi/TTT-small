        cnt = np.sum(mask)

                            options={'maxiter': 1500, 'ftol': 1e-16})
            hc2 = project_feasible(res2.x, n)
            c5c2 = compute_c5(hc2)
            if c5c2 < c5_best:
                c5_best = c5c2
                h_best = hc2.copy()

    # Phase 6: Upscale to finer grid
    if time.time() - t0 < budget_s - 45:
        n2 = min(n * 2, 320)
        if n2 > n:
            xo = np.arange(n, dtype=float)
            xn = np.linspace(0.0, n - 1, n2)
            hu = np.interp(xn, xo, h_best)
            hu = project_feasible(hu, n2)
            _TARGET = n2 * 0.5
            b2 = [(0.0, 1.0)] * n2
            cx2 = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}]
            for alpha in [500.0, 2000.0, 5000.0]:
                if time.time() - t0 > budget_s - 25:
                    break
                res = minimize(smax_obj, hu, args=(alpha,),
                               method='SLSQP', bounds=b2, constraints=cx2,
                               options={'maxiter': 2500, 'ftol': 1e-14})
                hc = project_feasible(res.x, n2)
                c5c = compute_c5(hc)
                if c5c < c5_best:
                    c5_best = c5c
                    h_best = hc.copy()
                    n = n2

    h_best = project_feasible(h_best, len(h_best))
    c5_best = compute_c5(h_best)
    return h_best, c5_best, len(h_best)
