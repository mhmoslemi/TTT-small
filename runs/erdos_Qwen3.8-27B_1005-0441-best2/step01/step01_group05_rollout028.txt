def _project(h, target_sum):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target_sum}"""
    h = np.clip(h, 0, 1)
    # Adjust free entries to meet sum constraint
    diff = target_sum - np.sum(h)
    free = (h > 0) & (h < 1)
    n_free = np.sum(free)
    if n_free == 0:
        return h
    h[free] += diff / n_free
    h = np.clip(h, 0, 1)
    # Second pass if needed
    diff = target_sum - np.sum(h)
    free = (h > 0) & (h < 1)
    n_free = np.sum(free)
    if n_free > 0:
        h[free] += diff / n_free
        h = np.clip(h, 0, 1)

Wait, I think there's an issue with my gradient. Let me be more careful.

The full correlation: np.correlate(a, v, mode='full') where a=h (length n), v=1-h (length n).

From numpy docs: c_k = sum_n a_{n+k} * v_n for the 'same' mode... actually let me look at the definition.

For 'full' mode, the output has length 2n-1. The k-th element (0-indexed) is:
c[k] = sum_j a[j] * v[j + k - (n-1)]

where the sum is over j such that both j and j+k-(n-1) are in [0, n-1].

So c[k] = sum_j h[j] * (1-h[j+k-(n-1)])

Let d = k - (n-1), ranging from -(n-1) to (n-1).

c[k] = sum_j h[j] * (1-h[j+d]) for valid j, j+d.

dc[k]/dh[i]:
- h appears in two places: as h[j] and as h[j+d]
- When j = i: contributes (1-h[i+d]) to dc/dh[i] (if i+d valid)
- When j+d = i, i.e., j = i-d: contributes -h[i-d] to dc/dh[i] (if i-d valid)

So dc[k]/dh[i] = (1-h[i+d]) * [0<=i+d<n] - h[i-d] * [0<=i-d<n]

And the gradient of the smooth objective:
grad[i] = sum_k p[k] * dx * dc[k]/dh[i]
= dx * sum_k p[k] * [(1-h[i+d_k]) * valid - h[i-d_k] * valid]

def _smooth_obj_grad(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    t = max(tau, 1e-10)
    exps = np.exp((corr - m) / t)
    wsum = np.sum(exps)
    if wsum < 1e-30:
        exps = np.zeros(2 * n - 1)
        exps[np.argmax(corr)] = 1.0
        wsum = 1.0
    p = exps / wsum
    obj = m + t * np.log(wsum)
    
    grad = np.zeros(n)
    ds = np.arange(2 * n - 1) - (n - 1)
    for kk in range(2 * n - 1):
        pk = p[kk]
        if pk < 1e-15:

def _project_box_sum(h, target):
    """Project onto {0<=x<=1, sum(x)=target} approximately."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)

        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = int(np.sum(free))
        if nf == 0:
            return h
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:

    best = h.copy()

    step = 0.3

    prev_h = None

        prev_h = h.copy()

        h_opt, c5_opt = _pg_optimize(h_pert, n, t_alloc, tau_init=0.02, tau_decay=0.98)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 4: SLSQP polish with decreasing temperature
    if _rem() > 40:
        for tau_polish in [1e-3, 5e-5, 1e-5]:
            if _rem() < 15:
                break
            h_opt, c5_opt = _slsqp_polish(best_h.copy(), n, min(40, _rem() - 10), tau=tau_polish)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # Phase 5: Final PG with very low temp to squeeze out last bits
    if _rem() > 20:
        h_opt, c5_opt = _pg_optimize(best_h.copy(), n, _rem() - 5, tau_init=0.005, tau_decay=0.995)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    best_h = _project_box_sum(best_h, target)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, n
