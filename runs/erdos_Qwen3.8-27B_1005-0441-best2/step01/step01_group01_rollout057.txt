```python
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15 * n:

        free = (h > 1e-14) & (h < 1.0 - 1e-14)
```

The key observation: for the correlation at lag k, only the pairs (i, i+k) matter. If I think of h as a "signal" and 1-h as its "complement", the correlation measures how well the signal aligns with the shifted complement.

But our problem is different: we're minimizing max_k sum h[i]*(1-h[i+k]), not the autocorrelation sum h[i]*h[i+k].

Let me relate them:
sum_i h[i]*(1-h[i+k]) = sum_i h[i] - sum_i h[i]*h[i+k] (for the valid range)

If h is binary (0 or 1), then h[i]*(1-h[i+k]) = 1 if h[i]=1 and h[i+k]=0. So it counts the number of "1→0" transitions at lag k.

For a binary sequence with exactly n/2 ones, the sum at lag 0 is 0 (since h[i]*(1-h[i]) = 0 for binary h). The sum at lag k is the number of positions where h has a 1 and h shifted by k has a 0.

To minimize the maximum over k, we want the "1→0" transitions to be evenly distributed across all lags.

This is essentially the problem of finding a binary sequence with minimal "maximum number of 1-to-0 transitions at any lag." This is related to the "cyclic autocorrelation" but for aperiodic (one-sided) correlation.

```python
def focused_gradient_step(h, n, max_iters=100):
    """Cyclical gradient steps on the worst lag."""
    dx = 2.0 / n
    h = h.copy()
    for _ in range(max_iters):
        corr = np.correlate(h, 1-h, mode='full')
        k_star = int(np.argmax(corr))
        shift = k_star - (n-1)
        # Compute gradient of f_{k_star}
        grad = np.zeros(n)
        if shift >= 0:
            grad[:n-shift] += (1 - h[shift:]) * dx
            grad[shift:] += (-h[:n-shift]) * dx
        else:
            s = -shift
            grad[s:] += (1 - h[:n-s]) * dx
            grad[:n-s] += (-h[s:]) * dx
        # Project gradient descent step with line search
        best_t = 0
        best_c = compute_c5(h)
        for t in [0.001, 0.005, 0.01, 0.02, 0.05]:
            h_trial = h - t * grad
            h_trial = project_feasible(h_trial, n)
            c = compute_c5(h_trial)
            if c < best_c:
                best_c = c
                best_t = t
        if best_t > 0:
            h = project_feasible(h - best_t * grad, n)
        else:
            break
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(500):
        d = target - np.sum(h)
        if abs(d) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(free)
        if cnt == 0:
            h += d / n
        else:
            h[free] += d / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def obj_topk(x, alpha, topk_n):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    k = min(topk_n, len(corr))
    if k < len(corr):
        part = np.argpartition(corr, -k)[-k:]
    else:
        part = np.arange(len(corr))
    topk = corr[part]
    shifted = alpha * (topk - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def obj_pen(x, alpha, topk_n, pen, n):
    base = obj_topk(x, alpha, topk_n)
    return base + pen * (np.sum(x) - n * 0.5) ** 2

def grad_f_k(h, k, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] += (1.0 - h[:n-k]) * dx
        grad[:n-k] += (-h[k:]) * dx
    else:
        s = -k
        grad[:n-s] += (1.0 - h[s:]) * dx
        grad[s:] += (-h[:n-s]) * dx
    return grad

def cyclical_grad_descent(h, n, max_rounds=200, time_limit_s=30.0):
    dx = 2.0 / n
    t_start = time.time()
    for _ in range(max_rounds):
        if time.time() - t_start > time_limit_s:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        m_idx = int(np.argmax(corr))
        k = m_idx - (n - 1)
        g = grad_f_k(h, k, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        best_c = compute_c5(h)
        best_h = h.copy()
        for scale in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
            h_try = project_feasible(h - scale * g / gn, n)
            c_try = compute_c5(h_try)
            if c_try < best_c - 1e-10:
                best_c = c_try
                best_h = h_try
        if np.allclose(best_h, h):
            break
        h = best_h
    return h

def directed_perturb(h, n, rng):
    corr_full = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    k_worst = int(np.argmax(corr_full))
    shift = k_worst - (n - 1)
    h_pert = h.copy()
    nperturb = 0.03
    if shift >= 0:
        both_hi = (h_pert[:n-shift] > 0.25) & (h_pert[shift:] > 0.25)
        idx = np.where(both_hi)[0]
        if len(idx) > 0:
            h_pert[idx] -= nperturb
            comp = (h_pert > 0.05) & (h_pert < 0.35)
            if np.sum(comp) > 0:
                h_pert[comp] += nperturb * len(idx) / max(1, np.sum(comp))
    else:
        s = -shift
        both_hi = (h_pert[s:] > 0.25) & (h_pert[:n-s] > 0.25)
        idx = np.where(both_hi)[0] + s
        if len(idx) > 0:
            h_pert[idx] -= nperturb
            comp = (h_pert > 0.05) & (h_pert < 0.35)
            if np.sum(comp) > 0:
                h_pert[comp] += nperturb * len(idx) / max(1, np.sum(comp))
    return project_feasible(h_pert, n)

                           options={'maxiter': 1000, 'ftol': 1e-14})
            hc = project_feasible(res.x, n)
            c = compute_c5(hc)
            if c < c5_best:
                c5_best = c
                h_best = hc.copy()

    # Phase 6: Random restarts with varying scale
    while time.time() - t0 < budget_s - 120:
        scale = 0.003 + 0.025 * rng.random() ** 1.5
        h_pert = h_best + scale * rng.standard_normal(n)
        h_pert = project_feasible(h_pert, n)
        topk_n = max(3, int(n_corr * 0.12))
        res = minimize(obj_topk, h_pert, args=(300.0, topk_n),
                       method='SLSQP', bounds=bounds,
                       options={'maxiter': 700, 'ftol': 1e-13})
        hc = project_feasible(res.x, n)
        c = compute_c5(hc)
        if c < c5_best:
            c5_best = c
            h_best = hc.copy()
        # Quick gradient polish after each restart
        if c < c5_best + 0.002:
            h_gd = cyclical_grad_descent(hc, n, max_rounds=50, time_limit_s=5.0)
            c_gd = compute_c5(h_gd)
            if c_gd < c5_best:
                c5_best = c_gd
                h_best = h_gd

    # Phase 7: Final ultra-tight SLSQP + gradient polish
    if time.time() - t0 < budget_s - 40:
        topk_n = max(2, int(n_corr * 0.04))
        res = minimize(obj_topk, h_best.copy(), args=(1500.0, topk_n),
                       method='SLSQP', bounds=bounds,
                       options={'maxiter': 5000, 'ftol': 1e-16})
        hc = project_feasible(res.x, n)
        c = compute_c5(hc)
        if c < c5_best:
            c5_best = c
            h_best = hc
        h_final = cyclical_grad_descent(h_best, n, max_rounds=200, time_limit_s=max(5, budget_s - 40 - (time.time()-t0)))
        c_final = compute_c5(h_final)
        if c_final < c5_best:
            c5_best = c_final
            h_best = h_final

    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)
    return h_best, c5_best, n
```
