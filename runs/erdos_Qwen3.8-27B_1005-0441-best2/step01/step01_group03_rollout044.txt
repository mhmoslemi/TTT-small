        m = (idx >= 0) & (idx < n)

    cur_c5 = float(np.max(cors)) * (2.0/n)

    zeros = np.where(h == 0.0)[0]

    T_min = 0.1
    step = 0
    while time.time() < deadline and T > T_min:

            break
        p = ones[rng.randint(len(ones))]
        q = zeros[rng.randint(len(zeros))]
        if p == q:
            continue
        d = swap_delta(h, p, q, n)
        new_cors = cors + d
        new_max = float(np.max(new_cors))
        cur_max = float(np.max(cors))

                zeros = np.where(h == 0.0)[0]

                best_c5 = new_max * (2.0/n)
                best_h = h.copy()
        T *= alpha
        step += 1
    return best_h, best_c5
