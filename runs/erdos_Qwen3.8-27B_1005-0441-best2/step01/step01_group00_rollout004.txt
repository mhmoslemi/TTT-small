    # Phase 5: Final coordinate micro-adjustments
    if best_h is not None and time.time() - t0 < budget_s * 0.99:
        n = len(best_h)
        ch = best_h.copy()
        cv = best_val
        for sweep in range(3):
            if time.time() - t0 > budget_s * 0.99:
                break
            for i in range(n):
                saved = ch[i]
                for delta in [0.005, -0.005, 0.002, -0.002, 0.01, -0.01]:
                    trial = saved + delta
                    if trial < 0 or trial > 1:
                        continue
                    ch[i] = trial
                    ch = _proj(ch, n)
                    v = _c5(ch)
                    if v < cv:
                        cv = v
                        break
                if ch[i] != saved:
                    pass  # kept improvement
                else:
                    ch[i] = saved
                    ch = _proj(ch, n)
            if cv < best_val:
                best_val = cv
                best_h = ch.copy()

    if best_h is None:
        n = 70
        best_h = np.full(n, 0.5)
        best_val = _c5(best_h)

    best_h = _proj(best_h, len(best_h))
    best_val = _c5(best_h)
    return (best_h, best_val, len(best_h))
