def project_feasible(h, target_sum):
    # Bisection to find the right shift for projection
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def compute_grads_smooth(h, dx, tau):
    """Compute gradient of smooth max using log-sum-exp."""
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    # Smooth max via log-sum-exp
    max_c = corr.max()
    shifted = tau * (corr - max_c)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exps = np.exp(shifted)
    weights = exps / exps.sum()
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            L = -lag
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
    return grad * dx

import numpy as np
import time, math
from scipy.optimize import minimize

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def smooth_max_obj_grad(h, tau):
    """Smooth max of correlations and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    shifted = np.clip(tau * (corr - mc), -60, 60)
    exps = np.exp(shifted)
    w = exps / exps.sum()
    
    # Smooth max value
    smax = mc + np.log(exps.sum()) / tau
    
    # Gradient
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-13:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:] += w[k] * (1.0 - h[:n - lag])
            grad[:n - lag] -= w[k] * h[lag:]
        else:
            L = -lag
            grad[:n - L] += w[k] * (1.0 - h[L:])
            grad[L:] -= w[k] * h[:n - L]
    
    return smax * dx, grad * dx

def gradient_descent_custom(h_init, time_budget, seed, lr=0.05, tau0=100.0, anneal=3.0):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    ts = n * 0.5
    h = project_feasible(h_init, ts)
    bh, bc = h.copy(), compute_c5(h)
    v = np.zeros(n)
    beta = 0.9
    t0 = time.time()
    dl = t0 + time_budget
    ni = 0
    while time.time() < dl - 0.2:
        tau = tau0 * (1.0 + anneal * (time.time() - t0) / max(time_budget, 1))
        _, g = smooth_max_obj_grad(h, tau)
        # Nesterov
        hl = project_feasible(h + beta * v, ts)
        _, gl = smooth_max_obj_grad(hl, tau)
        v = beta * v - lr * gl
        h = project_feasible(h + v, ts)
        c = compute_c5(h)
        if c < bc:
            bh, bc, ni = h.copy(), c, 0
            lr = min(lr * 1.005, 0.12)
        else:
            ni += 1
            if ni % 100 == 0:
                lr = max(lr * 0.85, 0.005)
        if ni > 800:
            h = project_feasible(bh + rng.randn(n) * 0.03, ts)
            v[:] = 0
            lr, ni = 0.05, 0
    return bh, bc

def binary_ls(h_frac, time_budget, seed, two_swap=False):
    rng = np.random.RandomState(seed)
    n = len(h_frac)
    ts = int(n // 2)
    h = (h_frac >= 0.5).astype(float)
    while h.sum() < ts:
        h[np.argmin(h)] = 1.0

        h[np.argmax(h)] = 0.0

            h[i], h[j] = h[j], h[i]

        else:

            if c < bc:
                bh, bc = h.copy(), c
            else:
                h[i1], h[i2], h[j1], h[j2] = 1, 1, 0, 0
    return bh, bc
