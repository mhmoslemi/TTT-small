def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def to_binary(h, n):
    """Convert to binary with exactly n/2 ones"""
    h = h.copy()
    # Sort by value, keep top n/2 as 1
    idx = np.argsort(h)[::-1][:n//2]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def binary_local_search(b, n, deadline, rng, max_stall=50):
    """Hill climbing on binary sequences via swaps"""
    best_b = b.copy()
    best_c5 = compute_c5(b)
    stall = 0
    
    while time.time() < deadline:
        improved = False
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        
        best_deltas = []
        for a in ones:
            for bpos in zeros:
                # Try swap
                b[a] = 0
                b[bpos] = 1
                c = compute_c5(b)
                b[a] = 1
                b[bpos] = 0
                best_deltas.append((c, a, bpos))
        
        best_deltas.sort()
        best_c, best_a, best_b = best_deltas[0]
        
        if best_c < best_c5 - 1e-12:
            b[best_a] = 0
            b[best_b] = 1
            best_c5 = best_c
            best_b = b.copy()
            stall = 0
            improved = True
        else:
            stall += 1
            if stall > max_stall:
                # Random restart or perturbation
                # Do a random swap
                a = rng.choice(ones)
                bpos = rng.choice(zeros)
                b[a] = 0
                b[bpos] = 1
                c = compute_c5(b)
                if c < best_c5 + 0.01:  # Allow slight worsening
                    best_b = b.copy()
                    best_c5 = min(best_c5, c)
                stall = 0
    
    return best_b, best_c5

def swap_delta(h, a, b, n):
    """Compute delta in correlation when h[a] 1->0 and h[b] 0->1"""
    delta = np.zeros(2*n - 1)
    # h[a-k] term: lag k, position a-k must be in [0,n-1]
    # a-k in [0,n-1] means k in [a-(n-1), a]

        if 0 <= a+k < n: d += h[a+k]
        if 0 <= b-k < n: d -= h[b-k]
        if 0 <= b+k < n: d -= h[b+k]
        delta[k + n - 1] = d
    return delta

def to_binary(h, n):
    # Keep exactly n//2 positions as 1 (the largest values)
    threshold = np.partition(h, n - n//2)[n - n//2]
    b = (h >= threshold).astype(float)
    # Handle ties
    count = int(b.sum())
    if count > n//2:

        tied = np.where((b == 1) & (h == threshold))[0]

- p+k = a, i.e., k = a-p. The column is (a-p)+n-1. Similarly always valid.

    deadline1 = t0 + budget_s * 0.45
    if h0 is not None and n_primary == len(h0):
        h_start = h0
    else:
        h_start = rng.rand(n_primary)
    bh, bc5 = binary_ls(h_start, n_primary, deadline1, rng)
    best_h, best_c5, best_n = bh, bc5, n_primary

    # Phase 2: Other n values (40% time)
    remaining_ns = ns_to_try[1:]
    if remaining_ns:
        per_try = (budget_s * 0.40) / len(remaining_ns)
        for n_try in remaining_ns:
            if time.time() > t0 + budget_s - 15:
                break
            dl = min(time.time() + per_try, t0 + budget_s - 15)
            h_rand = rng.rand(n_try)
            bh2, bc52 = binary_ls(h_rand, n_try, dl, rng)
            if bc52 < best_c5:
                best_h, best_c5, best_n = bh2, bc52, n_try

    # Phase 3: Refine best binary solution further (10% time)
    if best_h is not None and time.time() < t0 + budget_s - 10:
        dl3 = min(t0 + budget_s * 0.10, t0 + budget_s - 10)
        if dl3 > time.time():
            bh3, bc53 = binary_ls(best_h, best_n, dl3, rng)
            if bc53 < best_c5:
                best_h, best_c5 = bh3, bc53

    # Phase 4: Continuous refinement on best (last 5%)
    if best_h is not None and time.time() < t0 + budget_s - 3:
        dl4 = t0 + budget_s - 3
        bh4, bc54 = continuous_refine(best_h, best_n, dl4, rng)
        if bc54 < best_c5:
            best_h, best_c5 = bh4, bc54

    # Final validation
    c5_final = compute_c5(best_h)
    return best_h, float(c5_final), best_n
