import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _corr_grad_smoothed(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    val = (m + temp * np.log(np.sum(np.exp((corr - m) / temp)))) * dx
    w = np.exp((corr - m) / temp)
    w_sum = w.sum()
    w /= w_sum
    grad = np.zeros(n)
    for m_idx in range(len(corr)):
        wi = w[m_idx]
        if wi < 1e-13:
            continue
        lag = m_idx - (n - 1)
        if lag == 0:
            g = np.zeros(n)
            g[:] = 1.0 - h
            g[:] -= h
            g[:] = 1.0 - 2.0 * h
        elif lag > 0:
            g = np.zeros(n)
            g[0:n - lag] += (1.0 - h[lag:n])
            g[lag:n] -= h[0:n - lag]
        else:
            g = np.zeros(n)
            g[-lag:n] += (1.0 - h[0:n + lag])
            g[0:n + lag] -= h[-lag:n]
        grad += wi * g
    return val, grad * dx

def _slsqp(h_init, n, temp, max_iter, time_limit):
    t0 = time.time()
    bounds = [(0.0, 1.0)] * n

    def obj_and_jac(x):
        v, g = _corr_grad_smoothed(x, temp)
        return v, g

    def ceq(x):
        return np.sum(x) - n / 2.0

    def ceq_jac(x):
        return np.ones(n)

    cons = ({'type': 'eq', 'fun': ceq, 'jac': ceq_jac},)
    try:
        res = minimize(obj_and_jac, h_init, jac=True, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        h_new = np.clip(res.x, 0.0, 1.0)
        s = np.sum(h_new)
        tgt = float(n) / 2.0
        if s > 1e-12:
            h_new = h_new * tgt / s
            h_new = np.clip(h_new, 0.0, 1.0)
        return h_new
    except Exception:
        return h_init

def _adam_subgrad(h, n, lr=0.002, steps=2000, top_k=3, time_limit=40, t0=None):
    if t0 is None:
        t0 = time.time()
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    best = h.copy()
    best_v = compute_c5(best)
    for step in range(steps):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for mi in top_idx:
            lag = mi - (n - 1)
            if lag == 0:
                g = 1.0 - 2.0 * h
            elif lag > 0:
                g = np.zeros(n)
                g[0:n - lag] += (1.0 - h[lag:n])
                g[lag:n] -= h[0:n - lag]
            else:
                g = np.zeros(n)
                g[-lag:n] += (1.0 - h[0:n + lag])
                g[0:n + lag] -= h[-lag:n]
            grad += g
        grad /= top_k
        grad *= (2.0 / n)
        t = step + 1
        m_arr = beta1 * m_arr + (1 - beta1) * grad
        v_arr = beta2 * v_arr + (1 - beta2) * grad ** 2
        m_hat = m_arr / (1 - beta1 ** t)
        v_hat = v_arr / (1 - beta2 ** t)
        lr_t = lr * (1.0 - 0.5 * t / steps)
        h = h - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        tgt = float(n) / 2.0
        if s > 1e-12:
            h = h * tgt / s
            h = np.clip(h, 0.0, 1.0)
        v = compute_c5(h)
        if v < best_v - 1e-12:
            best = h.copy()
            best_v = v
    return best, best_v

def _perturb(h, scale, seed):
    rng = np.random.RandomState(seed)
    h_p = h + scale * rng.randn(len(h))
    h_p = np.clip(h_p, 0.0, 1.0)
    s = np.sum(h_p)
    tgt = float(len(h)) / 2.0
    if s > 1e-12:
        h_p = h_p * tgt / s
        h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def _sinusoidal_init(n, freq, phase, amp):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-12:
        h = h * (float(n) / 2.0) / s
        h = np.clip(h, 0.0, 1.0)
    return h

def _piecewise_linear_init(n, n_seg):
    per_seg = n // n_seg
    h = np.zeros(n)
    for i in range(n_seg):
        start = i * per_seg
        end = start + per_seg
        if i % 2 == 0:
            h[start:end] = 1.0
        else:
            h[start:end] = 0.0
    s = np.sum(h)
    tgt = float(n) / 2.0
    if s > 1e-12:
        h = h * tgt / s
    return np.clip(h, 0.0, 1.0)
