from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_fun(x, alpha):

    corr = np.correlate(x, 1.0 - x, mode='full')

def binary_local_search(h, n, rng, max_iters=5000):
    """Local search on binary subspace: flip pairs to maintain sum constraint."""

    h_bin = project_feasible(h_bin, n)
    best_c5 = compute_c5(h_bin)
    ones_idx = np.where(h_bin == 1.0)[0]
    zeros_idx = np.where(h_bin == 0.0)[0]
    for it in range(max_iters):
        improved = False
        i = ones_idx[rng.integers(len(ones_idx))]
        j = zeros_idx[rng.integers(len(zeros_idx))]
        h_bin[i], h_bin[j] = 0.0, 1.0
        c5_new = compute_c5(h_bin)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            ones_idx = np.where(h_bin == 1.0)[0]
            zeros_idx = np.where(h_bin == 0.0)[0]
            improved = True
        else:
            h_bin[i], h_bin[j] = 1.0, 0.0
        if not improved and it % 100 == 99:
            # Occasionally accept worse moves (simulated annealing)
            pass
    return h_bin, best_c5

def pair_swap_search(h, n, rng, max_rounds=200):
    """Continuous pair swap local search."""
    h = h.copy()
    best_c5 = compute_c5(h)
    deltas = [0.02, 0.05, 0.1, -0.02, -0.05, -0.1]
    for rnd in range(max_rounds):
        improved = False
        for _ in range(20):
            i, j = rng.choice(n, 2, replace=False)
            for d in deltas:
                new_h = h.copy()
                v1 = np.clip(h[i] + d, 0.0, 1.0)
                v2 = np.clip(h[j] - (v1 - h[i]), 0.0, 1.0)

                    best_c5 = c5_new
                    improved = True
