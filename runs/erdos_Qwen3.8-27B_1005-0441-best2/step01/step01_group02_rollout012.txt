def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max(corr, beta):
    mc = np.max(corr)
    return mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta

def smooth_grad_h(h, beta):
    """Gradient of smooth_max(corr(h), beta) * dx with respect to h."""
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    wsum = w.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    w = (w / wsum) * dx
    grad = np.zeros(n)
    # For each shift d, contribution to gradient
    for d in range(-(n-1), n):
        k = n - 1 + d
        wk = w[k]
        if wk < 1e-15:
            continue
        if d >= 0:
            # h[i+d]*(1-h[i]) for i in [0, n-d-1]
            # d/dh[j]: (1-h[j-d]) for j in [d, n-1]; -h[j+d] for j in [0, n-d-1]
            grad[d:] += wk * (1.0 - h[:n-d])
            grad[:n-d] -= wk * h[d:]
        else:

            # h[i-d2]*(1-h[i]) for i in [d2, n-1]

            grad[:n-d2] += wk * (1.0 - h[d2:])
            grad[d2:] -= wk * h[:n-d2]
    return grad

    n = len(h)
    h = np.array(h, dtype=float)

    hi = 2.0

        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
