import time
from scipy.optimize import minimize

def compute_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def get_c5(h, n):
    return float(np.max(compute_overlaps(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-14:
        mask = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        cnt = float(np.sum(mask))
        if cnt > 0:
            hp[mask] = np.clip(hp[mask] + diff / cnt, 0.0, 1.0)
    return hp

def sm_max(x, n, beta):
    h = x
    ov = compute_overlaps(h, n)
    m = np.max(ov)
    e = np.exp(beta * (ov - m))
    s = np.sum(e)
    return m + np.log(s) / beta

def sm_max_jac(x, n, beta):
    h = x
    ov = compute_overlaps(h, n)
    m = np.max(ov)
    e = np.exp(beta * (ov - m))
    s = np.sum(e)
    f = m + np.log(s) / beta
    w = beta * e / s
    dx = 2.0 / n
    n_full = 2 * n - 1
    grad = np.zeros(n)
    for k in range(n_full):
        if abs(w[k]) < 1e-16:
            continue
        lag = k - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            g[:n-lag] += 1.0 - h[lag:n]
            if lag > 0:
                g[lag:] -= h[:n-lag]
        else:
            ll = -lag
            g[ll:] += 1.0 - h[:n-ll]
            g[:n-ll] -= h[ll:]
        grad += w[k] * dx * g
    return f, grad

             'jac': lambda x: np.ones(n)}]

    for beta in [5, 10, 20, 50, 100, 200, 500, 1000]:
        if time.time() > deadline - 5:
            break
        def obj(x):
            return sm_max(x, n, beta)
        def jac(x):
            return sm_max_jac(x, n, beta)[1]
        res = minimize(obj, best_h, jac=jac, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': 300, 'ftol': 1e-14})
        if res.x is not None and getattr(res, 'fun', 1e9) < best_c5 + 0.01:
            h_new = project_feasible(np.clip(res.x, 0, 1), n)
            c5 = get_c5(h_new, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
    return best_h, best_c5

def opt_grad(h_start, n, deadline):
    """Softmax max gradient descent with line search."""
    best_h = h_start.copy()
    best_c5 = get_c5(h_start, n)
    h = h_start.copy()
    lr = 0.3
    stalled = 0
    it = 0
    beta_seq = [5, 10, 20, 50]
    for beta in beta_seq:
        while time.time() < deadline - 5:
            f, g = sm_max_jac(h, n, beta)
            # Project gradient to satisfy sum constraint
            g = g - np.mean(g)
            h_new = project_feasible(h - lr * g, n)
            c_new = get_c5(h_new, n)
            if c_new < best_c5 - 1e-13:
                best_c5 = c_new
                best_h = h_new.copy()
                stalled = 0
            else:

            h = h_new
            it += 1
            if stalled > 100:

                stalled = 0
                if lr < 0.001:

            if it % 200 == 0 and stalled > 0:

    return best_h, best_c5

from scipy.optimize import minimize

def compute_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def get_c5(h, n):
    return float(np.max(compute_overlaps(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-14:
        mask = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        cnt = float(np.sum(mask))
        if cnt > 0:
            hp[mask] = np.clip(hp[mask] + diff / cnt, 0.0, 1.0)
    return hp

def sm_max_val(x, n, beta):
    h = x
    ov = compute_overlaps(h, n)
    m = np.max(ov)
    e = np.exp(beta * (ov - m))
    s = np.sum(e)
    return m + np.log(s) / beta

def sm_max_jac(x, n, beta):
    h = x
    ov = compute_overlaps(h, n)
    m = np.max(ov)
    e = np.exp(beta * (ov - m))
    s = np.sum(e)
    f = m + np.log(s) / beta
    w = beta * e / s
    dx = 2.0 / n
    n_full = 2 * n - 1
    grad = np.zeros(n)
    for k in range(n_full):
        wk = w[k]
        if abs(wk) < 1e-16:

        lag = k - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            g[:n - lag] += 1.0 - h[lag:]
            if lag > 0:
                g[lag:] -= h[:n - lag]
        else:
            ll = -lag
            g[ll:] += 1.0 - h[:n - ll]
            g[:n - ll] -= h[ll:]
        grad += wk * dx * g
    return f, grad

            h_init = make_structured(n_try, rng, style)
            d_mid = min(time.time() + per_time * 0.45, t_limit - 2)
            bh, bc5 = opt_slsqp(h_init, n_try, d_mid)
            bh2, bc52 = opt_grad(bh, n_try, t_limit)
            if bc52 < bc5:
                bh, bc5 = bh2, bc52
            if bc5 < best_c5:
                best_h, best_c5, best_n = bh, bc5, n_try

    if best_h is not None and time.time() < t0 + budget_s - 10:
        d_final = t0 + budget_s - 5
        bh, bc5 = opt_slsqp(best_h, best_n, d_final)
        if bc5 < best_c5:
            best_c5 = bc5
            best_h = bh
        bh2, bc52 = opt_grad(best_h, best_n, d_final)
        if bc52 < best_c5:
            best_c5 = bc52
            best_h = bh2
        bh3, bc53 = binarize_then_refine(best_h, best_n, rng)
        if bc53 < best_c5:
            best_h, best_c5 = bh3, bc53

    h_final = project_feasible(best_h, best_n)
    c5_final = get_c5(h_final, best_n)
    h_final = np.clip(h_final, 0, 1)
    assert abs(np.sum(h_final) * (2.0 / best_n) - 1.0) < 1e-8
    return h_final, float(c5_final), best_n
