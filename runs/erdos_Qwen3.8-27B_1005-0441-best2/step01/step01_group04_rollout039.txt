                starts.append(_gen_block(n, p, d))
        for f in [1, 2, 3, 5]:
            for a_ in [0.35, 0.45, 0.5]:
                starts.append(_gen_sine(n, f, a_))
        for _ in range(3):
            starts.append(_gen_random(n, rng))

        # Try top candidates
        for h0 in starts[:12]:
            if not _time_left(40):
                break
            h_opt = _optimize_lbfgsb(h0, 300, 1e6, 200)
            h_opt = _fix_sum(h_opt)
            c5 = _eval_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
            if _time_left(30) and c5 < best_c5 + 0.005:
                h_cd = _coord_descent(h_opt, n_sweeps=5)
                c5_cd = _eval_c5(h_cd)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()

    # Phase 3: Intensive coordinate descent on best
    if best_h is not None and _time_left(60):
        h_cd = _coord_descent(best_h, n_sweeps=30)
        c5 = _eval_c5(h_cd)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cd.copy()

    # Phase 4: Another round of L-BFGS-B + coord descent
    if best_h is not None and _time_left(60):
        for alpha in [500, 2000]:
            if not _time_left(40):
                break
            h_opt = _optimize_lbfgsb(best_h, alpha, 1e7, 400)
            h_opt = _fix_sum(h_opt)
            c5 = _eval_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
            if _time_left(30):
                h_cd = _coord_descent(h_opt, n_sweeps=15)
                c5_cd = _eval_c5(h_cd)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()

    # Phase 5: Final fine-tuning with many coord descent sweeps
    if best_h is not None and _time_left(20):
        h_cd = _coord_descent(best_h, n_sweeps=50)
        c5 = _eval_c5(h_cd)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cd.copy()

    if best_h is None:
        best_h = np.full(80, 0.5)
        best_c5 = _eval_c5(best_h)

    best_h = _fix_sum(best_h)
    best_c5 = _eval_c5(best_h)
    return best_h, best_c5, len(best_h)
