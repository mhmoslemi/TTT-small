                idx = np.argsort(h)[:max(1, int(np.ceil(-diff)))]
            h[idx] += diff / len(idx)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def lbbfs_opt(h0, beta, maxiter=300):
    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_val, h0.copy(), jac=smooth_grad, method='L-BFGS-B',
                       bounds=bounds, args=(beta,),
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-11})
        h_out = project(res.x, target)
    except Exception:
        h_out = project(h0.copy(), target)
    return h_out, compute_c5(h_out)

def coord_descent(h0, target, n_sweeps=100, deltas=None):
    n = len(h0)
    if deltas is None:
        deltas = [0.02, 0.01, 0.005, 0.002, -0.002, -0.005, -0.01, -0.02]
    h = h0.copy()
    best = compute_c5(h)
    for _ in range(n_sweeps):
        improved = False
        for j in range(n):
            for delta in deltas:
                h_try = h.copy()
                h_try[j] += delta
                k = (j + 1) % n
                h_try[k] -= delta
                h_try = np.clip(h_try, 0.0, 1.0)
                if abs(h_try.sum() - target) > 1e-10:
                    h_try = project(h_try, target)
                c = compute_c5(h_try)
                if c < best - 1e-13:
                    h = h_try
                    best = c
                    improved = True
                    break
        if not improved:
            break
    return h, best

def make_structured_start(n, rng, idx):
    target = n / 2.0
    if idx == 0:
        h = np.zeros(n)
        h[rng.choice(n, n // 2, replace=False)] = 1.0
    elif idx == 1:
        f = rng.randint(2, 6)
        ph = rng.random() * 2 * np.pi
        h = 0.5 + 0.48 * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif idx == 2:
        h = np.zeros(n)
        block = rng.randint(2, n // 4 + 1)
        for start in range(0, n, block * 2):
            h[start:start + block] = 1.0
    elif idx == 3:
        h = np.zeros(n)
        h[rng.choice(n, n // 2, replace=False)] = 1.0
        h = project(h + 0.5 * rng.randn(n), target)
    else:
        h = np.zeros(n)
        h[rng.choice(n, n // 2, replace=False)] = 1.0
        h = project(h + 0.25 * rng.randn(n), target)
    return project(h, target)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None

    n = len(h_init) if h_init is not None else 70
    target = n / 2.0

    if h_init is not None:
        best_h = project(h_init.copy(), target)
    else:
        best_h = np.zeros(n)
        best_h[:n // 2] = 1.0
        best_h = project(best_h, target)
    best_c5 = compute_c5(best_h)

    # Phase 1: Multi-beta L-BFGS-B from initial
    beta_schedule = [5, 10, 20, 40, 80, 150, 300, 600]
    for beta in beta_schedule:
        t_rem = budget_s - (time.time() - t_start)
        if t_rem < 40:
            break
        h, c = lbbfs_opt(best_h, beta, maxiter=min(400, int(t_rem // 1.5)))
        if c < best_c5:
            best_c5 = c
            best_h = h

    # Phase 2: Structured + random restarts with multi-beta
    t_rem = budget_s - (time.time() - t_start)
    n_restarts = min(int(t_rem / 60), 12)
    for r in range(n_restarts):
        t_rem = budget_s - (time.time() - t_start)
        if t_rem < 60:
            break
        if r < 5:
            h0 = make_structured_start(n, rng, r % 5)
        else:
            scale = 0.1 + 0.05 * rng.random()
            h0 = project(best_h + scale * rng.randn(n), target)
        for beta in [15, 50, 150]:
            t_rem = budget_s - (time.time() - t_start)
            if t_rem < 25:
                break
            h, c = lbbfs_opt(h0, beta, maxiter=200)
            h0 = h
            if c < best_c5:
                best_c5 = c
                best_h = h

    # Phase 3: Fine-tune with more betas on best solution
    t_rem = budget_s - (time.time() - t_start)
    for beta in [400, 800, 1500]:
        if t_rem < 20:
            break
        h, c = lbbfs_opt(best_h, beta, maxiter=300)
        if c < best_c5:
            best_c5 = c
            best_h = h
        t_rem = budget_s - (time.time() - t_start)

    # Phase 4: Aggressive coordinate descent
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 60:
        h_cd, c_cd = coord_descent(best_h, target, n_sweeps=200,
                                   deltas=[0.01, 0.005, 0.002, 0.001,
                                           -0.001, -0.002, -0.005, -0.01])
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd

    # Phase 5: Short-NM style local search via small random steps
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 30:
        h_now = best_h.copy()
        scale = 0.03
        for _ in range(500):
            t_rem = budget_s - (time.time() - t_start)
            if t_rem < 15:
                break
            h_try = project(h_now + scale * rng.randn(n), target)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try
                h_now = h_try
            else:
                scale *= 0.97

    # Phase 6: Final coordinate polish
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 20:
        h_fn, c_fn = coord_descent(best_h, target, n_sweeps=100,
                                   deltas=[0.005, 0.002, 0.001, -0.001, -0.002, -0.005])
        if c_fn < best_c5:
            best_c5 = c_fn
            best_h = h_fn

    best_h = project(best_h, target)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, n)
