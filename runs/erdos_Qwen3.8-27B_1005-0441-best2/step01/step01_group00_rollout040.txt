    return np.correlate(h, 1.0 - h, mode='full')

    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _smoobj(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    shifted = c - cmax
    s = np.exp(shifted / eps)
    return (cmax + eps * np.log(s.sum())) * dx

def _smograd(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    shifted = c - cmax
    s = np.exp(shifted / eps)
    ssum = s.sum()

        d = n * 0.5 - h.sum()
        if abs(d) < 1e-14:
            break

        h = np.clip(h, 0, 1)

    return x.sum() - n * 0.5

    return np.ones(n)

    """Generate a random binary pattern with exactly n/2 ones."""
    indices = rng.permutation(n)

    h[indices[:n//2]] = 1.0

    """Hill climbing on binary pattern."""
    h = h_bin.copy()
    best = _c5(h)
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    for it in range(max_iter):
        c = _corr(h)
        k_worst = int(np.argmax(c))
        m = k_worst - (n - 1)

            good_zeros = [j for j in zeros if j >= m and (j - m) in ones or (j - m) not in range(n)]
        else:
            p = -m
            bad_ones = [i for i in ones if i >= p and (i - p) in zeros]
            good_zeros = [j for j in zeros if j < n - p and (j + p) in ones or (j + p) not in range(n)]
        
        if not bad_ones:
            break
            
        # Try random swap
        i = rng.choice(bad_ones)
        candidates = list(zeros)
        if not candidates:
            break
        j = rng.choice(candidates)
        if i == j:
            continue

        ones.discard(i); ones.add(j)
        zeros.discard(j); zeros.add(i)

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _smoobj(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    shifted = c - cmax
    s = np.exp(shifted / eps)
    return (cmax + eps * np.log(s.sum())) * dx

def _smograd(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    shifted = c - cmax
    s = np.exp(shifted / eps)
    ssum = s.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = s[k] / ssum
        if wk < 1e-12:
            continue
        m = k - (n - 1)
        if m >= 0:
            g[:n - m] += wk
            g[:n - m] -= wk * h[m:]
            g[m:] -= wk * h[:n - m]
        else:
            p = -m
            g[p:] += wk
            g[:n - p] -= wk * h[p:]
            g[p:] -= wk * h[:n - p]
    return g * dx

def _proj(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(2000):
        d = n * 0.5 - h.sum()
        if abs(d) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            h = np.clip(h + d / n, 0, 1)
            break
        h[free] += d / free.sum()
        h = np.clip(h, 0, 1)
    return h

def _eq_fun(x, n):
    return x.sum() - n * 0.5

def _eq_jac(x, n):
    return np.ones(n)

def _gen_binary(n, rng):
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    return h

def _binary_climb(h_bin, n, rng, max_iter=1200):
    h = h_bin.copy().astype(float)
    best = _c5(h)
    for _ in range(max_iter):
        c = _corr(h)
        k_w = int(np.argmax(c))
        m = k_w - (n - 1)
        if m >= 0:
            mask = (h[:n - m] == 1) & (h[m:] == 0)
            bad = np.where(mask)[0]
        else:
            p = -m
            mask = (h[p:] == 1) & (h[:n - p] == 0)
            bad = np.where(mask)[0] + p
        if len(bad) == 0:
            break
        i = bad[rng.randint(len(bad))]
        zi = np.where(h == 0)[0]
        if len(zi) == 0:
            break
        j = zi[rng.randint(len(zi))]
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        nv = _c5(h)
        if nv < best:
            best = nv
        else:
            h[i], h[j] = h[j], h[i]
    return h

def _opt_single(h0, n, t0, budget, rng):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
    h = h0.copy()
    best_val = _c5(h)
    best_h = h.copy()
    epsilons = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    maxits = [150, 250, 350, 400, 500, 600, 700]
    for idx in range(len(epsilons)):
        if time.time() - t0 > budget * 0.88:
            break
        eps = epsilons[idx]
        mt = maxits[idx]
        try:
            res = minimize(_smoobj, h.copy(), args=(eps,),
                          jac=_smograd, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': mt, 'ftol': 1e-13})
            h_new = _proj(res.x, n)
        except Exception:
            continue
        val = _c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = _proj(h_new, n)
        if idx in (2, 4) and time.time() - t0 < budget * 0.85:
            kick = rng.randn(n) * 0.07
            h = _proj(h + kick, n)
    return best_h, best_val
