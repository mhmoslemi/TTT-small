def project_h(h, n):
    """Project h onto {0 ≤ h ≤ 1, sum(h) = n/2}"""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    target = n / 2.0
    if s > 1e-12:
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
        # After clipping, sum might not be exactly n/2, do a second pass
        s2 = h.sum()
        if s2 > 1e-12:
            # Distribute remaining mass to elements not at bounds
            deficit = target - s2
            interior = (h > 0) & (h < 1)
            if interior.any():
                h[interior] += deficit / interior.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_tau = 0.01

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s > 1e-12:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        diff = target - s2
        if abs(diff) > 1e-12:
            interior = (h > 1e-8) & (h < 1.0 - 1e-8)
            if interior.sum() > 0:
                h[interior] += diff / interior.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def _sobj(h):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    t = _g_tau
    if t < 1e-10:
        return m
    return m + t * np.log(np.sum(np.exp((corr - m) / t)))

def _sequal(h):
    return np.sum(h) - _g_n / 2.0

def _sequal_jac(h):
    return np.ones(_g_n)

def slsqp_optimize(h_init, n, tau_schedule, t0, deadline):
    global _g_n, _g_tau
    _g_n = n
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': _sequal, 'jac': _sequal_jac},)
    best = h_init.copy()
    best_val = compute_c5(best)
    for tau in tau_schedule:
        if time.time() > deadline:
            break
        _g_tau = tau
        h = best.copy()
        n_inner = 4 if tau > 0.005 else 6
        for it in range(n_inner):
            if time.time() > deadline:
                break
            try:
                res = minimize(_sobj, h, method='SLSQP', bounds=bounds,
                             constraints=cons,
                             options={'maxiter': 350, 'ftol': 1e-15})
                h_new = np.clip(res.x, 0.0, 1.0)
                h_new = project_feasible(h_new, n)
                v = compute_c5(h_new)
                if v < best_val - 1e-14:
                    best = h_new.copy()
                    best_val = v
                    h = h_new
                else:
                    break
            except Exception:
                break
    return best, best_val

def active_grad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    k_star = int(np.argmax(corr))
    s = k_star - (n - 1)
    if s == 0:
        return dx * (1.0 - 2.0 * h)
    a = abs(s)
    grad = np.zeros(n)
    if s > 0:
        grad[a:] += dx * (1.0 - h[:n - a])
        grad[:n - a] -= dx * h[a:]
    else:
        grad[:n - a] += dx * (1.0 - h[a:])
        grad[a:] -= dx * h[:n - a]
    return grad

def subgrad_refine(h_init, n, t0, deadline):
    best = h_init.copy()
    best_c5 = compute_c5(best)
    h = h_init.copy()
    alpha = 5.0
    stall = 0
    while time.time() < deadline and stall < 80:
        grad = active_grad(h, n)
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best = h_new.copy()
            best_c5 = c5_new
            stall = 0
            alpha = min(alpha * 1.02, 50.0)
        else:
            stall += 1
            alpha *= 0.85
        h = h_new
        if alpha < 0.005:
            break
    return best, best_c5

def _perturb_struct(h, n, seed):
    rng = np.random.RandomState(seed)
    h_p = h.copy()
    for f in range(1, 4):
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.005, 0.03) / f
        h_p += amp * np.sin(2 * np.pi * f * np.arange(n) / n + phase)
    return project_feasible(h_p, n)

def _perturb_gauss(h, n, seed, scale):
    rng = np.random.RandomState(seed)
    return project_feasible(h + scale * rng.randn(n), n)

def _downsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(160, 0.5)
    n = len(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Main SLSQP optimization with decaying temperature
    deadline1 = t0 + min(250, budget_s * 0.30)
    tau1 = [0.04, 0.015, 0.007, 0.003, 0.001, 0.0004]
    best_h, best_c5 = slsqp_optimize(best_h, n, tau1, t0, deadline1)

    # Phase 2: Subgradient refinement on active constraint
    deadline2 = t0 + min(400, budget_s * 0.45)
    h_sg, c5_sg = subgrad_refine(best_h, n, t0, deadline2)
    if c5_sg < best_c5:
        best_h, best_c5 = h_sg, c5_sg

    # Phase 3: Low-temperature SLSQP polish
    deadline3 = t0 + min(600, budget_s * 0.65)
    tau3 = [0.0005, 0.0002, 0.0001, 0.00005]
    best_h, best_c5 = slsqp_optimize(best_h, n, tau3, t0, deadline3)

    # Phase 4: Try lower resolution (n//2) then upsample back
    deadline4 = t0 + min(750, budget_s * 0.80)
    if n > 80:
        n_low = n // 2
        h_low = _downsample(best_h, n_low)
        h_low_opt, c5_low = slsqp_optimize(h_low, n_low,
                                           [0.01, 0.003, 0.001, 0.0003], t0, deadline4)
        if c5_low < 0.45:
            h_up = _downsample(h_low_opt, n)
            h_up, c5_up = slsqp_optimize(h_up, n, [0.005, 0.001, 0.0003], t0, deadline4)
            if c5_up < best_c5:
                best_h, best_c5 = h_up, c5_up

    # Phase 5: Structured perturbations + re-optimize
    n_restarts = 0
    while budget_s - (time.time() - t0) > 70 and n_restarts < 30:
        if n_restarts % 3 == 0:
            h_pert = _perturb_struct(best_h, n, seed + 700 + n_restarts)
        elif n_restarts % 3 == 1:
            h_pert = _perturb_gauss(best_h, n, seed + 1000 + n_restarts, 0.015)
        else:
            h_pert = _perturb_gauss(best_h, n, seed + 2000 + n_restarts, 0.04)
        dl = t0 + budget_s - 40
        h_opt, c5_opt = slsqp_optimize(h_pert, n, [0.003, 0.001, 0.0003], t0, dl)
        if c5_opt < best_c5 - 1e-14:
            best_h, best_c5 = h_opt, c5_opt
        n_restarts += 1

    # Phase 6: Final SLSQP polish
    deadline_final = t0 + budget_s - 5
    best_h, best_c5 = slsqp_optimize(best_h, n, [0.0001, 0.00005], t0, deadline_final)

    # Final subgradient micro-adjustment

        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
