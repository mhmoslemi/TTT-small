import numpy as np
import time, math

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def smooth_max_gradient(h, dx, tau):
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    mc = corr.max()
    shifted = np.clip(tau * (corr - mc), -100, 0)
    exps = np.exp(shifted)
    weights = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-11:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            n_lag = n - lag
            if n_lag > 0:
                grad[:n_lag] += w * (1.0 - h[lag:n])
                grad[lag:n] -= w * h[:n_lag]
        else:
            L = -lag
            n_L = n - L
            if n_L > 0:
                grad[L:n] += w * (1.0 - h[:n_L])
                grad[:n_L] -= w * h[L:n]
    return grad * dx

def continuous_optimize(h_init, time_budget, seed, tau_start=30.0, tau_end=3.0):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    beta = 0.9
    lr = 0.1
    t0 = time.time()
    deadline = t0 + time_budget
    step = 0
    no_improve = 0
    while time.time() < deadline - 0.2:
        frac = (time.time() - t0) / max(time_budget, 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        grad = smooth_max_gradient(h, dx, tau)
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm * min(gnorm, 0.5)
        v = beta * v - lr * grad
        h_new = project_feasible(h + v, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        step += 1
        if no_improve > 200:

            v[:] = 0.0
            lr = max(lr * 0.9, 0.01)
            no_improve = 0
        elif step % 100 == 0:
            lr = max(lr * 0.97, 0.005)
    return best_h, best_c5

def binary_hill_climb(h_frac, time_budget, seed):

    n = len(h_frac)
    target_sum = int(n // 2)
    h = (h_frac > 0.5).astype(float)
    # Adjust to exact sum

    if diff > 0:
        zeros = np.where(h == 0)[0]
        rng.shuffle(zeros)
        h[zeros[:diff]] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        rng.shuffle(ones)
        h[ones[:(-diff)]] = 0.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    deadline = t0 + time_budget
    omh = 1.0 - h

        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Try all swaps, find best
        best_delta = 0
        best_i, best_j = -1, -1
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c = np.correlate(h, 1.0 - h, mode='full').max() * (2.0/n)
                if c < best_c5 - 1e-12:
                    best_delta = c - best_c5
                    best_i, best_j = i, j
                h[i] = 1.0
                h[j] = 0.0
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            best_h = h.copy()
            best_c5 += best_delta
        else:
            # Stuck - try a random pair to escape
            if len(ones) > 1 and len(zeros) > 1:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                c = compute_c5(h)
                if c < best_c5 * 1.01:  # accept slightly worse
                    pass  # keep the move to explore
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            break  # or continue randomly
    return best_h, best_c5, h

import numpy as np
import time, math

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def smooth_grad(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    s = np.clip(tau * (corr - mc), -80, 0)
    ex = np.exp(s)
    w = ex / ex.sum()
    g = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-11:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            m = n - lag
            if m > 0:
                g[:m] += wk * (1.0 - h[lag:])
                g[lag:] -= wk * h[:m]
        else:
            L = -lag
            m = n - L
            if m > 0:
                g[L:] += wk * (1.0 - h[:m])
                g[:m] -= wk * h[L:]
    return g * dx

def cont_opt(h0, tbud, seed, tau_hi=40.0, tau_lo=4.0, lr0=0.08):
    rng = np.random.RandomState(seed)
    n = len(h0)
    ts = n * 0.5
    dx = 2.0 / n
    h = project_feasible(h0.copy(), ts)
    bh, bc = h.copy(), compute_c5(h)
    v = np.zeros(n)
    lr = lr0
    t_s = time.time()
    dl = t_s + tbud
    ni = 0
    while time.time() < dl - 0.1:
        fr = min((time.time() - t_s) / max(tbud, 1e-6), 1.0)
        tau = tau_hi * (tau_lo / tau_hi) ** fr
        g = smooth_grad(h, dx, tau)
        gn = np.linalg.norm(g)
        if gn > 1e-14:
            g = g / gn * min(gn, 0.3)
        v = 0.9 * v - lr * g
        h = project_feasible(h + v, ts)
        c = compute_c5(h)
        if c < bc - 1e-12:
            bh, bc, ni = h.copy(), c, 0
        else:
            ni += 1
        if ni > 250:
            h = project_feasible(bh + rng.randn(n) * 0.025, ts)
            v = np.zeros(n)
            lr = max(lr * 0.8, 0.01)
            ni = 0
    return bh, bc

def hill_climb(h, dx, max_iter=400, time_limit=8.0):
    n = len(h)
    bh = h.copy()
    bc = float(np.correlate(h, 1.0 - h, mode='full').max() * dx)
    t0 = time.time()
    for _ in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_c, best_pj = bc, None
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                c = float(np.correlate(h, 1.0 - h, mode='full').max() * dx)
                h[i], h[j] = 1.0, 0.0
                if c < best_c - 1e-15:
                    best_c, best_pj = c, (i, j)
        if best_pj is None:
            break
        h[best_pj[0]], h[best_pj[1]] = 0.0, 1.0
        bh, bc = h.copy(), best_c
    return bh, bc

def bin_ils(hf, tbud, seed):
    rng = np.random.RandomState(seed)
    n = len(hf)
    ts = n // 2
    dx = 2.0 / n
    h = (hf > 0.5).astype(float)
    d = ts - int(h.sum())
    if d > 0:
        z = np.where(h == 0)[0]
        rng.shuffle(z)
        h[z[:d]] = 1
    elif d < 0:
        o = np.where(h == 1)[0]
        rng.shuffle(o)
        h[o[:(-d)]] = 0
    bh, bc = h.copy(), compute_c5(h)
    t0 = time.time()
    dl = t0 + tbud
    while time.time() < dl - 1.0:
        h_cur, c_cur = hill_climb(h.copy(), dx, max_iter=300, time_limit=6.0)
        if c_cur < bc - 1e-12:
            bh, bc = h_cur.copy(), c_cur
            h = h_cur.copy()
        if time.time() > dl - 1.5:
            break
        ks = rng.randint(1, 4)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) >= ks and len(zeros) >= ks:
            ki = rng.choice(ones, size=ks, replace=False)
            kj = rng.choice(zeros, size=ks, replace=False)
            h_k = h.copy()
            h_k[ki] = 0
            h_k[kj] = 1
            h_rec, c_rec = hill_climb(h_k, dx, max_iter=200, time_limit=4.0)
            if c_rec < bc - 1e-12:
                bh, bc = h_rec.copy(), c_rec
                h = h_rec.copy()
            elif c_rec < bc * 1.03:
                h = h_rec.copy()
    return bh, bc
