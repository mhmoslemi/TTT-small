    g = np.zeros(n)
    if m >= 0:
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]
    else:
        p = -m
        g[p:] += (1.0 - h[:n - p])
        g[:n - p] -= h[p:]
    return g * dx

def _smograd(h, eps):
    n = len(h)
    c = _corr(h)
    cmax = c.max()
    shifted = c - cmax
    s = np.exp(shifted / eps)
    ssum = s.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = s[k] / ssum
        if wk < 1e-14:
            continue
        m = k - (n - 1)
        g += wk * _grad_k(h, m)
    return g

def _proj(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        d = n * 0.5 - h.sum()
        if abs(d) < 1e-15:
            break
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        if free.sum() == 0:
            h = np.clip(h + d / n, 0, 1)
            break
        h[free] += d / free.sum()
        h = np.clip(h, 0, 1)
    return h

def _eq_fun(x, n):
    return x.sum() - n * 0.5

def _eq_jac(x, n):
    return np.ones(n)

def _nesterov_polish(h, n, t0, t_end):
    best_h = h.copy()
    best_val = _c5(h)
    x = h.copy()
    y = h.copy()
    t_k = 1.0
    lr0 = 0.4
    for it in range(20000):
        if time.time() > t_end:
            break
        c = _corr(y)
        k_star = int(np.argmax(c))
        m = k_star - (n - 1)
        g = _grad_k(y, m)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new
        lr = lr0 / (1.0 + 0.02 * it)
        x_new = _proj(x - lr * g, n)
        y = _proj(x_new + beta * (x_new - x), n)
        x = x_new
        t_k = t_new
        val = _c5(y)
        if val < best_val:
            best_val = val
            best_h = y.copy()
    return best_h, best_val

def _active_polish(h, n, t0, t_end, topk=2):
    best_h = h.copy()
    best_val = _c5(h)
    x = h.copy()
    lr = 0.02
    stall = 0
    for it in range(50000):
        if time.time() > t_end:
            break
        c = _corr(x)
        top_idxs = np.argsort(c)[-topk:]
        g = np.zeros(n)
        for k in top_idxs:
            g += _grad_k(x, k - (n - 1))
        g /= topk
        if np.linalg.norm(g) < 1e-14:
            break
        x_new = _proj(x - lr * g, n)
        val = _c5(x_new)
        if val < best_val - 1e-15:
            best_val = val
            best_h = x_new.copy()
            x = x_new
            lr = min(lr * 1.005, 0.08)
            stall = 0
        else:
            lr *= 0.97
            stall += 1
            if stall > 5000 or lr < 1e-10:
                break
    return best_h, best_val

def _slsqp_run(h0, n, eps, maxit, t0, t_limit):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_smoobj, h0.copy(), args=(eps,),
                       jac=_smograd, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxit, 'ftol': 1e-14})
        return _proj(res.x, n)
    except Exception:
        return None

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_val = np.inf

    starts = []
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        h_init = _proj(h_init, len(h_init))
        starts.append(h_init)
    except Exception:
        starts.append(np.full(100, 0.5))

    n0 = len(starts[0])
    for freq in [2, 3, 5, 7]:
        for amp in [0.35, 0.5]:
            ph = rng.rand() * 2 * np.pi
            hh = 0.5 + amp * np.sin(2.0 * np.pi * freq * np.arange(n0) / n0 + ph)
            hh = _proj(hh, n0)
            starts.append(hh)
    for n_try in [100, 140]:
        hh = 0.5 + 0.45 * np.sin(2.0 * np.pi * 3.0 * np.arange(n_try) / n_try)
        starts.append(_proj(hh, n_try))

    # Phase 1: SLSQP with correct gradient
    t1 = t0 + budget_s * 0.45
    for idx, h0 in enumerate(starts):
        if time.time() > t1:
            break
        n = len(h0)
        for eps in [0.08, 0.03, 0.01, 0.004, 0.0015]:
            if time.time() > t1:
                break
            rem = t1 - time.time()
            maxit = max(100, int(rem * 1.5))
            h_res = _slsqp_run(h0, n, eps, maxit, t0, t1)
            if h_res is None:
                continue
            val = _c5(h_res)
            if val < best_val:
                best_val = val
                best_h = h_res.copy()
                h0 = h_res.copy()

    # Phase 2: Nesterov subgradient
    t2 = t0 + budget_s * 0.65
    if best_h is not None and time.time() < t2:
        n = len(best_h)
        hp, vp = _nesterov_polish(best_h, n, t0, t2)
        if vp < best_val:
            best_val = vp
            best_h = hp

    # Phase 3: SLSQP ultra-polish
    t3 = t0 + budget_s * 0.85
    if best_h is not None and time.time() < t3:
        n = len(best_h)
        for eps in [0.003, 0.001, 0.0005, 0.0002]:
            if time.time() > t3:
                break
            rem = t3 - time.time()
            if rem < 5:
                break
            maxit = max(200, int(rem * 2))
            h_res = _slsqp_run(best_h, n, eps, maxit, t0, t3)
            if h_res is not None and _c5(h_res) < best_val:
                best_val = _c5(h_res)
                best_h = h_res

    # Phase 4: Active-set fine polish
    t4 = t0 + budget_s * 0.985
    if best_h is not None and time.time() < t4:
        n = len(best_h)
        for tk in [3, 1]:
            if time.time() > t4:
                break
            hp, vp = _active_polish(best_h, n, t0, t4, topk=tk)
            if vp < best_val:
                best_val = vp
                best_h = hp

    if best_h is None:
        n = 100
        best_h = np.full(n, 0.5)
        best_val = _c5(best_h)

    best_h = _proj(best_h, len(best_h))
    best_val = _c5(best_h)
    return (best_h, best_val, len(best_h))
