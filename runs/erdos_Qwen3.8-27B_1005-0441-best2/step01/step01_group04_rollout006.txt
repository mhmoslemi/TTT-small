def _eval_c5(h):
    dx = 2.0 / len(h)

def _fix_sum(h, n):

            for i in idx:

                add = min(diff, room[i])
                if add < 1e-15:
                    break
                h[i] += add
                diff -= add

    cmax = np.max(corr)
