    t_rem = budget_s - (time.time() - t_start)
    n_rest = min(int(t_rem / 60), 4) if t_rem > 60 else 0
    for i in range(n_rest):
        t_rem = budget_s - (time.time() - t_start)
        if t_rem < 40:
            break
        if i == 0:
            hp = np.roll(best_h, rng.randint(1, n))
        elif i == 1:
            hp = best_h + 0.12 * rng.randn(n)
        elif i == 2:
            hp = np.zeros(n)
            hp[rng.choice(n, n // 2, replace=False)] = 1.0
            hp = hp + 0.03 * rng.randn(n)
        else:
            alpha = 0.6180339887 * (1.0 + 0.05 * i)
            hp = np.zeros(n)
            for j in range(n):
                hp[j] = 1.0 if ((j * alpha) % 1.0) < 0.5 else 0.0
        hp = project(hp.astype(float))
        ho, co = cont_opt(hp, min(35, t_rem - 10), [50.0, 150.0, 500.0])
        if co < best_c5:
            best_c5 = co
            best_h = ho

    # Phase 4: Coordinate descent + final NM
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 8:
        t_lim = t_start + budget_s - 5
        hcd, ccd = coord_descent(best_h, max_passes=4, t_limit=t_lim)
        if ccd < best_c5:
            best_c5 = ccd
            best_h = hcd
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 12:
        hnm = nm_polish(best_h, maxiter=300)
        cnm = compute_c5(hnm)
        if cnm < best_c5:
            best_c5 = cnm
            best_h = hnm

    best_h = project(best_h)
    fc = compute_c5(best_h)
    if fc < best_c5:
        best_c5 = fc
    return (best_h, best_c5, n)
