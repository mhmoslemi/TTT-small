import numpy as np

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def to_binary(h, n):
    target = n // 2
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin

def precompute_grads(h):
    N = len(h)
    omh = 1.0 - h
    grads = np.zeros((N, 2*N-1))
    for i in range(N):
        # i as first position: s in [-i, N-1-i], clipped to [-(N-1), N-1]
        lo = max(-i, -(N-1))
        hi = min(N-1-i, N-1)
        if lo <= hi:
            s = np.arange(lo, hi+1)
            grads[i, N-1+s] += omh[i+s]
        # i as second position: s in [i-N+1, i], clipped
        lo2 = max(i-N+1, -(N-1))
        hi2 = min(i, N-1)
        if lo2 <= hi2:
            s2 = np.arange(lo2, hi2+1)
            grads[i, N-1+s2] -= h[i-s2]
    return grads
