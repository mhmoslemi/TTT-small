            if j1 == j2:
                no_improve += 1
                temp *= cool
                continue
            b[i1] = 0
            b[i2] = 0
            b[j1] = 1
            b[j2] = 1
            nc = compute_c5(b)
            delta = nc - curr_c5
            if delta <= 0 or (temp > 1e-10 and math.log(rng.random() + 1e-15) > -delta / temp):
                curr_c5 = nc
                if curr_c5 < best_c5:
                    best_c5 = curr_c5
                    best_b[:] = b
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                b[i1] = 1
                b[i2] = 1
                b[j1] = 0
                b[j2] = 0
                no_improve += 1
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i] = 0
            b[j] = 1
            nc = compute_c5(b)
            delta = nc - curr_c5
            if delta <= 0 or (temp > 1e-10 and math.log(rng.random() + 1e-15) > -delta / temp):
                curr_c5 = nc
                if curr_c5 < best_c5:
                    best_c5 = curr_c5
                    best_b[:] = b
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                b[i] = 1
                b[j] = 0
                no_improve += 1
        temp *= cool
        if temp < t_lo:
            temp = t_lo
    return best_b, best_c5

def bi_ls(b, time_budget):
    """Best-improvement local search: try all 1-swaps, then 2-swaps."""
    n = len(b)
    dx = 2.0 / n
    b = b.copy()
    best_c5 = compute_c5(b)
    t0 = time.time()
    improved_any = True
    while improved_any and (time.time() - t0) < time_budget - 1.0:
        improved_any = False
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        best_delta = 1e10
        best_i = best_j = -1
        for i in ones:
            b[i] = 0
            for j in zeros:
                b[j] = 1
                c5 = compute_c5(b)
                d = c5 - best_c5
                if d < best_delta:
                    best_delta = d
                    best_i, best_j = i, j
                b[j] = 0
            b[i] = 1
        if best_i >= 0 and best_delta < -1e-12:
            b[best_i] = 0
            b[best_j] = 1
            best_c5 += best_delta
            improved_any = True
        if (time.time() - t0) < time_budget - 5.0:
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            best_d2 = 1e10
            best_4 = None
            for ii in range(len(ones)):
                for jj in range(ii + 1, len(ones)):
                    i1, i2 = ones[ii], ones[jj]
                    b[i1] = 0
                    b[i2] = 0
                    for k in range(len(zeros)):
                        for l in range(k + 1, len(zeros)):
                            j1, j2 = zeros[k], zeros[l]
                            b[j1] = 1
                            b[j2] = 1
                            c5 = compute_c5(b)
                            d = c5 - best_c5
                            if d < best_d2:
                                best_d2 = d
                                best_4 = (i1, i2, j1, j2)
                            b[j2] = 0
                            b[j1] = 0
                    b[i2] = 1
                    b[i1] = 1
            if best_4 is not None and best_d2 < -1e-12:
                i1, i2, j1, j2 = best_4
                b[i1] = 0
                b[i2] = 0
                b[j1] = 1
                b[j2] = 1
                best_c5 += best_d2
                improved_any = True
    return b, best_c5

def frac_refine(h_init, time_budget, seed):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + time_budget
    lr = 0.04
    beta = 0.8
    v = np.zeros(n)
    tau = 100.0
    no_impr = 0
    while time.time() < deadline - 0.5:
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = corr.max()
        sh = np.clip(tau * (corr - mc), -40, 40)
        ex = np.exp(sh)
        sw = ex.sum()
        if sw < 1e-30:
            break
        wts = ex / sw
        grad = np.zeros(n)
        nc = len(corr)
        for k in range(nc):
            w = wts[k]
            if w < 1e-13:
                continue
            L = k - (n - 1)
            if L >= 0:
                grad[L:] += w * (1.0 - h[:n - L])
                grad[:n - L] -= w * h[L:]
            else:
                LL = -L
                grad[:n - LL] += w * (1.0 - h[LL:])
                grad[LL:] -= w * h[:n - LL]
        grad *= dx
        v = beta * v - lr * grad
        hn = project_feasible(h + v, target)
        c5n = compute_c5(hn)
        if c5n < best_c5:
            best_c5 = c5n
            best_h[:] = hn
            no_impr = 0
        else:
            no_impr += 1
        h[:] = hn
        lr *= 0.9997
        tau = min(tau * 1.004, 600.0)
        if no_impr > 2000:
            h[:] = project_feasible(best_h + rng.randn(n) * 0.03, target)
            v[:] = 0
            lr = 0.04
            tau = 100.0
            no_impr = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s
    best_h = None
    best_c5 = 1.0
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = len(h0)
    else:
        h0 = None
        n0 = 70
    n_list = [n0, 64, 80, 72]
    seen = set()
    nfl = []
    for x in n_list:
        if x not in seen and 0 < x < 1000:
            seen.add(x)
            nfl.append(x)
    for ni, n in enumerate(nfl):
        elapsed = time.time() - t_start
        remaining = budget_s - elapsed - 30
        if remaining < 20:
            break
        n_inits = 3 if ni == 0 else 2
        for trial in range(n_inits):
            elapsed = time.time() - t_start
            remaining = budget_s - elapsed - 25
            if remaining < 15:
                break
            t_sa = min(remaining / (n_inits - trial + 0.5) * 0.65, 280)
            t_ls = min(remaining / (n_inits - trial + 0.5) * 0.2, 60)
            t_fr = min(remaining / (n_inits - trial + 0.5) * 0.2 - 5, 50)
            if t_sa < 8:
                break
            if trial == 0 and n == n0 and h0 is not None:
                h_init = h0.copy()
            elif trial == 0:
                h_init = golden_init(n)
            elif trial == 1:
                h_init = tm_init(n)
            else:
                h_init = np.random.RandomState(seed + trial * 7 + n * 3).rand(n)
            bb, c5s = sa_binary(h_init, t_sa, seed + trial * 1000 + n * 100)
            if c5s < best_c5:
                best_c5 = c5s
                best_h = bb.copy()
            bb2, c5s2 = sa_binary(bb, min(t_sa * 0.6, 120), seed + 50000 + trial + n)
            if c5s2 < best_c5:
                best_c5 = c5s2
                best_h = bb2.copy()
                bb = bb2
            if t_ls > 5:
                bb3, c5ls = bi_ls(bb, t_ls)
                if c5ls < best_c5:
                    best_c5 = c5ls
                    best_h = bb3.copy()
            if t_fr > 5:
                hf, c5f = frac_refine(best_h, t_fr, seed + 77777 + trial + n)
                if c5f < best_c5:
                    best_c5 = c5f
                    best_h = hf.copy()
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - 5
    if remaining > 20 and best_h is not None:
        nb = len(best_h)
        bb, c5s = sa_binary(best_h, min(remaining * 0.5, 120), seed + 999999, 0.015, 0.00003)
        if c5s < best_c5:
            best_c5 = c5s
            best_h = bb.copy()
        rem2 = budget_s - (time.time() - t_start) - 3
        if rem2 > 8:
            hf, c5f = frac_refine(best_h, min(rem2, 40), seed + 55555)
            if c5f < best_c5:
                best_c5 = c5f
                best_h = hf.copy()
    if best_h is None:
        best_h = golden_init(nfl[0])
        best_c5 = compute_c5(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
