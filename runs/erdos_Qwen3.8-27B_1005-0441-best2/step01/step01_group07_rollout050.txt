def project_feasible(h, target_sum):
    """Project h onto {0 <= x <= 1, sum(x) = target_sum}"""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def compute_smooth_grad(h, dx, tau):
    """Compute gradient of smooth approximation of max_k corr[k] * dx"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    shifted = tau * (corr - mc)
    exps = np.exp(shifted)
    weights = exps / exps.sum()
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k] * dx
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:-lag] -= w * h[lag:]
        else:
            L = -lag
            grad[:-L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n-L]
    return grad

import numpy as np
import time, math

def project_feasible(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def topk_grad(h, k_top):
    """Gradient using equal weights on top-k lags"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[::-1][:k_top]
    w = np.zeros(len(corr))
    w[idx] = 1.0 / k_top
    grad = np.zeros(n)
    for kk in idx:
        wk = w[kk] * dx
        lag = kk - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wk * (1.0 - h[:n-lag])
            grad[:n-lag] -= wk * h[lag:]
        else:
            L = -lag
            grad[:n-L] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:n-L]
    return grad

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    exps = np.exp(tau * (corr - mc))
    w = exps / exps.sum()
    grad = np.zeros(n)

        if wk < 1e-13: continue
