import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        h = np.clip(h + (target - s) / n, 0.0, 1.0)
    return h

def _binarize(h):
    n = len(h)
    idx = np.argsort(h)[::-1]
    result = np.zeros(n)
    result[idx[:n // 2]] = 1.0
    return result

def _swap_search(h, t_end, max_rounds=1000):
    n = len(h)
    h = np.array(h, dtype=float)
    best_c5 = compute_c5(h)
    stall = 0
    for rnd in range(max_rounds):
        if time.time() > t_end:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 0.0
        best_i, best_j = -1, -1
        for i in ones:
            h[i] = 0.0
            for j in zeros:
                h[j] = 1.0
                c = compute_c5(h)
                h[j] = 0.0
                delta = c - best_c5
                if delta < best_delta - 1e-15:
                    best_delta = delta
                    best_i, best_j = i, j
            h[i] = 1.0
        if best_i < 0:
            stall += 1
            if stall > 3:
                break
            continue
        stall = 0
        h[best_i] = 0.0
        h[best_j] = 1.0
        best_c5 += best_delta
    return h, best_c5

def _sim_anneal(h, t_end, rng, t0=0.01, t1=0.0003, max_iter=50000):
    n = len(h)
    h = np.array(h, dtype=float)
    cur = compute_c5(h)

        if time.time() > t_end:

        frac = it / max_iter
        T = t0 * (t1 / t0) ** frac

        zeros = np.where(h < 0.5)[0]
        if not len(ones) or not len(zeros):
