```python
import time

def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def get_max_corr(h):
    return float(np.max(compute_correlation(h)))

    return float(np.max(compute_correlation(h))) * dx

def make_binary(h, n):
    """Round h to binary, fixing sum to n//2"""
    b = (h >= 0.5).astype(float)
    target = n // 2
    current = int(b.sum())
    if current > target:
        # Need to flip some 1s to 0s
        ones = np.where(b == 1)[0]
        # Flip the ones with lowest h values
        order = np.argsort(h[ones])
        for idx in order[:current - target]:
            b[ones[idx]] = 0
    elif current < target:
        zeros = np.where(b == 0)[0]
        order = np.argsort(-h[zeros])
        for idx in order[:target - current]:
            b[zeros[idx]] = 1
    return b

    no_improve = 0

        zeros = np.where(h == 0)[0]

                h[j] = 1.0
                new_max = get_max_corr(h)
                h[i] = 1.0
                h[j] = 0.0
                
                if new_max < best_new_max - 1e-10:

                    best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            best_max = best_new_max
            best_h = h.copy()
            no_improve = 0

                    j = zeros[rng.randint(len(zeros))]

                no_improve = 0

        c5 = bmax * 2.0 / n_try
        if c5 < best_c5:

            best_h = bh
```

```python
    assert abs(np.sum(best_h) * 2.0 / best_n - 1.0) < 1e-8
    return best_h, float(c5_final), best_n
```

a[p] contributes when i=p: a[p] * b[p - lag]. Valid when 0 ≤ p - lag < n.
b[p] contributes when i - lag = p, i.e., i = p + lag: a[p+lag] * b[p] = a[p+lag] * (1-h[p]). Valid when 0 ≤ p+lag < n.

So changing h[p] affects:
- For each lag where 0 ≤ p-lag < n: the term a[p]*b[p-lag] changes
- For each lag where 0 ≤ p+lag < n: the term a[p+lag]*b[p] changes

For our swap (h[i]: 1→0, h[j]: 0→1):
Δresult[k] for lag d = k-(n-1):
- From h[i] change (1→0):
  - a[i] term: Δ = (0-1)*b[i-d] = -b[i-d] = -(1-h[i-d]) [if 0≤i-d<n]
  - b[i] term: Δ = a[i+d]*(1-0) - a[i+d]*(1-1) = a[i+d] - 0 = a[i+d] = h[i+d] [if 0≤i+d<n]
- From h[j] change (0→1):
  - a[j] term: Δ = (1-0)*b[j-d] = b[j-d] = (1-h[j-d]) [if 0≤j-d<n]
  - b[j] term: Δ = a[j+d]*(1-1) - a[j+d]*(1-0) = 0 - a[j+d] = -h[j+d] [if 0≤j+d<n]

So Δcorr[d] = -(1-h[i-d])*[0≤i-d<n] + h[i+d]*[0≤i+d<n] + (1-h[j-d])*[0≤j-d<n] - h[j+d]*[0≤j+d<n]

This is O(1) per lag! And there are 2n-1 lags. So updating the full correlation after a swap is O(n).

Let me implement this for speed:

```python
def swap_update_corr(corr, h, i, j, n):
    """Update correlation vector after swapping h[i]=1->0 and h[j]=0->1"""
    # i was 1, j was 0
    d_range = np.arange(-(n-1), n)  # lags
    # For each lag d, compute delta
    # delta[d] = -(1-h[i-d]) if 0<=i-d<n + h[i+d] if 0<=i+d<n
    #            + (1-h[j-d]) if 0<=j-d<n - h[j+d] if 0<=j+d<n
    
    # Precompute for position i
    delta = np.zeros(2*n-1)
    # h[i] changes from 1 to 0:
    # a[i] term: -(1-h[i-d]) for d where 0<=i-d<n, i.e., i-n+1 <= d <= i
    lo_i = max(-(n-1), i-n+1)
    hi_i = min(n-1, i)
    for d in range(lo_i, hi_i+1):
        idx = d + (n-1)
        delta[idx] += -(1.0 - h[i-d])
    # b[i] term: +h[i+d] for d where 0<=i+d<n, i.e., -i <= d <= n-1-i
    lo_i2 = max(-(n-1), -i)
    hi_i2 = min(n-1, n-1-i)
    for d in range(lo_i2, hi_i2+1):
        idx = d + (n-1)
        delta[idx] += h[i+d]
    
    # h[j] changes from 0 to 1:
    lo_j = max(-(n-1), j-n+1)
    hi_j = min(n-1, j)
    for d in range(lo_j, hi_j+1):
        idx = d + (n-1)
        delta[idx] += (1.0 - h[j-d])
    lo_j2 = max(-(n-1), -j)
    hi_j2 = min(n-1, n-1-j)
    for d in range(lo_j2, hi_j2+1):
        idx = d + (n-1)
        delta[idx] += -h[j+d]
    
    return corr + delta
```

import time

def get_max_corr(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def get_c5(h, n):
    return get_max_corr(h) * 2.0 / n

def to_binary(h, n):
    b = np.zeros(n)
    target = n // 2
    order = np.argsort(-h)
    b[order[:target]] = 1.0
    return b

def local_search(h, n, deadline, rng, sa_temp=0):
    best_h = h.copy()
    best_max = get_max_corr(h)
    no_improve = 0
    
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
