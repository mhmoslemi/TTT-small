        grad[:n - e] += (-h[e:])
    return grad * (2.0 / n)

def _pgd(h, alpha, lr, n_iter):
    h = h.copy()
    n = len(h)
    best = _eval_c5(h)
    for it in range(n_iter):
        g = _grad_smoothed(h, alpha)
        step = lr
        improved = False
        for _ in range(10):
            h_new = _project(h - step * g, n)
            c5 = _eval_c5(h_new)
            if c5 < best - 1e-14:
                h = h_new
                best = c5
                improved = True
                break
            step *= 0.5
        if not improved:
            break
    return h, best

def _chase_max(h, n_iter):
    h = h.copy()
    n = len(h)
    best = _eval_c5(h)
    for it in range(n_iter):
        corr = _corr_vals(h)
        k = int(np.argmax(corr))
        d = k - (n - 1)
        g = _grad_lag(h, d)
        step = 0.05
        improved = False
        for _ in range(8):
            h_new = _project(h - step * g, n)
            c5 = _eval_c5(h_new)
            if c5 < best - 1e-14:
                h = h_new
                best = c5
                improved = True
                break
            step *= 0.5
        if not improved:
            break
    return h, best

def _coord_desc(h, n_iter, rng):
    h = h.copy()
    n = len(h)
    best = _eval_c5(h)
    deltas = [0.02, 0.05, 0.1, 0.15, 0.25]
    for it in range(n_iter):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        for delta in deltas:
            for sgn in (1.0, -1.0):
                h_try = h.copy()
                h_try[i] += sgn * delta
                h_try[j] -= sgn * delta
                if h_try[i] < -1e-9 or h_try[i] > 1 + 1e-9 or h_try[j] < -1e-9 or h_try[j] > 1 + 1e-9:
                    continue
                h_try = np.clip(h_try, 0.0, 1.0)
                c5 = _eval_c5(h_try)
                if c5 < best - 1e-10:
                    h = h_try
                    best = c5
                    break
    return h, best

def _block_init(n):
    period = 3
    h = np.where(np.arange(n) % period < 2, 0.8, 0.2)
    return _project(h.astype(float), n)

def _sin_init(n, freq):
    x = np.linspace(0, 2.0 * np.pi * freq, n, endpoint=False)
    h = 0.5 + 0.4 * np.sin(x)
    return _project(h, n)

def _rand_init(n, rng):
    return _project(rng.random(n), n)

def _lbgfsmm(h0, max_iter=300):
    n = len(h0)
    t0_init = _eval_c5(h0)
    x0 = np.concatenate([h0, [t0_init]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]

    def obj(x):
        return x[-1]

    def ineq(x):
        hv = x[:-1]
        tv = x[-1]
        c = np.correlate(hv, 1.0 - hv, mode='full') * (2.0 / n)
        return tv - c

    def eq(x):
        return x[:-1].sum() - n / 2.0

    try:
        res = minimize(obj, x0, method='SLSQP',
                       constraints=[{'type': 'ineq', 'fun': ineq},
                                    {'type': 'eq', 'fun': eq}],
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-13})
        return res.x[:-1]
    except Exception:
        return h0.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    best_c5 = np.inf
    best_h = None

    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if h_init is not None:
        n0 = len(h_init)
        h_init = _fix_sum(h_init, n0)
        c5 = _eval_c5(h_init)
        best_c5 = c5
        best_h = h_init.copy()

        # Phase 1: Projected gradient with increasing alpha
        h_c = h_init.copy()
        for alpha, lr, nit in [(20, 0.03, 300), (50, 0.02, 300), (100, 0.015, 300)]:
            if time.time() - t0 > budget_s - 150:
                break
            h_c, c5 = _pgd(h_c, alpha, lr, nit)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_c.copy()

        # Phase 2: Chase max
        if time.time() - t0 < budget_s - 80:
            h_c2, c5 = _chase_max(best_h, 500)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_c2.copy()

        # Phase 3: SLSQP minimax
        if time.time() - t0 < budget_s - 60:
            h_mm = _lbgfsmm(best_h, 400)
            h_mm = _fix_sum(h_mm, len(h_mm))
            c5 = _eval_c5(h_mm)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_mm.copy()

        # Phase 4: More chase + coordinate descent
        if time.time() - t0 < budget_s - 40:
            h_c3, c5 = _chase_max(best_h, 400)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_c3.copy()
        if time.time() - t0 < budget_s - 20:
            h_cd, c5 = _coord_desc(best_h, 500, rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cd.copy()

    # Try other n values
    for n in [40, 47, 50, 60, 67, 80]:
        if time.time() - t0 > budget_s - 60:
            break
        inits = [_block_init(n), _sin_init(n, 1), _sin_init(n, 2),
                 _rand_init(n, rng), _sin_init(n, 3)]
        for h0 in inits:
            if time.time() - t0 > budget_s - 30:
                break
            h_opt, c5 = _pgd(h0, 30, 0.03, 200)
            h_opt, c5 = _pgd(h_opt, 100, 0.015, 200)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
            h_ch, c5 = _chase_max(h_opt, 200)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_ch.copy()

    # Final polish
    if best_h is not None:
        if time.time() - t0 < budget_s - 30:
            h_f, c5 = _chase_max(best_h, 500)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_f.copy()
        if time.time() - t0 < budget_s - 15:
            h_cd, c5 = _coord_desc(best_h, 400, rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cd.copy()
        if time.time() - t0 < budget_s - 10:
            h_pg, c5 = _pgd(best_h, 500, 0.01, 200)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_pg.copy()

    if best_h is None:
        best_h = np.full(50, 0.5)
        best_c5 = _eval_c5(best_h)

    best_h = _fix_sum(best_h, len(best_h))
    best_c5 = _eval_c5(best_h)
    return best_h, best_c5, len(best_h)
