def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:

        cnt = np.sum(mask)
        if cnt > 0:

            np.clip(h, 0.0, 1.0, out=h)
    return h

    n = len(h0)
    h = h0.copy()
    y = h0.copy()
    t = 1.0
    # Lipschitz estimate: L = O(n * dx) roughly. Let's use a simple overestimate.
    # Actually L for the smooth max... let's just use a conservative value.

            # Golden section

from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += resid / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def grad_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = alpha * (corr - np.max(corr))
    shifted = np.clip(shifted, -500.0, 500.0)

    cw = np.concatenate(([0.0], np.cumsum(w)))
    sum_w = cw[1:n+1] - cw[:-n]
    sum_wh = np.correlate(w, h, mode='valid')
    return dx * (sum_w - 2.0 * sum_wh)

def fista(h0, alpha, max_iter, t_end):

    h = project_feasible(h0.copy(), n)
    y = h.copy()

    return h, best_f

def coord_sweep(h, n_sweeps, t_end):

    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for j in range(n):
            corr = np.correlate(h, 1.0 - h, mode='full')
            dx = 2.0 / n
            hi = h[j]
            m_lo = max(j, 0)
            m_hi = min(j + n, 2 * n - 1)
            m_idx = np.arange(m_lo, m_hi)
            k_idx = m_idx - j
            b = 1.0 - 2.0 * h[k_idx]
            a = corr[m_idx] - b * hi
            fixed_mask = np.ones(2 * n - 1, dtype=bool)
            fixed_mask[m_idx] = False
            C = np.max(corr[fixed_mask]) if np.any(fixed_mask) else 0.0
            a_min = np.min(a)
            a_max = np.max(a)

            b_neg = b[b < 0]
