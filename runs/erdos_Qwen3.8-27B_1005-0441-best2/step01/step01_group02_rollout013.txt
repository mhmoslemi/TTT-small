def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_detail(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def _smooth_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    return float((mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta) * dx)

def _smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-14:
            continue
        d = (n - 1) - i
        lo1 = max(0, -d)
        hi1 = min(n, n - d)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + d:hi1 + d])
        lo2 = max(0, d)
        hi2 = min(n, d + n)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2 - d:hi2 - d]
    return grad * dx

def _project(h):

def _subgradient_step(h, top_k=5, step_size=0.01):
    """Take a subgradient step focusing on worst shifts."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mc = np.max(corr)
    # Find top_k worst shifts
    idx = np.argsort(corr)[-top_k:]
    # Compute subgradient: for each worst shift, gradient of that correlation
    grad = np.zeros(n)
    for i in idx:
        d = (n - 1) - i
        lo1 = max(0, -d)
        hi1 = min(n, n - d)
        if lo1 < hi1:
            grad[lo1:hi1] += (1.0 - h[lo1 + d:hi1 + d])

    grad *= dx / top_k
    # Step

The trick for equality with L-BFGS-B: We can parameterize h[0..n-2] freely and set h[n-1] = n/2 - sum(h[0..n-2]). Then we only have bounds on h[0..n-2] plus the constraint 0 ≤ h[n-1] ≤ 1, i.e., n/2 - 1 ≤ sum(h[0..n-2]) ≤ n/2.

This is still an inequality constraint. Alternatively, we can just use a large penalty on (sum - n/2)².

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _smooth_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    return float((mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta) * dx)

def _smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-14:
            continue
        d = (n - 1) - i
        lo1 = max(0, -d)
        hi1 = min(n, n - d)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + d:hi1 + d])
        lo2 = max(0, d)
        hi2 = min(n, d + n)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2 - d:hi2 - d]
    return grad * dx

def _project(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
        free = (h > 0) & (h < 1)
        if free.sum() == 0:
            h = np.clip(h + (target - s) / n, 0, 1)
        else:
            adj = (target - s) / free.sum()
            h[free] += adj
            h = np.clip(h, 0.0, 1.0)
    return h

def _lbfgsb_phase(h0, beta, maxiter=3000, pen=5000.0):
    n = len(h0)
    target = n / 2.0
    def obj(x):
        return _smooth_val(x, beta) + pen * (x.sum() - target) ** 2
    def grad(x):
        return _smooth_grad(x, beta) + 2.0 * pen * (x.sum() - target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9, 'maxls': 50})
        return _project(res.x)
    except Exception:
        return _project(h0.copy())

def _slsqp_phase(h0, beta, maxiter=500):
    n = len(h0)
    target = n / 2.0
    def obj(x):
        return _smooth_val(x, beta)
    def Objective_jac(x):
        return _smooth_grad(x, beta)
    def con_fun(x):
        return x.sum() - target
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=Objective_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return _project(res.x)
    except Exception:
        return _project(h0.copy())

def _active_sg_step(h, top_k=6, step_size=0.008):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for i in idx:
        d = (n - 1) - i
        lo1 = max(0, -d)
        hi1 = min(n, n - d)
        if lo1 < hi1:
            grad[lo1:hi1] += (1.0 - h[lo1 + d:hi1 + d])
        lo2 = max(0, d)
        hi2 = min(n, d + n)
        if lo2 < hi2:
            grad[lo2:hi2] -= h[lo2 - d:hi2 - d]
    grad *= dx / top_k
    return _project(h - step_size * grad)

def _coord_descent(h, t_budget=30, rng=None):
    n = len(h)
    if rng is None:
        rng = np.random.RandomState(0)
    t0 = time.time()
    best_c = compute_c5(h)
    h = h.copy()
    while time.time() - t0 < t_budget:
        improved = False
        for j in rng.permutation(n):
            if time.time() - t0 > t_budget:
                break
            base = h[j]
            for delta in [0.005, -0.005, 0.01, -0.01, 0.002, -0.002, 0.02, -0.02]:
                h_try = h.copy()
                h_try[j] += delta
                if h_try[j] < 0 or h_try[j] > 1:
                    continue
                k1 = (j + 1) % n
                k2 = (j + n - 1) % n
                h_try[k1] -= delta * 0.5
                h_try[k2] -= delta * 0.5
                if h_try[k1] < 0 or h_try[k1] > 1 or h_try[k2] < 0 or h_try[k2] > 1:
                    continue
                h_try = _project(h_try)
                c_try = compute_c5(h_try)
                if c_try < best_c - 1e-13:
                    h = h_try
                    best_c = c_try
                    improved = True
                    break
            if time.time() - t0 > t_budget:
                break
        if not improved:
            for j in rng.permutation(n):
                if time.time() - t0 > t_budget:
                    break
                for delta in [0.001, -0.001, 0.002, -0.002]:
                    h_try = h.copy()
                    h_try[j] += delta
                    if h_try[j] < 0 or h_try[j] > 1:
                        continue
                    k1 = (j + 1) % n
                    h_try[k1] -= delta
                    if h_try[k1] < 0 or h_try[k1] > 1:
                        continue
                    h_try = _project(h_try)
                    c_try = compute_c5(h_try)
                    if c_try < best_c - 1e-13:
                        h = h_try
                        best_c = c_try
                        improved = True
                        break
            if not improved:
                break
    return h, best_c

def _sim_anneal(h, t_budget=30, rng=None):
    n = len(h)
    if rng is None:
        rng = np.random.RandomState(1)
    t0 = time.time()
    best_c = compute_c5(h)
    h_cur = h.copy()
    best_h = h.copy()
    T = 0.005
    while time.time() - t0 < t_budget:
        T *= 0.995
        if T < 1e-6:
            break
        j = rng.randint(n)
        delta = rng.choice([0.02, -0.02, 0.05, -0.05, 0.01, -0.01])
        h_try = h_cur.copy()
        h_try[j] += delta
        if h_try[j] < 0 or h_try[j] > 1:
            continue
        k = (j + rng.randint(1, n)) % n
        h_try[k] -= delta
        if h_try[k] < 0 or h_try[k] > 1:
            continue
        h_try = _project(h_try)
        c_try = compute_c5(h_try)
        c_cur = compute_c5(h_cur)
        if c_try < c_cur or rng.rand() < np.exp(-(c_try - c_cur) / T):
            h_cur = h_try
            if c_try < best_c - 1e-14:
                best_c = c_try
                best_h = h_try.copy()
    return best_h, best_c

def _nm_polish(h, t_budget=50, rng=None):
    n = len(h)
    t0 = time.time()
    maxiter = min(3000, max(200, int(50 * (t_budget - (time.time() - t0)))))
    def nm_obj(x):
        return compute_c5(_project(x))
    try:
        simplex = [h.copy()]
        for j in range(min(n, 20)):
            pt = h.copy()
            pt[j] += 0.015
            pt[(j + 1) % n] -= 0.015
            simplex.append(_project(pt))
        res = minimize(nm_obj, h.copy(), method='Nelder-Mead',
                      options={'maxiter': maxiter, 'xatol': 1e-11, 'fatol': 1e-14,
                               'initial_simplex': simplex[:n+1]})
        h_nm = _project(res.x)
        return h_nm, compute_c5(h_nm)
    except Exception:
        pass
    try:
        res = minimize(nm_obj, h.copy(), method='Nelder-Mead',
                      options={'maxiter': maxiter, 'xatol': 1e-11, 'fatol': 1e-14})
        h_nm = _project(res.x)
        return h_nm, compute_c5(h_nm)
    except Exception:
        return h, compute_c5(h)

def _optimize_round(h0, t_budget=200, t_start=None, rng=None):
    if t_start is None:
        t_start = time.time()
    if rng is None:
        rng = np.random.RandomState(42)
    h = _project(h0.copy().astype(float))
    best = compute_c5(h)
    hb = h.copy()

    # Phase A: L-BFGS-B with increasing beta
    for beta in [8.0, 20.0, 50.0, 120.0, 300.0]:
        if time.time() - t_start > t_budget - 80:
            break
        h = _lbfgsb_phase(h, beta, maxiter=2000)
        c = compute_c5(h)
        if c < best:
            best = c
            hb = h.copy()

    # Phase B: SLSQP refinement
    if time.time() - t_start < t_budget - 60:
        for beta in [100.0, 500.0]:
            if time.time() - t_start > t_budget - 50:
                break
            h = _slsqp_phase(h, beta, maxiter=400)
            c = compute_c5(h)
            if c < best:
                best = c
                hb = h.copy()

    # Phase C: Active-set subgradient
    if time.time() - t_start < t_budget - 40:
        h_sg = h.copy()
        lr = 0.012
        for _ in range(60):
            if time.time() - t_start > t_budget - 35:
                break
            h_sg = _active_sg_step(h_sg, top_k=6, step_size=lr)
            c = compute_c5(h_sg)
            if c < best:
                best = c
                hb = h_sg.copy()
            lr *= 0.97

    # Phase D: NM polish
    if time.time() - t_start < t_budget - 10:
        nm_bud = min(40, t_budget - (time.time() - t_start) - 5)
        h_nm, c_nm = _nm_polish(hb, t_budget=nm_bud, rng=rng)
        if c_nm < best:
            best = c_nm
            hb = h_nm

    return hb, best

def _structured_start(n, rng, typ):
    h = np.zeros(n)
    if typ == 0:
        h[:n // 2] = 1.0
    elif typ == 1:
        h[rng.choice(n, n // 2, replace=False)] = 1.0
    elif typ == 2:
        s1, s2 = n // 4, n // 2
        h[:s1] = 1.0
        h[s2:s2 + s1] = 1.0
    elif typ == 3:
        for i in range(n):
            if i % 3 != 2:
                h[i] = 1.0
    elif typ == 4:
        for i in range(n):
            if (i % 5) < 2 or (i % 5) == 4:
                h[i] = 1.0
    else:
        third = n // 3
        h[:third] = 1.0
        h[2 * third: 2 * third + third] = 1.0
    h = _project(h + 0.15 * rng.randn(n))
    return h
