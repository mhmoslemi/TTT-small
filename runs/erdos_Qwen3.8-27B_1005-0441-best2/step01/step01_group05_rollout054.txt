    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project_hs(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _smooth_obj_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = np.clip((corr - m) / max(t, 1e-12), -80, 80)
    e = np.exp(s)
    se = np.sum(e)
    w = e / se
    obj = (m + max(t, 1e-12) * np.log(se)) * dx
    c1 = np.convolve(1.0 - h, w, 'full')[:n]
    c2 = np.correlate(h, w, 'full')[:n]
    grad = dx * (c1 - c2)
    return obj, grad

def _pgd_phase(h, target, t, lr, n_iter, deadline):
    h = h.copy()
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        obj, grad = _smooth_obj_grad(h, t)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = lr / (gnorm + 1e-10)
        accepted = False
        for alpha in [step, step * 0.5, step * 0.25, step * 0.1, step * 0.05]:
            h_try = _project_hs(h - alpha * grad, target)
            obj_try, _ = _smooth_obj_grad(h_try, t)
            if obj_try < obj - 1e-15:
                h = h_try
                accepted = True
                break
        if not accepted:
            break
    return h

def _pgd_schedule(h, target, temps, lr, iters, deadline):
    best = h.copy()
    best_val = compute_c5(h)
    for t_i in range(len(temps)):
        if time.time() > deadline:
            break
        t = temps[t_i]
        cur_lr = lr * (0.8 ** t_i)
        h = _pgd_phase(h, target, t, cur_lr, iters, deadline)
        val = compute_c5(h)
        if val < best_val:
            best = h.copy()
            best_val = val
    return best, best_val

def _perturb(h, scale, rng):
    h_p = h + scale * rng.standard_normal(len(h))
    return _project_hs(h_p, float(len(h)) / 2.0)

def _binary_init(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    h = h + 0.01 * rng.standard_normal(n)
    return _project_hs(h, float(n) / 2.0)

def _two_block_init(n, gap_frac, rng):
    h = np.zeros(n)
    n_ones = n // 2
    half = n_ones // 2
    gap = int(n * gap_frac)
    p1 = rng.randint(0, max(1, n - 2 * half - gap))
    for i in range(half):
        h[(p1 + i) % n] = 1.0
        h[(p1 + half + gap + i) % n] = 1.0
    h = h + 0.02 * rng.standard_normal(n)
    return _project_hs(h, float(n) / 2.0)

def _upsample(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old, dtype=float) * 2.0 / n_old
    x_new = np.arange(n_new, dtype=float) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project_hs(h_new, float(n_new) / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(160, 0.5)

    n = len(h0)
    target = float(n) / 2.0
    h0 = _project_hs(h0, target)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Coarse projected gradient from initial
    temps1 = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    dl1 = min(t0 + 250, deadline)
    h_opt, c5_opt = _pgd_schedule(h0, target, temps1, 1.5, 400, dl1)
    if c5_opt < best_c5:
        best_h, best_c5 = h_opt, c5_opt

    # Phase 2: Fine tune
    temps2 = [0.0001, 0.00005, 0.00002]
    dl2 = min(t0 + 400, deadline)
    h_opt, c5_opt = _pgd_schedule(best_h, target, temps2, 0.8, 500, dl2)
    if c5_opt < best_c5:
        best_h, best_c5 = h_opt, c5_opt

    # Phase 3: Perturbations and re-optimize
    scales = [0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.015, 0.025, 0.04, 0.06]
    for i in range(len(scales)):
        if time.time() > deadline - 30:
            break
        h_pert = _perturb(best_h, scales[i], rng)
        temps_p = [0.005, 0.002, 0.0005, 0.0001]
        dl_p = min(time.time() + 50, deadline - 10)
        h_opt, c5_opt = _pgd_schedule(h_pert, target, temps_p, 1.0, 300, dl_p)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 4: Alternative initializations at same n
    alts = [_binary_init(n, rng), _two_block_init(n, 0.25, rng), _two_block_init(n, 0.4, rng)]
    for h_alt in alts:
        if time.time() > deadline - 30:
            break
        temps_a = [0.05, 0.01, 0.001, 0.0001]
        dl_a = min(time.time() + 60, deadline - 10)
        h_opt, c5_opt = _pgd_schedule(h_alt, target, temps_a, 1.2, 300, dl_a)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 5: Try nearby resolutions
    for n_try in [n - 40, n + 40]:
        if n_try < 80 or time.time() > deadline - 40:
            continue
        h_try = _upsample(best_h, n_try)
        tgt_try = float(n_try) / 2.0
        temps_r = [0.02, 0.005, 0.001, 0.0002]
        dl_r = min(time.time() + 70, deadline - 10)
        h_opt, c5_opt = _pgd_schedule(h_try, tgt_try, temps_r, 1.0, 300, dl_r)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 6: Final ultra-fine polish
    if time.time() < deadline - 5:
        n_f = len(best_h)
        tgt_f = float(n_f) / 2.0
        temps_f = [0.00005, 0.00001]
        h_opt, c5_opt = _pgd_schedule(best_h, tgt_f, temps_f, 0.3, 600, deadline)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Ensure final feasibility
    n_final = len(best_h)
    best_h = _project_hs(best_h, float(n_final) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
