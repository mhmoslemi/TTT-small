            child_vals[i] = compute_c5(children[i])

        # Combine and select
        combined = np.empty((pop_size + lam, n))
        combined_vals = np.empty(pop_size + lam)
        combined[:pop_size] = pop
        combined_vals[:pop_size] = pop_vals
        combined[pop_size:] = children
        combined_vals[pop_size:] = child_vals

        sel_order = np.argsort(combined_vals)
        for i in range(pop_size):
            r = sel_order[i]
            pop[i] = combined[r]
            pop_vals[i] = combined_vals[r]

        # Sigma adaptation: (1/5)-rule
        # Compare best offspring vs best parent
        best_off = child_vals.min()
        best_par = pop_vals.min()
        if best_off < best_par:
            sigmas *= 1.05
        else:
            sigmas *= 0.92
        # Per-individual adaptation
        avg_improve = (pop_vals[order[:pop_size//2]] - pop_vals.min()).mean()
        sigmas += 0.01 * (avg_improve > 0) * rng.standard_normal(pop_size)
        sigmas = np.clip(sigmas, 0.003, 0.25)

    return best_h, best_c5, pop, pop_vals, sigmas

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_deadline = t0 + budget_s

    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(160, 0.5)
    n = len(best_h)
    best_c5 = compute_c5(best_h)
    global _g_n
    _g_n = n

    # Phase 1: SLSQP polish (45s)
    t_p1 = 45
    if t_deadline - time.time() > t_p1 + 100:
        h1, v1 = _slsqp_polish(best_h.copy(), [0.02, 0.008, 0.003], max_iter_each=200)
        if v1 < best_c5:
            best_h, best_c5 = h1, v1

    # Phase 2: ES (bulk)
    t_p2_end = t_deadline - 120
    es_time = max(200, t_p2_end - time.time())
    if es_time > 60:
        pop_size = 50
        pop = np.zeros((pop_size, n))
        pop[0] = best_h
        rng = np.random.RandomState(seed + 7)
        for i in range(1, pop_size):
            mode = i % 4
            if mode == 0:
                pop[i] = best_h + rng.uniform(-0.04, 0.04, n)
            elif mode == 1:
                pop[i] = best_h + 0.08 * rng.randn(n)
            elif mode == 2:
                # Two-block structure
                pop[i] = np.zeros(n)
                half = n // 4
                offset = rng.randint(0, n // 2)
                pop[i][offset:offset+half] = 1.0
                off2 = (offset + half + rng.randint(n//8, n//4)) % n
                for j in range(half):
                    pop[i][(off2 + j) % n] = 1.0
            else:
                # Binary random
                pop[i] = (rng.permutation(n) < n // 2).astype(float)
            pop[i] = project_feasible(pop[i], n)
        pop_vals = np.array([compute_c5(pop[i]) for i in range(pop_size)])
        sigmas = np.full(pop_size, 0.06)

        best_h, best_c5, pop, pop_vals, sigmas = _run_es(
            pop, pop_vals, sigmas, n, seed + 100, es_time)

    # Phase 3: SLSQP polish of ES best (80s)
    t_p3_end = t_deadline - 40
    if time.time() < t_p3_end:
        temps3 = [0.01, 0.005, 0.002, 0.001, 0.0005]
        h3, v3 = _slsqp_polish(best_h.copy(), temps3, max_iter_each=300)
        if v3 < best_c5:
            best_h, best_c5 = h3, v3

    # Phase 4: Final micro-polish
    if time.time() < t_deadline - 5:
        temps4 = [2e-4, 1e-4, 5e-5]
        h4, v4 = _slsqp_polish(best_h.copy(), temps4, max_iter_each=150)
        if v4 < best_c5:
            best_h, best_c5 = h4, v4

    # Final feasibility
    best_h = project_feasible(best_h, n)
    # Ensure exact sum
    s = np.sum(best_h)
    target = n / 2.0
    if abs(s - target) > 1e-9:
        best_h = np.clip(best_h + (target - s) / n, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
