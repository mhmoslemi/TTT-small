from scipy.optimize import linprog, minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _compute_grads(h, n):
    """Compute gradient of each c_m w.r.t. h. Returns array of shape (2n-1, n)."""
    G = np.zeros((2*n-1, n))
    c = _corr(h)
    for k in range(2*n-1):
        m = k - (n-1)
        if m >= 0:
            # c_m = sum_{i=m}^{n-1} h[i](1-h[i-m])
            # dc_m/dh[j] = (1-h[j-m]) for j >= m
            #             + (-h[j+m]) for j <= n-1-m (if j+m < n)
            G[k, m:] += (1.0 - h[:n-m])
            if n-1-m >= 0:
                G[k, :n-1-m+1] -= h[m:]
        else:
            p = -m
            # c_m = sum_{i=0}^{n-1-p} h[i](1-h[i+p])

            #             + (-h[j-p]) for j >= p
            G[k, :n-p] += (1.0 - h[p:])
            G[k, p:] -= h[:n-p]
    return G, c

        for _ in range(20):
            ht = _proj((1 - alpha) * hc + alpha * h_lp, n)
            vt = _c5(ht)
            if vt < best_val - 1e-15:
                best_val = vt
                best_h = ht.copy()
                hc = ht
                improved = True
                break
            alpha *= 0.5
        if not improved:
            break
    return best_h, best_val

def _smoobj(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    s = np.exp((c - cmax) / eps)
    return (cmax + eps * np.log(s.sum())) * dx

def _smograd(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    s = np.exp((c - cmax) / eps)
    sw = s / s.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if sw[k] < 1e-14:
            continue
        m = k - (n - 1)
        if m >= 0:
            g[m:] += sw[k] * (1.0 - h[:n - m])
            g[:n - m] -= sw[k] * h[m:]
        else:
            p = -m
            g[:n - p] += sw[k] * (1.0 - h[p:])
            g[p:] -= sw[k] * h[:n - p]
    return g * dx

def _eq_fun(x, n):
    return x.sum() - n * 0.5

def _eq_jac(x, n):
    return np.ones(n)

def _smooth_opt(h, n, t0, budget_s, frac, eps_list):
    best_h = h.copy()
    best_val = _c5(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
    for eps in eps_list:
        if time.time() - t0 > budget_s * frac:
            break
        try:
            res = minimize(_smoobj, best_h.copy(), args=(eps,),
                          jac=_smograd, method='SLSQP',
                          bounds=bnds, constraints=cons,
                          options={'maxiter': 400, 'ftol': 1e-13})
            hr = _proj(res.x, n)
            vr = _c5(hr)
            if vr < best_val:
                best_val = vr
                best_h = hr
        except Exception:
            pass
    return best_h, best_val

def _subgrad(h, n, t0, budget_s, frac, niter=5000):
    best_h = h.copy()
    best_val = _c5(h)
    hc = h.copy()
    lr = 0.008
    for it in range(niter):
        if time.time() - t0 > budget_s * frac:
            break
        c = _corr(hc)
        ks = int(np.argmax(c))
        m = ks - (n - 1)
        g = np.zeros(n)
        if m >= 0:
            g[m:] += (1.0 - hc[:n - m])
            g[:n - m] -= hc[m:]
        else:
            p = -m
            g[:n - p] += (1.0 - hc[p:])
            g[p:] -= hc[:n - p]
        hn = _proj(hc - lr * g, n)
        vn = _c5(hn)
        if vn < best_val - 1e-15:
            best_val = vn
            best_h = hn.copy()
            hc = hn
            lr = min(lr * 1.03, 0.04)
        else:
            lr *= 0.92
            if lr < 1e-9:
                break
    return best_h, best_val

def _topk_smoobj(h, eps, K):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    idx = np.argsort(c)[-K:]
    cmax = c[idx].max()
    s = np.exp((c[idx] - cmax) / eps)
    return (cmax + eps * np.log(s.sum())) * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_val = np.inf

    # Get initial construction
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        h0 = _proj(h0, len(h0))
    except Exception:
        h0 = _proj(0.5 + 0.3 * np.sin(2 * np.pi * np.arange(70) / 70), 70)
    n0 = len(h0)

    # Phase 1: Smooth opt to warm-start
    h1, v1 = _smooth_opt(h0, n0, t0, budget_s, 0.08, [0.04, 0.015])
    best_h, best_val = h1, v1

    # Phase 2: SLP (main improvement engine)
    h2, v2 = _slp(best_h, n0, t0, budget_s, 0.40, 600)
    if v2 < best_val:
        best_val, best_h = v2, h2

    # Phase 3: Another round of SLP from best
    h3, v3 = _slp(best_h, n0, t0, budget_s, 0.55, 400)
    if v3 < best_val:
        best_val, best_h = v3, h3

    # Phase 4: Try alternate starting points / n values
    if time.time() - t0 < budget_s * 0.50:
        for n_try in [n0, n0 + 2, n0 + 4, n0 + 10]:
            if time.time() - t0 > budget_s * 0.70:
                break
            for freq in [1, 2, 3]:
                if time.time() - t0 > budget_s * 0.70:
                    break
                for a in [0.25, 0.38, 0.50]:
                    if time.time() - t0 > budget_s * 0.70:
                        break
                    hs = _proj(0.5 + a * np.sin(2 * np.pi * freq * np.arange(n_try) / n_try), n_try)
                    hs, vs = _smooth_opt(hs, n_try, t0, budget_s, 0.70, [0.02, 0.008])
                    if vs < best_val:
                        best_val, best_h = vs, hs
                    hs, vs = _slp(hs, n_try, t0, budget_s, 0.70, 200)
                    if vs < best_val:
                        best_val, best_h = vs, hs

    # Phase 5: Random restart with SLP
    if time.time() - t0 < budget_s * 0.70:
        for n_try in [n0, n0 + 10]:
            for _ in range(3):
                if time.time() - t0 > budget_s * 0.80:
                    break
                hs = _proj(rng.rand(n_try), n_try)
                hs, vs = _smooth_opt(hs, n_try, t0, budget_s, 0.80, [0.03])
                if vs < best_val:
                    best_val, best_h = vs, hs
                hs, vs = _slp(hs, n_try, t0, budget_s, 0.80, 200)
                if vs < best_val:
                    best_val, best_h = vs, hs

    # Phase 6: Final SLP polish on best
    if time.time() - t0 < budget_s * 0.85:
        best_h, best_val = _slp(best_h, len(best_h), t0, budget_s, 0.92, 500)

    # Phase 7: Subgradient final polish
    if time.time() - t0 < budget_s * 0.92:
        best_h, best_val = _subgrad(best_h, len(best_h), t0, budget_s, 0.98, 8000)

    # Final projection and evaluation
    best_h = _proj(best_h, len(best_h))
    best_val = _c5(best_h)

    if best_h is None:
        best_h = np.full(n0, 0.5)
        best_val = _c5(best_h)

    return (best_h, best_val, len(best_h))
