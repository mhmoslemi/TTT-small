    #       B_k = h[j-k]*bwd_valid + rest

    # So we just need A_k!
    
    # A_k for corr_idx = m:
    # k = m - n + 1
    # fwd: valid if 0 <= j+k < n, i.e., 0 <= j + m - n + 1 < n
    # bwd: valid if 0 <= j-k < n, i.e., 0 <= j - m + n - 1 < n, i.e., n-1-j <= m <= j+n-1... 
    
    # Let me just use the formula:
    # A[m] = (1 - h[j + k]) * I(j+k valid) - h[j-k] * I(j-k valid)
    # where k = m - n + 1
    
    # And B[m] = corr[m] - A[m] * h[j]  (this is exact since corr[m] = A[m]*h[j] + B[m])
    
    k_vals = np.arange(2*n-1) - (n-1)  # k values
    # j+k valid: 0 <= j+k < n
    fwd_mask = (j + k_vals >= 0) & (j + k_vals < n)
    # j-k valid: 0 <= j-k < n
    bwd_mask = (j - k_vals >= 0) & (j - k_vals < n)
    
    A = np.zeros(2*n-1)
    # Forward: A += 1 - h[j+k]
    if np.any(fwd_mask):
        A[fwd_mask] += 1.0 - h[j + k_vals[fwd_mask]]
    # Backward: A -= h[j-k]
    if np.any(bwd_mask):
        A[bwd_mask] -= h[j - k_vals[bwd_mask]]
    
    B = corr - A * old
    
    # Now find min over x in [0,1] of max_k (A_k*x + B_k)
    # Try x=0, x=1, and intersection of active lines
    f0 = np.max(B)  # f(0)
    f1 = np.max(A + B)  # f(1)
    
    # Find which k is active at x=0 and x=1
    k0 = np.argmax(B)
    k1 = np.argmax(A + B)
    
    candidates = [0.0, 1.0]
    if k0 != k1 and A[k1] != A[k0]:
        x_int = (B[k0] - B[k1]) / (A[k1] - A[k0])
        if 0.0 < x_int < 1.0:
            candidates.append(x_int)
    
    # Also try a few more points
    candidates.extend([0.1, 0.3, 0.5, 0.7, 0.9])
    
    best_x = 0.0
    best_val = np.inf
    for x in candidates:
        x = np.clip(x, 0.0, 1.0)
        val = np.max(A * x + B)
        if val < best_val:
            best_val = val
            best_x = x
    
    return best_x

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_step(h, j, corr):
    n = len(h)
    old = h[j]
    k_vals = np.arange(-(n - 1), n)
    fwd_mask = (j + k_vals >= 0) & (j + k_vals < n)
    bwd_mask = (j - k_vals >= 0) & (j - k_vals < n)
    A = np.zeros(2 * n - 1)
    if fwd_mask.any():
        A[fwd_mask] += 1.0 - h[j + k_vals[fwd_mask]]
    if bwd_mask.any():
        A[bwd_mask] -= h[j - k_vals[bwd_mask]]
    B = corr - A * old
    f0 = B.max()
    f1 = (A + B).max()
    best_x = 0.0 if f0 <= f1 else 1.0
    best_f = min(f0, f1)
    k0 = int(np.argmax(B))
    k1 = int(np.argmax(A + B))
    if k0 != k1 and abs(A[k1] - A[k0]) > 1e-12:
        x_int = (B[k0] - B[k1]) / (A[k1] - A[k0])
        if 0.0 <= x_int <= 1.0:
            f_int = np.max(A * x_int + B)
            if f_int < best_f:
                best_f = f_int
                best_x = float(x_int)
    for x in (0.25, 0.5, 0.75):
        f_x = np.max(A * x + B)
        if f_x < best_f:

            best_x = x
    delta = best_x - old
    if abs(delta) > 1e-14:
        h[j] = best_x
        corr += A * delta
    return best_x

def binary_round(h, n):
    h_bin = (h > 0.5).astype(float)
    diff = n / 2.0 - h_bin.sum()
    if diff > 0:
        zeros = np.where(h_bin == 0)[0]
        order = zeros[np.argsort(h[zeros])[::-1]]
        for i in range(min(int(diff), len(order))):

        order = ones[np.argsort(h[ones])]

            h_bin[order[i]] = 0.0
    return h_bin
