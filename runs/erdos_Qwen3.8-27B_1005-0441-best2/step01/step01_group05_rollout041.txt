        scale = 0.015 + 0.05 * (restart % 6) / 5.0
        h_pert = _perturb(best_h, scale, seed + 100 + restart)
        temps_r = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0003]
        max_t = min(35, _rem() - 25)
        if max_t < 15:
            break
        h_opt, c5_opt = _multi_temp_opt(h_pert, n_cur, temps_r,
                                         max_time=max_t, iters_per_temp=1)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
        restart += 1

    # Phase 5: Subgradient polish for fine-tuning
    if _rem() > 30:
        h_pol, c5_pol = _subgradient_polish(best_h.copy(), len(best_h),
                                             max_time=_rem() - 10, t0=t0)
        if c5_pol < best_c5:
            best_h = h_pol
            best_c5 = c5_pol

    # Phase 6: Final SLSQP polish at very low temp
    if _rem() > 20:
        temps_f = [0.0003, 0.0001, 0.00003]
        h_opt, c5_opt = _multi_temp_opt(best_h.copy(), len(best_h), temps_f,
                                         max_time=_rem() - 5, iters_per_temp=2)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    n_final = len(best_h)
    best_h = _normalize(best_h, n_final)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_final
