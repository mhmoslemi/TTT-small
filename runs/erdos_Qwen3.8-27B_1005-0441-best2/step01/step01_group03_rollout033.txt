```python
def to_binary(h, n):
    """Convert continuous h to binary with exactly n/2 ones."""
    # Round, then fix count
    b = (h >= 0.5).astype(float)
    cnt = int(b.sum())
    if cnt > n // 2:
        # Remove excess ones (those closest to 0.5)
        ones = np.where(b == 1)[0]
        order = np.argsort(h[ones])  # sort by value (closest to 0.5 first)
        for i in range(cnt - n//2):
            b[ones[order[i]]] = 0

        zeros = np.where(b == 0)[0]
        order = np.argsort(-h[zeros])  # sort by value descending (closest to 0.5 first)
        for i in range(n//2 - cnt):
```

```python
import numpy as np
```

```python
def to_binary(h, n):
    """Round h to binary with exactly n//2 ones."""
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:n//2]] = 1.0
    return b

def c5_val(b, n):
    dx = 2.0 / n
    corr = np.correlate(b, 1.0 - b, mode='full')
    return float(corr.max()) * dx

def local_search(b, n, deadline):
    """Hill climbing: find best improving swap at each step."""
    while time.time() < deadline:
        improved = False
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        curr_corr = np.correlate(b, 1.0 - b, mode='full')
        curr_max = float(curr_corr.max())
        best_val = curr_max
        best_swap = None
        
        for i in ones:
            for j in zeros:
                b[i], b[j] = 0, 1
                nm = float(np.correlate(b, 1.0 - b, mode='full').max())
                if nm < best_val:
                    best_val = nm
                    best_swap = (i, j)
                b[i], b[j] = 1, 0
        
        if best_swap is not None:
            b[best_swap[0]], b[best_swap[1]] = 0, 1
            improved = True
        
        if not improved:
            break
    
    return b, c5_val(b, n)

def anneal(b, n, deadline, rng):
    """Simulated annealing on binary vectors."""
    curr = b.copy()
    curr_c5 = c5_val(curr, n)
    best = curr.copy()
    best_c5 = curr_c5
    T = 0.003
    
    while time.time() < deadline:

        zeros = np.where(curr == 0)[0]

        j = zeros[rng.randint(len(zeros))]
```

```python
def compute_deltas(b, n):
    """Compute delta_i[p] for all positions i and lags p."""
    # p ranges from -(n-1) to (n-1), indexed 0 to 2n-2
    # p = m - (n-1) where m is the correlate index
    n_lags = 2*n - 1
    deltas = np.zeros((n, n_lags))
    for i in range(n):
        for p in range(-(n-1), n):
            # Contribution to R[n-1+p] from position i:
            # As b[t+p] where t = i-p: valid if 0 ≤ i-p < n
            #   value: b[i] * (1-b[i-p])
            # As (1-b[t]) where t = i: valid if 0 ≤ i+p < n  
            #   value: b[i+p] * (1-b[i])
            t1 = i - p
            t2 = i + p
            val = 0
            if b[i] == 1:
                if 0 <= t1 < n:
                    val += (1 - b[t1])  # this is the current contribution as first arg
                if 0 <= t2 < n:
                    val += 0  # (1-b[i]) = 0 when b[i]=1
            # Delta when b[i] goes 1→0:
            # First arg: was b[i]*(1-b[t1]) = 1*(1-b[t1]), becomes 0. Delta: -(1-b[t1])
            # Second arg: was b[t2]*(1-b[i]) = b[t2]*0 = 0, becomes b[t2]*1. Delta: +b[t2]
            d = 0
            if b[i] == 1:
                if 0 <= t1 < n:
                    d -= (1 - b[t1])
                if 0 <= t2 < n:
                    d += b[t2]
            elif b[i] == 0:
                # Delta when b[i] goes 0→1:
                # First arg: was 0, becomes (1-b[t1]). Delta: +(1-b[t1])
                # Second arg: was b[t2]*1, becomes 0. Delta: -b[t2]
                if 0 <= t1 < n:
                    d += (1 - b[t1])
                if 0 <= t2 < n:
                    d -= b[t2]
            deltas[i, n-1+p] = d
    return deltas
```
