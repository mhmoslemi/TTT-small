                    break
        if not improved:
            break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None

    if h_init is not None:
        h = proj_h(h_init.copy())
        n = len(h)
    else:
        n = 70
        h = np.zeros(n)
        h[:n // 2] = 1.0
        h = proj_h(h)

    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Build binary version
    h_bin = (h >= 0.5).astype(float)
    s = int(h_bin.sum())
    while s > n // 2:
        oi = np.where(h_bin == 1.0)[0]
        h_bin[rng.choice(oi)] = 0.0
        s -= 1
    while s < n // 2:
        zi = np.where(h_bin == 0.0)[0]
        h_bin[rng.choice(zi)] = 1.0
        s += 1

    # Phase 1: Binary local search (multiple restarts)
    if tl() > 250:
        h_b, c_b = binary_local_search(h_bin, max_passes=4)
        if c_b < best_c5:
            best_c5 = c_b
            best_h = proj_h(h_b)
        for rep in range(4):
            if tl() < 50:
                break
            h_b2 = h_bin.copy()
            for _ in range(3 + rep):
                oi = np.where(h_b2 == 1.0)[0]
                zi = np.where(h_b2 == 0.0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    h_b2[rng.choice(oi)] = 0.0
                    h_b2[rng.choice(zi)] = 1.0
            h_b3, c_b3 = binary_local_search(h_b2, max_passes=3)
            if c_b3 < best_c5:
                best_c5 = c_b3
                best_h = proj_h(h_b3)

    # Phase 2: Multi-scale SLSQP from continuous best
    if tl() > 180:
        h_work = best_h.copy()
        for i in range(6):
            if tl() < 30:
                break
            beta = 4.0 * (i + 1) ** 1.6
            h_work = slsqp_opt(h_work, beta, maxit=130)
            c = compute_c5(h_work)
            if c < best_c5:
                best_c5 = c
                best_h = h_work.copy()

    # Phase 3: PGD with increasing sharpness
    if tl() > 80:
        for beta in [25, 50, 100, 200, 400]:
            if tl() < 25:
                break
            h_pg = pgd_opt(best_h, beta, iters=80, lr_init=0.02)
            c = compute_c5(h_pg)
            if c < best_c5:
                best_c5 = c
                best_h = h_pg.copy()

    # Phase 4: Population-based search
    if tl() > 120:
        pop_size = 7
        pop = [best_h.copy() for _ in range(pop_size)]
        for i in range(1, pop_size):
            s_mod = i % 4
            if s_mod == 0:
                pop[i] = proj_h(best_h + 0.12 * rng.randn(n))
            elif s_mod == 1:
                p = np.zeros(n)
                p[rng.choice(n, n // 2, replace=False)] = 1.0
                pop[i] = proj_h(p)
            elif s_mod == 2:
                pop[i] = proj_h(best_h + 0.05 * rng.randn(n))
            else:
                p = np.zeros(n)
                idx = rng.permutation(n)
                p[idx[:n // 2]] = 1.0
                pop[i] = proj_h(p)

        gen = 0
        while tl() > 60:
            gen += 1
            new_pop = []
            for p in pop:
                beta = 15.0 + 20.0 * gen
                hp = pgd_opt(p, beta, iters=35, lr_init=0.012)
                new_pop.append(hp)
            for i in range(pop_size):
                r = rng.random()
                if r < 0.25:
                    new_pop[i] = proj_h(new_pop[i] + 0.10 * rng.randn(n))
                elif r < 0.45:
                    j = rng.randint(pop_size)
                    new_pop[i] = proj_h(0.5 * new_pop[i] + 0.5 * new_pop[j])
            for p in new_pop:
                c = compute_c5(p)
                if c < best_c5:
                    best_c5 = c
                    best_h = p.copy()
            scores = np.array([compute_c5(p) for p in new_pop])
            order = np.argsort(scores)
            pop = [new_pop[i].copy() for i in order[:max(3, pop_size // 2)]]
            pop.append(best_h.copy())
            while len(pop) < pop_size:
                pop.append(proj_h(best_h + 0.07 * rng.randn(n)))

    # Phase 5: SLSQP polish with high beta
    if tl() > 50:
        h_pol = slsqp_opt(best_h, 400, maxit=150)
        c = compute_c5(h_pol)
        if c < best_c5:
            best_c5 = c
            best_h = h_pol.copy()

    # Phase 6: Fine coordinate descent
    if tl() > 25:
        h_fin = best_h.copy()
        fc5 = best_c5
        for outer in range(30):
            if tl() < 8:
                break
            improved = False
            for j in range(n):
                for delta in [0.001, -0.001, 0.002, -0.002, 0.0005, -0.0005,
                              0.003, -0.003, 0.005, -0.005]:
                    ht = h_fin.copy()
                    ht[j] += delta
                    if ht[j] < 0.0 or ht[j] > 1.0:
                        continue
                    diff = -delta
                    for sp in [2, 3, 5]:
                        idxs = [(j + k) % n for k in range(1, sp + 1)]
                        ht2 = ht.copy()
                        ok = True
                        for k in idxs:
                            ht2[k] += diff / sp
                            if ht2[k] < 0.0 or ht2[k] > 1.0:
                                ok = False
                                break
                        if not ok:
                            continue
                        ht2 = proj_h(ht2)
                        c = compute_c5(ht2)
                        if c < fc5 - 1e-14:
                            h_fin = ht2
                            fc5 = c
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break
        if fc5 < best_c5:
            best_c5 = fc5
            best_h = h_fin.copy()

    best_h = proj_h(best_h)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, n)
