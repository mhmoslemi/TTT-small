    mc = corr.max()
    shifted = np.clip(tau * (corr - mc), -500, 500)

    weights = exps / exps.sum()

    kick_limit = 0
    while time.time() < dl - 0.5:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = corr.max()
        sidx = np.argsort(corr)[::-1]
        tl = sidx[:M]
        tv = corr[tl].copy()
        rm = float(corr[sidx[M]]) if M < len(corr) else 0.0
        best_nm = max_c + 1.0
        bi, bj = -1, -1
        for i in ones:
            ii = int(i)
            for j in zeros:
                jj = int(j)
                nm = rm
                for mi in range(M):
                    L = int(tl[mi] - (n - 1))
                    di = 0.0
                    dj = 0.0
                    if L >= 0:
                        if ii < n - L:
                            di -= (1.0 - h[ii + L])
                        if L <= ii:
                            di += h[ii - L]
                        if jj < n - L:
                            dj += (1.0 - h[jj + L])
                        if L <= jj:
                            dj -= h[jj - L]
                    else:
                        Lp = -L
                        if Lp <= ii:
                            di -= (1.0 - h[ii - Lp])
                        if ii < n - Lp:
                            di += h[ii + Lp]
                        if Lp <= jj:
                            dj += (1.0 - h[jj - Lp])
                        if jj < n - Lp:
                            dj -= h[jj + Lp]
                    nv = tv[mi] + di + dj
                    if nv > nm:
                        nm = nv
                if nm < best_nm:
                    best_nm = nm
                    bi, bj = ii, jj
        if bi < 0:
            kick_limit += 1
            if kick_limit > 30:
                break
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            h[i], h[j] = 0.0, 1.0
            cf = compute_c5(h)
            if cf > best_c5:
                h[i], h[j] = 1.0, 0.0
            continue
        else:
            kick_limit = 0
            h[bi], h[bj] = 0.0, 1.0
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5

def round_bin(h_frac, ts):
    order = np.argsort(-h_frac)
    hb = np.zeros(len(h_frac))
    hb[order[:ts]] = 1.0
    return hb

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.asarray(initial_h_values, dtype=float)
    else:
        h_init = np.full(70, 0.5)
    n = len(h_init)
    ts = n / 2.0
    ts_i = n // 2
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: Fractional optimization
    tr = dl - time.time() - 10
    fb = min(tr * 0.35, 350)
    hf = h_init
    cf = best_c5
    if fb > 15:
        hf, cf = frac_opt(h_init, fb, seed)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf.copy()

    # Phase 2: Binary search from rounded fractional
    tr = dl - time.time() - 10
    bb = min(tr * 0.3, 250)
    hb1 = round_bin(hf, ts_i)
    hbr, bc1 = bin_search(hb1, min(bb, 200), seed + 1)
    if bc1 < best_c5:
        best_c5 = bc1
        best_h = hbr.copy()

    # Phase 2b: Binary from initial rounded (different start)
    tr = dl - time.time() - 10
    if tr > 30:
        hb2 = round_bin(h_init, ts_i)
        hbr2, bc2 = bin_search(hb2, min(tr * 0.2, 80), seed + 3)
        if bc2 < best_c5:
            best_c5 = bc2
            best_h = hbr2.copy()

    # Phase 3: Re-optimize fractionally from best, then re-bin
    tr = dl - time.time() - 10
    if tr > 25:
        hf3, cf3 = frac_opt(best_h, min(tr * 0.3, 120), seed + 10)
        if cf3 < best_c5:
            best_c5 = cf3
            best_h = hf3.copy()
        tr2 = dl - time.time() - 5
        if tr2 > 20:
            hb3 = round_bin(hf3, ts_i)
            hbr3, bc3 = bin_search(hb3, min(tr2 * 0.5, 80), seed + 12)
            if bc3 < best_c5:
                best_c5 = bc3
                best_h = hbr3.copy()

    # Phase 4: Try other n values
    for nt in [56, 64, 72, 80, 48]:
        tr = dl - time.time() - 5
        if tr < 40:
            break
        if nt == n:
            continue
        ht = np.full(nt, 0.5)
        bf = min(tr * 0.3, 45)
        bb2 = min(tr * 0.3, 45)
        hft, cft = frac_opt(ht, bf, seed + nt)
        hbt = round_bin(hft, nt // 2)
        hbrt, bct = bin_search(hbt, bb2, seed + nt + 2)
        if bct < best_c5:
            best_c5 = bct
            best_h = hbrt.copy()

    # Phase 5: Final binary polish on best solution
    tr = dl - time.time() - 1
    if tr > 5:
        nb = len(best_h)
        hb4 = (best_h >= 0.5).astype(float)
        s4 = int(hb4.sum())
        ts4 = nb // 2
        if s4 > ts4:
            oi = np.argsort(hb4)
            for idx_ in range(s4 - ts4):
                hb4[int(oi[idx_])] = 0.0
        elif s4 < ts4:
            oi = np.argsort(-best_h)
            for idx_ in range(ts4 - s4):
                hb4[int(oi[idx_])] = 1.0
        hbr4, bc4 = bin_search(hb4, min(tr, 40), seed + 99)
        if bc4 < best_c5:
            best_c5 = bc4
            best_h = hbr4.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
