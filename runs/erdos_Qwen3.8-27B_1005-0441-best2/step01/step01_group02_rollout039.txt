import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
        h = np.clip(h + (target - s) / n, 0.0, 1.0)
    return h

def _corr_lse(h, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    val = mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta
    w = np.exp(beta * (corr - mc))
    w /= w.sum()
    return val, w, corr

def _lse_obj(h, beta, n):
    val, _, _ = _corr_lse(h, beta)
    return float(val * 2.0 / n)

def _lse_grad(h, beta, n):
    _, w, _ = _corr_lse(h, beta)
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        s = i - (n - 1)
        if s >= 0:
            lo = s
            hi = n
        else:
            lo = 0
            hi = n + s
        for m in range(lo, hi):
            grad[m] += wi * (1.0 - h[m - s])
        lo2 = 0
        hi2 = n if s >= 0 else n - s
        if s >= 0:
            lo2 = 0
            hi2 = n - s
        else:
            lo2 = -s
            hi2 = n
        for m in range(lo2, hi2):
            grad[m] -= wi * h[m + s]
    return grad * dx

def _fista_step(h, beta, n, step, t_param, h_prev):

    return h_new, h_momentum, t_new

def _slsqp_con_fun(x, n):

def _slsqp_con_jac(x, n):
    return np.ones(len(x))

def _slsqp_active_obj(x, n, act_idx):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    return float(np.sum(corr[act_idx]) * dx)

def _slsqp_active_jac(x, n, act_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in act_idx:
        s = i - (n - 1)
        if s >= 0:
            for m in range(s, n):
                grad[m] += (1.0 - x[m - s])
            for m in range(0, n - s):
                grad[m] -= x[m + s]
        else:
            t = -s
            for m in range(0, n - t):
                grad[m] += (1.0 - x[m + t])
            for m in range(t, n):
                grad[m] -= x[m - t]
    return grad * dx
