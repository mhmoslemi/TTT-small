def compute_overlap_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps
    # overlap[k] = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k]) * dx
    overlaps = np.zeros(n)
    for k in range(n):
        overlaps[k] = np.sum(h[:n-k] * (1 - h[k:])) * dx
    
    # Log-sum-exp for numerical stability
    m = np.max(overlaps)
    log_sum = m + tau * np.log(np.sum(np.exp((overlaps - m) / tau)))
    
    # Weights
    w = np.exp((overlaps - m) / tau)
    w /= np.sum(w)
    
    # Gradient: sum_k w_k * grad(overlap[k])
    # grad(overlap[k])[j] = (1-h[j+k])*1_{j<=n-1-k} - h[j-k]*1_{j>=k} (all * dx)
    grad = np.zeros(n)
    for k in range(n):
        # For positions j where 0 <= j <= n-1-k: contrib (1-h[j+k])
        # For positions j where k <= j <= n-1: contrib -h[j-k]
        wk = w[k] * dx
        if wk < 1e-15:

        for j in range(n - k):
            grad[j] += wk * (1 - h[j+k])
        for j in range(k, n):
            grad[j] -= wk * h[j-k]
    
    return log_sum, grad

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _overlap_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.empty(n)
    for k in range(n):
        c[k] = np.dot(h[:n-k], 1.0 - h[k:])
    m = np.max(c)
    exp_c = np.exp((c - m) / tau)
    s = np.sum(exp_c)
    w = exp_c / s
    obj = (m + tau * np.log(s)) * dx
    grad = np.zeros(n)
    for k in range(n):
        wk = w[k]
        if wk < 1e-14:
            continue
        grad[:n-k] += wk * (1.0 - h[k:])
        grad[k:] -= wk * h[:n-k]
    grad *= dx
    return obj, grad

def _enforce(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    s = np.sum(h)
    if s > 1e-12:
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
    return h

def _proj_grad(h, n, seed, max_time, tau0=0.04):
    np.random.seed(seed)
    t0 = time.time()
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    tau = tau0
    lr = 0.05
    stall = 0
    while time.time() - t0 < max_time:
        obj, grad = _overlap_and_grad(h, tau)
        h_try = _enforce(h - lr * grad, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-9:
            h = h_try
            best_h = h.copy()
            best_c5 = c5_try
            lr = min(lr * 1.05, 0.5)
            stall = 0
        else:
            stall += 1
            lr *= 0.7
            if stall > 80:
                tau = max(tau * 0.5, 0.002)
                h = _enforce(h + 0.015 * np.random.randn(n), n)
                lr = 0.03
                stall = 0
        tau = max(tau * 0.9995, 0.003)
    return best_h, best_c5

def _slsqp_refine(h, n, max_time, tau=0.005):
    t0 = time.time()
    bounds = [(0.0, 1.0)] * n
    def _f(hv):
        o, g = _overlap_and_grad(hv, tau)
        return o, g
    cons = ({'type': 'eq', 'fun': lambda hv: np.sum(hv) - n/2.0,
             'jac': lambda hv: np.ones(n)},)
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(3):
        if time.time() - t0 > max_time - 5:
            break
        try:
            res = minimize(_f, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, jac=True,
                           options={'maxiter': 500, 'ftol': 1e-14})
            h_new = _enforce(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-10:
                best_h = h_new.copy()
                best_c5 = c5_new
            h_cur = h_new.copy()
        except Exception:
            break
    return best_h, best_c5

def _upsample(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old, dtype=float) * 2.0 / n_old
    x_new = np.arange(n_new, dtype=float) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    s = np.sum(h_new)
    if s > 1e-12:
        h_new = h_new * (float(n_new) / 2.0) / s
    return np.clip(h_new, 0.0, 1.0)

def _binary_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    n_ones = n // 2
    h[rng.choice(n, n_ones, replace=False)] = 1.0
    return h

def _pulse_init(n, seed, gap_frac=0.3):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    n_ones = n // 2
    half = n_ones // 2
    gap = int(n * gap_frac)
    pos = rng.randint(0, max(1, n - half - gap))
    for i in range(half):
        h[(pos + i) % n] = 1.0
        h[(pos + half + gap + i) % n] = 1.0
    return h

def _perturb(h, scale, seed):
    rng = np.random.RandomState(seed)
    n = len(h)
    h_p = h + scale * rng.randn(n)
    return _enforce(h_p, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def _rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    else:
        n = 160
        h0 = _pulse_init(n, seed, 0.3)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Main projected gradient on initial
    if _rem() > 200:
        t1 = min(180, _rem() - 20)
        h1, c51 = _proj_grad(h0.copy(), n, seed, t1, tau0=0.05)
        if c51 < best_c5:
            best_h, best_c5 = h1, c51

    # Phase 2: SLSQP refine
    if _rem() > 120:
        h2, c52 = _slsqp_refine(best_h.copy(), n, min(100, _rem() - 20), tau=0.008)
        if c52 < best_c5:
            best_h, best_c5 = h2, c52

    # Phase 3: Alternative initializations
    inits = [
        _binary_init(n, seed + 10),
        _pulse_init(n, seed + 20, 0.25),
        _pulse_init(n, seed + 30, 0.4),
        _perturb(best_h, 0.12, seed + 40),
    ]
    for i, hi in enumerate(inits):
        if _rem() < 40:
            break
        t_alloc = min(40, _rem() - 15)
        hf, cf = _proj_grad(hi.copy(), n, seed + 100 + i, t_alloc, tau0=0.06)
        if cf < best_c5:
            best_h, best_c5 = hf, cf
        if _rem() > 30:
            hs, cs = _slsqp_refine(best_h.copy(), n, min(20, _rem() - 10), tau=0.008)
            if cs < best_c5:
                best_h, best_c5 = hs, cs

    # Phase 4: Upsample to 2n and optimize
    n2 = n * 2
    if _rem() > 100 and n2 <= 640:
        h_up = _upsample(best_h, n2)
        hu, cu = _proj_grad(h_up, n2, seed + 500, min(60, _rem() - 30), tau0=0.03)
        if cu < best_c5:
            best_h, best_c5 = hu, cu
        if _rem() > 30:
            hs, cs = _slsqp_refine(best_h.copy(), len(best_h), min(25, _rem() - 10), tau=0.005)
            if cs < best_c5:
                best_h, best_c5 = hs, cs

    # Phase 5: Random restarts at best resolution
    n_cur = len(best_h)
    restart = 0
    while _rem() > 25 and restart < 30:
        scale = 0.02 + 0.05 * (restart % 5) / 4.0
        hp = _perturb(best_h, scale, seed + 2000 + restart)
        mt = min(20, _rem() - 10)
        if mt < 8:
            break
        hr, cr = _proj_grad(hp, n_cur, seed + 2000 + restart, mt, tau0=0.03)
        if cr < best_c5:
            best_h, best_c5 = hr, cr
        restart += 1

    # Phase 6: Final SLSQP polish
    if _rem() > 15:
        hf, cf = _slsqp_refine(best_h.copy(), len(best_h), max(8, _rem() - 3), tau=0.003)
        if cf < best_c5:
            best_h, best_c5 = hf, cf

    # Ensure constraint
    best_h = _enforce(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
