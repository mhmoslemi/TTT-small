    zeros = np.where(h == 0.0)[0]
    best_c5 = compute_c5(h, n)
    best_i, best_j = -1, -1
    for i in ones:
        for j in zeros:
            h[i], h[j] = 0.0, 1.0
            c5 = compute_c5(h, n)
            h[i], h[j] = 1.0, 0.0
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_i, best_j = int(i), int(j)
    return best_i, best_j, best_c5

def local_search(h, n, deadline):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    while time.time() < deadline:
        i, j, new_c5 = best_swap(h, n)
        if i < 0:
            break
        h[i] = 0.0
        h[j] = 1.0
        if new_c5 < best_c5 - 1e-15:
            best_c5 = new_c5
            best_h = h.copy()
    return best_h, best_c5

def perturb(h, n, rng, n_swaps):
    h = h.copy()
    for _ in range(n_swaps):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i != j:
            h[i] = 0.0
            h[j] = 1.0
    return h

def perturb2(h, n, rng):
    """Swap two 1s with two 0s (4-position move)."""
    h = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) < 2 or len(zeros) < 2:
        return h
    i1, i2 = rng.choice(ones, size=2, replace=False)
    j1, j2 = rng.choice(zeros, size=2, replace=False)
    h[i1] = 0.0; h[i2] = 0.0
    h[j1] = 1.0; h[j2] = 1.0
    return h

def continuous_refine(h, n, deadline, rng):
    dx = 2.0 / n
    h = h.astype(float).copy()
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    steps = [0.1, 0.05, 0.02, 0.01, 0.005]
    for step in steps:
        it = 0
        while time.time() < deadline and it < 5000:
            ov = np.correlate(h, 1.0 - h, mode='full') * dx
            k_max = int(np.argmax(ov))
            lag = k_max - (n - 1)
            g = np.zeros(n)
            if lag >= 0:
                for a in range(n):
                    if a + lag < n:
                        g[a] += (1.0 - h[a + lag])
                    if a - lag >= 0:
                        g[a] -= h[a - lag]
            else:
                ll = -lag
                for a in range(ll, n):
                    g[a] += (1.0 - h[a - ll])
                for a in range(n - ll):
                    g[a] -= h[a + ll]
            # Project: h_new = clip(h - step*g/gmax, 0, 1) then fix sum
            gm = np.max(np.abs(g))
            if gm < 1e-12:
                break
            h_new = np.clip(h - step * g / gm, 0.0, 1.0)
            # Fix sum to target
            diff = target - np.sum(h_new)
            if abs(diff) > 1e-10:
                mask = (h_new > 1e-8) & (h_new < 1.0 - 1e-8)
                cnt = float(np.sum(mask))
                if cnt > 0:
                    h_new[mask] = np.clip(h_new[mask] + diff / cnt, 0.0, 1.0)
                    # Fix again
                    diff2 = target - np.sum(h_new)
                    if abs(diff2) > 1e-10:
                        h_new += diff2 / n
                        h_new = np.clip(h_new, 0.0, 1.0)
                    # Final fix
                    diff3 = target - np.sum(h_new)
                    if abs(diff3) > 1e-10:
                        idx1 = np.where(h_new > 0.01)[0]
                        idx0 = np.where(h_new < 0.99)[0]
                        if len(idx1) > 0 and len(idx0) > 0:
                            h_new[idx1] -= diff3 / (2.0 * len(idx1))
                            h_new[idx0] += diff3 / (2.0 * len(idx0))
                            h_new = np.clip(h_new, 0.0, 1.0)
            c5 = compute_c5(h_new, n)
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                best_h = h_new.copy()
                h = h_new
            else:
                it += 1
            it += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = 1.0
    best_n = 0

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except Exception:
        h0 = None
        n0 = 46

    def try_n(n, deadline, start_h=None):
        if start_h is not None:
            h = make_binary(np.clip(start_h, 0, 1), n)
        else:
            h = rng.rand(n)
            h = make_binary(h, n)
        return local_search(h, n, deadline)

    # Phase 1: Binary local search on initial (fast, converges in <1s)
    if h0 is not None:
        h_bin = make_binary(h0, n0)
        bh, bc5 = local_search(h_bin, n0, t0 + 30.0)
        best_h, best_c5, best_n = bh, bc5, n0

    # Phase 2: Perturbation + restart (main exploration, 55% budget)
    deadline2 = t0 + budget_s * 0.55
    rounds = 0
    while time.time() < deadline2:
        n_swaps = rng.randint(2, 7)
        if rng.random() < 0.3:
            h_p = perturb2(best_h, best_n, rng)
        else:
            h_p = perturb(best_h, best_n, rng, n_swaps)
        sub_dl = min(time.time() + 8.0, deadline2)
        bh, bc5 = local_search(h_p, best_n, sub_dl)
        if bc5 < best_c5 - 1e-14:
            best_h, best_c5 = bh, bc5
        rounds += 1

    # Phase 3: Try other n values (25% budget)
    n_candidates = [44, 48, 42, 50, 52, 40, 56, 46 if best_n != 46 else 54]
    n_candidates = [x for x in n_candidates if x != best_n and x >= 30 and x <= 64]
    deadline3 = t0 + budget_s * 0.80
    elapsed_for_n = deadline3 - time.time()
    per_n = max(elapsed_for_n / max(len(n_candidates), 1), 2.0)
    for nt in n_candidates:
        if time.time() > deadline3:
            break
        for st in range(4):
            if time.time() > deadline3:
                break
            if st == 0:
                hs = np.zeros(nt); hs[:nt // 2] = 1.0
            elif st == 1:
                hs = np.zeros(nt); hs[::2] = 1.0
                hs = make_binary(hs, nt)
            elif st == 2:
                hs = rng.rand(nt)
            else:
                # Structured: alternating blocks
                period = max(4, nt // 12)
                hs = np.tile([0.0, 0.0, 1.0, 1.0], nt // 4 + 1)[:nt]
                hs = make_binary(hs, nt)
            dl = min(time.time() + per_n / 4.0, deadline3 - 1)
            bh, bc5 = local_search(make_binary(hs, nt), nt, dl)
            if bc5 < best_c5 - 1e-14:
                best_h, best_c5, best_n = bh, bc5, nt

    # Phase 4: Continuous refinement on best (15% budget)
    if best_h is not None:
        deadline4 = t0 + budget_s - 8
        bh, bc5 = continuous_refine(best_h, best_n, deadline4, rng)
        if bc5 < best_c5 - 1e-14:
            best_h, best_c5 = bh, bc5

    # Phase 5: Final binary polish
    if time.time() < t0 + budget_s - 5:
        h_bin = make_binary(best_h, best_n)
        bh, bc5 = local_search(h_bin, best_n, t0 + budget_s - 3)
        if bc5 < best_c5 - 1e-14:
            best_h, best_c5 = bh, bc5

    # Validate
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-8, f"Sum constraint violated: {np.sum(best_h)} vs {best_n/2}"
    best_h = np.clip(best_h, 0.0, 1.0)
    # Re-fix sum after clip
    diff = best_n / 2.0 - np.sum(best_h)
    if abs(diff) > 1e-12:
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            best_h[mask] += diff / np.sum(mask)
    final_c5 = compute_c5(best_h, best_n)
    return best_h, float(final_c5), best_n
