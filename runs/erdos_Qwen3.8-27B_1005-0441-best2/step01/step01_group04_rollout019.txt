import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            i = int(np.argmax(1.0 - h))
            add = min(diff, 1.0 - h[i])
            if add < 1e-15:
                break
            h[i] += add
        else:
            i = int(np.argmin(h))
            sub = min(-diff, h[i])
            if sub < 1e-15:
                break
            h[i] -= sub
    return h

def _softmax_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    shifted = corr - cmax
    return float(cmax + np.log(np.sum(np.exp(alpha * shifted))) / alpha)

class _SMObj:
    def __init__(self, alpha):
        self.alpha = alpha
    def __call__(self, h):
        return _softmax_obj(h, self.alpha)

def _eq_sum(h):
    return float(np.sum(h) - len(h) / 2.0)

def _local_refine_sm(h0, alpha, max_iter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_SMObj(alpha), h0, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': _eq_sum}],
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        return _fix_sum(res.x, n)
    except Exception:
        return h0.copy()

def _mm_obj(x):
    return x[-1]

def _mm_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr

def _mm_eq(x):
    h = x[:-1]
    return float(np.sum(h) - len(h) / 2.0)

def _local_refine_mm(h0, max_iter):
    n = len(h0)
    t_init = _eval_c5(h0) + 0.003
    x0 = np.concatenate([h0, [t_init]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    try:
        res = minimize(_mm_obj, x0, method='SLSQP',
                       constraints=[{'type': 'ineq', 'fun': _mm_ineq},
                                    {'type': 'eq', 'fun': _mm_eq}],
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        return _fix_sum(res.x[:-1], n)
    except Exception:
        return h0.copy()
