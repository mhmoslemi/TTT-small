```python
def compute_gradient(h, weights, n):
    """Compute gradient of soft-max objective w.r.t. h."""
    # weights has length 2n-1
    # For each j: grad[j] = sum_k weights[k] * ((1-h[j-(k-n+1)])*[valid] - h[j+(k-n+1)]*[valid])
    
    lags = np.arange(2*n-1) - (n-1)  # length 2n-1, from -(n-1) to n-1
    grad = np.zeros(n)
    
    for j in range(n):
        # Term 1: (1-h[j-lag]) for valid j-lag
        idx1 = j - lags
        valid1 = (idx1 >= 0) & (idx1 < n)
        t1 = np.sum(weights[valid1] * (1 - h[idx1[valid1]]))
        
        # Term 2: h[j+lag] for valid j+lag
        idx2 = j + lags
        valid2 = (idx2 >= 0) & (idx2 < n)
        t2 = np.sum(weights[valid2] * h[idx2[valid2]])
        
        grad[j] = t1 - t2
    
    return grad
```

For binary h, I can try:

- For j: affects corr at lags where j±lag is in range → at most 2n-1 lags
- Same for m

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.9999999
            cnt = np.sum(mask)
            if cnt == 0: break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-9
            cnt = np.sum(mask)
            if cnt == 0: break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    m = c.max()
    shifted = beta * (c - m)
    shifted -= shifted.max()
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def compute_weights(corr, beta):
    m = corr.max()
    shifted = beta * (corr - m)
    shifted -= shifted.max()
    return np.exp(shifted) / np.sum(np.exp(shifted))

def grad_softmax(h, weights, n):
    lags = np.arange(2*n-1) - (n-1)
    grad = np.zeros(n)
    for j in range(n):
        idx1 = j - lags
        v1 = (idx1 >= 0) & (idx1 < n)
        t1 = np.dot(weights[v1], (1-h)[idx1[v1]])
        idx2 = j + lags
        v2 = (idx2 >= 0) & (idx2 < n)
        t2 = np.dot(weights[v2], h[idx2[v2]])
        grad[j] = t1 - t2
    return grad

def coord_descent(h, n, t0, time_limit, max_sweeps=200):
    best = h.copy()
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        for beta in [50, 200, 1000]:
            corr = np.correlate(best, 1.0-best, mode="full")
            w = compute_weights(corr, beta)
            g = grad_softmax(best, w, n)
            # Gradient descent step with projection
            lr = 0.5 / np.max(np.abs(g)) if np.max(np.abs(g)) > 0 else 0.01
            new_h = best - lr * g
            new_h = np.clip(new_h, 0, 1)
            new_h = project_feasible(new_h, n)
            c5_new = compute_c5(new_h)
            if c5_new < best_c5:
                best = new_h
                best_c5 = c5_new
            else:
                # Try smaller step
                lr *= 0.1
                new_h = best - lr * g
                new_h = np.clip(new_h, 0, 1)
                new_h = project_feasible(new_h, n)
                c5_new = compute_c5(new_h)
                if c5_new < best_c5:
                    best = new_h
                    best_c5 = c5_new
    return best, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(3000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    np.clip(h, 0.0, 1.0, out=h)
    return h

def _compute_grad(h, w, n):
    lags = np.arange(2 * n - 1) - (n - 1)
    omh = 1.0 - h
    grad = np.empty(n)
    for j in range(n):
        i1 = j - lags
        m1 = (i1 >= 0) & (i1 < n)
        i2 = j + lags
        m2 = (i2 >= 0) & (i2 < n)
        grad[j] = np.dot(w[m1], omh[i1[m1]]) - np.dot(w[m2], h[i2[m2]])
    return grad

def _softmax_obj_grad(h, n, beta, lam):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    m = c.max()
    s = beta * (c - m)
    s -= s.max()
    eg = np.exp(s)
    w = eg / np.sum(eg)
    f = (m + np.log(np.sum(eg)) / beta) * dx
    pen = lam * (np.sum(h) - n / 2.0) ** 2
    g = dx * _compute_grad(h, w, n) + lam * (np.sum(h) - n / 2.0) * np.ones(n)
    return f + pen, g

def lbfgsb_optimize(h, n, t0, time_limit):
    best = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for beta, lr in [(10, 1.0), (50, 0.5), (200, 0.3), (800, 0.2), (3000, 0.1)]:
        if time.time() - t0 > time_limit * 0.88:
            break
        lam_seq = [1.0, 10.0, 50.0, 200.0, 500.0, 2000.0, 8000.0]
        for lam in lam_seq:
            if time.time() - t0 > time_limit * 0.88:
                break
            try:
                res = minimize(_softmax_obj_grad, best, args=(n, beta, lam),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-10,
                                        'maxls': 50})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    hn = project_feasible(res.x, n)
                    c5n = compute_c5(hn)
                    if c5n < best_c5 - 1e-12:
                        best = hn.copy()
                        best_c5 = c5n
            except Exception:
                pass
    return best, best_c5

def proj_gradient_descent(h, n, t0, time_limit, max_iters=5000):
    best = h.copy()
    best_c5 = compute_c5(h)
    beta = 200
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        c = np.correlate(best, 1.0 - best, mode="full")
        m = c.max()
        s = beta * (c - m)
        s -= s.max()
        w = np.exp(s) / np.sum(np.exp(s))
        g = _compute_grad(best, w, n)
        lr = 0.3 / (np.max(np.abs(g)) + 1e-12)
        new_h = best - lr * g
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_feasible(new_h, n)
        c5_new = compute_c5(new_h)
        if c5_new < best_c5 - 1e-14:
            best = new_h
            best_c5 = c5_new
        elif it > 100 and it % 2 == 0:
            beta = max(beta * 1.5, 1e6)
    return best, best_c5

def binary_local_search(h_bin, n, t0, time_limit, max_passes=300):
    h = (h_bin > 0.5).astype(float).copy()
    need = n / 2.0 - np.sum(h)
    if need > 0:
        order = np.argsort(np.diff(h, prepend=0))
        for i in range(int(need)):
            h[order[i]] = 1.0
    elif need < 0:
        order = np.argsort(np.diff(h, prepend=0))
        for i in range(int(-need)):
            h[order[i]] = 0.0
    h = project_feasible(h, n)
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    best_c5 = float(corr.max() * 2.0 / n)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    for p in range(max_passes):
        if time.time() - t0 > time_limit:

        improved = False
        rng_pass = np.random.RandomState(p + 42)
        shuffled_ones = ones.copy()
        rng_pass.shuffle(shuffled_ones)
        for j in shuffled_ones:
            if h[j] != 1.0:

            best_m = -1
            best_max = corr.max()
            for m in zeros:
                if h[m] != 0.0:
                    continue
                h[j] = 0.0
                h[m] = 1.0
                c_new = np.correlate(h, 1.0 - h, mode="full")
                nm = c_new.max()
                if nm < best_max - 1e-12:

                    best_m = m
                h[j] = 1.0

            if best_m >= 0:

                if c5_new < best_c5:

    return h, best_c5
```
