        else:
            lr *= 0.6
            if lr < 1e-12:
                break
    return best_h, best_v

def coord_descent(h, t_budget, rng):
    n = len(h)
    best_c = compute_c5(h)
    t0 = time.time()
    scales = [1.0, 0.5, 0.25, 0.1, 0.04]
    for scale in scales:
        if time.time() - t0 > t_budget:
            break
        base_d = 0.08 * scale
        deltas = [base_d, -base_d, 2*base_d, -2*base_d, 0.5*base_d, -0.5*base_d]
        improved = True
        while improved and time.time() - t0 < t_budget:
            improved = False
            order = rng.permutation(n)
            for i in order:
                if time.time() - t0 > t_budget - 2:
                    break
                for j in order:
                    if i == j or abs(i - j) < 3:
                        continue
                    for d in deltas:
                        hi_new = h[i] - d
                        hj_new = h[j] + d
                        if hi_new < -1e-12 or hi_new > 1 + 1e-12:
                            continue
                        if hj_new < -1e-12 or hj_new > 1 + 1e-12:
                            continue
                        h[i] = hi_new
                        h[j] = hj_new
                        c = compute_c5(h)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                            break
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def local_optimize(h, t_budget, rng):
    n = len(h)
    target = n / 2.0
    h = project_feasible(h.copy(), target)
    best_h = h.copy()
    best_c = compute_c5(h)
    t0 = time.time()

    # Gradient phases with increasing beta
    remaining = t_budget - (time.time() - t0)
    if remaining > 40:
        for beta in [10.0, 25.0, 60.0, 120.0]:
            rem = t_budget - (time.time() - t0)
            if rem < 15:
                break
            h_g, c_g = projected_gradient(h, beta, steps=600, lr_init=0.05,
                                          t_limit=rem * 0.35)
            if c_g < best_c:
                best_c = c_g
                best_h = h_g.copy()
            h = h_g

    # Coordinate descent polish
    rem = t_budget - (time.time() - t0)
    if rem > 10:
        h_cd, c_cd = coord_descent(best_h.copy(), t_budget=rem, rng=rng)
        if c_cd < best_c:
            best_c = c_cd
            best_h = h_cd.copy()

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None

    if h_init is None:
        n = 70
        h_init = np.zeros(n)
        h_init[:n // 2] = 1.0
    else:
        h_init = h_init.copy()

    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init, target)
    best_c5 = compute_c5(best_h)

    # Phase 1: Main optimization from initial (250s)
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 120:
        h_opt, c_opt = local_optimize(best_h, t_budget=min(250, t_rem), rng=rng)
        if c_opt < best_c5:
            best_c5 = c_opt
            best_h = h_opt

    # Phase 2: Structured perturbations (use remaining time)
    t_rem = budget_s - (time.time() - t_start)
    perturb_types = [
        lambda b, s: b + 0.05 * rng.randn(n),
        lambda b, s: b + 0.1 * rng.randn(n),
        lambda b, s: b + 0.15 * rng.randn(n),
        lambda b, s: _flip_random(b, s, 5),
        lambda b, s: _flip_random(b, s, 10),
        lambda b, s: _binary_push(b, s, 0.3),
        lambda b, s: _shift_pattern(b, s, rng.randint(1, 7)),
        lambda b, s: _mirror_mix(b, s),
    ]

    n_rest = min(len(perturb_types), int(t_rem / 60))
    for pi in range(n_rest):
        t_rem = budget_s - (time.time() - t_start)
        if t_rem < 55:
            break
        h_pert = project_feasible(perturb_types[pi](best_h, 0), target)
        t_sub = min(50, t_rem - 5)
        hp, cp = local_optimize(h_pert, t_budget=t_sub, rng=rng)
        if cp < best_c5:
            best_c5 = cp
            best_h = hp

    # Phase 3: Random binary-like starts
    t_rem = budget_s - (time.time() - t_start)
    n_rand = min(int(t_rem / 50), 4) if t_rem > 50 else 0
    for i in range(n_rand):
        t_rem = budget_s - (time.time() - t_start)
        if t_rem < 45:
            break
        h_rand = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h_rand[idx] = 1.0
        t_sub = min(42, t_rem - 5)
        hr, cr = local_optimize(h_rand, t_budget=t_sub, rng=rng)
        if cr < best_c5:
            best_c5 = cr
            best_h = hr

    # Phase 4: Final intensive coordinate descent on best
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 20:
        h_fin, c_fin = coord_descent(best_h.copy(), t_budget=t_rem - 5, rng=rng)
        if c_fin < best_c5:
            best_c5 = c_fin
            best_h = h_fin

    # Final projection and validation
    best_h = project_feasible(best_h, target)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = best_h.copy()

    return (best_h, best_c5, n)

def _flip_random(h, seed_val, k):
    """Randomly flip k bits and redistribute."""
    n = len(h)
    h = h.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    if len(ones) < k or len(zeros) < k:
        return h
    r = np.random.RandomState(seed_val)
    flip_to_zero = r.choice(ones, min(k, len(ones)), replace=False)
    flip_to_one = r.choice(zeros, min(k, len(zeros)), replace=False)
    n_flip = min(len(flip_to_zero), len(flip_to_one))
    for i in range(n_flip):
        h[flip_to_zero[i]] = 0.0
        h[flip_to_one[i]] = 1.0
    return h

def _binary_push(h, seed_val, frac):
    """Push fractional values toward 0 or 1."""
    h = h.copy()
    mid = (h > 0.25) & (h < 0.75)
    n_mid = mid.sum()
    if n_mid == 0:
        return h
    # Push toward nearest binary
    h[mid & (h > 0.5)] = np.minimum(h[mid & (h > 0.5)] + frac, 1.0)
    h[mid & (h <= 0.5)] = np.maximum(h[mid & (h <= 0.5)] - frac, 0.0)
    return h

def _shift_pattern(h, seed_val, shift):
    """Circular shift by 'shift' positions."""
    n = len(h)
    return np.roll(h, shift)

def _mirror_mix(h, seed_val):
    """Mix h with its mirror."""
    n = len(h)
    mirror = np.flipud(h)
    return 0.5 * h + 0.5 * mirror
