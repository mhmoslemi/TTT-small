def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    """Project h to feasible set: 0<=h<=1, sum(h)=n/2"""
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    target = float(n) / 2.0
    if s > 1e-12:
        h = h * target / s
    h = np.clip(h, 0.0, 1.0)
    # Fix any remaining sum violation
    diff = target - np.sum(h)
    if abs(diff) > 1e-9:
        # Distribute the difference
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
            h = np.clip(h, 0.0, 1.0)
    return h

def _soft_max_obj(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    t = max(temp, 1e-10)
    val = (m + t * np.log(np.sum(np.exp((corr - m) / t)))) * dx
    return val

                               constraints=cons, options={'maxiter': 400, 'ftol': 1e-13})
                h_new = project_feasible(res.x, n)
                v = compute_c5(h_new)
                if v < best_val - 1e-13:
                    best = h_new
                    best_val = v
                h_cur = h_new
            except Exception:
                break
    return best, best_val

def _gauss_mut(h, sigma, rng):
    return h + sigma * rng.standard_normal(len(h))

def _block_swap_mut(h, rng, n_blocks=4):
    h_mut = h.copy()
    n = len(h)
    for _ in range(n_blocks):
        i = int(rng.integers(0, max(1, n // 3)))
        j = int(rng.integers(2 * n // 3, n))
        ln = int(rng.integers(2, max(3, n // 8)))
        tmp = h_mut[i:i+ln].copy()
        h_mut[i:i+ln] = h_mut[j:j+ln]
        h_mut[j:j+ln] = tmp
    return h_mut

def _low_freq_mut(h, rng, n=6):
    m = len(h)
    noise = np.zeros(m)
    for k in range(1, n + 1):
        amp = rng.standard_normal() / k
        ph = rng.uniform(0, 2 * np.pi)
        noise += amp * np.sin(2 * np.pi * k * np.arange(m) / m + ph)
    return h + 0.08 * noise

def _random_init(n, rng):
    h = rng.random(n)
    return project_feasible(h, n)

def _two_block_init(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    half = n_ones // 2
    gap = int(rng.integers(n // 10, n // 3))
    pos1 = int(rng.integers(0, max(1, n - 2 * half - gap)))
    pos2 = (pos1 + half + gap) % n
    for i in range(half):
        h[(pos1 + i) % n] = 1.0
        h[(pos2 + i) % n] = 1.0
    return h

def _multi_block_init(n, rng, n_blocks):
    h = np.zeros(n)
    n_ones = n // 2
    bs = max(1, n_ones // n_blocks)
    positions = np.sort(rng.choice(n, size=n_blocks, replace=False))
    for p in positions:
        for i in range(bs):
            h[(p + i) % n] = 1.0
    actual = int(np.sum(h > 0.5))
    target = n // 2
    if actual < target:
        zeros = np.where(h == 0)[0]
        if len(zeros) >= target - actual:
            extra = rng.choice(zeros, size=target - actual, replace=False)
            h[extra] = 1.0
    elif actual > target:
        ones = np.where(h == 1)[0]
        if len(ones) >= actual - target:
            rem = rng.choice(ones, size=actual - target, replace=False)
            h[rem] = 0.0
    return project_feasible(h, n)

def _evolve(h_init, n, pop_size, n_gen, rng, max_time):
    t0 = time.time()
    pop = [h_init.copy()]
    pops = [compute_c5(pop[0])]
    initial_muts = [
        _gauss_mut(h_init, 0.12, rng), _gauss_mut(h_init, 0.04, rng),
        _low_freq_mut(h_init, rng, 4), _low_freq_mut(h_init, rng, 8),
        _block_swap_mut(h_init, rng, 3), _block_swap_mut(h_init, rng, 5),
        _two_block_init(n, rng), _multi_block_init(n, rng, 3),
        _multi_block_init(n, rng, 5), _random_init(n, rng),
    ]
    for hm in initial_muts:
        hm = project_feasible(hm, n)
        pop.append(hm)
        pops.append(compute_c5(hm))
    while len(pop) < pop_size:
        idx = int(rng.integers(0, len(pop)))
        hn = project_feasible(_gauss_mut(pop[idx], 0.08, rng), n)
        pop.append(hn)
        pops.append(compute_c5(hn))
    sigmas = [0.06] * pop_size
    best_val = min(pops)
    keep = max(4, pop_size // 2)
    for gen in range(n_gen):
        if time.time() - t0 > max_time:
            break
        order = np.argsort(pops)
        pop = [pop[i] for i in order]
        pops = [pops[i] for i in order]
        sigmas = [sigmas[i] for i in order]
        if pops[0] < best_val:
            best_val = pops[0]
        for i in range(keep, pop_size):
            cand = rng.integers(0, keep, size=3)
            parent = int(cand[np.argmin([pops[c] for c in cand])])
            mt = rng.random()
            if mt < 0.45:
                hnew = _gauss_mut(pop[parent], sigmas[parent], rng)
            elif mt < 0.65:
                hnew = _low_freq_mut(pop[parent], rng, int(rng.integers(3, 9)))
            elif mt < 0.80:
                hnew = _block_swap_mut(pop[parent], rng, int(rng.integers(2, 6)))
            elif mt < 0.90:
                hnew = _gauss_mut(pop[parent], sigmas[parent] * 2.5, rng)
            else:
                hnew = _gauss_mut(pop[parent], sigmas[parent] * 0.3, rng)
            hnew = project_feasible(hnew, n)
            v = compute_c5(hnew)
            if v < pops[parent]:
                sigmas[parent] = min(0.25, sigmas[parent] * 1.15)
            else:
                sigmas[parent] = max(0.005, sigmas[parent] * 0.92)
            pop[i] = hnew
            pops[i] = v
    best_idx = int(np.argmin(pops))
    return pop[best_idx], pops[best_idx]

def _upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    else:
        n = 160
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    temps_full = [0.08, 0.03, 0.01, 0.003, 0.0005, 0.0001]

    # Phase 1: ES global search
    rem = budget_s - (time.time() - t0)
    if rem > 200:
        h_es, c5_es = _evolve(h0, n, 35, 8000, rng, min(300, rem - 150))
        if c5_es < best_c5:
            best_h, best_c5 = h_es, c5_es

    # Phase 2: SLSQP refine
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        h_r, c5_r = _slsqp_refine(best_h.copy(), n, seed, min(150, rem - 30), temps_full)
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r

    # Phase 3: ES again from refined
    rem = budget_s - (time.time() - t0)
    if rem > 150:
        h_es2, c5_es2 = _evolve(best_h, n, 30, 6000, rng, min(250, rem - 100))
        if c5_es2 < best_c5:
            best_h, best_c5 = h_es2, c5_es2

    # Phase 4: SLSQP again
    rem = budget_s - (time.time() - t0)
    if rem > 50:
        h_r2, c5_r2 = _slsqp_refine(best_h.copy(), n, seed + 50, min(120, rem - 20), temps_full)
        if c5_r2 < best_c5:
            best_h, best_c5 = h_r2, c5_r2

    # Phase 5: Try different n
    for n_try in [n - 40, n + 40, n + 80]:
        rem = budget_s - (time.time() - t0)
        if rem < 80 or n_try <= 40 or n_try > 400:
            break
        if n_try == n:
            continue
        if n_try < n:
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_tr = project_feasible(np.interp(x_new, x_old, best_h), n_try)
        else:
            h_tr = _upsample(best_h, n_try)
        h_r3, c5_r3 = _slsqp_refine(h_tr, n_try, seed + n_try, min(80, rem - 30), temps_full)
        if c5_r3 < best_c5:
            best_h, best_c5, n = h_r3, c5_r3, n_try

    # Phase 6: Final ES burst
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        h_es3, c5_es3 = _evolve(best_h, len(best_h), 25, 4000, rng, max(20, min(120, rem - 40)))
        if c5_es3 < best_c5:
            best_h, best_c5 = h_es3, c5_es3

    # Phase 7: Final polish
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        h_fin, c5_fin = _slsqp_refine(best_h.copy(), len(best_h), seed + 9999,
                                       max(5, rem - 3), [0.01, 0.001, 0.0001])
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin, c5_fin

    return best_h, best_c5, len(best_h)
