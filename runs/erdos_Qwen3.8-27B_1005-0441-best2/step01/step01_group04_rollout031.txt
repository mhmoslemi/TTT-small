import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _compute_corr(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project_sum(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def _subgrad_topk(h, corr, topk=8):
    n = len(h)
    top_idx = np.argsort(corr)[-topk:]
    alpha_w = 15.0
    cw = corr[top_idx]
    w = np.exp(alpha_w * (cw - cw.max()))
    w /= w.sum()
    grad = np.zeros(n)
    for idx, wi in zip(top_idx, w):
        k = int(idx) - (n - 1)
        if k >= 0:
            if k < n:
                i1 = np.arange(k, n)
                grad[i1] += wi * (1.0 - h[i1 - k])
            if k > 0 and k < n:
                i2 = np.arange(0, n - k)
                grad[i2] -= wi * h[i2 + k]
        else:
            kk = -k
            if kk < n:
                i1 = np.arange(0, n - kk)
                grad[i1] += wi * (1.0 - h[i1 + kk])
            if kk > 0 and kk < n:
                i2 = np.arange(kk, n)
                grad[i2] -= wi * h[i2 - kk]
    return grad

def _subgrad_run(h0, max_iter=3000, topk=8):
    n = len(h0)
    h = h0.copy()
    base_c5 = _eval_c5(h)
    step = 0.01
    patience = 0
    best_h = h.copy()
    best_c5 = base_c5
    for it in range(max_iter):
        corr = _compute_corr(h)
        c5 = np.max(corr)
        if c5 < best_c5 - 1e-8:
            best_h = h.copy()
            best_c5 = c5
            patience = 0
        else:
            patience += 1
        if patience > 300:
            step *= 0.7
            patience = 100
        grad = _subgrad_topk(h, corr, topk)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        h = h - step * grad / max(gnorm, 1e-10) * np.linalg.norm(grad)
        h = _project_sum(h, n)
        step = max(step * (1 + 1e-6 * it), 0.002)
    return best_h, best_c5

def _softmax_obj(h, alpha):
    corr = _compute_corr(h)
    cmax = np.max(corr)
    shifted = corr - cmax
    logsum = np.log(np.sum(np.exp(alpha * shifted)))
    return cmax + logsum / alpha

def _eq_sum(h):
    return float(np.sum(h) - len(h) / 2.0)

class _ObjF:
    def __init__(self, alpha):
        self.alpha = alpha
    def __call__(self, x):
        return _softmax_obj(x, self.alpha)

def _slsqp_refine(h0, alpha=100, max_iter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_ObjF(alpha), h0, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': _eq_sum}],
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        return _project_sum(res.x, n)
    except Exception:
        return h0.copy()

def _coord_descent(h, n_iters=50):
    n = len(h)
    for _ in range(n_iters):
        improved = False
        for i in range(n):
            c0 = _compute_corr(h)
            cmax = np.max(c0)
            best_val = h[i]
            for delta in [-0.05, -0.02, -0.01, 0.01, 0.02, 0.05]:
                h[i] = np.clip(h[i] + delta, 0, 1)
                h = _project_sum(h, n)
                c = _compute_corr(h)
                if np.max(c) < cmax - 1e-10:
                    best_val = h[i]
                    improved = True
            h[i] = best_val
        h = _project_sum(h, n)
        if not improved:
            break
    return h

def _make_start(n, rng, style):
    if style == 0:
        h = rng.random(n)
    elif style == 1:
        h = np.where(np.arange(n) % 3 < 2, 0.9, 0.1)
    elif style == 2:
        x = np.linspace(0, 2 * np.pi * 2, n, endpoint=False)
        h = 0.5 + 0.3 * np.sin(x) + 0.1 * np.sin(3 * x)
    elif style == 3:
        h = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
    elif style == 4:
        h = np.zeros(n)
        for j in range(n // 2):
            h[(j * 3) % n] = 1.0
    else:
        h = 0.5 + 0.2 * np.cos(2 * np.pi * np.arange(n) / n * 5)
    return _project_sum(h, n)

    best_h = None

    # Try initial_h_values if available

    if h_parent is not None and remaining() > 200:
