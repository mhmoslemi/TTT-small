```python
    # Get top M correlations
    top_idx = np.argsort(corr)[-M:]
    # Weights: exponential of (corr - max) / temp
    max_c = np.max(corr)
    temp = max_c - np.min(corr[top_idx]) + 1e-10
    w = np.exp((corr[top_idx] - max_c) / temp)
    w = w / np.sum(w)
    
    grad = np.zeros(n)
    for wi, ki in zip(w, top_idx):
        shift = ki - (n - 1)
        g = np.zeros(n)
        if shift >= 0:
            vi = np.arange(0, n - shift)
            g[vi] += (1 - h[vi + shift])
            vj = np.arange(shift, n)
            g[vj] -= h[vj - shift]
        else:
            s = -shift
            vi = np.arange(s, n)
            g[vi] += (1 - h[vi - s])
            vj = np.arange(0, n - s)
            g[vj] -= h[vj + s]
        grad += wi * g
    return grad * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project(h, target):
    """Project onto {0<=h<=1, sum=target}"""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h_p)
    if abs(diff) > 1e-12:
        mask = (h_p > 1e-8) & (h_p < 1.0 - 1e-8)
        cnt = np.sum(mask)
        if cnt > 0:
            h_p[mask] += diff / cnt
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def _grad_at_lag(h, n, shift):
    """Gradient of corr at a specific lag (shift) w.r.t. h"""
    g = np.zeros(n)
    if shift >= 0:
        vi = np.arange(0, n - shift)
        g[vi] += (1.0 - h[vi + shift])
        vj = np.arange(shift, n)
        g[vj] -= h[vj - shift]
    else:
        s = -shift
        vi = np.arange(s, n)
        g[vi] += (1.0 - h[vi - s])
        vj = np.arange(0, n - s)
        g[vj] -= h[vj + s]
    return g

def _momentum_optimize(h_init, n, t0, max_time, seed):
    """Momentum-based projected gradient with soft subgradient"""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    target = n / 2.0
    
    h = h_init.copy()
    best = h.copy()
    best_val = _compute_c5(best)
    vel = np.zeros(n)
    mu = 0.9  # momentum
    lr = 0.02  # initial learning rate
    no_improve = 0
    M_top = 5  # number of top lags to average
    
    while time.time() - t0 < max_time:
        corr = np.correlate(h, 1.0 - h, mode="full")
        # Get top M lags
        top_idx = np.argsort(corr)[-M_top:]
        max_c = np.max(corr)
        # Soft weights
        temps = max_c - corr[top_idx] + 1e-10
        w = np.exp(-temps / 0.3)
        w /= np.sum(w)
        
        # Compute combined gradient
        grad = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            shift = ki - (n - 1)
            grad += wi * _grad_at_lag(h, n, shift)
        grad *= dx
        
        # Momentum update
        vel = mu * vel - lr * grad
        h_new = h + vel
        h_new = _project(h_new, target)
        
        val = _compute_c5(h_new)
        if val < best_val - 1e-12:
            best = h_new.copy()
            best_val = val
            h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                # Reduce learning rate
                lr *= 0.95
                vel *= 0.5
                # Occasionally add small perturbation
                if no_improve > 100:
                    h_new = h + 0.01 * rng.randn(n)
                    h_new = _project(h_new, target)
                    val = _compute_c5(h_new)
                    if val < best_val:
                        best = h_new.copy()
                        best_val = val
                    h = h_new.copy()
                    no_improve = 0
                if no_improve > 200:
                    # Reset momentum
                    vel = np.zeros(n)
                    no_improve = 0
                    lr = max(0.001, lr * 1.5)
        
        # Periodically try larger M_top
        if no_improve > 80 and no_improve % 80 == 0:
            M_top = min(10, M_top + 2)
    
    return best, best_val
```

Actually, you know what, let me try a different angle. What if I use the fact that the problem is equivalent to maximizing min_k autocorr(k), and I use a "min-max" iterative approach?

At each iteration:
1. Compute all autocorrelations A(k) = sum_i h[i]*h[i+k] (for the overlap region)
2. Find the minimum: k_min
3. Compute the gradient of A(k_min) w.r.t. h: g[j] = h[j+k_min] + h[j-k_min] (where defined)
4. Update: h += lr * g (to increase the minimum autocorrelation)
5. Project

This is more directly targeting the bottleneck.

But wait, increasing A(k_min) might decrease A(k_max) below the current min. So it's a trade-off.

A better approach: use the "Dinkelbach" or "bisection" method:
- Bisect on t: is there a feasible h with A(k) ≥ t for all k?
- For a given t, the feasibility problem is: maximize min_k A(k) - t, or check if min_k A(k) ≥ t
- The inner optimization is still hard

1. Soft-max gradient descent with momentum

```python
    dx = 2.0 / n
    target = n / 2.0
    rng = np.random.RandomState(seed)
    h = h0.copy()
    best = h.copy()
    best_val = _compute_c5(best)
    vel = np.zeros(n)
    mu = 0.85
    lr = 0.015
    no_imp = 0
    M_top = 5
    step_count = 0

    while time.time() - t0 < max_time:
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argsort(corr)[-M_top:]

        spread = max_c - corr[top_idx]

        grad *= dx

        h_new = _project(h + vel, target)
        val = _compute_c5(h_new)
        step_count += 1

        if val < best_val - 1e-12:

            best_val = val
            h = h_new.copy()
            no_imp = 0

def _slsqp_obj(h, temp, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + temp * np.log(np.sum(np.exp((corr - m) / temp)))) * dx

def _slsqp_polish(h0, n, t0, max_time, seed):
    target = n / 2.0
    best = h0.copy()
    best_val = _compute_c5(best)
    bounds = [(0.0, 1.0)] * n
    temps = [0.02, 0.005, 0.001, 0.0003]

    for temp in temps:
        if time.time() - t0 > max_time - 5:
            break
        h = best.copy()

                break
            try:
                res = minimize(
                    lambda x: _slsqp_obj(x, temp, n), h,
                    method='SLSQP', bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                                  'jac': lambda x: np.ones(n)}],
                    options={'maxiter': 300, 'ftol': 1e-12})
                h_new = _project(res.x, target)
                v = _compute_c5(h_new)
                if v < best_val - 1e-12:
                    best = h_new.copy()
                    best_val = v
                h = h_new.copy()
            except Exception:
                break
    return best, best_val

def _structured_init(n, rng, k_pulses, gap_frac):

    per_pulse = n_ones // k_pulses

            if idx not in used:

    return _project(h, n / 2.0)
```
