import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vec(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _fix_sum(h):
    n = len(h)
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(10000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-12
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return h

class _SumCon:
    def __init__(self, n):
        self.n = n
    def __call__(self, h):
        return float(np.sum(h) - self.n / 2.0)

class _SO:
    def __init__(self, alpha):
        self.a = alpha
    def __call__(self, h):
        c = _corr_vec(h)
        m = float(np.max(c))
        s = self.a * (c - m)
        s = s - np.max(s)
        return m + np.log(np.sum(np.exp(s))) / self.a

def _do_slq(h0, alpha, iters):
    n = len(h0)
    try:
        r = minimize(_SO(alpha), h0.copy(), method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _SumCon(n)}],
                     options={'maxiter': iters, 'ftol': 1e-15})
        return _fix_sum(r.x)
    except Exception:
        return h0.copy()

def _coord_descent(h, max_rounds, rng):
    n = len(h)
    best_c5 = _eval_c5(h)
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for rnd in range(max_rounds):
        improved = False
        order = rng.permutation(n)
        for i in order:
            for d in deltas:
                for sign in [1, -1]:
                    hp = h.copy()
                    hp[i] = np.clip(hp[i] + sign * d, 0.0, 1.0)
                    hp = _fix_sum(hp)
                    c5 = _eval_c5(hp)
                    if c5 < best_c5 - 1e-15:
                        h = hp
                        best_c5 = c5
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    return h, best_c5

def _sa_explore(h, rng, t_start, t_end, n_iter):
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    for it in range(n_iter):
        T = t_start * (t_end / t_start) ** (it / max(1, n_iter - 1))
        h_new = cur_h.copy()
        n_mod = rng.integers(2, 6)
        idx = rng.choice(n, size=n_mod, replace=False)
        for i in idx:
            h_new[i] += rng.normal(0, 0.08)
        h_new = _fix_sum(h_new)
        c5_new = _eval_c5(h_new)
        if c5_new < cur_c5 or rng.random() < np.exp(-(c5_new - cur_c5) / max(T, 1e-10)):
            cur_h = h_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
    return best_h, best_c5

def _try_n(h_start, n, rng, budget_left):
    if n == len(h_start):
        h0 = h_start.copy()
    else:
        x_old = np.linspace(0, 2, len(h_start), endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h0 = np.interp(x_new, x_old, h_start, period=2.0)
        h0 = _fix_sum(h0)
    alphas = [30, 80, 200, 500, 1200, 3000, 6000]
    best = h0
    best_c5 = _eval_c5(h0)
    for a in alphas:
        if time.time() > budget_left:
            break
        h_opt = _do_slq(best, a, 800)
        c5 = _eval_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best = h_opt
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.default_rng(seed)
    best_c5 = np.inf
    best_h = None

    def elapsed():
        return time.time() - t0

    def remaining():
        return deadline - time.time()

    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if h_init is not None and len(h_init) > 0:
        h_init = _fix_sum(h_init)
        c5 = _eval_c5(h_init)
        best_c5 = c5
        best_h = h_init.copy()
        n_init = len(h_init)
    else:
        n_init = 80
        h_init = np.full(n_init, 0.5)
        best_c5 = _eval_c5(h_init)
        best_h = h_init.copy()

    # Phase 1: Progressive alpha SLSQP on initial (n_init)
    alphas_main = [20, 50, 100, 200, 500, 1000, 2000, 4000, 8000, 15000]
    h_cur = best_h.copy()
    for a in alphas_main:
        if remaining() < 80:
            break
        h_cur = _do_slq(h_cur, a, 1200)
        c5 = _eval_c5(h_cur)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cur.copy()

    # Phase 2: Perturbation + re-optimize loop on best
    for trial in range(8):
        if remaining() < 100:
            break
        scale = 0.03 + 0.02 * trial
        h_pert = _fix_sum(best_h + rng.normal(0, scale, len(best_h)))
        h_opt = _do_slq(h_pert, 500, 800)
        h_opt = _do_slq(h_opt, 2000, 800)
        c5 = _eval_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()

    # Phase 3: Try other n values (interpolation)
    for n_try in [60, 64, 72, 88, 96, 100, 47, 50, 53]:
        if remaining() < 80:
            break
        if n_try == len(best_h):
            continue
        h_try, c5_try = _try_n(best_h, n_try, rng, time.time() + min(remaining() - 10, 60))
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 4: Simulated annealing exploration
    if remaining() > 120:
        n_sa = min(3000, max(500, int(60 * 100000 / (len(best_h) ** 2))))
        h_sa, c5_sa = _sa_explore(best_h, rng, 0.01, 0.0001, n_sa)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        # Polish SA result
        if remaining() > 60:
            h_pol = _do_slq(h_sa, 2000, 600)
            c5_pol = _eval_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 5: Coordinate descent polish
    if remaining() > 40:
        n_rounds = min(200, max(20, int(remaining() * 0.4 * 50)))
        best_h, best_c5 = _coord_descent(best_h, n_rounds, rng)

    # Phase 6: Final high-alpha SLSQP
    if remaining() > 30:
        h_final = _do_slq(best_h, 20000, 1500)
        c5_final = _eval_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 7: One more coordinate descent
    if remaining() > 20:
        best_h, best_c5 = _coord_descent(best_h, 50, rng)

    # Final verification
    best_h = _fix_sum(best_h)
    best_c5 = _eval_c5(best_h)

    return best_h, best_c5, len(best_h)
