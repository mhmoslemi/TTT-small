    mc = np.max(corr)
    shifted = beta * (corr - mc)
    max_s = np.max(shifted)
    w = np.exp(shifted - max_s)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-15:
            continue
        d = m - (n - 1)
        lo1 = max(0, -d)
        hi1 = min(n, n - d)
        if lo1 < hi1:
            grad[lo1:hi1] += wm * (1.0 - h[lo1 + d:hi1 + d])
        lo2 = max(0, d)
        hi2 = min(n, d + n)
        if lo2 < hi2:
            grad[lo2:hi2] -= wm * h[lo2 - d:hi2 - d]
    return grad * dx


def project(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = free.sum()
        if nf == 0:
            h = np.clip(h + (target - s) / n, 0.0, 1.0)
        else:
            h[free] += (target - s) / nf
            h = np.clip(h, 0.0, 1.0)
    return h


def _slsqp_obj(x, beta):
    return smooth_val(x, beta)


def _slsqp_jac(x, beta):
    return smooth_grad(x, beta)


def _con_fun(x):
    return x.sum() - x.shape[0] / 2.0


def _con_jac(x):
    return np.ones(x.shape[0])


def slsqp_smooth(h0, beta, maxiter=250):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_slsqp_obj, h0, args=(beta,), jac=_slsqp_jac,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project(res.x)
    except Exception:
        return project(h0.copy())


def local_opt(h0, t_budget, n_betas=5):
    h = project(h0.copy().astype(float))
    best = compute_c5(h)
    hb = h.copy()
    t0 = time.time()
    for i in range(n_betas):
        if time.time() - t0 > t_budget - 5:
            break
        beta = 12.0 * (i + 1) ** 1.6
        h = slsqp_smooth(h, beta, maxiter=250)
        c = compute_c5(h)
        if c < best:
            best = c
            hb = h.copy()
    return hb, best


def nm_polish(h0, maxiter=400):
    h = project(h0.copy().astype(float))
    hcg = h * (1.0 - h)
    res = minimize(compute_c5, h, method='Nelder-Mead',
                   options={'maxiter': maxiter, 'xatol': 1e-11, 'fatol': 1e-13})
    h_new = project(res.x)
    return h_new, compute_c5(h_new)


def subgrad_descent(h0, n_iters, lr_init=0.002):
    n = len(h0)
    dx = 2.0 / n
    h = project(h0.copy().astype(float))
    best = compute_c5(h)
    hb = h.copy()
    for it in range(n_iters):
        lr = lr_init / (1.0 + 0.01 * it)
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_max = int(np.argmax(corr))
        d = k_max - (n - 1)
        grad = np.zeros(n)
        lo1 = max(0, -d)
        hi1 = min(n, n - d)
        if lo1 < hi1:
            grad[lo1:hi1] += (1.0 - h[lo1 + d:hi1 + d])
        lo2 = max(0, d)
        hi2 = min(n, d + n)
        if lo2 < hi2:
            grad[lo2:hi2] -= h[lo2 - d:hi2 - d]
        h = project(h - lr * grad * dx)
        c = compute_c5(h)
        if c < best:
            best = c
            hb = h.copy()
    return hb, best


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None

    if h_init is None:
        n = 70
        h_init = np.zeros(n)
        h_init[:n // 2] = 1.0

    n = len(h_init)
    best_h = project(h_init.copy())
    best_c5 = compute_c5(best_h)

    # Phase 1: Main SLSQP smooth-max optimization
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 120:
        h_opt, c_opt = local_opt(best_h, t_budget=min(420, t_rem - 20), n_betas=6)
        if c_opt < best_c5:
            best_c5 = c_opt
            best_h = h_opt

    # Phase 1b: NM polish
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 30:
        h_nm, c_nm = nm_polish(best_h, maxiter=min(500, int(t_rem * 2)))
        if c_nm < best_c5:
            best_c5 = c_nm
            best_h = h_nm

    # Phase 2: Subgradient descent on true max
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 30:
        n_sg = min(2000, int(t_rem * 8))
        h_sg, c_sg = subgrad_descent(best_h, n_iters=n_sg, lr_init=0.0015)
        if c_sg < best_c5:
            best_c5 = c_sg
            best_h = h_sg
        h_nm2, c_nm2 = nm_polish(best_h, maxiter=200)
        if c_nm2 < best_c5:
            best_c5 = c_nm2
            best_h = h_nm2

    # Phase 3: Population-based perturbation search
    t_rem = budget_s - (time.time() - t_start)
    n_rounds = min(int(t_rem / 55), 12) if t_rem > 55 else 0
    pop = [best_h.copy()]
    pop_c = [best_c5]

    for r in range(n_rounds):
        t_rem = budget_s - (time.time() - t_start)
        if t_rem < 50:
            break

        candidates = []
        for amp in [0.06, 0.12, 0.20, 0.30]:
            candidates.append(project(best_h + amp * rng.randn(n)))

        if len(pop) > 1:
            candidates.append(project(pop[1] + 0.10 * rng.randn(n)))

        if r % 4 == 2:
            h_rand = np.zeros(n)
            h_rand[rng.choice(n, n // 2, replace=False)] = 1.0
            candidates.append(project(h_rand))

        # Swap-based mutation: swap two random entries
        if r % 2 == 0:
            h_swap = best_h.copy()
            i_s = rng.randint(n)
            j_s = (i_s + rng.randint(1, n)) % n
            h_swap[i_s], h_swap[j_s] = h_swap[j_s], h_swap[i_s]
            candidates.append(project(h_swap))

        t_per = max(8.0, (t_rem - 10.0) / max(1, len(candidates)))
        for h_cand in candidates:
            t_rem = budget_s - (time.time() - t_start)
            if t_rem < 12:
                break
            h_new, c_new = local_opt(h_cand, t_budget=min(t_per, t_rem - 5), n_betas=3)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new
                pop = [h_new.copy()]
                pop_c = [c_new]
            else:
                if c_new < pop_c[-1] + 0.002:
                    pop.append(h_new)
                    pop_c.append(c_new)
                    if len(pop) > 6:
                        order = np.argsort(pop_c)
                        pop = [pop[i] for i in order[:6]]
                        pop_c = [pop_c[i] for i in order[:6]]

    # Phase 4: Coordinate descent with exact 1D optimization
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 25:
        h_now = best_h.copy()
        for sweep in range(30):
            t_rem = budget_s - (time.time() - t_start)
            if t_rem < 5:
                break
            improved = False
            for j in range(n):
                best_v = h_now[j]
                best_c = best_c5
                for frac in [0.2, 0.4, 0.6, 0.8, 0.005, -0.005, 0.01, -0.01, 0.02, -0.02]:
                    h_try = h_now.copy()
                    new_val = h_now[j] + frac
                    if new_val < -0.01 or new_val > 1.01:
                        continue
                    h_try[j] = np.clip(new_val, 0.0, 1.0)
                    diff = h_try.sum() - n / 2.0
                    if abs(diff) > 1e-12:
                        k_adj = (j + 11) % n
                        h_try[k_adj] = np.clip(h_try[k_adj] - diff, 0.0, 1.0)
                        if abs(h_try.sum() - n / 2.0) > 1e-8:
                            h_try = project(h_try)
                    c_try = compute_c5(h_try)
                    if c_try < best_c:
                        best_c = c_try
                        best_v = h_try[j]
                if best_c < best_c5 - 1e-13:
                    h_try = h_now.copy()
                    h_try[j] = best_v
                    h_try = project(h_try)
                    h_now = h_try
                    best_c5 = compute_c5(h_now)
                    improved = True
            if not improved:
                break

    # Phase 5: Final NM polish
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 15:
        h_fin, c_fin = nm_polish(best_h, maxiter=500)
        if c_fin < best_c5:
            best_c5 = c_fin
            best_h = h_fin

    best_h = project(best_h)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, n)
