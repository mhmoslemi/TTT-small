def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _proj(h, n):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        d = n * 0.5 - h.sum()
        if abs(d) < 1e-13:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        if free.sum() == 0:
            h = np.clip(h + d / n, 0, 1)
            break
        h[free] += d / free.sum()
        h = np.clip(h, 0, 1)
    return h

def _weighted_corr_grad(h, p, n):
    """Gradient of sum_k p_k * corr_k * dx"""
    dx = 2.0 / n
    g = np.zeros(n)
    n_lags = 2 * n - 1
    for k_idx in range(n_lags):
        pk = p[k_idx]
        if pk < 1e-15:

        m = k_idx - (n - 1)
        if m >= 0:
            g[:n-m] += pk * (1.0 - h[m:])
            g[m:] -= pk * h[:n-m]
        else:
            pp = -m
            g[:n-pp] += pk * (1.0 - h[pp:])
            g[pp:] -= pk * h[:n-pp]
    return g * dx

    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    shifted = c - cmax
    s = np.exp(shifted / eps)
