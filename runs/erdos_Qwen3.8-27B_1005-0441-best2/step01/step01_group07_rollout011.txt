def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):

        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:

    return float(corr.max() * dx)

def compute_grad_smooth(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    shifted = np.clip(tau * (corr - mc), -500, 500)
    exps = np.exp(shifted)
    weights = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-12:

        lag = k - (n - 1)

            grad[:n - lag] -= w * h[lag:]

            L = -lag
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
    return grad * dx
