So the integral is |A \ (A-k)| = |A| - |A ∩ (A-k)| = 1 - |A ∩ (A-k)|

So C₅ = max_k [1 - |A ∩ (A-k)|] = 1 - min_k |A ∩ (A-k)|

We want to MAXIMIZE the minimum self-overlap! This is the "Erdős problem" in its geometric formulation.

For a convex set (interval) of measure 1 in [0,2]: A = [0,1]. Then |A ∩ (A-k)| = 1-|k| for |k|<1, 0 otherwise. So min = 0 (at k=1 or k=2). C₅ = 1. Bad.

For a "spread out" set: A = [0, 0.5] ∪ [0.75, 1.25] or something. The goal is to make the autocorrelation as flat as possible.

The theoretical minimum for C₅ is related to the Fourier transform. By the uncertainty principle, a set of measure 1 in [0,2] must have some "concentration" in its autocorrelation.

Anyway, for the optimization, the key insight is: we want h to be "as binary as possible" (values close to 0 or 1) but with a structure that maximizes the minimum overlap. The values being in (0,1) is a relaxation that might help slightly.

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    s = beta * c
    s_max = s.max()
    val = (s_max + np.log(np.sum(np.exp(s - s_max)))) / beta * dx
    if flat_w > 0:
        mid = c[n // 4 : 2 * n - 1 - n // 4]
        val += flat_w * np.var(mid)
    return val

def _slsqp(h, n, beta, maxiter, ftol, flat_w):
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(soft_obj, h.copy(), args=(n, beta, flat_w),
                      method='SLSQP', bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return None

def try_binary(h, n):
    h_b = (h > 0.5).astype(float)
    need = int(round(n / 2.0 - np.sum(h_b)))
    if need > 0:
        order = np.argsort(np.abs(h - 0.5))
        for i in range(need):
            h_b[order[i]] = 1.0
    elif need < 0:
        order = np.argsort(np.abs(h - 0.5))
        for i in range(-need):
            h_b[order[i]] = 0.0
    return project_feasible(h_b, n)

def interp_n(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    xo = np.linspace(0, 2, n_old, endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), n_new)

def flip_search(h, n, rounds=60, candidates=15):
    h = project_feasible(h, n)
    c5 = compute_c5(h)
    for _ in range(rounds):
        improved = False
        idx_lo = np.argsort(h)[:candidates]
        idx_hi = np.argsort(h)[::-1][:candidates]
        for i in idx_lo:
            if h[i] > 0.01:
                old = h[i]
                h[i] = 0.0
                h2 = project_feasible(h, n)
                c5_2 = compute_c5(h2)
                if c5_2 < c5 - 1e-10:
                    h = h2.copy()
                    c5 = c5_2
                    improved = True
                    break
                h[i] = old
        if improved:
            continue
        for i in idx_hi:
            if h[i] < 0.99:
                old = h[i]
                h[i] = 1.0
                h2 = project_feasible(h, n)
                c5_2 = compute_c5(h2)
                if c5_2 < c5 - 1e-10:
                    h = h2.copy()
                    c5 = c5_2
                    improved = True
                    break
                h[i] = old
        if not improved:
            break
    return h, c5

def optimize_sequence(h, n, t0, budget_s, steps):
    h_curr = project_feasible(h, n)
    for beta, mi, fwl in steps:
        if time.time() - t0 > budget_s:
            break
        r = _slsqp(h_curr, n, beta, mi, 1e-12, fwl)
        if r is not None:
            h_curr = r
    return h_curr

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf

    if 'initial_h_values' in globals() and initial_h_values is not None:
        n0 = len(initial_h_values)
        h_start = project_feasible(initial_h_values.copy(), n0)
    else:
        n0 = 160
        h_start = np.full(n0, 0.5)

    # Phase 1: Multi-beta with flatness (exploration)
    steps1 = [(5, 60, 0.2), (20, 80, 0.1), (100, 100, 0.05), (500, 100, 0.0)]
    h_p = optimize_sequence(h_start, n0, t0, budget_s * 0.22, steps1)
    c5 = compute_c5(h_p)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h_p.copy()

    # Phase 2: High-precision polish (exploitation)
    if time.time() - t0 < budget_s * 0.5:
        steps2 = [(1000, 200, 0.0), (5000, 300, 0.0), (20000, 400, 0.0)]
        h_p2 = optimize_sequence(best_h, n0, t0, budget_s * 0.5, steps2)
        c5_2 = compute_c5(h_p2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h_p2

    # Phase 3: Binary round + reoptimize
    if time.time() - t0 < budget_s * 0.58:
        h_b = try_binary(best_h, n0)
        steps3 = [(50, 100, 0.05), (500, 150, 0.0), (3000, 200, 0.0)]
        h_p3 = optimize_sequence(h_b, n0, t0, budget_s * 0.58, steps3)
        c5_3 = compute_c5(h_p3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h_p3

    # Phase 4: Perturbation restarts
    n_best = len(best_h)
    for i in range(30):
        if time.time() - t0 > budget_s * 0.75:
            break
        if i < 10:
            sig = 0.015
        elif i < 20:
            sig = 0.04
        else:
            sig = 0.08
        h_pert = best_h + rng.randn(n_best) * sig
        h_pert = project_feasible(h_pert, n_best)
        steps4 = [(30, 60, 0.05), (300, 80, 0.0), (2000, 100, 0.0)]
        h_p4 = optimize_sequence(h_pert, n_best, t0, budget_s * 0.75, steps4)
        c5_4 = compute_c5(h_p4)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h_p4

    # Phase 5: Local flip search
    if time.time() - t0 < budget_s * 0.82:
        h_fl, c5_fl = flip_search(best_h.copy(), n_best, rounds=80, candidates=20)
        if c5_fl < best_c5:
            best_c5 = c5_fl
            best_h = h_fl
        # Re-optimize after flips
        if time.time() - t0 < budget_s * 0.78:
            steps5 = [(500, 200, 0.0), (5000, 300, 0.0)]
            h_p5 = optimize_sequence(best_h, n_best, t0, budget_s * 0.78, steps5)
            c5_5 = compute_c5(h_p5)
            if c5_5 < best_c5:
                best_c5 = c5_5
                best_h = h_p5
            h_fl2, c5_fl2 = flip_search(best_h.copy(), n_best, rounds=40, candidates=15)
            if c5_fl2 < best_c5:
                best_c5 = c5_fl2
                best_h = h_fl2

    # Phase 6: Try lower resolution
    if time.time() - t0 < budget_s * 0.45 and n0 > 160:
        n_alt = max(128, n0 // 2)
        h_alt = interp_n(best_h, n_alt)
        steps6 = [(20, 80, 0.1), (200, 120, 0.0), (1000, 150, 0.0), (5000, 200, 0.0)]
        h_p6 = optimize_sequence(h_alt, n_alt, t0, budget_s * 0.55, steps6)
        c5_6 = compute_c5(h_p6)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h_p6

    # Phase 7: Ultimate polish
    if time.time() - t0 < budget_s * 0.08 and best_h is not None:
        n_fin = len(best_h)
        r = _slsqp(best_h, n_fin, 100000, 500, 1e-14, 0.0)
        if r is not None:
            c5_f = compute_c5(r)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = r
        h_fl3, c5_fl3 = flip_search(best_h.copy(), n_fin, rounds=50, candidates=20)
        if c5_fl3 < best_c5:
            best_c5 = c5_fl3
            best_h = h_fl3

    # Final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)

    try:
        val = evaluate_erdos_solution(best_h)
        if val < final_c5:
            final_c5 = val
    except Exception:
        pass

    return (best_h, final_c5, n_final)
