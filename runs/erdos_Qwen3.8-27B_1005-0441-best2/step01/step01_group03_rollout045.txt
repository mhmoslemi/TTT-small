def multi_lag_gradient(h, n, top_k=5):
    """Compute weighted gradient from top-k worst lags."""
    dx = 2.0 / n
    ov = np.correlate(h, 1.0 - h, mode='full') * dx
    # Get top k lag indices
    k_idx = np.argsort(ov)[-top_k:]
    k_idx = k_idx[k_idx != (n-1)]  # skip lag 0 (always 0 for binary, small for continuous)
    
    g = np.zeros(n)
    for kk in k_idx:
        lag = int(kk) - (n - 1)
        # Gradient of overlap at this lag
        for j in range(n):
            # h[j] as first factor: (1-h[j+lag]) if valid
            jp = j + lag
            if 0 <= jp < n:
                g[j] += (1.0 - h[jp])
            # h[j] as second factor: -h[j-lag] if valid
            jm = j - lag
            if 0 <= jm < n:
                g[j] -= h[jm]
    return g * dx, ov

    g = np.zeros(n)
    # h[j] as first factor: contributes (1-h[j+lag]) for valid j
    if lag >= 0:
        # j+lag < n → j < n-lag
        # j ≥ 0

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def to_binary(h, n, rng):
    h = np.round(h).clip(0, 1)
    target = n // 2
    s = int(np.sum(h))
    if s > target:
        ones = np.where(h == 1)[0]
        rng.shuffle(ones)
        for idx in ones[:s - target]:
            h[idx] = 0
    elif s < target:
        zeros = np.where(h == 0)[0]
        rng.shuffle(zeros)
        for idx in zeros[:target - s]:
            h[idx] = 1
    return h

def grad_lag(h, lag, n):
    g = np.zeros(n)
    if lag >= 0:
        g[:n - lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n - lag]
    else:
        a = -lag
        g[a:] += (1.0 - h[:n - a])
        g[:n - a] -= h[a:]
    return g
