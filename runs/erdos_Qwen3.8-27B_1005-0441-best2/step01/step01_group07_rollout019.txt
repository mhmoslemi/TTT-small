    t0, dl = time.time(), time.time() + budget
    while time.time() < dl - 0.3:
        g = grad_smooth(h, dx, tau)
        gn = np.linalg.norm(g) + 1e-12
        v = beta * v - lr * g / gn * 0.5
        h = project_feasible(h + v, ts)
        c = compute_c5(h)
        if c < bc - 1e-10:
            bh, bc, ni = h.copy(), c, 0
        else:
            ni += 1
        tau = min(tau * 1.007, 200.0)
        if ni > 100:
            lr *= 0.92
        if ni > 400:
            h = project_feasible(bh + rng.randn(n) * 0.025, ts)
            v[:], lr, tau, ni = 0.0, lr0, tau0, 0
    return bh, bc

def bin_polish(h0, budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h0)
    ts = int(round(n / 2.0))
    h = (h0 >= 0.5).astype(float)
    d = ts - int(h.sum())
    for _ in range(abs(d)):
        if d > 0:
            h[np.argmin(h)] = 1.0
        else:
            h[np.argmax(h)] = 0.0
    bh, bc = h.copy(), compute_c5(h)
    t0, dl = time.time(), time.time() + budget
    while time.time() < dl - 0.3:
        oi = np.where(h == 1)[0]
        oz = np.where(h == 0)[0]
        if len(oi) == 0 or len(oz) == 0:
            break
        i = oi[rng.randint(len(oi))]
        j = oz[rng.randint(len(oz))]
        h[i], h[j] = 0.0, 1.0
        c = compute_c5(h)
        if c < bc - 1e-12:
            bh, bc = h.copy(), c
        else:
            h[i], h[j] = 1.0, 0.0
    return bh, bc

def coord_descent(h0, budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h0)
    ts = n / 2.0
    h = project_feasible(h0.copy(), ts)
    bh, bc = h.copy(), compute_c5(h)
    t0, dl = time.time(), time.time() + budget
    order = np.arange(n)
    it = 0
    while time.time() < dl - 0.3:
        if it >= n:
            order = rng.permutation(n)
        j = order[it % n]
        it += 1
        best_t, best_c = h[j], compute_c5(h)
        for trial in (0.0, 0.2, 0.5, 0.8, 1.0):
            h[j] = trial
            h = project_feasible(h, ts)
            c = compute_c5(h)
            if c < best_c:
                best_c, best_t = c, trial
        h[j] = best_t
        h = project_feasible(h, ts)
        if best_c < bc - 1e-12:
            bh, bc = h.copy(), best_c
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h70 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h70 = np.full(70, 0.5)
    bh, bc, bn = h70.copy(), compute_c5(h70), 70

    # Phase 1: smooth gradient at n=70
    tr = dl - time.time() - 160
    if tr > 40:
        h, c = smooth_opt(h70, min(tr, 320), seed, tau0=3.0, lr0=0.045)
        if c < bc:
            bh, bc, bn = h, c, 70

    # Phase 2: smooth gradient at n=140 (interpolated)
    tr = dl - time.time() - 120
    if tr > 40:
        h140 = np.interp(np.linspace(0, 69, 140), np.arange(70), bh)
        h140 = project_feasible(h140, 70.0)
        h, c = smooth_opt(h140, min(tr, 280), seed + 100, tau0=3.0, lr0=0.04)
        if c < bc:
            bh, bc, bn = h, c, 140

    # Phase 3: coordinate descent polish
    tr = dl - time.time() - 50
    if tr > 20:
        h, c = coord_descent(bh, min(tr, 60), seed + 300)
        if c < bc:
            bh, bc, bn = h, c, bn

    # Phase 4: binary polish
    tr = dl - time.time() - 40
    if tr > 15:
        h, c = bin_polish(bh, min(tr, 50), seed + 400)
        if c < bc:
            bh, bc, bn = h, c, bn

    # Phase 5: random restarts with smooth opt
    for i in range(6):
        tr = dl - time.time() - 10
        if tr < 15:
            break
        rt = np.random.RandomState(seed + 700 + i * 13)
        noise = 0.02 + 0.01 * i
        ht = project_feasible(bh + rt.randn(bn) * noise, bn / 2.0)
        alloc = min(tr / 7, 55)
        h, c = smooth_opt(ht, alloc, seed + 2000 + i, tau0=2.0 + i, lr0=0.03)
        if c < bc:
            bh, bc, bn = h, c, bn

    # Phase 6: final small-step polish
    tr = dl - time.time() - 1
    if tr > 5:
        h, c = smooth_opt(bh, min(tr, 20), seed + 9999, tau0=20.0, lr0=0.012, beta=0.5)
        if c < bc:
            bh, bc = h, c

    # Phase 7: final binary check
    tr = dl - time.time() - 0.5
    if tr > 8:
        h, c = bin_polish(bh, min(tr, 15), seed + 5000)
        if c < bc:
            bh, bc, bn = h, c, bn

    bc = compute_c5(bh)
    return bh, bc, bn
