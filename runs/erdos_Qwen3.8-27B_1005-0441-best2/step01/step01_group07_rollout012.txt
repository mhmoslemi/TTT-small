    positions = sorted([int((i + 0.5) / ts * n * 0.5 + rng.randn() * 1.5) if i < ts else int(n - 1 - (i - ts + 0.5) / ts * n * 0.5 + rng.randn() * 1.5) for i in range(ts)])
    for i in range(ts):
        h[min(max(positions[i], 0), n - 1)] = 1.0
    excess = int(h.sum()) - ts
    for _ in range(excess):
        h[np.argmax(h)] = 0.0
    deficit = ts - int(h.sum())
    for _ in range(deficit):
        h[np.argmin(h)] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    best_h, best_c5, best_n = None, 1.0, 0

    n_list = [70, 60, 80, 66, 72, 90, 84]
    if 'initial_h_values' in globals() and initial_h_values is not None:
        n_main = len(initial_h_values)
        if n_main not in n_list:
            n_list.insert(0, n_main)
        h_parent = np.asarray(initial_h_values, dtype=float)
    else:
        h_parent = None

    for ni, n in enumerate(n_list):
        t_rem = deadline - time.time() - 5
        if t_rem < 15:
            break
        if ni == 0:
            t_alloc = min(t_rem * 0.4, 380)
        else:
            t_alloc = min(t_rem / max(1, len(n_list) - ni), 100)
        if t_alloc < 8:
            continue
        ts = n / 2.0

        starts = []
        if ni == 0 and h_parent is not None:
            starts.append(project_feasible(h_parent.copy(), ts))
        starts.append(project_feasible(structured_init(n, seed + n), ts))
        starts.append(project_feasible(0.5 + 0.15 * np.random.RandomState(seed + n * 7).randn(n), ts))

        t_phase = t_alloc / 3.0
        for si, h_s in enumerate(starts):
            t_rem2 = deadline - time.time() - 3
            tc = min(t_phase, t_rem2 * 0.5)
            if tc < 3:
                continue
            hc, c5c = optimize_cont(h_s, tc, seed + n * 31 + si * 7,
                                     lr0=0.05, tau0=25.0 + si * 20)
            tb = min(t_phase * 0.6, deadline - time.time() - 2)
            if tb > 2:
                hb, c5b = bin_local(hc, tb, seed + n * 31 + si * 7 + 500)
                if c5b < c5c:
                    hc, c5c = hb, c5b
            if c5c < best_c5:
                best_h, best_c5, best_n = hc, c5c, n

    t_rem = deadline - time.time() - 1
    if t_rem > 5 and best_h is not None:
        ts = best_n // 2
        hp, c5p = bin_local(best_h, min(t_rem, 25), seed + 77777, use_2swap=True)
        if c5p < best_c5:
            best_h, best_c5 = hp, c5p

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
