    if best_h is not None and _el() < budget_s * 0.99:
        n = len(best_h)
        t_limit = min(_el() + (budget_s * 0.99 - _el()), budget_s * 0.99)
        h_sg, v_sg = _subgrad_refine(best_h, n, max_iter=10000,
                                     time_limit=budget_s * 0.99, t0=t0)
        if v_sg < best_val:
            best_val = v_sg
            best_h = h_sg

    # Phase 6: Try perturbation restart if time remains
    if best_h is not None and _el() < budget_s * 0.96:
        n = len(best_h)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _eq_n, 'jac': _eq_j, 'args': (n,)}]
        for pert_scale in [0.02, 0.01]:
            if _el() > budget_s * 0.97:
                break
            h_p = _proj(best_h + pert_scale * rng.randn(n), n)
            try:
                res = _min(_smoobj, h_p, args=(0.002,), jac=_smograd_h,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-13})
                h_c = _proj(res.x, n)
                if _c5(h_c) < best_val:
                    best_val = _c5(h_c)
                    best_h = h_c
            except Exception:
                pass

    best_h = _proj(best_h, len(best_h))
    best_val = _c5(best_h)
    return (best_h, best_val, len(best_h))
