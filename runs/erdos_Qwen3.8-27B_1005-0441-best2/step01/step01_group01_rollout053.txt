                           constraints=[{'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac}],
                           options={'maxiter': 3000, 'ftol': 1e-16})
            h_try = project_feasible(res.x, 80)
            c5_try = compute_c5(h_try)
            if c5_try < c5_best - 1e-10:
                c5_best = c5_try
                h_best = h_try.copy()
                n = 80
                bounds = bounds_80

    # Phase 3: Subgradient polish
    if time.time() - t0 < budget_s * 0.42:
        for ss in [0.02, 0.01, 0.005]:
            if time.time() - t0 > budget_s * 0.44:
                break
            h_sg, c5_sg = subgradient_step(h_best, n, ss, 2000)
            if c5_sg < c5_best - 1e-10:
                c5_best = c5_sg
                h_best = h_sg.copy()
            # Follow with SLSQP
            res = minimize(obj_func, h_best.copy(), args=(1000.0,),
                           method='SLSQP', bounds=bounds, constraints=constraint,
                           options={'maxiter': 2000, 'ftol': 1e-15})
            h_try = project_feasible(res.x, n)
            c5_try = compute_c5(h_try)
            if c5_try < c5_best - 1e-10:
                c5_best = c5_try
                h_best = h_try.copy()

    # Phase 4: L-BFGS-B with large penalty (fast per-iteration)
    if time.time() - t0 < budget_s * 0.52:
        for pen in [1000.0, 10000.0, 100000.0]:
            if time.time() - t0 > budget_s * 0.54:
                break
            res = minimize(obj_func_pen, h_best.copy(), args=(500.0, pen),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 800, 'ftol': 1e-17, 'gtol': 1e-12})
            h_try = project_feasible(res.x, n)
            c5_try = compute_c5(h_try)
            if c5_try < c5_best - 1e-10:
                c5_best = c5_try
                h_best = h_try.copy()

    # Phase 5: Swap perturbations + SLSQP
    cnt = 0
    while time.time() - t0 < budget_s - 150 and cnt < 40:
        cnt += 1
        n_swaps = 2 + int(6 * rng.random())
        h_p = swap_perturb(h_best, n, rng, n_swaps)
        res = minimize(obj_func, h_p.copy(), args=(600.0,),
                       method='SLSQP', bounds=bounds, constraints=constraint,
                       options={'maxiter': 2000, 'ftol': 1e-15})
        h_try = project_feasible(res.x, n)
        c5_try = compute_c5(h_try)
        if c5_try < c5_best - 1e-10:
            c5_best = c5_try
            h_best = h_try.copy()

    # Phase 6: Random Gaussian perturbations
    cnt = 0
    while time.time() - t0 < budget_s - 120 and cnt < 25:
        cnt += 1
        scale = 0.005 + 0.04 * rng.random()
        h_p = h_best + scale * rng.standard_normal(n)
        h_p = project_feasible(h_p, n)
        res = minimize(obj_func, h_p.copy(), args=(800.0,),
                       method='SLSQP', bounds=bounds, constraints=constraint,
                       options={'maxiter': 1500, 'ftol': 1e-15})
        h_try = project_feasible(res.x, n)
        c5_try = compute_c5(h_try)
        if c5_try < c5_best - 1e-10:
            c5_best = c5_try
            h_best = h_try.copy()

    # Phase 7: Alternative structured starts
    if time.time() - t0 < budget_s - 100:
        for kind in [1, 2, 4, 5]:
            if time.time() - t0 > budget_s * 0.82:
                break
            h_alt = make_pattern(n, kind, seed + 7 * kind)
            for alpha in [300.0, 1500.0]:
                if time.time() - t0 > budget_s * 0.84:
                    break
                res = minimize(obj_func, h_alt.copy(), args=(alpha,),
                               method='SLSQP', bounds=bounds, constraints=constraint,
                               options={'maxiter': 2000, 'ftol': 1e-15})
                h_try = project_feasible(res.x, n)
                c5_try = compute_c5(h_try)
                if c5_try < c5_best - 1e-10:
                    c5_best = c5_try
                    h_best = h_try.copy()

    # Phase 8: Final ultra-high-alpha SLSQP polish
    if time.time() - t0 < budget_s - 20:
        for alpha in [5000.0, 10000.0, 20000.0]:
            if time.time() - t0 > budget_s - 10:
                break
            res = minimize(obj_func, h_best.copy(), args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=constraint,
                           options={'maxiter': 3000, 'ftol': 1e-16})
            h_try = project_feasible(res.x, n)
            c5_try = compute_c5(h_try)
            if c5_try < c5_best - 1e-10:
                c5_best = c5_try
                h_best = h_try.copy()

    # Final subgradient micro-polish
    if time.time() - t0 < budget_s - 5:
        h_sg, c5_sg = subgradient_step(h_best, n, 0.003, 500)
        if c5_sg < c5_best - 1e-10:
            c5_best = c5_sg
            h_best = h_sg.copy()

    h_best = project_feasible(h_best, len(h_best))
    c5_best = compute_c5(h_best)
    return h_best, c5_best, len(h_best)
