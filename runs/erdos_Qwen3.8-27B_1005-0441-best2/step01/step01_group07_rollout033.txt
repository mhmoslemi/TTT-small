import numpy as np
import time, math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_binary(h_float, n):
    """Convert float h to binary with exactly n//2 ones."""
    target = n // 2
    h = (h_float >= 0.5).astype(float)
    diff = target - int(h.sum())
    if diff > 0:
        zeros = np.where(h == 0)[0]
        for _ in range(diff):
            h[np.random.randint(len(zeros))] = 1.0  # not random, just first
        # Actually need to be more careful
    # Simpler: sort by threshold
    h = np.zeros(n)
    order = np.argsort(-h_float)
    h[order[:target]] = 1.0
    return h

def sa_search(h, t_budget, seed, T0=2.0, T_min=0.1):

    rng = np.random.RandomState(seed)
    n = len(h)

    h = h.copy()
    best_h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(corr.max() * dx)
    t0 = time.time()
    deadline = t0 + t_budget
    
    T = T0
    no_improve = 0
    iters = 0
    
    while time.time() < deadline - 0.2:
        iters += 1
        r = rng.random()
        
        if r < 0.65:
            # Single swap
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            new_corr = compute_corr(h)
            new_c5 = float(new_corr.max() * dx)
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            elif rng.random() < math.exp(min((best_c5 - new_c5) * 300.0 / T, 50.0)):

            else:

            corr = new_corr
        elif r < 0.85:

            zeros = np.where(h == 0.0)[0]

            i1, i2 = rng.choice(len(ones), 2, replace=False)

def hill_climb(h, t_budget, seed):
    """Pure hill climbing - only accept improvements."""
    rng = np.random.RandomState(seed)
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    corr = compute_corr(h)

    while time.time() < deadline - 0.2:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

            corr = new_corr

import numpy as np
import time, math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_binary_from_float(hf, n):
    target = n // 2
    order = np.argsort(-hf)
    h = np.zeros(n)
    h[order[:target]] = 1.0
    return h

def sa_search(h, t_budget, seed, T0=2.0, T_min=0.03):
    rng = np.random.RandomState(seed)
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(corr.max() * dx)
    t0 = time.time()
    deadline = t0 + t_budget
    T = T0
    no_improve = 0
    iters = 0

    while time.time() < deadline - 0.15:
        iters += 1
        r = rng.random()

        if r < 0.62:
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            new_corr = compute_corr(h)
            new_c5 = float(new_corr.max() * dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            elif rng.random() < math.exp(min((best_c5 - new_c5) * 400.0 / T, 0.0)):
                pass
            else:
                h[i], h[j] = 1.0, 0.0
                continue
            corr = new_corr
        elif r < 0.82:
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                continue
            p = rng.permutation(len(ones))[:2]
            q = rng.permutation(len(zeros))[:2]
            a1, a2 = ones[p[0]], ones[p[1]]
            b1, b2 = zeros[q[0]], zeros[q[1]]
            if a1 == a2 or b1 == b2 or len(set((a1,a2,b1,b2))) < 4:
                continue
            h[a1], h[a2], h[b1], h[b2] = 0.0, 0.0, 1.0, 1.0
            new_corr = compute_corr(h)
            new_c5 = float(new_corr.max() * dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            elif rng.random() < math.exp(min((best_c5 - new_c5) * 400.0 / T, 0.0)):
                pass
            else:
                h[a1], h[a2], h[b1], h[b2] = 1.0, 1.0, 0.0, 0.0
                continue
            corr = new_corr
        else:
            top_k = min(5, len(corr))
            top_lags = np.argsort(corr)[::-1][:top_k]
            wl = top_lags[rng.randint(top_k)]
            L = wl - (n - 1)
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            did_move = False
            if L >= 0 and L < n:
                cands = [m for m in range(n - L) if h[m] > 0.5 and h[m+L] < 0.5]
            elif L < 0 and L > -n:
                La = -L
                cands = [m for m in range(La, n) if h[m] > 0.5 and h[m-La] < 0.5]
            else:
                cands = []
            if cands:
                m = cands[rng.randint(len(cands))]
                z = zeros[rng.randint(len(zeros))]
                if z != m:
                    h[m], h[z] = 0.0, 1.0
                    new_corr = compute_corr(h)
                    new_c5 = float(new_corr.max() * dx)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                        no_improve = 0
                    elif rng.random() < math.exp(min((best_c5 - new_c5) * 400.0 / T, 0.0)):
                        pass
                    else:
                        h[m], h[z] = 1.0, 0.0
                        continue
                    corr = new_corr
                    did_move = True
            if not did_move:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i], h[j] = 0.0, 1.0
                new_corr = compute_corr(h)
                new_c5 = float(new_corr.max() * dx)

            no_improve = 0
        else:
            T = max(T * 0.99997, T_min)

    return best_h, best_c5

def hill_climb(h, t_budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(corr.max() * dx)
    t0 = time.time()
    deadline = t0 + t_budget
    stuck = 0
    while time.time() < deadline - 0.1:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_corr = compute_corr(h)
        new_c5 = float(new_corr.max() * dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            corr = new_corr
            stuck = 0
        else:
            h[i], h[j] = 1.0, 0.0
            stuck += 1
            if stuck > 80000:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.asarray(initial_h_values, dtype=float)

        h_init = np.full(70, 0.5)

    best_h = h_init.copy()

    # Phase 1: Long SA from initial
    t_rem = deadline - time.time() - 60
    if t_rem > 40:
        h1, c5_1 = sa_search(h_init, min(t_rem * 0.55, 550), seed, T0=3.0, T_min=0.03)
        if c5_1 < best_c5:
            best_h, best_c5 = h1, c5_1

    # Phase 2: SA from best with different temp

    if t_rem > 30:
        h2, c5_2 = sa_search(best_h, min(t_rem * 0.4, 300), seed + 200, T0=1.0, T_min=0.02)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2

    # Phase 3: ILS with perturbations
    rng_main = np.random.RandomState(seed + 777)
    trial = 0
    while deadline - time.time() > 50:
        t_left = deadline - time.time() - 20
        t_alloc = min(t_left / max(1, 6 - trial % 6), 140)
        if t_alloc < 15:
            t_alloc = 15
        k_pert = rng_main.randint(3, 10) + trial % 4
        h_pert = best_h.copy()
        ones_p = np.where(h_pert == 1.0)[0]
        zeros_p = np.where(h_pert == 0.0)[0]
        for _ in range(k_pert):
            o = ones_p[rng_main.randint(len(ones_p))]
            z = zeros_p[rng_main.randint(len(zeros_p))]
            h_pert[o], h_pert[z] = 0.0, 1.0
            ones_p = np.where(h_pert == 1.0)[0]
            zeros_p = np.where(h_pert == 0.0)[0]
        h3, c5_3 = sa_search(h_pert, t_alloc, seed + 2000 + trial * 13,
                             T0=1.5, T_min=0.02)
        if c5_3 < best_c5:
            best_h, best_c5 = h3, c5_3
        trial += 1

    # Phase 4: Hill climb polish
    t_rem = deadline - time.time() - 1
    if t_rem > 5:
        h4, c5_4 = hill_climb(best_h, min(t_rem, 40), seed + 9999)
        if c5_4 < best_c5:
            best_h, best_c5 = h4, c5_4

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
