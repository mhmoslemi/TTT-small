                corr = nc
                c5_corr = new_max
                improved_this_round += 1
        if improved_this_round == 0:
            break
    
    dx = 2.0 / n
    return h, float(c5_corr * dx)

def soft_max_c5(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    if beta < 1:
        beta = 1
    shifted = beta * (c - m)
    shifted -= shifted.max()
    logsum = np.log(np.sum(np.exp(shifted)))
    return (m + logsum / beta) * dx, float(m * dx)

def constraint_eq(x):
    return np.sum(x) - len(x) / 2.0

def constraint_jac(x):
    return np.ones(len(x))

def optimize_continuous(h_start, n, t0, budget_s, max_frac=0.7):
    h = project_feasible(h_start, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_jac}]
    
    betas = [5, 20, 100, 500, 2000]
    for beta in betas:
        if time.time() - t0 > budget_s * max_frac:
            break
        def obj(x, b=beta):
            val, _ = soft_max_c5(x, n, b)
            return val
        try:
            maxiter = 200 if n <= 200 else 100
            res = minimize(obj, h, method='SLSQP', bounds=bounds,
                          constraints=constraints,
                          options={'maxiter': maxiter, 'ftol': 1e-10})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

def make_sinusoidal_init(n, f=1, phase=0):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * f * x + phase)

def make_block_init(n, nblocks=4):
    h = np.zeros(n)
    block_size = n // nblocks
    for b in range(0, nblocks, 2):
        start = b * block_size
        end = min(start + block_size, n)
        h[start:end] = 1.0
    return project_feasible(h, n)

def batch_local_search(h, corr, ones, zeros, rng, B=100, n=None):
    """Try B random pair swaps, return best improvement if any."""
    if n is None:
        n = len(h)
    max_k = 2 * n - 1
    
    i_batch = ones[rng.randint(len(ones), size=B)]
    j_batch = zeros[rng.randint(len(zeros), size=B)]
    
    best_idx = -1
    best_max = np.inf
    
    for idx in range(B):
        i, j = i_batch[idx], j_batch[idx]
        nc = corr.copy()  # O(2n-1)
        ki_lo = max(0, n - 1 - i)
        ki_hi = min(max_k - 1, n - 1 + i)
        kj_lo = max(0, n - 1 - j)
        kj_hi = min(max_k - 1, n - 1 + j)
        ii = np.arange(ki_lo, ki_hi + 1)
        qi = i + ii - (n - 1)
        nc[ii] -= (1.0 - h[qi])
        ij = np.arange(kj_lo, kj_hi + 1)
        qj = j + ij - (n - 1)
        nc[ij] += (1.0 - h[qj])
        nm = np.max(nc)
        if nm < best_max:
            best_max = nm
            best_idx = idx
    
    if best_idx >= 0 and best_max < np.max(corr) - 1e-12:
        i, j = i_batch[best_idx], j_batch[best_idx]
        # Recompute
        nc = corr.copy()
        ki_lo = max(0, n - 1 - i)
        ki_hi = min(max_k - 1, n - 1 + i)
        kj_lo = max(0, n - 1 - j)
        kj_hi = min(max_k - 1, n - 1 + j)
        ii = np.arange(ki_lo, ki_hi + 1)
        qi = i + ii - (n - 1)
        nc[ii] -= (1.0 - h[qi])
        ij = np.arange(kj_lo, kj_hi + 1)
        qj = j + ij - (n - 1)
        nc[ij] += (1.0 - h[qj])
        return i, j, nc, best_max
    return None

minimize max_k ∫ h(x) (1-h(x+k)) dx

subject to ∫ h = 1, 0≤h≤1

For the continuous case, if h is a step function with values in {0,1}, then h is the indicator of a set A ⊂ [0,2] with measure 1. The overlap at shift k is the measure of A ∩ (A+k)^c within the overlap region.

For a "nice" set A (like a union of intervals), the overlap function is piecewise linear, and we want to minimize its maximum.

The optimal structure for this 1D problem might be related to "balanced words" or "Sturmian words" in combinatorics on words. These are binary sequences with the most balanced possible autocorrelation.
