            npts = rng.randint(3, 8)
            idx = rng.choice(n, size=npts, replace=False)
            h[idx] += rng.randn(npts) * 0.08
            h = _proj_sum(h, n * 0.5)
            C = _corr(h)

    # Phase 2: SLSQP polish
    if time.time() - t0 < budget_s * 0.5:
        polished = _slsqp_polish(best_h, best_h.size, 0.55, t0, budget_s)
        pv = _c5(polished)
        if pv < best_val:
            best_val, best_h = pv, polished

    # Phase 3: pair-swap on polished + try larger n
    for n_try in [n0, 100, 140]:
        if time.time() - t0 > budget_s * 0.78:
            break
        if n_try == n0:
            h = best_h.copy()
        else:
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h = np.interp(x_new, x_old, best_h)
            h = _proj_sum(h, n_try * 0.5)
        C = _corr(h)
        n = n_try
        order = [(i, j) for i in range(n) for j in range(i + 1, n)]
        for sw in range(200):
            if time.time() - t0 > budget_s * 0.82:
                break
            rng.shuffle(order)
            v = _sweep(h, C, n, order, rng)
            if v < best_val:
                best_val, best_h = v, h.copy()
        if time.time() - t0 < budget_s * 0.88:
            polished = _slsqp_polish(best_h, best_h.size, 0.88, t0, budget_s)
            pv = _c5(polished)
            if pv < best_val:
                best_val, best_h = pv, polished

    # Phase 4: final pair-swap polish
    if time.time() - t0 < budget_s * 0.96:
        n = len(best_h)
        h = best_h.copy()
        C = _corr(h)
        order = [(i, j) for i in range(n) for j in range(i + 1, n)]
        for sw in range(500):
            if time.time() - t0 > budget_s * 0.98:
                break
            rng.shuffle(order)
            v = _sweep(h, C, n, order, rng)
            if v < best_val:
                best_val, best_h = v, h.copy()

    best_h = _proj_sum(best_h, len(best_h) * 0.5)
    best_val = _c5(best_h)
    return (best_h, best_val, len(best_h))
