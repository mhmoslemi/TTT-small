    return corr * dx

    return np.max(compute_all_g(h))

def project_sum(h, target):
    """Project h onto {0 ≤ h ≤ 1, sum(h) = target}."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_shift(h, s):
    """Gradient of g_s = sum_i h[i+s](1-h[i]) * dx w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    i_lo = max(0, -s)
    i_hi = min(n-1, n-1-s)
    if i_lo > i_hi:
        return grad
    i = np.arange(i_lo, i_hi + 1)
    j1 = i + s
    np.add.at(grad, j1, 1.0 - h[i])
    np.add.at(grad, i, -h[j1])
    return grad * dx

def coord_descent_sweep(h, g, n, dx, max_sweeps=5, time_limit=None, t0=None):
    """Coordinate descent: for each j, try h[j] = 0, 1 and keep best."""
    best_c5 = np.max(g)
    for sweep in range(max_sweeps):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        improved = False
        for j in range(n):
            v = h[j]
            # Compute delta for all shifts
            # delta[j,s] = (1-h[j-s])*I(0≤j-s<n) - h[j+s]*I(0≤j+s<n)
            s_vals = np.arange(-(n-1), n)
            js = j - s_vals  # j - s
            jp = j + s_vals  # j + s
            mask_js = (js >= 0) & (js < n)
            mask_jp = (jp >= 0) & (jp < n)
            delta = np.zeros(2*n-1)
            delta[mask_js] += (1.0 - h[js[mask_js]])
            delta[mask_jp] -= h[jp[mask_jp]]
            delta *= dx

                c5_new = np.max(g_new)
                if c5_new < best_c5 - 1e-15:
                    h[j] = t
                    g = g_new.copy()
                    best_c5 = c5_new
                    improved = True
                    # Recheck with t=0 or t=1
                    # Actually just continue to next j
                    break
        if not improved:
            break
    return h, g, best_c5

import numpy as np
import time

def compute_all_g(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_shift(h, s, dx):
    n = len(h)
    grad = np.zeros(n)
    i_lo = max(0, -s)
    i_hi = min(n - 1, n - 1 - s)
    if i_lo > i_hi:
        return grad
    i = np.arange(i_lo, i_hi + 1)
    j1 = i + s
    np.add.at(grad, j1, 1.0 - h[i])
    np.add.at(grad, i, -h[j1])
    return grad * dx

def solve_1d_minmax(g, delta, v):
    def compute_at(t):
        vals = g + (t - v) * delta
        mx = vals.max()
        idx = np.where(vals >= mx - 1e-10)[0]
        return mx, idx

        return 0.0

            hi = mid
        elif delta[idx_m].max() < -1e-10:
            lo = mid
        else:
            return mid
    return (lo + hi) / 2.0

def coord_descent(h, g, n, dx, max_sweeps=20, deadline=None, allow_frac=True):
    s_range = np.arange(-(n - 1), n)
    best_c5 = float(np.max(g))
    for sweep in range(max_sweeps):

            break
        improved = False
        for j in range(n):
            v = h[j]
            js_arr = j - s_range
            jp_arr = j + s_range
            mask_js = (js_arr >= 0) & (js_arr < n)
            mask_jp = (jp_arr >= 0) & (jp_arr < n)
            delta = np.zeros(2 * n - 1)
            delta[mask_js] += (1.0 - h[js_arr[mask_js]])
            delta[mask_jp] -= h[jp_arr[mask_jp]]
            delta *= dx
            candidates = [0.0, 1.0]
            if allow_frac:
                t_opt = solve_1d_minmax(g, delta, v)
                if 0.01 < t_opt < 0.99 and abs(t_opt - v) > 1e-6:
                    candidates.append(t_opt)
            best_t = v
            best_c5_j = best_c5
            for t in candidates:
                if abs(t - v) < 1e-10:
                    continue
                g_new = g + (t - v) * delta
                c5_new = np.max(g_new)
                if c5_new < best_c5_j - 1e-14:
                    best_c5_j = c5_new
                    best_t = t
            if best_t != v:

                g[:] = g + (best_t - v) * delta
                best_c5 = best_c5_j
                improved = True
        if not improved:
            break
    return h, g, best_c5

def subgrad_refine(h, n, deadline, n_iters=80):

    target = n / 2.0
    best_h = h.copy()
    g = compute_all_g(h)

        h_new = project_sum(h_new, target)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5
