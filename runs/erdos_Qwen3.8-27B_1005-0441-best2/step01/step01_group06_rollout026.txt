```python
def project_onto_feasible(h, n):
    """Project h onto {h: 0≤h≤1, sum(h)=n/2}"""
    target = n / 2.0
    # Bisection on lambda: h_i = clip(h_i - lambda, 0, 1), find lambda s.t. sum = target
    lo, hi = h.min() - target, h.max()
    for _ in range(60):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2, 0, 1)
    # Fix small numerical issues
    diff = target - h_proj.sum()
    if abs(diff) > 1e-10:
        # Distribute remaining
        mask = (h_proj > 0) & (h_proj < 1)
        if mask.sum() > 0:
            h_proj[mask] += diff / mask.sum()
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj
```

```python
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    # Fix tiny numerical errors
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        mask = (hp > 1e-7) & (hp < 1 - 1e-7)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def subgrad_h(h, n, k):
    """Subgradient of g_k at h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            grad[:m] += (1.0 - h[k:])
            grad[k:] += -h[:m]
    else:
        kp = -k
        m = n - kp
        if m > 0:
            grad[kp:] += (1.0 - h[:m])
            grad[:m] += -h[kp:]
    return grad * dx
```

```python
def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def get_g(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, n):
    h = h.astype(float)
    target = n * 0.5
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    d = target - hp.sum()
    if abs(d) > 1e-11:
        m = (hp > 1e-8) & (hp < 1.0 - 1e-8)
        if m.sum() > 0:
            hp[m] = np.clip(hp[m] + d / m.sum(), 0.0, 1.0)
    return hp

def subgrad_for_shift(h, n, s):
    dx = 2.0 / n
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        if m > 0:
            g[s:] += (1.0 - h[:m])
            g[:m] += -h[s:]
    else:
        p = -s
        m = n - p
        if m > 0:
            g[:m] += (1.0 - h[p:])
            g[p:] += -h[:m]
    return g * dx

def nesterov_opt(h_init, max_iters, t0, budget, top_k=10, mu=0.9):
    n = len(h_init)
    h = project_feasible(h_init.copy(), n)
    v = np.zeros(n)
    g_curr = get_g(h)
    best_c5 = float(g_curr.max())
    best_h = h.copy()
    eta = 0.008
    consecutive_bad = 0

    for it in range(max_iters):
        elapsed = time.time() - t0
        if elapsed > budget * 0.93:
            break

        idx_full = 2 * n - 1
        nk = min(top_k, idx_full)
        idx_top = np.argpartition(-g_curr, nk - 1)[:nk]
        grad = np.zeros(n)
        for idx in idx_top:
            k = int(idx) - (n - 1)
            grad += subgrad_for_shift(h, n, k)
        grad /= nk

        v = mu * v + eta * grad
        h_new = project_feasible(h - v, n)
        g_new = get_g(h_new)
        c5_new = float(g_new.max())
        c5_old = float(g_curr.max())

        if c5_new > c5_old * 1.02 + 1e-10:
            eta *= 0.7
            v = mu * 0.5 * v
            h_new = project_feasible(h - v, n)
            g_new = get_g(h_new)
            c5_new = float(g_new.max())
            consecutive_bad += 1
        else:
            consecutive_bad = 0
            if (it + 1) % 2000 < 500:
                eta *= 1.02

        h = h_new
        g_curr = g_new

        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h.copy()

        if consecutive_bad > 50:
            h = project_feasible(h + 0.025 * np.random.randn(n), n)
            g_curr = get_g(h)
            v = np.zeros(n)
            eta = 0.008
            consecutive_bad = 0

    return best_h, best_c5

def binary_sa(h_init, n, max_rounds, t0, budget, T0=0.008):
    h_bin = (h_init > 0.5).astype(float)
    s = int(h_bin.sum())
    target = n // 2
    if s != target:
        if s > target:
            ones = np.where(h_bin == 1.0)[0]
            to_flip = np.random.choice(ones, s - target, replace=False)
            h_bin[to_flip] = 0.0
        else:
            zeros = np.where(h_bin == 0.0)[0]
            to_flip = np.random.choice(zeros, target - s, replace=False)
            h_bin[to_flip] = 1.0

    c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = c5
    T = T0

    for r in range(max_rounds):
        if time.time() - t0 > budget * 0.94:
            break
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = ones[np.random.randint(len(ones))]
        i0 = zeros[np.random.randint(len(zeros))]
        h_bin[i1] = 0.0
        h_bin[i0] = 1.0
        c5_new = compute_c5(h_bin)

        if c5_new < c5:
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5

        else:

                c5 = c5_new
            else:
                h_bin[i1] = 1.0
                h_bin[i0] = 0.0

    return best_h, best_c5
```
