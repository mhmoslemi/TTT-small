            offspring.append(_feasible(h_mut, n))
            off_sig.append(max(0.005, min(0.14, sigma)))
        off_fit = [compute_c5(offspring[i]) for i in range(n_off)]
        all_h = pop + offspring
        all_fit = fitness + off_fit
        all_sig = sigmas + off_sig
        order = np.argsort(all_fit)
        keep = min(pop_size, len(order))
        pop = [all_h[int(order[i])].copy() for i in range(keep)]
        fitness = [all_fit[int(order[i])] for i in range(keep)]
        sigmas = [all_sig[int(order[i])] for i in range(keep)]
        med_fit = float(np.median(fitness))
        for i in range(keep):
            if fitness[i] < med_fit:
                sigmas[i] = min(0.14, sigmas[i] * 1.12)
            else:
                sigmas[i] = max(0.004, sigmas[i] * 0.88)
        cur_best = int(np.argmin(fitness))
        if fitness[cur_best] < best_val - 1e-10:
            best_val = fitness[cur_best]
            best_h = pop[cur_best].copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 50:
            n_r = max(3, keep // 4)
            for i in range(n_r):
                idx = int(rng.randint(keep // 3, keep))
                pop[idx] = _feasible(best_h + 0.07 * rng.randn(n), n)
                fitness[idx] = compute_c5(pop[idx])
                sigmas[idx] = 0.07
            no_improve = 0
    return best_h, best_val

def _structured_init(n, rng):
    h = np.zeros(n)
    block = max(2, n // 16)
    pos = 0
    on = rng.rand() > 0.5
    while pos < n:
        b = min(block, n - pos)
        if on:
            h[pos:pos+b] = 1.0
        pos += b
        on = not on
    h += 0.08 * rng.randn(n)
    return _feasible(h, n)

def _bi_pulse_init(n, rng):
    h = np.zeros(n)
    n1 = n // 4
    gap = int(n * rng.uniform(0.1, 0.4))
    p1 = int(rng.randint(0, max(1, n - 2 * n1 - gap)))
    p2 = (p1 + n1 + gap) % n
    for i in range(n1):
        h[(p1 + i) % n] = 1.0
        h[(p2 + i) % n] = 1.0
    h += 0.05 * rng.randn(n)
    return _feasible(h, n)

def _binary_init(n, rng):
    h = np.zeros(n)
    n1 = n // 2
    idx = rng.permutation(n)[:n1]
    h[idx] = 1.0
    h += 0.06 * rng.randn(n)
    return _feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(80, 0.5)
    n = len(h0)
    h0 = _feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Polish initial
    if budget_s - (time.time() - t0) > 120:
        tm = min(130, budget_s - (time.time() - t0) - 50)
        ho, cv = _slsqp_polish(h0.copy(), n, tm)
        if cv < best_c5:
            best_h, best_c5 = ho, cv

    # Phase 2: Multi-start diverse inits
    inits = [_structured_init(n, rng), _bi_pulse_init(n, rng),
             _feasible(h0 + 0.12 * rng.randn(n), n),
             _fe_asible_n_init(n, rng), _binary_init(n, rng)]
    if budget_s - (time.time() - t0) > 100:
        inits.append(_structured_init(n, rng))
        inits.append(_bi_pulse_init(n, rng))
    for hi in inits:
        if budget_s - (time.time() - t0) < 50:
            break
        tm = min(35, budget_s - (time.time() - t0) - 25)
        if tm < 12:
            break
        ho, cv = _slsqp_polish(hi, n, tm)
        if cv < best_c5:
            best_h, best_c5 = ho, cv

    # Phase 3: ES exploration
    if budget_s - (time.time() - t0) > 250:
        ps = max(25, min(50, int((budget_s - (time.time() - t0)) / 25)))
        tm = min(int((budget_s - (time.time() - t0)) * 0.35), 380)
        he, ce = _es_evolve(best_h.copy(), n, ps, tm, seed + 777)
        if ce < best_c5:
            best_h, best_c5 = he, ce
        if budget_s - (time.time() - t0) > 80:
            ho, cv = _slsqp_polish(best_h.copy(), n, min(80, budget_s - (time.time() - t0) - 30))
            if cv < best_c5:
                best_h, best_c5 = ho, cv

    # Phase 4: Nelder-Mead
    if budget_s - (time.time() - t0) > 50:
        hn, cn = _nm_polish(best_h.copy(), n, min(45, budget_s - (time.time() - t0) - 15))
        if cn < best_c5:
            best_h, best_c5 = hn, cn

    # Phase 5: Greedy CD
    if budget_s - (time.time() - t0) > 30:
        hg, cg = _greedy_cd(best_h.copy(), n, min(25, budget_s - (time.time() - t0) - 10))
        if cg < best_c5:
            best_h, best_c5 = hg, cg

    # Phase 6: Random restarts
    rcount = 0
    while budget_s - (time.time() - t0) > 50 and rcount < 20:
        scale = 0.02 + 0.06 * (rcount % 6) / 5.0
        hp = _feasible(best_h + scale * rng.randn(n), n)
        tm = min(25, budget_s - (time.time() - t0) - 20)
        if tm < 12:
            break
        ho, cv = _slsqp_polish(hp, n, tm)
        if cv < best_c5:
            best_h, best_c5 = ho, cv
        rcount += 1

    # Phase 7: Final polish
    if budget_s - (time.time() - t0) > 25:
        ho, cv = _slsqp_polish(best_h.copy(), n, max(12, budget_s - (time.time() - t0) - 5),
                               temps=[0.001, 0.0004, 0.00015, 0.00008])
        if cv < best_c5:
            best_h, best_c5 = ho, cv
    if budget_s - (time.time() - t0) > 15:
        hg, cg = _greedy_cd(best_h.copy(), n, max(8, budget_s - (time.time() - t0) - 5))
        if cg < best_c5:
            best_h, best_c5 = hg, cg

    best_h = _feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n

def _fe_asible_n_init(n, rng):
    h = np.full(n, 0.5)
    for i in range(n):
        h[i] = float(rng.randint(0, 2))
    return _feasible(h, n)
