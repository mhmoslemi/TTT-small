import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_all_lags(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    # Fine adjustment
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        # Adjust interior points
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += resid / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += resid / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_grad(h, alpha):
    """Gradient of smooth max objective w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500, 500)
    exps = np.exp(shifted)
    weights = exps / np.sum(exps)  # softmax weights
    
    # Gradient of smooth max = weighted average of gradients of each lag
    # d/dh[i] of lag at shift m:
    # lag[m] = sum_n h[n+m]*(1-h[n]) * dx (for valid n)
    # d/dh[i] = dx * [(1-h[i-m]) if valid + (-h[i+m]) if valid]
    grad = np.zeros(n)
    for k in range(2*n-1):
        if weights[k] < 1e-10:
            continue
        s = k - (n-1)  # actual shift
        # h[i] appears as h[n+m] when n = i-s, valid if 0≤i-s<n and 0≤i<n
        # contribution: (1-h[i-s]) for valid i
        lo1 = max(0, s)
        hi1 = min(n, n + s)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1)
            grad[idx] += weights[k] * dx * (1.0 - h[idx - s])
        # h[i] appears in (1-h[n]) when n = i+s, valid if 0≤i+s<n and 0≤i<n
        # contribution: -h[i+s] for valid i
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2)
            grad[idx] -= weights[k] * dx * h[idx + s]
    return grad

    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:

    c5_best = compute_c5(h_best)
    
    # Phase 1: Custom projected gradient with smooth max
    # Using adaptive step size (backtracking line search)
    for alpha in [50, 100, 200, 500, 1000]:
        for it in range(500):

            if el > budget_s * 0.40:

            g = smooth_max_grad(h_best, alpha)
            # Project gradient to satisfy sum constraint

            local_max = smooth_max(h_best, alpha)
            # Backtracking line search
            lr = 0.1
            for trial in range(30):

                h_new = project_feasible(h_new, n)
                obj_new = smooth_max(h_new, alpha)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += resid / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max_val(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -600.0, 600.0)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)

    total = np.sum(exps)
    weights = exps / total
    grad = np.zeros(n)
    for k in range(2 * n - 1):

        s = k - (n - 1)

            grad[idx] -= wk * dx * h[idx + s]

        if lo2 < hi2:
            idx = np.arange(lo2, hi2)
            grad[idx] -= h[idx + s]
