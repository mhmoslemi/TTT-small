def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()

def coord_descent(h, n, t0, budget_s, time_frac=0.85):

    Nh = 1.0 - h
    corr = np.correlate(h, Nh, mode="full").copy()

            old_h = h[i]
            id_minus = i - d_array
            id_plus = i + d_array
            alpha = ((id_minus >= 0) & (id_minus < n)).astype(float)
            beta = ((id_plus >= 0) & (id_plus < n)).astype(float)
            h_im = np.where(alpha > 0, h[np.clip(id_minus, 0, n - 1)], 0.0)

        shifted -= shifted.max()
        soft = (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

            best_c5 = c5_b
            best_h = h_b

    # Phase 6: SLSQP final polish
    if time.time() - t0 < budget_s * 0.92:
        h_sp, c5_sp = slsqp_polish(best_h, n, t0, budget_s, beta=5000)
        if c5_sp < best_c5:
            best_c5 = c5_sp
            best_h = h_sp

    # Phase 7: A few random perturbations
    for rs in range(6):
        if time.time() - t0 > budget_s * 0.96:
            break
        sigma = 0.02 + 0.02 * rs
        h_pert = project_feasible(best_h + rng.randn(n) * sigma, n)
        h_r, c5_r = coord_descent(h_pert, n, t0, budget_s, 0.96)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r

    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    try:
        val = evaluate_erdos_solution(best_h)
        if val < final_c5:
            final_c5 = val
    except Exception:
        pass

    return (best_h, final_c5, n)
