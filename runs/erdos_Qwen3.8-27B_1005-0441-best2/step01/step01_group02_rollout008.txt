```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(h, beta, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    sm = mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta
    s = h.sum() - n / 2.0
    return sm * dx + penalty * s * s

def smooth_grad(h, beta, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    w /= w.sum()
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-14:
            continue
        d = (n - 1) - i
        lo1 = max(0, -d)
        hi1 = min(n, n - d)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + d:hi1 + d])
        lo2 = max(0, d)
        hi2 = min(n, d + n)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2 - d:hi2 - d]
    s = h.sum() - n / 2.0
    grad = grad * dx + penalty * 2.0 * s * np.ones(n)
    return grad

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        h = np.clip(h + (target - s) / n, 0.0, 1.0)
    return h

def optimize_lbfgs(h0, n, beta, penalty, maxiter, t_limit):
    t0 = time.time()
    bounds = [(0.0, 1.0)] * n
    res = minimize(smooth_obj, h0, jac=lambda x: smooth_grad(x, beta, penalty),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'gtol': 1e-8, 'iprint': -1})
    return project(res.x, n)

def local_search(h, n, t_limit):
    """Pair-swap and snap local search"""
    t0 = time.time()
    h = h.copy()
    best = compute_c5(h)
    improved = True
    while improved and time.time() - t0 < t_limit:
        improved = False
        for i in range(n):
            for j in range(i+1, n):
                # Try transferring from i to j or j to i
                for delta in [0.05, 0.02, 0.01]:
                    h2 = h.copy()
                    h2[i] -= delta
                    h2[j] += delta
                    if 0 <= h2[i] <= 1 and 0 <= h2[j] <= 1:

                        if c2 < best - 1e-10:
                            h = h2
                            best = c2
                            improved = True
                            break
                    h2 = h.copy()
                    h2[i] += delta
                    h2[j] -= delta
                    if 0 <= h2[i] <= 1 and 0 <= h2[j] <= 1:
                        c2 = compute_c5(h2)
                        if c2 < best - 1e-10:
                            h = h2
                            best = c2
                            improved = True
                            break
    return h, best
```

```python
            break

        free = (h > 1e-10) & (h < 1 - 1e-10)
        if not free.any():
            break
        adj = (target - s) / free.sum()
        h[free] += adj
        h = np.clip(h, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode="full")

        lo2 = max(0, d)
        hi2 = min(n, d + n)
        if lo2 < hi2:
            grad[lo2:hi2] += wi * (-h[lo2 - d:hi2 - d])
    return grad * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not free.any():
            break
        h[free] += (target - s) / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def sc5_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta) * dx

def sc5_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    w /= w.sum()
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-14:
            continue
        d = (n - 1) - i
        lo1 = max(0, -d)
        hi1 = min(n, n - d)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1+d:hi1+d])
        lo2 = max(0, d)
        hi2 = min(n, d + n)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2-d:hi2-d]
    return grad * dx

def lbfgs_run(h0, n, beta, penalty, maxiter):
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        lambda x: sc5_val(x, beta) + penalty * (x.sum() - n/2.0)**2,
        h0,
        jac=lambda x: sc5_grad(x, beta) + 2*penalty*(x.sum()-n/2.0)*np.ones(n),
        method='L-BFGS-B', bounds=bounds,
        options={'maxiter': maxiter, 'gtol': 1e-10, 'iprint': -1}
    )
    return project_sum(res.x, n)

def pair_search(h, n, t_limit):
    t0 = time.time()
    h = h.copy().astype(float)
    best = compute_c5(h)
    improved = True
    while improved and time.time() - t0 < t_limit:
        improved = False
        best_c = compute_c5(h)
        for i in range(n):
            for j in range(i+1, n):
                for delta in (0.02, 0.05, 0.1, 0.5):
                    for sign in (1, -1):
                        h2 = h.copy()
                        h2[i] += sign * delta
                        h2[j] -= sign * delta
                        if h2[i] < -1e-9 or h2[i] > 1+1e-9 or h2[j] < -1e-9 or h2[j] > 1+1e-9:
                            continue
                        h2[i] = np.clip(h2[i], 0, 1)
                        h2[j] = np.clip(h2[j], 0, 1)
                        c2 = compute_c5(h2)
                        if c2 < best_c - 1e-11:
                            h = h2
                            best_c = c2
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        # Snap near-binary values
        for i in range(n):
            if h[i] > 0.98:
                h2 = h.copy()
                h2[i] = 1.0
                diff = 1.0 - h[i]
                # Find j to adjust
                j = np.argmin(np.abs(h2 - 0.5))
                h2[j] -= diff
                h2 = project_sum(h2, n)
                c2 = compute_c5(h2)
                if c2 < best_c - 1e-11:
                    h = h2
                    best_c = c2
                    improved = True
            elif h[i] < 0.02:
                h2 = h.copy()
                h2[i] = 0.0
                diff = h[i]
                j = np.argmin(np.abs(h2 - 0.5))
                h2[j] += diff
                h2 = project_sum(h2, n)
                c2 = compute_c5(h2)
                if c2 < best_c - 1e-11:
                    h = h2
                    best_c = c2
                    improved = True
    return h, best
```

```python
def anneal_search(h, n, t_limit, T0=0.001, T_end=0.0001, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    t0 = time.time()
    h = h.copy().astype(float)
    best = compute_c5(h)
    hb = h.copy()
    cur = best
    n_steps = 0
    while time.time() - t0 < t_limit:
        T = T0 * (T_end/T0) ** (min(1.0, (time.time()-t0)/t_limit))
        # Pick random pair
        i, j = rng.choice(n, 2, replace=False)
        delta = rng.uniform(0.01, 0.3)
        h[i] -= delta
        h[j] += delta
        if h[i] < -1e-9 or h[i] > 1+1e-9 or h[j] < -1e-9 or h[j] > 1+1e-9:
            h[i] += delta
            h[j] -= delta
            continue
        h[i] = np.clip(h[i], 0, 1)
        h[j] = np.clip(h[j], 0, 1)
        c = compute_c5(h)
        n_steps += 1
        if c < cur:
            cur = c
            if c < best:
                best = c
                hb = h.copy()
        elif rng.random() > np.exp(-(c - cur) / max(T, 1e-10)):
            h[i] += delta
            h[j] -= delta
    return hb, best
```
