    best_c5 = compute_c5(h)
    for beta in [5, 20, 80, 300, 1000, 5000, 20000]:
        if time.time() - t0 > time_limit:
            break
        h_new, c5_new = slsqp_round(h, n, beta, 200, t0, time_limit)
        if c5_new < best_c5:
            best = h_new.copy()
            best_c5 = c5_new
        h = h_new
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.full(320, 0.5)

    n = len(h_init)
    h_init = project_feasible(h_init, n)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: Continuous optimization from initial (35% budget)
    h, c5 = optimize_continuous(h_init.copy(), n, t0, budget_s * 0.30)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Binary local search (targeted)
    if time.time() - t0 < budget_s * 0.55:
        h_bin = to_binary(best_h, n)
        h_b, c5_b = targeted_binary_search(h_bin, n, t0, budget_s * 0.55, max_rounds=100)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
        # Continuous from binary
        if time.time() - t0 < budget_s * 0.65:
            h_c = h_b + rng.randn(n) * 0.02
            h_c = project_feasible(h_c, n)
            h2, c5_2 = optimize_continuous(h_c, n, t0, budget_s * 0.65)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()

    # Phase 3: Multiple perturbation restarts
    for i in range(10):
        if time.time() - t0 > budget_s * 0.82:
            break
        sigma = 0.015 + 0.015 * i
        h_pert = best_h + rng.randn(n) * sigma
        h_pert = project_feasible(h_pert, n)
        h, c5 = optimize_continuous(h_pert, n, t0, budget_s * 0.82)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Binary search from best continuous
    if time.time() - t0 < budget_s * 0.90:
        h_bin2 = to_binary(best_h, n)
        h_b2, c5_b2 = targeted_binary_search(h_bin2, n, t0, budget_s * 0.90, max_rounds=60)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()

    # Phase 5: Final high-precision polish
    if time.time() - t0 < budget_s * 0.95:
        h_f, c5_f = slsqp_round(best_h, n, 50000, 500, t0, budget_s * 0.97)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)

    try:
        val = evaluate_erdos_solution(best_h)
        if val is not None and val < final_c5:
            final_c5 = val
    except Exception:
        pass

    return (best_h, final_c5, n)
