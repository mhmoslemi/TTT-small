def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def pn_value(h, p):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.sum(corr**p) ** (1.0/p) * (2.0/len(h)))

def pn_grad(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    p1 = p - 1.0
    w = corr ** p1
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n) * dx
    w /= ws
    f = (np.sum(corr**p)) ** (1.0/p)
    grad = np.zeros(n)
    for i in range(2*n - 1):
        wi = w[i]
        if wi < 1e-14:

        d = (n - 1) - i
        lo1 = max(0, -d)
        hi1 = min(n, n - d)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + d:hi1 + d])
        lo2 = max(0, d)
        hi2 = min(n, d + n)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2 - d:hi2 - d]
    return grad * f * dx / (np.sum(corr**p) ** (1.0/p))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    return float((mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta) * dx)

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)

        lo2 = max(0, d)
        hi2 = min(n, d + n)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2 - d:hi2 - d]
    return grad * dx
