import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def subgradient(h, top_m=20):
    """Compute subgradient of max(corr) averaged over top-M lags. Vectorized."""
    corr = get_corr(h)
    n = len(h)
    top_idx = np.argpartition(corr, -top_m)[-top_m:]
    grad = np.zeros(n)
    n_minus_1 = n - 1
    for ki in top_idx:
        lag = ki - n_minus_1
        if lag > 0:
            grad[:n-lag] += (1.0 - h[lag:])
            grad[lag:] += (-h[:n-lag])
        elif lag < 0:
            l = -lag
            grad[l:] += (1.0 - h[:n-l])
            grad[:n-l] += (-h[l:])
        else:
            grad += (1.0 - 2.0 * h)
    grad /= len(top_idx)
    return grad, corr

    """Project h onto {0<=h<=1, sum(h)=n/2}."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        # Water-filling projection
        # If s > target: need to reduce; if s < target: need to increase
        # Simple approach: linear rescale then clip
        h *= target / s

        diff = target - np.sum(h)
        if diff > 0:

            room = 1.0 - h
            total_room = np.sum(room)
            if total_room > 0:
                h += diff * room / total_room
                h = np.clip(h, 0.0, 1.0)
        elif diff < 0:

            total_val = np.sum(h)
            if total_val > 0:
                h += diff * h / total_val
                h = np.clip(h, 0.0, 1.0)
    return h

            grad[lag:] += -h[:n-lag]
        elif lag < 0:
            l = -lag
            grad[l:] += 1.0 - h[:n-l]
            grad[:n-l] += -h[l:]
        else:
            grad += 1.0 - 2.0*h
    grad /= top_m
    return grad, np.max(corr)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = 0.5 * n
    s = h.sum()
    if s < 1e-14:
        h[:] = 0.5
        return h
    h *= tgt / s
    h = np.clip(h, 0.0, 1.0)
    diff = tgt - h.sum()
    if diff > 1e-12:
        room = np.maximum(0.0, 1.0 - h)
        tr = room.sum()
        if tr > 0: h += diff * room / tr; h = np.clip(h, 0.0, 1.0)
    elif diff < -1e-12:
        tv = h.sum()
        if tv > 0: h += diff * h / tv; h = np.clip(h, 0.0, 1.0)
    return h

def adam_loop(h0, n, dur, lr0=0.05, top_m=15, binary_every=500):
    t0 = time.time()
    h = h0.copy().astype(float)
    best_h, best_v = h.copy(), compute_c5(h)
    m = np.zeros(n); v = np.zeros(n)
    t = 0
    while time.time() - t0 < dur:
        t += 1
        lr = lr0 / (1.0 + 0.002 * t)
        g, mx = subgrad_max(h, top_m)
        m = 0.9*m + 0.1*g
        v = 0.999*v + 0.001*g*g
        mh = m / (1 - 0.9**t)
        vh = v / (1 - 0.999**t)
        h = project_feasible(h - lr * mh / (np.sqrt(vh) + 1e-8), n)
        cv = mx * (2.0/n)
        if cv < best_v - 1e-12:
            best_v = cv; best_h = h.copy()
        if binary_every > 0 and t % binary_every == 0:
            hb = (h > 0.5).astype(float)
            hb = project_feasible(hb, n)
            vb = compute_c5(hb)
            if vb < best_v:
                best_v = vb; best_h = hb.copy()
                h = hb.copy()
    return best_h, best_v

def binary_local_search(h0, n, dur):
    t0 = time.time()
    h = (h0 > 0.5).astype(float)
    h = project_feasible(h, n)
    best = h.copy()
    best_v = compute_c5(h)
    improved = True
    while improved and time.time() - t0 < dur:
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0: break
        for i in ones:
            if time.time() - t0 > dur: break
            for j in zeros:
                h[i] = 0.0; h[j] = 1.0
                v = compute_c5(h)
                h[i] = 1.0; h[j] = 0.0
                if v < best_v - 1e-12:
                    best_v = v; best[i] = 0.0; best[j] = 1.0
                    h[i] = 0.0; h[j] = 1.0
                    improved = True
                    break
    return best, best_v

Actually, let me think about this differently. The effect of swapping h[i]=1 with h[j]=0 on the correlation at lag k is:
- At position i: we lose the contribution h[i]*(1-h[i+k]) = 1*(1-h[i+k])
- At position j: we gain the contribution h[j](1-h[j+k]) = 1*(1-h[j+k])
- Also at positions that reference i or j as the second element:
  - At position i-k: we lose h[i-k]*(1-h[i]) = h[i-k]*1 = h[i-k]
  - At position j-k: we gain h[j-k]*(1-h[j]) = h[j-k]*1 = h[j-k]

So the change in c_k from swapping i (1→0) and j (0→1) is:
Δc_k = -[1-h[i+k]] + [1-h[j+k]] - h[i-k] + h[j-k]
     = h[j+k] - h[i+k] - h[i-k] + h[j-k]

(for valid indices; terms with out-of-range indices are 0)

This is O(1) per (i, j, k) triple, but we need the max over k, which makes it O(n) per swap. For 6400 swaps, that's 6400*160 = 1M operations per pass. Much faster!

def subgrad(h, top_m):

            best_h, best_c5 = hb2, vb2

    remaining = budget_s - (time.time() - t0)
    if remaining > 80 and n > 80:
        h80 = upsample_to(best_h, 80)
        h80, v80 = adam_loop(h80, 80, min(50, remaining - 20), lr=0.05, top_m=10)
        if v80 < best_c5:
            best_h, best_c5 = h80, v80
        remaining = budget_s - (time.time() - t0)
        if remaining > 40 and len(best_h) == 80:
            hb80, vb80 = bin_sa(h80, 80, min(30, remaining - 10), rng)
            if vb80 < best_c5:
                best_h, best_c5 = hb80, vb80

    remaining = budget_s - (time.time() - t0)
    if remaining > 80 and n < 200:
        h200 = upsample_to(best_h, 200)
        h200, v200 = adam_loop(h200, 200, min(50, remaining - 20), lr=0.03, top_m=15)
        if v200 < best_c5:
            best_h, best_c5 = h200, v200

    restart = 0
    while budget_s - (time.time() - t0) > 30 and restart < 40:
        scale = 0.03 + 0.06 * (restart % 5) / 4.0
        h_p = perturb_h(best_h, scale, rng)
        max_t = min(18, budget_s - (time.time() - t0) - 10)
        if max_t < 8:
            break
        h_r, v_r = adam_loop(h_p, len(best_h), max_t, lr=0.03, top_m=15)
        if v_r < best_c5:
            best_h, best_c5 = h_r, v_r
        restart += 1

    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        h_f, v_f = adam_loop(best_h, len(best_h), max(12, remaining - 5), lr=0.005, top_m=25)
        if v_f < best_c5:
            best_h, best_c5 = h_f, v_f

    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        hb_f, vb_f = bin_greedy(best_h, len(best_h), max(8, remaining - 3), rng)
        if vb_f < best_c5:
            best_h, best_c5 = hb_f, vb_f

    return best_h, best_c5, len(best_h)
