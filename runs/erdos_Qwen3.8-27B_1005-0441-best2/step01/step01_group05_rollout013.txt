    dx = 2.0 / n
    corb = np.correlate(h, 1.0 - h, "full")  # length 2n-1
    # Softmax weights

    # Compute gradient
    grad = np.zeros(n)
    # shifts: shift_k = k - (n-1) for k = 0, ..., 2n-2
    shifts = np.arange(2*n-1) - (n-1)  # from -(n-1) to (n-1)

    if s > 1e-12:
        h_new *= (n_new / 2.0) / s
    return np.clip(h_new, 0.0, 1.0)

def _ham_init(n, rng):
    h = 0.5 + 0.2 * np.sin(2 * np.pi * 3 * np.arange(n) / n)
    h = np.clip(h, 0.0, 1.0)
    return _proj(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s - 5

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(100, 0.5)
    n = len(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Polish initial
    dl = min(t0 + 130, t_end - 60)
    h, f = _pgd(best_h.copy(), n, dl, t_init=0.03)
    if f < best_c5:
        best_h, best_c5 = h, f

    # Phase 2: Second pass with lower temp
    if time.time() < t_end - 80:
        dl = min(time.time() + 80, t_end - 60)
        h, f = _pgd(best_h.copy(), n, dl, t_init=0.005)
        if f < best_c5:
            best_h, best_c5 = h, f

    # Phase 3: Alternative inits
    inits = [
        _binary_init(n, rng),
        _pulse_init(n, rng),
        _pulse_init(n, rng),
        _ham_init(n, rng),
        _proj(best_h + 0.15 * rng.randn(n), n / 2.0),
        _proj(best_h + 0.08 * rng.randn(n), n / 2.0),
    ]
    for i, hi in enumerate(inits):
        if time.time() > t_end - 50:
            break
        tl = min(50, t_end - time.time() - 30)
        if tl < 15:
            break
        h, f = _pgd(hi, n, time.time() + tl, t_init=0.05)
        if f < best_c5:
            best_h, best_c5 = h, f

    # Phase 4: Random restarts
    r = 0
    while time.time() < t_end - 25 and r < 200:
        sc = 0.02 + 0.07 * rng.random()
        hp = _proj(best_h + sc * rng.randn(n), n / 2.0)
        tl = min(18, t_end - time.time() - 12)
        if tl < 8:
            break
        h, f = _pgd(hp, n, time.time() + tl, t_init=0.01, max_iter=3000)
        if f < best_c5:
            best_h, best_c5 = h, f
        r += 1

    # Phase 5: Upsample and polish at higher res
    if time.time() < t_end - 60:
        n2 = n + 40
        if n2 > n:
            hu = _upsample(best_h, n2)
            h, f = _pgd(hu, n2, min(time.time() + 50, t_end - 20), t_init=0.01, max_iter=5000)
            if f < best_c5:
                best_h, best_c5 = h, f
                n = n2

    # Phase 6: Final polish
    if time.time() < t_end - 10:
        h, f = _pgd(best_h.copy(), len(best_h), t_end, t_init=0.003, max_iter=5000)
        if f < best_c5:
            best_h, best_c5 = h, f

    return best_h, best_c5, len(best_h)
