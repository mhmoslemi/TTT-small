import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(500):
        d = target - np.sum(h)
        if abs(d) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(free))
        if cnt == 0:
            h += d / n
            np.clip(h, 0.0, 1.0, out=h)
        else:
            h[free] += d / cnt
            np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        mid = int(np.argmin(np.abs(h - 0.5)))
        h[mid] = min(1.0, max(0.0, h[mid] + resid))
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def _obj_smooth(x, alpha, target_sum, pen_coeff):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    sh = np.clip(alpha * (corr - m), -500.0, 500.0)
    sm = m + np.log(np.sum(np.exp(sh))) / alpha
    pen = pen_coeff * (np.sum(x) - target_sum) ** 2
    return sm + pen

def _obj_active(x, alpha, target_sum, pen_coeff, active_idx):
    dx = 2.0 / len(x)
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    ca = corr[active_idx]
    ma = float(np.max(ca))
    sh = np.clip(alpha * (ca - ma), -500.0, 500.0)
    sm = ma + np.log(np.sum(np.exp(sh))) / alpha
    pen = pen_coeff * (np.sum(x) - target_sum) ** 2
    return sm + pen

def _sum_eq(x, target_sum):
    return np.sum(x) - target_sum

def _sum_eq_jac(x):
    return np.ones(len(x))

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        n0 = 160
        h0 = np.zeros(n0)
        b = n0 // 4
        h0[:b] = 1.0
        h0[-b:] = 1.0
        h0 = project_feasible(h0, n0)

    h_best = project_feasible(h0.copy(), len(h0))
    n = len(h_best)
    c5_best = compute_c5(h_best)
    target_sum = n * 0.5
    bounds = [(0.0, 1.0)] * n

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: L-BFGS-B smooth max, ramp alpha up
    for alpha, pen in [(40, 800), (100, 2000), (300, 5000), (600, 10000), (1200, 20000)]:
        if tl() < 90:
            break
        res = minimize(_obj_smooth, h_best.copy(), args=(alpha, target_sum, pen),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 400, 'ftol': 1e-18, 'gtol': 1e-14})
        hc = project_feasible(res.x, n)
        c5c = compute_c5(hc)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()

    # Phase 2: Active-set L-BFGS-B (focus on critical lags)
    for it in range(40):
        if tl() < 40:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode='full')
        m = float(np.max(corr))
        active = np.where(corr >= m * 0.99 + 1e-15)[0]
        if len(active) < 3:
            active = np.argsort(corr)[-4:]
        for alpha, pen in [(400, 5000), (800, 10000)]:
            if tl() < 30:
                break
            res = minimize(_obj_active, h_best.copy(),
                          args=(alpha, target_sum, pen, active),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-15})
            hc = project_feasible(res.x, n)
            c5c = compute_c5(hc)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc.copy()

    # Phase 3: SLSQP with equality constraint (more precise constraint handling)
    constraint = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}]
    for alpha in [500, 1000]:
        if tl() < 70:
            break
        res = minimize(_obj_smooth, h_best.copy(), args=(alpha, target_sum, 0.0),
                      method='SLSQP', bounds=bounds, constraints=constraint,
                      options={'maxiter': 600, 'ftol': 1e-14})
        hc = project_feasible(res.x, n)
        c5c = compute_c5(hc)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()

    # Phase 4: Multi-scale perturbations
    for scale in [0.08, 0.04, 0.02, 0.01, 0.005, 0.002]:
        if tl() < 60:
            break
        for _ in range(3):
            if tl() < 25:
                break
            h_p = h_best + scale * rng.standard_normal(n)
            h_p = project_feasible(h_p, n)
            res = minimize(_obj_smooth, h_p, args=(600, target_sum, 8000),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 300, 'ftol': 1e-18})
            hc = project_feasible(res.x, n)
            c5c = compute_c5(hc)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc.copy()

    # Phase 5: Structural restarts
    if tl() > 60:
        # Three-block
        hb3 = np.zeros(n)
        b3 = max(2, n // 6)
        hb3[:b3] = 1.0
        hb3[n // 2 - b3:n // 2 + b3] = 1.0
        hb3[-b3:] = 1.0
        hb3 = project_feasible(hb3, n)
        res = minimize(_obj_smooth, hb3, args=(400, target_sum, 5000),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 400, 'ftol': 1e-18})
        hc = project_feasible(res.x, n)
        c5c = compute_c5(hc)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()

        # Random binary with correct count
        idx = rng.permutation(n)[:n // 2]
        hbr = np.zeros(n)
        hbr[idx] = 1.0
        hbr = project_feasible(hbr, n)
        res = minimize(_obj_smooth, hbr, args=(400, target_sum, 5000),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 400, 'ftol': 1e-18})
        hc = project_feasible(res.x, n)
        c5c = compute_c5(hc)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()

    # Phase 6: Grid exploration
    for n_try in [n * 3 // 2, n * 2, n // 2 + 10]:
        if tl() < 60:
            break
        if n_try <= 24 or n_try > 480 or n_try == n:
            continue
        x_old = np.linspace(0.0, 1.0, n)
        x_new = np.linspace(0.0, 1.0, n_try)
        hi = np.interp(x_new, x_old, h_best)
        hi = project_feasible(hi, n_try)
        ts = n_try * 0.5
        bd = [(0.0, 1.0)] * n_try
        for alpha in [300, 800]:
            if tl() < 30:
                break
            res = minimize(_obj_smooth, hi, args=(alpha, ts, 5000),
                          method='L-BFGS-B', bounds=bd,
                          options={'maxiter': 400, 'ftol': 1e-18})
            hc = project_feasible(res.x, n_try)
            c5c = compute_c5(hc)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc.copy()

    # Final polish: small coordinate moves
    if tl() > 15:
        for _ in range(200):
            if tl() < 10:
                break
            i = int(rng.integers(0, n))
            for d in [-0.002, 0.002, -0.001, 0.001]:
                v = h_best[i] + d
                if v < 0 or v > 1:
                    continue
                h_t = h_best.copy()
                h_t[i] = v
                h_t = project_feasible(h_t, n)
                c5_t = compute_c5(h_t)
                if c5_t < c5_best - 1e-12:
                    h_best = h_t
                    c5_best = c5_t
                    break

    h_best = project_feasible(h_best, len(h_best))
    c5_best = compute_c5(h_best)
    return h_best, c5_best, len(h_best)
