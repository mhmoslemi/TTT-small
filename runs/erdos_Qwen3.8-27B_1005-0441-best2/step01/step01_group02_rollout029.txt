from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def shift_grad(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        if n - d > 0:
            grad[:n-d] += (1.0 - h[d:])
        if d < n:
            grad[d:] -= h[:n-d]
    else:
        dd = -d
        if n - dd > 0:
            grad[dd:] += (1.0 - h[:n-dd])
        if n - dd > 0:
            grad[:n-dd] -= h[dd:]
    return grad

def smooth_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    shifted = np.clip(corr - mc, -100.0, 0.0)
    return float((mc + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx)

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    shifted = np.clip(corr - mc, -100.0, 0.0)
    w = np.exp(beta * shifted)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        if w[i] < 1e-13:
            continue
        d = (n - 1) - i
        grad += w[i] * shift_grad(h, d)
    return grad * dx

def project(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h = np.clip(h + diff / n, 0.0, 1.0)
    return h

def fw_optimize(h0, max_iter=5000, t_budget=100):
    """Frank-Wolfe with the actual (non-smooth) objective"""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    
    for t in range(max_iter):
        if time.time() - t0 > t_budget:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        mc = corr.max()
        # Active set: shifts where corr is within 1% of max
        active = corr > mc - 0.01 * mc
        # Subgradient: average gradient over active shifts
        g = np.zeros(n)
        n_active = active.sum()
        for i in range(2 * n - 1):
            if not active[i]:
                continue
            d = (n - 1) - i
            g += shift_grad(h, d)
        g /= n_active
        g *= dx
        
        # FW step
        order = np.argsort(g)
        s = np.zeros(n)
        s[order[:n // 2]] = 1.0
        
        alpha = 2.0 / (t + 2)
        h_new = (1 - alpha) * h + alpha * s
        h_new = np.clip(h_new, 0, 1)
        
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
        h = h_new
    
    return project(best_h), best_c5

def lbfgs_penalty(h0, beta, mu, maxiter=300):
    """L-BFGS-B with penalty for sum constraint"""
    n = len(h0)
    target = n / 2.0
    
    def obj(x):
        return smooth_val(x, beta) + mu * (x.sum() - target) ** 2
    
    def jac(x):
        return smooth_grad(x, beta) + 2 * mu * (x.sum() - target) * np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=jac, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14,
                                               'gtol': 1e-10, 'maxls': 50})
        return project(res.x)
    except Exception:
        return project(h0.copy())

def coord_descent(h0, n_passes=50, t_budget=30):
    """Coordinate descent polish"""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    
    for p in range(n_passes):
        if time.time() - t0 > t_budget:
            break
        improved = False
        for j in range(n):
            if time.time() - t0 > t_budget:
                break
            # Try different values for h[j], adjusting h[j+1] for sum
            v0 = h[j]
            k = (j + 1) % n
            for delta in [-0.1, -0.05, -0.02, -0.01, 0.01, 0.02, 0.05, 0.1]:
                h[j] = v0 + delta
                h[k] -= delta
                if h[j] < 0 or h[j] > 1 or h[k] < 0 or h[k] > 1:
                    continue
                c = compute_c5(h)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    improved = True
                    break
            h[j] = v0
            h[k] = h[k] + delta if False else h[k]  # restore
        if not improved:
            break

# Top-level helper functions
def _con_fun(x):
    return x.sum() - len(x) / 2.0

def _con_jac(x):
    return np.ones(len(x))

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def shift_grad(h, d):
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        if n > d:
            g[:n-d] += 1.0 - h[d:]
            g[d:] -= h[:n-d]
    else:
        dd = -d
        if n > dd:
            g[dd:] += 1.0 - h[:n-dd]
            g[:n-dd] -= h[dd:]
    return g

def smooth_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = corr.max()
    s = np.clip(corr - mc, -50.0, 0.0)
    return (mc + np.log(np.sum(np.exp(beta * s))) / beta) * dx

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = corr.max()
    s = np.clip(corr - mc, -50.0, 0.0)
    w = np.exp(beta * s)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    g = np.zeros(n)
    for i in range(2*n-1):
        if w[i] < 1e-13:
            continue
        g += w[i] * shift_grad(h, (n-1)-i)
    return g * dx

def project(h):
    n = len(h)
    tgt = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = tgt - h.sum()
        if abs(diff) < 1e-15:
            break
        h = np.clip(h + diff / n, 0.0, 1.0)
    return h
