    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def _rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    else:
        n0 = 160
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Optimize initial with SLSQP (multiple temps)
    temps1 = [0.05, 0.02, 0.008, 0.003, 0.001, 0.0004, 0.0001]
    if _rem() > 80:
        h_opt, c5_opt = _optimize_slseqp(h0.copy(), n0, temps1, t0, min(180, _rem() - 60), seed)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 2: L-BFGS-B with penalty on current best
    if _rem() > 60:
        h_opt, c5_opt = _optimize_lbfgsb(best_h.copy(), n0, 0.002, t0, min(100, _rem() - 40))
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 3: Alternative initializations
    alts = []
    alts.append((_thue_morse_init(n0), "thue"))
    alts.append((_gapped_binary(n0, 0.15), "gapped15"))
    alts.append((_gapped_binary(n0, 0.3), "gapped30"))
    alts.append((_project(best_h + 0.1 * np.random.randn(n0), n0), "perturb"))

    for i, (h_alt, name) in enumerate(alts):
        if _rem() < 50:
            break
        temps2 = [0.03, 0.01, 0.003, 0.001]
        h_opt, c5_opt = _optimize_slseqp(h_alt, n0, temps2, t0, min(45, _rem() - 30), seed + 500 + i)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 4: Subgradient method for fine-tuning
    if _rem() > 60:
        h_opt, c5_opt = _subgrad_loop(best_h.copy(), n0, t0, min(80, _rem() - 30), seed + 200)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 5: Try n=80 (faster, more iterations possible)
    n_small = 80
    h_small = _upsample(best_h, n_small)
    if _rem() > 80:
        h_opt, c5_opt = _optimize_slseqp(h_small, n_small, [0.03, 0.01, 0.003, 0.001, 0.0003],
                                         t0, min(100, _rem() - 30), seed + 800)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        if _rem() > 40:
            h_opt2, c5_opt2 = _optimize_lbfgsb(h_opt, n_small, 0.001, t0, min(50, _rem() - 20))
            if c5_opt2 < best_c5:
                best_h, best_c5 = h_opt2, c5_opt2

    # Phase 6: Random restarts at best n
    n_cur = len(best_h)
    for restart in range(30):
        if _rem() < 25:
            break
        scale = 0.02 + 0.06 * (restart % 5) / 4.0
        h_pert = _project(best_h + scale * np.random.randn(n_cur), n_cur)
        temps3 = [0.01, 0.003, 0.001]
        max_t = min(30, _rem() - 15)
        if max_t < 10:
            break
        h_opt, c5_opt = _optimize_slseqp(h_pert, n_cur, temps3, t0, max_t, seed + 2000 + restart)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 7: Final SLSQP polish with very fine temp
    if _rem() > 30:
        globals()['_g_n'] = len(best_h)
        globals()['_g_temp'] = 0.0001
        globals()['_g_pen'] = 100.0
        n_fin = len(best_h)
        _g_n = n_fin
        _g_temp = 0.0001
        _g_pen = 1000.0
        bounds = [(0.0, 1.0)] * n_fin
        cons = ({'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac},)
        try:
            res = minimize(_objective, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 500, 'ftol': 1e-14})
            h_new = _project(res.x, n_fin)
            v = compute_c5(h_new)
            if v < best_c5:
                best_h, best_c5 = h_new, v
        except Exception:
            pass

    # Phase 8: Subgradient final pass
    if _rem() > 20:
        h_opt, c5_opt = _subgrad_loop(best_h.copy(), len(best_h), t0, min(15, _rem() - 5), seed + 9999)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Ensure feasibility
    best_h = _project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
