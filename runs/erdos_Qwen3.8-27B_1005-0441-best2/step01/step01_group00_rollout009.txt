def _eq_fun(x, n):
    return x.sum() - n * 0.5

def _eq_jac(x, n):
    return np.ones(n)

def _pair_grid_opt(h, i, j):
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-10:
        return h, False
    best_f = _c5(h)
    best_vi = h[i]
    grid = np.linspace(lo, hi, 40)
    for vi in grid:
        h[i] = vi
        h[j] = s - vi
        f = _c5(h)
        if f < best_f:
            best_f = f
            best_vi = vi
    h[i] = best_vi
    h[j] = s - best_vi
    return h, best_f < _c5(h) - 1e-15

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_val = np.inf

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        h_init = _proj(h_init, len(h_init))
        best_h = h_init.copy()
        best_val = _c5(best_h)
    except Exception:
        pass

    if best_h is None:
        n = 70
        best_h = np.full(n, 0.5)
        best_val = _c5(best_h)

    n = len(best_h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]

    # Phase 1: SLSQP with decreasing smoothing
    for eps in [0.02, 0.01, 0.005, 0.002, 0.001]:
        if time.time() - t0 > budget_s * 0.25:
            break
        try:
            res = minimize(_smoobj, best_h.copy(), args=(eps,),
                           jac=_smograd, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-14})
            h_res = _proj(res.x, n)
            val = _c5(h_res)
            if val < best_val:
                best_val = val
                best_h = h_res
        except Exception:
            pass

    # Phase 2: Weighted subgradient descent with adaptive lr
    if time.time() - t0 < budget_s * 0.55:
        h = best_h.copy()
        lr = 0.002
        stuck = 0
        for it in range(80000):
            if time.time() - t0 > budget_s * 0.55:
                break
            g = _subgrad_weighted(h, 0.004)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            h_new = _proj(h - lr * g, n)
            v = _c5(h_new)
            if v < best_val - 1e-15:
                best_val = v
                best_h = h_new.copy()
                h = h_new
                stuck = 0
            else:
                h = h_new  # still move, but don't update best
                stuck += 1
                lr *= 0.985
                if stuck > 3000:
                    lr = 0.002
                    stuck = 0
                    # Small random kick
                    mask = rng.rand(n) < 0.3
                    h = _proj(h + 0.003 * (rng.rand(n) - 0.5) * mask, n)

    # Phase 3: Pair coordinate descent (grid search)
    if time.time() - t0 < budget_s * 0.82:
        h = best_h.copy()
        idx = np.arange(n)
        sweeps = 0
        while time.time() - t0 < budget_s * 0.82:
            rng.shuffle(idx)
            improved = 0
            for cnt in range(n - 1):
                i = idx[cnt]
                j = idx[cnt + 1]
                if h[i] < 0.01 or h[i] > 0.99 or h[j] < 0.01 or h[j] > 0.99:
                    continue
                h, did_improve = _pair_grid_opt(h, i, j)
                v = _c5(h)
                if v < best_val - 1e-15:
                    best_val = v
                    best_h = h.copy()
                    improved += 1
            sweeps += 1
            if improved == 0 and sweeps > 3:
                break

    # Phase 4: Final fine subgradient polish
    if time.time() - t0 < budget_s * 0.95:
        h = best_h.copy()
        lr = 0.0008
        for it in range(50000):
            if time.time() - t0 > budget_s * 0.95:
                break
            g = _subgrad_weighted(h, 0.002)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            h_new = _proj(h - lr * g, n)
            v = _c5(h_new)
            if v < best_val - 1e-15:
                best_val = v
                best_h = h_new.copy()
                h = h_new
                lr = min(lr * 1.005, 0.003)
            else:
                lr *= 0.99
                h = h_new
                if lr < 1e-9:
                    break

    # Phase 5: Tiny random perturbations to escape local opt
    if time.time() - t0 < budget_s * 0.98:
        for attempt in range(200):
            if time.time() - t0 > budget_s * 0.98:
                break
            h_cand = best_h.copy()
            mask = rng.rand(n) < 0.15
            h_cand[mask] += (rng.rand(mask.sum()) - 0.5) * 0.01
            h_cand = _proj(h_cand, n)
            v = _c5(h_cand)
            if v < best_val:
                best_val = v
                best_h = h_cand
            else:
                # Try very small subgradient step from perturbed point
                g = _subgrad_weighted(h_cand, 0.003)
                for ls in range(50):
                    h_cand = _proj(h_cand - 0.0005 * g, n)
                    v = _c5(h_cand)
                    if v < best_val:
                        best_val = v
                        best_h = h_cand.copy()
                        break

    best_h = _proj(best_h, n)
    best_val = _c5(best_h)
    return (best_h, best_val, n)
