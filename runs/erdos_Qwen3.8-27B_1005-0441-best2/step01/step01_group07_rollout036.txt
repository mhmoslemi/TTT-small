    shifted = np.clip(tau * (corr - mc), -500, 500)

    grad = np.zeros(n)

import numpy as np
import time, math

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad(h, dx, tau):
    n = len(h)
    corr = compute_corr_full(h)
    mc = corr.max()
    shifted = np.clip(tau * (corr - mc), -500, 500)
    exps = np.exp(shifted)
    weights = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-13:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            L = -lag
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
    return grad * dx

def gradient_optimize(h_init, time_budget, seed, lr_init=0.05, tau_init=40.0):

    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    beta = 0.8
    lr = lr_init
    tau = tau_init
    t0 = time.time()
    deadline = t0 + time_budget
    no_improve = 0
    while time.time() < deadline - 0.3:

        h_new = project_feasible(h + v, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
    return best_h, best_c5

def to_binary(h_frac, n):
    target = n // 2
    h = (h_frac >= 0.5).astype(float)
    s = h.sum()
    if s > target:
        order = np.argsort(h_frac)[::-1]
        for i in order:
            if s <= target: break
            h[i] = 0.0; s -= 1
    elif s < target:
        order = np.argsort(h_frac)
        for i in order:
            if s >= target: break
            h[i] = 1.0; s += 1
    return h

def binary_best_improve(h_bin, time_budget, max_rounds=5000):
    """Exhaustive best-improvement local search for binary h."""
    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    deadline = t0 + time_budget
    rounds = 0
    while time.time() < deadline - 1.0 and rounds < max_rounds:
        corr = compute_corr_full(h)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_swap_c5 = best_c5
        best_swap = None
        # Try all swaps
        for a in ones:
            for b in zeros:
                h[a] = 0.0
                h[b] = 1.0
                c = compute_corr_full(h)
                c5_val = c.max() * dx
                if c5_val < best_swap_c5 - 1e-14:
                    best_swap_c5 = c5_val
                    best_swap = (a, b)
                h[a] = 1.0
                h[b] = 0.0
        if best_swap is None:

        a, b = best_swap
        h[a] = 0.0
        h[b] = 1.0
        best_c5 = best_swap_c5
        best_h = h.copy()
        rounds += 1
    return best_h, best_c5

def binary_sa(h_bin, time_budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.copy()

def eval_swap_delta(h, a, b, n):
    """Compute delta in correlation when h[a] 1→0, h[b] 0→1."""
    omh = 1.0 - h
    # delta[l] for l = -(n-1)..(n-1), stored at index l+(n-1)
    delta = np.zeros(2*n-1)
    # -(1-h[a-l]): valid when 0 <= a-l <= n-1, i.e., a-(n-1) <= l <= a
    for l in range(max(a-(n-1), -(n-1)), min(a, n-1)+1):
        idx = l + (n-1)
        delta[idx] -= omh[a-l]
    # +(1-h[b-l]): valid when 0 <= b-l <= n-1, i.e., b-(n-1) <= l <= b
    for l in range(max(b-(n-1), -(n-1)), min(b, n-1)+1):
        idx = l + (n-1)
        delta[idx] += omh[b-l]
    # +h[a+l]: valid when 0 <= a+l <= n-1, i.e., -a <= l <= n-1-a
    for l in range(max(-a, -(n-1)), min(n-1-a, n-1)+1):
        idx = l + (n-1)
        delta[idx] += h[a+l]
    # -h[b+l]: valid when 0 <= b+l <= n-1, i.e., -b <= l <= n-1-b
    for l in range(max(-b, -(n-1)), min(n-1-b, n-1)+1):
        idx = l + (n-1)
        delta[idx] -= h[b+l]
    return delta

import numpy as np
import time, math

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad(h, dx, tau):
    n = len(h)
    corr = compute_corr_full(h)
    mc = corr.max()
    shifted = np.clip(tau * (corr - mc), -500, 500)
    exps = np.exp(shifted)
    weights = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-13:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            L = -lag
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
    return grad * dx

def gradient_optimize(h_init, time_budget, seed, lr_init=0.05, tau_init=40.0):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    beta = 0.8
    lr = lr_init
    tau = tau_init
    t0 = time.time()
    deadline = t0 + time_budget
    no_improve = 0
    while time.time() < deadline - 0.3:
        if no_improve > 300:
            h = project_feasible(best_h + rng.randn(n) * 0.03, target_sum)
            v[:] = 0.0
            lr = lr_init
            tau = tau_init
            no_improve = 0
        elif no_improve > 100:
            lr = max(lr * 0.9, 0.002)
        grad = compute_grad(h, dx, tau)
        tau = min(tau * 1.003, 400.0)
        v = beta * v - lr * grad
        h_new = project_feasible(h + v, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
    return best_h, best_c5

def to_binary(h_frac, n):
    target = n // 2
    h = (h_frac >= 0.5).astype(float)
    s = int(h.sum())
    if s > target:
        order = np.argsort(h_frac)[::-1]
        for i in order:
            if s <= target:
                break
            h[i] = 0.0
            s -= 1
    elif s < target:
        order = np.argsort(h_frac)
        for i in order:
            if s >= target:
                break
            h[i] = 1.0
            s += 1
    return h

def binary_best_improve(h_bin, time_budget):
    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.copy()
    corr = compute_corr_full(h)
    best_c5 = float(corr.max() * dx)
    best_h = h.copy()
    t0 = time.time()
    deadline = t0 + time_budget
    while time.time() < deadline - 0.5:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 0.0
        best_a, best_b = -1, -1
        for a in ones:
            for b in zeros:
                d = -corr.copy()
                ao = a + n - 1
                bo = b + n - 1
                sl = max(0, min(ao, 2*n-2) - n + 1)
                sr = min(2*n-1, max(ao, n - 1) + n + 1)
                if sl < sr:
                    d[sl:sr] -= 1.0
                sl = max(0, min(bo, 2*n-2) - n + 1)
                sr = min(2*n-1, max(bo, n - 1) + n + 1)
                if sl < sr:
                    d[sl:sr] += 1.0
                sl = max(0, min(ao, 2*n-2) - n + 2)
                sr = min(2*n-1, max(ao, n) + n)
                if sl < sr:
                    d[sl:sr] += h[sl + 1 - ao:sr + 1 - ao]
                sl = max(0, min(bo, 2*n-2) - n + 2)
                sr = min(2*n-1, max(bo, n) + n)
                if sl < sr:
                    d[sl:sr] -= h[sl + 1 - bo:sr + 1 - bo]
                new_max = float(np.max(corr + d))
                new_val = new_max - float(corr.max())
                if new_val < best_delta:
                    best_delta = new_val
                    best_a, best_b = a, b
        if best_a < 0:
            break
        h[best_a] = 0.0
        h[best_b] = 1.0
        corr = compute_corr_full(h)
        best_c5 = float(corr.max() * dx)
        best_h = h.copy()
    return best_h, best_c5

def binary_sa(h_bin, time_budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    deadline = t0 + time_budget
    T = 0.005
    while time.time() < deadline - 0.5:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        h[a], h[b] = 0.0, 1.0
        c5_new = compute_c5(h)
        delta = c5_new - best_c5
        if delta < 0:
            best_c5 = c5_new
            best_h = h.copy()
        elif rng.rand() < math.exp(-delta / max(T, 1e-12)):
            pass
        else:
            h[a], h[b] = 1.0, 0.0
        T = max(T * 0.9997, 0.0005)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
