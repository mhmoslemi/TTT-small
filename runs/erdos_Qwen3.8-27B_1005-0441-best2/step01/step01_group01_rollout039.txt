    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.sum(free)
        if nf == 0:

        np.clip(h, 0.0, 1.0, out=h)

        if np.any(free):

            np.clip(h, 0.0, 1.0, out=h)

    dx = 2.0 / n

    h[:block] = 1.0
    h[-block:] = 1.0
    return project_feasible(h, n)

def make_three_block(n):
    h = np.zeros(n)
    b = max(1, n // 6)
    h[:b] = 1.0
    h[n//2 - b:n//2 + b] = 1.0
    h[-b:] = 1.0
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)

def make_four_block(n):
    h = np.zeros(n)
    b = max(1, n // 8)
    for i in range(4):
        center = int((i + 0.5) * n / 4.0)
        start = max(0, center - b)
        end = min(n, center + b)
        h[start:end] = 1.0
    return project_feasible(h, n)

def make_ramp(n):
    h = np.zeros(n)
    q = n // 4
    ramp_len = max(2, n // 16)
    h[:q - ramp_len] = 1.0
    h[q - ramp_len:q] = np.linspace(1.0, 0.0, ramp_len)
    h[n//2 - q + ramp_len:n//2 - q + 2*ramp_len] = np.linspace(0.0, 1.0, ramp_len) if 2*ramp_len <= n else 0
    h[n - q + ramp_len:] = 1.0
    h[n - q - (ramp_len - 1):n - q + ramp_len] = np.linspace(1.0, 0.0, 2*ramp_len - 1)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        h0 = make_two_block(160)

    n = len(h0)
    h_best = project_feasible(h0.copy(), n)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    def do_lbfgs(h_start, alpha, pen, maxiter, ftol, gtol):
        res = minimize(fun_obj, h_start, args=(alpha, pen), jac=fun_grad,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol, 'iprint': -1})
        return res.x

    # Phase 1: Coarse-to-fine optimization with analytical gradient
    phase1_configs = [(20.0, 50.0, 400, 1e-12, 1e-6),
                      (50.0, 100.0, 400, 1e-13, 1e-7),
                      (100.0, 200.0, 500, 1e-13, 1e-8),
                      (200.0, 500.0, 500, 1e-14, 1e-8)]
    for alpha, pen, mi, ft, gt in phase1_configs:
        if remaining() < 40:
            break
        x = do_lbfgs(h_best.copy(), alpha, pen, mi, ft, gt)
        hc = project_feasible(x, n)
        c5c = compute_c5(hc)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()

    # Phase 2: Multi-start from diverse structures
    starts = [h_best.copy(), make_two_block(n), make_three_block(n), make_four_block(n)]
    for _ in range(4):
        hr = rng.random(n)
        starts.append(project_feasible(hr, n))
    # Stochastic structured: random block sizes
    for _ in range(2):
        hs = np.zeros(n)
        nb = 2 + rng.integers(0, 3)
        for b_i in range(nb):
            w = max(1, n // (2 * nb) + rng.integers(-n // (4 * nb), n // (4 * nb)))
            pos = rng.integers(0, max(1, n - w))
            hs[pos:pos + w] = 1.0
        starts.append(project_feasible(hs, n))

    for sp in starts:
        if remaining() < 50:
            break
        for alpha, pen in [(100.0, 500.0), (300.0, 1000.0)]:
            if remaining() < 25:
                break
            x = do_lbfgs(sp.copy(), alpha, pen, 400, 1e-13, 1e-8)
            hc = project_feasible(x, n)
            c5c = compute_c5(hc)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc.copy()

    # Phase 3: Perturbation restarts (main exploration loop)
    no_improve = 0
    while remaining() > 150:
        pert = 0.005 + 0.04 * rng.random()
        h_pert = h_best + pert * rng.standard_normal(n)
        h_pert = project_feasible(h_pert, n)
        alpha = 200.0 + 200.0 * rng.random()
        pen = 500.0 + 500.0 * rng.random()
        x = do_lbfgs(h_pert, alpha, pen, 300, 1e-14, 1e-8)
        hc = project_feasible(x, n)
        c5c = compute_c5(hc)
        if c5c < c5_best - 1e-10:
            c5_best = c5c
            h_best = hc.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                # Larger perturbation to escape basin
                pert = 0.05 + 0.1 * rng.random()
                h_pert = h_best + pert * rng.standard_normal(n)
                h_pert = project_feasible(h_pert, n)
                no_improve = 0

    # Phase 4: High-alpha final refinement
    if remaining() > 20:
        for alpha in [500.0, 1000.0, 2000.0]:
            if remaining() < 10:
                break
            x = do_lbfgs(h_best.copy(), alpha, 2000.0, 400, 1e-15, 1e-10)
            hc = project_feasible(x, n)
            c5c = compute_c5(hc)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc.copy()

    # Phase 5: Try upscaling if time permits
    if remaining() > 80 and n < 320:
        n2 = min(n * 2, 320)
        if n2 > n:
            x_old = np.arange(n, dtype=float)
            x_new = np.linspace(0.0, float(n - 1), n2)
            h_up = np.interp(x_new, x_old, h_best)
            h_up = project_feasible(h_up, n2)
            b2 = [(0.0, 1.0)] * n2
            for alpha, pen in [(100.0, 500.0), (300.0, 1000.0)]:
                if remaining() < 30:
                    break
                res = minimize(fun_obj, h_up, args=(alpha, pen), jac=fun_grad,
                               method='L-BFGS-B', bounds=b2,
                               options={'maxiter': 400, 'ftol': 1e-14, 'gtol': 1e-8, 'iprint': -1})
                hc = project_feasible(res.x, n2)
                c5c = compute_c5(hc)
                if c5c < c5_best:
                    c5_best = c5c
                    h_best = hc.copy()
            n = len(h_best)

    # Final projection and evaluation
    h_best = project_feasible(h_best, len(h_best))
    c5_best = compute_c5(h_best)
    return h_best, c5_best, len(h_best)
