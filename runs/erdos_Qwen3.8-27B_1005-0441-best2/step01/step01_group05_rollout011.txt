def smooth_max_obj_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)

    exps = np.exp(shifted)
    s = np.sum(exps)
    val = (m + temp * np.log(s)) * dx
    w = exps / s
    grad = np.zeros(n)
    for s_idx in range(-(n - 1), n):
        wk = w[n - 1 + s_idx]
        if wk < 1e-13:
            continue
        lo1 = max(0, s_idx)
        hi1 = min(n, n + s_idx)
        if lo1 < hi1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1 - s_idx:hi1 - s_idx])

def project_constraint(h_new, n):
    target = float(n) / 2.0
    lo = h_new.min() - 1.0
    hi = h_new.max() + 1.0
    for _ in range(80):

        h = np.clip(h_new - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h_new - 0.5 * (lo + hi), 0.0, 1.0)

def downsample(h, n_new):
    n_old = len(h)
    if n_new >= n_old:
        h_new = h[:n_new] if n_new > n_old else h.copy()
    else:
        x_old = np.arange(n_old) * 2.0 / n_old
        x_new = np.arange(n_new) * 2.0 / n_new
        h_new = np.interp(x_new, x_old, h)
    s = h_new.sum()
    target = float(n_new) / 2.0
    if s > 1e-12:

    h_new = np.clip(h_new, 0.0, 1.0)

        if v2 < val:
            h = h_try
            step = min(step * 1.05, 10.0 / n)
        else:
            step *= 0.7
    return h

def optimize_lbfgsb(h, n, temps, max_time, t0):
    best = h.copy()
    best_val = compute_c5(best)
    for temp in temps:
        if time.time() - t0 > max_time:
            break
        h_cur = best.copy()
        for _ in range(3):
            if time.time() - t0 > max_time:
                break
            try:
                res = minimize(smooth_max_obj_grad, h_cur, jac=True,
                               method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * n,
                               options={'maxiter': 300, 'ftol': 1e-14,
                                         'gtol': 1e-8, 'iprint': -1})
                h_new = project_constraint(res.x, n)
                v = compute_c5(h_new)
                if v < best_val - 1e-12:
                    best = h_new
                    best_val = v
                h_cur = h_new
            except Exception:
                break
    return best, best_val

def make_symmetric_init(n, n_pulses=4, width_frac=0.35):
    h = np.zeros(n)
    xs = np.arange(n) * 2.0 / n
    for i in range(n_pulses):
        center = 2.0 * (i + 0.5) / n_pulses
        sigma = 2.0 * width_frac / n_pulses
        h += np.exp(-0.5 * ((xs - center) / sigma) ** 2)
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    target = float(n) / 2.0
    if s > 1e-12:
        h = h * target / s
    return np.clip(h, 0.0, 1.0)

def make_block_init(n, n_blocks=4, fill=0.5):
    h = np.zeros(n)
    block_size = n // n_blocks
    ones_per_block = int(block_size * fill)
    for b in range(n_blocks):
        start = b * block_size
        h[start:start + ones_per_block] = 1.0
    return h

def try_optimize(h_init, n, t0, max_time, seed):
    global np_rng
    np._seed = seed
    h = h_init.copy()
    h = project_constraint(h, n)
    # Phase A: Projected gradient pre-optimization
    pg_time = min(8.0, max_time * 0.15)
    if max_time > 10:
        h = projected_gradient(h, n, max_iter=3000, temp=0.05, step_scale=1.0)
    # Phase B: L-BFGS-B with decreasing temperature
    temps = [0.05, 0.02, 0.005, 0.001, 0.0003, 0.0001]
    h, v = optimize_lbfgsb(h, n, temps, max_time, t0)
    return h, v

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h0_parent = np.full(80, 0.5)

    best_h = h0_parent.copy()
    best_c5 = compute_c5(best_h)

    n_parent = len(h0_parent)

    # Try multiple resolutions
    n_candidates = sorted(set([n_parent, n_parent // 2, n_parent // 2 + 20,
                                max(40, n_parent // 4), max(60, n_parent // 3)]))
    n_candidates = [x for x in n_candidates if 30 <= x <= 400]

    # Sort by expected difficulty (smaller n first for quick wins)
    n_candidates.sort()

    for ni in n_candidates:
        if remaining() < 40:
            break
        alloc = min(120.0, remaining() * 0.25)
        if alloc < 20:
            break
        h_init = downsample(h0_parent, ni)
        h_opt, c5_opt = try_optimize(h_init, ni, t0, alloc, seed + ni)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Try symmetric/block initializations at a few resolutions
    for ni in [n_parent, n_parent // 2, max(60, n_parent // 3)]:
        if remaining() < 30:
            break
        alloc = min(60.0, remaining() * 0.2)
        if alloc < 15:
            break
        for init_fn in [make_symmetric_init, make_block_init]:
            if remaining() < 20:
                break
            h_init = init_fn(ni)
            h_opt, c5_opt = try_optimize(h_init, ni, t0, min(40.0, remaining() - 10), seed + 500 + ni)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Random restarts at best resolution
    n_best = len(best_h)
    restart_count = 0
    while remaining() > 30 and restart_count < 80:
        alloc = min(20.0, remaining() - 15)
        if alloc < 8:
            break
        scale = 0.02 + 0.06 * (restart_count % 10) / 9.0
        h_pert = best_h + scale * np.random.randn(n_best)
        h_pert = project_constraint(h_pert, n_best)
        h_opt, c5_opt = try_optimize(h_pert, n_best, t0, alloc, seed + 2000 + restart_count)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        restart_count += 1

    # Final polish
    if remaining() > 15:
        h_opt, c5_opt = try_optimize(best_h.copy(), n_best, t0,
                                     max(10.0, remaining() - 5), seed + 9999)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Snap near-binary values and re-evaluate
    h_snap = best_h.copy()
    h_snap[h_snap < 0.05] = 0.0
    h_snap[h_snap > 0.95] = 1.0
    h_snap = project_constraint(h_snap, n_best)
    c5_snap = compute_c5(h_snap)
    if c5_snap < best_c5:
        best_c5 = c5_snap
        best_h = h_snap

    return best_h, best_c5, len(best_h)
