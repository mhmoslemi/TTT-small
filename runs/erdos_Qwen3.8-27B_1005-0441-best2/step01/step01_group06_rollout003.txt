    """Gradient of c[lag] w.r.t. h, where c = correlate(h, 1-h, full)."""
    g = np.zeros(n)
    if lag >= 0:
        # c[lag] = sum_{i=0}^{n-1-lag} h[i+lag] * (1-h[i])
        # d/dh[j]: 
        #   h[j] appears in h[i+lag] when j = i+lag, i.e., i = j-lag (need i>=0, i<n-lag)
        #   => j >= lag, j < n, and j-lag < n-lag => j < n (always)
        #   contribution: (1-h[j-lag])
        #   h[j] appears in (1-h[i]) when j = i, i.e., i = j (need j+lag < n)
        #   => j < n-lag

    dx = 2.0/n
    c = np.correlate(h, 1-h, mode='full')
    # Find top k lags
    top_indices = np.argsort(c)[-top_k:]  # largest values
    # Softmax weights
    w = np.exp(beta * (c[top_indices] - c[top_indices].max()))
    w /= w.sum()
    
    g = np.zeros(n)
    for wi, ki in zip(w, top_indices):
        lag = ki - (n-1)
        g += wi * compute_grad_at_lag(h, lag, n)
    return g

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            cnt = np.sum(mask)
            if cnt == 0:
                break
            adj = min(diff / cnt, 1.0 - h[mask] + 1e-10)
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-10
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def project_fast(h, n):

def grad_at_lag(h, lag, n):
    g = np.zeros(n)
