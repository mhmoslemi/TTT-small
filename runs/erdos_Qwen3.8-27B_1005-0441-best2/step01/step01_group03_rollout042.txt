def soft_max_grad(overlaps, h, n, temperature=0.01):
    """Compute gradient using softmax weighting of top lags."""
    # Use top lags weighted by their overlap value
    max_ov = np.max(overlaps)
    weights = np.exp((overlaps - max_ov) / temperature)
    weights /= np.sum(weights)
    
    g = np.zeros(n)
    for m, w in enumerate(weights):
        if w < 1e-6:
            continue

        if lag >= 0:

            g[k:] += (-h[:n-k])
    else:

            g[k:] += (1.0 - h[:n-k])

    return g
