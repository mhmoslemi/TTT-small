from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(200):
        d = target - np.sum(h)
        if abs(d) < 1e-15:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += resid / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    shifted = np.clip(alpha * (corr - mc), -500, 500)
    return mc + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    log_w = np.clip(alpha * (corr - mc), -500, 500)
    log_w -= np.max(log_w)
    w = np.exp(log_w)
    w /= np.sum(w)
    
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        s = m - (n - 1)
        wm = w[m]
        if wm < 1e-14:
            continue
        if s >= 0:
            i_end = n - s
            if i_end > 0:
                grad[:i_end] += wm * one_minus_h[s:]
            if s < n:
                grad[s:] -= wm * h[:n - s]
        else:
            i_start = -s
            if i_start < n:
                grad[i_start:] += wm * one_minus_h[:n - i_start]
            i_end = n + s
            if i_end > 0:
                grad[:i_end] -= wm * h[-s:]
    return grad * dx

    h[:b] = 1.0
    h[n//2 - n//8 : n//2 + n//8] = 1.0
    h[-b:] = 1.0
