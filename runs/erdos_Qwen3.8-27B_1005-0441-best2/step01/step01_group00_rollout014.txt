import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _proj(h, n):
    target = n * 0.5
    v = h.astype(float)
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    out = np.clip(v - lam, 0.0, 1.0)
    d = target - out.sum()
    if abs(d) > 1e-12:
        free = (out > 1e-14) & (out < 1.0 - 1e-14)
        if free.sum() > 0:
            out[free] += d / free.sum()
            out = np.clip(out, 0.0, 1.0)
    return out

def _obj(h, eps):
    c = _corr(h)
    dx = 2.0 / len(h)
    cmax = c.max()
    return (cmax + eps * np.log(np.exp((c - cmax) / eps).sum())) * dx

def _grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    w = np.exp((c - cmax) / eps)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        m = k - (n - 1)
        if m >= 0:
            g[:n-m] += wk * (1.0 - h[m:])
            g[m:] -= wk * h[:n-m]
        else:
            p = -m
            g[p:] += wk * (1.0 - h[:n-p])
            g[:n-p] -= wk * h[p:]
    return g * dx

def _subgrad(h):
    """Gradient of the exact max (subgradient at argmax shift)"""
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    k_star = int(np.argmax(c))
    m = k_star - (n - 1)
    g = np.zeros(n)
    if m >= 0:
        g[:n-m] += (1.0 - h[m:])
        g[m:] -= h[:n-m]
    else:
        p = -m
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return g * dx

def _eq_fun(x, n):
    return x.sum() - n * 0.5

def _eq_jac(x, n):
    return np.ones(n)

def _slsqp_phase(h, n, eps_list, t_deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
    for eps in eps_list:
        if time.time() > t_deadline:
            break
        maxit = 400 if eps > 0.01 else 600
        try:
            res = minimize(_obj, h.copy(), args=(eps,),
                          jac=_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxit, 'ftol': 1e-14})
            h = _proj(res.x, n)
        except Exception:
            pass
    return h

def _pg_phase(h, n, eps, max_iter, t_deadline):
    lr = 0.15
    f_cur = _obj(h, eps)
    for it in range(max_iter):
        if time.time() > t_deadline:
            break
        g = _grad(h, eps)
        improved = False
        for alpha in [lr, lr*0.5, lr*0.25, lr*0.1]:
            h_new = _proj(h - alpha * g, n)
            f_new = _obj(h_new, eps)
            if f_new < f_cur * (1.0 - 1e-10):
                h = h_new
                f_cur = f_new
                improved = True
                break
        if improved:
            lr = min(lr * 1.03, 0.4)
        else:
            lr *= 0.7
            if lr < 1e-7:
                break
    return h

def _subgrad_phase(h, n, max_iter, t_deadline):
    h_best = h.copy()
    v_best = _c5(h)
    lr = 0.02
    for it in range(max_iter):
        if time.time() > t_deadline:
            break
        g = _subgrad(h)
        h_new = _proj(h - lr * g, n)
        v_new = _c5(h_new)
        if v_new < v_best - 1e-10:
            h = h_new
            v_best = v_new
            h_best = h.copy()
            lr = min(lr * 1.02, 0.08)
        else:
            lr *= 0.92
            if lr < 1e-7:
                break
    return h_best, v_best

def _local_search(h, n, rng, max_iter, t_deadline, best_val):
    h_cur = h.copy()
    v_cur = _c5(h_cur)
    deltas = [0.02, -0.02, 0.05, -0.05, 0.1, -0.1, 0.15, -0.15, 0.3, -0.3]
    for it in range(max_iter):
        if time.time() > t_deadline:
            break
        idx = rng.randint(n)
        improved = False
        for d in deltas:
            h_try = h_cur.copy()
            h_try[idx] = np.clip(h_try[idx] + d, 0.0, 1.0)
            h_try = _proj(h_try, n)
            v_try = _c5(h_try)
            if v_try < v_cur - 1e-9:
                h_cur = h_try
                v_cur = v_try
                improved = True
                break
        if not improved:
            i, j = rng.choice(n, 2, replace=False)
            for flip_i in [0.0, 1.0]:
                for flip_j in [0.0, 1.0]:
                    h_try = h_cur.copy()
                    h_try[i] = flip_i
                    h_try[j] = flip_j
                    h_try = _proj(h_try, n)
                    v_try = _c5(h_try)
                    if v_try < v_cur - 1e-9:
                        h_cur = h_try
                        v_cur = v_try
                        break
                if v_cur < _c5(h):
                    break
        if v_cur < best_val:
            h = h_cur.copy()
    return h, v_cur
