import numpy as np
from scipy.optimize import minimize
import time

_CUR_N = 160

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(free):
            h[free] += d / np.count_nonzero(free)
        else:
            h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        mid = (h > 0.01) & (h < 0.99)
        if np.any(mid):
            h[mid] += resid / np.count_nonzero(mid)
            np.clip(h, 0.0, 1.0, out=h)
    return h

def obj_lse(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    return (m + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def obj_pen(x, alpha, pen):
    n = len(x)
    return obj_lse(x, alpha) + pen * (np.sum(x) - n * 0.5) ** 2

def sum_eq_fun(x):
    return np.sum(x) - _CUR_N * 0.5

def sum_eq_jac(x):
    return np.ones(len(x))

def make_default(n):
    h = np.zeros(n)
    for i in range(n):
        if (i % 4) < 2:
            h[i] = 1.0
    return project_feasible(h, n)

def subgradient_step(h, lr, n, top_k=3):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    npts = len(h)
    shifts = np.arange(len(corr)) - (npts - 1)
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(npts)
    for idx in top_idx:
        k = int(shifts[idx])
        if k >= 0:
            for i in range(npts - k):
                grad[i] += (1.0 - h[i + k])
                grad[i + k] -= h[i]
        elif k < 0:
            kk = -k
            for i in range(npts - kk):
                grad[i + kk] += (1.0 - h[i])
                grad[i] -= h[i + kk]
    h_new = h - lr * grad
    return project_feasible(h_new, npts)
