import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_binary(h, n):
    """Project to nearest binary with sum = n/2"""
    h = (h > 0.5).astype(float)
    target = n // 2
    cur = int(np.sum(h))
    if cur > target:
        # Need to turn off (cur - target) bits
        order = np.argsort(h)[::-1]  # most "on" first
        for i in range(cur - target):
            h[order[i]] = 0.0
    elif cur < target:
        order = np.argsort(h)  # most "off" first
        for i in range(target - cur):
            h[order[i]] = 1.0
    return h

def compute_correlation(h):
    """Full correlation of h with 1-h"""
    return np.correlate(h, 1.0 - h, mode="full")

def swap_delta(c, h, p, q, n):
    """Compute change in correlation vector when swapping h[p] and h[q] (binary)"""
    # Assumes h[p]=1, h[q]=0 (we swap)
    # lags range from -(n-1) to +(n-1)
    lags = np.arange(-(n-1), n)  # length 2n-1
    # For position p (was 1, becomes 0): δ = -1
    # Δc[L] = -1 * [(1-h[p-L])*I(0≤p-L≤n-1) - h[p+L]*I(0≤p+L≤n-1)]
    # For position q (was 0, becomes 1): δ = +1
    # Δc[L] = +1 * [(1-h[q-L])*I(0≤q-L≤n-1) - h[q+L]*I(0≤q+L≤n-1)]
    
    # Validity masks for p
    valid_p1 = (p - lags >= 0) & (p - lags < n)
    valid_p2 = (p + lags >= 0) & (p + lags < n)
    # Validity masks for q
    valid_q1 = (q - lags >= 0) & (q - lags < n)
    valid_q2 = (q + lags >= 0) & (q + lags < n)
    
    # Clip indices for safe access
    p_lags1 = np.clip(p - lags, 0, n-1)
    p_lags2 = np.clip(p + lags, 0, n-1)
    q_lags1 = np.clip(q - lags, 0, n-1)
    q_lags2 = np.clip(q + lags, 0, n-1)
    
    delta_p = -(1.0 - h[p_lags1]) * valid_p1 + h[p_lags2] * valid_p2
    delta_q = (1.0 - h[q_lags1]) * valid_q1 - h[q_lags2] * valid_q2
    
    return delta_p + delta_q

            break

        q = zeros[rng.randint(len(zeros))]

            c = new_c

                best_c5 = new_c5
                best_h = h.copy()
            T = max(T * cooling, T_min)
        else:
            # Reject, stay
            pass

            improved = True
            while improved and time.time() - t0 < budget_s * 0.85:

                best_dc5 = 0
                best_pair = None
                for p_idx in range(min(len(ones), 50)):

                    for q_idx in range(min(len(zeros), 50)):
                        q_pos = zeros[q_idx]
                        if p_pos == q_pos:
                            continue

                        new_max = (c + dc).max()
                        dc5 = (new_max - c.max()) * dx

                            best_pair = (p_pos, q_pos)

                    h[p], h[q] = h[q], h[p]
                    c = c + dc
                    best_c5 = c.max() * dx
                    best_h = h.copy()
                    improved = True
    
    return best_h, best_c5, c

        t_gl = min(time.time() + t_alloc * 0.6, t0 + budget_s * 0.90)

        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Phase 1: Main n (from initial)
    n_main = n_init
    h_b = to_binary(h_init, n_main)
    try_config(h_b, n_main, budget_s * 0.18, seed)

    # Phase 2: Downsampled sizes
    for n_try in [n_main // 2, n_main // 4, 80, 64, 128, n_main // 3]:
        if n_try < 32 or n_try > 512 or n_try % 2 != 0:
            continue
        if n_try == n_main:
            continue
        if time.time() - t0 > budget_s * 0.75:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_d = np.interp(x_new, x_old, best_h)
        h_bd = to_binary(h_d, n_try)
        try_config(h_bd, n_try, budget_s * 0.07, seed + n_try)

    # Phase 3: Random restarts at best n
    n_best = len(best_h)
    for ri in range(6):
        if time.time() - t0 > budget_s * 0.86:
            break
        h_rand = make_structured_init(n_best, seed + 1000 + ri)
        try_config(h_rand, n_best, budget_s * 0.035, seed + 2000 + ri)

    # Phase 4: Final greedy polish
    if time.time() - t0 < budget_s * 0.90:
        t_fin = t0 + budget_s * 0.96
        h_f, c5_f = greedy_optimize(best_h, len(best_h), t_fin)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    n_final = len(best_h)
    assert np.sum(best_h) == n_final / 2, f"Sum constraint violated: {np.sum(best_h)}"
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    try:
        val = evaluate_erdos_solution(best_h)
        if val < best_c5:
            best_c5 = val
    except Exception:
        pass
    return (best_h, best_c5, n_final)
