import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    result = np.clip(h - lam, 0.0, 1.0)
    resid = target - np.sum(result)
    if abs(resid) > 1e-13:
        mask = (result > 1e-12) & (result < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            result[mask] += resid / cnt
            np.clip(result, 0.0, 1.0, out=result)
    return result

def build_indices(n):
    js = np.arange(n)[:, None]
    ms = np.arange(2 * n - 1)[None, :]
    idx1 = js + ms - (n - 1)
    m1 = (idx1 >= 0) & (idx1 < n)
    i1 = np.clip(idx1, 0, n - 1)
    idx2 = js + (n - 1) - ms
    m2 = (idx2 >= 0) & (idx2 < n)
    i2 = np.clip(idx2, 0, n - 1)
    return m1.astype(np.float64), i1, m2.astype(np.float64), i2

def adam_step(h, alpha, lr, m_v, v_v, t, mk1, id1, mk2, id2, n, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cd = corr * dx
    mx = np.max(cd)
    sh = alpha * (cd - mx)
    np.clip(sh, -500.0, 500.0, out=sh)
    es = np.exp(sh)
    sw = np.sum(es)
    w = es / sw
    obj = mx + np.log(sw) / alpha
    T1 = (1.0 - mk1 * h[id1]) @ w
    T2 = (mk2 * h[id2]) @ w
    grad = dx * (T1 - T2)
    b1, b2, ep = 0.85, 0.999, 1e-8
    m_v[:] = b1 * m_v + (1.0 - b1) * grad
    v_v[:] = b2 * v_v + (1.0 - b2) * (grad * grad)
    mh = m_v / (1.0 - b1 ** t)
    vh = v_v / (1.0 - b2 ** t)
    h_new = h - lr * mh / (np.sqrt(vh) + ep)
    h_new = project_feasible(h_new, n)
    return h_new, obj

    c5_best = compute_c5(h_best)
