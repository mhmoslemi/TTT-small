import numpy as np
from scipy.optimize import minimize
import time


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def compute_c5(h):
    return float(np.max(compute_overlaps(h)))


def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(200):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-10:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += resid / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max_info(overlaps, alpha):
    m = np.max(overlaps)
    shifted = np.clip(alpha * (overlaps - m), -500.0, 500.0)
    exp_s = np.exp(shifted)
    p = exp_s / np.sum(exp_s)
    val = m + np.log(np.sum(exp_s)) / alpha
    return val, p


def grad_c5_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    overlaps = compute_overlaps(h)
    _, p = smooth_max_info(overlaps, alpha)
    active = np.where(p > 1e-14)[0]
    grad = np.zeros(n)
    for idx in active:
        L = int(idx) - (n - 1)
        w = p[idx] * dx
        if L >= 0:
            grad[L:] += w * (1.0 - h[:n - L])
            grad[:n - L] -= w * h[L:]
        else:
            m = -L
            grad[:n - m] += w * (1.0 - h[m:])
            grad[m:] -= w * h[:n - m]
    return grad


def objective_func(x, alpha, pen):
    overlaps = compute_overlaps(x)
    val, _ = smooth_max_info(overlaps, alpha)
    s = np.sum(x) - len(x) * 0.5
    return val + pen * s * s


def grad_func(x, alpha, pen):
    g = grad_c5_smooth(x, alpha)
    s = np.sum(x) - len(x) * 0.5
    g += 2.0 * pen * s
    return g

    h[n - b:] = 1.0

import numpy as np
from scipy.optimize import minimize
import time


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def compute_c5(h):
    return float(np.max(compute_overlaps(h)))


def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(200):
        d = target - np.sum(h)
        if abs(d) < 1e-15:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-10:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += resid / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max_info(overlaps, alpha):
    m = np.max(overlaps)
    shifted = np.clip(alpha * (overlaps - m), -500.0, 500.0)
    exp_s = np.exp(shifted)
    p = exp_s / np.sum(exp_s)
    val = m + np.log(np.sum(exp_s)) / alpha
    return val, p


def grad_c5_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    overlaps = compute_overlaps(h)
    _, p = smooth_max_info(overlaps, alpha)
    grad = np.zeros(n)
    grad += p[n - 1] * dx * (1.0 - 2.0 * h)
    for m in range(1, n):
        w = 2.0 * p[n - 1 + m] * dx
        if w < 1e-15:
            continue
        grad[m:] += w * (1.0 - h[:n - m])
        grad[:n - m] -= w * h[m:]
    return grad


def objective_func(x, alpha, pen):
    overlaps = compute_overlaps(x)
    val, _ = smooth_max_info(overlaps, alpha)
    s = np.sum(x) - len(x) * 0.5
    return val + pen * s * s


def grad_func(x, alpha, pen):
    g = grad_c5_smooth(x, alpha)
    s = np.sum(x) - len(x) * 0.5
    g += 2.0 * pen * s * np.ones(len(x))
    return g

    val, _ = smooth_max_info(compute_overlaps(x), alpha)
    return val

    return np.sum(x) - len(x) * 0.5


def sum_eq_jac(x):

                                constraints=constraint2,

                hc2 = project_feasible(res2.x, len(h_best))
                c5c2 = compute_c5(hc2)
                if c5c2 < c5_best:
                    c5_best = c5c2
                    h_best = hc2.copy()
                    n = len(h_best)

    h_best = project_feasible(h_best, len(h_best))
    c5_best = compute_c5(h_best)
    return h_best, c5_best, len(h_best)
