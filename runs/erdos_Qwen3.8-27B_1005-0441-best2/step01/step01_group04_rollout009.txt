    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        c = int(np.count_nonzero(m))
        if c > 0:
            h[m] += diff / c
            h = np.clip(h, 0.0, 1.0)
    return h

class _SoftObj:
    def __init__(self, alpha, lam):
        self.alpha = alpha
        self.lam = lam
    def __call__(self, h):
        c = _corr(h)
        m = c.max()
        return m + np.log(np.sum(np.exp(self.alpha * (c - m)))) / self.alpha + self.lam * (np.sum(h) - len(h) / 2.0) ** 2

class _MMObj:
    def __call__(self, x):
        return x[-1]

class _MMIneq:
    def __call__(self, x):
        h, t = x[:-1], x[-1]
        return t - _corr(h)

class _MMEq:
    def __call__(self, x):
        h = x[:-1]
        return float(np.sum(h) - len(h) / 2.0)

def _lbnb(h0, alpha, lam, maxiter=500):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    obj = _SoftObj(alpha, lam)
    try:
        res = minimize(obj, h0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return res.x
    except Exception:
        return h0.copy()

def _slsqp_mm(h0, maxiter=500):
    n = len(h0)
    t_init = _eval_c5(h0)
    x0 = np.concatenate([h0, [t_init]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'ineq', 'fun': _MMIneq()}, {'type': 'eq', 'fun': _MMEq()}]
    try:
        res = minimize(_MMObj(), x0, method='SLSQP', constraints=cons,
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14})
        return res.x[:-1]
    except Exception:
        return h0.copy()

def _init_block(n):
    h = np.where((np.arange(n) % 4) < 2, 0.75, 0.25)
    return _project(h, n)

def _init_sin(n, f=1):
    x = np.linspace(0.0, 2.0 * np.pi * f, n, endpoint=False)
    h = 0.5 + 0.3 * np.sin(x)
    return _project(h, n)

def _init_rand(n, rng):
    return _project(rng.random(n), n)

def _try_update(h, best_h, best_c5):
    h = _project(h, len(h))
    c = _eval_c5(h)
    if c < best_c5:
        return h.copy(), c
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    best_h = None
    best_c5 = np.inf

    try:
        h_init = np.array(initial_h_values, dtype=float)
        best_h = _project(h_init, len(h_init))
        best_c5 = _eval_c5(best_h)
    except Exception:
        pass

    # Phase 1: L-BFGS-B with increasing alpha on the initial solution
    if best_h is not None:
        for alpha in [20, 50, 100, 300, 1000, 3000]:
            if time.time() - t0 > budget_s - 100:
                break
            h_new = _lbnb(best_h, alpha, 1e5, 800)
            best_h, best_c5 = _try_update(h_new, best_h, best_c5)

    # Phase 2: SLSQP minimax on best so far
    if best_h is not None and time.time() - t0 < budget_s - 80:
        best_h, best_c5 = _try_update(_slsqp_mm(best_h, 600), best_h, best_c5)

    # Phase 3: Explore other n values and initializations
    test_ns = [50, 60, 64, 70, 90, 100, 120]
    for n in test_ns:
        if time.time() - t0 > budget_s - 50:
            break
        inits = [_init_block(n), _init_sin(n, 1), _init_sin(n, 2), _init_rand(n, rng)]
        for h0 in inits:
            if time.time() - t0 > budget_s - 35:
                break
            h_new = _lbnb(h0, 50, 1e5, 400)
            h_new = _lbnb(h_new, 300, 1e5, 400)
            h_new = _lbnb(h_new, 1000, 1e5, 400)
            best_h, best_c5 = _try_update(h_new, best_h, best_c5)

    # Phase 4: Fine polish with small perturbations + high-alpha optimization
    if best_h is not None:
        max_polish = 50
        for i in range(max_polish):
            if time.time() - t0 > budget_s - 15:
                break
            scale = 0.012 / (1.0 + 0.4 * i)
            h_p = best_h + rng.normal(0.0, scale, len(best_h))
            h_new = _lbnb(h_p, 2000, 1e6, 300)
            best_h, best_c5 = _try_update(h_new, best_h, best_c5)

    # Phase 5: Final SLSQP minimax polish
    if best_h is not None and time.time() - t0 < budget_s - 10:
        best_h, best_c5 = _try_update(_slsqp_mm(best_h, 400), best_h, best_c5)

    # Phase 6: One more L-BFGS round at very high alpha
    if best_h is not None and time.time() - t0 < budget_s - 5:
        h_new = _lbnb(best_h, 5000, 1e6, 500)
        best_h, best_c5 = _try_update(h_new, best_h, best_c5)

    if best_h is None:
        best_h = _project(np.full(80, 0.5), 80)
        best_c5 = _eval_c5(best_h)

    best_h = _project(best_h, len(best_h))
    best_c5 = _eval_c5(best_h)
    return best_h, best_c5, len(best_h)
