    best_n = 70

    h_parent = None
    n_parent = 70
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.asarray(initial_h_values, dtype=float)
        n_parent = len(h_parent)

    n_candidates = [n_parent]
    for n in range(20, 251, 10):
        if n % 2 == 0 and n != n_parent:
            n_candidates.append(n)

    phase1_deadline = t0 + budget_s * 0.50
    results = []

    for n in n_candidates:
        if time.time() > phase1_deadline - 8:
            break
        n_inits = 3
        inits = make_inits(n, seed + n * 7, n_inits)
        if n == n_parent and h_parent is not None:
            inits.insert(0, h_parent.copy())
            n_inits = len(inits)
        t_avail = phase1_deadline - time.time() - 3
        t_per = max(t_avail / (n_inits + 1), 4.0)
        for mode in range(n_inits):
            if time.time() > phase1_deadline - 3:
                break
            t_alloc = min(t_per * (n_inits - mode), phase1_deadline - time.time() - 2)
            if t_alloc < 3:
                break
            h_try = project_feasible(inits[mode].copy(), n / 2.0)
            h_opt, c5_opt = grad_optimize(h_try, t_alloc, seed + n * 1000 + mode)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n
            results.append((c5_opt, n, h_opt.copy()))

    results.sort(key=lambda x: x[0])
    phase2_deadline = t0 + budget_s * 0.85
    for c5_r, n_r, h_r in results[:6]:
        if time.time() > phase2_deadline - 5:
            break
        for lr_try, nt_try in [(0.03, 6), (0.09, 10), (0.015, 4)]:
            if time.time() > phase2_deadline - 3:
                break
            t_alloc = min(phase2_deadline - time.time() - 3, 80.0)
            if t_alloc < 5:
                break
            h_opt, c5_opt = grad_optimize(h_r.copy(), t_alloc,
                                           seed + 50000 + n_r * 10 + int(lr_try * 1000),
                                           lr_init=lr_try, n_top=nt_try)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_r

    t_rem = deadline - time.time() - 1
    if t_rem > 5 and best_h is not None:
        h_pol, c5_pol = grad_optimize(best_h.copy(), t_rem, seed + 99999,
                                       lr_init=0.003, n_top=3, mom=0.9)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    if best_h is None:
        nf = n_parent if n_parent % 2 == 0 else n_parent + 1
        best_h = project_feasible(np.full(nf, 0.5), nf / 2.0)
        best_c5 = compute_c5(best_h)
        best_n = nf

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
