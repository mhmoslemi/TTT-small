    for period in [3, 4, 5, 7]:
        pat = np.where(np.arange(n) % period < (period + 1) // 2, 0.85, 0.15)
        starts.append(_proj(pat, n))
    for phase in [0.0, 0.7, 1.4]:
        hh = 0.5 + 0.46 * np.cos(np.pi * x + phase)
        starts.append(_proj(hh, n))
    return starts

def _interp_h(h_old, n_new):
    x_old = np.arange(len(h_old))
    x_new = np.linspace(0, len(h_old) - 1, n_new)
    return np.interp(x_new, x_old, h_old)

def _slsqp(h0, eps, n, maxit):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_smoobj, h0.copy(), args=(eps,),
                       jac=_smograd, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxit, 'ftol': 1e-13})
        return _proj(res.x, n)
    except Exception:
        return h0.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_val = np.inf

    def elapsed():
        return time.time() - t0

    # Phase 1: Build diverse starting pool
    pool = []
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        h_init = _proj(h_init, len(h_init))
        pool.append(h_init)
        v = _c5(h_init)
        if v < best_val:
            best_val = v
            best_h = h_init.copy()
    except Exception:
        pass

    for n in [70, 80, 90, 100, 110]:
        for s in _gen_starts(n, rng):
            pool.append(s)
            if _c5(s) < best_val:
                best_val = _c5(s)
                best_h = s.copy()

    # Sort pool by C5, keep top 12
    pool.sort(key=_c5)
    pool = pool[:12]

    # Phase 2: Multi-epsilon SLSQP on top starts
    for idx, h0 in enumerate(pool):
        if elapsed() > budget_s * 0.72:
            break
        n = len(h0)
        # Epsilon schedule: coarse to fine
        eps_list = [0.06, 0.03, 0.015, 0.007, 0.004]
        # For the parent (first entry), use smaller epsilons from the start
        if idx == 0 and len(pool) > 1 and pool[0].shape == pool[1].shape:
            eps_list = [0.04, 0.02, 0.01, 0.005, 0.003]
        for eps in eps_list:
            if elapsed() > budget_s * 0.74:
                break
            maxit = 400 + int(300 / (eps * 100))
            maxit = min(maxit, 1000)
            h_res = _slsqp(h0, eps, n, maxit)
            val = _c5(h_res)
            if val < best_val:
                best_val = val
                best_h = h_res.copy()
            h0 = h_res  # chain: next eps starts from previous result

    # Phase 3: Transfer best to nearby n values and re-optimize
    if best_h is not None and elapsed() < budget_s * 0.82:
        n_cur = len(best_h)
        for n_try in [n_cur - 10, n_cur + 10, n_cur + 20]:
            if n_try < 40 or n_try > 140 or elapsed() > budget_s * 0.86:
                continue
            h_interp = _interp_h(best_h, n_try)
            h_interp = _proj(h_interp, n_try)
            for eps in [0.03, 0.015, 0.008]:
                if elapsed() > budget_s * 0.88:
                    break
                h_res = _slsqp(h_interp, eps, n_try, 600)
                val = _c5(h_res)
                if val < best_val:
                    best_val = val
                    best_h = h_res.copy()
            # restore h_interp for next transfer
            h_interp = _proj(_interp_h(best_h if best_h.shape[0] != n_try else best_h, n_try), n_try)

    # Phase 4: Intensive refinement of the single best
    if best_h is not None and elapsed() < budget_s * 0.90:
        n = len(best_h)
        for eps in [0.003, 0.0015, 0.0008]:
            if elapsed() > budget_s * 0.93:
                break
            h_res = _slsqp(best_h, eps, n, 900)
            if _c5(h_res) < best_val:
                best_val = _c5(h_res)
                best_h = h_res

    # Phase 5: Softmax-weighted subgradient descent
    if best_h is not None and elapsed() < budget_s * 0.94:
        n = len(best_h)
        h = best_h.copy()
        lr = 0.008
        for it in range(4000):
            if elapsed() > budget_s * 0.97:
                break
            c = _corr(h)
            cmax = c.max()
            temp = 0.004
            shifted = c - cmax
            s = np.exp(shifted / temp)
            w = s / s.sum()
            g = np.zeros(n)
            for k in range(2 * n - 1):
                if w[k] < 1e-13:
                    continue
                m = k - (n - 1)
                if m >= 0:
                    g[m:] += w[k] * (1.0 - h[:n - m])
                    g[:n - m] -= w[k] * h[m:]
                else:
                    p = -m
                    g[:n - p] += w[k] * (1.0 - h[p:])
                    g[p:] -= w[k] * h[:n - p]
            h_new = _proj(h - lr * g, n)
            v = _c5(h_new)
            if v < best_val:
                best_val = v
                best_h = h_new.copy()
                h = h_new
                lr = min(lr * 1.01, 0.015)
            else:
                lr *= 0.98
                if lr < 1e-9:
                    break

    # Phase 6: Perturbation + quick re-opt
    if best_h is not None and elapsed() < budget_s * 0.93:
        n = len(best_h)
        for trial in range(3):
            if elapsed() > budget_s * 0.96:
                break
            h_pert = best_h + rng.randn(n) * 0.04
            h_pert = _proj(h_pert, n)
            h_res = _slsqp(h_pert, 0.004, n, 500)
            if _c5(h_res) < best_val:
                best_val = _c5(h_res)
                best_h = h_res

    if best_h is None:
        n = 100
        best_h = np.full(n, 0.5)
        best_val = _c5(best_h)

    best_h = _proj(best_h, len(best_h))
    best_val = _c5(best_h)
    return (best_h, best_val, len(best_h))
