    return h_opt, _eval_c5(h_opt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    best_c5 = np.inf
    best_h = None

    def remaining():
        return budget_s - (time.time() - t0)

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = None

    if h_init is not None:
        h_init = _fix_sum(h_init, len(h_init))
        best_h = h_init.copy()
        best_c5 = _eval_c5(h_init)
    else:
        n = 80
        x = np.linspace(0, 2, n)
        best_h = _fix_sum(0.5 + 0.35 * np.sin(np.pi * x) + 0.1 * np.sin(3 * np.pi * x), n)
        best_c5 = _eval_c5(best_h)

    # Phase 1: Softmax annealing (fast, low constraint overhead)
    if remaining() > 400:
        for alpha in [50, 150, 500, 1500]:
            if remaining() < 100:
                break
            h1 = _slsqp_soft(best_h, alpha, 600)
            h1 = _fix_sum(h1, len(h1))
            c = _eval_c5(h1)
            if c < best_c5:
                best_c5, best_h = c, h1.copy()

    # Phase 2: Active set min-max (efficient, fewer constraints)
    if remaining() > 300:
        h2, c2 = _active_set_optimize(best_h, 600)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 3: Full min-max SLSQP for precision
    if remaining() > 250:
        h3 = _slsqp_mm_full(best_h, 1000, 1e-14)
        h3 = _fix_sum(h3, len(h3))
        c = _eval_c5(h3)
        if c < best_c5:
            best_c5, best_h = c, h3.copy()

    # Phase 4: Perturbation + re-optimize (escape local minima)
    if remaining() > 200:
        for trial in range(20):
            if remaining() < 50:
                break
            scale = 0.005 + 0.01 * (trial % 4)
            hp = _fix_sum(best_h + rng.normal(0, scale, len(best_h)), len(best_h))
            h4, c4 = _active_set_optimize(hp, 400)
            if c4 < best_c5:
                best_c5, best_h = c4, h4

    # Phase 5: Try alternative n values
    for n_alt in [60, 70, 76, 84, 90, 100]:
        if remaining() < 100:
            break
        x = np.linspace(0, 2, n_alt)
        h_try = 0.5 + 0.35 * np.sin(np.pi * x) + 0.15 * np.sin(2.0 * np.pi * x)
        h_try = _fix_sum(h_try, n_alt)
        h_try = _slsqp_soft(h_try, 300, 400)
        h_try = _fix_sum(h_try, n_alt)
        h_try, c_try = _active_set_optimize(h_try, 400)
        if c_try < best_c5:
            best_c5, best_h = c_try, h_try

    # Phase 6: Coordinate descent fine-tuning
    if remaining() > 60:
        h6, c6 = _coord_descent(best_h, 5000, seed=seed)
        if c6 < best_c5:
            best_c5, best_h = c6, h6

    # Phase 7: Final active-set + full refinement
    if remaining() > 40:
        h7, c7 = _active_set_optimize(best_h, 800)
        if c7 < best_c5:
            best_c5, best_h = c7, h7

    # Phase 8: Final full SLSQP with tightest tolerance
    if remaining() > 25:
        h8 = _slsqp_mm_full(best_h, 500, 1e-15)
        h8 = _fix_sum(h8, len(h8))
        c = _eval_c5(h8)
        if c < best_c5:
            best_c5, best_h = c, h8.copy()

    best_h = _fix_sum(best_h, len(best_h))
    best_c5 = _eval_c5(best_h)
    return best_h, best_c5, len(best_h)
