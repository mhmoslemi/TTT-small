def project_feasible(h, n):

        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

        if cnt > 0:

            np.clip(proj, 0.0, 1.0, out=proj)

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    exp_c = np.exp(alpha * (corr - max_c))
    sum_exp = np.sum(exp_c)
    F = max_c + np.log(sum_exp) / alpha
    weights = exp_c / sum_exp
    g = np.zeros(n)
    for s in range(2 * n - 1):
        w = weights[s]
        if w < 1e-14:
            continue
        lag = s - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                g[lag:] += w * dx * (1.0 - h[:n - lag])
                g[:n - lag] -= w * dx * h[lag:]
        else:
            m = -lag
            if n - m > 0:
                g[:n - m] += w * dx * (1.0 - h[m:])
                g[m:] -= w * dx * h[:n - m]
    return F, g, max_c

def slsqp_obj(x, alpha_val):
    F, _, _ = smooth_max_grad(x, alpha_val)
    return F

def slsqp_grad(x, alpha_val):
    _, g, _ = smooth_max_grad(x, alpha_val)
    return g

def sum_eq_fun(x, n_val):
    return np.sum(x) - n_val * 0.5

def sum_eq_jac(x, n_val):
    return np.ones(len(x))

        h[:b] = 1.0

        b = n // 6

    elif pattern == 'four':

                       jac=lambda x: slsqp_grad(x, alpha),

    except Exception:
