import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def to_binary(h):
    """Project to binary with exactly n/2 ones"""
    n = len(h)
    h = np.round(h)
    h = np.clip(h, 0, 1)
    s = h.sum()
    target = n // 2
    if s == target:
        return h
    # Need to fix the count
    if s > target:
        # Change some 1s to 0s
        ones = np.where(h == 1)[0]
        to_flip = target - s  # negative
        indices = ones[:int(-to_flip)]
        h[indices] = 0
    else:
        zeros = np.where(h == 0)[0]
        to_flip = target - s
        indices = zeros[:int(to_flip)]
        h[indices] = 1
    return h

def local_search_binary(h, max_rounds=100, t_limit=None, t0=None):
    """Try all single swaps, accept best improving one each round"""
    n = len(h)
    best_c5 = compute_c5(h)
    for round_i in range(max_rounds):
        if t_limit and time.time() - t0 > t_limit:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_swap = None
        best_new_c5 = best_c5
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0, 1
                c = compute_c5(h)
                h[i], h[j] = 1, 0

                    best_new_c5 = c
                    best_swap = (i, j)
        if best_swap is None:
            break
        i, j = best_swap
        h[i], h[j] = 0, 1
        best_c5 = best_new_c5
    return h, best_c5

    return float((mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta) * dx)

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-14:
            continue
        d = (n - 1) - i
        lo1 = max(0, -d)
        hi1 = min(n, n - d)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + d:hi1 + d])
        lo2 = max(0, d)
        hi2 = min(n, d + n)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2 - d:hi2 - d]
    return grad * dx

def slsqp_opt(h, beta, maxiter=200):
    n = len(h)
    target = n / 2.0
    def obj(x):
        return smooth_val(x, beta)
    def obj_jac(x):
        return smooth_grad(x, beta)
    def con_fun(x):
        return x.sum() - target
    def con_jac(x):
        return np.ones(n)
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_jac, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        x = np.clip(res.x, 0, 1)
        x += (target - x.sum()) / n
        x = np.clip(x, 0, 1)
        return x
    except Exception:
        return h

def coord_descent(h, t_limit):
    n = len(h)
    t0 = time.time()
    target = n / 2.0
    best = compute_c5(h)
    for _ in range(100):
        if time.time() - t0 > t_limit:
            break
        improved = False
        for i in range(n):
            # Try to adjust h[i] and h[(i+1)%n] to fix sum
            for di in np.linspace(-0.05, 0.05, 11):
                if di == 0:
                    continue
                h_try = h.copy()
                h_try[i] = np.clip(h[i] + di, 0, 1)
                h_try[(i+1) % n] = np.clip(h[(i+1)%n] - (h_try[i] - h[i]), 0, 1)
                # Fix sum
                diff = target - h_try.sum()
                if abs(diff) > 1e-10:
                    # Apply to the element most in the interior
                    mid = (i+5) % n
                    h_try[mid] = np.clip(h_try[mid] + diff, 0, 1)
                    if abs(h_try.sum() - target) > 1e-8:
                        continue
                c = compute_c5(h_try)
                if c < best - 1e-12:
                    best = c
                    h = h_try
                    improved = True
                    break
        if not improved:
            break
    return h, best
