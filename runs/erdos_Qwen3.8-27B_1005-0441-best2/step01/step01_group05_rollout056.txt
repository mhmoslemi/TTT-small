def project_sum(h, target):

    cons = ({'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac},)
    for temp in [0.01, 0.003, 0.001, 0.0003]:
        if time.time() - t0 > max_time - 3:
            break
        _g_temp = temp
        h_curr = best.copy()
        for _ in range(3):
            if time.time() - t0 > max_time - 2:
                break
            try:
                res = minimize(_soft_obj, h_curr, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 300, 'ftol': 1e-13})
                h_new = np.clip(res.x, 0.0, 1.0)
                h_new = project_sum(h_new, target)
                v = compute_c5(h_new)
                if v < best_val - 1e-11:
                    best = h_new.copy()
                    best_val = v
                h_curr = h_new.copy()
            except Exception:
                break
    return best, best_val

def make_random_init(n, seed, density=0.5):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    n_ones = n // 2
    h[rng.choice(n, n_ones, replace=False)] = 1.0
    # Add structure: sort in blocks
    h = h[rng.permutation(n)]
    return h

def make_block_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    n_ones = n // 2
    pos = 0
    while n_ones > 0 and pos < n:
        run_len = min(rng.randint(1, max(2, n // 8)), n_ones, n - pos)
        h[pos:pos + run_len] = 1.0
        n_ones -= run_len
        gap = rng.randint(1, max(2, n // 10))
        pos += run_len + gap
    if n_ones > 0:
        zeros = np.where(h == 0)[0]
        pick = rng.choice(zeros, min(n_ones, len(zeros)), replace=False)
        h[pick] = 1.0
    return h

def make_symmetric_init(n, seed):
    rng = np.random.RandomState(seed)
    half_n = n // 2
    h_half = np.zeros(half_n)
    n_ones_half = half_n // 2
    h_half[rng.choice(half_n, n_ones_half, replace=False)] = 1.0
    if n % 2 == 0:
        h = np.concatenate([h_half, h_half[::-1]])
    else:
        h = np.concatenate([h_half, [0.0], h_half[::-1]])
    return h[:n]

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    else:
        n = 160
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    remaining_budget = budget_s

    # Phase 1: Heavy coordinate descent on initial
    t_rem = remaining_budget - (time.time() - t0)
    if t_rem > 50:
        sweeps = min(1000, int(t_rem * 5))
        h_opt, c5_opt = coord_descent(h0, n_sweeps=sweeps, n_grid=121, seed=seed)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 2: Alternative initializations
    inits = [
        make_random_init(n, seed + 1),
        make_block_init(n, seed + 2),
        make_symmetric_init(n, seed + 3),
        make_block_init(n, seed + 4),
        make_random_init(n, seed + 5),
    ]
    t_rem = remaining_budget - (time.time() - t0)
    for idx, h_init in enumerate(inits):
        if t_rem < 30:
            break
        t_rem = remaining_budget - (time.time() - t0)
        sweeps = min(400, int(t_rem * 2))
        h_opt, c5_opt = coord_descent(h_init, n_sweeps=sweeps, n_grid=81, seed=seed + 200 + idx)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        t_rem = remaining_budget - (time.time() - t0)
        if t_rem < 20:
            break

    # Phase 3: SLSQP polish from best
    t_rem = remaining_budget - (time.time() - t0)
    if t_rem > 30:
        h_opt, c5_opt = slsqp_polish(best_h, n, max_time=min(80, t_rem - 10), seed=seed + 500)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 4: Fine coordinate descent (high grid resolution)
    t_rem = remaining_budget - (time.time() - t0)
    if t_rem > 40:
        sweeps = min(2000, int(t_rem * 3))
        h_opt, c5_opt = coord_descent(best_h, n_sweeps=sweeps, n_grid=201, seed=seed + 888)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 5: Random perturbations + coordinate descent (escape local optima)
    t_rem = remaining_budget - (time.time() - t0)
    n_cur = len(best_h)
    rng = np.random.RandomState(seed + 4242)
    restart = 0
    while t_rem > 30 and restart < 30:
        scale = 0.02 + 0.03 * (restart % 5) / 4.0
        h_pert = best_h + scale * rng.randn(n_cur)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = project_sum(h_pert, n_cur / 2.0)
        h_opt, c5_opt = coord_descent(h_pert, n_sweeps=150, n_grid=101, seed=seed + 2000 + restart)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        restart += 1
        t_rem = remaining_budget - (time.time() - t0)

    # Phase 6: Final SLSQP
    t_rem = remaining_budget - (time.time() - t0)
    if t_rem > 25:
        h_opt, c5_opt = slsqp_polish(best_h, n, max_time=min(50, t_rem - 5), seed=seed + 7777)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 7: Final fine coordinate descent
    t_rem = remaining_budget - (time.time() - t0)
    if t_rem > 30:
        h_opt, c5_opt = coord_descent(best_h, n_sweeps=2000, n_grid=301, seed=seed + 9999)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Final projection to ensure exact feasibility
    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
