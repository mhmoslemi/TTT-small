def grid_min_1d(c0, v_old, consts, slopes, center, n_grid=400):
    xs = np.linspace(0.0, 1.0, n_grid)
    vals = consts[None, :] + (xs[:, None] * slopes[None, :])
    val0_col = c0 + (xs - v_old) * (1.0 - xs - v_old)
    vals[:, center] = val0_col
    best_idx = np.argmin(vals.max(axis=1))
    best_val = vals.max(axis=1)[best_idx]
    # Refine around best with finer grid
    x_lo = max(0.0, xs[best_idx] - (xs[1] - xs[0]) * 5)
    x_hi = min(1.0, xs[best_idx] + (xs[1] - xs[0]) * 5)
    xs2 = np.linspace(x_lo, x_hi, 200)
    vals2 = consts[None, :] + (xs2[:, None] * slopes[None, :])
    val0_col2 = c0 + (xs2 - v_old) * (1.0 - xs2 - v_old)
    vals2[:, center] = val0_col2
    best_idx2 = np.argmin(vals2.max(axis=1))
    return float(xs2[best_idx2])

def coord_descent(h, n, passes=2):
    center = n - 1
    m = 2 * n - 1
    for _ in range(passes):
        for i in range(n):
            v_old = h[i]
            corr = np.correlate(h, 1.0 - h, mode='full')
            c0 = corr[center]
            slopes = np.ones(m)
            for k in range(m):
                r = k - center
                if r == 0:
                    continue
                hp = h[i + r] if 0 <= i + r < n else 0.0
                hm = h[i - r] if 0 <= i - r < n else 0.0
                slopes[k] = 1.0 - hp - hm
            consts = corr - v_old * slopes
            consts[center] = c0
            v_new = grid_min_1d(c0, v_old, consts, slopes, center)
            h[i] = v_new
        h = project_feasible(h, n)
    return h

def make_two_block(n):
    h = np.zeros(n)
    b = n // 4
    h[:b] = 1.0
    h[-b:] = 1.0
    return project_feasible(h, n)

def make_three_block(n):
    h = np.zeros(n)
    b = max(1, n // 6)
    h[:b] = 1.0
    h[n // 2 - b:n // 2 + b] = 1.0
    h[-b:] = 1.0
    return project_feasible(h, n)

def make_sine(n):
    x = np.linspace(0, 2 * np.pi, n, endpoint=False)
    h = 0.5 + 0.5 * np.cos(x)
    return project_feasible(h, n)

def round_to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n // 2]] = 1.0
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h_best = make_two_block(80)
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    bounds = [(0.0, 1.0)] * n

    def slsqp_opt(h_init, n_pts, alpha, maxiter):
        res = minimize(
            obj_smooth, h_init.copy(), args=(alpha,),
            method='SLSQP',
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        return project_feasible(res.x, n_pts)

    # Phase 1: SLSQP with increasing alpha
    for alpha in [20, 60, 150, 400, 1200]:
        if time.time() - t0 > budget_s * 0.25:
            break
        hc = slsqp_opt(h_best, n, alpha, 1200)
        c5c = compute_c5(hc)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()

    # Phase 2: Coordinate descent
    if time.time() - t0 < budget_s * 0.35:
        h_cd = coord_descent(h_best.copy(), n, passes=4)
        c5cd = compute_c5(h_cd)
        if c5cd < c5_best:
            c5_best = c5cd
            h_best = h_cd.copy()

    # Phase 3: SLSQP from improved point with high alpha
    for alpha in [500, 2000, 8000]:
        if time.time() - t0 > budget_s * 0.50:
            break
        hc = slsqp_opt(h_best, n, alpha, 1000)
        c5c = compute_c5(hc)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()

    # Phase 4: Coordinate descent again
    if time.time() - t0 < budget_s * 0.60:
        h_cd = coord_descent(h_best.copy(), n, passes=5)
        c5cd = compute_c5(h_cd)
        if c5cd < c5_best:
            c5_best = c5cd
            h_best = h_cd.copy()

    # Phase 5: Binary rounding + re-optimize
    if time.time() - t0 < budget_s * 0.70:
        h_bin = round_to_binary(h_best, n)
        h_round = h_bin * 0.9 + 0.05  # soften slightly for optimization
        h_round = project_feasible(h_round, n)
        hc = slsqp_opt(h_round, n, 500, 800)
        c5c = compute_c5(hc)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()
            h_cd = coord_descent(h_best.copy(), n, passes=3)
            c5cd = compute_c5(h_cd)
            if c5cd < c5_best:
                c5_best = c5cd
                h_best = h_cd.copy()

    # Phase 6: Random restarts with diverse structures
    start_structs = [
        h_best.copy(),
        make_two_block(n),
        make_three_block(n),
        make_sine(n),
    ]
    for sp in start_structs:
        if time.time() - t0 > budget_s * 0.85:
            break
        for alpha in [100, 500]:
            if time.time() - t0 > budget_s * 0.85:
                break
            hc = slsqp_opt(sp, n, alpha, 600)
            c5c = compute_c5(hc)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc.copy()
                h_cd = coord_descent(h_best.copy(), n, passes=2)
                c5cd = compute_c5(h_cd)
                if c5cd < c5_best:
                    c5_best = c5cd
                    h_best = h_cd.copy()

    # Phase 7: Perturbation restarts
    while time.time() - t0 < budget_s * 0.92:
        pert = 0.02 + 0.06 * rng.random()
        h_p = h_best + pert * rng.standard_normal(n)
        h_p = project_feasible(h_p, n)
        hc = slsqp_opt(h_p, n, 300, 500)
        c5c = compute_c5(hc)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()
            h_cd = coord_descent(h_best.copy(), n, passes=2)
            c5cd = compute_c5(h_cd)
            if c5cd < c5_best:
                c5_best = c5cd
                h_best = h_cd.copy()

    # Phase 8: Final coordinate descent polish
    if time.time() - t0 < budget_s * 0.98:
        h_cd = coord_descent(h_best.copy(), n, passes=3)
        c5cd = compute_c5(h_cd)
        if c5cd < c5_best:
            c5_best = c5cd
            h_best = h_cd.copy()

    h_best = project_feasible(h_best, len(h_best))
    c5_best = compute_c5(h_best)
    return h_best, c5_best, len(h_best)
