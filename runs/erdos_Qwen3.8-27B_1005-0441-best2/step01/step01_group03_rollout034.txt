import numpy as np
import time

def compute_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def get_c5(h, n):
    return float(np.max(compute_overlaps(h, n)))

def project_binary(h, n):
    """Project to binary with exactly n/2 ones using threshold + adjustment."""
    m = n // 2
    # Find threshold
    sorted_h = np.sort(h)[::-1]
    h_bin = (h >= sorted_h[m-1]).astype(float)
    # Ensure exactly m ones
    cnt = int(h_bin.sum())
    if cnt > m:
        # Need to flip (cnt - m) ones to zeros, pick the smallest h values among ones
        one_idx = np.where(h_bin > 0.5)[0]
        # Sort by h value ascending
        order = np.argsort(h[one_idx])
        for i in range(cnt - m):
            h_bin[one_idx[order[i]]] = 0
    elif cnt < m:
        # Need to flip (m - cnt) zeros to ones, pick the largest h values among zeros
        zero_idx = np.where(h_bin < 0.5)[0]
        order = np.argsort(-h[zero_idx])
        for i in range(m - cnt):
            h_bin[zero_idx[order[i]]] = 1
    return h_bin

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-14:
        mask = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        cnt = float(np.sum(mask))
        if cnt > 0:
            hp[mask] = np.clip(hp[mask] + diff / cnt, 0.0, 1.0)
    return hp

def continuous_refine(h, n, deadline, rng):
    """Gradient descent refinement on continuous [0,1] values."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = get_c5(h, n)
    lr = 0.3
    stalled = 0
    
    while time.time() < deadline:
        ov = compute_overlaps(h, n)
        k_max = int(np.argmax(ov))
        lag = k_max - (n - 1)
        
        # Gradient of overlap at lag with respect to h
        g = np.zeros(n)
        for j in range(n):
            jl = j - lag
            if 0 <= jl < n:
                g[j] += 1.0 - h[jl]

                g[j] -= h[jr]

  - The terms involving j as k+d: h[k](1-h[j]) was h[k]*1, now h[k]*0. Change: -h[k] where k=j-d. Change: -h[j-d]

where terms are 0 if the index is out of bounds.

import time

def compute_ov_counts(h, n):
    """Compute overlap counts (unscaled) using correlation."""
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def get_c5(h, n):
    dx = 2.0 / n
    return float(np.max(compute_ov_counts(h, n))) * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-14:
        mask = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        cnt = float(np.sum(mask))
        if cnt > 0:
            hp[mask] = np.clip(hp[mask] + diff / cnt, 0.0, 1.0)
    return hp

    idx = np.argsort(-h)[:m]

    b[idx] = 1.0
