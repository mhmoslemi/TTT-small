def binary_local_search(h, n, max_passes=100):
    h = h.copy()
    dx = 2.0 / n
    improved = True
    passes = 0
    while improved and passes < max_passes:
        improved = False
        passes += 1
        # Compute current correlations
        c = np.correlate(h, 1-h, mode="full")
        current_max = c.max() * dx
        
        # Try flipping each bit
        for j in range(n):
            old_val = h[j]
            h[j] = 1 - old_val
            # Check sum constraint
            if np.sum(h) != n/2:
                h[j] = old_val
                continue
            c_new = np.correlate(h, 1-h, mode="full")
            new_max = c_new.max() * dx
            if new_max < current_max - 1e-15:
                current_max = new_max
                improved = True
                c = c_new
            else:
                h[j] = old_val
    
    return h, current_max

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.9999999
            cnt = np.sum(mask)
            if cnt == 0: break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-9
            cnt = np.sum(mask)
            if cnt == 0: break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def c_fun(x):
    return np.sum(x) - len(x) / 2.0

def c_jac(x):
    return np.ones(len(x))

def lse_obj(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(c)
    c_s = c - m
    return ((m + np.log(np.sum(np.exp(beta * c_s)))) / beta) * dx

def do_slsqp(h_in, n, beta, maxiter, ftol):
    h = project_feasible(h_in, n)
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(lse_obj, h, args=(n, beta), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def to_binary(h, n):
    h_bin = (h > 0.5).astype(float)
    need = n / 2.0 - np.sum(h_bin)
    if need > 0:
        order = np.argsort(0.5 - h)  # closest to 0.5 from below first
        for i in range(int(need)):
            h_bin[order[i]] = 1.0
    elif need < 0:
        order = np.argsort(0.5 - h)  # most negative = furthest above 0.5
        for i in range(int(-need)):
            h_bin[order[-1-i]] = 0.0

def bin_local_search(h_bin, n, max_passes, deadline):
    h = h_bin.astype(float).copy()
    dx = 2.0 / n
    nfull = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode="full")
    cur_max = c.max()

    D = np.zeros((n, nfull))
    for j in range(n):
        for i in range(j, min(j + n, nfull)):
            D[j, i] += 1.0 - h[j + n - 1 - i]
        for i in range(n - 1 - j, min(2 * n - 1 - j, nfull)):
            D[j, i] -= h[j + i - (n - 1)]
    
    ones_mask = (h == 1.0)
    zeros_mask = (h == 0.0)
    improved = True
    passes = 0
    
    while improved and passes < max_passes:
        if time.time() > deadline:
            break
        improved = False
        passes += 1
        ones_idx = np.where(ones_mask)[0]
        zeros_idx = np.where(zeros_mask)[0]
        
        best_pair = None
        best_val = cur_max
        best_new_c = None
        
        for j in ones_idx:
            base = c - D[j]
            new_cs = base[None, :] + D[zeros_idx, :]
            new_maxs = new_cs.max(axis=1)
            mi = np.argmin(new_maxs)
            if new_maxs[mi] < best_val - 1e-15:
                best_val = new_maxs[mi]
                best_pair = (j, zeros_idx[mi])
                best_new_c = new_cs[mi].copy()
        
        if best_pair is not None:

            h[j] = 0.0
            h[k] = 1.0
            ones_mask[j] = False
            zeros_mask[j] = True
            zeros_mask[k] = False
            ones_mask[k] = True
            c = best_new_c
            cur_max = best_val
            improved = True
    
    return h, cur_max * dx
