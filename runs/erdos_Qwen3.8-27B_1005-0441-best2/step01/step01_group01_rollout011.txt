def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def soft_max_obj(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c = corr * dx
    m = np.max(c)
    shifted = alpha * (c - m)
    shifted = np.clip(shifted, -350.0, 350.0)
    return m + np.log(np.sum(np.exp(shifted))) / alpha
