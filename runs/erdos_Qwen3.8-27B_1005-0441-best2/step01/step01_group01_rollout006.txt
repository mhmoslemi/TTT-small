import numpy as np
from scipy.optimize import minimize
import time

_TARGET_GLOBAL = 0.0

def _sum_constraint(x):
    return np.sum(x) - _TARGET_GLOBAL

def _sum_constraint_jac(x):
    return np.ones(len(x))

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-10:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += resid / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -600.0, 600.0)
    sm = m + np.log(np.sum(np.exp(shifted))) / alpha
    return sm * dx

def penalty_obj(h, alpha, mu):
    n = len(h)
    base = smooth_max_obj(h, alpha)
    pen = mu * (np.sum(h) - n * 0.5) ** 2
    return base + pen

def make_two_block(n):
    h = np.zeros(n)
    block = n // 4
    h[:block] = 1.0
    h[-block:] = 1.0
    return project_feasible(h, n)

def make_three_block(n):
    h = np.zeros(n)
    b = n // 6
    h[:b] = 1.0
    h[n//2 - b:n//2 + b] = 1.0
    h[-b:] = 1.0
    return project_feasible(h, n)

def make_four_block(n):
    h = np.zeros(n)
    b = n // 8
    for k in range(4):
        start = k * n // 4
        h[start:start+b] = 1.0
    return project_feasible(h, n)

def make_perturbed(h, rng, scale, local_frac):
    n = len(h)
    h_p = h.copy()
    if local_frac > 0:
        center = rng.integers(0, n)
        width = max(int(n * local_frac), 3)
        for i in range(2 * width):
            idx = (center + i - width) % n
            h_p[idx] += scale * rng.standard_normal()
    h_p += scale * 0.3 * rng.standard_normal(n)
    return project_feasible(h_p, n)

def run(seed=42, budget_s=1100, **kwargs):
    global _TARGET_GLOBAL
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        h0 = make_two_block(160)
    n = len(h0)
    _TARGET_GLOBAL = n * 0.5
    h_best = project_feasible(h0.copy(), n)
    c5_best = compute_c5(h_best)

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_constraint_jac}]

    # Phase 1: Progressive SLSQP with increasing alpha
    for alpha in [200.0, 500.0, 1000.0, 2000.0, 4000.0]:
        if time.time() - t0 > budget_s * 0.30:
            break
        res = minimize(
            smooth_max_obj, h_best.copy(), args=(alpha,),
            method='SLSQP', bounds=bounds, constraints=constraints,
            options={'maxiter': 3000, 'ftol': 1e-16}
        )
        hc = project_feasible(res.x, n)
        c5c = compute_c5(hc)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()

    # Phase 2: L-BFGS-B with increasing penalty
    for mu in [2000.0, 10000.0, 50000.0, 200000.0]:
        if time.time() - t0 > budget_s * 0.45:
            break
        res = minimize(
            penalty_obj, h_best.copy(), args=(1000.0, mu),
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': 600, 'ftol': 1e-18, 'gtol': 1e-15}
        )
        hc = project_feasible(res.x, n)
        c5c = compute_c5(hc)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()

    # Phase 3: Targeted multi-restart with SLSQP
    for trial in range(20):
        if time.time() - t0 > budget_s * 0.72:
            break
        if trial % 4 == 0:
            h_pert = make_perturbed(h_best, rng, 0.03, 0.0)
        elif trial % 4 == 1:
            h_pert = make_perturbed(h_best, rng, 0.05, 0.12)
        elif trial % 4 == 2:

        else:
            h_pert = make_perturbed(h_best, rng, 0.015, 0.06)
        alpha = 1500.0 + 500.0 * (trial % 4)
        res = minimize(
            smooth_max_obj, h_pert, args=(alpha,),
            method='SLSQP', bounds=bounds, constraints=constraints,
            options={'maxiter': 2500, 'ftol': 1e-15}
        )
        hc = project_feasible(res.x, n)
        c5c = compute_c5(hc)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()

        if time.time() - t0 > budget_s * 0.82:
            break
        for alpha in [800.0, 2000.0]:
            if time.time() - t0 > budget_s * 0.84:
                break
            res = minimize(
                smooth_max_obj, sp.copy(), args=(alpha,),
                method='SLSQP', bounds=bounds, constraints=constraints,
                options={'maxiter': 2000, 'ftol': 1e-14}
            )
            hc = project_feasible(res.x, n)
            c5c = compute_c5(hc)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc.copy()

    # Phase 5: Final polish with very high alpha
    if time.time() - t0 < budget_s * 0.90:
        for alpha in [3000.0, 6000.0]:
            if time.time() - t0 > budget_s * 0.92:
                break
            res = minimize(
                penalty_obj, h_best.copy(), args=(alpha, 100000.0),
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': 1500, 'ftol': 1e-18, 'gtol': 1e-16}
            )
            hc = project_feasible(res.x, n)
            c5c = compute_c5(hc)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc.copy()

    # Phase 6: Try slightly different grid sizes
    if time.time() - t0 < budget_s - 120:
        for n_trial in [n + 40, n - 40]:
            if time.time() - t0 > budget_s - 80:
                break
            if n_trial < 60 or n_trial > 400:
                continue
            _TARGET_GLOBAL = n_trial * 0.5
            x_old = np.arange(n, dtype=float)
            x_new = np.linspace(0.0, float(n - 1), n_trial)
            h_up = np.interp(x_new, x_old, h_best)
            h_up = project_feasible(h_up, n_trial)
            bounds2 = [(0.0, 1.0)] * n_trial
            constraints2 = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_constraint_jac}]
            best_trial = (c5_best, h_best.copy(), n)
            for alpha in [1000.0, 3000.0]:
                if time.time() - t0 > budget_s - 60:
                    break
                res = minimize(
                    smooth_max_obj, h_up.copy(), args=(alpha,),
                    method='SLSQP', bounds=bounds2, constraints=constraints2,
                    options={'maxiter': 2500, 'ftol': 1e-14}
                )
                hc = project_feasible(res.x, n_trial)
                c5c = compute_c5(hc)
                if c5c < best_trial[0]:
                    best_trial = (c5c, hc.copy(), n_trial)
            if best_trial[0] < c5_best:
                c5_best, h_best, n = best_trial
                _TARGET_GLOBAL = n * 0.5
                bounds = [(0.0, 1.0)] * n
                constraints = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_constraint_jac}]

    # Final feasibility check
    h_best = project_feasible(h_best, len(h_best))
    c5_best = compute_c5(h_best)
    return h_best, c5_best, len(h_best)
