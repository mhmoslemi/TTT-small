import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _fix_sum(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            i = int(np.argmax(1.0 - h))
            add = min(diff, 1.0 - h[i])
            if add < 1e-15:
                break
            h[i] += add
        else:
            i = int(np.argmin(h))
            sub = min(-diff, h[i])
            if sub < 1e-15:
                break
            h[i] -= sub
    return h

def _smooth_c5(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = c.max()
    if alpha > 1e6:
        return cm
    return cm + np.log(np.sum(np.exp(alpha * (c - cm)))) / alpha

def _slqp_obj(x, alpha):
    return _smooth_c5(np.clip(x, 0, 1), alpha)

def _slqp_step(h0, alpha, max_it):
    n = len(h0)
    try:
        res = minimize(_slqp_obj, h0, args=(alpha,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_it, 'ftol': 1e-13})
        return _fix_sum(res.x, n)
    except Exception:
        return h0.copy()

def _continuation(h0, alphas, max_it, t0, budget):
    h = h0.copy()
    for alpha in alphas:
        if time.time() - t0 > budget - 30:
            break
        h_new = _slqp_step(h, alpha, max_it)
        if _eval_c5(h_new) <= _eval_c5(h) + 1e-14:
            h = h_new
    return h

def _gen_starts(n, rng):
    starts = [rng.random(n)]
    for ratio in [0.3, 0.4, 0.5, 0.6, 0.7]:
        hb = np.zeros(n)
        hb[:int(n * ratio)] = 1.0
        starts.append(hb.copy())
    for p in [3, 4, 5, 7, 8]:
        for duty in range(1, p):
            pat = np.array([1.0 if (i % p) < duty else 0.0 for i in range(p)])
            starts.append(np.tile(pat, n // p + 1)[:n].copy())
    x = np.linspace(0, 4 * np.pi, n, endpoint=False)
    starts.append(0.5 + 0.45 * np.sin(x))
    phi = (1 + np.sqrt(5)) / 2
    xx = np.linspace(0, 2, n)
    starts.append(((xx % (1 / phi)) < (0.5 / phi)).astype(float))
    return [_fix_sum(s, n) for s in starts]

def _de_obj(x, n):
    hc = np.clip(x, 0, 1)
    dx = 2.0 / n
    c = np.correlate(hc, 1.0 - hc, mode='full') * dx
    return c.max() + 0.05 * np.std(c)

def _binary_refine(h, t0, budget):
    n = len(h)
    h = h.copy()
    c5 = _eval_c5(h)
    for _ in range(5):
        improved = False
        order = list(range(n))
        np.random.shuffle(order)
        for i in order:
            if time.time() - t0 > budget - 30:
                return h
            for val in [0.0, 1.0]:
                if abs(h[i] - val) < 1e-4:
                    continue
                saved = h[i]
                h[i] = val
                h = _fix_sum(h, n)
                c5_new = _eval_c5(h)
                if c5_new < c5 - 1e-12:
                    c5 = c5_new
                    improved = True
                else:
                    h[i] = saved
                    h = _fix_sum(h, n)
                    break
        if not improved:
            break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    best_c5 = np.inf
    best_h = None

    def _try(h):
        nonlocal best_c5, best_h
        h = _fix_sum(h, len(h))
        c5 = _eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Load initial
    try:
        h0 = np.array(initial_h_values, dtype=float)
        h0 = _fix_sum(h0, len(h0))
        _try(h0)
    except Exception:
        h0 = None

    # Continuation on initial
    if h0 is not None:
        h_opt = _continuation(h0, [50, 100, 200, 500, 1000, 3000, 10000], 400, t0, budget_s)
        _try(h_opt)

    # Multiple n, structured starts
    for n in [40, 50, 60, 80, 100]:
        if time.time() - t0 > budget_s - 180:
            break
        for h_s in _gen_starts(n, rng):
            if time.time() - t0 > budget_s - 60:
                break
            h_opt = _continuation(h_s, [80, 200, 800, 3000], 250, t0, budget_s)
            _try(h_opt)

    # Differential evolution for small n
    for n in [24, 32, 40, 48]:
        if time.time() - t0 > budget_s - 250:
            break
        try:
            res = differential_evolution(_de_obj, bounds=[(0, 1)] * n, args=(n,),
                                         seed=int(rng.integers(100000)), maxiter=150,
                                         tol=1e-9, popsize=8, polish=False)
            h_de = _fix_sum(res.x, n)
            h_pol = _continuation(h_de, [500, 2000, 8000], 250, t0, budget_s)
            _try(h_pol)
        except Exception:
            pass

    # Perturbation + re-optimize on best
    if best_h is not None:
        for trial in range(60):
            if time.time() - t0 > budget_s - 60:
                break
            scale = 0.08 * max(1.0, 1.0 - trial / 60.0)
            h_p = best_h + rng.normal(0, scale, len(best_h))
            h_p = _fix_sum(h_p, len(best_h))
            h_opt = _continuation(h_p, [200, 800, 3000], 200, t0, budget_s)
            _try(h_opt)

    # Binary refinement on best
    if best_h is not None and time.time() - t0 < budget_s - 25:
        h_bin = _binary_refine(best_h, t0, budget_s)
        _try(h_bin)

    # Final high-alpha polish
    if best_h is not None and time.time() - t0 < budget_s - 15:
        h_final = _continuation(best_h, [5000, 20000, 100000], 400, t0, budget_s)
        _try(h_final)

    if best_h is None:
        best_h = _fix_sum(np.full(40, 0.5), 40)
        best_c5 = _eval_c5(best_h)

    return best_h, best_c5, len(best_h)
