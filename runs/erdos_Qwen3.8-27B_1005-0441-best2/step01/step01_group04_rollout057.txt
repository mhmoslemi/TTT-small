import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo, hi = -5.0, 5.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _fix_sum(h, n):
    """Fix sum to n/2 by redistributing."""
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            max_room = room.max()
            if max_room < 1e-15:
                break
            h += (diff / max_room) * (room / room.sum()) if room.sum() > 0 else 0
            h = np.clip(h, 0, 1)
        else:
            h += diff / h.sum() * h if h.sum() > 0 else 0
            h = np.clip(h, 0, 1)
    return h

def _softmax_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = corr.max()
    shifted = alpha * (corr - cmax)
    shifted = np.clip(shifted, -100, 0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-30:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w = e / s
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - n + 1
        js = max(0, -lag)
        je = min(n-1, n-1-lag)
        if js <= je:
            grad[js:je+1] += wk * (1.0 - h[js+lag:je+1+lag])
        js2 = max(0, lag)
        je2 = min(n-1, n-1+lag)
        if js2 <= je2:
            grad[js2:je2+1] -= wk * h[js2-lag:je2+1-lag]
    grad *= dx
    return grad, cmax

def _proj_subgrad(h0, alphas, iters, lr_base):
    n = len(h0)
    h = h0.copy()
    best = h.copy()
    best_c5 = _eval_c5(h)
    for alpha in alphas:
        lr = lr_base / (1.0 + 0.05 * alpha)
        for it in range(iters):
            grad, _ = _softmax_grad(h, alpha)
            h = h - lr * grad
            h = _project(h, n)
            c5 = _eval_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best = h.copy()
        # anneal: cool down
    return best, best_c5

def _slsqp_minimax(h0):
    n = len(h0)
    t0 = _eval_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    
    def obj(x):
        return x[-1]
    
    def ineq(x):
        h = x[:-1]
        t = x[-1]
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - corr
    
    def eq(x):
        return x[:-1].sum() - n / 2.0
    
    try:
        res = minimize(obj, x0, method='SLSQP',
                       constraints=[{'type': 'ineq', 'fun': ineq},
                                    {'type': 'eq', 'fun': eq}],
                       bounds=bounds,
                       options={'maxiter': 300, 'ftol': 1e-10})
        if res.success or res.fun < t0:
            h_res = _project(res.x[:-1], n)
            return h_res
    except:
        pass
    return h0.copy()

def _slsqp_softmax(h0, alpha, maxiter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    
    def obj(h):
        corr = _corr_vals(h)
        cmax = corr.max()
        shifted = alpha * (corr - cmax)
        shifted = np.clip(shifted, -100, 0)
        return cmax + np.log(np.exp(shifted).sum()) / alpha
    
    def eq(h):
        return h.sum() - n / 2.0
    
    try:
        res = minimize(obj, h0, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': eq}],
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.fun < _eval_c5(h0) + 0.01:
            return _project(res.x, n)
    except:
        pass
    return h0.copy()

def _make_periodic(n, pattern):
    """Create periodic pattern of given length, repeated to fill n."""
    p = len(pattern)
    h = np.tile(pattern, n // p + 1)[:n]
    return _fix_sum(h, n)

def _make_sine(n, freq, amp):
    x = np.arange(n)
    h = 0.5 + amp * np.sin(2 * np.pi * freq * x / n)
    return _fix_sum(h, n)

def _interpolate(h_coarse, n_fine):
    """Interpolate coarse step function to fine grid."""
    n_coarse = len(h_coarse)
    x_coarse = np.linspace(0, 2, n_coarse, endpoint=False) + 1.0/n_coarse
    x_fine = np.linspace(0, 2, n_fine, endpoint=False) + 1.0/n_fine
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    return _fix_sum(h_fine, n_fine)

import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(h, n):
    h = np.asarray(h, dtype=float)
    lo, hi = -5.0, 5.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > n / 2.0:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _softmax_grad(h, alpha, topk=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = corr.max()
    if topk and topk < len(corr):
        idx = np.argsort(corr)[-topk:]
        w = np.zeros_like(corr)
        sub = corr[idx]
        sh = np.clip(alpha * (sub - sub.max()), -100, 0)
        e = np.exp(sh)
        w[idx] = e / e.sum()
    else:
        sh = np.clip(alpha * (corr - cmax), -100, 0)
        e = np.exp(sh)
        w = e / e.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - n + 1
        js, je = max(0, -lag), min(n-1, n-1-lag)
        if js <= je:
            grad[js:je+1] += wk * (1.0 - h[js+lag:je+1+lag])
        js2, je2 = max(0, lag), min(n-1, n-1+lag)
        if js2 <= je2:
            grad[js2:je2+1] -= wk * h[js2-lag:je2+1-lag]
    grad *= dx
    return grad, cmax

def _proj_subgrad(h0, alphas, iters, lr0, deadline, topk=None):
    n = len(h0)
    h = h0.copy()
    best, best_c5 = h.copy(), _eval_c5(h)
    for alpha in alphas:
        lr = lr0 / (1.0 + 0.02 * alpha)
        for it in range(iters):
            if time.time() > deadline:
                return best, best_c5
            grad, _ = _softmax_grad(h, alpha, topk)
            h = _project(h - lr * grad, n)
            c5 = _eval_c5(h)
            if c5 < best_c5:
                best_c5, best = c5, h.copy()
            lr *= 0.998
        lr = lr0 / (1.0 + 0.02 * alphas[-1])
    return best, best_c5

def _mm_obj(x):
    return x[-1]

def _mm_ineq(x):
    h, t = x[:-1], x[-1]
    return t - _corr_vals(h)

def _mm_eq(x):
    return x[:-1].sum() - len(x) / 2.0 + 1.0

def _slsqp_mm(h0, maxiter=500):
    n = len(h0)
    x0 = np.concatenate([h0, [_eval_c5(h0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    try:
        res = minimize(_mm_obj, x0, method='SLSQP',
                       constraints=[{'type': 'ineq', 'fun': _mm_ineq},
                                    {'type': 'eq', 'fun': _mm_eq}],
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_res = _project(res.x[:-1], n)
        if _eval_c5(h_res) < _eval_c5(h0) + 1e-12:
            return h_res
    except:
        pass
    return h0.copy()

def _sm_obj(h, alpha):
    corr = _corr_vals(h)
    cm = corr.max()
    sh = np.clip(alpha * (corr - cm), -100, 0)
    return cm + np.log(np.exp(sh).sum()) / alpha

def _slsqp_sm(h0, alpha, maxiter=300):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    def f(h): return _sm_obj(h, alpha)
    def ceq(h): return h.sum() - n / 2.0
    try:
        res = minimize(f, h0, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': ceq}],
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-12})
        h_res = _project(res.x, n)
        if _eval_c5(h_res) < _eval_c5(h0) + 1e-12:
            return h_res
    except:
        pass
    return h0.copy()
