d c[k] / d h[i]:
- When i is in [0, n-1-L]: the term h[i]*(1-h[i+L]) contributes (1-h[i+L])
- When i-L is in [0, n-1] (i.e., i in [L, n-1]) and i ≤ n-1-L... wait, I need to be more careful.

Actually, the term h[t]*(1-h[t+L]) involves h at positions t and t+L, where t ranges from 0 to n-1-L.
- d/dh[i] of h[t]*(1-h[t+L]) is non-zero when:
  - t = i (i ranges 0 to n-1-L): derivative is (1-h[i+L])
  - t+L = i, i.e., t = i-L (i ranges L to n-1, and i-L ranges 0 to n-1-L): derivative is -h[i-L]

So: d c[k]/d h[i] = (1-h[i+L]) * [i ≤ n-1-L] - h[i-L] * [i ≥ L]

For L ≥ 0:

- From h[j+L2] when j+L2 = i, i.e., j = i-L2: valid when 0 ≤ i-L2 ≤ n-1-L2, i.e., L2 ≤ i ≤ n-1. Contribution: (1-h[i-L2])
- From (1-h[j]) when j = i: valid when 0 ≤ i ≤ n-1-L2. Contribution: -h[i+L2]

So: d c[k]/d h[i] = (1-h[i-L2]) * [L2 ≤ i ≤ n-1] - h[i+L2] * [0 ≤ i ≤ n-1-L2]

For L > 0 (lag is positive):

    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def to_binary(h_frac, n):
    h = np.zeros(n)
    order = np.argsort(-h_frac)
    h[order[:n // 2]] = 1.0
    return h

def exhaustive_1swap(h, time_limit):
    n = len(h)
    h = h.copy()
    best_max = float(np.correlate(h, 1.0 - h, mode='full').max())
    best_h = h.copy()
    t0 = time.time()
    while time.time() - t0 < time_limit:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                cm = float(np.correlate(h, 1.0 - h, mode='full').max())
                h[i], h[j] = 1.0, 0.0
                if cm < best_max - 1e-12:
                    h[i], h[j] = 0.0, 1.0
                    best_max = cm
                    best_h = h.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, float(best_max * 2.0 / n)

def simulate_anneal_binary(h_init, time_limit, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    curr_max = float(np.correlate(h, 1.0 - h, mode='full').max())
    best_h = h.copy()
    best_max = curr_max
    t0 = time.time()
    T = 0.08
    while time.time() - t0 < time_limit:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_max = float(np.correlate(h, 1.0 - h, mode='full').max())
        delta = new_max - curr_max
        if delta < 0 or rng.rand() < math.exp(-delta / max(T, 1e-8)):
            curr_max = new_max
            if curr_max < best_max:
                best_max = curr_max
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T *= 0.9997
    return best_h, float(best_max * 2.0 / n)

def compute_subgrad(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k]
        if abs(w) < 1e-12:
            continue
        L = k - (n - 1)
        if L >= 0:
            m = n - L
            if m > 0:
                grad[:m] += w * (1.0 - h[L:])
                grad[L:] -= w * h[:m]
        else:
            L2 = -L
            m = n - L2
            if m > 0:
                grad[L2:] += w * (1.0 - h[:m])
                grad[:m] -= w * h[L2:]
    return grad

def fractional_polish(h_init, time_limit, seed):
    n = len(h_init)
    target_sum = n / 2.0
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    lr = 0.015
    beta = 0.9
    v = np.zeros(n)
    tau = 40.0
    no_improve = 0
    while time.time() - t0 < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = corr.max()
        shifted = np.clip((corr - mc) * tau, -500, 500)
        exps = np.exp(shifted)
        weights = exps / exps.sum()
        grad = compute_subgrad(h, weights)
        v = beta * v - lr * grad
        h_new = project_feasible(h + v, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        if no_improve > 200:
            lr = max(lr * 0.8, 0.002)
            tau *= 1.01
        if no_improve > 600:
            h = project_feasible(best_h + np.random.RandomState(seed + no_improve).randn(n) * 0.03, target_sum)
            v[:] = 0.0
            lr = 0.01
            tau = 40.0
            no_improve = 0
    return best_h, best_c5

def make_structured_init(n, seed):
    rng = np.random.RandomState(seed)
    patterns = []
    p1 = np.zeros(n)
    for i in range(n):
        p1[i] = 1.0 if (i * 2) % n < 1 else 0.0
    patterns.append(p1)
    p2 = np.zeros(n)
    period = max(3, n // 14)
    for i in range(n):
        p2[i] = 1.0 if (i % period) < period * 2 // 3 else 0.0
    patterns.append(p2)
    p3 = np.zeros(n)
    perm = rng.permutation(n)
    p3[perm[:n // 2]] = 1.0
    patterns.append(p3)
    # Block pattern
    p4 = np.zeros(n)
    block = n // 7
    for b in range(0, n, block * 2):
        p4[b:b + block] = 1.0
    patterns.append(p4)
    return patterns

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.asarray(initial_h_values, dtype=float)
    else:
        parent = np.full(70, 0.5)

    best_h = parent.copy()
    best_c5 = compute_c5(parent)
    best_n = len(parent)

    n_list = sorted(set([best_n, 64, 66, 68, 70, 72, 74, 76, 80]))
    n_list = [n for n in n_list if n >= 40 and n <= 100 and n % 2 == 0]

    for n in n_list:
        if time.time() > deadline - 25:
            break
        t_rem = deadline - time.time() - 15
        if t_rem < 15:
            break

        # Build initializations
        inits = []
        if n == len(parent):
            inits.append(parent.copy())
        else:
            x_old = np.linspace(0, 2, len(parent))
            x_new = np.linspace(0, 2, n)
            inits.append(np.interp(x_new, x_old, parent))
        inits.extend(make_structured_init(n, seed + n))

        best_n_h = None
        best_n_c5 = 1e9

        for ii, hfrag in enumerate(inits):
            if time.time() > deadline - 15:
                break
            hfrag = project_feasible(hfrag, n / 2.0)
            h_bin = to_binary(hfrag, n)

            # Binary exhaustive search
            t_alloc = min((deadline - time.time() - 12) * 0.25, 45)
            if t_alloc < 3:
                continue
            hb, c5b = exhaustive_1swap(h_bin, t_alloc)
            if c5b < best_n_c5:
                best_n_c5 = c5b
                best_n_h = hb.copy()

            # Simulated annealing from best binary
            if time.time() < deadline - 12 and best_n_h is not None:
                t_sa = min((deadline - time.time() - 10) * 0.2, 25)
                if t_sa > 3:
                    hs, c5s = simulate_anneal_binary(best_n_h, t_sa, seed + 50 + ii * 13)
                    if c5s < best_n_c5:
                        best_n_c5 = c5s
                        best_n_h = hs.copy()
                    # Then exhaustive again
                    t_ex2 = min((deadline - time.time() - 10) * 0.15, 20)
                    if t_ex2 > 2:
                        hb2, c5b2 = exhaustive_1swap(hs, t_ex2)
                        if c5b2 < best_n_c5:
                            best_n_c5 = c5b2
                            best_n_h = hb2.copy()

        # Fractional polishing from best binary for this n
        if time.time() < deadline - 10 and best_n_h is not None:
            t_fp = min((deadline - time.time() - 8) * 0.35, 35)
            if t_fp > 5:
                hf, c5f = fractional_polish(best_n_h.astype(float), t_fp, seed + 500 + n)
                if c5f < best_n_c5:
                    best_n_c5 = c5f
                    best_n_h = hf.copy()

        # Fractional from random init
        if time.time() < deadline - 10:
            t_fr = min((deadline - time.time() - 8) * 0.2, 25)
            if t_fr > 5:
                rng_r = np.random.RandomState(seed + 700 + n)
                h_rand = project_feasible(0.5 + rng_r.randn(n) * 0.25, n / 2.0)
                hr, c5r = fractional_polish(h_rand, t_fr, seed + 800 + n)
                if c5r < best_n_c5:
                    best_n_c5 = c5r
                    best_n_h = hr.copy()

        if best_n_h is not None and best_n_c5 < best_c5:
            best_h = best_n_h
            best_c5 = best_n_c5
            best_n = n

    # Final polish on global best
    t_rem = deadline - time.time() - 2
    if t_rem > 5:
        hf, c5f = fractional_polish(best_h.copy(), min(t_rem, 20), seed + 9999)
        if c5f < best_c5:
            best_h, best_c5 = hf, c5f

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
