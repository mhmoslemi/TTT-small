            g[j_lo2:j_hi2] -= w * h[j_lo2 - lag:j_hi2 - lag]
    return g, float(ov_max)

def optimize_phase(h0, n, deadline, rng, alpha_init=30.0, alpha_max=400.0):
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = get_c5(h0, n)
    v = np.zeros(n)
    step_size = 0.12
    beta = 0.85
    stalled = 0
    alpha = alpha_init
    iters = 0
    
    while time.time() < deadline:
        g, _ = compute_grad(h, n, alpha)
        g_norm = np.max(np.abs(g))
        if g_norm > 1e-15:
            g = g / g_norm
        v = beta * v + (1.0 - beta) * g
        h_new = project_feasible(h - step_size * v, n)
        c_new = get_c5(h_new, n)
        
        if c_new < best_c5 - 1e-13:
            best_c5 = c_new
            best_h = h_new.copy()
            h = h_new
            stalled = 0
        else:
            h = h_new
            stalled += 1
            if stalled > 40:
                step_size *= 0.65
                v *= 0.4
                stalled = 0
            if stalled > 120:
                noise = rng.randn(n) * 0.15
                h = project_feasible(best_h + noise, n)
                v = np.zeros(n)
                step_size = min(step_size * 2.0, 0.15)
                stalled = 0
                alpha = min(alpha * 1.5, alpha_max)
        
        iters += 1
        if iters % 400 == 0:
            alpha = min(alpha * 1.2, alpha_max)
    
    return best_h, best_c5

def binary_refine(h0, n, deadline, rng):
    h_bin = (h0 > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        extra = n_ones - target
        pick = rng.choice(idx, size=min(extra, len(idx)), replace=False)
        h_bin[pick] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        need = target - n_ones
        pick = rng.choice(idx, size=min(need, len(idx)), replace=False)
        h_bin[pick] = 1
    best_c5 = get_c5(h_bin, n)
    h_best = h_bin.copy()
    T = 0.03
    no_imp = 0
    while time.time() < deadline:
        ones_idx = np.where(h_best == 1)[0]
        zeros_idx = np.where(h_best == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = int(rng.choice(ones_idx))
        j = int(rng.choice(zeros_idx))
        h_new = h_best.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        c_new = get_c5(h_new, n)
        delta = c_new - best_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-6)):
            h_best = h_new
            if c_new < best_c5:
                best_c5 = c_new
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
        if no_imp > 300:
            T *= 0.75
            no_imp = 0
        if T < 0.002:
            break
    return h_best.astype(float), best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    best_h = None
    best_c5 = 1.0
    best_n = 0
    
    try:
        h0_arr = np.array(initial_h_values, dtype=float)
    except Exception:
        h0_arr = None
    
    if h0_arr is not None:
        n0 = len(h0_arr)
        h0_arr = project_feasible(h0_arr, n0)
        
        # Phase 1a: smooth continuous optimization
        dl1a = t0 + budget_s * 0.28
        bh, bc5 = optimize_phase(h0_arr, n0, dl1a, rng, alpha_init=25.0, alpha_max=200.0)
        best_h, best_c5, best_n = bh, bc5, n0
        
        # Phase 1b: binary refinement
        dl1b = t0 + budget_s * 0.38
        bh_bin, bc5_bin = binary_refine(best_h, best_n, dl1b, rng)
        if bc5_bin < best_c5:
            best_h, best_c5 = bh_bin, bc5_bin
        
        # Phase 1c: continuous refinement from best
        dl1c = t0 + budget_s * 0.55
        h_start = best_h.copy()
        if np.max(np.abs(h_start - np.round(h_start))) < 0.01:
            h_start = h_start + rng.randn(best_n) * 0.08
            h_start = project_feasible(h_start, best_n)
        bh, bc5 = optimize_phase(h_start, best_n, dl1c, rng, alpha_init=50.0, alpha_max=500.0)
        if bc5 < best_c5:
            best_h, best_c5 = bh, bc5
        
        # Phase 1d: another round of binary + continuous
        dl1d = t0 + budget_s * 0.62
        bh_bin2, bc5_bin2 = binary_refine(best_h, best_n, dl1d, rng)
        if bc5_bin2 < best_c5:
            best_h, best_c5 = bh_bin2, bc5_bin2
    else:
        best_n = 46
        h0_arr = np.zeros(best_n)
        h0_arr[:best_n // 2] = 1.0
        h0_arr = project_feasible(h0_arr, best_n)
        bh, bc5 = optimize_phase(h0_arr, best_n, t0 + budget_s * 0.5, rng)
        best_h, best_c5 = bh, bc5
    
    # Phase 2: try nearby n values
    if h0_arr is not None:
        nb = len(h0_arr)
        n_candidates = [x for x in range(max(34, nb - 4), min(64, nb + 5)) if x != nb and x % 2 == 0]
    else:
        n_candidates = [44, 48, 50, 52]
    
    phase2_start = max(time.time(), t0 + budget_s * 0.62)
    phase2_end = t0 + budget_s * 0.82
    if n_candidates and phase2_start < phase2_end:
        per_try = (phase2_end - phase2_start) / float(len(n_candidates))
        for n_try in n_candidates:
            if time.time() > phase2_end:
                break
            h_struct = np.zeros(n_try)
            h_struct[:n_try // 2] = 1.0
            h_struct = project_feasible(h_struct, n_try)
            dl_try = time.time() + per_try * 0.85
            bh, bc5 = optimize_phase(h_struct, n_try, dl_try, rng, alpha_init=40.0)
            if bc5 < best_c5:
                best_h, best_c5, best_n = bh, bc5, n_try
    
    # Phase 3: final sharpening
    if best_h is not None:
        dl_final = t0 + budget_s - 3.0
        if time.time() < dl_final:
            bh, bc5 = optimize_phase(best_h, best_n, dl_final, rng, alpha_init=150.0, alpha_max=600.0)
            if bc5 < best_c5:
                best_h, best_c5 = bh, bc5
    
    # Final validation
    h_final = project_feasible(best_h, best_n)
    c5_final = get_c5(h_final, best_n)
    dx_f = 2.0 / best_n
    if abs(np.sum(h_final) * dx_f - 1.0) > 1e-8:
        h_final = project_feasible(h_final, best_n)
        c5_final = get_c5(h_final, best_n)
    
    return h_final, float(c5_final), best_n
