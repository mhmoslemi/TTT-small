def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

    for _ in range(300):

        d = target - s

        mask = (h > 1e-10) & (h < 1.0 - 1e-10)

        if cnt > 0:
            h[mask] += d / cnt
        else:
            h += d / n
        np.clip(h, 0.0, 1.0, out=h)

        if np.sum(mask) > 0:

            np.clip(h, 0.0, 1.0, out=h)

def softmax_obj_grad(h, alpha, mu):

    # Phase 1: L-BFGS-B with alpha annealing
    for alpha in [40.0, 80.0, 150.0, 300.0, 600.0]:
        if time.time() - t0 > budget_s * 0.20:
            break
        for mu in [300.0, 2000.0]:
            if time.time() - t0 > budget_s * 0.22:
                break
            res = minimize(
                obj_grad, h_best.copy(), args=(alpha, mu),
                method='L-BFGS-B', bounds=bounds, jac=True,
                options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-12}
            )
            hc = project_feasible(res.x, n)
            c5c = compute_c5(hc)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc.copy()

    # Phase 2: Higher mu for tighter constraint
    for mu in [5000.0, 20000.0, 80000.0]:
        if time.time() - t0 > budget_s * 0.40:
            break
        res = minimize(
            obj_grad, h_best.copy(), args=(300.0, mu),
            method='L-BFGS-B', bounds=bounds, jac=True,
            options={'maxiter': 400, 'ftol': 1e-16, 'gtol': 1e-12}
        )
        hc = project_feasible(res.x, n)
        c5c = compute_c5(hc)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()

    # Phase 3: Multi-restart from diverse structures
    starts = []
    for k in range(4):
        starts.append(make_struct(n, k))
    for _ in range(4):
        h_r = rng.random(n)
        h_r = (h_r > rng.uniform(0.3, 0.7)).astype(float)
        starts.append(project_feasible(h_r, n))
    # Perturbed best
    for _ in range(3):
        hp = h_best + 0.03 * rng.standard_normal(n)
        starts.append(project_feasible(hp, n))

    for sp in starts:
        if time.time() - t0 > budget_s * 0.68:
            break
        for alpha in [100.0, 300.0]:
            if time.time() - t0 > budget_s * 0.70:
                break
            res = minimize(
                obj_grad, sp.copy(), args=(alpha, 5000.0),
                method='L-BFGS-B', bounds=bounds, jac=True,
                options={'maxiter': 400, 'ftol': 1e-15, 'gtol': 1e-11}
            )
            hc = project_feasible(res.x, n)
            c5c = compute_c5(hc)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc.copy()

    # Phase 4: Perturbation restarts (main search)
    while time.time() - t0 < budget_s - 80:
        pert = 0.005 + 0.03 * rng.random()
        h_pert = h_best + pert * rng.standard_normal(n)
        h_pert = project_feasible(h_pert, n)
        alpha = 200.0 + 400.0 * rng.random()
        res = minimize(
            obj_grad, h_pert, args=(alpha, 10000.0),
            method='L-BFGS-B', bounds=bounds, jac=True,
            options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-11}
        )
        hc = project_feasible(res.x, n)
        c5c = compute_c5(hc)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()

    # Phase 5: Final fine-tuning with very high alpha
    if time.time() - t0 < budget_s - 40:
        for alpha in [500.0, 1000.0, 2000.0]:
            if time.time() - t0 > budget_s - 40:
                break
            res = minimize(
                obj_grad, h_best.copy(), args=(alpha, 50000.0),
                method='L-BFGS-B', bounds=bounds, jac=True,
                options={'maxiter': 600, 'ftol': 1e-16, 'gtol': 1e-13}
            )
            hc = project_feasible(res.x, n)
            c5c = compute_c5(hc)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc.copy()

    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)
    return h_best, c5_best, n
