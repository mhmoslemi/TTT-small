def binary_local_search(h_bin, n, iterations=10000, deadline=None):
    """Local search: try 2-exchanges to minimize max overlap."""
    best_h = h_bin.copy()
    best_c5 = get_c5(h_bin, n)
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]

    for it in range(iterations):
        if deadline and time.time() > deadline:
            break
        a = ones[np.random.randint(len(ones))]
        b = zeros[np.random.randint(len(zeros))]

        h_new[a], h_new[b] = h_new[b], h_new[a]

        if c_new < best_c5:
            best_h = h_new
            best_c5 = c_new

def project_feasible(h, n):

        cnt = float(np.sum(mask))
        if cnt > 0:
            hp[mask] = np.clip(hp[mask] + diff / cnt, 0.0, 1.0)
    return hp

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def get_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_grad_top_lags(h, n, top_k=5, beta=10.0):
    """Compute softmax-weighted subgradient over top-k lags."""
    ov = get_overlaps(h, n)
    n_ov = len(ov)
    # Softmax weights
    ov_max = ov.max()
    w = np.exp(beta * (ov - ov_max))
    s = w.sum()
    if s < 1e-15:
        w = np.zeros(n_ov)
        w[np.argmax(ov)] = 1.0
    else:
        w /= s
    # Only use top-k lags
    idx = np.argsort(w)[-top_k:]
    w = w[idx]
    w /= w.sum()
    
    g = np.zeros(n)
    for i, kk in enumerate(idx):
        lag = kk - (n - 1)
        if lag == 0:
            g[:] += w[i] * (1.0 - 2.0 * h)  # d/dh of h*(1-h) = 1-2h
        else:
            l = abs(lag)
            if lag > 0:
                # A part: j in [0, n-l), add (1-h[j+lag])
                g[:n-l] += w[i] * (1.0 - h[lag:])
                # B part: j in [lag, n), subtract h[j-lag]
                g[lag:] -= w[i] * h[:n-lag]
            else:
                # lag < 0, l = -lag
                # A part: j in [l, n), add (1-h[j-l])
                g[l:] += w[i] * (1.0 - h[:n-l])
                # B part: j in [0, n-l), subtract h[j+l]
                g[:n-l] -= w[i] * h[l:]
    return g, ov

    dx = 2.0 / n

    best_c5 = get_c5(h, n)
    lr = 0.1  # initial step size (scaled by dx below)
    stalled = 0
    
    while time.time() < deadline:
        g, ov = compute_grad_top_lags(h, n, top_k, beta)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-10:
            break
        
        # Line search
        best_lr = lr
        best_c = best_c5
        for trial in range(8):

            c_trial = get_c5(h_trial, n)
            if c_trial < best_c - 1e-10:
                best_lr = lr
                best_c = c_trial
                best_h = h_trial
                break
            lr *= 0.7
        else:
            lr *= 0.5
            if lr < 0.001:
                # Restart with perturbation
                h = best_h + rng.randn(n) * 0.1
                h = project_feasible(h, n)
                lr = 0.1

        h = best_h
        if best_c < best_c5 - 1e-12:

import numpy as np

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-14:
        mask = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        cnt = float(np.sum(mask))
        if cnt > 0:
            hp[mask] = np.clip(hp[mask] + diff / cnt, 0.0, 1.0)
    return hp

        if time.time() > deadline3 - 30:
            break
        # Quick local search with limited time
        per_time = min(30.0, (deadline3 - time.time()) / max(1, len(n_tries) - n_tries.index(n_try)))
        dl = time.time() + per_time
        
        # Multiple starting patterns
        for init_type in ['block', 'periodic']:
            if time.time() > dl:
                break
            if init_type == 'block':
                h_try = np.zeros(n_try)
                h_try[:n_try//2] = 1.0
            else:
                # Periodic: 11001100... or 10011001...
                period = 4
                h_try = np.zeros(n_try)
                for i in range(n_try):
                    if i % period < 2:
                        h_try[i] = 1.0
                h_try = project_feasible(make_binary_from_continuous(h_try, n_try), n_try)
            
            bh3, bc5_3 = binary_hill_climb(h_try, n_try, dl, rng)
            if bc5_3 < best_c5:
                best_h, best_c5, best_n = bh3, bc5_3, n_try
    
    # Final validation
    h_final = project_feasible(best_h, best_n)
    c5_final = get_c5(h_final, best_n)
    if c5_final > best_c5 + 1e-10:
        h_final = best_h
        c5_final = best_c5
    
    return h_final, float(c5_final), best_n
