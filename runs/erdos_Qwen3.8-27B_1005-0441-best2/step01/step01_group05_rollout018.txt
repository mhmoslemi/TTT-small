def _subgrad_step(h, lr):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_max = int(np.argmax(corr))
    shift = k_max - (n - 1)
    n_abs = abs(shift)
    
    sg = np.zeros(n)
    if shift >= 0:
        if n_abs < n:
            sg[:n-n_abs] += (1.0 - h[n_abs:])
            sg[n_abs:] -= h[:n-n_abs]
    else:

            sg[n_abs:] += (1.0 - h[:n-n_abs])

    h_new = np.clip(h_new, 0.0, 1.0)

                h_new = np.clip(h_new, 0.0, 1.0)

def _solveslsqp(h, n, temp, max_iter, ftol):

        m = np.max(corr)

    def cons_eq(hv):
        return np.sum(hv) - n / 2.0
    bounds = [(0.0, 1.0)] * n

        if s > 1e-12:
            h_new = h_new * (n / 2.0) / s
            h_new = np.clip(h_new, 0.0, 1.0)
        return h_new

        return h

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    excess = np.maximum(0, corr - T)
    return penalty * np.sum(excess**2) + eps_var * np.var(corr)

def _upsample(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old, dtype=float) * 2.0 / n_old
    x_new = np.arange(n_new, dtype=float) * 2.0 / n_new

    if s > 1e-12:

    return np.clip(h_new, 0.0, 1.0)

def _subgrad_loop(h, n_steps, lr_init, lr_end, t0, deadline):

        h = _subgrad_step(h, lr)
        v = compute_c5(h)
        if v < best_val:
            best = h.copy()
            best_val = v
    return best, best_val

import time

_g_n = 160
_g_temp = 0.01
_g_T = 0.39
_g_pen = 100.0
_g_vareps = 1e-3

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _slsqp_obj(hv):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(hv, 1.0 - hv, mode="full")
    m = np.max(corr)
    t = max(_g_temp, 1e-10)
    return (m + t * np.log(np.sum(np.exp((corr - m) / t)))) * dx

def _cons_eq(hv):
    return np.sum(hv) - _g_n / 2.0

def _cons_jac(hv):
    return np.ones(_g_n)

def _thresh_obj(hv):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(hv, 1.0 - hv, mode="full") * dx
    excess = np.maximum(0.0, corr - _g_T)
    return _g_pen * np.sum(excess * excess) + _g_vareps * np.var(corr)

def _proj_sum(h, n):
    target = float(n) / 2.0
    s = np.sum(h)
    if s < 1e-12:
        return h
    h = h * target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        mask = (h > 1e-7) & (h < 1.0 - 1e-7)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def _slsqp_solve(h, temp, max_iter, ftol):
    global _g_n, _g_temp
    n = len(h)
    _g_n = n
    _g_temp = temp
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac},)
    try:
        res = minimize(_slsqp_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
        h_new = np.clip(res.x, 0.0, 1.0)
        return _proj_sum(h_new, n)
    except Exception:
        return h

def _subgrad_step(h, lr, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mv = np.max(corr)
    active = np.where(corr >= mv - eps)[0]
    sg = np.zeros(n)
    cnt = 0
    for ki in active:
        shift = int(ki) - (n - 1)
        ka = abs(shift)
        if ka >= n:
            continue
        if shift >= 0:
            sg[:n - ka] += (1.0 - h[ka:])
            sg[ka:] -= h[:n - ka]
        else:
            sg[ka:] += (1.0 - h[:n - ka])
            sg[:n - ka] -= h[ka:]
        cnt += 1
    if cnt > 0:
        sg /= cnt
    h_new = h - lr * dx * sg
    h_new = np.clip(h_new, 0.0, 1.0)
    return _proj_sum(h_new, n)

def _subgrad_loop(h, n_steps, lr_init, lr_end, deadline):
    best = h.copy()
    best_val = compute_c5(best)
    for i in range(n_steps):
        if time.time() > deadline:
            break
        frac = i / max(1, n_steps - 1)
        lr = lr_init * (lr_end / lr_init) ** frac
        h = _subgrad_step(h, lr, 0.002)
        v = compute_c5(h)
        if v < best_val:
            best = h.copy()
            best_val = v
    return best, best_val

def _perturb(h, scale):
    n = len(h)
    h_p = h + scale * np.random.randn(n)
    h_p = np.clip(h_p, 0.0, 1.0)

    h_new = np.interp(x_new, x_old, h)
    return _proj_sum(h_new, n_new)
