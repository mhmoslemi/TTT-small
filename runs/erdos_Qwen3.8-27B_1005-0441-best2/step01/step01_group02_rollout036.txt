Actually, let me precompute the correlation structure. For a given h, the correlation at shift d is:
c[d] = sum(h[i] * (1-h[i+d])) for valid i

If I change h[j] by delta, which entries change?
- For the term h[j]*(1-h[j+d']): c[d'] changes by delta*(1-h[j+d'])
- For the term h[j-d']*(1-h[j]): c[d'] changes by h[j-d']*(-delta)

So changing h[j] by delta affects ALL lags, but only by a small amount. For coordinate descent, it's simpler to just recompute.

For n=70, np.correlate with mode="full" on two length-70 arrays takes about 10-20 microseconds. So 10000 evaluations take about 100-200ms. Very fast.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = free.sum()
        if nf == 0:
            h = np.clip(h + diff / n, 0.0, 1.0)
        else:
            h[free] += diff / nf
            h = np.clip(h, 0.0, 1.0)
    return h

def _sobj(x, beta):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = corr.max()
    return float((mc + np.log(np.sum(np.exp(beta * (corr - mc))))) / beta * (2.0 / len(x)))

def _sobjj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = corr.max()
    w = np.exp(beta * (corr - mc))
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-14:
            continue
        d = i - (n - 1)
        lo1, hi1 = max(0, -d), min(n, n - d)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - x[lo1 + d:hi1 + d])
        lo2, hi2 = max(0, d), min(n, d + n)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * x[lo2 - d:hi2 - d]
    return grad * dx

def _con_fun(x):
    return x.sum() - len(x) / 2.0

def _con_jac(x):
    return np.ones(len(x))

def _slsqp(h0, beta, maxiter=200):
    n = len(h0)
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sobj, h0, jac=_sobjj, method='SLSQP',
                       args=(beta,), bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return _project(res.x)
    except Exception:
        return _project(h0.copy())

def _cd_single(h, deltas, t_end):
    n = len(h)
    best_c5 = compute_c5(h)
    for delta in deltas:
        for rnd in range(80):
            if time.time() > t_end - 2:
                break
            imp = False
            for j in range(n):
                for sgn in [1, -1]:
                    d = sgn * delta
                    h[j] += d
                    if h[j] < 0 or h[j] > 1:
                        h[j] -= d
                        continue
                    c = compute_c5(h)
                    if c < best_c5 - 1e-16:
                        best_c5 = c
                        imp = True
                    else:
                        h[j] -= d
                if imp:
                    break
            h = _project(h)
            bc = compute_c5(h)
            if bc < best_c5:
                best_c5 = bc
            if not imp:
                break
    return h, best_c5

def _cd_pair(h, deltas, t_end):
    n = len(h)
    best_c5 = compute_c5(h)
    for delta in deltas:
        for rnd in range(40):
            if time.time() > t_end - 2:
                break
            imp = False
            for j in range(n):
                for ko in [1, 2, 3, n // 2]:
                    k = (j + ko) % n
                    if k == j:
                        continue
                    for sgn in [1, -1]:
                        d = sgn * delta
                        h[j] += d
                        h[k] -= d
                        if h[j] < 0 or h[j] > 1 or h[k] < 0 or h[k] > 1:
                            h[j] -= d
                            h[k] += d
                            continue
                        c = compute_c5(h)
                        if c < best_c5 - 1e-16:
                            best_c5 = c
                            imp = True
                        else:
                            h[j] -= d
                            h[k] += d
                if imp:
                    break
            if not imp:
                break
    return h, best_c5

def _nm_polish(h, t_end, maxiter=400):
    n = len(h)
    def nm_obj(x):
        return compute_c5(_project(x))
    remaining = max(50, int(t_end - time.time() - 3))
    mi = min(maxiter, remaining)
    try:
        res = minimize(nm_obj, h.copy(), method='Nelder-Mead',
                       options={'maxiter': mi, 'xatol': 1e-10, 'fatol': 1e-13})
        h2 = _project(res.x)
        c2 = compute_c5(h2)
        return h2, c2
    except Exception:
        return h.copy(), compute_c5(h)

def _optimize_start(h0, t_budget):
    t_end = time.time() + t_budget
    h = _project(h0.astype(float).copy())
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for beta in [10.0, 30.0, 80.0, 200.0]:
        if time.time() > t_end - 15:
            break
        h = _slsqp(h, beta, maxiter=180)
        c = compute_c5(h)
        if c < best_c5:
            best_c5, best_h = c, h.copy()
    h = best_h.copy()
    h, c = _cd_single(h, [0.05, 0.02, 0.01, 0.005], t_end)
    if c < best_c5:
        best_c5, best_h = c, h.copy()
    h = best_h.copy()
    h, c = _cd_pair(h, [0.02, 0.01, 0.005], t_end)
    if c < best_c5:
        best_c5, best_h = c, h.copy()
    h = best_h.copy()
    h, c = _cd_single(h, [0.002, 0.001, 0.0005], t_end)
    if c < best_c5:
        best_c5, best_h = c, h.copy()
    if time.time() < t_end - 10:
        h2, c2 = _nm_polish(best_h, t_end)
        if c2 < best_c5:
            best_c5, best_h = c2, h2
    best_h, bc = _cd_single(best_h, [0.0003, 0.0001, 0.00005], t_end)
    if bc < best_c5:
        best_c5 = bc
        best_h = best_h
    return _project(best_h), best_c5

    rng = np.random.RandomState(seed)
