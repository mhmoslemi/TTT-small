Let me think about the 2-swap coordinate descent more carefully:
- We have h with sum = n/2
- Pick two indices i, j (i != j)
- Try: h[i] += delta, h[j] -= delta
- Feasibility: 0 <= h[i]+delta <= 1, 0 <= h[j]-delta <= 1
- The sum is automatically preserved
- We want to minimize C5

For the delta, we can try a few values: the maximum feasible delta, half of that, etc.

This is promising because it maintains the constraint exactly without projection.

For the subgradient method:
- Compute corr, find argmax k*
- Compute subgradient g (gradient of corr[k*] with respect to h)
- Step: h_new = h - alpha * g
- Project: h = project_feasible(h_new)
- Use Polyak step: alpha = (f(h) - f*) / ||g||^2, but we don't know f*
- Alternative: decreasing step size alpha_t = c / sqrt(t) or c / t

Let me also add a "binary round" step: after optimization, round h to nearest 0/1 (maintaining exactly n/2 ones), evaluate, and if better, keep it. Then un-round slightly.

```python
import numpy as np
from scipy.optimize import minimize
import time

_TARG = 35.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(x, target):
    x = np.clip(x, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _con_fun(x):
    return x.sum() - _TARG

def _con_jac(x):
    return np.ones(len(x))

def _sobj(h, beta):
    n = len(h)
    corr = compute_corr(h)
    mc = np.max(corr)
    shifted = beta * (corr - mc)
    mx = np.max(shifted)
    return (mc + mx + np.log(np.sum(np.exp(shifted - mx))) / beta) * (2.0 / n)

def _sjac(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    mc = np.max(corr)
    shifted = beta * (corr - mc)
    mx = np.max(shifted)
    w = np.exp(shifted - mx)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = (n - 1) - k
        if d >= 0:
            lo1, hi1 = 0, n - d
            grad[lo1:hi1] += wk * (1.0 - h[lo1 + d:hi1 + d])
            lo2, hi2 = d, n
            grad[lo2:hi2] -= wk * h[lo2 - d:hi2 - d]
        else:
            d2 = -d
            lo1, hi1 = d2, n
            grad[lo1:hi1] += wk * (1.0 - h[lo1 - d2:hi1 - d2])
            lo2, hi2 = 0, n - d2
            grad[lo2:hi2] -= wk * h[lo2 + d2:hi2 + d2]
    return grad * dx

def _slsqp(h0, beta, maxiter):
    n = len(h0)
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sobj, h0, jac=_sjac, args=(beta,),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, _TARG)
    except Exception:
        return project_feasible(h0.copy(), _TARG)

def _coord_2swap(h, target, n_trials, rng):
    n = len(h)
    h = h.copy()
    best = compute_c5(h)
    deltas = [0.01, 0.02, 0.05, 0.1, 0.3, 0.5, -0.01, -0.02, -0.05, -0.1, -0.3, -0.5]
    nd = len(deltas)
    for _ in range(n_trials):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = deltas[rng.randint(nd)]
        hi = h[i] + delta
        hj = h[j] - delta
        if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
            continue
        h[i] = hi
        h[j] = hj
        c = compute_c5(h)
        if c < best - 1e-15:
            best = c
        else:
            h[i] -= delta
            h[j] += delta
    return h, best

def _subgrad(h, steps, alpha0, target):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best = compute_c5(h)
    for t in range(steps):
        corr = compute_corr(h)
        k_star = int(np.argmax(corr))
        d = (n - 1) - k_star
        g = np.zeros(n)
        if d >= 0:
            lo1, hi1 = 0, n - d
            g[lo1:hi1] += (1.0 - h[lo1 + d:hi1 + d])
            lo2, hi2 = d, n
            g[lo2:hi2] -= h[lo2 - d:hi2 - d]
        else:
            d2 = -d
            lo1, hi1 = d2, n
            g[lo1:hi1] += (1.0 - h[lo1 - d2:hi1 - d2])
            lo2, hi2 = 0, n - d2
            g[lo2:hi2] -= h[lo2 + d2:hi2 + d2]
        alpha = alpha0 / (1.0 + 0.1 * t)
        h = project_feasible(h - alpha * g * dx, target)
        c = compute_c5(h)
        if c < best:
            best = c
    return h, best

def _nm_polish(h0, target, maxiter):
    n = len(h0)
    def nm_obj(x):
        return compute_c5(project_feasible(x, target))
    try:
        res = minimize(nm_obj, h0.copy(), method='Nelder-Mead',
                      options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-13})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h0.copy(), target)
```
