    """Compute subgradient of f_{p_lag} w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    # Term 1: (1 - h[j-p]) for j-p in [0, n)

def _smoothed_subgrad(h, K=5, alpha=50):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Get top K
    top_idx = np.argsort(corr)[::-1][:K]
    w = np.exp(alpha * (corr[top_idx] - corr[top_idx[0]]))
    w = w / w.sum()
    g = np.zeros(n)
    for wi, ki in zip(w, top_idx):
        p = ki - (n - 1)

        g[valid1] += wi * (1.0 - h[ip[valid1]])
        jp = j_range + p
        valid2 = (jp >= 0) & (jp < n)
        g[valid2] -= wi * h[jp[valid2]]

For the log-sum-exp:
log(sum(exp(alpha * corr_shift))) where corr_shift = corr - max(corr)

The max of corr_shift is 0, so the exp values are in (exp(-alpha*range), 1]. For alpha=5000 and range ~0.1, the smallest values are exp(-500) ≈ 0, which is fine. The numerical issue is when alpha * (corr_i - max) is very negative, exp underflows to 0, which is actually fine.

The gradient of the softmax objective:
d/dh_j [max(corr) + (1/alpha) * log(sum_i exp(alpha * (corr_i - max)))]
= (1/alpha) * sum_i alpha * exp(alpha*(corr_i - max)) / sum_i exp(alpha*(corr_i - max)) * dcorr_i/dh_j
= sum_i w_i * dcorr_i/dh_j (weighted sum of gradients of individual correlations)

This is exactly the smoothed subgradient! So the SLSQP approach and the subgradient approach are related, but SLSQP uses second-order information (Hessian approximation) while plain subgradient doesn't.

import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _fix_sum(h, n):
    h = np.array(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    target = n / 2.0
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            nr = room.sum()
            if nr < 1e-15:
                break
            h += diff * room / nr
        else:
            na = h.sum()
            if na < 1e-15:
                break
            h += diff * h / na
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if abs(diff) < 1e-15:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                h[i] += a
                diff -= a
            else:
                s = min(-diff, h[i])
                h[i] -= s
                diff += s
    return h

def _sg_vec(h, p, dx):
    n = len(h)
    j = np.arange(n)
    g = np.zeros(n)
    ip = j - p
    v = (ip >= 0) & (ip < n)
    g[v] += 1.0 - h[ip[v]]
    jp = j + p
    v = (jp >= 0) & (jp < n)
    g[v] -= h[jp[v]]
    return g * dx

def _smooth_sg(h, K, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = corr.max()
    top = np.argsort(corr)[::-1][:K]
    sw = corr[top] - cmax
    w = np.exp(alpha * sw)
    w /= w.sum()
    g = np.zeros(n)
    for wi, ki in zip(w, top):
        g += wi * _sg_vec(h, int(ki - (n - 1)), dx)
    return g

def _sm_obj(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = corr.max()
    sh = corr - cmax
    sh = np.maximum(sh, -60.0 / alpha)
    return cmax + np.log(np.sum(np.exp(alpha * sh))) / alpha

def _sm_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = corr.max()
    sh = corr - cmax
    sh = np.maximum(sh, -60.0 / alpha)
    w = np.exp(alpha * sh)
    w /= w.sum()
    g = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for ki in nz:
        g += w[ki] * _sg_vec(x, int(ki - (n - 1)), dx)
    return g

class _ObjF:
    def __init__(self, alpha):
        self.a = alpha
    def __call__(self, x):
        return _sm_obj(x, self.a)

class _JacF:
    def __init__(self, alpha):
        self.a = alpha
    def __call__(self, x):
        return _sm_grad(x, self.a)

def _eq(h):
    return h.sum() - len(h) / 2.0

def _slsqp(h0, alpha, niters):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_ObjF(alpha), h0, jac=_JacF(alpha),
                       method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': _eq}],
                       bounds=bounds,
                       options={'maxiter': niters, 'ftol': 1e-16})
        return res.x
    except Exception:
        return h0.copy()
