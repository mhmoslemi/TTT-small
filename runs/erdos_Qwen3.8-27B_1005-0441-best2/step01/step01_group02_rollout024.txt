    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        # Find which elements can absorb the adjustment
        deficit = target - s
        if deficit > 0:
            # Need to increase: add to elements < 1
            mask = h < 1.0 - 1e-15
            if not mask.any():
                break
            h[mask] += deficit / mask.sum()
        else:
            # Need to decrease: subtract from elements > 0
            mask = h > 1e-15
            if not mask.any():
                break
            h[mask] += deficit / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_descent(h, n, t_budget, step_scale=0.1):
    """Subgradient descent on the max correlation."""
    t0 = time.time()
    h = project_feasible(h.copy(), n)
    best = compute_c5(h)
    hb = h.copy()
    alpha = step_scale
    for iteration in range(100000):
        if time.time() - t0 > t_budget:
            break
        corr_vals = compute_corr(h)
        mc = corr_vals.max()
        # Find active lags (within 5% of max)
        threshold = mc * 0.95
        active = np.where(corr_vals >= threshold)[0]

        n_active = len(active)
        dx = 2.0 / n
        subgrad = np.zeros(n)
        for k_idx in active:
            lag = k_idx - (n - 1)
            for j in range(n):
                s = 1.0
                # (1 - h[j-lag]) term
                jl = j - lag
                if 0 <= jl < n:
                    s -= h[jl]
                # -h[j+lag] term
                jp = j + lag
                if 0 <= jp < n:
                    s -= h[jp]
                subgrad[j] += s
        subgrad *= dx / n_active

        h_new = h - alpha * subgrad
        h_new = project_feasible(h_new, n)

        if c < best:

            hb = h_new.copy()
            h = h_new
            alpha = min(alpha * 1.1, 0.5)
        else:
            alpha *= 0.8
            h = h_new  # still take the step
    
    return hb, best

    if lag >= 0:
        grad[:lag] = 1.0
        grad[lag:] = 1.0 - h[:n - lag]
        grad[:n - lag] -= h[lag:]
    else:
        l = -lag
        grad[:n - l] = 1.0 - h[l:]
        grad[n - l:] = 1.0
        grad[l:] -= h[:n - l]
    return grad

def sg_step(h, n, step_size, top_frac=0.3):
    """One subgradient step focusing on top lags."""
    corr_vals = compute_corr_vals(h)
    mc = corr_vals.max()
    threshold = mc - 0.02  # lags within 0.02 of max
    active = np.where(corr_vals >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr_vals)])
    
    dx = 2.0 / n
    subgrad = np.zeros(n)
    for idx in active:
        lag = idx - (n - 1)
        subgrad += lag_gradient(h, lag)
    subgrad *= dx / len(active)
    
    h_new = h - step_size * subgrad
    return project_feasible(h_new, n)

def subgrad_optimize(h, n, t_budget, init_step=0.15):
    t0 = time.time()
    h = project_feasible(h.copy(), n)
    best = compute_c5(h)
    hb = h.copy()
    step = init_step
    no_improve = 0
    while time.time() - t0 < t_budget:
        # Batch: do several small steps
        batch = 5
        for _ in range(batch):
            h_new = sg_step(h, n, step)
            c = compute_c5(h_new)
            if c < best - 1e-12:
                best = c
                hb = h_new.copy()
                h = h_new
                no_improve = 0
            else:
                no_improve += 1
                step *= 0.9
        if no_improve > 50:
            step = max(step * 0.5, 0.001)
            no_improve = 0
    return hb, best

def slsqp_smooth(h, n, beta, maxiter=200):
    """SLSQP on smoothed objective."""
    target = n / 2.0
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full") * dx
        mc = corr.max()
        # Smooth max
        shifted = beta * (corr - mc)
        shifted = np.clip(shifted, -50, 50)
        return mc + np.log(np.sum(np.exp(shifted))) / beta
    
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full") * dx
        mc = corr.max()
        w = np.exp(np.clip(beta * (corr - mc), -50, 50))
        ws = w.sum()
        if ws < 1e-300:
            return np.zeros(n)
        w /= ws
        g = np.zeros(n)
        for idx in range(len(corr)):
            if w[idx] < 1e-12:
                continue
            lag = idx - (n - 1)
            g += w[idx] * lag_gradient(x, lag)
        return g
    
    def con_fun(x):
        return x.sum() - target
    def con_jac(x):
        return np.ones(n)
    
    try:
        res = minimize(obj, h.copy(), jac=jac, method='SLSQP',
                      bounds=[(0,1)]*n,
                      constraints=[{'type':'eq','fun':con_fun,'jac':con_jac}],
                      options={'maxiter':maxiter,'ftol':1e-10})
        return project_feasible(res.x, n)
    except:
        return project_feasible(h, n)

def nm_polish(h, n, maxiter=300):
    target = n / 2.0
    def obj(x):
        return compute_c5(project_feasible(x, n))
    try:
        res = minimize(obj, h.copy(), method='Nelder-Mead',
                      options={'maxiter':maxiter,'xatol':1e-9,'fatol':1e-11})
        hp = project_feasible(res.x, n)
        return hp
    except:
        return project_feasible(h, n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(8000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def lag_gradient(h, lag):
