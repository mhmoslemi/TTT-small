import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def power_sum_obj(h, p, lam, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Shift to non-negative for power
    c = np.clip(corr, 1e-15, None)
    ps = (np.sum(c**p))**(1.0/p)
    penalty = lam * (np.sum(h) - n/2.0)**2
    return ps * dx + penalty

def optimize_power(h_init, n, p, lam, max_time, seed):
    """Optimize using L-BFGS-B with power-sum objective."""
    t0 = time.time()
    best = h_init.copy()
    best_val = compute_c5(best)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return power_sum_obj(x, p, lam, n)
    
    # L-BFGS-B optimization
    res = minimize(obj, h_init, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-10,
                            'maxfun': 20000})
    h_new = np.clip(res.x, 0.0, 1.0)
    # Project onto sum constraint
    s = np.sum(h_new)
    target = n / 2.0
    if s > 1e-12:
        h_new = h_new * target / s
        h_new = np.clip(h_new, 0.0, 1.0)
        # Re-project (clipping may have broken the sum)
        s = np.sum(h_new)
        if abs(s - target) > 1e-10:
            diff = target - s
            # Distribute diff among non-boundary elements
            mask = (h_new > 1e-6) & (h_new < 1 - 1e-6)
            if np.sum(mask) > 0:
                h_new[mask] += diff / np.sum(mask)
                h_new = np.clip(h_new, 0.0, 1.0)
    
    v = compute_c5(h_new)
    if v < best_val:
        best = h_new
        best_val = v
    return best, best_val

Or simply: F = sum_topK c_k^p * w + sum_all c_k^p * (1-w)

This focuses the optimization on reducing the worst peaks.

The minimum is 0 when h is the same at x and x+k (both 0 or both 1).

This suggests that a good h should have some "self-similarity" at all scales. A periodic pattern with period 2/n (alternating 0,1,0,1,...) would give zero overlap for even shifts but maximum overlap for odd shifts. Not good.

A "blocks" pattern like 1111000011110000... might work better. The key is to have the transition points (where h changes from 0 to 1 or 1 to 0) be spread out so that for any shift k, the number of transitions is small.

This is related to the concept of a "perfect difference set" or "Singer difference set" in combinatorics!

For a binary sequence of length n with n/2 ones, the cross-correlation with the complement at shift k counts the number of "1 followed by 0" or "0 followed by 1" pairs at distance k. We want to minimize the maximum such count.

This is similar to a "Golay sequence" or "m-sequence" design, but those are for autocorrelation.

    target = n / 2.0
    if abs(s - target) < 1e-14:
        return h
    diff = target - s
    if diff > 0:
        w = 1.0 - h
        if w.sum() > 1e-14:
            h = h + diff * w / w.sum()
    else:
        w = h.copy()
        if w.sum() > 1e-14:

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = np.maximum(corr, 1e-14)
