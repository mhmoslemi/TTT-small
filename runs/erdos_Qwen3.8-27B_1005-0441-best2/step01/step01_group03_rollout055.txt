import numpy as np
import time

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def get_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def get_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_lag(h, lag, n):
    g = np.zeros(n)
    for j in range(n):
        if 0 <= j - lag < n:
            g[j] += 1.0 - h[j - lag]
        if 0 <= j + lag < n:
            g[j] -= h[j + lag]
    return g

def gradient_optimize(h, n, deadline, rng, n_lags=3):

    best_h = h.copy()
    best_c5 = get_c5(h, n)
    lr = 0.3

        ov = get_overlaps(h, n)
        top_idx = np.argsort(ov)[-n_lags:]
        g = np.zeros(n)
        cnt = 0
        for ki in top_idx:
            lag = int(ki) - (n - 1)
            if lag == 0:
                continue
            g += grad_lag(h, lag, n)
            cnt += 1
        if cnt == 0:
            break
        g /= cnt
        h_new = h - lr * g * dx
        h_new = project_feasible(h_new, n)
        c_new = get_c5(h_new, n)

            best_c5 = c_new

def to_binary(h, n, rng):
    """Convert to binary with exactly n//2 ones."""
    h_bin = (h >= 0.5).astype(float)
    cnt = int(h_bin.sum())
    if cnt > n // 2:
        idx = np.where(h_bin == 1)[0]
        flip = rng.choice(idx, size=cnt - n//2, replace=False)
        h_bin[flip] = 0
    elif cnt < n // 2:
        idx = np.where(h_bin == 0)[0]
        flip = rng.choice(idx, size=n//2 - cnt, replace=False)
        h_bin[flip] = 1
    return h_bin

    best_c5 = get_c5(h_bin, n)
    best_h = h_bin.copy()
    cur_c5 = best_c5
    T = 0.003
    T_min = 1e-12
    stall = 0
    
    while time.time() < deadline:
        ones_idx = np.where(h_bin == 1)[0]
        zeros_idx = np.where(h_bin == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        # Make n_swaps random pair swaps
        for _ in range(n_swaps):

            j = zeros_idx[rng.randint(len(zeros_idx))]
            h_bin[i] = 0.0

Let me also add: **exhaustive local search** - for each position, try both values (0 and 1) and pick the better one, iterating until convergence. This is like coordinate descent on the binary variables.

For n=46: 46 positions, each try 2 values → 92 evaluations per full pass. Each evaluation is O(n²) = O(2116). So one pass is ~200K ops. Can do thousands of passes in a second.

But wait, if I change one position from 0 to 1, I need to also change another from 1 to 0 to maintain the sum. So it's always a pair swap.

**Exhaustive pair search**: For each of the C(23,1) = 23 possible 1-positions to turn off, and each of the 23 possible 0-positions to turn on: 23*23 = 529 pairs. For each, compute C₅. Pick the best. Repeat.

This is 529 * O(n²) per pass = 529 * 2116 ≈ 1.1M ops per pass. In numpy, each get_c5 call is maybe 10μs, so 529 calls = 5ms per pass. Can do ~200 passes per second. Over 1000 seconds, that's 200,000 passes. More than enough.

This exhaustive local search is much more reliable than random SA.
