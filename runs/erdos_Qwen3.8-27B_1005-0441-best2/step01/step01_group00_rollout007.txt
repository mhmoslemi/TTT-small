            if rem() < 25:
                break
            hr = _slsqp_pass(h, lags, n, bounds, cons, eps, 500)
            if hr is not None:
                v = _c5(hr)
                if v < best_val:
                    best_val = v
                    best_h = hr.copy()
                    h = hr.copy()
        # Verify all lags; if a non-active lag is now max, re-expand
        c = _corr(h)
        cmax = c.max()
        near_max = np.where(c > 0.999 * cmax)[0]
        if len(near_max) > len(lags) + 5:
            lags = np.sort(np.union1d(lags, near_max))
            lags = lags[-25:]

    # Phase 3: Subgradient polishing
    if rem() > 40:
        h = best_h.copy()
        lr = 0.003
        stale = 0
        for it in range(5000):
            if rem() < 8:
                break
            c = _corr(h)
            cmax = c.max()
            active = np.where(c > 0.998 * cmax)[0]
            g = np.zeros(n)
            for idx in active:
                m = idx - (n - 1)
                g += _single_grad(h, m, n)
            g /= max(len(active), 1)
            h_new = _proj(h - lr * g, n)
            v = _c5(h_new)
            if v < best_val - 1e-15:
                best_val = v
                best_h = h_new.copy()
                h = h_new
                stale = 0
                lr = min(lr * 1.005, 0.012)
            else:
                stale += 1
                lr *= 0.94
                if stale > 200 or lr < 1e-9:
                    break

    # Phase 4: Interpolate to 2n and refine
    if best_h is not None and rem() > 80:
        n_old = len(best_h)
        n_new = min(n_old * 2, 280)
        x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
        # Handle boundary: append first value for interpolation continuity
        h_ext = np.concatenate([best_h, best_h[:1]])
        x_ext = np.concatenate([x_old, [2.0]])
        h_hi = np.interp(x_new + 1.0 / n_new, x_ext, h_ext)
        h_hi = _proj(h_hi, n_new)
        b2 = [(0.0, 1.0)] * n_new
        c2 = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n_new,)}]
        al2 = np.arange(2 * n_new - 1)
        hh = h_hi
        for eps in [0.04, 0.015, 0.006]:
            if rem() < 40:
                break
            hr = _slsqp_pass(hh, al2, n_new, b2, c2, eps, 250)
            if hr is not None and _c5(hr) < best_val:
                best_val = _c5(hr)
                best_h = hr.copy()
                hh = hr.copy()
        # Focused pass on hires
        if rem() > 30:
            lags2 = _find_active(hh, M=15)
            for eps in [0.003, 0.001]:
                if rem() < 25:
                    break
                hr = _slsqp_pass(hh, lags2, n_new, b2, c2, eps, 400)
                if hr is not None and _c5(hr) < best_val:
                    best_val = _c5(hr)
                    best_h = hr.copy()

    # Phase 5: Final micro-polish
    if best_h is not None and rem() > 15:
        nn = len(best_h)
        b3 = [(0.0, 1.0)] * nn
        c3 = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (nn,)}]
        lg3 = _find_active(best_h, M=8)
        hr = _slsqp_pass(best_h, lg3, nn, b3, c3, 0.001, 500)
        if hr is not None and _c5(hr) < best_val:
            best_val = _c5(hr)
            best_h = hr

    if best_h is None:
        nr = 70
        best_h = np.full(nr, 0.5)
        best_val = _c5(best_h)

    best_h = _proj(best_h, len(best_h))
    best_val = _c5(best_h)
    return (best_h, best_val, len(best_h))
