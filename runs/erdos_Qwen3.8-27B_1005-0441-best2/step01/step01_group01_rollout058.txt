def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr)), corr

    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    resid = target - np.sum(proj)

            np.clip(proj, 0.0, 1.0, out=proj)

def grad_c5(h, n):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c5 = np.max(corr)
    thresh = c5 - 1e-12
    active = np.where(corr >= thresh)[0]
    grad = np.zeros(n)
    for k_idx in active:
        L = k_idx - (n - 1)
        # grad_L[j] = (1-h[j+L]) for valid j+L - h[j-L] for valid j-L
        if L >= 0:
            grad[:n-L] += 1.0 - h[L:]
            grad[L:] -= h[:n-L]
        else:
            grad[-L:] += 1.0 - h[:n+L]
            grad[:n+L] -= h[-L:]
    grad /= len(active)
    return c5, grad * dx

def binary_local_search(h, n, max_sweeps=200, time_limit=None, t0=None):
    """Swap-based local search on binary h."""
    h = (h > 0.5).astype(float)
    c5, _ = compute_c5(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for sweep in range(max_sweeps):
        if time_limit is not None and t0 is not None and time.time() - t0 > time_limit:

                h[a] = 1.0
                h[b] = 0.0

            zeros = np.where(h == 0.0)[0]
            improved = True
        
        if not improved:

    return h, c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr)), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    resid = target - np.sum(proj)
    if abs(resid) > 1e-12:
        mask = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += resid / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj

def grad_c5(h):
    """Compute subgradient of C5 wrt h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c5 = np.max(corr)
    thresh = c5 - 1e-12
    active = np.where(corr >= thresh)[0]
    grad = np.zeros(n)
    for k_idx in active:
        L = k_idx - (n - 1)
        if L >= 0:
            grad[:n-L] += 1.0 - h[L:]
            grad[L:] -= h[:n-L]
        else:
            grad[-L:] += 1.0 - h[:n+L]
            grad[:n+L] -= h[-L:]
    grad /= len(active)
    return c5, grad * dx

def smooth_max_obj(h, alpha):
    """Smooth approximation of C5 for gradient-based methods."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -400.0, 400.0)
    sm = m + np.log(np.sum(np.exp(shifted))) / alpha
    return sm

def smooth_max_grad(h, alpha):
    """Gradient of smooth max C5."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -400.0, 400.0)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    grad = np.zeros(n)
    for i, k_idx in enumerate(range(2*n-1)):
        if weights[i] < 1e-8:
            continue
        L = k_idx - (n - 1)
        if L >= 0:
            grad[:n-L] += weights[i] * (1.0 - h[L:])
            grad[L:] -= weights[i] * h[:n-L]
        else:
            grad[-L:] += weights[i] * (1.0 - h[:n+L])
            grad[:n+L] -= weights[i] * h[-L:]
    return grad * dx

def projected_gradient_step(h, grad, lr, n):
    """One step of projected gradient descent."""
    h_new = h - lr * grad
    return project_feasible(h_new, n)

def binary_local_search(h, n, t0, budget_frac=0.3, max_time_s=60):
    """Best-improvement swap local search."""
    h = (h > 0.5).astype(float)
    c5, _ = compute_c5(h)
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    
    time_limit = min(max_time_s, (budget_s * budget_frac) - (time.time() - t0))  # Can't do this without budget_s
    # Actually let's pass time limit differently
    
    while True:
        elapsed = time.time() - t0
        if elapsed > time_limit:
            break
        best_delta = 0
        best_a, best_b = -1, -1
        h_save = h.copy()
        for a in ones:
            for b in zeros:
                h = h_save.copy()
                h[a] = 0.0
                h[b] = 1.0
                c, _ = compute_c5(h)
                delta = c - c5
                if delta < best_delta - 1e-15:
                    best_delta = delta
                    best_a, best_b = int(a), int(b)
        if best_a < 0:
            break
        h[best_a] = 0.0
        h[best_b] = 1.0
        c5 += best_delta
        ones = np.where(h == 1.0)[0].tolist()
        zeros = np.where(h == 0.0)[0].tolist()
    
    c5, _ = compute_c5(h)
    return h, c5

def adam_optimize(h, n, t0, time_limit_s, max_iter=5000):
    """Adam optimizer with projection."""
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.1
    c5_best, _ = compute_c5(h)
    
    for t in range(1, max_iter+1):
        if time.time() - t0 > time_limit_s:
            break
        c5, grad = grad_c5(h)
        if c5 < c5_best:
            c5_best = c5
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h_new, n)
        # Decay learning rate
        lr *= 0.999
    return h, c5_best

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr)), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(h - lam, 0.0, 1.0)
    resid = target - np.sum(proj)
    if abs(resid) > 1e-12:
        mask = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += resid / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj

def grad_c5_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -400.0, 400.0)
    e = np.exp(shifted)
    w = e / e.sum()
    obj = m + np.log(e.sum()) / alpha
    grad = np.zeros(n)
    for i in range(2*n - 1):
        wi = w[i]
        if wi < 1e-10:
            continue
        L = i - (n - 1)
        if L >= 0:
            grad[:n-L] += wi * (1.0 - h[L:])
            grad[L:] -= wi * h[:n-L]
        else:
            grad[-L:] += wi * (1.0 - h[:n+L])
            grad[:n+L] -= wi * h[-L:]
    return obj, grad * dx

def binary_local_search(h, n, t0, time_limit):
    h = (h > 0.5).astype(float)
    c5, _ = compute_c5(h)
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    while time.time() - t0 < time_limit:
        best_delta = 1e-15
        best_a, best_b = -1, -1
        for a in ones:
            h[a] = 0.0
            for b in zeros:
                h[b] = 1.0
                c, _ = compute_c5(h)
                h[b] = 0.0
                delta = c - c5
                if delta < -best_delta:
                    best_delta = -delta
                    best_a, best_b = int(a), int(b)
            h[a] = 1.0
        if best_a < 0:
            break
        h[best_a] = 0.0
        h[best_b] = 1.0
        c5 -= best_delta
        ones = np.where(h == 1.0)[0].tolist()
        zeros = np.where(h == 0.0)[0].tolist()
    c5, _ = compute_c5(h)
    return h, c5

def sa_binary_search(h, n, t0, time_limit, rng):
    """Simulated annealing on binary h."""
    h = (h > 0.5).astype(float)
    c5, _ = compute_c5(h)
    T = 0.01
    best_h = h.copy()
    best_c5 = c5
    while time.time() - t0 < time_limit:
        a = int(rng.integers(n))
        b = int(rng.integers(n))
        if h[a] == h[b]:
            continue
        h[a], h[b] = h[b], h[a]
        c, _ = compute_c5(h)
        delta = c - c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
            c5 = c
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[a], h[b] = h[b], h[a]
        T *= 0.999
    return best_h, best_c5

def continuous_refine(h, n, t0, time_limit, alpha=100.0):
    """Adam-like gradient descent with soft-max objective."""
    h = project_feasible(h.copy(), n)
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr = 0.05
    c5_best, _ = compute_c5(h)
    for t in range(1, 50000):
        if time.time() - t0 > time_limit:
            break
        obj, grad = grad_c5_smooth(h, alpha)
        c5, _ = compute_c5(h)
        if c5 < c5_best:
            c5_best = c5
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        h = project_feasible(h - lr * m_hat / (np.sqrt(v_hat) + eps), n)
        if t % 500 == 0:
            lr *= 0.9
    return h, c5_best

def make_block(n, nblocks):
    h = np.zeros(n)
    block_size = n // (2 * nblocks)
    gap = n // nblocks
    for i in range(nblocks):
        start = i * gap
        h[start:start+block_size] = 1.0
    return project_feasible(h, n)

def make_periodic(n, period, duty):
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < duty:
            h[i] = 1.0
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)

    return h

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr)), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(h - lam, 0.0, 1.0)
    resid = target - np.sum(proj)
    if abs(resid) > 1e-12:
        mask = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            proj[mask] += resid / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj

def grad_c5_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -400.0, 400.0)
    e = np.exp(shifted)
    e_sum = e.sum()
    w = e / e_sum
    obj = m + np.log(e_sum) / alpha
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-10:
            continue
        L = i - (n - 1)
        if L >= 0:
            grad[:n - L] += wi * (1.0 - h[L:])
            grad[L:] -= wi * h[:n - L]
        else:
            grad[-L:] += wi * (1.0 - h[:n + L])
            grad[:n + L] -= wi * h[-L:]
    return obj, grad * dx

def binary_local_search(h, n, deadline):
    h = (h > 0.5).astype(float)
    c5, _ = compute_c5(h)
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    while time.time() < deadline:
        best_delta = 1e-15
        best_a, best_b = -1, -1
        for a in ones:
            h[a] = 0.0
            for b in zeros:
                h[b] = 1.0
                c, _ = compute_c5(h)
                h[b] = 0.0
                delta = c - c5
                if delta < -best_delta:
                    best_delta = -delta
                    best_a, best_b = int(a), int(b)
            h[a] = 1.0
        if best_a < 0:
            break
        h[best_a] = 0.0
        h[best_b] = 1.0
        c5 -= best_delta
        ones = np.where(h == 1.0)[0].tolist()
        zeros = np.where(h == 0.0)[0].tolist()
    c5, _ = compute_c5(h)
    return h, c5

def sa_binary_search(h, n, t0, deadline, rng):
    h = (h > 0.5).astype(float)
    c5, _ = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    T = 0.005
    while time.time() < deadline:
        a = int(rng.integers(n))
        b = int(rng.integers(n))
        if h[a] == h[b]:
            continue
        h[a], h[b] = h[b], h[a]
        c, _ = compute_c5(h)
        delta = c - c5
        accept = delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T))
        if accept:
            c5 = c
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[a], h[b] = h[b], h[a]
        T *= 0.9995
    return best_h, best_c5

def continuous_refine(h, n, deadline, alpha=100.0):
    h = project_feasible(h.copy(), n)
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr = 0.08
    decay = 0.9995
    c5_best, _ = compute_c5(h)
    for t in range(1, 100000):
        if time.time() > deadline:
            break
        _, grad = grad_c5_smooth(h, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        step_scale = min(1.0, 0.1 / (np.linalg.norm(step) + 1e-10))
        h = project_feasible(h - step * step_scale, n)
        c5, _ = compute_c5(h)
        if c5 < c5_best:
            c5_best = c5
        if t % 1000 == 0:
            lr *= 0.7
    return h, c5_best

def make_blocks(n, nblocks):
    h = np.zeros(n)
    seg = n // nblocks
    blk = seg // 2
    for i in range(nblocks):
        h[i * seg:i * seg + blk] = 1.0
    return project_feasible(h, n)

def make_periodic(n, period, duty):
    h = np.zeros(n)
    idx = np.arange(n)
    h[(idx % period) < duty] = 1.0
    return project_feasible(h, n)

def make_random_binary(n, rng):
    h = np.zeros(n)
    h[rng.permutation(n)[:n // 2]] = 1.0
    return h
