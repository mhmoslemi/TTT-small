import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project(h, target):
    n = len(h)
    h = h.copy()
    lo, hi = -1.0, 2.0
    for _ in range(80):
        tau = (lo + hi) / 2.0
        s = np.clip(h - tau, 0.0, 1.0).sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def _soft_max(corr, beta):
    mc = np.max(corr)
    return mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta

def _sm_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = (w / ws) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wi = w[k]
        if wi < 1e-15:
            continue
        d = k - (n - 1)
        if d > 0:
            grad[d:n] += wi * (1.0 - h[:n-d])
            grad[:n-d] -= wi * h[d:n]
        elif d < 0:
            dd = -d
            grad[:n-dd] += wi * (1.0 - h[dd:n])
            grad[dd:n] -= wi * h[:n-dd]
        else:
            grad += wi * (1.0 - 2.0 * h)
    return grad

def _pgd(h, betas, iters_per_beta, t_budget):
    n = len(h)
    target = n / 2.0
    t0 = time.time()
    best_c = compute_c5(h)
    best_h = h.copy()
    for beta in betas:
        hiter = h.copy()
        for it in range(iters_per_beta):
            if time.time() - t0 > t_budget:
                return best_h, best_c
            g = _sm_grad(hiter, beta)
            # Backtracking line search
            alpha = 0.1
            f_cur = _soft_max(np.correlate(hiter, 1-hiter, "full"), beta)
            for _ls in range(30):
                h_try = _project(hiter - alpha * g, target)
                f_try = _soft_max(np.correlate(h_try, 1-h_try, "full"), beta)
                if f_try <= f_cur - 1e-4 * alpha * np.dot(g, hiter - h_try):
                    break
                alpha *= 0.5
            hiter = h_try
            c = compute_c5(hiter)
            if c < best_c:
                best_c = c
                best_h = hiter.copy()
            h = hiter.copy()
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project(h, target):
    n = len(h)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        tau = (lo + hi) / 2.0
        s = np.clip(h - tau, 0.0, 1.0).sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def _sm_val(corr, beta):
    mc = np.max(corr)
    return float(mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta)

def _sm_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = (w / ws) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wi = w[k]
        if wi < 1e-15:
            continue
        d = k - (n - 1)
        if d > 0:
            grad[d:n] += wi * (1.0 - h[:n - d])
            grad[:n - d] -= wi * h[d:n]
        elif d < 0:
            dd = -d
            grad[:n - dd] += wi * (1.0 - h[dd:n])
            grad[dd:n] -= wi * h[:n - dd]
        else:
            grad += wi * (1.0 - 2.0 * h)
    return grad

def _pgd(h, beta, max_iter, step, t_dead):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iter):
        if time.time() > t_dead:
            break
        g = _sm_grad(h, beta)
        h_new = _project(h - step * g, target)
        if it % 5 == 0:
            c = compute_c5(h_new)
            if c < best_c:
                best_c = c
                best_h = h_new.copy()
        h = h_new
    c = compute_c5(h)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c

def _coord_desc(h, n_rounds, t_dead, rng):
    n = len(h)
    target = n / 2.0
    best = compute_c5(h)
    best_h = h.copy()
    for _r in range(n_rounds):
        if time.time() > t_dead:
            break
        improved = False
        for i in range(n):
            if (best_h[i] < 0.005):
                continue
            for j in range(n):
                if j == i or (best_h[j] > 0.995):
                    continue
                for delta in [0.01, -0.01, 0.02, -0.02, 0.005, -0.005]:
                    a = best_h[i] + delta
                    b = best_h[j] - delta
                    if a < 0 or a > 1 or b < 0 or b > 1:
                        continue
                    ht = best_h.copy()
                    ht[i] = a
                    ht[j] = b
                    ct = compute_c5(ht)
                    if ct < best - 1e-13:
                        best = ct
                        best_h = ht.copy()
                        improved = True
                        break
                if improved:
                    break
            if time.time() > t_dead:
                break
    return best_h, best

def _lbfgs_free(h, beta, t_dead):
    n = len(h)
    target = n / 2.0
    free = np.where((h > 0.002) & (h < 0.998))[0]
    if len(free) < 2:
        return h, compute_c5(h)
    if len(free) > 40:
        scores = np.abs(h[free] - 0.5)
        idx = np.argsort(scores)[:40]
        free = free[idx]
    h_fixed = h.copy()

    def obj(x):
        hf = h_fixed.copy()
        hf[free] = np.clip(x, 0.0, 1.0)
        hf = _project(hf, target)
        corr = np.correlate(hf, 1.0 - hf, "full")
        return _sm_val(corr, beta)

    def grad(x):
        hf = h_fixed.copy()
        hf[free] = np.clip(x, 0.0, 1.0)
        hf = _project(hf, target)
        gf = _sm_grad(hf, beta)
        return gf[free]

    x0 = h[free].copy()
    bounds = list(zip(np.zeros(len(free)), np.ones(len(free))))
    try:
        res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 300, 'gtol': 1e-7})
        hn = h_fixed.copy()
        hn[free] = np.clip(res.x, 0.0, 1.0)
        hn = _project(hn, target)
        cn = compute_c5(hn)
        if cn < compute_c5(h):
            return hn, cn
    except Exception:
        pass
    return h, compute_c5(h)

    rng = np.random.RandomState(seed)
    t_start = time.time()
    try:

    if fc < best_c:
        best_c = fc
    return (best_h, best_c, n)
