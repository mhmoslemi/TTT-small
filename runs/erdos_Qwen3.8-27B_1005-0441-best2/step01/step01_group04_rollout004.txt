    dx = 2.0 / n

    cm = np.max(corr)
    s = np.exp(alpha * (corr - cm))

def _smooth_max_grad(h, alpha):

import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        p = np.clip(h - mid, 0.0, 1.0)
        if np.sum(p) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _smooth_max_val(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(corr)
    s = np.exp(alpha * (corr - cm))
    return cm + np.log(np.sum(s)) / alpha

def _smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(corr)
    s = np.exp(alpha * (corr - cm))
    W = s / np.sum(s)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        for m in range(2 * n - 1):
            k = m - (n - 1)
            w = W[m]
            if w < 1e-15:
                continue
            if 0 <= j - k < n:
                g += w * (1.0 - h[j - k])
            if 0 <= j + k < n:
                g -= w * h[j + k]
        grad[j] = g * dx
    f_val = cm + np.log(np.sum(s)) / alpha
    return f_val, grad

def _proj_grad(h0, alpha, max_iter=500, lr0=0.04):
    n = len(h0)
    h = _project_feasible(h0.copy(), n)
    f, g = _smooth_max_grad(h, alpha)
    lr = lr0
    stall = 0
    for it in range(max_iter):
        h_new = _project_feasible(h - lr * g, n)
        f_new = _smooth_max_val(h_new, alpha)
        if f_new < f - 1e-15:
            h = h_new
            f = f_new
            f, g = _smooth_max_grad(h, alpha)
            lr = min(lr * 1.1, 1.0)
            stall = 0
        else:
            lr *= 0.5
            stall += 1
            if lr < 1e-12 or stall > 30:
                break
    return h, f

def _slsqp_opt(h0, alpha, maxiter=300):
    n = len(h0)
    def _obj(x):
        return _smooth_max_val(x, alpha)
    def _grad(x):
        return _smooth_max_grad(x, alpha)[1]
    def _eq(x):
        return np.sum(x) - n / 2.0
    def _eq_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq, 'jac': _eq_jac}]
    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return res.x
    except Exception:
        return h0.copy()

def _perturb(h, rng, scale):
    return _project_feasible(h + rng.normal(0, scale, len(h)), len(h))

def _random_h(n, rng):
    return _project_feasible(rng.random(n), n)

def _sine_h(n, freq, amp):
    x = np.linspace(0, 2.0 * np.pi * freq, n, endpoint=False)
    return _project_feasible(0.5 + amp * np.sin(x), n)

def _block_h(n, period, frac):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i % period) < int(frac * period) else 0.0
    return _project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    best_c5 = np.inf
    best_h = None

    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if h_init is not None:
        h_init = _project_feasible(h_init, len(h_init))
        c5_init = _eval_c5(h_init)
        best_c5 = c5_init
        best_h = h_init.copy()

    def _try_update(h):
        nonlocal best_c5, best_h
        c5 = _eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    alpha_schedule = [10, 30, 80, 200, 500, 1200, 3000, 8000]

    if h_init is not None:
        h_curr = h_init.copy()
        for alpha in alpha_schedule:
            if time.time() - t0 > budget_s - 250:
                break
            h_curr, _ = _proj_grad(h_curr, alpha, max_iter=400, lr0=0.03)
            _try_update(h_curr)
        if time.time() - t0 < budget_s - 150:
            h_s = _slsqp_opt(best_h, 800, 300)
            h_s = _project_feasible(h_s, len(h_s))
            _try_update(h_s)
        if time.time() - t0 < budget_s - 100:
            h_s2 = _proj_grad(h_s, 15000, max_iter=300, lr0=0.01)
            _try_update(h_s2)

    n_list = [47, 50, 53, 60, 67, 74, 80, 90]
    for n in n_list:
        if time.time() - t0 > budget_s - 120:
            break
        starts = [_random_h(n, rng), _sine_h(n, 1, 0.35), _sine_h(n, 2, 0.3),
                  _sine_h(n, 3, 0.25), _sine_h(n, 5, 0.2), _block_h(n, 3, 0.7),
                  np.full(n, 0.5)]
        for h0 in starts:
            if time.time() - t0 > budget_s - 60:
                break
            for alpha in [50, 200, 800]:
                if time.time() - t0 > budget_s - 40:
                    break
                h_opt, _ = _proj_grad(h0, alpha, max_iter=250, lr0=0.03)
                _try_update(h_opt)
            if time.time() - t0 > budget_s - 40:
                break
            h_opt2, _ = _proj_grad(h_opt, 3000, max_iter=250, lr0=0.015)
            _try_update(h_opt2)

    if best_h is not None:
        for _ in range(40):
            if time.time() - t0 > budget_s - 40:
                break
            h_p = _perturb(best_h, rng, 0.04)
            h_opt, _ = _proj_grad(h_p, 400, max_iter=250, lr0=0.02)
            _try_update(h_opt)
        for _ in range(30):
            if time.time() - t0 > budget_s - 30:
                break
            h_p = _perturb(best_h, rng, 0.015)
            h_opt, _ = _proj_grad(h_p, 2000, max_iter=250, lr0=0.01)
            _try_update(h_opt)

    if best_h is not None and time.time() - t0 < budget_s - 20:
        h_f = _slsqp_opt(best_h, 3000, 400)
        h_f = _project_feasible(h_f, len(h_f))
        _try_update(h_f)
        if time.time() - t0 < budget_s - 10:
            h_f2 = _proj_grad(h_f, 20000, max_iter=400, lr0=0.008)
            _try_update(h_f2)

    if best_h is None:
        best_h = np.full(47, 0.5)
        best_c5 = _eval_c5(best_h)

    best_h = _project_feasible(best_h, len(best_h))
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = _eval_c5(best_h)
    return best_h, best_c5, len(best_h)
