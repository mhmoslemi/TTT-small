import time

def project_sum(h, target, n):
    """Project h onto [0,1]^n ∩ {sum = target}"""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Add to smallest elements
            idx = np.argsort(h)
            for i in idx:
                if h[i] < 1.0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                    diff -= add
                    if abs(diff) < 1e-12:
                        break
        else:
            idx = np.argsort(-h)
            for i in idx:
                if h[i] > 0.0:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                    diff += sub
                    if abs(diff) < 1e-12:
                        break
    return h

    grad = np.zeros(n)
    if L >= 0:
        # f_L = dx * sum_{j=L}^{n-1} h[j-L]*(1-h[j])

        #          - dx*h[i-L] for i in [L, n-1] (second factor)

        for i in range(L, n):

        if abs(diff) < 1e-13:

        if diff > 0:

def eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(8000):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = 1.0 - h > 1e-14
            if not m.any():
                break
            h[m] += (1.0 - h[m]) * (d / (1.0 - h[m]).sum())
        else:
            m = h > 1e-14
            if not m.any():
                break
            h[m] -= h[m] * (-d / h[m].sum())
    return h

def grad_lag(h, L):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if L >= 0:
        m = n - L
        if m > 0:
            g[:m] += 1.0 - h[L:]
            g[L:] -= h[:m]
    else:
        Lp = -L
        m = n - Lp
        if m > 0:
            g[Lp:] += 1.0 - h[:m]
            g[:m] -= h[Lp:]
    return g * dx

def subgrad_loop(h, alpha, n_iter, t0, budget, margin):
    n = len(h)
    target = n / 2.0
    h = project_sum(h, target)
    best = h.copy()
    best_c5 = eval_c5(h)
    eta = 1.0
    momentum = np.zeros(n)
    beta = 0.7
    no_improve = 0
    for _ in range(n_iter):
        if time.time() - t0 > budget - margin:
            break
        c = compute_c(h)
        idx = np.argsort(c)[::-1][:min(12, len(c))]
        c_top = c[idx]
        w = np.exp(alpha * (c_top - c_top[0]))
        w /= w.sum()

            best_c5 = c5_new
            best = h_new.copy()
            h = h_new
            no_improve = 0
            eta = min(eta * 1.03, 5.0)
        else:
            no_improve += 1
            eta *= 0.94
            if eta < 0.02 or no_improve > 400:
                break
    return best, best_c5

def _slsqp_obj(x):
    cx = compute_c(x)
    m = cx.max()
    top = np.argsort(cx)[::-1][:6]
    return m + 0.03 * np.sum(cx[top] - m)

def _slsqp_eq(x):
    return x.sum() - x.size / 2.0

def slsqp_polish(h, t0, budget):
    n = len(h)
    target = n / 2.0
    try:
        res = minimize(_slsqp_obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _slsqp_eq}],
                       options={'maxiter': 400, 'ftol': 1e-14})
        hx = project_sum(res.x, target)
        if eval_c5(hx) < eval_c5(h):
            return hx
    except Exception:
        pass
    return h
