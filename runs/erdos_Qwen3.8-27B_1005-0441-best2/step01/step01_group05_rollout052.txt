    def energy(h):
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        if top_k >= len(corr):
            return np.sum(corr ** power)
        idx = np.argpartition(corr, -top_k)[-top_k:]
        return np.sum(corr[idx] ** power)
    
    def move(h, sigma):
        """Gaussian perturbation with constraint repair"""
        h_new = h + sigma * np.random.randn(n)
        h_new = np.clip(h_new, 0.0, 1.0)

            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            h_new[mask] = np.clip(h_new[mask] + step, 0.0, 1.0)
        return h_new

def _project(h, target):

    for _ in range(30):

        diff = target - s

def _corr_and_lags(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")

    scores = _evaluate_population(pop, top_k, power)
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_c5 = compute_c5(best_h)
    sigma = 0.06
    gen = 0
    while time.time() - t0 < max_time:
        gen_batch = 20
        for _ in range(gen_batch):
            if time.time() - t0 > max_time - 2:
                break
            order = np.argsort(scores)
            parents = [pop[order[i]] for i in range(min(8, pop_size))]
            children = []
            for _ in range(pop_size // 2):
                p = parents[np.random.randint(len(parents))]
                s = sigma * (0.5 + np.random.random() * 1.5)
                c = _project(p + s * np.random.randn(n), target)
                children.append(c)
                if time.time() - t0 > max_time - 2:
                    break
            if not children:
                break
            child_scores = _evaluate_population(children, top_k, power)
            all_ind = [pop[i] for i in range(pop_size) if i not in order[:pop_size//2]] + children
            all_sc = [scores[i] for i in range(pop_size) if i not in order[:pop_size//2]] + child_scores
            combined_order = np.argsort(all_sc)
            new_pop = [all_ind[i] for i in combined_order[:pop_size]]
            new_scores = [all_sc[i] for i in combined_order[:pop_size]]
            pop, scores = new_pop, new_scores
            ci = int(np.argmin(scores))
            c5 = compute_c5(pop[ci])
            if c5 < best_c5:
                best_c5 = c5
                best_h = pop[ci].copy()
        sigma = max(0.005, sigma * 0.92)
        gen += 1
    return best_h, best_c5

def _slight_upsample(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) * 2.0 / n_old
    x_new = np.arange(n_new) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _binary_hill_climb(h_init, max_time, seed):
    np.random.seed(seed)
    n = len(h_init)
    target = n / 2.0
    h = np.where(h_init > 0.5, 1.0, 0.0)
    h = _project(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    T = 0.01
    while time.time() - t0 < max_time:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c5_new = compute_c5(h)
        if c5_new < best_c5 or np.random.random() < np.exp(-(c5_new - compute_c5(best_h)) / T):
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
        else:
            h[i], h[j] = 1.0, 0.0
        T = max(0.001, T * 0.9995)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def _rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(160, 0.5)
    n = len(h0)
    target = n / 2.0
    h0 = _project(h0, target)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: PGD with multiple settings
    if _rem() > 120:
        for tk, pw, tm in [(5, 4, 60), (3, 6, 40), (7, 3, 30)]:
            if _rem() < tm + 10:
                break
            h_o, c5_o = _pgd_optimize(best_h.copy(), min(tm, _rem() - 10), seed + tk * 7 + pw, tk, pw)
            if c5_o < best_c5:
                best_h, best_c5 = h_o, c5_o

    # Phase 2: SLSQP refinement
    if _rem() > 60:
        h_o, c5_o = _slsqp_refine(best_h.copy(), min(50, _rem() - 10), seed + 100, 3, 6)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o

    # Phase 3: ES global search
    if _rem() > 200:
        h_o, c5_o = _es_loop(best_h.copy(), min(150, _rem() - 20), seed + 500, 25, 5, 4)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o
            # PGD after ES
            if _rem() > 50:
                h_o2, c5_o2 = _pgd_optimize(best_h.copy(), min(40, _rem() - 10), seed + 800, 3, 6)
                if c5_o2 < best_c5:
                    best_h, best_c5 = h_o2, c5_o2

    # Phase 4: Binary hill climb (exploit near-binary structure)
    if _rem() > 80:
        h_o, c5_o = _binary_hill_climb(best_h.copy(), min(60, _rem() - 10), seed + 2000)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o
        # Then PGD from binary solution (allow fractional)
        if _rem() > 40:
            h_soft = best_h + 0.05 * np.random.randn(len(best_h))
            h_soft = _project(h_soft, target)
            h_o2, c5_o2 = _pgd_optimize(h_soft, min(30, _rem() - 10), seed + 3000, 3, 5)
            if c5_o2 < best_c5:
                best_h, best_c5 = h_o2, c5_o2

    # Phase 5: Try n=120
    if _rem() > 100:
        n2 = 120
        h2 = _slight_upsample(best_h, n2)
        h_o, c5_o = _pgd_optimize(h2, min(50, _rem() - 10), seed + 4000, 5, 4)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o
        if _rem() > 30:
            h_o, c5_o = _slsqp_refine(best_h.copy(), min(25, _rem() - 5), seed + 4100, 3, 6)
            if c5_o < best_c5:
                best_h, best_c5 = h_o, c5_o

    # Phase 6: Final polish
    if _rem() > 20:
        h_o, c5_o = _pgd_optimize(best_h.copy(), min(15, _rem() - 3), seed + 9999, 3, 8)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o
    if _rem() > 10:
        h_o, c5_o = _slsqp_refine(best_h.copy(), min(8, _rem() - 2), seed + 9998, 2, 10)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o

    best_h = _project(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
