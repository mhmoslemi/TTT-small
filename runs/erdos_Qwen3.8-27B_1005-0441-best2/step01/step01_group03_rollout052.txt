def smooth_extra_grad(h, n, beta):
    """Compute gradient of smoothed max overlap."""
    dx = 2.0 / n
    # Compute all overlaps
    ov = np.correlate(h, 1.0 - h, mode='full') * dx
    # Softmax weights (stabilized)
    ov_max = ov.max()
    exp_ov = np.exp((ov - ov_max) * beta)
    w = exp_ov / exp_ov.sum()
    # Compute gradient
    grad = np.zeros(n)
    n1 = n - 1  # n-1
    for k in range(2 * n - 1):
        L = k - n1
        wk = w[k]
        if wk < 1e-12:

        for j in range(n):
            g = 0.0
            jm = j - L
            if 0 <= jm < n:
                g += 1.0 - h[jm]
            jp = j + L
            if 0 <= jp < n:
                g -= h[jp]
            grad[j] += wk * dx * g
    return grad

import numpy as np
import time

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-13:
        mask = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        cnt = float(np.sum(mask))
        if cnt > 0:
            hp[mask] = np.clip(hp[mask] + diff / cnt, 0.0, 1.0)
    return hp

def get_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def all_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def coord_update(h, n, j, n_grid=300):
    """Update coordinate j to minimize max overlap (ignoring sum constraint)."""
    dx = 2.0 / n
    # Compute coefficient a_k and intercept b_k for each lag
    # overlap(k) = a_k * h[j] + b_k
    nk = 2 * n - 1
    a = np.zeros(nk)
    ov = all_overlaps(h, n)
    n1 = n - 1
    for k in range(nk):
        L = k - n1
        # h[j] appears in:
        # 1) as h[i] when i=j: need 0≤j-L<n, contributes h[j]*(1-h[j-L])
        # 2) as h[i-L] when i-L=j → i=j+L: need 0≤j+L<n, contributes h[j+L]*(1-h[j])
        coeff = 0.0
        if 0 <= j - L < n:
            coeff += (1.0 - h[j - L])
        if 0 <= j + L < n:
            coeff -= h[j + L]
        a[k] = dx * coeff
    
    b = ov - a * h[j]
    
    # Minimize max_k (a_k * t + b_k) for t in [0,1]
    t_grid = np.linspace(0.0, 1.0, n_grid)
    vals = np.max(a[:, None] * t_grid[None, :] + b[:, None], axis=0)
    best_j = np.argmin(vals)
    return t_grid[best_j]

def coordinate_descent(h, n, deadline, rng, max_rounds=5000):
    best_h = h.copy()
    best_c5 = get_c5(h, n)
    n_rounds = 0
    
    while time.time() < deadline - 2:
        n_rounds += 1
        # Random order
        order = rng.permutation(n)
        for j in order:
            t_new = coord_update(h, n, j)
            h[j] = np.clip(t_new, 0.0, 1.0)
        
        # Project to feasible set
        h = project_feasible(h, n)
        c5 = get_c5(h, n)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def lse_gradient_step(h, n, beta, lr):
    dx = 2.0 / n
    ov = np.correlate(h, 1.0 - h, mode='full') * dx
    ov_max = ov.max()
    exp_ov = np.exp((ov - ov_max) * beta)
    w = exp_ov / exp_ov.sum()
    
    grad = np.zeros(n)
    n1 = n - 1
    for k in range(2 * n - 1):
        L = k - n1
        wk = w[k]
        if wk < 1e-12:
            continue
        for j in range(n):
            g = 0.0
            jm = j - L
            if 0 <= jm < n:
                g += 1.0 - h[jm]
            jp = j + L
            if 0 <= jp < n:
                g -= h[jp]
            grad[j] += wk * dx * g
    return h - lr * grad

Let me vectorize the coefficient computation:

For coordinate j and lag L:
coeff = (1-h[j-L]) * I(0≤j-L<n) - h[j+L] * I(0≤j+L<n)

For a fixed j, as L varies from -(n-1) to (n-1):
- j-L ranges from j+(n-1) down to j-(n-1)
- Valid when 0 ≤ j-L < n, i.e., j-n+1 ≤ L ≤ j
- j+L ranges from j-(n-1) to j+(n-1)  
- Valid when 0 ≤ j+L < n, i.e., -j ≤ L < n-j

So:
term1[L] = 1-h[j-L] for L in [j-n+1, j] (clamped to [-(n-1), n-1])
term2[L] = h[j+L] for L in [-j, n-j-1] (clamped)

I can compute these with numpy slicing:

```python
def compute_ab(h, n, j):
    dx = 2.0 / n
    nk = 2 * n - 1
    a = np.zeros(nk)
    # L ranges from -(n-1) to (n-1), indexed k = L + (n-1) = 0 to nk-1
    # term1: 1-h[j-L] valid for 0≤j-L<n → L in [j-n+1, j]
    L_min1 = max(j - n + 1, -(n-1))
    L_max1 = min(j, n-1)
    if L_min1 <= L_max1:
        k_min1 = L_min1 + (n-1)
        k_max1 = L_max1 + (n-1)
        idx = np.arange(L_min1, L_max1 + 1)
        a[k_min1:k_max1+1] += (1.0 - h[j - idx]) * dx
    
    # term2: -h[j+L] valid for 0≤j+L<n → L in [-j, n-j-1]
    L_min2 = max(-j, -(n-1))
    L_max2 = min(n - j - 1, n - 1)
    if L_min2 <= L_max2:
        k_min2 = L_min2 + (n-1)
        k_max2 = L_max2 + (n-1)
        idx = np.arange(L_min2, L_max2 + 1)
        a[k_min2:k_max2+1] -= h[j + idx] * dx
    
    return a
```

This is O(n) per coordinate (vectorized). Then the 1D optimization is O(nk * n_grid) = O(91 * 300) ≈ 27K per coordinate. Total per round: 46 * (46 + 27K) ≈ 1.2M. Very fast!

Let me rewrite more carefully. Actually, I realize I can simplify. For a given j:

a[k] = dx * [(1-h[j-L_k]) * I(0≤j-L_k<n) - h[j+L_k] * I(0≤j+L_k<n)]

where L_k = k - (n-1).

Let me just compute this with two masked arrays:

```python
def coord_update_fast(h, n, j, n_grid=200):
    dx = 2.0 / n
    nk = 2 * n - 1
    L = np.arange(nk) - (n - 1)  # L values: -(n-1) to (n-1)
    
    # Coefficient of h[j] in overlap(L)
    a = np.zeros(nk)
    mask1 = (j - L >= 0) & (j - L < n)
    mask2 = (j + L >= 0) & (j + L < n)
    a[mask1] += (1.0 - h[j - L[mask1]]) * dx
    a[mask2] -= h[j + L[mask2]] * dx
    
    ov = all_overlaps(h, n)
    b = ov - a * h[j]
    
    # 1D minimization
    t_grid = np.linspace(0.0, 1.0, n_grid)
    vals = np.max(a[:, None] * t_grid[None, :] + b[:, None], axis=0)
    return t_grid[np.argmin(vals)]
```

import numpy as np
import time

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-13:
        mask = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        cnt = float(np.sum(mask))
        if cnt > 0:
            hp[mask] = np.clip(hp[mask] + diff / cnt, 0.0, 1.0)
    return hp

def get_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def all_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def coord_opt(h, n):
    dx = 2.0 / n
    nk = 2 * n - 1
    L_arr = np.arange(nk) - (n - 1)
    t_grid = np.linspace(0.0, 1.0, 300)
    best_h = h.copy()
    best_c5 = get_c5(h, n)

    while time.time() < deadline_global - 2:
        ov = all_overlaps(h, n)
        order = rng_global.permutation(n)
        for j in order:
            a = np.zeros(nk)

            h[j] = t_opt
            ov = ov + delta * a

def lse_grad(h, n, beta):
    dx = 2.0 / n
    ov = np.correlate(h, 1.0 - h, mode='full') * dx
    exp_ov = np.exp((ov - ov.max()) * beta)
    w = exp_ov / exp_ov.sum()
    grad = np.zeros(n)
    n1 = n - 1
    for k in range(2 * n - 1):
        L = k - n1
        wk = w[k]
        if wk < 1e-12:
            continue
        for j in range(n):
            g = 0.0
            jm = j - L
            if 0 <= jm < n:
                g += 1.0 - h[jm]
            jp = j + L
            if 0 <= jp < n:
                g -= h[jp]
            grad[j] += wk * dx * g
    return grad
