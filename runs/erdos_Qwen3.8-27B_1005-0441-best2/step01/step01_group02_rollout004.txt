    return mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smax(corr, beta):
    mc = np.max(corr)
    if beta < 1.0:
        return float(mc)
    shifted = np.clip(beta * (corr - mc), -700.0, 700.0)
    return float(mc + np.log(np.sum(np.exp(shifted))) / beta)

def smax_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    w = np.exp(np.clip(beta * (corr - mc), -700.0, 700.0))
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-15:
            continue
        d = (n - 1) - idx
        lo1 = max(0, -d)
        hi1 = min(n, n - d)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1+d:hi1+d])
        lo2 = max(0, d)
        hi2 = min(n, d + n)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2-d:hi2-d]
    return grad * dx

def project(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
        h = np.clip(h + (target - s) / n, 0.0, 1.0)
    return h

def binary_climb(h_in, max_passes, t_limit):
    n = len(h_in)
    b = (h_in >= 0.5).astype(float)
    s = int(b.sum())
    if s > n // 2:
        idxs = np.where(b == 1)[0]
        for i in range(s - n // 2):
            b[idxs[i]] = 0.0
    elif s < n // 2:
        idxs = np.where(b == 0)[0]
        for i in range(n // 2 - s):
            b[idxs[i]] = 1.0
    best_c = compute_c5(b)
    t0 = time.time()
    for _ in range(max_passes):
        if time.time() - t0 > t_limit:
            break
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        improved = False
        for i in ones:
            if time.time() - t0 > t_limit:
                break
            for j in zeros:
                b[i], b[j] = 0.0, 1.0
                c = compute_c5(b)
                b[i], b[j] = 1.0, 0.0
                if c < best_c - 1e-15:
                    b[i], b[j] = 0.0, 1.0
                    best_c = c
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return b, best_c

def grad_descend(h0, betas, t_limit):
    h = h0.copy()
    best_c = compute_c5(h)
    t0 = time.time()
    n = len(h)
    v = np.zeros(n)
    for beta in betas:
        if time.time() - t0 > t_limit:
            break
        lr = 0.08
        mom = 0.25
        for it in range(3000):
            if time.time() - t0 > t_limit:
                break
            g = smax_grad(h, beta)
            g = g - np.mean(g)
            if np.max(np.abs(g)) < 1e-14:
                break
            v = mom * v + lr * g
            vmax = np.max(np.abs(v))
            if vmax > 0.5:
                v = v * 0.5 / vmax
            f_cur = smax(np.correlate(h, 1.0 - h, mode="full"), beta)
            step_scale = 1.0
            ok = False
            for _ls in range(30):
                h_try = project(h - step_scale * v)
                f_try = smax(np.correlate(h_try, 1.0 - h_try, mode="full"), beta)
                if f_try < f_cur:
                    h = h_try
                    ok = True
                    break
                step_scale *= 0.5
            if not ok:
                v *= 0.5
                if step_scale < 1e-12:
                    break
            else:
                lr = min(lr * 1.015, 0.25)
            c = compute_c5(h)
            if c < best_c:
                best_c = c
    return h, best_c

def coord_descent(h, t_limit):
    n = len(h)
    best_c = compute_c5(h)
    t0 = time.time()
    step = 0
    deltas = [0.01, -0.01, 0.005, -0.005, 0.02, -0.02, 0.003, -0.003]
    no_improve = 0
    while time.time() - t0 < t_limit:
        j = step % n
        improved = False
        for delta in deltas:
            h_try = h.copy()
            h_try[j] += delta
            if h_try[j] < -1e-12 or h_try[j] > 1.0 + 1e-12:
                continue
            k = (j + 1) % n
            h_try[k] -= delta
            if h_try[k] < -1e-12 or h_try[k] > 1.0 + 1e-12:
                continue
            h_try = project(h_try)
            c = compute_c5(h_try)
            if c < best_c - 1e-15:
                h = h_try
                best_c = c
                improved = True
                no_improve = 0
                break
        if not improved:
            no_improve += 1
            if no_improve > n * len(deltas):
                break
        else:
            no_improve = 0
        step += 1
    return h, best_c
