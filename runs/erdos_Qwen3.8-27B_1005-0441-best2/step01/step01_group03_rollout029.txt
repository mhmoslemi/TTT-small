def compute_c5(h, n):

    c = np.correlate(h, 1.0 - h, mode='full')

    h_b = (h >= 0.5).astype(np.float64)
    s = int(h_b.sum())
    if s > target:
        idx = np.where(h_b == 1)[0]
        order = np.argsort(h[idx])
        h_b[idx[order[:s-target]]] = 0.0
    elif s < target:
        idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[idx])
        h_b[idx[order[:target-s]]] = 1.0

def local_search(h, n, deadline):
    best = h.copy()

    while time.time() < deadline:

    if mode == 0:
        h[:half] = 1.0
    elif mode == 1:
        h[::2][:half] = 1.0
    elif mode == 2:
        for i in range(n):
            if (i % (half+1)) < half // 2:
                h[i] = 1.0
    else:
        h[:half] = 1.0
    for _ in range(10):
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] != h[j]:
            h[i], h[j] = h[j], h[i]
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    bh_val, bc_val, bn_val = None, 1.0, 0

    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
        n0 = len(h0)
    except Exception:
        h0 = None
        n0 = 46

    # Phase 1: SA on binarized initial (30% budget)
    if h0 is not None:
        hb = binarize(h0, n0)
    else:
        hb = make_start(n0, rng)
    d1 = t0 + budget_s * 0.30
    bh1, bc1 = sa(hb, n0, d1, rng, T0=0.018, Tmin=0.0004, dc=0.9999)
    bh_val, bc_val, bn_val = bh1, bc1, n0

    # Phase 2: Other n values with varied starts (48% budget)
    ncands = [28, 30, 32, 34, 36, 38, 40, 42, 44, 48, 50, 52, 56, 60, 64]
    ncands = [x for x in ncands if x != n0]
    n_exc = len(ncands)
    per = budget_s * 0.48 / max(n_exc, 1)
    for idx, nt in enumerate(ncands):
        if time.time() > t0 + budget_s - 45:
            break
        dt = min(time.time() + per, t0 + budget_s - 45)
        mode = idx % 3
        hs = make_start(nt, rng, mode=mode)
        T0s = 0.020 + 0.005 * (idx % 3)
        bh_t, bc_t = sa(hs, nt, dt, rng, T0=T0s, Tmin=0.0005, dc=0.99985)
        if bc_t < bc_val:
            bh_val, bc_val, bn_val = bh_t, bc_t, nt

    # Phase 3: Continuous refinement of best (22% budget)
    if bh_val is not None and time.time() < t0 + budget_s - 15:
        d3 = t0 + budget_s - 8
        bh3, bc3 = cont_refine(bh_val, bn_val, d3, rng)
        if bc3 < bc_val:
            bh_val, bc_val = bh3, bc3

    # Final: ensure feasibility and validate
    hf = np.clip(bh_val, 0.0, 1.0).astype(np.float64)
    tgt = bn_val / 2.0
    diff = tgt - hf.sum()
    if abs(diff) > 1e-10:
        hf = hf + diff / bn_val
        hf = np.clip(hf, 0.0, 1.0)
    c5f = compute_c5(hf, bn_val)
    return hf, float(c5f), bn_val
