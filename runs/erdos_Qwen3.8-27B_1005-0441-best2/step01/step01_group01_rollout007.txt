from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_all(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        mask = (h > 0) & (h < 1)
        if np.sum(mask) == 0:
            h += d / n
            np.clip(h, 0.0, 1.0, out=h)
        else:
            h[mask] += d / np.sum(mask)
            np.clip(h, 0.0, 1.0, out=h)
    # Final correction
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += resid / np.sum(mask)
            np.clip(h, 0.0, 1.0, out=h)
    return h

def gradient_lp(h, p):
    """Compute gradient of (sum c^p)^(1/p) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # c[k] for k = 0, ..., 2n-2
    # lag[k] = k - (n-1)
    # grad[j] = sum_k c[k]^(p-1) * (1-h[j+lag]) - h[j-lag] (valid indices)
    #         all divided by (sum c^p)^(1-1/p), times dx

    S = np.sum(cp)
    if S < 1e-30:
        return np.zeros(n)
    w = np.power(np.maximum(corr, 0), p-1) / (S ** (1 - 1.0/p)) * dx
    
    grad = np.zeros(n)

    for k in range(2*n - 1):
