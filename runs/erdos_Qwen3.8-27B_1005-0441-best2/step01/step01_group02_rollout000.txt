            best_h = h_nm

    # Phase 2: Diverse candidate exploration
    t_rem = budget_s - (time.time() - t_start)
    candidates = []
    for freq in [2, 3, 4, 5, 6, 7, 8, 10]:
        candidates.append(_gen_sine(n, freq, rng.rand() * 6.28, rng))
        candidates.append(_gen_sine(n, freq, rng.rand() * 6.28, rng))
    for _ in range(6):
        candidates.append(_gen_blocks(n, rng))
    for _ in range(6):
        candidates.append(_gen_rand_bin(n, rng))
    for _ in range(4):
        candidates.append(_project(best_h + 0.14 * rng.randn(n)))

    for h_cand in candidates:
        t_rem = budget_s - (time.time() - t_start)
        if t_rem < 35:
            break
        h_opt, c_opt = _quick_opt(h_cand, t_budget=min(11, t_rem - 10), n_beta=3)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Subgradient top-k method
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 25:
        h_sg, c_sg = _subgrad_topk(best_h, n_steps=300, k_top=3)
        if c_sg < best_c:
            best_c = c_sg
            best_h = h_sg
        h_nm, c_nm = _nm_polish(best_h, maxiter=500)
        if c_nm < best_c:
            best_c = c_nm
            best_h = h_nm

    # Phase 4: Intensive SLSQP + NM on best
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 30:
        for i in range(3):
            if time.time() - t_start > budget_s - 30:
                break
            beta = 40.0 + 40.0 * i
            h_new = _slsqp(best_h, beta, maxiter=350)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        h_nm, c_nm = _nm_polish(best_h, maxiter=700)
        if c_nm < best_c:
            best_c = c_nm
            best_h = h_nm

    # Phase 5: More perturbed restarts
    t_rem = budget_s - (time.time() - t_start)
    for i in range(8):
        t_rem = budget_s - (time.time() - t_start)
        if t_rem < 25:
            break
        scale = 0.06 + 0.02 * i
        h_pert = _project(best_h + scale * rng.randn(n))
        h_opt, c_opt = _quick_opt(h_pert, t_budget=min(15, t_rem - 5), n_beta=3)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        h_nm2, c_nm2 = _nm_polish(h_opt, maxiter=300)
        if c_nm2 < best_c:
            best_c = c_nm2
            best_h = h_nm2

    # Phase 6: Coordinate descent fine-tuning
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 15:
        n_passes = min(35, max(5, int((t_rem - 5) * 1.5)))
        h_cd, c_cd = _coord_descent(best_h, n_passes=n_passes)
        if c_cd < best_c:
            best_c = c_cd
            best_h = h_cd

    # Phase 7: Final NM + subgrad + coord
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 12:
        h_sg2, c_sg2 = _subgrad_topk(best_h, n_steps=200, k_top=2)
        if c_sg2 < best_c:
            best_c = c_sg2
            best_h = h_sg2
        h_nm3, c_nm3 = _nm_polish(best_h, maxiter=600)
        if c_nm3 < best_c:
            best_c = c_nm3
            best_h = h_nm3

    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 8:
        h_cd2, c_cd2 = _coord_descent(best_h, n_passes=20)
        if c_cd2 < best_c:
            best_c = c_cd2
            best_h = h_cd2

    best_h = _project(best_h)
    final_c = compute_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return (best_h, best_c, n)
