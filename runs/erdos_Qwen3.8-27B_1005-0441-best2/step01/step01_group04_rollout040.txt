If s < 0: f(h) = sum_{i=-s}^{n-1} h[i] * (1 - h[i+s])  [where i+s = i-|s|]

def _subgradient_step(h, lr, top_m=5):
    n = len(h)
    corr = _corr_array(h)
    cmax = corr.max()
    # Softmax weights over top lags
    alpha = 20.0
    shifted = corr - cmax
    weights = np.exp(alpha * shifted) / np.exp(alpha * shifted).sum()
    
    # Weighted gradient
    grad = np.zeros(n)
    # Only compute for top-m lags (for speed)
    top_idx = np.argsort(corr)[-top_m:]
    top_weights = weights[top_idx]
    top_weights = top_weights / top_weights.sum()
    
    n_center = n - 1  # center of correlation array
    for t in range(len(top_idx)):
        s = top_idx[t] - n_center
        grad += top_weights[t] * _grad_lag(h, s)
    
    h_new = h - lr * grad
    h_new = _project_sum(h_new, n)
    return h_new, cmax

    pbest_val = np.full(n_particles, np.inf)

    particles[0] = h_init.copy()

        pbest[i] = particles[i].copy()
        pbest_val[i] = _eval_c5(particles[i])

            gbest_val = pbest_val[i]

    w = 0.8
    c1 = 1.5
    c2 = 1.5
    v_max = 0.1
    
    for it in range(n_iter):

            r1 = rng.random(n)
            r2 = rng.random(n)
            velocities[i] = w * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])

                if val < gbest_val:
                    gbest_val = val
                    gbest = particles[i].copy()
        # Decrease w
        w = max(0.4, w - 0.8 / n_iter)
    
    return gbest, gbest_val

import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project_sum(h, n):
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def _grad_lag(h, s):
    n = len(h)
    m = np.arange(n)
    grad = np.zeros(n)
    if s >= 0:
        mf = (m + s) < n
        grad[mf] += (1.0 - h[m[mf] + s])
        ms = m >= s
        grad[ms] -= h[m[ms] - s]
    else:
        a = -s
        mf = m >= a
        grad[mf] += (1.0 - h[m[mf] - a])
        ms = (m + a) < n
        grad[ms] -= h[m[ms] + a]
    return grad

def _subgrad_optimize(h0, n, max_iter=10000, time_limit=60.0, t0=None):
    if t0 is None:
        t0 = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_val = _eval_c5(h)
    velocity = np.zeros(n)
    momentum = 0.85
    lr = 0.02
    stall = 0
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        corr = _corr_array(h)
        cmax = corr.max()
        
        # Softmax weights
        alpha = 15.0
        w = np.exp(alpha * (corr - cmax))
        wis = w.sum()
        if wis > 0:
            w /= wis
        else:
            w[np.argmax(corr)] = 1.0
        
        # Top-m weighting
        top_m = 5
        idx = np.argsort(corr)[-top_m:]
        w_top = w[idx].sum()
        if w_top > 0:
            w_top_arr = w[idx] / w_top
        else:
            w_top_arr = np.ones(top_m) / top_m
        
        n_center = n - 1
        grad = np.zeros(n)
        for t in range(top_m):
            s = idx[t] - n_center
            grad += w_top_arr[t] * _grad_lag(h, s)
        
        velocity = momentum * velocity - lr * grad
        h_new = _project_sum(h + velocity, n)
        val = _eval_c5(h_new)
        
        if val < best_val - 1e-12:
            best_val = val
            best_h = h_new.copy()
            h = h_new
            stall = 0
            lr = min(lr * 1.02, 0.08)
        else:
            h = h_new  # keep the step
            stall += 1
            lr *= 0.97
            if lr < 0.001:

            g[i] = (obj(h) - base) / eps
            h[i] -= eps
        return g

import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project_sum(h, n):
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _grad_corr(h, lag):
    """Gradient of corr[lag] (where lag is index in full correlation array) w.r.t. h"""
    n = len(h)
    s = lag - (n - 1)  # actual shift
    m = np.arange(n)
    grad = np.zeros(n)
    if s >= 0:
        # corr[n-1+s] = sum_{i=0}^{n-1-s} h[i] * (1-h[i+s])
        mf = m + s < n
        grad[mf] += (1.0 - h[m[mf] + s])
        ms = m >= s
        grad[ms] -= h[m[ms] - s]
    else:
        a = -s
        # corr[n-1-a] = sum_{i=a}^{n-1} h[i] * (1-h[i-a])
        mf = m >= a
        grad[mf] += (1.0 - h[m[mf] - a])
        ms = m + a < n
        grad[ms] -= h[m[ms] + a]
    return grad

def _subgrad_optimize(h0, n, max_iter, t_start, time_limit):

    vel = np.zeros(n)
    mu = 0.8

    stall = 0

    for it in range(max_iter):

        sh = corr - cmax
        w = np.exp(alpha_sm * sh)

        if ws < 1e-300:
            argmax_idx = np.argmax(corr)
            w = np.zeros_like(corr)
            w[argmax_idx] = 1.0
        else:
            w /= ws
        
        # Focus on top-m
        top_idx = np.argsort(corr)[-top_m:]
        wt = w[top_idx]
        wt_sum = wt.sum()
        if wt_sum < 1e-300:
            wt = np.ones(top_m) / top_m
        else:
            wt = wt / wt_sum
        
        grad = np.zeros(n)
        for t in range(top_m):
            grad += wt[t] * _grad_corr(h, top_idx[t])
        
        vel = mu * vel - lr * grad
        h_try = _project_sum(h + vel, n)

    V = np.zeros((n_part, n))
    PB = np.zeros((n_part, n))
