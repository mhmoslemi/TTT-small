```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def overlap_gradient(h, k, n):
    """Gradient of the overlap at shift k with respect to h.
    k is the lag: -(n-1) to (n-1), where 0 is the center.
    The overlap is sum_j h[j] * (1-h[j+k]) for valid j.
    """
    grad = np.zeros(n)
    # Term 1: h[j] * (1-h[j+k]) -> d/dh[i] = (1-h[i+k]) if 0<=i+k<n
    # Term 2: h[j-k] * (1-h[j]) -> d/dh[i] = -h[i-k] if 0<=i-k<n
    # Actually, the overlap at lag k (where k can be negative) is:
    # sum_j h[j] * (1-h[j+k]) for j where both j and j+k are in [0,n-1]
    
    # For positive k: valid j are 0 to n-1-k
    #   grad[i] gets (1-h[i+k]) for i in [0, n-1-k] (from first term)
    #   and -h[i-k] for i in [k, n-1] (from second term, where j=i-k, h[j]*(1-h[j+k])=h[i-k]*(1-h[i]))
    
    # For negative k (let k' = -k > 0): valid j are k' to n-1
    #   grad[i] gets (1-h[i-k']) for i in [k', n-1] (from first term, j=i, needs i-k' >= 0)
    #   and -h[i+k'] for i in [0, n-1-k'] (from second term)
    
    if k >= 0:
        valid_end = n - k
        if valid_end > 0:
            grad[0:valid_end] += (1.0 - h[k:n])
        if k > 0:
            grad[k:n] -= h[0:n-k]
    else:
        k2 = -k
        valid_start = k2
        if valid_start < n:
            grad[valid_start:n] += (1.0 - h[0:n-k2])
        if k2 > 0:
            grad[0:n-k2] -= h[k2:n]
    
    # Project gradient to satisfy sum constraint: subtract mean
    grad -= np.mean(grad)
    return grad

def subgradient_minmax(h, n, max_iters, lr_init=0.01, target_mr=2):
    """Subgradient method for min max_k overlap_k(h)"""
    dx = 2.0 / n
    h_cur = h.copy()
    h_avg = h.copy()  # Polyak averaging
    best_h = h.copy()
    best_c5 = compute_c5(h_cur)
    
    for it in range(max_iters):
        overlaps = compute_all_overlaps(h_cur)
        c5_cur = overlaps.max()
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        
        # Find hot set: top target_mr shifts
        sorted_idx = np.argsort(overlaps)[::-1]
        hot_set = sorted_idx[:target_mr]
        
        # Average gradient over hot set
        grad_avg = np.zeros(n)
        center = n - 1  # index in full correlation corresponding to lag 0
        for idx in hot_set:
            k = idx - center  # lag
            grad_avg += overlap_gradient(h_cur, k, n)
        grad_avg /= target_mr
        
        # Step size (decreasing)
        lr = lr_init / (1.0 + 0.01 * it)
        
        h_new = h_cur - lr * grad_avg
        h_new = project_feasible(h_new, n)
        
        # Polyak averaging
        h_avg = (it * h_avg + h_new) / (it + 1)
        
        h_cur = h_new
        
        if it % 1000 == 0 and it > 0:
            c5_avg = compute_c5(h_avg)
            if c5_avg < best_c5:
                best_c5 = c5_avg
                best_h = h_avg.copy()
    
    # Check averaged solution
    c5_avg = compute_c5(h_avg)
    if c5_avg < best_c5:
        best_c5 = c5_avg
        best_h = h_avg.copy()
    
    return best_h, best_c5

        # Second term: -h[i+k2] for i in [0, n-1-k2]
        if k2 > 0:
            grad[0:n-k2] -= h[k2:n]
    return grad
```

Let me verify with a small example. n=3, h=[h0,h1,h2], k=1:
f_1 = h0*(1-h1) + h1*(1-h2) (sum over j=0,1 since j+k<3 means j<2)

∂f_1/∂h0 = (1-h1) [from first term, j=0: h0*(1-h1)]
∂f_1/∂h1 = (1-h2) - h0 [from first term j=1: h1*(1-h2), and second term j=0: -h0*(1-h1)]
∂f_1/∂h2 = -h1 [from second term j=1: -h1*(1-h2)]

With my formula for k=1, n=3:
- First term: valid = n-k = 2, grad[0:2] += (1-h[1:3]) = (1-h1, 1-h2)
  So grad[0] += 1-h1, grad[1] += 1-h2 ✓
- Second term: grad[1:3] -= h[0:2] = (h0, h1)
  So grad[1] -= h0, grad[2] -= h1 ✓

Result: grad = [1-h1, 1-h2-h0, -h1] ✓

Now for k=-1, n=3:
f_{-1} = h1*(1-h0) + h2*(1-h1) (sum over j=1,2 since j-1>=0 means j>=1)

∂f_{-1}/∂h0 = -h1 [from j=1: h1*(1-h0), derivative w.r.t. h0 is -h1]
∂f_{-1}/∂h1 = (1-h0) - h2 [from j=1: h1*(1-h0) gives (1-h0), and j=2: h2*(1-h1) gives -h2]
∂f_{-1}/∂h2 = (1-h1) [from j=2: h2*(1-h1)]

With my formula for k=-1 (k2=1), n=3:
- First term: grad[1:3] += (1-h[0:2]) = (1-h0, 1-h1)
  So grad[1] += 1-h0, grad[2] += 1-h1 ✓
- Second term: grad[0:2] -= h[1:3] = (h1, h2)
  So grad[0] -= h1, grad[1] -= h2 ✓

Result: grad = [-h1, 1-h0-h2, 1-h1] ✓

The gradient formula is correct.

Now, for the subgradient method, I need to project the gradient to the constraint set. The constraint is sum(h) = n/2. The projection of a gradient onto the tangent space of this constraint is: grad_proj = grad - (sum(grad)/n) * ones, i.e., subtract the mean.

Then the step is: h_new = h - lr * grad_proj, followed by projection to [0,1] and fixing the sum.

5. Trying different resolutions

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def overlap_grad(h, k, n):
    grad = np.zeros(n)
    if k >= 0:
        if k < n:
            v = n - k
            if v > 0:
                grad[0:v] += (1.0 - h[k:n])
        if k > 0:
            grad[k:n] -= h[0:n-k]
    else:
        k2 = -k
        if k2 < n:
            v = n - k2
            if v > 0:
                grad[k2:n] += (1.0 - h[0:v])
        if k2 > 0:
            grad[0:n-k2] -= h[k2:n]
    return grad

def subgradient_minmax(h, n, max_iters, lr_init=0.005, hot=3, time_limit=None, t0=None):
    h_cur = h.copy()
    h_avg = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h_cur)
    dx = 2.0 / n
    center = n - 1
    for it in range(max_iters):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        overlaps = np.correlate(h_cur, 1.0 - h_cur, mode="full") * dx
        c5_cur = overlaps.max()
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        sidx = np.argsort(overlaps)[::-1]
        hot_idx = sidx[:hot]
        g = np.zeros(n)
        for idx in hot_idx:
            k = idx - center
            g += overlap_grad(h_cur, k, n)
        g /= hot
        g -= np.mean(g)
        lr = lr_init / (1.0 + 0.02 * it)
        h_new = project_feasible(h_cur - lr * g, n)
        h_avg = (it * h_avg + h_new) / (it + 1)
        h_cur = h_new
        if it % 500 == 499 and it > 0:
            c5a = compute_c5(h_avg)
            if c5a < best_c5:
                best_c5 = c5a
                best_h = h_avg.copy()
    c5a = compute_c5(h_avg)
    if c5a < best_c5:
        best_c5 = c5a
        best_h = h_avg.copy()
    return best_h, best_c5

def slsqp_refine(h, n, t0, budget_s, frac=0.15):
    if time.time() - t0 > budget_s * (1.0 - frac):
        return h, compute_c5(h)
    dx = 2.0 / n
    def soft_obj(x, n=n, dx=dx, beta=5000.0):
        c = np.correlate(x, 1.0 - x, mode="full")
        m = c.max()
        sh = beta * (c - m)
        sh -= sh.max()
        return (m + np.log(np.sum(np.exp(sh))) / beta) * dx + 1e6 * (np.sum(x) - n/2.0)**2
    def con_fun(x, n=n):
        return np.sum(x) - n / 2.0
    def con_jac(x, n=n):
        return np.ones(n)
    try:
        res = minimize(soft_obj, h, method='SLSQP',
                       bounds=[(0.0,1.0)]*n,
                       constraints=[{'type':'eq','fun':con_fun,'jac':con_jac}],
                       options={'maxiter':300,'ftol':1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            return h_new, c5_new
    except Exception:
        pass
    return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf
    best_n = 320

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = initial_h_values.copy().astype(float)
        n0 = len(h0)
    else:
        n0 = 320
        h0 = np.full(n0, 0.5)

    h0 = project_feasible(h0, n0)
    c5_0 = compute_c5(h0)
    best_h, best_c5, best_n = h0.copy(), c5_0, n0

    # Phase 1: Subgradient at base resolution
    h_s, c5_s = subgradient_minmax(h0, n0, max_iters=8000, lr_init=0.006, hot=3, time_limit=budget_s*0.35, t0=t0)
    if c5_s < best_c5:
        best_h, best_c5 = h_s, c5_s

    # Phase 2: SLSQP polish
    h_p, c5_p = slsqp_refine(best_h, n0, t0, budget_s, 0.20)
    if c5_p < best_c5:
        best_h, best_c5 = h_p, c5_p

    # Phase 3: More subgradient with varied params
    for lr_t, hot_t in [(0.003, 5), (0.01, 2), (0.004, 3)]:
        if time.time() - t0 > budget_s * 0.60:
            break
        h_s2, c5_s2 = subgradient_minmax(best_h, n0, max_iters=6000, lr_init=lr_t, hot=hot_t, time_limit=budget_s*0.60, t0=t0)
        if c5_s2 < best_c5:
            best_h, best_c5 = h_s2, c5_s2
        h_p2, c5_p2 = slsqp_refine(best_h, n0, t0, budget_s, 0.10)
        if c5_p2 < best_c5:
            best_h, best_c5 = h_p2, c5_p2

    # Phase 4: Restart from perturbations at same resolution
    for r in range(8):
        if time.time() - t0 > budget_s * 0.78:
            break
        scale = 0.02 + 0.01 * r
        h_pert = project_feasible(best_h + rng.randn(n0) * scale, n0)
        h_r, c5_r = subgradient_minmax(h_pert, n0, max_iters=4000, lr_init=0.005, hot=3, time_limit=budget_s*0.78, t0=t0)
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r
            h_p3, c5_p3 = slsqp_refine(best_h, n0, t0, budget_s, 0.08)
            if c5_p3 < best_c5:
                best_h, best_c5 = h_p3, c5_p3

    # Phase 5: Try upsampled resolution
    if time.time() - t0 < budget_s * 0.45 and n0 < 400:
        n1 = n0 * 2
        x_old = np.linspace(0, 2, n0, endpoint=False)
        x_new = np.linspace(0, 2, n1, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_feasible(h_up, n1)
        h_u, c5_u = subgradient_minmax(h_up, n1, max_iters=5000, lr_init=0.004, hot=3, time_limit=budget_s*0.70, t0=t0)
        if c5_u < best_c5:
            best_h, best_c5, best_n = h_u, c5_u, n1
        h_p4, c5_p4 = slsqp_refine(best_h, best_n, t0, budget_s, 0.12)
        if c5_p4 < best_c5:
            best_h, best_c5, best_n = h_p4, c5_p4, best_n

    # Phase 6: Final subgradient + SLSQP on best
    if time.time() - t0 < budget_s * 0.88:
        h_f, c5_f = subgradient_minmax(best_h, best_n, max_iters=3000, lr_init=0.003, hot=2, time_limit=budget_s*0.90, t0=t0)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f
    h_final, c5_final = slsqp_refine(best_h, best_n, t0, budget_s, 0.08)
    if c5_final < best_c5:
        best_h, best_c5 = h_final, c5_final
