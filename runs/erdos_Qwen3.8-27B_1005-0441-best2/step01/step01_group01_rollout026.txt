import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, n):
    """Project h onto {0<=q<=1, sum(q)=n/2} using water-filling."""
    target = n * 0.5
    # Binary search for lambda
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        q = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(q)
        if s > target:
            lo = mid
        else:
            hi = mid
    q = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune
    resid = target - np.sum(q)
    if abs(resid) > 1e-12:
        mask = (q > 1e-10) & (q < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            q[mask] += resid / cnt
            np.clip(q, 0.0, 1.0, out=q)
    return q

    s = alpha * corr * dx
    s -= np.max(s)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = alpha * corr * dx
    s -= np.max(s)
    exp_s = np.exp(s)
    p = exp_s / np.sum(exp_s)

        if w < 1e-12:

        else:

            part1[t:] = 1.0 - h[:n-t]
            part2[:n-t] = h[t:]

    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h0.copy()
    best_f = smoothed_obj(h0, alpha)
    
    for it in range(n_iters):
        # Nesterov step
        y = project_feasible(h + momentum * v, n)
        g = smoothed_grad(y, alpha)
        new_h = project_feasible(y - dt * g, n)
        f_new = smoothed_obj(new_h, alpha)
        
        # Backtracking if needed
        if f_new > best_f + 1e-6 * (it + 1):
            dt *= 0.5
            new_h = project_feasible(y - dt * g, n)
        
        v = new_h - h
        h = new_h
        
        if f_new < best_f:
            best_f = f_new
            best_h = h.copy()
        
        # Adaptive step size
        if it % 50 == 0:
            # Check if step is too large
            grad_norm = np.max(np.abs(g))
            if grad_norm > 0:
                dt = min(dt * 1.1, 1.0 / grad_norm)
    
    return best_h, best_f

        if c5_try < c5_best:
            c5_best = c5_try
            h_best = h_try

    # Phase 5: Final SLSQP polish
    if time.time() - t0 < budget_s - 40:
        _SUM_TARGET = n * 0.5
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
        res = minimize(slsqp_obj, h_best.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 600, 'ftol': 1e-15})
        h_sl = project_feasible(res.x, n)
        c5_sl = compute_c5(h_sl)
        if c5_sl < c5_best:
            c5_best = c5_sl
            h_best = h_sl

    # Final feasibility
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)
    return h_best, c5_best, n
