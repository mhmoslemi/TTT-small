import time, math

def project_feasible(h, target_sum):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def compute_maxcorr_c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = float(corr.max())
    return mc, mc * 2.0 / len(h)

def grad_c5(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    exps = np.exp(tau * (corr - corr.max()))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            m = n - lag
            grad[:m] += wk
            grad[:m] -= wk * h[lag:]
            grad[lag:] -= wk * h[:m]
        else:
            L = -lag
            m = n - L
            grad[L:] += wk
            grad[L:] -= wk * h[:m]
            grad[:m] -= wk * h[L:]
    return float(corr.max() * dx), grad * dx

def adam_opt(h_init, time_budget, seed, tau0=30.0, lr=0.05, tau_growth=1.005):
    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    tau = tau0
    t0 = time.time()
    deadline = t0 + time_budget
    t = 0
    while time.time() < deadline - 0.1:
        t += 1
        c5, grad = grad_c5(h, dx, tau)
        m_v = beta1 * m_v + (1.0 - beta1) * grad
        v_v = beta2 * v_v + (1.0 - beta2) * grad * grad
        m_hat = m_v / (1.0 - beta1 ** t)
        v_hat = v_v / (1.0 - beta2 ** t)
        h_new = project_feasible(h - lr * m_hat / (np.sqrt(v_hat) + eps), target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            tau = max(tau * 0.97, tau0)
        h = h_new
        tau = min(tau * tau_growth, 500.0)
    return best_h, best_c5

def binary_sa(h_frac, time_budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h_frac)
    target = n // 2
    h = (h_frac >= 0.5).astype(float)
    while h.sum() > target:
        h[rng.randint(n)] = 0.0
    while h.sum() < target:
        h[rng.randint(n)] = 1.0
    _, best_c5 = compute_maxcorr_c5(h)
    best_h = h.copy()
    T0, Tf = 0.8, 0.005
    t0 = time.time()
    deadline = t0 + time_budget
    while time.time() < deadline - 0.1:
        elapsed = time.time() - t0
        frac = min(elapsed / time_budget, 1.0)
        T = Tf * (T0 / Tf) ** (1.0 - frac)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        _, c5_new = compute_maxcorr_c5(h)
        delta = c5_new - best_c5
        if delta <= 0 or rng.rand() < math.exp(-delta / (T + 1e-10)):
            if delta < 0:
                best_h = h.copy()
                best_c5 = c5_new
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5
