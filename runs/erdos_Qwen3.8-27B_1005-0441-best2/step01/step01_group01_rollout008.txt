    shifted = np.clip(alpha * (corr - m), -400.0, 400.0)

def sum_constraint(x):
    return np.sum(x) - len(x) * 0.5

def make_smooth_blocks(n, n_blocks, transition):
    h = np.zeros(n)
    block = n // (2 * n_blocks)
    for b in range(n_blocks):
        start = b * (n // n_blocks)
        end = start + block
        if end > n:
            end = n
        for i in range(max(0, start - transition), min(n, end + transition)):
            dist_left = i - (start - transition)
            dist_right = (end + transition) - 1 - i
            v_left = np.tanh(dist_left / max(transition * 0.3, 0.5))
            v_right = np.tanh(dist_right / max(transition * 0.3, 0.5))
            h[i] = min(1.0, 0.5 * (v_left + v_right))
    return project_feasible(h, n)

def make_harmonic(n, freq, n_peaks=2):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.zeros(n)
    for k in range(n_peaks):
        center = 2.0 * (k + 0.5) / n_peaks
        h += 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * (x - center))
    h = h / np.max(h) if np.max(h) > 0 else h
    h = np.clip(h, 0, 1)
    h = project_feasible(h, n)
    return h

def binary_local_search(h, n, rng, max_rounds=300):
    h = (h > 0.5).astype(float)
    best_c5 = compute_c5_local(h)
    for r in range(max_rounds):
        improved = False
        for _ in range(80):
            i, j = rng.choice(n, 2, replace=False)
            if h[i] == h[j]:
                continue
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
            c5_new = compute_c5_local(h)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                improved = True
            else:
                h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        if not improved and r > 50:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h_best = make_smooth_blocks(160, 2, 8)
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5_local(h_best)

    # Phase 1: diverse structured starting points
    starts = [h_best.copy()]
    for nb in [2, 3]:
        for tr in [4, 8, 14, 20]:
            starts.append(make_smooth_blocks(n, nb, tr))
    starts.append(make_harmonic(n, 2, 2))
    starts.append(make_harmonic(n, 3, 2))
    starts.append(make_harmonic(n, 2, 3))
    # Binary patterns
    for _ in range(4):
        hb = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        hb[idx] = 1.0
        starts.append(hb)

    for s in starts:
        s = project_feasible(np.asarray(s, dtype=float), n)
        c5 = compute_c5_local(s)
        if c5 < c5_best:
            c5_best = c5
            h_best = s.copy()

    # Phase 2: binary local search from best binary-like solution
    h_bin = (h_best > 0.5).astype(float)
    h_bin = project_feasible(h_bin, n)
    if time.time() - t0 < budget_s * 0.25:
        h_ls, c5_ls = binary_local_search(h_bin, n, rng, max_rounds=400)
        if c5_ls < c5_best:
            c5_best = c5_ls
            h_best = h_ls.copy()

    # Phase 3: SLSQP continuous optimization with increasing alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]

    for alpha in [50.0, 150.0, 400.0, 1000.0, 2000.0]:
        if time.time() - t0 > budget_s * 0.45:
            break
        res = minimize(soft_max_obj, h_best.copy(), args=(alpha,),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': 2000, 'ftol': 1e-16})
        hc = project_feasible(res.x, n)
        c5c = compute_c5_local(hc)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()

    # Phase 4: Nelder-Mead local refinement on sub-vectors
    if time.time() - t0 < budget_s * 0.60:
        # Perturb and use Nelder-Mead (derivative-free)
        for nm_iter in range(5):
            if time.time() - t0 > budget_s * 0.60:
                break
            phase = 0.015 + 0.01 * rng.random()
            hp = project_feasible(h_best + phase * rng.standard_normal(n), n)
            res = minimize(soft_max_obj, hp, args=(600.0,),
                          method='Nelder-Mead',
                          options={'maxiter': 500, 'xatol': 1e-9, 'fatol': 1e-12})
            hc = project_feasible(res.x, n)
            c5c = compute_c5_local(hc)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc.copy()

    # Phase 5: structured restarts with varying geometry
    while time.time() - t0 < budget_s - 120:
        strategy = rng.integers(0, 3)
        if strategy == 0:
            sigma = 0.02 + 0.04 * rng.random()
            hp = project_feasible(h_best + sigma * rng.standard_normal(n), n)
        elif strategy == 1:
            # Re-generate smooth blocks with random params and blend
            nb = rng.integers(2, 4)
            tr = int(3 + 15 * rng.random())
            alt = make_smooth_blocks(n, nb, tr)
            mix = 0.3 + 0.4 * rng.random()
            hp = project_feasible(mix * h_best + (1 - mix) * alt, n)
        else:
            # Perturb only a random subset
            subset_size = max(10, n // 5)
            idx = rng.choice(n, subset_size, replace=False)
            hp = h_best.copy()
            hp[idx] += 0.1 * rng.standard_normal(subset_size)
            hp = project_feasible(hp, n)

        alpha = 300.0 + 700.0 * rng.random()
        res = minimize(soft_max_obj, hp, args=(alpha,),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': 1200, 'ftol': 1e-15})
        hc = project_feasible(res.x, n)
        c5c = compute_c5_local(hc)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()

    # Phase 6: Final high-precision polish
    if time.time() - t0 < budget_s - 30:
        for alpha in [3000.0, 5000.0]:
            if time.time() - t0 > budget_s - 10:
                break
            res = minimize(soft_max_obj, h_best.copy(), args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 3000, 'ftol': 1e-18})
            hc = project_feasible(res.x, n)
            c5c = compute_c5_local(hc)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc.copy()

    h_best = project_feasible(h_best, n)
    c5_best = compute_c5_local(h_best)
    return h_best, c5_best, n
