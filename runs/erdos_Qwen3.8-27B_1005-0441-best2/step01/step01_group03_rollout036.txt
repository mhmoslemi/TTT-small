import numpy as np
import time

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-14:
        mask = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        cnt = float(np.sum(mask))
        if cnt > 0:
            hp[mask] = np.clip(hp[mask] + diff / cnt, 0.0, 1.0)
    return hp

def get_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def get_c5(h, n):
    return float(np.max(get_overlaps(h, n)))

def optimize_h(h, n, deadline, rng, top_k=5, base_lr=0.35, sa_temp=0.006):
    dx = 2.0 / n
    best_h = h.copy()

    lr = base_lr
    T = sa_temp
    stalled = 0

    while time.time() < deadline:
        # Main loop with multi-lag gradient + SA
        batch_size = 200
        for _ in range(batch_size):
            if time.time() >= deadline:
                break
            ov = get_overlaps(h, n)
            top_idx = np.argsort(ov)[-top_k:]
            lags = np.array([int(kk) - (n - 1) for kk in top_idx])
            w = ov[top_idx].copy()
            wsum = w.sum()
            if wsum < 1e-15:
                w = np.ones(top_k) / top_k
            else:
                w = w / wsum

            g = np.zeros(n)
            for wi, lag in zip(w, lags):
                lag = int(lag)
                for j in range(n):
                    a = j - lag
                    if 0 <= a < n:
                        g[j] += wi * (1.0 - h[a])
                    b = j + lag
                    if 0 <= b < n:
                        g[j] -= wi * h[b]

            h_new = h - lr * g * dx

            c_new = get_c5(h_new, n)

            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
                best_h = h_new.copy()
                h = h_new
                stalled = 0
            else:
                accept_prob = np.exp(-(c_new - best_c5) / T) if T > 1e-12 else 0.0
                if rng.random() < accept_prob:
                    h = h_new
                stalled += 1

        # Cooling and adaptation
        T *= 0.96

            lr *= 0.65
            stalled = 0
            if lr < 0.008:
                h = best_h.copy()

                h = project_feasible(h + noise, n)
                lr = base_lr
                T = sa_temp
        elif stalled > 25:
            # Swap-based local search
            for _ in range(40):
                if time.time() >= deadline:

                ones = np.where(h > 0.9)[0]
