def project_binary(h, n):
    """Project to binary with exactly n/2 ones."""
    h = h.copy()
    # Sort by value, keep top n/2 as 1s
    thresh = np.sort(h)[::-1][n//2 - 1] if n % 2 == 0 else np.sort(h)[::-1][(n-1)//2]
    # For even n, need exactly n/2 ones
    if n % 2 == 0:
        num_ones = n // 2
    else:
        num_ones = n // 2  # and one 0.5
    
    if n % 2 == 0:
        h = (h >= thresh).astype(float)
        if np.sum(h) != num_ones:
            # Fix: set to threshold exactly
            h[h >= thresh] = 0
            h[h < thresh] = 1
            # Still might not be right, use partial sort
            idx = np.argsort(h)[::-1]
            h = np.zeros(n)
            h[idx[:num_ones]] = 1.0
    else:

        idx = np.argsort(h)[::-1]
        h = np.zeros(n)
        h[idx[:num_ones]] = 1.0

def gradient_refine(h, n, deadline, rng):
    """Gradient-based refinement for continuous h."""
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    lr = 0.3
    
    while time.time() < deadline:
        ov = np.correlate(h, 1.0 - h, mode='full') * dx
        k_max = int(np.argmax(ov))
        L = k_max - (n - 1)
        
        # Gradient of c[L] w.r.t. h[i]:
        # c[L] = sum_i h[i]*(1-h[i+L])
        # dc/dh[i] = (1-h[i+L]) - h[i-L]  (where indices valid)
        g = np.zeros(n)
        for i in range(n):
            val = 0.0
            if 0 <= i + L < n:
                val += (1.0 - h[i + L])
            if 0 <= i - L < n:
                val -= h[i - L]
            g[i] = val
        
        h_new = h - lr * g * dx
        # Project to [0,1] with sum = n/2
        h_new = np.clip(h_new, 0, 1)
        # Adjust sum
        diff = target - np.sum(h_new)
        if abs(diff) > 1e-10:
            # Distribute correction
            mask = (h_new > 1e-6) & (h_new < 1.0 - 1e-6)
            if np.sum(mask) > 0:
                h_new[mask] += diff / np.sum(mask)
                h_new = np.clip(h_new, 0, 1)
            # If still off, adjust 0/1 values
            remaining = target - np.sum(h_new)
            if remaining > 0:
                zeros_mask = (h_new < 0.5)
                if np.sum(zeros_mask) > 0:
                    idx = np.where(zeros_mask)[0][:int(abs(remaining))+1]
                    h_new[idx] = np.clip(h_new[idx] + remaining / len(idx), 0, 1)
            elif remaining < 0:
                ones_mask = (h_new > 0.5)
                if np.sum(ones_mask) > 0:
                    idx = np.where(ones_mask)[0][:int(abs(remaining))+1]
                    h_new[idx] = np.clip(h_new[idx] + remaining / len(idx), 0, 1)
        
        c_new = compute_c5(h_new, n)
        h = h_new
        if c_new < best_c5 - 1e-12:
            best_c5 = c_new
            best_h = h.copy()
    
    return best_h, best_c5

Actually wait, the inner loop is Python-level, which is slow. Let me vectorize it.

For a given lag L:
Δc[L] for swap (i,j) = h[i+L] - h[j+L] + h[i-L] - h[j-L]
= (h[i+L] + h[i-L]) - (h[j+L] + h[j-L])

So for each i in ones, compute s_i = h[i+L] + h[i-L] (with boundary handling)
For each j in zeros, compute t_j = h[j+L] + h[j-L]
We want s_i - t_j < 0, i.e., s_i < t_j.
Best swap: minimize s_i - t_j = min_i s_i - max_j t_j (independent!)

So the best swap for lag L is: i* = argmin_{i in ones} (h[i+L]+h[i-L]), j* = argmax_{j in zeros} (h[j+L]+h[j-L]).

This is O(n) per lag! Much better.

Then we just need to verify that this swap actually reduces the overall max (not just lag L). So we do at most 5 candidate swaps (one per top lag) and evaluate each fully.

def project_feasible_h(h, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0

        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    # Fix residual
    diff = target - np.sum(hp)
    if abs(diff) > 1e-12:
        frac = (hp > 1e-8) & (hp < 1.0 - 1e-8)
        cnt = np.sum(frac)
        if cnt > 0:
            hp[frac] = np.clip(hp[frac] + diff / cnt, 0.0, 1.0)
    return hp

def to_binary(h, n):
    """Round h to binary (or binary + one 0.5 for odd n)."""
    target = n // 2 if n % 2 == 0 else (n - 1) // 2
    h_sorted_idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[h_sorted_idx[:target]] = 1.0
    if n % 2 == 1:
        out[h_sorted_idx[target]] = 0.5
    return out

def binary_local_search(h, n, deadline, rng, max_stall=40):

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def project_feasible_h(h, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-12:
        frac = (hp > 1e-8) & (hp < 1.0 - 1e-8)
        cnt = np.sum(frac)
        if cnt > 0:
            hp[frac] = np.clip(hp[frac] + diff / cnt, 0.0, 1.0)
    return hp

def to_binary(h, n):
    target = n // 2 if n % 2 == 0 else (n - 1) // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    if n % 2 == 1:
        out[idx[target]] = 0.5
    return out

def binary_local_search(h, n, deadline, rng, max_stall=35):
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    dx = 2.0 / n
    stall = 0
    while time.time() < deadline:
        ones = np.where(h > 0.7)[0]
        zeros = np.where(h < 0.3)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        ov = np.correlate(h, 1.0 - h, mode='full') * dx
        c5 = float(np.max(ov))
        topk = np.argsort(ov)[-5:]
        best_candidate = None
        best_new_c5 = c5
        for li in topk:
            L = int(li) - (n - 1)
            if L == 0:
                continue
            s = np.zeros(len(ones))
            t = np.zeros(len(zeros))
            for idx in range(len(ones)):
                i = ones[idx]
                if i + L < n:
                    s[idx] += h[i + L]
                if i - L >= 0:
                    s[idx] += h[i - L]
            for idx in range(len(zeros)):
                j = zeros[idx]
                if j + L < n:
                    t[idx] += h[j + L]
                if j - L >= 0:
                    t[idx] += h[j - L]
            if s.min() >= t.max() - 1e-12:
                continue
            i_star = ones[np.argmin(s)]
            j_star = zeros[np.argmax(t)]
            h[i_star] = 0.0
            h[j_star] = 1.0
            nc5 = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
            h[i_star] = 1.0
            h[j_star] = 0.0
            if nc5 < best_new_c5 - 1e-14:
                best_new_c5 = nc5
                best_candidate = (i_star, j_star)
        if best_candidate is not None:
            i, j = best_candidate
            h[i] = 0.0
            h[j] = 1.0
            if best_new_c5 < best_c5:
                best_c5 = best_new_c5
                best_h = h.copy()
            if best_new_c5 > c5:
                stall += 1

                    oz = zeros[rng.randint(len(zeros))]

                    h[oz] = 1.0
            if compute_c5(h, n) > best_c5 + 0.04:

            stall = 0

def gradient_descent(h, n, deadline, rng):
    h = h.copy().astype(float)
    best_h = h.copy()

    dx = 2.0 / n

        if wts.sum() < 1e-15:
            wts = np.ones(3) / 3.0
        else:
            wts /= wts.sum()
        g = np.zeros(n)

            L = int(li) - (n - 1)

                continue
            for i in range(n):
                if i + L < n:
                    g[i] += w * (1.0 - h[i + L])
                if i - L >= 0:
                    g[i] -= w * h[i - L]
        h_new = project_feasible_h(h - lr * g * dx, n)
        nc5 = compute_c5(h_new, n)

    h[:n // 2] = 1.0

        h[n // 2] = 0.5
    return h

    if n % 2 == 1:
        h[-1] = 0.5 if np.sum(h) < n / 2.0 else 0.0
    return to_binary(h, n)
