    for _ in range(2000):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            i = int(np.argmax(1.0 - h))
            h[i] = min(1.0, h[i] + d)
        else:
            i = int(np.argmin(h))
            h[i] = max(0.0, h[i] + d)
    return h

def _grad_from_weights(h, w):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for d in range(1, n):
        wm = w[n - 1 + d]
        if wm > 1e-14:
            grad[d:] += wm * dx * (1.0 - h[:n - d])
            grad[:n - d] -= wm * dx * h[d:]
        wn = w[n - 1 - d]
        if wn > 1e-14:
            grad[:n - d] += wn * dx * (1.0 - h[d:])
            grad[d:] -= wn * dx * h[:n - d]
    w0 = w[n - 1]
    if w0 > 1e-14:
        grad += w0 * dx * (1.0 - 2.0 * h)
    return grad

def _softmax_obj(h, alpha):
    corr = _overlaps(h)
    m = corr.max()
    s = np.clip(alpha * (corr - m), -50, 50)
    e = np.exp(s)
    return m + np.log(e.sum()) / alpha

def _softmax_weights(h, alpha):
    corr = _overlaps(h)
    m = corr.max()
    s = np.clip(alpha * (corr - m), -50, 50)
    e = np.exp(s)
    return e / e.sum()

def _slsqp_opt(h0, alpha, maxiter=500):
    n = len(h0)
    def fun(x):
        return _softmax_obj(x, alpha)
    def jac(x):
        w = _softmax_weights(x, alpha)
        return _grad_from_weights(x, w)
    def eq(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        r = minimize(fun, h0, jac=jac, method='SLSQP',
                     constraints=[{'type': 'eq', 'fun': eq}],
                     bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14})
        return r.x
    except Exception:
        return h0.copy()

def _subgrad_nesterov(h0, n_iter=3000, step0=0.04, top_k=5):
    n = len(h0)
    h = h0.copy()
    vel = np.zeros(n)
    beta = 0.85
    best_h = h0.copy()
    best_c = _eval_c5(h0)
    for it in range(n_iter):
        ms = step0 / (1.0 + 0.005 * it)
        corr = _overlaps(h)
        top_idx = np.argsort(corr)[-top_k:]
        w = np.zeros(2 * n - 1)
        w[top_idx] = 1.0 / len(top_idx)
        g = _grad_from_weights(h, w)
        vel = beta * vel - ms * g
        h_new = _fix_sum(h + vel, n)
        c = _eval_c5(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    return best_h

def _binary_c5(b):
    n = len(b)
    dx = 2.0 / n
    ones = (b == 1).astype(float)
    zeros = (b == 0).astype(float)
    corr = np.correlate(ones, zeros, mode='full')
    return float(corr.max() * dx)

def _binary_local_search(b, n_iter=20000, seed=0):
    rng = np.random.default_rng(seed)
    n = len(b)
    b = b.astype(int).copy()
    bc = _binary_c5(b)
    best_b = b.copy()
    best_c = bc
    ones_idx = np.where(b == 1)[0]
    zeros_idx = np.where(b == 0)[0]
    T = 0.01
    for it in range(n_iter):
        T = 0.01 * (0.98 ** (it / 500.0))
        i = ones_idx[rng.integers(len(ones_idx))]
        j = zeros_idx[rng.integers(len(zeros_idx))]
        b[i], b[j] = b[j], b[i]
        nc = _binary_c5(b)
        if nc < bc or rng.random() < np.exp(-(nc - bc) / max(T, 1e-10)):
            bc = nc
            if bc < best_c:
                best_c = bc
                best_b = b.copy()
        else:
            b[i], b[j] = b[j], b[i]
    return best_b

def _smooth_binary(b, alpha=10.0):
    n = len(b)
    h = 0.5 + 0.5 * np.tanh(alpha * (b - 0.5))
    return _fix_sum(h, n)

def _random_h(n, rng):
    h = rng.random(n)
    return _fix_sum(h, n)

def _perturb(h, rng, scale):
    return _fix_sum(h + rng.normal(0, scale, len(h)), len(h))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    best_c5 = np.inf
    best_h = None

    def time_left():
        return budget_s - (time.time() - t0)

    # Start from initial_h_values if available
    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if h_init is not None and len(h_init) > 0:
        h_init = _fix_sum(h_init, len(h_init))
        c5 = _eval_c5(h_init)
        best_c5 = c5
        best_h = h_init.copy()

        # SLSQP with increasing alpha
        for alpha in [30, 80, 200, 500]:
            if time_left() < 80:
                break
            h_opt = _slsqp_opt(best_h, alpha, 500)
            h_opt = _fix_sum(h_opt, len(h_opt))
            c5 = _eval_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

        # Subgradient Nesterov refinement
        if time_left() > 60:
            h_sg = _subgrad_nesterov(best_h, 4000, 0.03, 5)
            c5 = _eval_c5(h_sg)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_sg.copy()

        # SLSQP again after subgradient
        if time_left() > 60:
            h_opt = _slsqp_opt(best_h, 300, 400)
            h_opt = _fix_sum(h_opt, len(h_opt))
            c5 = _eval_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

    # Try different n values with various starting points
    n_list = [47, 53, 64, 80, 100]
    for n in n_list:
        if time_left() < 60:
            break
        starts = [_random_h(n, rng)]
        x = np.linspace(0, 1, n, endpoint=False)
        for freq in [1, 2, 3]:
            starts.append(_fix_sum(0.5 + 0.4 * np.sin(2 * np.pi * freq * x), n))
        # Block pattern
        blk = np.tile([1, 1, 0], n // 3 + 1)[:n]
        starts.append(_fix_sum(blk.astype(float), n))
        # Thresholded version of best
        if best_h is not None:
            th = (best_h > 0.5).astype(float)
            starts.append(_fix_sum(np.resize(th, n), n))

        for h0 in starts[:3]:
            if time_left() < 40:
                break
            for alpha in [50, 200]:
                if time_left() < 30:
                    break
                h_opt = _slsqp_opt(h0, alpha, 300)
                h_opt = _fix_sum(h_opt, n)
                c5 = _eval_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()

    # Binary local search on best solution
    if best_h is not None and time_left() > 40:
        n = len(best_h)
        b_init = (best_h > 0.5).astype(int)
        b_init = _fix_sum(b_init.astype(float), n)
        b_init = (b_init > 0.5).astype(int)
        # Ensure exactly n//2 ones
        ones_count = b_init.sum()
        if ones_count > n // 2:
            idx = np.where(b_init == 1)[0]
            b_init[idx[-(ones_count - n//2):]] = 0
        elif ones_count < n // 2:
            idx = np.where(b_init == 0)[0]
            b_init[idx[:n//2 - ones_count]] = 1
        b_opt = _binary_local_search(b_init, 15000, seed)
        h_sm = _smooth_binary(b_opt, 8.0)
        h_sm = _slsqp_opt(h_sm, 100, 200)
        h_sm = _fix_sum(h_sm, n)
        c5 = _eval_c5(h_sm)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_sm.copy()
        # Also evaluate the binary directly
        cb = _binary_c5(b_opt)
        if cb < best_c5:
            best_c5 = cb
            best_h = b_opt.astype(float)
            best_h = _fix_sum(best_h, len(best_h))

    # Final perturbation rounds on best
    if best_h is not None:
        for _ in range(20):
            if time_left() < 30:
                break
            h_p = _perturb(best_h, rng, 0.02)
            h_opt = _slsqp_opt(h_p, 150, 250)
            h_opt = _fix_sum(h_opt, len(h_opt))
            c5 = _eval_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

    # Final SLSQP polish
    if best_h is not None and time_left() > 20:
        h_fin = _slsqp_opt(best_h, 400, 600)
        h_fin = _fix_sum(h_fin, len(h_fin))
        c5 = _eval_c5(h_fin)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_fin

    # Fallback
    if best_h is None:
        best_h = np.full(48, 0.5)
        best_c5 = _eval_c5(best_h)

    best_h = _fix_sum(best_h, len(best_h))
    best_c5 = _eval_c5(best_h)
    return best_h, best_c5, len(best_h)
