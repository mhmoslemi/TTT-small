import numpy as np
import time, math

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def compute_grad_softmax(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    shifted = np.clip(tau * (corr - mc), -500, 500)
    exps = np.exp(shifted)
    weights = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            L = -lag
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
    return grad * dx

def adam_project(h, grad, m, v, t, lr, b1, b2, eps):
    m = b1 * m + (1 - b1) * grad
    v = b2 * v + (1 - b2) * grad ** 2
    m_hat = m / (1 - b1 ** t)
    v_hat = v / (1 - b2 ** t)
    n = len(h)
    target_sum = n / 2.0
    # Adam step
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    h_new = h - step
    h_new = project_feasible(h_new, target_sum)
    return h_new, m, v

def binary_anneal(h_frac, time_budget, seed):

    n = len(h_frac)

    h = (h_frac >= 0.5).astype(float)

import numpy as np
import time, math

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def compute_grad_softmax(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    shifted = np.clip(tau * (corr - mc), -600, 600)
    exps = np.exp(shifted)
    s = exps.sum()
    if s < 1e-300:
        weights = np.zeros(len(corr))
        weights[mc == corr] = 1.0 / len(corr)
    else:
        weights = exps / s
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
        else:
            L = -lag
            if L < n:
                grad[:n - L] += w * (1.0 - h[L:])
                grad[L:] -= w * h[:n - L]
    return grad * dx

def adam_step(h, g, m, v, t, lr, b1, b2, eps, target_sum):
    m = b1 * m + (1.0 - b1) * g
    v = b2 * v + (1.0 - b2) * (g * g)
    m_hat = m / (1.0 - b1 ** t)
    v_hat = v / (1.0 - b2 ** t)
    h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
    h_new = project_feasible(h_new, target_sum)
    return h_new, m, v

        if c5 < best_c5:
            best_h, best_c5 = h5, c5

    # Phase 6: Binary annealing (100s)
    if best_h is not None and time.time() < deadline - 100:
        alloc = min(80, deadline - time.time() - 30)
        hb, cb = binary_anneal(best_h, alloc, seed + 606)
        if cb < best_c5:
            best_h, best_c5 = hb, cb

    # Phase 7: Exhaustive single swap polish (40s)
    if best_h is not None and time.time() < deadline - 40:
        n = len(best_h)
        if n > 40 and (max(best_h) - min(best_h)) > 0.5:
            hb2 = (best_h >= 0.5).astype(float)
            hs = int(hb2.sum())
            tgt = n // 2
            if hs < tgt:
                idx0 = np.where(hb2 == 0)[0]
                hb2[idx0[:tgt - hs]] = 1.0
            elif hs > tgt:
                idx1 = np.where(hb2 == 1)[0]
                hb2[idx1[:hs - tgt]] = 0.0
            h7, c7 = exhaustive_single_swap(hb2, max_rounds=20)
            if c7 < best_c5:
                best_h, best_c5 = h7, c7

    # Phase 8: Final fractional polish (30s)
    if best_h is not None and time.time() < deadline - 5:
        alloc = min(25, deadline - time.time() - 2)
        n = len(best_h)
        h8, c8 = optimize_phase(best_h, alloc, seed + 707, 200.0, 1500.0, 0.015, 0.001)
        if c8 < best_c5:
            best_h, best_c5 = h8, c8

    # Phase 9: One more binary anneal from updated best (20s)
    if best_h is not None and time.time() < deadline - 20:
        alloc = min(18, deadline - time.time() - 2)
        hb9, cb9 = binary_anneal(best_h, alloc, seed + 808)
        if cb9 < best_c5:
            best_h, best_c5 = hb9, cb9

    if best_h is None:
        best_h = project_feasible(h70, 35.0)
        best_c5 = compute_c5(best_h)

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
