def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(800):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def constraint_eq(x):
    return np.sum(x) - len(x) / 2.0

def constraint_eq_jac(x):
    return np.ones(len(x))

def soft_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    m = c.max()
    shifted = beta * (c - m)
    shifted -= shifted.max()
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    val = (m + np.log(sum_exp) / beta) * dx
    weights = exp_s / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        off = k - (n - 1)
        i_start = max(0, -off)
        i_end = min(n, n - off)
        if i_start >= i_end:
            continue
        i_idx = np.arange(i_start, i_end)
        j_idx = i_idx + off
        grad[i_idx] += w * (1.0 - h[j_idx])
        grad[j_idx] -= w * h[i_idx]
    return val, grad * dx

def soft_obj_only(h, beta):
    return soft_obj_and_grad(h, beta)[0]

def soft_grad_only(h, beta):
    return soft_obj_and_grad(h, beta)[1]

def subgradient_step(h, n, k_star):
    dx = 2.0 / n
    off = k_star - (n - 1)
    i_start = max(0, -off)
    i_end = min(n, n - off)
    i_idx = np.arange(i_start, i_end)
    j_idx = i_idx + off
    grad = np.zeros(n)
    grad[i_idx] += (1.0 - h[j_idx])
    grad[j_idx] -= h[i_idx]
    return grad * dx

def topk_focused_obj_and_grad(h, beta, k_focus, focus_weight):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    m = c.max()
    shifted = beta * (c - m)
    shifted -= shifted.max()
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    val = (m + np.log(sum_exp) / beta) * dx
    weights = exp_s / sum_exp.copy()
    for kf in k_focus:
        weights[kf] *= focus_weight
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        off = k - (n - 1)
        i_start = max(0, -off)
        i_end = min(n, n - off)
        if i_start >= i_end:
            continue
        i_idx = np.arange(i_start, i_end)
        j_idx = i_idx + off
        grad[i_idx] += w * (1.0 - h[j_idx])
        grad[j_idx] -= w * h[i_idx]
    return val, grad * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        n0 = len(initial_h_values)
        best_h = project_feasible(initial_h_values.copy(), n0)
    else:
        n0 = 160
        best_h = np.full(n0, 0.5)

    best_c5 = compute_c5(best_h)

    def remaining():
        return budget_s - (time.time() - t0)

    def slsqp_round(h_in, n, beta, maxiter, ftol):
        bounds = [(0.0, 1.0)] * n
        constraints = [{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}]
        res = minimize(soft_obj_only, h_in.copy(), args=(beta,),
                       method='SLSQP', bounds=bounds, constraints=constraints,
                       jac=lambda x: soft_grad_only(x, beta),
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
        return h_in

    # Phase 1: SLSQP with analytic gradient, multi-beta at base resolution
    n = n0
    for beta in [10, 50, 200, 800, 3000, 12000, 50000, 200000]:
        if remaining() < 80:
            break
        maxiter = 300 if beta < 100 else 150
        ftol = 1e-12 if beta < 1000 else 1e-14
        h_new = slsqp_round(best_h, n, beta, maxiter, ftol)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 2: Focus on top-k lag values
    if remaining() > 300:
        n = len(best_h)
        dx = 2.0 / n
        c = np.correlate(best_h, 1.0 - h, mode="full")
        c = np.correlate(best_h, 1.0 - best_h, mode="full")
        sorted_idx = np.argsort(c)[::-1]
        top_k = sorted_idx[:min(5, len(sorted_idx))].tolist()

        for focus_w in [2.0, 3.0, 5.0]:
            if remaining() < 60:
                break
            for beta in [500, 5000]:
                if remaining() < 40:
                    break
                bounds = [(0.0, 1.0)] * n
                constraints = [{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_jac}]
                def _typ_obj(x):
                    return topk_focused_obj_and_grad(x, beta, top_k, focus_w)[0]
                def _typ_grad(x):
                    return topk_focused_obj_and_grad(x, beta, top_k, focus_w)[1]
                res = minimize(_typ_obj, best_h.copy(), method='SLSQP',
                              bounds=bounds, constraints=constraints,
                              jac=_typ_grad,
                              options={'maxiter': 100, 'ftol': 1e-12})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    h_new = project_feasible(res.x, n)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new

    # Phase 3: Subgradient projected gradient on active lag
    if remaining() > 200:
        n = len(best_h)
        dx = 2.0 / n
        h_sg = best_h.copy()
        base_lr = 0.02 * dx
        for it in range(3000):
            if remaining() < 60:
                break
            c = np.correlate(h_sg, 1.0 - h_sg, mode="full")
            k_star = int(np.argmax(c))
            grad = subgradient_step(h_sg, n, k_star)
            lr = base_lr / (1.0 + 0.01 * it)
            h_next = h_sg - lr * grad
            h_next = project_feasible(h_next, n)
            c5_next = compute_c5(h_next)
            if c5_next < best_c5:
                best_c5 = c5_next
                best_h = h_next.copy()
            h_sg = h_next

    # Phase 4: Try lower resolution (n/2, n/4)
    if remaining() > 250:
        for n_try in [n0 // 2, n0 // 4, n0 // 3]:
            if remaining() < 80 or n_try < 32:
                break
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_try = np.interp(x_new, x_old, best_h)
            h_try = project_feasible(h_try, n_try)
            for beta in [100, 1000, 10000]:
                if remaining() < 50:
                    break
                h_new = slsqp_round(h_try, n_try, beta, 200, 1e-12)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
                    n = n_try
                    x_old = np.linspace(0, 2, n, endpoint=False)

    # Phase 5: Try slightly higher resolution
    if remaining() > 200 and n < 480:
        n_up = min(480, int(n * 1.5))
        if n_up > n + 10:
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_up, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = project_feasible(h_up, n_up)
            for beta in [200, 2000, 20000]:
                if remaining() < 50:
                    break
                h_new = slsqp_round(h_up, n_up, beta, 150, 1e-12)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
                    n = n_up

    # Phase 6: Final intensive SLSQP polishing at best resolution
    n = len(best_h)
    for beta in [50000, 200000, 500000, 1000000]:
        if remaining() < 25:
            break
        maxiter = 100 if beta > 100000 else 150
        h_new = slsqp_round(best_h, n, beta, maxiter, 1e-14)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 7: One more subgradient pass on the final solution
    if remaining() > 40:
        n = len(best_h)
        dx = 2.0 / n
        h_sg = best_h.copy()
        for it in range(1000):
            if remaining() < 15:
                break
            c = np.correlate(h_sg, 1.0 - h_sg, mode="full")
            k_star = int(np.argmax(c))
            grad = subgradient_step(h_sg, n, k_star)
            lr = 0.01 * dx / (1.0 + 0.005 * it)
            h_next = h_sg - lr * grad
            h_next = project_feasible(h_next, n)
            c5_next = compute_c5(h_next)
            if c5_next < best_c5:
                best_c5 = c5_next
                best_h = h_next.copy()
            h_sg = h_next

    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)

    try:
        val = evaluate_erdos_solution(best_h)
        if val < final_c5:
            final_c5 = val
    except Exception:
        pass

    return (best_h, final_c5, n_final)
