import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    hf = np.asarray(h, dtype=np.float64)
    corr = np.correlate(hf, 1.0 - hf, mode='full')
    return float(np.max(corr)) * dx

def make_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n, dtype=np.float64)
    b[idx[:target]] = 1.0
    return b

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-12:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            hp[mask] += diff / cnt
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def sa_binary(h_init, n, deadline, rng):
    h = h_init.astype(np.float64).copy()
    current_c5 = compute_c5(h, n)
    best_h = h.copy()
    best_c5 = current_c5
    T = 0.08
    min_T = 1e-6
    no_improve = 0
    iter_count = 0
    while time.time() < deadline:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        a = int(ones[rng.randint(len(ones))])
        b = int(zeros[rng.randint(len(zeros))])
        h[a] = 0.0
        h[b] = 1.0
        new_c5 = compute_c5(h, n)
        delta = new_c5 - current_c5
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[b] = 0.0
            h[a] = 1.0
            no_improve += 1
        iter_count += 1
        if iter_count % 3000 == 0:
            T *= 0.93
        if no_improve > 8000:
            T = max(T * 2.0, 0.03)
            no_improve = 0
        if T < min_T:
            T = min_T
    return best_h, best_c5

def sa_binary_2swap(h_init, n, deadline, rng):

    current_c5 = compute_c5(h, n)
    best_h = h.copy()
    best_c5 = current_c5
    T = 0.12
    no_improve = 0
    iter_count = 0
    while time.time() < deadline:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        use_2swap = (iter_count % 5 == 0) and len(ones) >= 2 and len(zeros) >= 2
        if use_2swap:
            pa = rng.permutation(len(ones))[:2]
            pb = rng.permutation(len(zeros))[:2]
            a1, a2 = int(ones[pa[0]]), int(ones[pa[1]])
            b1, b2 = int(zeros[pb[0]]), int(zeros[pb[1]])
            h[a1] = 0.0; h[a2] = 0.0; h[b1] = 1.0; h[b2] = 1.0
            new_c5 = compute_c5(h, n)
            delta = new_c5 - current_c5
            if delta <= 0 or rng.random() < np.exp(-delta / T):

                if new_c5 < best_c5:

                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[b1] = 0.0; h[b2] = 0.0; h[a1] = 1.0; h[a2] = 1.0
                no_improve += 1
        else:
            a = int(ones[rng.randint(len(ones))])
            b = int(zeros[rng.randint(len(zeros))])
            h[a] = 0.0; h[b] = 1.0
            new_c5 = compute_c5(h, n)

            if delta <= 0 or rng.random() < np.exp(-delta / T):

        if no_improve > 10000:

    return best_h, best_c5

def local_search(h, n, deadline):
    h = h.astype(np.float64).copy()
    current_c5 = compute_c5(h, n)
    best_h = h.copy()
    best_c5 = current_c5
    while time.time() < deadline:
        improved = False
        best_delta = 0.0
        best_a = -1
        best_b = -1
        ones = list(np.where(h == 1.0)[0])
        zeros = list(np.where(h == 0.0)[0])
        for a in ones:
            for b in zeros:
                h[a] = 0.0; h[b] = 1.0
                c = compute_c5(h, n)
                h[b] = 0.0; h[a] = 1.0
                d = current_c5 - c
                if d > best_delta:
                    best_delta = d
                    best_a = a
                    best_b = b
        if best_a >= 0:
            h[best_a] = 0.0; h[best_b] = 1.0
            current_c5 = compute_c5(h, n)
            best_c5 = current_c5
            best_h = h.copy()
            improved = True
        if not improved:
            break
    return best_h, best_c5

def continuous_optimize(h, n, deadline, rng):
    dx = 2.0 / n
    target = n / 2.0
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    lr = 0.4
    stalled = 0
    it = 0
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        top_k = np.argsort(corr)[-min(5, len(corr)):]
        g = np.zeros(n)
        for kk in top_k:
            lag = int(kk) - (n - 1)
            for j in range(n):
                jl = j - lag
                if 0 <= jl < n:
                    g[j] += 1.0 - h[jl]
                jr = j + lag
                if 0 <= jr < n:
                    g[j] -= h[jr]
        h_new = h - lr * g * dx
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c5 - 1e-14:
            best_c5 = c_new
            best_h = h_new.copy()
            stalled = 0
        else:
            stalled += 1
        h = h_new
        it += 1
        if stalled > 150:
            lr *= 0.65
            stalled = 0
            if lr < 0.003:
                h = best_h + rng.randn(n) * 0.12
                h = project_feasible(h, n)
                lr = 0.4
    return best_h, best_c5
