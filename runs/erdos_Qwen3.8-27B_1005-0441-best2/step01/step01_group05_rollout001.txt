    m = np.max(corr)

def project_h(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    target = n / 2.0
    if abs(s - target) > 1e-14:
        # Need to adjust to hit target
        if s > target:
            # Reduce values, preferentially reducing values closer to 1
            overshoot = s - target
            # Distribute reduction proportional to h
            frac = overshoot / max(s, 1e-14)
            h = h * (1.0 - frac)
            h = np.clip(h, 0.0, 1.0)
        else:
            # Increase values
            deficit = target - s
            h = h + deficit / n
            h = np.clip(h, 0.0, 1.0)
    # Final clip and check
    h = np.clip(h, 0.0, 1.0)
    return h

from scipy.optimize import minimize
import time

_g_n = 160
_g_temp = 0.001

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _obj(h):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    t = max(_g_temp, 1e-10)
    shifted = np.clip((corr - m) / t, -60, 0)
    return (m + t * np.log(np.sum(np.exp(shifted)))) * dx

def _cons_eq(h):
    return np.sum(h) - _g_n / 2.0

def _cons_eq_jac(h):
    return np.ones(_g_n)

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        if mask.sum() > 0:
            adj = diff / mask.sum()
            h[mask] += adj
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _slsqp(h_init, n, seed, max_time, temps=None):
    global _g_n, _g_temp
    t0 = time.time()
    _g_n = n
    best = h_init.copy()

                h_new = _project(h_new, n)

                if v < best_val - 1e-14:
                    best = h_new.copy()
                    best_val = v
                h = h_new.copy()
            except Exception:
                break
    return best, best_val

def _grad_max_corr(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    t = max(temp, 1e-10)
    w = np.exp(np.clip((corr - m) / t, -60, 0))
    sw = w.sum()
    if sw < 1e-30:
        return np.zeros(n)
    w = w / sw
    grad = np.zeros(n)
    n_corr = len(corr)
    for idx in range(n_corr):
        wi = w[idx]
        if wi < 1e-10:
            continue
        k = idx - (n - 1)
        if k >= 0:
            n_min, n_max = 0, n - 1 - k
        else:
            n_min, n_max = -k, n - 1
        for j in range(n):
            nv = j - k
            if n_min <= nv <= n_max:
                grad[j] += wi * (1.0 - h[nv])
            if n_min <= j <= n_max:
                jk = j + k
                if 0 <= jk < n:
                    grad[j] -= wi * h[jk]
    return grad * dx

def _pgd(h_init, n, max_time, lr0=0.1, lr_end=0.001, temp0=0.01, temp_end=1e-4):
    t0 = time.time()
    h = h_init.copy()
    best = h.copy()
    best_val = compute_c5(h)
    n_iter = 0
    while time.time() - t0 < max_time:
        frac = min(1.0, n_iter / max(1, max_time * 2))
        lr = lr0 * (lr_end / lr0) ** frac
        temp = temp0 * (temp_end / temp0) ** frac
        g = _grad_max_corr(h, n, temp)
        h_new = h - lr * g
        h_new = _project(h_new, n)
        v = compute_c5(h_new)
        if v < best_val:
            best = h_new.copy()
            best_val = v
        h = h_new
        n_iter += 1
    return best, best_val

    h_p = h + mod
    h_p = _project(h_p, n)

    h_new = np.interp(x_new, x_old, h)
    h_new = _project(h_new, n_new)
    return h_new
