import numpy as np
import time
import math

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def smooth_opt(h_init, t_budget, seed, lr0=0.06):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    ts = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), ts)
    bh = h.copy()
    bc = compute_c5(h)
    v = np.zeros(n)
    lr = lr0
    tau = 40.0
    t0 = time.time()
    dl = t0 + t_budget
    ni = 0
    while time.time() < dl - 0.3:
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = corr.max()
        sh = np.clip(tau * (corr - mc), -400, 400)
        w = np.exp(sh)
        w /= w.sum()
        g = np.zeros(n)
        for k in range(len(corr)):
            wk = w[k]
            if wk < 1e-12:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                m = n - lag
                g[lag:] += wk * (1.0 - h[:m])
                g[:m] -= wk * h[lag:]
            else:
                L = -lag
                g[:n-L] += wk * (1.0 - h[L:])
                g[L:] -= wk * h[:n-L]
        g *= dx
        v = 0.85 * v - lr * g
        h = project_feasible(h + v, ts)
        c = compute_c5(h)
        if c < bc - 1e-12:
            bh = h.copy()
            bc = c
            ni = 0
        else:
            ni += 1
        tau = min(tau * 1.005, 600.0)
        if ni > 300:
            if lr > 0.003:
                lr *= 0.7
            else:
                h = project_feasible(bh + rng.randn(n) * 0.04, ts)
                v[:] = 0
                lr = lr0
                tau = 40.0
                ni = 0
    return bh, bc

def bin_steepest(h_frac, t_budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h_frac)
    no = n // 2
    h = (h_frac >= 0.5).astype(float)
    s = int(h.sum())
    if s > no:
        oi = np.where(h == 1.0)[0]
        for i in range(s - no):
            h[oi[i]] = 0.0
    elif s < no:
        zi = np.where(h == 0.0)[0]
        for i in range(no - s):
            h[zi[i]] = 1.0
    bh = h.copy()
    bc = compute_c5(h)
    cc = bc
    t0 = time.time()
    dl = t0 + t_budget
    stuck = 0
    while time.time() < dl - 0.5:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_imp = 0.0
        best_mv = None
        for i in ones:
            h[i] = 0.0
            for j in zeros:
                h[j] = 1.0
                c = compute_c5(h)
                if c < cc - best_imp:
                    best_imp = cc - c
                    best_mv = (i, j)
                h[j] = 0.0
            h[i] = 1.0
        if best_mv:
            i, j = best_mv
            h[i] = 0.0
            h[j] = 1.0
            cc = compute_c5(h)
            if cc < bc:
                bc = cc
                bh = h.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 5:
                kk = rng.randint(2, min(6, len(ones), len(zeros)) + 1)
                ko = rng.choice(ones, kk, replace=False)
                kz = rng.choice(zeros, kk, replace=False)
                for a in ko:
                    h[a] = 0.0
                for b in kz:
                    h[b] = 1.0
                cc = compute_c5(h)
                stuck = 0
    return bh, bc

        h_init = np.asarray(initial_h_values, dtype=float)
        n = len(h_init)
    else:
        h_init = np.full(n, 0.5)

    # Phase 1: continuous optimization
    t_rem = deadline - time.time() - 5
    t1 = min(t_rem * 0.15, 100)
    h1, c1 = smooth_opt(h_init, t1, seed)
    if c1 < best_c5:
        best_h, best_c5, best_n = h1, c1, n

    # Phase 2: binary steepest from continuous (primary)
    t_rem = deadline - time.time() - 5
    t2 = min(t_rem * 0.30, 280)
    hb, cb = bin_steepest(h1, t2, seed + 100)
    if cb < best_c5:
        best_h, best_c5, best_n = hb, cb, n

    # Phase 3: binary steepest again with different kicks
    t_rem = deadline - time.time() - 5
    t3 = min(t_rem * 0.20, 180)
    hb2, cb2 = bin_steepest(h1, t3, seed + 300)
    if cb2 < best_c5:
        best_h, best_c5, best_n = hb2, cb2, n

    # Phase 4: fresh continuous from perturbed best + binary
    for trial in range(3):
        t_rem = deadline - time.time() - 5
        if t_rem < 80:
            break
        rng_t = np.random.RandomState(seed + 700 + trial * 100)
        h_try = best_h + rng_t.randn(len(best_h)) * 0.06
        t_cont = min(t_rem * 0.25, 50)
        h_opt, c_opt = smooth_opt(h_try, t_cont, seed + 1000 + trial * 50)
        if c_opt < best_c5:
            best_h, best_c5 = h_opt, c_opt
        t_rem = deadline - time.time() - 5
        if t_rem < 30:
            continue
        t_bin = min(t_rem * 0.35, 120)
        hb3, cb3 = bin_steepest(h_opt, t_bin, seed + 2000 + trial * 50)
        if cb3 < best_c5:
            best_h, best_c5, best_n = hb3, cb3, len(h_opt)

    # Phase 5: try n=50
    t_rem = deadline - time.time() - 5
    if t_rem > 100:
        n50 = 50
        rng50 = np.random.RandomState(seed + 5000)
        h50 = project_feasible(np.full(n50, 0.5) + rng50.randn(n50) * 0.1, n50 / 2.0)
        t50c = min(t_rem * 0.25, 40)
        h50o, c50o = smooth_opt(h50, t50c, seed + 5100)
        t_rem = deadline - time.time() - 5
        if t_rem > 30:
            t50b = min(t_rem * 0.30, 60)
            h50b, c50b = bin_steepest(h50o, t50b, seed + 5200)
            if c50b < best_c5:
                best_h, best_c5, best_n = h50b, c50b, n50

    # Phase 6: final polish
    t_rem = deadline - time.time() - 2
    if t_rem > 15:
        hp, cp = smooth_opt(best_h, min(t_rem * 0.4, 25), seed + 9999, lr0=0.005)
        if cp < best_c5:
            best_h, best_c5, best_n = hp, cp, len(best_h)
        t_rem2 = deadline - time.time() - 1
        if t_rem2 > 15:
            hb5, cb5 = bin_steepest(best_h, min(t_rem2 * 0.5, 30), seed + 8888)
            if cb5 < best_c5:
                best_h, best_c5, best_n = hb5, cb5, len(best_h)

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
