import numpy as np
import time, math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def lse_grad(h, dx, tau):
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    mc = corr.max()
    shifted = np.clip(tau * (corr - mc), -500, 500)
    exps = np.exp(shifted)
    ws = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = ws[k]
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:] += w * omh[:n - lag]
            grad[:n - lag] -= w * h[lag:]
        else:
            L = -lag
            grad[:n - L] += w * omh[L:]
            grad[L:] -= w * h[:n - L]
    return grad * dx

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def adam_opt(h_init, t_budget, seed, tau0=50.0, lr0=0.05):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    ts = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), ts)
    bh, bc = h.copy(), compute_c5(h)
    m1, m2 = np.zeros(n), np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    tau, t, ni = tau0, 0, 0
    t0 = time.time()
    dl = t0 + t_budget
    while time.time() < dl - 0.2:
        t += 1
        g = lse_grad(h, dx, tau)
        m1 = b1 * m1 + (1 - b1) * g
        m2 = b2 * m2 + (1 - b2) * g * g
        mh = m1 / (1 - b1 ** t)
        vh = m2 / (1 - b2 ** t)
        lr = lr0 / (1 + 0.003 * t)
        h = project_feasible(h - lr * mh / (np.sqrt(vh) + eps), ts)
        c = compute_c5(h)
        if c < bc:
            bh, bc, ni = h.copy(), c, 0
        else:
            ni += 1
        tau = min(tau * 1.004, 500.0)
        if ni > 400:
            h = project_feasible(bh + rng.randn(n) * 0.07, ts)
            m1, m2, t, tau, ni = np.zeros(n), np.zeros(n), 0, tau0, 0
    return bh, bc

def binary_sa(h_frac, t_budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h_frac)
    tc = n // 2
    h = (h_frac >= 0.5).astype(float)
    if h.sum() > tc:
        h[np.argsort(h)[::-1][:int(h.sum()) - tc]] = 0
    elif h.sum() < tc:
        h[np.argsort(h)[:tc - int(h.sum())]] = 1
    bh, bc = h.copy(), compute_c5(h)
    t0 = time.time()
    dl = t0 + t_budget
    T = 0.001
    while time.time() < dl - 0.3:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i, j = ones[rng.randint(len(ones))], zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0, 1
        c = compute_c5(h)
        delta = c - bc
        if delta < 0 or (T > 1e-7 and rng.rand() < math.exp(-delta / max(T, 1e-7))):
            bc, bh = c, h.copy()
        else:
            h[i], h[j] = 1, 0
        T *= 0.9995
    return bh, bc

def make_structured(n, seed, mode=0):
    rng = np.random.RandomState(seed)
    if mode == 0:
        p = rng.randint(3, 9)
        w = max(1, p // 2)
        ht = np.array([1.0 if i % p < w else 0.0 for i in range(n)])
    elif mode == 1:
        ht = 0.5 + 0.3 * np.sin(2 * np.pi * rng.randint(2, 6) * np.arange(n) / n + rng.rand())
    else:
        ht = rng.rand(n) * 0.3 + 0.35
    return ht

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        h0 = np.full(70, 0.5)
    n = len(h0)
    ts = n / 2.0
    h0 = project_feasible(h0, ts)
    bh, bc = h0.copy(), compute_c5(h0)

    # Phase 1: Main optimization from initial (35% of time)
    tr = dl - time.time()
    if tr > 80:
        h, c = adam_opt(h0, min(tr * 0.35, 350), seed, tau0=50, lr0=0.06)
        if c < bc:
            bh, bc = h, c
        hb, cb = binary_sa(bh, min(tr * 0.08, 60), seed + 7)
        if cb < bc:
            bh, bc = hb, cb

    # Phase 2: Multiple restarts from perturbed best (30% of time)
    n_trials = 5
    for i in range(n_trials):
        tr = dl - time.time()
        if tr < 50:
            break
        rt = np.random.RandomState(seed + 200 * (i + 1))
        pert = 0.05 + 0.04 * i
        ht = project_feasible(bh + rt.randn(n) * pert, ts)
        alloc = min(tr * 0.18, 120)
        h, c = adam_opt(ht, alloc, seed + 500 + i, tau0=40 + 20 * i, lr0=0.04)
        if c < bc:
            bh, bc = h, c

    # Phase 3: Structured starts with different n (20% of time)
    for nn in [60, 70, 80, 65]:
        tr = dl - time.time()
        if tr < 50:
            break
        if nn == n:
            ht = make_structured(nn, seed + 300, 0)
        else:
            ht = make_structured(nn, seed + 300 + nn, np.random.RandomState(seed + nn).randint(3))
        ht = project_feasible(ht, nn / 2.0)
        alloc = min(tr * 0.15, 100)
        h, c = adam_opt(ht, alloc, seed + 2000 + nn, tau0=50, lr0=0.05)
        if c < bc:
            bh, bc, n = h, c, nn
        hb, cb = binary_sa(h, min(alloc * 0.4, 35), seed + 3000 + nn)
        if cb < bc:
            bh, bc, n = hb, cb, nn

    # Phase 4: More diverse restarts on best n (10% of time)
    for i in range(3):
        tr = dl - time.time()
        if tr < 40:
            break
        rt = np.random.RandomState(seed + 5000 + i * 53)
        ht = make_structured(n, seed + 6000 + i, i % 3)
        h_mix = 0.5 * ht + 0.5 * bh
        h_mix = project_feasible(h_mix, n / 2.0)
        alloc = min(tr * 0.15, 80)
        h, c = adam_opt(h_mix, alloc, seed + 7000 + i, tau0=60, lr0=0.04)
        if c < bc:
            bh, bc = h, c
        hb, cb = binary_sa(h, min(alloc * 0.3, 25), seed + 8000 + i)
        if cb < bc:
            bh, bc = hb, cb

    # Phase 5: Final polish (remaining time)
    tr = dl - time.time()
    if tr > 15:
        h, c = adam_opt(bh, min(tr * 0.5, 50), seed + 9999, tau0=200, lr0=0.008)
        if c < bc:
            bh, bc = h, c
        hb, cb = binary_sa(bh, min(tr * 0.4, 30), seed + 9998)
        if cb < bc:
            bh, bc = hb, cb

    bc = compute_c5(bh)
    return bh, bc, n
