def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-14:
        mask = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        cnt = float(np.sum(mask))
        if cnt > 0:
            hp[mask] = np.clip(hp[mask] + diff / cnt, 0.0, 1.0)
    return hp

def get_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def get_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_lag_vec(h, lag, n):
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            g[:n - lag] += (1.0 - h[lag:n])
            g[lag:n] -= h[:n - lag]
    else:
        m = -lag
        if m < n:
            g[m:] += (1.0 - h[:n - m])
            g[:n - m] -= h[m:]
    return g

def multi_lag_optimize(h, n, deadline, rng, eps0=0.03):
