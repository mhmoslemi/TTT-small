import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            h[mask] = np.minimum(h[mask] + diff / mask.sum(), 1.0)
        else:
            mask = h > 0.0
            if not mask.any():
                break
            h[mask] = np.maximum(h[mask] + diff / mask.sum(), 0.0)
    return np.clip(h, 0.0, 1.0)

def _soft_max_parts(h, n, beta):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = c.max()
    s = beta * (c - m)
    s -= s.max()
    e = np.exp(s)
    S = e.sum()
    w = e / S
    val = (m + np.log(S) / beta) * (2.0 / n)
    return val, w, c

def _grad_soft_max(h, n, w):
    nn = 2 * n - 1
    h_ext = np.zeros(n + nn - 1)
    h_ext[n - 1:n - 1 + n] = h
    M = np.array([h_ext[m:m + nn] for m in range(n)])
    A = M @ w
    B = M @ w[::-1]
    return ((1.0 - A) - B) * (2.0 / n)

def _topk_parts(h, n, topk, beta):
    c = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(c)[-topk:]
    c_s = c[idx]
    m = c_s.max()
    s = beta * (c_s - m)
    s -= s.max()
    e = np.exp(s)
    S = e.sum()
    w_kv = e / S
    val = (m + np.log(S) / beta) * (2.0 / n)
    w_full = np.zeros(nn_full_len(n))
    for i, j in enumerate(idx):
        w_full[j] = w_kv[i]
    return val, w_full, c

    return 2 * n - 1

def slsqp_polish(h, n, t0, time_limit):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        m = c.max()
        s = 50.0 * (c - m)
        s -= s.max()
        return (m + np.log(np.exp(s).sum()) / 50.0) * dx
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        m = c.max()
        s = 50.0 * (c - m)
        s -= s.max()
        e = np.exp(s)
        w = e / e.sum()
        return _grad_soft_max(x, n, w)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n)
            return h_new

        pass
    return h

    # Phase 6: Final SLSQP + projected gradient combo
    if time.time() - t0 < budget_s * 0.70:
        h_pol = slsqp_polish(best_h.copy(), n, t0, budget_s * 0.10)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 7: Try upsampled optimization if time and size allow
    if time.time() - t0 < budget_s * 0.35 and n <= 320:
        n2 = n * 2
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_feasible(h_up, n2)
        h_opt, c5_opt = projected_gradient(h_up, n2, t0, budget_s * 0.25,
                                           max_iters=1000, topk=5)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final: ensure feasibility and compute
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)

    try:
        val = evaluate_erdos_solution(best_h)
        if val < final_c5:
            final_c5 = val
    except Exception:
        pass

    return (best_h, final_c5, n_final)
