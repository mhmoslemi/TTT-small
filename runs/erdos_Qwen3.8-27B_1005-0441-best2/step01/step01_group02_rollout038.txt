            nm = minimize(obj_fin, best_h.copy(), method='Nelder-Mead',
                         options={'maxiter': 600, 'xatol': 1e-11, 'fatol': 1e-13})
            h_f = _project(nm.x, n)
            c_f = compute_c5(h_f)
            if c_f < best_c5:
                best_c5 = c_f
                best_h = h_f
        except Exception:
            pass

    # Phase 6: Coordinate nudge
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 8:
        h_now = best_h.copy()
        for _ in range(3):
            improved = False
            for j in range(n):
                for d in [0.003, -0.003, 0.007, -0.007]:
                    h_try = h_now.copy()
                    h_try[j] += d
                    k2 = (j + 7) % n
                    h_try[k2] -= d
                    if h_try[j] >= 0 and h_try[j] <= 1 and h_try[k2] >= 0 and h_try[k2] <= 1:
                        h_try = _project(h_try, n)
                        c_try = compute_c5(h_try)
                        if c_try < best_c5 - 1e-13:
                            h_now = h_try
                            best_c5 = c_try
                            improved = True
                            break
                if improved:
                    break
        best_h = _project(h_now, n)
        c_final = compute_c5(best_h)
        if c_final < best_c5:
            best_c5 = c_final
            best_h = best_h

    best_h = _project(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, n)
