def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _smooth_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    return float((mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta) * dx)

def _smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-14:
            continue
        d = (n - 1) - i
        lo1 = max(0, -d)
        hi1 = min(n, n - d)
        if lo1 < hi1:

        lo2 = max(0, d)
        hi2 = min(n, d + n)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2 - d:hi2 - d]
