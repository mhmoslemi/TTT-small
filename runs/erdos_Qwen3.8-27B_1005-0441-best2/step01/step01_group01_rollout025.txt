    """Take one projected subgradient step on the true max."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    shift = k_star - (n - 1)
    
    # Gradient of corr[k_star] w.r.t. h
    # corr[k] = sum_i h[i]*(1-h[i-shift]) (roughly)
    # d/dh[j] = (1-h[j-shift]) - h[j+shift] (for valid indices)
    grad = np.zeros(n)
    for j in range(n):
        gs = j - shift
        gp = j + shift
        if 0 <= gs < n:
            grad[j] += (1.0 - h[gs])
        if 0 <= gp < n:
            grad[j] -= h[gp]
    grad *= dx
    
    # Projected gradient descent
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(300):
        d = target - np.sum(h)
        if abs(d) < 1e-15:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        cnt = int(np.sum(free))
        if cnt == 0:
            break
        h[free] += d / cnt
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = int(np.sum(free))
        if cnt > 0:
            h[free] += resid / cnt
    return h

def smooth_max_obj(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    z = np.clip(alpha * (corr - m), -500.0, 500.0)
    return (m + np.log(np.sum(np.exp(z))) / alpha) * dx

def true_max_obj(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.max(corr) * dx)
