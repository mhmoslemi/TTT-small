import numpy as np
import time

def get_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def get_c5(h, n):
    return float(np.max(get_overlaps(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-14:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        cnt = float(mask.sum())
        if cnt > 1e-10:
            hp[mask] = np.clip(hp[mask] + diff / cnt, 0.0, 1.0)
    return hp

def to_binary(h, n):
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:n // 2]] = 1.0
    return b

def sa_binary(h, n, deadline, rng):
    h = h.copy()
    cur = get_c5(h, n)
    best_c5 = cur
    best_h = h.copy()
    
    T = 0.015
    T_min = 1e-9
    cool = 0.9997
    no_improve = 0
    reheat_T = 0.01
    
    while time.time() < deadline:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        p = ones[rng.randint(len(ones))]
        q = zeros[rng.randint(len(zeros))]
        
        h[p] = 0.0
        h[q] = 1.0
        new_c5 = get_c5(h, n)
        delta = new_c5 - cur
        
        if delta < 0 or (T > T_min and rng.rand() < np.exp(-delta / T)):
            cur = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[p] = 1.0
            h[q] = 0.0
        
        T *= cool
        if no_improve > 3000:
            T = reheat_T
            no_improve = 0
            reheat_T *= 0.8
    
    return best_h, best_c5

def continuous_refine(h, n, deadline, rng):
    """Projected subgradient with momentum and multi-lag."""
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = get_c5(h, n)
    
    lr = 0.3
    momentum = np.zeros(n)
    mu = 0.7
    stalled = 0
    
    while time.time() < deadline:
        ov = get_overlaps(h, n)
        # Top 3 worst lags
        top_idx = np.argsort(ov)[-3:]
        top_ov = np.sort(ov[top_idx])[::-1]
        
        # Weighted subgradient
        g = np.zeros(n)
        for w, idx in zip(top_ov, top_idx):
            lag = int(idx) - (n - 1)
            for j in range(n):
                gm = 0.0
                jm = j - lag
                jp = j + lag
                if 0 <= jm < n:
                    gm += (1.0 - h[jm])
                if 0 <= jp < n:
                    gm -= h[jp]
                g[j] += w * gm
        
        # Normalize gradient
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        g = g / gnorm
        
        momentum = mu * momentum - lr * g * dx
        h_new = h + momentum
        h_new = project_feasible(h_new, n)
        
        c_new = get_c5(h_new, n)
        if c_new < best_c5 - 1e-12:
            best_c5 = c_new
            best_h = h_new.copy()
            stalled = 0

            momentum *= 0.5
            stalled = 0

                h = project_feasible(h, n)
                lr = 0.3
                momentum = np.zeros(n)
    
    return best_h, best_c5
