import numpy as np
import time, math

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def grads_smooth(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    s = np.clip(tau * (corr - mc), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-13:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            g[lag:] += wk * (1.0 - h[:n - lag])
            g[:n - lag] -= wk * h[lag:]
        else:
            L = -lag
            g[:n - L] += wk * (1.0 - h[L:])
            g[L:] -= wk * h[:n - L]
    return g * dx

def grad_opt(h0, tb, seed, lr0=0.05):
    rng = np.random.RandomState(seed)
    n = len(h0)
    ts = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h0.copy(), ts)
    bh, bc = h.copy(), compute_c5(h)
    v = np.zeros(n)
    lr, tau = lr0, 40.0
    ni = 0
    dl = time.time() + tb
    while time.time() < dl - 0.3:
        if ni > 300:
            h = project_feasible(bh + rng.randn(n) * 0.03, ts)
            v[:] = 0.0
            lr = lr0
            ni = 0
        elif ni > 100:
            lr = max(lr * 0.9, 0.002)
        g = grads_smooth(h, dx, tau)
        v = 0.8 * v - lr * g
        hn = project_feasible(h + v, ts)
        cn = compute_c5(hn)
        if cn < bc:
            bh, bc, ni = hn.copy(), cn, 0
        else:
            ni += 1
        h = hn
        tau = min(tau * 1.003, 400.0)
    return bh, bc

def to_bin(hf):
    n = len(hf)
    t = n // 2
    h = (hf >= 0.5).astype(float)
    d = t - int(h.sum())
    if d > 0:
        idx = np.argsort(-hf)
        h[idx[:d]] = 1.0
    elif d < 0:
        idx = np.argsort(hf)
        h[idx[:(-d)]] = 0.0
    return h

def sa_bin(h0, tb, seed):
    rng = np.random.RandomState(seed)
    n = len(h0)
    h = h0.copy()
    bh = h.copy()
    bc = compute_c5(h)
    cc = bc
    dl = time.time() + tb
    total = max(tb, 1.0)
    t_start = time.time()
    while time.time() < dl - 0.2:
        p = min(1.0, (time.time() - t_start) / total)

        i = ones[rng.randint(len(ones))]

            cc = cn
            if cn < bc:
