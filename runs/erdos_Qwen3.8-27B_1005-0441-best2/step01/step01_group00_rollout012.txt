def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _smooth_obj(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    shifted = (c - cmax) / eps
    shifted = np.clip(shifted, -50, 50)
    s = np.exp(shifted)
    return (cmax + eps * np.log(s.sum())) * dx

def _smooth_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    shifted = (c - cmax) / eps
    shifted = np.clip(shifted, -50, 50)
    s = np.exp(shifted)
    ssum = s.sum()
    w = s / ssum
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        m = k - (n - 1)
        if m >= 0:
            g[:n-m] += wk * (1.0 - h[m:])
            g[m:] -= wk * h[:n-m]
        else:
            p = -m
            g[p:] += wk * (1.0 - h[:n-p])
            g[:n-p] -= wk * h[p:]
    return g

def _proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        d = 0.5 * n - h.sum()
        if abs(d) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += d / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _eq_fun(x, n):
    return x.sum() - 0.5 * n

def _eq_jac(x, n):
    return np.ones(n)

def _subgrad_polish(h, n, max_iter, lr_init):
    h = h.copy()
    lr = lr_init
    g_avg = np.zeros(n)
    for it in range(max_iter):
        c = _corr(h)
        vals = sorted(c, reverse=True)
        threshold = vals[min(4, 2*n-2)]
        active = np.where(c >= threshold - 1e-10)[0]
        g = np.zeros(n)
        for k in active:
            m = k - (n - 1)
            if m >= 0:
                g[:n-m] += (1.0 - h[m:])
                g[m:] -= h[:n-m]
            else:
                p = -m
                g[p:] += (1.0 - h[:n-p])
                g[:n-p] -= h[p:]
        g /= len(active)
        # Remove mean to handle sum constraint approximately
        g -= g.mean()
        g_avg = 0.7 * g_avg + 0.3 * g
        h_new = _proj(h - lr * g_avg * (2.0 / n), n)
        v_new = _c5(h_new)
        v_cur = _c5(h)
        if v_new < v_cur:
            h = h_new
            lr = min(lr * 1.02, 0.08)
        else:
            lr *= 0.75
            if lr < 1e-9:
                break
    return h

def _coordinate_descent(h, n, max_rounds):
    h = h.copy()
    best_v = _c5(h)
    for r in range(max_rounds):
        improved = False
        for j in range(n):
            for delta in (0.02, -0.02, 0.01, -0.01):
                h[:]= h.copy()
                h[j] += delta
                h2 = _proj(h, n)
                v = _c5(h2)
                if v < best_v - 1e-12:
                    best_v = v
                    h = h2
                    improved = True
                    break
        if not improved:
            break
    return h, best_v

    best_h = h0.copy()
    best_val = _c5(h0)

    # Phase 1: SLSQP with decreasing epsilon on n0
    bounds = [(0.0, 1.0)] * n0
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n0,)}]
    h_curr = h0.copy()

    eps_schedule = [0.12, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001]
    for eps in eps_schedule:
        if remaining() < 100:
            break
        maxit = int(min(2500, 300 + 600 * (0.12 / eps) ** 0.4))
        try:
            res = minimize(_smooth_obj, h_curr.copy(), args=(eps,),
                          jac=_smooth_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxit, 'ftol': 1e-15})
            h_res = _proj(res.x, n0)
            v = _c5(h_res)
            if v < best_val:
                best_val = v
                best_h = h_res.copy()
            h_curr = _proj(0.3 * h_curr + 0.7 * res.x, n0)
        except Exception:
            pass

    # Phase 2: Try perturbations to escape local minima
    for pert in [0.005, 0.01, 0.02]:
        if remaining() < 80:
            break
        h_pert = _proj(best_h + pert * rng.randn(n0), n0)
        for eps in [0.01, 0.004]:
            if remaining() < 60:
                break
            try:
                res = minimize(_smooth_obj, h_pert.copy(), args=(eps,),
                              jac=_smooth_grad, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 500, 'ftol': 1e-14})
                h_res = _proj(res.x, n0)
                v = _c5(h_res)
                if v < best_val:
                    best_val = v
                    best_h = h_res.copy()
            except Exception:
                pass

    # Phase 3: Try other n values (interpolate and optimize)
    for n_try in [90, 110]:
        if remaining() < 150:
            break
        x_old = np.linspace(0, 2, n0)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_try = _proj(h_interp, n_try)
        bnds = [(0.0, 1.0)] * n_try
        cns = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n_try,)}]
        for eps in [0.02, 0.008, 0.003]:
            if remaining() < 80:
                break
            try:
                res = minimize(_smooth_obj, h_try.copy(), args=(eps,),
                              jac=_smooth_grad, method='SLSQP',
                              bounds=bnds, constraints=cns,
                              options={'maxiter': 600, 'ftol': 1e-14})
                h_res = _proj(res.x, n_try)
                v = _c5(h_res)
                if v < best_val:
                    best_val = v
                    best_h = h_res.copy()
                h_try = h_res
            except Exception:
                pass

    # Phase 4: Subgradient polish on best solution
    if best_h is not None:
        n = len(best_h)
        h_sub = _subgrad_polish(best_h, n, 5000, 0.03)
        v = _c5(h_sub)
        if v < best_val:
            best_val = v
            best_h = h_sub

    # Phase 5: Coordinate descent
    if best_h is not None and remaining() > 20:
        n = len(best_h)
        h_cd, v_cd = _coordinate_descent(best_h, n, 50)
        if v_cd < best_val:
            best_val = v_cd
            best_h = h_cd

    # Phase 6: Final SLSQP ultra-fine polish
    if best_h is not None and remaining() > 30:
        n = len(best_h)
        bnds = [(0.0, 1.0)] * n
        cns = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
        for eps in [0.0008, 0.0004]:
            if remaining() < 20:
                break
            try:
                res = minimize(_smooth_obj, best_h.copy(), args=(eps,),
                              jac=_smooth_grad, method='SLSQP',
                              bounds=bnds, constraints=cns,
                              options={'maxiter': 1500, 'ftol': 1e-16})
                h_res = _proj(res.x, n)
                v = _c5(h_res)
                if v < best_val:
                    best_val = v
                    best_h = h_res
            except Exception:
                pass

    # Final subgradient after all
    if best_h is not None and remaining() > 10:
        n = len(best_h)
        h_final = _subgrad_polish(best_h, n, 3000, 0.02)
        v = _c5(h_final)
        if v < best_val:
            best_val = v
            best_h = h_final

    best_h = _proj(best_h, len(best_h))
    best_val = _c5(best_h)
    return (best_h, best_val, len(best_h))
