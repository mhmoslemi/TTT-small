    s = _overlap_values(h)  # length 2n-1

import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _overlaps(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _smooth_max(s, alpha):
    m = np.max(s)
    return m + np.log(np.sum(np.exp(alpha * (s - m)))) / alpha

def _smooth_max_w(s, alpha):
    m = np.max(s)
    e = np.exp(alpha * (s - m))
    w = e / np.sum(e)
    return w

def _project_sum(h, n):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(proj)
    if abs(diff) > 1e-12:
        order = np.argsort(np.abs(proj - 0.5))[::-1]
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                d = min(diff, 1.0 - proj[idx])
                proj[idx] += d
                diff -= d
            else:
                d = min(-diff, proj[idx])
                proj[idx] -= d
                diff += d
    return proj

def _grad_c5_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    s = _overlaps(h)
    w = _smooth_max_w(s, alpha)
    grad = np.zeros(n)
    for ki in range(len(s)):
        wk = w[ki]
        if wk < 1e-10:
            continue
        k = ki - (n - 1)
        i_lo = max(0, -k)
        i_hi = min(n - 1, n - 1 - k)
        if i_lo > i_hi:
            continue
        js = np.arange(i_lo, i_hi + 1)
        # Source: d/dh[j] of h[j]*(1-h[j+k]) = (1-h[j+k]) for j in [i_lo,i_hi]
        jk = js + k
        valid = (jk >= 0) & (jk < n)
        src = np.zeros(n)
        src[js[valid]] = 1.0 - h[jk[valid]]
        # Target: d/dh[j] of h[j-k]*(1-h[j]) = -h[j-k] for j-k in [i_lo,i_hi]
        jt_lo = max(0, i_lo + k)
        jt_hi = min(n - 1, i_hi + k)
        if jt_lo <= jt_hi:
            jt = np.arange(jt_lo, jt_hi + 1)
            jtk = jt - k
            valid_t = (jtk >= i_lo) & (jtk <= i_hi)
            tgt = np.zeros(n)
            tgt[jt[valid_t]] = -h[jtk[valid_t]]
            grad += wk * dx * (src + tgt)
        else:
            grad += wk * dx * src
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _overlaps(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _smooth_max(s, alpha):
    m = np.max(s)
    return m + np.log(np.sum(np.exp(alpha * (s - m)))) / alpha

def _smooth_max_w(s, alpha):
    m = np.max(s)
    e = np.exp(alpha * (s - m))
    return e / np.sum(e)

def _project_sum(h, n):
    h = np.array(h, dtype=float)
    np.clip(h, 0.0, 1.0, out=h)
    target = n / 2.0
    s = float(np.sum(h))
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if float(np.sum(proj)) > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-13:
        order = np.argsort(np.abs(proj - 0.5))[::-1]
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                d = min(diff, 1.0 - proj[idx])
                proj[idx] += d
                diff -= d
            else:
                d = min(-diff, proj[idx])
                proj[idx] -= d
                diff += d
    return proj

def _grad_c5_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    s = _overlaps(h)
    w = _smooth_max_w(s, alpha)
    grad = np.zeros(n)
    for ki in range(len(s)):
        wk = w[ki]
        if wk < 1e-10:
            continue
        k = ki - (n - 1)
        i_lo = max(0, -k)
        i_hi = min(n - 1, n - 1 - k)
        if i_lo > i_hi:
            continue
        length = i_hi - i_lo + 1
        contrib = np.zeros(n)
        src_slice = slice(i_lo, i_hi + 1)
        src_h = slice(i_lo + k, i_hi + k + 1)
        contrib[src_slice] += 1.0 - h[src_h]
        tgt_lo, tgt_hi = i_lo + k, i_hi + k
        tgt_slice = slice(tgt_lo, tgt_hi + 1)
        tgt_h = slice(tgt_lo - k, tgt_hi - k + 1)
        contrib[tgt_slice] -= h[tgt_h]
        grad += wk * dx * contrib
    return grad

def _proj_grad_desc(h0, alpha, n_iter, step0, t_end):
    n = len(h0)
    h = np.array(h0, dtype=float)
    best_h = h.copy()
    best_c5 = _eval_c5(h)
    stall = 0
    for it in range(n_iter):
        if time.time() > t_end:
            break
        step = step0 / (1.0 + 0.05 * it)
        grad = _grad_c5_smooth(h, alpha)
        best_trial_c5 = best_c5 + 0.01
        best_trial_h = h
        for f in [2.0, 1.0, 0.5, 0.25, 0.1]:
            ht = _project_sum(h - step * f * grad, n)
            c5t = _eval_c5(ht)
            if c5t < best_trial_c5:
                best_trial_c5 = c5t
                best_trial_h = ht
        h = best_trial_h
        if best_trial_c5 < best_c5 - 1e-12:
            best_c5 = best_trial_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 60:
            h = _project_sum(h + np.random.default_rng(it).normal(0, 0.03, n), n)
            stall = 0
    return best_h, best_c5

def _slsqp_refine(h0, t_end):
    n = len(h0)
    alpha = 500

    def obj(x):
        return _smooth_max(_overlaps(x), alpha)

    def obj_grad(x):
        return _grad_c5_smooth(x, alpha)

    def eq(x):
        return float(np.sum(x) - n / 2.0)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': eq}],
                       bounds=bounds, options={'maxiter': 300, 'ftol': 1e-12})
        if time.time() > t_end:
            return h0, _eval_c5(h0)
        h = _project_sum(res.x, n)
        return h, _eval_c5(h)
    except Exception:
        return h0, _eval_c5(h0)

def _block_init(n, pattern=None):
    if pattern is None:
        n1 = n // 2
        blocks1 = [n1 // 3 + (1 if i < n1 % 3 else 0) for i in range(3)]
        blocks0 = [n // 2 - n1 // 2 + (1 if i < (n - n1) % 3 else 0) for i in range(3)]
        h = np.zeros(n)
        pos = 0
        for i in range(3):
            h[pos:pos + blocks1[i]] = 1.0
            pos += blocks1[i]
            b0 = blocks0[min(i, len(blocks0) - 1)]
            pos += b0
        h = _project_sum(h, n)
    else:
        h = np.zeros(n)
        for start, length, val in pattern:
            h[start:min(start + length, n)] = val
        h = _project_sum(h, n)
    return h

    return _project_sum(h + rng.normal(0, scale, len(h)), len(h))
