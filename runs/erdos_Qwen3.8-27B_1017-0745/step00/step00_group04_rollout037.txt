def lp_step(h, q, G, Delta, n):

    M = 2 * n - 1
    nv = 2 * n + 1
    e = dx * Delta * Delta
    # Objective: minimize t
    c = np.zeros(nv)
    c[2*n] = 1.0

        A[row, 2*n] = -1.0
        b[row] = -q[l] - e
        row += 1

from scipy.optimize import linprog
import math

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def is_feasible(h, n):
    if len(h) != n or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    S = n / 2.0
    return abs(float(np.sum(h)) - S) <= 1e-10 and abs(math.fsum(float(x) for x in h) - S) <= 1e-10

def project_simplex(v, S):
    n = len(v)
    v = np.clip(np.asarray(v, dtype=np.float64), -1e6, 1e6)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(3):
        res = S - math.fsum(float(x) for x in h)
        if abs(res) < 1e-13:
            break
        for i in range(n):
            if abs(res) < 1e-13:
                break
            if res > 0:
                a = min(res, 1.0 - h[i]); h[i] += a; res -= a
            else:
                a = min(-res, h[i]); h[i] -= a; res += a
    return np.clip(h, 0.0, 1.0)

def build_lags(h, n):
    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    q[n-1] = dx * np.sum(h * (1.0 - h))
    G[n-1, :] = dx * (1.0 - 2.0 * h)
    for r in range(1, n):
        cnt = n - r
        idx = np.arange(cnt)
        q[n-1-r] = dx * np.sum(h[idx] * (1.0 - h[idx + r]))
        G[n-1-r, idx] += dx * (1.0 - h[idx + r])
        G[n-1-r, idx + r] -= dx * h[idx]
        q[n-1+r] = dx * np.sum(h[idx + r] * (1.0 - h[idx]))
        G[n-1+r, idx] -= dx * h[idx + r]
        G[n-1+r, idx + r] += dx * (1.0 - h[idx])
    return q, G

def lp_step(h, q, G, Delta, n):
    dx = 2.0 / n
    M = 2 * n - 1
    nv = 2 * n + 1
    e = dx * Delta * Delta
    c = np.zeros(nv); c[2*n] = 1.0
    nrow = M + 4 * n + 1
    A = np.zeros((nrow, nv)); b = np.zeros(nrow)
    row = 0
    for l in range(M):
        A[row, :n] = G[l]; A[row, 2*n] = -1.0; b[row] = -q[l] - e; row += 1
    for i in range(n):
        A[row, i] = 1.0; A[row, n+i] = -1.0; row += 1
    for i in range(n):
        A[row, i] = -1.0; A[row, n+i] = -1.0; row += 1
    for i in range(n):
        A[row, i] = 1.0; b[row] = 1.0 - h[i]; row += 1
    for i in range(n):
        A[row, i] = -1.0; b[row] = h[i]; row += 1
    A[row, n:n*2] = 1.0; b[row] = Delta; row += 1
    Ae = np.zeros((1, nv)); Ae[0, :n] = 1.0
    be = np.zeros(1)
    bounds = [(None, None)] * n + [(0, None)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A, b_ub=b, A_eq=Ae, b_eq=be, bounds=bounds,
                      method='highs', options={'maxiter': 500})
        if res.success and res.x is not None and np.all(np.isfinite(res.x)):
            return res.x[:n].copy()
    except:
        pass
    return None

def trust_search(h0, n):
