    parent = globals().get('initial_h_values', None)
    parent_h = None
    if parent is not None:
        parent = np.array(parent, dtype=np.float64)
        if parent.ndim == 1 and 2 <= len(parent) < 1000:
            n = len(parent)
            if np.all(np.isfinite(parent)) and bool(np.all(np.abs(parent) <= 1e6)):
                parent_h = parent.copy()
    if n is None:
        nk = kwargs.get('n_points', None)
        if isinstance(nk, int) and 2 <= nk <= 999:
            n = nk
    if n is None:
        n = 45
    S = n / 2.0
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    if parent_h is not None:
        fb = project_K(parent_h, n, S)
        if fb is None:
            fb = np.full(n, 0.5)
    else:
        fb = np.full(n, 0.5)
    fb_score, _ = score(fb, n)
    best_h, best_score = fb.copy(), fb_score
    starts = []
    if parent_h is not None:
        starts.append(parent_h.copy())
        starts.append(parent_h[::-1].copy())
        starts.append((1.0 - parent_h).copy())
        starts.append((1.0 - parent_h[::-1]).copy())
        for sd in [0.08, 0.25]:
            p = project_K(parent_h + sd * rng.standard_normal(n), n, S)
            if p is not None:
                starts.append(p)
        p = project_K(rng.random(n), n, S)
        if p is not None:
            starts.append(p)
        p = project_K(rng.beta(0.5, 0.5, n), n, S)
        if p is not None:
            starts.append(p)
    else:
        starts.append(np.full(n, 0.5))
        for _ in range(3):
            p = project_K(rng.random(n), n, S)
            if p is not None:
                starts.append(p)
        for _ in range(2):
            p = project_K(rng.beta(0.5, 0.5, n), n, S)
            if p is not None:
                starts.append(p)
        p = project_K(rng.beta(2.0, 2.0, n), n, S)
        if p is not None:
            starts.append(p)
        m = int(rng.integers(1, max(2, min(5, n // 2 + 1))))
        ph = float(rng.uniform(0.0, 2.0 * math.pi))
        cos_v = 0.5 + 0.45 * np.cos(2.0 * math.pi * m * (np.arange(n) + 0.5) / n + ph)
        p = project_K(cos_v, n, S)
        if p is not None:
            starts.append(p)
    starts = [s for s in starts if feasible(s, n, S)]
    dedup = []
    for s in starts:
        if not any(float(np.max(np.abs(s - d))) <= 1e-6 for d in dedup):
            dedup.append(s)
    max_chains = 8 if n <= 64 else (6 if n <= 256 else 4)
    starts = dedup[:max_chains]
    try:
        for s in starts:
            bh, bs = metropolis_chain(s, n, S, rng)
            if bs < best_score - 1e-12:
                best_h, best_score = bh.copy(), bs
        nrestart = 2 if n <= 64 else 1
        sigmas = [0.08, 0.25] if n <= 64 else [0.15]
        for sd in sigmas[:nrestart]:
            rs = project_K(best_h + sd * rng.standard_normal(n), n, S)
            if rs is not None:
                bh, bs = metropolis_chain(rs, n, S, rng)
                if bs < best_score - 1e-12:
                    best_h, best_score = bh.copy(), bs
    except Exception:
        pass
    try:
        cands = [best_h.copy(), fb.copy()]
        uniq = []
        for c in cands:
            if not any(float(np.max(np.abs(c - d))) <= 1e-5 for d in uniq):
                uniq.append(c)
        scored = []
        for c in uniq:
            sc, _ = score(c, n)
            if np.isfinite(sc):
                scored.append((sc, c))
        scored.sort(key=_sort_key)
        nkeep = 3 if n <= 64 else (2 if n <= 256 else 1)
        for sv in scored[:nkeep]:
            bh, bs = local_refine(sv[1], n, S, rng)
            if bs < best_score - 1e-12:
                best_h, best_score = bh.copy(), bs
    except Exception:
        pass
    final = np.array(best_h, dtype=np.float64, copy=True)
    final_sc, _ = score(final, n)
    if not feasible(final, n, S) or not np.isfinite(final_sc) or final_sc >= fb_score:
        final = fb.copy()
        final_sc, _ = score(final, n)
    if not feasible(final, n, S):
        final = np.full(n, 0.5)
        final_sc, _ = score(final, n)
    return final.copy(), float(final_sc), int(n)
