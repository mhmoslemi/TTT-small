- ∂/∂h[i] of h[j]*(1-h[i]) = -h[j] (the (1-h[k]) part)
- ∂/∂h[j] of h[j]*(1-h[i]) = (1-h[i]) (the h[k+d] part)

import numpy as np
import math

def exact_score(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def is_feasible(h, n):
    S = n / 2.0
    if h.ndim != 1 or len(h) != n or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return False
    if abs(np.sum(h) - S) > 1e-10 or abs(math.fsum(float(x) for x in h) - S) > 1e-10:
        return False
    return True

def capped_simplex(v, n):
    S = n / 2.0
    v = np.clip(np.asarray(v, dtype=np.float64), -1e6, 1e6)
    lo_l = float(np.min(v)) - 1.0
    hi_l = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo_l + hi_l)
        t = np.clip(v - mid, 0.0, 1.0)
        if float(np.sum(t)) > S:
            lo_l = mid
        else:
            hi_l = mid
    h = np.clip(v - hi_l, 0.0, 1.0)
    res = S - math.fsum(float(x) for x in h)
    if res > 1e-14:
        for i in range(n):
            a = min(1.0 - h[i], res)
            h[i] += a; res -= a
            if res <= 1e-14: break
    elif res < -1e-14:
        for i in range(n):
            a = min(h[i], -res)
            h[i] -= a; res += a
            if res >= -1e-14: break
    h = np.clip(h, 0.0, 1.0)
    return h if is_feasible(h, n) else None

def grad_q(h, n, m):
    dx = 2.0 / n
    g = np.zeros(2 * n - 1)
    g[0] = dx * (1.0 - 2.0 * h[m])
    for d in range(1, n):
        cnt = n - d
        # q[2d-1]
        if m >= d:
            g[2*d-1] += dx * (1.0 - h[m-d])
        if m < cnt:
            g[2*d-1] += -dx * h[m+d]
        # q[2d]
        if m < cnt:
            g[2*d] += dx * (1.0 - h[m+d])
        if m >= d:
            g[2*d] += -dx * h[m-d]
    return g

def real_roots(A, B, C, lo, hi):
    roots = []
    tol = 1e-14 * max(1.0, abs(A), abs(B), abs(C))
    if abs(A) <= tol:
        if abs(B) > tol:
            r = -C / B
            if lo - 1e-10 <= r <= hi + 1e-10:
                roots.append(max(lo, min(hi, r)))
        return roots
    disc = B*B - 4.0*A*C
    eps = np.finfo(np.float64).eps
    if disc < -64.0 * eps * (B*B + abs(4.0*A*C) + 1.0):
        return roots
    if disc < 0:
        disc = 0.0
    sq = math.sqrt(disc)
    sign_B = 1.0 if B >= 0 else -1.0
    z = -0.5 * (B + sign_B * sq)
    if abs(z) > 1e-300:
        r1, r2 = z / A, C / z
    else:
        r1 = r2 = -B / (2.0 * A)
    for r in (r1, r2):
        if np.isfinite(r) and lo - 1e-10 <= r <= hi + 1e-10:
            roots.append(max(lo, min(hi, r)))
    return roots

def build_affine_hull(slopes, ints, lo, hi):
    # Returns list of (x_start, x_end, slope, intercept)
    n = len(slopes)
    if n == 0:
        return []
    order = np.lexsort((-ints, slopes))
    ds, dc = [], []
    for idx in order:
        s, c = float(slopes[idx]), float(ints[idx])
        if not np.isfinite(s) or not np.isfinite(c):
            continue
        if ds and abs(s - ds[-1]) < 1e-14 * max(1.0, abs(s)):
            if c > dc[-1]: dc[-1] = c
        else:
            ds.append(s); dc.append(c)
    hull_s, hull_c, hull_x = [], [], []
    for k in range(len(ds)):
        s, c = ds[k], dc[k]
        xc = -1e18
        while hull_s:
            sp, cp = hull_s[-1], hull_c[-1]
            if abs(s - sp) < 1e-15: break
            xc = (cp - c) / (s - sp)
            if xc <= hull_x[-1] + 1e-15:
                hull_s.pop(); hull_c.pop(); hull_x.pop()
            else: break
        if not hull_s:
            hull_s.append(s); hull_c.append(c); hull_x.append(-1e18)
        else:
            if abs(s - sp) < 1e-15: break
            hull_s.append(s); hull_c.append(c); hull_x.append(xc)
    segs = []
    for k in range(len(hull_s)):
        x1 = max(hull_x[k], lo)
        x2 = min(hull_x[k+1] if k+1 < len(hull_s) else 1e18, hi)
        if x2 - x1 > 1e-15:
            segs.append((x1, x2, hull_s[k], hull_c[k]))
    return segs

def pair_line_search(h, i, j, n, cur_score):
    if j - i <= 0: return None
    s = j - i
    m = 2 * n - 1
    dx = 2.0 / n
    q = np.zeros(m)
    q[0] = dx * np.sum(h * (1.0 - h))
    for d in range(1, n):
        cnt = n - d
        q[2*d-1] = dx * np.sum(h[d:d+cnt] * (1.0 - h[0:cnt]))
        q[2*d] = dx * np.sum(h[0:cnt] * (1.0 - h[d:d+cnt]))
    gi = grad_q(h, n, i)
    gj = grad_q(h, n, j)
    b = gi - gj
    a = np.zeros(m)
    a[0] = -2.0 * dx
    a[2*s-1] = dx
    a[2*s] = dx
    c = q  # c[l] = q[l] at t=0
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-14:
        return None
    # Affine indices
    quad_set = {0, 2*s-1, 2*s}
    aff_idx = [l for l in range(m) if l not in quad_set]
    aff_slopes = b[aff_idx]
    aff_ints = c[aff_idx]
    segs = build_affine_hull(aff_slopes, aff_ints, lo, hi)
    # Candidate t values
    cands = [lo, hi, 0.0]
    for (xs, xe, sl, ic) in segs:
        cands.append(max(xs, lo))
    # Quadratic vs affine segment roots
    for qi in [0, 2*s-1, 2*s]:
        for (xs, xe, sl, ic) in segs:
            A = a[qi]
            B = b[qi] - sl
            C = c[qi] - ic
            cands.extend(real_roots(A, B, C, max(xs, lo), min(xe, hi)))
    # Quadratic vs quadratic roots
    for q1 in [0, 2*s-1, 2*s]:
        for q2 in [2*s-1, 2*s]:
            if q1 >= q2: continue
            A = a[q1] - a[q2]
            B = b[q1] - b[q2]
            C = c[q1] - c[q2]
            cands.extend(real_roots(A, B, C, lo, hi))
    # Vertices of convex quadratics
    for qi in [2*s-1, 2*s]:  # a > 0 for these
        if a[qi] > 0:
            tv = -b[qi] / (2.0 * a[qi])
            if lo - 1e-10 <= tv <= hi + 1e-10:
                cands.append(max(lo, min(hi, tv)))
    # Deduplicate
    cands = [max(lo, min(hi, t)) for t in cands if np.isfinite(t)]
    cands = sorted(set(round(t, 12) for t in cands))  # hmm this loses precision
    # Better: numpy unique with tolerance
    if len(cands) > 1:
        cands_arr = np.array(cands)
        _, unique_idx = np.unique(np.round(cands_arr, 12), return_index=True)
        cands = sorted(cands_arr[unique_idx].tolist())
    # Evaluate predicted max
    best_trials = []
    for t in cands:
        qval = a * t * t + b * t + c
        pred = float(np.max(qval))
        best_trials.append((pred, t))
    best_trials.sort(key=lambda x: x[0])
    # Score top 6
    best_h = h
    best_score = cur_score
    for pred, t in best_trials[:6]:
        ps = h[i] + h[j]
        hi_new = h[i] + t
        hi_new = max(0.0, min(1.0, hi_new, ps))
        hj_new = ps - hi_new
        h_new = h.copy()
        h_new[i] = hi_new
        h_new[j] = hj_new
        if not is_feasible(h_new, n):
            h_new = capped_simplex(h_new, n)
            if h_new is None: continue
        sc = exact_score(h_new, n)
        if sc < best_score - 1e-12:
            best_score = sc
            best_h = h_new
    if best_score < cur_score - 1e-12:
        return best_h
    return None

def build_starts(n, rng, parent):
    S = n / 2.0
    starts = []
    # Constant
    starts.append(np.full(n, 0.5))
    # Block
    blk = np.zeros(n)
    k = n // 2
    blk[:k] = 1.0
    if n % 2 == 1: blk[k] = 0.5
    starts.append(blk)
    # Alternating
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    alt = capped_simplex(alt, n)
    if alt is not None: starts.append(alt)
    # Sinusoid
    idx = np.arange(n)
    sinv = 0.5 + 0.45 * np.sin(2.0 * np.pi * idx / n)
    sinv = capped_simplex(sinv, n)
    if sinv is not None: starts.append(sinv)
    # Parent
    if parent is not None:
        starts.insert(0, parent.copy())
        # Perturbations
        p1 = parent + rng.standard_normal(n) * 0.12
        p1 = capped_simplex(p1, n)
        if p1 is not None: starts.append(p1)
        p2 = parent + rng.standard_normal(n) * 0.35
        p2 = capped_simplex(p2, n)
        if p2 is not None: starts.append(p2)
        # Uniform random
        u = rng.random(n)
        u = capped_simplex(u, n)
        if u is not None: starts.append(u)
    else:
        for _ in range(4):
            u = rng.random(n)
            u = capped_simplex(u, n)
            if u is not None: starts.append(u)
    # Deduplicate
    unique = []
    for v in starts:
        if v is None: continue
        dup = False
        for w in unique:
            if np.max(np.abs(v - w)) < 1e-10:
                dup = True; break
        if not dup:
            unique.append(v)
    return unique
