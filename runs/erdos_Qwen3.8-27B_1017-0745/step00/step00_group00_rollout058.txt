def project_box_sum(v, S):
    n = len(v)
    v = np.clip(v, -np.inf, np.inf)
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-15:
        if resid > 0:
            for i in range(n):
                if resid <= 0: break
                room = 1.0 - y[i]
                add = min(room, resid)
                y[i] += add
                resid -= add
        else:
            for i in range(n):
                if resid >= 0: break
                room = y[i]
                sub = min(room, -resid)
                y[i] -= sub
                resid += sub
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1+1e-12):
        return None
    if abs(np.sum(y) - S) > 1e-10 * max(1, n):
        return None
    return y

def exact_score(h):

            sigma = min(0.25, 1.5 * sigma)

    final = project_box_sum(best, n / 2)
    if final is not None:
        s = exact_score(final)
        if np.isfinite(s):
            best = final
            best_score = s
    else:
        best = parent.copy()
        best_score = exact_score(best)
    if not np.isfinite(best_score):
        best = np.full(n, 0.5)
        best_score = exact_score(best)
    return (best, best_score, n)
