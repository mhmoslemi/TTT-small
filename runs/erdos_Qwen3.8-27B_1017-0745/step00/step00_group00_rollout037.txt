import numpy as np
from math import sqrt, copysign

def project_box_sum(v, S):
    n = len(v)
    lo_v = float(np.min(v)) - 1.0
    hi_v = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo_v + hi_v)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo_v = mid
        else:
            hi_v = mid
    y = np.clip(v - 0.5 * (lo_v + hi_v), 0.0, 1.0)
    res = S - np.sum(y)
    for i in range(n):
        if res > 0:
            d = min(res, 1.0 - y[i]); y[i] += d; res -= d
        elif res < 0:
            d = min(-res, y[i]); y[i] -= d; res += d
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1 + 1e-12) or abs(np.sum(y) - S) > 1e-10 * max(1, n):
        return None
    return y

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1 + 1e-12) or abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score_and_corr(h):
    n = len(h)
    if not is_feasible(h, n):
        return None, None
    corr = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(corr)):
        return None, None
    sc = float(np.max(corr) * 2.0 / n)
    if not np.isfinite(sc):
        return None, None
    return sc, corr

def quad_roots(a, b, c):
    if abs(a) < 1e-15:
        if abs(b) < 1e-15:
            return []
        r = -c / b
        return [r] if np.isfinite(r) else []
    disc = b * b - 4.0 * a * c
    th = 64.0 * np.finfo(float).eps * max(1.0, abs(b * b), abs(4.0 * a * c))
    if disc < -th:
        return []
    if disc < 0:
        disc = 0.0
    sd = sqrt(disc)
    q = -0.5 * (b + copysign(sd, b))
    out = []
    if abs(q) > 1e-15:
        r1, r2 = q / a, c / q
        if np.isfinite(r1): out.append(r1)
        if np.isfinite(r2): out.append(r2)
    else:
        r = -b / (2.0 * a)
        if np.isfinite(r): out.append(r)
    return out

def make_pair_coeffs(h, corr, n, a, b):
    A = corr.copy()
    B = np.zeros(2 * n - 1)
    D = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        bk = 0.0
        if 0 <= a - k < n: bk += 1.0 - h[a - k]
        if 0 <= b - k < n: bk -= 1.0 - h[b - k]
        if 0 <= a + k < n: bk -= h[a + k]
        if 0 <= b + k < n: bk += h[b + k]
        B[k + n - 1] = bk
    D[n - 1] = -2.0
    D[(a - b) + n - 1] = 1.0
    D[(b - a) + n - 1] = 1.0
    if not (np.all(np.isfinite(A)) and np.all(np.isfinite(B)) and np.all(np.isfinite(D))):
        return None
    return A, B, D

def build_hull(A, B, curved, lo, hi):
    n = len(A)
    ld = []
    for k in range(n):
        if k in curved:
            continue
        ld.append((B[k], -A[k], A[k]))
    if not ld:
        return [(lo, hi, None, None)]
    ld.sort()
    hull = []
    for m_ne, neg_c, c in ld:
        m, c_val = m_ne, c
        if not hull:
            hull.append([m, c_val, -1e18])
            continue
        x_cr = (hull[-1][1] - c_val) / (m - hull[-1][0])
        while hull and x_cr <= hull[-1][2] + 1e-15:
            hull.pop()
            if not hull:
                x_cr = -1e18
                break
            x_cr = (hull[-1][1] - c_val) / (m - hull[-1][0])
        hull.append([m, c_val, x_cr])
    segs = []
    for i in range(len(hull)):
        s = max(hull[i][2], lo)
        e = min(hull[i + 1][2], hi) if i + 1 < len(hull) else hi
        if e > s:
            segs.append((s, e, hull[i][0], hull[i][1]))
    return segs

def line_min_pair(h, corr, cur_sc, a, b, budget):
    n = len(h)
    lo = max(-h[a], h[b] - 1.0)
    hi = min(1.0 - h[a], h[b])
    if hi - lo < 1e-14:
        return None
    cf = make_pair_coeffs(h, corr, n, a, b)
    if cf is None:
        return None
    A, B, D = cf
    curved = {a - b + n - 1, b - a + n - 1, n - 1}
    segs = build_hull(A, B, curved, lo, hi)
    if not segs:
        return None
    cl = list(curved)
    cands = []
    for (s, e, m, c) in segs:
        cands.append(s)
        cands.append(e)
        for ki in curved:
            if abs(D[ki]) > 1e-15:
                st = -B[ki] / (2.0 * D[ki])
                if s - 1e-12 <= st <= e + 1e-12:
                    cands.append(max(s, min(e, st)))
        for ii in range(len(cl)):
            for jj in range(ii + 1, len(cl)):
                i1, i2 = cl[ii], cl[jj]
                for r in quad_roots(D[i1] - D[i2], B[i1] - B[i2], A[i1] - A[i2]):
                    if s - 1e-12 <= r <= e + 1e-12:
                        cands.append(max(s, min(e, r)))
        if m is not None:
            for ki in curved:
                for r in quad_roots(D[ki], B[ki] - m, A[ki] - c):
                    if s - 1e-12 <= r <= e + 1e-12:
                        cands.append(max(s, min(e, r)))
    cands.append(0.0)
    cands = [max(lo, min(hi, c)) for c in cands if lo - 1e-12 <= c <= hi + 1e-12]
    uniq = []
    for c in cands:
        if not any(abs(c - u) < 1e-12 * max(1.0, abs(c)) for u in uniq):
            uniq.append(c)
    scored = []
    for alpha in uniq:
        if abs(alpha) < 1e-14:
            continue
        val = None
        for (s, e, m, c) in segs:
            if s - 1e-14 <= alpha <= e + 1e-14:
                v = (c + m * alpha) if m is not None else -1e18
                for ki in curved:
                    q = A[ki] + B[ki] * alpha + D[ki] * alpha * alpha
                    if q > v:
                        v = q
                val = v
                break
        if val is not None and val > -1e17:
            scored.append((val, alpha))
    scored.sort()
    best_h, best_sc, best_cr = None, cur_sc, corr
    for i in range(min(5, len(scored))):
        if budget[0] <= 0:
            break
        budget[0] -= 1
        alpha = scored[i][1]
        if abs(alpha) < 1e-14:
            continue
        nh = h.copy()
        nh[a] = max(0.0, min(1.0, nh[a] + alpha))
        nh[b] = max(0.0, min(1.0, nh[b] - alpha))
        if not is_feasible(nh, n):
            continue
        sc, cr = exact_score_and_corr(nh)
        if sc is None:
            continue
        if sc < best_sc - 1e-12 * max(1.0, abs(best_sc)):
            best_sc, best_cr, best_h = sc, cr, nh
    if best_h is None:
        return None
    return best_h, best_cr, best_sc
