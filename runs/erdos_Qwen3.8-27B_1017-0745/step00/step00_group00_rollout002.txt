import numpy as np
from scipy.optimize import minimize

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    residual = S - float(np.sum(y))
    if abs(residual) > 1e-14:
        if residual > 0:
            for i in range(n):
                cap = 1.0 - y[i]
                add = min(residual, cap)
                y[i] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            for i in range(n):
                cap = y[i]
                rem = min(-residual, cap)
                y[i] -= rem
                residual += rem
                if abs(residual) < 1e-14:
                    break
    return y

def check_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    tol = 1e-10 * max(1, n)
    return abs(float(np.sum(h)) - n / 2.0) <= tol

def exact_score(h):
    n = len(h)
    if not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def lag_values_gradients(h):
    n = len(h)
    num_lags = 2 * n - 1
    Q = np.zeros(num_lags)
    G = np.zeros((num_lags, n))
    for idx in range(num_lags):
        k = idx - (n - 1)
        i_start = max(0, k)
        i_end = min(n, n + k)
        for i in range(i_start, i_end):
            j = i - k
            Q[idx] += h[i] * (1.0 - h[j])
            G[idx, i] += (1.0 - h[j])
            G[idx, j] -= h[i]
    return Q, G

def obj_func(z):
    return z[-1]

def obj_jac(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def ineq_func(z):
    h = z[:-1]
    t = z[-1]
    n = len(h)
    Q, _ = lag_values_gradients(h)
    return t - Q / n

def ineq_jac(z):
    h = z[:-1]
    n = len(h)
    _, G = lag_values_gradients(h)
    num_lags = G.shape[0]
    J = np.zeros((num_lags, n + 1))
    J[:, :n] = -G / n
    J[:, n] = 1.0
    return J

def eq_func(z):
    return float(np.sum(z[:-1])) / len(z[:-1]) - 0.5

def eq_jac(z):
    n = len(z) - 1
    J = np.zeros(n + 1)
    J[:n] = 1.0 / n
    return J

def get_parent():
    hv = globals().get('initial_h_values', None)
    if hv is None:
        return None
    hv = np.asarray(hv, dtype=np.float64)
    if hv.ndim != 1 or len(hv) < 2 or len(hv) > 999:
        return None
    if not np.all(np.isfinite(hv)):
        return None
    if np.any(hv < -1e-6) or np.any(hv > 1.0 + 1e-6):
        return None
    n = len(hv)
    if abs(float(np.sum(hv)) - n / 2.0) > 1e-6 * max(1, n):
        return None
    hv = project_box_sum(hv, n / 2.0)
    if not check_feasible(hv, n):
        return None
    return hv

def build_starts(parent, n, rng):
    starts = []
    if parent is not None:
        starts.append(parent.copy())
        starts.append(parent[::-1].copy())
        for sigma in [0.01, 0.03, 0.06, 0.12, 0.20, 0.05]:
            p = project_box_sum(parent + sigma * rng.standard_normal(n), n / 2.0)
            starts.append(p)
        for _ in range(4):
            u = rng.random(n)
            p = project_box_sum(u, n / 2.0)
            starts.append(p)
    else:
        starts.append(np.full(n, 0.5))
        for _ in range(8):
            u = rng.random(n)
            starts.append(project_box_sum(u, n / 2.0))
        params = [(0.25, 1, 0.37), (0.35, 2, 0.74), (0.45, 3, 1.11)]
        for A, m, ph in params:
            idx = np.arange(n, dtype=np.float64)
            u = 0.5 + A * np.cos(2.0 * np.pi * m * idx / n + ph)
            starts.append(project_box_sum(u, n / 2.0))
    return [s for s in starts if check_feasible(s, n)]

def optimize_from(h0, n):
    parent_n = len(h0)
    Q0, _ = lag_values_gradients(h0)
    t0 = min(1.0, float(np.max(Q0)) / parent_n + 1e-4)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'ineq', 'fun': ineq_func, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_func, 'jac': eq_jac},
    ]
    try:
        res = minimize(obj_func, z0, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-10, 'disp': False})
    except Exception:
        return None
    if res is None or not hasattr(res, 'x'):
        return None
    h_r = np.asarray(res.x[:n], dtype=np.float64)
    if not np.all(np.isfinite(h_r)):
        return None
    h_p = project_box_sum(h_r, n / 2.0)
    if not check_feasible(h_p, n):
        return None
    return h_p

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)

    parent = get_parent()
    if parent is not None:
        n = len(parent)
        best = parent.copy()
        best_score = exact_score(best)
    else:
        n = 45
        best = np.full(n, 0.5)
        best_score = exact_score(best)

    n_half = n / 2.0
    starts = build_starts(parent, n, rng)

    for h0 in starts:
        hr = optimize_from(h0, n)
        if hr is None:
            continue
        s = exact_score(hr)
        if s < best_score - 1e-15:
            best = hr.copy()
            best_score = s

    sigma = 0.05
    for _ in range(4):
        h0 = project_box_sum(best + sigma * rng.standard_normal(n), n_half)
        if not check_feasible(h0, n):
            break
        hr = optimize_from(h0, n)
        improved = False
        if hr is not None:
            s = exact_score(hr)
            if s < best_score - 1e-12:
                best = hr.copy()
                best_score = s
                improved = True
        if improved:
            sigma = max(0.01, 0.7 * sigma)
        else:
            sigma = min(0.25, 1.5 * sigma)

    best = project_box_sum(best, n_half)
    if check_feasible(best, n):
        c5 = exact_score(best)
    else:
        best = np.full(n, 0.5)
        c5 = exact_score(best)

    return best, c5, n
