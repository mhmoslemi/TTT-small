from scipy.optimize import minimize
import math

def capped_simplex_project(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or z.size < 2:
        return None
    n = z.size
    if not np.all(np.isfinite(z)):
        return None
    if np.max(np.abs(z)) > 1e12:
        return None

    high = np.max(z)
    for _ in range(80):
        mid = low + 0.5 * (high - low)
        y = np.clip(z - mid, 0.0, 1.0)
        if np.sum(y) > S:
            low = mid
        else:
            high = mid
    y = np.clip(z - (low + 0.5 * (high - low)), 0.0, 1.0)
    r = S - np.sum(y)
    if r > 0:
        for i in range(n):
            if r <= 0:
                break
            slack = 1.0 - y[i]
            if slack > 1e-15:
                add = min(r, slack)
                y[i] += add
                r -= add
    elif r < 0:
        for i in range(n):
            if r >= 0:
                break
            if y[i] > 1e-15:
                sub = min(-r, y[i])
                y[i] -= sub
                r += sub

        for i in range(n):
            if r > 0 and 1.0 - y[i] > 1e-15:
                y[i] = min(1.0, y[i] + r)
                break
            elif r < 0 and y[i] > 1e-15:
                y[i] = max(0.0, y[i] + r)
                break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    y = np.clip(y, 0.0, 1.0)
    if abs(np.sum(y) - S) > 1e-10:
        return None
    return y

def is_feasible(h, n):
    if h is None or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10:
        return False
    return True

def compute_lags(h):
    n = len(h)
    nlags = 2 * n - 1
    B = np.zeros(nlags, dtype=np.float64)
    G = np.zeros((nlags, n), dtype=np.float64)
    for d in range(-(n - 1), n):
        idx = d + (n - 1)
        lo = max(0, -d)
        hi = min(n, n - d)
        src = h[lo:hi]
        dst = h[lo + d:hi + d]
        B[idx] = np.dot(src, 1.0 - dst)
        G[idx, lo:hi] += (1.0 - dst)
        G[idx, lo + d:hi + d] -= src
    return B, G

def exact_score(h):
    h = np.asarray(h, dtype=np.float64)
    if len(h) < 2 or not np.all(np.isfinite(h)):
        return None
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    if not np.all(np.isfinite(vals)):
        return None
    return float(np.max(vals))

def nlp_obj(z):
    return z[-1]

def nlp_obj_grad(z):
    n = len(z) - 1
    g = np.zeros(len(z), dtype=np.float64)
    g[-1] = 1.0
    return g

def nlp_ineq(z):
    n = len(z) - 1
    h = z[:n]
    t = z[-1]
    B, _ = compute_lags(h)
    return t - B

def nlp_ineq_jac(z):
    n = len(z) - 1
    h = z[:n]
    _, G = compute_lags(h)
    J = np.zeros((2 * n - 1, n + 1), dtype=np.float64)
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def nlp_eq(z):
    n = len(z) - 1
    return np.sum(z[:n]) - n / 2.0

def nlp_eq_jac(z):
    n = len(z) - 1
    J = np.zeros(n + 1, dtype=np.float64)
    J[:n] = 1.0
    return J

def make_periodic(n, rng):
    maxf = max(1, min(6, (n - 1) // 2))
    q = rng.integers(1, maxf + 1)
    phi = rng.uniform(0.0, 2.0 * math.pi)
    i = np.arange(n, dtype=np.float64)
    v = 0.5 + 0.45 * np.cos(2.0 * math.pi * q * i / n + phi)
    return v

def make_binary(n, rng):
    v = np.zeros(n, dtype=np.float64)
    perm = rng.permutation(n)
    half = n // 2
    v[perm[:half]] = 1.0
    if n % 2 == 1:
        v[perm[half]] = 0.5
    return v

def dedup_check(v, existing, tol=1e-10):
    for e in existing:
        if np.max(np.abs(v - e)) < tol:
            return True
    return False
