def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M)
    Gm = np.zeros((M, n))
    q[0] = dx * np.sum(h * (1.0 - h))
    Gm[0] = dx * (1.0 - 2.0 * h)
    idx = 1
    for r in range(1, n):
        for i in range(n - r):
            q[idx] += dx * h[i+r] * (1.0 - h[i])
            Gm[idx, i] += -dx * h[i+r]
            Gm[idx, i+r] += dx * (1.0 - h[i])
            q[idx+1] += dx * h[i] * (1.0 - h[i+r])
            Gm[idx+1, i] += dx * (1.0 - h[i+r])
            Gm[idx+1, i+r] += -dx * h[i]
        idx += 2

import numpy as np

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M)
    Gm = np.zeros((M, n))
    q[0] = dx * np.sum(h * (1.0 - h))
    Gm[0] = dx * (1.0 - 2.0 * h)
    idx = 1
    for r in range(1, n):
        for i in range(n - r):
            q[idx] += dx * h[i + r] * (1.0 - h[i])
            Gm[idx, i] += -dx * h[i + r]
            Gm[idx, i + r] += dx * (1.0 - h[i])
            q[idx + 1] += dx * h[i] * (1.0 - h[i + r])
            Gm[idx + 1, i] += dx * (1.0 - h[i + r])
            Gm[idx + 1, i + r] += -dx * h[i]
        idx += 2
    if not (np.all(np.isfinite(q)) and np.all(np.isfinite(Gm))):
        return None, None
    return q, Gm

def lp_step(h, q, Gm, delta):
    n = len(h)

    M = 2 * n - 1
    e = dx * delta * delta

    nr = M + 4 * n + 1

    b_ub = np.zeros(nr)

    for i in range(n):

    for i in range(n):
        A_ub[rr, i] = 1.0; b_ub[rr] = 1.0 - h[i]; rr += 1
    for i in range(n):
        A_ub[rr, i] = -1.0; b_ub[rr] = h[i]; rr += 1
