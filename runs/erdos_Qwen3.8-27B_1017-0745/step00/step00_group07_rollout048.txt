        if ad[k] > 0:
            v = -bd[k]/(2*ad[k])
            if tlo-1e-10 <= v <= thi+1e-10: cands.append(max(tlo, min(thi, v)))
    bf, bt = inf, 0.0
    for t in cands:
        t = max(tlo, min(thi, t))
        f = max(ad[k]*t*t + bd[k]*t + cd[k] for k in qi)
        if hs:
            j = min(max(int(np.searchsorted(hst, t, side='right'))-1, 0), len(hs)-1)
            f = max(f, hs[j]*t + hc[j])
        if f < bf - 1e-15 or (abs(f-bf) <= 1e-15 and abs(t) < abs(bt)):
            bf, bt = f, t
    return bt if np.isfinite(bf) else None

def _add_start(starts, h, n, S):
    if h is None or not _feasible(h, n, S): return
    for e in starts:
        if np.max(np.abs(h - e)) < 1e-10: return
    starts.append(h.copy())

def _make_pairs(n, offsets):
    sp = set()
    for off in offsets:
        for i in range(n):
            a, b = i, (i+off) % n
            if a != b: sp.add((min(a,b), max(a,b)))
    return sorted(sp)

def run(seed=42, **kwargs):
    n = 80; parent = None
    pv = globals().get('initial_h_values')
    if pv is not None:
        pv = np.asarray(pv, dtype=np.float64)
        if pv.ndim == 1 and len(pv) >= 2:
            n = len(pv)
            if np.all(np.isfinite(pv)): parent = pv.copy()
    S = n / 2.0
    h_const = np.full(n, 0.5)
    fallbacks = [h_const.copy()]
    if parent is not None:
        if _feasible(parent, n, S): fallbacks.insert(0, parent.copy())
        else:
            pr = _proj(parent, S)
            if pr is not None: fallbacks.insert(0, pr)
    inc_h, inc_s = h_const.copy(), _score(h_const, n)
    for fb in fallbacks:
        s = _score(fb, n)
        if s is not None and s < inc_s: inc_h, inc_s = fb.copy(), s
    if n > 128 or n < 2: return (inc_h, inc_s, n)
    try: rng = np.random.default_rng(int(seed))
    except: rng = np.random.default_rng(42)
    starts = []
    if parent is not None:
        _add_start(starts, parent, n, S)
        _add_start(starts, parent[::-1].copy(), n, S)
    _add_start(starts, h_const, n, S)
    if parent is not None:
        _add_start(starts, _proj(parent + 0.12*rng.standard_normal(n), S), n, S)
    qp = int(rng.integers(1, min(8, max(1, n//2))+1))
    ph = float(rng.uniform(0, 6.2832))
    _add_start(starts, _proj(0.5+0.45*np.cos(2*3.14159265*qp*np.arange(n)/n+ph), S), n, S)
    perm = rng.permutation(n)
    hb = np.zeros(n); hb[perm[:n//2]] = 1.0
    if n % 2 == 1: hb[perm[n//2]] = 0.5
    _add_start(starts, _proj(hb, S), n, S)
    if len(starts) < 6 and parent is None:
        qp2 = int(rng.integers(1, min(8, max(1, n//2))+1))
        ph2 = float(rng.uniform(0, 6.2832))
        _add_start(starts, _proj(0.5+0.45*np.cos(2*3.14159265*qp2*np.arange(n)/n+ph2), S), n, S)
        _add_start(starts, _proj(0.5+0.30*rng.standard_normal(n), S), n, S)
        _add_start(starts, _proj(rng.uniform(0, 1, n), S), n, S)
    starts = starts[:6]
    for st in starts:
        s = _score(st, n)
        if s is not None and s < inc_s: inc_h, inc_s = st.copy(), s
    offsets = []
    d = 1
    while d <= n//2:
        offsets.append(d); d *= 2
    if n//2 not in offsets: offsets.append(n//2)
    sparse_pairs = np.array(_make_pairs(n, offsets))
    full_pairs = np.array([(i,j) for i in range(n) for j in range(i+1, n)])
    Btot = int(min(8000, max(500, 64000000//(n*n))))
    do_full = len(full_pairs) <= Btot
    B_spar = Btot - len(full_pairs) if do_full else Btot
    tc = 0
    for st in starts:
        hc = st.copy(); sc = _score(hc, n)
        if sc is None: continue
        rc = _raw(hc, n); stag = 0
        for sw in range(3):
            if stag >= 2: break
            pid = rng.permutation(len(sparse_pairs))
            acc = False
            for ii in pid:
                if tc >= B_spar: break
                tc += 1
                p, q = int(sparse_pairs[ii, 0]), int(sparse_pairs[ii, 1])
                t = _lmin(hc, rc, n, p, q)
                if t is None or abs(t) < 1e-14: continue
                ht = hc.copy(); ps = hc[p] + hc[q]
                pn = float(np.clip(hc[p]+t, max(0.0, ps-1.0), min(1.0, ps)))
                ht[p] = pn; ht[q] = ps - pn
                if not _feasible(ht, n, S):
                    ht = _proj(ht, S)
                    if ht is None or not _feasible(ht, n, S): continue
                st2 = _score(ht, n)
                if st2 is None: continue
                if st2 < inc_s: inc_h, inc_s = ht.copy(), st2
                if st2 < sc - 1e-13*max(1.0, abs(sc)):
                    hc = ht; sc = st2; rc = _raw(hc, n); acc = True
            if not acc: stag += 1
    if do_full:
        hf = inc_h.copy(); sf = _score(hf, n)
        if sf is not None:
            rf = _raw(hf, n)
            for ex in range(2):
                pid = rng.permutation(len(full_pairs))
                acc = False
                for ii in pid:
                    if tc >= Btot: break
                    tc += 1
                    p, q = int(full_pairs[ii, 0]), int(full_pairs[ii, 1])
                    t = _lmin(hf, rf, n, p, q)
                    if t is None or abs(t) < 1e-14: continue
                    ht = hf.copy(); ps = hf[p] + hf[q]
                    pn = float(np.clip(hf[p]+t, max(0.0, ps-1.0), min(1.0, ps)))
                    ht[p] = pn; ht[q] = ps - pn
                    if not _feasible(ht, n, S):
                        ht = _proj(ht, S)
                        if ht is None or not _feasible(ht, n, S): continue
                    st2 = _score(ht, n)
                    if st2 is None: continue
                    if st2 < inc_s: inc_h, inc_s = ht.copy(), st2
                    if st2 < sf - 1e-13*max(1.0, abs(sf)):
                        hf = ht; sf = st2; rf = _raw(hf, n); acc = True
                if not acc: break
    pool = [inc_h]
    for fb in fallbacks:
        if not any(np.array_equal(fb, x) for x in pool): pool.append(fb)
    best_h, best_s = None, inf
    for c in pool:
        c = c.copy()
        if not _feasible(c, n, S):
            c = _proj(c, S)
            if c is None: continue
        s = _score(c, n)
        if s is not None and s < best_s: best_h, best_s = c, s
    if best_h is None: best_h, best_s = h_const, _score(h_const, n)
    return (best_h, best_s, n)
