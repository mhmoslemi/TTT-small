def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(corr * dx))
    return val if math.isfinite(val) else float('inf')

def check_feasible(h, n):
    if h.ndim != 1 or len(h) != n or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return False
    S = n / 2.0
    if abs(float(np.sum(h)) - S) > 1e-10 or abs(math.fsum(float(x) for x in h) - S) > 1e-10:
        return False
    return True

def capped_simplex(v, n):
    S = n / 2.0
    v = np.clip(v.astype(np.float64), -1e6, 1e6)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - hi, 0.0, 1.0)
    res = S - math.fsum(float(x) for x in h)
    idx = 0
    while abs(res) > 1e-15 and idx < n:
        if res > 0:
            a = min(res, 1.0 - h[idx])
            h[idx] += a; res -= a
        else:
            a = min(-res, h[idx])
            h[idx] -= a; res += a

def solve_quad_roots(A, B, C, lo, hi):
    roots = []
    tol = 1e-14 * max(1.0, abs(A), abs(B), abs(C))
    if abs(A) <= tol:
        if abs(B) > tol:
            r = -C / B
            if lo - 1e-10 <= r <= hi + 1e-10:
                roots.append(max(lo, min(hi, r)))
        return roots
    d = B * B - 4.0 * A * C
    if d < -64.0 * np.finfo(np.float64).eps * (B * B + abs(4.0 * A * C) + 1.0):
        return roots
    if d < 0:
        d = 0.0
    sd = math.sqrt(d)
    z = -0.5 * (B + math.copysign(sd, B))
    if abs(z) > 1e-30:
        r1, r2 = z / A, C / z
    else:
        r1 = r2 = -B / (2.0 * A)
    for r in (r1, r2):
        if math.isfinite(r) and lo - 1e-10 <= r <= hi + 1e-10:
            roots.append(max(lo, min(hi, r)))
    return roots
