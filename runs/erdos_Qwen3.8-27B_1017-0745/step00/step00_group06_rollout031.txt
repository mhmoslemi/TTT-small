    starts = [fallback.copy()]
    starts.append(np.full(N, 0.5))
    fM = int(math.floor(M))
    rem = M - fM
    blk1 = np.zeros(N)
    blk1[:fM] = 1.0
    if rem > 0:
        blk1[fM] = rem
    starts.append(blk1)
    off = N // 3
    blk2 = np.zeros(N)
    for ii in range(fM):
        blk2[(off + ii) % N] = 1.0
    if rem > 0:
        blk2[(off + fM) % N] = rem
    starts.append(blk2)
    for k in range(1, 4):
        for ps in [0.0, math.pi / 2.0]:
            prof = 0.5 + 0.2 * np.sin(2.0 * math.pi * k * np.arange(N) / N + ps)
            proj = capped_simplex_project(prof, M)
            if proj is not None:
                starts.append(proj)
    if parent is not None:
        for sh in [N // 4, N // 2]:
            proj = capped_simplex_project(np.roll(fallback, sh), M)
            if proj is not None:
                starts.append(proj)
    for sc in [0.05, 0.12, 0.22]:
        noise = rng.standard_normal(N)
        sm = 0.5 * noise + 0.25 * np.roll(noise, 1) + 0.25 * np.roll(noise, -1)
        proj = capped_simplex_project(fallback + sc * sm, M)
        if proj is not None:
            starts.append(proj)

    unique_starts = [fallback.copy()]
    scored = []
    for s in starts[1:]:
        s = np.clip(s, 0.0, 1.0)
        dup = False
        for us in unique_starts:
            if np.max(np.abs(s - us)) < 1e-8:
                dup = True
                break
        if not dup:
            scored.append((s, exact_score(s, N)))
    scored.sort(key=lambda x: x[1])
    for s, sc in scored:
        if len(unique_starts) >= n_starts:
            break
        unique_starts.append(s.copy())
    while len(unique_starts) < n_starts:
        unique_starts.append(fallback.copy())

    for start in unique_starts:
        h = capped_simplex_project(start, M)
        if h is None:
            h = fallback.copy()
        h_score = exact_score(h, N)
        no_imp = 0
        for _sweep in range(max_sweeps):
            improved = False
            pairs = []
            seen_p = set()
            for ii in range(N):
                jj = (ii + 1) % N
                p = (min(ii, jj), max(ii, jj))
                if p not in seen_p:
                    pairs.append(p)
                    seen_p.add(p)
            att = 0
            while len(pairs) - N < n_rp and att < 300:
                ii = int(rng.integers(0, N))
                jj = int(rng.integers(0, N))
                if ii == jj:
                    att += 1
                    continue
                p = (min(ii, jj), max(ii, jj))
                if p not in seen_p:
                    pairs.append(p)
                    seen_p.add(p)
                att += 1
            perm = rng.permutation(len(pairs))
            for pm in perm:
                pi, pj = pairs[pm]
                res = search_pair(h, pi, pj, h_score, N, dx, G, nref, giter, M)
                if res is not None:
                    h = res
                    h_score = exact_score(h, N)
                    improved = True
                    if h_score < inc_score - 1e-12:
                        incumbent = h.copy()
                        inc_score = h_score
            if improved:
                no_imp = 0
            else:
                no_imp += 1
            if no_imp >= 2:
                break

    fh = capped_simplex_project(incumbent, M)
    if fh is None or not is_feasible(fh, N):
        fh = fallback
    fs = exact_score(fh, N)
    if not math.isfinite(fs):
        fh = np.full(N, 0.5)
        fs = exact_score(fh, N)
    return (fh.astype(np.float64), float(fs), N)
