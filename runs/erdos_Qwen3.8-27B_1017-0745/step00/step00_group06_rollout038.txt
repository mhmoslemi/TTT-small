                    if 0 <= i + q < N:
                        gi -= dx * h[i + q]
                    gj = 0.0
                    if 0 <= j - q < N:
                        gj += dx * (1.0 - h[j - q])
                    if 0 <= j + q < N:
                        gj -= dx * h[j + q]
                    b[qi] = gi - gj
                c = np.zeros(n_lags)
                c[N - 1] = -2.0 * dx
                dl = j - i
                c[N - 1 + dl] = dx
                c[N - 1 - dl] = dx
                if not (np.all(np.isfinite(r)) and np.all(np.isfinite(b)) and np.all(np.isfinite(c))):
                    continue
                tlo = max(-h[i], h[j] - 1.0)
                thi = min(1.0 - h[i], h[j])
                if thi - tlo < 1e-14:
                    continue
                tg = np.linspace(tlo, thi, grid_pts)
                fv = np.max(r[None, :] + b[None, :] * tg[:, None] + c[None, :] * tg[:, None], axis=1)
                bi = int(np.argmin(fv))
                cands = [tg[bi]]
                im = (fv[1:-1] <= fv[:-2]) & (fv[1:-1] <= fv[2:])
                basins = []
                inb, bmv, bmi = False, np.inf, 0
                for idx in range(1, grid_pts - 1):
                    if im[idx - 1]:
                        if not inb:
                            inb, bmv, bmi = True, fv[idx], idx
                        elif fv[idx] < bmv:
                            bmv, bmi = fv[idx], idx
                    else:
                        if inb:
                            basins.append((bmv, bmi))
                            inb = False
                if inb:
                    basins.append((bmv, bmi))
                basins.sort()
                for _, bsi in basins[:refine_ct]:
                    a2, b2_ = tg[max(0, bsi - 1)], tg[min(grid_pts - 1, bsi + 1)]
                    if b2_ > a2:
                        rt = golden_line(a2, b2_, r, b, c, gold_it)
                        if tlo <= rt <= thi:
                            cands.append(rt)
                best_c, best_sc = None, h_score
                for tc in cands:
                    if abs(tc) < 1e-14:
                        continue
                    if line_f(tc, r, b, c) >= h_score - 1e-12:
                        continue
                    y = h.copy()
                    y[i] += tc
                    y[j] -= tc
                    p = capped_simplex_project(y, N)
                    if p is None or not is_feasible(p, N) or np.max(np.abs(p - y)) > 1e-7:
                        continue
                    sc = exact_score(p, N)
                    if sc < best_sc - 1e-12:
                        best_sc, best_c = sc, p.copy()
                if best_c is not None:
                    h, h_score = best_c, best_sc
                    imp = True
                    if h_score < incumbent_score - 1e-12:
                        incumbent, incumbent_score = h.copy(), h_score
            if imp:
                empty = 0
            else:
                empty += 1
                if empty >= 2:
                    break
    final = capped_simplex_project(incumbent, N)
    if final is None or not is_feasible(final, N):
        final = fallback
    final = np.clip(final, 0.0, 1.0)
    return (final, exact_score(final, N), N)
