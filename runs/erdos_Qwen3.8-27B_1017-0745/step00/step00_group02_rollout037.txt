            if abs(A) > 1e-14 * sc:
                disc = B * B - 4.0 * A * C
                if disc < -1e-14 * (B * B + abs(4.0 * A * C) + 1e-300):
                    continue
                if disc < 0.0:
                    disc = 0.0
                sq = math.sqrt(disc)
                if B >= 0.0:
                    t1 = (-B - sq) / (2.0 * A)
                    t2 = (2.0 * C) / (-B - sq) if abs(B + sq) > 1e-300 else 0.0
                else:
                    t1 = (-B + sq) / (2.0 * A)
                    t2 = (2.0 * C) / (-B + sq) if abs(B - sq) > 1e-300 else 0.0
                for t in (t1, t2):
                    if np.isfinite(t):
                        tt = 1e-12 * max(1.0, abs(t_lo), abs(t_hi))
                        if t_lo - tt <= t <= t_hi + tt:
                            cands.append(max(t_lo, min(t_hi, t)))
            elif abs(B) > 1e-14 * sc:
                t = -C / B
                if np.isfinite(t):
                    tt = 1e-12 * max(1.0, abs(t_lo), abs(t_hi))
                    if t_lo - tt <= t <= t_hi + tt:
                        cands.append(max(t_lo, min(t_hi, t)))
    for r in range(m):
        if abs(c[r]) > 1e-300:
            t = -b[r] / (2.0 * c[r])
            if np.isfinite(t):
                tt = 1e-12 * max(1.0, abs(t_lo), abs(t_hi))
                if t_lo - tt <= t <= t_hi + tt:
                    cands.append(max(t_lo, min(t_hi, t)))
    cands.sort()
    dk = [cands[0]]
    for i in range(1, len(cands)):
        if abs(cands[i] - dk[-1]) > 1e-12 * max(1.0, abs(cands[i])):
            dk.append(cands[i])
    best_t, best_v = 0.0, float('inf')
    for st in range(0, len(dk), 256):
        chunk = np.array(dk[st:st + 256])
        vals = np.max(q[None, :] + chunk[:, None] * b[None, :] + (chunk[:, None] ** 2) * c[None, :], axis=1)
        for i in range(len(chunk)):
            if vals[i] < best_v - 1e-15:
                best_v = float(vals[i])
                best_t = float(chunk[i])
    return best_t, best_v

def grid_line_min(q, b, c, t_lo, t_hi):
    ts = np.linspace(t_lo, t_hi, 17)
    ts = np.unique(np.sort(np.append(ts, 0.0)))
    vals = np.max(q[None, :] + ts[:, None] * b[None, :] + (ts[:, None] ** 2) * c[None, :], axis=1)
    bi = int(np.argmin(vals))
    for _ in range(2):
        if bi == 0:
            bl, br = ts[0], ts[1]
        elif bi == len(ts) - 1:
            bl, br = ts[-2], ts[-1]
        else:
            bl, br = ts[bi - 1], ts[bi + 1]
        ts2 = np.linspace(bl, br, 9)
        ts = np.unique(np.sort(np.concatenate([ts[:bi], ts2, ts[bi + 1:]])))
        ts = np.unique(np.sort(np.append(ts, 0.0)))
        vals = np.max(q[None, :] + ts[:, None] * b[None, :] + (ts[:, None] ** 2) * c[None, :], axis=1)
        bi = int(np.argmin(vals))
    return float(ts[bi]), float(vals[bi])

def build_tree(n):
    records = []
    stack = [(0, n, 0)]
    while stack:
        a, b, dep = stack.pop()
        if b - a <= 1:
            continue
        m = (a + b) // 2
        records.append((dep, a, m, b))
        stack.append((m, b, dep + 1))
        stack.append((a, m, dep + 1))
    max_d = max(r[0] for r in records) if records else 0
    ordered = []
    for d in range(max_d + 1):
        for r in records:
            if r[0] == d:
                ordered.append(r)
    return ordered

def make_direction(rec, n):
    a, m, b = rec[1], rec[2], rec[3]
    L, R = m - a, b - m
    if L <= 0 or R <= 0:
        return None
    d = np.zeros(n)
    d[a:m] = 1.0 / L
    d[m:b] = -1.0 / R
    s = float(np.sum(d))
    if abs(s) > 1e-18:
        for i in range(a, b):
            if d[i] != 0.0:
                d[i] -= s
                break
    if abs(float(np.sum(d))) > 1e-14:
        return None
    return d

def run(seed=42, **kwargs):
    n = 45
    fallback_h = None
    parent_raw = None
    try:
        parent_raw = initial_h_values
    except NameError:
        pass
    if parent_raw is not None:
        try:
            h0 = np.array(parent_raw, dtype=np.float64).ravel()
            if 2 <= len(h0) <= 999:
                n = len(h0)
                if np.all(np.isfinite(h0)) and float(np.max(np.abs(h0))) <= 1e6:
                    if is_feasible(h0, n):
                        fallback_h = h0.copy()
                    else:
                        p = project_k(h0, n)
                        fallback_h = p.copy() if p is not None else np.full(n, 0.5)
                else:
                    fallback_h = np.full(n, 0.5)
        except Exception:
            n, fallback_h = 45, None
    if fallback_h is None:
        nv = kwargs.get('n_points', None)
        if isinstance(nv, (int, np.integer)) and not isinstance(nv, bool) and 2 <= nv <= 999:
            n = int(nv)
        fallback_h = np.full(n, 0.5)
    n = int(n)
    dx = 2.0 / n
    tree = build_tree(n)
    if n > 256 and len(tree) > 64:
        M = len(tree)
        tree = [tree[int(k * (M - 1) / 63)] for k in range(64)] if M > 1 else tree
    n_sweeps = 3 if n <= 128 else (2 if n <= 256 else 1)
    use_exact = n <= 48
    rng = np.random.default_rng(seed)
    starts = [fallback_h.copy()]
    try:
        if parent_raw is not None:
            h0 = np.array(parent_raw, dtype=np.float64).ravel()
            if len(h0) == n and np.all(np.isfinite(h0)):
                cj = project_k(1.0 - h0, n)
                if cj is not None:
                    starts.append(cj)
                nj = project_k(h0 + 0.12 * rng.standard_normal(n), n)
                if nj is not None:
                    starts.append(nj)
                bj = project_k(rng.beta(0.5, 0.5, n), n)
                if bj is not None:
                    starts.append(bj)
        else:
            uj = project_k(rng.uniform(0.0, 1.0, n), n)
            if uj is not None:
                starts.append(uj)
            bv = project_k(rng.beta(0.5, 0.5, n), n)
            if bv is not None:
                starts.append(bv)
    except Exception:
        pass
    mx_st = 4 if n <= 128 else (3 if n <= 256 else 2)
    deduped = [starts[0]]
    for s in starts[1:]:
        if any(float(np.max(np.abs(s - d))) <= 1e-8 for d in deduped):
            continue
        deduped.append(s)
        if len(deduped) >= mx_st:
            break
    starts = deduped[:mx_st]
    best_h, best_score = fallback_h.copy(), exact_score(fallback_h, n)
    for sh in starts:
        if not is_feasible(sh, n):
            continue
        try:
            cur, cs = sh.copy(), exact_score(sh, n)
            if cs < best_score - 1e-12:
                best_h, best_score = cur.copy(), cs
            for sw in range(n_sweeps):
                imp = False
                dord = list(range(len(tree)))
                if sw % 2 == 1:
                    dord.reverse()
                for di in dord:
                    d = make_direction(tree[di], n)
                    if d is None:
                        continue
                    t_lo, t_hi = -np.inf, np.inf
                    for i in range(n):
                        if d[i] != 0.0:
                            a1, a2 = -cur[i] / d[i], (1.0 - cur[i]) / d[i]
                            t_lo = max(t_lo, min(a1, a2))
                            t_hi = min(t_hi, max(a1, a2))
                    if not (np.isfinite(t_lo) and np.isfinite(t_hi)) or t_hi - t_lo <= 1e-12:
                        continue
                    q = dx * np.correlate(cur, 1.0 - cur, mode='full')
                    b = dx * (np.correlate(d, 1.0 - cur, mode='full') - np.correlate(cur, d, mode='full'))
                    c = -dx * np.correlate(d, d, mode='full')
                    if not (np.all(np.isfinite(q)) and np.all(np.isfinite(b)) and np.all(np.isfinite(c))):
                        continue
                    tb, pv = (exact_line_min(q, b, c, t_lo, t_hi) if use_exact
                              else grid_line_min(q, b, c, t_lo, t_hi))
                    if not np.isfinite(pv) or pv >= cs - 1e-12:
                        continue
                    hr = repair_small(cur + tb * d, n)
                    if hr is None:
                        continue
                    ts2 = exact_score(hr, n)
                    if ts2 < cs - 1e-12:
                        cur, cs = hr.copy(), ts2
                        imp = True
                        if cs < best_score - 1e-12:
                            best_h, best_score = cur.copy(), cs
                if not imp:
                    break
        except Exception:
            continue
    rh = best_h.copy()
    rs = exact_score(rh, n)
    if not is_feasible(rh, n) or not np.isfinite(rs):
        p = project_k(rh, n)
        if p is not None:
            rh, rs = p, exact_score(p, n)
    if not is_feasible(rh, n) or not np.isfinite(rs):
        fs = exact_score(fallback_h, n)
        if is_feasible(fallback_h, n) and np.isfinite(fs):
            rh, rs = fallback_h.copy(), fs
        else:
            rh = np.full(n, 0.5)
            rs = exact_score(rh, n)
    return (rh.copy(), float(rs), int(n))
