    B = np.zeros(2 * n - 1)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        row = d + (n - 1)
        lo = max(0, -d)
        hi = min(n, n - d)
        src = h[lo:hi]
        dst = h[lo + d:hi + d]
        B[row] = np.dot(src, 1.0 - dst)
        G[row, lo:hi] += (1.0 - dst)
        G[row, lo + d:hi + d] -= h[lo:hi]
    return B, G


def slep_obj(z):
    return z[-1]


def slep_obj_grad(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g


def slep_ineq(z):
    h = z[:-1]
    t = z[-1]
    B, _ = compute_lags(h)
    return t - B


def slep_ineq_jac(z):
    h = z[:-1]
    n = len(h)
    _, G = compute_lags(h)
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J


def slep_eq(z):
    return np.sum(z[:-1]) - len(z[:-1]) / 2.0


def slep_eq_jac(z):
    n = len(z) - 1
    J = np.zeros(n + 1)
    J[:n] = 1.0
    return J


def is_dup(a, b):
    return np.max(np.abs(a - b)) < 1e-10


def run(seed=42, **kwargs):
    n = None
    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        ihv_arr = np.asarray(ihv, dtype=np.float64)
        if ihv_arr.ndim == 1 and len(ihv_arr) >= 2:
            n = len(ihv_arr)
    if n is None:
        nk = kwargs.get('n_points', None)
        if nk is not None and isinstance(nk, (int, np.integer)) and 8 <= int(nk) <= 512:
            n = int(nk)
        else:
            n = 80
    S = n / 2.0
    try:
        rng = np.random.default_rng(int(seed))
    except (TypeError, ValueError):
        rng = np.random.default_generator(42)

    parent_proj = None
    if ihv is not None:
        ihv_arr = np.asarray(ihv, dtype=np.float64)
        if (ihv_arr.ndim == 1 and len(ihv_arr) == n
                and np.all(np.isfinite(ihv_arr)) and np.max(np.abs(ihv_arr)) <= 1e12):
            parent_proj = project_capped_simplex(ihv_arr, S)

    const_h = np.full(n, 0.5)
    best_h = const_h.copy()
    best_score = exact_score(const_h)
    if parent_proj is not None:
        ps = exact_score(parent_proj)
        if ps is not None and ps < best_score:
            best_h = parent_proj.copy()
            best_score = ps

    starts = []
    if parent_proj is not None:
        starts.append(parent_proj.copy())
    starts.append(const_h.copy())

    if n <= 128:
        if parent_proj is not None:
            rp = parent_proj[::-1].copy()
            if not any(is_dup(rp, s) for s in starts):
                starts.append(rp)
            pt = parent_proj + rng.normal(0, 0.18, n)
            pp = project_capped_simplex(pt, S)
            if pp is not None and not any(is_dup(pp, s) for s in starts):
                starts.append(pp)
        i_arr = np.arange(n, dtype=np.float64)
        q_max = min(6, max(1, (n - 1) // 2))
        for _ in range(2):
            q = int(rng.integers(1, q_max + 1))
            phi = float(rng.uniform(0, 2 * np.pi))
            per = 0.5 + 0.45 * np.cos(2 * np.pi * q * i_arr / n + phi)
            pp = project_capped_simplex(per, S)
            if pp is not None and not any(is_dup(pp, s) for s in starts):
                starts.append(pp)
        perm = rng.permutation(n)
        bv = np.zeros(n)
        bv[perm[:n // 2]] = 1.0
        if n % 2 == 1:
            bv[perm[n // 2]] = 0.5
        if not any(is_dup(bv, s) for s in starts):
            starts.append(bv.copy())
        gv = 0.5 + 0.3 * rng.standard_normal(n)
        gp = project_capped_simplex(gv, S)
        if gp is not None and not any(is_dup(gp, s) for s in starts):
            starts.append(gp)
        uv = rng.uniform(0, 1, n)
        up = project_capped_simplex(uv, S)
        if up is not None and not any(is_dup(up, s) for s in starts):
            starts.append(up)
    else:
        if parent_proj is not None:
            rp = parent_proj[::-1].copy()
            if not any(is_dup(rp, s) for s in starts):
                starts.append(rp)

    for sh in starts:
        ss = exact_score(sh)
        if ss is not None and ss < best_score:
            best_h = sh.copy()
            best_score = ss

    if n <= 128:
        bounds = [(0.0, 1.0)] * n + [(None, None)]
        cons = [
            {'type': 'ineq', 'fun': slep_ineq, 'jac': slep_ineq_jac},
            {'type': 'eq', 'fun': slep_eq, 'jac': slep_eq_jac},
        ]
        for sh in starts:
            try:
                B0, _ = compute_lags(sh)
                m0 = float(np.max(B0))
                t0 = m0 + 1e-8 * max(1.0, abs(m0))
                z0 = np.append(sh, t0)
                res = minimize(slep_obj, z0, jac=slep_obj_grad, bounds=bounds,
                               constraints=cons, method='SLSQP',
                               options={'maxiter': 180, 'ftol': 1e-9, 'disp': False})
                if res is not None and hasattr(res, 'x') and res.x is not None:
                    x = np.asarray(res.x, dtype=np.float64)
                    if x.shape == (n + 1,) and np.all(np.isfinite(x)):
                        hc = project_capped_simplex(x[:n], S)
                        if hc is not None:
                            sc = exact_score(hc)
                            if sc is not None and sc < best_score:
                                best_h = hc.copy()
                                best_score = sc
            except Exception:
                pass

    pool = [best_h, const_h]
    if parent_proj is not None:
        pool.append(parent_proj)

    final_h = const_h.copy()
    final_score = exact_score(const_h)
    for c in pool:
        if c is None:
            continue
        cp = project_capped_simplex(c, S)
        if cp is None:
            cp = c
        cs = exact_score(cp)
        if cs is not None and (final_score is None or cs < final_score):
            final_h = cp.copy()
            final_score = cs

    fs = exact_score(final_h)
    if fs is None:
        final_h = const_h.copy()
        fs = exact_score(const_h)

    return (final_h, fs, n)
