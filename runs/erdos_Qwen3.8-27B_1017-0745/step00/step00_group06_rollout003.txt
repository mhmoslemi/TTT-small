    candidates = [h_parent]
    h_uni = np.full(N, 0.5, dtype=np.float64)
    h_step = np.zeros(N, dtype=np.float64)
    n_ones = int(np.floor(m))
    h_step[:n_ones] = 1.0
    h_step[n_ones] = m - n_ones
    waves = [0.5 + 0.22 * np.sin(2.0 * np.pi * f * np.arange(N) / N) for f in (1, 3, 7)]
    raw_starts = [h_uni, h_step] + [project_feasible(w, N, m) for w in waves]
    for rs in raw_starts:
        is_dup = False
        for c in candidates:
            if np.max(np.abs(rs - c)) < 1e-8:
                is_dup = True
                break
        if not is_dup:
            candidates.append(rs)
    n_restarts = int(kwargs.get('n_restarts', 8))
    n_restarts = max(6, min(12, n_restarts))
    while len(candidates) < n_restarts:
        base = h_parent if rng.random() < 0.6 else h_uni
        std = [0.04, 0.10, 0.20][len(candidates) % 3]
        noise = rng.standard_normal(N) * std
        noise = 0.5 * noise + 0.25 * np.roll(noise, 1) + 0.25 * np.roll(noise, -1)
        cnd = project_feasible(base + noise, N, m)
        is_dup = False
        for c in candidates:
            if np.max(np.abs(cnd - c)) < 1e-8:
                is_dup = True
                break
        if not is_dup:
            candidates.append(cnd)
    try:
        maxiter = int(kwargs.get('maxiter', 120))
        maxiter = max(60, min(250, maxiter))
    except Exception:
        maxiter = 120
    ep_maxiter = min(400, 2 * maxiter)
    h_inc = project_feasible(h_parent, N, m)
    score_inc = evaluate_c5(h_inc)
    temps = [0.02, 0.005, 0.001, 0.0002]
    all_results = [h_inc]
    for start in candidates:
        h_cur = project_feasible(start, N, m)
        all_results.append(h_cur)
        for tau in temps:
            args_s = (N, d, tau, m)
            cons_s = [{'type': 'eq', 'fun': mass_eq_smooth, 'jac': mass_eq_smooth_jac}]
            bounds_s = [(0.0, 1.0)] * N
            try:
                res = minimize(smooth_obj, h_cur, jac=smooth_jac, bounds=bounds_s,
                               constraints=cons_s, args=args_s, method='SLSQP',
                               options={'ftol': 1e-10, 'maxiter': maxiter, 'disp': False})
                if res.x is not None and res.x.shape == (N,) and np.all(np.isfinite(res.x)):
                    h_cur = project_feasible(res.x, N, m)
                all_results.append(h_cur)
            except Exception:
                pass
    scored = []
    seen = []
    for hv in all_results:
        hv = project_feasible(hv, N, m)
        sc = evaluate_c5(hv)
        is_dup = False
        for (hs, ss) in seen:
            if np.max(np.abs(hv - hs)) < 1e-10:
                is_dup = True
                break
        if not is_dup:
            seen.append((hv, sc))
            scored.append((sc, hv))
    scored.sort(key=lambda x: x[0])
    top3 = [scored[i][1] for i in range(min(3, len(scored)))]
    for h_cand in top3:
        sc_cand = evaluate_c5(h_cand)
        z0 = np.concatenate([h_cand, [sc_cand + 1e-6]])
        args_e = (N, d, 0.0, m)
        cons_e = [
            {'type': 'eq', 'fun': mass_eq_epi, 'jac': mass_eq_epi_jac},
            {'type': 'ineq', 'fun': epi_ineq, 'jac': epi_ineq_jac}
        ]
        bounds_e = [(0.0, 1.0)] * N + [(0.0, 2.1)]
        try:
            res = minimize(epi_obj, z0, jac=epi_obj_jac, bounds=bounds_e,
                           constraints=cons_e, args=args_e, method='SLSQP',
                           options={'ftol': 1e-10, 'maxiter': ep_maxiter, 'disp': False})
            if res.x is not None and res.x.shape == (N + 1,) and np.all(np.isfinite(res.x)):
                h_e = project_feasible(res.x[:N], N, m)
                sc_e = evaluate_c5(h_e)
                if sc_e < score_inc - 1e-12:
                    h_inc = h_e
                    score_inc = sc_e
        except Exception:
            pass
    h_inc = project_feasible(h_inc, N, m)
    if not (np.all(h_inc >= -1e-12) and np.all(h_inc <= 1.0 + 1e-12) and
            abs(np.sum(h_inc) - m) < 1e-9 * max(1, N)):
        h_inc = np.full(N, 0.5, dtype=np.float64)
    score_inc = evaluate_c5(h_inc)
    return (h_inc, score_inc, N)
