import numpy as np
from scipy.optimize import minimize

def project_box_sum(z, m):
    n = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(n, 0.5)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(z - mid, 0.0, 1.0)))
        if s > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    res = m - float(np.sum(h))
    if res > 1e-12:
        for i in range(n):
            add = min(res, 1.0 - h[i])
            h[i] += add
            res -= add
            if res <= 1e-13:
                break
    elif res < -1e-12:
        for i in range(n):
            sub = min(-res, h[i])
            h[i] -= sub
            res += sub
            if res >= -1e-13:
                break
    res = m - float(np.sum(h))
    if abs(res) > 1e-10:
        idx = np.argsort(-h) if res > 0 else np.argsort(h)
        for i in idx:
            if res > 0:
                add = min(res, 1.0 - h[i])
                h[i] += add
                res -= add
            else:
                sub = min(-res, h[i])
                h[i] -= sub
                res += sub
            if abs(res) < 1e-13:
                break
    return h

def exact_score(h):
    n = len(h)
    if n == 0 or not np.all(np.isfinite(h)):
        return np.inf
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lag_q(h):
    n = len(h)
    q = np.zeros(2 * n - 1)
    for idx, s in enumerate(range(-(n - 1), n)):
        if s >= 0:
            L = n - s
            q[idx] = float(np.sum(h[0:L] * (1.0 - h[s:s + L])))
        else:
            d = -s
            L = n - d
            q[idx] = float(np.sum(h[d:d + L] * (1.0 - h[0:L])))
    return q

def lag_jac(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for idx, s in enumerate(range(-(n - 1), n)):
        if s >= 0:
            L = n - s
            G[idx, 0:L] += (1.0 - h[s:s + L])
            G[idx, s:s + L] -= h[0:L]
        else:
            d = -s
            L = n - d
            G[idx, d:d + L] += (1.0 - h[0:L])
            G[idx, 0:L] -= h[d:d + L]
    return G

    return z[n]
