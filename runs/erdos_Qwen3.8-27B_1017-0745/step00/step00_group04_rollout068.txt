def project_capped(v, n):
    S = n / 2.0
    v = np.clip(v, -1e6, 1e6)
    # Find lambda such that sum(clip(v - lambda, 0, 1)) = S
    # clip(v-lambda, 0, 1) = min(max(v-lambda, 0), 1)
    # As lambda increases, each component decreases from 1 to 0
    # At lambda = min(v)-1, all components are 1, sum = n > S (since S = n/2)
    # At lambda = max(v), all components are 0, sum = 0 < S
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo+hi)/2, 0, 1)
    # Correct residual
    residual = S - math.fsum(h)
    if abs(residual) > 1e-15:
        if residual > 0:
            for i in range(n):
                add = min(residual, 1 - h[i])
                h[i] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            for i in range(n):
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
                if abs(residual) < 1e-15:
                    break
    return h

def soft_obj_and_grad(h, tau, n):

    exps = np.exp(shifted)

    if not (np.all(best_h >= -1e-15) and np.all(best_h <= 1.0 + 1e-15) and abs(float(np.sum(best_h)) - S) < 1e-6):
        return h_const, score_const, n
    return best_h, final_score, n
