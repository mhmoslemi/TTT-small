import numpy as np
import scipy.optimize as sco

_G_N = 0
_G_D = 1.0
_G_TAU = 0.01

def exact_score(h):
    h = np.asarray(h, dtype=np.float64)
    N = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / N))

def get_corr_jac(h, d):
    N = len(h)
    R = np.correlate(h, 1.0 - h, mode='full') * d
    J = np.zeros((2 * N - 1, N), dtype=np.float64)
    for q in range(-(N - 1), N):
        row = q + N - 1
        js = max(0, -q)
        je = min(N, N - q)
        for j in range(js, je):
            J[row, j + q] += d * (1.0 - h[j])
            J[row, j] -= d * h[j + q]
    return R, J

def project_box_mass(x, m, N):
    x = np.array(x, dtype=np.float64)
    if x.shape != (N,) or not np.all(np.isfinite(x)) or np.max(np.abs(x)) > 1e6:

        if s > m:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    delta = m - np.sum(y)
    if delta > 0:
        for i in range(N):
            a = min(delta, 1.0 - y[i])
            if a <= 0:
