    h_fb = project_capped_simplex(h_fb, s)
    sc_fb = exact_score(h_fb)
    if sc_fb < sc_inc:
        h_inc, sc_inc = h_fb, sc_fb
    if not check_feasible(h_inc, n) or not np.isfinite(sc_inc):
        h_inc = np.full(n, 0.5)
        sc_inc = exact_score(h_inc)
    return (h_inc.astype(np.float64), float(sc_inc), int(n))
