    if not np.all(np.isfinite(h)) or abs(np.sum(h) - m) > 1e-8:
        return np.full(N, 0.5)
    return h


def lag_values(h):
    N = len(h)
    M = 2 * N - 1
    q = np.zeros(M, dtype=np.float64)
    om = 1.0 - h
    for k in range(N):
        L = N - k
        q[N - 1 + k] = np.dot(h[k:k + L], om[0:L])
    for k in range(1, N):
        L = N - k
        q[N - 1 - k] = np.dot(h[0:L], om[k:k + L])
    return q


def score_exact(h, N):
    if h is None or len(h) != N:
        return float('inf')
    h = np.asarray(h, dtype=np.float64)
    if not np.all(np.isfinite(h)):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / N))


def smooth_value(h, tau, N):
    q = lag_values(h)
    r = (2.0 / N) * q
    rmax = float(np.max(r))
    if tau < 1e-14:
        return rmax
    exponents = (r - rmax) / tau
    return rmax + tau * math.log(float(np.sum(np.exp(exponents))))


def smooth_value_grad(h, tau, N):
    q = lag_values(h)
    M = 2 * N - 1
    r = (2.0 / N) * q
    rmax = float(np.max(r))
    g = np.zeros(N, dtype=np.float64)
    om = 1.0 - h
    if tau < 1e-14:
        kmax = int(np.argmax(r)) - (N - 1)
        if kmax >= 0:
            L = N - kmax
            g[kmax:kmax + L] += om[0:L]
            g[0:L] -= h[kmax:kmax + L]
        else:
            d = -kmax
            L = N - d
            g[0:L] += om[d:d + L]
            g[d:d + L] -= h[0:L]
        return rmax, g
    exponents = (r - rmax) / tau
    exp_vals = np.exp(exponents)
    s = float(np.sum(exp_vals))
    p = exp_vals / s
    phi = rmax + tau * math.log(s)
    for idx in range(M):
        w = (2.0 / N) * p[idx]
        if w < 1e-16:
            continue
        k = idx - (N - 1)
        if k >= 0:
            L = N - k
            g[k:k + L] += w * om[0:L]
            g[0:L] -= w * h[k:k + L]
        else:
            d = -k
            L = N - d
            g[0:L] += w * om[d:d + L]
            g[d:d + L] -= w * h[0:L]
    return phi, g


def optimize_from_start(h_start, N, inc, inc_score, iters_per_stage):
    taus = [0.08, 0.02, 0.005, 0.001, 0.0002, 0.00004, 0.000008]
    h = h_start.copy()
    for tau in taus:
        stage_best = h.copy()
        stage_best_score = score_exact(h, N)
        if stage_best_score < inc_score:
            inc_score = stage_best_score
            inc = h.copy()
        for _it in range(iters_per_stage):
            phi, g = smooth_value_grad(h, tau, N)
            if not np.all(np.isfinite(g)):
                break
            if float(np.max(np.abs(g))) < 1e-12:
                break
            alpha = 1.0
            done_stage = False
            for _ls in range(24):
                y = project_box_sum(h - alpha * g, N)
                d = y - h
                dinf = float(np.max(np.abs(d)))
                if dinf < 1e-10:
                    break
                gd = float(np.dot(g, d))
                if gd >= 0:
                    alpha *= 0.5
                    if alpha < 1e-8:
                        break
                    continue
                phi_y = smooth_value(y, tau, N)
                sy = score_exact(y, N)
                if sy < inc_score:
                    inc_score = sy
                    inc = y.copy()
                if sy < stage_best_score:
                    stage_best_score = sy
                    stage_best = y.copy()
                if phi_y <= phi + 1e-4 * gd + 1e-14:
                    h = y
                    alpha = min(1.4 * alpha, 10.0)
                    if dinf < 1e-8:
                        done_stage = True
                    break
                else:
                    alpha *= 0.5
                    if alpha < 1e-8:
                        break
            if done_stage:
                break
            if alpha < 1e-8:
                break
        h = stage_best.copy()
        hs = score_exact(h, N)
        if hs < inc_score:
            inc_score = hs
            inc = h.copy()
    return inc, inc_score


def run(seed=42, **kwargs):
    N = None
    baseline = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if len(iv) > 0 and np.all(np.isfinite(iv)):
            N = len(iv)
            if (np.all(iv >= -1e-7) and np.all(iv <= 1.0 + 1e-7)
                    and abs(np.sum(iv) - N / 2.0) < 1e-4 * N):
                baseline = np.clip(iv, 0.0, 1.0)
                baseline = project_box_sum(baseline, N)
            else:
                baseline = project_box_sum(iv, N)
    if N is None:
        if 'n_points' in kwargs:
            try:
                nv = int(kwargs['n_points'])
                if 1 <= nv <= 999:
                    N = nv
            except (ValueError, TypeError):
                pass
        if N is None:
            N = 70
    if N == 1:
        hh = np.array([0.5])
        return hh, score_exact(hh, 1), 1
    if baseline is None:
        baseline = np.full(N, 0.5)

    inc = baseline.copy()
    inc_score = score_exact(inc, N)

    if N <= 100:
        n_starts = 8
        ipst = 60
    elif N <= 200:
        n_starts = 6
        ipst = 40
    elif N <= 500:
        n_starts = 4
        ipst = 20
    else:
        n_starts = 3
        ipst = 8

    rng = np.random.default_rng(seed)
    starts = [baseline.copy()]
    starts.append(project_box_sum(baseline + 0.20 * rng.standard_normal(N), N))
    starts.append(project_box_sum(rng.uniform(0.0, 1.0, N), N))

    R = min(6, (N - 1) // 2)
    if R > 0:
        ii = np.arange(N, dtype=np.float64)
        z4 = np.full(N, 0.5)
        for r in range(1, R + 1):
            A = rng.standard_normal()
            B = rng.standard_normal()
            z4 += (0.12 / r) * (A * np.sin(2.0 * np.pi * r * ii / N)
                                 + B * np.cos(2.0 * np.pi * r * ii / N))
        starts.append(project_box_sum(z4, N))
    else:
        starts.append(np.full(N, 0.5))

    use_complement = False
    if 'initial_h_values' in globals() and initial_h_values is not None:
        iv2 = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if len(iv2) == N and np.all(np.isfinite(iv2)):
            starts.append(project_box_sum(1.0 - np.clip(iv2, 0.0, 1.0), N))
            use_complement = True
    if not use_complement:
        starts.append(project_box_sum(0.5 + 0.4 * rng.standard_normal(N), N))

    starts.append(np.full(N, 0.5))
    starts.append(project_box_sum(baseline + 0.08 * rng.standard_normal(N), N))
    starts.append(project_box_sum(baseline + 0.40 * rng.standard_normal(N), N))

    for s in range(min(n_starts, len(starts))):
        try:
            hs = project_box_sum(starts[s], N)
            ss = score_exact(hs, N)
            if ss < inc_score:
                inc_score = ss
                inc = hs.copy()
            inc, inc_score = optimize_from_start(hs, N, inc, inc_score, ipst)
        except Exception:
            continue

    inc = project_box_sum(inc, N)
    final_score = score_exact(inc, N)
    if not np.isfinite(final_score) or len(inc) != N:
        inc = np.full(N, 0.5)
        final_score = score_exact(inc, N)
    return inc, final_score, N
