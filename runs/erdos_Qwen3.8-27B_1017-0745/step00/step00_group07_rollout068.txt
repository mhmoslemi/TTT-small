def _feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-10) and np.all(h <= 1.0 + 1e-10) and
            abs(np.sum(h) - n / 2.0) <= 1e-8)

def _proj(z, S):
    n = len(z)
    if not np.all(np.isfinite(z)) or np.max(np.abs(z)) > 1e12:
        return None
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0, 1)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - np.sum(y)
    if r > 0:
        for i in np.argsort(y):
            if y[i] < 1.0:
                d = min(r, 1.0 - y[i]); y[i] += d; r -= d
                if r <= 0: break
    elif r < 0:
        for i in np.argsort(y)[::-1]:
            if y[i] > 0.0:
                d = min(-r, y[i]); y[i] -= d; r += d
                if r >= 0: break
    y = np.clip(y, 0.0, 1.0)
    return y if (np.all(np.isfinite(y)) and abs(np.sum(y) - S) < 1e-9) else None

def _score(h):
    n = len(h)
    if n == 0 or not np.all(np.isfinite(h)):
        return None
    v = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(np.max(v)) if np.all(np.isfinite(v)) else None

def _lags(h):
    n = len(h); m = 2 * n - 1
    B = np.zeros(m); G = np.zeros((m, n))
    for k in range(m):
        d = k - (n - 1)
        lo = max(0, -d); hi = min(n, n - d)
        B[k] = np.dot(h[lo:hi], 1.0 - h[lo+d:hi+d])
        G[k, lo:hi] += 1.0 - h[lo+d:hi+d]
        G[k, lo+d:hi+d] -= h[lo:hi]
    return B, G

def _obj(z):
    return z[-1]

def _obj_g(z):
    g = np.zeros(len(z)); g[-1] = 1.0; return g

def _ineq(z):
    n = len(z) - 1
    B, _ = _lags(z[:n])
    return z[-1] - B

def _ineq_j(z):
    n = len(z) - 1
    _, G = _lags(z[:n])
    J = np.zeros((2*n-1, n+1)); J[:, :n] = -G; J[:, n] = 1.0
    return J

def _eq(z):
    return np.sum(z[:-1]) - (len(z) - 1) / 2.0

def _eq_j(z):
    j = np.zeros(len(z)); j[:-1] = 1.0; return j

def _periodic(n, q, phi):
    i = np.arange(n, dtype=float)
    return 0.5 + 0.45 * np.cos(2.0 * math.pi * q * i / n + phi)

def _binary(n, rng):
    h = np.zeros(n); p = rng.permutation(n)
    h[p[:n//2]] = 1.0
    if n % 2: h[p[n//2]] = 0.5
    return h

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    n = None
    parent = None
    igh = globals().get('initial_h_values', None)
    if igh is not None:
        try:
            p = np.asarray(igh, dtype=np.float64).ravel()
            if p.size >= 2:
                n = p.size
                pr = _proj(p.copy(), n / 2.0)
                if pr is not None and _feasible(pr, n):
                    parent = pr
        except Exception:
            pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 80))
        except Exception:
            n = 80
        if not (8 <= n <= 512):
            n = 80
    S = n / 2.0
    const = np.full(n, 0.5)
    best_h = const.copy()
    best_score = _score(best_h)

    raw_starts = []
    if parent is not None:
        raw_starts.append(parent.copy())
    raw_starts.append(const.copy())
    if parent is not None:
        raw_starts.append(parent[::-1].copy())
        raw_starts.append(parent + rng.normal(0, 0.18, n))
    if n <= 128:
        qmax = min(6, max(1, (n - 1) // 2))
        q1 = int(rng.integers(1, qmax + 1))
        phi1 = rng.uniform(0, 2 * math.pi)
        raw_starts.append(_periodic(n, q1, phi1))
        raw_starts.append(_binary(n, rng))
        q2 = int(rng.integers(1, qmax + 1))
        phi2 = rng.uniform(0, 2 * math.pi)
        raw_starts.append(_periodic(n, q2, phi2))

    starts = []
    for c in raw_starts:
        pr = _proj(np.asarray(c, dtype=np.float64), S)
        if pr is None or not _feasible(pr, n):
            continue
        dup = False
        for s in starts:
            if np.max(np.abs(pr - s)) < 1e-10:
                dup = True; break
        if not dup:
            starts.append(pr)

    for s in starts:
        sc = _score(s)
        if sc is not None and sc < best_score:
            best_score = sc; best_h = s.copy()

    if n <= 128:
        bounds = [(0, 1)] * n + [(None, None)]
        cons = [
            {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_j},
            {'type': 'eq', 'fun': _eq, 'jac': _eq_j}
        ]
        for h0 in starts:
            try:
                B0, _ = _lags(h0)
                t0 = float(np.max(B0)) + 1e-8
                z0 = np.append(h0, t0)
                res = minimize(_obj, z0, jac=_obj_g, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 200, 'ftol': 1e-10, 'disp': False})
                if res is not None and hasattr(res, 'x'):
                    h1 = np.asarray(res.x[:n], dtype=np.float64)
                    if np.all(np.isfinite(h1)):
                        pr = _proj(h1, S)
                        if pr is not None and _feasible(pr, n):
                            sc = _score(pr)
                            if sc is not None and sc < best_score:
                                best_score = sc; best_h = pr.copy()
            except Exception:
                pass
            try:
                h0b = h0 + rng.normal(0, 0.05, n)
                pr2 = _proj(h0b, S)
                if pr2 is not None and _feasible(pr2, n):
                    B2, _ = _lags(pr2)
                    t2 = float(np.max(B2)) + 1e-8
                    z2 = np.append(pr2, t2)
                    res2 = minimize(_obj, z2, jac=_obj_g, method='SLSQP',
                                    bounds=bounds, constraints=cons,
                                    options={'maxiter': 150, 'ftol': 1e-10, 'disp': False})
                    if res2 is not None and hasattr(res2, 'x'):
                        h2 = np.asarray(res2.x[:n], dtype=np.float64)
                        if np.all(np.isfinite(h2)):
                            pr3 = _proj(h2, S)
                            if pr3 is not None and _feasible(pr3, n):
                                sc3 = _score(pr3)
                                if sc3 is not None and sc3 < best_score:
                                    best_score = sc3; best_h = pr3.copy()
            except Exception:
                pass

    if not _feasible(best_h, n):
        best_h = const.copy()
        best_score = _score(best_h)
    c5 = _score(best_h)
    if c5 is None:
        best_h = const.copy()
        c5 = _score(best_h)
    return best_h.astype(np.float64), float(c5), int(n)
