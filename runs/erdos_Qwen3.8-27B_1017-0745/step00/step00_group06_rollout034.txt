The k-th element (0-indexed) corresponds to lag (k - (N-1)).
So element k corresponds to shift: h shifted right by (N-1-k), or equivalently, the correlation at lag (k - (N-1)).

Actually let me be more careful. `np.correlate(a, v, mode='full')` computes:
c_k = sum_j a[j] * v[j+k-(M-1)]

where M = max(len(a), len(v)). For same-length sequences of length N:
c_k = sum_j a[j] * v[j+k-(N-1)] for valid j.

In our case, a = h, v = 1-h. So:
c_k = sum_j h[j] * (1-h[j+k-(N-1)])

Let lag q = k - (N-1), so k = q + (N-1), range of q is [-(N-1), N-1].
c_{q+(N-1)} = sum_j h[j] * (1-h[j+q]) for valid j (where 0<=j<N and 0<=j+q<N).

This is the overlap of h with its q-shifted version against (1-h). So R_q = c_{q+(N-1)} * dx.

The gradient: dR_q/dh[a] for each a in [0,N):
R_q = dx * sum_j h[j] * (1-h[j+q]) for valid (j, j+q).

For a given index a:
- If a appears as j (first factor): contributes dx * (1-h[a+q]) for each valid j=a, i.e., if 0<=a+q<N
- If a appears as j+q (second factor): contributes -dx * h[a-q] for each valid j=a-q, i.e., if 0<=a-q<N

So: dR_q/dh[a] = dx*(1-h[a+q]) [if 0<=a+q<N] - dx*h[a-q] [if 0<=a-q<N]

        # First term: a+q in [0,N) => a in [-q, N-q) intersect [0,N) => a in [max(0,-q), min(N,N-q))

import numpy as np
from scipy.optimize import minimize

_P = {'dx': 0.02, 'm': 47.5, 'tau': 0.02}

def _score(h):
    h = np.asarray(h, dtype=np.float64)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / len(h)))

def _RJ(h):
    N = len(h)
    dx = _P['dx']
    omh = 1.0 - h
    R = np.zeros(2 * N - 1)
    J = np.zeros((2 * N - 1, N))
    for q in range(1 - N, N):
        r = q + N - 1
        jm, jM = max(0, -q), min(N, N - q)
        js = np.arange(jm, jM)
        R[r] = dx * np.dot(h[js], omh[js + q])
        a1m, a1M = max(0, -q), min(N, N - q)
        if a1M > a1m:
            J[r, a1m:a1M] += dx * omh[a1m + q:a1M + q]
        a2m, a2M = max(0, q), min(N, N + q)
        if a2M > a2m:
            J[r, a2m:a2M] -= dx * h[a2m - q:a2M - q]
    return R, J

def _proj(x, m):
    N = len(x)
    if not np.all(np.isfinite(x)) or np.max(np.abs(x)) > 1e6:
        return np.full(N, m / N)
    lo, hi = np.min(x) - 1.0, np.max(x)
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(x - mid, 0.0, 1.0)) > m:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    d = m - np.sum(y)
    for i in range(N):
        if abs(d) < 1e-14: break
        if d > 0:
            a = min(d, 1.0 - y[i]); y[i] += a; d -= a
        else:
            a = min(-d, y[i]); y[i] -= a; d += a
    return y

def _sf(x):
    R, _ = _RJ(x)
    a = R.max(); t = _P['tau']
    return float(a + t * np.log(np.sum(np.exp((R - a) / t))))

def _sj(x):
    R, J = _RJ(x)
    a = R.max(); t = _P['tau']
    w = np.exp((R - a) / t); w /= w.sum()
    return w @ J

def _eqf(x):
    return np.sum(x) - _P['m']

def _eqj(x):
    return np.ones(len(x))

def _ef(z):
    return z[-1]

def _ejf(z):
    r = np.zeros(len(z)); r[-1] = 1.0
    return r

def _eeq(z):
    return np.sum(z[:-1]) - _P['m']

def _eej(z):
    r = np.zeros(len(z)); r[:-1] = 1.0
    return r

def _eieq(z):
    h = z[:-1]
    R, _ = _RJ(h)
    return z[-1] - R

def _eiej(z):
    h = z[:-1]
    _, J = _RJ(h)
    return np.hstack([-J, np.ones((2 * len(h) - 1, 1))])

def _ck(c):
    return c[1]

def run(seed=42, **kwargs):
    global _P, initial_h_values
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size >= 2 and np.all(np.isfinite(p)):
            parent = p
    except Exception:
        pass
    if parent is not None:
        N = len(parent)
    else:
        N = 95
        if 'n_points' in kwargs:
            try: N = max(2, min(int(kwargs['n_points']), 999))
            except Exception: pass
    dx, m = 2.0 / N, N / 2.0
    _P = {'dx': dx, 'm': m, 'tau': 0.02}

    starts = []
    if parent is not None:
        starts.append(_proj(parent.copy(), m))
    starts.append(np.full(N, 0.5))
    st = np.zeros(N); k = int(m); st[:k] = 1.0
    if k < N and m > k: st[k] = m - k
    starts.append(st)
    for f in (1, 3, 7):
        i = np.arange(N)
        starts.append(_proj(0.5 + 0.22 * np.sin(2.0 * np.pi * f * i / N), m))

    uniq = [starts[0]]
    for s in starts[1:]:
        if all(np.max(np.abs(s - u)) > 1e-8 for u in uniq):
            uniq.append(s)

    nr = 8
    if 'n_restarts' in kwargs:
        try: nr = max(6, min(int(kwargs['n_restarts']), 12))
        except Exception: pass
    rng = np.random.default_rng(seed if isinstance(seed, (int, np.integer)) else 42)
    stds_arr = np.array([0.04, 0.10, 0.20])
    while len(uniq) < nr:
        nz = rng.standard_normal(N) * stds_arr[len(uniq) % 3]
        nz = 0.5 * nz + 0.25 * np.roll(nz, 1) + 0.25 * np.roll(nz, -1)
        c = _proj(uniq[0] + nz, m)
        if all(np.max(np.abs(c - u)) > 1e-8 for u in uniq):
            uniq.append(c)

    mi = 120
    if 'maxiter' in kwargs:
        try: mi = max(60, min(int(kwargs['maxiter']), 250))
        except Exception: pass

    incumbent = uniq[0].copy()
    inc_sc = _score(incumbent)
    cands = [(incumbent.copy(), inc_sc)]
    bounds = [(0.0, 1.0)] * N
    eqc = {'type': 'eq', 'fun': _eqf, 'jac': _eqj}

    for start in uniq:
        cur = start.copy()
        for tau in (0.02, 0.005, 0.001, 0.0002):
            _P['tau'] = tau
            try:
                res = minimize(_sf, cur, jac=_sj, method='SLSQP',
                               bounds=bounds, constraints=[eqc],
                               options={'ftol': 1e-10, 'maxiter': mi, 'disp': False})
                hx = np.asarray(res.x, dtype=np.float64)
                if hx.shape == (N,) and np.all(np.isfinite(hx)):
                    cur = _proj(hx, m)
            except Exception:
                pass
            sc = _score(cur)
            cands.append((cur.copy(), sc))

    cands.sort(key=_ck)
    top = []
    seen = []
    for h, s in cands:
        distinct = True
        for u in seen:
            if np.max(np.abs(h - u)) <= 1e-6:
                distinct = False
                break
        if distinct:
            top.append((h, s))
            seen.append(h.copy())
            if len(top) >= 3:
                break

    emi = min(400, 2 * mi)
    for h0, _ in top:
        z0 = np.concatenate([h0, [_score(h0) + 1e-6]])
        ebounds = [(0.0, 1.0)] * N + [(0.0, 2.1)]
        econs = [
            {'type': 'eq', 'fun': _eeq, 'jac': _eej},
            {'type': 'ineq', 'fun': _eieq, 'jac': _eiej}
        ]
        try:
            res = minimize(_ef, z0, jac=_ejf, method='SLSQP',
                           bounds=ebounds, constraints=econs,
                           options={'ftol': 1e-10, 'maxiter': emi, 'disp': False})
            hx = np.asarray(res.x, dtype=np.float64)[:-1]
            if hx.shape == (N,) and np.all(np.isfinite(hx)):
                hx = _proj(hx, m)
                sc = _score(hx)
                cands.append((hx, sc))
        except Exception:
            pass

    cands.sort(key=_ck)
    best_h, best_s = cands[0]
    best_h = _proj(best_h, m)
    best_s = _score(best_h)

    if (len(best_h) != N or not np.all(np.isfinite(best_h)) or
            np.any(best_h < -1e-12) or np.any(best_h > 1.0 + 1e-12) or
            abs(np.sum(best_h) - m) > 1e-9 * max(1, N)):
        best_h = np.full(N, 0.5)
        best_s = _score(best_h)

    return (best_h, float(best_s), N)
