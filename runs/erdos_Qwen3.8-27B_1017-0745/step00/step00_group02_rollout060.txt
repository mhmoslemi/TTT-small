import numpy as np
from scipy.optimize import minimize

def project_onto_K(y, n):
    """Project y onto {h: 0<=h_i<=1, sum(h)=n/2}"""
    S = n / 2.0
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(y - mid, 0.0, 1.0)
        s = proj.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    proj = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Correct residual
    residual = S - proj.sum()
    if abs(residual) > 0:
        if residual > 0:
            slack = 1.0 - proj
            idx = np.where(slack > 1e-12)[0]
            if len(idx) > 0:
                add = min(residual / len(idx), 0.5 * slack[idx].min())
                proj[idx] += add
            # If still off, distribute more
            residual = S - proj.sum()
            if abs(residual) > 1e-12:
                idx = np.where(proj < 1.0 - 1e-12)[0]
                if len(idx) > 0:
                    add = residual / len(idx)
                    proj[idx] = np.clip(proj[idx] + add, 0, 1)
        else:
            idx = np.where(proj > 1e-12)[0]
            if len(idx) > 0:
                rem = min(-residual / len(idx), 0.5 * proj[idx].min())
                proj[idx] -= rem
            residual = S - proj.sum()
            if abs(residual) > 1e-12:
                idx = np.where(proj > 1e-12)[0]
                if len(idx) > 0:
                    rem = residual / len(idx)
                    proj[idx] = np.clip(proj[idx] + rem, 0, 1)
    # Final clip
    proj = np.clip(proj, 0.0, 1.0)
    return proj

def exact_score(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_lags(h, n, dx):
    """Return array of 2n-1 lag values matching np.correlate(h,1-h,'full')*dx ordering"""
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_obj_and_grad(h, n, dx, tau):
    """Compute F_tau and its gradient."""
    q = compute_lags(h, n, dx)
    L = len(q)
    m = np.max(q)
    w = np.exp((q - m) / tau)
    sw = w.sum()
    F = m + tau * np.log(sw)
    p = w / sw
    # Gradient: sum_k p_k * grad_q_k
    # grad_q_k[j] = dx * [(1-h[j+k]) if 0<=j+k<n else 0] - [h[j-k] if 0<=j-k<n else 0]
    # k_signed = k - (n-1) for k=0..2n-2
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        ks = k - (n - 1)  # signed lag
        pk = p[k]
        if abs(pk) < 1e-15:
            continue
        # h[j] appears with +: term h[j]*(1-h[j+ks]) -> deriv (1-h[j+ks]) if 0<=j+ks<n
        # h[j] appears with -: term h[j-ks]*(1-h[j]) -> deriv -h[j-ks] if 0<=j-ks<n
        # For +: j from 0 to n-1-ks (if ks>=0) or j from 0 to n-1 (if ks<0, j+ks from -ks to n-1-ks)
        lo_p = max(0, -ks)
        hi_p = min(n, n - ks)
        # For -: j from ks to n-1 (if ks>=0) or j from 0 to n-1+ks (if ks<0)
        lo_m = max(0, ks)
        hi_m = min(n, n + ks)
        if hi_p > lo_p:
            grad[lo_p:hi_p] += pk * dx * (1.0 - h[lo_p + ks:hi_p + ks])
        if hi_m > lo_m:
            grad[lo_m:hi_m] -= pk * dx * h[lo_m - ks:hi_m - ks]
    return F, grad

def project_K(y, n):
    S = n / 2.0
    lo, hi = np.min(y) - 1.0, np.max(y) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    p = np.clip(y - lam, 0.0, 1.0)
    r = S - p.sum()
    if r > 1e-12:
        idx = np.where(p < 1.0 - 1e-12)[0]
        if len(idx):
            p[idx] = np.minimum(p[idx] + r / len(idx), 1.0)
    elif r < -1e-12:
        idx = np.where(p > 1e-12)[0]
        if len(idx):
            p[idx] = np.maximum(p[idx] + r / len(idx), 0.0)
    return np.clip(p, 0.0, 1.0)

def exact_score(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def get_lags(h, n, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_fg(h, n, dx, tau):
    q = get_lags(h, n, dx)
    L = len(q)
    m = q.max()
    w = np.exp((q - m) / tau)
    sw = w.sum()
    F = m + tau * np.log(sw)
    p = w / sw
    g = np.zeros(n)
    for k in range(L):
        pk = p[k]
        if pk < 1e-15:
            continue
        ks = k - (n - 1)
        if ks >= 0:
            if n - ks > 0:
                g[:n-ks] += pk * dx * (1.0 - h[ks:n])
                g[ks:] -= pk * dx * h[:n-ks]
        else:
            m2 = -ks
            if n - m2 > 0:
                g[m2:] += pk * dx * (1.0 - h[:n-m2])
                g[:n-m2] -= pk * dx * h[m2:]
    return F, g

    h = h0.copy()

        alpha = 1.0
        small_count = 0
        for _ in range(max_it):
            F, g = smooth_fg(h, n, dx, tau)
            if not np.all(np.isfinite(g)):
                break
            accepted = False
            for _ in range(max_ls):

        res = run_grad(pert, n, dx, taus_full[-5:], 40, 20)
        sc = exact_score(res, n)
        local.append(res.copy())
        if sc < best_score - 1e-12:
            best_score, best_h = sc, res.copy()

    if n <= 100:
        slist = [exact_score(cr, n) for cr in local]
        order = np.argsort(slist)
        sel = []
        for i in order:
            if all(np.max(np.abs(local[i] - s)) > 1e-5 for s in sel):
                sel.append(local[i].copy())
            if len(sel) >= 3:
                break
        if not sel:
            sel = [best_h.copy()]
        for cr in sel:
            t0 = min(1.0, exact_score(cr, n) + 1e-6)
            z0 = np.concatenate([cr, [t0]])
            bounds = [(0.0, 1.0)] * (n + 1)
            cons = [
                {'type': 'ineq', 'fun': _sl_ineq, 'jac': _sl_ineq_j, 'args': (n, dx)},
                {'type': 'eq', 'fun': _sl_eq, 'jac': _sl_eq_j, 'args': (n, S)}
            ]
            try:
                sol = minimize(_sl_f, z0, args=(n,), jac=_sl_fg,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 200, 'ftol': 1e-10, 'disp': False})
                hs = project_K(sol.x[:n], n)
                sc = exact_score(hs, n)
                if sc < best_score - 1e-12:
                    best_score, best_h = sc, hs.copy()
            except Exception:
                pass

    best_h = project_K(best_h, n)
    fb_sc = exact_score(fallback_h, n)
    best_score = exact_score(best_h, n)
    if best_score > fb_sc + 1e-12:
        best_h, best_score = fallback_h, fb_sc
    if not np.all(np.isfinite(best_h)) or np.abs(best_h.sum() - S) > 1e-8:
        best_h = np.full(n, 0.5)
        best_score = exact_score(best_h, n)
    best_h = np.clip(best_h, 0.0, 1.0)
    return (best_h.copy(), best_score, n)
