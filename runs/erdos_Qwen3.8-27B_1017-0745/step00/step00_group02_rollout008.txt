def _repair(h, n, S):
    h = h.copy()
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1 + 1e-12):
        return None
    h = np.clip(h, 0.0, 1.0)
    r = S - float(np.sum(h))
    if abs(r) > 1e-8:
        return None
    if r > 0:
        for i in range(n):
            add = min(r, 1.0 - h[i])
            h[i] += add
            r -= add
            if r <= 1e-13:
                break
    elif r < 0:
        for i in range(n):
            sub = min(-r, h[i])
            h[i] -= sub
            r += sub
            if r >= -1e-13:
                break
    if abs(float(np.sum(h)) - S) > 1e-10:
        return None
    return h

def _pair_bc(h, n, dx, i, j):
    nl = 2 * n - 1
    b = np.zeros(nl)
    for p, sgn in [(i, 1.0), (j, -1.0)]:
        for ell in range(-(n - 1), n):
            idx = ell + n - 1
            g = 0.0
            if 0 <= p - ell < n:
                g += (1.0 - h[p - ell])
            if 0 <= p + ell < n:
                g -= h[p + ell]
            b[idx] += sgn * dx * g
    c = np.zeros(nl)
    c[n - 1] = -2.0 * dx
    lg = j - i
    if lg != 0:
        c[n - 1 + lg] += dx
        c[n - 1 - lg] += dx
    return b, c

def _anneal(h0, n, dx, S, rng, n_props):
    cur = h0.copy().astype(np.float64)
    cur_sc, cur_q = _score(cur, n)
    best, best_sc = cur.copy(), cur_sc
    stages = [(0.20, 0.50), (0.08, 0.25), (0.03, 0.12), (0.01, 0.06)]
    for tm, st in stages:
        T = tm * dx
        for _ in range(n_props):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n - 1))
            if j >= i:
                j += 1
            t = float(rng.uniform(-st, st))
            lo = max(-cur[i], cur[j] - 1.0)
            hi = min(1.0 - cur[i], cur[j])
            if t < lo or t > hi:
                continue
            tr = cur.copy()
            tr[i] += t
            tr[j] -= t
            tr = _repair(tr, n, S)
            if tr is None:
                continue
            ts, tq = _score(tr, n)
            if ts >= float('inf'):
                continue
            d = ts - cur_sc
            if d <= 0.0 or float(rng.random()) < math.exp(-min(d / T, 700.0)):
                cur, cur_sc, cur_q = tr, ts, tq
                if ts < best_sc - 1e-12:
                    best_sc, best = ts, tr.copy()
    return best, best_sc

def _refine(h0, n, dx, S, rng):
    cur = h0.copy().astype(np.float64)
    cur_sc, cur_q = _score(cur, n)
    if cur_q is None:
        return cur, cur_sc
    for _sweep in range(2):
        committed = False
        s0 = cur_sc
        if n <= 64:
            prs = [(i, j) for i in range(n) for j in range(i + 1, n)]
            for k in range(len(prs) - 1, 0, -1):
                r = int(rng.integers(0, k + 1))
                prs[k], prs[r] = prs[r], prs[k]
        else:
            mx = min(400, 4 * n) if n <= 256 else 64
            prs, seen = [], set()
            for _ in range(mx * 10):
                if len(prs) >= mx:
                    break
                a, c2 = int(rng.integers(0, n)), int(rng.integers(0, n))
                if a == c2:
                    continue
                if a > c2:
                    a, c2 = c2, a
                if (a, c2) in seen:
                    continue
                seen.add((a, c2))
                prs.append((a, c2))
        for (ii, jj) in prs:
            lo = max(-cur[ii], cur[jj] - 1.0)
            hi = min(1.0 - cur[ii], cur[jj])
            if hi - lo < 1e-15:
                continue
            b, c = _pair_bc(cur, n, dx, ii, jj)
            grid = np.unique(np.sort(np.append(np.linspace(lo, hi, 17), 0.0)))
            for k in range(1, 6):
                grid = np.append(grid, [lo * (0.5 ** k), hi * (0.5 ** k)])
            grid = np.unique(np.sort(grid))
            grid = grid[(grid >= lo - 1e-14) & (grid <= hi + 1e-14)]
            best_t = 0.0
            best_pred = float('inf')
            for _it in range(3):
                ps = np.max(cur_q[None, :] + b[None, :] * grid[:, None] + c[None, :] * grid[:, None] ** 2, axis=1)
                bi = int(np.argmin(ps))
                best_pred, best_t = float(ps[bi]), float(grid[bi])
                if len(grid) < 3:
                    break
                if bi == 0:
                    nb = np.sort(np.unique(grid[:2]))
                elif bi == len(grid) - 1:
                    nb = np.sort(np.unique(grid[-2:]))
                else:
                    nb = np.sort(np.unique(grid[bi - 1:bi + 2]))
                ext = np.linspace(nb[0], nb[-1], 7)
                grid = np.unique(np.sort(np.concatenate([grid, ext])))
                grid = grid[(grid >= lo - 1e-14) & (grid <= hi + 1e-14)]
            if best_pred < cur_sc - 1e-12:
                tr = cur.copy()
                tr[ii] += best_t
                tr[jj] -= best_t
                tr = _repair(tr, n, S)
                if tr is not None:
                    ts, tq2 = _score(tr, n)
                    if ts < float('inf') and ts < cur_sc - 1e-12:
                        cur, cur_sc, cur_q = tr, ts, tq2
                        committed = True
        if not committed or cur_sc > s0 - 1e-10:
            break
    return cur, cur_sc

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)

    parent = globals().get('initial_h_values', None)
    n = 45
    fb = None
    if parent is not None:
        p = np.array(parent, dtype=np.float64)
        if p.ndim == 1 and 2 <= len(p) < 1000:
            n = len(p)
            if np.all(np.isfinite(p)) and np.all(np.abs(p) <= 1e6):
                dx0 = 2.0 / n
                S0 = n / 2.0
                if np.all(p >= 0) and np.all(p <= 1) and abs(float(np.sum(p)) - S0) <= 1e-10:
                    fb = p.copy()
                else:
                    pr = _project(p, n, S0)
                    fb = pr if pr is not None else np.full(n, 0.5)
            else:
                fb = np.full(n, 0.5)
    if fb is None:
        nk = kwargs.get('n_points', None)
        if isinstance(nk, int) and 2 <= nk <= 999:
            n = nk
        fb = np.full(n, 0.5)

    dx = 2.0 / n
    S = n / 2.0
    fb_sc, _ = _score(fb, n)
    gb, gb_sc = fb.copy(), fb_sc

    starts = []
    if parent is not None:
        p = np.array(parent, dtype=np.float64)
        if p.ndim == 1 and len(p) == n and np.all(np.isfinite(p)):
            if not (np.all(p >= 0) and np.all(p <= 1) and abs(float(np.sum(p)) - S) <= 1e-10):
                p = _project(p, n, S)
                if p is None:
                    p = np.full(n, 0.5)
            starts.append(p.copy())
            starts.append(p[::-1].copy())
            starts.append((1.0 - p).copy())
            starts.append((1.0 - p)[::-1].copy())
            for sig in [0.08, 0.25]:
                pr = _project(p + sig * rng.standard_normal(n), n, S)
                if pr is not None:
                    starts.append(pr)
            for fn in [lambda: rng.uniform(0, 1, n), lambda: rng.beta(0.5, 0.5, n)]:
                pr = _project(fn(), n, S)
                if pr is not None:
                    starts.append(pr)
    else:
        starts.append(np.full(n, 0.5))
        for _ in range(3):
            pr = _project(rng.uniform(0, 1, n), n, S)
            if pr is not None:
                starts.append(pr)
        for _ in range(2):
            pr = _project(rng.beta(0.5, 0.5, n), n, S)
            if pr is not None:
                starts.append(pr)
        pr = _project(rng.beta(2, 2, n), n, S)
        if pr is not None:
            starts.append(pr)
        mv = int(rng.integers(1, max(1, min(4, n // 2)) + 1))
        ph = float(rng.uniform(0, 2 * math.pi))
        xv = np.arange(n, dtype=np.float64) + 0.5
        pr = _project(0.5 + 0.45 * np.cos(2 * math.pi * mv * xv / n + ph), n, S)
        if pr is not None:
            starts.append(pr)

    uq = []
    for s in starts:
        if not any(np.max(np.abs(s - u)) <= 1e-6 for u in uq):
            uq.append(s)
    starts = uq

    nc = len(starts) if n <= 64 else min(6 if n <= 256 else 4, len(starts))
    np_ = 180 if n <= 64 else (40 if n <= 256 else 12)

    all_best = [(fb.copy(), fb_sc)]
    for cs in starts[:nc]:
        try:
            sc, _ = _score(cs, n)
            if sc < gb_sc - 1e-12:
                gb, gb_sc = cs.copy(), sc
            cb, cbs = _anneal(cs, n, dx, S, rng, np_)
            all_best.append((cb, cbs))
            if cbs < gb_sc - 1e-12:
                gb, gb_sc = cb.copy(), cbs
        except Exception:
            pass

    nes = [0.08, 0.25] if n <= 64 else [0.15]
    for sg in nes:
        try:
            nec = _project(gb + sg * rng.standard_normal(n), n, S)
            if nec is not None:
                cb, cbs = _anneal(nec, n, dx, S, rng, np_)
                all_best.append((cb, cbs))
                if cbs < gb_sc - 1e-12:
                    gb, gb_sc = cb.copy(), cbs
        except Exception:
            pass

    cands = []
    for (hv, sv) in all_best:
        if sv < float('inf') and not any(np.max(np.abs(hv - cv)) <= 1e-5 for cv, _ in cands):
            cands.append((hv.copy(), sv))
    cands.sort(key=lambda x: x[1])
    nr = 3 if n <= 64 else (2 if n <= 256 else 1)
    for (hc, _) in cands[:nr]:
        try:
            hr, hrs = _refine(hc, n, dx, S, rng)
            if hrs < gb_sc - 1e-12:
                gb, gb_sc = hr.copy(), hrs
        except Exception:
            pass

    final = _repair(gb.copy(), n, S)
    if final is None:
        final = _repair(fb.copy(), n, S)
    if final is None:
        final = np.full(n, 0.5)
    fsc, _ = _score(final, n)
    if fsc >= float('inf') or not _check_feasible(final, n):
        final = np.full(n, 0.5)
        fsc, _ = _score(final, n)
    return final.copy(), float(fsc), int(n)
