                    do_swap = (u == 0 or math.log(u) < L)
                if do_swap:
                    states[ri], states[si] = states[si], states[ri]
                    scores[ri], scores[si] = scores[si], scores[ri]
        if stagnation >= stag_lim and restarts < 2 and eval_count + R < B:
            restarts += 1
            stagnation = 0
            ns, nsc = [], []
            for i in range(R):
                if i == 0:
                    prof = best_state.copy()
                elif i <= 5:
                    prof = best_state + rng.standard_normal(n) * [0.02, 0.04, 0.08, 0.16, 0.28][i - 1]
                elif i == 6:
                    prof = rng.beta(0.5, 0.5, n)
                elif restarts == 1:
                    prof = rng.random(n)
                else:
                    bb = np.zeros(n)
                    bb[:n // 2] = 1.0
                    if n % 2 == 1:
                        bb[n // 2] = 0.5
                    prof = np.roll(bb, int(rng.integers(n)))
                pr = capped_simplex_project(prof, target)
                if pr is None:
                    pr = best_state.copy()
                sc = exact_score(pr, dx)
                if sc is None:
                    pr = best_state.copy()
                    sc = exact_score(pr, dx)
                ns.append(pr)
                nsc.append(sc)
                eval_count += 1
                if sc < best_score:
                    best_score = sc
                    best_state = pr.copy()
            states, scores = ns, nsc
    final = capped_simplex_project(best_state, target)
    if final is None:
        final = fallback
    fs = exact_score(final, dx)
    if fs is None:
        final = fallback
        fs = exact_score(final, dx)
    if fs is None:
        final = np.full(n, 0.5)
        fs = float(np.max(np.correlate(final, 1.0 - final, mode='full')) * dx)
    return final, float(fs), n
