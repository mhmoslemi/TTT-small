import numpy as np
from scipy.optimize import minimize

def exact_score(h):
    n = len(h)
    if not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def check_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def project_box_sum(v, S):
    n = len(v)
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    res = S - np.sum(y)
    if res > 0:
        caps = 1.0 - y
        for i in range(n):
            add = min(res, caps[i])
            y[i] += add
            res -= add
            if res <= 1e-14:
                break
    elif res < 0:
        res = -res
        for i in range(n):
            sub = min(res, y[i])
            y[i] -= sub
            res -= sub
            if res <= 1e-14:
                break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    if abs(np.sum(y) - S) > 1e-10 * max(1, n):
        return None
    return y

def compute_lags(h):
    n = len(h)
    n_lags = 2 * n - 1
    Q = np.zeros(n_lags)
    G = np.zeros((n_lags, n))
    for m in range(n_lags):
        ell = m - (n - 1)
        i_lo = max(0, -ell)
        i_hi = min(n, n - ell)
        for i in range(i_lo, i_hi):
            j = i + ell
            Q[m] += h[j] * (1.0 - h[i])
            if j != i:
                G[m, j] += (1.0 - h[i])
                G[m, i] -= h[j]
            else:
                G[m, i] += (1.0 - 2.0 * h[i])
    return Q, G

def slsqp_obj(z):
    return z[-1]

def slsqp_obj_jac(z):
    n = len(z) - 1
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def slsqp_ineq(z):
    h = z[:-1]
    n = len(h)
    t = z[-1]
    Q, _ = compute_lags(h)
    return t - Q / n

def slsqp_ineq_jac(z):
    h = z[:-1]
    n = len(h)
    _, G = compute_lags(h)
    n_lags = 2 * n - 1
    J = np.zeros((n_lags, n + 1))
    J[:, :n] = -G / n
    J[:, n] = 1.0
    return J

def slsqp_eq(z):
    h = z[:-1]
    return np.sum(h) / len(h) - 0.5

def slsqp_eq_jac(z):
    n = len(z) - 1
    g = np.zeros(len(z))

    return g
