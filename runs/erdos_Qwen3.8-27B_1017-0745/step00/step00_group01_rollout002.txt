        z0[:-1] = start
        z0[-1] = min(2.0, exact_score(start) + 1e-6)
        try:
            res = minimize(_obj, z0, method='SLSQP', jac=_obj_jac,
                           bounds=bounds, constraints=cons,
                           options={'ftol': 1e-10, 'maxiter': maxiter})
            if res is not None and getattr(res, 'x', None) is not None:
                x = np.asarray(res.x, dtype=np.float64)
                if x.size == n + 1 and np.all(np.isfinite(x)):
                    hc = project_capped_simplex(x[:-1], n)
                    sc = exact_score(hc)
                    if sc < best_score - 1e-12:
                        best_h = hc
                        best_score = sc
        except Exception:
            continue

    if n <= 128:
        best_h = pair_search(best_h, n)
        best_score = exact_score(best_h)

    best_h = project_capped_simplex(best_h, n)
    best_score = exact_score(best_h)
    ps = exact_score(parent)
    if best_score > ps + 1e-12:
        best_h = parent
        best_score = ps
    if not np.all(np.isfinite(best_h)) or best_h.min() < -1e-10 or best_h.max() > 1.0 + 1e-10:
        best_h = np.full(n, 0.5)
        best_score = exact_score(best_h)
    return best_h, float(best_score), n
