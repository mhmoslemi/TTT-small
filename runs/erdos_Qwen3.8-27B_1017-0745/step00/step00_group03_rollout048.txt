def gen_starts(x0, N, rng):
    starts = [x0.copy(), np.full(N, 0.5)]
    starts.append(project_box_sum(1.0 - x0, N))
    starts.append(project_box_sum(x0 + 0.15 * rng.standard_normal(N), N))
    starts.append(project_box_sum(rng.random(N), N))
    Rf = min(5, (N - 1) // 2)
    if Rf > 0:
        i = np.arange(N)
        z = np.full(N, 0.5)
        for r in range(1, Rf + 1):
            A = rng.standard_normal()
            B = rng.standard_normal()
            z += (0.12 / r) * (A * np.sin(2.0 * np.pi * r * i / N) + B * np.cos(2.0 * np.pi * r * i / N))
        starts.append(project_box_sum(z, N))
    else:
        starts.append(np.full(N, 0.5))
    unique = [starts[0]]
    for s in starts[1:]:
        if all(np.max(np.abs(s - u)) > 1e-8 for u in unique):
            unique.append(s)
    return unique


def local_solve(x0, N, dx, prob_objs, max_iter):
    d_var, u_var, t_var, x_p, r0_p, J0_p, r_nz_p, J_nz_p, R_p, prob = prob_objs
    x = x0.copy()
    F_cur = score_exact(x, N)
    R = min(1.0, float(np.sqrt(N)))
    R_max = min(4.0, float(np.sqrt(N)))
    R_min = 1e-4
    failures = 0
    nz_idx = list(range(N - 1)) + list(range(N, 2 * N - 1))
    for _ in range(max_iter):
        if R <= R_min or failures >= 8:
            break
        q, G = compute_qG(x, N)
        if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
            break
        r0 = dx * q[N - 1]
        J0 = dx * G[N - 1]
        r_nz = dx * q[nz_idx]
        J_nz = dx * G[nz_idx]
        x_p.value = x
        r0_p.value = r0
        J0_p.value = J0
        r_nz_p.value = r_nz
        J_nz_p.value = J_nz
        R_p.value = R
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=300,
                       abstol=1e-8, reltol=1e-8, feastol=1e-8)
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                R *= 0.5
                failures += 1
                continue
            d_sol = d_var.value
            if d_sol is None or d_sol.shape != (N,) or not np.all(np.isfinite(d_sol)):
                R *= 0.5
                failures += 1
                continue
        except Exception:
            R *= 0.5
            failures += 1
            continue
        x_new = project_box_sum(x + d_sol, N)
        d_eff = x_new - x
        if np.linalg.norm(d_eff) > R * (1.0 + 1e-6) + 1e-10:
            R *= 0.5
            failures += 1
            continue
        F_new = score_exact(x_new, N)
        tol = 1e-12 * max(1.0, abs(F_cur))
        if F_new < F_cur - tol:
            x = x_new
            F_cur = F_new
            failures = 0
            if np.linalg.norm(d_eff) >= 0.8 * R:
                R = min(1.5 * R, R_max)
        else:
            R *= 0.5
            failures += 1
        if np.linalg.norm(d_eff) < 1e-7:
            break
    return x, F_cur


def run(seed=42, **kwargs):
    N = 70
    x0 = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= len(parent) <= 1000:
            N = len(parent)
            x0 = parent.copy()
    except Exception:
        pass
    if x0 is None:
        try:
            nk = int(kwargs.get('n_points', 70))
            if 1 <= nk <= 999:
                N = nk
        except Exception:
            pass
    dx = 2.0 / N
    if x0 is None or x0.shape != (N,) or not np.all(np.isfinite(x0)):
        if x0 is not None and x0.shape == (N,):
            x0 = project_box_sum(x0, N)
        else:
            x0 = np.full(N, 0.5)
    else:
        if np.any(x0 < -0.5) or np.any(x0 > 1.5):
            x0 = np.full(N, 0.5)
        else:
            x0 = project_box_sum(x0, N)
    try:
        seed = int(seed) % (2 ** 32)
    except Exception:
        seed = 42
    rng = np.random.default_rng(seed)
    if N == 1:
        h = np.array([0.5])
        return (h, float(h[0] * (1.0 - h[0]) * dx), 1)
    starts = gen_starts(x0, N, rng)
    best_x = starts[0].copy()
    best_F = score_exact(best_x, N)
    for s in starts[1:]:
        F = score_exact(s, N)
        if F < best_F:
            best_F = F
            best_x = s.copy()
    if N <= 1000:
        if N <= 100:
            n_st, mi = min(len(starts), 6), 20
        elif N <= 250:
            n_st, mi = min(len(starts), 5), 10
        elif N <= 500:
            n_st, mi = min(len(starts), 4), 6
        else:
            n_st, mi = min(len(starts), 3), 3
        try:
            prob_objs = build_prob(N, dx)
        except Exception:
            prob_objs = None
        if prob_objs is not None:
            for s in starts[:n_st]:
                try:
                    xs, Fs = local_solve(s, N, dx, prob_objs, mi)
                    if Fs < best_F:
                        best_F = Fs
                        best_x = xs.copy()
                except Exception:
                    pass
    h_final = project_box_sum(best_x, N)
    F_final = score_exact(h_final, N)
    if not np.isfinite(F_final):
        h_final = np.full(N, 0.5)
        F_final = score_exact(h_final, N)
    return (h_final, float(F_final), N)
