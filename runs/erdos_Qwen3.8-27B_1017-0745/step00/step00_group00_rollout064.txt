            resid = resid + sub
    return y

def check_feasible(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score(h):
    n = len(h)
    if not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def lag_values(h, n):
    Q = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        total = 0.0
        i_lo = max(0, k)
        i_hi = min(n - 1, n - 1 + k)
        for i in range(i_lo, i_hi + 1):
            j = i - k
            total += h[i] * (1.0 - h[j])
        Q[k + n - 1] = total
    return Q

def lag_gradients(h, n):
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        ki = k + n - 1
        for r in range(n):
            val = 0.0
            if 0 <= r - k < n:
                val += (1.0 - h[r - k])
            if 0 <= r + k < n:
                val -= h[r + k]
            G[ki, r] = val
    return G

def slsqp_objective(z):
    return z[-1]

def slsqp_jac(z):
    n = len(z) - 1
    grad = np.zeros(n + 1)
    grad[-1] = 1.0
    return grad

def slsqp_ineq(z):
    n = len(z) - 1
    h = z[:n]
    t = z[-1]
    Q = lag_values(h, n)
    return t - Q / n

def slsqp_ineq_jac(z):
    n = len(z) - 1
    h = z[:n]
    G = lag_gradients(h, n)
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G / n
    J[:, n] = 1.0
    return J

def slsqp_eq(z):
    n = len(z) - 1
    h = z[:n]
    return np.sum(h) / n - 0.5

def slsqp_eq_jac(z):
    n = len(z) - 1
    J = np.zeros(n + 1)
    J[:n] = 1.0 / n
    return J

def run_slsqp(h0, n):
    Q0 = lag_values(h0, n)
    t0 = min(1.0, float(np.max(Q0)) / n + 1e-4)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * (n + 1)
    cons = [
        {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    try:
        res = minimize(slsqp_objective, z0, jac=slsqp_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-10, 'disp': False})
        if res.x is not None and len(res.x) == n + 1 and np.all(np.isfinite(res.x)):
            return res.x[:n]
    except Exception:
        pass
    return None

def run(seed=42, **kwargs):
    n = 45
    parent = None
    if 'initial_h_values' in globals():
        pv = np.array(initial_h_values, dtype=np.float64)
        if pv.ndim == 1 and 2 <= len(pv) <= 999 and np.all(np.isfinite(pv)):
            if np.all(pv >= -1e-6) and np.all(pv <= 1.0 + 1e-6):
                if abs(np.sum(pv) - len(pv) / 2.0) < 1e-6:
                    n = len(pv)
                    pv = np.clip(pv, 0.0, 1.0)
                    pv = project_box_sum(pv, n / 2.0)
                    parent = pv.copy()
    rng = np.random.default_rng(seed)
    starts = []
    if parent is not None:
        starts.append(parent.copy())
        starts.append(parent[::-1].copy())
        for sd in [0.01, 0.03, 0.06, 0.12, 0.20, 0.05]:
            p = parent + sd * rng.standard_normal(n)
            p = project_box_sum(p, n / 2.0)
            starts.append(p)
        for _ in range(4):
            u = rng.random(n)
            u = project_box_sum(u, n / 2.0)
            starts.append(u)
    else:
        starts.append(np.full(n, 0.5))
        for _ in range(8):
            u = rng.random(n)
            u = project_box_sum(u, n / 2.0)
            starts.append(u)
        for A, m, ph in [(0.25, 1, 0.37), (0.35, 2, 0.74), (0.45, 3, 1.11)]:
            idx = np.arange(n)
            u = 0.5 + A * np.cos(2.0 * np.pi * m * idx / n + ph)
            u = project_box_sum(u, n / 2.0)
            starts.append(u)
    best_h = None
    best_score = float('inf')
    if parent is not None:
        best_h = parent.copy()
        best_score = exact_score(parent)
    for h0 in starts:
        if not check_feasible(h0, n):
            h0 = project_box_sum(h0, n / 2.0)
            if not check_feasible(h0, n):
                continue
        sc = exact_score(h0)
        if sc < best_score:
            best_score = sc
            best_h = h0.copy()
        result = run_slsqp(h0, n)
        if result is not None:
            result = project_box_sum(result, n / 2.0)
            if check_feasible(result, n):
                sc2 = exact_score(result)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = result.copy()
    sigma = 0.05
    for _ in range(4):
        if best_h is None:
            break
        perturbed = best_h + sigma * rng.standard_normal(n)
        perturbed = project_box_sum(perturbed, n / 2.0)
        if not check_feasible(perturbed, n):
            sigma = min(0.25, 1.5 * sigma)
            continue
        result = run_slsqp(perturbed, n)
        improved = False
        if result is not None:
            result = project_box_sum(result, n / 2.0)
            if check_feasible(result, n):
                sc = exact_score(result)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_h = result.copy()
                    improved = True
        if improved:
            sigma = max(0.01, 0.7 * sigma)
        else:
            sigma = min(0.25, 1.5 * sigma)
    if best_h is not None:
        best_h = project_box_sum(best_h, n / 2.0)
        if check_feasible(best_h, n):
            best_score = exact_score(best_h)
        else:
            best_h = None
    if best_h is None:
        if parent is not None:
            best_h = parent.copy()
            best_score = exact_score(parent)
        else:
            best_h = np.full(n, 0.5)
            best_score = exact_score(best_h)
    return (best_h.astype(np.float64), best_score, n)
