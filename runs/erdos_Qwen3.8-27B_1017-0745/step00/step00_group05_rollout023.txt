def capped_simplex_project(v, target):

            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0).copy()

            break
        if r > 0:
            cap = 1.0 - y
            tot = float(np.sum(cap))
            if tot < 1e-15:
                break
            y += r * cap / tot
        else:
            tot = float(np.sum(y))
            if tot < 1e-15:
                break
            y += r * y / tot
        np.clip(y, 0.0, 1.0, out=y)
    if not np.all(np.isfinite(y)) or y.min() < -1e-12 or y.max() > 1.0 + 1e-12:
        return None
    if abs(float(np.sum(y)) - target) > 1e-9:
        return None
    return y


def exact_score(h, dx):
    h = np.asarray(h, dtype=np.float64)
    if h.ndim != 1 or not np.all(np.isfinite(h)):
        return np.inf
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))


def smooth_objective(x, tau, dx):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    s = corr * dx
    m = float(np.max(s))
    e = np.exp((s - m) / tau)
    se = float(np.sum(e))
    F = m + tau * np.log(se)
    p = e / se
    grad = np.zeros(n, dtype=np.float64)
    for j in range(2 * n - 1):
        t = j - (n - 1)
        pj = p[j]
        if pj < 1e-15:
            continue
        c_lo1 = max(0, t)
        c_hi1 = min(n, t + n)
        if c_hi1 > c_lo1:
            grad[c_lo1:c_hi1] += pj * (1.0 - x[c_lo1 - t:c_hi1 - t])
        c_lo2 = max(0, -t)
        c_hi2 = min(n, n - t)
        if c_hi2 > c_lo2:
            grad[c_lo2:c_hi2] -= pj * x[c_lo2 + t:c_hi2 + t]
    return F, grad


def _slsqp_fun(x, tau, dx):
    val, _ = smooth_objective(x, tau, dx)
    return val


def _slsqp_jac(x, tau, dx):
    _, grad = smooth_objective(x, tau, dx)
    return grad


def _eq_fun(x):
    return [float(np.sum(x)) - len(x) / 2.0]


def _eq_jac(x):
    return [np.ones(len(x), dtype=np.float64)]
