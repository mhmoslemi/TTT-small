import numpy as np

def project_box_sum(z, N):
    m = N / 2.0
    if z.shape != (N,) or not np.isfinite(z).all() or np.abs(z).max() > 1e6:
        return np.full(N, 0.5)
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    e = m - h.sum()
    if abs(e) > 1e-14:
        idx = np.arange(N)
        if e > 0:
            caps = 1.0 - h
            for i in idx:
                adj = min(e, caps[i])
                h[i] += adj
                e -= adj
                if e <= 1e-15:
                    break
        else:
            for i in idx:
                adj = min(-e, h[i])
                h[i] -= adj
                e += adj
                if e >= -1e-15:
                    break
    h = np.clip(h, 0.0, 1.0)
    if not np.isfinite(h).all() or abs(h.sum() - m) > 1e-10 * max(1, N):
        return np.full(N, 0.5)
    return h

def lag_values(h, N):
    M = 2 * N - 1
    q = np.zeros(M)
    for k in range(-(N - 1), N):
        idx = k + N - 1
        if k >= 0:
            L = N - k
            q[idx] = np.sum(h[k:k + L] * (1.0 - h[0:L]))
        else:
            d = -k
            L = N - d
            q[idx] = np.sum(h[0:L] * (1.0 - h[d:d + L]))
    return q

def smooth_value(h, tau, N):
    q = lag_values(h, N)
    r = (2.0 / N) * q
    rmax = r.max()
    ex = np.exp((r - rmax) / tau)
    return rmax + tau * np.log(ex.sum())

def smooth_value_grad(h, tau, N):
    q = lag_values(h, N)
    r = (2.0 / N) * q
    rmax = r.max()
    ex = np.exp((r - rmax) / tau)
    p = ex / ex.sum()
    phi = rmax + tau * np.log(ex.sum())
    grad = np.zeros(N)
    for k in range(-(N - 1), N):
        idx = k + N - 1
        w = (2.0 / N) * p[idx]
        if w < 1e-18:
            continue
        if k >= 0:
            L = N - k
            a = h[k:k + L]
            b = h[0:L]
            grad[k:k + L] += w * (1.0 - b)
            grad[0:L] -= w * a
        else:
            d = -k
            L = N - d
            a = h[0:L]
            b = h[d:d + L]
            grad[0:L] += w * (1.0 - b)
            grad[d:d + L] -= w * a
    return phi, grad

def score_exact(h, N):
    if h.shape != (N,) or not np.isfinite(h).all():
        return np.inf
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * (2.0 / N))
