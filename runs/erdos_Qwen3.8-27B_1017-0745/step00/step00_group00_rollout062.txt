    bounds = [(0.0, 1.0)] * (n + 1)

    if parent is not None:
        best = parent.copy()
    else:
        best = np.full(n, 0.5)
    best_score = exact_score(best)

    starts = [best.copy()]
    if parent is not None:
        starts.append(parent[::-1].copy())
        for sd in [0.01, 0.03, 0.06, 0.12, 0.20, 0.05]:
            v = parent + sd * rng.standard_normal(n)
            p = project_box_sum(v, S)
            if p is not None and check_feasible(p, n):
                starts.append(p)
    n_rand = 4 if parent is not None else 8
    for _ in range(n_rand):
        v = rng.uniform(0.0, 1.0, n)
        p = project_box_sum(v, S)
        if p is not None and check_feasible(p, n):
            starts.append(p)
    if parent is None:
        for A, m, ph in [(0.25, 1, 0.37), (0.35, 2, 0.74), (0.45, 3, 1.11)]:
            idx = np.arange(n)
            v = 0.5 + A * np.cos(2.0 * np.pi * m * idx / n + ph)
            p = project_box_sum(v, S)
            if p is not None and check_feasible(p, n):
                starts.append(p)

    for h0 in starts:
        h1 = solve_from(h0, n, S, bounds)
        if h1 is not None:
            sc = exact_score(h1)
            if sc < best_score - 1e-15:
                best_score = sc
                best = h1.copy()

    sigma = 0.05
    for _ in range(4):
        h0 = best + sigma * rng.standard_normal(n)
        p = project_box_sum(h0, S)
        if p is None or not check_feasible(p, n):
            break
        h1 = solve_from(p, n, S, bounds)
        improved = False
        if h1 is not None:
            sc = exact_score(h1)
            if sc < best_score - 1e-12:
                best_score = sc
                best = h1.copy()
                improved = True
        if improved:
            sigma = max(0.01, 0.7 * sigma)
        else:
            sigma = min(0.25, 1.5 * sigma)

    p = project_box_sum(best, S)
    if p is not None and check_feasible(p, n):
        best = p
    else:
        if parent is not None:
            best = parent
        else:
            best = np.full(n, 0.5)
    c5 = exact_score(best)
    if not np.isfinite(c5):
        c5 = 1.0
    return best, c5, n
