            y[m] += r * y[m] / np.sum(y[m])
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(np.sum(y) - target) > 1e-10:
        return None
    return y

def exact_score(h, dx):
    n = len(h)
    if n < 1 or np.any(~np.isfinite(h)) or np.min(h) < -1e-12 or np.max(h) > 1 + 1e-12:
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def smooth_fun(x, tau, dx):
    s = dx * np.correlate(x, 1.0 - x, mode='full')
    m = np.max(s)
    return m + tau * np.log(np.sum(np.exp((s - m) / tau)))

def smooth_grad(x, tau, dx):
    n = len(x)
    s = dx * np.correlate(x, 1.0 - x, mode='full')
    m = np.max(s)
    e = np.exp((s - m) / tau)
    p = e / np.sum(e)
    lags = np.arange(-(n - 1), n)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        t = lags[k]
        w = p[k] * dx
        if abs(w) < 1e-16:
            continue
        if t >= 0:
            i1 = np.arange(t, n)
            g[i1] += w * (1.0 - x[i1 - t])
            i2 = np.arange(0, n - t)
            g[i2] -= w * x[i2 + t]
        else:
            tt = -t
            i1 = np.arange(0, n - tt)
            g[i1] += w * (1.0 - x[i1 + tt])
            i2 = np.arange(tt, n)
            g[i2] -= w * x[i2 - tt]
    return g

def mass_eq(x, target):
    return np.sum(x) - target

def mass_eq_jac(x, target):
    return np.ones(x.shape)

def optimize_start(cur, n, dx, target, bounds, cons, tau_schedule, best_h, best_c):
    for tau in tau_schedule:
        try:
            res = minimize(smooth_fun, cur, method='SLSQP', jac=smooth_grad,
                           args=(tau, dx), bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-10, 'disp': False})
            x = res.x
            if x.shape == (n,) and np.all(np.isfinite(x)):
                x = capped_simplex_project(x, target)
                if x is not None:
                    c = exact_score(x, dx)
                    if np.isfinite(c) and c < best_c - 1e-12:
                        best_c = c
                        best_h = x.copy()
                    f_new = smooth_fun(x, tau, dx)
                    f_old = smooth_fun(cur, tau, dx)
                    if np.isfinite(f_new) and f_new <= f_old + 1e-10:
                        cur = x.copy()
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    n = 70
    parent = None
    try:
        _pv = np.array(initial_h_values, dtype=np.float64).ravel()
        if 2 <= len(_pv) <= 999 and np.all(np.isfinite(_pv)):
            n = len(_pv)
            if np.min(_pv) >= -1e-10 and np.max(_pv) <= 1 + 1e-10:
                parent = _pv.copy()
    except Exception:
        pass
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': mass_eq, 'jac': mass_eq_jac, 'args': (target,)}]
    tau_schedule = [0.02, 0.005, 0.001, 0.0002, 0.00005, 0.00001]
    if parent is not None:
        parent = capped_simplex_project(parent, target)
    best_h, best_c = None, np.inf
    if parent is not None:
        c = exact_score(parent, dx)
        if np.isfinite(c):
            best_h, best_c = parent.copy(), c
    starts = []
    if parent is not None:
        starts.append(parent)
        starts.append(parent[::-1].copy())
        for sig in [0.04, 0.10, 0.22, 0.40]:
            p = capped_simplex_project(parent + sig * rng.standard_normal(n), target)
            if p is not None:
                starts.append(p)
    for _ in range(2):
        p = capped_simplex_project(rng.random(n), target)
        if p is not None:
            starts.append(p)
    p = capped_simplex_project(rng.beta(0.5, 0.5, n), target)
    if p is not None:
        starts.append(p)
    starts.append(np.full(n, 0.5))
    for s in starts:
        c = exact_score(s, dx)
        if np.isfinite(c) and c < best_c - 1e-12:
            best_c, best_h = c, s.copy()
    for s in starts:
        best_h, best_c = optimize_start(s.copy(), n, dx, target, bounds, cons,
                                        tau_schedule, best_h, best_c)
    center = best_h if best_h is not None else np.full(n, 0.5)
    for sig in [0.02, 0.08, 0.20]:
        s = capped_simplex_project(center + sig * rng.standard_normal(n), target)
        if s is None:
            continue
        best_h, best_c = optimize_start(s.copy(), n, dx, target, bounds, cons,
                                        tau_schedule, best_h, best_c)
        center = best_h if best_h is not None else center
    if best_h is None:
        best_h = np.full(n, 0.5)
    best_h = capped_simplex_project(best_h, target)
    if best_h is None:
        best_h = np.full(n, 0.5)
    best_c = exact_score(best_h, dx)
    return (best_h, best_c, n)
