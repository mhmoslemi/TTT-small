def project_box_sum(v, S):
    n = len(v)
    if n == 0:
        return None
    low = np.min(v) - 1.0
    high = np.max(v)
    for _ in range(80):
        mid = 0.5 * (low + high)
        y = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(y)
        if s > S:
            low = mid
        else:
            high = mid
    y = np.clip(v - 0.5 * (low + high), 0.0, 1.0)
    # Correct residual
    res = S - np.sum(y)
    for i in range(n):
        if res > 0:
            d = min(res, 1.0 - y[i])
            y[i] += d
            res -= d
        else:
            d = min(-res, y[i])
            y[i] -= d
            res += d
        if abs(res) < 1e-14:
            break

def quad_roots(a, b, c):
    # a*x^2 + b*x + c = 0
    roots = []
    if abs(a) < 1e-16:
        if abs(b) < 1e-16:
            return roots
        roots.append(-c / b)
    else:
        disc = b * b - 4.0 * a * c
        tol = 64.0 * np.finfo(float).eps * max(1.0, b * b, abs(4.0 * a * c))
        if disc < -tol:
            return roots
        disc = max(0.0, disc)
        sq = math.sqrt(disc)
        q = -0.5 * (b + math.copysign(sq, b))
        if abs(q) > 1e-16:
            roots.append(q / a)
            roots.append(c / q)
        else:
            roots.append(-b / (2.0 * a))
    return [r for r in roots if np.isfinite(r)]

import numpy as np
import math

def project_box_sum(v, S):
    n = len(v)
    if n == 0:
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    res = S - float(np.sum(y))
    for i in range(n):
        if res > 1e-14:
            d = min(res, 1.0 - y[i])
            y[i] += d; res -= d
        elif res < -1e-14:
            d = min(-res, y[i])
            y[i] -= d; res += d
    if abs(S - float(np.sum(y))) > 1e-10 * max(1, n):
        return None
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1 + 1e-12):
        return None
    return np.clip(y, 0.0, 1.0)

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1 + 1e-12):
        return False
    return abs(float(np.sum(h)) - n / 2.0) <= 1e-10 * max(1, n)

def exact_score_and_corr(h):
    n = len(h)
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return None, None
    corr = np.correlate(h, 1.0 - h, mode='full')
    score = float(np.max(corr) * (2.0 / n))
    if not np.isfinite(score):
        return None, None
    return score, corr

def make_pair_coeffs(h, corr, a, b):
    n = len(h)
    m = 2 * n - 1
    A = corr.copy()
    B = np.zeros(m)
    D = np.zeros(m)
    diff = b - a
    for k in range(-(n - 1), n):
        idx = k + n - 1
        bk = 0.0
        ia = a - k
        if 0 <= ia < n:
            bk += (1.0 - h[ia])
        ib = b - k
        if 0 <= ib < n:
            bk -= (1.0 - h[ib])
        j1 = a + k
        if 0 <= j1 < n:
            bk -= h[j1]
        j2 = b + k
        if 0 <= j2 < n:
            bk += h[j2]
        B[idx] = bk
        if k == 0:
            D[idx] = -2.0
        elif k == diff or k == -diff:
            D[idx] = 1.0
    if not np.all(np.isfinite(B)):
        return None, None, None
    return A, B, D

def build_upper_hull(A, B, T_set, lo, hi):
    m = len(A)
    lines = [(B[i], A[i]) for i in range(m) if i not in T_set]
    if not lines:
        return [(lo, hi, None, None)]
    lines.sort(key=lambda x: (x[0], -x[1]))
    unique = []
    for s in lines:
        if unique and abs(unique[-1][0] - s[0]) < 1e-15:
            if s[1] > unique[-1][1]:
                unique[-1] = s
        else:
            unique.append(s)

    for sl in unique:

        if not hull:
            hull.append([mm, cc, -1e18])

def quad_roots(c2, c1, c0):
    roots = []
    if abs(c2) < 1e-16:
        if abs(c1) > 1e-16:
            r = -c0 / c1
            if np.isfinite(r):
                roots.append(r)
        return roots
    disc = c1 * c1 - 4.0 * c2 * c0
    tol = 64.0 * np.finfo(float).eps * max(1.0, c1 * c1, abs(4.0 * c2 * c0))
    if disc < -tol:
        return roots
    disc = max(0.0, disc)
    sq = math.sqrt(disc)
    q = -0.5 * (c1 + math.copysign(sq, c1))
    if abs(q) > 1e-16:
        for r in (q / c2, c0 / q):
            if np.isfinite(r):
                roots.append(r)
    else:
        r = -c1 / (2.0 * c2)
        if np.isfinite(r):
            roots.append(r)
    return roots

def poly_eval(alpha, A, B, D):
    vals = A + B * alpha + D * alpha * alpha
    return float(np.max(vals))

def line_minimize_pair(h, corr, cur_score, a, b, budget, scalls):

    lo = max(-h[a], h[b] - 1.0)
    hi = min(1.0 - h[a], h[b])

    A, B, D = make_pair_coeffs(h, corr, a, b)
    if A is None:
        return h, corr, cur_score, scalls
    diff = b - a

    seg_tol = 1e-12 * max(1.0, abs(lo), abs(hi))
    cands = set()

        cands.add(s0); cands.add(s1)

            cands.add(0.0)

                st = -B[ti] / (2.0 * D[ti])
                if s0 - seg_tol <= st <= s1 + seg_tol:
                    cands.add(max(s0, min(s1, st)))
        for i in range(len(T)):

                rs = quad_roots(D[T[i]] - D[T[j]], B[T[i]] - B[T[j]], A[T[i]] - A[T[j]])
                for r in rs:
                    if s0 - seg_tol <= r <= s1 + seg_tol:
                        cands.add(max(s0, min(s1, r)))
        if ml is not None:
            for ti in T:
                rs = quad_roots(D[ti], B[ti] - ml, A[ti] - cl)
                for r in rs:
                    if s0 - seg_tol <= r <= s1 + seg_tol:
                        cands.add(max(s0, min(s1, r)))
    clist = sorted(cands)

        if sc is not None and sc < best_sc - 1e-12 * max(1.0, abs(best_sc)):

def cosine_profile(n, m, phase):
    i = np.arange(n, dtype=np.float64)
    return 0.5 + 0.45 * np.cos(2.0 * np.pi * m * i / n + phase)

def alt_profile(n):
