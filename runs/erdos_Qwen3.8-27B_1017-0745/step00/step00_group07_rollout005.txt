    if n < 2 or not np.all(np.isfinite(h)):
        return None
    dx = 2.0 / n
    s = float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))
    return s if np.isfinite(s) else None

def lag_model(x):
    n = len(x)
    m = 2 * n - 1
    B = np.zeros(m)
    G = np.zeros((m, n))
    for idx in range(m):
        d = idx - (n - 1)
        lo = max(0, -d)
        hi = min(n, n - d)
        s1 = x[lo:hi]
        s2 = x[lo + d:hi + d]
        B[idx] = np.dot(s1, 1.0 - s2)
        G[idx, lo:hi] += 1.0 - s2
        G[idx, lo + d:hi + d] -= s1
    return B, G

def build_starts(n, S, rng, parent):
    starts = []
    lo_b, hi_b = np.zeros(n), np.ones(n)
    def try_add(v):
        if v is None or not is_feasible(v, S):
            return
        for s in starts:
            if np.max(np.abs(s - v)) < 1e-10:
                return
        starts.append(v.copy())
    if parent is not None:
        p = project_box_sum(parent, S, lo_b, hi_b)
        if p is not None:
            try_add(p)
            try_add(p[::-1].copy())
    else:
        try_add(np.full(n, 0.5))
    K = max(1, min(6, (n - 1) // 2))
    for _ in range(2):
        q = int(rng.integers(1, K + 1))
        ph = float(rng.uniform(0, 2 * np.pi))
        z = 0.5 + 0.45 * np.cos(2 * np.pi * q * np.arange(n) / n + ph)
        try_add(project_box_sum(z, S, lo_b, hi_b))
    w = rng.standard_normal(n)
    wf = np.fft.rfft(w)
    kf = np.arange(len(wf))
    wf *= np.exp(-0.5 * (kf / max(1.0, K / 2.0)) ** 2)
    noise = np.fft.irfft(wf, n)
    std = np.std(noise)
    if std > 1e-12:
        noise /= std
    try_add(project_box_sum(0.5 + 0.35 * noise, S, lo_b, hi_b))
    perm = rng.permutation(n)
    z = np.zeros(n)
    z[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        z[perm[n // 2]] = 0.5
    try_add(z)
    cap = 6 if n <= 128 else (4 if n <= 256 else 2)
    return starts[:cap]

def slp_optimize(x0, S, dx, max_iter):
    n = len(x0)
    x = x0.copy()
    score_x = exact_score(x)
    if score_x is None:
        return x, score_x, x, score_x
    best_x, best_s = x.copy(), score_x
    Delta = 0.20
    Dmin, Dmax = 1e-4, 0.5
    lp_fail, consec = 0, 0
    for _ in range(max_iter):
        if Delta < Dmin:
            break
        lower = np.clip(x - Delta, 0.0, 1.0)
        upper = np.clip(x + Delta, 0.0, 1.0)
        if np.sum(lower) > S + 1e-9 or np.sum(upper) < S - 1e-9:
            Delta = max(0.25 * Delta, Dmin)
            continue
        B, G = lag_model(x)
        m = len(B)
        A_ub = np.hstack([G, -np.ones((m, 1))])
        b_ub = G @ x - B
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([S])
        c = np.zeros(n + 1)
        c[n] = 1.0
        bounds = [(lower[i], upper[i]) for i in range(n)] + [(None, None)]
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'max_iter': 500})
        except Exception:
            lp_fail += 1
            Delta = max(0.25 * Delta, Dmin)
            if lp_fail >= 3:
                break
            continue
        if res.status != 0 or not np.all(np.isfinite(res.x)):
            lp_fail += 1
            Delta = max(0.25 * Delta, Dmin)
            if lp_fail >= 3:
                break
            continue
        y = project_box_sum(res.x[:n].copy(), S, lower, upper)
        if y is None:
            y = project_box_sum(res.x[:n].copy(), S, np.zeros(n), np.ones(n))
        if y is None or not is_feasible(y, S):
            Delta = max(0.25 * Delta, Dmin)
            continue
        Qm = np.max(B + G @ (y - x))
        pred = dx * (np.max(B) - Qm)
        score_y = exact_score(y)
        if score_y is None:
            Delta = max(0.25 * Delta, Dmin)
            continue
        if score_y < best_s - 1e-14 * max(1.0, abs(best_s)):
            best_s = score_y
            best_x = y.copy()
        actual = score_x - score_y
        ptol = 1e-12 * max(1.0, abs(score_x))
        if pred <= ptol:
            if actual > 0:
                x, score_x = y.copy(), score_y
            break
        rho = actual / pred
        if actual > 1e-12 * max(1.0, abs(score_x)) and rho >= 0.1:
            lp_fail = 0
            step = np.max(np.abs(y - x))
            if rho >= 0.75 and step >= 0.8 * Delta:
                Delta = min(2.0 * Delta, Dmax)
            elif rho < 0.25:
                Delta = max(0.25 * Delta, Dmin)
            if step < 1e-10:
                x, score_x = y, score_y
                break
            consec = consec + 1 if abs(score_x - score_y) < 1e-10 * max(1.0, abs(score_x)) else 0
            if consec >= 8:
                x, score_x = y.copy(), score_y
                break
            x, score_x = y, score_y
        else:
            Delta = max(0.25 * Delta, Dmin)
    return x, score_x, best_x, best_s

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = None
    try:
        pv = np.array(initial_h_values, dtype=np.float64)
        if pv.ndim == 1 and len(pv) >= 2 and np.all(np.isfinite(pv)):
            parent = pv.copy()
    except Exception:
        parent = None
    if parent is not None:
        n = len(parent)
    else:
        n = int(kwargs.get('n_points', 80))
        if n < 8 or n > 512:
            n = 80
    dx = 2.0 / n
    S = n / 2.0
    h0 = np.full(n, 0.5)
    best_h, best_score = h0.copy(), exact_score(h0)
    if parent is not None:
        hp = project_box_sum(parent, S, np.zeros(n), np.ones(n))
        if hp is not None:
            sp = exact_score(hp)
            if sp is not None and sp < best_score:
                best_score, best_h = sp, hp.copy()
    if n > 512:
        return best_h, float(best_score), n
    max_iter = 36 if n <= 128 else (24 if n <= 256 else 12)
    starts = build_starts(n, S, rng, parent)
    if not starts:
        starts = [h0.copy()]
    for s in starts:
        _, _, bx, bs = slp_optimize(s, S, dx, max_iter)
        if bs is not None and bs < best_score - 1e-14 * max(1.0, abs(best_score)):
            best_score, best_h = bs, bx.copy()
    final_s = exact_score(best_h)
    if final_s is None:
        best_h, final_s = h0.copy(), exact_score(h0)
    return best_h, float(final_s), n
