def project_capped_simplex(z, S):
    n = len(z)
    if not np.all(np.isfinite(z)) or np.max(np.abs(z)) > 1e12:
        return None
    low = np.min(z) - 1.0
    high = np.max(z)
    for _ in range(80):
        mid = low + 0.5 * (high - low)
        y = np.clip(z - mid, 0.0, 1.0)
        if np.sum(y) > S:
            low = mid
        else:
            high = mid
    y = np.clip(z - (low + 0.5 * (high - low)), 0.0, 1.0)
    r = S - np.sum(y)
    if r > 1e-12:
        for i in range(n):
            if r <= 1e-15:
                break
            sl = 1.0 - y[i]
            if sl > 0:
                add = min(r, sl)
                y[i] += add
                r -= add
    elif r < -1e-12:
        for i in range(n):
            if r >= -1e-15:
                break
            sl = y[i]
            if sl > 0:
                sub = min(-r, sl)
                y[i] -= sub
                r += sub

        return None
    y = np.clip(y, 0.0, 1.0)
    if abs(np.sum(y) - S) > 1e-10:
        return None
    return y

def compute_lags(h):
    n = len(h)
    B = np.zeros(2 * n - 1)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        src = h[lo:hi]
        dst = h[lo + d:hi + d]
        B[row] = np.dot(src, 1.0 - dst)
        G[row, lo:hi] += (1.0 - dst)
        G[row, lo + d:hi + d] -= src
    return B, G

        return None
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    if not np.all(np.isfinite(vals)):
        return None
    sc = float(np.max(vals))
    return sc if np.isfinite(sc) else None

    return z[-1] - B

    n = len(z) - 1
    _, G = compute_lags(z[:-1])
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
