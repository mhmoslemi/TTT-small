            npop = [None] * M
            ns = [999.0] * M
            for i in range(M):
                others = [j for j in range(M) if j != i]
                ch = rng.choice(len(others), size=3, replace=False)
                r1, r2, r3 = others[ch[0]], others[ch[1]], others[ch[2]]
                F = rng.uniform(0.45, 0.90)
                CR = rng.uniform(0.55, 0.95)
                v = snap[r1] + F * (snap[r2] - snap[r3])
                mask = rng.random(n) < CR
                jrand = rng.integers(0, n)
                mask[jrand] = True
                z = np.where(mask, v, snap[i])
                if not np.all(np.isfinite(z)):
                    npop[i] = snap[i]
                    ns[i] = snap_s[i]
                    continue
                u = project_capped(z, S)
                if u is None or not is_feasible(u, S):
                    npop[i] = snap[i]
                    ns[i] = snap_s[i]
                    continue
                bc = u
                bs = exact_score(u)
                if bs is None:
                    npop[i] = snap[i]
                    ns[i] = snap_s[i]
                    continue
                if n <= 128:
                    for alpha in (0.25, 0.50, 0.75):
                        seg = snap[i] + alpha * (u - snap[i])
                        seg = project_capped(seg, S)
                        if seg is not None and is_feasible(seg, S):
                            ss = exact_score(seg)
                            if ss is not None and ss < bs:
                                bc = seg
                                bs = ss
                if bs < snap_s[i]:
                    npop[i] = bc
                    ns[i] = bs
                    if bs < best_s:
                        best = bc.copy()
                        best_s = bs
                else:
                    npop[i] = snap[i]
                    ns[i] = snap_s[i]
            pop = npop
            pop_s = ns
            tol = 1e-13 * max(1.0, abs(base_sc))
            if best_s < base_sc - tol:
                stag = 0
                base_sc = best_s
            else:
                stag += 1
            if stag >= 18 and restarts < 2:
                elites = max(2, M // 4)
                for k in range(elites, M):
                    t = (restarts * M + k) % 5
                    if t < 4:
                        sg = [0.05, 0.15, 0.35, 0.60][t]
                        z = best + sg * rng.standard_normal(n)
                    else:
                        q = rng.integers(1, min(8, max(1, (n - 1) // 2)) + 1)
                        ph = rng.uniform(0, 2 * math.pi)
                        z = 0.5 + 0.45 * np.cos(2 * math.pi * q * np.arange(n) / n + ph)
                    u = project_capped(z, S)
                    if u is not None and is_feasible(u, S):
                        s = exact_score(u)
                        if s is not None:
                            pop[k] = u
                            pop_s[k] = s
                            if s < best_s:
                                best = u.copy()
                                best_s = s
                            else:
                                pop[k] = best.copy()
                                pop_s[k] = best_s
                        else:
                            pop[k] = best.copy()
                            pop_s[k] = best_s
                    else:
                        pop[k] = best.copy()
                        pop_s[k] = best_s
                stag = 0
                restarts += 1
            elif stag >= 42 and restarts >= 2:
                break
    pool_h = [best.copy(), const_fb.copy()]
    if s_parent is not None:
        pool_h.append(parent_fb.copy())
    final_h = const_fb.copy()
    final_s = s_const
    for hc in pool_h:
        hf = project_capped(hc, S)
        if hf is None or not is_feasible(hf, S):
            continue
        sf = exact_score(hf)
        if sf is not None and sf < final_s:
            final_h = hf
            final_s = sf
    final_s = exact_score(final_h)
    if final_s is None:
        final_h = const_fb
        final_s = exact_score(const_fb)
    return final_h, final_s, int(n)
