                starts.append(pr)
    starts.append(np.full(n, 0.5, dtype=np.float64))
    for _ in range(3):
        pr = _project(rng.uniform(0, 1, n), tgt)
        if pr is not None:
            starts.append(pr)
    for _ in range(2):
        pr = _project(rng.beta(0.5, 0.5, n), tgt)
        if pr is not None:
            starts.append(pr)

    all_starts = list(starts)
    for sig in (0.02, 0.08, 0.20):
        ct = best_h + sig * rng.standard_normal(n)
        pr = _project(ct, tgt)
        if pr is not None:
            all_starts.append(pr)

    for h0 in all_starts:
        cur = h0.copy()
        for tau in tau_sched:
            try:
                res = minimize(_lsq_obj, cur, method='SLSQP', jac=True,
                               args=(tau, dx), bounds=bounds, constraints=cons,
                               options={'maxiter': 200, 'ftol': 1e-10, 'disp': False})
                if res is not None and hasattr(res, 'x'):
                    cand = np.asarray(res.x, dtype=np.float64)
                    if cand.shape == (n,) and np.all(np.isfinite(cand)):
                        pc = _project(cand, tgt)
                        if pc is not None and _feasible(pc, n):
                            sc = _score(pc, dx)
                            if sc < best_c - 1e-12:
                                best_h, best_c = pc.copy(), sc
                            fv, _ = _lsq_obj(pc, tau, dx)
                            cv, _ = _lsq_obj(cur, tau, dx)
                            if np.isfinite(fv) and (not np.isfinite(cv) or fv <= cv + 1e-10):
                                cur = pc.copy()
            except Exception:
                pass

    final = _project(best_h, tgt)
    if final is not None and _feasible(final, n):
        best_h = final
    best_c = _score(best_h, dx)
    if not np.isfinite(best_c):
        best_h = np.full(n, 0.5, dtype=np.float64)
        best_c = _score(best_h, dx)
    return best_h, best_c, n
