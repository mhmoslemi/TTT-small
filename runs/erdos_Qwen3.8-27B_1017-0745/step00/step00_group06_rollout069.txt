    a = np.max(R)
    e = np.exp((R - a) / tau)
    return J.T @ (e / np.sum(e))

def _eqf(h):
    return float(np.sum(h) - _P['m'])

def _eqj(h):
    return np.ones(_P['N'])

def _obj_e(z):
    return float(z[-1])

def _jace(z):
    g = np.zeros(_P['N'] + 1)
    g[-1] = 1.0
    return g

def _eqf_e(z):
    return float(np.sum(z[:-1]) - _P['m'])

def _eqj_e(z):
    N = _P['N']
    return np.concatenate([np.ones(N), [0.0]])

def _ineqf(z):
    R, _ = corr_jac(z[:-1], _P['N'], _P['d'])
    return z[-1] - R

def _ineqj(z):
    N = _P['N']
    _, J = corr_jac(z[:-1], N, _P['d'])
    return np.hstack([-J, np.ones((2 * N - 1, 1))])

def _score_key(c):
    return c[1]

def run(seed=42, **kwargs):
    global _P
    N, parent = None, None
    try:
        pv = np.asarray(initial_h_values, dtype=np.float64)
        if pv.ndim == 1 and pv.shape[0] >= 2 and np.all(np.isfinite(pv)):
            N, parent = int(pv.shape[0]), pv.copy()
    except Exception:
        pass
    if N is None:
        N = 95
        try:
            N = max(2, min(999, int(kwargs['n_points'])))
        except Exception:
            pass
    m, d = N / 2.0, 2.0 / N
    base = project_feasible(parent if parent is not None else np.full(N, 0.5), N, m)
    if N > 999:
        return base, exact_score(base), N
    incumbent, inc_score = base.copy(), exact_score(base)
    try:
        rng = np.random.default_rng(int(seed) & 0x7FFFFFFFFFFFFFFF)
    except Exception:
        rng = np.random.default_rng(42)
    starts = [incumbent.copy()]
    step = np.zeros(N)
    fl = int(np.floor(m))
    step[:fl] = 1.0
    if fl < N:
        step[fl] = m - fl
    starts.append(project_feasible(step, N, m))
    i_arr = np.arange(N, dtype=np.float64)
    for f in [1, 3, 7]:
        starts.append(project_feasible(0.5 + 0.22 * np.sin(2.0 * np.pi * f * i_arr / N), N, m))
    n_r = 8
    try:
        n_r = max(6, min(12, int(kwargs['n_restarts'])))
    except Exception:
        pass
    stds = [0.04, 0.10, 0.20]
    k = len(starts)
    while len(starts) < n_r:
        bv = incumbent if k % 2 == 0 else np.full(N, 0.5)
        nz = rng.standard_normal(N) * stds[k % 3]
        nz = 0.5 * nz + 0.25 * np.roll(nz, 1) + 0.25 * np.roll(nz, -1)
        starts.append(project_feasible(bv + nz, N, m))
        k += 1
    ded = [starts[0]]
    for s in starts[1:]:
        if not any(np.max(np.abs(s - t)) < 1e-8 for t in ded):
            ded.append(s)
    starts = ded[:n_r]
    maxit = 120
    try:
        maxit = max(60, min(250, int(kwargs['maxiter'])))
    except Exception:
        pass
    taus = [0.02, 0.005, 0.001, 0.0002]
    candidates = [(incumbent.copy(), inc_score)]
    bnds = [(0.0, 1.0)] * N
    cons_s = [{'type': 'eq', 'fun': _eqf, 'jac': _eqj}]
    for s in starts:
        cur = s.copy()
        for tau in taus:
            _P.update({'N': N, 'd': d, 'm': m, 'tau': tau})
            try:
                res = minimize(_obj_s, cur, jac=_jac_s, method='SLSQP',
                               bounds=bnds, constraints=cons_s,
                               options={'ftol': 1e-10, 'maxiter': maxit, 'disp': False})
                hn = np.asarray(res.x, dtype=np.float64)
                if hn.shape == (N,) and np.all(np.isfinite(hn)):
                    cur = project_feasible(hn, N, m)
            except Exception:
                pass
            candidates.append((cur.copy(), exact_score(cur)))
    candidates.sort(key=_score_key)
    top = []
    for c in candidates:
        if not any(np.max(np.abs(c[0] - t[0])) < 1e-8 for t in top):
            top.append(c)
        if len(top) >= 3:
            break
    epi_mi = min(400, 2 * maxit)
    for c in top:
        h0 = c[0].copy()
        z0 = np.concatenate([h0, [exact_score(h0) + 1e-6]])
        _P.update({'N': N, 'd': d, 'm': m})
        cons_e = [{'type': 'eq', 'fun': _eqf_e, 'jac': _eqj_e},
                  {'type': 'ineq', 'fun': _ineqf, 'jac': _ineqj}]
        bnds_e = [(0.0, 1.0)] * N + [(0.0, 2.1)]
        try:
            res = minimize(_obj_e, z0, jac=_jace, method='SLSQP',
                          bounds=bnds_e, constraints=cons_e,
                          options={'ftol': 1e-10, 'maxiter': epi_mi, 'disp': False})
            zh = np.asarray(res.x, dtype=np.float64)
            if zh.shape == (N + 1,) and np.all(np.isfinite(zh)):
                hn = project_feasible(zh[:-1], N, m)
                candidates.append((hn.copy(), exact_score(hn)))
        except Exception:
            pass
    candidates.sort(key=_score_key)
    if candidates[0][1] < inc_score - 1e-12:
        incumbent, inc_score = candidates[0][0].copy(), candidates[0][1]
    fh = project_feasible(incumbent, N, m)
    ok = (fh.shape == (N,) and np.all(np.isfinite(fh))
          and np.all(fh >= -1e-15) and np.all(fh <= 1.0 + 1e-15)
          and abs(np.sum(fh) - m) < 1e-9 * max(1, N))
    if ok:
        fh = np.clip(fh, 0.0, 1.0)
        return fh, exact_score(fh), N
    fh = np.full(N, 0.5)
    return fh, exact_score(fh), N
