        pv = np.array(initial_h_values, dtype=np.float64).ravel()
        if pv.size > 0 and np.all(np.isfinite(pv)):
            parent = pv
    except (NameError, ValueError, TypeError):
        parent = None

    if parent is not None:
        N = parent.size
    else:
        N = 70
        try:
            nv = kwargs.get('n_points', None)
            if nv is not None:
                ni = int(nv)
                if 1 <= ni <= 999:
                    N = ni
        except (ValueError, TypeError):
            pass

    if N == 1:
        h1 = np.array([0.5])
        return h1, score_exact(h1, 1), 1

    m = N / 2.0
    if parent is not None:
        if np.all(parent >= 0) and np.all(parent <= 1) and abs(np.sum(parent) - m) < 1e-10 * max(1, N):
            baseline = parent.copy()
        else:
            baseline = project_box_sum(parent.copy(), N)
    else:
        baseline = np.full(N, 0.5)

    inc_score = score_exact(baseline, N)
    inc_vec = baseline.copy()

    if N <= 100:
        n_starts, iters_ps = 8, 60
    elif N <= 200:
        n_starts, iters_ps = 6, 40
    elif N <= 500:
        n_starts, iters_ps = 4, 20
    else:
        n_starts, iters_ps = 3, 8

    starts = [baseline.copy()]
    starts.append(project_box_sum(baseline + 0.20 * rng.standard_normal(N), N))
    starts.append(project_box_sum(rng.uniform(0.0, 1.0, N), N))
    starts.append(project_box_sum(fourier_profile(N, rng), N))
    if parent is not None:
        starts.append(project_box_sum(1.0 - baseline, N))
    else:
        starts.append(project_box_sum(0.5 + 0.4 * rng.standard_normal(N), N))
    starts.append(np.full(N, 0.5))
    starts.append(project_box_sum(baseline + 0.08 * rng.standard_normal(N), N))
    starts.append(project_box_sum(baseline + 0.40 * rng.standard_normal(N), N))
    starts = starts[:n_starts]

    for s in starts:
        try:
            sc = score_exact(s, N)
            if sc < inc_score:
                inc_score = sc
                inc_vec[:] = s
            inc_vec, inc_score = optimize_start(s.copy(), N, inc_vec, inc_score, iters_ps)
        except Exception:
            continue

    inc_vec = np.clip(inc_vec, 0.0, 1.0)
    if inc_vec.shape != (N,) or not np.all(np.isfinite(inc_vec)):
        inc_vec = np.full(N, 0.5)
    if abs(np.sum(inc_vec) - m) > 1e-10 * max(1, N):
        inc_vec = project_box_sum(inc_vec, N)
    final_score = score_exact(inc_vec, N)
    if not np.isfinite(final_score):
        inc_vec = np.full(N, 0.5)
        final_score = score_exact(inc_vec, N)
    return inc_vec, float(final_score), int(N)
