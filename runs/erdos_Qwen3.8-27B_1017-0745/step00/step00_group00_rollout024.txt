def project_box_sum(v, S):
    n = len(v)
    if not np.all(np.isfinite(v)) or np.max(np.abs(v)) > 1e12:
        return None
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    y = np.clip(v - lam, 0.0, 1.0)
    resid = S - np.sum(y)
    if resid > 0:
        for i in range(n):
            add = min(resid, 1.0 - y[i])

                resid -= add

def exact_score(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr) * (2.0 / n))

def smooth_eval(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr) * (2.0 / n))
    q = corr / n
    z = q / tau
    zmax = np.max(z)
    e = np.exp(z - zmax)
    p = e / np.sum(e)
    F = float(tau * (zmax + np.log(np.sum(e))))
    s1 = np.convolve(p, 1.0 - h, mode='full')
    p_rev = p[::-1]
    s2 = np.convolve(h, p_rev, mode='full')
    grad = (s1[n-1:2*n-1] - s2[n-1:2*n-1]) / n
    if not np.all(np.isfinite(grad)) or not np.isfinite(F):
        return None, None, c5
    return F, grad, c5
