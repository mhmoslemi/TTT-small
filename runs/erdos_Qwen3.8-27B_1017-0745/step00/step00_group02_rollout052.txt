        return None
    return h

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    n = None
    parent = None
    try:
        piv = globals().get('initial_h_values', None)
        if piv is not None:
            piv = np.asarray(piv, dtype=np.float64)
            if piv.ndim == 1 and 2 <= piv.shape[0] <= 999 and np.all(np.isfinite(piv)) and np.max(np.abs(piv)) < 1e6:
                n = piv.shape[0]
                parent = piv.copy()
    except Exception:
        pass
    if n is None:
        np_k = kwargs.get('n_points', 45)
        n = int(np_k) if isinstance(np_k, (int, np.integer)) and 2 <= int(np_k) <= 999 else 45
    dx = 2.0 / n
    S = n / 2.0
    fallback = project_onto_K(parent, n, S) if parent is not None else np.full(n, 0.5)
    if fallback is None:
        fallback = np.full(n, 0.5)
    best_h = fallback.copy()
    best_score = exact_score(best_h, dx)

    taus = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00001]
    if n > 100:
        taus = [0.02, 0.005, 0.001, 0.0001, 0.00001]

    local_results = []
    starts = make_starts(parent, n, S, rng)
    for s in starts:
        sc = exact_score(s, dx)
        if sc < best_score - 1e-12:
            best_h, best_score = s.copy(), sc
        rh = proj_grad(s, n, S, dx, taus)
        sc = exact_score(rh, dx)
        local_results.append((sc, rh.copy()))
        if sc < best_score - 1e-12:
            best_h, best_score = rh.copy(), sc

    pert_scales = [0.04, 0.12, 0.30, 0.50] if n <= 100 else [0.12]
    for sd in pert_scales:
        z = rng.standard_normal(n)
        s = project_onto_K(best_h + sd * z, n, S)
        if s is None:
            continue
        rh = proj_grad(s, n, S, dx, taus)
        sc = exact_score(rh, dx)
        local_results.append((sc, rh.copy()))
        if sc < best_score - 1e-12:
            best_h, best_score = rh.copy(), sc

    if n <= 100:
        local_results.sort(key=lambda x: x[0])
        picked = []
        for sc, rh in local_results:
            if len(picked) >= 3:
                break
            if all(np.max(np.abs(rh - pr)) > 1e-5 for _, pr in picked):
                picked.append((sc, rh))
        if not picked:
            picked.append((exact_score(fallback, dx), fallback.copy()))
        for _, ph in picked:
            sh = try_slsqp(ph, n, S, dx)
            if sh is not None:
                sh = project_onto_K(sh, n, S)
                if sh is not None:
                    sc = exact_score(sh, dx)
                    if sc < best_score - 1e-12:
                        best_h, best_score = sh.copy(), sc

    best_h = project_onto_K(best_h, n, S)
    if best_h is None:
        best_h = fallback.copy()
    final_score = exact_score(best_h, dx)
    fb_score = exact_score(fallback, dx)
    if final_score > fb_score:
        best_h, final_score = fallback.copy(), fb_score
    if not np.all(np.isfinite(best_h)) or best_h.min() < -1e-12 or best_h.max() > 1.0 + 1e-12:
        best_h = np.full(n, 0.5)
        final_score = exact_score(best_h, dx)
    best_h = np.clip(best_h, 0.0, 1.0)
    return (best_h.astype(np.float64), float(final_score), n)
