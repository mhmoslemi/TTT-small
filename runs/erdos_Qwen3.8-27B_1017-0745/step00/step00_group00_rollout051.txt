def project_box_sum(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(y)
        if s > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    # Correct residual
    resid = S - np.sum(y)
    if abs(resid) > 1e-15:
        if resid > 0:
            for i in range(n):
                add = min(resid, 1.0 - y[i])
                y[i] += add
                resid -= add
                if abs(resid) < 1e-15:
                    break
        else:
            for i in range(n):
                sub = min(-resid, y[i])
                y[i] -= sub
                resid += sub
                if abs(resid) < 1e-15:
                    break
    return y
