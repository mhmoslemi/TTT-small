    taus = [0.02, 0.005, 0.001, 0.0002, 0.00005, 0.00001]

    incumbent = None
    incumbent_score = np.inf

    if parent is not None and is_feasible(parent, n):
        incumbent = parent.copy()
        incumbent_score = exact_score(parent, dx)

    starts = []
    if parent is not None:
        starts.append(parent.copy())
        starts.append(parent[::-1].copy())
        for sig in [0.04, 0.10, 0.22, 0.40]:
            starts.append(parent + sig * rng.standard_normal(n))
    starts.append(rng.uniform(0.0, 1.0, size=n))
    starts.append(rng.beta(0.5, 0.5, size=n))
    starts.append(np.full(n, 0.5))

    if parent is None:
        for _ in range(4):
            starts.append(rng.uniform(0.0, 1.0, size=n))
        for _ in range(3):
            starts.append(rng.beta(0.5, 0.5, size=n))

    for s in starts:
        hres = optimize_from_start(s, n, dx, lags, target, taus)
        pr = capped_simplex_project(hres, target)
        if pr is None or not is_feasible(pr, n):
            continue
        sc = exact_score(pr, dx)
        if sc < incumbent_score - 1e-12:
            incumbent = pr.copy()
            incumbent_score = sc

    center = incumbent if incumbent is not None else np.full(n, 0.5)
    for sig in [0.02, 0.08, 0.20]:
        hres = optimize_from_start(center + sig * rng.standard_normal(n), n, dx, lags, target, taus)
        pr = capped_simplex_project(hres, target)
        if pr is not None and is_feasible(pr, n):
            sc = exact_score(pr, dx)
            if sc < incumbent_score - 1e-12:
                incumbent = pr.copy()
                incumbent_score = sc
        center = incumbent if incumbent is not None else np.full(n, 0.5)

    if incumbent is None:
        incumbent = np.full(n, 0.5)

    final = capped_simplex_project(incumbent, target)
    if final is None or not is_feasible(final, n):
        final = np.full(n, 0.5)

    c5 = exact_score(final, dx)
    if not np.isfinite(c5):
        final = np.full(n, 0.5)
        c5 = exact_score(final, dx)

    return (final, c5, n)
