def project_box_sum(z, N):
    z = np.asarray(z, dtype=np.float64)
    if z.shape != (N,) or not np.all(np.isfinite(z)) or np.any(np.abs(z) > 1e100):
        return np.full(N, 0.5)
    m = N / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        v = np.clip(z - mid, 0.0, 1.0)
        if np.sum(v) > m:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    h = np.clip(z - theta, 0.0, 1.0)
    e = m - np.sum(h)
    # bounded correction
    for _ in range(3):
        e = m - np.sum(h)
        if abs(e) < 1e-10 * max(1, N):
            break
        sign = 1.0 if e > 0 else -1.0
        for i in range(N):
            if e == 0:
                break
            delta = sign * min(abs(e), (1.0 - h[i]) if sign > 0 else h[i])
            h[i] += delta
            e -= delta

        return np.full(N, 0.5)
    h = np.clip(h, 0.0, 1.0)
    return h

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / N))

def lag_values_gradients(h, N):
    dx = 2.0 / N
    q = np.correlate(h, 1.0 - h, mode='full')
    G = np.zeros((2 * N - 1, N), dtype=np.float64)
    for k in range(-(N - 1), N):
        row = k + N - 1
        if k >= 0:
            L = N - k
            G[row, k:k + L] += (1.0 - h[:L])
            G[row, :L] -= h[k:k + L]
        else:
            s = -k
            L = N - s
            G[row, :L] += (1.0 - h[s:s + L])
            G[row, s:s + L] -= h[:L]
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None, None
    return q * dx, G * dx

def build_model(N):
    d = cp.Variable(N)
    u = cp.Variable()
    t = cp.Variable()
    x = cp.Parameter(N)
    r0 = cp.Parameter()
    J0 = cp.Parameter(N)
    rn = cp.Parameter(2 * N - 2)
    Jn = cp.Parameter((2 * N - 2, N))
    R = cp.Parameter()
    cons = [x + d >= 0, x + d <= 1, cp.sum(d) == 0,
            cp.norm(d, 2) <= R,
            u >= rn + Jn @ d,
            t >= u + (2.0 / N) * cp.sum_squares(d),
            t >= r0 + J0 @ d]
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, cons)
    return prob, d, u, t, x, r0, J0, rn, Jn, R

import numpy as np
import cvxpy as cp
import math


def project_box_sum(z, N):
    z = np.asarray(z, dtype=np.float64)
    if z.shape != (N,) or not np.all(np.isfinite(z)) or np.any(np.abs(z) > 1e100):
        return np.full(N, 0.5)
    m = N / 2.0
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        v = np.clip(z - mid, 0.0, 1.0)
        if np.sum(v) > m:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    h = np.clip(z - theta, 0.0, 1.0)
    for _ in range(3):
        e = m - np.sum(h)
        if abs(e) < 1e-10 * max(1, N):
            break
        sign = 1.0 if e > 0 else -1.0
        for i in range(N):
            if abs(e) < 1e-12:
                break
            cap = (1.0 - h[i]) if sign > 0 else h[i]
            delta = sign * min(abs(e), cap)
            h[i] += delta
            e -= delta
    if not np.all(np.isfinite(h)):
        return np.full(N, 0.5)
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return h


def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / N))


def lag_grads(h, N):
    dx = 2.0 / N
    raw = np.correlate(h, 1.0 - h, mode='full')
    G = np.zeros((2 * N - 1, N), dtype=np.float64)
    for k in range(-(N - 1), N):
        row = k + N - 1
        if k >= 0:
            L = N - k
            G[row, k:k + L] += (1.0 - h[:L])
            G[row, :L] -= h[k:k + L]
        else:
            s = -k
            L = N - s
            G[row, :L] += (1.0 - h[s:s + L])
            G[row, s:s + L] -= h[:L]
    if not np.all(np.isfinite(raw)) or not np.all(np.isfinite(G)):
        return None, None
    return raw * dx, G * dx


def build_model(N):
    d = cp.Variable(N)
    u = cp.Variable()
    t = cp.Variable()
    xP = cp.Parameter(N)
    r0P = cp.Parameter()
    J0P = cp.Parameter(N)
    nan_ = 2 * N - 2
    rnP = cp.Parameter(nan_)
    JnP = cp.Parameter((nan_, N))
    RP = cp.Parameter()
    cons = [
        xP + d >= 0,
        xP + d <= 1,
        cp.sum(d) == 0,
        cp.norm(d, 2) <= RP,
        u * cp.ones(nan_) >= rnP + JnP @ d,
        t >= u + (2.0 / N) * cp.sum_squares(d),
        t >= r0P + J0P @ d
    ]
    prob = cp.Problem(cp.Minimize(t), cons)
    return prob, d, u, t, xP, r0P, J0P, rnP, JnP, RP


def solve_local(start, N, max_iter):
    dx = 2.0 / N
    x = start.copy()
    f_cur = score_exact(x, N)
    if not np.isfinite(f_cur):
        return x, f_cur
    R = min(1.0, math.sqrt(float(N)))
    R_max = min(4.0, math.sqrt(float(N)))
    R_min = 1e-4
    fails = 0
    try:
        prob, d, u, t, xP, r0P, J0P, rnP, JnP, RP = build_model(N)
    except Exception:
        return x, f_cur
    for _ in range(max_iter):
        q, G = lag_grads(x, N)
        if q is None:
            break
        r0 = q[N - 1]
        J0 = G[N - 1]
        rn = np.delete(q, N - 1)
        Jn = np.delete(G, N - 1, axis=0)
        try:
            xP.value = x
            r0P.value = r0
            J0P.value = J0
            rnP.value = rn
            JnP.value = Jn
            RP.value = R
            d.value = np.zeros(N)
            u.value = float(np.max(rn))
            t.value = f_cur
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=200,
                       abstol=1e-8, reltol=1e-8, feastol=1e-8)
        except Exception:
            R = max(R * 0.5, R_min)
            fails += 1
            if fails >= 8:
                break
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            R = max(R * 0.5, R_min)
            fails += 1
            if fails >= 8:
                break
            continue
        dd = d.value
        if dd is None or dd.shape != (N,) or not np.all(np.isfinite(dd)):
            R = max(R * 0.5, R_min)
            fails += 1
            if fails >= 8:
                break
            continue
        trial = project_box_sum(x + dd, N)
        if np.any(trial < 0) or np.any(trial > 1) or not np.all(np.isfinite(trial)):
            R = max(R * 0.5, R_min)
            fails += 1
            if fails >= 8:
                break
            continue
        d_eff = trial - x
        if np.linalg.norm(d_eff) > R * (1.0 + 1e-6) + 1e-10:
            R = max(R * 0.5, R_min)
            fails += 1
            if fails >= 8:
                break
            continue
        f_new = score_exact(trial, N)
        if f_new < f_cur - 1e-12 * max(1.0, abs(f_cur)):
            U = r0 + float(J0 @ d_eff)
            d2 = float(np.dot(d_eff, d_eff))
            for i in range(len(rn)):
                val = rn[i] + float(Jn[i] @ d_eff) + dx * d2
                if val > U:
                    U = val
            actual = f_cur - f_new
            predicted = f_cur - U
            if predicted > 1e-12:
                ratio = actual / predicted
            elif actual > 0:
                ratio = 10.0
            else:
                ratio = 0.0
            x = trial
            f_cur = f_new
            fails = 0
            n_eff = np.linalg.norm(d_eff)
            if ratio >= 0.75 and n_eff >= 0.8 * R:
                R = min(1.5 * R, R_max)
            elif ratio < 0.1:
                R = max(R * 0.5, R_min)
            if n_eff < 1e-7:
                break
        else:
            R = max(R * 0.5, R_min)
            fails += 1
        if R <= R_min:
            break
    return x, f_cur

    for s in starts:

        for us in unique:

        c5 = score_exact(best, N)
    return best.astype(np.float64), float(c5), int(N)
