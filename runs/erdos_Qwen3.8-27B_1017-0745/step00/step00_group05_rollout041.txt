        return (fin, float(fs), n)

    B = max(200, min(12000, int(200000000 / (n * n))))
    max_att = 4 * B
    max_swp = 4 * B // R + 1
    restarts = 0
    stagnant = 0
    stag_lim = max(50, 2 * n)
    att = 0
    swp = 0
    stop = False

    while not stop and att < max_att and swp < max_swp and eval_count < B:
        imp = False
        for r in range(R):
            if stop or att >= max_att or eval_count >= B:
                break
            att += 1
            cand = propose_move(rstates[r], rhos[r], rng)
            if cand is None:
                continue
            cs = exact_score(cand, dx)
            eval_count += 1
            if cs is None:
                continue
            if cs < best_score:
                best_score = cs
                best_h = cand.copy()
                imp = True
                if best_score <= 0.3808:
                    stop = True
                    break
            delta = cs - rscores[r]
            if delta <= 0:
                rstates[r] = cand; rscores[r] = cs
            else:
                u = float(rng.random())
                if u == 0.0 or math.log(u) < -delta / temps[r]:
                    rstates[r] = cand; rscores[r] = cs
        swp += 1
        if swp % 5 == 0:
            par = (swp // 5) % 2
            pairs = [(2*i, 2*i+1) for i in range(4)] if par == 0 else [(2*i+1, 2*i+2) for i in range(3)]
            for ra, rb in pairs:
                if rb >= R:
                    continue
                L = (rscores[ra] - rscores[rb]) * (1.0/temps[ra] - 1.0/temps[rb])
                do_swap = L >= 0
                if not do_swap:
                    u = float(rng.random())
                    do_swap = (u == 0.0 or math.log(u) < L)
                if do_swap:
                    rstates[ra], rstates[rb] = rstates[rb], rstates[ra]
                    rscores[ra], rscores[rb] = rscores[rb], rscores[ra]
        stagnant = 0 if imp else stagnant + 1
        if stagnant >= stag_lim and restarts < 2 and eval_count + R < B:
            restarts += 1
            stagnant = 0
            rprof = [best_h.copy()]
            for sd in [0.02, 0.04, 0.08, 0.16, 0.28]:
                rprof.append(best_h + rng.standard_normal(n) * sd)
            rprof.append(rng.beta(0.5, 0.5, n))
            rprof.append(rng.random(n) if restarts == 1 else make_block(n, rng))
            ns = []; nc = []
            for pr in rprof:
                px = capped_simplex_project(pr, target)
                if px is None:
                    px = best_h.copy()
                sc = exact_score(px, dx)
                eval_count += 1
                if sc is None:
                    sc = best_score
                    px = best_h.copy()
                ns.append(px); nc.append(sc)
                if sc < best_score:
                    best_score = sc
                    best_h = px.copy()
            rstates = ns
            rscores = nc

    fin = capped_simplex_project(best_h, target)
    if fin is None:
        fin = best_h
    fs = exact_score(fin, dx)
    if fs is None or not math.isfinite(fs):
        fin2 = capped_simplex_project(uniform, target)
        if fin2 is None:
            fin2 = uniform
        fs = exact_score(fin2, dx)
        if fs is None:
            fs = 0.5
        fin = fin2
    if not chk_feasible(fin, target):
        fin2 = capped_simplex_project(uniform, target)
        if fin2 is not None:
            fin = fin2
            fs = exact_score(fin, dx)
        if fs is None:
            fs = 0.5
    return (fin, float(fs), n)
