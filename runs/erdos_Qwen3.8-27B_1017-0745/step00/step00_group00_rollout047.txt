def project_box_sum(v, S):
    n = len(v)
    if not np.all(np.isfinite(v)) or not math.isfinite(S):
        return None
    if S < -1.0 or S > float(n) + 1.0 or np.max(np.abs(v)) > 1e12:
        return None
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v))
    for _ in range(80):
        m = 0.5 * (lo + hi)
        if float(np.sum(np.clip(v - m, 0.0, 1.0))) > S:
            lo = m
        else:
            hi = m
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0).copy()
    res = S - float(np.sum(y))
    for _ in range(2):
        for i in range(n):
            if abs(res) < 1e-12:
                break
            a = min(res, 1.0 - y[i]) if res > 0 else -min(-res, y[i])
            y[i] += a
            res -= a
    if not np.all(np.isfinite(y)) or float(np.min(y)) < -1e-12 or float(np.max(y)) > 1.0 + 1e-12:
        return None
    if abs(float(np.sum(y)) - S) > 1e-10 * max(1, n):

    return y
