    return t_lo, t_hi

def exact_line_min(t_lo, t_hi, q, b, c):
    n_l = len(q)
    cands = [t_lo, t_hi, 0.0]
    for r in range(n_l):
        for s in range(r + 1, n_l):
            A, B, C = c[r] - c[s], b[r] - b[s], q[r] - q[s]
            sc = max(abs(A), abs(B), abs(C), 1e-300)
            if abs(A) < 1e-14 * sc:
                if abs(B) > 1e-300:
                    t = -C / B
                    if np.isfinite(t) and t_lo - 1e-12 <= t <= t_hi + 1e-12:
                        cands.append(t)
            else:
                disc = B * B - 4.0 * A * C
                tol = 1e-14 * (B * B + abs(4.0 * A * C) + 1e-300)
                if disc < -tol:
                    continue
                disc = max(disc, 0.0)
                sd = math.sqrt(disc)
                if B >= 0:
                    q1 = -0.5 * (B + sd)
                else:
                    q1 = -0.5 * (B - sd)
                roots = []
                if abs(q1) > 1e-300:
                    roots = [q1 / A, C / q1]
                else:
                    roots = [-B / (2.0 * A)] * 2
                for tv in roots:
                    if np.isfinite(tv) and t_lo - 1e-12 <= tv <= t_hi + 1e-12:
                        cands.append(tv)
        if abs(c[r]) > 1e-300:
            ts = -b[r] / (2.0 * c[r])
            if np.isfinite(ts) and t_lo - 1e-12 <= ts <= t_hi + 1e-12:
                cands.append(ts)
    cands = [max(t_lo, min(t_hi, t)) for t in cands]
    cands.sort()
    deduped = [cands[0]]
    for t in cands[1:]:
        if abs(t - deduped[-1]) > 1e-12 * max(1.0, abs(t)):
            deduped.append(t)
    best_t, best_val = 0.0, float('inf')
    for i in range(0, len(deduped), 256):
        ch = deduped[i:i + 256]
        for t in ch:
            v = float(np.max(q + t * b + t * t * c))
            if v < best_val:
                best_val, best_t = v, t
    return best_t, best_val

def grid_line_min(t_lo, t_hi, q, b, c):
    pts = np.unique(np.concatenate([np.linspace(t_lo, t_hi, 17), [0.0]]))
    best_t, best_val = 0.0, float('inf')
    scores = np.array([float(np.max(q + t * b + t * t * c)) for t in pts])
    idx = int(np.argmin(scores))
    best_val, best_t = scores[idx], pts[idx]
    for _ in range(2):
        if idx == 0:
            a_br, b_br = pts[0], pts[1]
        elif idx == len(pts) - 1:
            a_br, b_br = pts[-2], pts[-1]
        else:
            a_br, b_br = pts[idx - 1], pts[idx + 1]
        new_pts = np.unique(np.concatenate([pts, np.linspace(a_br, b_br, 10)]))
        pts = new_pts
        scores = np.array([float(np.max(q + t * b + t * t * c)) for t in pts])
        idx = int(np.argmin(scores))
        best_val, best_t = scores[idx], pts[idx]
    return best_t, best_val

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    n, fallback_h = None, None
    try:
        arr = np.array(initial_h_values, dtype=np.float64).flatten()
        if 2 <= len(arr) <= 999:
            n = len(arr)
            if np.all(np.isfinite(arr)) and np.max(np.abs(arr)) <= 1e6:
                if np.all(arr >= -1e-10) and np.all(arr <= 1.0 + 1e-10) and abs(arr.sum() - n / 2.0) <= 1e-10:
                    fallback_h = np.clip(arr, 0.0, 1.0).copy()
                else:
                    fallback_h = project_feasible(arr, n)
                    if fallback_h is None:
                        fallback_h = np.full(n, 0.5)
            else:
                fallback_h = np.full(n, 0.5)
    except Exception:
        pass
    if n is None:
        n = 45
    if fallback_h is None:
        fallback_h = np.full(n, 0.5)
    dx = 2.0 / n
    fb_score = exact_score(fallback_h, n)
    best_h, best_score = fallback_h.copy(), fb_score
    tree = build_tree(n)
    if len(tree) > 64:
        sel = sorted(set(int(round(k * (len(tree) - 1) / 63.0)) for k in range(64)))
        dirs = [tree[i] for i in sel]
    else:
        dirs = tree
    max_starts = 4 if n <= 128 else (3 if n <= 256 else 2)
    starts = [fallback_h.copy()]
    comp = 1.0 - fallback_h.copy()
    if is_feasible(comp, n):
        starts.append(comp)
    pn = project_feasible(fallback_h + 0.12 * rng.standard_normal(n), n)
    if pn is not None:
        starts.append(pn)
    pb = project_feasible(rng.beta(0.5, 0.5, size=n), n)
    if pb is not None:
        starts.append(pb)
    ded = [starts[0]]
    for s in starts[1:]:
        if all(np.max(np.abs(s - d2)) > 1e-8 for d2 in ded):
            ded.append(s)
    starts = ded[:max_starts]
    max_sweeps = 3 if n <= 128 else (2 if n <= 256 else 1)
    use_exact = n <= 48
    for start in starts:
        cur, cur_score = start.copy(), exact_score(start, n)
        if not np.isfinite(cur_score):
            continue
        improved = True
        for sw in range(max_sweeps):
            if not improved:
                break
            improved = False
            for rec in dirs:
                _, a, m, b_ = rec
                d = make_direction(a, m, b_, n)
                t_lo, t_hi = feasible_interval(cur, d)
                if t_lo is None or t_hi - t_lo < 1e-12:
                    continue
                omh = 1.0 - cur
                q = np.correlate(cur, omh, mode='full') * dx
                bv = dx * (np.correlate(d, omh, mode='full') - np.correlate(cur, d, mode='full'))
                cv = -dx * np.correlate(d, d, mode='full')
                if not (np.all(np.isfinite(q)) and np.all(np.isfinite(bv)) and np.all(np.isfinite(cv))):
                    continue
                if use_exact:
                    t_s, pv = exact_line_min(t_lo, t_hi, q, bv, cv)
                else:
                    t_s, pv = grid_line_min(t_lo, t_hi, q, bv, cv)
                if not np.isfinite(t_s) or pv >= cur_score - 1e-12:
                    continue
                ht = cur + t_s * d
                rep = small_repair(ht, n)
                if rep is None:
                    rep = project_feasible(ht, n)
                if rep is None:
                    continue
                ns = exact_score(rep, n)
                if not np.isfinite(ns) or ns >= cur_score - 1e-12:
                    continue
                cur, cur_score = rep, ns
                improved = True
                if cur_score < best_score - 1e-12:
                    best_h, best_score = cur.copy(), cur_score
    fh = best_h.copy()
    if not is_feasible(fh, n):
        p = project_feasible(fh, n)
        fh = p if p is not None else np.full(n, 0.5)
    fs = exact_score(fh, n)
    fbs = exact_score(fallback_h, n)
    if not np.isfinite(fs) or (np.isfinite(fbs) and fs >= fbs - 1e-12):
        fh = fallback_h.copy() if is_feasible(fallback_h, n) else np.full(n, 0.5)
        fs = exact_score(fh, n)
    if not is_feasible(fh, n):
        fh = np.full(n, 0.5)
        fs = exact_score(fh, n)
    return (fh.copy(), float(fs), int(n))
