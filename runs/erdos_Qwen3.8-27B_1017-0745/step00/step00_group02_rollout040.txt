        best_trial_score = cur
        for k in range(max_bt):
            alpha = 2.0 ** (-k)
            trial = h + alpha * d_feas
            tp = project_k(trial, n, S)
            if tp is None:
                continue
            ts = exact_score(tp, n)
            if ts < 0:
                continue
            if ts < best_trial_score - 1e-12:
                best_trial_score = ts
                best_h_trial = tp
        if best_h_trial is None:
            break
        improvement = cur - best_trial_score
        h = best_h_trial.copy()
        cur = best_trial_score
        if improvement < 1e-10:
            consec += 1
            if consec >= 3:
                break
        else:
            consec = 0
    return h

def compute_q_internal(h, n, dx):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if not np.all(np.isfinite(c)):
        return None
    return c

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)

    n = None
    h_parent = None
    try:
        raw = globals().get('initial_h_values', None)
        if raw is not None:
            arr = np.asarray(raw, dtype=np.float64).ravel()
            if 2 <= len(arr) <= 999:
                n = len(arr)
                if np.all(np.isfinite(arr)) and np.all(np.abs(arr) <= 1e6):
                    s_tmp = n / 2.0
                    if np.all(arr >= 0) and np.all(arr <= 1) and abs(float(np.sum(arr)) - s_tmp) <= 1e-12:
                        h_parent = arr.copy()
                    else:
                        h_parent = project_k(arr, n, s_tmp)
    except Exception:
        n = None
        h_parent = None

    if n is None:
        try:
            nv = kwargs.get('n_points', None)
            if nv is not None and not isinstance(nv, bool) and isinstance(nv, (int, np.integer)) and 2 <= int(nv) <= 999:
                n = int(nv)
        except Exception:
            n = 45
        if n is None:
            n = 45

    S = n / 2.0
    dx = 2.0 / n

    if h_parent is None or h_parent.shape != (n,) or not np.all(np.isfinite(h_parent)):
        h_parent = project_k(np.full(n, 0.5), n, S)
    if h_parent is None:
        h_parent = np.full(n, 0.5)

    fallback_h = h_parent.copy()
    fallback_score = exact_score(fallback_h, n)
    if fallback_score < 0:
        fallback_score = 1.0

    best_h = fallback_h.copy()
    best_score = fallback_score

    starts = make_starts(h_parent, n, S, rng)
    if not starts:
        starts = [fallback_h.copy()]

    max_iter = 40 if n <= 64 else (20 if n <= 256 else 8)

    for s in starts:
        ss = exact_score(s, n)
        if ss >= 0 and ss < best_score - 1e-12:
            best_h = s.copy()
            best_score = ss
        try:
            res = optimize_start(s, n, S, dx, max_iter)
            if res is not None:
                rs = exact_score(res, n)
                if rs >= 0 and rs < best_score - 1e-12:
                    best_h = res.copy()
                    best_score = rs
        except Exception:
            pass

    final_h = best_h.copy()
    final_score = exact_score(final_h, n)
    if final_score < 0:
        fp = project_k(final_h, n, S)
        if fp is not None:
            final_h = fp
            final_score = exact_score(final_h, n)
    if final_score < 0:
        final_h = fallback_h.copy()
        final_score = exact_score(final_h, n)
    if final_score < 0:
        final_h = np.full(n, 0.5)
        final_score = exact_score(final_h, n)
    if final_score < 0:
        final_score = float('inf')

    return (np.array(final_h, dtype=np.float64).copy(), float(final_score), int(n))
