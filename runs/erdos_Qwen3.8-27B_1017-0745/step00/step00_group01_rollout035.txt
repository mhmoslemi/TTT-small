                    inc_score = pop_sc[ri]
                    inc_vec = pop[ri].copy()
            no_imp = 0
            n_res += 1
        elif no_imp >= patience and n_res >= 2:
            break
    return inc_vec, inc_score

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0 and np.any(np.isfinite(p)):
            n = p.size
            parent = p.copy()
    except Exception:
        pass
    if n is None:
        try:
            nv = kwargs.get('n_points', 47)
            if isinstance(nv, bool):
                nv = 47
            nv = int(nv)
            n = nv if 1 <= nv <= 999 else 47
        except Exception:
            n = 47
    if n is None or n <= 0:
        n = 47
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)

    if parent is not None and parent.size == n:
        if is_feasible(parent, n):
            fallback = parent.copy()
        else:
            pr = project_capped(parent, n)
            fallback = pr if pr is not None else uniform_vec(n)
    else:
        fallback = uniform_vec(n)
    fb_score = exact_score(fallback, n)
    incumbent = fallback.copy()
    inc_score = fb_score

    if n == 1 or n > 999:
        return (incumbent, inc_score, n)

    if n <= 64:
        P, G, pat = 32, 160, 24
    elif n <= 128:
        P, G, pat = 20, 80, 16
    elif n <= 256:
        P, G, pat = 12, 32, 8
    else:
        P, G, pat = 8, 12, 4

    base = fallback.copy()
    props = make_proposals(base, n, rng, P)
    pop, pop_sc = build_population(props, n, P, inc_score)
    pop, pop_sc = fill_population(pop, pop_sc, base, n, P, rng)
    for i in range(len(pop)):
        if pop_sc[i] < inc_score - 1e-12:
            inc_score = pop_sc[i]
            incumbent = pop[i].copy()

    if P >= 4:
        iv, isc = de_search(pop, pop_sc, n, rng, G, pat)
        if isc < inc_score - 1e-12:
            inc_score = isc
            incumbent = iv

    final = project_capped(incumbent, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if fs >= float('inf') or fs > fb_score + 1e-12:
        final = fallback.copy()
        fs = fb_score
    return (final, fs, n)
