def project_onto_K(y, n, S):
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return None
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        x = np.clip(y - mid, 0.0, 1.0)
        s = float(np.sum(x))
        if abs(s - S) < 1e-13:
            break
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - float(np.sum(x))
    if abs(r) > 1e-14:
        if r > 0:
            slack = 1.0 - x
            slack[slack < 1e-15] = 0.0
            tot = float(np.sum(slack))
            if tot > 1e-15:
                x += r * slack / tot
        else:
            pos = x.copy()
            pos[pos < 1e-15] = 0.0
            tot = float(np.sum(pos))
            if tot > 1e-15:
                x += r * pos / tot
    x = np.clip(x, 0.0, 1.0)
    if abs(float(np.sum(x)) - S) > 1e-10:
        return None
    if np.any(x < -1e-12) or np.any(x > 1.0 + 1e-12):
        return None
    return x

def is_feasible(h, n, S):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(float(np.sum(h)) - S) > 1e-10:
        return False
    return True

def exact_score(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if not np.all(np.isfinite(c)):
        return None
    return float(np.max(c))

def build_gradient_matrix(h, n, dx):
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n), dtype=np.float64)
    for idx in range(n_lags):
        ell = idx - (n - 1)
        for j in range(n):
            g = 0.0
            jm = j - ell
            if 0 <= jm < n:
                g += (1.0 - h[jm])
            jp = j + ell
            if 0 <= jp < n:
                g -= h[jp]
            G[idx, j] = dx * g
    return G

def solve_qp(h, n, dx, S, q, G, C):
    try:
        import cvxpy as cp
    except Exception:
        return None
    d = cp.Variable(n, value=np.zeros(n))
    t = cp.Variable(value=C)
    cons = [q + G @ d <= t, h + d >= 0, h + d <= 1, cp.sum(d) == S - float(np.sum(h))]
    obj = cp.Minimize(t + dx * cp.sum_squares(d))
    prob = cp.Problem(obj, cons)
    for solver_name, kwargs in [('CVXOPT', {'max_iters': 150, 'tol_gap_abs': 1e-8, 'tol_gap_rel': 1e-8, 'tol_feas': 1e-8}),
                                 ('ECOS', {'max_iters': 200, 'abstol': 1e-9, 'reltol': 1e-9, 'feastol': 1e-9})]:
        try:
            prob.solve(solver=solver_name, **kwargs)
            if d.value is None or t.value is None:
                continue
            d_val = np.array(d.value, dtype=np.float64).flatten()
            t_val = float(t.value)
            if d_val.shape != (n,) or not np.all(np.isfinite(d_val)) or not np.isfinite(t_val):
                continue
            if np.max(np.abs(d_val)) > 1.0 + 1e-5:
                continue
            ld = q + G @ d_val
            if not np.all(np.isfinite(ld)):
                continue
            if np.max(ld) > t_val + 1e-5:
                continue
            if np.max(ld) + dx * float(np.sum(d_val**2)) > C + 1e-5:
                continue
            h_new = h + d_val
            if np.min(h_new) < -1e-5 or np.max(h_new) > 1.0 + 1e-5:
                continue
            if abs(float(np.sum(h_new)) - S) > 1e-5:
                continue
            return d_val
        except Exception:
            continue
    return None
