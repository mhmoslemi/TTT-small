    z = np.clip(z, -1e12, 1e12)
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h_tmp = np.clip(z - mid, 0.0, 1.0)
        if np.sum(h_tmp) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(n + 2):
        if abs(r) < 1e-12 * max(1, n):
            break
        if r > 0:
            slack = 1.0 - h
            idx = int(np.argmax(slack))
            amt = min(r, slack[idx])
            if amt <= 1e-15:
                break
            h[idx] += amt
            r -= amt
        else:
            amt = min(-r, h[int(np.argmin(h))])
            if amt <= 1e-15:
                break
            idx = int(np.argmin(h))
            h[idx] -= amt
            r += amt
    h = np.clip(h, 0.0, 1.0)
    if not check_feasibility(h, n):
        return np.full(n, 0.5, dtype=np.float64)
    return h

def exact_score(h):
    n = len(h)
    if not check_feasibility(h, n):
        return np.inf
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(c))

def smooth_value_weights(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = dx * np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(c))
    a = np.exp((c - c_max) / tau)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 0.0:
        return np.inf, None
    F = c_max + tau * np.log(s)
    P = a / s
    return F, P

def smooth_gradient(h, tau, P):
    n = len(h)
    dx = 2.0 / n
    w = 1.0 - h
    P_rev = P[::-1]
    U = np.convolve(P, w, mode='full')[n - 1:2 * n - 1]
    V = np.convolve(P_rev, h, mode='full')[n - 1:2 * n - 1]
    return dx * (U - V)

def build_starts(n, base, rng):
    starts = []
    props = [base, base[::-1].copy(), 1.0 - base[::-1].copy(), np.full(n, 0.5)]
    blk = np.zeros(n)
    half = n // 2
    blk[:half] = 1.0
    if n % 2 == 1:
        blk[half] = 0.5
    props.append(blk)
    alt = np.zeros(n)
    alt[0::2] = 1.0
    props.append(alt)
    for sd in [0.05, 0.15, 0.35]:
        props.append(base + sd * rng.standard_normal(n))
    props.append(rng.random(n))
    for p in props:
        p = np.asarray(p, dtype=np.float64)
        if p.shape != (n,) or not np.all(np.isfinite(p)):
            p = np.full(n, 0.5)
        pj = project_capped_simplex(p, n)
        dup = False
        for ex in starts:
            if np.max(np.abs(pj - ex)) < 1e-10:
                dup = True
                break
        if not dup:
            starts.append(pj)
    return starts

def pgd_from_start(x, taus, iters_per_tau):
    n = len(x)
    best_x = x.copy()
    best_score = exact_score(x)
    for tau in taus:
        alpha = 1.0
        stall = 0
        for it in range(iters_per_tau):
            F_val, P = smooth_value_weights(x, tau)
            if not np.isfinite(F_val) or P is None:
                break
            grad = smooth_gradient(x, tau, P)
            if not np.all(np.isfinite(grad)):
                break
            accepted = False
            y = x.copy()
            for bt in range(24):
                y = project_capped_simplex(x - alpha * grad, n)
                D = float(np.sum((y - x) ** 2))
                if np.max(np.abs(y - x)) <= 1e-9:
                    break
                F_y, _ = smooth_value_weights(y, tau)
                thresh = F_val - 1e-4 * D / alpha + 1e-14 * max(1.0, abs(F_val))
                if np.isfinite(F_y) and F_y <= thresh:
                    accepted = True
                    break
                alpha *= 0.5
            if not accepted:
                break
            F_y, _ = smooth_value_weights(y, tau)
            if not np.isfinite(F_y):
                break
            imp = abs(F_val - F_y)
            if imp <= 1e-12 * max(1.0, abs(F_val)):
                stall += 1
            else:
                stall = 0
            x = y
            alpha = min(2.0, 1.25 * alpha)
            if stall >= 5:
                break
            sc = exact_score(x)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_x = x.copy()
    return best_x, best_score

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(int(seed))
    except (TypeError, ValueError, OverflowError):
        rng = np.random.default_rng(42)

    try:
        base = np.array(initial_h_values, dtype=np.float64).ravel()
        if len(base) == 0:
            raise ValueError
    except (NameError, TypeError, ValueError):
        base = None

    if base is not None:
        n = len(base)
        if not np.all(np.isfinite(base)):
            base = np.full(n, 0.5)
    else:
        try:
            n = int(kwargs.get('n_points', 47))
        except (TypeError, ValueError):
            n = 47
        n = max(1, min(n, 999))
        base = np.full(n, 0.5)

    if n <= 64:
        max_starts, max_iters = 10, 100
    elif n <= 256:
        max_starts, max_iters = 5, 60
    else:
        max_starts, max_iters = 2, 25

    try:
        restarts = int(kwargs.get('restarts', max_starts))
    except (TypeError, ValueError):
        restarts = max_starts
    restarts = max(1, min(restarts, max_starts))

    try:
        iters_per_tau = int(kwargs.get('iters_per_tau', max_iters))
    except (TypeError, ValueError):
        iters_per_tau = max_iters
    iters_per_tau = max(1, min(iters_per_tau, max_iters))

    taus = [0.02, 0.005, 0.001, 0.0002, 0.00005, 0.00001]

    base_proj = project_capped_simplex(base, n)
    fallback_score = exact_score(base_proj)
    if not np.isfinite(fallback_score):
        base_proj = np.full(n, 0.5)
        fallback_score = exact_score(base_proj)

    incumbent = base_proj.copy()
    incumbent_score = fallback_score

    if n <= 999:
        start_pool = build_starts(n, base_proj, rng)
        start_pool = start_pool[:restarts]
        for s in start_pool:
            try:
                bx, bs = pgd_from_start(s, taus, iters_per_tau)
                if np.isfinite(bs) and bs < incumbent_score - 1e-12:
                    incumbent = bx
                    incumbent_score = bs
            except Exception:
                continue

    incumbent = project_capped_simplex(incumbent, n)
    final_score = exact_score(incumbent)

    fb2 = project_capped_simplex(base, n)
    fb2_score = exact_score(fb2)
    if np.isfinite(fb2_score) and fb2_score < final_score:
        incumbent = fb2
        final_score = fb2_score

    if not np.isfinite(final_score):
        incumbent = np.full(n, 0.5, dtype=np.float64)
        final_score = exact_score(incumbent)

    return incumbent, float(final_score), int(n)
