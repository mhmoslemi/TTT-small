import numpy as np

def capped_simplex_project(z, n):
    target = n / 2.0
    z = np.asarray(z, dtype=np.float64)
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return np.full(n, 0.5)
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(z - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) * 0.5, 0, 1)
    for _ in range(n + 2):
        res = target - h.sum()
        if abs(res) < 1e-15 * max(1, n):
            break
        if res > 0:
            idx = int(np.argmin(1.0 - h))
            h[idx] = min(1.0, h[idx] + res)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + res)
    h = np.clip(h, 0, 1)
    if not np.all(np.isfinite(h)) or abs(h.sum() - target) > 1e-9 * max(1, n):
        return np.full(n, 0.5)
    return h

def exact_score(h):
    h = np.asarray(h, dtype=np.float64)
    n = h.size
    if n == 0 or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def _corr_jac(h):
    n = h.size
    q = np.correlate(h, 1.0 - h, mode='full')
    J = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        i0 = max(d, 0)
        j0 = max(-d, 0)
        m = n - abs(d)
        row = d + n - 1
        if d == 0:
            J[row, i0:i0 + m] = 1.0 - 2.0 * h[i0:i0 + m]
        else:
            J[row, i0:i0 + m] += 1.0 - h[j0:j0 + m]
            J[row, j0:j0 + m] -= h[i0:i0 + m]
    return q, J

def _epi_obj(z, n):
    return z[-1]

def _epi_obj_jac(z, n):
    g = np.zeros(z.size)
    g[-1] = 1.0
    return g

def _epi_ineq(z, n):
    return z[-1] - (2.0 / n) * np.correlate(z[:-1], 1.0 - z[:-1], mode='full')

def _epi_ineq_jac(z, n):

    dx = 2.0 / n
    out = np.zeros((2 * n - 1, n + 1))
    out[:, :n] = -dx * J
    out[:, n] = 1.0
    return out

def _epi_eq_jac(z, n):

    return g

    steps = (0.04, 0.015, 0.005, 0.0015, 0.0005, 0.0001)
