def project_box_sum(v, S):
    # Project v onto {0<=x<=1, sum(x)=S}
    n = len(v)
    low = v.min() - 1.0
    high = v.max()
    for _ in range(80):
        mid = (low + high) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > S:
            low = mid
        else:
            high = mid
    y = np.clip(v - (low + high) / 2.0, 0.0, 1.0)
    # Correct residual
    resid = S - y.sum()
    if abs(resid) > 1e-14:
        if resid > 0:
            for i in range(n):
                cap = 1.0 - y[i]
                add = min(cap, resid)
                y[i] += add
                resid -= add
                if abs(resid) < 1e-14:
                    break

                sub = min(y[i], -resid)
                y[i] -= sub
                resid += sub

                    break
