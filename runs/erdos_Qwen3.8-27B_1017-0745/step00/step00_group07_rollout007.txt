        pred = dx * (np.max(B) - Qm)
        score_y = exact_score(y)
        if score_y is None:
            Delta = 0.25 * Delta
            continue
        actual = score_x - score_y
        tol = 1e-12 * max(1.0, abs(score_x))
        if pred <= tol:
            if actual > tol:
                x, score_x = y, score_y
            break
        rho = actual / pred
        if actual > tol and rho >= 0.1:
            x, score_x = y, score_y
            smalls = smalls + 1 if actual < 1e-10 * max(1.0, abs(score_x)) else 0
            if smalls >= 8:
                break
            if rho >= 0.75 and step >= 0.8 * Delta:
                Delta = min(2.0 * Delta, Dmax)
            elif rho < 0.25:
                Delta = max(0.25 * Delta, Dmin)
        else:
            Delta = 0.25 * Delta
    return x, score_x

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = None
    try:
        pv = np.array(globals().get('initial_h_values'), dtype=np.float64)
        if pv.ndim == 1 and pv.size >= 2 and np.all(np.isfinite(pv)):
            parent = pv.copy()
    except Exception:
        parent = None
    if parent is not None:
        n = len(parent)
    else:
        nkw = kwargs.get('n_points', None)
        n = int(nkw) if isinstance(nkw, (int, np.integer)) and 8 <= nkw <= 512 else 80
    S = n / 2.0
    lv, uv = np.zeros(n), np.ones(n)
    if parent is not None:
        pr = project_box_sum(parent, S, lv, uv)
        parent = pr if (pr is not None and check_feasible(pr, S)) else None
    fallback = np.full(n, 0.5)
    fs = exact_score(fallback)
    best_h, best_score = fallback.copy(), fs if fs is not None else 0.5
    if parent is not None:
        ps = exact_score(parent)
        if ps is not None and ps < best_score:
            best_score, best_h = ps, parent.copy()
    if n <= 512:
        mi = 36 if n <= 128 else (24 if n <= 256 else 12)
        for s in generate_starts(n, rng, parent):
            xn, sn = slp_search(s.copy(), S, n, mi)
            if sn is not None and sn < best_score - 1e-14 * max(1.0, abs(best_score)):
                best_score, best_h = sn, xn.copy()
    fin = project_box_sum(best_h, S, lv, uv)
    if fin is None or not check_feasible(fin, S):
        fin = fallback
    sc = exact_score(fin)
    if sc is None:
        fin, sc = fallback, exact_score(fallback)
        if sc is None:
            sc = 0.5
    return (fin, float(sc), int(n))
