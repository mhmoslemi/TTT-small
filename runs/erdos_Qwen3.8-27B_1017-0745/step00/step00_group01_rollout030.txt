    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):

    return abs(float(np.sum(h)) - n / 2.0) <= 1e-10 * max(1, n)

def project_capped(z, n):
    s = n / 2.0
    z = np.clip(np.asarray(z, dtype=np.float64), -1e12, 1e12)
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    v = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(v))
    for _ in range(n + 2):
        if abs(r) <= 1e-13:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - v))
            step = min(r, 1.0 - v[idx])
            if step <= 1e-15:
                break
            v[idx] += step
            r -= step
        else:
            idx = int(np.argmax(v))
            step = min(-r, v[idx])
            if step <= 1e-15:
                break
            v[idx] -= step
            r += step
    if not np.all(np.isfinite(v)) or np.any(v < -1e-12) or np.any(v > 1.0 + 1e-12):
        return None
    if abs(float(np.sum(v)) - s) > 1e-10 * max(1, n):
        return None
    return v

def exact_score(h, n):
    if h is None or len(h) != n or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_model(x, n):
    dx = 2.0 / n
    c = dx * np.correlate(x, 1.0 - x, mode='full')
    L = 2 * n - 1
    J = np.zeros((L, n), dtype=np.float64)
    for d in range(-(n - 1), n):
        row = d + n - 1
        i0, j0, m = max(d, 0), max(-d, 0), n - abs(d)
        J[row, i0:i0 + m] += 1.0 - x[j0:j0 + m]
        J[row, j0:j0 + m] -= x[i0:i0 + m]
    G = dx * J
    if not np.all(np.isfinite(c)) or not np.all(np.isfinite(G)):
        return None, None
    return c, G

def solve_lp(G, c, x, R, n):
    L = 2 * n - 1
    s = n / 2.0
    nrows = L + 2 * n + 1
    nvars = 2 * n + 1
    A_ub = np.zeros((nrows, nvars), dtype=np.float64)
    b_ub = np.zeros(nrows, dtype=np.float64)
    A_ub[:L, :n] = G
    A_ub[:L, 2 * n] = -1.0
    b_ub[:L] = G @ x - c
    A_ub[L:L + n, np.arange(n)] = 1.0
    A_ub[L:L + n, n + np.arange(n)] = -1.0
    b_ub[L:L + n] = x
    A_ub[L + n:L + 2 * n, np.arange(n)] = -1.0
    A_ub[L + n:L + 2 * n, n + np.arange(n)] = -1.0
    b_ub[L + n:L + 2 * n] = -x
    A_ub[L + 2 * n, n:2 * n] = 1.0
    b_ub[L + 2 * n] = R
    A_eq = np.zeros((1, nvars), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([s])
    c_obj = np.zeros(nvars)
    c_obj[2 * n] = 1.0
    bounds = [(0.0, 1.0)] * (2 * n) + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0 or not np.all(np.isfinite(res.x)):
            return None
        return res.x[:n].copy()
    except Exception:
        return None

def run(seed=42, **kwargs):
    n = 47
    base = None
    try:
        parent = initial_h_values
        if parent is not None:
            parent = np.asarray(parent, dtype=np.float64).ravel()
            if len(parent) > 0 and np.all(np.isfinite(parent)) and np.all(parent >= -1e-8) and np.all(parent <= 1.0 + 1e-8):
                n = len(parent)
                base = project_capped(parent, n)
    except Exception:
        pass
    if base is None:
        try:
            nop = int(kwargs.get('n_points', 47))
            if 1 <= nop <= 999:
                n = nop
        except Exception:
            pass
        base = np.full(n, 0.5, dtype=np.float64)
    fallback = base.copy()
    fb_score = exact_score(fallback, n)
    if n > 256:
        u = np.full(n, 0.5)
        if exact_score(u, n) < fb_score:
            return u, exact_score(u, n), int(n)
        return fallback, fb_score, int(n)
    if n <= 64:
        max_starts, max_iter = 9, 50
    elif n <= 128:
        max_starts, max_iter = 5, 30
    else:
        max_starts, max_iter = 2, 15
    R0 = max(0.1, min(1.5, 0.02 * n))
    Rmax = max(R0, min(8.0, 0.10 * n))
    try:
        ms = int(kwargs.get('max_starts', max_starts))
        max_starts = max(1, min(ms, [10, 6, 3][0 if n <= 64 else (1 if n <= 128 else 2)]))
    except Exception:
        pass
    try:
        mi = int(kwargs.get('max_iter', max_iter))
        max_iter = max(1, min(mi, [60, 40, 20][0 if n <= 64 else (1 if n <= 128 else 2)]))
    except Exception:
        pass
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    starts = []

    def _try_add(v):
        p = project_capped(v, n)
        if p is None:
            return
        for s in starts:
            if np.max(np.abs(p - s)) < 1e-10:
                return
        starts.append(p)
    _try_add(base)
    _try_add(base[::-1].copy())
    _try_add(1.0 - base[::-1].copy())
    _try_add(np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    _try_add(blk)
    alt = np.zeros(n)
    alt[0::2] = 1.0
    _try_add(alt)
    for std in [0.04, 0.12, 0.30]:
        _try_add(base + rng.standard_normal(n) * std)
    _try_add(rng.random(n))
    starts = starts[:max_starts]
    if not starts:
        starts = [base]
    best_h = fallback.copy()
    best_s = fb_score
    for start in starts:
        lx = start.copy()
        ls = exact_score(lx, n)
        if ls < best_s - 1e-12:
            best_s = ls
            best_h = lx.copy()
        if n <= 1:
            continue
        R = R0
        rej = 0
        for _ in range(max_iter):
            if R < 1e-6:
                break
            c, G = lag_model(lx, n)
            if c is None:
                break
            y = solve_lp(G, c, lx, R, n)
            if y is None:
                R *= 0.25
                rej += 1
                if rej >= 5:
                    break
                continue
            y = project_capped(y, n)
            if y is None:
                R *= 0.25
                rej += 1
                if rej >= 5:
                    break
                continue
            disp = float(np.sum(np.abs(y - lx)))
            if disp > R:
                fac = R / disp * (1.0 - 1e-12)
                y = lx + fac * (y - lx)
                y = project_capped(y, n)
                if y is None:
                    R *= 0.25
                    rej += 1
                    if rej >= 5:
                        break
                    continue
            if disp <= 1e-10:
                break
            model_val = float(np.max(c + G @ (y - lx)))
            pred = ls - model_val
            if pred <= 1e-12:
                break
            best_trial = None
            best_trial_s = ls
            used_alpha = 0.0
            for k in range(9):
                alpha = 0.5 ** k
                z = lx + alpha * (y - lx)
                z = project_capped(z, n)
                if z is None:
                    continue
                if float(np.sum(np.abs(z - lx))) > R * (1.0 + 1e-8):
                    sf = R / float(np.sum(np.abs(z - lx))) * (1.0 - 1e-12)
                    z = lx + sf * (z - lx)
                    z = project_capped(z, n)
                    if z is None:
                        continue
                zs = exact_score(z, n)
                if zs < best_trial_s - 1e-12:
                    best_trial_s = zs
                    best_trial = z
                    used_alpha = alpha
            if best_trial is None:
                R *= 0.25
                rej += 1
                if rej >= 5:
                    break
                continue
            actual = ls - best_trial_s
            rho = actual / pred if pred > 1e-12 else 0.0
            lx = best_trial.copy()
            ls = best_trial_s
            rej = 0
            if ls < best_s - 1e-12:
                best_s = ls
                best_h = lx.copy()
            if rho < 0.1:
                R *= 0.5
            elif rho >= 0.75 and used_alpha == 1.0 and disp >= 0.8 * R:
                R = min(2.0 * R, Rmax)
    final = project_capped(best_h, n)
    if final is None:
        final = fallback
    fs = exact_score(final, n)
    ffs = exact_score(fallback, n)
    if fs > ffs or not np.all(np.isfinite(final)):
        final = fallback
        fs = ffs
    return final, float(fs), int(n)
