        h_p = project_box_sum(best + sc * rng.standard_normal(N), m)
        s_p = score(h_p)
        if s_p < best_score:
            best = h_p.copy()
            best_score = s_p
        h_opt = solve_epigraph(h_p, N, m)
        s_opt = score(h_opt)
        if s_opt < best_score:
            best = h_opt.copy()
            best_score = s_opt
    best = project_box_sum(best, m)
    best = np.clip(best, 0.0, 1.0)
    final_score = score(best)
    if not np.isfinite(final_score):
        best = np.full(N, 0.5)
        final_score = score(best)
    return best, final_score, N
