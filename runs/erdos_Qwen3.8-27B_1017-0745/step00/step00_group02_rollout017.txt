    if h.shape != (n,) or not np.all(np.isfinite(h)):

    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(h.sum() - n / 2.0) <= tol

def _project_k(y, n, S):
    if not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0, 1).sum() > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0, 1)
    diff = S - x.sum()
    if abs(diff) < 1e-14:
        return x
    if diff > 0:
        mask = (1.0 - x) > 1e-8
    else:
        mask = x > 1e-8
    cnt = mask.sum()
    if cnt > 0:
        x[mask] += diff / cnt
        x = np.clip(x, 0, 1)
    return x

def _exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _lag_scores(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _sm_val(h, n, tau):
    Q = _lag_scores(h, n)
    m = Q.max()
    return m + tau * math.log(np.exp((Q - m) / tau).sum())

def _sm_grad(h, n, tau):
    Q = _lag_scores(h, n)
    m = Q.max()
    W = np.exp((Q - m) / tau)
    p = W / W.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        ell = k - (n - 1)
        if p[k] < 1e-15:
            continue
        a = max(0, -ell); b = min(n, n - ell)
        g[a:b] += p[k] * dx * (1.0 - h[a + ell:b + ell])
        c = max(0, ell); d2 = min(n, n + ell)
        g[c:d2] -= p[k] * dx * h[c - ell:d2 - ell]
    return g

def _proj_grad(h, n, S, tau_list):
    dx = 2.0 / n
    h = h.copy()
    bl, bs = h.copy(), _exact_score(h, n)
    for tau in tau_list:
        alpha = 1.0
        for _ in range(80):
            Q = _lag_scores(h, n)
            m = Q.max()
            W = np.exp((Q - m) / tau)
            Fv = m + tau * math.log(W.sum())
            p = W / W.sum()
            g = np.zeros(n)
            for k in range(2 * n - 1):
                ell = k - (n - 1)
                if p[k] < 1e-15:
                    continue
                a = max(0, -ell); b = min(n, n - ell)
                g[a:b] += p[k] * dx * (1.0 - h[a + ell:b + ell])
                c = max(0, ell); d2 = min(n, n + ell)
                g[c:d2] -= p[k] * dx * h[c - ell:d2 - ell]
            if not np.all(np.isfinite(g)):
                break
            y = _project_k(h - alpha * g, n, S)
            dlt = y - h
            if np.max(np.abs(dlt)) <= 1e-10:
                break
            gd = g.dot(dlt)
            if gd >= -1e-14:
                break
            acc = False
            for _ in range(25):
                if _sm_val(y, n, tau) <= Fv + 1e-4 * gd + 1e-14:
                    h = y; acc = True
                    s = _exact_score(h, n)
                    if s < bs:
                        bs = s; bl = h.copy()
                    break
                alpha *= 0.5
                y = _project_k(h - alpha * g, n, S)
                dlt = y - h
                gd = g.dot(dlt)
                if gd >= -1e-14:
                    break
            if not acc:
                break
            alpha = min(1.25 * alpha, 5.0)
    return bl, bs

def _slsqp_obj(z):
    return z[_SLSQP_N]

def _slsqp_obj_j(z):
    g = np.zeros(_SLSQP_N + 1)
    g[_SLSQP_N] = 1.0
    return g

def _slsqp_ieq(z):
    return z[_SLSQP_N] - _lag_scores(z[:_SLSQP_N], _SLSQP_N)

def _slsqp_ieq_j(z):
    h = z[:_SLSQP_N]
    n = _SLSQP_N
    L = 2 * n - 1
    J = np.zeros((L, n + 1))
    J[:, -1] = 1.0
    for k in range(L):
        ell = k - (n - 1)
        a = max(0, -ell); b = min(n, n - ell)
        J[k, a:b] -= _SLSQP_DX * (1.0 - h[a + ell:b + ell])
        c = max(0, ell); d2 = min(n, n + ell)
        J[k, c:d2] += _SLSQP_DX * h[c - ell:d2 - ell]
    return J

def _slsqp_eq(z):
    return z[:_SLSQP_N].sum() - _SLSQP_N / 2.0

def _slsqp_eq_j(z):
    g = np.zeros(_SLSQP_N + 1)
    g[:_SLSQP_N] = 1.0
    return g

def _try_slsqp(h_init, n, S):
    global _SLSQP_N, _SLSQP_DX
    if n > 100:
        return None
    _SLSQP_N = n
    _SLSQP_DX = 2.0 / n
    t0 = min(1.0, _exact_score(h_init, n) + 1e-8)
    z0 = np.concatenate([h_init, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_j},
        {'type': 'ineq', 'fun': _slsqp_ieq, 'jac': _slsqp_ieq_j}
    ]
    try:
        res = __import__('scipy.optimize', fromlist=['minimize']).minimize(
            _slsqp_obj, z0, jac=_slsqp_obj_j, bounds=bounds,
            constraints=cons, method='SLSQP',
            options={'maxiter': 250, 'ftol': 1e-10, 'disp': False})
        hs = res.x[:n]
        if not np.all(np.isfinite(hs)) or hs.shape != (n,):
            return None
        hs = np.clip(hs, 0, 1)
        hs = _project_k(hs, n, S)
        if not _check_feasible(hs, n):
            return None
        return hs
    except Exception:
        return None

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = None
    try:
        pv = globals().get('initial_h_values', None)
        if pv is not None:
            pv = np.asarray(pv, dtype=np.float64)
            if pv.ndim == 1 and 2 <= pv.shape[0] <= 999 and np.all(np.isfinite(pv)) and np.max(np.abs(pv)) < 1e6:
                parent = pv.copy()
    except Exception:
        parent = None
    if parent is not None:
        n = len(parent)
    else:
        n = int(kwargs.get('n_points', 45))
        if not (2 <= n <= 999):
            n = 45
    dx = 2.0 / n
    S = n / 2.0
    if parent is not None:
        fallback = _project_k(parent, n, S)
    else:
        fallback = np.full(n, 0.5)
    best_h = fallback.copy()
    best_score = _exact_score(best_h, n)
    tau_list = [0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4, 5e-5, 1e-5]
    starts = [fallback.copy(), (1.0 - fallback).copy(), fallback[::-1].copy(), (1.0 - fallback[::-1]).copy()]
    if parent is not None:
        starts.append(_project_k(parent + 0.08 * rng.standard_normal(n), n, S))
        starts.append(_project_k(parent + 0.25 * rng.standard_normal(n), n, S))
    starts.append(_project_k(rng.uniform(0, 1, n), n, S))
    starts.append(_project_k(rng.beta(0.5, 0.5, n), n, S))
    best_h_list = []
    for st in starts:
        if not _check_feasible(st, n):
            st = _project_k(st, n, S)
        s = _exact_score(st, n)
        if np.isfinite(s) and s < best_score - 1e-12:
            best_score = s; best_h = st.copy()
        for h_c in best_h_list:
            if np.max(np.abs(st - h_c)) < 1e-8:
                break
        else:
            best_h_list.append(st.copy())
        res_h, res_s = _proj_grad(st, n, S, tau_list)
        if np.isfinite(res_s) and res_s < best_score - 1e-12:
            best_score = res_s; best_h = res_h.copy()
        for h_c in best_h_list:
            if np.max(np.abs(res_h - h_c)) < 1e-5:
                break
        else:
            best_h_list.append(res_h.copy())
    for sig in [0.04, 0.12, 0.30, 0.50]:
        st = _project_k(best_h + sig * rng.standard_normal(n), n, S)
        res_h, res_s = _proj_grad(st, n, S, tau_list)
        if np.isfinite(res_s) and res_s < best_score - 1e-12:
            best_score = res_s; best_h = res_h.copy()
    if n <= 100:
        sorted_list = sorted(best_h_list, key=lambda x: _exact_score(x, n))
        slsqp_starts = []
        for h_c in sorted_list[:5]:
            if len(slsqp_starts) >= 3:
                break
            if all(np.max(np.abs(h_c - s)) > 1e-5 for s in slsqp_starts):
                slsqp_starts.append(h_c)
        if not slsqp_starts:
            slsqp_starts = [fallback.copy()]
        for h_init in slsqp_starts:
            hs = _try_slsqp(h_init, n, S)
            if hs is not None:
                s = _exact_score(hs, n)
                if np.isfinite(s) and s < best_score - 1e-12:
                    best_score = s; best_h = hs.copy()
    best_h = _project_k(best_h, n, S)
    if not _check_feasible(best_h, n):
        best_h = fallback.copy()
    fs = _exact_score(fallback, n)
    bs_score = _exact_score(best_h, n)
    if not np.isfinite(bs_score) or bs_score > fs:
        best_h = fallback.copy()
        bs_score = fs
    best_h = np.clip(best_h, 0.0, 1.0).astype(np.float64)
    final_score = float(np.max(np.correlate(best_h, 1.0 - best_h, mode='full')) * dx)
    return (best_h, final_score, n)
