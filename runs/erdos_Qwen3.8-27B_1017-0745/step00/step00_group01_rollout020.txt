    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return np.inf
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return np.inf
    if abs(np.sum(h) - n / 2.0) > 1e-8 * max(1, n):
        return np.inf
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)


def _lsq_and_grad(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = dx * corr
    c_max = c.max()
    a = np.exp((c - c_max) / tau)
    sa = a.sum()
    if sa <= 0.0 or not np.isfinite(sa):
        return np.inf, None
    F = c_max + tau * np.log(sa)
    p = a / sa
    conv1 = np.convolve(p, 1.0 - h, mode='full')
    U = conv1[n - 1:n - 1 + n]
    conv2 = np.convolve(p[::-1], h, mode='full')
    V = conv2[n - 1:n - 1 + n]
    grad = dx * (U - V)
    if not np.all(np.isfinite(grad)):
        return np.inf, None
    return F, grad


def _is_dup(s, starts):
    for e in starts:
        if np.max(np.abs(s - e)) < 1e-10:
            return True
    return False


def _make_starts(base, n, rng, max_starts):
    starts = []
    cands = [base.copy(), base[::-1].copy(), 1.0 - base[::-1].copy(), np.full(n, 0.5)]
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    cands.append(blk)
    alt = np.zeros(n)
    alt[0::2] = 1.0
    cands.append(alt)
    for sd in (0.05, 0.15, 0.35):
        cands.append(base + rng.standard_normal(n) * sd)
    cands.append(rng.random(n))
    for c in cands:
        if len(starts) >= max_starts:
            break
        s = _project_simplex(c, n)
        if not _is_dup(s, starts):
            starts.append(s)
    guard = 0
    while len(starts) < max_starts and guard < 20:
        guard += 1
        s = _project_simplex(base + rng.standard_normal(n) * 0.2, n)
        if not _is_dup(s, starts):
            starts.append(s)
    return starts


def run(seed=42, **kwargs):
    if not isinstance(seed, (int, np.integer)):
        seed = 42
    rng = np.random.default_rng(seed)
    base = None
    parent = globals().get('initial_h_values', None)
    if parent is not None:
        try:
            p = np.array(parent, dtype=np.float64).ravel()
            if p.size > 0:
                n = p.size
                if np.all(np.isfinite(p)):
                    base = _project_simplex(p.copy(), n)
                else:
                    base = np.full(n, 0.5)
        except Exception:
            base = None
    if base is None:
        n = 47
        if 'n_points' in kwargs:
            try:
                n = int(kwargs['n_points'])
                n = max(1, min(n, 999))
            except (ValueError, TypeError):
                n = 47
        base = np.full(n, 0.5)

    if n <= 64:
        ms_max, it_max = 12, 150
    elif n <= 256:
        ms_max, it_max = 6, 90
    else:
        ms_max, it_max = 2, 40
    if n <= 64:
        ms_def, it_def = 10, 100
    elif n <= 256:
        ms_def, it_def = 5, 60
    else:
        ms_def, it_def = 2, 25

    try:
        max_starts = int(kwargs.get('restarts', ms_def))
    except (ValueError, TypeError):
        max_starts = ms_def
    max_starts = max(1, min(max_starts, ms_max))

    try:
        iters_per_tau = int(kwargs.get('iters_per_tau', it_def))
    except (ValueError, TypeError):
        iters_per_tau = it_def
    iters_per_tau = max(1, min(iters_per_tau, it_max))

    temperatures = (0.02, 0.005, 0.001, 0.0002, 0.00005, 0.00001)
    fallback = base.copy()
    fallback_score = _exact_score(fallback, n)
    incumbent = base.copy()
    incumbent_score = _exact_score(incumbent, n)
    starts = _make_starts(base, n, rng, max_starts)

    for s in starts:
        sc = _exact_score(s, n)
        if np.isfinite(sc) and sc < incumbent_score - 1e-12:
            incumbent = s.copy()
            incumbent_score = sc

    if n <= 999:
        for si in range(len(starts)):
            x = starts[si].copy()
            try:
                for tau in temperatures:
                    F, grad = _lsq_and_grad(x, n, tau)
                    if not np.isfinite(F) or grad is None:
                        break
                    alpha = 1.0
                    stagnation = 0
                    for _it in range(iters_per_tau):
                        if not np.all(np.isfinite(grad)):
                            break
                        accepted = False
                        a_trial = alpha
                        for _tr in range(24):
                            y = _project_simplex(x - a_trial * grad, n)
                            if np.max(np.abs(y - x)) <= 1e-9:
                                break
                            D = np.sum((y - x) ** 2)
                            Fy, _ = _lsq_and_grad(y, n, tau)
                            if (np.isfinite(Fy) and
                                    Fy <= F - 1e-4 * D / a_trial + 1e-14 * max(1.0, abs(F))):
                                improvement = F - Fy
                                x = y
                                F = Fy
                                F, grad = _lsq_and_grad(x, n, tau)
                                alpha = min(2.0, 1.25 * a_trial)
                                accepted = True
                                if improvement <= 1e-12 * max(1.0, abs(F)):
                                    stagnation += 1
                                else:
                                    stagnation = 0
                                break
                            a_trial *= 0.5
                        if not np.all(np.isfinite(x)):
                            break
                        sc = _exact_score(x, n)
                        if np.isfinite(sc) and sc < incumbent_score - 1e-12:
                            incumbent = x.copy()
                            incumbent_score = sc
                        if not accepted or stagnation >= 5:
                            break
            except Exception:
                continue

    incumbent = _project_simplex(incumbent, n)
    final_score = _exact_score(incumbent, n)
    fallback = _project_simplex(fallback, n)
    fallback_score = _exact_score(fallback, n)
    if not np.isfinite(final_score) or final_score > fallback_score:
        incumbent = fallback
        final_score = fallback_score
    if not np.isfinite(final_score):
        incumbent = np.full(n, 0.5)
        final_score = _exact_score(incumbent, n)
    return incumbent, final_score, n
