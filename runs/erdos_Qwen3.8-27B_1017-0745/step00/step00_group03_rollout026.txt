        G[r, k:k + L] += nh[:L]
        G[r, :L] -= h[k:k + L]

            G[r2, :L] += nh[k:k + L]

        cp.norm(sdx * d) <= t - u,

    ]
    prob = cp.Problem(cp.Minimize(t), constraints)

def project_box_sum(z, N):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape[0] != N or not np.all(np.isfinite(z)) or np.max(np.abs(z)) > 1e100:
        return np.full(N, 0.5, dtype=np.float64)
    m = N / 2.0
    lo, hi = z.min() - 1.0, z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    e = m - h.sum()
    for _ in range(3):
        if abs(e) < 1e-12:
            break
        for i in range(N):
            if abs(e) < 1e-12:
                break
            if e > 0:
                a = min(e, 1.0 - h[i])
                h[i] += a
                e -= a
            else:
                a = min(-e, h[i])
                h[i] -= a
                e += a
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(h.sum() - m) > 1e-10 * max(1, N):
        return np.full(N, 0.5, dtype=np.float64)
    return h

        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / N))

def lag_grad(h, N):
    q = np.correlate(h, 1.0 - h, mode='full')
    G = np.zeros((2 * N - 1, N), dtype=np.float64)
    nh = 1.0 - h
    for k in range(N - 1, -1, -1):
        r = k + N - 1
        L = N - k
        G[r, k:k + L] += nh[:L]
        G[r, :L] -= h[k:k + L]
        if k > 0:
            r2 = N - 1 - k
            G[r2, :L] += nh[k:]
            G[r2, k:] -= h[:L]
    return q, G

    u = cp.Variable()
    t = cp.Variable()
    x_p = cp.Parameter(N)
    r0_p = cp.Parameter()
    J0_p = cp.Parameter(N)
    nz = np.array([r for r in range(2 * N - 1) if r != N - 1])
    M = len(nz)
    rn_p = cp.Parameter(M)
    Jn_p = cp.Parameter((M, N))
    R_p = cp.Parameter()
    sdx = math.sqrt(dx)
    nrm_d = cp.norm(d, 2)
    cons = [
        x_p + d >= 0,
        x_p + d <= 1,
        cp.sum(d) == 0,
        nrm_d <= R_p,
        u >= rn_p + Jn_p @ d,
        cp.norm(sdx * d) <= t - u,
        t >= r0_p + J0_p @ d,
    ]
    prob = cp.Problem(cp.Minimize(t), cons)
    return d, u, t, x_p, r0_p, J0_p, rn_p, Jn_p, R_p, prob, nz
