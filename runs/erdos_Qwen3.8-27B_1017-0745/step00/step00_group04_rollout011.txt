    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1

    G = np.zeros((M, n))
    q[n-1] = dx * np.dot(h, 1.0 - h)
    G[n-1] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        npd = n - d
        ip = n - 1 + d
        im = n - 1 - d
        q[ip] = dx * np.dot(h[d:n], 1.0 - h[:npd])
        q[im] = dx * np.dot(h[:npd], 1.0 - h[d:n])

import numpy as np

def lag_values(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    q[n - 1] = dx * np.dot(h, 1.0 - h)
    G[n - 1] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        npd = n - d
        ip = n - 1 + d
        im = n - 1 - d
        q[ip] = dx * np.dot(h[d:], 1.0 - h[:npd])
        q[im] = dx * np.dot(h[:npd], 1.0 - h[d:])
        G[ip, d:] += dx * (1.0 - h[:npd])
        G[ip, :npd] += -dx * h[d:]
        G[im, :npd] += dx * (1.0 - h[d:])
        G[im, d:] += -dx * h[:npd]
    return q, G
