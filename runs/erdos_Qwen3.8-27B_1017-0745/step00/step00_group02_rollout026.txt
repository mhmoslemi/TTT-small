import numpy as np
import math

_slp = {}

def _safe_parent():
    try:
        v = globals().get('initial_h_values', None)
        if v is None:
            return None
        v = np.array(v, dtype=np.float64).ravel()
        if v.size < 2 or v.size > 999:
            return None
        if not np.all(np.isfinite(v)) or np.max(np.abs(v)) > 1e6:
            return None
        return v
    except Exception:
        return None

def project_k(y, n, S):
    y = np.asarray(y, dtype=np.float64)
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return 0.5 * np.ones(n)
    lo, hi = float(np.min(y)) - 1.0, float(np.max(y)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        c = np.clip(y - mid, 0.0, 1.0)
        if c.sum() > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    x = np.clip(y - lam, 0.0, 1.0)
    res = S - x.sum()
    if abs(res) > 1e-13:
        if res > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += res / cnt
    x = np.clip(x, 0.0, 1.0)
    if abs(x.sum() - S) > 1e-10:
        return 0.5 * np.ones(n)
    return x

def exact_score(h, dx):
    if h.size < 2 or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def is_feasible(h, n, S):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)) or h.min() < -1e-9 or h.max() > 1.0 + 1e-9:
        return False
    return abs(h.sum() - S) <= 1e-8

def smooth_val(h, dx, tau):
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(q))
    w = np.exp((q - m) / tau)
    return m + tau * float(np.log(np.sum(w)))

def smooth_grad(h, dx, tau):
    n = h.size
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(q))
    w = np.exp((q - m) / tau)
    p = w / float(np.sum(w))
    L = 2 * n - 1
    g = np.zeros(n)
    for k in range(L):
        Lag = k - (n - 1)
        lo = max(0, -Lag)
        hi = min(n, n - Lag)
        if hi <= lo:
            continue
        a = h[lo:hi]
        b = h[lo + Lag:hi + Lag]
        g[lo + Lag:hi + Lag] += p[k] * dx * (1.0 - a)
        g[lo:hi] -= p[k] * dx * b
    return g

def projected_gradient(h0, n, S, dx, taus, max_it):
    h = h0.copy()
    for tau in taus:
        alpha = 1.0
        for _ in range(max_it):
            F = smooth_val(h, dx, tau)
            if not np.isfinite(F):
                return h
            g = smooth_grad(h, dx, tau)
            if not np.all(np.isfinite(g)):
                return h
            accept = False
            for _ in range(25):
                y = project_k(h - alpha * g, n, S)
                d = y - h
                if float(np.max(np.abs(d))) < 1e-10:
                    break
                gd = float(np.dot(g, d))
                if gd >= -1e-14:
                    break
                Fy = smooth_val(y, dx, tau)
                if Fy <= F + 1e-4 * gd + 1e-14:
                    h = y
                    accept = True
                    break
                alpha *= 0.5
            if not accept:
                break
            alpha = min(1.25 * alpha, 5.0)
    return h

def slsqp_obj(z):
    return z[-1]

def slsqp_obj_jac(z):
    n = _slp['n']
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return g

def slsqp_ineq(z):
    h = z[:-1]
    t = z[-1]
    dx = _slp['dx']
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - q

def slsqp_ineq_jac(z):
    h = z[:-1]
    n = _slp['n']
    dx = _slp['dx']
    L = 2 * n - 1
    J = np.zeros((L, n + 1))
    for k in range(L):
        Lag = k - (n - 1)
        lo = max(0, -Lag)
        hi = min(n, n - Lag)
        if hi <= lo:
            continue
        a = h[lo:hi]
        b = h[lo + Lag:hi + Lag]
        J[k, lo + Lag:hi + Lag] -= dx * (1.0 - a)
        J[k, lo:hi] += dx * b
        J[k, -1] = 1.0
    return J

def slsqp_eq(z):
    n = _slp['n']
    S = _slp['S']
    return np.array([np.sum(z[:n]) - S])

def slsqp_eq_jac(z):
    n = _slp['n']
    J = np.zeros((1, n + 1))
    J[0, :n] = 1.0
    return J

def slsqp_refine(h0, n, S, dx):
    if n > 100:
        return h0
    try:
        from scipy.optimize import minimize as _min
    except Exception:
        return h0
    _slp['n'] = n
    _slp['S'] = S
    _slp['dx'] = dx
    s0 = exact_score(h0, dx)
    z0 = np.concatenate([h0, [min(1.0, s0 + 1e-8)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    try:
        res = _min(slsqp_obj, z0, method='SLSQP', jac=slsqp_obj_jac,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': 250, 'ftol': 1e-10, 'disp': False})
        h_out = res.x[:n].copy()
        if h_out.size == n and np.all(np.isfinite(h_out)):
            if np.all(h_out >= -1e-6) and np.all(h_out <= 1.0 + 1e-6):
                if abs(np.sum(h_out) - S) < 1e-4:
                    h_out = project_k(np.clip(h_out, 0, 1), n, S)
                    return h_out
    except Exception:
        pass
    return h0

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = _safe_parent()
    if parent is not None:
        n = int(parent.size)
    else:
        n = kwargs.get('n_points', 45)
        if not isinstance(n, int) or n < 2 or n > 999:
            n = 45
    dx = 2.0 / n
    S = n / 2.0

    if parent is not None:

    else:
        fallback = 0.5 * np.ones(n)

    best_h = fallback.copy()
    best_s = exact_score(best_h, dx)

    # Build starts
    starts = [best_h.copy(), (1.0 - best_h).copy(),
              best_h[::-1].copy(), (1.0 - best_h[::-1]).copy()]
    if parent is not None:
        starts.append(project_k(parent + 0.08 * rng.standard_normal(n), n, S))
        starts.append(project_k(parent + 0.25 * rng.standard_normal(n), n, S))
    starts.append(project_k(rng.uniform(0, 1, n), n, S))
    starts.append(project_k(rng.beta(0.5, 0.5, n), n, S))

    # Score starts
    for s in starts:
        sc = exact_score(s, dx)
        if sc < best_s - 1e-12:
            best_h = s.copy()
            best_s = sc

    # Projected gradient
    if n <= 100:
        taus = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00001]
        max_it = 80
    else:
        taus = [0.02, 0.005, 0.001, 0.0001, 0.00001]
        max_it = 20

    local_results = []
    for s in starts:
        h_opt = projected_gradient(s, n, S, dx, taus, max_it)
        sc = exact_score(h_opt, dx)
        if sc < best_s - 1e-12:
            best_h = h_opt.copy()
            best_s = sc
        local_results.append(h_opt)

    # Basin perturbations
    sigmas = [0.04, 0.12, 0.30, 0.50] if n <= 100 else [0.12]
    for sig in sigmas:
        z = rng.standard_normal(n)
        pert = project_k(best_h + sig * z, n, S)
        h_opt = projected_gradient(pert, n, S, dx, taus, max_it)
        sc = exact_score(h_opt, dx)
        if sc < best_s - 1e-12:
            best_h = h_opt.copy()
            best_s = sc
        local_results.append(h_opt)

    # SLSQP refinement

        valid = [lr for lr in local_results if is_feasible(lr, n, S)]
        if valid:
            sc_arr = np.array([exact_score(v, dx) for v in valid])
            order = np.argsort(sc_arr)
            refined = []
            for idx in order:
                if len(refined) >= 3:
                    break
                cand = valid[idx].copy()
                ok = True
                for r in refined:
                    if float(np.max(np.abs(cand - r))) < 1e-5:
                        ok = False
                        break
                if ok:
                    refined.append(cand)

                refined.append(fallback.copy())
            for rs in refined:
                h_r = slsqp_refine(rs, n, S, dx)
                sc = exact_score(h_r, dx)

                    best_s = sc

    final_s = exact_score(best_h, dx)
    fb_s = exact_score(fallback, dx)
    if not is_feasible(best_h, n, S) or final_s > fb_s:

        final_s = fb_s
    if not is_feasible(best_h, n, S):

        final_s = exact_score(best_h, dx)

    best_h = np.asarray(best_h, dtype=np.float64)
    final_s = float(exact_score(best_h, dx))
    return (best_h, final_s, int(n))
