def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):

def haar_basis(n):
    if n <= 1:
        return np.zeros((0, n))
    dirs = []
    stack = [(0, n)]
    while stack:
        a, b = stack.pop()
        m = b - a
        if m < 2:
            continue
        mL = m // 2
        mR = m - mL
        sp = a + mL
        v = np.zeros(n)
        v[a:sp] = math.sqrt(mR / (m * mL))
        v[sp:b] = -math.sqrt(mL / (m * mR))
        dirs.append(v)
        if mL >= 2:
            stack.append((a, sp))
        if mR >= 2:
            stack.append((sp, b))
    return np.array(dirs)

def get_interval(x, v):
    nz = np.nonzero(v)[0]
    if len(nz) == 0:
        return (0.0, 0.0)
    lo = -np.inf
    hi = np.inf
    for i in nz:
        vi = v[i]
        r0 = -x[i] / vi
        r1 = (1.0 - x[i]) / vi

        hi = min(hi, max(r0, r1))

        return None
    lo = min(lo, 0.0)
    hi = max(hi, 0.0)

    return (lo, hi)
