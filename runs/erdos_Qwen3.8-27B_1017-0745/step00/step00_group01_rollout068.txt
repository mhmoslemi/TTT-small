        return fallback.astype(np.float64), fb_score, n

    sd = seed if isinstance(seed, (int, np.integer)) else 42
    rng = np.random.default_rng(int(sd))
    taurs = [0.02, 0.005, 0.001, 0.0002, 0.00005, 0.00001]

    starts = build_starts(base, n, rng)
    starts = starts[:n_starts]

    best_x = fallback.copy()
    best_score = fb_score

    for s in starts:
        sc = exact_score(s, n)
        if sc < best_score - 1e-12:
            best_score = sc
            best_x = s.copy()

    for s in starts:
        try:
            bx, bs = pgd_search(s, n, taurs, n_iter)
            if bs < best_score - 1e-12:
                best_score = bs
                best_x = bx.copy()
        except Exception:
            continue

    best_x = capped_simplex_project(best_x, n / 2.0)
    best_x = np.clip(best_x, 0.0, 1.0)
    final_score = exact_score(best_x, n)
    if np.isinf(final_score):
        best_x = np.full(n, 0.5)
        final_score = exact_score(best_x, n)
    if final_score > fb_score:
        best_x = fallback
        final_score = fb_score

    return best_x.astype(np.float64), final_score, n
