def project_box_sum(z, N):

import numpy as np

def project_box_sum(z, N):
    m = N / 2.0
    if z.shape != (N,) or not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(z - mid, 0.0, 1.0)
        if float(np.sum(h)) > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    res = m - float(np.sum(h))
    if res > 0:
        for i in range(N):
            adj = min(res, 1.0 - h[i])
            h[i] += adj
            res -= adj
            if abs(res) < 1e-12:
                break
    elif res < 0:
        for i in range(N):
            adj = min(-res, h[i])
            h[i] -= adj
            res += adj
            if abs(res) < 1e-12:

        else:

    return q

def _smooth_core(h, tau, N, want_grad):
    dx = 2.0 / N
    M = 2 * N - 1
    q = lag_values(h, N)
    r = dx * q
    rmax = float(np.max(r))
    shifted = (r - rmax) / tau
    ev = np.exp(shifted)
    tot = float(np.sum(ev))
    phi = rmax + tau * float(np.log(tot))
    p = ev / tot
    if not want_grad:
        return phi
    grad = np.zeros(N)
    for s in range(M):
        w = dx * p[s]
        if w < 1e-15:
            continue
        lag = s - (N - 1)
        if lag >= 0:
            L = N - lag
            grad[0:L] += w * (1.0 - h[lag:lag + L])
            grad[lag:N] -= w * h[0:L]
        else:
            d = -lag
            L = N - d
            grad[d:N] += w * (1.0 - h[0:L])
            grad[0:L] -= w * h[d:N]
    return phi, grad
