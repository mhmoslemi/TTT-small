        F_curr, g = surrogate_and_grad(h, dx, tau)
        if not np.all(np.isfinite(g)):
            continue
        eta = 1.0
        stall = 0
        for _ in range(per_level):
            trial_eta = eta
            accepted = False
            for _bs in range(16):
                if trial_eta < 1e-8:
                    break
                trial = project_capped_simplex(h - trial_eta * g, S)
                dv = trial - h
                if np.max(np.abs(dv)) < 1e-15:
                    break
                gd = np.dot(g, dv)
                if gd >= 0:
                    trial_eta *= 0.5
                    continue
                F_t, _ = surrogate_and_grad(trial, dx, tau)
                if F_t <= F_curr + 1e-4 * gd + 1e-12 * max(1.0, abs(F_curr)):
                    h = trial
                    F_curr = F_t
                    accepted = True
                    break
                trial_eta *= 0.5
            if not accepted:
                break
            eta = min(8.0, 1.2 * trial_eta)
            _, g = surrogate_and_grad(h, dx, tau)
            if not np.all(np.isfinite(g)):
                break
            if abs(F_curr) < 1e-10:
                stall += 1
                if stall >= 12:
                    break
            else:
                stall = 0
    return h

def pair_transfer(h, n, rng, P):
    h = h.copy()
    best_h = h.copy()
    best_s = exact_score(h)
    stages = [(0.08, 0.001), (0.025, 0.0003), (0.008, 0.00008), (0.002, 0.0)]
    per = max(1, P // len(stages))
    for slimit, T in stages:
        for _ in range(per):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            if rng.random() < 0.5:
                cap = min(1.0 - h[i], h[j])
                sgn = 1.0
            else:
                cap = min(h[i], 1.0 - h[j])
                sgn = -1.0
            mag = sgn * rng.random() * min(slimit, cap)
            if abs(mag) < 1e-12:
                continue
            trial = h.copy()
            trial[i] += mag
            trial[j] -= mag
            if trial[i] < -1e-15 or trial[i] > 1 + 1e-15 or trial[j] < -1e-15 or trial[j] > 1 + 1e-15:
                continue
            s_new = exact_score(trial)
            if s_new <= best_s + 1e-15:
                h = trial
                if s_new < best_s:
                    best_s = s_new
                    best_h = trial.copy()
            elif T > 0 and math.exp(-(s_new - best_s) / T) > rng.random():
                h = trial
    return best_h, best_s

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = None
    if 'initial_h_values' in globals():
        try:
            p = np.array(globals()['initial_h_values'], dtype=np.float64).ravel()
            if p.size >= 2 and np.all(np.isfinite(p)):
                parent = p
        except Exception:
            parent = None
    if parent is not None:
        n = len(parent)
    else:
        n = max(8, min(512, int(kwargs.get('n_points', 80))))
    dx = 2.0 / n
    S = n / 2.0
    if parent is not None:
        fallback = project_capped_simplex(parent, S)
    else:
        fallback = np.full(n, 0.5)
    best_h = fallback.copy()
    best_score = exact_score(fallback)
    starts = [fallback.copy()]
    if parent is not None:
        for sd in [0.04, 0.15, 0.35]:
            starts.append(project_capped_simplex(parent + sd * rng.standard_normal(n), S))
    q = int(rng.integers(1, n))
    phi = rng.random() * 2.0 * math.pi
    starts.append(project_capped_simplex(0.5 + 0.45 * np.cos(2.0 * math.pi * q * np.arange(n) / n + phi), S))
    starts.append(project_capped_simplex(rng.random(n), S))
    perm = rng.permutation(n)
    bb = np.zeros(n)
    bb[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        bb[perm[n // 2]] = 0.5
    starts.append(project_capped_simplex(bb, S))
    unique_starts = [starts[0]]
    for s in starts[1:]:
        if all(np.max(np.abs(s - u)) > 1e-10 for u in unique_starts):
            unique_starts.append(s)
    max_starts = 7 if n <= 256 else 3
    unique_starts = unique_starts[:max_starts]
    baseline = best_score
    for s in unique_starts:
        sc = exact_score(s)
        if sc < best_score:
            best_score = sc
            best_h = s.copy()
    B = min(480, max(12, int(4000000 / (n * n))))
    tau_sched = [0.02, 0.004, 0.0008, 0.00016, 0.000032, 0.0000064]
    for s in unique_starts:
        try:
            opt = optimize_smooth(s, n, dx, tau_sched, B, rng)
            sc = exact_score(opt)
            if sc < best_score:
                best_score = sc
                best_h = opt.copy()
        except Exception:
            pass
    if best_score > baseline - 1e-8 and n <= 256:
        for sd in [0.6, 1.0]:
            try:
                ns = project_capped_simplex(best_h + sd * rng.standard_normal(n), S)
                opt = optimize_smooth(ns, n, dx, tau_sched, max(6, B // 2), rng)
                sc = exact_score(opt)
                if sc < best_score:
                    best_score = sc
                    best_h = opt.copy()
            except Exception:
                pass
    P = max(80, min(800, int(10000000 / (n * n))))
    try:
        best_h, best_score = pair_transfer(best_h, n, rng, P)
    except Exception:
        pass
    best_h = np.clip(best_h, 0.0, 1.0)
    if abs(np.sum(best_h) - S) > 1e-10:
        best_h = project_capped_simplex(best_h, S)
    final_score = exact_score(best_h)
    if final_score >= fallback_score:
        best_h = fallback.copy()
        final_score = fallback_score
    if not np.all(np.isfinite(best_h)):
        best_h = fallback.copy()
        final_score = fallback_score
    return (best_h, final_score, n)
