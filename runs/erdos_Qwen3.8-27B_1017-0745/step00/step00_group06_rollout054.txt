    for _ in range(70):
        lam = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x - lam, 0.0, 1.0)))
        if s > m:
            lo = lam
        else:
            hi = lam
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    delta = m - float(np.sum(y))
    if delta > 1e-14:
        for i in range(n):
            a = min(delta, 1.0 - y[i])
            y[i] += a
            delta -= a
            if delta < 1e-14:
                break
    elif delta < -1e-14:
        for i in range(n):
            a = min(-delta, y[i])
            y[i] -= a
            delta += a
            if -delta < 1e-14:
                break
    return y


def _score_key(c):
    return c[1]


def build_starts(n, m, parent, rng, nrestarts):
    starts = []
    if parent is not None:
        starts.append(proj(parent, n, m))
    starts.append(np.full(n, 0.5))
    k = int(np.floor(m))
    sv = np.zeros(n)
    sv[:k] = 1.0
    sv[k] = m - k
    starts.append(sv.copy())
    for f in (1, 3, 7):
        i = np.arange(n)
        starts.append(proj(0.5 + 0.22 * np.sin(2.0 * np.pi * f * i / n), n, m))
    scales = [0.04, 0.10, 0.20]
    base = parent if parent is not None else np.full(n, 0.5)
    idx = 0
    while len(starts) < nrestarts:
        s = scales[idx % 3]
        noise = rng.standard_normal(n)
        noise = 0.5 * noise + 0.25 * np.roll(noise, 1) + 0.25 * np.roll(noise, -1)
        starts.append(proj(base + s * noise, n, m))
        idx += 1
    dedup = [starts[0]]
    for s in starts[1:]:
        if all(np.max(np.abs(s - d)) > 1e-8 for d in dedup):
            dedup.append(s)
        if len(dedup) >= nrestarts:
            break
    return dedup[:nrestarts]


def run(seed=42, **kwargs):
    global _N, _M, _D
    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if len(pv) >= 2 and np.all(np.isfinite(pv)):
            parent = pv.copy()
    except (NameError, Exception):
        parent = None
    if parent is not None:
        n = len(parent)
    else:
        try:
            n = int(kwargs.get('n_points', 95))
        except (ValueError, TypeError):
            n = 95
        n = max(2, min(n, 999))
    n = min(n, 999)
    _N = n
    m = n / 2.0
    _M = m
    _D = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    eq_s = {'fun': mfun_s, 'jac': mgrad_s, 'type': 'eq'}

    if parent is not None:
        incumbent = proj(parent, n, m)
    else:
        incumbent = np.full(n, 0.5)
    incumbent_score = exact_score(incumbent)

    try:
        rng = np.random.default_rng(int(seed))
    except (ValueError, TypeError):
        rng = np.random.default_rng(42)

    try:
        nrestarts = int(kwargs.get('n_restarts', 8))
    except (ValueError, TypeError):
        nrestarts = 8
    nrestarts = max(6, min(nrestarts, 12))

    try:
        maxiter = int(kwargs.get('maxiter', 120))
    except (ValueError, TypeError):
        maxiter = 120
    maxiter = max(60, min(maxiter, 250))

    starts = build_starts(n, m, parent, rng, nrestarts)
    candidates = [(incumbent.copy(), incumbent_score)]
    taus = [0.02, 0.005, 0.001, 0.0002]

    for s0 in starts:
        cur = s0.copy()
        candidates.append((cur.copy(), exact_score(cur)))
        for tau in taus:
            _TAU = tau
            try:
                res = minimize(smooth_fun, cur, jac=smooth_grad, method='SLSQP',
                               bounds=bounds, constraints=[eq_s],
                               options={'ftol': 1e-10, 'maxiter': maxiter, 'disp': False})
                if res.x is not None and np.all(np.isfinite(res.x)) and len(res.x) == n:
                    cur = proj(res.x, n, m)
                else:
                    continue
            except Exception:
                continue
            candidates.append((cur.copy(), exact_score(cur)))

    candidates.sort(key=_score_key)
    top = []
    for c in candidates:
        if all(np.max(np.abs(c[0] - t[0])) > 1e-8 for t in top):
            top.append(c)
        if len(top) >= 3:
            break

    epi_maxiter = min(400, 2 * maxiter)
    eq_e = {'fun': mfun_e, 'jac': mgrad_e, 'type': 'eq'}
    ineq_e = {'fun': epib_ineq, 'jac': epib_ineq_jac, 'type': 'ineq'}
    epi_bounds = bounds + [(0.0, 2.1)]

    for h_c, _ in top:
        z0 = np.concatenate([h_c, [exact_score(h_c) + 1e-6]])
        try:
            res = minimize(epib_fun, z0, jac=epib_grad, method='SLSQP',
                           bounds=epi_bounds, constraints=[eq_e, ineq_e],
                           options={'ftol': 1e-10, 'maxiter': epi_maxiter, 'disp': False})
            if res.x is not None and np.all(np.isfinite(res.x)) and len(res.x) == n + 1:
                h_new = proj(res.x[:n], n, m)
                s_new = exact_score(h_new)
                if s_new < incumbent_score - 1e-12:
                    incumbent = h_new
                    incumbent_score = s_new
        except Exception:
            pass

    incumbent = proj(incumbent, n, m)
    incumbent = np.clip(incumbent, 0.0, 1.0)
    if abs(float(np.sum(incumbent)) - m) > 1e-9 * max(1, n):
        incumbent = np.full(n, 0.5)
    c5 = exact_score(incumbent)
    return (incumbent, c5, n)
