_tau_h = [0.02]
_n_h = [95]

def _exact_score(h):
    h = np.asarray(h, dtype=np.float64)
    if h.ndim != 1 or not np.all(np.isfinite(h)):
        return np.inf
    n = len(h)
    if n < 2:
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _corr_jac(h):
    n = len(h)
    d = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode='full') * d
    jac = np.zeros((2 * n - 1, n), dtype=np.float64)
    for q in range(-(n - 1), n):
        row = q + n - 1
        jlo = max(0, -q)
        jhi = min(n, n - q)
        for j in range(jlo, jhi):
            jac[row, j + q] += d * (1.0 - h[j])
            jac[row, j] -= d * h[j + q]
    return r, jac

def _project(x, m, n):

            delta -= add
