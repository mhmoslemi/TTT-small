    z = np.asarray(z, dtype=np.float64).ravel()

    hi = np.max(z)

def project_box_sum(z, N):
    m = N / 2.0
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape[0] != N or not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)
    e = m - h.sum()
    if abs(e) > 1e-13:
        for i in range(N):
            if e > 0:
                adj = min(e, 1.0 - h[i])
                if adj <= 0:
                    break
                h[i] += adj
                e -= adj
            else:
                adj = min(-e, h[i])
                if adj <= 0:

                h[i] -= adj
                e += adj
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):

def lag_values(h, N):
    q = np.empty(2 * N - 1, dtype=np.float64)
    v = 1.0 - h
    for k in range(-(N - 1), N):
        if k >= 0:
            L = N - k
            q[N - 1 + k] = np.dot(h[k:k + L], v[0:L])
        else:
            d = -k
            L = N - d
            q[N - 1 + k] = np.dot(h[0:L], v[d:d + L])
    return q


def smooth_value(h, tau, N):
    q = lag_values(h, N)
    r = (2.0 / N) * q
    rmax = float(np.max(r))
    total = float(np.exp((r - rmax) / tau).sum())
    return rmax + tau * np.log(total)


def smooth_value_grad(h, tau, N):
    q = lag_values(h, N)
    r = (2.0 / N) * q
    rmax = float(np.max(r))

    total = float(ex.sum())

    phi = rmax + tau * np.log(total)
    v = 1.0 - h
    g = np.zeros(N, dtype=np.float64)

        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / N))
