def project_box_sum(z, N):
    m = N / 2.0
    if z.shape != (N,) or not np.all(np.isfinite(z)) or np.max(np.abs(z)) > 1e100:
        return np.full(N, 0.5)
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    e = m - np.sum(h)
    for _ in range(3):
        if abs(e) < 1e-14:
            break
        for i in range(N):
            if e > 0:
                amt = min(e, 1.0 - h[i])
                h[i] += amt; e -= amt
            elif e < 0:
                amt = min(-e, h[i])
                h[i] -= amt; e += amt
            else:
                break
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - m) > 1e-10 * max(1, N) or not np.all(np.isfinite(h)):
        return np.full(N, 0.5)
    return h

def is_feasible(h, N):
    m = N / 2.0
    return (h.shape == (N,) and np.all(np.isfinite(h)) and np.all(h >= -1e-12)
            and np.all(h <= 1.0 + 1e-12) and abs(np.sum(h) - m) <= 1e-10 * max(1, N))

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / N))

def lag_coefficients(h, d, N):
    M = 2 * N - 1
    A = np.zeros(M); B = np.zeros(M); C = np.zeros(M)
    for k in range(-(N-1), N):
        idx = k + N - 1
        if k >= 0:
            L = N - k; a, b = h[k:k+L], h[:L]; u, v = d[k:k+L], d[:L]
        else:
            s = -k; L = N - s; a, b = h[:L], h[s:s+L]; u, v = d[:L], d[s:s+L]
        A[idx] = np.dot(a, 1.0 - b)
        B[idx] = np.dot(u, 1.0 - b) - np.dot(a, v)
        C[idx] = -np.dot(u, v)
    if not (np.all(np.isfinite(A)) and np.all(np.isfinite(B)) and np.all(np.isfinite(C))):
        return None
    return A, B, C

def make_directions(h, N, rng, stage):

    dirs = []

    others = [i for i in range(N) if i != anchor]
    for ii in range(min(ns, len(others))):

    npair = sum(1 for d in dirs)
    used = set()

        if npair >= nr: break

            if (i,j) not in used:

def evaluate_pass(start_h, N, stage, rng):
    h = start_h.copy()
    cs = score_exact(h, N)
    for d in make_directions(h, N, rng, stage):
        try:
            if not np.all(np.isfinite(d)) or np.max(np.abs(d)) < 1e-15: continue
            d = d.copy(); d[0] -= np.sum(d)
            if np.max(np.abs(d)) < 1e-15: continue

            nh, ns = line_search(h, d, N, cs)
            if ns < cs - 1e-15:
                h, cs = nh, ns
        except Exception:
            pass
    return h, cs
