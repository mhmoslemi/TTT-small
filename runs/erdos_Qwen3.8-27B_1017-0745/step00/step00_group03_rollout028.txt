import math

def project_box_sum(z, N):
    if z.shape != (N,):
        return np.full(N, 0.5)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)

        return np.full(N, 0.5)

    for _ in range(3):
        if abs(e) < 1e-15:
            break
        for i in range(N):
            if e > 0:
                add = min(e, 1.0 - v[i])

    return np.full(N, 0.5)

def feasible(h, N):
    if h.shape != (N,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < 0.0) or np.any(h > 1.0):
        return False
    if abs(np.sum(h) - N / 2.0) > 1e-10 * max(1, N):
        return False
    return True

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    s = float(np.max(c)) * (2.0 / N)
    if not math.isfinite(s):
        return float('inf')
    return s

def make_fourier(rng, N):
    R = min(5, (N - 1) // 2)
    if R <= 0:
        return np.full(N, 0.5)
    i = np.arange(N, dtype=np.float64)
    z = np.full(N, 0.5)
    for r in range(1, R + 1):
        A = rng.standard_normal()
        B = rng.standard_normal()
        z += (0.12 / r) * (A * np.sin(2.0 * np.pi * r * i / N) + B * np.cos(2.0 * np.pi * r * i / N))
    return z

def init_population(rng, N, baseline, crd_flip):
    P = crd_flip[0]
    X = np.zeros((P, N), dtype=np.float64)
    fit = np.full(P, float('inf'), dtype=np.float64)
    X[0] = baseline
    fit[0] = score_exact(baseline, N)
    u = np.full(N, 0.5)
    X[1] = u
    fit[1] = score_exact(u, N)
    if crd_flip[1]:
        X[2] = project_box_sum(1.0 - baseline, N)
    else:
        X[2] = project_box_sum(rng.standard_normal(N), N)
    fit[2] = score_exact(X[2], N)
    sigmas = [0.02, 0.05, 0.10, 0.20, 0.40]
    si = 0
    for k in range(3, P):
        phase = (k - 3) % 8
        if phase < 5:
            z = baseline + sigmas[si % 5] * rng.standard_normal(N)
            si += 1
        elif phase == 5:
            z = rng.uniform(0.0, 1.0, size=N)
        elif phase == 6:
            z = rng.standard_normal(N)
        else:
            z = make_fourier(rng, N)
        X[k] = project_box_sum(z, N)
        fit[k] = score_exact(X[k], N)
    return X, fit

        h = np.array([0.5])

        final_score = score_exact(global_best, N)
    return global_best, final_score, N
