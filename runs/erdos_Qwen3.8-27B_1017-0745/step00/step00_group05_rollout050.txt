def capped_simplex_project(v, target):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if np.any(~np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - float(np.sum(y))
    for _ in range(20):
        if abs(r) < 1e-14:
            break
        if r > 0:
            cap = 1.0 - y
            tot = float(np.sum(cap))
            if tot <= 0:
                break
            y = np.clip(y + r / tot * cap, 0.0, 1.0)
        else:
            tot = float(np.sum(y))
            if tot <= 0:
                break
            y = np.clip(y - (-r) / tot * y, 0.0, 1.0)
        r = target - float(np.sum(y))
    if abs(r) > 1e-10 or np.any(~np.isfinite(y)):
        return None
    return y

def exact_score(h, dx):
    h = np.asarray(h, dtype=np.float64)
    if h.ndim != 1 or np.any(~np.isfinite(h)):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def is_feasible(h, n, tol=1e-10):
    h = np.asarray(h, dtype=np.float64)
    if h.ndim != 1 or len(h) != n:
        return False
    if np.any(~np.isfinite(h)):
        return False
    if np.min(h) < -1e-12 or np.max(h) > 1.0 + 1e-12:
        return False
    if abs(float(np.sum(h)) - n / 2.0) > tol:
        return False
    return True

def smooth_fun(x, tau, dx, n):

    corr = np.correlate(h, 1.0 - h, mode='full')

    m = float(np.max(s))

            grad[m3:m4+1] -= w * h[m3-t:m4-t+1]
    if not np.all(np.isfinite(grad)) or not np.isfinite(val):
        return float('inf'), np.zeros(n)
    return float(val), grad
