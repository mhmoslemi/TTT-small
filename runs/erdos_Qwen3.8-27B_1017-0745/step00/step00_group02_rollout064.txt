        if len(idx) > 0:
            x[idx] += diff / len(idx)
            x[idx] = np.clip(x[idx], 0, 1)
    return x


def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx


def smooth_value(h, n, dx, tau):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    w = np.exp((c - m) / tau)
    return m + tau * math.log(w.sum())


def smooth_grad(h, n, dx, tau):
    c = np.correlate(h, 1.0 - h, mode='full') * c_max_c(
        np.ones(2 * n - 1) * 0)  # placeholder, won't use
    return _sg(h, n, dx, tau)


def _sg(h, n, dx, tau):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    w = np.exp((c - m) / tau)
    p = w / w.sum()
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        k = ki - (n - 1)
        lo = max(0, -k)
        hi = min(n, n - k)
        if hi <= lo:
            continue
        a = h[lo:hi]
        b = h[lo + k:hi + k]
        grad[lo + k:hi + k] += p[ki] * dx * (1.0 - a)
        grad[lo:hi] += p[ki] * (-dx) * b
    return grad


def is_feasible(h, n, S):
    if len(h) != n or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return False
    return abs(h.sum() - S) <= 1e-6


def pg_continuation(h0, n, S, dx, best_ref, max_iter):
    h = h0.copy()
    taus = [0.02, 0.01, 0.005, 0.002, 0.0005, 0.0002, 0.0001, 0.00005, 0.00001]
    if n > 100:
        taus = [0.02, 0.005, 0.001, 0.0001, 0.00001]
    for tau in taus:
        F = smooth_value(h, n, dx, tau)
        if not np.isfinite(F):
            break
        alpha = 1.0
        for _ in range(max_iter):
            g = _sg(h, n, dx, tau)
            if not np.all(np.isfinite(g)):
                break
            y = project_to_K(h - alpha * g, n, S)
            d = y - h
            if np.max(np.abs(d)) <= 1e-10:
                break
            gd = float(g.dot(d))
            if gd >= -1e-12:
                break
            Fy = smooth_value(y, n, dx, tau)
            if Fy <= F + 1e-4 * gd + 1e-14:
                h = y
                F = Fy
                alpha = min(1.25 * alpha, 5.0)
                sc = exact_score(h)
                if sc < best_ref[0]:
                    best_ref[0] = sc
                    best_ref[1] = h.copy()
            else:
                alpha *= 0.5
                if alpha < 1e-10:
                    break
    return h


class SLSQPProb:
    def __init__(self, n, S, dx):
        self.n = n
        self.S = S
        self.dx = dx

    def obj(self, z):
        return z[self.n]

    def obj_jac(self, z):
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        return g

    def ineq(self, z):
        h = z[:self.n]
        return z[self.n] - np.correlate(h, 1.0 - h, mode='full') * self.dx

    def ineq_jac(self, z):
        h = z[:self.n]
        n, dx = self.n, self.dx
        jac = np.zeros((2 * n - 1, n + 1))
        for ki in range(2 * n - 1):
            k = ki - (n - 1)
            lo = max(0, -k)
            hi = min(n, n - k)
            if hi <= lo:
                continue
            jac[ki, lo + k:hi + k] = dx * (1.0 - h[lo:hi])
            jac[ki, lo:hi] += -dx * h[lo + k:hi + k]
            jac[ki, n] = 1.0
        return jac

    def eq(self, z):
        return z[:self.n].sum() - self.S

    def eq_jac(self, z):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g


def slsqp_refine(h0, n, S, dx, best_ref):
    from scipy.optimize import minimize
    t0 = min(1.0, exact_score(h0) + 1e-8)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0, 1)] * (n + 1)
    pr = SLSQPProb(n, S, dx)
    cons = [
        {'type': 'ineq', 'fun': pr.ineq, 'jac': pr.ineq_jac},
        {'type': 'eq', 'fun': pr.eq, 'jac': pr.eq_jac},
    ]
    try:
        res = minimize(pr.obj, z0, jac=pr.obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 250, 'ftol': 1e-10, 'disp': False})
        hx = res.x[:n]
        if np.all(np.isfinite(hx)) and np.all(hx >= -1e-6) and np.all(hx <= 1.0 + 1e-6):
            hx = project_to_K(hx, n, S)
            if is_feasible(hx, n, S):
                sc = exact_score(hx)
                if sc < best_ref[0]:
                    best_ref[0] = sc
                    best_ref[1] = hx.copy()
    except Exception:
        pass


def run(seed=42, **kwargs):
    parent = None
    gv = globals().get('initial_h_values', None)
    if gv is not None:
        try:
            pv = np.array(gv, dtype=np.float64).ravel()
            if 2 <= pv.size <= 999 and np.all(np.isfinite(pv)) and np.max(np.abs(pv)) < 1e6:
                parent = pv
        except Exception:
            parent = None

    if parent is not None:
        n = int(parent.size)
    else:
        n = int(kwargs.get('n_points', 45))
        if not (2 <= n <= 999):
            n = 45
        parent = np.full(n, 0.5)

    S = n / 2.0
    dx = 2.0 / n
    best_h = project_to_K(parent, n, S)
    best_score = exact_score(best_h)
    best_ref = [best_score, best_h.copy()]
    rng = np.random.default_rng(seed)
    max_iter = 80 if n <= 100 else 20

    starts = [best_h.copy()]
    starts.append(project_to_K(1.0 - best_h, n, S))
    starts.append(project_to_K(best_h[::-1].copy(), n, S))
    starts.append(project_to_K(1.0 - best_h[::-1].copy(), n, S))
    if parent is not None and n <= 100:
        starts.append(project_to_K(parent + 0.08 * rng.standard_normal(n), n, S))
        starts.append(project_to_K(parent + 0.25 * rng.standard_normal(n), n, S))
    starts.append(project_to_K(rng.random(n), n, S))
    starts.append(project_to_K(rng.beta(0.5, 0.5, n), n, S))

    if n > 100:
        starts = starts[:5]

    for st in starts:
        if not is_feasible(st, n, S):
            st = project_to_K(st, n, S)
        sc = exact_score(st)
        if sc < best_ref[0]:
            best_ref[0] = sc
            best_ref[1] = st.copy()
        pg_continuation(st, n, S, dx, best_ref, max_iter)

    perts = [0.04, 0.12, 0.30, 0.50] if n <= 100 else [0.12]
    for sig in perts:
        st = project_to_K(best_ref[1] + sig * rng.standard_normal(n), n, S)
        if is_feasible(st, n, S):
            sc = exact_score(st)
            if sc < best_ref[0]:
                best_ref[0] = sc
                best_ref[1] = st.copy()
            pg_continuation(st, n, S, dx, best_ref, max_iter)

    if n <= 100:
        sl_starts = [best_ref[1].copy()]
        for st in starts:
            if len(sl_starts) >= 3:
                break
            if np.max(np.abs(st - best_ref[1])) > 1e-5:
                if exact_score(st) < best_ref[0] + 0.05:
                    sl_starts.append(st.copy())
        for ss in sl_starts:
            slsqp_refine(ss, n, S, dx, best_ref)

    final_h = project_to_K(best_ref[1], n, S)
    if not is_feasible(final_h, n, S):
        final_h = project_to_K(parent, n, S)
    fb = project_to_K(best_h, n, S)
    final_score = exact_score(final_h)
    if final_score > exact_score(fb):
        final_h = fb
        final_score = exact_score(fb)

    return (final_h.astype(np.float64).copy(), float(final_score), int(n))
