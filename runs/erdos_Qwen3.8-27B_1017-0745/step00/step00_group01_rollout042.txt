def is_feasible(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-15) or np.any(h > 1 + 1e-15):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def capped_simplex_project(z, n):
    s = n / 2.0
    if not np.all(np.isfinite(z)) or z.shape != (n,):
        return np.full(n, 0.5)
    z = np.clip(z, -1e12, 1e12)
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(z - mid, 0.0, 1.0)
        if np.sum(h) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    # residual correction
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) <= 1e-12:
            break
        if r > 0:
            slack = 1.0 - h
            idx = np.argmax(slack)
            if slack[idx] <= 0:
                break
            move = min(r, slack[idx])
            h[idx] += move
            r -= move
        else:
            mass = h.copy()
            idx = np.argmax(mass)
            if mass[idx] <= 0:
                break
            move = min(-r, mass[idx])
            h[idx] -= move
            r += move
    h = np.clip(h, 0.0, 1.0)
    if not is_feasible(h, n):
        return np.full(n, 0.5)
    return h

def exact_score(h, n):

        return np.inf

def lse_value_and_weights(h, dx, n, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = dx * corr
    c_max = np.max(c)
    a = np.exp((c - c_max) / tau)
    s = np.sum(a)
    if not np.isfinite(s) or s <= 0:
        return None, None
    F = c_max + tau * np.log(s)
    P = a / s
    return F, P

def grad_lse(h, dx, n, tau):
    val, P = lse_value_and_weights(h, dx, n, tau)

        return None

    V = np.convolve(h, P[::-1], mode='full')[n-1 : 2*n-1]
    grad = dx * (U - V)
    if not np.all(np.isfinite(grad)):
        return None
    return grad

def lse_vals(h, dx, n, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = dx * corr
    cm = np.max(c)
    a = np.exp((c - cm) / tau)
    ss = np.sum(a)
    if not np.isfinite(ss) or ss <= 0:
        return None, None
    return cm + tau * np.log(ss), a / ss

def lse_grad(h, dx, n, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = dx * corr
    cm = np.max(c)
    a = np.exp((c - cm) / tau)
    ss = np.sum(a)
    if not np.isfinite(ss) or ss <= 0:
        return None
    P = a / ss
    t_idx = np.arange(n)
    K1 = t_idx[None, :] - t_idx[:, None]
    P1 = P[K1 + n - 1]
    P2 = P[-K1 + n - 1]
    s1 = np.sum((1.0 - h[None, :]) * P1, axis=1)
    s2 = np.sum(h[None, :] * P2, axis=1)
    grad = dx * (s1 - s2)
    if not np.all(np.isfinite(grad)):
        return None
    return grad

def build_starts(n, parent, rng):
    starts = []
    uniform = np.full(n, 0.5)
    if parent is not None:
        base = capped_simplex_project(parent.copy(), n)
    else:
        base = uniform
    starts.append(base)
    starts.append(capped_simplex_project(base[::-1].copy(), n))
    starts.append(capped_simplex_project(1.0 - base[::-1].copy(), n))
    starts.append(uniform.copy())
    block = np.zeros(n)
    block[:n // 2] = 1.0
    if n % 2 == 1:
        block[n // 2] = 0.5
    starts.append(capped_simplex_project(block, n))
    alt = np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n)])
    starts.append(capped_simplex_project(alt, n))
    for sd in [0.05, 0.15, 0.35]:
        starts.append(capped_simplex_project(base + rng.normal(0, sd, n), n))
    starts.append(capped_simplex_project(rng.uniform(0, 1, n), n))
    unique = []
    for s in starts:
        dup = False
        for u in unique:
            if np.max(np.abs(s - u)) < 1e-10:
                dup = True
                break
        if not dup:
            unique.append(s)
    return unique

def run(seed=42, **kwargs):
    try:
        myseed = int(seed)
    except:
        myseed = 42
    rng = np.random.default_rng(myseed)
    n = 47
    parent = None
    try:
        iv = initial_h_values
        iv = np.array(iv, dtype=np.float64).ravel()
        if len(iv) > 0 and np.all(np.isfinite(iv)):
            n = len(iv)
            parent = iv.copy()
    except:
        pass
    if n > 999:
        h = capped_simplex_project(parent, n) if parent is not None else np.full(n, 0.5)
        sc = exact_score(h, n)
        if not np.isfinite(sc):
            h = np.full(n, 0.5)
            sc = exact_score(h, n)
        return (h, sc, n)
    if n <= 64:
        max_starts, max_iter = 10, 100
    elif n <= 256:
        max_starts, max_iter = 5, 60
    else:
        max_starts, max_iter = 2, 25
    if 'restarts' in kwargs:
        try:
            r = int(kwargs['restarts'])
            cap = 12 if n <= 64 else 6 if n <= 256 else 2
            if 0 < r < cap:
                max_starts = r
        except:
            pass
    if 'iters_per_tau' in kwargs:
        try:
            it = int(kwargs['iters_per_tau'])
            cap = 150 if n <= 64 else 90 if n <= 256 else 40
            if 0 < it < cap:
                max_iter = it
        except:
            pass
    dx = 2.0 / n
    taus = [0.02, 0.005, 0.001, 0.0002, 0.00005, 0.00001]
    all_starts = build_starts(n, parent, rng)
    starts = all_starts[:max_starts]
    if parent is not None:
        h0 = capped_simplex_project(parent.copy(), n)
    else:
        h0 = np.full(n, 0.5)
    s0 = exact_score(h0, n)
    if not np.isfinite(s0):
        h0 = np.full(n, 0.5)
        s0 = exact_score(h0, n)
    inc_h = h0.copy()
    inc_s = s0
    fb_h = h0.copy()
    fb_s = s0
    for st in starts:
        sc = exact_score(st, n)
        if np.isfinite(sc) and sc < inc_s - 1e-12:
            inc_h = st.copy()
            inc_s = sc
    for st in starts:
        x = st.copy()
        try:
            for tau in taus:
                Fx, _ = lse_vals(x, dx, n, tau)
                if Fx is None:
                    continue
                g = lse_grad(x, dx, n, tau)
                if g is None:
                    continue
                alpha = 1.0
                small_cnt = 0
                for _ in range(max_iter):
                    y = capped_simplex_project(x - alpha * g, n)
                    diff = y - x
                    D = np.sum(diff * diff)
                    if np.max(np.abs(diff)) <= 1e-9 or D < 1e-30:
                        break
                    Fy, _ = lse_vals(y, dx, n, tau)
                    if Fy is None:
                        alpha *= 0.5
                        if alpha < 1e-12:
                            break
                        continue
                    thresh = Fx - 1e-4 * D / alpha + 1e-14 * max(1.0, abs(Fx))
                    if Fy <= thresh:
                        imp = Fx - Fy
                        if imp <= 1e-12 * max(1.0, abs(Fx)):
                            small_cnt += 1
                        else:
                            small_cnt = 0
                        if small_cnt >= 5:
                            break
                        x = y
                        Fx = Fy
                        g = lse_grad(x, dx, n, tau)
                        if g is None:
                            break
                        alpha = min(2.0, 1.25 * alpha)
                        sc = exact_score(x, n)
                        if np.isfinite(sc) and sc < inc_s - 1e-12:
                            inc_h = x.copy()
                            inc_s = sc
                    else:
                        alpha *= 0.5
                        if alpha < 1e-12:
                            break
        except:
            pass
    inc_h = capped_simplex_project(inc_h, n)
    inc_s = exact_score(inc_h, n)
    fb_s = exact_score(fb_h, n)
    if not np.isfinite(inc_s):
        inc_h = fb_h.copy()
        inc_s = fb_s
    if not np.isfinite(inc_s):
        inc_h = np.full(n, 0.5)
        inc_s = exact_score(inc_h, n)
    return (inc_h, inc_s, n)
