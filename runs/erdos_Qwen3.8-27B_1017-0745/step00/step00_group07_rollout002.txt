    if parent is not None:
        h_fallback = project_capped_simplex(parent, S)
    else:
        h_fallback = np.full(n, 0.5)
    c_fallback = evaluate_c5(h_fallback)
    starts = [h_fallback]
    if parent is not None:
        for sd in [0.04, 0.15, 0.35]:
            starts.append(project_capped_simplex(parent + sd * rng.standard_normal(n), S))
        q = int(rng.integers(1, max(2, n)))
        phi = float(rng.random() * 2.0 * math.pi)
        arr = np.arange(n, dtype=np.float64)
        starts.append(project_capped_simplex(0.5 + 0.45 * np.cos(2.0 * math.pi * q * arr / n + phi), S))
    else:
        for _ in range(2):
            q = int(rng.integers(1, max(2, n)))
            phi = float(rng.random() * 2.0 * math.pi)
            arr = np.arange(n, dtype=np.float64)
            starts.append(project_capped_simplex(0.5 + 0.45 * np.cos(2.0 * math.pi * q * arr / n + phi), S))
    starts.append(project_capped_simplex(rng.random(n), S))
    perm = rng.permutation(n)
    bb = np.zeros(n)
    bb[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        bb[perm[n // 2]] = 0.5
    starts.append(bb)
    dedup = [starts[0]]
    for s in starts[1:]:
        dup = False
        for d in dedup:
            if np.max(np.abs(s - d)) < 1e-10:
                dup = True
                break
        if not dup:
            dedup.append(s)
    starts = dedup[:7] if n <= 256 else dedup[:3]
    best_h = h_fallback.copy()
    best_c5 = c_fallback
    baseline_c5 = c_fallback
    for s in starts:
        c = evaluate_c5(s)
        if c < best_c5:
            best_c5 = c
            best_h = s.copy()
        if c < baseline_c5:
            baseline_c5 = c
    B = min(480, max(12, int(4000000 / (n * n))))
    for s in starts:
        h_opt = optimize_start(s, n, dx, S, B)
        c_opt = evaluate_c5(h_opt)
        if c_opt < best_c5:
            best_c5 = c_opt
            best_h = h_opt.copy()
    if best_c5 >= baseline_c5 - 1e-8 and n <= 256:
        for sd in [0.6, 1.0]:
            s2 = project_capped_simplex(best_h + sd * rng.standard_normal(n), S)
            h_opt = optimize_start(s2, n, dx, S, max(6, B // 2))
            c_opt = evaluate_c5(h_opt)
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()
    P = max(80, min(800, int(10000000 / (n * n))))
    pt_h, pt_c = pair_transfer_polish(best_h, n, S, P, rng)
    if pt_c < best_c5:
        best_c5 = pt_c
        best_h = pt_h
    best_h = project_capped_simplex(best_h, S)
    final_c5 = evaluate_c5(best_h)
    if final_c5 > c_fallback:
        best_h = h_fallback.copy()
        final_c5 = c_fallback
    return best_h, final_c5, n
