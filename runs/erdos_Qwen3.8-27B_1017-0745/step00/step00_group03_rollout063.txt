import numpy as np

def _project_box_sum(z, N):
    uniform = np.full(N, 0.5)
    if z.shape != (N,) or not np.all(np.isfinite(z)):
        return uniform
    m = N * 0.5
    lo = float(z.min()) - 1.0
    hi = float(z.max())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - hi, 0.0, 1.0).copy()
    e = m - h.sum()
    if e > 1e-12:
        for i in range(N):
            if h[i] < 1.0:
                adj = min(e, 1.0 - h[i])
                h[i] += adj
                e -= adj
                if e <= 0:
                    break
    elif e < -1e-12:
        for i in range(N):
            if h[i] > 0.0:
                adj = min(-e, h[i])
                h[i] -= adj
                e += adj
                if e >= 0:
                    break
    if (not np.all(np.isfinite(h)) or h.min() < -1e-12 or h.max() > 1.0 + 1e-12
            or abs(h.sum() - m) > 1e-10 * max(1, N)):
        return uniform
    return np.clip(h, 0.0, 1.0)

def _lag_values(h):
    N = len(h)
    M = 2 * N - 1
    q = np.zeros(M)
    for k in range(-(N - 1), N):
        if k >= 0:
            L = N - k
            q[k + N - 1] = np.dot(h[k:k + L], 1.0 - h[0:L])
        else:
            d = -k
            L = N - d
            q[k + N - 1] = np.dot(h[0:L], 1.0 - h[d:d + L])
    return q

def _score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / N))

def _smooth_val_grad(h, tau, N):
    dx = 2.0 / N
    q = _lag_values(h)
    r = dx * q
    rmax = float(np.max(r))
    exps = np.exp((r - rmax) / tau)
    s = float(np.sum(exps))
    p = exps / s
    phi = rmax + tau * np.log(s)
    g = np.zeros(N)
    for k in range(-(N - 1), N):
        w = dx * p[k + N - 1]
        if w < 1e-18:
            continue
        if k >= 0:
            L = N - k
            a = h[k:k + L]
            b = h[0:L]
            g[k:k + L] += w * (1.0 - b)
            g[0:L] -= w * a
        else:
            d = -k
            L = N - d
            a = h[0:L]
            b = h[d:d + L]
            g[0:L] += w * (1.0 - b)
            g[d:d + L] -= w * a
    if not np.all(np.isfinite(g)):
        return np.inf, np.zeros(N)
    return phi, g

def _smooth_val(h, tau, N):
    dx = 2.0 / N
    q = _lag_values(h)
    r = dx * q
    rmax = float(np.max(r))
    s = float(np.sum(np.exp((r - rmax) / tau)))
    return rmax + tau * np.log(s)

def _optimize(h0, N, inc_h, inc_score, iters):
    temps = [0.08, 0.02, 0.005, 0.001, 0.0002, 0.00004, 8e-6]
    h = h0.copy()
    for tau in temps:
        sb = h.copy()
        sb_score = _score_exact(sb, N)
        for _it in range(iters):
            phi_h, g = _smooth_val_grad(h, tau, N)
            if not np.isfinite(phi_h):
                break
            alpha = 1.0
            stp_ok = False
            last_disp = 1.0
            for _bs in range(24):
                y = _project_box_sum(h - alpha * g, N)
                d = y - h
                disp = float(np.max(np.abs(d)))
                if disp < 1e-10:
                    break
                gd = float(np.dot(g, d))
                phi_y = _smooth_val(y, tau, N)
                sc_y = _score_exact(y, N)
                if np.isfinite(sc_y):
                    if sc_y < inc_score:
                        inc_score = sc_y
                        inc_h[:] = y
                    if sc_y < sb_score:
                        sb_score = sc_y
                        sb[:] = y
                if gd < 0 and phi_y <= phi_h + 1e-4 * gd + 1e-14:
                    h[:] = y
                    last_disp = disp
                    stp_ok = True
                    break
                alpha *= 0.5
                if alpha < 1e-8:
                    break
            if stp_ok and last_disp < 1e-8:
                break
        h = sb.copy()
    return inc_h, inc_score
