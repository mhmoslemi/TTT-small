import numpy as np
import math

def is_feasible(h, n, S):
    try:
        if h.shape != (n,):
            return False
        if not np.all(np.isfinite(h)):
            return False
        if np.any(h < -1e-15) or np.any(h > 1.0 + 1e-15):
            return False
        s1 = float(np.sum(h))
        s2 = math.fsum(float(x) for x in h)
        return abs(s1 - S) <= 1e-10 and abs(s2 - S) <= 1e-10
    except Exception:
        return False

def project_capped_simplex(v, n, S):
    try:
        v = np.clip(np.asarray(v, dtype=np.float64).copy(), -1e6, 1e6)
        if v.shape != (n,) or not np.all(np.isfinite(v)):
            return None
        lo = float(v.min()) - 1.0
        hi = float(v.max())
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
                lo = mid
            else:
                hi = mid
        h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
        for _ in range(80):
            res = S - math.fsum(h.tolist())
            if abs(res) <= 1e-12:
                break
            if res > 0:
                for i in range(n):
                    t = min(1.0 - h[i], res)
                    if t > 1e-15:
                        h[i] += t
                        res -= t
                        if abs(res) <= 1e-12:
                            break
            else:
                for i in range(n):
                    t = min(h[i], -res)
                    if t > 1e-15:
                        h[i] -= t
                        res += t
                        if abs(res) <= 1e-12:
                            break
        if not is_feasible(h, n, S):
            for j in range(n):
                rest = math.fsum(h.tolist()) - h[j]
                val = S - rest
                if 0.0 <= val <= 1.0:
                    h[j] = val
                    if is_feasible(h, n, S):
                        break
        return h if is_feasible(h, n, S) else None
    except Exception:
        return None

def exact_score(h, n):
    try:
        dx = 2.0 / n
        return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))
    except Exception:
        return float('inf')

def try_add(pop_list, c, n, S):
    try:
        c = np.asarray(c, dtype=np.float64)
        if c.shape != (n,) or not np.all(np.isfinite(c)):
            return
        if not is_feasible(c, n, S):
            return
        for p in pop_list:
            if float(np.max(np.abs(c - p))) < 1e-10:
                return
        pop_list.append(c.copy())
    except Exception:
        pass

def make_pattern(idx, n):
    if idx == 0:
        return np.full(n, 0.5)
    elif idx == 1:
        v = np.zeros(n)
        v[:n // 2] = 1.0
        if n % 2 == 1:
            v[n // 2] = 0.5
        return v
    elif idx == 2:
        return np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    else:
        return np.clip(0.5 + 0.45 * np.sin(2.0 * np.pi * np.arange(n) / n), 0.0, 1.0)

def build_population(n, S, parent_seed, P, rng):
    pop_list = []
    if parent_seed is not None:
        try_add(pop_list, parent_seed, n, S)
        try_add(pop_list, parent_seed[::-1].copy(), n, S)
        try_add(pop_list, 1.0 - parent_seed.copy(), n, S)
        try_add(pop_list, 1.0 - parent_seed[::-1].copy(), n, S)
    for idx in range(4):
        try_add(pop_list, make_pattern(idx, n), n, S)
    if parent_seed is not None:
        sigmas = [0.04, 0.10, 0.22, 0.40]
        for g in range(min(12, P // 2)):
            z = rng.standard_normal(n)
            z -= z.mean()
            proj = project_capped_simplex(parent_seed + sigmas[g % 4] * z, n, S)
            if proj is not None:
                try_add(pop_list, proj, n, S)
    attempts = 0
    max_att = 20 * P
    while len(pop_list) < P and attempts < max_att:
        attempts += 1
        if parent_seed is not None and attempts % 3 == 0:
            z = rng.standard_normal(n)
            z -= z.mean()
            sig = [0.05, 0.10, 0.20][attempts % 3]
            proj = project_capped_simplex(parent_seed + sig * z, n, S)
        else:
            proj = project_capped_simplex(rng.random(n), n, S)
        if proj is not None:
            try_add(pop_list, proj, n, S)
    while len(pop_list) < P:
        pop_list.append(pop_list[0].copy())
    return np.array(pop_list[:P], dtype=np.float64)

def de_step(X, fit, F_arr, CR_arr, P, n, S, rng):
    X_new = X.copy()
    fit_new = fit.copy()
    F_new = F_arr.copy()
    CR_new = CR_arr.copy()
    order = np.argsort(fit)
    pbest_n = max(1, int(math.ceil(0.2 * P)))
    pbest_pool = order[:pbest_n]
    for i in range(P):
        F_t = F_arr[i]
        CR_t = CR_arr[i]
        if rng.random() < 0.1:
            F_t = 0.1 + 0.8 * rng.random()
        if rng.random() < 0.1:
            CR_t = rng.random()
        if rng.random() < 0.75:
            cands = [x for x in range(P) if x != i]
            r1, r2, r3 = int(rng.choice(cands, size=3, replace=False)), int(rng.choice(cands, size=3, replace=False)), 0
            idxs = rng.choice(cands, size=3, replace=False)
            r1, r2, r3 = int(idxs[0]), int(idxs[1]), int(idxs[2])
            mutant = X[r1] + F_t * (X[r2] - X[r3])
        else:
            pool = [int(x) for x in pbest_pool if x != i]
            if len(pool) == 0:
                pool = [int(x) for x in pbest_pool]
            p = pool[int(rng.integers(len(pool)))]
            cands2 = [x for x in range(P) if x != i and x != p]
            idxs2 = rng.choice(cands2, size=2, replace=False)
            r1, r2 = int(idxs2[0]), int(idxs2[1])
            mutant = X[i] + F_t * (X[p] - X[i]) + F_t * (X[r1] - X[r2])
        mask = rng.random(n) < CR_t
        trial = np.where(mask, mutant, X[i])
        fc = int(rng.integers(n))
        trial[fc] = mutant[fc]
        proj = project_capped_simplex(trial, n, S)
        if proj is None or not is_feasible(proj, n, S):
            continue
        if float(np.max(np.abs(proj - X[i]))) < 1e-14:
            continue
        sc = exact_score(proj, n)
        if not np.isfinite(sc):
            continue
        if sc < fit_new[i] - 1e-12:
            X_new[i] = proj
            fit_new[i] = sc
            F_new[i] = F_t
            CR_new[i] = CR_t
        elif abs(sc - fit_new[i]) <= 1e-12 and float(np.max(np.abs(proj - X[i]))) > 1e-6 and rng.random() < 0.02:
            X_new[i] = proj
            fit_new[i] = sc
            F_new[i] = F_t
            CR_new[i] = CR_t
    return X_new, fit_new, F_new, CR_new
