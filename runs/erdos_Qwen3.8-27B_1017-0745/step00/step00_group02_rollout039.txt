def _score(h, n, dx):
    if not np.all(np.isfinite(h)) or h.shape != (n,):
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    if not np.all(np.isfinite(q)):
        return float('inf')
    return float(np.max(q))

def _feasible(h, n, S):

            np.all(h >= 0) and np.all(h <= 1) and
            abs(float(np.sum(h)) - S) <= 1e-10)

def _project(y, n, S):

            break
    return h, sc

def _gen_pairs(n, rng):
    if n <= 64:
        all_p = [(i, j) for i in range(n) for j in range(i + 1, n)]
        idx = list(range(len(all_p)))
        rng.shuffle(idx)
        return [all_p[k] for k in idx]
    target = min(400, 4 * n) if n <= 256 else 64
    seen = set()
    out = []
    draws = 0
    while len(out) < target and draws < 10 * target:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n - 1))
        if j >= i:
            j += 1
        key = (i, j)
        if key not in seen:
            seen.add(key)
            out.append(key)
        draws += 1
    return out

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = globals().get('initial_h_values', None)
    S = None
    dx = None
    n = 45
    p = None
    if parent is not None:
        try:
            par = np.array(parent, dtype=np.float64).ravel()
            if par.ndim == 1 and 2 <= len(par) < 1000 and np.all(np.isfinite(par)) and np.max(np.abs(par)) <= 1e6:
                n = len(par)
                S = n / 2.0
                dx = 2.0 / n
                if np.all(par >= 0) and np.all(par <= 1) and abs(float(np.sum(par)) - S) <= 1e-10:
                    p = par.copy()
                else:
                    p = _project(par, n, S)
                if p is None:
                    p = 0.5 * np.ones(n)
        except Exception:
            n = 45
            S = n / 2.0
            dx = 2.0 / n
            p = 0.5 * np.ones(n)
    if S is None:
        S = n / 2.0
        dx = 2.0 / n
        p = 0.5 * np.ones(n)
    fallback_h = p.copy()
    fallback_sc = _score(fallback_h, n, dx)
    best_h = fallback_h.copy()
    best_sc = fallback_sc
    starts = []
    if p is not None and np.std(p) > 1e-15:
        starts.append(p.copy())
        starts.append(p[::-1].copy())
        starts.append((1.0 - p).copy())
        starts.append((1.0 - p)[::-1].copy())
        z1 = rng.standard_normal(n) * 0.08
        pr = _project(p + z1, n, S)
        if pr is not None:
            starts.append(pr)
        z2 = rng.standard_normal(n) * 0.25
        pr = _project(p + z2, n, S)
        if pr is not None:
            starts.append(pr)
    u = rng.uniform(0, 1, n)
    pr = _project(u, n, S)
    if pr is not None:
        starts.append(pr)
    bv = rng.beta(0.5, 0.5, n)
    pr = _project(bv, n, S)
    if pr is not None:
        starts.append(pr)
    if not starts:
        starts.append(0.5 * np.ones(n))
    starts = starts[:8]
    deduped = [starts[0]]
    for s in starts[1:]:
        if all(np.max(np.abs(s - d)) > 1e-6 for d in deduped):
            deduped.append(s)
    starts = deduped
    stages = [(0.20, 0.50), (0.08, 0.25), (0.03, 0.12), (0.01, 0.06)]
    if n <= 64:
        n_chain = len(starts)
        n_prop = 180
        n_extra = 2
    elif n <= 256:
        n_chain = min(6, len(starts))
        n_prop = 40
        n_extra = 1
    else:
        n_chain = min(4, len(starts))
        n_prop = 12
        n_extra = 1
    chain_bests = []
    for ci in range(n_chain):
        s_start = starts[ci]
        s_sc = _score(s_start, n, dx)
        if s_sc < best_sc - 1e-12:
            best_sc = s_sc
            best_h = s_start.copy()
        try:
            bh, bsc = _metropolis_chain(s_start, n, dx, S, rng, stages, n_prop)
            if bsc < best_sc - 1e-12:
                best_sc = bsc
                best_h = bh.copy()
            chain_bests.append((bh, bsc))
        except Exception:
            pass
    for _ in range(n_extra):
        sig = 0.08 if n <= 64 else 0.15
        z = rng.standard_normal(n) * sig
        s_start = _project(best_h + z, n, S)
        if s_start is None:
            continue
        s_sc = _score(s_start, n, dx)
        if s_sc < best_sc - 1e-12:
            best_sc = s_sc
            best_h = s_start.copy()
        try:
            bh, bsc = _metropolis_chain(s_start, n, dx, S, rng, stages, n_prop)
            if bsc < best_sc - 1e-12:
                best_sc = bsc
                best_h = bh.copy()
            chain_bests.append((bh, bsc))
        except Exception:
            pass
    states = [(fallback_h, fallback_sc)] + chain_bests
    dedup_states = [states[0]]
    for st in states[1:]:
        if all(np.max(np.abs(st[0] - ds[0])) > 1e-5 for ds in dedup_states):
            dedup_states.append(st)
    dedup_states.sort(key=lambda x: x[1])
    if n <= 64:
        n_refine = min(3, len(dedup_states))
    elif n <= 256:
        n_refine = min(2, len(dedup_states))
    else:
        n_refine = 1
    for ri in range(n_refine):
        try:
            rh, rsc = _local_refine(dedup_states[ri][0], n, dx, S, rng, 2)
            if rsc < best_sc - 1e-12:
                best_sc = rsc
                best_h = rh.copy()
        except Exception:
            pass
    final_h = _small_repair(best_h.copy(), n, S)
    if final_h is None:
        final_h = _small_repair(fallback_h.copy(), n, S)
    if final_h is None:
        final_h = 0.5 * np.ones(n)
    final_sc = _score(final_h, n, dx)
    if not _feasible(final_h, n, S):
        final_h = 0.5 * np.ones(n)
        final_sc = _score(final_h, n, dx)
    if not _feasible(final_h, n, S):
        final_h = 0.5 * np.ones(45)
        final_sc = _score(final_h, 45, 2.0 / 45)
        n = 45
    return (final_h.astype(np.float64).copy(), float(final_sc), int(n))
