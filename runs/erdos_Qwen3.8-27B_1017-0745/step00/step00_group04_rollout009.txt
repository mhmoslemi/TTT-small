def compute_lags(h, dx):
    n = len(h)
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    q[0] = dx * np.dot(h, 1.0 - h)
    G[0] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        ip = 2 * d - 1
        im = 2 * d
        q[ip] = dx * np.dot(h[d:], 1.0 - h[:m])
        G[ip, d:] = dx * (1.0 - h[:m])
        G[ip, :m] = -dx * h[d:]
        q[im] = dx * np.dot(h[:m], 1.0 - h[d:])
        G[im, :m] = dx * (1.0 - h[d:])
        G[im, d:] = -dx * h[:m]
    return q, G

    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

class SoftObj:
    def __init__(self, tau, dx):
        self.tau = tau
        self.dx = dx

    def __call__(self, h):
        q, G = compute_lags(h, self.dx)
        qmax = np.max(q)
        exps = np.exp((q - qmax) / self.tau)
        total = np.sum(exps)
        p = exps / total
        return qmax + self.tau * math.log(total), p @ G

        return np.sum(x) - self.n / 2.0

        g[-1] = 1.0
        return g

        return g

    def ineq_fun(self, z):
