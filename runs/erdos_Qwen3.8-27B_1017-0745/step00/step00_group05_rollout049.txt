def lag_gradient(h, k, n):
    t = int(k) - (n - 1)
    g = np.zeros(n, dtype=float)
    if t >= 0:
       if t < n:
          g[t:] += 1.0 - h[:n-t]
          g[:n-t] -= h[t:]
    else:
       s = -t
       if s < n:
          g[:n-s] += 1.0 - h[s:]
          g[s:] -= h[:n-s]
    return g

import math
import numpy as np

def exact_score(h, dx):
    h = np.asarray(h, dtype=float)
    if h.ndim != 1 or h.size == 0:
        return float('inf')
    if not np.all(np.isfinite(h)):
        return float('inf')
    if np.min(h) < -1e-7 or np.max(h) > 1.0 + 1e-7:
        return float('inf')
    v = np.correlate(h, 1.0 - h, mode='full') * float(dx)
    if v.size == 0:
        return float('inf')
    s = float(np.max(v))
    return s if np.isfinite(s) else float('inf')

def capped_simplex_project(v, target):
    nv = np.asarray(v, dtype=float).ravel()
    if nv.size == 0 or not np.all(np.isfinite(nv)):
        return None
    if float(np.max(np.abs(nv))) > 1e300 or target < -1e-12 or target > nv.size + 1e-12:
        return None
    low = float(np.min(nv)) - 1.0
    high = float(np.max(nv)) + 1.0
    for _ in range(80):
        mid = 0.5 * (low + high)
        y = np.clip(nv - mid, 0.0, 1.0)
        s = float(np.sum(y))
        if s > target:
            low = mid
        else:
            high = mid
    y = np.clip(nv - 0.5 * (low + high), 0.0, 1.0)
    for _ in range(6):
        r = float(target - np.sum(y))
        if abs(r) <= 1e-13:
            break
        if r > 0.0:
            for i in range(y.size):
                if r <= 0.0:
                    break
                cap = 1.0 - y[i]
                if cap <= 0.0:
                    continue
                add = r if r < cap else cap
                y[i] += add
                r -= add
        else:
            for i in range(y.size):
                if r >= 0.0:
                    break
                sub = -r if -r < y[i] else y[i]
                if sub <= 0.0:
                    continue
                y[i] -= sub
                r += sub
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or np.min(y) < 0.0 or np.max(y) > 1.0:
        return None
    return y if abs(float(np.sum(y)) - target) <= 1e-10 else None

def normalize_direction(d, n):
    nd = np.asarray(d, dtype=float).ravel()
    if nd.size != n or not np.all(np.isfinite(nd)):
        return None
    if float(np.linalg.norm(nd)) < 1e-14:
        return None
    nd = nd - float(np.mean(nd))
    nd[-1] -= float(np.sum(nd))
    m = float(np.max(np.abs(nd)))
    if not np.isfinite(m) or m < 1e-14:
        return None
    nd = nd / m
    nd[-1] -= float(np.sum(nd))
    if not np.all(np.isfinite(nd)):
        return None
    return nd

def lag_gradient(h, k, n):
    t = int(k) - (n - 1)
    g = np.zeros(n, dtype=float)
    if t >= 0:
        if t < n:
            g[t:] += 1.0 - h[:n - t]
            g[:n - t] -= h[t:]
    else:
        s = -t
        if s < n:
            g[:n - s] += 1.0 - h[s:]
            g[s:] -= h[:n - s]
    return g

def build_directions(h, rng, n):

    q = np.correlate(h, 1.0 - h, 'full')
    top = np.argsort(q)[-3:]
    sg = np.zeros(n, dtype=float)

def add_root(cand, x, lo, hi, tol):
    if not math.isfinite(x):
        return
    if x < lo - tol or x > hi + tol:
        return
    if x < lo:
        x = lo
    elif x > hi:
        x = hi
    cand.append(float(x))

def add_pair_roots(cand, A, B, C, lo, hi, tol):
    am = abs(A)
    if abs(B) > am:
        am = abs(B)
    if abs(C) > am:
        am = abs(C)
    if am <= 0.0:
        return
    scale = am if am > 1.0 else 1.0
    ct = 1e-13 * scale
    if am < ct:
        return
    if abs(A) <= ct:
        if abs(B) > ct:
            add_root(cand, -C / B, lo, hi, tol)
        return
    D = B * B - 4.0 * A * C
    dt = 1e-12 * max(1.0, abs(B * B), abs(4.0 * A * C))
    if D < -dt:
        return
    if D < 0.0:
        D = 0.0
    sd = math.sqrt(D)
    if B >= 0.0:
        q = -0.5 * (B + sd)
    else:
        q = -0.5 * (B - sd)
    if abs(q) > 1e-300:
        x1 = q / A
        x2 = C / q
    else:
        x1 = -B / (2.0 * A)
        x2 = x1
    add_root(cand, x1, lo, hi, tol)
    if x2 != x1:
        add_root(cand, x2, lo, hi, tol)

        elif abs(x - uniq[-1]) > 1e-12 * max(1.0, abs(x)):
            uniq.append(x)
    return uniq

def line_search(h, d, dx, n):
    target = 0.5 * float(n)
    cur = exact_score(h, dx)
    if not np.isfinite(cur):
        return h, cur
    d = normalize_direction(d, n)
    if d is None:
        return h, cur
    lo = 0.0
    hi = 0.0
    for i in range(n):
        di = float(d[i])
        if abs(di) <= 1e-15:
            continue
        hi1 = 1.0 - float(h[i])
        if di > 0.0:
            low_i = -float(h[i]) / di
            up_i = hi1 / di
        else:
            low_i = hi1 / di
            up_i = -float(h[i]) / di
        if low_i > lo:
            lo = low_i
        if up_i < hi:
            hi = up_i
    if not (math.isfinite(lo) and math.isfinite(hi)):
        return h, cur
    if lo < 0.0 < lo + 1e-12:
        lo = 0.0
    if hi > 0.0 > hi - 1e-12:
        hi = 0.0
    if lo > hi + 1e-12:
        return h, cur
    if lo > hi:
        lo = hi = 0.5 * (lo + hi)
    if hi - lo < 1e-14:
        return h, cur
    a = np.correlate(h, 1.0 - h, 'full')
    b = np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')
    c = -np.correlate(d, d, 'full')
    if not (np.all(np.isfinite(a)) and np.all(np.isfinite(b)) and np.all(np.isfinite(c))):
        return h, cur
    al = collect_line_candidates(lo, hi, a, b, c, n <= 80)
    L = len(al)
    if L == 0:
        return h, cur
    env = np.empty(L, dtype=float)
    for s in range(0, L, 256):
        e = min(L, s + 256)
        ab = np.asarray(al[s:e], dtype=float)
        ab2 = ab * ab
        vals = a[None, :] + b[None, :] * ab[:, None] + c[None, :] * ab2[:, None]
        env[s:e] = np.max(vals, axis=1)
    order = np.argsort(env, kind='stable')
    evals = []
    for j in range(min(8, L)):
        evals.append(int(order[j]))
    zi = -1
    for i in range(L):
        if abs(al[i]) <= 1e-14:
            zi = i
            break
    if zi >= 0:
        evals.append(zi)
    evals.append(0)
    evals.append(L - 1)
    seen = set()
    finals = []
    for i in evals:
        if 0 <= i < L and i not in seen:
            seen.add(i)
            finals.append(al[i])
    best_h = h
    best_s = cur
    for alpha in finals:
        if not math.isfinite(float(alpha)):
            continue
        y = capped_simplex_project(h + float(alpha) * d, target)
        if y is None:
            continue
        sc = exact_score(y, dx)
        if sc < best_s - 1e-12:
            best_s = sc
            best_h = y
    return best_h, best_s

def local_search(start, dx, rng, n, sweeps):
    target = 0.5 * float(n)
    h = np.asarray(start, dtype=float).copy()
    if h.size != n:
        return None
    h = capped_simplex_project(h, target)
    if h is None:
        return None
    cur = exact_score(h, dx)
    if not np.isfinite(cur):
        return None
    best_h = h.copy()
    best_s = cur
    for _ in range(sweeps):
        if best_s <= 0.3808:
            break
        dirs = build_directions(h, rng, n)
        if not dirs:
            break
        improved = False
        for d in dirs:
            res = line_search(h, d, dx, n)
            if res is None:
                continue
            nh, ns = res
            if np.isfinite(ns) and ns < best_s - 1e-12:
                nh = np.asarray(nh, dtype=float).copy()
                best_h = nh
                best_s = float(ns)
                h = nh
                improved = True
        if not improved:
            break
    return best_h, float(best_s)

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    n = 70
    parent_arr = None
    p = globals().get('initial_h_values', None)
    if p is not None:
        arr = np.asarray(p, dtype=float).ravel()
        if 2 <= arr.size <= 999:
            n = int(arr.size)
            target0 = 0.5 * float(n)
            if np.all(np.isfinite(arr)) and float(np.max(np.abs(arr))) <= 1e6:
                pr = capped_simplex_project(arr, target0)
                if pr is not None:
                    parent_arr = pr
    target = 0.5 * float(n)
    dx = 2.0 / float(n)
    u = np.full(n, 0.5, dtype=float)
    u = capped_simplex_project(u, target)
    if u is None:
        u = np.full(n, 0.5, dtype=float)
    block = np.zeros(n, dtype=float)
    block[:n // 2] = 1.0
    if n % 2 == 1:
        block[n // 2] = 0.5
    block = capped_simplex_project(block, target)
    if block is None:
        block = u.copy()
    safe_h = u.copy()
    safe_s = exact_score(safe_h, dx)
    starts = []
    if parent_arr is not None:
        starts.append(parent_arr)
        starts.append(parent_arr[::-1].copy())
        z = rng.standard_normal(n)
        starts.append(parent_arr + 0.04 * z)
        z = rng.standard_normal(n)
        starts.append(parent_arr + 0.12 * z)
    starts.append(u.copy())
    starts.append(block.copy())
    extra = 2 if parent_arr is None else 1
    for _ in range(extra):
        starts.append(rng.uniform(0.0, 1.0, size=n))
        starts.append(rng.beta(0.5, 0.5, size=n))
    raw = []
    for cand in starts:
        if cand is None:
            continue
        y = capped_simplex_project(cand, target)
        if y is not None:
            raw.append(y)
    dedup = []
    for y in raw:
        same = False
        for f in dedup:
            if np.max(np.abs(y - f)) < 1e-10:
                same = True
                break
        if not same:
            dedup.append(y)
    scored = []
    for y in dedup:
        sc = exact_score(y, dx)
        if np.isfinite(sc):
            scored.append((sc, y))
            if sc < safe_s - 1e-12:
                safe_h = y.copy()
                safe_s = float(sc)
    if not scored:
        scored = [(safe_s, safe_h.copy())]
    ordered = []
    remaining = list(scored)
    while remaining:
        mi = 0
        mv = remaining[0][0]
        for i in range(1, len(remaining)):
            if remaining[i][0] < mv:
                mi = i
                mv = remaining[i][0]
        ordered.append(remaining.pop(mi))
    psc = None
    if parent_arr is not None:
        for sc, y in scored:
            if np.max(np.abs(y - parent_arr)) < 1e-10:
                psc = (sc, y)
                break
    search = []
    if psc is not None:
        search.append(psc)
    for item in ordered:
        if len(search) >= 3:
            break
        y = item[1]
        same = False
        for _, y2 in search:
            if np.max(np.abs(y - y2)) < 1e-10:
                same = True
                break
        if not same:
            search.append(item)
    if not search:
        search = [(safe_s, safe_h.copy())]
    for idx, item in enumerate(search):
        if safe_s <= 0.3808:
            break
        sw = 2 if (n <= 80 and idx == 0) else 1
        res = local_search(item[1], dx, rng, n, sw)
        if res is not None:
            nh, ns = res
            if np.isfinite(ns) and ns < safe_s - 1e-12:
                safe_h = nh.copy()
                safe_s = float(ns)
    restart_scales = [0.02, 0.08]
    cnt = 2 if n <= 80 else 1
    for i in range(min(cnt, len(restart_scales))):
        if safe_s <= 0.3808:
            break
        z = rng.standard_normal(n)
        y = capped_simplex_project(safe_h + restart_scales[i] * z, target)
        if y is None:
            continue
        res = local_search(y, dx, rng, n, 1)
        if res is not None and np.isfinite(res[1]) and res[1] < safe_s - 1e-12:
            safe_h = res[0].copy()
            safe_s = float(res[1])
    final = capped_simplex_project(safe_h, target)
    if final is None:
        final = u.copy()
    sc = exact_score(final, dx)
    if not np.isfinite(sc):
        final = u.copy()
        sc = exact_score(final, dx)
    return (final, float(sc), int(n))
