from math import copysign, sqrt, floor

def capped_simplex_project(v, target):

    if n == 0 or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(4):
        r = target - float(np.sum(y))
        if abs(r) < 1e-14:
            break
        if r > 0:
            add = np.minimum(r, 1.0 - y)
            y = np.clip(y + add, 0.0, 1.0)
        else:

def normalize_direction(d):

def feasible_interval(h, d):

            lo = max(lo, -h[i] / d[i])

        hi = max(hi, 0.0)
    lo = min(lo, 0.0)
    hi = max(hi, 0.0)
    if hi - lo < -tol:

def quad_roots(A, B, C, lo, hi):
    roots = []
    s = max(1.0, abs(A), abs(B), abs(C))
    if abs(C) > 1e-14 * s:
        disc = B * B - 4.0 * C * A
        if disc < 0.0:
            if disc > -1e-10 * s:
                disc = 0.0
            else:
                return roots
        sq = sqrt(disc)
        q = -0.5 * (B + copysign(sq, B))
        if abs(q) > 1e-30:
            roots.append(q / C)
            if abs(A) > 1e-30:
                roots.append(A / q)
        else:
            roots.append(-B / (2.0 * C))
    elif abs(B) > 1e-14 * s:
        roots.append(-A / B)
    else:
        return roots
    out = []
    pad = 1e-12 * max(1.0, abs(lo), abs(hi))
    for r in roots:
        if np.isfinite(r) and r >= lo - pad and r <= hi + pad:
            out.append(max(lo, min(hi, r)))
    return out

def line_search_small(h, d, dx, a, b, c, alpha_lo, alpha_hi):

    cands = [alpha_lo, alpha_hi, 0.0]

            sp = -b[i] / (2.0 * c[i])
            if np.isfinite(sp) and alpha_lo <= sp <= alpha_hi:

    for i in range(m):
        for j in range(i + 1, m):
            A = a[i] - a[j]
            B = b[i] - b[j]
            C = c[i] - c[j]
            cands.extend(quad_roots(A, B, C, alpha_lo, alpha_hi))
    cands = sorted(set(round(x, 12) for x in cands))
    best_a, best_s = 0.0, float('inf')
    batch = 512
    for sl in range(0, len(cands), batch):
        alpha = np.array(cands[sl:sl+batch])
        vals = a + b * alpha[:, None] + c * alpha[:, None] ** 2
        env = np.max(vals, axis=1)
        best_local = int(np.argmin(env))
        if env[best_local] < best_s:
            best_s = float(env[best_local])
            best_a = float(alpha[best_local])
    return best_a

def build_directions(h, n, rng):

def local_search(start, dx, rng, n, target, sweep_limit):
    h = start.astype(np.float64).copy()
    sc = exact_score(h, dx)
    best_h = h.copy()
    best_sc = sc
    for _ in range(sweep_limit):
        if best_sc <= 0.3808:
            break
        dirs = build_directions(h, n, rng)
        a = np.correlate(h, 1.0 - h, mode='full')
        for d in dirs:
            b = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
            c = -np.correlate(d, d, mode='full')
            if not (np.all(np.isfinite(b)) and np.all(np.isfinite(c))):
                continue
            iv = feasible_interval(h, d)
            if iv is None:
                continue
            al, ah = iv
            if ah - al < 1e-12:
                continue
            alpha = line_search_small(h, d, dx, a, b, c, al, ah) if n <= 96 else line_search_large(h, d, dx, a, b, c, al, ah)
            prop = capped_simplex_project(h + alpha * d, target)
            if prop is None or not is_feasible(prop, n, target):
                continue
            psc = exact_score(prop, dx)
            if psc < sc - 1e-12:
                h = prop
                sc = psc
                if sc < best_sc:
                    best_sc = sc
                    best_h = h.copy()
                a = np.correlate(h, 1.0 - h, mode='full')
    return best_h, best_sc

    if parent is not None and valid_starts:
        sel.append(parent)
    sel.append(uniform)
    for sc, s in scored:
        if len(sel) >= 3:
            break
        dup = any(float(np.max(np.abs(s - v))) < 1e-10 for v in sel)
        if not dup:
            sel.append(s)
    sweeps = 2 if n <= 96 else 1
    for st in sel:
        if not is_feasible(st, n, target):
            continue
        bh, bs = local_search(st, dx, rng, n, target, sweeps)
        if bs < inc_sc - 1e-12:
            incumbent = bh
            inc_sc = bs
        if inc_sc <= 0.3808:
            break
    for sigma in [0.02, 0.08]:
        if n > 96 and sigma > 0.04:
            break
        if inc_sc <= 0.3808:
            break
        perturbed = capped_simplex_project(incumbent + sigma * rng.standard_normal(n), target)
        if perturbed is not None and is_feasible(perturbed, n, target):
            bh, bs = local_search(perturbed, dx, rng, n, target, 1)
            if bs < inc_sc - 1e-12:
                incumbent = bh
                inc_sc = bs
    final = capped_simplex_project(incumbent, target)
    if final is None or not is_feasible(final, n, target):
        final = uniform
    c5 = exact_score(final, dx)
    if not np.isfinite(c5):
        final = uniform
        c5 = exact_score(final, dx)
    return (final, c5, n)
