import numpy as np

def project_box_sum(z, N):
    if z is None or not hasattr(z, 'shape') or z.shape != (N,) or not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    if np.max(np.abs(z)) > 1e12:
        return np.full(N, 0.5)
    m = N / 2.0
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        v = np.clip(z - mid, 0.0, 1.0)
        if np.sum(v) > m:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    v = np.clip(z - theta, 0.0, 1.0)
    e = m - np.sum(v)
    for _ in range(4):
        if abs(e) < 1e-14:
            break
        for i in range(N):
            if e > 0:
                add = min(e, 1.0 - v[i])
                v[i] += add
                e -= add
            elif e < 0:
                sub = min(-e, v[i])
                v[i] -= sub
                e += sub
            if abs(e) < 1e-14:
                break
    if not np.all(np.isfinite(v)) or np.any(v < -1e-12) or np.any(v > 1.0 + 1e-12):
        return np.full(N, 0.5)
    v = np.clip(v, 0.0, 1.0)
    if abs(np.sum(v) - m) > 1e-10 * max(1, N):
        return np.full(N, 0.5)
    return v


def feasible(h, N):
    if h is None or not hasattr(h, 'shape') or h.shape != (N,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - N / 2.0) <= 1e-10 * max(1, N)


def score_exact(h, N):
    if h is None or not hasattr(h, 'shape') or h.shape != (N,):
        return float('inf')
    if not np.all(np.isfinite(h)):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(c)) * (2.0 / N)
    if not math.isfinite(val):
        return float('inf')
    return val


def make_fourier_profile(N, rng):
    R = min(5, (N - 1) // 2)
    if R <= 0:
        return np.full(N, 0.5)
    i = np.arange(N, dtype=float)
    z = np.full(N, 0.5)
    for r in range(1, R + 1):
        A = rng.standard_normal()
        B = rng.standard_normal()
        coeff = 0.12 / r
        z += coeff * (A * np.sin(2.0 * np.pi * r * i / N) + B * np.cos(2.0 * np.pi * r * i / N))
    return z
