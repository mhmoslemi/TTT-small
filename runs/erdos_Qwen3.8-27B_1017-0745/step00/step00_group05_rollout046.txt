            if s < ps[i] - 1e-12:
                npop[i], nps[i] = p, s
                adds.append(pop[i].copy())
                dl.append(ps[i] - s)
                fu.append(Fi)
                cu.append(Cri)
                if s < inc_s - 1e-12:
                    inc, inc_s = p.copy(), s
        pop, ps = npop, nps
        archive.extend(adds)
        while len(archive) > NP:
            archive.pop(int(rng.integers(0, len(archive))))
        if dl:
            wa = np.array(dl) / sum(dl)
            ma, ca = np.array(fu), np.array(cu)
            denom = float(np.sum(wa * ma))
            if denom > 1e-15:
                muF = float(np.clip(0.9 * muF + 0.1 * float(np.sum(wa * ma**2)) / denom, 0.05, 0.95))
            muCR = float(np.clip(0.9 * muCR + 0.1 * float(np.sum(wa * ca)), 0.05, 0.95))
        if inc_s < prev_s - 1e-12:
            stagnant = 0
        else:
            stagnant += 1
        if stagnant >= stag_lim and restarts < 2 and ec + (NP - max(2, NP // 4)) <= budget:
            en = max(2, NP // 4)
            o2 = np.argsort(ps)
            keep = [int(o2[j]) for j in range(en)]
            np2 = [pop[k].copy() for k in keep]
            np2[0] = inc.copy()
            nps2 = [_score(x, dx) for x in np2]
            sigs = [0.02, 0.06, 0.15, 0.30]
            for j in range(en, NP):
                if ec >= budget:
                    np2[j] = inc.copy()
                    nps2[j] = inc_s
                    continue
                if (j - en) % 2 == 0:
                    v = inc + sigs[(j - en) // 2 % 4] * rng.standard_normal(n)
                else:
                    v = _pstart(n, rng, inc)
                p, s = _pscore(v, m, n, dx)
                ec += 1
                if p is not None:
                    np2[j] = p
                    nps2[j] = s
                    if s < inc_s - 1e-12:
                        inc, inc_s = p.copy(), s
                else:
                    np2[j] = inc.copy()
                    nps2[j] = inc_s
            pop, ps = np2, nps2
            archive = []
            muF, muCR = 0.5, 0.7
            stagnant = 0
            restarts += 1
        if inc_s <= 0.3808:
            break

    pf = _proj(inc, m)
    final = pf if (pf is not None and _feas(pf, n)) else fb.copy()
    fs = _score(final, dx)
    if fs == float('inf') or not _feas(final, n):
        final = np.full(n, 0.5)
        fs = _score(final, dx)
    return final, fs, n
