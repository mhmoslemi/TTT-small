def capped_simplex_project(v, target):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0 or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - np.sum(y)
    if abs(r) > 1e-12:
        if r > 0:
            cap = 1.0 - y
            idx = np.where(cap > 1e-12)[0]
            if len(idx) > 0:
                y[idx] += r / cap[idx].sum()
        else:
            idx = np.where(y > 1e-12)[0]
            if len(idx) > 0:
                y[idx] += r / y[idx].sum()
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(np.sum(y) - target) > 1e-10:
        return None
    return y

        return np.inf
    if np.min(h) < -1e-10 or np.max(h) > 1.0 + 1e-10:
        return np.inf
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def smooth_value_gradient(h, tau, dx):
    n = len(h)
    c = dx * np.correlate(h, 1.0 - h, mode='full')
    m = np.max(c)
    e = np.exp((c - m) / tau)
    s = np.sum(e)
    F = m + tau * np.log(s)
    p = e / s
    grad = np.zeros(n, dtype=np.float64)

        t = idx - (n - 1)

            grad[u:] -= wt * h[:n - u]
    if not np.all(np.isfinite(grad)) or not np.isfinite(F):
        return (1e10, np.zeros(n))
    return (F, grad)
