from scipy.optimize import minimize

def exact_score(h, dx):
    if h is None or not np.all(np.isfinite(h)):
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def capped_simplex_project(v, target):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    lo, hi = np.min(v) - 1.0, np.max(v) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - np.sum(y)
    if abs(r) > 1e-14:
        if r > 0:
            cap = 1.0 - y
            idx = np.where(cap > 1e-14)[0]
            if len(idx) > 0:
                y[idx] += min(r / len(idx), cap[idx].min())
        else:
            idx = np.where(y > 1e-14)[0]
            if len(idx) > 0:
                y[idx] -= min(-r / len(idx), y[idx].min())
    if not np.all(np.isfinite(y)) or np.min(y) < -1e-10 or np.max(y) > 1.0 + 1e-10:
        return None
    if abs(np.sum(y) - target) > 1e-8:
        return None
    return y

def is_feasible(h, target):
    if h is None or getattr(h, 'ndim', 0) != 1 or not np.all(np.isfinite(h)):
        return False
    return (np.min(h) >= -1e-12 and np.max(h) <= 1.0 + 1e-12
            and abs(np.sum(h) - target) <= 1e-10)

def smooth_obj_grad(h, tau, dx, n):
    s = dx * np.correlate(h, 1.0 - h, mode='full')
    m = np.max(s)
    e = np.exp((s - m) / tau)
    Z = np.sum(e)
    F = m + tau * np.log(Z)
    p = e / Z
    grad = np.zeros(n, dtype=np.float64)
    for k in range(2 * n - 1):
        t = k - (n - 1)
        pt = p[k]
        if pt < 1e-15:
            continue
        jlo, jhi = max(0, -t), min(n, n - t)
        if jlo < jhi:
            idx = np.arange(jlo, jhi)
            grad[idx] += pt * (1.0 - h[idx + t])
        jlo2, jhi2 = max(0, t), min(n, n + t)
        if jlo2 < jhi2:
            idx2 = np.arange(jlo2, jhi2)
            grad[idx2] -= pt * h[idx2 - t]
    grad *= dx
    if not np.all(np.isfinite(grad)):
        return (np.inf, None)
    return (F, grad)

def smooth_wrapper(x, tau, dx, n):
    F, g = smooth_obj_grad(x, tau, dx, n)
    if g is None:
        return (1e30, np.zeros_like(x))
    return (F, g)

def mass_eq(x):
    return np.array([np.sum(x) - x.shape[0] / 2.0])

def run_one_start(h0, n, dx, target, best_h, best_score):
    h = h0.copy()
    tau_sched = [0.02, 0.005, 0.001, 2e-4, 5e-5, 1e-5]
    bounds = [(0.0, 1.0)] * n
    for tau in tau_sched:
