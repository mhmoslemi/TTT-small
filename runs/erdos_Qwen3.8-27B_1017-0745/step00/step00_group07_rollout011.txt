    cur_s = exact_score(h)
    best = h.copy()

    stages = [(0.08, 0.001), (0.025, 0.0003), (0.008, 0.00008), (0.002, 0.0)]
    per_stage = max(10, P // 4)

            if i == j:
                continue
            sign = 1.0 if rng.random() < 0.5 else -1.0
            if sign > 0:
                cap = min(1.0 - h[i], h[j])
            else:
                cap = min(h[i], 1.0 - h[j])
            mag = rng.random() * min(step_lim, cap)
            if mag < 1e-12:
                continue
            transfer = sign * mag
            trial = h.copy()
            trial[i] += transfer
            trial[j] -= transfer
            if trial[i] < -1e-12 or trial[i] > 1 + 1e-12 or trial[j] < -1e-12 or trial[j] > 1 + 1e-12:

                cur_s = new_s

    return best, best_s
