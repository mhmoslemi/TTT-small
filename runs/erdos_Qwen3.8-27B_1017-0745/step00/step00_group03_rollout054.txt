    starts.append(project_box_sum(baseline + 0.40 * rng.standard_normal(N), N))
    starts = starts[:n_starts]
    for s in starts:
        try:
            s = project_box_sum(s, N)
            sc = score_exact(s, N)
            if sc < inc_score:
                inc_score = sc
                inc_h[:] = s
            inc_h, inc_score = optimize_start(s, N, inc_h, inc_score, iters)
        except Exception:
            pass
    if inc_h.shape != (N,) or not np.all(np.isfinite(inc_h)):
        inc_h = np.full(N, 0.5)
    inc_h = np.clip(inc_h, 0.0, 1.0)
    if abs(inc_h.sum() - m) > 1e-9 * max(1, N):
        inc_h = project_box_sum(inc_h, N)
    fs = score_exact(inc_h, N)
    if not np.isfinite(fs):
        inc_h = np.full(N, 0.5)
        fs = score_exact(inc_h, N)
    return inc_h, fs, N
