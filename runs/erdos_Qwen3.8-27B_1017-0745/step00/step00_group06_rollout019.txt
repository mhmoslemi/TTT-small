                if cy < best_score - 1e-12:
                    best_score, best_h = cy, y.copy()
                move = np.max(np.abs(u_step))
                if rho > 0.75 and move >= 0.8 * delta:
                    delta = min(dmax, 1.5 * delta)
                elif rho < 0.25:
                    delta = max(dmin, delta / 2.0)
            else:
                delta = max(dmin, delta / 2.0)
                if cy < best_score - 1e-12:
                    best_score, best_h = cy, y.copy()
                if delta <= dmin:
                    break

    h_out = best_h
    sc = exact_score(h_out, N)
    if (not np.all(np.isfinite(h_out)) or h_out.shape != (N,)
            or np.any(h_out < -1e-12) or np.any(h_out > 1.0 + 1e-12)
            or abs(np.sum(h_out) - N / 2.0) > 1e-9 * max(1, N)):
        h_out = project_capped_simplex(h_out, N)
        if h_out is None:
            h_out = np.full(N, 0.5)
        sc = exact_score(h_out, N)
    return h_out.astype(np.float64), float(sc), N
