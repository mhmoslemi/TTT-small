def lag_coefficients(h, d, N):
    M = 2 * N - 1
    A = np.zeros(M)
    B = np.zeros(M)
    C = np.zeros(M)
    for k in range(-(N - 1), N):
        idx = k + N - 1
        if k >= 0:
            L = N - k
            a = h[k:k + L]
            b = h[0:L]
            u = d[k:k + L]
            v = d[0:L]
        else:
            s = -k
            L = N - s
            a = h[0:L]
            b = h[s:s + L]
            u = d[0:L]
            v = d[s:s + L]
        A[idx] = np.sum(a * (1.0 - b))
        B[idx] = np.sum(u * (1.0 - b) - a * v)
        C[idx] = -np.sum(u * v)

def line_search(h, d, N, curr_score):
    # d should be zero-sum, normalized
    d = np.asarray(d, dtype=np.float64)
    if not np.all(np.isfinite(d)) or np.all(np.abs(d) < 1e-15):
        return h, curr_score
    # Compute feasible interval
    dz = d[np.abs(d) > 1e-15]
    hz = h[np.abs(d) > 1e-15]
    r0 = -hz / dz
    r1 = (1.0 - hz) / dz
    alpha_lo = np.max(np.minimum(r0, r1))
    alpha_hi = np.min(np.maximum(r0, r1))
    if alpha_hi - alpha_lo < 1e-14:
        return h, curr_score
    A, B, C = lag_coefficients(h, d, N)
    if not (np.all(np.isfinite(A)) and np.all(np.isfinite(B)) and np.all(np.isfinite(C))):
        return h, curr_score
    
    # Generate candidates
    cands = set()
    cands.add(alpha_lo)
    cands.add(alpha_hi)
    cands.add(0.0)
    # Stationary points
    for j in range(len(A)):
        if abs(C[j]) > 1e-15:
            sp = -B[j] / (2.0 * C[j])
            if alpha_lo - 1e-10 <= sp <= alpha_hi + 1e-10:
                cands.add(sp)
    # Pairwise intersections
    M = len(A)
    for i in range(M):
        for j in range(i + 1, M):
            u = C[i] - C[j]
            v = B[i] - B[j]
            w = A[i] - A[j]
            tol = 1e-12 * max(abs(u), abs(v), abs(w), 1e-300)
            if abs(u) <= tol:
                # Linear
                if abs(v) > tol:
                    root = -w / v
                    if alpha_lo - 1e-10 <= root <= alpha_hi + 1e-10:
                        cands.add(root)
            else:
                D = v * v - 4.0 * u * w
                if D < 0:
                    if D > -1e-12 * (v * v + abs(4.0 * u * w) + 1e-300):
                        D = 0.0
                    else:
                        continue
                sqrtD = sqrt(D)
                q = -0.5 * (v + copysign(sqrtD, v))
                if abs(q) > 1e-300:
                    r1v = q / u
                    r2v = w / q
                else:
                    r1v = -v / (2.0 * u)
                    r2v = r1v
                for r in (r1v, r2v):
                    if np.isfinite(r) and alpha_lo - 1e-10 <= r <= alpha_hi + 1e-10:
                        cands.add(r)
    
    # Deduplicate
    cands_list = sorted(set(cands))
    # Remove near-duplicates
    deduped = []
    for c in cands_list:
        if not deduped:
            deduped.append(c)
        elif abs(c - deduped[-1]) > 1e-12 * max(1.0, abs(c)):
            deduped.append(c)
    
    # Clamp to interval
    deduped = [max(alpha_lo, min(alpha_hi, c)) for c in deduped]
    
    # Evaluate envelope
    alphas = np.array(deduped)
    # Batched evaluation
    best_pred = float('inf')
    best_idx = 0
    batch_size = 512
    all_preds = np.full(len(alphas), float('inf'))
    for s in range(0, len(alphas), batch_size):
        e = alphas[s:s + batch_size]
        env = A[None, :] + B[None, :] * e[:, None] + C[None, :] * e[:, None] ** 2
        preds = env.max(axis=1)
        all_preds[s:s + batch_size] = preds
    
    # Take top 16 + endpoints + zero
    order = np.argsort(all_preds)
    top = set(order[:16].tolist())
    # Add endpoints and zero
    top.add(0)  # index 0 is alpha_lo
    top.add(len(deduped) - 1)  # index last is alpha_hi
    for i, c in enumerate(deduped):
        if abs(c) < 1e-14:
            top.add(i)
            break
    
    # Evaluate exact score for top candidates
    best_h = h
    best_s = curr_score
    for idx in top:
        alpha = deduped[idx]
        h_new = h + alpha * d
        h_new = project_box_sum(h_new, N)
        s_new = score_exact(h_new, N)
        if s_new < best_s:
            best_s = s_new
            best_h = h_new
    
    return best_h, best_s

def make_directions(h, N, rng, stage):
    dirs = []
    if stage == 'main':
        n_star, n_rand, n_block, n_harm = (8, 6, 4, 2) if N <= 80 else ((6,4,2,1) if N <= 250 else (3,2,1,1))
    else:
        n_star, n_rand, n_block, n_harm = (4, 3, 2, 1) if N <= 80 else ((3,2,1,1) if N <= 250 else (1,1,1,0))
    
    # Star pairs: anchor at largest min(h,1-h)
    anchor = int(np.argmax(np.minimum(h, 1.0 - h)))
    used_pairs = set()
    others = [i for i in range(N) if i != anchor]
    # Choose evenly spaced
    if n_star > 0 and others:
        picks = [others[int(i * len(others) / n_star)] for i in range(min(n_star, len(others)))]
        for p in picks:
            d = np.zeros(N)
            d[p] = 1.0
            d[anchor] = -1.0
            dirs.append(d)
            used_pairs.add(frozenset({anchor, p}))
    
    # Random pairs
    for _ in range(n_rand):
        for _ in range(100):
            i, j = rng.integers(0, N, 2)
            if i == j:
                continue
            pr = frozenset({i, j})
            if pr in used_pairs:
                continue
            used_pairs.add(pr)
            d = np.zeros(N)
            d[i] = 1.0
            d[j] = -1.0
            dirs.append(d)
            break
        else:
            # Fallback
            for i in range(N - 1):
                pr = frozenset({i, i + 1})
                if pr not in used_pairs:
                    used_pairs.add(pr)
                    d = np.zeros(N)
                    d[i] = 1.0
                    d[i + 1] = -1.0
                    dirs.append(d)
                    break
    
    # Sparse blocks
    for _ in range(n_block):
        sz = rng.integers(3, min(12, N) + 1)
        idx = rng.choice(N, sz, replace=False)
        d = np.zeros(N)
        w = rng.standard_normal(sz)
        w -= w.mean()
        w[int(rng.integers(0, sz))] += d[idx.sum()]  # fix residual
        d[idx] = w
        if np.max(np.abs(d)) > 1e-15:
            d /= np.max(np.abs(d))
            dirs.append(d)
    
    # Harmonic
    for r in range(1, n_harm + 1):
        phase = rng.uniform(0, 2 * pi)
        t = np.arange(N) * (2.0 * pi * r / N)
        d = np.cos(t + phase)
        d -= d.mean()
        if np.max(np.abs(d)) > 1e-15:
            d /= np.max(np.abs(d))
            dirs.append(d)
    
    return dirs

            except Exception:
                pass
        incumbents.append((ss, hs.copy()))
    incumbents.sort(key=lambda x: x[0])
    final_h = project_box_sum(incumbents[0][1], N)
    final_s = score_exact(final_h, N)
    if not np.isfinite(final_s):
        final_h = np.full(N, 0.5)
        final_s = score_exact(final_h, N)
    return final_h, float(final_s), N
