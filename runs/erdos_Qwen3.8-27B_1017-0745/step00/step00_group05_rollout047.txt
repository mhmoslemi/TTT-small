            np.all(h >= -1e-14) and np.all(h <= 1 + 1e-14) and abs(h.sum() - t) < 1e-10)

def mblock(n):
    h = np.zeros(n); h[:n // 2] = 1.0
    if n % 2: h[n // 2] = 0.5
    return h

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    try:
        par = np.array(initial_h_values, dtype=np.float64).ravel()
        n = len(par)
    except Exception:
        par, n = None, 70
    if not 2 <= n <= 999:
        par, n = None, 70
    dx, tgt = 2.0 / n, float(n) // 2
    fb = np.full(n, 0.5)
    par_ok = False
    if par is not None and par.shape == (n,) and np.all(np.isfinite(par)) and np.all(np.abs(par)) <= 1e6:
        par_ok = True
        if np.all(par >= -1e-10) and np.all(par <= 1 + 1e-10) and abs(par.sum() - tgt) < 1e-8:
            pf = project(par, tgt)
            if pf is not None: fb = pf
    NP = min(64, max(16, 2 * n))
    budget = 12000 if n <= 128 else (3500 if n <= 512 else 900)
    ev = 0
    inc = fb.copy()
    inc_s = escore(inc, dx); ev += 1
    pop, ps = [inc.copy()], [inc_s]

    il = []
    if par_ok:
        il.extend([par.copy(), par[::-1].copy(), 1.0 - par.copy()])
        for s_ in [0.03, 0.08, 0.18, 0.30]:
            il.append(par + s_ * rng.standard_normal(n))
    il.extend([np.full(n, 0.5), mblock(n), rng.random(n), 0.5 + 0.15 * rng.standard_normal(n)])
    for h0 in il:
        if len(pop) >= NP or ev >= budget: break
        p = project(h0, tgt)
        if p is None: continue
        dup = False
        for e in pop:
            if np.max(np.abs(p - e)) < 1e-8: dup = True; break
        if dup: continue
        s = escore(p, dx); ev += 1
        if s < inc_s - 1e-12: inc = p.copy(); inc_s = s
        pop.append(p.copy()); ps.append(s)

    tr = 0
    while len(pop) < NP and ev < budget and tr < 200:
        tr += 1
        p = project(0.5 + 0.2 * rng.standard_normal(n), tgt)
        if p is None: continue
        s = escore(p, dx); ev += 1
        if s < inc_s - 1e-12: inc = p.copy(); inc_s = s
        pop.append(p.copy()); ps.append(s)

    muF, muCR = 0.5, 0.7
    archive = []
    stag, nrestart = 0, 0
    sth = 35 if n <= 128 else (18 if n <= 512 else 8)

    while ev + NP <= budget:
        order = np.argsort(ps)
        pn = max(2, int(0.15 * NP) + 1)
        pb = set(order[:pn].tolist())
        npop = [x.copy() for x in pop]
        nps = ps[:]
        garch, succ = [], []
        imp = False
        for i in range(NP):
            if ev >= budget: break
            F = muF
            for _ in range(8):
                F = muF + 0.1 * rng.standard_cauchy()
                if np.isfinite(F) and F > 0: break
            F = float(np.clip(F, 0.05, 1.0))
            CR = float(np.clip(muCR + 0.1 * rng.standard_normal(), 0, 1))
            if rng.random() < 0.8:
                pi = [j for j in pb if j != i]
                if not pi: pi = [0]
                xp = pi[rng.integers(len(pi))]
                c1 = [j for j in range(NP) if j != i and j != xp]
                r1 = c1[rng.integers(len(c1))]
                pool = list(range(NP)) + [NP + k for k in range(len(archive))]
                pool = [j for j in pool if j != i and j != r1]
                if not pool: pool = [i]
                r2 = pool[rng.integers(len(pool))]
                a2 = pop[r2] if r2 < NP else archive[r2 - NP]
                v = pop[i] + F * (pop[xp] - pop[i]) + F * (pop[r1] - a2)
            else:
                c1 = [j for j in range(NP) if j != i]
                r1 = c1[rng.integers(len(c1))]
                c2 = [j for j in c1 if j != r1]
                r2 = c2[rng.integers(len(c2))]
                c3 = [j for j in c2 if j != r2]
                r3 = c3[rng.integers(len(c3))]
                v = pop[r1] + F * (pop[r2] - pop[r3])
            jr = rng.integers(n)
            mk = rng.random(n) < CR
            mk[jr] = True
            trv = np.where(mk, v, pop[i])
            if not np.all(np.isfinite(trv)): continue
            p = project(trv, tgt)
            if p is None: continue
            s = escore(p, dx); ev += 1
            if s >= ps[i] - 1e-12: continue
            garch.append(pop[i].copy())
            npop[i] = p.copy(); nps[i] = s
            succ.append((F, CR, ps[i] - s))
            if s < inc_s - 1e-12: inc = p.copy(); inc_s = s; imp = True
        pop, ps = npop, nps
        archive.extend(garch)
        while len(archive) > NP:
            archive.pop(rng.integers(len(archive)))
        if succ:
            tot = sum(x[2] for x in succ)
            wfac = sum(x[2] / tot * x[0] for x in succ)
            wf = sum(x[2] / tot * x[0] ** 2 for x in succ) / max(wfac, 1e-15)
            muF = float(np.clip(0.9 * muF + 0.1 * wf, 0.05, 0.95))
            muCR = float(np.clip(0.9 * muCR + 0.1 * sum(x[2] / tot * x[1] for x in succ), 0.05, 0.95))
        stag = 0 if imp else stag + 1
        if inc_s <= 0.3808: break
        if stag >= sth and nrestart < 2:
            el = max(2, NP // 4)
            o2 = np.argsort(ps)
            elite = o2[:el].tolist()
            ok_inc = False
            for e in elite:
                if ps[e] <= inc_s + 1e-12: ok_inc = True; break
            if not ok_inc: elite[-1] = -1
            np2, nps2 = [], []
            for e in elite:
                if e >= 0: np2.append(pop[e].copy()); nps2.append(ps[e])
                else: np2.append(inc.copy()); nps2.append(inc_s)
            sc = [0.02, 0.06, 0.15, 0.30]
            si = 0; rt = 0
            while len(np2) < NP and ev < budget and rt < 100:
                rt += 1
                h0 = inc + sc[si % 4] * rng.standard_normal(n) if si % 2 == 0 else 0.5 + 0.2 * rng.standard_normal(n)
                si += 1
                p = project(h0, tgt)
                if p is None: continue
                s = escore(p, dx); ev += 1
                if s < inc_s - 1e-12: inc = p.copy(); inc_s = s
                np2.append(p); nps2.append(s)
            if len(np2) >= 2:
                pop, ps = np2, nps2
            archive = []
            muF, muCR = 0.5, 0.7
            stag = 0; nrestart += 1

    fp = project(inc, tgt)
    if fp is None or not isfeas(fp, n, tgt):
        fp = fb.copy()
    fs = escore(fp, dx)
    if not np.isfinite(fs) or not isfeas(fp, n, tgt):
        fp = np.full(n, 0.5); fs = escore(fp, dx)
    return fp, float(fs), int(n)
