            res = minimize(_obj, h, args=(tau, dx), jac=_jac,
                           method='SLSQP', bounds=bl, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-10, 'disp': False})
            if res.x is not None and len(res.x) == n and np.all(np.isfinite(res.x)):
                rh = capped_simplex_project(res.x, target)
                if rh is not None and check_feasible(rh, n):
                    hv, _ = smooth_val_grad(h, tau, dx)
                    rv, _ = smooth_val_grad(rh, tau, dx)
                    if np.isfinite(rv) and rv <= hv + 1e-10:
                        h = rh
        except Exception:
            pass
    return h

def run(seed=42, **kwargs):
    n = 70
    dx = 2.0 / n
    target = n / 2.0
    rng = np.random.default_rng(seed)

    parent = None
    try:
        pv = np.array(initial_h_values, dtype=np.float64).flatten()
        if 2 <= len(pv) <= 999 and np.all(np.isfinite(pv)):
            n = len(pv)
            dx = 2.0 / n
            target = n / 2.0
            if np.min(pv) >= -1e-6 and np.max(pv) <= 1.0 + 1e-6:
                parent = capped_simplex_project(pv, target)
    except Exception:
        pass

    uniform = np.full(n, 0.5)
    fallback = parent.copy() if parent is not None else uniform.copy()
    incumbent = fallback.copy()
    inc_score = exact_score(incumbent, dx)

    def _try_update(h_cand):
        nonlocal incumbent, inc_score
        hc = capped_simplex_project(h_cand, target)
        if hc is None or not check_feasible(hc, n):
            return
        sc = exact_score(hc, dx)
        if np.isfinite(sc) and sc < inc_score - 1e-12:
            incumbent = hc.copy()
            inc_score = sc

    starts = []
    if parent is not None:
        starts.append(parent.copy())
        starts.append(parent[::-1].copy())
        for sig in [0.04, 0.10, 0.22, 0.40]:
            starts.append(parent + sig * rng.standard_normal(n))
    starts.append(rng.uniform(0, 1, n))
    starts.append(rng.beta(0.5, 0.5, n))
    starts.append(np.full(n, 0.5))
    if parent is None:
        starts = [np.full(n, 0.5)]
        for _ in range(3):
            starts.append(rng.uniform(0, 1, n))
        for _ in range(2):
            starts.append(rng.beta(0.5, 0.5, n))

    for s in starts:
        res_h = optimize_from(s, target, dx, n)
        if res_h is not None:
            _try_update(res_h)

    for sig in [0.02, 0.08, 0.20]:
        center = incumbent + sig * rng.standard_normal(n)
        res_h = optimize_from(center, target, dx, n)
        if res_h is not None:
            _try_update(res_h)

    final = capped_simplex_project(incumbent, target)
    if final is None or not check_feasible(final, n):
        final = capped_simplex_project(fallback, target)
        if final is None or not check_feasible(final, n):
            final = uniform.copy()
    c5 = exact_score(final, dx)
    if not np.isfinite(c5):
        final = fallback.copy()
        c5 = exact_score(final, dx)
    return final, c5, n
