                        if h_new.min() >= -1e-5 and h_new.max() <= 1.0 + 1e-5:
                            return dv, tv
        except Exception:
            continue
    return None

def run(seed=42, **kwargs):
    n = 45
    parent = None
    try:
        ihv = globals().get('initial_h_values', None)
        if ihv is not None:
            arr = np.array(ihv, dtype=np.float64)
            if arr.ndim == 1 and 2 <= len(arr) <= 999:
                n = len(arr)
                S_tmp = n / 2.0
                if np.isfinite(arr).all() and np.abs(arr).max() <= 1e6:
                    if arr.min() >= 0 and arr.max() <= 1.0 and abs(arr.sum() - S_tmp) <= 1e-12:
                        parent = arr.copy()
                    else:
                        parent = project_to_K(arr, S_tmp)
    except Exception:
        parent = None

    dx = 2.0 / n
    S = n / 2.0

    if parent is None:
        if 'n_points' in kwargs:
            npv = kwargs['n_points']
            if isinstance(npv, (int, np.integer)) and not isinstance(npv, (bool, np.bool_)) and 2 <= int(npv) <= 999:
                n = int(npv)
                dx = 2.0 / n
                S = n / 2.0
        parent = np.full(n, 0.5, dtype=np.float64)

    fallback_h = parent.copy()
    _, fb_score = evaluate_lags(fallback_h, dx)
    if fb_score is None:
        fb_score = 1.0
    best_h = parent.copy()
    best_score = fb_score

    rng = np.random.default_rng(seed)
    starts = []

    def add_start(hc):
        if hc is None:
            return
        p = project_to_K(hc, S)
        if p is None:
            return
        for s in starts:
            if np.max(np.abs(p - s)) <= 1e-8:
                return
        starts.append(p)

    add_start(parent)
    add_start(1.0 - parent)
    add_start(parent[::-1].copy())
    add_start(parent + rng.normal(0, 0.08, n))
    add_start(parent + rng.normal(0, 0.25, n))
    try:
        add_start(rng.beta(0.5, 0.5, n))
    except Exception:
        pass
    if parent is None:
        add_start(rng.uniform(0, 1, n))
        mode = rng.integers(1, max(1, min(4, n // 2)) + 1)
        phase = rng.uniform(0, 2 * math.pi)
        idx = np.arange(n, dtype=np.float64)
        add_start(0.5 + 0.35 * np.cos(2 * math.pi * mode * (idx + 0.5) / n + phase))

    max_starts = 6 if n <= 64 else (4 if n <= 256 else 3)
    starts = starts[:max_starts]
    max_iter = 40 if n <= 64 else (20 if n <= 256 else 8)
    max_bt = 9 if n <= 64 else (6 if n <= 256 else 4)

    for start in starts:
        try:
            cur_h = start.copy()
            _, cur_score = evaluate_lags(cur_h, dx)
            if cur_score is None:
                continue
            if cur_score < best_score - 1e-12:
                best_h = cur_h.copy()
                best_score = cur_score
            tiny_count = 0
            for it in range(max_iter):
                q, cur_score = evaluate_lags(cur_h, dx)
                if q is None:
                    break
                G = compute_grad_matrix(cur_h, dx, n)
                if not np.isfinite(G).all():
                    break
                res = solve_qp_sub(cur_h, q, G, dx, S, n)
                if res is None:
                    break
                d_sol = res[0]
                h_ep = project_to_K(cur_h + d_sol, S)
                if h_ep is None:
                    break
                d_feas = h_ep - cur_h
                if np.max(np.abs(d_feas)) <= 1e-9:
                    break
                best_ts = cur_score
                best_th = None
                for k in range(max_bt):
                    alpha = 2.0 ** (-k)
                    ht = project_to_K(cur_h + alpha * d_feas, S)
                    if ht is None:
                        continue
                    _, ts = evaluate_lags(ht, dx)
                    if ts is None:
                        continue
                    if ts < best_ts:
                        best_ts = ts
                        best_th = ht
                if best_th is None or best_ts >= cur_score - 1e-12:
                    break
                dec = cur_score - best_ts
                cur_h = best_th.copy()
                cur_score = best_ts
                if cur_score < best_score - 1e-12:
                    best_h = cur_h.copy()
                    best_score = cur_score
                if dec < 1e-10:
                    tiny_count += 1
                    if tiny_count >= 3:
                        break
                else:
                    tiny_count = 0
        except Exception:
            continue

    final_h = best_h.copy()
    _, final_score = evaluate_lags(final_h, dx)
    if final_score is None or not np.isfinite(final_score):
        final_h = fallback_h.copy()
        _, final_score = evaluate_lags(final_h, dx)
    if final_score is None or not np.isfinite(final_score):
        final_h = np.full(n, 0.5, dtype=np.float64)
        _, final_score = evaluate_lags(final_h, dx)
    final_h = np.clip(final_h, 0.0, 1.0)
    final_h = project_to_K(final_h, S)
    if final_h is None:
        final_h = np.full(n, 0.5, dtype=np.float64)
        _, final_score = evaluate_lags(final_h, dx)
    else:
        _, final_score = evaluate_lags(final_h, dx)
    return final_h.astype(np.float64), float(final_score), int(n)
