import numpy as np
import math

def project_box_sum(v, S):
    n = len(v)
    if n == 0 or not np.all(np.isfinite(v)):
        return None
    if S < 0 or S > n:
        return None
    if float(np.max(np.abs(v))) > 1e12:
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        m = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - m, 0.0, 1.0)))
        if s > S:
            lo = m
        else:
            hi = m
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    residual = S - float(np.sum(y))
    for i in range(n):
        if abs(residual) < 1e-14:
            break
        if residual > 0:
            a = min(residual, 1.0 - y[i])
            y[i] += a
            residual -= a
        else:
            s2 = min(-residual, y[i])
            y[i] -= s2
            residual += s2
    if not np.all(np.isfinite(y)) or np.max(y) > 1.0 + 1e-10 or np.min(y) < -1e-10:
        return None
    if abs(float(np.sum(y)) - S) > 1e-10 * max(1, n):
        return None
    return y

def is_feasible(h, n):
    return (h.shape == (n,) and bool(np.all(np.isfinite(h))) and
            float(np.max(h)) <= 1.0 + 1e-12 and float(np.min(h)) >= -1e-12 and
            abs(float(np.sum(h)) - n / 2.0) <= 1e-10 * max(1, n))

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    if len(corr) != 2 * n - 1 or not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr)) * (2.0 / n)

    if 'initial_h_values' in globals():

            n = len(p)

    return np.full(45, 0.5), 45
