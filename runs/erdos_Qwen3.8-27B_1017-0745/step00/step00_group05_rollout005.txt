    return np.ones(len(h))

import numpy as np
from scipy.optimize import minimize


def capped_simplex_project(v, target):
    n = len(v)
    if n < 1 or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - np.sum(y)
    if abs(r) > 1e-10:
        if r > 0:
            cap = 1.0 - y
            avail = np.sum(cap)
            if avail > 1e-12:
                y = y + r * cap / avail
        else:
            avail = np.sum(y)
            if avail > 1e-12:
                y = y + r * y / avail
        y = np.clip(y, 0.0, 1.0)
        if abs(target - np.sum(y)) > 1e-10:
            return None
    return y


def exact_score(h, dx):
    if len(h) < 1 or not np.all(np.isfinite(h)):
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))


def feasible(h, n):
    if len(h) != n or not np.all(np.isfinite(h)):
        return False
    if np.min(h) < -1e-12 or np.max(h) > 1.0 + 1e-12:
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-10


def mass_residual(h):
    return np.sum(h) - len(h) / 2.0


def mass_jacobian(h):
    return np.ones(len(h))


def smooth_objective(x, tau, dx):
    n = len(x)
    s = dx * np.correlate(x, 1.0 - x, mode='full')
    m = np.max(s)
    e = np.exp((s - m) / tau)
    se = np.sum(e)
    F = m + tau * np.log(se)
    p = e / se
    lags = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for idx in range(len(lags)):
        t = lags[idx]
        pi = p[idx]
        if pi < 1e-15:
            continue
        i_lo = max(0, -t)
        i_hi = min(n - 1, n - 1 - t)
        for j in range(i_lo, i_hi + 1):
            grad[j] += pi * dx * (1.0 - x[j + t])
        j_lo2 = max(0, t)
        j_hi2 = n - 1 if t >= 0 else n - 1 + t
        for j in range(j_lo2, j_hi2 + 1):
            grad[j] -= pi * dx * x[j - t]
    return F, grad


def run_constrained(x0, n, dx, taus, maxiter=200):
    x = x0.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': mass_residual, 'jac': mass_jacobian}]
    for tau in taus:
        try:
            res = minimize(smooth_objective, x, args=(tau, dx),
                           method='SLSQP', jac=True, bounds=bounds,
                           constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-10, 'disp': False})
            if res.x is not None and np.all(np.isfinite(res.x)):
                proj = capped_simplex_project(res.x, n / 2.0)
                if proj is not None:
                    F_old = smooth_objective(x, tau, dx)[0]
                    F_new = smooth_objective(proj, tau, dx)[0]
                    if np.isfinite(F_new) and F_new <= F_old + 1e-10:
                        x = proj
        except Exception:
            pass
    return x
