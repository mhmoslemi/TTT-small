                    best_sc = tsc; best_h = trial.copy()
                if was_local:
                    local_att += 1
                delta = tsc - cur_sc
                if delta <= 0:
                    cur = trial.copy(); cur_sc = tsc
                    if was_local:
                        local_acc += 1
                else:
                    u = max(rng.random(), np.finfo(np.float64).tiny)
                    if math.log(u) < -delta / Tj:
                        cur = trial.copy(); cur_sc = tsc
                        if was_local:
                            local_acc += 1
            if j < 5 and local_att > 0:
                frac = local_acc / local_att
                if frac < 0.15:
                    r = max(0.03, 0.75 * r)
                elif frac > 0.45:
                    r = min(1.2, 1.20 * r)

    # Finalize
    fh = project_box_sum(best_h, S)
    if fh is None or not is_feasible(fh, n):
        fh = fb.copy()
    fs = exact_score(fh, n)
    if fs is None:
        fs = best_sc
    ffh = project_box_sum(fb, S)
    if ffh is not None and is_feasible(ffh, n):
        ffs = exact_score(ffh, n)
        if ffs is not None and ffs < fs:
            fh = ffh; fs = ffs
    else:
        cv = np.full(n, 0.5); fs = exact_score(cv, n)
        if fs is not None:
            fh = cv
    if fs is None:
        fs = float(np.max(np.correlate(fh, 1.0 - fh, mode='full')) * (2.0 / n))
    return (np.float64(fh), float(fs), n)
