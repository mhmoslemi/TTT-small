            if proj is not None:
                parent = proj.copy()
    taus = [0.02, 0.005, 0.001, 0.0002, 0.00005, 0.00001]
    starts = []
    if parent is not None:
        starts.append(parent.copy())
        starts.append(parent[::-1].copy())
        for sig in [0.04, 0.10, 0.22, 0.40]:
            pert = parent + sig * rng.standard_normal(n)
            pr = capped_simplex_project(pert, target)
            if pr is not None:
                starts.append(pr)
    starts.append(capped_simplex_project(rng.uniform(0, 1, n), target))
    starts.append(capped_simplex_project(rng.beta(0.5, 0.5, n), target))
    starts.append(np.full(n, 0.5, dtype=np.float64))
    incumbent = None
    incumbent_score = float('inf')
    for s in starts:
        if s is None:
            continue
        if not check_feasible(s, n):
            continue
        sc = exact_score(s, dx)
        if sc < incumbent_score - 1e-12:
            incumbent_score = sc
            incumbent = s.copy()
    for s in starts:
        if s is None or not check_feasible(s, n):
            continue
        result = run_continuous(s, n, dx, taus)
        if result is not None and check_feasible(result, n):
            sc = exact_score(result, dx)
            if sc < incumbent_score - 1e-12:
                incumbent_score = sc
                incumbent = result.copy()
    if incumbent is None:
        incumbent = np.full(n, 0.5, dtype=np.float64)
        incumbent_score = exact_score(incumbent, dx)
    for sig in [0.02, 0.08, 0.20]:
        pert = incumbent + sig * rng.standard_normal(n)
        pr = capped_simplex_project(pert, target)
        if pr is None:
            continue
        result = run_continuous(pr, n, dx, taus)
        if result is not None and check_feasible(result, n):
            sc = exact_score(result, dx)
            if sc < incumbent_score - 1e-12:
                incumbent_score = sc
                incumbent = result.copy()
    final = capped_simplex_project(incumbent, target)
    if final is None:
        final = np.full(n, 0.5, dtype=np.float64)
    final_score = exact_score(final, dx)
    if not np.isfinite(final_score):
        final_score = incumbent_score if np.isfinite(incumbent_score) else 1.0
    return (final, final_score, n)
