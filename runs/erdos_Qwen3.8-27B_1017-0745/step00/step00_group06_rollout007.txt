        if s > m:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - hi, 0.0, 1.0)
    delta = m - y.sum()
    if delta > 1e-14:
        for i in range(N):
            a = min(delta, 1.0 - y[i])
            y[i] += a; delta -= a
            if abs(delta) < 1e-14: break
    elif delta < -1e-14:
        for i in range(N):
            a = min(-delta, y[i])
            y[i] -= a; delta += a
            if abs(delta) < 1e-14: break
    return y

def smooth_obj(x, d, N, tau, m):
    R = compute_R(x)
    a = R.max()
    return a + tau * np.log(np.sum(np.exp((R - a) / tau)))

def smooth_grad(x, d, N, tau, m):
    R = compute_R(x)
    a = R.max()
    w = np.exp((R - a) / tau)
    w /= w.sum()
    return w @ compute_R_jac(x)

def mass_eq(x, d, N, tau, m):
    return x.sum() - m

def mass_eq_jac(x, d, N, tau, m):
    return np.ones(len(x))

def epi_obj(z, d, N, tau, m):
    return z[-1]

def epi_grad(z, d, N, tau, m):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def epi_ineq(z, d, N, tau, m):
    return z[-1] - compute_R(z[:-1])

def epi_ineq_jac(z, d, N, tau, m):
    h = z[:-1]
    J = compute_R_jac(h)
    Q = 2 * N - 1
    out = np.zeros((Q, N + 1))
    out[:, :N] = -J
    out[:, N] = 1.0
    return out

def epi_mass(z, d, N, tau, m):
    return z[:-1].sum() - m

def epi_mass_jac(z, d, N, tau, m):
    g = np.zeros(N + 1)
    g[:N] = 1.0
    return g

def run(seed=42, **kwargs):
    N = 95
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size >= 2 and np.all(np.isfinite(p)):
            parent = p.copy()
            N = parent.size
    except Exception:
        pass
    if parent is None:
        try:
            N = int(kwargs.get('n_points', 95))
        except Exception:
            N = 95
        N = max(2, min(N, 999))
    d = 2.0 / N
    m = N / 2.0
    uniform = np.full(N, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    starts = []
    if parent is not None:
        if N > 999:
            hp = project_onto_set(parent, m)
            sc = compute_score(hp)
            return (hp, float(sc), N)
        starts.append(project_onto_set(parent, m))
    starts.append(uniform.copy())
    step = np.zeros(N)
    fl = int(m)
    step[:fl] = 1.0
    if fl < N: step[fl] = m - fl
    starts.append(step)
    ii = np.arange(N)
    for f in [1, 3, 7]:
        wv = 0.5 + 0.22 * np.sin(2.0 * np.pi * f * ii / N)
        starts.append(project_onto_set(wv, m))
    dedup = []
    for s in starts:
        dup = False
        for d2 in dedup:
            if np.max(np.abs(s - d2)) < 1e-8:
                dup = True; break
        if not dup:
            dedup.append(s)
    starts = dedup
    try:
        n_restarts = max(6, min(int(kwargs.get('n_restarts', 8)), 12))
    except Exception:
        n_restarts = 8
    while len(starts) < n_restarts:
        base = parent if parent is not None else uniform
        sd = [0.04, 0.10, 0.20][len(starts) % 3]
        noise = rng.normal(0, sd, N)
        noise = 0.5*noise + 0.25*np.roll(noise,1) + 0.25*np.roll(noise,-1)
        starts.append(project_onto_set(base + noise, m))
    incumbent = starts[0].copy()
    inc_score = compute_score(incumbent)
    cands = [(inc_score, incumbent.copy())]
    try:
        maxiter = max(60, min(int(kwargs.get('maxiter', 120)), 250))
    except Exception:
        maxiter = 120
    taus = [0.02, 0.005, 0.001, 0.0002]
    bounds = [(0.0, 1.0)] * N
    smooth_cons = [{'type': 'eq', 'fun': mass_eq, 'jac': mass_eq_jac}]
    for s in starts:
        x = s.copy()
        for tau in taus:
            args_s = (d, N, tau, m)
            try:
                res = minimize(smooth_obj, x, jac=smooth_grad, args=args_s,
                               method='SLSQP', bounds=bounds, constraints=smooth_cons,
                               options={'ftol': 1e-10, 'maxiter': maxiter, 'disp': False})
                if res.x is not None and len(res.x) == N and np.all(np.isfinite(res.x)) and np.max(np.abs(res.x)) < 1e6:
                    x = project_onto_set(res.x, m)
            except Exception:
                pass
            sc = compute_score(x)
            cands.append((sc, x.copy()))
    top = []
    used = set()
    for _ in range(3):
        bi = -1; bs = float('inf')
        for i in range(len(cands)):
            if i not in used and cands[i][0] < bs:
                bs = cands[i][0]; bi = i
        if bi >= 0:
            used.add(bi); top.append(cands[bi][1].copy())
        else:
            break
    epi_bounds = [(0.0, 1.0)] * N + [(0.0, 2.1)]
    epi_args = (d, N, 0.0, m)
    epi_cons = [{'type': 'ine', 'fun': epi_ineq, 'jac': epi_ineq_jac},
                {'type': 'eq', 'fun': epi_mass, 'jac': epi_mass_jac}]
    for h0 in top:
        z0 = np.concatenate([h0, [compute_score(h0) + 1e-6]])
        try:
            res = minimize(epi_obj, z0, jac=epi_grad, args=epi_args,
                           method='SLSQP', bounds=epi_bounds, constraints=epi_cons,
                           options={'ftol': 1e-10, 'maxiter': min(400, 2*maxiter), 'disp': False})
            if res.x is not None and len(res.x) == N + 1 and np.all(np.isfinite(res.x)) and np.max(np.abs(res.x)) < 1e6:
                h_new = project_onto_set(res.x[:N], m)
                sc = compute_score(h_new)
                if sc < inc_score - 1e-12:
                    incumbent = h_new
                    inc_score = sc
        except Exception:
            pass
    incumbent = project_onto_set(incumbent, m)
    if not (incumbent.shape == (N,) and np.all(np.isfinite(incumbent)) and np.all(incumbent >= -1e-12) and np.all(incumbent <= 1.0 + 1e-12) and abs(incumbent.sum() - m) < 1e-8):
        incumbent = uniform.copy()
    c5 = compute_score(incumbent)
    return (incumbent.astype(np.float64), c5, N)
