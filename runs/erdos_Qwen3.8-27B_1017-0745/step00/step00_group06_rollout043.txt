import numpy as np
import math

def project_capped_simplex(v, N):
    if v.shape != (N,) or not np.all(np.isfinite(v)) or np.max(np.abs(v)) > 1e6:
        return None
    M = N / 2.0
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > M:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    delta = M - float(np.sum(y))
    for _ in range(3):
        if abs(delta) <= 1e-12 * max(1, N):
            break
        if delta > 0:
            for i in range(N):
                add = min(delta, 1.0 - y[i])
                if add > 0:
                    y[i] += add; delta -= add
        else:
            for i in range(N):
                sub = min(-delta, y[i])
                if sub > 0:
                    y[i] -= sub; delta += sub
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(float(np.sum(y)) - M) > 1e-9 * max(1, N):
        return None
    return y

def is_feasible(h, N):
    return (h.shape == (N,) and np.all(np.isfinite(h)) and np.min(h) >= 0.0
            and np.max(h) <= 1.0 and abs(float(np.sum(h)) - N / 2.0) <= 1e-9 * max(1, N))

def exact_c5(h, N):
    try:
        c = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / N))
        return c if np.isfinite(c) else float('inf')
    except Exception:
        return float('inf')

def lag_gradient(x, q, N):
    d = 2.0 / N
    g = np.zeros(N, dtype=np.float64)
    jlo = max(0, -q)
    jhi = min(N, N - q)
    for j in range(jlo, jhi):
        g[j + q] += d * (1.0 - x[j])
        g[j] -= d * x[j + q]
    return g

def fw_min_norm(G):
    if G.shape[0] == 1:
        return G[0].copy()
    gbar = G[0].copy().astype(np.float64)
    for _ in range(60):
        dots = G @ gbar
        k = int(np.argmin(dots))
        gap = float(np.dot(gbar, gbar)) - float(dots[k])
        w = G[k].astype(np.float64) - gbar
        ww = float(np.dot(w, w))
        if gap <= 1e-12 * max(1.0, float(np.dot(gbar, gbar))) or ww <= 1e-24:
            break
        alpha = min(1.0, max(0.0, gap / ww))
        gbar = gbar + alpha * w
    return gbar

def add_start(starts, cand, N):
    p = project_capped_simplex(cand, N)
    if p is None or not is_feasible(p, N):
        return
    for s in starts:
        if np.max(np.abs(p - s)) < 1e-8:
            return
    starts.append(p)

def smooth_noise(z):
    s = 0.5 * z + 0.25 * np.roll(z, 1) + 0.25 * np.roll(z, -1)
    s -= float(np.mean(s))
    return s

def optimize_start(x0, N, maxiter, rng):

    Cx = exact_c5(x, N)
    eta_base = 0.20
    eta_min = 1e-4
    no_improve = 0
    d = 2.0 / N
    for _ in range(maxiter):
        R = np.correlate(x, 1.0 - x, mode='full') * d
        Cx = float(np.max(R))

        active = []

            if Cx - float(R[k]) <= eps_act:
                active.append(int(k))
            if len(active) >= 8:
                break
        if not active:
            active = [int(order[0])]
        top2 = [int(order[0])]
        if len(order) > 1:
            top2.append(int(order[1]))
        directions = []
        G = np.array([lag_gradient(x, k - (N - 1), N) for k in active], dtype=np.float64)
        gbar = fw_min_norm(G)

                continue

            dup = False

                    dup = True
                    break
            if not dup:
                directions.append(uk)

                    break
        best_y = None
        best_score = Cx

        for u in directions:

                            continue
                        sc = exact_c5(y, N)

        if best_y is not None:
            x = best_y
            no_improve = 0
            if best_eta >= 0.75 * eta_base:
                eta_base = min(0.25, 1.25 * eta_base)
            else:

        else:
            no_improve += 1
            eta_base = max(eta_min, 0.5 * eta_base)
            if no_improve >= 5 and eta_base <= eta_min * 1.001:
