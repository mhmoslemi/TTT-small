import numpy as np
import math
from scipy.optimize import minimize

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def lags_grad(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        idx = np.arange(n - d)
        ixd = idx + d
        q[2*d-1] = dx * np.sum(h[ixd] * (1.0 - h[idx]))
        G[2*d-1, ixd] += dx * (1.0 - h[idx])
        G[2*d-1, idx] += -dx * h[ixd]
        q[2*d] = dx * np.sum(h[idx] * (1.0 - h[ixd]))
        G[2*d, idx] += dx * (1.0 - h[ixd])
        G[2*d, ixd] += -dx * h[idx]
    return q, G

def cpx_project(v, n):
    S = n / 2.0
    v = np.clip(v, -1e6, 1e6)
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    # residual correction
    resid = S - math.fsum(h.tolist())
    if abs(resid) > 1e-12:
        if resid > 0:
            cap = 1.0 - h
            for j in range(n):
                if cap[j] > 0:
                    adj = min(resid, cap[j])
                    h[j] += adj
                    resid -= adj
                    if abs(resid) < 1e-12: break
        else:
            for j in range(n):
                if h[j] > 0:
                    adj = min(-resid, h[j])
                    h[j] -= adj
                    resid += adj
                    if abs(resid) < 1e-12: break
    # final check
    if not (np.all(h >= 0) and np.all(h <= 1)):
        return np.full(n, 0.5)
    if abs(np.sum(h) - S) > 1e-10:
        # try to fix
        for j in range(n):
            val = S - math.fsum([x for i, x in enumerate(h) if i != j])
            if 0.0 <= val <= 1.0:
                h[j] = val
                break
    if not (np.all(h >= 0) and np.all(h <= 1)) or abs(np.sum(h) - S) > 1e-10:
        return np.full(n, 0.5)
    return h

import numpy as np
import math
from scipy.optimize import minimize

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def lags_grad(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        idx = np.arange(n - d)
        ixd = idx + d
        q[2 * d - 1] = dx * np.sum(h[ixd] * (1.0 - h[idx]))
        G[2 * d - 1, ixd] += dx * (1.0 - h[idx])
        G[2 * d - 1, idx] += -dx * h[ixd]
        q[2 * d] = dx * np.sum(h[idx] * (1.0 - h[ixd]))
        G[2 * d, idx] += dx * (1.0 - h[ixd])
        G[2 * d, ixd] += -dx * h[idx]
    return q, G

def cpx_project(v, n):
    S = n / 2.0
    v = np.clip(np.asarray(v, dtype=float), -1e6, 1e6)
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - math.fsum(h.tolist())
    if abs(resid) > 1e-13:
        if resid > 0:
            for j in range(n):
                adj = min(resid, 1.0 - h[j])
                h[j] += adj
                resid -= adj
                if abs(resid) < 1e-13:
                    break
        else:
            for j in range(n):
                adj = min(-resid, h[j])
                h[j] -= adj
                resid += adj
                if abs(resid) < 1e-13:
                    break
    if not (np.all(h >= 0) and np.all(h <= 1)) or abs(np.sum(h) - S) > 1e-10:
        return np.full(n, 0.5)
    return h

def soft_fun(h, tau):
    q, _ = lags_grad(h)
    qmax = np.max(q)
    return qmax + tau * np.log(np.sum(np.exp((q - qmax) / tau)))

    q, G = lags_grad(h)
    qmax = np.max(q)
    exps = np.exp((q - qmax) / tau)
    p = exps / np.sum(exps)
    return p @ G

def eq_fun(h, n):
    return np.sum(h) - n / 2.0

def eq_jac(h, n):
    return np.ones(n)

def epi_obj(z):
    return z[-1]

def epi_obj_jac(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def epi_eq_fun(z):
    n = len(z) - 1
    return np.sum(z[:-1]) - n / 2.0

def epi_eq_jac(z):
    n = len(z) - 1
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def epi_ineq_fun(z):
    h = z[:-1]
    q, _ = lags_grad(h)
    return z[-1] - q

def epi_ineq_jac(z):
    h = z[:-1]
    n = len(h)
    _, G = lags_grad(h)
    M = 2 * n - 1
    J = np.zeros((M, n + 1))
    J[:, :-1] = -G
    J[:, -1] = 1.0
    return J

    if res.x is not None and len(res.x) == n + 1 and np.all(np.isfinite(res.x)):
        return cpx_project(res.x[:n], n)
