                        g = g - g.mean()
                        gn = float(np.max(np.abs(g)))
                        if gn <= 1e-14:
                            continue
                        for sg in [1.0, -1.0]:
                            u = sg * g / gn
                            for s in range(3):
                                eta = e0 / (2.0 ** s)
                                y = project_capped_simplex(x + eta * u, N)
                                if y is None or not is_feasible(y, N):
                                    continue
                                if float(np.max(np.abs(y - x))) <= 1e-12:
                                    continue
                                sc = exact_c5(y, N)
                                if sc < best_s - 1e-12:
                                    best_s, best_t, best_e = sc, y, eta
            if best_t is not None:
                no_imp = 0
                eta_b = min(0.25, 1.25 * eta_b) if best_e >= 0.75 * eta_b else max(eta_mn, 1.25 * best_e)
                x = best_t
                Cx = best_s
                if Cx < inc_sc - 1e-12:
                    incumbent = x.copy()
                    inc_sc = Cx
            else:
                eta_b = max(eta_mn, 0.5 * eta_b)
    final = project_capped_simplex(incumbent, N)
    if final is None or not is_feasible(final, N):
        final = saved_fb
    if not is_feasible(final, N):
        final = np.full(N, 0.5, dtype=np.float64)
    c5 = exact_c5(final, N)
    return (final, c5, N)
