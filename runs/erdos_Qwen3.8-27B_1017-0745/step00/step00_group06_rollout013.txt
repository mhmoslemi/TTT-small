def _project_capped_simplex(v, N):
    """Project v onto {0<=x<=1, sum(x)=N/2}."""
    M = N / 2.0
    v = np.asarray(v, dtype=np.float64)
    if v.shape != (N,) or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        lam = 0.5 * (lo + hi)
        y = np.clip(v - lam, 0.0, 1.0)
        s = float(np.sum(y))
        if s > M:
            lo = lam
        else:
            hi = lam
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    # Repair residual
    residual = M - float(np.sum(y))
    for _ in range(2):
        if abs(residual) < 1e-12:
            break
        for i in range(N):
            if residual > 0 and y[i] < 1.0:
                add = min(residual, 1.0 - y[i])
                y[i] += add
                residual -= add

def _score(h):

        return float('inf')
    if not np.all(np.isfinite(h)):
        return float('inf')
    try:
        c = np.correlate(h, 1.0 - h, mode='full')
        return float(np.max(c)) * (2.0 / N)
    except Exception:
        return float('inf')

def _compute_RG(h):
    """Compute R (2N-1,) and G (2N-1, N)."""
    N = len(h)
    alpha = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')

    G = np.zeros((2 * N - 1, N), dtype=np.float64)

        j_hi = min(N, N - q)
        for j in range(j_lo, j_hi):
            r = j + q
            G[row, r] += alpha * (1.0 - h[j])
            G[row, j] -= alpha * h[j + q]
    return R, G

def _run_lp(x, R, G, delta):

    c[N] = 1.0

    b_ub = -R
    A_eq = np.zeros((1, nvar))
    A_eq[0, :N] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(N):
        lo = max(-x[i], -delta)
        hi = min(1.0 - x[i], delta)
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'display': False, 'time_limit': 10})
    except Exception:
        return None
    if res is None or res.status != 0 or not hasattr(res, 'x') or res.x is None:
        return None
    u = res.x[:N]
    if not np.all(np.isfinite(u)):
        return None
    return u

import numpy as np
from scipy.optimize import linprog

def _project_capped_simplex(v, N):
    M = N / 2.0
    v = np.asarray(v, dtype=np.float64)
    if v.shape != (N,) or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        lam = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - lam, 0.0, 1.0)))
        if s > M:
            lo = lam
        else:
            hi = lam
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    residual = M - float(np.sum(y))
    for _ in range(2):
        if abs(residual) < 1e-13:
            break
        for i in range(N):
            if residual > 0 and y[i] < 1.0:
                d = min(residual, 1.0 - y[i])
                y[i] += d
                residual -= d
            elif residual < 0 and y[i] > 0.0:
                d = min(-residual, y[i])
                y[i] -= d
                residual += d
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(float(np.sum(y)) - M) > 1e-9 * max(1, N):
        return None
    return y

def _score(h):
    N = len(h)
    if N < 2 or not np.all(np.isfinite(h)):
        return float('inf')
    try:
        return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / N)
    except Exception:
        return float('inf')

def _compute_RG(h):
    N = len(h)
    alpha = 2.0 / N
    R = np.correlate(h, 1.0 - h, mode='full') * alpha
    G = np.zeros((2 * N - 1, N), dtype=np.float64)
    for q in range(-(N - 1), N):
        row = q + N - 1
        j_lo = max(0, -q)
        j_hi = min(N, N - q)
        for j in range(j_lo, j_hi):
            G[row, j + q] += alpha * (1.0 - h[j])
            G[row, j] -= alpha * h[j + q]
    return R, G

def _solve_lp(x, R, G, delta):
    N = len(x)
    c = np.zeros(N + 1)
    c[N] = 1.0
    A_ub = np.hstack([G, -np.ones((2 * N - 1, 1))])
    b_ub = -R
    A_eq = np.zeros((1, N + 1))
    A_eq[0, :N] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-x[i], -delta), min(1.0 - x[i], delta)) for i in range(N)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'display': False, 'time_limit': 5})
    except Exception:
        return None
    if res is None or res.status != 0 or res.x is None:
        return None
    u = np.array(res.x[:N], dtype=np.float64)
    if not np.all(np.isfinite(u)):
        return None
    return u

def _make_seed(s):
    try:
        x = int(s)
        return x % (2**32) if x < 0 else x
    except Exception:
        return 42
