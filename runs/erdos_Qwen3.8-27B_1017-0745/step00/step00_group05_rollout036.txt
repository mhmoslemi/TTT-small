Output[n-1-k] = sum_i h[i] * (1-h[i-k]) for k > 0

Actually, the convention for numpy.correlate is:
c_k = sum_n a_{n+k} * v_n

where the sum is over all n where both indices are valid.

So for a = h, v = 1-h:
c_k = sum_{valid n} h[n+k] * (1-h[n])

This corresponds to the integral ∫ h(x+k)(1-h(x)) dx which equals ∫ h(x)(1-h(x-k)) dx by substitution.

The max over all lags covers both positive and negative shifts, so we get the correct C₅.

    stag_thresh = 35 if n <= 128 else (18 if n <= 512 else 8)
    max_restart = 2
    restarts = 0
    stagnant = 0
    mu_F = 0.5
    mu_CR = 0.7
    incumbent = fallback.copy()
    inc_score = exact_score(incumbent, dx)
    evals += 1
    pop = [incumbent.copy()]
    pscores = [inc_score]
    init_cands = []
    if usable:
        init_cands.append(parent.copy())
        init_cands.append(parent[::-1].copy())
        init_cands.append(1.0 - parent.copy())
        for sig in [0.03, 0.08, 0.18, 0.30]:
            init_cands.append(parent + sig * rng.standard_normal(n))
    init_cands.append(np.full(n, 0.5))
    init_cands.append(make_block(n, target))
    for _ in range(3):
        init_cands.append(gen_profile(rng, n, parent, usable))
    for ic in init_cands:
        if evals + 1 > budget or len(pop) >= NP:
            break
        p = capped_simplex_project(ic, target)
        if p is None or not is_feasible(p, n, target):
            continue
        dup = False
        for ex in pop:
            if np.max(np.abs(p - ex)) < 1e-8:
                dup = True
                break
        if dup:
            continue
        s = exact_score(p, dx)
        evals += 1
        pop.append(p)
        pscores.append(s)
        if s < inc_score - 1e-12:
            incumbent = p.copy()
            inc_score = s
    tries = 0
    while len(pop) < NP and tries < 10 * NP and evals + 1 <= budget:
        tries += 1
        p = capped_simplex_project(gen_profile(rng, n, parent, usable), target)
        if p is None or not is_feasible(p, n, target):
            continue
        dup = False
        for ex in pop:
            if np.max(np.abs(p - ex)) < 1e-8:
                dup = True
                break
        if dup:
            continue
        s = exact_score(p, dx)
        evals += 1
        pop.append(p)
        pscores.append(s)
        if s < inc_score - 1e-12:
            incumbent = p.copy()
            inc_score = s
    while len(pop) < NP:
        pop.append(pop[0].copy())
        pscores.append(pscores[0])
    archive = []
    while evals + NP <= budget:
        order = np.argsort(pscores)
        pbest_count = max(2, math.ceil(0.15 * NP))
        pbest = list(order[:pbest_count])
        new_pop = [x.copy() for x in pop]
        new_scores = list(pscores)
        gen_archive = []
        succ_F, succ_CR, succ_d = [], [], []
        gen_improved = False
        for i in range(NP):
            if evals + 1 > budget:
                break
            F_i = mu_F
            for _ in range(8):
                F_i = mu_F + 0.1 * rng.standard_cauchy()
                if F_i > 0 and np.isfinite(F_i):
                    break
            F_i = float(np.clip(F_i, 0.05, 1.0))
            CR_i = float(np.clip(rng.normal(mu_CR, 0.1), 0.0, 1.0))
            x_t = new_pop[i]
            if rng.random() < 0.8:
                pi = pbest[rng.integers(len(pbest))]
                r1 = rng.integers(NP)
                while r1 == i:
                    r1 = rng.integers(NP)
                r2 = rng.integers(NP)
                while r2 == i or r2 == r1:
                    r2 = rng.integers(NP)
                v = x_t + F_i * (new_pop[pi] - x_t) + F_i * (new_pop[r1] - new_pop[r2])
            else:
                r1 = rng.integers(NP)
                while r1 == i:
                    r1 = rng.integers(NP)
                r2 = rng.integers(NP)
                while r2 == i or r2 == r1:
                    r2 = rng.integers(NP)
                r3 = rng.integers(NP)
                while r3 == i or r3 == r1 or r3 == r2:
                    r3 = rng.integers(NP)
                v = new_pop[r1] + F_i * (new_pop[r2] - new_pop[r3])
            if not np.all(np.isfinite(v)):
                continue
            jrand = int(rng.integers(n))
            mask = rng.random(n) < CR_i
            mask[jrand] = True
            trial = np.where(mask, v, x_t)
            p = capped_simplex_project(trial, target)
            if p is None or not is_feasible(p, n, target):
                continue
            s = exact_score(p, dx)
            evals += 1
            if s < new_scores[i] - 1e-12:
                gen_archive.append(new_pop[i])
                new_pop[i] = p
                new_scores[i] = s
                succ_F.append(F_i)
                succ_CR.append(CR_i)
                succ_d.append(pscores[i] - s)
                if s < inc_score - 1e-12:
                    incumbent = p.copy()
                    inc_score = s
                    gen_improved = True
        pop = new_pop
        pscores = new_scores
        archive.extend(gen_archive)
        while len(archive) > NP:
            archive.pop(int(rng.integers(len(archive))))
        if succ_d:
            tot = sum(succ_d)
            if tot > 1e-15:
                w = [d / tot for d in succ_d]
                eh_F = sum(w[k] * succ_F[k] * succ_F[k] for k in range(len(w)))
                em_F = sum(w[k] * succ_F[k] for k in range(len(w)))
                if em_F > 1e-15:
                    mu_F = float(np.clip(0.9 * mu_F + 0.1 * eh_F / em_F, 0.05, 0.95))
                mu_CR = float(np.clip(0.9 * mu_CR + 0.1 * sum(w[k] * succ_CR[k] for k in range(len(w))), 0.05, 0.95))
        if gen_improved:
            stagnant = 0
        else:
            stagnant += 1
        if inc_score <= 0.3808:
            break
        if stagnant >= stag_thresh and restarts < max_restart:
            if evals + NP <= budget:
                elite = max(2, NP // 4)
                order = np.argsort(pscores)
                keep = list(order[:elite])
                if not any(is_feasible(pop[k], n, target) and abs(pscores[k] - inc_score) < 1e-10 for k in keep):
                    keep[-1] = int(np.argmin([pscores[k] for k in keep]))
                    pop[keep[-1]] = incumbent.copy()
                    pscores[keep[-1]] = inc_score
                sigs = [0.02, 0.06, 0.15, 0.30]
                si = 0
                for k in range(NP):
                    if k in [int(x) for x in keep]:
                        continue
                    if si % 2 == 0:
                        ph = incumbent + sigs[si % len(sigs)] * rng.standard_normal(n)
                    else:
                        ph = gen_profile(rng, n, parent, usable)
                    si += 1
                    p = capped_simplex_project(ph, target)
                    if p is None or not is_feasible(p, n, target):
                        p = incumbent.copy()
                    pop[k] = p
                    s = exact_score(p, dx)
                    evals += 1
                    pscores[k] = s
                    if s < inc_score - 1e-12:
                        incumbent = p.copy()
                        inc_score = s
                archive = []
                mu_F = 0.5
                mu_CR = 0.7
                stagnant = 0
                restarts += 1
    fin = capped_simplex_project(incumbent, target)
    if fin is None or not is_feasible(fin, n, target):
        fin = fallback.copy()
    final_score = exact_score(fin, dx)
    if not np.isfinite(final_score):
        fin = np.full(n, 0.5)
        final_score = exact_score(fin, dx)
    return fin, float(final_score), int(n)
