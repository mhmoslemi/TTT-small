            L = N - d
            q[idx] = np.sum(h[0:L] * (1.0 - h[d:d+L]))
    return q

    sh = np.sum(np.exp((r - rmax) / tau))
    phi = rmax + tau * math.log(sh)
    p = np.exp((r - rmax) / tau) / sh
    grad = np.zeros(N)
    for k in range(-(N - 1), N):
        w = dx * p[k + N - 1]
        if w < 1e-18:
            continue
        if k >= 0:
            L = N - k
            grad[k:k+L] += w * (1.0 - h[0:L])
            grad[0:L] -= w * h[k:k+L]
        else:
            d = -k
            L = N - d
            grad[0:L] += w * (1.0 - h[d:d+L])
            grad[d:d+L] -= w * h[0:L]
    return phi, grad

def smooth_value(h, tau, N):
    dx = 2.0 / N
    r = dx * lag_values(h, N)
    rmax = float(np.max(r))
    return rmax + tau * math.log(np.sum(np.exp((r - rmax) / tau)))

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / N)

def optimize_start(start, N, inc_h, inc_score, iters_per_tau):
    taus = [0.08, 0.02, 0.005, 0.001, 0.0002, 0.00004, 0.000008]
    h = start.copy()
    gb_h, gb_s = inc_h, inc_score
    for tau in taus:
        sg_h, sg_s = h.copy(), score_exact(h, N)
        alpha = 1.0
        for _ in range(iters_per_tau):
            phi, g = smooth_value_grad(h, tau, N)
            if not np.all(np.isfinite(g)):
                break
            acc, d_inf = False, 0.0
            for _ in range(24):
                y = project_box_sum(h - alpha * g, N)
                d = y - h
                d_inf = float(np.max(np.abs(d)))
                if d_inf < 1e-10:
                    break
                gd = float(np.dot(g, d))
                ys = score_exact(y, N)
                if ys < sg_s:
                    sg_h, sg_s = y.copy(), ys
                if ys < gb_s:
                    gb_h, gb_s = y.copy(), ys
                if gd < 0 and smooth_value(y, tau, N) <= phi + 1e-4 * gd + 1e-14:
                    h, alpha, acc = y, min(1.4 * alpha, 10.0), True
                    break
                alpha *= 0.5
                if alpha < 1e-8:
                    break
            if not acc:
                break
            if d_inf < 1e-8:
                break
        h = sg_h.copy()
    return gb_h, gb_s

def run(seed=42, **kwargs):
    N = None
    try:
        parent = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if parent.size > 0 and np.all(np.isfinite(parent)):
            N = int(parent.size)
    except Exception:
        pass
    if N is None:
        try:
            t = int(kwargs.get('n_points', 70))
            if 1 <= t <= 999:
                N = t
        except Exception:
            N = 70
    parent_valid = None
    try:
        parent2 = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if parent2.size == N and np.all(np.isfinite(parent2)) \
                and np.all(parent2 >= 0) and np.all(parent2 <= 1) \
                and abs(np.sum(parent2) - N / 2.0) < 1e-6:
            parent_valid = parent2.copy()
    except Exception:
        pass
    if parent_valid is not None:
        baseline = parent_valid
    else:
        baseline = np.full(N, 0.5)
    if N == 1:
        hv = np.array([0.5])
        return hv, score_exact(hv, 1), 1
    if N <= 100:
        nst, itp = 8, 60
    elif N <= 200:
        nst, itp = 6, 40
    elif N <= 500:
        nst, itp = 4, 20
    else:
        nst, itp = 3, 8
    rng = np.random.RandomState(int(seed) if seed is not None else 42)
    starts = [baseline]
    starts.append(project_box_sum(baseline + 0.20 * rng.randn(N), N))
    starts.append(project_box_sum(rng.uniform(0.0, 1.0, N), N))
    R = min(6, (N - 1) // 2)
    if R > 0:
        a = np.arange(N, dtype=np.float64)
        fz = np.full(N, 0.5)
        for r in range(1, R + 1):
            amp = 0.12 / r
            ar, br = rng.randn(), rng.randn()
            fz += amp * (ar * np.sin(2.0 * np.pi * r * a / N) + br * np.cos(2.0 * np.pi * r * a / N))
        starts.append(project_box_sum(fz, N))
    else:
        starts.append(np.full(N, 0.5))
    if parent_valid is not None:
        starts.append(project_box_sum(1.0 - baseline, N))
    else:
        starts.append(project_box_sum(np.full(N, 0.5) + 0.4 * rng.randn(N), N))
    starts.append(np.full(N, 0.5))
    if nst > 7:
        starts.append(project_box_sum(baseline + 0.08 * rng.randn(N), N))
    if nst > 8:
        starts.append(project_box_sum(baseline + 0.40 * rng.randn(N), N))
    starts = starts[:nst]
    inc = baseline.copy()
    inc_s = score_exact(inc, N)
    if not np.isfinite(inc_s):
        inc = np.full(N, 0.5)
        inc_s = score_exact(inc, N)
    for start in starts:
        try:
            st = np.asarray(start, dtype=np.float64).ravel()
            if st.shape[0] != N or not np.all(np.isfinite(st)):
                continue
            st = project_box_sum(st, N)
            ss = score_exact(st, N)
            if ss < inc_s:
                inc, inc_s = st.copy(), ss
            bh, bs = optimize_start(st, N, inc, inc_s, itp)
            if bs < inc_s:
                inc, inc_s = bh, bs
        except Exception:
            continue
    fin = project_box_sum(inc, N)
    fs = score_exact(fin, N)
    if not np.isfinite(fs):
        fin = np.full(N, 0.5)
        fs = score_exact(fin, N)
    return fin, fs, int(N)
