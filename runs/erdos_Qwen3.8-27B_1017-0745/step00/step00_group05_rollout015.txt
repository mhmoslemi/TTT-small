def capped_simplex_project(v, target):

            mid = 0.5 * (lo + hi)
            y = np.clip(v - mid, 0.0, 1.0)
            if np.sum(y) > target:
                lo = mid
            else:
                hi = mid
        y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

        return None
    return y

def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    if not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr))

def is_feasible(h, n, target):
    return (h is not None and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= 0.0) and np.all(h <= 1.0) and abs(np.sum(h) - target) <= 1e-10)

    b = np.zeros(n)

        sigma = rng.uniform(0.05, 0.40)
        v = 0.5 + rng.normal(0, sigma, n)
    else:
        v = rng.normal(0.5, 0.15, n)
