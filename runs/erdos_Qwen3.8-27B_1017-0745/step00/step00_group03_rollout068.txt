def project_box_sum(z, N):

    z = np.asarray(z, dtype=np.float64)

def score_exact(h, N):
    if h.ndim != 1 or h.shape[0] != N or not np.all(np.isfinite(h)):
        return np.inf
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / N)

def smooth_value_grad(h, tau, N):
    M = 2 * N - 1
    q = np.zeros(M)
    for k in range(-(N - 1), N):
        idx = k + N - 1
        if k >= 0:
            L = N - k
            q[idx] = np.dot(h[k:k + L], 1.0 - h[0:L])
        else:
            d = -k
            L = N - d
            q[idx] = np.dot(h[0:L], 1.0 - h[d:d + L])
    r = (2.0 / N) * q
    rmax = r.max()
    expts = np.exp((r - rmax) / tau)
    s = expts.sum()
    p = expts / s
    phi = rmax + tau * np.log(s)
    grad = np.zeros(N)
    for k in range(-(N - 1), N):
        idx = k + N - 1
        w = (2.0 / N) * p[idx]
        if w < 1e-15:
            continue
        if k >= 0:
            L = N - k
            grad[k:k + L] += w * (1.0 - h[0:L])
            grad[0:L] -= w * h[k:k + L]
        else:
            d = -k
            L = N - d
            grad[0:L] += w * (1.0 - h[d:d + L])
            grad[d:d + L] -= w * h[0:L]
    if not np.all(np.isfinite(grad)):
        return np.inf, np.zeros(N)
    return phi, grad

def smooth_value(h, tau, N):
    M = 2 * N - 1
    q = np.zeros(M)
    for k in range(-(N - 1), N):
        idx = k + N - 1
        if k >= 0:
            L = N - k
            q[idx] = np.dot(h[k:k + L], 1.0 - h[0:L])
        else:
            d = -k
            L = N - d
            q[idx] = np.dot(h[0:L], 1.0 - h[d:d + L])
    r = (2.0 / N) * q
    rmax = r.max()
    s = np.exp((r - rmax) / tau).sum()
    return rmax + tau * np.log(s)

def optimize_start(h0, N, inc_h, inc_score, iters):
    h = h0.copy()
    temps = [0.08, 0.02, 0.005, 0.001, 0.0002, 0.00004, 8e-6]
    for tau in temps:
        stage_best_h = h.copy()
        stage_best_score = score_exact(h, N)
        for it in range(iters):
            phi, grad = smooth_value_grad(h, tau, N)
            if not np.isfinite(phi) or not np.all(np.isfinite(grad)):
                break
            alpha = 1.0
            accepted = False
            for _ in range(24):
                y = project_box_sum(h - alpha * grad, N)
                d = y - h
                if np.max(np.abs(d)) < 1e-10:
                    break
                gd = np.dot(grad, d)
                if gd >= 0:
                    alpha *= 0.5
                    if alpha < 1e-8:
                        break
                    continue
                phi_y = smooth_value(y, tau, N)
                if phi_y <= phi + 1e-4 * gd + 1e-14:
                    h = y
                    accepted = True
                    alpha = min(1.4 * alpha, 10.0)
                    break
                else:
                    alpha *= 0.5
                    if alpha < 1e-8:
                        break
            # Even if not accepted by smooth criterion, check exact score
            y_exact = score_exact(h, N)
            if y_exact < inc_score:
                inc_score = y_exact
                inc_h = h.copy()
            if y_exact < stage_best_score:
                stage_best_score = y_exact
                stage_best_h = h.copy()
            if accepted and np.max(np.abs(h - stage_best_h)) < 1e-8:
                break
        h = stage_best_h
    # After all temps, check final
    final_score = score_exact(h, N)
    if final_score < inc_score:
        inc_score = final_score
        inc_h = h.copy()
    return inc_h, inc_score

import numpy as np

def project_box_sum(z, N):
    if z is None:
        return np.full(N, 0.5)
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or z.shape[0] != N or not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    m = N / 2.0
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    e = m - h.sum()
    if e > 1e-13:
        for i in range(N):
            if e < 1e-15:
                break
            adj = min(e, 1.0 - h[i])
            h[i] += adj
            e -= adj
    elif e < -1e-13:
        for i in range(N):
            if -e < 1e-15:
                break
            adj = min(-e, h[i])
            h[i] -= adj
            e += adj
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(h.sum() - m) > 1e-10 * max(1, N):
        return np.full(N, 0.5)
    return h

def score_exact(h, N):
    if h.ndim != 1 or h.shape[0] != N or not np.all(np.isfinite(h)):
        return np.inf
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / N)

def _lag_q(h, N):
    M = 2 * N - 1
    q = np.zeros(M)
    for k in range(-(N - 1), N):
        idx = k + N - 1
        if k >= 0:
            L = N - k
            q[idx] = np.dot(h[k:k + L], 1.0 - h[0:L])
        else:
            d = -k
            L = N - d
            q[idx] = np.dot(h[0:L], 1.0 - h[d:d + L])
    return q

def smooth_value_grad(h, tau, N):
    q = _lag_q(h, N)
    r = (2.0 / N) * q
    rmax = r.max()
    expts = np.exp((r - rmax) / tau)
    s = expts.sum()
    p = expts / s
    phi = float(rmax + tau * np.log(s))
    grad = np.zeros(N)
    M = 2 * N - 1
    for k in range(-(N - 1), N):
        idx = k + N - 1
        w = (2.0 / N) * p[idx]
        if w < 1e-15:
            continue
        if k >= 0:
            L = N - k
            grad[k:k + L] += w * (1.0 - h[0:L])
            grad[0:L] -= w * h[k:k + L]
        else:
            d = -k
            L = N - d
            grad[0:L] += w * (1.0 - h[d:d + L])
            grad[d:d + L] -= w * h[0:L]
    if not np.all(np.isfinite(grad)):
        return np.inf, np.zeros(N)
    return phi, grad

def smooth_value(h, tau, N):
    q = _lag_q(h, N)
    r = (2.0 / N) * q
    rmax = r.max()
    s = np.exp((r - rmax) / tau).sum()
    return float(rmax + tau * np.log(s))

                    alpha *= 0.5

                        break
                    continue
                py = smooth_value(y, tau, N)

                    alpha = min(1.4 * alpha, 10.0)

                alpha *= 0.5
