    fallback = parent.copy() if parent_feasible else uniform.copy()
    if n <= 128:
        budget = 12000
    elif n <= 512:
        budget = 3500
    else:
        budget = 900
    eval_count = [0]
    NP = min(64, max(16, 2 * n))
    pop = []
    scores = []
    if parent_feasible:
        add_to_pop(parent, pop, scores, eval_count, n, target, dx)
        add_to_pop(parent[::-1].copy(), pop, scores, eval_count, n, target, dx)
        add_to_pop(1.0 - parent, pop, scores, eval_count, n, target, dx)
        for sig in [0.03, 0.08, 0.18, 0.30]:
            add_to_pop(parent + sig * rng.standard_normal(n), pop, scores, eval_count, n, target, dx)
    add_to_pop(uniform, pop, scores, eval_count, n, target, dx)
    add_to_pop(make_block_profile(n), pop, scores, eval_count, n, target, dx)
    attempts = 0
    while len(pop) < NP and attempts < 10 * NP:
        attempts += 1
        add_to_pop(gen_profile(n, rng, parent), pop, scores, eval_count, n, target, dx)
    while len(pop) < NP:
        pop.append(pop[0].copy())
        scores.append(exact_score(pop[-1], dx))
        eval_count[0] += 1
    bi = int(np.argmin(scores))
    incumbent = pop[bi].copy()
    incumbent_score = scores[bi]
    mu_F, mu_CR = 0.5, 0.7
    archive = []
    stal_count = 0
    n_restarts = 0
    stag_thresh = 35 if n <= 128 else (18 if n <= 512 else 8)
    while eval_count[0] + NP <= budget:
        order = np.argsort(scores)
        pbest_n = max(2, int(ceil(0.15 * NP)))
        pbest = order[:pbest_n]
        new_pop = [pop[i].copy() for i in range(NP)]
        new_scores = list(scores)
        new_archive = list(archive)
        succeeded = []
        for i in range(NP):
            F_i = mu_F
            for _ in range(8):
                F_i = rng.cauchy(mu_F, 0.1)
                if np.isfinite(F_i) and F_i > 0:
                    break
            F_i = min(max(F_i, 0.05), 1.0)
            CR_i = min(max(rng.normal(mu_CR, 0.1), 0.0), 1.0)
            if rng.random() < 0.8:
                p_idx = pbest[int(rng.integers(len(pbest)))]
                cands = [x for x in range(NP) if x != i and x != p_idx]
                r1 = cands[int(rng.integers(len(cands)))]
                pool = [x for x in range(NP + len(new_archive)) if x != i and x != r1]
                r2 = pool[int(rng.integers(len(pool)))]
                x_p, x_r1 = pop[p_idx], pop[r1]
                x_r2 = pop[r2] if r2 < NP else new_archive[r2 - NP]
                v = pop[i] + F_i * (x_p - pop[i]) + F_i * (x_r1 - x_r2)
            else:
                cands = [x for x in range(NP) if x != i]
                r1 = cands.pop(int(rng.integers(len(cands))))
                r2 = cands.pop(int(rng.integers(len(cands))))
                cands2 = [x for x in cands]
                r3 = cands2[int(rng.integers(len(cands2)))]
                v = pop[r1] + F_i * (pop[r2] - pop[r3])
            jrand = int(rng.integers(n))
            mask = rng.random(n) < CR_i
            mask[jrand] = True
            trial = np.where(mask, v, pop[i])
            proj = capped_simplex_project(trial, target)
            if proj is None or not is_feasible(proj, n, target):
                continue
            s = exact_score(proj, dx)
            eval_count[0] += 1
            if s < new_scores[i] - 1e-12:
                if s < incumbent_score - 1e-12:
                    incumbent = proj.copy()
                    incumbent_score = s
                    stal_count = 0
                new_pop[i] = proj
                new_scores[i] = s
                new_archive.append(pop[i].copy())
                succeeded.append((F_i, CR_i, new_scores[i] * 0 + scores[i] - s))
        pop = new_pop
        scores = new_scores
        archive = new_archive
        while len(archive) > NP:
            archive.pop(int(rng.integers(len(archive))))
        if succeeded:
            total_d = sum(d for _, _, d in succeeded)
            if total_d > 1e-15:
                w = [d / total_d for _, _, d in succeeded]
                Fs = sum(w[i] * succeeded[i][0] for i in range(len(succeeded)))
                F2s = sum(w[i] * succeeded[i][0] ** 2 for i in range(len(succeeded)))
                if Fs > 0:
                    mu_F = min(max(0.9 * mu_F + 0.1 * F2s / Fs, 0.05), 0.95)
                CRm = sum(w[i] * succeeded[i][1] for i in range(len(succeeded)))
                mu_CR = min(max(0.9 * mu_CR + 0.1 * CRm, 0.05), 0.95)
        if not succeeded:
            stal_count += 1
        if incumbent_score <= 0.3808:
            break
        if stal_count >= stag_thresh and n_restarts < 2:
            ec = max(2, NP // 4)
            o2 = np.argsort(scores)
            elite_p = [pop[o2[k]].copy() for k in range(ec)]
            elite_s = [scores[o2[k]] for k in range(ec)]
            if not any(np.max(np.abs(e - incumbent)) < 1e-10 for e in elite_p):
                wi = int(np.argmax(elite_s))
                elite_p[wi] = incumbent.copy()
                elite_s[wi] = incumbent_score
            sig_cyc = [0.02, 0.06, 0.15, 0.30]
            si = 0
            for k in range(ec, NP):
                if (k - ec) % 2 == 0:
                    v2 = incumbent + sig_cyc[si % 4] * rng.standard_normal(n)
                    si += 1
                else:
                    v2 = gen_profile(n, rng, parent)
                pr = capped_simplex_project(v2, target)
                if pr is None or not is_feasible(pr, n, target):
                    pr = incumbent.copy()
                s2 = exact_score(pr, dx)
                eval_count[0] += 1
                elite_p.append(pr)
                elite_s.append(s2)
                if s2 < incumbent_score - 1e-12:
                    incumbent = pr.copy()
                    incumbent_score = s2
            pop, scores = elite_p, elite_s
            archive = []
            mu_F, mu_CR = 0.5, 0.7
            stal_count = 0
            n_restarts += 1
    proj_f = capped_simplex_project(incumbent, target)
    if proj_f is not None and is_feasible(proj_f, n, target):
        h_final = proj_f
    else:
        h_final = fallback
    if not is_feasible(h_final, n, target):
        h_final = uniform
    c5 = exact_score(h_final, dx)
    if not np.isfinite(c5):
        h_final = uniform
        c5 = exact_score(h_final, dx)
    return h_final, float(c5), int(n)
