import numpy as np

def is_feasible(h, n):
    if h is None or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < 0) or np.any(h > 1):
        return False
    if abs(np.sum(h) - n/2.0) > 1e-10 * max(1, n):
        return False
    return True

def project_capped_simplex(z, n):
    """Project z onto {0<=h<=1, sum(h)=n/2}."""
    s = n / 2.0
    if z is None or len(z) != n or not np.all(np.isfinite(z)):
        return np.full(n, 0.5)
    z = np.clip(z.copy(), -1e12, 1e12)
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(z - mid, 0.0, 1.0)
        if h.sum() > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    # Residual correction
    r = s - h.sum()
    for _ in range(n + 2):
        if abs(r) < 1e-12 * max(1, n):
            break
        if r > 0:
            slack = 1.0 - h
            idx = np.argmax(slack)
            if slack[idx] <= 0:
                break
            amount = min(r, slack[idx])
            h[idx] += amount
            r -= amount
        else:
            slack = h
            idx = np.argmax(slack)
            if slack[idx] <= 0:
                break
            amount = min(-r, slack[idx])
            h[idx] -= amount
            r += amount
    h = np.clip(h, 0.0, 1.0)
    if not is_feasible(h, n):
        return np.full(n, 0.5)
    return h

def exact_c5(h):
    n = len(h)
    if n == 0 or not np.all(np.isfinite(h)):
        return np.inf
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def lse_value_and_weights(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = corr.max()
    a = np.exp((corr - c_max) / tau)
    total = a.sum()
    if total <= 0 or not np.isfinite(total):
        return None, None
    val = c_max + tau * np.log(total)
    w = a / total
    return val, w

def lse_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    val, P = lse_value_and_weights(h, tau)
    if P is None:
        return None, None
    U = np.convolve(P, 1.0 - h, mode='full')[n-1 : 2*n-1]
    V = np.convolve(P[::-1], h, mode='full')[n-1 : 2*n-1]
    return val, dx * (U - V)

def gradient_descent(x0, n, tau, iters, rng):
    """Projected gradient descent at fixed temperature."""
    x = x0.copy()
    val, grad = lse_gradient(x, tau)
    if val is None or not np.isfinite(val):
        return x
    alpha = 1.0
    stagnant = 0
    for it in range(iters):
        if not np.all(np.isfinite(grad)):
            break
        y = project_capped_simplex(x - alpha * grad, n)
        D = np.sum((y - x)**2)
        if np.max(np.abs(y - x)) <= 1e-9:
            break
        val_y, _ = lse_value_and_weights(y, tau)
        if val_y is None or not np.isfinite(val_y):
            alpha *= 0.5
            if alpha < 1e-12:
                break
            continue
        armijo = val_y <= val - 1e-4 * D / alpha + 1e-14 * max(1.0, abs(val))
        if armijo:

import numpy as np


def is_feasible(h, n):
    if h is None or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True


def project_capped_simplex(z, n):
    s = n / 2.0
    if z is None or len(z) != n or not np.all(np.isfinite(z)):
        return np.full(n, 0.5)
    z = np.clip(z.astype(np.float64).copy(), -1e12, 1e12)
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(z - mid, 0.0, 1.0)
        if h.sum() > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - h.sum()
    for _ in range(n + 2):
        if abs(r) < 1e-13 * max(1, n):
            break
        if r > 0:
            slack = 1.0 - h
            idx = int(np.argmax(slack))
            if slack[idx] <= 1e-15:
                break
            amt = min(r, slack[idx])
            h[idx] += amt
            r -= amt
        else:
            idx = int(np.argmax(h))
            if h[idx] <= 1e-15:
                break
            amt = min(-r, h[idx])
            h[idx] -= amt
            r += amt
    h = np.clip(h, 0.0, 1.0)
    if not is_feasible(h, n):
        return np.full(n, 0.5)
    return h


def exact_c5(h):
    n = len(h)
    if n == 0 or not np.all(np.isfinite(h)):
        return np.inf
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)


def lse_value_and_weights(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = corr.max()
    a = np.exp((corr - c_max) / tau)
    total = a.sum()
    if total <= 0 or not np.isfinite(total):
        return None, None
    val = c_max + tau * np.log(total)
    w = a / total
    if not np.all(np.isfinite(w)):
        return None, None
    return val, w


def lse_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    val, P = lse_value_and_weights(h, tau)
    if P is None:
        return None, None
    U = np.convolve(P, 1.0 - h, mode='full')[n - 1:2 * n - 1]
    V = np.convolve(P[::-1], h, mode='full')[n - 1:2 * n - 1]
    g = dx * (U - V)
    if not np.all(np.isfinite(g)):
        return None, None
    return val, g


def build_starts(n, base, seed):
    rng = np.random.default_rng(seed)
    raw = []
    raw.append(base.copy())
    raw.append(base[::-1].copy())
    raw.append((1.0 - base[::-1]).copy())
    raw.append(np.full(n, 0.5).copy())
    blk = np.zeros(n)
    half = n // 2
    blk[:half] = 1.0
    if n % 2 == 1:
        blk[half] = 0.5
    raw.append(blk.copy())
    alt = np.zeros(n)
    alt[::2] = 1.0
    raw.append(alt.copy())
    for sd in [0.05, 0.15, 0.35]:
        raw.append(base + rng.standard_normal(n) * sd)
    raw.append(rng.random(n).copy())
    projected = [project_capped_simplex(s, n) for s in raw]
    deduped = [projected[0]]
    for p in projected[1:]:
        if not any(np.max(np.abs(p - q)) < 1e-10 for q in deduped):
            deduped.append(p)
    return deduped


def gradient_descent(x0, n, tau, iters):
    x = x0.copy()
    val, grad = lse_gradient(x, tau)
    if val is None or not np.isfinite(val):
        return x
    alpha = 1.0

                val = val_y

            break
        val, grad = lse_gradient(x, tau)
        if val is None or not np.isfinite(val):
