def compute_gradient(h, p, dx):
    n = len(h)
    A = 1.0 - h
    grad = np.zeros(n)
    for d in range(n):
        # T1 contribution: P[n-1+d] * A[d:] to grad[0:n-d]
        grad[:n-d] += p[n-1+d] * A[d:]
        # T2 contribution: -P[n-1+d] * h[:n-d] to grad[d:]
        grad[d:] -= p[n-1+d] * h[:n-d]
