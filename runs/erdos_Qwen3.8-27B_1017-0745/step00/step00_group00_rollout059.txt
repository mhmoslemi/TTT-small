    for h0 in starts:
        h_new = optimize_start(h0, n, S, bounds, constraints)
        if h_new is not None:
            sc = exact_score(h_new)
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new

    sigma = 0.05
    for _ in range(4):
        pert = best_h + sigma * np.random.randn(n)
        h_p = project_box_sum(pert, S)
        if h_p is None:
            sigma = min(0.25, 1.5 * sigma)
            continue
        h_new = optimize_start(h_p, n, S, bounds, constraints)
        if h_new is not None:
            sc = exact_score(h_new)
            if sc < best_score - 1e-12:
                best_score = sc
                best_h = h_new
                sigma = max(0.01, 0.7 * sigma)
            else:
                sigma = min(0.25, 1.5 * sigma)
        else:
            sigma = min(0.25, 1.5 * sigma)

    final_h = project_box_sum(best_h, S)
    if final_h is None or not check_feasible(final_h, n):
        if parent is not None:
            final_h = parent
        else:
            final_h = project_box_sum(np.full(n, 0.5), S)
    final_score = exact_score(final_h)
    return (np.array(final_h, dtype=np.float64), final_score, n)
