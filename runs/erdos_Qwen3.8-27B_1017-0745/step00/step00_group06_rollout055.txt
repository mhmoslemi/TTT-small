    try:
        p = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if p.size >= 2 and np.all(np.isfinite(p)):
            parent = p
    except Exception:
        pass

    if parent is not None:
        n = int(parent.size)
    else:
        n = kwargs.get('n_points', 95)
        try:
            n = int(n)
        except Exception:
            n = 95
        n = max(2, min(n, 999))

    d = 2.0 / n
    m = n / 2.0
    _state['n'] = n
    _state['d'] = d
    _state['m'] = m

    h_unif = np.full(n, 0.5)
    h0 = project_feasible(parent) if parent is not None else h_unif.copy()
    inc_h = h0.copy()
    inc_score = exact_score(inc_h)

    try:
        seed = int(seed)
    except Exception:
        seed = 42
    rng = np.random.default_rng(seed)

    starts = [h0.copy(), h_unif.copy()]
    hs = np.zeros(n)
    hs[:int(m)] = 1.0
    if m != int(m):
        hs[int(m)] = m - int(m)
    starts.append(project_feasible(hs))

    idx = np.arange(n)
    for f in [1, 3, 7]:
        starts.append(project_feasible(0.5 + 0.22 * np.sin(2.0 * np.pi * f * idx / n)))

    unique = [starts[0].copy()]
    for s in starts[1:]:
        if all(np.max(np.abs(s - u)) > 1e-8 for u in unique):
            unique.append(s.copy())

    n_restarts = kwargs.get('n_restarts', 8)
    try:
        n_restarts = int(n_restarts)
    except Exception:
        n_restarts = 8
    n_restarts = max(6, min(12, n_restarts))

    base = h0 if parent is not None else h_unif
    stds = [0.04, 0.10, 0.20]
    k = 0
    while len(unique) < n_restarts and k < 100:
        std = stds[k % 3]
        noise = rng.standard_normal(n) * std
        noise = 0.5 * noise + 0.25 * np.roll(noise, 1) + 0.25 * np.roll(noise, -1)
        hr = project_feasible(base + noise)
        if all(np.max(np.abs(hr - u)) > 1e-8 for u in unique):
            unique.append(hr.copy())
        k += 1

    maxiter = kwargs.get('maxiter', 120)
    try:
        maxiter = int(maxiter)
    except Exception:
        maxiter = 120
    maxiter = max(60, min(250, maxiter))

    taus = [0.02, 0.005, 0.001, 0.0002]
    bounds = [(0.0, 1.0)] * n
    constr = [{'type': 'eq', 'fun': mass_fun, 'jac': mass_jac}]
    candidates = [(inc_h.copy(), inc_score)]

    skip_opt = n > 999
    if not skip_opt:
        for start in unique:
            h_cur = start.copy()
            sc = exact_score(h_cur)
            candidates.append((h_cur.copy(), sc))
            if sc < inc_score - 1e-12:
                inc_score = sc
                inc_h = h_cur.copy()
            for tau in taus:
                _state['tau'] = tau
                try:
                    res = minimize(smooth_obj, h_cur, jac=smooth_jac, method='SLSQP',
                                  bounds=bounds, constraints=constr,
                                  options={'ftol': 1e-10, 'maxiter': maxiter, 'disp': False})
                    if res.x is not None and len(res.x) == n and np.all(np.isfinite(res.x)):
                        h_cur = project_feasible(res.x)
                except Exception:
                    pass
                sc = exact_score(h_cur)
                candidates.append((h_cur.copy(), sc))
                if sc < inc_score - 1e-12:
                    inc_score = sc
                    inc_h = h_cur.copy()

    candidates.sort(key=_sort_key)
    top = []
    for hc, sc in candidates:
        if all(np.max(np.abs(hc - t[0])) > 1e-6 for t in top):
            top.append((hc.copy(), sc))
        if len(top) >= 3:
            break

    if not skip_opt:
        epi_bounds = [(0.0, 1.0)] * n + [(0.0, 2.1)]
        epi_mi = min(400, 2 * maxiter)
        epi_constr = [
            {'type': 'eq', 'fun': epi_mass_fun, 'jac': epi_mass_jac},
            {'type': 'ineq', 'fun': epi_ineq_fun, 'jac': epi_ineq_jac}
        ]
        for hc, sc in top:
            z0 = np.concatenate([hc, [sc + 1e-6]])
            try:
                res = minimize(epi_obj, z0, jac=epi_jac, method='SLSQP',
                              bounds=epi_bounds, constraints=epi_constr,
                              options={'ftol': 1e-10, 'maxiter': epi_mi, 'disp': False})
                if res.x is not None and len(res.x) == n + 1 and np.all(np.isfinite(res.x)):
                    h_new = project_feasible(res.x[:n])
                    sc_new = exact_score(h_new)
                    if sc_new < inc_score - 1e-12:
                        inc_score = sc_new
                        inc_h = h_new.copy()
            except Exception:
                pass

    inc_h = project_feasible(inc_h)
    if not (np.all(inc_h >= -1e-9) and np.all(inc_h <= 1.0 + 1e-9) and abs(np.sum(inc_h) - m) < 1e-9 * max(1, n)):
        inc_h = project_feasible(h_unif)
    inc_score = exact_score(inc_h)
    return (inc_h.astype(np.float64), float(inc_score), n)
