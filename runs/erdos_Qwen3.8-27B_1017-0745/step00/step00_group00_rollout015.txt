def project_box_sum(v, S):
    n = len(v)
    if not np.all(np.isfinite(v)) or np.any(np.abs(v) > 1e12):
        return None
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    res = S - np.sum(y)
    if res > 0:
        for i in range(n):
            if res <= 0:
                break
            add = min(res, 1.0 - y[i])
            y[i] += add
            res -= add
    elif res < 0:
        for i in range(n):
            if res >= 0:
                break
            sub = min(-res, y[i])
            y[i] -= sub
            res += sub
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    if abs(np.sum(y) - S) > 1e-10 * max(1, n):
        return None
    y = np.clip(y, 0.0, 1.0)
    return y

            c2 = fftconvolve(p[::-1], h, mode='full')
            if not (np.all(np.isfinite(c1)) and np.all(np.isfinite(c2))):
                c1 = np.convolve(p, 1.0 - h, mode='full')
                c2 = np.convolve(p[::-1], h, mode='full')
        except Exception:
            c1 = np.convolve(p, 1.0 - h, mode='full')
            c2 = np.convolve(p[::-1], h, mode='full')
    grad = (c1[n - 1:2 * n - 1] - c2[n - 1:2 * n - 1]) / n
    if not np.all(np.isfinite(grad)):
        grad = None
    if not np.isfinite(F):
        F = None
    return F, grad, exact

def make_starts(parent, n, rng):
    S = n / 2.0
    starts = []
    if parent is not None:
        starts.append(parent.copy())
    if n <= 100:
        sigmas = [0.02, 0.05, 0.10, 0.18, 0.30]
        for sig in sigmas:
            v = parent + sig * rng.standard_normal(n) if parent is not None else sig * rng.standard_normal(n)
           proj = project_box_sum(v, S)
            if proj is not None:
                starts.append(proj)
        for _ in range(3):
            v = rng.uniform(0.0, 1.0, size=n)
            proj = project_box_sum(v, S)
            if proj is not None:
                starts.append(proj)
    elif n <= 300:
        sigmas = [0.03, 0.10, 0.22]
        for sig in sigmas:
            v = parent + sig * rng.standard_normal(n) if parent is not None else sig * rng.standard_normal(n)
            proj = project_box_sum(v, S)
            if proj is not None:
                starts.append(proj)
        for _ in range(2):
            v = rng.uniform(0.0, 1.0, size=n)
            proj = project_box_sum(v, S)
            if proj is not None:
                starts.append(proj)
    else:
        if parent is not None:
            v = parent + 0.10 * rng.standard_normal(n)
            proj = project_box_sum(v, S)
            if proj is not None:
                starts.append(proj)
        v = rng.uniform(0.0, 1.0, size=n)
        proj = project_box_sum(v, S)
        if proj is not None:
            starts.append(proj)
    if parent is None and n <= 100:
        c = np.full(n, 0.5)
        starts.insert(0, project_box_sum(c, S))
        idx = np.arange(n)
        for A, m, ph in [(0.20, 1, 0.31), (0.35, 2, 0.77), (0.45, 3, 1.13)]:
            cv = 0.5 + A * np.cos(2.0 * np.pi * m * idx / n + ph)
            proj = project_box_sum(cv, S)
            if proj is not None:
                starts.append(proj)
    elif parent is None and n <= 300:
        c = np.full(n, 0.5)
        starts.insert(0, project_box_sum(c, S))
        idx = np.arange(n)
        for A, m, ph in [(0.20, 1, 0.31), (0.35, 2, 0.77), (0.45, 3, 1.13)]:
            cv = 0.5 + A * np.cos(2.0 * np.pi * m * idx / n + ph)
            proj = project_box_sum(cv, S)
            if proj is not None:
                starts.append(proj)
    elif parent is None:
        c = np.full(n, 0.5)
        starts.insert(0, project_box_sum(c, S))

        proj = project_box_sum(cv, S)

    return starts

def optimize_stage(h, n, tau, max_iter, budget):
    cnt = [0]
    F, grad, exact = smooth_eval(h, tau)
    cnt[0] += 1
    if F is None or grad is None:
        return h, cnt[0]
    stall = 0
    for it in range(max_iter):
        if cnt[0] >= budget:
            break
        gmax = np.max(np.abs(grad))
        if gmax < 1e-13:
            break
        alpha = min(1e8, 0.08 / gmax)
        accepted = False
        for bt in range(12):
            if cnt[0] >= budget:
                break
            y = project_box_sum(h - alpha * grad, n / 2.0)
            if y is None:
                alpha *= 0.5
                continue
            d = y - h
            dirn = np.dot(grad, d)
            if not np.isfinite(dirn) or dirn >= 0:
                alpha *= 0.5
                continue
            Fy, gy, ey = smooth_eval(y, tau)
            cnt[0] += 1
            if Fy is not None and Fy <= F + 1e-4 * dirn + 1e-13 * max(1.0, abs(F)):
                dec = F - Fy
                h = y
                F = Fy
                grad = gy
                accepted = True
                if dec <= 1e-12 * max(1.0, abs(F)):
                    stall += 1
                else:
                    stall = 0
                if stall >= 4:
                    break
                if np.max(np.abs(d)) < 1e-8:
                    break
                break
            alpha *= 0.5
        if accepted and stall >= 4:
            break
        if not accepted:
            break
    return h, cnt[0]

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent = None
    n = 45
    S = n / 2.0
    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        try:
            arr = np.array(ihv, dtype=np.float64).ravel()
            if arr.size >= 2 and arr.size <= 999 and np.all(np.isfinite(arr)) and np.max(np.abs(arr)) <= 1e12:
                n = arr.size
                S = n / 2.0
                proj = project_box_sum(arr, S)
                if proj is not None:
                    parent = proj
        except Exception:
            pass
    if parent is None:
        n = 45
        S = n / 2.0
        parent = np.full(n, 0.5)
    best_h = parent.copy()
    best_score = exact_score(best_h)
    if best_score is None:
        best_score = 1.0
    budget_total = min(50000, max(100, int(1e8 / (n * n))))
    tau_list = [0.04, 0.015, 0.006, 0.002, 0.0007, 0.00025, 0.00008]
    if n <= 100:
        max_iter = 60
    elif n <= 300:
        max_iter = 25
    else:
        max_iter = 8
    starts = make_starts(parent, n, rng)
    eval_count = 0
    for st in starts:
        if eval_count >= budget_total:
            break
        sc = exact_score(st)
        if sc is not None and sc < best_score:
            best_h = st.copy()
            best_score = sc
        h_curr = st.copy()
        for tau in tau_list:
            if eval_count >= budget_total:
                break
            h_curr, used = optimize_stage(h_curr, n, tau, max_iter, budget_total - eval_count)
            eval_count += used
            sc = exact_score(h_curr)
            if sc is not None and sc < best_score:
                best_h = h_curr.copy()
                best_score = sc
    final = project_box_sum(best_h, S)
    if final is not None:
        best_h = final
    final_score = exact_score(best_h)
    if final_score is None:
        best_h = parent.copy()
        final_score = exact_score(best_h)
    if final_score is None:
        best_h = np.full(n, 0.5)
        final_score = exact_score(best_h)
    if final_score is None:
        final_score = 0.5
    return (best_h.astype(np.float64), float(final_score), int(n))
