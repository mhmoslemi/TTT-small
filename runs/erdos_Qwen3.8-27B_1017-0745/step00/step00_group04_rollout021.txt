import numpy as np
import math

def is_feasible(h, n):
    if len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-15) or np.any(h > 1 + 1e-15):
        return False
    S = n / 2.0
    if abs(float(np.sum(h)) - S) > 1e-10:
        return False
    if abs(math.fsum(float(x) for x in h) - S) > 1e-10:
        return False
    return True

def project_capped_simplex(v, n):
    S = n / 2.0
    v = np.clip(np.asarray(v, dtype=np.float64), -1e6, 1e6)
    if len(v) != n or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(v - mid, 0.0, 1.0)
        if float(np.sum(h)) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    res = math.fsum(float(x) for x in h) - S
    if abs(res) > 1e-12:
        delta = -res
        for i in range(n):
            if delta > 0:
                add = min(delta, 1.0 - h[i])
                if add > 1e-15:
                    h[i] += add
                    delta -= add
            else:
                sub = min(-delta, h[i])
                if sub > 1e-15:
                    h[i] -= sub
                    delta += sub
            if abs(delta) < 1e-12:
                break
    h = np.clip(h, 0.0, 1.0)
    if not is_feasible(h, n):
        return None
    return h

def exact_score(h, n):
    if not is_feasible(h, n):
        return None
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    score = float(np.max(corr * dx))
    if not math.isfinite(score):
        return None
    return score

def make_candidate(idx, n, rng, parent_seed):
    S = n / 2.0
    if idx == 0:
        return np.full(n, 0.5, dtype=np.float64)
    if idx == 1:
        h = np.zeros(n, dtype=np.float64)

        if n % 2 == 1:
            h[nf] = 0.5
        return h

        return np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n)], dtype=np.float64)

        t = np.arange(n, dtype=np.float64)

    if idx == 5 and parent_seed is not None:

    if parent_seed is not None and idx < 20:

        z = rng.standard_normal(n)
        z -= np.mean(z)
        return parent_seed + sigma * z
    return rng.uniform(0.0, 1.0, n)

def add_unique(pop, candidate, n):
    h = project_capped_simplex(candidate, n)
    if h is None:
        return None
    for p in pop:
        if float(np.max(np.abs(h - p))) < 1e-10:
            return None
    return h
