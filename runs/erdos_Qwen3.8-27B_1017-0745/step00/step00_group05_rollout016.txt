    lo = float(v.min()) - 1.0
    hi = float(v.max())
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    for _ in range(3):
        if abs(r) < 1e-14:
            break
        for i in range(n):
            if r > 0:
                a = min(r, 1.0 - y[i])
                y[i] += a
                r -= a
            else:
                a = min(-r, y[i])
                y[i] -= a
                r += a
    if not np.all(np.isfinite(y)) or y.min() < -1e-12 or y.max() > 1.0 + 1e-12 or abs(y.sum() - target) > 1e-10:
        return None
    return y


def is_feasible(h, n):
    if len(h) != n or not np.all(np.isfinite(h)):
        return False
    if h.min() < -1e-12 or h.max() > 1.0 + 1e-12:
        return False
    return abs(h.sum() - n / 2.0) <= 1e-10


def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return None
    c = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
    return c if np.isfinite(c) else None


def lag_data(h, n):
    q = np.correlate(h, 1.0 - h, mode='full')
    G = np.zeros((2 * n - 1, n))
    h_neg = 1.0 - h
    m = np.arange(n)
    for t_idx in range(2 * n - 1):
        t = t_idx - (n - 1)
        m1 = m - t
        m2 = m + t
        mask1 = (m1 >= 0) & (m1 < n)
        mask2 = (m2 >= 0) & (m2 < n)
        G[t_idx, mask1] += h_neg[m1[mask1]]
        G[t_idx, mask2] -= h[m2[mask2]]
    if not np.all(np.isfinite(G)) or not np.all(np.isfinite(q)):
        return None, None
    return q, G


def slp_search(h0, dx, rng, max_iter):
    n = len(h0)
    h = h0.copy()
    best = h.copy()
    best_s = exact_score(h, dx)
    rho = 0.08
    rho_min, rho_max = 1e-5, 0.25
    for _ in range(max_iter):
        if rho < rho_min:
            break
        q, G = lag_data(h, n)
        if q is None:
            break
        Q = float(np.max(q))
        A = np.zeros((2 * n - 1, n + 1))
        A[:, :n] = G
        A[:, n] = -1.0
        b_ub = -q.copy()
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = [0.0]
        bounds = []
        for i in range(n):
            bounds.append((max(-rho, -h[i]), min(rho, 1.0 - h[i])))
        bounds.append((None, None))
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'maxiter': 300})
        except Exception:
            rho = max(rho_min, rho / 2.0)
            continue
        if not res.success:
            rho = max(rho_min, rho / 2.0)
            continue
        x = res.x
        if len(x) != n + 1 or not np.all(np.isfinite(x)):
            rho = max(rho_min, rho / 2.0)
            continue
        d = x[:n]
        z = x[n]
        cand = capped_simplex_project(h + d, n / 2.0)
        if cand is None or not is_feasible(cand, n):
            rho = max(rho_min, rho / 2.0)
            continue
        s_new = exact_score(cand, dx)
        if s_new is None:
            rho = max(rho_min, rho / 2.0)
            continue
        if s_new < best_s - 1e-12:
            best = cand.copy()
            best_s = s_new
        q_new = float(np.max(np.correlate(cand, 1.0 - cand, mode='full')))
        predicted = Q - z
        actual = Q - q_new
        h_old = h.copy()
        accepted = False
        if predicted > 1e-10 * max(1.0, abs(Q)):
            ratio = actual / predicted
        else:
            ratio = 0.0
        if not np.isfinite(ratio):
            rho = max(rho_min, rho / 2.0)
            continue
        if actual > 1e-12 * max(1.0, abs(Q)):
            h = cand
            accepted = True
        if not accepted or ratio < 0.1:
            rho = max(rho_min, rho / 2.0)
        elif ratio > 0.75 and float(np.max(np.abs(cand - h_old))) >= 0.8 * rho:
            rho = min(rho_max, 1.5 * rho)
    return best


def pairwise_poll(h, dx, rng):
    n = len(h)
    if n > 256:
        return h.copy()
    cur = h.copy()
    cur_s = exact_score(cur, dx)
    if cur_s is None:
        return cur
    n_pairs = 24 if n <= 128 else 8
    radii = [0.08, 0.025, 0.008]
    for r in radii:
        for _ in range(n_pairs):
            a = int(rng.integers(n))
            b = int(rng.integers(n - 1))
            if b >= a:
                b += 1
            s_lo = max(-cur[a], cur[b] - 1.0, -r)
            s_hi = min(1.0 - cur[a], cur[b], r)
            if s_lo >= s_hi:
                continue
            trials = [s_lo, s_hi, -r, -r / 2.0, r / 2.0, r]
            seen = set()
            pair_best = None
            pair_best_s = None
            for s in trials:
                if abs(s) < 1e-14:
                    continue
                key = round(s, 12)
                if key in seen:
                    continue
                seen.add(key)
                trial = cur.copy()
                trial[a] += s
                trial[b] -= s
                proj = capped_simplex_project(trial, n / 2.0)
                if proj is None or not is_feasible(proj, n):
                    continue
                sv = exact_score(proj, dx)
                if sv is None:
                    continue
                if pair_best_s is None or sv < pair_best_s - 1e-12:
                    pair_best = proj
                    pair_best_s = sv
            if pair_best is not None and pair_best_s < cur_s - 1e-12:
                cur = pair_best
                cur_s = pair_best_s
    return cur


def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    n = 70
    parent = None
    try:
        pv = np.array(initial_h_values, dtype=np.float64).ravel()
        if pv.ndim == 1 and 2 <= len(pv) <= 999 and np.all(np.isfinite(pv)) and float(np.max(np.abs(pv))) <= 1e6:
            n = len(pv)
            parent = pv.copy()
    except Exception:
        pass
    dx = 2.0 / n
    target = float(n) / 2.0
    u = np.full(n, 0.5)
    fallback = None
    if parent is not None:
        if parent.min() >= -1e-10 and parent.max() <= 1.0 + 1e-10 and abs(parent.sum() - target) <= 1e-8:
            proj = capped_simplex_project(parent, target)
            if proj is not None and is_feasible(proj, n):
                fallback = proj.copy()
    if fallback is None:
        fallback = u.copy()
    fb_s = exact_score(fallback, dx)
    if fb_s is None:
        fb_s = float(np.max(np.correlate(u, 1.0 - u, mode='full')) * dx)
        fallback = u.copy()
    starts = []
    if parent is not None:
        proj_p = capped_simplex_project(parent, target)
        if proj_p is not None and is_feasible(proj_p, n):
            starts.append(proj_p)
        rev = capped_simplex_project(parent[::-1].copy(), target)
        if rev is not None and is_feasible(rev, n):
            starts.append(rev)
        for sc in [0.035, 0.12]:
            noisy = capped_simplex_project(parent + sc * rng.standard_normal(n), target)
            if noisy is not None and is_feasible(noisy, n):
                starts.append(noisy)
    for _ in range(2):
        r1 = capped_simplex_project(rng.random(n), target)
        if r1 is not None:
            starts.append(r1)
        r2 = capped_simplex_project(rng.beta(0.5, 0.5, size=n), target)
        if r2 is not None:
            starts.append(r2)
    starts.append(u.copy())
    unique_starts = []
    for s in starts:
        dup = False
        for u2 in unique_starts:
            if float(np.max(np.abs(s - u2))) < 1e-10:
                dup = True
                break
        if not dup:
            unique_starts.append(s)
    best_h = fallback.copy()
    best_score = fb_s
    max_iter = 50 if n <= 128 else (25 if n <= 512 else 12)
    for st in unique_starts:
        s0 = exact_score(st, dx)
        if s0 is not None and s0 < best_score - 1e-12:
            best_h = st.copy()
            best_score = s0
        if best_score <= 0.3808:
            break
        slp_res = slp_search(st, dx, rng, max_iter)
        s1 = exact_score(slp_res, dx)
        if s1 is not None and s1 < best_score - 1e-12:
            best_h = slp_res.copy()
            best_score = s1
        poll_res = pairwise_poll(slp_res, dx, rng)
        s2 = exact_score(poll_res, dx)
        if s2 is not None and s2 < best_score - 1e-12:
            best_h = poll_res.copy()
            best_score = s2
        if best_score <= 0.3808:
            break
    for sc in [0.025, 0.08]:
        if best_score <= 0.3808:
            break
        restart = capped_simplex_project(best_h + sc * rng.standard_normal(n), target)
        if restart is None or not is_feasible(restart, n):
            continue
        slp_res = slp_search(restart, dx, rng, max_iter)
        s1 = exact_score(slp_res, dx)
        if s1 is not None and s1 < best_score - 1e-12:
            best_h = slp_res.copy()
            best_score = s1
        poll_res = pairwise_poll(slp_res, dx, rng)
        s2 = exact_score(poll_res, dx)
        if s2 is not None and s2 < best_score - 1e-12:
            best_h = poll_res.copy()
            best_score = s2
    final = capped_simplex_project(best_h, target)
    if final is None or not is_feasible(final, n):
        final = fallback.copy()
    fs = exact_score(final, dx)
    if fs is None:
        final = u.copy()
        fs = exact_score(final, dx)
    if fs is None:
        fs = 0.5
    return (final, float(fs), int(n))
