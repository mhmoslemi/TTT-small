        q[ni] = dx * float(np.dot(h[:n - d], 1.0 - h[d:]))
        G[ni, :n - d] = dx * (1.0 - h[d:])
        G[ni, d:] = -dx * h[:n - d]
    return q, G


def lse_ev(x, temp):
    q, G = compute_q_G(x)
    qmax = float(np.max(q))
    sh = (q - qmax) / temp
    e = np.exp(sh)
    se = float(np.sum(e))
    p = e / se
    return qmax + temp * math.log(se), G.T @ p


def project_capped(v, n, S):
    if not np.all(np.isfinite(v)):
        return np.full(n, S / n)
    v = np.clip(v, -1e6, 1e6)
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v))
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = math.fsum(np.clip(v - mid, 0.0, 1.0).tolist())
        if s > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    for _ in range(30):
        res = S - math.fsum(h.tolist())
        if abs(res) < 1e-15:
            break
        if res > 0:
            room = 1.0 - h
            ix = int(np.argmax(room))
            h[ix] += min(res, room[ix])
        else:
            ix = int(np.argmax(h))
            h[ix] -= min(-res, h[ix])
    h = np.clip(h, 0.0, 1.0)
    if abs(math.fsum(h.tolist()) - S) > 1e-10:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        ix = int(np.argmax(mask.astype(float))) if np.any(mask) else 0
        o = h.tolist()
        o[ix] = 0.0
        h[ix] = S - math.fsum(o)
        if not (0.0 <= h[ix] <= 1.0):
            h = np.full(n, S / n)
        h = np.clip(h, 0.0, 1.0)
    return h


def _eq_f(x):
    return [float(np.sum(x[:_g_n])) - _g_S]


def _eq_j(x):
    z = np.zeros(len(x))
    z[:_g_n] = 1.0
    return [z]


def _eo_f(x):
    return x[_g_n]


def _eo_j(x):
    g = np.zeros(len(x))
    g[_g_n] = 1.0
    return g


def _ei_f(x):
    q, _ = compute_q_G(x[:_g_n])
    return x[_g_n] - q


def _ei_j(x):
    n = _g_n
    _, G = compute_q_G(x[:n])
    J = np.zeros((G.shape[0], n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J


def run(seed=42, **kwargs):
    global _g_n, _g_S, _g_dx
    rng = np.random.default_rng(seed)
    parent = None
    n = 42
    pv = globals().get('initial_h_values', None)
    if pv is not None:
        try:
            a = np.array(pv, dtype=np.float64).ravel()
            if a.size >= 2:
                n = int(a.size)
                if np.all(np.isfinite(a)):
                    S_v = float(n) / 2.0
                    parent = project_capped(a, n, S_v)
        except Exception:
            pass
    if parent is None and 'n_points' in kwargs:
        try:
            n = max(2, min(256, int(kwargs['n_points'])))
        except Exception:
            n = 42
    S = float(n) / 2.0
    dx = 2.0 / n
    _g_n, _g_S, _g_dx = n, S, dx

    h0 = np.full(n, 0.5)
    best_h, best_sc = h0.copy(), exact_score(h0)
    if parent is not None:
        ps = exact_score(parent)
        if ps < best_sc - 1e-12:
            best_h, best_sc = parent.copy(), ps

    if n > 256:
        return (best_h, float(best_sc), n)

    starts = []
    if parent is not None:
        starts.append(parent.copy())
        for sig in (0.10, 0.25, 0.50):
            starts.append(project_capped(parent + sig * rng.standard_normal(n), n, S))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    starts.append(project_capped(blk, n, S))
    alt = (np.arange(n) % 2 == 0).astype(float)
    starts.append(project_capped(alt, n, S))
    for _ in range(2 if parent is not None else 4):
        starts.append(project_capped(rng.random(n), n, S))
    if parent is None:
        tt = np.arange(n, dtype=np.float64)
        starts.append(project_capped(0.5 + 0.45 * np.sin(2.0 * np.pi * tt / n), n, S))
    dedup = [starts[0]]
    for s in starts[1:]:
        if not any(float(np.max(np.abs(s - d))) < 1e-10 for d in dedup):
            dedup.append(s)

    bounds = [(0.0, 1.0)] * n
    cons_m = [{'type': 'eq', 'fun': _eq_f, 'jac': _eq_j}]
    temps = (0.02, 0.005, 0.001, 0.0002)

    for st in dedup:
        hc = project_capped(st, n, S)
        sc = exact_score(hc)
        if sc < best_sc - 1e-12:
            best_sc, best_h = sc, hc.copy()
        for temp in temps:
            try:
                res = minimize(lse_ev, hc.copy(), args=(temp,),
                               method='SLSQP', jac=True, bounds=bounds,
                               constraints=cons_m,
                               options={'ftol': 1e-10, 'maxiter': 120})
                if res is not None and np.all(np.isfinite(res.x)):
                    hn = project_capped(res.x, n, S)
                    sn = exact_score(hn)
                    if sn < best_sc - 1e-12:
                        best_sc, best_h = sn, hn.copy()
                    vn, _ = lse_ev(hn, temp)
                    vo, _ = lse_ev(hc, temp)
                    if vn <= vo + 1e-12:
                        hc = hn
            except Exception:
                pass
        try:
            qc, _ = compute_q_G(hc)
            tm = float(np.max(qc))
            x0 = np.concatenate([hc, [tm + max(1e-8, 1e-6 * abs(tm))]])
            be = bounds + [(0.0, None)]
            ce = cons_m + [{'type': 'ineq', 'fun': _ei_f, 'jac': _ei_j}]
            res = minimize(_eo_f, x0, method='SLSQP', jac=_eo_j,
                           bounds=be, constraints=ce,
                           options={'ftol': 1e-10, 'maxiter': 200})
            if res is not None and np.all(np.isfinite(res.x)):
                he = project_capped(res.x[:n], n, S)
                se = exact_score(he)
                if se < best_sc - 1e-12:
                    best_sc, best_h = se, he.copy()
        except Exception:
            pass

    try:
        qb, _ = compute_q_G(best_h)
        tm = float(np.max(qb))
        x0 = np.concatenate([best_h, [tm + max(1e-8, 1e-6 * abs(tm))]])
        be = bounds + [(0.0, None)]
        ce = cons_m + [{'type': 'ineq', 'fun': _ei_f, 'jac': _ei_j}]
        res = minimize(_eo_f, x0, method='SLSQP', jac=_eo_j,
                       bounds=be, constraints=ce,
                       options={'ftol': 1e-10, 'maxiter': 200})
        if res is not None and np.all(np.isfinite(res.x)):
            hf = project_capped(res.x[:n], n, S)
            sf = exact_score(hf)
            if sf < best_sc - 1e-12:
                best_sc, best_h = sf, hf.copy()
    except Exception:
        pass

    best_h = project_capped(best_h, n, S)
    fs = exact_score(best_h)
    if not (np.all(best_h >= -1e-15) and np.all(best_h <= 1.0 + 1e-15)):
        best_h = np.full(n, 0.5)
        fs = exact_score(best_h)
    return (best_h, float(fs), n)
