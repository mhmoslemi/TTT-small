def project_capped_simplex(v, N):
    M = N / 2.0
    v = np.asarray(v, dtype=np.float64)
    if v.shape != (N,) or not np.all(np.isfinite(v)) or np.max(np.abs(v)) > 1e6:
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > M:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    delta = M - np.sum(y)
    for _ in range(3):
        if abs(delta) <= 1e-12 * max(1, N):
            break
        if delta > 0:
            for i in range(N):
                t = min(delta, 1.0 - y[i])
                y[i] += t
                delta -= t
                if abs(delta) < 1e-15:
                    break
        else:
            for i in range(N):
                t = min(-delta, y[i])
                y[i] -= t
                delta += t
                if abs(delta) < 1e-15:
                    break
    if not np.all(np.isfinite(y)):
        return None
    y = np.clip(y, 0.0, 1.0)
    if abs(np.sum(y) - M) > 1e-9 * max(1, N):
        return None
    return y


def is_feasible(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return False
    if np.min(h) < -1e-12 or np.max(h) > 1.0 + 1e-12:
        return False
    return abs(np.sum(h) - N / 2.0) <= 1e-9 * max(1, N)


def exact_c5(h, N):
    try:
        c = np.max(np.correlate(h, 1.0 - h, mode='full'))
        val = float(c) * (2.0 / N)
        return val if np.isfinite(val) else float('inf')
    except Exception:
        return float('inf')


def lag_gradient(x, q, d):
    N = len(x)
    g = np.zeros(N, dtype=np.float64)
    js = max(0, -q)
    je = min(N, N - q)
    for j in range(js, je):
        g[j + q] += d * (1.0 - x[j])
        g[j] -= d * x[j + q]
    return g


def frank_wolfe_min_norm(G):
    K = G.shape[0]
    if K == 0:
        return None
    if K == 1:
        return G[0].copy()
    g = G[0].copy()
    for _ in range(60):
        dots = G @ g
        k = int(np.argmin(dots))
        v = G[k]
        gap = g @ g - g @ v
        w = v - g
        ww = w @ w
        if gap <= 1e-12 * max(1.0, g @ g) or ww <= 1e-24:
            break
        alpha = min(1.0, max(0.0, gap / ww))
        g = g + alpha * w
    return g


def add_start(starts, candidate, N):

    if y is None or not is_feasible(y, N):
        return starts
    for s in starts:
        if np.max(np.abs(y - s)) < 1e-8:
            return starts
    starts.append(y)
    return starts
