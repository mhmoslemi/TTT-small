def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or len(z) == 0:
        return None
    n = len(z)
    if not np.all(np.isfinite(z)) or not np.isfinite(S) or S < 0 or S > n:
        return None
    if float(np.max(np.abs(z))) > 1e12:
        return None
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    if lo >= hi:
        hi = lo + 1e-12
    for _ in range(80):
        mid = lo + 0.5 * (hi - lo)
        y = np.clip(z - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(z - (lo + 0.5 * (hi - lo)), 0.0, 1.0)
    r = S - np.sum(y)
    if r > 1e-14:
        for i in range(n):
            if r <= 0:
                break
            s1 = 1.0 - y[i]
            if s1 > 0:
                a = min(r, s1)
                y[i] += a
                r -= a
    elif r < -1e-14:
        for i in range(n):

                break

                y[i] -= b
                r += b
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(np.sum(y) - S) > 1e-8:
        return None
    return y

    return True

    h = np.asarray(h, dtype=np.float64)
    if h.ndim != 1 or len(h) < 2:

    if not np.all(np.isfinite(h)) or np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):

    vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(vals)):
        return None
    c5 = float(np.max(vals))
    return c5 if np.isfinite(c5) else None


def compute_lags_grad(h):
    n = len(h)
    B = np.zeros(2 * n - 1)
    G = np.zeros((2 * n - 1, n))
    for idx, d in enumerate(range(-(n - 1), n)):
        lo = max(0, -d)
        hi = min(n, n - d)
        B[idx] = np.dot(h[lo:hi], 1.0 - h[lo + d:hi + d])
        G[idx, lo:hi] += 1.0 - h[lo + d:hi + d]
        G[idx, lo + d:hi + d] -= h[lo:hi]
    return B, G


def slsqp_obj(z):
    return z[-1]


def slsqp_obj_grad(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g


def slsqp_ineq(z):
    n = len(z) - 1
    B, _ = compute_lags_grad(z[:n])
    return z[-1] - B


def slsqp_ineq_jac(z):
    n = len(z) - 1
    _, G = compute_lags_grad(z[:n])
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G
    J[:, -1] = 1.0
    return J

    n = len(z) - 1

def slsqp_eq_jac(z):
