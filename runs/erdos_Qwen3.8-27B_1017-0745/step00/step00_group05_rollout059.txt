    corr = np.correlate(h, 1.0 - h, mode='full')
    s = corr * dx  # shape (2n-1,)
    m = s.max()
    exps = np.exp((s - m) / tau)
    w = exps / exps.sum()
    F = m + tau * np.log(exps.sum())
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        t = idx - (n - 1)
        p_t = w[idx]
        # dq_t/dh[j] = 1[j-t in [0,n)) * (1-h[j-t]) - 1[j+t in [0,n)) * h[j+t]

import numpy as np
from scipy.optimize import minimize


def _project(v, target):
    v = np.asarray(v, dtype=float)
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - y.sum()
    for _ in range(50):
        if abs(r) < 1e-14:
            break
        if r > 0:
            cap = 1.0 - y
            m = cap > 1e-15
            tc = cap[m].sum()
            if tc < 1e-15:
                break
            f = min(1.0, r / tc)
            y[m] += cap[m] * f
        else:
            m = y > 1e-15
            tc = y[m].sum()
            if tc < 1e-15:
                break
            f = min(1.0, -r / tc)
            y[m] -= y[m] * f
        r = target - y.sum()
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(y.sum() - target) > 1e-9:
        return None
    return y


def _score(h, dx):
    h = np.asarray(h, dtype=float)
    if h.ndim != 1 or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def _smooth_fg(x, tau, dx):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    s = corr * dx
    m = s.max()
    e = np.exp((s - m) / tau)
    st = e.sum()
    w = e / st
    val = m + tau * np.log(st)
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        t = idx - (n - 1)
        pt = w[idx]
        if pt < 1e-18:
            continue
        a1 = max(0, t)
        b1 = min(n, t + n)
        if a1 < b1:
            g[a1:b1] += pt * dx * (1.0 - x[a1 - t:b1 - t])
        a2 = max(0, -t)
        b2 = min(n, n - t)
        if a2 < b2:
            g[a2:b2] -= pt * dx * x[a2 + t:b2 + t]
    if not np.all(np.isfinite(g)):
        g[:] = 0.0
    return val, g


def _mass_eq(x):
    return [x.sum() - len(x) * 0.5]


def _mass_jac(x):
    return [np.ones(len(x))]


def _optimize_start(x0, tau_list, dx, target, bounds, incumbent, inc_score):
    x = _project(x0, target)
    if x is None:
        return incumbent, inc_score
    sc = _score(x, dx)
    if np.isfinite(sc) and sc < inc_score - 1e-12:
        incumbent, inc_score = x.copy(), sc
    for tau in tau_list:
        try:
            res = minimize(_smooth_fg, x, method='SLSQP', jac=True,
                           args=(tau, dx), bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _mass_eq,
                                         'jac': _mass_jac}],
                           options={'maxiter': 200, 'ftol': 1e-10,
                                    'disp': False})
            xv = np.asarray(res.x, dtype=float)
            if xv.shape != x.shape or not np.all(np.isfinite(xv)):
                continue
            xp = _project(xv, target)
            if xp is None:
                continue

            if np.isfinite(sc2) and sc2 < inc_score - 1e-12:
                incumbent, inc_score = xp.copy(), sc2
            f_cur, _ = _smooth_fg(x, tau, dx)
            f_new, _ = _smooth_fg(xp, tau, dx)
            if np.isfinite(f_new) and np.isfinite(f_cur) and f_new <= f_cur + 1e-10:
                x = xp
        except Exception:
            pass
    return incumbent, inc_score
