    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(3):
        r = target - float(np.sum(y))
        if abs(r) < 1e-14:
            break
        if r > 0:
            for i in range(n):
                add = min(r, 1.0 - y[i])
                if add > 0:
                    y[i] += add
                    r -= add
        else:
            for i in range(n):
                sub = min(-r, y[i])
                if sub > 0:
                    y[i] -= sub
                    r += sub
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    if abs(float(np.sum(y)) - target) > 1e-10:
        return None
    return y

def is_feasible(h, n):
    if len(h) != n or not np.all(np.isfinite(h)):
        return False
    if float(np.min(h)) < -1e-12 or float(np.max(h)) > 1.0 + 1e-12:
        return False
    if abs(float(np.sum(h)) - n / 2.0) > 1e-10:
        return False
    return True

def exact_score(h, n):
    if not is_feasible(h, n):
        return None
    dx = 2.0 / n
    val = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
    return val if np.isfinite(val) else None

def compute_lags_grad(h, n):
    q = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(q)):
        return None, None
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        t = k - (n - 1)
        js1 = max(0, -t)
        je1 = min(n - 1, n - 1 - t)
        if js1 <= je1:
            idx = np.arange(js1, je1 + 1)
            G[k, idx] += 1.0 - h[idx + t]
        js2 = max(0, t)
        je2 = min(n - 1, n - 1 + t)
        if js2 <= je2:
            idx = np.arange(js2, je2 + 1)
            G[k, idx] -= h[idx - t]
    if not np.all(np.isfinite(G)):
        return None, None
    return q, G

def slp_search(h0, n, max_iter):
    h = h0.copy()
    best = h0.copy()
    best_score = exact_score(best, n)
    if best_score is None:
        return best
    rho = 0.08
    rho_max = 0.25
    rho_min = 1e-5
    for _ in range(max_iter):
        if rho < rho_min:
            break
        q, G = compute_lags_grad(h, n)
        if q is None:
            break
        Q = float(np.max(q))
        c = np.zeros(n + 1)
        c[n] = 1.0
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -q
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        bounds = []
        for i in range(n):
            bounds.append((max(-rho, -h[i]), min(rho, 1.0 - h[i])))
        bounds.append((None, None))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'maxiter': 300})
        except Exception:
            rho = max(rho_min, rho * 0.5)
            continue
        if res is None or res.status != 0 or res.x is None:
            rho = max(rho_min, rho * 0.5)
            continue
        x = np.asarray(res.x, dtype=np.float64)
        if x.shape != (n + 1,) or not np.all(np.isfinite(x)):
            rho = max(rho_min, rho * 0.5)
            continue
        d = x[:n]
        z_val = x[n]
        cand = h + d
        proj = capped_simplex_project(cand, n / 2.0)
        if proj is None or not is_feasible(proj, n):
            rho = max(rho_min, rho * 0.5)
            continue
        score = exact_score(proj, n)
        if score is None:
            rho = max(rho_min, rho * 0.5)
            continue
        if score < best_score - 1e-12:
            best = proj.copy()
            best_score = score
        predicted = Q - z_val
        Q_new = float(np.max(np.correlate(proj, 1.0 - proj, mode='full')))
        actual = Q - Q_new
        h_before = h.copy()
        accepted = actual > 1e-12 * max(1.0, abs(Q))
        if accepted:
            h = proj
        ratio = 0.0
        if predicted > 1e-10 * max(1.0, abs(Q)):
            ratio = actual / predicted
        if (not accepted) or (not np.isfinite(ratio)) or ratio < 0.1:
            rho = max(rho_min, rho * 0.5)
        elif ratio > 0.75 and float(np.max(np.abs(proj - h_before))) >= 0.8 * rho:
            rho = min(rho_max, 1.5 * rho)
    return best

def pairwise_poll(h0, n, rng):
    h = h0.copy()
    best = h0.copy()
    best_sc = exact_score(best, n)
    if best_sc is None or n > 256:
        return best
    cur_sc = best_sc
    n_pairs = 24 if n <= 128 else 8
    radii = [0.08, 0.025, 0.008]
    for r in radii:
        for _ in range(n_pairs):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a == b:
                b = (a + 1) % n
            s_lo = max(-h[a], h[b] - 1.0, -r)
            s_hi = min(1.0 - h[a], h[b], r)
            if s_lo > s_hi:
                continue
            cands_s = [s_lo, s_hi, -r, -0.5 * r, 0.5 * r, r]
            seen = set()
            for s in cands_s:
                sc = float(np.clip(s, s_lo, s_hi))
                if abs(sc) < 1e-14:
                    continue
                key = round(sc, 14)
                if key in seen:
                    continue
                seen.add(key)
                trial = h.copy()
                trial[a] += sc
                trial[b] -= sc
                proj = capped_simplex_project(trial, n / 2.0)
                if proj is None or not is_feasible(proj, n):
                    continue
                sc_score = exact_score(proj, n)
                if sc_score is None:
                    continue
                if sc_score < cur_sc - 1e-12:
                    h = proj
                    cur_sc = sc_score
                    if sc_score < best_sc - 1e-12:
                        best = proj.copy()
                        best_sc = sc_score
    return best

def make_starts(n, rng, parent):
    target = n / 2.0
    starts = []
    if parent is not None:
        p = capped_simplex_project(parent, target)
        if p is not None and is_feasible(p, n):
            starts.append(p)
        pf = capped_simplex_project(parent[::-1].copy(), target)
        if pf is not None and is_feasible(pf, n):
            starts.append(pf)
        n1 = capped_simplex_project(parent + 0.035 * rng.standard_normal(n), target)
        if n1 is not None and is_feasible(n1, n):
            starts.append(n1)
        n2 = capped_simplex_project(parent + 0.12 * rng.standard_normal(n), target)
        if n2 is not None and is_feasible(n2, n):
            starts.append(n2)
    r1 = capped_simplex_project(rng.random(n), target)
    if r1 is not None:
        starts.append(r1)
    bv = capped_simplex_project(rng.beta(0.5, 0.5, size=n), target)
    if bv is not None:
        starts.append(bv)
    starts.append(np.full(n, 0.5))
    dedup = []
    for s in starts:
        dup = False
        for d in dedup:
            if float(np.max(np.abs(s - d))) < 1e-10:
                dup = True
                break
        if not dup:
            dedup.append(s)
    return dedup

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    n = 70
    parent = None
    try:
        iv = np.array(initial_h_values, dtype=np.float64).ravel()
        if 2 <= len(iv) <= 999:
            n = len(iv)
            parent = iv
    except Exception:
        pass
    dx = 2.0 / n
    target = n / 2.0
    fallback = np.full(n, 0.5)
    if parent is not None and np.all(np.isfinite(parent)):
        if float(np.max(parent)) <= 1e6 and float(np.min(parent)) >= -1e6:
            if float(np.min(parent)) >= -1e-10 and float(np.max(parent)) <= 1.0 + 1e-10:
                if abs(float(np.sum(parent)) - target) <= 1e-8:
                    pf = capped_simplex_project(parent, target)
                    if pf is not None:
                        fallback = pf
    fb_score = exact_score(fallback, n)
    if fb_score is None:
        fallback = np.full(n, 0.5)
        fb_score = exact_score(fallback, n)
    best = fallback.copy()
    best_score = fb_score
    starts = make_starts(n, rng, parent)
    max_it = 50 if n <= 128 else (25 if n <= 512 else 12)
    for s in starts:
        slp_res = slp_search(s, n, max_it)
        poll_res = pairwise_poll(slp_res, n, rng)
        for cand in [slp_res, poll_res]:
            sc = exact_score(cand, n)
            if sc is not None and sc < best_score - 1e-12:
                best = cand.copy()
                best_score = sc
        if best_score <= 0.3808:
            break
    for scale in [0.025, 0.08]:
        noisy = capped_simplex_project(best + scale * rng.standard_normal(n), target)
        if noisy is None or not is_feasible(noisy, n):
            continue
        slp_res = slp_search(noisy, n, max_it)
        poll_res = pairwise_poll(slp_res, n, rng)
        for cand in [slp_res, poll_res]:
            sc = exact_score(cand, n)
            if sc is not None and sc < best_score - 1e-12:
                best = cand.copy()
                best_score = sc
        if best_score <= 0.3808:
            break
    final = capped_simplex_project(best, target)
    if final is None or not is_feasible(final, n):
        final = fallback.copy()
    final_sc = exact_score(final, n)
    if final_sc is None:
        final = np.full(n, 0.5)
        final_sc = exact_score(final, n)
    return (final, final_sc, n)
