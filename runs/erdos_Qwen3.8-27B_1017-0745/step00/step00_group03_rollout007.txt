        p = np.asarray(parent, dtype=np.float64).ravel()
        if p.size >= 2 and np.all(np.isfinite(p)):
            n = p.size
        elif 'n_points' in kwargs and isinstance(kwargs['n_points'], int) and kwargs['n_points'] >= 2:
            n = kwargs['n_points']
    elif 'n_points' in kwargs and isinstance(kwargs['n_points'], int) and kwargs['n_points'] >= 2:
        n = kwargs['n_points']

    m = n / 2.0

    if n == 1:
        h = np.array([0.5])
        return h, score_c5(h), 1

    if parent is not None:
        p = np.asarray(parent, dtype=np.float64).ravel()
        if p.size == n and np.all(np.isfinite(p)):
            if (p >= 0).all() and (p <= 1).all() and abs(p.sum() - m) < 1e-10 * n:
                baseline = p.copy()
            else:
                baseline = project_box_sum(p, m)
        else:
            baseline = np.full(n, 0.5)
    else:
        baseline = np.full(n, 0.5)

    best = baseline.copy()
    best_score = score_c5(best)

    rng = np.random.default_rng(seed)

    starts = []
    starts.append(baseline.copy())
    starts.append(project_box_sum(1.0 - baseline, m))
    starts.append(np.full(n, 0.5))
    for sc in (0.05, 0.15, 0.35):
        starts.append(project_box_sum(baseline + sc * rng.standard_normal(n), m))
    starts.append(project_box_sum(rng.uniform(0.0, 1.0, n), m))
    z_smooth = np.full(n, 0.5)
    max_r = min(4, (n - 1) // 2)
    idx_arr = np.arange(n, dtype=np.float64)
    for r in range(1, max_r + 1):
        ar = rng.standard_normal()
        br = rng.standard_normal()
        z_smooth += (0.12 / r) * (ar * np.sin(2.0 * np.pi * r * idx_arr / n)
                                  + br * np.cos(2.0 * np.pi * r * idx_arr / n))
    starts.append(project_box_sum(z_smooth, m))

    bounds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [
        {'type': 'eq', 'fun': epi_eq, 'jac': epi_eq_jac, 'args': (n, m)},
        {'type': 'ineq', 'fun': epi_ineq, 'jac': epi_ineq_jac, 'args': (n, m)},
    ]
    opt_kwargs = {'maxiter': 300, 'ftol': 1e-10, 'disp': False}

    def optimize_from(h0):
        q0, _ = lag_values_and_jac(h0)
        t0 = max(float(q0.max()), 1e-10)
        z0 = np.concatenate([h0, [t0]])
        try:
            res = minimize(epi_obj, z0, jac=epi_obj_jac, args=(n, m),
                           method='SLSQP', bounds=bounds,
                           constraints=cons, options=opt_kwargs)
            if res.x is not None and res.x.size == n + 1 and np.all(np.isfinite(res.x[:n])):
                return project_box_sum(res.x[:n], m)
        except Exception:
            pass
        return h0.copy()

    for h0 in starts:
        s0 = score_c5(h0)
        if s0 < best_score - 1e-15:
            best_score = s0
            best = h0.copy()
        h_opt = optimize_from(h0)
        s_opt = score_c5(h_opt)
        if s_opt < best_score - 1e-15:
            best_score = s_opt
            best = h_opt.copy()

    hop_scales = (0.10, 0.05, 0.025, 0.0125)
    for hs in hop_scales:
        z_hop = best + hs * rng.standard_normal(n)
        h_hop = project_box_sum(z_hop, m)
        sh = score_c5(h_hop)
        if sh < best_score - 1e-15:
            best_score = sh
            best = h_hop.copy()
        h_opt = optimize_from(h_hop)
        s_opt = score_c5(h_opt)
        if s_opt < best_score - 1e-15:
            best_score = s_opt
            best = h_opt.copy()

    best = project_box_sum(best, m)
    final_score = score_c5(best)
    if not np.isfinite(final_score) or best.min() < -1e-10 or best.max() > 1.0 + 1e-10:
        best = np.full(n, 0.5)
        final_score = score_c5(best)

    return best.astype(np.float64), float(final_score), int(n)
