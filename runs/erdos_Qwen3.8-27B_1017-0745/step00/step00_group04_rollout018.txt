def compute_lags(h, n):
    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    # q_0
    q[0] = dx * float(np.sum(h * (1.0 - h)))
    G[0, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        idx_p = 2 * d - 1  # q_{d,+}
        idx_m = 2 * d      # q_{d,-}
        # q_{d,+} = dx * sum_{i=0}^{n-d-1} h[i+d]*(1-h[i])
        q[idx_p] = dx * float(np.sum(h[d:] * (1.0 - h[:-d])))
        # q_{d,-} = dx * sum_{i=0}^{n-d-1} h[i]*(1-h[i+d])
        q[idx_m] = dx * float(np.sum(h[:-d] * (1.0 - h[d:])))
        # Gradients for d,+
        G[idx_p, :n-d] += -dx * h[d:]       # -dx*h[i+d] for i=0..n-d-1
        G[idx_p, d:] += dx * (1.0 - h[:-d])  # dx*(1-h[i]) for i=0..n-d-1, component i+d
        # Gradients for d,-
        G[idx_m, :n-d] += dx * (1.0 - h[d:])   # dx*(1-h[i+d]) for i=0..n-d-1
        G[idx_m, d:] += -dx * h[:-d]            # -dx*h[i] for i=0..n-d-1, component i+d
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None, None
    return q, G

def active_bundle_gradient(q, G):
    Cq = float(np.max(q))
    tol = max(1e-12, 1e-7 * max(1.0, abs(Cq)))
    active = np.where(q >= Cq - tol)[0]
    if len(active) > 12:
        order = np.argsort(q[active])[::-1][:12]
        active = active[order]
    k = len(active)
    if k == 0:
        return None
    U = G[active, :].copy()
    U -= U.mean(axis=1, keepdims=True)
    K = U @ U.T
    K = 0.5 * (K + K.T)
    alpha = np.full(k, 1.0 / k)
    for _ in range(40):
        z = K @ alpha
        j = int(np.argmin(z))
        if float(alpha @ z) - float(z[j]) <= 1e-12:
            break
        v = np.zeros(k)
        v[j] = 1.0
        v -= alpha
        vkv = float(v @ K @ v)
        if vkv <= 1e-24:
            break
        gamma = float(np.clip(-float(v @ z) / vkv, 0.0, 1.0))
        alpha += gamma * v
        alpha = np.clip(alpha, 0.0, None)
        alpha /= float(np.sum(alpha))
    g = alpha @ U
    if not np.all(np.isfinite(g)):
        # Fallback: centered gradient of largest lag
        jmax = int(np.argmax(q))
        g = G[jmax, :] - float(np.mean(G[jmax, :]))
        if not np.all(np.isfinite(g)):
            return None
    return g

def local_search(h0, rng, n, max_iter):

    best = cur.copy()

        return best, best_score
    for it in range(max_iter):
        q, G = compute_lags(cur, n)
        improved = False
        if q is not None:
            g = active_bundle_gradient(q, G)
            if g is not None:
                for b in range(8):

            h += (res / tot) * cap
            h = np.clip(h, 0.0, 1.0)
    target = S - math.fsum(float(h[i]) for i in range(1, n))
    if 0.0 <= target <= 1.0:
        h[0] = target
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(float(np.sum(h)) - S) > 1e-9:
        return None
    return h

def compute_lags(h, n):
    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    q[0] = dx * float(np.sum(h * (1.0 - h)))
    G[0, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        ip = 2 * d - 1
        im = 2 * d
        q[ip] = dx * float(np.sum(h[d:] * (1.0 - h[:-d])))
        q[im] = dx * float(np.sum(h[:-d] * (1.0 - h[d:])))
        G[ip, :n - d] += -dx * h[d:]
        G[ip, d:] += dx * (1.0 - h[:-d])
        G[im, :n - d] += dx * (1.0 - h[d:])
        G[im, d:] += -dx * h[:-d]
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None, None
    return q, G

def active_bundle_gradient(q, G):
    Cq = float(np.max(q))
    tol = max(1e-12, 1e-7 * max(1.0, abs(Cq)))
    active = np.where(q >= Cq - tol)[0]
    if len(active) > 12:
        order = np.argsort(q[active])[::-1][:12]
        active = active[order]
    k = len(active)
    if k == 0:
        return None
    U = G[active, :].copy()
    U -= U.mean(axis=1, keepdims=True)
    K = U @ U.T
    K = 0.5 * (K + K.T)
    alpha = np.full(k, 1.0 / k)
    for _ in range(40):
        z = K @ alpha
        j = int(np.argmin(z))
        if float(alpha @ z) - float(z[j]) <= 1e-12:
            break
        v = np.zeros(k)
        v[j] = 1.0
        v -= alpha
        vkv = float(v @ K @ v)
        if vkv <= 1e-24:
            break
        gamma = float(np.clip(-float(v @ z) / vkv, 0.0, 1.0))
        alpha += gamma * v
        alpha = np.clip(alpha, 0.0, None)
        alpha /= float(np.sum(alpha))
    g = alpha @ U
    if not np.all(np.isfinite(g)):
        jmax = int(np.argmax(q))
        g = G[jmax, :] - float(np.mean(G[jmax, :]))
        if not np.all(np.isfinite(g)):
            return None
    return g

def local_search(h0, rng, n, max_iter):
    cur = project_capped_simplex(h0.copy(), n)
    if cur is None:
        cur = np.full(n, 0.5)
    cur_score = exact_score(cur, n)
    best = cur.copy()
    best_score = cur_score
    eta = 0.5
    if not math.isfinite(cur_score):
        return best, best_score
    for it in range(max_iter):
        q, G = compute_lags(cur, n)
        improved = False
        if q is not None:
            g = active_bundle_gradient(q, G)
            if g is not None:
                for b in range(8):
                    eta_try = eta * (0.5 ** b)
                    prop = project_capped_simplex(cur - eta_try * g, n)
                    if prop is None or np.max(np.abs(prop - cur)) < 1e-14:
                        continue
                    ps = exact_score(prop, n)
                    if ps < cur_score - 1e-12:
                        cur = prop
                        cur_score = ps
                        if ps < best_score - 1e-12:
                            best = cur.copy()
                            best_score = ps
                        eta = min(2.0, 1.25 * eta_try)
                        improved = True
                        break
        if not improved:
            eta = max(1e-6, eta / 2.0)
            if it < 40:
                sigma, T = 0.12, 0.002
            elif it < 80:
                sigma, T = 0.07, 0.0008
            else:
                sigma, T = 0.035, 0.00025
            best_prop = None
            bp_score = cur_score
            for _ in range(4):
                prop = project_capped_simplex(cur + sigma * rng.standard_normal(n), n)
                if prop is None:
                    continue
                ps = exact_score(prop, n)
                if ps < bp_score:
                    best_prop = prop
                    bp_score = ps
            if best_prop is not None:
                delta = bp_score - cur_score
                accept = True
                if delta > 0:
                    accept = rng.random() < math.exp(-delta / T)
                if accept:
                    cur = best_prop
                    cur_score = bp_score
                    eta = 0.25
                    if cur_score < best_score - 1e-12:
                        best = cur.copy()
                        best_score = cur_score
    best_score = exact_score(best, n)
    return best, best_score

def add_unique(starts, vec, n):
    pv = project_capped_simplex(vec, n)
    if pv is None or not is_feasible(pv, n):
        return
    for existing in starts:
        if np.max(np.abs(pv - existing)) < 1e-10:
            return
    starts.append(pv)

def run(seed=42, **kwargs):
    parent = globals().get('initial_h_values', None)
    n = None
    parent_vec = None
    if parent is not None:
        try:
            pa = np.array(parent, dtype=np.float64).ravel()
            if pa.size >= 2:
                n = pa.size
                if np.all(np.isfinite(pa)):
                    parent_vec = pa
        except Exception:
            pass
    if n is None:
        try:
            nn = int(kwargs.get('n_points', 42))
            if 2 <= nn <= 512:
                n = nn
        except Exception:
            n = 42
    try:
        s = int(seed)
    except Exception:
        s = 42
    rng = np.random.default_rng(s)

    const_h = np.full(n, 0.5)
    const_score = exact_score(const_h, n)

    parent_feasible = None
    parent_score = float('inf')
    if parent_vec is not None:
        pv = project_capped_simplex(parent_vec, n)
        if pv is not None and is_feasible(pv, n):
            parent_feasible = pv
            parent_score = exact_score(pv, n)

    if parent_feasible is not None and parent_score < const_score - 1e-12:
        incumbent = parent_feasible.copy()
        incumbent_score = parent_score
    else:
        incumbent = const_h.copy()
        incumbent_score = const_score

    if n > 128:
        if not is_feasible(incumbent, n):
            incumbent = const_h
            incumbent_score = const_score
        incumbent_score = exact_score(incumbent, n)
        return (incumbent, float(incumbent_score), int(n))

    starts = []
    if parent_feasible is not None:
        starts.append(parent_feasible.copy())
    add_unique(starts, const_h, n)

    block = np.zeros(n)
    block[:n // 2] = 1.0
    if n % 2 == 1:
        block[n // 2] = 0.5
    add_unique(starts, block, n)

    alt = np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n)])
    add_unique(starts, alt, n)

    idx = np.arange(n, dtype=np.float64)
    sinus = 0.5 + 0.45 * np.sin(2.0 * np.pi * idx / n)
    add_unique(starts, sinus, n)

    if parent_feasible is not None:
        for sig in [0.05, 0.15, 0.35, 0.15]:
            add_unique(starts, parent_feasible + sig * rng.standard_normal(n), n)
        for _ in range(2):
            add_unique(starts, rng.random(n), n)
    else:
        for _ in range(4):
            add_unique(starts, rng.random(n), n)

    starts = starts[:12]

    max_iter = 120 if n <= 64 else 80
    for st in starts:
        try:
            cand, cs = local_search(st, rng, n, max_iter)
            if cand is not None and is_feasible(cand, n):
                cs = exact_score(cand, n)
                if cs < incumbent_score - 1e-12:
                    incumbent = cand
                    incumbent_score = cs
        except Exception:
            pass

    if not is_feasible(incumbent, n):
        fixed = project_capped_simplex(incumbent, n)
        if fixed is not None and is_feasible(fixed, n):
            incumbent = fixed
        elif parent_feasible is not None:
            incumbent = parent_feasible
        else:
            incumbent = const_h
    incumbent_score = exact_score(incumbent, n)
    return (incumbent, float(incumbent_score), int(n))
