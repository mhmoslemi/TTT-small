                for alpha in (0.25, 0.5, 0.75):
                    xseg = project_capped(snap[i] + alpha * (u - snap[i]), n, S)
                    if xseg is None or not is_feasible(xseg, n, S):
                        continue
                    scs = exact_score(xseg, n)
                    if scs is not None and scs < bs:
                        bs, bc = scs, xseg
            if bs < best_score:
                best_score, best_h = bs, bc.copy()
            if bs < snap_scores[i]:
                pop[i] = bc
                pop_scores[i] = bs

        if best_score < gen_start_best - tol:
            stagnant = 0
        else:
            stagnant += 1

        if stagnant >= 18 and restarts < 2:
            restarts += 1
            stagnant = 0
            order = np.argsort(pop_scores)
            n_elite = max(2, M // 4)
            elite_set = set(order[:n_elite].tolist())
            bi = int(np.argmin(pop_scores))
            elite_set.add(bi)
            for j in range(M):
                if j in elite_set:
                    continue
                sigmas = [0.05, 0.15, 0.35, 0.60]
                z = best_h + sigmas[j % 4] * rng.standard_normal(n)
                proj = project_capped(z, n, S)
                if proj is not None and is_feasible(proj, n, S):
                    sc = exact_score(proj, n)
                    if sc is not None:
                        pop[j], pop_scores[j] = proj, sc
                        if sc < best_score:
                            best_score, best_h = sc, proj.copy()
                        continue
                pop[j], pop_scores[j] = best_h.copy(), best_score
        elif restarts >= 2 and stagnant >= 24:
            break

    pool = [(best_h, exact_score(best_h, n))]
    for fh, _ in fallbacks:
        if np.max(np.abs(fh - best_h)) > 1e-12:
            sf = exact_score(fh, n)
            if sf is not None:
                pool.append((fh, sf))

    final_h, final_score = pool[0]
    for fh, fs in pool[1:]:
        if fs < final_score:
            final_h, final_score = fh, fs

    if not is_feasible(final_h, n, S):
        proj = project_capped(final_h, n, S)
        if proj is not None:
            final_h = proj
        else:
            final_h = h_const
    final_score = exact_score(final_h, n)
    if final_score is None:
        final_h, final_score = h_const, exact_score(h_const, n)

    return (final_h, final_score, n)
