                elif rho >= 0.75 and ca == 1.0 and dn >= 0.8 * R:
                    R = min(2.0 * R, Rmax)
    final = project_capped(best, n)
    if final is None or not is_feasible(final, n):
        final = parent.copy()
    fs = exact_score(final, n)
    fb = exact_score(parent, n)
    if fs > fb:
        final = parent.copy()
        fs = fb
    c5_bound = exact_score(final, n)
    return (final, c5_bound, int(n))
