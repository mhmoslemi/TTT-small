import numpy as np

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1 or not np.all(np.isfinite(v)) or np.max(np.abs(v)) > 1e12:
        return None
    n = len(v)
    if n < 2 or S < -1e-12 or S > n + 1e-12:
        return None
    lo, hi = np.min(v) - 1.0, np.max(v)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    res = S - np.sum(y)
    tol = 1e-10 * max(1, n)
    if abs(res) > tol:
        for _ in range(2):
            for i in range(n):
                if res > 0:
                    add = min(res, 1.0 - y[i])
                    y[i] += add
                    res -= add
                else:
                    sub = min(-res, y[i])
                    y[i] -= sub
                    res += sub
    if not np.all(np.isfinite(y)) or np.min(y) < -1e-12 or np.max(y) > 1.0 + 1e-12:
        return None
    y = np.clip(y, 0.0, 1.0)
    if abs(np.sum(y) - S) > tol:
        return None
    return y

def is_feasible(h, n):
    if h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.min(h) < -1e-12 or np.max(h) > 1.0 + 1e-12:
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score(h, n):
    if not is_feasible(h, n):
        return None
    try:
        corr = np.correlate(h, 1.0 - h, mode='full')
        if len(corr) != 2 * n - 1 or not np.all(np.isfinite(corr)):
            return None
        return float(np.max(corr) * (2.0 / n))
    except Exception:
        mx = 0.0
        for k in range(-(n - 1), n):
            s = 0.0
            for i in range(max(0, k), min(n, n + k)):
                s += h[i] * (1.0 - h[i - k])
            mx = max(mx, s)
        return float(mx * (2.0 / n))

def random_zero_sum_direction(n, rng):
    d = rng.standard_normal(n)
    d -= np.mean(d)
    d[-1] = -np.sum(d[:-1])
    norm = np.linalg.norm(d)
    if not np.isfinite(norm) or norm < 1e-14:
        d = np.zeros(n)
        d[0], d[1] = 1.0, -1.0
        norm = np.sqrt(2.0)
    d /= norm
    if not np.all(np.isfinite(d)):
        return None
    return d

def chord_interval(h, d, n):
    lo, hi = -1e18, 1e18
    for i in range(n):
        if d[i] > 1e-15:
            lo = max(lo, -h[i] / d[i])
            hi = min(hi, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-15:
            lo = max(lo, (1.0 - h[i]) / d[i])
            hi = min(hi, -h[i] / d[i])
    if not (np.isfinite(lo) and np.isfinite(hi)):
        return None
    if lo > hi + 1e-14:
        return None
    if lo > 1e-10 or hi < -1e-10:
        return None
    return max(lo, -1e15), min(hi, 1e15)

def repair_trial(v, S, n):
    if not np.all(np.isfinite(v)):
        return None

        return None

                    res += sub

        return None, 0

        return trial, 1

        return trial, 2
