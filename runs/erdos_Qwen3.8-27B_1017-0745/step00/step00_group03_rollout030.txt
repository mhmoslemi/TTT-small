from math import sqrt, copysign

def project_box_sum(z, N):
    m = N / 2.0
    z = np.asarray(z, dtype=np.float64)
    if z.shape != (N,) or not np.all(np.isfinite(z)) or np.max(np.abs(z)) > 1e100:
        return np.full(N, 0.5)
    lo, hi = z.min() - 1.0, z.max()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    e = m - h.sum()
    if abs(e) > 1e-13:
        for _ in range(2):
            for i in range(N):
                if abs(e) < 1e-14:
                    break
                if e > 0:
                    d = min(e, 1.0 - h[i])
                    if d > 1e-15:
                        h[i] += d
                        e -= d
                else:
                    d = min(-e, h[i])
                    if d > 1e-15:
                        h[i] -= d
                        e += d
    return np.clip(h, 0.0, 1.0)

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / N))

def is_feasible(h, N):
    m = N / 2.0
    return (h.shape == (N,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-12 and h.max() <= 1.0 + 1e-12 and
            abs(h.sum() - m) <= 1e-10 * max(1, N))

def lag_coeffs(h, d, N):
    M = 2 * N - 1
    A = np.zeros(M)
    B = np.zeros(M)
    C = np.zeros(M)
    om = 1.0 - h
    for k in range(-(N - 1), N):
        j = k + N - 1
        if k >= 0:
            L = N - k
            a, b, u, v = h[k:k+L], om[:L], d[k:k+L], d[:L]
        else:
            s = -k
            L = N - s
            a, b, u, v = h[:L], om[s:s+L], d[:L], d[s:s+L]
        A[j] = np.sum(a * b)
        B[j] = np.sum(u * b - a * v)
        C[j] = -np.sum(u * v)
    return A, B, C

def line_search(h, d, N, cur_score):
    # compute feasible interval
    nz = np.abs(d) > 1e-15
    if not np.any(nz):
        return h, cur_score
    hn = h[nz]
    dn = d[nz]
    r0 = -hn / dn
    r1 = (1.0 - hn) / dn
    a_lo = np.maximum(r0, r1).max()
    a_hi = np.minimum(r0, r1).min()
    if a_hi - a_lo < 1e-14:
        return h, cur_score
    a_lo = max(a_lo, -np.inf)
    a_hi = min(a_hi, np.inf)
    
    M = 2 * N - 1
    A, B, C = lag_coeffs(h, d, N)
    if not (np.all(np.isfinite(A)) and np.all(np.isfinite(B)) and np.all(np.isfinite(C))):
        return h, cur_score
    
    # candidate alphas
    cands = [a_lo, a_hi, 0.0]
    # stationary points
    for k in range(M):
        if abs(C[k]) > 1e-15:
            sp = -B[k] / (2.0 * C[k])
            if a_lo - 1e-10 <= sp <= a_hi + 1e-10:
                cands.append(sp)
    # pairwise intersections for N <= 80
    if N <= 80:
        for k in range(M):
            for l in range(k + 1, M):
                u = C[k] - C[l]
                v = B[k] - B[l]
                w = A[k] - A[l]
                scale = max(abs(u), abs(v), abs(w), 1e-300)
                if abs(u) <= 1e-12 * scale:
                    if abs(v) > 1e-12 * scale:
                        r = -w / v
                        if a_lo - 1e-10 <= r <= a_hi + 1e-10:
                            cands.append(r)
                else:
                    D = v * v - 4.0 * u * w
                    tol = 1e-12 * (v * v + abs(4.0 * u * w) + 1e-300)
                    if D >= -tol:
                        D = max(D, 0.0)
                        sD = sqrt(D)
                        q = -0.5 * (v + copysign(sD, v) if v != 0 else sD)
                        if abs(q) > 1e-300:
                            r1_ = q / u
                            r2_ = w / q
                        else:

                        for r in [r1_, r2_]:
                            if a_lo - 1e-10 <= r <= a_hi + 1e-10:
                                cands.append(r)
    else:
        # grid for large N

        grid = np.linspace(a_lo, a_hi, n_grid)
        cands.extend(grid.tolist())
    
    # deduplicate

    cands.sort()
    dedup = []
    for c in cands:
        if not dedup or abs(c - dedup[-1]) > 1e-12 * max(1.0, abs(c)):
            dedup.append(c)

        return h, cur_score

    shortlist = []

        batch = alpha_arr[st:st+512]
        envelope = np.max(A[None, :] + B[None, :] * batch[:, None] + C[None, :] * batch[:, None] ** 2, axis=1)
        idx_in_batch = np.argsort(envelope)[:min(16, len(batch))]
        for idx in idx_in_batch:
            shortlist.append((envelope[idx], st + idx))
    
    shortlist.sort(key=lambda x: x[0])
    # pick top candidates, ensure endpoints and 0 are included
    selected = set()

        for i, dc in enumerate(dedup):
            if abs(dc - c) < 1e-12:
                selected.add(i)
                break
    
    # evaluate exact score for selected
    best_h = h
    best_score = cur_score
    for idx in sorted(selected):
        alpha = dedup[idx]
        hp = h + alpha * d
        hp = project_box_sum(hp, N)
        if is_feasible(hp, N):
            s = score_exact(hp, N)
            if s < best_score - 1e-15:
                best_h = hp
                best_score = s

def make_dirs(h, N, rng, n_star, n_rand, n_block, n_harm):
    dirs = []
    # star pairs
    if n_star > 0:
        anchor = int(np.argmax(np.minimum(h, 1.0 - h)))
        others = [i for i in range(N) if i != anchor]
        step = max(1, len(others) // n_star)
        for t in range(n_star):
            idx = others[min(t * step, len(others) - 1)]
            d = np.zeros(N)
            d[idx] = 1.0
            d[anchor] = -1.0
            dirs.append(d)
    # random pairs
    if n_rand > 0 and N >= 2:
        used = set()
        for _ in range(n_rand):
            for att in range(50):
                i, j = rng.integers(0, N, size=2)
                if i == j:
                    continue
                key = (min(i, j), max(i, j))
                if key not in used:
                    used.add(key)
                    d = np.zeros(N)
                    d[i] = 1.0
                    d[j] = -1.0
                    dirs.append(d)
                    break
            else:
                break
    # sparse blocks
    if n_block > 0 and N >= 3:
        for t in range(n_block):
            sz = int(rng.integers(3, min(12, N) + 1))
            sup = rng.choice(N, size=sz, replace=False)
            vals = rng.standard_normal(sz)
            vals -= vals.mean()
            vals[sz - 1] -= vals.sum()  # ensure zero sum
            if np.max(np.abs(vals)) < 1e-15:
                continue
            d = np.zeros(N)
            d[sup] = vals
            d /= np.max(np.abs(d))
            dirs.append(d)
    # harmonics
    if n_harm > 0 and N >= 3:
        max_r = (N - 1) // 2
        for t in range(min(n_harm, max_r)):
            r = t + 1
            phase = rng.uniform(0.0, 2.0 * np.pi)
            i = np.arange(N, dtype=np.float64)
            d = np.cos(2.0 * np.pi * r * i / N + phase)
            d -= d.mean()
            if np.max(np.abs(d)) < 1e-15:
                continue
            d /= np.max(np.abs(d))
            dirs.append(d)
    return dirs

def build_starts(N, rng, baseline):

    z = np.full(N, 0.5)
    for r in range(1, R + 1):
        a_r = rng.standard_normal()
        b_r = rng.standard_normal()
        z += (0.12 / r) * (a_r * np.sin(2 * np.pi * r * i / N) + b_r * np.cos(2 * np.pi * r * i / N))
    starts.append(project_box_sum(z, N))
    # Deduplicate
    dedup = []
    for s in starts:

        ok = True
        for d in dedup:

                break

    return dedup
