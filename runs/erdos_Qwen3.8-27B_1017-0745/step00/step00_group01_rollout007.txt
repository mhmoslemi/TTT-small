            maxiter = max(10, min(int(kwargs['maxiter']), 500))
        except Exception:
            pass

    try:
        seed = int(seed)
    except Exception:
        seed = 42
    rng = np.random.default_rng(seed)

    starts = make_starts(parent, n, rng)
    best_h = starts[0].copy()
    best_score = exact_score(best_h)

    for s in starts[1:]:
        sc = exact_score(s)
        if sc < best_score - 1e-12:
            best_h = s.copy()
            best_score = sc

    for s in starts:
        h_new = try_slsqp(s, n, maxiter)
        if h_new is not None:
            sc = exact_score(h_new)
            if sc < best_score - 1e-12:
                best_h = h_new.copy()
                best_score = sc

    if do_pair and n <= 128:
        best_h, best_score = pair_search(best_h, n, 3)

    best_h = project_capped_simplex(best_h, n)
    final_score = exact_score(best_h)
    parent_score = exact_score(parent)
    if parent_score < final_score - 1e-12:
        best_h = parent.copy()
        final_score = parent_score

    if not (np.all(np.isfinite(best_h)) and np.all(best_h >= 0) and np.all(best_h <= 1)):
        best_h = np.full(n, 0.5)
        final_score = exact_score(best_h)

    return best_h, final_score, int(n)
