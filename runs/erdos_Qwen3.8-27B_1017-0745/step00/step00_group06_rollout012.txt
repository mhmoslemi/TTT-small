def capped_simplex_project(v, N, M):
    if v.shape != (N,) or not np.all(np.isfinite(v)) or np.max(np.abs(v)) > 1e6:
        return None
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > M:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    delta = M - np.sum(y)
    for _ in range(3):
        if abs(delta) <= 1e-12 * max(1, N):
            break
        if delta > 0:
            for i in range(N):
                if delta <= 0:
                    break
                add = min(delta, 1.0 - y[i])
                y[i] += add
                delta -= add
        else:
            for i in range(N):
                if delta >= 0:
                    break
                sub = min(-delta, y[i])
                y[i] -= sub
                delta += sub
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    y = np.clip(y, 0.0, 1.0)
    if abs(np.sum(y) - M) > 1e-9 * max(1, N):
        return None
    return y

def exact_score(h, N):
    try:
        if h.shape != (N,) or not np.all(np.isfinite(h)):
            return float('inf')
        c = max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / N)
        if not np.isfinite(c):
            return float('inf')
        return float(c)
    except Exception:
        return float('inf')

def check_feasible(h, N):

        return False
    if np.any(h < 0) or np.any(h > 1.0):

    if abs(np.sum(h) - M) > 1e-9 * max(1, N):

    return True

def build_block_profile(N, M, offset):
    h = np.zeros(N, dtype=np.float64)
    fl = int(math.floor(M))
    fr = M - fl
    for j in range(fl):
        h[(offset + j) % N] = 1.0
    if fl < N and fr > 1e-12:
        h[(offset + fl) % N] = fr
    return h

def build_wave_profile(N, k, phase):
    i = np.arange(N, dtype=np.float64)
    return 0.5 + 0.22 * np.sin(2.0 * math.pi * k * i / N + phase)

def deduplicate_profile(cand, existing, tol=1e-8):
    for e in existing:
        if np.max(np.abs(cand - e)) < tol:
            return False
    return True
