import numpy as np
from scipy.optimize import minimize

def capped_simplex_project(v, target):
    n = len(v)
    if n < 2 or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - float(np.sum(y))
    if abs(r) > 1e-13:
        if r > 0:
            cap = 1.0 - y
            tot = float(np.sum(cap))
            if tot <= 1e-15:
                return None
            y = y + (r / tot) * cap
        else:
            tot = float(np.sum(y))
            if tot <= 1e-15:
                return None
            y = y + (r / tot) * y
    y = np.clip(y, 0.0, 1.0)
    if abs(float(np.sum(y)) - target) > 1e-10 or not np.all(np.isfinite(y)):
        return None
    return y

def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def is_feasible(h, target):
    if h.ndim != 1 or not np.all(np.isfinite(h)):
        return False
    if float(np.min(h)) < -1e-12 or float(np.max(h)) > 1.0 + 1e-12:
        return False
    return abs(float(np.sum(h)) - target) <= 1e-10

def smooth_fg(h, args):
    tau, dx = args
    s = dx * np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(s))
    e = np.exp((s - m) / tau)
    se = float(np.sum(e))
    if se < 1e-300:
        se = 1e-300
    p = e / se
    F = m + tau * float(np.log(se))
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-15:
            continue
        t = k - (n - 1)
        lo1, hi1 = max(0, t), min(n, t + n)
        g[lo1:hi1] += p[k] * dx * (1.0 - h[lo1 - t:hi1 - t])
        lo2, hi2 = max(0, -t), min(n, n - t)
        g[lo2:hi2] -= p[k] * dx * h[lo2 + t:hi2 + t]
    if not np.all(np.isfinite(g)):
        return F, None
    return F, g

    return [float(np.sum(h)) - len(h) / 2.0]

    return np.ones(len(h))
