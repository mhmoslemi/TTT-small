def compute_q_G(h, n):
    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    # q[0]
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        # q[2d-1] = dx * sum_{i=0}^{n-d-1} h[i+d]*(1-h[i])
        np_plus = n - d
        q[2*d-1] = dx * np.sum(h[d:d+np_plus] * (1.0 - h[:np_plus]))
        # G[2d-1, i+d] += dx*(1-h[i]), G[2d-1, i] += dx*(-h[i+d])

import numpy as np
import math

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(corr * dx))
    if not math.isfinite(val):
        return float('inf')
    return val

def is_feasible(h, n, S):
    if h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    s1 = float(np.sum(h))
    s2 = math.fsum(float(x) for x in h)
    if abs(s1 - S) > 1e-10 or abs(s2 - S) > 1e-10:
        return False
    return True

def capped_simplex(v, S):
    v = np.clip(v.astype(np.float64), -1e6, 1e6)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(v - mid, 0.0, 1.0)
        if float(np.sum(h)) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    # Correct residual
    s = float(np.sum(h))
    res = S - s
    if res > 0:
        cap = 1.0 - h
        cap = np.maximum(cap, 0.0)
        total_cap = float(np.sum(cap))
        if total_cap > 0:
            h += res * cap / total_cap
    elif res < 0:
        total_h = float(np.sum(h))
        if total_h > 0:
            h += (-res) * (h / total_h)
    # Clamp
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    s = float(np.sum(h))
    res = S - s
    if res > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
        if len(idx) > 0:
            h[idx[0]] = min(1.0, h[idx[0]] + res)
    elif res < 0:
        idx = np.where(h > 1e-12)[0]
        if len(idx) > 0:
            h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_q_G(h, n):
    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        q[2*d-1] = dx * np.sum(h[d:d+m] * (1.0 - h[:m]))
        G[2*d-1, d:d+m] += dx * (1.0 - h[:m])
        G[2*d-1, :m] += -dx * h[d:d+m]
        q[2*d] = dx * np.sum(h[:m] * (1.0 - h[d:d+m]))
        G[2*d, :m] += dx * (1.0 - h[d:d+m])
        G[2*d, d:d+m] += -dx * h[:m]
    return q, G

def solve_quad(A, B, C, lo, hi):
    tol = 1e-14 * max(1.0, abs(A), abs(B), abs(C))
    roots = []
    if abs(A) < tol:
        if abs(B) > tol:
            r = -C / B
            if lo - 1e-10 <= r <= hi + 1e-10:
                roots.append(min(max(r, lo), hi))
    else:
        disc = B * B - 4.0 * A * C
        thr = 64.0 * np.finfo(np.float64).eps * (B * B + abs(4.0 * A * C) + 1.0)
        if disc < -thr:
            return roots
        disc = max(0.0, disc)
        sd = math.sqrt(disc)
        q_val = -0.5 * (B + math.copysign(sd, B))
        if abs(q_val) > 1e-30:
            r1 = q_val / A
            r2 = C / q_val
        else:
            r1 = -B / (2.0 * A)
            r2 = r1
        for r in (r1, r2):
            if math.isfinite(r) and lo - 1e-10 <= r <= hi + 1e-10:
                roots.append(min(max(r, lo), hi))
    return roots

def build_hull(slopes, intercepts, lo, hi):
    # returns list of (slope, intercept, t_start, t_end)
    idx = np.lexsort(-intercepts)  # sort by slope asc, intercept desc
    # Actually lexsort sorts by last key first. So np.lexsort((-intercepts, slopes)) sorts by slopes asc, then -intercepts asc = intercepts desc
    idx = np.lexsort((-intercepts, slopes))
    hull = []  # list of [slope, intercept, start]
    for ix in idx:
        m = slopes[ix]
        c = intercepts[ix]
        while hull:
            om, oc, os_ = hull[-1]
            if abs(m - om) < 1e-14 * max(1.0, abs(m), abs(om)):
                # same slope, keep higher intercept (new one since we sorted desc)
                if c > oc:
                    hull.pop()
                    continue
                else:
                    break
            x_cross = (oc - c) / (m - om)
            if x_cross <= os_ + 1e-14:
                hull.pop()
            else:
                break
        if hull:
            om, oc, os_ = hull[-1]
            x_cross = (oc - c) / (m - om)
            hull.append([m, c, x_cross])
        else:
            hull.append([m, c, lo])
    # Convert to segments
    segments = []
    n_hull = len(hull)
    for k in range(n_hull):
        m, c, s = hull[k]
        e = hull[k+1][2] if k + 1 < n_hull else hi
        s = max(s, lo)
        e = min(e, hi)
        if s < e:
            segments.append((m, c, s, e))
    return segments

def pair_line_search(h, i, j, cur_score, n, S):
    dx = 2.0 / n
    M = 2 * n - 1
    q, G = compute_q_G(h, n)
    s = j - i
    a = np.zeros(M)
    a[0] = -2.0 * dx
    a[2*s-1] = dx
    a[2*s] = dx
    b = G[:, i] - G[:, j]
    c = q
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi <= lo:
        return None, cur_score
    # Affine indices
    curved = [0, 2*s-1, 2*s]
    affine_mask = np.ones(M, dtype=bool)
    affine_mask[curved] = False
    aff_idx = np.where(affine_mask)[0]
    aff_slopes = b[aff_idx]
    aff_intercepts = c[aff_idx]
    segments = build_hull(aff_slopes, aff_intercepts, lo, hi)
    # Candidate t values
    cands = set()
    cands.add(lo)
    cands.add(hi)
    cands.add(0.0)
    for seg in segments:
        cands.add(seg[2])
        cands.add(seg[3])
    # Quadratic-line intersections
    for li in curved:
        if abs(a[li]) > 1e-30:
            for (m, c_seg, s1, s2) in segments:
                A2 = a[li]
                B2 = b[li] - m
                C2 = c[li] - c_seg
                for r in solve_quad(A2, B2, C2, s1, s2):
                    cands.add(r)
    # Quadratic-quadratic intersections
    for li in range(3):
        for lj in range(li+1, 3):
            A2 = a[curved[li]] - a[curved[lj]]
            B2 = b[curved[li]] - b[curved[lj]]
            C2 = c[curved[li]] - c[curved[lj]]
            for r in solve_quad(A2, B2, C2, lo, hi):
                cands.add(r)
    # Vertices
    for li in curved:
        if a[li] > 1e-30:
            tv = -b[li] / (2.0 * a[li])
            if lo <= tv <= hi:
                cands.add(tv)
    # Evaluate candidates
    cands_list = sorted(cands)
    cands_list = [cv for cv in cands_list if lo - 1e-12 <= cv <= hi + 1e-12]
    if not cands_list:
        return None, cur_score
    # Evaluate polynomial max at each candidate
    best_t = 0.0
    best_pred = float('inf')
    eval_ts = []
    for t in cands_list:
        t = min(max(t, lo), hi)
        vals = a * t * t + b * t + c
        pm = float(np.max(vals))
        if pm < best_pred:
            best_pred = pm
            best_t = t
        eval_ts.append((pm, t))
    eval_ts.sort()
    # Score top candidates with exact evaluator
    best_score = cur_score
    best_h = None
    pair_sum = h[i] + h[j]
    lo_i = max(0.0, pair_sum - 1.0)
    hi_i = min(1.0, pair_sum)
    trial_count = 0
    for pm, t in eval_ts:
        if trial_count >= 8:
            break
        trial_count += 1
        t = min(max(t, lo), hi)
        hi_new = h[i] + t
        hi_new = min(max(hi_new, lo_i), hi_i)
        hj_new = pair_sum - hi_new
        h_trial = h.copy()
        h_trial[i] = hi_new
        h_trial[j] = hj_new
        if not is_feasible(h_trial, n, S):
            h_trial = capped_simplex(h_trial, S)

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(corr * dx))
    return val if math.isfinite(val) else float('inf')

def is_feasible(h, n, S):
    if h.ndim != 1 or len(h) != n or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False

def capped_simplex(v, S):

    lo_b, hi_b = float(np.min(v)) - 1.0, float(np.max(v))

            lo_b = mid
        else:
            hi_b = mid

        cap = np.maximum(1.0 - h, 0.0)
        tc = float(np.sum(cap))
        if tc > 0:
            h += res * cap / tc
    elif res < -1e-15:
        th = float(np.sum(h))
        if th > 0:
            h += res * (h / th)
    h = np.clip(h, 0.0, 1.0)

        if len(idx) > 0:

    dx = 2.0 / n

        G[2*d-1, d:d+m] += dx * (1.0 - h[:m])
        G[2*d-1, :m] += -dx * h[d:d+m]
        q[2*d] = dx * np.sum(h[:m] * (1.0 - h[d:d+m]))
        G[2*d, :m] += dx * (1.0 - h[d:d+m])
        G[2*d, d:d+m] += -dx * h[:m]

def solve_quad(A, B, C, lo, hi):
    tol = 1e-14 * max(1.0, abs(A), abs(B), abs(C))
    roots = []
    if abs(A) < tol:
        if abs(B) > tol:
            r = -C / B

    else:

        thr = 64.0 * np.finfo(np.float64).eps * (B*B + abs(4.0*A*C) + 1.0)
        if disc < -thr:
            return roots
