def exact_score(h, n):

    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_q_G(h, n):
    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    # Lag 0
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0] = dx * (1.0 - 2.0 * h)
    # Positive lags d=1..n-1 -> rows 1..n-1
    for d in range(1, n):
        row = d  # row index for q_{d,+}
        q[row] = dx * np.sum(h[d:] * (1.0 - h[:-d]))
        G[row, :-d] += -dx * h[d:]  # gradient wrt h[i] for i=0..n-d-1
        G[row, d:] += dx * (1.0 - h[:-d])  # gradient wrt h[i+d] for i=0..n-d-1
    # Negative lags d=1..n-1 -> rows n..2n-2
    for d in range(1, n):
        row = n + d - 1  # row index for q_{d,-}
        q[row] = dx * np.sum(h[:-d] * (1.0 - h[d:]))
        G[row, :-d] += dx * (1.0 - h[d:])  # gradient wrt h[i] for i=0..n-d-1
        G[row, d:] += -dx * h[:-d]  # gradient wrt h[i+d] for i=0..n-d-1
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None, None
    return q, G

def active_bundle_gradient(q, G):
    Cq = float(np.max(q))
    tol = max(1e-12, 1e-7 * max(1.0, abs(Cq)))
    active = np.where(q >= Cq - tol)[0]
    if len(active) > 12:
        order = np.argsort(q[active])[::-1][:12]
        active = active[order]
    U = G[active].copy()
    # Center each row
    U -= np.mean(U, axis=1, keepdims=True)
    K = U @ U.T
    K = 0.5 * (K + K.T)
    m = len(active)
    if m == 0:
        return None
    alpha = np.full(m, 1.0 / m)
    for _ in range(40):
        z = K @ alpha
        j = int(np.argmin(z))
        if float(alpha @ z) - float(z[j]) <= 1e-12:
            break
        v = np.zeros(m)
        v[j] = 1.0
        v -= alpha
        vKv = float(v @ K @ v)
        if vKv <= 1e-24:
            break
        gamma = float(np.clip(-(v @ z) / vKv, 0.0, 1.0))
        alpha += gamma * v
        alpha[alpha < 0] = 0.0
        alpha /= np.sum(alpha)
    g = alpha @ U
    if not np.all(np.isfinite(g)):
        g = U[0]  # fallback to single active gradient
        if not np.all(np.isfinite(g)):
            return None
    return g

def local_search(h0, rng, n, max_iter):
    h = project_capped_simplex(h0, n)
    if h is None:
        return None, None
    cur_score = exact_score(h, n)
    best = h.copy()
    best_score = cur_score
    eta = 0.5
    for it in range(max_iter):
        q, G = compute_q_G(h, n)
        if G is None:
            break
        g = active_bundle_gradient(q, G)
        accepted = False
        if g is not None:
            for b in range(8):
                eta_try = eta * (0.5 ** b)
                cand = project_capped_simplex(h - eta_try * g, n)
                if cand is None:
                    continue
                if np.max(np.abs(cand - h)) < 1e-14:
                    continue
                cs = exact_score(cand, n)
                if not np.isfinite(cs):
                    continue
                if cs < cur_score - 1e-12:
                    h = cand
                    cur_score = cs
                    if cur_score < best_score - 1e-12:
                        best = h.copy()
                        best_score = cur_score
                    eta = min(2.0, 1.25 * eta_try)
                    accepted = True
                    break
        if not accepted:
            eta = max(1e-6, eta / 2.0)

                sigma, T = 0.12, 0.002
            elif it < 80:
                sigma, T = 0.07, 0.0008
            else:
                sigma, T = 0.035, 0.00025
            best_cand = h
            best_cs = cur_score
            for _ in range(4):
                prop = h + sigma * rng.standard_normal(n)
                pc = project_capped_simplex(prop, n)
                if pc is None:
                    continue
                ps = exact_score(pc, n)
                if not np.isfinite(ps):

                if ps < best_cs:
                    best_cand = pc
                    best_cs = ps
            if best_cand is not h:
                delta = best_cs - cur_score

                        best_score = cur_score

        return None, None
    final_score = exact_score(best, n)

    return h

def active_bundle_gradient(q, G):
    Cq = float(np.max(q))
    tol = max(1e-12, 1e-7 * max(1.0, abs(Cq)))
    active = np.where(q >= Cq - tol)[0]
    if len(active) > 12:
        order = np.argsort(q[active])[::-1][:12]
        active = active[order]
    m = len(active)
    if m == 0:
        return None
    U = G[active].copy()
    U -= np.mean(U, axis=1, keepdims=True)
    K = U @ U.T
    K = 0.5 * (K + K.T)
    alpha = np.full(m, 1.0 / m)
    for _ in range(40):
        z = K @ alpha
        j = int(np.argmin(z))
        gap = float(alpha @ z) - float(z[j])
        if gap <= 1e-12:
            break
        v = np.zeros(m)
        v[j] = 1.0
        v -= alpha
        vKv = float(v @ K @ v)
        if vKv <= 1e-24:
            break
        gamma = float(np.clip(-(v @ z) / vKv, 0.0, 1.0))
        alpha += gamma * v
        alpha[alpha < 0] = 0.0
        sn = float(np.sum(alpha))
        if sn < 1e-15:
            alpha = np.full(m, 1.0 / m)
        else:
            alpha /= sn
    g = alpha @ U
    if not np.all(np.isfinite(g)):
        g = U[0]
        if not np.all(np.isfinite(g)):
            return None
    return g

def local_search(h0, rng, n, max_iter):
    h = project_capped_simplex(h0, n)
    if h is None:
        return None, None
    cur_score = exact_score(h, n)
    if not np.isfinite(cur_score):
        return None, None
    best = h.copy()
    best_score = cur_score
    eta = 0.5
    for it in range(max_iter):
        q, G = compute_q_G(h, n)
        if G is None:
            break
        g = active_bundle_gradient(q, G)
        accepted = False
        if g is not None:
            for b in range(8):
                eta_try = eta * (0.5 ** b)
                cand = project_capped_simplex(h - eta_try * g, n)
                if cand is None:
                    continue
                if np.max(np.abs(cand - h)) < 1e-14:
                    continue
                cs = exact_score(cand, n)
                if not np.isfinite(cs):
                    continue
                if cs < cur_score - 1e-12:
                    h = cand
                    cur_score = cs
                    if cur_score < best_score - 1e-12:
                        best = h.copy()
                        best_score = cur_score
                    eta = min(2.0, 1.25 * eta_try)
                    accepted = True
                    break
        if not accepted:
            eta = max(1e-6, eta / 2.0)
            if it < 40:
                sigma, T = 0.12, 0.002
            elif it < 80:
                sigma, T = 0.07, 0.0008
            else:
                sigma, T = 0.035, 0.00025
            best_cand = h
            best_cs = cur_score
            for _ in range(4):
                prop = h + sigma * rng.standard_normal(n)
                pc = project_capped_simplex(prop, n)
                if pc is None:
                    continue
                ps = exact_score(pc, n)
                if not np.isfinite(ps):
                    continue
                if ps < best_cs:
                    best_cand = pc
                    best_cs = ps
            if best_cand is not h:
                delta = best_cs - cur_score
                if delta <= 0.0 or rng.random() < math.exp(-delta / T):
                    h = best_cand
                    cur_score = best_cs
                    eta = 0.25
                    if cur_score < best_score - 1e-12:
                        best = h.copy()
                        best_score = cur_score
    if not is_feasible(best, n):
        return None, None
    fs = exact_score(best, n)
    return best, fs

def add_unique(starts, v):
    if v is None or not np.all(np.isfinite(v)):
        return
    for ex in starts:
        if np.max(np.abs(v - ex)) < 1e-10:
            return
    starts.append(v)

def run(seed=42, **kwargs):
    try:
        seed = int(seed)
    except (TypeError, ValueError):
        seed = 42
    rng = np.random.default_rng(seed)
    n = 42
    parent = None
    pv = globals().get('initial_h_values', None)
    if pv is not None:
        try:
            pv_arr = np.array(pv, dtype=np.float64)
            if pv_arr.ndim == 1 and len(pv_arr) >= 2:
                n = len(pv_arr)
                parent = pv_arr
        except Exception:
            pass
    if parent is None:
        nk = kwargs.get('n_points', None)
        if nk is not None:
            try:
                nc = int(nk)
                if 2 <= nc <= 512:
                    n = nc
            except (TypeError, ValueError):
                pass
    const_h = np.full(n, 0.5)
    const_score = exact_score(const_h, n)
    parent_feasible = None
    parent_score = None
    if parent is not None:
        if np.all(np.isfinite(parent)):
            ph = project_capped_simplex(parent, n)
            if ph is not None and is_feasible(ph, n):
                parent_feasible = ph
                parent_score = exact_score(ph, n)
    incumbent = const_h.copy()
    incumbent_score = const_score
    if parent_feasible is not None and parent_score < const_score - 1e-12:
        incumbent = parent_feasible.copy()
        incumbent_score = parent_score
    if n > 128:
        if not is_feasible(incumbent, n):
            incumbent = const_h
            incumbent_score = const_score
        return incumbent, float(incumbent_score), int(n)
    starts = []
    if parent_feasible is not None:
        add_unique(starts, parent_feasible)
    add_unique(starts, const_h)
    block = np.zeros(n)
    block[:n // 2] = 1.0
    if n % 2 == 1:
        block[n // 2] = 0.5
    add_unique(starts, block)
    alt = np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n)])
    add_unique(starts, project_capped_simplex(alt, n))
    idx = np.arange(n)
    sin_h = 0.5 + 0.45 * np.sin(2.0 * math.pi * idx / n)
    add_unique(starts, project_capped_simplex(sin_h, n))
    if parent_feasible is not None:
        for sig in (0.05, 0.15, 0.35):
            pert = parent_feasible + sig * rng.standard_normal(n)
            add_unique(starts, project_capped_simplex(pert, n))
        for _ in range(2):
            add_unique(starts, project_capped_simplex(rng.random(n), n))
    else:
        for _ in range(4):
            add_unique(starts, project_capped_simplex(rng.random(n), n))
    starts = starts[:12]
    max_iter = 120 if n <= 64 else 80
    for s in starts:
        try:
            rh, rs = local_search(s, rng, n, max_iter)
        except Exception:
            continue
        if rh is None or rs is None:
            continue
        if is_feasible(rh, n) and np.isfinite(rs) and rs < incumbent_score - 1e-12:
            incumbent = rh.copy()
            incumbent_score = rs
    candidates = [(incumbent, incumbent_score), (const_h, const_score)]
    if parent_feasible is not None:
        candidates.insert(1, (parent_feasible, parent_score))
    best_h, best_s = candidates[0][0].copy(), candidates[0][1]
    for ch, cs in candidates:
        if is_feasible(ch, n) and np.isfinite(cs) and cs < best_s - 1e-12:
            best_h, best_s = ch.copy(), cs
    if not is_feasible(best_h, n):
        rp = project_capped_simplex(best_h, n)
        if rp is not None and is_feasible(rp, n):
            best_h = rp
            best_s = exact_score(rp, n)
        else:
            best_h = const_h.copy()
            best_s = const_score
    if not np.isfinite(best_s):
        best_h = const_h.copy()
        best_s = const_score
    return best_h, float(best_s), int(n)
