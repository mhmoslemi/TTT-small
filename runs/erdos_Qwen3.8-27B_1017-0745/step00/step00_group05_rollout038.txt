def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def normalize_direction(d):

def quadratic_roots_in_interval(A, B, C, lo, hi):
    """Find real roots of A*x^2 + B*x + C = 0 in [lo, hi]."""
    tol = 1e-12 * max(1.0, abs(lo), abs(hi))
    roots = []
    if abs(C) < 1e-14 and abs(B) < 1e-14:
        return roots
    if abs(A) < 1e-14:
        # Linear: B*x + C = 0
        if abs(B) > 1e-14:
            r = -C / B
            if lo - tol <= r <= hi + tol:
                r = max(lo, min(hi, r))
                roots.append(r)
        return roots
    disc = B*B - 4.0*A*C
    scale = abs(A)*lo*lo + abs(B)*abs(lo) + abs(C)
    if disc < -1e-10 * max(1.0, scale):
        return roots
    if disc < 0:
        disc = 0.0
    sq = math.sqrt(disc)
    # Use stable formula
    if B >= 0:
        q = -0.5 * (B + sq)
    else:
        q = -0.5 * (B - sq)
    if abs(q) > 1e-14:
        r1 = q / A
        r2 = C / q
    else:
        r1 = -B / (2.0 * A)
        r2 = r1

            r = max(lo, min(hi, r))
            if not any(abs(r - x) < 1e-13 for x in roots):

import numpy as np
import math

def capped_simplex_project(v, target):
    n = len(v)
    if n == 0 or not np.all(np.isfinite(v)):
        return None
    v = np.asarray(v, dtype=np.float64).copy()
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(4):
        r = target - float(np.sum(y))
        if abs(r) < 1e-12:
            break
        for i in range(n):
            if r > 0:
                add = min(r, 1.0 - y[i])
                y[i] += add; r -= add
            elif r < 0:
                sub = min(-r, y[i])
                y[i] -= sub; r += sub
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(float(np.sum(y)) - target) > 1e-10:
        return None
    return y

def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def normalize_direction(d):
    d = np.asarray(d, dtype=np.float64).copy()
    if not np.all(np.isfinite(d)) or float(np.max(np.abs(d))) < 1e-15:
        return None
    d -= float(np.mean(d))
    d[-1] -= float(np.sum(d))
    m = float(np.max(np.abs(d)))
    if m < 1e-15:
        return None
    d /= m
    d[-1] -= float(np.sum(d))
    return d

def quadratic_roots(A, B, C, lo, hi):
    tol = 1e-12 * max(1.0, abs(lo), abs(hi))
    roots = []
    if abs(A) < 1e-14:
        if abs(B) > 1e-14:
            r = -C / B
            if lo - tol <= r <= hi + tol:
                roots.append(max(lo, min(hi, r)))
        return roots
    disc = B * B - 4.0 * A * C
    if disc < 0:
        if disc > -1e-10 * (abs(A) + abs(B) + abs(C)):
            disc = 0.0
        else:
            return roots
    sq = math.sqrt(disc)
    q = -0.5 * (B + sq) if B >= 0 else -0.5 * (B - sq)
    if abs(q) > 1e-14:
        r1, r2 = q / A, C / q
    else:

            r = max(lo, min(hi, r))
            if not any(abs(r - x) < 1e-12 for x in roots):
                roots.append(r)
    return roots

def build_directions(h, rng, n):

    corr = np.correlate(h, 1.0 - h, mode='full')
    top_lags = np.argsort(corr)[::-1][:3]

        t = int(ti) - (n - 1)
        g = np.zeros(n)
        for j in range(n):

                g[j] += (1.0 - h[j - t])
