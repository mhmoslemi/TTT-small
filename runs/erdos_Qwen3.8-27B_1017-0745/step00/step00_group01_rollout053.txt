    z = np.asarray(z, dtype=np.float64)

        return np.full(n, 0.5)
    z = np.clip(z, -1e12, 1e12)
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(z - mid, 0.0, 1.0)
        if np.sum(h) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) < 1e-14 * max(1, n):
            break
        if r > 0:
            slack = 1.0 - h
            idx = int(np.argmax(slack))
            step = min(r, slack[idx])
            if step < 1e-15:
                break
            h[idx] += step
        else:
            idx = int(np.argmax(h))
            step = min(-r, h[idx])
            if step < 1e-15:
                break
            h[idx] -= step
        r = s - np.sum(h)
    h = np.clip(h, 0.0, 1.0)
    if not check_feasibility(h, n):
        return np.full(n, 0.5)
    return h

def exact_score(h):
    n = len(h)
    if not check_feasibility(h, n):
        return np.inf
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return dx * float(np.max(c))

def compute_lse_and_weights(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    a = np.exp((c - c_max) / tau)
    s = np.sum(a)
    if not np.isfinite(s) or s <= 0:
        return np.inf, None
    return c_max + tau * np.log(s), a / s

def lse_gradient(h, P):
    n = len(h)
    dx = 2.0 / n
    U = np.convolve(P, 1.0 - h, mode='full')[n - 1:2 * n - 1]
    V = np.convolve(P[::-1], h, mode='full')[n - 1:2 * n - 1]
    g = dx * (U - V)
    if not np.all(np.isfinite(g)):
        return None
    return g

def build_starts(base, n, rng, max_starts):
    cands = []
    cands.append(base.copy())
    cands.append(base[::-1].copy())
    cands.append(1.0 - base[::-1].copy())
    cands.append(np.full(n, 0.5))
    b = np.zeros(n)
    b[:n // 2] = 1.0
    if n % 2 == 1:
        b[n // 2] = 0.5
    cands.append(b.copy())
    alt = np.zeros(n)
    alt[::2] = 1.0
    cands.append(alt.copy())
    for sd in [0.05, 0.15, 0.35]:
        cands.append(base + sd * rng.standard_normal(n))
    cands.append(rng.uniform(0.0, 1.0, n))
    starts = []
    for c in cands:
        p = project_capped_simplex(c, n / 2.0, n)
        dup = False
        for s in starts:
            if np.max(np.abs(p - s)) < 1e-10:
                dup = True
                break
        if not dup:
            starts.append(p)
        if len(starts) >= max_starts:
            break
    return starts

def grad_descent(x, tau, iters, n):
    f, P = compute_lse_and_weights(x, tau)
    if not np.isfinite(f) or P is None:
        return x
    g = lse_gradient(x, P)
    if g is None:
        return x
    alpha = 1.0
    small_count = 0
    f_prev_track = f
    for _ in range(iters):
        if not np.isfinite(f) or g is None:
            break
        accepted = False
        alpha_local = alpha
        for _bt in range(24):
            y = project_capped_simplex(x - alpha_local * g, n / 2.0, n)
            D = float(np.sum((y - x) ** 2))
            if np.max(np.abs(y - x)) <= 1e-9:
                return x
            fy, _ = compute_lse_and_weights(y, tau)
            thresh = 1e-4 * D / alpha_local + 1e-14 * max(1.0, abs(f))
            if np.isfinite(fy) and fy <= f - thresh:
                x = y
                f = fy
                small_count = small_count + 1 if (f_prev_track - f) <= 1e-12 * max(1.0, abs(f)) else 0
                f_prev_track = f
                f_new, P2 = compute_lse_and_weights(x, tau)
                g = lse_gradient(x, P2) if P2 is not None else None
                alpha = min(2.0, 1.25 * alpha_local)
                accepted = True
                if g is None or small_count >= 5:
                    return x
                break
            alpha_local *= 0.5
            if alpha_local < 1e-12:
                break
        if not accepted:
            break
    return x

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    n = None
    base = None
    try:
        ival = np.array(initial_h_values, dtype=np.float64).ravel()
        if ival.size > 0:
            n = ival.size
            base = ival.copy() if np.all(np.isfinite(ival)) else np.full(n, 0.5)
    except Exception:
        pass
    if n is None:
        n = 47
        base = np.full(n, 0.5)
        try:
            nv = int(kwargs.get('n_points', 47))
            if 1 <= nv <= 999:
                n = nv
                base = np.full(n, 0.5)
        except Exception:
            pass
    n_half = n / 2.0
    base = project_capped_simplex(base, n_half, n)
    if n <= 64:
        max_starts, iters_tau = 10, 100
        cap_s, cap_i = 12, 150
    elif n <= 256:
        max_starts, iters_tau = 5, 60
        cap_s, cap_i = 6, 90
    else:
        max_starts, iters_tau = 2, 25
        cap_s, cap_i = 2, 40
    try:
        rs = int(kwargs.get('restarts', max_starts))
        max_starts = max(1, min(rs, cap_s))
    except Exception:
        pass
    try:
        ip = int(kwargs.get('iters_per_tau', iters_tau))
        iters_tau = max(1, min(ip, cap_i))
    except Exception:
        pass
    fb = base.copy()
    fb_score = exact_score(fb)
    if not np.isfinite(fb_score):
        fb = np.full(n, 0.5)
        fb_score = exact_score(fb)
    incumbent = fb.copy()
    incumbent_score = fb_score
    if n <= 999:
        starts = build_starts(base, n, rng, max_starts)
        temps = [0.02, 0.005, 0.001, 0.0002, 0.00005, 0.00001]
        for st in starts:
            try:
                x = project_capped_simplex(st, n_half, n)
                sc = exact_score(x)
                if np.isfinite(sc) and sc < incumbent_score - 1e-12:
                    incumbent = x.copy()
                    incumbent_score = sc
                for tau in temps:
                    x = grad_descent(x, tau, iters_tau, n)
                    x = project_capped_simplex(x, n_half, n)
                    sc = exact_score(x)
                    if np.isfinite(sc) and sc < incumbent_score - 1e-12:
                        incumbent = x.copy()
                        incumbent_score = sc
            except Exception:
                continue
    incumbent = project_capped_simplex(incumbent, n_half, n)
    final_score = exact_score(incumbent)
    if not np.isfinite(final_score) or final_score > fb_score:
        incumbent = fb.copy()
        final_score = fb_score
    if not np.isfinite(final_score):
        incumbent = np.full(n, 0.5)
        final_score = exact_score(incumbent)
    return incumbent.astype(np.float64), float(final_score), int(n)
