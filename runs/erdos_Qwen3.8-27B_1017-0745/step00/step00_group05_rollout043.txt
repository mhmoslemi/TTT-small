        f = capped_simplex_project(incumbent, target)
        if f is not None and is_feasible(f, target): incumbent = f
        fs = exact_score(incumbent, dx)
        if fs is None or not math.isfinite(fs): incumbent = fallback; fs = fb_sc
        return (np.asarray(incumbent, dtype=np.float64), float(fs), n)
    T_min, T_max = 2e-5, 0.02
    temps = np.array([T_min*(T_max/T_min)**(r/(R-1)) for r in range(R)])
    radii = np.array([min(0.20, 0.015+0.18*math.sqrt(temps[r]/T_max)) for r in range(R)])
    B = max(200, min(12000, int(200000000/(n*n))))
    max_att = 4 * B; max_sw = int(math.ceil(4.0*B/R)) + 1
    stag = 0; stag_lim = max(50, 2*n); restarts = 0; sweep = 0; att = 0
    while sweep < max_sw and att < max_att and evals < B:
        sweep += 1; imp = False
        for r in range(R):
            if evals >= B or att >= max_att or inc_sc <= 0.3808: break
            att += 1
            cand = propose_move(states[r], radii[r], rng, n)
            if cand is None: continue
            sc = exact_score(cand, dx); evals += 1
            if sc is None or not math.isfinite(sc): continue
            if sc < inc_sc: inc_sc = sc; incumbent = cand.copy(); imp = True
            dlt = sc - scores[r]
            if dlt <= 0:
                states[r] = cand; scores[r] = sc
            else:
                u = float(rng.random())
                if u == 0.0 or math.log(u) < -dlt/temps[r]:
                    states[r] = cand; scores[r] = sc
        if inc_sc <= 0.3808: break
        stag = 0 if imp else stag + 1
        if sweep % 5 == 0:
            par = (sweep // 5) % 2
            if par == 0:
                pairs = [(0,1),(2,3),(4,5),(6,7)]
            else:
                pairs = [(1,2),(3,4),(5,6)]
            for (ar, br) in pairs:
                if ar >= R or br >= R: continue
                L = (scores[ar]-scores[br])*(1.0/temps[ar]-1.0/temps[br])
                do = (L >= 0)
                if not do:
                    u = float(rng.random())
                    do = (u == 0.0 or math.log(u) < L)
                if do:
                    states[ar], states[br] = states[br].copy(), states[ar].copy()
                    scores[ar], scores[br] = scores[br], scores[ar]
        if stag >= stag_lim and restarts < 2 and evals + R < B:
            restarts += 1
            ns = [None]*R; nsc = [None]*R
            ns[0] = incumbent.copy()
            sigs = [0.02, 0.04, 0.08, 0.16, 0.28]
            for sl in range(1, 6):
                pr = capped_simplex_project(incumbent + rng.standard_normal(n)*sigs[sl-1], target)
                ns[sl] = pr if pr is not None else incumbent.copy()
            pr = capped_simplex_project(rng.beta(0.5, 0.5, size=n), target)
            ns[6] = pr if pr is not None else incumbent.copy()
            if restarts == 1:
                prof = rng.random(n)
            else:
                blk = np.zeros(n); blk[:n//2] = 1.0
                if n % 2: blk[n//2] = 0.5
                prof = np.roll(blk, int(rng.integers(0, n)))
            pr = capped_simplex_project(prof, target)
            ns[7] = pr if pr is not None else incumbent.copy()
            for sl in range(R):
                if evals >= B: break
                sc = exact_score(ns[sl], dx); evals += 1
                nsc[sl] = sc if sc is not None else 1.0
                if nsc[sl] is not None and nsc[sl] < inc_sc:
                    inc_sc = nsc[sl]; incumbent = ns[sl].copy()
            states, scores = ns, nsc; stag = 0
            if inc_sc <= 0.3808: break
    fin = capped_simplex_project(incumbent, target)
    if fin is not None and is_feasible(fin, target):
        h_out = np.clip(fin, 0.0, 1.0)
    else:
        h_out = np.clip(fallback, 0.0, 1.0)
    c5 = exact_score(h_out, dx)
    if c5 is None or not math.isfinite(c5):
        h_out = np.clip(fallback, 0.0, 1.0); c5 = fb_sc
        if c5 is None or not math.isfinite(c5): c5 = 0.5
    h_out = np.asarray(h_out, dtype=np.float64)
    return (h_out, float(c5), n)
