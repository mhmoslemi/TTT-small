def exact_score(h, dx):

        return float('inf')

            h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def lag_vals(h, dx):

def sm_val(q, tau):

        return float('inf')
    w = np.exp((q - m) / tau)
    s = w.sum()

        if p[k] < 1e-15:
            continue
        ell = k - (n - 1)
        i1, v1 = j - ell, (j - ell >= 0) & (j - ell < n)
        i2, v2 = j + ell, (j + ell >= 0) & (j + ell < n)
        t1, t2 = np.zeros(n), np.zeros(n)
        t1[v1] = 1.0 - h[i1[v1]]
        t2[v2] = h[i2[v2]]
        g += p[k] * dx * (t1 - t2)
    return g

def continuation(h0, n, dx, taus, mi):
    h = h0.copy()
    S = n / 2.0
    for tau in taus:
        a, st = 1.0, 0
        for _ in range(mi):
            q = lag_vals(h, dx)
            F = sm_val(q, tau)
            if not np.isfinite(F):
                break
            g = sm_grad(h, n, dx, tau)
            if not np.all(np.isfinite(g)):
                break
            y = project_K(h - a * g, n, S)
            d = y - h
            if np.max(np.abs(d)) < 1e-10:
                break
            gd = g @ d
            if gd > -1e-14:
                break
            Fy = sm_val(lag_vals(y, dx), tau)
            if Fy <= F + 1e-4 * gd + 1e-14:
                st = st + 1 if (F - Fy) < 1e-12 else 0
                h = y
                if st >= 10:
                    break
                a = min(1.25 * a, 5.0)
            else:
                a *= 0.5
                if a < 1e-8:
                    break
    return h

def sop_obj(z, n, dx):
    return z[-1]

def sop_obj_jac(z, n, dx):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def sop_eq(z, n, S):
    return z[:n].sum() - S

def sop_eq_jac(z, n, S):
    g = np.ones(len(z))
    g[-1] = 0.0
    return g

def sop_ineq(z, n, dx):
    return z[-1] - lag_vals(z[:n], dx)

def sop_ineq_jac(z, n, dx):
    h = z[:n]
    nl = 2 * n - 1
    J = np.zeros((nl, n + 1))
    j = np.arange(n)
    for k in range(nl):
        ell = k - (n - 1)
        i1, v1 = j - ell, (j - ell >= 0) & (j - ell < n)
        i2, v2 = j + ell, (j + ell >= 0) & (j + ell < n)
        t1, t2 = np.zeros(n), np.zeros(n)
        t1[v1] = 1.0 - h[i1[v1]]
        t2[v2] = h[i2[v2]]
        J[k, :n] = -dx * (t1 - t2)
        J[k, n] = 1.0
    return J

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    init = globals().get('initial_h_values', None)
    nd = kwargs.get('n_points', 45)
    if not isinstance(nd, (int, np.integer)):
        nd = 45
    nd = max(2, min(999, int(nd)))
    ok = False
    if init is not None:
        init = np.array(init, dtype=np.float64).ravel()
        if 2 <= len(init) <= 999 and np.all(np.isfinite(init)) and np.max(np.abs(init)) < 1e6:
            ok = True
    n = len(init) if ok else nd
    if not ok:
        init = np.full(n, 0.5)
    dx, S = 2.0 / n, n / 2.0
    hp = project_K(init.copy(), n, S)
    best_h, best_s = hp.copy(), exact_score(hp, dx)
    fb_h, fb_s = hp.copy(), best_s

    starts = [best_h.copy(), (1.0 - best_h).copy(), best_h[::-1].copy(), (1.0 - best_h[::-1]).copy()]
    if ok:
        starts.append(project_K(best_h + 0.08 * rng.standard_normal(n), n, S))
        starts.append(project_K(best_h + 0.25 * rng.standard_normal(n), n, S))
    starts.append(project_K(rng.uniform(0, 1, n), n, S))
    starts.append(project_K(rng.beta(0.5, 0.5, n), n, S))
    if not ok:
        for _ in range(3):
            starts.append(project_K(rng.uniform(0, 1, n), n, S))
        for _ in range(2):
            starts.append(project_K(rng.beta(0.5, 0.5, n), n, S))
        starts.append(project_K(0.5 + 0.35 * rng.standard_normal(n), n, S))

    for s in starts:
        if feasible(s, n, S):
            sc = exact_score(s, dx)
            if sc < best_s - 1e-12:
                best_s, best_h = sc, s.copy()

    T = [0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4, 5e-5, 1e-5] if n <= 100 else [0.02, 0.005, 0.001, 1e-4, 1e-5]
    MI = 80 if n <= 100 else 20
    local = []
    for s in starts:
        if not feasible(s, n, S):
            continue
        hopt = project_K(continuation(s, n, dx, T, MI), n, S)
        if feasible(hopt, n, S):
            sc = exact_score(hopt, dx)
            if sc < best_s - 1e-12:
                best_s, best_h = sc, hopt.copy()
            local.append((sc, hopt.copy()))

    perts = [0.04, 0.12, 0.30, 0.50] if n <= 100 else [0.12]
    for sig in perts:
        hp2 = project_K(best_h + sig * rng.standard_normal(n), n, S)
        if feasible(hp2, n, S):
            hopt = project_K(continuation(hp2, n, dx, T, MI), n, S)
            if feasible(hopt, n, S):
                sc = exact_score(hopt, dx)
                if sc < best_s - 1e-12:
                    best_s, best_h = sc, hopt.copy()
                local.append((sc, hopt.copy()))

    if n <= 100:
        local_scores = [x[0] for x in local]
        order = list(np.argsort(local_scores))
        local = [local[i] for i in order]
        sel = []
        for sc, hv in local:
            if len(sel) >= 3:
                break
            close = False
            for _, sh in sel:
                if np.max(np.abs(hv - sh)) <= 1e-5:
                    close = True
                    break
            if not close:
                sel.append((sc, hv))
        if len(sel) == 0:
            sel.append((best_s, best_h.copy()))
        for sc, hv in sel:
            try:
                z0 = np.concatenate([hv, [min(1.0, sc + 1e-8)]])
                bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
                con = [{'type': 'eq', 'fun': sop_eq, 'jac': sop_eq_jac, 'args': (n, S)},
                       {'type': 'ineq', 'fun': sop_ineq, 'jac': sop_ineq_jac, 'args': (n, dx)}]
                res = minimize(sop_obj, z0, jac=sop_obj_jac, args=(n, dx), method='SLSQP',
                               bounds=bounds, constraints=con,
                               options={'maxiter': 200, 'ftol': 1e-10, 'disp': False})
                if res is not None and hasattr(res, 'x') and np.all(np.isfinite(res.x)):
                    hs = project_K(res.x[:n].copy(), n, S)
                    if feasible(hs, n, S):
                        ssc = exact_score(hs, dx)
                        if ssc < best_s - 1e-12:
                            best_s, best_h = ssc, hs.copy()
                        local.append((ssc, hs.copy()))
            except Exception:
                pass

    best_h = project_K(best_h, n, S)
    if not feasible(best_h, n, S):
        best_h, best_s = fb_h.copy(), fb_s
    else:
        fs = exact_score(best_h, dx)
        if fs > fb_s + 1e-10:
            best_h, best_s = fb_h.copy(), fb_s
        else:
            best_s = fs
    best_h = np.clip(best_h, 0.0, 1.0)
    final_s = exact_score(best_h, dx)
    return (best_h, final_s, n)
