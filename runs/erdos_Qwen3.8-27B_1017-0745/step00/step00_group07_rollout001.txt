    spec *= np.exp(-0.5 * (freq / cutoff) ** 2)
    noise = np.irfft(spec, n)
    std = noise.std()
    if std > 1e-12: noise /= std
    y = project_box_sum(0.5 + 0.35 * noise, S, zero, one)
    if y is not None and all(np.max(np.abs(y - s)) > 1e-10 for s in starts):
        starts.append(y)
    perm = rng.permutation(n)
    zb = np.zeros(n)
    zb[perm[:n // 2]] = 1.0
    if n % 2 == 1: zb[perm[n // 2]] = 0.5
    if check_feasible(zb) and all(np.max(np.abs(zb - s)) > 1e-10 for s in starts):
        starts.append(zb.copy())
    mx = 6 if n <= 128 else (4 if n <= 256 else 2)
    return starts[:mx]

def slp_search(x, score_x, n, max_iters):
    dx = 2.0 / n
    S = n / 2.0
    Delta = 0.20
    Dmin, Dmax = 1e-4, 0.5
    best_y = x.copy()
    best_score = score_x
    fail_count = 0
    consec_small = 0
    for _ in range(max_iters):
        if Delta < Dmin: break
        lower = np.maximum(0.0, x - Delta)
        upper = np.minimum(1.0, x + Delta)
        B, G = lag_model(x)
        m = len(B)
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = G @ x - B
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([S])
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        bounds = list(zip(lower, upper)) + [(None, None)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'max_iter': 500})
        except Exception:
            break
        if res.status != 0 or not np.all(np.isfinite(res.x)):
            Delta = 0.25 * Delta; fail_count += 1
            if fail_count >= 3: break
            continue
        y = project_box_sum(res.x[:n], S, lower, upper)
        if y is None:
            Delta = 0.25 * Delta; fail_count += 1
            if fail_count >= 3: break
            continue
        Qm = float(np.max(B + G @ (y - x)))
        pred = dx * (float(np.max(B)) - Qm)
        score_y = exact_score(y)
        if score_y is None:
            Delta = 0.25 * Delta; fail_count += 1
            if fail_count >= 3: break
            continue
        if score_y < best_score - 1e-14 * max(1.0, abs(best_score)):
            best_y = y.copy(); best_score = score_y
        actual = score_x - score_y
        ptol = 1e-12 * max(1.0, abs(score_x))
        if pred <= ptol:
            if actual > 1e-12:
                best_y = y.copy(); best_score = score_y
            break
        rho = actual / pred
        if actual > 1e-12 * max(1.0, abs(score_x)) and rho >= 0.1:
            fail_count = 0
            step = np.max(np.abs(y - x))
            if rho >= 0.75 and step >= 0.8 * Delta:
                Delta = min(2.0 * Delta, Dmax)
            elif rho < 0.25:
                Delta = max(0.25 * Delta, Dmin)
            if abs(actual) < 1e-10 * max(1.0, abs(score_x)):
                consec_small += 1
            else:
                consec_small = 0
            if consec_small >= 8: break
            x = y; score_x = score_y
        else:
            Delta = 0.25 * Delta; fail_count += 1
            if fail_count >= 3: break
    return best_y, best_score

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = None
    n = 80
    gv = globals().get('initial_h_values', None)
    if gv is not None:
        try:
            pv = np.asarray(gv, dtype=np.float64).ravel()
            if pv.ndim == 1 and pv.size >= 2 and np.all(np.isfinite(pv)):
                parent = pv.copy()
                n = pv.size
        except Exception:
            pass
    if parent is None and 'n_points' in kwargs:
        try:
            nv = int(kwargs['n_points'])
            if 8 <= nv <= 512: n = nv
        except Exception:
            pass
    S = n / 2.0
    const_vec = np.full(n, 0.5)
    fallback_score = exact_score(const_vec)
    if parent is not None:
        rep = project_box_sum(parent, S, np.zeros(n), np.ones(n))
        if rep is not None: parent = rep
    if n > 512:
        return const_vec, fallback_score, n
    starts = make_starts(n, parent, rng)
    best_h = const_vec.copy()
    best_c5 = fallback_score
    if parent is not None:
        ps = exact_score(parent)
        if ps is not None and ps < best_c5:
            best_h = parent.copy(); best_c5 = ps
    max_iters = 36 if n <= 128 else (24 if n <= 256 else 12)
    for s in starts:
        try:
            ss = exact_score(s)
            if ss is None: continue
            y, sc = slp_search(s, ss, n, max_iters)
            if sc < best_c5 - 1e-14 * max(1.0, abs(best_c5)):
                best_h = y; best_c5 = sc
        except Exception:
            continue
    if not check_feasible(best_h):
        best_h = const_vec
        best_c5 = fallback_score
    fc = exact_score(best_h)
    if fc is None:
        best_h = const_vec
        fc = fallback_score
    best_h = np.asarray(best_h, dtype=np.float64).ravel()
    return best_h, float(fc), int(len(best_h))
