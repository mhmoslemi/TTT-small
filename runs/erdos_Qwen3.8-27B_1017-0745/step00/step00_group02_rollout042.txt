            if prob.status not in ('optimal', 'optimal_inaccurate'):
                continue
            if d.value is None or t.value is None:
                continue
            dv = np.asarray(d.value, dtype=np.float64).ravel()
            tv = float(t.value)
            if dv.shape != (n,) or not np.all(np.isfinite(dv)):
                continue
            aff = q + G @ dv
            if not np.all(np.isfinite(aff)):
                continue
            if float(aff.max()) > tv + 1e-5:
                continue
            if float(aff.max()) + dx * float(np.dot(dv, dv)) > C + 1e-5:
                continue
            if float(dv.min()) < -1.0 - 1e-5 or float(dv.max()) > 1.0 + 1e-5:
                continue
            hx = h + dv
            if float(hx.min()) < -1e-5 or float(hx.max()) > 1.0 + 1e-5:
                continue
            if abs(float(hx.sum()) - S) > 1e-5:
                continue
            return dv
        except Exception:
            continue
    return None

def optimize_start(h0, dx, S, n, max_iter, max_bt):
    h = h0.copy()
    _, C_now = compute_score(h, dx)
    if C_now is None:
        return h, C_now
    stall = 0
    for _ in range(max_iter):
        q, C_now = compute_score(h, dx)
        if C_now is None:
            break
        G = compute_G(h, dx, n)
        if not np.all(np.isfinite(G)):
            break
        d = solve_qp(h, q, G, C_now, dx, S, n)
        if d is None:
            break
        h_proj = project_K(h + d, n, S)
        if h_proj is None:
            break
        d_feas = h_proj - h
        if float(np.abs(d_feas).max()) <= 1e-9:
            break
        best_sc = C_now
        best_cand = None
        for k in range(max_bt + 1):
            alpha = 2.0 ** (-k)
            cand = project_K(h + alpha * d_feas, n, S)
            if cand is None:
                continue
            _, sc = compute_score(cand, dx)
            if sc is not None and sc < best_sc:
                best_sc = sc
                best_cand = cand
        if best_cand is None:
            break
        if best_sc >= C_now - 1e-12:
            break
        dec = C_now - best_sc
        h = best_cand
        C_now = best_sc
        if dec < 1e-10:
            stall += 1
            if stall >= 3:
                break
        else:
            stall = 0
    return h, C_now

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)

    n = 45
    parent = None
    ihv = globals().get('initial_h_values', None)
    if ihv is not None:
        try:
            arr = np.array(ihv, dtype=np.float64).ravel()
            if 2 <= arr.size <= 999 and np.all(np.isfinite(arr)) and float(np.abs(arr).max()) <= 1e6:
                n = arr.size
                S_tmp = n / 2.0
                if np.all(arr >= -1e-12) and np.all(arr <= 1.0 + 1e-12) and abs(float(arr.sum()) - S_tmp) <= 1e-12:
                    parent = np.clip(arr, 0.0, 1.0)
                else:
                    proj = project_K(arr, n, S_tmp)
                    if proj is not None:
                        parent = proj
        except Exception:
            pass

    if parent is None:
        nk = kwargs.get('n_points', None)
        if isinstance(nk, int) and not isinstance(nk, bool) and 2 <= nk <= 999:
            n = nk

    dx = 2.0 / n
    S = n / 2.0

    if parent is not None:
        fallback = parent.copy()
    else:
        fallback = 0.5 * np.ones(n)

    _, fb_score = compute_score(fallback, dx)
    if fb_score is None or not np.isfinite(fb_score):
        fallback = 0.5 * np.ones(n)
        _, fb_score = compute_score(fallback, dx)

    best_h = fallback.copy()
    best_score = fb_score

    max_starts = 6 if n <= 64 else (4 if n <= 256 else 3)
    max_iter = 40 if n <= 64 else (20 if n <= 256 else 8)
    max_bt = 9 if n <= 64 else (6 if n <= 256 else 4)

    starts = []
    def add_start(hc):
        if hc is None:
            return
        proj = project_K(hc, n, S)
        if proj is None:
            return
        for existing in starts:
            if float(np.abs(proj - existing).max()) <= 1e-8:
                return
        if len(starts) < max_starts:
            starts.append(proj.copy())

    if parent is not None:
        add_start(parent)
        add_start(1.0 - parent)
        add_start(parent[::-1].copy())
        add_start(parent + 0.08 * rng.standard_normal(n))
        add_start(parent + 0.25 * rng.standard_normal(n))
        add_start(rng.beta(0.5, 0.5, size=n))
    else:
        add_start(0.5 * np.ones(n))
        add_start(rng.uniform(0, 1, size=n))
        add_start(rng.beta(0.5, 0.5, size=n))
        m_mode = int(rng.integers(1, max(1, min(4, n // 2)) + 1))
        phase = float(rng.uniform(0, 2 * math.pi))
        idx = np.arange(n, dtype=np.float64)
        trig = 0.5 + 0.35 * np.cos(2.0 * math.pi * m_mode * (idx + 0.5) / n + phase)
        add_start(trig)

    if not starts:
        starts.append(fallback.copy())

    archive = [(fallback.copy(), fb_score)]

    for si in range(len(starts)):
        sh = starts[si]
        _, sc = compute_score(sh, dx)
        if sc is None:
            continue
        if sc < all_x[1] for all_x in archive:
            pass
        archived = False
        for ah, asc in archive:
            if sc < asc - 1e-12 and float(np.abs(sh - ah).max()) > 1e-8:
                archive.append(sh.copy())
                archived = True
                break
        if not archived and (not archive or sc < min(a[1] for a in archive) - 1e-12):
            archive.append(sh.copy())
        if sc < best_score - 1e-12:
            best_score = sc
            best_h = sh.copy()

    for si in range(len(starts)):
        try:
            res_h, res_sc = optimize_start(starts[si], dx, S, n, max_iter, max_bt)
        except Exception:
            continue
        if res_sc is None or not np.isfinite(res_sc):
            continue
        if not is_feasible(res_h, S, n):
            proj = project_K(res_h, n, S)
            if proj is None:
                continue
            res_h = proj
            _, res_sc = compute_score(res_h, dx)
            if res_sc is None:
                continue
        if res_sc < best_score - 1e-12:
            best_score = res_sc
            best_h = res_h.copy()
        archived = False
        for ah, asc in archive:
            if res_sc < asc - 1e-12 and float(np.abs(res_h - ah).max()) > 1e-8:
                archive.append(res_h.copy())
                archived = True
                break
        if not archived:
            archive.append(res_h.copy())

    if not is_feasible(best_h, S, n):
        proj = project_K(best_h, n, S)
        if proj is not None:
            best_h = proj
        else:
            best_h = fallback.copy()

    if not is_feasible(best_h, S, n):
        best_h = fallback.copy()

    final_h = best_h.copy()
    final_h = np.clip(final_h, 0.0, 1.0)
    final_h = np.asarray(final_h, dtype=np.float64)
    _, final_sc = compute_score(final_h, dx)
    if final_sc is None or not np.isfinite(final_sc):
        final_h = fallback.copy()
        _, final_sc = compute_score(final_h, dx)
        if final_sc is None:
            final_sc = 1.0

    return final_h, float(final_sc), int(n)
