                    eb = min(0.25, 1.25 * eb) if be >= 0.75 * eb else max(em, 1.25 * be)
                    if Cx < best_score - 1e-12:
                        best, best_score = x.copy(), Cx
                else:
                    eb, ni = max(em, 0.5 * eb), ni + 1
            except Exception:
                ni += 1

    fp = project_capped_simplex(best, N)
    best = fp if fp is not None and is_feasible(fp, N) else fb
    cs = exact_c5(best, N)
    return (best, float(cs) if np.isfinite(cs) else float('inf'), N)
