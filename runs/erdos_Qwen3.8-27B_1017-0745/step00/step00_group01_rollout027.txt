        max_starts, iters_per_tau = min(int(kwargs.get('restarts', 5)), 6), min(int(kwargs.get('iters_per_tau', 60)), 90)
    else:
        max_starts, iters_per_tau = min(int(kwargs.get('restarts', 2)), 2), min(int(kwargs.get('iters_per_tau', 25)), 40)

    tau_schedule = [0.02, 0.005, 0.001, 0.0002, 0.00005, 0.00001]

    if parent is not None and np.all(np.isfinite(parent)):
        h_inc = project_capped_simplex(parent.copy(), n / 2.0, n)
    else:
        h_inc = np.full(n, 0.5)
    sc_inc = exact_score(h_inc)
    h_fb = h_inc.copy()
    sc_fb = sc_inc

    starts = generate_starts(n, rng, parent)
    starts = starts[:max_starts]

    for start in starts:
        sc_s = exact_score(start)
        if sc_s < sc_inc - 1e-12:
            h_inc = start.copy()
            sc_inc = sc_s
        x = start.copy()
        for tau in tau_schedule:
            x = pgd_stage(x, tau, iters_per_tau)
            sc_x = exact_score(x)
            if sc_x < sc_inc - 1e-12:
                h_inc = x.copy()
                sc_inc = sc_x

    h_inc = project_capped_simplex(h_inc, n / 2.0, n)
    sc_inc = exact_score(h_inc)
    sc_fb = exact_score(h_fb)
    if sc_fb < sc_inc:
        h_inc = h_fb.copy()
        sc_inc = sc_fb

    if not check_feasible(h_inc) or not np.isfinite(sc_inc):
        h_inc = np.full(n, 0.5)
        sc_inc = exact_score(h_inc)

    return (h_inc, sc_inc, n)
