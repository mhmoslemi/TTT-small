- Index k in the output corresponds to lag d = k - (n-1)
- For lag d (where d goes from -(n-1) to n-1):

def project_capped_simplex(z, n, s):
    # Project z onto {h in [0,1]^n : sum(h) = s}
    if z is None or len(z) != n or not np.all(np.isfinite(z)):
        return np.full(n, 0.5)
    z = np.asarray(z, dtype=np.float64)
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(80):
        mid = 0.5*(lo+hi)
        c = np.clip(z - mid, 0.0, 1.0)
        if c.sum() > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    # Residual correction
    for _ in range(n + 2):
        r = s - h.sum()
        if abs(r) < 1e-12:
            break
        if r > 0:
            idx = np.argmax(1.0 - h)
            add = min(r, 1.0 - h[idx])
            h[idx] += add
        else:
            idx = np.argmax(h)
            sub = min(-r, h[idx])
            h[idx] -= sub
    if not np.all(np.isfinite(h)) or h.min() < -1e-12 or h.max() > 1+1e-12 or abs(h.sum()-s) > 1e-9*max(1,n):
        return np.full(n, 0.5)
    h = np.clip(h, 0.0, 1.0)
    return h

def exact_score(h):

    if n == 0 or not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def corr_and_jac(h):

    q = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    J = np.zeros((2*n-1, n), dtype=np.float64)

        j0 = max(-d, 0)
        m = n - abs(d)
        for r in range(m):

            J[k, j0+r] -= h[i0+r]
    return q, J
