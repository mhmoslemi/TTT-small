```python
import numpy as np
from scipy.optimize import minimize
import functools

def _exact_score(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * (2.0/n))

def _corr_vec(h, d):
    n = len(h)
    return np.correlate(h, 1.0-h, mode='full') * d

def _corr_jac(h, d):
    n = len(h)
    J = np.zeros((2*n-1, n))
    for q in range(-n+1, n):
        row = q + n - 1
        j_lo = max(0, -q)
        j_hi = min(n, n-q)
        for j in range(j_lo, j_hi):
            J[row, j] += d*(1.0 - h[j+q])
            J[row, j+q] -= d*h[j]
    return J
```

```python
def _project(x, n, m):
    # Project onto {0<=y<=1, sum(y)=m}
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.shape != (n,) or not np.all(np.isfinite(x)):
        return np.full(n, 0.5)
    if np.abs(x).max() > 1e6:
        return np.full(n, 0.5)
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x))
    for _ in range(70):
        mid = 0.5*(lo+hi)
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > m:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5*(lo+hi), 0.0, 1.0)
    # Fix residual
    delta = m - y.sum()
    idx = np.arange(n)
    if delta > 0:
        for i in idx:
            add = min(delta, 1.0 - y[i])
            y[i] += add
            delta -= add
            if abs(delta) < 1e-15:
                break
    elif delta < 0:
        for i in idx:
            sub = min(-delta, y[i])
            y[i] -= sub
            delta += sub
            if abs(delta) < 1e-15:
                break
    return y
```
