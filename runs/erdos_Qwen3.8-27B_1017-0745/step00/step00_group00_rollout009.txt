                    if sc < best_sc - 1e-12 * max(1.0, abs(best_sc)):
                        best_h, best_sc = pr.copy(), sc
                    wi = int(np.argmax(scores))
                    mi = int(np.argmin(scores))
                    if wi == mi:
                        wi = (wi + 1) % P
                    X[wi], scores[wi] = pr, sc
                except Exception:
                    continue
            rlvl, no_imp = min(4, rlvl + 1), 0
            Fmean, CRmean = min(0.95, Fmean + 0.1), min(0.95, CRmean + 0.05)
    # Final validation
    fh = project_box_sum(best_h, S)
    if fh is not None and is_feasible(fh, n):
        fs = exact_score(fh, n)
        if np.isfinite(fs):
            best_h, best_sc = fh, fs
    else:
        best_h, best_sc = fallback_h.copy(), fb_score
    fp = project_box_sum(fallback_h, S)
    if fp is not None:
        fsc = exact_score(fp, n)
        if np.isfinite(fsc) and fsc < best_sc:
            best_h, best_sc = fp, fsc
    if not np.isfinite(best_sc):
        best_h, best_sc = np.full(n, 0.5, dtype=np.float64), exact_score(np.full(n, 0.5, dtype=np.float64), n)
    return (best_h.astype(np.float64), float(best_sc), n)
