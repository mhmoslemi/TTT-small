        return h, cur_score
    hs, ds = h[mask], d[mask]
    r0 = -hs / ds
    r1 = (1.0 - hs) / ds
    a_lo = np.max(np.minimum(r0, r1))
    a_hi = np.min(np.maximum(r0, r1))
    if not (np.isfinite(a_lo) and np.isfinite(a_hi)) or (a_hi - a_lo) <= 1e-14:
        return h, cur_score
    a_lo, a_hi = max(a_lo, -1e10), min(a_hi, 1e10)
    if a_hi - a_lo <= 1e-14:
        return h, cur_score
    A, B, C = lag_coefficients(h, d, N)
    if not (np.all(np.isfinite(A)) and np.all(np.isfinite(B)) and np.all(np.isfinite(C))):
        return h, cur_score
    M = len(A)
    alphas = [a_lo, a_hi, 0.0]
    for i in range(M):
        if abs(C[i]) > 1e-15:
            sp = -B[i] / (2.0 * C[i])
            if a_lo - 1e-10 <= sp <= a_hi + 1e-10:
                alphas.append(max(a_lo, min(a_hi, sp)))
    if N <= 80:
        for i in range(M):
            for j in range(i + 1, M):
                u = C[i] - C[j]
                v = B[i] - B[j]
                w = A[i] - A[j]
                sc = max(abs(u), abs(v), abs(w), 1e-300)
                if abs(u) <= 1e-12 * sc:
                    if abs(v) > 1e-300:
                        root = -w / v
                        if a_lo - 1e-10 <= root <= a_hi + 1e-10:
                            alphas.append(max(a_lo, min(a_hi, root)))
                else:
                    D = v * v - 4.0 * u * w
                    if D < 0:
                        if D < -1e-12 * (v * v + abs(4.0 * u * w) + 1e-300):
                            continue
                        D = 0.0
                    sqD = math.sqrt(D)
                    q = -0.5 * (v + math.copysign(sqD, v))
                    if abs(q) > 1e-300:
                        roots = [q / u, w / q]
                    else:
                        roots = [-v / (2.0 * u), -v / (2.0 * u)]
                    for root in roots:
                        if np.isfinite(root) and a_lo - 1e-10 <= root <= a_hi + 1e-10:
                            alphas.append(max(a_lo, min(a_hi, root)))
    alphas.sort()
    deduped = []
    for a in alphas:
        if not deduped or abs(a - deduped[-1]) > 1e-12 * max(1.0, abs(a)):
            deduped.append(a)
    if not deduped:
        return h, cur_score
    arr = np.array(deduped, dtype=np.float64)
    candidates = []
    bs = 512
    for st in range(0, len(arr), bs):
        batch = arr[st:st+bs]
        env = A[None, :] + B[None, :] * batch[:, None] + C[None, :] * (batch[:, None]) ** 2
        me = np.max(env, axis=1)
        for bi in range(len(batch)):
            candidates.append((float(me[bi]), st + bi))
    candidates.sort(key=lambda x: x[0])
    top_set = set(idx for _, idx in candidates[:16])
    top_set.add(0)
    top_set.add(len(arr) - 1)
    for i, a in enumerate(arr):
        if abs(a) < 1e-15:
            top_set.add(i)
            break
    best_h, best_s = h, cur_score
    for idx in top_set:
        h_new = project_box_sum(h + arr[idx] * d, N)
        s = score_exact(h_new, N)
        if s < best_s - 1e-15:
            best_s, best_h = s, h_new
    return best_h, best_s

def make_directions(h, N, rng, stage):
    if stage == 'main':
        cs = (8, 6, 4, 2) if N <= 80 else ((6, 4, 2, 1) if N <= 250 else (3, 2, 1, 1))
    else:
        cs = (4, 3, 2, 1) if N <= 80 else ((3, 2, 1, 1) if N <= 250 else (1, 1, 1, 0))
    n_star, n_rand, n_block, n_harm = cs
    dirs = []
    anchor = int(np.argmax(np.minimum(h, 1.0 - h)))
    others = [i for i in range(N) if i != anchor]
    for c in range(min(n_star, len(others))):
        pos = min(c * max(1, len(others) // n_star), len(others) - 1)
        d = np.zeros(N)
        d[others[pos]] = 1.0
        d[anchor] = -1.0
        dirs.append(d)
    used = set()
    for c in range(n_rand):
        added = False
        for _ in range(50):
            a, b = sorted(rng.integers(0, N, size=2).tolist())
            if a != b and (a, b) not in used:
                used.add((a, b))
                d = np.zeros(N)
                d[b], d[a] = 1.0, -1.0
                dirs.append(d)
                added = True
                break
        if not added and N >= 2:
            d = np.zeros(N)
            d[0], d[1] = 1.0, -1.0
            dirs.append(d)
    for c in range(n_block):
        ss = int(rng.integers(3, min(13, N + 1)))
        if ss < 3 or N < 3:
            continue
        supp = rng.choice(N, size=ss, replace=False)
        vals = rng.standard_normal(ss)
        vals -= np.mean(vals)
        vals[0] -= np.sum(vals)
        d = np.zeros(N)
        d[supp] = vals
        if np.max(np.abs(d)) > 1e-15:
            d /= np.max(np.abs(d))
            dirs.append(d)
    for c in range(min(n_harm, (N - 1) // 2)):
        r = c + 1
        ph = rng.uniform(0, 2 * math.pi)
        ia = np.arange(N)
        d = np.cos(2.0 * math.pi * r * ia / N + ph)
        d -= np.mean(d)
        d[0] -= np.sum(d)
        if np.max(np.abs(d)) > 1e-15:
            d /= np.max(np.abs(d))
            dirs.append(d)
    return dirs

def run(seed=42, **kwargs):
    try:
        seed = int(seed)
    except (ValueError, TypeError):
        seed = 42
    rng = np.random.default_rng(seed)
    N = None
    baseline = None
    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            arr = np.asarray(initial_h_values, dtype=np.float64).ravel()
            if len(arr) > 0:
                N = len(arr)
                if np.all(np.isfinite(arr)):
                    baseline = project_box_sum(np.clip(arr, 0, 1), N)
                else:
                    baseline = np.full(N, 0.5)
    except Exception:
        pass
    if N is None:
        N = int(kwargs.get('n_points', 70))
        if not (1 <= N <= 999):
            N = 70
        baseline = np.full(N, 0.5)
    incumbent = project_box_sum(baseline, N)
    inc_score = score_exact(incumbent, N)
    if N > 999:
        return incumbent, float(inc_score), int(N)
    starts = [incumbent.copy(), np.full(N, 0.5), project_box_sum(1.0 - baseline, N)]
    starts.append(project_box_sum(rng.uniform(0, 1, N), N))
    starts.append(project_box_sum(incumbent + 0.20 * rng.standard_normal(N), N))
    for _ in range(2):
        R = min(5, (N - 1) // 2)
        ia = np.arange(N)
        z3 = np.full(N, 0.5)
        for r in range(1, R + 1):
            z3 += (0.12 / r) * (rng.standard_normal() * np.sin(2 * math.pi * r * ia / N) +
                              rng.standard_normal() * np.cos(2 * math.pi * r * ia / N))
        starts.append(project_box_sum(z3, N))
    dedup_starts = [starts[0]]
    for s in starts[1:]:
        if all(np.max(np.abs(s - ds)) >= 1e-8 for ds in dedup_starts):
            dedup_starts.append(s)
    mx = 7 if N <= 80 else (5 if N <= 250 else 3)
    dedup_starts = dedup_starts[:mx]
    for s in dedup_starts:
        ss = score_exact(s, N)
        if ss < inc_score:
            inc_score, incumbent = ss, s.copy()
    local_results = []
    for start in dedup_starts:
        h_cur, s_cur = start.copy(), score_exact(start, N)
        for d in make_directions(h_cur, N, rng, 'main'):
            try:
                h_new, s_new = line_search(h_cur, d, N, s_cur)
                if s_new < s_cur - 1e-15:
                    h_cur, s_cur = h_new, s_new
            except Exception:
                pass
        local_results.append((h_cur, s_cur))
    local_results.sort(key=lambda x: x[1])
    ne = 3 if N <= 80 else (2 if N <= 250 else 1)
    elites = []
    for hr, sr in local_results:
        if all(np.max(np.abs(hr - eh)) >= 1e-8 for eh, _ in elites):
            elites.append((hr, sr))
        if len(elites) >= ne:
            break
    for eh, es in elites:
        h_cur, s_cur = eh.copy(), es
        for d in make_directions(h_cur, N, rng, 'reduced'):
            try:
                h_new, s_new = line_search(h_cur, d, N, s_cur)
                if s_new < s_cur - 1e-15:
                    h_cur, s_cur = h_new, s_new
            except Exception:
                pass
        if s_cur < inc_score:
            inc_score, incumbent = s_cur, h_cur.copy()
    scales = [0.08, 0.20] if N <= 80 else ([0.12] if N <= 250 else [])
    for scale in scales:
        h_p = project_box_sum(incumbent + scale * rng.standard_normal(N), N)
        s_p = score_exact(h_p, N)
        for d in make_directions(h_p, N, rng, 'reduced'):
            try:
                h_new, s_new = line_search(h_p, d, N, s_p)
                if s_new < s_p - 1e-15:
                    h_p, s_p = h_new, s_new
            except Exception:
                pass
        if s_p < inc_score:
            inc_score, incumbent = s_p, h_p.copy()
    if not feasible_check(incumbent, N):
        incumbent = project_box_sum(incumbent, N)
    if not feasible_check(incumbent, N):
        incumbent = np.full(N, 0.5)
    final_score = score_exact(incumbent, N)
    if not np.isfinite(final_score):
        incumbent = np.full(N, 0.5)
        final_score = score_exact(incumbent, N)
    return incumbent, float(final_score), int(N)
