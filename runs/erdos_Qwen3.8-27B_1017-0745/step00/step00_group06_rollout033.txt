def project_capped_simplex(v, N):
    M = N / 2.0
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > M:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    y = np.clip(v - lam, 0.0, 1.0)
    delta = M - np.sum(y)
    for _ in range(3):
        if delta > 0:
            for i in range(N):
                add = min(delta, 1.0 - y[i])
                y[i] += add
                delta -= add
        elif delta < 0:
            for i in range(N):
                sub = min(-delta, y[i])
                y[i] -= sub
                delta += sub
        if abs(delta) <= 1e-12 * max(1, N):
            break
    if not np.all(np.isfinite(y)):
        return None
    if np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    if abs(np.sum(y) - M) > 1e-9 * max(1, N):
        return None
    return np.clip(y, 0.0, 1.0)

        return False

        return False

        return False
    if abs(np.sum(h) - N / 2.0) > 1e-9 * max(1, N):
        return False
    return True

def exact_c5(h, N):
    try:
        if not is_feasible(h, N):
            return float('inf')
        d = 2.0 / N
        vals = np.correlate(h, 1.0 - h, mode='full') * d
        m = float(np.max(vals))
        if not np.isfinite(m):
            return float('inf')
        return m
    except Exception:
        return float('inf')

def lag_gradient(x, q, d, N):
    g = np.zeros(N, dtype=np.float64)
    jmin = max(0, -q)
    jmax = min(N, N - q)
    for j in range(jmin, jmax):
        g[j + q] += d * (1.0 - x[j])
        g[j] -= d * x[j + q]
    return g

def fw_min_norm(G, max_iter=60):
    n_rows = G.shape[0]
    gbar = G[0].copy()
    for _ in range(max_iter):
        dots = np.array([float(np.dot(G[k], gbar)) for k in range(n_rows)])
        kmin = int(np.argmin(dots))
        gap = float(np.dot(gbar, gbar)) - dots[kmin]
        w = G[kmin] - gbar
        dw = float(np.dot(w, w))
        if gap <= 1e-12 * max(1.0, float(np.dot(gbar, gbar))) or dw <= 1e-24:
            break
        alpha = min(1.0, max(0.0, gap / dw))
        gbar = gbar + alpha * w
    return gbar

def smooth_z(z):
    y = 0.5 * z + 0.25 * np.roll(z, 1) + 0.25 * np.roll(z, -1)
