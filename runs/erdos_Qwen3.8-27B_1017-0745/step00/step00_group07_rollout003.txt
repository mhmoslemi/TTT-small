    res = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'max_iter': 500})
    if res.status != 0 or not np.all(np.isfinite(res.x)):
        return None, None, None
    y = res.x[:n].copy()
    y = project_box_sum(y, S, lower, upper)
    if y is None:
        return None, None, None
    Qmodel = float(np.max(B + G.dot(y - x)))
    pred = float(np.max(B) - Qmodel)
    return y, Qmodel, pred

def build_starts(n, seed, parent):
    rng = np.random.default_rng(seed)
    S = n / 2.0
    starts = []
    gl = np.zeros(n)
    gu = np.ones(n)
    if parent is not None:
        starts.append(parent.copy())
        rev = parent[::-1].copy()
        if not np.allclose(rev, parent, atol=1e-10):
            starts.append(rev)
    else:
        starts.append(np.full(n, 0.5))
    K = max(1, min(6, (n - 1) // 2))
    for _ in range(2):
        q = int(rng.integers(1, K + 1))
        phase = float(rng.uniform(0, 2 * np.pi))
        i = np.arange(n)
        z = 0.5 + 0.45 * np.cos(2 * np.pi * q * i / n + phase)
        p = project_box_sum(z, S, gl, gu)
        if p is not None:
            starts.append(p)
    z = rng.standard_normal(n)
    fz = np.fft.rfft(z)
    kf = np.arange(len(fz))
    fz *= np.exp(-0.5 * (kf / max(1.0, K / 2.0)) ** 2)
    z = np.fft.irfft(fz, n=n)
    sd = np.std(z)
    if sd > 1e-12:
        z = z / sd
    z = 0.5 + 0.35 * z
    p = project_box_sum(z, S, gl, gu)
    if p is not None:
        starts.append(p)
    idx = rng.permutation(n)
    z2 = np.zeros(n)
    z2[idx[:n // 2]] = 1.0
    if n % 2 == 1:
        z2[idx[n // 2]] = 0.5
    if check_feasible(z2, S):
        starts.append(z2)
    deduped = []
    for s in starts:
        if not check_feasible(s, S):
            continue
        dup = any(np.max(np.abs(s - d)) < 1e-10 for d in deduped)
        if not dup:
            deduped.append(s)
    cap = 6 if n <= 128 else (4 if n <= 256 else 2)
    return deduped[:cap]

def run(seed=42, **kwargs):
    parent = None
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        try:
            p = np.array(iv, dtype=np.float64).ravel()
            if p.ndim == 1 and len(p) >= 2 and np.all(np.isfinite(p)):
                parent = p
        except Exception:
            parent = None
    if parent is not None:
        n = len(parent)
    else:
        n = kwargs.get('n_points', 80)
        if not isinstance(n, (int, np.integer)) or n < 8 or n > 512:
            n = 80
    S = n / 2.0
    h_const = np.full(n, 0.5)
    best_h = h_const.copy()
    best_score = exact_score(h_const)
    if parent is not None:
        p_rep = project_box_sum(parent, S, np.zeros(n), np.ones(n))
        if p_rep is not None:
            sc = exact_score(p_rep)
            if sc is not None and sc < best_score:
                best_h = p_rep.copy()
                best_score = sc
        elif check_feasible(parent, S):
            sc = exact_score(parent)
            if sc is not None and sc < best_score:
                best_h = parent.copy()
                best_score = sc
    if n > 512:
        return best_h, best_score, n
    starts = build_starts(n, seed, parent if check_feasible(parent, S) else None)
    Delta_min = 1e-4
    Delta_max = 0.5
    max_iter = 36 if n <= 128 else (24 if n <= 256 else 12)
    for start in starts:
        x = start.copy()
        score_x = exact_score(x)
        if score_x is None:
            continue
        if score_x < best_score - 1e-14 * max(1.0, abs(best_score)):
            best_h = x.copy()
            best_score = score_x
        Delta = 0.20
        fail_count = 0
        small_steps = 0
        for _it in range(max_iter):
            if fail_count >= 3 or Delta < Delta_min:
                break
            try:
                y, Qmodel, pred = slp_step(x, Delta, S)
            except Exception:
                y = None
            if y is None:
                fail_count += 1
                Delta = max(0.25 * Delta, Delta_min)
                continue
            fail_count = 0
            score_y = exact_score(y)
            if score_y is None:
                Delta = max(0.25 * Delta, Delta_min)
                continue
            if score_y < best_score - 1e-14 * max(1.0, abs(best_score)):
                best_h = y.copy()
                best_score = score_y
            actual = score_x - score_y
            pred_tol = 1e-12 * max(1.0, abs(score_x))
            if pred <= pred_tol:
                break
            rho = actual / pred
            step_norm = float(np.max(np.abs(y - x)))
            if actual > pred_tol and rho >= 0.1:
                if rho >= 0.75 and step_norm >= 0.8 * Delta:
                    Delta = min(2.0 * Delta, Delta_max)
                elif rho < 0.25:
                    Delta = max(0.25 * Delta, Delta_min)
                x = y
                score_x = score_y
                if actual < 1e-10 * max(1.0, abs(score_x)):
                    small_steps += 1
                else:
                    small_steps = 0
            else:
                Delta = max(0.25 * Delta, Delta_min)
            if small_steps >= 8:
                break
    final_score = exact_score(best_h)
    if final_score is None:
        best_h = h_const
        final_score = exact_score(h_const)
    return best_h, final_score, n
