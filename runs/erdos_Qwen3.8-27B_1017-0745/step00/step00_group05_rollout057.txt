            idx = np.where(y < 0.999)[0]
            if len(idx) > 0:
                y[idx[0]] = min(1.0, y[idx[0]] + r)
        else:
            idx = np.where(y > 0.001)[0]
            if len(idx) > 0:
                y[idx[0]] = max(0.0, y[idx[0]] + r)
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(y.sum()-target) > 1e-8:
        return None
    return y

def escore(h, dx):
    if h.ndim != 1 or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0-h, mode='full') * dx))

def svg(h, tau, dx):
    n = len(h)
    corr = np.correlate(h, 1.0-h, mode='full') * dx
    m = corr.max()
    e = np.exp((corr - m)/tau)
    se = e.sum()
    F = m + tau * math.log(se)
    p = e / se
    g = np.zeros(n)
    for k in range(len(corr)):
        w = p[k] * dx
        if w < 1e-14:
            continue
        t = k - (n-1)
        if t == 0:
            g += w*(1.0 - 2.0*h)
        elif t > 0:
            g[:n-t] += w*(1.0 - h[t:])
            g[t:] -= w*h[:n-t]
        else:
            tt = -t
            g[tt:] += w*(1.0 - h[:n-tt])
            g[:n-tt] -= w*h[tt:]
    if not np.all(np.isfinite(g)):
        return F + 1e8, g
    return F, g

def mr(h):
    return h.sum() - len(h)/2.0

def mj(h):
    return np.ones(len(h))

def upd_inc(h, dx, inc, sc):
    if h is None or h.ndim != 1 or not np.all(np.isfinite(h)):
        return inc, sc
    s = escore(h, dx)
    if np.isfinite(s) and s < sc - 1e-12:
        return h.copy(), s
    return inc, sc

def run_cont(x, taus, dx, n, bounds, constr):
    target = n/2.0
    for tau in taus:
        try:
            res = minimize(svg, x, args=(tau, dx), method='SLSQP', jac=True,
                          bounds=bounds, constraints=constr,
                          options={'maxiter':200, 'ftol':1e-10, 'disp':False})
            rx = res.x
            if rx.ndim == 1 and len(rx) == n and np.all(np.isfinite(rx)):
                pj = csp(rx, target)
                if pj is not None:
                    ov, _ = svg(x, tau, dx)
                    nv, _ = svg(pj, tau, dx)
                    if np.isfinite(nv) and nv <= ov + 1e-10:
                        x = pj
        except Exception:
            pass
    return x

def run(seed=42, **kwargs):
    parent = None
    n = 70
    try:
        pv = np.asarray(initial_h_values, dtype=float)
        if pv.ndim == 1 and 2 <= len(pv) <= 999:
            n = len(pv)
            if np.all(np.isfinite(pv)) and pv.min() >= -0.1 and pv.max() <= 1.1:
                parent = pv.copy()
    except Exception:
        pass
    dx = 2.0/n
    target = n/2.0
    bounds = [(0.0, 1.0)]*n
    constr = [{'type':'eq', 'fun':mr, 'jac':mj}]
    taus = [0.02, 0.005, 0.001, 0.0002, 0.00005, 0.00001]
    rng = np.random.default_rng(seed)

    best = np.full(n, 0.5)
    best_s = escore(best, dx)

    starts = []
    if parent is not None:
        pp = csp(parent, target)
        if pp is not None:
            best, best_s = upd_inc(pp, dx, best, best_s)
            starts.append(pp)
            rev = csp(pp[::-1].copy(), target)
            if rev is not None:
                starts.append(rev)
            for sig in [0.04, 0.10, 0.22, 0.40]:
                s = csp(pp + sig*rng.standard_normal(n), target)
                if s is not None:
                    starts.append(s)
    s1 = csp(rng.uniform(0.0, 1.0, n), target)
    if s1 is not None:
        starts.append(s1)
    s2 = csp(rng.beta(0.5, 0.5, n), target)
    if s2 is not None:
        starts.append(s2)
    starts.append(np.full(n, 0.5))

    for s in starts:
        proj = csp(s, target)
        if proj is None:
            continue
        best, best_s = upd_inc(proj, dx, best, best_s)
        res = run_cont(proj, taus, dx, n, bounds, constr)
        if res is not None and np.all(np.isfinite(res)):
            pj = csp(res, target)
            if pj is not None:
                best, best_s = upd_inc(pj, dx, best, best_s)

    for sig in [0.02, 0.08, 0.20]:
        pert = csp(best + sig*rng.standard_normal(n), target)
        if pert is not None:
            best, best_s = upd_inc(pert, dx, best, best_s)
            res = run_cont(pert, taus, dx, n, bounds, constr)
            if res is not None and np.all(np.isfinite(res)):
                pj = csp(res, target)
                if pj is not None:
                    best, best_s = upd_inc(pj, dx, best, best_s)

    final = csp(best, target)
    if final is None or not np.all(np.isfinite(final)):
        final = np.full(n, 0.5)
    final = np.clip(final, 0.0, 1.0)
    final_score = escore(final, dx)
    return final, final_score, n
