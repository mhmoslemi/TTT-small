def feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and h.min() >= -1e-12 and h.max() <= 1.0 + 1e-12
            and abs(np.sum(h) - n * 0.5) <= 1e-10)

def lags(h):
    n = len(h)
    nl = 2 * n - 1
    B = np.zeros(nl)
    G = np.zeros((nl, n))
    for d in range(-(n - 1), n):
        i = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        s = h[lo:hi]
        t = h[lo + d:hi + d]
        B[i] = np.dot(s, 1.0 - t)
        G[i, lo:hi] += 1.0 - t
        G[i, lo + d:hi + d] -= s
    return B, G

    n = len(h)
    if not feasible(h, n):
        return None
    v = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(v)):

def project_capped(z, S):
    n = len(z)
    if not np.all(np.isfinite(z)) or np.max(np.abs(z)) > 1e12:
        return None
    lo = np.min(z) - 1.0
    hi = np.max(z)

            lo = mid
        else:

            y[i] += a
            r -= a

        B[i] = np.dot(s, 1.0 - t)
        G[i, lo:hi] += 1.0 - t
