def capped_simplex_project(v, target):
    n = len(v)

        else:

                dd = min(r, 1.0 - y[i]); y[i] += dd; r -= dd
            else:
                dd = min(-r, y[i]); y[i] -= dd; r += dd

def norm_dir(d):

    m = float(np.max(np.abs(d)))

def qroots(A, B, C, lo, hi):
    res = []
    tol = 1e-13
    sc = max(1.0, abs(A), abs(B), abs(C))
    if abs(C) > tol * sc:

    out = []
    for r in res:

            out.append(min(hi, max(lo, r)))
