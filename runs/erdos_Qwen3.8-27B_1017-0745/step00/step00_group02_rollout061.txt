    L = 2 * n - 1
    J = np.zeros((L, L + 1))
    for idx in range(L):
        ell = idx - (n - 1)
        J[idx, -1] = 1.0
        i_lo = max(0, -ell)
        i_hi = min(n, n - ell)
        if i_lo < i_hi:
            J[idx, i_lo + ell:i_hi + ell] -= dx * (1.0 - h[i_lo:i_hi])
            J[idx, i_lo:i_hi] += dx * h[i_lo + ell:i_hi + ell]
    return J

def _sq_eq(z, args):
    return np.sum(z[:-1]) - args[2]

def _sq_eq_j(z, args):
    r = np.zeros(len(z))
    r[:-1] = 1.0
    return r

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    n = 45
    parent = globals().get('initial_h_values', None)
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.ndim == 1 and 2 <= len(p) <= 999 and np.all(np.isfinite(p)) and np.max(np.abs(p)) < 1e6:
            n = len(p)
    dx = 2.0 / n
    S = n / 2.0
    fb = _proj(np.full(n, 0.5), S, n)
    best_h = fb.copy()
    best_sc = _score(best_h, dx)
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.ndim == 1 and len(p) == n and np.all(np.isfinite(p)):
            fb = _proj(p, S, n)
            if fb is not None and abs(fb.sum() - S) < 1e-10:
                best_h = fb.copy()
                best_sc = _score(best_h, dx)
    local_results = [best_h.copy()]
    starts = [best_h.copy()]
    h_rev = best_h[::-1].copy()
    h_comp = 1.0 - best_h
    starts += [_proj(h_comp, S, n), _proj(h_rev, S, n), _proj(1.0 - h_rev, S, n)]
    starts += [_proj(best_h + 0.08 * rng.standard_normal(n), S, n)]
    starts += [_proj(best_h + 0.25 * rng.standard_normal(n), S, n)]
    starts += [_proj(rng.uniform(0, 1, n), S, n)]
    starts += [_proj(rng.beta(0.5, 0.5, n), S, n)]
    taus = [0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4, 5e-5, 1e-5]
    n_iter = 80
    for st in starts:
        h = _proj(st, S, n)
        if h is None:
            continue
        for tau in taus:
            alpha = 1.0
            for it in range(n_iter):
                q = _lags(h, dx)
                F = _smooth(q, tau)
                g = _sgrad(h, dx, tau)
                if not np.all(np.isfinite(g)):
                    break
                y = _proj(h - alpha * g, S, n)
                if y is None:
                    break
                d = y - h
                gd = np.dot(g, d)
                if np.max(np.abs(d)) < 1e-10 or gd > -1e-12:
                    break
                Fy = _smooth(_lags(y, dx), tau)
                if Fy <= F + 1e-4 * gd + 1e-14:
                    h = y
                    sc = _score(h, dx)
                    if sc < best_sc - 1e-12:
                        best_sc = sc
                        best_h = h.copy()
                    alpha = min(1.25 * alpha, 5.0)
                else:
                    alpha *= 0.5
                    if alpha < 1e-10:
                        break
            if h is not None and np.all(np.isfinite(h)):
                local_results.append(h.copy())
    for sig in [0.04, 0.12, 0.30, 0.50]:
        h = _proj(best_h + sig * rng.standard_normal(n), S, n)
        if h is None:
            continue
        for tau in taus:
            alpha = 1.0
            for it in range(n_iter):
                q = _lags(h, dx)
                F = _smooth(q, tau)
                g = _sgrad(h, dx, tau)
                if not np.all(np.isfinite(g)):
                    break
                y = _proj(h - alpha * g, S, n)
                if y is None:
                    break
                d = y - h
                gd = np.dot(g, d)
                if np.max(np.abs(d)) < 1e-10 or gd > -1e-12:
                    break
                Fy = _smooth(_lags(y, dx), tau)
                if Fy <= F + 1e-4 * gd + 1e-14:
                    h = y
                    sc = _score(h, dx)
                    if sc < best_sc - 1e-12:
                        best_sc = sc
                        best_h = h.copy()
                    alpha = min(1.25 * alpha, 5.0)
                else:
                    alpha *= 0.5
                    if alpha < 1e-10:
                        break
        if h is not None and np.all(np.isfinite(h)):
            local_results.append(h.copy())
    if n <= 100:
        local_results.sort(key=lambda x: _score(x, dx))
        sel = []
        for hr in local_results:
            if len(sel) >= 3:
                break
            if all(np.max(np.abs(hr - s)) > 1e-5 for s in sel):
                sel.append(hr)
        if not sel:
            sel = [best_h.copy()]
        slsqp_args = (n, dx, S)
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        cons = [
            {'type': 'ineq', 'fun': _sq_in, 'jac': _sq_in_j, 'args': slsqp_args},
            {'type': 'eq', 'fun': _sq_eq, 'jac': _sq_eq_j, 'args': slsqp_args},
        ]
        for hr in sel:
            t0 = min(1.0, _score(hr, dx) + 1e-8)
            z0 = np.concatenate([hr, [t0]])
            try:
                from scipy.optimize import minimize
                res = minimize(_sq_obj, z0, method='SLSQP', jac=_sq_obj_j,
                               args=slsqp_args, bounds=bounds, constraints=cons,
                               options={'maxiter': 250, 'ftol': 1e-10, 'disp': False})
                zh = res.x[:-1]
                if np.all(np.isfinite(zh)) and len(zh) == n:
                    zh = _proj(zh, S, n)
                    if zh is not None and np.all(np.isfinite(zh)):
                        sc = _score(zh, dx)
                        if sc < best_sc - 1e-12:
                            best_sc = sc
                            best_h = zh.copy()
            except Exception:
                pass
    best_h = _proj(best_h, S, n)
    if best_h is None:
        best_h = np.full(n, 0.5)
    final_sc = _score(best_h, dx)
    if not np.all(np.isfinite(best_h)) or abs(best_sc - final_sc) > 1e-6:
        best_h = _proj(fb, S, n)
        if best_h is None:
            best_h = np.full(n, 0.5)
        final_sc = _score(best_h, dx)
    np.clip(best_h, 0.0, 1.0, out=best_h)
    r = S - best_h.sum()
    if abs(r) > 1e-12:
        m = best_h < 1.0 if r > 0 else best_h > 0.0
        if m.sum() > 0:
            best_h[m] += r / m.sum()
            np.clip(best_h, 0.0, 1.0, out=best_h)
    final_sc = _score(best_h, dx)
    return (best_h, float(final_sc), int(n))
