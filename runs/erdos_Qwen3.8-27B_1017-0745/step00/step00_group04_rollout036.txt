def is_feasible(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return False
    S = n / 2.0
    if abs(np.sum(h) - S) > 1e-9:
        return False
    if abs(math.fsum(float(x) for x in h) - S) > 1e-9:
        return False
    return True

def project_capped_simplex(v, n):
    S = n / 2.0
    v = np.clip(np.asarray(v, dtype=np.float64), -1e6, 1e6)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(v - mid, 0.0, 1.0)
        if np.sum(h) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(200):
        res = S - math.fsum(float(x) for x in h)
        if abs(res) <= 1e-12:
            break
        if res > 0:
            for i in range(n):
                add = min(res, 1.0 - h[i])
                if add > 0:
                    h[i] += add
                    res -= add
                    if abs(res) <= 1e-12:
                        break
        else:
            for i in range(n):
                sub = min(-res, h[i])
                if sub > 0:
                    h[i] -= sub
                    res += sub
                    if abs(res) <= 1e-12:
                        break
    h = np.clip(h, 0.0, 1.0)
    if is_feasible(h, n):
        return h
    return None

def exact_score(h, n):
    if not is_feasible(h, n):
        return None
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    score = float(np.max(corr) * dx)
    if not math.isfinite(score):
        return None
    return score

        return
    for existing in pop:
        if np.max(np.abs(v - existing)) < 1e-10:
            return
    pop.append(v.copy())
