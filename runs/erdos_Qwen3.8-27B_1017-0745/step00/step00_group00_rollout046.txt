    F = tau * (zmax + np.log(se))
    c1 = do_convolve(p, 1.0 - h)
    c2 = do_convolve(p[::-1], h)
    grad = (c1[n - 1:2 * n - 1] - c2[n - 1:2 * n - 1]) / n
    if not np.isfinite(F) or not np.all(np.isfinite(grad)):
        return None, None, score
    return F, grad, score

def optimize_one(h, n, S, temps, max_iter, best_h, best_sc, budget):
    for tau in temps:
        if budget[0] <= 0:
            return
        res = smooth_eval(h, tau)
        if res[0] is None:
            continue
        F, grad, _ = res
        budget[0] -= 1
        stale = 0
        for it in range(max_iter):
            if budget[0] <= 0:
                return
            gmax = float(np.max(np.abs(grad)))
            if gmax < 1e-13:
                break
            alpha = min(1e8, 0.08 / gmax)
            did_accept = False
            for bt in range(12):
                if budget[0] <= 0:
                    return
                y = project_box_sum(h - alpha * grad, S)
                if y is None:
                    alpha *= 0.5
                    continue
                budget[0] -= 1
                d = y - h
                direc = float(np.dot(grad, d))
                if not np.isfinite(direc) or direc >= 0.0:
                    alpha *= 0.5
                    continue
                r2 = smooth_eval(y, tau)
                if r2[0] is None:
                    alpha *= 0.5
                    continue
                Fy, gy, sc_y = r2
                if is_feasible(y, n) and sc_y < best_sc[0]:
                    best_h[:] = y
                    best_sc[0] = sc_y
                if Fy <= F + 1e-4 * direc + 1e-13 * max(1.0, abs(F)):
                    F_prev = F
                    h = y
                    F = Fy
                    grad = gy
                    did_accept = True
                    if float(np.max(np.abs(d))) < 1e-8 or (F_prev - Fy) <= 1e-12 * max(1.0, abs(F)):
                        stale += 1
                    else:
                        stale = 0
                    if stale >= 4:
                        break
                else:
                    alpha *= 0.5
            if not did_accept or stale >= 4:
                break


def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent = None
    n = 45
    S_global = globals().get('initial_h_values', None)
    if S_global is not None:
        try:
            arr = np.array(S_global, dtype=np.float64)
            if arr.ndim == 1 and 2 <= len(arr) <= 999 and np.all(np.isfinite(arr)) and np.max(np.abs(arr)) <= 1e12:
                n = len(arr)
                S = n / 2.0
                proj = project_box_sum(arr, S)
                if proj is not None:
                    parent = proj
        except Exception:
            pass
    if parent is None:
        n = 45
        parent = np.full(n, 0.5)
    S = n / 2.0
    best_h = parent.copy()
    sc = exact_score(best_h)
    if sc is None:
        best_h = np.full(n, 0.5)
        sc = exact_score(best_h)
    best_sc = [sc]
    B = min(50000, max(100, int(1e8 / (n * n))))
    budget = [B]
    temperatures = [0.04, 0.015, 0.006, 0.002, 0.0007, 0.00025, 0.00008]
    if n <= 100:
        max_iter = 60
    elif n <= 300:
        max_iter = 25
    else:
        max_iter = 8
    starts = make_starts(n, parent, rng)
    for st in starts:
        if not is_feasible(st, n):
            continue
        sc_st = exact_score(st)
        if sc_st is not None and sc_st < best_sc[0]:
            best_h[:] = st
            best_sc[0] = sc_st
        if budget[0] <= 0:
            break
        try:
            optimize_one(st.copy(), n, S, temperatures, max_iter, best_h, best_sc, budget)
        except Exception:
            continue
    final_y = project_box_sum(best_h, S)
    if final_y is not None:
        best_h = final_y
    sc_final = exact_score(best_h)
    if sc_final is None:
        best_h = parent.copy()
        sc_final = exact_score(best_h)
        if sc_final is None:
            best_h = np.full(n, 0.5)
            sc_final = exact_score(best_h)
    return best_h, float(sc_final), int(n)
