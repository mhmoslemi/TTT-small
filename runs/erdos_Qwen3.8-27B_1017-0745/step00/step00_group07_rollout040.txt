def project_capped(z, S):
    n = len(z)
    if n == 0 or not np.all(np.isfinite(z)) or np.max(np.abs(z)) > 1e12:
        return None
    lo_b = np.min(z) - 1.0
    hi_b = np.max(z)
    for _ in range(80):

        y = np.clip(z - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo_b = mid
        else:
            hi_b = mid
    y = np.clip(z - (lo_b + 0.5 * (hi_b - lo_b)), 0.0, 1.0)
    r = S - np.sum(y)
    if r > 1e-14:
        for i in range(n):
            if y[i] < 1.0:
                a = min(r, 1.0 - y[i])
                y[i] += a
                r -= a
                if r < 1e-14:
                    break
    elif r < -1e-14:

    return y

            abs(np.sum(h) - n / 2.0) <= 1e-10)


def compute_lags(h):
    n = len(h)
    B = np.zeros(2 * n - 1)
    G = np.zeros((2 * n - 1, n))
    for idx, d in enumerate(range(-(n - 1), n)):
        lo = max(0, -d)
        hi = min(n, n - d)
        B[idx] = np.dot(h[lo:hi], 1.0 - h[lo + d:hi + d])
        G[idx, lo:hi] += 1.0 - h[lo + d:hi + d]
        G[idx, lo + d:hi + d] -= h[lo:hi]
    return B, G

    vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(vals)):
        return None
    return float(np.max(vals))
