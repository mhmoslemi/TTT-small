import numpy as np

def capped_simplex_project(v, M):
    N = len(v)
    v = np.asarray(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        s = float(np.sum(y))
        if s > M:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    delta = M - float(np.sum(y))
    for _ in range(2):
        if abs(delta) < 1e-15:
            break
        for idx in range(N):
            if delta > 0:
                amt = min(delta, 1.0 - y[idx])
                y[idx] += amt
                delta -= amt
            else:
                amt = min(-delta, y[idx])
                y[idx] -= amt
                delta += amt
    if abs(delta) > 1e-12:
        return None
    if not np.all(np.isfinite(y)) or np.min(y) < -1e-15 or np.max(y) > 1.0+1e-15:
        return None
    y = np.clip(y, 0.0, 1.0)
    if abs(float(np.sum(y)) - M) > 1e-9 * max(1, N):
        return None
    return y

def exact_score(h, N):

def compute_rbc(i, j, h, N, dx):
    # Returns r array (2N-1), b array (2N-1), and c as dict of 3 entries
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    r = corr  # indexed: lag q -> index q + (N-1)
    nl = 2 * N - 1
    b = np.zeros(nl)
    for q in range(-(N - 1), N):
        # g_{q,i} - g_{q,j}
        val = 0.0
        if 0 <= i - q < N:
            val += (1.0 - h[i - q])
        if 0 <= j - q < N:
            val -= (1.0 - h[j - q])
        if 0 <= i + q < N:
            val -= h[i + q]
        if 0 <= j + q < N:
            val += h[j + q]
        b[q + (N - 1)] = dx * val
    # c: only 3 nonzero entries
    # c[0] at lag 0 -> index N-1
    # c[j-i] at index (j-i)+(N-1)
    # c[i-j] at index (i-j)+(N-1)
    c = {}
    c[0] = -2.0 * dx
    if j - i != 0:
        c[j - i] = dx
        c[i - j] = dx
    return r, b, c

def eval_line(r, b, c, nl, N, t):
    # f(t) = max_q (r_q + b_q*t + c_q*t^2)
    val = r + b * t
    for q, cv in c.items():
        idx = q + (N - 1)
        if 0 <= idx < nl:
            val[idx] += cv * t * t
    return float(np.max(val))

def golden_min(f, a, b_val, n_iters):
    phi = (math.sqrt(5.0) - 1.0) / 2.0
    u = b_val - phi * (b_val - a)
    v = a + phi * (b_val - a)
    fu = f(u)
    fv = f(v)
    for _ in range(n_iters):
        if fu <= fv:
            b_val = v
            v = u
            fv = fu
            u = b_val - phi * (b_val - a)
            fu = f(u)
        else:
            a = u
            u = v
            fu = fv
            v = a + phi * (b_val - a)
            fv = f(v)
    return 0.5 * (a + b_val)
