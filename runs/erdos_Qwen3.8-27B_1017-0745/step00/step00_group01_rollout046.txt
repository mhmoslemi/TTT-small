def eval_envelope(a, b, c, t_arr):

    scores = np.empty(len(t_arr))

def line_search(x, v, n):

    if interval is None:
        return None
    lo, hi = interval
    coeffs = get_coeffs(x, v, n)
    if coeffs is None:
        return None
    a, b, c = coeffs
    cands = get_candidates(a, b, c, lo, hi)
    t_arr = np.array(cands, dtype=np.float64)
    best_ts = eval_envelope(a, b, c, t_arr)
    if 0.0 not in best_ts:
        best_ts = np.append(best_ts, 0.0)
    best_score = float('inf')
    best_h = None
    for t in best_ts:
        if abs(t) < 1e-15:

        y = x + t * v
        if not feasibility_check(y, n):
            y = project_capped(y, n)
            if y is None:

        sc = exact_score(y, n)
        if sc < best_score:
            best_score = sc
            best_h = y
    return best_h  # None means no improvement found
