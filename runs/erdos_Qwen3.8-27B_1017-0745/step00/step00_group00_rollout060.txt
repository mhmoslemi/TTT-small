        h_new = run_slsqp(h0, n, bounds, cons)
        if h_new is not None:
            h_new = project_box_sum(h_new, S)
            if feasibility_check(h_new, n):
                sc = exact_score(h_new)
                if sc < best_score:
                    best = h_new.copy()
                    best_score = sc

    sigma = 0.05
    for _ in range(4):
        h0 = project_box_sum(best + sigma * rng.standard_normal(n), S)
        h_new = run_slsqp(h0, n, bounds, cons)
        improved = False
        if h_new is not None:
            h_new = project_box_sum(h_new, S)
            if feasibility_check(h_new, n):
                sc = exact_score(h_new)
                if sc < best_score - 1e-12:
                    best = h_new.copy()
                    best_score = sc
                    improved = True
        sigma = max(0.01, 0.7 * sigma) if improved else min(0.25, 1.5 * sigma)

    best = project_box_sum(best, S)
    if not feasibility_check(best, n):
        best = project_box_sum(parent, S)
    final_score = exact_score(best)
    return (best, final_score, n)
