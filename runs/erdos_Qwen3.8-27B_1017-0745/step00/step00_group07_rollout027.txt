            if score_x < best_score:
                best_score = score_x
                best_h = x.copy()
            Delta = 0.20
            fail_count = 0
            stall = 0
            for _ in range(max_iter):
                if Delta < 1e-4 or fail_count >= 3:
                    break
                result = lp_step(x, Delta, S, n)
                if result is None:
                    fail_count += 1
                    Delta = max(0.25 * Delta, 1e-4)
                    continue
                y, pred_raw = result
                pred = pred_raw * (2.0 / n)
                score_y = exact_score(y)
                if score_y is None:
                    fail_count += 1
                    Delta = max(0.25 * Delta, 1e-4)
                    continue
                if score_y < best_score - 1e-14 * max(1.0, abs(best_score)):
                    best_score = score_y
                    best_h = y.copy()
                actual = score_x - score_y
                if pred <= 1e-12 * max(1.0, abs(score_x)):
                    break
                rho = actual / pred
                if actual > 1e-12 * max(1.0, abs(score_x)) and rho >= 0.1:
                    step_sz = np.max(np.abs(y - x))
                    x = y
                    score_x = score_y
                    fail_count = 0
                    if rho >= 0.75 and step_sz >= 0.8 * Delta:
                        Delta = min(2.0 * Delta, 0.5)
                    elif rho < 0.25:
                        Delta = max(0.25 * Delta, 1e-4)
                    stall = stall + 1 if actual < 1e-10 * max(1.0, abs(score_x)) else 0
                    if stall >= 8:
                        break
                else:
                    Delta = max(0.25 * Delta, 1e-4)
                    fail_count += 1
    final_h = fallback.copy()
    final_s = exact_score(fallback)
    if final_s is None:
        final_s = 1.0
    cs = exact_score(best_h)
    if cs is not None and cs < final_s:
        final_s = cs
        final_h = best_h.copy()
    if not check_feasible(final_h):
        rep = project_box_sum(final_h, S, np.zeros(n), np.ones(n))
        if rep is not None and check_feasible(rep):
            final_h = rep
            final_s = exact_score(rep)
    if not check_feasible(final_h) or final_s is None:
        final_h = np.full(n, 0.5)
        final_s = exact_score(final_h)
    return final_h, float(final_s), n
