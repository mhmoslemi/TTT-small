import math

def exact_score(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(corr * dx))
    return val

def is_feasible(h, n):
    if h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    S = n / 2.0
    s1 = abs(float(np.sum(h)) - S)
    s2 = abs(math.fsum(float(x) for x in h) - S)
    return s1 <= 1e-10 and s2 <= 1e-10

def project_capped_simplex(v, n):
    if v.ndim != 1 or len(v) != n:
        return None
    if not np.all(np.isfinite(v)):
        return None
    v = np.clip(v, -1e6, 1e6).astype(np.float64)
    S = n / 2.0
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(4):
        resid = S - math.fsum(float(x) for x in h)
        if abs(resid) < 1e-14:
            break
        if resid > 0:
            mask = h < 1.0 - 1e-14
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += resid / cnt
        else:
            mask = h > 1e-14
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += resid / cnt
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2):
        resid = S - math.fsum(float(x) for x in h)
        if abs(resid) < 1e-14:
            break
        if resid > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + resid)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + resid)
    if not np.all(np.isfinite(h)):
        return None
    return h

def compute_lags_and_grad(h, n):
    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        idx = np.arange(n - d)
        q[d] = dx * np.sum(h[idx + d] * (1.0 - h[idx]))
        G[d, idx + d] += dx * (1.0 - h[idx])
        G[d, idx] += dx * (-h[idx + d])
        q[n - 1 + d] = dx * np.sum(h[idx] * (1.0 - h[idx + d]))
        G[n - 1 + d, idx] += dx * (1.0 - h[idx + d])
        G[n - 1 + d, idx + d] += dx * (-h[idx])
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None, None
    return q, G

def active_bundle_gradient(q, G):
    Cq = float(np.max(q))
    tol = max(1e-12, 1e-7 * max(1.0, abs(Cq)))
    active = np.where(q >= Cq - tol)[0]
    if len(active) > 12:
        active = active[np.argsort(q[active])[::-1][:12]]
    m = len(active)
    if m == 0:
        return None
    U = G[active, :].copy()
    U -= np.mean(U, axis=1, keepdims=True)
    if not np.all(np.isfinite(U)):
        return None
    K = U @ U.T
    K = (K + K.T) / 2.0
    alpha = np.full(m, 1.0 / m)
    for _ in range(40):
        z = K @ alpha
        j = int(np.argmin(z))
        gap = float(alpha @ z - z[j])
        if gap <= 1e-12:
            break
        v = np.zeros(m)
        v[j] = 1.0
        v -= alpha
        kvv = float(v @ K @ v)
        if kvv <= 1e-24:
            break
        gamma = float(np.clip(-(v @ z) / kvv, 0.0, 1.0))
        alpha += gamma * v
        alpha = np.clip(alpha, 0.0, None)
        s = float(np.sum(alpha))
        if s < 1e-30:
            break
        alpha /= s
    g = alpha @ U
    if not np.all(np.isfinite(g)):
        li = int(np.argmax(q))
        g = G[li, :].copy()
        g -= np.mean(g)
        if not np.all(np.isfinite(g)):
            return None
    return g

def local_search(h0, rng, n, max_iter):
    h = project_capped_simplex(h0, n)
    if h is None:
        return None, None
    cur_score = exact_score(h, n)
    if not np.isfinite(cur_score):
        return None, None
    best_h = h.copy()
    best_score = cur_score
    eta = 0.5
    for it in range(max_iter):
        q, G = compute_lags_and_grad(h, n)
        if q is None:
            break
        g = active_bundle_gradient(q, G)
        improved = False
        if g is not None:
            for b in range(8):
                eta_try = eta * (0.5 ** b)
                proposal = project_capped_simplex(h - eta_try * g, n)
                if proposal is None:
                    continue
                if np.max(np.abs(proposal - h)) < 1e-14:
                    continue
                ps = exact_score(proposal, n)
                if not np.isfinite(ps):
                    continue
                if ps < cur_score - 1e-12:
                    h = proposal
                    cur_score = ps
                    if cur_score < best_score - 1e-12:
                        best_score = cur_score
                        best_h = h.copy()
                    eta = min(2.0, 1.25 * eta_try)
                    improved = True
                    break
        if not improved:
            eta = max(1e-6, eta / 2.0)
            if it < 40:
                sigma, T = 0.12, 0.002
            elif it < 80:
                sigma, T = 0.07, 0.0008
            else:
                sigma, T = 0.035, 0.00025
            best_shake_score = cur_score
            best_shake_h = None
            for _ in range(4):
                noise = rng.normal(0.0, sigma, n)
                prop = project_capped_simplex(h + noise, n)
                if prop is None:
                    continue
                spos = exact_score(prop, n)
                if not np.isfinite(spos):
                    continue
                if spos < best_shake_score:
                    best_shake_score = spos
                    best_shake_h = prop
            if best_shake_h is not None:
                delta = best_shake_score - cur_score
                accept = delta <= 1e-12
                if not accept and T > 0:
                    try:
                        accept = rng.random() < math.exp(-delta / T)
                    except OverflowError:
                        accept = False
                if accept:
                    h = best_shake_h
                    cur_score = best_shake_score
                    eta = 0.25
                    if cur_score < best_score - 1e-12:
                        best_score = cur_score
                        best_h = h.copy()
    return best_h, best_score

def add_unique(starts, v, tol=1e-10):
    for s in starts:
        if np.max(np.abs(s - v)) < tol:
            return
    starts.append(v)

def run(seed=42, **kwargs):
    parent = globals().get('initial_h_values', None)
    n = None
    parent_arr = None
    if parent is not None:
        try:
            pa = np.asarray(parent, dtype=np.float64)
            if pa.ndim == 1 and len(pa) >= 2:
                n = len(pa)
                if np.all(np.isfinite(pa)):
                    parent_arr = pa.copy()
        except Exception:
            pass
    if n is None:
        try:
            np_kw = int(kwargs.get('n_points', 42))
            if 2 <= np_kw <= 512:
                n = np_kw
        except Exception:
            n = 42
    if n < 2:
        n = 42
    try:
        seed_int = int(seed)
    except Exception:
        seed_int = 42
    rng = np.random.default_rng(seed_int)
    S = n / 2.0
    const_h = np.full(n, 0.5)
    const_score = exact_score(const_h, n)
    parent_fb = None
    parent_fb_score = None
    if parent_arr is not None:
        pf = None
        if np.all(parent_arr >= 0) and np.all(parent_arr <= 1):
            if abs(float(np.sum(parent_arr)) - S) <= 1e-12:
                pf = parent_arr.copy()
            else:
                pf = project_capped_simplex(parent_arr, n)
        else:
            pf = project_capped_simplex(parent_arr, n)
        if pf is not None:
            parent_fb = pf
            parent_fb_score = exact_score(pf, n)
    if parent_fb is not None and parent_fb_score < const_score - 1e-12:
        best_h = parent_fb.copy()
        best_score = parent_fb_score
    else:
        best_h = const_h.copy()
        best_score = const_score
    if n <= 128:
        starts = []
        if parent_fb is not None:
            add_unique(starts, parent_fb.copy())
        add_unique(starts, const_h.copy())
        block = np.zeros(n)
        block[:n // 2] = 1.0
        if n % 2 == 1:
            block[n // 2] = 0.5
        pb = project_capped_simplex(block, n)
        if pb is not None:
            add_unique(starts, pb)
        alt = np.zeros(n)
        alt[0::2] = 1.0
        pa2 = project_capped_simplex(alt, n)
        if pa2 is not None:
            add_unique(starts, pa2)
        i_arr = np.arange(n, dtype=np.float64)
        sin_v = 0.5 + 0.45 * np.sin(2.0 * np.pi * i_arr / n)
        ps2 = project_capped_simplex(sin_v, n)
        if ps2 is not None:
            add_unique(starts, ps2)
        if parent_fb is not None:
            for sig in (0.05, 0.15, 0.35, 0.15):
                pert = parent_fb + sig * rng.normal(0.0, 1.0, n)
                pp = project_capped_simplex(pert, n)
                if pp is not None:
                    add_unique(starts, pp)
        n_rand = 2 if parent_fb is not None else 4
        for _ in range(n_rand):
            rv = project_capped_simplex(rng.uniform(0.0, 1.0, n), n)
            if rv is not None:
                add_unique(starts, rv)
        starts = starts[:12]
        max_iter = 120 if n <= 64 else 80
        for st in starts:
            try:
                res = local_search(st, rng, n, max_iter)
            except Exception:
                continue
            if res is None:
                continue
            cand, cs = res
            if cand is None or cs is None:
                continue
            if not np.isfinite(cs):
                continue
            if not is_feasible(cand, n):
                continue
            if cs < best_score - 1e-12:
                best_score = cs
                best_h = cand.copy()
    if not is_feasible(best_h, n):
        rep = project_capped_simplex(best_h, n)
        if rep is not None and is_feasible(rep, n):
            best_h = rep
    if not is_feasible(best_h, n):
        if parent_fb is not None and is_feasible(parent_fb, n):
            best_h = parent_fb.copy()
        else:
            best_h = const_h.copy()
    final_score = exact_score(best_h, n)
    if not np.isfinite(final_score):
        best_h = const_h.copy()
        final_score = exact_score(best_h, n)
    return (best_h.astype(np.float64), float(final_score), int(n))
