def compute_lag_grad(h, n):
    q = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(q)):
        return None, None
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        t = k - (n - 1)
        # grad[j] = (1-h[j-t])*1[0<=j-t<n] - h[j+t]*1[0<=j+t<n]
        # First term: j-t in [0,n) => j in [t, t+n), but j in [0,n)
        # Second term: j+t in [0,n) => j in [-t, n-t)
        j_idx = np.arange(n)
        jt1 = j_idx - t  # j - t
        jt2 = j_idx + t  # j + t
        m1 = (jt1 >= 0) & (jt1 < n)
        m2 = (jt2 >= 0) & (jt2 < n)
        G[k, m1] += (1.0 - h[jt1[m1]])
        G[k, m2] -= h[jt2[m2]]
    if not np.all(np.isfinite(G)):
        return None, None
    return q, G

            if abs(r) < 1e-14:
                break
    if not np.all(np.isfinite(y)) or np.min(y) < -1e-12 or np.max(y) > 1.0 + 1e-12:
        return None
    if abs(np.sum(y) - target) > 1e-10:
        return None
    return y

def is_feasible(h, n):
    if len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.min(h) < -1e-12 or np.max(h) > 1.0 + 1e-12:
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10:
        return False
    return True

def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return None
    c = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(c)):
        return None
    return float(np.max(c) * dx)

def compute_lag_grad(h, n):
    q = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(q)):
        return None, None
    G = np.zeros((2 * n - 1, n))
    j_idx = np.arange(n)
    for k in range(2 * n - 1):
        t = k - (n - 1)
        jt1 = j_idx - t
        jt2 = j_idx + t
        m1 = (jt1 >= 0) & (jt1 < n)
        m2 = (jt2 >= 0) & (jt2 < n)
        G[k, m1] += (1.0 - h[jt1[m1]])
        G[k, m2] -= h[jt2[m2]]
    if not np.all(np.isfinite(G)):
        return None, None
    return q, G

def slp_search(h0, dx, rng):
    n = len(h0)
    h = h0.copy()
    best = h.copy()
    best_score = exact_score(h, dx)
    if best_score is None:
        return best, best_score
    rho = 0.08
    rho_max = 0.25
    rho_min = 1e-5
    max_iter = 50 if n <= 128 else (25 if n <= 512 else 12)
    target = n / 2.0
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    for _ in range(max_iter):
        if rho < rho_min:
            break
        q, G = compute_lag_grad(h, n)
        if q is None:
            break
        Q = float(np.max(q))
        m = 2 * n - 1
        A_ub = np.zeros((m, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = -q
        bounds = []
        for i in range(n):
            bounds.append((max(-rho, -h[i]), min(rho, 1.0 - h[i])))
        bounds.append((None, None))
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'maxiter': 300, 'presolve': True})
        except Exception:
            rho = max(rho_min, rho / 2.0)
            continue
        if res.status != 0 or res.x is None or len(res.x) != n + 1:
            rho = max(rho_min, rho / 2.0)
            continue
        d = res.x[:n]
        z_val = float(res.x[n])
        if not np.all(np.isfinite(d)):
            rho = max(rho_min, rho / 2.0)
            continue
        old_h = h.copy()
        cand = old_h + d
        proj = capped_simplex_project(cand, target)
        if proj is None or not is_feasible(proj, n):
            rho = max(rho_min, rho / 2.0)
            continue
        s_new = exact_score(proj, dx)
        if s_new is None:
            rho = max(rho_min, rho / 2.0)
            continue
        if s_new < best_score - 1e-12:
            best = proj.copy()
            best_score = s_new
        q_new = float(np.max(np.correlate(proj, 1.0 - proj, mode='full')))
        predicted = Q - z_val
        actual = Q - q_new
        accepted = actual > 1e-12 * max(1.0, abs(Q))
        if accepted:
            h = proj.copy()
            if predicted > 1e-10 * max(1.0, abs(Q)) and np.isfinite(predicted):
                ratio = actual / predicted
            else:
                ratio = 0.0
            if np.isfinite(ratio):
                if ratio > 0.75:
                    step_size = float(np.max(np.abs(proj - old_h)))
                    if step_size >= 0.8 * rho:
                        rho = min(rho_max, 1.5 * rho)
                elif ratio < 0.1:
                    rho = max(rho_min, rho / 2.0)
        else:
            rho = max(rho_min, rho / 2.0)
    return best, best_score

def pairwise_poll(h, dx, rng, n):
    target = n / 2.0
    best = h.copy()
    best_s = exact_score(best, dx)
    if best_s is None:
        return best
    if n > 256:
        return best
    n_pairs = 24 if n <= 128 else 8
    radii = [0.08, 0.025, 0.008]
    for r in radii:
        for _ in range(n_pairs):
            a = int(rng.integers(n))
            b = int(rng.integers(n))
            attempts = 0
            while b == a and attempts < 50:
                b = int(rng.integers(n))
                attempts += 1
            if b == a:
                continue
            lo_s = max(-best[a], best[b] - 1.0, -r)
            hi_s = min(1.0 - best[a], best[b], r)
            if lo_s >= hi_s - 1e-14:
                continue
            s_set = []
            for sv in [lo_s, hi_s, -r, -0.5 * r, 0.5 * r, r]:
                sv_c = max(lo_s, min(hi_s, sv))
                if abs(sv_c) > 1e-14:
                    s_set.append(round(sv_c, 12))
            s_set = list(set(s_set))
            pair_best = best_s
            pair_best_h = None
            for sv in s_set:
                if sv < lo_s - 1e-14 or sv > hi_s + 1e-14:
                    continue
                cand = best.copy()
                cand[a] += sv
                cand[b] -= sv
                proj = capped_simplex_project(cand, target)
                if proj is None or not is_feasible(proj, n):
                    continue
                sc = exact_score(proj, dx)
                if sc is not None and sc < pair_best - 1e-12:
                    pair_best = sc
                    pair_best_h = proj.copy()
            if pair_best_h is not None:
                best = pair_best_h
                best_s = pair_best
    return best

def get_parent():
    try:
        p = initial_h_values
    except NameError:
        return None, 0
    try:
        arr = np.array(p, dtype=np.float64).ravel()
    except Exception:
        return None, 0
    n = len(arr)
    if n < 2 or n > 999:
        return None, 0
    if not np.all(np.isfinite(arr)) or np.max(np.abs(arr)) > 1e6:
        return None, n
    tgt = n / 2.0
    if np.min(arr) >= -1e-10 and np.max(arr) <= 1.0 + 1e-10 and abs(np.sum(arr) - tgt) <= 1e-8:
        proj = capped_simplex_project(arr, tgt)
        if proj is not None:
            return proj, n
    return None, n

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent, parent_n = get_parent()
    n = parent_n if parent is not None else 70
    dx = 2.0 / n
    target = n / 2.0
    u = np.full(n, 0.5)
    fallback = parent.copy() if parent is not None else u.copy()
    fb_score = exact_score(fallback, dx)
    if fb_score is None:
        fb_score = 0.5
    starts = []
    if parent is not None:
        starts.append(parent.copy())
        starts.append(parent[::-1].copy())
        for scale in [0.035, 0.12]:
            proj = capped_simplex_project(parent + scale * rng.standard_normal(n), target)
            if proj is not None:
                starts.append(proj)
    starts.append(u.copy())
    for _ in range(3):
        proj = capped_simplex_project(rng.random(n), target)
        if proj is not None:
            starts.append(proj)
    for _ in range(2):
        proj = capped_simplex_project(rng.beta(0.5, 0.5, size=n), target)
        if proj is not None:
            starts.append(proj)
    unique_starts = []
    for s in starts:
        dup = False
        for us in unique_starts:
            if np.max(np.abs(s - us)) < 1e-10:
                dup = True
                break
        if not dup:
            unique_starts.append(s)
    starts = unique_starts
    best_h = fallback.copy()
    best_c = fb_score
    for s in starts:
        if not is_feasible(s, n):
            continue
        h_slp, sc = slp_search(s, dx, rng)
        if sc is not None and sc < best_c - 1e-12:
            best_h = h_slp.copy()
            best_c = sc
        h_pp = pairwise_poll(h_slp, dx, rng, n)
        sc_pp = exact_score(h_pp, dx)
        if sc_pp is not None and sc_pp < best_c - 1e-12:
            best_h = h_pp.copy()
            best_c = sc_pp
        if best_c <= 0.3808:
            break
    for scale in [0.025, 0.08]:
        if best_c <= 0.3808:
            break
        noisy = best_h + scale * rng.standard_normal(n)
        proj = capped_simplex_project(noisy, target)
        if proj is None or not is_feasible(proj, n):
            continue
        h_slp, sc = slp_search(proj, dx, rng)
        if sc is not None and sc < best_c - 1e-12:
            best_h = h_slp.copy()
            best_c = sc
        h_pp = pairwise_poll(h_slp, dx, rng, n)
        sc_pp = exact_score(h_pp, dx)
        if sc_pp is not None and sc_pp < best_c - 1e-12:
            best_h = h_pp.copy()
            best_c = sc_pp
    proj = capped_simplex_project(best_h, target)
    if proj is not None and is_feasible(proj, n):
        best_h = proj
    else:
        best_h = fallback.copy()
    final_c = exact_score(best_h, dx)
    if final_c is None or final_c > fb_score:
        best_h = fallback.copy()
        final_c = fb_score
    if final_c is None:
        best_h = u.copy()
        final_c = exact_score(u, dx)
        if final_c is None:
            final_c = 0.5
    return (best_h, final_c, n)
