                    if cur < best_s - 1e-12:
                        best_h = h.copy()
                        best_s = cur
                    eta = 0.25
                elif math.exp(-delta / T) > rng.random():
                    h = bp
                    cur = bs2
                    eta = 0.25
    best_s = float(exact_score(best_h, n))
    return best_h, best_s

def run(seed=42, **kwargs):
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    rng = np.random.default_rng(seed)
    n = 42
    parent_h = None
    parent = globals().get('initial_h_values', None)
    if parent is not None:
        try:
            pa = np.asarray(parent, dtype=np.float64)
            if pa.ndim == 1 and len(pa) >= 2:
                n = len(pa)
                if np.all(np.isfinite(pa)):
                    parent_h = pa.copy()
        except Exception:
            pass
    if parent is None or parent_h is None:
        try:
            nv = kwargs.get('n_points', None)
            if nv is not None:
                ni = int(nv)
                if 2 <= ni <= 512:
                    n = ni
        except Exception:
            pass
    const_h = np.full(n, 0.5)
    const_s = float(exact_score(const_h, n))
    inc_h, inc_s = const_h.copy(), const_s
    if parent_h is not None:
        if is_feasible(parent_h, n):
            ps = float(exact_score(parent_h, n))
        else:
            p = project_capped_simplex(parent_h, n)
            if p is None:
                parent_h = None
            else:
                parent_h = p
                ps = float(exact_score(p, n))
        if parent_h is not None and ps < inc_s - 1e-12:
            inc_h, inc_s = parent_h.copy(), ps
    starts = []
    try_add(starts, inc_h, n, 12)
    try_add(starts, const_h, n, 12)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    try_add(starts, blk, n, 12)
    alt = np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n)])
    try_add(starts, alt, n, 12)
    ii = np.arange(n, dtype=np.float64)
    sinv = 0.5 + 0.45 * np.sin(2.0 * np.pi * ii / n)
    try_add(starts, sinv, n, 12)
    if parent_h is not None:
        for sig in [0.05, 0.15, 0.35, 0.15]:
            try_add(starts, parent_h + sig * rng.standard_normal(n), n, 12)
        for _ in range(2):
            try_add(starts, rng.random(n), n, 12)
    else:
        for _ in range(4):
            try_add(starts, rng.random(n), n, 12)
    if n <= 128:
        mi = 120 if n <= 64 else 80
        for s in starts:
            try:
                bh, bs = local_search(s, rng, n, mi)
                if is_feasible(bh, n) and bs < inc_s - 1e-12:
                    inc_h, inc_s = bh.copy(), bs
            except Exception:
                pass
    cands = [(inc_h, inc_s), (const_h, const_s)]
    if parent_h is not None:
        cands.append((parent_h, float(exact_score(parent_h, n))))
    fh, fs = cands[0]
    for ch, cs in cands[1:]:
        if cs < fs - 1e-12:
            fh, fs = ch, cs
    if not is_feasible(fh, n):
        p = project_capped_simplex(fh, n)
        if p is not None and is_feasible(p, n):
            fh = p
        else:
            fh = const_h.copy()
        fs = float(exact_score(fh, n))
    else:
        fs = float(exact_score(fh, n))
    return fh, fs, n
