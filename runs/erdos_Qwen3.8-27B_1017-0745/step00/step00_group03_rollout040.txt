def project_box_sum(z, N):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or z.shape[0] != N or not np.all(np.isfinite(z)) or np.any(np.abs(z) > 1e100):
        return np.full(N, 0.5)
    m = N / 2.0
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)
    for _ in range(3):
        e = m - np.sum(h)
        if abs(e) < 1e-14:
            break
        h = np.clip(h + e / N, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(np.sum(h) - m) > 1e-10 * max(1, N):
        return np.full(N, 0.5)
    return h


def is_feasible(h, N):
    m = N / 2.0
    return (isinstance(h, np.ndarray) and h.shape == (N,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - m) <= 1e-10 * max(1, N))


def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / N))


def lag_values_gradients(h, N):
    q = np.correlate(h, 1.0 - h, mode='full')
    G = np.zeros((2 * N - 1, N), dtype=np.float64)
    for r in range(2 * N - 1):
        k = r - (N - 1)
        if k >= 0:
            L = N - k
            if L <= 0:
                continue
            G[r, k:] += 1.0 - h[:L]
            G[r, :L] -= h[k:k + L]
        else:
            s = -k
            L = N - s
            if L <= 0:
                continue
            G[r, :L] += 1.0 - h[s:s + L]
            G[r, s:] -= h[:L]
    return q, G


def build_solver(N, Delta):
    d = cp.Variable(N)
    u = cp.Variable()
    t = cp.Variable()
    x_p = cp.Parameter(N)
    r0_p = cp.Parameter()
    J0_p = cp.Parameter(N)
    r_nz_p = cp.Parameter(2 * N - 2)
    J_nz_p = cp.Parameter((2 * N - 2, N))
    R_p = cp.Parameter()
    con = [x_p + d >= 0, x_p + d <= 1, cp.sum(d) == 0, cp.norm(d, 2) <= R_p,
           u >= r_nz_p + J_nz_p @ d, t >= u + Delta * cp.sum_squares(d), t >= r0_p + J0_p @ d]
    prob = cp.Problem(cp.Minimize(t), con)
    return d, u, t, x_p, r0_p, J0_p, r_nz_p, J_nz_p, R_p, prob


def solve_model(prob, d_var, N):
    try:
        prob.solve(solver='ECOS', verbose=False, max_iter=300, abstol=1e-8, reltol=1e-8, feastol=1e-8)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    dv = d_var.value
    if dv is None or dv.shape != (N,) or not np.all(np.isfinite(dv)):
        return None
    return dv
