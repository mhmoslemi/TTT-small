def score_exact(h, N):

def lag_values(h, N):
    M = 2 * N - 1
    q = np.zeros(M)
    for k in range(0, N):
        L = N - k
        a = h[k:k + L]
        b = h[0:L]
        q[k + N - 1] = np.dot(a, 1.0 - b)

        L = N - k
        a = h[0:L]
        b = h[k:k + L]
        q[N - 1 - k] = np.dot(a, 1.0 - b)
    return q

    return float(rmax + tau * math.log(s))
