def project_box_sum(v, S):

            best_h = bh.copy()
    fin = project_box_sum(best_h, S)
    if fin is None:
        fin = parent.copy()
    sc = exact_score(fin)
    if sc is None or not np.isfinite(sc):
        fin = parent.copy()
        sc = exact_score(fin)
        if sc is None:
            sc = 1.0
    return (fin, float(sc), n)
