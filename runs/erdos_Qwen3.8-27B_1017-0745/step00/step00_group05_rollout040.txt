def quad_roots(A, B, C):
    # Solve A*x^2 + B*x + C = 0, return list of real roots
    roots = []
    tol = 1e-14 * max(1.0, abs(A), abs(B), abs(C))
    if abs(A) < tol:
        if abs(B) > tol:
            r = -C / B
            if np.isfinite(r):
                roots.append(r)
    else:
        disc = B * B - 4.0 * A * C
        if disc < -tol:
            return roots
        if disc < 0:
            disc = 0.0
        sq = math.sqrt(disc)
        if abs(B) < tol:
            r1 = sq / (2.0 * A)
            r2 = -sq / (2.0 * A)
            if np.isfinite(r1): roots.append(r1)
            if np.isfinite(r2) and abs(r2 - r1) > 1e-14: roots.append(r2)
        else:
            q = -0.5 * (B + math.copysign(sq, B))
            r1 = q / A
            r2 = C / q if abs(q) > tol else -B / (2.0 * A)
            if np.isfinite(r1): roots.append(r1)
            if np.isfinite(r2) and abs(r2 - r1) > 1e-14: roots.append(r2)

def build_directions(h, n, rng):
    dirs = []
    q = np.correlate(h, 1.0 - h, mode='full')
    top3 = np.argsort(-q)[:3]
    for t_idx in top3:
        t = t_idx - (n - 1)
        g = lag_grad(h, t, n)
        d = norm_dir(-g)
        if d is not None:
            dirs.append(d)
    n_cos = min(6, n - 1) if n <= 96 else 4
    idx = np.arange(n)
    for k in range(1, n_cos + 1):
        d = norm_dir(np.cos(math.pi * k * (idx + 0.5) / n))
        if d is not None:
            dirs.append(d)
    # Deterministic pairs
    n_pairs = 6 if n <= 96 else 4
    for r in range(n_pairs):
        a_i = int(r * n / n_pairs)
        b_i = (a_i + 1) % n if r % 2 == 0 else (a_i + max(1, n // 2)) % n
        if b_i == a_i:
            b_i = (a_i + 1) % n
        d = np.zeros(n)
        d[a_i] = 1.0
        d[b_i] = -1.0
        dirs.append(d)
    # Random pairs
    n_rpairs = 4 if n <= 96 else 2
    for _ in range(n_rpairs):
        pair = rng.integers(0, n, size=2)
        if pair[0] == pair[1]:
            pair[1] = (pair[1] + 1) % n
        d = np.zeros(n)
        d[pair[0]] = 1.0
        d[pair[1]] = -1.0
        dirs.append(d)
    # Random dense
    n_rand = 2 if n <= 96 else 1
    for _ in range(n_rand):
        d = norm_dir(rng.standard_normal(n))
        if d is not None:
            dirs.append(d)
    return dirs

def local_search(start, dx, rng, sweep_limit):
    n = len(start)
    target = float(n) / 2.0
    h = start.copy()
    best = h.copy()
    cur_score = exact_score(h, dx)
    best_score = cur_score
    for sweep in range(sweep_limit):
        if cur_score <= 0.3808:
            break
        dirs = build_directions(h, n, rng)
        for d in dirs:
            if d is None or not np.all(np.isfinite(d)):

            h2, s2 = line_search(h, d, dx, n)
            if s2 < cur_score - 1e-12:
                h = h2
                cur_score = s2
                if cur_score < best_score:
                    best_score = cur_score
                    best = h.copy()
    return best, best_score
