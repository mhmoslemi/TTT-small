        n_default = max(8, min(512, int(kwargs['n_points'])))
    gav = globals().get('initial_h_values', None)
    if gav is not None:
        try:
            p = np.array(gav, dtype=np.float64).ravel()
            if p.size >= 2 and np.all(np.isfinite(p)):
                parent = p.copy()
        except Exception:
            pass
    n = len(parent) if parent is not None else n_default
    dx = 2.0 / n
    S = n / 2.0
    fallback = project_capped_simplex(parent, S) if parent is not None else np.full(n, 0.5)
    best_h = fallback.copy()
    best_score = exact_score(fallback)
    # Build starts
    starts = [fallback]
    if parent is not None:
        for sd in [0.04, 0.15, 0.35]:
            starts.append(project_capped_simplex(parent + rng.normal(0, sd, n), S))
    q = int(rng.integers(1, n))
    phi = rng.uniform(0, 2 * np.pi)
    xs = np.arange(n, dtype=np.float64)
    starts.append(project_capped_simplex(0.5 + 0.45 * np.cos(2 * np.pi * q * xs / n + phi), S))
    starts.append(project_capped_simplex(rng.uniform(0, 1, n), S))
    perm = rng.permutation(n)
    binary = np.zeros(n)
    binary[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        binary[perm[n // 2]] = 0.5
    starts.append(binary)
    # Deduplicate
    filt = [starts[0]]
    for s in starts[1:]:
        if all(np.max(np.abs(s - f)) >= 1e-10 for f in filt):
            filt.append(s)
    max_starts = 7 if n <= 256 else 3
    filt = filt[:max_starts]
    # Score starts
    baseline = best_score
    for s in filt:
        sc = exact_score(s)
        if sc < best_score - 1e-12:
            best_score = sc
            best_h = s.copy()
    # Optimize
    B = min(480, max(12, 4000000 // (n * n)))
    for s in filt:
        try:
            ho = optimize_start(s, S, rng, B)
            sc = exact_score(ho)
            if sc < best_score - 1e-12:
                best_score = sc
                best_h = ho.copy()
        except Exception:
            pass
    # Diversification
    if best_score >= baseline - 1e-8 and n <= 256:
        for sd in [0.6, 1.0]:
            try:
                div = project_capped_simplex(best_h + rng.normal(0, sd, n), S)
                ho = optimize_start(div, S, rng, max(1, B // 2))
                sc = exact_score(ho)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_h = ho.copy()
            except Exception:
                pass
    # Pair-transfer polish
    P = max(80, min(800, 10000000 // (n * n)))
    steps = [0.08, 0.025, 0.008, 0.002]
    temps = [0.001, 0.0003, 0.00008, 0.0]
    cur = best_h.copy()
    cur_sc = best_score
    ps = P // 4
    for stage in range(4):
        sl = steps[stage]
        T = temps[stage]
        for _ in range(ps):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            sign = 1.0 if rng.random() < 0.5 else -1.0
            if sign > 0:
                cap = min(1.0 - cur[i], cur[j])
            else:
                cap = min(cur[i], 1.0 - cur[j])
            if cap < 1e-12:
                continue
            amt = rng.uniform(0, min(sl, cap)) * sign
            if abs(amt) < 1e-12:
                continue
            trial = cur.copy()
            trial[i] += amt
            trial[j] -= amt
            if trial[i] < -1e-12 or trial[i] > 1 + 1e-12 or trial[j] < -1e-12 or trial[j] > 1 + 1e-12:
                continue
            trial[i] = float(np.clip(trial[i], 0, 1))
            trial[j] = float(np.clip(trial[j], 0, 1))
            ts = exact_score(trial)
            if ts <= cur_sc or (T > 0 and math.log(rng.random()) < -(ts - cur_sc) / T):
                cur = trial
                cur_sc = ts
                if ts < best_score - 1e-14:
                    best_score = ts
                    best_h = cur.copy()
    # Final check
    best_h = np.clip(best_h, 0.0, 1.0)
    resid = S - np.sum(best_h)
    if abs(resid) > 1e-14:
        for i in range(n):
            if resid > 0 and best_h[i] < 1.0:
                best_h[i] = min(1.0, best_h[i] + resid)
                break
            elif resid < 0 and best_h[i] > 0.0:
                best_h[i] = max(0.0, best_h[i] + resid)
                break
    fs = exact_score(best_h)
    if not np.isfinite(fs) or fs > exact_score(fallback):
        best_h = fallback.copy()
        fs = exact_score(best_h)
    return (best_h, fs, n)
