        add_start(parent[::-1], starts, n, S)
        add_start(parent + 0.18 * rng.standard_normal(n), starts, n, S)

    max_q = min(6, max(1, (n - 1) // 2))
    xx = np.arange(n, dtype=np.float64)
    q1 = int(rng.integers(1, max_q + 1))
    phi1 = float(rng.uniform(0.0, 2.0 * np.pi))
    add_start(0.5 + 0.45 * np.cos(2.0 * np.pi * q1 * xx / n + phi1), starts, n, S)

    perm = rng.permutation(n)
    b = np.zeros(n)
    b[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        b[perm[n // 2]] = 0.5
    add_start(b, starts, n, S)

    if parent is None:
        q2 = int(rng.integers(1, max_q + 1))
        phi2 = float(rng.uniform(0.0, 2.0 * np.pi))
        add_start(0.5 + 0.45 * np.cos(2.0 * np.pi * q2 * xx / n + phi2), starts, n, S)
        add_start(0.5 + 0.30 * rng.standard_normal(n), starts, n, S)
        add_start(rng.uniform(0.0, 1.0, n), starts, n, S)

    max_starts = 6 if n <= 128 else 3
    starts = starts[:max_starts]

    best_h = None
    best_score = float('inf')
    for s in starts:
        sc = exact_score(s)
        if sc is not None and sc < best_score:
            best_score = sc
            best_h = s.copy()
    if best_h is None:
        best_h = const_h.copy()
        best_score = exact_score(const_h) if exact_score(const_h) is not None else 0.25

    if n <= 128:
        bounds = [(0.0, 1.0)] * n + [(None, None)]
        for h0 in starts:
            try:
                B0, _ = compute_lags(h0)
                t0 = float(np.max(B0)) + 1e-8 * max(1.0, abs(float(np.max(B0))))
                z0 = np.append(h0.copy(), t0)
                res = minimize(
                    fun=_obj, x0=z0, jac=_obj_grad, method='SLSQP',
                    bounds=bounds,
                    constraints=[
                        {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac},
                        {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}
                    ],
                    options={'maxiter': 180, 'ftol': 1e-9, 'disp': False}
                )
                if res is not None and hasattr(res, 'x') and len(res.x) == n + 1:
                    h_sol = res.x[:n]
                    if np.all(np.isfinite(h_sol)):
                        proj = project_capped_simplex(h_sol, S)
                        if proj is not None:
                            sc = exact_score(proj)
                            if sc is not None and sc < best_score:
                                best_score = sc
                                best_h = proj.copy()
            except Exception:
                pass

    pool = [best_h, const_h]
    if parent is not None:
        pool.append(parent)

    final_h = const_h.copy()
    final_score = exact_score(const_h)
    if final_score is None:
        final_score = 0.25

    for c in pool:
        if c is None:
            continue
        c = np.asarray(c, dtype=np.float64).ravel()
        if c.size != n or not np.all(np.isfinite(c)):
            continue
        if np.min(c) < -1e-12 or np.max(c) > 1.0 + 1e-12 or abs(np.sum(c) - S) > 1e-10:
            c = project_capped_simplex(c, S)
            if c is None:
                continue
        sc = exact_score(c)
        if sc is not None and sc < final_score:
            final_score = sc
            final_h = c.copy()

    final_h = np.clip(final_h, 0.0, 1.0)
    fs = exact_score(final_h)
    if fs is not None:
        final_score = fs
    else:
        final_h = np.full(n, 0.5)
        final_score = float(np.max(np.correlate(final_h, 1.0 - final_h, mode='full') * (2.0 / n)))

    return (final_h.astype(np.float64), float(final_score), n)
