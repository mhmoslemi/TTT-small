        if len(feasible) >= n_starts:
            break
    sds = [0.08, 0.18, 0.30]
    ni = 0
    while len(feasible) < n_starts and ni < 20:
        base = parent if np.all(np.isfinite(parent)) else np.full(N, 0.5)
        pert = base + sds[ni % 3] * rng.standard_normal(N)
        ps = project_capped(pert, N, M)
        if ps is not None:
            dup = False
            for f in feasible:
                if float(np.max(np.abs(ps - f))) < 1e-8:
                    dup = True
                    break
            if not dup:
                feasible.append(ps)
        ni += 1
    return feasible

def run(seed=42, **kwargs):
    N = None
    parent = None
    try:
        arr = np.asarray(initial_h_values, dtype=np.float64)
        if arr.ndim == 1 and arr.size >= 2 and np.all(np.isfinite(arr)) and float(np.max(np.abs(arr))) < 1e6:
            N = arr.size
            parent = arr.copy()
    except Exception:
        pass
    if N is None:
        try:
            N = int(kwargs.get('n_points', 95))
        except Exception:
            N = 95
        N = max(2, min(999, N))
    M = N / 2.0

    if parent is not None:
        parent = project_capped(parent, N, M)
    if parent is None:
        parent = np.full(N, 0.5)
    if not np.all(np.isfinite(parent)):
        parent = np.full(N, 0.5)

    inc = parent.copy()
    inc_score = exact_score(inc)
    if not np.isfinite(inc_score):
        inc = np.full(N, 0.5)
        inc_score = exact_score(inc)

    if N > 999:
        return inc, float(inc_score), N

    try:
        n_starts = int(kwargs.get('n_starts', 6))
    except Exception:
        n_starts = 6
    n_starts = max(1, min(8, n_starts))
    try:
        maxiter = int(kwargs.get('maxiter', 40))
    except Exception:
        maxiter = 40
    maxiter = max(10, min(80, maxiter))
    if N <= 200:
        n_starts = min(n_starts, 6)
        maxiter = min(maxiter, 40)
    elif N <= 500:
        n_starts = min(n_starts, 4)
        maxiter = min(maxiter, 20)
    else:
        n_starts = min(n_starts, 2)
        maxiter = min(maxiter, 10)

    try:
        rng_seed = int(seed) % (2**32)
    except Exception:
        rng_seed = 42
    rng = np.random.default_rng(rng_seed)

    feasible_starts = _make_starts(parent, N, M, rng, n_starts)

    for s in feasible_starts:
        sc = exact_score(s)
        if np.isfinite(sc) and sc < inc_score - 1e-12:
            inc = s.copy()
            inc_score = sc

    delta_min = 1e-4
    delta_max = 0.25

    for s in feasible_starts:
        x = s.copy()
        x_score = exact_score(x)
        delta = 0.10
        stop = False
        for _it in range(maxiter):
            if stop:
                break
            R, G = compute_R_G(x, N)
            if not (np.all(np.isfinite(R)) and np.all(np.isfinite(G))):
                break
            Nlag = 2 * N - 1
            c_obj = np.zeros(N + 1)
            c_obj[N] = 1.0
            A_ub = np.hstack([G, -np.ones((Nlag, 1))])
            b_ub = -R
            A_eq = np.zeros((1, N + 1))
            A_eq[0, :N] = 1.0
            b_eq = np.zeros(1)
            bounds = []
            for i in range(N):
                bounds.append((max(-x[i], -delta), min(1.0 - x[i], delta)))
            bounds.append((None, None))
            failed = False
            try:
                res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                              bounds=bounds, method='highs',
                              options={'presolve': True, 'disp': False})
                if res is None or not res.success or not np.all(np.isfinite(res.x)):
                    failed = True
            except Exception:
                failed = True
            if failed:
                delta = max(delta_min, 0.5 * delta)
                if delta - delta_min < 1e-10:
                    stop = True
                continue
            u = res.x[:N]
            y_raw = x + u
            step_mag = float(np.max(np.abs(u)))
            y = project_capped(y_raw, N, M)
            if y is None or float(np.max(np.abs(y - x))) > delta + 1e-7:
                delta = max(delta_min, 0.5 * delta)
                if delta - delta_min < 1e-10:
                    stop = True
                continue
            y_score = exact_score(y)
            if not np.isfinite(y_score):
                delta = max(delta_min, 0.5 * delta)
                if delta - delta_min < 1e-10:
                    stop = True
                continue
            L_val = float(np.max(R + G @ (y - x)))
            pred = x_score - L_val
            actual = x_score - y_score
            if y_score < inc_score - 1e-12:
                inc = y.copy()
                inc_score = y_score
            if actual > 1e-12:
                rho = actual / pred if pred > 1e-12 else 0.0
                x = y.copy()
                x_score = y_score
                if rho > 0.75 and step_mag >= 0.8 * delta:
                    delta = min(delta_max, 1.5 * delta)
            else:
                delta = max(delta_min, 0.5 * delta)
                if delta - delta_min < 1e-10:
                    stop = True

    final = project_capped(inc, N, M)
    if final is None or not np.all(np.isfinite(final)) or np.any(final < -1e-12) or np.any(final > 1.0 + 1e-12):
        final = np.full(N, 0.5)
    if abs(float(np.sum(final)) - M) > 1e-9 * max(1, N):
        final = np.full(N, 0.5)
    c5 = exact_score(final)
    if not np.isfinite(c5):
        final = np.full(N, 0.5)
        c5 = exact_score(final)
    return final.astype(np.float64), float(c5), N
