import numpy as np
from scipy.optimize import minimize
import math

def capped_simplex_project(z, S):
    z = np.asarray(z, dtype=np.float64)
    n = len(z)
    if n < 1 or not np.all(np.isfinite(z)):
        return None
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(80):
        mid = lo + 0.5 * (hi - lo)
        y = np.clip(z - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    mid = lo + 0.5 * (hi - lo)
    y = np.clip(z - mid, 0.0, 1.0)
    r = S - np.sum(y)
    if abs(r) < 1e-14:
        pass
    elif r > 0:
        for i in range(n):
            if r <= 0:
                break
            slack = 1.0 - y[i]
            if slack > 0:
                add = min(r, slack)
                y[i] += add
                r -= add
    elif r < 0:
        for i in range(n):
            if r >= 0:
                break
            slack = y[i]
            if slack > 0:
                sub = min(-r, slack)
                y[i] -= sub
                r += sub
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1 + 1e-12) or abs(np.sum(y) - S) > 1e-10:
        return None
    y = np.clip(y, 0.0, 1.0)
    return y

def check_feasible(h, n):
    if h is None or h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < 0) or np.any(h > 1):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10:
        return False
    return True

def exact_score(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n < 1 or not np.all(np.isfinite(h)):
        return None
    if np.any(h < 0) or np.any(h > 1):
        return None
    if abs(np.sum(h) - n / 2.0) > 1e-10:
        return None
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode="full") * dx
    if not np.all(np.isfinite(vals)):
        return None
    s = float(np.max(vals))
    if not math.isfinite(s):
        return None
    return s

def compute_lags(h):
    n = len(h)
    B = np.zeros(2 * n - 1)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        src = slice(lo, hi)
        dst = slice(lo + d, hi + d)
        B[row] = np.dot(h[src], 1.0 - h[dst])
        G[row, src] += (1.0 - h[dst])
        G[row, dst] -= h[src]
    return B, G

def _obj(z):
    return z[-1]

def _obj_grad(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def _ineq(z):
    n = len(z) - 1
    h = z[:n]
    t = z[-1]
    B, _ = compute_lags(h)
    return t - B

def _ineq_jac(z):
    n = len(z) - 1
    h = z[:n]
    _, G = compute_lags(h)
    J = np.hstack([-G, np.ones((2 * n - 1, 1))])
    return J

def _eq_con(z):
    n = len(z) - 1
    return z[:n].sum() - n / 2.0

def _eq_jac(z):
    n = len(z) - 1
    row = np.zeros(n + 1)
    row[:n] = 1.0
    return row.reshape(1, -1)

def build_starts(n, S, rng, parent):
