def smooth_gradient(h, n, tau):
    dx = 2.0 / n

    cm = float(np.max(c))
    a = np.exp((c - cm) / tau)
    ss = float(np.sum(a))

        return None

    V = signal.convolve(p[::-1], h, mode='full')
    sl = n - 1
    grad = dx * (U[sl:sl + n] - V[sl:sl + n])
