    n_levels = len(taus)
    iters_per_level = max(1, B // n_levels)
    best_score = compute_score(h)
    S = n / 2.0
    
    for li, tau in enumerate(taus):
        iters = iters_per_level + (1 if li < B % n_levels else 0)
        eta = 1.0
        h_curr = h.copy()
        F_curr, g_curr = surrogate_and_grad(h_curr, dx, tau)

            # Check for zero step
            # Try trial
            trial_ok = False
            for _ in range(16):
                if eta < 1e-8:
                    break
                trial = project_simplex(h_curr - eta * g_curr, S)
                d_step = trial - h_curr
                if np.all(np.abs(d_step) < 1e-14):

                    break

                    eta *= 0.5
                    continue

                    break
                else:
                    eta *= 0.5
            if not trial_ok:
                break
        
        h = h_curr

def surrogate_and_grad(h, dx, tau):

    scores = np.empty(2 * n - 1)

        d = idx - (n - 1)
        lo = max(0, -d)

    if parent is not None:
        h_start = project_simplex(parent.astype(np.float64).copy(), S)
    else:
        h_start = np.full(n, 0.5)

    fallback = h_start.copy()
    fallback_s = compute_score(fallback)

    starts = [h_start]
    i_arr = np.arange(n)
    if parent is not None:
        for sd in [0.04, 0.15, 0.35]:
            starts.append(project_simplex(parent + sd * rng.standard_n(n), S))
        q = int(rng.integers(1, n))
        phi = float(rng.uniform(0, 2 * math.pi))
        starts.append(project_simplex(0.5 + 0.45 * np.cos(2 * math.pi * q * i_arr / n + phi), S))
    else:
        for q in [3, 7]:
            phi = float(rng.uniform(0, 2 * math.pi))
            starts.append(project_simplex(0.5 + 0.45 * np.cos(2 * math.pi * q * i_arr / n + phi), S))
    starts.append(project_simplex(rng.random(n), S))
    perm = rng.permutation(n)
    h_bin = np.zeros(n)
    h_bin[perm[:int(n // 2)]] = 1.0
    if n % 2 == 1:
        h_bin[perm[int(n // 2)]] = 0.5
    starts.append(h_bin)

    unique = [starts[0]]
    for s in starts[1:]:
        if min(np.max(np.abs(s - u)) for u in unique) > 1e-10:
            unique.append(s)
    max_starts = 7 if n <= 256 else 3
    unique = unique[:max_starts]

    best_h = fallback.copy()
    best_score = fallback_s
    for s in unique:
        sc = compute_score(s)
        if sc < best_score:
            best_score = sc
            best_h = s.copy()

    B = min(240, max(12, int(500000 / (n * n))))
    for s in unique:
        try:
            h_opt, sc = optimize_start(s, dx, n, B, rng)
            if sc < best_score:
                best_score = sc
                best_h = h_opt.copy()
        except Exception:
            pass

    baseline = compute_score(unique[0])
    if best_score > baseline - 1e-8 and n <= 256:
        for sd in [0.6, 1.0]:
            s2 = project_simplex(best_h + sd * rng.standard_n(n), S)
            try:
                h_opt, sc = optimize_start(s2, dx, n, max(4, B // 2), rng)
                if sc < best_score:
                    best_score = sc
                    best_h = h_opt.copy()
            except Exception:
                pass

    P = max(80, min(800, int(10000000 / (n * n))))
    try:
        h_pol, s_pol = pair_transfer_polish(best_h, dx, rng, P)
        if s_pol < best_score:
            best_score = s_pol
            best_h = h_pol.copy()
    except Exception:
        pass

    final = project_simplex(np.clip(best_h, 0.0, 1.0), S)
    final_s = compute_score(final)
    if final_s < fallback_s:
        return (final.astype(np.float64), float(final_s), n)
    return (fallback.astype(np.float64), float(fallback_s), n)
