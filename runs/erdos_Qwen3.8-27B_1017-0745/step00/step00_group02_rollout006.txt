        mid = 0.5 * (lo + hi)
        x = np.clip(y - mid, 0.0, 1.0)
        s = float(np.sum(x))
        if abs(s - S) <= 1e-13 or hi - lo <= 1e-14 * max(1.0, abs(mid)):
            break
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - mid, 0.0, 1.0)
    r = S - float(np.sum(x))
    if r > 1e-13:
        for j in range(n):
            if r <= 0: break
            a = min(r, 1.0 - x[j]); x[j] += a; r -= a
    elif r < -1e-13:
        for j in range(n):
            if r >= 0: break
            a = min(-r, x[j]); x[j] -= a; r += a
    x = np.clip(x, 0.0, 1.0)
    if abs(float(np.sum(x)) - S) > 1e-10:
        return None
    return x

def exact_score(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c) * (2.0 / n))

def is_feasible(h, n):
    return h.shape == (n,) and np.all(np.isfinite(h)) and np.all(h >= 0) and np.all(h <= 1) and abs(float(np.sum(h)) - n / 2.0) <= 1e-10

def gen_candidate(rng, n, S, base, mode, scale):
    if mode == 0:
        y = rng.uniform(0, 1, n)
    elif mode == 1:
        y = rng.beta(0.5, 0.5, n)
    elif mode == 2:
        y = rng.beta(2.0, 2.0, n)
    elif mode == 3:
        b = base if base is not None else np.full(n, 0.5)
        y = b + scale * rng.standard_normal(n)
    elif mode == 4:
        K = int(rng.integers(1, min(6, n // 2) + 1))
        y = np.full(n, 0.5)
        idx = np.arange(n)
        for k in range(1, K + 1):
            ak = 0.30 / math.sqrt(k) * rng.standard_normal()
            bk = 0.30 / math.sqrt(k) * rng.standard_normal()
            y += ak * np.cos(2.0 * np.pi * k * idx / n) + bk * np.sin(2.0 * np.pi * k * idx / n)
    else:
        y = rng.uniform(0, 1, n)
    if not np.all(np.isfinite(y)):
        return None
    return project_to_K(y, n, S)

def run(seed=42, **kwargs):
    n = 45
    parent = None
    ihv = globals().get('initial_h_values')
    if ihv is not None:
        try:
            arr = np.asarray(ihv, dtype=np.float64)
            if arr.ndim == 1 and 2 <= len(arr) <= 999:
                n = len(arr)
                if np.all(np.isfinite(arr)) and np.all(np.abs(arr) <= 1e6):
                    S_t = n / 2.0
                    if np.all(arr >= 0) and np.all(arr <= 1) and abs(float(np.sum(arr)) - S_t) <= 1e-12:
                        parent = arr.copy()
                    else:
                        pp = project_to_K(arr, n, S_t)
                        if pp is not None:
                            parent = pp
        except Exception:
            pass
    if parent is None and 'n_points' in kwargs:
        kv = kwargs['n_points']
        if isinstance(kv, int) and not isinstance(kv, bool) and 2 <= kv <= 999:
            n = kv
    S = n / 2.0
    fallback = parent.copy() if parent is not None else np.full(n, 0.5)
    fb_score = exact_score(fallback, n)
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 64:
        P = max(48, min(160, 3 * n))
        G = 120; stag_th = 24; max_rs = 2
    elif n <= 256:
        P = 64; G = 40; stag_th = 12; max_rs = 1
    else:
        P = 32; G = 20; stag_th = 8; max_rs = 1
    scales = [0.03, 0.08, 0.18, 0.35, 0.60]
    pop = np.empty((P, n))
    sc = np.empty(P)
    Fv = np.empty(P)
    Cv = np.empty(P)
    pop[0] = fallback; sc[0] = fb_score
    Fv[0] = rng.uniform(0.4, 0.9); Cv[0] = rng.uniform(0.25, 0.95)
    slot = 1
    if parent is not None:
        for v in [parent[::-1].copy(), 1.0 - parent.copy(), (1.0 - parent[::-1]).copy()]:
            if slot >= P: break
            pp = project_to_K(v, n, S)
            if pp is not None:
                pop[slot] = pp; sc[slot] = exact_score(pp, n)
            else:
                pop[slot] = fallback; sc[slot] = fb_score
            Fv[slot] = rng.uniform(0.4, 0.9); Cv[slot] = rng.uniform(0.25, 0.95)
            slot += 1
    mi = 0
    while slot < P:
        mode = mi % 5
        scale = scales[mi % len(scales)]
        c = gen_candidate(rng, n, S, parent, mode, scale)
        if c is not None:
            pop[slot] = c; sc[slot] = exact_score(c, n)
        else:
            pop[slot] = fallback; sc[slot] = fb_score
        Fv[slot] = rng.uniform(0.4, 0.9); Cv[slot] = rng.uniform(0.25, 0.95)
        slot += 1; mi += 1
    arch_h = fallback.copy(); arch_s = fb_score
    for i in range(P):
        if sc[i] < arch_s - 1e-12:
            arch_h = pop[i].copy(); arch_s = sc[i]
    stagnation = 0; n_restarts = 0
    for gen in range(G):
        fp = pop.copy(); fs = sc.copy(); fF = Fv.copy(); fC = Cv.copy()
        idx_s = np.argsort(fs)
        p_sz = max(2, math.ceil(0.2 * P))
        pb = idx_s[:p_sz]
        npop = np.empty((P, n)); nsc = np.empty(P); nF = np.empty(P); nC = np.empty(P)
        improved = False
        for i in range(P):
            Fi = fF[i]; Ci = fC[i]
            if rng.random() < 0.1: Fi = rng.uniform(0.3, 0.9)
            if rng.random() < 0.1: Ci = rng.uniform(0.1, 0.95)
            pb_cands = [int(j) for j in pb if j != i]
            if not pb_cands: pb_cands = [i]
            p_idx = int(rng.choice(pb_cands))
            rem = [j for j in range(P) if j != i and j != p_idx]
            r1 = int(rng.choice(rem))
            rem2 = [j for j in rem if j != r1]
            r2 = int(rng.choice(rem2))
            v = fp[i] + Fi * (fp[p_idx] - fp[i]) + Fi * (fp[r1] - fp[r2])
            mask = rng.random(n) < Ci
            mask[int(rng.integers(0, n))] = True
            w = np.where(mask, v, fp[i])
            trial = project_to_K(w, n, S)
            if trial is None:
                trial = fallback.copy()
            ts = exact_score(trial, n)
            if ts < fs[i] - 1e-12:
                npop[i] = trial; nsc[i] = ts; nF[i] = Fi; nC[i] = Ci
            else:
                npop[i] = fp[i]; nsc[i] = fs[i]; nF[i] = fF[i]; nC[i] = fC[i]
            if ts < arch_s - 1e-12:
                arch_h = trial.copy(); arch_s = ts; improved = True
        pop = npop; sc = nsc; Fv = nF; Cv = nC
        if not any(np.max(np.abs(pop[i] - arch_h)) < 1e-12 for i in range(P)):
            wi = int(np.argmax(sc))
            pop[wi] = arch_h; sc[wi] = arch_s; Fv[wi] = 0.6; Cv[wi] = 0.5
        if improved:
            stagnation = 0
        else:
            stagnation += 1
        if n_restarts < max_rs and stagnation >= stag_th:
            stagnation = 0; n_restarts += 1
            sigma = 0.06 if n_restarts == 1 else 0.18
            order = np.argsort(sc)
            keep = len(order) // 2
            for k in range(keep, P):
                if (k - keep) % 2 == 0:
                    y = arch_h + sigma * rng.standard_normal(n)
                    c = project_to_K(y, n, S)
                else:
                    c = gen_candidate(rng, n, S, arch_h, int(rng.integers(0, 5)), scales[int(rng.integers(0, len(scales)))])
                if c is None: c = arch_h.copy()
                pop[k] = c; sc[k] = exact_score(c, n)
                Fv[k] = rng.uniform(0.4, 0.9); Cv[k] = rng.uniform(0.25, 0.95)
                if sc[k] < arch_s - 1e-12:
                    arch_h = c.copy(); arch_s = sc[k]
    best_h = arch_h.copy()
    best_s = exact_score(best_h, n)
    if not is_feasible(best_h, n):
        pj = project_to_K(best_h, n, S)
        if pj is not None and is_feasible(pj, n):
            best_h = pj; best_s = exact_score(best_h, n)
        else:
            best_h = fallback.copy(); best_s = fb_score
    fb_s = exact_score(fallback, n)
    if best_s > fb_s - 1e-12:
        best_h = fallback.copy(); best_s = fb_s
    if not is_feasible(best_h, n):
        best_h = np.full(n, 0.5); best_s = exact_score(best_h, n)
    return best_h, best_s, n
