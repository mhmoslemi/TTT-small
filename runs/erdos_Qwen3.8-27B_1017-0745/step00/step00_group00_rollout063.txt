import numpy as np
from scipy.optimize import minimize

def project_box_sum(v, S):
    n = len(v)
    if not np.all(np.isfinite(v)):
        return np.full(n, S / n)
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if resid > 1e-14:
        for i in range(n):
            add = min(resid, 1.0 - y[i])
            y[i] += add
            resid -= add
            if resid <= 0:
                break
    elif resid < -1e-14:
        for i in range(n):
            sub = min(-resid, y[i])
            y[i] -= sub
            resid += sub
            if resid >= 0:
                break
    return y

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n * 0.5) <= 1e-10 * max(1, n))

def exact_score(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_q_and_grad(h):
    n = len(h)
    nl = 2 * n - 1
    Q = np.zeros(nl)
    G = np.zeros((nl, n))
    for kk in range(nl):
        k = kk - (n - 1)
        ilo = max(0, -k)
        ihi = min(n, n - k)
        idx = np.arange(ilo, ihi)
        js = idx + k
        Q[kk] = np.sum(h[idx] * (1.0 - h[js]))
        G[kk, idx] += (1.0 - h[js])
        G[kk, js] -= h[idx]
    return Q, G

def obj_fn(z):
    return z[-1]

def obj_jac_fn(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def ineq_fn(z):
    h = z[:-1]
    Q, _ = lag_q_and_grad(h)
    return z[-1] - Q / len(h)

def ineq_jac_fn(z):
    h = z[:-1]
    n = len(h)
    _, G = lag_q_and_grad(h)
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G / n
    J[:, n] = 1.0
    return J

def eq_fn(z):
    return np.sum(z[:-1]) / len(z[:-1]) - 0.5

def eq_jac_fn(z):
    n = len(z[:-1])
    J = np.zeros(n + 1)
    J[:n] = 1.0 / n
    return J

def slsqp_optimize(h0):
    n = len(h0)
    Q, _ = lag_q_and_grad(h0)
    t0 = min(1.0, np.max(Q) / n + 1e-4)
    z0 = np.append(h0, t0)
    bounds = [(0.0, 1.0)] * (n + 1)
    cons = [{'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac_fn},
            {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac_fn}]
    try:
        res = minimize(obj_fn, z0, jac=obj_jac_fn, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 400, 'ftol': 1e-10, 'disp': False})
        if res.x.shape == (n + 1,) and np.all(np.isfinite(res.x)):
            return res.x[:-1].copy()
    except Exception:
        pass
    return None

def run(seed=42, **kwargs):
    n = 45
    parent = None
    if 'initial_h_values' in globals():
        ph = np.array(globals()['initial_h_values'], dtype=np.float64).ravel()
        if (ph.ndim == 1 and 2 <= ph.size <= 999 and np.all(np.isfinite(ph))
                and np.all(ph >= -1e-6) and np.all(ph <= 1.0 + 1e-6)
                and abs(np.sum(ph) - ph.size * 0.5) < 1e-6):
            n = ph.size
            parent = project_box_sum(np.clip(ph, 0.0, 1.0), n * 0.5)

    rng = np.random.default_rng(seed)
    best_h = parent.copy() if parent is not None else np.full(n, 0.5)
    best_score = exact_score(best_h)

    starts = [best_h.copy()]
    if parent is not None:
        starts.append(parent[::-1].copy())
        for sd in [0.01, 0.03, 0.06, 0.12, 0.20, 0.05]:
            starts.append(parent + sd * rng.standard_normal(n))
        for _ in range(5):
            starts.append(rng.random(n))
    else:
        for _ in range(8):
            starts.append(rng.random(n))
        for A, m, pv in [(0.25, 1, 0.37), (0.35, 2, 0.74), (0.45, 3, 1.11)]:
            starts.append(0.5 + A * np.cos(2.0 * np.pi * m * np.arange(n) / n + pv))

    for s in starts:
        hs = project_box_sum(np.asarray(s, dtype=np.float64), n * 0.5)
        if not check_feasible(hs, n):
            continue
        res = slsqp_optimize(hs)
        if res is None:
            continue
        hr = project_box_sum(res, n * 0.5)
        if not check_feasible(hr, n):
            continue
        sc = exact_score(hr)
        if sc < best_score - 1e-14:
            best_score = sc
            best_h = hr

    sigma = 0.05
    for _ in range(5):
        pert = best_h + sigma * rng.standard_normal(n)
        hp = project_box_sum(pert, n * 0.5)
        if not check_feasible(hp, n):
            break
        res = slsqp_optimize(hp)
        improved = False
        if res is not None:
            hr = project_box_sum(res, n * 0.5)
            if check_feasible(hr, n):
                sc = exact_score(hr)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_h = hr
                    improved = True
        if improved:
            sigma = max(0.01, 0.7 * sigma)
        else:
            sigma = min(0.25, 1.5 * sigma)

    final_h = project_box_sum(best_h, n * 0.5)
    if check_feasible(final_h, n):
        final_score = exact_score(final_h)
    elif parent is not None:
        final_h = parent
        final_score = exact_score(final_h)
    else:
        final_h = np.full(n, 0.5)
        final_score = exact_score(final_h)

    return (final_h, final_score, n)
