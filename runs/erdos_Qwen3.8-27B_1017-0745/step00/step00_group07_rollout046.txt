def build_upper_hull(bs, cs, idx_zero):
    # bs, cs are arrays of slopes and intercepts
    # idx_zero is a boolean mask of which lags are affine (a_d == 0)
    # Returns (sorted_lines, hull_breaks, hull_idx) where
    # hull_idx[k] is the index into sorted_lines of the k-th active line
    # hull_breaks[k] is the t-value where line k becomes active
    mask = idx_zero
    lines = list(zip(bs[mask], cs[mask]))
    if not lines:
        return None, None, None
    # Sort by slope
    lines.sort()
    # Remove duplicate slopes, keep max intercept
    unique = []
    i = 0
    while i < len(lines):
        mx = lines[i][1]
        j = i + 1
        while j < len(lines) and lines[j][0] == lines[i][0]:
            mx = max(mx, lines[j][1])
            j += 1
        unique.append((lines[i][0], mx))
        i = j
    # Build upper hull
