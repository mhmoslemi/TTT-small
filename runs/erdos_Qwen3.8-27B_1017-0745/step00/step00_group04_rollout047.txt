def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(corr * dx))
    return val if math.isfinite(val) else float('inf')

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    # zero lag
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0, :] = dx * (1.0 - 2.0 * h)
    # nonzero lags
    for d in range(1, n):
        cnt = n - d
        hp = h[d:cnt+d]  # h[d], h[d+1], ..., h[n-1]
        hl = h[0:cnt]    # h[0], h[1], ..., h[n-d-1]
        q[2*d-1] = dx * np.sum(hp * (1.0 - hl))
        q[2*d] = dx * np.sum(hl * (1.0 - hp))
        G[2*d-1, d:cnt+d] += dx * (1.0 - hl)
        G[2*d-1, 0:cnt] += -dx * hp
        G[2*d, 0:cnt] += dx * (1.0 - hp)
        G[2*d, d:cnt+d] += -dx * hl
    if not (np.isfinite(q).all() and np.isfinite(G).all()):
        return None, None
    return q, G

        r=_rest(rng.random(n),n)
        if r is not None: ss.append(r)
    fin=[]
    for s in ss:
        if _ok(s,n) and all(np.max(np.abs(s-f))>=1e-10 for f in fin):
            fin.append(s)
    return fin

def _pairs(n,rng):
    if n<=48: pr=[(i,j) for i in range(n) for j in range(i+1,n)]
    elif n<=64:
        ps=set()
        for r in(1,2,max(1,n//4),max(1,n//2)):
            for i in range(n):
                p=(i,(i+r)%n);ps.add((min(p),max(p)))
        pr=sorted(ps)
    else:
        ps=set()
        for r in(1,max(1,n//2)):
            for i in range(n):
                p=(i,(i+r)%n);ps.add((min(p),max(p)))
        pr=sorted(ps)
    return [pr[k] for k in rng.permutation(len(pr))]

def run(seed=42,**kwargs):
    n=42;parent=None
    try:
        iv=np.asarray(initial_h_values,dtype=np.float64)
        if iv.ndim==1 and len(iv)>=2:
            n=len(iv)
            if np.isfinite(iv).all():
                S=n/2.0
                if np.all(iv>=-1e-12) and np.all(iv<=1.0+1e-12) and abs(float(np.sum(iv))-S)<=1e-10 and abs(math.fsum(float(x) for x in iv)-S)<=1e-10:
                    parent=iv.copy()
                else:
                    p=_rest(iv,n)
                    if p is not None: parent=p
    except Exception: n=42;parent=None
    try:
        nk=kwargs.get('n_points',None)
        if nk is not None and not isinstance(nk,bool) and int(nk)==nk and 2<=int(nk)<=512 and parent is None:
            n=int(nk)
    except Exception: pass
    if parent is not None: n=len(parent)
    try: s=int(seed)
    except Exception: s=42
    rng=np.random.default_rng(s)
    const_h=np.full(n,0.5);const_sc=exact_score(const_h)
    inc=const_h.copy();inc_sc=const_sc
    pfb=None;pfb_sc=float('inf')
    if parent is not None:
        ps=exact_score(parent)
        if math.isfinite(ps) and _ok(parent,n):
            pfb=parent.copy();pfb_sc=ps
            if ps<inc_sc-1e-12: inc=parent.copy();inc_sc=ps
    if n<=128:
        ns=2 if n<=64 else 1;mx=8 if n<=48 else 4
        for st in _starts(n,rng,parent)[:mx]:
            if not _ok(st,n): continue
            ch=st.copy();cs=exact_score(ch)
            if not math.isfinite(cs): continue
            for _ in range(ns):
                prs=_pairs(n,rng);imp=False
                for(i,j) in prs:
                    try: res=_psearch(ch,i,j,cs)
                    except Exception: res=None
                    if res is not None:
                        ns2=exact_score(res)
                        if math.isfinite(ns2) and ns2<cs-1e-12:
                            ch=res;cs=ns2;imp=True
                if not imp: break
            if cs<inc_sc-1e-12: inc=ch.copy();inc_sc=cs
    cands=[(inc_sc,inc)]
    if pfb is not None: cands.append((pfb_sc,pfb))
    cands.append((const_sc,const_h));cands.sort(key=itemgetter(0))
    bs,bh=cands[0]
    if not _ok(bh,n):
        if pfb is not None: bh,bs=pfb,pfb_sc
        else: bh,bs=const_h,const_sc
    try:
        fs=exact_score(bh)
        if not math.isfinite(fs): bh,bs=const_h,const_sc
        else: bs=fs
    except Exception: bh,bs=const_h,exact_score(const_h)
    return (bh.astype(np.float64),float(bs),int(n))
