    if abs(h.sum() - m) > 1e-10 * N or not np.all(np.isfinite(h)):
        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)

def lag_coeffs(h, d, N):
    M = 2 * N - 1
    A = np.zeros(M); B = np.zeros(M); C = np.zeros(M)
    for k in range(-(N - 1), N):
        idx = k + N - 1
        if k >= 0:
            L = N - k; a = h[k:k+L]; b = h[:L]; u = d[k:k+L]; v = d[:L]
        else:
            s = -k; L = N - s; a = h[:L]; b = h[s:s+L]; u = d[:L]; v = d[s:s+L]
        A[idx] = np.sum(a * (1.0 - b))
        B[idx] = np.sum(u * (1.0 - b) - a * v)
        C[idx] = -np.sum(u * v)
    return A, B, C

def line_search(h, d, N, cur_score):
    A, B, C = lag_coeffs(h, d, N)
    if not (np.all(np.isfinite(A)) and np.all(np.isfinite(B)) and np.all(np.isfinite(C))):
        return h.copy(), cur_score
    M = 2 * N - 1
    a_lo, a_hi = 0.0, 0.0
    for i in range(N):
        if abs(d[i]) < 1e-300:
            continue
        r0 = -h[i] / d[i]; r1 = (1.0 - h[i]) / d[i]
        a_lo = max(a_lo, min(r0, r1)); a_hi = min(a_hi, max(r0, r1))
    if a_hi - a_lo < 1e-14:
        return h.copy(), cur_score
    cands = [a_lo, a_hi, 0.0]
    for k in range(M):
        if abs(C[k]) > 1e-300:
            sp = -B[k] / (2.0 * C[k])
            if a_lo - 1e-10 <= sp <= a_hi + 1e-10:
                cands.append(sp)
    for k in range(M):
        for l in range(k + 1, M):
            u = C[k] - C[l]; v = B[k] - B[l]; w = A[k] - A[l]
            sc = max(abs(u), abs(v), abs(w), 1e-300)
            if abs(u) <= 1e-12 * sc:
                if abs(v) > 1e-300:
                    cands.append(-w / v)
            else:
                D = v * v - 4.0 * u * w
                if D < -1e-12 * (v * v + abs(4.0 * u * w) + 1e-300):
                    continue
                D = max(D, 0.0)
                q = -0.5 * (v + copysign(sqrt(D), v))
                if abs(q) > 1e-300:
                    cands.append(q / u); cands.append(w / q)
                else:
                    r = -v / (2.0 * u); cands.append(r)
    cands = [a for a in cands if np.isfinite(a) and a_lo - 1e-10 <= a <= a_hi + 1e-10]
    cands = sorted(set(round(a, 13) for a in cands))
    if not cands:
        return h.copy(), cur_score
    n_c = len(cands)
    env = np.empty(n_c)
    for s in range(0, n_c, 512):
        e2 = min(s + 512, n_c)
        al = np.array(cands[s:e2])
        env[s:e2] = (A[None, :] + B[None, :] * al[:, None] + C[None, :] * al[:, None] ** 2).max(axis=1)
    order = np.argsort(env)
    ca = np.array(cands)
    extras = [int(np.argmin(np.abs(ca - a_lo))), int(np.argmin(np.abs(ca - a_hi))), int(np.argmin(ca))]
    sel = sorted(set(list(order[:16]) + extras))
    best_h, best_s = h.copy(), cur_score
    for i in sel:
        a = max(a_lo, min(a_hi, cands[i]))
        cand = project_box_sum(h + a * d, N)
        s_val = score_exact(cand, N)
        if s_val < best_s - 1e-15:
            best_s, best_h = s_val, cand
    return best_h, best_s

def make_directions(h, N, rng, main):
    dlist = []
    ns, nr, nb, nh = (8, 6, 4, 2) if main else (4, 3, 2, 1)
    anchor = int(np.argmax(np.minimum(h, 1.0 - h)))
    others = [i for i in range(N) if i != anchor]
    for j in range(min(ns, len(others))):
        idx = others[j * len(others) // ns]
        d = np.zeros(N); d[idx] = 1.0; d[anchor] = -1.0; dlist.append(d)
    used = set()
    for _ in range(nr):
        for _t in range(40):
            ii = int(rng.integers(N)); jj = int(rng.integers(N))
            key = (min(ii, jj), max(ii, jj))
            if ii != jj and key not in used:
                used.add(key)
                d = np.zeros(N); d[ii] = 1.0; d[jj] = -1.0; dlist.append(d); break
    for _ in range(nb):
        bs = int(rng.integers(3, min(12, N) + 1))
        supp = rng.choice(N, bs, replace=False)
        vals = rng.standard_normal(bs); vals -= vals.mean(); vals[0] -= vals.sum()
        if np.max(np.abs(vals)) > 1e-300:
            d = np.zeros(N); d[supp] = vals / np.max(np.abs(vals)); dlist.append(d)
    for r in range(1, nh + 1):
        if 2 * r >= N:
            break
        ph = rng.uniform(0.0, 2.0 * np.pi)
        idx = np.arange(N, dtype=np.float64)
        d = np.cos(2.0 * np.pi * r * idx / N + ph); d -= d.mean()
        if np.max(np.abs(d)) > 1e-300:
            d /= np.max(np.abs(d)); dlist.append(d)
    return dlist

def gen_starts(N, rng, baseline):
    starts = [baseline.copy(), np.full(N, 0.5)]
    starts.append(project_box_sum(1.0 - baseline, N))
    starts.append(project_box_sum(rng.uniform(0.0, 1.0, N), N))
    starts.append(project_box_sum(baseline + 0.2 * rng.standard_normal(N), N))
    i = np.arange(N, dtype=np.float64)
    for _ in range(2):
        z = np.full(N, 0.5)
        for r in range(1, min(5, (N - 1) // 2) + 1):
            z += (0.12 / r) * (rng.standard_normal() * np.sin(2 * np.pi * r * i / N) + rng.standard_normal() * np.cos(2 * np.pi * r * i / N))
        starts.append(project_box_sum(z, N))
    deduped = [starts[0]]
    for s in starts[1:]:
        ok = True
        for d in deduped:
            if np.max(np.abs(s - d)) < 1e-8:
                ok = False; break
        if ok:
            deduped.append(s)
    return deduped[:7]

def do_pass(h, N, rng, main):
    cur_s = score_exact(h, N)
    dirs = make_directions(h, N, rng, main)
    for d in dirs:
        try:
            h, cur_s = line_search(h, d, N, cur_s)
        except Exception:
            pass
    return h, cur_s

def run(seed=42, **kwargs):
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    rng = np.random.default_rng(seed)
    N = 70
    baseline = np.full(N, 0.5)
    try:
        p = initial_h_values
        if p is not None:
            p = np.asarray(p, dtype=np.float64).ravel()
            if p.size > 0 and np.all(np.isfinite(p)):
                N = p.size
                baseline = project_box_sum(p, N)
    except Exception:
        pass
    best_h = baseline.copy()
    best_s = score_exact(best_h, N)
    starts = gen_starts(N, rng, best_h)
    all_ends = []
    for st in starts:
        h_e, s_e = do_pass(st, N, rng, True)
        all_ends.append((h_e, s_e))
    scs = np.array([e[1] for e in all_ends])
    ord_ = np.argsort(scs)
    n_elite = min(3, len(all_ends))
    for jj in range(n_elite):
        oi = int(ord_[jj])
        h_e, s_e = all_ends[oi]
        if s_e < best_s:
            best_h, best_s = h_e.copy(), s_e
    for jj in range(n_elite):
        oi = int(ord_[jj])
        h_r, s_r = do_pass(all_ends[oi][0], N, rng, False)
        if s_r < best_s:
            best_h, best_s = h_r.copy(), s_r
    for scale in [0.08, 0.20]:
        pert = project_box_sum(best_h + scale * rng.standard_normal(N), N)
        s_p = score_exact(pert, N)
        if s_p < best_s:
            best_h, best_s = pert.copy(), s_p
        h_p, s_p = do_pass(pert, N, rng, False)
        if s_p < best_s:
            best_h, best_s = h_p.copy(), s_p
    best_h = project_box_sum(best_h, N)
    best_s = score_exact(best_h, N)
    if not np.isfinite(best_s):
        best_h = np.full(N, 0.5)
        best_s = score_exact(best_h, N)
    return best_h, best_s, N
