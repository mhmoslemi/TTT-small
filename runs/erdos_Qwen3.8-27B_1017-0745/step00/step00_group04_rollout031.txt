        if len(pop) >= P:
            break
        z = rng.standard_normal(n)
        z -= np.mean(z)
        p = project_capped_simplex(base0 + sigmas[i % 4] * z, n)
        if p is not None:
            c = add_unique(pop, p, n)
            if c is not None:
                pop.append(c)
    for _ in range(20 * P):
        if len(pop) >= P:
            break
        p = project_capped_simplex(rng.random(n), n)
        if p is not None:
            c = add_unique(pop, p, n)
            if c is not None:
                pop.append(c)
    while len(pop) < P:
        pop.append(pop[0].copy())
    return np.array(pop[:P], dtype=np.float64)

def de_search(X, fitness, n, rng, max_gens):
    P, dim = X.shape
    F = rng.uniform(0.5, 0.8, size=P)
    CR = np.full(P, 0.8)
    inc_score = float(np.min(fitness))
    inc_h = X[int(np.argmin(fitness))].copy()
    stagnant = 0
    restarts = 0
    rsig = [0.20, 0.10, 0.05]
    for gen in range(max_gens):
        Xn = X.copy()
        Fn = F.copy()
        CRn = CR.copy()
        ft = fitness.copy()
        gen_imp = False
        for i in range(P):
            Ft, CRt = F[i], CR[i]
            if rng.random() < 0.1:
                Ft = 0.1 + 0.8 * rng.random()
            if rng.random() < 0.1:
                CRt = rng.random()
            if rng.random() < 0.75:
                cands = [j for j in range(P) if j != i]
                r1, r2, r3 = rng.choice(cands, size=3, replace=False)
                mutant = X[r1] + Ft * (X[r2] - X[r3])
            else:
                nbest = max(1, int(math.ceil(0.2 * P)))
                order = np.argsort(fitness)
                pc = [int(order[j]) for j in range(nbest) if int(order[j]) != i]
                if not pc:
                    pc = [int(order[0])]
                p = int(pc[int(rng.integers(len(pc)))])
                rc = [j for j in range(P) if j != i and j != p]
                r1, r2 = rng.choice(rc, size=2, replace=False)
                mutant = X[i] + Ft * (X[p] - X[i]) + Ft * (X[r1] - X[r2])
            trial = X[i].copy()
            mask = rng.random(n) < CRt
            mask[int(rng.integers(n))] = True
            trial[mask] = mutant[mask]
            proj = project_capped_simplex(trial, n)
            if proj is None or not is_feasible(proj, n):
                continue
            if np.max(np.abs(proj - X[i])) < 1e-14:
                continue
            sc = exact_score(proj, n)
            if sc is None:
                continue
            if sc < fitness[i] - 1e-12:
                Xn[i], ft[i] = proj, sc
                Fn[i], CRn[i] = Ft, CRt
            elif abs(sc - fitness[i]) <= 1e-12 and np.max(np.abs(proj - X[i])) >= 1e-6:
                if rng.random() < 0.02:
                    Xn[i], ft[i] = proj, sc
                    Fn[i], CRn[i] = Ft, CRt
            if sc < inc_score - 1e-12:
                inc_score, inc_h = sc, proj.copy()
                gen_imp = True
        X, fitness, F, CR = Xn, ft, Fn, CRn
        stagnant = 0 if gen_imp else stagnant + 1
        if stagnant >= 30 and restarts < 3:
            restarts += 1
            stagnant = 0
            kcnt = max(2, P // 4) + 1
            order = np.argsort(fitness)
            keep_set = set(int(order[j]) for j in range(kcnt))
            sg = rsig[restarts - 1]
            for j in range(P):
                if j in keep_set:
                    continue
                base = inc_h if rng.random() < 0.75 else np.full(n, 0.5)
                z = rng.standard_normal(n)
                z -= np.mean(z)
                p = project_capped_simplex(base + sg * z, n)
                if p is not None:
                    X[j] = p
                    sc = exact_score(p, n)
                    fitness[j] = sc if sc is not None else 1.0
            F = rng.uniform(0.5, 0.8, size=P)
            CR = np.full(P, 0.8)
        elif stagnant >= 70 and restarts >= 3:
            break
    return inc_h, inc_score

def run(seed=42, **kwargs):
    n = 42
    parent_seed = None
    pr = globals().get('initial_h_values', None)
    if pr is not None:
        try:
            pa = np.array(pr, dtype=np.float64).flatten()
            if pa.ndim == 1 and pa.size >= 2:
                n = pa.size
                if np.all(np.isfinite(pa)):
                    S = n / 2.0
                    if np.all(pa >= -1e-15) and np.all(pa <= 1.0 + 1e-15):
                        if abs(np.sum(pa) - S) <= 1e-12 and abs(math.fsum(float(x) for x in pa) - S) <= 1e-12:
                            parent_seed = pa.copy()
                        else:
                            parent_seed = project_capped_simplex(np.clip(pa, 0, 1), n)
                    else:
                        parent_seed = project_capped_simplex(np.clip(pa, 0, 1), n)
        except Exception:
            pass
    if parent_seed is None:
        if 'n_points' in kwargs:
            try:
                nv = kwargs['n_points']
                if not isinstance(nv, bool) and float(nv) == int(nv) and 2 <= int(nv) <= 512:
                    n = int(nv)
            except Exception:
                pass
    try:
        s = int(seed)
    except Exception:
        s = 42
    rng = np.random.default_rng(s)
    const_h = np.full(n, 0.5, dtype=np.float64)
    cs = exact_score(const_h, n)
    par_h, par_score = None, None
    if parent_seed is not None and is_feasible(parent_seed, n):
        par_score = exact_score(parent_seed, n)
        if par_score is not None:
            par_h = parent_seed.copy()
    inc_h, inc_score = const_h.copy(), cs
    if par_score is not None and par_score < inc_score - 1e-12:
        inc_h, inc_score = par_h.copy(), par_score
    if n > 128:
        cands = [const_h]
        if par_h is not None:
            cands.append(par_h)
        idx = np.arange(n)
        sh = project_capped_simplex(0.5 + 0.45 * np.sin(2.0 * np.pi * idx / n), n)
        if sh is not None:
            cands.append(sh)
        for c in cands:
            if c is not None and is_feasible(c, n):
                sc = exact_score(c, n)
                if sc is not None and sc < inc_score - 1e-12:
                    inc_h, inc_score = c.copy(), sc
        return inc_h, float(inc_score), int(n)
    try:
        X = build_population(n, par_h, rng)
        ft = np.array([exact_score(X[i], n) if exact_score(X[i], n) is not None else 1.0 for i in range(X.shape[0])])
        mg = 180 if n <= 64 else 100
        ih2, is2 = de_search(X, ft, n, rng, mg)
        if is2 < inc_score - 1e-12:
            inc_h, inc_score = ih2.copy(), is2
    except Exception:
        pass
    if par_h is not None:
        ps = exact_score(par_h, n)
        if ps is not None and ps < inc_score - 1e-12:
            inc_h, inc_score = par_h.copy(), ps
    cfs = exact_score(const_h, n)
    if cfs is not None and cfs < inc_score - 1e-12:
        inc_h, inc_score = const_h.copy(), cfs
    if not is_feasible(inc_h, n):
        fp = project_capped_simplex(inc_h, n)
        if fp is not None:
            inc_h = fp
    fs = exact_score(inc_h, n)
    if fs is None:
        inc_h = const_h.copy()
        fs = exact_score(const_h, n)
    if fs is None:
        fs = 1.0
    return inc_h.astype(np.float64), float(fs), int(n)
