                baseline = np.clip(a, 0.0, 1.0)
                if abs(np.sum(baseline) - N / 2.0) > 1e-8 * max(1, N):
                    baseline = project_box_sum(baseline, N)
            else:
                baseline = project_box_sum(a, N)
    except NameError:
        pass
    except Exception:
        pass
    if N is None:
        try:
            nv = float(kwargs.get('n_points', 70))
            nv_i = int(nv)
            if abs(nv - nv_i) < 1e-9 and 1 <= nv_i <= 999:
                N = nv_i
        except Exception:
            N = 70
        baseline = np.full(N, 0.5)
    if N == 1:
        return baseline.copy(), score_exact(baseline, 1), 1
    inc_h = baseline.copy()
    inc_score = score_exact(inc_h, N)
    if not np.isfinite(inc_score):
        inc_h = project_box_sum(inc_h, N)
        inc_score = score_exact(inc_h, N)
    if N <= 100:
        n_starts, iters = 8, 60
    elif N <= 200:
        n_starts, iters = 6, 40
    elif N <= 500:
        n_starts, iters = 4, 20
    else:
        n_starts, iters = 3, 8
    rng = np.random.default_rng(seed)
    starts = build_starts(inc_h, N, rng)
    for si in range(min(n_starts, len(starts))):
        h0 = project_box_sum(starts[si], N)
        s0 = score_exact(h0, N)
        if np.isfinite(s0) and s0 < inc_score:
            inc_score = s0
            inc_h[:] = h0
        try:
            inc_h, inc_score = optimize_start(h0, N, inc_h, inc_score, iters)
        except Exception:
            pass
    h_final = project_box_sum(inc_h, N)
    s_final = score_exact(h_final, N)
    if not np.isfinite(s_final):
        h_final = np.full(N, 0.5)
        s_final = score_exact(h_final, N)
    return h_final, float(s_final), N
