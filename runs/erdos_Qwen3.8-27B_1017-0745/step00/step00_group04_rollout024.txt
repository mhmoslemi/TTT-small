import numpy as np
import math

def is_feasible(h, n):
    try:
        h = np.asarray(h, dtype=np.float64)
    except Exception:
        return False
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return False
    S = n / 2.0
    return abs(np.sum(h) - S) <= 1e-10 and abs(math.fsum(float(x) for x in h) - S) <= 1e-10

def project_capped_simplex(v, n):
    v = np.asarray(v, dtype=np.float64)
    if v.shape != (n,) or not np.all(np.isfinite(v)):
        return None
    v = np.clip(v, -1e6, 1e6)
    S = n / 2.0
    lo, hi = float(v.min()) - 1.0, float(v.max())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0, 1)) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0, 1)
    for _ in range(3):
        r = S - math.fsum(float(x) for x in h)
        if abs(r) <= 1e-12:
            break
        for i in range(n):
            if abs(r) < 1e-12:
                break
            if r > 0:
                d = min(1.0 - h[i], r)
                h[i] += d
                r -= d
            else:
                d = min(h[i], -r)
                h[i] -= d
                r += d
    if not is_feasible(h, n):
        return None
    return h

def exact_score(h, n):
