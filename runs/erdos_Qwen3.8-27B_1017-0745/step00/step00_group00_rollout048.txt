def is_feas(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and np.all(h >= -1e-12)
            and np.all(h <= 1.0 + 1e-12) and abs(float(h.sum()) - n * 0.5) < 1e-9 * n)

def esc(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * (2.0 / n), c

def qroots(a, b, c):
    if abs(a) < 1e-16:
        if abs(b) < 1e-16:
            return []
        return [-c / b]
    dsc = b * b - 4.0 * a * c
    if dsc < 0.0:
        t = 64.0 * 2.22e-16 * max(1.0, b * b, abs(4.0 * a * c))
        if dsc > -t:
            dsc = 0.0
        else:
            return []
    sd = math.sqrt(dsc)
    q = -0.5 * (b + math.copysign(sd, b))
    if abs(q) > 1e-30:
        r = [q / a, c / q]
    else:
        r = [-b / (2.0 * a)]
    return [x for x in r if math.isfinite(x)]

def pair_coeffs(h, corr, a, b, n):
    N = 2 * n - 1
    A = corr.copy()
    B = np.zeros(N)
    D = np.zeros(N)
    d = b - a
    for k in range(-(n - 1), n):
        bk = 0.0
        if 0 <= a - k < n:
            bk += 1.0 - h[a - k]
        if 0 <= b - k < n:
            bk -= 1.0 - h[b - k]
        if 0 <= a + k < n:
            bk -= h[a + k]
        if 0 <= b + k < n:
            bk += h[b + k]
        B[k + n - 1] = bk
        if k == 0:
            D[k + n - 1] = -2.0
        elif k == -d or k == d:
            D[k + n - 1] = 1.0
    if not np.all(np.isfinite(B)):
        return None
    return A, B, D

def hull_segs(A, B, cur, lo, hi):
    N = len(A)
    nc = [(i, A[i], B[i]) for i in range(N) if i not in cur]
    if not nc:
        return [(lo, hi, None)]
    nc.sort(key=_kline)
    dd = [nc[0]]
    for i in range(1, len(nc)):
        if nc[i][2] != dd[-1][2]:
            dd.append(nc[i])
        elif nc[i][1] > dd[-1][1]:
            dd[-1] = (nc[i][0], nc[i][1], nc[i][2])
    hl = [(dd[0][2], dd[0][1])]
    hx = [-1e18]
    for i in range(1, len(dd)):
        m, c = dd[i][2], dd[i][1]
        xs = -1e18
        while hl:
            m0, c0 = hl[-1]
            xc = (c0 - c) / (m - m0)
            if xc <= hx[-1] + 1e-15:
                hl.pop()
                hx.pop()
            else:
                xs = xc
                break
        hl.append((m, c))
        hx.append(xs)
    segs = []
    for j in range(len(hl)):
        s = max(hx[j], lo)
        if s < hi - 1e-14:
            segs.append((s, hi, (hl[j][0], hl[j][1])))
    return segs if segs else [(lo, hi, None)]

def lmin_pair(h, corr, sc, a, b, n):
    lo = max(-h[a], h[b] - 1.0)
    hi = min(1.0 - h[a], h[b])
    if hi - lo < 1e-14:
        return h, corr, sc
    cf = pair_coeffs(h, corr, a, b, n)
    if cf is None:
        return h, corr, sc
    A, B, D = cf
    d = b - a
    ci = [n - 1, n - 1 - d, n - 1 + d]
    cur = set(ci)
    segs = hull_segs(A, B, cur, lo, hi)
    cands = set()
    for s, e, ln in segs:
        cands.add(s)
        cands.add(e)
        for j in ci:
            if D[j] != 0.0:
                st = -B[j] / (2.0 * D[j])
                if s - 1e-12 <= st <= e + 1e-12:
                    cands.add(st)
        for i in range(3):
            for j in range(i + 1, 3):
                for r in qroots(D[ci[i]] - D[ci[j]], B[ci[i]] - B[ci[j]], A[ci[i]] - A[ci[j]]):
                    if s - 1e-12 <= r <= e + 1e-12:
                        cands.add(r)
            if ln is not None:
                for r in qroots(D[ci[i]], B[ci[i]] - ln[0], A[ci[i]] - ln[1]):
                    if s - 1e-12 <= r <= e + 1e-12:
                        cands.add(r)
    if lo <= 0.0 <= hi:
        cands.add(0.0)
    scored = []
    for al in cands:
        al = max(lo, min(hi, al))
        if abs(al) < 1e-14:
            continue
        obj = -1e30
        for s, e, ln in segs:
            if s - 1e-12 <= al <= e + 1e-12:
                if ln is not None:
                    obj = max(obj, ln[0] * al + ln[1])
                break
        for j in ci:
            obj = max(obj, A[j] + B[j] * al + D[j] * al * al)
        scored.append((obj, al))
    if not scored:
        return h, corr, sc
    scored.sort(key=_k0)
    bh, bc, bs = h, corr, sc
    for _, al in scored[:5]:
        h2 = h.copy()
        h2[a] = max(0.0, min(1.0, h2[a] + al))
        h2[b] = max(0.0, min(1.0, h2[b] - al))
        if not is_feas(h2, n):
            continue
        s2, c2 = esc(h2)
        if math.isfinite(s2) and s2 < bs - 1e-13:
            bh, bc, bs = h2, c2, s2
    return bh, bc, bs

def make_starts(n, parent, rng):
    S = n * 0.5
    starts = []
    if parent is not None:
        p = np.clip(parent.astype(np.float64), 0.0, 1.0)
        pv = project_box_sum(p, S)
        if pv is not None:
            starts.append(pv)
            starts.append(pv[::-1].copy())
        for sd in [0.05, 0.18]:
            v = project_box_sum(pv + rng.normal(0, sd, n), S)
            if v is not None:
                starts.append(v)
        for _ in range(2):
            v = project_box_sum(rng.uniform(0, 1, n), S)
            if v is not None:
                starts.append(v)
        v = project_box_sum(0.5 + 0.45 * np.cos(2.0 * math.pi * np.arange(n, dtype=np.float64) / n + 0.37), S)
        if v is not None:
            starts.append(v)
        v = np.zeros(n)
        v[:n // 2] = 1.0
        if n % 2 == 1:
            v[n // 2] = 0.5
        starts.append(v)
    else:
        starts.append(np.full(n, 0.5))
        v = np.zeros(n)
        v[:n // 2] = 1.0
        if n % 2 == 1:
            v[n // 2] = 0.5
        starts.append(v)
        v = project_box_sum(np.where(np.arange(n) % 2 == 0, 1.0, 0.0), S)
        if v is not None:
            starts.append(v)
        for (m, ph) in [(1, 0.37), (2, 0.71), (3, 1.13)]:
            v = project_box_sum(0.5 + 0.45 * np.cos(2.0 * math.pi * m * np.arange(n, dtype=np.float64) / n + ph), S)
            if v is not None:
                starts.append(v)
        for _ in range(2):
            v = project_box_sum(rng.uniform(0, 1, n), S)
            if v is not None:
                starts.append(v)
    uniq = []
    for s in starts:
        if not is_feas(s, n):
            continue
        dup = any(np.max(np.abs(s - u)) <= 1e-12 for u in uniq)
        if not dup:
            uniq.append(s)
    return uniq

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    g = globals()
    parent = g.get('initial_h_values', None)
    n = 45
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.ndim == 1 and 2 <= len(p) <= 999 and np.all(np.isfinite(p)) and np.all(p >= -1e-6) and np.all(p <= 1.0 + 1e-6):
            n = len(p)
        else:
            parent = None
    fb = project_box_sum(np.full(n, 0.5), n * 0.5)
    if fb is None:
        fb = np.full(n, 0.5)
    fb_sc, _ = esc(fb)
    best_h = fb.copy()
    best_sc = fb_sc
    budget = 150000
    starts = make_starts(n, parent, rng)
    n_sw = 3 if n <= 100 else (2 if n <= 300 else 1)
    pair_cap = min(n * (n - 1) // 2, max(1, 30000000 // (5 * n * n)))
    for st in starts:
        budget -= 1
        if budget <= 0:
            break
        sc, cr = esc(st)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_h = st.copy()
        h_loc = st.copy()
        c_loc = cr.copy()
        s_loc = sc
        for sw in range(n_sw):
            all_pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
            if len(all_pairs) > pair_cap:
                idx = rng.choice(len(all_pairs), size=pair_cap, replace=False)
                prs = [all_pairs[k] for k in idx]
            else:
                prs = all_pairs[:]
            rng.shuffle(prs)
            imp = False
            for a, b in prs:
                if budget < 7:
                    break
                sb = s_loc
                h_loc, c_loc, s_loc = lmin_pair(h_loc, c_loc, s_loc, a, b, n)
                budget -= 5
                if s_loc < sb - 1e-15:
                    imp = True
                    if s_loc < best_sc - 1e-13:
                        best_sc = s_loc
                        best_h = h_loc.copy()
            if not imp:
                break
            if abs(float(h_loc.sum()) - n * 0.5) > 1e-12 * n:
                hr = project_box_sum(h_loc, n * 0.5)
                if hr is not None:
                    h_loc = hr
                    s_loc, c_loc = esc(h_loc)
                    budget -= 1
        if s_loc < best_sc - 1e-13:
            best_sc = s_loc
            best_h = h_loc.copy()
    hf = project_box_sum(best_h, n * 0.5)
    if hf is not None and is_feas(hf, n):
        best_h = hf
    sc_f, _ = esc(best_h)
    if not math.isfinite(sc_f) or sc_f > best_sc + 1e-6:
        best_h = np.full(n, 0.5)
        sc_f, _ = esc(best_h)
    return (best_h.astype(np.float64), float(sc_f), n)
