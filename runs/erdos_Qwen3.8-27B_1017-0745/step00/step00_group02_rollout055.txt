from scipy.optimize import minimize

def project_K(y, S):
    n = len(y)
    if not np.all(np.isfinite(y)):
        return np.full(n, S / n)
    lo, hi = np.min(y) - 1.0, np.max(y)
    for _ in range(200):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = S - np.sum(x)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def exact_score(h, n):
    if not np.all(np.isfinite(h)) or len(h) != n:
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_scores(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def smooth_val(h, n, tau):
    qs = lag_scores(h, n)
    m = np.max(qs)
    w = np.exp((qs - m) / tau)
    return m + tau * np.log(np.sum(w))

def smooth_grad(h, n, tau):
    qs = lag_scores(h, n)
    m = np.max(qs)
    w = np.exp((qs - m) / tau)
    p = w / np.sum(w)
    dx = 2.0 / n
    g = np.zeros(n)
    for idx in range(2 * n - 1):

        if pk < 1e-18:

        lo = max(0, -k)
        hi = min(n, n - k)
        a = h[lo:hi]
        b = h[lo + k:hi + k]
        g[lo + k:hi + k] += pk * dx * (1.0 - a)
        g[lo:hi] += pk * dx * (-b)
    return g

def pgd_optimize(h0, n, tau_sched, max_iter=80, max_ls=25):
    h = h0.copy()
    for tau in tau_sched:
        F = smooth_val(h, n, tau)
        alpha = 1.0
        stall = 0
        for it in range(max_iter):
            g = smooth_grad(h, n, tau)
            if not np.all(np.isfinite(g)):
                break
            y = project_K(h - alpha * g, n * 0.5)
            d = y - h
            if np.max(np.abs(d)) < 1e-10:
                break
            gd = np.dot(g, d)
            if gd > -1e-14:
                break
            Fy = smooth_val(y, n, tau)
            if Fy <= F + 1e-4 * gd + 1e-14:
                F = Fy
                h = y.copy()
                if F - stall < 1e-12:
                    stall += F
                alpha = min(1.25 * alpha, 5.0)
            else:
                alpha *= 0.5
                if it >= max_ls:
                    break
        if not np.all(np.isfinite(h)):
            break
    return h

def slsqp_obj(x, n, S):
    return x[n]

def slsqp_grad(x, n, S):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def slsqp_ineq(x, n, S):
    h = x[:n]
    t = x[n]
    qs = lag_scores(h, n)
    return t - qs

def slsqp_ineq_jac(x, n, S):
    h = x[:n]
    dx = 2.0 / n
    J = np.zeros((2 * n - 1, n + 1))
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        lo = max(0, -k)
        hi = min(n, n - k)
        a = h[lo:hi]
        b = h[lo + k:hi + k]
        J[idx, lo + k:hi + k] = -(1.0 - a) * dx
        J[idx, lo:hi] += (-b) * dx
        J[idx, n] = 1.0
    return J

def slsqp_eq(x, n, S):
    return np.sum(x[:n]) - S

def slsqp_eq_jac(x, n, S):
    J = np.zeros(n + 1)
    J[:n] = 1.0
    return J

def slsqp_refine(h0, n):
    S = n * 0.5
    x0 = np.append(h0, min(1.0, exact_score(h0, n) + 1e-8))
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'fun': lambda x, n=n, S=S: slsqp_ineq(x, n, S),
         'jac': lambda x, n=n, S=S: slsqp_ineq_jac(x, n, S), 'type': 'ineq'},
        {'fun': lambda x, n=n, S=S: slsqp_eq(x, n, S),
         'jac': lambda x, n=n, S=S: slsqp_eq_jac(x, n, S), 'type': 'eq'},
    ]
    res = minimize(slsqp_obj, x0, jac=lambda x, n=n, S=S: slsqp_grad(x, n, S),
