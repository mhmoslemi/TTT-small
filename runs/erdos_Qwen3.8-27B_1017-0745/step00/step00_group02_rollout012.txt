    if not check_feasible(final, n) or final_score > fb2 - 1e-12:
        final = fallback.copy()
        final_score = exact_score(final, n)
    if not check_feasible(final, n):
        final = np.full(n, 0.5, dtype=np.float64)
        final_score = exact_score(final, n)
    return final.copy(), float(final_score), int(n)
