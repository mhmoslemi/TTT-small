import numpy as np
import math

def exact_score(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = float(np.max(corr) * dx)
    return c if math.isfinite(c) else float('inf')

def is_feasible(h, n):
    if not isinstance(h, np.ndarray) or h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return False
    S = n / 2.0
    return abs(float(np.sum(h)) - S) <= 1e-10 and abs(math.fsum(float(x) for x in h) - S) <= 1e-10

def project_capped_simplex(v, n):
    S = n / 2.0
    v = np.asarray(v, dtype=np.float64)
    if v.shape != (n,) or not np.all(np.isfinite(v)):
        return None
    v = np.clip(v, -1e6, 1e6)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        hc = np.clip(v - mid, 0.0, 1.0)
        if float(np.sum(hc)) > S:
            lo = mid
        else:
            hi = mid
    hc = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    res = S - math.fsum(float(x) for x in hc)
    if abs(res) > 1e-14:
        cap = (1.0 - hc) if res > 0 else hc
        tc = float(np.sum(cap))
        if tc > 1e-15:
            hc = np.clip(hc + res * cap / tc, 0.0, 1.0)
    for _ in range(3):
        diff = S - math.fsum(float(x) for x in hc)
        if abs(diff) < 1e-14:
            break
        slack = (1.0 - hc) if diff > 0 else hc
        j = int(np.argmax(slack))
        if slack[j] >= abs(diff):
            hc[j] += diff
    return hc if is_feasible(hc, n) else None

def active_bundle_gradient(q, G, n):
    Cq = float(np.max(q))
    tol = max(1e-12, 1e-7 * max(1.0, abs(Cq)))
    active = np.where(q >= Cq - tol)[0]
    if len(active) > 12:
        active = active[np.argsort(q[active])[::-1][:12]]
    m = len(active)
    if m == 0:
        return None
    U = G[active, :].copy()
    U -= U.mean(axis=1, keepdims=True)
    if m == 1:
        return U[0] if np.all(np.isfinite(U[0])) else None
    K = 0.5 * (U @ U.T + (U @ U.T).T)
    alpha = np.ones(m) / m
    for _ in range(40):
        z = K @ alpha
        j = int(np.argmin(z))
        if float(alpha @ z) - float(z[j]) <= 1e-12:
            break
        v = np.eye(m)[j] - alpha
        vKv = float(v @ K @ v)
        if vKv <= 1e-24:
            break
        gamma = float(np.clip(-float(v @ z) / vKv, 0.0, 1.0))
        alpha = np.clip(alpha + gamma * v, 0.0, None)
        alpha /= alpha.sum()
    g = alpha @ U
    return g if np.all(np.isfinite(g)) else (U[np.argmax(q[active])] if np.all(np.isfinite(U[np.argmax(q[active])])) else None)
