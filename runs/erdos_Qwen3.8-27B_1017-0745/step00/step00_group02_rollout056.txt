import numpy as np
import math

def project_K(y, n, S):
    y = np.asarray(y, dtype=np.float64)
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = S - np.sum(x)
    if abs(diff) > 1e-13:
        if diff > 0:
            m = x < 1.0 - 1e-14
        else:
            m = x > 1e-14
        c = int(np.sum(m))
        if c > 0:
            x[m] = np.clip(x[m] + diff / c, 0.0, 1.0)
    if abs(np.sum(x) - S) > 1e-9:
        return np.full(n, 0.5)
    return x

def exact_score(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def check_feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            np.all(h >= -1e-10) and np.all(h <= 1.0 + 1e-10) and
            abs(np.sum(h) - S) < 1e-7)

def smooth_val(h, n, tau):
    dx = 2.0 / n
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(q))
    return m + tau * np.log(np.sum(np.exp((q - m) / tau)))

def smooth_grad(h, n, tau):
    dx = 2.0 / n
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(q))
    w = np.exp((q - m) / tau)
    p = w / np.sum(w)
    L = 2 * n - 1
    g = np.zeros(n)
    for k in range(L):
        j = k - (n - 1)
        if j >= 0:
            hi = n - j
            g[:hi] += p[k] * dx * (1.0 - h[j:j + hi])
            g[j:j + hi] -= p[k] * dx * h[:hi]
        else:
            mm = -j
            g[mm:] += p[k] * dx * (1.0 - h[:n - mm])
            g[:n - mm] -= p[k] * dx * h[mm:]
    return g

                break
            gd = float(np.dot(g, d))

def _s_obj_jac(z, n, S):

    dx = 2.0 / n
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    return z[n] - q

def _s_ineq_jac(z, n, S):
    h = z[:n]
    dx = 2.0 / n
    L = 2 * n - 1
    J = np.zeros((L, n + 1))
    for k in range(L):

            J[k, mm:] -= dx * (1.0 - h[:n - mm])
            J[k, :n - mm] += dx * h[mm:]
        J[k, n] = 1.0
    return J

def _s_eq(z, n, S):
    return np.sum(z[:n]) - S

def _s_eq_jac(z, n, S):

def slsqp_refine(h0, n, S):
    try:
        from scipy.optimize import minimize
    except ImportError:
        return None
    if n > 100:
        return None
    h0 = project_K(h0, n, S)
    t0 = min(1.0, exact_score(h0, n) + 1e-8)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * (n + 1)
    cons = [
        {'type': 'ineq', 'fun': _s_ineq, 'jac': _s_ineq_jac, 'args': (n, S)},
        {'type': 'eq', 'fun': _s_eq, 'jac': _s_eq_jac, 'args': (n, S)},
    ]
    try:
        res = minimize(_s_obj, z0, method='SLSQP', jac=_s_obj_jac,
                       bounds=bounds, constraints=cons, args=(n, S),
                       options={'maxiter': 300, 'ftol': 1e-10, 'disp': False})
        if not hasattr(res, 'x'):
            return None

            return None
        h = project_K(h, n, S)

            return None
        return h
