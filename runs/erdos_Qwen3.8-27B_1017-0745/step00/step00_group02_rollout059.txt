import numpy as np
import math

def check_feasible(h, n):
    if h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10:
        return False
    return True

def project_k(y, n):
    if not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    S = n / 2.0
    lo, hi = np.min(y) - 1.5, np.max(y) + 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-16:
            break
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    res = S - np.sum(h)
    if abs(res) > 1e-13:
        if res > 0:
            sl = np.where(h < 1.0 - 1e-14, 1.0 - h, 0.0)
            ts = np.sum(sl)
            if ts > 0:
                h += res * sl / ts
        else:
            po = np.where(h > 1e-14, h, 0.0)
            tp = np.sum(po)
            if tp > 0:
                h += res * po / tp
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(np.sum(h) - S) > 1e-10:
        return np.full(n, 0.5)
    return h

def exact_score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def smooth_value(h, tau):
    dx = 2.0 / len(h)
    qs = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(qs)
    w = np.exp((qs - m) / tau)
    return float(m + tau * math.log(np.sum(w)))

def smooth_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    nl = 2 * n - 1
    qs = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(qs)
    w = np.exp((qs - m) / tau)
    p = w / np.sum(w)
    grad = np.zeros(n)
    for li in range(nl):
        ell = li - (n - 1)
        lo = max(0, -ell)
        hi = min(n, n - ell)
        ai = np.arange(lo, hi)
        if ell >= 0:
            X, Y = ai + ell, ai
        else:
            X, Y = ai, ai + ell
        grad[X] += p[li] * dx * (1.0 - h[Y])
        grad[Y] -= p[li] * dx * h[X]
    return grad

def projected_continue(h, n, tau_list, max_iter=80):
    dx = 2.0 / n
    for tau in tau_list:
        f_cur = smooth_value(h, tau)
        accept_count = 0
        for it in range(max_iter):
            g = smooth_gradient(h, tau)
            if not np.all(np.isfinite(g)):
                return h, f_cur
            alpha = 1.0
            for _ in range(25):
                y = project_k(h - alpha * g, n)
                d = y - h
                if np.max(np.abs(d)) <= 1e-12:
                    break
                gd = np.dot(g, d)
                if gd > -1e-14:
                    break
                f_new = smooth_value(y, tau)
                if f_new <= f_cur + 1e-4 * gd + 1e-14:
                    if f_new < f_cur - 1e-12:
                        accept_count = 0
                    else:
                        accept_count += 1
                    h = y
                    f_cur = f_new
                    break
                alpha *= 0.5
            else:
                break
            if accept_count >= 10:
                break
        if not np.all(np.isfinite(h)):
            return None, f_cur
    return h, f_cur

def slsqp_obj(z, n):
    return z[-1]

def slsqp_obj_jac(z, n):
    j = np.zeros(len(z))
    j[-1] = 1.0
    return j

def slsqp_con(z, n, dx, S):
    h = z[:n]
    t = z[-1]
    qs = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - qs

def slsqp_con_jac(z, n, dx, S):
    h = z[:n]
    nl = 2 * n - 1
    J = np.zeros((nl, n + 1))
    for li in range(nl):
        ell = li - (n - 1)
        lo = max(0, -ell)
        hi = min(n, n - ell)
        ai = np.arange(lo, hi)
        if ell >= 0:
            X, Y = ai + ell, ai
        else:
            X, Y = ai, ai + ell
        J[li, Y] -= dx * h[X]
        J[li, X] += dx * (1.0 - h[Y])
        J[li, -1] = 1.0
    return J

def slsqp_eq(z, n, dx, S):
    return np.sum(z[:n]) - S

def slsqp_eq_jac(z, n, dx, S):
    j = np.zeros(n + 1)
    j[:n] = 1.0
    return j

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    raw = globals().get('initial_h_values', None)
    n = None
    parent = None
    if raw is not None:
        r = np.array(raw, dtype=np.float64).ravel()
        if r.ndim == 1 and 2 <= len(r) <= 999 and np.all(np.isfinite(r)) and np.max(np.abs(r)) <= 1e6:
            n = len(r)
            parent = r
    if n is None:
        n = int(kwargs.get('n_points', 45))
        if not (2 <= n <= 999):
            n = 45
    dx = 2.0 / n
    S = n / 2.0
    tau_big = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00001]
    if n > 100:
        tau_big = [0.02, 0.005, 0.001, 0.0001, 0.00001]
    if parent is not None:
        init = project_k(parent, n)
    else:
        init = np.full(n, 0.5)
    best_h = init.copy()
    best_score = exact_score(best_h)
    candidates = [init.copy()]

    def try_archive(h_cand):
        nonlocal best_h, best_score
        h_cand = np.clip(h_cand, 0.0, 1.0)
        if not check_feasible(h_cand, n):
            return
        sc = exact_score(h_cand)
        if sc < best_score - 1e-12:
            best_h = h_cand.copy()
            best_score = sc

    if parent is not None:
        starts = [project_k(parent, n),
                  project_k(1.0 - parent, n),
                  project_k(parent[::-1].copy(), n),
                  project_k(1.0 - parent[::-1].copy(), n),
                  project_k(parent + 0.08 * rng.standard_normal(n), n),
                  project_k(parent + 0.25 * rng.standard_normal(n), n),
                  project_k(rng.uniform(0, 1, n), n),
                  project_k(rng.beta(0.5, 0.5, n), n)]
    else:
        starts = [np.full(n, 0.5)]
        for _ in range(3):
            starts.append(project_k(rng.uniform(0, 1, n), n))
        for _ in range(3):
            starts.append(project_k(rng.beta(0.5, 0.5, n), n))
        starts.append(project_k(0.5 + 0.35 * rng.standard_normal(n), n))

    max_iter = 80 if n <= 100 else 20
    for st in starts:
        try_archive(st)
        h_out, _ = projected_continue(st.copy(), n, tau_big, max_iter)
        if h_out is not None:
            try_archive(h_out)
            candidates.append(h_out.copy())

    n_perturb = 4 if n <= 100 else 1
    sigmas = [0.04, 0.12, 0.30, 0.50][:n_perturb]
    for sig in sigmas:
        h_pert = project_k(best_h + sig * rng.standard_normal(n), n)
        try_archive(h_pert)
        h_out, _ = projected_continue(h_pert.copy(), n, tau_big, max_iter)
        if h_out is not None:
            try_archive(h_out)
            candidates.append(h_out.copy())

    if n <= 100:
        candidates.sort(key=lambda x: exact_score(x))
        refined = []
        for c in candidates:
            if all(np.max(np.abs(c - r)) > 1e-5 for r in refined):
                refined.append(c)
                if len(refined) >= 3:
                    break
        if not refined:
            refined = [best_h.copy()]
        try:
            from scipy.optimize import minimize as sp_min
            for r_h in refined:
                t0 = min(1.0, exact_score(r_h) + 1e-8)
                z0 = np.concatenate([r_h, [t0]])
                bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
                cons = [
                    {'type': 'ineq', 'fun': slsqp_con, 'jac': slsqp_con_jac, 'args': (n, dx, S)},
                    {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac, 'args': (n, dx, S)}
                ]
                res = sp_min(slsqp_obj, z0, args=(n,), jac=lambda z, nn: slsqp_obj_jac(z, nn),
                             method='SLSQP', bounds=bounds, constraints=cons,
                             options={'maxiter': 250, 'ftol': 1e-10, 'disp': False})
                if res is not None and hasattr(res, 'x') and np.all(np.isfinite(res.x)):
                    h_sol = project_k(res.x[:n], n)
                    try_archive(h_sol)
        except Exception:
            pass

    final_h = project_k(best_h, n)
    if not check_feasible(final_h, n):
        final_h = project_k(init, n)
    if not check_feasible(final_h, n):
        final_h = np.full(n, 0.5)
    final_score = exact_score(final_h)
    return (final_h, final_score, n)
