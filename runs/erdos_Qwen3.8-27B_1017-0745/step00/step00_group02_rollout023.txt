                disc = max(0.0, disc)
                sq = math.sqrt(disc)
                if B > 0:
                    qv = -0.5 * (B + sq)
                else:
                    qv = -0.5 * (B - sq)
                r1 = qv / A
                r2 = C / qv if abs(qv) > 1e-300 else 0.0
                for rt in (r1, r2):
                    if math.isfinite(rt) and tlo - 1e-12 <= rt <= thi + 1e-12:
                        cands.append(max(tlo, min(thi, rt)))
            elif abs(B) > 1e-14 * sc:
                rt = -C / B
                if math.isfinite(rt) and tlo - 1e-12 <= rt <= thi + 1e-12:
                    cands.append(max(tlo, min(thi, rt)))
    for r in range(M):
        if abs(c[r]) > 1e-300:
            tp = -b[r] / (2.0 * c[r])
            if math.isfinite(tp) and tlo <= tp <= thi:
                cands.append(tp)
    cands.sort()
    dd = [cands[0]]
    for v in cands[1:]:
        if abs(v - dd[-1]) > 1e-12 * max(1.0, abs(v)):
            dd.append(v)
    best_t, best_v = 0.0, float('inf')
    chunk = 256
    for i in range(0, len(dd), chunk):
        ts = np.array(dd[i:i + chunk])
        vals = q[None, :] + ts[:, None] * b[None, :] + ts[:, None] ** 2 * c[None, :]
        fms = np.max(vals, axis=1)
        li = int(np.argmin(fms))
        if fms[li] < best_v:
            best_v = float(fms[li])
            best_t = float(ts[li])
    return best_t, best_v

def _grid_line(q, b, c, tlo, thi):
    grid = np.unique(np.concatenate([np.linspace(tlo, thi, 17), [0.0]]))
    best_t, best_v = 0.0, float('inf')
    for t in grid:
        vals = q + t * b + t * t * c
        fm = float(np.max(vals))
        if fm < best_v:
            best_v = fm
            best_t = float(t)
    bi = int(np.argmin(np.abs(grid - best_t)))
    lo_i, hi_i = max(0, bi - 1), min(len(grid) - 1, bi + 1)
    fine = np.unique(np.concatenate([grid, np.linspace(grid[lo_i], grid[hi_i], 9)]))
    for t in fine:
        vals = q + t * b + t * t * c
        fm = float(np.max(vals))
        if fm < best_v:
            best_v = fm
            best_t = float(t)
    return best_t, best_v

def run(seed=42, **kwargs):
    n, parent = 45, None
    try:
        p = np.array(globals()['initial_h_values'], dtype=np.float64).ravel()
        if 2 <= len(p) <= 999:
            n = len(p)
            if np.all(np.isfinite(p)) and np.all(np.abs(p) <= 1e6):
                parent = p.copy()
    except Exception:
        pass
    if parent is None and 'n_points' in kwargs:
        try:
            nk = kwargs['n_points']
            if isinstance(nk, int) and not isinstance(nk, bool) and 2 <= nk <= 999:
                n = nk
        except Exception:
            pass
    dx, S = 2.0 / n, float(n) / 2.0
    if parent is not None and _feasible(parent, n, S):
        fb = parent.copy()
    elif parent is not None:
        pj = _project(parent, n, S)
        fb = pj if pj is not None else np.full(n, 0.5)
    else:
        fb = np.full(n, 0.5)
    fb_score = _score(fb, n)
    best_h, best_sc = fb.copy(), fb_score
    rng = np.random.default_rng(seed)
    max_starts = 4 if n <= 128 else (3 if n <= 256 else 2)
    starts = [fb.copy()]
    if parent is not None:
        for cand in (1.0 - parent, parent + 0.12 * rng.standard_normal(n), rng.beta(0.5, 0.5, n)):
            if len(starts) >= max_starts:
                break
            pj = _project(cand, n, S)
            if pj is not None:
                starts.append(pj)
    else:
        for cand in (rng.uniform(0, 1, n), rng.beta(0.5, 0.5, n)):
            if len(starts) >= max_starts:
                break
            pj = _project(cand, n, S)
            if pj is not None:
                starts.append(pj)
    dd = [starts[0]]
    for s in starts[1:]:
        if all(np.max(np.abs(s - d0)) > 1e-8 for d0 in dd):
            dd.append(s)
    starts = dd[:max_starts]
    tree = _tree(n)
    if n > 256 and len(tree) > 64:
        tree = [tree[int(k * (len(tree) - 1) / 63)] for k in range(64)]
    nsweep = 3 if n <= 128 else (2 if n <= 256 else 1)
    use_exact = n <= 48
    for st in starts:
        try:
            ch = st.copy()
            cs = _score(ch, n)
            if not math.isfinite(cs) or not _feasible(ch, n, S):
                continue
            if cs < best_sc - 1e-12:
                best_h, best_sc = ch.copy(), cs
            for _ in range(nsweep):
                imp = False
                for (_, a, m, b) in tree:
                    d = _direction(n, a, m, b)
                    tlo, thi = _feasible_interval(ch, d)
                    if not (math.isfinite(tlo) and math.isfinite(thi)) or thi - tlo <= 1e-12:
                        continue
                    q = dx * np.correlate(ch, 1.0 - ch, mode='full')
                    bc = dx * (np.correlate(d, 1.0 - ch, mode='full') - np.correlate(ch, d, mode='full'))
                    cc = -dx * np.correlate(d, d, mode='full')
                    if not (np.all(np.isfinite(q)) and np.all(np.isfinite(bc)) and np.all(np.isfinite(cc))):
                        continue
                    if use_exact:
                        tb, pv = _exact_line(q, bc, cc, tlo, thi)
                    else:
                        tb, pv = _grid_line(q, bc, cc, tlo, thi)
                    if pv >= cs - 1e-12:
                        continue
                    ht = _repair(ch + tb * d, n, S)
                    if ht is None:
                        continue
                    ns = _score(ht, n)
                    if not math.isfinite(ns) or ns >= cs - 1e-12:
                        continue
                    ch, cs = ht, ns
                    imp = True
                    if cs < best_sc - 1e-12:
                        best_h, best_sc = ch.copy(), cs
                if not imp:
                    break
        except Exception:
            continue
    fh = best_h.copy()
    if not _feasible(fh, n, S):
        pj = _project(fh, n, S)
        fh = pj if pj is not None else fb.copy()
    fs = _score(fh, n)
    if not math.isfinite(fs) or fs >= _score(fb, n):
        fh, fs = fb.copy(), _score(fb, n)
    if not math.isfinite(fs):
        fh = np.full(n, 0.5)
        fs = _score(fh, n)
    return fh.copy(), float(fs), int(n)
