    G = np.zeros((M, n))
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        idx = np.arange(n - d)
        hi = h[idx]
        hp = h[idx + d]
        q[d] = dx * np.sum(hp * (1.0 - hi))
        G[d, idx] += -dx * hp
        G[d, idx + d] += dx * (1.0 - hi)
        qi = n + d - 1
        q[qi] = dx * np.sum(hi * (1.0 - hp))
        G[qi, idx] += dx * (1.0 - hp)
        G[qi, idx + d] += -dx * hi
    return q, G


def capped_projection(v, S):
    n = len(v)
    v = np.clip(v, -1e6, 1e6)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    mid = 0.5 * (lo + hi)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(v - mid, 0.0, 1.0)
        if math.fsum(h) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - mid, 0.0, 1.0)
    res = S - math.fsum(h)
    if abs(res) > 1e-14:
        if res > 0:
            order = np.argsort(1.0 - h)[::-1]
            for j in order:
                a = min(1.0 - h[j], res)
                h[j] += a
                res -= a
                if abs(res) < 1e-14:
                    break
        else:
            order = np.argsort(h)[::-1]
            for j in order:
                a = min(h[j], -res)
                h[j] -= a
                res += a
                if abs(res) < 1e-14:
                    break
    s2 = math.fsum(h)
    if abs(s2 - S) > 1e-10:
        intm = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(intm) > 0:
            j = intm[0]
            oth = S - (s2 - h[j])
            if 0.0 <= oth <= 1.0:
                h[j] = oth
    return h


def soft_obj_fn(h, tau):
    q, G = compute_lags(h)
    qm = float(np.max(q))
    sh = (q - qm) / tau
    p = np.exp(sh)
    ps = float(np.sum(p))
    val = qm + tau * math.log(ps)
    pp = p / ps
    grad = pp @ G
    return val, grad


def eq_con(h):
    return math.fsum(h) - len(h) / 2.0


def eq_con_jac(h):
    return np.ones(len(h))


def epi_obj_fn(x):
    return x[len(x) - 1]


def epi_obj_grad(x):
    nv = len(x)
    g = np.zeros(nv)
    g[nv - 1] = 1.0
    return x[nv - 1], g


def epi_eq(x):
    n = len(x) - 1
    return math.fsum(x[:n]) - n / 2.0


def epi_eq_jac(x):
    n = len(x) - 1
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g


def epi_ineq(x):
    n = len(x) - 1
    q, _ = compute_lags(x[:n])
    return x[n] - q


def epi_ineq_jac(x):
    n = len(x) - 1
    M = 2 * n - 1
    _, G = compute_lags(x[:n])
    J = np.zeros((M, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J


def run(seed=42, **kwargs):
    n = 42
    best_h = None
    best_score = None
    try:
        parent = np.array(initial_h_values, dtype=float).ravel()
        if len(parent) >= 2:
            n = len(parent)
            if np.all(np.isfinite(parent)):
                best_h = capped_projection(parent, n / 2.0)
                best_score = exact_score(best_h)
    except Exception:
        pass
    if best_h is None:
        n = int(kwargs.get('n_points', 42))
        n = max(2, min(n, 256))
        best_h = np.full(n, 0.5)
        best_score = exact_score(best_h)
    S = n / 2.0
    rng = np.random.default_rng(seed)
    starts = [best_h.copy()]
    for sig in [0.10, 0.25, 0.50]:
        starts.append(capped_projection(best_h + sig * rng.standard_normal(n), S))
    blk = np.zeros(n)
    n1 = n // 2
    blk[:n1] = 1.0
    if n % 2 == 1:
        blk[n1] = 0.5
    starts.append(capped_projection(blk, S))
    alt = np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n)])
    starts.append(capped_projection(alt, S))
    for _ in range(2):
        starts.append(capped_projection(rng.random(n), S))
    idx = np.arange(n)
    starts.append(capped_projection(0.5 + 0.45 * np.sin(2.0 * np.pi * idx / n), S))
    uniq = []
    for s in starts:
        if not np.all(np.isfinite(s)):
            continue
        dup = any(np.max(np.abs(s - u)) < 1e-10 for u in uniq)
        if not dup:
            uniq.append(s)
    do_opt = n <= 256
    bounds = [(0.0, 1.0)] * n
    if do_opt:
        temps = [0.02, 0.005, 0.001, 0.0002]
        con_eq = {'type': 'eq', 'fun': eq_con, 'jac': eq_con_jac}
        for st in uniq:
            cur = st.copy()
            for tau in temps:
                try:
                    res = minimize(soft_obj_fn, cur, args=(tau,), method='SLSQP',
                                   jac=True, bounds=bounds, constraints=[con_eq],
                                   options={'ftol': 1e-10, 'maxiter': 120})
                    if res is not None and hasattr(res, 'x') and np.all(np.isfinite(res.x)):
                        pr = capped_projection(res.x, S)
                        sc = exact_score(pr)
                        if sc < best_score - 1e-12:
                            best_h = pr.copy()
                            best_score = sc
                        vn = soft_obj_fn(pr, tau)[0]
                        vo = soft_obj_fn(cur, tau)[0]
                        if vn <= vo + 1e-12:
                            cur = pr
                except Exception:
                    pass
            try:
                qc, _ = compute_lags(cur)
                t0 = float(np.max(qc)) + max(1e-8, 1e-6 * abs(float(np.max(qc))))
                x0 = np.concatenate([cur, [t0]])
                con_e = {'type': 'eq', 'fun': epi_eq, 'jac': epi_eq_jac}
                con_i = {'type': 'ineq', 'fun': epi_ineq, 'jac': epi_ineq_jac}
                r2 = minimize(epi_obj_fn, x0, method='SLSQP', jac=True,
                              bounds=bounds + [(0.0, None)],
                              constraints=[con_e, con_i],
                              options={'ftol': 1e-10, 'maxiter': 200})
                if r2 is not None and hasattr(r2, 'x') and np.all(np.isfinite(r2.x)):
                    pr = capped_projection(r2.x[:n], S)
                    sc = exact_score(pr)
                    if sc < best_score - 1e-12:
                        best_h = pr.copy()
                        best_score = sc
            except Exception:
                pass
    try:
        qb, _ = compute_lags(best_h)
        t0 = float(np.max(qb)) + max(1e-8, 1e-6 * abs(float(np.max(qb))))
        x0 = np.concatenate([best_h, [t0]])
        con_e = {'type': 'eq', 'fun': epi_eq, 'jac': epi_eq_jac}
        con_i = {'type': 'ineq', 'fun': epi_ineq, 'jac': epi_ineq_jac}
        r3 = minimize(epi_obj_fn, x0, method='SLSQP', jac=True,
                      bounds=bounds + [(0.0, None)],
                      constraints=[con_e, con_i],
                      options={'ftol': 1e-10, 'maxiter': 200})
        if r3 is not None and hasattr(r3, 'x') and np.all(np.isfinite(r3.x)):
            pr = capped_projection(r3.x[:n], S)
            sc = exact_score(pr)
            if sc < best_score - 1e-12:
                best_h = pr.copy()
                best_score = sc
    except Exception:
        pass
    best_h = capped_projection(best_h, S)
    final_score = exact_score(best_h)
    if not (np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)):
        best_h = np.full(n, 0.5)
        final_score = exact_score(best_h)
    return best_h, float(final_score), n
