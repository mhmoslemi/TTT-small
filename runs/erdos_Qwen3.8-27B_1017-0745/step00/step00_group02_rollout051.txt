    best_h = fallback.copy()
    best_score = _score(best_h, dx)
    locals_list = [best_h.copy()]

    def _archive(h):
        nonlocal best_h, best_score
        if _feasible(h, n):
            sc = _score(h, dx)
            if sc < best_score - 1e-12:
                best_h = h.copy()
                best_score = sc
            locals_list.append(h.copy())

    starts = [best_h.copy()]
    if parent is not None:
        starts.append(_project(1.0 - parent, S))
        starts.append(_project(parent[::-1].copy(), S))
        starts.append(_project(1.0 - parent[::-1].copy(), S))
        starts.append(_project(parent + 0.08 * rng.standard_normal(n), S))
        starts.append(_project(parent + 0.25 * rng.standard_normal(n), S))
    starts.append(_project(rng.uniform(0, 1, n), S))
    starts.append(_project(rng.beta(0.5, 0.5, n), S))

    tau_fine = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00001]
    if n > 100:
        tau_fine = [0.02, 0.005, 0.001, 0.0001, 0.00001]

    for s in starts:
        _archive(s)
        h_opt = _pg(s, dx, tau_fine)
        _archive(h_opt)

    sigma_list = [0.04, 0.12, 0.30, 0.50] if n <= 100 else [0.12]
    for sig in sigma_list:
        pt = _project(best_h + sig * rng.standard_normal(n), S)
        _archive(pt)
        h_opt = _pg(pt, dx, tau_fine)
        _archive(h_opt)

    if n <= 100:
        scored = []
        for hr in locals_list:
            scored.append((_score(hr, dx), hr))
        scored.sort(key=lambda t: t[0])
        refined_starts = []
        for sc, hr in scored:
            if len(refined_starts) >= 3:
                break
            if all(np.max(np.abs(hr - r)) > 1e-5 for r in refined_starts):
                refined_starts.append(hr)
        if not refined_starts:
            refined_starts = [best_h.copy()]
        for hr in refined_starts:
            sl = _slsqp_refine(hr, dx, n, S)
            if sl is not None:
                sl = _project(sl, S)
                _archive(sl)

    best_h = _project(best_h, S)
    if not _feasible(best_h, n):
        best_h = fallback.copy()
    best_score = _score(best_h, dx)
    return (best_h.astype(np.float64), best_score, n)
