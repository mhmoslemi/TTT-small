    dx = 2.0 / N
    nz = 2 * N - 2
    d = cp.Variable(N); u = cp.Variable(); t = cp.Variable()
    Px = cp.Parameter(N); Pr0 = cp.Parameter(); PJ0 = cp.Parameter(N)
    Prnz = cp.Parameter(nz); PJnz = cp.Parameter((nz, N))
    PR = cp.Parameter()
    cons = [Px + d >= 0, Px + d <= 1, cp.sum(d) == 0, cp.norm(d, 2) <= PR,
            u >= Prnz + PJnz @ d, t >= u + dx * cp.sum_squares(d),
            t >= Pr0 + PJ0 @ d]
    return cp.Problem(cp.Minimize(t), cons), (d, u, t, Px, Pr0, PJ0, Prnz, PJnz, PR)

def local_opt(start, N, n_iter):
    dx = 2.0 / N
    cur = start.copy(); cs = score_exact(cur, N)
    R = min(1.0, math.sqrt(N)); Rmax = min(4.0, math.sqrt(N)); Rmin = 1e-4
    fails = 0
    try:
        prob, (d, u, t, Px, Pr0, PJ0, Prnz, PJnz, PR) = build_model(N)
    except Exception:
        return cur, cs
    row0 = N - 1
    nz_idx = [i for i in range(2 * N - 1) if i != row0]
    for _ in range(n_iter):
        if R < Rmin or fails >= 8:
            break
        q, G = lag_vals_grads(cur, N)
        if q is None:
            break
        Px.value = cur; Pr0.value = q[row0]; PJ0.value = G[row0]
        Prnz.value = q[nz_idx]; PJnz.value = G[nz_idx]
        PR.value = R
        try:
            prob.solve(solver=cp.ECOS, max_iters=200, abstol=1e-8,
                       reltol=1e-8, feastol=1e-8, verbose=False)
        except Exception:
            R = max(R / 2.0, Rmin); fails += 1; continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            R = max(R / 2.0, Rmin); fails += 1; continue
        dv = d.value
        if dv is None or len(dv) != N or not np.all(np.isfinite(dv)):
            R = max(R / 2.0, Rmin); fails += 1; continue
        trial = project_box_sum(cur + dv, N)
        de = trial - cur
        if np.linalg.norm(de) > R * (1 + 1e-6) + 1e-10:
            R = max(R / 2.0, Rmin); fails += 1; continue
        ns = score_exact(trial, N)
        if ns < cs - 1e-12 * max(1.0, abs(cs)):
            cur = trial; cs = ns; fails = 0
            if np.linalg.norm(de) >= 0.8 * R:
                R = min(R * 1.5, Rmax)
            if np.linalg.norm(de) < 1e-7:
                break
        else:
            R = max(R / 2.0, Rmin); fails += 1
    return cur, cs

def run(seed=42, **kwargs):
    N = None; baseline = None
    try:
        ival = np.array(initial_h_values, dtype=np.float64).ravel()
        if ival.ndim == 1 and ival.size >= 2:
            N = int(ival.size)
            fint = np.where(np.isfinite(ival), ival, 0.5)
            if np.min(fint) >= -1e-10 and np.max(fint) <= 1 + 1e-10:
                fc = np.clip(fint, 0, 1)
                if abs(np.sum(fc) - N / 2.0) <= 1e-10 * N:
                    baseline = fc.copy()
                else:
                    baseline = project_box_sum(fc, N)
            else:
                baseline = project_box_sum(fint, N)
    except Exception:
        pass
    if N is None:
        try:
            N = int(kwargs.get('n_points', 70))
            if N < 2 or N > 999:
                N = 70
        except Exception:
            N = 70
    if baseline is None:
        baseline = np.full(N, 0.5)
    if N < 2:
        return np.array([0.5]), 0.5, 1
    try:
        rng = np.random.default_rng(seed if isinstance(seed, (int, np.integer)) else 42)
    except Exception:
        rng = np.random.default_rng(42)
    starts = [baseline.copy(), np.full(N, 0.5)]
    starts.append(project_box_sum(1.0 - baseline, N))
    starts.append(project_box_sum(baseline + 0.15 * rng.standard_normal(N), N))
    starts.append(project_box_sum(rng.uniform(0, 1, N), N))
    Rf = min(5, (N - 1) // 2)
    if Rf > 0:
        i = np.arange(N, dtype=np.float64)
        zf = np.full(N, 0.5)
        for r in range(1, Rf + 1):
            A, B = rng.standard_normal(), rng.standard_normal()
            zf += (0.12 / r) * (A * np.sin(2 * np.pi * r * i / N) + B * np.cos(2 * np.pi * r * i / N))
        starts.append(project_box_sum(zf, N))
    else:
        starts.append(np.full(N, 0.5))
    deduped = []
    for s in starts:
        ok = True
        for dd in deduped:
            if np.max(np.abs(s - dd)) < 1e-8:
                ok = False; break
        if ok and np.all(np.isfinite(s)):
            deduped.append(s)
    if not deduped:
        deduped = [np.full(N, 0.5)]
    if N <= 100:
        ns_, ni_ = min(6, len(deduped)), 20
    elif N <= 250:
        ns_, ni_ = min(5, len(deduped)), 10
    elif N <= 500:
        ns_, ni_ = min(4, len(deduped)), 6
    elif N <= 1000:
        ns_, ni_ = min(3, len(deduped)), 3
    else:
        ns_, ni_ = 1, 0
    best_h = deduped[0].copy(); best_c5 = score_exact(best_h, N)
    for s in deduped[1:]:
        c = score_exact(s, N)
        if c < best_c5:
            best_h, best_c5 = s.copy(), c
    if ni_ > 0:
        for s in deduped[:ns_]:
            oh, oc = local_opt(s, N, ni_)
            if oc < best_c5:
                best_h, best_c5 = oh, oc
    best_h = np.clip(best_h, 0, 1)
    if abs(np.sum(best_h) - N / 2.0) > 1e-10 * N:
        best_h = project_box_sum(best_h, N)
    if not np.all(np.isfinite(best_h)) or np.min(best_h) < -1e-12 or np.max(best_h) > 1 + 1e-12:
        best_h = np.full(N, 0.5)
    best_c5 = score_exact(best_h, N)
    if not np.isfinite(best_c5):
        best_h = np.full(N, 0.5)
        best_c5 = score_exact(best_h, N)
    return best_h.astype(np.float64), float(best_c5), int(N)
