```python
        base = uniform_h(n)
    if not is_feasible(base, n):
        pr = project_capped(base, n)
        base = pr if pr is not None else uniform_h(n)

    fallback = base.copy()
    fb_score = exact_score(fallback, n)
    incumbent, inc_score = base.copy(), fb_score

    if n == 1:
        return fallback, float(fb_score), 1
    if n > 999:
        return fallback, float(fb_score), n

    if n <= 64:
        P, G, pat = 32, 160, 24
    elif n <= 128:
        P, G, pat = 20, 80, 16
    elif n <= 256:
        P, G, pat = 12, 32, 8
    else:
        P, G, pat = 8, 12, 4

    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)

    pop = build_pop(base, n, rng, P)
    scores = np.array([exact_score(m, n) for m in pop])
    for i in range(P):
        if scores[i] < inc_score - 1e-12:
            inc_score = scores[i]
            incumbent = pop[i].copy()

    no_imp, n_restart = 0, 0
    prev_best = inc_score

    try:
        for g in range(G):
            snap = [m.copy() for m in pop]
            ss = scores.copy()
            for i in range(P):
                try:
                    others = [j for j in range(P) if j != i]
                    perm = rng.choice(len(others), 3, replace=False)
                    a, b, c = others[int(perm[0])], others[int(perm[1])], others[int(perm[2])]
                    F = rng.uniform(0.45, 0.90)
                    CR_g = 0.90 - 0.25 * g / max(1, G - 1)
                    jr = int(rng.integers(0, n))
                    mk = rng.random(n) < CR_g
                    mk[jr] = True
                    donor = snap[a] + F * (snap[b] - snap[c])
                    w = np.where(mk, donor, snap[i])
                    trial = project_capped(w, n)
                    if trial is None:
                        continue
                    sc = exact_score(trial, n)
                    if sc < ss[i] - 1e-12:
                        pop[i] = trial
                        scores[i] = sc
                        if sc < inc_score - 1e-12:
                            inc_score = sc
                            incumbent = trial.copy()
                except Exception:
                    continue

            if inc_score < prev_best - 1e-12:
                no_imp = 0
            else:
                no_imp += 1
            prev_best = inc_score

            if no_imp >= pat:
                if n_restart >= 2:
                    break
                order = np.argsort(scores)
                elite = [pop[int(order[k])].copy() for k in range(P // 2)]
                elite_sc = [scores[int(order[k])] for k in range(P // 2)]
                sd = 0.10 if n_restart == 0 else 0.30
                for k in range(P // 2, P):
                    slot = k - P // 2
                    if slot % 3 == 2:
                        prop = rng.uniform(0, 1, n).astype(np.float64)
                    else:
                        prop = incumbent + rng.normal(0, sd, n)
                    h = project_capped(prop, n)
                    if h is None:
                        h = uniform_h(n)
                    sc = exact_score(h, n)
                    pop[k] = h
                    scores[k] = sc
                    if sc < inc_score - 1e-12:
                        inc_score = sc
                        incumbent = h.copy()
                pop = elite + pop[P // 2:]
                scores = np.array(elite_sc + list(scores[P // 2:]))
                no_imp = 0
                n_restart += 1
    except Exception:
        pass

    final = project_capped(incumbent, n)
    if final is None:
        final = fallback
    final_sc = exact_score(final, n)
    if not np.isfinite(final_sc):
        final = fallback
        final_sc = fb_score
    if fb_score <= final_sc:
        final = fallback
        final_sc = fb_score
    final_sc = exact_score(final, n)
    if not np.isfinite(final_sc):
        final = uniform_h(n)
        final_sc = exact_score(final, n)
    return final, float(final_sc), int(n)
```
