import numpy as np
import math
from scipy.optimize import linprog

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def is_feasible(h, n):
    if h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    S = n / 2.0
    if abs(np.sum(h) - S) > 1e-10:
        return False
    if abs(math.fsum(float(x) for x in h) - S) > 1e-10:
        return False
    return True

def project_capped(v, n):
    S = n / 2.0
    v = np.clip(np.asarray(v, dtype=np.float64), -1e6, 1e6)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        clip_sum = np.sum(np.clip(v - mid, 0.0, 1.0))
        if clip_sum > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(v - lam, 0.0, 1.0)
    for _ in range(3):
        res = S - math.fsum(float(x) for x in h)
        if abs(res) < 1e-13:
            break
        if res > 0:
            for i in range(n):
                if res <= 1e-15:
                    break
                add = min(res, 1.0 - h[i])
                h[i] += add
                res -= add
        else:
            for i in range(n):
                if res >= -1e-15:
                    break
                sub = min(-res, h[i])
                h[i] -= sub
                res += sub
    res2 = S - np.sum(h)
    if abs(res2) > 1e-13:
        for i in range(n):
            if abs(res2) < 1e-15:
                break
            adj = min(abs(res2), max(0.0, (1.0 - h[i]) if res2 > 0 else h[i])) * (1.0 if res2 > 0 else -1.0)
            h[i] += adj
            res2 -= adj if res2 > 0 else -adj
    if not is_feasible(h, n):
        j = n // 2
        others = math.fsum(float(h[k]) for k in range(n) if k != j)
        vals_j = S - others
        if 0.0 <= vals_j <= 1.0:
            h[j] = vals_j
    if is_feasible(h, n):
        return h
    return None

def compute_lags(h, n):
    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0] = dx * (1.0 - 2.0 * h)
    for r in range(1, n):
        idx_pos = 2 * r - 1
        idx_neg = 2 * n - 1 - 2 * r
        for i in range(n - r):
            q[idx_pos] += dx * h[i + r] * (1.0 - h[i])
            q[idx_neg] += dx * h[i] * (1.0 - h[i + r])
            G[idx_pos, i] += -dx * h[i + r]
            G[idx_pos, i + r] += dx * (1.0 - h[i])
            G[idx_neg, i] += dx * (1.0 - h[i + r])
            G[idx_neg, i + r] += -dx * h[i]
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None, None
    return q, G

def lp_step(h, q, G, dx, delta, n):
    M = 2 * n - 1
    e = dx * delta * delta
    nv = 2 * n + 1
    c = np.zeros(nv)
    c[2 * n] = 1.0
    rows = M + 4 * n + 1
    A_ub = np.zeros((rows, nv))
    b_ub = np.zeros(rows)
    for l in range(M):
        A_ub[l, 0:n] = G[l]
        A_ub[l, 2 * n] = -1.0
        b_ub[l] = -q[l] - e
    row = M
    for i in range(n):
        A_ub[row, i] = 1.0
        A_ub[row, n + i] = -1.0
        b_ub[row] = 0.0
        row += 1
        A_ub[row, i] = -1.0
        A_ub[row, n + i] = -1.0
        b_ub[row] = 0.0
        row += 1
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    for i in range(n):
        A_ub[row, n + i] = 1.0
    b_ub[row] = delta
    A_eq = np.zeros((1, nv))
    A_eq[0, 0:n] = 1.0
    b_eq = np.array([0.0])
    bounds = [(None, None)] * n + [(0.0, None)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'maxiter': 500})
        if res.status == 0 and res.x is not None:
            x = np.asarray(res.x, dtype=np.float64)
            if x.shape == (nv,) and np.all(np.isfinite(x)):
                return x[0:n]
    except Exception:
        pass
    return None

def trust_search(h0, n):
    dx = 2.0 / n
    hp = project_capped(h0, n)
    if hp is None:
        return h0
    hcur = hp.copy()
    ccur = exact_score(hcur)
    delta = max(0.25, 0.03 * n)
    delta_max = max(1.0, 0.12 * n)
    delta_min = 1e-4
    delta = min(delta, delta_max)
    for _ in range(50):
        if delta < delta_min:
            break
        q, G = compute_lags(hcur, n)
        if q is None:
            break
        p = lp_step(hcur, q, G, dx, delta, n)
        if p is None:
            delta *= 0.5
            continue
        htrial = hcur + p
        htrial = project_capped(htrial, n)
        if htrial is None:
            delta *= 0.5
            continue
        m = np.sum(np.abs(htrial - hcur))
        if m > delta:
            alpha = delta * (1.0 - 1e-12) / m
            htrial = hcur + alpha * (htrial - hcur)
            htrial = project_capped(htrial, n)
            if htrial is None:
                delta *= 0.5
                continue
        if not np.all(np.isfinite(htrial)):
            delta *= 0.5
            continue
        if np.max(np.abs(htrial - hcur)) < 1e-14:
            delta *= 0.5
            continue
        ctrial = exact_score(htrial)
        if ctrial < ccur - 1e-12:
            disp = htrial - hcur
            lm = np.max(q + G @ disp)
            pred = ccur - lm
            ared = ccur - ctrial
            hcur = htrial
            ccur = ctrial
            if pred > 1e-12:
                rho = ared / pred
                if rho < 0.1:
                    delta *= 0.5
                elif rho > 0.75 and np.sum(np.abs(disp)) >= 0.8 * delta:
                    delta = min(2.0 * delta, delta_max)
            else:
                delta *= 0.5
        else:
            delta *= 0.5
    return hcur

def make_starts(n, rng, parent):
    starts = []
    seen = []

    def add_unique(hv):

    i = np.arange(n, dtype=np.float64)

    if parent is not None:

        add_unique(rng.uniform(0.0, 1.0, n))
