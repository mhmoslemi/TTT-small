def lag_values_and_grad(h, dx):
    n = len(h)
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    # d=0, index n-1
    q[n-1] = dx * np.dot(h, 1.0 - h)
    G[n-1] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        # positive lag d, index n-1+d
        idx_p = n - 1 + d
        overlap = h[:n-d] * (1.0 - h[d:])
        q[idx_p] = dx * np.sum(overlap)
        # gradient
        grad = np.zeros(n)
        grad[:n-d] += dx * (1.0 - h[d:n])
        grad[d:] -= dx * h[:n-d]
        G[idx_p] = grad
        # negative lag d, index n-1-d
        idx_m = n - 1 - d
        overlap_m = h[d:] * (1.0 - h[:n-d])

        grad_m[d:] += dx * (1.0 - h[:n-d])

import math

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def lag_qG(h, dx):
    n = len(h)
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    q[n-1] = dx * np.dot(h, 1.0 - h)
    G[n-1] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        ip = n - 1 + d
        q[ip] = dx * np.sum(h[:n-d] * (1.0 - h[d:]))
        gp = np.zeros(n)
        gp[:n-d] += dx * (1.0 - h[d:])
        gp[d:] -= dx * h[:n-d]
        G[ip] = gp
        im = n - 1 - d
        q[im] = dx * np.sum(h[d:] * (1.0 - h[:n-d]))
        gm = np.zeros(n)
        gm[d:] += dx * (1.0 - h[:n-d])
        gm[:n-d] -= dx * h[d:]
        G[im] = gm
    return q, G

def project_feasible(v, n):

    v = np.clip(v.astype(float), -1e6, 1e6)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))

    qmax = float(np.max(q))
    sh = (q - qmax) / tau
    ev = np.exp(sh)
    sv = np.sum(ev)
    val = qmax + tau * math.log(sv)
    p = ev / sv
    grad = p @ G
    return val, grad

def epi_ineq(z, dx, n):
