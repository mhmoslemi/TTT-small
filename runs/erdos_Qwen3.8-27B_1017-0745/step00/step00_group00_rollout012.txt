            return np.asarray(res.x[:-1], dtype=np.float64)
    except Exception:
        pass
    return None

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)

    n = 45
    parent = None
    if 'initial_h_values' in globals():
        pv = np.asarray(globals()['initial_h_values'], dtype=np.float64)
        if pv.ndim == 1 and 2 <= len(pv) <= 999 and np.all(np.isfinite(pv)):
            if np.all(pv >= -1e-6) and np.all(pv <= 1.0 + 1e-6):
                n = len(pv)
                pv_clip = np.clip(pv, 0.0, 1.0)
                S = n / 2.0
                parent = project_box_sum(pv_clip, S)
                if parent is None:
                    parent = pv_clip.copy()
                    parent[0] += S - float(np.sum(parent))
                    parent = project_box_sum(parent, S)
                if parent is None:
                    parent = None
                    n = 45
    if parent is None:
        n = 45
        parent = np.full(n, 0.5)
    S = n / 2.0
    best = parent.copy()
    best_score = exact_score(best)

    starts = [best.copy(), best[::-1].copy()]
    sigmas_list = [0.01, 0.03, 0.06, 0.12, 0.20, 0.05]
    for s in sigmas_list:
        pert = best + s * rng.standard_normal(n)
        p = project_box_sum(pert, S)
        if p is not None:
            starts.append(p.copy())
    for _ in range(4):
        u = rng.uniform(0.0, 1.0, n)
        p = project_box_sum(u, S)
        if p is not None:
            starts.append(p.copy())

    for h0 in starts:
        if not check_feasible(h0, n):
            continue
        hr = solve_one(h0, n)
        if hr is not None:
            hr = project_box_sum(hr, S)
            if hr is not None and check_feasible(hr, n):
                sc = exact_score(hr)
                if sc < best_score - 1e-12:
                    best = hr.copy()
                    best_score = sc

    sigma = 0.05
    for _ in range(4):
        pert = best + sigma * rng.standard_normal(n)
        p = project_box_sum(pert, S)
        if p is None:
            break
        hr = solve_one(p, n)
        improved = False
        if hr is not None:
            hr = project_box_sum(hr, S)
            if hr is not None and check_feasible(hr, n):
                sc = exact_score(hr)
                if sc < best_score - 1e-12:
                    best = hr.copy()
                    best_score = sc
                    improved = True
        if improved:
            sigma = max(0.01, 0.7 * sigma)
        else:
            sigma = min(0.25, 1.5 * sigma)

    best = project_box_sum(best, S)
    if best is None or not check_feasible(best, n):
        fb = parent.copy() if parent is not None else np.full(n, 0.5)
        best = project_box_sum(fb, S)
        if best is None:
            best = np.full(n, 0.5)
    best = np.clip(best, 0.0, 1.0)
    best_score = exact_score(best)
    best = np.asarray(best, dtype=np.float64)
    return (best, best_score, n)
