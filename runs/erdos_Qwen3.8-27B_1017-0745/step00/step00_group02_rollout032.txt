    return abs(float(np.sum(h)) - S) <= 1e-10

def project_k(y, n, S):
    y = np.asarray(y, dtype=np.float64)
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return None
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = float(np.sum(x))
        if s > S:
            lo = mid
        else:
            hi = mid
        if abs(s - S) < 1e-13 or (hi - lo) < 1e-14 * max(1.0, abs(mid)):
            break
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    r = S - float(np.sum(x))
    for i in range(n):
        if r > 1e-13:
            a = min(r, 1.0 - x[i])
            x[i] += a
            r -= a
        elif r < -1e-13:
            a = min(-r, x[i])
            x[i] -= a
            r += a
        else:
            break
    x = np.clip(x, 0.0, 1.0)
    if abs(float(np.sum(x)) - S) > 1e-10:
        return None
    return x

def exact_score(h, dx):
    try:
        corr = np.correlate(h, 1.0 - h, mode='full')
        if not np.all(np.isfinite(corr)):
            return float('inf')
        return float(np.max(corr) * dx)
    except Exception:
        return float('inf')

def gen_cand(rng, n, S, base, mode, scale):
    try:
        if mode == 0:
            y = np.random.uniform(0, 1, n)
        elif mode == 1:
            y = np.random.beta(0.5, 0.5, n)
        elif mode == 2:
            y = np.random.beta(2, 2, n)
        elif mode == 3:
            K = int(rng.integers(1, min(6, n // 2) + 1))
            y = np.full(n, 0.5)
            ii = np.arange(n, dtype=np.float64)
            for k in range(1, K + 1):
                a = float(rng.normal(0, 0.30 / math.sqrt(k)))
                b = float(rng.normal(0, 0.30 / math.sqrt(k)))
                y += a * np.cos(2.0 * np.pi * k * ii / n)
                y += b * np.sin(2.0 * np.pi * k * ii / n)
        elif mode >= 4:
            y = base + scale * rng.standard_normal(n)
        else:
            y = np.full(n, 0.5)
        return project_k(y, n, S)
    except Exception:
        return None

def run(seed=42, **kwargs):
    n = 45
    fallback = None
    try:
        parent = globals().get('initial_h_values', None)
        if parent is not None:
            p = np.asarray(parent, dtype=np.float64)
            if p.ndim == 1 and 2 <= p.shape[0] <= 999:
                n = p.shape[0]
                S_t = n / 2.0
                if np.all(np.isfinite(p)) and float(np.max(np.abs(p))) < 1e6:
                    pp = np.clip(p, 0.0, 1.0)
                    if abs(float(np.sum(pp)) - S_t) <= 1e-12:
                        fallback = pp.copy()
                    else:
                        fallback = project_k(pp, n, S_t)
                if fallback is None:
                    fallback = project_k(p, n, S_t)
                if fallback is None:
                    fallback = np.full(n, 0.5)
    except Exception:
        pass
    if fallback is None:
        np_ = kwargs.get('n_points', None)
        if isinstance(np_, int) and not isinstance(np_, bool) and 2 <= np_ <= 999:
            n = np_
        fallback = np.full(n, 0.5)

    S = n / 2.0
    dx = 2.0 / n
    if n <= 64:
        P = max(48, min(160, 3 * n))
        G, stag_t, max_rs = 120, 24, 2
    elif n <= 256:
        P, G, stag_t, max_rs = 64, 40, 12, 1
    else:
        P, G, stag_t, max_rs = 32, 20, 8, 1

    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)

    pop = [fallback.copy() for _ in range(P)]
    scores = [exact_score(pop[i], dx) for i in range(P)]
    Fv = [float(rng.uniform(0.4, 0.9)) for _ in range(P)]
    Cv = [float(rng.uniform(0.25, 0.95)) for _ in range(P)]

    idx = 1
    try:
        parent = globals().get('initial_h_values', None)
        if parent is not None:
            p = np.asarray(parent, dtype=np.float64)
            if p.ndim == 1 and p.shape[0] == n:
                for var in [p[::-1].copy(), (1.0 - p).copy(), (1.0 - p)[::-1].copy()]:
                    if idx >= P:
                        break
                    if np.all(np.isfinite(var)) and abs(float(np.sum(var)) - S) < 1e-8:
                        pop[idx] = np.clip(var, 0.0, 1.0)
                        scores[idx] = exact_score(pop[idx], dx)
                        idx += 1
    except Exception:
        pass

    modes = [0, 1, 2, 3, 4, 4, 4, 3, 4, 4]
    scales = [0.0, 0.0, 0.0, 0.0, 0.03, 0.08, 0.18, 0.0, 0.35, 0.60]
    mi = 0
    while idx < P:
        m = modes[mi % 10]
        sc = scales[mi % 10]
        c = gen_cand(rng, n, S, fallback, m, sc)
        if c is None:
            c = fallback.copy()
        pop[idx] = c
        scores[idx] = exact_score(c, dx)
        idx += 1
        mi += 1

    arch_h = fallback.copy()
    arch_s = exact_score(arch_h, dx)
    for i in range(P):
        if check_feasible(pop[i], n, S) and scores[i] < arch_s - 1e-12:
            arch_h = pop[i].copy()
            arch_s = scores[i]

    stag = 0
    n_restarts = 0
    prev_arch = arch_s

    for gen in range(G):
        fp = [pop[i].copy() for i in range(P)]
        fs = scores[:]
        fF = Fv[:]
        fC = Cv[:]
        order = [int(x) for x in np.argsort(np.array(fs))]
        pbs = max(2, math.ceil(0.2 * P))
        pp = order[:pbs]

        for i in range(P):
            cp = [j for j in pp if j != i]
            if not cp:
                cp = [j for j in range(P) if j != i]
            p = cp[int(rng.integers(len(cp)))]
            av = [j for j in range(P) if j != i and j != p]
            r1 = av[int(rng.integers(len(av)))]
            av2 = [j for j in av if j != r1]
            r2 = av2[int(rng.integers(len(av2)))]
            Fi = fF[i]
            Ci = fC[i]
            if float(rng.random()) < 0.1:
                Fi = float(rng.uniform(0.3, 0.9))
            if float(rng.random()) < 0.1:
                Ci = float(rng.uniform(0.1, 0.95))
            v = fp[i] + Fi * (fp[p] - fp[i]) + Fi * (fp[r1] - fp[r2])
            mask = rng.random(n) < Ci
            mask[int(rng.integers(n))] = True
            w = np.where(mask, v, fp[i])
            trial = project_k(w, n, S)
            if trial is not None and check_feasible(trial, n, S):
                ts = exact_score(trial, dx)
                if ts < fs[i] - 1e-12:
                    pop[i] = trial
                    scores[i] = ts
                    Fv[i] = Fi
                    Cv[i] = Ci
                    if ts < arch_s - 1e-12:
                        arch_h = trial.copy()
                        arch_s = ts
                else:
                    pop[i] = fp[i]
                    scores[i] = fs[i]
            else:
                pop[i] = fp[i]
                scores[i] = fs[i]

        found = False
        for i in range(P):
            if float(np.max(np.abs(pop[i] - arch_h))) < 1e-12:
                found = True
                break
        if not found:
            worst = int(np.argmax(np.array(scores)))
            pop[worst] = arch_h.copy()
            scores[worst] = arch_s
            Fv[worst] = 0.6
            Cv[worst] = 0.5

        if arch_s < prev_arch - 1e-12:
            stag = 0
        else:
            stag += 1
        prev_arch = arch_s

        if stag >= stag_t and n_restarts < max_rs:
            sig = 0.06 if n_restarts == 0 else 0.18
            so = [int(x) for x in np.argsort(np.array(scores))]
            keep_set = set(so[:math.ceil(P // 2)])
            repl = [j for j in range(P) if j not in keep_set]
            for ri in range(len(repl)):
                if ri < len(repl) // 2:
                    c = gen_cand(rng, n, S, arch_h, 4, sig)
                else:
                    m = modes[(mi + ri) % 10]
                    c = gen_cand(rng, n, S, arch_h, m, scales[m % 10])
                if c is None:
                    c = arch_h.copy()
                pop[repl[ri]] = c
                scores[repl[ri]] = exact_score(c, dx)
                Fv[repl[ri]] = float(rng.uniform(0.4, 0.9))
                Cv[repl[ri]] = float(rng.uniform(0.25, 0.95))
            n_restarts += 1
            stag = 0

    try:
        final = arch_h.copy()
        if not check_feasible(final, n, S):
            final = project_k(final, n, S)
            if final is None:
                final = fallback.copy()
    except Exception:
        final = fallback.copy()

    try:
        fb_s = exact_score(fallback, dx)
        fa = exact_score(final, dx)
        if check_feasible(final, n, S) and fa < fb_s - 1e-12:
            result = final
        else:
            result = fallback.copy()
    except Exception:
        result = np.full(n, 0.5)

    if not check_feasible(result, n, S):
        result = np.full(n, 0.5)
    result = np.clip(result, 0.0, 1.0).astype(np.float64)
    sc = exact_score(result, dx)
    return result, sc, int(n)
