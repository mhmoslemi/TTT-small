    return s1 <= 1e-10 and s2 <= 1e-10

            lo = mid
        else:
            hi = mid

def active_bundle_gradient(q, G):
    Cq = float(np.max(q))
    tol = max(1e-12, 1e-7 * max(1.0, abs(Cq)))
    active = np.where(q >= Cq - tol)[0]
    if len(active) > 12:
        order = np.argsort(q[active])[::-1][:12]
        active = active[order]
    k = len(active)
    if k == 0:
        return None
    U = G[active].copy()
    # center each row
    for i in range(k):
        U[i] -= np.mean(U[i])
    K = U @ U.T
    K = 0.5 * (K + K.T)
    alpha = np.full(k, 1.0 / k)
    for _ in range(40):
        z = K @ alpha
        j = int(np.argmin(z))
        gap = float(alpha @ z) - float(z[j])
        if gap <= 1e-12:
            break
        v = np.zeros(k)
        v[j] = 1.0
        v -= alpha
        vKv = float(v @ K @ v)
        if vKv <= 1e-24:
            break
        vKz = float(v @ K @ z)
        gamma = float(np.clip(-vKz / vKv, 0.0, 1.0))
        alpha += gamma * v
        alpha = np.clip(alpha, 0.0, None)
        s = float(np.sum(alpha))
        if s > 0:
            alpha /= s
        else:
            alpha = np.full(k, 1.0 / k)
    g = alpha @ U
    if not np.all(np.isfinite(g)):
        # fallback to single largest
        idx = int(np.argmax(q))
        g = G[idx].copy()
        g -= np.mean(g)
        if not np.all(np.isfinite(g)):
            return None
    return g

            break
        g = active_bundle_gradient(q, G)
        accepted = False
        if g is not None and np.sum(g**2) > 1e-24:
            for b in range(8):

import numpy as np

def exact_score(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    S = n / 2.0
    return abs(float(np.sum(h)) - S) <= 1e-10 and abs(math.fsum([float(x) for x in h]) - S) <= 1e-10

def project_capped_simplex(v, n):
    if v.shape != (n,) or not np.all(np.isfinite(v)):
        return None
    v = np.clip(v, -1e6, 1e6)
    S = n / 2.0
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        t = np.clip(v - mid, 0.0, 1.0)
        if float(np.sum(t)) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - hi, 0.0, 1.0)
    res = S - math.fsum([float(x) for x in h])

        if res > 0:
            for i in np.argsort(h)[::-1]:
                a = min(res, 1.0 - h[i])
                if a > 0:
                    h[i] += a
                    res -= a
                    if abs(res) < 1e-14:
                        break

            for i in np.argsort(h):
                a = min(-res, h[i])
                if a > 0:
                    h[i] -= a
                    res += a

                        break

    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))

    for d in range(1, n):
        m = n - d
        q[d] = dx * np.sum(h[d:n] * (1.0 - h[:m]))
        G[d, d:n] += dx * (1.0 - h[:m])
        G[d, :m] += -dx * h[d:n]
        qi = n + d - 1
        q[qi] = dx * np.sum(h[:m] * (1.0 - h[d:n]))
        G[qi, :m] += dx * (1.0 - h[d:n])
        G[qi, d:n] += -dx * h[:m]
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None, None
    return q, G

    Cq = float(np.max(q))
    tol = max(1e-12, 1e-7 * max(1.0, abs(Cq)))
    active = np.where(q >= Cq - tol)[0]
    if len(active) > 12:
        active = active[np.argsort(q[active])[::-1][:12]]
