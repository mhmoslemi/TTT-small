from scipy.optimize import linprog

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def project_capped(z, n):

        else:
            hi = mid
    lam = 0.5 * (lo + hi)

            r -= dt

            dt = min(h[i], -r)

            r += dt

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-12) or np.any(h > 1 + 1e-12):
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(q) * (2.0 / n))

def lag_model(x, n):
    s = n / 2.0
    L = 2 * n - 1
    c = (2.0 / n) * np.correlate(x, 1.0 - x, mode='full')
    G = np.zeros((L, n), dtype=np.float64)
    for d in range(-(n - 1), n):
        row = d + n - 1
        i0 = max(d, 0)
        j0 = max(-d, 0)
        m = n - abs(d)
        if d == 0:
            G[row, i0:i0 + m] = 1.0 - 2.0 * x[i0:i0 + m]
        else:
            G[row, i0:i0 + m] += 1.0 - x[j0:j0 + m]
            G[row, j0:j0 + m] -= x[i0:i0 + m]
    if not np.all(np.isfinite(c)) or not np.all(np.isfinite(G)):
        return None
    return c, G

def lp_step(x, R, c, G, n):
    s = n / 2.0
    L, nv = G.shape
    nvar = 2 * n + 1
    A_ub = np.zeros((L + 2 * n + 1, nvar))
    b_ub = np.zeros(L + 2 * n + 1)
    A_ub[:L, :n] = G
    A_ub[:L, 2*n] = -1.0
    b_ub[:L] = G @ x - c
    for i in range(n):
        A_ub[L + i, i] = 1.0
        A_ub[L + i, n + i] = -1.0
        b_ub[L + i] = x[i]
        A_ub[L + n + i, i] = -1.0
        A_ub[L + n + i, n + i] = -1.0
        b_ub[L + n + i] = -x[i]
    A_ub[-1, n:2*n] = 1.0
    b_ub[-1] = R
    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    beq = np.array([s])
    c_obj = np.zeros(nvar)
    c_obj[2*n] = 1.0
    bounds = [(0, 1)] * n + [(0, 1)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=beq,
                      bounds=bounds, method='highs', options={'presolve': True, 'maxiter': 500})
    except Exception:
        return None
    if res.status != 0 or not np.all(np.isfinite(res.x[:nvar])):
        return None
    return res.x[:n].copy()
