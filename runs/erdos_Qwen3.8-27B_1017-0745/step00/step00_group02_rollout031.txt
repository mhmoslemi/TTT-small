            elif abs(B) > 1e-14 * sc:
                r = -C / B
                if np.isfinite(r) and t_lo - tol <= r <= t_hi + tol:
                    cands.append(min(max(r, t_lo), t_hi))
    for i in range(m):
        if abs(c[i]) > 1e-300:
            r = -b[i] / (2.0 * c[i])
            if np.isfinite(r) and t_lo - tol <= r <= t_hi + tol:
                cands.append(min(max(r, t_lo), t_hi))
    cands = sorted(cands)
    ded = [cands[0]]
    for v in cands[1:]:
        if abs(v - ded[-1]) > 1e-12 * max(1.0, abs(v)):
            ded.append(v)
    best_t, best_v = ded[0], np.inf
    for idx in range(len(ded)):
        t = ded[idx]
        v = float(np.max(q + t * b + t * t * c))
        if v < best_v:
            best_v = v
            best_t = t
    return best_t, best_v

def line_min_grid(q, b, c, t_lo, t_hi, n):
    pts = np.linspace(t_lo, t_hi, 17)
    pts = np.sort(np.unique(np.concatenate([pts, [0.0]])))
    best_t, best_v = 0.0, np.inf
    for idx in range(len(pts)):
        t = pts[idx]
        v = float(np.max(q + t * b + t * t * c))
        if v < best_v:
            best_v = v
            best_t = t
    for _ in range(2):
        bi = int(np.argmin([float(np.max(q + pts[i] * b + pts[i] * pts[i] * c)) for i in range(len(pts))]))
        lo_i = max(0, bi - 1)
        hi_i = min(len(pts) - 1, bi + 1)
        sub = np.linspace(pts[lo_i], pts[hi_i], 11)
        pts = np.sort(np.unique(np.concatenate([pts, sub])))
        for idx in range(len(pts)):
            t = pts[idx]
            v = float(np.max(q + t * b + t * t * c))
            if v < best_v:
                best_v = v
                best_t = t
    return best_t, best_v

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    n = 45
    fallback_h = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64).ravel()
        if 2 <= len(parent) <= 999:
            n = len(parent)
            if np.all(np.isfinite(parent)) and np.max(np.abs(parent)) <= 1e6:
                if np.all(parent >= 0) and np.all(parent <= 1) and abs(float(np.sum(parent)) - n / 2.0) <= 1e-10:
                    fallback_h = parent.copy()
                else:
                    p = project_onto_K(parent, n)
                    fallback_h = p if p is not None else np.full(n, 0.5)
            else:
                fallback_h = np.full(n, 0.5)
    except Exception:
        n = 45
        fallback_h = None
    if fallback_h is None:
        nk = kwargs.get('n_points', None)
        if isinstance(nk, int) and not isinstance(nk, bool) and 2 <= nk <= 999:
            n = nk
        fallback_h = np.full(n, 0.5)
    dx = 2.0 / n
    fallback_score = exact_score(fallback_h, n)
    best_h = fallback_h.copy()
    best_score = fallback_score
    starts = [fallback_h.copy()]
    try:
        starts.append(1.0 - fallback_h)
    except Exception:
        pass
    try:
        p = project_onto_K(fallback_h + 0.12 * rng.standard_normal(n), n)
        if p is not None:
            starts.append(p)
    except Exception:
        pass
    try:
        p = project_onto_K(rng.beta(0.5, 0.5, n), n)
        if p is not None:
            starts.append(p)
    except Exception:
        pass
    ded = [starts[0]]
    for s in starts[1:]:
        if not any(float(np.max(np.abs(s - d))) <= 1e-8 for d in ded):
            ded.append(s)
    max_starts = 4 if n <= 128 else (3 if n <= 256 else 2)
    ded = ded[:max_starts]
    tree = build_tree(n)
    max_sweeps = 3 if n <= 128 else (2 if n <= 256 else 1)
    for st in ded:
        if not check_feasible(st, n):
            continue
        cur_h = st.copy()
        cur_score = exact_score(cur_h, n)
        if cur_score < best_score - 1e-12:
            best_h = cur_h.copy()
            best_score = cur_score
        for sweep in range(max_sweeps):
            improved = False
            dirs = tree
            if len(dirs) > 64:
                sel = []
                for k in range(64):
                    idx = int(k * (len(dirs) - 1) / 63)
                    sel.append(dirs[idx])
                dirs = sel
            if len(dirs) > 64 and sweep % 2 == 1:
                dirs = dirs[::-1]
            for (dpth, a, m, b) in dirs:
                dv = make_dir(n, a, m, b)
                nz = np.abs(dv) > 1e-15
                if not np.any(nz):
                    continue
                idxs = np.where(nz)[0]
                tls, tus = [], []
                for ii in idxs:
                    if dv[ii] > 0:
                        tls.append(-cur_h[ii] / dv[ii])
                        tus.append((1.0 - cur_h[ii]) / dv[ii])
                    else:
                        tls.append((1.0 - cur_h[ii]) / dv[ii])
                        tus.append(-cur_h[ii] / dv[ii])
                t_lo = max(tls)
                t_hi = min(tus)
                if not (np.isfinite(t_lo) and np.isfinite(t_hi)):
                    continue
                if t_hi - t_lo <= 1e-12:
                    continue
                h1mh = 1.0 - cur_h
                qq = np.correlate(cur_h, h1mh, mode='full') * dx
                bb = dx * (np.correlate(dv, h1mh, mode='full') - np.correlate(cur_h, dv, mode='full'))
                cc = -dx * np.correlate(dv, dv, mode='full')
                if not (np.all(np.isfinite(qq)) and np.all(np.isfinite(bb)) and np.all(np.isfinite(cc))):
                    continue
                if n <= 48:
                    t_best, pred = line_min_exact(qq, bb, cc, t_lo, t_hi, n)
                else:
                    t_best, pred = line_min_grid(qq, bb, cc, t_lo, t_hi, n)
                if pred >= cur_score - 1e-12:
                    continue
                htr = cur_h + t_best * dv
                hr = repair_roundoff(htr, n)
                if hr is None:
                    continue
                ts = exact_score(hr, n)
                if ts < cur_score - 1e-12:
                    cur_h = hr.copy()
                    cur_score = ts
                    improved = True
                    if ts < best_score - 1e-12:
                        best_h = cur_h.copy()
                        best_score = ts
            if not improved:
                break
    final_h = best_h.copy()
    if not check_feasible(final_h, n):
        p = project_onto_K(final_h, n)
        final_h = p if p is not None else fallback_h.copy()
    final_score = exact_score(final_h, n)
    if final_score > fallback_score + 1e-12:
        final_h = fallback_h.copy()
        final_score = fallback_score
    return final_h.copy(), float(final_score), int(n)
