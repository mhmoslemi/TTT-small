    v = np.clip(v, 0.0, 1.0)  # initial clip
    # Bisection for tau: sum(clip(v - tau, 0, 1)) = S
    lo = np.min(v) - 1.0

            add = min(space, residual)
            y[i] += add

            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    residual = S - float(np.sum(y))
    if residual > 1e-12:
        for i in range(n):
            add = min(1.0 - y[i], residual)
            if add <= 0:
                continue
            y[i] += add
            residual -= add
            if abs(residual) < 1e-14:
                break
    elif residual < -1e-12:
        for i in range(n):
            take = min(y[i], -residual)
            if take <= 0:
                continue
            y[i] -= take
            residual += take
            if abs(residual) < 1e-14:
                break
    return y

def is_feasible(h, n):

def exact_score(h):
    n = len(h)
    if not np.all(np.isfinite(h)):
        return np.inf
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def compute_Q_grad(h):
    n = len(h)
    nlags = 2 * n - 1
    Q = np.zeros(nlags)
    G = np.zeros((nlags, n))
    for k in range(-(n - 1), n):
        idx = k + n - 1
        lo = max(0, k)
        hi = min(n - 1, n - 1 + k)
        if lo <= hi:
            i_arr = np.arange(lo, hi + 1)
            Q[idx] = float(np.dot(h[i_arr], 1.0 - h[i_arr - k]))
        if k >= 0:
            if k > 0:
                r1 = np.arange(k, n)
                G[idx, r1] += (1.0 - h[r1 - k])
            r2 = np.arange(0, n - k)
            G[idx, r2] -= h[r2 + k]
        else:
            m = -k
            r1 = np.arange(0, n - m)
            G[idx, r1] += (1.0 - h[r1 + m])
            r2 = np.arange(m, n)
            G[idx, r2] -= h[r2 - m]
    return Q, G
