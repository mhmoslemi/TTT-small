4. The j-th line on the hull is active from x_{j-1} to x_j (with x_0 = -inf, x_last = +inf).
5. Clip to [lo, hi].

Wait, for the upper hull with lines sorted by increasing slope, the hull is built from left to right. Line with smallest slope is active for small x, line with largest slope is active for large x. The intersection of line i and line i+1 on the hull gives the transition point.

Standard algorithm:
- Sort by slope (for equal slope, keep max intercept)
- Maintain a list of hull lines
- For each new line (m, c):
  - While the intersection of the new line with the last hull line is <= the intersection of the last two hull lines, pop the last hull line.
  - Add the new line.

The intersection of line (m1, c1) and (m2, c2) is x = (c1-c2)/(m2-m1) (the x where m1*x+c1 = m2*x+c2, so x = (c1-c2)/(m2-m1)).

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0 or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))

        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        s = float(np.sum(y))
        if s > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    res = S - float(np.sum(y))
    for i in range(n):
        if abs(res) < 1e-15:
            break

            add = min(res, 1.0 - y[i])
            y[i] += add
            res -= add
        else:
            sub = min(-res, y[i])
            y[i] -= sub
            res += sub
    for i in range(n):

def is_feasible(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(float(np.sum(h)) - n/2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score_and_corr(h):
    n = len(h)
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return None, None
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None, None

        return None, None
    corr = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(corr)):
        return None, None
    score = float(np.max(corr)) * (2.0 / n)
    if not np.isfinite(score):
        return None, None
    return score, corr

def quad_roots(a, b, c):
    roots = []
    if abs(a) < 1e-15:
        if abs(b) < 1e-15:
            return roots
        r = -c / b
        if np.isfinite(r):
            roots.append(r)
        return roots
    disc = b*b - 4.0*a*c
    tol = 64.0 * np.finfo(np.float64).eps * max(1.0, b*b, abs(4.0*a*c))
    if disc < -tol:
        return roots
    if disc < 0:
        disc = 0.0
    sq = math.sqrt(disc)
    if b >= 0:
        q = -0.5 * (b + sq)
    else:
        q = -0.5 * (b - sq)
    if abs(q) > 1e-30:
        roots.append(q / a)
        roots.append(c / q)
    else:
        roots.append(-b / (2.0 * a))
    roots = [r for r in roots if np.isfinite(r)]
    return roots

def make_pair_coeffs(h, corr, a, b):
    n = len(h)
    m = 2*n - 1
    A = corr.copy()
    B = np.zeros(m)
    D = np.zeros(m)
    D[a] = -2.0  # k=0 maps to index a=n-1
    # Actually k=0 maps to index n-1
    D[n-1] = -2.0
    # k = a-b maps to index (a-b)+(n-1)
    D[a-b+n-1] = 1.0
    # k = b-a maps to index (b-a)+(n-1)
    D[b-a+n-1] = 1.0
    for k in range(-(n-1), n):
        idx = k + n - 1
        bv = 0.0
        ia1 = a - k
        if 0 <= ia1 < n:
            bv += 1.0 - h[ia1]
        ia2 = a + k
        if 0 <= ia2 < n:
            bv -= h[ia2]
        ib1 = b - k
        if 0 <= ib1 < n:
            bv -= 1.0 - h[ib1]
        ib2 = b + k
        if 0 <= ib2 < n:
            bv += h[ib2]
        B[idx] = bv
    if not np.all(np.isfinite(B)) or not np.all(np.isfinite(A)):
        return None, None, None
    return A, B, D

def upper_hull(A, B, curved_mask, lo, hi):
    # Get non-curved lines
    lines = []
    for idx in range(len(A)):
        if not curved_mask[idx]:
            m = B[idx]
            c = A[idx]
            if np.isfinite(m) and np.isfinite(c):
                lines.append((m, c))
    if not lines:
        return [(lo, hi, None)]
    # Sort by slope, then by intercept descending for equal slopes
    lines.sort(key=lambda x: (x[0], -x[1]))
    # Remove duplicate slopes (keep max intercept)
    deduped = []
    for m, c in lines:
        if deduped and deduped[-1][0] == m:
            continue  # already kept the one with higher intercept
        deduped.append((m, c))
    lines = deduped
    # Build upper hull
    hull = []  # list of (m, c, start_x)
    for m, c in lines:
        if not hull:
            hull.append((m, c, -np.inf))
            continue
        # Intersection with last hull line
        lm, lc = hull[-1][0], hull[-1][1]
        if m == lm:
            continue
        x_int = (lc - c) / (m - lm)
        while len(hull) >= 2 and x_int <= hull[-1][2] + 1e-15:
            hull.pop()
            if not hull:
                break
            lm, lc = hull[-1][0], hull[-1][1]
            if m == lm:
                break
            x_int = (lc - c) / (m - lm)
        if hull:
            if m != hull[-1][0]:
                if not np.isfinite(x_int):
                    return None
                hull.append((m, c, x_int))
        else:
            hull.append((m, c, -np.inf))
    # Generate segments clipped to [lo, hi]
    segments = []
    for i, (m, c, x_start) in enumerate(hull):
        x_end = hull[i+1][2] if i+1 < len(hull) else np.inf
        s_lo = max(lo, x_start)
        s_hi = min(hi, x_end)
        if s_lo < s_hi - 1e-15:
            segments.append((s_lo, s_hi, (m, c)))
        elif s_lo == s_hi and lo <= s_lo <= hi:
            segments.append((s_lo, s_hi, (m, c)))
    if not segments:
        segments = [(lo, hi, hull[0][:2] if hull else None)]
    return segments

def line_minimize_pair(h, corr, current_score, a, b, n, score_budget):
    lo = max(-h[a], h[b] - 1.0)
    hi = min(1.0 - h[a], h[b])
    if hi - lo < 1e-14:
        return h, corr, current_score, 0
    A, B, D = make_pair_coeffs(h, corr, a, b)
    if A is None:
        return h, corr, current_score, 0
    # Identify curved indices
    curved = np.zeros(2*n-1, dtype=bool)
    curved[n-1] = True  # k=0
    curved[a-b+n-1] = True  # k=a-b
    curved[b-a+n-1] = True  # k=b-a
    curved_idx = [i for i in range(2*n-1) if curved[i]]
    segments = upper_hull(A, B, curved, lo, hi)
    if segments is None:
        return h, corr, current_score, 0
    # Generate candidates
    candidates = []
    eps = 1e-12 * max(1.0, abs(lo), abs(hi))
    for s_lo, s_hi, line in segments:
        # Endpoints
        candidates.append(s_lo)
        candidates.append(s_hi)
        # 0, length-1
        if lo <= 0.0 <= hi:
            if s_lo - eps <= 0.0 <= s_hi + eps:
                candidates.append(0.0)
        # Stationary points of curved quadratics
        for ci in curved_idx:
            if abs(D[ci]) < 1e-15:
                continue
            alpha_star = -B[ci] / (2.0 * D[ci])
            if s_lo - eps <= alpha_star <= s_hi + eps:
                candidates.append(alpha_star)
        # Intersections of curved with curved
        for i in range(len(curved_idx)):
            for j in range(i+1, len(curved_idx)):
                ci, cj = curved_idx[i], curved_idx[j]
                da = D[ci] - D[cj]
                ba = B[ci] - B[cj]
                ca = A[ci] - A[cj]
                roots = quad_roots(da, ba, ca)
                for r in roots:
                    if s_lo - eps <= r <= s_hi + eps:
                        candidates.append(r)
        # Intersections of curved with active line
        if line is not None:
            lm, lc = line
            for ci in curved_idx:
                da = D[ci]
                ba = B[ci] - lm
                ca = A[ci] - lc
                roots = quad_roots(da, ba, ca)
                for r in roots:
                    if s_lo - eps <= r <= s_hi + eps:
                        candidates.append(r)
    # Score candidates by polynomial objective
    scored = []
    for alpha in candidates:
        if alpha < lo - eps or alpha > hi + eps:
            continue
        alpha = max(lo, min(hi, alpha))
        # Compute polynomial objective at alpha
        # Find which segment alpha is in
        obj = -np.inf
        for s_lo, s_hi, line in segments:
            if s_lo - eps <= alpha <= s_hi + eps:
                if line is not None:
                    obj = max(obj, line[1] + line[0]*alpha)
                break
        for ci in curved_idx:
            val = A[ci] + B[ci]*alpha + D[ci]*alpha*alpha
            obj = max(obj, val)
        if np.isfinite(obj):
            scored.append((obj, alpha))
    scored.sort(key=lambda x: x[0])
    # Deduplicate alphas
    selected = []
    for _, alpha in scored:
        dup = False
        for sel_alpha in selected:
            if abs(alpha - sel_alpha) < 1e-12 * max(1.0, abs(alpha)):
                dup = True
                break
        if not dup:
            selected.append(alpha)
        if len(selected) >= 5:
            break
    # Rescore with exact method
    best_prop_h = None
    best_prop_corr = None
    best_prop_score = current_score
    costs = 0
    for alpha in selected:
        if costs >= score_budget:
            break
        if abs(alpha) < 1e-15:
            continue
        h_new = h.copy()
        h_new[a] += alpha
        h_new[b] -= alpha
        h_new[a] = max(0.0, min(1.0, h_new[a]))
        h_new[b] = max(0.0, min(1.0, h_new[b]))
        if not is_feasible(h_new, n):
            continue
        score, corr_new = exact_score_and_corr(h_new)
        costs += 1
        if score is not None and score < best_prop_score - 1e-12*max(1.0, abs(best_prop_score)):
            best_prop_score = score
            best_prop_h = h_new
            best_prop_corr = corr_new
    if best_prop_h is not None:
        return best_prop_h, best_prop_corr, best_prop_score, costs
    return h, corr, current_score, costs
