from scipy.optimize import minimize

def capped_simplex_project(v, target):
    n = len(v)
    if np.any(~np.isfinite(v)):
        return None
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - mid, 0.0, 1.0).copy()
    r = target - float(np.sum(y))
    if abs(r) > 1e-13:
        if r > 0:
            idx = np.where(1.0 - y > 1e-14)[0]
        else:
            idx = np.where(y > 1e-14)[0]
        if len(idx) > 0:
            y[idx] += r / len(idx)
            np.clip(y, 0.0, 1.0, out=y)
            r2 = target - float(np.sum(y))
            if abs(r2) > 1e-12:
                if r2 > 0:
                    idx2 = np.where(1.0 - y > 1e-14)[0]
                else:
                    idx2 = np.where(y > 1e-14)[0]
                if len(idx2) > 0:
                    y[idx2[0]] += r2 / len(idx2)
    if not np.all(np.isfinite(y)) or float(np.min(y)) < -1e-12 or float(np.max(y)) > 1.0 + 1e-12:
        return None
    if abs(float(np.sum(y)) - target) > 1e-8:
        return None
    return y

def exact_score(h, dx):
    if np.any(~np.isfinite(h)):
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def smooth_fg(h, tau, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(c))
    e = np.exp((c - m) / tau)
    se = float(np.sum(e))
    if se < 1e-300 or not np.isfinite(se):
        return (inf_val(), np.zeros(n))
    p = e / se
    val = m + tau * np.log(se)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-15:
            continue
        t = k - n + 1
        i1_lo, i1_hi = max(0, t), min(n, n + t)
        if i1_hi > i1_lo:
            i1 = np.arange(i1_lo, i1_hi)
            g[i1] += w * (1.0 - h[i1 - t])
        i2_lo, i2_hi = max(0, -t), min(n, n - t)
        if i2_hi > i2_lo:
            i2 = np.arange(i2_lo, i2_hi)
            g[i2] -= w * h[i2 + t]
    g *= dx
    if not np.all(np.isfinite(g)):
        return (inf_val(), np.zeros(n))
    return (val, g)

def inf_val():
    return np.inf

def mass_eq(h):
    return float(np.sum(h)) - len(h) / 2.0

def mass_jac(h):
    return np.ones(len(h))

def opt_schedule(h0, dx, target):
    n = len(h0)
    taus = [0.02, 0.005, 0.001, 0.0002, 0.00005, 0.00001]
    h = h0.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': mass_eq, 'jac': mass_jac}]
    for tau in taus:
        try:
            res = minimize(smooth_fg, h, method='SLSQP', args=(tau, dx),
                           jac=True, bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-10, 'disp': False})
            if res.x is not None and len(res.x) == n and np.all(np.isfinite(res.x)):

                    v_cur, _ = smooth_fg(h, tau, dx)
                    if np.isfinite(v_new) and v_new <= v_cur + 1e-10:
                        h = proj
        except Exception:
            pass
    return h
