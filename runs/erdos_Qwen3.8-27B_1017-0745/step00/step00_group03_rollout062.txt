def project_box_sum(z, N):
    if z is None:
        return np.full(N, 0.5)
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape[0] != N or not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    m = N / 2.0
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z))
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)
    e = m - h.sum()
    for _ in range(4):
        if abs(e) < 1e-14:
            break
        if e > 0:
            for i in range(N):
                adj = min(1.0 - h[i], e)
                h[i] += adj
                e -= adj
                if e <= 0:
                    break
        else:
            for i in range(N):
                adj = min(h[i], -e)
                h[i] -= adj
                e += adj
                if e >= 0:
                    break
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(h.sum()) - m) > 1e-10 * max(1, N):
        return np.full(N, 0.5)
    return h

def lag_values(h):
    N = len(h)
    q = np.zeros(2 * N - 1)
    for k in range(-(N - 1), N):
        idx = k + N - 1
        if k >= 0:
            L = N - k
            q[idx] = np.dot(h[k:], 1.0 - h[:L])
        else:
            d = -k
            L = N - d
            q[idx] = np.dot(h[:L], 1.0 - h[d:d + L])
    return q

def score_exact(h, N):

def smooth_value(h, tau, N):
    q = lag_values(h)
    r = (2.0 / N) * q
    rmax = float(np.max(r))
    return rmax + tau * float(np.log(np.exp((r - rmax) / tau).sum()))

def compute_smooth_grad(h, tau, N):
    q = lag_values(h)
    r = (2.0 / N) * q
    rmax = float(np.max(r))
    vals = np.exp((r - rmax) / tau)
    s = float(vals.sum())
    p = vals / s
    phi = rmax + tau * np.log(s)
    grad = np.zeros(N)
    for idx in range(2 * N - 1):
        w = (2.0 / N) * p[idx]
        if w < 1e-15:
            continue
        k = idx - N + 1
        if k >= 0:
            L = N - k
            grad[k:k + L] += w * (1.0 - h[:L])
            grad[:L] -= w * h[k:k + L]
        else:
            d = -k
            L = N - d
            grad[:L] += w * (1.0 - h[d:d + L])
            grad[d:d + L] -= w * h[:L]
    return phi, grad
