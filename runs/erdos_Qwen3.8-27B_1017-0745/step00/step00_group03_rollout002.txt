                    s_cur = s_new
                    if s_cur < inc_s:
                        inc, inc_s = h_new.copy(), s_cur
            except Exception:
                pass
    # Final validation
    if not check_feasible(inc, N):
        inc = project_box_sum(inc, N)
    inc = np.clip(inc, 0.0, 1.0)
    inc_s = score_exact(inc, N)
    if not np.isfinite(inc_s):
        inc = np.full(N, 0.5); inc_s = score_exact(inc, N)
    return inc, inc_s, N
