    v = np.clip(v, -1e6, 1e6)

        h = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > S:
            lo = mid
        else:

                    res -= add

                if rem > 0:

            if res.x is not None and len(res.x) == n + 1 and np.all(np.isfinite(res.x)):
                hx = project_simplex(res.x[:n], n, S)
                sc = exact_score(hx)
                if sc < best_score - 1e-12:
                    best_h, best_score = hx, sc
        except Exception:
            pass

    best_h = project_simplex(best_h, n, S)
    best_score = exact_score(best_h)
    return (best_h, best_score, n)
