def project_box_sum(z, N):
    if z is None or not hasattr(z, '__len__') or len(z) != N:
        return np.full(N, 0.5)
    z = np.asarray(z, dtype=np.float64)
    if not np.all(np.isfinite(z)) or np.max(np.abs(z)) > 1e6:
        return np.full(N, 0.5)
    m = N / 2.0
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        hc = np.clip(z - mid, 0.0, 1.0)
        if np.sum(hc) > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)

def lag_values(h):
    N = len(h)
    q = np.zeros(2 * N - 1)
    for k in range(N):
        L = N - k
        q[k + N - 1] = float(np.dot(h[k:k+L], 1.0 - h[0:L]))
    for d in range(1, N):
        L = N - d
        q[N - 1 - d] = float(np.dot(h[0:L], 1.0 - h[d:d+L]))
    return q

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / N))

    r = (2.0 / N) * q
    rmax = float(np.max(r))
    exps = np.exp((r - rmax) / tau)
    s = float(np.sum(exps))
    return rmax + tau * np.log(s)
