def lp_step(h, q, G, delta, dx):
    n = len(h)
    M = 2 * n - 1
    e = dx * delta * delta
    # Variables: p[0:n], u[0:n], t  => 2n+1 vars
    # Minimize t => c = [0,...,0, 1, 0,...,0]
    c = np.zeros(2 * n + 1)
    c[2 * n] = 1.0  # t is last variable
    
    # Inequalities A_ub @ x <= b_ub
    # Lag constraints: q_l + G_l . p + e <= t => G_l . p - t <= -q_l - e
    # Box: -h[i] <= p[i] <= 1-h[i] => p[i] <= 1-h[i], -p[i] <= h[i]
    # Trust: p[i] <= u[i], -p[i] <= u[i], sum u <= delta
    
    n_rows = M + 4 * n + 1
    A = np.zeros((n_rows, 2 * n + 1))
    b = np.zeros(n_rows)
    
    row = 0
    for l in range(M):
        A[row, :n] = G[l]
        A[row, 2 * n] = -1.0
        b[row] = -q[l] - e
        row += 1
    for i in range(n):
        A[row, i] = 1.0; A[row, n + i] = -1.0; b[row] = 0.0; row += 1
        A[row, i] = -1.0; A[row, n + i] = -1.0; b[row] = 0.0; row += 1
    for i in range(n):
        A[row, i] = 1.0; b[row] = 1.0 - h[i]; row += 1
        A[row, i] = -1.0; b[row] = h[i]; row += 1
    A[row, n:2*n] = 1.0; b[row] = delta; row += 1
    
    # Equality: sum p = 0
    A_eq = np.zeros((1, 2 * n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    bounds = [(None, None)] * n + [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'maxiter': 500, 'presolve': True})
    if res is None or not res.success or res.x is None:
        return None
    p = res.x[:n]
    if not np.all(np.isfinite(p)):
        return None
    return p

    candidates = [(best, best_score)]
    if parent_fallback is not None:
        candidates.append((parent_fallback, parent_score))
    candidates.append((const_fallback, const_score))
    candidates.sort(key=lambda x: x[1])
    h_final, sc_final = candidates[0]
    if not is_feasible(h_final, n):
        h_final = project_simplex(h_final, n)
    if h_final is None or not is_feasible(h_final, n):
        if parent_fallback is not None:
            h_final, sc_final = parent_fallback, parent_score
        else:
            h_final, sc_final = const_fallback, const_score
    sc_final = exact_score(h_final)
    h_final = np.asarray(h_final, dtype=np.float64)
    return (h_final, float(sc_final), n)
