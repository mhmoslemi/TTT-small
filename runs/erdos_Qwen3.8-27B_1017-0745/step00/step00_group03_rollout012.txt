import numpy as np
from scipy.optimize import minimize


def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(5):
        res = m - float(np.sum(h))
        if abs(res) < 1e-12:
            break
        if res > 0:
            for i in range(N):
                room = 1.0 - h[i]
                if room > 0:
                    add = min(room, res)
                    h[i] += add
                    res -= add
                    if abs(res) < 1e-12:
                        break
        else:
            for i in range(N):
                if h[i] > 0:
                    sub = min(h[i], -res)
                    h[i] -= sub
                    res += sub
                    if abs(res) < 1e-12:
                        break
    if not np.all(np.isfinite(h)) or np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)


def score(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / N)


def lag_values_and_jac(h):
    N = len(h)
    q = np.zeros(2 * N - 1)
    G = np.zeros((2 * N - 1, N))
    for idx in range(2 * N - 1):
        s = idx - (N - 1)
        if s >= 0:
            L = N - s
            a = h[0:L]
            b = h[s:s + L]
            q[idx] = float(np.sum(b * (1.0 - a)))
            G[idx, 0:L] -= b
            G[idx, s:s + L] += (1.0 - a)
        else:
            d = -s
            L = N - d
            a = h[0:L]
            b = h[d:d + L]
            q[idx] = float(np.sum(a * (1.0 - b)))
            G[idx, 0:L] += (1.0 - b)
            G[idx, d:d + L] -= a
    return q, G


def epigraph_obj(z, N, m):
    return float(z[N])


def epigraph_obj_grad(z, N, m):
    g = np.zeros(N + 1)
    g[N] = 1.0
    return g


def eq_fun(z, N, m):
    return float(np.sum(z[:N]) - m)


def eq_grad(z, N, m):
    g = np.zeros(N + 1)
    g[:N] = 1.0
    return g


def ineq_fun(z, N, m):
    h = z[:N]
    t = z[N]
    q, _ = lag_values_and_jac(h)
    return t - q


def ineq_grad(z, N, m):
    h = z[:N]
    _, G = lag_values_and_jac(h)
    g = np.zeros((2 * N - 1, N + 1))
    g[:, :N] = -G
    g[:, N] = 1.0
    return g


def run_slsqp(h0, N, m):
    q, _ = lag_values_and_jac(h0)
    t0 = float(np.max(q))
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * N + [(0.0, float(N))]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_grad},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_grad},
    ]
    try:
        res = minimize(epigraph_obj, z0, jac=epigraph_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons, args=(N, m),
                       options={'maxiter': 250, 'ftol': 1e-9})
        x = res.x
        if x is not None and len(x) == N + 1 and np.all(np.isfinite(x[:N])):
            return x[:N].copy()
    except Exception:
        pass
    return h0.copy()


def smooth_start(rng, N):
    n_modes = min(4, (N - 1) // 2)
    z = np.full(N, 0.5)
    idx = np.arange(N)
    for r in range(1, n_modes + 1):
        A = (0.12 / r) * rng.standard_normal()
        B = (0.12 / r) * rng.standard_normal()
        z += A * np.sin(2.0 * np.pi * r * idx / N) + B * np.cos(2.0 * np.pi * r * idx / N)
    return z


def run(seed=42, **kwargs):
    N = 70
    parent = None
    pv = globals().get('initial_h_values', None)
    if pv is not None:
        pv = np.asarray(pv, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 1:
            N = int(pv.size)
            if np.all(np.isfinite(pv)):
                parent = pv.copy()
    m = N / 2.0
    rng = np.random.default_rng(seed)

    if parent is not None:
        baseline = project_box_sum(parent, m)
    else:
        baseline = np.full(N, 0.5)

    best = baseline.copy()
    best_score = score(best, N)

    if N <= 2:
        return best, float(best_score), int(N)

    starts = [baseline]
    if parent is not None:
        starts.append(project_box_sum(1.0 - parent, m))
    else:
        starts.append(project_box_sum(rng.standard_normal(N), m))
    starts.append(np.full(N, 0.5))
    for sc in (0.05, 0.15, 0.35):
        starts.append(project_box_sum(baseline + sc * rng.standard_normal(N), m))
    starts.append(project_box_sum(rng.uniform(0.0, 1.0, N), m))
    starts.append(project_box_sum(smooth_start(rng, N), m))

    for h0 in starts:
        s = score(h0, N)
        if s < best_score:
            best_score = s
            best = h0.copy()
        res = run_slsqp(h0, N, m)
        res = project_box_sum(res, m)
        s = score(res, N)
        if s < best_score:
            best_score = s
            best = res.copy()

    for sc in (0.10, 0.05, 0.025, 0.0125):
        h0 = project_box_sum(best + sc * rng.standard_normal(N), m)
        s = score(h0, N)
        if s < best_score:
            best_score = s
            best = h0.copy()
        res = run_slsqp(h0, N, m)
        res = project_box_sum(res, m)
        s = score(res, N)
        if s < best_score:
            best_score = s
            best = res.copy()

    best = project_box_sum(best, m)
    if not np.all(np.isfinite(best)) or np.any(best < 0.0) or np.any(best > 1.0):
        best = np.full(N, 0.5)
    best_score = score(best, N)
    return best, float(best_score), int(N)
