    for r in range(1, R + 1):
        zf += (0.12 / r) * (rng.standard_normal() * np.sin(2 * np.pi * r * ii / N) + rng.standard_normal() * np.cos(2 * np.pi * r * ii / N))
    starts.append(project_box_sum(zf, N))
    if parent is not None:
        starts.append(project_box_sum(1.0 - baseline, N))
    else:
        starts.append(project_box_sum(0.5 + 0.4 * rng.standard_normal(N), N))
    starts.append(np.full(N, 0.5))
    starts.append(project_box_sum(baseline + 0.08 * rng.standard_normal(N), N))
    starts.append(project_box_sum(baseline + 0.40 * rng.standard_normal(N), N))
    if N <= 100:
        n_starts, ipp = 8, 60
    elif N <= 200:
        n_starts, ipp = 6, 40
    elif N <= 500:
        n_starts, ipp = 4, 20
    else:
        n_starts, ipp = 3, 8
    starts = starts[:n_starts]
    best_h = baseline.copy()
    best_score = score_exact(best_h, N)
    for s in starts:
        s = np.asarray(s, dtype=np.float64)
        if s.shape != (N,) or not np.all(np.isfinite(s)):
            continue
        s = project_box_sum(s, N)
        ss = score_exact(s, N)
        if np.isfinite(ss) and ss < best_score:
            best_h = s.copy()
            best_score = ss
        try:
            best_h, best_score = optimize(s, N, best_h, best_score, ipp)
        except Exception:
            pass
    best_h = np.asarray(best_h, dtype=np.float64)
    if best_h.shape != (N,) or not np.all(np.isfinite(best_h)):
        best_h = np.full(N, 0.5)
    if np.any(best_h < 0) or np.any(best_h > 1) or abs(np.sum(best_h) - m) > 1e-10 * max(1, N):
        best_h = project_box_sum(best_h, N)
    final_score = score_exact(best_h, N)
    if not np.isfinite(final_score):
        best_h = np.full(N, 0.5)
        final_score = score_exact(best_h, N)
    return best_h, float(final_score), int(N)
