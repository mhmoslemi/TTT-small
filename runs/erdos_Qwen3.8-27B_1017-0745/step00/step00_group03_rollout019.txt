    elif e < -1e-14:
        for i in range(N):
            if h[i] <= 0:
                continue
            sub = min(-e, h[i])
            h[i] -= sub
            e += sub
            if e >= -1e-14:
                break
    if not np.all(np.isfinite(h)) or np.min(h) < -1e-12 or np.max(h) > 1.0 + 1e-12:
        return np.full(N, 0.5)
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - m) > 1e-10 * max(1, N):
        return np.full(N, 0.5)
    return h

def lag_values(h):
    N = len(h)
    M = 2 * N - 1
    q = np.zeros(M)
    for k in range(-(N - 1), N):
        j = k + N - 1
        if k >= 0:
            L = N - k
            q[j] = np.dot(h[k:k + L], 1.0 - h[0:L])
        else:
            d = -k
            L = N - d
            q[j] = np.dot(h[0:L], 1.0 - h[d:d + L])
    return q

def smooth_value(h, tau, N):
    q = lag_values(h)
    r = (2.0 / N) * q
    rmax = np.max(r)
    ex = np.exp((r - rmax) / tau)
    return float(rmax + tau * np.log(np.sum(ex)))

def smooth_value_grad(h, tau, N):
    q = lag_values(h)
    r = (2.0 / N) * q
    rmax = np.max(r)
    ex = np.exp((r - rmax) / tau)
    s = np.sum(ex)
    p = ex / s
    phi = float(rmax + tau * np.log(s))
    grad = np.zeros(N)
    dx = 2.0 / N
    for k in range(-(N - 1), N):
        j = k + N - 1
        w = dx * p[j]
        if w < 1e-16:
            continue
        if k >= 0:
            L = N - k
            a = h[k:k + L]
            b = h[0:L]
            grad[k:k + L] += w * (1.0 - b)
            grad[0:L] -= w * a
        else:
            d = -k
            L = N - d
            a = h[0:L]
            b = h[d:d + L]
            grad[0:L] += w * (1.0 - b)
            grad[d:d + L] -= w * a
    return phi, grad

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / N))

def optimize_from(h_start, N, incumbent, inc_score, iters_per_temp):
    temps = [0.08, 0.02, 0.005, 0.001, 0.0002, 0.00004, 0.000008]
    h = h_start.copy()
    for tau in temps:
        stage_best = h.copy()
        stage_best_score = score_exact(h, N)
        alpha0 = 1.0
        for _ in range(iters_per_temp):
            phi_h, g = smooth_value_grad(h, tau, N)
            if not np.all(np.isfinite(g)):
                break
            alpha = alpha0
            accepted = False
            for _ in range(24):
                y = project_box_sum(h - alpha * g, N)
                d = y - h
                if np.max(np.abs(d)) < 1e-10:
                    break
                gd = float(np.dot(g, d))
                phi_y = smooth_value(y, tau, N)
                y_score = score_exact(y, N)
                if np.isfinite(y_score) and y_score < inc_score:
                    incumbent[:] = y
                    inc_score = y_score
                if np.isfinite(y_score) and y_score < stage_best_score:
                    stage_best_score = y_score
                    stage_best[:] = y
                if gd < 0 and phi_y <= phi_h + 1e-4 * gd + 1e-14:
                    h = y.copy()
                    alpha0 = min(1.4 * alpha, 10.0)
                    accepted = True
                    break
                alpha *= 0.5
                if alpha < 1e-8:
                    break
            if not accepted:
                break
        h = stage_best.copy()
    return incumbent, inc_score

def run(seed=42, **kwargs):
    N = 70
    baseline = None
    try:
        _ih = initial_h_values
        if _ih is not None:
            _arr = np.array(_ih, dtype=np.float64).ravel()
            if _arr.size > 0 and np.all(np.isfinite(_arr)):
                N = _arr.size
                baseline = np.clip(_arr, 0.0, 1.0)
    except Exception:
        pass
    if baseline is None:
        if 'n_points' in kwargs:
            try:
                _n = int(kwargs['n_points'])
                if 1 <= _n <= 999:
                    N = _n
            except Exception:
                pass
        baseline = np.full(N, 0.5)
    else:
        m = N / 2.0
        if abs(baseline.sum() - m) > 1e-8 or np.min(baseline) < -1e-12 or np.max(baseline) > 1.0 + 1e-12:
            baseline = project_box_sum(baseline, N)
    if N == 1:
        h1 = np.array([0.5])
        return h1, score_exact(h1, 1), 1
    inc = baseline.copy()
    inc_score = score_exact(inc, N)
    if N <= 100:
        n_starts, iters = 8, 60
    elif N <= 200:
        n_starts, iters = 6, 40
    elif N <= 500:
        n_starts, iters = 4, 20
    else:
        n_starts, iters = 3, 8
    rng = np.random.default_rng(int(seed) if seed is not None else 42)
    starts = [inc.copy(),
              project_box_sum(inc + 0.20 * rng.standard_normal(N), N),
              project_box_sum(rng.uniform(0.0, 1.0, N), N)]
    R = min(6, (N - 1) // 2)
    ii = np.arange(N, dtype=np.float64)
    four = np.full(N, 0.5)
    for r in range(1, R + 1):
        four += (0.12 / r) * (rng.standard_normal() * np.sin(2.0 * np.pi * r * ii / N)
                              + rng.standard_normal() * np.cos(2.0 * np.pi * r * ii / N))
    starts.append(project_box_sum(four, N))
    starts.append(project_box_sum(1.0 - inc, N))
    starts.append(np.full(N, 0.5).copy())
    starts.append(project_box_sum(inc + 0.08 * rng.standard_normal(N), N))
    starts.append(project_box_sum(inc + 0.40 * rng.standard_normal(N), N))
    starts = starts[:n_starts]
    for s in starts:
        s = project_box_sum(s, N)
        s_score = score_exact(s, N)
        if np.isfinite(s_score) and s_score < inc_score:
            inc[:] = s
            inc_score = s_score
        try:
            inc, inc_score = optimize_from(s, N, inc, inc_score, iters)
        except Exception:
            pass
    inc = np.clip(inc, 0.0, 1.0)
    m = N / 2.0
    if abs(inc.sum() - m) > 1e-10 * max(1, N):
        inc = project_box_sum(inc, N)
    final_score = score_exact(inc, N)
    if not np.isfinite(final_score):
        inc = np.full(N, 0.5)
        final_score = score_exact(inc, N)
    return inc, final_score, int(N)
