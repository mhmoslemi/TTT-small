import numpy as np
import math

def feasibility_check(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def project_capped_simplex(z, n):
    if z.shape != (n,) or not np.all(np.isfinite(z)) or np.any(np.abs(z) > 1e12):
        return None
    s = n / 2.0
    lo_b, hi_b = np.min(z) - 1.0, np.max(z)
    for _ in range(80):
        lam = 0.5 * (lo_b + hi_b)
        if np.sum(np.clip(z - lam, 0.0, 1.0)) > s:
            lo_b = lam
        else:
            hi_b = lam
    h = np.clip(z - 0.5 * (lo_b + hi_b), 0.0, 1.0)
    tol = 1e-10 * max(1, n)
    for _ in range(2 * n):
        r = s - np.sum(h)
        if abs(r) <= tol:
            break
        if r > 0:
            msk = h < 1.0 - 1e-15
            if not np.any(msk):
                break
            j = int(np.where(msk)[0][np.argmax(1.0 - h[msk])])
            h[j] = min(1.0, h[j] + r)
        else:
            msk = h > 1e-15
            if not np.any(msk):
                break
            j = int(np.where(msk)[0][np.argmax(h[msk])])
            h[j] = max(0.0, h[j] + r)
    h = np.clip(h, 0.0, 1.0)
    return h if feasibility_check(h, n) else None

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return float('inf')
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full')
    if len(q) != 2 * n - 1 or not np.all(np.isfinite(q)):
        return float('inf')
    return float(np.max(q) * (2.0 / n))

def haar_basis(n):
    if n <= 1:
        return np.zeros((0, n))
    V = np.zeros((n - 1, n))
    stack = [(0, n)]
    di = 0
    while stack:
        a, b = stack.pop()
        m = b - a
        if m < 2:
            continue
        mid = a + m // 2
        mL, mR = mid - a, b - mid
        V[di, a:mid] = math.sqrt(mR / (m * mL))
        V[di, mid:b] = -math.sqrt(mL / (m * mR))
        di += 1
        stack.append((a, mid))
        stack.append((mid, b))
    return V[:di]

def line_search(x, v, dx, n, pi, pj):
    lo, hi = float('inf'), float('-inf')
    for i in range(n):
        vi = v[i]
        if abs(vi) < 1e-15:
            continue
        r0, r1 = -x[i] / vi, (1.0 - x[i]) / vi
        lo = max(lo, min(r0, r1))
        hi = min(hi, max(r0, r1))
    if not (np.isfinite(lo) and np.isfinite(hi)):
        return x, exact_score(x, n)
    lo, hi = min(lo, 0.0), max(hi, 0.0)
    if hi - lo < 1e-14:
        return x, exact_score(x, n)
    omx = 1.0 - x
    cc = dx * np.correlate(x, omx, mode='full')
    bc = dx * (np.correlate(v, omx, mode='full') - np.correlate(x, v, mode='full'))
    ac = -dx * np.correlate(v, v, mode='full')
    if not (np.all(np.isfinite(ac)) and np.all(np.isfinite(bc)) and np.all(np.isfinite(cc))):
        return x, exact_score(x, n)
    tt = 1e-12 * max(1.0, abs(lo), abs(hi), abs(hi - lo))
    cands = [lo, hi, 0.0]
    for k in range(len(pi)):
        d, e = int(pi[k]), int(pj[k])
        ad, bd, cd = ac[d] - ac[e], bc[d] - bc[e], cc[d] - cc[e]
        sc_ = max(1.0, abs(ad), abs(bd), abs(cd))
        if abs(ad) <= 1e-14 * sc_:
            if abs(bd) > 1e-14 * sc_:
                t = -cd / bd
                if lo - tt <= t <= hi + tt:
                    cands.append(max(lo, min(hi, t)))
        else:
            disc = bd * bd - 4.0 * ad * cd
            dtol = 1e-14 * max(1.0, bd * bd, abs(4.0 * ad * cd))
            if disc < -dtol:
                continue
            disc = max(0.0, disc)
            sq = math.sqrt(disc)
            qr = -0.5 * (bd + math.copysign(sq, bd)) if abs(bd) > 1e-300 else -0.5 * (bd + sq)
            if abs(qr) > 1e-300:
                t1, t2 = qr / ad, cd / qr
            else:
                t1 = t2 = -bd / (2.0 * ad)
            for tv in (t1, t2):
                if lo - tt <= tv <= hi + tt:
                    cands.append(max(lo, min(hi, tv)))
    for d in range(len(ac)):
        if abs(ac[d]) > 1e-14 * max(1.0, abs(bc[d]), abs(cc[d])):
            t = -bc[d] / (2.0 * ac[d])
            if lo - tt <= t <= hi + tt:
                cands.append(max(lo, min(hi, t)))
    cands.sort()
    merged = [cands[0]]
    for t in cands[1:]:
        if t - merged[-1] > tt:
            merged.append(t)
    best_t, best_sc = 0.0, exact_score(x, n)
    pool_ts, pool_vs = [0.0], [best_sc]
    for ci in range(0, len(merged), 512):
        chunk = np.array(merged[ci:ci + 512])
        vals = np.max(ac[None, :] * chunk[:, None] ** 2 + bc[None, :] * chunk[:, None] + cc[None, :], axis=1)
        fm = np.isfinite(vals)
        if np.any(fm):
            order = np.argsort(vals)
            for oi in order[:min(8, len(order))]:
                if fm[oi]:
                    pool_ts.append(float(chunk[oi]))
                    pool_vs.append(float(vals[oi]))
    for t in pool_ts:
        if t == 0.0:
            continue
        y = x + t * v
        if not feasibility_check(y, n):
            y = project_capped_simplex(y, n)
            if y is None:
                continue
        sc = exact_score(y, n)
        if sc < best_sc - 1e-12:
            best_sc, best_t = sc, t
    if best_t != 0.0:
        return x + best_t * v, best_sc
    return x, best_sc

def clamp_int(val, default, mx):
    if val is None or isinstance(val, bool) or not isinstance(val, (int, np.integer)) or val < 1:
        return default
    return min(int(val), mx)

def run(seed=42, **kwargs):
    if not isinstance(seed, (int, float)) or isinstance(seed, bool):
        seed = 42
    seed = int(seed)
    n, base = 47, None
    try:
        ihv = initial_h_values
        if ihv is not None:
            arr = np.asarray(ihv, dtype=np.float64).ravel()
            if arr.size > 0:
                n = arr.size
                if np.all(np.isfinite(arr)) and np.all(arr >= -1e-8) and np.all(arr <= 1.0 + 1e-8):
                    arr = np.clip(arr, 0.0, 1.0)
                    if feasibility_check(arr, n):
                        base = arr.copy()
                    else:
                        base = project_capped_simplex(arr, n)
                if base is None:
                    base = np.full(n, 0.5)
    except (NameError, AttributeError, TypeError, ValueError):
        pass
    if base is None:
        nv = kwargs.get('n_points', None)
        if nv is not None and isinstance(nv, (int, np.integer)) and not isinstance(nv, bool) and 1 <= nv <= 999:
            n = int(nv)
        base = np.full(n, 0.5)
    fallback = base.copy()
    fb_sc = exact_score(fallback, n)
    if n == 1:
        u = np.full(1, 0.5)
        return (u, float(exact_score(u, 1)), 1)
    if n <= 48:
        ms, mw_ = 6, 3
    elif n <= 64:
        ms, mw_ = 3, 2
    else:
        ms, mw_ = 0, 0
    ns = clamp_int(kwargs.get('starts'), 5 if n <= 48 else 2, ms) if ms > 0 else 0
    nw = clamp_int(kwargs.get('sweeps'), 2 if n <= 48 else 1, mw_) if mw_ > 0 else 0
    rng = np.random.default_rng(seed)
    starts, dups = [], []
    raw_list = [base.copy(), base[::-1].copy(), 1.0 - base.copy(),
                base + rng.standard_normal(n) * 0.12, np.full(n, 0.5)]
    blk = np.zeros(n); blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    raw_list.append(blk)
    alt = np.zeros(n); alt[::2] = 1.0
    raw_list.append(alt)
    raw_list.append(base + rng.standard_normal(n) * 0.35)
    raw_list.append(rng.uniform(0.0, 1.0, n))
    raw_list.append(base + rng.standard_normal(n) * 0.45)
    for _ in range(30):
        raw_list.append(base + rng.standard_normal(n) * rng.uniform(0.05, 0.5))
    for raw in raw_list:
        hp = project_capped_simplex(raw, n)
        if hp is None:
            continue
        dup = False
        for dv in dups:
            if np.max(np.abs(hp - dv)) < 1e-10:
                dup = True
                break
        if dup:
            continue
        dups.append(hp)
        starts.append(hp)
        if len(starts) >= max(ns, 1):
            break
    while len(starts) < max(ns, 1):
        starts.append(base.copy())
    incumbent, inc_sc = fallback.copy(), fb_sc
    start_scores = []
    for st in starts:
        sc = exact_score(st, n)
        start_scores.append(sc)
        if sc < inc_sc - 1e-12:
            incumbent, inc_sc = st.copy(), sc
    if ns > 0 and n <= 64:
        V = haar_basis(n)
        m = 2 * n - 1
        pi = np.array([d for d in range(m) for e in range(d + 1, m)], dtype=np.int32)
        pj = np.array([e for d in range(m) for e in range(d + 1, m)], dtype=np.int32)
        dx = 2.0 / n
        for si in range(min(ns, len(starts))):
            x = starts[si].copy()
            lsc = start_scores[si]
            for sw in range(nw):
                rng_sw = np.random.default_rng(seed + sw * 1000 + si)
                perm = rng_sw.permutation(n - 1)
                imp = False
                for di in range(n - 1):
                    d = int(perm[di])
                    try:
                        xn, sn = line_search(x, V[d], dx, n, pi, pj)
                        if sn < lsc - 1e-12:
                            x, lsc, imp = xn, sn, True
                            if sn < inc_sc - 1e-12:
                                incumbent, inc_sc = x.copy(), sn
                    except Exception:
                        pass
                if not imp:
                    break
    sp = project_capped_simplex(incumbent, n)
    if sp is not None:
        ss = exact_score(sp, n)
        if np.isfinite(ss) and ss < fb_sc - 1e-12:
            incumbent, inc_sc = sp, ss
    if feasibility_check(incumbent, n):
        fs = exact_score(incumbent, n)
        if np.isfinite(fs) and fs < fb_sc - 1e-12:
            return (incumbent.astype(np.float64), float(fs), int(n))
    return (fallback.astype(np.float64), float(fb_sc), int(n))
