import numpy as np

def project_box_sum(z, N):
    m = 0.5 * N
    if len(z) != N or not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(4):
        e = m - np.sum(h)
        if abs(e) < 1e-13:
            break
        if e > 0:
            room = np.maximum(1.0 - h, 0.0)
            tr = np.sum(room)
            if tr < 1e-15:
                break
            h = np.minimum(h + e * room / tr, 1.0)
        else:
            av = np.maximum(h, 0.0)
            ta = np.sum(av)
            if ta < 1e-15:
                break
            h = np.maximum(h + e * av / ta, 0.0)
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(np.sum(h) - m) > 1e-10 * max(1, N):
        return np.full(N, 0.5)
    return h

def lag_values(h, N):
    M = 2 * N - 1
    q = np.zeros(M)
    for k in range(-(N - 1), N):
        L = N - abs(k)
        if k >= 0:
            q[k + N - 1] = np.dot(h[k:k + L], 1.0 - h[0:L])
        else:
            d = -k
            q[k + N - 1] = np.dot(h[0:L], 1.0 - h[d:d + L])
    return q

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / N))

def smooth_value(h, tau, N):
    q = lag_values(h, N)
    r = (2.0 / N) * q
    rmax = np.max(r)
    sh = (r - rmax) / tau
    ms = np.max(sh)
    return rmax + tau * (ms + np.log(np.sum(np.exp(sh - ms))))
