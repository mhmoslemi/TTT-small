        ns, nr, nb, nh = (8, 6, 4, 2) if main else (4, 3, 2, 1)
    elif N <= 250:
        ns, nr, nb, nh = (6, 4, 2, 1) if main else (3, 2, 1, 1)
    else:
        ns, nr, nb, nh = (3, 2, 1, 1) if main else (1, 1, 1, 0)
    dirs = []
    anchor = int(np.argmax(np.minimum(h, 1.0 - h)))
    others = [i for i in range(N) if i != anchor]
    paired = set()
    for j in range(min(ns, len(others))):
        idx = others[int(j * len(others) / ns)]
        d = np.zeros(N)
        d[idx], d[anchor] = 1.0, -1.0
        dirs.append(d)
        paired.add((min(anchor,idx), max(anchor,idx)))
    att = 0
    while len(dirs) < ns + nr and att < 200:
        att += 1
        i, j = sorted(rng.choice(N, size=2, replace=False))
        key = (i, j)
        if key in paired:
            continue
        paired.add(key)
        d = np.zeros(N)
        d[i], d[j] = 1.0, -1.0
        dirs.append(d)
    for _ in range(nb):
        sz = int(rng.integers(3, min(13, N+1)))
        if sz < 3:
            continue
        sup = rng.choice(N, size=sz, replace=False)
        vals = rng.standard_normal(sz)
        vals -= np.mean(vals)
        vals[sup[0]] -= np.sum(vals)
        d = np.zeros(N)
        d[sup] = vals
        nrm = np.max(np.abs(d))
        if nrm > 1e-14:
            dirs.append(d / nrm)
    mx = max(1, (N-1)//2)
    t = np.arange(N, dtype=np.float64)
    for c in range(min(nh, mx)):
        r = c + 1
        ph = rng.uniform(0, 2*math.pi)
        d = np.cos(2.0*math.pi*r*t/N + ph)
        d -= np.mean(d)
        d[0] -= np.sum(d)
        nrm = np.max(np.abs(d))
        if nrm > 1e-14:
            dirs.append(d / nrm)
    return dirs

def do_pass(h, N, rng, main):
    sc = score_exact(h, N)
    for d in get_directions(h, N, rng, main):
        try:
            nh, nsc = line_search(h, d, N, sc)
            if nsc < sc:
                h, sc = nh, nsc
        except Exception:
            pass
    return h, sc

def run(seed=42, **kwargs):
    N = 70
    parent = None
    try:
        pv = np.array(initial_h_values, dtype=np.float64).flatten()
        if len(pv) > 0:
            N = len(pv)
            if np.all(np.isfinite(pv)):
                parent = pv.copy()
    except Exception:
        pass
    if parent is None and 'n_points' in kwargs:
        try:
            N = int(kwargs['n_points'])
            N = max(1, min(999, N))
        except Exception:
            pass
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    rng = np.random.default_rng(seed)
    if parent is not None and is_feasible(parent, N):
        baseline = parent
    elif parent is not None:
        baseline = project_box_sum(parent, N)
    else:
        baseline = np.full(N, 0.5)
    incumbent = baseline.copy()
    inc_sc = score_exact(incumbent, N)

    starts = [baseline.copy(), np.full(N, 0.5)]
    if parent is not None:
        starts.append(project_box_sum(1.0 - parent, N))
    else:
        starts.append(project_box_sum(rng.uniform(0, 1, N), N))
    starts.append(project_box_sum(rng.uniform(0, 1, N), N))
    starts.append(project_box_sum(baseline + 0.2*rng.standard_normal(N), N))
    R = min(5, (N-1)//2)
    t = np.arange(N, dtype=np.float64)
    for _ in range(2):
        z = np.full(N, 0.5)
        for r in range(1, R+1):
            z += (0.12/r)*(rng.standard_normal()*np.sin(2*math.pi*r*t/N) + rng.standard_normal()*np.cos(2*math.pi*r*t/N))
        starts.append(project_box_sum(z, N))
    mxs = 7 if N <= 80 else (5 if N <= 250 else 3)
    ustarts = []
    for s in starts:
        s = project_box_sum(s, N)
        dup = any(np.max(np.abs(s - u)) < 1e-8 for u in ustarts)
        if not dup:
            ustarts.append(s)
        if len(ustarts) >= mxs:
            break
    results = []
    for st in ustarts:
        hh, ss = do_pass(st.copy(), N, rng, True)
        if ss < inc_sc:
            incumbent, inc_sc = hh.copy(), ss
        results.append((hh, ss))
    results.sort(key=lambda x: x[1])
    nel = 3 if N <= 80 else (2 if N <= 250 else 1)
    uel = []
    for eh, es in results:
        if not any(np.max(np.abs(eh - ue)) < 1e-8 for ue, _ in uel):
            uel.append((eh, es))
        if len(uel) >= nel:
            break
    for eh, es in uel:
        hh, ss = do_pass(eh.copy(), N, rng, False)
        if ss < inc_sc:
            incumbent, inc_sc = hh.copy(), ss
    if N <= 80:
        scales = [0.08, 0.20]
    elif N <= 250:
        scales = [0.12]
    else:
        scales = []
    for sc_ in scales:
        perm = project_box_sum(incumbent + sc_*rng.standard_normal(N), N)
        sh = score_exact(perm, N)
        h0 = perm if sh < inc_sc else incumbent.copy()
        s0 = min(sh, inc_sc)
        hh, ss = do_pass(h0, N, rng, False)
        if ss < inc_sc:
            incumbent, inc_sc = hh.copy(), ss
    if not is_feasible(incumbent, N):
        incumbent = project_box_sum(incumbent, N)
    if not is_feasible(incumbent, N):
        incumbent = np.full(N, 0.5)
    final = score_exact(incumbent, N)
    if not np.isfinite(final):
        incumbent = np.full(N, 0.5)
        final = score_exact(incumbent, N)
    return incumbent, float(final), int(N)
