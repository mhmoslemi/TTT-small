def random_zero_sum_direction(n, rng):
    while True:
        w = rng.standard_normal(n)
        w -= np.mean(w)
        w[n - 1] = -float(np.sum(w[:n - 1]))

            w = np.zeros(n)

            return w / np.sqrt(2.0)
