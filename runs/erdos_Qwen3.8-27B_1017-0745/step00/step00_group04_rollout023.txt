def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = float(np.max(corr * dx))
    if not math.isfinite(s):
        return None
    return s

def is_feasible(h):
    if h.ndim != 1 or not np.all(np.isfinite(h)):
        return False
    n = len(h)
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    S = n / 2.0
    if abs(float(np.sum(h)) - S) > 1e-10:
        return False
    if abs(math.fsum(float(x) for x in h) - S) > 1e-10:
        return False
    return True

def restore_simplex(v, n):

    v = np.clip(v.astype(np.float64), -1e6, 1e6)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))

def exact_score(h):
    n = len(h)
    if n < 2 or not np.all(np.isfinite(h)):
        return None
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = float(np.max(corr * dx))
    return s if math.isfinite(s) else None

def is_feasible(h):
    if h.ndim != 1 or len(h) < 2 or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return False
    S = len(h) / 2.0
    if abs(float(np.sum(h)) - S) > 1e-10:
        return False
    return True

def restore_simplex(v, n):
    S = n / 2.0
    v = np.clip(v.astype(np.float64), -1e6, 1e6)
    lo_b = float(np.min(v)) - 1.0
    hi_b = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo_b + hi_b)
        h = np.clip(v - mid, 0.0, 1.0)
        if float(np.sum(h)) > S:
            lo_b = mid
        else:
            hi_b = mid
    h = np.clip(v - 0.5 * (lo_b + hi_b), 0.0, 1.0)
    residual = S - math.fsum(float(x) for x in h)
    if residual > 1e-15:
        for idx in range(n):
            cap = 1.0 - h[idx]
            add = min(residual, cap)
            h[idx] += add
            residual -= add
            if residual <= 1e-15:
                break
    elif residual < -1e-15:
        res = -residual
        for idx in range(n):
            rem = min(res, h[idx])
            h[idx] -= rem
            res -= rem
            if res <= 1e-15:
                break
    s2 = math.fsum(float(x) for x in h)

            h[idx] += diff
            break
        elif diff < 0 and h[idx] >= -diff:
            h[idx] += diff
            break

def compute_lags(h):
    n = len(h)

    m = 2 * n - 1
    q = np.zeros(m)
    G = np.zeros((m, n))
    q[0] = dx * float(np.sum(h * (1.0 - h)))
    G[0, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        hq = h[:n - d]
        hp = h[d:]
        q[2 * d - 1] = dx * float(np.sum(hp * (1.0 - hq)))
        q[2 * d] = dx * float(np.sum(hq * (1.0 - hp)))
        G[2 * d - 1, d:] += dx * (1.0 - hq)
        G[2 * d - 1, :n - d] += -dx * hp
        G[2 * d, :n - d] += dx * (1.0 - hp)
        G[2 * d, d:] += -dx * hq
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):

def solve_quad(A, B, C, lo, hi):
    tol = 1e-14 * max(1.0, abs(A), abs(B), abs(C))
    roots = []
    if abs(A) <= tol:
        if abs(B) > tol:
            r = -C / B

    disc = B * B - 4.0 * A * C

        return roots
    if disc < 0:
        disc = 0.0

    z = -0.5 * (B + math.copysign(sq, B)) if abs(B) > 1e-30 else -sq

        s = int(seed)
    except Exception:
        s = 42
    rng = np.random.default_rng(s)
    constant_h = np.full(n, 0.5)
    const_sc = exact_score(constant_h)
    if const_sc is None:
        const_sc = 0.375
    incumbent = constant_h.copy()
    incumbent_sc = const_sc
    if parent_h is not None:
        pf_sc = exact_score(parent_h)
        if pf_sc is not None and pf_sc < incumbent_sc - 1e-12:
            incumbent, incumbent_sc = parent_h.copy(), pf_sc
    if n > 128:
        blk = np.zeros(n); blk[:n//2] = 1.0
        if n % 2 == 1: blk[n//2] = 0.5
        bs = exact_score(blk) if is_feasible(blk) else None
        if bs is not None and bs < incumbent_sc - 1e-12:
            incumbent, incumbent_sc = blk, bs
        fs = exact_score(incumbent)
        if fs is None:
            incumbent, fs = constant_h, const_sc
        return (incumbent.astype(np.float64), float(fs), int(n))
    starts = build_starts(n, rng, parent_h)
    if not starts:
        starts = [constant_h.copy()]
    pairs = get_pairs(n, rng)
    n_sweeps = 2 if n <= 64 else 1
    for start in starts:
        h_cur = start.copy()
        sc_cur = exact_score(h_cur)
        if sc_cur is None:
            continue
        if sc_cur < incumbent_sc - 1e-12:
            incumbent, incumbent_sc = h_cur.copy(), sc_cur
        if not pairs:
            continue
        for _sweep in range(n_sweeps):
            improved = False
            for (pi, pj) in pairs:
                res = pair_line_search(h_cur, pi, pj, sc_cur)
                if res is not None:
                    ns = exact_score(res)
                    if ns is not None and ns < sc_cur - 1e-12:
                        h_cur, sc_cur = res, ns
                        improved = True
                        if sc_cur < incumbent_sc - 1e-12:
                            incumbent, incumbent_sc = h_cur.copy(), sc_cur
            if not improved:
                break
    cands = [(incumbent, incumbent_sc), (constant_h, const_sc)]
    if parent_h is not None:
        ps = exact_score(parent_h)
        if ps is not None:
            cands.append((parent_h, ps))
    best_h, best_sc = cands[0]
    for ch, cs in cands[1:]:
        if cs < best_sc - 1e-12:
            best_h, best_sc = ch, cs
    if not is_feasible(best_h):
        rh = restore_simplex(best_h, n)
        best_h = rh if rh is not None else constant_h.copy()
    fs = exact_score(best_h)
    if fs is None:
        best_h = constant_h.copy()
        fs = const_sc
    return (best_h.astype(np.float64), float(fs), int(n))
