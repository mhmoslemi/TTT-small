def is_feasible(h, n, S):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-14) or np.any(h > 1 + 1e-14):
        return False
    if abs(np.sum(h) - S) > 1e-9:
        return False
    return True

def project_capped_simplex(z, S):
    n = len(z)
    if n == 0 or not np.all(np.isfinite(z)) or np.max(np.abs(z)) > 1e12:
        return None
    low = np.min(z) - 1.0
    high = np.max(z)
    for _ in range(80):
        mid = low + 0.5 * (high - low)
        y = np.clip(z - mid, 0.0, 1.0)
        if np.sum(y) > S:
            low = mid
        else:
            high = mid
    y = np.clip(z - 0.5*(low+high), 0.0, 1.0)
    res = S - np.sum(y)
    if abs(res) > 1e-14:
        idx = np.argsort(1.0 - y) if res > 0 else np.argsort(y)
        for i in idx:
            if res > 0:
                delta = min(res, 1.0 - y[i])
                y[i] += delta
                res -= delta
            else:
                delta = min(-res, y[i])
                y[i] -= delta
                res += delta
            if abs(res) < 1e-14:
                break
    if np.any(y < -1e-14) or np.any(y > 1.0 + 1e-14) or abs(np.sum(y) - S) > 1e-9:
        return None
    y = np.clip(y, 0.0, 1.0)
    return y

def exact_score(h, n):
    if not is_feasible(h, n, n/2.0):
        return None, None
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    s = float(np.max(vals))
    if not np.isfinite(s):
        return None, None
    raw = corr[::-1].copy()
    return s, raw

def quad_roots(A, B, C):
    scale = max(1.0, abs(B), abs(C))
    if abs(A) < 1e-14 * scale:
        if abs(B) < 1e-14 * scale:
            return []
        r = -C / B
        return [r] if np.isfinite(r) else []
    disc = B*B - 4.0*A*C
    tol = 1e-12 * scale * scale
    if disc < -tol:
        return []
    if disc < 0:
        disc = 0.0
    sq = math.sqrt(disc)
    if abs(B) < 1e-14 * scale:
        r1 = sq / (2.0*A)
        r2 = -sq / (2.0*A)
    elif B > 0:
        qv = -0.5*(B + sq)
        r1 = qv / A
        r2 = C / qv if qv != 0 else r1
    else:
        qv = -0.5*(B - sq)
        r1 = qv / A
        r2 = C / qv if qv != 0 else r1
    roots = []
    for r in [r1, r2]:
        if np.isfinite(r):
            roots.append(r)
    return roots

def pair_line_min(h, raw, n, p, q):

    if abs(B) < 1e-14 * scale:
        r1, r2 = sq / (2.0 * A), -sq / (2.0 * A)
    elif B > 0:
        qv = -0.5 * (B + sq)
        r1, r2 = qv / A, C / qv
    else:
        qv = -0.5 * (B - sq)
        r1, r2 = qv / A, C / qv
    return [r for r in (r1, r2) if np.isfinite(r)]

def build_hull(b_arr, c_arr):
    m = len(b_arr)
    if m == 0:
        return None
    order = np.argsort(b_arr)
    fb, fc = [], []
    for i in order:
        b, c = float(b_arr[i]), float(c_arr[i])
        if fb and abs(b - fb[-1]) < 1e-15:
            if c > fc[-1]:
                fc[-1] = c
        else:
            fb.append(b)
            fc.append(c)
    hb, hc, hs = [], [], []
    for i in range(len(fb)):
        b, c = fb[i], fc[i]
        while hb:
            db = b - hb[-1]
            if abs(db) < 1e-15:
                if c > hc[-1]:
                    hc[-1] = c
                break
            x = (hc[-1] - c) / db
            if len(hs) > 1 and x <= hs[-1] + 1e-15:
                hb.pop(); hc.pop(); hs.pop()
            else:
                break
        if not hb:
            hb.append(b); hc.append(c); hs.append(-np.inf)
        else:
            db = b - hb[-1]
            if abs(db) < 1e-15:
                if c > hc[-1]:
                    hc[-1] = c
            else:
                x = (hc[-1] - c) / db
                if not np.isfinite(x):
                    x = np.inf if x > 0 else -np.inf
                hb.append(b); hc.append(c); hs.append(x)
    return np.array(hb), np.array(hc), np.array(hs)

def eval_envelope(t, a_map, quad_d, b_arr, c_arr, half, hull):
    val = -1e30
    if hull is not None:
        hb, hc, hs = hull
        idx = int(np.searchsorted(hs, t, side='right')) - 1
        idx = max(0, min(idx, len(hb) - 1))
        val = max(val, hb[idx] * t + hc[idx])
    for d in quad_d:
        a = a_map.get(d, 0.0)
        val = max(val, a * t * t + b_arr[d + half] * t + c_arr[d + half])
    return val

def pair_line_min(h, raw, n, p, q):
    t_lo = max(-h[p], h[q] - 1.0)
    t_hi = min(1.0 - h[p], h[q])
    if not (np.isfinite(t_lo) and np.isfinite(t_hi)) or t_hi - t_lo < 1e-14:
        return None
    gap = q - p
    half = n - 1
    n_lags = 2 * n - 1
    b_arr = np.zeros(n_lags)
    for d in range(-half, half + 1):
        lo = max(0, -d)
        hi = min(n, n - d)
        gp = 0.0
        if lo <= p < hi:
            gp += 1.0 - h[p + d]
        if lo + d <= p < hi + d:
            gp -= h[p - d]
        gq = 0.0
        if lo <= q < hi:
            gq += 1.0 - h[q + d]
        if lo + d <= q < hi + d:
            gq -= h[q - d]
        b_arr[d + half] = gp - gq
    a_map = {0: -2.0, gap: 1.0, -gap: 1.0}
    quad_d = [0, gap, -gap]
    cands = [t_lo, t_hi, 0.0]
    af_b, af_c = [], []
    for idx in range(n_lags):
        d = idx - half
        if d in a_map:
            continue
        af_b.append(b_arr[idx])
        af_c.append(raw[idx])
    hull = build_hull(np.array(af_b), np.array(af_c)) if af_b else None
    if hull is not None:
        hb, hc, hs = hull
        for i in range(len(hs)):
            if t_lo < hs[i] < t_hi:
                cands.append(hs[i])
        for seg in range(len(hb)):
            s_lo = t_lo if seg == 0 else max(t_lo, hs[seg])
            s_hi = t_hi if seg == len(hb) - 1 else min(t_hi, hs[seg + 1])
            if s_lo >= s_hi:
                continue
            for d in quad_d:
                a = a_map.get(d, 0.0)
                bq = b_arr[d + half]
                cq = raw[d + half]
                for r in quad_roots(a, bq - hb[seg], cq - hc[seg]):
                    if s_lo - 1e-10 <= r <= s_hi + 1e-10:
                        cands.append(max(t_lo, min(t_hi, r)))
    for i in range(3):
        for j in range(i + 1, 3):
            d1, d2 = quad_d[i], quad_d[j]
            a1, a2 = a_map.get(d1, 0), a_map.get(d2, 0)
            b1, b2 = b_arr[d1 + half], b_arr[d2 + half]
            c1, c2 = raw[d1 + half], raw[d2 + half]
            for r in quad_roots(a1 - a2, b1 - b2, c1 - c2):
                if t_lo <= r <= t_hi:
                    cands.append(r)
    for d in quad_d:
        a = a_map.get(d, 0)
        if a > 0:
            v = -b_arr[d + half] / (2.0 * a)
            if t_lo <= v <= t_hi:
                cands.append(v)
    best_t, best_f = 0.0, eval_envelope(0.0, a_map, quad_d, b_arr, raw, half, hull)
    for t in cands:
        if t < t_lo - 1e-10 or t > t_hi + 1e-10:
            continue
        t = max(t_lo, min(t_hi, t))
        f = eval_envelope(t, a_map, quad_d, b_arr, raw, half, hull)
        if f < best_f - 1e-14 or (abs(f - best_f) <= 1e-14 and abs(t) < abs(best_t)):
            best_f, best_t = f, t
    if abs(best_t) < 1e-14:
        return None
    return best_t

def do_sweep(pairs, h_cur, score_cur, raw_cur, n, budget, best_h, best_score):
    accepted = 0
    for (p, q) in pairs:
        if budget[0] >= budget[1]:
            break
        budget[0] += 1
        t = pair_line_min(h_cur, raw_cur, n, p, q)
        if t is None:
            continue
        h_trial = h_cur.copy()
        ps = h_cur[p] + h_cur[q]
        p_new = np.clip(h_cur[p] + t, max(0.0, ps - 1.0), min(1.0, ps))
        q_new = ps - p_new
        h_trial[p] = p_new
        h_trial[q] = q_new
        sc, raw_new = exact_score(h_trial, n)
        if sc is None:
            continue
        if sc < best_score[0] - 1e-12:
            best_score[0] = sc
            best_h[:] = h_trial
        if sc < score_cur - 1e-13 * max(1.0, abs(score_cur)):
            h_cur[:] = h_trial
            score_cur = sc
            raw_cur[:] = raw_new
            accepted += 1
    return score_cur, accepted

def run(seed=42, **kwargs):
    if seed is None:
        seed = 42
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    rng = np.random.default_rng(seed)
    n, h_parent = None, None
    if 'initial_h_values' in globals():
        try:
            iv = np.array(globals()['initial_h_values'], dtype=np.float64)
            if iv.ndim == 1 and len(iv) >= 1:
                n = len(iv)
                h_parent = iv.copy()
        except Exception:
            pass
    if n is None:
        n = kwargs.get('n_points', 80)
        if not isinstance(n, int) or n < 8 or n > 512:
            n = 80
    S = n / 2.0
    h_const = np.full(n, 0.5)
    h_fallback = None
    if h_parent is not None and np.all(np.isfinite(h_parent)):
        if is_feasible(h_parent, n, S):
            h_fallback = h_parent.copy()
        else:
            pr = project_capped_simplex(h_parent, S)
            if pr is not None:
                h_fallback = pr
    if n <= 1:
        sc = exact_score(h_const, n)[0]
        return h_const, sc, n
    best_h = h_const.copy()
    best_score = exact_score(h_const, n)[0]
    if h_fallback is not None:
        sf = exact_score(h_fallback, n)
        if sf[0] is not None and sf[0] < best_score:
            best_score = sf[0]
            best_h = h_fallback.copy()
    if n > 128:
        return best_h, best_score, n
    starts = [best_h.copy()]
    if h_fallback is not None and not np.allclose(h_fallback, h_const):
        starts.append(h_fallback.copy())
        starts.append(h_fallback[::-1].copy())
        pr = project_capped_simplex(h_fallback + 0.12 * rng.standard_normal(n), S)
        if pr is not None:
            starts.append(pr)
    q = int(rng.integers(1, min(9, max(1, n // 2)) + 1))
    phi = float(rng.uniform(0, 2 * math.pi))
    z = 0.5 + 0.45 * np.cos(2 * math.pi * q * np.arange(n) / n + phi)
    pr = project_capped_simplex(z, S)
    if pr is not None:
        starts.append(pr)
    perm = rng.permutation(n)
    z_bin = np.zeros(n)
    z_bin[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        z_bin[perm[n // 2]] = 0.5
    if is_feasible(z_bin, n, S):
        starts.append(z_bin)
    unique_starts = []
    for s in starts:
        if not is_feasible(s, n, S):
            continue
        dup = any(np.max(np.abs(s - us)) < 1e-10 for us in unique_starts)
        if not dup:
            unique_starts.append(s)
            sc = exact_score(s, n)[0]
            if sc is not None and sc < best_score:
                best_score = sc
                best_h = s.copy()
    unique_starts = unique_starts[:6]
    Btot = min(8000, max(500, int(64000000.0 / (n * n))))
    n_pairs_full = n * (n - 1) // 2
    do_full_first = n_pairs_full <= Btot
    budget = [n_pairs_full if do_full_first else 0, Btot]
    best_score_mutable = [best_score]
    best_h_mutable = best_h.copy()
    if do_full_first:
        all_pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        rng.shuffle(all_pairs)
        h_cur = best_h_mutable.copy()
        sc, raw_cur = exact_score(h_cur, n)
        sc, acc = do_sweep(all_pairs, h_cur, sc, raw_cur, n, budget, best_h_mutable, best_score_mutable)
        if sc < best_score_mutable[0]:
            best_score_mutable[0] = sc
            best_h_mutable[:] = h_cur
        if acc > 0 and budget[0] + n_pairs_full <= Btot:
            rng.shuffle(all_pairs)
            sc, raw_cur = exact_score(h_cur, n)
            sc, acc2 = do_sweep(all_pairs, h_cur, sc, raw_cur, n, budget, best_h_mutable, best_score_mutable)
            if sc < best_score_mutable[0]:
                best_score_mutable[0] = sc
                best_h_mutable[:] = h_cur
    best_score = best_score_mutable[0]
    best_h = best_h_mutable.copy()
    offsets = set()
    p2 = 1
    while p2 <= n // 2:
        offsets.add(p2)
        p2 *= 2
    if n // 2 > 0:
        offsets.add(n // 2)
    sparse_pairs, seen = [], set()
    for off in sorted(offsets):
        for i in range(n):
            j = (i + off) % n
            if i != j:
                pair = (min(i, j), max(i, j))
                if pair not in seen:
                    seen.add(pair)
                    sparse_pairs.append(pair)
    sparse_pairs.sort()
    for start in unique_starts[:6]:
        if budget[0] >= Btot:
            break
        h_cur = start.copy()
        sc, raw_cur = exact_score(h_cur, n)
        if sc is None:
            continue
        stagnant = 0
        for _ in range(3):
            if budget[0] >= Btot:
                break
            sp = sparse_pairs[:]
            rng.shuffle(sp)
            sc, acc = do_sweep(sp, h_cur, sc, raw_cur, n, budget, best_h, best_score_mutable)
            if sc < best_score_mutable[0]:
                best_score_mutable[0] = sc
                best_h[:] = h_cur
            if acc == 0:
                stagnant += 1
                if stagnant >= 2:
                    break
            else:
                stagnant = 0
    best_score = best_score_mutable[0]
    pool = [best_h.copy()]
    if h_fallback is not None:
        pool.append(h_fallback.copy())
    pool.append(h_const.copy())
    final_h, final_s = None, float('inf')
    for cand in pool:
        if not is_feasible(cand, n, S):
            pr = project_capped_simplex(cand, S)
            if pr is None:
                continue
            cand = pr
        sc, _ = exact_score(cand, n)
        if sc is not None and sc < final_s:
            final_s, final_h = sc, cand.copy()
    if final_h is None:
        final_h = h_const.copy()
        final_s = exact_score(final_h, n)[0]
    return final_h, final_s, n
