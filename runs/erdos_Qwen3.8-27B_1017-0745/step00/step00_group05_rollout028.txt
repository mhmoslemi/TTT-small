import numpy as np
import math

def capped_simplex_project(v, target):
    n = len(v)
    if n == 0:
        return None
    if not np.all(np.isfinite(v)):
        return None
    if np.max(np.abs(v)) > 1e12:
        return None
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - np.sum(y)
    for _ in range(3):
        if abs(res) < 1e-15:
            break
        if res > 0:
            for i in range(n):
                if res < 1e-15:
                    break
                add = min(res, 1.0 - y[i])
                y[i] += add
                res -= add
        else:
            for i in range(n):
                if -res < 1e-15:
                    break
                sub = min(-res, y[i])
                y[i] -= sub
                res += sub
    if not np.all(np.isfinite(y)) or np.min(y) < -1e-12 or np.max(y) > 1.0 + 1e-12:
        return None
    if abs(np.sum(y) - target) > 1e-9:
        return None
    return np.clip(y, 0.0, 1.0)

def is_feasible(h, target):
    if h.ndim != 1:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.min(h) < 0.0 or np.max(h) > 1.0:
        return False
    if abs(np.sum(h) - target) > 1e-10:
        return False
    return True

def repair_mass(h, target):
    n = len(h)
    err = target - np.sum(h)
    if abs(err) <= 1e-10:
        return h
    if abs(err) > 1e-9:
        return None
    for _ in range(3):
        if abs(err) < 1e-14:
            break
        if err > 0:
            for i in range(n):
                if err < 1e-14:
                    break
                a = min(err, 1.0 - h[i])
                h[i] += a
                err -= a
        else:
            for i in range(n):
                if -err < 1e-14:
                    break
                a = min(-err, h[i])
                h[i] -= a
                err += a
    if not np.all(np.isfinite(h)) or np.min(h) < -1e-12 or np.max(h) > 1.0 + 1e-12:
        return None
    if abs(np.sum(h) - target) > 1e-10:
        return None
    return h

def exact_score(h, dx):
    if h.ndim != 1 or not np.all(np.isfinite(h)):
        return None
    corr = np.correlate(h, 1.0 - h, mode="full")
    s = float(np.max(corr) * dx)
    if not np.isfinite(s):
        return None
    return s

def propose_move(h, rho, rng, n):
    try:
        mt = rng.random()
        amp = rng.uniform(-rho, rho)
        if abs(amp) < 1e-15:
            return None
        if mt < 0.45:
            a = int(rng.integers(n))
            u = int(rng.integers(n - 1))
            b = u if u < a else u + 1
            d = np.zeros(n)
            d[a] = 1.0
            d[b] = -1.0
        elif mt < 0.70:
            kvals = sorted(set([min(4, n), min(8, n), min(16, n)]))
            k = int(rng.choice(kvals))
            idx = rng.choice(n, size=k, replace=False)
            sv = rng.standard_normal(k)
            sv -= np.mean(sv)
            sv[-1] = -np.sum(sv[:-1])
            mx = np.max(np.abs(sv))
            if not np.isfinite(mx) or mx < 1e-12:
                a = int(rng.integers(n))
                u = int(rng.integers(n - 1))
                b = u if u < a else u + 1
                d = np.zeros(n)
                d[a] = 1.0
                d[b] = -1.0
            else:
                d = np.zeros(n)
                d[idx] = sv / mx
        elif mt < 0.90:
            blens = [x for x in [4, 8, 16, 32] if x <= n]
            bl = int(rng.choice(blens)) if blens else n
            start = int(rng.integers(n - bl + 1))
            d = np.zeros(n)
            half = bl // 2
            d[start:start + half] = 1.0
            d[start + bl - half:start + bl] = -1.0
        else:
            kk = int(rng.integers(1, min(8, n - 1) + 1))
            d = np.cos(np.pi * kk * (np.arange(n) + 0.5) / n)
            d -= np.mean(d)
            d[int(rng.integers(n))] = -np.sum(d)
            mx = np.max(np.abs(d))
            if not np.isfinite(mx) or mx < 1e-12:
                a = int(rng.integers(n))
                u = int(rng.integers(n - 1))
                b = u if u < a else u + 1
                d = np.zeros(n)
                d[a] = 1.0
                d[b] = -1.0
            else:
                d = d / mx
        cand = h + amp * d
        if np.min(cand) < 0.0 or np.max(cand) > 1.0:
            return None
        cand = repair_mass(cand.copy(), float(n) / 2.0)
        if cand is None or not is_feasible(cand, float(n) / 2.0):
            return None
        return cand
    except Exception:
        return None

def make_block_profile(n, rng):
    arr = np.zeros(n)
    half = n // 2
    arr[:half] = 1.0
    if n % 2 == 1:
        arr[half] = 0.5
    shift = int(rng.integers(n))
    arr = np.roll(arr, shift)
    return arr

def build_initial_states(n, rng, parent, fallback):
    target = float(n) / 2.0
    states = []
    profiles = []
    if parent is not None:
        profiles.append(parent.copy())
        profiles.append(parent[::-1].copy())
        profiles.append(parent + rng.standard_normal(n) * 0.025)
        profiles.append(parent + rng.standard_normal(n) * 0.08)
        profiles.append(1.0 - parent)
        profiles.append(rng.random(n))
        profiles.append(rng.beta(0.5, 0.5, n))
        profiles.append(make_block_profile(n, rng))
    else:
        profiles.append(fallback.copy())
        profiles.append(make_block_profile(n, rng))
        profiles.append(rng.random(n))
        profiles.append(rng.beta(0.5, 0.5, n))
        profiles.append(rng.beta(0.5, 0.5, n))
        profiles.append((rng.random(n) > 0.5).astype(float))
        profiles.append(fallback + rng.standard_normal(n) * 0.25)
        profiles.append(rng.random(n))
    for p in profiles:
        if is_feasible(p, target):
            states.append(p.copy())
        else:
            proj = capped_simplex_project(p, target)
            if proj is None:
                states.append(fallback.copy())
            else:
                states.append(proj)
    return states[:8]

def run(seed=42, **kwargs):

    except Exception:
        rng = np.random.default_rng(42)

    if final is None or not is_feasible(final, target):
        final = fallback.copy()
    fs = exact_score(final, dx)
    if fs is None:
        fs = exact_score(fallback, dx)
    if fs is None:
        fs = float('inf')
    return (final, fs, n)
