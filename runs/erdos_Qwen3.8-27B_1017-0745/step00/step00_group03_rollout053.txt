    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    e = m - np.sum(h)
    idx = np.arange(N)
    for _ in range(2):
        for ii in idx:
            if abs(e) < 1e-15:
                break
            if e > 0:
                adj = min(e, 1.0 - h[ii])
                h[ii] += adj
                e -= adj
            else:
                adj = min(-e, h[ii])
                h[ii] -= adj
                e += adj
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(np.sum(h) - m) > 1e-10 * max(1, N):
        return uniform
    return h

def lag_values(h, N):
    M = 2 * N - 1
    q = np.zeros(M)
    for k in range(-(N - 1), N):
        idx = k + N - 1
        if k >= 0:
            L = N - k
            a = h[k:k + L]
            b = h[0:L]
            q[idx] = np.dot(a, 1.0 - b)
        else:
            d = -k
            L = N - d
            a = h[0:L]
            b = h[d:d + L]
            q[idx] = np.dot(a, 1.0 - b)
    return q

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / N))

def smooth_value(h, tau, N):
    q = lag_values(h, N)
    r = (2.0 / N) * q
    rmax = np.max(r)
    if not np.isfinite(rmax):
        return np.inf
    s = np.sum(np.exp((r - rmax) / tau))
    if s <= 0 or not np.isfinite(s):
        return np.inf
    return rmax + tau * np.log(s)

def smooth_value_grad(h, tau, N):
    q = lag_values(h, N)
    r = (2.0 / N) * q
    rmax = np.max(r)
    s = np.sum(np.exp((r - rmax) / tau))
    if s <= 0 or not np.isfinite(s):
        return np.inf, np.zeros(N)
    phi = rmax + tau * np.log(s)
    p = np.exp((r - rmax) / tau) / s
    grad = np.zeros(N)
    for k in range(-(N - 1), N):
        idx = k + N - 1
        w = (2.0 / N) * p[idx]
        if abs(w) < 1e-30:
            continue
        if k >= 0:
            L = N - k
            a = h[k:k + L]
            b = h[0:L]
            grad[k:k + L] += w * (1.0 - b)
            grad[0:L] -= w * a
        else:
            d = -k
            L = N - d
            a = h[0:L]
            b = h[d:d + L]
            grad[0:L] += w * (1.0 - b)
            grad[d:d + L] -= w * a
    return phi, grad

def run_opt(h0, N, inx, ins, iters):
    temps = [0.08, 0.02, 0.005, 0.001, 0.0002, 0.00004, 0.000008]
    h = h0.copy()
    for tau in temps:
        sb = h.copy()
        sbs = score_exact(sb, N)
        alpha_carry = 1.0
        for it in range(iters):
            phi_h, g = smooth_value_grad(h, tau, N)
            if not np.isfinite(phi_h) or not np.all(np.isfinite(g)):
                break
            alpha = alpha_carry
            accepted = False
            tiny = False
            for trial in range(24):
                y = project_box_sum(h - alpha * g, N)
                d = y - h
                if np.max(np.abs(d)) < 1e-10:
                    break
                gd = np.dot(g, d)
                phi_y = smooth_value(y, tau, N)
                sy = score_exact(y, N)
                if sy < ins:
                    ins = sy
                    inx[:] = y
                if sy < sbs:
                    sbs = sy
                    sb[:] = y
                if gd < 0 and np.isfinite(phi_y) and phi_y <= phi_h + 1e-4 * gd + 1e-14:
                    h[:] = y
                    alpha_carry = min(1.4 * alpha, 10.0)
                    accepted = True
                    if np.max(np.abs(d)) < 1e-8:
                        tiny = True
                    break
                alpha *= 0.5
                if alpha < 1e-8:
                    break
            if not accepted or tiny:
                break
        h = sb.copy()
    return inx, ins

def run(seed=42, **kwargs):
    N = 70
    has_parent = False
    parent = None
    try:
        iv = initial_h_values
        if iv is not None:
            iv = np.asarray(iv, dtype=np.float64).ravel()
            if iv.size > 0 and np.all(np.isfinite(iv)):
                N = iv.size
                parent = iv.copy()
                has_parent = True
    except Exception:
        has_parent = False
    if not has_parent and 'n_points' in kwargs:
        try:
            vp = int(kwargs['n_points'])
            if 1 <= vp <= 999:
                N = vp
        except Exception:
            pass
    try:
        s = int(seed)
    except Exception:
        s = 42
    rgen = np.random.default_rng(s)
    if N == 1:
        h1 = np.array([0.5])
        return h1, score_exact(h1, 1), 1
    m = N / 2.0
    if has_parent:
        baseline = parent.copy()
        if (np.min(baseline) < -1e-10 or np.max(baseline) > 1.0 + 1e-10
                or abs(np.sum(baseline) - m) > 1e-8 * max(1, N)):
            baseline = project_box_sum(baseline, N)
    else:
        baseline = np.full(N, 0.5)
    inx = baseline.copy()
    ins = score_exact(inx, N)
    if N <= 100:
        n_starts, iters = 8, 60
    elif N <= 200:
        n_starts, iters = 6, 40
    elif N <= 500:
        n_starts, iters = 4, 20
    else:
        n_starts, iters = 3, 8
    starts = [baseline.copy()]
    starts.append(project_box_sum(baseline + 0.20 * rgen.standard_normal(N), N))
    starts.append(project_box_sum(rgen.random(N), N))
    R = min(6, (N - 1) // 2)
    four = np.full(N, 0.5)
    ii = np.arange(N)
    for r in range(1, R + 1):
        amp = 0.12 / r
        four += amp * rgen.standard_normal() * np.sin(2.0 * np.pi * r * ii / N)
        four += amp * rgen.standard_normal() * np.cos(2.0 * np.pi * r * ii / N)
    starts.append(project_box_sum(four, N))
    if has_parent:
        starts.append(project_box_sum(1.0 - baseline, N))
    else:
        starts.append(project_box_sum(np.full(N, 0.5) + 0.4 * rgen.standard_normal(N), N))
    starts.append(np.full(N, 0.5))
    starts.append(project_box_sum(baseline + 0.08 * rgen.standard_normal(N), N))
    starts.append(project_box_sum(baseline + 0.40 * rgen.standard_normal(N), N))
    starts = starts[:n_starts]
    for st in starts:
        try:
            sc = score_exact(st, N)
            if sc < ins:
                ins = sc
                inx = st.copy()
            inx2 = inx.copy()
            ins2 = ins
            inx3, ins3 = run_opt(st.copy(), N, inx2, ins2, iters)
            if ins3 < ins:
                ins = ins3
                inx = inx3.copy()
        except Exception:
            pass
    final = project_box_sum(inx, N)
    sfinal = score_exact(final, N)
    if not np.isfinite(sfinal):
        final = np.full(N, 0.5)
        sfinal = score_exact(final, N)
    return final, sfinal, N
