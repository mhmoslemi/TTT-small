def slsqp_obj(z):
    return z[-1]

def slsqp_obj_jac(z):
    n = len(z) - 1
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def slsqp_ineq(z):
    n = len(z) - 1
    h = z[:n]
    t = z[n]
    Q, _ = compute_lags_and_grad(h)
    return t - (2.0 / n) * Q

def slsqp_ineq_jac(z):
    n = len(z) - 1
    h = z[:n]
    _, G = compute_lags_and_grad(h)
    n_lags = 2 * n - 1
    J = np.zeros((n_lags, n + 1))
    J[:, :n] = -(2.0 / n) * G
    J[:, n] = 1.0
    return J

def slsqp_eq(z):
    n = len(z) - 1
    return np.sum(z[:n]) / n - 0.5

def slsqp_eq_jac(z):
    n = len(z) - 1
    J = np.zeros(n + 1)

    if not np.all(np.isfinite(y)):
        return None
    if np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    if abs(float(np.sum(y)) - S) > 1e-10 * max(1.0, n):
        return None
    return y

def check_feasible(h, n):
    if h is None or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(float(np.sum(h)) - n / 2.0) > 1e-10 * max(1.0, n):
        return False
    return True

def exact_score(h):
    n = len(h)
    if n < 2 or not np.all(np.isfinite(h)):
        return np.inf
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_lags_and_grad(h):
    n = len(h)
    n_lags = 2 * n - 1
    Q = np.zeros(n_lags)
    G = np.zeros((n_lags, n))
    for ks in range(n_lags):
        k = ks - (n - 1)
        lo_i = max(0, k)
        hi_i = min(n - 1, n - 1 + k)
        for i in range(lo_i, hi_i + 1):
            j = i - k
            Q[ks] += h[i] * (1.0 - h[j])
            G[ks, i] += (1.0 - h[j])
            G[ks, j] -= h[i]
    return Q, G

def slsqp_obj(z):
    return z[-1]

def slsqp_obj_jac(z):
    m = len(z)
    g = np.zeros(m)
    g[-1] = 1.0
    return g

def slsqp_ineq(z):
    n = len(z) - 1
    h = z[:n]
    t = z[n]
    Q, _ = compute_lags_and_grad(h)
    return t - (2.0 / n) * Q

def slsqp_ineq_jac(z):
    n = len(z) - 1
    h = z[:n]
    _, G = compute_lags_and_grad(h)
    n_lags = 2 * n - 1
    J = np.zeros((n_lags, n + 1))
    J[:, :n] = -(2.0 / n) * G
    J[:, n] = 1.0
    return J

def slsqp_eq(z):
    n = len(z) - 1
    return float(np.sum(z[:n])) / n - 0.5

def slsqp_eq_jac(z):
    n = len(z) - 1
    J = np.zeros(n + 1)
    J[:n] = 1.0 / n
    return J

def run_slsqp(h0, n, bounds, cons):
    S = n / 2.0
    Q0, _ = compute_lags_and_grad(h0)
    t0 = min(1.0, float(np.max(Q0)) * 2.0 / n + 1e-4)
    z0 = np.concatenate([h0, [t0]])
    res = minimize(slsqp_obj, z0, jac=slsqp_obj_jac, method='SLSQP',
                  bounds=bounds, constraints=cons,
                  options={'maxiter': 300, 'ftol': 1e-10, 'disp': False})
    if res.x is not None and len(res.x) == n + 1 and np.all(np.isfinite(res.x)):
        return res.x[:n]
    return None

def run(seed=42, **kwargs):
    n = 45
    parent = None
    if 'initial_h_values' in globals():
        try:
            p = np.array(initial_h_values, dtype=np.float64).ravel()
            if (2 <= len(p) <= 999 and np.all(np.isfinite(p))
                    and np.all(p >= -1e-6) and np.all(p <= 1.0 + 1e-6)
                    and abs(float(np.sum(p)) - len(p) / 2.0) < 1e-4):
                n = len(p)
                p = np.clip(p, 0.0, 1.0)
                pr = project_box_sum(p, n / 2.0)
                if pr is not None and check_feasible(pr, n):
                    parent = pr
        except Exception:
            parent = None

    S = n / 2.0
    best_h = parent.copy() if parent is not None else np.full(n, 0.5)
    best_score = exact_score(best_h)
    S2 = 2.0 / n
    rng = np.random.default_rng(seed)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]

    starts = [best_h.copy()]
    if parent is not None:
        starts.append(parent[::-1].copy())
        for sd in [0.01, 0.03, 0.06, 0.12, 0.20, 0.05]:
            pert = parent + sd * rng.standard_normal(n)
            pr = project_box_sum(pert, S)
            if pr is not None:
                starts.append(pr)
        for _ in range(4):
            pr = project_box_sum(rng.uniform(0, 1, n), S)
            if pr is not None:
                starts.append(pr)
    else:
        for _ in range(8):
            pr = project_box_sum(rng.uniform(0, 1, n), S)
            if pr is not None:
                starts.append(pr)
        for A, m, ph in [(0.25, 1, 0.37), (0.35, 2, 0.74), (0.45, 3, 1.11)]:
            u = 0.5 + A * np.cos(2.0 * np.pi * m * np.arange(n) / n + ph)
            pr = project_box_sum(u, S)
            if pr is not None:
                starts.append(pr)

    for h0 in starts:
        try:
            hc = run_slsqp(h0, n, bounds, cons)
            if hc is not None:
                hc = project_box_sum(hc, S)
                if hc is not None and check_feasible(hc, n):
                    sc = exact_score(hc)
                    if sc < best_score:
                        best_score = sc
                        best_h = hc.copy()
        except Exception:
            pass

    sigma = 0.05
    for _ in range(4):
        improved = False
        try:
            pert = best_h + sigma * rng.standard_normal(n)
            h0 = project_box_sum(pert, S)
            if h0 is not None and check_feasible(h0, n):
                hc = run_slsqp(h0, n, bounds, cons)
                if hc is not None:
                    hc = project_box_sum(hc, S)
                    if hc is not None and check_feasible(hc, n):
                        sc = exact_score(hc)
                        if sc < best_score - 1e-12:
                            best_score = sc
                            best_h = hc.copy()
                            improved = True
        except Exception:
            pass
        if improved:
            sigma = max(0.01, 0.7 * sigma)
        else:
            sigma = min(0.25, 1.5 * sigma)

    final_h = project_box_sum(best_h, S)
    if final_h is None or not check_feasible(final_h, n):
        final_h = parent.copy() if parent is not None else np.full(n, 0.5)
    final_score = exact_score(final_h)
    return (final_h, final_score, n)
