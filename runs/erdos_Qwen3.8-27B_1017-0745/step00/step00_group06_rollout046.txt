    add_start(starts, fallback + 0.12 * make_smooth(rng.standard_normal(N)), N)
    if len(starts) < max_starts:
        for tr in range(32):
            if len(starts) >= max_starts:
                break
            sc = [0.05, 0.10, 0.18, 0.28][tr % 4]
            add_start(starts, fallback + sc * make_smooth(rng.standard_normal(N)), N)
    starts = starts[:max_starts]

    for s in starts:
        c = exact_c5(s, N)
        if c < incumbent_score - 1e-12:
            incumbent, incumbent_score = s.copy(), c

    d = 2.0 / N
    for start_vec in starts:
        x = start_vec.copy()
        Cx = exact_c5(x, N)
        if Cx < incumbent_score - 1e-12:
            incumbent, incumbent_score = x.copy(), Cx
        eta_base, eta_min, no_imp = 0.20, 1e-4, 0
        for _ in range(max_iter):
            if no_imp >= 5 and eta_base <= eta_min * (1 + 1e-12):
                break
            R = np.correlate(x, 1.0 - x, mode='full') * d
            Cx = float(np.max(R))
            eps_a = max(1e-12, 1e-9 * max(1.0, abs(Cx)))
            order = np.argsort(R)[::-1]
            active, seen = [], set()
            for r in order:
                if Cx - R[r] <= eps_a and r not in seen:
                    active.append(int(r))
                    seen.add(int(r))
                if len(active) >= 8:
                    break
            if not active:
                active = [int(order[0])]
            for r in order[:2]:
                if int(r) not in seen:
                    active.append(int(r))
                    seen.add(int(r))
            G_list = [lag_gradient(x, r - (N - 1), d, N) for r in active[:8]]
            if not G_list:
                continue
            G = np.array(G_list)
            gbar = frank_wolfe_min_norm(G)
            directions = []
            for dc in [-gbar] + [-G_list[i] for i in range(min(2, len(G_list)))]:
                u = normalize_dir(dc)
                if u is None:
                    continue
                dup = False
                for ex in directions:
                    if np.max(np.abs(u - ex)) < 1e-8:
                        dup = True
                        break
                if not dup:
                    directions.append(u)
            if (no_imp > 0 and no_imp % 3 == 0) or not directions:
                for sg in [1, -1]:
                    u_e = normalize_dir(sg * make_smooth(rng.standard_normal(N)))
                    if u_e is None:
                        continue
                    dup = False
                    for ex in directions:
                        if np.max(np.abs(u_e - ex)) < 1e-8:
                            dup = True
                            break
                    if not dup:
                        directions.append(u_e)
            best_trial, best_score, best_eta = None, Cx, 0.0
            for u in directions:
                for s2 in range(5):
                    eta_s = max(eta_min, eta_base / (2.0 ** s2))
                    y = project_capped_simplex(x + eta_s * u, N)
                    if y is None or not is_feasible(y, N) or np.max(np.abs(y - x)) <= 1e-12:
                        continue
                    cy = exact_c5(y, N)
                    if np.isfinite(cy) and cy < best_score - 1e-12:
                        best_trial, best_score, best_eta = y, cy, eta_s
            if best_trial is not None:
                x, Cx = best_trial, best_score
                if Cx < incumbent_score - 1e-12:
                    incumbent, incumbent_score = x.copy(), Cx
                no_imp = 0
                eta_base = min(0.25, 1.25 * eta_base) if best_eta >= 0.75 * eta_base else max(eta_min, 1.25 * best_eta)
            else:
                eta_base = max(eta_min, 0.5 * eta_base)
                no_imp += 1

    x = project_capped_simplex(incumbent, N)
    if x is None or not is_feasible(x, N):
        x = fallback
    c5 = exact_c5(x, N)
    c5_fb = exact_c5(fallback, N)
    if c5_fb < c5 - 1e-12:
        x, c5 = fallback, c5_fb
    if c5 == float('inf'):
        x = np.full(N, 0.5)
        c5 = exact_c5(x, N)
    return (x, float(c5), N)
