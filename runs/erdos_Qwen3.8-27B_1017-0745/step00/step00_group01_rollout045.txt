import numpy as np

def check_feasible(h, n):
    h = np.asarray(h, dtype=np.float64)
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def project_capped_simplex(z, n):
    z = np.asarray(z, dtype=np.float64)
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return np.full(n, 0.5, dtype=np.float64)
    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(z - mid, 0.0, 1.0)
        if np.sum(proj) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) < 1e-12:
            break
        if r > 0:
            slack = 1.0 - h
            idx = int(np.argmax(slack))
            amt = min(r, slack[idx])
            if amt < 1e-15:
                break
            h[idx] += amt
            r -= amt
        else:
            idx = int(np.argmin(h))
            amt = min(-r, h[idx])
            if amt < 1e-15:
                break
            h[idx] -= amt
            r += amt
    h = np.clip(h, 0.0, 1.0)
    if not check_feasible(h, n):
        return np.full(n, 0.5, dtype=np.float64)
    return h

def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64)
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return np.inf
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return np.inf
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def f_tau_and_weights(h, n, tau):
    c = compute_corr(h, n)
    c_max = float(np.max(c))
    a = np.exp((c - c_max) / tau)
    total = float(np.sum(a))
    if not np.all(np.isfinite(a)) or total <= 0.0:
        return np.inf, None
    p = a / total
    val = c_max + tau * np.log(total)
    if not np.isfinite(val):
        return np.inf, None
    return val, p

def f_tau_value(h, n, tau):
    v, _ = f_tau_and_weights(h, n, tau)
    return v

def grad_f_tau(h, n, p):
    dx = 2.0 / n
    grad = np.zeros(n, dtype=np.float64)
    for u in range(2 * n - 1):
        d = u - (n - 1)
        w = p[u]
        if w < 1e-18:
            continue
        m = n - abs(d)
        ks = np.arange(max(d, 0), max(d, 0) + m)
        grad[ks] += w * (1.0 - h[ks - d])
        ks2 = np.arange(max(-d, 0), max(-d, 0) + m)
        grad[ks2] -= w * h[ks2 + d]
    grad *= dx
    if not np.all(np.isfinite(grad)):
        return None
    return grad

def build_starts(n, parent, rng, max_starts):
    if parent is not None and np.all(np.isfinite(parent)):
        base = project_capped_simplex(parent, n)
    else:
        base = np.full(n, 0.5, dtype=np.float64)
    rev = base[::-1].copy()
    omr = 1.0 - rev
    uniform = np.full(n, 0.5, dtype=np.float64)
    block = np.zeros(n, dtype=np.float64)
    half = n // 2
    block[:half] = 1.0
    if n % 2 == 1:
        block[half] = 0.5
    block = project_capped_simplex(block, n)
    alt = np.zeros(n, dtype=np.float64)
    alt[::2] = 1.0
    alt = project_capped_simplex(alt, n)
    n1 = project_capped_simplex(base + 0.05 * rng.standard_normal(n), n)
    n2 = project_capped_simplex(base + 0.15 * rng.standard_normal(n), n)
    n3 = project_capped_simplex(base + 0.35 * rng.standard_normal(n), n)
    ru = project_capped_simplex(rng.uniform(0.0, 1.0, size=n), n)
    cands = [base, rev, omr, uniform, block, alt, n1, n2, n3, ru]
    starts = []
    for c in cands:
        ph = project_capped_simplex(c, n)
        dup = False
        for ex in starts:
            if np.max(np.abs(ph - ex)) < 1e-10:
                dup = True
                break
        if not dup:
            starts.append(ph)
        if len(starts) >= max_starts:
            break
    return starts

def optimize_start(x, n, taus, max_iters):
    for tau in taus:
        alpha = 1.0
        f_val, p = f_tau_and_weights(x, n, tau)
        if p is None or not np.isfinite(f_val):
            continue
        grad = grad_f_tau(x, n, p)
        if grad is None:
            continue
        stall = 0
        f_prev = f_val
        for _ in range(max_iters):
            y = project_capped_simplex(x - alpha * grad, n)
            D = float(np.sum((y - x) ** 2))
            if np.max(np.abs(y - x)) <= 1e-9:
                break
            f_y = f_tau_value(y, n, tau)
            if not np.isfinite(f_y):
                alpha *= 0.5
                if alpha < 1e-12:
                    break
                continue
            thresh = f_val - 1e-4 * D / alpha + 1e-14 * max(1.0, abs(f_val))
            if f_y <= thresh:
                x = y
                f_val = f_y
                c = compute_corr(x, n)
                cmax = float(np.max(c))
                a = np.exp((c - cmax) / tau)
                tot = float(np.sum(a))
                if tot <= 0 or not np.all(np.isfinite(a)):
                    break
                p = a / tot
                grad = grad_f_tau(x, n, p)
                if grad is None:
                    break
                alpha = min(2.0, 1.25 * alpha)

                    stall += 1
                    if stall >= 5:
                        break
                else:
                    stall = 0
                f_prev = f_val
            else:
                alpha *= 0.5
                if alpha < 1e-12:
                    break
    return x
