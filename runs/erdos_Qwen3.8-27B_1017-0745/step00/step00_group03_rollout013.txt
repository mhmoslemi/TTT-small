def project_box_sum(z, N):

def smooth_value_grad(h, tau, N):
    dx = 2.0 / N
    q = lag_values(h, N)
    M = 2 * N - 1
    r = dx * q
    rmax = np.max(r)
    ex = np.exp((r - rmax) / tau)
    s = np.sum(ex)
    p = ex / s
    phi = rmax + tau * np.log(s)
    g = np.zeros(N, dtype=np.float64)
    for k in range(0, N):
        L = N - k

        b = h[0:L]

            inc_h = s.copy()
        try:
            inc_h, inc_score = optimize_start(s, N, inc_h, inc_score, iters)
        except Exception:
            pass
    final = project_box_sum(inc_h, N) if inc_h.shape == (N,) else np.full(N, 0.5, dtype=np.float64)
    c5 = score_exact(final, N)
    if not np.isfinite(c5):
        final = np.full(N, 0.5, dtype=np.float64)
        c5 = score_exact(final, N)
    return final, float(c5), int(N)
