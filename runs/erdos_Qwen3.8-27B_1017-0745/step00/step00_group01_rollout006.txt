    n = len(h)
    if not check_feasible(h, n):
        return float('inf')
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(c))

def smooth_obj_and_weights(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = dx * np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(c))
    a = np.exp((c - c_max) / tau)
    s = float(np.sum(a))
    F = c_max + tau * np.log(s)
    p = a / s  # length 2n-1, p[u] corresponds to lag u-(n-1)
    return F, p, c

def smooth_gradient(h, p):
    n = len(h)
    dx = 2.0 / n
    # p has length 2n-1, p[u] = p_lag where lag = u - (n-1)
    P = p  # already in the right order
    # U = conv(P, 1-h)[n-1 : 2n-1]
    # V = conv(reverse(P), h)[n-1 : 2n-1]
    U = np.convolve(P, 1.0 - h, mode='full')[n-1:2*n-1]
    V = np.convolve(P[::-1], h, mode='full')[n-1:2*n-1]
    return dx * (U - V)

def build_starts(base, n, seed):

    starts.append(base.copy())
    # reversal

    # uniform

    k = n // 2

    if n % 2 == 1:

    starts.append(blk)
    # alternating

    alt[::2] = 1.0
    # need to project to fix mass

    # noisy versions

def projected_gradient_desc(x, n, tau_schedule, max_iter, max_backtrack):
    best_x = x.copy()
    best_score = exact_score(x)
    for tau in tau_schedule:
        alpha = 1.0
        for it in range(max_iter):
            F, p, c = smooth_obj_and_weights(x, tau)
            if not np.isfinite(F):
                break
            grad = smooth_gradient(x, p)
            if not np.all(np.isfinite(grad)):
                break
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-9:
                break
            # Armijo backtracking
            accepted = False
            for bt in range(max_backtrack):
                y = project_capped_simplex(x - alpha * grad, n)
                D = float(np.sum((y - x) ** 2))
                if max(np.max(np.abs(y - x))) <= 1e-9:
                    break
                Fy, _, _ = smooth_obj_and_weights(y, tau)
                armijo_rhs = F - 1e-4 * D / alpha + 1e-14 * max(1.0, abs(F))
                if np.isfinite(Fy) and Fy <= armijo_rhs:
                    accepted = True
                    break
                alpha *= 0.5
                if alpha < 1e-12:
                    break
            if not accepted:
                break
            # evaluate exact score
            sy = exact_score(y)
            if np.isfinite(sy) and sy < best_score - 1e-12:
                best_score = sy
                best_x = y.copy()
            x = y
            alpha = min(2.0, 1.25 * alpha)
