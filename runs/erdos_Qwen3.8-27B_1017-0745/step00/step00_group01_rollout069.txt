import numpy as np

def _feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if h.min() < -1e-12 or h.max() > 1.0 + 1e-12:
        return False
    return abs(h.sum() - 0.5 * n) <= 1e-10 * max(1.0, n)

def _project(z, n):
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return np.full(n, 0.5, dtype=np.float64)
    target = 0.5 * n
    z = np.clip(z, -1e12, 1e12)
    lo, hi = z.min() - 1.0, z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(n + 2):
        r = target - h.sum()
        if abs(r) <= 1e-14 * max(1.0, n):
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            step = min(r, 1.0 - h[idx])
            if step <= 0:
                break
            h[idx] += step
        else:
            idx = int(np.argmax(h))
            step = min(-r, h[idx])
            if step <= 0:
                break
            h[idx] -= step
    h = np.clip(h, 0.0, 1.0)
    if not _feasible(h, n):
        return np.full(n, 0.5, dtype=np.float64)
    return h

def _score(h, n):
    if not _feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _svfg(h, n, tau):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = c.max()
    a = np.exp((c - cm) / tau)
    s = a.sum()
    F = cm + tau * np.log(s)
    p = a / s
    U = np.convolve(p, 1.0 - h, mode='full')[n - 1:2 * n - 1]
    V = np.convolve(p[::-1], h, mode='full')[n - 1:2 * n - 1]
    g = dx * (U - V)
    if not np.all(np.isfinite(g)) or not np.isfinite(F):
        return None, None
    return F, g

def _pd_stage(h, n, tau, max_iters):
    alpha = 1.0
    stag = 0
    for _ in range(max_iters):
        F, g = _svfg(h, n, tau)
        if F is None:
            break
        accepted = False
        for _ in range(24):
            y = _project(h - alpha * g, n)
            D = float(np.sum((y - h) ** 2))
            if float(np.max(np.abs(y - h))) <= 1e-9:
                return h
            Fy, _ = _svfg(y, n, tau)
            if Fy is None:
                alpha *= 0.5
                continue
            if Fy <= F - 1e-4 * D / alpha + 1e-14 * max(1.0, abs(F)):
                if abs(Fy - F) <= 1e-12 * max(1.0, abs(F)):
                    stag += 1
                else:
                    stag = 0
                if stag >= 5:

                accepted = True
                break
            alpha *= 0.5
            if alpha < 1e-12:

        if not accepted:
