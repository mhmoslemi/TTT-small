                    r1 = int(rng.integers(NP))
                pool = list(range(NP)) + list(range(NP, NP + archive_n))
                pool = [j for j in pool if j != i and j != r1]
                r2g = pool[int(rng.integers(len(pool)))]
                xr2 = pop[r2g] if r2g < NP else archive[r2g - NP]
                v = pop[i] + f_i * (xp - pop[i]) + f_i * (pop[r1] - xr2)
            else:
                idxs = [j for j in range(NP) if j != i]
                pm = rng.permutation(len(idxs))
                v = pop[idxs[pm[0]]] + f_i * (pop[idxs[pm[1]]] - pop[idxs[pm[2]]])
            jrand = int(rng.integers(n))
            mask = rng.random(n) < cr_i
            mask[jrand] = True
            trial = np.empty(n)
            trial[mask] = v[mask]
            trial[~mask] = pop[i][~mask]
            if not np.all(np.isfinite(trial)):
                continue
            y = capped_simplex_project(trial, target)
            if y is None or not is_feasible(y, n, target):
                continue
            sc = exact_score(y, dx)
            evals += 1
            if not math.isfinite(sc):
                continue
            if sc < new_pops[i] - 1e-12:
                gen_archive.append(new_pop[i].copy())
                delta = new_pops[i] - sc
                new_pops[i] = sc
                new_pop[i] = y.copy()
                successes.append((f_i, cr_i, delta))
                if sc < incumbent_score - 1e-12:
                    incumbent = y.copy()
                    incumbent_score = sc
                    stagnant = 0
        pop = new_pop
        pops = new_pops
        archive.extend(gen_archive)
        while len(archive) > NP:
            archive.pop(int(rng.integers(len(archive))))
        if successes:
            tot_d = sum(d for _, _, d in successes)
            if tot_d > 1e-30:
                ws = np.array([d / tot_d for _, _, d in successes])
                Fs = np.array([f for f, _, _ in successes])
                CRs = np.array([c for _, c, _ in successes])
                mu_F = min(max(0.9 * mu_F + 0.1 * float(np.sum(ws * Fs**2) / np.sum(ws * Fs)), 0.05), 0.95)
                mu_CR = min(max(0.9 * mu_CR + 0.1 * float(np.sum(ws * CRs)), 0.05), 0.95)
        stagnant += 1
        if stagnant >= stag_limit and restarts < max_restarts:
            ec = max(2, NP // 4)
            o2 = np.argsort(pops)
            keep = [int(o2[k]) for k in range(ec)]
            for i, k in enumerate(keep):
                pop[i] = pop[k].copy()
                pops[i] = pops[k]
            for i in range(ec, NP):
                if evals >= budget:
                    break
                if i % 2 == 0:
                    sg = [0.02, 0.06, 0.15, 0.30][(i - ec) // 2 % 4]
                    v = incumbent + sg * rng.standard_normal(n)
                else:
                    v = rng.uniform(0, 1, n)
                y, sc = project_or_fallback(v, target, n, fallback, dx)
                evals += 1
                pop[i] = y
                pops[i] = sc
                if sc < incumbent_score - 1e-12:
                    incumbent = y.copy()
                    incumbent_score = sc
            archive = []
            mu_F, mu_CR, stagnant, restarts = 0.5, 0.7, 0, restarts + 1
        if incumbent_score <= 0.3808:
            break

    # Final
    final = capped_simplex_project(incumbent, target)
    if final is None or not is_feasible(final, n, target):
        final = fallback.copy()
    fs = exact_score(final, dx)
    if not math.isfinite(fs):
        final = np.full(n, 0.5)
        fs = exact_score(final, dx)
    return final, float(fs), n
