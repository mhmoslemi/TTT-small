import numpy as np
import math

def project_box_sum(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return None
    lo_b, hi_b = float(min(v) - 1.0), float(max(v))
    for _ in range(80):
        mid = 0.5 * (lo_b + hi_b)
        y = np.clip(v - mid, 0.0, 1.0)
        s = y.sum()
        if s > S:
            lo_b = mid
        else:
            hi_b = mid
    y = np.clip(v - 0.5*(lo_b + hi_b), 0.0, 1.0)
    resid = S - y.sum()
    if resid > 0:
        for i in range(n):
            add = min(resid, 1.0 - y[i])
            y[i] += add
            resid -= add
            if resid <= 0: break
    elif resid < 0:
        for i in range(n):
            sub = min(-resid, y[i])
            y[i] -= sub
            resid += sub
            if resid >= 0: break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1+1e-12):
        return None
    if abs(y.sum() - S) > 1e-10 * max(1, n):
        return None
    return y

def is_feasible(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1 + 1e-12):
        return False
    if abs(h.sum() - n/2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score_and_corr(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(corr)):
        return None, None
    score = float(np.max(corr)) * (2.0 / n)
    return score, corr

def make_pair_coeff(h, corr, n, a, b):
    # A = corr (length 2n-1), index j corresponds to lag k = j-(n-1)
    A = corr.copy()
    B = np.zeros(2*n-1)
    D = np.zeros(2*n-1)
    for j in range(2*n-1):
        k = j - (n-1)  # lag
        # B_k
        bk = 0.0
        ia = a - k
        if 0 <= ia < n:
            bk += 1.0 - h[ia]
        ib = b - k
        if 0 <= ib < n:
            bk -= 1.0 - h[ib]
        # subtract h[a]*d[a-k] and h[b]*d[b-k]
        # d[x]=1 if x==a, -1 if x==b, 0 otherwise
        if a - k == a:  # k==0
            bk -= h[a]
        elif a - k == b:  # k==a-b
            bk += h[a]
        if b - k == a:  # k==b-a
            bk -= h[b]
        elif b - k == b:  # k==0
            bk += h[b]
        B[j] = bk
        # D_k = -(d[a]*d[a-k] + d[b]*d[b-k])
        dk = 0.0
        da_k = 0.0
        if a - k == a: da_k = 1.0
        elif a - k == b: da_k = -1.0
        db_k = 0.0
        if b - k == a: db_k = 1.0
        elif b - k == b: db_k = -1.0
        dk = -(1.0 * da_k + (-1.0) * db_k)
        D[j] = dk
    if not (np.all(np.isfinite(A)) and np.all(np.isfinite(B)) and np.all(np.isfinite(D))):
        return None, None, None
    return A, B, D

def quad_roots(a, b, c):
    # Return real roots of a*x^2 + b*x + c = 0
    roots = []
    if abs(a) < 1e-30:
        if abs(b) < 1e-30:
            return []
        r = -c / b
        if np.isfinite(r):
            roots.append(r)
        return roots
    disc = b*b - 4.0*a*c
    tol = 64.0 * np.finfo(np.float64).eps * max(1.0, b*b, abs(4.0*a*c))
    if disc < -tol:
        return []
    if disc < 0:
        disc = 0.0
    sq = math.sqrt(max(0.0, disc))
    q = -0.5 * (b + math.copysign(sq, b))
    if abs(q) > 1e-30:
        r1 = q / a
        r2 = c / q
    else:
        r1 = -b / (2.0 * a)
        r2 = r1
    for r in [r1, r2]:
        if np.isfinite(r):
            roots.append(r)
    return roots

def upper_hull(A, B, valid_indices, lo, hi):
    # Build upper envelope of lines A[k] + B[k]*x for k in valid_indices over [lo,hi]
    if len(valid_indices) == 0:
        return [(lo, hi, None, None)]  # no line
    lines = []
    for k in valid_indices:
        lines.append((B[k], A[k]))
    # Sort by slope, then by intercept (descending)
    lines.sort(key=lambda t: (t[0], -t[1]))
    # Remove duplicates: keep highest intercept for same slope
    unique = []
    for m, c in lines:
        if unique and abs(unique[-1][0] - m) < 1e-15:
            continue  # already have higher intercept (since we sorted by -intercept)
        unique.append((m, c))
    # Build upper hull
    hull = []  # list of (slope, intercept, start_x)
    for m, c in unique:
        while hull:
            m0, c0, x0 = hull[-1]
            if abs(m - m0) < 1e-15:
                hull.pop()
                continue
            x_cross = (c0 - c) / (m - m0)
            if x_cross <= x0 + 1e-14:
                hull.pop()
            else:
                break
        if not hull:
            hull.append((m, c, -1e18))
        else:
            m0, c0, x0 = hull[-1]
            x_cross = (c0 - c) / (m - m0)
            hull.append((m, c, x_cross))
    # Clip to [lo, hi] and return segments
    segments = []
    for i in range(len(hull)):
        m, c, x_start = hull[i]
        x_end = hull[i+1][2] if i+1 < len(hull) else 1e18
        seg_lo = max(lo, x_start)
        seg_hi = min(hi, x_end)
        if seg_hi > seg_lo + 1e-15:
            segments.append((seg_lo, seg_hi, m, c))
    return segments

def line_minimize_pair(h, corr, current_score, n, a, b, score_budget):
    lo = max(-h[a], h[b] - 1.0)
    hi = min(1.0 - h[a], h[b])
    if hi - lo < 1e-14:
        return h, corr, current_score, score_budget
    A, B, D = make_pair_coeff(h, corr, n, a, b)
    if A is None:
        return h, corr, current_score, score_budget
    # Curved indices: lag=0 → index n-1, lag=a-b → index a-b+n-1, lag=b-a → index b-a+n-1
    curved_lags = {0, a-b, b-a}
    curved_idx = set()
    for lag in curved_lags:
        idx = lag + n - 1
        if 0 <= idx < 2*n-1:
            curved_idx.add(idx)
    non_curved = [j for j in range(2*n-1) if j not in curved_idx]
    segments = upper_hull(A, B, non_curved, lo, hi)
    candidates = []
    tol = 1e-12 * max(1.0, abs(lo), abs(hi))
    for seg_lo, seg_hi, m_line, c_line in segments:
        # Endpoints
        candidates.append(seg_lo)
        candidates.append(seg_hi)
        # Stationary points of convex quadratics (D>0): lags a-b and b-a
        for lag in [a-b, b-a]:
            idx = lag + n - 1
            if 0 <= idx < 2*n-1 and abs(D[idx]) > 1e-15:
                v = -B[idx] / (2.0 * D[idx])
                if seg_lo - tol <= v <= seg_hi + tol:
                    candidates.append(max(lo, min(hi, v)))
        # Pairwise intersections among curved terms
        curved_list = list(curved_idx)
        for i_c in range(len(curved_list)):
            for j_c in range(i_c+1, len(curved_list)):
                i, j = curved_list[i_c], curved_list[j_c]
                roots = quad_roots(D[i]-D[j], B[i]-B[j], A[i]-A[j])
                for r in roots:
                    if seg_lo - tol <= r <= seg_hi + tol:
                        candidates.append(max(lo, min(hi, r)))
            # Curved vs line
            if m_line is not None:
                roots = quad_roots(D[i_c], B[i_c] - m_line, A[i_c] - c_line)
                for r in roots:
                    if seg_lo - tol <= r <= seg_hi + tol:
                        candidates.append(max(lo, min(hi, r)))
        # alpha=0
        if seg_lo <= 0.0 <= seg_hi:
            candidates.append(0.0)
    # Deduplicate and evaluate
    candidates = list(set(round(c, 14) for c in candidates))
    # Evaluate polynomial objective at each candidate, keep top 5
    scored = []
    for alpha in candidates:
        alpha = max(lo, min(hi, alpha))
        if abs(alpha) < 1e-14:
            continue
        obj = -np.inf
        for idx in curved_idx:
            val = A[idx] + B[idx]*alpha + D[idx]*alpha*alpha
            if val > obj:
                obj = val
        for seg_lo2, seg_hi2, m2, c2 in segments:
            if seg_lo2 - tol <= alpha <= seg_hi2 + tol:
                if m2 is not None:
                    lval = c2 + m2 * alpha
                    if lval > obj:
                        obj = lval
                break
        if np.isfinite(obj):
            scored.append((obj, alpha))
    scored.sort(key=lambda t: t[0])
    # Take top 5 distinct alphas
    best_alphas = []
    for _, al in scored[:20]:
        if all(abs(al - ba) > 1e-12 * max(1.0, abs(al)) for ba in best_alphas):
            best_alphas.append(al)
        if len(best_alphas) >= 5:
            break
    if not best_alphas:
        return h, corr, current_score, score_budget
    # Exact rescore
    best_local = (current_score, h.copy(), corr.copy())
    for alpha in best_alphas:
        if abs(alpha) < 1e-14:
            continue
        h_new = h.copy()
        h_new[a] += alpha
        h_new[b] -= alpha
        h_new[a] = max(0.0, min(1.0, h_new[a]))
        h_new[b] = max(0.0, min(1.0, h_new[b]))
        if not is_feasible(h_new, n):
            continue
        sc, co = exact_score_and_corr(h_new)
        score_budget -= 1
        if sc is None or not np.isfinite(sc):
            continue
        if sc < best_local[0] - 1e-12 * max(1.0, abs(best_local[0])):
            best_local = (sc, h_new, co)
    if score_budget <= 0:
        return best_local[1], best_local[2], best_local[0], score_budget
    if best_local[0] < current_score - 1e-12 * max(1.0, abs(current_score)):
        return best_local[1], best_local[2], best_local[0], score_budget
    return h, corr, current_score, score_budget
