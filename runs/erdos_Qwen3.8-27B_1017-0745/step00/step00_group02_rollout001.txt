def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def project_k(y, n):
    S = n / 2.0
    if not np.all(np.isfinite(y)) or y.shape != (n,):
        return None
    lo_l = float(np.min(y)) - 1.0
    hi_l = float(np.max(y)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo_l + hi_l)
        x = np.clip(y - mid, 0.0, 1.0)
        s = float(np.sum(x))
        if s > S:
            lo_l = mid
        else:
            hi_l = mid
        if abs(s - S) < 1e-13 or (hi_l - lo_l) < 1e-14 * max(1.0, abs(mid)):
            break
    x = np.clip(y - mid, 0.0, 1.0)
    resid = S - float(np.sum(x))
    if abs(resid) > 1e-10:
        if resid > 0:
            for i in range(n):
                slack = 1.0 - x[i]
                add = min(slack, resid)
                if add > 0:
                    x[i] += add
                    resid -= add
                if abs(resid) < 1e-12:
                    break
        else:
            for i in range(n):
                if x[i] > 1e-14:
                    rem = min(x[i], -resid)
                    x[i] -= rem
                    resid += rem
                if abs(resid) < 1e-12:
                    break
    x = np.clip(x, 0.0, 1.0)
    if not (np.all(x >= 0.0) and np.all(x <= 1.0) and abs(float(np.sum(x)) - S) <= 1e-10):
        return None
    return x

def feasible_interval(h, d):
    mask = np.abs(d) > 1e-15
    if not np.any(mask):
        return None
    tlo = 0.0
    thi = 0.0
    for i in np.where(mask)[0]:
        a = -h[i] / d[i]
        b = (1.0 - h[i]) / d[i]
        lo_i, hi_i = min(a, b), max(a, b)
        tlo = max(tlo, lo_i)
        thi = min(thi, hi_i)
    if tlo > thi + 1e-12:
        return None
    return (max(0.0, tlo), min(0.0, thi))

def build_tree(n):
    records = []
    def rec(a, b, depth):
        if b - a <= 1:
            return
        m = (a + b) // 2
        records.append((depth, a, m, b))
        rec(a, m, depth + 1)
        rec(m, b, depth + 1)
    rec(0, n, 0)
    max_d = max(r[0] for r in records) + 1
    ordered = []
    for d in range(max_d):
        for r in records:
            if r[0] == d:
                ordered.append(r)
    return ordered

def make_dir(a, m, b, n):
    d = np.zeros(n)
    L = m - a
    R = b - m
    d[a:m] = 1.0 / L
    d[m:b] = -1.0 / R
    s = float(np.sum(d))
    if abs(s) > 1e-15:
        nz = np.where(np.abs(d) > 1e-15)[0]
        d[nz[0]] -= s
    return d

def exact_line_min(tlo, thi, q, b, c):

    cands = [tlo, thi, 0.0]
    for r in range(ncorr):
