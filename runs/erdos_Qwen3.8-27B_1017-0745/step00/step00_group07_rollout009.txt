            if pj is not None and is_feasible(pj, n, S):
                pool_h[k] = pj
            else:
                pool_sc[k] = float('inf')
                continue
        sc = exact_score(pool_h[k])
        if sc is not None:
            pool_sc[k] = sc
        else:
            pool_sc[k] = float('inf')
        if pool_sc[k] < pool_sc[fi]:
            fi = k

    h_out = pool_h[fi].copy()
    c5_out = exact_score(h_out)
    if c5_out is None or not np.isfinite(c5_out):
        h_out = h_const.copy()
        c5_out = exact_score(h_out)
    n_out = len(h_out)

    return (h_out, float(c5_out), int(n_out))
