import numpy as np
from scipy.optimize import minimize


def capped_simplex_project(v, target):
    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - np.sum(y)
    if abs(r) > 1e-13:
        if r > 0:
            cap = 1.0 - y
            avail = np.sum(cap)
            if avail < 1e-15:
                return None
            y += r * cap / avail
        else:
            cap = y.copy()
            avail = np.sum(cap)
            if avail < 1e-15:
                return None
            y += r * cap / avail
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(np.sum(y) - target) > 1e-9:
        return None
    return y


def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return np.inf
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))


def mass_residual(h):
    return np.sum(h) - len(h) / 2.0


def mass_jacobian(h):
    return np.ones(len(h))


def smooth_obj(h, tau, dx, n):
    s = dx * np.correlate(h, 1.0 - h, mode='full')
    m = np.max(s)
    e = np.exp((s - m) / tau)
    ss = np.sum(e)
    if ss < 1e-300:
        return float('inf')
    return float(m + tau * np.log(ss))


def smooth_jac(h, tau, dx, n):
    s = dx * np.correlate(h, 1.0 - h, mode='full')
    m = np.max(s)
    e = np.exp((s - m) / tau)
    ss = np.sum(e)
    if ss < 1e-300:
        return np.zeros(n)
    p = e / ss
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        t = k - (n - 1)
        pt = p[k] * dx
        if pt < 1e-18:
            continue
        if t >= 0:
            live = n - t
            if live > 0:
                grad[t:] += pt * (1.0 - h[:live])
                grad[:live] -= pt * h[t:]
        else:
            u = -t
            live = n - u
            if live > 0:
                grad[:live] += pt * (1.0 - h[u:])
                grad[u:] -= pt * h[:live]
    return grad


def is_feasible(h, target, tol=1e-10):
    if not np.all(np.isfinite(h)):
        return False
    if np.min(h) < -1e-12 or np.max(h) > 1.0 + 1e-12:
        return False
    return abs(np.sum(h) - target) <= tol


def run_optimization(h0, n, dx, target, bounds, constraints, taus, maxiter, ftol):
    h_cur = h0.copy()
    for tau in taus:
        args = (tau, dx, n)
        try:
            res = minimize(smooth_obj, h_cur, method='SLSQP', args=args,
                           jac=smooth_jac, bounds=bounds, constraints=constraints,
                           options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
            if res.x is not None and np.all(np.isfinite(res.x)):
                proj = capped_simplex_project(res.x, target)
                if proj is not None and is_feasible(proj, target):
                    f_new = smooth_obj(proj, tau, dx, n)
                    f_old = smooth_obj(h_cur, tau, dx, n)
                    if np.isfinite(f_new) and f_new <= f_old + 1e-10:
                        h_cur = proj
        except Exception:
            pass
    return h_cur


def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    n = 70
    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:

                        and abs(np.sum(pv) - n / 2.0) <= 1e-8):
                    parent = pv.copy()
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': mass_residual, 'jac': mass_jacobian}]
    h_parent = None
    if parent is not None:
        h_parent = capped_simplex_project(parent, target)
    starts = []
    if h_parent is not None:
        starts.append(h_parent.copy())
        starts.append(h_parent[::-1].copy())
        for sig in [0.04, 0.10, 0.22, 0.40]:
            pert = h_parent + sig * rng.standard_normal(n)
            proj = capped_simplex_project(pert, target)
            if proj is not None:
                starts.append(proj)
    for _ in range(3):
        v = rng.uniform(0.0, 1.0, n)
        proj = capped_simplex_project(v, target)
        if proj is not None:
            starts.append(proj)

        v = rng.beta(0.5, 0.5, n)
        proj = capped_simplex_project(v, target)

            starts.append(proj)
    starts.append(np.full(n, 0.5))
    h_best = None

    if h_parent is not None:

        c5_best = exact_score(h_parent, dx)

            continue

    for s in starts:

            continue

            sc = exact_score(h_opt, dx)
            if np.isfinite(sc) and sc < c5_best - 1e-12:
                c5_best = sc
                h_best = h_opt.copy()

    h_final = capped_simplex_project(h_best, target)

            h_final = np.full(n, 0.5)

        c5_best = exact_score(h_final, dx)
    return (h_final, c5_best, n)
