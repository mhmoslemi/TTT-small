def smooth_value_grad(h, tau, N):
    q = lag_values(h, N)
    dx = 2.0 / N
    r = dx * q
    rmax = float(np.max(r))
    ev = np.exp((r - rmax) / tau)
    s = float(np.sum(ev))
    p = ev / s
    phi = rmax + tau * np.log(s)
    grad = np.zeros(N)
    oh = 1.0 - h
    for idx in range(2 * N - 1):
        if p[idx] < 1e-15:
            continue
        w = dx * p[idx]
        k = idx - (N - 1)
        if k >= 0:
            L = N - k
            grad[k:k+L] += w * oh[0:L]
            grad[0:L] -= w * h[k:k+L]
        else:
            d = -k
            L = N - d
            grad[0:L] += w * oh[d:d+L]
            grad[d:d+L] -= w * h[0:L]
    if not np.all(np.isfinite(grad)):
        return phi, np.zeros(N)
    return phi, grad

def smooth_value_only(h, tau, N):
    q = lag_values(h, N)
    dx = 2.0 / N
    r = dx * q
    rmax = float(np.max(r))
    return float(rmax + tau * np.log(np.sum(np.exp((r - rmax) / tau))))

def optimize_from(cur, N, inc_vec, inc_score, iters):
    try:
        temps = [0.08, 0.02, 0.005, 0.001, 0.0002, 0.00004, 0.000008]
        for tau in temps:
            stage_best = cur.copy()
            stage_best_sc = score_exact(cur, N)
            for _ in range(iters):
                phi, g = smooth_value_grad(cur, tau, N)
                if not np.isfinite(phi) or not np.all(np.isfinite(g)):
                    break
                alpha = 1.0
                accepted = False
                stalled = False
                for _t in range(24):
                    y = project_box_sum(cur - alpha * g, N)
                    d = y - cur
                    if np.max(np.abs(d)) < 1e-10:
                        stalled = True
                        break
                    gd = float(np.dot(g, d))
                    phi_y = smooth_value_only(y, tau, N)
                    sc_y = score_exact(y, N)
                    if np.isfinite(sc_y) and sc_y < inc_score:
                        inc_vec = y.copy()
                        inc_score = sc_y
                    if np.isfinite(sc_y) and sc_y < stage_best_sc:
                        stage_best = y.copy()
                        stage_best_sc = sc_y
                    if gd < 0 and phi_y <= phi + 1e-4 * gd + 1e-14:
                        if sc_y < inc_score:
                            pass
                        cur = y
                        accepted = True
                        if np.max(np.abs(d)) < 1e-8:
                            break
                        alpha = min(1.4 * alpha, 10.0)
                        break
                    alpha *= 0.5
                    if alpha < 1e-8:
                        break
                sc_cur = score_exact(cur, N)
                if np.isfinite(sc_cur) and sc_cur < inc_score:
                    inc_vec = cur.copy()
                    inc_score = sc_cur
                if np.isfinite(sc_cur) and sc_cur < stage_best_sc:
                    stage_best = cur.copy()
                    stage_best_sc = sc_cur
                if stalled or (not accepted and alpha < 1e-8):
                    cur = stage_best.copy()
                    break
            cur = stage_best.copy()
    except Exception:
        pass
    return inc_vec, inc_score

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    N = None
    parent = None
    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            arr = np.asarray(initial_h_values, dtype=np.float64).ravel()
            if arr.size > 0:
                N = arr.size
                if np.all(np.isfinite(arr)):
                    parent = arr
    except Exception:
        pass
    if N is None:
        try:
            N = int(kwargs.get('n_points', 70))
            if N < 1:
                N = 70
        except Exception:
            N = 70
    if N == 1:
        return np.array([0.5]), 0.5 * 2.0, 1
    m = N / 2.0
    if parent is not None:
        try:
            if np.all(parent >= -1e-12) and np.all(parent <= 1.0+1e-12) and abs(np.sum(parent)-m) < 1e-8*max(1,N):
                baseline = np.clip(parent, 0.0, 1.0)
            else:
                baseline = project_box_sum(parent, N)
        except Exception:
            baseline = project_box_sum(parent, N) if np.all(np.isfinite(parent)) else np.full(N, 0.5)
    else:
        baseline = np.full(N, 0.5)
    inc_vec = baseline.copy()
    inc_score = score_exact(baseline, N)
    if not np.isfinite(inc_score):
        inc_vec = np.full(N, 0.5)
        inc_score = score_exact(inc_vec, N)
    if N <= 100:
        n_starts, iters = 8, 60
    elif N <= 200:
        n_starts, iters = 6, 40
    elif N <= 500:
        n_starts, iters = 4, 20
    else:
        n_starts, iters = 3, 8
    starts = [inc_vec.copy()]
    starts.append(project_box_sum(baseline + 0.20 * rng.standard_normal(N), N))
    starts.append(project_box_sum(rng.uniform(0.0, 1.0, size=N), N))
    R = min(6, max(1, (N - 1) // 2))
    ii = np.arange(N, dtype=np.float64)
    four = np.full(N, 0.5)
    for r in range(1, R + 1):
        A = rng.standard_normal()
        B = rng.standard_normal()
        four += (0.12 / r) * (A * np.sin(2.0 * np.pi * r * ii / N) + B * np.cos(2.0 * np.pi * r * ii / N))
    starts.append(project_box_sum(four, N))
    if parent is not None and np.all(np.isfinite(parent)):
        starts.append(project_box_sum(1.0 - parent, N))
    else:
        starts.append(project_box_sum(np.full(N, 0.5) + 0.4 * rng.standard_normal(N), N))
    starts.append(np.full(N, 0.5))
    starts.append(project_box_sum(baseline + 0.08 * rng.standard_normal(N), N))
    starts.append(project_box_sum(baseline + 0.40 * rng.standard_normal(N), N))
    starts = starts[:n_starts]
    for st in starts:
        try:
            st = project_box_sum(st, N)
            sc = score_exact(st, N)
            if np.isfinite(sc) and sc < inc_score:
                inc_vec = st.copy()
                inc_score = sc
            inc_vec, inc_score = optimize_from(st, N, inc_vec, inc_score, iters)
        except Exception:
            continue
    try:
        inc_vec = np.asarray(inc_vec, dtype=np.float64).ravel()
        if inc_vec.shape[0] != N or not np.all(np.isfinite(inc_vec)):
            raise ValueError("bad shape")
        if np.any(inc_vec < -1e-12) or np.any(inc_vec > 1.0 + 1e-12):
            inc_vec = project_box_sum(inc_vec, N)
        fs = score_exact(inc_vec, N)
        if not np.isfinite(fs):
            inc_vec = np.full(N, 0.5)
            fs = score_exact(inc_vec, N)
    except Exception:
        inc_vec = np.full(N, 0.5)
        fs = score_exact(inc_vec, N)
    return inc_vec, float(fs), N
