            sz = exact_score(z, n)
            if sz < best_s:
                best_s, best_z, best_alpha = sz, z, alpha
        if best_z is None or best_s >= score - 1e-12:
            R *= 0.25
            rejs += 1
            continue
        actual = score - best_s
        rho = actual / pred if pred > 1e-12 else 0.0
        x, score = best_z.copy(), best_s
        rejs = 0
        disp = np.sum(np.abs(x - best_z + best_z - x))  # placeholder
        if rho < 0.1:
            R *= 0.5
        elif rho >= 0.75 and best_alpha == 1.0 and np.sum(np.abs(best_z - y)) < 1e-10 and a >= 0.8 * R:
            R = min(2.0 * R, Rmax)
        R = max(R, 1e-6)
        R = min(R, Rmax)
    return x, score

def build_starts(base, n, rng, count):
    pool = []
    def add(v):
        if v is None:
            return
        for s in pool:
            if np.max(np.abs(v - s)) < 1e-10:
                return
        pool.append(v)
    add(base.copy())
    add(project_capped(base[::-1], n))
    add(project_capped(1.0 - base[::-1], n))
    add(np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    add(project_capped(blk, n))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    add(project_capped(alt, n))
    for sd in (0.04, 0.12, 0.30):
        add(project_capped(base + rng.normal(0, sd, n), n))
    add(project_capped(rng.uniform(0, 1, n), n))
    return pool[:count]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    fallback = project_capped(parent, n) if parent is not None else np.full(n, 0.5)
    if fallback is None:
        fallback = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 64:
        t_s, t_i = 9, 50
    elif n <= 128:
        t_s, t_i = 5, 30
    elif n <= 256:
        t_s, t_i = 2, 15
    else:
        t_s, t_i = 0, 0
    starts = build_starts(fallback, n, rng, t_s) if t_s > 0 else [fallback]
    gb, gs = fallback.copy(), exact_score(fallback, n)
    for st in starts:
        s = exact_score(st, n)
        if s < gs - 1e-12:
            gs, gb = s, st.copy()
    if n > 1 and t_i > 0:
        for st in starts:
            xo, so = optimize_start(st, n, t_i)
            if so < gs - 1e-12:
                gs, gb = so, xo.copy()
    gb = project_capped(gb, n)
    if gb is None or not is_feasible(gb, n):
        gb = fallback
    if exact_score(gb, n) > exact_score(fallback, n):
        gb = fallback
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = np.full(n, 0.5)
    return final, float(exact_score(final, n)), int(n)
