        h = proj(parent, N, N / 2.0)
        return h, exact_score(h), N
    m, d = N / 2.0, 2.0 / N
    bounds = [(0.0, 1.0)] * N
    rng = np.random.default_rng(seed)
    starts = []
    if parent is not None:
        starts.append(proj(parent, N, m))
    starts.append(np.full(N, 0.5))
    st = np.zeros(N)
    st[:int(m)] = 1.0
    if int(m) < N:
        st[int(m)] = m - int(m)
    starts.append(proj(st, N, m))
    for f in (1, 3, 7):
        starts.append(proj(0.5 + 0.22 * np.sin(2 * np.pi * f * np.arange(N) / N), N, m))
    n_r = kwargs.get('n_restarts', 8)
    if not isinstance(n_r, int) or n_r < 6:
        n_r = 6
    n_r = min(n_r, 12)
    base = starts[0] if parent is not None else np.full(N, 0.5)
    sds = (0.04, 0.10, 0.20)
    while len(starts) < n_r:
        nz = rng.normal(0, sds[len(starts) % 3], N)
        nz = 0.5 * nz + 0.25 * np.roll(nz, 1) + 0.25 * np.roll(nz, -1)
        starts.append(proj(base + nz, N, m))
    uq = [starts[0]]
    for s in starts[1:]:
        if not any(np.abs(s - u).max() < 1e-8 for u in uq):
            uq.append(s)
    starts = uq[:n_r]
    best_h = proj(starts[0], N, m)
    best_c = exact_score(best_h)
    maxit = kwargs.get('maxiter', 120)
    if not isinstance(maxit, int) or not 60 <= maxit <= 250:
        maxit = 120
    cands = [best_h.copy()]
    taus = (0.02, 0.005, 0.001, 0.0002)
    for s in starts:
        h_c = proj(s, N, m)
        for tau in taus:
            try:
                res = minimize(s_obj, h_c, args=(tau, N, d), jac=s_jac,
                               method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': meq_s, 'jac': meq_s_jac, 'args': m}],
                               ftol=1e-10, maxiter=maxit, disp=False)
                if res.x is not None and res.x.size == N and np.all(np.isfinite(res.x)):
                    h_c = proj(res.x, N, m)
            except Exception:
                pass
            c = exact_score(h_c)
            if c < best_c - 1e-12:
                best_c, best_h = c, h_c.copy()
            cands.append(h_c.copy())
    cs = sorted(cands, key=exact_score)
    pl = []
    for h in cs:
        if len(pl) >= 3:
            break
        if not any(np.abs(h - p).max() < 1e-6 for p in pl):
            pl.append(h)
    eB = bounds + [(0.0, 2.1)]
    eC = [{'type': 'eq', 'fun': meq_e, 'jac': meq_e_jac, 'args': m},
          {'type': 'ineq', 'fun': epi_ineq, 'jac': epi_ineq_jac, 'args': (N, d)}]
    eMAX = min(400, 2 * maxit)
    for h0 in pl:
        z0 = np.concatenate([h0, [exact_score(h0) + 1e-6]])
        try:
            res = minimize(epi_obj, z0, args=(N, d), jac=epi_obj_jac,
                           method='SLSQP', bounds=eB, constraints=eC,
                           ftol=1e-10, maxiter=eMAX, disp=False)
            if res.x is not None and res.x.size == N + 1 and np.all(np.isfinite(res.x)):
                h_e = proj(res.x[:N], N, m)
                c = exact_score(h_e)
                if c < best_c - 1e-12:
                    best_c, best_h = c, h_e.copy()
        except Exception:
            pass
    best_h = proj(best_h, N, m)
    best_c = exact_score(best_h)
    if best_c >= 1e9:
        best_h = np.full(N, 0.5)
        best_c = exact_score(best_h)
    return best_h, best_c, N
