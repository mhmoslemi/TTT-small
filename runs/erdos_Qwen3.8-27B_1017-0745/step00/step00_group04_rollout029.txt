                    mutant = X[int(idx[0])] + Ft * (X[int(idx[1])] - X[int(idx[2])])
                else:
                    pb_c = [j for j in pb_pool if j != i]
                    if not pb_c:
                        pb_c = [pb_pool[0]]
                    pi = pb_c[int(rng.integers(len(pb_c)))]
                    rem = [j for j in cands if j != pi]
                    idx = rng.choice(rem, size=2, replace=False)
                    mutant = X[i] + Ft * (X[pi] - X[i]) + Ft * (X[int(idx[0])] - X[int(idx[1])])
                trial = X[i].copy()
                mask = rng.random(n) < CRt
                trial[mask] = mutant[mask]
                trial[int(rng.integers(n))] = mutant[int(rng.integers(n))]
                proj = project_capped_simplex(trial, n)
                if proj is None or not is_feasible(proj, n):
                    continue
                if np.max(np.abs(proj - X[i])) < 1e-14:
                    continue
                sc = exact_score(proj, n)
                if sc is None:
                    continue
                if sc < fitness[i] - 1e-12:
                    X_next[i] = proj
                    fit_next[i] = sc
                    F_next[i] = Ft
                    CR_next[i] = CRt
                    if sc < inc_score - 1e-12:
                        inc = proj.copy()
                        inc_score = sc
                        improved_gen = True
                elif abs(sc - fitness[i]) <= 1e-12 and np.max(np.abs(proj - X[i])) >= 1e-6 and rng.random() < 0.02:
                    X_next[i] = proj
                    fit_next[i] = sc
                    F_next[i] = Ft
                    CR_next[i] = CRt
            X, fitness, F_arr, CR_arr = X_next, fit_next, F_next, CR_next
            stagnant = 0 if improved_gen else stagnant + 1
            if stagnant >= 30 and restarts < 3:
                n_keep = max(2, P // 4) - 1
                ki = np.argsort(fitness)[:n_keep]
                new_pop = [X[j].copy() for j in ki]
                rs_sig = [0.20, 0.10, 0.05][min(restarts, 2)]
                for _ in range(P - len(new_pop)):
                    if rng.random() < 0.75:
                        z = rng.standard_normal(n)
                        z -= np.mean(z)
                        v = project_capped_simplex(inc + rs_sig * z, n)
                    else:
                        v = project_capped_simplex(rng.uniform(0, 1, n), n)
                    new_pop.append(v if v is not None else new_pop[0].copy())
                while len(new_pop) < P:
                    new_pop.append(new_pop[0].copy())
                X = np.array(new_pop[:P], dtype=np.float64)
                fitness = np.full(P, np.inf)
                for i in range(P):
                    sc = exact_score(X[i], n)
                    if sc is not None:
                        fitness[i] = sc
                        if sc < inc_score - 1e-12:
                            inc = X[i].copy()
                            inc_score = sc
                F_arr = rng.uniform(0.5, 0.8, P)
                CR_arr = np.full(P, 0.8)
                stagnant = 0
                restarts += 1
            elif stagnant >= 70 and restarts >= 3:
                break
    best_h = inc.copy()
    best_sc = inc_score
    if parent_score is not None and parent_score < best_sc - 1e-12:
        best_h, best_sc = parent_seed.copy(), parent_score
    if const_score is not None and const_score < best_sc - 1e-12:
        best_h, best_sc = const_h.copy(), const_score
    if not is_feasible(best_h, n):
        rep = project_capped_simplex(best_h, n)
        if rep is not None:
            best_h = rep
    best_sc = exact_score(best_h, n)
    if best_sc is None:
        best_h, best_sc = const_h.copy(), const_score
    return (best_h.astype(np.float64), float(best_sc), int(n))
