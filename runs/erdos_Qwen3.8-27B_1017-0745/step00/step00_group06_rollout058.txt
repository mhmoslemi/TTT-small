    try:
        n_restarts = max(6, min(12, int(kwargs.get('n_restarts', 8))))
    except Exception:
        n_restarts = 8
    maxiter = 120
    try:
        maxiter = max(60, min(250, int(kwargs.get('maxiter', 120))))
    except Exception:
        maxiter = 120
    epi_iters = min(400, 2 * maxiter)

    rng = np.random.default_rng(int(seed) if isinstance(seed, (int, np.integer)) else 42)
    starts = [inc.copy(), np.full(N, 0.5)]
    step_v = np.zeros(N)
    step_v[:int(m)] = 1.0
    fr = m - int(m)
    if 0 < fr < 1:
        step_v[int(m)] = fr
    starts.append(step_v)
    idx = np.arange(N, dtype=np.float64)
    for f in [1, 3, 7]:
        starts.append(_proj(0.5 + 0.22 * np.sin(2 * np.pi * f * idx / N), m, N))
    base = parent.copy() if parent is not None else np.full(N, 0.5)
    stds = [0.04, 0.10, 0.20]
    while len(starts) < n_restarts:
        s = stds[len(starts) % 3]
        noise = rng.standard_normal(N) * s
        noise = 0.5 * noise + 0.25 * np.roll(noise, 1) + 0.25 * np.roll(noise, -1)
        starts.append(_proj(base + noise, m, N))
    deduped = [starts[0]]
    for s in starts[1:]:
        ok = True
        for d2 in deduped:
            if np.max(np.abs(s - d2)) < 1e-8:
                ok = False
                break
        if ok:
            deduped.append(s)
    starts = deduped[:n_restarts]

    bounds = [(0.0, 1.0)] * N
    candidates = [inc.copy()]

    for start in starts:
        h_cur = start.copy()
        for tau in [0.02, 0.005, 0.001, 0.0002]:
            _G['tau'] = tau
            try:
                res = minimize(_smooth_obj, h_cur, jac=_smooth_jac,
                               bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': _mass_eq, 'jac': _mass_eq_jac}],
                               method='SLSQP', ftol=1e-10, disp=False,
                               options={'maxiter': maxiter})
                if res.x is not None and len(res.x) == N and np.all(np.isfinite(res.x)):
                    h_cur = _proj(res.x, m, N)
            except Exception:
                pass
            candidates.append(h_cur.copy())

    scored = [(c, _exact_score(c)) for c in candidates]
    scored.sort(key=_score_key)
    top_sels = []
    for h_c, sc in scored:
        ok = all(np.max(np.abs(h_c - t)) > 1e-8 for t, _ in top_sels)
        if ok:
            top_sels.append((h_c, sc))
        if len(top_sels) >= 3:
            break

    epi_bounds = [(0.0, 1.0)] * N + [(0.0, 2.1)]
    for h_c, sc in top_sels:
        z0 = np.concatenate([h_c, [sc + 1e-6]])
        try:
            res = minimize(_epi_obj, z0, jac=_epi_obj_jac,
                           bounds=epi_bounds,
                           constraints=[
                               {'type': 'eq', 'fun': _epi_mass, 'jac': _epi_mass_jac},
                               {'type': 'ineq', 'fun': _epi_ineq, 'jac': _epi_ineq_jac}
                           ],
                           method='SLSQP', ftol=1e-10, disp=False,
                           options={'maxiter': epi_iters})
            if res.x is not None and len(res.x) == N + 1 and np.all(np.isfinite(res.x)):
                h_ep = _proj(res.x[:N], m, N)
                sc_ep = _exact_score(h_ep)
                if sc_ep < inc_score - 1e-12:
                    inc = h_ep
                    inc_score = sc_ep
        except Exception:
            pass

    inc = _proj(inc, m, N)
    if not (np.all(inc >= -1e-12) and np.all(inc <= 1.0 + 1e-12)) or abs(np.sum(inc) - m) > 1e-9 * max(1, N):
        inc = np.full(N, 0.5)
    inc = np.clip(inc, 0.0, 1.0)
    c5 = _exact_score(inc)
    return (inc.astype(np.float64), float(c5), int(N))
