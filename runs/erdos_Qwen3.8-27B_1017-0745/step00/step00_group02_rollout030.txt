def exact_score(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr) * dx)

def joint_project(y, n):
    S = n / 2.0
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return None
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        x = np.clip(y - mid, 0.0, 1.0)
        s = float(np.sum(x))
        if s > S:
            lo = mid
        else:
            hi = mid
        if abs(s - S) < 1e-13 or (hi - lo) < 1e-14:
            break
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - float(np.sum(x))
    if abs(resid) > 1e-12:
        if resid > 0:
            for i in range(n):

                x[i] += add
                resid -= add
                if resid <= 1e-14:

                    break

def depth_key(r):
    return r[0]

def build_tree(n):
    records = []
    stack = [(0, n, 0)]
    while stack:
        a, b, d = stack.pop()
        if b - a <= 1:
            continue
        m = (a + b) // 2
        records.append((d, a, m, b))
        stack.append((m, b, d + 1))
        stack.append((a, m, d + 1))

        th = (1.0 - h[i]) / d[i]
        if tl > th:

            t_lo = tl
