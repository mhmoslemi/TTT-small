    scales = (0.05, 0.10, 0.18, 0.28)
    for att in range(32):
        if len(starts) >= 32:
            break
        z2 = rng.standard_normal(N)
        sm2 = smooth_vec(z2)
        starts = add_start(starts, fallback + scales[att % 4] * sm2, N)
    return starts

def run(seed=42, **kwargs):
    N = None
    parent = None
    try:
        ih = globals().get('initial_h_values', None)
        if ih is not None:
            arr = np.asarray(ih, dtype=np.float64)
            if arr.ndim == 1 and arr.size >= 2 and np.all(np.isfinite(arr)) and np.max(np.abs(arr)) <= 1e6:
                N = int(arr.size)
                parent = arr.copy()
    except Exception:
        pass
    if N is None:
        try:
            N = max(2, min(999, int(float(kwargs.get('n_points', 95)))))
        except Exception:
            N = 95
    if parent is not None and N != parent.size:
        N = parent.size

    try:
        s = int(seed) % (2 ** 32)
    except Exception:
        s = 42
    rng = np.random.default_rng(s)

    if parent is not None:
        fallback = project_capped_simplex(parent, N)
        if fallback is None or not is_feasible(fallback, N):
            fallback = np.full(N, 0.5)
    else:
        fallback = np.full(N, 0.5)
    saved_fb = fallback.copy()
    incumbent = fallback.copy()
    incumbent_score = exact_c5(incumbent, N)
    if not is_feasible(incumbent, N) or incumbent_score == float('inf'):
        incumbent = np.full(N, 0.5)
        incumbent_score = exact_c5(incumbent, N)

    if N > 999:
        h_fin = project_capped_simplex(incumbent, N)
        if h_fin is None or not is_feasible(h_fin, N):
            h_fin = saved_fb
        return (h_fin.astype(np.float64), float(exact_c5(h_fin, N)), N)

    if N <= 128:
        max_st, def_it = 8, 80
    elif N <= 256:
        max_st, def_it = 4, 40
    else:
        max_st, def_it = 2, 15
    try:
        n_st = max(1, min(max_st, int(kwargs.get('n_starts', max_st))))
    except Exception:
        n_st = max_st
    try:
        maxit = max(10, min(def_it, int(kwargs.get('maxiter', def_it))))
    except Exception:
        maxit = def_it

    all_starts = build_starts(N, rng, fallback)
    starts = all_starts[:n_st]

    for st in starts:
        sc = exact_c5(st, N)
        if sc < incumbent_score - 1e-12:
            incumbent = st.copy()
            incumbent_score = sc

    d = 2.0 / N
    for st in starts:
        x = st.copy()
        Cx = exact_c5(x, N)
        eta_base = 0.20
        eta_min = 1e-4
        no_imp = 0
        for _ in range(maxit):
            R = np.correlate(x, 1.0 - x, mode='full') * d
            Cx = float(np.max(R))
            order = np.argsort(R)[::-1]
            eps = max(1e-12, 1e-9 * max(1.0, abs(Cx)))
            active = []
            for idx in order:
                if Cx - R[idx] <= eps:
                    active.append(int(idx))
                if len(active) >= 8:
                    break
            if not active:
                active = [int(order[0])]
            top2 = [int(order[0])] + ([int(order[1])] if len(order) > 1 else [])

            G_list = []
            seen = set()
            for row in active:
                q = row - (N - 1)
                if q not in seen:
                    G_list.append(lag_gradient(x, q, d, N))
                    seen.add(q)
            for row in top2:
                q = row - (N - 1)
                if q not in seen:
                    G_list.append(lag_gradient(x, q, d, N))
                    seen.add(q)
            if not G_list:
                no_imp += 1
                continue
            G = np.array(G_list, dtype=np.float64)
            gbar = frank_wolfe_min_norm(G)

            dirs = []
            mg = np.max(np.abs(gbar))
            if mg > 1e-14:
                dirs.append(-gbar / mg)
            for t in range(min(2, len(G))):
                mt = np.max(np.abs(G[t]))
                if mt > 1e-14:
                    nd = -G[t] / mt
                    skip = False
                    for ex in dirs:
                        if np.max(np.abs(nd - ex)) < 1e-8:
                            skip = True
                            break
                    if not skip:
                        dirs.append(nd)

            best_tri = None
            best_sc = Cx
            best_eta = 0.0
            for u in dirs:
                for se in range(5):
                    eta_s = max(eta_min, eta_base / (2.0 ** se))
                    y = project_capped_simplex(x + eta_s * u, N)
                    if y is None or not is_feasible(y, N):
                        continue
                    if np.max(np.abs(y - x)) <= 1e-12:
                        continue
                    sc = exact_c5(y, N)
                    if sc < best_sc - 1e-12:
                        best_sc = sc
                        best_tri = y
                        best_eta = eta_s

            if best_tri is None and (no_imp + 1) % 3 == 0:
                e0 = min(0.10, max(eta_min, eta_base))
                for _ev in range(2):
                    sm = smooth_vec(rng.standard_normal(N))
                    for sgn in (1.0, -1.0):
                        for se in range(3):
                            eta_s = e0 / (2.0 ** se)
                            y = project_capped_simplex(x + sgn * eta_s * sm, N)
                            if y is None or not is_feasible(y, N):
                                continue
                            if np.max(np.abs(y - x)) <= 1e-12:
                                continue
                            sc = exact_c5(y, N)
                            if sc < best_sc - 1e-12:
                                best_sc = sc
                                best_tri = y
                                best_eta = eta_s

            if best_tri is not None:
                x = best_tri
                Cx = best_sc
                no_imp = 0
                if best_eta >= 0.75 * eta_base:
                    eta_base = min(0.25, 1.25 * eta_base)
                else:
                    eta_base = max(eta_min, 1.25 * best_eta)
                if Cx < incumbent_score - 1e-12:
                    incumbent = x.copy()
                    incumbent_score = Cx
            else:
                eta_base = max(eta_min, 0.5 * eta_base)
                no_imp += 1
                if no_imp >= 5 and eta_base <= eta_min * 1.01:
                    break

    h_fin = project_capped_simplex(incumbent, N)
    if h_fin is None or not is_feasible(h_fin, N):
        h_fin = saved_fb
    c5_fin = exact_c5(h_fin, N)
    c5_fb = exact_c5(saved_fb, N)
    if c5_fb < c5_fin - 1e-12:
        h_fin = saved_fb
        c5_fin = c5_fb
    if not is_feasible(h_fin, N) or c5_fin == float('inf'):
        h_fin = np.full(N, 0.5)
        c5_fin = exact_c5(h_fin, N)
    return (h_fin.astype(np.float64), float(c5_fin), N)
