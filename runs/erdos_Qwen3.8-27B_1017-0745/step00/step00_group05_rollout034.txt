            sF.append(F_i); sCR.append(CR_i); sD.append(pop_sc[i] - s)
            if s < inc_sc - 1e-12:
                incumbent, inc_sc = pr.copy(), s
        pop, pop_sc = new_pop, new_sc
        if sF:
            td = sum(sD)
            ws = [d / td for d in sD]
            mF2 = sum(w * f * f for w, f in zip(ws, sF))
            mF = sum(w * f for w, f in zip(ws, sF))
            if mF > 0:
                mu_F = float(np.clip(0.9 * mu_F + 0.1 * mF2 / mF, 0.05, 0.95))
            mu_CR = float(np.clip(0.9 * mu_CR + 0.1 * sum(w * c for w, c in zip(ws, sCR)), 0.05, 0.95))
        while len(archive) > NP:
            archive.pop(int(rng.integers(0, len(archive))))
        if inc_sc <= 0.3808:
            break
        if min(pop_sc) < inc_sc - 1e-12:
            stag = 0
        else:
            stag += 1
        if stag >= stag_thr and rst_cnt < max_restarts:
            ordr = np.argsort(pop_sc)
            keep = [int(x) for x in ordr[:elite]]
            for k in keep:
                if abs(pop_sc[k] - inc_sc) < 1e-12:
                    break
            else:
                keep[-1] = int(np.argmin(pop_sc))
            np2 = [pop[k].copy() for k in keep]
            ns2 = [pop_sc[k] for k in keep]
            ci = 0
            while len(np2) < NP and ec[0] + NP <= budget:
                if ci % 2 == 0:
                    sg = sigmas[(ci // 2) % 4]
                    h2 = incumbent + sg * rng.standard_normal(n)
                else:
                    h2 = make_rand_profile(n, rng, incumbent)
                pr = capped_simplex_project(h2, target)
                if pr is not None:
                    s2 = exact_score(pr, dx); ec[0] += 1
                    if np.isfinite(s2):
                        np2.append(pr.copy()); ns2.append(s2)
                        if s2 < inc_sc - 1e-12:
                            incumbent, inc_sc = pr.copy(), s2
                ci += 1
            while len(np2) < NP:
                np2.append(incumbent.copy()); ns2.append(inc_sc)
            pop, pop_sc, archive = np2, ns2, []
            mu_F, mu_CR, stag, rst_cnt = 0.5, 0.7, 0, rst_cnt + 1

    fp = capped_simplex_project(incumbent, target)
    if fp is not None and is_feasible(fp, n, target):
        hf = fp
    else:
        hf = capped_simplex_project(fallback, target)
        if hf is None:
            hf = np.full(n, 0.5)
    c5 = exact_score(hf, dx)
    if not np.isfinite(c5):
        hf = np.full(n, 0.5)
        c5 = exact_score(hf, dx)
    return hf, float(c5), int(n)
