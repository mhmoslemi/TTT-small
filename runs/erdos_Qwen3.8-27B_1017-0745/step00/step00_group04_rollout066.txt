            if res_e.x.shape == (n + 1,) and np.all(np.isfinite(res_e.x)):
                hc = project_feasible(res_e.x[:n], n)
                if valid_check(hc, n):
                    hs = exact_score(hc)
                    if hs < best_score - 1e-12:
                        best_h = hc.copy()
                        best_score = hs

    # Final epigraph polish from global best
    t0 = float(np.max(compute_q_G(best_h)[0])) + max(1e-8, 1e-6)
    z0 = np.concatenate([best_h, [t0]])
    bounds_e = [(0.0, 1.0)] * n + [(0.0, None)]
    cons_e = [{'type': 'eq', 'fun': epig_eq, 'jac': epig_eq_jac},
              {'type': 'ineq', 'fun': epig_ineq, 'jac': epig_ineq_jac}]
    try:
        res_f = minimize(epig_obj, z0, method='SLSQP', jac=epig_obj_grad,
                         bounds=bounds_e, constraints=cons_e,
                         options={'ftol': 1e-10, 'maxiter': 200})
    except Exception:
        res_f = None
    if res_f is not None and res_f.x is not None:
        if res_f.x.shape == (n + 1,) and np.all(np.isfinite(res_f.x)):
            hc = project_feasible(res_f.x[:n], n)
            if valid_check(hc, n):
                hs = exact_score(hc)
                if hs < best_score - 1e-12:
                    best_h = hc.copy()
                    best_score = hs

    best_h = project_feasible(best_h, n)
    if not valid_check(best_h, n):
        best_h = np.full(n, 0.5)
    return best_h, exact_score(best_h), n
