def capped_simplex_project(v, target):
    n = len(v)
    if n == 0 or not np.all(np.isfinite(v)):
        return None
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - np.sum(y)
    for _ in range(3):
        if abs(r) < 1e-12:
            break
        idx = 0
        while abs(r) > 1e-13 and idx < n:
            if r > 0:
                amt = min(r, 1.0 - y[idx])
                y[idx] += amt
                r -= amt
            else:
                amt = min(-r, y[idx])
                y[idx] -= amt
                r += amt
            idx += 1
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    if abs(np.sum(y) - target) > 1e-10:
        return None
    return y

def compute_grads(h):

def pair_poll(h, dx, rng, n):
    best = h.copy()
    best_score = exact_score(h, dx)
    if best_score is None:
        return h
    if n > 256:
        return best
    if n <= 128:
        n_pairs = 24
    else:
        n_pairs = 8
    for radius in [0.08, 0.025, 0.008]:
        for _ in range(n_pairs):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            while b == a:
                b = int(rng.integers(0, n))
            s_lo = max(-best[a], best[b] - 1.0)
            s_hi = min(1.0 - best[a], best[b])
            s_lo = max(s_lo, -radius)
            s_hi = min(s_hi, radius)
            if s_hi <= s_lo:
                continue
            s_vals = sorted(set([s_lo, s_hi, -radius, -radius/2, radius/2, radius]))
            s_vals = [s for s in s_vals if s_lo <= s <= s_hi and abs(s) > 1e-14]
            pair_best_score = best_score
            pair_best = None
            for s in s_vals:
                cand = best.copy()
                cand[a] += s
                cand[b] -= s
                cand = capped_simplex_project(cand, n / 2.0)
                if cand is None or not check_feasible(cand, n):
                    continue
                sc = exact_score(cand, dx)
                if sc is not None and sc < pair_best_score - 1e-12:
                    pair_best_score = sc
                    pair_best = cand.copy()
            if pair_best is not None:
                best = pair_best
                best_score = pair_best_score
    return best

import numpy as np
from scipy.optimize import linprog

def capped_simplex_project(v, target):
    n = len(v)
    if n == 0 or not np.all(np.isfinite(v)):
        return None
    if target <= 0:
        return np.zeros(n, dtype=np.float64)
    if target >= n:
        return np.ones(n, dtype=np.float64)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0).copy()
    r = target - np.sum(y)
    for _ in range(3):
        if abs(r) < 1e-12:
            break
        idx = 0
        while abs(r) > 1e-13 and idx < n:
            if r > 0:
                amt = min(r, 1.0 - y[idx])
                y[idx] += amt
                r -= amt
            else:
                amt = min(-r, y[idx])
                y[idx] -= amt
                r += amt
            idx += 1
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    if abs(np.sum(y) - target) > 1e-10:
        return None
    return y

def check_feasible(h, n):
    if h is None or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-10

def exact_score(h, dx):
    c = np.correlate(h, 1 - h, mode='full') * dx
    val = float(np.max(c))
    return val if np.isfinite(val) else None

def compute_grads(h):
    n = len(h)
    q = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    G = np.zeros((2 * n - 1, n), dtype=np.float64)
    for t in range(-(n - 1), n):
        row = t + n - 1
        j1lo = max(0, t)
        j1hi = min(n, n + t)
        if j1lo < j1hi:
            G[row, j1lo:j1hi] += 1.0 - h[j1lo - t:j1hi - t]
        j2lo = max(0, -t)
        j2hi = min(n, n - t)
        if j2lo < j2hi:
            G[row, j2lo:j2hi] -= h[j2lo + t:j2hi + t]
    if not (np.all(np.isfinite(G)) and np.all(np.isfinite(q))):
        return None, None
    return q, G

def slp_search(h0, dx, rng, n):
    h = h0.copy()
    best = h.copy()
    best_sc = exact_score(h, dx)
    if best_sc is None:
        return h
    max_it = 50 if n <= 128 else (25 if n <= 512 else 12)
    rho = 0.08
    rho_max = 0.25
    rho_min = 1e-5
    for _ in range(max_it):
        if rho < rho_min:
            break
        q, G = compute_grads(h)
        if q is None:
            break
        Q = float(np.max(q))
        m = 2 * n - 1
        A_ub = np.column_stack([G, -np.ones(m)])
        b_ub = -q
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        lb = np.maximum(-rho, -h)
        ubd = np.minimum(rho, 1.0 - h)
        bounds = list(zip(lb, ubd)) + [(None, None)]
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'maxiter': 300})
        except Exception:
            rho = max(rho_min, rho / 2.0)
            continue
        if res is None or not getattr(res, 'success', False) or res.x is None:
            rho = max(rho_min, rho / 2.0)
            continue
        if len(res.x) != n + 1 or not np.all(np.isfinite(res.x)):
            rho = max(rho_min, rho / 2.0)
            continue
        d = res.x[:n]
        z_val = float(res.x[-1])
        cand = h + d
        proj = capped_simplex_project(cand, n / 2.0)
        if proj is None or not check_feasible(proj, n):
            rho = max(rho_min, rho / 2.0)
            continue

        if sc is not None and sc < best_sc - 1e-12:
            best = proj.copy()
            best_sc = sc
        q_new = float(np.max(np.correlate(proj, 1 - proj, mode='full')))
        predicted = Q - z_val
        actual = Q - q_new
        h_before = h.copy()
        rho_before = rho

            rho = max(rho_min, rho / 2.0)
            continue
        ratio = actual / predicted if predicted > 1e-10 * max(1.0, abs(Q)) else 0.0
        accepted = actual > 1e-12 * max(1.0, abs(Q))
        if not accepted or ratio < 0.1:
            rho = max(rho_min, rho / 2.0)
        elif ratio > 0.75 and np.max(np.abs(proj - h_before)) >= 0.8 * rho_before:
            h = proj
            rho = min(rho_max, 1.5 * rho)
        else:
            h = proj
    return best

def pair_poll(h, dx, rng, n):
    best = h.copy()
    best_sc = exact_score(h, dx)
    if best_sc is None or n > 256:
        return best
    n_pairs = 24 if n <= 128 else 8
    for radius in (0.08, 0.025, 0.008):
        for _ in range(n_pairs):

            b = int(rng.integers(0, n))
            while b == a:
                b = int(rng.integers(0, n))
            slo = max(-best[a], best[b] - 1.0, -radius)
            shi = min(1.0 - best[a], best[b], radius)
            if shi <= slo:
                continue
            s_list = sorted(set([slo, shi, -radius, -radius / 2, radius / 2, radius]))
            s_list = [s for s in s_list if slo - 1e-14 <= s <= shi + 1e-14 and abs(s) > 1e-14]
            p_best_sc = best_sc
            p_best = None
            for s in s_list:
                cand = best.copy()
                cand[a] += s

                cand = capped_simplex_project(cand, n / 2.0)
                if cand is None or not check_feasible(cand, n):
                    continue

                if sc is not None and sc < p_best_sc - 1e-12:
                    p_best_sc = sc
                    p_best = cand.copy()
            if p_best is not None:
                best = p_best
                best_sc = p_best_sc
    return best
