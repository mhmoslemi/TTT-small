import numpy as np
import math

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1 or v.size < 2 or not np.all(np.isfinite(v)) or np.any(np.abs(v) > 1e12):
        return None
    n = v.size
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0).copy()
    residual = S - float(np.sum(y))
    for _ in range(2):
        if residual > 0:
            for i in range(n):
                if residual <= 1e-15:
                    break
                add = min(residual, 1.0 - y[i])
                y[i] += add
                residual -= add
        elif residual < 0:
            for i in range(n):
                if residual >= -1e-15:
                    break
                sub = min(-residual, y[i])
                y[i] -= sub
                residual += sub
    res = S - float(np.sum(y))
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    if abs(res) > 1e-10 * max(1.0, float(n)):
        return None
    return y

def is_feasible(h, n):
    if h.ndim != 1 or h.size != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(float(np.sum(h)) - n / 2.0) > 1e-10 * max(1.0, float(n)):
        return False
    return True

def exact_score(h):
    n = h.size
    if not is_feasible(h, n):
        return None
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def smooth_eval(h, tau):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    q = corr / float(n)
    exact = float(np.max(corr) * (2.0 / n))
    z = q / tau
    zmax = float(np.max(z))
    e = np.exp(z - zmax)
    s = float(np.sum(e))
    if s <= 0.0 or not math.isfinite(s):
        return None, None, exact
    p = e / s
    F = tau * (zmax + math.log(s))
    rp = p[::-1].copy()
    omh = 1.0 - h
    conv1 = np.convolve(p, omh, mode='full')
    conv2 = np.convolve(rp, h, mode='full')
    g = (conv1[n - 1:2 * n - 1] - conv2[n - 1:2 * n - 1]) / float(n)
    if not np.all(np.isfinite(g)):
        return None, None, exact
    if not math.isfinite(F):
        return None, None, exact
    return F, g, exact
