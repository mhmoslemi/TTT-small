import numpy as np
import math

def exact_score(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    sc = float(np.max(corr) * dx)
    return sc if math.isfinite(sc) and sc >= -1e-15 else float('inf')

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    S = n / 2.0
    if abs(float(np.sum(h)) - S) > 1e-9 or abs(math.fsum(float(x) for x in h) - S) > 1e-9:
        return False
    return True

def project_capped_simplex(v, n):
    S = n / 2.0
    v = np.clip(np.asarray(v, dtype=np.float64), -1e6, 1e6)
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

        res = S - math.fsum(float(x) for x in h)
        if abs(res) < 1e-14:
            break
        if res > 0:
            cap = 1.0 - h
            if np.sum(cap) > 0:
                h += res * cap / np.sum(cap)
        else:
            if np.sum(h) > 0:
                h += res * h / np.sum(h)

        rem = S - s_o + float(h[idx])
        if -1e-14 <= rem <= 1.0 + 1e-14:

        return None

        return None
    return h

def compute_q_G(h, n):
    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M, dtype=np.float64)
    G = np.zeros((M, n), dtype=np.float64)
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        ip = np.arange(d, d + m)
        iq = np.arange(0, m)
        q[2 * d - 1] = dx * np.sum(h[ip] * (1.0 - h[iq]))
        G[2 * d - 1, ip] += dx * (1.0 - h[iq])
        G[2 * d - 1, iq] -= dx * h[ip]
        q[2 * d] = dx * np.sum(h[iq] * (1.0 - h[ip]))
        G[2 * d, iq] += dx * (1.0 - h[ip])
        G[2 * d, ip] -= dx * h[iq]
    return q, G

def active_bundle_gradient(q, G, n):
    Cq = float(np.max(q))
    tol = max(1e-12, 1e-7 * max(1.0, abs(Cq)))
    active = np.where(q >= Cq - tol)[0]
    if len(active) > 12:
        active = active[np.argsort(q[active])[-12:]]
    p = len(active)
    if p == 0:
        return None
    U = G[active, :].copy()
    U -= U.mean(axis=1, keepdims=True)
    K = U @ U.T
    K = 0.5 * (K + K.T)
    alpha = np.full(p, 1.0 / p)
    for _ in range(40):
        z = K @ alpha
        j = int(np.argmin(z))
        if float(alpha[j]) - float(z[j]) / p <= 1e-12 and p > 0:
            pass
        gap = float(alpha @ z) - float(z[j])
        if gap <= 1e-12:
            break
        v = -alpha
        v[j] += 1.0
        denom = float(v @ (K @ v))
        if denom <= 1e-24:
            break
        gamma = max(0.0, min(1.0, -float(v @ z) / denom))
        alpha = alpha + gamma * v
        alpha = np.clip(alpha, 0.0, None)
        s = alpha.sum()
        if s > 0:
            alpha /= s
    g = alpha @ U
    if not np.all(np.isfinite(g)) or np.linalg.norm(g) < 1e-15:
        idx = int(np.argmax(q))
        g = G[idx, :] - float(np.mean(G[idx, :]))
        if not np.all(np.isfinite(g)):
            return None
    return g

def local_search(h0, rng, n, max_iter):

    if h is None:
        h = np.full(n, 0.5)
    cur = h.copy()

    best_sc = cur_sc

        g = active_bundle_gradient(q, G, n)

                        best_sc = sc

                sigma, T = 0.035, 0.00025

                    best = cur.copy()
                    best_sc = cur_sc

    return best, best_sc

def make_starts(n, parent_h, rng):

    def add_unique(v):
