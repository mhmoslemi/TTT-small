        n_starts, iters = 6, 40
    elif N <= 500:
        n_starts, iters = 4, 20
    else:
        n_starts, iters = 3, 8
    n_starts = min(n_starts, len(starts))
    for si in range(n_starts):
        h_s = starts[si].copy()
        if h_s.shape != (N,) or not np.all(np.isfinite(h_s)):
            h_s = project_box_sum(h_s, N)
        s_s = score_exact(h_s, N)
        if np.isfinite(s_s) and s_s < inc_s:
            inc_s = s_s
            inc_h[:] = h_s
        try:
            inc_h, inc_s = optimize_start(h_s, N, inc_h.copy(), inc_s, iters)
        except Exception:
            pass
    if inc_h.shape != (N,) or not np.all(np.isfinite(inc_h)):
        inc_h = np.full(N, 0.5)
    inc_h = project_box_sum(inc_h, N)
    final_s = score_exact(inc_h, N)
    if not np.isfinite(final_s):
        inc_h = np.full(N, 0.5)
        final_s = score_exact(inc_h, N)
    return inc_h, float(final_s), N
