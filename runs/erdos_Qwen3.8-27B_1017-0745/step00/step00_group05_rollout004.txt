import numpy as np
from scipy.optimize import minimize


def capped_simplex_project(v, target):
    v = np.asarray(v, dtype=np.float64).ravel()
    if not np.all(np.isfinite(v)):
        return None
    n = len(v)
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(50):
        r = target - np.sum(y)
        if abs(r) < 1e-10:
            break
        if r > 0:
            mask = y < 1.0 - 1e-12
            if not np.any(mask):
                break
            avail = 1.0 - y[mask]
            sa = np.sum(avail)
            if sa < 1e-14:
                break
            y[mask] += r * avail / sa
            y = np.clip(y, 0.0, 1.0)
        else:
            mask = y > 1e-12
            if not np.any(mask):
                break
            avail = y[mask]
            sa = np.sum(avail)
            if sa < 1e-14:
                break
            y[mask] += r * avail / sa
            y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)):
        return None
    if np.min(y) < -1e-10 or np.max(y) > 1.0 + 1e-10:
        return None
    if abs(np.sum(y) - target) > 1e-8:
        return None
    return y


def exact_score(h, dx):
    if h is None:
        return np.inf
    h = np.asarray(h, dtype=np.float64)
    if h.ndim != 1 or not np.all(np.isfinite(h)):
        return np.inf
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def is_feasible(h, n):
    if h is None or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.min(h) < -1e-10 or np.max(h) > 1.0 + 1e-10:
        return False
    if abs(np.sum(h) - n * 0.5) > 1e-9:
        return False
    return True


def smooth_value_gradient(h, tau, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = corr * dx
    m = np.max(s)
    e = np.exp((s - m) / tau)
    se = np.sum(e)
    p = e / se
    F = m + tau * np.log(se)
    grad = np.zeros(n, dtype=np.float64)
    for idx in range(2 * n - 1):
        L = idx - (n - 1)
        w = dx * p[idx]
        if w < 1e-16:
            continue
        j0 = max(0, L)
        j1 = min(n, L + n)
        if j1 > j0:
            grad[j0:j1] += w * (1.0 - h[j0 - L:j1 - L])
        j0b = max(0, -L)
        j1b = min(n, n - L)
        if j1b > j0b:
            grad[j0b:j1b] -= w * h[j0b + L:j1b + L]
    if not np.all(np.isfinite(grad)):
        return (np.inf, None)
    return (F, grad)


def mass_eq(x):
    return np.sum(x) - len(x) * 0.5


def mass_eq_jac(x):
    return np.ones(len(x))


def optimize_from_start(h0, dx, taus, maxiter):
    n = len(h0)
    target = n * 0.5
    h = h0.copy()
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': mass_eq, 'jac': mass_eq_jac}
    for tau in taus:
        try:
            res = minimize(smooth_value_gradient, h, args=(tau, dx),
                           method='SLSQP', jac=True, bounds=bounds,
                           constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-10, 'disp': False})
        except Exception:
            continue
        if res.x is not None and np.all(np.isfinite(res.x)):
            hp = capped_simplex_project(res.x, target)
            if hp is not None:
                F_new, _ = smooth_value_gradient(hp, tau, dx)
                F_cur, _ = smooth_value_gradient(h, tau, dx)
                if np.isfinite(F_new) and F_new <= F_cur + 1e-10:
                    h = hp
    return h
