def check_feasible(h, n, mass_tol=1e-10):

    return True

            break
        if hi - lo < 1e-14 * max(1.0, abs(lam)):
            break
        if s > S:
            lo = lam
        else:
            hi = lam
        lam = 0.5 * (lo + hi)

    return y
