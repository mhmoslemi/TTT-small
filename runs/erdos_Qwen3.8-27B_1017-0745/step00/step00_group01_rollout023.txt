import numpy as np

def check_feasible(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-15) or np.any(h > 1 + 1e-15):
        return False
    mass_err = abs(np.sum(h) - n / 2.0)
    if mass_err > 1e-10 * max(1, n):
        return False
    return True

def project_capped_simplex(z, n):
    s = n / 2.0
    if not np.all(np.isfinite(z)):
        return np.full(n, 0.5)
    z = np.clip(z, -1e12, 1e12)
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(z - mid, 0.0, 1.0)
        if np.sum(h) > s:
            lo = mid
        else:
            hi = mid
    mid = 0.5 * (lo + hi)
    h = np.clip(z - mid, 0.0, 1.0)
    # Residual correction
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) < 1e-14:
            break
        if r > 0:
            slack = 1.0 - h
            idx = np.argmax(slack)
            move = min(r, slack[idx])
        else:
            move = min(-r, h[idx])  # need most available mass
            # actually we need the coordinate with most h to take from
            idx = np.argmax(h)
            move = min(-r, h[idx])
        h[idx] += move if r > 0 else -move
        r = s - np.sum(h)
    h = np.clip(h, 0.0, 1.0)
    if not check_feasible(h, n):
        return np.full(n, 0.5)
    return h

def exact_score(h, n):

        return np.inf
    dx = 2.0 / n
    val = np.correlate(h, 1.0 - h, mode='full') * dx

    c_max = np.max(c)

def check_feasible(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def project_capped_simplex(z, n):
    s = n / 2.0

    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return np.full(n, 0.5)

        mid = 0.5 * (lo + hi)
        h = np.clip(z - mid, 0.0, 1.0)
        if np.sum(h) > s:
            lo = mid
        else:
            hi = mid
    mid = 0.5 * (lo + hi)
    h = np.clip(z - mid, 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 2):

            if move <= 0:

        return np.inf
    dx = 2.0 / n
    val = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(val))

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = float(np.max(c))
    a = np.exp((c - c_max) / tau)
    a_sum = float(np.sum(a))
    F = c_max + tau * np.log(a_sum)
    p = a / a_sum
    return F, p, c

def smooth_gradient(h, n, tau):
    F, p, c = smooth_value_and_weights(h, n, tau)
    if not np.isfinite(F) or not np.all(np.isfinite(p)):
        return None, None
    dx = 2.0 / n
    grad = np.zeros(n)
    n1 = n - 1
    for idx in range(2 * n - 1):
        pd = p[idx]
        if pd < 1e-16:
            continue
        d = idx - n1
        if d >= 0:
            m = n - d
            if m > 0:
                grad[d:d + m] += pd * (1.0 - h[0:m])
                grad[0:m] -= pd * h[d:d + m]
        else:
            D = -d
            m = n - D
            if m > 0:
                grad[0:m] += pd * (1.0 - h[D:D + m])
                grad[D:D + m] -= pd * h[0:m]
    grad *= dx
    return grad, F
