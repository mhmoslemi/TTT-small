def smooth_value_gradient(h, tau, dx, n):
    s = np.correlate(h, 1.0 - h, mode='full') * dx
    smax = np.max(s)
    ex = np.exp((s - smax) / tau)
    tot = np.sum(ex)
    if tot <= 0:
        return 1e10, np.zeros(n)
    F = smax + tau * np.log(tot)
    p = ex / tot
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        pk = p[m] * dx
        js, je = max(0, k), min(n, n + k)
        if js < je:
            idx = np.arange(js, je)
            grad[idx] += pk * (1.0 - h[idx - k])
        js, je = max(0, -k), min(n, n - k)
        if js < je:
            idx = np.arange(js, je)
            grad[idx] -= pk * h[idx + k]
    if not (np.isfinite(F) and np.all(np.isfinite(grad))):
        return 1e10, np.zeros(n)
    return F, grad

def mass_residual(h):
    return np.sum(h) - len(h) / 2.0

def mass_jacobian(h):
    return np.ones(len(h))

def check_feasible(h, n):
    if h is None or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.min(h) < -1e-12 or np.max(h) > 1 + 1e-12:
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10:
        return False
    return True

def run_single_start(h0, n, dx, bounds, cons, lses):
    target = n / 2.0
    best = h0.copy()
    best_score = exact_score(h0, dx)
    current = h0.copy()
    for tau in lses:
        try:
            res = minimize(smooth_value_gradient, current, args=(tau, dx, n),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-10, 'disp': False})
            x_new = res.x
            if x_new.shape == (n,) and np.all(np.isfinite(x_new)):
                x_new = capped_simplex_project(x_new, target)
                if x_new is not None and check_feasible(x_new, n):
                    sc = exact_score(x_new, dx)
                    if sc < best_score - 1e-12:
                        best = x_new.copy()
                        best_score = sc
                    F_old, _ = smooth_value_gradient(current, tau, dx, n)
                    F_new, _ = smooth_value_gradient(x_new, tau, dx, n)
                    if F_new <= F_old + 1e-10:
                        current = x_new
        except Exception:
            pass
    return best, best_score

def run(seed=42, **kwargs):
    n = 70
    parent = None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64)
        if iv.ndim == 1 and 2 <= len(iv) <= 999:
            n = len(iv)
            if np.all(np.isfinite(iv)) and np.min(iv) >= -1e-10 and np.max(iv) <= 1 + 1e-10:
                parent = iv.copy()
    except Exception:
        pass
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': mass_residual, 'jac': mass_jacobian}]
    lses = [0.02, 0.005, 0.001, 0.0002, 0.00005, 0.00001]
    rng = np.random.default_rng(seed)

    parent_proj = None
    if parent is not None:
        parent_proj = capped_simplex_project(parent, target)

    inc_vec = None
    inc_score = float('inf')

    starts = []
    if parent_proj is not None:
        starts.append(parent_proj)
        starts.append(parent_proj[::-1].copy())
        for sig in [0.04, 0.10, 0.22, 0.40]:
            p = parent_proj + sig * rng.standard_normal(n)
            pp = capped_simplex_project(p, target)
            if pp is not None:
                starts.append(pp)
    u = rng.random(n)
    pu = capped_simplex_project(u, target)
    if pu is not None:
        starts.append(pu)
    b = rng.beta(0.5, 0.5, size=n)
    pb = capped_simplex_project(b, target)
    if pb is not None:
        starts.append(pb)
    starts.append(np.full(n, 0.5))

    for s in starts:
        if not check_feasible(s, n):
            continue
        sc = exact_score(s, dx)
        if sc < inc_score - 1e-12:
            inc_score = sc
            inc_vec = s.copy()
        best, bs = run_single_start(s, n, dx, bounds, cons, lses)
        if bs < inc_score - 1e-12:
            inc_score = bs
            inc_vec = best.copy()

    center = inc_vec if inc_vec is not None else np.full(n, 0.5)
    for sig in [0.02, 0.08, 0.20]:
        p = center + sig * rng.standard_normal(n)
        pp = capped_simplex_project(p, target)
        if pp is not None and check_feasible(pp, n):
            best, bs = run_single_start(pp, n, dx, bounds, cons, lses)
            if bs < inc_score - 1e-12:
                inc_score = bs
                inc_vec = best.copy()
                center = inc_vec.copy()

    if inc_vec is None:
        inc_vec = np.full(n, 0.5)

    final = capped_simplex_project(inc_vec, target)
    if final is None or not check_feasible(final, n):
        final = np.full(n, 0.5)
    c5 = exact_score(final, dx)
    return final, c5, n
