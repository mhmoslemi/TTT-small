def compute_lags(h, dx):

    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    q[n-1] = dx * np.dot(h, 1.0 - h)

    epi_bounds_f = [(0.0, 1.0)] * n + [(0.0, None)]
    epi_cons_f = [eq_con, {'type': 'ineq', 'fun': epi_ineq, 'jac': epi_ineq_jac, 'args': (dx,)}]
    try:
        rf = minimize(epi_obj_grad, xf, args=(dx,), method='SLSQP',
                      jac=True, bounds=epi_bounds_f, constraints=epi_cons_f,
                      options={'ftol': 1e-10, 'maxiter': 200})
        if hasattr(rf, 'x') and rf.x is not None and np.all(np.isfinite(rf.x[:n])):
            hx = project_simplex(rf.x[:n], n)
            ch = exact_score(hx, dx)
            if ch < best_c5 - 1e-12:
                best_h = hx.copy()
                best_c5 = ch
    except Exception:
        pass
    best_h = project_simplex(best_h, n)
    best_c5 = exact_score(best_h, dx)
    return (best_h, best_c5, n)
