        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - y.sum()
    for _ in range(3):
        if r > 1e-14:
            for i in range(n):
                if r <= 1e-14:
                    break
                y[i] += min(r, 1.0 - y[i])
                r -= min(r, 1.0 - y[i])
        elif r < -1e-14:
            for i in range(n):
                if r >= -1e-14:
                    break
                y[i] -= min(-r, y[i])
                r += min(-r, y[i])
        else:
            break
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(y.sum() - target) > 1e-9:
        return None
    return y

def check_feasible(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if h.min() < -1e-12 or h.max() > 1.0 + 1e-12:
        return False
    if abs(h.sum() - n / 2.0) > 1e-9:
        return False
    return True

def compute_score(h, dx):
    val = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
    return val if np.isfinite(val) else None

def compute_lags_grad(h, n):
    q = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    G = np.zeros((2 * n - 1, n), dtype=np.float64)
    j = np.arange(n)
    for k in range(2 * n - 1):
        t = k - (n - 1)
        idx1 = j - t
        m1 = (idx1 >= 0) & (idx1 < n)
        t1 = np.where(m1, 1.0 - h[idx1], 0.0)
        idx2 = j + t
        m2 = (idx2 >= 0) & (idx2 < n)
        t2 = np.where(m2, h[idx2], 0.0)
        G[k] = t1 - t2
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None, None
    return q, G

def slp_search(h0, dx, n, rng):
    h = h0.copy()
    best = h.copy()
    best_score = compute_score(h, dx)
    rho = 0.08
    rho_max = 0.25
    rho_min = 1e-5
    max_iter = 50 if n <= 128 else (25 if n <= 512 else 12)
    for _ in range(max_iter):
        if rho < rho_min:
            break
        q, G = compute_lags_grad(h, n)
        if q is None:
            rho = max(rho_min, rho * 0.5)
            continue
        Q = float(np.max(q))
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.hstack([G, -np.ones((2 * n - 1, 1))])
        b_ub = -q
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        bounds = []
        for i in range(n):
            bounds.append((max(-rho, -h[i]), min(rho, 1.0 - h[i])))
        bounds.append((None, None))
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'maxiter': 300})
        except Exception:
            rho = max(rho_min, rho * 0.5)
            continue
        if not res.success or res.x is None or len(res.x) != n + 1:
            rho = max(rho_min, rho * 0.5)
            continue
        if not np.all(np.isfinite(res.x)):
            rho = max(rho_min, rho * 0.5)
            continue
        d = res.x[:n]
        z = float(res.x[-1])
        cand = capped_simplex_project(h + d, n / 2.0)
        if cand is None or not check_feasible(cand, n):
            rho = max(rho_min, rho * 0.5)
            continue
        s_new = compute_score(cand, dx)
        if s_new is None:
            rho = max(rho_min, rho * 0.5)
            continue
        h_before = h.copy()
        if s_new < best_score - 1e-12:
            best = cand.copy()
            best_score = s_new
        Q_new = float(np.max(np.correlate(cand, 1.0 - cand, mode='full')))
        predicted = Q - z
        actual = Q - Q_new
        ref = max(1.0, abs(Q))
        if predicted > 1e-10 * ref:
            ratio = actual / predicted
        else:
            ratio = 0.0
        accepted = actual > 1e-12 * ref
        if accepted and np.isfinite(ratio) and ratio > 0.75:
            if float(np.max(np.abs(cand - h_before))) >= 0.8 * rho:
                rho = min(rho_max, 1.5 * rho)
        elif (not accepted) or (np.isfinite(ratio) and ratio < 0.1):
            rho = max(rho_min, rho * 0.5)
        if accepted:
            h = cand
    return best

def pairwise_poll(h0, dx, n, rng):
    h = h0.copy()
    best = h.copy()
    best_score = compute_score(h, dx)
    radii = [0.08, 0.025, 0.008]
    n_pairs = 24 if n <= 128 else (8 if n <= 256 else 0)
    for r in radii:
        for _ in range(n_pairs):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            for _ tries in range(10):
                if b != a:
                    break
                b = int(rng.integers(0, n))
            if b == a:
                continue
            s_lo = max(-h[a], h[b] - 1.0, -r)
            s_hi = min(1.0 - h[a], h[b], r)
            if s_lo >= s_hi:
                continue
            cs_list = [s_lo, s_hi]
            for sv in (-r, -0.5 * r, 0.5 * r, r):
                sv_c = np.clip(sv, s_lo, s_hi)
                if all(abs(sv_c - x) > 1e-14 for x in cs_list):
                    cs_list.append(float(sv_c))
            bp_score = best_score
            bp_cand = None
            for s in cs_list:
                if abs(s) <= 1e-14:
                    continue
                hh = h.copy()
                hh[a] += s
                hh[b] -= s
                proj = capped_simplex_project(hh, n / 2.0)
                if proj is None or not check_feasible(proj, n):
                    continue
                sc = compute_score(proj, dx)
                if sc is not None and sc < bp_score - 1e-12:
                    bp_score = sc
                    bp_cand = proj
            if bp_cand is not None:
                h = bp_cand
                if bp_score < best_score - 1e-12:
                    best = h.copy()
                    best_score = bp_score
    return best

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent = None
    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            pv = np.array(initial_h_values, dtype=np.float64).ravel()
            if 2 <= len(pv) <= 999 and np.all(np.isfinite(pv)) and float(np.max(np.abs(pv))) <= 1e6:
                parent = pv
    except Exception:
        parent = None
    if parent is not None:
        n = len(parent)
    else:
        n = 70
    dx = 2.0 / n
    target = n / 2.0
    u = np.full(n, 0.5)
    starts = []
    if parent is not None:
        pp = capped_simplex_project(parent, target)
        if pp is not None and check_feasible(pp, n):
            starts.append(pp)
        starts.append(capped_simplex_project(parent[::-1].copy(), target))
        for sc in (0.035, 0.12):
            starts.append(capped_simplex_project(parent + sc * rng.standard_normal(n), target))
        starts.append(capped_simplex_project(rng.random(n), target))
        starts.append(capped_simplex_project(rng.beta(0.5, 0.5, size=n), target))
    starts.append(u.copy())
    if parent is None:
        for _ in range(3):
            starts.append(capped_simplex_project(rng.random(n), target))
        for _ in range(2):
            starts.append(capped_simplex_project(rng.beta(0.5, 0.5, size=n), target))
    scored_starts = []
    seen = []
    for s in starts:
        if s is None or not check_feasible(s, n):
            continue
        dup = False
        for v in seen:
            if float(np.max(np.abs(s - v))) < 1e-10:
                dup = True
                break
        if not dup:
            seen.append(s.copy())
            scored_starts.append(s)
    fallback = u.copy()
    fallback_score = compute_score(fallback, dx)
    if parent is not None:
        pp = capped_simplex_project(parent, target)
        if pp is not None and check_feasible(pp, n):
            ps = compute_score(pp, dx)
            if ps is not None and ps < fallback_score:
                fallback = pp.copy()
                fallback_score = ps
    incumbent = fallback.copy()
    incumbent_score = fallback_score
    for s in scored_starts:
        sc = compute_score(s, dx)
        if sc is not None and sc < incumbent_score - 1e-12:
            incumbent = s.copy()
            incumbent_score = sc
        r1 = slp_search(s, dx, n, rng)
        rs1 = compute_score(r1, dx)
        if rs1 is not None and rs1 < incumbent_score - 1e-12:
            incumbent = r1.copy()
            incumbent_score = rs1
        r2 = pairwise_poll(r1, dx, n, rng)
        rs2 = compute_score(r2, dx)
        if rs2 is not None and rs2 < incumbent_score - 1e-12:
            incumbent = r2.copy()
            incumbent_score = rs2
        if incumbent_score <= 0.3808:
            break
    for sc in (0.025, 0.08):
        rs = capped_simplex_project(incumbent + sc * rng.standard_normal(n), target)
        if rs is not None and check_feasible(rs, n):
            r1 = slp_search(rs, dx, n, rng)
            s1 = compute_score(r1, dx)
            if s1 is not None and s1 < incumbent_score - 1e-12:
                incumbent = r1.copy()
                incumbent_score = s1
            r2 = pairwise_poll(r1, dx, n, rng)
            s2 = compute_score(r2, dx)
            if s2 is not None and s2 < incumbent_score - 1e-12:
                incumbent = r2.copy()
                incumbent_score = s2
        if incumbent_score <= 0.3808:
            break
    final = capped_simplex_project(incumbent, target)
    if final is None or not check_feasible(final, n):
        final = fallback.copy()
    final_score = compute_score(final, dx)
    if final_score is None:
        final = u.copy()
        final_score = compute_score(final, dx)
    return (final, final_score, n)
