    m = 2 * n - 1
    b = np.zeros(m)
    for ell in range(-(n - 1), n):
        v = 0.0
        if 0 <= i - ell < n:
            v += (1.0 - h[i - ell])
        if 0 <= j - ell < n:
            v -= (1.0 - h[j - ell])
        if 0 <= i + ell < n:
            v -= h[i + ell]
        if 0 <= j + ell < n:
            v += h[j + ell]
        b[ell + (n - 1)] = dx * v
    c = np.zeros(m)
    c[n - 1] = -2.0 * dx
    c[n - 1 + (j - i)] = dx
    c[n - 1 + (i - j)] = dx
    return b, c

def metropolis_chain(h0, n, rng, n_prop):
    dx = 2.0 / n
    h = h0.copy().astype(np.float64)
    _, sc = score_and_lags(h, n)
    best_h, best_sc = h.copy(), sc
    stages = [(0.20, 0.50), (0.08, 0.25), (0.03, 0.12), (0.01, 0.06)]
    for tm, step in stages:
        T = tm * dx
        for _ in range(n_prop):
            i = int(rng.integers(n))
            j = int(rng.integers(n - 1))
            if j >= i:
                j += 1
            t = rng.uniform(-step, step)
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if t < lo or t > hi or lo >= hi:
                continue
            ht = h.copy()
            ht[i] += t
            ht[j] -= t
            ht = repair_small(ht, n)
            if ht is None:
                continue
            _, sct = score_and_lags(ht, n)
            if sct >= float('inf'):
                continue
            d = sct - sc
            if d <= 0 or rng.random() < math.exp(-min(d / T, 700.0)):
                h, sc = ht, sct
                if sct < best_sc - 1e-12:
                    best_sc, best_h = sct, h.copy()
    return best_h, best_sc

def local_refine(h0, n, rng):
    h = h0.copy().astype(np.float64)
    q, sc = score_and_lags(h, n)
    max_pairs = 400 if n > 64 else n * (n - 1) // 2
    for sweep in range(2):
        committed = False
        sc_start = sc
        if n <= 64:
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
            rng.shuffle(pairs)
        else:
            pairs, att = [], 0
            while len(pairs) < max_pairs and att < 10 * max_pairs:
                i, j = int(rng.integers(n)), int(rng.integers(n))
                if i == j:
                    att += 1
                    continue
                if i > j:
                    i, j = j, i
                if (i, j) not in pairs:
                    pairs.append((i, j))
                att += 1
        for (i, j) in pairs:
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if hi - lo < 1e-14:
                continue
            b, c = build_coeffs(h, n, i, j)
            grid = np.linspace(lo, hi, 17)
            grid = np.append(grid, 0.0)
            for kk in range(1, 6):
                grid = np.append(grid, [lo * (0.5 ** kk), hi * (0.5 ** kk)])
            grid = np.sort(np.unique(grid))
            grid = grid[(grid >= lo - 1e-14) & (grid <= hi + 1e-14)]
            for _ in range(2):
                pred = q[None, :] + grid[:, None] * b[None, :] + (grid[:, None]) ** 2 * c[None, :]
                ps = np.max(pred, axis=1)
                bi = int(np.argmin(ps))
                if bi == 0:
                    br = grid[1:3]
                elif bi == len(grid) - 1:
                    br = grid[-3:-1]
                else:
                    br = grid[bi - 1:bi + 2]
                grid = np.sort(np.unique(np.concatenate([grid, np.linspace(br[0], br[1], 9)[1:-1]])))
                grid = grid[(grid >= lo - 1e-14) & (grid <= hi + 1e-14)]
            pred = q[None, :] + grid[:, None] * b[None, :] + (grid[:, None]) ** 2 * c[None, :]
            ps = np.max(pred, axis=1)
            bi = int(np.argmin(ps))
            if ps[bi] >= sc - 1e-12:
                continue
            ht = h.copy()
            ht[i] += grid[bi]
            ht[j] -= grid[bi]
            ht = repair_small(ht, n)
            if ht is None:
                continue
            qt, sct = score_and_lags(ht, n)
            if sct < sc - 1e-12:
                h, q, sc = ht, qt, sct
                committed = True
        if not committed or (sc_start - sc) < 1e-10:
            break
    return h, sc

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    n = 45
    parent_h = None
    parent = globals().get('initial_h_values', None)
    if parent is not None:
        parent = np.asarray(parent, dtype=np.float64)
        if parent.ndim == 1 and 2 <= len(parent) < 1000:
            n = len(parent)
            if np.all(np.isfinite(parent)) and np.max(np.abs(parent)) <= 1e6:
                ph = np.clip(parent, 0.0, 1.0).copy()
                S = n / 2.0
                r = S - float(np.sum(ph))
                for idx in range(n):
                    if abs(r) < 1e-13:
                        break
                    if r > 0:
                        a = min(r, 1.0 - ph[idx])
                        ph[idx] += a
                        r -= a
                    else:
                        a = min(-r, ph[idx])
                        ph[idx] -= a
                        r += a
                parent_h = ph
            else:
                parent_h = np.full(n, 0.5)
    if parent_h is None:
        np_ = kwargs.get('n_points', None)
        if isinstance(np_, (int, np.integer)) and 2 <= np_ <= 999:
            n = int(np_)
        parent_h = np.full(n, 0.5)
    starts = [parent_h.copy(), parent_h[::-1].copy(), (1.0 - parent_h).copy(), (1.0 - parent_h[::-1]).copy()]
    for sd in [0.08, 0.25]:
        p = project_feasible(parent_h + sd * rng.standard_normal(n), n)
        if p is not None:
            starts.append(p)
    for fn in [rng.random, lambda: rng.beta(0.5, 0.5, n)]:
        p = project_feasible(fn(), n)
        if p is not None:
            starts.append(p)
    unique = []
    for s in starts:
        if not np.all(np.isfinite(s)):
            continue
        if all(np.max(np.abs(s - u)) > 1e-6 for u in unique):
            unique.append(s)
    fallback_h = parent_h.copy()
    if not check_feasible(fallback_h, n):
        fallback_h = np.full(n, 0.5)
    fallback_sc = score_and_lags(fallback_h, n)[1]
    best_h, best_sc = fallback_h.copy(), fallback_sc
    n_chains = len(unique) if n <= 64 else min(6, len(unique)) if n <= 256 else min(4, len(unique))
    n_prop = 180 if n <= 64 else 40 if n <= 256 else 12
    for si in range(min(n_chains, len(unique))):
        try:
            s = unique[si]
            _, ss = score_and_lags(s, n)
            if ss < best_sc - 1e-12:
                best_sc, best_h = ss, s.copy()
            bh, bsc = metropolis_chain(s, n, rng, n_prop)
            if bsc < best_sc - 1e-12:
                best_sc, best_h = bsc, bh
        except Exception:
            pass
    n_restart = 2 if n <= 64 else 1
    sigmas = [0.08, 0.25] if n <= 64 else [0.15]
    for si in range(n_restart):
        try:
            p = project_feasible(best_h + sigmas[si] * rng.standard_normal(n), n)
            if p is not None:
                bh, bsc = metropolis_chain(p, n, rng, n_prop)
                if bsc < best_sc - 1e-12:
                    best_sc, best_h = bsc, bh
        except Exception:
            pass
    try:
        rh, rsc = local_refine(best_h, n, rng)
        if rsc < best_sc - 1e-12:
            best_sc, best_h = rsc, rh
    except Exception:
        pass
    try:
        fh = repair_small(best_h.copy(), n)
        if fh is None or not check_feasible(fh, n):
            fh = fallback_h
        s_f = score_and_lags(fh, n)[1]
        if s_f >= float('inf') or s_f > fallback_sc:
            fh, s_f = fallback_h, fallback_sc
    except Exception:
        fh = np.full(n, 0.5)
        s_f = score_and_lags(fh, n)[1]
    fh = np.clip(fh, 0.0, 1.0).astype(np.float64)
    s_f = float(np.max(np.correlate(fh, 1.0 - fh, mode='full')) * (2.0 / n))
    return fh, s_f, int(n)
