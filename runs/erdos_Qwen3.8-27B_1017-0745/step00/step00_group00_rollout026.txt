import numpy as np
from scipy.optimize import minimize

def project_box_sum(v, S):
    n = len(v)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    resid = S - np.sum(y)
    # correct residual
    if resid > 0:
        idxs = np.argsort(1.0 - y)[::-1]
        for idx in idxs:
            add = min(resid, 1.0 - y[idx])
            y[idx] += add
            resid -= add
            if resid < 1e-12:
                break
    elif resid < 0:
        idxs = np.argsort(y)
        for idx in idxs:
            sub = min(-resid, y[idx])
            y[idx] -= sub
            resid += sub
            if resid > -1e-12:
                break
    return y

def check_feasible(h, n):
    if h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score(h):
    if not np.all(np.isfinite(h)):
        return np.inf
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def obj(z):
    return z[-1]

def obj_jac(z):
    n = len(z) - 1
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def ineq(z):
    n = len(z) - 1
    h = z[:n]
    t = z[n]
    c = np.correlate(h, 1.0 - h, mode='full')
    return t - c / n

def ineq_jac(z):
    n = len(z) - 1
    h = z[:n]
    nlag = 2 * n - 1
    J = np.zeros((nlag, n + 1))
    for m in range(nlag):
        k = m - (n - 1)
        J[m, n] = 1.0
        # dQ_k/dh[r] = (1-h[r+k])*I(0<=r+k<n) - h[r-k]*I(0<=r-k<n)
        for r in range(n):
            val = 0.0
            if 0 <= r + k < n:
                val += (1.0 - h[r + k])
            if 0 <= r - k < n:
                val -= h[r - k]
            J[m, r] = -val / n
    return J

def eq(z):
    n = len(z) - 1
    return np.sum(z[:n]) / n - 0.5

def eq_jac(z):
    n = len(z) - 1
    g = np.zeros(len(z))
    g[:n] = 1.0 / n
    return g

def solve_slsqp(h0, n):
    c0 = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = min(1.0, float(np.max(c0)) / n + 1e-4)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    constraints = [
        {'type': 'ineq', 'fun': ineq, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq, 'jac': eq_jac}
    ]
    res = minimize(obj, z0, method='SLSQP', jac=obj_jac,
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': 300, 'ftol': 1e-10, 'disp': False})
    return res

import numpy as np
from scipy.optimize import minimize

def project_box_sum(v, S):
    n = len(v)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if resid > 1e-13:
        idxs = np.argsort(1.0 - y)[::-1]
        for idx in idxs:
            add = min(resid, 1.0 - y[idx])
            y[idx] += add
            resid -= add
            if resid < 1e-13:
                break
    elif resid < -1e-13:
        idxs = np.argsort(y)
        for idx in idxs:
            sub = min(-resid, y[idx])
            y[idx] -= sub
            resid += sub
            if resid > -1e-13:
                break
    return y

def check_feasible(h, n):
    if h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score(h):
    if not np.all(np.isfinite(h)):
        return np.inf
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def obj_fn(z):
    return z[-1]

def obj_jac_fn(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def ineq_fn(z):
    n = len(z) - 1
    h = z[:n]
    t = z[n]
    c = np.correlate(h, 1.0 - h, mode='full')
    return t - c / n

def ineq_jac_fn(z):
    n = len(z) - 1
    h = z[:n]
    nlag = 2 * n - 1
    J = np.zeros((nlag, n + 1))
    inv_n = 1.0 / n

        J[m, n] = 1.0

                g -= (1.0 - h[r - k]) * inv_n

            J[m, r] = g

def eq_fn(z):
    n = len(z) - 1
    return np.sum(z[:n]) / n - 0.5

def eq_jac_fn(z):
    n = len(z) - 1
    g = np.zeros(len(z))
    g[:n] = 1.0 / n
    return g

def solve_slsqp(h0, n):
    c0 = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = min(1.0, float(np.max(c0)) / n + 1e-4)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * (n + 1)
    cons = [
        {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac_fn},
        {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac_fn}
    ]
    return minimize(obj_fn, z0, method='SLSQP', jac=obj_jac_fn,
                    bounds=bounds, constraints=cons,
                    options={'maxiter': 300, 'ftol': 1e-10, 'disp': False})
