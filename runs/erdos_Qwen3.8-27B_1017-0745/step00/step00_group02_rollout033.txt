                best_h = ht
        if best_h is None or best_sc >= C - 1e-12:
            break
        h = best_h.copy()
        q, C = compute_scores(h, n)
    return h, C

def run(seed=42, **kwargs):
    n = None
    parent = None
    fallback_h = None
    try:
        raw = globals().get('initial_h_values', None)
        if raw is not None:
            arr = np.asarray(raw, dtype=np.float64).ravel()
            if arr.ndim == 1 and 2 <= len(arr) <= 999:
                n = len(arr)
                dx = 2.0 / n
                S = float(n) / 2.0
                if np.all(np.isfinite(arr)) and np.all(np.abs(arr) <= 1e6):
                    if np.all(arr >= -1e-12) and np.all(arr <= 1.0 + 1e-12) and abs(np.sum(arr) - S) <= 1e-12:
                        parent = arr.copy()
                        fallback_h = parent.copy()
                    else:
                        proj = project_K(np.clip(arr, 0.0, 1.0), n, S)
                        if proj is not None:
                            parent = proj
                            fallback_h = proj.copy()
                if fallback_h is None:
                    fallback_h = np.full(n, 0.5, dtype=np.float64)
    except Exception:
        pass
    if n is None:
        nk = kwargs.get('n_points', None)
        if isinstance(nk, int) and not isinstance(nk, bool) and 2 <= nk <= 999:
            n = nk
        else:
            n = 45
        fallback_h = np.full(n, 0.5, dtype=np.float64)
    dx = 2.0 / n
    S = float(n) / 2.0
    if fallback_h is None:
        fallback_h = np.full(n, 0.5, dtype=np.float64)
    fb_score = float(np.max(np.correlate(fallback_h, 1.0 - fallback_h, mode='full')) * dx)
    best_h = fallback_h.copy()
    best_score = fb_score
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    starts = []
    if parent is not None:
        starts.append(parent.copy())
        starts.append(1.0 - parent.copy())
        starts.append(parent[::-1].copy())
        for sd in [0.08, 0.25]:
            p = project_K(parent + rng.normal(0, sd, n), n, S)
            if p is not None:
                starts.append(p)
        p = project_K(rng.beta(0.5, 0.5, n), n, S)
        if p is not None:
            starts.append(p)
    else:
        starts.append(np.full(n, 0.5, dtype=np.float64))
        p = project_K(rng.uniform(0, 1, n), n, S)
        if p is not None:
            starts.append(p)
        p = project_K(rng.beta(0.5, 0.5, n), n, S)
        if p is not None:
            starts.append(p)
        m = int(rng.integers(1, max(1, min(4, n // 2)) + 1))
        ph = rng.uniform(0, 2 * math.pi)
        ca = 0.5 + 0.35 * np.cos(2 * math.pi * m * (np.arange(n) + 0.5) / n + ph)
        p = project_K(ca, n, S)
        if p is not None:
            starts.append(p)
    max_starts = 6 if n <= 64 else (4 if n <= 256 else 3)
    deduped = []
    for s in starts[:2 * max_starts]:
        dup = False
        for d in deduped:
            if float(np.max(np.abs(s - d))) <= 1e-8:
                dup = True
                break
        if not dup:
            deduped.append(s)
        if len(deduped) >= max_starts:
            break
    starts = deduped
    max_iters = 40 if n <= 64 else (20 if n <= 256 else 8)
    for st in starts:
        try:
            _, sc = compute_scores(st, n)
            if sc is not None and sc < best_score - 1e-12:
                best_h = st.copy()
                best_score = sc
            ho, Co = optimize_start(st, n, dx, S, max_iters)
            if Co < best_score - 1e-12:
                best_h = ho.copy()
                best_score = Co
        except Exception:
            continue
    _, fs = compute_scores(best_h, n)
    if fs is None or fs > fb_score + 1e-6:
        best_h = fallback_h.copy()
        fs = fb_score
    final = float(np.max(np.correlate(best_h, 1.0 - best_h, mode='full')) * dx)
    return np.asarray(best_h, dtype=np.float64).copy(), final, int(n)
