import numpy as np
import cvxpy as cp

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / N
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def check_feasible(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return False
    if h.min() < -1e-10 or h.max() > 1.0 + 1e-10:
        return False
    if abs(h.sum() - N / 2.0) > 1e-10 * max(1, N):
        return False
    return True

def project_box_sum(z, N):
    m = N / 2.0
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (N,) or not np.all(np.isfinite(z)) or np.max(np.abs(z)) > 1e100:
        return np.full(N, 0.5)
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(80):
        theta = 0.5 * (lo + hi)
        v = np.clip(z - theta, 0.0, 1.0)
        if v.sum() > m:
            lo = theta
        else:
            hi = theta
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    e = m - h.sum()
    for _ in range(2):
        if e > 0:
            for i in range(N):
                if e <= 0:
                    break
                s = min(e, 1.0 - h[i])
                h[i] += s
                e -= s
        elif e < 0:
            for i in range(N):
                if e >= 0:
                    break
                s = min(-e, h[i])
                h[i] -= s
                e += s
        else:
            break
    if abs(e) > 1e-10 * max(1, N) or not np.all(np.isfinite(h)):
        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)

def lag_values_gradients(h, N):
    q = np.correlate(h, 1.0 - h, mode='full')
    G = np.zeros((2 * N - 1, N))
    for row in range(2 * N - 1):
        k = row - (N - 1)
        if k >= 0:
            L = N - k
            G[row, k:k + L] += (1.0 - h[0:L])
            G[row, 0:L] -= h[k:k + L]
        else:
            s = -k
            L = N - s
            G[row, 0:L] += (1.0 - h[s:s + L])
            G[row, s:s + L] -= h[0:L]
    return q, G

def build_solver(N, dx):
    nl = 2 * N - 1
    d = cp.Variable(N)
    u = cp.Variable()
    t = cp.Variable()
    x_p = cp.Parameter(N)
    r0_p = cp.Parameter()
    J0_p = cp.Parameter(N)
    rnz_p = cp.Parameter(nl - 1)
    Jnz_p = cp.Parameter((nl - 1, N))
    R_p = cp.Parameter(nonneg=True)
    cts = [
        x_p + d >= 0,
        x_p + d <= 1,
        cp.sum(d) == 0,
        cp.norm(d, 2) <= R_p,
        u >= rnz_p + Jnz_p @ d,
        t >= u + dx * cp.sum_squares(d),
        t >= r0_p + J0_p @ d,
    ]
    prob = cp.Problem(cp.Minimize(t), cts)
    return d, u, t, x_p, r0_p, J0_p, rnz_p, Jnz_p, R_p, prob

def run_local_path(h0, N, dx, iters, rng):
    cur = h0.astype(np.float64).copy()
    Fc = score_exact(cur, N)
    if not np.isfinite(Fc):
        return cur, Fc
    nl = 2 * N - 1
    try:
        d_v, u_v, t_v, x_p, r0_p, J0_p, rnz_p, Jnz_p, R_p, prob = build_solver(N, dx)
        have_solver = True
    except Exception:
        have_solver = False
        return cur, Fc
    R0 = min(1.0, np.sqrt(N))
    Rmax = min(4.0, np.sqrt(N))
    Rmin = 1e-4
    R = R0
    cf = 0
    for _ in range(iters):
        if R < Rmin:
            break
        q, G = lag_values_gradients(cur, N)
        if not (np.all(np.isfinite(q)) and np.all(np.isfinite(G))):
            R *= 0.5
            cf += 1
            if cf >= 8:
                break
            continue
        ra = q * dx
        Ja = G * dx
        r0 = ra[N - 1]
        J0 = Ja[N - 1]
        idx = [i for i in range(nl) if i != N - 1]
        rnz = ra[idx]
        Jnz = Ja[idx]
        x_p.value = cur
        r0_p.value = r0
        J0_p.value = J0
        rnz_p.value = rnz
        Jnz_p.value = Jnz
        R_p.value = R
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=200,
                       abstol=1e-9, reltol=1e-9, feastol=1e-9)
        except Exception:
            R *= 0.5
            cf += 1
            if cf >= 8:
                break
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            R *= 0.5
            cf += 1
            if cf >= 8:
                break
            continue
        dv = d_v.value
        if dv is None or dv.shape != (N,) or not np.all(np.isfinite(dv)):
            R *= 0.5
            cf += 1
            if cf >= 8:
                break
            continue
        trial = project_box_sum(cur + dv, N)
        if not check_feasible(trial, N):
            R *= 0.5
            cf += 1
            if cf >= 8:
                break
            continue
        de = trial - cur
        if np.linalg.norm(de) > R * (1 + 1e-6) + 1e-10:
            R *= 0.5
            cf += 1
            if cf >= 8:
                break
            continue
        Fn = score_exact(trial, N)
        tol = 1e-12 * max(1.0, abs(Fc))
        if Fn < Fc - tol:
            actual = Fc - Fn
            U = max(r0 + J0 @ de, np.max(rnz + Jnz @ de) + dx * float(de @ de))
            pred = Fc - U
            cur = trial
            Fc = Fn
            cf = 0
            if np.linalg.norm(de) < 1e-7:
                break
            ratio = actual / pred if pred > 1e-12 else (1e9 if actual > 0 else 0.0)
            if ratio >= 0.75 and np.linalg.norm(de) >= 0.8 * R:
                R = min(1.5 * R, Rmax)
            elif ratio < 0.1:
                R *= 0.5
        else:
            R *= 0.5
            cf += 1
            if cf >= 8:
                break
    return cur, Fc

def run(seed=42, **kwargs):

    parent = None
    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:

            if arr.size > 1 and np.all(np.isfinite(arr)):

    else:

    base = starts[0]

    starts.append(c)
    starts.append(project_box_sum(base + 0.15 * rng.standard_normal(N), N))
    starts.append(project_box_sum(rng.uniform(0, 1, N), N))
    i = np.arange(N, dtype=np.float64)

        z += (0.12 / r) * (A * np.sin(2.0 * np.pi * r * i / N) + B * np.cos(2.0 * np.pi * r * i / N))
    starts.append(project_box_sum(z, N))
    uniq = []
    for s in starts[:6]:
        s = np.asarray(s, dtype=np.float64)
        if not check_feasible(s, N):
            s = project_box_sum(s, N)
        dup = False
        for u in uniq:

                dup = True
                break
        if not dup:
            uniq.append(s)
    if N <= 100:
        ms, il = 6, 20
    elif N <= 250:

    elif N <= 500:

    else:

    if not np.isfinite(fs):
        best_h = np.full(N, 0.5)
        fs = score_exact(best_h, N)
    return best_h, float(fs), int(N)
