            J[row, :m] = 1.0 - 2.0 * h[:m]
        else:
            J[row, i0:i0 + m] += 1.0 - h[j0:j0 + m]
            J[row, j0:j0 + m] -= h[i0:i0 + m]
    return q, J

def epigraph_obj(z, n, dx, s):
    return float(z[n])

def epigraph_obj_jac(z, n, dx, s):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def epigraph_ineq(z, n, dx, s):
    q, _ = compute_q_and_J(z[:n])
    return z[n] - dx * q

def epigraph_ineq_jac(z, n, dx, s):
    _, J = compute_q_and_J(z[:n])
    out = np.zeros((2 * n - 1, n + 1))
    out[:, :n] = -dx * J
    out[:, n] = 1.0
    return out

def epigraph_eq(z, n, dx, s):
    return float(z[:n].sum() - s)

def epigraph_eq_jac(z, n, dx, s):
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def pair_transfer(h, n):
    dx = 2.0 / n
    best_h = h.copy()
    best_score = exact_score(best_h)
    steps = [0.04, 0.015, 0.005, 0.0015, 0.0005, 0.0001]
    for step in steps:
        for _sw in range(3):
            snap_score = best_score
            sw_best_h = best_h.copy()
            sw_best_score = best_score
            for i in range(n):
                for j in range(i + 1, n):
                    a = min(step, 1.0 - best_h[i], best_h[j])
                    if a > 1e-12:
                        ht = best_h.copy()
                        ht[i] += a
                        ht[j] -= a
                        sc = exact_score(ht)
                        if sc < sw_best_score - 1e-12:
                            sw_best_score = sc
                            sw_best_h = ht.copy()
                    b = min(step, 1.0 - best_h[j], best_h[i])
                    if b > 1e-12:
                        ht = best_h.copy()
                        ht[i] -= b
                        ht[j] += b
                        sc = exact_score(ht)
                        if sc < sw_best_score - 1e-12:
                            sw_best_score = sc
                            sw_best_h = ht.copy()
            if sw_best_score < snap_score - 1e-12:
                best_h = sw_best_h
                best_score = sw_best_score
            else:
                break
    return best_h, best_score

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    n = 47
    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if pv.ndim == 1 and len(pv) > 0 and np.all(np.isfinite(pv)) and pv.min() > -1e-8 and pv.max() < 1.0 + 1e-8:
            n = len(pv)
            parent = project_capped_simplex(pv, n)
    except (NameError, Exception):
        pass
    if parent is None:
        try:
            n = int(kwargs.get('n_points', 47))
            if not (1 <= n <= 999):
                n = 47
        except (ValueError, TypeError):
            n = 47
        parent = np.full(n, 0.5)

    s = n / 2.0
    dx = 2.0 / n
    best_h = parent.copy()
    best_score = exact_score(best_h)
    parent_fallback = parent.copy()
    parent_fallback_score = exact_score(parent_fallback)

    # Build starts list
    starts = [parent.copy()]
    rev = parent[::-1].copy()
    starts.append(project_capped_simplex(rev, n))
    starts.append(project_capped_simplex(1.0 - rev, n))
    starts.append(np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    starts.append(project_capped_simplex(blk, n))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    starts.append(project_capped_simplex(alt, n))
    for sd in [0.02, 0.06, 0.15, 0.35]:
        starts.append(project_capped_simplex(parent + rng.normal(0, sd, n), n))
    starts.append(project_capped_simplex(rng.uniform(0, 1, n), n))
    starts.append(project_capped_simplex(rng.standard_normal(n), n))

    # Deduplicate
    deduped = [starts[0]]
    for v in starts[1:]:
        unique = True
        for e in deduped:
            if np.max(np.abs(v - e)) < 1e-10:
                unique = False
                break
        if unique:
            deduped.append(v)
    starts = deduped

    # Determine effort
    max_restarts = min(len(starts), kwargs.get('max_restarts', 12))
    if n <= 64:
        maxiter = min(int(kwargs.get('maxiter', 250)), 500)
    elif n <= 256:
        maxiter = min(int(kwargs.get('maxiter', 180)), 500)
    else:
        maxiter = min(int(kwargs.get('maxiter', 100)), 500)

    # SLSQP optimization
    if n <= 999:
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': epigraph_ineq, 'jac': epigraph_ineq_jac},
            {'type': 'eq', 'fun': epigraph_eq, 'jac': epigraph_eq_jac},
        ]
        args = (n, dx, s)
        for start in starts[:max_restarts]:
            z0 = np.zeros(n + 1)
            z0[:n] = start
            z0[n] = min(2.0, exact_score(start) + 1e-6)
            try:
                res = minimize(
                    epigraph_obj, z0, args=args, method='SLSQP',
                    jac=epigraph_obj_jac, constraints=cons, bounds=bounds,
                    options={'ftol': 1e-10, 'maxiter': maxiter}
                )
                if res.x is not None and len(res.x) == n + 1 and np.all(np.isfinite(res.x)):
                    h_try = project_capped_simplex(res.x[:n], n)
                    sc = exact_score(h_try)
                    if sc < best_score - 1e-12:
                        best_h = h_try
                        best_score = sc
            except Exception:
                continue

    # Pair transfer refinement
    if n <= 128:
        best_h, best_score = pair_transfer(best_h, n)

    # Final validation and fallback
    best_h = project_capped_simplex(best_h, n)
    final_score = exact_score(best_h)
    if final_score > parent_fallback_score + 1e-9:
        best_h = parent_fallback.copy()
        final_score = parent_fallback_score
    if not (np.all(np.isfinite(best_h)) and best_h.min() >= -1e-9 and best_h.max() <= 1.0 + 1e-9 and abs(best_h.sum() - s) < 1e-8 * max(1.0, float(n))):
        best_h = np.full(n, 0.5)
        final_score = exact_score(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_score = exact_score(best_h)
    return best_h, final_score, n
