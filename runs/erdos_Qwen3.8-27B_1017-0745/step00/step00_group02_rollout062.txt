import numpy as np
from scipy.optimize import minimize

def _check(h, S):
    return (h is not None and getattr(h, 'ndim', -1) == 1 and np.isfinite(h).all()
            and np.all(h >= -1e-12) and np.all(h <= 1 + 1e-12)
            and abs(h.sum() - S) <= 1e-10)

def _project(y, n):
    S = n * 0.5
    if not np.isfinite(y).all():
        return np.full(n, 0.5)
    lo, hi = y.min() - 1.0, y.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0, 1).sum() > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0, 1)
    r = S - h.sum()
    if abs(r) > 1e-13:
        if r > 0:
            for i in np.argsort(1 - h)[::-1]:
                d = min(r, 1 - h[i]); h[i] += d; r -= d
                if abs(r) < 1e-13: break
        else:
            for i in np.argsort(h):
                d = min(-r, h[i]); h[i] -= d; r += d
                if abs(r) < 1e-13: break
    return np.clip(h, 0, 1)

def _score(h, n):
    dx = 2.0 / n
    if not (np.isfinite(h).all() and len(h) == n):
        return float('inf')
    return float(np.correlate(h, 1 - h, mode='full').max() * dx)

def _sval(q, tau):
    m = q.max(); w = np.exp((q - m) / tau)
    return m + tau * np.log(w.sum())

def _sgrad(h, n, tau):
    dx = 2.0 / n
    q = np.correlate(h, 1 - h, mode='full') * dx
    m = q.max(); w = np.exp((q - m) / tau); p = w / w.sum()
    g = np.zeros(n)
    for ell in range(-(n - 1), n):
        pw = p[ell + n - 1] * dx
        if pw < 1e-15: continue
        if ell >= 0:
            g[ell:] += pw * (1 - h[:n - ell])
            g[:n - ell] -= pw * h[ell:]
        else:
            e = -ell
            g[:n - e] += pw * (1 - h[e:])
            g[e:] -= pw * h[:n - e]
    return g

def _pgd(h0, n, tau_list):
    h = h0.copy()
    for tau in tau_list:
        F = _sval(np.correlate(h, 1 - h, mode='full') * (2.0 / n), tau)
        stale = 0
        for it in range(80):
            g = _sgrad(h, n, tau)
            if not np.isfinite(g).any(): break
            alpha = 1.0; accepted = False
            for _ in range(25):
                y = _project(h - alpha * g, n)
                d = y - h
                if np.abs(d).max() < 1e-10 or np.dot(g, d) >= -1e-14: break
                Fy = _sval(np.correlate(y, 1 - y, mode='full') * (2.0 / n), tau)
                if Fy <= F + 1e-4 * np.dot(g, d) + 1e-14:
                    dec = F - Fy; h = y; F = Fy; accepted = True; break
                alpha *= 0.5
            if not accepted: break
            if dec < 1e-12:
                stale += 1
                if stale >= 10: break
            else:
                stale = 0
    return h

def _slsqp_obj(z, n, S):
    return z[-1]

def _slsqp_jac(z, n, S):
    j = np.zeros(n + 1); j[-1] = 1.0
    return j

def _slsqp_con(z, n, S):
    h = z[:n]; t = z[-1]; dx = 2.0 / n
    return t - np.correlate(h, 1 - h, mode='full') * dx

def _slsqp_conjac(z, n, S):
    h = z[:n]; dx = 2.0 / n; L = 2 * n - 1
    J = np.zeros((L, n + 1)); J[:, -1] = 1.0
    for ell in range(-(n - 1), n):
        row = ell + n - 1
        if ell >= 0:
            J[row, ell:] -= dx * (1 - h[:n - ell])
            J[row, :n - ell] += dx * h[ell:]
        else:
            e = -ell
            J[row, :n - e] -= dx * (1 - h[e:])
            J[row, e:] += dx * h[:n - e]
    return J

def _slsqp_eq(z, n, S):
    return z[:n].sum() - S

def _slsqp_eqjac(z, n, S):
    j = np.zeros(n + 1); j[:n] = 1.0
    return j

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = globals().get('initial_h_values', None)
    n = None
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64).ravel()
        if 2 <= p.size <= 999 and np.isfinite(p).all() and np.abs(p).max() < 1e6:
            n = p.size
    if n is None:
        n = kwargs.get('n_points', 45)
        if not (isinstance(n, (int, np.integer)) and 2 <= n <= 999):
            n = 45
    n = int(n); S = n * 0.5
    if parent is not None and n == np.asarray(parent).ravel().size:
        fb = _project(np.asarray(parent, dtype=np.float64).ravel(), n)
    else:
        fb = np.full(n, 0.5)
    best = fb.copy(); best_sc = _score(fb, n)
    small = n <= 100
    tau_list = ([0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 1e-5]
                if small else [0.02, 0.005, 0.001, 1e-4, 1e-5])
    starts = []
    if _check(fb, S): starts.append(fb.copy())
    starts.append(_project(1 - fb, n))
    rv = _project(fb[::-1].copy(), n)
    starts.append(rv)
    starts.append(_project(1 - rv, n))
    for sd in (0.08, 0.25):
        starts.append(_project(fb + sd * rng.standard_normal(n), n))
    starts.append(_project(rng.uniform(0, 1, n), n))
    starts.append(_project(rng.beta(0.5, 0.5, n), n))
    local_results = []
    for st in starts:
        if not _check(st, S): continue
        sc = _score(st, n)
        if sc < best_sc - 1e-12: best = st.copy(); best_sc = sc
        ho = _pgd(st, n, tau_list)
        if _check(ho, S):
            sc2 = _score(ho, n)
            local_results.append((sc2, ho.copy()))
            if sc2 < best_sc - 1e-12: best = ho.copy(); best_sc = sc2
    for sig in ([0.04, 0.12, 0.30, 0.50] if small else [0.12]):
        hp = _project(best + sig * rng.standard_normal(n), n)
        if not _check(hp, S): continue
        ho = _pgd(hp, n, tau_list)
        if _check(ho, S):
            sc = _score(ho, n)
            local_results.append((sc, ho.copy()))
            if sc < best_sc - 1e-12: best = ho.copy(); best_sc = sc
    if small:
        local_results.sort(key=lambda x: x[0])
        sel = []
        for sc, hh in local_results:
            if all(np.abs(hh - s[1]).max() > 1e-5 for s in sel):
                sel.append((sc, hh))
            if len(sel) >= 3: break
        if not sel: sel = [(best_sc, best.copy())]
        bnds = [(0, 1)] * n + [(0, 1)]
        cons = [{'type': 'ineq', 'fun': _slsqp_con, 'jac': _slsqp_conjac},
                {'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eqjac}]
        for sc0, h0 in sel[:3]:
            z0 = np.concatenate([h0, [min(1.0, sc0 + 1e-8)]])
            try:
                res = minimize(_slsqp_obj, z0, args=(n, S), method='SLSQP',
                               bounds=bnds, jac=_slsqp_jac, constraints=cons,
                               options={'maxiter': 250, 'ftol': 1e-10, 'disp': False})
                if np.isfinite(res.fun)):
                    hs = _project(res.x[:n], n)
                    if _check(hs, S):
                        sc = _score(hs, n)
                        local_results.append((sc, hs.copy()))
                        if sc < best_sc - 1e-12: best = hs.copy(); best_sc = sc
            except Exception:
                pass
    best = _project(best, n)
    sc_best = _score(best, n)
    sc_fb = _score(fb, n)
    if not _check(best, S) or sc_best > sc_fb:
        best = fb.copy(); sc_best = sc_fb
    if not _check(best, S):
        best = np.full(n, 0.5); sc_best = _score(best, n)
    return best.copy(), float(sc_best), n
