            lo = mid
        else:
            hi = mid

                        y[j] -= tt
                        py = capped_simplex_projection(y, N)
                        if py is None or np.max(np.abs(py - y)) > 1e-7:
                            continue
                        if not is_feasible(py, N):
                            continue
                        ps = exact_score(py, N)
                        if ps < best_s - 1e-12:
                            best_s = ps
                            best_y = py
                    if best_y is not None:
                        h = best_y
                        h_score = best_s
                        swept = True
                        if h_score < incumbent_score - 1e-12:
                            incumbent = h.copy()
                            incumbent_score = h_score
                if not swept:
                    no_imp += 1
                    if no_imp >= 2:
                        break
                else:
                    no_imp = 0

    final = capped_simplex_projection(incumbent, N)
    if final is None or not is_feasible(final, N):
        final = fallback
    final_score = exact_score(final, N)
    if final_score == float('inf'):
        final = np.full(N, 0.5)
        final_score = exact_score(final, N)
    return final.astype(np.float64), float(final_score), int(N)
