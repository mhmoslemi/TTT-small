                if np.all(np.isfinite(p)) and float(np.max(np.abs(p))) <= 1e6:
                    parent = p
        except Exception:
            pass
    dx = 2.0 / n
    target = float(n) / 2.0
    u = np.full(n, 0.5)
    fallback = u.copy()
    fallback_score = exact_score(fallback, dx)
    if parent is not None:
        pf = capped_simplex_project(parent, target)
        if pf is not None and is_feasible(pf, n):
            if (float(np.min(parent)) >= -1e-10 and float(np.max(parent)) <= 1.0 + 1e-10
                    and abs(float(np.sum(parent)) - target) <= 1e-8):
                fallback = pf.copy()
                fallback_score = exact_score(fallback, dx)
    starts = []
    if parent is not None:
        pp = capped_simplex_project(parent, target)
        if pp is not None:
            starts.append(pp)
        rr = capped_simplex_project(parent[::-1].copy(), target)
        if rr is not None:
            starts.append(rr)
        for s in [0.035, 0.12]:
            noisy = parent + s * rng.standard_normal(n)
            pr = capped_simplex_project(noisy, target)
            if pr is not None:
                starts.append(pr)
    for s in [None, None]:
        rv = capped_simplex_project(rng.random(n), target)
        if rv is not None:
            starts.append(rv)
    bv = capped_simplex_project(rng.beta(0.5, 0.5, size=n), target)
    if bv is not None:
        starts.append(bv)
    starts.append(u.copy())
    if parent is None:
        for _ in range(2):
            rv = capped_simplex_project(rng.random(n), target)
            if rv is not None:
                starts.append(rv)
        for _ in range(2):
            bv = capped_simplex_project(rng.beta(0.5, 0.5, size=n), target)
            if bv is not None:
                starts.append(bv)
    seen = []
    unique_starts = []
    for s in starts:
        if s is None or not is_feasible(s, n):
            continue
        dup = False
        for t in seen:
            if float(np.max(np.abs(s - t))) < 1e-10:
                dup = True
                break
        if not dup:
            seen.append(s)
            unique_starts.append(s)
    best_h = fallback.copy()
    best_score = fallback_score
    for s in unique_starts:
        sc = exact_score(s, dx)
        if sc < best_score - 1e-12:
            best_h = s.copy()
            best_score = sc
        if best_score <= 0.3808:
            break
        r1 = slp_search(s, dx, n, rng)
        sc1 = exact_score(r1, dx)
        if sc1 < best_score - 1e-12:
            best_h = r1.copy()
            best_score = sc1
        r2 = pairwise_poll(r1, dx, n, rng)
        sc2 = exact_score(r2, dx)
        if sc2 < best_score - 1e-12:
            best_h = r2.copy()
            best_score = sc2
        if best_score <= 0.3808:
            break
    for rsc in [0.025, 0.08]:
        if best_score <= 0.3808:
            break
        noisy = best_h + rsc * rng.standard_normal(n)
        pr = capped_simplex_project(noisy, target)
        if pr is None or not is_feasible(pr, n):
            continue
        r1 = slp_search(pr, dx, n, rng)
        sc1 = exact_score(r1, dx)
        if sc1 < best_score - 1e-12:
            best_h = r1.copy()
            best_score = sc1
        r2 = pairwise_poll(r1, dx, n, rng)
        sc2 = exact_score(r2, dx)
        if sc2 < best_score - 1e-12:
            best_h = r2.copy()
            best_score = sc2
    final = capped_simplex_project(best_h, target)
    if final is None or not is_feasible(final, n):
        final = fallback.copy()
    fs = exact_score(final, dx)
    if not np.isfinite(fs):
        final = u.copy()
        fs = exact_score(final, dx)
    return (final, fs, n)
