def project_box_sum(z, m):

            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(3):
        res = m - h.sum()
        if abs(res) < 1e-13:
            break
        if res > 0:
            for i in range(N):
                if res <= 0:
                    break
                a = min(res, 1.0 - h[i])
                h[i] += a
                res -= a
        else:
