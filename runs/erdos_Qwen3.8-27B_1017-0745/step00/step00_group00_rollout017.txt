    if not np.all(np.isfinite(y)) or np.min(y) < -1e-12 or np.max(y) > 1.0 + 1e-12:
        return None
    if abs(float(np.sum(y)) - S) > 1e-10 * max(1, n):
        return None
    return np.clip(y, 0.0, 1.0)

def is_feasible(h, n):
    if h.ndim != 1 or len(h) != n or not np.all(np.isfinite(h)):
        return False
    if np.min(h) < -1e-12 or np.max(h) > 1.0 + 1e-12:
        return False
    if abs(float(np.sum(h)) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score(h):
    n = len(h)
    if not is_feasible(h, n):
        return None
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / n)

def smooth_eval(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    q = corr / float(n)
    exact = float(np.max(corr) * 2.0 / n)
    z = q / tau
    zmax = float(np.max(z))
    e = np.exp(z - zmax)
    s = float(np.sum(e))
    p = e / s
    F = tau * (zmax + np.log(s))
    grad = np.zeros(n)
    for r in range(n):
        s2 = 0.0
        for k in range(-(n - 1), n):
            idx = k + (n - 1)
            term = 0.0
            if 0 <= r + k < n:
                term += (1.0 - h[r + k])
            if 0 <= r - k < n:
                term -= h[r - k]
            s2 += p[idx] * term
        grad[r] = s2 / float(n)
    if not np.all(np.isfinite(grad)) or not np.isfinite(F):
        return None, None, exact
    return F, grad, exact

def make_starts(n, parent, rng):
    starts = []
    if parent is not None:
        starts.append(parent.copy())
        if n <= 100:
            for sig in [0.02, 0.05, 0.10, 0.18, 0.30]:
                starts.append(parent + sig * rng.standard_normal(n))
            for _ in range(3):
                starts.append(rng.uniform(size=n))
        elif n <= 300:
            for sig in [0.03, 0.10, 0.22]:
                starts.append(parent + sig * rng.standard_normal(n))
            for _ in range(2):
                starts.append(rng.uniform(size=n))
        else:
            starts.append(parent + 0.10 * rng.standard_normal(n))
            starts.append(rng.uniform(size=n))
    else:
        starts.append(np.full(n, 0.5))
        if n <= 100:
            for A, m, ph in [(0.20, 1, 0.31), (0.35, 2, 0.77), (0.45, 3, 1.13)]:
                x = np.arange(n)
                starts.append(0.5 + A * np.cos(2.0 * np.pi * m * x / n + ph))
            for _ in range(5):
                starts.append(rng.uniform(size=n))
        elif n <= 300:
            for A, m, ph in [(0.20, 1, 0.31), (0.35, 2, 0.77), (0.45, 3, 1.13)]:
                x = np.arange(n)
                starts.append(0.5 + A * np.cos(2.0 * np.pi * m * x / n + ph))
            for _ in range(2):
                starts.append(rng.uniform(size=n))
        else:
            x = np.arange(n)
            starts.append(0.5 + 0.20 * np.cos(2.0 * np.pi * x / n + 0.31))
            starts.append(rng.uniform(size=n))
    feasible = []
    S = n / 2.0
    for v in starts:
        y = project_box_sum(v, S)
        if y is not None:
            feasible.append(y)
    return feasible

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent = None
    n = 45
    S = n / 2.0
    ih = globals().get('initial_h_values', None)
    if ih is not None:
        try:
            arr = np.array(ih, dtype=np.float64, copy=True)
            if (arr.ndim == 1 and 2 <= len(arr) <= 999
                    and np.all(np.isfinite(arr)) and np.max(np.abs(arr)) <= 1e12):
                n = len(arr)
                S = n / 2.0
                proj = project_box_sum(arr, S)
                if proj is not None:
                    parent = proj
        except Exception:
            parent = None
    if parent is None:
        n = 45
        S = n / 2.0
        parent = np.full(n, 0.5)
    best_h = parent.copy()
    best_score = exact_score(best_h)
    if best_score is None:
        return parent, 1.0, n
    starts = make_starts(n, parent, rng)
    if not starts:
        starts = [parent.copy()]
    temps = [0.04, 0.015, 0.006, 0.002, 0.0007, 0.00025, 0.00008]
    if n <= 100:
        max_iter = 60
    elif n <= 300:
        max_iter = 25
    else:
        max_iter = 8
    budget = min(50000, max(100, int(1e8 / (n * n))))
    eval_count = 0
    for start in starts:
        if eval_count >= budget:
            break
        h = start.copy()
        sc = exact_score(h)
        if sc is not None and sc < best_score:
            best_score = sc
            best_h = h.copy()
        for tau in temps:
            if eval_count >= budget:
                break
            res = smooth_eval(h, tau)
            if res is None or res[0] is None:
                continue
            F, grad, _ = res
            eval_count += 1
            max_g = float(np.max(np.abs(grad)))
            if max_g < 1e-13:
                continue
            no_accept = 0
            for it in range(max_iter):
                if eval_count >= budget:
                    break
                alpha = min(1e8, 0.08 / max_g)
                accepted = False
                for bt in range(12):
                    if eval_count >= budget:
                        break
                    y = project_box_sum(h - alpha * grad, S)
                    if y is None:
                        alpha *= 0.5
                        continue
                    d = y - h
                    directional = float(np.dot(grad, d))
                    if not np.isfinite(directional) or directional >= 0:
                        alpha *= 0.5
                        continue
                    res2 = smooth_eval(y, tau)
                    eval_count += 1
                    if res2 is None or res2[0] is None:
                        alpha *= 0.5
                        continue
                    F2, grad2, sc2 = res2
                    if sc2 is not None and sc2 < best_score:
                        best_score = sc2
                        best_h = y.copy()
                    arm = 1e-4 * directional + 1e-13 * max(1.0, abs(F))
                    if F2 <= F + arm:
                        dec = F - F2
                        h = y
                        F = F2
                        grad = grad2
                        max_g = float(np.max(np.abs(grad)))
                        accepted = True
                        if np.max(np.abs(d)) < 1e-8 or dec <= 1e-12 * max(1.0, abs(F)):
                            no_accept += 1
                        else:
                            no_accept = 0
                        if no_accept >= 4:
                            break
                        break
                    alpha *= 0.5
                if not accepted:
                    break
    final = project_box_sum(best_h, S)
    if final is not None:
        fs = exact_score(final)
        if fs is not None:
            return final, fs, n
    fs = exact_score(best_h)
    if fs is not None:
        return best_h, fs, n
    return parent, 1.0, n
