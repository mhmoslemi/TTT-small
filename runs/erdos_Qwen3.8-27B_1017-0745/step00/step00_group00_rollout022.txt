def project_box_sum(v, S):
    n = len(v)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0).copy()
    resid = S - float(np.sum(y))
    if abs(resid) > 1e-14:
        if resid > 0:
            for i in range(n):
                add = min(1.0 - y[i], resid)
                y[i] += add
                resid -= add
                if abs(resid) < 1e-14:
                    break
        else:
            for i in range(n):
                sub = min(y[i], -resid)
                y[i] -= sub
                resid += sub
                if abs(resid) < 1e-14:
                    break
    return y

    n = len(h)

def lag_values(h):
    n = len(h)
    Q = np.zeros(2 * n - 1)
    for k in range(-n + 1, n):
        idx = k + n - 1
        lo = max(0, k)
        hi = min(n - 1, n - 1 + k)
        for i in range(lo, hi + 1):
            Q[idx] += h[i] * (1.0 - h[i - k])
    return Q

def lag_grads(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for k in range(-n + 1, n):
        idx = k + n - 1
        lo = max(0, k)
        hi = min(n - 1, n - 1 + k)
        for i in range(lo, hi + 1):
            j = i - k
            G[idx, i] += 1.0 - h[j]
            G[idx, j] -= h[i]
    return G

def solve_slsqp(h0, bounds):
    n = len(h0)
    Q0 = lag_values(h0)
    t0 = min(1.0, float(np.max(Q0)) / n + 1e-4)
    z0 = np.concatenate([h0, [t0]])
    cons = [
        {'type': 'ineq', 'fun': ineq_func, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_func, 'jac': eq_jac}
