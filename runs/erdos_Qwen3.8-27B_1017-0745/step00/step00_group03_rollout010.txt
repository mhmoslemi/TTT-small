import numpy as np
from scipy.optimize import minimize

def score(h):
    h = np.asarray(h, dtype=np.float64)
    if h.ndim != 1 or h.size < 2 or not np.all(np.isfinite(h)):
        return np.inf
    N = h.size
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(z, m):
    N = z.size
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(5):
        res = m - np.sum(h)
        if abs(res) < 1e-12:
            break
        order = np.argsort(h) if res > 0 else np.argsort(-h)
        for i in order:
            if res > 0:
                d = min(res, 1.0 - h[i])
                h[i] += d
                res -= d
            else:
                d = min(-res, h[i])
                h[i] -= d
                res += d
            if abs(res) < 1e-14:
                break
    return h

def lag_q(h):
    N = h.size
    q = np.zeros(2 * N - 1)
    for s in range(-(N - 1), N):
        si = s + N - 1
        if s >= 0:
            L = N - s
            q[si] = np.dot(h[:L], 1.0 - h[s:s + L])
        else:
            d = -s
            L = N - d
            q[si] = np.dot(h[d:], 1.0 - h[:L])
    return q

def lag_jac(h):
    N = h.size
    G = np.zeros((2 * N - 1, N))
    for s in range(-(N - 1), N):
        si = s + N - 1
        if s >= 0:
            L = N - s
            G[si, :L] += 1.0 - h[s:s + L]
            G[si, s:] -= h[:L]
        else:
            d = -s
            L = N - d
            G[si, d:] += 1.0 - h[:L]
            G[si, :L] -= h[d:]
    return G

def obj_fun(z, N, m):
    return z[N]

def obj_jac(z, N, m):
    g = np.zeros(N + 1)
    g[N] = 1.0
    return g

def eq_fun(z, N, m):
    return np.sum(z[:N]) - m

def eq_jac(z, N, m):
    J = np.zeros((1, N + 1))
    J[0, :N] = 1.0
    return J

def ineq_fun(z, N, m):
    return z[N] - lag_q(z[:N])

def ineq_jac(z, N, m):
    G = lag_jac(z[:N])
    J = np.zeros((2 * N - 1, N + 1))
    J[:, :N] = -G
    J[:, N] = 1.0
    return J

def try_optimize(h0, N, m):
    t0 = float(np.max(lag_q(h0)))
    x0 = np.concatenate([h0, [t0]])
    bounds = list(zip(np.zeros(N), np.ones(N))) + [(0.0, float(N))]
    cons = [{"type": "eq", "fun": eq_fun, "jac": eq_jac},
            {"type": "ineq", "fun": ineq_fun, "jac": ineq_jac}]
    try:
        res = minimize(obj_fun, x0, args=(N, m), jac=obj_jac, method="SLSQP",
                       bounds=bounds, constraints=cons, maxiter=250, ftol=1e-9)
        if res.x.size == N + 1 and np.all(np.isfinite(res.x[:N])):
            return project_box_sum(res.x[:N], m)
    except Exception:
        pass
    return h0.copy()

def run(seed=42, **kwargs):
    N = 70
    parent = None
    if 'initial_h_values' in globals():
        p = globals()['initial_h_values']
        if p is not None:
            arr = np.asarray(p, dtype=np.float64)
            if arr.ndim == 1 and arr.size >= 2 and np.all(np.isfinite(arr)):
                parent = arr
    if parent is not None:
        N = parent.size
    m = N / 2.0
    uni = np.full(N, 0.5)
    if parent is not None:
        if (np.all(parent >= 0.0) and np.all(parent <= 1.0)
                and abs(np.sum(parent) - m) < 1e-10):
            best = parent.copy()
        else:
            best = project_box_sum(parent, m)
    else:
        best = uni.copy()
    best_score = score(best)
    if N <= 1:
        return (best, best_score, N)
    rng = np.random.default_rng(seed)
    starts = [best.copy(), uni.copy()]
    starts.append(project_box_sum(1.0 - best, m))
    for sc in [0.05, 0.15, 0.35]:
        starts.append(project_box_sum(best + sc * rng.standard_normal(N), m))
    starts.append(project_box_sum(rng.uniform(0, 1, N), m))
    zm = np.full(N, 0.5)
    n_modes = min(4, (N - 1) // 2)
    idx = np.arange(N)
    for r in range(1, n_modes + 1):
        A = rng.standard_normal()
        B = rng.standard_normal()
        zm += (0.12 / r) * (A * np.sin(2 * np.pi * r * idx / N)
                             + B * np.cos(2 * np.pi * r * idx / N))
    starts.append(project_box_sum(zm, m))
    for h0 in starts:
        s0 = score(h0)
        if s0 < best_score:
            best, best_score = h0.copy(), s0
        h_opt = try_optimize(h0, N, m)
        s_opt = score(h_opt)
        if s_opt < best_score:
            best, best_score = h_opt.copy(), s_opt
    for sc in [0.10, 0.05, 0.025, 0.0125]:
        h0 = project_box_sum(best + sc * rng.standard_normal(N), m)
        s0 = score(h0)
        if s0 < best_score:
            best, best_score = h0.copy(), s0
        h_opt = try_optimize(h0, N, m)
        s_opt = score(h_opt)
        if s_opt < best_score:
            best, best_score = h_opt.copy(), s_opt
    best = project_box_sum(best, m)
    best_score = score(best)
    if (not np.all(np.isfinite(best)) or not np.all(best >= 0)
            or not np.all(best <= 1)):
        best, best_score = uni.copy(), score(uni)
    return (best, best_score, N)
