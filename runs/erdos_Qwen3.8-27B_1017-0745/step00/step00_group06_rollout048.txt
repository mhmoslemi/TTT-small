    if parent is None:

        try:

                if ni >= 2 and ni <= 999 and abs(ni - float(np_)) < 1e-9:

            pass

    dx = 2.0 / N

    max_sweeps = kwargs.get('max_sweeps', 8)

    # Generate and score starts
    all_starts = make_starts(incumbent, N, rng)
    scored = []
    for s in all_starts:
        p = project_capped_simplex(s, N)
        if p is not None and is_feasible(p, N):
            sc = exact_score(p)
            scored.append((sc, p))
    scored.sort(key=lambda x: x[0])
    # Ensure incumbent is first
    starts_list = [incumbent]
    for sc, p in scored:
        if len(starts_list) >= n_starts:
            break
        if np.max(np.abs(p - starts_list[0])) > 1e-8:
            # Check not duplicate of existing
            dup = False
            for ex in starts_list:
                if np.max(np.abs(p - ex)) < 1e-8:
                    dup = True
                    break
            if not dup:
                starts_list.append(p)
    # If we don't have enough, add more from scored
    if len(starts_list) < n_starts:
        for sc, p in scored:
            if len(starts_list) >= n_starts:
                break
            dup = False
            for ex in starts_list:
                if np.max(np.abs(p - ex)) < 1e-8:
                    dup = True

        a, b2 = i, (i + 1) % N
        if a > b2:

                pairs_set = set(pairs)

                    i = int(rng.integers(N))
                    j = int(rng.integers(N))
                    if i == j:
                        continue
                    if (i, j) in pairs_set or (j, i) in pairs_set:
                        continue
                    if abs(i - j) <= 1 or (abs(i - j) == N - 1):
                        continue
                    a, b2 = (i, j) if i < j else (j, i)
                    if (a, b2) not in pairs_set:
                        pairs_set.add((a, b2))
                        pairs.append((a, b2))
                        if len(pairs) - len(cyclic_list) >= n_rand:
                            break
            else:
                pairs = cyclic_list
            rng_shuffled = list(pairs)
            rng.shuffle(rng_shuffled) if hasattr(rng, 'shuffle') else None
            # numpy generator doesn't have shuffle on lists, use permutation
            perm = rng.permutation(len(rng_shuffled))
            rng_shuffled = [rng_shuffled[k] for k in perm]

            for (i, j) in rng_shuffled:
                t_lo = max(-h[i], h[j] - 1.0)
                t_hi = min(1.0 - h[i], h[j])
                if t_hi - t_lo < 1e-14:

                # Compute r, b, c vectors
                out = np.correlate(h, 1.0 - h, mode='full')

                tb = np.where(vb, 1.0 - h[idx_b[vb]], 0.0)
                tc = np.where(vc, h[idx_c[vc]], 0.0)

                b_vec = dx * (ta - tb - tc + td)

                c_vec[N - 1] = -2.0 * dx
