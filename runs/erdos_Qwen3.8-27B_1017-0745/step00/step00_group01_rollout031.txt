def project_capped_simplex(z, n):
    if z is None or not hasattr(z, 'shape') or z.ndim != 1 or len(z) != n:

        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(z - lam, 0.0, 1.0)

        res = target - h.sum()
        if abs(res) < 1e-12:

            idx = np.argmax(1.0 - h)
            h[idx] = min(1.0, h[idx] + res)

def exact_score(h):
    n = len(h)
    if n == 0 or not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def corr_jac(h):
    n = len(h)
    q = np.correlate(h, 1.0 - h, mode='full')
    J = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        d = k - (n - 1)
        ad = abs(d)
        m = n - ad
        i0 = max(d, 0)
        j0 = max(-d, 0)
        for r in range(m):
            ii = i0 + r
            jj = j0 + r
            J[k, ii] += (1.0 - h[jj])
            J[k, jj] -= h[ii]
    return q, J

def epi_obj_c(z):
    return z[-1]

def epi_obj_jac_c(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def epi_ineq_c(z):
    n = len(z) - 1
    h = z[:n]
    t = z[-1]
    dx = 2.0 / n
    q = np.correlate(h, 1.0 - h, mode='full')
    return t - dx * q

def epi_ineq_jac_c(z):
    n = len(z) - 1
    h = z[:n]
    dx = 2.0 / n
    _, J = corr_jac(h)
    out = np.zeros((2 * n - 1, n + 1))
    out[:, :n] = -dx * J
    out[:, n] = 1.0
    return out

def epi_eq_c(z):
    n = len(z) - 1
    return z[:n].sum() - n / 2.0

def epi_eq_jac_c(z):
    n = len(z) - 1
    j = np.zeros(n + 1)
    j[:n] = 1.0
