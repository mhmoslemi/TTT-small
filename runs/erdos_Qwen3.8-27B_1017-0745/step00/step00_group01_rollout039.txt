    for d in range(a_arr.shape[0]):
        if abs(a_arr[d]) > 1e-14:
            t = -b_arr[d] / (2.0 * a_arr[d])
            if np.isfinite(t):
                cands.append(t)
    filt = sorted(min(max(t, lo_t), hi_t) for t in cands if lo_t - tol_t <= t <= hi_t + tol_t)
    if not filt:
        return x, None
    merged = [filt[0]]
    for t in filt[1:]:
        if t - merged[-1] > tol_t:
            merged.append(t)
    retain = []
    for st in range(0, len(merged), 512):
        ts = np.array(merged[st:st + 512])
        vals = np.max(a_arr[None, :] * ts[:, None] ** 2 + b_arr[None, :] * ts[:, None] + c_arr[None, :], axis=1)
        for ii in range(len(ts)):
            if np.isfinite(vals[ii]):
                retain.append((vals[ii], ts[ii]))
    retain.sort(key=lambda p: p[0])
    test_ts = [m[1] for m in retain[:8]] + [0.0]
    local_score = exact_score(x, n)
    best_x, best_s = x, local_score
    for t in test_ts:
        if abs(t) < 1e-15:
            continue
        y = x + t * v
        if not feasible_check(y, n):
            y = project_capped(y, n)
            if y is None:
                continue
        sc = exact_score(y, n)
        if sc < best_s - 1e-12:
            best_s, best_x = sc, y
    return best_x, best_s

def run(seed=42, **kwargs):
    try:
        n, parent = None, None
        try:
            raw = np.array(initial_h_values, dtype=np.float64).flatten()
            if raw.size >= 1 and np.all(np.isfinite(raw)) and np.max(np.abs(raw)) <= 1e8:
                n, parent = raw.size, raw.copy()
        except Exception:
            pass
        if n is None:
            npts = kwargs.get('n_points', 47)
            n = int(npts) if isinstance(npts, (int, np.integer)) and not isinstance(npts, bool) and 1 <= npts <= 999 else 47
        s = n / 2.0
        base = None
        if parent is not None and parent.shape == (n,):
            if np.all(parent >= -1e-8) and np.all(parent <= 1 + 1e-8):
                p = np.clip(parent, 0, 1)
                base = p if abs(p.sum() - s) <= 1e-10 * max(1, n) else project_capped(p, n)
        if base is None:
            base = np.full(n, 0.5, dtype=np.float64)
        fallback = base.copy()
        incumbent, inc_score = fallback.copy(), exact_score(fallback, n)
        if n == 1:
            return (np.array([0.5]), exact_score(np.array([0.5]), 1), 1)
        rng = np.random.default_rng(seed if isinstance(seed, (int, np.integer)) and not isinstance(seed, bool) else 42)
        raws = [base.copy(), base[::-1].copy(), 1.0 - base.copy(), base + rng.normal(0, 0.12, n),
                np.full(n, 0.5), np.zeros(n), np.zeros(n), base + rng.normal(0, 0.35, n),
                rng.uniform(0, 1, n), base + rng.normal(0, 0.45, n)]
        raws[5][:n // 2] = 1.0
        if n % 2 == 1:
            raws[5][n // 2] = 0.5
        raws[6][::2] = 1.0
        used = []
        for rs in raws:
            pr = project_capped(np.asarray(rs, dtype=np.float64), n)
            if pr is None:
                continue
            if not any(np.max(np.abs(pr - u)) < 1e-10 for u in used):
                used.append(pr)
        max_starts = min(len(used), 5 if n <= 48 else 2)
        max_sweeps = 2 if n <= 48 else (1 if n <= 64 else 0)
        kw_s = kwargs.get('starts', None)
        if isinstance(kw_s, (int, np.integer)) and not isinstance(kw_s, bool) and kw_s > 0:
            max_starts = min(kw_s, 6 if n <= 48 else 3, len(used))
        kw_sw = kwargs.get('sweeps', None)
        if isinstance(kw_sw, (int, np.integer)) and not isinstance(kw_sw, bool) and kw_sw > 0:
            max_sweeps = min(kw_sw, 3 if n <= 48 else 2)
        do_search = max_sweeps > 0 and n <= 64
        basis = haar_basis(n) if do_search else None
        pair_idx = make_pair_indices(n) if do_search else None
        for si in range(max_starts):
            x = used[si].copy()
            x_score = exact_score(x, n)
            if x_score < inc_score - 1e-12:
                incumbent, inc_score = x.copy(), x_score
            if not do_search:
                continue
            improved = True
            for _ in range(max_sweeps):
                if not improved:
                    break
                improved = False
                for di in rng.permutation(n - 1):
                    try:
                        nx, ns = line_search(x, basis[di], pair_idx, n)
                        if ns is not None and ns < x_score - 1e-12:
                            x, x_score = nx, ns
                            improved = True
                            if x_score < inc_score - 1e-12:
                                incumbent, inc_score = x.copy(), x_score
                    except Exception:
                        continue
        final = project_capped(incumbent, n)
        if final is None or not feasible_check(final, n):
            final = fallback
        fs, fb = exact_score(final, n), exact_score(fallback, n)
        if fs <= fb + 1e-12:
            return (final.astype(np.float64), float(fs), int(n))
        return (fallback.astype(np.float64), float(fb), int(n))
    except Exception:
        h = np.full(47, 0.5, dtype=np.float64)
        return (h, exact_score(h, 47), 47)
