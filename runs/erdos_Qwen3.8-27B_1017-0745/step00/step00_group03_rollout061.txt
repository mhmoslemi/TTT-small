def project_box_sum(z, N):
    if z is None or not hasattr(z, '__len__'):
        return np.full(N, 0.5)
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (N,) or not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    m = N / 2.0
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    res = m - np.sum(h)
    if res > 1e-12:
        order = np.argsort(1.0 - h)
        for i in order:
            if res <= 1e-14:
                break
            a = min(1.0 - h[i], res)
            h[i] += a
            res -= a
    elif res < -1e-12:
        order = np.argsort(h)
        res = -res
        for i in order:

        return np.full(N, 0.5)
    return h


def lag_values(h, N):
    M = 2 * N - 1
    q = np.zeros(M)

        idx = k + N - 1

            L = N - k
            q[idx] = np.dot(h[k:k + L], 1.0 - h[0:L])
        else:
            d = -k
            L = N - d
            q[idx] = np.dot(h[0:L], 1.0 - h[d:d + L])
    return q

    r = (2.0 / N) * q
    rmax = float(np.max(r))
    vals = np.exp((r - rmax) / tau)
    return rmax + tau * np.log(np.sum(vals))

    q = lag_values(h, N)
    r = (2.0 / N) * q
    rmax = float(np.max(r))
    exps = np.exp((r - rmax) / tau)
    total = float(np.sum(exps))
    p = exps / total
    phi = rmax + tau * np.log(total)
    dx = 2.0 / N
    grad = np.zeros(N)
    for k in range(-(N - 1), N):
        idx = k + N - 1
        w = dx * p[idx]
        if abs(w) < 1e-16:
            continue
        if k >= 0:
            L = N - k
            a = h[k:k + L]
            b = h[0:L]
            grad[k:k + L] += w * (1.0 - b)
            grad[0:L] -= w * a
        else:
            d = -k
            L = N - d
            a = h[0:L]
            b = h[d:d + L]
            grad[0:L] += w * (1.0 - b)
            grad[d:d + L] -= w * a
    return phi, grad
