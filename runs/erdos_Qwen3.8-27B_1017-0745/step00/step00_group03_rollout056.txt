import numpy as np

def project_box_sum(z, N):
    m = N / 2.0
    if z is None or not hasattr(z, 'shape') or z.shape != (N,) or not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    z = np.asarray(z, dtype=np.float64).copy()
    lo, hi = float(z.min()) - 1.0, float(z.max())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(z - mid, 0.0, 1.0)
        if h.sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    e = m - h.sum()
    if e > 1e-14:
        for i in range(N):
            if e <= 0:
                break
            a = min(e, 1.0 - h[i])
            h[i] += a
            e -= a
    elif e < -1e-14:
        e2 = -e
        for i in range(N):
            if e2 <= 0:
                break
            a = min(e2, h[i])
            h[i] -= a
            e2 -= a
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)):
        return np.full(N, 0.5)
    return h

def lag_values(h, N):
    M = 2 * N - 1
    q = np.zeros(M)
    for k in range(-(N - 1), N):
        if k >= 0:
            L = N - k
            q[k + N - 1] = np.sum(h[k:k + L] * (1.0 - h[0:L]))
        else:
            d = -k
            L = N - d
            q[k + N - 1] = np.sum(h[0:L] * (1.0 - h[d:d + L]))
    return q

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / N)

def smooth_value(h, tau, N):
    q = lag_values(h, N)
    r = (2.0 / N) * q
    rmax = float(np.max(r))
    exps = np.exp((r - rmax) / tau)
    s = float(np.sum(exps))
    if s <= 0.0:
        return rmax
    return rmax + tau * np.log(s)

def smooth_value_grad(h, tau, N):
    q = lag_values(h, N)
    r = (2.0 / N) * q
    rmax = float(np.max(r))
    exps = np.exp((r - rmax) / tau)
    s = float(np.sum(exps))
    if s <= 0.0:
        s = 1.0
        exps = np.zeros(len(r))
        exps[int(np.argmax(r))] = 1.0
    p = exps / s
    phi = rmax + tau * np.log(s)
    dx = 2.0 / N
    grad = np.zeros(N)
    for k in range(-(N - 1), N):
        w = dx * p[k + N - 1]
        if w < 1e-16:
            continue
        if k >= 0:
            L = N - k
            grad[k:k + L] += w * (1.0 - h[0:L])
            grad[0:L] -= w * h[k:k + L]
        else:
            d = -k
            L = N - d
            grad[0:L] += w * (1.0 - h[d:d + L])
            grad[d:d + L] -= w * h[0:L]
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(N)
    return phi, grad

def optimize_start(h0, N, iters_per_stage, inc_h, inc_score):
    temperatures = [0.08, 0.02, 0.005, 0.001, 0.0002, 0.00004, 0.000008]
    h = h0.copy()

        inc_h = h.copy()

            phi, g = smooth_value_grad(stage_h, tau, N)

                    break
