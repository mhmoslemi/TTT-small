            delta = max(delta * 0.5, delta_min)
            if delta <= delta_min * 1.01:
                break
            continue
        u_delta = y - x
        pred_max = float(np.max(R + G.dot(u_delta)))
        p = c_x - pred_max
        a = c_x - c_y
        if a > 1e-12:
            x = y.copy()
            rho = a / p if p > 1e-12 else 0.0
            step_actual = float(np.max(np.abs(u_delta)))
            if rho > 0.75 and step_actual >= 0.8 * delta:
                delta = min(delta_max, 1.5 * delta)
            elif rho < 0.25:
                delta = max(delta * 0.5, delta_min)
                if delta <= delta_min * 1.01:
                    break
        else:
            rho = a / p if p > 1e-12 else 0.0
            if rho < 0.25:
                delta = max(delta * 0.5, delta_min)
                if delta <= delta_min * 1.01:
                    break
    return x


def run(seed=42, **kwargs):
    parent = None
    try:
        p = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if len(p) >= 2 and np.all(np.isfinite(p)) and float(np.max(np.abs(p))) < 1e6:
            parent = p
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
    else:
        n = 95
        if 'n_points' in kwargs:
            try:
                n = int(kwargs['n_points'])
            except Exception:
                pass
        n = max(2, min(n, 999))

    M = n / 2.0
    fallback = np.full(n, 0.5, dtype=np.float64)
    best_h = fallback.copy()
    best_score = score_h(fallback)

    if parent is not None and n > 999:
        ph = project_capped_simplex(parent, M)
        if ph is not None and score_h(ph) < best_score:
            best_h = ph
            best_score = score_h(ph)
        return best_h, best_score, n

    maxiter = 40
    if 'maxiter' in kwargs:
        try:
            maxiter = max(10, min(int(kwargs['maxiter']), 80))
        except Exception:
            pass
    n_starts = 6
    if 'n_starts' in kwargs:
        try:
            n_starts = max(1, min(int(kwargs['n_starts']), 8))
        except Exception:
            pass

    if n <= 200:
        max_starts, max_iters = 6, 40
    elif n <= 500:
        max_starts, max_iters = 4, 20
    else:
        max_starts, max_iters = 2, 10
    maxiter = min(maxiter, max_iters)
    n_starts = min(n_starts, max_starts)

    starts = []
    if parent is not None:
        ph = project_capped_simplex(parent, M)
        if ph is not None:
            starts.append(ph)
    starts.append(np.full(n, 0.5, dtype=np.float64))
    blk = np.zeros(n, dtype=np.float64)
    bloc = int(np.floor(M))
    blk[:bloc] = 1.0
    frac = M - bloc
    if frac > 0 and bloc < n:
        blk[bloc] = frac
    starts.append(blk)

    try:
        s2 = int(seed) % (2 ** 32)
    except Exception:
        s2 = 42
    rng = np.random.default_rng(s2)
    i = np.arange(n)
    for k in [1, 3, 7]:
        prof = 0.5 + 0.28 * np.sin(2.0 * np.pi * k * i / n)
        pp = project_capped_simplex(prof, M)
        if pp is not None:
            starts.append(pp)

    deduped = []
    for s in starts:
        if s is None:
            continue
        dup = False
        for d in deduped:
            if float(np.max(np.abs(s - d))) < 1e-8:
                dup = True
                break
        if not dup:
            deduped.append(s)

    base = deduped[0] if deduped else fallback
    noise_scales = [0.08, 0.18, 0.30]
    ni = 0
    while len(deduped) < n_starts and ni < 20:
        noise = rng.normal(0.0, noise_scales[ni % 3], size=n)
        pp = project_capped_simplex(base + noise, M)
        if pp is not None:
            dup = False
            for d in deduped:
                if float(np.max(np.abs(pp - d))) < 1e-8:
                    dup = True
                    break
            if not dup:
                deduped.append(pp)
        ni += 1

    for s in deduped:
        sc = score_h(s)
        if sc < best_score - 1e-12:
            best_score = sc
            best_h = s.copy()

    for s in deduped:
        result = run_local_search(s, maxiter)
        if result is not None:
            sc = score_h(result)
            if sc < best_score - 1e-12:
                best_score = sc
                best_h = result.copy()

    final = project_capped_simplex(best_h, M)
    if final is None:
        best_h = fallback
    else:
        best_h = final
    best_score = score_h(best_h)
    if not np.isfinite(best_score) or best_score > 2.0:
        best_h = fallback
        best_score = score_h(fallback)
    return best_h, best_score, n
