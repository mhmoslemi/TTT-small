    S = n / 2.0
    if not check_feasible(h, n, S):
        return None, None
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full').copy()
    score = float(np.max(raw)) * dx
    if not np.isfinite(score):
        return None, None
    return score, raw

def quad_roots(A, B, C):
    scale = max(1.0, abs(B), abs(C))
    if abs(A) < 1e-14 * scale:
        if abs(B) < 1e-14 * scale:
            return []
        return [-C / B]
    disc = B * B - 4.0 * A * C
    if disc < -1e-12 * scale * scale:
        return []
    disc = max(0.0, disc)
    sq = sqrt(disc)
    q = -0.5 * (B + copysign(sq, B))
    rts = []
    if abs(q) > 1e-30:
        rts.append(q / A)
        rts.append(C / q)
    else:
        rts.append(-B / (2.0 * A))
    return [r for r in rts if np.isfinite(r)]

def pair_line_min(h, raw, p, q, n):
    t_lo = max(-h[p], h[q] - 1.0)
    t_hi = min(1.0 - h[p], h[q])
    if t_hi - t_lo < 1e-14 or not np.isfinite(t_lo) or not np.isfinite(t_hi):
        return None
    gap = abs(q - p)
    if gap == 0:
        return None
    c = raw
    b = np.zeros(2 * n - 1)
    for d in range(-(n - 1), n):
        lo_d = max(0, -d)
        hi_d = min(n, n - d)
        gp = 0.0
        if lo_d <= p < hi_d and 0 <= p + d < n:
            gp += 1.0 - h[p + d]
        if lo_d <= p - d < hi_d:
            gp -= h[p - d]
        gq = 0.0
        if lo_d <= q < hi_d and 0 <= q + d < n:
            gq += 1.0 - h[q + d]
        if lo_d <= q - d < hi_d:
            gq -= h[q - d]
        b[d + n - 1] = gp - gq
    a = np.zeros(2 * n - 1)
    a[n - 1] = -2.0
    a[gap + n - 1] = 1.0
    a[-gap + n - 1] = 1.0
    if not np.all(np.isfinite(b)):
        return None
    # Build upper hull of affine lines
    am = (a == 0)
    hsl, hic, hst = [], [], []
    if np.any(am):
        sl = b[am].copy()
        ic = c[am].copy()
        order = np.argsort(sl)
        sl, ic = sl[order], ic[order]
        for i in range(len(sl)):
            if hsl and abs(sl[i] - hsl[-1]) < 1e-14:
                if ic[i] > hic[-1]:
                    hic[-1] = ic[i]
                continue
            if hsl:
                ds = sl[i] - hsl[-1]
                if abs(ds) < 1e-14:
                    continue
                xc = (hic[-1] - ic[i]) / ds
                while len(hsl) > 1 and xc <= hst[-1] + 1e-14:
                    hsl.pop(); hic.pop(); hst.pop()
                    ds = sl[i] - hsl[-1]
                    xc = (hic[-1] - ic[i]) / ds
                hst.append(xc)
            else:
                hst.append(-inf)
            hsl.append(sl[i]); hic.append(ic[i])
    # Special quadratics
    sds = [n - 1, gap + n - 1, -gap + n - 1]
    sa = [a[sds[0]], a[sds[1]], a[sds[2]]]
    sb = [b[sds[0]], b[sds[1]], b[sds[2]]]
    sc = [c[sds[0]], c[sds[1]], c[sds[2]]]
    cands = [t_lo, t_hi, 0.0]
    for i in range(1, len(hst)):
        if t_lo < hst[i] < t_hi:
            cands.append(hst[i])
    for seg in range(len(hsl)):
        s_lo = max(t_lo, hst[seg])
        s_hi = min(t_hi, hst[seg + 1] if seg + 1 < len(hsl) else t_hi)
        if s_lo >= s_hi:
            continue
        for k in range(3):
            for r in quad_roots(sa[k], sb[k] - hsl[seg], sc[k] - hic[seg]):
                if s_lo - 1e-10 <= r <= s_hi + 1e-10:
                    cands.append(r)
    for k1 in range(3):
        for k2 in range(k1 + 1, 3):
            for r in quad_roots(sa[k1] - sa[k2], sb[k1] - sb[k2], sc[k1] - sc[k2]):
                if t_lo - 1e-10 <= r <= t_hi + 1e-10:
                    cands.append(r)
    for k in range(3):
        if sa[k] > 0:
            v = -sb[k] / (2.0 * sa[k])
            if t_lo - 1e-10 <= v <= t_hi + 1e-10:
                cands.append(v)
    best_t, best_F = 0.0, inf
    for t in cands:
        if not np.isfinite(t):
            continue
        t = max(t_lo, min(t_hi, t))
        f = -inf
        if hsl:
            idx = len(hsl) - 1
            for i in range(len(hst)):
                if t < hst[i]:
                    idx = i - 1
                    break
            idx = max(0, idx)
            f = hsl[idx] * t + hic[idx]
        for k in range(3):
            qv = sa[k] * t * t + sb[k] * t + sc[k]
            if qv > f:
                f = qv
        if f < best_F - 1e-15 or (abs(f - best_F) <= 1e-15 and abs(t) < abs(best_t)):
            best_F, best_t = f, t
    if abs(best_t) < 1e-14:
        return None
    return float(best_t)

def build_sparse_pairs(n):
    offsets = []
    v = 1
    while v <= n // 2:
        offsets.append(v)
        v *= 2
    if n // 2 not in offsets:
        offsets.append(n // 2)
    pairs = set()
    for off in offsets:
        for i in range(n):
            j = (i + off) % n
            if i != j:
                pairs.add((min(i, j), max(i, j)))
    return sorted(pairs)

def run(seed=42, **kwargs):
    if seed is None:
        seed = 42
    try:
        seed = int(seed)
    except (ValueError, TypeError):
        seed = 42
    rng = np.random.default_rng(seed)
    n = 80
    parent = globals().get('initial_h_values', None)
    have_parent = False
    if parent is not None:
        try:
            parr = np.asarray(parent, dtype=np.float64).ravel()
            if len(parr) >= 1:
                n = len(parr)
                if np.all(np.isfinite(parr)):
                    S = n / 2.0
                    if check_feasible(parr, n, S):
                        parr = parr.copy()
                        have_parent = True
                    else:
                        pr = project_capped_simplex(parr, S)
                        if pr is not None:
                            parr = pr
                            have_parent = True
        except Exception:
            pass
    S = n / 2.0
    dx = 2.0 / n
    h_const = np.full(n, 0.5)
    inc_h = h_const.copy()
    inc_score, _ = score_exact(inc_h, n)
    if inc_score is None:
        inc_score = 0.5
    if have_parent:
        ps, _ = score_exact(parr, n)
        if ps is not None and ps < inc_score:
            inc_score, inc_h = ps, parr.copy()
    if n > 128:
        return inc_h, float(inc_score), int(n)
    # Build starts
    starts = []
    def try_add_start(hc):
        if hc is None or not np.all(np.isfinite(hc)):
            return
        hc = np.clip(hc, 0.0, 1.0)
        if not check_feasible(hc, n, S):
            pr = project_capped_simplex(hc, S)
            if pr is None:
                return
            hc = pr
        for ex in starts:
            if float(np.max(np.abs(hc - ex))) < 1e-10:
                return
        starts.append(hc)
    if have_parent:
        try_add_start(parr.copy())
        try_add_start(parr[::-1].copy())
        try_add_start(h_const.copy())
        g = rng.standard_normal(n) * 0.12
        try_add_start(project_capped_simplex(parr + g, S))
        qq = int(rng.integers(1, min(8, max(1, n // 2)) + 1))
        ph = rng.uniform(0, 2 * np.pi)
        try_add_start(project_capped_simplex(0.5 + 0.45 * np.cos(2 * np.pi * qq * np.arange(n) / n + ph), S))
        perm = rng.permutation(n)
        bb = np.zeros(n)
        bb[perm[:n // 2]] = 1.0
        if n % 2 == 1:
            bb[perm[n // 2]] = 0.5
        try_add_start(bb)
    else:
        try_add_start(h_const.copy())
        for _ in range(2):
            qq = int(rng.integers(1, min(8, max(1, n // 2)) + 1))
            ph = rng.uniform(0, 2 * np.pi)
            try_add_start(project_capped_simplex(0.5 + 0.45 * np.cos(2 * np.pi * qq * np.arange(n) / n + ph), S))
        try_add_start(project_capped_simplex(0.5 + 0.3 * rng.standard_normal(n), S))
        try_add_start(project_capped_simplex(rng.uniform(0, 1, n), S))
        perm = rng.permutation(n)
        bb = np.zeros(n)
        bb[perm[:n // 2]] = 1.0
        if n % 2 == 1:
            bb[perm[n // 2]] = 0.5
        try_add_start(bb)
    try_add_start(inc_h.copy())
    # Budget
    full_size = n * (n - 1) // 2
    Btot = min(8000, max(500, 64000000 // (n * n)))
    reserve_full = full_size <= Btot
    reserve_n = full_size if reserve_full else 0
    sparse_budget = Btot - reserve_n
    sparse_pairs = build_sparse_pairs(n)
    # Multistart sparse phase
    for st in starts:
        cur = st.copy()
        cur_score, cur_raw = score_exact(cur, n)
        if cur_score is None:
            continue
        stagnant = 0
        for sweep in range(3):
            if sparse_budget <= 0:
                break
            order = list(range(len(sparse_pairs)))
            rng.shuffle(order)
            accepted = 0
            for idx in order:
                if sparse_budget <= 0:
                    break
                sparse_budget -= 1
                p, q = sparse_pairs[idx]
                t = pair_line_min(cur, cur_raw, p, q, n)
                if t is None:
                    continue
                ps = cur[p] + cur[q]
                pn = float(np.clip(cur[p] + t, max(0.0, ps - 1.0), min(1.0, ps)))
                qn = ps - pn
                trial = cur.copy()
                trial[p] = pn
                trial[q] = qn
                if not check_feasible(trial, n, S):
                    trial = project_capped_simplex(trial, S)
                    if trial is None or not check_feasible(trial, n, S):
                        continue
                ts, tr = score_exact(trial, n)
                if ts is None:
                    continue
                if ts < inc_score - 1e-15:
                    inc_score, inc_h = ts, trial.copy()
                if ts < cur_score - 1e-13 * max(1.0, abs(cur_score)):
                    cur, cur_score, cur_raw = trial, ts, tr
                    accepted += 1
            if accepted > 0:
                stagnant = 0
            else:
                stagnant += 1
            if stagnant >= 2:
                break
    # Full sweep
    if reserve_full and sparse_budget + reserve_n > 0:
        cur = inc_h.copy()
        cur_score, cur_raw = score_exact(cur, n)
        if cur_score is None:
            return inc_h, float(inc_score), int(n)
        all_pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        for fsweep in range(2):
            if reserve_n <= 0:
                break
            reserve_n -= full_size
            order = list(range(len(all_pairs)))
            rng.shuffle(order)
            accepted = 0
            for idx in order:
                p, q = all_pairs[idx]
                t = pair_line_min(cur, cur_raw, p, q, n)
                if t is None:
                    continue
                ps = cur[p] + cur[q]
                pn = float(np.clip(cur[p] + t, max(0.0, ps - 1.0), min(1.0, ps)))
                qn = ps - pn
                trial = cur.copy()
                trial[p] = pn
                trial[q] = qn
                if not check_feasible(trial, n, S):
                    trial = project_capped_simplex(trial, S)
                    if trial is None or not check_feasible(trial, n, S):
                        continue
                ts, tr = score_exact(trial, n)
                if ts is None:
                    continue
                if ts < inc_score - 1e-15:
                    inc_score, inc_h = ts, trial.copy()
                if ts < cur_score - 1e-13 * max(1.0, abs(cur_score)):
                    cur, cur_score, cur_raw = trial, ts, tr
                    accepted += 1
            if accepted == 0:
                break
    # Final
    pool = [inc_h]
    if have_parent:
        pool.append(parr.copy())
    pool.append(h_const.copy())
    best_h, best_s = None, inf
    for ph in pool:
        ph = ph.copy()
        if not check_feasible(ph, n, S):
            pr = project_capped_simplex(ph, S)
            if pr is None:
                continue
            ph = pr
        ss, _ = score_exact(ph, n)
        if ss is not None and ss < best_s:
            best_s, best_h = ss, ph
    if best_h is None:
        best_h = h_const.copy()
        best_s, _ = score_exact(best_h, n)
        if best_s is None:
            best_s = 0.5
    return best_h.astype(np.float64), float(best_s), int(n)
