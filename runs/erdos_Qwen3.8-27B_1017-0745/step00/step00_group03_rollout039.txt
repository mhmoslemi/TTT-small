import numpy as np
from math import sqrt, copysign

def project_box_sum(z, N):
    m = N / 2.0
    uniform = np.full(N, 0.5)
    if z.shape != (N,) or not np.all(np.isfinite(z)) or np.max(np.abs(z)) > 1e100:
        return uniform
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(z - mid, 0.0, 1.0)))
        if s > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    e = m - float(np.sum(h))
    for _ in range(3):
        if e > 1e-12:
            order = np.argsort(-(1.0 - h))
            for i in order:
                if e <= 1e-12: break
                add = min(e, 1.0 - h[i])
                h[i] += add; e -= add
        elif e < -1e-12:
            order = np.argsort(-h)
            for i in order:
                if e >= -1e-12: break
                sub = min(-e, h[i])
                h[i] -= sub; e += sub
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - m) > 1e-8 * max(1, N):
        return uniform
    return h

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / N))

def lag_coefficients(h, d, N):
    M = 2 * N - 1
    A = np.zeros(M); B = np.zeros(M); C = np.zeros(M)
    for k in range(-(N - 1), N):
        idx = k + N - 1
        if k >= 0:
            L = N - k; a, b = h[k:k+L], h[0:L]; u, v = d[k:k+L], d[0:L]
        else:
            s = -k; L = N - s; a, b = h[0:L], h[s:s+L]; u, v = d[0:L], d[s:s+L]
        A[idx] = float(np.sum(a * (1.0 - b)))
        B[idx] = float(np.sum(u * (1.0 - b) - a * v))
        C[idx] = -float(np.sum(u * v))
    return A, B, C

def line_search(h, d, N, cur_score):
    M = 2 * N - 1
    A, B, C = lag_coefficients(h, d, N)
    if not (np.all(np.isfinite(A)) and np.all(np.isfinite(B)) and np.all(np.isfinite(C))):
        return h, cur_score
    nz = np.abs(d) > 1e-15
    if not np.any(nz):
        return h, cur_score
    al = h[nz] / d[nz]; be = (1.0 - h[nz]) / d[nz]
    lo = max(float(np.max(np.minimum(al, be))), -1e6)
    hi = min(float(np.min(np.maximum(al, be))), 1e6)
    if hi - lo < 1e-14:
        return h, cur_score
    cands = [lo, hi, 0.0]
    nzC = np.abs(C) > 1e-15
    if np.any(nzC):
        sp = -B[nzC] / (2.0 * C[nzC])
        cands.extend(sp[np.isfinite(sp)].tolist())
    for i in range(M):
        for j in range(i + 1, M):
            u = C[i] - C[j]; v = B[i] - B[j]; w = A[i] - A[j]
            sc = max(abs(u), abs(v), abs(w), 1e-300)
            if abs(u) <= 1e-12 * sc:
                if abs(v) > 1e-12 * sc:
                    r = -w / v
                    if np.isfinite(r): cands.append(r)
            else:
                D = v * v - 4.0 * u * w
                if D < -1e-12 * (v * v + abs(4.0 * u * w) + 1e-300): continue
                if D < 0: D = 0.0
                sq = sqrt(D)
                q = -0.5 * (v + copysign(sq, v))
                if abs(q) > 1e-300:
                    r1 = q / u; r2 = w / q
                else:
                    r1 = -v / (2.0 * u); r2 = r1
                if np.isfinite(r1): cands.append(r1)
                if np.isfinite(r2): cands.append(r2)
    in_r = [a for a in cands if np.isfinite(a) and lo - 1e-10 <= a <= hi + 1e-10]
    in_r = [max(lo, min(hi, a)) for a in in_r]
    in_r.sort()
    dd = [in_r[0]] if in_r else []
    for a in in_r[1:]:
        if abs(a - dd[-1]) > 1e-12 * max(1.0, abs(a)): dd.append(a)
    if not dd: return h, cur_score
    aa = np.array(dd)
    pred = np.empty(len(aa))
    for s0 in range(0, len(aa), 256):
        e0 = min(s0 + 256, len(aa))
        b_aa = aa[s0:e0]
        vals = A[None, :] + b_aa[:, None] * B[None, :] + (b_aa[:, None] ** 2) * C[None, :]
        pred[s0:e0] = np.max(vals, axis=1)
    n_sel = min(16, len(pred))
    top = np.argsort(pred)[:n_sel]
    for mx in [0, len(dd) - 1]:
        if mx < len(pred) and mx not in top: top = np.append(top, mx)
    top = np.unique(top)
    best_h, best_s = h, cur_score
    for ti in top:

        s_new = score_exact(h_new, N)
        if s_new < best_s - 1e-15: best_h, best_s = h_new, s_new
    return best_h, best_s

def make_directions(h, N, rng, stage):
    dirs = []
    if stage == 'main':
        n_sp, n_rp, n_sb, n_ha = (8, 6, 4, 2) if N <= 80 else (6, 4, 2, 1)
    else:
        n_sp, n_rp, n_sb, n_ha = (4, 3, 2, 1) if N <= 80 else (3, 2, 1, 1)
    # star pairs
    anchor = int(np.argmax(np.minimum(h, 1.0 - h)))
    others = np.array([i for i in range(N) if i != anchor])
    for ii in range(min(n_sp, len(others))):
        j = int(others[np.array_split(len(others), min(n_sp, len(others)))[ii][0]]) if len(others) >= n_sp else int(rng.integers(N))
        if j == anchor: continue
        d = np.zeros(N); d[j] = 1.0; d[anchor] = -1.0
        dirs.append(d)
    # random pairs
    used_pairs = set()
    for _ in range(n_rp):
        for att in range(50):
            a, b = int(rng.integers(N)), int(rng.integers(N))
            if a == b: continue
            key = (min(a, b), max(a, b))
            if key in used_pairs: continue
            used_pairs.add(key)
            d = np.zeros(N); d[a] = 1.0; d[b] = -1.0
            dirs.append(d); break
    # sparse blocks
    for _ in range(n_sb):
        if N < 3: break
        sz = int(rng.integers(3, min(13, N + 1)))
        idx = rng.choice(N, size=sz, replace=False)
        z = rng.standard_normal(sz) - np.mean(rng.standard_normal(sz))
        d = np.zeros(N); d[idx] = z
        d[idx[0]] -= np.sum(d)
        nn = np.max(np.abs(d))
        if nn > 1e-14: dirs.append(d / nn)
    # harmonics
    max_ha = min(n_ha, (N - 1) // 2)
    for r in range(1, max_ha + 1):
        phase = float(rng.uniform(0, 2 * np.pi))
        i = np.arange(N)
        d = np.cos(2.0 * np.pi * r * i / N + phase)
        d -= np.mean(d)
        d[0] -= np.sum(d)
        nn = np.max(np.abs(d))
        if nn > 1e-14: dirs.append(d / nn)
    return dirs
