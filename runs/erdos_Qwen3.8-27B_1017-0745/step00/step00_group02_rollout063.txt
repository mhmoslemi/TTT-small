def _s_obj_g(z, n, dx, Sv):
    gz = np.zeros(n + 1)
    gz[n] = 1.0
    return gz

def _s_ineq(z, n, dx, Sv):
    return z[n] - lag_scores(z[:n])

def _s_ineq_g(z, n, dx, Sv):
    h = z[:n]
    n2 = 2 * n - 1
    J = np.zeros((n2, n + 1))
    for li in range(n2):
        l = li - (n - 1)
        lo = max(0, -l)
        hi = min(n, n - l)
        J[li, lo:hi] -= dx * h[lo + l:hi + l]
        J[li, lo + l:hi + l] += dx * (1.0 - h[lo:hi])
        J[li, n] = 1.0
    return J

def _s_eq(z, n, dx, Sv):
    return float(np.sum(z[:n]) - Sv)

def _s_eq_g(z, n, dx, Sv):
    gz = np.zeros(n + 1)
    gz[:n] = 1.0
    return gz

def slsqp_refine(h0, n, S):
    try:
        from scipy.optimize import minimize
    except Exception:
        return None
    dx = 2.0 / n
    Sv = float(S)
    z0 = np.empty(n + 1)
    z0[:n] = h0
    z0[n] = min(1.0, exact_score(h0) + 1e-8)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'ineq', 'fun': _s_ineq, 'jac': _s_ineq_g, 'args': (n, dx, Sv)},
        {'type': 'eq', 'fun': _s_eq, 'jac': _s_eq_g, 'args': (n, dx, Sv)},
    ]
    try:
        res = minimize(_s_obj, z0, method='SLSQP', jac=_s_obj_g,
                       args=(n, dx, Sv), bounds=bounds, constraints=cons,
                       options={'maxiter': 250, 'ftol': 1e-10, 'disp': False})
        if np.all(np.isfinite(res.x)):
            return project_onto_K(res.x[:n], n, S)
    except Exception:
        pass
    return None

def check_feasible(h, n, S):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return False
    if abs(float(np.sum(h)) - S) > 1e-8:
        return False
    return True

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    h_parent = None
    ig = globals().get('initial_h_values', None)
    if ig is not None:
        hp = np.asarray(ig, dtype=np.float64)
        if hp.ndim == 1 and 2 <= len(hp) <= 999:
            if np.all(np.isfinite(hp)) and float(np.max(np.abs(hp))) < 1e6:
                h_parent = hp.copy()
    if h_parent is not None:
        n = len(h_parent)
    elif 'n_points' in kwargs and isinstance(kwargs['n_points'], int) and 2 <= kwargs['n_points'] <= 999:
        n = kwargs['n_points']
    else:
        n = 45
    dx = 2.0 / n
    S = n / 2.0
    if h_parent is not None:
        h_fallback = project_onto_K(h_parent, n, S)
    else:
        h_fallback = np.full(n, 0.5)
    best_h = h_fallback.copy()
    best_score = exact_score(h_fallback)
    starts = [best_h.copy()]
    if h_parent is not None:
        hp = project_onto_K(h_parent, n, S)
        starts.append(1.0 - hp)
        starts.append(hp[::-1].copy())
        starts.append(1.0 - hp[::-1])
        starts.append(project_onto_K(hp + 0.08 * rng.standard_normal(n), n, S))
        starts.append(project_onto_K(hp + 0.25 * rng.standard_normal(n), n, S))
    starts.append(project_onto_K(rng.uniform(0, 1, n), n, S))
    starts.append(project_onto_K(rng.beta(0.5, 0.5, n), n, S))
    local_results = []
    for s in starts:
        s = project_onto_K(s, n, S)
        sc = exact_score(s)
        if sc < best_score - 1e-12:
            best_h = s.copy()
            best_score = sc
        local_results.append(s.copy())
    tau_list = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00001]
    if n > 100:
        tau_list = [0.02, 0.005, 0.001, 0.0001, 0.00001]
    for i in range(len(local_results)):
        h_opt = pg_optimize(local_results[i], n, S, tau_list)
        sc = exact_score(h_opt)
        if sc < best_score - 1e-12:
            best_h = h_opt.copy()
            best_score = sc
        local_results[i] = h_opt.copy()
    for sigma in [0.04, 0.12, 0.30, 0.50]:
        z = rng.standard_normal(n)
        s = project_onto_K(best_h + sigma * z, n, S)
        h_opt = pg_optimize(s, n, S, tau_list)
        sc = exact_score(h_opt)
        if sc < best_score - 1e-12:
            best_h = h_opt.copy()
            best_score = sc
        local_results.append(h_opt.copy())
    if n <= 100:
        scored = sorted([(exact_score(r), r) for r in local_results])
        selected = []
        for sc, h in scored:
            if len(selected) >= 3:
                break
            if all(float(np.max(np.abs(h - s))) > 1e-5 for s in selected):
                selected.append(h)
        for h_start in selected:
            h_ref = slsqp_refine(h_start, n, S)
            if h_ref is not None:
                sc = exact_score(h_ref)
                if sc < best_score - 1e-12:
                    best_h = h_ref.copy()
                    best_score = sc
    best_h = project_onto_K(best_h, n, S)
    if not check_feasible(best_h, n, S):
        best_h = h_fallback.copy()
    h_fb = project_onto_K(h_fallback, n, S)
    if exact_score(h_fb) < exact_score(best_h):
        best_h = h_fb
    if not check_feasible(best_h, n, S):
        best_h = np.full(n, 0.5)
        best_h[n // 2] += S - float(np.sum(best_h))
    c5 = exact_score(best_h)
    return best_h, c5, n
