        if abs(cv[r]) > 1e-15:
            t = -bv[r] / (2.0 * cv[r])
            if np.isfinite(t) and t_lo - 1e-12 <= t <= t_hi + 1e-12:
                cands.append(max(t_lo, min(t_hi, t)))
    cands.sort()
    deduped = [cands[0]]
    for t in cands[1:]:
        if abs(t - deduped[-1]) > 1e-12 * max(1.0, abs(t)):
            deduped.append(t)
    best_t, best_val = 0.0, float('inf')
    for i in range(0, len(deduped), 256):
        chunk = np.array(deduped[i:i + 256], dtype=np.float64)
        vals = np.max(q[None, :] + chunk[:, None] * bv[None, :] + chunk[:, None] ** 2 * cv[None, :], axis=1)
        jmin = int(np.argmin(vals))
        if vals[jmin] < best_val:
            best_val = float(vals[jmin])
            best_t = float(chunk[jmin])
    return best_t, best_val

def grid_line_min(q, bv, cv, t_lo, t_hi):
    grid = np.linspace(t_lo, t_hi, 17)
    if abs(grid[0]) > 1e-15:
        grid = np.unique(np.append(grid, 0.0))
        grid.sort()
    best_t, best_val = 0.0, float('inf')
    for _ in range(3):
        vals = np.max(q[None, :] + grid[:, None] * bv[None, :] + grid[:, None] ** 2 * cv[None, :], axis=1)
        jmin = int(np.argmin(vals))
        if vals[jmin] < best_val:
            best_val = float(vals[jmin])
            best_t = float(grid[jmin])
        if _ == 2:
            break
        if jmin == 0:
            lo_i, hi_i = 0, 1
        elif jmin == len(grid) - 1:
            lo_i, hi_i = len(grid) - 2, len(grid) - 1
        else:
            lo_i, hi_i = jmin - 1, jmin + 1
        new_pts = np.linspace(grid[lo_i], grid[hi_i], 9)
        grid = np.unique(np.concatenate([grid, new_pts]))
        grid.sort()
    return best_t, best_val

def feasible_interval(h, d):
    t_lo = 0.0
    t_hi = 0.0
    nz = np.where(d != 0.0)[0]
    for i in nz:
        di = d[i]
        if di > 0.0:
            t_lo = max(t_lo, -h[i] / di)
            t_hi = min(t_hi, (1.0 - h[i]) / di)
        else:
            t_lo = max(t_lo, (1.0 - h[i]) / di)
            t_hi = min(t_hi, -h[i] / di)
    return t_lo, t_hi

def run(seed=42, **kwargs):
    n = None
    parent_h = None
    try:
        p = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 2 <= len(p) <= 999:
            n = len(p)
            if np.all(np.isfinite(p)) and np.max(np.abs(p)) <= 1e6:
                if np.all(p >= -1e-15) and np.all(p <= 1.0 + 1e-15) and abs(np.sum(p) - n / 2.0) <= 1e-10:
                    parent_h = np.clip(p, 0.0, 1.0).copy()
                else:
                    proj = project_to_K(p, n)
                    parent_h = proj if proj is not None else np.full(n, 0.5)
            else:
                parent_h = np.full(n, 0.5)
    except Exception:
        pass
    if n is None:
        try:
            nn = kwargs.get('n_points', None)
            if isinstance(nn, int) and not isinstance(nn, bool) and 2 <= nn <= 999:
                n = nn
        except Exception:
            pass
    if n is None:
        n = 45
    if parent_h is None:
        parent_h = np.full(n, 0.5)
    dx = 2.0 / n

    fallback_h = parent_h.copy()
    fallback_score = exact_score(fallback_h, n)
    if not np.isfinite(fallback_score):
        fallback_h = np.full(n, 0.5)
        fallback_score = exact_score(fallback_h, n)

    best_h = parent_h.copy()
    best_score = fallback_score
    rng = np.random.default_rng(seed)

    starts = [fallback_h.copy()]
    comp = 1.0 - parent_h
    if is_feasible(comp, n):
        starts.append(comp.copy())
    proj1 = project_to_K(parent_h + 0.12 * rng.standard_normal(n), n)
    if proj1 is not None:
        starts.append(proj1.copy())
    proj2 = project_to_K(rng.beta(0.5, 0.5, size=n), n)
    if proj2 is not None:
        starts.append(proj2.copy())
    dedup_starts = [starts[0]]
    for s in starts[1:]:
        keep = True
        for d in dedup_starts:
            if np.max(np.abs(s - d)) <= 1e-8:
                keep = False
                break
        if keep:
            dedup_starts.append(s)
    max_starts = 4 if n <= 128 else (3 if n <= 256 else 2)
    dedup_starts = dedup_starts[:max_starts]

    records = build_tree_records(n)
    if n > 256 and len(records) > 64:
        idxs = [int(k * (len(records) - 1) / 63.0) for k in range(64)]
        sel_records = [records[i] for i in idxs]
    else:
        sel_records = records
    max_sweeps = 3 if n <= 128 else (2 if n <= 256 else 1)

    for sh in dedup_starts:
        if not is_feasible(sh, n):
            continue
        cur_h = sh.copy()
        cur_score = exact_score(cur_h, n)
        if not np.isfinite(cur_score):
            continue
        if cur_score < best_score - 1e-12:
            best_h = cur_h.copy()
            best_score = cur_score
        for sweep in range(max_sweeps):
            accepted = False
            order = list(range(len(sel_records)))
            if sweep % 2 == 1:
                order.reverse()
            for di in order:
                rec = sel_records[di]
                d = make_dir(n, rec[1], rec[2], rec[3])
                if d is None:
                    continue
                t_lo, t_hi = feasible_interval(cur_h, d)
                if not (np.isfinite(t_lo) and np.isfinite(t_hi)) or t_hi - t_lo <= 1e-12:
                    continue
                q = dx * np.correlate(cur_h, 1.0 - cur_h, mode='full')
                bv = dx * (np.correlate(d, 1.0 - cur_h, mode='full') - np.correlate(cur_h, d, mode='full'))
                cv = -dx * np.correlate(d, d, mode='full')
                if not (np.all(np.isfinite(q)) and np.all(np.isfinite(bv)) and np.all(np.isfinite(cv))):
                    continue
                if n <= 48:
                    t_star, val_pred = exact_line_min(q, bv, cv, t_lo, t_hi)
                else:
                    t_star, val_pred = grid_line_min(q, bv, cv, t_lo, t_hi)
                if val_pred >= cur_score - 1e-12:
                    continue
                h_trial = cur_h + t_star * d
                repaired = repair_small(h_trial, n)
                if repaired is None:
                    continue
                sc = exact_score(repaired, n)
                if not np.isfinite(sc) or sc >= cur_score - 1e-12:
                    continue
                cur_h = repaired
                cur_score = sc
                accepted = True
                if cur_score < best_score - 1e-12:
                    best_h = cur_h.copy()
                    best_score = cur_score
            if not accepted:
                break

    final_h = best_h.copy()
    if not is_feasible(final_h, n):
        proj = project_to_K(final_h, n)
        if proj is not None and is_feasible(proj, n):
            final_h = proj
        else:
            final_h = fallback_h.copy()
    final_score = exact_score(final_h, n)
    fb_score = exact_score(fallback_h, n)
    if not np.isfinite(final_score) or final_score > fb_score:
        final_h = fallback_h.copy()
        final_score = fb_score
    if not np.isfinite(final_score):
        final_h = np.full(n, 0.5)
        final_score = exact_score(final_h, n)
    return final_h.astype(np.float64), float(final_score), int(n)
