Now for the Frank-Wolfe on the active bundle:
- Select active lags (top 12 by q value, within tolerance)
- Center their gradients: U = G_active - mean(G_active, axis=1, keepdims=True)
- K = U @ U.T (12x12 at most)
- Minimize 0.5 * ||U.T @ alpha||^2 = 0.5 * alpha.T @ K @ alpha
  subject to alpha >= 0, sum(alpha) = 1

- alpha = uniform
- For iter in range(40):

  - j = argmin(z)

import numpy as np
import math

def exact_score(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = float(np.max(corr * dx))
    return c if math.isfinite(c) else float('inf')

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    s = float(np.sum(h))
    if abs(s - n/2.0) > 1e-10:
        return False
    s2 = math.fsum(float(x) for x in h)
    if abs(s2 - n/2.0) > 1e-10:
        return False
    return True

def project_capped_simplex(v, n):
    S = n / 2.0
    if v.shape != (n,) or not np.all(np.isfinite(v)):
        return None
    v = np.clip(v, -1e6, 1e6)
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_cand = np.clip(v - mid, 0.0, 1.0)
        if float(np.sum(h_cand)) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    resid = S - math.fsum(float(x) for x in h)
    if abs(resid) > 1e-14:
        if resid > 0:
            mask = h < 1.0 - 1e-12
            cnt = int(np.sum(mask))
            if cnt > 0:
                h[mask] += resid / cnt
        else:
            mask = h > 1e-12
            cnt = int(np.sum(mask))
            if cnt > 0:
                h[mask] += resid / cnt
    h = np.clip(h, 0.0, 1.0)
    resid2 = S - float(np.sum(h))
    if abs(resid2) > 1e-14:
        if resid2 > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += resid2 / cnt
            h = np.clip(h, 0.0, 1.0)
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return None
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return None
    if abs(float(np.sum(h)) - S) > 1e-8:
        return None
    return h

def lag_data(h, n):
    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        idx = np.arange(n - d)
        q[d] = dx * np.sum(h[idx + d] * (1.0 - h[idx]))
        G[d, idx + d] += dx * (1.0 - h[idx])
        G[d, idx] += -dx * h[idx + d]
        qn = n - 1 + d
        q[qn] = dx * np.sum(h[idx] * (1.0 - h[idx + d]))
        G[qn, idx] += dx * (1.0 - h[idx + d])
        G[qn, idx + d] += -dx * h[idx]
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None, None
    return q, G

def active_bundle_gradient(q, G):
    Cq = float(np.max(q))
    tol = max(1e-12, 1e-7 * max(1.0, abs(Cq)))
    active = np.where(q >= Cq - tol)[0]
    if len(active) > 12:
        order = np.argsort(q[active])[::-1][:12]
        active = active[order]
    U = G[active, :].copy()
    U -= np.mean(U, axis=1, keepdims=True)
    if not np.all(np.isfinite(U)):
        return None
    if len(active) == 1:
        return U[0]
    K = U @ U.T
    K = (K + K.T) / 2.0
    m = len(active)
    alpha = np.full(m, 1.0 / m)
    for _ in range(40):
        z = K @ alpha
        j = int(np.argmin(z))
        if float(alpha @ z) - z[j] <= 1e-12:
            break
        v = np.zeros(m)
        v[j] = 1.0
        v -= alpha
        vKv = float(v @ K @ v)
        if vKv <= 1e-24:
            break
        gamma = np.clip(-float(v @ z) / vKv, 0.0, 1.0)
        alpha += gamma * v
        alpha = np.clip(alpha, 0.0, None)
        s = float(np.sum(alpha))
        if s < 1e-30:
            break
        alpha /= s
    g = alpha @ U
    if not np.all(np.isfinite(g)):
        j0 = int(np.argmax(q))
        g0 = G[j0, :] - np.mean(G[j0, :])
        return g0 if np.all(np.isfinite(g0)) else None
    return g

def local_search(h0, rng, n, max_iter):
    h = project_capped_simplex(h0, n)
    if h is None:
        h = np.full(n, 0.5)
    cur_score = exact_score(h, n)
    best_h = h.copy()
    best_score = cur_score
    eta = 0.5
    for it in range(max_iter):
        q, G = lag_data(h, n)
        if q is None:
            break
        g = active_bundle_gradient(q, G)
        improved = False
        if g is not None:
            for b in range(8):
                eta_try = eta * (0.5 ** b)
                prop = project_capped_simplex(h - eta_try * g, n)
                if prop is None or np.max(np.abs(prop - h)) < 1e-14:
                    continue
                sc = exact_score(prop, n)
                if sc < cur_score - 1e-12:
                    h = prop
                    cur_score = sc
                    if sc < best_score - 1e-12:
                        best_h = h.copy()
                        best_score = sc
                    eta = min(2.0, 1.25 * eta_try)
                    improved = True
                    break
        if not improved:
            eta = max(1e-6, eta / 2.0)
            if it < 40:
                sigma, T = 0.12, 0.002
            elif it < 80:
                sigma, T = 0.07, 0.0008
            else:
                sigma, T = 0.035, 0.00025
            best_prop = None
            best_prop_sc = cur_score
            for _ in range(4):
                prop = project_capped_simplex(h + sigma * rng.standard_normal(n), n)
                if prop is None:
                    continue
                sc = exact_score(prop, n)
                if sc < best_prop_sc:
                    best_prop = prop
                    best_prop_sc = sc
            if best_prop is not None:
                delta = best_prop_sc - cur_score
                accept = False
                if delta <= 0:
                    accept = True
                else:
                    accept = rng.random() < math.exp(-delta / T)
                if accept:
                    h = best_prop
                    cur_score = best_prop_sc
                    eta = 0.25
                    if cur_score < best_score - 1e-12:
                        best_h = h.copy()
                        best_score = cur_score
    final_score = exact_score(best_h, n)
    return best_h, final_score

                if uv is not None:
                    add_unique(starts, uv)
        else:
            for _ in range(4):
                uv = project_capped_simplex(rng.random(n), n)
                if uv is not None:
                    add_unique(starts, uv)
        starts = starts[:12]
    max_iter = 120 if n <= 64 else (80 if n <= 128 else 0)
    if max_iter > 0:
        for sv in starts:
            try:
                lv, lsc = local_search(sv, rng, n, max_iter)
                if is_feasible(lv, n) and lsc < global_best_sc - 1e-12:
                    global_best_h, global_best_sc = lv.copy(), lsc
            except Exception:
                pass
    candidates = [(global_best_h, global_best_sc)]
    if parent_fb is not None:
        candidates.append((parent_fb, parent_fb_sc))
    candidates.append((const_h, const_sc))
    fh, fs = min(candidates, key=lambda x: x[1])
    if not is_feasible(fh, n):
        rp = project_capped_simplex(fh, n)
        if rp is not None and is_feasible(rp, n):
            fh, fs = rp, exact_score(rp, n)
        else:
            fh, fs = const_h, const_sc
    fs = exact_score(fh, n)
    if not math.isfinite(fs):
        fs = const_sc
        fh = const_h
    return fh.astype(np.float64), float(fs), int(n)
