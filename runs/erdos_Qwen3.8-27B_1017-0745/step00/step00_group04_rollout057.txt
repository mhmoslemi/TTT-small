def project_capped_simplex(v, S, N):
    v = np.clip(v, -1e6, 1e6)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(v - mid, 0.0, 1.0)
        s = float(np.sum(h))
        if s > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    # mass correction
    res = S - float(np.sum(h))
    if abs(res) > 1e-14:
        if res > 0:
            mask = h < 1.0
            cap = np.sum(1.0 - h[mask])
            add = min(res, cap)
            n_add = int(np.sum(mask))
            if n_add > 0:
                h[mask] += add / n_add
        else:
            mask = h > 0.0
            cap = np.sum(h[mask])
            rem = min(-res, cap)
            n_rem = int(np.sum(mask))
            if n_rem > 0:
                h[mask] -= rem / n_rem
    # final check
    for i in range(N):
        h[i] = min(max(h[i], 0.0), 1.0)
    s2 = float(np.sum(h))
    if abs(s2 - S) > 1e-10:
        # find interior coordinate to fix
        for j in range(N):
            if 0.0 < h[j] < 1.0:
                others = float(np.sum(h)) - h[j]
                h[j] = S - others
                if 0.0 <= h[j] <= 1.0:
                    break
    if not np.all((h >= 0) & (h <= 1)) or abs(float(np.sum(h)) - S) > 1e-9:
        h = np.full(N, 0.5)
    return h

def soft_fun(h, dx, N, tau):
    q, G = compute_lags(h, dx, N)
    qmax = np.max(q)
    shifted = (q - qmax) / tau
    exps = np.exp(shifted)
    logsum = np.log(np.sum(exps))
    phi = qmax + tau * logsum
    weights = exps / np.sum(exps)
    grad = G.T @ weights
    return phi, grad

                        se = exact_score(h_e)
                        if se < best_score - 1e-12:
                            best_score = se
                            best_h = h_e.copy()
        except Exception:
            pass
    # Final polish on global best
    q0, _ = compute_lags(best_h, dx, N)
    t0 = float(np.max(q0)) + max(1e-8, 1e-6 * abs(float(np.max(q0))))
    z0 = np.append(best_h, t0)
    ep_bounds = bounds + [(0.0, None)]
    ep_con = [{'type': 'eq', 'fun': epi_mass_eq, 'jac': epi_mass_jac},
              {'type': 'ineq', 'fun': epi_ineq, 'jac': epi_ineq_jac}]
    try:
        res3 = minimize(epi_fun, z0, method='SLSQP', jac=epi_jac,
                        bounds=ep_bounds, constraints=ep_con,
                        options={'ftol': 1e-10, 'maxiter': 200})
        if res3 is not None and res3.x is not None:
            z_new = np.asarray(res3.x, dtype=float).ravel()
            if len(z_new) == N + 1 and np.all(np.isfinite(z_new)):
                h_f = project_capped(z_new[:N], S, N)
                if feasible_check(h_f, S, N):
                    sf = exact_score(h_f)
                    if sf < best_score - 1e-12:
                        best_score = sf
                        best_h = h_f.copy()
    except Exception:
        pass
    best_h = project_capped(best_h, S, N)
    if not feasible_check(best_h, S, N):
        best_h = np.full(N, 0.5)
    final_score = float(exact_score(best_h))
    return (best_h, final_score, int(N))
