    return abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n)


def project_capped_simplex(z, s):
    n = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(n, 0.5)
    z = np.clip(z, -1e12, 1e12)
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(z - mid, 0.0, 1.0)
        if np.sum(h) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) <= 1e-13 * max(1, n):
            break
        if r > 0:
            slack = 1.0 - h
            idx = int(np.argmax(slack))
            move = min(r, slack[idx])
            if move <= 0:
                break
            h[idx] += move
            r -= move
        else:
            idx = int(np.argmax(h))
            move = min(-r, h[idx])
            if move <= 0:
                break
            h[idx] -= move
            r += move
    h = np.clip(h, 0.0, 1.0)
    if not feasibility_check(h, n):
        return np.full(n, 0.5)
    return h


def exact_score(h):
    n = len(h)
    if not feasibility_check(h, n):
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, 'full')
    return dx * np.max(corr)


def smooth_value(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = dx * np.correlate(h, 1.0 - h, 'full')
    cm = np.max(c)
    a = np.exp((c - cm) / tau)
    return cm + tau * np.log(np.sum(a))


def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = dx * np.correlate(h, 1.0 - h, 'full')
    cm = np.max(c)
    a = np.exp((c - cm) / tau)
    p = a / np.sum(a)
    g = np.zeros(n)
    for u in range(2 * n - 1):
        if p[u] < 1e-16:
            continue
        d = u - (n - 1)
        i0 = max(d, 0)
        j0 = max(-d, 0)
        m = n - abs(d)
        for r in range(m):
            i = i0 + r
            j = j0 + r
            g[i] += p[u] * (1.0 - h[j])
            g[j] += -p[u] * h[i]
    return dx * g


def build_starts(parent_arr, n, rng):
    s = n / 2.0
    cands = []
    if parent_arr is not None:
        base = project_capped_simplex(parent_arr.astype(np.float64).copy(), s)
    else:
        base = np.full(n, 0.5)
    cands.append(base)
    cands.append(project_capped_simplex(base[::-1].copy(), s))
    cands.append(project_capped_simplex(1.0 - base[::-1].copy(), s))
    cands.append(np.full(n, 0.5))
    blk = np.zeros(n)
    k = n // 2
    blk[:k] = 1.0
    if n % 2 == 1:
        blk[k] = 0.5
    cands.append(project_capped_simplex(blk, s))
    alt = np.zeros(n)
    alt[::2] = 1.0
    cands.append(project_capped_simplex(alt, s))
    for std in [0.05, 0.15, 0.35]:
        cands.append(project_capped_simplex(base + rng.normal(0, std, n), s))
    cands.append(project_capped_simplex(rng.uniform(0, 1, n), s))
    starts = []
    for c in cands:
        if not feasibility_check(c, n):
            continue
        dup = False
        for s2 in starts:
            if np.max(np.abs(c - s2)) < 1e-10:
                dup = True
                break
        if not dup:
            starts.append(c)
    return starts


def run(seed=42, **kwargs):
    n = 47
    parent_arr = None
    if 'initial_h_values' in globals():
        try:
            ph = np.array(initial_h_values, dtype=np.float64).flatten()
            if ph.size > 0 and np.all(np.isfinite(ph)):
                parent_arr = ph.copy()
                n = len(ph)
        except Exception:
            pass
    if parent_arr is None and 'n_points' in kwargs:
        try:
            nv = int(kwargs['n_points'])
            if 1 <= nv <= 999:
                n = nv
        except Exception:
            pass

    s = n / 2.0
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)

    if n > 999:
        if parent_arr is not None:
            h_fb = project_capped_simplex(parent_arr, s)
        else:
            h_fb = np.full(n, 0.5)
        sc = exact_score(h_fb)
        if not np.isfinite(sc):
            h_fb = np.full(n, 0.5)
            sc = exact_score(h_fb)
        return h_fb, float(sc), n

    if n <= 64:
        cap_starts, cap_iters = 12, 150
        n_starts, iters_tau = 10, 100
    elif n <= 256:
        cap_starts, cap_iters = 6, 90
        n_starts, iters_tau = 5, 60
    else:
        cap_starts, cap_iters = 2, 40
        n_starts, iters_tau = 2, 25

    if 'restarts' in kwargs:
        try:
            n_starts = max(1, min(int(kwargs['restarts']), cap_starts))
        except Exception:
            pass
    if 'iters_per_tau' in kwargs:
        try:
            iters_tau = max(1, min(int(kwargs['iters_per_tau']), cap_iters))
        except Exception:
            pass

    temps = [0.02, 0.005, 0.001, 0.0002, 0.00005, 0.00001]
    starts = build_starts(parent_arr, n, rng)
    while len(starts) < n_starts:
        starts.append(project_capped_simplex(rng.uniform(0, 1, n), s))
    starts = starts[:n_starts]

    if parent_arr is not None:
        h_inc = project_capped_simplex(parent_arr, s)
    else:
        h_inc = np.full(n, 0.5)
    sc_inc = exact_score(h_inc)
    h_fb = h_inc.copy()
    sc_fb = sc_inc

    for h_s in starts:
        sc_s = exact_score(h_s)
        if np.isfinite(sc_s) and sc_s < sc_inc - 1e-12:
            h_inc = h_s.copy()
            sc_inc = sc_s

    for h_s in starts:
        x = h_s.copy()
        try:
            for tau in temps:
                Fx = smooth_value(x, tau)
                if not np.isfinite(Fx):
                    break
                g = smooth_grad(x, tau)
                if not np.all(np.isfinite(g)):
                    break
                alpha = 1.0
                stall = 0
                for _ in range(iters_tau):
                    y = project_capped_simplex(x - alpha * g, s)
                    D = np.sum((y - x) ** 2)
                    if D <= 1e-18:
                        break
                    Fy = smooth_value(y, tau)
                    if not np.isfinite(Fy):
                        alpha *= 0.5
                        if alpha < 1e-12:
                            break
                        continue
                    if Fy <= Fx - 1e-4 * D / alpha + 1e-14 * max(1.0, abs(Fx)):
                        imp = Fx - Fy
                        if imp <= 1e-12 * max(1.0, abs(Fx)):
                            stall += 1
                            if stall >= 5:
                                break
                        else:
                            stall = 0
                        x = y.copy()
                        Fx = Fy
                        g = smooth_grad(x, tau)
                        if not np.all(np.isfinite(g)):
                            break
                        alpha = min(2.0, 1.25 * alpha)
                        sc_x = exact_score(x)
                        if np.isfinite(sc_x) and sc_x < sc_inc - 1e-12:
                            h_inc = x.copy()
                            sc_inc = sc_x
                    else:
                        alpha *= 0.5
                        if alpha < 1e-12:
                            break
        except Exception:
            continue

    h_final = project_capped_simplex(h_inc, s)
    sc_final = exact_score(h_final)
    sc_fb_final = exact_score(h_fb)
    if np.isfinite(sc_fb_final) and sc_fb_final < sc_final:
        h_final = h_fb.copy()
        sc_final = sc_fb_final
    if not np.isfinite(sc_final) or not feasibility_check(h_final, n):
        h_final = np.full(n, 0.5)
        sc_final = exact_score(h_final)
    return h_final.astype(np.float64), float(sc_final), int(n)
