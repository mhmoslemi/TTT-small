def feasibility_check(h, n, S):

    if abs(np.sum(h) - S) > 1e-10:
        return False
    return True

def project_to_K(y, n, S):
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return None
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        x = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > S:
            lo = mid
        else:
            hi = mid
        if abs(s - S) < 1e-13 or (hi - lo) < 1e-15:
            break
    x = np.clip(y - 0.5*(lo+hi), 0.0, 1.0)
    r = S - np.sum(x)
    if abs(r) > 1e-12:
        if r > 0:
            slack = 1.0 - x
            total_slack = np.sum(slack)

        else:
            total_pos = np.sum(x)
            if total_pos < -r:
                return None
            x += r / total_pos * x

def _feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and

            abs(np.sum(h) - S) <= 1e-10)

def _project(y, n, S):

        return None
    lo, hi = float(np.min(y)) - 1.0, float(np.max(y)) + 1.0

        s = np.sum(np.clip(y-mid, 0.0, 1.0))
