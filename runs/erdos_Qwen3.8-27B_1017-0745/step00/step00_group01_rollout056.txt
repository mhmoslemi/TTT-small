def check_feasibility(h, n):
    if h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-15) or np.any(h > 1.0 + 1e-15):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1.0, n)


def project_capped_simplex(z, n):
    if z is None or z.ndim != 1 or len(z) != n or not np.all(np.isfinite(z)):
        return np.full(n, 0.5, dtype=np.float64)
    z = np.clip(z, -1e12, 1e12).astype(np.float64)
    s = n / 2.0
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(z - mid, 0.0, 1.0)
        if np.sum(h) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) <= 1e-13:
            break
        if r > 0:
            slack = 1.0 - h
            i = int(np.argmax(slack))
            move = min(r, slack[i])
            if move <= 0:
                break
            h[i] += move
            r -= move
        else:
            i = int(np.argmax(h))
            move = min(-r, h[i])
            if move <= 0:
                break
            h[i] -= move
            r += move
    h = np.clip(h, 0.0, 1.0)
    if not check_feasibility(h, n):
        return np.full(n, 0.5, dtype=np.float64)
    return h


def exact_score(h):
    n = len(h)
    if n == 0 or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-15) or np.any(h > 1.0 + 1e-15):
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full')
    return float(dx * np.max(corr))


def smooth_value_only(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = dx * corr
    c_max = float(np.max(c))
    a = np.exp((c - c_max) / tau)
    sa = float(np.sum(a))
    if not np.isfinite(sa) or sa <= 0:
        return float('inf')
    return c_max + tau * math.log(sa)


def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = dx * corr
    c_max = float(np.max(c))
    a = np.exp((c - c_max) / tau)
    sa = float(np.sum(a))
    if not np.isfinite(sa) or sa <= 0:
        return None
    p = a / sa
    U = np.convolve(p, 1.0 - h, mode='full')
    V = np.convolve(p[::-1], h, mode='full')
    grad = dx * (U[n - 1:2 * n - 1] - V[n - 1:2 * n - 1])
    if not np.all(np.isfinite(grad)):
        return None
    return grad


def make_starts(n, base, rng):
    starts = []
    starts.append(base.copy())
    starts.append(base[::-1].copy())
    starts.append((1.0 - base[::-1]).copy())
    starts.append(np.full(n, 0.5, dtype=np.float64))
    block = np.zeros(n, dtype=np.float64)
    k = n // 2
    block[:k] = 1.0
    if n % 2 == 1:
        block[k] = 0.5
    starts.append(block)
    alt = np.zeros(n, dtype=np.float64)
    alt[0::2] = 1.0
    starts.append(alt)
    for sd in [0.05, 0.15, 0.35]:
        starts.append(base + rng.normal(0, sd, n))
    starts.append(rng.uniform(0, 1, n))
    projected = [project_capped_simplex(s, n) for s in starts]
    unique = []
    for s in projected:
        dup = False
        for u in unique:
            if np.max(np.abs(s - u)) < 1e-10:
                dup = True
                break
        if not dup:
            unique.append(s)
    return unique

    except NameError:
        pass
    except Exception:
        pass

    if n is None:
        nkw = kwargs.get('n_points', 47)
        if isinstance(nkw, (int, np.integer)) and 1 <= nkw <= 999:
            n = int(nkw)
        elif isinstance(nkw, float) and nkw == int(nkw) and 1 <= nkw <= 999:
            n = int(nkw)
        else:
            n = 47
        base = np.full(n, 0.5, dtype=np.float64)

    try:
        s = int(seed)
    except Exception:
        s = 42
    rng = np.random.default_rng(s)

    if n <= 64:
        max_starts, max_iter = 10, 100
    elif n <= 256:
        max_starts, max_iter = 5, 60
    else:
        max_starts, max_iter = 2, 25

    rk = kwargs.get('restarts', None)
    if isinstance(rk, (int, np.integer)) and rk > 0:
        cap = 12 if n <= 64 else (6 if n <= 256 else 2)
        max_starts = min(int(rk), cap)
    ik = kwargs.get('iters_per_tau', None)
    if isinstance(ik, (int, np.integer)) and ik > 0:
        cap = 150 if n <= 64 else (90 if n <= 256 else 40)
        max_iter = min(int(ik), cap)

    fallback = base.copy()
    fallback_score = exact_score(fallback)

    if n > 999:
        final = project_capped_simplex(fallback, n)
        sc = exact_score(final)
        if not check_feasibility(final, n):
            final = np.full(n, 0.5, dtype=np.float64)
            sc = exact_score(final)
        return (final, sc, n)

    starts = make_starts(n, base, rng)
    if len(starts) > max_starts:
        starts = starts[:max_starts]

    temps = [0.02, 0.005, 0.001, 0.0002, 0.00005, 0.00001]
    incumbent = fallback.copy()
    incumbent_score = fallback_score

    for start in starts:
        try:
            sc = exact_score(start)
            if sc < incumbent_score - 1e-12:
                incumbent = start.copy()
                incumbent_score = sc
            x = start.copy()
            for tau in temps:
                alpha = 1.0
                stall = 0
                F_x, _ = None, None
                for it in range(max_iter):
                    F_x = smooth_value_only(x, tau)
                    if not np.isfinite(F_x):
                        break
                    grad = smooth_grad(x, tau)
                    if grad is None:
                        break
                    y = project_capped_simplex(x - alpha * grad, n)
                    D = float(np.sum((y - x) ** 2))
                    if float(np.max(np.abs(y - x))) <= 1e-9:
                        break
                    F_y = smooth_value_only(y, tau)
                    if not np.isfinite(F_y):
                        alpha *= 0.5
                        if alpha < 1e-12:
                            break
                        continue
                    tol = 1e-4 * D / alpha + 1e-14 * max(1.0, abs(F_x))
                    if F_y <= F_x - tol:
                        imp = F_x - F_y
                        if imp <= 1e-12 * max(1.0, abs(F_x)):
                            stall += 1
                        else:
                            stall = 0
                        x = y
                        alpha = min(2.0, 1.25 * alpha)
                        sc_x = exact_score(x)
                        if sc_x < incumbent_score - 1e-12:
                            incumbent = x.copy()
                            incumbent_score = sc_x
                        if stall >= 5:
                            break
                    else:
                        alpha *= 0.5
                        if alpha < 1e-12:
                            break
                sc_x = exact_score(x)
                if sc_x < incumbent_score - 1e-12:
                    incumbent = x.copy()
                    incumbent_score = sc_x
        except Exception:
            continue

    final = project_capped_simplex(incumbent, n)
    final_score = exact_score(final)
    fb_score = exact_score(fallback)
    if final_score > fb_score - 1e-12:
        final = fallback.copy()
        final_score = fb_score
    if not check_feasibility(final, n):
        final = np.full(n, 0.5, dtype=np.float64)
        final_score = exact_score(final)

    return (final, final_score, n)
