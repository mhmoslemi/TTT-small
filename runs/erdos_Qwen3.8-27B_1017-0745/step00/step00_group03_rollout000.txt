    starts.append(project_box_sum(make_fourier_start(N, rng, 3, 0.15), m))

    # Optimize each start
    for h0 in starts:
        s0 = score_exact(h0, N)
        if s0 < best_score:
            best = h0.copy()
            best_score = s0
        h_opt = slsqp_solve(h0, N, m)
        s_opt = score_exact(h_opt, N)
        if s_opt < best_score:
            best = h_opt.copy()
            best_score = s_opt

    # Basin hops
    for sc in [0.10, 0.05, 0.025, 0.0125]:
        hop = best + sc * rng.standard_normal(N)
        hop = project_box_sum(hop, m)
        s_hop = score_exact(hop, N)
        if s_hop < best_score:
            best = hop.copy()
            best_score = s_hop
        h_opt = slsqp_solve(hop, N, m)
        s_opt = score_exact(h_opt, N)
        if s_opt < best_score:
            best = h_opt.copy()
            best_score = s_opt

    # Final validation
    best = project_box_sum(best, m)
    c5 = score_exact(best, N)
    if not np.isfinite(c5):
        best = np.full(N, 0.5)
        c5 = score_exact(best, N)
    return best, c5, int(N)
