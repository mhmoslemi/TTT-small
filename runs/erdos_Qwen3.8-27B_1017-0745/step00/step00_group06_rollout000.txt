                res = minimize(smooth_obj, cur, jac=smooth_jac, method='SLSQP',
                               bounds=bounds_h, constraints=cons_h, options=opts_h)
                if res.x is not None and res.x.shape == (N,) and np.all(np.isfinite(res.x)):
                    cur = project_bm(res.x)
                else:
                    cur = project_bm(cur)
            except Exception:
                cur = project_bm(cur)
            sc = exact_score(cur)
            candidates.append((cur.copy(), sc))
            if sc < s_best:
                s_best = sc
                h_best = cur.copy()
    candidates.sort(key=lambda t: t[1])
    best_cands = []
    for c, s in candidates:
        if all(np.max(np.abs(c - bc)) > 1e-6 for bc, _ in best_cands):
            best_cands.append((c, s))
        if len(best_cands) >= 3:
            break
    bounds_e = [(0.0, 1.0)] * N + [(0.0, 2.1)]
    cons_e = [
        {'type': 'eq', 'fun': mass_eq_e, 'jac': mass_eq_e_jac},
        {'type': 'ineq', 'fun': epi_ineq, 'jac': epi_ineq_jac}
    ]
    opts_e = {'ftol': 1e-10, 'maxiter': min(400, 2 * maxiter), 'disp': False}
    for hc, _ in best_cands:
        z0 = np.zeros(N + 1)
        z0[:-1] = hc
        z0[-1] = exact_score(hc) + 1e-6
        try:
            res = minimize(epi_obj, z0, jac=epi_obj_jac, method='SLSQP',
                           bounds=bounds_e, constraints=cons_e, options=opts_e)
            if res.x is not None and res.x.shape == (N + 1,) and np.all(np.isfinite(res.x)):
                h_try = project_bm(res.x[:-1])
            else:
                h_try = project_bm(hc)
        except Exception:
            h_try = project_bm(hc)
        sc = exact_score(h_try)
        if sc < s_best - 1e-12:
            s_best = sc
            h_best = h_try.copy()
    h_final = project_bm(h_best)
    if not (np.all(h_final >= -1e-12) and np.all(h_final <= 1.0 + 1e-12) and
            abs(np.sum(h_final) - _m) < 1e-9 * max(1, N)):
        h_final = np.full(N, 0.5)
    c5_final = exact_score(h_final)
    return (h_final, c5_final, N)
