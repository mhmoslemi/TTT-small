    if h.ndim != 1 or len(h) != n or not np.all(np.isfinite(h)):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr) * (2.0 / n))

def project_to_K(y, S):
    n = len(y)
    if not np.all(np.isfinite(y)):
        return None
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        x = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > S:
            lo = mid
        else:
            hi = mid
        if abs(s - S) < 1e-13 or (hi - lo) < 1e-14 * max(1.0, abs(mid)):
            break
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(x)
    if resid > 0:
        slacks = 1.0 - x
        idx = np.argsort(slacks)[::-1]
        for j in idx:
            add = min(resid, slacks[j])
            x[j] += add
            resid -= add
            if resid < 1e-15:
                break
    elif resid < 0:
        rem = -resid
        idx = np.argsort(x)
        for j in idx:
            take = min(rem, x[j])
            x[j] -= take
            rem -= take
            if rem < 1e-15:
                break
    if np.any(x < -1e-10) or np.any(x > 1.0 + 1e-10):
        return None
    x = np.clip(x, 0.0, 1.0)
    if abs(np.sum(x) - S) > 1e-10:
        return None
    return x

def repair_small(h, S):
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    x = np.clip(h, 0.0, 1.0)
    resid = S - np.sum(x)
    if abs(resid) > 1e-8:
        return None
    if resid > 0:
        idx = np.argsort(1.0 - x)[::-1]
        for j in idx:
            add = min(resid, 1.0 - x[j])
            x[j] += add
            resid -= add
            if resid < 1e-15:
                break
    elif resid < 0:
        rem = -resid
        idx = np.argsort(x)
        for j in idx:
            take = min(rem, x[j])
            x[j] -= take
            rem -= take
            if rem < 1e-15:
                break
    if np.any(x < 0.0) or np.any(x > 1.0) or abs(np.sum(x) - S) > 1e-10:
        return None
    return x

def build_tree(n):
    records = []
    stack = [(0, n, 0)]
    while stack:
        a, b, depth = stack.pop()
        if b - a <= 1:
            continue
        m = (a + b) // 2
        records.append((depth, a, m, b))
        stack.append((a, m, depth + 1))
        stack.append((m, b, depth + 1))
    max_d = max(r[0] for r in records) if records else 0
    ordered = []
    for d in range(max_d + 1):
        for r in records:
            if r[0] == d:
                ordered.append(r)
    return ordered

def make_direction(rec, n):
    depth, a, m, b = rec
    d = np.zeros(n)
    L = m - a
    R = b - m
    d[a:m] = 1.0 / L
    d[m:b] = -1.0 / R
    s = np.sum(d)
    if abs(s) > 1e-15:
        nz = np.nonzero(d)[0]
        d[nz[0]] -= s
    return d

def exact_line_min(q, b, c, t_lo, t_hi):
    m = len(q)
    cands = [t_lo, t_hi, 0.0]
    tol_ext = 1e-12 * max(1.0, abs(t_lo), abs(t_hi))
    for r in range(m):
        for s2 in range(r + 1, m):
            A = c[r] - c[s2]
            B = b[r] - b[s2]
            C = q[r] - q[s2]
            scale = max(abs(A), abs(B), abs(C), 1e-300)
            if abs(A) < 1e-14 * scale:
                if abs(B) > 1e-14 * scale:
                    t = -C / B
                    if np.isfinite(t) and t_lo - tol_ext <= t <= t_hi + tol_ext:
                        cands.append(max(t_lo, min(t_hi, t)))
            else:
                disc = B * B - 4.0 * A * C
                tol_disc = 1e-14 * (B * B + abs(4.0 * A * C) + 1e-300)
                if disc < -tol_disc:
                    continue
                if disc < 0:
                    disc = 0.0
                sq = math.sqrt(disc)
                if B > 0:
                    t1 = (-B - sq) / (2.0 * A)
                    t2 = (-B + sq) / (2.0 * A)
                elif B < 0:
                    t1 = (-B + sq) / (2.0 * A)
                    t2 = (-B - sq) / (2.0 * A)
                else:
                    t1 = t2 = sq / (2.0 * abs(A))
                for t in [t1, t2]:
                    if np.isfinite(t) and t_lo - tol_ext <= t <= t_hi + tol_ext:
                        cands.append(max(t_lo, min(t_hi, t)))
    for r in range(m):
        if abs(c[r]) > 1e-300:
            t = -b[r] / (2.0 * c[r])
            if np.isfinite(t) and t_lo - tol_ext <= t <= t_hi + tol_ext:
                cands.append(max(t_lo, min(t_hi, t)))
    cands = sorted(set(round(t, 15) for t in cands))
    best_t = 0.0
    best_val = float('inf')
    for i in range(0, len(cands), 256):
        ts = np.array(cands[i:i + 256])
        vals = q[None, :] + ts[:, None] * b[None, :] + ts[:, None] ** 2 * c[None, :]
        for j in range(len(ts)):
            v = float(np.max(vals[j]))
            if v < best_val:
                best_val = v
                best_t = float(ts[j])
    return best_t, best_val

def grid_line_min(q, b, c, t_lo, t_hi):
    cands = np.unique(np.append(np.linspace(t_lo, t_hi, 17), 0.0))
    vals = np.max(q[None, :] + cands[:, None] * b[None, :] + cands[:, None] ** 2 * c[None, :], axis=1)
    best_idx = int(np.argmin(vals))
    for _ in range(2):
        i0 = max(0, best_idx - 1)
        i1 = min(len(cands) - 1, best_idx + 1)
        new_pts = np.linspace(cands[i0], cands[i1], 9)
        cands = np.unique(np.append(cands, new_pts))
        vals = np.max(q[None, :] + cands[:, None] * b[None, :] + cands[:, None] ** 2 * c[None, :], axis=1)
        best_idx = int(np.argmin(vals))
    return float(cands[best_idx]), float(vals[best_idx])

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        pv = globals().get('initial_h_values', None)
        if pv is not None:
            parent = np.array(pv, dtype=np.float64).ravel()
            if 2 <= len(parent) <= 999:
                n = len(parent)
    except Exception:
        parent = None
    if n is None:
        try:
            nk = kwargs.get('n_points', None)
            if nk is not None and isinstance(nk, int) and not isinstance(nk, bool) and 2 <= nk <= 999:
                n = nk
        except Exception:
            pass
    if n is None:
        n = 45
    S = n / 2.0
    dx = 2.0 / n
    fallback_h = np.full(n, 0.5)
    if parent is not None:
        try:
            if np.all(np.isfinite(parent)) and np.all(np.abs(parent) <= 1e6):
                if np.all(parent >= 0) and np.all(parent <= 1) and abs(np.sum(parent) - S) < 1e-10:
                    fallback_h = parent.copy()
                else:
                    proj = project_to_K(parent, S)
                    if proj is not None:
                        fallback_h = proj
        except Exception:
            fallback_h = np.full(n, 0.5)
    fallback_score = evaluate_exact(fallback_h, n)
    best_h = fallback_h.copy()
    best_score = fallback_score
    rng = np.random.default_rng(seed)
    starts = [fallback_h.copy()]
    if parent is not None:
        comp = 1.0 - parent
        if np.all(comp >= 0) and np.all(comp <= 1) and abs(np.sum(comp) - S) < 1e-8:
            starts.append(comp.copy())
        else:
            pc = project_to_K(1.0 - parent, S)
            if pc is not None:
                starts.append(pc)
        noisy = parent + 0.12 * rng.standard_normal(n)
        pn = project_to_K(noisy, S)
        if pn is not None:
            starts.append(pn)
        beta_v = rng.beta(0.5, 0.5, n)
        pb = project_to_K(beta_v, S)
        if pb is not None:
            starts.append(pb)
    else:
        uni = rng.uniform(0, 1, n)
        pu = project_to_K(uni, S)
        if pu is not None:
            starts.append(pu)
        beta_v = rng.beta(0.5, 0.5, n)
        pb = project_to_K(beta_v, S)
        if pb is not None:
            starts.append(pb)
    max_starts = 4 if n <= 128 else (3 if n <= 256 else 2)
    starts = starts[:max_starts]
    dedup = [starts[0]]
    for s in starts[1:]:
        if all(float(np.max(np.abs(s - d))) > 1e-8 for d in dedup):
            dedup.append(s)
    starts = dedup
    tree_records = build_tree(n)
    if n <= 256 and len(tree_records) <= 256:
        dir_recs = tree_records
    else:
        M = len(tree_records)
        if M > 64:
            dir_recs = [tree_records[int(k * (M - 1) / 63)] for k in range(64)]
        else:
            dir_recs = tree_records
    use_exact = n <= 48
    max_sweeps = 3 if n <= 128 else (2 if n <= 256 else 1)
    for start in starts:
        cur_h = start.copy()
        cur_score = evaluate_exact(cur_h, n)
        if cur_score < best_score - 1e-12:
            best_h = cur_h.copy()
            best_score = cur_score
        try:
            for sweep in range(max_sweeps):
                improved = False
                recs = dir_recs if sweep % 2 == 0 else list(reversed(dir_recs))
                for rec in recs:
                    d = make_direction(rec, n)
                    nz = np.nonzero(d)[0]
                    if len(nz) == 0:
                        continue
                    t_lo = -1e18
                    t_hi = 1e18
                    for i in nz:
                        if d[i] > 0:
                            t_lo = max(t_lo, -cur_h[i] / d[i])
                            t_hi = min(t_hi, (1.0 - cur_h[i]) / d[i])
                        else:
                            t_lo = max(t_lo, (1.0 - cur_h[i]) / d[i])
                            t_hi = min(t_hi, -cur_h[i] / d[i])
                    if t_hi - t_lo < 1e-12:
                        continue
                    q = dx * np.correlate(cur_h, 1.0 - cur_h, mode='full')
                    bv = dx * (np.correlate(d, 1.0 - cur_h, mode='full') - np.correlate(cur_h, d, mode='full'))
                    cv = -dx * np.correlate(d, d, mode='full')
                    if not (np.all(np.isfinite(q)) and np.all(np.isfinite(bv)) and np.all(np.isfinite(cv))):
                        continue
                    if use_exact:
                        t_best, pred_val = exact_line_min(q, bv, cv, t_lo, t_hi)
                    else:
                        t_best, pred_val = grid_line_min(q, bv, cv, t_lo, t_hi)
                    if not np.isfinite(t_best) or pred_val >= cur_score - 1e-12:
                        continue
                    h_trial = cur_h + t_best * d
                    h_rep = repair_small(h_trial, S)
                    if h_rep is None:
                        continue
                    new_score = evaluate_exact(h_rep, n)
                    if new_score < cur_score - 1e-12:
                        cur_h = h_rep
                        cur_score = new_score
                        improved = True
                        if cur_score < best_score - 1e-12:
                            best_h = cur_h.copy()
                            best_score = cur_score
                if not improved:
                    break
        except Exception:
            break
    fh = best_h.copy()
    if not np.all(np.isfinite(fh)) or np.any(fh < -1e-10) or np.any(fh > 1.0 + 1e-10) or abs(np.sum(fh) - S) > 1e-10:
        fh = project_to_K(fh, S)
        if fh is None:
            fh = np.full(n, 0.5)
    fs = evaluate_exact(fh, n)
    fb = evaluate_exact(fallback_h, n)
    if fs < fb - 1e-12:
        return fh.copy(), fs, n
    return fallback_h.copy(), fb, n
