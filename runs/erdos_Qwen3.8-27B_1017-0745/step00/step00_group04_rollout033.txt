            blk[int(n / 2)] = 0.5
        patterns.append(blk)
        idx_f = np.arange(n, dtype=np.float64)
        sin_p = project_capped_simplex(0.5 + 0.45 * np.sin(2.0 * math.pi * idx_f / n), n)
        if sin_p is not None:
            patterns.append(sin_p)
        for pat in patterns:
            sc = exact_score(pat, n)
            if sc is not None and sc < incumbent_score - 1e-12:
                incumbent = pat.copy()
                incumbent_score = sc
        return (incumbent.astype(np.float64), float(incumbent_score), n)
    P = min(64, max(24, 2 * n))
    max_gen = 180 if n <= 64 else 100
    try:
        X = build_pop(n, parent, rng, P)
        fitness = np.full(P, 999.0, dtype=np.float64)
        for i in range(P):
            sc = exact_score(X[i], n)
            if sc is not None:
                fitness[i] = sc
                if sc < incumbent_score - 1e-12:
                    incumbent = X[i].copy()
                    incumbent_score = sc
        F_arr = rng.uniform(0.5, 0.8, size=P)
        CR_arr = np.full(P, 0.8, dtype=np.float64)
        stagnant = 0
        restarts = 0
        keep_count = max(2, P // 4) - 1
        for gen in range(max_gen):
            X_next = X.copy()
            fit_next = fitness.copy()
            improved = False
            for i in range(P):
                F_t = F_arr[i]
                CR_t = CR_arr[i]
                if rng.random() < 0.1:
                    F_t = 0.1 + 0.8 * rng.random()
                if rng.random() < 0.1:
                    CR_t = rng.random()
                if rng.random() < 0.75:
                    cands = [j for j in range(P) if j != i]
                    r1, r2, r3 = rng.choice(cands, size=3, replace=False)
                    mutant = X[r1] + F_t * (X[r2] - X[r3])
                else:
                    order = np.argsort(fitness)
                    pn = max(1, int(math.ceil(0.2 * P)))
                    pc = [int(order[k]) for k in range(pn) if order[k] != i]
                    if not pc:
                        pc = [int(order[0])]
                    p = int(rng.choice(pc, size=1)[0])
                    rc = [j for j in range(P) if j != i and j != p]
                    r1, r2 = rng.choice(rc, size=2, replace=False)
                    mutant = X[i] + F_t * (X[p] - X[i]) + F_t * (X[r1] - X[r2])
                mask = rng.random(n) < CR_t
                mask[int(rng.integers(0, n))] = True
                trial = np.where(mask, mutant, X[i])
                trial_p = project_capped_simplex(trial, n)
                if trial_p is None or not np.all(np.isfinite(trial_p)):
                    continue
                if np.max(np.abs(trial_p - X[i])) < 1e-14:
                    continue
                sc = exact_score(trial_p, n)
                if sc is None:
                    continue
                if sc < fitness[i] - 1e-12:
                    X_next[i] = trial_p
                    fit_next[i] = sc
                    F_arr[i] = F_t
                    CR_arr[i] = CR_t
                    if sc < incumbent_score - 1e-12:
                        incumbent = trial_p.copy()
                        incumbent_score = sc
                        improved = True
                elif abs(sc - fitness[i]) <= 1e-12 and np.max(np.abs(trial_p - X[i])) >= 1e-6:
                    if rng.random() < 0.02:
                        X_next[i] = trial_p
                        fit_next[i] = sc
                        F_arr[i] = F_t
                        CR_arr[i] = CR_t
            X = X_next
            fitness = fit_next
            if improved:
                stagnant = 0
            else:
                stagnant += 1
            if stagnant >= 30 and restarts < 3:
                sig_r = [0.20, 0.10, 0.05]
                sigma_r = sig_r[restarts]
                kc = min(keep_count, P)
                order = np.argsort(fitness)
                keep_idx = sorted(int(order[k]) for k in range(kc))
                X_keep = X[keep_idx].copy()
                fit_keep = fitness[keep_idx].copy()
                n_new = P - len(keep_idx)
                new_rows = np.zeros((n_new, n))
                for k in range(n_new):
                    if rng.random() < 0.75:
                        z = rng.standard_normal(n)
                        z -= np.mean(z)
                        cand = incumbent + sigma_r * z
                    else:
                        cand = rng.uniform(0.0, 1.0, size=n)
                    cp = project_capped_simplex(cand, n)
                    new_rows[k] = cp if cp is not None else const_h
                X = np.vstack([X_keep, new_rows])
                nf = np.full(n_new, 999.0)
                for k in range(n_new):
                    sc = exact_score(X[kc + k], n)
                    if sc is not None:
                        nf[k] = sc
                        if sc < incumbent_score - 1e-12:
                            incumbent = X[kc + k].copy()
                            incumbent_score = sc
                fitness = np.concatenate([fit_keep, nf])
                F_arr = rng.uniform(0.5, 0.8, size=P)
                CR_arr = np.full(P, 0.8, dtype=np.float64)
                stagnant = 0
                restarts += 1
            elif stagnant >= 70 and restarts >= 3:
                break
    except Exception:
        pass
    if not is_feasible(incumbent, n):
        fixed = project_capped_simplex(incumbent, n)
        if fixed is not None:
            incumbent = fixed
    final_score = exact_score(incumbent, n)
    if final_score is None:
        if parent is not None and parent_score is not None:
            return (parent.astype(np.float64), float(parent_score), n)
        incumbent = const_h
        final_score = exact_score(const_h, n)
    return (incumbent.astype(np.float64), float(final_score), n)
