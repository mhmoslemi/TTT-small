            L = N - d
            a = h[0:L]
            b = h[d:d + L]
            grad[0:L] += w * (1.0 - b)
            grad[d:d + L] -= w * a
    return phi, grad

def smooth_value(h, tau, N):
    q = lag_values(h, N)
    r = (2.0 / N) * q
    rmax = np.max(r)
    Z = np.sum(np.exp((r - rmax) / tau))
    return rmax + tau * np.log(Z)

def optimize_start(h_start, N, inc_h, inc_score, iters):
    temps = [0.08, 0.02, 0.005, 0.001, 0.0002, 0.00004, 0.000008]
    h = h_start.copy()
    for tau in temps:
        sb = h.copy()
        sb_score = inc_score
        for _ in range(iters):
            try:
                phi, g = smooth_value_grad(h, tau, N)
            except Exception:
                break
            if not np.all(np.isfinite(g)):
                break
            alpha = 1.0
            accepted = False
            for _ in range(24):
                y = project_box_sum(h - alpha * g, N)
                d = y - h
                if np.max(np.abs(d)) < 1e-10:
                    break
                gd = np.dot(g, d)
                svi = smooth_value(y, tau, N)
                si = score_exact(y, N)
                if si < inc_score:
                    inc_h = y.copy()
                    inc_score = si
                if si < sb_score:
                    sb = y.copy()
                    sb_score = si
                if gd < 0 and svi <= phi + 1e-4 * gd + 1e-14:
                    h = y.copy()
                    accepted = True
                    alpha = min(1.4 * alpha, 10.0)
                    if np.max(np.abs(d)) < 1e-8:
                        break
                    break
                alpha *= 0.5
                if alpha < 1e-8:
                    break
            if not accepted:
                h = sb.copy()
                break
        h = sb.copy()
    return inc_h, inc_score

def make_start_portfolio(baseline, N, rng):
    starts = [baseline.copy()]
    starts.append(project_box_sum(baseline + 0.20 * rng.standard_normal(N), N))
    starts.append(project_box_sum(rng.uniform(0.0, 1.0, N), N))
    R = min(6, (N - 1) // 2)
    i = np.arange(N, dtype=np.float64)
    z = np.full(N, 0.5)
    for r in range(1, R + 1):
        A = rng.standard_normal()
        B = rng.standard_normal()
        z += (0.12 / r) * (A * np.sin(2.0 * np.pi * r * i / N) + B * np.cos(2.0 * np.pi * r * i / N))
    starts.append(project_box_sum(z, N))
    starts.append(project_box_sum(1.0 - baseline, N))
    starts.append(np.full(N, 0.5))
    starts.append(project_box_sum(baseline + 0.08 * rng.standard_normal(N), N))
    starts.append(project_box_sum(baseline + 0.40 * rng.standard_normal(N), N))
    return starts

def run(seed=42, **kwargs):
    N = 70
    baseline = None
    try:
        parent = initial_h_values
        if parent is not None:
            parent = np.asarray(parent, dtype=np.float64)
            if parent.ndim == 1 and len(parent) > 0:
                N = int(len(parent))
                if np.all(np.isfinite(parent)):
                    baseline = project_box_sum(parent, N)
                else:
                    baseline = np.full(N, 0.5)
    except Exception:
        pass
    if baseline is None:
        if 'n_points' in kwargs:
            try:
                N = int(kwargs['n_points'])
            except (ValueError, TypeError):
                N = 70
            if N < 1:
                N = 70
        baseline = np.full(N, 0.5)
    if N == 1:
        h = np.array([0.5])
        s = score_exact(h, 1)
        return h, s, 1
    seed = int(seed) if seed is not None else 42
    rng = np.random.default_rng(seed)
    inc_h = baseline.copy()
    inc_score = score_exact(inc_h, N)
    if N <= 100:
        n_starts = 8
        iters = 60
    elif N <= 200:
        n_starts = 6
        iters = 40
    elif N <= 500:
        n_starts = 4
        iters = 20
    else:
        n_starts = 3
        iters = 8
    starts = make_start_portfolio(baseline, N, rng)
    starts = starts[:n_starts]
    for st in starts:
        st = project_box_sum(st, N)
        ss = score_exact(st, N)
        if ss < inc_score:
            inc_h = st.copy()
            inc_score = ss
        try:
            inc_h, inc_score = optimize_start(st, N, inc_h, inc_score, iters)
        except Exception:
            pass
    inc_h = project_box_sum(inc_h, N)
    final_score = score_exact(inc_h, N)
    if not np.isfinite(final_score):
        inc_h = np.full(N, 0.5)
        final_score = score_exact(inc_h, N)
    inc_h = np.clip(inc_h, 0.0, 1.0)
    if abs(inc_h.sum() - N / 2.0) > 1e-9:
        inc_h = project_box_sum(inc_h, N)
        final_score = score_exact(inc_h, N)
    return inc_h, float(final_score), int(N)
