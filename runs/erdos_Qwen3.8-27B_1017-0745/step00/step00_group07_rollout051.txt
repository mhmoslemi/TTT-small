import numpy as np
from scipy.optimize import minimize
import math


def _is_feasible(h, n):
    if h is None or not isinstance(h, np.ndarray) or h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(float(np.sum(h)) - n / 2.0) <= 1e-10


def _project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    n = len(z)
    if n == 0 or not np.all(np.isfinite(z)) or np.max(np.abs(z)) > 1e12:
        return None
    S = float(S)
    if S < 0 or S > n:
        return None
    if np.max(z) < -1.0:
        return None
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = lo + 0.5 * (hi - lo)
        y = np.clip(z - mid, 0.0, 1.0)
        s = float(np.sum(y))
        if s > S:
            lo = mid
        else:
            hi = mid
    mid = lo + 0.5 * (hi - lo)
    y = np.clip(z - mid, 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 0:
        for i in range(n):
            if r <= 0:
                break
            add = min(r, 1.0 - y[i])
            if add > 0:
                y[i] += add
                r -= add
    elif r < 0:
        for i in range(n):
            if r >= 0:
                break
            sub = min(-r, y[i])
            if sub > 0:
                y[i] -= sub
                r += sub
    r = S - float(np.sum(y))
    if abs(r) > 1e-12:
        for i in range(n):
            if abs(r) < 1e-14:
                break
            if r > 0 and y[i] < 1.0:
                y[i] = min(1.0, y[i] + r)
                r = 0.0
            elif r < 0 and y[i] > 0.0:
                y[i] = max(0.0, y[i] + r)
                r = 0.0
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    if abs(float(np.sum(y)) - S) > 1e-10:
        return None
    return np.clip(y, 0.0, 1.0)


def _lag_sums_and_grads(h):
    n = len(h)
    B = np.zeros(2 * n - 1, dtype=np.float64)
    G = np.zeros((2 * n - 1, n), dtype=np.float64)
    for idx, d in enumerate(range(-(n - 1), n)):
        lo = max(0, -d)
        hi = min(n, n - d)
        if lo >= hi:
            continue
        src = np.arange(lo, hi)
        dst = np.arange(lo + d, hi + d)
        h_dst = h[dst]
        B[idx] = float(np.dot(h[src], 1.0 - h_dst))
        G[idx, src] += (1.0 - h_dst)
        G[idx, dst] -= h[src]
    return B, G


def _exact_score(h):
    n = len(h)
    if n == 0 or not np.all(np.isfinite(h)):
        return None
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    if not np.all(np.isfinite(vals)):
        return None
    return float(np.max(vals))


def _obj(z):
    return z[-1]


def _obj_grad(z):
    n = len(z) - 1
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return g


def _ineq(z):
    n = len(z) - 1
    h = z[:n]
    t = z[-1]
    B, _ = _lag_sums_and_grads(h)
    return t - B


def _ineq_jac(z):
    n = len(z) - 1
    h = z[:n]
    _, G = _lag_sums_and_grads(h)
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J


def _eq(z):
    n = len(z) - 1
    return float(np.sum(z[:n]) - n / 2.0)


def _eq_jac(z):
    n = len(z) - 1
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g


def _make_starts(parent, n, rng):
    starts = []
    S = n / 2.0

    def _try_add(z):
        if z is None or len(z) != n:
            return
        for s in starts:
            if np.max(np.abs(s - z)) < 1e-10:
                return
        starts.append(z)

    if parent is not None:
        proj = _project_capped_simplex(parent, S)
        if proj is not None:
            _try_add(proj)
        rev = np.asarray(parent, dtype=np.float64)[::-1].copy()
        projr = _project_capped_simplex(rev, S)
        if projr is not None:
            _try_add(projr)
        pert = parent + rng.normal(0, 0.18, n)
        projp = _project_capped_simplex(pert, S)
        if projp is not None:
            _try_add(projp)

    const = np.full(n, 0.5, dtype=np.float64)
    _try_add(const)

    max_q = max(1, min(6, (n - 1) // 2))
    q = int(rng.integers(1, max_q + 1))
    phi = rng.uniform(0, 2 * math.pi)
    ii = np.arange(n, dtype=np.float64)
    peri = 0.5 + 0.45 * np.cos(2.0 * math.pi * q * ii / n + phi)
    projp = _project_capped_simplex(peri, S)
    if projp is not None:
        _try_add(projp)

    perm = rng.permutation(n)
    binv = np.zeros(n, dtype=np.float64)
    k = n // 2
    binv[perm[:k]] = 1.0
    if n % 2 == 1:
        binv[perm[k]] = 0.5
    binv = _project_capped_simplex(binv, S)
    _try_add(binv)

    attempts = 0
    while len(starts) < 6 and attempts < 18:
        attempts += 1
        proposal = rng.normal(0.5, 0.25, n)
        proj = _project_capped_simplex(proposal, S)
        _try_add(proj)
    return starts

    best_score = _exact_score(best_h)
    if best_score is None or not math.isfinite(best_score):

    starts = _make_starts(parent, n, rng)

        if sc is not None and sc < best_score:

    if n <= 128:

                z0 = np.concatenate([s, [t0]])
                ineq = {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac}
                eq = {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}
                res = minimize(
