                    next_pop[i] = snap[i]; next_s[i] = snap_s[i]
                    continue
                u = project_capped_simplex(z, S, n)
                if u is None:
                    next_pop[i] = snap[i]; next_s[i] = snap_s[i]
                    continue
                s_u = score_c5(u)
                best_p = u
                best_ps = s_u if s_u is not None else float('inf')
                if n <= 128 and s_u is not None:
                    for alpha in (0.25, 0.50, 0.75):
                        seg = snap[i] + alpha * (u - snap[i])
                        sp = project_capped_simplex(seg, S, n)
                        if sp is not None:
                            ss = score_c5(sp)
                            if ss is not None and ss < best_ps:
                                best_ps = ss; best_p = sp
                if best_ps < inc_s:
                    incumbent = best_p.copy(); inc_s = best_ps
                if s_u is None:
                    next_pop[i] = snap[i]; next_s[i] = snap_s[i]
                elif best_ps < snap_s[i]:
                    next_pop[i] = best_p; next_s[i] = best_ps
                else:
                    next_pop[i] = snap[i]; next_s[i] = snap_s[i]

            pop = next_pop
            pop_s = next_s
            tol = 1e-13 * max(1.0, abs(gen_best))
            if inc_s < gen_best - tol:
                stag = 0; stall = 0
            else:
                stag += 1; stall += 1

            if stag >= 18 and restarts < 2:
                restarts += 1
                order = sorted(range(M), key=lambda k: pop_s[k])
                en = max(2, M // 4)
                new_pop = [pop[order[k]].copy() for k in range(en)]
                new_s = [pop_s[order[k]] for k in range(en)]
                sr = si_global
                for _ in range(M - en):
                    z_r = gen_candidates(n, incumbent, rng, sr)
                    sr += 1
                    u_r = project_capped_simplex(z_r, S, n)
                    if u_r is not None:
                        s_r = score_c5(u_r)
                    else:
                        s_r = None
                    if s_r is not None:
                        new_pop.append(u_r); new_s.append(s_r)
                        if s_r < inc_s:
                            inc_s = s_r; incumbent = u_r.copy()
                    else:
                        new_pop.append(const_fb.copy())
                        new_s.append(score_c5(const_fb))
                si_global = sr
                pop = new_pop[:M]; pop_s = new_s[:M]
                stag = 0; stall = 0

            if restarts >= 2 and stall >= 24:
                break

    pool = [incumbent, const_fb]
    if parent_fb is not None:
        pool.append(parent_fb)
    best_h = None
    best_final = float('inf')
    for c in pool:
        if c is None:
            continue
        cc = np.clip(c, 0.0, 1.0).astype(np.float64)
        pc = project_capped_simplex(cc, S, n)
        if pc is not None:
            s = score_c5(pc)
            if s is not None and s < best_final:
                best_final = s; best_h = pc.copy()
    if best_h is None:
        best_h = const_fb; best_final = score_c5(const_fb)
    return best_h.astype(np.float64), float(best_final), n
