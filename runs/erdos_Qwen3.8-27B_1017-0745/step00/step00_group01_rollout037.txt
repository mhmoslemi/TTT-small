        gb_start = gbs
        F = rng.uniform(0.45, 0.90)
        CR = 0.90 - 0.25 * g / max(1, G - 1)
        Xold = [p.copy() for p in pop]
        Xs = list(ps)
        npp = [p.copy() for p in pop]
        nps = list(ps)
        for i in range(len(pop)):
            try:
                ot = [j for j in range(len(pop)) if j != i]
                abc = rng.choice(ot, 3, replace=False)
                a, b, c = int(abc[0]), int(abc[1]), int(abc[2])
                dr = Xold[a] + F * (Xold[b] - Xold[c])
                jr = int(rng.integers(0, n))
                mk = rng.random(n) < CR
                mk[jr] = True
                w = np.where(mk, dr, Xold[i])
                tr = project_capped(w, n)
                if tr is None:
                    continue
                ts = exact_score(tr, n)
                if ts < Xs[i] - 1e-12:
                    npp[i], nps[i] = tr, ts
                    if ts < gbs - 1e-12:
                        gb, gbs = tr.copy(), ts
            except Exception:
                continue
        pop, ps = npp, nps
        if gbs < gb_start - 1e-12:
            ni = 0
        else:
            ni += 1
            if ni >= pat and rs < 2:
                idx = np.argsort(ps)
                half = len(pop) // 2
                keep = [pop[idx[j]].copy() for j in range(half)]
                sc_rs = 0.10 if rs == 0 else 0.30
                for k in range(len(pop) - half):
                    if k % 3 == 2:
                        prop = rng.random(n).astype(np.float64)
                    else:
                        prop = gb + sc_rs * rng.standard_normal(n)
                    pr = project_capped(prop, n)
                    if pr is None:
                        pr = np.full(n, 0.5, dtype=np.float64)
                    keep.append(pr)
                pop = keep
                ps = [exact_score(p, n) for p in pop]
                for i in range(len(pop)):
                    if ps[i] < gbs - 1e-12:
                        gb, gbs = pop[i].copy(), ps[i]
                ni, rs = 0, rs + 1
            elif ni >= pat:
                break

    pr = project_capped(gb, n)
    if pr is not None:
        gb = pr
    fs = exact_score(gb, n)
    fbs = exact_score(fb, n)
    if fbs <= fs:
        gb, fs = fb.copy(), fbs
    if fs == float('inf'):
        gb = np.full(n, 0.5, dtype=np.float64)
        fs = exact_score(gb, n)
    return gb, float(fs), n
