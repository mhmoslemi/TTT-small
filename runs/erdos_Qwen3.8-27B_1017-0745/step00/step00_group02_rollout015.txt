def _score(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def _project(y, n):
    S = n / 2.0
    if not np.all(np.isfinite(y)):
        return None
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if hi - lo <= 1e-14 * max(1.0, abs(mid)):
            break
        if np.clip(y - mid, 0, 1).sum() > S:
            hi = mid
        else:
            lo = mid
    x = np.clip(y - 0.5 * (lo + hi), 0, 1)
    r = S - x.sum()
    if r > 1e-13:
        for i in range(n):
            if abs(r) < 1e-13: break
            a = min(r, 1.0 - x[i]); x[i] += a; r -= a
    elif r < -1e-13:
        for i in range(n):
            if abs(r) < 1e-13: break
            a = min(-r, x[i]); x[i] -= a; r += a
    if abs(x.sum() - S) > 1e-10:
        return None
    return x

def _repair(h, n):

        return None

def _preG(h, n, dx):
    G = np.zeros((n, 2 * n - 1))

        for ell in range(-(n - 1), n):

            if 0 <= pp < n: v += 1.0 - h[pp]
            pp = p + ell
            if 0 <= pp < n: v -= h[pp]
            G[p, k] = dx * v
    return G
