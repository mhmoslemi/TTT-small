import math

_S = {}

def exact_score(h):
    h = np.asarray(h, dtype=np.float64).flatten()
    if h.size < 2 or not np.all(np.isfinite(h)):
        return 1e6
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / h.size))

def project(x, N, m):
    x = np.asarray(x, dtype=np.float64).flatten()
    if x.size != N or not np.all(np.isfinite(x)) or np.max(np.abs(x)) > 1e6:
        return np.full(N, 0.5)
    lo, hi = x.min() - 1.0, x.max()
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > m:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    delta = m - y.sum()
    idx = np.arange(N)
    if delta > 1e-15:
        for i in idx:
            a = min(delta, 1.0 - y[i])
            y[i] += a; delta -= a
            if delta <= 1e-15: break
    elif delta < -1e-15:
        for i in idx:
            a = min(-delta, y[i])
            y[i] -= a; delta += a
            if delta >= -1e-15: break
    y = np.clip(y, 0.0, 1.0)
    return y

def _R_J(h, N, d):
    R = np.correlate(h, 1.0 - h, mode='full') * d
    J = np.zeros((2 * N - 1, N))
    for q in range(-N + 1, N):

        for j in range(a, b):

import numpy as np
from scipy.optimize import minimize
import math

_S = {}

def exact_score(h):
    h = np.asarray(h, dtype=np.float64).flatten()
    if h.size < 2 or not np.all(np.isfinite(h)):
        return 1e6
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / h.size))

def project(x, N, m):
    x = np.asarray(x, dtype=np.float64).flatten()
    if x.size != N or not np.all(np.isfinite(x)) or np.max(np.abs(x)) > 1e6:
        return np.full(N, 0.5)
    lo, hi = x.min() - 1.0, x.max()
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > m:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    delta = m - y.sum()
    if delta > 1e-15:
        for i in range(N):
            a = min(delta, 1.0 - y[i])
            y[i] += a; delta -= a
            if delta <= 1e-15:
                break
    elif delta < -1e-15:
        for i in range(N):
            a = min(-delta, y[i])
            y[i] -= a; delta += a
            if delta >= -1e-15:
                break
    y = np.clip(y, 0.0, 1.0)
    return y

def _R_J(h, N, d):
    R = np.correlate(h, 1.0 - h, mode='full') * d
    J = np.zeros((2 * N - 1, N))
    for q in range(-N + 1, N):
        r = q + N - 1
        a, b = max(0, -q), min(N, N - q)
        for j in range(a, b):
            J[r, j + q] += d * (1.0 - h[j])
            J[r, j] -= d * h[j + q]
    return R, J

def _sf(x):
    N, d, tau = _S['N'], _S['d'], _S['tau']
    R, J = _R_J(x, N, d)
    a = R.max()
    ex = np.exp((R - a) / tau)
    s = ex.sum()
    val = a + tau * math.log(s)
    grad = J.T @ (ex / s)
    return val, grad

def _sobj(x):
    v, _ = _sf(x)
    return v

def _sjac(x):
    _, g = _sf(x)
    return g

def _meq(x):
    return np.sum(x) - _S['m']

def _meq_jac(x):
    return np.ones(_S['N'])

def _eobj(z):
    return z[-1]

def _eobj_jac(z):
    g = np.zeros(_S['N'] + 1)
    g[-1] = 1.0
    return g

def _epeq(z):
    return z[:-1].sum() - _S['m']

def _epeq_jac(z):
    g = np.zeros(_S['N'] + 1)
    g[:-1] = 1.0
    return g

def _epineq(z):
    N, d = _S['N'], _S['d']
    R, _ = _R_J(z[:-1], N, d)
    return z[-1] - R

def _epineq_jac(z):
    N, d = _S['N'], _S['d']
    _, J = _R_J(z[:-1], N, d)
    out = np.zeros((2 * N - 1, N + 1))
    out[:, :N] = -J
    out[:, N] = 1.0
    return out

    h_init = None
    try:
        h_raw = np.array(initial_h_values, dtype=np.float64).flatten()
        if h_raw.size >= 2 and np.all(np.isfinite(h_raw)):

    except Exception:
        h_init = None

    if h_init is not None:

    else:

            N = 95
        N = max(2, min(999, N))

    m = N / 2.0
    d = 2.0 / N
    _S['N'] = N
    _S['d'] = d
    _S['m'] = m

    if h_init is not None:
        h_cur = project(h_init, N, m)
    else:

    best_h = h_cur.copy()
    best_score = exact_score(h_cur)

    try:

    except Exception:
        rng = np.random.default_rng(42)

    hf = int(np.floor(m))

    rem = m - hf
