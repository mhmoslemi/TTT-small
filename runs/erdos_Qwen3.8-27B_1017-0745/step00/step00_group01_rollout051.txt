def capped_simplex_project(z, n, s):
    """Project z onto {0<=h<=1, sum(h)=s}."""
    if z is None or len(z) != n or not np.all(np.isfinite(z)):
        return np.full(n, 0.5)
    z = np.clip(z.astype(np.float64), -1e12, 1e12)

            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    # Residual correction
    r = s - np.sum(h)
    for _ in range(n + 2):

            idx = np.argmax(headroom)

            r -= move

            idx = np.argmax(slack)

            r += move
    h = np.clip(h, 0.0, 1.0)
