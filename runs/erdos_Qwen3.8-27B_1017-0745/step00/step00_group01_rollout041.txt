    s = n / 2.0
    tol_mass = 1e-10 * max(1, n)
    if z.shape != (n,) or not np.all(np.isfinite(z)) or np.any(np.abs(z) > 1e12):
        return None
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(z - mid, 0.0, 1.0)
        if np.sum(h) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(n + 2):
        r = s - np.sum(h)
        if abs(r) < tol_mass:
            break
        if r > 0:
            hr = 1.0 - h
            idx = int(np.argmax(hr))
            step = min(r, hr[idx])
        else:
            idx = int(np.argmax(h))
            step = min(-r, h[idx])
        if step <= 1e-15:
            break
        h[idx] += step if r > 0 else -step
    h = np.clip(h, 0.0, 1.0)
    return h if check_feasible(h, n, tol_mass) else None

def exact_score(h, n):
    tol_mass = 1e-10 * max(1, n)
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return float('inf')
    if abs(np.sum(h) - n / 2.0) > tol_mass:
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full')
    if q.shape != (2 * n - 1,) or not np.all(np.isfinite(q)):
        return float('inf')
    return float(np.max(q) * (2.0 / n))

def build_haar_basis(n):
    if n <= 1:
        return np.empty((0, n), dtype=np.float64)
    dirs = []
    stack = [(0, n)]
    while stack:
        a, b = stack.pop()
        m = b - a
        if m < 2:
            continue
        mid = a + m // 2
        mL, mR = mid - a, b - mid
        v = np.zeros(n, dtype=np.float64)
        v[a:mid] = math.sqrt(float(mR) / (m * mL))
        v[mid:b] = -math.sqrt(float(mL) / (m * mR))
        if np.all(np.isfinite(v)) and abs(np.sum(v)) < 1e-8:
            dirs.append(v)
        if mL >= 2:
            stack.append((a, mid))
        if mR >= 2:
            stack.append((mid, b))
    return np.array(dirs, dtype=np.float64)

def feasible_interval(x, v):
    lo, hi = float('-inf'), float('inf')
    for i in range(len(x)):
        vi = v[i]
        if vi == 0.0:
            continue
        r0, r1 = -x[i] / vi, (1.0 - x[i]) / vi
        if r0 > r1:
            r0, r1 = r1, r0
        lo, hi = max(lo, r0), min(hi, r1)
    if not (np.isfinite(lo) and np.isfinite(hi)) or lo > hi:
        return None
    if lo > 1e-10 or hi < -1e-10:
        return None
    lo, hi = max(lo, -1e10), min(hi, 1e10)
    if hi - lo < 1e-14:
        return None
    return (lo, hi)

def line_coeffs(x, v, n):
    dx = 2.0 / n
    c = dx * np.correlate(x, 1.0 - x, mode='full')
    b = dx * (np.correlate(v, 1.0 - x, mode='full') - np.correlate(x, v, mode='full'))
    a = -dx * np.correlate(v, v, mode='full')
    L = 2 * n - 1
    if c.shape != (L,) or not (np.all(np.isfinite(a)) and np.all(np.isfinite(b)) and np.all(np.isfinite(c))):
        return None
    return (a, b, c)

def line_candidates(a, b, c, lo, hi, pairs):
    tol_t = 1e-12 * max(1.0, abs(lo), abs(hi), abs(hi - lo))
    cands = [lo, hi, 0.0]
    L = len(a)
    for (i, j) in pairs:
        ad, bd, cd = a[i] - a[j], b[i] - b[j], c[i] - c[j]
        scale = max(1.0, abs(ad), abs(bd), abs(cd))
        if abs(ad) > 1e-14 * scale:
            disc = bd * bd - 4.0 * ad * cd
            tol_d = 1e-14 * max(1.0, bd * bd, abs(4.0 * ad * cd))
            if disc < -tol_d:
                continue
            disc = max(0.0, disc)
            sq = math.sqrt(disc)
            qr = -0.5 * (bd + sq if bd >= 0 else bd - sq)
            if abs(qr) > 1e-30:
                t1, t2 = qr / ad, cd / qr
            else:
                t1 = t2 = -bd / (2.0 * ad)
            for t in (t1, t2):
                if np.isfinite(t) and lo - tol_t <= t <= hi + tol_t:
                    cands.append(t)
        elif abs(bd) > 1e-14 * scale:
            t = -cd / bd
            if np.isfinite(t) and lo - tol_t <= t <= hi + tol_t:
                cands.append(t)
    for d in range(L):
        if abs(a[d]) > 1e-14:
            t = -b[d] / (2.0 * a[d])
            if np.isfinite(t) and lo - tol_t <= t <= hi + tol_t:
                cands.append(t)
    cands = [max(lo, min(hi, t)) for t in cands]
    cands.sort()
    merged = [cands[0]]
    for t in cands[1:]:
        if t - merged[-1] > tol_t:
            merged.append(t)
    return merged

def do_line_search(x, v, n, pairs, cur_score):
    fi = feasible_interval(x, v)
    if fi is None:
        return None
    lo, hi = fi
    lc = line_coeffs(x, v, n)
    if lc is None:
        return None
    a, b, c = lc
    cands = line_candidates(a, b, c, lo, hi, pairs)
    if not cands:
        return None
    tol_t = 1e-12 * max(1.0, abs(lo), abs(hi), abs(hi - lo))
    tol_mass = 1e-10 * max(1, n)
    store = [(float(np.max(c)), 0.0)]
    chunk = 512
    for k in range(0, len(cands), chunk):
        tc = np.array(cands[k:k + chunk], dtype=np.float64)
        mat = a[None, :] * (tc * tc)[:, None] + b[None, :] * tc[:, None] + c[None, :]
        fv = np.all(np.isfinite(mat), axis=1)
        if not np.all(fv):
            tc, mat = tc[fv], mat[fv]
            if tc.size == 0:
                continue
        vals = np.max(mat, axis=1)
        for m in range(len(vals)):
            if np.isfinite(vals[m]):
                store.append((float(vals[m]), float(tc[m])))
    store.sort(key=lambda z: z[0])
    top = [0.0]
    for val, t in store:
        if len(top) >= 9:
            break
        is_dup = any(abs(t - tt) < tol_t for tt in top)
        if not is_dup:
            top.append(t)
    best_y, best_sc = None, cur_score
    for t in top:
        if abs(t) < tol_t:
            continue
        y = x + t * v
        if not check_feasible(y, n, tol_mass):
            y = project_simplex(y, n)
            if y is None:
                continue
        sc = exact_score(y, n)
        if sc < best_sc:
            best_sc, best_y = sc, y
    return best_y

def run(seed=42, **kwargs):
    n, base = 47, None
    try:
        parent = np.array(initial_h_values, dtype=np.float64).ravel()
        if parent.size > 0 and np.all(np.isfinite(parent)):
            n = parent.size
            p = np.clip(parent, 0.0, 1.0)
            base = project_simplex(p, n)
            if base is None:
                base = p.copy() if abs(np.sum(p) - n / 2.0) <= 1e-8 * n else uniform_vec(n)
    except Exception:
        pass
    if base is None:
        try:
            nv = kwargs.get('n_points', None)
            if nv is not None and isinstance(nv, (int, np.integer)) and not isinstance(nv, bool) and 1 <= int(nv) <= 999:
                n = int(nv)
        except Exception:
            pass
        base = uniform_vec(n)
    if n <= 1:
        h = np.array([0.5], dtype=np.float64)
        return (h, 0.0, 1)
    tol_mass = 1e-10 * max(1, n)
    if not check_feasible(base, n, tol_mass):
        pr = project_simplex(base, n)
        base = pr if pr is not None else uniform_vec(n)
    fallback = base.copy()
    fb_score = exact_score(fallback, n)
    try:
        if isinstance(seed, (int, np.integer)) and not isinstance(seed, bool):
            seed = int(seed)
        else:
            seed = 42
    except Exception:
        seed = 42
    n_start = 5 if n <= 48 else (2 if n <= 64 else 1)
    n_sweep = 2 if n <= 48 else (1 if n <= 64 else 0)
    try:
        sv = kwargs.get('starts', None)
        if sv is not None and isinstance(sv, (int, np.integer)) and not isinstance(sv, bool) and sv > 0:
            n_start = min(int(sv), 6 if n <= 48 else (3 if n <= 64 else 1))
    except Exception:
        pass
    try:
        sw2 = kwargs.get('sweeps', None)
        if sw2 is not None and isinstance(sw2, (int, np.integer)) and not isinstance(sw2, bool) and sw2 > 0:
            n_sweep = min(int(sw2), 3 if n <= 48 else (2 if n <= 64 else 0))
    except Exception:
        pass
    rng = np.random.RandomState(seed)
    pool = [base.copy()]
    proposals = [base[::-1].copy(), 1.0 - base.copy(), base + rng.normal(0, 0.12, n), uniform_vec(n)]
    blk = np.zeros(n, dtype=np.float64); blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    proposals.append(blk)
    alt = np.zeros(n, dtype=np.float64); alt[0::2] = 1.0
    if n % 2 == 1:
        alt[-1] = 0.5
    proposals.extend([alt, base + rng.normal(0, 0.35, n), base + rng.normal(0, 0.45, n), rng.uniform(0, 1, n)])
    for rp in proposals:
        if len(pool) >= n_start:
            break
        pr = project_simplex(rp, n)
        if pr is None:
            continue
        if not any(np.max(np.abs(pr - p)) < 1e-10 for p in pool):
            pool.append(pr)
    gi = 0
    while len(pool) < n_start and gi < 30:
        gi += 1
        pr = project_simplex(base + rng.normal(0, 0.15 + 0.01 * gi, n), n)
        if pr is not None and not any(np.max(np.abs(pr - p)) < 1e-10 for p in pool):
            pool.append(pr)
    while len(pool) < n_start:
        pool.append(base.copy())
    incumbent, inc_sc = fallback.copy(), fb_score
    for st in pool:
        sc = exact_score(st, n)
        if sc < inc_sc - 1e-12:
            incumbent, inc_sc = st.copy(), sc
    if n <= 64 and n_sweep > 0:
        basis = build_haar_basis(n)
        L = 2 * n - 1
        pairs = [(i, j) for i in range(L) for j in range(i + 1, L)]
        for si in range(len(pool)):
            x, ls = pool[si].copy(), exact_score(pool[si], n)
            for _sw in range(n_sweep):
                improved = False
                for d_idx in rng.permutation(len(basis)):
                    try:
                        res = do_line_search(x, basis[d_idx], n, pairs, ls)
                        if res is not None:
                            sc = exact_score(res, n)
                            if sc < ls - 1e-12:
                                x, ls, improved = res, sc, True
                                if sc < inc_sc - 1e-12:
                                    incumbent, inc_sc = res.copy(), sc
                    except Exception:
                        pass
                if not improved:
                    break
    final = project_simplex(incumbent, n)
    if final is None:
        final = incumbent
    if not check_feasible(final, n, tol_mass):
        final = fallback.copy()
    fsc = exact_score(final, n)
    if fsc > fb_score + 1e-12:
        final, fsc = fallback.copy(), fb_score
    if not (np.all(np.isfinite(final)) and final.shape == (n,)):
        final, fsc = uniform_vec(n), exact_score(uniform_vec(n), n)
    return (final, float(fsc), int(n))
