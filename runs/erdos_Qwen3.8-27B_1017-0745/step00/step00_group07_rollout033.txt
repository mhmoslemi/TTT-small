            y[i] += add
            resid -= add
    elif resid < -1e-14:
        for i in range(n):
            if resid >= 0:
                break
            sub = min(-resid, y[i])
            y[i] -= sub
            resid += sub
    if not np.all(np.isfinite(y)) or np.any(y < 0) or np.any(y > 1):
        return None
    if abs(np.sum(y) - S) > 1e-10:
        return None
    return y

def exact_score(h):
    n = len(h)
    if not is_feasible(h, n, n * 0.5):
        return None
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    if not np.all(np.isfinite(vals)):
        return None
    return float(np.max(vals))

def make_candidate(n, base, rng, kind):
    if kind == 0:
        sig = [0.05, 0.15, 0.35, 0.60][int(rng.integers(4))]
        return base + sig * rng.standard_normal(n)
    if kind == 1:
        q = int(rng.integers(1, max(2, min(8, (n - 1) // 2) + 1)))
        phi = rng.uniform(0, 2 * math.pi)
        return 0.5 + 0.45 * np.cos(2 * math.pi * q * np.arange(n) / n + phi)
    if kind == 2:
        return rng.uniform(0, 1, n)
    if kind == 3:
        return 0.5 + 0.5 * rng.standard_normal(n)
    z = np.zeros(n)
    perm = rng.permutation(n)
    z[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        z[perm[n // 2]] = 0.5
    return z

def run(seed=42, **kwargs):
    n = None
    parent = None
    iva = globals().get('initial_h_values', None)
    if iva is not None:
        try:
            arr = np.asarray(iva, dtype=np.float64)
            if arr.ndim == 1 and arr.size >= 2:
                n = int(arr.size)
                parent = arr.copy()
        except Exception:
            pass
    if n is None:
        n = 80
    if n < 8:
        n = 8
    S = n * 0.5

    h_const = np.full(n, 0.5)
    score_const = exact_score(h_const)
    if score_const is None:
        score_const = 0.5

    h_parent = None
    score_parent = None
    if parent is not None and len(parent) == n and np.all(np.isfinite(parent)):
        if abs(np.sum(parent) - S) <= 1e-10 and np.all(parent >= 0) and np.all(parent <= 1):
            h_parent = parent.copy()
        else:
            pr = project_simplex(parent, S)
            if pr is not None:
                h_parent = pr
        if h_parent is not None:
            score_parent = exact_score(h_parent)

    h_best = h_const.copy()
    c5_best = score_const
    if h_parent is not None and score_parent is not None and score_parent < c5_best:
        h_best = h_parent.copy()
        c5_best = score_parent

    if n > 512:
        return (h_best, c5_best, n)

    try:
        s = int(seed)
    except Exception:
        s = 42
    rng = np.random.default_rng(s)

    if n <= 128:
        M, G = 32, 100
    elif n <= 256:
        M, G = 24, 60
    else:
        M, G = 16, 30

    base = h_parent if h_parent is not None else h_const
    pop = [None] * M
    psc = [1.0] * M

    pop[0] = base.copy()
    sc = exact_score(pop[0])
    psc[0] = sc if sc is not None else 1.0
    if sc is not None and sc < c5_best:
        c5_best = sc
        h_best = pop[0].copy()

    idx = 1
    rev = base[::-1].copy()
    sc_r = exact_score(rev)
    if sc_r is not None and np.max(np.abs(rev - pop[0])) > 1e-10:
        pop[1] = rev
        psc[1] = sc_r
        if sc_r < c5_best:
            c5_best = sc_r
            h_best = rev.copy()
        idx = 2

    kind = idx
    att = 0
    while idx < M and att < 10 * M:
        att += 1
        z = make_candidate(n, base, rng, kind % 5)
        kind += 1
        if not np.all(np.isfinite(z)):
            continue
        proj = project_simplex(z, S)
        if proj is None or not is_feasible(proj, n, S):
            continue
        skip = False
        for j in range(idx):
            if np.max(np.abs(proj - pop[j])) < 1e-10:
                skip = True
                break
        if skip:
            continue
        sc = exact_score(proj)
        if sc is None:
            continue
        pop[idx] = proj
        psc[idx] = sc
        if sc < c5_best:
            c5_best = sc
            h_best = proj.copy()
        idx += 1

    for j in range(idx, M):
        pop[j] = base.copy()
        psc[j] = psc[0]

    # Count distinct
    dcnt = 0
    for i in range(M):
        d = True
        for j in range(i):
            if np.max(np.abs(pop[i] - pop[j])) < 1e-10:
                d = False
                break
        if d:
            dcnt += 1
    if dcnt < 4:
        return (h_best, c5_best, n)

    stagnant = 0
    restarts = 0
    search_done = False

    for gen in range(G):
        if search_done:
            break
        gen_start_best = c5_best
        tol = 1e-13 * max(1.0, abs(gen_start_best))
        snap = [pop[i].copy() for i in range(M)]
        ssc = psc[:]

        for i in range(M):
            others = [j for j in range(M) if j != i]
            r1, r2, r3 = int(rng.integers(len(others))), int(rng.integers(len(others))), int(rng.integers(len(others)))
            # Ensure distinct
            r1 = others[r1 % len(others)]
            r2 = others[r2 % len(others)]
            while r2 == r1:
                r2 = others[int(rng.integers(len(others)))]
            r3 = others[r3 % len(others)]
            while r3 == r1 or r3 == r2:
                r3 = others[int(rng.integers(len(others)))]

            F = rng.uniform(0.45, 0.90)
            CR = rng.uniform(0.55, 0.95)
            v = snap[r1] + F * (snap[r2] - snap[r3])
            mask = rng.random(n) < CR
            mask[int(rng.integers(n))] = True
            z = np.where(mask, v, snap[i])
            if not np.all(np.isfinite(z)):
                continue
            u = project_simplex(z, S)
            if u is None or not is_feasible(u, n, S):
                continue
            sc_u = exact_score(u)
            if sc_u is None:
                continue
            best_sc = sc_u
            best_h = u
            if n <= 128:
                for alpha in (0.25, 0.50, 0.75):
                    seg = snap[i] + alpha * (u - snap[i])
                    sp = project_simplex(seg, S)
                    if sp is None or not is_feasible(sp, n, S):
                        continue
                    sc_s = exact_score(sp)
                    if sc_s is not None and sc_s < best_sc:
                        best_sc = sc_s
                        best_h = sp
            if best_sc < c5_best:
                c5_best = best_sc
                h_best = best_h.copy()
            if best_sc < ssc[i] - 1e-15:
                pop[i] = best_h
                psc[i] = best_sc

        if c5_best < gen_start_best - tol:
            stagnant = 0
        else:
            stagnant += 1

        if stagnant >= 18 and restarts < 2:
            restarts += 1
            stagnant = 0
            order = np.argsort(psc)
            n_elite = max(2, M // 4)
            elite_set = set(int(order[k]) for k in range(n_elite))
            kr = 0
            for j in range(M):
                if j in elite_set:
                    continue
                z = make_candidate(n, h_best, rng, kr % 5)
                kr += 1
                if not np.all(np.isfinite(z)):
                    pop[j] = h_best.copy()
                    psc[j] = c5_best
                    continue
                proj = project_simplex(z, S)
                if proj is None or not is_feasible(proj, n, S):
                    pop[j] = h_best.copy()
                    psc[j] = c5_best
                    continue
                sc = exact_score(proj)
                if sc is None:
                    pop[j] = h_best.copy()
                    psc[j] = c5_best
                    continue
                pop[j] = proj
                psc[j] = sc
                if sc < c5_best:
                    c5_best = sc
                    h_best = proj.copy()
        elif stagnant >= 42 and restarts >= 2:
            search_done = True

    # Final selection
    final_h = h_const
    final_sc = score_const
    pool_h = [h_best, h_const]
    if h_parent is not None:
        pool_h.append(h_parent)
    for cand in pool_h:
        if cand is None:
            continue
        p = project_simplex(cand, S)
        if p is None:
            continue
        sc = exact_score(p)
        if sc is not None and sc < final_sc:
            final_sc = sc
            final_h = p

    return (final_h, final_sc, n)
