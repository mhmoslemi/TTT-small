def project_box_sum(z, N):
    m = N / 2.0
    if not np.all(np.isfinite(z)) or np.max(np.abs(z)) > 1e100:
        return np.full(N, 0.5)
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        v = np.clip(z - mid, 0.0, 1.0)
        if np.sum(v) > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    e = m - np.sum(h)

                break
            if e > 0:

import numpy as np
import cvxpy as cp
import math


def project_box_sum(z, N):
    m = N / 2.0
    if not np.all(np.isfinite(z)) or np.max(np.abs(z)) > 1e100:
        return np.full(N, 0.5)
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        v = np.clip(z - mid, 0.0, 1.0)
        if np.sum(v) > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    e = m - np.sum(h)
    for _ in range(3):
        for i in range(N):
            if abs(e) < 1e-15:
                break
            if e > 0:
                add = min(e, 1.0 - h[i])
                h[i] += add
                e -= add
            else:
                sub = min(-e, h[i])
                h[i] -= sub
                e += sub
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(np.sum(h) - m) > 1e-9 * max(1, N):
        return np.full(N, 0.5)
    return h


def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / N))


def is_feasible(h, N):
    m = N / 2.0
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - m) <= 1e-10 * max(1, N)


def lag_values_gradients(h, N):
    q = np.correlate(h, 1.0 - h, mode='full')
    G = np.zeros((2 * N - 1, N))
    for k in range(-(N - 1), N):
        row = k + N - 1
        if k >= 0:
            L = N - k
            G[row, k:] += (1.0 - h[:L])
            G[row, :L] -= h[k:]
        else:
            s = -k
            L = N - s
            G[row, :L] += (1.0 - h[s:])
            G[row, s:] -= h[:L]
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None, None
    return q, G


def build_model(N, delta):
    d = cp.Variable(N)
    u = cp.Variable()
    t = cp.Variable()
    x_p = cp.Parameter(N)
    r0_p = cp.Parameter()
    J0_p = cp.Parameter(N)
    nz = 2 * N - 2
    rnz_p = cp.Parameter(nz)
    Jnz_p = cp.Parameter((nz, N))
    R_p = cp.Parameter(nonneg=True)
    ones_nz = np.ones(nz)
    cons = [
        x_p + d >= 0,
        x_p + d <= 1,
        cp.sum(d) == 0,
        cp.norm(d, 2) <= R_p,
        u * ones_nz >= rnz_p + Jnz_p @ d,
        t >= u + delta * cp.sum_squares(d),
        t >= r0_p + J0_p @ d,
    ]
    prob = cp.Problem(cp.Minimize(t), cons)
    return d, u, t, x_p, r0_p, J0_p, rnz_p, Jnz_p, R_p, prob


def solve_local(N, delta, h0, max_iter):

    if N < 2:
        return h, best
    d, u, t, x_p, r0_p, J0_p, rnz_p, Jnz_p, R_p, prob = build_model(N, delta)
    R = min(1.0, math.sqrt(N))
    R_max = min(4.0, math.sqrt(N))
    R_min = 1e-4
    fails = 0
    zero_idx = N - 1
    nz_idx = np.array([i for i in range(2 * N - 1) if i != zero_idx])

        if fails >= 8 or R <= R_min:
            break
        q, G = lag_values_gradients(h, N)
        if q is None:
            break
        r = delta * q
        Jm = delta * G
        x_p.value = h
        r0_p.value = r[zero_idx]
        J0_p.value = Jm[zero_idx]
        rnz_p.value = r[nz_idx]
        Jnz_p.value = Jm[nz_idx]
        R_p.value = R
        try:

                       abstol=1e-8, reltol=1e-8, feastol=1e-8)
        except Exception:

            fails += 1
            continue

            fails += 1
            continue

                break

            R *= 0.5

    return h, best
