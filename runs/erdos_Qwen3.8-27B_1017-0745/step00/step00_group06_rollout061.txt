_dg = 0.0
_mg = 0.0

def _sobj(z):

    R = np.correlate(h, 1.0 - h, mode='full') * _dg
    a = np.max(R)
    return a + _taug * np.log(np.sum(np.exp((R - a) / _taug)))

    h = z[:_Ng]
    R = np.correlate(h, 1.0 - h, mode='full') * _dg
    a = np.max(R)
    w = np.exp((R - a) / _taug)
    w /= np.sum(w)
    g = np.zeros(_Ng)
    for q in range(-(_Ng - 1), _Ng):
        r = q + _Ng - 1

            continue

        for j in range(jlo, jhi):
            g[j + q] += w[r] * _dg * (1.0 - h[j])
