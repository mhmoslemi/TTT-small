        n_starts = 6
    n_starts = max(1, min(n_starts, 8))
    if N > 500:
        n_starts = min(n_starts, 2)
        maxiter = min(maxiter, 10)
    elif N > 200:
        n_starts = min(n_starts, 4)
        maxiter = min(maxiter, 20)
    starts = build_starts(parent, N, M, n_starts, rng)
    best_h = None
    best_c = float('inf')
    for s in starts:
        c = compute_score(s)
        if c < best_c - 1e-12:
            best_c = c
            best_h = s.copy()
    if best_h is None:
        best_h = np.full(N, 0.5)
        best_c = compute_score(best_h)
    if N <= 999:
        for s in starts:
            result = run_local_lp(s, N, maxiter)
            c = compute_score(result)
            if c < best_c - 1e-12:
                best_c = c
                best_h = result.copy()
    if best_h is None or not np.all(np.isfinite(best_h)) or len(best_h) != N:
        best_h = np.full(N, 0.5)
        best_c = compute_score(best_h)
    fp = project_capped(best_h, N, M)
    if fp is not None:
        best_h = fp
        best_c = compute_score(best_h)
    return best_h, float(best_c), int(N)
