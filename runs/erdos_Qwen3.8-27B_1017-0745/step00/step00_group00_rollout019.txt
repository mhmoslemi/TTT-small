        s_hi = starts[i + 1] if i + 1 < len(hull) else 1e18
        s_hi = min(s_hi, hi)
        if s_lo < s_hi - 1e-14:
            segments.append((s_lo, s_hi, hull[i][2]))
    return segments if segments else [(lo, hi, -1)]

def line_minimize(h, corr, score, a, b, budget):
    n = len(h)
    lo = max(-h[a], h[b] - 1.0)
    hi = min(1.0 - h[a], h[b])
    if hi - lo < 1e-14:
        return h, corr, score, 0
    A, B, D = make_pair_coeffs(h, corr, a, b)
    if A is None:
        return h, corr, score, 0
    c0, ca, cb = n - 1, a - b + n - 1, b - a + n - 1
    exclude = frozenset([c0, ca, cb])
    segments = build_hull(B, A, exclude, lo, hi)
    candidates = []
    for s_lo, s_hi, kline in segments:
        candidates.append(s_lo); candidates.append(s_hi)
        for ck in (c0, ca, cb):
            if abs(D[ck]) > 1e-14:
                sp = -B[ck] / (2.0 * D[ck])
                if s_lo - 1e-12 <= sp <= s_hi + 1e-12:
                    candidates.append(sp)
            if kline >= 0:
                da = D[ck]
                db = B[ck] - B[kline]
                dc = A[ck] - A[kline]
                for r in quad_roots(da, db, dc):
                    if s_lo - 1e-12 <= r <= s_hi + 1e-12:
                        candidates.append(r)
        for i2 in range(3):
            for j2 in range(i2 + 1, 3):
                ci, cj = (c0, ca, cb)[i2], (c0, ca, cb)[j2]
                da = D[ci] - D[cj]
                db = B[ci] - B[cj]
                dc = A[ci] - A[cj]
                for r in quad_roots(da, db, dc):
                    if s_lo - 1e-12 <= r <= s_hi + 1e-12:
                        candidates.append(r)
        if s_lo - 1e-12 <= 0.0 <= s_hi + 1e-12:
            candidates.append(0.0)
    # Score candidates by polynomial objective
    scored = []
    for alpha in candidates:
        if alpha < lo - 1e-12 or alpha > hi + 1e-12:
            continue
        alpha = max(lo, min(hi, alpha))
        if abs(alpha) < 1e-14:
            continue
        obj = -1e18
        for seg in segments:
            sl, sh, kl = seg
            if sl - 1e-11 <= alpha <= sh + 1e-11:
                if kl >= 0:
                    obj = max(obj, A[kl] + B[kl] * alpha)
                for ck in (c0, ca, cb):
                    obj = max(obj, A[ck] + B[ck] * alpha + D[ck] * alpha * alpha)
                break
        if obj > -1e17:
            scored.append((obj, alpha))
    scored.sort(key=lambda t: t[0])
    seen = []
    top = []
    for obj, alpha in scored:
        if any(abs(alpha - s) < 1e-12 for s in seen):
            continue
        seen.append(alpha); top.append(alpha)
        if len(top) >= 5:
            break
    best = (h, corr, score)
    budget_used = 0
    for alpha in top:
        if budget_used >= budget:
            break
        alpha = max(lo, min(hi, alpha))
        hh = h.copy()
        hh[a] += alpha
        hh[b] -= alpha
        hh[a] = max(0.0, min(1.0, hh[a]))
        hh[b] = max(0.0, min(1.0, hh[b]))
        if not is_feasible(hh, n):
            budget_used += 1
            continue
        sc, cr = exact_score_and_corr(hh)
        budget_used += 1
        if sc is None:
            continue
        if sc < best[2] - 1e-12 * max(1.0, abs(best[2])):
            best = (hh, cr, sc)
    return best[0], best[1], best[2], budget_used

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    iv = globals().get('initial_h_values', None)
    parent = None
    n = 45
    if iv is not None:
        try:
            iv = np.asarray(iv, dtype=np.float64)
            if iv.ndim == 1 and 2 <= len(iv) <= 999 and np.all(np.isfinite(iv)):
                if np.min(iv) >= -1e-6 and np.max(iv) <= 1.0 + 1e-6:
                    n = len(iv)
                    p = project_box_sum(iv, n / 2.0)
                    if p is not None and is_feasible(p, n):
                        parent = p
        except Exception:
            pass
    S = n / 2.0
    if parent is not None:
        fallback = parent.copy()
    else:
        fallback = np.full(n, 0.5, dtype=np.float64)
    fs, _ = exact_score_and_corr(fallback)
    if fs is None:
        fallback = np.full(45, 0.5, dtype=np.float64)
        n = 45; S = 22.5
        fs, _ = exact_score_and_corr(fallback)
        if fs is None:
            return fallback, 0.5, 45
    best_h, best_sc = fallback.copy(), fs
    # Build starts
    starts = []
    if parent is not None:
        starts.append(parent.copy())
        starts.append(parent[::-1].copy())
        for sd in (0.05, 0.18):
            starts.append(parent + rng.standard_normal(n) * sd)
    starts.append(rng.uniform(0, 1, n))
    starts.append(rng.uniform(0, 1, n))
    idx = np.arange(n)
    starts.append(0.5 + 0.45 * np.cos(2 * np.pi * idx / n + 0.37))
    blk = np.zeros(n); blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    starts.append(blk)
    # Project and deduplicate
    proj_starts = []
    for s in starts:
        p = project_box_sum(s, S)
        if p is not None and is_feasible(p, n):
            if not any(np.max(np.abs(p - q)) <= 1e-12 for q in proj_starts):
                proj_starts.append(p)
    budget = 150000
    pair_cap = min(n * (n - 1) // 2, max(1, int(30000000 / (5 * n * n))))
    n_sweeps = 3 if n <= 100 else (2 if n <= 300 else 1)
    for st in proj_starts:
        if budget <= 0:
            break
        hh = st.copy()
        sc, cr = exact_score_and_corr(hh)
        if sc is None:
            continue
        budget -= 1
        if sc < best_sc - 1e-12 * max(1.0, abs(best_sc)):
            best_sc = sc; best_h = hh.copy()
        for sweep in range(n_sweeps):
            if budget <= 0:
                break
            accepted = 0
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
            rng.shuffle(pairs)
            pairs = pairs[:pair_cap]
            for (a, b) in pairs:
                if budget <= 0:
                    break
                hh, cr2, sc2, used = line_minimize(hh, cr, sc, a, b, budget)
                budget -= used
                if sc2 < sc - 1e-12 * max(1.0, abs(sc)):
                    hh, cr, sc = hh, cr2, sc2
                    accepted += 1
                    if sc < best_sc - 1e-12 * max(1.0, abs(best_sc)):
                        best_sc = sc; best_h = hh.copy()
            if abs(S - float(np.sum(hh))) > 1e-10 * max(1, n):
                hh = project_box_sum(hh, S)
                if hh is None or not is_feasible(hh, n):
                    break
                sc, cr = exact_score_and_corr(hh)
                budget -= 1
                if sc is None:
                    break
            if accepted == 0:
                break
    # Final validation
    pb = project_box_sum(best_h, S)
    if pb is not None and is_feasible(pb, n):
        fh = pb
    else:
        fh = fallback
    sf, cf = exact_score_and_corr(fh)
    if sf is None:
        fh = np.full(n, 0.5); sf, _ = exact_score_and_corr(fh)
        if sf is None:
            fh = np.full(45, 0.5); n = 45; sf = 0.5
    # Compare with fallback
    ff, _ = exact_score_and_corr(fallback)
    if ff is not None and ff < sf:
        fh, sf = fallback, ff
    return fh, sf, len(fh)
