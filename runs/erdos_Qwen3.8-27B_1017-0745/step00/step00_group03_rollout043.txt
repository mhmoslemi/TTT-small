                b = h[0:L]

                G[row, s:s + L] += -a

    u = cp.Variable()
    t = cp.Variable()

    J0_p = cp.Parameter(N)
    r_nz_p = cp.Parameter(len(nz_rows))

        t >= u + delta * cp.sum_squares(d),
        t >= r0_p + J0_p @ d,

import numpy as np
import cvxpy as cp


def project_box_sum(z, N):
    z = np.asarray(z, dtype=np.float64)
    if z.shape != (N,) or not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    if np.max(np.abs(z)) > 1e100:
        return np.full(N, 0.5)
    m = N / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        v = np.clip(z - mid, 0.0, 1.0)
        if np.sum(v) > m:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    h = np.clip(z - theta, 0.0, 1.0)
    for _ in range(3):
        e = m - np.sum(h)
        if abs(e) < 1e-13:
            break
        if e > 0:
            for i in range(N):
                if h[i] < 1.0:
                    add = min(e, 1.0 - h[i])
                    h[i] += add
                    e -= add
                    if abs(e) < 1e-14:
                        break
        else:
            for i in range(N):
                if h[i] > 0.0:
                    sub = min(-e, h[i])
                    h[i] -= sub
                    e += sub
                    if abs(e) < 1e-14:
                        break
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(np.sum(h) - m) > 1e-9 * max(1, N):
        return np.full(N, 0.5)
    return h


def is_feasible(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - N / 2.0) > 1e-10 * max(1, N):
        return False
    return True


def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / N))


def lag_values_gradients(h, N):
    q = np.correlate(h, 1.0 - h, mode='full')
    G = np.zeros((2 * N - 1, N), dtype=np.float64)
    for row in range(2 * N - 1):
        k = row - (N - 1)
        if k >= 0:
            L = N - k
            if L > 0:
                G[row, k:k + L] += (1.0 - h[0:L])
                G[row, 0:L] += -h[k:k + L]
        else:
            s = -k
            L = N - s
            if L > 0:
                G[row, 0:L] += (1.0 - h[s:s + L])
                G[row, s:s + L] += -h[0:L]
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None, None
    return q, G


def build_model(N, delta):
    n_lags = 2 * N - 1
    nz_mask = np.ones(n_lags, dtype=bool)
    nz_mask[N - 1] = False
    nz_idx = np.where(nz_mask)[0]
    n_nz = len(nz_idx)
    d = cp.Variable(N)
    u = cp.Variable()
    t = cp.Variable()
    x_p = cp.Parameter(N)
    r0_p = cp.Parameter()
    J0_p = cp.Parameter(N)
    r_nz_p = cp.Parameter(n_nz)
    J_nz_p = cp.Parameter((n_nz, N))
    R_p = cp.Parameter(nonneg=True)
    cons = [
        x_p + d >= 0,
        x_p + d <= 1,
        cp.sum(d) == 0,
        cp.norm(d, 2) <= R_p,
        np.ones(n_nz) * u >= r_nz_p + J_nz_p @ d,
        t >= u + delta * cp.sum_squares(d),
        t >= r0_p + J0_p @ d,
    ]
    prob = cp.Problem(cp.Minimize(t), cons)
    return d, u, t, x_p, r0_p, J0_p, r_nz_p, J_nz_p, R_p, prob, nz_idx
