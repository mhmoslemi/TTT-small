import numpy as np
from scipy import optimize
import math


def project_feasible(y, S):
    n = len(y)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y))
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    res = S - float(np.sum(h))
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.where(1.0 - h > 1e-12)[0]
            if len(idx) > 0:
                h[idx] += min(res / len(idx), 1e-6)
                h = np.clip(h, 0.0, 1.0)
        else:
            idx = np.where(h > 1e-12)[0]
            if len(idx) > 0:
                h[idx] += max(res / len(idx), -1e-6)
                h = np.clip(h, 0.0, 1.0)
    return h


def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def is_feasible(h, S, tol=1e-10):
    return (h.ndim == 1 and len(h) >= 2 and np.all(np.isfinite(h))
            and np.min(h) >= -1e-9 and np.max(h) <= 1.0 + 1e-9
            and abs(float(np.sum(h)) - S) <= tol)


def smooth_value(h, dx, tau):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(c))
    w = np.exp((c - m) / tau)
    return m + tau * math.log(float(np.sum(w)))


def smooth_grad(h, dx, tau):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(c))
    w = np.exp((c - m) / tau)
    p = w / float(np.sum(w))
    grad = np.zeros(n)
    shifts = np.arange(-(n - 1), n)
    for k in range(2 * n - 1):
        sh = int(shifts[k])
        v1 = np.zeros(n)
        v2 = np.zeros(n)
        if sh >= 0:
            v1[:n - sh] = 1.0 - h[sh:]
            v2[sh:] = h[:n - sh]
        else:
            s = -sh
            v1[s:] = 1.0 - h[:n - s]
            v2[:n - s] = h[s:]
        grad += p[k] * dx * (v1 - v2)
    return grad


def run_gradient(h0, S, dx, taus, max_iter=80):
    n = len(h0)
    h = h0.copy()
    for tau in taus:
        alpha = 1.0
        small = 0
        for _ in range(max_iter):
            F = smooth_value(h, dx, tau)
            g = smooth_grad(h, dx, tau)
            if not np.all(np.isfinite(g)):
                break
            y = project_feasible(h - alpha * g, S)
            d = y - h
            if np.max(np.abs(d)) < 1e-10:
                break
            gd = float(np.dot(g, d))
            if gd > -1e-14:
                break
            Fy = smooth_value(y, dx, tau)
            if Fy <= F + 1e-4 * gd + 1e-14:
                h = y.copy()
                small = small + 1 if (F - Fy) < 1e-12 else 0
                if small >= 10:
                    break
                alpha = min(1.25 * alpha, 5.0)
            else:
                alpha *= 0.5
                if alpha < 1e-15:
                    break
    return h


def _slsqp_obj(z, n, dx):
    return float(z[n])


def _slsqp_obj_jac(z, n, dx):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g


def _slsqp_ineq(z, n, dx):
    h = z[:n]
    t = z[n]
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - c


def _slsqp_ineq_jac(z, n, dx):
    h = z[:n]
    J = np.zeros((2 * n - 1, n + 1))
    shifts = np.arange(-(n - 1), n)
    for k in range(2 * n - 1):
        sh = int(shifts[k])
        J[k, n] = 1.0
        v1 = np.zeros(n)
        v2 = np.zeros(n)
        if sh >= 0:
            v1[:n - sh] = 1.0 - h[sh:]
            v2[sh:] = h[:n - sh]
        else:
            s = -sh
            v1[s:] = 1.0 - h[:n - s]
            v2[:n - s] = h[s:]
        J[k, :n] = -dx * (v1 - v2)
    return J


def _slsqp_eq(z, n, dx, S):
    return float(np.sum(z[:n])) - S


def _slsqp_eq_jac(z, n, dx, S):
    g = np.ones(n + 1)
    g[n] = 0.0
    return g


def slsqp_refine(h0, S, dx, n):
    t0 = min(1.0, exact_score(h0, dx) + 1e-8)
    z0 = np.concatenate([h0, [t0]])
    bds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'fun': _slsqp_eq, 'jac': _slsqp_eq_jac, 'type': 'eq', 'args': (n, dx, S)},
        {'fun': _slsqp_ineq, 'jac': _slsqp_ineq_jac, 'type': 'ineq', 'args': (n, dx)},
    ]
    try:
        res = optimize.minimize(_slsqp_obj, z0, jac=_slsqp_obj_jac, args=(n, dx),
                                bounds=bds, constraints=cons, method='SLSQP',
                                options={'maxiter': 250, 'ftol': 1e-10, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_feasible(np.clip(res.x[:n], 0.0, 1.0), S)
            if is_feasible(h_new, S):
                return h_new
    except Exception:
        pass
    return None


def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = None
    if 'initial_h_values' in globals():
        pv = np.asarray(globals()['initial_h_values'], dtype=np.float64)
        if (pv.ndim == 1 and 2 <= len(pv) <= 999
                and np.all(np.isfinite(pv)) and np.all(np.abs(pv) < 1e6)):
            parent = pv.copy()
    if parent is not None:
        n = len(parent)
    else:
        n = kwargs.get('n_points', 45)
        if not isinstance(n, int) or n < 2 or n > 999:
            n = 45
    dx = 2.0 / n
    S = n / 2.0
    fallback = project_feasible(parent, S) if parent is not None else np.full(n, 0.5)
    if not is_feasible(fallback, S):
        fallback = np.full(n, 0.5)
    best_h = fallback.copy()
    best_sc = exact_score(best_h, dx)

    starts = [fallback.copy()]
    if parent is not None:
        starts.append(project_feasible(1.0 - parent, S))
        starts.append(project_feasible(parent[::-1].copy(), S))
        starts.append(project_feasible(1.0 - parent[::-1].copy(), S))
        for sd in [0.08, 0.25]:
            starts.append(project_feasible(parent + sd * rng.standard_normal(n), S))
    starts.append(project_feasible(rng.random(n), S))
    starts.append(project_feasible(rng.beta(0.5, 0.5, n), S))

    taus = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00001]
    results = []
    for st in starts:
        sc = exact_score(st, dx)
        if is_feasible(st, S) and sc < best_sc - 1e-12:
            best_h, best_sc = st.copy(), sc
        h_out = run_gradient(st, S, dx, taus)
        sc_out = exact_score(h_out, dx)
        if is_feasible(h_out, S) and sc_out < best_sc - 1e-12:
            best_h, best_sc = h_out.copy(), sc_out
        results.append((h_out.copy(), sc_out))

    for sig in [0.04, 0.12, 0.30, 0.50]:
        st = project_feasible(best_h + sig * rng.standard_normal(n), S)
        h_out = run_gradient(st, S, dx, taus)
        sc_out = exact_score(h_out, dx)
        if is_feasible(h_out, S) and sc_out < best_sc - 1e-12:
            best_h, best_sc = h_out.copy(), sc_out
        results.append((h_out.copy(), sc_out))

    if n <= 100:
        scores_arr = np.array([r[1] for r in results])
        order = np.argsort(scores_arr)
        picked = []
        for oi in order[:min(10, len(order))]:
            cand = results[oi][0]
            distinct = True
            for pc in picked:
                if np.max(np.abs(cand - pc)) < 1e-5:
                    distinct = False
                    break
            if distinct:
                picked.append(cand)
                if len(picked) >= 3:
                    break
        for pc in picked:
            hr = slsqp_refine(pc, S, dx, n)
            if hr is not None:
                sc_hr = exact_score(hr, dx)
                if sc_hr < best_sc - 1e-12:
                    best_h, best_sc = hr.copy(), sc_hr

    best_h = project_feasible(best_h, S)
    fb_sc = exact_score(fallback, dx)
    if not is_feasible(best_h, S) or exact_score(best_h, dx) > fb_sc + 1e-12:
        best_h = fallback.copy()
    best_h = np.clip(best_h, 0.0, 1.0)
    final_sc = exact_score(best_h, dx)
    return (best_h.copy(), final_sc, n)
