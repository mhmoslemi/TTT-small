        src = h[lo:hi]
        dst = h[lo + d:hi + d]

    return lags

                    continue
                F_t = smooth_obj_and_grad(trial, dx, tau)[0]
                if F_t <= F + 1e-4 * gd + 1e-12 * max(1.0, abs(F)):
                    h = trial
                    accepted = True
                    eta = min(8.0, 1.2 * eta)
                    break
                eta *= 0.5
                if eta < 1e-8:
                    break
            if not accepted:
                break
            if abs(F_t - F) <= 1e-10 * max(1.0, abs(F)):
                stall += 1
            else:
                stall = 0
            if stall >= 12:
                break
    return h

def pair_transfer_polish(h0, seed):
    n = len(h0)
    S = n / 2.0
    rng = np.random.default_rng(seed + 9999)
    h_cur = h0.copy()
    best_h = h0.copy()
    best_s = exact_score(best_h)
    P = max(80, min(800, int(10000000 / (n * n))))
    stages = [(0.08, 0.001), (0.025, 0.0003), (0.008, 0.00008), (0.002, 0.0)]
    Pp = P // 4
    for si in range(4):
        step_lim, T = stages[si]
        p = Pp if si < 3 else P - Pp * 3
        for _ in range(p):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            sign = 1.0 if rng.random() < 0.5 else -1.0
            if sign > 0:
                cap = min(1.0 - h_cur[i], h_cur[j])
            else:
                cap = min(h_cur[i], 1.0 - h_cur[j])
            mag = rng.random() * min(step_lim, cap)
            if mag < 1e-12:
                continue
            trial = h_cur.copy()
            trial[i] += sign * mag
            trial[j] -= sign * mag
            if trial[i] < -1e-12 or trial[i] > 1.0 + 1e-12 or trial[j] < -1e-12 or trial[j] > 1.0 + 1e-12:
                continue
            s_new = exact_score(trial)
            if s_new <= best_s:
                h_cur = trial
                best_h = trial
                best_s = s_new
            elif T > 0 and rng.random() < math.exp(-(s_new - best_s) / T):
                h_cur = trial
    return best_h, best_s

def run(seed=42, **kwargs):
    n_kw = kwargs.get('n_points', None)
    use_parent = False
    parent = None
    ph = globals().get('initial_h_values', None)
    if ph is not None:
        try:
            parent = np.array(ph, dtype=np.float64).flatten()
            if parent.size >= 2 and np.all(np.isfinite(parent)):
                use_parent = True
        except Exception:
            pass
    if use_parent:
        n = parent.size
    elif n_kw is not None:
        n = max(8, min(int(n_kw), 512))
    else:
        n = 80
    dx = 2.0 / n
    S = n / 2.0
    rng = np.random.default_rng(seed)
    h_const = np.full(n, 0.5)
    s_const = exact_score(h_const)
    best_h = h_const.copy()
    best_s = s_const
    if use_parent:
        h_par = project_simplex(parent, S)
        s_par = exact_score(h_par)
        if s_par < best_s:
            best_h = h_par.copy()
            best_s = s_par
    starts = []
    i_arr = np.arange(n)
    if use_parent:
        starts.append(parent.copy())
        for sd in [0.04, 0.15, 0.35]:
            starts.append(project_simplex(parent + rng.standard_normal(n) * sd, S))
        q = int(rng.integers(1, n))
        phi = rng.random() * 2.0 * math.pi
        starts.append(project_simplex(0.5 + 0.45 * np.cos(2.0 * math.pi * q * i_arr / n + phi), S))
    else:
        starts.append(h_const.copy())
        for _ in range(2):
            q = int(rng.integers(1, n))
            phi = rng.random() * 2.0 * math.pi
            starts.append(project_simplex(0.5 + 0.45 * np.cos(2.0 * math.pi * q * i_arr / n + phi), S))
    starts.append(project_simplex(rng.random(n), S))
    perm = rng.permutation(n)
    h_bin = np.zeros(n)
    h_bin[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        h_bin[perm[n // 2]] = 0.5
    starts.append(h_bin)
    dedup = []
    for s in starts:
        dup = False
        for sd in dedup:
            if np.max(np.abs(s - sd)) < 1e-10:
                dup = True
                break
        if not dup:
            dedup.append(s)
    max_starts = 7 if n <= 256 else 3
    dedup = dedup[:max_starts]
    baseline_s = best_s
    for s in dedup:
        sv = exact_score(s)
        if sv < best_s:
            best_h = s.copy()
            best_s = sv
    tau_sched = [0.02, 0.004, 0.0008, 0.00016, 0.000032, 0.0000064]
    B = min(480, max(12, int(4000000 / (n * n))))
    for s in dedup:
        res = smooth_optimize(s, dx, tau_sched, B)
        sv = exact_score(res)
        if sv < best_s - 1e-12:
            best_h = res.copy()
            best_s = sv
    if best_s >= baseline_s - 1e-8 and n <= 256:
        for sd in [0.6, 1.0]:
            pert = project_simplex(best_h + rng.standard_normal(n) * sd, S)
            res = smooth_optimize(pert, dx, tau_sched, max(1, B // 2))
            sv = exact_score(res)
            if sv < best_s - 1e-12:
                best_h = res.copy()
                best_s = sv
    if n <= 256:
        best_h, best_s = pair_transfer_polish(best_h, seed)
    best_h = np.clip(best_h, 0.0, 1.0)
    r = S - np.sum(best_h)
    if abs(r) > 1e-10:
        for i in range(n):
            if abs(r) < 1e-14:
                break
            if r > 0:
                a = min(r, 1.0 - best_h[i])
                best_h[i] += a
                r -= a
            else:
                a = min(-r, best_h[i])
                best_h[i] -= a
                r += a
    final_score = exact_score(best_h)
    if final_score > s_const:
        best_h = h_const.copy()
        final_score = s_const
    else:
        best_s = final_score
    return best_h, float(best_s), int(n)
