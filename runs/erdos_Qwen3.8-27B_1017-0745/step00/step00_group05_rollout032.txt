def capped_simplex_project(v, target):
    n = len(v)
    if not np.all(np.isfinite(v)) or np.max(np.abs(v)) > 1e12:
        return None
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - np.sum(y)
    for _ in range(3):
        if res > 1e-12:
            for i in range(n):
                add = min(res, 1.0 - y[i])
                y[i] += add
                res -= add
                if res <= 1e-12:
                    break
        elif res < -1e-12:
            for i in range(n):
                sub = min(-res, y[i])
                y[i] -= sub
                res += sub
                if res >= -1e-12:
                    break
        else:
            break

        return None
    y = np.clip(y, 0.0, 1.0)
    if abs(np.sum(y) - target) > 1e-10:
        return None
    return y

    if np.min(h) < 0.0 or np.max(h) > 1.0:
