import numpy as np
import math

def is_feasible(h, n):
    try:
        h = np.asarray(h, dtype=np.float64)
    except Exception:
        return False
    if h.ndim != 1 or h.shape[0] != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if not (h.min() >= -1e-15 and h.max() <= 1.0 + 1e-15):
        return False
    S = n / 2.0
    if abs(np.sum(h) - S) > 1e-10:
        return False
    if abs(math.fsum(float(x) for x in h) - S) > 1e-10:
        return False
    return True

def project_capped_simplex(v, n):
    try:
        v = np.asarray(v, dtype=np.float64)
        if v.ndim != 1 or v.shape[0] != n:
            return None
        v = np.clip(v, -1e6, 1e6)
    except Exception:
        return None
    S = n / 2.0
    lo = float(v.min()) - 1.0
    hi = float(v.max())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(v - mid, 0.0, 1.0)
        if np.sum(h) > S:
            lo = mid
        else:
            hi = mid
    mid = 0.5 * (lo + hi)
    h = np.clip(v - mid, 0.0, 1.0)
    # residual correction
    for _ in range(200):
        res = S - math.fsum(float(x) for x in h)
        if abs(res) <= 1e-12:
            break
        sgn = 1.0 if res > 0 else -1.0
        changed = False
        for j in range(n):
            if sgn > 0 and h[j] < 1.0:
                d = min(sgn * res, 1.0 - h[j])
                if d > 1e-15:
                    h[j] += d
                    changed = True
                    break
            elif sgn < 0 and h[j] > 0.0:
                d = abs(min(sgn * res, h[j]))
                if d > 1e-15:
                    h[j] -= d
                    changed = True
                    break
        if not changed:
            break
    # final check
    if not is_feasible(h, n):
        # try setting one interior coordinate
        for j in range(n):
            others = math.fsum(float(x) for k, x in enumerate(h) if k != j)
            val = S - others
            if 0.0 <= val <= 1.0:
                h[j] = val
                break
    if not is_feasible(h, n):
        return None
    return h

def exact_score(h, n):

        if not is_feasible(h, n):
            return None
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        score = float(np.max(corr * dx))
        if not math.isfinite(score):
            return None
        return score
    except Exception:
        return None
