import numpy as np

import math

def compute_lags(h, n):
    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    q[n-1] = dx * np.dot(h, 1 - h)
    G[n-1, :] = dx * (1 - 2 * h)
    for d in range(1, n):
        m = n - d
        q[n-1+d] = dx * np.dot(h[d:], 1 - h[:m])
        G[n-1+d, d:] += dx * (1 - h[:m])
        G[n-1+d, :m] -= dx * h[d:]
        q[n-1-d] = dx * np.dot(h[:m], 1 - h[d:])
        G[n-1-d, :m] += dx * (1 - h[d:])
        G[n-1-d, d:] -= dx * h[:m]
    return q, G

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

    res = S - math.fsum(h.tolist())
    if abs(res) > 1e-13:
        mask = (h < 1.0) if res > 0 else (h > 0.0)

    q, _ = compute_lags(z[:n], n)
    qm = np.max(q)
    return qm + tau * math.log(math.fsum(np.exp((q - qm) / tau).tolist()))

def soft_jac(z, n, tau):
    q, G = compute_lags(z[:n], n)
    qm = np.max(q)
    e = np.exp((q - qm) / tau)
    return G.T @ (e / np.sum(e))

def sm_eq(z, S):
    return np.sum(z) - S

def sm_eq_jac(z, S):
    return np.ones(len(z))

def epi_fun(z, n):
    return z[n]

def epi_jac(z, n):
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return g

def epi_eq(z, n):
    return np.sum(z[:n]) - n / 2.0

def epi_eq_jac(z, n):
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def epi_ineq(z, n):
    q, _ = compute_lags(z[:n], n)
    return z[n] - q

def epi_ineq_jac(z, n):
    _, G = compute_lags(z[:n], n)
    M = 2 * n - 1
    J = np.zeros((M, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J
