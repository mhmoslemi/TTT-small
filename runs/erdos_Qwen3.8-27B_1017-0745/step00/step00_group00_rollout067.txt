import numpy as np
from scipy.optimize import minimize

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    res = S - y.sum()
    if abs(res) > 1e-14:
        if res > 0:
            order = np.argsort(1.0 - y)[::-1]
        else:
            order = np.argsort(y)
        for i in order:
            if abs(res) < 1e-15:
                break
            if res > 0:
                a = min(1.0 - y[i], res)
                y[i] += a
                res -= a
            else:
                a = min(y[i], -res)
                y[i] -= a
                res += a
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    if abs(y.sum() - S) > 1e-10 * max(1, n):
        return None
    return np.clip(y, 0.0, 1.0)

def check_feasible(h, n):
    if h is None or len(h) != n or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(h.sum() - n * 0.5) <= 1e-10 * max(1, n)

def exact_score(h):
    n = len(h)
    if not check_feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def lag_grad(h, n):
    nl = 2 * n - 1
    Qs = np.zeros(nl)
    G = np.zeros((nl, n))
    for idx in range(nl):
        k = idx - n + 1
        lo = max(0, -k)
        hi = min(n, n - k)
        if lo >= hi:
            continue
        ns = np.arange(lo, hi)
        Qs[idx] = np.sum(h[ns + k] * (1.0 - h[ns]))
        s1lo, s1hi = max(0, k), min(n, n + k)
        if s1lo < s1hi:
            ms = np.arange(s1lo, s1hi)
            G[idx, ms] += 1.0 - h[ms - k]
        s2lo, s2hi = max(0, -k), min(n, n - k)
        if s2lo < s2hi:
            ms2 = np.arange(s2lo, s2hi)
            G[idx, ms2] -= h[ms2 + k]
    return Qs, G

def _obj(z):
    return z[-1]

def _obj_grad(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def _ineq(z):
    n = len(z) - 1
    h, t = z[:n], z[n]
    Qs, _ = lag_grad(h, n)
    return t - Qs / n

def _ineq_jac(z):
    n = len(z) - 1
    h = z[:n]
    _, G = lag_grad(h, n)
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G / n
    J[:, n] = 1.0
    return J

def _eq(z):
    n = len(z) - 1
    return z[:n].sum() / n - 0.5

def _eq_jac(z):
    n = len(z) - 1
    J = np.zeros(n + 1)
    J[:n] = 1.0 / n
    return J

def try_slsqp(h0, n, bounds):
    Qs, _ = lag_grad(h0, n)
    t0 = min(1.0, float(np.max(Qs)) / n + 1e-4)
    z0 = np.concatenate([h0, [t0]])
    constraints = [
        {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac},
        {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}
    ]
    try:
        res = minimize(_obj, z0, jac=_obj_grad, bounds=bounds,
                       constraints=constraints, method='SLSQP',
                       options={'maxiter': 300, 'ftol': 1e-10, 'disp': False})
    except Exception:
        return None
    if res is None:
        return None
    h = np.asarray(res.x[:n], dtype=np.float64)
    if not np.all(np.isfinite(h)):
        return None
    return h

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = None
    n = 45
    if 'initial_h_values' in globals():
        iv = globals()['initial_h_values']
        if iv is not None:

                    n = len(arr)
                    pr = project_box_sum(arr, n * 0.5)
                    if pr is not None:
                        parent = pr
    if parent is None:
        parent = np.full(n, 0.5)
    n = len(parent)
    S = n * 0.5
    best = parent.copy()
    best_score = exact_score(best)
    if not np.isfinite(best_score):
        best = np.full(n, 0.5)
        best_score = exact_score(best)

    starts = [parent.copy(), parent[::-1].copy()]
    for sigma in [0.01, 0.03, 0.06, 0.12, 0.20, 0.05]:
        pert = parent + sigma * rng.standard_normal(n)
        p = project_box_sum(pert, S)
        if p is not None:
            starts.append(p)
    for _ in range(4):
        u = rng.uniform(0, 1, n)
        p = project_box_sum(u, S)
        if p is not None:
            starts.append(p)

    for h0 in starts:
        h1 = try_slsqp(h0, n, bounds)
        if h1 is None:
            continue
        h1 = project_box_sum(h1, S)
        if h1 is None or not check_feasible(h1, n):
            continue
        sc = exact_score(h1)
        if sc < best_score:
            best = h1
            best_score = sc

    for _ in range(4):
        pert = best + sigma * rng.standard_normal(n)
        p = project_box_sum(pert, S)
        if p is None or not check_feasible(p, n):

            continue
