import numpy as np
from scipy import signal

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1 or not np.all(np.isfinite(v)) or np.max(np.abs(v)) > 1e12:
        return None
    n = len(v)
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(80):
        m = 0.5 * (lo + hi)
        if np.sum(np.clip(v - m, 0.0, 1.0)) > S:
            lo = m
        else:
            hi = m
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    res = S - np.sum(y)
    for _ in range(2):
        if abs(res) < 1e-13:
            break
        for i in range(n):
            if abs(res) < 1e-13:
                break
            if res > 0:
                a = min(res, 1.0 - y[i])
                y[i] += a
                res -= a
            else:
                a = min(-res, y[i])
                y[i] -= a
                res += a
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(np.sum(y) - S) > 1e-10 * max(1, n):
        return None
    return y

def is_feasible(h, n):
    if h.ndim != 1 or len(h) != n or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) < 1e-10 * max(1, n)

def exact_score(h):
    n = len(h)
    if not is_feasible(h, n):
        return None
    corr = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(corr)):
        return None
    return float(np.max(corr) * 2.0 / n)

def smooth_eval(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    sc = float(np.max(corr) * 2.0 / n)
    q = corr / n
    z = q / tau
    zmax = np.max(z)
    e = np.exp(z - zmax)
    es = np.sum(e)
    if es <= 0 or not np.isfinite(es):
        return None, None, sc
    p = e / es
    F = float(tau * (zmax + np.log(es)))
    p_rev = p[::-1]
    if n <= 128:
        c1 = np.convolve(p, 1.0 - h, mode='full')
        c2 = np.convolve(p_rev, h, mode='full')
    else:
        try:
            c1 = signal.fftconvolve(p, 1.0 - h, mode='full')
            c2 = signal.fftconvolve(p_rev, h, mode='full')
            if not (np.all(np.isfinite(c1)) and np.all(np.isfinite(c2))):
                raise ValueError
        except Exception:
            c1 = np.convolve(p, 1.0 - h, mode='full')
            c2 = np.convolve(p_rev, h, mode='full')
    g = (c1[n - 1:2 * n - 1] - c2[n - 1:2 * n - 1]) / n
    if not np.all(np.isfinite(g)):
        return None, None, sc
    return F, g, sc
