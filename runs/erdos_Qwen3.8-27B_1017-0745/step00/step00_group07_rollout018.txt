import numpy as np
import math

def project_capped_simplex(z, S):
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(z - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - np.sum(y)
    if r > 1e-15:
        for i in range(len(y)):
            if r <= 1e-15:
                break
            slack = 1.0 - y[i]
            if slack > 1e-15:
                amt = min(r, slack)
                y[i] += amt
                r -= amt
    elif r < -1e-15:
        for i in range(len(y)):
            if r >= -1e-15:
                break
            slack = y[i]
            if slack > 1e-15:
                amt = min(-r, slack)
                y[i] -= amt
                r += amt
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)):
        y = np.full(len(z), S / len(z))
    return y

def exact_score(h):
    dx = 2.0 / len(h)
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    result = float(np.max(vals))
    return result if np.isfinite(result) else float('inf')

def compute_lag_scores(h, dx):
    n = len(h)
    scores = np.zeros(2 * n - 1)
    for d in range(-(n - 1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        scores[d + n - 1] = dx * np.sum(h[lo:hi] * (1.0 - h[lo + d:hi + d]))
    return scores

def surrogate_and_grad(h, dx, tau):
    n = len(h)
    scores = compute_lag_scores(h, dx)
    amax = float(np.max(scores))
    exp_s = np.exp((scores - amax) / tau)
    w = exp_s / float(np.sum(exp_s))
    F = amax + tau * math.log(float(np.sum(exp_s)))
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wd = w[idx]
        if wd < 1e-14:
            continue
        d = idx - (n - 1)
        lo = max(0, -d)
        hi = min(n, n - d)
        src = h[lo:hi]
        dst = h[lo + d:hi + d]
        grad[lo:hi] += wd * dx * (1.0 - dst)
        grad[lo + d:hi + d] -= wd * dx * src
    return F, grad

def optimize_start(h_init, dx, S, n, B, rng):
    tau_list = [0.02, 0.004, 0.0008, 0.00016, 0.000032, 0.0000064]
    h = h_init.copy()
    best_h = h.copy()
    best_sc = exact_score(h)
    iters_per = B // len(tau_list)
    for k in range(len(tau_list)):
        tau = tau_list[k]
        budget = iters_per + (B - iters_per * len(tau_list)) if k < (B % len(tau_list)) else iters_per
        eta = 1.0
        consecutive = 0
        prev_F = None
        for it in range(budget):
            F, g = surrogate_and_grad(h, dx, tau)
            if not np.isfinite(F) or not np.all(np.isfinite(g)):
                break
            trial = None
            for _ in range(16):
                if eta < 1e-8:
                    break
                step = h - eta * g
                trial = project_capped_simplex(step, S)
                d = trial - h
                if np.max(np.abs(d)) < 1e-16:
                    eta = 0.0
                    break
                gd = float(np.dot(g, d))
                if gd >= 0:
                    eta *= 0.5
                    continue
                F_t, _ = surrogate_and_grad(trial, dx, tau)
                if not np.isfinite(F_t):
                    eta *= 0.5
                    continue
                rhs = F + 1e-4 * gd + 1e-12 * max(1.0, abs(F))
                if F_t <= rhs:
                    accepted = True
                    break
                else:
                    accepted = False
                    eta *= 0.5
            if eta == 0.0 or trial is None or not np.all(np.isfinite(trial)):
                break
            sc_t = exact_score(trial)
            if sc_t < best_sc - 1e-12:
                best_sc = sc_t
                best_h = trial.copy()
            if prev_F is not None:
                if abs(F - prev_F) <= 1e-10 * max(1.0, abs(F)):
                    consecutive += 1
                else:
                    consecutive = 0
                if consecutive >= 12:
                    break
            prev_F = F
            h = trial
            if accepted:
                eta = min(8.0, 1.2 * eta)
    return best_h, best_sc

def pair_transfer_polish(h_init, dx, n, rng):
    P = max(80, min(800, 10000000 // (n * n)))
    per_stage = P // 4
    step_limits = [0.08, 0.025, 0.008, 0.002]
    temps = [0.001, 0.0003, 0.00008, 0.0]
    h = h_init.copy()
    best_h = h.copy()
    best_sc = exact_score(h)
    cur_sc = best_sc
    for stage in range(4):
        sl = step_limits[stage]
        T = temps[stage]
        for _ in range(per_stage):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            sign = 1.0 if rng.random() < 0.5 else -1.0
            if sign > 0:
                cap = min(1.0 - h[i], h[j])
            else:
                cap = min(h[i], 1.0 - h[j])
            if cap < 1e-12:
                continue
            mag = rng.random() * min(sl, cap)
            if mag < 1e-12:
                continue
            delta = sign * mag
            trial = h.copy()
            trial[i] += delta
            trial[j] -= delta
            sc_t = exact_score(trial)
            if sc_t <= cur_sc or (T > 0 and np.exp(-(sc_t - cur_sc) / T) >= rng.random()):
                h = trial
                cur_sc = sc_t
                if sc_t < best_sc - 1e-12:
                    best_sc = sc_t
                    best_h = trial.copy()
    return best_h, best_sc

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    h_parent = None
    n_parent = 0
    if 'initial_h_values' in globals():
        try:
            raw = np.asarray(globals()['initial_h_values'], dtype=np.float64)
            if raw.ndim > 1:
                raw = raw.ravel()
            if raw.size >= 2 and np.all(np.isfinite(raw)):
                h_parent = raw.copy()
                n_parent = raw.size
        except Exception:
            pass
    if n_parent > 0:
        n = n_parent
    elif 'n_points' in kwargs:
        n = int(kwargs['n_points'])
        n = max(8, min(512, n))
    else:
        n = 80
    dx = 2.0 / n
    S = n / 2.0
    # Fallback
    h_fallback = np.full(n, 0.5)
    sc_fallback = exact_score(h_fallback)
    # Parent repair
    if h_parent is not None:
        h_parent = project_capped_simplex(h_parent, S)
        sc_parent = exact_score(h_parent)
        if sc_parent < sc_fallback:
            h_fallback = h_parent.copy()
            sc_fallback = sc_parent
    # Build starts
    starts = []
    if h_parent is not None:
        starts.append(h_parent.copy())
        for sd in [0.04, 0.15, 0.35]:
            pert = h_parent + rng.normal(0, sd, n)
            starts.append(project_capped_simplex(pert, S))
        q = int(rng.integers(1, n))
        phi = float(rng.uniform(0, 2 * math.pi))
        xx = np.arange(n)
        cyc = 0.5 + 0.45 * np.cos(2 * math.pi * q * xx / n + phi)
        starts.append(project_capped_simplex(cyc, S))
        starts.append(project_capped_simplex(rng.uniform(0, 1, n), S))
        perm = rng.permutation(n)
        binv = np.zeros(n)
        binv[perm[:n // 2]] = 1.0
        if n % 2 == 1:
            binv[perm[n // 2]] = 0.5
        starts.append(binv.copy())
    else:
        starts.append(np.full(n, 0.5))
        for qq in [2, 5]:
            phi = float(rng.uniform(0, 2 * math.pi))
            xx = np.arange(n)
            cyc = 0.5 + 0.45 * np.cos(2 * math.pi * qq * xx / n + phi)
            starts.append(project_capped_simplex(cyc, S))
        starts.append(project_capped_simplex(rng.uniform(0, 1, n), S))
    # Deduplicate
    unique_starts = [starts[0]]
    for s in starts[1:]:
        if all(np.max(np.abs(s - u)) > 1e-10 for u in unique_starts):
            unique_starts.append(s)
    max_starts = 7 if n <= 256 else 3
    unique_starts = unique_starts[:max_starts]
    # Score starts
    best_h = h_fallback.copy()
    best_sc = sc_fallback
    baseline_sc = best_sc
    for s in unique_starts:
        sc = exact_score(s)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_h = s.copy()
        baseline_sc = min(baseline_sc, sc)
    # Optimize each start
    B = min(480, max(12, 4000000 // (n * n)))
    for s in unique_starts:
        h_opt, sc_opt = optimize_start(s, dx, S, n, B, rng)
        if np.all(np.isfinite(h_opt)) and sc_opt < best_sc - 1e-12:
            best_sc = sc_opt
            best_h = h_opt.copy()
    # Diversification if no improvement
    if best_sc > baseline_sc - 1e-8 and n <= 256:
        for sd in [0.6, 1.0]:
            pert = best_h + rng.normal(0, sd, n)
            pert = project_capped_simplex(pert, S)
            h_opt, sc_opt = optimize_start(pert, dx, S, n, B // 2, rng)
            if np.all(np.isfinite(h_opt)) and sc_opt < best_sc - 1e-12:
                best_sc = sc_opt
                best_h = h_opt.copy()
    # Pair transfer polish
    h_pol, sc_pol = pair_transfer_polish(best_h, dx, n, rng)
    if sc_pol < best_sc - 1e-12:
        best_sc = sc_pol
        best_h = h_pol.copy()
    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    r = S - np.sum(best_h)
    if abs(r) > 1e-14:
        best_h = project_capped_simplex(best_h, S)
    final_sc = exact_score(best_h)
    if final_sc < sc_fallback - 1e-12:
        h_out = best_h
        c5_out = final_sc
    else:
        h_out = h_fallback
        c5_out = sc_fallback
    c5_out = exact_score(h_out)
    return (h_out, c5_out, len(h_out))
