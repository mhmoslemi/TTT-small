    g = np.zeros(len(z)); g[:-1] = 1.0; return g

def eiv_f(z):
    R, _ = lag_rg(z[:-1])
    return z[-1] - R

def eiv_j(z):
    _, J = lag_rg(z[:-1])
    g = np.zeros((J.shape[0], J.shape[1] + 1))
    g[:, :-1] = -J; g[:, -1] = 1.0
    return g

def _pick_top3(cands):
    cands = sorted(cands, key=lambda c: c[0]) if cands else []
    picked = []
    for c, h in cands:
        if all(np.max(np.abs(h - ph)) > 1e-10 for _, ph in picked):
            picked.append((c, h))
        if len(picked) == 3:
            break
    return picked

def run(seed=42, **kwargs):
    global _N, _m, _tau
    N = 95
    h0 = None
    try:
        ih = np.asarray(initial_h_values, dtype=np.float64)
        if ih.ndim == 1 and ih.size >= 2 and np.all(np.isfinite(ih)):
            N = ih.size
            h0 = proj_bm(ih.copy())
    except (NameError, Exception):
        pass
    if h0 is None:
        try:
            N = int(kwargs.get('n_points', 95))
        except (ValueError, TypeError):
            N = 95
        N = max(2, min(N, 999))
        h0 = np.full(N, 0.5)
    if N > 999:
        h0 = proj_bm(h0)
        return (h0, exact_score(h0), N)
    _N = N
    _m = N * 0.5
    h0 = proj_bm(h0)
    best_h = h0.copy()
    best_c = exact_score(best_h)

    rng = np.random.default_rng(42 if seed is None else seed)
    starts = [h0.copy(), np.full(N, 0.5)]
    step = np.zeros(N)
    step[:int(_m)] = 1.0
    if N % 2 == 1:
        step[int(_m)] = 0.5
    starts.append(proj_bm(step.copy()))
    iarr = np.arange(N, dtype=np.float64)
    for f in [1, 3, 7]:
        starts.append(proj_bm(0.5 + 0.22 * np.sin(2.0 * np.pi * f * iarr / N)))
    try:
        nr = max(6, min(int(kwargs.get('n_restarts', 8)), 12))
    except (ValueError, TypeError):
        nr = 8
    sds = [0.04, 0.10, 0.20]
    while len(starts) < nr:
        base = h0
        sd = sds[(len(starts) - 4) % 3] if len(starts) >= 4 else 0.10
        noise = rng.normal(0, sd, N)
        ns = 0.5 * noise + 0.25 * np.roll(noise, 1) + 0.25 * np.roll(noise, -1)
        starts.append(proj_bm(base + ns))
    dedup = [starts[0]]
    for s in starts[1:]:
        if all(np.max(np.abs(s - d)) > 1e-8 for d in dedup):
            dedup.append(s)
    starts = dedup

    try:
        mi = max(60, min(int(kwargs.get('maxiter', 120)), 250))
    except (ValueError, TypeError):
        mi = 120
    bounds = [(0.0, 1.0)] * N
    con_eq = {'type': 'eq', 'fun': mcf, 'jac': mcj}

    cands = [(best_c, best_h.copy())]
    for s in starts:
        cands.append((exact_score(s), s.copy()))

    temps = [0.02, 0.005, 0.001, 0.0002]
    for st in starts:
        cur = st.copy()
        for tau in temps:
            _tau = tau
            try:
                res = minimize(sfun, cur, jac=sjac, method='SLSQP',
                               bounds=bounds, constraints=[con_eq],
                               options={'maxiter': mi, 'ftol': 1e-10, 'disp': False})
                if res.x is not None and len(res.x) == N and np.all(np.isfinite(res.x)):
                    cur = proj_bm(res.x)
            except Exception:
                pass
            sc = exact_score(cur)
            cands.append((sc, cur.copy()))
            if sc < best_c - 1e-12:
                best_c = sc
                best_h = cur.copy()

    top3 = _pick_top3(cands)
    epi_mi = min(400, 2 * mi)
    epi_bounds = [(0.0, 1.0)] * N + [(0.0, 2.1)]
    con_eeq = {'type': 'eq', 'fun': eeq_f, 'jac': eeq_j}
    con_eiv = {'type': 'ineq', 'fun': eiv_f, 'jac': eiv_j}
    for _, eh in top3:
        z0 = np.concatenate([eh, [exact_score(eh) + 1e-6]])
        try:
            res = minimize(eobj, z0, jac=eobj_j, method='SLSQP',
                           bounds=epi_bounds, constraints=[con_eeq, con_eiv],
                           options={'maxiter': epi_mi, 'ftol': 1e-10, 'disp': False})
            if res.x is not None and len(res.x) == N + 1 and np.all(np.isfinite(res.x)):
                eh2 = proj_bm(res.x[:N])
                sc2 = exact_score(eh2)
                cands.append((sc2, eh2.copy()))
                if sc2 < best_c - 1e-12:
                    best_c = sc2
                    best_h = eh2.copy()
        except Exception:
            pass

    best_h = proj_bm(best_h)
    final_c = exact_score(best_h)
    if not (np.all(best_h >= 0) and np.all(best_h <= 1) and abs(best_h.sum() - _m) < 1e-8):
        best_h = np.full(N, 0.5)
        final_c = exact_score(best_h)
    return (best_h.astype(np.float64), final_c, N)
