                if float(np.max(np.abs(d))) < 1e-10:
                    break
                gd = float(np.dot(g, d))
                sc_y = score_exact(y, N)
                if sc_y < inc_score:
                    inc_score = sc_y
                    inc_vec = y.copy()
                if sc_y < stage_best_score:
                    stage_best_score = sc_y
                    stage_best = y.copy()
                phi_y = smooth_value(y, tau, N)
                if gd < 0.0 and phi_y <= phi + 1e-4 * gd + 1e-14:
                    h = y
                    alpha = min(1.4 * alpha, 10.0)
                    accepted = True
                    break
                alpha *= 0.5
                if alpha < 1e-8:
                    break
            if not accepted:
                break
            if float(np.max(np.abs(d))) < 1e-8:
                break
        h = stage_best.copy()
    if stage_best_score < inc_score:
        inc_score = stage_best_score
        inc_vec = stage_best.copy()
    return inc_vec, inc_score

def build_starts(baseline, N, rng):
    starts = []
    starts.append(baseline.copy())
    starts.append(project_box_sum(baseline + 0.20 * rng.standard_normal(N), N))
    starts.append(project_box_sum(rng.uniform(0.0, 1.0, size=N), N))
    R = min(6, (N - 1) // 2)
    i = np.arange(N, dtype=np.float64)
    fourier = np.full(N, 0.5)
    for r in range(1, R + 1):
        A = rng.standard_normal()
        B = rng.standard_normal()
        fourier += (0.12 / r) * (A * np.sin(2.0 * np.pi * r * i / N) + B * np.cos(2.0 * np.pi * r * i / N))
    starts.append(project_box_sum(fourier, N))
    starts.append(project_box_sum(1.0 - baseline, N))
    starts.append(np.full(N, 0.5))
    starts.append(project_box_sum(baseline + 0.08 * rng.standard_normal(N), N))
    starts.append(project_box_sum(baseline + 0.40 * rng.standard_normal(N), N))
    return starts

def run(seed=42, **kwargs):
    N = 70
    parent_arr = None
    try:
        parent_arr = np.asarray(initial_h_values, dtype=np.float64)
        if parent_arr.ndim == 1 and parent_arr.size > 0 and np.all(np.isfinite(parent_arr)):
            N = parent_arr.size
    except Exception:
        pass
    if parent_arr is None:
        try:
            np2 = int(kwargs.get('n_points', 70))
            if 1 <= np2 <= 999:
                N = np2
        except Exception:
            pass
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    rng = np.random.default_rng(seed)
    M = N // 2
    uniform = np.full(N, 0.5)
    if N <= 1:
        h = np.array([0.5])
        return h, score_exact(h, max(1, N)), 1
    baseline = None
    try:
        if parent_arr is not None and parent_arr.ndim == 1 and parent_arr.size == N and np.all(np.isfinite(parent_arr)):
            if np.all(parent_arr >= -1e-8) and np.all(parent_arr <= 1.0 + 1e-8) and abs(parent_arr.sum() - M) < 1e-6 * max(1, N):
                baseline = np.clip(parent_arr, 0.0, 1.0).copy()
            else:
                baseline = project_box_sum(parent_arr, N)
    except Exception:
        pass
    if baseline is None:
        baseline = uniform.copy()
    inc_vec = baseline.copy()
    inc_score = score_exact(inc_vec, N)
    if not np.isfinite(inc_score):
        inc_vec = uniform.copy()
        inc_score = score_exact(inc_vec, N)
    if N <= 100:
        n_starts, iters = 8, 60
    elif N <= 200:
        n_starts, iters = 6, 40
    elif N <= 500:
        n_starts, iters = 4, 20
    else:
        n_starts, iters = 3, 8
    all_starts = build_starts(baseline, N, rng)
    for s in range(min(n_starts, len(all_starts))):
        try:
            start = all_starts[s].copy()
            start = project_box_sum(start, N)
            sc_s = score_exact(start, N)
            if sc_s < inc_score:
                inc_score = sc_s
                inc_vec = start.copy()
            inc_vec, inc_score = optimize_start(start, N, inc_vec, inc_score, iters)
        except Exception:
            pass
    inc_vec = project_box_sum(inc_vec, N)
    if not np.all(np.isfinite(inc_vec)) or np.any(inc_vec < 0) or np.any(inc_vec > 1):
        inc_vec = uniform.copy()
    final_score = score_exact(inc_vec, N)
    if not np.isfinite(final_score):
        inc_vec = uniform.copy()
        final_score = score_exact(inc_vec, N)
    return inc_vec, float(final_score), int(N)
