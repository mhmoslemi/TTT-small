import numpy as np
from scipy.optimize import minimize


def _score_key(c):
    return c[1]


def exact_score(h):
    h = np.asarray(h, dtype=np.float64)
    N = len(h)
    if N < 2 or not np.all(np.isfinite(h)):
        return 999.0
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / N))


def lag_vals(h, N, d):
    return np.correlate(h, 1.0 - h, mode='full') * d


def lag_jac(h, N, d):
    J = np.zeros((2 * N - 1, N))
    for q in range(-(N - 1), N):
        ri = q + N - 1
        jlo = max(0, -q)
        jhi = min(N, N - q)
        for j in range(jlo, jhi):
            J[ri, j + q] += d * (1.0 - h[j])
            J[ri, j] -= d * h[j + q]
    return J


def project(x, N, m):
    x = np.array(x, dtype=np.float64).ravel()
    if x.shape != (N,) or not np.all(np.isfinite(x)) or np.max(np.abs(x)) > 1e6:
        return np.full(N, 0.5)
    lo, hi = x.min() - 1.0, x.max()
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(x - mid, 0.0, 1.0)) > m:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0).copy()
    delta = m - y.sum()
    if delta > 1e-14:
        for i in range(N):
            a = min(delta, 1.0 - y[i])
            y[i] += a
            delta -= a
            if delta <= 1e-14:
                break
    elif delta < -1e-14:
        for i in range(N):
            a = min(-delta, y[i])
            y[i] -= a
            delta += a
            if -delta <= 1e-14:
                break
    return y


def smooth_fun(h, N, d, tau):
    R = lag_vals(h, N, d)
    a = R.max()
    return float(a + tau * np.log(np.sum(np.exp((R - a) / tau))))


def smooth_jac(h, N, d, tau):
    R = lag_vals(h, N, d)
    a = R.max()
    e = np.exp((R - a) / tau)
    w = e / e.sum()
    return lag_jac(h, N, d).T @ w


def meq_fun(h, N, d):
    return float(h.sum() - N * 0.5)


def meq_jac(h, N, d):
    return np.ones(N)


def epi_fun(z, N, d):
    return float(z[-1])


def epi_jac(z, N, d):
    g = np.zeros(N + 1)
    g[-1] = 1.0
    return g


def epi_inq(z, N, d):
    h = z[:N]
    return z[-1] - lag_vals(h, N, d)


def epi_inq_jac(z, N, d):
    h = z[:N]
    J = lag_jac(h, N, d)
    Ji = np.zeros((2 * N - 1, N + 1))
    Ji[:, :N] = -J
    Ji[:, N] = 1.0
    return Ji


def epi_meq(z, N, d):
    return float(z[:N].sum() - N * 0.5)


def epi_meq_jac(z, N, d):

    return g
