                    hv = best_h + sigma * rng.standard_normal(n)
                proj = project_box_sum(hv, S)
                if proj is None or not is_feasible(proj, n):
                    continue
                sc = exact_score(proj, n)
                budget_used += 1
                if not np.isfinite(sc):
                    continue
                if sc < best_score - 1e-12 * max(1.0, abs(best_score)):
                    best_h = proj.copy()
                    best_score = sc
                wi = find_worst_excluding_best(pop_scores)
                pop[wi] = proj.copy()
                pop_scores[wi] = sc
            restart_level += 1
            Fmean = min(0.95, Fmean + 0.10)
            CRmean = min(0.95, CRmean + 0.05)
            no_improve = 0

    final_h = project_box_sum(best_h, S)
    if final_h is None or not is_feasible(final_h, n):
        final_h = fb.copy()
        final_score = exact_score(final_h, n)
    else:
        final_score = exact_score(final_h, n)
    fb_proj = project_box_sum(fb, S)
    if fb_proj is not None and is_feasible(fb_proj, n):
        fb_sc = exact_score(fb_proj, n)
        if fb_sc < final_score:
            final_h = fb_proj
            final_score = fb_sc
    return final_h, float(final_score), int(n)
