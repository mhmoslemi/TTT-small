import numpy as np
from scipy.optimize import minimize


def project_box_sum(v, S):
    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if resid > 1e-12:
        for i in range(n):
            if y[i] < 1.0 - 1e-14:
                a = min(resid, 1.0 - y[i])
                y[i] += a
                resid -= a
                if resid <= 1e-13:
                    break
    elif resid < -1e-12:
        for i in range(n):
            if y[i] > 1e-14:
                a = min(-resid, y[i])
                y[i] -= a
                resid += a
                if resid >= -1e-13:
                    break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    if abs(np.sum(y) - S) > 1e-9 * max(1, n):
        return None
    return y


def exact_score(h):
    if h is None or not np.all(np.isfinite(h)):
        return float('inf')
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def is_feasible(h, n):
    if h is None or h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-9 * max(1, n):
        return False
    return True


def get_parent():
    if 'initial_h_values' not in globals():
        return None
    raw = globals()['initial_h_values']
    if raw is None:
        return None
    p = np.array(raw, dtype=np.float64).ravel()
    if p.size < 2 or p.size > 999:
        return None
    if not np.all(np.isfinite(p)):
        return None
    if np.any(p < -1e-4) or np.any(p > 1.0 + 1e-4):
        return None
    n = p.size
    if abs(np.sum(p) - n / 2.0) > 1e-4:
        return None
    p = np.clip(p, 0.0, 1.0)
    pr = project_box_sum(p, n / 2.0)
    return pr if pr is not None else p


def slsqp_obj(z):
    return z[-1]


def slsqp_obj_jac(z):
    j = np.zeros(z.size)
    j[-1] = 1.0
    return j


def slsqp_ineq(z):
    n = z.size - 1
    h, t = z[:n], z[n]
    return t - np.correlate(h, 1.0 - h, mode='full') / n


def slsqp_ineq_jac(z):
    n = z.size - 1
    h = z[:n]
    J = np.zeros((2 * n - 1, n + 1))
    for idx in range(2 * n - 1):
        l = idx - (n - 1)
        a, b = max(0, l), min(n - 1, n - 1 + l)
        if a <= b:
            J[idx, a:b + 1] -= (1.0 - h[a - l:b + 1 - l]) / n
        c, d = max(0, -l), min(n - 1, n - 1 - l)
        if c <= d:
            J[idx, c:d + 1] += h[c + l:d + 1 + l] / n
        J[idx, n] = 1.0
    return J


def slsqp_eq(z):
    n = z.size - 1
    return np.sum(z[:n]) / n - 0.5


def slsqp_eq_jac(z):
    n = z.size - 1
    j = np.zeros(n + 1)
    j[:n] = 1.0 / n
    return j


def optimize_start(h0, n, S):
    lags = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = min(1.0, np.max(lags) / n + 1e-4)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac},
    ]
    res = minimize(slsqp_obj, z0, jac=slsqp_obj_jac, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': 300, 'ftol': 1e-10, 'disp': False})
    if res.x is None or res.x.size != n + 1:
        return None
    return np.clip(res.x[:n], 0.0, 1.0)


def run(seed=42, **kwargs):
    parent = get_parent()
    if parent is not None:
        n = len(parent)
    else:
        n = 45
        parent = np.full(n, 0.5)
    S = n / 2.0
    rng = np.random.default_rng(seed)

    starts = [parent.copy(), parent[::-1].copy()]
    for sd in [0.01, 0.03, 0.06, 0.12, 0.20, 0.05]:
        p = project_box_sum(parent + sd * rng.standard_normal(n), S)
        if p is not None:
            starts.append(p)
    for _ in range(4):
        p = project_box_sum(rng.uniform(0, 1, n), S)
        if p is not None:
            starts.append(p)
    for A, m, ph in [(0.25, 1, 0.37), (0.35, 2, 0.74), (0.45, 3, 1.11)]:
        u = 0.5 + A * np.cos(2.0 * np.pi * m * np.arange(n) / n + ph)
        p = project_box_sum(u, S)
        if p is not None:
            starts.append(p)

    best_h = parent.copy()
    best_score = exact_score(best_h)

    for h0 in starts:
        if not is_feasible(h0, n):
            p = project_box_sum(h0, S)
            if p is None:
                continue
            h0 = p
        try:
            h_new = optimize_start(h0, n, S)
        except Exception:
            continue
        if h_new is None:
            continue
        h_new = project_box_sum(h_new, S)
        if h_new is None or not is_feasible(h_new, n):
            continue
        sc = exact_score(h_new)
        if sc < best_score - 1e-14:
            best_h = h_new.copy()
            best_score = sc

    sigma = 0.05
    for _ in range(4):
        h_pert = project_box_sum(best_h + sigma * rng.standard_normal(n), S)
        if h_pert is None:
            break
        try:
            h_new = optimize_start(h_pert, n, S)
        except Exception:
            h_new = None
        improved = False
        if h_new is not None:
            h_new = project_box_sum(h_new, S)
            if h_new is not None and is_feasible(h_new, n):
                sc = exact_score(h_new)
                if sc < best_score - 1e-12:
                    best_h = h_new.copy()
                    best_score = sc
                    improved = True
        sigma = max(0.01, 0.7 * sigma) if improved else min(0.25, 1.5 * sigma)

    best_h = project_box_sum(best_h, S)
    if best_h is None or not is_feasible(best_h, n):
        best_h = parent.copy()
    c5 = exact_score(best_h)
    return best_h, c5, n
