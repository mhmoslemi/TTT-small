        p = clip_proj(base + 0.2 * rng.randn(n), n)
        pool.append(p if p is not None else base.copy())
    return pool[:5]

def run(seed=42, **kwargs):
    n = 47
    base = None
    try:
        ih = np.asarray(initial_h_values, dtype=np.float64)
        if ih.ndim == 1 and len(ih) > 0:
            n = int(len(ih))
            if np.all(np.isfinite(ih)) and np.all(ih >= -1e-8) and np.all(ih <= 1.0 + 1e-8):
                base = ih.copy() if feasible_check(ih, n) else project_capped_simplex(ih, n)
            if base is None:
                base = uniform_fallback(n)
    except Exception:
        pass
    if base is None:
        npk = kwargs.get('n_points', None)
        if isinstance(npk, (int, np.integer)) and not isinstance(npk, bool) and 1 <= npk <= 999:
            n = int(npk)
        base = uniform_fallback(n)
    fallback = base.copy()
    if n == 1:
        h = uniform_fallback(1)
        return (h, exact_score(h, 1), 1)
    if n <= 64:
        pool = build_pool(base, n, seed)
    else:
        pool = [base.copy()]
    incumbent = None
    incumbent_score = float('inf')
    for p in pool:
        sc = exact_score(p, n)
        if sc < incumbent_score - 1e-12:
            incumbent = p.copy()
            incumbent_score = sc
    if n <= 48:
        ms, msw = min(5, len(pool)), 2
    elif n <= 64:
        ms, msw = min(2, len(pool)), 1
    else:
        ms, msw = 0, 0
    if ms > 0:
        dirs_arr = haar_basis(n)
        nd = dirs_arr.shape[0]
        for si in range(ms):
            x = pool[si].copy()
            lsc = exact_score(x, n)
            for sw in range(msw):
                if nd == 0:
                    break
                order = np.random.RandomState(seed + si * 1000 + sw).permutation(nd)
                imp = False
                for di in order:
                    try:
                        v = dirs_arr[di]
                        fi = feasible_interval(x, v)
                        if fi is None:
                            continue
                        lo, hi = fi
                        c_arr = (2.0 / n) * np.correlate(x, 1.0 - x, mode='full')
                        b_arr = (2.0 / n) * (np.correlate(v, 1.0 - x, mode='full') - np.correlate(x, v, mode='full'))
                        a_arr = -(2.0 / n) * np.correlate(v, v, mode='full')
                        if not (np.all(np.isfinite(a_arr)) and np.all(np.isfinite(b_arr)) and np.all(np.isfinite(c_arr))):
                            continue
                        cands = line_candidates(a_arr, b_arr, c_arr, lo, hi)
                        scores = []
                        for idx in range(0, len(cands), 512):
                            cc = np.array(cands[idx:idx + 512], dtype=np.float64)
                            vals = a_arr[None, :] * (cc[:, None] ** 2) + b_arr[None, :] * cc[:, None] + c_arr[None, :]
                            env = np.max(vals, axis=1)
                            for ci in range(len(cc)):
                                if np.isfinite(env[ci]):
                                    scores.append((float(env[ci]), cands[idx + ci]))
                        scores.sort()
                        best_xs, best_sc = x, lsc
                        for (_, t) in scores[:8]:
                            y = x + t * v
                            if not np.all(np.isfinite(y)):
                                continue
                            if not feasible_check(y, n):
                                y = project_capped_simplex(y, n)
                                if y is None:
                                    continue
                            sc = exact_score(y, n)
                            if sc < best_sc - 1e-12:
                                best_sc = sc
                                best_xs = y
                        if best_sc < lsc - 1e-12:
                            x = best_xs
                            lsc = best_sc
                            imp = True
                            if lsc < incumbent_score - 1e-12:
                                incumbent = x.copy()
                                incumbent_score = lsc
                    except Exception:
                        continue
                if not imp:
                    break
    final = incumbent if incumbent is not None else fallback
    proj = project_capped_simplex(final, n)
    if proj is not None:
        final = proj
    if not feasible_check(final, n):
        final = fallback
    fs = exact_score(final, n)
    fsf = exact_score(fallback, n)
    if fs > fsf - 1e-12:
        final, fs = fallback, fsf
    return (final, fs, n)
