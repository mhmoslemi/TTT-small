import numpy as np
import math

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = float(np.max(corr) * dx)
    if not math.isfinite(s):
        return float('inf')
    return s

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-15) or np.any(h > 1 + 1e-15):
        return False
    S = n / 2.0
    if abs(np.sum(h) - S) > 1e-10:
        return False
    if abs(math.fsum(float(x) for x in h) - S) > 1e-10:
        return False
    return True

def capped_simplex(v, n):
    S = n / 2.0
    v = np.clip(v.astype(np.float64), -1e6, 1e6)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    # Correct residual mass
    resid = S - math.fsum(float(x) for x in h)
    if resid > 0:
        cap = 1.0 - h
        idx = np.argsort(-cap)
        for k in idx:
            add = min(resid, cap[k])
            if add <= 0:
                break
            h[k] += add
            resid -= add
            if abs(resid) < 1e-15:
                break
    elif resid < 0:
        idx = np.argsort(-h)
        for k in idx:
            rem = min(-resid, h[k])
            if rem <= 0:
                break
            h[k] -= rem
            resid += rem
            if abs(resid) < 1e-15:
                break
    h = np.clip(h, 0.0, 1.0)
    if is_feasible(h, n):
        return h
    return None

def compute_qG(h, n):
    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        k = np.arange(n - d)
        hp = k + d
        q[2 * d - 1] = dx * np.sum(h[hp] * (1.0 - h[k]))
        q[2 * d] = dx * np.sum(h[k] * (1.0 - h[hp]))
        G[2 * d - 1, hp] += dx * (1.0 - h[k])
        G[2 * d - 1, k] += dx * (-h[hp])
        G[2 * d, k] += dx * (1.0 - h[hp])
        G[2 * d, hp] += dx * (-h[k])
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None, None
    return q, G

def quad_roots(A, B, C, lo, hi):
    tol = 1e-14 * max(1.0, abs(A), abs(B), abs(C))
    if abs(A) <= tol:
        if abs(B) <= tol:
            return []
        r = -C / B
        if lo - 1e-10 <= r <= hi + 1e-10:
            return [max(lo, min(hi, r))]
        return []
    disc = B * B - 4.0 * A * C
    eps = np.finfo(np.float64).eps
    if disc < -64.0 * eps * (B * B + abs(4.0 * A * C) + 1.0):
        return []
    if disc < 0:
        disc = 0.0
    sqrt_d = math.sqrt(disc)
    z = -0.5 * (B + math.copysign(sqrt_d, B) if B != 0 else sqrt_d)
    roots = []
    if z != 0.0:
        r1 = z / A
        r2 = C / z
    else:
        r1 = -B / (2.0 * A)
        r2 = r1
    for r in [r1, r2]:
        if math.isfinite(r) and lo - 1e-10 <= r <= hi + 1e-10:
            roots.append(max(lo, min(hi, r)))
    return roots

def build_hull(lines, lo, hi):
    # lines: list of (slope, intercept)
    if not lines:
        return []
    # Sort by slope, for equal slopes keep max intercept
    from operator import itemgetter
    # Use lexsort: primary=slope asc, secondary=intercept desc
    slopes = np.array([l[0] for l in lines])
    intercepts = np.array([l[1] for l in lines])
    order = np.lexsort((-intercepts, slopes))
    # Deduplicate: for same slope, keep first (max intercept)
    filtered = []
    prev_slope = None
    for idx in order:
        s, c = slopes[idx], intercepts[idx]
        if prev_slope is not None and abs(s - prev_slope) < 1e-14 * max(1.0, abs(s), abs(prev_slope)):
            continue
        filtered.append((float(s), float(c)))
        prev_slope = s
    if not filtered:
        return []
    # Build upper envelope
    hull = []  # each entry: (slope, intercept, start_x)
    hull.append((filtered[0][0], filtered[0][1], -1e18))
    for s_new, c_new in filtered[1:]:
        s_old, c_old, _ = hull[-1]
        x_cross = (c_old - c_new) / (s_new - s_old)
        while len(hull) > 1 and x_cross <= hull[-1][2]:
            hull.pop()
            s_old, c_old, _ = hull[-1]
            x_cross = (c_old - c_new) / (s_new - s_old)
        hull.append((s_new, c_new, x_cross))
    # Clip to [lo, hi]
    segments = []
    for s, c, x0 in hull:
        seg_lo = max(x0, lo)
        seg_hi = hi
        if seg_lo < seg_hi:
            segments.append((s, c, seg_lo, seg_hi))
    # Also return breakpoints within [lo, hi]
    breakpoints = []
    for seg in segments:
        breakpoints.append(seg[2])
    return segments, breakpoints

def pair_line_search(h, i, j, n, current_score):
    dx = 2.0 / n
    s = j - i
    q, G = compute_qG(h, n)
    if q is None:
        return None, current_score
    M = 2 * n - 1
    a = np.zeros(M)
    b = np.zeros(M)
    c = q.copy()
    b[:] = G[:, i] - G[:, j]
    a[0] = -2.0 * dx
    a[2 * s - 1] = dx
    a[2 * s] = dx
    # Feasible interval for t
    hi_t = min(1.0 - h[i], h[j])
    lo_t = max(-h[i], h[j] - 1.0)
    if lo_t >= hi_t:
        return None, current_score
    curved_idx = [0, 2 * s - 1, 2 * s]
    affine_idx = [l for l in range(M) if l not in curved_idx]
    # Build affine hull
    affine_lines = [(b[l], c[l]) for l in affine_idx]
    hull_result = build_hull(affine_lines, lo_t, hi_t)
    if hull_result is None:
        return None, current_score
    segments, bps = hull_result
    # Candidate t values
    cands = [lo_t, hi_t, 0.0]
    cands.extend(bps)
    # Quadratic vs affine roots
    for li in curved_idx:
        for (ms, cs, seg_lo, seg_hi) in segments:
            A = a[li]
            B = b[li] - ms
            C = c[li] - cs
            roots = quad_roots(A, B, C, seg_lo, seg_hi)
            cands.extend(roots)
    # Pairwise quadratic roots
    for x in range(3):
        for y in range(x + 1, 3):
            A = a[curved_idx[x]] - a[curved_idx[y]]
            B = b[curved_idx[x]] - b[curved_idx[y]]
            C = c[curved_idx[x]] - c[curved_idx[y]]
            roots = quad_roots(A, B, C, lo_t, hi_t)
            cands.extend(roots)
    # Vertices of convex quadratics
    for li in curved_idx:
        if a[li] > 1e-15:
            tv = -b[li] / (2.0 * a[li])
            if lo_t <= tv <= hi_t:
                cands.append(tv)
    # Deduplicate and sort
    cands = [t for t in cands if math.isfinite(t) and lo_t - 1e-10 <= t <= hi_t + 1e-10]
    cands = [max(lo_t, min(hi_t, t)) for t in cands]
    cands.sort()
    deduped = []
    for t in cands:
        if not deduped or abs(t - deduped[-1]) > 1e-12:
            deduped.append(t)
    if not deduped:
        return None, current_score
    # Score candidates by polynomial max
    def poly_max(t):
        vals = a * t * t + b * t + c
        return float(np.max(vals))
    # Also need affine envelope value
    def full_max(t):
        quad_max = float(np.max(a * t * t + b * t + c))
        aff_vals = np.array([b[l] * t + c[l] for l in affine_idx])
        aff_max = float(np.max(aff_vals))
        return max(quad_max, aff_max)
    scored = [(full_max(t), t) for t in deduped]
    scored.sort(key=lambda x: x[0])
    # Evaluate top candidates with exact score
    best_h = None
    best_score = current_score
    evaluated = 0
    for (pm, t) in scored:
        if evaluated >= 6:
            break
        # Construct trial
        pair_sum = h[i] + h[j]
        new_i = h[i] + t
        new_i = max(0.0, min(1.0, new_i))
        new_i = max(max(0.0, pair_sum - 1.0), min(min(1.0, pair_sum), new_i))
        new_j = pair_sum - new_i
        trial = h.copy()
        trial[i] = new_i
        trial[j] = new_j
        if not is_feasible(trial, n):
            trial = capped_simplex(trial, n)
            if trial is None:
                evaluated += 1
                continue
        sc = exact_score(trial)
        evaluated += 1
        if sc < best_score - 1e-12:
            best_score = sc
            best_h = trial
    if best_h is not None:
        return best_h, best_score
    return None, current_score

def add_unique(starts, h, n):
    if not is_feasible(h, n):
        return
    for s in starts:
        if np.max(np.abs(h - s)) < 1e-10:
            return
    starts.append(h.copy())

def get_pair_schedule(n, rng):
    if n <= 48:
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    elif n <= 64:
        rs = [1, 2, max(1, n // 4), max(1, n // 2)]
        pairs = set()
        for i in range(n):
            for r in rs:
                j = (i + r) % n
                if i == j:
                    continue
                pairs.add((min(i, j), max(i, j)))
        pairs = sorted(pairs)
    elif n <= 128:
        rs = [1, max(1, n // 2)]
        pairs = set()
        for i in range(n):
            for r in rs:
                j = (i + r) % n
                if i == j:
                    continue
                pairs.add((min(i, j), max(i, j)))
        pairs = sorted(pairs)
    else:
        return []
    order = rng.permutation(len(pairs))
    return [pairs[k] for k in order]

def local_search(h, n, rng, num_sweeps):
    cur = h.copy()
    cur_score = exact_score(cur)
    for sweep in range(num_sweeps):
        pairs = get_pair_schedule(n, rng)
        improved = False
        for (i, j) in pairs:
            new_h, new_sc = pair_line_search(cur, i, j, n, cur_score)
            if new_h is not None:
                cur = new_h
                cur_score = new_sc
                improved = True
        if not improved:
            break
    return cur, cur_score

            if abs(resid) < 1e-15:
                break
    h = np.clip(h, 0.0, 1.0)
    return h if is_feasible(h, n) else None

def compute_qG(h, n):
    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        k = np.arange(n - d)
        hp = k + d
        q[2*d-1] = dx * np.sum(h[hp] * (1.0 - h[k]))
        q[2*d] = dx * np.sum(h[k] * (1.0 - h[hp]))
        G[2*d-1, hp] += dx * (1.0 - h[k])
        G[2*d-1, k] -= dx * h[hp]
        G[2*d, k] += dx * (1.0 - h[hp])
        G[2*d, hp] -= dx * h[k]
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None, None
    return q, G

def quad_roots(A, B, C, lo, hi):
    if not all(math.isfinite(x) for x in [A, B, C]):
        return []
    tol = 1e-14 * max(1.0, abs(A), abs(B), abs(C))
    if abs(A) <= tol:
        if abs(B) <= tol:
            return []
        r = -C / B
        return [max(lo, min(hi, r))] if lo - 1e-10 <= r <= hi + 1e-10 else []
    disc = B*B - 4.0*A*C
    eps = np.finfo(np.float64).eps
    if disc < -64.0*eps*(B*B + abs(4.0*A*C) + 1.0):
        return []
    disc = max(0.0, disc)
    sd = math.sqrt(disc)
    z = -0.5*(B + sd) if B >= 0 else -0.5*(B - sd)
    roots = []
    if abs(z) > 1e-30:
        r1, r2 = z/A, C/z
    else:
        r1 = r2 = -B/(2.0*A)
    for r in [r1, r2]:
        if math.isfinite(r) and lo - 1e-10 <= r <= hi + 1e-10:
            roots.append(max(lo, min(hi, r)))
    return roots

def build_affine_hull(b_arr, c_arr, idx_list, lo, hi):
    if not idx_list:
        return [], []
    slopes = np.array([b_arr[i] for i in idx_list])
    intercepts = np.array([c_arr[i] for i in idx_list])
    order = np.lexsort((-intercepts, slopes))
    fs, fc = [], []
    ps = None
    for idx in order:
        s2, c2 = float(slopes[idx]), float(intercepts[idx])
        if ps is not None and abs(s2-ps) < 1e-14*max(1.0, abs(s2), abs(ps)):
            continue
        fs.append(s2); fc.append(c2); ps = s2
    if not fs:
        return [], []
    hs, hc, hstart = [fs[0]], [fc[0]], [-1e18]
    for i2 in range(1, len(fs)):
        sn, cn = fs[i2], fc[i2]
        xc = (hc[-1] - cn) / (sn - hs[-1])
        while len(hs) > 1 and xc <= hstart[-1]:
            hs.pop(); hc.pop(); hstart.pop()
            xc = (hc[-1] - cn) / (sn - hs[-1])
        hs.append(sn); hc.append(cn); hstart.append(xc)
    segs, bps = [], []
    for i2 in range(len(hs)):
        sl = max(hstart[i2], lo)
        if sl < hi:
            segs.append((hs[i2], hc[i2], sl, hi))
            if hstart[i2] >= lo:
                bps.append(hstart[i2])
    return segs, bps

def pair_line_search(h, i, j, n, cur_sc):
    dx = 2.0 / n
    s = j - i
    q, G = compute_qG(h, n)
    if q is None:
        return None, cur_sc
    M = 2*n - 1
    a = np.zeros(M)
    b = G[:, i] - G[:, j]
    c = q.copy()
    a[0] = -2.0*dx
    a[2*s-1] = dx
    a[2*s] = dx
    lo_t = max(-h[i], h[j]-1.0)
    hi_t = min(1.0-h[i], h[j])
    if hi_t - lo_t < 1e-15:
        return None, cur_sc
    curved = [0, 2*s-1, 2*s]
    aff = [l for l in range(M) if l not in curved]
    segs, bps = build_affine_hull(b, c, aff, lo_t, hi_t)
    cands = [lo_t, hi_t, 0.0] + bps
    for li in curved:
        for (ms, cs, sl, sh) in segs:
            cands.extend(quad_roots(a[li], b[li]-ms, c[li]-cs, sl, sh))
    for x in range(3):
        for y in range(x+1, 3):
            cands.extend(quad_roots(a[curved[x]]-a[curved[y]], b[curved[x]]-b[curved[y]], c[curved[x]]-c[curved[y]], lo_t, hi_t))
    for li in curved:
        if a[li] > 1e-15:
            tv = -b[li]/(2.0*a[li])
            if lo_t <= tv <= hi_t:
                cands.append(tv)
    cands = sorted(set(round(t, 12) for t in cands if math.isfinite(t) and lo_t-1e-10 <= t <= hi_t+1e-10))
    if not cands:
        return None, cur_sc
    scored = []
    for t in cands:
        qm = float(np.max(a*t*t + b*t + c))
        am = float(np.max(b[aff]*t + c[aff])) if aff else 0.0
        scored.append((max(qm, am), t))
    scored.sort()
    best_h, best_sc = None, cur_sc
    ev = 0
    for (_, t) in scored:
        if ev >= 6:
            break
        psum = h[i] + h[j]
        ni = max(max(0.0, psum-1.0), min(min(1.0, psum), h[i]+t))
        trial = h.copy()
        trial[i] = ni
        trial[j] = psum - ni
        if not is_feasible(trial, n):
            trial = capped_simplex(trial, n)
            if trial is None:
                ev += 1
                continue
        sc = exact_score(trial)
        ev += 1
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_h = trial
    return (best_h, best_sc) if best_h is not None else (None, cur_sc)

def add_unique(starts, h, n):
    if not is_feasible(h, n):
        return
    for s2 in starts:
        if float(np.max(np.abs(h - s2))) < 1e-10:
            return
    starts.append(h.copy())

def get_pairs(n, rng):
    if n <= 48:
        pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
    elif n <= 128:
        rs = [1, 2, max(1, n//4), max(1, n//2)] if n <= 64 else [1, max(1, n//2)]
        ps = set()
        for i in range(n):
            for r in rs:
                j = (i + r) % n
                if i != j:
                    ps.add((min(i, j), max(i, j)))
        pairs = sorted(ps)
    else:
        return []
    perm = rng.permutation(len(pairs))
    return [pairs[k] for k in perm]

def local_search(h0, n, rng, nsweeps):
    cur = h0.copy()
    csc = exact_score(cur)
    for _ in range(nsweeps):
        pairs = get_pairs(n, rng)
        improved = False
        for (i, j) in pairs:
            nh, nsc = pair_line_search(cur, i, j, n, csc)
            if nh is not None:
                cur, csc, improved = nh, nsc, True
        if not improved:
            break
    return cur, csc

def run(seed=42, **kwargs):
    try:
        try:
            seed = int(seed)
        except (ValueError, TypeError):
            seed = 42
        rng = np.random.default_rng(seed)
        n = 42
        if 'n_points' in kwargs:
            np_ = kwargs['n_points']
            if not isinstance(np_, bool) and 2 <= int(np_) <= 512:
                try:
                    n = int(np_)
                except (ValueError, TypeError):
                    pass
        parent = None
        try:
            pv = np.array(initial_h_values, dtype=np.float64).flatten()
            if pv.size >= 2:
                n = pv.size
                if np.all(np.isfinite(pv)):
                    S = n / 2.0
                    if np.all(pv >= -1e-12) and np.all(pv <= 1.0+1e-12) and abs(np.sum(pv)-S) <= 1e-12:
                        parent = pv.copy()
                    else:
                        parent = capped_simplex(pv, n)
        except (NameError, AttributeError, ValueError, TypeError):
            parent = None
        const_h = np.full(n, 0.5)
        const_sc = exact_score(const_h)
        ginc, gsc = const_h.copy(), const_sc
        pfb = None
        if parent is not None:
            psc = exact_score(parent)
            if psc < gsc - 1e-12:
                ginc, gsc = parent.copy(), psc
            pfb = parent.copy()
        starts = []
        if n <= 128:
            if parent is not None:
                add_unique(starts, parent, n)
            add_unique(starts, const_h, n)
            blk = np.zeros(n)
            blk[:n//2] = 1.0
            if n % 2 == 1:
                blk[n//2] = 0.5
            add_unique(starts, blk, n)
            alt = np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n)])
            ap = capped_simplex(alt, n)
            if ap is not None:
                add_unique(starts, ap, n)
            if n <= 48:
                sp = capped_simplex(0.5 + 0.45*np.sin(2.0*np.pi*np.arange(n)/n), n)
                if sp is not None:
                    add_unique(starts, sp, n)
                if parent is not None:
                    p1 = capped_simplex(parent + 0.12*rng.standard_normal(n), n)
                    if p1 is not None:
                        add_unique(starts, p1, n)
                    p2 = capped_simplex(parent + 0.35*rng.standard_normal(n), n)
                    if p2 is not None:
                        add_unique(starts, p2, n)
                nr = 2 if parent is not None else 4
                for _ in range(nr):
                    rp = capped_simplex(rng.uniform(0, 1, n), n)
                    if rp is not None:
                        add_unique(starts, rp, n)
            if 49 <= n <= 128:
                starts = starts[:4]
        if not starts:
            starts = [const_h.copy()]
        nsweeps = 2 if n <= 64 else 1
        for st in starts:
            try:
                rh, rs2 = local_search(st, n, rng, nsweeps)
                if is_feasible(rh, n) and rs2 < gsc - 1e-12:
                    ginc, gsc = rh, rs2
            except Exception:
                pass
        best_h, best_s = ginc, gsc
        if pfb is not None:
            pf2 = exact_score(pfb)
            if pf2 < best_s - 1e-12:
                best_h, best_s = pfb, pf2
        if const_sc < best_s - 1e-12:
            best_h, best_s = const_h, const_sc
        if not is_feasible(best_h, n):
            best_h = capped_simplex(best_h, n)
            if best_h is None:
                best_h, best_s = const_h, const_sc
            else:
                best_s = exact_score(best_h)
        return best_h.astype(np.float64), float(best_s), int(n)
    except Exception:
        h_f = np.full(42, 0.5)
        return h_f, float(exact_score(h_f)), 42
