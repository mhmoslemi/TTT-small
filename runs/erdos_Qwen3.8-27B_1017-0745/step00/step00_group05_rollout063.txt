import numpy as np
from scipy.optimize import minimize

def capped_simplex_project(v, target):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - float(np.sum(y))
    if abs(r) > 1e-13:
        if r > 0:
            cap = 1.0 - y
            tc = float(np.sum(cap))
            if tc > 0:
                y = y + r * cap / tc
        else:
            cap = y.copy()
            tc = float(np.sum(cap))
            if tc > 0:
                y = y + r * cap / tc
        y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(float(np.sum(y)) - target) > 1e-10:
        return None
    if np.min(y) < -1e-12 or np.max(y) > 1.0 + 1e-12:
        return None
    return y

def check_feasible(h, target):
    return (h.ndim == 1 and np.all(np.isfinite(h))
            and float(np.min(h)) >= -1e-12 and float(np.max(h)) <= 1.0 + 1e-12
            and abs(float(np.sum(h)) - target) <= 1e-10)

def exact_score(h, dx):
    if h.ndim != 1 or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def smooth_val_grad(h, args):
    tau, dx, lags, n = args
    s = dx * np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(s))
    e = np.exp((s - m) / tau)
    Z = float(np.sum(e))
    F = m + tau * np.log(Z)
    p = e / Z
    grad = np.zeros(n)
    for ki in range(len(lags)):
        if p[ki] < 1e-16:
            continue
        t = int(lags[ki])
        c = p[ki] * dx
        a, b = max(0, t), min(n, n + t)
        if b > a:
            idx = np.arange(a, b)
            grad[idx] += c * (1.0 - h[idx - t])
        a, b = max(0, -t), min(n, n - t)
        if b > a:
            idx = np.arange(a, b)
            grad[idx] -= c * h[idx + t]
    return F, grad

def eq_fun(h):
    return [float(np.sum(h)) - h.size / 2.0]

def eq_jac(h):
    return np.ones(h.size)

def solve_stage(h0, n, dx, lags, tau):
    try:
        res = minimize(smooth_val_grad, h0.copy(),
                       args=(tau, dx, lags, n), jac=True, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 200, 'ftol': 1e-10, 'disp': False})
        x = res.x
        if x is not None and x.shape == (n,) and np.all(np.isfinite(x)):
            return x
    except Exception:
        pass
    return None

def run_continuation(h_start, n, dx, lags, target, tau_sched):
    h_cur = h_start.copy()
    all_pts = [h_cur.copy()]
    for tau in tau_sched:
        raw = solve_stage(h_cur, n, dx, lags, tau)
        if raw is not None:

            if proj is not None and check_feasible(proj, target):
                all_pts.append(proj.copy())
                Fn, _ = smooth_val_grad(proj, (tau, dx, lags, n))
                Fo, _ = smooth_val_grad(h_cur, (tau, dx, lags, n))
                if np.isfinite(Fn) and Fn <= Fo + 1e-10:
                    h_cur = proj
    all_pts.append(h_cur.copy())
    return all_pts
