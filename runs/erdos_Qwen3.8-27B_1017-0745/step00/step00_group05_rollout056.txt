            cap = y.copy()
            avail = cap.sum()
            if avail > 1e-15:
                y = y + r * cap / avail
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(y.sum() - target) > 1e-10:
        return None
    return y


def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return np.inf
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_fj(x, tau, dx, n):
    s = dx * np.correlate(x, 1.0 - x, mode='full')
    m = s.max()
    e = np.exp((s - m) / tau)
    F = m + tau * np.log(e.sum())
    p = e / e.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        t = k - (n - 1)
        if t >= 0:
            js = np.arange(t, n)
            grad[js] += p[k] * dx * (1.0 - x[js - t])
            js2 = np.arange(0, n - t)
            grad[js2] -= p[k] * dx * x[js2 + t]
        else:
            u = -t
            js = np.arange(0, n - u)
            grad[js] += p[k] * dx * (1.0 - x[js + u])
            js2 = np.arange(u, n)
            grad[js2] -= p[k] * dx * x[js2 - u]
    return F, grad


def mass_eq(x, target):
    return x.sum() - target


def mass_eq_jac(x, target):
    return np.ones(len(x))


def do_optimize(h0, n, dx, target, bounds, tau_sched):
    current = h0.copy()
    for tau in tau_sched:
        try:
            cons = [{'type': 'eq', 'fun': mass_eq, 'jac': mass_eq_jac, 'args': (target,)}]
            res = minimize(smooth_fj, current, jac=True, args=(tau, dx, n),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-10, 'disp': False})
            if res.x is not None and res.x.shape == (n,) and np.all(np.isfinite(res.x)):
                proj = capped_simplex_project(res.x, target)
                if proj is not None:
                    F_new, _ = smooth_fj(proj, tau, dx, n)
                    F_cur, _ = smooth_fj(current, tau, dx, n)
                    if np.isfinite(F_new) and F_new <= F_cur + 1e-10:
                        current = proj
        except Exception:
            pass
    return current


def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    n = 70
    parent = None
    try:
        iv = initial_h_values
        if isinstance(iv, np.ndarray) and iv.ndim == 1 and 2 <= len(iv) <= 999:
            n = len(iv)
            ivc = iv.astype(np.float64).copy()
            if (np.all(np.isfinite(ivc)) and ivc.min() >= -1e-10
                    and ivc.max() <= 1.0 + 1e-10
                    and abs(ivc.sum() - n / 2.0) <= 1e-8):
                parent = capped_simplex_project(ivc, n / 2.0)
    except (NameError, Exception):
        pass

    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    tau_sched = [0.02, 0.005, 0.001, 2e-4, 5e-5, 1e-5]

    incumbent = None
    incumbent_score = np.inf
    if parent is not None:
        incumbent = parent.copy()
        incumbent_score = exact_score(parent, dx)

    starts = []
    if parent is not None:
        starts.append(parent.copy())
        starts.append(parent[::-1].copy())
        for sigma in [0.04, 0.10, 0.22, 0.40]:
            pert = parent + sigma * rng.standard_normal(n)
            p = capped_simplex_project(pert, target)
            if p is not None:
                starts.append(p)
    starts.append(np.full(n, 0.5))
    p = capped_simplex_project(rng.uniform(0.0, 1.0, n), target)
    if p is not None:
        starts.append(p)
    p = capped_simplex_project(rng.beta(0.5, 0.5, n), target)
    if p is not None:
        starts.append(p)
    p = capped_simplex_project(rng.beta(0.5, 0.5, n), target)
    if p is not None:
        starts.append(p)

    for s in starts:
        opt = do_optimize(s, n, dx, target, bounds, tau_sched)
        sc = exact_score(opt, dx)
        if np.isfinite(sc) and sc < incumbent_score - 1e-12:
            incumbent_score = sc
            incumbent = opt.copy()

    center = incumbent if incumbent is not None else np.full(n, 0.5)
    for sigma in [0.02, 0.08, 0.20]:
        pert = center + sigma * rng.standard_normal(n)
        p = capped_simplex_project(pert, target)
        if p is not None:
            opt = do_optimize(p, n, dx, target, bounds, tau_sched)
            sc = exact_score(opt, dx)
            if np.isfinite(sc) and sc < incumbent_score - 1e-12:
                incumbent_score = sc
                incumbent = opt.copy()
        center = incumbent if incumbent is not None else np.full(n, 0.5)

    if incumbent is None:
        incumbent = np.full(n, 0.5)
    final = capped_simplex_project(incumbent, target)
    if final is None:
        final = np.full(n, 0.5)
    final_score = exact_score(final, dx)
    if not np.isfinite(final_score):
        final = np.full(n, 0.5)
        final_score = exact_score(final, dx)
    return (final, final_score, n)
