import numpy as np
import math

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(corr * dx))
    return val if math.isfinite(val) else float('inf')

def is_feasible(h, n):
    if h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-15) or np.any(h > 1.0 + 1e-15):
        return False
    S = n / 2.0
    if abs(float(np.sum(h)) - S) > 1e-10:
        return False
    if abs(math.fsum(float(x) for x in h) - S) > 1e-10:
        return False
    return True

def project_capped_simplex(v, n):
    if v.ndim != 1 or len(v) != n or not np.all(np.isfinite(v)):
        return None
    v = np.clip(v.astype(np.float64), -1e6, 1e6)

            lo = mid
        else:
            hi = mid

    for _ in range(3):
        res = S - math.fsum(float(x) for x in h)
        if abs(res) < 1e-14:
            break

            idx = np.where(slack > 1e-14)[0]

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    # Lag 0
    q[0] = dx * float(np.sum(h * (1.0 - h)))
    G[0, :] = dx * (1.0 - 2.0 * h)
    # Positive lags d = 1..n-1
    for d in range(1, n):
        idx_pos = 2 * d - 1  # q_{d,+}
        idx_neg = 2 * d  # q_{d,-}
        m = n - d
        # q_{d,+} = dx * sum_{i=0}^{m-1} h[i+d]*(1-h[i])
        q[idx_pos] = dx * float(np.sum(h[d:] * (1.0 - h[:m])))
        # Gradient for q_{d,+}
        G[idx_pos, :m] += -dx * h[d:]  # from (1-h[i]) term, d/dh[i]
        G[idx_pos, d:] += dx * (1.0 - h[:m])  # from h[i+d] term, d/dh[i+d]
        # q_{d,-} = dx * sum_{i=0}^{m-1} h[i]*(1-h[i+d])
        q[idx_neg] = dx * float(np.sum(h[:m] * (1.0 - h[d:])))
        G[idx_neg, :m] += dx * (1.0 - h[d:])  # from h[i] term
        G[idx_neg, d:] += -dx * h[:m]  # from (1-h[i+d]) term
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None, None
    return q, G

def active_bundle_gradient(q, G):
    Cq = float(np.max(q))
    tol = max(1e-12, 1e-7 * max(1.0, abs(Cq)))
    active = np.where(q >= Cq - tol)[0]
    if len(active) > 12:
        idx_sorted = np.argsort(q[active])[-12:]
        active = active[idx_sorted]
    k = len(active)
    if k == 0:
        return None
    U = G[active].copy()
    for r in range(k):
        U[r] -= np.mean(U[r])
    K = U @ U.T
    K = 0.5 * (K + K.T)
    alpha = np.full(k, 1.0 / k)
    for _ in range(40):
        z = K @ alpha
        gap = float(alpha @ z) - float(z[np.argmin(z)])
        if gap <= 1e-12:
            break
        j = int(np.argmin(z))
        v_dir = np.zeros(k)
        v_dir[j] = 1.0
        v_dir -= alpha
        denom = float(v_dir @ K @ v_dir)
        if denom <= 1e-24:
            break
        gamma = float(np.clip(-float(v_dir @ z) / denom, 0.0, 1.0))
        alpha += gamma * v_dir
        alpha = np.clip(alpha, 0.0, None)
        alpha /= alpha.sum()
    g = alpha @ U
    if not np.all(np.isfinite(g)):
        return None
    if np.max(np.abs(g)) < 1e-15:
        # Fallback: centered gradient of largest lag
        i0 = int(np.argmax(q))
        g = G[i0].copy()
        g -= np.mean(g)
        if not np.all(np.isfinite(g)):
            return None
    return g

def local_search(h0, rng, n, max_iter):
    S = n / 2.0
    h = project_capped_simplex(h0.copy(), n)
    if h is None:
        return h0, exact_score(h0)
    cur_score = exact_score(h)
    best = h.copy()
    best_score = cur_score
    eta = 0.5
    for it in range(max_iter):
        q, G = compute_lags(h)
        if q is None:
            break
        g = active_bundle_gradient(q, G)
        accepted = False

                trial = project_capped_simplex(h - eta_try * g, n)
                if trial is None or np.max(np.abs(trial - h)) < 1e-14:
                    continue
                s = exact_score(trial)
                if s < cur_score - 1e-12:
                    h = trial
                    cur_score = s
                    if s < best_score - 1e-12:
                        best = h.copy()
                        best_score = s
                    eta = min(2.0, 1.25 * eta_try)
                    accepted = True
                    break
        if not accepted:
            eta = max(1e-6, eta / 2.0)
            # Shake
            if it < 40:
                sigma, T = 0.12, 0.002
            elif it < 80:
                sigma, T = 0.07, 0.0008
            else:
                sigma, T = 0.035, 0.00025
            best_shake = None
            best_shake_score = cur_score
            for _ in range(4):
                prop = project_capped_simplex(h + sigma * rng.standard_normal(n), n)
                if prop is None:
                    continue
                ps = exact_score(prop)
                if best_shake is None or ps < best_shake_score:
                    best_shake = prop
                    best_shake_score = ps
            if best_shake is not None:
                delta = best_shake_score - cur_score
                if delta <= 1e-12 or rng.random() < math.exp(-delta / T):
                    h = best_shake
                    cur_score = best_shake_score
                    eta = 0.25
                    if cur_score < best_score - 1e-12:
                        best = h.copy()
                        best_score = cur_score
    best_score = exact_score(best)
    return best, best_score

def add_unique(lst, vec, tol=1e-10):

        G[in_, :m] += dx * (1.0 - h[d:])
        G[in_, d:] += -dx * h[:m]
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None, None
    return q, G

def active_bundle_gradient(q, G):
    Cq = float(np.max(q))
    tol = max(1e-12, 1e-7 * max(1.0, abs(Cq)))
    active = np.where(q >= Cq - tol)[0]
    if len(active) > 12:
        idx_s = np.argsort(q[active])[-12:]
        active = active[idx_s]
    k = len(active)
    if k == 0:
        return None
    U = G[active].copy()
    for r in range(k):
        U[r] -= np.mean(U[r])
    K = U @ U.T
    K = 0.5 * (K + K.T)
    alpha = np.full(k, 1.0 / k)
    for _ in range(40):
        z = K @ alpha
        gap = float(alpha @ z) - float(z[np.argmin(z)])
        if gap <= 1e-12:
            break
        j = int(np.argmin(z))
        v_dir = np.zeros(k)
        v_dir[j] = 1.0
        v_dir -= alpha
        denom = float(v_dir @ K @ v_dir)
        if denom <= 1e-24:
            break
        gamma = float(np.clip(-float(v_dir @ z) / denom, 0.0, 1.0))
        alpha += gamma * v_dir
        alpha = np.clip(alpha, 0.0, None)
        alpha /= alpha.sum()
    g = alpha @ U
    if not np.all(np.isfinite(g)):
        return None
    if np.max(np.abs(g)) < 1e-15:
        i0 = int(np.argmax(q))
        g = G[i0].copy()
        g -= np.mean(g)
        if not np.all(np.isfinite(g)):
            return None
    return g

def local_search(h0, rng, n, max_iter):
    h = project_capped_simplex(h0.copy(), n)
    if h is None:
        h = h0.copy()
    cur_score = exact_score(h)
    best = h.copy()
    best_score = cur_score
    eta = 0.5
    for it in range(max_iter):
        q, G = compute_lags(h)
        if q is None:
            break
        g = active_bundle_gradient(q, G)
        accepted = False
        if g is not None:
            for b in range(8):
                eta_try = eta * (0.5 ** b)
                trial = project_capped_simplex(h - eta_try * g, n)
                if trial is None or np.max(np.abs(trial - h)) < 1e-14:
                    continue
                s = exact_score(trial)
                if s < cur_score - 1e-12:
                    h = trial
                    cur_score = s
                    if s < best_score - 1e-12:
                        best = h.copy()
                        best_score = s
                    eta = min(2.0, 1.25 * eta_try)
                    accepted = True
                    break
        if not accepted:
            eta = max(1e-6, eta / 2.0)
            if it < 40:
                sigma, T = 0.12, 0.002
            elif it < 80:
                sigma, T = 0.07, 0.0008
            else:
                sigma, T = 0.035, 0.00025
            best_shake, best_shake_s = None, cur_score
            for _ in range(4):
                prop = project_capped_simplex(h + sigma * rng.standard_normal(n), n)
                if prop is None:
                    continue
                ps = exact_score(prop)
                if best_shake is None or ps < best_shake_s:
                    best_shake, best_shake_s = prop, ps
            if best_shake is not None:
                delta = best_shake_s - cur_score
                if delta <= 1e-12 or rng.random() < math.exp(-delta / T):
                    h = best_shake
                    cur_score = best_shake_s
                    eta = 0.25
                    if cur_score < best_score - 1e-12:
                        best = h.copy()
                        best_score = cur_score
    best_score = exact_score(best)
    return best, best_score

def add_unique(lst, vec, tol=1e-10):
    if vec is None or not np.all(np.isfinite(vec)):
        return lst
    for ex in lst:
        if np.max(np.abs(vec - ex)) < tol:
            return lst
    lst.append(vec.copy())
    return lst

def run(seed=42, **kwargs):
    parent = globals().get('initial_h_values', None)
    n = 42
    parent_vec = None
    try:
        seed = int(seed)
    except (ValueError, TypeError):
        seed = 42
    if parent is not None:
        try:
            p = np.array(parent, dtype=np.float64).ravel()
            if p.ndim == 1 and len(p) >= 2:
                n = len(p)
                if np.all(np.isfinite(p)):
                    parent_vec = p.copy()
        except Exception:
            pass
    if parent_vec is None and 'n_points' in kwargs:
        try:
            kn = int(kwargs['n_points'])
            if 2 <= kn <= 512:
                n = kn
        except (ValueError, TypeError):
            pass
    rng = np.random.default_rng(seed)
    S = n / 2.0
    const_h = np.full(n, 0.5)
    const_score = exact_score(const_h)
    incumbent = const_h.copy()
    incumbent_score = const_score
    parent_fb, parent_fb_score = None, const_score
    if parent_vec is not None:
        pf = None
        if (np.all(parent_vec >= 0) and np.all(parent_vec <= 1)
                and abs(float(np.sum(parent_vec)) - S) < 1e-12):
            pf = parent_vec.copy()
        else:
            pf = project_capped_simplex(parent_vec, n)
        if pf is not None and is_feasible(pf, n):
            ps = exact_score(pf)
            if math.isfinite(ps):
                parent_fb = pf
                parent_fb_score = ps
                if ps < incumbent_score - 1e-12:
                    incumbent = pf.copy()
                    incumbent_score = ps
    if n > 128:
        h_out = np.clip(incumbent, 0.0, 1.0)
        sc = exact_score(h_out)
        if not is_feasible(h_out, n):
            h_out, sc = const_h, const_score
        return (h_out, float(sc), n)
    starts = []
    if parent_fb is not None:
        add_unique(starts, parent_fb)
    add_unique(starts, const_h)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    blk = project_capped_simplex(blk, n)
    add_unique(starts, blk)
    alt = np.zeros(n)
    alt[0::2] = 1.0
    alt = project_capped_simplex(alt, n)
    add_unique(starts, alt)
    sinc = 0.5 + 0.45 * np.sin(2.0 * np.pi * np.arange(n) / n)
    sinc = project_capped_simplex(sinc, n)
    add_unique(starts, sinc)
    if parent_fb is not None:
        for sig in [0.05, 0.15, 0.35, 0.15]:
            pert = project_capped_simplex(parent_fb + sig * rng.standard_normal(n), n)
            add_unique(starts, pert)
        for _ in range(2):
            add_unique(starts, project_capped_simplex(rng.random(n), n))
    else:
        for _ in range(4):
            add_unique(starts, project_capped_simplex(rng.random(n), n))
    starts = starts[:12]
    max_iter = 120 if n <= 64 else 80
    for st in starts:
        try:
            cand, cs = local_search(st, rng, n, max_iter)
            if cand is not None and is_feasible(cand, n) and math.isfinite(cs):
                if cs < incumbent_score - 1e-12:
                    incumbent = cand.copy()
                    incumbent_score = cs
        except Exception:
            pass
    h_out = incumbent.copy()
    if not is_feasible(h_out, n):
        rep = project_capped_simplex(h_out, n)
        h_out = rep if rep is not None else const_h
    h_out = np.clip(h_out, 0.0, 1.0)
    sc = exact_score(h_out)
    if not is_feasible(h_out, n) or not math.isfinite(sc):
        if parent_fb is not None:
            h_out, sc = parent_fb, parent_fb_score
        else:
            h_out, sc = const_h, const_score
    return (h_out, float(sc), n)
