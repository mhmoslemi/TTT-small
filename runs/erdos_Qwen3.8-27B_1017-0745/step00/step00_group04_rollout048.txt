import numpy as np
import math

def _exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _is_feasible(h, n):
    if h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-15) or np.any(h > 1.0 + 1e-15):
        return False
    S = n / 2.0
    if abs(np.sum(h) - S) > 1e-10:
        return False
    if abs(math.fsum(float(x) for x in h) - S) > 1e-10:
        return False
    return True

def _project(v, n):
    S = n / 2.0
    v = np.clip(np.asarray(v, dtype=np.float64), -1e6, 1e6)
    lo_l = float(np.min(v)) - 2.0
    hi_l = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = (lo_l + hi_l) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        if np.sum(h) > S:
            lo_l = mid
        else:
            hi_l = mid
    h = np.clip(v - (lo_l + hi_l) / 2.0, 0.0, 1.0)
    # Correct residual
    residual = S - math.fsum(float(x) for x in h)
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            for k in idx:
                add = min(residual, 1.0 - h[k])
                if add <= 0:
                    break
                h[k] += add
                residual -= add
        else:
            idx = np.where(h > 1e-12)[0]
            for k in idx:
                rem = min(-residual, h[k])
                if rem <= 0:
                    break
                h[k] -= rem
                residual += rem
    h = np.clip(h, 0.0, 1.0)
    # Final mass correction on a single coordinate
    s = math.fsum(float(x) for x in h)
    diff = S - s
    if abs(diff) > 1e-14:
        for k in range(n):
            if diff > 0 and h[k] < 1.0 - abs(diff):
                h[k] += diff
                break
            elif diff < 0 and h[k] > abs(diff):

    return h

def _compute_qG(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M, dtype=np.float64)
    G = np.zeros((M, n), dtype=np.float64)
    # Lag 0
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0, :] = dx * (1.0 - 2.0 * h)
    # Non-zero lags
    for d in range(1, n):
        nd = n - d
        q[2*d-1] = dx * np.sum(h[d:d+nd] * (1.0 - h[0:nd]))
        q[2*d] = dx * np.sum(h[0:nd] * (1.0 - h[d:d+nd]))
        G[2*d-1, d:d+nd] += dx * (1.0 - h[0:nd])
        G[2*d-1, 0:nd] += -dx * h[d:d+nd]
        G[2*d, 0:nd] += dx * (1.0 - h[d:d+nd])
        G[2*d, d:d+nd] += -dx * h[0:nd]
    return q, G

def _real_roots(A, B, C, lo, hi):
    """Solve A*t^2 + B*t + C = 0 for t in [lo, hi]."""
    tol = 1e-14 * max(1.0, abs(A), abs(B), abs(C))
    roots = []
    if abs(A) <= tol:
        if abs(B) > tol:
            r = -C / B
            if lo - 1e-10 <= r <= hi + 1e-10:
                roots.append(max(lo, min(hi, r)))
        return roots
    disc = B * B - 4.0 * A * C
    disc_tol = 64.0 * np.finfo(float).eps * (B * B + abs(4.0 * A * C) + 1.0)
    if disc < -disc_tol:
        return roots
    if disc < 0:
        disc = 0.0
    sqrt_d = math.sqrt(disc)
    if B >= 0:
        z = -0.5 * (B + sqrt_d)
    else:
        z = -0.5 * (B - sqrt_d)
    if abs(z) > 1e-300:
        r1 = z / A
        r2 = C / z
    else:
        r1 = r2 = -B / (2.0 * A)
    for r in [r1, r2]:
        if np.isfinite(r) and lo - 1e-10 <= r <= hi + 1e-10:
            roots.append(max(lo, min(hi, r)))
    # If r1 == r2 (double root), avoid duplicate
    if len(roots) == 2 and abs(roots[0] - roots[1]) < 1e-12:
        roots = [roots[0]]
    return roots

- The envelope is a convex piecewise-linear function
- Each line on the envelope is active on an interval [x_start, x_end]
- x_start is where the new line overtakes the previous one
- For the first line, x_start = -∞
- For the last line, x_end = +∞

The algorithm:

for each (m, c) in sorted order:
    while hull is not empty:
