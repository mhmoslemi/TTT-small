        c_obj[n] = 1.0
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'maxiter': 300})
        except Exception:
            rho = max(rho_min, rho / 2.0)
            continue
        if res is None or not getattr(res, 'success', False):
            rho = max(rho_min, rho / 2.0)
            continue
        x = getattr(res, 'x', None)
        if x is None or x.shape != (n + 1,) or not np.all(np.isfinite(x)):
            rho = max(rho_min, rho / 2.0)
            continue
        d = x[:n]
        z_val = float(x[n])
        candidate = h + d
        proj = capped_simplex_project(candidate, n / 2.0)
        if proj is None or not is_feasible(proj, n):
            rho = max(rho_min, rho / 2.0)
            continue
        score = exact_score(proj, dx)
        if score is None:
            rho = max(rho_min, rho / 2.0)
            continue
        if score < best_score - 1e-12:
            best_score = score
            best_h = proj.copy()
        q_new = np.correlate(proj, 1.0 - proj, mode='full')
        Q_new = float(np.max(q_new))
        predicted = Q - z_val
        actual = Q - Q_new
        h_before = h.copy()
        accepted = False
        if abs(predicted) > 1e-10 * max(1.0, abs(Q)):
            ratio = actual / predicted
        else:
            ratio = 0.0
        if actual > 1e-12 * max(1.0, abs(Q)):
            h = proj.copy()
            accepted = True
        if not accepted or ratio < 0.1:
            rho = max(rho_min, rho / 2.0)
        elif accepted and ratio > 0.75 and np.max(np.abs(proj - h_before)) >= 0.8 * rho:
            rho = min(rho_max, 1.5 * rho)
    return best_h, best_score

def pairwise_poll(h, dx, n, rng):
    if n > 256:
        return h.copy()
    best_h = h.copy()
    best_score = exact_score(h, dx)
    if best_score is None:
        return h
    n_pairs = 24 if n <= 128 else 8
    for r in [0.08, 0.025, 0.008]:
        for _ in range(n_pairs):
            a = int(rng.integers(n))
            b = int(rng.integers(n))
            while b == a:
                b = int(rng.integers(n))
            s_lo = max(-best_h[a], best_h[b] - 1.0, -r)
            s_hi = min(1.0 - best_h[a], best_h[b], r)
            if s_lo > s_hi:
                continue
            cands = [s_lo, s_hi, -r, -0.5 * r, 0.5 * r, r]
            seen = []
            for s in cands:
                sc = min(max(s, s_lo), s_hi)
                if abs(sc) <= 1e-14:
                    continue
                if any(abs(sc - vv) < 1e-12 for vv in seen):
                    continue
                seen.append(sc)
            for s in seen:
                prop = best_h.copy()
                prop[a] += s
                prop[b] -= s
                proj = capped_simplex_project(prop, n / 2.0)
                if proj is None or not is_feasible(proj, n):
                    continue
                sc = exact_score(proj, dx)
                if sc is not None and sc < best_score - 1e-12:
                    best_score = sc
                    best_h = proj.copy()
    return best_h

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    global initial_h_values
    n = 70
    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=np.float64)
        if pv.ndim == 1 and 2 <= pv.shape[0] <= 999:
            n = pv.shape[0]
            if np.all(np.isfinite(pv)) and np.max(np.abs(pv)) <= 1e6:
                parent = pv.copy()
    except Exception:
        pass
    dx = 2.0 / n
    target = float(n) / 2.0
    u = np.full(n, 0.5)
    valid_parent = False
    if parent is not None:
        if parent.min() >= -1e-10 and parent.max() <= 1.0 + 1e-10 and abs(parent.sum() - target) <= 1e-8:
            valid_parent = True
    if valid_parent:
        fp = capped_simplex_project(parent, target)
        fallback_h = fp if (fp is not None and is_feasible(fp, n)) else u.copy()
    else:
        fallback_h = u.copy()
    starts = []
    if parent is not None:
        p1 = capped_simplex_project(parent, target)
        if p1 is not None and is_feasible(p1, n):
            starts.append(p1)
        p2 = capped_simplex_project(parent[::-1].copy(), target)
        if p2 is not None and is_feasible(p2, n):
            starts.append(p2)
        for scale in [0.035, 0.12]:
            pn = capped_simplex_project(parent + scale * rng.standard_normal(n), target)
            if pn is not None and is_feasible(pn, n):
                starts.append(pn)
        rv = capped_simplex_project(rng.random(n), target)
        if rv is not None:
            starts.append(rv)
        bv = capped_simplex_project(rng.beta(0.5, 0.5, size=n), target)
        if bv is not None:
            starts.append(bv)
    starts.append(u.copy())
    unique_starts = []
    for s in starts:
        if s is None or not is_feasible(s, n):
            continue
        if not any(np.max(np.abs(s - us)) < 1e-10 for us in unique_starts):
            unique_starts.append(s)
    best_h = fallback_h.copy()
    best_score = exact_score(best_h, dx)
    if best_score is None:
        best_score = 1.0
    for st in unique_starts:
        sc = exact_score(st, dx)
        if sc is not None and sc < best_score - 1e-12:
            best_score = sc
            best_h = st.copy()
        if best_score <= 0.3808:
            break
        sh, ss = slp_search(st, dx, n)
        if ss is not None and ss < best_score - 1e-12:
            best_score = ss
            best_h = sh.copy()
        ph = pairwise_poll(sh, dx, n, rng)
        psc = exact_score(ph, dx)
        if psc is not None and psc < best_score - 1e-12:
            best_score = psc
            best_h = ph.copy()
        if best_score <= 0.3808:
            break
    for rscale in [0.025, 0.08]:
        noisy = best_h + rscale * rng.standard_normal(n)
        rn = capped_simplex_project(noisy, target)
        if rn is not None and is_feasible(rn, n):
            sh, ss = slp_search(rn, dx, n)
            if ss is not None and ss < best_score - 1e-12:
                best_score = ss
                best_h = sh.copy()
            ph = pairwise_poll(sh, dx, n, rng)
            psc = exact_score(ph, dx)
            if psc is not None and psc < best_score - 1e-12:
                best_score = psc
                best_h = ph.copy()
        if best_score <= 0.3808:
            break
    final = capped_simplex_project(best_h, target)
    if final is None or not is_feasible(final, n):
        final = fallback_h.copy()
    final_score = exact_score(final, dx)
    if final_score is None:
        final = u.copy()
        final_score = exact_score(final, dx)
    if final_score is None:
        final_score = 1.0
    return (final, float(final_score), n)
