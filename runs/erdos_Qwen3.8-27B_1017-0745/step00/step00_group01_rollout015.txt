    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(dx * np.max(c))


def compute_c(h, dx):
    return dx * np.correlate(h, 1.0 - h, mode="full")


def smooth_value(c, tau):
    c_max = np.max(c)
    a = np.exp((c - c_max) / tau)
    return c_max + tau * np.log(np.sum(a))


def softmax_weights(c, tau):
    c_max = np.max(c)
    a = np.exp((c - c_max) / tau)
    return a / np.sum(a)


def compute_gradient(h, p, dx):
    n = len(h)
    grad = np.zeros(n)
    omh = 1.0 - h
    for u in range(2 * n - 1):
        pd = p[u]
        if pd < 1e-15:
            continue
        d = u - (n - 1)
        i0 = max(d, 0)
        j0 = max(-d, 0)
        m = n - abs(d)
        idx_i = np.arange(i0, i0 + m)
        idx_j = np.arange(j0, j0 + m)
        grad[idx_i] += pd * omh[idx_j]
        grad[idx_j] -= pd * h[idx_i]
    return dx * grad


def make_starts(base, n, rng):
    s = n / 2.0
    starts = []
    starts.append(base.copy())
    starts.append(base[::-1].copy())
    starts.append(1.0 - base[::-1].copy())
    starts.append(np.full(n, 0.5))
    bv = np.zeros(n)
    bv[:n // 2] = 1.0
    if n % 2 == 1:
        bv[n // 2] = 0.5
    starts.append(bv)
    av = np.zeros(n)
    av[::2] = 1.0
    starts.append(av)
    for sd in [0.05, 0.15, 0.35]:
        starts.append(base + sd * rng.standard_normal(n))
    starts.append(rng.uniform(0.0, 1.0, n))
    projected = [project_capped_simplex(st, s) for st in starts]
    unique = []
    for st in projected:
        dup = False
        for u in unique:
            if np.max(np.abs(st - u)) < 1e-10:
                dup = True
                break
        if not dup:
            unique.append(st)
    return unique


def projected_gradient_descent(x0, n, taus, iters_per_tau):
    s = n / 2.0
    dx = 2.0 / n
    x = x0.copy()
    armijo_c = 1e-4
    max_bt = 24
    best_x = x0.copy()
    best_sc = exact_score(x0)
    for tau in taus:
        alpha = 1.0
        stall = 0
        for _it in range(iters_per_tau):
            c = compute_c(x, dx)
            if not np.all(np.isfinite(c)):
                break
            Fx = smooth_value(c, tau)
            if not np.isfinite(Fx):
                break
            p = softmax_weights(c, tau)
            if not np.all(np.isfinite(p)):
                break
            grad = compute_gradient(x, p, dx)
            if not np.all(np.isfinite(grad)):
                break
            if np.max(np.abs(grad)) < 1e-10:
                break
            accepted = False
            for _bt in range(max_bt):
                y = project_capped_simplex(x - alpha * grad, s)
                D = np.sum((y - x) ** 2)
                if np.max(np.abs(y - x)) <= 1e-9 or D < 1e-20:
                    break
                cy = compute_c(y, dx)
                if not np.all(np.isfinite(cy)):
                    alpha *= 0.5
                    continue
                Fy = smooth_value(cy, tau)
                tol = 1e-14 * max(1.0, abs(Fx))
                if Fy <= Fx - armijo_c * D / alpha + tol:
                    imp = Fx - Fy
                    stall = stall + 1 if imp <= 1e-12 * max(1.0, abs(Fx)) else 0
                    x = y
                    sc = exact_score(x)
                    if np.isfinite(sc) and sc < best_sc - 1e-12:
                        best_sc = sc
                        best_x = x.copy()
                    alpha = min(2.0, 1.25 * alpha)
                    accepted = True
                    break
                alpha *= 0.5
            if stall >= 5:
                break
            if not accepted:
                break
    return best_x


def run(seed=42, **kwargs):
    n = None
    base = None
    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            arr = np.array(initial_h_values, dtype=np.float64).flatten()
            if len(arr) > 0:
                n = len(arr)
                base = arr if np.all(np.isfinite(arr)) else np.full(n, 0.5)
    except (NameError, TypeError, ValueError):
        pass
    if n is None or n < 1:
        n = 47
        if base is None:
            base = np.full(n, 0.5)
    skip_search = n > 999
    if n <= 64:
        max_starts, iters_per_tau = 10, 100
    elif n <= 256:
        max_starts, iters_per_tau = 5, 60
    else:
        max_starts, iters_per_tau = 2, 25
    try:
        r = int(kwargs.get('restarts', -1))
        if r > 0:
            max_starts = min(r, 12) if n <= 64 else (min(r, 6) if n <= 256 else min(r, 2))
    except (ValueError, TypeError):
        pass
    try:
        ip = int(kwargs.get('iters_per_tau', -1))
        if ip > 0:
            iters_per_tau = min(ip, 150) if n <= 64 else (min(ip, 90) if n <= 256 else min(ip, 40))
    except (ValueError, TypeError):
        pass
    try:
        rng = np.random.default_rng(int(seed))
    except (ValueError, TypeError):
        rng = np.random.default_rng(42)
    s = n / 2.0
    base = project_capped_simplex(base, s)
    fallback = base.copy()
    fb_score = exact_score(fallback)
    incumbent = base.copy()
    inc_score = fb_score
    if skip_search:
        fin = project_capped_simplex(incumbent, s)
        sc = exact_score(fin)
        if not np.isfinite(sc) or not check_feasible(fin, n):
            fin, sc = np.full(n, 0.5), exact_score(np.full(n, 0.5))
        return fin.astype(np.float64), float(sc), int(n)
    all_starts = make_starts(base, n, rng)
    starts = all_starts[:max_starts]
    for st in starts:
        sc = exact_score(st)
        if np.isfinite(sc) and sc < inc_score - 1e-12:
            incumbent, inc_score = st.copy(), sc
    taus = [0.02, 0.005, 0.001, 0.0002, 0.00005, 0.00001]
    for st in starts:
        try:
            res = projected_gradient_descent(st, n, taus, iters_per_tau)
            sc = exact_score(res)
            if np.isfinite(sc) and sc < inc_score - 1e-12:
                incumbent, inc_score = res.copy(), sc
        except Exception:
            continue
    final = project_capped_simplex(incumbent, s)
    final_score = exact_score(final)
    fb = project_capped_simplex(fallback, s)
    fb_score = exact_score(fb)
    if not np.isfinite(fb_score):
        fb, fb_score = np.full(n, 0.5), exact_score(np.full(n, 0.5))
    if not np.isfinite(final_score) or final_score > fb_score:
        final, final_score = fb, fb_score
    if not check_feasible(final, n):
        final, final_score = np.full(n, 0.5), exact_score(np.full(n, 0.5))
    return final.astype(np.float64), float(final_score), int(n)
