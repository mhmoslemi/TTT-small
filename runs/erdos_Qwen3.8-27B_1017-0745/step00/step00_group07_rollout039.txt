def project_box_sum(z, S, lower, upper):
    n = len(z)
    z = np.array(z, dtype=np.float64)
    lower = np.array(lower, dtype=np.float64)
    upper = np.array(upper, dtype=np.float64)
    if not np.all(np.isfinite(z)) or not np.all(np.isfinite(lower)) or not np.all(np.isfinite(upper)):
        return None
    if np.any(upper < lower - 1e-14):
        return None
    z = np.clip(z, lower, upper)
    S = float(S)
    sl, su = float(np.sum(lower)), float(np.sum(upper))
    if S < sl - 1e-10 or S > su + 1e-10:
        return None
    z = np.clip(z, lower, upper)
    s = float(np.sum(z))
    if abs(s - S) < 1e-10:
        return z
    # rescale z for stability
    max_abs = np.max(np.abs(z))
    if max_abs > 1e-30:
        z = z / max_abs
    z = z - np.median(z) + S / n  # center roughly

        sy = float(np.sum(y))

        else:
            hi = mid
    y = np.clip(z - 0.5*(lo+hi), lower, upper)
    # fix residual
    residual = S - float(np.sum(y))

def exact_score(h):
    n = len(h)
    if n < 2 or not np.all(np.isfinite(h)):
        return None
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    score = float(np.max(c)) * dx
    if not np.isfinite(score):
        return None
    return score

def lag_model(x):
    n = len(x)
    m = 2 * n - 1
    B = np.zeros(m)
    G = np.zeros((m, n))
    for d in range(-(n-1), n):
        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        B[row] = float(np.dot(x[lo:hi], 1.0 - x[lo+d:hi+d]))
        G[row, lo:hi] += 1.0 - x[lo+d:hi+d]
        if d != 0:
            G[row, lo+d:hi+d] -= x[lo:hi]
        # wait, for d=0: lo=0, hi=n, G[row, 0:n] += 1-x[0:n], G[row, 0:n] -= x[0:n] => 1-2x
        # Actually the formula is: G[d,q] = (1-x[q+d]) * 1_{lo<=q<hi} - x[q-d] * 1_{lo+d<=q<hi+d}
        # For d=0: first part: q in [0,n), value 1-x[q]; second part: q in [0,n), value x[q]
        # So G[0,q] = 1-x[q] - x[q] = 1-2x[q]. Correct.
        # For d!=0: first part is q in [lo, hi), second part is q in [lo+d, hi+d)
        # We add 1-x[q+d] to columns lo:hi, and subtract x[q-d] from columns lo+d:hi+d
        # But we need to be careful: the second term subtracts x[q-d] for q in [lo+d, hi+d)
        # So for column j in [lo+d, hi+d), we subtract x[j-d]
        # That's the same as: for the slice lo+d:hi+d, subtract x[lo:hi]
        # Wait, x[j-d] for j in lo+d..hi+d-1 means x[(lo+d-d)..(hi+d-d)] = x[lo..hi-1] = x[lo:hi]
        # Yes! So G[row, lo+d:hi+d] -= x[lo:hi] is correct.
    return B, G

    if score_x is None:
        return x, None

    Delta_max = 0.5

    small_count = 0

            continue

        fail_count = 0

            elif rho < 0.25:
                Delta = max(0.25 * Delta, Delta_min)

                small_count += 1
            else:
                small_count = 0
            if small_count >= 8:
                break
        else:
            Delta = 0.25 * Delta
    return best_local, best_score
