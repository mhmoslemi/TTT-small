import numpy as np
from scipy.optimize import minimize


def project_capped_simplex(z, S):
    n = len(z)
    if n < 1 or not np.all(np.isfinite(z)) or np.max(np.abs(z)) > 1e12:
        return None
    low = np.min(z) - 1.0
    high = np.max(z)
    for _ in range(80):
        mid = low + 0.5 * (high - low)
        y = np.clip(z - mid, 0.0, 1.0)
        if np.sum(y) > S:
            low = mid
        else:
            high = mid
    y = np.clip(z - (low + 0.5 * (high - low)), 0.0, 1.0)
    r = S - np.sum(y)
    if r > 0:
        for i in range(n):
            if y[i] < 1.0:
                a = min(r, 1.0 - y[i])
                y[i] += a
                r -= a
                if abs(r) < 1e-15:
                    break
    elif r < 0:
        for i in range(n):
            if y[i] > 0.0:
                a = min(-r, y[i])
                y[i] -= a
                r += a
                if abs(r) < 1e-15:
                    break
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(np.sum(y) - S) > 1e-10:
        return None
    return y


def check_feasible(h, n):
    if h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10:
        return False
    return True


def exact_score(h):
    n = len(h)
    if n < 1 or not np.all(np.isfinite(h)):
        return None
    vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(vals)):
        return None
    return float(np.max(vals))


def compute_lags(h):
    n = len(h)
    B = np.zeros(2 * n - 1)
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        d = k - (n - 1)
        lo = max(0, -d)
        hi = min(n, n - d)
        B[k] = np.dot(h[lo:hi], 1.0 - h[lo + d:hi + d])
        if hi > lo:
            G[k, lo:hi] += 1.0 - h[lo + d:hi + d]
            G[k, lo + d:hi + d] -= h[lo:hi]
    return B, G


def slsqp_obj(z):
    return z[-1]


def slsqp_obj_jac(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g


def slsqp_ineq(z):
    n = len(z) - 1
    B, _ = compute_lags(z[:n])
    return z[-1] - B


def slsqp_ineq_jac(z):
    n = len(z) - 1
    _, G = compute_lags(z[:n])
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J


def slsqp_eq(z):
    n = len(z) - 1
    return np.sum(z[:n]) - n / 2.0


def slsqp_eq_jac(z):
    n = len(z) - 1
    J = np.zeros((1, n + 1))
    J[0, :n] = 1.0
    return J


def run(seed=42, **kwargs):
    n = 80
    parent = None
    try:
        pv = globals().get('initial_h_values', None)
        if pv is not None:
            pv = np.asarray(pv, dtype=np.float64).ravel()
            if len(pv) >= 2 and np.all(np.isfinite(pv)):
                n = len(pv)
                p = project_capped_simplex(pv, n / 2.0)
                if p is not None:
                    parent = p
    except Exception:
        n = 80
    if parent is None and 'n_points' in kwargs:
        try:
            nk = int(kwargs['n_points'])
            if 8 <= nk <= 512:
                n = nk
        except Exception:
            pass
    S = n / 2.0
    rng = np.random.default_rng(seed)
    const_h = np.full(n, 0.5)
    starts = []
    if parent is not None:
        starts.append(parent.copy())
    starts.append(const_h.copy())
    if parent is not None:
        starts.append(parent[::-1].copy())
    max_starts = 6 if n <= 128 else 3
    attempts = 0
    while len(starts) < max_starts and attempts < 18:
        attempts += 1
        i_arr = np.arange(n)
        if attempts == 1 and parent is not None:
            h_new = parent + 0.18 * rng.standard_normal(n)
        elif attempts in (2, 3):
            q = int(rng.integers(1, min(6, (n - 1) // 2) + 1))
            phi = rng.uniform(0.0, 2.0 * np.pi)
            h_new = 0.5 + 0.45 * np.cos(2.0 * np.pi * q * i_arr / n + phi)
        elif attempts == 4:
            perm = rng.permutation(n)
            h_new = np.zeros(n)
            h_new[perm[:n // 2]] = 1.0
            if n % 2 == 1:
                h_new[perm[n // 2]] = 0.5
        else:
            h_new = rng.uniform(0.0, 1.0, n)
        p = project_capped_simplex(h_new, S)
        if p is not None:
            dup = False
            for s in starts:
                if np.max(np.abs(p - s)) < 1e-10:
                    dup = True
                    break
            if not dup:
                starts.append(p.copy())
    best_h = None
    best_score = np.inf
    for h in starts:
        if check_feasible(h, n):
            sc = exact_score(h)
            if sc is not None and sc < best_score:
                best_score = sc
                best_h = h.copy()
    if n <= 128:
        bounds = [(0.0, 1.0)] * n + [(None, None)]
        cons = [
            {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac},
            {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
        ]
        for h0 in starts[:max_starts]:
            if not check_feasible(h0, n):
                continue
            B0, _ = compute_lags(h0)
            t0 = float(np.max(B0)) + 1e-8 * max(1.0, abs(float(np.max(B0))))
            z0 = np.concatenate([h0, [t0]])
            try:
                res = minimize(slsqp_obj, z0, jac=slsqp_obj_jac,
                               method='SLSQP', bounds=bounds,
                               constraints=cons,
                               options={'maxiter': 180, 'ftol': 1e-9,
                                        'disp': False})
            except Exception:
                continue
            if res is None or not hasattr(res, 'x') or len(res.x) != n + 1:
                continue
            h_sol = np.asarray(res.x[:n], dtype=np.float64)
            if not np.all(np.isfinite(h_sol)):
                continue
            p = project_capped_simplex(h_sol, S)
            if p is None or not check_feasible(p, n):
                continue
            sc = exact_score(p)
            if sc is not None and sc < best_score:
                best_score = sc
                best_h = p.copy()
    if best_h is None:
        best_h = const_h.copy()
        best_score = exact_score(best_h)
    p = project_capped_simplex(best_h, S)
    if p is not None:
        best_h = p
    sc = exact_score(best_h)
    if sc is None:
        best_h = const_h.copy()
        sc = exact_score(best_h)
    return (best_h, float(sc), n)
