        if e > 0:
            room = 1.0 - h
            add = e
            for j in np.argsort(-room):
                if add <= 0:
                    break
                step = min(add, room[j])
                h[j] += step
                add -= step
        else:
            sub = -e
            for j in np.argsort(h):
                if sub <= 0:
                    break
                step = min(sub, h[j])
                h[j] -= step
                sub -= step
        e = m - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return np.full(N, 0.5)
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - m) > 1e-10 * max(1, N):
        return np.full(N, 0.5)
    return h

def lag_q_G(h, N):
    dx = 2.0 / N
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    G = np.zeros((2 * N - 1, N))
    for lag in range(2 * N - 1):
        k = lag - (N - 1)
        if k >= 0:
            L = N - k
            G[lag, k:k+L] += (1.0 - h[0:L])
            G[lag, 0:L] -= h[k:k+L]
        else:
            s = -k
            L = N - s
            G[lag, 0:L] += (1.0 - h[s:s+L])
            G[lag, s:s+L] -= h[0:L]
    G *= dx
    return q, G

def build_model(N, dx):
    d = cp.Variable(N)
    u = cp.Variable()
    t = cp.Variable()
    x_p = cp.Parameter(N)
    r0_p = cp.Parameter()
    J0_p = cp.Parameter(N)
    r_nz_p = cp.Parameter(2 * N - 2)
    J_nz_p = cp.Parameter((2 * N - 2, N))
    R_p = cp.Parameter()
    cons = [
        x_p + d >= 0,
        x_p + d <= 1,
        cp.sum(d) == 0,
        cp.norm(d, 2) <= R_p,
        u * cp.ones(2 * N - 2) >= r_nz_p + J_nz_p @ d,
        t >= u + dx * cp.sum_squares(d),
        t >= r0_p + J0_p @ d,
    ]
    prob = cp.Problem(cp.Minimize(t), cons)
    return prob, d, u, t, x_p, r0_p, J0_p, r_nz_p, J_nz_p, R_p

def make_starts(N, rng, baseline):
    starts = []
    if baseline is not None:
        starts.append(baseline.copy())
    starts.append(np.full(N, 0.5))
    if baseline is not None:
        z = 1.0 - baseline.copy()
    else:
        z = rng.standard_normal(N) + 0.5
    starts.append(project_box_sum(z, N))
    if baseline is not None:
        z2 = baseline.copy() + 0.15 * rng.standard_normal(N)
    else:
        z2 = 0.5 + 0.15 * rng.standard_normal(N)
    starts.append(project_box_sum(z2, N))
    starts.append(project_box_sum(rng.uniform(0, 1, N), N))
    Rf = min(5, (N - 1) // 2)
    z4 = np.full(N, 0.5)
    ii = np.arange(N)
    for r in range(1, Rf + 1):
        coeff = 0.12 / r
        A = rng.standard_normal()
        B = rng.standard_normal()
        z4 += coeff * (A * np.sin(2 * np.pi * r * ii / N) + B * np.cos(2 * np.pi * r * ii / N))
    starts.append(project_box_sum(z4, N))
    uniq = []
    for s in starts:
        dup = False
        for u2 in uniq:
            if np.max(np.abs(s - u2)) < 1e-8:
                dup = True
                break
        if not dup:
            uniq.append(s)
    return uniq

def solve_local(start, N, dx, max_iter, mp):
    prob, d_var, u_var, t_var, x_p, r0_p, J0_p, r_nz_p, J_nz_p, R_p = mp
    current = start.copy()
    F_curr = score_exact(current, N)
    R = min(1.0, np.sqrt(N))
    R_max = min(4.0, np.sqrt(N))
    R_min = 1e-4
    fail_count = 0
    nz_idx = np.array([i for i in range(2 * N - 1) if i != N - 1])
    for _ in range(max_iter):
        if R <= R_min or fail_count >= 8:
            break
        q_s, G_s = lag_q_G(current, N)
        r0 = q_s[N - 1]
        J0 = G_s[N - 1, :]
        r_nz = q_s[nz_idx]
        J_nz = G_s[nz_idx, :]
        x_p.value = current
        r0_p.value = r0
        J0_p.value = J0
        r_nz_p.value = r_nz
        J_nz_p.value = J_nz
        R_p.value = R
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=200,
                       abstol=1e-8, reltol=1e-8, feastol=1e-8)
        except Exception:
            R = max(R / 2.0, R_min)
            fail_count += 1
            continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            R = max(R / 2.0, R_min)
            fail_count += 1
            continue
        d_val = np.array(d_var.value).flatten()
        if d_val.size != N or not np.all(np.isfinite(d_val)):
            R = max(R / 2.0, R_min)
            fail_count += 1
            continue
        trial = project_box_sum(current + d_val, N)
        d_eff = trial - current
        if np.linalg.norm(d_eff) > R * (1 + 1e-6) + 1e-10:
            R = max(R / 2.0, R_min)
            fail_count += 1
            continue
        F_new = score_exact(trial, N)
        if F_new < F_curr - 1e-12 * max(1.0, abs(F_curr)):
            actual = F_curr - F_new
            U_nz = r_nz + J_nz @ d_eff
            U = max(r0 + J0 @ d_eff, np.max(U_nz) + dx * float(np.dot(d_eff, d_eff)))
            predicted = F_curr - U
            if predicted > 1e-12:
                ratio = actual / predicted
            elif actual > 0:
                ratio = 100.0
            else:
                ratio = 0.0
            if ratio >= 0.75 and np.linalg.norm(d_eff) >= 0.8 * R:
                R = min(1.5 * R, R_max)
            elif ratio < 0.1:
                R = max(R / 2.0, R_min)
            current = trial
            F_curr = F_new
            fail_count = 0
            if np.linalg.norm(d_eff) < 1e-7:
                break
        else:
            R = max(R / 2.0, R_min)
            fail_count += 1
    return current, F_curr

def run(seed=42, **kwargs):
    N = 70
    baseline = None
    try:
        parent = np.array(initial_h_values, dtype=np.float64).flatten()
        if parent.size > 0 and np.all(np.isfinite(parent)):
            N = parent.size
            if np.all(parent >= -1e-12) and np.all(parent <= 1.0 + 1e-12):
                p = np.clip(parent, 0.0, 1.0)
                if abs(np.sum(p) - N / 2.0) <= 1e-10 * max(1, N):
                    baseline = p
                else:
                    baseline = project_box_sum(p, N)
            else:
                baseline = project_box_sum(parent, N)
    except Exception:
        pass
    if baseline is None:
        try:
            np_ = int(kwargs.get('n_points', 0))
            if 1 <= np_ <= 999:
                N = np_
        except Exception:
            pass
        baseline = np.full(N, 0.5)

    if N == 1:
        h1 = np.array([0.5])
        return h1, score_exact(h1, 1), 1

    dx = 2.0 / N
    rng = np.random.default_rng(seed)

    if N <= 100:
        max_starts, max_iter = 6, 20
    elif N <= 250:
        max_starts, max_iter = 5, 10
    elif N <= 500:
        max_starts, max_iter = 4, 6
    else:
        max_starts, max_iter = 3, 3

    starts = make_starts(N, rng, baseline)
    starts = starts[:max_starts]

    incumbent = starts[0].copy()
    best_score = score_exact(incumbent, N)
    for s in starts[1:]:
        sc = score_exact(s, N)
        if sc < best_score:
            best_score = sc
            incumbent = s.copy()

    if N <= 1000:
        try:
            mp = build_model(N, dx)
            for s in starts:
                opt_h, opt_sc = solve_local(s, N, dx, max_iter, mp)
                if opt_sc < best_score - 1e-12 * max(1.0, abs(best_score)):
                    best_score = opt_sc
                    incumbent = opt_h.copy()
        except Exception:
            pass

    h_final = project_box_sum(incumbent, N)
    sc_final = score_exact(h_final, N)
    if not np.isfinite(sc_final):
        h_final = np.full(N, 0.5)
        sc_final = score_exact(h_final, N)
    return h_final.astype(np.float64), float(sc_final), int(N)
