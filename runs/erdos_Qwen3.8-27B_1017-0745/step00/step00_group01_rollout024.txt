import numpy as np
import math

def check_feasibility(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1 + 1e-12):
        return False
    tol = 1e-10 * max(1, n)
    if abs(np.sum(h) - n / 2.0) > tol:
        return False
    return True

def uniform_h(n):
    return np.full(n, 0.5, dtype=np.float64)

def capped_simplex_project(z, n):
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    if np.any(np.abs(z) > 1e12):
        return None
    s = n / 2.0
    tol = 1e-10 * max(1, n)
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(z - mid, 0.0, 1.0)
        if np.sum(h) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)

        if abs(r) <= tol:
            break
        if r > 0:
            idx = np.argmax(1.0 - h)

            h[idx] += add
        else:
            idx = np.argmax(h)
            sub = min(-r, h[idx])
            if sub <= 1e-15:
                break
            h[idx] -= sub
    h = np.clip(h, 0.0, 1.0)

    return None

def exact_score(h, n):
    if not check_feasibility(h, n):
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full')
    if q.shape != (2 * n - 1,):
        return float('inf')
    if not np.all(np.isfinite(q)):
        return float('inf')
    return float(np.max(q) * (2.0 / n))

def haar_basis(n):
    if n <= 1:
        return np.empty((0, n), dtype=np.float64)
    directions = []
    stack = [(0, n)]
    while stack:
        a, b = stack.pop()
        m = b - a
        if m < 2:
            continue
        mid = a + m // 2
        mL = mid - a
        mR = b - mid
        v = np.zeros(n, dtype=np.float64)
        v[a:mid] = math.sqrt(mR / (m * mL))
        v[mid:b] = -math.sqrt(mL / (m * mR))
        if np.all(np.isfinite(v)) and abs(np.sum(v)) < 1e-10 * m:
            directions.append(v)
        stack.append((mid, b))
        stack.append((a, mid))
    return np.array(directions, dtype=np.float64)

def feasible_interval(x, v):
    lo, hi = 0.0, 0.0
    nz = np.nonzero(np.abs(v) > 1e-15)[0]
    if len(nz) == 0:
        return None
    lo = -np.inf
    hi = np.inf
    for i in nz:
        vi = v[i]
        r0 = -x[i] / vi
        r1 = (1.0 - x[i]) / vi
        a2, b2 = min(r0, r1), max(r0, r1)
        lo = max(lo, a2)
        hi = min(hi, b2)
    if not (np.isfinite(lo) and np.isfinite(hi)):
        return None
    if lo > hi + 1e-12:
        return None
    if hi - lo < 1e-14:
        return None
    return lo, hi

def line_coefficients(x, v, n):
    dx = 2.0 / n
    c = dx * np.correlate(x, 1.0 - x, mode='full')
    b = dx * (np.correlate(v, 1.0 - x, mode='full') - np.correlate(x, v, mode='full'))
    a = -dx * np.correlate(v, v, mode='full')

        return None

        return None

def line_candidates(a, b, c, lo, hi, pair_ids):

def evaluate_envelope(a, b, c, ts):
    results = []

    for s in range(0, len(ts), chunk):
        tc = ts[s:s + chunk]
        vals = tc[:, None] * tc[:, None] * a[None, :] + tc[:, None] * b[None, :] + c[None, :]
        mx = np.max(vals, axis=1)
        results.append(mx)
    return np.concatenate(results)

    if iv is None:
        return None

        return None

        env = evaluate_envelope(a, b, c, cands)

        return None

        return None

        return None

        sc = exact_score(y, n)
        if not math.isinf(sc) and sc < best_s:
            best_s = sc
            best_y = y.copy()
    return best_y, best_s
