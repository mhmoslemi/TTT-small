def try_slsqp(h0, n):
    try:
        ctx = SLSQPCtx(n)
        t0 = min(1.0, exact_score(h0) + 1e-8)
        z0 = np.concatenate([h0, [t0]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        cons = [
            {'type': 'ineq', 'fun': ctx.ineq, 'jac': ctx.ineq_jac},
            {'type': 'eq', 'fun': ctx.eq, 'jac': ctx.eq_jac},
        ]
        res = minimize(ctx.obj, z0, jac=ctx.obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 250, 'ftol': 1e-10, 'disp': False})

            h_slqp = try_slsqp(sel, n)
            if h_slqp is not None:
                sc = exact_score(h_slqp)
                if sc < best_score - 1e-12:
                    best_h = h_slqp.copy()
                    best_score = sc

    best_h = project_to_K(best_h, n)
    final_score = exact_score(best_h)
    if not np.all(np.isfinite(best_h)) or final_score > best_score + 1e-9:
        best_h = project_to_K(0.5 + 0.0, n)
        final_score = exact_score(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    return best_h.copy(), final_score, n
