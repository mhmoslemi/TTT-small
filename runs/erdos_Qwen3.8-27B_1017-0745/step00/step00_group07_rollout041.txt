import numpy as np
import math

def _feasible(h, n, S):
    if not isinstance(h, np.ndarray) or h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-15) or np.any(h > 1.0 + 1e-15):
        return False
    if abs(np.sum(h) - S) > 1e-10:
        return False
    return True

def _project(z, S):
    n = len(z)
    if not np.all(np.isfinite(z)) or np.max(np.abs(z)) > 1e12:
        return None
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(80):
        mid = lo + 0.5 * (hi - lo)
        y = np.clip(z - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(z - (lo + 0.5*(hi-lo)), 0.0, 1.0)
    # residual correction
    for _ in range(10):
        resid = S - np.sum(y)
        if abs(resid) < 1e-12:
            break
        if resid > 0:
            slack = 1.0 - y
            idx = np.where(slack > 1e-13)[0]
            if len(idx) == 0: break
            add = min(resid, np.sum(slack[idx]))
            y[idx] += add * slack[idx] / np.sum(slack[idx])
        else:
            idx = np.where(y > 1e-13)[0]
            if len(idx) == 0: break
            sub = min(-resid, np.sum(y[idx]))
            y[idx] -= sub * y[idx] / np.sum(y[idx])
    y = np.clip(y, 0.0, 1.0)
    if abs(np.sum(y) - S) > 1e-10 or not np.all(np.isfinite(y)):
        return None
    return y

def _score(h, n):
    S = n / 2.0
    if not _feasible(h, n, S):
        return None, None
    corr = np.correlate(h, 1.0 - h, 'full')
    val = float(np.max(corr) * (2.0 / n))
    if not math.isfinite(val):
        return None, None
    raw = corr[::-1].copy()
    return val, raw

def _qroots(A, B, C):
    # returns list of real roots of A*t^2 + B*t + C = 0
    sc = max(1.0, abs(B), abs(C))
    if abs(A) < 1e-14 * sc:
        if abs(B) < 1e-14 * sc:
            return []
        return [float(-C / B)]
    disc = B * B - 4.0 * A * C
    tol = 1e-12 * sc * sc
    if disc < -tol:
        return []
    if disc < 0:
        disc = 0.0
    sq = math.sqrt(disc)
    if abs(B) < 1e-14 * sc:
        r1 = sq / A
        r2 = -sq / A
        return sorted([float(r1), float(r2)]) if abs(r1 - r2) > 1e-12 else [float(r1)]
    qv = -0.5 * (B + math.copysign(sq, B))
    r1 = float(qv / A)
    r2 = float(C / qv)
    if abs(r1 - r2) < 1e-12 * max(1.0, abs(r1)):
        return [r1]
    return sorted([r1, r2])

def _hull(slopes, intercepts):
    # Build upper envelope of lines y=slope*t+intercept
    # Returns (hull_slopes, hull_intercepts, hull_starts) arrays
    M = len(slopes)
    if M == 0:
        return np.array([]), np.array([]), np.array([])
    order = np.argsort(slopes, kind='stable')
    hs, hi_, hx = [], [], []
    # group by slope, keep max intercept
    i = 0
    while i < M:
        s = slopes[order[i]]
        best_c = -np.inf
        j = i
        while j < M and slopes[order[j]] == s:
            best_c = max(best_c, intercepts[order[j]])
            j += 1
        # add line (s, best_c)
        if not hs:
            hs.append(s); hi_.append(best_c); hx.append(-np.inf)
        else:
            # intersection with last
            dx = s - hs[-1]
            if abs(dx) < 1e-15:
                if best_c > hi_[-1]:
                    hs[-1] = s; hi_[-1] = best_c
                i = j; continue
            x = (hi_[-1] - best_c) / dx
            while len(hx) > 1 and x <= hx[-1] + 1e-15:
                hx.pop(); hi_.pop(); hs.pop()
                dx = s - hs[-1]
                x = (hi_[-1] - best_c) / dx
            hs.append(s); hi_.append(best_c); hx.append(x)
        i = j
    return np.array(hs), np.array(hi_), np.array(hx)

def _eval_hull(t, hs, hi_, hx):
    if len(hs) == 0:
        return -np.inf
    idx = np.searchsorted(hx, t, side='right') - 1
    idx = max(0, min(idx, len(hs) - 1))
    return hs[idx] * t + hi_[idx]

    t_lo = max(-h[p], h[q] - 1.0)
    t_hi = min(1.0 - h[p], h[q])
    if not (t_lo <= t_hi) or not np.isfinite(t_lo) or not np.isfinite(t_hi):
        return None
    if t_hi - t_lo < 1e-15:
        return 0.0
    gap = q - p  # signed
    # special lags: 0, gap, -gap
    special = {0, gap, -gap}
    d_min = -(n - 1)
    # Compute b_d and c_d for all lags
    # c_d = raw[d - d_min] = raw[d + n - 1]
    # b_d = g[d,p] - g[d,q]
    # For a given d: g[d,r] = 1_{lo<=r<hi}*(1-h[r+d]) - 1_{lo<=r-d<hi}*h[r-d]
    # where lo=max(0,-d), hi=min(n,n-d)
    # We only need b_d for all d to build the hull, and a_d for the 3 special ones.
    
    b_vals = np.zeros(2*n-1)
    c_vals = raw.copy()  # c_vals[i] = B_{i-(n-1)}
    
    for d in range(-(n-1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        # g[d, p]: p in [lo, hi)? and check r-d in [lo, hi)?
        gp = 0.0
        if lo <= p < hi:
            gp += (1.0 - h[p + d])
        if lo <= p - d < hi:
            gp -= h[p - d]
        gq = 0.0
        if lo <= q < hi:
            gq += (1.0 - h[q + d])
        if lo <= q - d < hi:
            gq -= h[q - d]
        b_vals[d + n - 1] = gp - gq
    
    # a_d: -2 at d=0, +1 at d=±gap
    a_special = {}
    a_special[0] = -2.0
    a_special[gap] = 1.0
    a_special[-gap] = 1.0
    
    # Build hull from affine lags (d not in special)
    aff_idx = []
    for d in range(-(n-1), n):
        if d not in special:
            aff_idx.append(d + n - 1)
    if len(aff_idx) > 0:
        aff_slopes = b_vals[aff_idx]
        aff_ints = c_vals[aff_idx]
        hs, hi_, hx = _hull(aff_slopes, aff_ints)
    else:
        hs, hi_, hx = np.array([]), np.array([]), np.array([])
    
    # Build candidate set
    cands = [t_lo, t_hi, 0.0]
    # hull breakpoints
    for x in hx:
        if t_lo <= x <= t_hi:
            cands.append(float(x))
    
    # For each hull segment, find roots of Q_i - L = 0
    # Hull segments: [hx[i], hx[i+1]] for i in range(len(hs)), clipped to [t_lo, t_hi]
    seg_bounds = [t_lo]
    for x in hx:
        if t_lo < x < t_hi:
            seg_bounds.append(float(x))
    seg_bounds.append(t_hi)
    
    # special quadratic params
    spec_lags = [0, gap, -gap]
    for seg_i in range(len(seg_bounds) - 1):
        sa, sb2 = seg_bounds[seg_i], seg_bounds[seg_i + 1]
        if sb2 - sa < 1e-15:
            continue
        # which hull line is active in this segment?
        mid = 0.5 * (sa + sb2)
        idx = np.searchsorted(hx, mid, side='right') - 1
        if len(hs) > 0:
            idx = max(0, min(idx, len(hs) - 1))
            # active line: hs[idx]*t + hi_[idx]
            for sl in spec_lags:
                si = sl + n - 1
                a_d = a_special[sl]
                b_d = b_vals[si]
                c_d = c_vals[si]
                # Q(t) - L(t) = a_d*t^2 + (b_d - hs[idx])*t + (c_d - hi_[idx])
                roots = _qroots(a_d, b_d - hs[idx], c_d - hi_[idx])
                for r in roots:
                    if sa <= r <= sb2:
                        cands.append(r)
        # else: no affine, skip (just quadratics)
    
    # if no hull, add roots of Q_i - Q_j = 0 and vertices
    # Pairwise intersections of special quadratics
    for i2 in range(3):
        for j2 in range(i2 + 1, 3):
            si, sj = spec_lags[i2], spec_lags[j2]
            ii, jj = si + n - 1, sj + n - 1
            da = a_special[si] - a_special[sj]
            db = b_vals[ii] - b_vals[jj]
            dc = c_vals[ii] - c_vals[jj]
            roots = _qroots(da, db, dc)
            for r in roots:
                if t_lo <= r <= t_hi:
                    cands.append(r)
    
    # Vertices of convex quadratics (a>0)
    for sl in spec_lags:
        a_d = a_special[sl]
        if a_d > 0:
            v = -b_vals[sl + n - 1] / (2.0 * a_d)
            if t_lo <= v <= t_hi:
                cands.append(float(v))
    
    # Evaluate F at all candidates
    best_t = 0.0
    best_F = np.inf
    for t in cands:
        if not np.isfinite(t) or t < t_lo - 1e-12 or t > t_hi + 1e-12:
            continue
        t = max(t_lo, min(t_hi, t))
        F = _eval_hull(t, hs, hi_, hx)
        for sl in spec_lags:
            si = sl + n - 1
            a_d = a_special[sl]
            F = max(F, a_d * t * t + b_vals[si] * t + c_vals[si])
        if F < best_F - 1e-15:
            best_F = F
            best_t = float(t)
        elif abs(F - best_F) <= 1e-15 and abs(t) < abs(best_t):
            best_t = float(t)
    
    if not np.isfinite(best_F):
        return None
    return best_t

    best_h, best_sc = pool[0], inc_score
    for cand in pool:
        if not _feasible(cand, n, S):
            pr = _project(cand, S)
            if pr is not None: cand = pr
            else: continue
        sc, _ = _score(cand, n)
        if sc is not None and sc < best_sc:
            best_sc = sc; best_h = cand.copy()
    final_sc, _ = _score(best_h, n)
    if final_sc is None: final_sc = best_sc
    return best_h.astype(np.float64), float(final_sc), n
