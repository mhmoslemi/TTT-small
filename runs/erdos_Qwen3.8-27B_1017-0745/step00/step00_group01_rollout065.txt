def check_feasible(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < 0) or np.any(h > 1):
        return False

        return False
    return True

def project_capped_simplex(z, n):
    s = n / 2.0
    h_uniform = np.full(n, 0.5)
    if not np.isfinite(z).all() or z.shape != (n,):
        return h_uniform
    z = np.clip(z, -1e12, 1e12)
    h = clip_simplex(z, s)
    for _ in range(n + 2):
        r = s - np.sum(h)
        if abs(r) <= 1e-10 * max(1, n):
            break
        if r > 0:
            slack = 1.0 - h
            idx = np.argmax(slack)
            step = min(r, slack[idx])
        else:
            slack = h
            idx = np.argmax(slack)
            step = min(-r, slack[idx])
        if step < 1e-15:
            break
        h[idx] += np.sign(r) * step
    h = np.clip(h, 0, 1)
    if not check_feasible(h, n):
        return h_uniform
    return h

def clip_simplex(z, s):
    n = len(z)
    lo = np.min(z) - 1.0

        if np.sum(val) > s:

        else:
