def solve_quad(cA, cB, cC):
    roots = []
    if abs(cA) < 1e-15:
        if abs(cB) < 1e-15:
            return roots
        r = -cC / cB
        if np.isfinite(r):
            roots.append(r)
        return roots
    disc = cB*cB - 4.0*cA*cC
    if disc < 0.0:
        tol = 64.0 * np.finfo(float).eps * max(1.0, cB*cB, abs(4.0*cA*cC))
        if disc < -tol:
            return roots
        disc = 0.0
    sq = sqrt(disc)
    q = -0.5*(cB + copysign(sq, cB if cB != 0 else 1.0))
    if abs(q) > 1e-30:
        for r in (q/cA, cC/q):
            if np.isfinite(r):
                roots.append(r)
    else:

import numpy as np
from math import copysign, sqrt

def solve_quad(cA, cB, cC):
    roots = []
    if abs(cA) < 1e-15:
        if abs(cB) < 1e-15:
            return roots
        r = -cC / cB
        if np.isfinite(r):
            roots.append(r)
        return roots
    disc = cB*cB - 4.0*cA*cC
    if disc < 0.0:
        tol = 64.0 * np.finfo(float).eps * max(1.0, cB*cB, abs(4.0*cA*cC))
        if disc < -tol:
            return roots
        disc = 0.0
    sq = sqrt(disc)
    q = -0.5*(cB + copysign(sq, cB if cB != 0 else 1.0))
    if abs(q) > 1e-30:
        for r in (q/cA, cC/q):
            if np.isfinite(r):
                roots.append(r)
    else:
        r = -cB/(2.0*cA)
        if np.isfinite(r):
            roots.append(r)
    return roots

def exact_score_corr(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n), corr

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and h.min() >= -1e-12 and h.max() <= 1.0 + 1e-12
            and abs(h.sum() - n / 2.0) <= 1e-10 * max(1, n))

def project_box_sum(v, S):
    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    lo, hi = float(v.min()) - 1.0, float(v.max())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    res = S - float(y.sum())
    if abs(res) > 1e-12:
        for i in range(n):
            if res > 0 and y[i] < 1.0:
                d = min(1.0 - y[i], res); y[i] += d; res -= d
                if abs(res) < 1e-13: break
            elif res < 0 and y[i] > 0.0:
                d = min(y[i], -res); y[i] -= d; res += d
                if abs(res) < 1e-13: break
    if not np.all(np.isfinite(y)) or y.min() < -1e-10 or y.max() > 1.0 + 1e-10:
        return None
    if abs(float(y.sum()) - S) > 1e-10 * max(1, n):
        return None
    return y

def make_BD(h, a, b):
    n = len(h)
    B = np.zeros(2 * n - 1)
    D = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        idx = k + n - 1
        bk = 0.0
        if 0 <= a - k < n: bk += 1.0 - h[a - k]
        if 0 <= b - k < n: bk -= 1.0 - h[b - k]
        if 0 <= a + k < n: bk -= h[a + k]
        if 0 <= b + k < n: bk += h[b + k]
        B[idx] = bk
        if k == 0: D[idx] = -2.0
        elif k == a - b: D[idx] = 1.0
        elif k == b - a: D[idx] = 1.0
    return B, D

def build_hull(corr, B, Tset, lo, hi):
    n2 = len(corr)
    items = []
    for k in range(n2):
        if k not in Tset:
            items.append((B[k], -corr[k]))
    if not items:
        return [(lo, hi, 0.0, 0.0)]
    items.sort()
    dm, dc, prev = [], [], None
    for sl, n_ic in items:
        if sl == prev:
            continue
        dm.append(sl); dc.append(-n_ic); prev = sl
    hm, hc, hs = [], [], []
    for i in range(len(dm)):
        m, c = dm[i], dc[i]
        if not hm:
            hm.append(m); hc.append(c); hs.append(-1e18)
        else:
            xc = (hc[-1] - c) / (m - hm[-1])
            while len(hm) > 1 and xc <= hs[-1] + 1e-14:
                hm.pop(); hc.pop(); hs.pop()
                xc = (hc[-1] - c) / (m - hm[-1])
            hm.append(m); hc.append(c); hs.append(xc)
    segs = []
    for i in range(len(hm)):
        s = max(hs[i], lo)
        e = hi if i == len(hm) - 1 else min(hs[i + 1], hi)
        if s < e - 1e-15:
            segs.append((s, e, hm[i], hc[i]))
    if not segs:
        segs = [(lo, hi, 0.0, 0.0)]
    return segs

def line_search(h, corr, score, a, b):
    n = len(h)
    lo = max(-h[a], h[b] - 1.0)
    hi = min(1.0 - h[a], h[b])
    if hi - lo < 1e-14:
        return h, corr, score
    B, D = make_BD(h, a, b)
    T = [n - 1, a - b + n - 1, b - a + n - 1]
    Tset = set(T)
    segs = build_hull(corr, B, Tset, lo, hi)
    scored = []
    for (s, e, m, c) in segs:
        cand_al = [s, e]
        for t in T:
            if D[t] != 0.0:
                st = -B[t] / (2.0 * D[t])
                if s - 1e-12 <= st <= e + 1e-12:
                    cand_al.append(max(s, min(e, st)))
        for i in range(3):
            for j in range(i + 1, 3):
                for r in solve_quad(D[T[i]] - D[T[j]], B[T[i]] - B[T[j]], corr[T[i]] - corr[T[j]]):
                    if s - 1e-12 <= r <= e + 1e-12:
                        cand_al.append(max(s, min(e, r)))
        for t in T:
            if D[t] != 0.0:
                for r in solve_quad(D[t], B[t] - m, corr[t] - c):
                    if s - 1e-12 <= r <= e + 1e-12:
                        cand_al.append(max(s, min(e, r)))
        if lo <= 0.0 <= hi:
            cand_al.append(0.0)
        for al in cand_al:
            if al < lo - 1e-12 or al > hi + 1e-12:
                continue
            al = max(lo, min(hi, al))
            val = m * al + c
            for t in T:
                v = corr[t] + B[t] * al + D[t] * al * al
                if v > val:
                    val = v
            scored.append((val, al))
    scored.sort()
    top5, seen = [], []
    for val, al in scored:
        dup = False
        for sa in seen:
            if abs(al - sa) < 1e-12 * max(1.0, abs(al)):
                dup = True; break
        if not dup:
            top5.append(al); seen.append(al)
            if len(top5) >= 5:
                break
    best_h, best_corr, best_sc = h, corr, score
    for al in top5:
        if abs(al) < 1e-14:
            continue
        hn = h.copy()
        hn[a] = max(0.0, min(1.0, hn[a] + al))
        hn[b] = max(0.0, min(1.0, hn[b] - al))
        if not is_feasible(hn, n):
            continue
        ns, nc = exact_score_corr(hn)
        if ns < best_sc - 1e-12 * max(1.0, abs(best_sc)):
            best_h, best_corr, best_sc = hn, nc, ns
    return best_h, best_corr, best_sc

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    n = 45
    parent = None
    if 'initial_h_values' in globals() and globals()['initial_h_values'] is not None:
        try:
            pv = np.array(globals()['initial_h_values'], dtype=np.float64).ravel()
            if 2 <= pv.size <= 999 and np.all(np.isfinite(pv)) and pv.min() >= -1e-6 and pv.max() <= 1.0 + 1e-6:
                n = pv.size
                S_tmp = n / 2.0
                parent = project_box_sum(pv, S_tmp)
                if parent is None or not is_feasible(parent, n):
                    parent = None
        except Exception:
            parent = None
    S = n / 2.0
    if parent is None:
        n = 45
        S = n / 2.0
        parent = np.full(n, 0.5)
    budget = 150000
    starts = [parent.copy()]
    if n <= 100:
        starts.append(parent[::-1].copy())
        starts.append(project_box_sum(parent + rng.normal(0, 0.05, n), S))
        starts.append(project_box_sum(parent + rng.normal(0, 0.18, n), S))
        starts.append(project_box_sum(rng.uniform(0, 1, n), S))
        starts.append(project_box_sum(rng.uniform(0, 1, n), S))
        x = np.arange(n, dtype=np.float64)
        starts.append(project_box_sum(0.5 + 0.45 * np.cos(2 * np.pi * x / n + 0.37), S))
        blk = np.zeros(n)
        hk = n // 2
        blk[:hk] = 1.0
        if n % 2 == 1:
            blk[hk] = 0.5
        starts.append(project_box_sum(blk, S))
    elif n <= 300:
        starts.append(parent[::-1].copy())
        starts.append(project_box_sum(parent + rng.normal(0, 0.12, n), S))
        starts.append(project_box_sum(rng.uniform(0, 1, n), S))
        x = np.arange(n, dtype=np.float64)
        starts.append(project_box_sum(0.5 + 0.45 * np.cos(2 * np.pi * x / n + 0.37), S))
    else:
        starts.append(parent[::-1].copy())
        starts.append(project_box_sum(rng.uniform(0, 1, n), S))
    valid_starts = []
    for s in starts:
        if s is not None and is_feasible(s, n):
            dup = False
            for fs in valid_starts:
                if float(np.max(np.abs(s - fs))) <= 1e-12:
                    dup = True; break
            if not dup:
                valid_starts.append(s)
    if not valid_starts:
        valid_starts = [parent.copy()]
    n_sweeps = 3 if n <= 100 else (2 if n <= 300 else 1)
    npairs_total = n * (n - 1) // 2
    pair_cap = min(npairs_total, max(1, 30000000 // (5 * n * n)))
    best_h, best_corr, best_score = None, None, None
    for st in valid_starts:
        if budget < 10:
            break
        s0, c0 = exact_score_corr(st)
        budget -= 1
        if best_score is None or s0 < best_score - 1e-12:
            best_h, best_corr, best_score = st.copy(), c0.copy(), s0
        lh, lc, ls = st.copy(), c0.copy(), s0
        for sw in range(n_sweeps):
            if budget < pair_cap * 5 + 5:
                break
            pairs = [(a, b) for a in range(n) for b in range(a + 1, n)]
            if len(pairs) > pair_cap:
                sel = rng.choice(len(pairs), size=pair_cap, replace=False)
                pairs = [pairs[i] for i in sel]
            rng.shuffle(pairs)
            improved = False
            for (a, b) in pairs:
                if budget < 5:
                    break
                nh, nc, ns = line_search(lh, lc, ls, a, b)
                budget -= 5
                if ns < ls - 1e-12 * max(1.0, abs(ls)):
                    lh, lc, ls = nh, nc, ns
                    improved = True
                    if best_score is None or ls < best_score - 1e-12:
                        best_h, best_corr, best_score = lh.copy(), lc.copy(), ls
            if not improved:
                break
            if abs(float(lh.sum()) - S) > 1e-12 * max(1, n):
                lp = project_box_sum(lh, S)
                if lp is not None and is_feasible(lp, n):
                    lh = lp
                    if budget > 0:
                        ls, lc = exact_score_corr(lh)
                        budget -= 1
    if best_h is None:
        best_h = np.full(n, 0.5, dtype=np.float64)
    ph = project_box_sum(best_h, S)
    if ph is not None and is_feasible(ph, n):
        best_h = ph
    fs = float(exact_score_corr(best_h)[0])
    return best_h.astype(np.float64), fs, int(n)
