            mask[int(rng.integers(n))] = True
            w = np.where(mask, v, fpop[i])
            trial = project_box_mass(w, n, S)
            if trial is None:
                trial = fpop[i].copy()
            ts = exact_score(trial, n)
            if ts < fsc[i] - 1e-12:
                npop[i] = trial; nsc[i] = ts; nF[i] = F; nCR[i] = CR
                if ts < arch_score - 1e-12:
                    arch_h = trial.copy(); arch_score = ts

        pop = npop; sc = nsc; Fv = nF; CRv = nCR

        found = False
        for i in range(P):
            if np.max(np.abs(pop[i] - arch_h)) < 1e-12:
                found = True; break
        if not found:
            wi = int(np.argmax(sc))
            pop[wi] = arch_h.copy(); sc[wi] = arch_score
            Fv[wi] = 0.6; CRv[wi] = 0.5

        if arch_score >= prev_arch - 1e-12:
            stagnation += 1
        else:
            stagnation = 0

        if stagnation >= rth and n_restarts < MR:
            n_restarts += 1
            stagnation = 0
            si = 0.06 if n_restarts == 1 else 0.18
            rank = np.argsort(sc)
            keep = list(rank[:math.ceil(P / 2)])
            rep = [j for j in range(P) if j not in keep]
            for idx, j in enumerate(rep):
                if idx % 2 == 0:
                    c = gen_candidate(rng, n, S, arch_h, 0, si)
                else:
                    c = gen_candidate(rng, n, S, base, int(rng.integers(5)), 0.15)
                if c is None:
                    c = fallback.copy()
                pop[j] = c; sc[j] = exact_score(c, n)
                Fv[j] = float(rng.uniform(0.4, 0.9)); CRv[j] = float(rng.uniform(0.25, 0.95))
                if sc[j] < arch_score - 1e-12:
                    arch_h = c.copy(); arch_score = sc[j]

    # Final selection
    final_h = arch_h.copy()
    final_s = exact_score(final_h, n)
    if not check_feasible(final_h, n, S):
        fixed = project_box_mass(final_h, n, S)
        if fixed is not None and check_feasible(fixed, n, S):
            final_h = fixed; final_s = exact_score(final_h, n)
        else:
            final_h = fallback; final_s = fb_score

    fb_s = exact_score(fallback, n)
    if check_feasible(fallback, n, S) and fb_s < final_s - 1e-12:
        final_h = fallback.copy(); final_s = fb_s

    if not check_feasible(final_h, n, S):
        final_h = fallback.copy(); final_s = fb_score
        if not check_feasible(final_h, n, S):
            final_h = 0.5 * np.ones(n, dtype=np.float64)
            final_s = exact_score(final_h, n)

    final_s = exact_score(final_h, n)
    if not np.all(np.isfinite(final_h)) or np.any(final_h < 0) or np.any(final_h > 1) or abs(float(np.sum(final_h)) - S) > 1e-10:
        final_h = 0.5 * np.ones(n, dtype=np.float64)
        final_s = exact_score(final_h, n)

    return final_h.copy(), final_s, int(n)
