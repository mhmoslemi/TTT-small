    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    for j in range(M):
        s = j - (n - 1)
        i_lo = max(0, -s)
        i_hi = min(n - 1, n - 1 - s)
        if i_lo > i_hi:
            continue
        idx = np.arange(i_lo, i_hi + 1)
        q[j] = dx * np.sum(h[idx] * (1.0 - h[idx + s]))
        # Gradient row
        # Part 1: (1 - h[k+s]) for k where 0<=k+s<n, i.e., k in [max(0,-s), min(n-1,n-1-s)]
        k1_lo = max(0, -s)
        k1_hi = min(n - 1, n - 1 - s)
        if k1_lo <= k1_hi:
            k1 = np.arange(k1_lo, k1_hi + 1)
            G[j, k1] += dx * (1.0 - h[k1 + s])
        # Part 2: -h[k-s] for k where 0<=k-s<n, i.e., k in [max(0,s), min(n-1,n-1+s)]
        k2_lo = max(0, s)
        k2_hi = min(n - 1, n - 1 + s)
        if k2_lo <= k2_hi:
            k2 = np.arange(k2_lo, k2_hi + 1)
            G[j, k2] -= dx * h[k2 - s]

        q[j] = dx * np.dot(h[idx], 1.0 - h[idx + s])
        k1_lo = max(0, -s)
        k1_hi = min(n - 1, n - 1 - s)
        if k1_lo <= k1_hi:
            k1 = np.arange(k1_lo, k1_hi + 1)
            G[j, k1] += dx * (1.0 - h[k1 + s])
        k2_lo = max(0, s)
        k2_hi = min(n - 1, n - 1 + s)
        if k2_lo <= k2_hi:
            k2 = np.arange(k2_lo, k2_hi + 1)
            G[j, k2] -= dx * h[k2 - s]
    return q, G

def _exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _project(v):
    n = len(v)
    S = 0.5 * n
    v = np.clip(v.astype(np.float64), -1e6, 1e6)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(v - mid, 0.0, 1.0)
        if float(np.sum(h)) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - float(np.sum(h))
    if abs(r) > 1e-12:
        if r > 0:
            room = 1.0 - h
            tot = float(np.sum(room))
            if tot > 1e-15:
                h = np.clip(h + r * room / tot, 0.0, 1.0)
        else:
            tot = float(np.sum(h))
            if tot > 1e-15:
                h = np.clip(h + r * h / tot, 0.0, 1.0)
    r2 = S - float(np.sum(h))
    if abs(r2) > 1e-10:
        j = n // 2
        if 0.0 <= h[j] + r2 <= 1.0:
            h[j] += r2
    if not np.all((h >= 0.0) & (h <= 1.0)) or abs(float(np.sum(h)) - S) > 1e-9:
        h = np.full(n, 0.5)
    return h

def _soft_fun(h, tau):
    q, G = _lag_qG(h)
    qmax = float(np.max(q))
    shifted = (q - qmax) / tau
    exps = np.exp(shifted)
    total = float(np.sum(exps))
    f = qmax + tau * np.log(total)
    g = G.T @ (exps / total)
    return f, g

def _eq_fun(x):
    return float(np.sum(x)) - 0.5 * len(x)

def _eq_jac(x):
    return np.ones(len(x))

def _epi_obj(z):
    n = len(z) - 1
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return z[-1], g

def _epi_eq_fun(z):
    n = len(z) - 1
    return float(np.sum(z[:n])) - 0.5 * n

def _epi_eq_jac(z):
    n = len(z) - 1
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def _epi_ineq_fun(z):
    h = z[:len(z) - 1]
    q, _ = _lag_qG(h)
    return z[-1] - q

def _epi_ineq_jac(z):
    h = z[:len(z) - 1]
    n = len(h)
    _, G = _lag_qG(h)
    M = G.shape[0]
    J = np.zeros((M, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def _block_vec(n):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    if n % 2 == 1:
        h[n // 2] = 0.5
    return h

def _alt_vec(n):
    return np.where(np.arange(n) % 2 == 0, 1.0, 0.0)

def run(seed=42, **kwargs):
    n = 42
    parent = None
    if 'initial_h_values' in globals():
        try:
            p = np.array(initial_h_values, dtype=np.float64).ravel()
            if p.size >= 2 and np.all(np.isfinite(p)):
                n = p.size
                parent = p.copy()
        except Exception:
            pass
    if parent is None:
        nk = kwargs.get('n_points', None)
        if nk is not None and 2 <= int(nk) <= 256:
            n = int(nk)
    h_fb = np.full(n, 0.5)
    s_fb = _exact_score(h_fb)
    h_inc = _project(parent) if parent is not None else h_fb.copy()
    s_inc = _exact_score(h_inc)
    if n > 256:
        return (h_inc, float(s_inc), n)
    rng = np.random.default_rng(seed)
    starts = [h_inc]
    if parent is not None:
        for sig in [0.10, 0.25, 0.50]:
            starts.append(_project(parent + sig * rng.standard_normal(n)))
    starts.append(_project(_block_vec(n)))
    starts.append(_project(_alt_vec(n)))
    for _ in range(3):
        starts.append(_project(rng.random(n)))
    if parent is None:
        idx = np.arange(n)
        starts.append(_project(0.5 + 0.45 * np.sin(2.0 * np.pi * idx / n)))
    dedup = [starts[0]]
    for s in starts[1:]:
        dup = False
        for d in dedup:
            if float(np.max(np.abs(s - d))) < 1e-10:
                dup = True
                break
        if not dup:
            dedup.append(s)
    starts = dedup
    bounds = [(0.0, 1.0)] * n
    eq_con = {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}
    epi_eq = {'type': 'eq', 'fun': _epi_eq_fun, 'jac': _epi_eq_jac}
    epi_in = {'type': 'ineq', 'fun': _epi_ineq_fun, 'jac': _epi_ineq_jac}
    z_bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    taus = [0.02, 0.005, 0.001, 0.0002]
    for h_start in starts:
        h_cur = h_start.copy()
        for tau in taus:
            try:
                res = minimize(_soft_fun, h_cur, method='SLSQP', jac=True,
                              args=(tau,), bounds=bounds, constraints=[eq_con],
                              options={'ftol': 1e-10, 'maxiter': 120})
                if res.x is not None and res.x.shape == (n,) and np.all(np.isfinite(res.x)):
                    h_new = _project(res.x)
                    s_new = _exact_score(h_new)
                    if s_new < s_inc - 1e-12:
                        h_inc = h_new
                        s_inc = s_new
                    f_new = _soft_fun(h_new, tau)[0]
                    f_old = _soft_fun(h_cur, tau)[0]
                    if f_new <= f_old + 1e-12:
                        h_cur = h_new
            except Exception:
                pass
        try:
            q0, _ = _lag_qG(h_cur)
            t0 = float(np.max(q0)) + max(1e-8, 1e-6 * abs(float(np.max(q0))))
            z0 = np.append(h_cur, t0)
            res = minimize(_epi_obj, z0, method='SLSQP', jac=True,
                          bounds=z_bounds, constraints=[epi_eq, epi_in],
                          options={'ftol': 1e-10, 'maxiter': 200})
            if res.x is not None and res.x.shape == (n + 1,) and np.all(np.isfinite(res.x)):
                h_new = _project(res.x[:n])
                s_new = _exact_score(h_new)
                if s_new < s_inc - 1e-12:
                    h_inc = h_new
                    s_inc = s_new
        except Exception:
            pass
    try:
        q0, _ = _lag_qG(h_inc)
        t0 = float(np.max(q0)) + max(1e-8, 1e-6 * abs(float(np.max(q0))))
        z0 = np.append(h_inc, t0)
        res = minimize(_epi_obj, z0, method='SLSQP', jac=True,
                      bounds=z_bounds, constraints=[epi_eq, epi_in],
                      options={'ftol': 1e-10, 'maxiter': 200})
        if res.x is not None and res.x.shape == (n + 1,) and np.all(np.isfinite(res.x)):
            h_new = _project(res.x[:n])
            s_new = _exact_score(h_new)
            if s_new < s_inc - 1e-12:
                h_inc = h_new
                s_inc = s_new
    except Exception:
        pass
    h_inc = _project(h_inc)
    s_inc = _exact_score(h_inc)
    if not np.all((h_inc >= 0.0) & (h_inc <= 1.0)) or abs(float(np.sum(h_inc)) - 0.5 * n) > 1e-9:
        h_inc = h_fb
        s_inc = s_fb
    return (h_inc, float(s_inc), n)
