def exact_score(h, n):
    if h.ndim != 1 or len(h) != n:
        return float('inf')
    if not np.all(np.isfinite(h)):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr) * (2.0 / n))

def project_to_K(y, n):
    if not np.all(np.isfinite(y)) or len(y) != n:
        return None
    S = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0

        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = float(np.sum(x))
        if s > S:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(y - lam, 0.0, 1.0)

        return None

        slack = 1.0 - x

            residual -= add

    R = b - m

def build_tree(n):
    records = []
    def rec(a, b, depth):
        if b - a <= 1:
            return
        m = (a + b) // 2
        records.append((depth, a, m, b))
        rec(a, m, depth + 1)
        rec(m, b, depth + 1)
    rec(0, n, 0)
    # sort by depth
    records.sort(key=lambda r: r[0])
    return records

        return None
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    S = n / 2.0
    x = np.clip(h, 0.0, 1.0).astype(np.float64)
    residual = S - float(np.sum(x))
    if abs(residual) > 1e-8:
        return None
    if residual > 0:
        order = np.argsort(x)
        for idx in order:
            add = min(residual, 1.0 - x[idx])
            x[idx] += add
            residual -= add
            if abs(residual) < 1e-14:
                break
    elif residual < 0:
        order = np.argsort(-x)
        for idx in order:
            rem = min(-residual, x[idx])
            x[idx] -= rem
            residual += rem
            if abs(residual) < 1e-14:
                break
    if np.any(x < 0.0) or np.any(x > 1.0) or abs(float(np.sum(x)) - S) > 1e-10:
        return None
    return x

def make_direction(a, m, b, n):
    d = np.zeros(n, dtype=np.float64)
    L = m - a
    R = b - m
    d[a:m] = 1.0 / L
    d[m:b] = -1.0 / R
    s = float(np.sum(d))
    if abs(s) > 1e-15:
        nz = np.nonzero(d)[0]
        d[nz[0]] -= s
    return d

def tree_sort_key(r):
    return r[0]

def build_tree(n):
    records = []
    stack = [(0, n, 0)]
    while stack:
        a, b, depth = stack.pop()
        if b - a <= 1:
            continue
        m = (a + b) // 2
        records.append((depth, a, m, b))
        stack.append((m, b, depth + 1))
        stack.append((a, m, depth + 1))
    records.sort(key=tree_sort_key)
    return records

def exact_line_min(q, b, c, t_lo, t_hi):
    t_candidates = [t_lo, t_hi, 0.0]
    m = len(q)
    for r in range(m):
        for s in range(r + 1, m):
            A = c[r] - c[s]
            B = b[r] - b[s]
            C = q[r] - q[s]
            tol = 1e-14 * max(1.0, abs(B), abs(C))
            if abs(A) < tol:
                if abs(B) < tol:
                    continue
                t = -C / B
                if np.isfinite(t) and t_lo - 1e-12 <= t <= t_hi + 1e-12:
                    t_candidates.append(float(t))
            else:
                disc = B * B - 4.0 * A * C
                neg_tol = 1e-14 * (B * B + abs(4.0 * A * C) + 1e-300)
                if disc < -neg_tol:
                    continue
                if disc < 0.0:
                    disc = 0.0
                sq = math.sqrt(disc)
                qv = -0.5 * (B + math.copysign(sq, B))
                t1 = qv / A
                t2 = C / qv if abs(qv) > 1e-300 else -B / (2.0 * A)
                for t in (t1, t2):
                    if np.isfinite(t) and t_lo - 1e-12 <= t <= t_hi + 1e-12:
                        t_candidates.append(float(t))
        if abs(c[r]) > 1e-300:
            t_st = -b[r] / (2.0 * c[r])
            if np.isfinite(t_st) and t_lo - 1e-12 <= t_st <= t_hi + 1e-12:
                t_candidates.append(float(t_st))
    t_candidates.sort()
    deduped = []
    for t in t_candidates:
        if not deduped or abs(t - deduped[-1]) > 1e-12 * max(1.0, abs(t)):
            deduped.append(t)
    clamped = [max(t_lo, min(t_hi, t)) for t in deduped]
    if not clamped:
        return 0.0
    best_t = clamped[0]
    best_val = float('inf')
    chunk = 256
    for i in range(0, len(clamped), chunk):
        ts = np.array(clamped[i:i+chunk], dtype=np.float64)
        vals = q[None, :] + ts[:, None] * b[None, :] + ts[:, None]**2 * c[None, :]
        row_max = np.max(vals, axis=1)
        for j in range(len(ts)):
            if row_max[j] < best_val:
                best_val = float(row_max[j])
                best_t = clamped[i + j]
    return best_t

def grid_line_min(q, b, c, t_lo, t_hi):
    ts = np.sort(np.unique(np.concatenate([np.linspace(t_lo, t_hi, 17), [0.0]])))
    vals = q[None, :] + ts[:, None] * b[None, :] + ts[:, None]**2 * c[None, :]
    row_max = np.max(vals, axis=1)
    bi = int(np.argmin(row_max))
    if bi == 0:
        lo_i, hi_i = 0, 1
    elif bi == len(ts) - 1:
        lo_i, hi_i = len(ts) - 2, len(ts) - 1
    else:
        lo_i, hi_i = bi - 1, bi + 1
    ts2 = np.linspace(ts[lo_i], ts[hi_i], 9)
    ts_all = np.sort(np.concatenate([ts, ts2]))
    vals2 = q[None, :] + ts_all[:, None] * b[None, :] + ts_all[:, None]**2 * c[None, :]
    row_max2 = np.max(vals2, axis=1)
    bi2 = int(np.argmin(row_max2))
    return float(ts_all[bi2])

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    n = None
    parent = None
    try:
        pr = globals().get('initial_h_values', None)
        if pr is not None:
            p = np.asarray(pr, dtype=np.float64).ravel()
            if 2 <= len(p) <= 999 and np.all(np.isfinite(p)) and float(np.max(np.abs(p))) <= 1e6:
                n = len(p)
                parent = p.copy()
    except Exception:
        parent = None
        n = None
    if n is None:
        try:
            nk = kwargs.get('n_points', None)
            if nk is not None and isinstance(nk, int) and not isinstance(nk, bool) and 2 <= nk <= 999:
                n = nk
        except Exception:
            pass
    if n is None:
        n = 45
    dx = 2.0 / n
    S = n / 2.0
    if parent is not None:
        proj = project_to_K(parent, n)
        fallback_h = proj.copy() if proj is not None else np.full(n, 0.5, dtype=np.float64)
    else:
        fallback_h = np.full(n, 0.5, dtype=np.float64)
    fallback_score = exact_score(fallback_h, n)
    tree = build_tree(n)
    if n <= 48:
        max_sweeps, use_exact = 3, True
    elif n <= 128:
        max_sweeps, use_exact = 3, False
    elif n <= 256:
        max_sweeps, use_exact = 2, False
    else:
        max_sweeps, use_exact = 1, False
    if n > 256 and len(tree) > 64:
        idx = [int(k * (len(tree) - 1) / 63) for k in range(64)]
        tree_use = [tree[i] for i in idx]
    else:
        tree_use = tree
    starts = [fallback_h.copy()]
    if parent is not None:
        comp = project_to_K(1.0 - parent, n)
        if comp is not None:
            starts.append(comp)
        noisy = project_to_K(parent + 0.12 * rng.standard_normal(n), n)
        if noisy is not None:
            starts.append(noisy)
        bp = project_to_K(rng.beta(0.5, 0.5, n), n)
        if bp is not None:
            starts.append(bp)
    else:
        for _ in range(2):
            pj = project_to_K(rng.uniform(0, 1, n), n)
            if pj is not None:
                starts.append(pj)
        mf = int(rng.integers(1, max(1, min(4, n // 2)) + 1))
        ph = rng.uniform(0, 2 * math.pi)
        cv = 0.5 + 0.35 * np.cos(2 * math.pi * mf * (np.arange(n) + 0.5) / n + ph)
        pj = project_to_K(cv, n)
        if pj is not None:
            starts.append(pj)
    max_starts = 4 if n <= 128 else (3 if n <= 256 else 2)
    starts = starts[:max_starts]
    deduped = [starts[0]]
    for s in starts[1:]:
        if all(float(np.max(np.abs(s - d))) > 1e-8 for d in deduped):
            deduped.append(s)
    starts = deduped
    best_h = fallback_h.copy()
    best_score = fallback_score
    for start in starts:
        try:
            cur_h = start.copy()
            cur_score = exact_score(cur_h, n)
            if not np.isfinite(cur_score):
                continue
            if cur_score < best_score - 1e-12:
                best_h, best_score = cur_h.copy(), cur_score
            for sweep in range(max_sweeps):
                improved = False
                order = list(range(len(tree_use)))
                if sweep % 2 == 1:
                    order = order[::-1]
                for di in order:
                    try:
                        dep, a, m, b = tree_use[di]
                        d = make_direction(a, m, b, n)
                        nz = np.nonzero(d)[0]
                        t_lo, t_hi = -np.inf, np.inf
                        for i in nz:
                            if d[i] > 0:
                                t_lo = max(t_lo, -cur_h[i] / d[i])
                                t_hi = min(t_hi, (1.0 - cur_h[i]) / d[i])
                            else:
                                t_lo = max(t_lo, (1.0 - cur_h[i]) / d[i])
                                t_hi = min(t_hi, -cur_h[i] / d[i])
                        if not (np.isfinite(t_lo) and np.isfinite(t_hi)) or t_hi - t_lo < 1e-12:
                            continue
                        q = dx * np.correlate(cur_h, 1.0 - cur_h, mode='full')
                        bc = dx * (np.correlate(d, 1.0 - cur_h, mode='full') - np.correlate(cur_h, d, mode='full'))
                        cc = -dx * np.correlate(d, d, mode='full')
                        if not (np.all(np.isfinite(q)) and np.all(np.isfinite(bc)) and np.all(np.isfinite(cc))):
                            continue
                        t_opt = exact_line_min(q, bc, cc, t_lo, t_hi) if use_exact else grid_line_min(q, bc, cc, t_lo, t_hi)
                        pred = float(np.max(q + t_opt * bc + t_opt * t_opt * cc))
                        if pred >= cur_score - 1e-12:
                            continue
                        h_trial = cur_h + t_opt * d
                        h_rep = repair_small(h_trial, n)
                        if h_rep is None:
                            continue
                        s_rep = exact_score(h_rep, n)
                        if not np.isfinite(s_rep) or s_rep >= cur_score - 1e-12:
                            continue
                        cur_h, cur_score = h_rep, s_rep
                        improved = True
                        if cur_score < best_score - 1e-12:
                            best_h, best_score = cur_h.copy(), cur_score
                    except Exception:
                        continue
                if not improved or best_score - cur_score < 1e-10:
                    break
        except Exception:
            continue
    final_h = best_h
    final_score = exact_score(final_h, n)
    valid = (np.isfinite(final_score) and np.all(final_h >= 0.0) and np.all(final_h <= 1.0)
             and abs(float(np.sum(final_h)) - S) <= 1e-10)
    if not valid:
        final_h = fallback_h
        final_score = exact_score(final_h, n)
        valid = (np.isfinite(final_score) and np.all(final_h >= 0.0) and np.all(final_h <= 1.0)
                 and abs(float(np.sum(final_h)) - S) <= 1e-10)
    if not valid:
        final_h = np.full(n, 0.5, dtype=np.float64)
        final_score = exact_score(final_h, n)
    return final_h.copy(), float(final_score), int(n)
