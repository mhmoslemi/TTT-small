import numpy as np

from operator import itemgetter

_N, _DX, _S = 0, 0.0, 0.0

def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return 1e18
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_k(y, S, n):
    if not np.all(np.isfinite(y)):
        return 0.5 * np.ones(n)
    lo = float(np.min(y)) - 1.5
    hi = float(np.max(y)) + 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.where(h < 0.999 if diff > 0 else h > 0.001)[0]
        if len(idx) == 0:
            idx = np.arange(n)
        h[idx] = np.clip(h[idx] + diff / len(idx), 0.0, 1.0)
    if abs(S - np.sum(h)) > 1e-10:
        h = np.clip(h + (S - np.sum(h)) / n, 0.0, 1.0)
        if abs(S - np.sum(h)) > 1e-10:
            return 0.5 * np.ones(n)
    return h

def lse_val(h, dx, tau):
    q = np.correlate(h, 1.0 - h, mode="full") * dx
    m = float(np.max(q))
    return m + tau * np.log(np.sum(np.exp((q - m) / tau)))

def lse_grad(h, dx, tau):
    n = len(h)
    q = np.correlate(h, 1.0 - h, mode="full") * dx
    m = float(np.max(q))
    w = np.exp((q - m) / tau)
    p = w / np.sum(w)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-15:
            continue
        ell = k - (n - 1)
        lo = max(0, -ell)
        hi = min(n, n - ell)
        g[lo + ell:hi + ell] += p[k] * dx * (1.0 - h[lo:hi])
        g[lo:hi] -= p[k] * dx * h[lo + ell:hi + ell]
    return g

def proj_grad(h0, dx, S, n, taus):
    h = project_k(h0, S, n).copy()
    for tau in taus:
        alpha, stale = 1.0, 0
        for _ in range(80):
            f = lse_val(h, dx, tau)
            g = lse_grad(h, dx, tau)
            if not (np.all(np.isfinite(g)) and np.isfinite(f)):
                break
            y = project_k(h - alpha * g, S, n)
            d = y - h
            if np.max(np.abs(d)) <= 1e-10:
                break
            gd = float(np.dot(g, d))
            if gd >= -1e-14:
                break
            if lse_val(y, dx, tau) <= f + 1e-4 * gd + 1e-14:
                dec = f - lse_val(h, dx, tau)
                h = y
                alpha = min(1.25 * alpha, 5.0)
                stale = stale + 1 if dec < 1e-12 else 0
                if stale >= 10:
                    break
            else:
                alpha *= 0.5
                if alpha < 1e-12:
                    break
    return project_k(h, S, n)
