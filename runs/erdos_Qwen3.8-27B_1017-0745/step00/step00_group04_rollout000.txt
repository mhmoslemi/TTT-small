    if parent is not None and np.all(np.isfinite(parent)):
        hp = project_feasible(parent[:n], n)
        starts.append(hp)
        for sigma in (0.10, 0.25, 0.50):
            starts.append(project_feasible(hp + sigma * rng.standard_normal(n), n))
    hblk = np.zeros(n)
    hblk[:n // 2] = 1.0
    if n % 2 == 1:
        hblk[n // 2] = 0.5
    starts.append(hblk.copy())
    halt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    starts.append(project_feasible(halt, n))
    for _ in range(2):
        starts.append(project_feasible(rng.random(n), n))
    dedup = [starts[0]]
    for s in starts[1:]:
        if all(np.max(np.abs(s - d)) > 1e-10 for d in dedup):
            dedup.append(s)
    taus = (0.02, 0.005, 0.001, 0.0002)
    bounds = [(0.0, 1.0)] * n
    cons_eq = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n,)}]
    for start in dedup:
        cur = start.copy()
        for tau in taus:
            try:
                res = minimize(soft_fun, cur, args=(tau, n), method='SLSQP',
                               jac=soft_jac, bounds=bounds, constraints=cons_eq,
                               options={'ftol': 1e-10, 'maxiter': 120})
                if res.x is not None and len(res.x) == n and np.all(np.isfinite(res.x)):
                    proj = project_feasible(res.x, n)
                    sc = exact_score(proj)
                    if sc < best_score - 1e-12:
                        best_h, best_score = proj, sc
                    v0, _ = soft_value(cur, tau, n)
                    v1, _ = soft_value(proj, tau, n)
                    if v1 <= v0 + 1e-12:
                        cur = proj
            except Exception:
                pass
        cur = epipolish(cur, n)
        sc = exact_score(cur)
        if sc < best_score - 1e-12:
            best_h, best_score = cur, sc
    best_h = epipolish(best_h, n)
    best_score = exact_score(best_h)
    return best_h, float(best_score), int(n)
