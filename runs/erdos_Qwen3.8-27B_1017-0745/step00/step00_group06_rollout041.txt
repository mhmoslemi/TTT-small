                        sc2 += 1
                    pc = project_capped_simplex(raw, N, M)
                    if pc is not None and _dedup_ok(pc, newpop):
                        newpop.append(pc)
                        ok = True
                        break
                if not ok:
                    newpop.append(fallback.copy())
            X = np.array(newpop, dtype=np.float64)
            S = np.array([exact_score(X[i], N) for i in range(P)], dtype=np.float64)
            for i in range(P):
                if S[i] < gbest - 1e-12:
                    gbest = S[i]
                    ginc = X[i].copy()

    fh = project_capped_simplex(ginc, N, M)
    if fh is None or not check_feasible(fh, N):
        fh = fallback.copy()
    if not check_feasible(fh, N):
        fh = np.full(N, 0.5)
    fs = exact_score(fh, N)
    ff = exact_score(fallback, N)
    if ff < fs - 1e-12:
        fh = fallback.copy()
        fs = ff
    return (fh.astype(np.float64), float(fs), int(N))
