        sc_arr = np.array([exact_score(c) for c in candidates])
        order = np.argsort(sc_arr)
        selected = []
        for idx in order:
            c = candidates[idx]
            distinct = True
            for s in selected:
                if np.max(np.abs(c - s)) <= 1e-5:
                    distinct = False
                    break
            if distinct:
                selected.append(c)
            if len(selected) >= 3:
                break
        for h0 in selected:
            hr = slsqp_refine(h0, n, S, dx)
            if hr is not None:
                sc = exact_score(hr)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_h = hr.copy()
                candidates.append(hr.copy())

    best_h = project_K(best_h, S)
    if not is_feasible(best_h, S):
        best_h = np.full(n, 0.5)
    final_score = exact_score(best_h)
    final_score = min(final_score, exact_score(np.full(n, 0.5)))
    if final_score < best_score:
        best_h = np.full(n, 0.5)
        best_score = final_score
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_K(best_h, S)
    best_score = exact_score(best_h)
    return (best_h, best_score, n)
