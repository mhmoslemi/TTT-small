def compute_gradient(h, p, n):
    grad = np.zeros(n)
    # p has length 2n-1
    one_minus_h = 1.0 - h
    for j in range(2*n-1):
        k = j - (n-1)
        pk = p[j]
        if abs(pk) < 1e-16:
            continue
        # First term: I(0<=r-k<n) * (1-h[r-k])
        # r ranges: k<=r<n+k and 0<=r<n
        r1_lo = max(0, k)
        r1_hi = min(n-1, n-1+k)
        if r1_lo <= r1_hi:
            grad[r1_lo:r1_hi+1] += pk * one_minus_h[r1_lo-k:r1_hi-k+1]
        # Second term: I(0<=r+k<n) * h[r+k]
        r2_lo = max(0, -k)
        r2_hi = min(n-1, n-1-k)
        if r2_lo <= r2_hi:
            grad[r2_lo:r2_hi+1] -= pk * h[r2_lo+k:r2_hi+k+1]
    grad /= n
    return grad

import numpy as np

def project_box_sum(v, S):
    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    if np.max(np.abs(v)) > 1e12:
        return None
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    y = np.clip(v - lam, 0.0, 1.0)
    res = S - np.sum(y)
    if abs(res) > 1e-13:
        # Correct residual
        idx = np.arange(n)
        if res > 0:
            cap = 1.0 - y
            for i in idx:
                if res <= 0:
                    break
                add = min(res, cap[i])
                y[i] += add
                res -= add

    q = corr / n
    es = float(np.max(corr) * 2.0 / n)
    z = q / tau
    zm = np.max(z)
    e = np.exp(z - zm)
    p = e / np.sum(e)
    F = tau * (zm + np.log(np.sum(e)))
    grad = compute_gradient(h, p, n)
    if not np.isfinite(F) or not np.all(np.isfinite(grad)):
        return None, None, es
    return F, grad, es

def make_starts(n, rng, parent):
    S = n / 2.0
    raw = []
    if parent is not None:
        raw.append(parent.copy())
        if n <= 100:
            for s in [0.02, 0.05, 0.10, 0.18, 0.30]:
                raw.append(parent + s * rng.standard_normal(n))
            for _ in range(3): raw.append(rng.uniform(0, 1, n))
        elif n <= 300:
            for s in [0.03, 0.10, 0.22]:
                raw.append(parent + s * rng.standard_normal(n))
            for _ in range(2): raw.append(rng.uniform(0, 1, n))
        else:
            raw.append(parent + 0.1 * rng.standard_normal(n))
            raw.append(rng.uniform(0, 1, n))
    else:
        raw.append(np.full(n, 0.5))
        i = np.arange(n)
        for A, m, ph in [(0.2, 1, 0.31), (0.35, 2, 0.77), (0.45, 3, 1.13)]:
            raw.append(0.5 + A * np.cos(2 * np.pi * m * i / n + ph))
        for _ in range(3 if n <= 100 else 2): raw.append(rng.uniform(0, 1, n))
    out = []
    for v in raw:
        pv = project_box_sum(np.asarray(v, dtype=np.float64), S)
        if pv is not None: out.append(pv)
    return out

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    n, S, parent = 45, 22.5, None
    try:
        pv = np.array(globals()['initial_h_values'], dtype=np.float64).flatten()
        if 2 <= len(pv) <= 999 and np.all(np.isfinite(pv)) and np.max(np.abs(pv)) <= 1e12:
            n = len(pv); S = n / 2.0
            proj = project_box_sum(pv, S)
            if proj is not None: parent = proj
    except Exception:
        pass
    if parent is None:
        parent = np.full(45, 0.5); n, S = 45, 22.5
    best_h = parent.copy()
    best_score = exact_score(best_h)
    if best_score is None: best_score = 1.0
    budget = min(50000, max(100, int(1e8 // (n * n))))
    ev = 0
    temps = [0.04, 0.015, 0.006, 0.002, 0.0007, 0.00025, 0.00008]
    mx_it = 60 if n <= 100 else (25 if n <= 300 else 8)
    starts = make_starts(n, rng, parent)
    if not starts: starts = [parent.copy()]
    for st in starts:
        if ev >= budget: break
        h = st.copy()
        for tau in temps:
            if ev >= budget: break
            res = smooth_eval(h, tau)
            if res[0] is None: continue
            F, grad, _ = res; ev += 1
            consec = 0
            stop = False
            for _it in range(mx_it):
                if ev >= budget or stop: break
                gmax = np.max(np.abs(grad))
                if gmax < 1e-13: break
                alpha = min(1e8, 0.08 / gmax)
                acc = False
                for _bt in range(12):
                    if ev >= budget: break
                    y = project_box_sum(h - alpha * grad, S)
                    if y is None:
                        alpha *= 0.5; continue
                    d = y - h
                    direc = np.dot(grad, d)
                    if not np.isfinite(direc) or direc >= 0:
                        alpha *= 0.5; continue
                    ry = smooth_eval(y, tau); ev += 1
                    if ry[0] is None:
                        es = exact_score(y)
                        if es is not None and es < best_score:
                            best_h, best_score = y.copy(), es
                        alpha *= 0.5; continue
                    Fy, gy, esy = ry
                    if esy is not None and esy < best_score:
                        best_h, best_score = y.copy(), esy
                    if Fy <= F + 1e-4 * direc + 1e-13 * max(1.0, abs(F)):
                        fdec = F - Fy
                        h, F, grad = y, Fy, gy
                        acc = True
                        if np.max(np.abs(d)) < 1e-8:
                            stop = True; break
                        if fdec <= 1e-12 * max(1.0, abs(F)):
                            consec += 1
                        else:
                            consec = 0
                        if consec >= 4: stop = True
                        break
                    alpha *= 0.5
                if not acc: break
    if not is_feasible(best_h, n):
        p2 = project_box_sum(best_h, S)
        best_h = p2 if p2 is not None else parent.copy()
    fs = exact_score(best_h)
    if fs is None:
        best_h = parent.copy(); fs = exact_score(best_h)
        if fs is None: fs = best_score
    return best_h.astype(np.float64), float(fs), n
