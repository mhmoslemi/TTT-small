            for e in deduped:
                if np.max(np.abs(sh - e)) < 1e-10:
                    dup = True
                    break
            if not dup:
                deduped.append(sh)
            if len(deduped) >= 6:
                break
    else:
        deduped = []
        for sh in starts:
            dup = False
            for e in deduped:
                if np.max(np.abs(sh - e)) < 1e-10:
                    dup = True
                    break
            if not dup:
                deduped.append(sh)
            if len(deduped) >= 3:
                break

    for sh in deduped:
        sc = exact_score(sh)
        if sc is not None and sc < best_score:
            best_score = sc
            best_h = sh.copy()

    if n <= 128:
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        ineq_c = {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac}
        eq_c = {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}
        for sh in deduped:
            B0, _ = compute_lags(sh)
            t0 = float(np.max(B0)) + 1e-8 * max(1.0, abs(float(np.max(B0))))
            z0 = np.concatenate([sh, [t0]])
            try:
                res = minimize(_obj, z0, jac=_obj_grad,
                               constraints=[ineq_c, eq_c], bounds=bounds,
                               method='SLSQP',
                               options={'maxiter': 180, 'ftol': 1e-9, 'disp': False})
                if res is not None and hasattr(res, 'x') and res.x is not None:
                    zx = res.x
                    if len(zx) == n + 1 and np.all(np.isfinite(zx)):
                        hc = project_capped_simplex(zx[:n], S)
                        if hc is not None and is_feasible(hc, n):
                            sc = exact_score(hc)
                            if sc is not None and sc < best_score:
                                best_score = sc
                                best_h = hc.copy()
            except Exception:
                pass

    if not is_feasible(best_h, n):
        best_h = const_h.copy()
        best_score = exact_score(const_h)
    final_score = exact_score(best_h)
    if final_score is None:
        best_h = const_h
        final_score = exact_score(best_h)
    return best_h, final_score, n
