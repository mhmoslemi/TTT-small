    starts = [base.copy(), base[::-1].copy(), 1.0 - base[::-1].copy(), np.full(n, 0.5)]
    nv = np.zeros(n)
    nv[:n // 2] = 1.0
    if n % 2 == 1:
        nv[n // 2] = 0.5
    starts.append(nv)
    alt = np.zeros(n)
    alt[::2] = 1.0
    starts.append(alt)
    for sd in (0.05, 0.15, 0.35):
        starts.append(base + rng.standard_normal(n) * sd)
    starts.append(rng.uniform(0.0, 1.0, size=n))
    proj = []
    for st in starts:
        ps = project_capped_simplex(st, n)
        dup = False
        for q in proj:
            if np.max(np.abs(ps - q)) < 1e-10:
                dup = True
                break
        if not dup:
            proj.append(ps)
    return proj

def run(seed=42, **kwargs):
    try:
        seed = int(seed)
    except (ValueError, TypeError):
        seed = 42
    rng = np.random.default_rng(seed)
    parent = None
    try:
        pv = np.array(initial_h_values, dtype=np.float64).ravel()
        if pv.size > 0 and np.all(np.isfinite(pv)):
            parent = pv.copy()
    except Exception:
        parent = None
    n_in = kwargs.get('n_points', None)
    if parent is not None:
        n = len(parent)
    elif n_in is not None:
        try:
            n = int(n_in)
            if n < 1 or n > 999:
                n = 47
        except (ValueError, TypeError):
            n = 47
    else:
        n = 47
    if parent is None:
        base = np.full(n, 0.5, dtype=np.float64)
    else:
        base = project_capped_simplex(parent, n)
    fallback = np.full(n, 0.5, dtype=np.float64)
    fallback_score = exact_score(fallback)
    if n > 999:
        sc = exact_score(base)
        if sc > fallback_score:
            return fallback, fallback_score, n
        return base, sc, n
    all_starts = build_starts(base, n, rng)
    cap_s = 10 if n <= 64 else (5 if n <= 256 else 2)
    cap_i = 100 if n <= 64 else (60 if n <= 256 else 25)
    max_stars = min(len(all_starts), cap_s)
    max_iters = cap_i
    try:
        rs = int(kwargs.get('restarts', 0))
        c2 = 12 if n <= 64 else (6 if n <= 256 else 2)
        if rs > 0:
            max_stars = min(max_stars, max(1, min(rs, c2)))
    except Exception:
        pass
    try:
        ip = int(kwargs.get('iters_per_tau', 0))
        c3 = 150 if n <= 64 else (90 if n <= 256 else 40)
        if ip > 0:
            max_iters = min(max_iters, max(1, min(ip, c3)))
    except Exception:
        pass
    temperatures = [0.02, 0.005, 0.001, 0.0002, 0.00005, 0.00001]
    incumbent = base.copy()
    incumbent_score = exact_score(incumbent)
    if not np.isfinite(incumbent_score):
        incumbent = fallback.copy()
        incumbent_score = fallback_score
    for st in all_starts[:max_stars]:
        sc = exact_score(st)
        if sc < incumbent_score - 1e-12:
            incumbent = st.copy()
            incumbent_score = sc
    for si in range(min(max_stars, len(all_starts))):
        x = all_starts[si].copy()
        try:
            for tau in temperatures:
                alpha = 1.0
                stall = 0
                for it in range(max_iters):
                    val, p = smooth_value(x, tau)
                    if not np.isfinite(val) or p is None:
                        break
                    grad = smooth_gradient(x, tau, p)
                    if not np.all(np.isfinite(grad)):
                        break
                    accepted = False
                    for bt in range(24):
                        y = project_capped_simplex(x - alpha * grad, n)
                        D = float(np.sum((y - x) ** 2))
                        if D <= 1e-18:
                            break
                        val_y, _ = smooth_value(y, tau)
                        if np.isfinite(val_y):
                            armijo = val - 1e-4 * D / alpha + 1e-14 * max(1.0, abs(val))
                            if val_y <= armijo:
                                imp = val - val_y
                                x = y
                                accepted = True
                                if imp <= 1e-12 * max(1.0, abs(val)):
                                    stall += 1
                                else:
                                    stall = 0
                                break
                        alpha *= 0.5
                    if not accepted:
                        break
                    if stall >= 5:
                        break
                    sc = exact_score(x)
                    if np.isfinite(sc) and sc < incumbent_score - 1e-12:
                        incumbent = x.copy()
                        incumbent_score = sc
                    alpha = min(2.0, 1.25 * alpha)
        except Exception:
            continue
    final = project_capped_simplex(incumbent, n)
    final_sc = exact_score(final)
    fb_sc = exact_score(fallback)
    if not np.isfinite(final_sc) or final_sc > fb_sc:
        final = fallback.copy()
        final_sc = fb_sc
    if not np.isfinite(final_sc):
        final = np.full(n, 0.5, dtype=np.float64)
        final_sc = exact_score(final)
    return final, final_sc, int(n)
