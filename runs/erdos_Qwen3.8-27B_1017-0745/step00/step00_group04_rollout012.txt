def exact_score(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def compute_lags(h, n):
    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0, :] = dx * (1.0 - 2.0 * h)
    idx = 1
    for r in range(1, n):
        nterms = n - r
        ip = h[r:]
        ip0 = h[:nterms]
        q[idx] = dx * np.sum(ip * (1.0 - ip0))
        G[idx, :nterms] += -dx * ip
        G[idx, r:r+nterms] += dx * (1.0 - ip0)
        idx += 1
        q[idx] = dx * np.sum(ip0 * (1.0 - ip))
        G[idx, :nterms] += dx * (1.0 - ip)
        G[idx, r:r+nterms] += -dx * ip0
        idx += 1
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None, None
    return q, G
