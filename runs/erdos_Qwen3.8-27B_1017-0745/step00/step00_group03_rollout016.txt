import numpy as np
from scipy.optimize import minimize

def project_box_sum(z, m, N):
    if z is None or z.shape != (N,) or not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    res = m - np.sum(h)
    if abs(res) > 1e-12:
        if res > 0:
            for i in range(N):
                cap = 1.0 - h[i]
                adj = min(res, cap)
                h[i] += adj
                res -= adj
                if abs(res) < 1e-12:
                    break
        else:
            for i in range(N):
                adj = min(-res, h[i])
                h[i] -= adj
                res += adj
                if abs(res) < 1e-12:
                    break
    if not np.all(np.isfinite(h)) or np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10) or abs(np.sum(h) - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return h

            # So G[j, d:d+L] += -a ✓
    return q, G

Let me verify: for k >= 0, f_k = sum_{n=0}^{L-1} h[n+k](1-h[n])
- df_k/dh[r] for r in [0, L): this r is the "n" index, so h[r] is in (1-h[r]), derivative = -h[r+k]. For r=0..L-1, h[r+k] = h[k:k+L] = a. So G[j, 0:L] += -a. ✓
- df_k/dh[r] for r in [k, N): this r is the "n+k" index (r = n+k, n = r-k in [0, L)), so h[r] is the "a" factor, derivative = 1-h[r-k]. For r=k..N-1, h[r-k] = h[0:N-k] = h[0:L] = b. So G[j, k:k+L] += 1-b. ✓

For k < 0 (d = -k), f_k = sum_{n=d}^{N-1} h[n-d](1-h[n]) = sum_{m=0}^{L-1} h[m](1-h[m+d])

    _, G = compute_lag_jac(h, N)

import numpy as np
from scipy.optimize import minimize

def project_box_sum(z, m, N):
    if z is None or np.asarray(z).shape != (N,) or not np.all(np.isfinite(np.asarray(z))):
        return np.full(N, 0.5)
    z = np.asarray(z, dtype=np.float64)
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    res = m - np.sum(h)
    if res > 1e-12:
        for i in range(N):
            adj = min(res, 1.0 - h[i])
            h[i] += adj
            res -= adj
            if res < 1e-12:
                break
    elif res < -1e-12:
        for i in range(N):
            adj = min(-res, h[i])
            h[i] -= adj
            res += adj
            if res > -1e-12:
                break
    if not np.all(np.isfinite(h)) or np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10) or abs(np.sum(h) - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return h

def compute_lag_jac(h, N):
    q = np.zeros(2 * N - 1)
    G = np.zeros((2 * N - 1, N))
    for j in range(2 * N - 1):
        k = j - (N - 1)
        if k >= 0:
            L = N - k
            a = h[k:k + L]
            b = h[0:L]
            q[j] = np.dot(a, 1.0 - b)
            G[j, 0:L] = -a
            G[j, k:k + L] += 1.0 - b
        else:
            d = -k
            L = N - d
            a = h[0:L]
            b = h[d:d + L]
            q[j] = np.dot(a, 1.0 - b)
            G[j, 0:L] = 1.0 - b
            G[j, d:d + L] += -a
    return q, G
