from scipy.optimize import minimize
import math

def lag_values_and_grads(h):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    q = np.zeros(m)
    G = np.zeros((m, n))
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        # forward lag d at index d
        q[d] = dx * np.sum(h[d:] * (1.0 - h[:-d]))
        G[d, d:n] += dx * (1.0 - h[:n-d])
        G[d, 0:n-d] -= dx * h[d:n]
        # backward lag d at index 2n-1-d
        q[2*n-1-d] = dx * np.sum(h[:n-d] * (1.0 - h[d:]))
        G[2*n-1-d, 0:n-d] += dx * (1.0 - h[d:n])
        G[2*n-1-d, d:n] -= dx * h[:n-d]
    return q, G

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_feasible(v, n, S):
    v = np.nan_to_num(v, nan=0.5, posinf=1e6, neginf=-1e6)
    v = np.clip(v, -1e6, 1e6)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(v - mid, 0.0, 1.0)
        s = math.fsum(h)
        if s > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    # correct residual
    res = S - math.fsum(h)
    for _ in range(10):
        if abs(res) < 1e-12:
            break
        if res > 0:
            avail = np.where(h < 1.0, 1.0 - h, 0.0)
            tot = avail.sum()
            if tot < 1e-15:
                break
            h += res * avail / tot
        else:
            avail = np.where(h > 0.0, h, 0.0)
            tot = avail.sum()
            if tot < 1e-15:
                break
            h += res * avail / tot
        h = np.clip(h, 0.0, 1.0)
        res = S - math.fsum(h)
    # final check
    h = np.clip(h, 0.0, 1.0)
    if abs(float(np.sum(h)) - S) > 1e-10:

        others = math.fsum(np.delete(h, 1))
        hv = S - others
        if 0.0 <= hv <= 1.0:

    p /= psum
    obj = qmax + tau * math.log(psum)
    grad = p @ G
    return obj, grad

def lag_vals_grads(h):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    q = np.zeros(m)
    G = np.zeros((m, n))
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        q[d] = dx * np.sum(h[d:] * (1.0 - h[:-d]))
        G[d, d:] += dx * (1.0 - h[:n-d])
        G[d, :n-d] -= dx * h[d:]
        q[2*n-1-d] = dx * np.sum(h[:n-d] * (1.0 - h[d:]))
        G[2*n-1-d, :n-d] += dx * (1.0 - h[d:])

            nv = int(kwargs['n_points'])
            if 2 <= nv <= 256:
                n = nv
        except Exception:
            pass
    S = n / 2.0
    rng = np.random.default_rng(seed)
    h_const = np.full(n, 0.5)
    best_h = h_const.copy()
    best_score = exact_score(h_const)
    if parent is not None:
        h_par = proj_feasible(parent, n, S)
        sc = exact_score(h_par)
        if sc < best_score:
            best_h = h_par.copy()
            best_score = sc
    else:
        h_par = best_h.copy()
    starts = [h_par.copy()]
    for sg in [0.10, 0.25, 0.50]:
        starts.append(proj_feasible(h_par + sg * rng.standard_normal(n), n, S))
    hb = np.zeros(n)
    hb[:n // 2] = 1.0
    if n % 2 == 1:
        hb[n // 2] = 0.5
    starts.append(proj_feasible(hb, n, S))
    ha = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    starts.append(proj_feasible(ha, n, S))
    for _ in range(3):
        starts.append(proj_feasible(rng.random(n), n, S))
    dedup = []
    for s in starts:
        dup = False
        for d in dedup:
            if np.max(np.abs(s - d)) < 1e-10:
                dup = True
                break
        if not dup:
            dedup.append(s)
    starts = dedup
    bounds = [(0.0, 1.0)] * n
    temps = [0.02, 0.005, 0.001, 0.0002]
    for s in starts:
        h_cur = s.copy()
        for tau in temps:
            try:
                res = minimize(soft_fun, h_cur, args=(tau,), jac=soft_jac,
                               method='SLSQP', bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': mass_eq, 'jac': mass_eq_jac}],
                               ftol=1e-10, maxiter=120)
                if res.x is not None and len(res.x) == n and np.all(np.isfinite(res.x)):
                    h_new = proj_feasible(res.x, n, S)
                    sc_new = exact_score(h_new)
                    if sc_new < best_score - 1e-12:
                        best_h = h_new.copy()
                        best_score = sc_new
                    obj_cur = soft_fun(h_cur, tau)
                    obj_new = soft_fun(h_new, tau)
                    if obj_new <= obj_cur + 1e-12:
                        h_cur = h_new
            except Exception:
                pass
        t0 = exact_score(h_cur) + max(1e-8, 1e-6 * abs(exact_score(h_cur)))
        x0 = np.zeros(n + 1)
        x0[:n] = h_cur
        x0[n] = t0
        xb = [(0.0, 1.0)] * n + [(0.0, None)]
        try:
            res2 = minimize(epi_obj, x0, jac=epi_obj_jac, method='SLSQP',
                            bounds=xb,
                            constraints=[{'type': 'eq', 'fun': epi_eq, 'jac': epi_eq_jac},
                                         {'type': 'ineq', 'fun': epi_ineq, 'jac': epi_ineq_jac}],
                            ftol=1e-10, maxiter=200)
            if res2.x is not None and len(res2.x) == n + 1 and np.all(np.isfinite(res2.x)):
                h_pol = proj_feasible(res2.x[:n], n, S)
                sc_pol = exact_score(h_pol)
                if sc_pol < best_score - 1e-12:
                    best_h = h_pol.copy()
                    best_score = sc_pol
        except Exception:
            pass
    t0 = best_score + max(1e-8, 1e-6 * abs(best_score))
    x0 = np.zeros(n + 1)
    x0[:n] = best_h
    x0[n] = t0
    xb = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res3 = minimize(epi_obj, x0, jac=epi_obj_jac, method='SLSQP',
                        bounds=xb,
                        constraints=[{'type': 'eq', 'fun': epi_eq, 'jac': epi_eq_jac},
                                     {'type': 'ineq', 'fun': epi_ineq, 'jac': epi_ineq_jac}],
                        ftol=1e-10, maxiter=200)
        if res3.x is not None and len(res3.x) == n + 1 and np.all(np.isfinite(res3.x)):
            h_fin = proj_feasible(res3.x[:n], n, S)
            sc_fin = exact_score(h_fin)
            if sc_fin < best_score - 1e-12:
                best_h = h_fin.copy()
                best_score = sc_fin
    except Exception:
        pass
    best_h = proj_feasible(best_h, n, S)
    best_score = exact_score(best_h)
    return (best_h, best_score, n)
