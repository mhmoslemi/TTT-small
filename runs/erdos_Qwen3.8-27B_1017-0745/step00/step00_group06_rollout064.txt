def _exact_score(h):
    h = np.asarray(h, dtype=np.float64)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / len(h)))

def _corr_and_jac(h):
    N = len(h)
    d = 2.0 / N
    R = np.correlate(h, 1.0 - h, mode='full') * d
    J = np.zeros((2 * N - 1, N))
    for i in range(2 * N - 1):
        k = i - (N - 1)
        j_lo = max(0, -k)
        j_hi = min(N, N - k)
        for j in range(j_lo, j_hi):
            J[i, j + k] += d * (1.0 - h[j])
            J[i, j] -= d * h[j + k]
    return R, J

def _project(x, N, m):
    x = np.asarray(x, dtype=np.float64)
    if x.shape != (N,) or not np.all(np.isfinite(x)) or np.max(np.abs(x)) > 1e6:
        return np.full(N, 0.5)
    lo, hi = float(np.min(x)) - 1.0, float(np.max(x))

            lo = mid
        else:
            hi = mid
    y = np.clip(x - mid, 0.0, 1.0)
    delta = m - np.sum(y)
    idx = 0
    while abs(delta) > 1e-14 and idx < N:
        if delta > 0:
            t = min(delta, 1.0 - y[idx])
        else:
            t = -min(-delta, y[idx])
        y[idx] += t
        delta -= t
        idx += 1
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)):
