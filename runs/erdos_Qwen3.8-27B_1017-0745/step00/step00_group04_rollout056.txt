        mid = 0.5 * (lo + hi)
        if math.fsum(np.clip(v - mid, 0.0, 1.0).tolist()) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    res = S - math.fsum(h.tolist())
    if abs(res) > 1e-14:
        if res > 0:
            m = h < 1.0
            cap = float(np.sum(1.0 - h[m]))
            if cap > 1e-14:
                h[m] += min(1.0, res / cap) * (1.0 - h[m])
        else:
            m = h > 0.0
            cap = float(np.sum(h[m]))
            if cap > 1e-14:
                h[m] -= min(1.0, -res / cap) * h[m]
    if abs(math.fsum(h.tolist()) - S) > 1e-10:
        j = int(np.argmax(h * (1.0 - h)))
        others = S - math.fsum([float(h[k]) for k in range(n) if k != j])
        if 0.0 <= others <= 1.0:
            h = h.copy(); h[j] = others
        else:
            h = np.full(n, 0.5)
    return np.clip(h, 0.0, 1.0)

def _compute_lags(h, dx):
    n = len(h)
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        L = n - d
        q[d] = dx * np.sum(h[d:d+L] * (1.0 - h[:L]))
        q[n + d - 1] = dx * np.sum(h[:L] * (1.0 - h[d:d+L]))
        for i in range(L):
            G[d, i + d] += dx * (1.0 - h[i])
            G[d, i] -= dx * h[i + d]
            G[n + d - 1, i] += dx * (1.0 - h[i + d])
            G[n + d - 1, i + d] -= dx * h[i]
    return q, G

def _soft_fun(x, dx, tau):
    q, G = _compute_lags(x, dx)
    qm = np.max(q)
    ex = np.exp((q - qm) / tau)
    s = np.sum(ex)
    val = qm + tau * math.log(s)
    return val, (ex / s) @ G

def _epi_fun(z, dx):
    return z[-1], np.r_[np.zeros(len(z) - 1), 1.0]

def _epi_eq(z, n):
    return np.sum(z[:-1]) - n / 2.0

def _epi_eq_jac(z, n):
    j = np.zeros(len(z))
    j[:-1] = 1.0
    return j

def _epi_ineq(z, dx):
    q, _ = _compute_lags(z[:-1], dx)
    return z[-1] - q

def _epi_ineq_jac(z, dx):
    h = z[:-1]
    n = len(h)
    _, G = _compute_lags(h, dx)
    J = np.zeros((2 * n - 1, len(z)))
    J[:, :-1] = -G
    J[:, -1] = 1.0
    return J

def _make_block(n):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    if n % 2 == 1:
        h[n // 2] = 0.5
    return h

def _make_alt(n):
    return np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n)])

def _epi_polish(h0, dx, n):
    h = _project_feasible(h0, n)
    q, _ = _compute_lags(h, dx)
    t0 = float(np.max(q)) + max(1e-8, 1e-6 * abs(float(np.max(q))))
    z0 = np.r_[h, t0]
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'eq', 'fun': _epi_eq, 'jac': _epi_eq_jac, 'args': (n,)},
        {'type': 'ineq', 'fun': _epi_ineq, 'jac': _epi_ineq_jac, 'args': (dx,)}
    ]
    try:
        res = minimize(_epi_fun, z0, args=(dx,), jac=True, method='SLSQP',
                       bounds=bounds, constraints=cons, ftol=1e-10, maxiter=200)
        if res.x is not None and np.all(np.isfinite(res.x)) and len(res.x) == n + 1:
            return _project_feasible(res.x[:-1], n)
    except Exception:
        pass
    return h

def run(seed=42, **kwargs):
    parent = globals().get('initial_h_values', None)
    n = 42
    has_parent = False
    parent_h = None
    if parent is not None:
        try:
            p = np.array(parent, dtype=np.float64).ravel()
            if len(p) >= 2:
                n = len(p)
                if np.all(np.isfinite(p)):
                    has_parent = True
                    parent_h = p
        except Exception:
            pass
    if not has_parent and 'n_points' in kwargs:
        try:
            n = int(kwargs['n_points'])
            if 2 <= n <= 256:
                pass
            else:
                n = 42
        except Exception:
            n = 42
    if n > 256:
        if has_parent:
            h = _project_feasible(parent_h, n)
            sc = _exact_score(h)
            return h, sc, n
        n = 42
    dx = 2.0 / n
    S = n / 2.0
    # Initial incumbent
    h0 = np.full(n, 0.5)
    best_h = _project_feasible(h0, n)
    best_score = _exact_score(best_h)
    if has_parent:
        ph = _project_feasible(parent_h, n)
        ps = _exact_score(ph)
        if ps < best_score:
            best_h, best_score = ph, ps
    # Build starts
    rng = np.random.default_rng(seed)
    starts = []
    if has_parent:
        starts.append(parent_h.copy())
        for sig in (0.10, 0.25, 0.50):
            starts.append(parent_h + sig * rng.standard_normal(n))
    starts.append(_make_block(n))
    starts.append(_make_alt(n))
    n_rand = 2 if has_parent else 4
    for _ in range(n_rand):
        starts.append(rng.random(n))
    if not has_parent:
        i = np.arange(n, dtype=np.float64)
        starts.append(0.5 + 0.45 * np.sin(2.0 * np.pi * i / n))
    # Deduplicate
    dedup = []
    for s in starts:
        proj = _project_feasible(s, n)
        if not any(np.max(np.abs(proj - d)) < 1e-10 for d in dedup):
            dedup.append(proj)
    starts = dedup
    # Multistart SLSQP with log-sum-exp continuation
    taus = (0.02, 0.005, 0.001, 0.0002)
    bnds = [(0.0, 1.0)] * n
    eq_c = {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (S,)}
    for start in starts:
        cur = _project_feasible(start, n)
        for tau in taus:
            try:
                res = minimize(_soft_fun, cur, args=(dx, tau), jac=True,
                               method='SLSQP', bounds=bnds, constraints=[eq_c],
                               ftol=1e-10, maxiter=120)
                if res.x is not None and np.all(np.isfinite(res.x)) and len(res.x) == n:
                    cand = _project_feasible(res.x, n)
                    sc = _exact_score(cand)
                    if sc < best_score - 1e-12:
                        best_h, best_score = cand, sc
                    q_c, _ = _compute_lags(cur, dx)
                    q_new, _ = _compute_lags(cand, dx)
                    if np.max(q_new) <= np.max(q_c) + tau * math.log(2 * n - 1) + 1e-10:
                        cur = cand
                    else:
                        cur = _project_feasible(res.x, n)
            except Exception:
                cur = _project_feasible(cur, n)
        # Epigraph polish
        polished = _epi_polish(cur, dx, n)
        sc = _exact_score(polished)
        if sc < best_score - 1e-12:
            best_h, best_score = polished, sc
    # Final polish on global best
    final = _epi_polish(best_h, dx, n)
    fs = _exact_score(final)
    if fs < best_score - 1e-12:
        best_h, best_score = final, fs
    # Final validation
    best_h = _project_feasible(best_h, n)
    if np.any(best_h < -1e-12) or np.any(best_h > 1.0 + 1e-12) or abs(math.fsum(best_h.tolist()) - S) > 1e-8:
        best_h = np.full(n, 0.5)
    best_score = _exact_score(best_h)
    return best_h, best_score, n

def _eq_fun(z, S):
    return np.sum(z) - S

def _eq_jac(z, S):
    return np.ones(len(z))
