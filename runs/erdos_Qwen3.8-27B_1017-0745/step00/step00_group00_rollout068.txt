import numpy as np
from scipy.optimize import minimize

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    lo, hi = np.min(v) - 1.0, np.max(v) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    res = S - np.sum(y)
    if res > 0:
        for i in range(n):
            d = min(1.0 - y[i], res)
            y[i] += d
            res -= d
            if res < 1e-14:
                break
    elif res < 0:
        for i in range(n):
            d = min(y[i], -res)
            y[i] -= d
            res += d
            if -res < 1e-14:
                break
    if not np.all(np.isfinite(y)):
        return None
    if np.min(y) < -1e-12 or np.max(y) > 1.0 + 1e-12:
        return None
    if abs(np.sum(y) - S) > 1e-10 * max(1, n):
        return None
    return y

def is_feasible(h, n):
    if h is None or h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.min(h) < -1e-12 or np.max(h) > 1.0 + 1e-12:
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score(h):
    n = len(h)
    if not is_feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

class Solver:
    def __init__(self, n):
        self.n = n
    def obj(self, z):
        return z[-1]
    def obj_jac(self, z):
        g = np.zeros(self.n + 1)
        g[-1] = 1.0
        return g
    def ineq(self, z):
        n = self.n
        h = z[:n]
        return z[-1] - np.correlate(h, 1.0 - h, mode='full') / n
    def ineq_jac(self, z):
        n = self.n
        h = z[:n]
        nl = 2 * n - 1
        J = np.zeros((nl, n + 1))
        for m in range(nl):
            k = m - (n - 1)
            for r in range(n):
                g = 0.0
                if 0 <= r - k < n:
                    g += 1.0 - h[r - k]
                if 0 <= r + k < n:
                    g -= h[r + k]
                J[m, r] = -g / n
            J[m, -1] = 1.0
        return J
    def eq_c(self, z):
        return np.sum(z[:self.n]) / self.n - 0.5
    def eq_j(self, z):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0 / self.n
        return g
    def solve(self, h0):
        n = self.n
        Q = np.correlate(h0, 1.0 - h0, mode='full')
        t0 = min(1.0, float(np.max(Q)) / n + 1e-4)
        z0 = np.concatenate([h0, [t0]])
        bounds = [(0.0, 1.0)] * (n + 1)
        cons = [
            {'type': 'ineq', 'fun': self.ineq, 'jac': self.ineq_jac},
            {'type': 'eq', 'fun': self.eq_c, 'jac': self.eq_j},
        ]
        try:
            res = minimize(self.obj, z0, jac=self.obj_jac, bounds=bounds,
                          constraints=cons, method='SLSQP',
                          options={'maxiter': 300, 'ftol': 1e-10, 'disp': False})
            if res.x is not None and len(res.x) == n + 1 and np.all(np.isfinite(res.x)):
                return res.x[:n].copy()
        except Exception:
            pass
        return None

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    n = 45
    parent = None
    raw = globals().get('initial_h_values', None)
    if raw is not None:
        hv = np.asarray(raw, dtype=np.float64)
        if hv.ndim == 1 and 2 <= len(hv) <= 999:
            if np.all(np.isfinite(hv)) and np.min(hv) > -0.01 and np.max(hv) < 1.01:
                if abs(np.sum(hv) - len(hv) / 2.0) < 1e-4:
                    y = project_box_sum(hv, len(hv) / 2.0)
                    if y is not None and is_feasible(y, len(hv)):
                        parent = y
                        n = len(hv)
    if parent is None:
        parent = np.full(45, 0.5, dtype=np.float64)
        n = 45
    S = n / 2.0
    best = parent.copy()
    best_score = exact_score(best)
    sv = Solver(n)

    starts = [best.copy(), best[::-1].copy()]
    for sd in [0.01, 0.03, 0.06, 0.12, 0.20, 0.05]:
        p = best + sd * rng.standard_normal(n)
        y = project_box_sum(p, S)
        if y is not None:
            starts.append(y)
    for _ in range(4):
        u = rng.uniform(0.0, 1.0, n)
        y = project_box_sum(u, S)
        if y is not None:
            starts.append(y)
    for (A, m, ph) in [(0.25, 1, 0.37), (0.35, 2, 0.74), (0.45, 3, 1.11)]:
        i = np.arange(n, dtype=np.float64)
        u = 0.5 + A * np.cos(2.0 * np.pi * m * i / n + ph)
        y = project_box_sum(u, S)
        if y is not None:
            starts.append(y)

    for h0 in starts:
        h0p = project_box_sum(h0, S)
        if h0p is None or not is_feasible(h0p, n):
            continue
        res_h = sv.solve(h0p)
        if res_h is None:
            continue
        res_h = project_box_sum(res_h, S)
        if res_h is None or not is_feasible(res_h, n):
            continue
        s = exact_score(res_h)
        if s < best_score - 1e-14:
            best = res_h.copy()
            best_score = s

    sigma = 0.05
    for _ in range(4):
        p = best + sigma * rng.standard_normal(n)
        y = project_box_sum(p, S)
        if y is None or not is_feasible(y, n):
            sigma = min(0.25, 1.5 * sigma)
            continue
        res_h = sv.solve(y)
        if res_h is None:
            sigma = min(0.25, 1.5 * sigma)
            continue
        res_h = project_box_sum(res_h, S)
        if res_h is None or not is_feasible(res_h, n):
            sigma = min(0.25, 1.5 * sigma)
            continue
        s = exact_score(res_h)
        if s < best_score - 1e-12:
            best = res_h.copy()
            best_score = s
            sigma = max(0.01, 0.7 * sigma)
        else:
            sigma = min(0.25, 1.5 * sigma)

    final = project_box_sum(best, S)
    if final is None or not is_feasible(final, n):
        final = parent.copy()
    c5 = exact_score(final)
    if c5 == float('inf'):
        final = parent.copy()
        c5 = exact_score(final)
    return (final.astype(np.float64), c5, n)
