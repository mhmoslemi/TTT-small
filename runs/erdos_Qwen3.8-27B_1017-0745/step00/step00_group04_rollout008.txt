import numpy as np
from scipy.optimize import minimize
import math

def exact_score(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / len(h)))

def capped_project(v, S):
    n = len(v)
    if not np.all(np.isfinite(v)):
        return np.full(n, 0.5)
    v = np.clip(v.astype(float), -1e6, 1e6)
    lo, hi = float(v.min()) - 1.0, float(v.max())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if math.fsum(np.clip(v - mid, 0, 1)) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    res = S - math.fsum(h)
    for j in range(n):
        if abs(res) < 1e-14:
            break
        if res > 0 and h[j] < 1.0:
            d = min(res, 1.0 - h[j]); h[j] += d; res -= d
        elif res < 0 and h[j] > 0.0:
            d = min(-res, h[j]); h[j] -= d; res += d
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12) or abs(float(np.sum(h)) - S) > 1e-10:
        return np.full(n, 0.5)
    return np.clip(h, 0.0, 1.0)

def compute_lags(h):
    n = len(h); dx = 2.0 / n; M = 2 * n - 1
    q = np.zeros(M); G = np.zeros((M, n))
    q[0] = dx * np.sum(h * (1.0 - h)); G[0, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        a = np.arange(n - d)
        q[n - 1 + d] = dx * np.sum(h[a + d] * (1.0 - h[a]))
        G[n - 1 + d, a + d] += dx * (1.0 - h[a]); G[n - 1 + d, a] += -dx * h[a + d]
        q[n - 1 - d] = dx * np.sum(h[a] * (1.0 - h[a + d]))
        G[n - 1 - d, a] += dx * (1.0 - h[a + d]); G[n - 1 - d, a + d] += -dx * h[a]
    return q, G

def soft_func(h, tau):
    q, G = compute_lags(h); qm = q.max()
    s = (q - qm) / tau; e = np.exp(s)
    p = e / (e.sum() + 1e-300)
    return qm + tau * math.log(e.sum() + 1e-300), G.T @ p

def mass_eq(x):
    return float(np.sum(x) - len(x) / 2.0)

def mass_eq_jac(x):
    return np.ones(len(x))

def epi_obj(x):
    return float(x[-1])

def epi_obj_jac(x):
    g = np.zeros(len(x)); g[-1] = 1.0; return g

def epi_eq(x):
    n = len(x) - 1; return float(np.sum(x[:n]) - n / 2.0)

def epi_eq_jac(x):
    n = len(x) - 1; J = np.zeros(n + 1); J[:n] = 1.0; return J

def epi_ineq(x):
    n = len(x) - 1; q, _ = compute_lags(x[:n]); return x[-1] - q

def epi_ineq_jac(x):
    n = len(x) - 1; M = 2 * n - 1
    _, G = compute_lags(x[:n])
    J = np.zeros((M, n + 1)); J[:, :n] = -G; J[:, -1] = 1.0; return J

def make_block(n):
    h = np.zeros(n); h[:n // 2] = 1.0
    if n % 2 == 1: h[n // 2] = 0.5
    return h

def make_alt(n):
    h = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(float)
    if n % 2 == 1: h[n // 2] = 0.5
    return h

def run(seed=42, **kwargs):
    n = 42
    parent = None
    try:

        if len(p) >= 2:
            parent = p; n = len(p)
    except:

    if parent is None and 'n_points' in kwargs:
        np_ = kwargs['n_points']
        if isinstance(np_, int) and 2 <= np_ <= 256:
            n = np_
    S = n / 2.0
    rng = np.random.default_rng(seed)

    h_const = np.full(n, 0.5)
    best_h = h_const.copy()
    best_score = exact_score(h_const)

    if parent is not None and np.all(np.isfinite(parent)):
        hp = capped_project(parent, S)
        sp = exact_score(hp)
        if sp < best_score:
            best_h, best_score = hp, sp

    starts = [best_h.copy()]
    if parent is not None and np.all(np.isfinite(parent)):
        for sigma in (0.10, 0.25, 0.50):
            starts.append(capped_project(parent + sigma * rng.standard_normal(n), S))
    starts.append(capped_project(make_block(n), S))
    starts.append(capped_project(make_alt(n), S))
    if parent is None:
        i = np.arange(n)
        starts.append(capped_project(0.5 + 0.45 * np.sin(2 * np.pi * i / n), S))
    for _ in range(2 if parent is not None else 4):
        starts.append(capped_project(rng.uniform(0, 1, n), S))

    dedup = [starts[0]]
    for s in starts[1:]:
        if min(np.max(np.abs(s - s2)) for s2 in dedup) > 1e-10:
            dedup.append(s)
    starts = dedup

    bounds = [(0.0, 1.0)] * n
    cons_s = [{'type': 'eq', 'fun': mass_eq, 'jac': mass_eq_jac}]
    tempers = (0.02, 0.005, 0.001, 0.0002)
    polished = []

    for start in starts:
        curr = start.copy()
        for tau in tempers:
            try:
                res = minimize(soft_func, curr, args=(tau,), method='SLSQP',
                               jac=True, bounds=bounds, constraints=cons_s,
                               options={'ftol': 1e-10, 'maxiter': 120})
                if res.x is not None and res.x.shape == (n,) and np.all(np.isfinite(res.x)):
                    hp = capped_project(res.x, S)
                    s_val = exact_score(hp)
                    if s_val < best_score - 1e-12:
                        best_h, best_score = hp, s_val
                    curr = hp
            except:
                pass
        polished.append(curr)

    epi_cons = [
        {'type': 'eq', 'fun': epi_eq, 'jac': epi_eq_jac},
        {'type': 'ineq', 'fun': epi_ineq, 'jac': epi_ineq_jac}
    ]
    bounds_epi = [(0.0, 1.0)] * n + [(0.0, None)]

    def do_epi_polish(h0):
        global best_h, best_score
        q0, _ = compute_lags(h0)
        t0 = float(q0.max()) + max(1e-8, 1e-6 * abs(float(q0.max())))
        x0 = np.concatenate([h0, [t0]])
        try:
            res = minimize(epi_obj, x0, method='SLSQP', jac=epi_obj_jac,
                           bounds=bounds_epi, constraints=epi_cons,
                           options={'ftol': 1e-10, 'maxiter': 200})
            if res.x is not None and res.x.shape == (n + 1,) and np.all(np.isfinite(res.x)):
                hp = capped_project(res.x[:n], S)
                s_val = exact_score(hp)
                if s_val < best_score - 1e-12:
                    best_h, best_score = hp, s_val
        except:
            pass

    for pt in polished:
        do_epi_polish(pt.copy())
    do_epi_polish(best_h.copy())

    best_h = capped_project(best_h, S)
    best_score = exact_score(best_h)
    if best_score < 0 or not np.isfinite(best_score):
        best_h = h_const; best_score = exact_score(h_const)
    return best_h, best_score, n
