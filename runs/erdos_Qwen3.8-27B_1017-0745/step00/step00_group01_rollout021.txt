def project_capped_simplex(z, n):
    """Project z onto {h: 0<=h_i<=1, sum(h)=n/2}."""
    target = n / 2.0
    if not np.all(np.isfinite(z)) or len(z) != n:
        return np.full(n, 0.5)
    z = np.asarray(z, dtype=np.float64)
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    # Correct residual
    for _ in range(n + 2):
        res = target - h.sum()
        if abs(res) < 1e-14:
            break
        if res > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        headroom = (1.0 - h) if res > 0 else h
        j = idx[np.argmax(headroom[idx])]
        amt = min(abs(res), headroom[j])
        h[j] += np.sign(res) * amt

    if not np.all(np.isfinite(h)) or h.sum() - target > 1e-9 * max(1, n):

import numpy as np

def project_capped_simplex(z, n):
    target = n / 2.0
    if not np.all(np.isfinite(z)) or len(z) != n:
        return np.full(n, 0.5)
    z = np.asarray(z, dtype=np.float64)
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(n + 2):
        res = target - h.sum()
        if abs(res) < 1e-14:
            break
        if res > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        headroom = (1.0 - h) if res > 0 else h
        j = idx[np.argmax(headroom[idx])]
        amt = min(abs(res), headroom[j])
        h[j] += np.sign(res) * amt
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(h.sum() - target) > 1e-9 * max(1, n):
        return np.full(n, 0.5)
    return h

def exact_score(h):
    n = len(h)

        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def corr_and_jac(h):
    n = len(h)
    q = np.correlate(h, 1.0 - h, mode='full')
    J = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        d = k - (n - 1)
        i0 = max(d, 0)
        j0 = max(-d, 0)
        m = n - abs(d)
        for r in range(m):
            J[k, i0 + r] += (1.0 - h[j0 + r])
            J[k, j0 + r] -= h[i0 + r]
    return q, J

    return z[-1]

    n = args[0]

    return g

    n, dx = args
    h = z[:n]
    return z[-1] - dx * np.correlate(h, 1.0 - h, mode='full')

    n, dx = args
    h = z[:n]
    _, J = corr_and_jac(h)
    row = np.zeros((2 * n - 1, n + 1))
    row[:, :n] = -dx * J
    row[:, n] = 1.0
