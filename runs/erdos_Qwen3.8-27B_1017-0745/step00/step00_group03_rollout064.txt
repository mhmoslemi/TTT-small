import numpy as np

def project_box_sum(z, N):
    if z is None:
        return np.full(N, 0.5)
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or z.shape[0] != N or not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    if np.max(np.abs(z)) > 1e6:
        return np.full(N, 0.5)
    m = N / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)
    for _ in range(5):
        e = m - np.sum(h)
        if abs(e) < 1e-14:
            break
        if e > 0:
            for i in range(N):
                adj = min(e, 1.0 - h[i])
                if adj > 0:
                    h[i] += adj
                    e -= adj
                    if e < 1e-14:
                        break
        else:
            for i in range(N):
                adj = min(-e, h[i])
                if adj > 0:
                    h[i] -= adj
                    e += adj
                    if e > -1e-14:
                        break
    if (not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1 + 1e-12)
            or abs(np.sum(h) - m) > 1e-10 * max(1, N)):
        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)

def lag_values(h):
    N = len(h)
    M = 2 * N - 1
    q = np.zeros(M)
    one_minus_h = 1.0 - h
    for k in range(-(N - 1), N):
        idx = k + N - 1
        if k >= 0:
            L = N - k
            a = h[k:k + L]
            b = h[0:L]
            q[idx] = np.dot(a, one_minus_h[0:L])
        else:
            d = -k
            L = N - d
            a = h[0:L]
            b = h[d:d + L]
            q[idx] = np.dot(a, one_minus_h[d:d + L])
    return q

def smooth_value_grad(h, tau, N):
    q = lag_values(h)
    r = (2.0 / N) * q
    rmax = np.max(r)
    ex = np.exp((r - rmax) / tau)
    s = np.sum(ex)
    p = ex / s
    phi = rmax + tau * np.log(s)
    grad = np.zeros(N)
    for k in range(-(N - 1), N):
        idx = k + N - 1
        w = (2.0 / N) * p[idx]
        if w < 1e-15:
            continue
        if k >= 0:
            L = N - k
            a = h[k:k + L]
            b = h[0:L]
            grad[k:k + L] += w * (1.0 - b)
            grad[0:L] -= w * a
        else:
            d = -k
            L = N - d
            a = h[0:L]
            b = h[d:d + L]
            grad[0:L] += w * (1.0 - b)
            grad[d:d + L] -= w * a
    return phi, grad

def smooth_value(h, tau, N):
    q = lag_values(h)
    r = (2.0 / N) * q
    rmax = np.max(r)
    ex = np.exp((r - rmax) / tau)
    s = np.sum(ex)
    return rmax + tau * np.log(s)

def score_exact(h, N):
    if (h is None or len(h) != N or h.ndim != 1
            or not np.all(np.isfinite(h))):
        return np.inf
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / N))
