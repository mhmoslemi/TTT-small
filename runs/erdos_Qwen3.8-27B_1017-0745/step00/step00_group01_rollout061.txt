def is_feasible(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1 + 1e-12):
        return False
    mass = float(np.sum(h, dtype=np.float64))
    if abs(mass - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def project_capped_simplex(z, n):

        return np.full(n, 0.5, dtype=np.float64)

        if s_val > s:

        else:

import numpy as np


def is_feasible(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    mass = float(np.sum(h, dtype=np.float64))
    if abs(mass - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True


def project_capped_simplex(z, n):
    s = n / 2.0
    z = np.array(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return np.full(n, 0.5, dtype=np.float64)
    z = np.clip(z, -1e12, 1e12)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        hc = np.clip(z - mid, 0.0, 1.0)
        if float(np.sum(hc, dtype=np.float64)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h, dtype=np.float64))
    for _ in range(n + 2):
        if abs(r) <= 1e-14:
            break
        if r > 0:
            slack = 1.0 - h
            idx = int(np.argmax(slack))
            move = min(r, float(slack[idx]))
            if move <= 0:
                break
            h[idx] += move
            r -= move
        else:
            avail = h.copy()
            idx = int(np.argmax(avail))
            move = min(-r, float(avail[idx]))
            if move <= 0:
                break
            h[idx] -= move
            r += move
    h = np.clip(h, 0.0, 1.0)
    if not is_feasible(h, n):
        return np.full(n, 0.5, dtype=np.float64)
    return h


def exact_score(h):
    n = len(h)
    if not is_feasible(h, n):
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(dx * np.max(corr))


def smooth_value_weights(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = dx * corr
    c_max = float(np.max(c))
    a = np.exp((c - c_max) / tau)
    s_a = float(np.sum(a, dtype=np.float64))
    if s_a <= 0:
        return None, None
    F = c_max + tau * np.log(s_a)
    p = a / s_a
    if not np.isfinite(F) or not np.all(np.isfinite(p)):
        return None, None
    return float(F), p


def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    _, p = smooth_value_weights(h, tau)
    if p is None:
        return None
    P = p
    U = np.convolve(P, 1.0 - h, mode="full")[(n - 1):(2 * n - 1)]
    V = np.convolve(P[::-1], h, mode="full")[(n - 1):(2 * n - 1)]
    grad = dx * (U - V)
    if not np.all(np.isfinite(grad)):
        return None
    return grad


def build_starts(n, rng, parent):
    if parent is not None:
        base = parent.astype(np.float64).copy()
    else:
        base = np.full(n, 0.5, dtype=np.float64)
    cands = []
    cands.append(base)
    cands.append(base[::-1].copy())
    cands.append(1.0 - base[::-1].copy())
    cands.append(np.full(n, 0.5, dtype=np.float64))
    block = np.zeros(n, dtype=np.float64)
    block[:n // 2] = 1.0
    if n % 2 == 1:
        block[n // 2] = 0.5
    cands.append(block)
    alt = np.zeros(n, dtype=np.float64)
    alt[::2] = 1.0
    cands.append(alt)
    for sd in [0.05, 0.15, 0.35]:
        cands.append(base + rng.normal(0, sd, n))
    cands.append(rng.uniform(0.0, 1.0, n))
    seen = []
    for c in cands:
        proj = project_capped_simplex(c, n)
        dup = False
        for s in seen:
            if np.max(np.abs(proj - s)) < 1e-10:
                dup = True
                break
        if not dup:
            seen.append(proj)
    return seen


def pgd_stage(x, tau, max_iter, n):
    F_x, _ = smooth_value_weights(x, tau)
    if F_x is None:
        return x
    grad = smooth_grad(x, tau)
    if grad is None:
        return x
    alpha = 1.0
    stagnation = 0
    for _ in range(max_iter):
        y = project_capped_simplex(x - alpha * grad, n)
        D = float(np.sum((y - x) ** 2))
        if D <= 0.0 or np.max(np.abs(y - x)) <= 1e-9:
            break
        F_y, _ = smooth_value_weights(y, tau)
        if F_y is None:
            break
        tol = 1e-14 * max(1.0, abs(F_x))
        if F_y <= F_x - 1e-4 * D / alpha + tol:
            improvement = F_x - F_y
            x = y
            F_x = F_y
            grad = smooth_grad(x, tau)
            if grad is None:
                break
            alpha = min(2.0, 1.25 * alpha)
            if improvement <= 1e-12 * max(1.0, abs(F_x)):
                stagnation += 1
            else:
                stagnation = 0
            if stagnation >= 5:
                break
        else:
            alpha *= 0.5
            if alpha < 1e-12:
                break
    return x
