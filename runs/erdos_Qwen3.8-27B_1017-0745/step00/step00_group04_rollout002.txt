def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(corr) * dx)
    if not math.isfinite(val):
        return float('inf')
    return val

def feasible(h, n, tol=1e-10):

    return True

def restore_simplex(v, n):

                if np.all(pv >= -1e-12) and np.all(pv <= 1.0 + 1e-12) and abs(np.sum(pv) - S) <= 1e-10:
                    parent = np.clip(pv, 0.0, 1.0)
                    if abs(np.sum(parent) - S) > 1e-12:
                        parent = restore_simplex(pv, n)
                    if parent is not None:
                        parent_sc = exact_score(parent)
    except:
        parent = None
    
    # Incumbent
    inc_h = const_h.copy()
    inc_sc = const_sc
    if parent is not None and parent_sc < inc_sc - 1e-12:
        inc_h = parent.copy()
        inc_sc = parent_sc
    
    if n > 128:
        if parent is not None:
            sc = exact_score(parent)
            if sc < inc_sc - 1e-12:
                inc_h = parent.copy()
                inc_sc = sc
        return (inc_h, float(inc_sc), n)
    
    # Build starts
    starts = []
    
    def add_start(h):
        if h is None or not feasible(h, n):
            return
        for s in starts:
            if np.max(np.abs(h - s)) < 1e-10:
                return
        starts.append(h.copy())
    
    if parent is not None:
        add_start(parent)
    add_start(const_h)
    
    # Block vector
    blk = np.zeros(n)
    half = n // 2
    blk[:half] = 1.0
    if n % 2 == 1:
        blk[half] = 0.5
    add_start(blk)
    
    # Alternating
    alt = np.array([1.0 if k % 2 == 0 else 0.0 for k in range(n)])
    alt = project(alt, n)
    add_start(alt)
    
    # Sinusoid
    idx = np.arange(n)
    sinv = 0.5 + 0.45 * np.sin(2.0 * math.pi * idx / n)
    sinv = project(sinv, n)
    add_start(sinv)
