        t0 = float(np.max(q_cur)) + max(1e-8, 1e-6 * abs(float(np.max(q_cur))))
        z0 = np.concatenate([cur, [t0]])
        z_bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        eeq = {'type': 'eq', 'fun': _epi_eq_fun, 'jac': _epi_eq_jac}
        ine = {'type': 'ineq', 'fun': _epi_ineq, 'jac': _epi_ineq_jac}
        try:
            res2 = minimize(_epi_obj, z0, method='SLSQP',
                            jac=_epi_obj, bounds=z_bounds,
                            constraints=[eeq, ine],
                            options={'ftol': 1e-10, 'maxiter': 200})
            if res2.x is not None and len(res2.x) == n + 1 and np.all(np.isfinite(res2.x)):
                hc = project_feasible(res2.x[:n], n)
                sc = exact_score(hc)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_h = hc.copy()
        except Exception:
            pass
    # Final epigraph polish from global best
    q_b, _ = compute_lags_and_grads(best_h)
    t_b = float(np.max(q_b)) + max(1e-8, 1e-6 * abs(float(np.max(q_b))))
    z_b = np.concatenate([best_h, [t_b]])
    z_bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    eeq = {'type': 'eq', 'fun': _epi_eq_fun, 'jac': _epi_eq_jac}
    ine = {'type': 'ineq', 'fun': _epi_ineq, 'jac': _epi_ineq_jac}
    try:
        res3 = minimize(_epi_obj, z_b, method='SLSQP',
                        jac=_epi_obj, bounds=z_bounds,
                        constraints=[eeq, ine],
                        options={'ftol': 1e-10, 'maxiter': 200})
        if res3.x is not None and len(res3.x) == n + 1 and np.all(np.isfinite(res3.x)):
            hc = project_feasible(res3.x[:n], n)
            sc = exact_score(hc)
            if sc < best_score - 1e-12:
                best_score = sc
                best_h = hc.copy()
    except Exception:
        pass
    # Final validation
    final = project_feasible(best_h, n)
    if not (np.all(final >= 0) and np.all(final <= 1)):
        final = np.full(n, 0.5)
    final_score = exact_score(final)
    return (final, final_score, n)
