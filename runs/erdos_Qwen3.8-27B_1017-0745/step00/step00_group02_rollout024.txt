import numpy as np
from scipy.optimize import minimize

def _feasible(h, n):
    if h.ndim != 1 or len(h) != n or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-8

def _project(y, n):
    S = n / 2.0
    y = np.asarray(y, dtype=np.float64)
    lo, hi = float(np.min(y)) - 1.0, float(np.max(y))
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0, 1)) > S:
            lo = mid
        else:
            hi = mid
    v = np.clip(y - 0.5 * (lo + hi), 0, 1)
    diff = S - np.sum(v)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.where(v < 1.0 - 1e-12)[0]
        else:
            idx = np.where(v > 1e-12)[0]
        if len(idx):
            v[idx] += diff / len(idx)
    v = np.clip(v, 0, 1)
    if abs(np.sum(v) - S) > 1e-9:
        v = np.full(n, 0.5)
    return v

def _score(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _lags(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _sval(h, n, tau):
    s = _lags(h, n)
    m = float(np.max(s))
    w = np.exp((s - m) / tau)
    return m + tau * np.log(np.sum(w))

def _sgrad(h, n, tau):
    s = _lags(h, n)
    m = float(np.max(s))
    w = np.exp((s - m) / tau)
    p = w / np.sum(w)
    dx = 2.0 / n
    L = 2 * n - 1
    g = np.zeros(n)
    for k in range(L):
        lag = k - (n - 1)
        c = p[k] * dx
        if lag >= 0:
            g[lag:] += c * (1.0 - h[:n - lag])
            g[:n - lag] += c * (-h[lag:])
        else:
            m2 = -lag
            g[:n - m2] += c * (1.0 - h[m2:])
            g[m2:] += c * (-h[:n - m2])
    return g

def _pg(h, n, taus, nit=80, nls=25):
    for tau in taus:
        stuck = 0
        for _ in range(nit):
            f = _sval(h, n, tau)
            g = _sgrad(h, n, tau)
            if not np.all(np.isfinite(g)):
                break
            alpha = 1.0
            ok = False
            for _ in range(nls):
                y = _project(h - alpha * g, n)
                d = y - h
                if np.max(np.abs(d)) < 1e-10:
                    ok = True
                    break
                gd = float(np.dot(g, d))
                if gd >= -1e-14:
                    ok = True
                    break
                fy = _sval(y, n, tau)
                if fy <= f + 1e-4 * gd + 1e-14:
                    dec = f - fy
                    h = y
                    ok = True
                    stuck = stuck + 1 if dec < 1e-12 else 0
                    break
                alpha *= 0.5
            if stuck >= 10 or not ok:
                break
    return h

class _SLSQ:
    def __init__(self, n):
        self.n = n
        self.dx = 2.0 / n
        self.S = n / 2.0
    def f(self, z):
        return z[self.n]
    def fj(self, z):
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        return g
    def ineq(self, z):
        return z[self.n] - _lags(z[:self.n], self.n)
    def ineq_j(self, z):
        h = z[:self.n]
        n, dx = self.n, self.dx
        L = 2 * n - 1
        J = np.zeros((L, n + 1))
        J[:, n] = 1.0
        for k in range(L):
            lag = k - (n - 1)
            if lag >= 0:
                J[k, lag:n] -= dx * (1.0 - h[:n - lag])
                J[k, :n - lag] += dx * h[lag:]
            else:
                m2 = -lag
                J[k, :n - m2] -= dx * (1.0 - h[m2:])
                J[k, m2:] += dx * h[:n - m2]
        return J
    def eq(self, z):
        return np.sum(z[:self.n]) - self.S
    def eq_j(self, z):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g

def _slsqp(h0, n):
    pr = _SLSQ(n)
    z0 = np.concatenate([h0, [min(1.0, _score(h0, n) + 1e-8)]])
    bounds = [(0, 1)] * n + [(0, 1)]
    cons = [
        {'type': 'ineq', 'fun': pr.ineq, 'jac': pr.ineq_j},
        {'type': 'eq', 'fun': pr.eq, 'jac': pr.eq_j},
    ]
    try:
        res = minimize(pr.f, z0, method='SLSQP', jac=pr.fj,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 250, 'ftol': 1e-10, 'disp': False})
    except Exception:
        return None
    h = res.x[:n]
    if not np.all(np.isfinite(h)):
        return None
    h = _project(h, n)
    return h if _feasible(h, n) else None

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = globals().get('initial_h_values', None)
    has_parent = False
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.ndim == 1 and 2 <= len(p) <= 999 and np.all(np.isfinite(p)) and np.max(np.abs(p)) < 1e6:
            n = len(p)
            fallback = _project(p, n)
            has_parent = True
    if not has_parent:
        n = int(kwargs.get('n_points', 45))
        if n < 2 or n > 999:
            n = 45
        fallback = np.full(n, 0.5)

    best_h = fallback.copy()
    best_s = _score(best_h, n)
    local_best = [best_h.copy()]

    starts = [best_h.copy()]
    if has_parent:
        starts.append(_project(1.0 - best_h, n))
        starts.append(_project(best_h[::-1].copy(), n))
        starts.append(_project(1.0 - best_h[::-1].copy(), n))
        starts.append(_project(best_h + 0.08 * rng.standard_normal(n), n))
        starts.append(_project(best_h + 0.25 * rng.standard_normal(n), n))
        starts.append(_project(rng.uniform(0, 1, n), n))
        starts.append(_project(rng.beta(0.5, 0.5, n), n))
    else:
        for i in range(3):
            starts.append(_project(rng.uniform(0, 1, n), n))
        for i in range(3):
            starts.append(_project(rng.beta(0.5, 0.5, n), n))
        starts.append(_project(np.full(n, 0.5) + 0.35 * rng.standard_normal(n), n))

    sm = n <= 100
    taus = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00001] if sm else [0.02, 0.005, 0.001, 0.0001, 0.00001]
    nit = 80 if sm else 20

    def _try_candidate(hc):
        nonlocal best_h, best_s
        hc = _project(hc, n)
        if _feasible(hc, n):
            sc = _score(hc, n)
            if sc < best_s - 1e-12:
                best_h = hc.copy()
                best_s = sc
            local_best.append(hc.copy())

    for h0 in starts:
        hs = _pg(_project(h0, n).copy(), n, taus, nit)
        _try_candidate(hs)

    n_pert = 4 if sm else 1
    perts = [0.04, 0.12, 0.30, 0.50][:n_pert]
    for sig in perts:
        h0 = _project(best_h + sig * rng.standard_normal(n), n)
        hs = _pg(h0.copy(), n, taus, nit)
        _try_candidate(hs)

    if sm:
        local_best.sort(key=lambda x: _score(x, n))
        sels = []
        for hb in local_best:
            if all(np.max(np.abs(hb - s)) > 1e-5 for s in sels) and len(sels) < 3:
                sels.append(hb)
        if fallback not in sels and len(sels) < 3:
            sels.append(fallback.copy())
        for hs in sels:
            hr = _slsqp(hs, n)
            if hr is not None:
                _try_candidate(hr)

    best_h = _project(best_h, n)
    fb = _project(fallback, n)
    if not _feasible(best_h, n) or _score(best_h, n) > _score(fb, n):
        best_h = fb.copy()
    if not _feasible(best_h, n):
        best_h = np.full(n, 0.5)
    best_s = _score(best_h, n)
    return best_h, best_s, n
