    v = np.asarray(v, dtype=np.float64).copy()

        deduped = []
        for s in starts:
            dup = False
            for d in deduped:
                if np.max(np.abs(s - d)) < 1e-10:
                    dup = True
                    break
            if not dup:
                deduped.append(s)
        starts = deduped
        bounds_h = [(0.0, 1.0)] * n
        eq_con = {'type': 'eq', 'fun': _eq_soft, 'jac': _eq_soft_jac}
        temps = [0.02, 0.005, 0.001, 0.0002]
        bounds_z = bounds_h + [(0.0, 1e10)]
        ineq_con = {'type': 'ineq', 'fun': _epi_ineq, 'jac': _epi_ineq_jac}
        eq_con_e = {'type': 'eq', 'fun': _epi_eq, 'jac': _epi_eq_jac}
        for start in starts:
            x_cur = start.copy()
            soft_val = _soft_fun(x_cur)[0]
            for tau in temps:
                _soft_tau = tau
                try:
                    res = minimize(_soft_fun, x_cur, jac=True, method='SLSQP',
                                   bounds=bounds_h, constraints=[eq_con],
                                   options={'ftol': 1e-10, 'maxiter': 120})
                    if res.x is not None and res.x.shape == (n,):
                        if np.all(np.isfinite(res.x)):
                            x_new = _project(res.x, n)
                            if _validate(x_new, n):
                                nv = _soft_fun(x_new)[0]
                                sc = _exact_score(x_new)
                                if sc < best_score - 1e-12:
                                    best_score = sc
                                    best_h = x_new.copy()
                                if nv <= soft_val + 1e-12:
                                    x_cur = x_new
                                    soft_val = nv
                except Exception:
                    pass
            q_c, _ = _lags_and_grad(x_cur)
            t0 = float(np.max(q_c)) + max(1e-8, 1e-6 * abs(float(np.max(q_c))))
            z0 = np.concatenate([x_cur, [t0]])
            try:
                res = minimize(_epi_obj, z0, jac=_epi_obj_jac, method='SLSQP',
                               bounds=bounds_z, constraints=[eq_con_e, ineq_con],
                               options={'ftol': 1e-10, 'maxiter': 200})
                if res.x is not None and res.x.shape == (n + 1,):
                    if np.all(np.isfinite(res.x)):
                        h_c = _project(res.x[:n], n)
                        if _validate(h_c, n):
                            sc = _exact_score(h_c)
                            if sc < best_score - 1e-12:
                                best_score = sc
                                best_h = h_c.copy()
            except Exception:
                pass
        q_c, _ = _lags_and_grad(best_h)
        t0 = float(np.max(q_c)) + max(1e-8, 1e-6 * abs(float(np.max(q_c))))
        z0 = np.concatenate([best_h, [t0]])
        try:
            res = minimize(_epi_obj, z0, jac=_epi_obj_jac, method='SLSQP',
                           bounds=bounds_z, constraints=[eq_con_e, ineq_con],
                           options={'ftol': 1e-10, 'maxiter': 200})
            if res.x is not None and res.x.shape == (n + 1,):
                if np.all(np.isfinite(res.x)):
                    h_c = _project(res.x[:n], n)
                    if _validate(h_c, n):
                        sc = _exact_score(h_c)
                        if sc < best_score - 1e-12:
                            best_score = sc
                            best_h = h_c.copy()
        except Exception:
            pass
    best_h = _project(best_h, n)
    if not _validate(best_h, n):
        best_h = h_const.copy()
        best_score = _exact_score(h_const)
    best_score = _exact_score(best_h)
    return (best_h, float(best_score), n)
