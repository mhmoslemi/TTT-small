            h[i] += min(r, 1.0 - h[i])
        else:
            i = int(np.argmax(h))
            if h[i] <= 1e-15:
                break
            h[i] -= min(-r, h[i])
    h = np.clip(h, 0.0, 1.0)
    return h if check_feasible(h, n) else None


def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full')
    if q.size != 2 * n - 1 or not np.all(np.isfinite(q)):
        return float('inf')
    return float(np.max(q) * 2.0 / n)


def build_haar(n):
    if n <= 1:
        return np.empty((0, n))
    dirs, stack = [], [(0, n)]
    while stack:
        a, b = stack.pop()
        m = b - a
        if m < 2:
            continue
        mid = a + m // 2
        mL, mR = mid - a, b - mid
        v = np.zeros(n)
        v[a:mid] = sqrt(mR / (m * mL))
        v[mid:b] = -sqrt(mL / (m * mR))
        dirs.append(v)
        stack.append((a, mid))
        stack.append((mid, b))
    dirs.reverse()
    return np.array(dirs)


def line_search(x, v, n):
    dx = 2.0 / n
    lo, hi = -1e15, 1e15
    for i in range(n):
        if abs(v[i]) < 1e-15:
            continue
        r0, r1 = -x[i] / v[i], (1.0 - x[i]) / v[i]
        lo = max(lo, min(r0, r1))
        hi = min(hi, max(r0, r1))
    if lo > hi + 1e-12 or not (np.isfinite(lo) and np.isfinite(hi)):
        return x, float('inf')
    lo, hi = min(lo, 0.0), max(hi, 0.0)
    if hi - lo < 1e-14:
        return x, float('inf')
    ones = 1.0 - x
    c = dx * np.correlate(x, ones, mode='full')
    b = dx * (np.correlate(v, ones, mode='full') - np.correlate(x, v, mode='full'))
    a = -dx * np.correlate(v, v, mode='full')
    nl = 2 * n - 1
    if a.shape != (nl,) or not np.all(np.isfinite(a)) or not np.all(np.isfinite(b)) or not np.all(np.isfinite(c)):
        return x, float('inf')
    tol_t = 1e-12 * max(1.0, abs(lo), abs(hi), abs(hi - lo))
    cands = [lo, hi, 0.0]
    for i in range(nl):
        for j in range(i + 1, nl):
            ad, bd, cd = a[i] - a[j], b[i] - b[j], c[i] - c[j]
            sc = max(1.0, abs(bd), abs(4.0 * ad * cd))
            if abs(ad) > 1e-14 * sc:
                disc = bd * bd - 4.0 * ad * cd
                if disc < -1e-14 * max(1.0, bd * bd):
                    continue
                disc = max(0.0, disc)
                sq = sqrt(disc)
                q = -0.5 * (bd + copysign(sq, bd))
                if abs(q) > 1e-30:
                    roots = [cd / q, q / ad]
                else:
                    roots = [-bd / (2.0 * ad)]
            elif abs(bd) > 1e-14:
                roots = [-cd / bd]
            else:
                continue
            for t in roots:
                if np.isfinite(t) and lo - tol_t <= t <= hi + tol_t:
                    cands.append(max(lo, min(hi, t)))
    for d in range(nl):
        if abs(a[d]) > 1e-15:
            t = -b[d] / (2.0 * a[d])
            if np.isfinite(t) and lo - tol_t <= t <= hi + tol_t:
                cands.append(max(lo, min(hi, t)))
    cands.sort()
    merged = [cands[0]]
    for t in cands[1:]:
        if t - merged[-1] > tol_t:
            merged.append(t)
    t_arr = np.array(merged)
    vals = c[np.newaxis, :] + b[np.newaxis, :] * t_arr[:, np.newaxis] + a[np.newaxis, :] * (t_arr[:, np.newaxis] ** 2)
    model = np.max(vals, axis=1)
    order = np.argsort(model)
    best_exact = exact_score(x, n)
    for idx in order[:12]:
        t = t_arr[idx]
        if abs(t) < 1e-15:
            continue
        y = x + t * v
        if not check_feasible(y, n):
            y = project_capped(y, n)
            if y is None:
                continue
        sc = exact_score(y, n)
        if np.isfinite(sc) and sc < best_exact - 1e-12:
            best_exact = sc
            x = y
    return x, best_exact


def add_to_pool(starts, h, n):
    if h is None or h.shape != (n,):
        return
    if not np.all(np.isfinite(h)):
        return
    proj = project_capped(h.copy(), n)
    if proj is None:
        return
    for s in starts:
        if np.max(np.abs(proj - s)) < 1e-10:
            return
    starts.append(proj)


def run(seed=42, **kwargs):
    n, base = 47, None
    try:
        iv = np.array(initial_h_values, dtype=np.float64)
        if iv.ndim == 1 and iv.size > 0:
            n = iv.size
            if np.all(np.isfinite(iv)) and np.all(iv >= -1e-8) and np.all(iv <= 1.0 + 1e-8):
                base = np.clip(iv, 0.0, 1.0)
    except (NameError, Exception):
        pass
    if base is None:
        npk = kwargs.get('n_points', None)
        if isinstance(npk, (int, np.integer)) and not isinstance(npk, bool) and 1 <= npk <= 999:
            n = int(npk)
    n = max(1, min(n, 999))
    rng = np.random.default_rng(seed if isinstance(seed, (int, np.integer)) and not isinstance(seed, bool) else 42)
    fallback = np.full(n, 0.5)
    if base is not None:
        fb = project_capped(base.copy(), n)
        if fb is not None:
            fallback = fb
    starts = []
    if base is not None:
        add_to_pool(starts, base, n)
        add_to_pool(starts, base[::-1].copy(), n)
        add_to_pool(starts, 1.0 - base.copy(), n)
        add_to_pool(starts, base + 0.12 * rng.standard_normal(n), n)
    add_to_pool(starts, np.full(n, 0.5), n)
    bv = np.zeros(n)
    bv[:n // 2] = 1.0
    if n % 2 == 1:
        bv[n // 2] = 0.5
    add_to_pool(starts, bv, n)
    av = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    if n % 2 == 1:
        av[n - 1] = 0.5
    add_to_pool(starts, av, n)
    if base is not None:
        add_to_pool(starts, base + 0.35 * rng.standard_normal(n), n)
    add_to_pool(starts, rng.uniform(0.0, 1.0, n), n)
    if base is not None:
        add_to_pool(starts, base + 0.45 * rng.standard_normal(n), n)
    if not starts:
        starts.append(fallback.copy())
    if n <= 48:
        n_starts, n_sweeps = 5, 2
    elif n <= 64:
        n_starts, n_sweeps = 2, 1
    else:
        n_starts, n_sweeps = 3, 0
    cs = kwargs.get('starts', None)
    if isinstance(cs, (int, np.integer)) and not isinstance(cs, bool) and cs > 0:
        n_starts = min(cs, 6 if n <= 48 else 3)
    cw = kwargs.get('sweeps', None)
    if isinstance(cw, (int, np.integer)) and not isinstance(cw, bool) and cw > 0:
        n_sweeps = min(cw, 3 if n <= 48 else 2)
    n_starts = min(len(starts), n_starts) if n_starts > 0 else 1
    if n_starts < 1:
        n_starts = 1
    incumbent = fallback.copy()
    incumbent_score = exact_score(incumbent, n)
    for s in starts:
        sc = exact_score(s, n)
        if np.isfinite(sc) and sc < incumbent_score - 1e-12:
            incumbent = s.copy()
            incumbent_score = sc
    if n <= 64 and n > 1 and n_sweeps > 0:
        basis = build_haar(n)
        nd = basis.shape[0]
        if nd > 0:
            for si in range(min(n_starts, len(starts))):
                x, xsc = starts[si].copy(), exact_score(starts[si], n)
                for sw in range(n_sweeps):
                    perm = rng.permutation(nd)
                    improved = False
                    for d in perm:
                        try:
                            x_new, sc_new = line_search(x, basis[d], n)
                            if np.isfinite(sc_new) and sc_new < xsc - 1e-12:
                                x, xsc = x_new, sc_new
                                improved = True
                                if xsc < incumbent_score - 1e-12:
                                    incumbent = x.copy()
                                    incumbent_score = xsc
                        except Exception:
                            pass
                    if not improved:
                        break
    final = project_capped(incumbent.copy(), n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs >= exact_score(fallback, n) + 1e-12:
        final = fallback.copy()
        fs = exact_score(final, n)
    if not np.isfinite(fs):
        final = np.full(n, 0.5)
        fs = exact_score(final, n)
    return final, float(fs), int(n)
