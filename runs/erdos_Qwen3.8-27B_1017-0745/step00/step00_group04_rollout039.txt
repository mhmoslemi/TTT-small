    S = n / 2.0
    v = np.clip(v, -1e6, 1e6)
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    for _ in range(3):
        res = S - math.fsum(h.tolist())
        if abs(res) < 1e-14:

        if res > 0:
            for i in range(n):
                if res <= 0: break

                h[i] -= sub

def lp_step(h, q, G, delta):
    n = len(h)
    M = 2 * n - 1
    dx = 2.0 / n
    e = dx * delta * delta
    # Variables: p[0:n], u[0:n], t (index 2n)
    nv = 2 * n + 1
    c = np.zeros(nv)
    c[2*n] = 1.0  # minimize t
    
    # Inequality rows
    # M lag rows: G[l]·p - t <= -q[l] - e
    A_ub = np.zeros((M + 4*n + 1, nv))
    b_ub = np.zeros(M + 4*n + 1)
    for l in range(M):
        A_ub[l, :n] = G[l]
        A_ub[l, 2*n] = -1.0
        b_ub[l] = -q[l] - e
    row = M
    for i in range(n):
        # p[i] - u[i] <= 0
        A_ub[row, i] = 1.0
        A_ub[row, n+i] = -1.0
        b_ub[row] = 0.0
        row += 1
    for i in range(n):
        # -p[i] - u[i] <= 0
        A_ub[row, i] = -1.0
        A_ub[row, n+i] = -1.0
        b_ub[row] = 0.0
        row += 1
    for i in range(n):
        # p[i] <= 1-h[i]
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - h[i]
        row += 1
    for i in range(n):
        # -p[i] <= h[i]
        A_ub[row, i] = -1.0
        b_ub[row] = h[i]
        row += 1
    # sum u[i] <= delta
    A_ub[row, n:2*n] = 1.0
    b_ub[row] = delta
    
    # Equality: sum p[i] = 0
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0

        h_t = project_simplex(h + p, n)
        if h_t is None:
            delta *= 0.5
            continue
        m = np.sum(np.abs(h_t - h))
        if m > delta:
            a = delta * (1 - 1e-12) / m
            h_t = project_simplex(h + a * (h_t - h), n)
            if h_t is None:
                delta *= 0.5
                continue
        if np.max(np.abs(h_t - h)) < 1e-14:
            delta *= 0.5
            continue
        C_t = exact_score(h_t)
        d_h = h_t - h
        pred = C_cur - float(np.max(q + G @ d_h))
        if C_t < C_cur - 1e-12:
            ared = C_cur - C_t
            h = h_t
            C_cur = C_t
            if pred > 1e-12:
                rho = ared / pred
                if rho < 0.1:
                    delta *= 0.5
                elif rho > 0.75 and m >= 0.8 * delta:
                    delta = min(2.0 * delta, delta_max)
            else:
                delta *= 0.5
        else:
            delta *= 0.5
    return h, C_cur

def add_unique(lst, v, n):
    if v is None or not is_feasible(v, n):
        return
    for u in lst:
        if np.max(np.abs(v - u)) < 1e-10:
            return
    lst.append(v)
    if len(lst) >= 8:
        return

def run(seed=42, **kwargs):
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    rng = np.random.default_rng(seed)

    parent_raw = globals().get('initial_h_values', None)
    parent = None
    if parent_raw is not None:
        try:
            p = np.asarray(parent_raw, dtype=np.float64)
            if p.ndim == 1 and len(p) >= 2:
                parent = p.copy()
        except Exception:
            pass

    n = len(parent) if parent is not None else None
    if n is None:
        try:
            np_ = int(kwargs.get('n_points', 42))
            if 2 <= np_ <= 256:
                n = np_
        except Exception:
            n = 42
    S = n / 2.0

    const_h = np.full(n, 0.5)
    const_h = project_simplex(const_h, n)
    if const_h is None:
        const_h = np.full(n, 0.5)
    C_const = exact_score(const_h)

    parent_fb = None
    C_parent_fb = float('inf')
    if parent is not None and np.all(np.isfinite(parent)):
        if np.all(parent >= -1e-12) and np.all(parent <= 1.0 + 1e-12) and abs(parent.sum() - S) < 1e-12:
            parent_fb = np.clip(parent, 0.0, 1.0)
        else:
            parent_fb = project_simplex(parent, n)
        if parent_fb is not None and is_feasible(parent_fb, n):
            C_parent_fb = exact_score(parent_fb)

    best_h = parent_fb if parent_fb is not None else const_h
    best_C = C_parent_fb if parent_fb is not None else C_const
    if C_const < best_C - 1e-12:
        best_h, best_C = const_h, C_const

    if n > 128:
        if parent_fb is not None and C_parent_fb < best_C - 1e-12:
            best_h, best_C = parent_fb, C_parent_fb
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        blk = project_simplex(blk, n)
        if blk is not None:
            Cb = exact_score(blk)
            if Cb < best_C - 1e-12:
                best_h, best_C = blk, Cb
        h_out = np.array(best_h, dtype=np.float64)
        return h_out, float(best_C), int(n)

    starts = []
    add_unique(starts, parent_fb, n)
    add_unique(starts, const_h, n)

    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    add_unique(starts, project_simplex(blk, n), n)

    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    add_unique(starts, project_simplex(alt, n), n)

    idx = np.arange(n)
    sin_v = 0.5 + 0.45 * np.sin(2.0 * math.pi * idx / n)
    add_unique(starts, project_simplex(sin_v, n), n)

    if parent_fb is not None:
        g1 = parent_fb + 0.12 * rng.normal(size=n)
        add_unique(starts, project_simplex(g1, n), n)
        g2 = parent_fb + 0.30 * rng.normal(size=n)
        add_unique(starts, project_simplex(g2, n), n)
        rv = rng.uniform(size=n)
        add_unique(starts, project_simplex(rv, n), n)
    else:
        for _ in range(4):
            rv = rng.uniform(size=n)
            add_unique(starts, project_simplex(rv, n), n)

    if not starts:
        starts = [const_h]

    for s in starts:
        C_s = exact_score(s)
        if C_s < best_C - 1e-12:
            best_h, best_C = s.copy(), C_s
        try:
            h_opt, C_opt = trust_search(s, n)
        except Exception:
            continue
        if C_opt < best_C - 1e-12:
            best_h, best_C = h_opt, C_opt

    if parent_fb is not None and C_parent_fb < best_C - 1e-12:
        best_h, best_C = parent_fb, C_parent_fb
    if C_const < best_C - 1e-12:
        best_h, best_C = const_h, C_const

    h_out = np.array(best_h, dtype=np.float64)
    if not is_feasible(h_out, n):
        h_out = project_simplex(h_out, n)
        if h_out is None:
            h_out = const_h.copy()
    C_out = exact_score(h_out)
    if not np.isfinite(C_out):
        h_out = parent_fb if parent_fb is not None else const_h
        C_out = exact_score(h_out)
    return h_out, float(C_out), int(n)
