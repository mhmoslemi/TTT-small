import numpy as np
import math

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = float(np.max(corr * dx))
    if not math.isfinite(c):
        return float('inf')
    return c

def is_feasible(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-15) or np.any(h > 1.0 + 1e-15):
        return False
    S = n / 2.0
    if abs(float(np.sum(h)) - S) > 1e-10:
        return False
    if abs(math.fsum(float(x) for x in h) - S) > 1e-10:
        return False
    return True

def restore(v, n):

        else:
            hi = mid

    residual = S - math.fsum(float(x) for x in h)

            add = min(residual, 1.0 - h[k])
            if add > 0:

                residual -= add

def compute_qG(h, n, dx):
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    # Zero lag
    q[0] = dx * float(np.sum(h * (1.0 - h)))
    G[0, :] = dx * (1.0 - 2.0 * h)
    # Shift lags
    for d in range(1, n):
        cnt = n - d
        h_a = h[d:d+cnt]  # h[i+d]
        h_b = h[0:cnt]    # h[i]
        q[2*d-1] = dx * float(np.sum(h_a * (1.0 - h_b)))
        q[2*d] = dx * float(np.sum(h_b * (1.0 - h_a)))
        # G[2d-1, m]: d/dh[m] of q[2d-1]
        # q[2d-1] = dx * sum_{i=0}^{cnt-1} (h[i+d] - h[i+d]*h[i])
        # d/dh[m]: 
        #   if m in [d, n-1] (m = i+d): +dx
        #   if m in [0, cnt-1] (m = i): -dx * h[m+d]
        #   if m in [d, n-1] (m = i+d, from product): -dx * h[m-d]
        idx_plus = np.arange(d, d+cnt)  # positions where m = i+d
        G[2*d-1, idx_plus] += dx
        G[2*d-1, 0:cnt] -= dx * h[0:cnt + d]  # -dx * h[i+d] for i in [0,cnt-1]
        # Wait, for m in [d, n-1], the product term: -dx * h[m-d]

        # G[2d, m]: d/dh[m] of q[2d]

        #   if m in [0, cnt-1] (m = i): +dx
        #   if m in [d, n-1] (m = i+d): -dx * h[m-d]
        #   if m in [0, cnt-1] (m = i, from product): -dx * h[m+d]
        G[2*d, 0:cnt] += dx
        G[2*d, d:d+cnt] -= dx * h[0:cnt]
        G[2*d, 0:cnt] -= dx * h[d:d+cnt]

def real_roots(A, B, C, lo, hi):
    tol = 1e-14 * max(1.0, abs(A), abs(B), abs(C))
    roots = []
    if abs(A) <= tol:
        if abs(B) > tol:
            r = -C / B
            if lo - 1e-10 <= r <= hi + 1e-10:
                roots.append(max(lo, min(hi, r)))
        return roots
    disc = B*B - 4*A*C
    if disc < -64*np.finfo(float).eps*(B*B + abs(4*A*C) + 1):
        return roots
    if disc < 0:
        disc = 0.0
    sqrt_d = math.sqrt(disc)
    z = -0.5 * (B + math.copysign(sqrt_d, B))
    if abs(z) > tol:
        r1 = z / A
        r2 = C / z
    else:
        r1 = r2 = -B / (2*A)
    for r in [r1, r2]:
        if math.isfinite(r) and lo - 1e-10 <= r <= hi + 1e-10:
            roots.append(max(lo, min(hi, r)))
    return roots

def build_hull(slopes, intercepts, lo, hi):

        return [], []

    # np.lexsort sorts by last key first

    hull_c = []
    hull_start = []
    
    for idx in order:
        s_val = float(slopes[idx])
        c_val = float(intercepts[idx])
        # Merge with last if slopes are very close
        if hull_s and abs(s_val - hull_s[-1]) < 1e-14 * max(1.0, abs(s_val), abs(hull_s[-1])):
            if c_val > hull_c[-1]:

                    c_old = hull_c[-2]
                    m_new = hull_s[-1]
                    c_new = hull_c[-1]
                    x = (c_old - c_new) / (m_new - m_old)
                    hull_start[-1] = max(lo, x)

            hull_s.append(s_val)
            hull_c.append(c_val)
            hull_start.append(lo)

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = float(np.max(corr * dx))
    return c if math.isfinite(c) else float('inf')

    top_k = min(6, len(ta))
    top = np.argsort(pm)[:top_k]
    best_h, best_sc = h.copy(), cur_sc
    psum = h[i] + h[j]
    for ti in top:
        t = float(ta[ti])
        ni = h[i] + t
        ni = max(0.0, min(1.0, ni))
        ni = max(psum - 1.0, min(psum, ni))
        if ni < -1e-12 or ni > 1.0 + 1e-12:
            continue
        nj = psum - ni
        ht = h.copy()
        ht[i] = ni
        ht[j] = nj
        if not is_feasible(ht, n):
            ht = restore(ht, n)
            if ht is None:
                continue
        sc = exact_score(ht)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_h = ht
    return best_h, best_sc

def make_starts(h_par, n, rng):
    starts = []
    h_c = np.full(n, 0.5)
    starts.append(h_c)
    hb = np.zeros(n)
    half = n // 2
    hb[:half] = 1.0
    if n % 2 == 1:
        hb[half] = 0.5
    starts.append(hb)
    ha = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    haa = restore(ha, n)
    if haa is not None:
        starts.append(haa)
    hs = 0.5 + 0.45 * np.sin(2.0 * np.pi * np.arange(n) / n)
    hss = restore(hs, n)
    if hss is not None:
        starts.append(hss)
    if h_par is not None and is_feasible(h_par, n):
        starts.append(h_par.copy())
    if n <= 48:
        if h_par is not None and is_feasible(h_par, n):
            for sd in (0.12, 0.35):
                hp = restore(h_par + rng.normal(0, sd, n), n)
                if hp is not None:
                    starts.append(hp)
            hr = restore(rng.uniform(0, 1, n), n)
            if hr is not None:
                starts.append(hr)
        else:
            for _ in range(4):
                hr = restore(rng.uniform(0, 1, n), n)
                if hr is not None:
                    starts.append(hr)
    dedup = []
    for sv in starts:
        if not is_feasible(sv, n):
            continue
        ok = True
        for ex in dedup:
            if np.max(np.abs(sv - ex)) < 1e-10:
                ok = False
                break
        if ok:
            dedup.append(sv)
    return dedup

def run(seed=42, **kwargs):
    n = 42
    h_par = None
    try:
        pv = np.array(initial_h_values, dtype=float)
        if pv.ndim == 1 and len(pv) >= 2:
            n = len(pv)
            h_par = pv.copy()
    except Exception:
        pass
    if h_par is None or not np.all(np.isfinite(h_par)) or len(h_par) != n:
        h_par = None
        try:
            nk = kwargs.get('n_points', None)
            if nk is not None and not isinstance(nk, bool) and float(nk) == int(nk) and 2 <= int(nk) <= 512:
                n = int(nk)
        except Exception:
            pass
    try:
        ss = int(seed)
    except Exception:
        ss = 42
    rng = np.random.default_rng(ss)
    dx = 2.0 / n
    h_const = np.full(n, 0.5)
    sc_const = exact_score(h_const)
    h_fb = None
    if h_par is not None:
        if is_feasible(h_par, n):
            h_fb = h_par.copy()
        else:
            hr = restore(h_par, n)
            if hr is not None:
                h_fb = hr
    h_best, sc_best = h_const.copy(), sc_const
    if h_fb is not None:
        scf = exact_score(h_fb)
        if scf < sc_best - 1e-12:
            sc_best, h_best = scf, h_fb.copy()
    if n > 128:
        return (np.array(h_best, dtype=float), float(sc_best), int(n))
    starts_list = make_starts(h_fb, n, rng)
    if not starts_list:
        starts_list = [h_const]
    if n <= 48:
        pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
    elif n <= 64:
        ps = set()
        for r in (1, 2, max(1, n//4), max(1, n//2)):
            for i in range(n):
                j2 = (i + r) % n
                if i != j2:
                    ps.add((min(i, j2), max(i, j2)))
        pairs = sorted(ps)
    else:
        ps = set()
        for r in (1, max(1, n//2)):
            for i in range(n):
                j2 = (i + r) % n
                if i != j2:
                    ps.add((min(i, j2), max(i, j2)))
        pairs = sorted(ps)
    nsw = 2 if n <= 64 else 1
    for st in starts_list:
        h_cur, sc_cur = st.copy(), exact_score(st)
        for _sw in range(nsw):
            perm = list(rng.permutation(len(pairs)))
            imp = False
            for pi in perm:
                pi2, pj = pairs[pi]
                hn, sn = pair_search(h_cur, pi2, pj, n, dx, sc_cur)
                if sn < sc_cur - 1e-12:
                    h_cur, sc_cur = hn, sn
                    imp = True
            if not imp:
                break
        if is_feasible(h_cur, n):
            scc = exact_score(h_cur)
            if scc < sc_best - 1e-12:
                sc_best, h_best = scc, h_cur.copy()
    if h_fb is not None and is_feasible(h_fb, n):
        scf2 = exact_score(h_fb)
        if scf2 < sc_best - 1e-12:
            sc_best, h_best = scf2, h_fb.copy()
    if not is_feasible(h_best, n):
        h_best = restore(h_best, n)
        if h_best is None:
            h_best = h_fb if h_fb is not None else h_const
            sc_best = exact_score(h_best)
        else:
            sc_best = exact_score(h_best)
    return (np.array(h_best, dtype=float), float(sc_best), int(n))
