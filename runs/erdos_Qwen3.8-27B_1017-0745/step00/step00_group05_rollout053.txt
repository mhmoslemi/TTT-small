        for _ in range(4):
            pr = capped_simplex_project(rng.uniform(0, 1, n), target)
            if pr is not None:
                starts.append(pr)
        for _ in range(3):
            pr = capped_simplex_project(rng.beta(0.5, 0.5, n), target)
            if pr is not None:
                starts.append(pr)

    tau_schedule = [0.02, 0.005, 0.001, 0.0002, 0.00005, 0.00001]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': mass_eq, 'jac': mass_eq_jac}]

    def try_solve(h_cur, tausteps, inc, incsc):
        for tau in tausteps:
            try:
                res = minimize(obj_and_grad, h_cur, args=(tau, dx),
                               method='SLSQP', jac=True, bounds=bounds,
                               constraints=cons,
                               options={'maxiter': 200, 'ftol': 1e-10, 'disp': False})
            except Exception:
                continue
            if res.x is None or res.x.shape != (n,) or not np.all(np.isfinite(res.x)):
                continue
            h_new = capped_simplex_project(res.x, target)
            if h_new is None or not is_feasible(h_new, target):
                continue
            sc = exact_score(h_new, dx)
            if np.isfinite(sc) and sc < incsc - 1e-12:
                inc = h_new.copy()
                incsc = sc
            F_new, _ = obj_and_grad(h_new, tau, dx)
            F_cur, _ = obj_and_grad(h_cur, tau, dx)
            if np.isfinite(F_new) and F_new <= F_cur + 1e-10:
                h_cur = h_new
        return h_cur, inc, incsc

    h_cur0, incumbent, incumbent_score = starts[0], incumbent, incumbent_score
    for i in range(len(starts)):
        st = starts[i]
        if st is None:
            continue
        if not is_feasible(st, target):
            st = capped_simplex_project(st, target)
            if st is None:
                continue
        sc = exact_score(st, dx)
        if np.isfinite(sc) and sc < incumbent_score - 1e-12:
            incumbent = st.copy()
            incumbent_score = sc
        _, incumbent, incumbent_score = try_solve(st.copy(), tau_schedule, incumbent, incumbent_score)

    center = incumbent if incumbent is not None else np.full(n, 0.5)
    for sigma in [0.02, 0.08, 0.20]:
        pert = capped_simplex_project(center + sigma * rng.standard_normal(n), target)
        if pert is None or not is_feasible(pert, target):
            continue
        sc = exact_score(pert, dx)
        if np.isfinite(sc) and sc < incumbent_score - 1e-12:
            incumbent = pert.copy()
            incumbent_score = sc
        center, incumbent, incumbent_score = try_solve(pert.copy(), tau_schedule, incumbent, incumbent_score)
        center = incumbent if incumbent is not None else np.full(n, 0.5)

    if incumbent is None:
        incumbent = capped_simplex_project(np.full(n, 0.5), target)
        if incumbent is None:
            incumbent = np.full(n, 0.5)
    if not is_feasible(incumbent, target):
        incumbent = capped_simplex_project(incumbent, target)
        if incumbent is None:
            incumbent = np.full(n, 0.5)

    c5 = exact_score(incumbent, dx)
    if not np.isfinite(c5):
        c5 = 0.5
    return (incumbent, c5, n)
