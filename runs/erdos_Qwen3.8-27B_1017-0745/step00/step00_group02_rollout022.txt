import numpy as np
import math

def score_and_lags(h, n):
    dx = 2.0 / n
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    s = float(np.max(q))
    if not np.isfinite(s):
        return h, float('inf'), q
    return h, s, q

def check_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(float(np.sum(h)) - n / 2.0) > 1e-10:
        return False
    return True

def project_feasible(y, n, S):
    if not np.all(np.isfinite(y)):
        return None
    lo_b = float(np.min(y)) - 1.0
    hi_b = float(np.max(y)) + 1.0
    mid = 0.5 * (lo_b + hi_b)
    for _ in range(100):
        mid = 0.5 * (lo_b + hi_b)
        if hi_b - lo_b <= 1e-14 * max(1.0, abs(mid)):
            break
        x = np.clip(y - mid, 0.0, 1.0)
        s = float(np.sum(x))
        if s > S:
            lo_b = mid
        else:
            hi_b = mid
    x = np.clip(y - mid, 0.0, 1.0)
    r = S - float(np.sum(x))
    if r > 1e-13:
        for i in range(n):
            if r <= 1e-13:
                break
            add = min(r, 1.0 - x[i])
            x[i] += add
            r -= add
    elif r < -1e-13:
        for i in range(n):
            if r >= -1e-13:
                break
            sub = min(-r, x[i])
            x[i] -= sub
            r += sub
    if abs(r) > 1e-10 or np.any(x < -1e-12) or np.any(x > 1.0 + 1e-12):
        return None
    return np.clip(x, 0.0, 1.0)

def repair_small_residual(h, n, S):
    if not np.all(np.isfinite(h)):
        return None
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    h = np.clip(h.copy(), 0.0, 1.0)
    r = S - float(np.sum(h))
    if abs(r) > 1e-8:
        return None
    if r > 1e-13:
        for i in range(n):
            if r <= 1e-13:
                break
            add = min(r, 1.0 - h[i])
            h[i] += add
            r -= add
    elif r < -1e-13:
        for i in range(n):
            if r >= -1e-13:
                break
            sub = min(-r, h[i])
            h[i] -= sub
            r += sub
    if abs(r) > 1e-10:
        return None
    return h

def compute_pair_coeffs(h, n, i, j, dx):
    b = np.zeros(2 * n - 1)
    c = np.zeros(2 * n - 1)
    c[n - 1] = -2.0 * dx
    diff = j - i
    if diff != 0:
        c[n - 1 + diff] += dx
        c[n - 1 - diff] += dx
    for ell_idx in range(2 * n - 1):
        ell = ell_idx - (n - 1)
        v = 0.0
        if 0 <= i - ell < n:
            v += 1.0 - h[i - ell]
        if 0 <= j - ell < n:
            v -= 1.0 - h[j - ell]
        if 0 <= i + ell < n:
            v -= h[i + ell]
        if 0 <= j + ell < n:
            v += h[j + ell]
        b[ell_idx] = dx * v
    return b, c

def eval_predicted(q, b, c, t_vals):
    res = q[None, :] + t_vals[:, None] * b[None, :] + (t_vals * t_vals)[:, None] * c[None, :]
    return np.max(res, axis=1)

def anneal_chain(h, n, dx, rng, T_mults, step_scales, props):
    S = n / 2.0
    h = h.copy()
    _, cur_score, _ = score_and_lags(h, n)
    best_h = h.copy()
    best_score = cur_score
    for s in range(len(T_mults)):
        T = T_mults[s] * dx
        step = step_scales[s]
        for _ in range(props):
            i = int(rng.integers(n))
            j = int(rng.integers(n - 1))
            if j >= i:
                j += 1
            t = float(rng.uniform(-step, step))
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if t < lo or t > hi or lo == hi:
                continue
            h_t = h.copy()
            h_t[i] += t
            h_t[j] -= t
            h_t = repair_small_residual(h_t, n, S)
            if h_t is None:

                h = h_t
                cur_score = ts
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_h = h.copy()
    return h, cur_score, best_h, best_score

            grid = np.sort(np.unique(grid))

            pred = eval_predicted(cur_q, bv, cv, grid)

            hc = h.copy()
            hc[i] += tb
            hc[j] -= tb
            hc = repair_small_residual(hc, n, S)

                continue
            _, cs, cq = score_and_lags(hc, n)

                continue
            h = hc
            cur_score = cs
            cur_q = cq
            improved = True
        if not improved or (ss_score - cur_score) < 1e-10:
            break
    return h, cur_score

    return item[1]
