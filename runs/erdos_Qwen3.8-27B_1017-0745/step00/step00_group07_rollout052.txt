def epi_fun(z):
    return z[-1]

def epi_fun_grad(z):
    g = np.zeros(len(z))

    return g

def epi_ineq(z):
    n = len(z) - 1
    B, _ = compute_lags(z[:n])
    return z[-1] - B

def epi_ineq_jac(z):
    n = len(z) - 1
    _, G = compute_lags(z[:n])
    J = np.column_stack([-G, np.ones(2 * n - 1)])
    return J

def epi_eq(z):
    n = len(z) - 1
    return np.sum(z[:n]) - n / 2.0

def epi_eq_jac(z):
