import numpy as np
from scipy.optimize import minimize
import math

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0, :] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        ip, im = 2 * d - 1, 2 * d
        hs, ho = h[d:], h[:n - d]
        q[ip] = dx * np.sum(hs * (1.0 - ho))
        G[ip, :n - d] = -dx * hs
        G[ip, d:] = dx * (1.0 - ho)
        q[im] = dx * np.sum(ho * (1.0 - hs))
        G[im, :n - d] = dx * (1.0 - hs)
        G[im, d:] = -dx * ho
    return q, G

def exact_score(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def project_simplex(v, n):
    S = n / 2.0
    v = np.clip(np.asarray(v, dtype=float), -1e6, 1e6)
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    res = S - float(np.sum(h))
    if abs(res) > 1e-12:
        for j in range(n):
            if 0.0 < h[j] < 1.0:
                others = float(np.sum(np.delete(h, j)))
                if 0.0 <= S - others <= 1.0:
                    h[j] = S - others
                break
    return np.clip(h, 0.0, 1.0)

def soft_func(h, tau):
    q, _ = compute_lags(h)
    qmax = float(np.max(q))
    sh = (q - qmax) / tau
    ws = float(np.sum(np.exp(sh)))
    return qmax + tau * math.log(ws)

def soft_jac(h, tau):
    q, G = compute_lags(h)
    qmax = float(np.max(q))
    w = np.exp((q - qmax) / tau)
    p = w / float(np.sum(w))
    return p @ G

def mass_eq(h):
    return np.sum(h) - len(h) / 2.0

def mass_eq_jac(h):
    return np.ones(len(h))

def epi_obj(z):
    return z[-1]

def epi_obj_jac(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def epi_eq(z):
    return np.sum(z[:-1]) - len(z) / 2.0 + 0.5

def epi_eq_jac(z):
    g = np.ones(len(z))
    g[-1] = 0.0
    return g

def epi_ineq(z):
    n = len(z) - 1
    q, _ = compute_lags(z[:n])
    return z[-1] - q

def epi_ineq_jac(z):
    n = len(z) - 1
    _, G = compute_lags(z[:n])
    M = 2 * n - 1
    J = np.zeros((M, n + 1))
    J[:, :n] = -G
    J[:, -1] = 1.0
    return J

def build_starts(n, parent, rng):
    starts = []
    if parent is not None:
        starts.append(parent)
        for sig in [0.10, 0.25, 0.50]:
            starts.append(parent + sig * rng.standard_normal(n))
    blk = np.zeros(n)
    half = n // 2
    blk[:half] = 1.0
    if n % 2 == 1:
        blk[half] = 0.5
    starts.append(blk)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    starts.append(alt)
    n_rand = 2 if parent is not None else 4
    for _ in range(n_rand):
        starts.append(rng.random(n))
    proj = []
    for s in starts:
        ps = project_simplex(s, n)
        if not proj or np.max(np.abs(ps - np.array(proj))) > 1e-10:
            proj.append(ps)
    return proj

def run(seed=42, **kwargs):
    global initial_h_values
    rng = np.random.default_rng(seed)
    parent = None
    n = 42
    try:
        p = np.asarray(initial_h_values, dtype=float).ravel()
        if p.size >= 2:
            n = p.size
            if np.all(np.isfinite(p)):
                parent = project_simplex(p, n)
    except (NameError, Exception):
        pass
    if 'n_points' in kwargs and parent is None:
        try:
            nn = int(kwargs['n_points'])
            if 2 <= nn <= 256:
                n = nn
        except Exception:
            pass
    S = n / 2.0
    const = np.full(n, 0.5)
    best_h = const.copy()
    best_c5 = exact_score(const)
    if parent is not None:
        pc5 = exact_score(parent)
        if pc5 < best_c5 - 1e-12:
            best_h, best_c5 = parent.copy(), pc5
    bounds = [(0.0, 1.0)] * n
    cons_soft = [{'type': 'eq', 'fun': mass_eq, 'jac': mass_eq_jac}]
    temps = [0.02, 0.005, 0.001, 0.0002]
    starts = build_starts(n, parent, rng)
    for s0 in starts:
        h_cur = s0.copy()
        sch = exact_score(h_cur)
        if sch < best_c5 - 1e-12:
            best_h, best_c5 = h_cur.copy(), sch
        for tau in temps:
            try:
                res = minimize(soft_func, h_cur, jac=soft_jac, method='SLSQP',
                               args=(tau,), bounds=bounds, constraints=cons_soft,
                               options={'ftol': 1e-10, 'maxiter': 120})
                if hasattr(res, 'x') and res.x is not None:
                    h_new = project_simplex(res.x, n)
                    if np.all(np.isfinite(h_new)):
                        nm = float(np.sum(h_new))
                        if abs(nm - S) > 1e-9:
                            h_new = project_simplex(h_new, n)
                        cv, _ = soft_func(h_new, tau)
                        cv_cur, _ = soft_func(h_cur, tau)
                        if cv <= cv_cur + 1e-12:
                            h_cur = h_new
                        cs = exact_score(h_cur)
                        if cs < best_c5 - 1e-12:
                            best_h, best_c5 = h_cur.copy(), cs
            except Exception:
                pass
        z0 = np.zeros(n + 1)
        q0, _ = compute_lags(h_cur)
        t0 = float(np.max(q0)) + max(1e-8, 1e-6 * abs(float(np.max(q0))))
        z0[:n] = h_cur
        z0[-1] = t0
        bnd_epi = bounds + [(0.0, None)]
        cons_epi = [
            {'type': 'eq', 'fun': epi_eq, 'jac': epi_eq_jac},
            {'type': 'ineq', 'fun': epi_ineq, 'jac': epi_ineq_jac}
        ]
        try:

            if hasattr(res2, 'x') and res2.x is not None:
                h_ep = project_simplex(res2.x[:n], n)
                if np.all(np.isfinite(h_ep)):
                    ce = exact_score(h_ep)
                    if ce < best_c5 - 1e-12:
                        best_h, best_c5 = h_ep.copy(), ce
        except Exception:
            pass
    z0 = np.zeros(n + 1)
    q0, _ = compute_lags(best_h)
    t0 = float(np.max(q0)) + max(1e-8, 1e-6 * abs(float(np.max(q0))))
    z0[:n] = best_h
    z0[-1] = t0
    bnd_epi = bounds + [(0.0, None)]
    cons_epi = [
        {'type': 'eq', 'fun': epi_eq, 'jac': epi_eq_jac},
        {'type': 'ineq', 'fun': epi_ineq, 'jac': epi_ineq_jac}
    ]
    try:
        res3 = minimize(epi_obj, z0, jac=epi_obj_jac, method='SLSQP',
                        bounds=bnd_epi, constraints=cons_epi,
                        options={'ftol': 1e-10, 'maxiter': 200})
        if hasattr(res3, 'x') and res3.x is not None:
            h_fin = project_simplex(res3.x[:n], n)
            if np.all(np.isfinite(h_fin)):
                cf = exact_score(h_fin)
                if cf < best_c5 - 1e-12:
                    best_h, best_c5 = h_fin.copy(), cf
    except Exception:
        pass
    best_h = project_simplex(best_h, n)
    if not np.all((best_h >= 0.0) & (best_h <= 1.0)) or abs(float(np.sum(best_h)) - S) > 1e-9:
        best_h = const.copy()
        best_c5 = exact_score(const)
    else:
        best_c5 = exact_score(best_h)
    return best_h, float(best_c5), int(n)
