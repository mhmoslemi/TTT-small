    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-10

def exact_score(h, n):
    dx = 2.0 / n
    q = np.correlate(h, 1.0 - h, mode='full')
    c = float(np.max(q)) * dx
    return c if np.isfinite(c) else None

def compute_lags_grads(h, n):
    q = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(q)):
        return None, None
    G = np.zeros((2 * n - 1, n), dtype=np.float64)
    for t in range(-(n - 1), n):
        row = t + n - 1
        j0 = max(0, t)
        j1 = min(n, t + n)
        if j0 < j1:
            G[row, j0:j1] += 1.0 - h[j0 - t:j1 - t]
        k0 = max(0, -t)
        k1 = min(n, n - t)
        if k0 < k1:
            G[row, k0:k1] -= h[k0 + t:k1 + t]
    if not np.all(np.isfinite(G)):
        return None, None
    return q, G

def slp_search(h0, n, dx, rng):
    current = h0.copy()
    best = h0.copy()
    best_score = exact_score(current, n)
    if best_score is None:
        return best
    rho = 0.08
    rho_max, rho_min = 0.25, 1e-5
    max_iter = 50 if n <= 128 else (25 if n <= 512 else 12)
    for _ in range(max_iter):
        if rho < rho_min:
            break
        q, G = compute_lags_grads(current, n)
        if q is None:
            break
        Q = float(np.max(q))
        m = 2 * n - 1
        A_ub = np.column_stack([G, -np.ones(m)])
        b_ub = -q.copy()
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        bounds = [(max(-rho, -current[i]), min(rho, 1.0 - current[i])) for i in range(n)]
        bounds.append((None, None))
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'maxiter': 300})
        except Exception:
            rho = max(rho_min, rho * 0.5)
            continue
        if not res.success or res.x is None or len(res.x) != n + 1 or not np.all(np.isfinite(res.x)):
            rho = max(rho_min, rho * 0.5)
            continue
        d = res.x[:n]
        z_lp = res.x[n]
        cand = current + d
        proj = capped_simplex_project(cand, n / 2.0)
        if proj is None or not check_feasible(proj, n):
            rho = max(rho_min, rho * 0.5)
            continue
        cand_score = exact_score(proj, n)
        if cand_score is None:
            rho = max(rho_min, rho * 0.5)
            continue
        if cand_score < best_score - 1e-12:
            best = proj.copy()
            best_score = cand_score
        q_new = float(np.max(np.correlate(proj, 1.0 - proj, mode='full')))
        predicted = Q - z_lp
        actual = Q - q_new
        rho_before = rho
        step_mag = float(np.max(np.abs(proj - current)))
        accepted = actual > 1e-12 * max(1.0, abs(Q))
        if accepted:
            current = proj.copy()
        denom = max(1.0, abs(Q)) * 1e-10
        if predicted < denom:
            rho = max(rho_min, rho * 0.5) if not accepted else rho
        elif not accepted or actual / predicted < 0.1:
            rho = max(rho_min, rho * 0.5)
        elif accepted and actual / predicted > 0.75 and step_mag >= 0.8 * rho_before:
            rho = min(rho_max, rho * 1.5)
    return best

def pairwise_poll(h, n, dx, rng):
    current = h.copy()
    best = h.copy()
    best_score = exact_score(current, n)
    if best_score is None or n > 256:
        return best
    n_pairs = 24 if n <= 128 else 8
    for r in [0.08, 0.025, 0.008]:
        for _ in range(n_pairs):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n - 1))
            if b >= a:
                b += 1
            s_lo = max(-current[a], current[b] - 1.0, -r)
            s_hi = min(1.0 - current[a], current[b], r)
            if s_hi - s_lo < 1e-14:
                continue
            pair_best, pair_best_sc = None, best_score
            for s in [s_lo, s_hi, min(r, s_hi), max(-r, s_lo), 0.0]:
                sc = s
                if sc < s_lo or sc > s_hi or abs(sc) < 1e-14:
                    continue
                cand = current.copy()
                cand[a] += sc
                cand[b] -= sc
                proj = capped_simplex_project(cand, n / 2.0)
                if proj is None or not check_feasible(proj, n):
                    continue
                ps = exact_score(proj, n)
                if ps is not None and ps < pair_best_sc - 1e-12:
                    pair_best, pair_best_sc = proj, ps
            if pair_best is not None:
                current = pair_best
                if pair_best_sc < best_score - 1e-12:
                    best, best_score = pair_best.copy(), pair_best_sc
    return best

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    n = 70
    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=np.float64)
        if pv.ndim == 1 and 2 <= pv.shape[0] <= 999:
            n = pv.shape[0]
            if np.all(np.isfinite(pv)) and float(np.max(np.abs(pv))) <= 1e6:
                parent = pv.copy()
    except Exception:
        pass
    dx = 2.0 / n
    target = n / 2.0
    u = np.full(n, 0.5)
    fallback, fallback_score = u.copy(), exact_score(u, n)
    if parent is not None:
        if np.min(parent) >= -1e-10 and np.max(parent) <= 1.0 + 1e-10 and abs(np.sum(parent) - target) <= 1e-8:
            pp = capped_simplex_project(parent, target)
            if pp is not None and check_feasible(pp, n):
                fallback = pp.copy()
                fs = exact_score(fallback, n)
                if fs is not None:
                    fallback_score = fs
    starts = []
    if parent is not None:
        starts.append(fallback.copy())
        starts.append(fallback[::-1].copy())
        for sc in [0.035, 0.12]:
            p = capped_simplex_project(fallback + sc * rng.standard_normal(n), target)
            if p is not None:
                starts.append(p)
    else:
        starts.append(u.copy())
        for _ in range(3):
            p = capped_simplex_project(rng.random(n), target)
            if p is not None:
                starts.append(p)
    for _ in range(2):
        p = capped_simplex_project(rng.beta(0.5, 0.5, size=n), target)
        if p is not None:
            starts.append(p)
    valid_starts = []
    for s in starts:
        if s is None or s.shape != (n,) or not np.all(np.isfinite(s)):
            continue
        if not check_feasible(s, n):
            s = capped_simplex_project(s, target)
            if s is None or not check_feasible(s, n):
                continue
        if any(np.max(np.abs(s - v)) < 1e-10 for v in valid_starts):
            continue
        valid_starts.append(s)
    incumbent, incumbent_score = fallback.copy(), fallback_score
    for s in valid_starts:
        sc = exact_score(s, n)
        if sc is not None and sc < incumbent_score - 1e-12:
            incumbent, incumbent_score = s.copy(), sc
        sr = slp_search(s, n, dx, rng)
        sr_sc = exact_score(sr, n)
        if sr_sc is not None and sr_sc < incumbent_score - 1e-12:
            incumbent, incumbent_score = sr.copy(), sr_sc
        pr = pairwise_poll(sr, n, dx, rng)
        pr_sc = exact_score(pr, n)
        if pr_sc is not None and pr_sc < incumbent_score - 1e-12:
            incumbent, incumbent_score = pr.copy(), pr_sc
        if incumbent_score <= 0.3808:
            break
    for sc in [0.025, 0.08]:
        noisy = capped_simplex_project(incumbent + sc * rng.standard_normal(n), target)
        if noisy is None or not check_feasible(noisy, n):
            continue
        ns = exact_score(noisy, n)
        if ns is not None and ns < incumbent_score - 1e-12:
            incumbent, incumbent_score = noisy.copy(), ns
        sr = slp_search(noisy, n, dx, rng)
        sr_sc = exact_score(sr, n)
        if sr_sc is not None and sr_sc < incumbent_score - 1e-12:
            incumbent, incumbent_score = sr.copy(), sr_sc
        pr = pairwise_poll(sr, n, dx, rng)
        pr_sc = exact_score(pr, n)
        if pr_sc is not None and pr_sc < incumbent_score - 1e-12:
            incumbent, incumbent_score = pr.copy(), pr_sc
        if incumbent_score <= 0.3808:
            break
    final = capped_simplex_project(incumbent, target)
    if final is None or not check_feasible(final, n):
        final = fallback
    final_score = exact_score(final, n)
    if final_score is None or not check_feasible(final, n):
        final, final_score = u, exact_score(u, n)
    return (final, final_score, n)
