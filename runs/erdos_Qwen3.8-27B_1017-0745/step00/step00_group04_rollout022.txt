def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def feas_ok(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    S = n / 2.0
    return abs(np.sum(h) - S) <= 1e-10 and abs(math.fsum(float(x) for x in h) - S) <= 1e-10

def project_capped(v, n):
    S = n / 2.0
    v = np.clip(v.astype(np.float64), -1e6, 1e6)
    lo, hi = min(v) - 1.0, max(v)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        c = np.clip(v - mid, 0.0, 1.0)
        if np.sum(c) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(10):
        res = S - math.fsum(float(x) for x in h)
        if abs(res) < 1e-14:
            break
        if res > 0:
            for i in range(n):
                a = min(res, 1.0 - h[i]); h[i] += a; res -= a
                if res <= 0: break
        else:
            for i in range(n):
                a = min(-res, h[i]); h[i] -= a; res += a
                if res >= 0: break
    h = np.clip(h, 0.0, 1.0)
    return h if feas_ok(h, n) else None

def compute_lags(h, n):
    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0] = dx * (1.0 - 2.0 * h)
    idx = 1
    for r in range(1, n):
        ns = n - r
        q[idx] = dx * np.sum(h[r:r+ns] * (1.0 - h[:ns]))
        G[idx, :ns] += -dx * h[r:r+ns]
        G[idx, r:r+ns] += dx * (1.0 - h[:ns])
        idx += 1
        q[idx] = dx * np.sum(h[:ns] * (1.0 - h[r:r+ns]))
        G[idx, :ns] += dx * (1.0 - h[r:r+ns])
        G[idx, r:r+ns] += -dx * h[:ns]
        idx += 1
    return q, G

def lp_step(h, q, G, n, delta):
    dx = 2.0 / n
    M = 2 * n - 1
    e = dx * delta * delta
    nv = 2 * n + 1
    c = np.zeros(nv); c[nv-1] = 1.0
    nr = M + 4 * n + 1
    A = np.zeros((nr, nv))
    b = np.zeros(nr)
    r = 0
    for l in range(M):
        A[r, :n] = G[l]
        A[r, nv-1] = -1.0
        b[r] = -q[l] - e
        r += 1
    I = np.eye(n)
    A[r:r+n, :n] = I; A[r:r+n, n:2*n] = -I; r += n
    A[r:r+n, :n] = -I; A[r:r+n, n:2*n] = -I; r += n
    A[r:r+n, :n] = I; b[r:r+n] = 1.0 - h; r += n
    A[r:r+n, :n] = -I; b[r:r+n] = h; r += n
    A[r, n:2*n] = 1.0; b[r] = delta
    A_eq = np.zeros((1, nv)); A_eq[0, :n] = 1.0
    bd = [(None, None)] * n + [(0.0, None)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=np.zeros(1), bounds=bd, method='highs')
    except Exception:
        return None
    if res.success and res.x is not None and len(res.x) == nv and np.all(np.isfinite(res.x)):
        return res.x[:n]
    return None

def trust_search(h0, n):
    h = h0.copy()
    C = exact_score(h)
    dmax = max(1.0, 0.12 * n)
    delta = min(max(0.25, 0.03 * n), dmax)
    for _ in range(50):
        if delta < 1e-4:
            break
        q, G = compute_lags(h, n)
        if not np.all(np.isfinite(G)):
            delta *= 0.5; continue
        p = lp_step(h, q, G, n, delta)
        if p is None:
            delta *= 0.5; continue
        oh = h.copy()
        ht = project_capped(oh + p, n)
        if ht is None:
            delta *= 0.5; continue
        m = np.sum(np.abs(ht - oh))
        if m > delta:
            a = delta * (1.0 - 1e-12) / m
            ht = project_capped(oh + a * (ht - oh), n)
            if ht is None:
                delta *= 0.5; continue
        if np.max(np.abs(ht - oh)) < 1e-14:
            delta *= 0.5; continue
        Ct = exact_score(ht)
        if not np.isfinite(Ct):
            delta *= 0.5; continue
        if Ct < C - 1e-12:
            disp = ht - oh
            pred = C - np.max(q + G @ disp)
            ared = C - Ct
            h = ht; C = Ct
            if pred > 1e-12:
                rho = ared / pred
                if rho < 0.1: delta *= 0.5
                elif rho > 0.75 and m >= 0.8 * delta: delta = min(2.0 * delta, dmax)
            else:
                delta *= 0.5
        else:
            delta *= 0.5
    return h, C
