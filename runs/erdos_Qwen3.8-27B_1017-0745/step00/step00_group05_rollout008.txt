            restart_count += 1
    final_proj, final_sc = project_and_check(inc_h, n, target)
    if final_proj is None:
        final_proj, final_sc = project_and_check(fallback, n, target)
    if final_proj is None:
        final_proj = np.full(n, 0.5)
        final_sc = exact_score(final_proj, dx)
    return (final_proj, final_sc, n)
