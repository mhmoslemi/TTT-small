        else:

    if not (np.isfinite(a_lo) and np.isfinite(a_hi)):
        return None
    if a_hi < a_lo - 1e-12:
        return None
    a_lo = min(a_lo, -1e-15); a_hi = max(a_hi, 1e-15)
    return a_lo, a_hi

def quad_roots(A, B, C):
    scale = max(1.0, abs(A), abs(B), abs(C))
    tol = 1e-13 * scale
    if abs(A) < tol:
        if abs(B) < tol:
            return []
        return [-C / B]
    disc = B * B - 4.0 * A * C
    if disc < -tol:
        return []
    disc = max(disc, 0.0)
    sq = math.sqrt(disc)
    q = -0.5 * (B + sq) if B >= 0 else -0.5 * (B - sq)
    if abs(q) < 1e-300:
        return [-B / (2.0 * A)]
    r1 = q / A
    r2 = C / q if abs(C) > 1e-300 else r1
    return [r1, r2]

def line_search(h, d, dx, n):
    coeffs = compute_line_coeffs(h, d)
    if coeffs is None:
        return None
    a, b, c = coeffs
    iv = feasible_interval(h, d)
    if iv is None:
        return None
    a_lo, a_hi = iv
    m = len(a)
    cands = [a_lo, a_hi, 0.0]
    if n <= 96:
        for r in range(m):
            for s in range(r + 1, m):
                for rt in quad_roots(c[r]-c[s], b[r]-b[s], a[r]-a[s]):
                    if np.isfinite(rt) and a_lo - 1e-12 <= rt <= a_hi + 1e-12:
                        cands.append(rt)
            if abs(c[r]) > 1e-15:
                sp = -b[r] / (2.0 * c[r])
                if np.isfinite(sp) and a_lo <= sp <= a_hi:
                    cands.append(sp)
    else:
        for k in range(65):
            cands.append(a_lo + k * (a_hi - a_lo) / 64.0)
        idx = np.argsort(a)[-24:]
        for j2 in range(len(idx)):
            r = idx[j2]
            if abs(c[r]) > 1e-15:
                sp = -b[r] / (2.0 * c[r])
                if np.isfinite(sp) and a_lo <= sp <= a_hi:
                    cands.append(sp)
            for k2 in range(j2 + 1, len(idx)):
                s = idx[k2]
                for rt in quad_roots(c[r]-c[s], b[r]-b[s], a[r]-a[s]):
                    if np.isfinite(rt) and a_lo <= rt <= a_hi:
                        cands.append(rt)
    cands.sort()
    filt = [cands[0]]
    for cv in cands[1:]:
        if cv - filt[-1] > 1e-12 * max(1.0, abs(cv)):
            filt.append(cv)
    # Evaluate in batches
    scored_alphas = []
    for i0 in range(0, len(filt), 512):
        batch = np.array(filt[i0:i0+512])
        vals = a[None,:] + b[None,:]*batch[:,None] + c[None,:]*batch[:,None]**2
        scored_alphas.extend(zip(list(batch), list(np.max(vals, axis=1))))
    scored_alphas.sort(key=lambda x: (x[1], abs(x[0])))
    kw = 6 if n <= 96 else 4
    short = [sa[0] for sa in scored_alphas[:kw]]
    if 0.0 not in short:
        short[-1] = 0.0
    best_h, best_sc = None, float('inf')
    target = n / 2.0
    for alpha in short:
        h_new = h + alpha * d
        hp = capped_simplex_project(h_new, target)
        if hp is None:
            continue
        sc = exact_score(hp, dx)
        if sc < best_sc:
            best_sc = sc; best_h = hp
    return (best_h, best_sc) if best_h is not None else None

def build_directions(h, rng, n):
    dirs = []
    q = np.correlate(h, 1.0 - h, mode='full')
    top3 = np.argsort(q)[-3:]
    for t_idx in top3:
        t = t_idx - (n - 1)
        g = np.zeros(n)
        for i in range(n):
            j = i - t
            if 0 <= j < n:
                g[i] += (1.0 - h[j])
            j2 = i + t
            if 0 <= j2 < n:
                g[i] -= h[j2]
        d = normalize_direction(-g)
        if d is not None:
            dirs.append(d)
    kmax = min(6, n-1) if n <= 96 else 4
    for k in range(1, kmax + 1):
        d = normalize_direction(np.cos(np.pi * k * (np.arange(n)+0.5)/n))
        if d is not None:
            dirs.append(d)
    npairs = 6 if n <= 96 else 4
    for r in range(npairs):
        ai = int(math.floor(r * n / npairs))
        bi = (ai + 1) % n if r % 2 == 0 else (ai + max(1, n//2)) % n
        if bi == ai:
            bi = (ai + 1) % n
        d = np.zeros(n); d[ai] = 1.0; d[bi] = -1.0
        dirs.append(d)
    nrand = 4 if n <= 96 else 2
    for _ in range(nrand):
        pair = rng.choice(n, size=2, replace=False)
        d = np.zeros(n); d[int(pair[0])] = 1.0; d[int(pair[1])] = -1.0
        dirs.append(d)
    ndense = 2 if n <= 96 else 1
    for _ in range(ndense):
        d = normalize_direction(rng.standard_normal(n))
        if d is not None:
            dirs.append(d)
    return dirs

def local_search(h0, dx, rng, n, sweep_limit):
    h = np.array(h0, dtype=np.float64)
    cur_score = exact_score(h, dx)
    best_h, best_score = np.array(h, dtype=np.float64), cur_score
    for sweep in range(sweep_limit):
        dirs = build_directions(h, rng, n)
        improved = False
        for d in dirs:
            nd = normalize_direction(d)
            if nd is None:
                continue
            res = line_search(h, nd, dx, n)
            if res is None:
                continue
            hun, sc = res
            if sc < cur_score - 1e-12:
                h = np.array(hun); cur_score = sc
                improved = True
                if sc < best_score:
                    best_score = sc; best_h = np.array(h)
                if sc <= 0.3808:
                    return best_h, best_score
        if not improved:
            break
    return best_h, best_score

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    n, parent, parent_fallback = 70, None, None
    try:
        ivil = np.array(initial_h_values, dtype=np.float64)
        if ivil.ndim == 1 and 2 <= len(ivil) <= 999:
            n = len(ivil)
            if np.all(np.isfinite(ivil)) and np.max(np.abs(ivil)) <= 1e6:
                p = capped_simplex_project(ivil, n/2.0)
                if p is not None:
                    parent = p
                if np.all(ivil >= -1e-10) and np.all(ivil <= 1+1e-10) and abs(float(np.sum(ivil))-n/2.0) <= 1e-8:
                    fb = capped_simplex_project(ivil, n/2.0)
                    if fb is not None:
                        parent_fallback = fb
    except Exception:
        n = 70
    dx = 2.0 / n
    target = n / 2.0
    u = np.full(n, 0.5)
    block = np.zeros(n)
    block[:n//2] = 1.0
    if n % 2 == 1:
        block[n//2] = 0.5
    starts = []
    if parent is not None:
        starts.extend([parent, parent[::-1].copy()])
        for sc in (0.04, 0.12):
            pp = capped_simplex_project(parent + sc*rng.standard_normal(n), target)
            if pp is not None:
                starts.append(pp)
        for _ in range(2):
            pp = capped_simplex_project(rng.uniform(size=n), target)
            if pp is not None:
                starts.append(pp)
            pp = capped_simplex_project(rng.beta(0.5, 0.5, n), target)
            if pp is not None:
                starts.append(pp)
    starts.extend([u, block])
    if parent is None:
        for _ in range(2):
            pp = capped_simplex_project(rng.uniform(size=n), target)
            if pp is not None:
                starts.append(pp)
            pp = capped_simplex_project(rng.beta(0.5, 0.5, n), target)
            if pp is not None:
                starts.append(pp)
    unique = [starts[0]]
    for s in starts[1:]:
        if all(np.max(np.abs(s-uu)) > 1e-10 for uu in unique):
            unique.append(s)
    incumbent = np.array(u)
    inc_score = exact_score(u, dx)
    scored = []
    for s in unique:
        sc = exact_score(s, dx)
        scored.append((sc, s))
        if sc < inc_score - 1e-12:
            inc_score = sc; incumbent = np.array(s)
    scored.sort(key=lambda x: x[0])
    sel = []
    if parent is not None:
        for sc, s in scored:
            if np.max(np.abs(s - parent)) < 1e-10:
                sel.append(s); break
    for sc, s in scored:
        if all(np.max(np.abs(s-ss)) > 1e-10 for ss in sel) and len(sel) < 3:
            sel.append(s)
            if len(sel) >= 3:
                break
    slimit = 2 if n <= 96 else 1
    for ss in sel:
        bh, bsc = local_search(ss, dx, rng, n, slimit)
        if bsc < inc_score - 1e-12:
            inc_score = bsc; incumbent = bh
        if inc_score <= 0.3808:
            break
    nre = 2 if n <= 96 else 1
    for i in range(nre):
        sig = 0.02 if i == 0 else 0.08
        pp = capped_simplex_project(incumbent + sig*rng.standard_normal(n), target)
        if pp is not None:
            bh, bsc = local_search(pp, dx, rng, n, 1)
            if bsc < inc_score - 1e-12:
                inc_score = bsc; incumbent = bh
        if inc_score <= 0.3808:
            break
    final = capped_simplex_project(incumbent, target)
    if final is None:
        final = parent_fallback if parent_fallback is not None else u
    final = np.clip(final, 0.0, 1.0)
    c5 = exact_score(final, dx)
    if not np.isfinite(c5):
        final = u; c5 = exact_score(u, dx)
    return (final, c5, n)
