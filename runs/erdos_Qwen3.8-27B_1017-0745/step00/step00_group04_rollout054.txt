import numpy as np
from scipy.optimize import minimize
import math

def lag_values(h):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        q[n-1+d] = dx * np.sum(h[d:m+d] * (1.0 - h[:m]))
        G[n-1+d, d:m+d] += dx * (1.0 - h[:m])
        G[n-1+d, :m] += -dx * h[d:m+d]
        q[n-1-d] = dx * np.sum(h[:m] * (1.0 - h[d:m+d]))
        G[n-1-d, :m] += dx * (1.0 - h[d:m+d])
        G[n-1-d, d:m+d] += -dx * h[:m]
    return q, G

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_feasible(v, n):
    S = n / 2.0
    v = np.asarray(v, dtype=float)
    v = np.clip(v, -1e6, 1e6)
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    res = S - math.fsum(h)
    if res > 1e-14:
        cap = 1.0 - h
        tc = np.sum(cap)
        if tc > 1e-14:
            h = np.clip(h + res * cap / tc, 0.0, 1.0)
    elif res < -1e-14:
        tt = np.sum(h)
        if tt > 1e-14:
            h = np.clip(h + res * h / tt, 0.0, 1.0)
    res2 = S - math.fsum(h)
    for j in range(n):
        if abs(res2) < 1e-12:
            break
        if res2 > 0 and h[j] < 1.0:
            a = min(res2, 1.0 - h[j]); h[j] += a; res2 -= a
        elif res2 < 0 and h[j] > 0:
            a = min(-res2, h[j]); h[j] -= a; res2 += a
    if not (np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) - S) < 1e-9):
        h = np.full(n, 0.5)
    return h

def soft_func(x, tau):
    q, _ = lag_values(x)
    qm = np.max(q)
    return qm + tau * np.log(np.sum(np.exp((q - qm) / tau)))

def soft_grad(x, tau):
    q, G = lag_values(x)
    qm = np.max(q)
    e = np.exp((q - qm) / tau)
    return G.T @ (e / np.sum(e))

def eq_sum(x, n):
    return np.sum(x) - n / 2.0

def eq_sum_grad(x, n):
    return np.ones(n)

def epi_obj(z):
    return z[-1]

def epi_obj_grad(z):
    g = np.zeros(len(z)); g[-1] = 1.0
    return g

def epi_eq(z):
    return np.sum(z[:-1]) - len(z[:-1]) / 2.0

def epi_eq_grad(z):
    g = np.zeros(len(z)); g[:-1] = 1.0
    return g

def epi_ineq(z):
    return z[-1] - lag_values(z[:-1])[0]

def epi_ineq_grad(z):
    n = len(z) - 1
    _, G = lag_values(z[:-1])
    J = np.zeros((2*n-1, n+1))
    J[:, :n] = -G; J[:, n] = 1.0
    return J

def build_starts(n, parent, rng):
    starts = []
    if parent is not None:
        p = project_feasible(parent, n)
        starts.append(p)
        for sig in [0.10, 0.25, 0.50]:
            starts.append(project_feasible(p + sig * rng.standard_normal(n), n))
    else:
        starts.append(np.full(n, 0.5))
    b = np.zeros(n); b[:n//2] = 1.0
    if n % 2 == 1:
        b[n//2] = 0.5
    starts.append(project_feasible(b, n))
    alt = np.zeros(n); alt[0::2] = 1.0; alt[1::2] = 0.5
    starts.append(project_feasible(alt, n))
    i = np.arange(n)
    starts.append(project_feasible(0.5 + 0.45*np.sin(2*np.pi*i/n), n))
    nr = 2 if parent is not None else 4
    for _ in range(nr):
        starts.append(project_feasible(rng.random(n), n))
    seen = []
    for s in starts:
        dup = False
        for t in seen:
            if np.max(np.abs(s - t)) < 1e-10:
                dup = True; break
        if not dup:
            seen.append(s)
    return seen

def polish_epigraph(x, n):
    q_c, _ = lag_values(x)
    t0 = float(np.max(q_c)) + max(1e-8, 1e-6*abs(float(np.max(q_c))))
    z0 = np.concatenate([x, [t0]])
    zb = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = minimize(epi_obj, z0, jac=epi_obj_grad, method='SLSQP',
                       bounds=zb,
                       constraints=[{'type':'eq','fun':epi_eq,'jac':epi_eq_grad},
                                    {'type':'ineq','fun':epi_ineq,'jac':epi_ineq_grad}],
                       options={'ftol':1e-10,'maxiter':200})
        if res.x is not None and len(res.x)==n+1 and np.all(np.isfinite(res.x)):
            return project_feasible(res.x[:n], n)
    except Exception:
        pass
    return None
