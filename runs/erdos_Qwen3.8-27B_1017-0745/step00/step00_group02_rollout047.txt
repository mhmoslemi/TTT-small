def _proj_K(y, S, n):
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return None
    lo, hi = float(y.min()) - 1.0, float(y.max()) + 1.0
    mid = 0.5 * (lo + hi)

    c = np.correlate(h, 1.0 - h, mode='full')

        return None
    return float(c.max() * dx)


def _grad_mat(h, n, dx):
    G = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        ell = k - (n - 1)
        for j in range(n):
            v = 0.0
            jm = j - ell
            jp = j + ell
            if 0 <= jm < n:
                v += 1.0 - h[jm]
            if 0 <= jp < n:
                v -= h[jp]
            G[k, j] = dx * v
    return G
