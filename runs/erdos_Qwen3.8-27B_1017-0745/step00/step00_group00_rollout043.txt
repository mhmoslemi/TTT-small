        cands.append(parent + rng.normal(0, 0.10, n))
        cands.append(make_cosine_fallback(n, rng))
        cands.append(cosine(n, 1, 0.37))
    else:
        cands.append(make_cosine_fallback(n, rng))
    for c in cands:
        pp = project_box_sum(c, S)
        if pp is not None and is_feasible(pp, n):
            if all(np.max(np.abs(pp - s)) > 1e-12 for s in starts):
                starts.append(pp)
    for s in starts[1:]:
        if calls['c'] >= lim:
            break
        sc = exact_score(s, n)
        if sc is not None:
            calls['c'] += 1
            if sc < best_s:
                best_s, best_h = sc, s.copy()

    cl = len(starts)
    for ci in range(len(starts)):
        if calls['c'] >= lim:
            break
        rem = lim - calls['c']
        quota = max(10, rem // max(1, cl))
        cl -= 1
        hc = starts[ci].copy()
        sc = exact_score(hc, n)
        if sc is None:
            hc, sc = fallback.copy(), fs
        else:
            calls['c'] += 1
            if sc < best_s:
                best_s, best_h = sc, hc.copy()

        pl = min(24, max(4, quota // 8)) if n <= 100 else 4
        pd = []
        hb, sb = hc.copy(), sc
        for _ in range(4 * pl):
            if len(pd) >= pl or calls['c'] >= lim:
                break
            trial, _ = do_move(hc, 0.35, rng, n, S)
            if trial is None:
                continue
            st = exact_score(trial, n)
            if st is None:
                continue
            calls['c'] += 1
            pd.append(st - sc)
            if st < sb:
                sb, hb = st, trial.copy()
            if st < best_s:
                best_s, best_h = st, trial.copy()
        if pd:
            da = np.array(pd)
            pos = da[da > 0]
            dr = np.median(pos) if len(pos) else (np.median(np.abs(da[da != 0])) if np.any(da != 0) else 0.001)
            T0 = max(1e-5, min(0.05, dr / math.log(2.0))) if np.isfinite(dr) and dr > 0 else 0.002
        else:
            T0 = 0.002
        if sb < sc:
            hc, sc = hb.copy(), sb

        cap = 300 if n <= 100 else (80 if n <= 300 else 30)
        tsel = max(0, quota - 2 * pl)
        r = 0.35
        for j in range(6):
            Tj = T0 * (0.5 ** j)
            sb2 = min(cap, (tsel + 5 - j) // (6 - j))
            if sb2 <= 0:
                break
            la, lc = 0, 0
            for _ in range(sb2):
                if calls['c'] >= lim:
                    break
                trial, is_loc = do_move(hc, r, rng, n, S)
                if trial is None:
                    if is_loc:
                        la += 1
                    continue
                if is_loc:
                    la += 1
                st = exact_score(trial, n)
                if st is None:
                    continue
                calls['c'] += 1
                if st < best_s:
                    best_s, best_h = st, trial.copy()
                d = st - sc
                if d <= 0:
                    hc, sc = trial.copy(), st
                    if is_loc:
                        lc += 1
                else:
                    u = max(rng.random(), np.finfo(np.float64).tiny)
                    if math.log(u) < -d / Tj:
                        hc, sc = trial.copy(), st
                        if is_loc:
                            lc += 1
            if la > 0:
                fr = lc / la
                if fr < 0.15:
                    r = max(0.03, 0.75 * r)
                elif fr > 0.45:
                    r = min(1.2, 1.20 * r)

    fh = project_box_sum(best_h, S)
    if fh is not None and is_feasible(fh, n):
        fs2 = exact_score(fh, n)
        if fs2 is not None:
            best_h, best_s = fh.copy(), fs2
    if not is_feasible(best_h, n):
        best_h = fallback.copy()
        best_s = fs
    sr = exact_score(best_h, n)
    if sr is None:
        best_h = fallback.copy()
        sr = exact_score(best_h, n)
        if sr is None:
            best_h = np.full(n, 0.5); sr = 0.5
    return best_h, float(sr), n

def make_cosine_fallback(n, rng):
    return rng.uniform(0.0, 1.0, n)

def do_move(h, r, rng, n, S):
    d = rand_dir(n, rng)
    if d is None:
        return None, False
    iv = chord(h, d)
    if iv is None:
        return None, False
    lo, hi = iv
    if hi - lo < 1e-12:
        return None, False
    is_loc = rng.random() >= 0.25
    if is_loc:
        alpha = r * (2.0 * rng.random() - 1.0)
        if alpha < lo or alpha > hi:
            return None, True
    else:
        alpha = lo + rng.random() * (hi - lo)
    trial = h + alpha * d
    trial = repair(trial, n, S)
    if trial is None:
        return None, is_loc
    return trial, is_loc
