        if abs(r) < 1e-15:
            break
        for i in range(n):
            if abs(r) < 1e-15:
                break
            if r > 0:
                a = min(r, 1.0 - y[i]); y[i] += a; r -= a
            else:
                a = min(-r, y[i]); y[i] -= a; r += a
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(float(np.sum(y)) - target) > 1e-10:
        return None
    return y

def exact_score(h, dx):
    if h.ndim != 1 or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def norm_dir(d):
    d = d.astype(np.float64).copy()
    if len(d) == 0 or not np.all(np.isfinite(d)):
        return None
    d -= np.mean(d)
    n = len(d)
    d[-1] -= np.sum(d)
    m = float(np.max(np.abs(d)))
    if m < 1e-15:
        return None
    d /= m
    d[-1] -= np.sum(d)
    return d

def alpha_interval(h, d):
    al, ah = -1e10, 1e10
    for i in range(len(h)):
        if abs(d[i]) < 1e-15:
            continue
        if d[i] > 0:
            al = max(al, -h[i] / d[i])
            ah = min(ah, (1.0 - h[i]) / d[i])
        else:
            al = max(al, (1.0 - h[i]) / d[i])
            ah = min(ah, -h[i] / d[i])
    if not (np.isfinite(al) and np.isfinite(ah)):
        return None
    if ah - al < -1e-12:
        return None
    if al > 1e-12 or ah < -1e-12:
        return None
    al = max(al, -1e-12); ah = min(ah, 1e-12)
    if ah < al:
        al = ah = 0.0
    return al, ah

def solve_quad(A, B, C, lo, hi):
    tol = 1e-14 * max(1.0, abs(A), abs(B), abs(C))
    tep = 1e-12 * max(1.0, abs(lo), abs(hi))
    roots = []
    if abs(A) < tol:
        if abs(B) >= tol:
            r = -C / B
            if lo - tep <= r <= hi + tep:
                roots.append(r)
    else:
        disc = B * B - 4.0 * A * C
        if disc < -tol * 10:
            return roots
        if disc < 0:
            disc = 0.0
        sq = sqrt(disc)
        sB = 1.0 if B >= 0 else -1.0
        q = -0.5 * (B + sB * sq)
        if abs(q) > 1e-30:
            for r in [q / A, C / q]:
                if np.isfinite(r) and lo - tep <= r <= hi + tep:
                    roots.append(r)
        else:
            r = -B / (2.0 * A)
            if np.isfinite(r) and lo - tep <= r <= hi + tep:
                roots.append(r)
    return roots

def do_line_search(h, d, dx, n):
    m = 2 * n - 1
    a = np.correlate(h, 1.0 - h, mode='full')
    b = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    c = -np.correlate(d, d, mode='full')
    if not (np.all(np.isfinite(a)) and np.all(np.isfinite(b)) and np.all(np.isfinite(c))):
        return 0.0
    iv = alpha_interval(h, d)
    if iv is None:
        return 0.0
    al, ah = iv
    if ah <= al:
        return 0.0
    cands = [al, ah, 0.0]
    if n <= 96:
        for r in range(m):
            for s in range(r + 1, m):
                cands.extend(solve_quad(c[s]-c[r], b[s]-b[r], a[s]-a[r], al, ah))
            if abs(c[r]) > 1e-14:
                sp = -b[r] / (2.0 * c[r])
                if np.isfinite(sp) and al <= sp <= ah:
                    cands.append(sp)
    else:
        cands.extend(np.linspace(al, ah, 65).tolist())
        ti = np.argsort(a)[-24:]
        for r in ti:
            if abs(c[r]) > 1e-14:
                sp = -b[r] / (2.0 * c[r])
                if np.isfinite(sp) and al <= sp <= ah:
                    cands.append(sp)
        for i in range(len(ti)):
            for j in range(i+1, len(ti)):
                cands.extend(solve_quad(c[ti[j]]-c[ti[i]], b[ti[j]]-b[ti[i]], a[ti[j]]-a[ti[i]], al, ah))
    cands = np.array(cands)
    cands = cands[np.isfinite(cands)]
    if len(cands) == 0:
        return 0.0
    cands.sort()
    merged = [cands[0]]
    for v in cands[1:]:
        if abs(v - merged[-1]) > 1e-12 * max(1.0, abs(v)):
            merged.append(v)
    best_val, best_a = float('inf'), 0.0
    for st in range(0, len(merged), 512):
        batch = np.array(merged[st:st+512])
        vals = a[:, None] + b[:, None]*batch[None,:] + c[:, None]*batch[None,:]**2
        env = np.max(vals, axis=0)
        for i in range(len(env)):
            if env[i] < best_val - 1e-12:
                best_val, best_a = env[i], batch[i]
            elif abs(env[i] - best_val) <= 1e-12 and abs(batch[i]) < abs(best_a):
                best_a = batch[i]
    return float(best_a)

def build_dirs(h, n, rng):
    dirs = []
    q = np.correlate(h, 1.0 - h, mode='full')
    for t in np.argsort(q)[-3:]:
        lag = int(t) - (n - 1)
        g = np.zeros(n)
        for j in range(n):
            if 0 <= j - lag < n:
                g[j] += (1.0 - h[j - lag])
            if 0 <= j + lag < n:
                g[j] -= h[j + lag]
        d = norm_dir(-g)
        if d is not None:
            dirs.append(d)
    for k in range(1, min(7, n)):
        d = np.cos(np.pi * k * (np.arange(n) + 0.5) / n)
        d = norm_dir(d)
        if d is not None:
            dirs.append(d)
    for r in range(6):
        ai = int(floor(r * n / 6.0))
        bi = (ai + 1) % n if r % 2 == 0 else (ai + max(1, n//2)) % n
        if bi == ai:
            bi = (ai + 1) % n
        d = np.zeros(n); d[ai] = 1.0; d[bi] = -1.0
        d = norm_dir(d)
        if d is not None:
            dirs.append(d)
    for _ in range(4):
        idx = rng.choice(n, size=2, replace=False)
        d = np.zeros(n); d[idx[0]] = 1.0; d[idx[1]] = -1.0
        d = norm_dir(d)
        if d is not None:
            dirs.append(d)
    for _ in range(2):
        d = norm_dir(rng.standard_normal(n))
        if d is not None:
            dirs.append(d)
    uniq = []
    for d in dirs:
        if not any(np.max(np.abs(d - u)) < 1e-10 for u in uniq):
            uniq.append(d)
    return uniq

def local_search(start, dx, rng, n):
    h = start.astype(np.float64).copy()
    cur_sc = exact_score(h, dx)
    if cur_sc >= float('inf'):
        return h, cur_sc
    best, best_sc = h.copy(), cur_sc
    nsw = 2 if n <= 96 else 1
    for _ in range(nsw):
        if cur_sc <= 0.3808:
            break
        for d in build_dirs(h, n, rng):
            a = do_line_search(h, d, dx, n)
            if abs(a) > 1e-14:
                proj = capped_simplex_project(h + a * d, float(n)/2.0)
                if proj is not None:
                    ns = exact_score(proj, dx)
                    if ns < cur_sc - 1e-12:
                        h, cur_sc = proj, ns
                        if cur_sc < best_sc:
                            best, best_sc = h.copy(), cur_sc
    return best, best_sc

def run(seed=42, **kwargs):
    n, dx, target = 70, 2.0/70, 35.0
    parent = None
    try:
        iv = np.array(initial_h_values, dtype=np.float64).ravel()
        if 2 <= len(iv) <= 999 and np.all(np.isfinite(iv)) and np.max(np.abs(iv)) < 1e6:
            n = len(iv); dx = 2.0/n; target = float(n)/2.0
            p = capped_simplex_project(iv, target)
            if p is not None:
                parent = p
    except Exception:
        pass
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    u = np.full(n, 0.5)
    blk = np.zeros(n); blk[:n//2] = 1.0
    if n % 2 == 1:
        blk[n//2] = 0.5
    blk = capped_simplex_project(blk, target) or u.copy()
    starts = [u.copy(), blk.copy()]
    if parent is not None:
        starts.insert(0, parent)
        starts.append(parent[::-1].copy())
        for sig in [0.04, 0.12]:
            p = capped_simplex_project(parent + sig * rng.standard_normal(n), target)
            if p is not None:
                starts.append(p)
    for _ in range(2):
        p = capped_simplex_project(rng.uniform(0, 1, n), target)
        if p is not None:
            starts.append(p)
    p = capped_simplex_project(rng.beta(0.5, 0.5, n), target)
    if p is not None:
        starts.append(p)
    scored = []
    for s in starts:
        sc = exact_score(s, dx)
        if sc < float('inf'):
            scored.append((sc, s))
    scored.sort(key=lambda x: x[0])
    sel = []
    if parent is not None:
        for i, (sc, s) in enumerate(scored):
            if np.max(np.abs(s - parent)) < 1e-10:
                sel.append(s); break
    for sc, s in scored:
        if len(sel) >= 3:
            break
        if not any(np.max(np.abs(s - x)) < 1e-10 for x in sel):
            sel.append(s)
    inc = parent if parent is not None else u.copy()
    inc_sc = exact_score(inc, dx)
    for s in sel:
        res, sc = local_search(s, dx, rng, n)
        if sc < inc_sc - 1e-12:
            inc, inc_sc = res, sc
        if inc_sc <= 0.3808:
            break
    nrest = 2 if n <= 96 else 1
    for ri in range(nrest):
        sig = 0.02 if ri == 0 else 0.08
        p = capped_simplex_project(inc + sig * rng.standard_normal(n), target)
        if p is not None:
            res, sc = local_search(p, dx, rng, n)
            if sc < inc_sc - 1e-12:
                inc, inc_sc = res, sc
        if inc_sc <= 0.3808:
            break
    final = capped_simplex_project(inc, target)
    if final is None:
        final = parent if parent is not None else u.copy()
    fs = exact_score(final, dx)
    if fs >= float('inf'):
        final, fs = u.copy(), exact_score(u, dx)
    return final, fs, n
