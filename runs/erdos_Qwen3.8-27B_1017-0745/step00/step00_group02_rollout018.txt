import numpy as np
import math

def project_onto_k(y, n):
    S = n / 2.0
    if not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        c = np.clip(y - mid, 0.0, 1.0)
        if c.sum() > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(30):
        diff = S - c.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = c < 1.0 - 1e-12
        else:
            mask = c > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        c[mask] += diff / cnt
        np.clip(c, 0.0, 1.0, out=c)
    return c

def exact_score(h):
    n = len(h)
    if not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(h.sum() - n / 2.0) <= 1e-10)

def lag_scores(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_value(h, tau):
    qs = lag_scores(h)
    m = qs.max()
    w = np.exp((qs - m) / tau)
    return float(m + tau * np.log(w.sum()))

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    qs = lag_scores(h)
    m = qs.max()
    w = np.exp((qs - m) / tau)
    p = w / w.sum()
    g = np.zeros(n)
    for l in range(2 * n - 1):
        if p[l] < 1e-15:
            continue
        k = l - (n - 1)
        lo1, hi1 = max(0, k), min(n, n + k)
        if hi1 > lo1:
            g[lo1:hi1] += p[l] * dx * (1.0 - h[lo1 - k:hi1 - k])
        lo2, hi2 = max(0, -k), min(n, n - k)
        if hi2 > lo2:
            g[lo2:hi2] -= p[l] * dx * h[lo2 + k:hi2 + k]
    return g

def proj_grad_descent(h0, tau_list):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_s = exact_score(h0)
    for tau in tau_list:
        alpha = 1.0
        consec = 0
        for _ in range(80):
            F = smooth_value(h, tau)
            g = smooth_grad(h, tau)
            if not np.all(np.isfinite(g)):
                break
            accepted = False
            for _hs in range(25):
                y = project_onto_k(h - alpha * g, n)
                d = y - h
                if np.max(np.abs(d)) <= 1e-10:
                    break
                gd = g.dot(d)
                if gd > -1e-12:
                    break
                Fy = smooth_value(y, tau)
                if Fy <= F + 1e-4 * gd + 1e-14:
                    dec = F - Fy
                    h = y
                    s = exact_score(h)
                    if s < best_s - 1e-12:
                        best_s = s
                        best_h = h.copy()
                    consec = consec + 1 if dec < 1e-12 else 0
                    alpha = min(1.25 * alpha, 5.0)
                    accepted = True
                    break
                alpha *= 0.5
            if not accepted:
                break
            if consec >= 10:
                break
    return best_h, best_s

def slsqp_obj(x):
    return x[-1]

def slsqp_obj_jac(x):
    n = len(x) - 1
    g = np.zeros(len(x))
    g[-1] = 1.0
    return g

def slsqp_ineq(x):
    n = len(x) - 1
    h = x[:n]
    return x[-1] - lag_scores(h)

def slsqp_ineq_jac(x):
    n = len(x) - 1
    h = x[:n]
    dx = 2.0 / n
    jac = np.zeros((2 * n - 1, n + 1))
    jac[:, -1] = 1.0
    for l in range(2 * n - 1):
        k = l - (n - 1)
        lo1, hi1 = max(0, k), min(n, n + k)
        if hi1 > lo1:
            jac[l, lo1:hi1] -= dx * (1.0 - h[lo1 - k:hi1 - k])
        lo2, hi2 = max(0, -k), min(n, n - k)
        if hi2 > lo2:
            jac[l, lo2:hi2] += dx * h[lo2 + k:hi2 + k]
    return jac

def slsqp_eq(x):
    n = len(x) - 1
    return x[:n].sum() - n / 2.0

def slsqp_eq_jac(x):
    n = len(x) - 1
    g = np.zeros(len(x))
    g[:n] = 1.0
    return g

def slsqp_refine(h0, n):
    dx = 2.0 / n
    S = n / 2.0
    s0 = exact_score(h0)
    x0 = np.concatenate([h0, [min(1.0, s0 + 1e-8)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    con = [
        {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    try:
        from scipy.optimize import minimize
        res = minimize(slsqp_obj, x0, method='SLSQP', jac=slsqp_obj_jac,
                       bounds=bounds, constraints=con,
                       options={'maxiter': 300, 'ftol': 1e-12, 'disp': False})
        hx = res.x[:n]
        if not np.all(np.isfinite(hx)):
            return h0, s0
        np.clip(hx, 0.0, 1.0, out=hx)
        hx = project_onto_k(hx, n)
        if not check_feasible(hx, n):
            return h0, s0
        s = exact_score(hx)
        if s < s0 - 1e-12:
            return hx, s
        return h0, s0
    except Exception:
        return h0, s0

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = globals().get('initial_h_values', None)
    n = 45
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.ndim == 1 and 2 <= len(p) <= 999 and np.all(np.isfinite(p)) and np.max(np.abs(p)) < 1e6:
            n = len(p)

        else:

    else:

        if isinstance(np_kw, int) and 2 <= np_kw <= 999:
            n = np_kw

    best_h = fallback.copy()
    best_s = exact_score(best_h)
    tau_list = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 1e-5]
    candidates = []

    def try_candidate(hc):
        nonlocal best_h, best_s
        np.clip(hc, 0.0, 1.0, out=hc)
        hc = project_onto_k(hc, n)
        if not check_feasible(hc, n):
            return hc
        s = exact_score(hc)
        if s < best_s - 1e-12:
            best_s = s
            best_h = hc.copy()
        candidates.append((hc.copy(), s))
        return hc

    starts = []
    starts.append(fallback.copy())
    starts.append(1.0 - fallback)
    starts.append(fallback[::-1].copy())
    starts.append(1.0 - fallback[::-1].copy())
    starts.append(fallback + 0.08 * rng.standard_normal(n))
    starts.append(fallback + 0.25 * rng.standard_normal(n))
    starts.append(rng.uniform(0, 1, n))
    starts.append(rng.beta(0.5, 0.5, n))

    local_results = []
    for st in starts:
        hc = try_candidate(st)
        bh, bs = proj_grad_descent(hc, tau_list)
        bh = try_candidate(bh)
        local_results.append((bh.copy(), exact_score(bh)))

    for sig in [0.04, 0.12, 0.30, 0.50]:
        pert = best_h + sig * rng.standard_normal(n)
        hc = try_candidate(pert)
        bh, bs = proj_grad_descent(hc, tau_list)
        bh = try_candidate(bh)
        local_results.append((bh.copy(), exact_score(bh)))

    if n <= 100:
        local_results.sort(key=lambda x: x[1])

            if len(refined) >= 3:
                break

        final_h = fallback.copy()
    final_s = exact_score(final_h)
    fb_s = exact_score(fallback)
    if final_s > fb_s + 1e-12:
        final_h = fallback.copy()
        final_s = fb_s
    final_h = project_onto_k(final_h, n)
    final_s = exact_score(final_h)
    return final_h.astype(np.float64), float(final_s), int(n)
