                sc = exact_score(h_new)
                candidates.append((h_new.copy(), sc))
                if sc < incumbent_score - 1e-12:
                    incumbent = h_new.copy()
                    incumbent_score = sc
                h_cur = h_new

    candidates.sort(key=lambda x: x[1])
    top3 = []
    for h_c, sc in candidates:
        if not any(np.max(np.abs(h_c - t[0])) < 1e-6 for t in top3):
            top3.append((h_c, sc))
        if len(top3) >= 3:
            break

    if N <= 999 and top3:
        mi_epi = min(500, max(240, 2 * maxiter))
        bounds_epi = bounds + [(0.0, 2.1)]
        cons_e = [
            {'type': 'ineq', 'fun': epi_ineq, 'jac': epi_ineq_jac},
            {'type': 'eq', 'fun': mass_eq_ext, 'jac': mass_eq_ext_jac}
        ]
        for h_c, sc in top3:
            z0 = np.concatenate([h_c, [sc + 1e-6]])
            try:
                res = minimize(epi_obj, z0, jac=epi_obj_jac,
                               bounds=bounds_epi, constraints=cons_e,
                               method='SLSQP', ftol=1e-10, disp=False,
                               options={'maxiter': mi_epi})
                h_new = project_feasible(res.x[:-1], N, m)
            except Exception:
                h_new = project_feasible(h_c, N, m)
            sc_new = exact_score(h_new)
            if sc_new < incumbent_score - 1e-12:
                incumbent = h_new.copy()
                incumbent_score = sc_new

    incumbent = project_feasible(incumbent, N, m)
    final_score = exact_score(incumbent)
    if not np.all(incumbent >= -1e-12) or not np.all(incumbent <= 1 + 1e-12) or abs(np.sum(incumbent) - m) > 1e-9 * max(1, N):
        incumbent = np.full(N, 0.5, dtype=np.float64)
        final_score = exact_score(incumbent)
    return (incumbent, final_score, N)
