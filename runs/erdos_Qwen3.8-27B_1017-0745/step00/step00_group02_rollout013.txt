            p = np.array(par, dtype=np.float64).ravel()
            if p.ndim == 1 and 2 <= len(p) <= 999 and np.all(np.isfinite(p)) and np.abs(p).max() <= 1e6:
                n = len(p)
                if np.all(p >= 0) and np.all(p <= 1) and abs(p.sum() - n / 2.0) <= 1e-12:
                    fb = p.copy()
                else:
                    pr = _project(p, n)
                    fb = pr if pr is not None else np.full(n, 0.5, dtype=np.float64)
        except Exception:
            fb = None

    if fb is None:
        kn = kwargs.get('n_points', None)
        if isinstance(kn, int) and not isinstance(kn, bool) and 2 <= kn <= 999:
            n = kn
        fb = np.full(n, 0.5, dtype=np.float64)

    base = fb.copy() if par is not None else None

    if n <= 64:
        P, G, st, mx = max(48, min(160, 3 * n)), 120, 24, 2
    elif n <= 256:
        P, G, st, mx = 64, 40, 12, 1
    else:
        P, G, st, mx = 32, 20, 8, 1

    pop = np.zeros((P, n), dtype=np.float64)
    sc = np.full(P, 1e30, dtype=np.float64)
    Fv = np.full(P, 0.6, dtype=np.float64)
    Cv = np.full(P, 0.5, dtype=np.float64)

    pop[0] = fb
    sc[0] = _score(fb, n)
    Fv[0] = rng.uniform(0.4, 0.9)
    Cv[0] = rng.uniform(0.25, 0.95)

    sl = 1
    if base is not None:
        for v in (base[::-1].copy(), 1.0 - base, (1.0 - base)[::-1].copy()):
            if sl >= P:
                break
            pop[sl] = v
            sc[sl] = _score(v, n)
            Fv[sl] = rng.uniform(0.4, 0.9)
            Cv[sl] = rng.uniform(0.25, 0.95)
            sl += 1

    scs = [0.03, 0.08, 0.18, 0.35, 0.60]
    mi = 0
    while sl < P:
        bd = base if base is not None else np.full(n, 0.5, dtype=np.float64)
        c = _gen_cand(rng, n, bd, mi % 8, scs[mi % 5])
        pop[sl] = c
        sc[sl] = _score(c, n)
        Fv[sl] = rng.uniform(0.4, 0.9)
        Cv[sl] = rng.uniform(0.25, 0.95)
        sl += 1
        mi += 1

    arch = fb.copy()
    asc = sc[0]
    stag = 0
    nres = 0

    for _g in range(G):
        fp = pop.copy()
        fs = sc.copy()
        npop = pop.copy()
        nsc = sc.copy()
        nF = Fv.copy()
        nC = Cv.copy()

        ord_ = np.argsort(fs)
        pb = max(2, math.ceil(0.2 * P))
        ppool = ord_[:pb]

        improved = False
        for i in range(P):
            cp = [j for j in ppool if j != i]
            if not cp:
                cp = list(ppool)
            p = cp[int(rng.integers(len(cp)))]
            av = [j for j in range(P) if j != i and j != p]
            r1 = av[int(rng.integers(len(av)))]
            av2 = [j for j in av if j != r1]
            r2 = av2[int(rng.integers(len(av2)))]

            F = Fv[i] if rng.random() > 0.1 else rng.uniform(0.3, 0.9)
            Cr = Cv[i] if rng.random() > 0.1 else rng.uniform(0.1, 0.95)

            v = fp[i] + F * (fp[p] - fp[i]) + F * (fp[r1] - fp[r2])
            mk = rng.random(n) < Cr
            mk[int(rng.integers(n))] = True
            w = np.where(mk, v, fp[i])

            tr = _project(w, n)
            if tr is None:
                continue
            ts = _score(tr, n)
            if not math.isfinite(ts):
                continue
            if ts < fs[i] - 1e-12:
                npop[i] = tr
                nsc[i] = ts
                nF[i] = F
                nC[i] = Cr
                if ts < asc - 1e-12:
                    arch = tr.copy()
                    asc = ts
                    improved = True

        pop, sc, Fv, Cv = npop, nsc, nF, nC

        in_pop = False
        for i in range(P):
            if np.max(np.abs(pop[i] - arch)) < 1e-12:
                in_pop = True
                break
        if not in_pop:
            wi = int(np.argmax(sc))
            pop[wi] = arch
            sc[wi] = asc
            Fv[wi] = 0.6
            Cv[wi] = 0.5

        if improved:
            stag = 0
        else:
            stag += 1

        if stag >= st and nres < mx:
            nres += 1
            stag = 0
            half = math.ceil(P // 2)
            sigma = 0.06 if nres == 1 else 0.18
            for j in range(half, P):
                if (j - half) % 2 == 0:
                    c = _gen_cand(rng, n, arch, 0, sigma)
                else:
                    c = _gen_cand(rng, n, arch, (j + 1) % 8, scs[j % 5])
                pop[j] = c
                sc[j] = _score(c, n)
                Fv[j] = rng.uniform(0.4, 0.9)
                Cv[j] = rng.uniform(0.25, 0.95)

    best = arch.copy()
    bs = _score(best, n)
    if not _feasible(best, n) or not math.isfinite(bs):
        pr = _project(best, n)
        if pr is not None:
            best = pr
            bs = _score(best, n)
    fbs = _score(fb, n)
    if not _feasible(best, n) or not math.isfinite(bs) or not (bs < fbs - 1e-12):
        best = fb.copy()
        bs = _score(best, n)
    if not _feasible(best, n) or not math.isfinite(bs):
        best = np.full(n, 0.5, dtype=np.float64)
        bs = _score(best, n)

    bs = _score(best, n)
    return best.copy(), bs, int(n)
