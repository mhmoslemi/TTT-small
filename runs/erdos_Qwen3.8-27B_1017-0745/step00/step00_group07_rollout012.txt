        if pred <= 1e-12 * max(1.0, abs(score_x)):
            break
        actual = score_x - score_y
        rho = actual / pred
        if actual > 1e-12 * max(1.0, abs(score_x)) and rho >= 0.1:
            ss = np.max(np.abs(y - x))
            x = y.copy()
            score_x = score_y
            fails = 0
            if rho >= 0.75 and ss >= 0.8 * Delta:
                Delta = min(2.0 * Delta, D_max)
            elif rho < 0.25:
                Delta = max(0.25 * Delta, D_min)
            consec = consec + 1 if actual < 1e-10 * max(1.0, abs(score_x)) else 0
            if consec >= 8:
                break
        else:
            Delta = max(0.25 * Delta, D_min)
    return best_x, best_c


def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = globals().get('initial_h_values', None)
    parent_arr = None
    n = None
    if parent is not None:
        try:
            p = np.array(parent, dtype=np.float64).ravel()
            if len(p) >= 2 and np.all(np.isfinite(p)):
                n = len(p)
                S = n / 2.0
                parent_arr = project_box_sum(p, S, np.zeros(n), np.ones(n))
                if parent_arr is None:
                    parent_arr = np.full(n, 0.5)
        except Exception:
            pass
    if parent_arr is None:
        n = kwargs.get('n_points', 80)
        if not isinstance(n, (int, np.integer)) or n < 8 or n > 512:
            n = 80
        S = n / 2.0
        parent_arr = np.full(n, 0.5)
    dx = 2.0 / n
    best_h = parent_arr.copy()
    best_c = exact_score(best_h)
    if n > 512:
        return best_h, best_c, int(n)
    max_it = 36 if n <= 128 else (24 if n <= 256 else 12)
    K = max(1, min(6, (n - 1) // 2))
    starts = []
    add_start(starts, parent_arr.copy(), S)
    add_start(starts, parent_arr[::-1].copy(), S)
    zeros = np.zeros(n)
    ones = np.ones(n)
    for _ in range(2):
        q = int(rng.integers(1, K + 1))
        ph = float(rng.uniform(0, 2 * math.pi))
        z = 0.5 + 0.45 * np.cos(2 * math.pi * q * np.arange(n) / n + ph)
        add_start(starts, project_box_sum(z, S, zeros, ones), S)
    noise = rng.standard_normal(n)
    freq = rfft(noise)
    kk = np.arange(len(freq))
    cutoff = max(1.0, K / 2.0)
    freq = freq * np.exp(-0.5 * (kk / cutoff) ** 2)
    sm = irfft(freq, n=n)
    sd = np.std(sm)
    if sd > 1e-12:
        sm = sm / sd
    add_start(starts, project_box_sum(0.5 + 0.35 * sm, S, zeros, ones), S)
    perm = rng.permutation(n)
    z2 = np.zeros(n)
    z2[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        z2[perm[n // 2]] = 0.5
    add_start(starts, project_box_sum(z2, S, zeros, ones), S)
    cap = 6 if n <= 128 else (4 if n <= 256 else 2)
    starts = starts[:cap]
    for s in starts:
        try:
            h_opt, c_opt = slp_search(s, S, max_it, dx)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
        except Exception:
            continue
    if not is_feasible(best_h, S):
        best_h = project_box_sum(best_h, S, np.zeros(n), np.ones(n))
        if best_h is None:
            best_h = np.full(n, 0.5)
    best_c = exact_score(best_h)
    return best_h, best_c, int(n)
