import numpy as np

def project_K(y, n, S):
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        c = np.clip(y - mid, 0.0, 1.0)
        if float(np.sum(c)) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    res = S - float(np.sum(h))
    if abs(res) > 1e-13:
        idx = np.where((h < 1.0 - 1e-14) if res > 0 else (h > 1e-14))[0]
        if len(idx) > 0:
            h[idx] += res / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def score(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def feasible(h, n, S):
    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-9) and np.all(h <= 1.0 + 1e-9) and
            abs(float(np.sum(h)) - S) < 1e-7)

def svalue(q, tau):
    m = float(np.max(q))
    w = np.exp((q - m) / tau)
    return m + tau * float(np.log(float(np.sum(w))))

def sgrad(h, q, tau, dx):
    n = len(h)
    m = float(np.max(q))
    w = np.exp((q - m) / tau)
    p = w / float(np.sum(w))
    g = np.zeros(n)
    for k in range(len(q)):
        ell = k - (n - 1)
        lo = max(0, -ell)
        hi = min(n, n - ell)
        if lo >= hi:
            continue
        g[lo + ell:hi + ell] += p[k] * dx * (1.0 - h[lo:hi])
        g[lo:hi] -= p[k] * dx * h[lo + ell:hi + ell]
    return g

class Slqp:
    def __init__(self, n, S, dx):
        self.n = n
        self.S = S
        self.dx = dx
    def f(self, x):
        return x[self.n]
    def fj(self, x):
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        return g
    def cin(self, x):
        h = x[:self.n]
        return x[self.n] - np.correlate(h, 1.0 - h, mode='full') * self.dx
    def cinj(self, x):
        n = self.n
        h = x[:n]
        jc = np.zeros((2 * n - 1, n + 1))
        for k in range(2 * n - 1):
            ell = k - (n - 1)
            lo = max(0, -ell)
            hi = min(n, n - ell)
            if lo < hi:
                jc[k, lo + ell:hi + ell] = -self.dx * (1.0 - h[lo:hi])
                jc[k, lo:hi] = -self.dx * h[lo + ell:hi + ell]
            jc[k, n] = 1.0
        return jc
    def ceq(self, x):
        return float(np.sum(x[:self.n])) - self.S
    def ceqj(self, x):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g

def continuation(h, n, S, dx, taus, maxit):
    h = h.copy()
    for tau in taus:
        q = np.correlate(h, 1.0 - h, mode='full') * dx
        for _ in range(maxit):
            g = sgrad(h, q, tau, dx)
            if not np.all(np.isfinite(g)):
                break
            alpha = 1.0
            accepted = False
            for _ in range(25):
                y = project_K(h - alpha * g, n, S)
                d = y - h
                if float(np.max(np.abs(d))) < 1e-12:
                    break
                gd = float(np.dot(g, d))
                if gd > -1e-14:
                    break
                qy = np.correlate(y, 1.0 - y, mode='full') * dx
                if svalue(qy, tau) <= svalue(q, tau) + 1e-4 * gd + 1e-14:
                    h = y
                    q = qy
                    accepted = True
                    break
                alpha *= 0.5
            if not accepted:
                break
    return h
