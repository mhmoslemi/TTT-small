            break

            b = h[0:L]
            u = d[k:k + L]

        B[idx] = np.sum(u * (1.0 - b) - a * v)

def make_directions(h, N, rng, stage):
    if N <= 80:
        counts = (8, 6, 4, 2) if stage == 'main' else (4, 3, 2, 1)
    elif N <= 250:
        counts = (6, 4, 2, 1) if stage == 'main' else (3, 2, 1, 1)
    else:
        counts = (3, 2, 1, 1) if stage == 'main' else (1, 1, 1, 0)
    dirs = []
    n_star, n_rand, n_block, n_harm = counts
    # Star pairs
    anchor = int(np.argmax(np.minimum(h, 1.0 - h)))
    others = [i for i in range(N) if i != anchor]
    for t in range(n_star):
        j = others[int(t * len(others) / n_star)]
        d = np.zeros(N)
        d[j] = 1.0
        d[anchor] = -1.0
        dirs.append(d)
    # Random pairs
    used = set()
    for _ in range(n_rand):
        for _attempt in range(20):
            i, j = sorted(rng.choice(N, size=2, replace=False))
            if (i, j) not in used:
                used.add((i, j))
                break
        d = np.zeros(N)
        d[i] = 1.0
        d[j] = -1.0
        dirs.append(d)
    # Sparse blocks
    for _ in range(n_block):
        sz = int(rng.integers(3, min(13, N+1)))
        support = rng.choice(N, size=sz, replace=False)
        d = np.zeros(N)
        vals = rng.standard_normal(sz)
        vals -= vals.mean()
        vals[sz-1] -= vals.sum()  # ensure zero sum
        d[support] = vals
        if np.max(np.abs(d)) > 1e-15:
            d /= np.max(np.abs(d))
            dirs.append(d)
    # Harmonics
    for r in range(1, min(n_harm, (N-1)//2) + 1):
        phase = rng.uniform(0, 2*math.pi)
        i = np.arange(N)
        d = np.cos(2*math.pi*r*i/N + phase)
        d -= d.mean()
        d[int(rng.integers(N))] -= d.sum()
        if np.max(np.abs(d)) > 1e-15:
            d /= np.max(np.abs(d))
            dirs.append(d)
    return dirs

    if a_hi - a_lo < 1e-14:
        return h, cs
    A, B, C = lag_coefficients(h, d, N)
    if not (np.all(np.isfinite(A)) and np.all(np.isfinite(B)) and np.all(np.isfinite(C))):
        return h, cs
    M = 2 * N - 1
    cands = [a_lo, a_hi, 0.0]
    for j in range(M):
        if abs(C[j]) > 1e-15:
            sp = -B[j] / (2.0 * C[j])
            if a_lo - 1e-10 <= sp <= a_hi + 1e-10:
                cands.append(sp)
    if N <= 80:
        for j in range(M):
            for l2 in range(j + 1, M):
                u, v, w = C[j] - C[l2], B[j] - B[l2], A[j] - A[l2]
                sc = max(abs(u), abs(v), abs(w), 1e-300)
                if abs(u) <= 1e-12 * sc:
                    if abs(v) > 1e-300 * sc:
                        r = -w / v
                        if a_lo - 1e-10 <= r <= a_hi + 1e-10:
                            cands.append(r)
                else:
                    D = v * v - 4.0 * u * w
                    if D >= -1e-12 * (v * v + abs(4.0 * u * w) + 1e-300):
                        D = max(D, 0.0)
                        q = -0.5 * (v + math.copysign(math.sqrt(D), v))
                        if abs(q) > 1e-300:
                            rr1, rr2 = q / u, w / q
                        else:
                            rr1 = rr2 = -v / (2.0 * u)
                        for rr in (rr1, rr2):
                            if np.isfinite(rr) and a_lo - 1e-10 <= rr <= a_hi + 1e-10:
                                cands.append(rr)
    cands = np.array([max(a_lo, min(a_hi, c)) for c in cands if np.isfinite(c)])
    cands.sort()
    dd = [float(cands[0])]
    for c in cands[1:]:
        if abs(c - dd[-1]) > 1e-12 * max(1.0, abs(c)):
            dd.append(float(c))
    arr = np.array(dd)
    env = A[None, :] + B[None, :] * arr[:, None] + C[None, :] * arr[:, None] ** 2
    pred = env.max(axis=1)
    order = np.argsort(pred)
    sel = set(order[:16].tolist())
    sel.add(0)
    sel.add(len(dd) - 1)
    sel.add(int(np.argmin(np.abs(arr))))
    best_h, best_s = h, cs
    for i in sorted(sel):
        hv = project_box_sum(h + arr[i] * d, N)
        sv = score_exact(hv, N)
        if sv < best_s - 1e-15:
            best_s, best_h = sv, hv
    return best_h, best_s

def make_directions(h, N, rng, stage):
    if N <= 80:
        ct = (8, 6, 4, 2) if stage == 'main' else (4, 3, 2, 1)
    elif N <= 250:
        ct = (6, 4, 2, 1) if stage == 'main' else (3, 2, 1, 1)
    else:
        ct = (3, 2, 1, 1) if stage == 'main' else (1, 1, 1, 0)
    n_st, n_rp, n_bl, n_ha = ct
    dirs = []
    anchor = int(np.argmax(np.minimum(h, 1.0 - h)))
    others = [i for i in range(N) if i != anchor]
    for t in range(n_st):
        j = others[t * len(others) // n_st]
        d = np.zeros(N)
        d[j] = 1.0
        d[anchor] = -1.0
        dirs.append(d)
    used = set()
    for _ in range(n_rp):
        ii, jj = 0, 1
        for _a in range(20):
            ii, jj = sorted(rng.choice(N, size=2, replace=False).tolist())
            if (ii, jj) not in used:
                used.add((ii, jj))
                break
        d = np.zeros(N)
        d[ii] = 1.0
        d[jj] = -1.0
        dirs.append(d)
    for _ in range(n_bl):
        sz = int(rng.integers(3, min(13, N + 1)))
        sup = rng.choice(N, size=sz, replace=False)
        d = np.zeros(N)
        vals = rng.standard_normal(sz)
        vals -= vals.mean()
        d[sup] = vals
        d[sup[0]] -= d.sum()
        if np.max(np.abs(d)) > 1e-15:
            dirs.append(d / np.max(np.abs(d)))
    for r in range(1, min(n_ha, (N - 1) // 2) + 1):
        ph = rng.uniform(0.0, 2.0 * math.pi)
        ii = np.arange(N)
        d = np.cos(2.0 * math.pi * r * ii / N + ph)
        d -= d.mean()
        d[0] -= d.sum()
        if np.max(np.abs(d)) > 1e-15:
            dirs.append(d / np.max(np.abs(d)))
    return dirs

def sort_key(x):
    return x[0]

def run(seed=42, **kwargs):
    try:
        s = int(seed)
    except (ValueError, TypeError):
        s = 42
    N = 70
    baseline = np.full(N, 0.5)
    try:
        ihv = np.asarray(initial_h_values, dtype=np.float64)
        if ihv.ndim == 1 and len(ihv) >= 10:
            N = len(ihv)
            if np.all(np.isfinite(ihv)):
                baseline = project_box_sum(ihv, N)
    except Exception:
        try:
            npv = int(kwargs.get('n_points', 70))
            if 1 <= npv <= 999:
                N = npv
        except (ValueError, TypeError):
            pass
    baseline = project_box_sum(baseline, N)
    best_h = baseline.copy()
    best_s = score_exact(baseline, N)
    if not np.isfinite(best_s):
        best_h = np.full(N, 0.5)
        best_s = score_exact(best_h, N)
    rng = np.random.default_rng(s)
    starts = [baseline.copy(), np.full(N, 0.5)]
    try:
        ihv = np.asarray(initial_h_values, dtype=np.float64)
        if ihv.ndim == 1 and len(ihv) == N and np.all(np.isfinite(ihv)):
            starts.append(project_box_sum(1.0 - ihv, N))
    except Exception:
        pass
    starts.append(project_box_sum(rng.uniform(0.0, 1.0, N), N))
    starts.append(project_box_sum(baseline + 0.2 * rng.standard_normal(N), N))
    for _ in range(2):
        R = min(5, (N - 1) // 2)
        z = np.full(N, 0.5)
        for rr in range(1, R + 1):
            ii = np.arange(N)
            z += (0.12 / rr) * (rng.standard_normal() * np.sin(2.0 * math.pi * rr * ii / N)
                                  + rng.standard_normal() * np.cos(2.0 * math.pi * rr * ii / N))
        starts.append(project_box_sum(z, N))
    us = [starts[0]]
    for st in starts[1:]:
        if all(np.max(np.abs(st - u)) > 1e-8 for u in us):
            us.append(st)
    max_st = 7 if N <= 80 else (5 if N <= 250 else 3)
    us = us[:max_st]
    results = []
    for st in us:
        lh = st.copy()
        ls = score_exact(lh, N)
        if ls < best_s - 1e-15:
            best_h, best_s = lh.copy(), ls
        for d in make_directions(lh, N, rng, 'main'):
            try:
                lh, ls = line_search(lh, d, N, ls)
                if ls < best_s - 1e-15:
                    best_h, best_s = lh.copy(), ls
            except Exception:
                pass
        results.append((ls, lh.copy()))
    results.sort(key=sort_key)
    n_el = 3 if N <= 80 else (2 if N <= 250 else 1)
    for _, el in results[:n_el]:
        lh = el.copy()
        ls = score_exact(lh, N)
        for d in make_directions(lh, N, rng, 'reduced'):
            try:
                lh, ls = line_search(lh, d, N, ls)
                if ls < best_s - 1e-15:
                    best_h, best_s = lh.copy(), ls
            except Exception:
                pass
    scales = [0.08, 0.20] if N <= 80 else ([0.12] if N <= 250 else [])
    for sc in scales:
        pz = project_box_sum(best_h + sc * rng.standard_normal(N), N)
        lh = pz.copy()
        ls = score_exact(lh, N)
        if ls < best_s - 1e-15:
            best_h, best_s = lh.copy(), ls
        for d in make_directions(lh, N, rng, 'reduced'):
            try:
                lh, ls = line_search(lh, d, N, ls)
                if ls < best_s - 1e-15:
                    best_h, best_s = lh.copy(), ls
            except Exception:
                pass
    best_h = project_box_sum(best_h, N)
    fs = score_exact(best_h, N)
    if not np.isfinite(fs):
        best_h = np.full(N, 0.5)
        fs = score_exact(best_h, N)
    return best_h, fs, int(N)
