def slsqp_refine(h0, n, dx, S):
    h = project_box_sum(h0, n, S)
    t0 = min(1.0, exact_score(h, n) + 1e-8)
    x0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * (n + 1)
    sc = _SC(n, dx, S)
    cons = [
        {'type': 'ineq', 'fun': sc.ineq, 'jac': sc.ineq_j},
        {'type': 'eq', 'fun': sc.eq, 'jac': sc.eq_j}
    ]
    obj = _SC_obj(n)
    try:
        res = so.minimize(obj.fun, x0, jac=obj.jac, bounds=bounds, constraints=cons,
                          method='SLSQP', options={'maxiter': 250, 'ftol': 1e-10, 'disp': False})
        h_res = res.x[:n]
        if np.all(np.isfinite(h_res)) and np.all(h_res >= -1e-6) and np.all(h_res <= 1.0 + 1e-6) \
           and abs(h_res.sum() - S) < 1e-4:
            h_res = project_box_sum(h_res, n, S)
            return h_res
    except Exception:
        pass
    return h0

class _SC_obj:
    def __init__(self, n):
        self.n = n
    def fun(self, x):
        return x[self.n]
    def jac(self, x):
        g = np.zeros(self.n + 1)
        g[self.n] = 1.0
        return g

def archive_candidate(cand, n, S, best_h, best_score):
    if cand is None or not np.all(np.isfinite(cand)):
        return best_h, best_score
    cand = np.clip(project_box_sum(cand, n, S), 0.0, 1.0)
    if not is_feasible(cand, n, S):
        return best_h, best_score
    sc = exact_score(cand, n)
    if sc < best_score - 1e-12:
        return cand.copy(), sc
    return best_h, best_score

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = globals().get('initial_h_values', None)
    if parent is not None:
        try:
            parent = np.asarray(parent, dtype=np.float64)
            if parent.ndim == 1 and 2 <= parent.size <= 999 and np.all(np.isfinite(parent)) and np.max(np.abs(parent)) < 1e6:
                n = parent.size
            else:
                n = 45
        except Exception:
            n = 45
    else:
        n = 45
    dx = 2.0 / n
    S = n / 2.0
    if parent is not None:
        try:
            parent = np.asarray(parent, dtype=np.float64)
            if parent.ndim == 1 and parent.size == n:
                best_h = project_box_sum(parent, n, S)
            else:
                best_h = np.full(n, 0.5)
        except Exception:
            best_h = np.full(n, 0.5)
    else:
        best_h = np.full(n, 0.5)
    best_score = exact_score(best_h, n)
    fallback_h = best_h.copy()
    fallback_score = best_score
    tau_sched = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00001]
    starts = [best_h.copy()]
    starts.append(project_box_sum(1.0 - best_h, n, S))
    starts.append(project_box_sum(best_h[::-1].copy(), n, S))
    starts.append(project_box_sum(1.0 - best_h[::-1].copy(), n, S))
    starts.append(project_box_sum(best_h + 0.08 * rng.standard_normal(n), n, S))
    starts.append(project_box_sum(best_h + 0.25 * rng.standard_normal(n), n, S))
    starts.append(project_box_sum(rng.uniform(0, 1, n), n, S))
    starts.append(project_box_sum(rng.beta(0.5, 0.5, n), n, S))
    local_results = []
    for s in starts:
        best_h, best_score = archive_candidate(s, n, S, best_h, best_score)
        res = projected_gradient(s, n, dx, S, tau_sched, rng)
        local_results.append(res)
        best_h, best_score = archive_candidate(res, n, S, best_h, best_score)
    for sigma in [0.04, 0.12, 0.30, 0.50]:
        pert = project_box_sum(best_h + sigma * rng.standard_normal(n), n, S)
        best_h, best_score = archive_candidate(pert, n, S, best_h, best_score)
        res = projected_gradient(pert, n, dx, S, tau_sched, rng)
        local_results.append(res)
        best_h, best_score = archive_candidate(res, n, S, best_h, best_score)
    if n <= 100:
        local_results.sort(key=lambda v: exact_score(v, n))
        unique = []
        for v in local_results:
            if all(np.max(np.abs(v - u)) > 1e-5 for u in unique):
                unique.append(v.copy())
            if len(unique) >= 3:
                break
        if not unique:
            unique = [best_h.copy()]
        for h_init in unique:
            res = slsqp_refine(h_init, n, dx, S)
            best_h, best_score = archive_candidate(res, n, S, best_h, best_score)
    best_h = project_box_sum(best_h, n, S)
    final_score = exact_score(best_h, n)
    if not is_feasible(best_h, n, S) or final_score > fallback_score:
        best_h = fallback_h.copy()
        final_score = fallback_score
    best_h = np.clip(best_h, 0.0, 1.0)
    final_score = exact_score(best_h, n)
    return best_h.astype(np.float64), float(final_score), int(n)
