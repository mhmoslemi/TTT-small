import numpy as np

class IneqCon:
    def __init__(self, n, dx):
        self.n = n
        self.dx = dx

    def fun(self, z):
        h = z[:self.n]
        return z[self.n] - np.correlate(h, 1.0 - h, mode='full') * self.dx

    def jac(self, z):
        h = z[:self.n]
        n, dx = self.n, self.dx
        L = 2 * n - 1
        J = np.zeros((L, n + 1))
        for k in range(L):
            ell = k - (n - 1)
            for j in range(n):
                g = 0.0
                if 0 <= j - ell < n:
                    g += 1.0 - h[j - ell]
                if 0 <= j + ell < n:
                    g -= h[j + ell]
                J[k, j] = -dx * g
            J[k, n] = 1.0
        return J


class EqCon:
    def __init__(self, n, S):
        self.n = n
        self.S = S

    def fun(self, z):
        return np.array([np.sum(z[:self.n]) - self.S])

    def jac(self, z):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g


def obj_fn(z):
    return z[-1]

    g = np.zeros(len(z))

        return None

            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

    if abs(err) > 1e-14:
        mask = h < 1.0 if err > 0 else h > 0.0
        cnt = np.sum(mask)

            h[mask] += err / cnt
    h = np.clip(h, 0.0, 1.0)

    return h

    m = np.max(c)

    sw = np.sum(w)

                g -= h[j + ell]
