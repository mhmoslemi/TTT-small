def exact_score(h):
    n = len(h)
    if not np.isfinite(h).all():
        return float('inf')
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if resid > 0:
        for i in range(n):
            cap = 1.0 - y[i]
            add = min(cap, resid)
            y[i] += add
            resid -= add
            if resid <= 1e-14:
                break
    elif resid < 0:
        for i in range(n):
            sub = min(y[i], -resid)
            y[i] -= sub
            resid += sub
            if resid >= -1e-14:
                break
    if not np.isfinite(y).all():
        return None
    if np.max(np.abs(y)) > 1.0 + 1e-12 or np.min(y) < -1e-12:
        return None
    if abs(np.sum(y) - S) > 1e-10 * max(1, n):
        return None
    return y

def check_feasible(h, n):

        return False

        return False
    S = n / 2.0

        return False
    if abs(np.sum(h) - S) > 1e-10 * max(1, n):
        return False
    return True

def lag_grad(h):
    n = len(h)
    nlags = 2 * n - 1
    Q = np.zeros(nlags)
    G = np.zeros((nlags, n))
    for k in range(-(n-1), n):
        idx = k + n - 1
        lo_i = max(0, k)
        hi_i = min(n-1, n-1+k)
        for i in range(lo_i, hi_i+1):
            j = i - k
            Q[idx] += h[i] * (1.0 - h[j])
            G[idx, i] += (1.0 - h[j])
            G[idx, j] -= h[i]
    return Q, G

def sl_obj(z):
    return z[-1]

def sl_obj_jac(z):
    n = len(z) - 1
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return g

def sl_ineq(z):
    h = z[:-1]
    t = z[-1]
    n = len(h)
    Q, _ = lag_grad(h)
    return t - Q / n

def sl_ineq_jac(z):
    h = z[:-1]
    n = len(h)
    Q, G = lag_grad(h)
    nlags = len(Q)
    J = np.zeros((nlags, n + 1))
    J[:, :n] = -G / n
    J[:, n] = 1.0
    return J

def sl_eq(z):
    h = z[:-1]
    n = len(h)
    return np.sum(h) / n - 0.5

def sl_eq_jac(z):
    n = len(z) - 1
    g = np.zeros(n + 1)
    g[:n] = 1.0 / n
    return g

                        best_score = sc
        except Exception:
            continue

    sigma = 0.05
    for _ in range(5):
        improved = False
        for _attempt in range(4):
            try:
                h_p = best + sigma * rng.standard_normal(n)
                h_p = project_box_sum(h_p, S)
                if h_p is None:
                    continue
                Qp, _ = lag_grad(h_p)
                tp = min(1.0, float(np.max(Qp)) / n + 1e-4)
                zp = np.append(h_p, tp)
                res = minimize(sl_obj, zp, jac=sl_obj_jac, method='SLSQP',
                               bounds=bounds, constraints=cons, options=opts)
                if res.x is not None and len(res.x) == n + 1:
                    h_new = project_box_sum(res.x[:n], S)
                    if h_new is not None and check_feasible(h_new, n):
                        sc = exact_score(h_new)
                        if sc < best_score - 1e-12:
                            best = h_new
                            best_score = sc
                            improved = True
            except Exception:
                continue
        if improved:
            sigma = max(0.01, 0.7 * sigma)
        else:
            sigma = min(0.25, 1.5 * sigma)

    final = project_box_sum(best, S)
    if final is None or not check_feasible(final, n):
        if parent is not None:
            final = project_box_sum(parent, S)
        if final is None or not check_feasible(final, n):
            final = np.full(n, 0.5)
            final = project_box_sum(final, S)
    final_score = exact_score(final)
    return (final, final_score, n)
