from scipy.optimize import minimize

def capped_simplex_project(v, target):
    n = len(v)
    if n == 0 or not np.all(np.isfinite(v)):
        return None
    v = np.array(v, dtype=np.float64).copy()
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-14:
        cap = (1.0 - y) if r > 0 else y.copy()
        tc = cap.sum()
        if tc > 1e-14:
            y += r * cap / tc
        else:
            return None
    y = np.clip(y, 0.0, 1.0)
    if abs(y.sum() - target) > 1e-10 or not np.all(np.isfinite(y)):

    return y

    return np.inf

    return h.sum() - len(h) / 2.0
