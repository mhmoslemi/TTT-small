    roots.append(q / C)

        roots.append(A / q)
    return roots

def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def capped_simplex_project(v, target):
    n = len(v)
    if not np.all(np.isfinite(v)) or np.max(np.abs(v)) > 1e6:
        return None
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0).copy()
    r = target - float(np.sum(y))
    for _ in range(4):
        if abs(r) < 1e-14:
            break
        for i in range(n):
            if r > 0:
                a = min(r, 1.0 - y[i])
            else:
                a = min(-r, y[i])
            if a > 0:
                y[i] += a if r > 0 else -a
                r -= a if r > 0 else a
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(float(np.sum(y)) - target) > 1e-10:
        return None
    return y

def norm_dir(d, n):
    d = d.astype(float).copy()
    if not np.all(np.isfinite(d)) or np.max(np.abs(d)) < 1e-15:
        return None
    d -= np.mean(d)
    d[-1] -= np.sum(d)
    d /= np.max(np.abs(d))
    d[-1] -= np.sum(d)
    return d

def qroots(A, B, C):
    tol = 1e-14 * max(1.0, abs(A), abs(B), abs(C))
    if abs(C) < tol:
        return [-A / B] if abs(B) >= tol else []
    disc = B * B - 4.0 * C * A
    if disc < -tol * max(abs(B * B), abs(4.0 * C * A), 1e-30):
        return []
    disc = max(disc, 0.0)
    sq = math.sqrt(disc)
    q = -0.5 * (B + math.copysign(sq, B))
    if abs(q) < tol:
        return [sq / (2.0 * C), -sq / (2.0 * C)]
    return [q / C, A / q]
