    c = np.correlate(h, 1.0 - h, mode='full')

    return (h.ndim == 1 and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(np.sum(h) - S) <= 1e-10)

def project_to_K(y, S, n):
    if y.ndim != 1 or len(y) != n or not np.all(np.isfinite(y)):
        return None
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))

            lo = mid
        else:
            hi = mid

            break

        mask = slack > 1e-10
        if np.sum(mask) > 0:

            x[mask] += amt
    elif r < -1e-12:

    return None

    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if not np.all(np.isfinite(c)):
        return None
    return c

def grad_matrix(h, dx, n):
    G = np.zeros((2*n-1, n))
    for ell in range(-(n-1), n):
        r = ell + (n-1)
        for j in range(n):
            v = 0.0
            if 0 <= j - ell < n:
                v += 1.0 - h[j - ell]
            if 0 <= j + ell < n:
                v -= h[j + ell]
            G[r, j] = dx * v
    return G

def solve_qp(h, q, G, dx, S, n):
    dv = cp.Variable(n)
    tv = cp.Variable()
    cons = [q + G @ dv <= tv,
            h + dv >= 0, h + dv <= 1,
            cp.sum(dv) == S - float(np.sum(h))]
    obj = cp.Minimize(tv + dx * cp.sum_squares(dv))
    for solver, kwargs in [(cp.CVXOPT, {'max_iters': 150}),
                           (cp.ECOS, {'max_iters': 200})]:
        try:
            prob = cp.Problem(obj, cons)
            prob.solve(solver=solver, **kwargs)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                d = dv.value
                if d is None or not np.all(np.isfinite(d)) or d.shape != (n,):
                    continue
                nd = h + d
                if np.any(nd < -1e-5) or np.any(nd > 1.0+1e-5):
                    continue
                if abs(np.sum(nd) - S) > 1e-5:
                    continue
                if not np.all(np.isfinite(nd)):
                    continue
                return d
        except Exception:
            continue
    return None

    small_decr = 0
    for it in range(max_iter):

            break

import numpy as np
import math
import cvxpy as cp


def exact_score(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def is_feasible(h, S):
    return (h.ndim == 1 and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(float(np.sum(h)) - S) <= 1e-10)


def project_to_K(y, S, n):
    if y.ndim != 1 or len(y) != n or not np.all(np.isfinite(y)):
        return None
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - float(np.sum(x))
    if r > 1e-12:
        mask = (1.0 - x) > 1e-10
        if np.sum(mask) > 0:
            amt = min(r, float(np.sum((1.0 - x)[mask]))) / int(np.sum(mask))
            x[mask] += amt
    elif r < -1e-12:
        mask = x > 1e-10
        if np.sum(mask) > 0:
            amt = max(r, -float(np.sum(x[mask]))) / int(np.sum(mask))
            x[mask] += amt
    if is_feasible(x, S):
        return x.copy()
    return None


def lag_scores(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if not np.all(np.isfinite(c)):
        return None
    return c


def grad_matrix(h, dx, n):
    G = np.zeros((2 * n - 1, n))
    for ell in range(-(n - 1), n):
        r = ell + (n - 1)
        for j in range(n):
            v = 0.0
            if 0 <= j - ell < n:
                v += 1.0 - h[j - ell]
            if 0 <= j + ell < n:
                v -= h[j + ell]
            G[r, j] = dx * v
    return G


def solve_qp(h, q, G, dx, S, n):
    dv = cp.Variable(n)
    tv = cp.Variable()
    cons = [q + G @ dv <= tv,
            h + dv >= 0, h + dv <= 1,
            cp.sum(dv) == S - float(np.sum(h))]
    obj = cp.Minimize(tv + dx * cp.sum_squares(dv))
    for solver, kw in [(cp.CVXOPT, {'max_iters': 150}), (cp.ECOS, {'max_iters': 200})]:
        try:
            prob = cp.Problem(obj, cons)
            prob.solve(solver=solver, **kw)
            if prob.status in ('optimal', 'optimal_inaccurate') and dv.value is not None:
                d = np.asarray(dv.value, dtype=np.float64)
                if d.shape != (n,) or not np.all(np.isfinite(d)):
                    continue
                nd = h + d
                if np.any(nd < -1e-5) or np.any(nd > 1.0 + 1e-5):
                    continue
                if abs(float(np.sum(nd)) - S) > 1e-5:
                    continue
                return d
        except Exception:
            continue
    return None


def optimize_start(h0, dx, S, n, max_iter, max_bk):
    h = h0.copy()
    cur = exact_score(h, dx)
    small_cnt = 0
    for _ in range(max_iter):
        q = lag_scores(h, dx)
        if q is None:
            break
        G = grad_matrix(h, dx, n)
        d = solve_qp(h, q, G, dx, S, n)
        if d is None:
            break
        hp = project_to_K(h + d, S, n)
        if hp is None:
            break
        d_feas = hp - h
        if np.max(np.abs(d_feas)) <= 1e-9:
            break
        best_sc = cur
        best_h = None
        for k in range(max_bk):
            a = 2.0 ** (-k)
            cand = project_to_K(h + a * d_feas, S, n)
            if cand is None:
                continue
            sc = exact_score(cand, dx)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_h = cand
        if best_h is None:
            break
        if cur - best_sc < 1e-10:
            small_cnt += 1
            h = best_h.copy()
            cur = best_sc
            if small_cnt >= 3:
                break
        else:
            small_cnt = 0
            h = best_h.copy()
            cur = best_sc
    return h, cur


def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)

    parent = None
    n = 45
    try:
        ph = globals().get('initial_h_values', None)
        if ph is not None:
            ph = np.array(ph, dtype=np.float64).ravel()
            if 2 <= len(ph) <= 999 and np.all(np.isfinite(ph)):
                n = len(ph)
                S_tmp = n / 2.0
                if np.all(ph >= -1e-12) and np.all(ph <= 1.0 + 1e-12) and abs(np.sum(ph) - S_tmp) <= 1e-12:
                    parent = ph.copy()
                else:
                    if np.all(np.abs(ph) <= 1e6):
                        p = project_to_K(ph, S_tmp, n)
                        if p is not None:
                            parent = p
    except Exception:
        pass

    if parent is None and kwargs.get('n_points') is not None:
        try:
            kn = int(kwargs['n_points'])
            if 2 <= kn <= 999:
                n = kn
        except Exception:
            pass

    S = n / 2.0
    dx = 2.0 / n
    fallback = np.full(n, 0.5, dtype=np.float64)

    starts = []
    if parent is not None:
        starts.append(parent.copy())
        starts.append((1.0 - parent).copy())
        starts.append(parent[::-1].copy())
        for sd in [0.08, 0.25]:
            nz = parent + sd * rng.standard_normal(n)
            p = project_to_K(nz, S, n)
            if p is not None:
                starts.append(p)
        p = project_to_K(rng.beta(0.5, 0.5, n), S, n)
        if p is not None:
            starts.append(p)
    else:
        starts.append(fallback.copy())
        p = project_to_K(rng.random(n), S, n)
        if p is not None:
            starts.append(p)
        p = project_to_K(rng.beta(0.5, 0.5, n), S, n)
        if p is not None:
            starts.append(p)
        m = int(rng.integers(1, max(1, min(4, n // 2)) + 1))
        phase = float(rng.uniform(0.0, 2.0 * math.pi))
        trig = 0.5 + 0.35 * np.cos(2.0 * math.pi * m * (np.arange(n) + 0.5) / n + phase)
        p = project_to_K(trig, S, n)
        if p is not None:
            starts.append(p)

    max_starts = 6 if n <= 64 else (4 if n <= 256 else 3)
    deduped = []
    for s in starts:
        if not is_feasible(s, S):
            continue
        dup = False
        for d0 in deduped:
            if np.max(np.abs(s - d0)) <= 1e-8:
                dup = True
                break
        if not dup:
            deduped.append(s)
        if len(deduped) >= max_starts:
            break

    best_h = fallback.copy()
    best_sc = exact_score(fallback, dx)

    max_iter = 40 if n <= 64 else (20 if n <= 256 else 8)
    max_bk = 9 if n <= 64 else (6 if n <= 256 else 4)

    for s in deduped:
        sc = exact_score(s, dx)
        if sc < best_sc - 1e-12:
            best_h = s.copy()
            best_sc = sc
        try:
            h_opt, sc_opt = optimize_start(s, dx, S, n, max_iter, max_bk)
            if sc_opt < best_sc - 1e-12:
                best_h = h_opt.copy()
                best_sc = sc_opt
        except Exception:
            pass

    if not is_feasible(best_h, S):
        p = project_to_K(best_h, S, n)
        if p is not None:
            best_h = p.copy()
        if not is_feasible(best_h, S):
            best_h = fallback.copy()

    final_sc = exact_score(best_h, dx)
    return best_h.astype(np.float64).copy(), final_sc, int(n)
