def lag_model(x):
    n = len(x)
    m = 2 * n - 1
    B = np.zeros(m)
    G = np.zeros((m, n))
    for dd in range(-(n-1), n):
        row = dd + (n - 1)
        lo = max(0, -dd)
        hi = min(n, n - dd)
        length = hi - lo
        if length <= 0:
            continue
        B[row] = np.dot(x[lo:hi], 1.0 - x[lo+dd:hi+dd])
        G[row, lo:hi] += (1.0 - x[lo+dd:hi+dd])

        hi2 = hi + dd
        if lo2 >= 0 and hi2 <= n and hi2 > lo2:
            G[row, lo2:hi2] -= x[lo:hi]
    return B, G

    return y, Qmodel, B

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = None
    n = None
    pv = globals().get('initial_h_values', None)
    if pv is not None:
        try:
            pv = np.asarray(pv, dtype=np.float64)
            if pv.ndim == 1 and len(pv) >= 2 and np.all(np.isfinite(pv)):
                n = len(pv)
                p = project_box_sum(pv, 0.5 * n, np.zeros(n), np.ones(n))
                parent = p if p is not None else np.full(n, 0.5)
        except Exception:
            parent = None
    if parent is None:
        n = 80
        try:
            nv = int(kwargs.get('n_points', 80))
            if 8 <= nv <= 512:
                n = nv
        except Exception:
            pass
        parent = np.full(n, 0.5)
    S = 0.5 * n
    dx = 2.0 / n
    best = parent.copy()
    best_score = exact_score(best)
    if best_score is None:
        best = np.full(n, 0.5)
        best_score = exact_score(best)
    if n > 512:
        sc = exact_score(best)
        return best, sc, n
    if n <= 128:
        max_starts, max_iter = 6, 36
    elif n <= 256:
        max_starts, max_iter = 4, 24
    else:
        max_starts, max_iter = 2, 12
    starts = make_starts(n, rng, parent)[:max_starts]
    D0, Dmin, Dmax = 0.20, 1e-4, 0.5
    for start in starts:
        try:
            x = start.copy()
            score_x = exact_score(x)
            if score_x is None or not feasible_check(x):
                continue
            if score_x < best_score - 1e-14 * max(1.0, abs(best_score)):
                best, best_score = x.copy(), score_x
            Delta = D0
            consec_fail, small_steps = 0, 0
            for _ in range(max_iter):
                y, Qmodel, B = slp_step(x, S, Delta)
                if y is None:
                    consec_fail += 1
                    Delta = max(0.25 * Delta, Dmin)
                    if consec_fail >= 3 or Delta <= Dmin:
                        break
                    continue
                consec_fail = 0
                score_y = exact_score(y)
                if score_y is None or not feasible_check(y):
                    Delta = 0.25 * Delta
                    if Delta < Dmin:
                        break
                    continue
                if score_y < best_score - 1e-14 * max(1.0, abs(best_score)):
                    best, best_score = y.copy(), score_y
                actual = score_x - score_y
                pred = dx * (float(np.max(B)) - Qmodel)
                if pred <= 1e-12 * max(1.0, abs(score_x)):
                    break
                rho = actual / pred
                if actual > 1e-12 * max(1.0, abs(score_x)) and rho >= 0.1:
                    ms = float(np.max(np.abs(y - x)))
                    if ms < 1e-10:
                        break
                    x, score_x = y, score_y
                    if rho >= 0.75 and ms >= 0.8 * Delta:
                        Delta = min(2.0 * Delta, Dmax)
                    elif rho < 0.25:
                        Delta = max(0.25 * Delta, Dmin)
                    small_steps = small_steps + 1 if abs(actual) < 1e-10 * max(1.0, abs(score_x)) else 0
                    if small_steps >= 8:
                        break
                else:
                    Delta = 0.25 * Delta
                    if Delta < Dmin:
                        break
        except Exception:
            break
    sc = exact_score(best)
    if sc is None or not feasible_check(best):
        best = np.full(n, 0.5)
        sc = exact_score(best)
    return best, sc, n
