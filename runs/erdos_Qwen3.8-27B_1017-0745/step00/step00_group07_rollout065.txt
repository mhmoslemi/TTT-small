import numpy as np
from scipy.optimize import minimize
import math

def is_feasible(h, n):
    if h.ndim != 1 or h.shape[0] != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-15) or np.any(h > 1.0 + 1e-15):
        return False
    return abs(float(np.sum(h)) - n / 2.0) <= 1e-10

def project_capped(z, S):
    n = len(z)
    if not np.all(np.isfinite(z)) or np.max(np.abs(z)) > 1e12:
        return None
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z))
    for _ in range(80):
        mid = lo + 0.5 * (hi - lo)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(z - (lo + 0.5 * (hi - lo)), 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-14:
        for i in range(n):
            if r <= 0: break
            s = 1.0 - y[i]
            if s > 1e-15:
                a = min(r, s); y[i] += a; r -= a
    elif r < -1e-14:
        for i in range(n):
            if r >= 0: break
            s = y[i]
            if s > 1e-15:
                a = min(-r, s); y[i] -= a; r += a
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(float(np.sum(y)) - S) > 1e-10:
        return None
    return y

def compute_lags(h):
    n = len(h)
    B = np.zeros(2 * n - 1)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        r = d + n - 1
        lo, hi = max(0, -d), min(n, n - d)
        src, dst = h[lo:hi], h[lo + d:hi + d]
        B[r] = np.dot(src, 1.0 - dst)
        G[r, lo:hi] += 1.0 - dst
        G[r, lo + d:hi + d] -= src
    return B, G

def exact_score(h):
    n = len(h)
    if n < 2 or not np.all(np.isfinite(h)):
        return float('inf')
    v = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(np.max(v)) if np.all(np.isfinite(v)) else float('inf')

def slsqp_obj(z):
    return z[-1]

def slsqp_obj_grad(z):
    g = np.zeros(len(z)); g[-1] = 1.0; return g

def slsqp_ineq(z):
    n = len(z) - 1
    B, _ = compute_lags(z[:n])
    return z[-1] - B

def slsqp_ineq_jac(z):
    n = len(z) - 1
    _, G = compute_lags(z[:n])
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G; J[:, n] = 1.0
    return J

def slsqp_eq(z):
    n = len(z) - 1
    return float(np.sum(z[:n])) - n / 2.0

def slsqp_eq_jac(z):
    J = np.zeros(len(z)); J[:-1] = 1.0; return J

def try_add(starts, h, n, S):
    if h is None: return
    h = np.asarray(h, dtype=np.float64)
    if h.ndim != 1 or h.shape[0] != n or not np.all(np.isfinite(h)): return
    if np.max(np.abs(h)) > 1e12: return
    pr = project_capped(h, S)
    if pr is None or not is_feasible(pr, n): return
    for s in starts:
        if np.max(np.abs(pr - s)) < 1e-10: return
    starts.append(pr)

def build_starts(n, S, seed, parent):
    rng = np.random.default_rng(seed)
    starts = []
    const = np.full(n, 0.5)
    if parent is not None:
        try_add(starts, parent, n, S)
    try_add(starts, const, n, S)
    if parent is not None:
        try_add(starts, parent[::-1].copy(), n, S)
        try_add(starts, parent + 0.18 * rng.standard_normal(n), n, S)
    if n > 4:
        qmax = min(6, (n - 1) // 2)
        if qmax >= 1:
            q = int(rng.integers(1, qmax + 1))
            phi = float(rng.uniform(0, 2 * math.pi))
            ii = np.arange(n, dtype=np.float64)
            try_add(starts, 0.5 + 0.45 * np.cos(2 * math.pi * q * ii / n + phi), n, S)
    perm = rng.permutation(n)
    bb = np.zeros(n)
    bb[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        bb[perm[n // 2]] = 0.5
    try_add(starts, bb, n, S)
    if parent is None:
        for _ in range(4):
            try_add(starts, 0.5 + 0.30 * rng.standard_normal(n), n, S)
            if len(starts) >= 6: break
    att = 0
    while len(starts) < 6 and att < 12:
        if att % 2 == 0:
            try_add(starts, 0.5 + 0.25 * rng.standard_normal(n), n, S)
        else:
            try_add(starts, rng.uniform(0, 1, n), n, S)
        att += 1
    return starts
