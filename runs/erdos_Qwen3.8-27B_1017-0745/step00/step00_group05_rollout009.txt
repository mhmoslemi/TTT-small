            if not any(np.max(np.abs(pop[k] - incumbent)) < 1e-10 for k in keep):
                keep[-1] = -1
            np2, ns2 = [], []
            for k in keep:
                if k == -1:
                    np2.append(incumbent.copy())
                    ns2.append(inc_score)
                else:
                    np2.append(pop[k].copy())
                    ns2.append(scores[k])
            sigmas = [0.02, 0.06, 0.15, 0.30]
            si = 0
            while len(np2) < NP and eval_count + NP <= budget:
                if si % 2 == 0:
                    sv = sigmas[(si // 2) % 4]
                    vv = incumbent + sv * rng.standard_normal(n)
                else:
                    vv = random_profile(n, rng, incumbent)
                si += 1
                pj = capped_simplex_project(vv, target)
                if pj is None or not is_feasible(pj, target):
                    continue
                eval_count += 1
                sc = exact_score(pj, dx)
                if not isfinite(sc):
                    continue
                np2.append(pj)
                ns2.append(sc)
                if sc < inc_score - 1e-12:
                    incumbent, inc_score = pj.copy(), sc
            while len(np2) < NP:
                np2.append(incumbent.copy())
                ns2.append(inc_score)
            pop, scores = np2, ns2
            archive, mu_F, mu_CR = [], 0.5, 0.7
    final = capped_simplex_project(incumbent, target)
    if final is None or not is_feasible(final, target):
        final = fp
    fs = exact_score(final, dx)
    if not isfinite(fs):
        final, fs = fp, exact_score(fp, dx)
    return (final, fs, n)
