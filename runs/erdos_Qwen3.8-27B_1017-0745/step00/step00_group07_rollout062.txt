import numpy as np
from scipy.optimize import minimize


def capped_simplex_projection(z, S):
    z = np.asarray(z, dtype=np.float64).ravel()
    n = len(z)
    if n < 1 or not np.all(np.isfinite(z)):
        return None
    if S < -1e-12 or S > n + 1e-12:
        return None
    if float(np.max(np.abs(z))) > 1e12:
        return None
    low = float(np.min(z)) - 1.0
    high = float(np.max(z))
    for _ in range(80):
        mid = low + 0.5 * (high - low)
        y = np.clip(z - mid, 0.0, 1.0)
        if np.sum(y) > S:
            low = mid
        else:
            high = mid
    y = np.clip(z - 0.5 * (low + high), 0.0, 1.0)
    r = S - np.sum(y)
    if r > 1e-14:
        for i in range(n):
            sl = 1.0 - y[i]
            if sl > 1e-15:
                a = min(r, sl)
                y[i] += a
                r -= a
                if abs(r) < 1e-14:
                    break
    elif r < -1e-14:
        for i in range(n):
            sl = y[i]
            if sl > 1e-15:
                a = min(-r, sl)
                y[i] -= a
                r += a
                if abs(r) < 1e-14:
                    break
    if not np.all(np.isfinite(y)):
        return None
    y = np.clip(y, 0.0, 1.0)
    if abs(np.sum(y) - S) > 1e-10:
        return None
    return y


def compute_lags(h):
    n = len(h)
    B = np.zeros(2 * n - 1)
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        di = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        src = np.arange(lo, hi)
        dst = src + d
        B[di] = np.dot(h[src], 1.0 - h[dst])
        G[di, src] += (1.0 - h[dst])
        G[di, dst] -= h[src]
    return B, G


def exact_score(h):
    h = np.asarray(h, dtype=np.float64).ravel()
    n = len(h)
    if n < 2 or not np.all(np.isfinite(h)):
        return None
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(np.sum(h) - n / 2.0) > 1e-10:
        return None
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    if not np.all(np.isfinite(vals)):
        return None
    s = float(np.max(vals))
    return s if np.isfinite(s) else None


def nlp_obj(z):
    return z[-1]


def nlp_obj_grad(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g


def nlp_ineq(z):
    n = len(z) - 1
    B, _ = compute_lags(z[:n])
    return z[-1] - B


def nlp_ineq_jac(z):
    n = len(z) - 1
    _, G = compute_lags(z[:n])
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J


def nlp_eq(z):
    n = len(z) - 1
    return np.sum(z[:n]) - n / 2.0


def nlp_eq_jac(z):
    n = len(z) - 1
    J = np.zeros(n + 1)
    J[:n] = 1.0
    return J

    if 'initial_h_values' in globals():
        try:

    S = n / 2.0

    starts.append(const_h.copy())

    if len(starts) < max_starts:
        perm = rng.permutation(n)
        bb = np.zeros(n)

        if n % 2 == 1:
            bb[perm[n // 2]] = 0.5

        if not dup:

                best_score = s

    if n <= 128:
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]

                        if s is not None and s < best_score:
                            best_score = s
                            best_h = h_try.copy()
            except Exception:
                pass

    final_score = exact_score(best_h)
    if final_score is None:
        best_h = const_h.copy()
        final_score = exact_score(const_h)

    return (best_h.copy(), float(final_score), n)
