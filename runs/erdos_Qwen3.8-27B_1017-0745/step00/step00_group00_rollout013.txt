        quota = max(1, (budget[1] - budget[0]) // max(1, chains_left))
        current = start.copy(); cur_sc = start_sc
        pilot_data = []
        pb_best = current.copy(); pb_sc = cur_sc
        p_att = min(pilot_lim * 4, max(4, quota // 2))
        for _ in range(p_att):
            if budget[0] >= budget[1]:
                break
            tr = make_proposal(current, 0.35, rng, n)
            if tr is None:
                continue
            sc = budgeted_score(tr, n, budget)
            if sc is not None:
                pilot_data.append(sc)
                if sc < pb_sc:
                    pb_sc = sc; pb_best = tr.copy()
                if len(pilot_data) >= pilot_lim:
                    break
        deltas = [p - start_sc for p in pilot_data]
        pos = [d for d in deltas if d > 1e-15]
        dref = float(np.median(pos)) if pos else (float(np.median([abs(d) for d in deltas if abs(d) > 1e-15])) if any(abs(d) > 1e-15 for d in deltas) else 0.001)
        T0 = max(1e-5, min(0.05, dref / 0.69314718056))
        if not np.isfinite(T0) or T0 <= 0:
            T0 = 0.002
        if pb_sc < cur_sc:
            current = pb_best.copy(); cur_sc = pb_sc
            if cur_sc < best_score:
                best_score = cur_sc; best_h = current.copy()
        rem = max(0, quota - len(pilot_data))
        alloc = []
        left = rem
        for si in range(n_stages):
            a = min(stage_moves, max(0, left // (n_stages - si)))
            alloc.append(a); left -= a
        if left > 0:
            alloc[-1] += left
        r = 0.35
        for stage in range(n_stages):
            T_J = T0 * (cooling ** stage)
            la, lacc = 0, 0
            for _ in range(alloc[stage]):
                if budget[0] >= budget[1]:
                    break
                try:
                    d = random_dir(n, rng)
                    iv = chord_interval(current, d)
                    if iv is None:
                        continue
                    lo, hi = iv
                    is_loc = rng.random() >= 0.25
                    tr = None
                    if is_loc:
                        la += 1
                        a = r * (2.0 * rng.random() - 1.0)
                        if lo <= a <= hi:
                            tr = current + a * d
                    else:
                        tr = current + (lo + rng.random() * (hi - lo)) * d
                    if tr is None:
                        continue
                    tr = repair_trial(tr, n)
                    if tr is None:
                        continue
                    ts = budgeted_score(tr, n, budget)
                    if ts is None:
                        continue
                    if ts < best_score:
                        best_score = ts; best_h = tr.copy()
                    accepted = ts <= cur_sc
                    if not accepted:
                        du = max(rng.random(), np.finfo(np.float64).tiny)
                        if math.log(du) < -(ts - cur_sc) / T_J:
                            accepted = True
                    if accepted:
                        current = tr; cur_sc = ts
                        if is_loc:
                            lacc += 1
                except Exception:
                    pass
            if la > 0:
                fr = lacc / la
                if fr < 0.15:
                    r = max(0.03, 0.75 * r)
                elif fr > 0.45:
                    r = min(1.2, 1.20 * r)
    final = project_box_sum(best_h, S)
    if final is not None and is_feasible(final, n):
        fs = exact_score(final, n)
        if fs is not None:
            return (final, fs, n)
    fb = project_box_sum(fallback, S)
    if fb is not None and is_feasible(fb, n):
        s = exact_score(fb, n)
        if s is not None:
            return (fb, s, n)
    c = np.full(n, 0.5)
    return (c, 0.5, n)
