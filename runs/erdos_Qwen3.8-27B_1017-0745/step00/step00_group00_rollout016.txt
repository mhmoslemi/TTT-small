        return float('inf')
    return float(np.max(c) * (2.0 / n))

def project_and_score(h, S, fb):
    n = len(h)
    p = project_box_sum(h, S)
    if p is None or not is_feasible(p, n):
        return fb.copy(), exact_score(fb, n)
    return p, exact_score(p, n)

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    n, S = 45, 22.5
    fallback_h = np.full(n, 0.5, dtype=np.float64)
    parent_h = None
    try:
        ih = globals().get('initial_h_values', None)
        if ih is not None:
            ih = np.asarray(ih, dtype=np.float64)
            if ih.ndim == 1 and 2 <= len(ih) <= 999 and np.all(np.isfinite(ih)):
                if float(np.min(ih)) >= -1e-8 and float(np.max(ih)) <= 1.0 + 1e-8:
                    n, S = len(ih), len(ih) / 2.0
                    p = project_box_sum(ih, S)
                    if p is not None and is_feasible(p, n):
                        parent_h = p.copy()
                        fallback_h = p.copy()
    except Exception:
        pass
    if parent_h is None:
        n, S = 45, 22.5
        fallback_h = np.full(n, 0.5, dtype=np.float64)
    best_score = exact_score(fallback_h, n)
    best_h = fallback_h.copy()

    P = 24 if n <= 100 else (16 if n <= 300 else 10)
    max_gen = 200 if n <= 100 else (120 if n <= 300 else 60)
    max_budget = min(3000, max(P + 2, floor(20000000 / (n * n))))
    evals = 0

    pop, ps = [None] * P, [0.0] * P
    cands = []
    if parent_h is not None:
        cands.append(parent_h)
        cands.append(parent_h[::-1].copy())
        for sig in [0.02, 0.06, 0.12, 0.25]:
            cands.append(parent_h + sig * rng.standard_normal(n))
    for _ in range(4):
        cands.append(rng.uniform(0, 1, n))
    for m in range(1, 5):
        t = np.arange(n, dtype=np.float64)
        cands.append(0.5 + 0.35 * np.cos(2.0 * np.pi * m * t / n + 0.37 * m))
    blk = np.zeros(n); blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    cands.append(blk)
    ci = 0
    for idx in range(P):
        if ci < len(cands):
            h_init = cands[ci]; ci += 1
        else:
            h_init = rng.uniform(0, 1, n) if idx % 2 == 0 else fallback_h + 0.1 * rng.standard_normal(n)
        proj, sc = project_and_score(h_init, S, fallback_h)
        evals += 1
        pop[idx] = proj
        ps[idx] = sc
        if sc < best_score - 1e-12 * max(1.0, abs(best_score)):
            best_h, best_score = proj.copy(), sc

    Fmean, CRmean = 0.65, 0.85
    pbest_sz = max(2, ceil(0.2 * P))
    no_imp, restart_lv = 0, 0

    for gen in range(max_gen):
        if evals >= max_budget:
            break
        Xold = [p.copy() for p in pop]
        s_old = list(ps)
        Xnew, s_new = list(Xold), list(s_old)
        order = np.argsort(s_old)
        pbest_pool = [int(x) for x in order[:pbest_sz]]
        F_list, CR_list = [], []
        gen_best = best_score
        for i in rng.permutation(P):
            if evals >= max_budget:
                break
            Fi = Fmean + 0.1 * rng.standard_cauchy()
            if not np.isfinite(Fi):
                Fi = Fmean
            Fi = float(np.clip(Fi, 0.1, 1.0))
            CRI = float(np.clip(CRmean + 0.1 * rng.standard_normal(), 0.05, 1.0))
            if rng.random() < 0.75 and P > 3:
                pool = [c for c in pbest_pool if c != i]
                pb = int(pool[rng.integers(len(pool))]) if pool else int(rng.integers(P))
                others = [c for c in range(P) if c != i and c != pb]
                r1 = int(others[rng.integers(len(others))])
                r2_pool = [c for c in others if c != r1]
                r2 = int(r2_pool[rng.integers(len(r2_pool))])
                v = Xold[i] + Fi * (Xold[pb] - Xold[i]) + Fi * (Xold[r1] - Xold[r2])
            elif P > 3:
                others = [c for c in range(P) if c != i]
                r0 = int(others[rng.integers(len(others))])
                rest = [c for c in others if c != r0]
                r1 = int(rest[rng.integers(len(rest))])
                rest2 = [c for c in rest if c != r1]
                r2 = int(rest2[rng.integers(len(rest2))])
                v = Xold[r0] + Fi * (Xold[r1] - Xold[r2])
            else:
                continue
            jrand = int(rng.integers(n))
            mask = rng.random(n) < CRI
            mask[jrand] = True
            trial = np.where(mask, v, Xold[i])
            proj, sc = project_and_score(trial, S, fallback_h)
            evals += 1
            if not np.isfinite(sc):
                continue
            tol = 1e-12 * max(1.0, abs(s_old[i]))
            if sc < s_old[i] - tol:
                Xnew[i], s_new[i] = proj, sc
                F_list.append(Fi); CR_list.append(CRI)
            if sc < gen_best - 1e-12 * max(1.0, abs(gen_best)):
                best_h, best_score, gen_best = proj.copy(), sc, sc
        pop, ps = Xnew, s_new
        if F_list:
            Fa = np.array(F_list); Ca = np.array(CR_list)
            Fmean = float(np.clip(0.9 * Fmean + 0.1 * float(np.sum(Fa ** 2) / np.sum(Fa)), 0.1, 0.95))
            CRmean = float(np.clip(0.9 * CRmean + 0.1 * float(np.mean(Ca)), 0.05, 0.95))
        if best_score < gen_best:
            no_imp = 0
        else:
            no_imp += 1
        if no_imp >= 8 and evals < max_budget and restart_lv <= 4:
            for j in range(max(1, P // 4)):
                if evals >= max_budget:
                    break
                if j % 3 == 0:
                    imm = rng.uniform(0, 1, n)
                else:
                    sig = min(0.30, 0.05 * 1.6 ** restart_lv)
                    imm = best_h + sig * rng.standard_normal(n)
                proj, sc = project_and_score(imm, S, fallback_h)
                evals += 1
                if not np.isfinite(sc):
                    continue
                if sc < best_score - 1e-12 * max(1.0, abs(best_score)):
                    best_h, best_score = proj.copy(), sc
                bidx = int(np.argmin(ps))
                wx = [c for c in range(P) if c != bidx]
                widx = wx[int(np.argmax([ps[c] for c in wx]))]
                pop[widx], ps[widx] = proj, sc
            restart_lv += 1
            Fmean = min(0.95, Fmean + 0.10)
            CRmean = min(0.95, CRmean + 0.05)
            no_imp = 0

    final_h = project_box_sum(best_h, S)
    if final_h is None or not is_feasible(final_h, n):
        final_h = fallback_h.copy()
    final_sc = exact_score(final_h, n)
    fb_sc = exact_score(fallback_h, n)
    if not np.isfinite(final_sc) or (np.isfinite(fb_sc) and fb_sc < final_sc):
        final_h, final_sc = fallback_h.copy(), fb_sc
    if not np.isfinite(final_sc):
        final_h = np.full(n, 0.5, dtype=np.float64)
        final_sc = exact_score(final_h, n)
    return (final_h.astype(np.float64), float(final_sc), n)
