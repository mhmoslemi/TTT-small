    tol = 1e-10 * max(1, n)
    if abs(float(np.sum(h)) - n / 2.0) > tol:
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full')
    if q.shape != (2 * n - 1,) or not np.all(np.isfinite(q)):
        return float('inf')
    return float((2.0 / n) * np.max(q))

def uniform_vec(n):
    return np.full(n, 0.5, dtype=np.float64)

def try_add(pop, vec, n, rng):
    p = project_capped(vec, n)
    if p is None:
        return False
    for e in pop:
        if np.max(np.abs(p - e)) < 1e-10:
            return False
    pop.append(p)
    return True
