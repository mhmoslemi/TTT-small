def project_capped_simplex(v, N, M):
    """Project v onto {x: 0<=x_i<=1, sum(x)=M} via bisection on lambda."""
    if v.shape != (N,) or not np.all(np.isfinite(v)):
        return None
    if np.max(np.abs(v)) > 1e6:
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > M:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    delta = M - float(np.sum(y))
    tol = 1e-12 * max(1, N)
    for _ in range(2):
        if abs(delta) <= tol:
            break
        if delta > 0:
            for i in range(N):
                if delta > 1e-15:
                    add = min(delta, 1.0 - y[i])
                    y[i] += add
                    delta -= add
        else:
            for i in range(N):
                if -delta > 1e-15:
                    sub = min(-delta, y[i])
                    y[i] -= sub
                    delta += sub
        delta = M - float(np.sum(y))
    if abs(delta) > 1e-9 * max(1, N):
        return None
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
