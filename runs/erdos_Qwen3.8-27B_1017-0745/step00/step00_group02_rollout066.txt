    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def smooth_obj_val(h, n, tau):
    dx = 2.0 / n
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(q.max())
    w = np.exp((q - m) / tau)
    return m + tau * math.log(float(w.sum()))

def smooth_grad(h, n, tau):
    dx = 2.0 / n
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(q.max())
    w = np.exp((q - m) / tau)
    p = w / float(w.sum())
    g = np.zeros(n)
    for k in range(2 * n - 1):
        pk = float(p[k])
        if pk < 1e-15:
            continue
        j_lo = max(0, -k)
        j_hi = min(n, n - k)
        for j in range(j_lo, j_hi):
            g[j + k] += pk * dx * (1.0 - h[j])
            g[j] -= pk * dx * h[j + k]
    return g

def pg_optimize(h0, n, tau_list, max_iter):
    h = h0.copy()
    for tau in tau_list:
        alpha = 1.0
        for _ in range(max_iter):
            fv = smooth_obj_val(h, n, tau)
            if not np.isfinite(fv):
                break
            g = smooth_grad(h, n, tau)
            if not np.all(np.isfinite(g)):
                break
            accepted = False
            for _ls in range(30):
                y = project_onto_K(h - alpha * g, n)
                d = y - h
                if np.max(np.abs(d)) < 1e-12:
                    break
                gd = float(g @ d)
                if gd > -1e-14:
                    break
                fy = smooth_obj_val(y, n, tau)
                if np.isfinite(fy) and fy <= fv + 1e-4 * gd + 1e-14:
                    h = y
                    accepted = True
                    break
                alpha *= 0.5
            if not accepted:
                break
            alpha = min(1.25 * alpha, 5.0)
    return h

def _s_obj(z):
    return z[_slsqp_n]

def _s_obj_g(z):
    g = np.zeros(_slsqp_n + 1)
    g[_slsqp_n] = 1.0
    return g

def _s_ineq(z):
    n = _slsqp_n
    dx = 2.0 / n
    h = z[:n]
    t = z[n]
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - q

def _s_ineq_j(z):
    n = _slsqp_n
    dx = 2.0 / n
    h = z[:n]
    J = np.zeros((2 * n - 1, n + 1))
    J[:, n] = 1.0
    for k in range(2 * n - 1):
        j_lo = max(0, -k)
        j_hi = min(n, n - k)
        for j in range(j_lo, j_hi):
            J[k, j + k] -= dx * (1.0 - h[j])
            J[k, j] += dx * h[j + k]
    return J

def _s_eq(z):
    return np.array([z[:_slsqp_n].sum() - _slsqp_n / 2.0])

def _s_eq_j(z):
    n = _slsqp_n
    J = np.zeros((1, n + 1))
    J[0, :n] = 1.0
    return J

def slsqp_refine(h0, n):
    global _slsqp_n
    try:
        from scipy.optimize import minimize
    except ImportError:
        return None
    _slsqp_n = n
    dx = 2.0 / n
    S = n / 2.0
    sc = exact_score(h0, n)
    z0 = np.concatenate([h0, [min(1.0, sc + 1e-8)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    constr = [
        {'type': 'ineq', 'fun': _s_ineq, 'jac': _s_ineq_j},
        {'type': 'eq', 'fun': _s_eq, 'jac': _s_eq_j},
    ]
    try:
        res = minimize(_s_obj, z0, method='SLSQP', jac=_s_obj_g,
                       bounds=bounds, constraints=constr,
                       options={'maxiter': 300, 'ftol': 1e-11, 'disp': False})
        if res is not None and res.x is not None:
            hn = res.x[:n]
            if np.all(np.isfinite(hn)) and len(hn) == n:
                hn = project_onto_K(hn, n)
                if check_feasible(hn, n):
                    return hn
    except Exception:
        pass
    return None

def run(seed=42, **kwargs):
    global _slsqp_n
    rng = np.random.default_rng(seed)
    n = None
    h_parent = None
    if 'initial_h_values' in globals():
        hp = globals()['initial_h_values']
        if hp is not None:
            try:
                hp = np.array(hp, dtype=np.float64).ravel()
                if 2 <= len(hp) <= 999 and np.all(np.isfinite(hp)) and np.max(np.abs(hp)) < 1e6:
                    h_parent = hp.copy()
                    n = len(hp)
            except Exception:
                pass
    if n is None:
        n = 45
        h_parent = np.full(n, 0.5)
    dx = 2.0 / n
    h_fb = project_onto_K(h_parent, n)
    if not check_feasible(h_fb, n):
        h_fb = np.full(n, 0.5)
    best_h = h_fb.copy()
    best_sc = exact_score(best_h, n)
    if not np.isfinite(best_sc):
        best_sc = 1.0

    starts = []
    starts.append(h_fb.copy())
    starts.append(project_onto_K(1.0 - h_fb, n))
    starts.append(project_onto_K(h_fb[::-1].copy(), n))
    starts.append(project_onto_K(1.0 - h_fb[::-1].copy(), n))
    starts.append(project_onto_K(h_fb + 0.08 * rng.standard_normal(n), n))
    starts.append(project_onto_K(h_fb + 0.25 * rng.standard_normal(n), n))
    starts.append(project_onto_K(rng.uniform(0, 1, n), n))
    starts.append(project_onto_K(rng.beta(0.5, 0.5, n), n))

    if n <= 100:
        taus = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00001]
        mx = 80
    else:
        taus = [0.02, 0.005, 0.001, 0.0001, 0.00001]
        mx = 25

    local_res = []
    for s in starts:
        if not check_feasible(s, n):
            s = project_onto_K(s, n)
        sc = exact_score(s, n)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best_sc = sc
            best_h = s.copy()
        ho = pg_optimize(s, n, taus, mx)
        if check_feasible(ho, n):
            sc2 = exact_score(ho, n)
            if np.isfinite(sc2) and sc2 < best_sc - 1e-12:
                best_sc = sc2
                best_h = ho.copy()
            local_res.append(ho)
        else:
            local_res.append(s.copy())

    if n <= 100:
        for sig in [0.04, 0.12, 0.30, 0.50]:
            pert = project_onto_K(best_h + sig * rng.standard_normal(n), n)
            if not check_feasible(pert, n):
                continue
            ho = pg_optimize(pert, n, taus, mx)
            if check_feasible(ho, n):
                sc3 = exact_score(ho, n)
                if np.isfinite(sc3) and sc3 < best_sc - 1e-12:
                    best_sc = sc3
                    best_h = ho.copy()
                local_res.append(ho)

    if n <= 100:
        scored = []
        seen = []
        for hr in local_res:
            hh = project_onto_K(hr, n)
            if not check_feasible(hh, n):
                continue
            sv = exact_score(hh, n)
            if not np.isfinite(sv):
                continue
            scored.append((sv, hh))
        scored.sort(key=lambda x: x[0])
        cands = []
        for sv, hh in scored:
            if len(cands) >= 3:
                break
            ok = True
            for ch in cands:
                if np.max(np.abs(hh - ch)) < 1e-5:
                    ok = False
                    break
            if ok:
                cands.append(hh)
        if len(cands) == 0:
            cands = [best_h.copy()]
        for cnd in cands:
            hr2 = slsqp_refine(cnd, n)
            if hr2 is not None and check_feasible(hr2, n):
                sc4 = exact_score(hr2, n)
                if np.isfinite(sc4) and sc4 < best_sc - 1e-12:
                    best_sc = sc4
                    best_h = hr2.copy()

    best_h = project_onto_K(best_h, n)
    if not check_feasible(best_h, n):
        best_h = h_fb.copy()
    final_sc = exact_score(best_h, n)
    fb_sc = exact_score(h_fb, n)
    if np.isfinite(fb_sc) and (not np.isfinite(final_sc) or fb_sc < final_sc):
        best_h = h_fb.copy()
        final_sc = fb_sc
    if not np.isfinite(final_sc):
        best_h = np.full(n, 0.5)
        final_sc = exact_score(best_h, n)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_K(best_h, n)
    final_sc = exact_score(best_h, n)
    return best_h.copy(), float(final_sc), int(n)
