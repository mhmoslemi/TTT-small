import numpy as np
import math

def is_feasible(h, n, S):
    if h.ndim != 1 or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1 + 1e-12):
        return False
    if abs(np.sum(h) - S) > 1e-10:
        return False
    return True

def project_capped_simplex(z, S):
    if len(z) == 0 or not np.all(np.isfinite(z)) or np.max(np.abs(z)) > 1e12:
        return None
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(80):
        mid = lo + 0.5 * (hi - lo)
        y = np.clip(z - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(z - hi, 0.0, 1.0)
    residual = S - np.sum(y)
    if residual > 0:
        slack = 1.0 - y
        idx = np.where(slack > 1e-15)[0]
        add = min(residual, np.sum(slack[idx]))
        if add > 0 and len(idx) > 0:
            y[idx] += add / len(idx)
    elif residual < 0:
        slack = y.copy()
        idx = np.where(slack > 1e-15)[0]
        sub = min(-residual, np.sum(slack[idx]))
        if sub > 0 and len(idx) > 0:
            y[idx] -= sub / len(idx)
    y = np.clip(y, 0.0, 1.0)
    r2 = S - np.sum(y)
    if abs(r2) > 1e-10:
        return None
    if not np.all(np.isfinite(y)) or np.any(y < 0) or np.any(y > 1):
        return None
    return y

def exact_score(h, n):
    if not np.all(np.isfinite(h)):
        return None
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(corr) * (2.0 / n))
    if not np.isfinite(val):
        return None
    return val

def score_with_raw(h, n):
    s = exact_score(h, n)
    if s is None:
        return None, None
    corr = np.correlate(h, 1.0 - h, mode='full')
    # corr[i] corresponds to lag d = i - (n-1)
    # B_d = corr[n-1+d]
    raw = corr.copy()  # index i ↔ lag d=i-(n-1), so B_d = raw[n-1+d]
    return s, raw

def quad_roots(A, B, C):
    scale = max(1.0, abs(B), abs(C))
    if abs(A) < 1e-14 * scale:
        if abs(B) < 1e-14 * scale:
            return []
        r = -C / B
        if np.isfinite(r):
            return [r]
        return []
    disc = B * B - 4.0 * A * C
    tol = -1e-10 * scale * scale
    if disc < tol:
        return []
    if disc < 0:
        disc = 0.0
    sq = math.sqrt(disc)
    qv = -0.5 * (B + math.copysign(sq, B))
    r1 = C / qv if abs(qv) > 1e-300 else qv / A
    r2 = qv / A
    rr = [r1, r2]
    out = []
    for r in rr:
        if np.isfinite(r):
            out.append(r)
    return out

def build_hull(sl, ic):
    # sl: slopes, ic: intercepts, both 1D arrays
    # Returns: starts (array), indices into sorted (sl, ic) pairs
    m = len(sl)
    if m == 0:
        return None, None
    order = np.argsort(sl, kind='stable')
    # Group by slope, keep max intercept
    unique_idx = []
    i = 0
    while i < m:
        j = i
        while j < m and sl[order[j]] == sl[order[i]]:
            j += 1
        best = i
        for k in range(i+1, j):
            if ic[order[k]] > ic[order[best]]:
                best = k
        unique_idx.append(order[best])
        i = j
    sl_u = sl[unique_idx]
    ic_u = ic[unique_idx]
    m2 = len(sl_u)
    if m2 == 0:
        return None, None
    starts = np.empty(m2)
    starts[0] = -np.inf
    hull_sl = np.empty(m2)
    hull_ic = np.empty(m2)
    hull_sl[0] = sl_u[0]
    hull_ic[0] = ic_u[0]
    cnt = 1
    for k in range(1, m2):
        # crossing of line k with last hull line
        ds = sl_u[k] - hull_sl[cnt-1]
        if abs(ds) < 1e-30:
            # parallel, skip (shouldn't happen after unique)
            continue
        x = (hull_ic[cnt-1] - ic_u[k]) / ds
        if not np.isfinite(x):

        while cnt > 1 and x <= starts[cnt-1] + 1e-15:
            cnt -= 1
        starts[cnt] = x
        hull_sl[cnt] = sl_u[k]
        hull_ic[cnt] = ic_u[k]
        cnt += 1
    if cnt == 0:
        return None, None
    return starts[:cnt], (hull_sl[:cnt], hull_ic[:cnt])

def hull_eval(starts, hull_data, t):
    sl, ic = hull_data
    m = len(starts)
    # binary search: find largest k with starts[k] <= t
    lo, hi = 0, m - 1
    if t < starts[0]:
        return sl[0] * t + ic[0]
    while lo < hi:
        mid = (lo + hi + 1) // 2
        if starts[mid] <= t:
            lo = mid
        else:
            hi = mid - 1
    return sl[lo] * t + ic[lo]

def pair_line_minimum(h, raw, n, p, q):
    # p < q, transfer t from q to p: h[p]+=t, h[q]-=t
    if p == q:
        return None
    t_lo = max(-h[p], h[q] - 1.0)
    t_hi = min(1.0 - h[p], h[q])
    if not np.isfinite(t_lo) or not np.isfinite(t_hi):
        return None
    if t_hi - t_lo < -1e-12:
        return None
    t_lo = max(t_lo, -1.0)
    t_hi = min(t_hi, 1.0)
    if t_hi - t_lo < 1e-15:
        return 0.0
    
    gap = q - p
    nd = 2 * n - 1
    # B_d at index n-1+d in raw
    # Compute b_d = g[d,p] - g[d,q] for all d
    # g[d,r] = (1-h[r+d])*I(lo<=r<hi) - h[r-d]*I(lo+d<=r<hi+d)
    # where lo=max(0,-d), hi=min(n, n-d)
    
    b_arr = np.zeros(nd)
    for d in range(-(n-1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        g=0.0
        # term 1: (1-h[r+d]) if lo<=r<hi
        r = p
        if lo <= r < hi:
            g += (1.0 - h[r + d])
        r = q
        if lo <= r < hi:
            g -= (1.0 - h[r + d])
        # term 2: -h[r-d] if lo+d<=r<hi+d
        r = p
        if lo + d <= r < hi + d:
            g -= h[r - d]
        r = q
        if lo + d <= r < hi + d:
            g += h[r - d]
        b_arr[n - 1 + d] = g
    
    c_arr = raw  # B_d = raw[n-1+d]
    
    # Special lags: d=0 (index n-1), d=gap (index n-1+gap), d=-gap (index n-1-gap)
    special_indices = set()
    specials = []
    for d in [0, gap, -gap]:
        idx = n - 1 + d
        if 0 <= idx < nd and idx not in special_indices:
            special_indices.add(idx)
            a = 0.0
            if d == 0:
                a = -2.0
            else:
                a = 1.0
            specials.append((idx, a, b_arr[idx], c_arr[idx]))
    
    # Affine lines: all indices not in special_indices
    affine_idx = [i for i in range(nd) if i not in special_indices]
    
    candidates = [t_lo, t_hi, 0.0]
    
    if len(affine_idx) > 0:
        af_sl = b_arr[affine_idx]
        af_ic = c_arr[affine_idx]
        starts, hull_data = build_hull(af_sl, af_ic)
        if starts is not None:
            # Add hull breakpoints in [t_lo, t_hi]
            for st in starts:
                if t_lo <= st <= t_hi and np.isfinite(st):
                    candidates.append(st)
            
            # For each hull segment, find intersections with special quadratics
            m_h = len(starts)
            for seg in range(m_h):
                seg_lo = max(t_lo, starts[seg])
                seg_hi = min(t_hi, starts[seg+1] if seg + 1 < m_h else np.inf)
                if seg_lo >= seg_hi:
                    continue
                sl_s = hull_data[0][seg]
                ic_s = hull_data[1][seg]
                for (idx, a, b, c) in specials:
                    # Q(t) = a*t^2 + b*t + c, line = sl_s*t + ic_s
                    # a*t^2 + (b-sl_s)*t + (c-ic_s) = 0
                    roots = quad_roots(a, b - sl_s, c - ic_s)
                    for r in roots:
                        if seg_lo - 1e-10 <= r <= seg_hi + 1e-10:
                            r = max(seg_lo, min(seg_hi, r))
                            candidates.append(r)
        else:
            starts = None
    else:
        starts = None
    
    # Pairwise intersections of special quadratics
    for i_s in range(len(specials)):
        for j_s in range(i_s + 1, len(specials)):
            idx_i, a_i, b_i, c_i = specials[i_s]
            idx_j, a_j, b_j, c_j = specials[j_s]
            # (a_i - a_j)*t^2 + (b_i - b_j)*t + (c_i - c_j) = 0
            roots = quad_roots(a_i - a_j, b_i - b_j, c_i - c_j)
            for r in roots:
                if t_lo - 1e-10 <= r <= t_hi + 1e-10:
                    r = max(t_lo, min(t_hi, r))
                    candidates.append(r)
    
    # Vertices of convex quadratics (a>0)
    for (idx, a, b, c) in specials:
        if a > 0:
            v = -b / (2.0 * a)
            if t_lo - 1e-10 <= v <= t_hi + 1e-10:
                v = max(t_lo, min(t_hi, v))
                candidates.append(v)
    
    # Evaluate F(t) at all candidates
    best_t = 0.0
    best_f = np.inf
    for t in candidates:
        t = max(t_lo, min(t_hi, t))
        if not np.isfinite(t):
            continue
        f_val = -np.inf
        if starts is not None:
            f_val = hull_eval(starts, hull_data, t)
        for (idx, a, b, c) in specials:
            qv = a * t * t + b * t + c
            if qv > f_val:
                f_val = qv
        if f_val < best_f - 1e-14:
            best_f = f_val
            best_t = t
        elif abs(f_val - best_f) <= 1e-14:
            if abs(t) < abs(best_t):
                best_t = t
    
    return best_t

                if len(ph_arr) >= 1 and ph_arr.ndim == 1:
                    n = len(ph_arr)
                    parent_h = ph_arr.copy()
            except Exception:
                pass
    
    if n is None:
        if 'n_points' in kwargs:

                n = int(kwargs['n_points'])
            except Exception:

    parent_fallback = None

            if np.all(parent_h >= 0) and np.all(parent_h <= 1) and abs(np.sum(parent_h) - S) <= 1e-10:
                parent_fallback = parent_h.copy()

    fallbacks = [h_const]
    if parent_fallback is not None:
        fallbacks.insert(0, parent_fallback)

    incumbent_score = np.inf
    for fb in fallbacks:
        s = exact_score(fb, n)
        if s is not None and s < incumbent_score:

            incumbent = fb.copy()

    if n > 128:
