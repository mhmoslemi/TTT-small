def is_feasible(h, n):
    if h is None or len(h) != n:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-15) or np.any(h > 1 + 1e-15):
        return False
    mass = float(np.sum(h))
    s = n / 2.0
    if abs(mass - s) > 1e-10 * max(1, n):
        return False
    return True

def project_capped_simplex(z, n):
    s = n / 2.0
    if z is None or len(z) != n or not np.all(np.isfinite(z)):
        return np.full(n, 0.5)
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    lo = float(z.min()) - 1.0
    hi = float(z.max())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(z - mid, 0.0, 1.0)
        if proj.sum() > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - hi, 0.0, 1.0)
    # Residual correction
    r = s - float(np.sum(h))
    for _ in range(n + 2):
        if abs(r) < 1e-14:
            break
        if r > 0:
            slack = 1.0 - h
            idx = int(np.argmax(slack))
            move = min(r, slack[idx])
            if move < 1e-16:
                break
            h[idx] += move
            r -= move
        else:
            slack = h.copy()
            idx = int(np.argmax(slack))
            move = min(-r, slack[idx])
            if move < 1e-16:
                break
            h[idx] -= move
            r += move
    h = np.clip(h, 0.0, 1.0)
    if not is_feasible(h, n):
        return np.full(n, 0.5)
    return h

def score(h):
    n = len(h)
    if not is_feasible(h, n):
        return float('inf')
    dx = 2.0 / n
