            d = np.zeros(n)
            d[a] = 1.0
            d[b] = -1.0
        else:
            d /= norm
    if d is None:
        return None
    amp = float(rng.uniform(-rho, rho))
    if abs(amp) < 1e-15:
        return None
    cand = h + amp * d
    if float(np.min(cand)) < 0.0 or float(np.max(cand)) > 1.0:
        return None
    if not repair_mass(cand, target):
        return None
    if not is_feasible(cand, n, target):
        return None
    return cand

def run(seed=42, **kwargs):
    n = 70
    parent = None
    try:
        pv = np.array(initial_h_values, dtype=np.float64)
        if pv.ndim == 1 and 2 <= pv.size <= 999:
            n = pv.size
            if np.all(np.isfinite(pv)) and float(np.max(np.abs(pv))) <= 1e6:
                parent = pv.copy()
    except Exception:
        pass
    dx = 2.0 / n
    target = float(n) / 2.0
    fallback = np.full(n, 0.5)
    validated_parent = None
    if parent is not None:
        if float(np.min(parent)) >= -1e-10 and float(np.max(parent)) <= 1.0 + 1e-10 and abs(float(np.sum(parent)) - target) <= 1e-8:
            vp = capped_simplex_project(parent, target)
            if vp is not None:
                validated_parent = vp
    if validated_parent is not None:
        fallback = validated_parent.copy()
    rng = np.random.default_rng(seed)
    R = 8
    profiles = []
    base = validated_parent.copy() if validated_parent is not None else fallback.copy()
    if parent is not None and validated_parent is not None:
        p = validated_parent.copy()
        profiles.append(p.copy())
        profiles.append(p[::-1].copy())
        profiles.append(capped_simplex_project(p + rng.standard_normal(n) * 0.025, target))
        profiles.append(capped_simplex_project(p + rng.standard_normal(n) * 0.08, target))
        profiles.append(capped_simplex_project(1.0 - p, target))
        profiles.append(capped_simplex_project(rng.random(n), target))
        profiles.append(capped_simplex_project(rng.beta(0.5, 0.5, n), target))
        profiles.append(make_block_profile(n, int(rng.integers(0, n))))
    else:
        profiles.append(fallback.copy())
        profiles.append(make_block_profile(n, 0))
        profiles.append(capped_simplex_project(rng.random(n), target))
        profiles.append(capped_simplex_project(rng.beta(0.5, 0.5, n), target))
        profiles.append(capped_simplex_project(rng.random(n), target))
        profiles.append(capped_simplex_project(rng.beta(0.5, 0.5, n), target))
        bp = np.zeros(n)
        bp[rng.choice(n, size=n // 2, replace=False)] = 1.0
        profiles.append(capped_simplex_project(bp, target))
        profiles.append(capped_simplex_project(np.full(n, 0.5) + rng.standard_normal(n) * 0.25, target))
    replicas = []
    for prof in profiles:
        replicas.append(prof if prof is not None else fallback.copy())
    T_min, T_max = 2e-5, 0.02
    temps = np.array([T_min * (T_max / T_min) ** (r / (R - 1)) for r in range(R)])
    rhos = np.minimum(0.20, 0.015 + 0.18 * np.sqrt(temps / T_max))
    eval_count = 0
    incumbent = fallback.copy()
    inc_score = exact_score(incumbent, dx)
    eval_count += 1
    scores = []
    for r in range(R):
        s = exact_score(replicas[r], dx)
        eval_count += 1
        if s is None:
            replicas[r] = incumbent.copy()
            s = inc_score
        scores.append(s)
        if s < inc_score:
            inc_score = s
            incumbent = replicas[r].copy()
    B = max(200, min(12000, 200000000 // (n * n)))
    max_sweeps = int(4 * B / 8) + 2
    stag_limit = max(50, 2 * n)
    restart_count = 0
    stag = 0
    early_stop = (inc_score <= 0.3808)
    sweep = 0
    while eval_count < B and sweep < max_sweeps and not early_stop:
        improved = False
        for r in range(R):
            if eval_count >= B or early_stop:
                break
            cand = propose_move(replicas[r], rhos[r], rng, n, target)
            if cand is not None:
                cs = exact_score(cand, dx)
                eval_count += 1
                if cs is not None:
                    if cs < inc_score:
                        inc_score = cs
                        incumbent = cand.copy()
                        improved = True
                        if cs <= 0.3808:
                            early_stop = True
                            break
                    delta = cs - scores[r]
                    if delta <= 0:
                        replicas[r] = cand
                        scores[r] = cs
                    else:
                        u = rng.random()
                        if u == 0.0 or math.log(u) < -delta / temps[r]:
                            replicas[r] = cand
                            scores[r] = cs
        sweep += 1
        if sweep % 5 == 0:
            pairs = [(0,1),(2,3),(4,5),(6,7)] if (sweep // 5) % 2 == 0 else [(1,2),(3,4),(5,6)]
            for (ri, si) in pairs:
                L = (scores[ri] - scores[si]) * (1.0 / temps[ri] - 1.0 / temps[si])
                if L >= 0.0:
                    replicas[ri], replicas[si] = replicas[si], replicas[ri]
                    scores[ri], scores[si] = scores[si], scores[ri]
                else:
                    u = rng.random()
                    if u == 0.0 or math.log(u) < L:
                        replicas[ri], replicas[si] = replicas[si], replicas[ri]
                        scores[ri], scores[si] = scores[si], scores[ri]
        stag = 0 if improved else stag + 1
        if stag >= stag_limit and restart_count < 2 and eval_count + R < B:
            noises = [0.02, 0.04, 0.08, 0.16, 0.28]
            new_reps = [incumbent.copy()]
            for sd in noises:
                np_ = capped_simplex_project(incumbent + rng.standard_normal(n) * sd, target)
                new_reps.append(np_ if np_ is not None else incumbent.copy())
            np_ = capped_simplex_project(rng.beta(0.5, 0.5, n), target)
            new_reps.append(np_ if np_ is not None else incumbent.copy())
            if restart_count == 0:
                np_ = capped_simplex_project(rng.random(n), target)
            else:
                np_ = capped_simplex_project(make_block_profile(n, int(rng.integers(0, n))), target)
            new_reps.append(np_ if np_ is not None else incumbent.copy())
            new_scores = []
            for i in range(R):
                s = exact_score(new_reps[i], dx)
                eval_count += 1
                if s is None:
                    new_reps[i] = incumbent.copy()
                    s = inc_score
                new_scores.append(s)
                if s < inc_score:
                    inc_score = s
                    incumbent = new_reps[i].copy()
            replicas = new_reps
            scores = new_scores
            restart_count += 1
            stag = 0
    final = capped_simplex_project(incumbent, target)
    if final is None:
        final = fallback.copy()
    final_score = exact_score(final, dx)
    if final_score is None or not is_feasible(final, n, target):
        final = fallback.copy()
        final_score = exact_score(final, dx)
        if final_score is None:
            final = np.full(n, 0.5)
            final_score = float(np.max(np.correlate(final, 1.0 - final, mode='full') * dx))
    if not is_feasible(final, n, target):
        fp = capped_simplex_project(final, target)
        if fp is not None:
            final = fp
        final_score = exact_score(final, dx)
    return (final, float(final_score), n)
