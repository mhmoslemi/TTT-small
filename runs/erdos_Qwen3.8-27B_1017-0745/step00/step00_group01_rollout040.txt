    m = len(a_arr)
    tol_t = 1e-12 * max(1.0, abs(lo), abs(hi), abs(hi - lo))
    cands = [lo, hi, 0.0]
    for d in range(m):
        if abs(a_arr[d]) > 1e-14:
            sp = -b_arr[d] / (2.0 * a_arr[d])
            if sp >= lo - tol_t and sp <= hi + tol_t:
                cands.append(sp)
    for d in range(m):
        for e in range(d + 1, m):
            ad = a_arr[d] - a_arr[e]
            bd = b_arr[d] - b_arr[e]
            cd = c_arr[d] - c_arr[e]
            scale = max(1.0, abs(bd), abs(ad), abs(cd))
            if abs(ad) < 1e-14 * scale:
                if abs(bd) > 1e-14 * scale:
                    r = -cd / bd
                    if r >= lo - tol_t and r <= hi + tol_t:
                        cands.append(r)
            else:
                disc = bd * bd - 4.0 * ad * cd
                tol_d = 1e-14 * max(1.0, bd * bd, abs(4.0 * ad * cd))
                if disc < -tol_d:
                    continue
                if disc < 0:
                    disc = 0.0
                sq = math.sqrt(disc)
                q = -0.5 * (bd + math.copysign(sq, bd))
                if abs(q) > 1e-15:
                    r1 = q / ad
                    r2 = cd / q
                else:
                    r1 = -bd / (2.0 * ad)
                    r2 = r1
                for r in (r1, r2):
                    if np.isfinite(r) and r >= lo - tol_t and r <= hi + tol_t:
                        cands.append(r)
    cands = np.array(cands, dtype=np.float64)
    cands = cands[np.isfinite(cands)]
    cands = np.clip(cands, lo, hi)
    cands = np.sort(cands)
    merged = [cands[0]]
    for c in cands[1:]:
        if c - merged[-1] > tol_t:
            merged.append(c)
    cands = np.array(merged)
    top_t = [0.0]
    top_sc = [0.0]
    chunk = 512
    for i in range(0, len(cands), chunk):
        tc = cands[i:i + chunk]
        result = np.zeros(tc.shape)
        for j in range(m):
            val = a_arr[j] * tc ** 2 + b_arr[j] * tc + c_arr[j]
            np.maximum(result, val, out=result)
        finite = np.isfinite(result)
        if not np.any(finite):
            continue
        li = np.where(finite)[0]
        lv = result[li]
        lt = tc[li]
        order = np.argsort(lv)[:8]
        for idx in order:
            top_t.append(float(lt[idx]))
            top_sc.append(float(lv[idx]))
    if len(top_sc) > 32:
        pairs = list(zip(top_t, top_sc))
        pairs.sort(key=lambda p: p[1])
        top_t = [p[0] for p in pairs[:8]]
        top_sc = [p[1] for p in pairs[:8]]
    best_x = x
    best_sc = current_score
    seen = set()
    for t_val in top_t:
        key = round(t_val, 12)
        if key in seen:
            continue
        seen.add(key)
        if abs(t_val) < 1e-15:
            continue
        y = x + t_val * v
        if not feasibility_check(y, n):
            y = capped_simplex_project(y, n)
            if y is None:
                continue
        sc = exact_score(y, n)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best_x = y
            best_sc = sc
    return best_x, best_sc

def run(seed=42, **kwargs):
    n = None
    base = None
    try:
        if 'initial_h_values' in globals():
            iv = initial_h_values
            if iv is not None:
                arr = np.asarray(iv, dtype=np.float64).ravel()
                if arr.size > 0:
                    n = arr.size
                    if np.all(np.isfinite(arr)) and np.all(arr >= -1e-8) and np.all(arr <= 1.0 + 1e-8):
                        clipped = np.clip(arr, 0.0, 1.0)
                        if feasibility_check(clipped, n):
                            base = clipped.copy()
                        else:
                            base = capped_simplex_project(clipped, n)
    except Exception:
        pass
    if n is None:
        nkw = kwargs.get('n_points', 47)
        if isinstance(nkw, (int, np.integer)) and not isinstance(nkw, bool) and 1 <= nkw <= 999:
            n = int(nkw)
        else:
            n = 47
    if base is None:
        base = np.full(n, 0.5, dtype=np.float64)
        if not feasibility_check(base, n):
            base = capped_simplex_project(base, n)
            if base is None:
                base = np.full(n, 0.5, dtype=np.float64)
    fallback = base.copy()
    fallback_score = exact_score(fallback, n)
    best_global = fallback.copy()
    best_global_score = fallback_score
    rng = np.random.RandomState(seed if isinstance(seed, (int, np.integer)) and not isinstance(seed, bool) else 42)
    sn = kwargs.get('starts', None)
    sw = kwargs.get('sweeps', None)
    if n <= 48:
        max_starts = 5 if not isinstance(sn, (int, np.integer)) or isinstance(sn, bool) or sn < 1 else min(sn, 6)
        max_sweeps = 2 if not isinstance(sw, (int, np.integer)) or isinstance(sw, bool) or sw < 1 else min(sw, 3)
    elif n <= 64:
        max_starts = 2 if not isinstance(sn, (int, np.integer)) or isinstance(sn, bool) or sn < 1 else min(sn, 3)
        max_sweeps = 1 if not isinstance(sw, (int, np.integer)) or isinstance(sw, bool) or sw < 1 else min(sw, 2)
    else:
        max_starts = 2
        max_sweeps = 0
    starts_list = [base]
    def _try_add(v):
        if v is None or not feasibility_check(v, n):
            return
        for ex in starts_list:
            if np.max(np.abs(v - ex)) < 1e-10:
                return
        starts_list.append(v)
    _try_add(base[::-1].copy())
    _try_add(1.0 - base)
    _try_add(capped_simplex_project(base + 0.12 * rng.randn(n), n))
    _try_add(np.full(n, 0.5, dtype=np.float64))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    _try_add(capped_simplex_project(blk, n))
    alt = np.zeros(n)
    alt[::2] = 1.0
    _try_add(capped_simplex_project(alt, n))
    _try_add(capped_simplex_project(base + 0.35 * rng.randn(n), n))
    _try_add(capped_simplex_project(rng.rand(n), n))
    _try_add(capped_simplex_project(base + 0.45 * rng.randn(n), n))
    att = 0
    while len(starts_list) < max_starts and att < 30:
        _try_add(capped_simplex_project(base + 0.2 * rng.randn(n), n))
        att += 1
    while len(starts_list) < max_starts:
        starts_list.append(base.copy())
    starts_list = starts_list[:max_starts]
    if n > 1 and max_sweeps > 0:
        basis = build_haar_basis(n)
        n_dirs = basis.shape[0]
        if n_dirs > 0:
            for si in range(len(starts_list)):
                x = starts_list[si].copy()
                x_score = exact_score(x, n)
                if not np.isfinite(x_score):
                    continue
                for swp in range(max_sweeps):
                    perm = rng.permutation(n_dirs)
                    improved = False
                    for di in perm:
                        v = basis[di]
                        fi = feasible_interval(x, v, n)
                        if fi is None:
                            continue
                        try:
                            x_new, sc_new = line_search(x, v, n, fi[0], fi[1], x_score)
                        except Exception:
                            continue
                        if np.isfinite(sc_new) and sc_new < x_score - 1e-12:
                            x = x_new
                            x_score = sc_new
                            improved = True
                            if x_score < best_global_score - 1e-12:
                                best_global = x.copy()
                                best_global_score = x_score
                    if not improved:
                        break
    final = capped_simplex_project(best_global, n)
    if final is None or not feasibility_check(final, n):
        final = fallback
    final = np.clip(final, 0.0, 1.0)
    if not feasibility_check(final, n):
        final = fallback
    final_score = exact_score(final, n)
    if not np.isfinite(final_score):
        final = np.full(n, 0.5, dtype=np.float64)
        final_score = exact_score(final, n)
    return (final, float(final_score), int(n))
