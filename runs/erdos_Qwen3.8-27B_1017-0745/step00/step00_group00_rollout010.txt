        p = project_box_sum(u, n / 2.0)
        if is_feasible(p, n):
            starts.append(p)

    for h0 in starts:
        try:
            h1 = try_optimize(h0)
            if h1 is not None:
                s = exact_score(h1)
                if s < best_score:
                    best_h = h1.copy()
                    best_score = s
        except Exception:
            continue

    sigma = 0.05
    for _ in range(4):
        try:
            pert = best_h + sigma * rng.standard_normal(n)
            p = project_box_sum(pert, n / 2.0)
            if not is_feasible(p, n):
                break
            h1 = try_optimize(p)
            improved = False
            if h1 is not None:
                s = exact_score(h1)
                if s < best_score - 1e-12:
                    best_h = h1.copy()
                    best_score = s
                    improved = True
            if improved:
                sigma = max(0.01, 0.7 * sigma)
            else:
                sigma = min(0.25, 1.5 * sigma)
        except Exception:
            break

    best_h = project_box_sum(best_h, n / 2.0)
    if not is_feasible(best_h, n):
        best_h = parent.copy()
    best_score = exact_score(best_h)
    return (best_h, best_score, n)
