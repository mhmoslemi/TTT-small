```python
import numpy as np
from scipy.optimize import minimize

def exact_score(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_k(y, S, n):
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - float(np.sum(h))
    if abs(r) > 1e-12:
        if r > 0:
            idx = np.where(1.0 - h > 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) > 0:
            h[idx] += r / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def feasible(h, n, S):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-9) and np.all(h <= 1.0 + 1e-9)
            and abs(float(np.sum(h)) - S) < 1e-7)

def lag_scores(h, dx, n):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_val(q, tau):
    m = float(np.max(q))
    w = np.exp((q - m) / tau)
    return m + tau * float(np.log(np.sum(w)))

def smooth_w(q, tau):
    m = float(np.max(q))
    w = np.exp((q - m) / tau)
    return w / float(np.sum(w))

def smooth_grad(h, tau, dx, n):
    q = lag_scores(h, dx, n)
    p = smooth_w(q, tau)
    g = np.zeros(n)
    for ii in range(len(p)):
        if p[ii] < 1e-14:
            continue
        ell = ii - (n - 1)
        il = max(0, -ell)
        ih = min(n, n - ell)
        a = h[il:ih]
        b = h[il + ell:ih + ell]
        g[il + ell:ih + ell] += p[ii] * dx * (1.0 - a)
        g[il:ih] -= p[ii] * dx * b
    return g

def cont_opt(h0, tau_list, dx, n, S, max_iter=80, max_ls=25):
    h = h0.copy()
    for tau in tau_list:
        alpha = 1.0
        acc = 0
        while acc < max_iter:
            q = lag_scores(h, dx, n)
            f = smooth_val(q, tau)
            g = smooth_grad(h, tau, dx, n)
            if not np.all(np.isfinite(g)):
                break
            a = alpha
            found = False
            for _ in range(max_ls):
                y = project_k(h - a * g, S, n)
                d = y - h
                if np.max(np.abs(d)) < 1e-10:
                    break
                gd = float(np.dot(g, d))
                if gd > -1e-12:
                    break
                fy = smooth_val(lag_scores(y, dx, n), tau)
                if fy <= f + 1e-4 * gd + 1e-14:
                    h = y
                    alpha = min(1.25 * a, 5.0)
                    acc += 1
                    found = True
                    break
                a *= 0.5
            if not found:
                break
    return h

def slsqp_obj(z, n):
    return z[n]

def slsqp_obj_jac(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def slsqp_ineq(z, n, dx):
    return z[n] - lag_scores(z[:n], dx, n)

def slsqp_ineq_jac(z, n, dx):
    h = z[:n]
    J = np.zeros((2 * n - 1, n + 1))
    for ii in range(2 * n - 1):
        ell = ii - (n - 1)
        il = max(0, -ell)
        ih = min(n, n - ell)
        a = h[il:ih]
        b = h[il + ell:ih + ell]
        J[ii, il + ell:ih + ell] -= dx * (1.0 - a)
        J[ii, il:ih] += dx * b
        J[ii, n] = 1.0
    return J

def slsqp_eq(z, n, S):
    return float(np.sum(z[:n])) - S

def slsqp_eq_jac(z, n, S):
    g = np.ones(n + 1)
    g[n] = 0.0
    return g

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = None
    pv = globals().get('initial_h_values')
    if pv is not None:
        pv = np.asarray(pv, dtype=np.float64)
        if pv.ndim == 1 and 2 <= len(pv) <= 999 and np.all(np.isfinite(pv)) and np.all(np.abs(pv) < 1e6):
            parent = pv.copy()
    if parent is not None:
        n = len(parent)
    else:
        n = int(kwargs.get('n_points', 45))
        if not (2 <= n <= 999):
            n = 45
    dx = 2.0 / n
    S = n / 2.0

    if parent is not None:
        fallback = project_k(parent, S, n)
    else:
        fallback = np.full(n, 0.5)
    best_h = fallback.copy()
    best_s = exact_score(best_h, dx)

    starts = [fallback.copy()]
    if parent is not None:
        starts.append(project_k(1.0 - fallback, S, n))
        starts.append(project_k(fallback[::-1].copy(), S, n))
        starts.append(project_k(1.0 - fallback[::-1].copy(), S, n))
        starts.append(project_k(fallback + 0.08 * rng.standard_normal(n), S, n))
        starts.append(project_k(fallback + 0.25 * rng.standard_normal(n), S, n))
    starts.append(project_k(rng.uniform(0.0, 1.0, n), S, n))
    starts.append(project_k(rng.beta(0.5, 0.5, n), S, n))

    for s in starts:
        sc = exact_score(s, dx)
        if sc < best_s - 1e-12:
            best_h = s.copy()
            best_s = sc

    if n <= 100:
        tau_list = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00001]
    else:
        tau_list = [0.02, 0.005, 0.001, 0.0001, 0.00001]

    local_cands = []
    for s in starts:
        h_opt = cont_opt(s, tau_list, dx, n, S)
        local_cands.append(h_opt.copy())
        sc = exact_score(h_opt, dx)
        if sc < best_s - 1e-12:
            best_h = h_opt.copy()
            best_s = sc

    n_pert = 4 if n <= 100 else 1
    for i in range(n_pert):
        sigma = [0.04, 0.12, 0.30, 0.50][i]
        z = project_k(best_h + sigma * rng.standard_normal(n), S, n)
        h_opt = cont_opt(z, tau_list, dx, n, S)
        sc = exact_score(h_opt, dx)
        if sc < best_s - 1e-12:
            best_h = h_opt.copy()
            best_s = sc

    if n <= 100:
        all_cands = local_cands + [best_h]
        scores = [exact_score(c, dx) for c in all_cands]
        order = sorted(range(len(all_cands)), key=lambda i: scores[i])
        sel = []
        for idx in order:
            if len(sel) >= 3:
                break
            distinct = True
            for s2 in sel:
                if np.max(np.abs(all_cands[idx] - all_cands[s2])) < 1e-5:
                    distinct = False
                    break
            if distinct:
                sel.append(idx)
        if len(sel) == 0:
            sel = [order[0]]
        for idx in sel:
            h0 = all_cands[idx]
            t0 = min(1.0, exact_score(h0, dx) + 1e-8)
            z0 = np.concatenate([h0, [t0]])
            bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
            cons = [
                {'type': 'ineq', 'fun': lambda z, n=n, dx=dx: slsqp_ineq(z, n, dx),
                 'jac': lambda z, n=n, dx=dx: slsqp_ineq_jac(z, n, dx)},
                {'type': 'eq', 'fun': lambda z, n=n, S=S: slsqp_eq(z, n, S),
                 'jac': lambda z, n=n, S=S: slsqp_eq_jac(z, n, S)}
            ]
            try:
                res = minimize(slsqp_obj, z0, args=(n,), jac=lambda z, n=n: slsqp_obj_jac(z, n),
                               bounds=bounds, constraints=cons, method='SLSQP',
                               options={'maxiter': 250, 'ftol': 1e-10, 'disp': False})
                if res.success or res.x is not None:
                    h_sol = project_k(res.x[:n], S, n)
                    if feasible(h_sol, n, S):
                        sc = exact_score(h_sol, dx)
                        if sc < best_s - 1e-12:
                            best_h = h_sol.copy()
                            best_s = sc
            except Exception:
                pass

    best_h = project_k(best_h, S, n)
    best_h = np.clip(best_h, 0.0, 1.0)
    if not feasible(best_h, n, S):
        best_h = fallback.copy()
    final_s = exact_score(best_h, dx)
    return (best_h, final_s, n)
```
