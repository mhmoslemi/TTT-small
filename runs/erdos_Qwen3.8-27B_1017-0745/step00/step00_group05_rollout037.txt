        profs = [fallback.copy(), blk, rng.uniform(0, 1, n), rng.uniform(0, 1, n),
                 rng.beta(0.5, 0.5, n), rng.beta(0.5, 0.5, n),
                 rng.integers(0, 2, n).astype(np.float64), 0.5 + rng.normal(0, 0.25, n)]
    for prof in profs:
        p = capped_simplex_project(prof, target, n)
        if p is None:
            p = fallback.copy()
        s = exact_score(p, n, dx)
        if s is None:
            s = fb_s
        states.append(p); scores.append(s)
        if s < inc_s:
            incumbent = p.copy(); inc_s = s
    T_min = 2e-5; T_max = 0.02
    temps = np.array([T_min * (T_max / T_min) ** (r / 7.0) for r in range(R)])
    radii = np.array([min(0.20, 0.015 + 0.18 * math.sqrt(temps[r] / T_max)) for r in range(R)])
    B = max(200, min(12000, int(200000000 // (n * n))))
    ev = R + 1; mx_att = 4 * B; mx_sw = (4 * B) // R + 1
    stag = 0; mx_stag = max(50, 2 * n); nrest = 0; stop = inc_s <= 0.3808
    sw = 0; att = 0
    while not stop and att < mx_att and sw < mx_sw and ev < B:
        imp = False
        for r in range(R):
            if stop or att >= mx_att or sw >= mx_sw or ev >= B:
                break
            cand = propose_move(states[r], n, radii[r], rng)
            att += 1
            if cand is None:
                continue
            cs = exact_score(cand, n, dx); ev += 1
            if cs is None:
                continue
            if cs < inc_s:
                incumbent = cand.copy(); inc_s = cs; imp = True
                if inc_s <= 0.3808:
                    stop = True; break
            dlt = cs - scores[r]
            if dlt <= 0:
                states[r] = cand; scores[r] = cs
            else:
                u = rng.random()
                if u == 0 or math.log(u) < -dlt / temps[r]:
                    states[r] = cand; scores[r] = cs
        sw += 1
        stag = 0 if imp else stag + 1
        if sw % 5 == 0 and not stop:
            par = (sw // 5) % 2
            pairs = [(0,1),(2,3),(4,5),(6,7)] if par == 0 else [(1,2),(3,4),(5,6)]
            for a, b in pairs:
                L = (scores[a] - scores[b]) * (1.0 / temps[a] - 1.0 / temps[b])
                if L >= 0 or (rng.random() > 0 and math.log(rng.random()) < L):
                    states[a], states[b] = states[b].copy(), states[a].copy()
                    scores[a], scores[b] = scores[b], scores[a]
        if stag >= mx_stag and nrest < 2 and ev + R < B:
            ns = []; nsc = []
            noises = [0.02, 0.04, 0.08, 0.16, 0.28]
            for j in range(R):
                if j == 0:
                    pr = incumbent.copy()
                elif j <= 5:
                    pr = incumbent + rng.normal(0, noises[j - 1], n)
                elif j == 6:
                    pr = rng.beta(0.5, 0.5, n)
                else:
                    if nrest == 0:
                        pr = rng.uniform(0, 1, n)
                    else:
                        bl2 = np.zeros(n); bl2[:n // 2] = 1.0
                        if n % 2: bl2[n // 2] = 0.5
                        pr = np.roll(bl2, int(rng.integers(0, n)))
                p = capped_simplex_project(pr, target, n)
                if p is None:
                    p = incumbent.copy()
                s = exact_score(p, n, dx); ev += 1
                if s is None:
                    s = inc_s
                ns.append(p); nsc.append(s)
                if s < inc_s:
                    incumbent = p.copy(); inc_s = s
            states = ns; scores = nsc; nrest += 1; stag = 0
            stop = inc_s <= 0.3808
    fh = capped_simplex_project(incumbent, target, n)
    if fh is None:
        fh = fallback.copy()
    c5 = exact_score(fh, n, dx)
    if c5 is None:
        fh = fallback.copy(); c5 = exact_score(fh, n, dx)
        if c5 is None:
            c5 = 0.0
    fh = np.clip(fh, 0.0, 1.0)
    c5 = exact_score(fh, n, dx)
    if c5 is None:
        c5 = 0.0
    return (fh, float(c5), n)
