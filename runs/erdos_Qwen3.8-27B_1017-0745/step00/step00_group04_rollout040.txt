Key mathematical structure:
- n points, dx = 2/n, sum(h) = n/2
- C5 = max_k ∫ h(x)(1-h(x+k)) dx over all shifts k
- Discretized: C5 = max(np.correlate(h, 1-h, 'full') * dx)
- For pair exchange h -> h + t*(e_i - e_j):
  - q[0] = dx * sum h[i](1-h[i]) is quadratic in t (from h_i(1-h_i) and h_j(1-h_j))
  - q[2s-1] and q[2s] for s=j-i are quadratic
  - All other lags are affine in t

import numpy as np
from math import fsum, copysign, sqrt, pi, isfinite

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(corr * dx))
    if not isfinite(val):
        return float('inf')
    return val

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-15) or np.any(h > 1.0 + 1e-15):
        return False
    S = n / 2.0
    if abs(np.sum(h) - S) > 1e-10 or abs(fsum(float(x) for x in h) - S) > 1e-10:
        return False
    return True

def restore_feasible(v, n):
    S = n / 2.0
    v = np.clip(v, -1e6, 1e6)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        hc = np.clip(v - mid, 0.0, 1.0)
        if np.sum(hc) > S:
            lo = mid
        else:
            hi = mid
    hc = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    residual = S - fsum(float(x) for x in hc)
    if residual > 0:
        slack = 1.0 - hc
        for idx in np.argsort(-slack):
            add = min(residual, float(slack[idx]))
            if add > 1e-15:
                hc[idx] += add
                residual -= add
                if abs(residual) < 1e-12:
                    break
    elif residual < 0:
        slack = hc.copy()
        for idx in np.argsort(-slack):
            rem = min(-residual, float(slack[idx]))
            if rem > 1e-15:
                hc[idx] -= rem
                residual += rem
                if abs(residual) < 1e-12:
                    break
    for idx in range(n):
        if 1e-15 < hc[idx] < 1.0 - 1e-15:
            others = fsum(float(x) for k, x in enumerate(hc) if k != idx)
            hc[idx] = S - others
            if 0.0 <= hc[idx] <= 1.0 and is_feasible(hc, n):
                return hc
    if is_feasible(hc, n):
        return hc
    return None

def quadratic_roots(A, B, C, lo, hi):
    tol = 1e-14 * max(1.0, abs(A), abs(B), abs(C))
    roots = []
    if abs(A) <= tol:
        if abs(B) > tol:
            r = -C / B
            if lo - 1e-10 <= r <= hi + 1e-10 and isfinite(r):
                roots.append(min(max(r, lo), hi))
        return roots
    disc = B * B - 4.0 * A * C
    disc_tol = -64.0 * np.finfo(float).eps * (B * B + abs(4.0 * A * C) + 1.0)
    if disc < disc_tol:
        return roots
    if disc < 0:
        disc = 0.0
    sd = sqrt(disc)
    z = -0.5 * (B + copysign(sd, B))
    if abs(z) > 1e-30:
        r1 = z / A
        r2 = C / z
    else:
        r1 = -B / (2.0 * A)
        r2 = r1
    for r in (r1, r2):
        if lo - 1e-10 <= r <= hi + 1e-10 and isfinite(r):
            roots.append(min(max(r, lo), hi))
    return roots

def affine_hull(slopes, intercepts, lo, hi):
    # Returns list of (start_t, slope, intercept) for upper envelope
    order = np.lexsort((-intercepts, slopes))
    hull = []  # list of (x_start, m, c)
    for idx in order:
        m = float(slopes[idx])
        c = float(intercepts[idx])
        if not (isfinite(m) and isfinite(c)):
            continue
        if hull:
            # merge if slope very close
            if abs(m - hull[-1][1]) < 1e-14 * max(1.0, abs(m), abs(hull[-1][1])):
                if c > hull[-1][2]:
                    hull[-1] = (hull[-1][0], m, c)
                continue
        while hull:
            x0, m0, c0 = hull[-1]
            m_new, c_new = m, c
            if m_new <= m0:
                break
            x_cross = (c0 - c_new) / (m_new - m0)
            if x_cross <= x0 + 1e-15:
                hull.pop()
            else:
                hull.append((x_cross, m_new, c_new))
                break
        else:
            hull.append((lo, m, c))
    # Clip to [lo, hi]
    segs = []
    for x0, m, c in hull:
        xs = max(x0, lo)
        xe = hi
        if xs < xe:
            segs.append((xs, xe, m, c))
    if not segs:
        segs.append((lo, hi, 0.0, 0.0))
    return segs

def pair_line_search(h, i, j, n, cur_score):
    dx = 2.0 / n
    s = j - i
    # Build q and G
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    # q[0] = dx * sum h[k](1-h[k])
    q[0] = dx * np.sum(h * (1.0 - h))
    for k in range(n):
        G[0, k] = dx * (1.0 - 2.0 * h[k])
    for d in range(1, n):
        # q[2d-1] = dx * sum_{i=0}^{n-d-1} h[i+d]*(1-h[i])
        # q[2d] = dx * sum_{i=0}^{n-d-1} h[i]*(1-h[i+d])
        idx = np.arange(n - d)
        hp = h[idx + d]
        hm = h[idx]
        q[2 * d - 1] = dx * np.sum(hp * (1.0 - hm))
        q[2 * d] = dx * np.sum(hm * (1.0 - hp))
        G[2 * d - 1, idx + d] += dx * (1.0 - hm)
        G[2 * d - 1, idx] += -dx * hp
        G[2 * d, idx] += dx * (1.0 - hp)
        G[2 * d, idx + d] += -dx * hm
    # For pair (i,j): a[l], b[l], c[l]
    a = np.zeros(M)
    a[0] = -2.0 * dx
    a[2 * s - 1] = dx
    a[2 * s] = dx
    b = G[:, i] - G[:, j]
    c = q.copy()
    # Feasible interval
    lo_t = max(-h[i], h[j] - 1.0)
    hi_t = min(1.0 - h[i], h[j])
    if lo_t >= hi_t:
        return None
    # Curved terms: indices 0, 2s-1, 2s
    curved = [0, 2 * s - 1, 2 * s]
    affine_idx = [l for l in range(M) if l not in curved]
    if not affine_idx:
        affine_idx = []
    # Build affine hull for affine terms
    if affine_idx:
        aff_slopes = b[affine_idx]
        aff_intercepts = c[affine_idx]
        segs = affine_hull(aff_slopes, aff_intercepts, lo_t, hi_t)
    else:
        segs = [(lo_t, hi_t, 0.0, 0.0)]
    # Candidates
    cands = [lo_t, hi_t, 0.0]
    for seg in segs:
        cands.append(seg[0])
        cands.append(seg[1])
    # Quadratic-line roots
    for cl in curved:
        for (xs, xe, m_line, c_line) in segs:
            A = a[cl]
            B = b[cl] - m_line
            C = c[cl] - c_line
            roots = quadratic_roots(A, B, C, xs, xe)
            cands.extend(roots)
    # Quadratic-quadratic roots
    for p in range(3):
        for q2 in range(p + 1, 3):
            A = a[curved[p]] - a[curved[q2]]
            B = b[curved[p]] - b[curved[q2]]
            C = c[curved[p]] - c[curved[q2]]
            roots = quadratic_roots(A, B, C, lo_t, hi_t)
            cands.extend(roots)
    # Vertex of positive-curvature quadratics
    for cl in curved:
        if a[cl] > 1e-15:
            tv = -b[cl] / (2.0 * a[cl])
            if lo_t - 1e-10 <= tv <= hi_t + 1e-10:
                cands.append(min(max(tv, lo_t), hi_t))
    # Deduplicate
    cands = [t for t in cands if isfinite(t) and lo_t - 1e-10 <= t <= hi_t + 1e-10]
    cands = [min(max(t, lo_t), hi_t) for t in cands]
    cands.sort()
    deduped = []
    for t in cands:
        if not deduped or abs(t - deduped[-1]) > 1e-12:
            deduped.append(t)
    # Score candidates
    best_t = 0.0
    best_score = cur_score
    for t in deduped[:8]:
        new_i = h[i] + t
        new_j = h[j] - t
        if new_i < -1e-12 or new_i > 1.0 + 1e-12 or new_j < -1e-12 or new_j > 1.0 + 1e-12:
            continue
        trial = h.copy()
        trial[i] = min(max(new_i, 0.0), 1.0)
        trial[j] = min(max(new_j, 0.0), 1.0)
        if not is_feasible(trial, n):
            fixed = restore_feasible(trial, n)
            if fixed is None:
                continue
            trial = fixed
        sc = exact_score(trial)
        if sc < best_score - 1e-12:
            best_score = sc
            best_t = t
    if best_score < cur_score - 1e-12:
        new_i = h[i] + best_t
        new_j = h[j] - best_t
        trial = h.copy()
        trial[i] = min(max(new_i, 0.0), 1.0)
        trial[j] = min(max(new_j, 0.0), 1.0)
        if not is_feasible(trial, n):
            fixed = restore_feasible(trial, n)
            if fixed is None:
                return h
            trial = fixed
        final_sc = exact_score(trial)
        if final_sc < cur_score - 1e-12 and is_feasible(trial, n):
            return trial
    return h

def project_to_simplex(v, n):
    """Project v onto {0<=h<=1, sum(h)=n/2}"""
    return restore_feasible(v.astype(float), n)

import numpy as np
from math import fsum, copysign, sqrt, pi, isfinite

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(corr * dx))
    return val if isfinite(val) else float('inf')

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return False
    S = n / 2.0
    if abs(float(np.sum(h)) - S) > 1e-10 or abs(fsum(float(x) for x in h) - S) > 1e-10:
        return False
    return True

def restore_feasible(v, n):
    S = n / 2.0
    v = np.clip(v.astype(float), -1e6, 1e6)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        hc = np.clip(v - mid, 0.0, 1.0)
        if float(np.sum(hc)) > S:
            lo = mid
        else:
            hi = mid
    hc = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    residual = S - fsum(float(x) for x in hc)
    if residual > 0:
        order = np.argsort(-(1.0 - hc))
        for idx in order:
            add = min(residual, float(1.0 - hc[idx]))
            if add > 1e-15:
                hc[idx] += add
                residual -= add
                if abs(residual) < 1e-12:
                    break
    elif residual < 0:
        order = np.argsort(-hc)
        for idx in order:
            rem = min(-residual, float(hc[idx]))
            if rem > 1e-15:
                hc[idx] -= rem
                residual += rem
                if abs(residual) < 1e-12:
                    break
    for idx in range(n):
        if 1e-14 < hc[idx] < 1.0 - 1e-14:
            others = fsum(float(x) for k, x in enumerate(hc) if k != idx)
            val = S - others
            if 0.0 <= val <= 1.0:
                hc2 = hc.copy()
                hc2[idx] = val
                if is_feasible(hc2, n):
                    return hc2
    if is_feasible(hc, n):
        return hc
    return None

def quad_roots(A, B, C, lo, hi):
    tol = 1e-14 * max(1.0, abs(A), abs(B), abs(C))
    roots = []
    if abs(A) <= tol:
        if abs(B) > tol:
            r = -C / B
            if lo - 1e-10 <= r <= hi + 1e-10 and isfinite(r):
                roots.append(min(max(r, lo), hi))
        return roots
    disc = B * B - 4.0 * A * C
    if disc < -64.0 * np.finfo(float).eps * (B * B + abs(4.0 * A * C) + 1.0):
        return roots
    if disc < 0:
        disc = 0.0
    sd = sqrt(disc)
    z = -0.5 * (B + copysign(sd, B))
    if abs(z) > 1e-30:
        r1, r2 = z / A, C / z
    else:
        r1 = r2 = -B / (2.0 * A)
    for r in (r1, r2):
        if lo - 1e-10 <= r <= hi + 1e-10 and isfinite(r):
            roots.append(min(max(r, lo), hi))
    return roots

def build_affine_hull(slopes, intercepts, lo, hi):
    order = np.lexsort((-intercepts, slopes))
    hull = []
    for idx in order:
        m = float(slopes[idx])
        c = float(intercepts[idx])
        if not (isfinite(m) and isfinite(c)):
            continue
        if hull and abs(m - hull[-1][1]) < 1e-14 * max(1.0, abs(m), abs(hull[-1][1])):
            if c > hull[-1][2]:
                hull[-1] = (hull[-1][0], m, c)
            continue
        while hull:
            x0, m0, c0 = hull[-1]
            if m <= m0:
                break
            xc = (c0 - c) / (m - m0)
            if xc <= x0 + 1e-15:
                hull.pop()
            else:
                hull.append((xc, m, c))
                break
        else:
            hull.append((lo, m, c))
    segs = []
    for x0, m, c in hull:
        xs = max(x0, lo)
        if xs < hi:
            segs.append((xs, hi, m, c))
    if not segs:
        segs = [(lo, hi, 0.0, 0.0)]
    return segs

def pair_line_search(h, i, j, n, cur_score):
    dx = 2.0 / n
    s = j - i
    M = 2 * n - 1
    q = np.zeros(M)
    G = np.zeros((M, n))
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        idx = np.arange(n - d)
        hp, hm = h[idx + d], h[idx]
        q[2 * d - 1] = dx * np.sum(hp * (1.0 - hm))
        q[2 * d] = dx * np.sum(hm * (1.0 - hp))
        G[2 * d - 1, idx + d] += dx * (1.0 - hm)
        G[2 * d - 1, idx] += -dx * hp
        G[2 * d, idx] += dx * (1.0 - hp)
        G[2 * d, idx + d] += -dx * hm
    a = np.zeros(M)
    a[0] = -2.0 * dx
    a[2 * s - 1] = dx
    a[2 * s] = dx
    b = G[:, i] - G[:, j]
    c = q.copy()
    lo_t = max(-h[i], h[j] - 1.0)
    hi_t = min(1.0 - h[i], h[j])
    if lo_t >= hi_t - 1e-15:
        return None
    curved = (0, 2 * s - 1, 2 * s)
    aff_mask = np.ones(M, dtype=bool)
    aff_mask[list(curved)] = False
    aff_idx = np.where(aff_mask)[0]
    if len(aff_idx) > 0:
        segs = build_affine_hull(b[aff_idx], c[aff_idx], lo_t, hi_t)
    else:
        segs = [(lo_t, hi_t, 0.0, 0.0)]
    cands = [lo_t, hi_t, 0.0]
    for seg in segs:
        cands.append(seg[0])
        cands.append(seg[1])
    for cl in curved:
        for (xs, xe, ml, cl_) in segs:
            cands.extend(quad_roots(a[cl], b[cl] - ml, c[cl] - cl_, xs, xe))
        for cl2 in curved:
            if cl2 <= cl:
                continue
            cands.extend(quad_roots(a[cl] - a[cl2], b[cl] - b[cl2], c[cl] - c[cl2], lo_t, hi_t))
        if a[cl] > 1e-15:
            tv = -b[cl] / (2.0 * a[cl])
            if lo_t - 1e-10 <= tv <= hi_t + 1e-10 and isfinite(tv):
                cands.append(min(max(tv, lo_t), hi_t))
    cands = [t for t in cands if isfinite(t) and lo_t - 1e-10 <= t <= hi_t + 1e-10]
    cands = sorted(set(min(max(t, lo_t), hi_t) for t in cands))
    deduped = []
    for t in cands:
        if not deduped or abs(t - deduped[-1]) > 1e-12:
            deduped.append(t)
    best_t, best_sc = 0.0, cur_score
    for t in deduped[:8]:
        ni, nj = h[i] + t, h[j] - t
        if ni < -1e-12 or ni > 1.0 + 1e-12 or nj < -1e-12 or nj > 1.0 + 1e-12:
            continue
        trial = h.copy()
        trial[i], trial[j] = min(max(ni, 0.0), 1.0), min(max(nj, 0.0), 1.0)
        if not is_feasible(trial, n):
            fx = restore_feasible(trial, n)
            if fx is None:
                continue
            trial = fx
        sc = exact_score(trial)
        if sc < best_sc - 1e-12:
            best_sc, best_t = sc, t
    if best_sc < cur_score - 1e-12:
        trial = h.copy()
        trial[i] = min(max(h[i] + best_t, 0.0), 1.0)
        trial[j] = min(max(h[j] - best_t, 0.0), 1.0)
        if not is_feasible(trial, n):
            fx = restore_feasible(trial, n)
            if fx is None:
                return h
            trial = fx
        if is_feasible(trial, n) and exact_score(trial) < cur_score - 1e-12:
            return trial
    return h
