import numpy as np
from scipy.optimize import minimize

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(v - tau, 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-15:
        for i in np.argsort(1.0 - y):
            d = min(r, 1.0 - y[i])
            y[i] += d
            r -= d
            if r <= 0:
                break
    elif r < -1e-15:
        for i in np.argsort(y):
            d = min(-r, y[i])
            y[i] -= d
            r += d
            if r >= 0:
                break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-9) or np.any(y > 1.0 + 1e-9):
        return None
    if abs(float(np.sum(y)) - S) > 1e-10 * max(1, len(y)):
        return None
    return y

def check_feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(float(np.sum(h)) - 0.5 * n) < 1e-10 * max(1, n))

def exact_score(h):
    n = len(h)
    if not check_feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_Q_G(h, n):
    nv = 2 * n - 1
    Q = np.zeros(nv)
    G = np.zeros((nv, n))
    for idx, s in enumerate(range(-(n - 1), n)):
        lo_i = max(0, -s)
        hi_i = min(n, n - s)
        for i in range(lo_i, hi_i):
            j = i + s
            Q[idx] += h[i] * (1.0 - h[j])
            G[idx, i] += (1.0 - h[j])
            G[idx, j] -= h[i]
    return Q, G

def slsq_obj(z):
    return z[-1]

def slsq_obj_grad(z):
    return np.r_[np.zeros(len(z) - 1), 1.0]

def slsq_ineq(z):
    n = len(z) - 1
    Q, _ = lag_Q_G(z[:n], n)
    return z[-1] - Q / n

def slsq_ineq_grad(z):
    n = len(z) - 1
    _, G = lag_Q_G(z[:n], n)
    nv = 2 * n - 1
    J = np.zeros((nv, n + 1))
    J[:, :n] = -G / n
    J[:, n] = 1.0
    return J

def slsq_eq(z):
    n = len(z) - 1
    return float(np.sum(z[:n])) / n - 0.5

def slsq_eq_grad(z):
    n = len(z) - 1
    g = np.zeros(n + 1)
    g[:n] = 1.0 / n
    return g

def _get_parent():
    n = 45
    if 'initial_h_values' in globals():
        ph = np.asarray(globals()['initial_h_values'], dtype=np.float64)
        if ph.ndim == 1 and 2 <= len(ph) <= 999 and np.all(np.isfinite(ph)):
            if np.all(ph >= -1e-6) and np.all(ph <= 1.0 + 1e-6):
                n = len(ph)
                p = project_box_sum(np.clip(ph, 0.0, 1.0), 0.5 * n)
                if p is not None:
                    return p, n
    return project_box_sum(np.full(n, 0.5), 0.5 * n), n

def _build_starts(parent, n, rng):
    starts = [parent.copy(), parent[::-1].copy()]
    sds = [0.01, 0.02, 0.05, 0.10, 0.15, 0.25, 0.08, 0.04]
    for sd in sds:
        p = project_box_sum(parent + sd * rng.standard_normal(n), 0.5 * n)
        if p is not None:
            starts.append(p)
    for _ in range(4):
        p = project_box_sum(rng.random(n), 0.5 * n)
        if p is not None:
            starts.append(p)
    for (A, m, ph) in [(0.3, 1, 0.5), (0.4, 2, 1.2), (0.2, 3, 2.1)]:
        u = 0.5 + A * np.cos(2.0 * np.pi * m * np.arange(n) / n + ph)
        p = project_box_sum(u, 0.5 * n)
        if p is not None:
            starts.append(p)
    return starts

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent, n = _get_parent()
    best_h = parent.copy()
    best_c5 = exact_score(parent)

    bounds = [(0.0, 1.0)] * (n + 1)
    cons = [{'type': 'ineq', 'fun': slsq_ineq, 'jac': slsq_ineq_grad},
            {'type': 'eq', 'fun': slsq_eq, 'jac': slsq_eq_grad}]
    opts = {'maxiter': 400, 'ftol': 1e-12, 'disp': False}

    starts = _build_starts(parent, n, rng)

    for h0 in starts:
        Q0, _ = lag_Q_G(h0, n)
        t0 = min(1.0, float(np.max(Q0)) / n + 2e-4)
        t0 = max(t0, 1e-6)
        z0 = np.append(h0, t0)
        try:
            res = minimize(slsq_obj, z0, jac=slsq_obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons, options=opts)
            h_new = np.array(res.x[:n], dtype=np.float64)
            h_new = project_box_sum(h_new, 0.5 * n)
            if h_new is not None and check_feasible(h_new, n):
                c = exact_score(h_new)
                if c < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c
        except Exception:
            pass

    sigma = 0.04
    for _ in range(6):
        h0 = project_box_sum(best_h + sigma * rng.standard_normal(n), 0.5 * n)
        if h0 is None:
            break
        Q0, _ = lag_Q_G(h0, n)
        t0 = min(1.0, float(np.max(Q0)) / n + 2e-4)
        t0 = max(t0, 1e-6)
        z0 = np.append(h0, t0)
        improved = False
        try:
            res = minimize(slsq_obj, z0, jac=slsq_obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons, options=opts)
            h_new = np.array(res.x[:n], dtype=np.float64)
            h_new = project_box_sum(h_new, 0.5 * n)
            if h_new is not None and check_feasible(h_new, n):
                c = exact_score(h_new)

    if best_h is not None and check_feasible(best_h, n):
        best_c5 = exact_score(best_h)
        return (best_h, best_c5, n)
    fb = project_box_sum(parent, 0.5 * n)
    if fb is None:
        fb = np.full(n, 0.5)
    return (fb, exact_score(fb), n)
