        if all(np.max(np.abs(s - u)) >= 1e-10 for u in unique):
            unique.append(s)
    return unique

def pair_search(x, n, max_sweeps=3, tol=1e-12):
    target = n / 2.0
    steps = [0.04, 0.015, 0.005, 0.0015, 0.0005, 0.0001]
    cs = exact_score(x)
    for step in steps:
        improved = False
        for _ in range(max_sweeps):
            best_p = None
            best_s = cs
            for i in range(n):
                for j in range(i + 1, n):
                    a = min(step, 1.0 - x[i], x[j])
                    if a > 1e-12:
                        x[i] += a; x[j] -= a
                        xp = np.clip(x, 0.0, 1.0)
                        sc = exact_score(xp)
                        if sc < best_s - tol:
                            best_s = sc; best_p = xp.copy()
                        x[i] -= a; x[j] += a
                    b = min(step, 1.0 - x[j], x[i])
                    if b > 1e-12:
                        x[j] += b; x[i] -= b
                        xp = np.clip(x, 0.0, 1.0)
                        sc = exact_score(xp)
                        if sc < best_s - tol:
                            best_s = sc; best_p = xp.copy()
                        x[j] -= b; x[i] += b
            if best_p is not None:
                x = capped_simplex_project(best_p, n, target)
                cs = exact_score(x)
                improved = True
            else:
                break
    return x, cs

def run(seed=42, **kwargs):
    global _N
    try:
        seed = int(seed)
    except (ValueError, TypeError):
        seed = 42
    rng = np.random.default_rng(seed)

    n = None
    base = None
    parent_fb = None
    try:
        iv = np.array(initial_h_values, dtype=np.float64).ravel()
        if len(iv) > 0 and np.all(np.isfinite(iv)):
            n = int(len(iv))
            if iv.min() >= -1e-8 and iv.max() <= 1.0 + 1e-8:
                base = capped_simplex_project(iv, n, n / 2.0)
                parent_fb = base.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
            if n < 1 or n > 999:
                n = 47
        except Exception:
            n = 47
        base = np.full(n, 0.5)
    target = n / 2.0

    if n <= 64:
        nrestart, nmaxit = 12, 250
    elif n <= 256:
        nrestart, nmaxit = 6, 180
    elif n <= 999:
        nrestart, nmaxit = 2, 100
    else:
        nrestart, nmaxit = 0, 0
    try:
        nrestart = min(int(kwargs.get('restarts', nrestart)), 12)
    except Exception:
        pass
    try:
        nmaxit = min(int(kwargs.get('maxiter', nmaxit)), 500)
    except Exception:
        pass
    try:
        psw = min(int(kwargs.get('pair_sweeps', 3)), 5)
    except Exception:
        psw = 3

    starts = make_starts(base, n, rng)
    best = base.copy()
    best_sc = exact_score(best)

    if nrestart > 0 and nmaxit > 0:
        _N = n
        dx = 2.0 / n
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': _ineq_fun, 'jac': _ineq_jac},
            {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac},
        ]
        for idx in range(min(nrestart, len(starts))):
            h0 = starts[idx]
            t0 = min(2.0, exact_score(h0) + 1e-6)
            z0 = np.concatenate([h0, [t0]])
            try:
                res = minimize(_obj, z0, jac=_obj_jac, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'ftol': 1e-10, 'maxiter': nmaxit})
                hr = np.asarray(res.x[:n], dtype=np.float64)
                hp = capped_simplex_project(hr, n, target)
                sc = exact_score(hp)
                if sc < best_sc - 1e-12:
                    best = hp
                    best_sc = sc
            except Exception:
                pass

    if n <= 128 and best_sc < float('inf'):
        best, best_sc = pair_search(best.copy(), n, max_sweeps=psw)

    best = capped_simplex_project(best, n, target)
    best_sc = exact_score(best)
    if parent_fb is not None:
        pf_sc = exact_score(parent_fb)
        if pf_sc < best_sc - 1e-12:
            best = parent_fb
            best_sc = pf_sc
    if not np.all(np.isfinite(best)) or best.min() < 0.0 or best.max() > 1.0:
        best = np.full(n, 0.5)
        best_sc = exact_score(best)
    return (best, best_sc, int(n))
