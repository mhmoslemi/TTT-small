        if len(z1) != n + 1 or not np.all(np.isfinite(z1)):
            return None
        h1 = project_box_sum(z1[:n], S)
        if h1 is None or not validate_h(h1, n):
            return None
        return h1

    for h0 in starts:
        h1 = solve_start(h0)
        if h1 is not None:
            s1 = exact_score(h1)
            if np.isfinite(s1) and s1 < best_score:
                best_h = h1.copy()
                best_score = s1

    sigma = 0.05
    for _ in range(4):
        pert = best_h + sigma * rng.standard_normal(n)
        p = project_box_sum(pert, S)
        if p is None:
            sigma = min(0.25, 1.5 * sigma)
            continue
        h1 = solve_start(p)
        improved = False
        if h1 is not None:
            s1 = exact_score(h1)
            if np.isfinite(s1) and s1 < best_score - 1e-12:
                best_h = h1.copy()
                best_score = s1
                improved = True
        if improved:
            sigma = max(0.01, 0.7 * sigma)
        else:
            sigma = min(0.25, 1.5 * sigma)

    final_h = project_box_sum(best_h, S)
    if final_h is None or not validate_h(final_h, n):
        final_h = parent.copy()
    final_score = exact_score(final_h)
    if not np.isfinite(final_score):
        final_h = np.full(n, 0.5)
        final_score = exact_score(final_h)
    return (final_h.astype(np.float64), final_score, n)
