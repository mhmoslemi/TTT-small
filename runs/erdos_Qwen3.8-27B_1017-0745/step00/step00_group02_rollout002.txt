def check_feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            np.all(h >= -1e-9) and np.all(h <= 1.0 + 1e-9) and
            abs(h.sum() - S) <= 1e-8)

def smooth_obj(h, dx, n, tau):
    qs = np.empty(2 * n - 1)
    idx = 0
    for k in range(-(n - 1), n):
        i_s, i_e = max(0, -k), min(n, n - k)
        j_s, j_e = max(0, k), min(n, n + k)
        qs[idx] = dx * np.sum(h[j_s:j_e] * (1.0 - h[i_s:i_e]))
        idx += 1
    m = qs.max()
    return m + tau * np.log(np.sum(np.exp((qs - m) / tau)))

def smooth_grad(h, dx, n, tau):
    qs = np.empty(2 * n - 1)
    idx = 0
    for k in range(-(n - 1), n):
        i_s, i_e = max(0, -k), min(n, n - k)
        j_s, j_e = max(0, k), min(n, n + k)
        qs[idx] = dx * np.sum(h[j_s:j_e] * (1.0 - h[i_s:i_e]))
        idx += 1
    m = qs.max()
    w = np.exp((qs - m) / tau)
    p = w / w.sum()
    grad = np.zeros(n)
    idx = 0
    for k in range(-(n - 1), n):
        i_s, i_e = max(0, -k), min(n, n - k)
        j_s, j_e = max(0, k), min(n, n + k)
        a = h[j_s:j_e]
        b = h[i_s:i_e]
        grad[j_s:j_e] += p[idx] * dx * (1.0 - b)
        grad[i_s:i_e] -= p[idx] * dx * a
        idx += 1
    return grad

def pg_step(h, dx, n, S, tau, max_iter=80, max_ls=25):
    F_cur = smooth_obj(h, dx, n, tau)
    alpha = 1.0
    for _ in range(max_iter):
        g = smooth_grad(h, dx, n, tau)
        if not np.all(np.isfinite(g)):
            break
        accepted = False
        a = alpha
        for _ in range(max_ls):
            y = project_to_K(h - a * g, n, S)
            d = y - h
            if np.max(np.abs(d)) <= 1e-12:
                break
            gd = g @ d
            if gd >= -1e-14:
                break
            F_new = smooth_obj(y, dx, n, tau)
            if F_new <= F_cur + 1e-4 * gd + 1e-14:
                decrease = F_cur - F_new
                h = y
                F_cur = F_new
                alpha = min(1.25 * a, 5.0)
                accepted = True
                if decrease < 1e-12:
                    break
                break
            a *= 0.5
        if not accepted:
            break
    return h

def sl_sqp_obj(x, n, S):
    return x[n]

def sl_sqp_obj_jac(x, n, S):
    j = np.zeros(n + 1)
    j[n] = 1.0
    return j

def sl_sqp_ineq(x, n, S, dx):
    h = x[:n]
    return x[n] - np.correlate(h, 1.0 - h, mode='full') * dx

def sl_sqp_ineq_jac(x, n, S, dx):
    h = x[:n]
    J = np.zeros((2 * n - 1, n + 1))
    for idx, k in enumerate(range(-(n - 1), n)):
        i_s, i_e = max(0, -k), min(n, n - k)
        j_s, j_e = max(0, k), min(n, n + k)
        J[idx, j_s:j_e] = -dx * (1.0 - h[i_s:i_e])
        J[idx, i_s:i_e] += dx * h[j_s:j_e]
        J[idx, n] = 1.0
    return J

def sl_sqp_eq(x, n, S):
    return x[:n].sum() - S

def sl_sqp_eq_jac(x, n, S):
    j = np.zeros(n + 1)
    j[:n] = 1.0
    return j

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = globals().get('initial_h_values', None)
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64).reshape(-1)
        if (p.size >= 2 and p.size <= 999 and np.all(np.isfinite(p)) and
                np.max(np.abs(p)) < 1e6):
            n = p.size
        else:
            n = int(kwargs.get('n_points', 45))
            if n < 2 or n > 999:
                n = 45
            p = np.full(n, 0.5)
    else:
        n = int(kwargs.get('n_points', 45))
        if n < 2 or n > 999:
            n = 45
        p = np.full(n, 0.5)
    dx = 2.0 / n
    S = 0.5 * n
    h0 = project_to_K(p, n, S)
    best_h = h0.copy()
    best_c = exact_score(h0, dx)

    def archive(hc):
        nonlocal best_h, best_c
        hc = project_to_K(hc, n, S)
        if check_feasible(hc, n, S):
            sc = exact_score(hc, dx)
            if sc < best_c - 1e-12:
                best_h = hc.copy()
                best_c = sc

    starts = [h0.copy(), 1.0 - h0.copy(), h0[::-1].copy(), (1.0 - h0[::-1]).copy()]
    starts.append(project_to_K(h0 + 0.08 * rng.standard_normal(n), n, S))
    starts.append(project_to_K(h0 + 0.25 * rng.standard_normal(n), n, S))
    starts.append(project_to_K(rng.uniform(0, 1, n), n, S))
    starts.append(project_to_K(rng.beta(0.5, 0.5, n), n, S))

    taus = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00001]
    local_candidates = []
    for s in starts:
        h_cur = s.copy()
        for t in taus:
            h_cur = pg_step(h_cur, dx, n, S, t)
            archive(h_cur)
        local_candidates.append(h_cur.copy())

    for sigma in [0.04, 0.12, 0.30, 0.50]:
        h_pert = project_to_K(best_h + sigma * rng.standard_normal(n), n, S)
        h_cur = h_pert.copy()
        for t in taus:
            h_cur = pg_step(h_cur, dx, n, S, t)
            archive(h_cur)
        local_candidates.append(h_cur.copy())

    if n <= 100:
        local_candidates.append(best_h.copy())
        local_candidates.sort(key=lambda x: exact_score(x, dx))
        distinct = []
        for c in local_candidates:
            if all(np.max(np.abs(c - d)) > 1e-5 for d in distinct):
                distinct.append(c)
            if len(distinct) >= 3:
                break
        for h_init in distinct:
            t0 = min(1.0, exact_score(h_init, dx) + 1e-8)
            x0 = np.concatenate([h_init, [t0]])
            bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
            try:
                res = minimize(sl_sqp_obj, x0, method='SLSQP',
                               jac=lambda x, nn, ss: sl_sqp_obj_jac(x, nn, ss),
                               args=(n, S),
                               bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': sl_sqp_eq,
                                             'jac': sl_sqp_eq_jac, 'args': (n, S)},
                                            {'type': 'ineq', 'fun': sl_sqp_ineq,
                                             'jac': sl_sqp_ineq_jac, 'args': (n, S, dx)}],
                               options={'maxiter': 250, 'ftol': 1e-10, 'disp': False})
                h_sol = res.x[:n]
                if (h_sol.shape == (n,) and np.all(np.isfinite(h_sol)) and
                        np.all(h_sol >= -1e-6) and np.all(h_sol <= 1.0 + 1e-6)):
                    h_sol = project_to_K(h_sol, n, S)
                    if check_feasible(h_sol, n, S):
                        archive(h_sol)
            except Exception:
                pass

    best_h = project_to_K(best_h, n, S)
    if not check_feasible(best_h, n, S):
        best_h = project_to_K(h0, n, S)
    if not check_feasible(best_h, n, S):
        best_h = np.full(n, 0.5)
    best_c = exact_score(best_h, dx)
    if not check_feasible(h0, n, S):
        h0 = np.full(n, 0.5)
    c0 = exact_score(h0, dx)
    if c0 < best_c - 1e-12:
        best_h = h0.copy()
        best_c = c0
    if best_c >= 0.38085:
        alt = project_to_K(h0, n, S)
        ca = exact_score(alt, dx)
        if ca < best_c:
            best_h = alt.copy()
            best_c = ca
    return (best_h.copy(), float(best_c), int(n))
