def project_capped(z, n):
    s = n / 2.0
    z = np.asarray(z, dtype=np.float64)
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return np.full(n, 0.5)
    lo, hi = z.min() - 1.0, z.max()
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(z - mid, 0.0, 1.0).sum() > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)
    for _ in range(n + 2):
        r = s - h.sum()
        if abs(r) < 1e-13:
            break
        if r > 0:
            i = int(np.argmax(1.0 - h))
            h[i] = min(1.0, h[i] + r)
        else:
            i = int(np.argmax(h))
            h[i] = max(0.0, h[i] + r)
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(h.sum() - s) > 1e-9 * max(1, n):
        return np.full(n, 0.5)
    return h

def exact_score(h):
    if h is None or len(h) == 0 or not np.all(np.isfinite(h)):
        return float('inf')
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)))

def corr_jac(h):
    n = len(h)
    q = np.correlate(h, 1.0 - h, mode="full")
    J = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        i0, j0 = max(d, 0), max(-d, 0)
        L = n - abs(d)
        for r in range(L):
            i, j = i0 + r, j0 + r
            J[row, i] += 1.0 - h[j]
            J[row, j] -= h[i]
    return q, J

def obj_fun(z, n, dx):
    return z[n]

def obj_jac(z, n, dx):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def ineq_fun(z, n, dx):
    h, t = z[:n], z[n]
    return t - dx * np.correlate(h, 1.0 - h, mode="full")

def ineq_jac(z, n, dx):
    _, J = corr_jac(z[:n])
    out = np.zeros((2 * n - 1, n + 1))
    out[:, :n] = -dx * J
    out[:, n] = 1.0
    return out

    return z[:n].sum() - n / 2.0

def eq_jac(z, n, dx):

    cands.append(base.copy())
    cands.append(base[::-1].copy())
    cands.append(1.0 - base[::-1])

    blk = np.zeros(n)

    alt = np.zeros(n)

    cands.append(alt)
    for sd in [0.02, 0.06, 0.15, 0.35]:
        cands.append(base + rng.normal(0.0, sd, n))
    cands.append(rng.uniform(0.0, 1.0, n))
    cands.append(rng.standard_normal(n))
    starts = []

def pair_search(h, n, max_sweeps=3):

        for _ in range(max_sweeps):
