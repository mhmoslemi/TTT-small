import numpy as np
from scipy.optimize import minimize

_N = 0
_M = 0

def project_box_sum(z, m):
    n = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(n, 0.5)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    res = m - np.sum(h)
    for _ in range(4):
        if res > 1e-14:
            for i in range(n):
                room = 1.0 - h[i]
                if room > 0:
                    a = min(room, res)
                    h[i] += a
                    res -= a
                    if res < 1e-14:
                        break
        elif res < -1e-14:
            for i in range(n):
                if h[i] > 0:
                    a = min(h[i], -res)
                    h[i] -= a
                    res += a
                    if res > -1e-14:
                        break
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(m - np.sum(h)) > 1e-9 * max(1, n):
        return np.full(n, 0.5)
    return h

def score(h):
    n = len(h)
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def lag_q_and_jac(h):
    n = len(h)
    nl = 2 * n - 1
    q = np.zeros(nl)
    G = np.zeros((nl, n))
    for s in range(-(n - 1), n):
        idx = s + n - 1
        if s >= 0:
            L = n - s
            a = h[s:s + L]
            b = h[0:L]
            q[idx] = np.sum(a * (1.0 - b))
            G[idx, s:s + L] += (1.0 - b)
            G[idx, 0:L] -= a
        else:
            d = -s
            L = n - d
            a = h[0:L]
            b = h[d:d + L]
            q[idx] = np.sum(a * (1.0 - b))
            G[idx, 0:L] += (1.0 - b)
            G[idx, d:d + L] -= a
    return q, G

def _obj(z):
    return z[-1]

def _obj_jac(z):
    g = np.zeros(_N + 1)
    g[-1] = 1.0
    return g

def _eq_con(z):
    return np.sum(z[:-1]) - _M

def _eq_jac(z):
    g = np.zeros(_N + 1)
    g[:-1] = 1.0
    return g

def _ineq_con(z):
    h = z[:-1]
    t = z[-1]
    q, _ = lag_q_and_jac(h)
    return t - q

def _ineq_jac(z):
    h = z[:-1]
    _, G = lag_q_and_jac(h)
    J = np.zeros((G.shape[0], _N + 1))
    J[:, :-1] = -G
    J[:, -1] = 1.0
    return J

def _solve(h0):
    n = _N
    m = _M
    q, _ = lag_q_and_jac(h0)
    t0 = max(float(np.max(q)), 0.0)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, float(n))]
    cons = [
        {'type': 'eq', 'fun': _eq_con, 'jac': _eq_jac},
        {'type': 'ineq', 'fun': _ineq_con, 'jac': _ineq_jac},
    ]
    try:
        res = minimize(_obj, z0, method='SLSQP', jac=_obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-9})
        if res is not None and hasattr(res, 'x') and res.x is not None:
            if len(res.x) == n + 1 and np.all(np.isfinite(res.x)):
                return project_box_sum(res.x[:-1], m)
    except Exception:
        pass
    return project_box_sum(h0, m)

def _try_candidate(h0, best, best_score):
    s0 = score(h0)
    if s0 < best_score:
        best = h0.copy()
        best_score = s0
    cand = _solve(h0)
    s1 = score(cand)
    if s1 < best_score:
        best = cand.copy()
        best_score = s1
    return best, best_score

def run(seed=42, **kwargs):
    global _N, _M
    parent = None
    if 'initial_h_values' in globals():
        p = globals()['initial_h_values']
        if p is not None and hasattr(p, '__len__') and len(p) > 0:
            p = np.asarray(p, dtype=np.float64)
            if p.ndim == 1 and len(p) > 0:
                if np.all(np.isfinite(p)):
                    parent = p.copy()
                else:
                    parent = None
    if parent is not None:
        n = len(parent)
    else:
        try:
            n = int(kwargs.get('n_points', 70))
            if n <= 0:
                n = 70
        except (TypeError, ValueError):
            n = 70
    _N = n
    m = n / 2.0
    _M = m

    if n == 1:
        h = np.array([0.5])
        return h, score(h), 1

    if parent is not None:
        if np.all(parent >= 0.0) and np.all(parent <= 1.0) and abs(np.sum(parent) - m) < 1e-10:
            baseline = parent.copy()
        else:
            baseline = project_box_sum(parent, m)
    else:
        baseline = np.full(n, 0.5)

    best = baseline.copy()
    best_score = score(best)

    rng = np.random.default_rng(seed)

    starts = [baseline.copy()]
    comp = project_box_sum(1.0 - baseline, m)
    starts.append(comp)
    starts.append(np.full(n, 0.5))
    for sc in [0.05, 0.15, 0.35]:
        starts.append(project_box_sum(baseline + sc * rng.standard_normal(n), m))
    starts.append(project_box_sum(rng.uniform(0.0, 1.0, n), m))
    # smooth Fourier start
    n_modes = min(4, (n - 1) // 2)
    zz = np.full(n, 0.5)
    for r in range(1, n_modes + 1):
        amp = 0.12 / r
        A = rng.standard_normal()
        B = rng.standard_normal()
        zz += amp * (A * np.sin(2.0 * np.pi * r * np.arange(n) / n) +
                     B * np.cos(2.0 * np.pi * r * np.arange(n) / n))
    starts.append(project_box_sum(zz, m))

    for h0 in starts:
        best, best_score = _try_candidate(h0, best, best_score)

    # Local basin hops with descending scale
    for sc in [0.10, 0.05, 0.025, 0.0125]:
        h0 = project_box_sum(best + sc * rng.standard_normal(n), m)
        best, best_score = _try_candidate(h0, best, best_score)

    # Final projection and validation
    best = project_box_sum(best, m)
    best_score = score(best)
    if not np.isfinite(best_score):
        best = np.full(n, 0.5)
        best_score = score(best)

    return best, best_score, n
