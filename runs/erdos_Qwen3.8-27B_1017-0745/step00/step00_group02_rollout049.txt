    _HAS_CP = False

def _project_k(y, S, n):
    y = np.asarray(y, dtype=np.float64)
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return None
    lo, hi = float(y.min()) - 1.0, float(y.max()) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0, 1).sum())
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - float(x.sum())
    if abs(r) > 1e-12:
        k = max(1, int(abs(r)) + 1)
        idx = np.argsort(1.0 - x) if r > 0 else np.argsort(x)
        x[idx[:k]] += r / k
        x = np.clip(x, 0.0, 1.0)
    if not (np.all(x >= 0) and np.all(x <= 1) and abs(float(x.sum()) - S) < 1e-10):
        return None
    return x

def _score(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(c)):
        return None
    return float(c.max() * dx)

def _valid(h, S, n):
    return (h.shape == (n,) and bool(np.all(np.isfinite(h))) and
            bool(np.all(h >= 0) and np.all(h <= 1)) and
            abs(float(h.sum()) - S) < 1e-10)

def _compute_G(h, dx, n):
    G = np.zeros((2 * n - 1, n), dtype=np.float64)
    js = np.arange(n)
    for ell in range(-(n - 1), n):
        g = np.zeros(n)
        jp = js - ell
        m1 = (jp >= 0) & (jp < n)
        g[m1] += 1.0 - h[jp[m1]]
        jm = js + ell
        m2 = (jm >= 0) & (jm < n)
        g[m2] -= h[jm[m2]]
        G[ell + n - 1] = dx * g
    return G

def _solve_qp(h, q, G, dx, n, S, C):
    if not _HAS_CP:
        return None
    d = cp.Variable(n)
    t = cp.Variable()
    rhs = S - float(h.sum())
    cons = [q + G @ d <= t, h + d >= 0, h + d <= 1, cp.sum(d) == rhs]
    prob = cp.Problem(cp.Minimize(t + dx * cp.sum_squares(d)), cons)
    for solver in [cp.CVXOPT, cp.ECOS]:
        try:
            prob.solve(solver=solver, max_iters=200, abstol=1e-8, reltol=1e-8)
        except Exception:
            continue
        if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
            continue
        try:
            dv = np.array(d.value).flatten().astype(np.float64)
            tv = float(t.value)
        except Exception:
            continue
        if not (np.all(np.isfinite(dv)) and np.isfinite(tv)):
            continue
        aff = q + G @ dv
        if not np.all(np.isfinite(aff)) or float(aff.max()) > tv + 1e-5:
            continue
        if float(aff.max()) + dx * float(dv @ dv) > C + 1e-5:
            continue
        if float(np.abs(dv).max()) > 1.0 + 1e-5:
            continue
        hd = h + dv
        if float(np.max(np.maximum(0.0, -hd, hd - 1.0))) > 1e-5:
            continue
        if abs(float(hd.sum()) - S) > 1e-5:
            continue
        return dv
    return None

def _opt_start(h0, dx, n, S, max_iter, max_bt):
    h = h0.copy()
    cur = _score(h, n)
    if cur is None:
        return h, None
    stall = 0
    for _ in range(max_iter):
        q = np.correlate(h, 1.0 - h, mode='full') * dx
        G = _compute_G(h, dx, n)
        dv = _solve_qp(h, q, G, dx, n, S, cur)
        if dv is None:
            break
        hp = _project_k(h + dv, S, n)
        if hp is None:
            break
        d_f = hp - h
        if float(np.abs(d_f).max()) < 1e-9:
            break
        best_a, best_s, best_c = 0.0, cur, None
        for k in range(max_bt + 1):
            a = 2.0 ** (-k)
            c = _project_k(h + a * d_f, S, n)
            if c is not None:
                sc = _score(c, n)
                if sc is not None and sc < best_s - 1e-12:
                    best_a, best_s, best_c = a, sc, c
        if best_a == 0.0:
            break
        if cur - best_s < 1e-10:
            stall += 1
            if stall >= 3:
                break
        else:
            stall = 0
        h, cur = best_c.copy(), best_s
    return h, cur

def _add_start(starts, s, S, n):
    p = _project_k(s, S, n)
    if p is None:
        return
    for e in starts:
        if float(np.abs(p - e).max()) < 1e-8:
            return
    starts.append(p)

def _make_starts(parent, n, S, rng, has_parent):
    starts = []
    if has_parent:
        starts.append(parent.copy())
        _add_start(starts, 1.0 - parent, S, n)
        _add_start(starts, parent[::-1].copy(), S, n)
        _add_start(starts, parent + rng.standard_normal(n) * 0.08, S, n)
        _add_start(starts, parent + rng.standard_normal(n) * 0.25, S, n)
        _add_start(starts, rng.beta(0.5, 0.5, n), S, n)
    else:
        starts.append(np.full(n, 0.5))
        _add_start(starts, rng.uniform(0, 1, n), S, n)
        _add_start(starts, rng.beta(0.5, 0.5, n), S, n)
        m = int(rng.integers(1, max(2, min(5, n // 2 + 1))))
        ph = float(rng.uniform(0, 2 * math.pi))
        _add_start(starts, 0.5 + 0.35 * np.cos(2 * math.pi * m * (np.arange(n) + 0.5) / n + ph), S, n)
    cap = 6 if n <= 64 else (4 if n <= 256 else 3)
    return starts[:cap]

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    n, parent = None, None
    try:
        _iv = globals().get('initial_h_values', None)
        if _iv is not None:
            pv = np.array(_iv, dtype=np.float64).flatten()
            if 2 <= len(pv) <= 999 and np.all(np.isfinite(pv)) and float(np.abs(pv).max()) <= 1e6:
                n = len(pv)
                S = float(n) / 2.0
                if bool(np.all(pv >= 0) and np.all(pv <= 1)) and abs(float(pv.sum()) - S) < 1e-12:
                    parent = pv.copy()
                else:
                    p2 = _project_k(pv, S, n)
                    if p2 is not None:
                        parent = p2
    except Exception:
        pass
    if n is None:
        np_ = kwargs.get('n_points', 45)
        n = np_ if isinstance(np_, int) and not isinstance(np_, bool) and 2 <= np_ <= 999 else 45
    S = float(n) / 2.0
    dx = 2.0 / n
    fallback = parent.copy() if parent is not None else np.full(n, 0.5)
    fb_sc = _score(fallback, n)
    if fb_sc is None:
        fallback = np.full(n, 0.5)
        fb_sc = _score(fallback, n)
    best_h, best_sc = fallback.copy(), fb_sc
    starts = _make_starts(parent, n, S, rng, parent is not None)
    mi = 40 if n <= 64 else (20 if n <= 256 else 8)
    mb = 8 if n <= 64 else (5 if n <= 256 else 3)
    for s in starts:
        try:
            oh, oc = _opt_start(s, dx, n, S, mi, mb)
            if oh is not None and oc is not None and oc < best_sc - 1e-12:
                best_h, best_sc = oh.copy(), oc
        except Exception:
            continue
    if not _valid(best_h, S, n):
        best_h = _project_k(best_h, S, n)
        if best_h is None:
            best_h = fallback.copy()
    if not _valid(fallback, S, n):
        fallback = np.full(n, 0.5)
        fb_sc = _score(fallback, n)
    if _valid(fallback, S, n) and fb_sc is not None and fb_sc < best_sc - 1e-12:
        best_h, best_sc = fallback.copy(), fb_sc
    if not _valid(best_h, S, n):
        best_h = np.full(n, 0.5)
    best_h = np.clip(best_h.astype(np.float64), 0.0, 1.0)
    fin = _score(best_h, n)
    if fin is None:
        best_h = np.full(n, 0.5)
        fin = _score(best_h, n)
    return best_h.copy(), fin, int(n)
