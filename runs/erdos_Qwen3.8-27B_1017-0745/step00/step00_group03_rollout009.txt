import numpy as np
from scipy.optimize import minimize

def project_box_sum(z, m):
    n = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(n, 0.5)
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        clipped = np.clip(z - mid, 0.0, 1.0)
        s = np.sum(clipped)
        if s > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    resid = m - np.sum(h)
    if resid > 0:
        for i in range(n):
            space = 1.0 - h[i]
            adj = min(resid, space)
            h[i] += adj
            resid -= adj
            if resid <= 1e-15:
                break
    elif resid < 0:
        for i in range(n):
            adj = min(-resid, h[i])
            h[i] -= adj
            resid += adj
            if resid >= -1e-15:
                break
    if not np.all(np.isfinite(h)) or np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return np.full(n, 0.5)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - m) > 1e-6:
        return np.full(n, 0.5)
    return h

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def lag_values_and_jac(h):
    N = len(h)
    S = 2 * N - 1
    q = np.zeros(S)
    G = np.zeros((S, N))
    for s in range(-(N-1), N):
        idx = s + (N - 1)
        if s >= 0:
            L = N - s
            a = h[:L]
            b = h[s:s+L]
            q[idx] = np.sum(a * (1.0 - b))
            G[idx, :L] += (1.0 - b)
            G[idx, s:s+L] -= a
        else:
            d = -s
            L = N - d
            a = h[d:d+L]
            b = h[:L]
            q[idx] = np.sum(a * (1.0 - b))
            G[idx, d:d+L] += (1.0 - b)
            G[idx, :L] -= a
    return q, G

def obj_epigraph(z, N, m):
    return z[N]

def obj_epigraph_jac(z, N, m):
    g = np.zeros(N + 1)
    g[N] = 1.0
    return g

def eq_constraint(z, N, m):
    return np.sum(z[:N]) - m

def eq_constraint_jac(z, N, m):
    j = np.zeros(N + 1)
    j[:N] = 1.0
    return j

def ineq_constraint(z, N, m):
    h = z[:N]
    t = z[N]
    q, _ = lag_values_and_jac(h)
    return t - q

def ineq_constraint_jac(z, N, m):
    h = z[:N]
    _, G = lag_values_and_jac(h)
    S = 2 * N - 1
    J = np.zeros((S, N + 1))
    J[:, :N] = -G
    J[:, N] = 1.0
    return J

def build_constraints(N, m):
    cons = [
        {'type': 'eq', 'fun': lambda z: eq_constraint(z, N, m), 'jac': lambda z: eq_constraint_jac(z, N, m)},
        {'type': 'ineq', 'fun': lambda z: ineq_constraint(z, N, m), 'jac': lambda z: ineq_constraint_jac(z, N, m)},
    ]
    return cons

import numpy as np
from scipy.optimize import minimize

_N = 70
_m = 35.0

def project_box_sum(z, m):
    n = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(n, 0.5)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    resid = m - np.sum(h)
    if resid > 1e-12:
        for i in range(n):
            sp = 1.0 - h[i]
            adj = min(resid, sp)
            h[i] += adj
            resid -= adj
            if resid <= 1e-14:
                break
    elif resid < -1e-12:
        for i in range(n):
            adj = min(-resid, h[i])
            h[i] -= adj
            resid += adj
            if resid >= -1e-14:
                break
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - m) > 1e-6:
        return np.full(n, 0.5)
    return h

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def lag_values_and_jac(h):
    N = len(h)
    S = 2 * N - 1
    q = np.zeros(S)
    G = np.zeros((S, N))
    for s in range(-(N - 1), N):
        idx = s + (N - 1)
        if s >= 0:
            L = N - s
            a = h[:L]
            b = h[s:s + L]
            q[idx] = np.sum(a * (1.0 - b))
            G[idx, :L] += (1.0 - b)
            G[idx, s:s + L] -= a
        else:
            d = -s
            L = N - d
            a = h[d:d + L]
            b = h[:L]
            q[idx] = np.sum(a * (1.0 - b))
            G[idx, d:d + L] += (1.0 - b)
            G[idx, :L] -= a
    return q, G

def obj_fun(z):
    return z[_N]

def obj_jac(z):
    g = np.zeros(_N + 1)
    g[_N] = 1.0
    return g

def eq_fun(z):
    return np.sum(z[:_N]) - _m

def eq_jac(z):
    j = np.zeros(_N + 1)
    j[:_N] = 1.0
    return j

def ineq_fun(z):
    h = z[:_N]
    t = z[_N]
    q, _ = lag_values_and_jac(h)
    return t - q

def ineq_jac(z):
    h = z[:_N]
    _, G = lag_values_and_jac(h)
    S = 2 * _N - 1
    J = np.zeros((S, _N + 1))
    J[:, :_N] = -G
    J[:, _N] = 1.0
    return J

def try_solve(h0):
    q0, _ = lag_values_and_jac(h0)
    t0 = float(np.max(q0))
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * _N + [(0.0, float(_N))]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
    ]
    try:
        res = minimize(obj_fun, z0, method='SLSQP', jac=obj_jac,
                       bounds=bounds, constraints=cons,
                       maxiter=300, ftol=1e-10, options={'disp': False})
        if res.x is not None and len(res.x) == _N + 1 and np.all(np.isfinite(res.x)):
            return project_box_sum(res.x[:_N], _m)
    except Exception:
        pass
    return h0.copy()

def run(seed=42, **kwargs):
    global _N, _m
    rng = np.random.default_rng(seed)

    parent = globals().get('initial_h_values', None)
    if parent is not None:
        parent = np.asarray(parent, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            _N = parent.size
            _m = _N / 2.0
            if np.all(np.isfinite(parent)):
                if np.all(parent >= 0) and np.all(parent <= 1) and abs(np.sum(parent) - _m) < 1e-10:
                    best = parent.copy()
                else:
                    best = project_box_sum(parent, _m)
            else:
                best = np.full(_N, 0.5)
        else:
            _N = 70
            _m = 35.0
            best = np.full(_N, 0.5)
    else:
        n_req = kwargs.get('n_points', None)
        if n_req is not None and isinstance(n_req, int) and n_req > 0:
            _N = n_req
        else:
            _N = 70
        _m = _N / 2.0
        best = np.full(_N, 0.5)

    best_score = score_exact(best, _N)
    if _N == 1:
        return best, best_score, _N

    starts = [best.copy()]
    starts.append(project_box_sum(1.0 - best, _m))
    starts.append(np.full(_N, 0.5))
    for sc in [0.05, 0.15, 0.35]:
        starts.append(project_box_sum(best + sc * rng.standard_normal(_N), _m))
    starts.append(project_box_sum(rng.uniform(0, 1, _N), _m))
    z_f = np.full(_N, 0.5)
    nm = min(4, (_N - 1) // 2)
    idx = np.arange(_N)
    for r in range(1, nm + 1):
        amp = 0.15 / r
        z_f += amp * rng.standard_normal() * np.sin(2.0 * np.pi * r * idx / _N)
        z_f += amp * rng.standard_normal() * np.cos(2.0 * np.pi * r * idx / _N)
    starts.append(project_box_sum(z_f, _m))

    for h0 in starts:
        s = score_exact(h0, _N)
        if s < best_score:
            best = h0.copy()
            best_score = s
        hc = try_solve(h0)
        s = score_exact(hc, _N)
        if s < best_score:
            best = hc.copy()
            best_score = s

    for sc in [0.10, 0.05, 0.025, 0.0125]:
        h0 = project_box_sum(best + sc * rng.standard_normal(_N), _m)
        s = score_exact(h0, _N)
        if s < best_score:
            best = h0.copy()
            best_score = s
        hc = try_solve(h0)
        s = score_exact(hc, _N)
        if s < best_score:
            best = hc.copy()
            best_score = s

    best = project_box_sum(best, _m)
    best_score = score_exact(best, _N)
    if not np.isfinite(best_score):
        best = np.full(_N, 0.5)
        best_score = score_exact(best, _N)
    return best, best_score, _N
