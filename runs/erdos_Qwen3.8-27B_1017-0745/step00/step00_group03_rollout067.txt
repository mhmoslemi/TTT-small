    return phi, grad

def optimize_start(h0, N, inc_vec, inc_score, iters):
    temps = [0.08, 0.02, 0.005, 0.001, 0.0002, 0.00004, 0.000008]
    h = h0.copy()
    for tau in temps:
        best_h = h.copy()
        best_s = score_exact(h, N)
        alpha_s = 1.0
        for _ in range(iters):
            phi, g = smooth_value_grad(h, tau, N)
            if not np.all(np.isfinite(g)):
                break
            alpha = alpha_s
            accepted = False
            stop = False
            for _ in range(24):
                y = project_box_sum(h - alpha * g, N)
                d = y - h
                dm = np.max(np.abs(d))
                if dm < 1e-10:
                    break
                gd = np.dot(g, d)
                phi_y = smooth_value(y, tau, N)
                s_y = score_exact(y, N)
                if s_y < best_s:
                    best_s = s_y
                    best_h = y.copy()
                if s_y < inc_score:
                    inc_score = s_y
                    inc_vec[:] = y
                if gd < 0 and phi_y <= phi + 1e-4 * gd + 1e-14:
                    h = y
                    alpha_s = min(1.4 * alpha, 10.0)
                    accepted = True
                    if dm < 1e-8:
                        stop = True
                    break
                alpha *= 0.5
                if alpha < 1e-8:
                    break
            if stop or not accepted:
                h = best_h.copy()
                break
        h = best_h.copy()
    sf = score_exact(h, N)
    if sf < inc_score:
        inc_score = sf
        inc_vec[:] = h
    return inc_vec.copy(), inc_score

def run(seed=42, **kwargs):
    try:
        sd = int(seed)
    except (ValueError, TypeError):
        sd = 42
    rng = np.random.default_rng(sd)
    has_parent = False
    N = 70
    baseline = np.full(N, 0.5)
    try:
        parent = initial_h_values
        if parent is not None:
            p = np.asarray(parent, dtype=np.float64)
            if p.ndim == 1 and p.size > 0:
                N = p.size
                if np.all(np.isfinite(p)):
                    baseline = np.clip(p, 0.0, 1.0)
                    has_parent = True
                else:
                    baseline = project_box_sum(p, N)
            else:
                baseline = np.full(N, 0.5)
    except NameError:
        pass
    except Exception:
        baseline = np.full(N, 0.5)
    if not has_parent and 'n_points' in kwargs:
        try:
            np_ = int(kwargs['n_points'])
            if 1 <= np_ <= 999:
                N = np_
                baseline = np.full(N, 0.5)
        except Exception:
            pass
    baseline = project_box_sum(baseline, N)
    base_score = score_exact(baseline, N)
    inc_vec = baseline.copy()
    inc_score = base_score
    if N == 1:
        return inc_vec, inc_score, N
    if N <= 100:
        n_starts, iters = 8, 60
    elif N <= 200:
        n_starts, iters = 6, 40
    elif N <= 500:
        n_starts, iters = 4, 20
    else:
        n_starts, iters = 3, 8
    b = baseline
    starts = [b.copy(),
              project_box_sum(b + 0.20 * rng.standard_normal(N), N),
              project_box_sum(rng.uniform(0, 1, N), N)]
    R = min(6, (N - 1) // 2)
    z = np.full(N, 0.5)
    ar = np.arange(N)
    for r in range(1, R + 1):
        amp = 0.12 / r
        A = rng.standard_normal()
        B = rng.standard_normal()
        z += amp * (A * np.sin(2 * np.pi * r * ar / N) + B * np.cos(2 * np.pi * r * ar / N))
    starts.append(project_box_sum(z, N))
    if has_parent:
        starts.append(project_box_sum(1.0 - b, N))
    else:
        starts.append(project_box_sum(0.5 + 0.4 * rng.standard_normal(N), N))
    starts.append(project_box_sum(np.full(N, 0.5), N))
    starts.append(project_box_sum(b + 0.08 * rng.standard_normal(N), N))
    starts.append(project_box_sum(b + 0.40 * rng.standard_normal(N), N))
    for i in range(min(n_starts, len(starts))):
        s = project_box_sum(starts[i], N)
        ss = score_exact(s, N)
        if ss < inc_score:
            inc_score = ss
            inc_vec[:] = s
        try:
            inc_vec, inc_score = optimize_start(s, N, inc_vec, inc_score, iters)
        except Exception:
            pass
    inc_vec = project_box_sum(inc_vec, N)
    fs = score_exact(inc_vec, N)
    if not np.isfinite(fs):
        inc_vec = np.full(N, 0.5)
        fs = score_exact(inc_vec, N)
    return inc_vec, float(fs), N
