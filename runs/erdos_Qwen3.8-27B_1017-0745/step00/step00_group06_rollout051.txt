        {'type': 'eq', 'fun': _ep_eq, 'jac': _ep_eq_jac},
        {'type': 'ineq', 'fun': _ep_ineq, 'jac': _ep_ineq_jac}
    ]
    for h0 in top:
        t0 = exact_score(h0) + 1e-6
        z0 = np.concatenate([h0, [t0]])
        try:
            res = minimize(_ep_fun, z0, jac=_ep_jac, method='SLSQP',
                           bounds=bnds_e, constraints=cons_e,
                           options={'ftol': 1e-10, 'maxiter': mio, 'disp': False})
            if res.x is not None and res.x.shape == (N + 1,) and np.all(np.isfinite(res.x)):
                ht = proj(res.x[:-1], m)
                ct = exact_score(ht)
                if ct < best_c - 1e-12:
                    best_c = ct
                    best_h = ht
        except Exception:
            pass

    best_h = proj(best_h, m)
    best_c = exact_score(best_h)
    if not (np.all(best_h >= -1e-14) and np.all(best_h <= 1.0 + 1e-14) and abs(np.sum(best_h) - m) < 1e-9 * max(1, N)):
        best_h = np.full(N, 0.5)
        best_c = exact_score(best_h)
    return (best_h, float(best_c), int(N))
