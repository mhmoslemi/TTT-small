def is_feasible(h, n):

        return False

        return False

        return False
    mass_err = abs(np.sum(h) - n / 2.0)
    return mass_err <= 1e-10 * max(1, n)

def capped_simplex_project(z, n):
    if not np.all(np.isfinite(z)):
        return np.full(n, 0.5)
    z = np.clip(z, -1e12, 1e12).astype(np.float64)

        else:
            hi = mid

import numpy as np


def is_feasible(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n)


def capped_simplex_project(z, n):
    if not np.all(np.isfinite(z)):
        return np.full(n, 0.5)
    z = np.clip(z, -1e12, 1e12).astype(np.float64)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(z - mid, 0.0, 1.0)
        if np.sum(h) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) < 1e-13:
            break
        if r > 0:
            slack = 1.0 - h
            idx = int(np.argmax(slack))
            delta = min(r, slack[idx])
            if delta <= 0:
                break
            h[idx] += delta
            r -= delta
        else:
            idx = int(np.argmax(h))
            delta = min(-r, h[idx])
            if delta <= 0:
                break
            h[idx] -= delta
            r += delta
    h = np.clip(h, 0.0, 1.0)
    if not is_feasible(h, n):
        return np.full(n, 0.5)
    return h


def exact_score(h):
    n = len(h)
    if not is_feasible(h, n):
        return np.inf
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))


def smooth_value(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    cm = np.max(c)
    a = np.exp((c - cm) / tau)
    val = cm + tau * np.log(np.sum(a))
    return val if np.isfinite(val) else np.inf


def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    cm = np.max(c)
    a = np.exp((c - cm) / tau)
    p = a / np.sum(a)
    U = np.convolve(p, 1.0 - h, mode="full")[n - 1:2 * n - 1]
    V = np.convolve(p[::-1], h, mode="full")[n - 1:2 * n - 1]
    g = dx * (U - V)
    return g if np.all(np.isfinite(g)) else None


def build_starts(n, rng, parent):
    starts = []
    base = capped_simplex_project(parent, n)
    starts.append(base)
    starts.append(capped_simplex_project(base[::-1], n))
    starts.append(capped_simplex_project(1.0 - base[::-1], n))
    starts.append(np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    starts.append(capped_simplex_project(blk, n))
    alt = np.zeros(n)
    alt[0::2] = 1.0
    starts.append(capped_simplex_project(alt, n))
    for sd in (0.05, 0.15, 0.35):
        starts.append(capped_simplex_project(base + rng.normal(0, sd, n), n))
    starts.append(capped_simplex_project(rng.uniform(0, 1, n), n))
    dedup = []
    for s in starts:
        if not any(np.max(np.abs(s - d)) < 1e-10 for d in dedup):
            dedup.append(s)
    return dedup


def pgd_stage(x, tau, max_iter):
    n = len(x)
    alpha = 1.0
    Fx = smooth_value(x, tau)
    if not np.isfinite(Fx):
        return x
    g = smooth_grad(x, tau)
    if g is None:
        return x
    stall = 0
    for _ in range(max_iter):
        y = capped_simplex_project(x - alpha * g, n)
        D = np.sum((y - x) ** 2)
        if np.max(np.abs(y - x)) <= 1e-9:
            break
        Fy = smooth_value(y, tau)
        if not np.isfinite(Fy):
            alpha *= 0.5
            if alpha < 1e-12:
                break
            continue
        if Fy <= Fx - 1e-4 * D / alpha + 1e-14 * max(1.0, abs(Fx)):
            improve = Fx - Fy
            x = y
            Fx = Fy
            g = smooth_grad(x, tau)
            if g is None:
                break
            if improve <= 1e-12 * max(1.0, abs(Fx)):
                stall += 1
                if stall >= 5:
                    break
            else:
                stall = 0
            alpha = min(2.0, 1.25 * alpha)
        else:
            alpha *= 0.5
            if alpha < 1e-12:
                break
    return x
