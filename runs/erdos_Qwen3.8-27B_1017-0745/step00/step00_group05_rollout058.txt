            y = np.clip(y, 0.0, 1.0)
    r2 = target - np.sum(y)
    if abs(r2) > 5e-13:
        idx = np.where((y < 1.0 - 1e-14) if r2 > 0 else (y > 1e-14))[0]
        if len(idx) > 0:
            y[idx] += r2 / len(idx)
            y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(np.sum(y) - target) > 1e-10:
        return None
    return y


def exact_score(h, dx):
    if h.ndim != 1 or len(h) < 2 or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))


def check_feasible(h, target):
    return (h.ndim == 1 and np.all(np.isfinite(h)) and np.min(h) >= -1e-12
            and np.max(h) <= 1.0 + 1e-12 and abs(np.sum(h) - target) <= 1e-10)


def smooth_obj(h, tau, dx, lags):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    s = dx * c
    m = np.max(s)
    e = np.exp((s - m) / tau)
    se = np.sum(e)
    F = m + tau * np.log(se)
    p = e / se
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k] * dx
        if abs(w) < 1e-15:
            continue
        t = int(lags[k])
        js = max(0, t)
        je = min(n, t + n)
        if je > js:
            grad[js:je] += w * (1.0 - h[js - t:je - t])
        js2 = max(0, -t)
        je2 = min(n, n - t)
        if je2 > js2:
            grad[js2:je2] -= w * h[js2 + t:je2 + t]
    if not np.all(np.isfinite(grad)):
        return (F, None)
    return (F, grad)


def mass_fun(h):
    return np.array([np.sum(h) - len(h) / 2.0])


def mass_jac(h):
    return np.ones(len(h))


def optimize_tau(h0, tau_sched, dx, lags, bounds, cons, target):
    h = h0.copy()
    for tau in tau_sched:
        try:
            res = minimize(smooth_obj, h, args=(tau, dx, lags),
                           jac=True, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-10, 'disp': False})
        except Exception:
            continue
        if res is not None and hasattr(res, 'x') and res.x is not None:
            if res.x.shape == (len(h0),) and np.all(np.isfinite(res.x)):
                hp = capped_simplex_project(res.x, target)
                if hp is not None and check_feasible(hp, target):
                    Fn, _ = smooth_obj(hp, tau, dx, lags)
                    Fc, _ = smooth_obj(h, tau, dx, lags)
                    if np.isfinite(Fn) and (not np.isfinite(Fc) or Fn < Fc + 1e-10):
                        h = hp
    return h


def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = None
    n = 70
    try:
        pv = np.asarray(initial_h_values, dtype=np.float64)
        if pv.ndim == 1 and 2 <= len(pv) <= 999:
            n = len(pv)
            t0 = n / 2.0
            if np.all(np.isfinite(pv)) and np.min(pv) >= -1e-10 and np.max(pv) <= 1.0 + 1e-10:
                if abs(np.sum(pv) - t0) <= 1e-8:
                    parent = capped_simplex_project(pv.copy(), t0)
    except NameError:
        pass
    dx = 2.0 / n
    target = n / 2.0
    lags = np.arange(-(n - 1), n, dtype=np.int64)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': mass_fun, 'jac': mass_jac}]
    tau_sched = [0.02, 0.005, 0.001, 0.0002, 0.00005, 0.00001]

    starts = []
    if parent is not None:
        starts.append(parent.copy())
        starts.append(parent[::-1].copy())
        for sig in [0.04, 0.10, 0.22, 0.40]:
            starts.append(parent + sig * rng.standard_normal(n))
    else:
        starts.append(0.5 * np.ones(n))
    starts.append(rng.random(n))
    starts.append(rng.beta(0.5, 0.5, n))
    if parent is None:
        starts.append(rng.random(n))
        starts.append(rng.beta(0.5, 0.5, n))
        starts.append(0.5 * np.ones(n))

    proj_starts = []
    for s in starts:
        ps = capped_simplex_project(np.asarray(s, dtype=np.float64), target)
        if ps is not None and check_feasible(ps, target):
            proj_starts.append(ps)
    if len(proj_starts) == 0:
        proj_starts = [0.5 * np.ones(n)]

    inc_h = None
    inc_c = float('inf')
    for ps in proj_starts:
        sc = exact_score(ps, dx)
        if sc < inc_c - 1e-12:
            inc_c = sc
            inc_h = ps.copy()

    def try_update(h_cand):
        nonlocal_inc_c = None  # placeholder; handled below
        return None

    # Process each start through tau schedule
    for ps in proj_starts:
        ho = optimize_tau(ps, tau_sched, dx, lags, bounds, cons, target)
        sc = exact_score(ho, dx)
        if np.isfinite(sc) and sc < inc_c - 1e-12:
            inc_c = sc
            inc_h = ho.copy()

    # Adaptive restarts centered on incumbent
    center = inc_h if inc_h is not None else 0.5 * np.ones(n)
    for sig in [0.02, 0.08, 0.20]:
        pert = center + sig * rng.standard_normal(n)
        ps = capped_simplex_project(pert, target)
        if ps is None:
            continue
        ho = optimize_tau(ps, tau_sched, dx, lags, bounds, cons, target)
        sc = exact_score(ho, dx)
        if np.isfinite(sc) and sc < inc_c - 1e-12:
            inc_c = sc
            inc_h = ho.copy()
            center = inc_h.copy()

    # Final validation and fallback
    if inc_h is None or not check_feasible(inc_h, target):
        inc_h = 0.5 * np.ones(n)
        inc_c = exact_score(inc_h, dx)
    inc_h = capped_simplex_project(inc_h, target)
    if inc_h is None or not check_feasible(inc_h, target):
        inc_h = 0.5 * np.ones(n)
    inc_c = exact_score(inc_h, dx)
    return (inc_h, inc_c, n)
