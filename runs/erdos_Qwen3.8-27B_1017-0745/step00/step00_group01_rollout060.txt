def logsumexp_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    # Compute all lag values
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    c = dx * corr  # c[m] where m = d + (n-1)
    c_max = np.max(c)
    a = np.exp((c - c_max) / tau)
    s = np.sum(a)
    p = a / s  # p[m] = p_{m-(n-1)}
    
    # Gradient: grad[k] = dx * sum_d p[d] * [(1-h[k-d]) - h[k+d]] (with boundary checks)
    # d = m - (n-1)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        d = m - (n - 1)
        pm = p[m]
        if pm == 0:
            continue
        # (1-h[k-d]) for k where 0 <= k-d < n
        # k ranges from max(0, d) to min(n-1, n-1+d)
        k_lo = max(0, d)
        k_hi = min(n - 1, n - 1 + d)
        if k_lo <= k_hi:
            grad[k_lo:k_hi+1] += pm * (1.0 - h[k_lo:k_hi+1 - d])
        # -h[k+d] for k where 0 <= k+d < n
        # k ranges from max(0, -d) to min(n-1, n-1-d)

        k_hi2 = min(n - 1, n - 1 - d)
        if k_lo2 <= k_hi2:
            grad[k_lo2:k_hi2+1] -= pm * h[k_lo2:k_hi2+1 + d]
    
    return dx * grad

import numpy as np

def capped_simplex_project(z, n):
    s = n / 2.0
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return np.full(n, 0.5)
    z = np.clip(z, -1e12, 1e12)
    lo = np.min(z) - 1.0
    hi = np.max(z)
    if hi - lo < 1e-15:
        hn = np.full(n, s / n)
        return hn
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        hn = np.clip(z - mid, 0.0, 1.0)
        sn = np.sum(hn)
        if sn > s:
            lo = mid
        else:
            hi = mid
    mid = 0.5 * (lo + hi)
    hn = np.clip(z - mid, 0.0, 1.0)
    r = s - np.sum(hn)
    # Residual correction
    for _ in range(n + 2):
        if abs(r) < 1e-14:
            break
        if r > 0:
            slack = 1.0 - hn
            idx = np.argmax(slack)
            move = min(r, slack[idx])
            if move <= 1e-16:
                break
            hn[idx] += move
            r -= move
        else:
            slack = hn
            idx = np.argmax(slack)
            move = min(-r, slack[idx])
            if move <= 1e-16:
                break
            hn[idx] -= move
            r += move
    hn = np.clip(hn, 0.0, 1.0)
    if not np.all((hn >= 0) & (hn <= 1)) or abs(np.sum(hn) - s) > 1e-10 * max(1, n):
        return np.full(n, 0.5)
    return hn

def is_feasible(h, n):

        return False
    if not np.all(np.isfinite(h)):
        return False
    if not (np.all(h >= -1e-12) and np.all(h <= 1 + 1e-12)):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score(h):
    n = len(h)
    if not is_feasible(h, n):
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def logsumexp_value(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = dx * corr
    c_max = np.max(c)
    a = np.exp((c - c_max) / tau)
    return c_max + tau * np.log(np.sum(a))

def logsumexp_grad(h, tau):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = dx * corr
    c_max = np.max(c)
    a = np.exp((c - c_max) / tau)
    s_a = np.sum(a)
    if s_a <= 0 or not np.isfinite(s_a):
        return None
    p = a / s_a
    grad = np.zeros(n)

        d = m - (n - 1)
        pm = p[m]
        if pm < 1e-15:
            continue
        k_lo = max(0, d)
        k_hi = min(n - 1, n - 1 + d)
        if k_lo <= k_hi:
            grad[k_lo:k_hi + 1] += pm * (1.0 - h[k_lo - d:k_hi - d + 1])

        if k_lo2 <= k_hi2:
            grad[k_lo2:k_hi2 + 1] -= pm * h[k_lo2 + d:k_hi2 + d + 1]

def capped_simplex_project(z, n):
    s = n / 2.0
    if z.shape != (n,) or not np.all(np.isfinite(z)):

    z = np.clip(z, -1e12, 1e12)

        else:
            hi = mid
