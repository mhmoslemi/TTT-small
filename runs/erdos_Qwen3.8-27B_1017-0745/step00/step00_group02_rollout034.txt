    m = 2 * n - 1
    G = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        shift = k - (n - 1)
        for j in range(n):
            g = 0.0
            jp = j + shift
            if 0 <= jp < n:
                g += (1.0 - h[jp])
            jm = j - shift
            if 0 <= jm < n:
                g -= h[jm]
            G[k, j] = dx * g
    return G

        final_s = float(np.max(np.correlate(final_h, 1.0 - final_h, mode='full'))) * dx
    return final_h, float(final_s), int(n)
