        env = np.max(a[None, :] + b[None, :] * batch[:, None] + c[None, :] * batch[:, None] ** 2, axis=1)
        j = np.argmin(env)
        if env[j] < best_v:
            best_v = float(env[j])
            best_a = float(batch[j])
    return best_a

def local_search(start, dx, n, target, rng, sweeps):
    h = np.array(start, dtype=np.float64)
    cur_score = exact_score(h, dx)
    best = h.copy()
    best_score = cur_score
    for _ in range(sweeps):
        dirs = build_directions(h, n, rng)
        for d in dirs:
            if d is None:
                continue
            ba = line_search(h, d, dx, n)
            if ba is None:
                continue
            if abs(ba) < 1e-15:
                continue
            prop = capped_simplex_project(h + ba * d, target)
            if prop is None or not is_feasible(prop, n, target):
                continue
            ps = exact_score(prop, dx)
            if ps < cur_score - 1e-12:
                h, cur_score = prop, ps
                if cur_score < best_score:
                    best, best_score = h.copy(), cur_score
        if best_score <= 0.3808:
            break
    return best, best_score

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    n_target = 70
    parent = None
    fallback = None
    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            pv = np.array(initial_h_values, dtype=np.float64)
            if pv.ndim == 1 and 2 <= len(pv) <= 999:
                n = len(pv)
                n_target = n
                if np.all(np.isfinite(pv)) and np.max(np.abs(pv)) <= 1e6:
                    parent = pv.copy()
                    if np.min(pv) >= -1e-10 and np.max(pv) <= 1.0 + 1e-10:
                        t = float(n) / 2.0
                        if abs(np.sum(pv) - t) <= 1e-8:
                            fallback = capped_simplex_project(pv, t)
    except Exception:
        pass
    n = n_target
    dx = 2.0 / n
    target = float(n) / 2.0
    u = np.full(n, 0.5)
    starts = []
    if parent is not None:
        proj = capped_simplex_project(parent, target)
        if proj is None:
            proj = u.copy()
        starts.append(proj)
        starts.append(capped_simplex_project(parent[::-1], target) if capped_simplex_project(parent[::-1], target) is not None else u.copy())
        for sc in (0.04, 0.12):
            z = capped_simplex_project(parent + sc * rng.standard_normal(n), target)
            if z is not None:
                starts.append(z)
        ru = capped_simplex_project(rng.uniform(0, 1, n), target)
        if ru is not None:
            starts.append(ru)
        rb = capped_simplex_project(np.array([rng.beta(0.5, 0.5) for _ in range(n)]), target)
        if rb is not None:
            starts.append(rb)
    starts.append(u.copy())
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    starts.append(blk)
    if parent is None:
        for _ in range(2):
            ru = capped_simplex_project(rng.uniform(0, 1, n), target)
            if ru is not None:
                starts.append(ru)
            rb = capped_simplex_project(np.array([rng.beta(0.5, 0.5) for _ in range(n)]), target)
            if rb is not None:
                starts.append(rb)
    # deduplicate
    deduped = []
    for s in starts:
        if s is None:
            continue
        dup = False
        for d in deduped:
            if np.max(np.abs(s - d)) < 1e-10:
                dup = True
                break
        if not dup:
            deduped.append(s)
    starts = deduped
    incumbent = fallback if fallback is not None else u.copy()
    inc_score = exact_score(incumbent, dx)
    scored = []
    for s in starts:
        sc = exact_score(s, dx)
        scored.append((sc, s))
        if sc < inc_score - 1e-12:
            incumbent, inc_score = s.copy(), sc
    scored.sort(key=lambda x: x[0])
    selected = [incumbent]
    for sc, s in scored:
        if len(selected) >= 3:
            break
        if all(np.max(np.abs(s - x)) > 1e-10 for x in selected):
            selected.append(s)
    sweeps_max = 2 if n <= 96 else 1
    for s in selected:
        res_h, res_sc = local_search(s, dx, n, target, rng, sweeps_max)
        if res_sc < inc_score - 1e-12:
            incumbent, inc_score = res_h, res_sc
        if inc_score <= 0.3808:
            break
    restarts = 2 if n <= 96 else 1
    for sigma in (0.02, 0.08)[:restarts]:
        rp = capped_simplex_project(incumbent + sigma * rng.standard_normal(n), target)
        if rp is None:
            continue
        res_h, res_sc = local_search(rp, dx, n, target, rng, 1)
        if res_sc < inc_score - 1e-12:
            incumbent, inc_score = res_h, res_sc
        if inc_score <= 0.3808:
            break
    final = capped_simplex_project(incumbent, target)
    if final is None or not is_feasible(final, n, target):
        final = fallback if fallback is not None else u.copy()
    final = np.clip(final, 0.0, 1.0)
    final_score = exact_score(final, dx)
    if not np.isfinite(final_score):
        final = u.copy()
        final_score = exact_score(final, dx)
    return (final, final_score, n)
