                    if y is None or not is_feasible(y, n):
                        R *= 0.25
                        rejc += 1
                        continue
                best_z = None
                best_zs = float('inf')
                best_al = 0.0
                for ai in range(9):
                    alpha = 0.5 ** ai
                    z = x + alpha * (y - x)
                    z = project_capped(z, n)
                    if z is None or not is_feasible(z, n):
                        continue
                    zs = exact_score(z, n)
                    if zs < cx - 1e-12 and zs < best_zs:
                        best_z = z
                        best_zs = zs
                        best_al = alpha
                if best_z is None:
                    R *= 0.25
                    rejc += 1
                    continue
                old_x = x
                actual = cx - best_zs
                Mx = float(np.max(c + G @ (best_z - x)))
                pred = cx - Mx
                rho = actual / pred if pred > 1e-12 else 0.0
                x = best_z.copy()
                cx = best_zs
                rejc = 0
                if cx < best_score - 1e-12:
                    best = x.copy()
                    best_score = cx
                if rho < 0.1:
                    R *= 0.5
                elif rho >= 0.75 and best_al == 1.0:
                    disp = float(np.sum(np.abs(best_z - old_x)))
                    if disp >= 0.8 * R:
                        R = min(2.0 * R, Rmax)
    fp = project_capped(best, n)
    if fp is not None and is_feasible(fp, n):
        fs = exact_score(fp, n)
        if fs <= best_score + 1e-9:
            best = fp
            best_score = fs
    if best_score > exact_score(fb, n) + 1e-9:
        best = fb
        best_score = exact_score(fb, n)
    best_score = exact_score(best, n)
    return best, best_score, n
