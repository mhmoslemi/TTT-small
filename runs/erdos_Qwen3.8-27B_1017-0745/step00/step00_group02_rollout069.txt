def _smooth_grad(h, n, tau):
    dx = 2.0 / n
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(q)
    e = np.exp((q - m) / tau)
    p = e / np.sum(e)
    g = np.zeros(n)
    for j in range(n):
        # k ranges from j to j+n-1
        k_lo, k_hi = j, j + n  # k in [j, j+n-1]
        # h[k-j] for k in [j, j+n-1] is h[0], h[1], ..., h[n-1] = h itself
        # So g[j] = dx * sum_{k=j}^{j+n-1} p[k] * (1 - 2*h[k-j])
        # = dx * sum_{r=0}^{n-1} p[j+r] * (1 - 2*h[r])
        g[j] = dx * np.dot(p[j:j+n], 1.0 - 2.0 * h)
    return g

_SLSQP = {}

def _proj(y, n, S):
    y = np.asarray(y, dtype=np.float64)
    if not np.all(np.isfinite(y)) or len(y) != n:
        return np.full(n, S / n)
    lo, hi = float(np.min(y)) - 1.0, float(np.max(y)) + 1.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    d = S - float(np.sum(h))
    if d > 1e-12:
        slack = np.maximum(1.0 - h, 0.0)
        ts = float(np.sum(slack))
        if ts > 0:
            h += d * slack / ts
    elif d < -1e-12:
        pos = np.maximum(h, 0.0)
        tp = float(np.sum(pos))
        if tp > 0:
            h += d * pos / tp
    h = np.clip(h, 0.0, 1.0)

    if len(h) != n or not np.all(np.isfinite(h)):
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _lags(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _sv(q, tau):
    m = float(np.max(q))
    if not np.isfinite(m):
        return np.inf
    e = np.exp((q - m) / tau)
    return m + tau * float(np.log(np.sum(e)))

def _sg(h, n, tau):
    dx = 2.0 / n
    q = _lags(h, n)
    m = float(np.max(q))
    e = np.exp((q - m) / tau)
    p = e / float(np.sum(e))
    g = np.zeros(n)
    for j in range(n):

            float(np.min(h)) >= -1e-9 and float(np.max(h)) <= 1.0 + 1e-9 and
            abs(float(np.sum(h)) - S) <= 1e-9)

def _pgd(h0, n, S, taus, mx):

        a = 1.0
        f = _sv(_lags(h, n), tau)

def _tobj_jac(z):
    J = np.zeros(_SLSQP['n'] + 1)
