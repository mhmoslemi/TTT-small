def pair_transfer_polish(h, n_iter, rng):
    n = len(h)
    h = h.copy()
    best = exact_score(h)
    best_h = h.copy()
    stages = [(0.08, 0.001), (0.025, 0.0003), (0.008, 0.00008), (0.002, 0.0)]
    per_stage = max(1, n_iter // 4)
    for step_lim, T in stages:
        for _ in range(per_stage):
            i, j = rng.choice(n, size=2, replace=False)
            sign = 1.0 if rng.random() < 0.5 else -1.0
            if sign > 0:
                cap = min(1.0 - h[i], h[j])
            else:
                cap = min(h[i], 1.0 - h[j])
            mag = rng.random() * min(step_lim, cap)
            if mag < 1e-12:
                continue
            trial = h.copy()
            trial[i] += sign * mag
            trial[j] -= sign * mag
            if trial[i] < -1e-12 or trial[i] > 1.0 + 1e-12 or trial[j] < -1e-12 or trial[j] > 1.0 + 1e-12:
                continue
            trial = np.clip(trial, 0.0, 1.0)
            s_trial = exact_score(trial)
            s_curr = exact_score(h)
            delta = s_trial - s_curr
            if delta <= 0 or (T > 0 and rng.random() < math.exp(-delta / T)):
                h = trial
                if s_trial < best - 1e-12:
                    best = s_trial
                    best_h = trial.copy()
    return best_h, best

        grad[lo2:hi2] -= wd * h[lo:hi]
    return F, grad

def optimize_start(h0, dx, tau_schedule, B):
    n = len(h0)
    best = exact_score(h0)
    best_h = h0.copy()
    h = h0.copy()
    n_levels = len(tau_schedule)
    B_per = max(1, B // n_levels)
    for lv in range(n_levels):
        tau = tau_schedule[lv]
        eta = 1.0
        F_prev, g_prev = surrogate_and_grad(h, dx, tau)
        stall = 0
        for _ in range(B_per):
            if not np.all(np.isfinite(g_prev)):
                break
            trial = project_capped_simplex(h - eta * g_prev, 0.5 * n)
            dv = trial - h
            if np.max(np.abs(dv)) < 1e-14:
                break
            gd = float(np.dot(g_prev, dv))
            if gd >= 0:
                eta *= 0.5
                if eta < 1e-8:
                    break
                continue
            F_trial, G_trial = surrogate_and_grad(trial, dx, tau)
            if not np.isfinite(F_trial):
                eta *= 0.5
                if eta < 1e-8:
                    break
                continue
            armijo = F_prev + 1e-4 * gd + 1e-12 * max(1.0, abs(F_prev))
            if F_trial <= armijo:
                imp = F_prev - F_trial
                h = trial
                F_prev = F_trial
                g_prev = G_trial
                eta = min(8.0, 1.2 * eta)
                if abs(imp) < 1e-10 * max(1.0, abs(F_prev)):
                    stall += 1
                    if stall >= 12:
                        break
                else:
                    stall = 0
                s = exact_score(h)
                if s < best - 1e-12:
                    best = s
                    best_h = h.copy()
            else:
                eta *= 0.5
                if eta < 1e-8:
                    break
    return best_h, best

def pair_transfer_polish(h, n_iter, rng):
    n = len(h)
    h = h.copy()
    best = exact_score(h)
    best_h = h.copy()
    stages = [(0.08, 0.001), (0.025, 0.0003), (0.008, 0.00008), (0.002, 0.0)]
    per_stage = max(1, n_iter // 4)
    for step_lim, T in stages:
        s_curr = best
        for _ in range(per_stage):
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            sign = 1.0 if rng.random() < 0.5 else -1.0
            if sign > 0:
                cap = min(1.0 - h[i], h[j])
            else:
                cap = min(h[i], 1.0 - h[j])
            mag = rng.random() * min(step_lim, cap)
            if mag < 1e-12:
                continue
            trial = h.copy()
            trial[i] += sign * mag
            trial[j] -= sign * mag
            if trial[i] < -1e-12 or trial[i] > 1.0 + 1e-12 or trial[j] < -1e-12 or trial[j] > 1.0 + 1e-12:
                continue
            trial = np.clip(trial, 0.0, 1.0)
            s_trial = exact_score(trial)
            delta = s_trial - s_curr
            if delta <= 1e-15 or (T > 0 and rng.random() < math.exp(-delta / T)):
                h = trial
                s_curr = s_trial
                if s_trial < best - 1e-12:
                    best = s_trial
                    best_h = h.copy()
    return best_h, best

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = None
    pv = globals().get('initial_h_values', None)
    if pv is not None:
        try:
            p = np.array(pv, dtype=np.float64).flatten()
            if p.size >= 2 and np.all(np.isfinite(p)):
                parent = p
        except Exception:
            pass
    if parent is not None:
        n = parent.size
        dx = 2.0 / n
        S = 0.5 * n
    else:
        n = int(kwargs.get('n_points', 80))
        n = max(8, min(512, n))
        dx = 2.0 / n
        S = 0.5 * n
        parent = np.full(n, 0.5)
    h_parent = project_capped_simplex(parent, S)
    best = exact_score(h_parent)
    best_h = h_parent.copy()
    starts = [h_parent.copy()]
    if parent is not None and n <= 512:
        for sd in [0.04, 0.15, 0.35]:
            starts.append(project_capped_simplex(parent + rng.normal(0, sd, n), S))
    q = int(rng.integers(1, n))
    phi = float(rng.uniform(0, 2 * math.pi))
    ii = np.arange(n, dtype=np.float64)
    periodic = 0.5 + 0.45 * np.cos(2 * math.pi * q * ii / n + phi)
    starts.append(project_capped_simplex(periodic, S))
    starts.append(project_capped_simplex(rng.random(n), S))
    perm = rng.permutation(n)
    binary = np.zeros(n)
    binary[perm[:int(n // 2)]] = 1.0
    if n % 2 == 1:
        binary[perm[int(n // 2)]] = 0.5
    starts.append(binary)
    if parent is not None:
        max_starts = 7 if n <= 256 else 3
    else:
        max_starts = 5 if n <= 256 else 3
    unique_starts = [starts[0]]
    for s in starts[1:]:
        if all(float(np.max(np.abs(s - u))) > 1e-10 for u in unique_starts):
            unique_starts.append(s)
    unique_starts = unique_starts[:max_starts]
    for s in unique_starts:
        sc = exact_score(s)
        if sc < best - 1e-12:
            best = sc
            best_h = s.copy()
    baseline = best
    B = min(480, max(12, 4000000 // (n * n)))
    tau_sched = [0.02, 0.004, 0.0008, 0.00016, 0.000032, 0.0000064]
    for s in unique_starts:
        try:
            oh, os_ = optimize_start(s.copy(), dx, tau_sched, B)
            if os_ < best - 1e-12:
                best = os_
                best_h = oh.copy()
        except Exception:
            pass
    if best > baseline - 1e-8 and n <= 256:
        for sd in [0.6, 1.0]:
            try:
                ns = project_capped_simplex(best_h + rng.normal(0, sd, n), S)
                oh, os_ = optimize_start(ns, dx, tau_sched, max(1, B // 2))
                if os_ < best - 1e-12:
                    best = os_
                    best_h = oh.copy()
            except Exception:
                pass
    P = max(80, min(800, 10000000 // (n * n)))
    try:
        best_h, best = pair_transfer_polish(best_h, P, rng)
    except Exception:
        pass
    best_h = project_capped_simplex(best_h, S)
    sb = exact_score(best_h)
    sf = exact_score(h_parent)
    if sf < sb:
        best_h, best = h_parent.copy(), sf
    else:
        best = sb
    return (best_h, best, n)
