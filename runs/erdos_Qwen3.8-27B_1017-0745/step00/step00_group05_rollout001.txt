        if best_score <= 0.3808:
            break
        restart = capped_simplex_project(best_h + scale * rng.standard_normal(n), target)
        if restart is None:
            continue
        rsc = exact_score(restart, dx)
        if rsc < best_score - 1e-12:
            best_score = rsc
            best_h = restart.copy()
        res_h, res_score = slp_search(restart, dx, n, rng)
        if res_score < best_score - 1e-12:
            best_score = res_score
            best_h = res_h.copy()
        poll_h, poll_score = pairwise_poll(res_h, dx, n, rng)
        if poll_score < best_score - 1e-12:
            best_score = poll_score
            best_h = poll_h.copy()
    final = capped_simplex_project(best_h, target)
    if final is None or not is_feasible(final, n):
        final = fallback
    final_score = exact_score(final, dx)
    if not np.isfinite(final_score) or not is_feasible(final, n):
        final = u
        final_score = exact_score(final, dx)
    return final, final_score, n
