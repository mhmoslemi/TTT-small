            else:
                nsp[i] = frozen[i]
                nsc[i] = fs[i]
                _nf[i] = ff[i]
                _nc[i] = fc[i]

        pop = nsp
        scores = nsc
        Fs = _nf
        CRs = _nc
        stagnation = stagnation + 1 if not gi else 0

        in_pop = False
        for i in range(P):
            if np.max(np.abs(pop[i] - best)) < 1e-12:
                in_pop = True
                break
        if not in_pop:
            wi = int(np.argmax(scores))
            pop[wi] = best
            scores[wi] = best_score
            Fs[wi] = 0.6
            CRs[wi] = 0.5

        if stagnation >= re_thresh and n_re < max_re:
            n_re += 1
            sigma = 0.06 if n_re == 1 else 0.18
            o2 = np.argsort(scores)
            keep = max(1, math.ceil(P // 2))
            for sl in range(keep, P):
                if (sl - keep) % 2 == 0:
                    y = best + sigma * rng.standard_normal(n)
                    cand = project_k(y, n, S)
                    if cand is None:
                        cand = best.copy()
                else:
                    m = modes[int(rng.integers(0, 5))]
                    s2 = scales[int(rng.integers(0, 5))]
                    cand = gen_candidate(rng, n, best, m, s2)
                pop[sl] = cand
                scores[sl] = exact_score(cand, n)
                Fs[sl] = rng.uniform(0.4, 0.9)
                CRs[sl] = rng.uniform(0.25, 0.95)
                if scores[sl] < best_score - 1e-12 and check_feasible(cand, n):
                    best = cand.copy()
                    best_score = scores[sl]
            stagnation = 0

    final = best.copy()
    final_score = exact_score(final, n)
    if not check_feasible(final, n):
        p = project_k(final, n, S)
        if p is not None and check_feasible(p, n):
            final, final_score = p, exact_score(p, n)
        else:
            final, final_score = fallback.copy(), exact_score(fallback, n)
    if final_score > fb_score:
        final, final_score = fallback.copy(), exact_score(fallback, n)
    if not check_feasible(final, n):
        final = np.full(n, 0.5)
        final_score = exact_score(final, n)
    return final, final_score, int(n)
