    return J

def _add_start(starts, h, n, S):
    if h is None:
        return
    h = np.asarray(h, dtype=np.float64)
    if h.ndim != 1 or len(h) != n or not np.all(np.isfinite(h)):
        return
    p = _project_capped(h, S)
    if p is None:
        return
    for s in starts:
        if np.max(np.abs(p - s)) < 1e-10:
            return
    starts.append(p)

def _slsqp_solve(h0, n, S):
    B0, _ = _lags(h0)
    t0 = float(np.max(B0)) + 1e-8 * max(1.0, abs(float(np.max(B0))))
    z0 = np.concatenate([h0, [t0]])
    cons = [{'type': 'ineq', 'fun': _ineq_f, 'jac': _ineq_j},
            {'type': 'eq', 'fun': _eq_f, 'jac': _eq_j}]
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    res = minimize(_obj, z0, jac=_obj_g, constraints=cons, bounds=bounds,
                   method='SLSQP', options={'maxiter': 200, 'ftol': 1e-10, 'disp': False})
    if res is not None and hasattr(res, 'x') and res.x is not None:
        hx = np.asarray(res.x[:n], dtype=np.float64)
        if hx.ndim == 1 and len(hx) == n and np.all(np.isfinite(hx)):
            return _project_capped(hx, S)
    return None

def run(seed=42, **kwargs):
    n = 80
    try:
        if 'n_points' in kwargs and kwargs['n_points'] is not None:
            n_alt = int(kwargs['n_points'])
            if 8 <= n_alt <= 512:
                n = n_alt
    except (ValueError, TypeError):
        pass
    S = n / 2.0
    parent = None
    try:
        ihv = initial_h_values
        if ihv is not None and np.asarray(ihv).ndim == 1 and len(np.asarray(ihv)) >= 2:
            n = len(np.asarray(ihv))
            S = n / 2.0
            iv = np.asarray(ihv, dtype=np.float64)
            if np.all(np.isfinite(iv)) and np.max(np.abs(iv)) <= 1e12:
                p = _project_capped(iv, S)
                if p is not None:
                    parent = p.copy()
    except Exception:
        pass
    h_const = np.full(n, 0.5, dtype=np.float64)
    rng = np.random.default_rng(int(seed) if seed is not None else 42)
    starts = []
    _add_start(starts, parent, n, S)
    _add_start(starts, h_const, n, S)
    if parent is not None:
        _add_start(starts, parent[::-1].copy(), n, S)
        _add_start(starts, parent + rng.normal(0, 0.18, n), n, S)
        qf = int(rng.integers(1, min(7, max(2, (n - 1) // 2)) + 1))
        phi = float(rng.uniform(0, 2 * np.pi))
        idx = np.arange(n, dtype=np.float64)
        _add_start(starts, 0.5 + 0.45 * np.cos(2 * np.pi * qf * idx / n + phi), n, S)
        perm = rng.permutation(n)
        bv = np.zeros(n)
        bv[perm[:int(n // 2)]] = 1.0
        if n % 2 == 1:
            bv[perm[int(n // 2)]] = 0.5
        _add_start(starts, bv, n, S)
    if parent is None:
        for _ in range(2):
            qf = int(rng.integers(1, 5))
            phi = float(rng.uniform(0, 2 * np.pi))
            idx = np.arange(n, dtype=np.float64)
            _add_start(starts, 0.5 + 0.45 * np.cos(2 * np.pi * qf * idx / n + phi), n, S)
        _add_start(starts, 0.5 + 0.30 * rng.standard_normal(n), n, S)
        _add_start(starts, rng.uniform(0, 1, n), n, S)
        perm = rng.permutation(n)
        bv = np.zeros(n)
        bv[perm[:int(n // 2)]] = 1.0
        if n % 2 == 1:
            bv[perm[int(n // 2)]] = 0.5
        _add_start(starts, bv, n, S)
    att = 0
    while len(starts) < 6 and att < 18:
        base = starts[att % len(starts)] if starts else h_const
        _add_start(starts, base + rng.normal(0, 0.22, n), n, S)
        att += 1
    if not starts:
        starts = [h_const.copy()]
    best_h = None
    best_sc = np.inf
    for h0 in starts:
        sc = _score(h0)
        if sc is not None and sc < best_sc:
            best_sc = sc
            best_h = h0.copy()
    if n <= 128:
        max_starts = min(len(starts), 6)
        for si in range(max_starts):
            try:
                hx = _slsqp_solve(starts[si], n, S)
                if hx is not None:
                    sc = _score(hx)
                    if sc is not None and sc < best_sc:
                        best_sc = sc
                        best_h = hx.copy()
            except Exception:
                pass
    pool = []
    if best_h is not None:
        pool.append(best_h)
    if parent is not None:
        pool.append(parent)
    pool.append(h_const)
    final_h = None
    final_sc = np.inf
    for c in pool:
        c = np.asarray(c, dtype=np.float64)
        if c.ndim != 1 or len(c) != n or not np.all(np.isfinite(c)):
            continue
        p = _project_capped(c, S)
        if p is None:
            continue
        sc = _score(p)
        if sc is not None and sc < final_sc:
            final_sc = sc
            final_h = p
    if final_h is None:
        final_h = h_const
        final_sc = _score(final_h)
        if final_sc is None:
            final_sc = 0.5
    return final_h, float(final_sc), n
