import numpy as np
import math

def project_box_sum(z, N):
    z = np.asarray(z, dtype=np.float64)
    if z.shape != (N,) or not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    m = N / 2.0
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    e = m - float(np.sum(h))
    if abs(e) > 1e-13:
        for i in range(N):
            if abs(e) < 1e-15:
                break
            if e > 0:
                room = min(e, 1.0 - h[i])
                h[i] += room
                e -= room
            else:
                room = min(-e, h[i])
                h[i] -= room
                e += room
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - m) > 1e-9 * max(1, N):
        return np.full(N, 0.5)
    return h

def lag_values(h):
    N = len(h)
    M = 2 * N - 1
    q = np.zeros(M, dtype=np.float64)
    for k in range(-(N - 1), N):
        idx = k + N - 1
        if k >= 0:
            L = N - k
            q[idx] = np.dot(h[k:k+L], 1.0 - h[0:L])
        else:
            d = -k
            L = N - d
            q[idx] = np.dot(h[0:L], 1.0 - h[d:d+L])
    return q

def smooth_value_grad(h, tau, N):
    dx = 2.0 / N
    q = lag_values(h)
    r = dx * q
    rmax = float(np.max(r))
    ex = np.exp((r - rmax) / tau)
    s = float(np.sum(ex))
    p = ex / s
    phi = rmax + tau * math.log(s)
    grad = np.zeros(N, dtype=np.float64)

        w = dx * p[k + N - 1]
        if w < 1e-15:
            continue
        if k >= 0:
            L = N - k
            a = h[k:k+L]
            b = h[0:L]
            grad[k:k+L] += w * (1.0 - b)

            d = -k
            L = N - d
            a = h[0:L]
            b = h[d:d+L]
            grad[0:L] += w * (1.0 - b)
            grad[d:d+L] -= w * a
