    except Exception:
        rng = np.random.default_rng(42)
    starts = [best_h.copy()]
    max_starts = 6 if n <= 128 else 3
    fm = min(6, max(1, (n - 1) // 2))
    if parent is not None:
        starts.append(parent[::-1].copy())
        hp = _project(parent + 0.18 * rng.standard_normal(n), S)
        if hp is not None:
            starts.append(hp)
        hp2 = _project(_periodic(n, int(rng.integers(1, fm + 1)), float(rng.uniform(0, 6.2832))), S)
        if hp2 is not None:
            starts.append(hp2)
        hb = _project(_binary(n, rng), S)
        if hb is not None:
            starts.append(hb)
    else:
        hu = _project(rng.uniform(0, 1, n), S)
        if hu is not None:
            starts.append(hu)
        for _ in range(2):
            hp = _project(_periodic(n, int(rng.integers(1, fm + 1)), float(rng.uniform(0, 6.2832))), S)
            if hp is not None:
                starts.append(hp)
        hn = _project(0.5 + 0.30 * rng.standard_normal(n), S)
        if hn is not None:
            starts.append(hn)
        hb = _project(_binary(n, rng), S)
        if hb is not None:
            starts.append(hb)
    vstarts = []
    for s in starts:
        dup = any(np.max(np.abs(s - v)) < 1e-10 for v in vstarts)
        if not dup and _feasible(s, n):
            vstarts.append(s.copy())
            if len(vstarts) >= max_starts:
                break
    for s in vstarts:
        sc = _score(s)
        if sc is not None and sc < best_sc:
            best_sc = sc
            best_h = s.copy()
    if n <= 128:
        bnds = [(0.0, 1.0)] * n + [(None, None)]
        ic = {'fun': _ineq, 'jac': _ineq_jac, 'type': 'ineq'}
        ec = {'fun': _eq, 'jac': _eq_jac, 'type': 'eq'}
        for s in vstarts:
            B0, _ = _lags(s)
            t0 = float(np.max(B0)) + 1e-8 * max(1.0, abs(float(np.max(B0))))
            z0 = np.concatenate([s, [t0]])
            try:
                res = minimize(_obj, z0, jac=_obj_grad, method='SLSQP',
                               bounds=bnds, constraints=[ic, ec],
                               options={'maxiter': 180, 'ftol': 1e-9, 'disp': False})
                if res is not None and res.x is not None:
                    zr = np.asarray(res.x, dtype=np.float64)
                    if zr.shape == (n + 1,) and np.all(np.isfinite(zr)):
                        hr = _project(zr[:n], S)
                        if hr is not None and _feasible(hr, n):
                            sc = _score(hr)
                            if sc is not None and sc < best_sc:
                                best_sc = sc
                                best_h = hr.copy()
            except Exception:
                pass
    if not _feasible(best_h, n):
        bh = _project(best_h, S)
        best_h = bh if bh is not None else h_const.copy()
    rm = np.sum(best_h) - S
    if abs(rm) > 1e-12:
        best_h = np.clip(best_h - rm / n, 0.0, 1.0)
    fs = _score(best_h)
    if fs is None:
        best_h = h_const.copy()
        fs = _score(best_h)
    return (best_h.astype(np.float64), fs, n)
