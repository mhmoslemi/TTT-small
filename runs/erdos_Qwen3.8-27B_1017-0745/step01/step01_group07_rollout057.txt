- We want to minimize max_k ∫ h(x)(1-h(x+k)) dx
- The k=0 term is ∫ h(x)(1-h(x)) dx = ∫ h - ∫ h² = 1 - ∫h²

- The danger is intermediate lags where h and (1-h) have significant overlap.

import numpy as np

def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    n = len(z)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > n or np.max(np.abs(z)) > 1e12:
        return None
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    y = np.clip(z - lam, 0.0, 1.0)
    r = S - np.sum(y)
    step = 1 if r > 0 else -1
    for i in range(n):
        if r * step <= 0:
            break
        if step > 0:
            a = min(r, 1.0 - y[i])
            y[i] += a
            r -= a
        else:
            a = min(-r, y[i])
            y[i] -= a
            r += a
    if abs(np.sum(y) - S) > 1e-10 or not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1 + 1e-12):
        return None
    return np.clip(y, 0.0, 1.0)

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-9)

def exact_score(h, n):
    vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(vals)):
        return None
    return float(np.max(vals))

def pair_line_search(h, cur_score, p, r, n):
    L = max(-h[p], h[r] - 1.0)
    U = min(1.0 - h[p], h[r])
    if U - L <= 1e-12:
        return None
    best_t = 0.0
    best_f = cur_score
    # Evaluate at 17 points + t=0
    ts = np.linspace(L, U, 17)
    if 0.0 not in ts:  # ensure t=0 is included
        ts = np.append(ts, 0.0)
    for t in ts:
        hn = h.copy()
        hn[p] += t
        hn[r] -= t
        f = exact_score(hn, n)
        if f is not None and (f < best_f - 1e-14 or (abs(f - best_f) <= 1e-14 and abs(t) < abs(best_t))):
            best_f = f
            best_t = t
    # 4 refinement rounds
    radius = (U - L) / 16.0
    for _ in range(4):
        lo_r = max(L, best_t - radius)
        hi_r = min(U, best_t + radius)
        for t in np.linspace(lo_r, hi_r, 5):
            hn = h.copy()
            hn[p] += t
            hn[r] -= t
            f = exact_score(hn, n)
            if f is not None and (f < best_f - 1e-14 or (abs(f - best_f) <= 1e-14 and abs(t) < abs(best_t))):
                best_f = f
                best_t = t
        radius /= 4.0
    if best_f >= cur_score - 1e-12:
        return None
    hn = h.copy()
    hn[p] += best_t
    hn[r] -= best_t
    if not is_feasible(hn, n):
        hn2 = project_capped_simplex(hn, n / 2.0)
        if hn2 is None:
            return None
        sc = exact_score(hn2, n)
        if sc is None or sc >= cur_score - 1e-12:
            return None
        return hn2, sc
    return hn, best_f

def polish_pairs(h, cur_score, n, rng, max_sweeps):
    P = min(n * (n - 1) // 2, max(24, 16 * n))
    pairs = []
    # adjacent pairs
    adj = min(n - 1, max(1, P // 2))
    if adj <= n - 1:
        idx = np.linspace(0, n - 2, adj, dtype=int)
        for i in idx:
            pairs.append((i, i + 1))
    # random pairs
    remaining = P - len(pairs)
    tries = 0
    used = set(pairs)
    while len(pairs) < P and tries < remaining * 5 + 100:
        tries += 1
        i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
        if i == j:
            continue
        a, b = min(i, j), max(i, j)
        if (a, b) in used:
            continue
        used.add((a, b))
        pairs.append((a, b))
    rng.shuffle(pairs)
    h_cur = h.copy()
    sc_cur = cur_score
    for sweep in range(max_sweeps):
        accepted = 0
        for (p, r) in pairs:
            res = pair_line_search(h_cur, sc_cur, p, r, n)
            if res is not None:
                h_cur, sc_cur = res
                accepted += 1
        if accepted == 0:
            break
    return h_cur, sc_cur

def anneal_trajectory(h, cur_score, n, rng, epochs, K):
    h_cur = h.copy()
    sc_cur = cur_score
    scale = 0.12
    for e in range(epochs):
        T = max(5e-5, 3e-3 * (0.4 ** e))
        accepted = 0
        for _ in range(K):
            p, r = int(rng.integers(0, n)), int(rng.integers(0, n))
            if p == r:
                continue
            t = float(rng.uniform(-scale, scale))
            hn = h_cur.copy()
            hn[p] += t
            hn[r] -= t
            if hn[p] < -1e-12 or hn[p] > 1 + 1e-12 or hn[r] < -1e-12 or hn[r] > 1 + 1e-12:
                continue
            if abs(np.sum(hn) - n / 2.0) > 1e-9:
                continue
            sc = exact_score(hn, n)
            if sc is None:
                continue
            delta = sc - sc_cur
            if delta <= 0 or (np.log(max(rng.random(), 1e-300)) > -delta / T if T > 0 else delta <= 0):
                h_cur = hn
                sc_cur = sc
                accepted += 1
        ratio = accepted / max(K, 1)
        if ratio < 0.12:
            scale = min(0.4, 1.25 * scale)
        elif ratio > 0.55:
            scale = max(0.01, 0.75 * scale)
        # one polish sweep
        h_cur, sc_cur = polish_pairs(h_cur, sc_cur, n, rng, 1)
    return h_cur, sc_cur

def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    n = len(z)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > n or np.max(np.abs(z)) > 1e12:
        return None
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    y = np.clip(z - lam, 0.0, 1.0)

        if r * step <= 0:
            break
