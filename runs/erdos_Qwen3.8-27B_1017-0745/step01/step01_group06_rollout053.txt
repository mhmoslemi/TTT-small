- J[k, :n] = -G[k, :]
- J[k, n] = 1

For the equality constraint sum(d) = 0:

import numpy as np
from scipy.optimize import minimize

def project_K(y, n, S):
    if not np.all(np.isfinite(y)):
        return np.full(n, S / n)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    err = S - h.sum()
    if abs(err) > 1e-12:
        if err > 0:
            idx = np.where(1.0 - h > 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) > 0:
            h[idx] = np.clip(h[idx] + err / len(idx), 0.0, 1.0)
    if abs(h.sum() - S) > 1e-8:
        return np.full(n, S / n)
    return h

def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def lag_vals(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def build_G(h, n, dx):
    G = np.zeros((2 * n - 1, n))
    j = np.arange(n)
    for k in range(2 * n - 1):
        ell = k - (n - 1)
        if ell == 0:
            G[k, :] = dx * (1.0 - 2.0 * h)
        else:
            i1 = j - ell
            v1 = (i1 >= 0) & (i1 < n)
            i2 = j + ell
            v2 = (i2 >= 0) & (i2 < n)
            G[k, :] = 0.0
            G[k, v1] += dx * (1.0 - h[i1[v1]])
            G[k, v2] -= dx * h[i2[v2]]
    return G

def tr_obj(z, eta):
    d = z[:-1]
    return z[-1] + np.dot(d, d) / (2.0 * eta)

def tr_obj_jac(z, eta):
    d = z[:-1]
    g = d / eta
    g = np.append(g, 1.0)
    return g

def tr_ineq(z, q, G):
    d = z[:-1]
    return z[-1] - q - G @ d

def tr_ineq_jac(z, q, G):
    nl = len(q)
    J = np.empty((nl, len(q) + 1))
    J[:, :-1] = -G
    J[:, -1] = 1.0
    return J

def tr_eq(z):
    return z[:-1].sum()

def tr_eq_jac(z):
    g = np.ones(len(z))
    g[-1] = 0.0
    return g

        for i in range(n0):
            lo = max(i / n0, j / m)
            hi = min((i + 1) / n0, (j + 1) / m)
            if hi > lo:
                s += hp[i] * (hi - lo)
        out[j] = m * s
    return project_K(out, m, m / 2.0)

def tr_obj(z, eta):
    d = z[:-1]
    return float(z[-1] + np.dot(d, d) / (2.0 * eta))

def tr_obj_jac(z, eta):
    return np.append(z[:-1] / eta, 1.0)

def tr_ineq(z, q, G):
    return z[-1] - q - G @ z[:-1]

def tr_ineq_jac(z, q, G):
    nl = len(q)
    J = np.empty((nl, nl + 1))
    J[:, :-1] = -G
    J[:, -1] = 1.0
    return J

def tr_eq(z):
    return float(z[:-1].sum())

def tr_eq_jac(z):
    g = np.ones(len(z))
    g[-1] = 0.0
    return g

def local_solve(h0, n, dx, S, delta0, eta0, max_it, si_max):
    h = project_K(h0.copy(), n, S)
    delta, eta, fail = delta0, eta0, 0
    for _ in range(max_it):
        q = lag_vals(h, dx)
        G = build_G(h, n, dx)
        dlo = np.maximum(-delta, -h + 1e-10)
        dhi = np.minimum(delta, 1.0 - h - 1e-10)
        if np.any(dlo >= dhi):
            delta *= 0.5
            fail += 1
            if fail >= 8 or delta < 1e-5:
                break
            continue
        bnds = list(zip(dlo, dhi)) + [(None, None)]
        cons = [{'type': 'ineq', 'fun': tr_ineq, 'jac': tr_ineq_jac, 'args': (q, G)},
                {'type': 'eq', 'fun': tr_eq, 'jac': tr_eq_jac}]
        z0 = np.concatenate([np.zeros(n), [float(q.max())]])
        try:
            res = minimize(tr_obj, z0, jac=tr_obj_jac, args=(eta,), method='SLSQP',
                           bounds=bnds, constraints=cons,
                           options={'maxiter': si_max, 'ftol': 1e-10, 'disp': False})
        except Exception:
            delta *= 0.5
            fail += 1
            if fail >= 8 or delta < 1e-5:
                break
            continue
        if not hasattr(res, 'x') or res.x is None or not np.all(np.isfinite(res.x)):
            delta *= 0.5
            fail += 1
            if fail >= 8 or delta < 1e-5:
                break
            continue
        d_s = res.x[:-1]
        if np.abs(d_s).max() > delta + 1e-4 or abs(d_s.sum()) > 1e-4:
            delta *= 0.5
            fail += 1
            if fail >= 8 or delta < 1e-5:
                break
            continue
        ht = project_K(h + d_s, n, S)
        if not feasible(ht, n, S):
            delta *= 0.5
            fail += 1
            if fail >= 8 or delta < 1e-5:
                break
            continue
        da = ht - h
        if np.abs(da).max() < 1e-9:
            break
        pred = float(q.max()) - float((q + G @ da).max())
        actual = float(q.max()) - exact_score(ht, dx)
        if actual > 1e-12 and (pred <= 1e-12 or (pred > 1e-12 and actual / pred >= 0.05)):
            h = ht
            fail = 0
            if pred > 0:
                ratio = actual / pred
                if ratio > 0.6 and np.abs(da).max() >= 0.7 * delta:
                    delta = min(0.5, 2.0 * delta)
                    eta = min(2.0, 1.4 * eta)
                elif ratio < 0.2:
                    delta *= 0.7
        else:
            fail += 1
            delta *= 0.5
            eta = max(1e-4, 0.7 * eta)
        if fail >= 10 or delta < 1e-5:
            break
    return project_K(h, n, S)

def make_starts(base, n, S, rng, is_parent):
    starts = [project_K(base, n, S), project_K(1.0 - base, n, S), project_K(base[::-1], n, S)]
    starts.append(project_K(base + 0.08 * rng.standard_normal(n), n, S))
    starts.append(project_K(base + 0.25 * rng.standard_normal(n), n, S))
    j = np.arange(n)
    starts.append(project_K(0.5 + 0.45 * np.cos(2.0 * np.pi * j / n), n, S))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    starts.append(project_K(blk, n, S))
    starts.append(project_K(rng.uniform(0, 1, n), n, S))
    if is_parent:
        perm = rng.permutation(n)
        starts.append(project_K(base[perm], n, S))
    seen, dedup = [], []
    for s in starts:
        dup = False
        for sh in seen:
            if np.abs(s - sh).max() <= 1e-6:
                dup = True
                break
        if not dup:
            seen.append(s)
            dedup.append(s)
    return dedup

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    init = globals().get('initial_h_values', None)
    ok = False
    if init is not None:
        init = np.array(init, dtype=np.float64).ravel()
        if 2 <= len(init) <= 999 and np.all(np.isfinite(init)) and np.max(np.abs(init)) < 1e6:
            ok = True
    n0 = len(init) if ok else 45
    if ok:
        init = project_K(init.copy(), n0, n0 / 2.0)
    best_h, best_n, best_s = None, n0, float('inf')
    resolv = []
    if ok:
        for m in [n0, max(2, n0 - 1), min(999, n0 + 1)]:
            if m not in resolv:
                resolv.append(m)
    else:
        nd = kwargs.get('n_points', 45)
        if isinstance(nd, (int, np.integer)) and 2 <= nd <= 999:
            resolv.append(int(nd))
        else:
            resolv.append(45)
    for m in resolv:
        dx, S = 2.0 / m, m / 2.0
        if ok and m == n0:
            base = init.copy()
        elif ok:
            base = transfer(init, n0, m)
        else:
            base = np.full(m, 0.5)
        ip = (m == n0 and ok)
        starts = make_starts(base, m, S, rng, ip)
        for s in starts:
            if not feasible(s, m, S):
                continue
            sc = exact_score(s, dx)
            if sc < best_s - 1e-12:
                best_s, best_h, best_n = sc, s.copy(), m
        max_it = 25 if m <= 96 else (12 if m <= 256 else 5)
        si_max = 50 if m <= 96 else (30 if m <= 256 else 15)
        n_start = min(len(starts), 7 if m <= 96 else 3)
        for s in starts[:n_start]:
            if not feasible(s, m, S):
                continue
            ho = project_K(local_solve(s, m, dx, S, 0.12, 0.5, max_it, si_max), m, S)
            if feasible(ho, m, S):
                sc = exact_score(ho, dx)
                if sc < best_s - 1e-12:
                    best_s, best_h, best_n = sc, ho.copy(), m
        if m <= 96:
            hr = project_K(best_h if best_n == m else base + 0.18 * rng.standard_normal(m), m, S)
            ho = project_K(local_solve(hr, m, dx, S, 0.10, 0.4, max_it, si_max), m, S)
            if feasible(ho, m, S):
                sc = exact_score(ho, dx)
                if sc < best_s - 1e-12:
                    best_s, best_h, best_n = sc, ho.copy(), m
    if best_h is None or not feasible(best_h, best_n, best_n / 2.0):
        fbn, fbdx = n0, 2.0 / n0
        best_h = project_K(init.copy() if ok else np.full(fbn, 0.5), fbn, fbn / 2.0)
        best_n = fbn
    best_h = project_K(best_h, best_n, best_n / 2.0)
    final_s = exact_score(best_h, 2.0 / best_n)
    if not np.isfinite(final_s):
        fb = np.full(n0, 0.5)
        return (fb, float(np.max(np.correlate(fb, 0.5, mode='full')) * 2.0 / n0), n0)
    return (best_h, final_s, best_n)
