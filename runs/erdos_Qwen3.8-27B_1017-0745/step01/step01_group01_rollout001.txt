def project_capped(z, n):
    s = n / 2.0
    if not np.all(np.isfinite(z)):
        return None
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if abs(hi - lo) < 1e-15:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(n + 2):
        if abs(r) < 1e-12 * max(1, n):
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            if 1.0 - h[idx] <= 1e-15:
                break
            h[idx] += min(r, 1.0 - h[idx])
        else:
            idx = int(np.argmax(h))
            if h[idx] <= 1e-15:
                break
            h[idx] += max(r, -h[idx])
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(np.sum(h) - s) > 1e-10 * max(1, n):
        return None
    return h

        return float('inf')

        return float('inf')

        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_vals_grads(h, n):
    L = 2 * n - 1
    v = np.zeros(L)
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        idx = np.arange(lo, hi)
        jdx = idx + d
        v[row] = float(np.sum(h[idx] * (1.0 - h[jdx])))
        G[row, idx] += (1.0 - h[jdx])
        G[row, jdx] -= h[idx]
    return v, G

def _obj(z, n):
    return z[-1]

def _obj_jac(z, n):
    j = np.zeros(n + 1)
    j[-1] = 1.0
    return j

def _eq_fun(z, n):
    return np.sum(z[:n]) - n / 2.0

def _eq_jac(z, n):
    j = np.ones(n + 1)
    j[-1] = 0.0
    return j

def _ineq_fun(z, n):
    h = z[:n]
    t = z[-1]
    v, _ = lag_vals_grads(h, n)
    return t - (2.0 / n) * v

def _ineq_jac(z, n):
    h = z[:n]
    _, G = lag_vals_grads(h, n)
    L = 2 * n - 1
    j = np.zeros((L, n + 1))
    j[:, :n] = -(2.0 / n) * G
    j[:, n] = 1.0
    return j

def _add_start(starts, v, n, dedup_tol=1e-8):
    h2 = project_capped(v, n)
    if h2 is None:
        return
    for ex in starts:
        if float(np.max(np.abs(h2 - ex))) < dedup_tol:
            return
    starts.append(h2)

def _pairwise_polish(x, n, sweeps):
    best = x.copy()
    best_s = exact_score(best, n)
    for _ in range(sweeps):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                lo = max(-best[i], best[j] - 1.0)
                hi = min(1.0 - best[i], best[j])
                if lo >= hi - 1e-14:
                    continue
                cands = [0.0, lo + 1e-12, hi - 1e-12]
                for step in (0.001, 0.003, 0.01, 0.03, 0.1, 0.3):
                    for sgn in (1, -1):
                        dd = sgn * step
                        if lo <= dd <= hi:
                            cands.append(dd)
                best_d, best_sc = 0.0, best_s
                for dd in cands:
                    dd = max(lo, min(hi, dd))
                    trial = best.copy()
                    trial[i] += dd
                    trial[j] -= dd
                    sc = exact_score(trial, n)
                    if sc < best_sc - 1e-12:
                        best_d, best_sc = dd, sc
                if best_d != 0.0:
                    best[i] += best_d
                    best[j] -= best_d
                    best_s = best_sc
                    improved = True
        if not improved:
            break
    return best, best_s
