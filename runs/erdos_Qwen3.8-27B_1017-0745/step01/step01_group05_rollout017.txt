                    best_s = s
                    best_h = hc.copy()
            else:
                scores.append(float('inf'))
        order = np.argsort(scores)
        selected = []
        for i in order:
            if scores[i] < 1e30:
                selected.append(i)
            if len(selected) == mu:
                break
        if len(selected) < mu:
            break
        m_new = np.zeros(d)
        for i, idx in enumerate(selected):
            m_new += w[i] * samples[idx]
        raw_m = anchor + rho * (B @ m_new)
        if np.all(np.isfinite(raw_m)):
            hm = project_capped(raw_m, n)
            if check_feasible(hm, n):
                sm = exact_score(hm, n)
                if sm < best_s - 1e-13:
                    best_s = sm
                    best_h = hm.copy()
        y_w = (m_new - m) / max(sigma, 1e-12)
        p_sigma = (1 - c_sigma) * p_sigma + math.sqrt(c_sigma * (2 - c_sigma) * mu_eff) * (C_inv_sqrt @ y_w)
        denom = math.sqrt(max(1e-30, 1 - (1 - c_sigma) ** (2 * (g + 1))))
        h_sigma = 1.0 if np.linalg.norm(p_sigma) / denom < (1.4 + 2.0 / (d + 1)) * chi_d else 0.0
        p_c = (1 - c_c) * p_c + h_sigma * math.sqrt(c_c * (2 - c_c) * mu_eff) * y_w
        C_new = (1 - c_1 - c_mu + c_1 * (1 - h_sigma) * c_c * (2 - c_c)) * C + c_1 * np.outer(p_c, p_c)
        for i, idx in enumerate(selected):
            y_i = (samples[idx] - m) / max(sigma, 1e-12)
            C_new += c_mu * w[i] * np.outer(y_i, y_i)
        ratio = np.linalg.norm(p_sigma) / max(chi_d, 1e-12)
        sigma *= math.exp(np.clip(c_sigma / d_sigma * (ratio - 1), -2, 2))
        sigma = float(np.clip(sigma, 1e-4, 0.5))
        m = np.clip(m_new, -30, 30)
        C = 0.5 * (C_new + C_new.T)
        if rho * sigma * math.sqrt(float(np.max(ev))) < 1e-5:
            stagn += 1
        else:
            stagn = 0
        if stagn >= 5:
            break
    return best_h, best_s

def local_polish(h, n, rng, n_iter=500):
    best_h = h.copy()
    best_s = exact_score(h, n)
    for _ in range(n_iter):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = rng.uniform(-0.15, 0.15)
        trial = best_h.copy()
        trial[i] += delta
        trial[j] -= delta
        if trial[i] < -1e-12 or trial[i] > 1 + 1e-12 or trial[j] < -1e-12 or trial[j] > 1 + 1e-12:
            continue
        trial = project_capped(trial, n)
        s = exact_score(trial, n)
        if s < best_s - 1e-13:
            best_s = s
            best_h = trial.copy()
    return best_h, best_s

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent = globals().get('initial_h_values', None)
    n = 45
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.ndim == 1 and 2 <= len(p) <= 999 and np.all(np.isfinite(p)) and np.max(np.abs(p)) < 1e6:
            n = len(p)
            if check_feasible(p, n):
                fallback = p.copy()
            else:
                fallback = project_capped(p, n)
        else:
            fallback = np.full(n, 0.5)
    else:
        np_kw = kwargs.get('n_points', None)
        if isinstance(np_kw, int) and not isinstance(np_kw, bool) and 2 <= np_kw <= 999:
            n = np_kw
        fallback = np.full(n, 0.5)
    fallback = project_capped(fallback, n)
    best_h = fallback.copy()
    best_s = exact_score(fallback, n)
    if n <= 64:
        B = helmert_basis(n)
        d = n - 1
        rho = math.sqrt(n / d)
    else:
        d = min(48, n - 1)
        B = random_basis(n, d, rng)
        rho = math.sqrt(n / d)
    if n <= 100:
        pop_size = max(10, min(16, 4 + int(3 * math.log(max(d, 2)))))
        ngen = 40
        n_restarts = 3
    elif n <= 400:
        pop_size = 12
        ngen = 14
        n_restarts = 2
    else:
        pop_size = 10
        ngen = 8
        n_restarts = 2
    anchors = [best_h.copy(), 1.0 - best_h, best_h[::-1].copy(), 1.0 - best_h[::-1].copy()]
    unique_anchors = []
    for a in anchors:
        a = project_capped(a, n)
        dup = False
        for u in unique_anchors:
            if np.max(np.abs(a - u)) < 1e-10:
                dup = True
                break
        if not dup:
            unique_anchors.append(a)
    z = rng.standard_normal(n)
    z -= z.mean()
    zn = float(np.sqrt(np.sum(z ** 2)))
    if zn > 1e-12:
        pert = project_capped(best_h + 0.10 * z / zn, n)
        dup = False
        for u in unique_anchors:
            if np.max(np.abs(pert - u)) < 1e-6:
                dup = True
                break
        if not dup:
            unique_anchors.append(pert)
    sigmas = [0.05, 0.12, 0.04]
    for r in range(min(n_restarts, len(unique_anchors))):
        try:
            anchor = unique_anchors[r]
            sa = exact_score(anchor, n)
            if sa < best_s - 1e-13:
                best_s = sa
                best_h = anchor.copy()
            sigma0 = sigmas[r % len(sigmas)]
            if n > 64 and r > 0:
                B_r = random_basis(n, d, rng)
            else:
                B_r = B
            rho_r = rho
            bh, bs = cma_restart(anchor, B_r, rho_r, sigma0, ngen, pop_size, n, rng)
            if bs < best_s - 1e-13:
                best_s = bs
                best_h = bh.copy()
        except Exception:
            pass
    try:
        ph, ps = local_polish(best_h, n, rng, 600)
        if ps < best_s - 1e-13:
            best_s = ps
            best_h = ph.copy()
    except Exception:
        pass
    final_h = project_capped(best_h, n)
    if not check_feasible(final_h, n):
        final_h = fallback.copy()
    final_s = exact_score(final_h, n)
    fb_s = exact_score(fallback, n)
    if final_s > fb_s:
        final_h = fallback.copy()
        final_s = fb_s
    return final_h.astype(np.float64), float(final_s), int(n)
