        amax, d_star = -1e30, 0

            qq = x[q0:q0+m]
            up = u[p0:p0+m]
            uq = u[q0:q0+m]
            a_d = dx * (np.dot(p, 1.0-qq) + np.dot(1.0-qq, up) - np.dot(p, uq))
            if a_d > amax:

        p0, q0 = max(d_star, 0), max(-d_star, 0)
        m = n - abs(d_star)

        qq = x[q0:q0+m]

        g[q0:q0+m] -= dx * p

            break
