            c = exact_score(trial)
            if c < best_c:
                best = trial.copy()
                best_c = c
            d = c - cur_c
            if d <= 0:
                cur, cur_c = trial, c
                acc += 1
            elif d / T < 745 and rng.random() < math.exp(-d / T):
                cur, cur_c = trial, c
                acc += 1
            if tot % 50 == 0 and tot > 0:
                r = acc / tot
                sigma = min(0.25, max(0.003, sigma * math.exp(0.5 * (r - 0.25))))
                acc, tot = 0, 0
        T *= 0.55
        sigma = max(0.003, sigma * 0.75)
        rep = project_box_sum(cur, 0.5 * n)
        if rep is not None and check_feasible(rep, n) and np.max(np.abs(rep - cur)) < 1e-8:
            cur = rep
            cur_c = exact_score(cur)
            if cur_c < best_c:
                best = cur.copy()
                best_c = cur_c
    return best, best_c

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent, n = get_parent()
    fallback = parent.copy()
    fb_c = exact_score(fallback)
    best = parent.copy()
    best_c = exact_score(parent)

    scored = []
    for s in build_starts(parent, n, rng):
        c = exact_score(s)
        scored.append((c, s))
    scored.sort(key=_scored_key)
    for c, s in scored:
        if c < best_c:
            best = s.copy()
            best_c = c

    n_cycles = 4 if n <= 64 else (3 if n <= 256 else 2)
    for cyc in range(n_cycles):
        try:
            if cyc == 0:
                h0 = scored[0][1].copy()
            elif cyc == 1 and len(scored) > 1:
                h0 = scored[1][1].copy()
            else:
                sd = 0.10 if cyc == 2 else 0.25
                h0 = project_box_sum(best + sd * rng.standard_normal(n), 0.5 * n)
                if h0 is None or not check_feasible(h0, n):
                    h0 = best.copy()
            c0 = exact_score(h0)
            if c0 < best_c:
                best, best_c = h0.copy(), c0
            lb, lc = anneal_cycle(h0, n, rng)
            if lc < best_c:
                best, best_c = lb.copy(), lc
        except Exception:
            pass

    rep = project_box_sum(best, 0.5 * n)
    if rep is not None and check_feasible(rep, n):
        best = rep
    final_c = exact_score(best)
    if final_c < fb_c:
        return (best, final_c, n)
    return (fallback, fb_c, n)
