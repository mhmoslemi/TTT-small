def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-14) and np.all(h <= 1.0 + 1e-14)
            and abs(np.sum(h) - n / 2.0) <= 1e-10)

def exact_lags(h, n):
    vals = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode='full') * (2.0 / n)
    if vals.shape[0] != 2 * n - 1 or not np.all(np.isfinite(vals)):
        return None, None
    return vals, float(np.max(vals))

def line_eval(q, b, a, t):
    return float(np.max(q + b * t + a * t * t))

def build_starts(n, S, parent, rng):
    starts = [np.full(n, 0.5)]
    if parent is not None:
        starts.append(parent.copy())
        starts.append(1.0 - parent.copy())
        starts.append(parent[::-1].copy())
    idx = np.arange(n, dtype=np.float64)
    M = min(4, (n - 1) // 2)
    u = np.full(n, 0.5)
    for m in range(1, M + 1):
        u += rng.normal(0, 0.12 / np.sqrt(m)) * np.cos(2 * np.pi * m * idx / n + rng.uniform(0, 6.2832))
    pu = proj_simplex(u, S)
    if pu is not None and is_feasible(pu, n):
        starts.append(pu)
    perm = rng.permutation(n)
    bb = np.zeros(n); bb[perm[:n // 2]] = 1.0
    if n % 2 == 1: bb[perm[n // 2]] = 0.5
    starts.append(bb.copy())
    u2 = rng.uniform(0, 1, n)
    p2 = proj_simplex(u2, S)
    if p2 is not None and is_feasible(p2, n):
        starts.append(p2)
    ded = []
    for s in starts:
        s = np.asarray(s, dtype=np.float64)
        dup = False
        for d in ded:
            if np.max(np.abs(s - d)) < 1e-10: dup = True; break
        if not dup and is_feasible(s, n):
            ded.append(s)
    cap = 6 if n <= 128 else 4 if n <= 256 else 3
    return ded[:cap]

def pair_line_search(h, q, score, p, r, n, dx):
    L = max(-h[p], h[r] - 1.0)
    U = min(1.0 - h[p], h[r])
    if U - L <= 1e-12:
        return None
    d_arr = np.arange(2 * n - 1, dtype=np.int64) - (n - 1)
    i1 = r - d_arr; i2 = p - d_arr; i3 = p + d_arr; i4 = r + d_arr
    b = np.zeros(2 * n - 1, dtype=np.float64)
    m1 = (i1 >= 0) & (i1 < n); b[m1] += h[i1[m1]]
    m2 = (i2 >= 0) & (i2 < n); b[m2] -= h[i2[m2]]
    m3 = (i3 >= 0) & (i3 < n); b[m3] -= h[i3[m3]]
    m4 = (i4 >= 0) & (i4 < n); b[m4] += h[i4[m4]]
    b *= dx
    a = np.zeros(2 * n - 1, dtype=np.float64)
    a[n - 1] = -2.0 * dx
    a[n - 1 + (r - p)] += dx
    a[n - 1 + (p - r)] += dx
    ts = np.unique(np.append(np.linspace(L, U, 17), 0.0))
    best_v = 1e30; best_t = 0.0
    for t in ts:
        v = line_eval(q, b, a, float(t))
        if v < best_v - 1e-15 or (abs(v - best_v) <= 1e-15 and abs(t) < abs(best_t)):
            best_v = v; best_t = float(t)
    if best_v >= score - 1e-12:
        return None
    radius = (U - L) / 16.0
    for _ in range(4):
        lo_r = max(L, best_t - radius); hi_r = min(U, best_t + radius)
        rt = np.unique(np.append(np.linspace(lo_r, hi_r, 5), best_t))
        for t in rt:
            v = line_eval(q, b, a, float(t))
            if v < best_v - 1e-15 or (abs(v - best_v) <= 1e-15 and abs(t) < abs(best_t)):
                best_v = v; best_t = float(t)
        radius /= 4.0
    if best_v >= score - 1e-12:
        return None
    hnew = h.copy(); hnew[p] += best_t; hnew[r] -= best_t
    if not is_feasible(hnew, n):
        hnew = proj_simplex(hnew, n / 2.0)
        if hnew is None or not is_feasible(hnew, n):
            return None
    qn, sn = exact_lags(hnew, n)
    if qn is None or sn is None or sn >= score - 1e-12:
        return None
    return hnew, sn, qn

def polish(h, score, n, dx, rng, max_sweeps):
    q, _ = exact_lags(h, n)
    if q is None:
        return h, score
    P = min(n * (n - 1) // 2, 2 * n, max(24, 80000 // n))
    for _ in range(max_sweeps):
        accepted = 0
        pairs = []
        adj = min(n - 1, max(1, P // 2))
        ai = np.linspace(0, n - 2, adj, dtype=int) if P // 2 < n - 1 else np.arange(n - 1)
        for i in ai:
            pairs.append((int(i), int(i) + 1))
        seen = set(pairs)
        attempts = 0
        while len(pairs) < P and attempts < 5000:
            attempts += 1
            i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
            if i == j: continue
            if i > j: i, j = j, i
            if (i, j) in seen: continue
            seen.add((i, j)); pairs.append((i, j))
        rng.shuffle(pairs)
        for (p, r_) in pairs:
            res = pair_line_search(h, q, score, p, r_, n, dx)
            if res is not None:
                h, score, q = res; accepted += 1
        if accepted == 0:
            break
    return h, score

def anneal(h, score, n, dx, rng, epochs):
    K = min(1000, max(80, 80000 // n))
    scale = 0.12
    for e in range(epochs):
        T = max(5e-5, 3e-3 * (0.4 ** e))
        acc = 0
        for _ in range(K):
            p = int(rng.integers(0, n)); r = int(rng.integers(0, n))
            if p == r: continue
            L = max(-h[p], h[r] - 1.0); U = min(1.0 - h[p], h[r])
            t = rng.uniform(-scale, scale)
            if t < L or t > U: continue
            hnew = h.copy(); hnew[p] += t; hnew[r] -= t
            if not is_feasible(hnew, n):
                hnew = proj_simplex(hnew, n / 2.0)
                if hnew is None or not is_feasible(hnew, n): continue
            qn, sn = exact_lags(hnew, n)
            if qn is None: continue
            delta = sn - score
            if delta <= 0:
                h, score = hnew, sn; acc += 1
            elif delta < 700 * T and rng.random() < np.exp(-delta / T):
                h, score = hnew, sn; acc += 1
        ratio = acc / max(1, K)
        if ratio < 0.12: scale = min(0.4, 1.25 * scale)
        elif ratio > 0.55: scale = max(0.01, 0.75 * scale)
        h, score = polish(h, score, n, dx, rng, 1)
    return h, score

def run(seed=42, **kwargs):
    if seed is None: seed = 42
    try: seed = int(seed) % (2 ** 32)
    except (TypeError, ValueError): seed = 42
    rng = np.random.default_rng(seed)
    n = None; parent = None
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        try:
            iv_arr = np.asarray(iv, dtype=np.float64)
            if iv_arr.ndim == 1 and 2 <= iv_arr.size <= 999:
                n = iv_arr.size; S = n / 2.0
                if is_feasible(iv_arr, n):
                    parent = iv_arr.copy()
                else:
                    pr = proj_simplex(iv_arr, S)
                    if pr is not None and is_feasible(pr, n): parent = pr
        except Exception: pass
    if n is None:
        try: n = int(kwargs.get('n_points', 80))
        except (TypeError, ValueError): n = 80
        if n < 8 or n > 999: n = 80
    S = n / 2.0; dx = 2.0 / n
    const = np.full(n, 0.5)
    best_h = const.copy()
    best_score = exact_score_fn(const, n)
    if best_score is None or not np.isfinite(best_score): best_score = 0.5
    if parent is not None:
        ps = exact_score_fn(parent, n)
        if ps is not None and np.isfinite(ps) and ps < best_score:
            best_score = ps; best_h = parent.copy()
    starts = build_starts(n, S, parent, rng)
    for s in starts:
        sc = exact_score_fn(s, n)
        if sc is not None and np.isfinite(sc) and sc < best_score:
            best_score = sc; best_h = s.copy()
    n_sw = 4 if n <= 128 else 3 if n <= 256 else 2 if n <= 512 else 1
    n_ep = 5 if n <= 128 else 4 if n <= 256 else 3 if n <= 512 else 2
    for s in starts:
        try:
            hh = s.copy(); ss = exact_score_fn(hh, n)
            if ss is None or not np.isfinite(ss): continue
            hh, ss = anneal(hh, ss, n, dx, rng, n_ep)
            hh, ss = polish(hh, ss, n, dx, rng, n_sw)
            if ss is not None and np.isfinite(ss) and ss < best_score - 1e-12:
                best_score = ss; best_h = hh.copy()
        except Exception: pass
    try:
        bh, bs = polish(best_h, best_score, n, dx, rng, n_sw)
        if bs is not None and np.isfinite(bs) and bs < best_score:
            best_score = bs; best_h = bh.copy()
    except Exception: pass
    try:
        comp = 1.0 - best_h.copy()
        cs = exact_score_fn(comp, n)
        if cs is not None and is_feasible(comp, n):
            ch, cs2 = polish(comp, cs, n, dx, rng, n_sw)
            if cs2 is not None and np.isfinite(cs2) and cs2 < best_score:
                best_score = cs2; best_h = ch.copy()
    except Exception: pass
    if not is_feasible(best_h, n):
        best_h = proj_simplex(best_h, S)
        if best_h is None: best_h = const.copy()
    fs = exact_score_fn(best_h, n)
    if fs is None or not np.isfinite(fs):
        if parent is not None and is_feasible(parent, n):
            best_h = parent.copy(); fs = exact_score_fn(parent, n)
        if fs is None or not np.isfinite(fs):
            best_h = const.copy(); fs = 0.5
    return best_h, float(fs), n

def exact_score_fn(h, n):
    _, s = exact_lags(h, n)
    return s
