def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(np.sum(h) - n / 2.0) <= 1e-9)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def lag_values(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def lag_jac(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        p0, q0, m = max(d, 0), max(-d, 0), n - abs(d)
        G[d + n - 1, p0:p0 + m] += dx * (1.0 - h[q0:q0 + m])
        G[d + n - 1, q0:q0 + m] -= dx * h[p0:p0 + m]
    return G

    return z[n]

def obj_jac(z, n):
    g = np.zeros(n + 1); g[n] = 1.0
    return g

def inq_fun(z, n):

    if not np.all(np.isfinite(h)):
        return np.full(2 * n - 1, -1e10)
    return t - lag_values(h, n)

    if not np.all(np.isfinite(h)):
        return np.zeros((2 * n - 1, n + 1))
    G = lag_jac(h, n)
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def optimize_epigraph(start, n, maxiter=220):

    if h0 is None:
        return None, float('inf')
    cur = h0.copy()
    cur_sc = exact_score(cur, n)

import numpy as np
from scipy.optimize import minimize

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    if abs(r) > 1e-13:
        if r > 0:
            for i in np.argsort(h):
                if r <= 1e-14: break
                d = min(r, 1.0 - h[i]); h[i] += d; r -= d
        else:
            for i in np.argsort(-h):
                if r >= -1e-14: break
                d = max(r, -h[i]); h[i] += d; r -= d
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - s) > 1e-9:
        return None
    return h

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(np.sum(h) - n / 2.0) <= 1e-9)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def lag_values(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def lag_jac(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        p0, q0, m = max(d, 0), max(-d, 0), n - abs(d)
        G[d + n - 1, p0:p0 + m] += dx * (1.0 - h[q0:q0 + m])
        G[d + n - 1, q0:q0 + m] -= dx * h[p0:p0 + m]
    return G

def obj_fn(z, n):
    return z[n]

def obj_jac(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def inq_fun(z, n):
    h = z[:n]
    t = z[n]
    if not np.all(np.isfinite(h)):
        return np.full(2 * n - 1, -1e10)
    return t - lag_values(h, n)

def inq_jac(z, n):
    h = z[:n]
    if not np.all(np.isfinite(h)):
        return np.zeros((2 * n - 1, n + 1))
    G = lag_jac(h, n)
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def eq_fun(z, n):
    return np.sum(z[:n]) - n / 2.0

def eq_jac(z, n):
    J = np.zeros(n + 1)
    J[:n] = 1.0
    return J

def optimize_epigraph(start, n, maxiter=300, n_pass=3):
    h0 = project_capped(start, n)
    if h0 is None:
        return None, float('inf')
    cur = h0.copy()
    cur_sc = exact_score(cur, n)
    for _p in range(n_pass):
        t0 = float(np.max(lag_values(cur, n))) + 1e-8
        z0 = np.concatenate([cur, [t0]])
        try:
            res = minimize(obj_fn, z0, args=(n,), jac=obj_jac, method='SLSQP',
                           constraints=[
                               {'type': 'ineq', 'fun': inq_fun, 'jac': inq_jac, 'args': (n,)},
                               {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac, 'args': (n,)},
                           ],
                           bounds=[(0, 1)] * n + [(0, 1)],
                           options={'ftol': 1e-10, 'maxiter': maxiter})
        except Exception:
            break
        if res.x is None or res.x.shape != (n + 1,) or not np.all(np.isfinite(res.x)):
            break
        y = project_capped(res.x[:n], n)
        if y is None:
            break
        best_z, best_s = cur, cur_sc
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
            z = project_capped(cur + alpha * (y - cur), n)
            if z is None:
                continue
            sz = exact_score(z, n)
            if sz < best_s:
                best_s, best_z = sz, z
        if best_s >= cur_sc - 1e-12:
            break
        cur = best_z.copy()
        cur_sc = best_s
    return cur, cur_sc

def active_grad(h, n, tol=1e-3):
    dx = 2.0 / n
    q = lag_values(h, n)
    mx = float(np.max(q))
    active = np.where(q >= mx - tol)[0]
    g = np.zeros(n)
    for idx in active:
        d = int(idx) - (n - 1)
        p0, q0, m = max(d, 0), max(-d, 0), n - abs(d)
        g[p0:p0 + m] += dx * (1.0 - h[q0:q0 + m])
        g[q0:q0 + m] -= dx * h[p0:p0 + m]
    g /= float(len(active))
    return g, q

def subgradient_refine(h, n, n_iter=40):
    dx = 2.0 / n
    h = h.copy().astype(np.float64)
    sc = exact_score(h, n)
    if not np.isfinite(sc):
        return h, sc
    eta = 0.15 / dx
    eta_max = 1.5 / dx
    fails = 0
    for _ in range(n_iter):
        if fails >= 8 or eta < 1e-9 / dx:
            break
        g, _q = active_grad(h, n, tol=1e-3)
        z = project_capped(h - eta * g, n)
        if z is None:
            eta *= 0.5; fails += 1; continue
        v = z - h
        if np.max(np.abs(v)) < 1e-11:
            eta *= 0.5; fails += 1; continue
        best_z, best_s = None, sc
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            zz = project_capped(h + alpha * v, n)
            if zz is None:
                continue
            sz = exact_score(zz, n)
            if sz < best_s:
                best_s, best_z = sz, zz
        if best_z is None or best_s >= sc - 1e-13:
            eta *= 0.5; fails += 1
        else:
            h, sc, fails = best_z, best_s, 0
            eta = min(1.15 * eta, eta_max)
    return h, sc

def build_starts(fb, n, rng):

        dup = False
