import numpy as np
from scipy.optimize import minimize

def project_box_sum(z, N, m):
    if z.shape != (N,) or not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    if np.max(np.abs(z)) > 1e12:
        return np.full(N, 0.5)
    z = z - np.median(z)
    lo, hi = z.min() - 1.0, z.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    for _ in range(2):
        if r > 0:
            for i in range(N):
                adj = min(r, 1.0 - h[i]); h[i] += adj; r -= adj
                if r <= 1e-14: break
        elif r < 0:
            for i in range(N):
                adj = min(-r, h[i]); h[i] -= adj; r += adj
                if r >= -1e-14: break
        else:
            break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1 + 1e-10:
        return np.full(N, 0.5)
    if abs(h.sum() - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)

def exact_score(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    corr = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(corr)):
        return np.inf
    return float(np.max(corr) * 2.0 / N)

def lag_data(h, N):
    L = 2 * N - 1
    q = np.zeros(L)
    G = np.zeros((L, N))
    for s in range(-(N - 1), N):
        r = s + N - 1
        if s >= 0:
            ell = N - s
            a = h[:ell]; b = h[s:s+ell]
            q[r] = np.dot(a, 1.0 - b)
            G[r, :ell] += (1.0 - b); G[r, s:N] -= a
        else:
            d = -s; ell = N - d
            a = h[d:]; b = h[:ell]
            q[r] = np.dot(a, 1.0 - b)
            G[r, d:N] += (1.0 - b); G[r, :ell] -= a
    return q, G

def mm_obj(z, x, N):
    d = z[:N] - x
    return np.dot(d, d) + z[N]

def mm_obj_jac(z, x, N):
    g = np.zeros(N + 1)
    g[:N] = 2.0 * (z[:N] - x); g[N] = 1.0
    return g

def mm_eq(z, N, m):
    return np.array([np.sum(z[:N]) - m])

def mm_eq_jac(z, N, m):
    J = np.zeros((1, N + 1)); J[0, :N] = 1.0
    return J

def mm_ineq(z, N, x, q_x, G_x):
    return z[N] - q_x - G_x @ (z[:N] - x)

def mm_ineq_jac(z, N, x, q_x, G_x):
    J = np.zeros((2 * N - 1, N + 1))
    J[:, :N] = -G_x; J[:, N] = 1.0
    return J

def mm_step(x, N, m):
    q_x, G_x = lag_data(x, N)
    if not np.all(np.isfinite(q_x)) or not np.all(np.isfinite(G_x)):
        return x.copy()
    z0 = np.concatenate([x, [max(np.max(q_x), 0.0)]])
    bounds = [(0.0, 1.0)] * N + [(-4.0 * N, 6.0 * N)]
    cons = [
        {'type': 'eq', 'fun': mm_eq, 'jac': mm_eq_jac, 'args': (N, m)},
        {'type': 'ineq', 'fun': mm_ineq, 'jac': mm_ineq_jac, 'args': (N, x, q_x, G_x)},
    ]
    try:
        res = minimize(mm_obj, z0, jac=mm_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons, args=(x, N),
                       options={'maxiter': 120, 'ftol': 1e-10, 'disp': False})
    except Exception:
        return x.copy()
    if res.x is not None and len(res.x) == N + 1 and np.all(np.isfinite(res.x[:N])):
        return project_box_sum(res.x[:N], N, m)
    return x.copy()

def mm_local(x0, N, m, max_iter):
    x = x0.copy()
    best = x.copy()
    best_score = exact_score(x, N)
    cur_score = best_score
    alphas = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625]
    for it in range(max_iter):
        y_qp = mm_step(x, N, m)
        best_cs = cur_score
        best_cand = None
        for a in alphas:
            cand = (1.0 - a) * x + a * y_qp
            cand = project_box_sum(cand, N, m)
            sc = exact_score(cand, N)
            if sc < best_cs - 1e-14:
                best_cs = sc; best_cand = cand
        if best_cand is None or cur_score - best_cs <= 1e-12:
            break
        x = best_cand; cur_score = best_cs
        if cur_score < best_score:
            best_score = cur_score; best = x.copy()
        if np.max(np.abs(best_cand - x0)) < 1e-10 and it > 3:
            break
    return best, best_score
