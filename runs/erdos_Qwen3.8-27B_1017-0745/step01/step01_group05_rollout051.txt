            G[row, lo - k:hi - k] -= dx * h[lo:hi]

    n = len(z)

    idx = np.where(cond)[0]

        return np.full(n, 1.0 / n)

    if abs(s - 1.0) < 1e-14:
        return p
    pos = np.where(p > 0.0)[0]
    if len(pos) > 0:
        p[pos[0]] += 1.0 - s
        p = np.maximum(p, 0.0)
        p /= p.sum()
    return p
