def project_K(y, n, S):

        return np.full(n, S / n)
    lo, hi = y.min() - 1.0, y.max() + 1.0

            hi = mid

def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def lag_vec(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_grad(h, dx):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    j = np.arange(n)
    for k in range(2 * n - 1):
        ell = k - (n - 1)
        i1, v1 = j - ell, (j - ell >= 0) & (j - ell < n)
        i2, v2 = j + ell, (j + ell >= 0) & (j + ell < n)
        t1, t2 = np.zeros(n), np.zeros(n)
        t1[v1] = 1.0 - h[i1[v1]]
        t2[v2] = h[i2[v2]]
        G[k] = dx * (t1 - t2)
    return G

def transfer_parent(hp, n0, m):
