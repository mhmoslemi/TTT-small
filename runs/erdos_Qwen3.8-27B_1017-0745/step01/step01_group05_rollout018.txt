        if p.ndim == 1 and 2 <= len(p) <= 999 and np.all(np.isfinite(p)) and np.max(np.abs(p)) < 1e6:
            n = len(p)
            if check_feasible(p, n):
                fallback = p.copy()
            else:
                fallback = project_capped(p, n)
    else:
        nk = kwargs.get('n_points', None)
        if isinstance(nk, int) and not isinstance(nk, bool) and 2 <= nk <= 999:
            n = nk
        fallback = np.full(n, 0.5)
    if not check_feasible(fallback, n):
        fallback = np.full(n, 0.5)
    best_h = fallback.copy()
    best_s = exact_score(fallback, n)
    if n <= 64:
        B = make_helmert(n)
        d = n - 1
    else:
        d = 48
        B = make_basis_random(n, d, rng)
        if B is None:
            return fallback, best_s, n
    starts = []
    for s_raw in [fallback, 1.0 - fallback, fallback[::-1].copy(), 1.0 - fallback[::-1].copy()]:
        s = project_capped(s_raw, n)
        if not check_feasible(s, n):
            continue
        dup = any(np.max(np.abs(s - u)) < 1e-10 for u in starts)
        if not dup:
            starts.append(s)
            ss = exact_score(s, n)
            if ss < best_s - 1e-12:
                best_s = ss
                best_h = s.copy()
    sigmas = [0.06, 0.10, 0.04]
    gens = 36 if n <= 100 else (12 if n <= 400 else 6)
    for r in range(min(3, len(starts))):
        try:
            Br = B if n <= 64 else make_basis_random(n, d, rng)
            if Br is None:
                continue
            bh, bs = cma_restart(starts[r].copy(), Br, sigmas[r], gens, rng, n)
            if check_feasible(bh, n) and bs < best_s - 1e-12:
                best_s = bs
                best_h = bh.copy()
        except:
            pass
    if n <= 100:
        for attempt in range(3):
            z = rng.standard_normal(n)
            z -= z.mean()
            nz = math.sqrt(float(z @ z))
            if nz > 1e-10:
                z /= nz
                pert = project_capped(best_h + 0.08 * z, n)
                if check_feasible(pert, n) and np.max(np.abs(pert - best_h)) > 1e-6:
                    try:
                        Br = B if n <= 64 else make_basis_random(n, d, rng)
                        if Br is not None:
                            bh, bs = cma_restart(pert, Br, 0.04, gens, rng, n)
                            if check_feasible(bh, n) and bs < best_s - 1e-12:
                                best_s = bs
                                best_h = bh.copy()
                    except:
                        pass
                    break
    final_h = project_capped(best_h, n)
    if not check_feasible(final_h, n):
        final_h = fallback.copy()
    final_s = exact_score(final_h, n)
    fb_s = exact_score(fallback, n)
    if fb_s <= final_s + 1e-12:
        final_h = fallback.copy()
        final_s = fb_s
    return final_h.astype(np.float64), float(final_s), int(n)
