import numpy as np

def proj_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)):
        return None
    n = len(z)
    if S < 0 or S > n or np.max(np.abs(z)) > 1e12:
        return None
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    x = np.clip(z - lam, 0.0, 1.0)
    r = S - np.sum(x)
    for i in range(n):
        if r > 1e-15:
            a = min(r, 1.0 - x[i])
            x[i] += a
            r -= a
        elif r < -1e-15:
            a = min(-r, x[i])
            x[i] -= a
            r += a
        else:
            break
    if abs(np.sum(x) - S) > 1e-9 or not np.all(np.isfinite(x)):
        return None

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-9)

def exact_score(h, n):
    vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(vals)):
        return None
    return float(np.max(vals))

def make_line_coeffs(h, p, r, n):
    d = np.arange(-(n - 1), n, dtype=np.int64)
    b = np.zeros(2 * n - 1)
    a = np.zeros(2 * n - 1)

    idx = p + d; m = (idx >= 0) & (idx < n); b[m] -= h[idx[m]]

    a[n - 1 + (p - r)] = 1.0

def line_obj(vals_c, b, a, dx, t):
    q = vals_c + b * t + a * t * t

    U = min(1.0 - h[p], h[r])
    if U - L < 1e-12:
        return None

    # Coarse: 17 points
    ts = np.linspace(L, U, 17)
    ts = np.unique(np.concatenate([ts, [0.0]]))
    ts = ts[(ts >= L) & (ts <= U)]
    f0 = dx * float(np.max(q_vals))

    for t in ts:

        if f < best_f - 1e-14:

        hi_r = min(U, best_t + radius)

    P = min(n * (n - 1) // 2, max(24, 80000 // n))
    # Adjacent pairs
    adj = [(i, i + 1) for i in range(n - 1)]
    pairs = adj[:min(len(adj), P // 2)]
    if len(pairs) < P:
        remaining = P - len(pairs)
        used = set(pairs)
        attempts = 0
        while len(pairs) < P and attempts < 200:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                i, j = min(i, j), max(i, j)
                if (i, j) not in used:
                    pairs.append((i, j))
                    used.add((i, j))
            attempts += 1

def anneal_traj(h, n, dx, rng, epochs, nprop):

        T = max(5e-5, 3e-3 * (0.4 ** e))

            h_new[r] -= t
            if not is_feasible(h_new, n):
                h_new = proj_simplex(h_new, n / 2.0)

                    continue

                continue

                acc += 1

        if score < best_sc:
            best_h, best_sc = h.copy(), score

def is_feasible(h, n):

            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-9)

def exact_score(h, n):

    if not np.all(np.isfinite(vals)):
        return None
    return float(np.max(vals))

    b = np.zeros(2 * n - 1)
    a = np.zeros(2 * n - 1)
    idx = r + d; m = (idx >= 0) & (idx < n); b[m] += h[idx[m]]
    idx = p + d; m = (idx >= 0) & (idx < n); b[m] -= h[idx[m]]
    idx = p - d; m = (idx >= 0) & (idx < n); b[m] -= h[idx[m]]
    idx = r - d; m = (idx >= 0) & (idx < n); b[m] += h[idx[m]]
    a[n - 1] = -2.0
    a[n - 1 + (r - p)] = 1.0
    a[n - 1 + (p - r)] = 1.0

def line_eval(qc, b, a, t):
    q = qc + b * t + a * t * t
    return float(np.max(q))

        except (TypeError, ValueError):
            n = 80
        n = max(8, min(999, n))
    S = n / 2.0
    dx = 2.0 / n
    h_const = np.full(n, 0.5)
    best_h = h_const.copy()
    best_score = exact_score(h_const, n)
    if best_score is None:
        best_score = 0.5
    if parent is not None:
        ps = exact_score(parent, n)
        if ps is not None and ps < best_score:
            best_score, best_h = ps, parent.copy()
    starts = build_starts(n, S, parent, rng)
    for s in starts:
        sc = exact_score(s, n)
        if sc is not None and sc < best_score:
            best_score, best_h = sc, s.copy()
    nstarts = min(len(starts), 5 if n <= 128 else 3)
    epochs = 5 if n <= 128 else 3
    nprop = min(1000, max(80, 60000 // n))
    for s in starts[:nstarts]:
        th, ts_ = anneal_traj(s.copy(), n, dx, rng, epochs, nprop)
        if ts_ < best_score:
            best_score, best_h = ts_, th
    best_h, best_score = polish_sweep(best_h, best_score, n, dx, rng, 4)
    if not is_feasible(best_h, n):
        best_h = proj_simplex(best_h, S)
        if best_h is None:
            best_h = h_const.copy()
            best_score = 0.5
    fs = exact_score(best_h, n)
    if fs is not None:
        best_score = fs
    return best_h, float(best_score), n
