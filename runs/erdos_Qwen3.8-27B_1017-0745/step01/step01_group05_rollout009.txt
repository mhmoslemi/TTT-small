def project_simplex(z):
    z = np.asarray(z, dtype=np.float64)
    m = len(z)
    if m == 0:
        return z
    if not np.all(np.isfinite(z)):
        return np.full(m, 1.0 / m)
    s = np.sort(z)[::-1]
    cs = np.cumsum(s)
    rho = np.arange(1, m + 1)
    mask = (s - (cs - 1.0) / rho) > 0
    if not mask.any():
        return np.full(m, 1.0 / m)
    r = int(mask.sum())
    theta = (cs[r - 1] - 1.0) / r
    out = np.maximum(z - theta, 0.0)
    diff = 1.0 - out.sum()
    if abs(diff) < 1e-12:
        return out
    idx = np.argmax(out)
    out[idx] += diff
    return out

import numpy as np
import math

def project_capped(y, n):
    S = n / 2.0
    y = np.asarray(y, dtype=np.float64)
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return np.full(n, 0.5), False
    lo = float(y.min()) - 1.0
    hi = float(y.max())
    lam = np.clip(y.mean() - 0.5, lo, hi)
    c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(60):
        r = c.sum() - S
        if abs(r) <= 1e-13:
            break
        if r > 0:
            lo = lam
        else:
            hi = lam
        a = int(np.sum((y > lam) & (y < lam + 1.0)))
        if a > 0:
            lam_new = lam + r / a
        else:
            lam_new = 0.5 * (lo + hi)
        if not np.isfinite(lam_new) or lam_new <= lo or lam_new >= hi:
            lam_new = 0.5 * (lo + hi)
        lam = lam_new
        c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(3):
        diff = S - c.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = c < 1.0 - 1e-12
        else:
            mask = c > 1e-12
        cnt = int(mask.sum())
        if cnt == 0:
            break
        c[mask] += diff / cnt
        np.clip(c, 0.0, 1.0, out=c)
    if abs(c.sum() - S) > 1e-10:
        idx = int(np.argmax(1.0 - c)) if S > c.sum() else int(np.argmax(c))
        c[idx] += S - c.sum()
        np.clip(c, 0.0, 1.0, out=c)
    ok = bool(np.all(c >= 0) and np.all(c <= 1) and abs(c.sum() - S) <= 1e-10)
    return c, ok

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(h.sum() - n / 2.0) <= 1e-10)

def lag_scores(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def exact_score(h, n):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(lag_scores(h, n)))

def build_G(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        lo = max(0, k)
        hi = min(n, n + k)
        row = k + n - 1
        if hi > lo:
            G[row, lo:hi] += dx * (1.0 - h[lo - k:hi - k])
            G[row, lo - k:hi - k] -= dx * h[lo:hi]
    return G

def project_simplex(z):
    z = np.asarray(z, dtype=np.float64)
    m = len(z)
    if m == 0:
        return z.copy()
    if not np.all(np.isfinite(z)):
        return np.full(m, 1.0 / m)
    s = np.sort(z)[::-1]
    cs = np.cumsum(s)
    rho = np.arange(1, m + 1)
    mask = (s - (cs - 1.0) / rho) > 0
    if not mask.any():
        return np.full(m, 1.0 / m)
    r = int(mask.sum())
    theta = (cs[r - 1] - 1.0) / r
    out = np.maximum(z - theta, 0.0)
    diff = 1.0 - out.sum()
    if abs(diff) > 1e-14:
        out[int(np.argmax(out))] += diff
    return out

def dual_eval(h, q, G, p_vec, L, n):
    gmix = G.T @ p_vec
    y, _ = project_capped(h - gmix / L, n)
    d = y - h
    u = q + G @ d
    D_val = float(p_vec @ u + 0.5 * L * (d @ d))
    return d, u, D_val

def solve_majorant(h, q, G, n, max_inner):
    L = 4.0 / n
    GtG = float(np.sum(G * G))
    Ldual = max(1e-12, 1.05 * GtG / L)
    m = 2 * n - 1
    qmax = float(q.max())
    tol_act = max(1e-12, 1e-8 * max(1.0, abs(qmax)))
    active = np.where(q >= qmax - tol_act)[0]
    if len(active) == 0:
        p0 = np.zeros(m)
        p0[int(np.argmax(q))] = 1.0
    else:
        p0 = np.zeros(m)
        p0[active] = 1.0 / len(active)
    d_best, u_best, D_best = dual_eval(h, q, G, p0, L, n)
    x = p0.copy()
    y_pt = x.copy()
    t_ig = 1.0
    D_prev = D_best
    small_step = 0
    for it in range(max_inner):
        gy = G.T @ y_pt
        y_trial, _ = project_capped(h - gy / L, n)
        d_y = y_trial - h
        u_y = q + G @ d_y
        x_trial = project_simplex(y_pt + u_y / Ldual)
        d_x, u_x, D_x = dual_eval(h, q, G, x_trial, L, n)
        if D_x < D_prev - 1e-14:
            x2 = project_simplex(x + u_y / Ldual)
            d2, u2, D2 = dual_eval(h, q, G, x2, L, n)
            if D2 >= D_prev - 1e-14:
                x = x2
                y_pt = x.copy()
                t_ig = 1.0
                D_prev = D2
                if D2 > D_best:
                    D_best = D2
                    d_best = d2
                    u_best = u2
            else:
                break
        else:
            D_prev = D_x
            if D_x > D_best:
                D_best = D_x
                d_best = d_x
                u_best = u_x
            t_new = (1.0 + math.sqrt(1.0 + 4.0 * t_ig * t_ig)) / 2.0
            y_pt = x_trial + ((t_ig - 1.0) / t_new) * (x_trial - x)
            x = x_trial
            t_ig = t_new
        if np.max(np.abs(u_y)) < 1e-10:
            small_step += 1
        else:
            small_step = 0
        gap = float(u_best.max() - x @ u_best)
        if gap <= max(1e-10, 1e-8 * max(1.0, abs(float(u_best.max())))):
            break
        if small_step >= 4:
            break
    return d_best

def optimize_start(h0, n, max_outer, max_inner):
    h, _ = project_capped(h0, n)
    s = exact_score(h, n)
    best_h = h.copy()
    best_s = s
    L = 4.0 / n
    stall = 0
    for outer in range(max_outer):
        q = lag_scores(h, n)
        G = build_G(h, n)
        if not np.all(np.isfinite(G)):
            break
        d = solve_majorant(h, q, G, n, max_inner)
        if not np.all(np.isfinite(d)) or np.max(np.abs(d)) <= 1e-10:
            break
        v = G @ d
        norm2 = float(d @ d)
        qmax = float(q.max())
        accepted = False
        for j in range(20):
            alpha = 1.0 / (2 ** j)
            M_val = float(np.max(q + alpha * v)) + 0.5 * L * alpha * alpha * norm2
            if M_val >= qmax - 1e-14:
                continue
            h_trial = h + alpha * d
            h_trial, ok = project_capped(h_trial, n)
            if not ok:
                continue
            s_trial = exact_score(h_trial, n)
            if s_trial < s - 1e-12:
                h = h_trial
                s = s_trial
                if s < best_s - 1e-12:
                    best_s = s
                    best_h = h.copy()
                accepted = True
                break
        if not accepted:
            break
        if best_s - s < 1e-11:
            stall += 1
        else:
            stall = 0
        if stall >= 3:
            break
    return best_h, best_s

def run(seed=42, **kwargs):
