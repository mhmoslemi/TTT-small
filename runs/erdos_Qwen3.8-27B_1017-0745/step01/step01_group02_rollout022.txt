        return float('inf')
    c = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
    return c if np.isfinite(c) else float('inf')

def lag_vg(h, s, n):
    dx = 2.0 / n
    lo = max(0, -s)
    hi = min(n, n - s)
    q = 0.0

        q += h[i] * (1.0 - h[j])
        g[i] += (1.0 - h[j])
        g[j] -= h[i]

        vals[idx] = v
        grads[idx] = g

    d = h - x
    dx = 2.0 / n
    best = -float('inf')

        if ell > best:
            best = ell
            best_s = s
    return best, best_s
