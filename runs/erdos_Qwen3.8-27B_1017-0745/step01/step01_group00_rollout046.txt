def golden_min(fn, a, b, iters=52):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = fn(c)
    fd = fn(d)
    for _ in range(iters):
        if fc < fd:
            b, d, fd = d, c, fc
            c = b - gr * (b - a)
            fc = fn(c)
        else:
            a, c, fc = c, d, fd
            d = a + gr * (b - a)
            fd = fn(d)
    # Return best of endpoints and interior points
    best_t, best_v = a, fn(a)
    for t in (b, c, d, 0.5*(a+b)):
        v = fn(t)
        if v < best_v:
            best_v, best_t = v, t
    return best_t, best_v

def bisect_root(fn, a, b, iters=52):
    fa, fb = fn(a), fn(b)
    if fa * fb > 0:
        return None
    for _ in range(iters):
        m = 0.5 * (a + b)
        fm = fn(m)
        if fa * fm <= 0:
            b, fb = m, fm
        else:
            a, fa = m, fm
    return 0.5 * (a + b)

        return None
    D = lag_values(h, n)
    # Precompute A and B for each lag
    nlag = 2 * n - 1
    A = np.zeros(nlag)
    B = np.zeros(nlag)
    for k in range(nlag):

        if d == 0:

        elif abs(d) == ri:

        B[k] = lag_grad_at(h, d, i, n, dx) - lag_grad_at(h, d, j, n, dx)

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n))

def project_capped(z, n):
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for i in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0 and h[i] < 1.0:
            a = min(r, 1.0 - h[i]); h[i] += a; r -= a
        elif r < 0 and h[i] > 0.0:
            a = min(-r, h[i]); h[i] -= a; r += a
    if abs(r) > 1e-10 * max(1, n):
        return None
    return h

        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def lag_grad_at(h, d, k, n, dx):

    m = n - abs(d)
    g = 0.0
    if k >= p0 and k - p0 < m:
        g += dx * (1.0 - h[q0 + k - p0])
    if k >= q0 and k - q0 < m:
        g -= dx * h[p0 + k - q0]
    return g

        if fa * fm <= 0: b, fb = m, fm
        else: a, fa = m, fm

            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n))

def project_capped(z, n):

    s = n / 2.0

        else:

        g += dx * (1.0 - h[q0 + k - p0])
    if k >= q0 and k - q0 < m:

def local_descent(start, n, rng, n_moves):
    h = project_capped(start, n)
    if h is None:
        return None, float('inf')
    score = exact_score(h, n)
    use_all = n <= 80
    for _ in range(n_moves):
        if use_all:
            ii, jj = np.triu_indices(n, 1)
            pairs = list(zip(ii.tolist(), jj.tolist()))
        else:
            npairs = min(3000, 4 * n)
            ii = rng.integers(0, n, size=npairs)
            jj = rng.integers(0, n, size=npairs)
            pairs = [(a, b) for a, b in zip(ii.tolist(), jj.tolist()) if a < b]
        best_s, best_t, best_pair = score, None, None
        for (a, b) in pairs:
            t = pair_line_min(h, a, b, n)
            if t is None or abs(t) < 1e-14:
                continue
            trial = h.copy()
            trial[a] += t
            trial[b] -= t
            trial = project_capped(trial, n)
            if trial is None or not is_feasible(trial, n):
                continue
            s = exact_score(trial, n)
            if s < best_s:
                best_s, best_t, best_pair = s, t, (a, b)
        if best_pair is None or best_s >= score - 1e-12:
            break
        h = h.copy()
        h[best_pair[0]] += best_t
        h[best_pair[1]] -= best_t
        h = project_capped(h, n)
        if h is None:
            break
        score = exact_score(h, n)
        if score > best_s + 1e-10:
            score = best_s
    return h, score
