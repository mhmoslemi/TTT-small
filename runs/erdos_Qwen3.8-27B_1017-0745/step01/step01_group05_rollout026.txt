def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(h.sum() - n / 2.0) <= 1e-10)

def project_capped(y, n):
    S = n / 2.0
    if y.shape != (n,) or not np.all(np.isfinite(y)) or np.max(np.abs(y)) > 1e12:

        if np.clip(y - mid, 0, 1).sum() > S:
            lo = mid
        else:
            hi = mid
    c = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(5):
        r = S - c.sum()
        if abs(r) < 1e-14:
            break
        if r > 0:
            cap = 1.0 - c
            if cap.sum() < 1e-14:
                break
            c += r * cap / cap.sum()
        else:
            if c.sum() < 1e-14:
                break
            c += r * c / c.sum()
    np.clip(c, 0.0, 1.0, out=c)
    if check_feasible(c, n):
        return c, True
    return np.full(n, 0.5), False

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    if corr.shape != (2 * n - 1,) or not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr * (2.0 / n)))

def make_helmert(n):
    d = n - 1
    B = np.zeros((n, d))
    for j in range(d):
        sc = 1.0 / math.sqrt((j + 1) * (j + 2))
        B[:j + 1, j] = sc
        B[j + 1, j] = -(j + 1) * sc
    return B

def decode_point(anchor, B, rho, x, n):
    raw = anchor + rho * (B @ x)
    if not np.all(np.isfinite(raw)):
        return None, float('inf')
    h, ok = project_capped(raw, n)
    if not ok:
        return None, float('inf')
    return h, exact_score(h, n)
