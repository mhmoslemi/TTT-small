def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    return abs(np.sum(h) - n / 2.0) <= 5e-12 * max(1, n)

def project_capped(v, n):
    if v.shape != (n,) or not np.all(np.isfinite(v)):
        return None
    v = np.clip(v, -1e12, 1e12)
    s = n / 2.0
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if mid <= lo or mid >= hi:
            break
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(2):
        if abs(r) < 1e-14:
            break
        if r > 0:
            for i in range(n):
                if r <= 0:
                    break
                room = 1.0 - h[i]
                add = min(r, room)
                h[i] += add
                r -= add
        else:
            for i in range(n):
                if r >= 0:
                    break
                rem = h[i]
                sub = min(-r, rem)
                h[i] -= sub
                r += sub
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or np.any(h < 0) or np.any(h > 1.0):
        return None
    if abs(np.sum(h) - s) > 5e-12 * max(1, n):

    return h

def exact_score(h, n):

def lag_values(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    g = np.zeros(L)
    for d in range(-(n - 1), n):
        row = d + n - 1
        if d >= 0:
            m = n - d
            g[row] = dx * np.dot(h[:m], 1.0 - h[d:d + m])
        else:
            da = -d
            m = n - da
            g[row] = dx * np.dot(h[da:], 1.0 - h[:m])
    return g

def smooth_value(g, tau):
    gm = np.max(g)
    ex = np.exp((g - gm) / tau)
    return tau * (np.log(np.sum(ex)) + gm / tau)

            grad[:m] += -dx * p * h[da:]
    if not np.all(np.isfinite(grad)):
        return None, None
    return smooth_value(g, tau), grad
