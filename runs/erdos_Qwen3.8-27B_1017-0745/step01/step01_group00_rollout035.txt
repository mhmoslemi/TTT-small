    except Exception:
        pass

    try:
        hc2, sc2 = coord_descent(gb, n, 4, 0.05)
        if sc2 < gs - 1e-12:
            gs, gb = sc2, hc2.copy()
    except Exception:
        pass

    try:
        hp, sp = pair_refine(gb, n, 2)
        if sp < gs - 1e-12:
            gs, gb = sp, hp.copy()
    except Exception:
        pass

    try:
        hc3, sc3 = coord_descent(gb, n, 3, 0.03)
        if sc3 < gs - 1e-12:
            gs, gb = sc3, hc3.copy()
    except Exception:
        pass

    final = project_capped(gb, n)
    if final is None or exact_score(final, n) > fb_score:
        final = fallback
    sf = exact_score(final, n)
    return final, float(sf), n
