def project_box_sum(z, N, m):
    if z.shape != (N,) or not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    for _ in range(2):
        if abs(r) < 1e-14:
            break
        for i in range(N):
            if abs(r) < 1e-14:
                break
            if r > 0:
                adj = min(r, 1.0 - h[i])
            else:
                adj = max(r, -h[i])
            h[i] += adj
            r = m - h.sum()
