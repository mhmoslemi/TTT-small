def project_capped(y, n):
    if y.shape != (n,) or not np.all(np.isfinite(y)) or np.max(np.abs(y)) > 1e12:
        return np.full(n, 0.5), False
    S = n / 2.0
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        c = np.clip(y - mid, 0.0, 1.0)
        if c.sum() > S:
            lo = mid
        else:
            hi = mid
    c = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(5):
        r = S - c.sum()
        if abs(r) < 1e-14:
            break
        if r > 0:
            slack = 1.0 - c
            tot = slack.sum()
            if tot < 1e-15:
                break
            c += r * slack / tot
        else:
            tot = c.sum()
            if tot < 1e-15:
                break
            c += r * c / tot
        np.clip(c, 0.0, 1.0, out=c)
    if abs(c.sum() - S) > 1e-10 or c.min() < -1e-12 or c.max() > 1.0 + 1e-12:

    return c, True

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):

    if corr.shape != (2 * n - 1,) or not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr * dx))

    mueff = 1.0 / float(np.sum(w * w))
    cc = (4.0 + mueff / d) / (d + 4.0 + 2.0 * mueff / d)
    csig = (mueff + 2.0) / (d + mueff + 5.0)
    c1 = 2.0 / ((d + 1.3) ** 2 + mueff)
    cmu = min(1.0 - c1, 2.0 * (mueff - 2.0 + 1.0 / mueff) / ((d + 2.0) ** 2 + mueff))
    dsig = 1.0 + 2.0 * max(0.0, math.sqrt((mueff - 1.0) / (d + 1.0)) - 1.0) + csig
    chid = math.sqrt(d) * (1.0 - 1.0 / (4.0 * d) + 1.0 / (21.0 * d * d))
    nreset = 0
    stall = 0
    for gen in range(ngen):
        C = 0.5 * (C + C.T)
        try:
            evals, evecs = np.linalg.eigh(C)
            if not np.all(np.isfinite(evals)):
                raise np.linalg.LinAlgError("nonfinite")
            evals = np.clip(evals, 1e-8, 1e3)
            C = evecs @ np.diag(evals) @ evecs.T
            A = evecs @ np.diag(np.sqrt(evals))
            Cinvsh = evecs @ np.diag(1.0 / np.sqrt(evals))
        except Exception:
            if nreset >= 1:
                break
            nreset += 1
            C = np.eye(d)
            pc, ps = np.zeros(d), np.zeros(d)
            sigma *= 0.5
            continue
        xs = np.zeros((lam, d))
        scores = np.full(lam, float('inf'))
        for j in range(lam):
            z = np.zeros(d) if (gen == 0 and j == 0) else rng.standard_normal(d)
            xj = m + sigma * (A @ z)
            xj = np.clip(xj, -30.0, 30.0)
            raw = anchor + rho * (B @ xj)
            if not np.all(np.isfinite(raw)):
                continue
            hj, ok = project_capped(raw, n)
            if not ok:
                continue
            sj = exact_score(hj, n)
            xs[j] = xj
            scores[j] = sj
            if np.isfinite(sj) and sj < sbest:
                sbest = sj
                nbest = hj.copy()
        fin_idx = np.where(np.isfinite(scores))[0]
        if len(fin_idx) < mu:
            break
        fin_idx = fin_idx[np.argsort(scores[fin_idx])]
        sel = fin_idx[:mu]
        xs_sel = xs[sel]
        m_new = np.sum(w[:, None] * xs_sel, axis=0)
        raw_m = anchor + rho * (B @ m_new)
        if np.all(np.isfinite(raw_m)):
            hm, okm = project_capped(raw_m, n)
            if okm:
                sm = exact_score(hm, n)
                if np.isfinite(sm) and sm < sbest:
                    sbest = sm
                    nbest = hm.copy()
        if sigma > 1e-15:
            yw = (m_new - m) / sigma
            ys = (xs_sel - m) / sigma
        else:
            yw, ys = np.zeros(d), np.zeros_like(xs_sel)
        ps = (1.0 - csig) * ps + math.sqrt(max(1e-30, csig * (2.0 - csig) * mueff)) * (Cinvsh @ yw)
        d1 = math.sqrt(max(1e-30, 1.0 - (1.0 - csig) ** (2 * (gen + 1))))
        hs_val = 1.0 if np.linalg.norm(ps) / d1 < (1.4 + 2.0 / (d + 1.0)) * chid else 0.0
        pc = (1.0 - cc) * pc + hs_val * math.sqrt(max(1e-30, cc * (2.0 - cc) * mueff)) * yw
        rm = np.zeros((d, d))
        for i in range(mu):
            rm += w[i] * np.outer(ys[i], ys[i])
        Cmat = ((1.0 - c1 - cmu + c1 * (1.0 - hs_val) * cc * (2.0 - cc)) * C
                + c1 * np.outer(pc, pc) + cmu * rm)
        C = 0.5 * (Cmat + Cmat.T)
        ea = (csig / dsig) * (np.linalg.norm(ps) / chid - 1.0)
        ea = max(-2.0, min(2.0, ea))
        sigma = float(np.clip(sigma * math.exp(ea), 1e-4, 0.5))
        m = np.clip(m_new, -30.0, 30.0)
        if rho * sigma * math.sqrt(max(np.max(evals), 1e-30)) < 1e-5:
            stall += 1
        else:
            stall = 0
        if stall >= 5:
            break
    return nbest, sbest

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = globals().get('initial_h_values', None)
    n = 45
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.ndim == 1 and 2 <= len(p) <= 999 and np.all(np.isfinite(p)) and np.max(np.abs(p)) < 1e6:
            n = len(p)
            fb, ok = project_capped(p.copy(), n)
            if not ok:
                fb = np.full(n, 0.5)
        else:
            fb = np.full(n, 0.5)
    else:
        nkw = kwargs.get('n_points', None)
        if isinstance(nkw, int) and not isinstance(nkw, bool) and 2 <= nkw <= 999:
            n = nkw
        fb = np.full(n, 0.5)
    fb, ok = project_capped(fb, n)
    if not ok:
        fb = np.full(n, 0.5)
    best_h = fb.copy()
    best_s = exact_score(fb, n)
    if n <= 64:
        B = make_helmert(n)
    else:
        dim = min(48, n - 1) if n <= 400 else 32
        Btmp = rng.standard_normal((n, dim))
        Btmp -= Btmp.mean(axis=0, keepdims=True)
        try:
            _, _, Vt = np.linalg.svd(Btmp, full_matrices=False)
            B = Vt.T[:, :dim]
        except Exception:
            B = make_helmert(min(n, 64))
    d = B.shape[1]
    rho = math.sqrt(n / d)
    if n <= 100:
        lam = max(10, min(16, 4 + int(3 * math.log(max(d, 2)))))
        ngen = 36
    elif n <= 400:
        lam, ngen = 12, 12
    else:
        lam, ngen = 10, 6
    anchors = [fb.copy()]
    for c in [1.0 - fb, fb[::-1].copy(), 1.0 - fb[::-1].copy()]:
        t, ok = project_capped(c, n)
        if ok and all(np.max(np.abs(t - a)) >= 1e-10 for a in anchors):
            anchors.append(t)
    try:
        bh, bs = cma_restart(anchors[0], B, rho, 0.06, ngen, lam, rng, n)
        if bs < best_s:
            best_s, best_h = bs, bh
    except Exception:
        pass
    a1 = anchors[1] if len(anchors) > 1 else None
    if a1 is None:
        z = rng.standard_normal(n)
        z -= z.mean()
        zn = np.sqrt(np.mean(z * z))
        if zn > 1e-10:
            z /= zn
            a1, ok = project_capped(fb + 0.1 * z, n)
            if not ok:
                a1 = anchors[0]
        else:
            a1 = anchors[0]
    try:
        bh, bs = cma_restart(a1, B, rho, 0.10, ngen, lam, rng, n)
        if bs < best_s:
            best_s, best_h = bs, bh
    except Exception:
        pass
    if n <= 100:
        a2 = None
        for _ in range(4):
            z = rng.standard_normal(n)
            z -= z.mean()
            zn = np.sqrt(np.mean(z * z))
            if zn > 1e-10:
                z /= zn
                p3, ok = project_capped(best_h + 0.08 * z, n)
                if ok and all(np.max(np.abs(p3 - a)) >= 1e-10 for a in anchors):
                    a2 = p3
                    break
        if a2 is not None:
            try:
                bh, bs = cma_restart(a2, B, rho, 0.04, ngen, lam, rng, n)
                if bs < best_s:
                    best_s, best_h = bs, bh
            except Exception:
                pass
    try:
        bh, bs = cma_restart(best_h.copy(), B, rho, 0.015, min(25, ngen), lam, rng, n)
        if bs < best_s:
            best_s, best_h = bs, bh
    except Exception:
        pass
    fh, fok = project_capped(best_h.copy(), n)
    if not fok:
        fh = fb.copy()
    fs = exact_score(fh, n)
    fsb = exact_score(fb, n)
    if not np.isfinite(fs) or fs > fsb:
        fh, fs = fb.copy(), fsb
    return fh.astype(np.float64), float(fs), int(n)
