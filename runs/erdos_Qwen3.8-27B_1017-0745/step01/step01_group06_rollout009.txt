import numpy as np
from math import sqrt

def project_K(y, n, S):
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return np.full(n, S / n)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    e = S - h.sum()
    if abs(e) > 1e-12:
        if e > 0:
            for j in range(n):
                d = min(e, 1.0 - h[j])
                h[j] += d
                e -= d
                if e <= 0:
                    break
        else:
            e = -e
            for j in range(n):
                d = min(e, h[j])
                h[j] -= d
                e -= d
                if e <= 0:
                    break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1.0 + 1e-10 or abs(h.sum() - S) > 1e-8:
        return np.full(n, S / n)
    return h

def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def pair_coeffs(h, p, r, n, dx):
    nl = 2 * n - 1
    b = np.zeros(nl)
    a = np.zeros(nl)
    for ell in range(-(n-1), n):
        be = 0.0
        if 0 <= p - ell < n:
            be += (1.0 - h[p - ell])
        if 0 <= r - ell < n:
            be -= (1.0 - h[r - ell])
        if 0 <= p + ell < n:
            be -= h[p + ell]
        if 0 <= r + ell < n:
            be += h[r + ell]
        b[ell + n - 1] = dx * be
    d = abs(p - r)
    a[n - 1] = -2.0 * dx
    if d > 0:
        a[n - 1 + d] = dx
        a[n - 1 - d] = dx
    return b, a

def gen_coeffs(h, z, n, dx):
    nl = 2 * n - 1
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    b = dx * (np.correlate(z, 1.0 - h, mode='full') - np.correlate(h, z, mode='full'))
    a = -dx * np.correlate(z, z, mode='full')
    if b.shape != (nl,) or a.shape != (nl,):
        return None, None, None
    if not (np.all(np.isfinite(b)) and np.all(np.isfinite(a))):
        return None, None, None
    return q, b, a

def feasible_interval(h, z, n):
    lo, hi = 0.0, 0.0
    first = True
    for j in range(n):
        if abs(z[j]) < 1e-16:
            continue
        if z[j] > 0:
            v1, v2 = -h[j] / z[j], (1.0 - h[j]) / z[j]
        else:
            v1, v2 = (1.0 - h[j]) / z[j], -h[j] / z[j]
        if first:
            lo, hi = v1, v2
            first = False
        else:
            lo = max(lo, v1)
            hi = min(hi, v2)
    if first:
        return None
    if hi - lo < 1e-14:
        return None
    return lo, hi

def line_search(h, q, z, b, a, lo, hi, n, dx, n_coarse, n_branches, n_ref, pts_ref):
    nl = 2 * n - 1
    alphas = np.linspace(lo, hi, n_coarse)
    # Compute F for each alpha
    scores = np.zeros(n_coarse)
    for k in range(n_coarse):
        al = alphas[k]
        vals = q + al * b + al * al * a
        scores[k] = np.max(vals)
    best_coarse = np.argmin(scores)
    # Find local minima
    mins = []
    for k in range(n_coarse):
        if k == 0:
            if scores[0] <= scores[1]:
                mins.append(0)
        elif k == n_coarse - 1:
            if scores[k] <= scores[k-1]:
                mins.append(k)
        else:
            if scores[k] <= scores[k-1] and scores[k] <= scores[k+1]:
                mins.append(k)
    if len(mins) == 0:
        mins = [best_coarse]
    # Sort by score, separate by at least 2
    mins_sorted = sorted(mins, key=lambda k: (scores[k], abs(alphas[k])))
    sel = []
    for k in mins_sorted:
        ok = True
        for s in sel:
            if abs(k - s) < 2:
                ok = False
                break
        if ok:
            sel.append(k)
        if len(sel) >= n_branches:
            break
    if len(sel) == 0:
        sel = [best_coarse]
    
    best_alpha = 0.0
    best_score = exact_score(h, dx)
    
    for bk in sel:
        # Bracket around bk
        b_lo = max(bk - 1, 0)
        b_hi = min(bk + 1, n_coarse - 1)
        a_lo, a_hi = alphas[b_lo], alphas[b_hi]
        cur_best_a = alphas[bk]
        cur_best_s = scores[bk]
        for _ in range(n_ref):
            pts = np.linspace(a_lo, a_hi, pts_ref)
            sv = np.zeros(pts_ref)
            for k in range(pts_ref):
                al = pts[k]
                vals = q + al * b + al * al * a
                sv[k] = np.max(vals)
            bk2 = np.argmin(sv)
            if sv[bk2] < cur_best_s - 1e-15:
                cur_best_s = sv[bk2]
                cur_best_a = pts[bk2]
            # Shrink bracket
            if bk2 == 0:
                a_hi = pts[1] if pts_ref > 1 else a_hi
            elif bk2 == pts_ref - 1:
                a_lo = pts[pts_ref - 2] if pts_ref > 1 else a_lo
            else:
                a_lo = pts[bk2 - 1]
                a_hi = pts[bk2 + 1]
        # Evaluate this branch's best
        if cur_best_s < best_score - 1e-13:
            best_score = cur_best_s
            best_alpha = cur_best_a
    
    if best_alpha == 0.0:
        return h, q, exact_score(h, dx)
    
    trial = h + best_alpha * z
    trial = project_K(trial, n, n / 2.0)
    if not feasible(trial, n, n / 2.0):
        return h, q, exact_score(h, dx)
    s = exact_score(trial, dx)
    if s < exact_score(h, dx) - 1e-12:
        return trial, np.correlate(trial, 1.0 - trial, mode='full') * dx, s
    return h, q, exact_score(h, dx)

def local_descent(h, n, dx, rng, max_sweeps, n_pairs, n_sparse, nc, nb, nr, pr):
    S = n / 2.0
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    sc = exact_score(h, dx)
    changed_sweeps = 0
    for sw in range(max_sweeps):
        any_change = False
        # Pair directions
        if n_pairs >= n * (n - 1) // 2:
            pairs = [(p, r) for p in range(n) for r in range(p + 1, n)]
        else:
            pairs = set()
            att = 0
            while len(pairs) < n_pairs and att < 10000:
                p, r = rng.integers(n), rng.integers(n)
                if p != r:
                    if p > r:
                        p, r = r, p
                    pairs.add((p, r))
                att += 1
            # Fill deterministically if needed
            if len(pairs) < n_pairs:
                for p in range(n):
                    for r in range(p + 1, n):
                        if (p, r) not in pairs:
                            pairs.add((p, r))
                            if len(pairs) >= n_pairs:
                                break
                    if len(pairs) >= n_pairs:
                        break
            pairs = list(pairs)
        rng.shuffle(pairs)
        for (p, r) in pairs:
            b, a = pair_coeffs(h, p, r, n, dx)
            lo = max(-h[p], h[r] - 1.0)
            hi = min(1.0 - h[p], h[r])
            if hi - lo < 1e-14:
                continue
            z = np.zeros(n)
            z[p] = 1.0
            z[r] = -1.0
            h2, q2, s2 = line_search(h, q, z, b, a, lo, hi, n, dx, nc, nb, nr, pr)
            if s2 < sc - 1e-12:
                h, q, sc = h2, q2, s2
                any_change = True
        # Sparse directions
        for _ in range(n_sparse):
            sz = max(2, min(12, n))
            idx = rng.choice(n, sz, replace=False)
            vals = rng.standard_normal(sz)
            vals -= vals.mean()
            if np.max(np.abs(vals)) < 1e-14:

            z = np.zeros(n)
            z[idx] = vals
            z[idx[-1]] = -z[idx[:-1]].sum()

            if q3 is None:

            if iv is None:

            h2, q2, s2 = line_search(h, q, z, b3, a3, iv[0], iv[1], n, dx, nc, nb, nr, pr)
            if s2 < sc - 1e-12:
                h, q, sc = h2, q2, s2
                any_change = True
        if any_change:
            changed_sweeps = 0
        else:
            changed_sweeps += 1
            if changed_sweeps >= 2:
                break
    h = project_K(h, n, S)
    if not feasible(h, n, S):
        return None
    return h

    if not np.all(np.isfinite(h)): return float('inf')
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def fchk(h, n, S):
    return h.shape == (n,) and np.all(np.isfinite(h)) and h.min() >= -1e-10 and h.max() <= 1 + 1e-10 and abs(h.sum() - S) <= 1e-8

def pc(h, p, r, n, dx):
    nl = 2 * n - 1
    b = np.zeros(nl); a = np.zeros(nl)
    for ell in range(-(n - 1), n):
        be = 0.0
        if 0 <= p - ell < n: be += 1 - h[p - ell]
        if 0 <= r - ell < n: be -= 1 - h[r - ell]
        if 0 <= p + ell < n: be -= h[p + ell]
        if 0 <= r + ell < n: be += h[r + ell]
        b[ell + n - 1] = dx * be
    d = abs(p - r)
    a[n - 1] = -2.0 * dx
    if d > 0:
        a[n - 1 + d] = dx; a[n - 1 - d] = dx
    return b, a

def fiv(h, z, n):
    lo = hi = 0.0; first = True
    for j in range(n):
        if abs(z[j]) < 1e-16: continue
        if z[j] > 0: v1, v2 = -h[j] / z[j], (1 - h[j]) / z[j]
        else: v1, v2 = (1 - h[j]) / z[j], -h[j] / z[j]
        if first: lo, hi = v1, v2; first = False
        else: lo, hi = max(lo, v1), min(hi, v2)
    if first or hi - lo < 1e-14: return None
    return lo, hi

def lsearch(h, q, z, b, a, lo, hi, n, dx, nc, nb, nr, pr):
    S = n / 2.0
    alphas = np.linspace(lo, hi, nc)
    M = q[None, :] + alphas[:, None] * b[None, :] + (alphas[:, None] ** 2) * a[None, :]
    scg = np.max(M, axis=1)
    best_k = int(np.argmin(scg))
    mins = []
    for k in range(nc):
        if k == 0:
            if scg[0] <= scg[1]: mins.append(0)
        elif k == nc - 1:
            if scg[k] <= scg[k - 1]: mins.append(k)
        elif scg[k] <= scg[k - 1] and scg[k] <= scg[k + 1]: mins.append(k)
    if not mins: mins = [best_k]
    mins_s = sorted(mins, key=lambda k: (scg[k], abs(alphas[k])))
    sel = []
    for k in mins_s:
        if all(abs(k - s) >= 2 for s in sel): sel.append(k)
        if len(sel) >= nb: break
    if not sel: sel = [best_k]
    ba, bs = 0.0, float('inf')
    for bk in sel:
        a_lo = alphas[max(bk - 1, 0)]; a_hi = alphas[min(bk + 1, nc - 1)]
        ca, cs = alphas[bk], scg[bk]
        for _ in range(nr):
            pts = np.linspace(a_lo, a_hi, pr)
            M2 = q[None, :] + pts[:, None] * b[None, :] + (pts[:, None] ** 2) * a[None, :]
            sv = np.max(M2, axis=1)
            bk2 = int(np.argmin(sv))
            if sv[bk2] < cs: cs, ca = sv[bk2], pts[bk2]
            if bk2 == 0: a_hi = pts[min(1, pr - 1)]
            elif bk2 == pr - 1: a_lo = pts[max(pr - 2, 0)]
            else: a_lo, a_hi = pts[bk2 - 1], pts[bk2 + 1]
        if cs < bs: bs, ba = cs, ca
    if ba == 0.0: return None
    tr = h + ba * z
    tr = project_K(tr, n, S)
    if not fchk(tr, n, S): return None
    s = esc(tr, dx)
    if s < esc(h, dx) - 1e-12: return tr
    return None

def ldesc(h, n, dx, rng, sw, nsparse, nc, nb, nr, pr):
    S = n / 2.0
    q = np.correlate(h, 1 - h, mode='full') * dx
    sc = esc(h, dx)
    idle = 0
    for _ in range(sw):
        changed = False
        pairs = [(p, r) for p in range(n) for r in range(p + 1, n)]
        rng.shuffle(pairs)
        for (p, r) in pairs:
            b, a = pc(h, p, r, n, dx)
            lo = max(-h[p], h[r] - 1.0); hi = min(1.0 - h[p], h[r])
            if hi - lo < 1e-14: continue
            z = np.zeros(n); z[p] = 1.0; z[r] = -1.0
            tr = lsearch(h, q, z, b, a, lo, hi, n, dx, nc, nb, nr, pr)
            if tr is not None:
                h = tr; q = np.correlate(h, 1 - h, mode='full') * dx; sc = esc(h, dx); changed = True
        for _ in range(nsparse):
            sz = max(2, min(12, n))
            idx = rng.choice(n, sz, replace=False)
            vals = rng.standard_normal(sz); vals -= vals.mean()
            if np.max(np.abs(vals)) < 1e-14: continue
            z = np.zeros(n); z[idx] = vals
            z[idx[-1]] = -z[idx[:-1]].sum(); z /= np.max(np.abs(z))
            b = dx * (np.correlate(z, 1 - h, mode='full') - np.correlate(h, z, mode='full'))
            a = -dx * np.correlate(z, z, mode='full')
            if not (np.all(np.isfinite(b)) and np.all(np.isfinite(a))): continue
            iv = fiv(h, z, n)
            if iv is None: continue
            tr = lsearch(h, q, z, b, a, iv[0], iv[1], n, dx, nc, nb, nr, pr)
            if tr is not None:
                h = tr; q = np.correlate(h, 1 - h, mode='full') * dx; sc = esc(h, dx); changed = True
        if changed: idle = 0
        else:
            idle += 1
            if idle >= 2: break
    h = project_K(h, n, S)
    if not fchk(h, n, S): return None
    return h

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    init = globals().get('initial_h_values', None)
    ok = False
    if init is not None:
        try:
            init = np.array(init, dtype=np.float64).ravel()
        except Exception:
            init = None
        if init is not None and 2 <= len(init) <= 999 and np.all(np.isfinite(init)):
            if init.min() >= -1e-8 and init.max() <= 1 + 1e-8 and abs(init.sum() - len(init) / 2) <= 1e-6:
                ok = True
    if ok:
        n = len(init); base = project_K(init.copy(), n, n / 2.0)
    else:
        nd = kwargs.get('n_points', 45)
        if not isinstance(nd, (int, np.integer)) or isinstance(nd, bool): nd = 45
        n = max(2, min(999, int(nd))); base = np.full(n, 0.5)
    dx, S = 2.0 / n, n / 2.0
    if n <= 64: nc, nb, nr, pr, sw, nsparse, nshake = 25, 2, 4, 9, 4, 8, 3
    elif n <= 128: nc, nb, nr, pr, sw, nsparse, nshake = 17, 2, 3, 7, 3, 6, 1
    elif n <= 256: nc, nb, nr, pr, sw, nsparse, nshake = 13, 1, 2, 7, 2, 6, 1
    else: nc, nb, nr, pr, sw, nsparse, nshake = 9, 1, 1, 5, 1, 4, 0
    js = np.arange(n)
    starts = [base.copy()]
    starts.append(project_K(base + 0.05 * rng.standard_normal(n), n, S))
    starts.append(project_K(base + 0.15 * rng.standard_normal(n), n, S))
    starts.append(project_K(base + 0.30 * rng.standard_normal(n), n, S))
    starts.append(project_K(rng.uniform(0, 1, n), n, S))
    starts.append(project_K(rng.beta(0.5, 0.5, n), n, S))
    blk = np.zeros(n); blk[:int(S)] = 1.0
    if S - int(S) > 1e-12: blk[int(S)] = S - int(S)
    starts.append(project_K(blk[rng.permutation(n)], n, S))
    starts.append(project_K(0.5 + 0.45 * np.cos(2 * np.pi * js / n), n, S))
    starts.append(project_K(0.5 + 0.40 * np.sin(2 * np.pi * js / n), n, S))
    ramp = (2.0 * js / n) % 1.0
    starts.append(project_K(ramp, n, S))
    starts.append(project_K(1.0 - base, n, S))
    starts.append(project_K(base[::-1], n, S))
    starts.append(project_K((1.0 - base)[::-1], n, S))
    dedup = [base.copy()]
    for s in starts[1:]:
        if not fchk(s, n, S): continue
        if all(np.max(np.abs(s - ds)) > 1e-7 for ds in dedup): dedup.append(s)
        if len(dedup) >= 12: break
    starts = dedup
    best_h, best_s = base.copy(), esc(base, dx)
    for s in starts:
        sc0 = esc(s, dx)
        if sc0 < best_s - 1e-12: best_s, best_h = sc0, s.copy()
        hopt = ldesc(s, n, dx, rng, sw, nsparse, nc, nb, nr, pr)
        if hopt is not None:
            sc = esc(hopt, dx)
            if sc < best_s - 1e-12: best_s, best_h = sc, hopt.copy()
    sh_sds = [0.04, 0.12, 0.20][:nshake]
    for sd in sh_sds:
        hsh = project_K(best_h + sd * rng.standard_normal(n), n, S)
        if fchk(hsh, n, S):
            hopt = ldesc(hsh, n, dx, rng, sw, nsparse, nc, nb, nr, pr)
            if hopt is not None:
                sc = esc(hopt, dx)
                if sc < best_s - 1e-12: best_s, best_h = sc, hopt.copy()
    best_h = project_K(best_h, n, S)
    fb_h, fb_s = base.copy(), esc(base, dx)
    if not fchk(best_h, n, S):
        best_h, best_s = fb_h, fb_s
    else:
        fs = esc(best_h, dx)
        if fs > fb_s + 1e-10: best_h, best_s = fb_h, fb_s
        else: best_s = fs
    best_h = np.clip(best_h, 0, 1)
    best_h = project_K(best_h, n, S)
    final_s = esc(best_h, dx)
    return (best_h, final_s, n)
