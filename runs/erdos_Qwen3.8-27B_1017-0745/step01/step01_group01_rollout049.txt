def lag_gradient(x, active, n):
    omx = 1.0 - x
    G = np.zeros((len(active), n))
    for k, row in enumerate(active):

        xi = float(x[i])
        lo = -min(xi, step)

def project_capped(z, n):

            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15 * max(1.0, abs(mid)):

    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    res = s - float(np.sum(h))

def lag_values(x, n):

        m = n - abs(d)

        k += 1
    return vals

def lag_gradient(x, active, n):
    omx = 1.0 - x

        j0 = -d if d < 0 else 0
        m = n - abs(d)
        G[k, i0:i0 + m] += omx[j0:j0 + m]
        G[k, j0:j0 + m] -= x[i0:i0 + m]
    return G
