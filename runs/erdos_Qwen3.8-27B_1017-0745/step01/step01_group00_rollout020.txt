        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0 and 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None:
        n = 47
    if parent is None:
        parent = np.full(n, 0.5)
    fallback = project_capped(parent, n)
    if fallback is None:
        fallback = np.full(n, 0.5)
    fallback = project_capped(fallback, n)
    if fallback is None:
        fallback = np.full(n, 0.5)
    fb_score = exact_score(fallback, n)
    rng = np.random.default_rng(int(seed))
    if n == 1:
        return np.array([0.5]), 0.0, 1
    if n <= 64:
        ps, mg, sl, mr, pr, pd = 40, 120, 20, 3, 28, 10
    elif n <= 128:
        ps, mg, sl, mr, pr, pd = 36, 90, 16, 2, 20, 8
    elif n <= 256:
        ps, mg, sl, mr, pr, pd = 28, 55, 12, 1, 12, 6
    elif n <= 512:
        ps, mg, sl, mr, pr, pd = 20, 30, 10, 1, 8, 4
    else:
        ps, mg, sl, mr, pr, pd = 14, 18, 8, 1, 5, 4
    scale_n = float(np.clip(np.sqrt(47.0 / n), 0.5, 1.5))
    starts = make_starts(fallback, n, rng)
    all_sc = [exact_score(s, n) for s in starts]
    if np.max(np.all(np.abs(np.array(starts) - fallback[None, :]), axis=1)) > 1e-10:
        starts.append(fallback.copy())
        all_sc.append(fb_score)
    else:
        starts.insert(0, fallback.copy())
        all_sc.insert(0, fb_score)
    order = np.argsort(all_sc)
    scored_starts = [(all_sc[k], starts[k]) for k in order]
    pop = []
    pop_sc = []
    for sc_, st_ in scored_starts[:ps]:
        pop.append(st_.copy())
        pop_sc.append(sc_)
    tried = 0
    while len(pop) < ps and tried < 6 * ps:
        tried += 1
        r = rng.random()
        best_i = int(np.argmin(pop_sc))
        if r < 0.45:
            sig = rng.choice([0.03, 0.08, 0.16, 0.32]) * scale_n
            v = project_capped(pop[best_i] + rng.normal(0, sig, n), n)
        elif r < 0.70:
            sig = rng.choice([0.05, 0.12, 0.25]) * scale_n
            v = project_capped(pop[int(rng.integers(len(pop)))] + rng.normal(0, sig, n), n)
        elif r < 0.85:
            v = project_capped(rng.uniform(0, 1, n), n)
        else:
            perm = rng.permutation(n)
            bb = np.zeros(n)
            bb[perm[:n // 2]] = 1.0
            if n % 2:
                bb[perm[n // 2]] = 0.5
            v = project_capped(bb, n)
        if v is None:
            continue
        dup = any(np.max(np.abs(v - pp)) < 1e-8 for pp in pop)
        if not dup:
            pop.append(v)
            pop_sc.append(exact_score(v, n))
    while len(pop) < ps:
        pop.append(pop[int(np.argmin(pop_sc))].copy())
        pop_sc.append(min(pop_sc))
    F_arr = rng.uniform(0.45, 0.85, ps)
    CR_arr = rng.uniform(0.40, 0.90, ps)
    gi = int(np.argmin(pop_sc))
    global_best = pop[gi].copy()
    global_score = pop_sc[gi]
    stalls = 0
    restarts = 0
    for _gen in range(mg):
        snap = [p.copy() for p in pop]
        scs = list(pop_sc)
        bi = int(np.argmin(scs))
        xb = snap[bi].copy()
        gen_improved = False
        for i in range(ps):
            Fi = F_arr[i]
            CRi = CR_arr[i]
            if rng.random() < 0.1:
                Fi = 0.35 + 0.55 * rng.random()
            if rng.random() < 0.1:
                CRi = 0.10 + 0.85 * rng.random()
            others = [j for j in range(ps) if j != i]
            idxs = rng.choice(others, size=3, replace=False)
            r1, r2, r3 = int(idxs[0]), int(idxs[1]), int(idxs[2])
            if rng.random() < 0.5:
                mut = snap[r1] + Fi * (snap[r2] - snap[r3])
            else:
                mut = snap[i] + Fi * (xb - snap[i]) + Fi * (snap[r2] - snap[r3])
            mask = rng.random(n) < CRi
            mask[int(rng.integers(n))] = True
            trial = np.where(mask, mut, snap[i])
            v = project_capped(trial, n)
            if v is None:
                continue
            scv = exact_score(v, n)
            if scv < scs[i] - 1e-12:
                pop[i] = v
                pop_sc[i] = scv
                F_arr[i] = Fi
                CR_arr[i] = CRi
                if scv < global_score - 1e-12:
                    global_score = scv
                    global_best = v.copy()
                    gen_improved = True
            else:
                stalls += 1
        if gen_improved:
            stalls = max(0, stalls - ps)
        if global_score < min(pop_sc) - 1e-10:
            wi = int(np.argmax(pop_sc))
            pop[wi] = global_best.copy()
            pop_sc[wi] = global_score
        if stalls >= sl and restarts < mr:
            restarts += 1
            stalls = 0
            oor = np.argsort(pop_sc)
            nk = max(2, ps // 4)
            ki = oor[:nk]
            npop = [pop[k].copy() for k in ki]
            nsc = [pop_sc[k] for k in ki]
            if all(np.max(np.abs(x - global_best)) > 1e-10 for x in npop):
                npop.append(global_best.copy())
                nsc.append(global_score)
            rsig = min([0.06, 0.16, 0.32][min(restarts - 1, 2)] * scale_n, 0.45)
            fill = 0
            while len(npop) < ps and fill < 4 * ps:
                fill += 1
                rr = rng.random()
                if rr < 0.5:
                    vv = project_capped(global_best + rng.normal(0, rsig, n), n)
                elif rr < 0.75:
                    vv = project_capped(npop[int(rng.integers(len(npop)))] + rng.normal(0, rsig * 0.5, n), n)
                elif rr < 0.875:
                    vv = project_capped(rng.uniform(0, 1, n), n)
                else:
                    perm = rng.permutation(n)
                    bbv = np.zeros(n)
                    bbv[perm[:n // 2]] = 1.0
                    if n % 2:
                        bbv[perm[n // 2]] = 0.5
                    vv = project_capped(bbv, n)
                if vv is None:
                    continue
                if not any(np.max(np.abs(vv - x)) < 1e-8 for x in npop):
                    npop.append(vv)
                    nsc.append(exact_score(vv, n))
            while len(npop) < ps:
                npop.append(global_best.copy())
                nsc.append(global_score)
            pop = npop
            pop_sc = nsc
            F_arr = rng.uniform(0.45, 0.85, ps)
            CR_arr = rng.uniform(0.40, 0.90, ps)
    delta = min(0.25, 0.12 * scale_n)
    for _rnd in range(pr):
        if delta < 1e-4:
            break
        bt, bts = None, global_score
        dirs = []
        for _ in range(pd // 2):
            d = rng.standard_normal(n)
            d -= d.mean()
            mx = float(np.max(np.abs(d)))
            if mx > 1e-15 and np.isfinite(mx):
                dirs.append(d / mx)
        for _ in range(pd - pd // 2):
            i1 = int(rng.integers(len(pop)))
            i2 = int(rng.integers(len(pop)))
            if i1 == i2:
                i2 = (i2 + 1) % len(pop)
            d = pop[i1] - pop[i2]
            d -= d.mean()
            mx = float(np.max(np.abs(d)))
            if mx > 1e-15 and np.isfinite(mx):
                dirs.append(d / mx)
        for v in dirs:
            tlo, thi = -1e10, 1e10
            for ii in range(n):
                if v[ii] > 1e-15:
                    tlo = max(tlo, -global_best[ii] / v[ii])
                    thi = min(thi, (1.0 - global_best[ii]) / v[ii])
                elif v[ii] < -1e-15:
                    tlo = max(tlo, (1.0 - global_best[ii]) / v[ii])
                    thi = min(thi, -global_best[ii] / v[ii])
            if tlo >= thi:
                continue
            tvals = []
            for t in (-delta, -delta * 0.5, delta * 0.5, delta):
                tc = float(np.clip(t, tlo + 1e-12, thi - 1e-12))
                if abs(tc) > 1e-12 and all(abs(tc - tv) > 1e-12 for tv in tvals):
                    tvals.append(tc)
            for t in tvals:
                pv = project_capped(global_best + t * v, n)
                if pv is None or not is_feasible(pv, n):
                    continue
                sc = exact_score(pv, n)
                if sc < bts - 1e-12:
                    bts = sc
                    bt = pv
        if bt is not None and bts < global_score - 1e-12:
            global_best = bt.copy()
            global_score = bts
            delta = min(0.25, 1.2 * delta)
        else:
            delta *= 0.5
    try:
        cs_h, cs_s = coord_search(global_best, n, 2)
        if cs_s < global_score - 1e-12:
            global_best, global_score = cs_h, cs_s
    except Exception:
        pass
    final = project_capped(global_best, n)
    if final is None or not is_feasible(final, n):
        final = fallback
        fs = fb_score
    else:
        fs = exact_score(final, n)
        if fs > fb_score + 1e-12:
            final = fallback
            fs = fb_score
    return final, float(fs), int(n)
