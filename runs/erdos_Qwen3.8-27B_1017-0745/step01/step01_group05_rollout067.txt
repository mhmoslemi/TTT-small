        if not check_feasible(st, n):
            st = np.full(n, 0.5)
        dup = False
        for us in unique_starts:
            if float(np.max(np.abs(st - us))) < 1e-8:
                dup = True
                break
        if not dup:
            unique_starts.append(st)
    for st in unique_starts:
        try:
            bh, bs = optimize_start(st, n, mo, mi)
            if bs < best_s - 1e-12:
                best_s = bs
                best_h = bh.copy()
        except Exception:
            pass
    if n <= 100:
        try:
            st2 = project_capped(best_h + 0.025 * rng_global.standard_normal(n), n)
            bh2, bs2 = optimize_start(st2, n, mo // 2, mi)
            if bs2 < best_s - 1e-12:
                best_s = bs2
                best_h = bh2.copy()
        except Exception:
            pass
        try:
            ph, ps = pair_polish(best_h, n, 200)
            if ps < best_s - 1e-12:
                best_s = ps
                best_h = ph.copy()
        except Exception:
            pass
    final_h = project_capped(best_h, n)
    final_s = exact_score(final_h, n)
    if not check_feasible(final_h, n) or final_s > fb_s + 1e-12:
        final_h = fallback.copy()
        final_s = fb_s
    return final_h.astype(np.float64), float(final_s), int(n)
