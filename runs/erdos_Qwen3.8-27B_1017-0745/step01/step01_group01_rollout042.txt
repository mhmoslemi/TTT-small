```python
def line_coefficients(h, p, q, n, c):
    L = 2 * n - 1
    a = np.zeros(L)
    b = np.zeros(L)
    Delta = q - p
    a[n-1] = -2.0
    a[n-1+Delta] = 1.0
    a[n-1-Delta] = 1.0
    for d in range(-(n-1), n):
        row = d + n - 1
        Dp = 0.0
        Dq = 0.0
        if 0 <= p+d < n:
            Dp += 1.0 - h[p+d]
        if 0 <= p-d < n:
            Dp -= h[p-d]
        if 0 <= q+d < n:
            Dq += 1.0 - h[q+d]
        if 0 <= q-d < n:
            Dq -= h[q-d]
        b[row] = Dp - Dq
    return a, b
```

```python
def exact_line_minimizer(h, p, q, n, c, a, b):
    lo = max(-h[p], h[q] - 1.0)
    hi = min(1.0 - h[p], h[q])
    if hi - lo < 1e-14:
        return 0.0
    
    L = 2 * n - 1
    Delta = q - p
    special = [n-1, n-1+Delta, n-1-Delta]
    
    # Collect affine lines (non-special)
    affine_idx = []
    for row in range(L):
        if row not in special and a[row] == 0.0:
            affine_idx.append(row)
    
    # Build upper hull of affine lines
    # Sort by slope b[row], for equal slope keep max c[row]
    if affine_idx:
        lines = [(b[i], c[i]) for i in affine_idx]
        lines.sort(key=lambda x: (x[0], -x[1]))
        # Remove duplicates (same slope, keep max intercept)
        deduped = [lines[0]]
        for i in range(1, len(lines)):
            if lines[i][0] == deduped[-1][0]:
                if lines[i][1] > deduped[-1][1]:
                    deduped[-1] = lines[i]
            else:
                deduped.append(lines[i])
        
        # Build upper hull
        hull = []  # list of (slope, intercept, start)
        for m, cc in deduped:
            if not hull:
                hull.append((m, cc, float('-inf')))
            else:
                x = (hull[-1][1] - cc) / (m - hull[-1][0]) if m != hull[-1][0] else float('-inf')
                while len(hull) > 1 and x <= hull[-1][2] + 1e-15:
                    hull.pop()
                    x = (hull[-1][1] - cc) / (m - hull[-1][0]) if m != hull[-1][0] else float('-inf')
                hull.append((m, cc, x))
    
    # Generate candidates
    candidates = [lo, hi, 0.0]
    
    if affine_idx and hull:
        # Add hull switch points
        for i in range(1, len(hull)):
            if lo <= hull[i][2] <= hi:
                candidates.append(hull[i][2])
        
        # For each hull segment, find intersections with special quadratics
        for seg_i in range(len(hull)):
            m_k, c_k, start_k = hull[seg_i]
            end_k = hull[seg_i+1][2] if seg_i + 1 < len(hull) else float('inf')
            seg_lo = max(lo, start_k)
            seg_hi = min(hi, end_k)
            if seg_lo >= seg_hi:
                continue
            for s in special:
                # (a_s)δ² + (b_s - m_k)δ + (c_s - c_k) = 0
                rs = quadratic_roots(a[s], b[s] - m_k, c[s] - c_k, seg_lo, seg_hi)
                candidates.extend(rs)
    
    # Intersections between special quadratics
    for i in range(len(special)):
        for j in range(i+1, len(special)):
            s1, s2 = special[i], special[j]
            # (a1-a2)δ² + (b1-b2)δ + (c1-c2) = 0
            rs = quadratic_roots(a[s1]-a[s2], b[s1]-b[s2], c[s1]-c[s2], lo, hi)
            candidates.extend(rs)
    
    # Vertices of convex special quadratics
    for s in special:
        if a[s] > 0 and b[s] != 0:
            v = -b[s] / (2.0 * a[s])
            if lo <= v <= hi:
                candidates.append(v)
    
    # Evaluate
    candidates = np.array([x for x in candidates if np.isfinite(x) and lo - 1e-12 <= x <= hi + 1e-12])
    candidates = np.clip(candidates, lo, hi)
    
    best_val = float('inf')
    best_delta = 0.0
    for d in candidates:
        # Evaluate max of all lags at delta=d
        vals = c + b*d + a*d*d
        v = np.max(vals)
        if v < best_val:
            best_val = v
            best_delta = d
    
    return best_delta
```

```python
def quadratic_roots(A, B, C, lo, hi):
    if abs(A) <= 1e-14:
        if abs(B) <= 1e-14:
            return []
        r = -C / B
        if lo - 1e-12 <= r <= hi + 1e-12:
            return [max(lo, min(hi, r))]
        return []
    disc = B*B - 4*A*C
    if disc < -1e-14 * (B*B + 4*abs(A*C) + 1):
        return []
    disc = max(0.0, disc)
    sqrt_d = math.sqrt(disc)
    q = -0.5 * (B + math.copysign(sqrt_d, B))
    roots = []
    if abs(q) > 1e-14:
        r1 = q / A
        r2 = C / q
    else:
        r1 = r2 = -B / (2*A)
    tol = 1e-12 * max(1.0, abs(lo), abs(hi))
    if lo - tol <= r1 <= hi + tol:
        roots.append(max(lo, min(hi, r1)))
    if abs(r2 - r1) > 1e-12 and lo - tol <= r2 <= hi + tol:
        roots.append(max(lo, min(hi, r2)))
    return roots
```

```python
def build_edge_list(n):
    edges = []
    for i in range(n-1):
        edges.append((i, i+1))
    r = 2
    while r < n:
        i = 0
        while i + r < n:
            edges.append((i, i+r))
            i += 2*r
        r *= 2
    if n > 2 and (0, n-1) not in edges:
        edges.append((0, n-1))
    # Remove duplicates
    seen = set()
    unique = []
    for p, q in edges:
        if p > q:
            p, q = q, p
        if (p, q) not in seen:
            seen.add((p, q))
            unique.append((p, q))
    return unique
```

```python
def project_capped(z, n):
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if mid == lo or mid == hi:
            break
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(3):
        if abs(r) < 1e-13 * max(1, n):
            break
        if r > 0:
            for i in range(n):
                if r <= 0:
                    break

                    adj = min(r, space)
                    h[i] += adj
                    r -= adj
        else:

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return float('inf')

        return float('inf')
    try:
        val = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
    except Exception:
        return float('inf')
    return val if np.isfinite(val) else float('inf')

def lag_raw_values(h, n):
    L = 2 * n - 1

        row = d + n - 1
        if d >= 0:
            m = n - d
            c[row] = np.sum(h[:m] * (1.0 - h[d:d + m]))
        else:
            m = n + d
            c[row] = np.sum(h[-m:] * (1.0 - h[:m]))
    return c

def quadratic_roots(A, B, C, lo, hi):
    if abs(A) <= 1e-14:
        if abs(B) <= 1e-14:
            return []
        r = -C / B
        tol = 1e-12 * max(1.0, abs(lo), abs(hi))
        if lo - tol <= r <= hi + tol:
            return [max(lo, min(hi, r))]
        return []
    disc = B * B - 4.0 * A * C
    if disc < -1e-14 * (B * B + 4.0 * abs(A * C) + 1.0):
        return []
    disc = max(0.0, disc)
    sq = math.sqrt(disc)
    q = -0.5 * (B + math.copysign(sq, B if B != 0 else 1.0))
    roots = []
    tol = 1e-12 * max(1.0, abs(lo), abs(hi))
    if abs(q) > 1e-14:
        r1, r2 = q / A, C / q
    else:

    if abs(r2 - r1) > 1e-12 and lo - tol <= r2 <= hi + tol:
        roots.append(max(lo, min(hi, r2)))
    return roots

def line_coeffs(h, p, q, n):
    L = 2 * n - 1
    a = np.zeros(L)
    b = np.zeros(L)
    D = q - p
    a[n - 1] = -2.0
    a[n - 1 + D] = 1.0
    a[n - 1 - D] = 1.0
    for d in range(-(n - 1), n):

        if 0 <= p - d < n:
            dp -= h[p - d]
        if 0 <= q + d < n:
            dq += 1.0 - h[q + d]
        if 0 <= q - d < n:
            dq -= h[q - d]
        b[row] = dp - dq
    return a, b

def build_edges(n):
    edges = []
    for i in range(n - 1):
        edges.append((i, i + 1))
    r = 2
    while r < n:
        i = 0
        while i + r < n:
            edges.append((i, i + r))
            i += 2 * r
        r *= 2
    if n > 2:
        edges.append((0, n - 1))
    seen = set()
    out = []
    for p, q in edges:
        if p > q:
            p, q = q, p
        if (p, q) not in seen:
            seen.add((p, q))
            out.append((p, q))
    return out
```

```python
import numpy as np
import math

def project_capped(z, n):
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if mid == lo or mid == hi:
            break
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(3):
        if abs(r) < 1e-13 * max(1, n):
            break
        if r > 0:
            for i in range(n):
                if r <= 0:
                    break
                sp = 1.0 - h[i]
                if sp > 0:
                    adj = min(r, sp)
                    h[i] += adj
                    r -= adj
        else:
            for i in range(n):
                if r >= 0:
                    break
                adj = min(-r, h[i])
                if adj > 0:
                    h[i] -= adj
                    r += adj
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(np.sum(h) - s) > 5e-12 * max(1, n):
        return None
    return h

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return float('inf')
    if abs(np.sum(h) - n / 2.0) > 5e-10 * max(1, n):
        return float('inf')
    try:
        val = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
    except Exception:
        return float('inf')
    return val if np.isfinite(val) else float('inf')

def lag_raw(h, n):
    L = 2 * n - 1
    c = np.zeros(L)
    for d in range(-(n - 1), n):
        row = d + n - 1
        if d >= 0:
            m = n - d
            c[row] = np.sum(h[:m] * (1.0 - h[d:d + m]))
        else:
            m = n + d
            c[row] = np.sum(h[-m:] * (1.0 - h[:m]))
    return c

def quad_roots(A, B, C, lo, hi):
    if abs(A) <= 1e-14:
        if abs(B) <= 1e-14:
            return []
        r = -C / B
        tol = 1e-12 * max(1.0, abs(lo), abs(hi))
        return [max(lo, min(hi, r))] if lo - tol <= r <= hi + tol else []
    disc = B * B - 4.0 * A * C
    if disc < -1e-14 * (B * B + 4.0 * abs(A * C) + 1.0):
        return []
    disc = max(0.0, disc)
    sq = math.sqrt(disc)
    q = -0.5 * (B + math.copysign(sq, B if B != 0 else 1.0))
    roots = []
    tol = 1e-12 * max(1.0, abs(lo), abs(hi))
    if abs(q) > 1e-14:
        r1, r2 = q / A, C / q
    else:
        r1 = r2 = -B / (2.0 * A)
    if lo - tol <= r1 <= hi + tol:
        roots.append(max(lo, min(hi, r1)))
    if abs(r2 - r1) > 1e-12 and lo - tol <= r2 <= hi + tol:
        roots.append(max(lo, min(hi, r2)))
    return roots

def line_coeffs(h, p, q, n):
    L = 2 * n - 1
    a = np.zeros(L)
    b = np.zeros(L)
    D = q - p
    a[n - 1] = -2.0
    a[n - 1 + D] = 1.0
    a[n - 1 - D] = 1.0
    for d in range(-(n - 1), n):
        row = d + n - 1
        dp = 0.0
        dq = 0.0
        if 0 <= p + d < n:
            dp += 1.0 - h[p + d]
        if 0 <= p - d < n:
            dp -= h[p - d]
        if 0 <= q + d < n:
            dq += 1.0 - h[q + d]
        if 0 <= q - d < n:
            dq -= h[q - d]
        b[row] = dp - dq
    return a, b

def exact_line_min(h, p, q, n, c, a, b):
    lo = max(-h[p], h[q] - 1.0)
    hi = min(1.0 - h[p], h[q])
    if hi - lo < 1e-14:
        return 0.0
    L = 2 * n - 1
    D = q - p
    special = [n - 1, n - 1 + D, n - 1 - D]
    cands = [lo, hi, 0.0]
    aff = [(b[i], c[i]) for i in range(L) if i not in special]
    if aff:
        aff.sort(key=lambda x: (x[0], -x[1]))
        dd = [aff[0]]
        for i in range(1, len(aff)):
            if aff[i][0] == dd[-1][0]:
                if aff[i][1] > dd[-1][1]:
                    dd[-1] = aff[i]
            else:
                dd.append(aff[i])
        hull = []
        for ms, cc in dd:
            if not hull:
                hull.append([ms, cc, float('-inf')])
            elif ms == hull[-1][0]:
                if cc > hull[-1][1]:
                    hull[-1] = [ms, cc, hull[-1][2]]
            else:
                x = (hull[-1][1] - cc) / (ms - hull[-1][0])
                while len(hull) > 1 and x <= hull[-1][2] + 1e-15:
                    hull.pop()
                    x = (hull[-1][1] - cc) / (ms - hull[-1][0])
                hull.append([ms, cc, x])
        for si in range(len(hull)):
            if si > 0 and lo - 1e-12 <= hull[si][2] <= hi + 1e-12:
                cands.append(max(lo, min(hi, hull[si][2])))
            mk, ck = hull[si][0], hull[si][1]
            seg_lo = max(lo, hull[si][2])
            seg_hi = min(hi, hull[si + 1][2]) if si + 1 < len(hull) else hi
            if seg_lo < seg_hi:
                for s in special:
                    cands.extend(quad_roots(a[s], b[s] - mk, c[s] - ck, seg_lo, seg_hi))
        for i in range(len(special)):
            for j in range(i + 1, len(special)):
                s1, s2 = special[i], special[j]
                cands.extend(quad_roots(a[s1] - a[s2], b[s1] - b[s2], c[s1] - c[s2], lo, hi))

        if a[s] > 0 and abs(b[s]) > 1e-15:

        return 0.0
    best_v, best_d = float('inf'), 0.0
    for d in ca:
        vals = c + b * d + a * d * d
        v = float(np.max(vals))
        if v < best_v:
            best_v, best_d = v, float(d)
    return best_d

def build_edges(n):
    edges = []
    for i in range(n - 1):
        edges.append((i, i + 1))
    r = 2
    while r < n:
        i = 0
        while i + r < n:
            edges.append((i, i + r))
            i += 2 * r
        r *= 2
    if n > 2:
        edges.append((0, n - 1))
    seen = set()
    out = []
    for p, q in edges:
        if p > q:
            p, q = q, p
        if (p, q) not in seen:
            seen.add((p, q))
            out.append((p, q))
    return out
```
