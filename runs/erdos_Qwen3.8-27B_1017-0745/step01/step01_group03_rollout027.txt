import numpy as np
import math

def project_box_sum(u, S):
    u = np.asarray(u, dtype=np.float64)
    n = len(u)
    lo = float(np.min(u)) - 1.0
    hi = float(np.max(u))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(u - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(u - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-14:
        if resid > 0:
            for i in range(n):
                add = min(resid, 1.0 - y[i])
                y[i] += add
                resid -= add
                if abs(resid) < 1e-15:
                    break
        else:
            for i in range(n):
                sub = min(-resid, y[i])
                y[i] -= sub
                resid += sub
                if abs(resid) < 1e-15:
                    break
    if not np.all(np.isfinite(y)):
        return None
    if np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    if abs(np.sum(y) - S) > 1e-10 * max(1.0, float(n)):
        return None
    y = np.clip(y, 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 0:
        if resid > 0 and y[0] < 1.0:
            y[0] += min(resid, 1.0 - y[0])
        elif resid < 0 and y[0] > 0.0:
            y[0] += max(resid, -y[0])
    return y

def check_feasible(h, n):
    if h is None or h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1.0, float(n)):
        return False
    return True

def exact_lags(h):
    return np.correlate(h, 1.0 - h, mode='full')

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    lags = exact_lags(h)
    if not np.all(np.isfinite(lags)):
        return float('inf')
    return float(np.max(lags) * (2.0 / n))

def pair_delta_lags(h, p, q, a, n):
    Q = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        d = 0.0
        pk = p - k
        qk = q - k
        pkk = p + k
        qkk = q + k
        if 0 <= pk < n:
            d += a * (1.0 - h[pk])
        if 0 <= qk < n:
            d -= a * (1.0 - h[qk])
        if 0 <= pkk < n:
            d -= h[pkk] * a
        if 0 <= qkk < n:
            d += a * h[qkk]
        if k == 0:
            d -= 2.0 * a * a
        elif k == p - q:
            d += a * a
        elif k == q - p:
            d += a * a
        Q[k + n - 1] = d
    return Q

        acc_count = 0
        stage_h = h.copy()
        stage_C = C
        for _ in range(moves_per_stage):
            p, q = rng.choice(n, size=2, replace=False)
            ip = int(p)
            iq = int(q)
            if ip > iq:
                ip, iq = iq, ip

                h[ip] += a
                h[iq] -= a
                Q = Q_t
                C = C_t
