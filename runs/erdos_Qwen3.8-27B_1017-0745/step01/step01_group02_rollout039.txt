def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    if S < 0 or S > n:
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(v - tau, 0.0, 1.0)
    r = S - float(np.sum(y))
    tol = 5e-13 * max(1, n)
    if r > tol:
        for i in range(n):
            d = min(r, 1.0 - y[i])
            y[i] += d
            r -= d
            if r <= tol:
                break
    elif r < -tol:
        for i in range(n):
            d = min(-r, y[i])
            y[i] -= d
            r += d
            if r >= -tol:
                break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-9) or np.any(y > 1.0 + 1e-9):
        return None
    return y

def quad_roots(A, B, C):
    if abs(A) < 1e-15:
        if abs(B) < 1e-15:
            return []
        return [-C / B]
    disc = B * B - 4.0 * A * C
    scale = max(1.0, B * B, abs(4.0 * A * C))
    if disc < -1e-14 * scale:
        return []
    if disc < 0:
        disc = 0.0
    sqrt_d = math.sqrt(disc)
    q = -0.5 * (B + math.copysign(sqrt_d, B) if B != 0 else sqrt_d)
    if q != 0:
        r1 = q / A
        r2 = C / q
    else:
        r1 = r2 = -B / (2.0 * A)
    roots = []
    for r in [r1, r2]:
        if np.isfinite(r):
            roots.append(r)
    return roots

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    if S < -1e-9 or S > n + 1e-9:
        return None
    lo = float(np.min(v)) - 1.0
