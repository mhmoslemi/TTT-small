import math

def project_onto_k(y, n):
    S = n / 2.0
    if not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0

        else:

        if cnt == 0:

        np.clip(c, 0.0, 1.0, out=c)

    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lag_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def helmert_dir(j, n):
    d = np.zeros(n)
    s = math.sqrt((j + 1.0) * (j + 2.0))
    d[:j+1] = 1.0 / s
    d[j+1] = -(j + 1.0) / s
    return d

def line_search(h, d, n):
    # Find min of max_l (q_l + t*b_l + t^2*a_l) over feasible t
    lo, hi = 0.0, 0.0
    for i in range(n):
        di = d[i]
        if abs(di) < 1e-15:
            continue
        if di > 0:
            lo = max(lo, -h[i] / di)
            hi = min(hi, (1.0 - h[i]) / di)
        else:
            lo = max(lo, (1.0 - h[i]) / di)
            hi = min(hi, -h[i] / di)
    if hi - lo < 1e-12:
        return None
    # Compute coefficients
    q = lag_vec(h)  # dx * correlate(h, 1-h)
    b = -np.correlate(d, h, mode='full') - np.correlate(h, d, mode='full')
    b = b * (2.0 / n)
    a = -np.correlate(d, d, mode='full')
    a = a * (2.0 / n)
    m = 2 * n - 1
    # Collect candidate t values
    cands = [lo, hi, 0.0]
    # Stationary points
    for l in range(m):
        if abs(a[l]) > 1e-15:
            tp = -b[l] / (2.0 * a[l])
            if lo <= tp <= hi:
                cands.append(tp)
    # Pairwise intersections
    for i in range(m):
        for j in range(i+1, m):
            da = a[i] - a[j]
            db = b[i] - b[j]
            dc = q[i] - q[j]
            if abs(da) < 1e-15:
                if abs(db) > 1e-15:
                    tp = -dc / db
                    if lo <= tp <= hi:
                        cands.append(tp)
            else:
                disc = db*db - 4.0*da*dc
                if disc >= -1e-12:
                    if disc < 0:
                        disc = 0.0
                    sq = math.sqrt(disc)
                    tp1 = (-db + sq) / (2.0 * da)
                    tp2 = (-db - sq) / (2.0 * da)
                    if lo <= tp1 <= hi:
                        cands.append(tp1)
                    if lo <= tp2 <= hi:
                        cands.append(tp2)
    # Evaluate
    if len(cands) == 0:
        return None
    cands = np.array(cands)
    cands = cands[(cands >= lo) & (cands <= hi)]
    if len(cands) == 0:
        return None
    best_t = cands[0]
    best_v = np.inf
    for t in cands:
        vals = q + t * b + t * t * a
        v = vals.max()
        if v < best_v:
            best_v = v
            best_t = t
    return best_t

def polish_start(h0, n, rng, max_sweeps=2):
    h = h0.copy()
    s = exact_score(h)
    for sweep in range(max_sweeps):
        improved = False
        perm = rng.permutation(n - 1)

                hn = project_onto_k(hn, n)
                sn = exact_score(hn)
                if sn < s - 1e-12:
                    h = hn
                    s = sn
                    improved = True
        if not improved:
            break
    return h, s

    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.ndim == 1 and 2 <= len(p) <= 999 and np.all(np.isfinite(p)) and np.max(np.abs(p)) < 1e6:
            n = len(p)
            fallback = project_onto_k(p, n)
        else:
            fallback = np.full(n, 0.5)
    else:
        np_kw = kwargs.get('n_points', None)
        if isinstance(np_kw, int) and 2 <= np_kw <= 999:
            n = np_kw
        fallback = np.full(n, 0.5)
    fb_s = exact_score(fallback)
    best_h, best_s = fallback.copy(), fb_s

    def try_cand(hc):
        nonlocal best_h, best_s
        hc = project_onto_k(np.clip(hc, 0.0, 1.0), n)
        s = exact_score(hc)
        if s < best_s - 1e-13:
            best_s, best_h = s, hc.copy()
        return hc, s

    # Polish parent
    h1, s1 = polish(fallback, n, rng, sweeps=3)
    try_cand(h1)
    # Polish complement
    h2, s2 = polish(1.0 - fallback, n, rng, sweeps=2)
    try_cand(h2)
    # Polish reversed
    h3, s3 = polish(fallback[::-1], n, rng, sweeps=2)
    try_cand(h3)
    # Random perturbations + polish
    for sig in [0.05, 0.15, 0.30, 0.50, 0.80]:
        pert = fallback + sig * rng.standard_normal(n)
        hp, sp = polish(pert, n, rng, sweeps=2)
        try_cand(hp)
    # Random direction polish on best
    hr, sr = rand_dir_polish(best_h, n, rng, ndirs=12)
    try_cand(hr)
    # Second pass polish
    h4, s4 = polish(best_h, n, rng, sweeps=3)
    try_cand(h4)
    # SDP relaxation (optional, n<=50)
    if n <= 50:
        dx = 2.0 / n
        h_rel, X_rel = solve_sdp(n, dx, best_s)
        if h_rel is not None:
            hc, _ = try_cand(h_rel)
            for alpha in [0.25, 0.5, 0.75, 1.0]:
                try_cand(fallback + alpha * (h_rel - fallback))
            hp, sp = polish(hc, n, rng, sweeps=3)
            try_cand(hp)
            props = cov_proposals(hc, X_rel, n, rng)
            for pr in props:
                hp2, sp2 = polish(pr, n, rng, sweeps=2)
                try_cand(hp2)
    # Final random restarts
    for _ in range(4):
        d = rng.standard_normal(n)
        d -= d.mean()
        hp, sp = polish(best_h + 0.1 * d, n, rng, sweeps=2)
        try_cand(hp)
    # Final polish
    hf, sf = polish(best_h, n, rng, sweeps=4)
    try_cand(hf)
    # Validate and return
    best_h = project_onto_k(best_h, n)
    final_s = exact_score(best_h)
    if final_s > fb_s + 1e-12:
        best_h, final_s = fallback.copy(), fb_s
    best_h = project_onto_k(best_h, n)
    final_s = exact_score(best_h)
    return best_h.astype(np.float64), float(final_s), int(n)
