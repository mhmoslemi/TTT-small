The problem is to minimize C₅ = max_k ∫ h(x)(1-h(x+k)) dx over step functions h: [0,2]→[0,1] with ∫h = 1.

The discretization uses n points over [0,2], so dx = 2/n, and the constraint is sum(h) * dx = 1, i.e., sum(h) = n/2.

The score is computed as max(np.correlate(h, 1-h, mode="full")) * dx.

import numpy as np
import math

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.ndim != 1 or len(v) == 0 or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(v - tau, 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-14:
        idx = np.argsort(1.0 - y)
        for i in idx:
            d = min(r, 1.0 - y[i])
            y[i] += d
            r -= d
            if r <= 0:
                break
    elif r < -1e-14:
        idx = np.argsort(y)
        for i in idx:
            d = min(-r, y[i])
            y[i] -= d
            r += d
            if r >= 0:
                break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-10) or np.any(y > 1.0 + 1e-10):
        return None
    n = len(y)
    if abs(float(np.sum(y)) - S) > 1e-8 * max(1, n):
        return None
    return np.clip(y, 0.0, 1.0)

def check_feasible(h, n):
    if len(h) != n or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(float(np.sum(h)) - 0.5 * n) <= 1e-8 * max(1, n)

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    try:
        c = np.correlate(h, 1.0 - h, mode='full')
        if not np.all(np.isfinite(c)) or len(c) != 2 * n - 1:
            return float('inf')
        return float(np.max(c) * (2.0 / n))
    except Exception:
        return float('inf')

def make_direction(n, rng):
    if n <= 64:
        sizes = [2, 4, 8, n]
    elif n <= 256:
        sizes = [2, 8, 16, n]
    else:
        sizes = [2, 8, 32, n]
    sizes = [s for s in sizes if s <= n]
    for _ in range(20):
        k = sizes[rng.integers(0, len(sizes))]
        idx = rng.choice(n, size=k, replace=False)
        g = rng.standard_normal(k)
        g = g - np.mean(g)
        g[-1] = -math.fsum(g[:-1])
        amax = float(np.max(np.abs(g)))
        if amax > 1e-12:
            d = np.zeros(n)
            g = g / amax
            d[idx] = g
            return d
    idx = rng.permutation(n)[:2]
    d = np.zeros(n)
    d[idx[0]] = 1.0
    d[idx[1]] = -1.0
    return d

def feasible_interval(h, d):
    t_lo = -1e30
    t_hi = 1e30
    for k in range(len(h)):
        dk = d[k]
        if abs(dk) < 1e-15:
            continue
        if dk > 0:
            t_lo = max(t_lo, -h[k] / dk)
            t_hi = min(t_hi, (1.0 - h[k]) / dk)
        else:
            t_lo = max(t_lo, (1.0 - h[k]) / dk)
            t_hi = min(t_hi, -h[k] / dk)
    if not (np.isfinite(t_lo) and np.isfinite(t_hi)):
        return None
    if t_lo > -t_hi + 1e-12 or t_hi < t_lo + 1e-14:
        return None
    return (t_lo, t_hi)

def propose_feasible(h, n, sigma, rng):
    d = make_direction(n, rng)
    interval = feasible_interval(h, d)
    if interval is None:
        return None
    t_lo, t_hi = interval
    t = rng.normal(0.0, sigma)
    if t < t_lo or t > t_hi:
        return None
    trial = h + t * d
    if check_feasible(trial, n):
        if np.max(np.abs(trial - h)) > 1e-12:
            return trial
        return None
    if np.all(trial >= -1e-10) and np.all(trial <= 1.0 + 1e-10):
        repaired = project_box_sum(trial, 0.5 * n)
        if repaired is not None and check_feasible(repaired, n):
            if np.max(np.abs(repaired - h)) > 1e-12 and np.max(np.abs(repaired - trial)) < 1e-8:
                return repaired
    return None

        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(float(np.sum(h)) - 0.5 * n) <= 1e-8 * max(1, n)

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    try:
        c = np.correlate(h, 1.0 - h, mode='full')
        if not np.all(np.isfinite(c)) or len(c) != 2 * n - 1:
            return float('inf')
        return float(np.max(c) * (2.0 / n))
    except Exception:
        return float('inf')

def make_direction(n, rng):
    if n <= 64:
        sizes = [2, 4, 8, n]
    elif n <= 256:
        sizes = [2, 8, 16, n]
    else:
        sizes = [2, 8, 32, n]
    sizes = [s for s in sizes if s <= n]
    for _ in range(20):
        k = int(sizes[rng.integers(0, len(sizes))])

        g = rng.standard_normal(k)
        g = g - np.mean(g)
        g[-1] = -math.fsum(g[:-1].tolist())
        amax = float(np.max(np.abs(g)))
        if amax > 1e-12:

            d[idx] = g / amax
            return d
    idx = rng.permutation(n)[:2]
    d = np.zeros(n)
    d[idx[0]] = 1.0
    d[idx[1]] = -1.0
    return d

def feasible_interval(h, d):
    t_lo = -1e30
    t_hi = 1e30
    for k in range(len(h)):
        dk = d[k]
        if abs(dk) < 1e-15:

        if dk > 0:

        else:
            t_lo = max(t_lo, (1.0 - h[k]) / dk)
            t_hi = min(t_hi, -h[k] / dk)
    if not (np.isfinite(t_lo) and np.isfinite(t_hi)):
        return None
    if t_hi - t_lo < 1e-14:
        return None
    return (t_lo, t_hi)

def propose_feasible(h, n, sigma, rng):
    d = make_direction(n, rng)
    interval = feasible_interval(h, d)
    if interval is None:
        return None
    t_lo, t_hi = interval
    t = rng.normal(0.0, sigma)
    if t < t_lo or t > t_hi:
        return None
    trial = h + t * d
    if check_feasible(trial, n):
        if float(np.max(np.abs(trial - h))) > 1e-12:

        return None

                return repaired
    return None

def calibrate(start, n, rng, sigma0):
    ups = []
    best_c = exact_score(start, n)
    best_v = start.copy()
    for _ in range(24):
        trial = propose_feasible(start, n, sigma0, rng)
        if trial is None:
            continue
        c = exact_score(trial, n)
        if not np.isfinite(c):
            continue
        if c < best_c:
            best_c = c
            best_v = trial.copy()
        delta = c - best_c
        if delta > 0:
            ups.append(delta)
    if ups:
        m = float(np.median(ups))
        t0 = max(2e-5, min(2e-2, -m / math.log(0.8)))
    else:
        t0 = 1e-3
    return t0, best_v, best_c

def anneal_cycle(start, n, rng, n_levels, n_prop, sigma0):
    cur = start.copy()
    cur_sc = exact_score(cur, n)
    best_v = cur.copy()
    best_sc = cur_sc
    t0, cb_v, cb_sc = calibrate(cur, n, rng, sigma0)
    if cb_sc < best_sc:
        best_v = cb_v.copy()
        best_sc = cb_sc
    T = t0
    sigma = sigma0
    for lev in range(n_levels):
        T = max(1e-7, t0 * (0.55 ** lev))
        for pr in range(n_prop):
            trial = propose_feasible(cur, n, sigma, rng)
            if trial is None:
                continue
            c = exact_score(trial, n)
            if not np.isfinite(c):
                continue
            if c < best_sc:
                best_sc = c
                best_v = trial.copy()
            delta = c - cur_sc
            accept = False
            if delta <= 0:
                accept = True
            else:
                if delta / T < 745:
                    accept = rng.random() < math.exp(-delta / T)
            if accept:
                cur = trial.copy()
                cur_sc = c
        if (pr + 1) % 50 < n_prop:
            pass
        sigma = max(0.003, min(0.25, sigma * 0.75))
        rp = project_box_sum(cur, 0.5 * n)
        if rp is not None and check_feasible(rp, n) and float(np.max(np.abs(rp - cur))) < 1e-8:
            cur = rp
            cur_sc = exact_score(cur, n)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_v = cur.copy()
    return best_v, best_sc

def get_parent():
    if 'initial_h_values' in globals():
        ph = np.asarray(globals()['initial_h_values'], dtype=np.float64)
        if ph.ndim == 1 and 2 <= len(ph) <= 999 and np.all(np.isfinite(ph)):
            if np.all(ph >= -1e-6) and np.all(ph <= 1.0 + 1e-6):
                n = len(ph)
                if abs(float(np.sum(ph)) - 0.5 * n) <= 1e-6 * max(1, n):
                    p = project_box_sum(np.clip(ph, 0.0, 1.0), 0.5 * n)
                    if p is not None and check_feasible(p, n):
                        return p.copy(), n
    n = 45
    return np.full(n, 0.5), n

def build_starts(parent, n, rng):
    starts = []
    def add(v):
        p = project_box_sum(v, 0.5 * n)
        if p is None or not check_feasible(p, n):
            return
        for s in starts:
            if float(np.max(np.abs(p - s))) < 1e-10:
                return
        starts.append(p.copy())
    add(parent)
    add(1.0 - parent)
    add(parent[::-1])
    add(np.roll(parent, 1))
    add(parent + 0.05 * rng.standard_normal(n))
    add(parent + 0.15 * rng.standard_normal(n))
    add(rng.random(n))
    i = np.arange(n, dtype=np.float64)
    phi = rng.random() * 2.0 * np.pi
    theta = rng.random() * 2.0 * np.pi
    add(0.5 + 0.28 * np.cos(2.0 * np.pi * i / n + phi) + 0.12 * np.sin(4.0 * np.pi * i / n + theta))
    add(0.5 + 0.35 * np.sin(2.0 * np.pi * 2.0 * i / n + phi))
    add(parent + 0.08 * rng.standard_normal(n))
    return starts

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent, n = get_parent()
    fb_h = parent.copy()
    fb_sc = exact_score(fb_h, n)
    best_h = fb_h.copy()
    best_sc = fb_sc
    starts = build_starts(parent, n, rng)
    scored = []
    for s in starts:
        sc = exact_score(s, n)
        if np.isfinite(sc):
            scored.append((sc, s.copy()))
    scored.sort(key=lambda x: x[0])
    if not scored:
        scored = [(best_sc, best_h.copy())]
    if scored[0][0] < best_sc:
        best_sc = scored[0][0]
        best_h = scored[0][1].copy()
    if n <= 64:
        n_cycles, n_levels, n_prop, sigma0 = 4, 6, 240, 0.08
        restart_sds = [0.10, 0.25]
    elif n <= 256:
        n_cycles, n_levels, n_prop, sigma0 = 3, 5, 90, 0.06
        restart_sds = [0.18]
    else:
        n_cycles, n_levels, n_prop, sigma0 = 2, 4, 24, 0.04
        restart_sds = [0.12]
    for cy in range(n_cycles):
        try:
            if cy == 0:
                start_v = scored[0][1].copy()
            elif cy == 1 and len(scored) > 1:
                start_v = scored[1][1].copy()
            else:
                sd = restart_sds[(cy - 2) % len(restart_sds)] if cy >= 2 else restart_sds[0]
                start_v = None
                for _ in range(5):
                    cand = project_box_sum(best_h + sd * rng.standard_normal(n), 0.5 * n)
                    if cand is not None and check_feasible(cand, n):
                        start_v = cand
                        break
                if start_v is None:
                    start_v = best_h.copy()
            sc = exact_score(start_v, n)
            if sc < best_sc:
                best_sc = sc
                best_h = start_v.copy()
            lv, lsc = anneal_cycle(start_v, n, rng, n_levels, n_prop, sigma0)
            if lsc < best_sc:
                best_h = lv.copy()
                best_sc = lsc
        except Exception:
            pass
    fp = project_box_sum(best_h, 0.5 * n)
    if fp is not None and check_feasible(fp, n):
        best_h = fp
        best_sc = exact_score(best_h, n)
    final_sc = exact_score(best_h, n)
    if not np.isfinite(final_sc):
        fp2 = project_box_sum(fb_h, 0.5 * n)
        if fp2 is not None and check_feasible(fp2, n):
            best_h = fp2
        else:
            best_h = np.full(n, 0.5)
        final_sc = exact_score(best_h, n)
    if not np.isfinite(final_sc):
        best_h = np.full(45, 0.5)
        n = 45
        final_sc = exact_score(best_h, n)
    return (best_h, final_sc, n)
