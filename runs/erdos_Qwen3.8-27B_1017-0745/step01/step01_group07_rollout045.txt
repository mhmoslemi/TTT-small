        if n < 8 or n > 999:
            n = 80
    S = n / 2.0
    Dcap = 40 if n <= 128 else (32 if n <= 256 else (20 if n <= 512 else 12))
    D = min(n - 1, Dcap)
    B = build_dct_basis(n, D)
    if B is None:
        B = np.eye(n)[:, :D]
        B -= B.mean(axis=0, keepdims=True)
    h_const = np.full(n, 0.5)
    best_score = exact_score(h_const, n)
    if best_score is None:
        best_score = 0.5
    best_h = h_const.copy()
    if parent is not None:
        ps = exact_score(parent, n)
        if ps is not None and ps < best_score:
            best_score = ps; best_h = parent.copy()
    anchors = []
    if parent is not None:
        anchors.append(parent.copy())
        pp = project_capped_simplex(parent + rng.normal(0, 0.08, n), S)
        if pp is not None and is_feasible(pp, n): anchors.append(pp)
        anchors.append(parent[::-1].copy())
        cp = project_capped_simplex(1.0 - parent, S)
        if cp is not None and is_feasible(cp, n): anchors.append(cp)
    else:
        anchors.append(h_const.copy())
        z0 = rng.normal(0, 0.12, D)
        a1 = project_capped_simplex(h_const + B @ z0, S)
        if a1 is not None and is_feasible(a1, n): anchors.append(a1)
        perm = rng.permutation(n)
        bv = np.zeros(n); bv[perm[:n // 2]] = 1.0
        if n % 2: bv[perm[n // 2]] = 0.5
        anchors.append(bv)
    scored = []
    for a in anchors:
        a2 = np.asarray(a, dtype=np.float64)
        if not is_feasible(a2, n): continue
        if all(np.max(np.abs(a2 - s)) >= 1e-9 for s in scored):
            scored.append(a2)
    max_a = 4 if n <= 128 else (3 if n <= 256 else (2 if n <= 512 else 1))
    scored = scored[:max_a]
    gen_cap = 80 if n <= 128 else (45 if n <= 256 else (18 if n <= 512 else 6))
    for a in scored:
        sc = exact_score(a, n)
        if sc is not None and sc < best_score:
            best_score = sc; best_h = a.copy()
        bh, bs = cma_run(a, B, S, n, D, gen_cap, rng, best_score, best_h)
        best_h, best_score = bh, bs
    nr = 2 if n <= 128 else (1 if n <= 256 else 0)
    sf = 1.0
    sb = np.clip(0.10 * np.sqrt(n / D), 0.05, 0.70)
    for _ in range(nr):
        pre = best_score
        bh, bs = cma_run(best_h.copy(), B, S, n, D, gen_cap, rng, best_score, best_h, np.clip(sb * sf, 0.03, 1.5))
        best_h, best_score = bh, bs
        sf = max(0.5, 0.65 * sf) if best_score < pre - 1e-12 else min(6, 1.6 * sf)
    fh = best_h
    fs = exact_score(fh, n)
    if not is_feasible(fh, n) or fs is None:
        if parent is not None:
            fh = parent.copy(); fs = exact_score(fh, n)
        if not is_feasible(fh, n) or fs is None:
            fh = h_const.copy(); fs = exact_score(fh, n)
            if fs is None: fs = 0.5
    return fh, fs, n
