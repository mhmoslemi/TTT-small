        return np.full(n, 1.0 / n)

    diff = 1.0 - p.sum()
    if abs(diff) > 1e-15 and p.sum() > 0:

        p[idx] += diff
    return p

import numpy as np

def project_capped(y, n):
    if not np.all(np.isfinite(y)):
        return np.full(n, 0.5), False
    S = n / 2.0
    lo = float(y.min()) - 1.0
    hi = float(y.max())
    lam = float(np.clip(np.mean(y) - 0.5, lo, hi))
    for _ in range(80):
        c = np.clip(y - lam, 0.0, 1.0)
        r = c.sum() - S
        if abs(r) < 1e-14:
            break
        if r > 0:
            lo = lam
        else:
            hi = lam
        a = int(np.sum((y - lam > 0) & (y - lam < 1)))
        if a > 0:
            new_lam = lam + r / a
        else:
            new_lam = 0.5 * (lo + hi)
        if new_lam <= lo + 1e-16 or new_lam >= hi - 1e-16:
            new_lam = 0.5 * (lo + hi)
        lam = new_lam
    c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(5):
        diff = S - c.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = c < 1.0 - 1e-12
        else:
            mask = c > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        c[mask] += diff / cnt
        np.clip(c, 0.0, 1.0, out=c)
    ok = (np.all(c >= -1e-12) and np.all(c <= 1.0 + 1e-12) and abs(c.sum() - S) < 1e-10)
    return c, ok

def exact_score(h, n):
    if not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(h.sum() - n / 2.0) <= 1e-10)

def lag_scores(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def build_G(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        lo = max(0, k)
        hi = min(n, n + k)
        row = k + n - 1
        if hi > lo:
            G[row, lo:hi] += dx * (1.0 - h[lo - k:hi - k])
            G[row, lo - k:hi - k] -= dx * h[lo:hi]
    return G

def project_simplex(z):
    m = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(m, 1.0 / m)
    s = np.sort(z)[::-1]
    cs = np.cumsum(s)
    idx = np.arange(1, m + 1, dtype=np.float64)
    valid = s - (cs - 1.0) / idx > 0
    if not valid.any():
        return np.full(m, 1.0 / m)
    r = int(np.where(valid)[0][-1])
    theta = (cs[r] - 1.0) / (r + 1)
    p = np.maximum(z - theta, 0.0)
    diff = 1.0 - p.sum()
    if abs(diff) > 1e-14 and p.sum() > 0:
        p[int(np.argmax(p))] += diff
    return p

def dual_state(p, h, q, G, n, L):
    gmix = G.T @ p
    y, _ = project_capped(h - gmix / L, n)
    d = y - h
    u = q + G @ d
    D = float(p @ u + 0.5 * L * d @ d)
    gap = float(u.max() - p @ u)
    return d, u, D, gap

def solve_majorant(h, q, G, n, max_inner):
    L = 4.0 / n
    n_lags = len(q)
    Ldual = max(1e-12, 1.2 * float(np.sum(G * G)) / L)
    qmax = float(q.max())
    tol = max(1e-12, 1e-8 * max(1.0, abs(qmax)))
    mask = q >= qmax - tol
    if not mask.any():
        mask = np.zeros(n_lags, dtype=bool)
        mask[int(np.argmax(q))] = True
    p = np.zeros(n_lags)
    p[mask] = 1.0 / float(mask.sum())
    best_d = None
    for _ in range(max_inner):
        d, u, D, gap = dual_state(p, h, q, G, n, L)
        if not np.all(np.isfinite(D)):
            break
        if best_d is None or D > float(np.dot(p, u) + 0.5 * L * d @ d):
            best_d = d.copy()
        if gap <= max(1e-10, 1e-8 * max(1.0, abs(float(u.max())))):
            break
        step = 1.0 / Ldual
        x = project_simplex(p + u * step)
        d2, u2, D2, _ = dual_state(x, h, q, G, n, L)
        if D2 < D - 1e-13:
            x = project_simplex(p + u * step * 0.5)
            d2, u2, D2, _ = dual_state(x, h, q, G, n, L)
        if np.max(np.abs(x - p)) < 1e-13:
            p = x
            break
        p = x
    if best_d is None:
        d, u, D, gap = dual_state(p, h, q, G, n, L)
        best_d = d
    return best_d

def optimize_start(h0, n, max_outer, max_inner):

            break
        v = G @ d
        norm2 = float(d @ d)

        accepted = False

            ht, ok = project_capped(h + alpha * d, n)

                best_s = st
                best_h = ht.copy()

    return best_h, best_s

def coord_polish(h, n, max_rounds):
    h = h.copy()
    s = exact_score(h, n)
    S = n / 2.0
    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                if h[j] < 1e-8 or h[i] > 1.0 - 1e-8:
                    continue
                if h[i] < 1e-8 or h[j] > 1.0 - 1e-8:
                    continue
                for eps in [0.02, 0.01, 0.005]:
                    delta = min(eps, h[i] - 1e-9, 1.0 - 1e-9 - h[j])
                    if delta <= 0:
                        continue
                    ht = h.copy()
                    ht[i] -= delta
                    ht[j] += delta
                    if not check_feasible(ht, n):
                        continue
                    st = exact_score(ht, n)
                    if st < s - 1e-13:
                        s = st
                        h = ht
                        improved = True
                        break
        if not improved:
            break
    return h, s
