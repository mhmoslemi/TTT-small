import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-14:
        if resid > 0:
            order = np.argsort(1.0 - y)
            for i in order:
                add = min(resid, 1.0 - y[i])
                y[i] += add
                resid -= add
                if abs(resid) < 1e-15:
                    break
        else:
            order = np.argsort(y)
            for i in order:
                sub = min(-resid, y[i])
                y[i] -= sub
                resid += sub
                if abs(resid) < 1e-15:
                    break
    if not np.all(np.isfinite(y)):
        return None
    if np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    if abs(np.sum(y) - S) > 1e-10 * max(1.0, float(n)):
        return None
    y = np.clip(y, 0.0, 1.0)
    return y

def exact_score(h, n):
    if not np.all(np.isfinite(h)):
        return np.inf
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return np.inf
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(corr)) * (2.0 / n)
    return val if np.isfinite(val) else np.inf

def check_feasible(h, n):
    if h is None or h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 5e-10 * max(1.0, float(n)):
        return False
    return True

def line_minimax(x, y, n, S):
    if not check_feasible(x, n) or not check_feasible(y, n):
        return x, exact_score(x, n)
    d = y - x
    n_lags = 2 * n - 1
    A = np.zeros(n_lags)
    B = np.zeros(n_lags)
    D = np.zeros(n_lags)
    for k in range(-(n - 1), n):
        idx = k + n - 1
        i_lo = max(0, k)
        i_hi = min(n - 1, n - 1 + k)
        if i_lo > i_hi:
            continue
        i = np.arange(i_lo, i_hi + 1)
        j = i - k
        A[idx] = -np.sum(d[i] * d[j])
        B[idx] = np.sum(d[i] * (1.0 - x[j]) - x[i] * d[j])
        D[idx] = np.sum(x[i] * (1.0 - x[j]))
    alphas = [0.0, 1.0]
    for idx in range(n_lags):
        a2 = A[idx]
        if abs(a2) > 1e-14:
            sp = -B[idx] / (2.0 * a2)
            if -1e-10 <= sp <= 1.0 + 1e-10:
                alphas.append(np.clip(sp, 0.0, 1.0))
    for a_idx in range(n_lags):
        for b_idx in range(a_idx + 1, n_lags):
            da = A[a_idx] - A[b_idx]
            db = B[a_idx] - B[b_idx]
            dd = D[a_idx] - D[b_idx]
            if abs(da) < 1e-14 and abs(db) < 1e-14:
                continue
            if abs(da) < 1e-14:
                r = -dd / db
            else:
                disc = db * db - 4.0 * da * dd
                if disc < -1e-10 * (abs(db * db) + abs(4.0 * da * dd) + 1e-30):
                    continue
                disc = max(0.0, disc)
                sq = np.sqrt(disc)
                r1 = (-db + sq) / (2.0 * da)
                r2 = (-db - sq) / (2.0 * da)
                if -1e-10 <= r1 <= 1.0 + 1e-10:
                    alphas.append(np.clip(r1, 0.0, 1.0))
                if -1e-10 <= r2 <= 1.0 + 1e-10:
                    alphas.append(np.clip(r2, 0.0, 1.0))
    alphas = np.array(alphas, dtype=np.float64)
    alphas = alphas[(alphas >= 0.0) & (alphas <= 1.0)]
    if len(alphas) == 0:
        alphas = np.array([0.0, 1.0])
    alphas = np.unique(np.round(alphas, 12))
    best_score = np.inf
    best_a = 0.0
    for a_val in alphas:
        pt = x + a_val * d
        sc = np.max(A * a_val * a_val + B * a_val + D)
        if sc < best_score:
            best_score = sc
            best_a = a_val
    pt = x + best_a * d
    pt = project_box_sum(pt, S)
    if pt is None or not check_feasible(pt, n):
        pt = x
    sc = exact_score(pt, n)
    sc0 = exact_score(x, n)
    sc1 = exact_score(y, n)
    if sc1 < sc and sc1 < sc0:
        pt, sc = y, sc1
    return pt, sc

def prod_idx(i, j, nn):
    if i > j:
        i, j = j, i
    return i * nn - i * (i - 1) // 2 + (j - i)

def solve_trust_lp(x, r, n, S):
    P = n * (n + 1) // 2
    nvar = n + P + 1
    lo_v = np.maximum(0.0, x - r)
    up_v = np.minimum(1.0, x + r)
    rows_ub, cols_ub, vals_ub, rhs_ub = [], [], [], []
    row = 0
    pairs = []
    for i in range(n):
        for j in range(i, n):
            pi = prod_idx(i, j, n)
            zi = n + pi
            li, ui = lo_v[i], up_v[i]
            lj, uj = lo_v[j], up_v[j]
            if i == j:
                li2, ui2 = li, ui
            else:
                li2, ui2 = lj, uj
# McCormick: z >= li*vj + lj*vi - li*lj  =>  -li*vi -lj*vj + z <= -li*lj
            rows_ub.append(row); cols_ub.append(i); vals_ub.append(-li); rhs_ub.append(-li*li2)
            rows_ub.append(row); cols_ub.append(j); vals_ub.append(-lj); rhs_ub.append(-li*li2)
            rows_ub.append(row); cols_ub.append(zi); vals_ub.append(1.0); rhs_ub.append(-li*li2)
            row += 1
# z >= ui*vj + uj*vi - ui*uj  =>  -ui*vi -uj*vj + z <= -ui*uj
            rows_ub.append(row); cols_ub.append(i); vals_ub.append(-ui); rhs_ub.append(-ui*ui2)
            rows_ub.append(row); cols_ub.append(j); vals_ub.append(-uj); rhs_ub.append(-ui*ui2)
            rows_ub.append(row); cols_ub.append(zi); vals_ub.append(1.0); rhs_ub.append(-ui*ui2)
            row += 1
# z <= ui*vj + lj*vi - ui*lj  =>  -ui*vi -lj*vj - z <= -ui*lj
            rows_ub.append(row); cols_ub.append(i); vals_ub.append(-ui); rhs_ub.append(-ui*li2)
            rows_ub.append(row); cols_ub.append(j); vals_ub.append(-lj); rhs_ub.append(-ui*li2)
            rows_ub.append(row); cols_ub.append(zi); vals_ub.append(-1.0); rhs_ub.append(-ui*li2)
            row += 1
# z <= li*vj + uj*vi - li*uj  =>  -li*vi -uj*vj - z <= -li*uj
            rows_ub.append(row); cols_ub.append(i); vals_ub.append(-li); rhs_ub.append(-li*ui2)
            rows_ub.append(row); cols_ub.append(j); vals_ub.append(-uj); rhs_ub.append(-li*ui2)
            rows_ub.append(row); cols_ub.append(zi); vals_ub.append(-1.0); rhs_ub.append(-li*ui2)
            row += 1
            pairs.append((i, j, zi))
    t_idx = n + P
    for k in range(-(n - 1), n):
        i_lo = max(0, k)
        i_hi = min(n - 1, n - 1 + k)
        if i_lo > i_hi:
            continue
        rows_cur, cols_cur, vals_cur = [], [], []
        for ii in range(i_lo, i_hi + 1):
            jj = ii - k
            rows_cur.append(row); cols_cur.append(ii); vals_cur.append(1.0)
            pi = prod_idx(ii, jj, n)
            rows_cur.append(row); cols_cur.append(n + pi); vals_cur.append(-1.0)
        rows_cur.append(row); cols_cur.append(t_idx); vals_cur.append(-1.0)
        rows_ub.extend(rows_cur); cols_ub.extend(cols_cur); vals_ub.extend(vals_cur)
        rhs_ub.append(0.0)
        row += 1
    A_ub = coo_matrix((vals_ub, (rows_ub, cols_ub)), shape=(row, nvar)).tocsr()
    b_ub = np.array(rhs_ub, dtype=np.float64)
    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S], dtype=np.float64)
    c = np.zeros(nvar)
    c[t_idx] = 1.0
    lbs = np.zeros(nvar)
    ubs = np.zeros(nvar)
    lbs[:n] = lo_v
    ubs[:n] = up_v
    for i in range(n):
        for j in range(i, n):
            pi = prod_idx(i, j, n)
            zl = lo_v[i] * lo_v[j]
            zu = up_v[i] * up_v[j]
            lbs[n + pi] = max(0.0, zl)
            ubs[n + pi] = min(1.0, zu)
    ubs[t_idx] = float(n)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=list(zip(lbs, ubs)), method='highs',
                      options={'presolve': True})
        if res.status == 0 and res.x is not None and np.all(np.isfinite(res.x)):
            return res.x[:n]
    except Exception:
        pass
    return None

def local_lp_search(start, n, S):

    if h0 is None or not check_feasible(h0, n):
        return None
    current = h0.copy()
    cur_score = exact_score(current, n)
    r = 0.20
    fails = 0
    max_attempts = 8 if n <= 64 else 5
    for _ in range(max_attempts):
        if r < 1e-3 and fails >= 2:
            break
        endpoint = solve_trust_lp(current, r, n, S)
        if endpoint is not None and len(endpoint) == n and np.all(np.isfinite(endpoint)):
            if np.all(endpoint >= -1e-7) and np.all(endpoint <= 1.0 + 1e-7):
                ep = np.clip(endpoint, 0.0, 1.0)

                            cur_score = sc

    return current
