def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if resid > 1e-13:

            y[i] += a

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if resid > 1e-13:
        for i in range(n):
            a = min(resid, 1.0 - y[i])
            y[i] += a
            resid -= a
            if resid < 1e-15:
                break
    elif resid < -1e-13:
        for i in range(n - 1, -1, -1):
            a = min(-resid, y[i])
            y[i] -= a
            resid += a
            if -resid < 1e-15:
                break
    y = np.clip(y, 0.0, 1.0)
    if abs(np.sum(y) - S) > 1e-10 * max(1.0, float(n)):
        return None
    return y

def exact_score(h):
    n = len(h)
    if not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def feasible(h, n):

            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(np.sum(h) - n / 2.0) < 1e-10 * max(1.0, float(n)))

def lse_value(Q, tau):
    M = np.max(Q)
    ex = np.exp((Q - M) / tau)
    s = np.sum(ex)
    if not np.isfinite(s) or s <= 0:
        return None
    return M + tau * np.log(s)

def lse_weights(Q, tau):
    M = np.max(Q)
    ex = np.exp((Q - M) / tau)
    s = np.sum(ex)
    if not np.isfinite(s) or s <= 0:
        return None
    return ex / s

def compute_grad(h, n, w):
    t1 = fftconvolve(w, 1.0 - h, mode='full')
    t2 = fftconvolve(w[::-1], h, mode='full')
    return t1[n - 1:2 * n - 1] - t2[n - 1:2 * n - 1]

def harmonic_noise(n, rng):
    m1 = int(rng.integers(1, max(2, n // 2)))
    m2 = int(rng.integers(1, max(2, n // 2)))
    ph1 = float(rng.uniform(0, 2 * np.pi))
    ph2 = float(rng.uniform(0, 2 * np.pi))
    x = np.arange(n, dtype=np.float64)
    z = np.cos(2 * np.pi * m1 * x / n + ph1) + np.cos(2 * np.pi * m2 * x / n + ph2)
    z -= np.mean(z)
    rms = np.sqrt(np.mean(z ** 2))
    if rms < 1e-10:
        z = rng.standard_normal(n)
        z -= np.mean(z)
        rms = np.sqrt(np.mean(z ** 2))
    return z / rms
