import numpy as np
from scipy.optimize import minimize

_mm_state = {}

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    S = max(0.0, min(float(n), S))
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(v - tau, 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-14:
        for i in np.argsort(1.0 - y):
            d = min(r, 1.0 - y[i])
            y[i] += d
            r -= d
            if r <= 1e-14:
                break
    elif r < -1e-14:
        for i in np.argsort(y):
            d = min(-r, y[i])
            y[i] -= d
            r += d
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-9) or np.any(y > 1.0 + 1e-9):
        return None
    if abs(float(np.sum(y)) - S) > 1e-10 * max(1, n):
        return None
    return y

def check_feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(float(np.sum(h)) - 0.5 * n) < 1e-10 * max(1, n))

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    try:
        corr = np.correlate(h, 1.0 - h, mode='full')
        if len(corr) != 2 * n - 1 or not np.all(np.isfinite(corr)):
            return float('inf')
        s = float(np.max(corr) * (2.0 / n))
        return s if np.isfinite(s) else float('inf')
    except Exception:
        return float('inf')

def lag_vg(h, s, n):
    dx = 2.0 / n
    lo = max(0, -s)
    hi = min(n, n - s)
    cnt = hi - lo
    if cnt <= 0:
        return 0.0, np.zeros(n)
    ii = np.arange(lo, hi)
    jj = ii + s
    q = dx * float(np.sum(h[ii] * (1.0 - h[jj])))
    g = np.zeros(n)
    g[ii] += (1.0 - h[jj])
    g[jj] -= h[ii]
    g *= dx
    return q, g

def all_lags_val_grad(h, n):
    nv = 2 * n - 1
    vals = np.zeros(nv)
    grads = np.zeros((nv, n))
    for k, s in enumerate(range(-(n - 1), n)):
        v, g = lag_vg(h, s, n)
        vals[k] = v
        grads[k] = g
    return vals, grads

def sep_envelope(x, d, n):
    best = -1e18
    best_k = 0
    dx = 2.0 / n
    for k, s in enumerate(range(-(n - 1), n)):
        lo = max(0, -s)
        hi = min(n, n - s)
        ii = np.arange(lo, hi)
        jj = ii + s
        ell = float(np.sum(x[ii] * (1.0 - x[jj]) + (1.0 - x[jj]) * d[ii] - x[ii] * d[jj])) * dx
        if ell > best:
            best = ell
            best_k = k
    return best, best_k

def mm_obj(z):
    x = _mm_state['x']
    L = _mm_state['L']
    d = z[:-1] - x
    return 0.5 * L * float(np.dot(d, d)) + z[-1]

def mm_obj_g(z):
    x = _mm_state['x']
    L = _mm_state['L']
    n = len(z) - 1
    g = np.zeros(n + 1)
    g[:n] = L * (z[:n] - x)
    g[n] = 1.0
    return g

def mm_ineq(z):
    qJ = _mm_state['qJ']
    GJ = _mm_state['GJ']
    x = _mm_state['x']
    d = z[:len(x)] - x
    return z[-1] - qJ - GJ @ d

def mm_ineq_j(z):
    GJ = _mm_state['GJ']
    n = GJ.shape[1]
    J = np.zeros((GJ.shape[0], n + 1))
    J[:, :n] = -GJ
    J[:, n] = 1.0
    return J

def mm_eq(z):
    n = len(z) - 1
    return float(np.sum(z[:n])) - 0.5 * n

def mm_eq_j(z):
    n = len(z) - 1
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def solve_mm(x, n, active_idx=None):
    dx = 2.0 / n
    L = 2.0 * dx
    nv = 2 * n - 1
    vals, grads = all_lags_val_grad(x, n)
    if active_idx is None:
        active_idx = np.arange(nv)
    qJ = vals[active_idx]
    GJ = grads[active_idx]
    z0 = np.append(x.copy(), float(np.max(qJ)))
    _mm_state['x'] = x.copy()
    _mm_state['L'] = L
    _mm_state['qJ'] = qJ.copy()
    _mm_state['GJ'] = GJ.copy()
    bounds = [(0.0, 1.0)] * n + [(-1.0, 1.0)]
    cons = [{'type': 'ineq', 'fun': mm_ineq, 'jac': mm_ineq_j},
            {'type': 'eq', 'fun': mm_eq, 'jac': mm_eq_j}]
    try:
        res = minimize(mm_obj, z0, jac=mm_obj_g, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 100, 'ftol': 1e-10, 'disp': False})
        y_raw = np.asarray(res.x[:n], dtype=np.float64)
        if not np.all(np.isfinite(y_raw)):
            return x, False
        y = project_box_sum(y_raw, 0.5 * n)
        if y is None:
            return x, False
        d = y - x
        EJ = float(np.max(qJ + GJ @ d))
        full_val, full_k = sep_envelope(x, d, n)
        complete = full_val <= EJ + 1e-9 * max(1.0, abs(EJ))
        model_val = 0.5 * L * float(np.dot(d, d)) + EJ
        if model_val > float(np.max(qJ)) + 1e-7 and not complete:
            return x, False
        return y, complete
    except Exception:
        return x, False

def mm_descent(h0, n, max_steps=30):
    h = h0.copy()
    sc = exact_score(h, n)
    active = None
    for _ in range(max_steps):
        prop, complete = solve_mm(h, n, active)
        if not np.all(np.isfinite(prop)):
            break
        if float(np.max(np.abs(prop - h))) < 1e-11:
            break
        improved = False
        for k in range(9):
            alpha = 2.0 ** (-k)
            cand = project_box_sum(h + alpha * (prop - h), 0.5 * n)
            if cand is None:
                continue
            c = exact_score(cand, n)
            if c < sc - 1e-12 * max(1.0, abs(sc)):
                h = cand.copy()
                sc = c
                improved = True
                break
        if not improved:
            break
        if sc <= 1e-15:
            break
    return h, sc
