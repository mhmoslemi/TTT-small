import numpy as np

def project_capped_simplex(y, n, S):
    if y is None or not np.all(np.isfinite(y)) or len(y) != n:
        return np.full(n, S / n)
    y = np.asarray(y, dtype=np.float64).copy()
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    e = S - h.sum()
    if abs(e) > 1e-12:
        for j in range(n):
            if e > 0:
                add = min(e, 1.0 - h[j])
                h[j] += add
                e -= add
            else:
                sub = min(-e, h[j])
                h[j] -= sub
                e += sub
            if abs(e) < 1e-12:
                break
    if abs(h.sum() - S) > 1e-10:
        return np.full(n, S / n)
    return h

def feasible_check(h, n, S):

            np.all(np.isfinite(h)) and h.min() >= -1e-10 and
            h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

    if not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def build_factors(n, dx):
    gamma = 1.25 * dx
    Lps, Lns, cs = [], [], []
    Id = np.eye(n)
    for d in range(n):
        B = np.zeros((n, n), dtype=np.float64)
        if d == 0:
            B = dx * np.eye(n)
            c = np.full(n, dx)
        else:
            idx = np.arange(n - d)
            B[idx, idx + d] = dx / 2.0
            B[idx + d, idx] = dx / 2.0
            cp_v = np.zeros(n); cp_v[d:] = dx
            cm_v = np.zeros(n); cm_v[:n-d] = dx
            c = [cp_v, cm_v]
        P = 0.5 * (gamma * Id + B)
        N = 0.5 * (gamma * Id - B)
        try:
            Lp = np.linalg.cholesky(P)
            Ln = np.linalg.cholesky(N)
        except np.linalg.LinAlgError:
            return None
        if not np.all(np.isfinite(Lp)) or not np.all(np.isfinite(Ln)):
            return None
        Lps.append(Lp)
        Lns.append(Ln)
        cs.append(c)
    return Lps, Lns, cs

def mm_solve(cp, n, S, rho, factors, x, solver=None):
    Lps, Lns, cs = factors
    y = cp.Variable(n)
    t = cp.Variable()
    xpar = cp.Parameter(n)
    vpars = [cp.Parameter(n) for _ in range(n)]
    apars = [cp.Parameter() for _ in range(n)]
    cons = [y >= 0, y <= 1, cp.sum(y) == S]
    for d in range(n):
        quad_part = cp.sum_squares(Lns[d].T @ y) - 2.0 * (vpars[d] @ y) + apars[d]
        if d == 0:
            cons.append(cs[d] @ y + quad_part <= t)
        else:
            cons.append(cs[d][0] @ y + quad_part <= t)
            cons.append(cs[d][1] @ y + quad_part <= t)
    obj = cp.Minimize(t + (rho / 2.0) * cp.sum_squares(y - xpar))
    prob = cp.Problem(obj, cons)
    xpar.value = x
    for d in range(n):
        v = Lps[d] @ (Lps[d].T @ x)
        vpars[d].value = v
        apars[d].value = float(x @ v)
    y.value = x
    t.value = score_exact(x, n) + 1e-8
    try:
        prob.solve(solver=solver, warm_start=True, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if y.value is None or t.value is None:
        return None
    if not np.all(np.isfinite(y.value)) or not np.isfinite(t.value):
        return None
    if y.value.min() < -1e-5 or y.value.max() > 1.0 + 1e-5:
        return None
    if abs(y.value.sum() - S) > 1e-5:
        return None
    return np.asarray(y.value, dtype=np.float64).copy()

                x = yc
                sc = sc_new
                no_improve = 0
            else:
                no_improve += 1
                if no_improve >= 2:
                    break
    return x, sc

import numpy as np

def project_capped_simplex(y, n, S):
    if y is None or not np.all(np.isfinite(y)) or len(y) != n:
        return np.full(n, S / n)
    y = np.asarray(y, dtype=np.float64).copy()
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    e = S - h.sum()
    if abs(e) > 1e-12:
        for j in range(n):
            if abs(e) < 1e-13:
                break
            if e > 0:
                a = min(e, 1.0 - h[j])
                h[j] += a
                e -= a
            else:
                a = min(-e, h[j])
                h[j] -= a
                e += a
    if abs(h.sum() - S) > 1e-10:
        return np.full(n, S / n)
    return h

def feasible_check(h, n, S):
    return (isinstance(h, np.ndarray) and h.shape == (n,) and
            np.all(np.isfinite(h)) and h.min() >= -1e-10 and
            h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def score_exact(h, n):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def build_factors(n, dx):
    gamma = 1.25 * dx
    Id = np.eye(n, dtype=np.float64)
    Lps, Lns, cs = [], [], []
    for d in range(n):
        B = np.zeros((n, n), dtype=np.float64)
        if d == 0:
            B = dx * np.eye(n)
            c = [np.full(n, dx)]
        else:
            idx = np.arange(n - d)
            B[idx, idx + d] = dx / 2.0
            B[idx + d, idx] = dx / 2.0
            cpv = np.zeros(n); cpv[d:] = dx
            cmv = np.zeros(n); cmv[:n - d] = dx
            c = [cpv, cmv]
        P = 0.5 * (gamma * Id + B)
        N = 0.5 * (gamma * Id - B)
        try:
            Lp = np.linalg.cholesky(P)
            Ln = np.linalg.cholesky(N)
        except np.linalg.LinAlgError:
            return None
        if not (np.all(np.isfinite(Lp)) and np.all(np.isfinite(Ln))):
            return None
        Lps.append(Lp)
        Lns.append(Ln)
        cs.append(c)
    return Lps, Lns, cs

def mm_solve(cp, sol_mod, n, S, rho, factors, x):
    Lps, Lns, cs = factors
    y = cp.Variable(n)
    t = cp.Variable()
    xpar = cp.Parameter(n)
    vpars = [cp.Parameter(n) for _ in range(n)]
    apars = [cp.Parameter() for _ in range(n)]
    cons = [y >= 0, y <= 1, cp.sum(y) == S]
    for d in range(n):
        quad = cp.sum_squares(Lns[d].T @ y) - 2.0 * (vpars[d] @ y) + apars[d]
        for ci in range(len(cs[d])):
            cons.append(cs[d][ci] @ y + quad <= t)
    obj = cp.Minimize(t + (rho / 2.0) * cp.sum_squares(y - xpar))
    prob = cp.Problem(obj, cons)
    xpar.value = x
    for d in range(n):
        v = Lps[d] @ (Lps[d].T @ x)
        vpars[d].value = v
        apars[d].value = float(x @ v)
    y.value = x
    t.value = score_exact(x, n) + 1e-8
    try:
        prob.solve(solver=sol_mod, warm_start=True, verbose=False,
                   max_iters=200, abstol=1e-8, reltol=1e-8, feastol=1e-8)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if y.value is None or not np.all(np.isfinite(y.value)):
        return None
    if y.value.min() < -1e-4 or y.value.max() > 1.0 + 1e-4:
        return None
    if abs(y.value.sum() - S) > 1e-4:
        return None
    return np.asarray(y.value, dtype=np.float64).copy()

def mm_local(x0, n, dx, factors, solvers, rho_stages, max_iter):
    x = x0.copy()
    S = n / 2.0
    sc = score_exact(x, n)
    for rho in rho_stages:
        no_imp = 0
        for _ in range(max_iter):
            raw = None
            for slv in solvers:
                raw = mm_solve(slv[0], slv[1], n, S, rho, factors, x)
                if raw is not None:
                    break
            if raw is None:
                break
            yc = project_capped_simplex(raw, n, S)
            if not feasible_check(yc, n, S):
                break
            sc_new = score_exact(yc, n)
            if sc_new < sc - 1e-12:
                x, sc, no_imp = yc, sc_new, 0
            else:
                no_imp += 1
                if no_imp >= 2:
                    break
    return x, sc
