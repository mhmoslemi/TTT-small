    _try_add(pop, np.roll(parent, 1).copy(), S, n)
    for sd in [0.03, 0.08, 0.18, 0.35]:
        _try_add(pop, parent + sd * rng.standard_normal(n), S, n)
    for _ in range(4):
        _try_add(pop, rng.random(n), S, n)
    for m in range(1, min(5, n // 2 + 1)):
        phi, th = rng.uniform(0, 2 * np.pi), rng.uniform(0, 2 * np.pi)
        m2 = min(m + 1, n // 2)
        u = 0.5 + 0.28 * np.cos(2 * np.pi * m * idx / n + phi) + 0.12 * np.sin(2 * np.pi * m2 * idx / n + th)
        _try_add(pop, u, S, n)
    att = 0
    while len(pop) < P and att < 30 * P:
        att += 1
        if att % 2 == 0:
            _try_add(pop, parent + rng.uniform(0.03, 0.35) * rng.standard_normal(n), S, n)
        else:
            _try_add(pop, rng.random(n), S, n)
    while len(pop) < P:
        pop.append(parent.copy())
    return np.array(pop)

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent, n = _get_parent()
    S = 0.5 * n
    best_h, best_c5 = parent.copy(), exact_score(parent, n)
    if n <= 64:
        P, G, stag_lim, max_rs = 96, 300, 50, 3
    elif n <= 128:
        P, G, stag_lim, max_rs = 56, 150, 30, 2
    else:
        P, G, stag_lim, max_rs = 20, 32, 10, 1
    top_cnt = max(2, int(math.ceil(0.20 * P)))
    mu_F, mu_CR = 0.55, 0.70
    pop = _build_initial(parent, n, P, rng)
    scores = np.array([exact_score(pop[i], n) for i in range(P)])
    for i in range(P):
        if not np.isfinite(scores[i]):
            pop[i] = parent.copy(); scores[i] = best_c5
        if scores[i] < best_c5:
            best_c5 = scores[i]; best_h = pop[i].copy()
    archive, restarts, stag = [], 0, 0
    for gen in range(G):
        snap = [pop[i].copy() for i in range(P)]
        snap_sc = scores.copy()
        order = np.argsort(snap_sc)
        n_pop = [snap[i].copy() for i in range(P)]
        n_sc = snap_sc.copy()
        garch = list(archive)
        aF, aCR = [], []
        gimp = False
        for i in range(P):
            tc = [order[k] for k in range(min(top_cnt, P)) if order[k] != i]
            if not tc: tc = [order[0]]
            pb = tc[rng.integers(len(tc))]
            c2 = [j for j in range(P) if j != i and j != pb]
            r1 = c2[rng.integers(len(c2))]
            r2p = [j for j in range(P) if j != i and j != pb and j != r1]
            if garch: r2p.append(-1)
            r2c = r2p[rng.integers(len(r2p))]
            F_i = mu_F
            for _ in range(10):
                U = rng.uniform(1e-9, 1.0 - 1e-9)
                F_i = mu_F + 0.1 * math.tan(math.pi * (U - 0.5))
                if F_i > 0: break
            F_i = min(max(F_i, 0.05), 1.0)
            CR_i = min(max(float(rng.normal(mu_CR, 0.1)), 0.0), 1.0)
            x_i, x_p, x_r1 = snap[i], snap[pb], snap[r1]
            x_r2 = garch[rng.integers(len(garch))] if r2c == -1 else snap[r2c]
            donor = x_i + F_i * (x_p - x_i) + F_i * (x_r1 - x_r2)
            mask = rng.random(n) < CR_i
            mask[rng.integers(n)] = True
            trial = np.where(mask, donor, x_i)
            proj = project_box_sum(trial, S)
            if proj is None or not check_feasible(proj, n):
                continue
            sc = exact_score(proj, n)
            if not np.isfinite(sc):
                continue
            thr = snap_sc[i] - 1e-14 * max(1.0, abs(snap_sc[i]))
            if sc < thr:
                n_pop[i] = proj.copy(); n_sc[i] = sc
                archive.append(snap[i].copy())
                aF.append(F_i); aCR.append(CR_i)
                if sc < best_c5:
                    best_c5 = sc; best_h = proj.copy(); gimp = True
        for i in range(P):
            pop[i] = n_pop[i]; scores[i] = n_sc[i]
        while len(archive) > P:
            archive.pop(int(rng.integers(len(archive))))
        if aF:
            mu_F = 0.9 * mu_F + 0.1 * sum(f * f for f in aF) / sum(aF)
            mu_F = min(max(mu_F, 0.05), 1.0)
        if aCR:
            mu_CR = min(max(0.9 * mu_CR + 0.1 * float(np.mean(aCR)), 0.0), 1.0)
        stag = 0 if gimp else stag + 1
        if stag >= stag_lim:
            if restarts >= max_rs:
                break
            restarts += 1
            o2 = np.argsort(scores)
            el_n = max(2, P // 4)
            np2 = [pop[o2[0]].copy()]
            for k in range(1, el_n):
                c = pop[o2[k]].copy()
                if _unique(np2, c): np2.append(c)
            sd_sel = [0.04, 0.12, 0.30][restarts % 3]
            imm_n = int(0.7 * (P - len(np2)))
            for _ in range(imm_n):
                _try_add(np2, best_h + sd_sel * rng.standard_normal(n), S, n)
            for _ in range(P - len(np2)):
                _try_add(np2, rng.random(n), S, n)
            while len(np2) < P:
                np2.append(best_h.copy())
            for i in range(P):
                pop[i] = np2[i]; scores[i] = exact_score(pop[i], n)
                if scores[i] < best_c5:
                    best_c5 = scores[i]; best_h = pop[i].copy()
            archive = []; mu_F, mu_CR, stag = 0.55, 0.70, 0
    fin = project_box_sum(best_h, S)
    if fin is not None and check_feasible(fin, n):
        fc = exact_score(fin, n)
        if fc <= exact_score(parent, n):
            return (fin, fc, n)
    return (parent, exact_score(parent, n), n)
