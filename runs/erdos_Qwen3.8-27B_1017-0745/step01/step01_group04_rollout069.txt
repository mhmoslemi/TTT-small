def project_box_sum(z,m):
    N=len(z)
    if N==0 or not np.all(np.isfinite(z)): return np.full(N,0.5)
    lo=float(np.min(z))-1.0; hi=float(np.max(z))
    if not np.isfinite(lo) or not np.isfinite(hi): return np.full(N,0.5)
    for _ in range(70):
        mid=0.5*(lo+hi)
        s=np.sum(np.clip(z-mid,0.0,1.0))
        if s>m: lo=mid
        else: hi=mid
    h=np.clip(z-0.5*(lo+hi),0.0,1.0)
    r=m-h.sum()
    if r>0:
       for i in range(N):
          if r<=1e-14: break
          c=1.0-h[i]
          a=min(r,c)
          if a>0: h[i]+=a; r-=a
    elif r<0:

          c=h[i]

    if h.min()< -1e-9 or h.max()>1+1e-9 or abs(h.sum()-m)>1e-7*max(1,N):
       return np.full(N,0.5)
    return np.clip(h,0,1)

def project_box_sum(z, m):

def project_box_sum(z, m):
    z = np.asarray(z, float)
    N = len(z)
    if N == 0 or not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z))
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - float(h.sum())
    if r > 0.0:
        for i in range(N):
            if r <= 1e-14:
                break
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
    elif r < 0.0:
        for i in range(N):
            if r >= -1e-14:
                break
            a = min(-r, h[i])
            h[i] -= a
            r += a
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - m) > 1e-7 * max(1, N):
        return np.full(N, 0.5)
    return h

def fine_adjust(h, m):
    r = m - float(h.sum())
    if r > 0.0:
        for i in range(len(h)):
            if h[i] <= 1.0 - r + 1e-15:
                h[i] += r
                break
    elif r < 0.0:
        for i in range(len(h)):
            if h[i] >= -r - 1e-15:
                h[i] += r
                break
    return h

def lag_scores(h, N):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(N))

def smooth_logsum(c, tau):
    cm = float(np.max(c))
    if not math.isfinite(cm):
        return cm, np.zeros_like(c)
    e = np.exp((c - cm) / tau)
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
        return cm, p
    return cm + tau * math.log(s), e / s

def grad_from_weights(h, w, N):
    g = np.zeros(N, dtype=float)
    dx = 2.0 / float(N)
    wp = w[N - 1]
    if wp:
        g += dx * wp * (1.0 - 2.0 * h)
    for s in range(1, N):
        L = N - s
        wp = w[N - 1 + s]
        if wp:
            g[:L] += dx * wp * (1.0 - h[s:])
            g[s:] -= dx * wp * h[:L]
        wn = w[N - 1 - s]
        if wn:
            g[s:] += dx * wn * (1.0 - h[:L])
            g[:L] -= dx * wn * h[s:]
    return g

            stop = False
            for _ in range(bt):
                step = a / (1.0 + gn)
                if step <= 1e-11:
                    break
                y = project_box_sum(h - step * g, m)
                if not np.all(np.isfinite(y)):
                    break
                d = y - h
                dl = float(np.max(np.abs(d)))
                cy = lag_scores(y, N)
                if not np.all(np.isfinite(cy)):
                    break
                Cv = float(cy.max())
                if Cv < bc:
                    bc = Cv
                    bh = y.copy()
                if dl <= 1e-9:
                    acc = True
                    trial = y
                    used = a
                    stop = True
                    break
                Fy, _ = smooth_logsum(cy, tau)
                gd = float(np.dot(g, d))
                if math.isfinite(Fy) and ((gd < -1e-14 and Fy <= F + 1e-4 * gd + 1e-12) or Fy < F - 1e-11):
                    acc = True
                    trial = y
                    used = a
                    if F - Fy <= 1e-10 * max(1.0, abs(F)):
                        stagn += 1
                    else:
                        stagn = 0
                    break
                a *= 0.5
            if not acc:
                break
            h = project_box_sum(trial, m)
            if stop:
                break
            alpha = min(1.25 * used, 10.0)
            c = lag_scores(h, N)
            F, p = smooth_logsum(c, tau)
            if not math.isfinite(F):
                break
            g = grad_from_weights(h, p, N)
            g -= float(np.mean(g))
            if stagn >= 6:
                break
    best = project_box_sum(bh, m)
    cb = lag_scores(best, N)
    bcb = float(cb.max()) if np.all(np.isfinite(cb)) else np.inf
    if not math.isfinite(bcb) or bcb > bc + 1e-9:
        return bh, bc
    return best, bcb

def active_refine(h0, N, m, iters, kick=0.02, rng=None):
    h = project_box_sum(h0, m)
    c = lag_scores(h, N)
    bh = h.copy()
    bc = float(c.max())
    no = 0
    for _ in range(iters):
        C = float(c.max())
        if not math.isfinite(C):
            break
        tol = max(1e-8, 0.005 * C + 1e-5)
        mask = c >= C - tol
        cnt = int(mask.sum())
        if cnt <= 0:
            break
        w = mask.astype(float) / float(cnt)
        g = grad_from_weights(h, w, N)
        g -= float(np.mean(g))
        gn = float(np.linalg.norm(g))
        improved = False
        if gn > 1e-13:
            a = 0.5
            for _ in range(14):
                step = a / (1.0 + gn)
                if step <= 1e-12:
                    break
                y = project_box_sum(h - step * g, m)
                if not np.all(np.isfinite(y)):
                    break
                cy = lag_scores(y, N)
                if not np.all(np.isfinite(cy)):
                    break
                Cv = float(cy.max())
                if Cv < bc:
                    bc = Cv
                    bh = y.copy()
                if Cv < C - 1e-12:
                    h = y
                    c = cy
                    improved = True
                    break
                a *= 0.5
        if improved:
            no = 0
        else:
            no += 1
            if no >= 5:
                if rng is None:
                    break
                h = project_box_sum(h + kick * rng.standard_normal(N), m)
                c = lag_scores(h, N)
                if not np.all(np.isfinite(c)):
                    h = project_box_sum(bh, m)
                    c = lag_scores(h, N)
                no = 0
    best = project_box_sum(bh, m)
    cb = lag_scores(best, N)
    bcb = float(cb.max()) if np.all(np.isfinite(cb)) else np.inf
    if not math.isfinite(bcb) or bcb > bc + 1e-9:
        return bh, bc
    return best, bcb

def search_from(h0, N, m, rng, taus, maxit, ait):
    h, c = pgl_search(h0, N, m, taus, maxit)
    if not math.isfinite(c):
        h = np.full(N, 0.5)
        c = float(lag_scores(h, N).max())
    if len(taus) > 3:
        h2, c2 = pgl_search(h, N, m, taus[-3:], max(6, maxit // 3))
        if math.isfinite(c2) and c2 < c:
            h, c = h2, c2
    h3, c3 = active_refine(h, N, m, ait, 0.02, rng)
    if math.isfinite(c3) and c3 < c:
        h, c = h3, c3
    h = project_box_sum(h, m)
    c = float(lag_scores(h, N).max())
    return h, c

def make_fourier_start(N, rng, n_modes, amp):
    idx = np.arange(N, dtype=float)
    z = np.full(N, 0.5, dtype=float)
    nm = min(n_modes, (N - 1) // 2)
    for r in range(1, nm + 1):
        a = float(rng.standard_normal())
        b = float(rng.standard_normal())
        theta = 2.0 * math.pi * r * idx / float(N)
        z += (amp / float(r)) * (a * np.sin(theta) + b * np.cos(theta))
    return z

def add_start(out, z, m, tol=1e-8):
    z = project_box_sum(z, m)
    if not np.all(np.isfinite(z)):
        return
    for y in out:
        if np.max(np.abs(z - y)) < tol:
            return
    out.append(z)

def top_half(vals, N, k):
    b = np.zeros(N, dtype=float)
    if 0 < k < N:
        b[np.argsort(-vals)[:k]] = 1.0
    elif k >= N:
        b.fill(1.0)
    return b

def binary_local(N, rng, init, maxit):
    half = N // 2
    bits = np.zeros(N, dtype=float)
    if init is not None:
        b = np.asarray(init, float)
        if b.size == N and abs(b.sum() - half) < 1e-7:
            b = (b > 0.5).astype(float)
            if abs(b.sum() - half) < 1e-7:
                bits = b
    if abs(bits.sum() - half) > 1e-7:
        bits[:half] = 1.0
        rng.shuffle(bits)
    best = bits.copy()
    bc = float(lag_scores(best, N).max())
    no = 0
    for _ in range(maxit):
        ones = np.flatnonzero(bits > 0.5)
        zeros = np.flatnonzero(bits < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = int(ones[rng.integers(ones.size)])
        j = int(zeros[rng.integers(zeros.size)])
        bits[i], bits[j] = bits[j], bits[i]
        cv = float(lag_scores(bits, N).max())
        if cv < bc - 1e-12:
            best = bits.copy()
            bc = cv
            no = 0
        else:
            bits[i], bits[j] = bits[j], bits[i]
            no += 1
        if no >= 120:
            for _ in range(4):
                ones = np.flatnonzero(bits > 0.5)
                zeros = np.flatnonzero(bits < 0.5)
                if ones.size == 0 or zeros.size == 0:
                    break
                i = int(ones[rng.integers(ones.size)])
                j = int(zeros[rng.integers(zeros.size)])
                bits[i], bits[j] = bits[j], bits[i]
            cv = float(lag_scores(bits, N).max())
            if cv < bc:
                best = bits.copy()
                bc = cv
            no = 0
    return best.astype(float), bc

def run(seed=42, **kw):
    raw = globals().get("initial_h_values", None)
    N = -1
    have = False
    if raw is not None:
        a = np.asarray(raw, float)
        if a.ndim == 1 and a.size > 1 and np.all(np.isfinite(a)):
            N = int(a.size)
            have = True
    if N < 0:
        try:
            N = int(kw.get("n_points", 70))
        except Exception:
            N = 70
    if N < 2:
        N = 70
    if not have and N > 512:
        N = 512
    m = 0.5 * float(N)
    if have:
        parent = project_box_sum(raw, m)
    else:
        parent = np.full(N, 0.5)
    parent = fine_adjust(parent, m)
    if not np.all(np.isfinite(parent)):
        parent = np.full(N, 0.5)
    pc = float(lag_scores(parent, N).max())
    if N > 600:
        return parent, pc, int(N)
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    if N <= 128:
        taus = [0.03, 0.008, 0.002, 0.0005, 0.00012, 3e-5, 1e-5]
        hard = pc > 0.3807
        maxit = 40 if hard else 28
        ait = 35 if hard else 22
        bmax = 1500 if hard else 900
    elif N <= 256:
        taus = [0.03, 0.008, 0.002, 0.0005, 0.00012, 3e-5]
        maxit = 20
        ait = 15
        bmax = 500
    else:
        taus = [0.025, 0.006, 0.0015, 0.0004, 1e-4]
        maxit = 12
        ait = 8
        bmax = 150
    bh = None
    if N % 2 == 0 and N <= 120:
        bits0 = top_half(parent, N, N // 2)
        bh, _ = binary_local(N, rng, bits0, bmax)
    starts = []
    add_start(starts, parent, m)
    if bh is not None:
        add_start(starts, bh, m)
    add_start(starts, 1.0 - parent, m)
    add_start(starts, 0.5 * (parent + 1.0 - parent[::-1]), m)
    add_start(starts, parent + 0.06 * rng.standard_normal(N), m)
    add_start(starts, parent + 0.20 * rng.standard_normal(N), m)
    add_start(starts, 0.5 + 0.5 * rng.uniform(size=N), m)
    add_start(starts, make_fourier_start(N, rng, 3, 0.12), m)
    add_start(starts, make_fourier_start(N, rng, 5, 0.16), m)
    K = 8 if N <= 128 else (6 if N <= 256 else 4)
    if len(starts) > K:
        starts = starts[:K]
    best = parent.copy()
    bc = pc
    for st in starts:
        h, c = search_from(st, N, m, rng, taus, maxit, ait)
        if c < bc:
            best = h.copy()
            bc = c
    tail = taus[-3:]
    hops = (0.10, 0.04, 0.015) if N <= 128 else ((0.08,) if N <= 256 else ())
    for sc in hops:
        hop = project_box_sum(best + sc * rng.standard_normal(N), m)
        h, c = search_from(hop, N, m, rng, tail, max(6, maxit // 2), max(4, ait // 2))
        if c < bc:
            best = h.copy()
            bc = c
    for cand in (0.5 * (best + 1.0 - best[::-1]), 0.5 * (best + best[::-1])):
        cand = project_box_sum(cand, m)
        cc = float(lag_scores(cand, N).max())
        if cc < bc:
            best = cand.copy()
            bc = cc
    h, c = active_refine(best, N, m, max(5, ait // 2), 0.012, rng)
    if c < bc:
        best = h.copy()
        bc = c
    parent = project_box_sum(parent, m)
    parent = fine_adjust(parent, m)
    pc = float(lag_scores(parent, N).max())
    if pc < bc:
        best = parent.copy()
        bc = pc
    best = project_box_sum(best, m)
    best = fine_adjust(best, m)
    best = np.clip(best, 0.0, 1.0)
    cb = lag_scores(best, N)
    final = float(cb.max()) if np.all(np.isfinite(cb)) else np.inf
    if not math.isfinite(final):
        best = parent.copy()
        final = pc
    if best.min() < -1e-7 or best.max() > 1.0 + 1e-7 or abs(best.sum() - m) > 1e-6 * max(1, N):
        best = project_box_sum(parent, m)
        best = fine_adjust(best, m)
        cb = lag_scores(best, N)
        final = float(cb.max()) if np.all(np.isfinite(cb)) else pc
    return best.astype(float), float(final), int(N)
