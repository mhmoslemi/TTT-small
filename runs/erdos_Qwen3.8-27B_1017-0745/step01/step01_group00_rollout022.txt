- Decision vector: z = [y_1, ..., y_n, t] where y in [0,1]^n, t free
- Objective: t + beta*dx*||y-x||^2
- Constraints: t >= q[d] + G[d,:] @ (y-x) for all d (2n-1 constraints)
- Equality: sum(y) = n/2
- Bounds: 0 <= y_i <= 1

For SLSQP, inequalities are g_i(z) >= 0.
So: t - q[d] - G[d,:] @ (y-x) >= 0 for each d.

The gradient of the objective w.r.t. z:
- d/dy_i: 2*beta*dx*(y_i - x_i)
- d/dt: 1

The Jacobian of constraint d:
- d/dy_i: -G[d,i]
- d/dt: 1

Equality constraint: sum(y) - n/2 = 0
Jacobian: [1, 1, ..., 1, 0]

import numpy as np
from scipy.optimize import minimize

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def project_capped(z, n):
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    s = n / 2.0
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    return h

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def lag_model(x, n):
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.zeros(L)
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        p0 = max(d, 0)
        q0 = max(-d, 0)
        m = n - abs(d)
        p = x[p0:p0 + m]
        qv = x[q0:q0 + m]
        q[row] = dx * np.dot(p, 1.0 - qv)
        G[row, p0:p0 + m] += dx * (1.0 - qv)
        G[row, q0:q0 + m] -= dx * p
    return q, G

def _mm_obj(z, x, q, G, beta, dx, n):
    y = z[:n]
    return z[n] + beta * dx * np.dot(y - x, y - x)

def _mm_grad(z, x, q, G, beta, dx, n):
    g = np.empty(n + 1)
    g[:n] = 2.0 * beta * dx * (z[:n] - x)
    g[n] = 1.0
    return g

def _mm_ineq(z, x, q, G, n):
    return z[n] - q - G @ (z[:n] - x)

def _mm_ineq_jac(z, x, q, G, n):
    J = np.empty((2 * n - 1, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def _mm_eq(z, x, q, G, n):
    return float(np.sum(z[:n]) - n / 2.0)

def _mm_eq_jac(z, x, q, G, n):
    v = np.zeros(n + 1)
    v[:n] = 1.0
    return v

def solve_qp(x, n, beta, maxiter):
    dx = 2.0 / n
    q, G = lag_model(x, n)
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None
    x0 = np.empty(n + 1)
    x0[:n] = x
    x0[n] = float(np.max(q)) + 1e-10
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    args_main = (x, q, G, beta, dx, n)
    args_c = (x, q, G, n)
    cons = [
        {'type': 'ineq', 'fun': _mm_ineq, 'jac': _mm_ineq_jac, 'args': args_c},
        {'type': 'eq', 'fun': _mm_eq, 'jac': _mm_eq_jac, 'args': args_c},
    ]
    try:
        res = minimize(_mm_obj, x0, args=args_main, jac=_mm_grad,
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-12})
    except Exception:
        return None
    if res.x is None or res.x.shape != (n + 1,) or not np.all(np.isfinite(res.x)):
        return None
    return res.x[:n].copy()

def subgrad_solve(x, n, beta, n_iter):
    dx = 2.0 / n
    y = x.copy()
    best_y, best_m = y.copy(), float('inf')
    for k in range(n_iter):
        u = y - x
        avals = np.empty(2 * n - 1)
        for d in range(-(n - 1), n):
            p0, q0, m = max(d, 0), max(-d, 0), n - abs(d)
            p = x[p0:p0 + m]
            qv = x[q0:q0 + m]
            avals[d + n - 1] = dx * (np.dot(p, 1.0 - qv) + np.dot(1.0 - qv, u[p0:p0 + m]) - np.dot(p, u[q0:q0 + m]))
        row = int(np.argmax(avals))
        d = row - (n - 1)
        p0, q0, m = max(d, 0), max(-d, 0), n - abs(d)
        g = np.zeros(n)
        g[p0:p0 + m] += dx * (1.0 - x[q0:q0 + m])
        g[q0:q0 + m] -= dx * x[p0:p0 + m]
        v = g + 2.0 * beta * dx * u
        nv = np.linalg.norm(v)
        if nv < 1e-15:
            break
        eta = min(0.5 / (nv + 1e-12), 1.0 / (2.0 * beta * dx * np.sqrt(float(k + 1))))
        yn = project_capped(y - eta * v, n)
        if yn is None:
            break
        if np.max(np.abs(yn - y)) < 1e-9:
            y = yn
            break
        y = yn
        mv = avals[row] + beta * dx * np.dot(y - x, y - x)
        if mv < best_m:
            best_m, best_y = mv, y.copy()
    return best_y

def build_starts(fb, n, rng):
    cands = [fb.copy(), fb[::-1].copy(), 1.0 - fb.copy(), 1.0 - fb[::-1].copy()]
    cands.append(np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    cands.append(blk)
    cands.append(np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64))
    cands.append(fb + rng.normal(0, 0.08, n))
    cands.append(fb + rng.normal(0, 0.20, n))
    cands.append(fb + rng.normal(0, 0.45, n))
    cands.append(rng.uniform(0, 1, n))
    pool = []
    for c in cands:
        h = project_capped(c, n)
        if h is None or not is_feasible(h, n):
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(h - s)) < 1e-8:
                dup = True
                break
        if not dup:
            pool.append(h)
    return pool

def optimize_start(x0, n, n_outer, n_inner, slsqp_mi):
    x = x0.copy().astype(np.float64)
    score = exact_score(x, n)
    if not np.isfinite(score):
        return x, score
    beta = 1.0
    rejs = 0
    for _ in range(n_outer):
        if rejs >= 5 or beta > 64.0:
            break
        if n <= 128:
            y = solve_qp(x, n, beta, slsqp_mi)
        else:
            y = subgrad_solve(x, n, beta, n_inner)
        if y is None:
            beta *= 2.0
            rejs += 1
            continue
        y = project_capped(y, n)
        if y is None or not is_feasible(y, n):
            beta *= 2.0
            rejs += 1
            continue
        if np.max(np.abs(y - x)) < 1e-10:
            break
        best_z, best_s = None, score
        for kk in range(5):
            alpha = 1.0 / (2 ** kk)
            z = project_capped(x + alpha * (y - x), n)
            if z is None or not is_feasible(z, n):
                continue
            sz = exact_score(z, n)
            if sz < best_s:
                best_s, best_z = sz, z
        if best_z is None or best_s >= score - 1e-12:
            beta *= 2.0
            rejs += 1
            continue
        x, score = best_z.copy(), best_s
        rejs = 0
        beta = max(1.0, beta / 1.5)
    return x, exact_score(x, n)
