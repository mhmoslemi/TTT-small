def pair_delta(h, p, q, a, n):
    # Returns delta for all lags, length 2n-1
    delta = np.zeros(2*n-1)
    for k in range(-(n-1), n):
        d = 0.0
        # Term A: v[i]*(1-h[i-k]) for i in {p,q} with 0<=i-k<n
        pk = p - k
        if 0 <= pk < n:
            d += a * (1.0 - h[pk])
        qk = q - k
        if 0 <= qk < n:
            d += (-a) * (1.0 - h[qk])
        # Term B: -h[i]*v[i-k] for i={p+k, q+k} with 0<=i<n
        pk2 = p + k
        if 0 <= pk2 < n:
            d += -a * h[pk2]
        qk2 = q + k
        if 0 <= qk2 < n:
            d += a * h[qk2]
        # Term C: -v[i]*v[i-k]
        if k == 0:
            d += -2.0 * a * a
        elif k == p - q:
            d += a * a
        elif k == q - p:
            d += a * a
        delta[k + n - 1] = d
    return delta

import numpy as np

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n < 2 or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-14:
        if resid > 0:
            order = np.argsort(1.0 - y)
            for i in order:
                add = min(resid, 1.0 - y[i])
                if add <= 0: break
                y[i] += add
                resid -= add
                if abs(resid) < 1e-15: break
        else:
            order = np.argsort(y)
            for i in order:
                sub = min(-resid, y[i])
                if sub <= 0: break
                y[i] -= sub
                resid += sub
                if abs(resid) < 1e-15: break
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(np.sum(y) - S) > 1e-10 * n:
        return None
    return y

def exact_score(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def pair_delta(h, p, q, a, n):

    for k in range(-(n - 1), n):
        d = 0.0
        pk = p - k
        if 0 <= pk < n:

        qk = q - k

    best_h = h.copy()
    best_score = cur_score
    temps = [min(0.03, 0.003 * (45.0 / n)) * (0.35 ** s) for s in range(6)]
    radii = [0.20, 0.12, 0.07, 0.04, 0.02, 0.008]
    for stage in range(6):

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n < 2 or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-14:
        if resid > 0:
            order = np.argsort(1.0 - y)
            for i in order:
                add = min(resid, 1.0 - y[i])
                if add <= 0:
                    break
                y[i] += add
                resid -= add
                if abs(resid) < 1e-15:
                    break
        else:
            order = np.argsort(y)
            for i in order:
                sub = min(-resid, y[i])
                if sub <= 0:
                    break
                y[i] -= sub
                resid += sub
                if abs(resid) < 1e-15:
                    break
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(np.sum(y) - S) > 1e-10 * n:
        return None
    return y

            d += -a * h[pk2]

        if 0 <= qk2 < n:
            d += a * h[qk2]
        if k == 0:
            d += -2.0 * a * a
        elif k == p - q or k == q - p:
            d += a * a
        delta[k + n - 1] = d
