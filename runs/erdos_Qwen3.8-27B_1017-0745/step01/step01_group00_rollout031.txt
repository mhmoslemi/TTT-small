def project_capped(z, n):
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    idx = np.arange(n)
    if r > 0:
        for i in idx:
            if r <= 1e-14: break
            add = min(r, 1.0 - h[i])
            h[i] += add
            r -= add
    else:
        r = -r
        for i in idx:
            if r <= 1e-14: break
            sub = min(r, h[i])
            h[i] -= sub
            r -= sub
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    return h

    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def smooth_value_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    Q = np.zeros(L)
    for d in range(-(n-1), n):

        b = max(-d, 0)
        m = n - abs(d)
        Q[d + n - 1] = dx * np.sum(h[a:a+m] * (1.0 - h[b:b+m]))
    q_max = np.max(Q)
    exps = np.exp((Q - q_max) / tau)
    w = exps / np.sum(exps)
    F = q_max + tau * np.log(np.sum(exps))
    g = np.zeros(n)
    for d in range(-(n-1), n):
        a = max(d, 0)
        b = max(-d, 0)
        m = n - abs(d)
        wd = w[d + n - 1]
        g[a:a+m] += wd * dx * (1.0 - h[b:b+m])
        g[b:b+m] -= wd * dx * h[a:a+m]

        1.0 - fallback[::-1],

def project_capped(z, n):
    z = np.clip(z.astype(np.float64), -1e12, 1e12)

        else: hi = mid
