def build_upper_hull(lines):
    # lines: list of (slope, intercept)
    # Returns list of (slope, intercept, x_start, x_end) for hull segments
    if not lines:
        return []
    # Sort by slope, then by intercept descending (keep max intercept for same slope)
    lines.sort()
    # Remove duplicates keeping max intercept
    unique = []
    for m, c in lines:
        if unique and unique[-1][0] == m:
            if c > unique[-1][1]:
                unique[-1] = (m, c)
        else:
            unique.append((m, c))
    # Build hull
    hull = []  # list of (m, c, x_start)
    for m, c in unique:
        if not hull:
            hull.append([m, c, -float('inf')])
            continue
        # Intersection with previous
        pm, pc, px = hull[-1]
        if m == pm:
            continue  # shouldn't happen after dedup
        x_cross = (pc - c) / (m - pm)
        while hull and x_cross <= hull[-1][2]:
            hull.pop()
            if not hull:
                x_cross = -float('inf')
                break
            pm, pc, px = hull[-1]
            if m == pm:
                continue
            x_cross = (pc - c) / (m - pm)
        hull.append([m, c, x_cross if hull else -float('inf')])
    # Actually the last start is already set
    # Add x_end
    result = []
    for i, (m, c, xs) in enumerate(hull):
        xe = hull[i+1][2] if i + 1 < len(hull) else float('inf')
        result.append((m, c, xs, xe))
    return result

import numpy as np

def project_capped(z, n):
    s = n / 2.0
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    lo, hi = float(z.min()) - 1.0, float(z.max()) + 1.0
    for _ in range(100):
        mid = lo + (hi - lo) * 0.5
        if mid == lo or mid == hi:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(h.sum())
    for _ in range(3):
        if abs(r) <= 1e-13 * max(1, n):
            break
        if r > 0:
            for i in range(n):
                if r <= 1e-15:
                    break
                add = min(r, 1.0 - h[i])
                h[i] += add
                r -= add
        else:
            for i in range(n):
                if r >= -1e-15:
                    break
                rem = min(-r, h[i])
                h[i] -= rem
                r += rem
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(h.sum()) - s) > 5e-12 * max(1, n):
        return None
    return h

def is_feasible(h, n):
    return h.shape == (n,) and np.all(np.isfinite(h)) and np.all(h >= -1e-10) and np.all(h <= 1.0 + 1e-10) and abs(float(h.sum()) - n / 2.0) <= 5e-10 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    try:
        return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
    except Exception:
        return float('inf')

def lag_raw_values(h, n):
    L = 2 * n - 1
    c = np.zeros(L)
    for d in range(-(n - 1), n):
        i0 = max(0, -d)
        j0 = max(0, d)
        m = n - abs(d)
        c[d + n - 1] = float(np.sum(h[i0:i0+m] * (1.0 - h[j0:j0+m])))
    if not np.all(np.isfinite(c)):
        return None
    return c

        DP = 0.0
        DQ = 0.0
        pd = p + d

        pmd = p - d
        if 0 <= pmd < n:

        return 0.0

               n-1+D: (a[n-1+D], b[n-1+D], c[n-1+D]),
               n-1-D: (a[n-1-D], b[n-1-D], c[n-1-D])}

        hs = []
