import numpy as np
from scipy.signal import fftconvolve

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) < 1e-14:
        return y
    if resid > 0:
        order = np.argsort(1.0 - y, kind='stable')
        for i in order:
            add = min(resid, 1.0 - y[i])
            if add <= 0:
                break
            y[i] += add
            resid -= add
            if abs(resid) < 1e-15:
                break
    else:
        order = np.argsort(y, kind='stable')
        for i in order:
            sub = min(-resid, y[i])
            if sub <= 0:
                break
            y[i] -= sub
            resid += sub
            if abs(resid) < 1e-15:
                break
    if abs(resid) > 1e-10:
        return None
    y = np.clip(y, 0.0, 1.0)
    return y

def exact_score(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def smooth_value_grad(h, theta):
    n = len(h)
    tau = n * theta
    # Compute all Q_k
    Q = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        i_lo = max(0, k)
        i_hi = min(n - 1, n - 1 + k)
        if i_lo > i_hi:
            continue
        i = np.arange(i_lo, i_hi + 1)
        j = i - k
        Q[k + n - 1] = np.sum(h[i] * (1.0 - h[j]))
    # Log-sum-exp
    m = np.max(Q)
    exp_vals = np.exp((Q - m) / tau)
    s = np.sum(exp_vals)
    if not np.isfinite(s) or s <= 0:
        return None, None
    w = exp_vals / s
    F = m + tau * np.log(s)
    # Gradient
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for k in range(-(n - 1), n):
        wk = w[k + n - 1]
        if abs(wk) < 1e-15:
            continue
        i_lo = max(0, k)
        i_hi = min(n - 1, n - 1 + k)
        if i_lo > i_hi:
            continue
        i = np.arange(i_lo, i_hi + 1)
        j = i - k
        grad[i] += wk * one_minus_h[j]
        grad[j] -= wk * h[i]
    if not np.all(np.isfinite(grad)):
        return None, None
    return F, grad

def make_harmonic_noise(n, rng):
    x = np.zeros(n)
    for _ in range(2):
        m = rng.integers(1, n // 2 + 1)
        ph = rng.uniform(0, 2 * np.pi)
        x += np.cos(2 * np.pi * m * np.arange(n) / n + ph)
    x -= np.mean(x)
    rms = np.sqrt(np.mean(x**2))
    if rms < 1e-10:
        return rng.standard_normal(n)
    return x / rms
