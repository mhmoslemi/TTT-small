            ts = exact_score(trial, n)
            if ts < best_s:
                best_s, best_h = ts, trial
        if best_s < sc - 1e-12:
            h = best_h
            sc = best_s
            fails = 0
            eta = min(1.2 * eta, eta_max)
        else:
            eta *= 0.5
            fails += 1
    return h, sc

def build_starts(fallback, n, rng, count):
    pool = []
    def _add(v):
        for s in pool:
            if np.max(np.abs(v - s)) < 1e-8:
                return
        pool.append(v)
    _add(fallback.copy())
    r = project_capped(fallback[::-1], n)
    if r is not None:
        _add(r)
    r = project_capped(1.0 - fallback[::-1], n)
    if r is not None:
        _add(r)
    _add(np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    _add(project_capped(blk, n) if project_capped(blk, n) is not None else np.full(n, 0.5))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    r = project_capped(alt, n)
    if r is not None:
        _add(r)
    tri = np.abs(np.arange(n) - n / 2.0) / (n / 2.0)
    r = project_capped(1.0 - tri, n)
    if r is not None:
        _add(r)
    for sd in (0.08, 0.20, 0.50):
        r = project_capped(fallback + rng.normal(0, sd, n), n)
        if r is not None:
            _add(r)
    r = project_capped(rng.uniform(0, 1, n), n)
    if r is not None:
        _add(r)
    return pool[:count]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    n = max(1, min(999, n))
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if parent is not None:
        fallback = project_capped(parent, n)
    else:
        fallback = None
    if fallback is None or not is_feasible(fallback, n):
        fallback = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        return np.array([0.5]), 0.5, 1
    if n <= 64:
        n_starts, n_passes, maxiter = 8, 3, 250
    elif n <= 128:
        n_starts, n_passes, maxiter = 4, 2, 200
    elif n <= 256:
        n_starts, n_passes, maxiter = 2, 0, 0
    else:
        n_starts, n_passes, maxiter = 1, 0, 0
    starts = build_starts(fallback, n, rng, n_starts)
    gb, gs = fallback.copy(), exact_score(fallback, n)
    for st in starts:
        sct = exact_score(st, n)
        if sct < gs - 1e-12:
            gs, gb = sct, st.copy()
    if n <= 128 and n_passes > 0:
        for st in starts:
            try:
                xo, so = polish(st, n, n_passes, maxiter)
                if so < gs - 1e-12:
                    gs, gb = so, xo.copy()
            except Exception:
                pass
    else:
        if n <= 128:
            nit = 45
        elif n <= 256:
            nit = 30
        else:
            nit = 18
        for st in starts:
            try:
                xo, so = subgrad_search(st, n, nit)
                if so < gs - 1e-12:
                    gs, gb = so, xo.copy()
            except Exception:
                pass
    gb = project_capped(gb, n)
    if gb is None or not is_feasible(gb, n):
        gb = fallback
    if exact_score(gb, n) > exact_score(fallback, n) + 1e-12:
        gb = fallback
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = np.full(n, 0.5)
    return final, float(exact_score(final, n)), int(n)
