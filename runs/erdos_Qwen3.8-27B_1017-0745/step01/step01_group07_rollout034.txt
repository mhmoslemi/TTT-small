The task is to find a step function h: [0,2] → [0,1] that minimizes C₅ = max_k ∫ h(x)(1-h(x+k)) dx, subject to h ∈ [0,1] and ∫₀² h(x)dx = 1.

The strategy proposes using DC (Difference of Convex) Majorization-Minimization with a convex epigraph subproblem. Let me implement this.

Key ideas:
1. For each lag d, the quadratic overlap function q_d(h) = dx * sum h[i]*(1-h[i+d]) can be decomposed as q_d(h) = A_d(h) - B_d(h) where both A_d and B_d are convex.
2. At anchor a, replace B_d with its tangent to get a convex upper bound M_d(y;a).
3. Solve the convex subproblem: min t + eta/2*||y-a||² s.t. M_d(y;a) <= t for all d, 0<=y<=1, sum(y)=S, |y-a|<=rho.
4. Use line search with exact scoring to accept improvements.

import numpy as np
import cvxpy as cp
import math

def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):
        return None
    if np.max(np.abs(z)) > 1e9:
        return None
    n = len(z)
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    x = np.clip(z - lam, 0.0, 1.0)
    r = S - np.sum(x)
    for i in range(n):
        if r > 0:
            a = min(r, 1.0 - x[i])
            x[i] += a; r -= a
        elif r < 0:
            a = min(-r, x[i])
            x[i] -= a; r += a
        else:
            break
    if abs(np.sum(x) - S) > 1e-10 or not np.all(np.isfinite(x)):
        return None
    return np.clip(x, 0.0, 1.0)

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) - n/2.0) <= 1e-10)

def exact_score(h, n):
    try:
        vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if vals.size != 2*n - 1 or not np.all(np.isfinite(vals)):
            return None
        return float(np.max(vals))
    except Exception:
        return None

def solve_dc(anchor, n, S, dx, rho, eta):
    y = cp.Variable(n)
    t = cp.Variable()
    cons = [y >= 0, y <= 1, cp.sum(y) == S, y - anchor <= rho, anchor - y <= rho]
    for d in range(-(n-1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        y0 = y[lo:hi]
        y1 = y[lo+d:hi+d]
        s = np.asarray(anchor[lo:hi] + anchor[lo+d:hi+d])
        lhs = dx * (cp.sum(y0) + 0.25*cp.sum_squares(y0 - y1) - 0.5*cp.sum(s * (y0 + y1)) + 0.25*float(np.dot(s, s)))
        cons.append(lhs <= t)
    obj = cp.Minimize(t + 0.5 * eta * cp.sum_squares(y - anchor))
    prob = cp.Problem(obj, cons)
    if not prob.is_dcp():
        return None
    for solver in [cp.ECOS, cp.CVXOPT, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                if y.value is not None and np.all(np.isfinite(y.value)):
                    return np.asarray(y.value, dtype=np.float64)
        except Exception:
            pass
    return None

def optimize_dc(x, n, S, dx, archive, max_attempts):
    x = x.copy()
    f = exact_score(x, n)
    if f is None:
        return archive
    if f < archive[1]:
        archive = (x.copy(), f)
    rho, eta = 0.20, 0.20 * dx
    rho_min, eta_min = 0.005, 0.02 * dx
    fail_cnt, noimp_cnt, small_cnt = 0, 0, 0
    for _ in range(max_attempts):
        cand = solve_dc(x, n, S, dx, rho, eta)
        if cand is None:
            rho = max(rho_min, rho * 0.5)
            fail_cnt += 1
            if fail_cnt >= 2 or rho <= rho_min:
                break
            continue
        proj = project_capped_simplex(cand, S)
        if proj is None:
            rho = max(rho_min, rho * 0.5)
            fail_cnt += 1
            if fail_cnt >= 2 or rho <= rho_min:
                break
            continue
        best_trial, best_f = None, f
        for j in range(7):
            alpha = 2.0 ** (-j)
            trial = x + alpha * (proj - x)
            tp = project_capped_simplex(trial, S)
            if tp is None:
                continue
            sc = exact_score(tp, n)
            if sc is not None and sc < best_f - 1e-12:
                best_f, best_trial = sc, tp.copy()
        if best_trial is None:
            rho = max(rho_min, rho * 0.5)
            eta = max(eta_min, eta * 0.5)
            noimp_cnt += 1
            if noimp_cnt >= 3 or (rho <= rho_min and eta <= eta_min):
                break
            continue
        decrease = f - best_f
        disp = np.max(np.abs(best_trial - x))
        x, f = best_trial, best_f
        if f < archive[1]:
            archive = (x.copy(), f)
        fail_cnt = 0
        noimp_cnt = 0
        if decrease <= 1e-10:
            small_cnt += 1
        else:
            small_cnt = 0
        if small_cnt >= 2:
            break
        if disp <= 1e-7:
            break
        if disp >= 0.8 * rho and decrease > 1e-5:
            rho = min(0.40, 1.25 * rho)
    return archive

def build_starts(n, S, parent, rng):
    starts = []
    const = np.full(n, 0.5)
    starts.append(const)
    if parent is not None:
        starts.append(parent.copy())
        comp = 1.0 - parent
        starts.append(comp.copy())
        rev = parent[::-1].copy()
        starts.append(rev)
        pert = parent + rng.normal(0, 0.05, n)
        p = project_capped_simplex(pert, S)
        if p is not None:
            starts.append(p)
    if n <= 256:
        M = min(6, (n - 1) // 2)
        idx = np.arange(n, dtype=np.float64) + 0.5
        vec = np.full(n, 0.5)
        for m in range(1, M + 1):
            amp = 0.10 / np.sqrt(m) * rng.standard_normal()
            phi = rng.uniform(0, 2 * np.pi)
            vec += amp * np.cos(2 * np.pi * m * idx / n + phi)
        lf = project_capped_simplex(vec, S)
        if lf is not None:
            starts.append(lf)
        if parent is None:
            perm = rng.permutation(n)
            bv = np.zeros(n)
            bv[perm[:n//2]] = 1.0
            if n % 2 == 1:
                bv[perm[n//2]] = 0.5
            starts.append(bv.copy())
            vu = project_capped_simplex(rng.random(n), S)
            if vu is not None:
                starts.append(vu)
    deduped = []
    for s in starts:
        dup = False
        for d in deduped:
            if np.max(np.abs(np.asarray(s) - np.asarray(d))) < 1e-9:
                dup = True
                break
        if not dup:
            arr = np.asarray(s, dtype=np.float64)
            if is_feasible(arr, n):
                deduped.append(arr)
    return deduped

def run(seed=42, **kwargs):
    if seed is None:
        seed = 42
    try:
        seed = int(seed)
    except (TypeError, ValueError):
        seed = 42
    seed = seed % (2**32)
    rng = np.random.default_rng(seed)
    n = None
    parent = None
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        try:
            iv_arr = np.asarray(iv, dtype=np.float64)
            if iv_arr.ndim == 1 and 8 <= iv_arr.size <= 999:
                n = iv_arr.size
        except Exception:
            pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 80))
        except Exception:
            n = 80
        if not (8 <= n <= 999):
            n = 80
    S = n / 2.0
    dx = 2.0 / n
    const = np.full(n, 0.5, dtype=np.float64)
    best_h, best_score = const.copy(), exact_score(const, n)
    if best_score is None:
        best_score = 0.25  # analytic for constant: dx*n/4 = 0.5*0.5 = 0.25
    # Wait, for constant h=0.5, correlate(h, 1-h) at lag d: (n-|d|)*0.5*0.5 = (n-|d|)/4
    # max at d=0: n/4, times dx = 2/n gives (n/4)*(2/n) = 0.5
    # So exact_score should give 0.5
    if best_score is not None and abs(best_score - 0.5) > 0.1:
        best_score = 0.5
    archive = (best_h.copy(), best_score)
    if iv is not None and n is not None:
        try:
            iv_arr = np.asarray(iv, dtype=np.float64)
            if iv_arr.ndim == 1 and len(iv_arr) == n and np.all(np.isfinite(iv_arr)):
                if np.all(iv_arr >= 0) and np.all(iv_arr <= 1) and abs(np.sum(iv_arr) - S) <= 1e-6:
                    parent = iv_arr.copy()
                else:
                    parent = project_capped_simplex(iv_arr, S)
                if parent is not None and is_feasible(parent, n):
                    ps = exact_score(parent, n)
                    if ps is not None and ps < best_score:
                        best_score = ps
                        best_h = parent.copy()
                        archive = (best_h.copy(), best_score)
        except Exception:
            pass
    starts = build_starts(n, S, parent, rng)
    if n <= 256:
        max_att = 8 if n <= 128 else 5
        for s in starts:
            sc = exact_score(s, n)
            if sc is not None and sc < archive[1]:
                archive = (s.copy(), sc)
            archive = optimize_dc(s, n, S, dx, archive, max_att)
        # Additional restart
        if n <= 128:
            restart = archive[0] + 0.03 * rng.standard_normal(n)
            rp = project_capped_simplex(restart, S)
            if rp is not None and is_feasible(rp, n):
                dup = False
                for s in starts:
                    if np.max(np.abs(rp - s)) < 1e-9:
                        dup = True
                        break
                if not dup:
                    archive = optimize_dc(rp, n, S, dx, archive, 8)
    final_h, final_score = archive
    if not is_feasible(final_h, n):
        final_h = project_capped_simplex(final_h, S)
        if final_h is None:
            final_h = const.copy()
    fs = exact_score(final_h, n)
    if fs is None:
        if parent is not None:
            final_h = parent.copy()
            fs = exact_score(final_h, n)
        if fs is None:
            final_h = const.copy()
            fs = 0.5
    return final_h, fs, n

import numpy as np
import cvxpy as cp
import math

def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):
        return None
    if np.max(np.abs(z)) > 1e9:
        return None
    n = len(z)
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    x = np.clip(z - lam, 0.0, 1.0)
    r = S - np.sum(x)
    for i in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            a = min(r, 1.0 - x[i])
            x[i] += a; r -= a
        else:
            a = min(-r, x[i])
            x[i] -= a; r += a
    if abs(np.sum(x) - S) > 1e-10 or not np.all(np.isfinite(x)):
        return None
    return np.clip(x, 0.0, 1.0)

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-10)

def exact_score(h, n):
    try:
        vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if vals.size != 2 * n - 1 or not np.all(np.isfinite(vals)):
            return None
        return float(np.max(vals))
    except Exception:
        return None

def solve_dc(anchor, n, S, dx, rho, eta):
    y = cp.Variable(n)
    t = cp.Variable()
    cons = [y >= 0, y <= 1, cp.sum(y) == S, y - anchor <= rho, anchor - y <= rho]
    for d in range(-(n - 1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        y0 = y[lo:hi]
        y1 = y[lo + d:hi + d]
        s = anchor[lo:hi] + anchor[lo + d:hi + d]
        lhs = dx * (cp.sum(y0) + 0.25 * cp.sum_squares(y0 - y1)
                    - 0.5 * cp.sum(s * (y0 + y1)) + 0.25 * float(np.dot(s, s)))
        cons.append(lhs <= t)
    obj = cp.Minimize(t + 0.5 * eta * cp.sum_squares(y - anchor))
    prob = cp.Problem(obj, cons)
    if not prob.is_dcp():
        return None
    for solver in [cp.ECOS, cp.CVXOPT, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                if y.value is not None and np.all(np.isfinite(y.value)):
                    return np.asarray(y.value, dtype=np.float64)
        except Exception:
            pass
    return None

def optimize_dc(x, n, S, dx, archive, max_att):
    x = x.copy()
    f = exact_score(x, n)
    if f is None:
        return archive
    if f < archive[1]:
        archive = (x.copy(), f)
    rho, eta = 0.20, 0.20 * dx
    rho_min, eta_min = 0.005, 0.02 * dx
    fail_cnt, noimp_cnt, small_cnt = 0, 0, 0
    for _ in range(max_att):
        cand = solve_dc(x, n, S, dx, rho, eta)
        if cand is None:
            rho = max(rho_min, rho * 0.5)
            fail_cnt += 1
            if fail_cnt >= 2 or rho <= rho_min:
                break
            continue
        proj = project_capped_simplex(cand, S)
        if proj is None:
            rho = max(rho_min, rho * 0.5)
            fail_cnt += 1
            if fail_cnt >= 2 or rho <= rho_min:
                break
            continue
        best_trial, best_f = None, f
        for j in range(7):
            alpha = 2.0 ** (-j)
            trial = x + alpha * (proj - x)
            tp = project_capped_simplex(trial, S)
            if tp is None:
                continue
            sc = exact_score(tp, n)
            if sc is not None and sc < best_f - 1e-12:
                best_f, best_trial = sc, tp.copy()
        if best_trial is None:
            rho = max(rho_min, rho * 0.5)
            eta = max(eta_min, eta * 0.5)
            noimp_cnt += 1
            if noimp_cnt >= 3 or (rho <= rho_min and eta <= eta_min):
                break
            continue
        decrease = f - best_f
        disp = np.max(np.abs(best_trial - x))
        x, f = best_trial, best_f
        if f < archive[1]:
            archive = (x.copy(), f)
        fail_cnt, noimp_cnt = 0, 0
        if decrease <= 1e-10:
            small_cnt += 1
        else:
            small_cnt = 0
        if small_cnt >= 2 or disp <= 1e-7:
            break
        if disp >= 0.8 * rho and decrease > 1e-5:
            rho = min(0.40, 1.25 * rho)
    return archive

def build_starts(n, S, parent, rng):
    starts = []
    const = np.full(n, 0.5)
    starts.append(const)
    if parent is not None:
        starts.append(parent.copy())
        starts.append((1.0 - parent).copy())
        starts.append(parent[::-1].copy())
        pert = parent + rng.normal(0, 0.06, n)
        p = project_capped_simplex(pert, S)
        if p is not None:
            starts.append(p)
        pert2 = parent + rng.normal(0, 0.15, n)
        p2 = project_capped_simplex(pert2, S)
        if p2 is not None:
            starts.append(p2)
    M = min(5, (n - 1) // 2)
    idx = np.arange(n, dtype=np.float64) + 0.5
    vec = np.full(n, 0.5)
    for m in range(1, M + 1):
        amp = (0.12 / np.sqrt(m)) * rng.standard_normal()
        phi = rng.uniform(0, 2 * np.pi)
        vec += amp * np.cos(2 * np.pi * m * idx / n + phi)
    lf = project_capped_simplex(vec, S)
    if lf is not None:
        starts.append(lf)
    if parent is None:
        perm = rng.permutation(n)
        bv = np.zeros(n)
        bv[perm[:n // 2]] = 1.0
        if n % 2 == 1:
            bv[perm[n // 2]] = 0.5
        starts.append(bv.copy())
    deduped = []
    for s in starts:
        arr = np.asarray(s, dtype=np.float64)
        dup = any(np.max(np.abs(arr - d)) < 1e-9 for d in deduped)
        if not dup and is_feasible(arr, n):
            deduped.append(arr)
    return deduped
