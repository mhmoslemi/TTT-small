        res = minimize(_slsqp_obj, z0, method='SLSQP', jac=_slsqp_obj_jac,
                       args=(n,), bounds=bounds, cons=cons,
                       options={'maxiter': 350, 'ftol': 1e-13})
    except Exception:
        return None, None
    if res.x is None or res.x.shape != (n + 1,) or not np.all(np.isfinite(res.x)):
        return None, None
    h_res = project_capped(res.x[:n], n)
    if h_res is None:
        return None, None
    return h_res, float(exact_score(h_res, n))

def polish(h0, n, n_passes):
    h = h0.copy()
    score = exact_score(h, n)
    for _ in range(n_passes):
        h_res, s_res = run_epigraph(h, n)
        if h_res is None or s_res is None:
            break
        best_h, best_s = h, score
        for k in range(6):
            alpha = 1.0 / (2.0 ** k)
            z = project_capped(h + alpha * (h_res - h), n)
            if z is None:
                continue
            sz = exact_score(z, n)
            if sz < best_s - 1e-13:
                best_s, best_h = sz, z.copy()
        if best_s < score - 1e-13:
            h, score = best_h, best_s
        else:
            break
    return h, score

def _add_start(pool, v, n):
    if v is None:
        return
    for s in pool:
        if np.max(np.abs(v - s)) < 1e-10:
            return
    pool.append(v)

def build_starts(parent, n, rng, count):
    pool = []
    _add_start(pool, parent, n)
    _add_start(pool, project_capped(parent[::-1], n), n)
    _add_start(pool, project_capped(1.0 - parent[::-1], n), n)
    _add_start(pool, np.full(n, 0.5), n)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    _add_start(pool, blk, n)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    _add_start(pool, project_capped(alt, n), n)
    _add_start(pool, project_capped(parent + rng.normal(0, 0.05, n), n), n)
    _add_start(pool, project_capped(parent + rng.normal(0, 0.12, n), n), n)
    _add_start(pool, project_capped(parent + rng.normal(0, 0.25, n), n), n)
    _add_start(pool, project_capped(rng.uniform(0, 1, n), n), n)
    return pool[:count]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if parent is not None and parent.size == n:
        fc = np.clip(parent, 0.0, 1.0)
        fallback = project_capped(fc, n)
        if fallback is None:
            fallback = np.full(n, 0.5)
    else:
        fallback = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        return fallback, float(exact_score(fallback, n)), 1
    n_starts = min(10, max(4, n // 5))
    starts = build_starts(fallback, n, rng, n_starts)
    gb, gs = fallback.copy(), exact_score(fallback, n)
    for st in starts:
        s = exact_score(st, n)
        if s < gs - 1e-13:
            gs, gb = s, st.copy()
    if n <= 64:
        npass = 3
        mit = 350
    elif n <= 128:
        npass = 2
        mit = 250
    else:
        npass = 1
        mit = 150
    for st in starts:
        try:
            ho, so = polish(st, n, npass)
            if ho is not None and so < gs - 1e-13:
                gs, gb = so, ho.copy()
        except Exception:
            continue
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    if fs > exact_score(fallback, n) + 1e-13:
        final = fallback
        fs = exact_score(final, n)
    return final, float(fs), int(n)
