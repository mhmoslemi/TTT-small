def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:

        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)

def pair_delta(h, p, q, a, n):
    # Compute DeltaQ for all lags k=-(n-1)..n-1
    n_lags = 2 * n - 1
    DQ = np.zeros(n_lags)
    for k in range(-(n - 1), n):
        idx = k + n - 1
        dk = 0.0
        nm1 = p - k
        if 0 <= nm1 < n:
            dk += a * (1.0 - h[nm1])
        nm2 = q - k
        if 0 <= nm2 < n:
            dk -= a * (1.0 - h[nm2])
        nm3 = p + k
        if 0 <= nm3 < n:
            dk -= a * h[nm3]
        nm4 = q + k
        if 0 <= nm4 < n:
            dk += a * h[nm4]
        if k == p - q:
            dk -= a * a
        if k == q - p:
            dk -= a * a
        DQ[idx] = dk
    return DQ

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-14:
        if resid > 0:
            order = np.argsort(1.0 - y)
            for i in order:
                add = min(resid, 1.0 - y[i])
                y[i] += add
                resid -= add
                if resid < 1e-15:
                    break
        else:
            order = np.argsort(y)
            for i in order:
                sub = min(-resid, y[i])
                y[i] -= sub
                resid += sub
                if -resid < 1e-15:
                    break
    y = np.clip(y, 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 5e-12 * max(1.0, float(n)):
        return None
    if resid > 0 and y[0] < 1.0:
        y[0] += min(resid, 1.0 - y[0])
    elif resid < 0 and y[0] > 0.0:

            if not math.isfinite(C_t) or C_t >= C_cur - 1e-13:
                continue
            h[p] += a
            h[q] -= a
            lags = lags_t
            C_cur = C_t
            if C_cur < C_best:
                C_best = C_cur
                h_best = h.copy()
        lags = np.correlate(h, 1.0 - h, mode='full')
        C_cur = float(np.max(lags) * (2.0 / n))
        sc = exact_score(h, n)
        if sc < C_best:
            C_best = sc
            h_best = h.copy()
    return h_best, C_best

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    n = 45
    parent = None
    try:
        pv = np.array(initial_h_values, dtype=np.float64)
        if pv.ndim == 1 and 2 <= len(pv) <= 999 and np.all(np.isfinite(pv)):
            if np.all(pv >= -1e-9) and np.all(pv <= 1.0 + 1e-9):
                n = len(pv)
                tgt = n / 2.0
                if abs(np.sum(pv) - tgt) < 1e-6 * max(1.0, float(n)):
                    pv = np.clip(pv, 0.0, 1.0)
                    parent = project_box_sum(pv, tgt)
    except Exception:
        pass
    S = n / 2.0
    fallback = project_box_sum(np.full(n, 0.5), S)
    if parent is None or not feasible(parent, n, S):
        parent = fallback
    starts = build_starts(parent, n, S, rng)
    if not starts:
        starts = [fallback]
    best_h = starts[0].copy()
    best_sc = exact_score(best_h, n)
    for s in starts[1:]:
        sc = exact_score(s, n)
        if sc < best_sc:
            best_sc = sc
            best_h = s.copy()
    fallback_sc = exact_score(fallback, n)
    B_total = max(600, min(12000, 540000 // n))
    mps = max(60, B_total // len(starts))
    for s in starts:
        try:
            h_new, sc_new = anneal_start(s, n, S, rng, mps)
            if h_new is not None and sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_h = h_new
        except Exception:
            pass
    try:
        h_pol, sc_pol = greedy_polish(best_h, n, S, rng)
        if sc_pol < best_sc - 1e-14:
            best_sc = sc_pol
            best_h = h_pol
    except Exception:
        pass
    final_h = project_box_sum(best_h, S)
    if final_h is None or not feasible(final_h, n, S):
        final_h = fallback
        final_sc = fallback_sc
    else:
        final_sc = exact_score(final_h, n)
        if not math.isfinite(final_sc) or final_sc > fallback_sc:
            final_h = fallback
            final_sc = fallback_sc
    return (np.array(final_h, dtype=np.float64), final_sc, n)
