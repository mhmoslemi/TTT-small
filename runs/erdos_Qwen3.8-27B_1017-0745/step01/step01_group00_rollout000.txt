Let me analyze the problem and the strategy carefully.

The Erdős minimum overlap problem asks us to find a step function h: [0,2] → [0,1] with ∫h = 1 that minimizes the maximum overlap integral C₅ = max_k ∫ h(x)(1-h(x+k)) dx.

The previous program achieved C₅ ≤ 0.380993. The target is 0.3808 or lower.

The strategy suggests using a curvature-majorization (MM) approach where:
1. At current point x, form a convex upper bound on each lag
2. Minimize the convex epigraph (QP) using SLSQP for small n
3. Use matrix-free projected subgradient for larger n
4. Accept only based on exact score

Key mathematical insight: For each lag d, Q_d(y) = Q_d(x) + g_d^T(y-x) + 0.5*u^T H_d u, and since the largest eigenvalue of H_d is ≤ 2*dx (bound -dx off-diagonal + -2dx diagonal), we have:
Q_d(y) ≤ Q_d(x) + g_d^T(y-x) + dx*||y-x||²

So the convex majorant is: min t + beta*dx*||y-x||²  s.t. t ≥ Q_d(x) + g_d^T(y-x) for all d, y ∈ K.

import numpy as np
from scipy.optimize import minimize

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def project_capped(z, n):
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for i in range(n + 2):
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    return h

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr) * (2.0 / n))

def lag_model(x, n):
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.zeros(L)
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        p0 = max(d, 0)
        q0 = max(-d, 0)
        m = n - abs(d)
        p = x[p0:p0 + m]
        qv = x[q0:q0 + m]
        q[row] = dx * np.sum(p * (1.0 - qv))
        G[row, p0:p0 + m] += dx * (1.0 - qv)
        G[row, q0:q0 + m] -= dx * p
    return q, G

def slsqp_obj(z, x, q, G, dx, beta, n):
    y = z[:n]
    t = z[n]
    u = y - x
    return t + beta * dx * np.dot(u, u)

def slsqp_grad(z, x, dx, beta, n):
    y = z[:n]
    u = y - x
    g = np.zeros(n + 1)
    g[:n] = 2.0 * beta * dx * u
    g[n] = 1.0
    return g

def slsqp_ineq(z, x, q, G, n):
    y = z[:n]
    t = z[n]
    u = y - x
    return t - q - G @ u

def slsqp_ineq_jac(z, x, q, G, n):
    # Jacobian of ineq w.r.t. z = [y, t]
    # d/dy: -G, d/dt: 1
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def slsqp_eq(z, n):
    return np.array([np.sum(z[:n]) - n / 2.0])

def slsqp_eq_jac(z, n):
    J = np.zeros((1, n + 1))
    J[0, :n] = 1.0
    return J

def matrix_free_subgrad(x, n, beta, dx, n_inner):
    # Minimize max_d [Q_d(x) + g_d^T(y-x)] + beta*dx*||y-x||^2
    # via projected subgradient
    y = x.copy().astype(np.float64)
    best_y = x.copy().astype(np.float64)
    best_model = float('inf')
    s = n / 2.0
    for k in range(n_inner):
        u = y - x
        # Compute all lag affine values
        Amax = -np.inf
        best_d = 0
        best_sli = None
        for d in range(-(n - 1), n):
            p0 = max(d, 0)
            q0 = max(-d, 0)
            m = n - abs(d)
            p = x[p0:p0 + m]
            qv = x[q0:q0 + m]
            Ap = dx * (np.dot(p, 1.0 - qv) + np.dot(1.0 - qv, u[p0:p0+m]) - np.dot(p, u[q0:q0+m]))
            if Ap > Amax:
                Amax = Ap
                best_d = d
                best_sli = (p0, q0, m)
        model_val = Amax + beta * dx * np.dot(u, u)
        if model_val < best_model:
            best_model = model_val
            best_y = y.copy()
        # Subgradient of active lag + proximal term
        p0, q0, m = best_sli
        g = np.zeros(n)
        g[p0:p0+m] += dx * (1.0 - x[q0:q0+m])
        g[q0:q0+m] -= dx * x[p0:p0+m]
        v = g + 2.0 * beta * dx * u
        nv = np.linalg.norm(v)
        if not np.isfinite(nv):
            break
        eta = min(0.5 / (nv + 1e-12), 1.0 / (2.0 * beta * dx * np.sqrt(k + 1)))
        y_new = project_capped(y - eta * v, n)
        if y_new is None or not np.all(np.isfinite(y_new)):
            break
        if np.max(np.abs(y_new - y)) < 1e-9:
            break
        y = y_new
    return best_y

def optimize_start(start, n, n_outer, n_inner, slsqp_maxiter):

    score = exact_score(x, n)
    if np.isinf(score):

    dx = 2.0 / n
    beta = 1.0
    rejs = 0
    alphas = [1.0, 0.5, 0.25, 0.125, 0.0625]

        # Get proposal from convex majorizer
        y_prop = None
        if n <= 128:
            q, G = lag_model(x, n)
            if q is None or not np.all(np.isfinite(q)):
                break
            z0 = np.zeros(n + 1)
            z0[:n] = x
            z0[n] = np.max(q) + 1e-6
            cons = [
                {'type': 'ineq', 'fun': lambda z: slsqp_ineq(z, x, q, G, n),
                 'jac': lambda z: slsqp_ineq_jac(z, x, q, G, n)},
                {'type': 'eq', 'fun': lambda z: slsqp_eq(z, n),
                 'jac': lambda z: slsqp_eq_jac(z, n)},
            ]
            bounds = [(0, 1)] * n + [(None, None)]
            try:
                res = minimize(slsqp_obj, z0, args=(x, q, G, dx, beta, n),
                               jac=lambda z: slsqp_grad(z, x, dx, beta, n),
                               bounds=bounds, constraints=cons,
                               method='SLSQP',
                               options={'maxiter': slsqp_maxiter, 'ftol': 1e-10})
                if res.x is not None and res.x.size == n + 1 and np.all(np.isfinite(res.x)):
                    y_prop = project_capped(res.x[:n], n)
            except Exception:
                y_prop = None
        else:
            y_prop = matrix_free_subgrad(x, n, beta, dx, n_inner)
        
        if y_prop is None or not is_feasible(y_prop, n) or np.max(np.abs(y_prop - x)) < 1e-9:
            beta = min(64.0, 2.0 * beta)
            rejs += 1
            continue
        
        # Line search
        best_z, best_s = None, score
        for a in alphas:
            z = project_capped(x + a * (y_prop - x), n)
            if z is None or not is_feasible(z, n):
                continue
            sz = exact_score(z, n)
            if sz < best_s:
                best_s = sz
                best_z = z
        
        if best_z is not None and best_s < score - 1e-12:
            x = best_z.copy()
            score = best_s
            rejs = 0
            beta = max(1.0, beta / 1.5)
        else:
            beta = min(64.0, 2.0 * beta)
            rejs += 1
    return x, score
