- When i+d = p (i.e., i = p-d): h_δ(p-d) * (1 - h_δ(p)) if p-d valid

- When i+d = q (i.e., i = q-d): h_δ(q-d) * (1 - h_δ(q)) if q-d valid

Let me define D_d(k) = [k+d in [0,n)] * (1-h(k+d)) - [k-d in [0,n)] * h(k-d)

Then the linear coefficient for coordinate k changing by δ is: b_d(k) = D_d(k)

For pair (p,q) with v_p=1, v_q=-1:
b_d = D_d(p) - D_d(q)

And the quadratic coefficient:
a_d = -sum_i v_i * v_{i+d}
For d=0: a_0 = -v_p*v_p - v_q*v_q = -1-1 = -2
For d=Δ: a_Δ = -v_p*v_{p+Δ} = -v_p*v_q = -(-1) = 1 (since p+Δ=q)
For d=-Δ: a_{-Δ} = -v_q*v_{q-Δ} = -v_q*v_p = -(-1) = 1
For all other d: a_d = 0

So the three quadratic lags are 0, Δ, and -Δ.

        row = d + n - 1
        i0, j0, m = max(0, -d), max(0, d), n - abs(d)
        c[row] = float(np.sum(h[i0:i0+m] * (1.0 - h[j0:j0+m])))
    return c if np.all(np.isfinite(c)) else None

def quad_roots(A, B, C, lo, hi):
    t = 1e-14
    if abs(A) <= t * max(1.0, abs(A), abs(B), abs(C)):
        if abs(B) <= t * max(1.0, abs(B), abs(C)):
            return []
        r = -C / B
        return [float(np.clip(r, lo, hi))] if lo - 1e-12 <= r <= hi + 1e-12 else []
    disc = B * B - 4.0 * A * C
    if disc < -t * (B * B + 4.0 * abs(A * C) + 1.0):
        return []
    disc = max(0.0, disc)
    sq = math.sqrt(disc)
    q = -0.5 * (B + math.copysign(sq, B))
    r1, r2 = (q / A, C / q) if abs(q) > t else (-B / (2.0 * A), -B / (2.0 * A))
    out = []
    for r in (r1, r2):
        if np.isfinite(r) and lo - 1e-12 <= r <= hi + 1e-12:
            out.append(float(np.clip(r, lo, hi)))
    return out

def line_min(h, p, q, n, c):
    L = 2 * n - 1
    lo = max(-h[p], h[q] - 1.0)
    hi = min(1.0 - h[p], h[q])
    if hi - lo <= 1e-14:
        return 0.0
    D = q - p
    a = np.zeros(L)
    a[n-1] = -2.0
    a[n-1+D] = 1.0
    a[n-1-D] = 1.0
    b = np.zeros(L)
    for d in range(-(n-1), n):
        row = d + n - 1
        dp = (1.0 - h[p+d]) if 0 <= p+d < n else 0.0
        dp -= h[p-d] if 0 <= p-d < n else 0.0
        dq = (1.0 - h[q+d]) if 0 <= q+d < n else 0.0
        dq -= h[q-d] if 0 <= q-d < n else 0.0
        b[row] = dp - dq
    if not np.all(np.isfinite(b)):
        return 0.0
    special = {n-1, n-1+D, n-1-D}
    aff = [(b[r], c[r]) for r in range(L) if r not in special]
    aff.sort(key=lambda t: (t[0], -t[1]))
    dedup = []
    for sl, ic in aff:
        if dedup and abs(dedup[-1][0] - sl) < 1e-15:
            if ic > dedup[-1][1]:
                dedup[-1] = (sl, ic)
        else:
            dedup.append((sl, ic))
    hull = []
    for sl, ic in dedup:
        if not hull:
            hull.append([sl, ic, -1e30])
            continue
        while True:
            dsl = sl - hull[-1][0]
            if abs(dsl) < 1e-15:
                if ic > hull[-1][1]:
                    hull[-1][1] = ic
                break
            cx = (hull[-1][1] - ic) / dsl
            if len(hull) == 1 or cx > hull[-1][2] + 1e-15:
                hull.append([sl, ic, cx])
                break
            hull.pop()
            if not hull:
                hull.append([sl, ic, -1e30])
                break
    cands = [lo, hi, 0.0]
    cands.extend(h[2] for h in hull)
    for idx in range(len(hull)):
        s_lo = max(hull[idx][2], lo)
        s_hi = hull[idx+1][2] if idx + 1 < len(hull) else 1e30
        s_hi = min(s_hi, hi)
        if s_lo >= s_hi:
            continue
        ma, ca = hull[idx][0], hull[idx][1]
        for s in (n-1, n-1+D, n-1-D):
            cands.extend(quad_roots(a[s], b[s]-ma, c[s]-ca, s_lo, s_hi))
        sp = [n-1, n-1+D, n-1-D]
        for i in range(3):
            for j in range(i+1, 3):
                cands.extend(quad_roots(a[sp[i]]-a[sp[j]], b[sp[i]]-b[sp[j]], c[sp[i]]-c[sp[j]], s_lo, s_hi))
            if a[sp[i]] > 0:
                vx = -b[sp[i]] / (2.0 * a[sp[i]])
                if s_lo - 1e-12 <= vx <= s_hi + 1e-12:
                    cands.append(float(np.clip(vx, s_lo, s_hi)))
    cands = [float(np.clip(x, lo, hi)) for x in cands if np.isfinite(x)]
    if len(cands) > 1:
        cs = sorted(cands)
        u = [cs[0]]
        for v in cs[1:]:
            if v - u[-1] > 1e-10:
                u.append(v)
        cands = u
    bd, bv = 0.0, float('inf')
    for d in cands:
        v = float(np.max(c + b * d + a * d * d))
        if v < bv:
            bv, bd = v, d
    return bd if np.isfinite(bd) else 0.0

def make_edges(n):
    e = [(i, i+1) for i in range(n-1)]
    r = 2
    while r < n:
        i = 0
        while i + r < n:
            if (i, i+r) not in e:
                e.append((i, i+r))
            i += r
        r *= 2
    if n > 2 and (0, n-1) not in e:
        e.append((0, n-1))
    return e

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 0 < len(iv) <= 999:
            n = len(iv)
            if np.all(np.isfinite(iv)) and np.all(iv >= -1e-8) and np.all(iv <= 1+1e-8):
                parent = iv.copy()
    except Exception:
        pass
    if n is None:
        try: n = int(kwargs.get('n_points', 47))
        except Exception: n = 47
    n = max(1, min(n, 999))
    try: sd = int(seed)
    except Exception: sd = 42
    rng = np.random.default_rng(sd % (2**32))
    uniform = np.full(n, 0.5)
    if n <= 64: ns_def, sw_def = 6, 3
    elif n <= 96: ns_def, sw_def = 4, 2
    else: ns_def, sw_def = 4, 0
    try: ns = max(1, min(int(kwargs.get('starts', ns_def)), 10))
    except Exception: ns = ns_def
    try: swp = max(0, min(int(kwargs.get('max_sweeps', sw_def)), 4))
    except Exception: swp = sw_def
    raw = []
    if parent is not None:
        raw.extend([parent, parent[::-1], 1.0-parent, 1.0-parent[::-1],
                    parent+rng.normal(0,0.04,n), parent+rng.normal(0,0.12,n),
                    np.roll(parent, n//3)])
    raw.append(uniform)
    blk = np.zeros(n); blk[:n//2] = 1.0
    if n % 2 == 1: blk[n//2] = 0.5
    raw.append(blk)
    raw.append(0.5 + 0.45 * np.cos(2.0 * np.pi * np.arange(n) / n))
    raw.append(0.5 + 0.35 * np.cos(4.0 * np.pi * np.arange(n) / n))
    raw.append(rng.uniform(0, 1, n))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    raw.append(alt)
    feasible, parent_proj = [], None
    for v in raw:
        h2 = project_capped(v, n)
        if h2 is None: continue
        if any(float(np.max(np.abs(h2 - e))) < 1e-9 for e in feasible): continue
        if parent is not None and parent_proj is None and float(np.max(np.abs(h2 - parent))) < 1e-6:
            parent_proj = h2.copy()
        feasible.append(h2)
    best, best_sc = uniform.copy(), exact_score(uniform, n)
    for fs in feasible:
        sc = exact_score(fs, n)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best, best_sc = fs.copy(), sc
    edges = make_edges(n)
    for st in feasible[:ns]:
        x, xs = st.copy(), exact_score(st, n)
        for sweep in range(swp):
            if sweep > 0:
                rng.shuffle(edges)
            moved = False
            for (p, q) in edges:
                c = lag_raw(x, n)
                if c is None: continue
                delta = line_min(x, p, q, n, c)
                if abs(delta) < 1e-14: continue
                bt, bts = None, xs
                for k in range(7):
                    dt = delta * (0.5 ** k)
                    if abs(dt) < 1e-14: break
                    y = x.copy(); y[p] += dt; y[q] -= dt
                    y = project_capped(y, n)
                    if y is None: continue
                    sc = exact_score(y, n)
                    if sc < bts - 1e-12:
                        bt, bts = y, sc
                if bt is not None:
                    x, xs = bt, bts
                    moved = True
                    if xs < best_sc - 1e-12:
                        best, best_sc = x.copy(), xs
            if not moved: break
    cands_fin = [best]
    if parent_proj is not None: cands_fin.append(parent_proj)
    cands_fin.append(uniform)
    fb = None; fbs = float('inf')
    for cv in cands_fin:
        sc = exact_score(cv, n)
        if np.isfinite(sc) and sc < fbs:
            fb, fbs = cv.copy(), sc
    if fb is None: fb, fbs = uniform.copy(), exact_score(uniform, n)
    fp = project_capped(fb, n)
    if fp is not None:
        fsp = exact_score(fp, n)
        if fsp <= fbs + 1e-12:
            fb, fbs = fp, fsp
    if not np.isfinite(fbs):
        fb, fbs = uniform.copy(), exact_score(uniform, n)
    return fb, float(fbs), int(n)
