```python
def compute_B(h, i, j, n, dx):
    """Compute B_d = grad_d[i] - grad_d[j] for all d."""
    B = np.zeros(2*n-1)
    for d in range(-(n-1), n):
        a = max(d, 0)
        b = max(-d, 0)
        m = n - abs(d)
        # grad at i
        gi = 0.0
        si = i - a
        if 0 <= si < m:
            gi += 1.0 - h[si + b]
        si2 = i - b
        if 0 <= si2 < m:
            gi -= h[si2 + a]
        # grad at j
        gj = 0.0
        sj = j - a
        if 0 <= sj < m:
            gj += 1.0 - h[sj + b]
        sj2 = j - b
        if 0 <= sj2 < m:
            gj -= h[sj2 + a]
        B[d + n - 1] = dx * (gi - gj)
    return B
```

```python
    z = np.clip(z, -1e12, 1e12)

            continue
        nh, ns = local_descent(pp, n, rng, moves_per_hop, 4, 2)

        accept = delta <= 1e-12

        final = fallback
        fs = exact_score(final, n)
    return final, float(fs), int(n)
```
