def smooth_obj_grad(h, tau, N):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2N-1
    c = dx * corr  # these are the C values at each lag
    
    # log-sum-exp for stability
    cmax = np.max(c)
    e = np.exp((c - cmax) / tau)
    F = cmax + tau * np.log(np.sum(e))
    p = e / np.sum(e)
    
    # gradient
    grad = np.zeros(N)
    for k_idx in range(2 * N - 1):
        k = k_idx - (N - 1)  # actual lag
        w = dx * p[k_idx]
        if w < 1e-15:
            continue
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            d = k
            # (1-h[j-d]) for j in [d, N-1]
            grad[d:N] += w * (1.0 - h[0:N-d])
            # -h[j+d] for j in [0, N-1-d]
            grad[0:N-d] -= w * h[d:N]
        else:
            d = -k
            # (1-h[j+d]) for j in [0, N-1-d]
            grad[0:N-d] += w * (1.0 - h[d:N])
            # -h[j-d] for j in [d, N-1]
            grad[d:N] -= w * h[0:N-d]
    
    return F, grad

import numpy as np

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    if r > 1e-14:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
            if r <= 1e-14:
                break
    elif r < -1e-14:
        for i in range(N):
            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1 + 1e-10:
        return np.full(N, 0.5)
    if abs(h.sum() - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)

import numpy as np

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    if r > 1e-14:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
            if r <= 1e-14:
                break
    elif r < -1e-14:
        for i in range(N):
            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1 + 1e-10:
        return np.full(N, 0.5)
    if abs(h.sum() - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / N)

def smooth_obj_grad(h, tau, N):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = dx * corr
    cmax = np.max(c)
    e = np.exp((c - cmax) / tau)
    ss = np.sum(e)
    F = cmax + tau * np.log(ss)
    p = e / ss
    grad = np.zeros(N)
    for k_idx in range(2 * N - 1):
        k = k_idx - (N - 1)
        w = dx * p[k_idx]
        if w < 1e-15:
            continue
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            d = k
            grad[d:N] += w * (1.0 - h[0:N - d])
            grad[0:N - d] -= w * h[d:N]
        else:
            d = -k
            grad[0:N - d] += w * (1.0 - h[d:N])
            grad[d:N] -= w * h[0:N - d]
    return F, grad

def optimize_pgd(h0, N, m, tau_sched, max_it):
    h = h0.copy()
    best_local = h.copy()
    best_local_score = score_exact(h, N)
    for tau in tau_sched:
        alpha = 1.0
        for it in range(max_it):
            F, grad = smooth_obj_grad(h, tau, N)
            if not np.all(np.isfinite(grad)):
                break
            y = project_box_sum(h - alpha * grad, m)
            dy = y - h
            if not np.all(np.isfinite(dy)):
                break
            if np.max(np.abs(dy)) < 1e-10:
                s_y = score_exact(y, N)
                if s_y < best_local_score:
                    best_local = y.copy()
                    best_local_score = s_y
                break
            F_y = smooth_obj_grad(y, tau, N)[0]
            gd = np.dot(grad, dy)
            if gd < 0 and F_y <= F + 1e-4 * gd:
                h = y
                alpha = min(1.3 * alpha, 10.0)
                s_h = score_exact(h, N)
                if s_h < best_local_score:
                    best_local = h.copy()
                    best_local_score = s_h
            else:
                alpha *= 0.5
                if alpha < 1e-9:
                    s_h = score_exact(h, N)
                    if s_h < best_local_score:
                        best_local = h.copy()
                        best_local_score = s_h
                    break
    s_fin = score_exact(h, N)
    if s_fin < best_local_score:
        best_local = h.copy()
        best_local_score = s_fin
    return best_local

def make_fourier(N, rng, n_modes, amp):
    z = np.full(N, 0.5)
    idx = np.arange(N)
    ns = min(n_modes, (N - 1) // 2)
    for r in range(1, ns + 1):
        A = rng.standard_normal()
        B = rng.standard_normal()
        z += (amp / r) * (A * np.sin(2.0 * np.pi * r * idx / N) + B * np.cos(2.0 * np.pi * r * idx / N))
    return z

def make_block(N, m, rng):
    n_blocks = rng.integers(3, 8)
    z = np.zeros(N)
    widths = rng.dirichlet(np.ones(n_blocks) * 2.0) * N
    pos = 0.0
    for i in range(n_blocks):
        w = int(round(widths[i]))
        if i % 2 == 0:
            z[int(pos):int(pos) + w] = 1.0
        else:
            z[int(pos):int(pos) + w] = 0.0
        pos += w
    return z

def run(seed=42, **kwargs):
    raw = globals().get('initial_h_values', None)
    N = None
    if raw is not None:
        raw = np.asarray(raw, dtype=np.float64)
        if raw.ndim == 1 and raw.size > 1 and np.all(np.isfinite(raw)):
            N = raw.size
    if N is None:
        N = kwargs.get('n_points', 70)
        if not isinstance(N, (int, np.integer)) or N < 2 or N > 999:
            N = 70
    m = N / 2.0

    if raw is not None and raw.ndim == 1 and raw.size == N and np.all(np.isfinite(raw)):
        baseline = project_box_sum(raw.copy(), m)
    else:
        baseline = np.full(N, 0.5)

    best = baseline.copy()
    best_score = score_exact(best, N)
    rng = np.random.default_rng(seed)

    starts = [baseline.copy()]
    comp = project_box_sum(1.0 - baseline, m)
    if np.max(np.abs(comp - baseline)) > 1e-10:
        starts.append(comp)
    for sc in [0.04, 0.12, 0.25]:
        starts.append(project_box_sum(baseline + sc * rng.standard_normal(N), m))
    starts.append(project_box_sum(0.5 + 0.5 * rng.uniform(size=N), m))
    starts.append(project_box_sum(make_fourier(N, rng, 4, 0.12), m))
    starts.append(project_box_sum(make_fourier(N, rng, 6, 0.22), m))
    starts.append(project_box_sum(make_block(N, m, rng), m))
    starts.append(project_box_sum(make_block(N, m, rng), m))

    unique_starts = [starts[0]]
    for s in starts[1:]:
        if all(np.max(np.abs(s - u)) > 1e-8 for u in unique_starts):
            unique_starts.append(s)
    starts = unique_starts[:10]

    tau_sched = [0.02, 0.006, 0.002, 0.0006, 0.0002, 0.00006, 0.000015]
    max_it = 60

    for h0 in starts:
        s0 = score_exact(h0, N)
        if s0 < best_score:
            best = h0.copy()
            best_score = s0
        h_opt = optimize_pgd(h0, N, m, tau_sched, max_it)
        s_opt = score_exact(h_opt, N)
        if s_opt < best_score:
            best = h_opt.copy()
            best_score = s_opt

    for sc in [0.05, 0.02]:
        hop = project_box_sum(best + sc * rng.standard_normal(N), m)
        s_hop = score_exact(hop, N)
        if s_hop < best_score:
            best = hop.copy()
            best_score = s_hop
        h_opt = optimize_pgd(hop, N, m, tau_sched, max_it)
        s_opt = score_exact(h_opt, N)
        if s_opt < best_score:
            best = h_opt.copy()
            best_score = s_opt

    best = project_box_sum(best, m)
    c5 = score_exact(best, N)
    if not np.isfinite(c5):
        best = baseline.copy()
        c5 = score_exact(best, N)
    sb = score_exact(baseline, N)
    if sb < c5:
        best = baseline
        c5 = sb
    return best, float(c5), int(N)
