import numpy as np

def project_capped(z, n):
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = np.argmax(1.0 - h)
            if h[idx] >= 1.0 - 1e-14:
                break
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def dedup_add(pool, v, tol=1e-8):
    if v is None:
        return
    for s in pool:
        if np.max(np.abs(v - s)) < tol:
            return
    pool.append(v)

def build_pool(fallback, n, rng, pop_size):
    pool = [fallback.copy()]
    dedup_add(pool, project_capped(fallback[::-1], n))
    dedup_add(pool, project_capped(1.0 - fallback[::-1], n))
    dedup_add(pool, np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    dedup_add(pool, project_capped(blk, n))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    dedup_add(pool, project_capped(alt, n))
    perm = rng.permutation(n)
    bb = np.zeros(n)
    bb[perm[:n // 2]] = 1.0
    if n % 2:
        bb[perm[n // 2]] = 0.5
    dedup_add(pool, project_capped(bb, n))
    for sd in (0.08, 0.22):
        dedup_add(pool, project_capped(fallback + rng.normal(0, sd, n), n))
    for _ in range(2):
        dedup_add(pool, project_capped(rng.uniform(0, 1, n), n))
    for sh in (n // 3, n // 2):
        if sh > 0:
            dedup_add(pool, project_capped(np.roll(fallback, sh), n))
    scores = [exact_score(p, n) for p in pool]
    if len(pool) > pop_size:
        order = np.argsort(scores)
        keep = [order[0], order[1]]
        for idx in order[2:]:
            keep.append(idx)
            if len(keep) >= pop_size:
                break
        pool = [pool[i].copy() for i in keep]
        scores = [scores[i] for i in keep]
    while len(pool) < pop_size:
        if len(pool) > 0:
            base = pool[rng.integers(0, len(pool))]
            sig = np.sqrt(47.0 / n)
            sig = np.clip(sig, 0.5, 1.5)
            sig = min(0.32, 0.03 * (2 ** rng.integers(0, 3))) * sig
            v = project_capped(base + rng.normal(0, sig, n), n)
        else:
            v = project_capped(rng.uniform(0, 1, n), n)
        dedup_add(pool, v, tol=1e-10)
        if len(pool) >= pop_size + 10:
            break
    while len(pool) < pop_size:
        pool.append(pool[0].copy())
    return pool

def de_step(pop, scores, F_arr, CR_arr, n, rng, best_idx):
    M = len(pop)
    new_pop = [p.copy() for p in pop]
    new_scores = list(scores)
    new_F = list(F_arr)
    new_CR = list(CR_arr)
    improved = False
    for i in range(M):
        F_i = F_arr[i] if rng.random() > 0.1 else 0.35 + 0.55 * rng.random()
        CR_i = CR_arr[i] if rng.random() > 0.1 else 0.10 + 0.85 * rng.random()
        other = [j for j in range(M) if j != i]
        r1 = other[rng.integers(0, len(other))]
        r2 = other[rng.integers(0, len(other))]
        r3 = other[rng.integers(0, len(other))]
        if r2 == r3:
            r3 = (r3 + 1) % M
            if r3 == i:
                r3 = (r3 + 1) % M
        if rng.random() < 0.5:
            mutant = pop[r1] + F_i * (pop[r2] - pop[r3])
        else:
            mutant = pop[i] + F_i * (pop[best_idx] - pop[i] + pop[r2] - pop[r3])
        mask = rng.random(n) < CR_i
        fc = rng.integers(0, n)
        mask[fc] = True
        trial_raw = np.where(mask, mutant, pop[i])
        trial = project_capped(trial_raw, n)
        if trial is None:
            continue
        ts = exact_score(trial, n)
        if ts < new_scores[i] - 1e-12:
            new_pop[i] = trial
            new_scores[i] = ts
            new_F[i] = F_i
            new_CR[i] = CR_i
            improved = True
            if ts < new_scores[new_scores[0:scores[best_idx]].count(scores[best_idx])] if False else True:
                pass
    bi = int(np.argmin(new_scores))
    return new_pop, new_scores, new_F, new_CR, bi, improved

def pattern_poll(h, pop, scores, n, rng, rounds, ndirs):
    best = h.copy()
    best_s = exact_score(best, n)
    scale_n = np.clip(np.sqrt(47.0 / n), 0.5, 1.5)
    delta = min(0.25, 0.12 * scale_n)
    M = len(pop)
    for _ in range(rounds):
        if delta < 1e-4:
            break
        dirs = []
        for _ in range(ndirs):
            if len(pop) > 1 and rng.random() < 0.5:
                i = rng.integers(0, M)
                j = rng.integers(0, M)
                if i == j:
                    j = (j + 1) % M
                v = pop[i] - pop[j]
            else:
                v = rng.normal(0, 1, n)
            v = v - np.mean(v)
            mx = np.max(np.abs(v))
            if mx > 1e-12 and np.isfinite(mx):
                v = v / mx
                dirs.append(v)
        best_trial = None
        best_trial_s = best_s
        for v in dirs:
            t_lo, t_hi = -1e30, 1e30
            pos = v > 1e-12
            neg = v < -1e-12
            if np.any(pos):
                t_lo = max(t_lo, np.min(-best[pos] / v[pos]))
                t_hi = min(t_hi, np.min((1.0 - best[pos]) / v[pos]))
            if np.any(neg):
                t_lo = max(t_lo, np.min((1.0 - best[neg]) / v[neg]))
                t_hi = min(t_hi, np.min(-best[neg] / v[neg]))
            t_lo = max(t_lo, -delta)
            t_hi = min(t_hi, delta)
            if t_hi - t_lo < 1e-12:
                continue
            tvals = np.array([-delta, -0.5 * delta, 0.5 * delta, delta])
            tvals = tvals[(tvals > t_lo + 1e-12) & (tvals < t_hi - 1e-12)]
            for tv in tvals:
                cand = best + tv * v
                cd = project_capped(cand, n)
                if cd is None:
                    continue
                cs = exact_score(cd, n)
                if cs < best_trial_s - 1e-12:
                    best_trial_s = cs
                    best_trial = cd
        if best_trial is not None:
            best = best_trial
            best_s = best_trial_s
            delta = min(0.25, 1.2 * delta)
        else:
            delta *= 0.5
    return best, best_s
