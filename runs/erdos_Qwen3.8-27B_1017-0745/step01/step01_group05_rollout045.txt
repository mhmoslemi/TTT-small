            cands.extend(t_lin[(t_lin >= lo) & (t_lin <= hi)].tolist())
        # Quadratic case
        quad = m2 & (np.abs(aa) >= 1e-15)
        if quad.any():
            aa_q = aa[quad]; bb_q = bb[quad]; cc_q = cc[quad]
            disc = bb_q*bb_q - 4.0*aa_q*cc_q
            good = disc >= -1e-10
            if good.any():
                sq = np.sqrt(np.maximum(disc[good], 0.0))
                t1 = (-bb_q[good] - sq) / (2.0 * aa_q[good])
                t2 = (-bb_q[good] + sq) / (2.0 * aa_q[good])
                for t in [t1, t2]:
                    cands.extend(t[(t >= lo) & (t <= hi)].tolist())
    ca = np.clip(np.array(cands), lo, hi)
    # Deduplicate
    ca = np.unique(np.round(ca, 8))
    best_t, best_v = 0.0, cur_s
    for s0 in range(0, len(ca), 256):
        ch = ca[s0:s0+256]
        V = A[None,:]*ch[:,None]**2 + B[None,:]*ch[:,None] + C[None,:]
        mx = V.max(axis=1)
        i = int(np.argmin(mx))
        if mx[i] < best_v - 1e-14: best_v, best_t = float(mx[i]), float(ch[i])
    if best_t == 0.0 or best_v >= cur_s - 1e-14: return h, cur_s
    hn = project_onto_k(h + best_t * d, n)
    if not ok(hn, n): return h, cur_s
    sn = sc(hn, n)
    return (hn, sn) if sn < cur_s - 1e-14 else (h, cur_s)
