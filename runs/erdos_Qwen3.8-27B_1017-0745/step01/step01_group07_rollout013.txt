q_d = dx * sum_{i ∈ I_d} h[i] * (1 - h[i+d])

where I_d = {i : 0 ≤ i < n, 0 ≤ i+d < n}

The derivative with respect to h[j]:
- If j ∈ I_d: contributes dx * (1 - h[j+d])
- If j-d ∈ I_d (i.e., j = i+d for some i ∈ I_d, so 0 ≤ j-d < n, 0 ≤ j < n): contributes dx * (-h[j-d])

So G[d_row, j] = dx * (1-h[j+d]) * [j∈I_d] - dx * h[j-d] * [j-d∈I_d]

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csc_matrix, csr_matrix, eye as sp_eye

def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):
        return None
    if np.max(np.abs(z)) > 1e12:
        return None
    n = len(z)
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    y = np.clip(z - lam, 0.0, 1.0)
    r = S - np.sum(y)
    if r > 0:
        for i in range(n):
            if r <= 0: break
            a = min(r, 1.0 - y[i])
            y[i] += a; r -= a
    elif r < 0:
        for i in range(n):
            if r >= 0: break
            a = min(-r, y[i])
            y[i] -= a; r += a
    if abs(np.sum(y) - S) > 1e-10 or not np.all(np.isfinite(y)):
        return None
    return np.clip(y, 0.0, 1.0)

def is_feasible(h, n):
    return (h is not None and h.ndim == 1 and len(h) == n
            and np.all(np.isfinite(h)) and np.all(h >= -1e-15)
            and np.all(h <= 1.0 + 1e-15) and abs(np.sum(h) - n / 2.0) <= 1e-10)

def exact_score(h, n):
    try:
        vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if len(vals) != 2 * n - 1 or not np.all(np.isfinite(vals)):
            return None
        return float(np.max(vals))
    except Exception:
        return None

def build_lag_model(h, n):
    dx = 2.0 / n
    q = np.correlate(h, 1.0 - h, mode='full')[::-1] * dx
    # q[d + n - 1] corresponds to lag d
    m = 2 * n - 1
    rows = []
    cols = []
    data = []
    for d in range(-(n-1), n):
        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        for i in range(lo, hi):
            # dh[i] term: dx * (1 - h[i+d])
            rows.append(row)
            cols.append(i)
            data.append(dx * (1.0 - h[i + d]))
            # dh[i+d] term: -dx * h[i]
            rows.append(row)
            cols.append(i + d)
            data.append(-dx * h[i])
    gc = coo_matrix((data, (rows, cols)), shape=(m, n))
    G = gc.tocsr()
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G.data)):
        return None, None
    return q, G

def solve_lp_step(x, q, G, rho, n):
    m = 2 * n - 1
    dx = 2.0 / n
    mu = 0.002 * dx
    # Variables: delta (n), u (n), t (1) → 2n+1
    nv = 2 * n + 1
    # Objective
    c = np.zeros(nv)
    c[n:2*n] = mu
    c[2*n] = 1.0  # t coefficient
    
    # Inequality constraints: A_ub @ vars <= b_ub
    # 1) G @ delta - t <= -q  →  [G, 0(n), -1] vars <= -q
    # 2) delta - u <= 0       →  [I, -I, 0] vars <= 0
    # 3) -delta - u <= 0      →  [-I, -I, 0] vars <= 0
    
    # Build sparse
    # Block 1: (m, 2n+1)
    # Block 2: (n, 2n+1)
    # Block 3: (n, 2n+1)
    
    # For block 1, G is (m, n), we need [G, zeros(m,n), -ones(m,1)]
    # For block 2, we need [I_n, -I_n, zeros(n,1)]
    # For block 3, we need [-I_n, -I_n, zeros(n,1)]
    
    total_rows = m + 2 * n
    all_rows = []
    all_cols = []
    all_data = []
    
    # Block 1: G @ delta - t <= -q
    Gc = G.tocoo()
    for i in range(Gc.nnz):
        all_rows.append(Gc.row[i])
        all_cols.append(Gc.col[i])
        all_data.append(Gc.data[i])
    # -t part for each row
    for i in range(m):
        all_rows.append(i)
        all_cols.append(2 * n)
        all_data.append(-1.0)
    
    # Block 2: delta - u <= 0
    offset = m
    for i in range(n):
        all_rows.append(offset + i)
        all_cols.append(i)
        all_data.append(1.0)
        all_rows.append(offset + i)
        all_cols.append(n + i)
        all_data.append(-1.0)
    
    # Block 3: -delta - u <= 0
    offset2 = m + n
    for i in range(n):
        all_rows.append(offset2 + i)
        all_cols.append(i)
        all_data.append(-1.0)
        all_rows.append(offset2 + i)
        all_cols.append(n + i)
        all_data.append(-1.0)
    
    A_ub = coo_matrix((all_data, (all_rows, all_cols)), shape=(total_rows, nv)).tocsr()
    b_ub = np.concatenate([-q, np.zeros(2 * n)])
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    lo_delta = np.maximum(-rho, -x)
    hi_delta = np.minimum(rho, 1.0 - x)
    bounds = []
    for i in range(n):
        bounds.append((lo_delta[i], hi_delta[i]))
    for i in range(n):
        bounds.append((0.0, None))
    bounds.append((None, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res is not None and res.status == 0:
            delta = res.x[:n]
            if np.all(np.isfinite(delta)) and abs(np.sum(delta)) <= 1e-6:
                return delta
    except Exception:
        pass
    return None

def optimize_trajectory(x0, n, rng, best_h, best_score):

    dx = 2.0 / n
    x = x0.copy()
    f = exact_score(x, n)
    if f is None or not np.isfinite(f):
        return best_h, best_score
    
    if f < best_score - 1e-12:
        best_h = x.copy()
        best_score = f
    
    if n <= 128:
        max_iter = 30
    elif n <= 256:
        max_iter = 22
    elif n <= 512:
        max_iter = 14
    else:
        max_iter = 6
    
    rho = 0.08
    rho_max = 0.25
    rho_min = 2e-4
    consec_fail = 0
    
    for it in range(max_iter):
        if consec_fail >= 4 or rho < rho_min:
            break
        
        q, G = build_lag_model(x, n)
        if q is None or G is None:
            consec_fail += 1
            rho = max(rho_min, rho * 0.5)
            continue
        
        delta = solve_lp_step(x, q, G, rho, n)
        if delta is None or np.max(np.abs(delta)) < 1e-9:
            if delta is None:
                consec_fail += 1
                rho = max(rho_min, rho * 0.5)
            break
        
        # Line search
        accepted = False
        best_beta = None
        best_x_new = None
        best_f_new = None
        
        for j in range(9):
            beta = 2.0 ** (-j)
            cand = x + beta * delta
            proj = project_capped_simplex(cand, S)
            if proj is None or not is_feasible(proj, n):
                continue
            sc = exact_score(proj, n)
            if sc is None or not np.isfinite(sc):
                continue
            if sc < best_score - 1e-12:
                best_h = proj.copy()
                best_score = sc
            if sc < f - 1e-12:
                best_beta = beta
                best_x_new = proj
                best_f_new = sc
                accepted = True
                break

            consec_fail += 1
            rho = max(rho_min, rho * 0.5)
            continue

        s = best_x_new - x
        # Predicted decrease
        pred = f - float(np.max(q + G.dot(s)))
        act = f - best_f_new
        if pred <= 1e-12 or (pred > 0 and act / pred < 0.25):
            rho = max(rho_min, rho * 0.5)

            rho = min(rho_max, rho * 1.5)

        f = best_f_new
        consec_fail = 0

def build_starts(n, S, parent, rng):
    starts = []
    const = np.full(n, 0.5)

    if n <= 128:

        max_starts = 3
    else:
        max_starts = 2
    
    if parent is not None:
        starts.append(parent.copy())
        for sd in [0.04, 0.12]:
            pert = parent + rng.normal(0.0, sd, n)
            p = project_capped_simplex(pert, S)
            if p is not None:
                starts.append(p)
        starts.append(parent[::-1].copy())
    
    # Low-frequency
    idx = np.arange(n, dtype=np.float64)
    M = min(5, (n - 1) // 2)
    u = np.full(n, 0.5)
    for m in range(1, M + 1):
        amp = rng.normal(0.0, 0.10 / np.sqrt(m))
        phi = rng.uniform(0.0, 2.0 * np.pi)
        u += amp * np.cos(2.0 * np.pi * m * (idx + 0.5) / n + phi)
    lp = project_capped_simplex(u, S)
    if lp is not None:
        starts.append(lp)
    
    # Binary
    perm = rng.permutation(n)
    binv = np.zeros(n)
    binv[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        binv[perm[n // 2]] = 0.5
    starts.append(binv.copy())
    
    # Deduplicate
    deduped = []
    for s in starts:
        if s is None or not is_feasible(s, n):
            continue
        dup = False
        for d in deduped:
            if np.max(np.abs(s - d)) < 1e-9:
                dup = True
                break
        if not dup:
            deduped.append(s)
        if len(deduped) >= max_starts:
            break
    return deduped
