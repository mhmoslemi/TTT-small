            if iv_arr.ndim == 1 and 8 <= iv_arr.size <= 999:
                n = iv_arr.size
                S_tmp = n / 2.0
                proj = project_capped(iv_arr, S_tmp)
                if proj is not None and is_feasible(proj, n):
                    parent = proj
        except Exception:
            pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 80))
        except (TypeError, ValueError):
            n = 80
        if n < 8 or n > 999:
            n = 80
    S = n / 2.0
    left, right, lag_id = build_pairs(n)
    h_const = np.full(n, 0.5)
    best_h = h_const.copy()
    best_score = exact_score(h_const)
    if best_score is None:
        best_score = 0.5
    if parent is not None:
        ps = exact_score(parent)
        if ps is not None and np.isfinite(ps) and ps < best_score:
            best_score = ps
            best_h = parent.copy()
    starts = [h_const.copy()]
    if parent is not None:
        starts.append(parent.copy())
        for sd in [0.05, 0.15]:
            p = project_capped(parent + sd * rng.normal(0, 1, n), S)
            if p is not None and is_feasible(p, n):
                starts.append(p)
        starts.append(parent[::-1].copy())
    idx = np.arange(n, dtype=np.float64)
    M = min(4, (n - 1) // 2)
    lowf = np.full(n, 0.5)
    for m in range(1, M + 1):
        phi = rng.uniform(0.0, 2.0 * np.pi)
        amp = (0.15 / np.sqrt(float(m))) * rng.normal(0.0, 1.0)
        lowf += amp * np.cos(2.0 * np.pi * m * (idx + 0.5) / float(n) + phi)
    p = project_capped(lowf, S)
    if p is not None and is_feasible(p, n):
        starts.append(p)
    ru = project_capped(rng.uniform(0.0, 1.0, n), S)
    if ru is not None and is_feasible(ru, n):
        starts.append(ru)
    deduped = []
    for s in starts:
        dup = False
        for d in deduped:
            if np.max(np.abs(s - d)) < 1e-8:
                dup = True
                break
        if not dup:
            deduped.append(s)
    if n <= 128:
        max_iter = 150
    elif n <= 256:
        max_iter = 90
    else:
        max_iter = 50
    for s in deduped:
        try:
            bx, bs = pg_optimize(s, n, S, left, right, lag_id, max_iter)
            sc = exact_score(bx)
            if sc is not None and np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_h = bx.copy()
        except Exception:
            pass
    n_kicks = 3 if n <= 128 else (2 if n <= 256 else 1)
    sigma = 0.06
    for _k in range(n_kicks):
        kick = project_capped(best_h + sigma * rng.normal(0.0, 1.0, n), S)
        if kick is None or not is_feasible(kick, n):
            sigma = min(0.25, 1.6 * sigma)
            continue
        try:
            bx, bs = pg_optimize(kick, n, S, left, right, lag_id, max_iter)
            sc = exact_score(bx)
            if sc is not None and np.isfinite(sc) and sc < best_score - 1e-13:
                best_score = sc
                best_h = bx.copy()
                sigma = max(0.02, 0.5 * sigma)
            else:
                sigma = min(0.25, 1.5 * sigma)
        except Exception:
            sigma = min(0.25, 1.5 * sigma)
    final = project_capped(best_h, S)
    if final is None or not is_feasible(final, n):
        if parent is not None:
            final = parent.copy()
        else:
            final = h_const.copy()
    sc = exact_score(final)
    if sc is None or not np.isfinite(sc):
        final = h_const.copy()
        sc = exact_score(final)
        if sc is None:
            sc = 0.5
    return final, sc, n
