def project_capped(z, n):
    s = n / 2.0
    if not np.all(np.isfinite(z)):
        return None
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,):
        return None
    z = np.clip(z, -1e12, 1e12)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15 * max(1.0, abs(mid)):
            break
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    tol = 1e-13 * max(1, n)
    if abs(r) > tol:

            for i in range(n):
                if abs(r) <= tol:
                    break
                add = min(r, 1.0 - h[i])
                if add > 0:
                    h[i] += add
                    r -= add
        else:

                    break
                sub = min(-r, h[i])

                    h[i] -= sub
                    r += sub

    if not np.all(np.isfinite(h)) or abs(np.sum(h) - s) > 5e-12 * max(1, n):
        return None
    return h

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):

        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_values_grads(h, n):
    L = 2 * n - 1
    v = np.zeros(L)
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        i0 = max(0, -d)
        m = n - abs(d)
        ii = np.arange(i0, i0 + m)
        jj = ii + d
        v[row] = float(np.sum(h[ii] * (1.0 - h[jj])))
        G[row, ii] += (1.0 - h[jj])
        G[row, jj] -= h[ii]
    return v, G

    return z[-1]

def _obj_jac(z, n):
    g = np.zeros(z.shape[0])
    g[-1] = 1.0
    return g

def _eq_fun(z, n):
    return float(np.sum(z[:n]) - n / 2.0)

def _eq_jac(z, n):
    g = np.zeros(z.shape[0])
    g[:n] = 1.0
    return g

def _ineq_fun(z, n):
    h = z[:n]
    t = z[-1]
    dx = 2.0 / n
    v, _ = lag_values_grads(h, n)
    return t - dx * v

def _ineq_jac(z, n):
    h = z[:n]
    dx = 2.0 / n
    L = 2 * n - 1
    _, G = lag_values_grads(h, n)
    J = np.zeros((L, z.shape[0]))
    J[:, :n] = -dx * G
    J[:, -1] = 1.0
    return J

def polish(h, n, sweeps):
    best = h.copy()
    best_s = exact_score(best, n)
    dvals = np.array([0.001, 0.003, 0.01, 0.03, 0.1, 0.3])
    for _ in range(sweeps):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                lo = max(-best[i], best[j] - 1.0)
                hi = min(1.0 - best[i], best[j])
                if lo >= hi:
                    continue
                cands = np.array([0.0, lo + 1e-12, hi - 1e-12])
                for dv in dvals:
                    for sgn in (1.0, -1.0):
                        vv = sgn * dv
                        if lo < vv < hi:
                            cands = np.append(cands, vv)
                bs = best_s
                bt = None
                for delta in cands:
                    trial = best.copy()
                    trial[i] += delta
                    trial[j] -= delta
                    if trial[i] < -1e-10 or trial[i] > 1.0 + 1e-10:
                        continue
                    if trial[j] < -1e-10 or trial[j] > 1.0 + 1e-10:
                        continue
                    st = exact_score(trial, n)
                    if st < bs - 1e-13:
                        bs = st
                        bt = trial
                if bt is not None:
                    best = bt
                    best_s = bs
                    improved = True
        if not improved:
            break
    return best, best_s
