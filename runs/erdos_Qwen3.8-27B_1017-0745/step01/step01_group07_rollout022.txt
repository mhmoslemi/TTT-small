```python
def polish_pair(h, cur_score, n, dx, rng, max_sweeps):
    qlags = exact_lags(h, n)
    P = min(n*(n-1)//2, max(24, 80000//n))
    # Build pair list: adjacent + random
    pairs = []
    n_adj = min(n-1, max(1, P//2))
    adj_step = max(1, (n-1) // n_adj)
    for i in range(0, n-1, adj_step):
        pairs.append((i, i+1))
        if len(pairs) >= n_adj:
            break
    while len(pairs) < P:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i < j and (i,j) not in pairs:
            pairs.append((i,j))
        if len(pairs) > P * 3:
            break
    rng.shuffle(pairs)
    for _ in range(max_sweeps):
        any_acc = False
        for (p, r) in pairs:
            res = line_search(h, qlags, cur_score, p, r, n, dx)
            if res is not None:
                h, cur_score = res
                qlags = exact_lags(h, n)
                any_acc = True
        if not any_acc:
            break
    return h, cur_score
```
