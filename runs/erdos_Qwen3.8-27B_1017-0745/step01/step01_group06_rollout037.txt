        nd = kwargs.get('n_points', 45)
        if not isinstance(nd, (int, np.integer)) or isinstance(nd, bool):
            nd = 45
        n = max(2, min(999, int(nd)))
        parent = np.full(n, 0.5)
    dx, S = 2.0 / n, n / 2.0
    base = project_K(parent.copy(), n, S)
    fb_h = base.copy()
    fb_s = exact_score(fb_h, dx)
    best_h, best_s = base.copy(), fb_s
    if not np.isfinite(fb_s):
        fb_h = np.full(n, 0.5)
        fb_s = exact_score(fb_h, dx)
        best_h, best_s = fb_h.copy(), fb_s
    if n <= 64:
        P, G, NR, ST = 72, 120, 2, 18
    elif n <= 128:
        P, G, NR, ST = 48, 60, 1, 12
    elif n <= 256:
        P, G, NR, ST = 32, 30, 1, 8
    else:
        P, G, NR, ST = 20, 12, 0, 0
    cands = [base.copy(), project_K(1.0 - base, n, S), project_K(base[::-1].copy(), n, S),
             project_K(1.0 - base[::-1].copy(), n, S)]
    for sg in [0.03, 0.12, 0.30]:
        cands.append(project_K(base + sg * rng.standard_normal(n), n, S))
    cands.append(project_K(rng.uniform(0, 1, n), n, S))
    cands.append(project_K(rng.beta(0.5, 0.5, n), n, S))
    bv = np.zeros(n)
    bv[:int(S)] = 1.0
    if S > int(S):
        bv[int(S)] = S - int(S)
    cands.append(project_K(bv[rng.permutation(n)], n, S))
    t = np.arange(n)
    cands.append(project_K(0.5 + 0.45 * np.cos(2.0 * np.pi * t / n), n, S))
    pop = np.zeros((P, n))
    sc = np.full(P, float('inf'))
    for i in range(P):
        if i < len(cands):
            c = cands[i].copy()
        else:
            r = rng.random()
            if r < 0.4:
                sg = 10.0 ** rng.uniform(np.log10(0.015), np.log10(0.35))
                c = project_K(base + sg * rng.standard_normal(n), n, S)
            elif r < 0.7:
                c = project_K(rng.uniform(0, 1, n), n, S)
            else:
                c = project_K(rng.beta(0.5, 0.5, n), n, S)
        if feasible(c, n, S):
            s = exact_score(c, dx)
            pop[i], sc[i] = c, s
            if s < best_s - 1e-12:
                best_s, best_h = s, c.copy()
        else:
            pop[i], sc[i] = base.copy(), fb_s
    mu_F, mu_CR, stall, nrestart = 0.55, 0.75, 0, 0
    aidx = np.arange(P)
    for gen in range(G):
        sp, ss = pop.copy(), sc.copy()
        order = np.argsort(ss, kind='stable')
        tk = max(2, int(np.ceil(0.2 * P)))
        top = order[:tk].copy()
        nsuc, Fs, CRs = 0, [], []
        pb = best_s
        for i in range(P):
            F = float(np.clip(mu_F + 0.1 * rng.standard_normal(), 0.1, 1.0))
            CR = float(np.clip(mu_CR + 0.1 * rng.standard_normal(), 0.0, 1.0))
            if rng.random() < 0.8:
                p = int(top[rng.integers(tk)])
                while p == i and len(top) > 1:
                    p = int(top[rng.integers(tk)])
                r1 = int(rng.integers(P))
                while r1 == i or r1 == p:
                    r1 = int(rng.integers(P))
                r2 = int(rng.integers(P))
                while r2 == i or r2 == p or r2 == r1:
                    r2 = int(rng.integers(P))
                v = sp[i] + F * (sp[p] - sp[i]) + F * (sp[r1] - sp[r2])
            else:
                r1 = int(rng.integers(P))
                while r1 == i:
                    r1 = int(rng.integers(P))
                r2 = int(rng.integers(P))
                while r2 == i or r2 == r1:
                    r2 = int(rng.integers(P))
                r3 = int(rng.integers(P))
                while r3 == i or r3 == r1 or r3 == r2:
                    r3 = int(rng.integers(P))
                v = sp[r1] + F * (sp[r2] - sp[r3])
            if not np.all(np.isfinite(v)):
                continue
            sr = int(rng.integers(n))
            u = sp[i].copy()
            mask = rng.random(n) < CR
            mask[sr] = True
            u[mask] = v[mask]
            u = project_K(u, n, S)
            if not feasible(u, n, S):
                continue
            s = exact_score(u, dx)
            if not np.isfinite(s):
                continue
            if s < ss[i] - 1e-12:
                pop[i] = u
                sc[i] = s
                nsuc += 1
                Fs.append(F)
                CRs.append(CR)
                if s < best_s - 1e-12:
                    best_s, best_h = s, u.copy()
        if nsuc > 0:
            Fa = np.array(Fs)
            mu_F = float(np.clip(0.9 * mu_F + 0.1 * float((Fa * Fa).sum() / Fa.sum()), 0.1, 1.0))
            mu_CR = float(np.clip(0.9 * mu_CR + 0.1 * float(np.mean(np.array(CRs))), 0.0, 1.0))
        if best_s < pb - 1e-12:
            stall = 0
        else:
            stall += 1
        if stall >= ST and nrestart < NR and gen < G - 1:
            nrestart += 1
            st = np.argsort(sc, kind='stable')
            keep = st[:max(2, int(np.ceil(0.2 * P)))]
            sigs = [0.08, 0.18, 0.35]
            for pos in range(len(keep), P):
                r = rng.random()
                if r < 0.75:
                    sg = sigs[pos % 3]
                    c = project_K(best_h + sg * rng.standard_normal(n), n, S)
                else:
                    c = project_K(rng.beta(0.5, 0.5, n), n, S)
                if feasible(c, n, S):
                    s = exact_score(c, dx)
                    pop[pos], sc[pos] = c, s
                    if s < best_s - 1e-12:
                        best_s, best_h = s, c.copy()
                else:
                    pop[pos], sc[pos] = best_h.copy(), best_s
            mu_F, mu_CR, stall = 0.7, 0.8, 0
    # Refinement
    if n <= 64:
        rvar = [0.05, 0.02, 0.008, 0.003]
        rcount = 100
    elif n <= 128:
        rvar = [0.04, 0.015, 0.005]
        rcount = 40
    elif n <= 256:
        rvar = [0.025, 0.008]
        rcount = 20
    else:
        rvar = [0.012, 0.004]
        rcount = 8
    for sg in rvar:
        for _ in range(rcount):
            d = make_dir_full(rng, n) if rng.random() < 0.75 else make_dir_sparse(rng, n)
            if d is None:
                continue
            trial = project_K(best_h + sg * d, n, S)
            if not feasible(trial, n, S):
                continue
            ts = exact_score(trial, dx)
            if np.isfinite(ts) and ts < best_s - 1e-12:
                best_s, best_h = ts, trial.copy()
    # Final validation
    best_h = project_K(best_h, n, S)
    if not feasible(best_h, n, S):
        best_h, best_s = fb_h.copy(), fb_s
    else:
        fs = exact_score(best_h, dx)
        ffs = exact_score(fb_h, dx)
        if np.isfinite(fs) and fs <= ffs + 1e-10:
            best_s = fs
        else:
            best_h, best_s = fb_h.copy(), ffs
    best_h = np.clip(best_h, 0.0, 1.0)
    final_s = exact_score(best_h, dx)
    if not np.isfinite(final_s):
        final_s = fb_s
        best_h = fb_h.copy()
    return (best_h, float(final_s), int(n))
