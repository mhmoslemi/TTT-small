Let me carefully analyze the problem and strategy to implement a DC-MM (Difference of Convex - Majorization Minimization) approach for the Erdős minimum overlap problem.

Key ideas:
1. We have h: [0,2] → [0,1] with ∫h = 1, discretized as n points
2. We want to minimize max_k ∫h(x)(1-h(x+k))dx
3. The discretized version: minimize max_k (correlate(h, 1-h, 'full') * dx)

The strategy uses DC decomposition:
- Write each lag as q_ell(h) = c_ell^T h - h^T B_d h
- Split B_d = P_d - N_d where P_d, N_d are positive definite
- At current x, the convex upper model is qhat_ell(y|x) = c_ell^T y + ||Ln_d^T y||^2 - 2(P_d x)^T y + x^T P_d x
- Minimize max_ell qhat_ell(y|x) + (rho/2)||y-x||^2 subject to 0≤y≤1, sum(y)=S

This is a convex SOCP solvable by CVXPY with ECOS or CVXOPT.

  So c_{-d}[j] = dx for j < n-d, 0 otherwise.

- For ell = 0: q_0(h) = dx * sum h[i]*(1-h[i]) = dx * sum h[i] - dx * sum h[i]^2 = S*dx - h^T B_0 h
  So c_0[j] = dx for all j.

Now the convex model:
qhat_ell(y|x) = c_ell^T y + y^T N_d y - 2(P_d x)^T y + x^T P_d x

And the subproblem:
minimize max_ell qhat_ell(y|x) + (rho/2)||y-x||^2
s.t. 0 ≤ y ≤ 1, sum(y) = S

This can be written as:
minimize t + (rho/2)||y-x||^2
s.t. qhat_ell(y|x) ≤ t for all ell
     0 ≤ y ≤ 1, sum(y) = S

In CVXPY, (rho/2)||y-x||^2 = (rho/2)*sum_squares(y - x). Since x is a parameter, this is convex.

Each qhat_ell(y|x) = y^T N_d y + (c_ell - 2*P_d*x)^T y + x^T P_d x

The y^T N_d y term: since N_d = Ln_d @ Ln_d.T, we have y^T N_d y = ||Ln_d.T @ y||^2. In CVROPY this is sum_squares(Ln_d.T @ y).

So the constraint is:
sum_squares(Ln_d.T @ y) + (c_ell - 2*P_d@x) @ y + x @ (P_d @ x) ≤ t

This is a convex quadratic constraint (sum of squares of affine functions + affine ≤ linear), which is an SOCP-representable constraint.

import numpy as np
import cvxpy as cp
import scipy.linalg as sla

def project_capped(y, n, S):
    if not np.all(np.isfinite(y)):
        return np.full(n, S / n)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    err = S - h.sum()
    if abs(err) > 1e-12:
        idx = range(n) if err > 0 else range(n - 1, -1, -1)
        for j in idx:
            if err > 0:
                add = min(err, 1.0 - h[j])
                h[j] += add
                err -= add
            else:
                sub = min(-err, h[j])
                h[j] -= sub
                err += sub
            if abs(err) < 1e-12:
                break
    if abs(h.sum() - S) > 1e-8:
        return np.full(n, S / n)
    return h

def score_exact(h, n):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def build_factors(n, dx):
    """Build Cholesky factors for P_d and N_d for all d=0..n-1."""
    gamma = 1.25 * dx
    I = np.eye(n)
    Lp = []
    Ln = []
    for d in range(n):
        B = np.zeros((n, n))
        if d == 0:
            B = dx * I
        else:
            for i in range(n - d):
                B[i, i + d] = dx / 2
                B[i + d, i] = dx / 2
        P = 0.5 * (gamma * I + B)
        N = 0.5 * (gamma * I - B)
        try:
            Lp_d = np.linalg.cholesky(P)
            Ln_d = np.linalg.cholesky(N)
        except np.linalg.LinAlgError:
            return None
        Lp.append(Lp_d)
        Ln.append(Ln_d)
    return Lp, Ln

def make_model(n, S, rho, Lp, Ln):
    """Build CVXPY model. Returns (prob, x_par, v_pars, a_pars, y, t)."""
    y = cp.Variable(n)
    t = cp.Variable()
    x_par = cp.Parameter(n)
    # For each d, we need v_par[d] = P_d @ x and a_par[d] = x @ (P_d @ x)
    v_pars = [cp.Parameter(n) for _ in range(n)]
    a_pars = [cp.Parameter() for _ in range(n)]
    
    dx = 2.0 / n
    constraints = [y >= 0, y <= 1, cp.sum(y) == S]
    
    for ell in range(-(n-1), n):
        d = abs(ell)
        # c_ell vector
        c = np.zeros(n)
        if d == 0:
            c[:] = dx
        elif ell > 0:
            c[d:] = dx
        else:
            c[:n-d] = dx
        
        # qhat_ell(y|x) = c^T y + y^T N_d y - 2*(P_d x)^T y + x^T P_d x
        # = c^T y + ||Ln_d^T y||^2 - 2*v_par[d]^T y + a_par[d]
        Ln_d = Ln[d]
        constraints.append(
            cp.sum_squares(Ln_d.T @ y) + (c - 2 * v_pars[d]) @ y + a_pars[d] <= t
        )
    
    obj = cp.Minimize(t + (rho / 2) * cp.sum_squares(y - x_par))
    prob = cp.Problem(obj, constraints)
    return prob, x_par, v_pars, a_pars, y, t

def dc_mm_step(x, n, S, dx, Lp, Ln, rho, solver):
    """One DC-MM iteration from x. Returns new x or None."""
    # Build model
    prob, x_par, v_pars, a_pars, y_var, t_var = make_model(n, S, rho, Lp, Ln)
    
    # Set parameters
    x_par.value = x
    for d in range(n):
        v_pars[d].value = Lp[d] @ (Lp[d].T @ x)
        a_pars[d].value = x @ (Lp[d] @ (Lp[d].T @ x))
    
    y_var.value = x
    t_var.value = score_exact(x, n) + 1e-8
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if y_var.value is None or t_var.value is None:
        return None
    if not np.all(np.isfinite(y_var.value)) or not np.isfinite(t_var.value):
        return None
    
    y_raw = y_var.value
    if y_raw.min() < -1e-5 or y_raw.max() > 1.0 + 1e-5:
        return None
    if abs(y_raw.sum() - S) > 1e-5:
        return None
    
    y_new = project_capped(y_raw, n, S)
    if not feasible(y_new, n, S):
        return None
    
    s_new = score_exact(y_new, n)
    if not np.isfinite(s_new):
        return None
    
    return y_new, s_new

    rng = np.random.default_rng(seed)
    init = globals().get('initial_h_values', None)
    
    n = 45
    if init is not None:
        init = np.array(init, dtype=np.float64).ravel()
        if 2 <= len(init) <= 999 and np.all(np.isfinite(init)):

        else:

    else:

    S = n / 2.0
    dx = 2.0 / n
    fb = project_capped(base, n, S)
    fb_s = score_exact(fb, n)
    best_h, best_s = fb.copy(), fb_s
    
    # Generate starts
    starts = [fb.copy()]
    starts.append(project_capped(1.0 - fb, n, S))
    starts.append(project_capped(fb[::-1].copy(), n, S))
    starts.append(project_capped(1.0 - fb[::-1].copy(), n, S))
    for sig in [0.04, 0.12, 0.25]:
        starts.append(project_capped(fb + sig * rng.standard_normal(n), n, S))
    starts.append(project_capped(rng.uniform(0, 1, n), n, S))
    starts.append(project_capped(rng.beta(0.5, 0.5, n), n, S))
    
    # Scoring starts
    for s in starts:
        if feasible(s, n, S):

            if sc < best_s - 1e-12:

            Lp, Ln = factors
            rho_sched = [0.12, 0.04, 0.012]
            max_solves = 4
            # Select top starts
            scored = [(score_exact(s, n), s) for s in starts if feasible(s, n, S)]
            scored.sort(key=lambda x: x[0])
            sel_starts = [scored[0][1]]

                close = any(np.max(np.abs(sv - ss)) <= 1e-5 for _, ss in sel_starts)

    else:

        if fs > fb_s + 1e-10:

        else:
            best_s = fs
    best_h = np.clip(best_h, 0.0, 1.0)
    return (best_h, score_exact(best_h, n), n)

import numpy as np
import cvxpy as cp

def project_capped(y, n, S):
    if not np.all(np.isfinite(y)):
        return np.full(n, S / n)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    err = S - h.sum()
    if abs(err) > 1e-12:
        for j in range(n):
            if abs(err) < 1e-12:
                break
            if err > 0:
                add = min(err, 1.0 - h[j])
                h[j] += add
                err -= add
            else:
                sub = min(-err, h[j])
                h[j] -= sub
                err += sub
    if abs(h.sum() - S) > 1e-8:
        return np.full(n, S / n)
    return h

def score_exact(h, n):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def build_factors(n, dx):
    gamma = 1.25 * dx
    I = np.eye(n)
    Lp_l, Ln_l = [], []
    for d in range(n):
        B = np.zeros((n, n))
        if d == 0:
            B = dx * I
        else:
            rng = range(n - d)
            idx = np.arange(n - d)
            B[idx, idx + d] = dx / 2
            B[idx + d, idx] = dx / 2
        P = 0.5 * (gamma * I + B)
        N = 0.5 * (gamma * I - B)
        try:
            Lp_l.append(np.linalg.cholesky(P))
            Ln_l.append(np.linalg.cholesky(N))
        except np.linalg.LinAlgError:
            return None
    return Lp_l, Ln_l

def make_model(n, S, rho, Lp, Ln):
    dx = 2.0 / n
    y = cp.Variable(n)
    t = cp.Variable()
    x_par = cp.Parameter(n)
    v_pars = [cp.Parameter(n) for _ in range(n)]
    a_pars = [cp.Parameter() for _ in range(n)]
    cons = [y >= 0, y <= 1, cp.sum(y) == S]
    for ell in range(-(n - 1), n):
        d = abs(ell)
        c = np.zeros(n)
        if d == 0:
            c[:] = dx
        elif ell > 0:
            c[d:] = dx
        else:
            c[:n - d] = dx
        cons.append(cp.sum_squares(Ln[d].T @ y) + (c - 2 * v_pars[d]) @ y + a_pars[d] <= t)
    obj = cp.Minimize(t + (rho / 2.0) * cp.sum_squares(y - x_par))
    return cp.Problem(obj, cons), x_par, v_pars, a_pars, y, t

def dc_solve(x, n, S, Lp, Ln, rho, prob, x_par, v_pars, a_pars, y_v, t_v, solver):
    x_par.value = x
    for d in range(n):
        px = Lp[d] @ (Lp[d].T @ x)
        v_pars[d].value = px
        a_pars[d].value = float(x @ px)
    y_v.value = x
    t_v.value = score_exact(x, n) + 1e-8
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    yr, tr = y_v.value, t_v.value
    if yr is None or tr is None or not np.all(np.isfinite(yr)) or not np.isfinite(tr):
        return None
    if yr.min() < -1e-4 or yr.max() > 1.0 + 1e-4 or abs(yr.sum() - S) > 1e-4:
        return None
    y_new = project_capped(yr, n, S)
    if not feasible(y_new, n, S):
        return None
    s_new = score_exact(y_new, n)
    if not np.isfinite(s_new):
        return None
    return y_new, s_new

def try_solve(x, n, S, Lp, Ln, rho, solver_list):
    dx = 2.0 / n
    for solver in solver_list:
        try:
            m, x_p, v_p, a_p, y_v, t_v = make_model(n, S, rho, Lp, Ln)
            r = dc_solve(x, n, S, Lp, Ln, rho, m, x_p, v_p, a_p, y_v, t_v, solver)
            if r is not None:
                return r
        except Exception:
            continue
    return None

def pair_polish(h, n, S, rng, iters=600):
    h = h.copy()
    cur_s = score_exact(h, n)
    for _ in range(iters):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        for delta in (0.02, 0.05, 0.1):
            h2 = h.copy()
            h2[i] -= delta
            h2[j] += delta
            if h2[i] < -1e-12 or h2[j] > 1.0 + 1e-12:
                continue
            h2 = np.clip(h2, 0.0, 1.0)
            h2 = project_capped(h2, n, S)
            s2 = score_exact(h2, n)
            if s2 < cur_s - 1e-12:
                h, cur_s = h2, s2
                break
    return h, cur_s
