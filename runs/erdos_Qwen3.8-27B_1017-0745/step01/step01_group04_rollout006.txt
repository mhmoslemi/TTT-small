        return np.full(N, 0.5)
    lo, hi = z.min() - 1.0, z.max()

        else:
            hi = mid

    if r > 0:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
            if r <= 1e-14:
                break
    elif r < 0:
        for i in range(N):
            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1 + 1e-10:

    cmax = np.max(c)
    e = np.exp((c - cmax) / tau)
    se = np.sum(e)
    p = e / se
    F = cmax + tau * np.log(se)
    grad = np.zeros(N)
    for s in range(-(N - 1), N):
        r_idx = N - 1 + s
        w = dx * p[r_idx]
        if w < 1e-15:
            continue

        elif s > 0:

            grad[d:] += w * (1.0 - h[:L])
            grad[:L] -= w * h[d:]
    return F, grad, c
