def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n < 1 or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-14:
        if resid > 0:
            for idx in np.argsort(1.0 - y):
                add = min(resid, 1.0 - y[idx])
                y[idx] += add
                resid -= add
                if abs(resid) < 1e-15:
                    break
        else:
            for idx in np.argsort(y):
                sub = min(-resid, y[idx])
                y[idx] -= sub
                resid += sub
                if abs(resid) < 1e-15:
                    break
    y = np.clip(y, 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 0:
        if resid > 0 and y[0] < 1.0:
            y[0] += min(resid, 1.0 - y[0])
        elif resid < 0 and y[0] > 0.0:
            y[0] += max(resid, -y[0])
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    if abs(np.sum(y) - S) > 1e-10 * max(1.0, float(n)):
        return None
    return y

def check_feasible(h, n):
    if h is None or h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 5e-10 * max(1.0, float(n))

def exact_score(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def line_minimax(x, y, n, S):
    d = y - x
    n_lags = 2 * n - 1
    A = np.zeros(n_lags)
    B = np.zeros(n_lags)
    D = np.zeros(n_lags)
    for k in range(-(n - 1), n):
        il = max(0, k)
        ih = min(n - 1, n - 1 + k)
        i = np.arange(il, ih + 1)
        j = i - k
        D[k + n - 1] = np.sum(x[i] * (1.0 - x[j]))
        B[k + n - 1] = np.sum(d[i] * (1.0 - x[j]) - x[i] * d[j])
        A[k + n - 1] = -np.sum(d[i] * d[j])
    alphas = [0.0, 1.0]
    for k in range(n_lags):
        if abs(A[k]) > 1e-14:
            a = -B[k] / (2.0 * A[k])
            if -1e-10 <= a <= 1.0 + 1e-10:
                alphas.append(float(np.clip(a, 0.0, 1.0)))
    for k in range(n_lags):
        for l in range(k + 1, n_lags):
            da, db, dd = A[k] - A[l], B[k] - B[l], D[k] - D[l]
            if abs(da) < 1e-14:
                if abs(db) > 1e-14:
                    a = -dd / db
                    if -1e-10 <= a <= 1.0 + 1e-10:
                        alphas.append(float(np.clip(a, 0.0, 1.0)))
            else:
                disc = db * db - 4.0 * da * dd
                if disc > -1e-12 * (db * db + 4.0 * abs(da) * abs(dd) + 1e-30):
                    sqrt_d = np.sqrt(max(0.0, disc))
                    for a in [(-db + sqrt_d) / (2.0 * da), (-db - sqrt_d) / (2.0 * da)]:
                        if -1e-10 <= a <= 1.0 + 1e-10:
                            alphas.append(float(np.clip(a, 0.0, 1.0)))
    alphas = np.unique(np.array([a for a in alphas if 0.0 <= a <= 1.0]))
    best_sc = np.inf
    best_h = x.copy()
    for a in alphas:
        h_p = project_box_sum(x + a * d, S)
        if h_p is None:
            continue
        sc = exact_score(h_p)
        if sc < best_sc:
            best_sc = sc
            best_h = h_p
    return best_h, best_sc

def build_lp(current, r, n, S):
    lo = np.maximum(0.0, current - r)
    hi = np.minimum(1.0, current + r)
    P = n * (n + 1) // 2
    nv = n + P + 1
    vo, zo, to = 0, n, n + P
    rows, cols, vals = [], [], []
    rhs = []
    def add_ineq(coeffs, rhs_val):
        for var, c in coeffs:
            if abs(c) > 1e-15:
                rows.append(len(rhs))
                cols.append(var)
                vals.append(c)
        rhs.append(rhs_val)
    # McCormick
    for i in range(n):
        for j in range(i, n):
            p = i * n - i * (i - 1) // 2 + (j - i)
            pi, pj = lo[i], lo[j]
            qi, qj = hi[i], hi[j]
            pp = vo + i
            qp = vo + j
            zp = zo + p
            # 1: lj*vi + li*vj - z <= li*lj
            add_ineq([(qp, pj), (pp, pi), (zp, -1.0)], pi * pj)
            # 2: qj*vi + qi*vj - z <= qi*qj
            add_ineq([(qp, qj), (pp, qi), (zp, -1.0)], qi * qj)
            # 3: z - qi*vj - lj*vi <= -qi*lj
            add_ineq([(qp, -qi), (pp, -pj), (zp, 1.0)], -qi * pj)
            # 4: z - li*vj - qj*vi <= -li*qj
            add_ineq([(qp, -pi), (pp, -qj), (zp, 1.0)], -pi * qj)
    # Lag constraints: sum_i v_i - sum_i z_minmax <= t
    for k in range(-(n - 1), n):
        il = max(0, k)
        ih = min(n - 1, n - 1 + k)
        coeffs = [(to, -1.0)]
        for ii in range(il, ih + 1):
            jj = ii - k
            mn, mx = (ii, jj) if ii <= jj else (jj, ii)
            p = mn * n - mn * (mn - 1) // 2 + (mx - mn)
            coeffs.append((vo + ii, 1.0))
            coeffs.append((zo + p, -1.0))
        add_ineq(coeffs, 0.0)
    # Equality handled separately
    A_ub = coo_matrix((vals, (rows, cols)), shape=(len(rhs), nv)).tocsc()
    b_ub = np.array(rhs)
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    c = np.zeros(nv)
    c[to] = 1.0
    bnds = [(lo[i], hi[i]) for i in range(n)]
    for i in range(n):
        for j in range(i, n):
            p = i * n - i * (i - 1) // 2 + (j - i)
            bnds.append((lo[i] * lo[j], hi[i] * hi[j]))
    bnds.append((0.0, float(n)))
    return linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                   bounds=bnds, method='highs')

def project_box_sum(v, S):

    n = len(v)
    if n < 1 or not np.all(np.isfinite(v)):
        return None

        if np.sum(y) > S:

    resid = S - np.sum(y)
    if abs(resid) > 1e-14:

            for idx in np.argsort(1.0 - y):
                add = min(resid, 1.0 - y[idx])
                y[idx] += add
                resid -= add
                if abs(resid) < 1e-15:

def build_lp(current, r, n, S):

    hi = np.minimum(1.0, current + r)
    P = n * (n + 1) // 2
    nv = n + P + 1
    vo, zo, to = 0, n, n + P
    rr, cc, vv, rhs = [], [], [], []
    for i in range(n):
        for j in range(i, n):
            p = i * n - i * (i - 1) // 2 + (j - i)
            pi, pj, qi, qj = lo[i], lo[j], hi[i], hi[j]
            pp, qp, zp = vo + i, vo + j, zo + p
            for (c1, c2, c3, rv) in [(pj, pi, -1.0, pi * pj), (qj, qi, -1.0, qi * qj),
                                      (-qi, -pj, 1.0, -qi * pj), (-pi, -qj, 1.0, -pi * qj)]:
                row_idx = len(rhs)
                rhs.append(rv)
                if abs(c1) > 1e-15:
                    rr.append(row_idx); cc.append(qp); vv.append(c1)
                if abs(c2) > 1e-15:
                    rr.append(row_idx); cc.append(pp); vv.append(c2)
                if abs(c3) > 1e-15:
                    rr.append(row_idx); cc.append(zp); vv.append(c3)
    for k in range(-(n - 1), n):
        il, ih = max(0, k), min(n - 1, n - 1 + k)
        row_idx = len(rhs)
        rhs.append(0.0)
        rr.append(row_idx); cc.append(to); vv.append(-1.0)
        for ii in range(il, ih + 1):
            jj = ii - k
            mn, mx = (ii, jj) if ii <= jj else (jj, ii)
            p = mn * n - mn * (mn - 1) // 2 + (mx - mn)
            rr.append(row_idx); cc.append(vo + ii); vv.append(1.0)
            rr.append(row_idx); cc.append(zo + p); vv.append(-1.0)
    A_ub = coo_matrix((vv, (rr, cc)), shape=(len(rhs), nv)).tocsc()
    b_ub = np.array(rhs)
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    c = np.zeros(nv)
    c[to] = 1.0

    r = 0.20
