import numpy as np
import cvxpy as cp

def project_capped_simplex(y, n, S):
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return np.full(n, S / n)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    e = S - h.sum()
    if abs(e) > 1e-10:
        idx = range(n) if e > 0 else range(n - 1, -1, -1)
        for j in idx:
            room = (1.0 - h[j]) if e > 0 else h[j]
            adj = min(room, abs(e))
            h[j] += adj if e > 0 else -adj
            e += adj if e < 0 else -adj
            if abs(e) < 1e-12:
                break
    if abs(h.sum() - S) > 1e-8 or h.min() < -1e-10 or h.max() > 1.0 + 1e-10:
        return np.full(n, S / n)
    return h

def score_exact(h, n):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def is_feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def build_factors(n, dx):
    gamma = 1.25 * dx
    Lns, Lps = [], []
    for d in range(n):
        B = np.zeros((n, n))
        if d == 0:
            np.fill_diagonal(B, dx)
        else:
            idx = np.arange(n - d)
            B[idx, idx + d] = dx / 2
            B[idx + d, idx] = dx / 2
        Pm = (gamma * np.eye(n) + B) / 2.0
        Nm = (gamma * np.eye(n) - B) / 2.0
        try:
            Lps.append(np.linalg.cholesky(Pm))
            Lns.append(np.linalg.cholesky(Nm))
        except np.linalg.LinAlgError:
            return None, None, None
    c_list = []
    for ell in range(-(n - 1), n):
        d = abs(ell)
        c = np.zeros(n)
        if d == 0:
            c[:] = dx
        elif ell > 0:
            c[d:] = dx
        else:
            c[:n - d] = dx
        c_list.append(c)
    return Lns, Lps, c_list

def solve_mm_step(y, n, S, rho, Lns, Lps, c_list, dx):
    x_par = cp.Parameter(n)
    v_pars = [cp.Parameter(n) for _ in range(n)]
    a_pars = [cp.Parameter() for _ in range(n)]
    yv = cp.Variable(n)
    tv = cp.Variable()
    cons = [yv >= 0, yv <= 1, cp.sum(yv) == S]
    for k in range(2 * n - 1):
        d = abs(k - (n - 1))
        cons.append(c_list[k] @ yv + cp.sum_squares(Lns[d].T @ yv) - 2 * v_pars[d] @ yv + a_pars[d] <= tv)
    obj = cp.Minimize(tv + (rho / 2.0) * cp.sum_squares(yv - x_par))
    prob = cp.Problem(obj, cons)
    # Set parameters
    x_par.value = y
    for d in range(n):
        vx = Lps[d] @ (Lps[d].T @ y)
        v_pars[d].value = vx
        a_pars[d].value = float(y @ vx)
    yv.value = y
    tv.value = score_exact(y, n) + 1e-8
    solver_name = None
    res = None
    for sname in ['ECOS', 'MOSEK', 'CVXOPT']:
        try:
            prob.solve(solver=getattr(cp, sname), verbose=False, warm_start=True)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver_name = sname
                res = yv.value
                break
        except Exception:
            continue
    if res is None or not np.all(np.isfinite(res)):
        return None
    return res

    init = globals().get('initial_h_values', None)
    n = 45
    base = None
    if init is not None:
        init = np.array(init, dtype=np.float64).ravel()
        if 2 <= len(init) <= 999 and np.all(np.isfinite(init)) and np.max(np.abs(init)) < 1e6:
            n = len(init)
            base = init
    S, dx = n / 2.0, 2.0 / n
    fb = project_capped_simplex(base.copy(), n, S) if base is not None else np.full(n, 0.5)
    best_h, best_s = fb.copy(), score_exact(fb, n)
    starts = [best_h.copy(), (1.0 - best_h).copy(), best_h[::-1].copy(), (1.0 - best_h[::-1]).copy()]
    for sig in [0.04, 0.10, 0.20, 0.35]:
        starts.append(project_capped_simplex(best_h + sig * rng.standard_normal(n), n, S))
    starts.append(project_capped_simplex(rng.uniform(0, 1, n), n, S))
    ph = rng.uniform(0, 2 * np.pi)
    starts.append(project_capped_simplex(0.5 + 0.4 * np.cos(2 * np.pi * np.arange(n) / n * 2.0 + ph), n, S))
    starts.append(project_capped_simplex(0.5 + 0.45 * np.sin(2 * np.pi * np.arange(n) / n * 3.0 + ph), n, S))
    validated = []
    for s in starts:
        if is_feasible(s, n, S):
            sc = score_exact(s, n)
            if sc < best_s - 1e-12:
                best_s, best_h = sc, s.copy()
            dup = any(np.max(np.abs(s - vh)) < 1e-7 for _, vh in validated)
            if not dup:
                validated.append((sc, s.copy()))
    validated.sort(key=lambda x: x[0])
    mm_done = False
    if n <= 80:
        try:
            Lns, Lps, c_list = build_factors(n, dx)
        except Exception:
            Lns = None
        if Lns is not None:
            solver = None
            for sn in ['ECOS', 'MOSEK', 'CVXOPT', 'SCS']:
                if hasattr(cp, sn):
                    solver = sn
                    break
            if solver is not None:
                mm_done = True
                n_starts = min(4, len(validated))
                for _, sv in validated[:n_starts]:
                    try:
                        xres, sres = dc_mm_run(sv, n, S, [0.15, 0.05, 0.015, 0.005], Lns, Lps, c_list, solver)
                        if sres < best_s - 1e-12:
                            best_s, best_h = sres, xres
                    except Exception:
                        pass
                for sig in [0.06, 0.15]:
                    pp = project_capped_simplex(best_h + sig * rng.standard_normal(n), n, S)
                    if is_feasible(pp, n, S):
                        try:
                            xres, sres = dc_mm_run(pp, n, S, [0.10, 0.03, 0.01, 0.004], Lns, Lps, c_list, solver)
                            if sres < best_s - 1e-12:
                                best_s, best_h = sres, xres
                        except Exception:
                            pass
    if n <= 100:
        n_starts = min(3, len(validated))
        for _, sv in validated[:n_starts]:
            xw = sv.copy()
            scw = score_exact(xw, n)
            for _ in range(300):
                lags = np.correlate(xw, 1.0 - xw, mode='full') * dx
                km = int(np.argmax(lags))
                ell = km - (n - 1)
                g = np.zeros(n)
                j = np.arange(n)
                i1, v1 = j - ell, (j - ell >= 0) & (j - ell < n)
                i2, v2 = j + ell, (j + ell >= 0) & (j + ell < n)
                t1, t2 = np.zeros(n), np.zeros(n)
                t1[v1] = 1.0 - xw[i1[v1]]
                t2[v2] = xw[i2[v2]]
                g = dx * (t1 - t2)
                step = 0.005
                for _s in range(50):
                    yn = project_capped_simplex(xw - step * g, n, S)
                    sn = score_exact(yn, n)
                    if sn < scw - 1e-13:
                        xw, scw = yn, sn
                        if scw < best_s - 1e-12:
                            best_s, best_h = scw, xw.copy()
                        break
                    step *= 0.5
                else:
                    break
    best_h = project_capped_simplex(best_h, n, S)
    fs = score_exact(best_h, n)
    fbs = score_exact(fb, n)
    if fs > fbs + 1e-10:
        best_h, fs = fb, fbs
    return (best_h, float(fs), n)
