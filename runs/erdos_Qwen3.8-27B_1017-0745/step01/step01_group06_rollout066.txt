    j = np.arange(n)
    for k in range(nl):

        i2, v2 = j + ell, (j + ell >= 0) & (j + ell < n)

        t1[v1] = 1.0 - h[i1[v1]]
        t2[v2] = h[i2[v2]]
        G[k] = dx * (t1 - t2)
    return G

    d = z[:-1]
    return z[-1] + np.dot(d, d) / (2.0 * eta)

def tr_obj_jac(z, eta):
    g = np.empty_like(z)
    g[:-1] = z[:-1] / eta
    g[-1] = 1.0

    d = z[:-1]
    t = z[-1]
    return t - q - G @ d

    return g

        G = lag_grad(h, n, dx)
        qmax = q.max()
        z0 = np.concatenate([np.zeros(n), [qmax]])
        lb = np.maximum(-delta, -h)
        ub = np.minimum(delta, 1.0 - h)
        bounds = list(zip(lb, ub)) + [(None, None)]
        con = [{'type': 'ineq', 'fun': tr_ineq, 'jac': tr_ineq_jac, 'args': (q, G)},

            continue

        actual = sc_cur - sc_new

            delta *= 0.5
            consec_fail += 1
            if consec_fail >= 4 or delta < 1e-6:

        ratio = actual / max(pred, 1e-15)

            delta *= 0.5
            eta = max(eta * 0.5, 1e-4)

        elif ratio > 0.75 and np.max(np.abs(da)) > 0.8 * delta:
            delta = min(0.4, delta * 1.8)
            eta = min(eta * 1.4, 2.0)
        if np.max(np.abs(da)) < 1e-9:

    return h

def _tr_obj(z, eta):
    d = z[:-1]
    return z[-1] + np.dot(d, d) / (2.0 * eta)

def _tr_obj_j(z, eta):
    g = np.empty_like(z)
    g[:-1] = z[:-1] / eta
    g[-1] = 1.0
    return g

def _tr_in(z, q, G):
    return z[-1] - q - G @ z[:-1]

def _tr_in_j(z, q, G):
    J = np.empty((G.shape[0], G.shape[1] + 1))
    J[:, :-1] = -G
    J[:, -1] = 1.0
    return J

    return z[:-1].sum()

def _tr_eq_j(z, n, S):

def trust_region(h, n, dx, max_it=25):
    S = n / 2.0
    delta, eta = 0.10, 0.5
    cf = 0
    for _ in range(max_it):
        q = np.correlate(h, 1.0 - h, mode='full') * dx
        G = lag_grad(h, n, dx)
        z0 = np.concatenate([np.zeros(n), [q.max()]])
        lb = np.maximum(-delta, -h)
        ub = np.minimum(delta, 1.0 - h)
        bnds = list(zip(lb, ub)) + [(None, None)]
        cons = [{'type': 'ineq', 'fun': _tr_in, 'jac': _tr_in_j, 'args': (q, G)},
                {'type': 'eq', 'fun': _tr_eq, 'jac': _tr_eq_j, 'args': (n, S)}]
        try:
            r = minimize(_tr_obj, z0, jac=_tr_obj_j, args=(eta,),
                         method='SLSQP', bounds=bnds, constraints=cons,
                         options={'maxiter': 35, 'ftol': 1e-12})
        except Exception:
            r = None
        if r is None or not np.all(np.isfinite(r.x)):
            delta *= 0.5; cf += 1
            if cf >= 4 or delta < 1e-6: break
            continue
        ht = project_K(h + r.x[:-1], n, S)
        if not feasible(ht, n, S) or ht is None:
            delta *= 0.5; cf += 1
            if cf >= 4 or delta < 1e-6: break
            continue
        da = ht - h
        sc0 = exact_score(h, dx)
        sc1 = exact_score(ht, dx)
        if sc1 >= sc0 - 1e-13:
            delta *= 0.5; cf += 1
            if cf >= 4 or delta < 1e-6: break
            continue
        h = ht; cf = 0
        pred = q.max() - (q + G @ da).max()
        ratio = (sc0 - sc1) / max(pred, 1e-15)
        if ratio < 0.15:
            delta *= 0.5; eta = max(eta * 0.5, 1e-4)
        elif ratio > 0.7 and np.max(np.abs(da)) > 0.8 * delta:
            delta = min(0.4, delta * 1.7); eta = min(eta * 1.3, 2.0)
        if np.max(np.abs(da)) < 1e-9: break
    return h

def lse_continuation(h, n, dx, taus, mi=60):
    S = n / 2.0
    h = h.copy()
    for tau in taus:
        a = 1.0
        for _ in range(mi):
            q = np.correlate(h, 1.0 - h, mode='full') * dx
            m = q.max()
            w = np.exp((q - m) / tau)
            p = w / w.sum()
            g = np.zeros(n)
            j = np.arange(n)
            for k in range(2 * n - 1):
                if p[k] < 1e-15: continue
                ell = k - (n - 1)
                v1 = (j + ell >= 0) & (j + ell < n)
                v2 = (j - ell >= 0) & (j - ell < n)
                t = np.zeros(n)
                if np.any(v1): t[v1] = 1.0 - h[j[v1] + ell]
                if np.any(v2): t[v2] -= h[j[v2] - ell]
                g += p[k] * dx * t
            y = project_K(h - a * g, n, S)
            d = y - h
            if np.max(np.abs(d)) < 1e-10: break
            F0 = m + tau * np.log(w.sum())
            F1 = lse_val(y, n, dx, tau)
            gd = g @ d
            if F1 <= F0 + 1e-4 * gd:
                h = y; a = min(1.2 * a, 5.0)
            else:
                a *= 0.5
                if a < 1e-8: break
    return h

def lse_val(h, n, dx, tau):
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    m = q.max()
    return m + tau * np.log(np.sum(np.exp((q - m) / tau)))

def transfer(h0, n0, m):

            y[j] += h0[i] * ov
        y[j] *= m
    return y

import numpy as np
from scipy.optimize import minimize

def project_K(y, n, S):
    if not np.all(np.isfinite(y)):
        return np.full(n, S / n)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    err = S - h.sum()
    if abs(err) > 1e-12:
        if err > 0:
            idx = np.where(1.0 - h > 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) > 0:
            h[idx] = np.clip(h[idx] + err / len(idx), 0.0, 1.0)
    return h if abs(h.sum() - S) <= 1e-8 else np.full(n, S / n)

def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def lag_grad(h, n, dx):
    G = np.zeros((2 * n - 1, n))
    j = np.arange(n)
    for k in range(2 * n - 1):
        ell = k - (n - 1)
        v1 = (j + ell >= 0) & (j + ell < n)
        v2 = (j - ell >= 0) & (j - ell < n)
        t = np.zeros(n)
        if np.any(v2):
            t[v2] += 1.0 - h[j[v2] - ell]
        if np.any(v1):
            t[v1] -= h[j[v1] + ell]
        G[k] = dx * t
    return G

def _tr_obj(z, eta):
    d = z[:-1]
    return z[-1] + np.dot(d, d) / (2.0 * eta)

def _tr_obj_j(z, eta):
    g = np.empty_like(z)
    g[:-1] = z[:-1] / eta
    g[-1] = 1.0
    return g

def _tr_in(z, q, G):
    return z[-1] - q - G @ z[:-1]

def _tr_in_j(z, q, G):
    J = np.empty((G.shape[0], G.shape[1] + 1))
    J[:, :-1] = -G
    J[:, -1] = 1.0
    return J

def _tr_eq(z, n, S):
    return z[:-1].sum()

def _tr_eq_j(z, n, S):
    g = np.zeros(len(z))
    g[:-1] = 1.0
    return g

def trust_region(h, n, dx, max_it=20):
    S = n / 2.0
    delta, eta = 0.12, 0.5
    cf = 0
    for _ in range(max_it):
        q = np.correlate(h, 1.0 - h, mode='full') * dx
        G = lag_grad(h, n, dx)
        z0 = np.concatenate([np.zeros(n), [q.max()]])
        lb = np.maximum(-delta, -h)
        ub = np.minimum(delta, 1.0 - h)
        bnds = list(zip(lb, ub)) + [(None, None)]
        cons = [{'type': 'ineq', 'fun': _tr_in, 'jac': _tr_in_j, 'args': (q, G)},
                {'type': 'eq', 'fun': _tr_eq, 'jac': _tr_eq_j, 'args': (n, S)}]
        try:
            r = minimize(_tr_obj, z0, jac=_tr_obj_j, args=(eta,),
                         method='SLSQP', bounds=bnds, constraints=cons,
                         options={'maxiter': 30, 'ftol': 1e-12})
        except Exception:
            r = None
        if r is None or not np.all(np.isfinite(r.x)):
            delta *= 0.5; cf += 1
            if cf >= 4 or delta < 1e-6: break
            continue
        ht = project_K(h + r.x[:-1], n, S)
        sc0 = exact_score(h, dx)
        sc1 = exact_score(ht, dx)
        if not feasible(ht, n, S) or sc1 >= sc0 - 1e-13:
            delta *= 0.5; cf += 1
            if cf >= 4 or delta < 1e-6: break
            continue
        h = ht; cf = 0
        da = ht - h
        pred = q.max() - (q + G @ da).max()
        ratio = (sc0 - sc1) / max(pred, 1e-15)
        if ratio < 0.15:
            delta *= 0.5; eta = max(eta * 0.5, 1e-4)
        elif ratio > 0.7 and np.max(np.abs(da)) > 0.8 * delta:
            delta = min(0.4, delta * 1.7); eta = min(eta * 1.3, 2.0)
        if np.max(np.abs(da)) < 1e-9: break
    return h
