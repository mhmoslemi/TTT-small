    for k in range(2 * n - 1):

        v1 = (j - ell >= 0) & (j - ell < n)

        if len(idx) > 0:
            h[idx] = np.clip(h[idx] + e / len(idx), 0.0, 1.0)
    if abs(h.sum() - S) > 1e-8:

    if not np.all(np.isfinite(h)):

        t2 = np.zeros(n); t2[v2] = h[j[v2] + ell]
