def de_evolve(pop, scores, best, best_score, n, gens, rng, patience, restarts):
    if n <= 1 or gens <= 0 or len(pop) < 4: return pop, scores, best, best_score
    npop = len(pop); stag=0; rcount=0; scales=(0.03,0.07,0.15)
    for _ in range(gens):
        old_pop = [x.copy() for x in pop]
        old_sc = scores.copy()
        npop_old = len(pop)
        next_pop = [x.copy() for x in pop]
        next_sc = scores.copy()
        improved=False
        b = int(np.argmin(old_sc))
        for i in range(npop_old):
            if rng.random() < 0.8:
                while True:
                    r1 = int(rng.integers(npop_old)); r2 = int(rng.integers(npop_old)); r3 = int(rng.integers(npop_old))
                    if r1 != i and r2 != i and r3 != i and r1 != r2 and r1 != r3 and r2 != r3: break
                F = 0.45 + 0.5 * float(rng.random())
                v = old_pop[r1] + F * (old_pop[r2] - old_pop[r3])
            else:
                while True:
                    r1 = int(rng.integers(npop_old)); r2 = int(rng.integers(npop_old))
                    if r1 != i and r2 != i and r1 != r2: break
                F = 0.45 + 0.5 * float(rng.random())
                v = old_pop[i] + F * (old_pop[b] - old_pop[i]) + F * (old_pop[r1] - old_pop[r2])
            CR = 0.75 + 0.2 * float(rng.random())
            mask = rng.random(n) < CR
            mask[int(rng.integers(n))] = True
            trial = project_capped(np.where(mask, v, old_pop[i]), n)
            if trial is None: continue
            if float(np.max(np.abs(trial - old_pop[i]))) <= 1e-12: continue
            sc = exact_score(trial, n)
            if np.isfinite(sc) and sc < old_sc[i] - 1e-12:
                next_pop[i] = trial; next_sc[i] = sc
                if sc < best_score - 1e-12:
                    best = trial.copy(); best_score = sc; improved = True
        pop, scores = next_pop, next_sc
        if improved: stag = 0
        else: stag += 1
        if stag >= patience:
            if restarts > 0:
                pop, scores, best, best_score = restart_pop(pop, scores, best, best_score, n, rng, scales[min(rcount, 2)])

def parse_int(v, d, lo, hi):

        if isinstance(v, (bool, np.bool_)):
            return d

        if not math.isfinite(x):
            return d

        return lo if y < lo else hi if y > hi else y

def project_capped(z, n):
    s = 0.5 * n
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if mid == lo or mid == hi:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    tol = max(1e-13, 5e-12 * max(1, n))

    rcount = 0
    rs = (0.03, 0.07, 0.15)
    for _ in range(gens):
        old_pop = [x.copy() for x in pop]
        old_sc = scores.copy()
        next_pop = [x.copy() for x in pop]
        next_sc = scores.copy()
        any_improve = False
        npop = len(pop)
        b = int(np.argmin(old_sc))
        for i in range(npop):
            if rng.random() < 0.8:
                while True:
                    r1 = int(rng.integers(npop))
                    r2 = int(rng.integers(npop))
                    r3 = int(rng.integers(npop))
                    if r1 != i and r2 != i and r3 != i and r1 != r2 and r1 != r3 and r2 != r3:
                        break
                F = 0.45 + 0.5 * float(rng.random())
                v = old_pop[r1] + F * (old_pop[r2] - old_pop[r3])
            else:
                while True:
                    r1 = int(rng.integers(npop))
                    r2 = int(rng.integers(npop))
                    if r1 != i and r2 != i and r1 != r2:
                        break
                F = 0.45 + 0.5 * float(rng.random())
                v = old_pop[i] + F * (old_pop[b] - old_pop[i]) + F * (old_pop[r1] - old_pop[r2])
            CR = 0.75 + 0.2 * float(rng.random())
            mask = rng.random(n) < CR
            mask[int(rng.integers(n))] = True
            trial = project_capped(np.where(mask, v, old_pop[i]), n)
            if trial is None:
                continue
            if float(np.max(np.abs(trial - old_pop[i]))) <= 1e-12:
                continue
            sc = exact_score(trial, n)
            if np.isfinite(sc) and sc < old_sc[i] - 1e-12:
                next_pop[i] = trial
                next_sc[i] = sc
                any_improve = True
                if sc < best_score - 1e-12:
                    best = trial.copy()
                    best_score = sc
        pop, scores = next_pop, next_sc
        if any_improve:
            stag = 0
        else:
            stag += 1
        if stag >= patience:
            if restarts > 0:
                pop, scores, best, best_score = restart_pop(pop, scores, best, best_score, n, rng, rs[min(rcount, 2)])
                restarts -= 1
                rcount += 1
                stag = 0
            else:
                break
    return pop, scores, best, best_score

def local_polish(h, n, rng, moves):
    h = np.array(h, dtype=np.float64).copy()
    best = exact_score(h, n)
    if n <= 1 or moves <= 0 or not np.isfinite(best):
        u = np.full(n, 0.5)
        return u, exact_score(u, n)
    q = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(q))
    act = np.where(q >= 0.9 * mx)[0]
    if act.size == 0:
        act = np.array([n - 1])
    for it in range(moves):
        if it % 10 == 0:
            q = np.correlate(h, 1.0 - h, mode='full')
            mx = float(np.max(q))
            act = np.where(q >= 0.9 * mx)[0]
            if act.size == 0:
                act = np.array([n - 1])
        k = int(act[int(rng.integers(act.size))])
        lag = k - (n - 1)
        if lag == 0 or abs(lag) >= n:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
        else:
            d = abs(lag)
            i = int(rng.integers(n - d))
            j = i + d
        gi, gj = float(h[i]), float(h[j])
        lo, hi = -gi, 1.0 - gj
        if hi - lo <= 1e-11:
            continue
        lo += 1e-12
        hi -= 1e-12
        if hi <= lo:
            continue
        bt, bs = 0.0, best
        for t in np.linspace(lo, hi, 13):
            if abs(t) < 1e-12:
                continue
            h[i] = gi + t
            h[j] = gj - t
            sc = exact_score(h, n)
            if np.isfinite(sc) and sc < bs - 1e-13:
                bs = sc
                bt = float(t)
        h[i] = gi
        h[j] = gj
        if bs < best - 1e-12:
            trial = h.copy()
            trial[i] = gi + bt
            trial[j] = gj - bt
            trial = np.clip(trial, 0.0, 1.0)
            hp = project_capped(trial, n)
            if hp is not None:
                scp = exact_score(hp, n)
                if np.isfinite(scp) and scp < best - 1e-13:
                    h = hp
                    best = scp
    return h, best

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if len(iv) > 0:
            n = len(iv)
            if np.all(np.isfinite(iv)) and np.all(iv >= -1e-8) and np.all(iv <= 1.0 + 1e-8):
                parent = iv.copy()
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = parse_int(kwargs.get('n_points', 47), 47, 1, 999)
    n = int(n)
    uniform = np.full(n, 0.5)
    parent_h = None
    if parent is not None:
        if is_feasible(parent, n):
            parent_h = parent.copy()
        else:
            p2 = project_capped(parent, n)
            if p2 is not None and exact_score(p2, n) < float('inf'):
                parent_h = p2
    fallback = parent_h if parent_h is not None else uniform.copy()
    fb = exact_score(fallback, n)
    if not np.isfinite(fb):
        fallback = uniform.copy()
        fb = exact_score(uniform, n)
    seed_val = parse_int(seed, 42, -(2 ** 62), 2 ** 62)
    rng = np.random.default_rng(seed_val % (2 ** 32))
    starts = build_starts(parent_h, n, rng, uniform)
    best = fallback.copy()
    best_score = fb
    for s in starts:
        sc = exact_score(s, n)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best = s.copy()
            best_score = sc
    if n <= 64:
        NP = min(80, max(20, 4 * n))
        gens, restarts, moves = 200, 2, 180
    elif n <= 128:
        NP, gens, restarts, moves = 56, 100, 1, 100
    elif n <= 256:
        NP, gens, restarts, moves = 32, 40, 0, 50
    else:
        NP, gens, restarts, moves = 16, 0, 0, 0
    if n <= 256:
        NP = parse_int(kwargs.get('pop_size', NP), NP, 8, 96)
        gcap = 250 if n <= 64 else (120 if n <= 128 else 50)
        gens = parse_int(kwargs.get('generations', gens), gens, 0, gcap)
        restarts = parse_int(kwargs.get('max_restarts', restarts), restarts, 0, 3)
        moves = parse_int(kwargs.get('local_moves', moves), moves, 0, 300)
    patience = max(1, min(40, max(8, gens // 4))) if gens > 0 else 1
    pop = init_population(starts, best, n, rng, uniform, NP) if n <= 256 else starts
    pop, scores, best, best_score = score_pop(pop, n, best, best_score)
    if gens > 0 and n <= 256:
        pop, scores, best, best_score = de_evolve(pop, scores, best, best_score, n, gens, rng, patience, restarts)
    if moves > 0 and pop:
        order = np.argsort(scores)
        topk = min(3, len(order)) if n <= 128 else 1
        per = max(1, moves // topk)
        for ri in range(topk):
            polished, ps = local_polish(pop[int(order[ri])].copy(), n, rng, per)
            if np.isfinite(ps) and ps < best_score - 1e-12:
                best = polished.copy()
                best_score = ps
    cands = [best, fallback, uniform]
    if pop:
        cands.append(pop[int(np.argmin(scores))])
    selected = None
    selsc = float('inf')
    for c in cands:
        if c is None:
            continue
        c = np.asarray(c, dtype=np.float64).copy()
        sc = exact_score(c, n)
        if np.isfinite(sc) and sc < selsc:
            selected = c
            selsc = sc
    if selected is None:
        selected = uniform.copy()
        selsc = exact_score(selected, n)
    rep = project_capped(selected, n)
    if rep is not None:
        rsc = exact_score(rep, n)
        if np.isfinite(rsc) and rsc <= selsc + 1e-12:
            selected = rep
            selsc = rsc
    selected = np.clip(selected, 0.0, 1.0)
    final_rep = project_capped(selected, n)
    if final_rep is not None:
        selected = final_rep
    fsc = exact_score(selected, n)
    if not np.isfinite(fsc):
        selected = uniform.copy()
        fsc = exact_score(selected, n)
    return selected, float(fsc), int(n)
