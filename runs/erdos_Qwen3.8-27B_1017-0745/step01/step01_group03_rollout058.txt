Key insight: The objective is C₅ = max_k Q_k(h) * (2/n) where Q_k = sum_i h[i](1-h[i-k]) over valid indices.

For the LP:
- Variables: v[0..n-1] (the h values), z[i,j] for products v[i]*v[j] (i≤j), and t
- McCormick constraints for each product
- Lag constraints: sum_i (v[i] - z[min(i,j),max(i,j)]) ≤ t for each lag
- Equality: sum(v) = S
- Minimize t

For n=45: P = 45*46/2 = 1035 products, total vars = 45 + 1035 + 1 = 1081
Inequality rows: 4*1035 + (2*45-1) = 4140 + 89 = 4229
That's manageable.

For line minimax:
- Along h(a) = x + a*d for a in [0,1]
- Each lag Q_k is quadratic in a: A_k*a² + B_k*a + D_k
- Find min of max of these quadratics by checking endpoints, stationary points, and pairwise intersections

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0 or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5*(lo+hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if resid > 0:
        order = np.argsort(1.0 - y)
        for i in order:
            add = min(resid, 1.0 - y[i])
            y[i] += add
            resid -= add
            if resid < 1e-15: break
    elif resid < 0:
        order = np.argsort(y)
        for i in order:
            sub = min(-resid, y[i])
            y[i] -= sub
            resid += sub
            if -resid < 1e-15: break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1.0+1e-12):
        return None
    if abs(np.sum(y) - S) > 1e-9*max(1.0, float(n)):
        return None
    return np.clip(y, 0.0, 1.0)

def exact_score(h, n):
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0+1e-12):
        return np.inf
    corr = np.correlate(h, 1.0-h, mode='full')
    return float(np.max(corr) * (2.0/n))

def check_feasible(h, n):
    if h is None or h.shape != (n,): return False
    if not np.all(np.isfinite(h)): return False
    if np.any(h < -1e-12) or np.any(h > 1.0+1e-12): return False
    if abs(np.sum(h) - n/2.0) > 1e-9*max(1.0, float(n)): return False
    return True

def compute_lag_coeffs(h, d, n):
    # Returns A, B, D for each lag: Q_k(a) = A_k*a^2 + B_k*a + D_k
    # h(a) = h + a*d
    # Q_k = sum_i h(i,a)*(1-h(i-k,a)) = sum_i (h[i]+a*d[i])*(1-h[j]-a*d[j])
    # where j = i-k
    # = sum_i h[i] - h[i]*h[j] + a*d[i] - a*d[i]*h[j] - a*h[i]*d[j] + a^2*d[i]*h[j]*d[j]... 
    
    # Actually Q_k = sum_i h[i,a] - h[i,a]*h[j,a]
    # = sum_i (h[i]+a*d[i]) - (h[i]+a*d[i])(h[j]+a*d[j])
    # = sum_i (h[i]+a*d[i]) - h[i]*h[j] - a*h[i]*d[j] - a*d[i]*h[j] - a^2*d[i]*d[j]
    # D_k = sum_i h[i] - h[i]*h[j]
    # B_k = sum_i d[i] - h[i]*d[j] - d[i]*h[j]
    # A_k = sum_i -d[i]*d[j]
    
    A = np.zeros(2*n-1)
    B = np.zeros(2*n-1)
    D = np.zeros(2*n-1)
    for k in range(-(n-1), n):
        idx = k + n - 1
        i_lo = max(0, k)
        i_hi = min(n-1, n-1+k)
        if i_lo > i_hi: continue
        i = np.arange(i_lo, i_hi+1)
        j = i - k
        # h(a) - h(a)*h_j(a)
        # = h_i + a*d_i - (h_i + a*d_i)(h_j + a*d_j)
        # = h_i - h_i*h_j + a*(d_i - h_i*d_j - d_i*h_j) + a^2*(-d_i*d_j)
        D[idx] = np.sum(h[i] - h[i]*h[j])
        B[idx] = np.sum(d[i] - h[i]*d[j] - d[i]*h[j])
        A[idx] = np.sum(-d[i]*d[j])
    return A, B, D

def line_minimax(x, y, n, S):
    if not check_feasible(x, n) or not check_feasible(y, n):
        return x, exact_score(x, n)
    d = y - x
    A, B, D = compute_lag_coeffs(x, d, n)
    # Find candidates: endpoints, stationary points, pairwise intersections
    alphas = [0.0, 1.0]
    # Stationary points
    for k in range(2*n-1):
        if abs(A[k]) > 1e-15:
            a = -B[k] / (2.0*A[k])
            if -1e-10 <= a <= 1.0+1e-10:
                alphas.append(max(0.0, min(1.0, a)))
    # Pairwise intersections (sample a subset for speed)
    n_lags = 2*n-1
    for k1 in range(n_lags):
        for k2 in range(k1+1, n_lags):
            da = A[k1]-A[k2]
            db = B[k1]-B[k2]
            dc = D[k1]-D[k2]
            if abs(da) > 1e-15:
                disc = db*db - 4.0*da*dc
                if disc >= -1e-12:
                    disc = max(0.0, disc)
                    sq = np.sqrt(disc)
                    a1 = (-db+sq)/(2.0*da)
                    a2 = (-db-sq)/(2.0*da)
                    for a in [a1, a2]:
                        if -1e-10 <= a <= 1.0+1e-10:
                            alphas.append(max(0.0, min(1.0, a)))
            elif abs(db) > 1e-15:
                a = -dc/db
                if -1e-10 <= a <= 1.0+1e-10:
                    alphas.append(max(0.0, min(1.0, a)))
    # Evaluate and select best
    alphas = sorted(set([max(0.0, min(1.0, a)) for a in alphas if np.isfinite(a)]))
    # Deduplicate
    dedup = []
    for a in alphas:
        if not dedup or a - dedup[-1] > 1e-10:
            dedup.append(a)
    
    # For n=45, n_lags=89, pairwise is ~4005 pairs, fine
    # Evaluate all candidates
    best_a = 0.0
    best_val = np.inf
    for a in dedup:
        vals = A*a*a + B*a + D
        mx = float(np.max(vals))
        if mx < best_val:
            best_val = mx
            best_a = a
    
    h_new = x + best_a * d
    h_new = project_box_sum(h_new, S)
    if h_new is not None and check_feasible(h_new, n):
        sc = exact_score(h_new, n)
        if sc < np.inf:
            return h_new, sc
    return x, exact_score(x, n)

def build_lp(x, r, n, S):
    # Trust box: l[i] = max(0, x[i]-r), u[i] = min(1, x[i]+r)
    l = np.maximum(0.0, x - r)
    u = np.minimum(1.0, x + r)
    P = n*(n+1)//2
    # Variable order: v[0..n-1], z[0..P-1], t
    # z index for (i,j) with i<=j: i*n - i*(i-1)//2 + (j-i)
    def zidx(i, j):
        if i > j: i, j = j, i
        return i*n - i*(i-1)//2 + (j-i)
    
    n_vars = n + P + 1
    t_idx = n + P
    
    # Objective: minimize t
    c = np.zeros(n_vars)
    c[t_idx] = 1.0
    
    # McCormick constraints for each pair (i,j), i<=j
    # z >= l_i*v_j + l_j*v_i - l_i*l_j  =>  -v_i*l_j - v_j*l_i + z >= -l_i*l_j  =>  l_j*v_i + l_i*v_j - z <= l_i*l_j
    # z >= u_i*v_j + u_j*v_i - u_i*u_j  =>  -u_j*v_i - u_i*v_j + z <= 0  wait...
    # Let me redo: McCormick for z = v_i * v_j, l_i <= v_i <= u_i, l_j <= v_j <= u_j:
    # z >= l_i*v_j + l_j*v_i - l_i*l_j  -->  l_j*v_i + l_i*v_j - z <= l_i*l_j
    # z >= u_i*v_j + u_j*v_i - u_i*u_j  -->  u_j*v_i + u_i*v_j - z <= u_i*u_j  WAIT
    # Actually: z >= u_i*v_j + u_j*v_i - u_i*u_j
    # Rearranging: -u_j*v_i - u_i*v_j + z >= -u_i*u_j
    # In <= form: u_j*v_i + u_i*v_j - z <= u_i*u_j  NO
    # z >= u_i*v_j + u_j*v_i - u_i*u_j
    # 0 >= u_i*v_j + u_j*v_i - u_i*u_j - z
    # z <= ... no
    # -u_j*v_i - u_i*v_j + z >= -u_i*u_j
    # Multiplying by -1: u_j*v_i + u_i*v_j - z <= u_i*u_j  ... NO, that's wrong sign
    # Let me be very careful:
    # McCormick: z >= max(l_i*v_j + l_j*v_i - l_i*l_j, u_i*v_j + u_j*v_i - u_i*u_j)
    # z <= min(l_i*v_j + u_j*v_i - l_i*u_j, u_i*v_j + l_j*v_i - u_i*l_j)
    
    # Inequality 1: z >= l_i*v_j + l_j*v_i - l_i*l_j
    #   l_j*v_i + l_i*v_j - z <= l_i*l_j
    # Inequality 2: z >= u_i*v_j + u_j*v_i - u_i*u_j
    #   u_j*v_i + u_i*v_j - z <= u_i*u_j  ... wait
    #   z - u_i*v_j - u_j*v_i >= -u_i*u_j
    #   -z + u_i*v_j + u_j*v_i <= u_i*u_j  ... NO
    #   z >= u_i*v_j + u_j*v_i - u_i*u_j
    #   0 >= u_i*v_j + u_j*v_i - u_i*u_j - z
    #   u_j*v_i + u_i*v_j - z <= u_i*u_j  ... this is WRONG
    #   Let me just write: -u_j*v_i - u_i*v_j + z >= -u_i*u_j
    #   In <= form: u_j*v_i + u_i*v_j - z <= u_i*u_j  ... 
    #   Actually: (z - u_i*v_j - u_j*v_i) >= -u_i*u_j
    #   Multiply by -1: -(z - u_i*v_j - u_j*v_i) <= u_i*u_j
    #   -z + u_i*v_j + u_j*v_i <= u_i*u_j
    #   So: u_i*v_j + u_j*v_i - z <= u_i*u_j  ... YES that's correct!
    
    # Wait, for i=j: u_i*v_i + u_i*v_i - z <= u_i^2 means 2*u_i*v_i - z <= u_i^2
    # And z >= u_i*v_i + u_i*v_i - u_i^2 means z >= 2*u_i*v_i - u_i^2
    # For v_i in [0, u_i]: 2*u_i*v_i - u_i^2 <= v_i^2? 
    # 2*u_i*v_i - u_i^2 = u_i(2*v_i - u_i)
    # v_i^2 - u_i(2*v_i - u_i) = v_i^2 - 2*u_i*v_i + u_i^2 = (v_i - u_i)^2 >= 0
    # So v_i^2 >= 2*u_i*v_i - u_i^2. Good, z >= 2*u_i*v_i - u_i^2 is valid.
    
    # Inequality 3: z <= l_i*v_j + u_j*v_i - l_i*u_j
    #   l_j... wait no, the formula is:
    #   z <= u_i*v_j + l_j*v_i - u_i*l_j  (this is the "l" upper bound)
    #   z <= l_i*v_j + u_j*v_i - l_i*u_j  (this is the "u" upper bound)
    # Wait, let me look up McCormick correctly for z = x*y on [l_x,u_x] x [l_y,u_y]:
    # z >= l_x*y + l_y*x - l_x*l_y
    # z >= u_x*y + u_y*x - u_x*u_y
    # z <= l_x*y + u_y*x - l_x*u_y  ... NO
    # z <= u_x*y + l_y*x - u_x*l_y  ... 
    # Actually the standard McCormick:
    # z >= (l_x + l_y)*z/(l_x*l_y)... no that's not right either.
    # Let me just use the linear underestimators and overestimators:
    # Under: z >= l_x*y + l_y*x - l_x*l_y
    # Under: z >= u_x*y + u_y*x - u_x*u_y
    # Over:  z <= l_x*y + u_y*x - l_x*u_y  ... 
    # Hmm, I think the overestimators are:
    # z <= u_x*y + l_y*x - u_x*l_y
    # z <= l_x*y + u_y*x - l_x*u_y
    
    # Let me verify with i=j, x=y=v, l=0, u=1:
    # z >= 0*v + 0*v - 0 = 0  (trivial)
    # z >= 1*v + 1*v - 1 = 2v-1
    # z <= 1*v + 0*v - 0 = v
    # z <= 0*v + 1*v - 0 = v
    # So z <= v and z >= 2v-1, with 0<=v<=1. Since z=v^2:
    # v^2 <= v for v in [0,1] ✓
    # v^2 >= 2v-1 = (v-1)^2 - ... wait (v-1)^2 = v^2-2v+1, so v^2 >= 2v-1 ⟺ v^2-2v+1>=0 ⟺ (v-1)^2>=0 ✓
    
    # OK so for pair (i,j) with i<=j, bounds [l_i,u_i] for v_i and [l_j,u_j] for v_j:
    # z >= l_i*v_j + l_j*v_i - l_i*l_j  -->  l_j*v_i + l_i*v_j - z <= l_i*l_j
    # z >= u_i*v_j + u_j*v_i - u_i*u_j  -->  -u_j*v_i - u_i*v_j + z >= -u_i*u_j  -->  u_j*v_i + u_i*v_j - z <= u_i*u_j
    # z <= u_i*v_j + l_j*v_i - u_i*l_j  -->  l_j*v_i + u_i*v_j - z <= u_i*l_j
    # z <= l_i*v_j + u_j*v_i - l_i*u_j  -->  u_j*v_i + l_i*v_j - z <= l_i*u_j
    
    # Collect inequality rows
    rows, cols, vals, rhs = [], [], [], []
    
    for i in range(n):
        for j in range(i, n):
            zi = zidx(i, j)
            li, ui = l[i], u[i]
            lj, uj = l[j], u[j]
            vi = i  # variable index for v_i
            vj = j  # variable index for v_j
            zv = n + zi  # variable index for z
            
            # Row 1: l_j*v_i + l_i*v_j - z <= l_i*l_j
            rows.append(len(rhs)); cols.extend([vi, vj, zv]); vals.extend([lj, li, -1.0]); rhs.append(li*lj)
            # Row 2: u_j*v_i + u_i*v_j - z <= u_i*u_j
            rows.append(len(rhs)); cols.extend([vi, vj, zv]); vals.extend([uj, ui, -1.0]); rhs.append(ui*uj)
            # Row 3: l_j*v_i + u_i*v_j - z <= u_i*l_j
            rows.append(len(rhs)); cols.extend([vi, vj, zv]); vals.extend([lj, ui, -1.0]); rhs.append(ui*lj)
            # Row 4: u_j*v_i + l_i*v_j - z <= l_i*u_j
            rows.append(len(rhs)); cols.extend([vi, vj, zv]); vals.extend([uj, li, -1.0]); rhs.append(li*uj)
    
    # Lag constraints: sum_{i in I_k} (v[i] - z[min(i,j),max(i,j)]) - t <= 0
    for k in range(-(n-1), n):
        i_lo = max(0, k)
        i_hi = min(n-1, n-1+k)
        if i_lo > i_hi: continue
        i = np.arange(i_lo, i_hi+1)
        j = i - k
        for ii, jj in zip(i, j):
            mn, mx2 = min(ii,jj), max(ii,jj)
            zi = zidx(mn, mx2)
            rows.append(len(rhs)); cols.extend([ii, n+zi, t_idx]); vals.extend([1.0, -1.0, -1.0]); rhs.append(0.0)
    
    # Equality: sum(v) = S
    r0 = len(rhs)
    # We'll handle equality separately
    
    A_ub = coo_matrix((vals, (rows, cols)), shape=(len(rhs), n_vars)).tocsr()
    b_ub = np.array(rhs)
    
    A_eq_data = np.ones(n)
    A_eq = coo_matrix((A_eq_data, (np.arange(n), np.arange(n))), shape=(1, n_vars)).tocsr()
    b_eq = np.array([S])
    
    # Variable bounds
    lb = np.concatenate([l, np.array([l[i]*l[j] for i in range(n) for j in range(i,n)]), [0.0]])
    ub = np.concatenate([u, np.array([u[i]*u[j] for i in range(n) for j in range(i,n)]), [float(n)]])
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=list(zip(lb, ub)), method='highs',
                  options={'presolve': True, 'time_limit': 60})
    if res is not None and res.status == 0 and res.x is not None:
        return res.x[:n], res.x[t_idx]
    return None, None

def local_lp_search(start, n, S, max_iters=8):
    if not check_feasible(start, n):
        return start, exact_score(start, n)
    current = start.copy()
    best = start.copy()
    best_sc = exact_score(start, n)
    r = 0.20
    fails = 0
    for it in range(max_iters):
        if r < 1e-3 and fails >= 2:
            break
        endpoint, _ = build_lp(current, r, n, S)
        if endpoint is None:
            r *= 0.5
            fails += 1
            continue
        endpoint = np.clip(endpoint, 0.0, 1.0)
        endpoint = project_box_sum(endpoint, S)
        if endpoint is None or not check_feasible(endpoint, n):
            r *= 0.5
            fails += 1
            continue
        if np.max(np.abs(endpoint - current)) < 1e-12:
            r *= 0.5
            fails += 1
            continue
        h_new, sc = line_minimax(current, endpoint, n, S)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best = h_new
            current = h_new
            r = min(0.25, 1.2*r)
            fails = 0
        else:
            r *= 0.5
            fails += 1
    return best, best_sc

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    # Load parent

                tgt = n/2.0
                if abs(np.sum(pv) - tgt) < 1e-6*max(1.0, float(n)):
                    pv = np.clip(pv, 0.0, 1.0)
                    parent = project_box_sum(pv, tgt)
    except Exception:
        pass
    
    S = n/2.0
    if parent is None:
        parent = project_box_sum(np.full(n, 0.5), S)
    
    # Generate starts
    starts = [parent.copy(), parent[::-1].copy(), (1.0-parent).copy()]
    # Perturbed starts
    for amp in [0.04, 0.12]:
        p = project_box_sum(parent + amp*rng.standard_normal(n), S)
        if p is not None: starts.append(p)
    # Harmonic
    freqs = rng.integers(1, max(2, n//2), size=2)
    phases = rng.uniform(0, 2*np.pi, size=2)
    harm = np.zeros(n)
    for f, ph in zip(freqs, phases):
        harm += np.cos(2.0*np.pi*f*np.arange(n)/n + ph)
    hrm = np.sqrt(np.mean(harm**2))
    if hrm > 1e-12:
        hnorm = (harm - hmean) / hrm  # hmean = np.mean(harm)
        p = project_box_sum(parent + 0.08*hnorm, S)
        if p is not None: starts.append(p)
    
    # More random starts
    for _ in range(3):
        p = project_box_sum(rng.random(n), S)
        if p is not None: starts.append(p)
    
    # Deduplicate
    dedup_starts = [starts[0]]
    for s in starts[1:]:
        if not any(np.max(np.abs(s - d)) < 1e-10 for d in dedup_starts):
            dedup_starts.append(s)
    
    # Score all starts
    best_h = parent.copy()
    best_sc = exact_score(parent, n)
    for s in dedup_starts:
        sc = exact_score(s, n)
        if sc < best_sc:
            best_sc = sc
            best_h = s
    
    # LP-based local search
    if n <= 96:
        max_it = 8 if n <= 64 else 5
        for s in dedup_starts:
            h_new, sc = local_lp_search(s, n, S, max_it)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_h = h_new
    
    # Perturbation refinement from best
    for amp in [0.005, 0.02, 0.06, 0.12]:
