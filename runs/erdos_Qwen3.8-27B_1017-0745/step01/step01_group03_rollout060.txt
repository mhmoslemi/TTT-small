import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-14:
        if resid > 0:
            order = np.argsort(1.0 - y)
            for i in order:
                add = min(resid, 1.0 - y[i])
                y[i] += add; resid -= add
                if abs(resid) < 1e-15: break
        else:
            order = np.argsort(y)
            for i in order:
                sub = min(-resid, y[i])
                y[i] -= sub; resid += sub
                if abs(resid) < 1e-15: break
    y = np.clip(y, 0.0, 1.0)
    if abs(np.sum(y) - S) > 1e-9 * max(1.0, float(n)):
        return None
    return y

def exact_score(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    sc = float(np.max(c) * 2.0 / n)
    return sc if np.isfinite(sc) else float('inf')

def check_feasible(h, n):
    if h is None or h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 5e-10 * max(1.0, float(n))

def pair_idx(i, j, n):
    if i > j: i, j = j, i
    return i * n - i * (i - 1) // 2 + (j - i)

def line_minimax(x, y, n, S):
    d = y - x
    n_lags = 2 * n - 1
    A = np.zeros(n_lags); B = np.zeros(n_lags); D = np.zeros(n_lags)
    for k in range(-(n - 1), n):
        idx = k + n - 1
        ilo = max(0, k); ihi = min(n - 1, n - 1 + k)
        if ilo > ihi: continue
        ii = np.arange(ilo, ihi + 1); jj = ii - k
        D[idx] = np.sum(x[ii] * (1.0 - x[jj]))
        B[idx] = np.sum(d[ii] - x[ii] * d[jj] - d[ii] * x[jj])
        A[idx] = -np.sum(d[ii] * d[jj])
    cands = [0.0, 1.0]
    for idx in range(n_lags):
        if abs(A[idx]) > 1e-15:
            a = -B[idx] / (2.0 * A[idx])
            if -1e-10 <= a <= 1.0 + 1e-10:
                cands.append(max(0.0, min(1.0, a)))
    for idx in range(n_lags):
        for ldx in range(idx + 1, n_lags):
            da = A[idx] - A[ldx]; db = B[idx] - B[ldx]; dc = D[idx] - D[ldx]
            if abs(da) < 1e-15:
                if abs(db) > 1e-15:
                    a = -dc / db
                    if -1e-10 <= a <= 1.0 + 1e-10:
                        cands.append(max(0.0, min(1.0, a)))
            else:
                disc = db * db - 4.0 * da * dc
                if disc >= -1e-12:
                    sq = np.sqrt(max(0.0, disc))
                    for a in [(-db + sq) / (2.0 * da), (-db - sq) / (2.0 * da)]:
                        if np.isfinite(a) and -1e-10 <= a <= 1.0 + 1e-10:
                            cands.append(max(0.0, min(1.0, a)))
    cands = np.unique(np.clip(np.array(cands), 0.0, 1.0))
    best_a, best_q = 0.0, float('inf')
    for st in range(0, len(cands), 2048):
        ch = cands[st:st + 2048]
        Q = D + ch[:, None] * (B + ch[:, None] * A)
        mq = np.max(Q, axis=1)
        im = np.argmin(mq)

    for da in [-1e-5, 1e-5, -1e-6, 1e-6]:

def build_lp(x, r, n, S):

            if i != j:
                ru.append(zi); cu.append(i); vu.append(-lj)

            ru.append(zi); cu.append(zi); vu.append(1.0)
            if i != j:

    for k in range(-(n - 1), n):
        ilo = max(0, k); ihi = min(n - 1, n - 1 + k)

    bds.append((0.0, float(n)))
    try:

def local_lp_search(start, n, S, max_att=8):

            cur = nh; cs = ns; fails = 0; r = min(0.25, 1.2 * r)
