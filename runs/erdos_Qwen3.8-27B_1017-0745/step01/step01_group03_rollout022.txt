def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if resid > 1e-14:
        for i in range(n):
            add = min(resid, 1.0 - y[i])
            y[i] += add
            resid -= add
            if resid < 1e-15:
                break
    elif resid < -1e-14:
        for i in range(n):
            sub = min(-resid, y[i])
            y[i] -= sub
            resid += sub
            if -resid < 1e-15:
                break
    y = np.clip(y, 0.0, 1.0)
    if abs(np.sum(y) - S) > 5e-12 * max(1.0, float(n)):
        y[0] += S - np.sum(y)
        y = np.clip(y, 0.0, 1.0)
    return y

def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    sv = np.sort(v)[::-1]
    cssv = np.cumsum(sv)
    rho = np.where(sv - (cssv - 1.0) / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        return np.full(n, 1.0 / n)
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    y = np.maximum(v - theta, 0.0)
    resid = 1.0 - np.sum(y)
    if abs(resid) > 1e-15:
        idx = np.argmax(y)
        y[idx] += resid
        y = np.maximum(y, 0.0)
    return y

def exact_score(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def build_Q_G(h):
    n = len(h)
    L = 2 * n - 1
    Q = np.zeros(L)
    G = np.zeros((L, n))
    for k in range(-(n - 1), n):
        ki = k + n - 1
        i_lo = max(0, k)
        i_hi = min(n - 1, n - 1 + k)
        if i_lo > i_hi:
            continue
        i_arr = np.arange(i_lo, i_hi + 1)
        j_arr = i_arr - k
        Q[ki] = np.sum(h[i_arr] * (1.0 - h[j_arr]))
        G[ki, i_arr] += (1.0 - h[j_arr])
        G[ki, j_arr] += (-h[i_arr])
    return Q, G

def solve_surrogate(x, Q, G, S, n, warm_w):
    L = 2 * n - 1
    if warm_w is not None and np.all(np.isfinite(warm_w)) and np.all(warm_w >= -1e-12):
        w = project_simplex(warm_w.copy())
    else:
        w = np.zeros(L)
        w[np.argmax(Q)] = 1.0
    v = w.copy()
    t = 1.0
    Ldual = max(1e-8, 0.5001 * np.sum(G * G))
    Imax = max(100, min(320, math.ceil(14000.0 / n)))
    gap_check = max(5, Imax // 25)
    best_y = x.copy()
    best_model = float(np.max(Q))
    best_w = w.copy()
    best_phi = -1e30
    for it in range(Imax):
        z = G.T @ w
        y = project_box_sum(x - 0.5 * z, S)
        d = y - x
        p = G @ d
        vals = Q + p
        model = float(np.dot(d, d) + np.max(vals))
        if model < best_model:
            best_model = model
            best_y = y.copy()
        # Dual gradient and phi
        phi = float(np.dot(w, Q) + np.dot(d, d) + np.dot(z, d))
        if phi > best_phi:
            best_phi = phi
            best_w = w.copy()
        gap = float(np.max(vals) - np.dot(w, vals))
        if gap < 0:
            gap = 0.0
        # FISTA gradient step
        grad = Q + p
        w_new = project_simplex(v + grad / Ldual)
        t_new = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t * t))
        v_new = w_new + ((t - 1.0) / t_new) * (w_new - w)
        w, t, v = w_new, t_new, v_new
        if (it + 1) % gap_check == 0:
            z2 = G.T @ w
            y2 = project_box_sum(x - 0.5 * z2, S)
            d2 = y2 - x
            p2 = G @ d2
            vals2 = Q + p2
            model2 = float(np.dot(d2, d2) + np.max(vals2))
            if model2 < best_model:
                best_model = model2
                best_y = y2.copy()
            phi2 = float(np.dot(w, Q) + np.dot(d2, d2) + np.dot(z2, d2))
            gap2 = float(np.max(vals2) - np.dot(w, vals2))
            if gap2 < 0:

                w = best_w
                break
    z = G.T @ best_w
    y = project_box_sum(x - 0.5 * z, S)
    d = y - x
    p = G @ d
    vals = Q + p
    model = float(np.dot(d, d) + np.max(vals))

        best_model = model
        best_y = y.copy()
    if not np.all(np.isfinite(best_y)):
        best_y = x.copy()
    return best_y, best_w

    dd = float(np.dot(d, d))

            cur_score = s_new

def project_box_sum(v, S):

def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    sv = np.sort(v)[::-1]
    cssv = np.cumsum(sv)
    ar = np.arange(1, n + 1, dtype=np.float64)
    rho = np.where(sv - (cssv - 1.0) / ar > 0)[0]
    if len(rho) == 0:
        return np.full(n, 1.0 / n)
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    y = np.maximum(v - theta, 0.0)
    resid = 1.0 - np.sum(y)
    if abs(resid) > 1e-14:
        idx = int(np.argmax(y))
        y[idx] += resid
        y = np.maximum(y, 0.0)
    return y
