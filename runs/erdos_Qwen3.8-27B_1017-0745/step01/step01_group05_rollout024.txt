import numpy as np
import math

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(h.sum() - n / 2.0) <= 1e-10)

def project_capped(y, n):
    S = n / 2.0
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return np.full(n, 0.5), False
    lo = float(y.min()) - 1.0
    hi = float(y.max())
    lam = 0.5 * (lo + hi)
    for _ in range(80):
        c = np.clip(y - lam, 0.0, 1.0)
        r = c.sum() - S
        if abs(r) < 1e-14:
            break
        if r > 0:
            lo = lam
        else:
            hi = lam
        a = ((y - lam > 1e-15) & (y - lam < 1.0 - 1e-15)).sum()
        if a > 0:
            lam_new = lam + r / a
        else:
            lam_new = 0.5 * (lo + hi)
        if np.isfinite(lam_new) and lo < lam_new < hi:
            lam = lam_new
        else:
            lam = 0.5 * (lo + hi)
    c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(3):
        r = S - c.sum()
        if abs(r) < 1e-14:
            break
        if r > 0:
            mask = c < 1.0 - 1e-12
        else:
            mask = c > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        c[mask] += r / cnt
        np.clip(c, 0.0, 1.0, out=c)
    ok = check_feasible(c, n)
    return c, ok

def lag_scores(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    qs = lag_scores(h, n)
    if qs.shape != (2 * n - 1,) or not np.all(np.isfinite(qs)):
        return float('inf')
    return float(qs.max())

def build_G(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        row = k + n - 1
        if k >= 0:
            # First term: j from k to n-1
            G[row, k:] += dx * (1.0 - h[:n - k])
            # Second term: j from 0 to n-k-1
            G[row, :n - k] -= dx * h[k:]
        else:
            m = -k
            # First term: j from 0 to n-1-m
            G[row, :n - m] += dx * (1.0 - h[m:])
            # Second term: j from m to n-1
            G[row, m:] -= dx * h[:n - m]
    return G

def project_simplex(z):
    n = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(n, 1.0 / n)
    sz = np.sort(z)[::-1]
    cs = np.cumsum(sz)
    rho = np.arange(1, n + 1)
    cond = sz - (cs - 1.0) / rho > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    r = int(np.max(np.where(cond)[0]))
    theta = (cs[r] - 1.0) / (r + 1)
    p = np.maximum(z - theta, 0.0)
    s = p.sum()
    if s > 0:
        p /= s
        # fix roundoff
        p[p > 0] += (1.0 - p.sum()) / max(1, (p > 0).sum())
    return np.maximum(p, 0.0)

def dual_state(p, h, q, G, L, n):
    gp = G.T @ p
    y, ok = project_capped(h - gp / L, n)
    if not ok:
        return None
    d = y - h
    u = q + G @ d
    D = float(p @ u + (L / 2.0) * (d @ d))
    return d, u, D

    L = 2.0 * dx
    Ldual = max(1e-12, 1.05 * float(np.sum(G * G)) / L)
    # Initialize p on active lags

    m = len(q)

    p[active] = 1.0 / len(active)

    y_vec = p.copy()
    theta = 1.0
    best_D = -float('inf')
    best_d = None
    best_u = None
    stall = 0

        if st is None:

            st3 = dual_state(x_trial2, h, q, G, L, n)
            if st3 is not None and st3[2] >= best_D:
                x_trial, st2 = x_trial2, st3
                d_t, u_t, D_t = st2
            else:
                # Keep best
                break
        if D_t > best_D:
            best_D = D_t
            best_d = d_t.copy()
            best_u = u_t.copy()
        # Update
        if D_t < best_D - 1e-13:
            x = x_trial.copy()
            y_vec = x.copy()
            theta = 1.0
        else:
            theta_new = (1.0 + math.sqrt(1.0 + 4.0 * theta * theta)) / 2.0

        # Stopping
        gap = float(best_u.max() - p[active[0]] @ best_u)  # Hmm this is wrong

            stall += 1

                break
        else:

        p[idx] = 1.0
        st = dual_state(p, h, q, G, L, n)

    p = np.maximum(z - theta, 0.0)

        pos = p > 0
        p[pos] += (1.0 - s) / max(1, int(pos.sum()))

        d = solve_majorant(h, q, G, n, max_inner)
        if d is None or np.max(np.abs(d)) < 1e-10:
            break
        v = G @ d
        norm2 = float(d @ d)
        L = 4.0 / n
        qmax = float(q.max())
        prev_s = s
        accepted = False
        for hh in range(24):
            alpha = 1.0 / (2.0 ** hh)
            if alpha < 1e-15:
                break
            M_val = float(np.max(q + alpha * v)) + 0.5 * L * alpha * alpha * norm2
            if M_val >= qmax - 1e-14:
                continue
            ht = h + alpha * d
            ht, tok = project_capped(ht, n)
            if not tok:
                continue
            st = exact_score(ht, n)
            if st < s - 1e-12:
                h = ht
                s = st
                if s < best_s - 1e-12:
                    best_s = s
                    best_h = h.copy()
                small = small + 1 if (prev_s - st) < 1e-11 else 0
                accepted = True
                break
        if not accepted:
            break
        if small >= 2:
            break
    return best_h

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = globals().get('initial_h_values', None)
    n = 45
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.ndim == 1 and 2 <= len(p) <= 999 and np.all(np.isfinite(p)) and np.max(np.abs(p)) < 1e6:
            n = len(p)
            fallback, fok = project_capped(p.copy(), n)
            if not fok:
                fallback = np.full(n, 0.5)
        else:
            fallback = np.full(n, 0.5)
    else:
        nk = kwargs.get('n_points', None)
        if isinstance(nk, int) and not isinstance(nk, bool) and 2 <= nk <= 999:
            n = nk
        fallback = np.full(n, 0.5)
    best_h = fallback.copy()
    best_s = exact_score(best_h, n)
    if n <= 100:
        mo, mi, ns = 80, 80, 7
    elif n <= 400:
        mo, mi, ns = 12, 20, 4
    else:
        mo, mi, ns = 4, 10, 2
    starts = [fallback.copy(), 1.0 - fallback, fallback[::-1].copy(),
              1.0 - fallback[::-1].copy(),
              fallback + 0.05 * rng.standard_normal(n),
              fallback + 0.15 * rng.standard_normal(n),
              rng.uniform(0, 1, n)]
    starts = starts[:ns]
    ustarts = []
    for st in starts:
        dup = any(np.max(np.abs(st - us)) < 1e-8 for us in ustarts)
        if not dup:
            ustarts.append(st)
    for st in ustarts:
        try:
            bh = optimize_start(st, n, mo, mi)
            bh, bok = project_capped(bh, n)
            if bok:
                bs = exact_score(bh, n)
                if bs < best_s - 1e-12:
                    best_s = bs
                    best_h = bh.copy()
        except Exception:
            pass
    if n <= 100:
        try:
            rst = best_h + 0.025 * rng.standard_normal(n)
            bh = optimize_start(rst, n, mo // 2, mi)
            bh, bok = project_capped(bh, n)
            if bok:
                bs = exact_score(bh, n)
                if bs < best_s - 1e-12:
                    best_s = bs
                    best_h = bh.copy()
        except Exception:
            pass
    final_h, fok = project_capped(best_h, n)
    if not fok:
        final_h = fallback.copy()
    fs = exact_score(final_h, n)
    fbs = exact_score(fallback, n)
    if fs > fbs + 1e-12:
        final_h = fallback.copy()
        fs = fbs
    final_h, fok2 = project_capped(final_h, n)
    if not fok2:
        final_h = fallback.copy()
        fs = exact_score(fallback, n)
    else:
        fs = exact_score(final_h, n)
    return final_h.astype(np.float64), float(fs), int(n)
