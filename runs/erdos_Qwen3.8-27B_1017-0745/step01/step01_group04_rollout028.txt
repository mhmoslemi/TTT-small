def lag_data(h, N):
    L = 2 * N - 1
    q = np.zeros(L)
    G = np.zeros((L, N))
    for s in range(-(N-1), N):
        idx = s + N - 1
        if s >= 0:
            ell = N - s
            q[idx] = np.dot(h[:ell], 1.0 - h[s:s+ell])
            G[idx, :ell] += (1.0 - h[s:s+ell])
            G[idx, s:s+ell] -= h[:ell]
        else:
            d = -s
            ell = N - d
            q[idx] = np.dot(h[d:], 1.0 - h[:ell])
            G[idx, d:] += (1.0 - h[:ell])
            G[idx, :ell] -= h[d:]
    return q, G
