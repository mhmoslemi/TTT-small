            if hc is None:
                continue
            ss = exact_score(hc, n)
            if ss < snap_s[i] - 1e-12:
                npop[i] = t
                nscores[i] = ss
        pop = npop
        scores = nscores
        improved = False
        for i in range(P):
            if scores[i] < best_s - 1e-12:
                h = decode_cand(anchor, B, sigma, pop[i], n, d)
                if h is not None:
                    s2 = exact_score(h, n)
                    if s2 < best_s - 1e-12:
                        best_h = h
                        best_s = s2
                        improved = True
        if improved:
            stag = 0
        else:
            stag += 1
        if stag >= 4 and not restarted and gen < G - 1:
            restarted = True
            stag = 0
            best_u = np.zeros(d)
            for i in range(P):
                if scores[i] <= best_s + 1e-10:
                    best_u = pop[i].copy()
                    break
            pop[0] = best_u
            for i in range(1, P):
                pop[i] = reflect_val(best_u + 0.35 * rng.standard_normal(d))
                h = decode_cand(anchor, B, sigma, pop[i], n, d)
                scores[i] = exact_score(h, n) if h is not None else float('inf')
    return best_h, best_s

def local_polish(h0, n, rng, iters=8):
    h = h0.copy()
    s = exact_score(h, n)
    for _ in range(iters):
        improved = False
        for i in range(n):
            for delta in [0.02, -0.02, 0.05, -0.05, 0.1, -0.1]:
                trial = h.copy()
                trial[i] = np.clip(trial[i] + delta, 0.0, 1.0)
                r = n / 2.0 - trial.sum()
                if abs(r) < 1e-14:
                    continue
                if r > 0:
                    cap = 1.0 - trial
                    tot = cap.sum()
                    if tot < 1e-14:
                        continue
                    trial += r * cap / tot
                else:
                    tot = trial.sum()
                    if tot < 1e-14:
                        continue
                    trial += r * trial / tot
                np.clip(trial, 0.0, 1.0, out=trial)
                if not feasible(trial, n):
                    continue
                ts = exact_score(trial, n)
                if ts < s - 1e-12:
                    h = trial
                    s = ts
                    improved = True
                    break
        if not improved:
            break
    return h, s

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = globals().get('initial_h_values', None)
    n = 45
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.ndim == 1 and 2 <= len(p) <= 999 and np.all(np.isfinite(p)) and np.max(np.abs(p)) < 1e6:
            n = len(p)
            fallback, ok = project_capped(p.copy(), n)
            if not ok:
                fallback = np.full(n, 0.5)
        else:
            fallback = np.full(n, 0.5)
    else:
        np_kw = kwargs.get('n_points', None)
        if isinstance(np_kw, int) and not isinstance(np_kw, bool) and 2 <= np_kw <= 999:
            n = np_kw
        fallback = np.full(n, 0.5)
    best_h = fallback.copy()
    best_s = exact_score(fallback, n)
    B, d = make_basis(n)
    if d >= 2:
        z = rng.standard_normal(n)
        z -= z.mean()
        zn = math.sqrt(float(z @ z))
        alt = fallback.copy()
        if zn > 1e-10:
            alt_try, ok = project_capped(fallback + 0.12 * z / zn, n)
            if ok:
                alt = alt_try
        alt_s = exact_score(alt, n)
        if alt_s < best_s - 1e-12:
            best_s = alt_s
            best_h = alt.copy()
        if n <= 100:
            P = max(12, min(48, 2 * d + 8))
            G = 18
        elif n <= 400:
            P = max(10, min(32, d + 8))
            G = 6
        else:
            P = max(8, min(16, d))
            G = 3
        for anchor, sig in [(fallback, 0.10), (alt, 0.18), (best_h, 0.05)]:
            try:
                sh, ss = evolve_stage(anchor, B, sig, P, G, rng, n, d)
                if sh is not None:
                    s2 = exact_score(sh, n)
                    if s2 < best_s - 1e-12:
                        best_s = s2
                        best_h = sh
            except Exception:
                pass
    try:
        ph, ps = local_polish(best_h, n, rng, iters=10)
        if ps < best_s - 1e-12:
            best_s = ps
            best_h = ph
    except Exception:
        pass
    final_h, ok = project_capped(best_h.copy(), n)
    if not ok:
        final_h = fallback
    final_s = exact_score(final_h, n)
    fb_s = exact_score(fallback, n)
    if not np.isfinite(final_s) or final_s > fb_s + 1e-12:
        final_h = fallback.copy()
        final_s = fb_s
    final_s = exact_score(final_h, n)
    if not np.isfinite(final_s):
        final_h = np.full(n, 0.5)
        final_s = exact_score(final_h, n)
    return final_h.astype(np.float64), float(final_s), int(n)
