import numpy as np
import math

def proj_capped_simplex(y, S):
    y = np.asarray(y, dtype=np.float64)
    if y.ndim != 1 or not np.all(np.isfinite(y)) or S < 0 or S > len(y):
        return None
    if np.max(np.abs(y)) > 1e12:
        return None
    n = len(y)
    lo, hi = np.min(y) - 1.0, np.max(y)
    for _ in range(80):
        mid = lo + 0.5 * (hi - lo)
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = lo + 0.5 * (hi - lo)
    x = np.clip(y - lam, 0.0, 1.0)
    r = S - np.sum(x)
    if r > 0:
        for i in range(n):
            if r <= 0: break
            a = min(r, 1.0 - x[i]); x[i] += a; r -= a
    elif r < 0:
        for i in range(n):
            if r >= 0: break
            a = min(-r, x[i]); x[i] -= a; r += a
    if abs(np.sum(x) - S) > 1e-10 or not np.all(np.isfinite(x)):
        return None
    return np.clip(x, 0.0, 1.0)

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-10)

def exact_score(h, n):
    try:
        vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if vals.shape[0] != 2 * n - 1 or not np.all(np.isfinite(vals)):
            return None
        return float(np.max(vals))
    except Exception:
        return None

def build_dct_basis(n, D):
    D = min(D, n - 1)
    if D < 1:
        return None
    L = math.ceil(D / 2)
    K = list(range(1, L + 1))
    R = D - L
    if R > 0:
        if R == 1:
            K.append(n - 1)
        else:
            raw = np.linspace(L + 1, n - 1, R)
            for v in raw:
                iv = int(round(v))
                if iv not in K:
                    K.append(iv)
    i = L + 1
    while len(K) < D and i <= n - 1:
        if i not in K:
            K.append(i)
        i += 1
    K = sorted(set(K))[:D]
    D = len(K)
    idx = np.arange(n, dtype=np.float64) + 0.5
    B = np.zeros((n, D))
    for j, k in enumerate(K):
        B[:, j] = np.sqrt(2.0 / n) * np.cos(np.pi * k * idx / n)
    B -= B.mean(axis=0, keepdims=True)
    Q, _ = np.linalg.qr(B)
    if Q.shape != (n, D) or not np.all(np.isfinite(Q)):
        return None
    return Q[:, :D]

def cma_run(anchor, basis, S, n, rng, gen_cap, patience, sigma_init):
    D = basis.shape[1]
    P = 4 + int(3.0 * math.log(D)) if D > 1 else 4
    mu = max(2, P // 2)
    w = np.array([math.log(mu + 0.5) - math.log(i + 1) for i in range(mu)])
    w /= w.sum()
    mu_eff = 1.0 / np.sum(w ** 2)
    c_sigma = (mu_eff + 2.0) / (D + mu_eff + 5.0)
    d_sigma = 1.0 + 2.0 * max(0.0, math.sqrt((mu_eff - 1.0) / (D + 1.0)) - 1.0) + c_sigma
    c_c = (4.0 + mu_eff / D) / (D + 4.0 + 2.0 * mu_eff / D)
    c1 = 2.0 / ((D + 1.3) ** 2 + mu_eff)
    c_mu = min(1.0 - c1, 2.0 * (mu_eff - 2.0 + 1.0 / mu_eff) / ((D + 2.0) ** 2 + mu_eff))
    chi_D = math.sqrt(D) * (1.0 - 1.0 / (4.0 * D) + 1.0 / (21.0 * D * D))
    m = np.zeros(D)
    C = np.eye(D)
    p_sigma = np.zeros(D)
    p_c = np.zeros(D)
    sigma = max(0.05, min(0.70, sigma_init))
    best_sc = None
    no_improve = 0
    for g in range(gen_cap):
        eigvals, Vt = np.linalg.eigh(C)
        if not np.all(np.isfinite(eigvals)) or not np.all(np.isfinite(Vt)):
            break
        eigvals = np.clip(eigvals, 1e-8, 1e8)
        A = (Vt.T * np.sqrt(eigvals)).T
        C_invhalf = Vt.T @ np.diag(1.0 / np.sqrt(eigvals)) @ Vt
        z_list, h_list, sc_list, step_list = [], [], [], []
        for _ in range(P):
            r = rng.standard_normal(D)
            step = A @ r
            z = m + sigma * step
            h_c = proj_capped_simplex(anchor + basis @ z, S)
            if h_c is None:
                continue
            sc = exact_score(h_c, n)
            if sc is None:
                continue
            z_list.append(z); h_list.append(h_c); sc_list.append(sc); step_list.append(step)
        if len(sc_list) < mu:
            break
        order = np.argsort(sc_list, kind='stable')[:mu]
        sorted_z = np.array([z_list[i] for i in order])
        sorted_s = [sc_list[i] for i in order]
        sorted_h = [h_list[i] for i in order]
        sorted_step = np.array([step_list[i] for i in order])
        run_best = sorted_s[0]
        if best_sc is None or run_best < best_sc - 1e-12:
            best_sc = run_best
            no_improve = 0
        else:
            no_improve += 1
        yield sorted_h, sorted_s, best_sc
        if no_improve >= patience:
            break
        if sigma < 1e-3 and no_improve >= 8:
            break
        m_old = m.copy()
        m_new = w @ sorted_z
        y_w = (m_new - m_old) / sigma
        p_sigma = (1 - c_sigma) * p_sigma + math.sqrt(c_sigma * (2 - c_sigma) * mu_eff) * (C_invhalf @ y_w)
        h_sigma = 1.0 if np.linalg.norm(p_sigma) / math.sqrt(1.0 - (1 - c_sigma) ** (2 * (g + 1))) < (1.4 + 2.0 / (D + 1.0)) * chi_D else 0.0
        p_c = (1 - c_c) * p_c + h_sigma * math.sqrt(c_c * (2 - c_c) * mu_eff) * y_w
        C = (1 - c1 - c_mu) * C + c1 * (np.outer(p_c, p_c) + (1 - h_sigma) * c_c * (2 - c_c) * C)
        for i in range(mu):
            C += c_mu * w[i] * np.outer(sorted_step[i], sorted_step[i])
        C = 0.5 * (C + C.T)
        exp_arg = c_sigma / d_sigma * (np.linalg.norm(p_sigma) / chi_D - 1.0)
        exp_arg = max(-1.0, min(1.0, exp_arg))
        sigma = max(1e-3, min(1.5, sigma * math.exp(exp_arg)))
        m = m_new
        if not np.all(np.isfinite(m)) or not np.all(np.isfinite(C)) or not np.isfinite(sigma):
