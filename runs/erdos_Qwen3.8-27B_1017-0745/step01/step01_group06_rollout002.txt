def project_K(y, n, S):
    if not np.all(np.isfinite(y)):
        return np.full(n, S / n)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    err = S - h.sum()
    if abs(err) > 1e-12:
        if err > 0:
            cap = 1.0 - h
            idx = np.where(cap > 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) > 0:
            h[idx] = np.clip(h[idx] + err / len(idx), 0.0, 1.0)
    if abs(h.sum() - S) > 1e-8:
        return np.full(n, S / n)
    return h

def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

            h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_grad_matrix(h, n, dx):
    nl = 2 * n - 1
    G = np.zeros((nl, n))
    j = np.arange(n)
    for k in range(nl):
        ell = k - (n - 1)
        i1 = j - ell
        v1 = (i1 >= 0) & (i1 < n)
        i2 = j + ell
        v2 = (i2 >= 0) & (i2 < n)
        t1 = np.zeros(n); t1[v1] = 1.0 - h[i1[v1]]
        t2 = np.zeros(n); t2[v2] = h[i2[v2]]
        G[k] = dx * (t1 - t2)
    return G

def qp_obj(z, n, eta):
    d = z[:n]
    return z[n] + np.dot(d, d) / (2.0 * eta)

def qp_obj_jac(z, n, eta):
    d = z[:n]
    g = np.concatenate([d / eta, np.array([1.0])])
    return g

    d = z[:n]
    return z[n] - q - G @ d

    nl = len(q)
    J = np.zeros((nl, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

    y = np.zeros(m)
    for j in range(m):
        lo_old = max(0, int(j * n0 / m))

            ov = max(0.0, min(a1, b1) - max(a0, b0))

import numpy as np
from scipy.optimize import minimize

def project_K(y, n, S):
    if not np.all(np.isfinite(y)):
        return np.full(n, S / n)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    err = S - h.sum()
    if abs(err) > 1e-12:
        if err > 0:
            idx = np.where(1.0 - h > 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) > 0:
            h[idx] = np.clip(h[idx] + err / len(idx), 0.0, 1.0)
    if abs(h.sum() - S) > 1e-8:
        return np.full(n, S / n)
    return h

def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def lag_vec(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_grad_matrix(h, n, dx):
    nl = 2 * n - 1
    G = np.zeros((nl, n))
    j = np.arange(n)
    for k in range(nl):
        ell = k - (n - 1)
        i1 = j - ell
        v1 = (i1 >= 0) & (i1 < n)
        i2 = j + ell
        v2 = (i2 >= 0) & (i2 < n)
        t1 = np.zeros(n); t1[v1] = 1.0 - h[i1[v1]]
        t2 = np.zeros(n); t2[v2] = h[i2[v2]]
        G[k] = dx * (t1 - t2)
    return G

def qp_obj(z, n, eta):
    d = z[:n]
    return z[n] + np.dot(d, d) / (2.0 * eta)

def qp_obj_jac(z, n, eta):
    g = np.empty(n + 1)
    g[:n] = z[:n] / eta
    g[n] = 1.0
    return g

def qp_ineq(z, n, q, G):
    d = z[:n]
    return z[n] - q - G @ d

def qp_ineq_jac(z, n, q, G):
    nl = len(q)
    J = np.zeros((nl, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def qp_eq(z, n):
    return z[:n].sum()

def qp_eq_jac(z, n):
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def transfer_parent(hp, n0, m):
    if n0 == m:
        return hp.copy()
    y = np.zeros(m)
    for j in range(m):
        total = 0.0
        for i in range(n0):
            ov = max(0.0, min((i + 1) / n0, (j + 1) / m) - max(i / n0, j / m))
            total += hp[i] * ov
        y[j] = m * total
    return y

def make_starts(base, n, S, rng, has_par):
    starts = [base.copy(), (1.0 - base).copy(), base[::-1].copy(), (1.0 - base[::-1]).copy()]
    for sig in [0.06, 0.15, 0.30]:
        starts.append(project_K(base + sig * rng.standard_normal(n), n, S))
    starts.append(project_K(rng.uniform(0, 1, n), n, S))
    starts.append(project_K(rng.beta(0.3, 0.3, n), n, S))
    jj = np.arange(n)
    starts.append(project_K(0.5 + 0.45 * np.cos(2 * np.pi * jj / n), n, S))
    starts.append(project_K(0.5 + 0.4 * np.sin(4 * np.pi * jj / n) + 0.15 * np.cos(6 * np.pi * jj / n), n, S))
    blk = np.zeros(n)
    blk[:int(S)] = 1.0
    if int(S) < n:
        blk[int(S)] = S - int(S)
    starts.append(project_K(blk, n, S))
    if has_par:
        starts.append(project_K(base[rng.permutation(n)], n, S))
    starts.append(project_K(base + 0.5 * (base - base.mean()), n, S))
    dedup = []
    for s in starts:
        if not feasible(s, n, S):
            continue
        dup = False
        for ds in dedup:
            if np.max(np.abs(s - ds)) <= 1e-6:
                dup = True
                break
        if not dup:
            dedup.append(s)
    return dedup

def trust_region_local(h0, n, dx, max_iter, slm, d0, e0):
    S = n / 2.0
    h = project_K(h0.copy(), n, S)
    delta, eta = d0, e0
    cur = exact_score(h, dx)
    crej, cfail = 0, 0
    for _ in range(max_iter):
        q = lag_vec(h, dx)
        G = lag_grad_matrix(h, n, dx)
        qmax = float(np.max(q))
        z0 = np.concatenate([np.zeros(n), [qmax + 1e-10]])
        bounds = [(max(-delta, -h[j]), min(delta, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        cons = [{'type': 'ineq', 'fun': qp_ineq, 'jac': qp_ineq_jac, 'args': (n, q, G)},
                {'type': 'eq', 'fun': qp_eq, 'jac': qp_eq_jac, 'args': (n,)}]
        try:
            res = minimize(qp_obj, z0, jac=qp_obj_jac, args=(n, eta), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': slm, 'ftol': 1e-10, 'disp': False})
            if res is None or not hasattr(res, 'x') or not np.all(np.isfinite(res.x)):
                raise ValueError
            d = res.x[:n]
            for j in range(n):
                if d[j] < max(-delta, -h[j]) - 1e-5 or d[j] > min(delta, 1.0 - h[j]) + 1e-5:
                    raise ValueError
            if abs(d.sum()) > 5e-4:
                raise ValueError
            ht = project_K(h + d, n, S)
            if not feasible(ht, n, S):
                raise ValueError
            da = ht - h
            ns = exact_score(ht, dx)
            if not np.isfinite(ns):
                raise ValueError
            pred = qmax - float(np.max(q + G @ da))
            act = qmax - ns
            if act > 1e-12 and (pred <= 1e-12 or act / max(pred, 1e-15) >= 0.08):
                h, cur = ht, ns
                crej = cfail = 0
                r = act / max(pred, 1e-15) if pred > 1e-12 else 0.5
                if pred <= 1e-12 or r < 0.25:
                    delta *= 0.5
                elif r > 0.7 and np.max(np.abs(da)) >= 0.75 * delta:
                    delta = min(0.4, 1.8 * delta)
                    eta = min(2.0, 1.4 * eta)
                if np.max(np.abs(da)) < 1e-8:
                    break
            else:
                crej += 1
                delta *= 0.5
                eta = max(1e-4, eta * 0.6)
                if crej >= 7:
                    break
        except Exception:
            delta *= 0.5
            cfail += 1
            if cfail >= 4:
                break
        if delta < 1e-5:
            break
    h = project_K(h, n, S)
    return h, exact_score(h, dx)

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    init = globals().get('initial_h_values', None)

    best_h = project_K(best_h, best_n, best_n / 2.0)

    return (best_h, float(fs), int(best_n))
