def smooth_grad(h, tau, n):
    n_lags = 2 * n - 1
    Q = np.zeros(n_lags)
    for k in range(-(n - 1), n):
        i_lo, i_hi = max(0, k), min(n - 1, n - 1 + k)
        if i_lo > i_hi:
            continue
        i = np.arange(i_lo, i_hi + 1)
        Q[k + n - 1] = np.sum(h[i] * (1.0 - h[i - k]))
    m = float(np.max(Q))
    e = np.exp((Q - m) / tau)
    s = float(np.sum(e))
    if s <= 0 or not np.isfinite(s):
        return None
    w = e / s
    W = np.zeros(3 * n - 1)
    W[n - 1:3 * n - 2] = w
    A = np.zeros(3 * n - 1)
    A[n - 1:2 * n - 1] = 1.0 - h
    B = np.zeros(3 * n - 1)
    B[n - 1:2 * n - 1] = h
    c1 = fftconvolve(W, A, mode='full')
    c2 = fftconvolve(W, B, mode='full')
    grad = np.zeros(n)
    Q_len = len(Q)
    off1 = len(c1) - Q_len + n - 1
    off2 = len(c2) - Q_len + n - 1
    for r in range(n):
        v1, v2 = 0.0, 0.0
        for k in range(max(0, off1 - r), min(off1 - r + Q_len, Q_len)):
            kk = r - off1 + k
            if 0 <= kk < n:
                v1 += w[k] * (1.0 - h[kk])
        for kk2 in range(max(0, -off2 + r), min(n, Q_len - off2 + r)):
            k = off2 + r - kk2
            if 0 <= k < Q_len:
                v2 += w[k] * h[kk2]
        grad[r] = v1 - v2
    if not np.all(np.isfinite(grad)):
        return None
    return grad

def gen_harmonic_noise(n, rng):
    freqs = rng.integers(1, max(2, n // 2), size=2)
    phases = rng.uniform(0, 2.0 * np.pi, size=2)
    coeffs = rng.uniform(0.5, 1.5, size=2)
    idx = np.arange(n).astype(np.float64)
    v = np.zeros(n)
    for f, ph, cf in zip(freqs, phases, coeffs):
        v += cf * np.cos(2.0 * np.pi * f * idx / n + ph)
    v -= v.mean()
    rms = np.sqrt(np.mean(v ** 2))
    if rms < 1e-10:
        return rng.standard_normal(n)
    return v / rms

def gen_gaussian_noise(n, rng):
    v = rng.standard_normal(n)
    v -= v.mean()
    rms = np.sqrt(np.mean(v ** 2))
    return v / rms if rms > 1e-10 else v
