def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    z = z - np.median(z)
    if np.max(np.abs(z)) > 1e12:
        return np.full(N, 0.5)
    lo, hi = z.min() - 1.0, z.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    for _ in range(2):
        if abs(r) < 1e-12:
            break
        for i in range(N):
            if r > 0:
                adj = min(r, 1.0 - h[i])
                h[i] += adj
                r -= adj
            else:
                adj = min(-r, h[i])
                h[i] -= adj
                r += adj
            if abs(r) < 1e-14:
                break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1 + 1e-10:
        return np.full(N, 0.5)
    if abs(h.sum() - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return h

def exact_score(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    c = np.correlate(h, 1.0 - h, mode='full')
    s = float(np.max(c) * 2.0 / N)
    return s if np.isfinite(s) else np.inf

def lag_data(h, N):
    L = 2 * N - 1
    q = np.zeros(L)
    G = np.zeros((L, N))
    for s in range(-(N - 1), N):
        r = s + N - 1
        if s >= 0:
            ell = N - s
            q[r] = np.dot(h[:ell], 1.0 - h[s:s + ell])
            G[r, :ell] += 1.0 - h[s:s + ell]
            G[r, s:s + ell] -= h[:ell]
        else:

            ell = N - d
            q[r] = np.dot(h[d:], 1.0 - h[:ell])
            G[r, d:] += 1.0 - h[:ell]
            G[r, :ell] -= h[d:]
    return q, G

    x, N, m = x_data
    y = z[:N]
    return np.sum((y - x) ** 2) + z[N]

def mm_obj_jac(z, x_data):
    x, N, m = x_data
    g = np.zeros(N + 1)
    g[:N] = 2.0 * (z[:N] - x)
    g[N] = 1.0
    return g

    y = z[:N]
    return z[N] - q - G @ (y - x)

    x, q, G, N = x_data
    L = 2 * N - 1
    J = np.zeros((L, N + 1))
    J[:, :N] = -G
    J[:, N] = 1.0
    return J

    x, N, m = x_data
    return np.array([np.sum(z[:N]) - m])

def mm_eq_jac(z, x_data):
    x, N, m = x_data
    J = np.zeros((1, N + 1))
    J[0, :N] = 1.0
    return J

    m = None
    raw = globals().get('initial_h_values', None)
    if raw is not None:
        try:
            raw = np.asarray(raw, dtype=np.float64)
            if raw.ndim == 1 and raw.size >= 2 and np.all(np.isfinite(raw)):
                N = raw.size
                m = float(N) / 2.0
        except Exception:
            raw = None
    if N is None:
        N = kwargs.get('n_points', 70)
        if not isinstance(N, (int, np.integer)) or isinstance(N, bool) or N < 2 or N > 999:
            N = 70
        m = float(N) / 2.0
    if N == 1:
        h = np.array([0.5])
        return h, float(exact_score(h, 1)), 1

    if raw is not None and np.all(np.isfinite(raw)):
        if raw.min() >= -1e-12 and raw.max() <= 1 + 1e-12 and abs(raw.sum() - m) < 1e-7:
            baseline = np.clip(raw.copy(), 0.0, 1.0)
        else:
            baseline = project_box_sum(raw, m)
    else:
        baseline = np.full(N, 0.5)
    baseline = project_box_sum(baseline, m)
    baseline_s = exact_score(baseline, N)
    rng = np.random.default_rng(seed)

    starts = [baseline.copy()]
    starts.append(project_box_sum(1.0 - baseline, m))
    starts.append(project_box_sum(baseline[::-1].copy(), m))
    sh = max(1, N // 3)
    starts.append(project_box_sum(np.roll(baseline, sh), m))
    for sc in [0.04, 0.08, 0.15, 0.25]:
        starts.append(project_box_sum(baseline + sc * rng.standard_normal(N), m))
    starts.append(project_box_sum(rng.uniform(0.0, 1.0, size=N), m))
    starts.append(project_box_sum(fourier_start(N, rng, 5, 0.10), m))
    starts.append(project_box_sum(fourier_start(N, rng, 3, 0.18), m))
    starts.append(project_box_sum(fourier_start(N, rng, 7, 0.06), m))

    unique_starts = []
    for st in starts:
        dup = False
        for u in unique_starts:
            if np.max(np.abs(st - u)) <= 1e-10:
                dup = True
                break
        if not dup:
            unique_starts.append(st)
    starts = unique_starts[:8]

    best = baseline.copy()
    best_s = baseline_s
    for h0 in starts:
        s0 = exact_score(h0, N)
        if s0 < best_s - 1e-12:
            best = h0.copy()
            best_s = s0
        h_opt, s_opt = mm_run(h0, N, m, max_iter=40)
        if s_opt < best_s - 1e-12:
            best = h_opt.copy()
            best_s = s_opt

    for sc in [0.03, 0.06, 0.015]:
        hop = project_box_sum(best + sc * rng.standard_normal(N), m)
        h_opt, s_opt = mm_run(hop, N, m, max_iter=28)
        if s_opt < best_s - 1e-12:
            best = h_opt.copy()
            best_s = s_opt

    for sc in [0.02, 0.04]:
        hop = project_box_sum(best + sc * rng.standard_normal(N), m)
        h_opt, s_opt = mm_run(hop, N, m, max_iter=32)
        if s_opt < best_s - 1e-12:
            best = h_opt.copy()
            best_s = s_opt

    best = project_box_sum(best, m)
    c5 = exact_score(best, N)
    base_final = project_box_sum(baseline, m)
    base_s = exact_score(base_final, N)
    if not np.isfinite(c5) or c5 >= base_s - 1e-12:
        best = base_final
        c5 = base_s
    if not np.isfinite(c5):
        best = np.full(N, 0.5)
        c5 = exact_score(best, N)
    return np.asarray(best, dtype=np.float64), float(c5), int(N)
