            best_x = h_to_x(h_mean, B, N)
            sth = h_mean.copy()
        order = np.argsort(S, kind='stable')
        sel = order[:mu_sel]
        mu_new = np.zeros(d)
        for i in range(mu_sel):
            mu_new += w[i] * X[sel[i]]
        h_m2 = u + B @ mu_new
        h_m2 = project_box_sum(h_m2, m)
        mu_new = h_to_x(h_m2, B, N)
        y_w = (mu_new - mu) / sigma
        p_sigma = (1 - c_sigma) * p_sigma + sqrt(c_sigma * (2 - c_sigma) * mu_eff) * (c_inv_sqrt @ y_w)
        t = gen + 1
        h_sigma = 1.0 if np.linalg.norm(p_sigma) / sqrt(1 - (1 - c_sigma) ** (2 * t)) < (1.4 + 2.0 / (d + 1)) * chi_d else 0.0
        p_c = (1 - c_c) * p_c + h_sigma * sqrt(c_c * (2 - c_c) * mu_eff) * y_w
        cov_step = np.zeros((d, d))
        for i in range(mu_sel):
            y_i = (X[sel[i]] - mu) / sigma
            cov_step += w[i] * np.outer(y_i, y_i)
        C = (1 - c1 - c_mu) * C + c1 * (np.outer(p_c, p_c) + (1 - h_sigma) * c_c * (2 - c_c) * C) + c_mu * cov_step
        C = 0.5 * (C + C.T)
        exp_arg = np.clip((c_sigma / d_sigma) * (np.linalg.norm(p_sigma) / chi_d - 1.0), -1.0, 1.0)
        sigma = py_max(1e-4, py_min(0.5, sigma * np.exp(exp_arg)))
        mu = mu_new
        gen += 1
        if best_score < gen_start_best - 1e-12:
            stag = 0
        else:
            stag += 1
            if stag >= stag_limit:
                break
        if not np.all(np.isfinite(C)):
            C = np.eye(d)
            p_sigma = np.zeros(d)
            p_c = np.zeros(d)
            sigma = py_max(1e-4, sigma * 0.5)
    h_best = u + B @ best_x
    h_best = project_box_sum(h_best, m)
    s_best = score_exact(h_best, N)
    return h_best, s_best

def refine_pair_swap(h, m, N, rng, n_iter):
    best = h.copy()
    best_s = score_exact(best, N)
    for _ in range(n_iter):
        i = rng.integers(N)
        j = rng.integers(N)
        if i == j:
            continue
        d = best[i] - best[j]
        if d > 0:
            delta = -py_max(0, py_min(d, 1 - best[j])) * rng.uniform(0.3, 1.0)
        else:
            delta = py_min(0, py_min(d, -best[i])) * rng.uniform(0.3, 1.0)
        trial = best.copy()
        trial[i] += delta
        trial[j] -= delta
        trial = np.clip(trial, 0.0, 1.0)
        trial = project_box_sum(trial, m)
        s = score_exact(trial, N)
        if s < best_s - 1e-14:
            best = trial
            best_s = s
    return best, best_s

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    raw = globals().get('initial_h_values', None)
    N = None
    m = None
    if raw is not None:
        raw = np.asarray(raw, dtype=np.float64)
        if raw.ndim == 1 and raw.size > 1 and np.all(np.isfinite(raw)):
            N = raw.size
            m = N / 2.0
    if N is None:
        N = kwargs.get('n_points', 70)
        if not isinstance(N, (int, np.integer)) or isinstance(N, bool) or N < 1 or N > 999:
            N = 70
        m = N / 2.0
    if N == 1:
        h = np.array([0.5])
        return h, score_exact(h, 1), 1
    if raw is not None and np.all(np.isfinite(raw)):
        if raw.min() >= -1e-12 and raw.max() <= 1 + 1e-12 and abs(raw.sum() - m) < 1e-8:
            baseline = np.clip(raw.copy(), 0.0, 1.0)
        else:
            baseline = project_box_sum(raw, m)
    else:
        baseline = np.full(N, 0.5)
    baseline = project_box_sum(baseline, m)
    base_score = score_exact(baseline, N)
    if 2 <= N <= 96:
        n_runs, max_gen, stag_limit = 3, 100, 24
        sigmas = [0.12, 0.07, 0.035]
    elif 97 <= N <= 160:
        n_runs, max_gen, stag_limit = 2, 60, 18
        sigmas = [0.10, 0.05]
    elif 161 <= N <= 256:
        n_runs, max_gen, stag_limit = 1, 30, 12
        sigmas = [0.08]
    else:
        n_runs, max_gen, stag_limit = 0, 0, 0
        sigmas = []
    B = build_helmert(N)
    best_h = baseline.copy()
    best_c5 = base_score
    for r in range(n_runs):
        try:
            h_res, s_res = cma_run(best_h, B, m, N, sigmas[r], max_gen, stag_limit, rng)
            if s_res < best_c5 - 1e-12:
                best_c5 = s_res
                best_h = h_res.copy()
        except Exception:
            pass
    try:
        best_h, best_c5 = refine_pair_swap(best_h, m, N, rng, 500)
    except Exception:
        pass
    if best_c5 > base_score:
        best_h = baseline
        best_c5 = base_score
    best_h = project_box_sum(best_h, m)
    final_score = score_exact(best_h, N)
    if not np.isfinite(final_score) or final_score > base_score + 1e-12:
        best_h = project_box_sum(baseline, m)
        final_score = score_exact(best_h, N)
    if not np.isfinite(final_score):
        best_h = np.full(N, 0.5)
        final_score = score_exact(best_h, N)
    return best_h, final_score, int(N)
