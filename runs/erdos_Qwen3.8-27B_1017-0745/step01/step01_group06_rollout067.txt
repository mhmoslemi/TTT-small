        if not feasible(ht, n, S):
            delta *= 0.5
            cr += 1
            if cr >= 8:
                break
            continue
        da = ht - h
        q_old = lag_vals(h, dx)
        G = build_G(h, n, dx)
        q_new = lag_vals(ht, dx)
        pred = q_old.max() - (q_old + G @ da).max()
        act = q_old.max() - q_new.max()
        if act > 1e-12 and (pred <= 1e-12 or act / max(pred, 1e-15) >= 0.1):
            h = ht
            cr = 0
            ratio = act / max(pred, 1e-15)
            if ratio > 0.75 and np.max(np.abs(da)) >= 0.8 * delta:
                delta, eta = min(0.5, 2 * delta), min(2.0, 1.5 * eta)
            elif ratio < 0.25:
                delta *= 0.5
        else:
            cr += 1
            delta, eta = delta * 0.5, max(1e-4, eta * 0.5)
            if cr >= 8:
                break
        if np.max(np.abs(da)) < 1e-8:
            break
    return h

# Direct SLSQP epigraph callbacks
def d_obj(z, _):
    return z[-1]

def d_obj_jac(z, _):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def d_ineq(z, n, dx, nl):
    h = z[:-1]
    return z[-1] - np.correlate(h, 1.0 - h, mode='full') * dx

def d_ineq_jac(z, n, dx, nl):
    h = z[:-1]
    j = np.arange(n)
    J = np.zeros((nl, n + 1))
    for k in range(nl):
        ell = k - (n - 1)
        i1, v1 = j - ell, (j - ell >= 0) & (j - ell < n)
        i2, v2 = j + ell, (j + ell >= 0) & (j + ell < n)
        t1, t2 = np.zeros(n), np.zeros(n)
        t1[v1] = 1.0 - h[i1[v1]]
        t2[v2] = h[i2[v2]]
        J[k, :-1] = -dx * (t1 - t2)
        J[k, -1] = 1.0
    return J

def d_eq(z, S):
    return z[:-1].sum() - S

def d_eq_jac(z, S):
    g = np.zeros(len(z))
    g[:-1] = 1.0
    return g

def direct_solve(h0, n, S, dx, maxiter):
    nl = 2 * n - 1
    h = h0.copy()
    t0 = exact_score(h, dx) + 1e-6
    z0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    con = [
        {'type': 'eq', 'fun': d_eq, 'jac': d_eq_jac, 'args': (S,)},
        {'type': 'ineq', 'fun': d_ineq, 'jac': d_ineq_jac, 'args': (n, dx, nl)}
    ]
    try:
        res = minimize(d_obj, z0, jac=d_obj_jac, args=(), method='SLSQP',
                       bounds=bounds, constraints=con,
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return h
    if res is not None and hasattr(res, 'x') and np.all(np.isfinite(res.x)):
        hs = project_K(res.x[:-1].copy(), n, S)
        if feasible(hs, n, S):
            return hs
    return h

def transfer(h0, n0, m):
    if n0 == m:
        return h0.copy()
    y = np.zeros(m)
    for j in range(m):
        s = 0.0
        for i in range(n0):
            ov = min((i + 1) / n0, (j + 1) / m) - max(i / n0, j / m)
            if ov > 0:
                s += h0[i] * ov
        y[j] = m * s
    S = m / 2.0
    return project_K(y, m, S)
