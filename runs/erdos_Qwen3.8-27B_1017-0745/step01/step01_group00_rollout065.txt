import numpy as np
from scipy.optimize import minimize

def project_capped(z, n):
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    s = n / 2.0
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for i in range(n + 2):
        if abs(r) < 1e-14:
            break
        idx = i % n
        if r > 0:
            add = min(r, 1.0 - h[idx])
            h[idx] += add
            r -= add
        else:
            sub = min(-r, h[idx])
            h[idx] -= sub
            r += sub
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(np.sum(h) - s) > 1e-9 * max(1, n):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def compute_G(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        p0 = max(d, 0)
        q0 = max(-d, 0)
        m = n - abs(d)
        row = d + n - 1
        G[row, p0:p0 + m] += dx * (1.0 - h[q0:q0 + m])
        G[row, q0:q0 + m] -= dx * h[p0:p0 + m]
    return G

def obj_fun(z):
    return z[-1]

def obj_jac(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def eq_fun(z):
    n = len(z) - 1
    return np.sum(z[:n]) - 0.5 * n

def eq_jac(z):
    n = len(z) - 1
    J = np.zeros(len(z))
    J[:n] = 1.0
    return J

def ineq_fun(z):
    n = len(z) - 1
    h = z[:n]
    q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return z[-1] - q

def ineq_jac(z):
    n = len(z) - 1
    h = z[:n]
    G = compute_G(h, n)
    L = 2 * n - 1
    J = np.zeros((L, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def slsqp_solve(h, n, maxiter=400):
    q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    t0 = float(np.max(q)) + 1e-6
    z0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    cons = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
    ]
    try:
        res = minimize(obj_fun, z0, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
    except Exception:
        return None
    if res.x is None or res.x.shape != (n + 1,):
        return None
    return project_capped(res.x[:n], n)

        if best_s >= score - 1e-14:
            break
        cur, score = best_z, best_s
    return cur, score
