def project_K(y, n, S):
    # Capped simplex projection
    if not np.all(np.isfinite(y)):
        return np.full(n, S / n)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    err = S - h.sum()
    if abs(err) > 1e-12:
        if err > 0:
            idx = np.where(1.0 - h > 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) > 0:
            h[idx] = np.clip(h[idx] + err / len(idx), 0.0, 1.0)
    if abs(h.sum() - S) > 1e-8:

def lag_grad_matrix(h, n, dx):
    G = np.zeros((2 * n - 1, n))
    j = np.arange(n)
    for ell in range(-(n - 1), n):
        k = ell + (n - 1)

```python
def tr_obj(z, eta):
    d, t = z[:-1], z[-1]
    return t + np.dot(d, d) / (2.0 * eta)

def tr_obj_jac(z, eta):
    d, t = z[:-1], z[-1]
    g = np.zeros_like(z)
    g[:-1] = d / eta
    g[-1] = 1.0
    return g

def tr_ineq(z, q, G):
    d, t = z[:-1], z[-1]
    return t - q - G.dot(d)

def tr_ineq_jac(z, q, G):
    n = len(z) - 1
    J = np.zeros((len(q), len(z)))
    J[:, :-1] = -G
    J[:, -1] = 1.0
    return J

def tr_eq(z, n):
    return z[:n].sum()

def tr_eq_jac(z, n):
    g = np.zeros(len(z))
    g[:n] = 1.0
    return g
```

def solve_tr_subproblem(h, n, dx, delta, eta):
    S = n / 2.0
    q = lag_vals(h, dx)
    G = lag_grad_matrix(h, n, dx)
    z0 = np.concatenate([np.zeros(n), [q.max()]])
    # Bounds for d: max(-delta, -h[j]) to min(delta, 1-h[j])
    d_lo = np.maximum(-delta, -h)
    d_hi = np.minimum(delta, 1.0 - h)
    bounds = [(d_lo[j], d_hi[j]) for j in range(n)] + [(None, None)]
    con = [
        {'type': 'eq', 'fun': tr_eq, 'jac': tr_eq_jac, 'args': (n,)},
        {'type': 'ineq', 'fun': tr_ineq, 'jac': tr_ineq_jac, 'args': (q, G)},
    ]
    res = minimize(tr_obj, z0, jac=tr_obj_jac, args=(eta,),
                   method='SLSQP', bounds=bounds, constraints=con,
                   options={'maxiter': 50, 'ftol': 1e-10, 'disp': False})
    return res, q, G
```

def tr_local(h, n, dx, delta0, eta0, max_iter, rng):
    S = n / 2.0
    h = project_K(h.copy(), n, S)
    delta, eta = delta0, eta0
    cur_score = exact_score(h, dx)
    consec_rej = 0
    consec_fail = 0
    for _ in range(max_iter):
        res, q, G = solve_tr_subproblem(h, n, dx, delta, eta)
        # Validate result
        if (res is None or not hasattr(res, 'x') or not np.all(np.isfinite(res.x))
                or len(res.x) != n + 1):
            consec_fail += 1
            if consec_fail >= 4:
                break
            delta *= 0.5
            eta = max(1e-4, eta * 0.5)
            if delta < 1e-5:
                break
            continue
        z = res.x
        d = z[:n]
        # Project h + d onto feasible set
        h_trial = project_K(h + d, n, S)
        if not feasible(h_trial, n, S):
            consec_fail += 1
            if consec_fail >= 4:
                break
            delta *= 0.5
            if delta < 1e-5:
                break
            continue
        d_actual = h_trial - h
        cur_max = q.max()
        pred = cur_max - (q + G.dot(d_actual)).max()
        trial_score = exact_score(h_trial, dx)
        actual = cur_max - trial_score
        consec_fail = 0
        if np.max(np.abs(d_actual)) < 1e-8:
            break
        if actual > 1e-12 and (pred <= 1e-12 or actual / max(pred, 1e-15) >= 0.1):
            h = h_trial
            cur_score = trial_score
            consec_rej = 0
            if pred > 1e-12:
                ratio = actual / pred
                if ratio < 0.25:
                    delta *= 0.5
                elif ratio > 0.75 and np.max(np.abs(d_actual)) >= 0.8 * delta:
                    delta = min(0.5, 2.0 * delta)
                    eta = min(2.0, 1.5 * eta)
        else:
            consec_rej += 1
            delta *= 0.5
            eta = max(1e-4, 0.5 * eta)
            if consec_rej >= 8:
                break
            if delta < 1e-5:
                break
    return project_K(h, n, S), exact_score(h, dx)

def transfer_resolution(h_parent, n0, m):
    # Cell overlap formula
    y = np.zeros(m)
    for j in range(m):
        # New cell j covers [j/m, (j+1)/m] in normalized coords
        # Sum overlap with parent cells
        s = 0.0
        for i in range(n0):
            lo = max(i / n0, j / m)
            hi = min((i + 1) / n0, (j + 1) / m)
            ov = max(0.0, hi - lo)
            s += h_parent[i] * ov
        y[j] = m * s
    return y
```

    best_h = project_K(best_h, best_n, best_n/2.0)
    if not feasible(best_h, best_n, best_n/2.0):
        best_h, best_s, best_n = fb.copy(), fb_s, n0
    else:
        fs = exact_score(best_h, 2.0/best_n)
        if fs > fb_s + 1e-10:
            best_h, best_s, best_n = fb.copy(), fb_s, n0
        else:
            best_s = fs
    best_h = np.clip(best_h, 0.0, 1.0)
    return (best_h, exact_score(best_h, 2.0/best_n), best_n)
