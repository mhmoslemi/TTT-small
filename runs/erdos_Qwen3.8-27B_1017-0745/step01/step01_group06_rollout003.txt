        for i in range(np_):
            iLo, iHi = i / np_, (i + 1) / np_
            ov = max(0.0, min(jHi, iHi) - max(jLo, iLo))
            tot += parent[i] * ov
        y[j] = m * tot
    return project_cap(y, m, S)

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    init = globals().get('initial_h_values', None)
    n0, pv, ph = 45, False, None
    if init is not None:
        try:
            init = np.array(init, dtype=np.float64).ravel()
            if 2 <= len(init) <= 999 and np.all(np.isfinite(init)) and init.min() >= -1e-8 and init.max() <= 1.0 + 1e-8:
                n0 = len(init)
                ph = project_cap(init, n0, n0 / 2.0)
                pv = True
        except Exception:
            pass
    nd = kwargs.get('n_points', None)
    if nd is not None:
        try:
            nd = int(nd)
            if 2 <= nd <= 999:
                n0 = nd
        except Exception:
            pass
    rl = [n0] if not pv else sorted(set([n0, max(2, n0 - 1), min(999, n0 + 1)]))
    fb_h, fb_s, fb_n = np.full(n0, 0.5), exact_score(np.full(n0, 0.5), 2.0 / n0), n0
    if pv:
        fb_h, fb_s, fb_n = ph.copy(), exact_score(ph, 2.0 / n0), n0
    bh, bs, bn = fb_h.copy(), fb_s, fb_n
    for m in rl:
        S, dx = m / 2.0, 2.0 / m
        nit = 28 if m <= 96 else (12 if m <= 256 else 5)
        mi = 50 if m <= 96 else (30 if m <= 256 else 15)
        ns = 8 if m <= 96 else (3 if m <= 256 else 1)
        if pv and m == fb_n:
            base = ph.copy()
        elif pv:
            base = transfer(ph, fb_n, m)
        else:
            base = np.full(m, 0.5)
        starts = [base, project_cap(1.0 - base, m, S), project_cap(base[::-1].copy(), m, S)]
        starts.append(project_cap(base + 0.08 * rng.standard_normal(m), m, S))
        starts.append(project_cap(base + 0.25 * rng.standard_normal(m), m, S))
        starts.append(project_cap(rng.uniform(0, 1, m), m, S))
        starts.append(project_cap(rng.beta(0.5, 0.5, m), m, S))
        hb = np.zeros(m); hb[:int(S)] = 1.0
        if S - int(S) > 0:
            hb[int(S)] = S - int(S)
        starts.append(project_cap(hb, m, S))
        jj = np.arange(m)
        starts.append(project_cap(0.5 + 0.45 * np.cos(2.0 * np.pi * jj / m), m, S))
        dedup = []
        for s in starts:
            dup = any(np.max(np.abs(s - d_)) <= 1e-6 for d_ in dedup)
            if not dup:
                dedup.append(s)
        dedup = dedup[:ns]
        scored = sorted([(exact_score(s, dx), i) for i, s in enumerate(dedup) if feasible(s, m, S)])
        for sc, i in scored:
            if sc < bs - 1e-12:
                bs, bh, bn = sc, dedup[i].copy(), m
        for sc, i in scored[:min(4, len(scored))]:
            hopt = local_opt(dedup[i], m, dx, rng, nit, mi)
            if feasible(hopt, m, S):
                hs = exact_score(hopt, dx)
                if hs < bs - 1e-12:
                    bs, bh, bn = hs, hopt.copy(), m
        if m <= 96:
            hr = project_cap((bh if bn == m else base) + 0.18 * rng.standard_normal(m), m, S)
            hopt = local_opt(hr, m, dx, rng, nit, mi)
            if feasible(hopt, m, S):
                hs = exact_score(hopt, dx)
                if hs < bs - 1e-12:
                    bs, bh, bn = hs, hopt.copy(), m
    bh = project_cap(bh, bn, bn / 2.0)
    fs = exact_score(bh, 2.0 / bn)
    if not feasible(bh, bn, bn / 2.0) or fs > fb_s + 1e-10:
        bh, fs, bn = fb_h.copy(), fb_s, fb_n
    bh = np.clip(bh, 0.0, 1.0)
    bh = project_cap(bh, bn, bn / 2.0)
    return (bh, exact_score(bh, 2.0 / bn), bn)
