                ch = [j for j in range(P) if j != i]
                idx = np.random.default_rng().choice(len(ch), 3, replace=False)  # NO, use rng
                # Actually just use rng.integers
                r1, r2, r3 = rng.choice(ch, size=3, replace=False)
                v = pop[r1] + F_i*(pop[r2]-pop[r3])
            
            jrand = int(rng.integers(0, n))
            mask = rng.random(n) < CR_i
            mask[jrand] = True
            trial = np.where(mask, v, pop[i])
            trial = project_box_sum(trial, S)
            if trial is None or not check_feasible(trial, n):
                continue
            if np.max(np.abs(trial - pop[i])) < 1e-14:
                continue
            ts = exact_score(trial)
            if ts < scores[i] - 1e-14:
                new_pop[i] = trial
                new_scores[i] = ts
                new_F[i] = F_i
                new_CR[i] = CR_i
                if ts < best_s - 1e-14:
                    best_s = ts
                    best_h = trial.copy()
        
        pop = new_pop
        scores = new_scores
        F_arr = new_F
        CR_arr = new_CR
        
        if best_s < prev_best - 1e-12:
            non_imp = 0
        else:
            non_imp += 1
        
        div = math.sqrt(np.mean(np.var(pop, axis=0))) if P > 1 else 0.0
        
        if (non_imp >= 12 or div < 1e-3) and restarts < 3:
            worst = np.argsort(scores)[::-1][:R]
            sigs = [0.03, 0.10, 0.25]
            for ci, wi in enumerate(worst):
                if ci % 4 == 3:
                    cand2 = rng.random(n)
                else:
                    cand2 = best_h + sigs[ci % 3] * rng.standard_normal(n)
                proj = project_box_sum(cand2, S)
                if proj is not None and check_feasible(proj, n):
                    pop[wi] = proj
                    scores[wi] = exact_score(proj)
                    F_arr[wi] = 0.5
                    CR_arr[wi] = 0.9
                    if scores[wi] < best_s:
                        best_s = scores[wi]
                        best_h = proj.copy()
