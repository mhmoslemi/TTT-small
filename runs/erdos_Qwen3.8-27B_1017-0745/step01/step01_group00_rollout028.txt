    # f takes (t, *args)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c, *args)
    fd = f(d, *args)
    for _ in range(iters):
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr * (b - a)

    fb = f(b, *args)

    fa = f(a, *args)
    fb = f(b, *args)
    if fa * fb > 0:
        return None
    for _ in range(iters):
        mid = 0.5 * (a + b)
        fm = f(mid, *args)
        if fa * fm <= 0:
            b = mid; fb = fm
        else:
            a = mid; fa = fm
    return 0.5 * (a + b)
