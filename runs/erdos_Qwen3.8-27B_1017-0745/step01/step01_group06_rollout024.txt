                    best_sc, best_h = scores[i], X[i].copy()
                    imp = True
            stall = 0 if imp else stall + 1
            if stall >= STL and rst < RST and gen < GENS - 1:
                rst += 1
                order = np.argsort(scores)
                Xn = np.array([X[order[i]].copy() for i in range(bc)])
                Sn = np.array([scores[order[i]] for i in range(bc)])
                for slot in range(bc, P):
                    alt = (slot - bc) % 4
                    sig_r = [0.08, 0.18, 0.35]
                    c = best_h + sig_r[alt] * rng.standard_normal(n) if alt < 3 else rng.beta(0.5, 0.5, n)
                    h = project_K(np.array(c, dtype=np.float64), n, S)
                    if is_feasible(h, n, S):
                        Xn = np.vstack([Xn, h.reshape(1, -1)])
                        Sn = np.append(Sn, exact_score(h, n))
                    else:
                        Xn = np.vstack([Xn, best_h.reshape(1, -1)])
                        Sn = np.append(Sn, best_sc)
                X, scores, mu_F, mu_CR, stall = Xn, Sn, 0.7, 0.8, 0
        best_h, best_sc = refine_best(best_h, n, S, rng, SIG, NP2)
        best_h = project_K(best_h, n, S)
        if not is_feasible(best_h, n, S) or exact_score(best_h, n) > fb_sc + 1e-10:
            best_h, best_sc = fallback.copy(), fb_sc
        else:
            best_sc = exact_score(best_h, n)
        best_h = np.clip(best_h, 0.0, 1.0)
        return (best_h, float(best_sc), int(n))
    except Exception:
        if base is not None and is_feasible(base, n, S):
            return (base.copy(), float(exact_score(base, n)), int(n))
        h = np.full(45, 0.5)
        return (h, float(exact_score(h, 45)), 45)
