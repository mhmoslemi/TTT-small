def project_simplex(v):

    n = len(v)
    idx = np.argsort(v)[::-1]
    sv = v[idx]
    cs = np.cumsum(sv)
    rho = np.max(np.where(sv - (cs - 1.0) / np.arange(1, n + 1) > 0))

    if warm is not None and warm.shape == (L,) and np.all(np.isfinite(warm)):
        w = project_simplex(warm)
        if w is None:
            w = np.zeros(L); w[np.argmax(Q)] = 1.0
    else:
        w = np.zeros(L); w[np.argmax(Q)] = 1.0
    
    v = w.copy()
    t = 1.0
    Ldual = max(1e-8, 0.5 * np.sum(G * G) * 1.01)
    
    Imax = max(100, min(320, int(np.ceil(14000.0 / n))))
    best_y = x.copy()
    best_model = float(np.max(Q) + 0.0)  # model value at x
    best_w = w.copy()
    best_phi = -float('inf')
    
    for it in range(Imax):
        # Evaluate at v
        z = G.T @ v
        y_v = project_box_sum(x - z / 2.0, S)
        if y_v is None:
            y_v = x.copy()
        d = y_v - x
        p = G @ d  # this is G*(y-x) for the gradient
        vals = Q + p
        model_v = float(np.dot(d, d) + np.max(vals))
        
        if model_v < best_model:
            best_model = model_v
            best_y = y_v.copy()
        
        # Gradient of dual at v
        grad_v = Q + G @ d  # same as vals
        
        # FISTA update
        w_new = project_simplex(v + grad_v / Ldual)
        if w_new is None:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        v_new = w_new + ((t - 1.0) / t_new) * (w_new - v)
        v = v_new
        w = w_new
        t = t_new
        
        # Gap check every few iterations
        if (it + 1) % max(5, Imax // 25) == 0:
            z_w = G.T @ w

            phi_w = float(np.dot(w, Q) + np.dot(d_w, d_w) + np.dot(z_w, d_w))

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0 or not np.all(np.isfinite(v)):
        return None
    lo, hi = float(np.min(v)) - 1.5, float(np.max(v)) + 0.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if resid > 1e-15:
        for i in np.argsort(1.0 - y):
            a = min(resid, 1.0 - y[i])
            y[i] += a
            resid -= a
            if resid <= 1e-15:
                break
    elif resid < -1e-15:
        for i in np.argsort(y):
            a = min(-resid, y[i])
            y[i] -= a
            resid += a
            if resid >= -1e-15:
                break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None

        return None

            j = i - k
            G[row, i] += 1.0 - x[j]
            G[row, j] += -x[i]
    return G

def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    idx = np.argsort(v)[::-1]
    sv = v[idx]
    cs = np.cumsum(sv)
    rho = np.max(np.where(sv - (cs - 1.0) / np.arange(1, n + 1) > 0))
    theta = (cs[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = np.sum(w)
    if s < 1e-15:
        w = np.zeros(n)
        w[np.argmax(v)] = 1.0
        return w
    w /= s
    return w
