Key points:

2. Replace B_d by its tangent at anchor point a, giving a convex majorizer M_d(y;a)
3. Solve convex subproblem with CVXPY conic solver
4. Use trust-region with backtracking line search
5. Exact score acceptance

For lag d, with valid pairs (i, i+d) where 0<=i<n and 0<=i+d<n:
q_d(y) = dx * sum_i y[i]*(1-y[i+d])

DC decomposition:
-uv = ((u-v)^2 - (u+v)^2)/4

So y[i]*(1-y[i+d]) = y[i] - y[i]*y[i+d]

So q_d(y) = dx * sum_i [y[i] + (y[i]-y[i+d])^2/4 - (y[i]+y[i+d])^2/4]

A_d(y) = dx * sum_i [y[i] + (y[i]-y[i+d])^2/4]  (convex)
B_d(y) = dx/4 * sum_i (y[i]+y[i+d])^2  (convex)

B_d'(y;a) = dx/2 * sum_i (a[i]+a[i+d])*(y[i]+y[i+d]) - dx/4 * sum_i (a[i]+a[i+d])^2

M_d(y;a) = A_d(y) - B_d'(y;a)

        if r <= 0: break
        a = min(r, 1.0 - x[i])

def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):
        return None
    if np.max(np.abs(z)) > 1e9:
        return None
    n = len(z)
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(80):
        mid = lo + 0.5 * (hi - lo)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - np.sum(x)
    for i in range(n):
        if r <= 0: break
        a = min(r, 1.0 - x[i]); x[i] += a; r -= a
    if r < 0:
        for i in range(n):
            if r >= 0: break
            a = min(-r, x[i]); x[i] -= a; r += a
    if abs(np.sum(x) - S) > 1e-10 or not np.all(np.isfinite(x)):
        return None
    return np.clip(x, 0.0, 1.0)

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-9)

        vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if vals.size != 2 * n - 1 or not np.all(np.isfinite(vals)):
            return None
        return float(np.max(vals))
    except Exception:
        return None

    S, dx = n / 2.0, 2.0 / n
    y = cp.Variable(n)
    t = cp.Variable()
    cons = [y >= 0, y <= 1, cp.sum(y) == S, y - anchor <= rho, anchor - y <= rho]
    for d in range(-(n - 1), n):
        lo, hi = max(0, -d), min(n, n - d)
        y0, y1 = y[lo:hi], y[lo + d:hi + d]
        s = anchor[lo:hi] + anchor[lo + d:hi + d]
        expr = dx * (cp.sum(y0) + 0.25 * cp.sum_squares(y0 - y1)
                     - 0.5 * cp.sum(s * (y0 + y1)) + 0.25 * cp.dot(s, s))
        cons.append(expr <= t)
    obj = cp.Minimize(t + 0.5 * eta * cp.sum_squares(y - anchor))
    prob = cp.Problem(obj, cons)
    for solver in ['ECOS', 'CVXOPT', 'SCS', 'MOSEK']:

            if prob.status in ('optimal', 'optimal_inaccurate'):
                yv = y.value
                if yv is not None:
                    yv = np.asarray(yv, dtype=np.float64)
                    if yv.shape == (n,) and np.all(np.isfinite(yv)):
                        return yv
        except Exception:
            continue
    return None
