import numpy as np
from scipy.signal import fftconvolve

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.ndim != 1:
        return None
    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(v - tau, 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-15:
        for i in np.argsort(1.0 - y)[::-1]:
            d = min(r, 1.0 - y[i])
            y[i] += d
            r -= d
            if r <= 1e-14:
                break
    elif r < -1e-15:
        for i in np.argsort(y)[::-1]:
            d = min(-r, y[i])
            y[i] -= d
            r += d
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-9) or np.any(y > 1.0 + 1e-9):
        return None
    if abs(float(np.sum(y)) - S) > 1e-10 * max(1, n):
        return None
    return y

def check_feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(float(np.sum(h)) - 0.5 * n) < 1e-10 * max(1, n))

def exact_score(h):
    n = len(h)

For h=0.5 (uniform): ∫ h(x)h(x+k) dx = 0.25 * (2-|k|) for |k|<2, so the min at k=0 is 0.5, giving C_5 = 1-0.5 = 0.5.

For the optimal solution, we want the autocorrelation to be as flat as possible (but it must be 1 at k=0 since ∫h² ≤ ∫h = 1, with equality only if h is 0 or 1 a.e.).

import numpy as np
from scipy.signal import fftconvolve

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(v - tau, 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-14:
        for i in np.argsort(1.0 - y)[::-1]:
            d = min(r, 1.0 - y[i])
            y[i] += d
            r -= d
            if r <= 1e-14:
                break
    elif r < -1e-14:
        for i in np.argsort(y)[::-1]:
            d = min(-r, y[i])
            y[i] -= d
            r += d
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-9) or np.any(y > 1.0 + 1e-9):
        return None
    if abs(float(np.sum(y)) - S) > 1e-9 * n:
        return None
    return y

def check_feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(float(np.sum(h)) - 0.5 * n) < 1e-9 * max(1, n))

def exact_score(h):
    n = len(h)
    if not check_feasible(h, n):
        return float('inf')
    val = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
    return val if np.isfinite(val) else float('inf')

def _get_parent():
    gv = globals().get('initial_h_values', None)
    if gv is not None:
        ph = np.asarray(gv, dtype=np.float64)
        if ph.ndim == 1 and 2 <= len(ph) <= 999 and np.all(np.isfinite(ph)):
            if np.all(ph >= -1e-6) and np.all(ph <= 1.0 + 1e-6):
                n = len(ph)
                S = 0.5 * n
                if abs(float(np.sum(ph)) - S) < 1e-4 * max(1, n):
                    p = project_box_sum(np.clip(ph, 0.0, 1.0), S)
                    if p is not None:
                        return p, n
    return project_box_sum(np.full(45, 0.5), 22.5), 45

def lse_value_grad(h, n, tau):
    b = 1.0 - h
    r = np.correlate(h, b, mode='full')
    q = r * (2.0 / n)
    qmax = float(np.max(q))
    shifted = (q - qmax) / tau
    exps = np.exp(shifted)
    s = float(np.sum(exps))
    val = qmax + tau * np.log(s)
    p = exps / s
    term1 = fftconvolve(p[::-1], b, mode='full')[n - 1:2 * n - 1]
    corr2 = np.correlate(p, h, mode='full')
    term2 = corr2[2 * n - 2:n - 2:-1]
    grad = (2.0 / n) * (term1.astype(np.float64) - term2.astype(np.float64))
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n)
    return val, grad

def proj_stage(h, n, tau, max_iter):
    S = 0.5 * n
    h = h.copy()
    F, g = lse_value_grad(h, n, tau)
    if not np.isfinite(F):
        return h, h.copy(), exact_score(h)
    alpha = 1.0
    best_h = h.copy()
    best_c = exact_score(h)
    stall = 0
    for it in range(max_iter):
        accepted = False
        for _bt in range(30):
            trial = project_box_sum(h - alpha * g, S)
            if trial is None:
                alpha *= 0.5
                continue
            delta = trial - h
            if np.max(np.abs(delta)) < 1e-12:
                alpha *= 0.5
                continue
            gd = float(np.dot(g, delta))
            if gd >= -1e-12:
                break
            Ft, _ = lse_value_grad(trial, n, tau)
            if not np.isfinite(Ft):
                alpha *= 0.5
                continue
            if Ft <= F + 1e-4 * gd + 1e-13:
                g_old = g.copy()
                h = trial
                F = Ft
                F, g = lse_value_grad(h, n, tau)
                if not np.isfinite(F):
                    break
                d = h - (h - delta)
                y = g - g_old
                dy = float(np.dot(d, y))
                if dy > 1e-14:
                    alpha = np.clip(float(np.dot(d, d)) / dy, 1e-4, 10.0)
                else:
                    alpha = min(10.0, 1.3 * alpha)
                accepted = True
                break
            alpha *= 0.5
        if not accepted:
            break
        c = exact_score(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if np.max(np.abs(g)) < 1e-7:
            stall += 1
        else:
            stall = 0
        if stall >= 6:
            break
    return h, best_h, best_c

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent, n = _get_parent()
    S = 0.5 * n
    best_h = parent.copy()
    best_c = exact_score(parent)

    starts = [parent.copy(), (1.0 - parent).copy(), parent[::-1].copy()]
    for sd in [0.06, 0.15, 0.30, 0.50]:
        s = project_box_sum(parent + sd * rng.standard_normal(n), S)
        if s is not None:
            starts.append(s)
    starts.append(project_box_sum(rng.random(n), S) or parent.copy())
    t = np.arange(n, dtype=np.float64)
    for (A1, m1, P1, A2, m2, P2) in [(0.35, 1, 0.4, 0.15, 3, 1.1),
                                      (0.25, 2, 0.9, 0.20, 5, 2.3),
                                      (0.40, 1, 1.5, 0.10, 7, 0.3)]:
        u = 0.5 + A1 * np.cos(2 * np.pi * m1 * t / n + P1) + A2 * np.sin(2 * np.pi * m2 * t / n + P2)
        s = project_box_sum(u, S)
        if s is not None:
            starts.append(s)
    starts = [s for s in starts if s is not None]

    deduped = [starts[0]]
    for s in starts[1:]:
        if not any(np.max(np.abs(s.astype(np.float64) - d)) < 1e-7 for d in deduped):
            deduped.append(s)
    starts = deduped

    taus = [0.05 * (0.28 ** j) for j in range(8)]
    M = max(40, min(150, 6000 // n))

    for start in starts:
        if not check_feasible(start, n):
            continue
        c0 = exact_score(start)
        if c0 < best_c:
            best_c = c0
            best_h = start.copy()
        h_cur = start.copy()
        for tau in taus:
            h_cur, sb_h, sb_c = proj_stage(h_cur, n, tau, M)
            if sb_c < best_c:
                best_c = sb_c
                best_h = sb_h.copy()

    for sigma in [0.03, 0.08, 0.15, 0.25, 0.40]:
        s = project_box_sum(best_h + sigma * rng.standard_normal(n), S)
        if s is None:
            continue
        h_cur = s.copy()
        for tau in taus:
            h_cur, sb_h, sb_c = proj_stage(h_cur, n, tau, max(20, M // 2))
            if sb_c < best_c:
                best_c = sb_c
                best_h = sb_h.copy()

    for _ in range(3):
        s = project_box_sum(best_h + 0.10 * rng.standard_normal(n), S)
        if s is None:
            continue
        h_cur = s.copy()
        for tau in taus[-4:]:
            h_cur, sb_h, sb_c = proj_stage(h_cur, n, tau, max(20, M // 2))
            if sb_c < best_c:
                best_c = sb_c
                best_h = sb_h.copy()

    final = project_box_sum(best_h, S)
    if final is not None and check_feasible(final, n):
        fc = exact_score(final)
        if fc < best_c:
            best_c = fc
            best_h = final.copy()
    pb = project_box_sum(parent, S)
    if pb is not None:
        pc = exact_score(pb)
        if pc < best_c:
            best_c = pc
            best_h = pb.copy()
    if not np.isfinite(best_c):
        best_h = parent.copy()
        best_c = exact_score(best_h)
    return (best_h.astype(np.float64), best_c, n)
