import numpy as np

def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):
        return None
    if np.max(np.abs(z)) > 1e12:
        return None
    n = len(z)
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)
    r = S - np.sum(y)
    if r > 0:
        for i in range(n):
            if r <= 0: break
            a = min(r, 1.0 - y[i]); y[i] += a; r -= a
    elif r < 0:
        for i in range(n):
            if r >= 0: break
            a = min(-r, y[i]); y[i] -= a; r += a
    if abs(np.sum(y) - S) > 1e-10 or not np.all(np.isfinite(y)):
        return None
    return np.clip(y, 0.0, 1.0)

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-14) and np.all(h <= 1.0 + 1e-14)
            and abs(np.sum(h) - n / 2.0) <= 1e-10)

def exact_lags(h, n):
    vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if vals.shape != (2 * n - 1,) or not np.all(np.isfinite(vals)):
        return None, None
    return vals, float(np.max(vals))

def pair_line_search(h, q, p, r, n, dx):
    if p == r: return None
    L, U = max(-h[p], h[r] - 1.0), min(1.0 - h[p], h[r])
    if U - L <= 1e-12: return None
    a = np.zeros(2 * n - 1)
    a[n - 1] = -2.0 * dx
    dd = r - p
    a[n - 1 + dd] = dx
    a[n - 1 - dd] = dx
    b = np.zeros(2 * n - 1)
    for d in range(-(n - 1), n):
        s = 0.0
        if 0 <= p + d < n: s += 1.0 - h[p + d]
        if 0 <= r + d < n: s -= 1.0 - h[r + d]
        if 0 <= p - d < n: s -= h[p - d]
        if 0 <= r - d < n: s += h[r - d]
        b[n - 1 + d] = dx * s
    start_sc = float(np.max(q))
    best_t, best_v = 0.0, start_sc
    ts = np.linspace(L, U, 17)
    ts[np.argmin(np.abs(ts))] = 0.0
    for t in ts:
        v = float(np.max(q + b * t + a * t * t))
        if v < best_v - 1e-14: best_v, best_t = v, float(t)
    rad = (U - L) / 16.0
    for _ in range(4):
        lo_r, hi_r = max(L, best_t - rad), min(U, best_t + rad)
        for t in np.linspace(lo_r, hi_r, 5):
            v = float(np.max(q + b * t + a * t * t))
            if v < best_v - 1e-14: best_v, best_t = v, float(t)
        rad *= 0.25
    if best_v >= start_sc - 1e-12: return None
    return best_t

def polish(h, q, score, n, dx, rng, P):
    adj = min(n - 1, max(1, P // 2))
    pairs = [(int(i), int(i) + 1) for i in np.linspace(0, n - 2, adj, dtype=int)]
    guard = 0
    while len(pairs) < P and guard < P * 10:
        p, r = int(rng.integers(0, n)), int(rng.integers(0, n))
        guard += 1
        if p == r: continue
        if p > r: p, r = r, p
        if not any(pairs[i] == (p, r) for i in range(len(pairs))):
            pairs.append((p, r))
    pairs = pairs[:P]
    perm = rng.permutation(len(pairs))
    improved = False
    for i in perm:
        p, r = pairs[i]
        res = pair_line_search(h, q, p, r, n, dx)
        if res is None: continue
        ht = h.copy()
        ht[p] += res; ht[r] -= res
        if not is_feasible(ht, n): continue
        q2, s2 = exact_lags(ht, n)
        if q2 is not None and s2 < score - 1e-12:
            h[:] = ht; q, score = q2, s2
            improved = True
    return h, q, score, improved

def build_starts(n, S, parent, rng):
    starts = []
    if parent is not None: starts.append(parent.copy())
    idx = np.arange(n, dtype=np.float64)
    M = min(4, max(1, (n - 1) // 2))
    u = np.full(n, 0.5)
    for m in range(1, M + 1):
        A = float(rng.normal(0.0, 0.12 / np.sqrt(m)))
        phi = float(rng.uniform(0.0, 2.0 * np.pi))
        u += A * np.cos(2.0 * np.pi * m * (idx + 0.5) / n + phi)
    pu = project_capped_simplex(u, S)
    if pu is not None: starts.append(pu)
    perm = rng.permutation(n)
    bv = np.zeros(n); bv[perm[:n // 2]] = 1.0
    if n % 2 == 1: bv[perm[n // 2]] = 0.5
    starts.append(bv)
    ur = project_capped_simplex(rng.uniform(0.0, 1.0, n), S)
    if ur is not None: starts.append(ur)
    if parent is not None:
        starts.append(parent[::-1].copy())
        cp = 1.0 - parent
        if is_feasible(cp, n): starts.append(cp)
    deduped = []
    for s in starts:
        s = np.asarray(s, dtype=np.float64)
        if not is_feasible(s, n):
            s = project_capped_simplex(s, S)
            if s is None: continue
        dup = False
        for d in deduped:
            if np.max(np.abs(s - d)) < 1e-10: dup = True; break
        if not dup: deduped.append(s)
    return deduped[:6]
