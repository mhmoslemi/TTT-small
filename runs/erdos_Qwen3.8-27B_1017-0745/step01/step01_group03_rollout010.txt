def project_box_sum(v, S):

    n = len(v)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):

        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-14:
        if resid > 0:
            order = np.argsort(1.0 - y)
            for i in order:
                add = min(resid, 1.0 - y[i])
                y[i] += add
                resid -= add
                if abs(resid) < 1e-15:
                    break
        else:
            order = np.argsort(y)
            for i in order:
                sub = min(-resid, y[i])
                y[i] -= sub
                resid += sub

                    if scores[int(wi)] < best_sc - 1e-14:
                        best_sc = scores[int(wi)]
                        best_h = c.copy()
        if restarts >= 4 and no_improve >= 20 and D < 1e-5:
            break

    best_h, best_sc = local_refine(rng, best_h, n, S, ntrials=300)

    final_h = project_box_sum(best_h, S)
    if final_h is None:
        final_h = fallback
    final_sc = exact_score(final_h, n)
    if not np.isfinite(final_sc) or final_sc > fb_score:
        final_h = fallback
        final_sc = fb_score
    final_h = np.clip(final_h, 0.0, 1.0)
    return (np.array(final_h, dtype=np.float64), float(final_sc), int(n))
