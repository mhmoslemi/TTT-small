```python
def lag_model(x, n):
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.zeros(L)
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        p0 = max(d, 0)
        q0 = max(-d, 0)
        m = n - abs(d)
        p = slice(p0, p0 + m)
        qq = slice(q0, q0 + m)
        q[d + n - 1] = dx * np.sum(x[p] * (1.0 - x[qq]))
        G[d + n - 1, p] += dx * (1.0 - x[qq])
        G[d + n - 1, qq] -= dx * x[p]
    return q, G
```

def solve_mm_slsqp(x, n, beta, maxiter):
    _ctx['x'] = x
    _ctx['ny'] = n
    _ctx['dx'] = 2.0 / n
    _ctx['beta'] = beta
    _ctx['s'] = n / 2.0
    q, G = lag_model(x, n)
    _ctx['q'] = q
    _ctx['G'] = G
    z0 = np.empty(n + 1)
    z0[:n] = x
    z0[n] = np.max(q) + 1e-8
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    constr = [
        {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac},
        {'type': 'eq', 'fun': _eq, 'jac': _eq_jac},
    ]
    try:
        res = minimize(_obj, z0, jac=_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=constr,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
    except Exception:
        return None
    y = res.x[:n]
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return None
    return y

def solve_mm_subgrad(x, n, beta, n_inner):
    dx = 2.0 / n
    y = x.copy()
    best_model = float('inf')
    best_y = x.copy()
    for k in range(n_inner):
        u = y - x
        # Compute model values for all lags, find active one
        a_max = -np.inf
        g_active = np.zeros(n)
        for d in range(-(n - 1), n):
            p0 = max(d, 0)
            q0 = max(-d, 0)
            m = n - abs(d)
            p = p0 + np.arange(m)
            qq = q0 + np.arange(m)
            a_d = dx * (np.dot(x[p], 1.0 - x[qq]) + np.dot(1.0 - x[qq], u[p]) - np.dot(x[p], u[qq]))
            if a_d > a_max:
                a_max = a_d
                ga = np.zeros(n)
                ga[p] += dx * (1.0 - x[qq])
                ga[qq] -= dx * x[p]
                g_active = ga
        model_val = a_max + beta * dx * np.dot(u, u)
        if model_val < best_model:
            best_model = model_val
            best_y = y.copy()
        v = g_active + 2.0 * beta * dx * u
        nv = np.linalg.norm(v)
        if nv < 1e-15:
            break
        eta = min(0.5 / (nv + 1e-12), 1.0 / (2.0 * beta * dx * np.sqrt(k + 1.0)))
        y_new = project_capped(y - eta * v, n)
        if y_new is None:
            break
        if np.max(np.abs(y_new - y)) < 1e-10:
            y = y_new
            break
        y = y_new
    return best_y

from scipy.optimize import minimize

_ctx = {}

def project_capped(z, n):
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    h = np.clip(h, 0.0, 1.0)

        return None
    return h

    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c * (2.0 / n)))

def lag_model(x, n):
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.zeros(L)
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        p0, q0 = max(d, 0), max(-d, 0)
        m = n - abs(d)
        px, qx = x[p0:p0+m], x[q0:q0+m]
        q[d + n - 1] = dx * np.dot(px, 1.0 - qx)
        G[d + n - 1, p0:p0+m] += dx * (1.0 - qx)
        G[d + n - 1, q0:q0+m] -= dx * px
    return q, G

def _obj(z):
    ny = _ctx['ny']; y = z[:ny]; t = z[ny]; u = y - _ctx['x']
    return t + _ctx['beta'] * _ctx['dx'] * np.dot(u, u)

def _obj_jac(z):
    ny = _ctx['ny']; g = np.empty(z.shape)
    g[:ny] = 2.0 * _ctx['beta'] * _ctx['dx'] * (z[:ny] - _ctx['x'])
    g[ny] = 1.0
    return g

    ny = _ctx['ny']; y = z[:ny]; t = z[ny]
    return t - _ctx['q'] - _ctx['G'] @ (y - _ctx['x'])

    ny = _ctx['ny']; L = 2 * ny - 1
    J = np.empty((L, ny + 1))
    J[:, :ny] = -_ctx['G']
    J[:, ny] = 1.0
    return J

    _ctx['beta'] = beta; _ctx['s'] = n / 2.0

            m = n - abs(d)

            if a_d > a_max:
