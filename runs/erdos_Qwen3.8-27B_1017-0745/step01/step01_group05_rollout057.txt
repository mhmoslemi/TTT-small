def project_simplex(z):

    if m == 0:
        return np.array([], dtype=np.float64)
    if not np.all(np.isfinite(z)):
        return np.full(m, 1.0 / m, dtype=np.float64)
    sv = np.sort(z)[::-1]
    cumsum = np.cumsum(sv)
    rho = np.where(sv - (cumsum - 1.0) / np.arange(1, m + 1) > 0)[0]
    if len(rho) == 0:
        return np.full(m, 1.0 / m, dtype=np.float64)
    p = int(rho[-1]) + 1
    theta = (cumsum[p - 1] - 1.0) / p
    return np.maximum(z - theta, 0.0)

def project_capped(y, n):
    s = n / 2.0
    if y is None or getattr(y, 'shape', None) != (n,) or not np.all(np.isfinite(y)):
        return np.full(n, 0.5), False
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    lam = float(np.clip(y.mean() - 0.5, lo, hi))
    for _ in range(60):
        c = np.clip(y - lam, 0.0, 1.0)
        r = c.sum() - s
        if abs(r) <= 1e-13:
            break
        lo = lam if r > 0 else lo
        hi = lam if r < 0 else hi
        a = int(np.sum((y > lam) & (y < lam + 1.0)))
        lam_new = lam + r / a if a > 0 else 0.5 * (lo + hi)
        lam = lam_new if np.isfinite(lam_new) and lo < lam_new < hi else 0.5 * (lo + hi)

def check_feasible(h, n):

def lag_scores(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def exact_score(h, n):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(lag_scores(h, n)))

def build_G(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(-(n - 1), n):
        r = k + n - 1
        lo1, hi1 = max(0, k), min(n, n + k)
        if hi1 > lo1:
            G[r, lo1:hi1] += dx * (1.0 - h[lo1 - k:hi1 - k])
        lo2, hi2 = max(0, -k), min(n, n - k)
        if hi2 > lo2:
            G[r, lo2:hi2] -= dx * h[lo2 + k:hi2 + k]
    return G

def project_simplex(z):
    m = len(z)
    if m <= 0 or not np.all(np.isfinite(z)):
        return np.full(m, 1.0 / m) if m > 0 else np.array([], dtype=np.float64)
    sv = np.sort(z)[::-1]
    cs = np.cumsum(sv)
    idx = np.arange(1, m + 1, dtype=np.float64)
    rho = np.where(sv - (cs - 1.0) / idx > 0)[0]
    if len(rho) == 0:
        return np.full(m, 1.0 / m)
    p = int(rho[-1]) + 1
    theta = (cs[p - 1] - 1.0) / p
    return np.maximum(z - theta, 0.0)

def majorant_state(h, q, G, p_vec, L, n):
    gp = (G.T @ p_vec) / L
    zz, _ = project_capped(h - gp, n)
    d = zz - h
    u = q + G @ d
    D_val = float(p_vec @ u) + 0.5 * L * float(d @ d)
    return d, u, D_val
