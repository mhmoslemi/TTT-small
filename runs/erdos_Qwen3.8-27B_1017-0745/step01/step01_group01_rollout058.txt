        sd = 42
    if sd == 0:
        sd = 42
    rng = np.random.default_rng(sd)
    if n <= 64:
        ns_def, mi_def, ps_def = 8, 250, 1
    elif n <= 128:
        ns_def, mi_def, ps_def = 5, 160, 1
    elif n <= 256:
        ns_def, mi_def, ps_def = 3, 80, 0
    else:
        ns_def, mi_def, ps_def = 2, 0, 0
    try:
        n_starts = max(1, min(int(kwargs.get('starts', ns_def)), 10))
    except Exception:
        n_starts = ns_def
    try:
        max_iter = max(0, min(int(kwargs.get('max_iter', mi_def)), 300))
    except Exception:
        max_iter = mi_def
    try:
        n_sweeps = max(0, min(int(kwargs.get('polish_sweeps', ps_def)), 2))
    except Exception:
        n_sweeps = ps_def
    seen = []
    add_start(seen, fallback, n)
    if parent is not None:
        add_start(seen, fallback[::-1], n)
        add_start(seen, 1.0 - fallback, n)
        add_start(seen, 1.0 - fallback[::-1], n)
        add_start(seen, fallback + 0.025 * rng.standard_normal(n), n)
        add_start(seen, fallback + 0.10 * rng.standard_normal(n), n)
    add_start(seen, uniform, n)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    add_start(seen, blk, n)
    for q in (1, 2, 3):
        add_start(seen, 0.5 + 0.45 * np.cos(2.0 * np.pi * q * np.arange(n) / n), n)
    add_start(seen, 0.5 + 0.3 * np.cos(2.0 * np.pi * np.arange(n) / n) + 0.2 * np.cos(4.0 * np.pi * np.arange(n) / n), n)
    add_start(seen, rng.uniform(0, 1, n), n)
    saw = np.linspace(0.05, 0.95, n)
    add_start(seen, saw, n)
    starts = seen[:n_starts]
    best = fallback.copy()
    best_score = fb_score
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_score - 1e-12:
            best, best_score = st.copy(), ss
    if n <= 256 and max_iter > 0:
        for st in starts:
            x0 = st.copy()
            s0 = exact_score(x0, n)
            if not np.isfinite(s0):
                continue
            t0 = min(1.0, s0 + 1e-8)
            z0 = np.concatenate([x0, [t0]])
            bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
            constrs = [
                {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac, 'args': (n,)},
                {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac, 'args': (n,)}
            ]
            try:
                res = minimize(obj_fn, z0, args=(n,), method='SLSQP',
                               jac=obj_jac, bounds=bounds, constraints=constrs,
                               options={'maxiter': max_iter, 'ftol': 1e-10})
                if res.x is not None and len(res.x) == n + 1:
                    yh = project_capped(res.x[:n], n)
                    if yh is not None:
                        ys = exact_score(yh, n)
                        if np.isfinite(ys) and ys < best_score - 1e-12:
                            best, best_score = yh.copy(), ys
            except Exception:
                pass
    if n <= 64 and n_sweeps > 0:
        best, best_score = pairwise_polish(best, n, n_sweeps)
    final = project_capped(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final = project_capped(fallback, n)
        if final is None:
            final = uniform.copy()
        fs = exact_score(final, n)
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final, float(fs), int(n)
