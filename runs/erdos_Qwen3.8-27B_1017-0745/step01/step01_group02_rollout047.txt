def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    if S < -1e-12 or S > n + 1e-12:
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-13:
        for i in np.argsort(1.0 - y):
            d = min(r, 1.0 - y[i])
            y[i] += d
            r -= d
            if r <= 0:
                break
    elif r < -1e-13:
        for i in np.argsort(y):
            d = min(-r, y[i])
            y[i] -= d
            r += d
            if r >= 0:
                break

        return None

        return None
    return y

def check_feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and

def exact_score(h):
    n = len(h)
    if not check_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    if len(c) != 2 * n - 1 or not np.all(np.isfinite(c)):
        return float('inf')
    v = float(np.max(c) * (2.0 / n))
    return v if np.isfinite(v) else float('inf')

def lag_vg(h, s, n):
    dx = 2.0 / n
    lo = max(0, -s)
    hi = min(n, n - s)
    q = 0.0
    g = np.zeros(n)
    for i in range(lo, hi):
        j = i + s
        q += h[i] * (1.0 - h[j])
        g[i] += (1.0 - h[j])
        g[j] -= h[i]

def mm_step(x, n):
    dx = 2.0 / n
    L = 2.0 * dx
    nv = 2 * n - 1
    qJ = np.zeros(nv)
    GJ = np.zeros((nv, n))
    for idx, s in enumerate(range(-(n - 1), n)):
        q, g = lag_vg(x, s, n)
        qJ[idx] = q
        GJ[idx] = g
    _mm['x'] = x
    _mm['L'] = L
    _mm['qJ'] = qJ
    _mm['GJ'] = GJ
    _mm['S'] = 0.5 * n
    _mm['n'] = n
    
    t0 = max(float(np.max(qJ)), 0.0) + 1e-4
    z0 = np.append(x.copy(), t0)
    bounds = [(0.0, 1.0)] * n + [(-np.inf, np.inf)]
    cons = [
        {'type': 'ineq', 'fun': mm_ineq, 'jac': mm_ineq_grad},
        {'type': 'eq', 'fun': mm_eq, 'jac': mm_eq_grad}
    ]
    try:
        res = minimize(mm_obj, z0, jac=mm_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-12, 'disp': False})
        y = np.array(res.x[:n], dtype=np.float64)
    except Exception:
        return x.copy()
    yp = project_box_sum(y, 0.5 * n)
    if yp is None or not check_feasible(yp, n):
        return x.copy()
    return yp
