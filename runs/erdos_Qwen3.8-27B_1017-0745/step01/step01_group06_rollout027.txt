import numpy as np

def project_K(y, n, S):
    """Project y onto {h: 0<=h<=1, sum(h)=S} via bisection on threshold."""
    if not np.all(np.isfinite(y)) or y.shape != (n,):
        return np.full(n, S / n)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5*(lo+hi), 0.0, 1.0)
    e = S - h.sum()
    if abs(e) > 1e-12:
        if e > 0:
            idx = np.where(1.0 - h > 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) > 0:
            h[idx] += e / len(idx)
            h = np.clip(h, 0.0, 1.0)
    if not feasible_check(h, n, S):
        return np.full(n, S / n)
    return h

def feasible_check(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lag_vec(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def line_search(h, q, cur_score, z, b, a, lo, hi, dx, n, coarse_pts=25, n_branches=2, ref_rounds=4, ref_pts=9):
    """Find best alpha in [lo,hi] minimizing max(q + alpha*b + alpha^2*a)."""
    if hi - lo < 1e-14:
        return None
    # Coarse grid
    alphas = np.linspace(lo, hi, coarse_pts)
    # Evaluate F_line on grid
    # F_line[j] = max_r (q[r] + alphas[j]*b[r] + alphas[j]^2*a[r])
    # Shape: (coarse_pts, 2*n-1)
    F = q[None, :] + alphas[:, None] * b[None, :] + (alphas ** 2)[:, None] * a[None, :]
    Fm = F.max(axis=1)  # shape (coarse_pts,)
    
    # Find coarse local minima
    minima_idx = []
    for j in range(coarse_pts):
        left_ok = (j == 0) or (Fm[j] <= Fm[j-1])
        right_ok = (j == coarse_pts - 1) or (Fm[j] <= Fm[j+1])
        if left_ok and right_ok and np.isfinite(Fm[j]):
            minima_idx.append(j)
    
    if not minima_idx:
        minima_idx = [int(np.argmin(Fm))]
    
    # Sort by Fm value, break ties by |alpha - (lo+hi)/2|
    mid_a = 0.5 * (lo + hi)
    minima_idx.sort(key=lambda j: (Fm[j], abs(alphas[j] - mid_a)))
    
    # Select n_branches distinct minima (separated by >= 2)
    sel = []
    for j in minima_idx:
        if len(sel) >= n_branches:
            break
        if all(abs(j - s) >= 2 for s in sel):
            sel.append(j)
    while len(sel) < n_branches and len(minima_idx) > 0:
        j = minima_idx[len(sel) % len(minima_idx)]
        sel.append(j)
    
    best_alpha, best_fm = 0.0, cur_score
    for j in sel:
        # Refine around alphas[j]
        a_lo = alphas[max(0, j-1)]
        a_hi = alphas[min(coarse_pts-1, j+1)]
        ra = alphas[j]
        rf = Fm[j]
        for _ in range(ref_rounds):
            if a_hi - a_lo < 1e-15:
                break
            pts = np.linspace(a_lo, a_hi, ref_pts)
            Fp = q[None,:] + pts[:,None]*b[None,:] + (pts**2)[:,None]*a[None,:]
            Fmp = Fp.max(axis=1)
            k = int(np.argmin(Fmp[np.isfinite(Fmp)])) if np.any(np.isfinite(Fmp)) else 0
            # Fix: find argmin among finite
            fin_mask = np.isfinite(Fmp)
            if not np.any(fin_mask):
                break
            k = int(np.where(fin_mask)[0][np.argmin(Fmp[fin_mask])])
            if Fmp[k] < rf:
                rf = Fmp[k]
                ra = pts[k]
            a_lo = pts[max(0, k-1)]
            a_hi = pts[min(ref_pts-1, k+1)]
        
        if rf < best_fm:
            best_fm = rf
            best_alpha = ra
    
    if best_alpha == 0.0 and best_fm >= cur_score - 1e-12:
        return None
    # Check if it improves
    if best_fm >= cur_score - 1e-12:
        return None
    
    trial = h + best_alpha * z
    # Repair
    S = n / 2.0
    trial = project_K(trial, n, S)
    if not feasible_check(trial, n, S):
        return None
    sc = exact_score(trial, dx)
    if sc >= cur_score - 1e-12:
        return None
    q_new = lag_vec(trial, dx)
    return (trial, q_new, sc)

def local_descent(h0, n, dx, rng, max_sweeps=3, n_pairs=None, n_sparse=8, coarse_pts=25, n_branches=2, ref_rounds=4, ref_pts=9):
    S = n / 2.0
    h = project_K(h0.copy(), n, S)
    if not feasible_check(h, n, S):
        return None
    q = lag_vec(h, dx)
    sc = exact_score(h, dx)
    
    # Generate pair list
    if n_pairs is None:
        n_pairs = n * (n - 1) // 2
    pairs = []
    if n_pairs <= n * (n-1) // 2:
        all_pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
        rng.shuffle(all_pairs)
        pairs = all_pairs[:n_pairs]
    else:
        pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
    # Shuffle
    pairs = list(pairs)
    rng.shuffle(pairs)
    
    for sweep in range(max_sweeps):
        changed = False
        # Re-shuffle pairs each sweep
        rng.shuffle(pairs)
        
        for (p, r) in pairs:
            if p >= r:
                continue
            # Direction z = e_p - e_r
            # Feasible interval
            av_lo = max(-h[p], h[r] - 1.0)
            av_hi = min(1.0 - h[p], h[r])
            if av_hi - av_lo < 1e-14:
                continue
            # Compute b, a for this direction
            # b = dx * (corr(z, 1-h) - corr(h, z))
            # a = -dx * corr(z, z)
            # For efficiency with sparse z:
            z = np.zeros(n)
            z[p] = 1.0
            z[r] = -1.0
            b = dx * (np.correlate(z, 1.0 - h, mode='full') - np.correlate(h, z, mode='full'))
            a = -dx * np.correlate(z, z, mode='full')
            
            res = line_search(h, q, sc, z, b, a, av_lo, av_hi, dx, n,
                             coarse_pts, n_branches, ref_rounds, ref_pts)
            if res is not None:
                h, q, sc = res
                changed = True
        
        # Sparse zero-sum directions
        for _ in range(n_sparse):
            support_size = min(2 + rng.integers(0, min(10, n-1)), n)
            support = rng.choice(n, size=support_size, replace=False)
            vals = rng.standard_normal(support_size)
            vals -= vals.mean()
            if np.max(np.abs(vals)) < 1e-14:
                continue
            z = np.zeros(n)
            z[support] = vals / np.max(np.abs(vals))
            # Make sum exactly zero
            z[support[-1]] -= z[support].sum() - 0.0
            # Actually just subtract mean already done, but fix residual
            res_sum = z.sum()
            if abs(res_sum) > 1e-15:
                z[support[0]] -= res_sum
            
            # Feasible interval
            pos = z > 1e-14
            neg = z < -1e-14
            lo_a = 0.0
            hi_a = 0.0
            if np.any(pos):
                lo_a = max(lo_a, (-h[pos] / z[pos]).min())
                hi_a = min(hi_a, ((1.0 - h[pos]) / z[pos]).min())
            if np.any(neg):
                lo_a = max(lo_a, ((1.0 - h[neg]) / z[neg]).min())
                hi_a = min(hi_a, (-h[neg] / z[neg]).min())
            
            if hi_a - lo_a < 1e-14:
                continue
            
            b = dx * (np.correlate(z, 1.0 - h, mode='full') - np.correlate(h, z, mode='full'))
            a = -dx * np.correlate(z, z, mode='full')
            if not (np.all(np.isfinite(b)) and np.all(np.isfinite(a))):
                continue
            
            res = line_search(h, q, sc, z, b, a, lo_a, hi_a, dx, n,
                             coarse_pts, n_branches, ref_rounds, ref_pts)
            if res is not None:
                h, q, sc = res
                changed = True
        
        if not changed:
            break

def make_starts(base, n, dx, S, rng):
    """Generate algorithmic starting points."""
    starts = [base.copy()]
    
    # Complement, reversal, reversed complement
    starts.append(base[::-1].copy())
    starts.append((1.0 - base).copy())
    starts.append((1.0 - base[::-1]).copy())
    
    # Noisy
    starts.append(base + 0.05 * rng.standard_normal(n))
    starts.append(base + 0.15 * rng.standard_normal(n))
    
    # Uniform random
    starts.append(rng.uniform(0, 1, n))
    
    # Beta
    starts.append(np.minimum(1.0, np.maximum(0.0, rng.beta(0.5, 0.5, n))))
    
    # Block
    blk = np.zeros(n)
    n_ones = int(S)
    blk[:n_ones] = 1.0
    frac = S - n_ones
    if frac > 1e-10:
        blk[n_ones] = frac
    perm = rng.permutation(n)
    starts.append(blk[perm])
    
    # Sinusoid
    jj = np.arange(n)
    starts.append(0.5 + 0.45 * np.cos(2.0 * np.pi * jj / n))
    
    # Project all and deduplicate
    projected = []
    seen = []
    for s in starts:
        p = project_K(s, n, S)
        if not feasible_check(p, n, S):
            continue
        dup = False
        for sp in seen:
            if np.max(np.abs(p - sp)) <= 1e-7:
                dup = True
                break
        if not dup:
            projected.append(p)
            seen.append(p.copy())
    
    return projected

import numpy as np

def project_K(y, n, S):
    if not np.all(np.isfinite(y)) or y.shape != (n,):
        return np.full(n, S / n)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    e = S - h.sum()
    if abs(e) > 1e-12:
        if e > 0:
            idx = np.where(1.0 - h > 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) > 0:
            h[idx] = np.clip(h[idx] + e / len(idx), 0.0, 1.0)
    return _feasible(h, n, S) ? h : np.full(n, S / n) if False else (h if _feasible(h, n, S) else np.full(n, S / n))

def _feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _line_score(q, b, a, alpha):
    return np.max(q + alpha * b + alpha * alpha * a)

def line_search(h, q, sc, z, b, a, lo, hi, n, dx, cp=25, nb=2, rr=4, rp=9):
    if hi - lo < 1e-14:
        return None
    alphas = np.linspace(lo, hi, cp)
    F = (q[None, :] + alphas[:, None] * b[None, :] +
         (alphas ** 2)[:, None] * a[None, :]).max(axis=1)
    if not np.all(np.isfinite(F)):
        return None
    mids = []
    for j in range(cp):
        lk = (j == 0) or (F[j] <= F[j - 1])
        rk = (j == cp - 1) or (F[j] <= F[j + 1])
        if lk and rk:
            mids.append(j)
    if not mids:
        mids = [int(np.argmin(F))]
    mids.sort(key=lambda j: (F[j], abs(alphas[j])))
    sel = []
    for j in mids:
        if len(sel) >= nb:

        if all(abs(j - s) >= 2 for s in sel):
            sel.append(j)

    best_a, best_f = 0.0, sc

        for _ in range(rr):
            if a_hi - a_lo < 1e-15:
                break
            pts = np.linspace(a_lo, a_hi, rp)
            Fp = (q[None, :] + pts[:, None] * b[None, :] +
                  (pts ** 2)[:, None] * a[None, :]).max(axis=1)

def make_starts(base, n, S, rng):
