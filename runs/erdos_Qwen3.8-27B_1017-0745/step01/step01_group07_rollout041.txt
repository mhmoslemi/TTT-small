    if parent is not None:
        ps = exact_score(parent)
        if ps is not None and np.isfinite(ps) and ps < best_score:
            best_score = ps
            best_h = parent.copy()
    left, right, lag_id = build_pairs(n)
    if n <= 128:
        max_it = 120
    elif n <= 256:
        max_it = 80
    elif n <= 512:
        max_it = 40
    else:
        max_it = 20
    tau_sched = [0.02, 0.005, 0.001, 0.0002, 0.00005]
    starts = build_starts(n, S, parent, rng)
    for s in starts:
        sc = exact_score(s)
        if sc is not None and np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_h = s.copy()
    for s in starts:
        bx, bsc = projected_descent(s, tau_sched, left, right, lag_id, n, S, max_it)
        if bsc is not None and np.isfinite(bsc) and bsc < best_score - 1e-12:
            best_score = bsc
            best_h = bx.copy()
    n_kicks = 3 if n <= 128 else (2 if n <= 256 else (1 if n <= 512 else 0))
    sigma = 0.04
    for _ in range(n_kicks):
        kick = project_capped_simplex(best_h + sigma * rng.normal(0.0, 1.0, n), S)
        if kick is None:
            continue
        bx, bsc = projected_descent(kick, tau_sched, left, right, lag_id, n, S, max_it)
        if bsc is not None and np.isfinite(bsc) and bsc < best_score - 1e-12:
            best_score = bsc
            best_h = bx.copy()
            sigma = max(0.02, 0.5 * sigma)
        else:
            sigma = min(0.20, 1.8 * sigma)
    fh = project_capped_simplex(best_h, S)
    if fh is not None and is_feasible(fh, n):
        fs = exact_score(fh)
        if fs is not None and np.isfinite(fs):
            return fh, fs, n
    if parent is not None and is_feasible(parent, n):
        ps = exact_score(parent)
        if ps is not None and np.isfinite(ps):
            return parent.copy(), ps, n
    fs = exact_score(const_fb)
    if fs is None:
        fs = 0.5
    return const_fb, fs, n
