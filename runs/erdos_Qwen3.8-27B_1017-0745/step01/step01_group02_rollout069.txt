def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    if S < -1e-12 or S > n + 1e-12:
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(v - tau, 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-15:
        for i in np.argsort(1.0 - y):
            d = min(r, 1.0 - y[i])
            y[i] += d
            r -= d
            if r <= 0:
                break
    elif r < -1e-15:
        for i in np.argsort(y):
            d = min(-r, y[i])
            y[i] -= d
            r += d
            if r >= 0:
                break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-9) or np.any(y > 1.0 + 1e-9):
        return None
    if abs(float(np.sum(y)) - S) > 1e-10 * max(1, n):
        return None
    return y

def check_feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(float(np.sum(h)) - 0.5 * n) < 1e-10 * max(1, n))

def exact_score(h):
    n = len(h)
    if not check_feasible(h, n):
        return float('inf')
    try:
        corr = np.correlate(h, 1.0 - h, mode='full')
        if len(corr) != 2 * n - 1 or not np.all(np.isfinite(corr)):
            return float('inf')
        return float(np.max(corr) * (2.0 / n))
    except Exception:
        return float('inf')

def lag_vg(h, s, n):
    dx = 2.0 / n
    lo = max(0, -s)
    hi = min(n, n - s)
    if lo >= hi:
        return 0.0, np.zeros(n)
    i = np.arange(lo, hi)
    j = i + s
    q = dx * float(np.sum(h[i] * (1.0 - h[j])))
    g = np.zeros(n)
    g[i] += dx * (1.0 - h[j])
    g[j] -= dx * h[i]
    return q, g

def separation(x, d, n):
    dx = 2.0 / n
    best_val = -float('inf')
    best_s = 0
    for s in range(-(n - 1), n):
        lo = max(0, -s)
        hi = min(n, n - s)
        if lo >= hi:
            continue
        i = np.arange(lo, hi)
        j = i + s
        q = dx * float(np.sum(x[i] * (1.0 - x[j])))
        gr = dx * float(np.sum((1.0 - x[j]) * d[i] - x[i] * d[j]))
        ell = q + gr
        if ell > best_val:
            best_val = ell
            best_s = s
    return best_val, best_s

def mm_obj(z):
    x = _mm['x']
    L = _mm['L']
    y = z[:-1]
    return 0.5 * L * float(np.dot(y - x, y - x)) + z[-1]

def mm_obj_grad(z):

    L = _mm['L']
    n = len(x)
    g = np.zeros(n + 1)
    g[:n] = L * (z[:n] - x)
    g[n] = 1.0
    return g

def mm_ineq(z):
    y = z[:-1]
    x = _mm['x']
    d = y - x
    qJ = _mm['qJ']
    GJ = _mm['GJ']
    return z[-1] - qJ - GJ @ d

    GJ = _mm['GJ']
    J = np.hstack([-GJ, np.ones((len(GJ), 1))])
    return J

def mm_eq(z):

def mm_eq_grad(z):

    g = np.zeros(n + 1)

def solve_mm(x, n):

    L = 2.0 * dx
    max_solves = 8
    cut_cap = min(2 * n - 1, 96)
    
    # Build initial active set

    all_s = list(range(-(n-1), n))
    for s in all_s:
        q, _ = lag_vg(x, s, n)
        all_q.append((q, s))
    all_q.sort(key=lambda t: (-t[0], t[1]))
    
    if n <= 64:
        active_set = list(all_s)
    else:
        active_set = [s for _, s in all_q[:min(8, len(all_q))]]

    for _ in range(max_solves):

        qJ = np.zeros(len(active_set))
        GJ = np.zeros((len(active_set), n))
        for k, s in enumerate(active_set):

            z_new = res.x

            EJ = float(np.max(qJ + GJ @ d)) if len(active_set) > 0 else 0.0

    """Project onto [0,1]^n with sum = S."""
    v = np.asarray(v, dtype=np.float64).copy()
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    n = len(v)

        return None

            y_cur = yn
        except Exception:
            break
    return y_cur

    max_sv, cap = 6, min(2 * n - 1, 96)
    ql = []
    for s in range(-(n - 1), n):
        q, _ = lv(x, s, n)
        ql.append((q, s))
    ql.sort(key=lambda t: (-t[0], t[1]))
    act = [s for _, s in ql] if n <= 64 else [s for _, s in ql[:8]]
    y_cur = x.copy()
    for _ in range(max_sv):
        act = act[:cap]
        qJ = np.array([lv(x, s, n)[0] for s in act])
        GJ = np.array([lv(x, s, n)[1] for s in act])
        _mm['x'], _mm['L'], _mm['n'] = x, L, n
        _mm['qJ'], _mm['GJ'] = qJ, GJ
        t0 = float(np.max(qJ + GJ @ (y_cur - x)))
        z0 = np.concatenate([y_cur, [t0]])
        bnds = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
        cons = [{'type': 'ineq', 'fun': mi, 'jac': mig},
                {'type': 'eq', 'fun': me, 'jac': meg}]
        try:
            res = minimize(mo, z0, jac=mog, method='SLSQP', bounds=bnds,
                           constraints=cons, options={'maxiter': 150, 'ftol': 1e-10})
            yr = np.clip(res.x[:n], 0.0, 1.0)
            yn = pb(yr, 0.5 * n)
            if yn is None:
                break
            d = yn - x
            EJ = float(np.max(qJ + GJ @ d))
            fv, fs = sep(x, d, n)
            if fv > EJ + 1e-9 * max(1.0, abs(EJ)):
                if fs not in act and len(act) < cap:
                    act.append(fs)
                    y_cur = yn
                    continue
            if 0.5 * L * float(np.dot(d, d)) + EJ > float(np.max(qJ)) + 1e-7:
                break
            y_cur = yn
        except Exception:
            break
    return y_cur
