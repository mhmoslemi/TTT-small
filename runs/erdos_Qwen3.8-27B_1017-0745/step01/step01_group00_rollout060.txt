    dx = 2.0 / n

    q0 = max(-d, 0)

    # Second factor: -h[p0+r] * h[q0+r]
    g[q0:q0+m] -= dx * h[p0:p0+m]
    return g

    return t - q

        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(q)):
        return float('inf')
    return float(np.max(q))

def lag_values(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def lag_gradient(d, h, n):
    dx = 2.0 / n
    g = np.zeros(n)
    p0 = max(d, 0)
    q0 = max(-d, 0)
    m = n - abs(d)
    g[p0:p0 + m] += dx * (1.0 - h[q0:q0 + m])
    g[q0:q0 + m] -= dx * h[p0:p0 + m]
    return g

def _eps_obj(z, n):
    return z[n]

def _eps_obj_jac(z, n):
    j = np.zeros(n + 1)
    j[n] = 1.0
    return j

def _eps_ineq(z, n):
    h = z[:n]
    return z[n] - np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eps_ineq_jac(z, n):
    h = z[:n]
    nlag = 2 * n - 1
    J = np.zeros((nlag, n + 1))
    for d in range(-(n - 1), n):
        J[d + n - 1, :n] = -lag_gradient(d, h, n)
        J[d + n - 1, n] = 1.0
    return J

def _eps_eq(z, n):
    return np.sum(z[:n]) - n / 2.0

def _eps_eq_jac(z, n):
    J = np.zeros(n + 1)
    J[:n] = 1.0
    return J

def slsqp_epigraph(h0, n, maxiter=250):
    q = lag_values(h0, n)
    t0 = float(np.max(q)) + max(1e-10, 1e-8 * max(1.0, abs(float(np.max(q)))))
    z0 = np.concatenate([h0.astype(np.float64), [t0]])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    cons = [
        {'type': 'ineq', 'fun': _eps_ineq, 'jac': _eps_ineq_jac},
        {'type': 'eq', 'fun': _eps_eq, 'jac': _eps_eq_jac},
    ]
    try:
        res = minimize(_eps_obj, z0, args=(n,), method='SLSQP',
                       jac=_eps_obj_jac, bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
    except Exception:
        return None
    if res.x is None or res.x.shape != (n + 1,) or not np.all(np.isfinite(res.x)):
        return None
    return res.x[:n]

def slsqp_polish(h0, n, n_passes=3):
    h = project_capped(h0, n)
    if h is None:
        return None, float('inf')
    score = exact_score(h, n)
    for _ in range(n_passes):
        y = slsqp_epigraph(h, n)
        if y is None:
            break
        y = project_capped(y, n)
        if y is None:
            break
        best_z, best_s = h.copy(), score
        for k in range(6):
            alpha = 1.0 / (2.0 ** k)
            z = project_capped(h + alpha * (y - h), n)
            if z is None:
                continue
            sz = exact_score(z, n)
            if sz < best_s:
                best_s, best_z = sz, z
        if best_s >= score - 1e-12:
            break
        h, score = best_z.copy(), best_s
    return h, score

def coord_refine(h, n, n_rounds=3):
    h = h.copy().astype(np.float64)
    score = exact_score(h, n)
    dx = 2.0 / n
    for _ in range(n_rounds):
        improved = False
        for i in range(n):
            for delta in (0.02, 0.01, 0.005, -0.02, -0.01, -0.005):
                h[i] += delta
                if h[i] < 0 or h[i] > 1:
                    h[i] -= delta
                    continue
                h2 = project_capped(h, n)
                if h2 is None:
                    h[i] -= delta
                    continue
                s2 = exact_score(h2, n)
                if s2 < score - 1e-13:
                    h, score = h2, s2
                    improved = True
                    break
                h[i] -= delta
        if not improved:
            break
    return h, score

def subgrad_search(h, n, n_iter=45):
    dx = 2.0 / n
    h = h.copy().astype(np.float64)
    score = exact_score(h, n)
    eta = min(0.15 / dx, 2.0 / dx)
    eta_max = 2.0 / dx
    fails = 0
    for _ in range(n_iter):
        if fails >= 8 or eta < 1e-8 / dx:
            break
        q = lag_values(h, n)
        d = int(np.argmax(q)) - (n - 1)
        g = lag_gradient(d, h, n)
        z = project_capped(h - eta * g, n)
        if z is None:
            eta *= 0.5
            fails += 1
            continue
        v = z - h
        if np.max(np.abs(v)) < 1e-10:
            eta *= 0.5
            fails += 1
            continue
        best_z, best_s = h.copy(), score
        for k in range(5):
            alpha = 1.0 / (2.0 ** k)
            ct = project_capped(h + alpha * v, n)
            if ct is None:
                continue
            cs = exact_score(ct, n)
            if cs < best_s:
                best_s, best_z = cs, ct
        if best_s < score - 1e-12:
            h, score = best_z.copy(), best_s
            fails = 0
            eta = min(1.2 * eta, eta_max)
        else:
            eta *= 0.5
            fails += 1
    return h, score

def build_starts(fallback, n, rng, count):
    pool = [fallback.copy()]
    cands = [
        fallback[::-1].copy(),
        1.0 - fallback[::-1],
        np.full(n, 0.5),
    ]
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    cands.append(blk)
    cands.append(np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64))
    for sd in (0.06, 0.15, 0.30):
        cands.append(fallback + rng.normal(0, sd, n))
    cands.append(rng.uniform(0, 1, n))
    t = np.linspace(0, 1, n)
    cands.append(1.0 - np.abs(2.0 * t - 1.0))
    for c in cands:
        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-8:
                dup = True
                break
        if not dup:
            pool.append(p)
    return pool[:count]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if parent is not None:
        fallback = project_capped(parent, n)
        if fallback is None:
            fallback = np.full(n, 0.5)
    else:
        fallback = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    fb_score = exact_score(fallback, n)
    gb, gs = fallback.copy(), fb_score
    if n == 1:
        return fallback, float(gs), 1
    if n <= 64:
        starts = build_starts(fallback, n, rng, 10)
    elif n <= 128:
        starts = build_starts(fallback, n, rng, 3)
    else:
        starts = build_starts(fallback, n, rng, 2)
    for st in starts:
        s = exact_score(st, n)
        if s < gs - 1e-12:
            gs, gb = s, st.copy()
    if n <= 64:
        for st in starts:
            try:
                rh, rs = slsqp_polish(st, n, n_passes=3)
                if rh is not None and rs < gs - 1e-12:
                    gs, gb = rs, rh.copy()
            except Exception:
                pass
        try:
            rh, rs = coord_refine(gb, n, n_rounds=4)
            if rh is not None and rs < gs - 1e-12:
                gs, gb = rs, rh.copy()
        except Exception:
            pass
        for st in starts[:4]:
            try:
                rh, rs = slsqp_polish(st, n, n_passes=2)
                if rh is not None and rs < gs - 1e-12:
                    gs, gb = rs, rh.copy()
            except Exception:
                pass
        try:
            rh, rs = coord_refine(gb, n, n_rounds=2)
            if rh is not None and rs < gs - 1e-12:
                gs, gb = rs, rh.copy()
        except Exception:
            pass
    else:
        for st in starts:
            try:
                rh, rs = subgrad_search(st, n, 45 if n <= 128 else 25)
                if rh is not None and rs < gs - 1e-12:
                    gs, gb = rs, rh.copy()
            except Exception:
                pass
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    if fs > fb_score + 1e-12:
        final, fs = fallback, fb_score
    final = project_capped(final, n)
    if final is None:
        final = np.full(n, 0.5)
        fs = exact_score(final, n)
    return final, float(fs), int(n)
