        elif r < 0.85:
            c = project_capped(rng.uniform(0, 1, n), n)
        else:
            c = np.zeros(n)
            c[rng.permutation(n)[:n // 2]] = 1.0
            if n % 2:
                c[rng.permutation(n)[n // 2]] = 0.5
        if c is None:
            c = gb.copy()
        pop.append(c)

    scores = np.array([exact_score(p, n) for p in pop])
    if gs < float(np.min(scores)) - 1e-12:
        pop[int(np.argmax(scores))] = gb.copy()
        scores[int(np.argmax(scores))] = gs
    F_arr = rng.uniform(0.45, 0.85, M)
    CR_arr = rng.uniform(0.4, 0.9, M)
    stall = 0
    nrest = 0
    prev_gs = gs

    for gen in range(G):
        psap = [p.copy() for p in pop]
        ssap = scores.copy()
        bi = int(np.argmin(ssap))
        xb = psap[bi]
        npop = [p.copy() for p in pop]
        nsco = scores.copy()
        nF = F_arr.copy()
        nCR = CR_arr.copy()
        for i in range(M):
            fi = F_arr[i]
            cri = CR_arr[i]
            if rng.random() < 0.1:
                fi = 0.35 + 0.55 * rng.random()
            if rng.random() < 0.1:
                cri = 0.10 + 0.85 * rng.random()
            if rng.random() < 0.5:
                oth = [j for j in range(M) if j != i]
                ix = rng.choice(oth, 3, replace=False)
                mut = psap[ix[0]] + fi * (psap[ix[1]] - psap[ix[2]])
            else:
                oth = [j for j in range(M) if j != i]
                ix = rng.choice(oth, 2, replace=False)
                mut = psap[i] + fi * (xb - psap[i]) + fi * (psap[ix[0]] - psap[ix[1]])
            msk = rng.random(n) < cri
            msk[int(rng.integers(n))] = True
            tr = np.where(msk, mut, psap[i])
            tr = project_capped(tr, n)
            if tr is None:
                continue
            ts = exact_score(tr, n)
            if ts < ssap[i] - 1e-12:
                npop[i] = tr
                nsco[i] = ts
                nF[i] = fi
                nCR[i] = cri
        pop, scores, F_arr, CR_arr = npop, nsco, nF, nCR
        if gs < float(np.min(scores)) - 1e-12:
            wi = int(np.argmax(scores))
            pop[wi] = gb.copy()
            scores[wi] = gs
        if gs < prev_gs - 1e-12:
            stall = 0
        else:
            stall += 1
        prev_gs = gs
        if stall > sl and nrest < nr:
            nrest += 1
            stall = 0
            ord_ = np.argsort(scores)
            kk = max(2, M // 4)
            ki = ord_[:kk]
            keep_pop = [pop[int(k)].copy() for k in ki]
            keep_sc = [float(scores[int(k)]) for k in ki]
            gb_kept = any(abs(kc - gs) < 1e-10 for kc in keep_sc)
            if not gb_kept:
                wi2 = int(np.argmax(keep_sc))
                keep_pop[wi2] = gb.copy()
                keep_sc[wi2] = gs
            sr = min([0.06, 0.16, 0.32][min(nrest - 1, 2)] * sc, 0.45)
            while len(keep_pop) < M:
                r = rng.random()
                if r < 0.5:
                    c = project_capped(gb + rng.normal(0, sr, n), n)
                elif r < 0.75:
                    src2 = keep_pop[int(rng.integers(len(keep_pop)))]
                    c = project_capped(src2 + rng.normal(0, sr * 0.5, n), n)
                elif r < 0.875:
                    c = project_capped(rng.uniform(0, 1, n), n)
                else:
                    c = np.zeros(n)
                    c[rng.permutation(n)[:n // 2]] = 1.0
                    if n % 2:
                        c[rng.permutation(n)[n // 2]] = 0.5
                if c is None:
                    c = gb.copy()
                keep_pop.append(c)
            pop = keep_pop
            scores = np.array([exact_score(p, n) for p in pop])
            F_arr = rng.uniform(0.45, 0.85, M)
            CR_arr = rng.uniform(0.4, 0.9, M)

    delta = min(0.25, 0.12 * sc)
    xb = gb.copy()
    ord_ = np.argsort(scores)
    bq = max(2, M // 4)
    sidx = ord_[:bq]
    for rnd in range(pr):
        if delta < 1e-4:
            break
        bt, bs = None, gs
        hnf = pd // 2
        for d in range(pd):
            if d < hnf:
                v = rng.normal(0, 1, n)
                v -= np.mean(v)
            else:
                i1 = int(sidx[int(rng.integers(len(sidx)))])
                i2 = int(sidx[int(rng.integers(len(sidx)))])
                if i1 == i2:
                    i2 = (i2 + 1) % len(sidx)
                v = pop[i1].copy() - pop[i2].copy()
                v -= np.mean(v)
            vm = float(np.max(np.abs(v)))
            if vm < 1e-14 or not np.isfinite(vm):
                continue
            v /= vm
            tlo, thi = -np.inf, np.inf
            pos = v > 1e-14
            neg = v < -1e-14
            if np.any(pos):
                tlo = max(tlo, float(np.max(-xb[pos] / v[pos])))
            if np.any(neg):
                thi = min(thi, float(np.max((1.0 - xb[neg]) / v[neg])))
            if tlo >= thi - 1e-14:
                continue
            tested = []
            for t in (-delta, -0.5 * delta, 0.5 * delta, delta):
                tc = float(np.clip(t, tlo, thi))
                if abs(tc) < 1e-12:
                    continue
                if any(abs(tc - tt) < 1e-12 for tt in tested):
                    continue
                tested.append(tc)
                tr2 = project_capped(xb + tc * v, n)
                if tr2 is None:
                    continue
                ts2 = exact_score(tr2, n)
                if ts2 < bs - 1e-12:
                    bs, bt = ts2, tr2.copy()
        if bt is not None:
            gs, gb, xb = bs, bt, bt.copy()
            delta = min(0.25, 1.2 * delta)
        else:
            delta *= 0.5

    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb.copy()
    fs = exact_score(final, n)
    fbs = exact_score(fb, n)
    if fs > fbs + 1e-12:
        final, fs = fb.copy(), fbs
    return final, float(fs), int(n)
