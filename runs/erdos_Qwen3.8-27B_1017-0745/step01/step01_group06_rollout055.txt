    for k in range(2 * n - 1):

            hi = min(delta, 1.0 - h[j])

                           method='SLSQP', bounds=bounds, constraints=con,

            consec_fail += 1
            if consec_fail >= 4 or delta < 1e-5:

def tr_ineq_jac(z, n, q, G):
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def tr_eq(z, n):
    return z[:-1].sum()

def tr_eq_jac(z, n):
    g = np.zeros(len(z))
    g[:-1] = 1.0
    return g

def trust_region(h, n, dx, S, delta0, eta0, max_iter, slsqp_max):
    delta, eta = delta0, eta0
    consec_fail = 0
    for _ in range(max_iter):
        q = lag_vec(h, dx)
        G = lag_grad(h, n, dx)
        t0 = float(q.max())
        z0 = np.concatenate([np.zeros(n), [t0]])
        bounds = [(max(-delta, -h[j]), min(delta, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        con = [{'type': 'ineq', 'fun': tr_ineq, 'jac': tr_ineq_jac, 'args': (n, q, G)},
               {'type': 'eq', 'fun': tr_eq, 'jac': tr_eq_jac, 'args': (n,)}]
        try:
            res = minimize(tr_obj, z0, jac=tr_obj_jac, args=(n, eta), method='SLSQP',
                           bounds=bounds, constraints=con,
                           options={'maxiter': slsqp_max, 'ftol': 1e-12, 'disp': False})
        except Exception:
            delta *= 0.5
            consec_fail += 1
            if consec_fail >= 4 or delta < 1e-5:
                break
            continue
        if not hasattr(res, 'x') or not np.all(np.isfinite(res.x)) or len(res.x) != n + 1:
            delta *= 0.5
            consec_fail += 1
            if consec_fail >= 4 or delta < 1e-5:
                break
            continue
        d = res.x[:n]
        if abs(d.sum()) > 1e-4 or np.any(d < -h - 1e-4) or np.any(d > 1.0 - h - 1e-4):
            delta *= 0.5
            consec_fail += 1
            if consec_fail >= 4 or delta < 1e-5:
                break
            continue
        h_t = project_K(h + d, n, S)
        if not feasible(h_t, n, S):
            delta *= 0.5
            consec_fail += 1
            if consec_fail >= 4 or delta < 1e-5:
                break
            continue
        d_act = h_t - h
        F = float(q.max())
        F_t = float(lag_vec(h_t, dx).max())
        if np.max(np.abs(d_act)) < 1e-9:
            break
        pred = F - float((q + G @ d_act).max())
        actual = F - F_t
        if actual > 1e-12:
            h = h_t
            consec_fail = 0
            if pred > 1e-12:
                ratio = actual / pred
                if ratio < 0.25:
                    delta *= 0.5
                elif ratio > 0.75 and np.max(np.abs(d_act)) >= 0.8 * delta:
                    delta = min(0.5, 2.0 * delta)
                    eta = min(2.0, 1.5 * eta)
        else:
            delta *= 0.5
            eta = max(1e-4, eta * 0.5)
            consec_fail += 1
            if consec_fail >= 8 or delta < 1e-5:
                break
    return project_K(h, n, S)

def polish(h, n, dx, S, rng, iters=200):
    best = h.copy()
    bs = exact_score(best, dx)
    for _ in range(iters):
        j = rng.integers(n)
        h2 = best.copy()
        h2[j] += rng.choice([-1.0, 1.0]) * rng.uniform(0.01, 0.06)
        h2 = project_K(h2, n, S)
        if feasible(h2, n, S):
            sc = exact_score(h2, dx)
            if sc < bs - 1e-12:
                best, bs = h2, sc
    return best, bs

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    init = globals().get('initial_h_values', None)
    ok = False
    n0 = 45
    if init is not None:
        init = np.array(init, dtype=np.float64).ravel()
        if (2 <= len(init) <= 999 and np.all(np.isfinite(init)) and
                init.min() >= -1e-6 and init.max() <= 1.0 + 1e-6):
            n0 = len(init)
            if abs(init.sum() - n0 / 2.0) <= 1e-6:
                ok = True
            else:
                init = project_K(init, n0, n0 / 2.0)
                ok = True
    best_h, best_s, best_n = None, float('inf'), 0
    fb_h, fb_s, fb_n = None, float('inf'), 0

    res_list = []
    if ok:
        res_list = [n0, max(2, n0 - 1), min(999, n0 + 1)]
        res_list = list(dict.fromkeys(res_list))
    else:
        res_list = [n0]

    for m in res_list:
        S = m / 2.0
        dx = 2.0 / m
        if ok and m == n0:
            h_base = project_K(init.copy(), m, S)
        elif ok:
            h_base = transfer_parent(init, n0, m)
        else:
            h_base = np.full(m, 0.5)
        sc_base = exact_score(h_base, dx)
        if m == n0 and ok:
            fb_h, fb_s, fb_n = h_base.copy(), sc_base, m

        starts = [h_base.copy(), (1.0 - h_base).copy(), h_base[::-1].copy(),
                  (1.0 - h_base[::-1]).copy()]
        if ok:
            starts.append(project_K(h_base + 0.10 * rng.standard_normal(m), m, S))
            starts.append(project_K(h_base + 0.30 * rng.standard_normal(m), m, S))
            perm = rng.permutation(m)
            starts.append(project_K(h_base[perm], m, S))
        else:
            starts.append(project_K(np.full(m, 0.5 + 0.35 * rng.standard_normal(m)), m, S))
        starts.append(project_K(rng.uniform(0, 1, m), m, S))
        starts.append(project_K(0.5 + 0.45 * np.cos(2.0 * np.pi * np.arange(m) / m), m, S))

        dedup = []
        for s in starts:
            s = project_K(s, m, S)
            if not feasible(s, m, S):
                continue
            close = False
            for ds in dedup:
                if np.max(np.abs(s - ds)) <= 1e-6:
                    close = True
                    break
            if not close:
                dedup.append(s)

        mi = 30 if m <= 100 else 12
        slq = 50 if m <= 100 else 30
        for s in dedup[:8 if m <= 100 else 3]:
            h_opt = trust_region(s.copy(), m, dx, S, 0.12, 0.5, mi, slq)
            if feasible(h_opt, m, S):
                sc = exact_score(h_opt, dx)
                if sc < best_s - 1e-12:
                    best_s, best_h, best_n = sc, h_opt.copy(), m
                if sc < best_s - 1e-12 or (best_h is None):
                    best_s, best_h, best_n = sc, h_opt.copy(), m

        if m <= 100:
            h_p, sp = polish(best_h.copy() if best_n == m else dedup[0].copy(), m, dx, S, rng, 300)
            if feasible(h_p, m, S) and sp < best_s - 1e-12:
                best_s, best_h, best_n = sp, h_p.copy(), m

    if best_h is None:
        if fb_h is not None:
            best_h, best_s, best_n = fb_h, fb_s, fb_n
        else:
            m = n0
            best_h = np.full(m, 0.5)
            best_s = exact_score(best_h, 2.0 / m)
            best_n = m

    best_h = project_K(best_h, best_n, best_n / 2.0)
    final_s = exact_score(best_h, 2.0 / best_n)
    if fb_h is not None and final_s > fb_s + 1e-10:
        best_h, final_s, best_n = fb_h, fb_s, fb_n
    return (best_h, final_s, best_n)
