def project_capped(v, n):

    s = n / 2.0

def _proj(v, n):
    s = n / 2.0
    v = np.asarray(v, dtype=np.float64)
    if v.shape != (n,) or not np.all(np.isfinite(v)):
        return None
    v = np.clip(v, -1e6, 1e6)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if mid == lo or mid == hi:
            break
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    if r > 1e-14:
        for idx in np.argsort(1.0 - h):
            if r <= 1e-15:
                break
            a = min(r, 1.0 - h[idx]); h[idx] += a; r -= a
    elif r < -1e-14:
        for idx in np.argsort(h):
            if r >= -1e-15:
                break
            a = max(r, -h[idx]); h[idx] += a; r -= a
    h = np.clip(h, 0.0, 1.0)
    if abs(float(np.sum(h)) - s) > 1e-10 * n or not np.all(np.isfinite(h)):
        return None
    return h

def _score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return float('inf')
    if abs(float(np.sum(h)) - n / 2.0) > 1e-9 * n:
        return float('inf')
    try:
        c = np.correlate(h, 1.0 - h, mode='full')
    except Exception:
        return float('inf')
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c) * (2.0 / n))

def _sg(h, n, tau):
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    g = raw * dx
    gm = float(np.max(g))
    ex = np.exp((g - gm) / tau)
    se = float(np.sum(ex))
    p = ex / se
    F = gm + tau * np.log(se)
    grad = np.zeros(n)
    ks = np.where(p > 1e-14)[0]
    for k in ks:
        d = int(k) - (n - 1)
        wk = p[k] * dx
        i0 = max(0, -d); i1 = min(n - 1, n - 1 - d)
        if i0 <= i1:
            grad[i0:i1+1] += wk * (1.0 - h[i0+d:i1+d+1])
        j0 = max(0, d); j1 = min(n - 1, d + n - 1)
        if j0 <= j1:
            grad[j0:j1+1] -= wk * h[j0-d:j1-d+1]
    return F, grad

def _pollish(h, n, n_sweeps):
    """Coordinate nudge polish."""
    best = h.copy()
    best_s = _score(h, n)
    for sw in range(n_sweeps):
        for i in range(n):
            for delta in (-0.05, -0.02, -0.005, 0.005, 0.02, 0.05):
                t = best.copy()
                t[i] = max(0.0, min(1.0, t[i] + delta))
                tp = _proj(t, n)
                if tp is None:
                    continue
                ts = _score(tp, n)
                if ts < best_s - 1e-13:
                    best = tp; best_s = ts
    return best, best_s

def _opt(h0, n, tau_hi, tau_lo, nlev, niter):
    h = h0.copy()
    alpha = n / 4.0
    for lev in range(nlev):
        t = tau_hi * (tau_lo / tau_hi) ** (lev / max(1, nlev - 1))
        Fv, gv = _sg(h, n, t)
        prev_F = Fv
        prev_g = gv.copy()
        x_old = h.copy()
        for it in range(niter):
            yv = _proj(h - alpha * gv, n)
            if yv is None:
                alpha *= 0.5
                if alpha < 1e-8:
                    break
                continue
            nyv = _sg(yv, n, t)
            if not np.isfinite(nyv[0]) or nyv[0] >= prev_F - 1e-15:
                alpha *= 0.5
                if alpha < 1e-8:
                    break
                continue
            s_vec = yv - x_old
            x_old = h
            h = yv
            prev_F = nyv[0]
            y_vec = nyv[1] - prev_g
            sy = float(np.dot(s_vec, y_vec))
            ss = float(np.dot(s_vec, s_vec))
            if sy > 1e-14:
                alpha = ss / sy
            else:
                alpha = alpha * 2.0
            alpha = max(1e-5, min(alpha, 5.0 * n))
            prev_g = nyv[1].copy()
            if max(abs(s_vec)) < 1e-11:
                break
    return h

def _starts(n, rng, parent):
    out = []
    def add(v):
        hp = _proj(v, n)
        if hp is None:
            return
        for s in out:
            if float(np.max(np.abs(hp - s))) < 1e-10:
                return
        out.append(hp)
    uni = np.full(n, 0.5)
    add(uni)
    if parent is not None:
        add(parent.copy())
        add(parent[::-1].copy())
        add(1.0 - parent[::-1].copy())
        add(1.0 - parent.copy())
        for sd in (0.02, 0.05, 0.10, 0.18, 0.30):
            add(parent + rng.normal(0, sd, n))
    idx = np.arange(n, dtype=np.float64)
    for q in (1, 2, 3, 5, 7):
        add(0.5 + 0.49 * np.cos(2.0 * np.pi * q * idx / n))
        add(0.5 + 0.49 * np.sin(2.0 * np.pi * q * idx / n))
    blk = np.zeros(n); blk[:n//2] = 1.0
    if n % 2 == 1:
        blk[n//2] = 0.5
    add(blk)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    add(alt)
    for _ in range(4):
        add(rng.uniform(0, 1, n))
    return out

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= len(iv) <= 999:
            n = len(iv)
            if np.all(np.isfinite(iv)) and np.all(iv >= -1e-8) and np.all(iv <= 1.0 + 1e-8):
                parent = iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
    if n < 1 or n > 999:
        n = 47
    try:
        sd = int(seed)
    except Exception:
        sd = 42
    sd = sd % (2**32)
    rng = np.random.default_rng(sd)
    # Resolve parent
    if parent is not None:
        parent = _proj(parent, n)
    # Build starts
    st_list = _starts(n, rng, parent)
    max_starts = min(14, len(st_list))
    st_list = st_list[:max_starts]
    # Score all starts
    best_h, best_s = None, float('inf')
    uni = np.full(n, 0.5)
    us = _score(uni, n)
    best_h, best_s = uni.copy(), us
    if parent is not None:
        ps = _score(parent, n)
        if ps < best_s - 1e-13:
            best_h, best_s = parent.copy(), ps
    for st in st_list:
        ss = _score(st, n)
        if ss < best_s - 1e-13:
            best_h, best_s = st.copy(), ss
    # Optimize each start with continuation smoothed PGD
    if n <= 256:
        tau_hi, tau_lo = 0.04, 0.0005
        nlev = 10
        niter = 80
        for si, st in enumerate(st_list):
            h_opt = _opt(st, n, tau_hi, tau_lo, nlev, niter)
            hs = _score(h_opt, n)
            if hs < best_s - 1e-13:
                best_h, best_s = h_opt.copy(), hs
        # Second round: shorter, from improved solutions
        tau_hi2, tau_lo2 = 0.01, 0.0002
        nlev2, niter2 = 6, 120
        for st in st_list[:6]:
            h_opt = _opt(st, n, tau_hi2, tau_lo2, nlev2, niter2)
            hs = _score(h_opt, n)
            if hs < best_s - 1e-13:
                best_h, best_s = h_opt.copy(), hs
    # Polish
    if n <= 300:
        best_h, best_s = _pollish(best_h, n, 6)
    # Final validation
    fp = _proj(best_h, n)
    if fp is not None:
        fs = _score(fp, n)
        if fs <= best_s + 1e-12:
            best_h, best_s = fp, fs
    if not np.isfinite(best_s):
        best_h = uni.copy()
        best_s = _score(uni, n)
    return best_h, float(best_s), int(n)
