The overlap integral C5 = max_k ∫ h(x)(1-h(x+k)) dx. We want to minimize this.

One key insight: if h is a step function that's 0 on one part and 1 on the other (a "bang-bang" solution), then for shifts where the 1-region maps to the 0-region, the product h(x)(1-h(x+k)) is maximized. So we want to arrange the 1-region such that for all shifts, the overlap of the 1-region with the 0-region (shifted) is minimized.

This is related to the concept of "almost orthogonal" binary sequences. For a binary sequence of N bits with N/2 ones and N/2 zeros, we want to minimize the maximum cross-correlation between the sequence and its complement at all shifts.

Actually, for a binary (0/1) sequence, h(x)(1-h(x+k)) is 1 where h(x)=1 and h(x+k)=0, and 0 otherwise. So the sum is the number of positions where h has a 1 and the shifted h has a 0.

For a perfectly periodic pattern like 101010..., at even shifts the overlap is 0 and at odd shifts it's N/2. The max would be N/2 * dx = N/2 * 2/N = 1. That's bad.

For a pattern like 1100 1100 1100..., at shift 2: h=1100 shifted by 2 is 0011, overlap = 0. At shift 1: 1001, overlap = 2. At shift 4: 1100, overlap = 0. Hmm.

import numpy as np

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo, hi = z.min() - 1.0, z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    if r > 1e-14:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
            if r <= 1e-14:
                break
    elif r < -1e-14:
        for i in range(N):
            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1 + 1e-10:
        return np.full(N, 0.5)
    if abs(h.sum() - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / N)

def smooth_value_grad(h, dx, tau):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = dx * corr
    cmax = np.max(c)
    e = np.exp((c - cmax) / tau)
    se = np.sum(e)
    F = cmax + tau * np.log(se)
    p = e / se
    grad = np.zeros(N)
    dx_p0 = dx * p[N - 1]
    grad += dx_p0 * (1.0 - 2.0 * h)
    for s in range(1, N):
        ps = dx * p[N - 1 + s]
        if ps != 0.0:
            L = N - s
            grad[s:N] += ps * (1.0 - h[0:L])
            grad[0:L] -= ps * h[s:N]
    for s in range(1, N):
        ps = dx * p[N - 1 - s]
        if ps != 0.0:
            L = N - s
            grad[0:L] += ps * (1.0 - h[s:N])
            grad[s:N] -= ps * h[0:L]
    return F, grad, c

def optimize_one(h0, N, m, dx, tau_schedule, max_iter):
    h = h0.copy()
    best_h = h.copy()
    best_score = score_exact(h, N)
    alpha_init = 1.0
    for tau in tau_schedule:
        F, grad, _ = smooth_value_grad(h, dx, tau)
        if not np.all(np.isfinite(grad)):
            break
        alpha = alpha_init
        stagnant = 0
        for it in range(max_iter):
            y = project_box_sum(h - alpha * grad, m)
            d = y - h
            if not np.all(np.isfinite(d)):
                break
            if np.max(np.abs(d)) < 1e-10:
                break
            s_y = score_exact(y, N)
            if s_y < best_score:
                best_score = s_y
                best_h = y.copy()
            Fy, _, _ = smooth_value_grad(y, dx, tau)
            gd = np.dot(grad, d)
            if gd < 0.0 and Fy <= F + 1e-4 * gd + 1e-12:
                F_prev = F
                h = y
                F = Fy
                _, grad, _ = smooth_value_grad(h, dx, tau)
                if not np.all(np.isfinite(grad)):
                    break
                dec = F_prev - F
                if dec <= 1e-12 * max(1.0, abs(F_prev)):
                    stagnant += 1
                else:
                    stagnant = 0
                if stagnant >= 5:
                    break
                alpha = min(alpha * 1.25, 10.0)
            else:
                alpha *= 0.5
                if alpha < 1e-14:
                    break
    return best_h, best_score

def make_fourier_start(N, rng, n_modes, amp):
    z = np.full(N, 0.5)
    ns = min(n_modes, (N - 1) // 2)
    idx = np.arange(N)
    for r in range(1, ns + 1):
        A = rng.standard_normal()
        B = rng.standard_normal()
        z += (amp / r) * (A * np.sin(2.0 * np.pi * r * idx / N) + B * np.cos(2.0 * np.pi * r * idx / N))
    return z

    if N <= 128:

    else:

    tau_schedule = [0.03, 0.01, 0.004, 0.0015, 0.0005, 0.00015, 0.00005, 0.00001]

    starts.append(project_box_sum(1.0 - best, m))
