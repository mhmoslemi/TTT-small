import numpy as np
import math

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and np.all(h >= 0.0) and np.all(h <= 1.0) and abs(h.sum() - n / 2.0) <= 1e-10)

def project_capped(y, n):
    S = n / 2.0
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    lo = float(y.min()) - 1.0
    hi = float(y.max())
    lam = float(np.clip(y.mean() - 0.5, lo, hi))
    for _ in range(60):
        c = np.clip(y - lam, 0.0, 1.0)
        r = c.sum() - S
        if abs(r) <= 1e-13:
            break
        if r > 0:
            lo = lam
        else:
            hi = lam
        a = np.sum((y - lam > 0) & (y - lam < 1))
        if a > 0:
            lam_new = lam + r / a
            if lo < lam_new < hi:
                lam = lam_new
            else:
                lam = 0.5 * (lo + hi)
        else:
            lam = 0.5 * (lo + hi)
    c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(3):
        r = S - c.sum()
        if abs(r) < 1e-14:
            break
        if r > 0:
            mask = c < 1.0 - 1e-12
        else:
            mask = c > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        c[mask] += r / cnt
        np.clip(c, 0.0, 1.0, out=c)
    return c

def lag_scores(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    qs = lag_scores(h, n)
    if qs.shape != (2 * n - 1,) or not np.all(np.isfinite(qs)):
        return float('inf')
    return float(qs.max())

def build_G(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        lo = max(0, k)
        hi = min(n, n + k)
        row = k + n - 1
        sl1 = slice(lo, hi)
        sl2 = slice(lo - k, hi - k)
        G[row, sl1] += dx * (1.0 - h[sl2])
        G[row, sl2] -= dx * h[sl1]
    return G

def project_simplex(z):
    n = len(z)
    if n <= 1:
        return np.array([1.0]) if n == 1 else np.array([0.0])
    s = np.sort(z)[::-1]
    css = np.cumsum(s)
    rho = np.nonzero(s - (css - 1.0) / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        return np.full(n, 1.0 / n)
    r = rho[-1]
    theta = (css[r] - 1.0) / (r + 1)
    p = np.maximum(z - theta, 0.0)
    p /= p.sum()
    return p

def dual_eval(h, p, G, q, L, n):
    gmix = G.T @ p
    y = h - gmix / L
    y = project_capped(y, n)
    d = y - h
    u = q + G @ d
    D = p @ u + 0.5 * L * d @ d
    gap = u.max() - p @ u
    return d, u, D, gap

def solve_majorant(h, q, G, n, max_inner):
    dx = 2.0 / n
    L = 2.0 * dx
    Ldual = max(1e-12, 1.1 * np.sum(G * G) / L)
    m = q.shape[0]
    qmax = float(q.max())
    tol = max(1e-12, 1e-8 * max(1.0, abs(qmax)))
    active = np.nonzero(q >= qmax - tol)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(q))])
    p = np.zeros(m)
    p[active] = 1.0 / len(active)
    x = project_simplex(p)
    yv = x.copy()
    theta = 1.0
    best_D = -float('inf')
    best_d = None
    best_u = None
    consec_small = 0
    for _ in range(max_inner):
        d_y, u_y, D_y, gap_y = dual_eval(h, yv, G, q, L, n)
        if D_y > best_D:
            best_D = D_y
            best_d = d_y.copy()
            best_u = u_y.copy()
        if gap_y <= max(1e-10, 1e-8 * max(1.0, abs(u_y.max()))):
            break
        grad_y = u_y  # dual gradient is q + G@d = u
        x_trial = project_simplex(yv + grad_y / Ldual)
        d_t, u_t, D_t, _ = dual_eval(h, x_trial, G, q, L, n)
        stepnorm = np.max(np.abs(x_trial - x))
        if D_t < best_D - 1e-13:
            x_trial = project_simplex(x + u_y / Ldual)
            d_t, u_t, D_t, _ = dual_eval(h, x_trial, G, q, L, n)
            if D_t < best_D - 1e-13:
                if stepnorm < 1e-10:
                    consec_small += 1
                    if consec_small >= 3:
                        break
                continue
            yv = x_trial
            theta = 1.0
        else:
            if stepnorm < 1e-10:
                consec_small += 1
                if consec_small >= 3:
                    break
            else:
                consec_small = 0
            theta_new = (1.0 + math.sqrt(1.0 + 4.0 * theta * theta)) / 2.0
            yv = x_trial + ((theta - 1.0) / theta_new) * (x_trial - x)
            yv = project_simplex(yv)
            x = x_trial
            theta = theta_new
    if best_d is None:
        best_d, best_u, _, _ = dual_eval(h, x, G, q, L, n)
    return best_d, best_u

def optimize_start(h0, n, max_outer, max_inner):
    h = project_capped(h0.copy(), n)
    best_h = h.copy()
    best_s = exact_score(h, n)
    dx = 2.0 / n
    L = 2.0 * dx
    for _ in range(max_outer):
        q = lag_scores(h, n)
        if not np.all(np.isfinite(q)):
            break
        G = build_G(h, n)
        if not np.all(np.isfinite(G)):
            break
        d, u = solve_majorant(h, q, G, n, max_inner)
        if d is None or not np.all(np.isfinite(d)):
            break
        if np.max(np.abs(d)) <= 1e-12:
            break
        v = G @ d
        n2 = d @ d
        qmax = float(q.max())
        accepted = False
        for i in range(19):
            alpha = 1.0 / (2.0 ** i)
            Mval = float(np.max(q + alpha * v) + 0.5 * L * alpha * alpha * n2)
            if Mval >= qmax - 1e-14:
                continue
            h_trial = h + alpha * d
            h_trial = project_capped(h_trial, n)
            if not check_feasible(h_trial, n):
                continue
            s = exact_score(h_trial, n)
            if s < best_s - 1e-12:
                best_s = s
                best_h = h_trial.copy()
                h = h_trial.copy()
                accepted = True
                if s < qmax - 1e-11:
                    break
                break
        if not accepted:
            break
    return best_h, best_s

    parent = globals().get('initial_h_values', None)
    n = 45
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)

        sigs = [0.05, 0.15]
    elif n <= 400:

        sigs = [0.05, 0.15]

        sigs = [0.05]

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and np.all(h >= 0.0)
            and np.all(h <= 1.0) and abs(h.sum() - n / 2.0) <= 1e-10)

def project_capped(y, n):
    S = n / 2.0
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    lo = float(y.min()) - 1.0
    hi = float(y.max())
    lam = float(np.clip(np.mean(y) - 0.5, lo, hi))
    for _ in range(60):
        c = np.clip(y - lam, 0.0, 1.0)
        r = c.sum() - S
        if abs(r) <= 1e-13:
            break
        if r > 0:
            lo = lam
        else:
            hi = lam
        a = int(np.sum((y - lam > 0) & (y - lam < 1)))
        if a > 0:
            ln = lam + r / a
            lam = ln if lo < ln < hi else 0.5 * (lo + hi)
        else:

        cnt = int(mask.sum())
        if cnt == 0:
            break
        c[mask] += r / cnt
        np.clip(c, 0.0, 1.0, out=c)
    return c

def lag_scores(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    qs = lag_scores(h, n)
    if not np.all(np.isfinite(qs)):
        return float('inf')
    return float(np.max(qs))

def build_G(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        lo, hi = max(0, k), min(n, n + k)
        G[k + n - 1, lo:hi] += dx * (1.0 - h[lo - k:hi - k])
        G[k + n - 1, lo - k:hi - k] -= dx * h[lo:hi]
    return G

def project_simplex(z):
    m = len(z)
    if m <= 1:
        return np.full(m, 1.0 / max(m, 1))
    s = np.sort(z)[::-1]
    css = np.cumsum(s)
    idx = np.arange(1, m + 1, dtype=np.float64)
    rho = np.nonzero(s - (css - 1.0) / idx > 0)[0]
    if len(rho) == 0:
        return np.full(m, 1.0 / m)
    r = int(rho[-1])
    theta = (css[r] - 1.0) / (r + 1.0)
    p = np.maximum(z - theta, 0.0)
    s2 = p.sum()
    if s2 < 1e-15:
        return np.full(m, 1.0 / m)
    p /= s2
    return p

def dual_eval(h, p, G, q, L, n):
    y = project_capped(h - G.T @ p / L, n)
    d = y - h
    u = q + G @ d
    D = float(p @ u + 0.5 * L * d @ d)
    gap = float(np.max(u) - p @ u)
    return d, u, D, gap

def solve_majorant(h, q, G, n, max_inner):
