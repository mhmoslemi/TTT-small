import numpy as np
from scipy.optimize import minimize

_MM = {}

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-14:
        for i in np.argsort(1.0 - y):
            d = min(r, 1.0 - y[i]); y[i] += d; r -= d
            if r <= 1e-14: break
    elif r < -1e-14:
        for i in np.argsort(y):
            d = min(-r, y[i]); y[i] -= d; r += d
            if r >= -1e-14: break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-9) or np.any(y > 1.0 + 1e-9):
        return None
    if abs(float(np.sum(y)) - S) > 1e-9 * max(1, n):
        return None
    return y

def ok(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and np.all(h >= -1e-12)
            and np.all(h <= 1.0 + 1e-12) and abs(float(np.sum(h)) - 0.5 * n) < 1e-9 * max(1, n))

def score(h):
    n = len(h)
    if not ok(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    if c.shape[0] != 2 * n - 1 or not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c) * (2.0 / n))

def lvh(h, s, n):
    dx = 2.0 / n
    lo, hi = max(0, -s), min(n, n - s)
    q = 0.0
    g = np.zeros(n)
    for i in range(lo, hi):
        j = i + s
        q += h[i] * (1.0 - h[j])
        g[i] += 1.0 - h[j]
        g[j] -= h[i]
    return q * dx, g * dx

def _mo(z):
    d = z[:-1] - _MM['x']
    return 0.5 * _MM['L'] * float(d @ d) + z[-1]

def _mog(z):
    n = _MM['n']
    g = np.empty(n + 1)
    g[:n] = _MM['L'] * (z[:n] - _MM['x'])
    g[n] = 1.0
    return g

def _mi(z):
    d = z[:-1] - _MM['x']
    return z[-1] - _MM['qJ'] - _MM['GJ'] @ d

def _mij(z):
    n = _MM['n']; GJ = _MM['GJ']
    J = np.empty((GJ.shape[0], n + 1))
    J[:, :n] = -GJ
    J[:, n] = 1.0
    return J

def _me(z):
    return float(np.sum(z[:-1])) - _MM['n'] * 0.5

def _mej(z):
    g = np.zeros(_MM['n'] + 1)
    g[:-1] = 1.0
    return g

def solve_mm(x, n, Lfac):
    dx = 2.0 / n
    L = Lfac * dx
    nv = 2 * n - 1
    svals = np.arange(-(n - 1), n)
    qs = np.empty(nv); Gs = np.empty((nv, n))
    for k in range(nv):
        qs[k], Gs[k] = lvh(x, int(svals[k]), n)
    if n <= 64:
        active = list(range(nv))
    else:
        active = list(np.argsort(qs)[-min(nv, 64):])
    _MM['x'], _MM['L'], _MM['n'] = x, L, n
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    cons = [{'type': 'ineq', 'fun': _mi, 'jac': _mij},
            {'type': 'eq', 'fun': _me, 'jac': _mej}]
    best_y = x.copy(); best_mv = float('inf')
    max_att = max(4, min(8, nv // 20 + 4))
    for att in range(max_att):
        GJ, qJ = Gs[active], qs[active]
        _MM['GJ'], _MM['qJ'] = GJ, qJ
        y0 = x.copy() if att == 0 else best_y.copy()
        t0 = float(np.max(qJ + GJ @ (y0 - x))) + 1e-6
        z0 = np.append(y0, t0)
        try:
            res = minimize(_mo, z0, jac=_mog, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 150, 'ftol': 1e-11, 'disp': False})
        except Exception:
            break
        z = res.x
        if z is None or len(z) != n + 1 or not np.all(np.isfinite(z)):
            break
        yn = project_box_sum(z[:n], 0.5 * n)
        if yn is None:
            yn = x.copy()
        d = yn - x
        EJ = float(np.max(qJ + GJ @ d))
        mv = 0.5 * L * float(d @ d) + EJ
        if mv < best_mv:
            best_mv = mv; best_y = yn.copy()
        fm, fk = -1e30, -1
        for k in range(nv):
            e = qs[k] + Gs[k] @ d
            if e > fm: fm, fk = e, k
        if fm > EJ + 1e-8 and fk not in active and len(active) < min(nv, 96):
            active.append(fk)
        else:
            break
    return best_y

def mm_desc(h0, n, steps, Lfac):

    for _ in range(steps):

        d = y - h

                break
        if bc >= c - 1e-13:
            break
        h, c = bh, bc
