Let me analyze the problem and strategy carefully.

The task is to find a step function h: [0,2] → [0,1] that minimizes C₅ = max_k ∫ h(x)(1-h(x+k)) dx, subject to h(x) ∈ [0,1] and ∫₀² h(x) dx = 1.

The discretization uses n_points samples over [0,2], with dx = 2/n_points. The constraint is sum(h) * dx = 1, i.e., sum(h) = n_points/2.

The evaluation is C₅ = max(np.correlate(h, 1-h, mode="full")) * dx.

Current record: C₅ ≤ 0.38085, previous program achieved 0.381024.

The strategy proposes a Curvature-Majorized Majorization-Minimization (MM) approach:
- For each lag s, q_s(h) is quadratic with Hessian having largest eigenvalue ≤ 2
- So q_s(y) ≤ q_s(x) + g_s(x)ᵀ(y-x) + ||y-x||² for all y
- The MM step minimizes the convex quadratic upper bound over the feasible set
- Use SLSQP on this convex QP

import numpy as np
from scipy.optimize import minimize

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)) or abs(z).max() > 1e12:
        return np.full(N, 0.5)
    zc = z - np.median(z)
    lo, hi = zc.min() - 1.0, zc.max()
    for _ in range(100):
        mid = 0.5*(lo+hi)
        if np.clip(zc - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(zc - 0.5*(lo+hi), 0.0, 1.0)
    r = m - h.sum()
    for _ in range(2):
        if r > 1e-14:
            for i in range(N):
                a = min(r, 1.0 - h[i])
                h[i] += a; r -= a
                if r <= 1e-14: break
        elif r < -1e-14:
            for i in range(N):
                a = min(-r, h[i])
                h[i] -= a; r += a
                if r >= -1e-14: break
        else:
            break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1+1e-10:
        return np.full(N, 0.5)
    if abs(h.sum() - m) > 1e-8*max(1,N):
        return np.full(N, 0.5)
    return h

def is_feasible(h, N, m):
    return (h.shape == (N,) and np.all(np.isfinite(h)) and h.min() >= -1e-10
            and h.max() <= 1+1e-10 and abs(h.sum()-m) <= 1e-10*max(1,N))

def exact_score(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    corr = np.correlate(h, 1.0-h, mode='full')
    if not np.all(np.isfinite(corr)):
        return np.inf
    return float(np.max(corr) * 2.0 / N)

def lag_data(h, N):
    L = 2*N - 1
    q = np.zeros(L)
    G = np.zeros((L, N))
    for s in range(-(N-1), N):
        r = s + N - 1
        if s >= 0:
            ell = N - s
            q[r] = np.dot(h[:ell], 1.0 - h[s:s+ell])
            G[r, :ell] = 1.0 - h[s:s+ell]
            G[r, s:s+ell] -= h[:ell]
        else:
            d = -s
            ell = N - d
            q[r] = np.dot(h[d:], 1.0 - h[:ell])
            G[r, d:] = 1.0 - h[:ell]
            G[r, :ell] -= h[d:]
    return q, G

def _qp_obj(z, x, N, m):
    y = z[:N]
    return np.dot(y - x, y - x) + z[N]

def _qp_obj_jac(z, x, N, m):
    g = np.zeros(N+1)
    g[:N] = 2.0 * (z[:N] - x)
    g[N] = 1.0
    return g

def _qp_eq(z, x, N, m):
    return np.sum(z[:N]) - m

def _qp_eq_jac(z, x, N, m):
    J = np.zeros((1, N+1))
    J[0, :N] = 1.0
    return J

def _qp_ineq(z, x, N, m, q, G):
    # w - q - G@(y-x) >= 0 for all lags
    return z[N] - q - G @ (z[:N] - x)

def _qp_ineq_jac(z, x, N, m, q, G):
    J = np.zeros((G.shape[0], N+1))
    J[:, :N] = -G
    J[:, N] = 1.0
    return J

def mm_step(x, N, m, max_iter=40):
    x_cur = x.copy()
    score_cur = exact_score(x_cur, N)
    best_h = x_cur.copy()
    best_s = score_cur
    for it in range(max_iter):
        q, G = lag_data(x_cur, N)
        if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
            break
        t0 = float(np.max(q))
        z0 = np.concatenate([x_cur, [t0]])
        bounds = [(0.0, 1.0)]*N + [(-1e6, 1e6)]
        cons = [
            {'type': 'eq', 'fun': _qp_eq, 'jac': _qp_eq_jac, 'args': (x_cur, N, m, q, G)},
            {'type': 'ineq', 'fun': _qp_ineq, 'jac': _qp_ineq_jac, 'args': (x_cur, N, m, q, G)},
        ]
        try:
            res = minimize(_qp_obj, z0, jac=_qp_obj_jac, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          args=(x_cur, N, m, q, G),
                          options={'maxiter': 120, 'ftol': 1e-10, 'disp': False})
        except Exception:
            break
        if res.x is None or len(res.x) != N+1 or not np.all(np.isfinite(res.x[:N])):
            break
        y_prop = project_box_sum(res.x[:N], m)
        if not is_feasible(y_prop, N, m):
            break
        # Line search
        alphas = np.array([1, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125, 0.00390625])
        best_candidate = None
        best_cand_score = score_cur
        for a in alphas:
            cand = project_box_sum((1-a)*x_cur + a*y_prop, m)
            if not is_feasible(cand, N, m):
                continue
            sc = exact_score(cand, N)
            if np.isfinite(sc) and sc < best_cand_score:
                best_cand_score = sc
                best_candidate = cand
        if best_candidate is None or not (best_cand_score < score_cur - 1e-12):
            break
        x_cur = best_candidate
        score_cur = best_cand_score
        if score_cur < best_s:
            best_s = score_cur
            best_h = x_cur.copy()
        if np.max(np.abs(x_cur - best_h)) <= 1e-8:
            break
    return best_h, best_s

def fourier_start(N, rng, amp, rmax):
    z = np.full(N, 0.5)
    ns = min(rmax, (N-1)//2)
    idx = np.arange(N)
    for r in range(1, ns+1):
        A = rng.standard_normal()
        B = rng.standard_normal()
        z += (amp/r) * (A*np.sin(2*np.pi*r*idx/N) + B*np.cos(2*np.pi*r*idx/N))
    return z

def run(seed=42, **kwargs):
    N = None
    m = None
    raw = globals().get('initial_h_values', None)
    if raw is not None:
        try:
            raw = np.asarray(raw, dtype=np.float64)
            if raw.ndim == 1 and raw.size > 1 and np.all(np.isfinite(raw)):
                N = raw.size
                m = N / 2.0
        except Exception:
            pass
    if N is None:
        N = kwargs.get('n_points', 70)
        if not isinstance(N, (int, np.integer)) or isinstance(N, bool) or N < 1 or N > 999:
            N = 70
        m = N / 2.0
    if N == 1:
        h = np.array([0.5])
        return h, 0.5, 1

    # Baseline
    if raw is not None and raw.ndim == 1 and raw.size == N and np.all(np.isfinite(raw)):
        baseline = project_box_sum(raw, m)
    else:
        baseline = np.full(N, 0.5)
    base_score = exact_score(baseline, N)
    global_best = baseline.copy()
    global_best_score = base_score

    if N > 256:
        return global_best, global_best_score, int(N)

    rng = np.random.default_rng(seed)

    # Determine tier parameters
    if N <= 96:
        n_starts, mm_iters, n_restarts = 8, 40, 2
        restart_sigmas = [0.05, 0.015]
        restart_iters = [24, 24]
    elif N <= 160:
        n_starts, mm_iters, n_restarts = 5, 25, 1
        restart_sigmas = [0.04]
        restart_iters = [16]
    else:
        n_starts, mm_iters, n_restarts = 3, 12, 0
        restart_sigmas = []
        restart_iters = []

    # Build starts
    starts = [global_best.copy()]
    comp = project_box_sum(1.0 - global_best, m)
    starts.append(comp)
    starts.append(project_box_sum(global_best[::-1], m))
    shift = max(1, N//3)
    starts.append(project_box_sum(np.roll(global_best, shift), m))
    starts.append(project_box_sum(global_best + 0.05*rng.standard_normal(N), m))
    starts.append(project_box_sum(global_best + 0.15*rng.standard_normal(N), m))
    starts.append(project_box_sum(0.5 + 0.5*rng.uniform(size=N), m))
    starts.append(project_box_sum(fourier_start(N, rng, 0.12, 6), m))

    # Deduplicate
    unique_starts = []
    for s in starts:
        dup = False
        for u in unique_starts:
            if np.max(np.abs(s - u)) <= 1e-10:
                dup = True
                break
        if not dup and len(unique_starts) < n_starts:
            unique_starts.append(s)
    while len(unique_starts) < n_starts:
        sig = rng.uniform(0.03, 0.25)
        alt = project_box_sum(global_best + sig*rng.standard_normal(N), m)
        dup = False
        for u in unique_starts:
            if np.max(np.abs(alt - u)) <= 1e-10:
                dup = True
                break
        if dup:
            alt = project_box_sum(0.5 + 0.5*rng.uniform(size=N), m)
        unique_starts.append(alt)

    # Run MM on each start
    for h0 in unique_starts:
        s0 = exact_score(h0, N)
        if np.isfinite(s0) and s0 < global_best_score - 1e-12:
            global_best = h0.copy()
            global_best_score = s0
        try:
            h_mm, s_mm = mm_step(h0, N, m, mm_iters)
            if np.isfinite(s_mm) and s_mm < global_best_score - 1e-12:
                global_best = h_mm.copy()
                global_best_score = s_mm
        except Exception:
            pass

    # Perturbation restarts
    for i in range(n_restarts):
        sig = restart_sigmas[i]
        hop = project_box_sum(global_best + sig*rng.standard_normal(N), m)
        sh = exact_score(hop, N)
        if np.isfinite(sh) and sh < global_best_score - 1e-12:
            global_best = hop.copy()
            global_best_score = sh
        try:
            h_mm, s_mm = mm_step(hop, N, m, restart_iters[i])
            if np.isfinite(s_mm) and s_mm < global_best_score - 1e-12:
                global_best = h_mm.copy()
                global_best_score = s_mm
        except Exception:
            pass

    # Final validation
    global_best = project_box_sum(global_best, m)
    final_score = exact_score(global_best, N)
    if not np.isfinite(final_score) or final_score > global_best_score + 1e-8:
        global_best = project_box_sum(baseline, m)
        final_score = exact_score(global_best, N)
    if not np.isfinite(final_score):
        global_best = np.full(N, 0.5)
        final_score = exact_score(global_best, N)
    return global_best, float(final_score), int(N)

import numpy as np
from scipy.optimize import minimize

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)) or np.abs(z).max() > 1e12:
        return np.full(N, 0.5)
    zc = z - np.median(z)
    lo, hi = zc.min() - 1.0, zc.max()
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(zc - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(zc - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    for _ in range(2):
        if r > 1e-14:
            for i in range(N):
                a = min(r, 1.0 - h[i])
                h[i] += a
                r -= a
                if r <= 1e-14:
                    break
        elif r < -1e-14:
            for i in range(N):
                a = min(-r, h[i])
                h[i] -= a
                r += a
                if r >= -1e-14:
                    break
        else:
            break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1 + 1e-10:
        return np.full(N, 0.5)
    if abs(h.sum() - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return h

def is_feasible(h, N, m):
    return (h.shape == (N,) and np.all(np.isfinite(h)) and h.min() >= -1e-10
            and h.max() <= 1 + 1e-10 and abs(h.sum() - m) <= 1e-10 * max(1, N))

def exact_score(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    corr = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(corr)):
        return np.inf
    return float(np.max(corr) * 2.0 / N)

def lag_data(h, N):
    L = 2 * N - 1
    q = np.zeros(L)
    G = np.zeros((L, N))
    for s in range(-(N - 1), N):
        r = s + N - 1
        if s >= 0:
            ell = N - s
            q[r] = np.dot(h[:ell], 1.0 - h[s:s + ell])
            G[r, :ell] = 1.0 - h[s:s + ell]
            G[r, s:s + ell] -= h[:ell]
        else:
            d = -s
            ell = N - d
            q[r] = np.dot(h[d:], 1.0 - h[:ell])
            G[r, d:] = 1.0 - h[:ell]
            G[r, :ell] -= h[d:]
    return q, G

def _qp_fun(z, x, N, m, q, G):
    y = z[:N]
    return np.dot(y - x, y - x) + z[N]

def _qp_fun_jac(z, x, N, m, q, G):
    g = np.zeros(N + 1)
    g[:N] = 2.0 * (z[:N] - x)
    g[N] = 1.0
    return g

def _qp_eq(z, x, N, m, q, G):
    return np.sum(z[:N]) - m

def _qp_eq_jac(z, x, N, m, q, G):
    J = np.zeros((1, N + 1))
    J[0, :N] = 1.0
    return J

def _qp_ineq(z, x, N, m, q, G):
    return z[N] - q - G @ (z[:N] - x)

def _qp_ineq_jac(z, x, N, m, q, G):
    L = G.shape[0]
    J = np.zeros((L, N + 1))
    J[:, :N] = -G
    J[:, N] = 1.0
    return J

def mm_step(x, N, m, max_iter):
    x_cur = x.copy()
    score_cur = exact_score(x_cur, N)
    best_h = x_cur.copy()
    best_s = score_cur
    for _ in range(max_iter):
        q, G = lag_data(x_cur, N)
        if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
            break
        t0 = float(np.max(q))
        z0 = np.concatenate([x_cur, [t0]])
        bounds = [(0.0, 1.0)] * N + [(0.0, 2.0 * float(N))]
        cons = [
            {'type': 'eq', 'fun': _qp_eq, 'jac': _qp_eq_jac, 'args': (x_cur, N, m, q, G)},
            {'type': 'ineq', 'fun': _qp_ineq, 'jac': _qp_ineq_jac, 'args': (x_cur, N, m, q, G)},
        ]
        try:
            res = minimize(_qp_fun, z0, jac=_qp_fun_jac, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           args=(x_cur, N, m, q, G),
                           options={'maxiter': 120, 'ftol': 1e-10, 'disp': False})
        except Exception:
            break
        if res.x is None or len(res.x) != N + 1 or not np.all(np.isfinite(res.x[:N])):
            break
        y_prop = project_box_sum(res.x[:N], m)
        if not is_feasible(y_prop, N, m):
            break
        alphas = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125, 0.00390625])
        best_cand = None
        best_sc = score_cur
        for a in alphas:
            cand = project_box_sum((1.0 - a) * x_cur + a * y_prop, m)
            if not is_feasible(cand, N, m):
                continue
            sc = exact_score(cand, N)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_cand = cand
        if best_cand is None or not (best_sc < score_cur - 1e-12):
            break
        x_cur = best_cand
        score_cur = best_sc
        if score_cur < best_s:
            best_s = score_cur
            best_h = x_cur.copy()
    return best_h, best_s

def fourier_start(N, rng, amp, rmax):
    z = np.full(N, 0.5)
    ns = min(rmax, (N - 1) // 2)
    idx = np.arange(N)
    for r in range(1, ns + 1):
        A = rng.standard_normal()
        B = rng.standard_normal()
        z += (amp / r) * (A * np.sin(2.0 * np.pi * r * idx / N) + B * np.cos(2.0 * np.pi * r * idx / N))
    return z

def dedup_start(new, existing, tol=1e-10):
    for u in existing:
        if np.max(np.abs(new - u)) <= tol:
            return False
    return True

def run(seed=42, **kwargs):
    N = None
    m = None
    raw = globals().get('initial_h_values', None)
    if raw is not None:
        try:
            raw = np.asarray(raw, dtype=np.float64)

        except Exception:
            pass
    if N is None:

            N = 70
        m = N / 2.0
    if N == 1:
        h = np.array([0.5])
        return h, 0.5, 1

    if raw is not None and raw.ndim == 1 and raw.size == N and np.all(np.isfinite(raw)):
        baseline = project_box_sum(raw, m)
    else:
        baseline = np.full(N, 0.5)
    base_score = exact_score(baseline, N)

    global_best_score = base_score

    if N > 256:

    rng = np.random.default_rng(seed)

    if N <= 96:

    elif N <= 160:

        r_sigs = [0.04]
        r_iters = [16]
    else:

        r_sigs = []
        r_iters = []

    starts = [global_best.copy()]
    starts.append(project_box_sum(1.0 - global_best, m))
    starts.append(project_box_sum(global_best[::-1], m))
    sh = max(1, N // 3)

    starts.append(project_box_sum(global_best + 0.05 * rng.standard_normal(N), m))
    starts.append(project_box_sum(global_best + 0.15 * rng.standard_normal(N), m))

    starts.append(project_box_sum(fourier_start(N, rng, 0.12, 6), m))

    unique_starts = []

        extra += 1

            if np.isfinite(sh) and sh < global_best_score - 1e-12:

        final_score = exact_score(global_best, N)
    return global_best, float(final_score), int(N)
