                    c = rng.beta(0.5, 0.5, n)
                else:
                    c = best_h + cyc[ci % 3] * rng.standard_normal(n)
                ci += 1
                cp = project_K(c, n, S)
                if feasible(cp, n, S):
                    npop.append(cp)
            while len(npop) < P:
                npop.append(project_K(best_h + 0.1 * rng.standard_normal(n), n, S))
            pop = np.array(npop[:P], dtype=np.float64)
            scores = np.array([exact_score(pop[i], dx) for i in range(P)])
            mu_F, mu_CR, stall = 0.7, 0.8, 0

    for sig in sigmas:
        for _ in range(NPROP):
            if rng.random() < 0.75:
                z = rng.standard_normal(n)
                z -= z.mean()
                z[-1] = -z[:-1].sum()
            else:
                k = int(rng.integers(2, min(8, n) + 1))
                supp = rng.choice(n, size=k, replace=False)
                z = np.zeros(n)
                vals = rng.standard_normal(k)
                vals -= vals.mean()
                vals[-1] = -vals[:-1].sum()
                z[supp] = vals
            mz = np.max(np.abs(z))
            if mz < 1e-15:
                continue
            z /= mz
            trial = project_K(best_h + sig * z, n, S)
            if not feasible(trial, n, S):
                continue
            sc = exact_score(trial, dx)
            if sc < best_s - 1e-12:
                best_s, best_h = sc, trial.copy()

    best_h = project_K(best_h, n, S)
    if not feasible(best_h, n, S):
        best_h, best_s = fb_h.copy(), fb_s
    else:
        fs = exact_score(best_h, dx)
        if fs > fb_s + 1e-10:
            best_h, best_s = fb_h.copy(), fb_s
        else:
            best_s = fs
    best_h = np.clip(best_h, 0.0, 1.0)
    final_s = exact_score(best_h, dx)
    return (best_h.astype(np.float64), float(final_s), int(n))
