def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    S = float(S)
    if v.ndim != 1 or n < 1 or not np.all(np.isfinite(v)) or S < 0 or S > n:
        return None
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):

        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    y = np.clip(v - t, 0.0, 1.0)
    r = S - float(y.sum())
    if r > 1e-14:
        for i in np.argsort(1.0 - y)[::-1]:
            d = min(r, 1.0 - y[i]); y[i] += d; r -= d
            if r <= 0: break
    elif r < -1e-14:

    return (len(h) == n and np.all(np.isfinite(h)) and
            h.min() >= -1e-12 and h.max() <= 1.0 + 1e-12 and
            abs(float(h.sum()) - 0.5 * n) < 1e-10 * max(1, n))

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    v = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)
    return v if np.isfinite(v) else float('inf')

def smooth_F(h, n, tau):
    q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    qm = float(np.max(q))
    if tau < 1e-15:
        return qm
    sh = (q - qm) / tau
    return qm + tau * float(np.log(np.sum(np.exp(sh))))

def smooth_grad(h, n, tau):
    q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    qm = float(np.max(q))
    p = np.exp((q - qm) / tau)

                acc += p[i] * (1.0 - h[k + s])
            if 0 <= k - s < n:
                acc -= p[i] * h[k - s]
        g[k] = dx * acc
    return g

    if not np.isfinite(best_c):
        best_h = project_box_sum(np.full(n, 0.5), S)
        best_c = exact_score(best_h, n)
    starts = build_starts(parent, n, rng)
    M = max(30, min(120, 5400 // n))
    taus = [0.04 * (0.3 ** j) for j in range(7)]
    for h0 in starts:
        h_cur = h0.copy()
        for tau in taus:
            try:
                h_next, sb_h, sb_c = project_stage(h_cur, tau, M)
                h_cur = h_next
                if sb_c < best_c:
                    best_c = sb_c
                    best_h = sb_h.copy()
            except Exception:
                pass
    for sigma in [0.04, 0.12, 0.30]:
        h0 = project_box_sum(best_h + sigma * rng.standard_normal(n), S)
        if h0 is None:
            continue
        h_cur = h0.copy()
        for tau in taus:
            try:
                h_next, sb_h, sb_c = project_stage(h_cur, tau, max(20, M // 2))
                h_cur = h_next
                if sb_c < best_c:
                    best_c = sb_c
                    best_h = sb_h.copy()
            except Exception:
                pass
    pm = polish_nm(best_h, n)
    if pm is not None:
        c = exact_score(pm, n)
        if c < best_c:
            best_c = c
            best_h = pm.copy()
    pm2 = polish_nm(best_h + 0.02 * rng.standard_normal(n), n)
    if pm2 is not None:
        c = exact_score(pm2, n)
        if c < best_c:
            best_c = c
            best_h = pm2.copy()
    best_h = project_box_sum(best_h, S)
    if best_h is None or not check_feasible(best_h, n):
        best_h = parent
        best_c = exact_score(parent, n)
    best_c = exact_score(best_h, n)
    return (best_h, best_c, n)
