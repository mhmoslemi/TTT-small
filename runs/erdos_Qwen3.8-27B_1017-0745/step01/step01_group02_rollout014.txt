            ns = sigmas[restarts % len(sigmas)]
            fill = max(0, P - len(new_pop))
            for k in range(fill):
                if k < int(0.7 * fill):
                    cand = project_box_sum(best_h + ns * rng.standard_normal(n), S)
                else:
                    cand = project_box_sum(rng.random(n), S)
                if cand is None or not check_feasible(cand, n):
                    cand = best_h.copy()
                new_pop.append(cand)
                new_cv.append(exact_score(cand, n))
            sb, cv = new_pop[:P], new_cv[:P]
            archive = []
            mu_F, mu_CR = 0.55, 0.70
            no_imp = 0
            restarts += 1
    sigma = 0.02
    for _ in range(10):
        h0 = project_box_sum(best_h + sigma * rng.standard_normal(n), S)
        if h0 is None:
            break
        st = exact_score(h0, n)
        if np.isfinite(st) and st < best_c5:
            best_h, best_c5 = h0.copy(), st
            sigma = max(0.004, 0.6 * sigma)
        else:
            sigma = min(0.15, 1.5 * sigma)
    best_h = project_box_sum(best_h, S)
    if best_h is None:
        best_h = parent
    bf = exact_score(parent, n)
    sf = exact_score(best_h, n)
    if np.isfinite(bf) and (not np.isfinite(sf) or bf < sf):
        best_h, sf = parent, bf
    if not np.isfinite(sf):
        best_h = np.full(n, 0.5)
        sf = exact_score(best_h, n)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, S)
    if best_h is None:
        best_h = parent
    return (best_h, exact_score(best_h, n), n)
