import numpy as np
from scipy.signal import fftconvolve

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(60):
        mid = 0.5*(lo+hi)
        y = np.clip(v-mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-14:
        if resid > 0:
            idx = np.argsort(1.0 - y)
            for i in idx:
                a = min(resid, 1.0-y[i]); y[i] += a; resid -= a
                if resid <= 1e-15: break
        else:
            idx = np.argsort(y)
            for i in idx:
                s = min(-resid, y[i]); y[i] -= s; resid += s
                if resid >= -1e-15: break
    y = np.clip(y, 0.0, 1.0)
    if abs(np.sum(y)-S) > 1e-10*max(1.0, float(n)):
        return None
    return y

def exact_score(h, n):
    dx = 2.0/n
    c = np.correlate(h, 1.0-h, mode='full')
    return float(np.max(c)*dx)

def feasible(h, n, S):
    return h.shape==(n,) and np.all(np.isfinite(h)) and np.all(h>=-1e-12) and np.all(h<=1.0+1e-12) and abs(np.sum(h)-S) < 5e-12*max(1.0,float(n))

def compute_QG(h, n):
    L = 2*n-1
    Q = np.zeros(L); G = np.zeros((L, n))
    for k in range(-(n-1), n):
        ri = k + n - 1
        i0 = max(0, k); i1 = min(n-1, n-1+k)
        if i0 > i1: continue
        i = np.arange(i0, i1+1); j = i - k
        Q[ri] = np.sum(h[i]*(1.0-h[j]))
        G[ri, i] += (1.0-h[j])
        G[ri, j] += (-h[i])
    return Q, G

def proj_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    sv = np.sort(v)[::-1]
    cssv = np.cumsum(sv)
    rho = np.where(sv > (cssv-1.0)/np.arange(1, n+1)[::-1])[0]
    if len(rho) == 0:
        return np.ones(n)/n
    r = rho[0]
    theta = (cssv[r]-1.0)/(r+1)
    w = np.maximum(v-theta, 0.0)
    r2 = 1.0 - np.sum(w)
    if abs(r2) > 1e-14:
        pos = w > 0
        if r2 > 0 and np.any(pos):
            w[np.argmax(w)] += r2
        elif r2 < 0 and np.any(pos):
            w[pos] -= r2*1.0/pos.sum()
    w = np.maximum(w, 0.0)
    w /= max(w.sum(), 1e-30)
    return w

def fista_solve(x, Q, G, S, warm, n):
    L = 2*n-1
    if warm is not None and len(warm) == L and np.all(np.isfinite(warm)):
        w = proj_simplex(warm)
    else:
        w = np.zeros(L); w[np.argmax(Q)] = 1.0
    Ldual = max(1e-8, 0.5001*float(np.sum(G*G)))
    Imax = max(100, min(320, math.ceil(14000.0/n)))
    gap_tol = 1e-7*max(1.0, float(np.max(np.abs(Q))))
    v = w.copy(); t = 1.0
    best_y = x.copy(); best_model = float('inf')
    for it in range(Imax):
        z = G.T @ v
        y = project_box_sum(x - 0.5*z, S)
        if y is None: y = x.copy()
        d = y - x
        p = G @ d
        vals = Q + p
        model = float(d@d + np.max(vals))
        if model < best_model:
            best_model = model; best_y = y.copy()
        grad = Q + p
        if it % max(5, Imax//25) == 0 or it == Imax-1:
            ws = proj_simplex(w)
            z2 = G.T @ ws
            y2 = project_box_sum(x - 0.5*z2, S)
            if y2 is not None:
                d2 = y2-x; p2 = G@d2; v2 = Q+p2
                m2 = float(d2@d2 + np.max(v2))
                if m2 < best_model: best_model = m2; best_y = y2.copy()
                gap = float(np.max(v2) - np.dot(ws, v2))
                if gap <= gap_tol: break
        w_new = proj_simplex(v + grad/Ldual)
        t_new = 0.5*(1.0 + math.sqrt(1.0 + 4.0*t*t))
        v_new = w_new + ((t-1.0)/t_new)*(w_new - w)
        w = w_new; v = v_new; t = t_new
    return best_y, w

def mm_local(x0, n, S):

            if best_sc_ls - min(best_sc, 0.3) < 1e-10: break
