def compute_g(h, j, N):
    s = np.arange(-(N-1), N)  # length 2N-1
    # First term: valid when 0 <= j+s < N
    valid1 = (j + s >= 0) & (j + s < N)
    term1 = np.zeros(2*N-1)
    term1[valid1] = 1.0 - h[j + s[valid1]]

    term2 = np.zeros(2*N-1)
    term2[valid2] = h[j - s[valid2]]
    return term1 - term2

import numpy as np

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    if r > 0:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
            if r <= 1e-14:
                break
    elif r < 0:
        for i in range(N):
            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1 + 1e-10:
        return np.full(N, 0.5)
    if abs(h.sum() - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)

def exact_score(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / N)

def get_q(h, N):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr  # length 2N-1, index k corresponds to shift s = k-(N-1)

def compute_g(h, j, N):
    s = np.arange(-(N-1), N)
    g = np.zeros(2*N-1)
    v1 = (j + s >= 0) & (j + s < N)
    g[v1] = 1.0 - h[j + s[v1]]
    v2 = (j - s >= 0) & (j - s < N)
    g[v2] -= h[j - s[v2]]
    return g

def pair_line_search(h, q, p, r, N):
    t_lo = max(-h[p], h[r] - 1.0)
    t_hi = min(1.0 - h[p], h[r])
    if t_hi - t_lo < 1e-12:
        return 0.0
    # Compute b
    gp = compute_g(h, p, N)
    gr = compute_g(h, r, N)
    b = gp - gr
    # a is sparse
    a = np.zeros(2*N-1)
    a[N-1] = -2.0
    d = r - p
    a[N-1 + d] = 1.0
    a[N-1 - d] = 1.0
    # Coarse mesh
    mesh = np.linspace(t_lo, t_hi, 9)

        mesh = np.concatenate([[0.0], mesh])
    elif mesh[-1] < 0:
        mesh = np.concatenate([mesh, [0.0]])
    else:
        mesh = np.concatenate([mesh, [0.0]])
    mesh = np.sort(mesh)
    mesh = mesh[np.concatenate([[True], np.diff(mesh) > 1e-14])]
    
    def eval_t(t):
        return np.max(q + b*t + a*t*t)
    
    best_t = 0.0
    best_val = eval_t(0.0)
    for t in mesh:
        v = eval_t(t)
        if v < best_val - 1e-15:
            best_val = v
            best_t = t
        elif abs(v - best_val) <= 1e-15 and abs(t) < abs(best_t):
            best_t = t

    hi_r = t_hi

            hi_r = mesh[i+1]

def pair_descent(h, N, m, rng, n_sweeps=2):
    h = h.copy()
    q = get_q(h, N)
    cur_score = float(np.max(q) * 2.0 / N)
    for sweep in range(n_sweeps):
        pairs = [(p, r) for p in range(N) for r in range(p+1, N)]
        rng.shuffle(pairs)
        improved = False
        for p, r in pairs:
            t = pair_line_search(h, q, p, r, N)
            if abs(t) > 1e-12:
                pt = h[p] + h[r]
                hp_new = np.clip(h[p] + t, max(0.0, pt - 1.0), min(1.0, pt))
                hr_new = pt - hp_new
                h_new = h.copy()
                h_new[p] = hp_new
                h_new[r] = hr_new
                if np.all(np.isfinite(h_new)) and h_new.min() >= -1e-12 and h_new.max() <= 1+1e-12 and abs(h_new.sum() - m) < 1e-8:
                    h_new = np.clip(h_new, 0.0, 1.0)
                    new_score = exact_score(h_new, N)
                    if new_score < cur_score - 1e-12:
                        h = h_new
                        q = get_q(h, N)
                        cur_score = new_score
                        improved = True
        if not improved:
            break
    return h, cur_score

def sim_anneal(h, N, m, rng, n_proposals=160):
    h_cur = h.copy()
    best = h.copy()
    cur_score = exact_score(h_cur, N)
    best_score = cur_score
    for j in range(n_proposals):
        f = j / max(1, n_proposals - 1)
        sigma = 0.12 * (0.025 / 0.12) ** f
        T = 0.002 * (0.0001 / 0.002) ** f
        p = rng.integers(N)
        r = rng.integers(N)
        while r == p:
            r = rng.integers(N)
        t_lo = max(-h_cur[p], h_cur[r] - 1.0)
        t_hi = min(1.0 - h_cur[p], h_cur[r])
        s_lo = max(t_lo, -sigma)
        s_hi = min(t_hi, sigma)
        if s_hi - s_lo < 1e-12:
            continue
        t = rng.uniform(s_lo, s_hi)
        pt = h_cur[p] + h_cur[r]
        hp_new = np.clip(h_cur[p] + t, max(0.0, pt - 1.0), min(1.0, pt))
        hr_new = pt - hp_new
        h_new = h_cur.copy()
        h_new[p] = hp_new
        h_new[r] = hr_new
        if not np.all(np.isfinite(h_new)) or h_new.min() < -1e-12 or h_new.max() > 1+1e-12:
            continue
        h_new = np.clip(h_new, 0.0, 1.0)
        new_score = exact_score(h_new, N)
        if not np.isfinite(new_score):
            continue
        delta = new_score - cur_score
        if delta <= 0 or (T > 0 and rng.uniform() < np.exp(-delta / T) if delta / T < 700 else False):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best = h_new.copy()
                best_score = new_score
    return best, best_score

    if c5b < c5 - 1e-12:
        best, c5 = baseline, c5b
    if not np.isfinite(c5):
        best = np.full(N, 0.5)
        c5 = exact_score(best, N)
    return best, c5, int(N)
