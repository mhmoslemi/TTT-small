def line_minimax(x, y, n):
    d = y - x
    n_lags = 2*n - 1
    A = np.zeros(n_lags)
    B = np.zeros(n_lags)
    D = np.zeros(n_lags)
    for k in range(-(n-1), n):
        ki = k + n - 1
        i_lo, i_hi = max(0,k), min(n-1, n-1+k)
        i = np.arange(i_lo, i_hi+1)
        j = i - k
        A[ki] = -np.sum(d[i]*d[j])
        B[ki] = np.sum(d[i]*(1-x[j]) - x[i]*d[j])
        D[ki] = np.sum(x[i]*(1-x[j]))
    # Collect candidate alphas
    alphas = {0.0, 1.0}
    # Stationary points
    for ki in range(n_lags):
        if abs(A[ki]) > 1e-14:
            a = -B[ki] / (2*A[ki])
            if -1e-10 <= a <= 1+1e-10:
                alphas.add(max(0.0, min(1.0, a)))
    # Pairwise intersections
    for ki in range(n_lags):
        for lj in range(ki+1, n_lags):
            da = A[ki] - A[lj]
            db = B[ki] - B[lj]
            dc = D[ki] - D[lj]
            if abs(dc) < 1e-15 and abs(db) < 1e-15 and abs(da) < 1e-15:
                continue
            if abs(da) < 1e-14:
                if abs(db) > 1e-14:
                    a = -dc / db
                    if -1e-10 <= a <= 1+1e-10:
                        alphas.add(max(0.0, min(1.0, a)))
            else:
                disc = db*db - 4*da*dc
                if disc >= -1e-14:
                    disc = max(0.0, disc)
                    sd = np.sqrt(disc)
                    for a in [(-db+sd)/(2*da), (-db-sd)/(2*da)]:
                        if -1e-10 <= a <= 1+1e-10:
                            alphas.add(max(0.0, min(1.0, a)))
    alphas = sorted(alphas)
    # Evaluate
    best_a, best_q = 0.0, np.max(D)
    all_a = np.array(alphas)
    # Evaluate in chunks
    for s in range(0, len(all_a), 2048):
        chunk = all_a[s:s+2048]
        Q = np.outer(chunk, A) * chunk + np.outer(chunk, B) + D  # shape (len(chunk), n_lags)
        # Actually: Q[ai, ki] = A[ki]*ai^2 + B[ki]*ai + D[ki]
        Q = A[None,:] * chunk[:,None]**2 + B[None,:] * chunk[:,None] + D[None,:]
        mx = np.max(Q, axis=1)
        for idx in range(len(chunk)):
            if mx[idx] < best_q - 1e-15:
                best_q = mx[idx]
                best_a = chunk[idx]
    return best_a, best_q

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v))
    for _ in range(80):
        mid = 0.5*(lo+hi)
        if np.sum(np.clip(v-mid, 0, 1)) > S: lo = mid
        else: hi = mid
    y = np.clip(v - 0.5*(lo+hi), 0, 1)
    r = S - np.sum(y)
    if abs(r) > 1e-14:
        ord_ = np.argsort(1-y) if r > 0 else np.argsort(y)
        for i in ord_:
            if r > 0:
                a = min(r, 1-y[i]); y[i] += a; r -= a
            else:
                a = min(-r, y[i]); y[i] -= a; r += a
            if abs(r) < 1e-15: break
    y = np.clip(y, 0, 1)
    if not np.all(np.isfinite(y)): return None
    return y

def exact_score(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def feasible(h, n):
    return (h is not None and h.shape==(n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1+1e-12)
            and abs(np.sum(h)-n/2.0) < 1e-9*max(1.0,n))

def pidx(i, j, n):
    if i > j: i, j = j, i
    return i*n - i*(i-1)//2 + (j-i)

def solve_lp(x, r, n, S):
    P = n*(n+1)//2
    nv = n + P + 1
    lv = np.maximum(0.0, x-r); uv = np.minimum(1.0, x+r)
    R, C, V, b = [], [], [], []
    for i in range(n):
        for j in range(i, n):
            pi = n + pidx(i,j,n)
            a, b_ = lv[i], uv[i]; c, d = lv[j], uv[j]
            for rhs, zc, jc, ic in [(a*c,-1.0,a,c),(b_*d,-1.0,b_,d),
                                     (-c*b_,1.0,-b_,-c),(-a*d,1.0,-a,-d)]:
                row = len(b); b.append(rhs)
                R.append(row); C.append(pi); V.append(zc)
                R.append(row); C.append(j); V.append(jc)
                R.append(row); C.append(i); V.append(ic)
    for k in range(-(n-1), n):
        row = len(b); b.append(0.0)
        for i in range(max(0,k), min(n-1,n-1+k)+1):
            j = i-k
            R.append(row); C.append(i); V.append(1.0)
            R.append(row); C.append(n+pidx(i,j,n)); V.append(-1.0)
        R.append(row); C.append(n+P); V.append(-1.0)
    m = len(b)
    A_ub = coo_matrix((V,(R,C)),shape=(m,nv)).tocsr()
    A_eq = coo_matrix((np.ones(n),(np.arange(n),np.arange(n))),shape=(1,nv)).tocsr()
    c_obj = np.zeros(nv); c_obj[-1] = 1.0
    lb = np.concatenate([lv, np.zeros(P), [0.0]])
    ub = np.concatenate([uv, np.ones(P), [float(n)]])
    for i in range(n):
        for j in range(i,n):
            pi = n+pidx(i,j,n)
            lb[pi] = lv[i]*lv[j]; ub[pi] = uv[i]*uv[j]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=np.array(b), A_eq=A_eq,
                      b_eq=np.array([S]), bounds=list(zip(lb,ub)),
                      method='highs', options={'time_limit':60})
        if res.status==0 and len(res.x)==nv and np.all(np.isfinite(res.x)):
            return res.x[:n]
    except: pass
    return None

def line_minimax(x, y, n):
    d = y-x; nl = 2*n-1
    A = np.zeros(nl); B = np.zeros(nl); D = np.zeros(nl)
    for k in range(-(n-1), n):
        ki = k+n-1
        i = np.arange(max(0,k), min(n-1,n-1+k)+1)
        j = i-k
        A[ki] = -np.sum(d[i]*d[j])
        B[ki] = np.sum(d[i]*(1-x[j]) - x[i]*d[j])
        D[ki] = np.sum(x[i]*(1-x[j]))
    alphas = [0.0, 1.0]
    for ki in range(nl):
        if abs(A[ki]) > 1e-14:
            a = -B[ki]/(2*A[ki])
            if -1e-10 <= a <= 1+1e-10: alphas.append(max(0,min(1,a)))
    for ki in range(nl):
        for lj in range(ki+1, nl):
            da, db, dc = A[ki]-A[lj], B[ki]-B[lj], D[ki]-D[lj]
            if abs(da)<1e-14:
                if abs(db)>1e-14:
                    a = -dc/db
                    if -1e-10<=a<=1+1e-10: alphas.append(max(0,min(1,a)))
            else:
                disc = db*db-4*da*dc
                if disc >= -1e-12:
                    sd = np.sqrt(max(0.0,disc))
                    for a in [(-db+sd)/(2*da),(-db-sd)/(2*da)]:
                        if -1e-10<=a<=1+1e-10: alphas.append(max(0,min(1,a)))
    # Deduplicate
    alphas = sorted(set(round(a,10) for a in alphas))
    best_a, best_q = 0.0, np.max(D)
    for a in alphas:
        q = np.max(A*a*a + B*a + D)
        if q < best_q - 1e-15:
            best_q = q; best_a = a
    return best_a
