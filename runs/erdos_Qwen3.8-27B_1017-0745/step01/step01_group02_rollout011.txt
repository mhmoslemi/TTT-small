import numpy as np
import math

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(v - tau, 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-14:
        for i in np.argsort(1.0 - y)[::-1]:
            d = min(r, 1.0 - y[i])
            y[i] += d
            r -= d
            if r <= 1e-14:
                break
    elif r < -1e-14:
        for i in np.argsort(y)[::-1]:
            d = min(-r, y[i])
            y[i] -= d
            r += d
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-10) or np.any(y > 1.0 + 1e-10):
        return None
    if abs(float(np.sum(y)) - S) > 1e-10 * max(1, n):
        return None
    np.clip(y, 0.0, 1.0, out=y)
    return y

def check_feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(float(np.sum(h)) - 0.5 * n) < 1e-10 * max(1, n))

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    val = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
    return val if np.isfinite(val) else float('inf')

def full_conv(a, b):
    try:
        from scipy.signal import fftconvolve
        r = fftconvolve(a, b, mode='full')
        if np.all(np.isfinite(r)):
            return r
    except Exception:
        pass
    return np.convolve(a, b, mode='full')

def smooth_value_grad(h, n, tau):
    if tau <= 0 or not np.all(np.isfinite(h)):
        return float('inf'), None
    b = 1.0 - h
    r = full_conv(h[::-1].copy(), b)
    if len(r) != 3 * n - 2:
        return float('inf'), None
    q = (2.0 / n) * r
    qmax = float(np.max(q))
    if not np.isfinite(qmax):
        return float('inf'), None
    ex = np.exp((q - qmax) / tau)
    s = float(np.sum(ex))
    if s <= 0 or not np.isfinite(s):
        return float('inf'), None
    F = qmax + tau * math.log(s)
    p = ex / s
    p_rev = p[::-1]
    A = full_conv(p_rev, b)[n - 1:2 * n - 1]
    B = full_conv(p, h)[n - 1:2 * n - 1]
    g = (2.0 / n) * (A - B)
    if not np.all(np.isfinite(g)):
        return float('inf'), None
    return F, g
