def smooth_obj_grad(h, tau, N):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    cmax = c.max()
    exp_c = np.exp((c - cmax) / tau)
    p = exp_c / exp_c.sum()
    F = cmax + tau * np.log(exp_c.sum())
    
    grad = np.zeros(N)
    for s in range(-(N-1), N):
        r = N - 1 + s  # index in c/p
        ps = p[r]
        if ps < 1e-15:
            continue
        if s == 0:
            grad += dx * ps * (1.0 - 2.0 * h)
        elif s > 0:
            d = s
            grad[:N-d] += dx * ps * (1.0 - h[d:])
            grad[d:] -= dx * ps * h[:N-d]
        else:
            d = -s
            grad[d:] += dx * ps * (1.0 - h[:N-d])
            grad[:N-d] -= dx * ps * h[d:]
    return F, grad, c

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    if r > 1e-14:
        for i in range(N):
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
            if r <= 1e-14:
                break
    elif r < -1e-14:
        for i in range(N):
            a = min(-r, h[i])
            h[i] -= a
            r += a
            if r >= -1e-14:
                break
    if h.min() < -1e-10 or h.max() > 1.0 + 1e-10:
        return np.full(N, 0.5)
    if abs(h.sum() - m) > 1e-7:
        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)

    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / N)

def smooth_obj_grad(h, tau, N):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    cmax = c.max()
    ec = np.exp((c - cmax) / tau)
    sec = ec.sum()
    p = ec / sec
    F = cmax + tau * np.log(sec)
    grad = np.zeros(N)
    for s in range(-(N - 1), N):
        r = N - 1 + s
        ps = p[r]
        if ps < 1e-10:
            continue
        w = dx * ps
        if s == 0:
            grad += w * (1.0 - 2.0 * h)
        elif s > 0:
            d = s
            grad[:N - d] += w * (1.0 - h[d:])
            grad[d:] -= w * h[:N - d]
        else:
            d = -s
            grad[d:] += w * (1.0 - h[:N - d])
            grad[:N - d] -= w * h[d:]
    return F, grad

            accepted = False

                    continue

                    break

                    best_score = sy

        tau_sched = [0.05, 0.015, 0.005, 0.0015, 0.0005, 0.00015, 0.00005]
        max_iter = 50
        for h0 in unique_starts:
            bh, bs = optimize_start(h0, N, m, tau_sched, max_iter)
            if bs < best_score:
                best_h = bh.copy()
                best_score = bs
        for sc in [0.04, 0.015]:
            hop = project_box_sum(best_h + sc * rng.standard_normal(N), m)
            bh, bs = optimize_start(hop, N, m, tau_sched, max_iter)
            if bs < best_score:
                best_h = bh.copy()
                best_score = bs
    best_h = project_box_sum(best_h, m)
    c5 = exact_score(best_h, N)
    if not np.isfinite(c5) or c5 > baseline_score:
        best_h = baseline.copy()
        c5 = baseline_score
    return (best_h, c5, int(N))
