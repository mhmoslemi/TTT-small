import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)):
        return None
    n = len(z)
    if S < 0 or S > n or np.max(np.abs(z)) > 1e12:
        return None
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    y = np.clip(z - lam, 0.0, 1.0)
    r = S - np.sum(y)
    if r > 0:
        for i in range(n):
            if r <= 0: break
            a = min(r, 1.0 - y[i])
            y[i] += a
            r -= a
    elif r < 0:
        for i in range(n):
            if r >= 0: break
            a = min(-r, y[i])
            y[i] -= a
            r += a
    if abs(np.sum(y) - S) > 1e-10 or not np.all(np.isfinite(y)) or np.any(y < -1e-15) or np.any(y > 1 + 1e-15):
        return None
    return np.clip(y, 0.0, 1.0)

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-15) and np.all(h <= 1 + 1e-15)
            and abs(np.sum(h) - n / 2.0) <= 1e-9)

def exact_lags(h, n):
    if not is_feasible(h, n):
        return None
    vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if len(vals) != 2 * n - 1 or not np.all(np.isfinite(vals)):
        return None
    return vals

def exact_score(h, n):
    q = exact_lags(h, n)
    if q is None:
        return None
    return float(np.max(q))

def build_lag_model(h, n):
    q = exact_lags(h, n)
    if q is None:
        return None, None
    dx = 2.0 / n
    m = 2 * n - 1
    rows, cols, data = [], [], []
    for row in range(m):
        d = row - (n - 1)
        lo = max(0, -d)
        hi = min(n, n - d)
        for i in range(lo, hi):
            rows.append(row); cols.append(i); data.append(dx * (1.0 - h[i + d]))
            rows.append(row); cols.append(i + d); data.append(-dx * h[i])
    G = sp.coo_matrix((data, (rows, cols)), shape=(m, n)).tocsr()
    if not np.all(np.isfinite(G.data)):
        return None, None
    return q, G

def solve_lp_step(x, q, G, rho, n, mu):
    m = 2 * n - 1
    A1 = sp.sparse.hstack([G, sp.csc_matrix((m, n)), -sp.csc_matrix(np.eye(m, 1))])
    A2 = sp.sparse.hstack([sp.eye(n, format='csc'), -sp.eye(n, format='csc'), sp.csc_matrix((n, 1))])
    A3 = sp.sparse.hstack([-sp.eye(n, format='csc'), -sp.eye(n, format='csc'), sp.csc_matrix((n, 1))])
    A_ub = sp.vstack([A1, A2, A3]).tocsr()
    b_ub = np.concatenate([-q, np.zeros(n), np.zeros(n)])
    A_eq = sp.sparse.hstack([sp.csc_matrix(np.ones((1, n))), sp.csc_matrix((1, n + 1))]).tocsr()
    b_eq = np.zeros(1)
    c = np.concatenate([np.zeros(n), mu * np.ones(n), np.ones(1)])
    lbs = np.maximum(-rho, -x)
    ubs = np.minimum(rho, 1.0 - x)
    bounds = list(zip(lbs, ubs)) + [(0.0, None)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'maxiter': 300, 'presolve': True})
    except Exception:
        return None
    if res is None or not res.success:
        return None
    delta = res.x[:n]
    if not np.all(np.isfinite(delta)) or abs(np.sum(delta)) > 1e-5:
        return None
    return delta

def optimize_trajectory(h0, n, rng, best_h, best_score):
    S = n / 2.0
    dx = 2.0 / n
    mu = 0.002 * dx
    x = h0.copy()
    f = exact_score(x, n)
    if f is None:
        return best_h, best_score
    rho = 0.08
    rho_max, rho_min = 0.25, 2e-4
    fails = 0
    max_iter = 30 if n <= 128 else (22 if n <= 256 else (14 if n <= 512 else 6))
    for _ in range(max_iter):
        if rho < rho_min or fails >= 4:
            break
        q, G = build_lag_model(x, n)
        if G is None:
            fails += 1
            rho *= 0.5
            continue
        delta = solve_lp_step(x, q, G, rho, n, mu)
        if delta is None or np.max(np.abs(delta)) < 1e-9:
            fails += 1
            rho *= 0.5
            continue
        best_beta = None
        best_cand = None
        best_f_new = f
        for j in range(9):
            beta = 2.0 ** (-j)
            cand = project_capped_simplex(x + beta * delta, S)
            if cand is None:
                continue
            fc = exact_score(cand, n)
            if fc is None:
                continue
            if fc < best_score - 1e-12:
                best_h = cand.copy()
                best_score = fc
            if fc < best_f_new - 1e-12:
                best_f_new = fc
                best_cand = cand
                best_beta = beta
        if best_cand is None:
            fails += 1
            rho *= 0.5
            continue
        s = best_cand - x
        q_new = exact_lags(best_cand, n)
        pred = f - float(np.max(q + G @ s)) if q_new is not None else 0.0
        act = f - best_f_new
        if pred <= 1e-12 or act / max(pred, 1e-12) < 0.25:
            rho = max(rho_min, rho * 0.5)
        elif act / max(pred, 1e-12) > 0.75 and np.max(np.abs(s)) >= 0.8 * rho:
            rho = min(rho_max, rho * 1.5)
        x = best_cand
        f = best_f_new
        fails = 0
        if np.max(np.abs(s)) < 1e-8:
            break
    if f < best_score - 1e-12:
        best_h = x.copy()
        best_score = f
    return best_h, best_score

def build_starts(n, S, parent, rng):
    starts = []
    const = np.full(n, 0.5)
    if parent is not None:
        starts.append(parent.copy())
        for sig in [0.04, 0.12]:
            p = project_capped_simplex(parent + sig * rng.normal(0.0, 1.0, n), S)
            if p is not None:
                starts.append(p)
        starts.append(parent[::-1].copy())
        p_alt = project_capped_simplex(1.0 - parent, S)
        if p_alt is not None:

        u += a * np.cos(2.0 * np.pi * m * idx / n + phi)
    p = project_capped_simplex(u, S)
    if p is not None:
        starts.append(p)
    perm = rng.permutation(n)
    binv = np.zeros(n)
    binv[perm[:n // 2]] = 1.0
    p = project_capped_simplex(binv, S)
    if p is not None:
        starts.append(p)
    deduped = []
    for s in starts:
        if s is None or not is_feasible(s, n):
            continue
        dup = False
        for d in deduped:
            if np.max(np.abs(s - d)) < 1e-9:
                dup = True
                break
        if not dup:
            deduped.append(s)
    limit = 6 if n <= 128 else (4 if n <= 256 else (3 if n <= 512 else 2))
    return deduped[:limit]

def run(seed=42, **kwargs):
    if seed is None:
        seed = 42
    try:
        seed = int(seed)
    except (TypeError, ValueError):
        seed = 42
    rng = np.random.default_rng(seed)
    n = None
    parent = None
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        try:
            iv_arr = np.asarray(iv, dtype=np.float64)
            if iv_arr.ndim == 1 and 8 <= iv_arr.size <= 999 and np.all(np.isfinite(iv_arr)) and np.max(np.abs(iv_arr)) <= 1e12:
                n = iv_arr.size
                S = n / 2.0
                if is_feasible(iv_arr, n):
                    parent = iv_arr.copy()
                else:
                    pr = project_capped_simplex(iv_arr, S)
                    if pr is not None and is_feasible(pr, n):
                        parent = pr
        except Exception:
            pass
    if n is None:
        n = kwargs.get('n_points', 80)
        try:
            n = int(n)
        except (TypeError, ValueError):
            n = 80
        if n < 8 or n > 999:
            n = 80
    S = n / 2.0
    h_const = np.full(n, 0.5)
    best_h = h_const.copy()
    best_score = exact_score(h_const, n)
    if best_score is None:
        best_score = 0.5
    if parent is not None:
        ps = exact_score(parent, n)
        if ps is not None and ps < best_score:
            best_score = ps
            best_h = parent.copy()
    starts = build_starts(n, S, parent, rng)
    for s in starts:
        sc = exact_score(s, n)
        if sc is not None and sc < best_score:
            best_score = sc
            best_h = s.copy()
    for s in starts:
        best_h, best_score = optimize_trajectory(s, n, rng, best_h, best_score)
    # restarts
    n_restart = 2 if n <= 128 else (1 if n <= 256 else 0)
    sigma = 0.05
    for _ in range(n_restart):
        pert = project_capped_simplex(best_h + sigma * rng.normal(0.0, 1.0, n), S)
        if pert is not None and is_feasible(pert, n):
            old_score = best_score
            best_h, best_score = optimize_trajectory(pert, n, rng, best_h, best_score)
            if best_score < old_score - 1e-12:
                sigma = max(0.02, 0.5 * sigma)
            else:
                sigma = min(0.18, 1.8 * sigma)
    # final validation
    if not is_feasible(best_h, n):
        rep = project_capped_simplex(best_h, S)
        if rep is not None and is_feasible(rep, n):
            best_h = rep
    final_score = exact_score(best_h, n)
    if final_score is None:
        if parent is not None and is_feasible(parent, n):
            best_h = parent.copy()
            final_score = exact_score(parent, n)
        if final_score is None:
            best_h = h_const.copy()
            final_score = 0.5
    return best_h, float(final_score), int(n)

import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)):
        return None
    n = len(z)
    if S < 0 or S > n or np.max(np.abs(z)) > 1e12:
        return None
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    y = np.clip(z - lam, 0.0, 1.0)
    r = S - np.sum(y)
    if r > 0:
        for i in range(n):
            if r <= 0:
                break
            a = min(r, 1.0 - y[i])
            y[i] += a
            r -= a
    elif r < 0:
        for i in range(n):
            if r >= 0:
                break
            a = min(-r, y[i])
            y[i] -= a
            r += a
    if abs(np.sum(y) - S) > 1e-10 or not np.all(np.isfinite(y)):
        return None
    return np.clip(y, 0.0, 1.0)

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-15) and np.all(h <= 1.0 + 1e-15)
            and abs(np.sum(h) - n / 2.0) <= 1e-9)

def exact_lags(h, n):
    if not is_feasible(h, n):
        return None
    vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if len(vals) != 2 * n - 1 or not np.all(np.isfinite(vals)):
        return None
    return vals

def exact_score(h, n):
    q = exact_lags(h, n)
    if q is None:
        return None
    return float(np.max(q))

def build_lag_model(h, n):
    q = exact_lags(h, n)
    if q is None:
        return None, None
    dx = 2.0 / n
    m = 2 * n - 1
    rows, cols, data = [], [], []
    for row in range(m):
        d = row - (n - 1)
        lo = max(0, -d)
        hi = min(n, n - d)
        for i in range(lo, hi):
            rows.append(row)
            cols.append(i)
            data.append(-dx * h[i + d])
            rows.append(row)
            cols.append(i + d)
            data.append(dx * (1.0 - h[i]))
    G = sp.coo_matrix((data, (rows, cols)), shape=(m, n)).tocsr()
    if not np.all(np.isfinite(G.data)):
        return None, None
    return q, G

def solve_lp_step(x, q, G, rho, n, mu):
    m = 2 * n - 1
    I_n = sp.eye(n, format='csc')
    A1 = sp.sparse.hstack([G.tocsc(), sp.csc_matrix((m, n)), -sp.csc_matrix(np.eye(m, 1))])
    A2 = sp.sparse.hstack([I_n, -I_n.copy(), sp.csc_matrix((n, 1))])
    A3 = sp.sparse.hstack([-I_n.copy(), -I_n.copy(), sp.csc_matrix((n, 1))])
    A_ub = sp.vstack([A1, A2, A3]).tocsr()
    b_ub = np.concatenate([-q, np.zeros(n), np.zeros(n)])
    A_eq = sp.sparse.hstack([sp.csc_matrix(np.ones((1, n))), sp.csc_matrix((1, n + 1))]).tocsr()
    b_eq = np.zeros(1)
    c = np.concatenate([np.zeros(n), mu * np.ones(n), np.ones(1)])
    lbs = np.maximum(-rho, -x)
    ubs = np.minimum(rho, 1.0 - x)
    bounds = list(zip(lbs, ubs)) + [(0.0, None)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'maxiter': 500, 'presolve': True})
    except Exception:
        return None
    if res is None or not res.success:
        return None
    delta = np.asarray(res.x[:n], dtype=np.float64)
    if not np.all(np.isfinite(delta)) or abs(np.sum(delta)) > 1e-5:
        return None
    return delta

def optimize_trajectory(h0, n, rng, best_h, best_score):
    S = n / 2.0
    dx = 2.0 / n
    mu = 0.002 * dx
    x = h0.copy()
    f = exact_score(x, n)
    if f is None:
        return best_h, best_score
    rho = 0.08
    rho_max, rho_min = 0.25, 2e-4
    fails = 0
    if n <= 128:
        max_iter = 30
    elif n <= 256:
        max_iter = 22
    elif n <= 512:
        max_iter = 14
    else:
        max_iter = 6
    for _ in range(max_iter):
        if rho < rho_min or fails >= 4:
            break
        q, G = build_lag_model(x, n)
        if G is None:
            fails += 1
            rho *= 0.5
            continue
        delta = solve_lp_step(x, q, G, rho, n, mu)
        if delta is None or np.max(np.abs(delta)) < 1e-9:
            fails += 1
            rho *= 0.5
            continue
        best_cand = None
        best_f_new = f
        for j in range(9):
            beta = 2.0 ** (-j)
            cand = project_capped_simplex(x + beta * delta, S)
            if cand is None:
                continue
            fc = exact_score(cand, n)
            if fc is None:
                continue
            if fc < best_score - 1e-12:
                best_h = cand.copy()
                best_score = fc
            if fc < best_f_new - 1e-12:
                best_f_new = fc
                best_cand = cand
        if best_cand is None:
            fails += 1
            rho *= 0.5
            continue
        s = best_cand - x
        q_new = exact_lags(best_cand, n)
        if q_new is not None:
            lin_max = float(np.max(q + G @ s))
            pred = f - lin_max
        else:
            pred = 0.0
        act = f - best_f_new
        if pred <= 1e-12 or act / max(pred, 1e-12) < 0.25:
            rho = max(rho_min, rho * 0.5)
        elif act / max(pred, 1e-12) > 0.75 and np.max(np.abs(s)) >= 0.8 * rho:
            rho = min(rho_max, rho * 1.5)
        x = best_cand
        f = best_f_new
        fails = 0
        if np.max(np.abs(s)) < 1e-8:
            break
    if f < best_score - 1e-12:
        best_h = x.copy()
        best_score = f
    return best_h, best_score

def build_starts(n, S, parent, rng):
    starts = []
    const = np.full(n, 0.5)
    if parent is not None:
        starts.append(parent.copy())
        for sig in [0.04, 0.12]:
            p = project_capped_simplex(parent + sig * rng.normal(0.0, 1.0, n), S)
            if p is not None:
                starts.append(p)
        starts.append(parent[::-1].copy())
        p_alt = project_capped_simplex(1.0 - parent, S)
        if p_alt is not None:
            starts.append(p_alt)
    else:
        starts.append(const.copy())
    idx = np.arange(n, dtype=np.float64)
    M = min(5, max(1, (n - 1) // 2))
    u = np.full(n, 0.5)
    for m2 in range(1, M + 1):
        a = rng.normal(0.0, 0.10 / np.sqrt(m2))
        phi = rng.uniform(0.0, 2.0 * np.pi)
        u += a * np.cos(2.0 * np.pi * m2 * idx / n + phi)
    p = project_capped_simplex(u, S)
    if p is not None:
        starts.append(p)
    perm = rng.permutation(n)
    binv = np.zeros(n)
    binv[perm[:n // 2]] = 1.0
    p = project_capped_simplex(binv, S)
    if p is not None:
        starts.append(p)
    deduped = []
    for s in starts:
        if s is None or not is_feasible(s, n):
            continue
        dup = False
        for d in deduped:
            if np.max(np.abs(s - d)) < 1e-9:
                dup = True
                break
        if not dup:
            deduped.append(s)
    if n <= 128:
        limit = 6
    elif n <= 256:
        limit = 4
    elif n <= 512:
        limit = 3
    else:
        limit = 2
    return deduped[:limit]

def run(seed=42, **kwargs):
    if seed is None:
        seed = 42
    try:
        seed = int(seed)
    except (TypeError, ValueError):
        seed = 42
    rng = np.random.default_rng(seed)
    n = None
    parent = None
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        try:
            iv_arr = np.asarray(iv, dtype=np.float64)
            if (iv_arr.ndim == 1 and 8 <= iv_arr.size <= 999
                    and np.all(np.isfinite(iv_arr)) and np.max(np.abs(iv_arr)) <= 1e12):
                n = iv_arr.size
                S_tmp = n / 2.0
                if is_feasible(iv_arr, n):
                    parent = iv_arr.copy()
                else:
                    pr = project_capped_simplex(iv_arr, S_tmp)
                    if pr is not None and is_feasible(pr, n):
                        parent = pr
        except Exception:
            pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 80))
        except (TypeError, ValueError):
            n = 80
        if n < 8 or n > 999:
            n = 80
    S = n / 2.0
    h_const = np.full(n, 0.5)
    best_h = h_const.copy()
    best_score = exact_score(h_const, n)
    if best_score is None:
        best_score = 0.5
    if parent is not None:
        ps = exact_score(parent, n)
        if ps is not None and np.isfinite(ps) and ps < best_score:
            best_score = ps
            best_h = parent.copy()
    starts = build_starts(n, S, parent, rng)
    for s in starts:
        sc = exact_score(s, n)
        if sc is not None and np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_h = s.copy()
    for s in starts:
        best_h, best_score = optimize_trajectory(s, n, rng, best_h, best_score)
    if n <= 128:
        n_restart = 2
    elif n <= 256:
        n_restart = 1
    else:
        n_restart = 0
    sigma = 0.05
    for _ in range(n_restart):
        pert = project_capped_simplex(best_h + sigma * rng.normal(0.0, 1.0, n), S)
        if pert is not None and is_feasible(pert, n) and np.max(np.abs(pert - best_h)) > 1e-9:
            old_score = best_score
            best_h, best_score = optimize_trajectory(pert, n, rng, best_h, best_score)
            if best_score < old_score - 1e-12:
                sigma = max(0.02, 0.5 * sigma)
            else:
                sigma = min(0.18, 1.8 * sigma)
    if not is_feasible(best_h, n):
        rep = project_capped_simplex(best_h, S)
        if rep is not None and is_feasible(rep, n):
            best_h = rep
    final_score = exact_score(best_h, n)
    if final_score is None or not np.isfinite(final_score):
        if parent is not None and is_feasible(parent, n):
            best_h = parent.copy()
            final_score = exact_score(parent, n)
        if final_score is None or not np.isfinite(final_score):
            best_h = h_const.copy()
            final_score = 0.5
    return best_h, float(final_score), int(n)
