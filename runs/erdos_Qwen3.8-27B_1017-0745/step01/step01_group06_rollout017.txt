import numpy as np
from scipy.optimize import minimize

def project_K(y, n, S):
    if not np.all(np.isfinite(y)):
        return np.full(n, S / n)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    err = S - h.sum()
    if abs(err) > 1e-12:
        if err > 0:
            idx = np.where(1.0 - h > 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) > 0:
            h[idx] = np.clip(h[idx] + err / len(idx), 0.0, 1.0)
    if abs(h.sum() - S) > 1e-8:
        return np.full(n, S / n)
    return h

def exact_score(h, dx):
    if h.size == 0 or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lag_vec(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_grad_matrix(h, n, dx):
    G = np.zeros((2 * n - 1, n))
    j = np.arange(n)
    for k in range(2 * n - 1):
        ell = k - (n - 1)
        a = j - ell
        v1 = (a >= 0) & (a < n)
        b = j + ell
        v2 = (b >= 0) & (b < n)
        t1 = np.zeros(n); t2 = np.zeros(n)
        t1[v1] = 1.0 - h[a[v1]]
        t2[v2] = h[b[v2]]
        G[k] = dx * (t1 - t2)
    return G

def feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def tr_obj(z, eta):
    d = z[:-1]
    return z[-1] + 0.5 * (d @ d) / eta

def tr_obj_jac(z, eta):
    n = len(z) - 1
    g = np.empty(len(z))
    g[:n] = z[:n] / eta
    g[n] = 1.0
    return g

def tr_ineq(z, q, G):
    return z[-1] - q - G @ z[:-1]

def tr_ineq_jac(z, q, G):
    n = G.shape[1]
    J = np.empty((G.shape[0], n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def tr_eq(z, n, S):
    return z[:n].sum()

def tr_eq_jac(z, n, S):
    g = np.zeros(len(z))
    g[:len(z) - 1] = 1.0
    return g

def tr_solve(h, n, dx, delta, eta, maxiter):
    q = lag_vec(h, dx)
    G = lag_grad_matrix(h, n, dx)
    z0 = np.zeros(n + 1)
    z0[-1] = q.max()
    bounds = [(max(-delta, -h[j]), min(delta, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    con = [
        {'type': 'eq', 'fun': tr_eq, 'jac': tr_eq_jac, 'args': (n, n / 2.0)},
        {'type': 'ineq', 'fun': tr_ineq, 'jac': tr_ineq_jac, 'args': (q, G)}
    ]
    try:
        res = minimize(tr_obj, z0, jac=tr_obj_jac, args=(eta,),
                       method='SLSQP', bounds=bounds, constraints=con,
                       options={'maxiter': maxiter, 'ftol': 1e-10, 'disp': False})
        if res is not None and hasattr(res, 'x') and np.all(np.isfinite(res.x)) and res.x.size == n + 1:
            return res.x
    except Exception:
        pass
    return None
