    best = baseline.copy()
    best_score = exact_score(best, N)
    if N <= 999:
        rng = np.random.default_rng(seed)
        starts = [best.copy()]
        comp = 1.0 - best
        starts.append(project_feasible(comp, m))
        for sc in [0.04, 0.12, 0.25]:
            starts.append(project_feasible(best + sc * rng.standard_normal(N), m))
        starts.append(project_feasible(0.5 + 0.5 * rng.uniform(size=N), m))
        starts.append(project_feasible(make_fourier(N, rng, 6, 0.12), m))
        starts.append(project_feasible(make_fourier(N, rng, 4, 0.22), m))
        starts.append(project_feasible(make_fourier(N, rng, 8, 0.08), m))
        if N <= 128:
            n_starts = 9
            max_upd = 50
            hop_scales = [0.04, 0.015]
        elif N <= 256:
            n_starts = 7
            max_upd = 30
            hop_scales = [0.04]
        else:
            n_starts = 5
            max_upd = 16
            hop_scales = []
        starts = starts[:n_starts]
        seen = []
        for st in starts:
            if np.all(np.isfinite(st)):
                if all(np.abs(st - s).max() > 1e-10 for s in seen):
                    seen.append(st.copy())
        for h0 in seen:
            s0 = exact_score(h0, N)
            if s0 < best_score:
                best = h0.copy()
                best_score = s0
            h_opt, s_opt = optimize_start(h0, N, m, max_upd)
            if s_opt < best_score:
                best = h_opt.copy()
                best_score = s_opt
        for sc in hop_scales:
            hop = project_feasible(best + sc * rng.standard_normal(N), m)
            s_hop = exact_score(hop, N)
            if s_hop < best_score:
                best = hop.copy()
                best_score = s_hop
            h_opt, s_opt = optimize_start(hop, N, m, max_upd)
            if s_opt < best_score:
                best = h_opt.copy()
                best_score = s_opt
    best = project_feasible(best, m)
    c5 = exact_score(best, N)
    base_s = exact_score(baseline, N)
    if base_s < c5:
        best = baseline.copy()
        c5 = base_s
    if not np.isfinite(c5):
        best = np.full(N, 0.5)
        c5 = exact_score(best, N)
    return best, float(c5), int(N)
