def subgradient_search(h0, n, n_iter):
    h = h0.copy()
    score = exact_score(h, n)
    dx = 2.0 / n
    eta = 0.2 / dx
    eta_max = 2.0 / dx
    fails = 0
    for _ in range(n_iter):
        if fails >= 8 or eta < 1e-8 / dx:
            break
        q = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = int(np.argmax(q))
        d = idx - (n - 1)
        p0 = max(d, 0)
        q0 = max(-d, 0)
        m = n - abs(d)
        g = np.zeros(n)
        g[p0:p0 + m] += dx * (1.0 - h[q0:q0 + m])
        g[q0:q0 + m] -= dx * h[p0:p0 + m]
        z = project_capped(h - eta * g, n)
        if z is None or not is_feasible(z, n):
            eta *= 0.5
            fails += 1
            continue
        v = z - h
        if np.max(np.abs(v)) < 1e-10:
            eta *= 0.5
            fails += 1
            continue
        best_t, best_s = h, score
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            trial = project_capped(h + alpha * v, n)
            if trial is None or not is_feasible(trial, n):
                continue

            if s < best_s - 1e-12:
                best_s, best_t = s, trial
        if best_s >= score - 1e-12:
            eta *= 0.5
            fails += 1
            continue
        h = best_t
        score = best_s
        fails = 0
        eta = min(1.2 * eta, eta_max)
    return h, score

    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)

        if abs(r) < 1e-14:
            break

            a = min(r, 1.0 - h[i]); h[i] += a; r -= a
        else:
            a = min(-r, h[i]); h[i] -= a; r += a
