import numpy as np

def project_box_sum(z, N, m):
    if z.shape != (N,) or not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    for _ in range(3):
        if r > 1e-14:
            for i in range(N):
                adj = min(r, 1.0 - h[i])
                if adj > 0:
                    h[i] += adj
                    r -= adj
                    if r <= 1e-14:
                        break
        elif r < -1e-14:
            for i in range(N):
                adj = min(-r, h[i])
                if adj > 0:
                    h[i] -= adj
                    r += adj
                    if r >= -1e-14:
                        break
        else:
            break
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(h.sum() - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return h

def is_feasible(h, N, m):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return False
    if h.min() < -1e-12 or h.max() > 1.0 + 1e-12:
        return False
    if abs(h.sum() - m) > 1e-10 * max(1, N):
        return False
    return True

def exact_score(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    corr = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(corr)):
        return np.inf
    return float(np.max(corr) * (2.0 / N))

def make_fourier(N, rng, n_modes, amp):
    z = np.full(N, 0.5)
    R = min(n_modes, (N - 1) // 2)
    if R == 0:
        return z
    idx = np.arange(N)
    for r in range(1, R + 1):
        a = rng.standard_normal()
        b = rng.standard_normal()
        z += (amp / r) * (a * np.cos(2.0 * np.pi * r * idx / N) + b * np.sin(2.0 * np.pi * r * idx / N))
    return z

def build_population(baseline, N, m, P, rng):
    pop = [baseline.copy()]
    def try_add(cand):
        cand = project_box_sum(cand, N, m)
        if not is_feasible(cand, N, m):
            return
        for e in pop:
            if np.max(np.abs(cand - e)) <= 1e-10:
                return
        pop.append(cand)
    comp = 1.0 - baseline
    try_add(comp)
    perm = rng.permutation(N)
    try_add(baseline[perm])
    for sd in [0.02, 0.06, 0.15, 0.30]:
        try_add(baseline + sd * rng.standard_normal(N))
    try_add(rng.uniform(0.0, 1.0, size=N))
    try_add((rng.random(N) > 0.5).astype(float))
    try_add(make_fourier(N, rng, 6, 0.12))
    try_add(make_fourier(N, rng, 4, 0.24))
    attempts = 0
    while len(pop) < P and attempts < 20 * P:
        if attempts % 2 == 0:
            sd = 10.0 ** rng.uniform(np.log10(0.03), np.log10(0.35))
            try_add(baseline + sd * rng.standard_normal(N))
        else:
            try_add(rng.uniform(0.0, 1.0, size=N))
        attempts += 1
    while len(pop) < P:
        pop.append(pop[0].copy())
    return pop[:P]

def de_search(pop, N, m, P, max_gen, stag_limit, max_restarts, F_De, CR, rng):
    scores = np.array([exact_score(pop[i], N) for i in range(P)])
    best_h = pop[0].copy()
    best_s = scores[0]
    for i in range(P):
        if scores[i] < best_s:
            best_s = scores[i]
            best_h = pop[i].copy()
    stag = 0
    restarts = 0
    total_gen = 0
    while total_gen < max_gen:
        snap_pop = [pop[i].copy() for i in range(P)]
        snap_scores = scores.copy()
        prev_best = best_s
        for i in range(P):
            others = [j for j in range(P) if j != i]
            r1, r2, r3 = rng.choice(others, size=3, replace=False)
            v = snap_pop[r1] + F_De * (snap_pop[r2] - snap_pop[r3])
            mask = rng.random(N) < CR
            j_rand = rng.integers(0, N)
            mask[j_rand] = True
            u = np.where(mask, v, snap_pop[i])
            if not np.all(np.isfinite(u)):
                continue
            trial = project_box_sum(u, N, m)
            if not is_feasible(trial, N, m):
                continue
            ts = exact_score(trial, N)
            if ts < snap_scores[i] - 1e-12:
                pop[i] = trial
                scores[i] = ts
                if ts < best_s:
                    best_s = ts
                    best_h = trial.copy()
        total_gen += 1
        if best_s < prev_best - 1e-12:
            stag = 0
        else:
            stag += 1
        if stag >= stag_limit:
            if restarts >= max_restarts:
                if stag >= 2 * stag_limit:
                    break
            else:
                new_pop = [best_h.copy()]
                order = np.argsort(scores)
                n_keep = max(2, P // 4)
                for k in range(min(n_keep, P)):
                    idx = order[k]
                    if not any(np.max(np.abs(pop[idx] - new_pop[j])) < 1e-10 for j in range(len(new_pop))):
                        new_pop.append(pop[idx].copy())
                n_noise = int(round(3.0 * (P - len(new_pop)) * 0.75))
                n_rand = P - len(new_pop) - n_noise
                sig = 0.14 if restarts == 0 else 0.05
                for _ in range(n_noise):
                    c = project_box_sum(best_h + sig * rng.standard_normal(N), N, m)
                    if is_feasible(c, N, m):
                        new_pop.append(c)
                for _ in range(n_rand):
                    c = project_box_sum(rng.uniform(0.0, 1.0, size=N), N, m)
                    if is_feasible(c, N, m):
                        new_pop.append(c)
                while len(new_pop) < P:
                    new_pop.append(new_pop[0].copy())
                pop = new_pop[:P]
                scores = np.array([exact_score(pop[i], N) for i in range(P)])
                for i in range(P):
                    if scores[i] < best_s:
                        best_s = scores[i]
                        best_h = pop[i].copy()
                stag = 0
                restarts += 1
    return best_h, best_s

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
