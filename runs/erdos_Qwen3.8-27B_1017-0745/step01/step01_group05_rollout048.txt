            if not np.all(np.isfinite(G)) or np.max(np.abs(np.diag(G) - 1)) > 1e-6:
                continue
            if np.max(np.abs(B.sum(axis=0))) > 1e-6:
                continue
            return B
        except Exception:
            pass
    return None

def cma_restart(anchor, B, sigma0, n, rng, n_gen):
    d = B.shape[1]
    rho = math.sqrt(n / float(d))
    lam = max(10, min(16, 4 + int(3 * math.log(max(d, 2)))))
    mu = lam // 2
    lw = np.zeros(mu)
    for i in range(mu):
        lw[i] = math.log(mu + 0.5) - math.log(i + 1)
    w = lw / lw.sum()
    mu_eff = 1.0 / float(np.sum(w * w))
    cc = (4.0 + mu_eff / d) / (d + 4.0 + 2.0 * mu_eff / d)
    csig = (mu_eff + 2.0) / (d + mu_eff + 5.0)
    c1 = 2.0 / ((d + 1.3) ** 2 + mu_eff)
    cmu = min(1.0 - c1, 2.0 * (mu_eff - 2.0 + 1.0 / mu_eff) / ((d + 2.0) ** 2 + mu_eff))
    dsig = 1.0 + 2.0 * max(0.0, math.sqrt((mu_eff - 1.0) / (d + 1.0)) - 1.0) + csig
    chi_d = math.sqrt(d) * (1.0 - 1.0 / (4.0 * d) + 1.0 / (21.0 * d * d))
    m = np.zeros(d)
    C = np.eye(d)
    pc = np.zeros(d)
    ps = np.zeros(d)
    sigma = sigma0
    best_h = anchor.copy()
    best_s = exact_score(anchor, n)
    resets = 0
    stall = 0
    for gen in range(1, n_gen + 1):
        C = 0.5 * (C + C.T)
        try:
            ev, evec = np.linalg.eigh(C)
        except Exception:
            if resets >= 1:
                break
            C = np.eye(d)
            pc, ps = np.zeros(d), np.zeros(d)
            sigma = max(1e-4, sigma * 0.5)
            resets += 1
            continue
        ev = np.clip(ev, 1e-8, 1e3)
        C = evec @ np.diag(ev) @ evec.T
        C = 0.5 * (C + C.T)
        A = evec @ np.diag(np.sqrt(ev))
        Cinv = evec @ np.diag(1.0 / np.sqrt(ev)) @ evec.T
        samples = []
        scores = []
        for j in range(lam):
            if gen == 1 and j == 0:
                x = m.copy()
            else:
                x = m + sigma * (A @ rng.standard_normal(d))
            x = np.clip(x, -30, 30)
            raw = anchor + rho * (B @ x)
            hc, ok = project_capped(raw, n)
            if ok:
                sc = exact_score(hc, n)
                if np.isfinite(sc):
                    samples.append(x)
                    scores.append(sc)
                    if sc < best_s - 1e-12:
                        best_s = sc
                        best_h = hc.copy()
                    continue
            samples.append(None)
            scores.append(float('inf'))
        valid = [i for i in range(lam) if np.isfinite(scores[i])]
        if len(valid) < mu:
            break
        valid.sort(key=lambda i: scores[i])
        sel = valid[:mu]
        m_new = np.zeros(d)
        for i, idx in enumerate(sel):
            m_new += w[i] * samples[idx]
        m_new = np.clip(m_new, -30, 30)
        raw = anchor + rho * (B @ m_new)
        hc, ok = project_capped(raw, n)
        if ok:
            sc = exact_score(hc, n)
            if np.isfinite(sc) and sc < best_s - 1e-12:
                best_s = sc
                best_h = hc.copy()
        y_w = (m_new - m) / sigma
        ps = (1.0 - csig) * ps + math.sqrt(csig * (2.0 - csig) * mu_eff) * (Cinv @ y_w)
        denom = math.sqrt(1.0 - (1.0 - csig) ** (2 * gen))
        hsig = 1.0 if (np.linalg.norm(ps) / denom) < (1.4 + 2.0 / (d + 1)) * chi_d else 0.0
        pc = (1.0 - cc) * pc + hsig * math.sqrt(cc * (2.0 - cc) * mu_eff) * y_w
        Cn = (1.0 - c1 - cmu + c1 * (1.0 - hsig) * cc * (2.0 - cc)) * C
        Cn += c1 * np.outer(pc, pc)
        for i in range(mu):
            yi = (samples[sel[i]] - m) / sigma
            Cn += cmu * w[i] * np.outer(yi, yi)
        C = 0.5 * (Cn + Cn.T)
        ea = (csig / dsig) * (np.linalg.norm(ps) / chi_d - 1.0)
        ea = float(np.clip(ea, -2.0, 2.0))
        sigma = float(np.clip(sigma * math.exp(ea), 1e-4, 0.5))
        m = m_new
        if rho * sigma * math.sqrt(float(np.max(ev))) < 1e-5:
            stall += 1
            if stall >= 5:
                break
        else:
            stall = 0
    return best_h, best_s

def complement_vec(h):
    return 1.0 - h

def reverse_vec(h):
    return h[::-1].copy()

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = globals().get('initial_h_values', None)
    n = 45
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.ndim == 1 and 2 <= len(p) <= 999 and np.all(np.isfinite(p)) and np.max(np.abs(p)) < 1e6:
            n = len(p)
            if check_feasible(p, n):
                fallback = p.copy()
            else:
                fallback, ok = project_capped(p, n)
                if not ok:
                    fallback = np.full(n, 0.5)
        else:
            fallback = np.full(n, 0.5)
    else:
        nk = kwargs.get('n_points', None)
        if isinstance(nk, int) and not isinstance(nk, bool) and 2 <= nk <= 999:
            n = nk
        fallback = np.full(n, 0.5)
    best_h = fallback.copy()
    best_s = exact_score(best_h, n)
    if n <= 64:
        B = make_helmert(n)
    else:
        B = make_rand_basis(n, 48, rng)
        if B is None:
            B = make_helmert(65)
            n_b = 65
            B = B[:n, :]
    d_used = B.shape[1]
    starts = [fallback.copy()]
    for t in range(3):
        if t == 0:
            s = complement_vec(fallback)
        elif t == 1:
            s = reverse_vec(fallback)
        else:
            s = complement_vec(reverse_vec(fallback))
        s, ok = project_capped(s, n)
        if ok and np.max(np.abs(s - best_h)) > 1e-10:
            starts.append(s)
            sc = exact_score(s, n)
            if sc < best_s:
                best_s = sc
                best_h = s.copy()
    n_gen = 36 if n <= 100 else (12 if n <= 400 else 6)
    restart_defs = []
    restart_defs.append((fallback, 0.06))
    if len(starts) > 1:
        restart_defs.append((starts[1], 0.10))
    else:
        restart_defs.append((fallback, 0.10))
    z = rng.standard_normal(n)
    z -= z.mean()
    nz = np.linalg.norm(z)
    if nz > 1e-12:
        z /= nz
    pert = best_h + 0.08 * z
    pp, pok = project_capped(pert, n)
    if pok and np.max(np.abs(pp - best_h)) > 1e-10:
        restart_defs.append((pp, 0.04))
    for ri in range(len(restart_defs)):
        try:
            a, sig0 = restart_defs[ri]
            if not check_feasible(a, n):
                a, ok = project_capped(a, n)
                if not ok:
                    continue
            bh, bs = cma_restart(a, B, sig0, n, rng, n_gen)
            if bs < best_s - 1e-12:
                best_s = bs
                best_h = bh.copy()
        except Exception:
            pass
    # Fine polishing pass
    try:
        z2 = rng.standard_normal(n)
        z2 -= z2.mean()
        nz2 = np.linalg.norm(z2)
        if nz2 > 1e-12:
            z2 /= nz2
        pt = best_h + 0.02 * z2
        pt, pok = project_capped(pt, n)
        if pok:
            bh, bs = cma_restart(pt, B, 0.015, n, rng, 30)
            if bs < best_s - 1e-12:
                best_s = bs
                best_h = bh.copy()
    except Exception:
        pass
    final_h, fok = project_capped(best_h, n)
    if not fok:
        final_h = fallback.copy()
    fs = exact_score(final_h, n)
    fs_f = exact_score(fallback, n)
    if fs_f <= fs:
        final_h = fallback.copy()
        fs = fs_f
    return final_h.astype(np.float64), float(fs), int(n)
