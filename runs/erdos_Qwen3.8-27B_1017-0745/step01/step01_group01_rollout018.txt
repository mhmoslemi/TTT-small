            final = h0.copy()
        fs = exact_score(final, n)
        if not np.isfinite(fs):
            final, fs = uniform.copy(), 0.25
        return final.copy(), float(fs), int(n)
    pop = []
    for st in starts:
        if len(pop) >= NP:
            break
        pop.append(st.copy())
    add_unique(pop, best, n)
    fill_scales = (0.015, 0.04, 0.10, 0.22)
    fi = 0
    while len(pop) < NP and fi < 200:
        base = best if rng.random() < 0.5 else pop[int(rng.integers(0, len(pop)))]
        scale = fill_scales[fi % 4]
        cand = base + rng.normal(0, scale, n)
        if not add_unique(pop, cand, n):
            add_unique(pop, rng.uniform(0, 1, n), n)
        fi += 1
    while len(pop) < NP:
        pop.append(best.copy())
    pop_scores = np.array([exact_score(p, n) for p in pop])
    bi = int(np.argmin(pop_scores))
    if pop_scores[bi] < best_score - 1e-12:
        best_score, best = pop_scores[bi], pop[bi].copy()
    F_lo, F_hi = 0.45, 0.95
    CR_lo, CR_hi = 0.75, 0.95
    patience = max(12, min(40, gens // 4))
    stagnation = 0
    n_restarts = 0
    for gen in range(gens):
        old_pop = np.array([p.copy() for p in pop])
        old_scores = pop_scores.copy()
        b = int(np.argmin(old_scores))
        prev_best = best_score
        for i in range(NP):
            if rng.random() < 0.8:
                cands = [j for j in range(NP) if j != i]
                perm = rng.permutation(len(cands))[:3]
                r1, r2, r3 = cands[perm[0]], cands[perm[1]], cands[perm[2]]
                F = F_lo + rng.random() * (F_hi - F_lo)
                v = old_pop[r1] + F * (old_pop[r2] - old_pop[r3])
            else:
                cands = [j for j in range(NP) if j != i and j != b]
                if len(cands) < 2:
                    continue
                perm = rng.permutation(len(cands))[:2]
                r1, r2 = cands[perm[0]], cands[perm[1]]
                F = F_lo + rng.random() * (F_hi - F_lo)
                v = old_pop[i] + F * (old_pop[b] - old_pop[i]) + F * (old_pop[r1] - old_pop[r2])
            CR = CR_lo + rng.random() * (CR_hi - CR_lo)
            mask = rng.random(n) < CR
            mask[int(rng.integers(0, n))] = True
            trial = np.where(mask, v, old_pop[i])
            proj = project_capped(trial, n)
            if proj is None:
                continue
            if vec_dist(proj, old_pop[i]) <= 1e-12:
                continue
            ts = exact_score(proj, n)
            if np.isfinite(ts) and ts < old_scores[i] - 1e-12:
                pop[i] = proj
                pop_scores[i] = ts
                if ts < best_score - 1e-12:
                    best_score, best = ts, proj.copy()
        if best_score < prev_best - 1e-12:
            stagnation = 0
        else:
            stagnation += 1
        if stagnation >= patience:
            if n_restarts >= max_rs:
                break
            n_restarts += 1
            n_elite = max(2, math.ceil(0.2 * NP))
            order = np.argsort(pop_scores)
            keep = [pop[order[k]].copy() for k in range(min(n_elite, len(pop)))]
            if not any(vec_dist(best, k) < 1e-10 for k in keep):
                keep.insert(0, best.copy())
            base_scale = (0.03, 0.07, 0.15)[(n_restarts - 1) % 3] * (0.5 + rng.random())
            while len(keep) < NP:
                base = best if rng.random() < 0.4 else keep[int(rng.integers(0, len(keep)))]
                add_unique(keep, base + rng.normal(0, base_scale, n), n)
                add_unique(keep, rng.uniform(0, 1, n), n)
            while len(keep) < NP:
                keep.append(best.copy())
            pop = keep[:NP]
            pop_scores = np.array([exact_score(p, n) for p in pop])
            b2 = int(np.argmin(pop_scores))
            if pop_scores[b2] < best_score - 1e-12:
                best_score, best = pop_scores[b2], pop[b2].copy()
            stagnation = 0
    cand_list = [best, h0, uniform]
    final, final_score = None, float('inf')
    for cand in cand_list:
        cs = exact_score(cand, n)
        if np.isfinite(cs) and cs < final_score:
            final_score, final = cs, cand
    if final is None:
        final, final_score = uniform.copy(), exact_score(uniform, n)
    rep = project_capped(final, n)
    if rep is not None:
        rs = exact_score(rep, n)
        if np.isfinite(rs) and rs <= final_score + 1e-12:
            final, final_score = rep, rs
    if not np.isfinite(final_score):
        final, final_score = uniform.copy(), exact_score(uniform, n)
    return final.copy(), float(final_score), int(n)
