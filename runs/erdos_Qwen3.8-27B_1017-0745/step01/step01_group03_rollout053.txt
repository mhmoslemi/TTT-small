def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0 or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if resid > 1e-14:
        order = np.argsort(1.0 - y)
        for i in order:
            add = min(resid, 1.0 - y[i])
            y[i] += add
            resid -= add
            if resid < 1e-15:
                break
    elif resid < -1e-14:
        order = np.argsort(y)
        for i in order:
            sub = min(-resid, y[i])
            y[i] -= sub
            resid += sub
            if -resid < 1e-15:
                break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):

    c = np.correlate(h, 1.0 - h, mode='full')

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0 or not np.all(np.isfinite(v)):
        return None
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if resid > 1e-14:
        for i in np.argsort(1.0 - y):
            a = min(resid, 1.0 - y[i])
            y[i] += a; resid -= a
            if resid < 1e-15: break
    elif resid < -1e-14:
        for i in np.argsort(y):
            a = min(-resid, y[i])
            y[i] -= a; resid += a
            if -resid < 1e-15: break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1.0+1e-12):
        return None

def exact_score(h, n):
    if not np.all(np.isfinite(h)): return float('inf')
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    s = float(np.max(c) * dx)
    return s if np.isfinite(s) else float('inf')

def check_feasible(h, n):

    if np.any(h < -1e-12) or np.any(h > 1.0+1e-12): return False
    if abs(np.sum(h) - n/2.0) > 5e-10: return False
    return True

def pidx(i, j, n):
    if i > j: i, j = j, i
    return i * n - i * (i-1) // 2 + (j - i)

def build_lp(x, r, n, S):

    nvar = n + P + 1
    l = np.maximum(0.0, x - r); u = np.minimum(1.0, x + r)
    obj = np.zeros(nvar); obj[-1] = 1.0

        for j in range(i, n):
            pi = n + pidx(i, j, n)
            li, ui, lj, uj = l[i], u[i], l[j], u[j]
            if i == j:
                rows += [len(rhs)]*3
                cols += [pi, i, pi, i, pi, i]
                vals += [-1, 2*li, -1, 2*ui, 1, -(li+ui)]
                rhs += [li*li, ui*ui, -li*ui]
            else:

                cols += [pi, i, j, pi, i, j, pi, i, j, pi, i, j]
                vals += [-1,li,lj, -1,ui,uj, 1,-ui,-lj, 1,-li,-uj]
                rhs += [li*lj, ui*uj, -ui*lj, -li*uj]
    for k in range(-(n-1), n):

        if i_lo > i_hi: continue

def line_minimax(x, y, n, S):
