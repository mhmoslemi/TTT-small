def smooth_obj_grad(h, tau, dx):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = dx * corr
    cmax = c.max()
    e = np.exp((c - cmax) / tau)
    sum_e = e.sum()
    p = e / sum_e
    F = cmax + tau * np.log(sum_e)
    
    grad = np.zeros(N)
    # s = 0, r = N-1
    p0 = p[N-1]
    if p0 > 1e-15:
        grad += p0 * (1.0 - 2.0 * h)
    
    # s > 0, r = N-1+s, s = 1, ..., N-1
    for s in range(1, N):
        pr = p[N-1+s]
        if pr > 1e-15:
            L = N - s
            grad[s:N] += pr * (1.0 - h[0:L])
            grad[0:L] -= pr * h[s:N]

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(z - lam, 0.0, 1.0)
    r = m - h.sum()
    if r > 1e-14:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
            if r <= 1e-14:

            adj = min(-r, h[i])
            h[i] -= adj
            r += adj

        return np.full(N, 0.5)
    if abs(h.sum() - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)

def exact_score(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / N)

def smooth_obj_grad(h, tau, dx):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = dx * corr
    cmax = c.max()
    e = np.exp((c - cmax) / tau)
    sum_e = e.sum()
    p = e / sum_e
    F = cmax + tau * np.log(sum_e)
    grad = np.zeros(N)
    p0 = p[N - 1]
    if p0 > 1e-15:
        grad += p0 * (1.0 - 2.0 * h)
    for s in range(1, N):
        pr = p[N - 1 + s]
        if pr > 1e-15:
            L = N - s
            grad[s:] += pr * (1.0 - h[:L])
            grad[:L] -= pr * h[s:]
    for d in range(1, N):
        pr = p[N - 1 - d]
        if pr > 1e-15:
            L = N - d
            grad[:L] += pr * (1.0 - h[d:])
            grad[d:] -= pr * h[:L]
    grad *= dx
    return F, grad

def optimize_start(h0, N, m, dx, tau_sched, max_iter):
    best_h = h0.copy()
    best_sc = exact_score(h0, N)
    h = h0.copy()
    for tau in tau_sched:
        F, grad = smooth_obj_grad(h, tau, dx)
        if not np.isfinite(F) or not np.all(np.isfinite(grad)):
            break
        alpha = 1.0
        for _ in range(max_iter):
            y = project_box_sum(h - alpha * grad, m)
            d = y - h
            if not np.all(np.isfinite(y)):
                break
            step = np.abs(d).max()
            if step <= 1e-10:
                break
            Fy, _ = smooth_obj_grad(y, tau, dx)
            if not np.isfinite(Fy):
                alpha *= 0.5
                continue
            sy = exact_score(y, N)
            if np.isfinite(sy) and sy < best_sc:
                best_sc = sy
                best_h = y.copy()
            gd = np.dot(grad, d)
            if gd < 0 and Fy <= F + 1e-4 * gd + 1e-12:
                h = y.copy()
                F = Fy
                grad = smooth_obj_grad(h, tau, dx)[1]
                alpha = min(1.25 * alpha, 10.0)
            else:
                alpha *= 0.5
                if alpha < 1e-12:
                    break
    return best_h, best_sc

def make_fourier_start(N, rng, n_modes, amp):
    z = np.full(N, 0.5)
    ns = min(n_modes, (N - 1) // 2)
    idx = np.arange(N)
    for r in range(1, ns + 1):
        A = rng.standard_normal()
        B = rng.standard_normal()
        z += (amp / r) * (A * np.sin(2.0 * np.pi * r * idx / N) +
                         B * np.cos(2.0 * np.pi * r * idx / N))
    return z

def run(seed=42, **kwargs):
    N = None
    m = None
    raw = globals().get('initial_h_values', None)
    if raw is not None:
        raw = np.asarray(raw, dtype=np.float64)
        if raw.ndim == 1 and raw.size > 1 and np.all(np.isfinite(raw)):
            N = raw.size
            m = N / 2.0
    if N is None:
        N = kwargs.get('n_points', 70)
        if not isinstance(N, (int, np.integer)) or N < 2:
            N = 70
        m = N / 2.0
    dx = 2.0 / N

    if raw is not None and np.all(np.isfinite(raw)):
        baseline = project_box_sum(np.clip(raw, 0.0, 1.0), m)
    else:
        baseline = np.full(N, 0.5)
    baseline = project_box_sum(baseline, m)
    best_h = baseline.copy()
    best_sc = exact_score(best_h, N)
    rng = np.random.default_rng(seed)

    # Build start list
    starts = [baseline.copy(), np.full(N, 0.5)]
    comp = project_box_sum(1.0 - baseline, m)
    starts.append(comp)
    perm = baseline[np.argsort(rng.permutation(N))]
    starts.append(project_box_sum(perm, m))
    for sc in [0.04, 0.12]:
        starts.append(project_box_sum(baseline + sc * rng.standard_normal(N), m))
    starts.append(project_box_sum(rng.uniform(0.0, 1.0, size=N), m))
    starts.append(project_box_sum(make_fourier_start(N, rng, 6, 0.12), m))
    starts.append(project_box_sum(make_fourier_start(N, rng, 4, 0.22), m))

    # Deduplicate
    unique_starts = []
    for s in starts:
        if not np.all(np.isfinite(s)):
            continue
        dup = False
        for u in unique_starts:
            if np.abs(s - u).max() <= 1e-10:
                dup = True
                break
        if not dup:
            unique_starts.append(s)
    starts = unique_starts[:8]

    # Tau schedule and iteration budget based on N
    tau_sched = [0.02, 0.006, 0.002, 0.0006, 0.0002, 0.00006, 0.000015]
    if N <= 128:
        max_iter = 50
        n_hops = 2
    elif N <= 256:
        max_iter = 30
        n_hops = 1
    else:
        max_iter = 16
        n_hops = 0

    # Optimize each start
    for h0 in starts:
        s0 = exact_score(h0, N)
        if s0 < best_sc:
            best_sc = s0
            best_h = h0.copy()
        bh, bs = optimize_start(h0, N, m, dx, tau_sched, max_iter)
        if bs < best_sc:
            best_sc = bs
            best_h = bh.copy()

    # Basin hops
    hop_scales = [0.04, 0.015] if n_hops == 2 else [0.04]
    for i in range(min(n_hops, len(hop_scales))):
        sc = hop_scales[i]
        hop = project_box_sum(best_h + sc * rng.standard_normal(N), m)
        sh = exact_score(hop, N)
        if sh < best_sc:
            best_sc = sh
            best_h = hop.copy()
        bh, bs = optimize_start(hop, N, m, dx, tau_sched, max_iter)
        if bs < best_sc:
            best_sc = bs
            best_h = bh.copy()

    # Final: compare with projected baseline
    final = project_box_sum(best_h, m)
    final_sc = exact_score(final, N)
    base_sc = exact_score(baseline, N)
    if base_sc < final_sc:
        final = baseline.copy()
        final_sc = base_sc
    if not np.isfinite(final_sc):
        final = np.full(N, 0.5)
        final_sc = exact_score(final, N)
    return final, final_sc, int(N)
