The key insight from the strategy is to use a Majorization-Minimization (MM) approach instead of direct SLSQP on the nonconvex problem. The idea is:

1. At each feasible point x, build a convex quadratic upper bound for the maximum of all lag overlaps
2. Minimize this convex upper bound over the feasible set
3. Accept the result only if it actually improves the exact score

The MM approach works because:
- Each lag overlap q_s(h) is a quadratic function
- The Hessian of each q_s has spectral norm bounded by L = 2*dx = 4/n
- So q_s(y) ≤ q_s(x) + ∇q_s(x)·(y-x) + (L/2)||y-x||² for all y
- The maximum of these upper bounds is a convex function
- Minimizing this convex function gives a valid descent direction

import numpy as np
from scipy.optimize import minimize
import math

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    S = float(S)
    if S < 0 or S > n:
        return None
    lo = float(np.min(v)) - 1.5
    hi = float(np.max(v)) + 1.5
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-14:
        for i in np.argsort(1.0 - y):
            d = min(r, 1.0 - y[i])
            y[i] += d
            r -= d
            if r <= 1e-14:
                break
    elif r < -1e-14:
        for i in np.argsort(y):
            d = min(-r, y[i])
            y[i] -= d
            r += d
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-9) or np.any(y > 1.0 + 1e-9):
        return None
    if abs(float(np.sum(y)) - S) > 1e-10 * max(1, n):
        return None
    return y

def check_feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(float(np.sum(h)) - 0.5 * n) < 1e-10 * max(1, n))

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    if c.shape[0] != 2 * n - 1:
        return float('inf')
    s = float(np.max(c)) * (2.0 / n)
    return s if np.isfinite(s) else float('inf')

def lag_val_grad(h, s, n):
    dx = 2.0 / n
    lo_i = max(0, -s)
    hi_i = min(n, n - s)
    q = 0.0
    g = np.zeros(n)
    for i in range(lo_i, hi_i):
        j = i + s
        q += h[i] * (1.0 - h[j])
        g[i] += dx * (1.0 - h[j])
        g[j] -= dx * h[i]
    return q * dx, g

def get_parent():
    n = 45
    ph = globals().get('initial_h_values', None)
    if ph is not None:
        ph = np.asarray(ph, dtype=np.float64)
        if ph.ndim == 1 and 2 <= len(ph) <= 999 and np.all(np.isfinite(ph)):
            if np.all(ph >= -1e-6) and np.all(ph <= 1.0 + 1e-6):
                n = len(ph)
                if check_feasible(ph, n):
                    return ph.copy(), n
                p = project_box_sum(np.clip(ph, 0.0, 1.0), 0.5 * n)
                if p is not None and check_feasible(p, n):
                    return p, n
    return np.full(n, 0.5), n

def mm_obj(z, x, L):
    d = z[:-1] - x
    return 0.5 * L * np.dot(d, d) + z[-1]

def mm_obj_jac(z, x, L):
    g = np.empty(len(z))
    g[:-1] = L * (z[:-1] - x)
    g[-1] = 1.0
    return g

def mm_eq(z, n):
    return float(np.sum(z[:-1])) - 0.5 * n

def mm_eq_jac(z, n):
    g = np.zeros(len(z))
    g[:-1] = 1.0
    return g

def mm_ineq(z, qJ, GJ, x):
    return z[-1] - qJ - GJ @ (z[:-1] - x)

def mm_ineq_jac(z, qJ, GJ, x):
    nr = GJ.shape[0]
    nz = len(z)
    J = np.zeros((nr, nz))
    J[:, :-1] = -GJ
    J[:, -1] = 1.0
    return J

def solve_mm(x, n):
    dx = 2.0 / n
    L = 2.0 * dx
    use_all = n <= 64
    nv = 2 * n - 1
    if use_all:
        s_list = list(range(-(n - 1), n))
    else:
        vals = np.zeros(nv)
        for idx, s in enumerate(range(-(n - 1), n)):
            vals[idx], _ = lag_val_grad(x, s, n)
        top = np.argsort(-vals)[:8]
        s_list = [range(-(n - 1), n)[i] for i in top]
    qJ = np.array([lag_val_grad(x, s, n)[0] for s in s_list])
    GJ = np.array([lag_val_grad(x, s, n)[1] for s in s_list])
    t0 = float(np.max(qJ)) + 1e-6
    z0 = np.append(x.copy(), t0)
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    cons = [
        {'type': 'eq', 'fun': mm_eq, 'jac': mm_eq_jac, 'args': (n,)},
        {'type': 'ineq', 'fun': mm_ineq, 'jac': mm_ineq_jac, 'args': (qJ, GJ, x)},
    ]
    try:
        res = minimize(mm_obj, z0, jac=mm_obj_jac, args=(x, L),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-12, 'disp': False})
        z_r = np.asarray(res.x, dtype=np.float64)
        if z_r.shape[0] != n + 1 or not np.all(np.isfinite(z_r)):
            return x.copy()
        y_r = z_r[:-1]
        if np.any(y_r < -1e-4) or np.any(y_r > 1.0 + 1e-4):
            return x.copy()
        if abs(float(np.sum(y_r)) - 0.5 * n) > 1e-4 * n:
            return x.copy()
        slacks = z_r[-1] - qJ - GJ @ (y_r - x)
        if np.any(slacks < -1e-4):
            return x.copy()
        y_p = project_box_sum(np.clip(y_r, 0.0, 1.0), 0.5 * n)
        if y_p is None:
            return x.copy()
        return y_p
    except Exception:
        return x.copy()

def mm_descent(h0, n, max_steps):
    h = h0.copy()
    c_cur = exact_score(h, n)
    for _ in range(max_steps):
        y_new = solve_mm(h, n)
        if not np.all(np.isfinite(y_new)):
            break
        d = y_new - h
        if np.max(np.abs(d)) < 1e-12:
            break
        best_a, best_c = 0.0, c_cur
        for k in range(10):
            a = 2.0 ** (-k)
            cand = project_box_sum(h + a * d, 0.5 * n)
            if cand is None:
                continue
            c = exact_score(cand, n)
            if c < best_c - 1e-13:
                best_a, best_c = a, c
        if best_a > 0:
            old = c_cur
            h = project_box_sum(h + best_a * d, 0.5 * n)
            if h is None:
                break
            c_cur = exact_score(h, n)
            if c_cur >= old - 1e-10 * max(1.0, abs(old)):
                break
        else:
            break
    return h, exact_score(h, n)

def build_starts(parent, n, rng):
    starts = [parent.copy(), (1.0 - parent).copy(), parent[::-1].copy()]

    unique = [starts[0]]

            unique.append(s)
    return unique

def run(seed=42, **kwargs):
