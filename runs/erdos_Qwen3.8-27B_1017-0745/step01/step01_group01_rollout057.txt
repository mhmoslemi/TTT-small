            z0 = np.concatenate([x0, [t0]])
            try:
                res = minimize(slsqp_obj, z0, method='SLSQP', jac=slsqp_obj_grad,
                               bounds=bnds, constraints=cons, args=(n,),
                               options={'maxiter': max_iter, 'ftol': 1e-10})
            except Exception:
                continue
            if res.x is not None and len(res.x) == n + 1 and np.all(np.isfinite(res.x)):
                hp = project_capped(res.x[:n], n)
                if hp is not None:
                    hs = exact_score(hp, n)
                    if np.isfinite(hs) and hs < best_score - 1e-12:
                        best, best_score = hp.copy(), hs
    if n <= 64 and n_polish > 0:
        polished = pairwise_polish(best, n, n_polish)
        pf = project_capped(polished, n)
        if pf is not None:
            ps = exact_score(pf, n)
            if np.isfinite(ps) and ps < best_score - 1e-12:
                best, best_score = pf.copy(), ps
    final = project_capped(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final, fs = fallback.copy(), fb_score
    if not np.isfinite(fs):
        final, fs = uniform.copy(), exact_score(uniform, n)
    return final, float(fs), int(n)
