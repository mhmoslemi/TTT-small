from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if resid > 0:
        order = np.argsort(1.0 - y)
        for idx in order:
            add = min(resid, 1.0 - y[idx])
            y[idx] += add
            resid -= add
            if resid < 1e-14:
                break
    elif resid < 0:
        order = np.argsort(y)
        for idx in order:
            sub = min(-resid, y[idx])
            y[idx] -= sub
            resid += sub
            if -resid < 1e-14:
                break
    y = np.clip(y, 0.0, 1.0)
    if abs(np.sum(y) - S) > 1e-10 * max(1.0, float(n)):
        return None
    return y

def exact_score(h, n):
    if h is None or not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    sc = float(np.max(c) * dx)
    return sc if np.isfinite(sc) else float('inf')

def check_feasible(h, n):
    if h is None or h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-9 * max(1.0, float(n)):
        return False
    return True

def line_minimax(x, y, n):
    d = y - x
    n_lags = 2 * n - 1
    A = np.zeros(n_lags)
    B = np.zeros(n_lags)
    D = np.zeros(n_lags)
    for k in range(-(n - 1), n):
        ilo = max(0, k)
        ihi = min(n - 1, n - 1 + k)
        if ilo > ihi:
            continue
        i = np.arange(ilo, ihi + 1)
        j = i - k
        A[k + n - 1] = -np.sum(d[i] * d[j])
        B[k + n - 1] = np.sum(d[i] * (1.0 - x[j])) - np.sum(x[i] * d[j])
        D[k + n - 1] = np.sum(x[i] * (1.0 - x[j]))
    # Candidate alphas
    alphas = [0.0, 1.0]
    for ki in range(n_lags):
        if abs(A[ki]) > 1e-14:
            a_star = -B[ki] / (2.0 * A[ki])
            if -1e-10 <= a_star <= 1.0 + 1e-10:
                alphas.append(np.clip(a_star, 0.0, 1.0))
    # Pairwise intersections
    for ki in range(n_lags):
        for kj in range(ki + 1, n_lags):
            da = A[ki] - A[kj]
            db = B[ki] - B[kj]
            dd = D[ki] - D[kj]
            if abs(da) < 1e-14:
                if abs(db) > 1e-14:
                    a_r = -dd / db
                    if -1e-10 <= a_r <= 1.0 + 1e-10:
                        alphas.append(np.clip(a_r, 0.0, 1.0))
            else:
                disc = db * db - 4.0 * da * dd
                if disc >= -1e-10:
                    disc = max(0.0, disc)
                    sq = np.sqrt(disc)
                    for a_r in [(-db + sq) / (2.0 * da), (-db - sq) / (2.0 * da)]:
                        if -1e-10 <= a_r <= 1.0 + 1e-10:
                            alphas.append(np.clip(a_r, 0.0, 1.0))
    alphas = np.array(alphas)
    alphas = np.sort(np.unique(np.round(alphas, 10)))
    # Evaluate
    if len(alphas) > 0:
        idx = np.arange(n_lags)
        vals = A[idx]*alphas[:,None]**2 + B[idx]*alphas[:,None] + D[idx]
        best_a = alphas[0]
        best_val = np.max(vals)
        for ai in range(len(alphas)):
            mv = np.max(vals[ai])
            if mv < best_val - 1e-14:
                best_val = mv
                best_a = alphas[ai]
        # Verify with exact score

        h_cand = project_box_sum(h_cand, n / 2.0)
        if h_cand is not None and check_feasible(h_cand, n):
            return h_cand
    return None

    if v.size == 0 or not np.all(np.isfinite(v)):
        return None
    n = v.size
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - y.sum()
    if r > 1e-13:
        for i in np.argsort(1.0 - y):
            a = min(r, 1.0 - y[i]); y[i] += a; r -= a
            if r < 1e-13: break
    elif r < -1e-13:
        for i in np.argsort(y):
            a = min(-r, y[i]); y[i] -= a; r += a
            if -r < 1e-13: break

    A = np.zeros(nl); B = np.zeros(nl); D = np.zeros(nl)
