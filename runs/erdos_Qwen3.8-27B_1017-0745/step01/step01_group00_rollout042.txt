        if rl is not None:
            cands.append(rl)
        if rr is not None:
            cands.append(rr)
        if rl is not None and rl > L + 1e-12:
            tGL, _ = golden_min_1d(eval_G, L, rl)
            cands.append(tGL)
        if rr is not None and rr < U - 1e-12:
            tGR, _ = golden_min_1d(eval_G, rr, U)
            cands.append(tGR)
    best_t, best_c = 0.0, eval_C(0.0)
    for t in cands:
        t = max(L, min(U, t))
        ct = eval_C(t)
        if ct < best_c - 1e-14 or (abs(ct - best_c) < 1e-14 and abs(t) < abs(best_t)):
            best_t, best_c = t, ct
    if abs(best_t) < 1e-13:
        return None
    return best_t

def local_search(h0, n, max_moves, rng):
    h = h0.copy()
    score = exact_score(h, n)
    idx_arr = np.arange(n)
    for _ in range(max_moves):
        D = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        k_top = min(8, 2 * n - 1)
        top_lags = np.argsort(D)[-k_top:]
        pair_set = set()
        for dl in top_lags:
            d = dl - (n - 1)
            p0, q0 = max(d, 0), max(-d, 0)
            m = n - abs(d)
            gd = np.zeros(n)
            for r in range(m):
                gi = p0 + r
                gj = q0 + r
                gd[gi] += (1.0 - h[gj])
                gd[gj] -= h[gi]
            gd *= (2.0 / n)
            gmax = np.argmax(gd)
            gmin = np.argmin(gd)
            for a in range(n):
                if a != gmax:
                    pair_set.add((min(a, gmax), max(a, gmax)))
            for a in range(n):
                if a != gmin:
                    pair_set.add((min(a, gmin), max(a, gmin)))
        for _ in range(5):
            a, b = rng.integers(0, n, 2)
            if a != b:
                pair_set.add((min(a, b), max(a, b)))
        pairs = list(pair_set)
        if not pairs:
            break
        pairs.sort(key=lambda p: abs(D[np.argmax(np.abs(B_screen(h, p[0], p[1], n))))])
        best_trial, best_sc = None, score
        for (i, j) in pairs[:25]:
            t = pair_line_search(h, i, j, n)
            if t is None:
                continue
            trial = h.copy()
            trial[i] += t
            trial[j] -= t
            trial = project_capped(trial, n)
            if trial is None or not is_feasible(trial, n):
                continue
            sc = exact_score(trial, n)
            if sc < best_sc - 1e-13:
                best_sc, best_trial = sc, trial
        if best_trial is None or best_sc >= score - 1e-12:
            break
        h = best_trial
        score = best_sc
    return h, score

def B_screen(h, i, j, n):
    Lags = 2 * n - 1
    B = np.zeros(Lags)
    dx = 2.0 / n
    for k in range(Lags):
        d = k - (n - 1)
        p0, q0 = max(d, 0), max(-d, 0)
        m = n - abs(d)
        b = 0.0
        ri, rj, si, sj = i - p0, j - p0, i - q0, j - q0
        if 0 <= ri < m:
            b += 1.0 - h[q0 + ri]
        if 0 <= rj < m:
            b -= 1.0 - h[q0 + rj]
        if 0 <= si < m:
            b -= h[p0 + si]
        if 0 <= sj < m:
            b += h[p0 + sj]
        B[k] = dx * b
    return B

def build_starts(n, rng, fallback):
    pool = []
    def add(v):
        pv = project_capped(v, n)
        if pv is None:
            return
        for s in pool:
            if np.max(np.abs(pv - s)) < 1e-8:
                return
        pool.append(pv)
    add(fallback.copy())
    add(fallback[::-1].copy())
    add(1.0 - fallback[::-1])
    add(np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    add(blk)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    add(alt)
    for sd in (0.06, 0.15, 0.35):
        add(fallback + rng.normal(0, sd, n))
    add(rng.uniform(0, 1, n))
    for _ in range(4):
        perm = rng.permutation(n)
        bp = np.zeros(n)
        cnt = n // 2
        bp[perm[:cnt]] = 1.0
        if n % 2:
            bp[perm[cnt]] = 0.5
        add(bp)
    return pool

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0 and p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None:
        n = 47
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if parent is not None:
        parent = np.clip(parent, 0, 1)
        fallback = project_capped(parent, n)
        if fallback is None:
            fallback = np.full(n, 0.5)
    else:
        fallback = np.full(n, 0.5)
    fb_score = exact_score(fallback, n)
    if n <= 64:
        n_starts, max_mv, n_hops, hop_mv = 10, 30, 8, 12
    elif n <= 128:
        n_starts, max_mv, n_hops, hop_mv = 7, 20, 5, 8
    else:
        n_starts, max_mv, n_hops, hop_mv = 4, 10, 3, 5
    starts = build_starts(n, rng, fallback)
    if len(starts) > n_starts:
        starts = starts[:n_starts]
    gbest, gs = fallback.copy(), fb_score
    for st in starts:
        sc = exact_score(st, n)
        if sc < gs - 1e-13:
            gs, gbest = sc, st.copy()
    for st in starts:
        ht, st_sc = local_search(st, n, max_mv, rng)
        if st_sc < gs - 1e-13:
            gs, gbest = st_sc, ht.copy()
    walker, wsc = gbest.copy(), gs
    scale = np.clip(np.sqrt(47.0 / n), 0.5, 1.5)
    for hop in range(n_hops):
        sd = (0.04, 0.10, 0.20, 0.30)[hop % 4] * scale
        prop = project_capped(walker + rng.normal(0, sd, n), n)
        if prop is None:
            continue
        hp, hsc = local_search(prop, n, hop_mv, rng)
        delta = hsc - wsc
        T = 0.0015 * (0.25 + 0.75 * (1.0 - hop / max(1, n_hops)))
        accept = delta <= 1e-12 or rng.random() < np.exp(-min(700.0, delta / T))
        if accept:
            walker, wsc = hp.copy(), hsc
        if hsc < gs - 1e-13:
            gs, gbest = hsc, hp.copy()
    final = project_capped(gbest, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    if exact_score(final, n) > fb_score + 1e-12:
        final = fallback
    fsc = exact_score(final, n)
    if not np.isfinite(fsc):
        final = np.full(n, 0
