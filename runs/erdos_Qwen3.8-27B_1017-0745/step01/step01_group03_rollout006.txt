def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0 or not np.all(np.isfinite(v)):
        return None
    lo = float(v.min()) - 1.0
    hi = float(v.max())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if resid > 1e-15:
        order = np.argsort(1.0 - y)
        for i in order:
            add = min(resid, 1.0 - y[i])
            y[i] += add
            resid -= add
            if resid < 1e-15:
                break
    elif resid < -1e-15:
        order = np.argsort(y)
        for i in order:
            sub = min(-resid, y[i])
            y[i] -= sub
            resid += sub
            if resid > -1e-15:
                break
    y = np.clip(y, 0.0, 1.0)
    r2 = S - np.sum(y)
    if abs(r2) > 0:
        if r2 > 0 and y[0] < 1.0:
            y[0] += min(r2, 1.0 - y[0])
        elif r2 < 0 and y[0] > 0.0:
            y[0] += max(r2, -y[0])
    if abs(np.sum(y) - S) > 1e-10 or np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    return np.clip(y, 0.0, 1.0)

def feasible(h, n):
    if h is None or h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    return np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and abs(np.sum(h) - n / 2.0) <= 1e-10

def exact_lags(h):
    return np.correlate(h, 1.0 - h, mode='full')

def exact_score(h):
    n = len(h)
    if not feasible(h, n):
        return float('inf')
    c = exact_lags(h)
    s = float(c.max()) * (2.0 / n)
    return s if np.isfinite(s) else float('inf')

def pair_delta(h, p, q, a, n):
    DQ = np.zeros(2 * n - 1)
    apq = a * a
    for k in range(-(n - 1), n):
        d = 0.0
        j = p - k
        if 0 <= j < n:
            d += a * (1.0 - h[j])
        j = q - k
        if 0 <= j < n:
            d -= a * (1.0 - h[j])
        j = p + k
        if 0 <= j < n:
            d -= h[j] * a
        j = q + k
        if 0 <= j < n:
            d += h[j] * a
        if k == 0:
            d -= 2.0 * apq
        elif k == p - q:
            d += apq
        elif k == q - p:
            d += apq
        DQ[k + n - 1] = d
    return DQ

def greedy_polish(h_in, n, rng):
