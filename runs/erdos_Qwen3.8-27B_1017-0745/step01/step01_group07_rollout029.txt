    except Exception:
        return None

def exact_lags(h, n):
    return (np.correlate(h, 1.0 - h, mode='full') * (2.0 / n))[::-1]

def line_obj(t, c, b, a):
    return float(np.max(c + b * t + a * t * t))

def pair_line_search(h, q, p, r, n):
    dx = 2.0 / n
    L = max(-h[p], h[r] - 1.0)
    U = min(1.0 - h[p], h[r])
    if U - L <= 1e-14:
        return None
    c = q
    a = np.zeros(2 * n - 1)
    a[n - 1] = -2.0 * dx
    a[n - 1 + r - p] = dx
    a[n - 1 + p - r] = dx
    b = np.zeros(2 * n - 1)
    for d in range(-(n - 1), n):
        idx = n - 1 + d
        s1 = 0.0; s2 = 0.0
        if 0 <= p + d < n: s1 += 1.0 - h[p + d]
        if 0 <= r + d < n: s1 -= 1.0 - h[r + d]
        if 0 <= p - d < n: s2 += h[p - d]
        if 0 <= r - d < n: s2 -= h[r - d]
        b[idx] = dx * (s1 - s2)
    best_t = 0.0
    best_f = line_obj(0.0, c, b, a)
    ts = np.linspace(L, U, 17)
    for t in ts:
        f = line_obj(t, c, b, a)
        if f < best_f - 1e-15 or (abs(f - best_f) <= 1e-15 and abs(t) < abs(best_t)):
            best_f = f; best_t = t
    rad = (U - L) / 16.0
    for _ in range(4):
        lo2 = max(L, best_t - rad)
        hi2 = min(U, best_t + rad)
        for t in np.linspace(lo2, hi2, 5):
            f = line_obj(t, c, b, a)
            if f < best_f - 1e-15 or (abs(f - best_f) <= 1e-15 and abs(t) < abs(best_t)):
                best_f = f; best_t = t
        rad *= 0.25
    cur_max = float(np.max(c))
    if best_f >= cur_max - 1e-13:
        return None
    h_new = h.copy()
    h_new[p] += best_t
    h_new[r] -= best_t
    if not np.all(h_new >= -1e-12) or not np.all(h_new <= 1.0 + 1e-12):
        h_new = project_capped_simplex(h_new, n / 2.0)
        if h_new is None:
            return None
    q_new = exact_lags(h_new, n)
    sc_new = float(np.max(q_new))
    if sc_new >= cur_max - 1e-13:
        return None
    return h_new, sc_new, q_new

def build_starts(n, S, parent, rng):
    starts = []
    if parent is not None:
        starts.append(parent.copy())
        starts.append(parent[::-1].copy())
        starts.append(1.0 - parent.copy())
    M = min(4, (n - 1) // 2)
    idx = np.arange(n, dtype=np.float64)
    u = np.full(n, 0.5)
    for m in range(1, M + 1):
        phi = rng.uniform(0, 2 * np.pi)
        amp = rng.normal(0, 0.12 / np.sqrt(m))
        u += amp * np.cos(2 * np.pi * m * (idx + 0.5) / n + phi)
    p = project_capped_simplex(u, S)
    if p is not None:
        starts.append(p)
    perm = rng.permutation(n)
    bv = np.zeros(n)
    bv[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        bv[perm[n // 2]] = 0.5
    starts.append(bv.copy())
    ur = rng.random(n)
    p2 = project_capped_simplex(ur, S)
    if p2 is not None:
        starts.append(p2)
    if parent is None:
        starts.append(np.full(n, 0.5))
    out = []
    for s in starts:
        dup = False
        for d in out:
            if np.max(np.abs(s - d)) < 1e-10:
                dup = True; break
        if not dup and is_feasible(s, n):
            out.append(s)
    return out

def polish_sweep(h, q, score, n, rng, max_sweeps):
    dx = 2.0 / n
    P = min(n * (n - 1) // 2, 2 * n, max(24, 80000 // n))
    pairs = []
    nadj = min(n - 1, max(1, P // 2))
    if nadj <= n - 1:
        adj_idx = np.linspace(0, n - 2, nadj, dtype=int).tolist()
    else:
        adj_idx = list(range(n - 1))
    for i in adj_idx:
        pairs.append((i, i + 1))
    attempts = 0
    while len(pairs) < P and attempts < 300:
        i = int(rng.integers(0, n)); j = int(rng.integers(0, n))
        if i != j:
            found = False
            for pp in pairs:
                if (pp[0] == i and pp[1] == j) or (pp[0] == j and pp[1] == i):
                    found = True; break
            if not found:
                pairs.append((i, j))
        attempts += 1
    order = rng.permutation(len(pairs))
    h_cur = h.copy(); q_cur = q.copy()
    for sweep in range(max_sweeps):
        accepted = 0
        for oi in order:
            p, r = pairs[oi]
            res = pair_line_search(h_cur, q_cur, p, r, n)
            if res is not None:
                h_cur, sc_cur, q_cur = res
                accepted += 1
        if accepted == 0:
            break
    sc_final = float(np.max(q_cur))
    return h_cur, sc_final, q_cur

def anneal_trajectory(h0, n, dx, rng):
    h = h0.copy()
    sc = exact_score(h, n)
    if sc is None:
        return h0.copy(), 0.5
    best_h = h.copy(); best_sc = sc
    E = 5 if n <= 128 else (4 if n <= 256 else 3)
    K = min(1000, max(80, 80000 // n))
    scale = 0.12
    for e in range(E):
        T = max(5e-5, 3e-3 * (0.4 ** e))
        accepted = 0
        for _ in range(K):
            p = int(rng.integers(0, n)); r = int(rng.integers(0, n))
            if p == r:
                continue
            t = float(rng.uniform(-scale, scale))
            L = max(-h[p], h[r] - 1.0)
            U = min(1.0 - h[p], h[r])
            if t < L - 1e-15 or t > U + 1e-15:
                continue
            t = max(L, min(U, t))
            h_new = h.copy()
            h_new[p] += t; h_new[r] -= t
            if not np.all(h_new >= -1e-12) or not np.all(h_new <= 1.0 + 1e-12):
                h_new = project_capped_simplex(h_new, n / 2.0)
                if h_new is None:
                    continue
            sc_new = exact_score(h_new, n)
            if sc_new is None:
                continue
            if sc_new < sc - 1e-15:
                h = h_new; sc = sc_new; accepted += 1
                if sc < best_sc:
                    best_h = h.copy(); best_sc = sc
            else:
                delta = sc_new - sc
                if delta < 700.0 * T:
                    if float(rng.random()) < float(np.exp(-delta / T)):
                        h = h_new; sc = sc_new; accepted += 1
        ratio = accepted / K
        if ratio < 0.12:
            scale = min(0.4, 1.25 * scale)
        elif ratio > 0.55:
            scale = max(0.01, 0.75 * scale)
        q = exact_lags(h, n)
        h, sc, q = polish_sweep(h, q, sc, n, rng, 1)
        if sc < best_sc:
            best_h = h.copy(); best_sc = sc
    q = exact_lags(best_h, n)
    max_sw = 4 if n <= 128 else (3 if n <= 256 else 2)
    best_h, best_sc, _ = polish_sweep(best_h, q, best_sc, n, rng, max_sw)
    return best_h, best_sc

def run(seed=42, **kwargs):
    if seed is None:
        seed = 42
    try:
        seed = int(seed) % (2 ** 32)
    except (TypeError, ValueError):
        seed = 42
    rng = np.random.default_rng(seed)
    n = None
    parent = None
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        try:
            iv_arr = np.asarray(iv, dtype=np.float64)
            if iv_arr.ndim == 1 and 2 <= iv_arr.size <= 999:
                if np.all(np.isfinite(iv_arr)) and np.max(np.abs(iv_arr)) <= 1e12:
                    n = iv_arr.size
                    S = n / 2.0
                    if np.all(iv_arr >= 0) and np.all(iv_arr <= 1) and abs(np.sum(iv_arr) - S) <= 1e-12:
                        parent = iv_arr.copy()
                    else:
                        pr = project_capped_simplex(iv_arr, S)
                        if pr is not None and is_feasible(pr, n):
                            parent = pr
        except Exception:
            pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 80))
        except (TypeError, ValueError):
            n = 80
        if n < 8 or n > 999:
            n = 80
    S = n / 2.0
    dx = 2.0 / n
    h_const = np.full(n, 0.5)
    best_h = h_const.copy()
    best_sc = 0.5
    sc0 = exact_score(h_const, n)
    if sc0 is not None:
        best_sc = sc0
    if parent is not None:
        sp = exact_score(parent, n)
        if sp is not None and sp < best_sc:
            best_h = parent.copy(); best_sc = sp
    starts = build_starts(n, S, parent, rng)
    for s in starts:
        ss = exact_score(s, n)
        if ss is not None and ss < best_sc:
            best_h = s.copy(); best_sc = ss
    traj_cap = 6 if n <= 128 else (4 if n <= 256 else 3)
    for s in starts[:traj_cap]:
        try:
            th, tsc = anneal_trajectory(s, n, dx, rng)
            if tsc < best_sc:
                best_h = th.copy(); best_sc = tsc
        except Exception:
            pass
    q_best = exact_lags(best_h, n)
    max_sw = 4 if n <= 128 else (3 if n <= 256 else 2)
    fh, fsc, _ = polish_sweep(best_h, q_best, best_sc, n, rng, max_sw)
    if fsc < best_sc:
        best_h = fh.copy(); best_sc = fsc
    if not is_feasible(best_h, n):
        best_h = project_capped_simplex(best_h, S)
        if best_h is None:
            best_h = h_const.copy()
    final_sc = exact_score(best_h, n)
    if final_sc is None:
        final_sc = 0.5
    return best_h, final_sc, n
