            sig = min(0.30, 1.20 * sig)
    return bh, bf

def zt_polish(h, n, S, dx, rng, sigs, cnt):
    f = exact_score(h, dx)
    for s in sigs:
        for _ in range(cnt):
            t = make_prop(h, n, S, s, rng)
            if t is not None:
                fn = exact_score(t, dx)
                if fn < f - 1e-12:
                    h, f = t.copy(), fn
    return h, f

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    init = globals().get('initial_h_values', None)
    nd = 45
    nd_kw = kwargs.get('n_points', None)
    if nd_kw is not None and isinstance(nd_kw, (int, np.integer)) and not isinstance(nd_kw, bool) and 2 <= int(nd_kw) <= 999:
        nd = int(nd_kw)
    parent_ok = False
    parent_arr = None
    if init is not None:
        try:
            parent_arr = np.array(init, dtype=np.float64).ravel()
        except Exception:
            parent_arr = None
        if parent_arr is not None and 2 <= len(parent_arr) <= 999 and np.all(np.isfinite(parent_arr)) and parent_arr.min() >= -1e-8 and parent_arr.max() <= 1.0 + 1e-8 and abs(parent_arr.sum() - len(parent_arr) / 2.0) <= 1e-6:
            parent_ok = True
            nd = len(parent_arr)
    n = nd
    S = n / 2.0
    dx = 2.0 / n
    if parent_ok:
        fb = project_K(parent_arr, n, S)
    else:
        fb = np.full(n, 0.5)
    if not is_feas(fb, n, S):
        fb = np.full(n, S / n)
    fb_score = exact_score(fb, dx)
    best_h, best_f = fb.copy(), fb_score
    pool = []
    if parent_ok:
        add_to_pool(pool, parent_arr, n, S)
        add_to_pool(pool, 1.0 - parent_arr, n, S)
        add_to_pool(pool, parent_arr[::-1], n, S)
        add_to_pool(pool, 1.0 - parent_arr[::-1], n, S)
        add_to_pool(pool, parent_arr + 0.12 * rng.standard_normal(n), n, S)
        add_to_pool(pool, parent_arr + 0.28 * rng.standard_normal(n), n, S)
        add_to_pool(pool, parent_arr + 0.50 * rng.standard_normal(n), n, S)
    add_to_pool(pool, rng.uniform(0, 1, n), n, S)
    add_to_pool(pool, rng.beta(0.5, 0.5, n), n, S)
    bv = np.zeros(n)
    nb = int(np.floor(S))
    bv[:nb] = 1.0
    if S - nb > 1e-10:
        bv[nb] = S - nb
    bv = bv[rng.permutation(n)]
    add_to_pool(pool, bv, n, S)
    ph = float(rng.random() * 2.0 * pi)
    jarr = np.arange(n, dtype=np.float64)
    sv = 0.5 + 0.45 * np.cos(2.0 * pi * jarr / n + ph)
    add_to_pool(pool, sv, n, S)
    if not parent_ok:
        add_to_pool(pool, rng.uniform(0, 1, n), n, S)
        add_to_pool(pool, rng.beta(0.3, 0.3, n), n, S)
    if len(pool) == 0:
        pool.append(fb.copy())
    scores = np.array([exact_score(p, dx) for p in pool])
    order = np.argsort(scores)
    if n <= 64:
        max_starts, nlev, nprop, npilot, pol_cnt = 8, 16, 90, 32, 80
    elif n <= 128:
        max_starts, nlev, nprop, npilot, pol_cnt = 5, 10, 40, 20, 40
    elif n <= 256:
        max_starts, nlev, nprop, npilot, pol_cnt = 3, 8, 18, 12, 20
    else:
        max_starts, nlev, nprop, npilot, pol_cnt = 2, 6, 6, 8, 8
    pol_sigs = [0.025, 0.008]
    start_count = min(max_starts, len(pool))
    for si in range(start_count):
        idx = int(order[si])
        s = pool[idx]
        try:
            bh, bf = anneal_chain(s, n, S, dx, rng, nlev, nprop, npilot)
            if is_feas(bh, n, S) and bf < best_f - 1e-12:
                best_h, best_f = bh, bf
        except Exception:
            pass
    try:
        best_h, best_f = zt_polish(best_h, n, S, dx, rng, pol_sigs, pol_cnt)
    except Exception:
        pass
    for _ in range(4):
        sig2 = 0.05 + 0.10 * rng.random()
        hp2 = project_K(best_h + sig2 * rng.standard_normal(n), n, S)
        try:
            bh2, bf2 = anneal_chain(hp2, n, S, dx, rng, max(6, nlev // 2), max(30, nprop // 2), max(8, npilot // 2))
            if is_feas(bh2, n, S) and bf2 < best_f - 1e-12:
                best_h, best_f = bh2, bf2
            best_h, best_f = zt_polish(best_h, n, S, dx, rng, [0.015], pol_cnt)
        except Exception:
            pass
    best_h = project_K(best_h, n, S)
    if not is_feas(best_h, n, S):
        best_h, best_f = fb.copy(), fb_score
    else:
        fs = exact_score(best_h, dx)
        if fs > fb_score + 1e-10:
            best_h, best_f = fb.copy(), fb_score
        else:
            best_f = fs
    best_h = np.clip(best_h, 0.0, 1.0)
    return (best_h, float(exact_score(best_h, dx)), int(n))
