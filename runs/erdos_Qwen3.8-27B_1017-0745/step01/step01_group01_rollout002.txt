Now for the lag values and gradients:

For lag d (from -(n-1) to n-1):
- lo = max(0, -d), hi = min(n, n-d)
- v[d+n-1] = sum_{i=lo}^{hi-1} h[i] * (1 - h[i+d])
- Grad: for each i in [lo, hi-1], j = i+d:
  - ∂f_d/∂h[i] += (1 - h[j])
  - ∂f_d/∂h[j] += (-h[i])

Wait, for d=0: j = i, so both terms contribute to the same index:
∂f_0/∂h[i] = (1 - h[i]) + (-h[i]) = 1 - 2*h[i]

That's correct.

Let me code this up:

```python
def lag_values_and_grads(h, n):
    n_lags = 2 * n - 1
    v = np.zeros(n_lags)
    G = np.zeros((n_lags, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        m = hi - lo
        if m <= 0:
            continue
        i_idx = np.arange(lo, hi)
        j_idx = i_idx + d
        v[row] = np.dot(h[i_idx], 1.0 - h[j_idx])
        G[row, i_idx] += (1.0 - h[j_idx])
        G[row, j_idx] -= h[i_idx]
    return v, G
```

This is O(n^2) total which is fine for n=47.

Now the pairwise polish:
- For each pair (i, j) with i < j:
- Transfer δ: h[i] += δ, h[j] -= δ
- Feasible: max(-h[i], h[j]-1) <= δ <= min(1-h[i], h[j])
- Try δ = 0, lower, upper, and ±{0.001, 0.003, 0.01, 0.03, 0.1, 0.3} clipped to feasible range
- Accept best if improves score by > 1e-12

import numpy as np
from scipy.optimize import minimize

def project_capped(z, n):
    s = n / 2.0
    if not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if hi - lo < 1e-15:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(n + 2):
        if abs(r) < 1e-13 * max(1, n):
            break
        if r > 0:
            for i in range(n):
                if r <= 0:
                    break
                amt = min(r, 1.0 - h[i])
                h[i] += amt
                r -= amt
        else:
            for i in range(n):
                if r >= 0:
                    break
                amt = max(r, -h[i])
                h[i] += amt
                r -= amt
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 5e-12 * max(1, n):
        return None
    return h

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return float('inf')
    if abs(float(np.sum(h)) - n / 2.0) > 1e-9 * max(1, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

        G[row, j_idx] -= h[i_idx]

def slsqp_ineq(z, n):

def slsqp_ineq_jac(z, n):

def pairwise_polish(h, n, sweeps):
    for _ in range(sweeps):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                lo = max(-h[i], h[j] - 1.0)
                hi = min(1.0 - h[i], h[j])
                if lo > hi:
                    continue
                deltas = [0.0, lo, hi]
                for a in (0.001, 0.003, 0.01, 0.03, 0.1, 0.3):
                    for sgn in (1, -1):
                        d = sgn * a
                        d = max(lo, min(hi, d))
                        deltas.append(d)
                cur_score = exact_score(h, n)
                best_d, best_s = 0.0, cur_score
                for d in deltas:
                    h2 = h.copy()
                    h2[i] += d
                    h2[j] -= d
                    s2 = exact_score(h2, n)
                    if np.isfinite(s2) and s2 < best_s - 1e-12:
                        best_s, best_d = s2, d
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    improved = True
        if not improved:
            break
    return h
