def pair_delta_lags(h, p, q, a, n):
    # h: current state, p<q, a: change to h[p], -a to h[q]
    # returns DeltaQ array of length 2*n-1
    d = np.zeros(n)
    d[p] = a
    d[q] = -a
    DeltaQ = np.zeros(2*n-1)
    for k in range(-(n-1), n):
        idx = k + n - 1
        val = 0.0
        # First sum: d_i * (1-h[i-k]) for i in {p,q} where i-k in [0,n)
        for r, dr in ((p, a), (q, -a)):
            j = r - k
            if 0 <= j < n:
                val += dr * (1.0 - h[j])
        # Second sum: -h[i] * d_{i-k} for i in {p+k, q+k} where i in [0,n)
        for r, dr in ((p, a), (q, -a)):
            i = r + k
            if 0 <= i < n:
                val += -h[i] * dr
        # Third sum: -d_i * d_{i-k} for i where both i and i-k in {p,q}
        for i, di in ((p, a), (q, -a)):
            j = i - k
            if 0 <= j < n:
                dj = a if j == p else (-a if j == q else 0.0)
                if dj != 0.0:
                    val += -di * dj
        DeltaQ[idx] = val
    return DeltaQ

import numpy as np

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5*(lo+hi)
        y = np.clip(v-mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v-0.5*(lo+hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if resid > 1e-14:
        for i in range(n):
            if resid <= 1e-15: break
            add = min(resid, 1.0-y[i])
            y[i] += add; resid -= add
    elif resid < -1e-14:
        for i in range(n):
            if resid >= -1e-15: break
            sub = min(-resid, y[i])
            y[i] -= sub; resid += sub
    y = np.clip(y, 0.0, 1.0)
    if abs(np.sum(y)-S) > 5e-12*max(1.0,float(n)):
        return None
    return y

    return True

def pair_delta(h, p, q, a, n):
    dQ = np.zeros(2*n-1)
    for k in range(-(n-1), n):
        idx = k+n-1
        val = 0.0
        j = p-k
        if 0 <= j < n: val += a*(1.0-h[j])
        j = q-k
        if 0 <= j < n: val += -a*(1.0-h[j])
        i = p+k

        i = q+k
        if 0 <= i < n: val += h[i]*a
        j = p-k

            elif j == q: val += a*a
        j = q-k
        if 0 <= j < n:
            if j == q: val += -a*a
            elif j == p: val += a*a
