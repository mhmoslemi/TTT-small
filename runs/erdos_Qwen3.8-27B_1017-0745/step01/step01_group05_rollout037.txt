import numpy as np

def project_box_sum(y, n):
    S = n / 2.0
    if not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        c = np.clip(y - mid, 0.0, 1.0)
        if c.sum() > S:
            lo = mid
        else:
            hi = mid
    c = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(50):
        diff = S - c.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = c < 1.0 - 1e-14
        else:
            mask = c > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        c[mask] += diff / cnt
        np.clip(c, 0.0, 1.0, out=c)
    return c

def exact_score(h):
    n = len(h)
    if not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def check_feasible(h, n):

            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(h.sum() - n / 2.0) <= 1e-10)

def exchange_step(h, i, j, lags, n):
    """Try pair swap (i,j), return (new_h, new_lags, score) or None."""
    result = pair_scan(h, i, j, lags, n)

        return None

    new_h[i] += best_t
    new_h[j] -= best_t

    new_lags = lag_scores(new_h)
    score = float(np.max(new_lags))
    if score >= float(np.max(lags)) - 1e-12:

def basin_hopping(h0, n, rng, n_prop=18):
    """Metropolis basin hopping."""
    h_cur = h0.copy()
    s_cur = exact_score(h_cur)
    best_h = h_cur.copy()
    best_s = s_cur
    sigma = 0.1
    temps = [0.0015, 0.0008, 0.0003]
    block_lens = [8, 6, 4]
    prop_idx = 0
    for blk, (T, blen) in enumerate(zip(temps, block_lens)):
        accepted = 0
        for _ in range(blen):
            if prop_idx >= n_prop:
                break
            prop_idx += 1
            z = rng.standard_normal(n)
            z -= z.mean()
            cand = project_box_sum(h_cur + sigma * z, n)
            if not check_feasible(cand, n):
                continue
            cand_h, cand_s = polish(cand, n, rng, max_sweeps=1, pair_budget=min(n*(n-1)//2, max(8, 3*n//2)))
            delta = cand_s - s_cur
            if delta < -1e-14:
                h_cur, s_cur = cand_h, cand_s
                accepted += 1
            elif rng.random() < math.exp(-min(700.0, delta / T)):
                h_cur, s_cur = cand_h, cand_s
                accepted += 1
            if s_cur < best_s - 1e-14:
                best_s = s_cur
                best_h = h_cur.copy()
        rate = accepted / blen if blen > 0 else 0
        if rate > 0.6:
            sigma = min(sigma * 1.3, 0.3)
        elif rate < 0.2:
            sigma = max(sigma * 0.7, 0.02)
    return best_h, best_s
