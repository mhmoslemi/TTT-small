    n = int(n)
    tau = float(tau)
    if h.ndim != 1 or h.size != n or tau <= 0.0 or not math.isfinite(tau) or not np.all(np.isfinite(h)):
        return float('inf'), None
    q = (2.0 / float(n)) * pconv(h[::-1], 1.0 - h)
    if q.size != 2 * n - 1 or not np.all(np.isfinite(q)):
        return float('inf'), None
    qm = float(np.max(q))
    e = np.exp((q - qm) / tau)
    s = float(np.sum(e))
    if not math.isfinite(s) or s <= 0.0:
        return float('inf'), None
    return qm + tau * math.log(s), e / s

def smgrad(h, p, n):
    h = np.asarray(h, dtype=np.float64)
    n = int(n)
    if h.size != n or p is None:
        return np.zeros(n, dtype=np.float64)
    A = pconv(p[::-1], 1.0 - h)
    B = pconv(p, h)
    if A.size < 2 * n - 1 or B.size < 2 * n - 1:
        return np.zeros(n, dtype=np.float64)
    g = (2.0 / float(n)) * (A[n - 1:2 * n - 1] - B[n - 1:2 * n - 1])
    return g if np.all(np.isfinite(g)) else np.zeros(n, dtype=np.float64)

def sobj(h, tau, n):
    v, _ = smval(h, tau, n)
    return v if math.isfinite(v) else 1e8

def sjac(h, tau, n):
    _, p = smval(h, tau, n)
    return np.zeros(int(n), dtype=np.float64) if p is None else smgrad(h, p, int(n))

def seq(h, n):
    return float(np.sum(h) - 0.5 * int(n))

def seqj(h, n):
    return np.ones(int(n), dtype=np.float64)

def slsq_s(h, n, tau, it):
    h = clean(h, n)
    if h is None:
        return None
    n = int(n)
    try:
        r = minimize(sobj, h.copy(), jac=sjac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': seq, 'jac': seqj, 'args': (n,)}],
                     args=(float(tau), n),
                     options={'maxiter': int(it), 'ftol': 1e-12, 'disp': False})
        if r is not None and getattr(r, 'x', None) is not None:
            return clean(r.x, n)
    except Exception:
        pass
    return None

def eQ(h, n):
    h = np.asarray(h, dtype=np.float64)
    n = int(n)
    if h.size != n or not np.all(np.isfinite(h)):
        return np.zeros(2 * n - 1, dtype=np.float64)
    return pconv(h[::-1], 1.0 - h)

def eG(h, n):
    n = int(n)
    h = np.asarray(h, dtype=np.float64)
    G = np.zeros((2 * n - 1, n), dtype=np.float64)
    if h.size != n or not np.all(np.isfinite(h)):
        return G
    for k in range(2 * n - 1):
        s = k - (n - 1)
        lo = max(0, -s)
        hi = min(n, n - s)
        if hi > lo:
            G[k, lo:hi] += 1.0 - h[lo + s:hi + s]
            G[k, lo + s:hi + s] -= h[lo:hi]
    return G

def eob(z, n):
    t = float(z[-1])
    return t if math.isfinite(t) else 1e6

def eog(z, n):
    g = np.zeros(int(n) + 1, dtype=np.float64)
    g[-1] = 1.0
    return g

def ein(z, n):
    n = int(n)
    q = eQ(z[:n], n)
    return z[-1] - (2.0 / float(n)) * q

def eij(z, n):
    n = int(n)
    G = eG(z[:n], n)
    J = np.zeros((2 * n - 1, n + 1), dtype=np.float64)
    J[:, :n] = -(2.0 / float(n)) * G
    J[:, n] = 1.0
    return J if np.all(np.isfinite(J)) else np.zeros((2 * n - 1, n + 1), dtype=np.float64)

def ee(z, n):
    n = int(n)
    return float(np.sum(z[:n]) - 0.5 * n)

def eej(z, n):
    g = np.zeros(int(n) + 1, dtype=np.float64)
    g[:int(n)] = 1.0
    return g

def epi(h, n, it):
    h = clean(h, n)
    if h is None:
        return None
    n = int(n)
    q = eQ(h, n)
    t = (float(np.max(q)) * 2.0 / float(n) + 1e-4) if q.size else 0.3
    if not math.isfinite(t):
        t = 0.3
    t = min(max(t, 1e-6), 1.0)
    z = np.append(h, t)
    cons = [{'type': 'ineq', 'fun': ein, 'jac': eij, 'args': (n,)},
            {'type': 'eq', 'fun': ee, 'jac': eej, 'args': (n,)}]
    try:
        r = minimize(eob, z, jac=eog, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n + [(0.0, 2.0)],
                     constraints=cons, args=(n,),
                     options={'maxiter': int(it), 'ftol': 1e-12, 'disp': False})
        if r is not None and getattr(r, 'x', None) is not None:
            h2 = clean(r.x[:n], n)
            if h2 is not None:
                return h2
    except Exception:
        pass
    return h

def add(st, h, n):
    p = clean(h, n)
    if p is None:
        return
    for q in st:
        if float(np.max(np.abs(p - q))) < 1e-12:
            return
    st.append(p)

def quant(v, n):
    v = np.asarray(v, dtype=np.float64)
    n = int(n)
    if v.size != n:
        return None
    t = 0.5 * n
    k = int(math.floor(t))
    r = t - k
    q = np.zeros(n, dtype=np.float64)
    if k > 0:
        q[np.argsort(-v)[:k]] = 1.0
    if r > 1e-12 and k < n:
        q[np.argsort(-v)[k]] = r
    return q

def build(par, n, rng):
    st = []
    add(st, par, n)
    add(st, 1.0 - par, n)
    add(st, par[::-1], n)
    add(st, 0.5 * (par + par[::-1]), n)
    add(st, 0.5 * (par + (1.0 - par[::-1])), n)
    sq = np.zeros(n, dtype=np.float64)
    sq[:n // 2] = 1.0
    add(st, sq, n)
    q1 = quant(par, n)
    q2 = quant(1.0 - par, n)
    q3 = quant(par[::-1], n)
    if q1 is not None: add(st, q1, n)
    if q2 is not None: add(st, q2, n)
    if q3 is not None: add(st, q3, n)
    for sd in (0.03, 0.06, 0.12, 0.22):
        if len(st) >= 18:
            break
        add(st, par + sd * rng.standard_normal(n), n)
    for _ in range(2):
        if len(st) >= 18:
            break
        u = rng.random(n)
        add(st, u, n)
        q = quant(u, n)
        if q is not None:
            add(st, q, n)
    i = np.arange(n, dtype=np.float64)
    for m in range(1, 4):
        for ph in (0.0, 1.0, 2.0):
            if len(st) >= 18:
                break
            u = 0.5 + 0.32 * np.cos(2.0 * math.pi * m * i / n + ph) + \
                0.14 * np.sin(2.0 * math.pi * (m + 1) * i / n + 0.6 * ph + 0.5)
            add(st, u, n)
        if len(st) >= 18:
            break
    for m in range(1, 3):
        if len(st) >= 18:
            break
        add(st, 0.5 + 0.75 * np.sin(2.0 * math.pi * m * i / n + 0.5), n)
    return st

def pstage(h_in, n, tau, it):
    n = int(n)
    h = clean(h_in, n)
    if h is None:
        return None, None, float('inf')
    bh = h.copy()
    bc = score(h, n)
    F, p = smval(h, tau, n)
    if p is None:
        return h, bh, bc
    g = smgrad(h, p, n)
    alpha = 1.0
    small = 0
    for ii in range(int(it)):
        if float(np.max(np.abs(g))) < 1e-13:
            break
        ok = False
        for _ in range(22):
            c = project_box_sum(h - alpha * g, 0.5 * n)
            if c is None:
                break
            d = c - h
            if not np.all(np.isfinite(d)) or float(np.max(np.abs(d))) <= 1e-12:
                break
            gd = float(np.dot(g, d))
            if gd >= -1e-13:
                break
            Ft, p2 = smval(c, tau, n)
            if math.isfinite(Ft) and Ft <= F + 1e-4 * gd + 1e-12 * max(1.0, abs(F)):
                F_old = F
                go = g
                h = c
                F = Ft
                p = p2
                g = smgrad(h, p, n)
                if not np.all(np.isfinite(g)):
                    g = np.zeros(n, dtype=np.float64)
                dy = float(np.dot(d, g - go))
                if dy > 1e-14:
                    a2 = float(np.dot(d, d)) / dy
                    if math.isfinite(a2) and a2 > 0.0:
                        alpha = min(10.0, max(1e-4, a2))
                    else:
                        alpha = 0.5
                else:
                    alpha = min(10.0, 1.5 * alpha)
                if abs(F - F_old) < 1e-12:
                    small += 1
                else:
                    small = 0
                ok = True
                break
            alpha *= 0.5
            if alpha < 1e-10:
                break
        if not ok:
            break
        if small >= 8:
            break
        if (ii + 1) % 10 == 0:
            c = score(h, n)
            if c < bc:
                bc = c
                bh = h.copy()
    c = score(h, n)
    if c < bc:
        bc = c
        bh = h.copy()
    return h, bh, bc

def basin(best, bc, n, rng):
    n = int(n)
    h = clean(best, n)
    if h is None:
        return best, bc
    bc = score(h, n)
    if n <= 100:
        T = (0.03, 0.008, 0.002, 0.0006)
        I = (40, 40, 30, 20)
    elif n <= 300:
        T = (0.03, 0.008, 0.002)
        I = (30, 25, 20)
    else:
        T = (0.03, 0.008)
        I = (20, 15)
    cur = h
    for tau, it in zip(T, I):
        cur, sh, sc = pstage(cur, n, tau, it)
        if sh is not None and sc < bc - 1e-14:
            bc = sc
            best = sh.copy()
    sig = (0.05, 0.12, 0.25) if n <= 200 else (0.12,)
    for sg in sig:
        cur = clean(best + sg * rng.standard_normal(n), n)
        if cur is None:
            continue
        for tau, it in zip(T[:-1], I[:-1]):
            cur, sh, sc = pstage(cur, n, tau, it)
            if sh is not None and sc < bc - 1e-14:
                bc = sc
                best = sh.copy()
            if cur is None:
                break
    return best, bc

def get_parent():
    hv = globals().get('initial_h_values', None)
    if hv is not None:
        ph = np.asarray(hv, dtype=np.float64)
        if (ph.ndim == 1 and 2 <= ph.size <= 2000 and np.all(np.isfinite(ph)) and
                np.all(ph >= -1e-6) and np.all(ph <= 1.0 + 1e-6)):
            p = project_box_sum(np.clip(ph, 0.0, 1.0), 0.5 * ph.size)
            if p is not None and feas(p, ph.size):
                return p, int(ph.size)
    return project_box_sum(np.full(45, 0.5, dtype=np.float64), 22.5), 45

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        try:
            rng = np.random.default_rng(42)
        except Exception:
            rng = np.random.RandomState(seed)

    par, n = get_parent()
    best = par.copy()
    bc = score(best, n)

    starts = build(par, n, rng)
    so = []
    for idx, h in enumerate(starts):
        c = score(h, n)
        so.append((c, idx))
        if c < bc - 1e-14:
            bc = c
            best = h.copy()
    so.sort()

    if n <= 100:
        T = (0.08, 0.02, 0.005, 0.0012)
        I = (70, 100, 140, 180)
    elif n <= 300:
        T = (0.08, 0.02, 0.005)
        I = (50, 80, 110)
    else:
        T = (0.08, 0.02)
        I = (35, 55)

    lim = 4 if n <= 100 else (2 if n <= 300 else 1)
    extra = []
    add(extra, best, n)
    add(extra, par, n)

    for _, idx in so[:lim]:
        cur = clean(starts[idx], n)
        if cur is None:
            continue
        for tau, it in zip(T, I):
            h = slsq_s(cur, n, tau, it)
            if h is None:
                h = cur
            c = score(h, n)
            if c < bc - 1e-14:
                bc = c
                best = h.copy()
            cur = h
        if len(extra) < 10:
            add(extra, cur, n)

    cur = best
    for tau, it in zip(T[-2:], I[-2:]):
        h = slsq_s(cur, n, tau, it)
        if h is None:
            h = cur
        c = score(h, n)
        if c < bc - 1e-14:
            bc = c
            best = h.copy()
        cur = h
    if len(extra) < 10:
        add(extra, cur, n)

    best, bc = basin(best, bc, n, rng)
    if len(extra) < 10:
        add(extra, best, n)

    ex = []
    for idx, h in enumerate(extra):
        c = score(h, n)
        ex.append((c, idx))
    ex.sort()

    pool = []
    add(pool, best, n)
    add(pool, par, n)
    for _, idx in ex[:2]:
        if len(pool) < 3:
            add(pool, extra[idx], n)
    if len(pool) < 3:
        cand = clean(best + 0.07 * rng.standard_normal(n), n)
        if cand is not None:
            add(pool, cand, n)

    ei = 220 if n <= 100 else (140 if n <= 300 else 80)
    for h in pool:
        h2 = epi(h, n, ei)
        if h2 is not None:
            c = score(h2, n)
            if c < bc - 1e-14:
                bc = c
                best = h2.copy()

    fb = clean(par, n)
    if fb is not None:
        fc = score(fb, n)
        if fc < bc:
            bc = fc
            best = fb

    fin = clean(best, n)
    if fin is None:
        fin = project_box_sum(np.full(n, 0.5, dtype=np.float64), 0.5 * n)
        if fin is None:
            fin = np.full(n, 0.5, dtype=np.float64)
    cf = score(fin, n)
    if not math.isfinite(cf):
        cf = score(fb if fb is not None else fin, n)
        if not math.isfinite(cf):
            cf = float('inf')
    return fin, float(cf), int(n)
