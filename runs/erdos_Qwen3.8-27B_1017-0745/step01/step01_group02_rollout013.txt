import numpy as np

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(v - tau, 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-14:
        for i in np.argsort(1.0 - y):
            if r <= 0:
                break
            d = min(r, 1.0 - y[i])
            y[i] += d
            r -= d
    elif r < -1e-14:
        for i in np.argsort(y):
            if r >= 0:
                break
            d = min(-r, y[i])
            y[i] -= d
            r += d
    if not np.all(np.isfinite(y)) or np.any(y < -1e-10) or np.any(y > 1.0 + 1e-10):
        return None
    if abs(float(np.sum(y)) - S) > 1e-10 * max(1, n):
        return None
    return y

def check_feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(float(np.sum(h)) - 0.5 * n) < 1e-10 * max(1, n))

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    s = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
    return s if np.isfinite(s) else float('inf')

def smooth_val_grad(h, n, tau):
    # Compute q_s = dx * c_s for all lags s
    c = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    dx = 2.0 / n
    q = c * dx  # length 2n-1, q[j] corresponds to lag s = j - (n-1)
    qmax = float(np.max(q))
    # Stable log-sum-exp
    exponents = (q - qmax) / tau
    # Clip for numerical stability
    exponents = np.clip(exponents, -500, 0)
    weights = np.exp(exponents)
    wsum = float(np.sum(weights))
    if wsum <= 0 or not np.isfinite(wsum):
        return qmax, None
    p = weights / wsum  # length 2n-1
    F = qmax + tau * np.log(wsum)
    # Gradient: grad[k] = dx * sum_j p[j] * [(1-h[k-s_j])*1[valid] - h[k+s_j]*1[valid]]
    # where s_j = j - (n-1)
    b = 1.0 - h
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        s = j - (n - 1)
        if p[j] < 1e-15:
            continue
        # Term 1: (1-h[k-s]) for valid k-s
        # k ranges 0..n-1, k-s ranges 0..n-1 => s <= k <= n-1+s
        lo_k = max(0, s)
        hi_k = min(n, n + s)
        for k in range(lo_k, hi_k):
            grad[k] += p[j] * b[k - s]
        # Term 2: -h[k+s] for valid k+s
        lo_k = max(0, -s)
        hi_k = min(n, n - s)
        for k in range(lo_k, hi_k):
            grad[k] -= p[j] * h[k + s]
    grad *= dx
    return F, grad

def project_box_sum(v, S):

    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None

def smooth_val_grad(h, n, tau):
    if tau <= 0:
        return float('inf'), None
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    q = c * dx
    qmax = float(np.max(q))

    if not np.all(np.isfinite(grad)):

    return F, grad
