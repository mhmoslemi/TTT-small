import numpy as np
from scipy.optimize import minimize

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)) or N < 1:
        return np.full(N, 0.5)
    z = z - np.median(z)
    lo, hi = z.min() - 1.0, z.max()
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    if r > 1e-14:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
            if r <= 1e-14:
                break
    elif r < -1e-14:
        for i in range(N):
            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1 + 1e-10:
        return np.full(N, 0.5)
    if abs(h.sum() - m) > 1e-8:
        return np.full(N, 0.5)
    return h

def exact_score(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    corr = np.correlate(h, 1.0 - h, mode='full')
    sc = float(np.max(corr) * 2.0 / N)
    return sc if np.isfinite(sc) else np.inf

def lag_data(h, N):
    L = 2 * N - 1
    q = np.zeros(L)
    G = np.zeros((L, N))
    for s in range(-(N-1), N):
        r = s + N - 1
        if s >= 0:
            ell = N - s
            a = h[:ell]
            b = h[s:s+ell]
            q[r] = np.dot(a, 1.0 - b)
            G[r, :ell] += (1.0 - b)
            G[r, s:s+ell] -= a
        else:
            d = -s
            ell = N - d
            a = h[d:]
            b = h[:ell]
            q[r] = np.dot(a, 1.0 - b)
            G[r, d:] += (1.0 - b)
            G[r, :ell] -= a
    return q, G

def qp_obj(z, x, m):
    y = z[:len(x)]
    return np.sum((y - x)**2) + z[len(x)]

def qp_obj_jac(z, x, m):
    n = len(x)
    g = np.zeros(n + 1)
    g[:n] = 2.0 * (z[:n] - x)
    g[n] = 1.0
    return g

def qp_ineq_fun(z, x, q, G, m):
    n = len(x)
    y = z[:n]
    return z[n] - q - G @ (y - x)

def qp_ineq_jac(z, x, q, G, m):
    n = len(x)
    L = len(q)
    J = np.zeros((L, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def qp_eq_fun(z, x, q, G, m):
    return np.array([np.sum(z[:len(x)]) - m])

def qp_eq_jac(z, x, q, G, m):
    n = len(x)
    J = np.zeros((1, n + 1))
    J[0, :n] = 1.0
    return J

def mm_qp_solve(x, N, m, q, G):
    t0 = max(np.max(q), 0.0)
    z0 = np.concatenate([x, [t0]])
    bounds = [(0.0, 1.0)] * N + [(None, None)]
    cons = [
        {'type': 'eq', 'fun': qp_eq_fun, 'jac': qp_eq_jac, 'args': (x, q, G, m)},
        {'type': 'ineq', 'fun': qp_ineq_fun, 'jac': qp_ineq_jac, 'args': (x, q, G, m)},
    ]
    try:
        res = minimize(qp_obj, z0, jac=qp_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       args=(x, q, G, m),
                       options={'maxiter': 200, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return x
    if res.x is not None and len(res.x) == N + 1 and np.all(np.isfinite(res.x[:N])):
        return project_box_sum(res.x[:N], m)
    return x

def mm_local(x, N, m, max_iter=30):
    h_cur = x.copy()
    s_cur = exact_score(h_cur, N)
    h_best = h_cur.copy()
    s_best = s_cur
    for _ in range(max_iter):
        q, G = lag_data(h_cur, N)
        if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
            break
        y_qp = mm_qp_solve(h_cur, N, m, q, G)
        if not np.all(np.isfinite(y_qp)):
            break
        # Line search along x -> y_qp
        best_alpha = 1.0
        best_s = s_cur
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            cand = project_box_sum((1 - alpha) * h_cur + alpha * y_qp, m)
            sc = exact_score(cand, N)
            if sc < best_s:
                best_s = sc
                best_alpha = alpha
                best_cand = cand
        if best_s < s_cur - 1e-12:
            h_cur = best_cand
            s_cur = best_s
            if s_cur < s_best:
                s_best = s_cur
                h_best = h_cur.copy()
            if np.max(np.abs(h_cur - y_qp)) <= 1e-8:
                pass  # still continue
        else:
            break
    return h_best

def make_starts(baseline, N, m, rng):
    starts = [baseline.copy()]
    comp = 1.0 - baseline
    starts.append(project_box_sum(comp, m))
    rev = baseline[::-1].copy()
    starts.append(project_box_sum(rev, m))
    sh = N // 3
    sh = max(1, min(sh, N - 1))
    starts.append(project_box_sum(np.roll(baseline, sh), m))
    for sc in [0.05, 0.15]:
        starts.append(project_box_sum(baseline + sc * rng.standard_normal(N), m))
    starts.append(project_box_sum(rng.uniform(0.0, 1.0, N), m))
    # Fourier
    for amp in [0.12, 0.20]:
        z = np.full(N, 0.5)
        idx = np.arange(N)
        R = min(6, (N-1) // 2)
        for r in range(1, R + 1):
            a = rng.standard_normal()
            b = rng.standard_normal()
            z += (amp / r) * (a * np.sin(2*np.pi*r*idx/N) + b * np.cos(2*np.pi*r*idx/N))
        starts.append(project_box_sum(z, m))
    # Deduplicate
    unique = []
    for s in starts:
        if len(s.shape) != 1 or s.size != N or not np.all(np.isfinite(s)):
            continue
        dup = False
        for u in unique:
            if np.max(np.abs(s - u)) <= 1e-10:
                dup = True
                break
        if not dup:
            unique.append(s)
    return unique[:8]

def run(seed=42, **kwargs):
    N = None
    raw = globals().get('initial_h_values', None)
    if raw is not None:
        try:
            raw = np.asarray(raw, dtype=np.float64)
        except Exception:
            raw = None
        if raw is not None and raw.ndim == 1 and raw.size > 1 and np.all(np.isfinite(raw)):
            N = raw.size
    if N is None:
        N = kwargs.get('n_points', 70)
        if not isinstance(N, (int, np.integer)) or isinstance(N, bool) or N < 2:
            N = 70
    m = N / 2.0
    if N == 1:
        h = np.array([0.5])
        return h, exact_score(h, 1), 1

    # Baseline
    if raw is not None and np.all(np.isfinite(raw)):
        baseline = project_box_sum(raw, m)
    else:
        baseline = np.full(N, 0.5)
    s_base = exact_score(baseline, N)

    rng = np.random.default_rng(seed)
    h_best = baseline.copy()
    s_best = s_base

    # Determine search parameters based on N
    if N <= 96:
        n_starts = 8
        mm_iters = 30
        n_restarts = 2
        restart_it = 20
        restart_sigs = [0.05, 0.015]
    elif N <= 160:
        n_starts = 5
        mm_iters = 20
        n_restarts = 1
        restart_it = 12
        restart_sigs = [0.04]
    else:
        n_starts = 3
        mm_iters = 10
        n_restarts = 0
        restart_sigs = []

    # Generate and evaluate starts
    starts = make_starts(baseline, N, m, rng)
    for s in range(n_starts):
        if s >= len(starts):
            break
        h0 = starts[s]
        sc = exact_score(h0, N)
        if sc < s_best - 1e-12:
            h_best = h0.copy()
            s_best = sc
        try:
            h_mm = mm_local(h0, N, m, mm_iters)
            sc_mm = exact_score(h_mm, N)
            if sc_mm < s_best - 1e-12:
                h_best = h_mm.copy()
                s_best = sc_mm
        except Exception:
            pass

    # Perturbation restarts
    for i in range(n_restarts):
        sig = restart_sigs[i]
        try:
            hop = project_box_sum(h_best + sig * rng.standard_normal(N), m)
            sc_hop = exact_score(hop, N)
            if sc_hop < s_best - 1e-12:
                h_best = hop.copy()
                s_best = sc_hop
            h_mm = mm_local(hop, N, m, restart_it)
            sc_mm = exact_score(h_mm, N)
            if sc_mm < s_best - 1e-12:
                h_best = h_mm.copy()
                s_best = sc_mm
        except Exception:
            pass

    # Final validation
    h_best = project_box_sum(h_best, m)
    s_best = exact_score(h_best, N)
    if not np.isfinite(s_best) or s_best >= s_base:
        h_best = baseline
        s_best = s_base
    h_best = project_box_sum(h_best, m)
    s_best = exact_score(h_best, N)
    if not np.isfinite(s_best):
        h_best = np.full(N, 0.5)
        s_best = exact_score(h_best, N)
    return h_best.astype(np.float64), float(s_best), int(N)

def project_box_sum(z, m):
    N = len(z)
    if N < 1 or not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    z = z - np.median(z)
    lo, hi = z.min() - 1.0, z.max()
    for _ in range(100):
        mid = 0.5 * (lo + hi)

            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
