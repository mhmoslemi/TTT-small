        pos = np.where(p > 0)[0]
        if len(pos) > 0:
            p[pos[0]] -= r
    return p

def dual_eval(p_vec, h, G, q, L, n):
    gmix = G.T @ p_vec
    y, ok = project_capped(h - gmix / L, n)
    if not ok:
        return None
    d = y - h
    u = q + G @ d
    D = float(p_vec @ u + 0.5 * L * d @ d)
    return d, u, D

def solve_majorant(h, q, G, n, max_inner):
    dx = 2.0 / n
    L = 2.0 * dx
    Ldual = max(1e-12, 1.15 * float(np.sum(G * G)) / L)
    m = len(q)
    qmax = float(q.max())
    tol_q = max(1e-12, 1e-5 * max(1.0, abs(qmax)))
    active = np.where(q >= qmax - tol_q)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(q))])
    x = np.zeros(m)
    x[active] = 1.0 / len(active)
    y = x.copy()
    t = 1.0
    best_d = None
    for it in range(max_inner):
        st = dual_eval(y, h, G, q, L, n)
        if st is None:
            break
        d_y, u_y, D_y = st
        if best_d is None:
            best_d = d_y.copy()
        grad_y = u_y
        x_trial = project_simplex(y + grad_y / Ldual)
        st2 = dual_eval(x_trial, h, G, q, L, n)
        if st2 is None:
            break
        if np.max(np.abs(x_trial - x)) < 1e-12 and it > 5:
            break
        t_new = (1.0 + math.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_trial + ((t - 1.0) / t_new) * (x_trial - x)
        x = x_trial.copy()
        t = t_new
    st_final = dual_eval(x, h, G, q, L, n)
    if st_final is not None and st_final[0] is not None:
        best_d = st_final[0]
    if best_d is None or not np.all(np.isfinite(best_d)):
        p_unit = np.zeros(m)
        p_unit[int(np.argmax(q))] = 1.0
        st = dual_eval(p_unit, h, G, q, L, n)
        best_d = st[0] if st is not None else np.zeros(n)
    return best_d

def optimize_start(h0, n, max_outer, max_inner):
    h, ok = project_capped(h0.copy(), n)
    if not ok:
        h = np.full(n, 0.5)
    s = exact_score(h, n)
    best_h = h.copy()
    best_s = s
    stall = 0
    for _ in range(max_outer):
        q = lag_scores(h, n)
        G = build_G(h, n)
        if not np.all(np.isfinite(G)):
            break
        d = solve_majorant(h, q, G, n, max_inner)
        if d is None or not np.all(np.isfinite(d)):
            break
        if np.max(np.abs(d)) < 1e-11:
            break
        L = 4.0 / n
        norm2 = float(d @ d)
        v = G @ d
        accepted = False
        for j in range(20):
            alpha = 1.0 / (1 << j)
            qmax = float(q.max())
            model_val = float(np.max(q + alpha * v) + 0.5 * L * alpha * alpha * norm2)
            if model_val >= qmax - 1e-14:
                continue
            h_trial, ok2 = project_capped(h + alpha * d, n)
            if not ok2:
                continue
            s_trial = exact_score(h_trial, n)
            if s_trial < best_s - 1e-13:
                h = h_trial
                s = s_trial
                best_h = h.copy()
                best_s = s
                stall = 0
                accepted = True
                break
        if not accepted:
            break
        if s > best_s + 1e-13:
            h = best_h.copy()
            s = best_s
        if best_s > stall:
            stall += 1
        else:
            stall = max(stall, 0)
        if stall >= 3:
            break
    return best_h, best_s

def slsqp_obj(x):
    return x[-1]

def slsqp_obj_jac(x):
    g = np.zeros(len(x))
    g[-1] = 1.0
    return g

def slsqp_ineq(x):
    n = len(x) - 1
    h = x[:n]
    return x[-1] - lag_scores(h, n)

def slsqp_ineq_jac(x):
    n = len(x) - 1
    h = x[:n]
    dx = 2.0 / n
    jac = np.zeros((2 * n - 1, n + 1))
    jac[:, -1] = 1.0
    for k in range(-(n - 1), n):
        row = k + n - 1
        lo = max(0, k)
        hi = min(n, n + k)
        if hi <= lo:
            continue
        jac[row, lo:hi] -= dx * (1.0 - h[lo - k:hi - k])
        jac[row, lo - k:hi - k] += dx * h[lo:hi]
    return jac

def slsqp_eq(x):
    n = len(x) - 1
    return x[:n].sum() - n / 2.0

def slsqp_eq_jac(x):
    g = np.zeros(len(x))
    g[:len(x) - 1] = 1.0
    return g

def slsqp_refine(h0, n):
    s0 = exact_score(h0, n)
    x0 = np.concatenate([h0, [min(1.0, s0 + 1e-6)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    con = [
        {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    try:
        from scipy.optimize import minimize
        res = minimize(slsqp_obj, x0, method='SLSQP', jac=slsqp_obj_jac,
                       bounds=bounds, constraints=con,
                       options={'maxiter': 400, 'ftol': 1e-13, 'disp': False})
        hx = res.x[:n]
        if not np.all(np.isfinite(hx)):
            return h0, s0
        hx, ok = project_capped(hx, n)
        if not ok:
            return h0, s0
        s = exact_score(hx, n)
        return (hx, s) if s < s0 - 1e-13 else (h0, s0)
    except Exception:
        return h0, s0

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(h.sum() - n / 2.0) <= 1e-10)

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = globals().get('initial_h_values', None)
    n = 45
    fallback = np.full(n, 0.5)
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.ndim == 1 and 2 <= len(p) <= 999 and np.all(np.isfinite(p)) and np.max(np.abs(p)) < 1e6:
            n = len(p)
            fallback, ok = project_capped(p, n)
            if not ok:
                fallback = np.full(n, 0.5)
    else:
        nk = kwargs.get('n_points', None)
        if isinstance(nk, int) and not isinstance(nk, bool) and 2 <= nk <= 999:
            n = nk
    fb_s = exact_score(fallback, n)
    best_h = fallback.copy()
    best_s = fb_s
    max_outer = 80 if n <= 100 else (12 if n <= 400 else 5)
    max_inner = 80 if n <= 100 else (20 if n <= 400 else 12)
    starts = [fallback.copy(), (1.0 - fallback).copy(), fallback[::-1].copy(),
              (1.0 - fallback[::-1]).copy(), fallback + 0.06 * rng.standard_normal(n),
              fallback + 0.20 * rng.standard_normal(n), rng.uniform(0, 1, n)]
    max_starts = 7 if n <= 100 else (4 if n <= 400 else 2)
    for i in range(min(len(starts), max_starts)):
        try:
            sh, ss = optimize_start(starts[i], n, max_outer, max_inner)
            if check_feasible(sh, n) and ss < best_s - 1e-12:
                best_s = ss
                best_h = sh.copy()
        except Exception:
            pass
    try:
        pert = best_h + 0.03 * rng.standard_normal(n)
        sh, ss = optimize_start(pert, n, max_outer // 2, max_inner)
        if check_feasible(sh, n) and ss < best_s - 1e-12:
            best_s = ss
            best_h = sh.copy()
    except Exception:
        pass
    if n <= 120:
        try:
            sh, ss = slsqp_refine(best_h, n)
            if ss < best_s - 1e-12:
                best_s = ss
                best_h = sh.copy()
        except Exception:
            pass
    final_h, ok = project_capped(best_h, n)
    if not ok or not check_feasible(final_h, n):
        final_h = fallback.copy()
    final_s = exact_score(final_h, n)
    fb_s2 = exact_score(fallback, n)
    if final_s > fb_s2 + 1e-13:
        final_h = fallback.copy()
        final_s = fb_s2
    final_h, _ = project_capped(final_h, n)
    final_s = exact_score(final_h, n)
    return final_h.astype(np.float64), float(final_s), int(n)
