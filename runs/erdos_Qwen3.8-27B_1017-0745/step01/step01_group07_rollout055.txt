def anneal_trajectory(h0, score0, q0, n, dx, rng, E, K):
    h, score, q = h0.copy(), score0, q0.copy()
    scale = 0.12
    for e in range(E):

        accepted = 0
        for _ in range(K):
            p, r = rng.integers(0, n, 2)
            if p == r: continue
            L = max(-h[p], h[r] - 1.0)
            U = min(1.0 - h[p], h[r])
            if U - L <= 1e-12: continue
            t = rng.uniform(-scale, scale)
            if t < L or t > U: continue
            hn = h.copy(); hn[p] += t; hn[r] -= t
            if not is_feasible(hn, n):

                if hn is None: continue

            if qn is None: continue

            if delta <= 0 or rng.random() < np.exp(-delta / T):
                h, score, q = hn, sn, qn
                accepted += 1
        if accepted / max(K, 1) < 0.12: scale = min(0.4, 1.25*scale)
        elif accepted / max(K, 1) > 0.55: scale = max(0.01, 0.75*scale)
