import numpy as np
import math

def project_K(y, n, S):
    if not np.all(np.isfinite(y)):
        return np.full(n, S / n)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    e = S - h.sum()
    if abs(e) > 1e-12:
        order = np.arange(n)
        for j in order:
            if e > 0:
                d = min(e, 1.0 - h[j])
                h[j] += d
                e -= d
            else:
                d = min(-e, h[j])
                h[j] -= d
                e += d
            if abs(e) < 1e-14:
                break
    if abs(h.sum() - S) > 1e-10 or h.min() < -1e-10 or h.max() > 1.0 + 1e-10:
        return np.full(n, S / n)
    return h

def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def gen_direction(n, rng):
    r = rng.random()
    if r < 0.60:
        if n < 2:
            return np.zeros(n)
        i, j = rng.choice(n, size=2, replace=False)
        z = np.zeros(n)
        z[i] = 1.0
        z[j] = -1.0
        return z
    elif r < 0.90:
        k = rng.integers(3, min(9, n + 1))
        idx = rng.choice(n, size=k, replace=False)
        vals = rng.standard_normal(k - 1)
        z = np.zeros(n)
        z[idx[:k-1]] = vals
        z[idx[k-1]] = -math.fsum(vals)
    else:
        z = rng.standard_normal(n)
        z -= z.mean()
        pivot = rng.integers(0, n)
        z[pivot] = -math.fsum(z[np.arange(n) != pivot])
    m = np.max(np.abs(z))
    if m < 1e-15 or not np.isfinite(m):
        return None
    return z / m

def make_proposal(h, n, S, sigma, rng):
    z = gen_direction(n, rng)
    if z is None:
        return None
    alpha = rng.uniform(-sigma, sigma)
    trial = h + alpha * z
    if not np.all(np.isfinite(trial)):
        return None
    if trial.min() < -1e-10 or trial.max() > 1.0 + 1e-10:
        return None
    e = S - trial.sum()
    if abs(e) > 1e-8:
        return None
    if abs(e) > 1e-12:
        if e > 0:
            j = int(np.argmax(1.0 - trial))
            if 1.0 - trial[j] < 1e-14:
                return None
            trial[j] = min(1.0, trial[j] + e)
        else:
            j = int(np.argmax(trial))
            if trial[j] < 1e-14:
                return None
            trial[j] = max(0.0, trial[j] + e)
    if not feasible(trial, n, S):
        return None
    return trial

def calibrate_temp(h0, f0, n, dx, rng, pilot_n):
    deltas = []
    sigma = 0.08
    for _ in range(pilot_n):
        trial = make_proposal(h0, n, S_tmp_global, sigma, rng)
        if trial is not None:
            d = exact_score(trial, dx) - f0
            if d > 1e-12:
                deltas.append(d)
            elif d < -1e-12:
                deltas.append(0.0)
    if len(deltas) > 0:
        up = [d for d in deltas if d > 1e-12]
        if len(up) > 0:
            med = float(np.median(up))
            T0 = -med / math.log(0.8)
            T0 = max(1e-5, min(5e-2, T0))
        else:
            T0 = 1e-3
    else:
        T0 = 1e-3
    Tmin = max(1e-6, 0.001 * T0)
    return T0, Tmin

def anneal_chain(h0, n, S, dx, rng, nlevels, nprops, sigma_init):
    global S_tmp_global
    S_tmp_global = S
    h = h0.copy()
    f = exact_score(h, dx)
    hb, fb = h.copy(), f
    if not np.isfinite(f):
        return hb, fb
    T0, Tmin = calibrate_temp(h, f, n, dx, rng, 24)
    sigma = sigma_init
    for lvl in range(nlevels):
        if nlevels <= 1:
            T = T0
        else:
            frac = float(lvl) / float(nlevels - 1)
            T = T0 * (Tmin / T0) ** frac
        acc = 0
        for _ in range(nprops):
            trial = make_proposal(h, n, S, sigma, rng)
            if trial is None:
                continue
            f_new = exact_score(trial, dx)
            delta = f_new - f
            if delta <= 0:
                accept = True
            else:
                ratio = delta / T
                if ratio > 745:
                    accept = False
                else:
                    accept = rng.random() < math.exp(-ratio)
            if accept:
                h, f = trial, f_new
                acc += 1
                if f < fb - 1e-12:
                    fb, hb = f, h.copy()
        rate = acc / float(nprops)
        if rate < 0.15:
            sigma = max(1e-4, 0.7 * sigma)
        elif rate > 0.45:
            sigma = min(0.30, 1.2 * sigma)
    return hb, fb

def build_starts(base, n, S, rng):
    starts = [base.copy(), (1.0 - base).copy(), base[::-1].copy(), (1.0 - base[::-1]).copy()]
    for sd in [0.05, 0.15, 0.30]:
        starts.append(project_K(base + sd * rng.standard_normal(n), n, S))
    starts.append(project_K(rng.uniform(0, 1, n), n, S))
    starts.append(project_K(rng.beta(0.5, 0.5, n), n, S))
    perm = rng.permutation(n)
    blk = np.zeros(n)
    blk[perm[:n//2]] = 1.0
    blk[perm[n//2]] = n / float(n - n // 2) if n % 2 else 1.0
    blk = project_K(blk, n, S)
    starts.append(blk)
    phase = rng.uniform(0, 2 * np.pi)
    js = np.arange(n)
    sin = 0.5 + 0.45 * np.cos(2.0 * np.pi * js / n + phase)
    starts.append(project_K(sin, n, S))
    dedup = []
    for s in starts:
        if not feasible(s, n, S):
            continue
        dup = False
        for d in dedup:
            if np.max(np.abs(s - d)) <= 1e-7:
                dup = True
                break
        if not dup:
            dedup.append(s)
    return dedup[:8]

def final_polish(h0, n, S, dx, rng, nprops):
    h = h0.copy()
    f = exact_score(h, dx)
    for sig in [0.02, 0.006]:
        for _ in range(nprops):
            trial = make_proposal(h, n, S, sig, rng)
            if trial is not None:
                f_new = exact_score(trial, dx)
                if f_new < f - 1e-12:
                    h, f = trial, f_new
    return h, f

S_tmp_global = 22.5

def run(seed=42, **kwargs):
    global S_tmp_global
    rng = np.random.default_rng(seed)
    init = globals().get('initial_h_values', None)
    ok = False
    if init is not None:
        init = np.array(init, dtype=np.float64).ravel()
        if 2 <= len(init) <= 999 and np.all(np.isfinite(init)) and init.min() >= -1e-8 and init.max() <= 1.0 + 1e-8:
            ok = True
    n = len(init) if ok else 45
    dx = 2.0 / n
    S = n / 2.0
    S_tmp_global = S
    if ok:
        base = project_K(init.copy(), n, S)
    else:
        base = np.full(n, 0.5)
    fb_h, fb_s = base.copy(), exact_score(base, dx)
    best_h, best_s = fb_h.copy(), fb_s
    starts = build_starts(base, n, S, rng)
    for s in starts:
        sc = exact_score(s, dx)
        if sc < best_s - 1e-12:
            best_s, best_h = sc, s.copy()
    if n <= 64:
        nstarts, nlevels, nprops, npolish = 8, 14, 80, 80
    elif n <= 128:
        nstarts, nlevels, nprops, npolish = 5, 10, 40, 40
    elif n <= 256:
        nstarts, nlevels, nprops, npolish = 3, 8, 18, 20
    else:
        nstarts, nlevels, nprops, npolish = 2, 6, 6, 8
    scored = sorted([(exact_score(s, dx), s) for s in starts if feasible(s, n, S)])
    sel = [scored[i][1] for i in range(min(nstarts, len(scored)))]
    for s in sel:
        try:
            hb, fb = anneal_chain(s, n, S, dx, rng, nlevels, nprops, 0.08)
            if feasible(hb, n, S):
                sc = exact_score(hb, dx)
                if sc < best_s - 1e-12:
                    best_s, best_h = sc, hb.copy()
        except Exception:
            pass
    hp, fs = final_polish(best_h, n, S, dx, rng, npolish)
    if feasible(hp, n, S) and fs < best_s - 1e-12:
        best_h, best_s = hp, fs
    best_h = project_K(best_h, n, S)
    final_s = exact_score(best_h, dx)
    if not feasible(best_h, n, S) or final_s > fb_s + 1e-10:
        best_h, final_s = fb_h.copy(), fb_s
    else:
        best_s = final_s
    best_h = np.clip(best_h, 0.0, 1.0)
    final_s = exact_score(best_h, dx)
    return (best_h, final_s, n)
