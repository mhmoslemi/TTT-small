```python
import numpy as np
from scipy.optimize import minimize

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    if S < -1e-10 or S > n + 1e-10:
        return None
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(v - tau, 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-14:
        for i in np.argsort(1.0 - y):
            d = min(r, 1.0 - y[i])
            y[i] += d
            r -= d
            if r <= 1e-14:
                break
    elif r < -1e-14:
        for i in np.argsort(y):
            d = min(-r, y[i])
            y[i] -= d
            r += d
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-10) or np.any(y > 1.0 + 1e-10):
        return None
    if abs(float(np.sum(y)) - S) > 1e-9 * max(1, n):
        return None
    return y

def check_feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(float(np.sum(h)) - 0.5 * n) < 1e-9 * max(1, n))

def exact_score(h):
    n = len(h)
    if not check_feasible(h, n):
        return float('inf')
    try:
        c = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
        if not np.isfinite(c):
            return float('inf')
        return c
    except Exception:
        return float('inf')

def lag_val_grad(h, s, n):
    dx = 2.0 / n
    lo_i = max(0, -s)
    hi_i = min(n, n - s)
    q = 0.0
    g = np.zeros(n)
    for i in range(lo_i, hi_i):
        j = i + s
        q += h[i] * (1.0 - h[j])
        g[i] += (1.0 - h[j])
        g[j] -= h[i]
    return dx * q, dx * g

def get_parent():
    n = 45
    if 'initial_h_values' in globals():
        ph = np.asarray(globals()['initial_h_values'], dtype=np.float64)
        if ph.ndim == 1 and 2 <= len(ph) <= 999 and np.all(np.isfinite(ph)):
            if np.all(ph >= -1e-6) and np.all(ph <= 1.0 + 1e-6):
                n = len(ph)
                p = project_box_sum(np.clip(ph, 0.0, 1.0), 0.5 * n)
                if p is not None and check_feasible(p, n):
                    return p, n
    return project_box_sum(np.full(n, 0.5), 0.5 * n), n

def build_starts(parent, n, rng):
    starts = [parent.copy(), (1.0 - parent).copy(), parent[::-1].copy()]
    for sd in [0.06, 0.15, 0.25]:
        p = project_box_sum(parent + sd * rng.standard_normal(n), 0.5 * n)
        if p is not None:
            starts.append(p)
    p = project_box_sum(rng.random(n), 0.5 * n)
    if p is not None:
        starts.append(p)
    u = 0.5 + 0.30 * np.cos(2.0 * np.pi * np.arange(n) / n + 0.7) + 0.12 * np.sin(4.0 * np.pi * np.arange(n) / n + 1.3)
    p = project_box_sum(u, 0.5 * n)
    if p is not None:
        starts.append(p)
    # deduplicate
    unique = []
    for s in starts:
        if s is None:
            continue
        dup = False
        for u2 in unique:
            if np.max(np.abs(s - u2)) < 1e-10:
                dup = True
                break
        if not dup:
            unique.append(s)
    return unique
```

```python
_MM = {}

def mm_obj(z):
    x = _MM['x']
    L = _MM['L']
    y = z[:-1]
    t = z[-1]
    d = y - x
    return 0.5 * L * np.dot(d, d) + t

def mm_obj_grad(z):
    x = _MM['x']
    L = _MM['L']
    y = z[:-1]
    g = np.empty(len(z))
    g[:-1] = L * (y - x)
    g[-1] = 1.0
    return g

def mm_ineq(z):
    x = _MM['x']
    qJ = _MM['qJ']
    GJ = _MM['GJ']
    y = z[:-1]
    t = z[-1]
    d = y - x
    return t - qJ - GJ @ d

def mm_ineq_grad(z):
    GJ = _MM['GJ']
    n = GJ.shape[1]
    J = np.hstack([-GJ, np.ones((GJ.shape[0], 1))])
    return J

def mm_eq(z):
    n = _MM['n']
    return float(np.sum(z[:-1])) - 0.5 * n

def mm_eq_grad(z):
    n = _MM['n']
    g = np.zeros(n + 1)
    g[:-1] = 1.0
    return g
```

```python
def solve_mm(x, n):
    dx = 2.0 / n
    L = 2.0 * dx
    # Compute all lag values and gradients
    nv = 2 * n - 1
    q_all = np.zeros(nv)
    G_all = np.zeros((nv, n))
    for idx, s in enumerate(range(-(n-1), n)):
        q, g = lag_val_grad(x, s, n)
        q_all[idx] = q
        G_all[idx] = g
    
    # For n <= 64, use all lags; otherwise use top lags
    if n <= 64:
        active = np.arange(nv)
    else:
        top_k = min(nv, 96)
        active = np.argsort(-q_all)[:top_k]
    
    qJ = q_all[active]
    GJ = G_all[active]
    
    _MM['x'] = x
    _MM['L'] = L
    _MM['qJ'] = qJ
    _MM['GJ'] = GJ
    _MM['n'] = n
    
    # Initialize
    y0 = x.copy()
    d0 = np.zeros(n)
    t0 = np.max(qJ + GJ @ d0) + 2e-4
    z0 = np.append(y0, t0)
    
    bounds = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
    cons = [{'type': 'ineq', 'fun': mm_ineq, 'jac': mm_ineq_grad},
            {'type': 'eq', 'fun': mm_eq, 'jac': mm_eq_grad}]
    opts = {'maxiter': 150, 'ftol': 1e-12, 'disp': False}
    
    try:
        res = minimize(mm_obj, z0, jac=mm_obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons, options=opts)
    except Exception:
        return x.copy()
    
    out = res.x
    if len(out) != n + 1 or not np.all(np.isfinite(out)):
        return x.copy()
    
    y_new = out[:-1]
    # Project to feasible set
    y_new = project_box_sum(y_new, 0.5 * n)
    if y_new is None:
        return x.copy()
    
    return y_new
```

```python
def mm_descent(h0, n, max_steps=25):
    h = h0.copy()
    score = exact_score(h)
    for _ in range(max_steps):
        proposal = solve_mm(h, n)
        if proposal is None or np.max(np.abs(proposal - h)) < 1e-10:
            break
        # Line search
        best_alpha = -1
        best_sc = score
        for k in range(9):
            alpha = 2.0 ** (-k)
            pt = h + alpha * (proposal - h)
            pt = project_box_sum(pt, 0.5 * n)
            if pt is None:
                continue
            sc = exact_score(pt)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_alpha = alpha
        if best_alpha < 0:
            break
        h = project_box_sum(h + best_alpha * (proposal - h), 0.5 * n)
        if h is None:
            break
        new_score = exact_score(h)
        if new_score >= score - 1e-12:
            break
        if score - new_score < 1e-10 * max(1.0, abs(score)):
            h = h.copy()
            score = new_score
            break
        score = new_score
    return h, score
```

```python
import numpy as np
from scipy.optimize import minimize

_MM = {}

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    if S < -1e-10 or S > n + 1e-10:
        return None
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(v - tau, 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-14:
        for i in np.argsort(1.0 - y):
            d = min(r, 1.0 - y[i])
            y[i] += d
            r -= d
            if r <= 1e-14:
                break
    elif r < -1e-14:
        for i in np.argsort(y):
            d = min(-r, y[i])
            y[i] -= d
            r += d
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-10) or np.any(y > 1.0 + 1e-10):
        return None
    if abs(float(np.sum(y)) - S) > 1e-9 * max(1, n):
        return None
    return y

def check_feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(float(np.sum(h)) - 0.5 * n) < 1e-9 * max(1, n))

def exact_score(h):
    n = len(h)
    if not check_feasible(h, n):
        return float('inf')
    try:
        c = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
        return c if np.isfinite(c) else float('inf')
    except Exception:
        return float('inf')

def lag_val_grad(h, s, n):
    dx = 2.0 / n
    lo_i = max(0, -s)
    hi_i = min(n, n - s)
    q = 0.0
    g = np.zeros(n)
    for i in range(lo_i, hi_i):
        j = i + s
        q += h[i] * (1.0 - h[j])
        g[i] += (1.0 - h[j])
        g[j] -= h[i]
    return dx * q, dx * g

def get_parent():
    n = 45
    if 'initial_h_values' in globals():
        ph = np.asarray(globals()['initial_h_values'], dtype=np.float64)
        if ph.ndim == 1 and 2 <= len(ph) <= 999 and np.all(np.isfinite(ph)):
            if np.all(ph >= -1e-6) and np.all(ph <= 1.0 + 1e-6):
                n = len(ph)
                p = project_box_sum(np.clip(ph, 0.0, 1.0), 0.5 * n)
                if p is not None and check_feasible(p, n):
                    return p, n
    return project_box_sum(np.full(n, 0.5), 0.5 * n), n

def build_starts(parent, n, rng):
    starts = [parent.copy(), (1.0 - parent).copy(), parent[::-1].copy()]
    for sd in [0.05, 0.12, 0.22]:
        p = project_box_sum(parent + sd * rng.standard_normal(n), 0.5 * n)
        if p is not None:
            starts.append(p)
    p = project_box_sum(rng.random(n), 0.5 * n)
    if p is not None:
        starts.append(p)
    idx = np.arange(n)
    u = 0.5 + 0.28 * np.cos(2.0 * np.pi * idx / n + 0.7) + 0.10 * np.sin(4.0 * np.pi * idx / n + 1.3)
    p = project_box_sum(u, 0.5 * n)
    if p is not None:
        starts.append(p)
    u2 = 0.5 + 0.35 * np.cos(np.pi * idx / n - 0.3)
    p2 = project_box_sum(u2, 0.5 * n)
    if p2 is not None:
        starts.append(p2)
    unique = []
    for s in starts:
        dup = False
        for u2 in unique:
            if np.max(np.abs(s - u2)) < 1e-10:
                dup = True
                break
        if not dup:
            unique.append(s)
    return unique

def mm_obj(z):
    x = _MM['x']
    L = _MM['L']
    d = z[:-1] - x
    return 0.5 * L * np.dot(d, d) + z[-1]

def mm_obj_grad(z):
    x = _MM['x']
    L = _MM['L']
    g = np.empty(len(z))
    g[:-1] = L * (z[:-1] - x)
    g[-1] = 1.0
    return g

def mm_ineq(z):
    qJ = _MM['qJ']
    GJ = _MM['GJ']
    x = _MM['x']
    return z[-1] - qJ - GJ @ (z[:-1] - x)

def mm_ineq_grad(z):
    GJ = _MM['GJ']
    return np.hstack([-GJ, np.ones((GJ.shape[0], 1))])

def mm_eq(z):
    return float(np.sum(z[:-1])) - 0.5 * _MM['n']

def mm_eq_grad(z):
    g = np.zeros(_MM['n'] + 1)
    g[:-1] = 1.0
    return g

def solve_mm(x, n):
    dx = 2.0 / n
    L = 2.0 * dx
    nv = 2 * n - 1
    q_all = np.zeros(nv)
    G_all = np.zeros((nv, n))
    for idx, s in enumerate(range(-(n - 1), n)):
        q, g = lag_val_grad(x, s, n)
        q_all[idx] = q
        G_all[idx] = g
    if n <= 64:
        active = np.arange(nv)
    else:
        k = min(nv, 96)
        active = np.argsort(-q_all)[:k]
    qJ = q_all[active]
    GJ = G_all[active]
    _MM['x'] = x
    _MM['L'] = L
    _MM['qJ'] = qJ
    _MM['GJ'] = GJ
    _MM['n'] = n
    y0 = x.copy()
    t0 = float(np.max(qJ)) + 2e-4
    t0 = max(t0, 1e-6)
    z0 = np.append(y0, t0)
    bounds = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
    cons = [{'type': 'ineq', 'fun': mm_ineq, 'jac': mm_ineq_grad},
            {'type': 'eq', 'fun': mm_eq, 'jac': mm_eq_grad}]
    opts = {'maxiter': 150, 'ftol': 1e-13, 'disp': False}
    try:
        res = minimize(mm_obj, z0, jac=mm_obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons, options=opts)
    except Exception:
        return x.copy()
    out = res.x
    if len(out) != n + 1 or not np.all(np.isfinite(out)):
        return x.copy()
    y_new = project_box_sum(out[:-1], 0.5 * n)
    if y_new is None or not check_feasible(y_new, n):
        return x.copy()
    return y_new

def mm_descent(h0, n, max_steps):
    h = h0.copy()
    score = exact_score(h)
    if not np.isfinite(score):
        return h, score
    for _ in range(max_steps):
        proposal = solve_mm(h, n)
        if proposal is None:
            break
        diff = proposal - h
        if np.max(np.abs(diff)) < 1e-11:
            break
        best_alpha = -1
        best_sc = score
        n_alpha = 9 if n <= 128 else 6
        for k in range(n_alpha):
            alpha = 2.0 ** (-k)
            pt = h + alpha * diff
            pt = project_box_sum(pt, 0.5 * n)
            if pt is None:
                continue
            sc = exact_score(pt)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_alpha = alpha
        if best_alpha < 0:
            break
        h_new = project_box_sum(h + best_alpha * diff, 0.5 * n)
        if h_new is None or not check_feasible(h_new, n):
            break
        new_score = exact_score(h_new)
        if new_score >= score - 1e-13:
            break
        h = h_new.copy()
        score = new_score
        if score < 0.3808:
            break
    return h, score

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent, n = get_parent()
    best_h = parent.copy()
    best_c5 = exact_score(parent)
    starts = build_starts(parent, n, rng)
    M = max(10, min(35, 1800 // n))
    for h0 in starts:
        try:
            h_new, sc = mm_descent(h0, n, M)
            if np.isfinite(sc) and sc < best_c5:
                best_h = h_new.copy()
                best_c5 = sc
        except Exception:
            pass
    sds = [0.03, 0.08, 0.18, 0.30] if n <= 64 else [0.08, 0.20]
    for sd in sds:
        h0 = project_box_sum(best_h + sd * rng.standard_normal(n), 0.5 * n)
        if h0 is None:
            continue
        try:
            h_new, sc = mm_descent(h0, n, max(5, M // 2))
            if np.isfinite(sc) and sc < best_c5:
                best_h = h_new.copy()
                best_c5 = sc
        except Exception:
            pass
    best_h = project_box_sum(best_h, 0.5 * n)
    if best_h is not None and check_feasible(best_h, n):
        best_c5 = exact_score(best_h)
    else:
        best_h = parent.copy()
        best_c5 = exact_score(parent)
    ps = exact_score(parent)
    if ps < best_c5:
        return (parent, ps, n)
    return (best_h, best_c5, n)
```
