    if disc < -1e-12 * max(1.0, B * B, abs(4.0 * A * C)): return []
    if disc < 0: disc = 0.0
    sq = math.sqrt(disc)
    if abs(q := -0.5 * (B + math.copysign(sq, B))) < 1e-30:
        roots = [sq / (2.0 * A)] if sq > 1e-30 else []
    else:
        roots = [q / A, C / q]
    return [r for r in roots if np.isfinite(r)]

def build_hull(segs, lo, hi):
    if not segs: return []
    idx = np.argsort(segs[:, 0])
    lines = [(float(segs[idx[k], 0]), float(segs[idx[k], 1])) for k in range(len(idx))]
    dedup = []
    for m, c in lines:
        if dedup and abs(dedup[-1][0] - m) < 1e-18:
            if c > dedup[-1][1]: dedup[-1] = (m, c)
        else: dedup.append((m, c))
    if len(dedup) <= 1:
        if dedup: return [(dedup[0], lo, hi)]
        return []
    hull = [dedup[0]]; starts = [-np.inf]
    for m, c in dedup[1:]:
        if abs(m - hull[-1][0]) < 1e-18: continue
        cross = (hull[-1][1] - c) / (m - hull[-1][0])
        while len(hull) > 1 and cross <= starts[-1] + 1e-18:
            hull.pop(); starts.pop()
            cross = (hull[-1][1] - c) / (m - hull[-1][0])
        if len(hull) == 1 and cross <= starts[0]:
            hull = [(m, c)]; starts = [max(-hp.inf, lo)]
        else:
            hull.append((m, c)); starts.append(cross)
    out = []
    for i in range(len(hull)):
        s_lo = max(starts[i], lo)
        s_hi = min(starts[i + 1] if i + 1 < len(starts) else np.inf, hi)
        if s_lo < s_hi - 1e-15: out.append((hull[i], s_lo, s_hi))
    return out

def pair_try(h, q, cur, i, j, n):
    dx = 2.0 / n
    lo_t = max(-h[i], h[j] - 1.0)
    hi_t = min(1.0 - h[i], h[j])
    if hi_t - lo_t < 1e-13: return None
    L = 2 * n - 1
    s_arr = np.arange(L, dtype=np.float64) - (n - 1)
    bp = np.zeros(L)
    for p in range(L):
        s = int(p - (n - 1)); v = 0.0
        if 0 <= i + s < n: v += 1.0 - h[i + s]
        if 0 <= j + s < n: v -= 1.0 - h[j + s]
        if 0 <= i - s < n: v += h[i - s]
        if 0 <= j - s < n: v -= h[j - s]
        bp[p] = dx * v
    cp = q.copy()
    qp_idx = [n - 1, n - 1 + j - i, n - 1 + i - j]
    ap = np.zeros(L)
    ap[n - 1] = -2.0 * dx
    if 0 <= n - 1 + j - i < L: ap[n - 1 + j - i] = dx
    if 0 <= n - 1 + i - j < L: ap[n - 1 + i - j] = dx
    affine_mask = ap < 1e-30
    segs = np.stack([bp[affine_mask], cp[affine_mask]], axis=1) if np.any(affine_mask) else np.empty((0, 2))
    hull = build_hull(segs, lo_t, hi_t)
    cands = [lo_t, hi_t, 0.0]
    for _, s0, s1 in hull: cands.append(0.5 * (s0 + s1))
    for s0, s1 in [(seg[1], seg[2]) for seg in hull]: cands += [s0, s1]
    for qi in qp_idx:
        a, b, c = ap[qi], bp[qi], cp[qi]
        if abs(a) > 1e-30:
            tv = -b / (2.0 * a)
            if lo_t - 1e-13 <= tv <= hi_t + 1e-13: cands.append(tv)
    for a_i in range(len(qp_idx)):
        for a_j in range(a_i + 1, len(qp_idx)):
            p1, p2 = qp_idx[a_i], qp_idx[a_j]
            da = ap[p1] - ap[p2]; db = bp[p1] - bp[p2]; dc = cp[p1] - cp[p2]
            for r in quad_roots(da, db, dc):
                if lo_t - 1e-13 <= r <= hi_t + 1e-13: cands.append(r)
    for (m, c0), s0, s1 in hull:
        for qi in qp_idx:
            a, b, c = ap[qi], bp[qi] - m, cp[qi] - c0
            for r in quad_roots(a, b, c):
                if s0 - 1e-12 <= r <= s1 + 1e-12: cands.append(r)
    cands = np.array([min(max(c, lo_t), hi_t) for c in cands if np.isfinite(c)])
    if len(cands) == 0: return None
    cands = np.unique(np.round(cands, 12))
    best_t, best_val = 0.0, cur
    for t in cands:
        val = cp + bp * t + ap * t * t
        mv = float(np.max(val))
        if mv < best_val - 1e-14: best_val = mv; best_t = float(t)
    if abs(best_t) < 1e-15: return None
    h_new = h.copy(); h_new[i] += best_t; h_new[j] -= best_t
    h_new = project_box_sum(h_new, 0.5 * n)
    if h_new is None: return None
    sc, _ = score_and_corr(h_new, n)
    if sc < cur - 1e-14: return h_new, sc
    return None

def local_descent(h, n, rng, max_sweeps):
    h = h.copy()
    sc, q = score_and_corr(h, n)
    if sc == float('inf'): return h, sc
    pairs = list(range(n * (n - 1) // 2)) if n <= 64 else \
        list(rng.choice(n * (n - 1) // 2, size=min(256, n * (n - 1) // 2), replace=False))
    for _ in range(max_sweeps):
        improved = False
        rng.shuffle(pairs)
        for pi in pairs:
            ii, jj = pi // (n - 1), pi % (n - 1)
            if ii >= jj: ii, jj = n - 1 - 1 - ii, (pi % (n - 1)) + n - 1
            # redo: map single index to pair
            a, b = 0, 0
            rem = pi
            for a in range(n - 1):
                if rem < n - 1 - a: b = a + 1; break
                rem -= (n - 1 - a)
            else:
                continue
            try:
                res = pair_try(h, q, sc, a, b, n)
            except Exception:
                continue
            if res is not None:
                h, sc = res
                q = (2.0 / n) * np.correlate(h, 1.0 - h, mode='full')
                improved = True
        if not improved: break
    return h, sc

def build_starts(parent, n, rng):
    starts = [parent.copy(), (1.0 - parent).copy(), parent[::-1].copy()]
    for sd in [0.06, 0.20, 0.12]:
        p = project_box_sum(parent + sd * rng.standard_normal(n), 0.5 * n)
        if p is not None: starts.append(p)
    p = project_box_sum(rng.random(n), 0.5 * n)
    if p is not None: starts.append(p)
    phi = rng.uniform(0, 2 * np.pi); th = rng.uniform(0, 2 * np.pi)
    u = 0.5 + 0.30 * np.cos(2 * np.pi * np.arange(n) / n + phi) + 0.15 * np.sin(4 * np.pi * np.arange(n) / n + th)
    p = project_box_sum(u, 0.5 * n)
    if p is not None: starts.append(p)
    seen = []; unique = []
    for s in starts:
        if not check_feasible(s, n): continue
        dup = False
        for u2 in seen:
            if np.max(np.abs(s - u2)) < 1e-10: dup = True; break
        if not dup: seen.append(s); unique.append(s)
    return unique

def run(seed=42, **kwargs):
    try: rng = np.random.default_rng(seed)
    except Exception: rng = np.random.default_rng(42)
    parent, n = get_parent()
    best_h, best_c5 = parent.copy(), 0.0
    best_c5, _ = score_and_corr(best_h, n)
    if best_c5 == float('inf'): best_c5 = 0.39; best_h = np.full(n, 0.5)
    starts = build_starts(parent, n, rng)
    ms = 3 if n <= 64 else 2
    for s in starts:
        h, sc = local_descent(s, n, rng, ms)
        if sc < best_c5: best_h, best_c5 = h, sc
    for sigma in [0.04, 0.12, 0.28]:
        h0 = project_box_sum(best_h + sigma * rng.standard_normal(n), 0.5 * n)
        if h0 is None: continue
        h, sc = local_descent(h0, n, rng, 2 if n <= 64 else 1)
        if sc < best_c5: best_h, best_c5 = h, sc
    best_h = project_box_sum(best_h, 0.5 * n)
    if best_h is not None and check_feasible(best_h, n):
        best_c5, _ = score_and_corr(best_h, n)
    else:
        best_h = parent.copy(); best_c5, _ = score_and_corr(best_h, n)
    if best_c5 == float('inf'): best_h = np.full(n, 0.5); best_c5, _ = score_and_corr(best_h, n)
    return (best_h, best_c5, n)
