def project_capped_simplex(y, n, S):
    if not np.all(np.isfinite(y)) or len(y) != n:
        return np.full(n, S / n)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    e = S - h.sum()
    if abs(e) > 1e-12:
        if e > 0:
            for j in range(n):
                room = 1.0 - h[j]
                add = min(room, e)
                h[j] += add
                e -= add
                if abs(e) < 1e-13:
                    break
        else:
            for j in range(n):
                sub = min(h[j], -e)
                h[j] -= sub
                e += sub
                if abs(e) < 1e-13:
                    break
    return h

def exact_score(h, n):
    if len(h) != n or not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
