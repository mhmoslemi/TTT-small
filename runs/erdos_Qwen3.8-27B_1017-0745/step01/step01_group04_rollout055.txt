import numpy as np

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > m:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(z - lam, 0.0, 1.0)
    r = m - h.sum()
    if r > 1e-14:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
            if r <= 1e-14:
                break
    elif r < -1e-14:
        for i in range(N):
            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
            if r >= -1e-14:
                break
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1.0 + 1e-10:
        return np.full(N, 0.5)
    return h

def exact_score(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / N)

def smooth_obj_grad(h, tau, dx):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = dx * corr  # length 2N-1
    cmax = c.max()
    exp_c = np.exp((c - cmax) / tau)
    sum_exp = exp_c.sum()
    F = cmax + tau * np.log(sum_exp)
    p = exp_c / sum_exp  # softmax weights, length 2N-1
    # Gradient: sum_s p[s] * dx * dq_s/dh
    # corr index j corresponds to lag k = j - (N-1)
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        k = j - (N - 1)
        w = dx * p[j]
        if w < 1e-15:
            continue
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            d = k
            # For j_idx in [d, N-1]: add (1-h[j_idx - d])
            grad[d:] += w * (1.0 - h[:N - d])
            # For j_idx in [0, N-1-d]: add -h[j_idx + d]
            grad[:N - d] -= w * h[d:]
        else:

            # For j_idx in [d, N-1]: add -h[j_idx - d]
            grad[d:] -= w * h[:N - d]
    return F, grad, c

            break

def make_fourier_start(N, rng, n_modes, amp):
    z = np.full(N, 0.5)
    ns = min(n_modes, (N - 1) // 2)

        z += (amp / r) * (A * np.sin(2.0 * np.pi * r * idx / N) + B * np.cos(2.0 * np.pi * r * idx / N))

import numpy as np

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(z - lam, 0.0, 1.0)
    r = m - h.sum()
    if r > 1e-14:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
            if r <= 1e-14:
                break
    elif r < -1e-14:
        for i in range(N):
            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
            if r >= -1e-14:
                break
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1.0 + 1e-10:
        return np.full(N, 0.5)
    return h

def exact_score(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / N)

def smooth_obj_grad(h, tau, dx):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = dx * corr
    cmax = c.max()
    exp_c = np.exp((c - cmax) / tau)
    sum_exp = exp_c.sum()
    F = cmax + tau * np.log(sum_exp)
    p = exp_c / sum_exp
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        k = j - (N - 1)
        w = dx * p[j]
        if w < 1e-16:
            continue
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            d = k
            grad[:N - d] += w * (1.0 - h[d:])
            grad[d:] -= w * h[:N - d]
        else:
            d = -k
            grad[d:] += w * (1.0 - h[:N - d])
            grad[:N - d] -= w * h[d:]
    return F, grad

def optimize_start(h0, N, m, dx, tau_sched, max_iter):
    best_h = h0.copy()
    best_score = exact_score(h0, N)
    h = h0.copy()
    for tau in tau_sched:
        F_cur, g_cur = smooth_obj_grad(h, tau, dx)
        if not np.isfinite(F_cur) or not np.all(np.isfinite(g_cur)):
            break
        alpha = 1.0
        for it in range(max_iter):
            y = project_box_sum(h - alpha * g_cur, m)
            d = y - h
            if not np.all(np.isfinite(d)):
                break
            if np.max(np.abs(d)) < 1e-11:
                alpha *= 0.5
                if alpha < 1e-13:
                    break
                continue
            s_y = exact_score(y, N)
            if np.isfinite(s_y) and s_y < best_score:
                best_score = s_y
                best_h = y.copy()
            F_y, _ = smooth_obj_grad(y, tau, dx)
            gdotd = np.dot(g_cur, d)
            if gdotd < 0 and F_y <= F_cur + 1e-4 * gdotd + 1e-12:
                dec = F_cur - F_y
                h = y
                F_cur = F_y
                g_cur = smooth_obj_grad(h, tau, dx)[1]
                if not np.all(np.isfinite(g_cur)):
                    break
                alpha = min(1.25 * alpha, 10.0)
                if dec < 1e-13 * max(1.0, abs(F_cur)):
                    alpha *= 0.5
            else:
                alpha *= 0.5
                if alpha < 1e-13:
                    break
    return best_h, best_score

def make_fourier_start(N, rng, n_modes, amp):
    z = np.full(N, 0.5)
    ns = min(n_modes, (N - 1) // 2)
    idx = np.arange(N)
    for r in range(1, ns + 1):
        A = rng.standard_normal()
        B = rng.standard_normal()
        z += (amp / r) * (A * np.sin(2.0 * np.pi * r * idx / N) + B * np.cos(2.0 * np.pi * r * idx / N))
    return z

def run(seed=42, **kwargs):
    N = None
    m = None
    raw = globals().get('initial_h_values', None)
    if raw is not None:
        raw = np.asarray(raw, dtype=np.float64)
        if raw.ndim == 1 and raw.size >= 2 and np.all(np.isfinite(raw)):
            N = raw.size
            m = N / 2.0
    if N is None:
        N = kwargs.get('n_points', 70)
        if not isinstance(N, (int, np.integer)) or N < 2:
            N = 70
        m = N / 2.0
    dx = 2.0 / N
    rng = np.random.default_rng(seed)

    if raw is not None and raw.ndim == 1 and raw.size == N and np.all(np.isfinite(raw)):
        baseline = project_box_sum(raw.copy(), m)
    else:
        baseline = np.full(N, 0.5)

    best_h = baseline.copy()
    best_score = exact_score(baseline, N)

    if N <= 128:
        n_starts, max_iter, n_hops = 8, 50, 2
    elif N <= 256:
        n_starts, max_iter, n_hops = 6, 35, 1
    elif N <= 512:
        n_starts, max_iter, n_hops = 4, 20, 0
    else:
        n_starts, max_iter, n_hops = 3, 10, 0

    tau_sched = [0.02, 0.008, 0.003, 0.001, 0.0004, 0.00012, 0.00003]

    starts = [baseline.copy()]
    starts.append(project_box_sum(1.0 - baseline, m))
    starts.append(project_box_sum(baseline + 0.05 * rng.standard_normal(N), m))
    starts.append(project_box_sum(baseline + 0.15 * rng.standard_normal(N), m))
    starts.append(project_box_sum(baseline + 0.30 * rng.standard_normal(N), m))
    starts.append(project_box_sum(rng.uniform(size=N), m))
    starts.append(project_box_sum(make_fourier_start(N, rng, 6, 0.12), m))
    starts.append(project_box_sum(make_fourier_start(N, rng, 4, 0.25), m))
    starts.append(project_box_sum(0.5 + 0.4 * rng.uniform(size=N), m))

    deduped = [starts[0]]
    for s in starts[1:]:
        if all(np.max(np.abs(s - x)) > 1e-10 for x in deduped):
            deduped.append(s)
    starts = deduped[:n_starts]

    for h0 in starts:
        s0 = exact_score(h0, N)
        if s0 < best_score:
            best_score = s0
            best_h = h0.copy()
        h_opt, s_opt = optimize_start(h0, N, m, dx, tau_sched, max_iter)
        if s_opt < best_score:
            best_score = s_opt
            best_h = h_opt.copy()

    hop_scales = [0.04, 0.015][:n_hops]
    for sc in hop_scales:
        hop = project_box_sum(best_h + sc * rng.standard_normal(N), m)
        s_hop = exact_score(hop, N)
        if s_hop < best_score:
            best_score = s_hop
            best_h = hop.copy()
        h_opt, s_opt = optimize_start(hop, N, m, dx, tau_sched, max_iter)
        if s_opt < best_score:
            best_score = s_opt
            best_h = h_opt.copy()

    best_h = project_box_sum(best_h, m)
    c5 = exact_score(best_h, N)
    base_check = exact_score(baseline, N)
    if base_check < c5:
        best_h = baseline.copy()
        c5 = base_check
    if not np.isfinite(c5):
        best_h = np.full(N, 0.5)
        c5 = exact_score(best_h, N)
    return best_h, float(c5), int(N)
