        return np.full(n, S / n)
    return h

def feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-10 and h.max() <= 1 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_vec(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def pair_b(h, n, p, r):
    dx = 2.0 / n
    b = np.zeros(2 * n - 1)
    for ell in range(-(n - 1), n):
        lo_i, hi_i = max(0, -ell), min(n, n - ell)
        dp = 0.0
        if lo_i <= p - ell < hi_i: dp += 1.0 - h[p - ell]
        if lo_i <= p < hi_i: dp -= h[p + ell]
        dr = 0.0
        if lo_i <= r - ell < hi_i: dr += 1.0 - h[r - ell]
        if lo_i <= r < hi_i: dr -= h[r + ell]
        b[ell + n - 1] = dx * (dp - dr)
    return b

def pair_a(n, p, r):
    dx = 2.0 / n
    a = np.zeros(2 * n - 1)
    a[n - 1] = -2.0 * dx
    a[p - r + n - 1] = dx
    a[r - p + n - 1] = dx
    return a

def line_model(q, b, a, alphas):
    return np.max(q[None, :] + b[None, :] * alphas[:, None] + a[None, :] * alphas[:, None] ** 2)

def line_search(h, q, sc, z, b, a, lo, hi, n, nc=25, nb=2, nr=4, nrp=9):
    S = n / 2.0
    if hi - lo < 1e-14:
        return h, q, sc
    al = np.linspace(lo, hi, nc)
    mv = line_model(q, b, a, al)
    cands = [0.0]
    mns = []
    for k in range(nc):
        if k == 0:
            if mv[0] <= mv[1]: mns.append(k)
        elif k == nc - 1:
            if mv[k] <= mv[k - 1]: mns.append(k)
        else:
            if mv[k] <= mv[k - 1] and mv[k] <= mv[k + 1]: mns.append(k)
    mns.sort(key=lambda k: (mv[k], abs(al[k])))
    chosen = []
    for k in mns:
        if len(chosen) >= nb: break
        if any(abs(k - c) < 2 for c in chosen): continue
        chosen.append(k)
    if not chosen and mns: chosen = [mns[0]]
    for ck in chosen:
        lo_b, hi_b = al[max(0, ck - 1)], al[min(nc - 1, ck + 1)]
        ba, bv = al[ck], mv[ck]
        for _ in range(nr):
            mids = np.linspace(lo_b, hi_b, nrp)
            mvals = line_model(q, b, a, mids)
            bk = int(np.argmin(mvals))
            if mvals[bk] < bv: bv, ba = mvals[bk], mids[bk]
            lo_b, hi_b = mids[max(0, bk - 1)], mids[min(nrp - 1, bk + 1)]
        cands.append(ba)
    best_h, best_sc = h.copy(), sc
    for alv in cands:
        tr = h + alv * z
        tr = project_K(tr, n, S)
        if not feasible(tr, n, S): continue
        s2 = exact_score(tr, n)
        if s2 < best_sc - 1e-12:
            best_sc, best_h = s2, tr
    return best_h, lag_vec(best_h, n), best_sc

def sparse_dir(n, rng):
    sz = int(rng.integers(2, min(12, n) + 1))
    idx = rng.choice(n, size=sz, replace=False)
    vals = rng.standard_normal(sz)
    vals -= vals.mean()
    vals[-1] = -vals[:-1].sum()
    m = np.max(np.abs(vals))
    if m < 1e-15: return None
    z = np.zeros(n)
    z[idx] = vals / m
    z[idx[-1]] -= z.sum()
    return z

def local_descent(h, n, rng, ms=3, ns=8):
    S = n / 2.0
    h = project_K(h, n, S)
    if not feasible(h, n, S): h = np.full(n, S / n)
    q = lag_vec(h, n)
    sc = exact_score(h, n)
    dx = 2.0 / n
    pairs = [(p, r) for p in range(n) for r in range(p + 1, n)]
    unch = 0
    for _ in range(ms):
        changed = False
        perm = rng.permutation(len(pairs))
        for ki in perm:
            p, r = pairs[ki]
            lo = max(-h[p], h[r] - 1.0)
            hi = min(1.0 - h[p], h[r])
            if hi - lo < 1e-14: continue
            b = pair_b(h, n, p, r)
            a = pair_a(n, p, r)
            z = np.zeros(n); z[p], z[r] = 1.0, -1.0
            h2, q2, sc2 = line_search(h, q, sc, z, b, a, lo, hi, n)
            if sc2 < sc - 1e-14:
                h, q, sc = h2, q2, sc2
                changed = True
        for _ in range(ns):
            z = sparse_dir(n, rng)
            if z is None: continue
            b = dx * (np.correlate(z, 1.0 - h, mode='full') - np.correlate(h, z, mode='full'))
            a = -dx * np.correlate(z, z, mode='full')
            if not np.all(np.isfinite(b)) or not np.all(np.isfinite(a)): continue
            lo, hi = -inf, inf
            for j in range(n):
                if z[j] > 1e-15:
                    lo = max(lo, -h[j] / z[j]); hi = min(hi, (1.0 - h[j]) / z[j])
                elif z[j] < -1e-15:
                    lo = max(lo, (1.0 - h[j]) / z[j]); hi = min(hi, -h[j] / z[j])
            lo, hi = max(lo, -1.0), min(hi, 1.0)
            if hi - lo < 1e-14: continue
            h2, q2, sc2 = line_search(h, q, sc, z, b, a, lo, hi, n)
            if sc2 < sc - 1e-14:
                h, q, sc = h2, q2, sc2
                changed = True
        unch = 0 if changed else unch + 1
        if unch >= 2: break
    return h, sc

def build_starts(base, n, rng):
    S = n / 2.0
    starts = [base.copy(), (1.0 - base).copy(), base[::-1].copy(), (1.0 - base[::-1]).copy()]
    starts.append(project_K(base + 0.05 * rng.standard_normal(n), n, S))
    starts.append(project_K(base + 0.15 * rng.standard_normal(n), n, S))
    starts.append(project_K(rng.uniform(0, 1, n), n, S))
    starts.append(project_K(rng.beta(0.5, 0.5, n), n, S))
    bv = np.zeros(n); n1 = n // 2
    bv[:n1] = 1.0
    rem = S - n1
    if rem > 1e-12: bv[n1] = rem
    starts.append(project_K(bv[rng.permutation(n)], n, S))
    jj = np.arange(n)
    starts.append(project_K(0.5 + 0.45 * np.cos(2 * np.pi * jj / n), n, S))
    out = []
    for s in starts:
        s = project_K(s, n, S)
        if not feasible(s, n, S): continue
        if any(np.max(np.abs(s - o)) <= 1e-7 for o in out): continue
        out.append(s)
    return out[:10]

def run(seed=42, **kwargs):
    try: rng = np.random.default_rng(seed)
    except Exception: rng = np.random.default_rng(42)
    init = globals().get('initial_h_values', None)
    n = 45; pok = False
    if init is not None:
        try:
            init = np.array(init, dtype=np.float64).ravel()
            if 2 <= len(init) <= 999 and np.all(np.isfinite(init)):
                if init.min() >= -1e-8 and init.max() <= 1 + 1e-8:
                    n = len(init)
                    if abs(init.sum() - n / 2.0) <= 1e-6: pok = True
        except Exception: pass
    if not pok:
        nd = kwargs.get('n_points', 45)
        if isinstance(nd, (int, np.integer)) and not isinstance(nd, bool) and 2 <= nd <= 999:
            n = int(nd)
    n = max(2, min(999, n))
    S = n / 2.0
    base = project_K(init.copy() if pok else np.full(n, 0.5), n, S)
    fb_h, fb_s = base.copy(), exact_score(base, n)
    best_h, best_s = fb_h.copy(), fb_s
    ms = 3 if n <= 64 else 2
    ns = 8 if n <= 64 else 6
    starts = build_starts(base, n, rng)
    for s in starts:
        hs, sc = local_descent(s.copy(), n, rng, ms, ns)
        if sc < best_s - 1e-12: best_s, best_h = sc, hs.copy()
    for sig in [0.04, 0.12]:
        sh = project_K(best_h + sig * rng.standard_normal(n), n, S)
        if feasible(sh, n, S):
            hs, sc = local_descent(sh, n, rng, ms, ns)
            if sc < best_s - 1e-12: best_s, best_h = sc, hs.copy()
    best_h = project_K(best_h, n, S)
    fs = exact_score(best_h, n)
    if not feasible(best_h, n, S) or fs > fb_s + 1e-10:
        best_h, best_s = fb_h.copy(), fb_s
    else:
        best_s = fs
    return (np.clip(best_h, 0.0, 1.0), exact_score(best_h, n), n)
