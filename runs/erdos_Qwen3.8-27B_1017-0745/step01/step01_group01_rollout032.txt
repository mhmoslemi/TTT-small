    for q in (1, 2):
        add_s(0.5 + 0.45 * np.cos(2.0 * np.pi * q * ix / n))
    add_s(rng.uniform(0, 1, n))
    add_s(rng.beta(0.5, 0.5, n))
    best, best_s = fb.copy(), fs
    for st in starts:
        ss = exact_score(st, n)
        if math.isfinite(ss) and ss < best_s - 1e-12:
            best, best_s = st.copy(), ss
    if n == 1 or n > 256:
        f = project_capped(best, n)
        if f is None:
            f = fb.copy()
        return f, float(exact_score(f, n)), int(n)
    # Calibrate temperature
    M = 5 if n <= 64 else (4 if n <= 128 else 3)
    steps = 800 if n <= 64 else (400 if n <= 128 else 120)
    max_restart = 2 if n <= 64 else 1
    pilot_n = 24
    pos_deltas = []
    xref = best.copy()
    ref_s = best_s
    for _ in range(pilot_n):
        v = gen_direction(n, rng)
        if v is None:
            continue
        cb = chord_bounds(xref, v)
        if cb is None:
            continue
        a, b = cb
        if rng.random() < 0.20:
            t = rng.uniform(a, b)
        else:
            rho_p = 0.08
            t = rng.uniform(-rho_p * (b - a), rho_p * (b - a))
            if t < a or t > b:
                continue
        y = xref + t * v
        yr = roundoff_fix(y, n)
        if yr is None:
            continue
        s_y = exact_score(yr, n)
        if math.isfinite(s_y) and s_y > ref_s:
            pos_deltas.append(s_y - ref_s)
        if math.isfinite(s_y) and s_y < best_s - 1e-12:
            best, best_s = yr.copy(), s_y
    if pos_deltas:
        d_med = float(np.median(pos_deltas))
        T_hot = max(1e-5, min(0.02, d_med / (-math.log(0.8))))
    else:
        T_hot = 0.002
    T_cold = max(1e-6, 0.02 * T_hot)
    temps = np.array([T_cold * (T_hot / T_cold) ** (j / (M - 1)) for j in range(M)])
    # Set up replicas
    rep_h = [None] * M
    rep_s = [0.0] * M
    rep_h[0] = best.copy()
    rep_s[0] = best_s
    assigned = 0
    for st in starts:
        if assigned >= M:
            break
        if np.max(np.abs(st - best)) < 1e-10:
            continue
        rep_h[assigned + 1] = st.copy() if assigned + 1 < M else None
        if rep_h[assigned + 1] is not None:
            rep_s[assigned + 1] = exact_score(st, n)
            assigned += 1
    while assigned < M - 1:
        base = best if rng.random() < 0.5 else starts[int(rng.integers(len(starts)))].copy() if starts else best
        sdx = [0.03, 0.08, 0.18][assigned % 3]
        cand = project_capped(base + rng.normal(0, sdx, n), n)
        if cand is None:
            cand = base.copy()
        dup = any(np.max(np.abs(cand - rep_h[k])) < 1e-10 for k in range(assigned + 1) if rep_h[k] is not None)
        if not dup:
            rep_h[assigned + 1] = cand
            rep_s[assigned + 1] = exact_score(cand, n)
            assigned += 1
    for j in range(1, M):
        if rep_h[j] is None:
            rep_h[j] = best.copy()
            rep_s[j] = best_s
    # Main loop
    sweep_budget = max(0, min(int(kwargs.get('steps', steps)), 2000 if n <= 64 else (800 if n <= 128 else 200)))
    patience = max(50, sweep_budget // 5)
    stale = 0
    n_restart_used = 0
    for sweep in range(sweep_budget):
        for j in range(M):
            x = rep_h[j]
            Tj = temps[j]
            rho_j = 0.02 + 0.12 * j / (M - 1)
            v = gen_direction(n, rng)
            if v is None:
                continue
            cb = chord_bounds(x, v)
            if cb is None:
                continue
            a, b = cb
            if rng.random() < 0.20:
                t = rng.uniform(a, b)
            else:
                t = rng.uniform(-rho_j * (b - a), rho_j * (b - a))
                if t < a or t > b:
                    continue
            y = x + t * v
            yr = roundoff_fix(y, n)
            if yr is None or np.max(np.abs(yr - x)) < 1e-14:
                continue
            s_y = exact_score(yr, n)
            if not math.isfinite(s_y):
                continue
            delta = s_y - rep_s[j]
            if delta <= 0:
                rep_h[j] = yr
                rep_s[j] = s_y
            else:
                ratio = -delta / Tj
                if ratio > 745:
                    continue
                if math.log(rng.random()) < ratio:
                    rep_h[j] = yr
                    rep_s[j] = s_y
            if s_y < best_s - 1e-12:
                best, best_s = yr.copy(), s_y
                stale = 0
            else:
                stale += 1
        # Swaps
        parity = sweep % 2
        for j in range(parity, M - 1, 2):
            j2 = j + 1
            log_r = (rep_s[j] - rep_s[j2]) * (1.0 / temps[j] - 1.0 / temps[j2])
            if log_r >= 0:
                do_swap = True
            else:
                if log_r < -745:
                    do_swap = False
                else:
                    do_swap = math.log(rng.random()) < log_r
            if do_swap:
                rep_h[j], rep_h[j2] = rep_h[j2], rep_h[j]
                rep_s[j], rep_s[j2] = rep_s[j2], rep_s[j]
        if stale >= patience:
            if n_restart_used < max_restart:
                for j in range(max(1, M // 2), M):
                    base = best if j % 2 == 0 else (starts[int(rng.integers(len(starts)))].copy() if starts else best)
                    sdx = [0.03, 0.08, 0.18][(j + n_restart_used) % 3]
                    cand = project_capped(base + rng.normal(0, sdx, n), n)
                    if cand is None:
                        cand = base.copy()
                    rep_h[j] = cand
                    rep_s[j] = exact_score(cand, n)
                    if rep_s[j] < best_s - 1e-12:
                        best, best_s = cand.copy(), rep_s[j]
                n_restart_used += 1
                stale = 0
            else:
                break
    f = project_capped(best, n)
    if f is None:
        f = fb.copy()
    final_s = exact_score(f, n)
    if not math.isfinite(final_s) or final_s > fs:
        f = fb.copy()
        final_s = fs
    if not math.isfinite(final_s):
        f = uniform.copy()
        final_s = exact_score(f, n)
    return f, float(final_s), int(n)
