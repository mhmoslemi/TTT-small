import numpy as np
from scipy.optimize import minimize

def project_capped(v, n):
    v = np.asarray(v, dtype=np.float64).ravel()
    if v.size != n:
        return None
    if not np.all(np.isfinite(v)):
        return None
    target = 0.5 * n
    v = np.clip(v, -1e12, 1e12)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if mid == lo or mid == hi:
            break
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    for _ in range(n + 5):
        r = target - float(np.sum(h))
        if abs(r) < 1e-13:
            break
        if r > 0.0:
            idx = int(np.argmax(1.0 - h))
            delta = min(r, 1.0 - h[idx])
            if delta <= 0.0:
                break
            h[idx] += delta
        else:
            idx = int(np.argmax(h))
            delta = min(-r, h[idx])
            if delta <= 0.0:
                break
            h[idx] -= delta
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - target) > 1e-8 * max(1, n):
        return None
    return h

def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.size != n or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-8) or np.any(h > 1.0 + 1e-8):
        return float('inf')
    if abs(float(np.sum(h)) - 0.5 * n) > 1e-8 * max(1, n):
        return float('inf')
    try:
        q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if not np.all(np.isfinite(q)):
            return float('inf')
        return float(np.max(q))
    except Exception:
        return float('inf')

def smooth_core(x, n, tau):
    z = np.asarray(x, dtype=np.float64).ravel()
    if z.size != n or not np.all(np.isfinite(z)):
        return 1e100, None
    if float(np.max(np.abs(z))) > 10.0:
        return 1e100, None
    if tau < 1e-12:
        tau = 1e-12
    L = 2 * n - 1
    dx = 2.0 / n
    omx = 1.0 - z
    vals = np.empty(L, dtype=np.float64)
    idx = 0
    for d in range(-(n - 1), n):
        i0 = max(0, -d)
        j0 = max(0, d)
        m = n - abs(d)
        vals[idx] = dx * float(np.dot(z[i0:i0 + m], omx[j0:j0 + m]))
        idx += 1
    if not np.all(np.isfinite(vals)):
        return 1e100, None
    mx = float(np.max(vals))
    exps = np.exp((vals - mx) / tau)
    s = float(np.sum(exps))
    if not np.isfinite(s) or s <= 0.0:
        return 1e100, None
    w = exps / s
    return mx + tau * float(np.log(s)), w

def smooth_obj(x, n, tau):
    f, _ = smooth_core(x, n, tau)
    return float(f)

def smooth_grad(x, n, tau):
    f, w = smooth_core(x, n, tau)
    if w is None:
        return np.zeros(x.shape if hasattr(x, 'shape') else (n,), dtype=np.float64)
    z = np.asarray(x, dtype=np.float64).ravel()
    dx = 2.0 / n
    omx = 1.0 - z
    grad = np.zeros(n, dtype=np.float64)
    idx = 0
    for d in range(-(n - 1), n):
        i0 = max(0, -d)
        j0 = max(0, d)
        m = n - abs(d)
        pw = dx * w[idx]
        if pw != 0.0:
            grad[i0:i0 + m] += pw * omx[j0:j0 + m]
            grad[j0:j0 + m] -= pw * z[i0:i0 + m]
        idx += 1
    if not np.all(np.isfinite(grad)):
        return np.zeros(n, dtype=np.float64)
    return grad

def mass_fun(x, n):
    return np.array([float(np.sum(x) - 0.5 * n)])

def as_bounds(n):
    return [(0.0, 1.0)] * n

def smooth_stage(x, n, tau, stage_iter):
    h = project_capped(x, n)
    if h is None:
        return None
    if stage_iter <= 0:
        return h
    try:
        cons = [{'type': 'eq', 'fun': mass_fun, 'args': (n,)}]
        res = minimize(smooth_obj, h, args=(n, tau), jac=smooth_grad, method='SLSQP',
                       bounds=as_bounds(n), constraints=cons,
                       options={'maxiter': int(stage_iter), 'ftol': 1e-9, 'disp': False})
    except Exception:
        return h
    if res is None:
        return h
    y = np.asarray(res.x, dtype=np.float64).ravel()
    y = project_capped(y, n)
    if y is None:
        return h
    return y

def add_start(starts, v, n):
    h2 = project_capped(v, n)
    if h2 is None:
        return
    for ex in starts:
        if float(np.max(np.abs(h2 - ex))) < 1e-10:
            return
    starts.append(h2)

def local_polish(h, n, seed, max_sweeps):
    h = project_capped(h, n)
    if h is None:
        return None, float('inf')
    best = exact_score(h, n)
    if not np.isfinite(best):
        return h, best
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(0)
    deltas = (0.07, -0.07, 0.03, -0.03, 0.01, -0.01)

                sc = exact_score(zp, n)
                if sc < best - 1e-12:
                    h = zp
                    best = sc
                    sweep_improved = True
                    break
        if not sweep_improved:
            break
    return h, best

        x = int(value)

    if x < low:

    if x > high:

import numpy as np
from scipy.optimize import minimize


def project_capped(v, n):
    v = np.asarray(v, dtype=np.float64).ravel()
    if v.size != n:
        return None
    if not np.all(np.isfinite(v)):
        return None
    target = 0.5 * n
    v = np.clip(v, -1e12, 1e12)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if mid == lo or mid == hi:
            break
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(n + 5):
        r = target - float(np.sum(h))
        if abs(r) < 1e-13:
            break
        if r > 0.0:
            idx = int(np.argmax(1.0 - h))
            delta = min(r, 1.0 - h[idx])
            if delta <= 0.0:
                break
            h[idx] += delta
        else:
            idx = int(np.argmax(h))
            delta = min(-r, h[idx])
            if delta <= 0.0:
                break
            h[idx] -= delta
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - target) > 1e-8 * max(1, n):
        return None
    return h


def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.size != n or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-8) or np.any(h > 1.0 + 1e-8):
        return float('inf')
    if abs(float(np.sum(h)) - 0.5 * n) > 1e-8 * max(1, n):
        return float('inf')
    try:
        q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if not np.all(np.isfinite(q)):
            return float('inf')
        return float(np.max(q))
    except Exception:
        return float('inf')


def smooth_core(x, n, tau):
    z = np.asarray(x, dtype=np.float64).ravel()
    if z.size != n or not np.all(np.isfinite(z)):
        return 1e100, None
    if float(np.max(np.abs(z))) > 10.0:
        return 1e100, None
    if tau < 1e-12:
        tau = 1e-12
    L = 2 * n - 1
    dx = 2.0 / n
    omx = 1.0 - z
    vals = np.empty(L, dtype=np.float64)
    idx = 0
    for d in range(-(n - 1), n):
        i0 = max(0, -d)
        j0 = max(0, d)
        m = n - abs(d)
        vals[idx] = dx * float(np.dot(z[i0:i0 + m], omx[j0:j0 + m]))
        idx += 1
    if not np.all(np.isfinite(vals)):
        return 1e100, None
    mx = float(np.max(vals))
    exps = np.exp((vals - mx) / tau)
    s = float(np.sum(exps))
    if not np.isfinite(s) or s <= 0.0:
        return 1e100, None
    w = exps / s
    return mx + tau * float(np.log(s)), w


def smooth_obj(x, n, tau):
    f, _ = smooth_core(x, n, tau)
    return float(f)


def smooth_grad(x, n, tau):
    _, w = smooth_core(x, n, tau)
    if w is None:
        return np.zeros(n, dtype=np.float64)
    z = np.asarray(x, dtype=np.float64).ravel()
    dx = 2.0 / n
    omx = 1.0 - z
    grad = np.zeros(n, dtype=np.float64)
    idx = 0
    for d in range(-(n - 1), n):
        i0 = max(0, -d)
        j0 = max(0, d)
        m = n - abs(d)
        pw = dx * w[idx]
        if pw != 0.0:
            grad[i0:i0 + m] += pw * omx[j0:j0 + m]
            grad[j0:j0 + m] -= pw * z[i0:i0 + m]
        idx += 1
    if not np.all(np.isfinite(grad)):
        return np.zeros(n, dtype=np.float64)
    return grad


def mass_fun(x, n):
    return np.array([float(np.sum(x) - 0.5 * n)])


def as_bounds(n):
    return [(0.0, 1.0)] * n


def smooth_stage(x, n, tau, stage_iter):
    h = project_capped(x, n)
    if h is None:
        return None
    if stage_iter <= 0:
        return h
    try:
        cons = [{'type': 'eq', 'fun': mass_fun, 'args': (n,)}]
        res = minimize(
            smooth_obj, h, args=(n, tau), jac=smooth_grad,
            method='SLSQP', bounds=as_bounds(n), constraints=cons,
            options={'maxiter': int(stage_iter), 'ftol': 1e-9, 'disp': False}
        )
    except Exception:
        return h
    if res is None:
        return h
    y = np.asarray(res.x, dtype=np.float64).ravel()
    y = project_capped(y, n)
    if y is None:
        return h
    return y


def add_start(starts, v, n):
    h2 = project_capped(v, n)
    if h2 is None:
        return
    for ex in starts:

            return
    starts.append(h2)


def worst_lag_polish(h, n, max_iter):
    h = project_capped(h, n)
    if h is None:
        return None, float('inf')
    best = exact_score(h, n)
    if not np.isfinite(best):
        return h, best
    alpha = 0.1
    dx = 2.0 / n
    for _ in range(max_iter):
        z = h
        L = 2 * n - 1
        vals = np.empty(L, dtype=np.float64)
        omz = 1.0 - z
        idx = 0
        for d in range(-(n - 1), n):
            i0 = max(0, -d)
            j0 = max(0, d)
            m = n - abs(d)
            vals[idx] = dx * float(np.dot(z[i0:i0 + m], omz[j0:j0 + m]))
            idx += 1
        if not np.all(np.isfinite(vals)):
            break
        imax = int(np.argmax(vals))
        d = -(n - 1) + imax
        i0 = max(0, -d)
        j0 = max(0, d)
        m = n - abs(d)
        grad = np.zeros(n, dtype=np.float64)
        grad[i0:i0 + m] += dx * (1.0 - z[j0:j0 + m])
        grad[j0:j0 + m] -= dx * z[i0:i0 + m]
        trial = project_capped(z - alpha * grad, n)
        sc = exact_score(trial, n)
        if np.isfinite(sc) and sc < best - 1e-12:
            h = trial
            best = sc
            alpha = min(0.8, alpha * 1.3)
        else:
            alpha *= 0.5
            if alpha < 1e-6:
                break
    return h, best


def local_polish(h, n, seed, max_sweeps):
    h = project_capped(h, n)
    if h is None:
        return None, float('inf')
    best = exact_score(h, n)
    if not np.isfinite(best):
        return h, best
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(0)
    deltas = (0.07, -0.07, 0.03, -0.03, 0.01, -0.01)
    for _ in range(max_sweeps):
        sweep_improved = False
        order = rng.permutation(n)
        for i in order:
            for d in deltas:
                z = h.copy()
                z[i] += d
                zp = project_capped(z, n)
                if zp is None:
                    continue
                sc = exact_score(zp, n)
                if sc < best - 1e-12:
                    h = zp
                    best = sc
                    sweep_improved = True
                    break
        if not sweep_improved:
            break
    return h, best


def clamp_int(value, default, low, high):
    try:
        x = int(value)
    except Exception:
        return default
    if x < low:
        return low
    if x > high:
        return high
    return x
