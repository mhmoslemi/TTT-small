        return h, exact_score(h, 1), 1
    if raw is not None and np.all(np.isfinite(raw)):
        baseline = project_box_sum(raw, N, m)
    else:
        baseline = np.full(N, 0.5)
    best = baseline.copy()
    best_sc = exact_score(best, N)
    rng = np.random.default_rng(seed)
    if N <= 256:
        starts = make_starts(baseline, N, m, rng)
        max_outer = 40 if N <= 96 else (25 if N <= 160 else 12)
        for h0 in starts:
            sc0 = exact_score(h0, N)
            if np.isfinite(sc0) and sc0 < best_sc - 1e-12:
                best = h0.copy(); best_sc = sc0
            hr, scr = mm_local(h0, N, m, max_outer)
            if np.isfinite(scr) and scr < best_sc - 1e-12:
                best = hr.copy(); best_sc = scr
        for sig in [0.05, 0.015]:
            pert = project_box_sum(best + sig * rng.standard_normal(N), N, m)
            hr, scr = mm_local(pert, N, m, 24)
            if np.isfinite(scr) and scr < best_sc - 1e-12:
                best = hr.copy(); best_sc = scr
        pert2 = project_box_sum(best + 0.03 * rng.standard_normal(N), N, m)
        hr, scr = mm_local(pert2, N, m, 20)
        if np.isfinite(scr) and scr < best_sc - 1e-12:
            best = hr.copy(); best_sc = scr
    best = project_box_sum(best, N, m)
    c5 = exact_score(best, N)
    bl_sc = exact_score(baseline, N)
    if not np.isfinite(c5) or c5 > bl_sc:
        best = project_box_sum(baseline, N, m)
        c5 = exact_score(best, N)
    if not np.isfinite(c5):
        best = np.full(N, 0.5); c5 = exact_score(best, N)
    return best.astype(np.float64), float(c5), int(N)
