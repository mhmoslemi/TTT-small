            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)

        return float('inf')

    if corr.shape != (2 * n - 1,) or not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr * dx))

import numpy as np
import math

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(h.sum() - n / 2.0) <= 1e-10)

def project_capped(y, n):
    S = n / 2.0
    if y.shape != (n,) or not np.all(np.isfinite(y)) or np.max(np.abs(y)) > 1e12:
        return np.full(n, 0.5), False
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        c = np.clip(y - mid, 0.0, 1.0)
        if c.sum() > S:
            lo = mid
        else:
            hi = mid
    c = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(5):
        r = S - c.sum()
        if abs(r) < 1e-14:
            break
        if r > 0:
            sl = 1.0 - c
            t = sl.sum()
            if t > 1e-14:
                c += r * sl / t
        else:
            t = c.sum()
            if t > 1e-14:
                c += r * c / t
        np.clip(c, 0.0, 1.0, out=c)
    if check_feasible(c, n):
        return c, True
    return np.full(n, 0.5), False

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if corr.shape != (2 * n - 1,) or not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr * dx))

def make_basis(n, d, rng):
    if d >= n:
        d = n - 1
    if d <= 0:
        return None
    B = np.zeros((n, d))
    for j in range(d):
        denom = math.sqrt((j + 1) * (j + 2))
        B[:j + 1, j] = 1.0 / denom
        B[j + 1, j] = -(j + 1) / denom
    return B

def make_basis_rand(n, d, rng):
    for _ in range(3):
        M = rng.standard_normal((n, d))
        M -= M.mean(axis=0)
        try:
            Q, R = np.linalg.qr(M)
            B = Q[:, :d]
            if (np.all(np.isfinite(B)) and np.allclose(B.T @ B, np.eye(d), atol=1e-8)
                    and np.all(np.abs(B.sum(axis=0)) < 1e-6)):
                return B
        except Exception:
            pass
    return None

def cma_restart(anchor, B, sigma0, n, gens, rng):
    d = B.shape[1]
    rho = math.sqrt(n / d)
    lam = max(10, min(16, 4 + int(3.0 * math.log(d))))
    mu = lam // 2
    w = np.zeros(mu)
    for i in range(mu):
        w[i] = math.log(mu + 0.5) - math.log(i + 1)
    w /= w.sum()
    mu_eff = 1.0 / float(np.sum(w * w))
    c_sigma = (mu_eff + 2.0) / (d + mu_eff + 5.0)
    c_c = (4.0 + mu_eff / d) / (d + 4.0 + 2.0 * mu_eff / d)
    c_1 = 2.0 / ((d + 1.3) ** 2 + mu_eff)
    c_mu = min(1.0 - c_1, 2.0 * (mu_eff - 2.0 + 1.0 / mu_eff) / ((d + 2.0) ** 2 + mu_eff))
    d_sigma = 1.0 + 2.0 * max(0.0, math.sqrt((mu_eff - 1.0) / (d + 1.0)) - 1.0) + c_sigma
    chi_d = math.sqrt(d) * (1.0 - 1.0 / (4.0 * d) + 1.0 / (21.0 * d * d))
    m = np.zeros(d)
    C = np.eye(d)
    p_c = np.zeros(d)
    p_sigma = np.zeros(d)
    sigma = sigma0
    resets = 0
    stall = 0
    best_h = anchor.copy()
    best_s = exact_score(anchor, n)
    score_arr = np.full(lam, float('inf'))
    x_arr = np.zeros((lam, d))
    for gen in range(gens):
        C = 0.5 * (C + C.T)
        try:
            ev, evec = np.linalg.eigh(C)
            ev = np.clip(ev, 1e-8, 1e3)
            C = evec @ np.diag(ev) @ evec.T
            A = evec @ np.diag(np.sqrt(ev))
            Cns = evec @ np.diag(1.0 / np.sqrt(ev)) @ evec.T
        except Exception:
            if resets >= 1:
                break
            resets += 1
            C = np.eye(d)
            p_c[:] = 0.0
            p_sigma[:] = 0.0
            sigma *= 0.5
            A = Cns = np.eye(d)
            ev = np.ones(d)
        for j in range(lam):
            if gen == 0 and j == 0:
                x_arr[j] = m
            else:
                z = rng.standard_normal(d)
                x_arr[j] = m + sigma * (A @ z)
            np.clip(x_arr[j], -30.0, 30.0, out=x_arr[j])
            raw = anchor + rho * (B @ x_arr[j])
            if not np.all(np.isfinite(raw)):
                continue
            hc, ok = project_capped(raw, n)
            if not ok:
                continue
            sc = exact_score(hc, n)
            score_arr[j] = sc
            if sc < best_s - 1e-12:
                best_s = sc
                best_h = hc.copy()
        nvalid = int(np.sum(np.isfinite(score_arr)))
        if nvalid < mu:
            break
        order = np.argsort(score_arr)
        sel = order[:mu]
        m_new = np.zeros(d)
        for i in range(mu):
            m_new += w[i] * x_arr[sel[i]]
        y_w = (m_new - m) / sigma
        p_sigma = (1.0 - c_sigma) * p_sigma + math.sqrt(c_sigma * (2.0 - c_sigma) * mu_eff) * (Cns @ y_w)
        g = gen + 1
        denom = math.sqrt(1.0 - (1.0 - c_sigma) ** (2 * g))
        h_sig = 1.0 if np.linalg.norm(p_sigma) / denom < (1.4 + 2.0 / (d + 1.0)) * chi_d else 0.0
        p_c = (1.0 - c_c) * p_c + h_sig * math.sqrt(c_c * (2.0 - c_c) * mu_eff) * y_w
        sum_yy = np.zeros((d, d))
        for i in range(mu):
            yi = (x_arr[sel[i]] - m) / sigma
            sum_yy += w[i] * np.outer(yi, yi)
        fact = 1.0 - c_1 - c_mu + c_1 * (1.0 - h_sig) * c_c * (2.0 - c_c)
        C = fact * C + c_1 * np.outer(p_c, p_c) + c_mu * sum_yy
        C = 0.5 * (C + C.T)
        ea = (c_sigma / d_sigma) * (np.linalg.norm(p_sigma) / chi_d - 1.0)
        ea = max(-2.0, min(2.0, ea))
        sigma = max(1e-4, min(0.5, sigma * math.exp(ea)))
        m = np.clip(m_new, -30.0, 30.0)
        if not np.all(np.isfinite(C)) or not np.all(np.isfinite(m)):
            break
        if rho * sigma * math.sqrt(ev.max()) < 1e-5:
            stall += 1
            if stall >= 5:
                break
        else:
            stall = 0
    return best_h, best_s

        except Exception:
