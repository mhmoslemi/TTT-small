            if delta <= 0:
                accept = True
            else:
                exp_arg = -delta / temps[j]
                accept = (rng.random() < math.exp(exp_arg)) if exp_arg > -745 else False
            if accept:
                states[j] = yr; scores[j] = sy
                if sy < best_score - 1e-12:
                    best, best_score = yr.copy(), sy
                    stagnant = 0
            elif sy < best_score - 1e-12:
                best, best_score = yr.copy(), sy
                stagnant = 0
        # Adjacent swaps
        parity = sweep % 2
        for j in range(parity, M - 1, 2):
            i, k = j, j + 1
            log_r = (scores[i] - scores[k]) * (1.0 / temps[i] - 1.0 / temps[k])
            if log_r >= 0 or math.log(rng.random()) < log_r:
                states[i], states[k] = states[k], states[i]
                scores[i], scores[k] = scores[k], scores[i]
        stagnant += 1
        if stagnant >= patience:
            if rest_count < rest_bud:
                rest_count += 1
                for j in range(max(1, M // 2), M):
                    base = best if rest_count % 2 == 0 else proj_starts[int(rng.integers(0, len(proj_starts)))]
                    pert = base + rng.normal(0, 0.10, n)
                    h2 = project_capped(pert, n)
                    if h2 is None: h2 = base.copy()
                    states[j] = h2; scores[j] = exact_score(h2, n)
                    if scores[j] < best_score - 1e-12:
                        best, best_score = h2.copy(), scores[j]
                stagnant = 0
            else:
                break

    final = project_capped(best, n)
    if final is None: final = uniform.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > best_score: final, fs = best.copy(), best_score
    if not np.isfinite(fs): final, fs = uniform.copy(), exact_score(uniform, n)
    return final, float(fs), int(n)
