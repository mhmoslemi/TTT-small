import numpy as np
import math

def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):
        return None
    if np.max(np.abs(z)) > 1e12:
        return None
    n = len(z)
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(80):
        mid = lo + 0.5 * (hi - lo)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = lo + 0.5 * (hi - lo)
    y = np.clip(z - lam, 0.0, 1.0)
    r = S - np.sum(y)
    for i in range(n):
        if abs(r) < 1e-12:
            break
        if r > 0:
            a = min(r, 1.0 - y[i])
            y[i] += a
            r -= a
        else:
            a = min(-r, y[i])
            y[i] -= a
            r += a
    if abs(np.sum(y) - S) > 1e-10 or not np.all(np.isfinite(y)):
        return None
    return np.clip(y, 0.0, 1.0)

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) - n / 2.0) <= 1e-10)

def exact_lags(h, n):
    vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if vals.shape[0] != 2 * n - 1 or not np.all(np.isfinite(vals)):
        return None, None
    return vals, float(np.max(vals))

def exact_score(h, n):
    _, sc = exact_lags(h, n)
    return sc

def pair_line_search(h, q, p, r, n, dx):
    L = max(-h[p], h[r] - 1.0)
    U = min(1.0 - h[p], h[r])
    if U - L <= 1e-12:
        return None
    # Compute a and b coefficients
    nd = 2 * n - 1
    a = np.zeros(nd)
    b = np.zeros(nd)
    a[n - 1] = -2.0 * dx
    if r != p:
        idx1 = n - 1 + (r - p)
        idx2 = n - 1 + (p - r)
        if 0 <= idx1 < nd:
            a[idx1] = dx
        if 0 <= idx2 < nd:
            a[idx2] = dx
    for d_idx in range(nd):
        d = d_idx - (n - 1)
        t1 = (1.0 - h[p + d]) if 0 <= p + d < n else 0.0
        t2 = (1.0 - h[r + d]) if 0 <= r + d < n else 0.0
        t3 = h[p - d] if 0 <= p - d < n else 0.0
        t4 = h[r - d] if 0 <= r - d < n else 0.0
        b[d_idx] = dx * (t1 - t2 - t3 + t4)
    # Line search
    best_t = 0.0
    best_f = float(np.max(q))
    # Coarse grid
    pts = np.linspace(L, U, 17)
    for t in pts:
        f = float(np.max(q + b * t + a * t * t))
        if f < best_f - 1e-12 or (abs(f - best_f) <= 1e-12 and abs(t) < abs(best_t)):
            best_f = f
            best_t = t
    # Refine
    radius = (U - L) / 16.0
    for _ in range(4):
        lo_r = max(L, best_t - radius)
        hi_r = min(U, best_t + radius)
        if hi_r - lo_r < 1e-14:
            break
        pts_r = np.linspace(lo_r, hi_r, 5)
        for t in pts_r:
            f = float(np.max(q + b * t + a * t * t))
            if f < best_f - 1e-12 or (abs(f - best_f) <= 1e-12 and abs(t) < abs(best_t)):
                best_f = f
                best_t = t
        radius /= 4.0
    if best_f >= float(np.max(q)) - 1e-12:
        return None
    # Construct new h
    h_new = h.copy()
    h_new[p] += best_t
    h_new[r] -= best_t
    S = n / 2.0
    if not is_feasible(h_new, n):
        h_new = project_capped_simplex(h_new, S)
        if h_new is None:
            return None
    q_new, sc_new = exact_lags(h_new, n)
    if q_new is None:
        return None
    if sc_new >= float(np.max(q)) - 1e-12:
        return None
    return h_new, sc_new, q_new

def build_starts(n, S, parent, rng):
    starts = []
    if parent is not None:
        starts.append(parent.copy())
    # Low-frequency
    idx = np.arange(n, dtype=np.float64)
    M = min(4, (n - 1) // 2)
    per = np.full(n, 0.5)
    for m in range(1, M + 1):
        phi = float(rng.uniform(0.0, 2.0 * np.pi))
        A = rng.normal(0.0, 0.12 / math.sqrt(m))
        per += A * np.cos(2.0 * np.pi * m * (idx + 0.5) / n + phi)
    pp = project_capped_simplex(per, S)
    if pp is not None:
        starts.append(pp)
    # Random
    rr = rng.random(n)
    pr = project_capped_simplex(rr, S)
    if pr is not None:
        starts.append(pr)
    # Binary
    perm = rng.permutation(n)
    bnv = np.zeros(n)
    bnv[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        bnv[perm[n // 2]] = 0.5
    starts.append(bnv.copy())
    if parent is not None:
        starts.append(parent[::-1].copy())
    # Deduplicate
    deduped = []
    for s in starts:
        ok = True
        for d in deduped:
            if np.max(np.abs(s - d)) < 1e-10:
                ok = False
                break
        if ok and is_feasible(s, n):
            deduped.append(s)
    return deduped

def run(seed=42, **kwargs):
    if seed is None:
        seed = 42
    try:
        seed = int(seed)
    except (TypeError, ValueError):
        seed = 42
    seed = seed % (2**32)
    rng = np.random.default_rng(seed)
    n = None
    parent = None
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        try:
            iv_arr = np.asarray(iv, dtype=np.float64)

                        parent = iv_arr.copy()
                    else:

                        if proj is not None and is_feasible(proj, n):
                            parent = proj
        except Exception:

    if n is None:
        try:
            n = int(kwargs.get('n_points', 80))
        except (TypeError, ValueError):
            n = 80
        if n < 8 or n > 999:
            n = 80
    S = n / 2.0
    dx = 2.0 / n
    h_const = np.full(n, 0.5)
    best_h = h_const.copy()
    _, best_score = exact_lags(h_const, n)
    if best_score is None:
        best_score = 0.5
    if parent is not None:
        _, ps = exact_lags(parent, n)
        if ps is not None and ps < best_score:
            best_score = ps
            best_h = parent.copy()
    if n <= 128:
        cap = 6
    elif n <= 256:
        cap = 4
    elif n <= 512:
        cap = 3
    else:
        cap = 2

    for s in starts:
        _, sc = exact_lags(s, n)
        if sc is not None and sc < best_score:
            best_score = sc
            best_h = s.copy()
    # Determine parameters
    if n <= 128:

    elif n <= 256:
        E, extra_sweeps = 4, 3
    elif n <= 512:

        h_cur = h0.copy()
        q_cur, sc_cur = exact_lags(h_cur, n)
        if q_cur is None:
            continue
        scale = 0.12
        for e in range(E):
            T = max(5e-5, 3e-3 * (0.4 ** e))
            accepted = 0
            for _ in range(K):
                p = int(rng.integers(0, n))
                while True:
                    r = int(rng.integers(0, n))
                    if r != p:
                        break
                t = float(rng.uniform(-scale, scale))
                L = max(-h_cur[p], h_cur[r] - 1.0)
                U = min(1.0 - h_cur[p], h_cur[r])
                if t < L or t > U:
                    continue
                h_new = h_cur.copy()
                h_new[p] += t
                h_new[r] -= t
                if not is_feasible(h_new, n):

                    if h_new is None:

                if sc_new is None:
                    continue
                if sc_new < best_score:
                    best_score = sc_new
                    best_h = h_new.copy()
                delta = sc_new - sc_cur
                if delta <= 0 or (delta < 700 * T and rng.random() < math.exp(-delta / T)):
                    h_cur = h_new
                    sc_cur = sc_new
                    q_cur, _ = exact_lags(h_cur, n)
                    if q_cur is None:
                        q_cur, _ = exact_lags(h0, n)
                        h_cur = h0.copy()
                        sc_cur = float(np.max(q_cur))
                        break
                    accepted += 1
            ratio = accepted / max(1, K)
            if ratio < 0.12:
                scale = min(0.4, 1.25 * scale)
            elif ratio > 0.55:
                scale = max(0.01, 0.75 * scale)
            # One polishing sweep
            pairs = build_pairs(n, P, rng)
            for (pi, ri) in pairs:
                res = pair_line_search(h_cur, q_cur, pi, ri, n, dx)
                if res is not None:
                    h_cur, sc_cur, q_cur = res
                    if sc_cur < best_score:
                        best_score = sc_cur
                        best_h = h_cur.copy()
    
    # Final polishing from best
    h_final = best_h.copy()
    q_final, sc_final = exact_lags(h_final, n)
    if q_final is not None:
        for _ in range(extra_sweeps):
            improved = False
            pairs = build_pairs(n, P, rng)
            for (pi, ri) in pairs:
                res = pair_line_search(h_final, q_final, pi, ri, n, dx)
                if res is not None:
                    h_final, sc_final, q_final = res
                    improved = True
            if not improved:
                break
    if not is_feasible(h_final, n):
        h_final = project_capped_simplex(h_final, S)
        if h_final is None:
            h_final = h_const.copy()
    sc_final = exact_score(h_final, n)
    if sc_final is None:
        if parent is not None and is_feasible(parent, n):
            h_final = parent.copy()
            sc_final = exact_score(parent, n)
        if sc_final is None:
            h_final = h_const.copy()
            sc_final = 0.5

def build_pairs(n, P, rng):
    pairs = []
    # Adjacent pairs
    n_adj = min(n - 1, max(1, P // 2))
    if n_adj == n - 1:
        adj = list(range(n - 1))
    else:
        adj = [int(round(i * (n - 2) / max(1, n_adj - 1))) for i in range(n_adj)]
        adj = list(set(adj))
        rng.shuffle(adj)
        adj = adj[:n_adj]
    for i in adj:
        pairs.append((i, i + 1))
    # Fill remaining with random
    remaining = P - len(pairs)
    existing = set(pairs)
    for _ in range(min(remaining * 3, 5000)):
        if len(pairs) >= P:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        if i > j:
            i, j = j, i
        if (i, j) not in existing:
            pairs.append((i, j))
            existing.add((i, j))
    # Fallback enumeration
    if len(pairs) < P:
        for i in range(n):
            if len(pairs) >= P:
                break
            for j in range(i + 1, n):
                if len(pairs) >= P:
                    break
                if (i, j) not in existing:
                    pairs.append((i, j))
                    existing.add((i, j))
    rng.shuffle(pairs)
