            if use_slsqp:
                cur, cur_score = st.copy(), exact_score(st, n)
                for _ in range(3):
                    y = slsqp_epigraph(cur, n, maxiter_sl)
                    if y is None or not is_feasible(y, n):
                        break
                    best_z, best_s = cur, cur_score
                    for k in range(6):
                        alpha = 1.0 / (2.0 ** k)
                        trial = project_capped(cur + alpha * (y - cur), n)
                        if trial is None or not is_feasible(trial, n):
                            continue
                        ts = exact_score(trial, n)
                        if ts < best_s - 1e-12:
                            best_s, best_z = ts, trial
                    if best_s < cur_score - 1e-12:
                        cur, cur_score = best_z, best_s
                        if best_s < gs - 1e-12:
                            gs, gb = best_s, best_z.copy()
                    else:
                        break
            else:
                h_out, s_out = subgrad_step(st, n, max_iter_sg)
                if h_out is not None and s_out < gs - 1e-12:
                    gs, gb = s_out, h_out.copy()
        except Exception:
            pass
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n) or exact_score(final, n) > gs:
        final = project_capped(fallback, n)
    if final is None or not is_feasible(final, n):
        final = np.full(n, 0.5)
    return final, float(exact_score(final, n)), int(n)
