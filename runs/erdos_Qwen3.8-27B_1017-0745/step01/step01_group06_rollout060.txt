            pass
    if ok:
        n0 = len(init)
        hp = project_K(init.copy(), n0, n0 / 2.0)
    else:
        n0 = 45
        hp = np.full(n0, 0.5)
    resolutions = [n0]
    for nm in [n0 - 1, n0 + 1]:
        if 2 <= nm <= 999 and nm not in resolutions:
            resolutions.append(nm)
    best_h, best_n, best_s = hp.copy(), n0, exact_score(hp, n0)

    for m in resolutions:
        S = m / 2.0
        base = transfer_h(hp, n0, m) if m != n0 else hp.copy()
        base = project_K(base, m, S)
        starts = [base.copy(), (1.0 - base).copy(), base[::-1].copy()]
        starts.append(project_K(base + 0.08 * rng.standard_normal(m), m, S))
        starts.append(project_K(base + 0.25 * rng.standard_normal(m), m, S))
        starts.append(project_K(rng.uniform(0, 1, m), m, S))
        starts.append(project_K(rng.beta(0.5, 0.5, m), m, S))
        blk = np.zeros(m)
        blk[:int(S)] = 1.0
        if m % 2 == 1:
            blk[int(S)] = 0.5
        starts.append(project_K(blk.copy(), m, S))
        jj = np.arange(m)
        starts.append(project_K(np.clip(0.5 + 0.4 * np.cos(2 * np.pi * jj / m), 0, 1), m, S))
        ded = []
        seen = []
        for s in starts:
            dup = any(np.max(np.abs(s - t)) <= 1e-6 for t in seen)
            if not dup and feasible(s, m, S):
                ded.append(s)
                seen.append(s)
        mi = 28 if m <= 96 else (12 if m <= 256 else 5)
        si = 50 if m <= 96 else (30 if m <= 256 else 15)
        ns = 0
        for s in ded:
            if ns >= (8 if m <= 96 else (3 if m <= 256 else 1)):
                break
            ns += 1
            h1 = trust_local(s, m, mi, si)
            if feasible(h1, m, S):
                sc = exact_score(h1, m)
                if sc < best_s - 1e-12:
                    best_h, best_n, best_s = h1.copy(), m, sc
        if m <= 96:
            h1 = trust_local(project_K(base + 0.18 * rng.standard_normal(m), m, S), m, mi, si)
            if feasible(h1, m, S):
                sc = exact_score(h1, m)
                if sc < best_s - 1e-12:
                    best_h, best_n, best_s = h1.copy(), m, sc
        h2 = polish_slsqp(best_h if best_n == m else transfer_h(best_h, best_n, m), m, 150)
        if feasible(h2, m, S):
            sc = exact_score(h2, m)
            if sc < best_s - 1e-12:
                best_h, best_n, best_s = h2.copy(), m, sc

    best_h = project_K(best_h, best_n, best_n / 2.0)
    if not feasible(best_h, best_n, best_n / 2.0):
        best_h, best_n, best_s = hp.copy(), n0, exact_score(hp, n0)
    else:
        fs = exact_score(best_h, best_n)
        if fs > best_s + 1e-10:
            best_h, best_n, best_s = hp.copy(), n0, exact_score(hp, n0)
        else:
            best_s = fs
    best_h = np.clip(best_h, 0.0, 1.0)
    return (best_h, exact_score(best_h, best_n), best_n)
