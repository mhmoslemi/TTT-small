To do better, we need h to be non-uniform. The idea is to create a pattern where h is high in some regions and low in others, arranged so that for every shift k, the overlap between h and 1-h is small.

This is related to the concept of "cross-correlation minimization" or "autospectral design."

For the n=80 case, the discrete problem is:
Minimize max_d Σ_i h[i](1-h[i+d]) subject to 0≤h≤1, Σh = n/2.

This is a nonconvex problem (the objective is the max of bilinear terms).

- gr = zeros(n); gr[right_idx] += -h[left_idx]

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and h.min() >= -1e-12 and h.max() <= 1.0 + 1e-12
            and abs(h.sum() - n * 0.5) <= 1e-9)

def exact_score(h):
    n = len(h)
    v = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(v)):
        return None
    return float(v.max())

def build_pairs(n):
    idx = np.arange(n, dtype=np.int32)
    L = np.repeat(idx, n)
    R = np.tile(idx, n)
    lid = (R - L + n - 1).astype(np.int32)
    return L, R, lid

def comp_q(h, L, R, lid, dx, n):
    B = np.bincount(lid, weights=h[L] * (1.0 - h[R]), minlength=2 * n - 1)
    return dx * B

def lse_val(h, tau, L, R, lid, dx, n):
    q = comp_q(h, L, R, lid, dx, n)
    m = float(q.max())
    if not np.isfinite(m): return None
    s = np.sum(np.exp((q - m) / tau))
    return m + tau * float(np.log(s))

def lse_grad(h, tau, L, R, lid, dx, n):
    q = comp_q(h, L, R, lid, dx, n)
    m = float(q.max())
    e = np.exp((q - m) / tau)
    p = e / float(e.sum())
    gl = np.bincount(L, weights=p * (1.0 - h[R]), minlength=n).astype(np.float64)
    gr = np.bincount(R, weights=-p * h[L], minlength=n).astype(np.float64)
    g = dx * (gl + gr)
    if not np.all(np.isfinite(g)): return None
    return g

def exact_score(h):
    n = len(h)
    v = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(v)):
        return None
    return float(v.max())

def build_pairs(n):
    idx = np.arange(n, dtype=np.int32)
    return np.repeat(idx, n), np.tile(idx, n), (np.tile(idx, n) - np.repeat(idx, n) + n - 1).astype(np.int32)

def comp_q(h, L, R, lid, dx, n):
    return dx * np.bincount(lid, weights=h[L] * (1.0 - h[R]), minlength=2 * n - 1)

def lse_val(h, tau, L, R, lid, dx, n):
    q = comp_q(h, L, R, lid, dx, n)
    m = float(q.max())
    return m + tau * float(np.log(np.sum(np.exp((q - m) / tau))))
