    return y

def check_feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(float(np.sum(h)) - 0.5 * n) < 1e-10 * max(1, n))

def exact_score(h):
    n = len(h)
    if not check_feasible(h, n):
        return float('inf')
    try:
        c = np.correlate(h, 1.0 - h, mode='full')
        if len(c) != 2 * n - 1 or not np.all(np.isfinite(c)):
            return float('inf')
        return float(np.max(c) * (2.0 / n))
    except Exception:
        return float('inf')

def lag_value_grad(h, s, n):
    dx = 2.0 / n
    lo = max(0, -s)
    hi = min(n, n - s)
    q = 0.0
    g = np.zeros(n)
    for i in range(lo, hi):
        j = i + s
        q += h[i] * (1.0 - h[j])
        g[i] += (1.0 - h[j])
        g[j] -= h[i]
    return dx * q, dx * g

def get_parent():
    n = 45
    ph = globals().get('initial_h_values', None)
    if ph is not None:
        ph = np.asarray(ph, dtype=np.float64)
        if ph.ndim == 1 and 2 <= len(ph) <= 999 and np.all(np.isfinite(ph)):
            n = len(ph)
            if np.all(ph >= -1e-8) and np.all(ph <= 1.0 + 1e-8):
                p = project_box_sum(np.clip(ph, 0.0, 1.0), 0.5 * n)
                if p is not None and check_feasible(p, n):
                    return p, n
    return project_box_sum(np.full(n, 0.5), 0.5 * n), n

_mm_x = None
_mm_L = 0.0
_mm_GJ = None
_mm_qJ = None
_mm_n = 0

def mm_obj(z):
    y = z[:_mm_n]
    d = y - _mm_x
    return 0.5 * _mm_L * float(d @ d) + z[_mm_n]

def mm_obj_grad(z):
    y = z[:_mm_n]
    g = np.zeros(len(z))
    g[:_mm_n] = _mm_L * (y - _mm_x)
    g[_mm_n] = 1.0
    return g

def mm_ineq(z):
    y = z[:_mm_n]
    d = y - _mm_x
    return z[_mm_n] - _mm_qJ - _mm_GJ @ d

def mm_ineq_jac(z):
    J = np.zeros((len(_mm_qJ), len(z)))
    J[:, :_mm_n] = -_mm_GJ
    J[:, _mm_n] = 1.0
    return J

def mm_eq(z):
    return float(np.sum(z[:_mm_n])) - 0.5 * _mm_n

def mm_eq_grad(z):
    g = np.zeros(len(z))
    g[:_mm_n] = 1.0
    return g

def solve_mm(x, n):
    global _mm_x, _mm_L, _mm_GJ, _mm_qJ, _mm_n
    _mm_n = n
    _mm_x = x
    dx = 2.0 / n
    _mm_L = 2.0 * dx
    lags = list(range(-(n - 1), n))
    vals, grads = [], []
    for s in lags:
        q, g = lag_value_grad(x, s, n)
        vals.append(q)
        grads.append(g)
    vals = np.array(vals)
    grads = np.array(grads)
    if n <= 64:
        idx = np.arange(len(lags))
    else:
        k = min(2 * n - 1, 96)
        idx = np.argsort(-vals, kind='stable')[:k]
    _mm_GJ = grads[idx]
    _mm_qJ = vals[idx]
    y0 = x.copy()
    t0 = float(np.max(_mm_qJ))
    z0 = np.append(y0, t0)
    bounds = [(0.0, 1.0)] * n + [(-np.inf, np.inf)]
    cons = [{'type': 'ineq', 'fun': mm_ineq, 'jac': mm_ineq_jac},
            {'type': 'eq', 'fun': mm_eq, 'jac': mm_eq_grad}]
    opts = {'maxiter': 150, 'ftol': 1e-11, 'disp': False}
    try:
        res = minimize(mm_obj, z0, jac=mm_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons, options=opts)
        y = np.array(res.x[:n], dtype=np.float64)
    except Exception:
        return x.copy()
    if not np.all(np.isfinite(y)):
        return x.copy()
    y = project_box_sum(y, 0.5 * n)
    if y is None:
        return x.copy()
    return y

def local_mm_descent(start, n, max_iter=35):
    x = start.copy()
    cx = exact_score(x)
    for _ in range(max_iter):
        y = solve_mm(x, n)
        if np.max(np.abs(y - x)) < 1e-10:
            break
        accepted = False
        for k in range(10):
            alpha = 2.0 ** (-k)
            cand = project_box_sum(x + alpha * (y - x), 0.5 * n)
            if cand is None:
                continue
            c = exact_score(cand)
            if c < cx - 1e-12:
                x = cand.copy()
                cx = c
                accepted = True
                break
        if not accepted:
            break
        if k >= 9:
            break
    return x, cx

def build_starts(parent, n, rng):
    starts = [parent.copy(), (1.0 - parent).copy(), parent[::-1].copy()]
    for sd in [0.05, 0.10, 0.20, 0.30]:
        p = project_box_sum(parent + sd * rng.standard_normal(n), 0.5 * n)
        if p is not None:
            starts.append(p)
    p = project_box_sum(rng.random(n), 0.5 * n)
    if p is not None:
        starts.append(p)
    i = np.arange(n)
    u = 0.5 + 0.30 * np.cos(2.0 * np.pi * i / n + 0.7) + 0.12 * np.sin(4.0 * np.pi * i / n + 1.3)
    p = project_box_sum(u, 0.5 * n)
    if p is not None:
        starts.append(p)
    u2 = 0.5 + 0.25 * np.cos(4.0 * np.pi * i / n + 2.1) + 0.10 * np.sin(6.0 * np.pi * i / n + 0.4)
    p2 = project_box_sum(u2, 0.5 * n)
    if p2 is not None:
        starts.append(p2)
    u3 = 0.5 + 0.35 * np.cos(2.0 * np.pi * i / n) * np.cos(6.0 * np.pi * i / n)
    p3 = project_box_sum(u3, 0.5 * n)
    if p3 is not None:
        starts.append(p3)
    unique = []
    for s in starts:
        if s is None:
            continue
        dup = False
        for u in unique:
            if np.max(np.abs(s - u)) < 1e-10:
                dup = True
                break
        if not dup:
            unique.append(s)
    return unique

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent, n = get_parent()
    best_h = parent.copy()
    best_c5 = exact_score(parent)
    starts = build_starts(parent, n, rng)
    for h0 in starts:
        x, cx = local_mm_descent(h0, n, max_iter=35)
        if cx < best_c5:
            best_h = x.copy()
            best_c5 = cx
    for sd in [0.04, 0.10, 0.25]:
        p = project_box_sum(best_h + sd * rng.standard_normal(n), 0.5 * n)
        if p is None:
            continue
        x, cx = local_mm_descent(p, n, max_iter=20)
        if cx < best_c5:
            best_h = x.copy()
            best_c5 = cx
    for sd in [0.02, 0.06]:
        p = project_box_sum(best_h + sd * rng.standard_normal(n), 0.5 * n)
        if p is None:
            continue
        x, cx = local_mm_descent(p, n, max_iter=10)
        if cx < best_c5:
            best_h = x.copy()
            best_c5 = cx
    best_h = project_box_sum(best_h, 0.5 * n)
    if best_h is not None and check_feasible(best_h, n):
        final_c5 = exact_score(best_h)
        if np.isfinite(final_c5):
            best_c5 = final_c5
    else:
        best_h = parent.copy()
        best_c5 = exact_score(best_h)
    if not np.isfinite(best_c5) or best_c5 > 0.5:
        fb = project_box_sum(np.full(n, 0.5), 0.5 * n)
        if fb is not None:
            best_h = fb
            best_c5 = exact_score(fb)
    return (best_h, best_c5, n)
