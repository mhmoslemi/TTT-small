def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-15:
        if resid > 0:
            order = np.argsort(1.0 - y)
            for i in order:
                add = min(resid, 1.0 - y[i])
                y[i] += add
                resid -= add
                if abs(resid) < 1e-15:
                    break
        else:
            order = np.argsort(y)
            for i in order:
                sub = min(-resid, y[i])
                y[i] -= sub
                resid += sub
                if abs(resid) < 1e-15:
                    break
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(np.sum(y) - S) > 1e-9 * n:
        return None
    return y

def exact_score(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = float(np.max(corr) * 2.0 / n)
    return s if np.isfinite(s) else float('inf')

def check_feasible(h, n):
    if h is None or h.shape != (n,) or not np.all(np.isfinite(h)):

    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-9 * n:
        return False
    return True

def prod_idx(i, j, n):
    return i * n - i * (i - 1) // 2 + (j - i)

def solve_lp(x, r, n, S):
    l = np.maximum(0.0, x - r)
    u = np.minimum(1.0, x + r)
    P = n * (n + 1) // 2
    nv = n + P + 1
    t_idx = n + P
    rows, cols, vals, rhs = [], [], [], []
    for i in range(n):
        for j in range(i, n):
            pi = n + prod_idx(i, j, n)
            li, lj, ui, uj = l[i], l[j], u[i], u[j]
            rows.append(len(rhs)); cols += [i, j, pi]; vals += [lj, li, -1.0]; rhs.append(li * lj)
            rows.append(len(rhs)); cols += [i, j, pi]; vals += [uj, ui, -1.0]; rhs.append(ui * uj)
            rows.append(len(rhs)); cols += [i, j, pi]; vals += [-lj, -ui, 1.0]; rhs.append(-ui * lj)
            rows.append(len(rhs)); cols += [i, j, pi]; vals += [-uj, -li, 1.0]; rhs.append(-li * uj)
    for k in range(-(n - 1), n):
        i_lo, i_hi = max(0, k), min(n - 1, n - 1 + k)
        if i_lo > i_hi:
            continue
        row = len(rhs)
        for i in range(i_lo, i_hi + 1):
            j = i - k
            a, b = min(i, j), max(i, j)
            cols += [i, n + prod_idx(a, b, n), t_idx]
            vals += [1.0, -1.0, -1.0]
        rows.extend([row] * (i_hi - i_lo + 1) + [row])
        rhs.append(0.0)
    A_ub = coo_matrix((vals, (rows, cols)), shape=(len(rhs), nv)).tocsr()
    b_ub = np.array(rhs)
    A_eq = coo_matrix((np.ones(n), (np.arange(n), np.arange(n))), shape=(1, nv)).tocsr()
    b_eq = np.array([S])
    c_obj = np.zeros(nv)
    c_obj[t_idx] = 1.0
    lb = np.concatenate([l, np.zeros(P), [0.0]])
    ub = np.concatenate([u, np.zeros(P), [float(n)]])

def line_minimax(x, y, n, S):
    d = y - x
    n_lags = 2 * n - 1

    B = np.zeros(n_lags)
    D = np.zeros(n_lags)
    for k in range(-(n - 1), n):
        ki = k + n - 1

        i = np.arange(i_lo, i_hi + 1)
        j = i - k
        D[ki] = np.sum(x[i] * (1.0 - x[j]))
        B[ki] = np.sum(d[i] * (1.0 - x[j]) - x[i] * d[j])
        A[ki] = -np.sum(d[i] * d[j])
    alphas = [0.0, 1.0]
    for ki in range(n_lags):
        if abs(A[ki]) > 1e-14:
            a = -B[ki] / (2.0 * A[ki])
            if -1e-9 <= a <= 1.0 + 1e-9:
                alphas.append(float(np.clip(a, 0.0, 1.0)))
    for p in range(n_lags):
        for q in range(p + 1, n_lags):
            da, db, dc = A[p] - A[q], B[p] - B[q], D[p] - D[q]
            if abs(da) > 1e-14:
                disc = db * db - 4.0 * da * dc
                if disc >= -1e-10:
                    disc = max(0.0, disc)
                    sq = np.sqrt(disc)
                    for a in [(-db - sq) / (2.0 * da), (-db + sq) / (2.0 * da)]:
                        if -1e-9 <= a <= 1.0 + 1e-9:
                            alphas.append(float(np.clip(a, 0.0, 1.0)))
            elif abs(db) > 1e-14:
                a = -dc / db
                if -1e-9 <= a <= 1.0 + 1e-9:
                    alphas.append(float(np.clip(a, 0.0, 1.0)))
    alphas = np.array(alphas)
    alphas = np.clip(alphas, 0.0, 1.0)
    alphas = np.unique(np.round(alphas, 10))
    best_a, best_f = 0.0, float('inf')
    for a in alphas:
        Q = A * a * a + B * a + D
        f = float(np.max(Q))
        if f < best_f:
            best_f = f
            best_a = float(a)
    h_new = project_box_sum(x + best_a * d, S)
    if h_new is None:
        h_new = project_box_sum(y, S)
    return h_new
