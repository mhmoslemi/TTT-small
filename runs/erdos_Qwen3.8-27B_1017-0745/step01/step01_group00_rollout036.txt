    tryadd(fb[::-1])
    tryadd(1.0 - fb[::-1])
    tryadd(np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    tryadd(blk)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    tryadd(alt)
    for sd in (0.06, 0.15, 0.30):
        tryadd(fb + rng.normal(0, sd, n))
    tryadd(rng.uniform(0, 1, n))
    if n % 2 == 0:
        for _ in range(4):
            pat = np.zeros(n)
            pat[rng.permutation(n)[:n // 2]] = 1.0
            tryadd(pat)
    else:
        for _ in range(4):
            pat = np.zeros(n)
            perm = rng.permutation(n)
            pat[perm[:n // 2]] = 1.0
            pat[perm[n // 2]] = 0.5
            tryadd(pat)
    return pool[:max(count, 1)]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    fb = project_capped(parent, n) if parent is not None else None
    if fb is None:
        fb = np.full(n, 0.5)
    rng = np.random.default_rng(int(seed))
    if n <= 64:
        n_starts, n_moves, n_screen, n_random, n_hops, hop_moves = 10, 24, 5, 3, 6, 10
    elif n <= 128:
        n_starts, n_moves, n_screen, n_random, n_hops, hop_moves = 7, 16, 5, 2, 4, 7
    elif n <= 256:
        n_starts, n_moves, n_screen, n_random, n_hops, hop_moves = 4, 8, 4, 2, 3, 4
    else:
        n_starts, n_moves, n_screen, n_random, n_hops, hop_moves = 3, 5, 3, 1, 2, 3
    starts = build_starts(fb, n, rng, n_starts)
    gbest, gscore = fb.copy(), exact_score(fb, n)
    starts.sort(key=lambda x: exact_score(x, n))
    for st in starts:
        s = exact_score(st, n)
        if s < gscore - 1e-12:
            gscore, gbest = s, st.copy()
    for st in starts:
        xo, so = local_descent(st, n, rng, n_moves, n_screen, n_random)
        if so < gscore - 1e-12:
            gscore, gbest = so, xo.copy()
    walker, wscore = gbest.copy(), gscore
    T0 = 0.0015
    sds = [0.04, 0.10, 0.20, 0.30]
    scale = max(0.5, min(1.5, np.sqrt(47.0 / n)))
    for hop in range(n_hops):
        sd = sds[hop % 4] * scale
        prop = walker + rng.normal(0, sd, n)
        prop = project_capped(prop, n)
        if prop is None or not is_feasible(prop, n):
            continue
        xo, ps = local_descent(prop, n, rng, hop_moves, n_screen, max(1, n_random // 2))
        if ps < gscore - 1e-12:
            gscore, gbest = ps, xo.copy()
        delta = ps - wscore
        if delta <= 1e-12:
            walker, wscore = xo.copy(), ps
        else:
            T = T0 * (0.25 + 0.75 * (1.0 - hop / max(1, n_hops)))
            if rng.random() < np.exp(-min(700.0, delta / T)):
                walker, wscore = xo.copy(), ps
    gbest = project_capped(gbest, n)
    if gbest is None or not is_feasible(gbest, n) or exact_score(gbest, n) > exact_score(fb, n) + 1e-12:
        gbest = fb
    gscore = exact_score(gbest, n)
    if gscore == float('inf'):
        gbest = np.full(n, 0.5)
        gscore = exact_score(gbest, n)
    return gbest, float(gscore), int(n)
