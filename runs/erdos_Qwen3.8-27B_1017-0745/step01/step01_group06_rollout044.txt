            lo = mid
        else:
            hi = mid

    nl = 2*n-1

    j = np.arange(n)
    for k in range(nl):
        ell = k-(n-1)
        i1, v1 = j-ell, (j-ell >= 0)&(j-ell < n)
        i2, v2 = j+ell, (j+ell >= 0)&(j+ell < n)
        t1, t2 = np.zeros(n), np.zeros(n)
        t1[v1] = 1.0-h[i1[v1]]
        t2[v2] = h[i2[v2]]
        G[k] = dx*(t1-t2)

    j = np.arange(n)
    for k in range(2 * n - 1):

        t1, t2 = np.zeros(n), np.zeros(n)

        t1[v1] = 1.0 - h[j[v1] - ell]
        t2[v2] = h[j[v2] + ell]
