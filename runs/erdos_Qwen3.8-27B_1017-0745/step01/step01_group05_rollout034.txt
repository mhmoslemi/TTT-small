        j_hi = min(n, hi - k)

        return np.array([], dtype=np.float64)

    rho = np.where(s - (css - 1.0) / np.arange(1, m + 1) > 0)[0]

    r = rho[0]
    theta = (css[r] - 1.0) / (r + 1.0)
    p = np.maximum(z - theta, 0.0)
    p /= p.sum()
    return p

        if r > 0:
            lo = lam
        else:
            hi = lam
        a = int(np.sum((y > lam) & (y < lam + 1.0)))
        if a > 0:
            lam_new = lam + r / a
        else:
            lam_new = 0.5 * (lo + hi)
        if np.isfinite(lam_new) and lo < lam_new < hi:
            lam = lam_new
        else:
            lam = 0.5 * (lo + hi)
    c = np.clip(y - lam, 0.0, 1.0).astype(np.float64)
    for _ in range(3):
        r = S - c.sum()
        if abs(r) < 1e-14:
            break
        mask = c < 1.0 - 1e-12 if r > 0 else c > 1e-12
        cnt = int(mask.sum())
        if cnt == 0:
            break
        c[mask] += r / cnt
        np.clip(c, 0.0, 1.0, out=c)
    ok = check_feasible(c, n)
    if not ok:
        c = np.full(n, 0.5)
    return c, True

def lag_scores(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    q = lag_scores(h, n)
    if not np.all(np.isfinite(q)):
        return float('inf')
    return float(q.max())

def build_G(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        row = k + n - 1
        lo = max(0, k)
        hi = min(n, n + k)
        if hi <= lo:
            continue
        G[row, lo:hi] += dx * (1.0 - h[lo - k:hi - k])
        j_lo = max(0, lo - k)
        j_hi = min(n, hi - k)
        if j_hi > j_lo:
            G[row, j_lo:j_hi] -= dx * h[j_lo + k:j_hi + k]
    return G

def project_simplex(z):
    m = len(z)
    if m <= 1:
        return np.ones(m, dtype=np.float64) / max(m, 1)
    s = np.sort(z)[::-1].astype(np.float64)
    css = np.cumsum(s)
    idx = np.arange(1, m + 1, dtype=np.float64)
    rho_mask = s - (css - 1.0) / idx > 0
    if not rho_mask.any():
        return np.full(m, 1.0 / m, dtype=np.float64)
    r = int(np.where(rho_mask)[0][0])
    theta = (css[r] - 1.0) / (r + 1.0)
    p = np.maximum(z - theta, 0.0).astype(np.float64)
    ps = p.sum()
    if ps < 1e-15:
        return np.full(m, 1.0 / m, dtype=np.float64)
    p /= ps
    return p

def dual_state(p, h, q, G, L, n):
    gmix = G.T @ p
    y, ok = project_capped(h - gmix / L, n)
    if not ok:
        return None
    d = y - h
    u = q + G @ d
    if not np.all(np.isfinite(u)):
        return None
    D = float(p @ u + 0.5 * L * float(d @ d))
    gap = float(u.max() - p @ u)
    return d, u, D, gap

def solve_majorant(h, q, G, n, max_inner):
    dx = 2.0 / n
    L = 2.0 * dx
    m = 2 * n - 1
    Ldual = max(1e-12, 1.05 * float(np.sum(G * G)) / L)
    qmax = float(q.max())
    tol = max(1e-12, 1e-8 * max(1.0, abs(qmax)))
    mask = q >= qmax - tol
    p = np.zeros(m)
    if mask.sum() > 0:
        p[mask] = 1.0 / mask.sum()
    else:
        p[int(np.argmax(q))] = 1.0
    p = project_simplex(p)
    best_D = float('-inf')
    best_d = None
    x = p.copy()
    yv = x.copy()
    theta = 1.0
    stall = 0
    for _ in range(max_inner):
        st = dual_state(yv, h, q, G, L, n)
        if st is None:
            break
        d_y, u_y, D_y, gap_y = st
        if D_y > best_D:
            best_D = D_y
            best_d = d_y.copy()
        if gap_y <= max(1e-10, 1e-8 * max(1.0, abs(float(u_y.max())))):
            break
        x_trial = project_simplex(yv + u_y / Ldual)
        st2 = dual_state(x_trial, h, q, G, L, n)
        if st2 is None:
            break
        d_x, u_x, D_x, _ = st2
        if D_x < D_y - 1e-13:
            x_trial = project_simplex(x + u_y / Ldual)
            st2 = dual_state(x_trial, h, q, G, L, n)
            if st2 is None:
                break
            d_x, u_x, D_x, _ = st2
            if D_x < D_y - 1e-13:
                break
        if D_x > best_D:
            best_D = D_x
            best_d = d_x.copy()
        sn = float(np.linalg.norm(x_trial - x))
        if sn < 1e-10:
            stall += 1
            if stall >= 3:
                break
        else:
            stall = 0
        th_new = (1.0 + math.sqrt(1.0 + 4.0 * theta * theta)) / 2.0
        yv = project_simplex(x_trial + ((theta - 1.0) / th_new) * (x_trial - x))
        x = x_trial
        theta = th_new
    if best_d is None:
        st = dual_state(p, h, q, G, L, n)
        if st is None:
            return None
        best_d = st[0].copy()
    return best_d

def optimize_start(h0, n, max_outer, max_inner):
    h, ok = project_capped(h0.astype(np.float64).copy(), n)
    if not ok:
        return h0.copy()
    s = exact_score(h, n)
    best_h = h.copy()
    best_s = s
    stall = 0
    for _ in range(max_outer):
        q = lag_scores(h, n)
        G = build_G(h, n)
        if not np.all(np.isfinite(G)):
            break
        d = solve_majorant(h, q, G, n, max_inner)
        if d is None or not np.all(np.isfinite(d)):
            break
        if np.max(np.abs(d)) <= 1e-10:
            break
        v = G @ d
        norm2 = float(d @ d)
        qmax = float(q.max())
        L = 4.0 / n
        acc = False
        for i in range(19):
            alpha = 2.0 ** (-float(i))
            M = float(np.max(q + alpha * v)) + 0.5 * L * alpha * alpha * norm2
            if M >= qmax - 1e-14:
                continue
            ht = h + alpha * d
            ht, ok2 = project_capped(ht, n)
            if not ok2:
                continue
            st = exact_score(ht, n)
            if st < s - 1e-12:
                h = ht
                s = st
                if s < best_s:
                    best_s = s
                    best_h = h.copy()
                acc = True
                break
        if not acc:
            break
        if s > best_s - 1e-11:
            stall += 1
        else:
            stall = 0
        if stall >= 2:
            break
    return best_h

def coord_refine(h, n, n_passes):
    S = n / 2.0
    s = exact_score(h, n)
    deltas = [0.03, -0.03, 0.015, -0.015, 0.007, -0.007]
    for _ in range(n_passes):
        improved = False
        for i in range(n):
            for delta in deltas:
                h2 = h.copy()
                h2[i] += delta
                h2 = np.clip(h2, 0.0, 1.0)
                h2 *= (S / h2.sum()) if h2.sum() > 1e-15 else 1.0
                h2 = np.clip(h2, 0.0, 1.0)
                h2, ok = project_capped(h2, n)
                if not ok:
                    continue
                s2 = exact_score(h2, n)
                if s2 < s - 1e-11:
                    h = h2
                    s = s2
                    improved = True
                    break
        if not improved:
            break
    return h, s

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = globals().get('initial_h_values', None)
    n = 45
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.ndim == 1 and 2 <= len(p) <= 999 and np.all(np.isfinite(p)) and np.max(np.abs(p)) < 1e6:
            n = len(p)
            fb, _ = project_capped(p.copy(), n)
        else:
            fb = np.full(n, 0.5)
    else:
        nk = kwargs.get('n_points', None)
        if isinstance(nk, int) and not isinstance(nk, bool) and 2 <= nk <= 999:
            n = nk
        fb = np.full(n, 0.5)
    fb_s = exact_score(fb, n)
    best_h = fb.copy()
    best_s = fb_s
    if n <= 100:
        max_outer, max_inner = 80, 60
    elif n <= 400:
        max_outer, max_inner = 15, 20
    else:
        max_outer, max_inner = 5, 12
    starts = [fb.copy(), (1.0 - fb).copy(), fb[::-1].copy(), (1.0 - fb[::-1]).copy()]
    starts.append(project_capped(fb + 0.05 * rng.standard_normal(n), n)[0])
    starts.append(project_capped(fb + 0.15 * rng.standard_normal(n), n)[0])
    starts.append(project_capped(rng.uniform(0, 1, n), n)[0])
    seen = []
    deduped = []
    for st in starts:
        dup = any(np.max(np.abs(st - sv)) < 1e-8 for sv in seen)
        if not dup:
            deduped.append(st)
            seen.append(st)
    if n > 400:
        deduped = deduped[:2]
    elif n > 100:
        deduped = deduped[:4]
    for st in deduped:
        try:
            bh = optimize_start(st, n, max_outer, max_inner)
            bh, ok = project_capped(bh, n)
            if not ok:
                continue
            bs = exact_score(bh, n)
            if bs < best_s - 1e-12:
                best_h = bh.copy()
                best_s = bs
        except Exception:
            pass
    if n <= 100:
        try:
            pert = project_capped(best_h + 0.025 * rng.standard_normal(n), n)[0]
            bh = optimize_start(pert, n, max_outer // 2, max_inner)
            bh, ok = project_capped(bh, n)
            if ok:
                bs = exact_score(bh, n)
                if bs < best_s - 1e-12:
                    best_h = bh.copy()
                    best_s = bs
        except Exception:
            pass
    if n <= 200:
        try:
            best_h, best_s = coord_refine(best_h, n, 5)
        except Exception:
            pass
    final_h, ok = project_capped(best_h, n)
    if not ok:
        final_h = fb.copy()
    final_s = exact_score(final_h, n)
    if final_s > fb_s + 1e-12:
        final_h = fb.copy()
        final_s = fb_s
    final_h, _ = project_capped(final_h, n)
    final_s = exact_score(final_h, n)
    return final_h.astype(np.float64), float(final_s), int(n)
