import numpy as np
from scipy.optimize import minimize

def project_capped(z, n):
    z = np.clip(np.asarray(z, dtype=np.float64).ravel(), -1e12, 1e12)
    if z.shape != (n,):
        return None
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for i in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            h[i] = min(h[i] + r, 1.0)
        else:
            h[i] = max(h[i] + r, 0.0)
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    return h

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n))

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def smooth_val_grad(h, tau, n):
    dx = 2.0 / n
    Q = np.empty(2 * n - 1)
    for d in range(-(n - 1), n):
        a = max(d, 0); b = max(-d, 0); m = n - abs(d)
        Q[d + n - 1] = dx * np.dot(h[a:a+m], 1.0 - h[b:b+m])
    if not np.all(np.isfinite(Q)):
        return 1e10, np.zeros(n)
    qm = Q.max()
    e = np.exp(np.clip((Q - qm) / tau, -50, 50))
    se = e.sum()
    if se <= 0 or not np.isfinite(se):
        return 1e10, np.zeros(n)
    w = e / se
    F = qm + tau * np.log(se)
    g = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[d + n - 1]
        if wd < 1e-14:
            continue
        a = max(d, 0); b = max(-d, 0); m = n - abs(d)
        g[a:a+m] += wd * dx * (1.0 - h[b:b+m])
        g[b:b+m] -= wd * dx * h[a:a+m]
    if not np.all(np.isfinite(g)):
        return 1e10, np.zeros(n)
    return F, g

def mass_fun(h, n):
    return np.sum(h) - n / 2.0

def mass_jac(h, n):
    return np.ones(n)

def build_starts(fb, n, rng, cnt):
    pool = []
    cands = [fb.copy(), fb[::-1].copy(), 1.0 - fb[::-1], np.full(n, 0.5)]
    blk = np.zeros(n); blk[:n//2] = 1.0
    if n % 2:
        blk[n//2] = 0.5
    cands.append(blk)
    cands.append(np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64))
    cands.append(fb + rng.normal(0, 0.10, n))
    cands.append(fb + rng.normal(0, 0.25, n))
    cands.append(fb + rng.normal(0, 0.40, n))
    cands.append(rng.uniform(0, 1, n))
    for c in cands:
        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-8:
                dup = True; break
        if not dup:
            pool.append(p)
        if len(pool) >= cnt:
            break
    if not pool:
        pool.append(fb.copy())
    return pool

def slsqp_call(h0, tau, n, maxit):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': mass_fun, 'jac': mass_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_val_grad, h0.copy(), args=(tau, n),
                       method='SLSQP', jac=True, bounds=bounds,
                       constraints=cons,
                       options={'ftol': 1e-12, 'maxiter': maxit, 'disp': False})
    except Exception:
        return h0
    if res.x is None or res.x.shape != (n,) or not np.all(np.isfinite(res.x)):
        return h0
    return np.asarray(res.x, dtype=np.float64)

def optimize_one(h0, n, tau_sched, maxit_per_stage):
    h = h0.copy()
    for tau in tau_sched:
        hr = slsqp_call(h, tau, n, maxit_per_stage)
        hr = project_capped(hr, n)
        if hr is None or not is_feasible(hr, n):
            continue
        cur_s = exact_score(h, n)
        best_z, best_s = h.copy(), cur_s
        for alpha in (1.0, 0.5, 0.25, 0.125):
            z = project_capped(h + alpha * (hr - h), n)
            if z is None or not is_feasible(z, n):
                continue
            s = exact_score(z, n)
            if s < best_s - 1e-14:
                best_s, best_z = s, z.copy()
        if best_s < cur_s - 1e-14:
            h = best_z
    return h, exact_score(h, n)

def coord_polish(h, n, n_passes, eps):
    h = h.copy()
    s = exact_score(h, n)
    for _ in range(n_passes):
        improved = False
        idx = np.argsort(h)  # try low values first
        for i in idx:
            for delta in (eps, -eps):
                z = h.copy()
                z[i] += delta
                z2 = project_capped(z, n)
                if z2 is None or not is_feasible(z2, n):
                    continue
                s2 = exact_score(z2, n)
                if s2 < s - 1e-14:
                    h = z2; s = s2; improved = True
                    break
        if not improved:
            break
    return h, s
