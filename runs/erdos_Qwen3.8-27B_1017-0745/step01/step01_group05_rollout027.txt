        if a > 0 and np.isfinite(lam + r / a):
            lam_n = lam + r / a
        else:
            lam_n = 0.5 * (lo + hi)
        if lo < lam_n < hi:
            lam = lam_n
        else:
            lam = 0.5 * (lo + hi)
    c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(5):
        r = S - c.sum()
        if abs(r) < 1e-14:
            break
        mask = (c < 1.0 - 1e-12) if r > 0 else (c > 1e-12)
        cnt = mask.sum()
        if cnt == 0:
            break
        c[mask] += r / cnt
        np.clip(c, 0.0, 1.0, out=c)
    ok = (c.shape == (n,) and np.all(np.isfinite(c)) and c.min() >= -1e-12
          and c.max() <= 1.0 + 1e-12 and abs(c.sum() - S) < 1e-10)
    if not ok:
        return np.full(n, 0.5), False
    return c, True

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and h.min() >= -1e-12
            and h.max() <= 1.0 + 1e-12 and abs(h.sum() - n / 2.0) < 1e-10)

def lag_scores(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    qs = lag_scores(h, n)
    if qs.size != 2 * n - 1 or not np.all(np.isfinite(qs)):
        return float('inf')
    return float(qs.max())

def build_G(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        row = k + n - 1
        lo = max(0, k)
        hi = min(n, n + k)
        if hi > lo:
            G[row, lo:hi] += dx * (1.0 - h[lo - k:hi - k])
            G[row, lo - k:hi - k] -= dx * h[lo:hi]
    if not np.all(np.isfinite(G)):
        return None
    return G

def project_simplex(z):
    m = len(z)
    if m <= 1:
        return np.ones(m) / max(m, 1)
    sz = np.sort(z)[::-1]
    css = np.cumsum(sz)
    rho = np.where(sz - (css - 1.0) / np.arange(1, m + 1) > 0)[0]
    if rho.size == 0:
        return np.full(m, 1.0 / m)
    r = int(rho[-1])
    theta = (css[r] - 1.0) / (r + 1)
    p = np.maximum(z - theta, 0.0)
    diff = 1.0 - p.sum()
    if diff != 0.0 and p.max() > 0:
        p[np.argmax(p)] += diff
    return p

def solve_majorant(h, q, G, n, max_inner):
    dx = 2.0 / n
    L = 2.0 * dx
    m = q.size
    qmax = q.max()
    tol = max(1e-12, 1e-8 * max(1.0, abs(qmax)))
    active = np.where(q >= qmax - tol)[0]
    if active.size == 0:
        active = np.array([int(np.argmax(q))])
    p = np.zeros(m)
    p[active] = 1.0 / active.size
    mu = max(1e-12, 1.05 * float(np.sum(G * G)) / L)
    best_D = -1e30
    best_d = None
    stagnant = 0
    for it in range(max_inner):
        gmix = G.T @ p
        y, ok = project_capped(h - gmix / L, n)
        if not ok:
            break
        d = y - h
        u = q + G @ d
        D = float(p @ u + 0.5 * L * (d @ d))
        gap = float(u.max() - p @ u)
        if D > best_D:
            best_D = D
            best_d = d.copy()
        if gap <= max(1e-10, 1e-8 * max(1.0, abs(float(u.max())))):
            break
        p_new = project_simplex(p + u / mu)
        step = np.max(np.abs(p_new - p))
        if step < 1e-12:
            stagnant += 1
            if stagnant >= 4:
                break
        else:
            stagnant = 0
        p = p_new
    return best_d

def optimize_start(h0, n, max_outer, max_inner):
    h, ok = project_capped(h0, n)
    if not ok:
        return h0.copy(), float('inf')
    s0 = exact_score(h, n)
    best_h = h.copy()
    best_s = s0
    dx = 2.0 / n
    L = 2.0 * dx
    for outer in range(max_outer):
        q = lag_scores(h, n)
        G = build_G(h, n)
        if G is None:
            break
        d = solve_majorant(h, q, G, n, max_inner)
        if d is None or not np.all(np.isfinite(d)):
            break
        if np.max(np.abs(d)) < 1e-10:
            break
        v = G @ d
        norm2 = float(d @ d)
        qmax_cur = float(q.max())
        accepted = False
        for j in range(20):
            alpha = 1.0 / (2.0 ** j)
            M_alpha = float(np.max(q + alpha * v)) + 0.5 * L * alpha * alpha * norm2
            if M_alpha >= qmax_cur - 1e-14:
                continue
            h_trial = h + alpha * d
            h_trial, ok2 = project_capped(h_trial, n)
            if not ok2:
                continue
            s_t = exact_score(h_trial, n)
            if s_t < best_s - 1e-12:
                best_s = s_t
                best_h = h_trial.copy()
                h = h_trial.copy()
                accepted = True
                break
        if not accepted:
            break
    return best_h, best_s

def coord_refine(h, n, n_iter=80):
    dx = 2.0 / n
    best_h = h.copy()
    best_s = exact_score(h, n)
    rng = np.random.default_rng(12345)
    for _ in range(n_iter):
        i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
        if i == j:
            continue
        for delta in [0.02, 0.05, 0.1]:
            for sign in [1, -1]:
                h_t = best_h.copy()
                h_t[i] += sign * delta
                h_t[j] -= sign * delta
                h_t, ok = project_capped(h_t, n)
                if not ok:
                    continue
                s_t = exact_score(h_t, n)
                if s_t < best_s - 1e-12:
                    best_s = s_t
                    best_h = h_t.copy()
                    break
    return best_h, best_s

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = globals().get('initial_h_values', None)
    n = 45
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.ndim == 1 and 2 <= len(p) <= 999 and np.all(np.isfinite(p)) and np.max(np.abs(p)) < 1e6:
            n = len(p)
            fallback, ok = project_capped(p, n)
        else:
            fallback, _ = project_capped(np.full(n, 0.5), n)
    else:
        np_kw = kwargs.get('n_points', None)
        if isinstance(np_kw, int) and not isinstance(np_kw, bool) and 2 <= np_kw <= 999:
            n = np_kw
        fallback, _ = project_capped(np.full(n, 0.5), n)
    fb_s = exact_score(fallback, n)
    best_h = fallback.copy()
    best_s = fb_s
    if n <= 100:
        max_outer, max_inner = 80, 80
    elif n <= 400:
        max_outer, max_inner = 12, 18
    else:
        max_outer, max_inner = 5, 10
    starts = [fallback.copy(), (1.0 - fallback).copy(), fallback[::-1].copy(),
              (1.0 - fallback[::-1]).copy()]
    for sig in [0.05, 0.12, 0.20]:
        starts.append(fallback + sig * rng.standard_normal(n))
    starts.append(rng.uniform(0, 1, n))
    dedup = []
    for st in starts:
        if all(np.max(np.abs(st - d)) > 1e-8 for d in dedup):
            dedup.append(st)
    starts = dedup
    max_starts = 7 if n <= 100 else (4 if n <= 400 else 2)
    starts = starts[:max_starts]
    for st in starts:
        try:
            bh, bs = optimize_start(st, n, max_outer, max_inner)
            if check_feasible(bh, n) and bs < best_s - 1e-12:
                best_s = bs
                best_h = bh.copy()
        except Exception:
            pass
    if n <= 100:
        try:
            pert = best_h + 0.025 * rng.standard_normal(n)
            bh, bs = optimize_start(pert, n, max_outer // 2, max_inner)
            if check_feasible(bh, n) and bs < best_s - 1e-12:
                best_s = bs
                best_h = bh.copy()
        except Exception:
            pass
        try:
            bh, bs = coord_refine(best_h, n, 120)
            if check_feasible(bh, n) and bs < best_s - 1e-12:
                best_s = bs
                best_h = bh.copy()
        except Exception:
            pass
    final_h, ok = project_capped(best_h, n)
    if not ok or not check_feasible(final_h, n):
        final_h = fallback.copy()
    final_s = exact_score(final_h, n)
    if final_s > fb_s + 1e-12:
        final_h = fallback.copy()
        final_s = fb_s
    final_h, _ = project_capped(final_h, n)
    final_s = exact_score(final_h, n)
    return final_h.astype(np.float64), float(final_s), int(n)
