        if rep is not None and check_feasible(rep, n) and np.max(np.abs(rep - cur)) <= 1e-8:
            cur = rep
            cur_sc = exact_score(cur, n)
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc
    return best, best_sc

def _get_parent():
    n = 45
    if 'initial_h_values' in globals():
        ph = np.asarray(globals()['initial_h_values'], dtype=np.float64)
        if ph.ndim == 1 and 2 <= len(ph) <= 999 and np.all(np.isfinite(ph)):
            if np.all(ph >= -1e-8) and np.all(ph <= 1.0 + 1e-8):
                n = len(ph)
                p = project_box_sum(np.clip(ph, 0.0, 1.0), 0.5 * n)
                if p is not None:
                    return p, n
    return project_box_sum(np.full(n, 0.5), 0.5 * n), n

def _build_starts(parent, n, rng):
    starts = [parent.copy(), (1.0 - parent).copy(), parent[::-1].copy()]
    for sd in [0.04, 0.10, 0.20, 0.30]:
        p = project_box_sum(parent + sd * rng.standard_normal(n), 0.5 * n)
        if p is not None:
            starts.append(p)
    p = project_box_sum(rng.random(n), 0.5 * n)
    if p is not None:
        starts.append(p)
    i = np.arange(n)
    phi = rng.uniform(0, 2 * math.pi)
    theta = rng.uniform(0, 2 * math.pi)
    u = 0.5 + 0.30 * np.cos(2 * math.pi * i / n + phi) + 0.12 * np.sin(4 * math.pi * i / n + theta)
    p = project_box_sum(u, 0.5 * n)
    if p is not None:
        starts.append(p)
    u2 = 0.5 + 0.35 * np.cos(2 * math.pi * i / n + phi * 1.7)
    p2 = project_box_sum(u2, 0.5 * n)
    if p2 is not None:
        starts.append(p2)
    uniq = []
    for s in starts:
        dup = False
        for u in uniq:
            if np.max(np.abs(s - u)) < 1e-10:
                dup = True
                break
        if not dup:
            uniq.append(s)
    return uniq

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent, n = _get_parent()
    best_h = parent.copy()
    best_c5 = exact_score(parent, n)

    if n <= 64:
        n_cycles, n_levels, n_props, sigma0 = 5, 6, 250, 0.08
        restart_sds = [0.08, 0.18, 0.30]
    elif n <= 256:
        n_cycles, n_levels, n_props, sigma0 = 3, 5, 90, 0.06
        restart_sds = [0.15, 0.25]
    else:
        n_cycles, n_levels, n_props, sigma0 = 2, 4, 24, 0.04
        restart_sds = [0.12]

    starts = _build_starts(parent, n, rng)
    scored = [(exact_score(s, n), s) for s in starts]
    scored = [(sc, s) for sc, s in scored if np.isfinite(sc)]
    scored.sort(key=lambda x: x[0])
    for sc, s in scored:
        if sc < best_c5:
            best_c5 = sc
            best_h = s.copy()

    for cyc in range(n_cycles):
        if cyc == 0:
            st = scored[0][1].copy() if scored else parent.copy()
        elif cyc == 1 and len(scored) > 1:
            st = scored[1][1].copy()
        else:
            ri = (cyc - 2) % len(restart_sds) if cyc >= 2 else 0
            sd = restart_sds[ri]
            st = None
            for _ in range(4):
                st = project_box_sum(best_h + sd * rng.standard_normal(n), 0.5 * n)
                if st is not None and check_feasible(st, n):
                    break
            if st is None:
                st = best_h.copy()
        st_sc = exact_score(st, n)
        if np.isfinite(st_sc) and st_sc < best_c5:
            best_c5 = st_sc
            best_h = st.copy()
        try:
            lb, lsc = anneal_cycle(st, n, rng, n_levels, n_props, sigma0)
            if np.isfinite(lsc) and lsc < best_c5:
                best_c5 = lsc
                best_h = lb.copy()
        except Exception:
            pass

    fh = project_box_sum(best_h, 0.5 * n)
    if fh is not None and check_feasible(fh, n):
        fc = exact_score(fh, n)
        if np.isfinite(fc):
            return (fh, fc, n)
    fb = project_box_sum(parent, 0.5 * n)
    if fb is not None and check_feasible(fb, n):
        rc = exact_score(fb, n)
        if np.isfinite(rc):
            return (fb, rc, n)
    fb = np.full(n, 0.5)
    return (fb, exact_score(fb, n), n)
