def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):

        v = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

    L = 2 * n - 1

        j0 = max(0, d)
        m = n - abs(d)
        c[d + n - 1] = float(np.dot(h[i0:i0 + m], 1.0 - h[j0:j0 + m]))
    return c

    v = 0.0
    j = idx + d
    if 0 <= j < n:
        v += 1.0 - h[j]
    j2 = idx - d
    if 0 <= j2 < n:
        v -= h[j2]
    return v

def pair_ab(h, p, q, n, c):
    L = 2 * n - 1
    a = np.zeros(L)
    b = np.zeros(L)
    Dlt = q - p
    a[n - 1] = -2.0
    a[n - 1 + Dlt] = 1.0
    a[n - 1 - Dlt] = 1.0

        b[d + n - 1] = D_val(d, p, h, n) - D_val(d, q, h, n)
    if not (np.all(np.isfinite(a)) and np.all(np.isfinite(b))):

            disc = 0.0
        else:
            return roots
