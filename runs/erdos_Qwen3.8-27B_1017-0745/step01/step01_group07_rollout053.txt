import numpy as np
import math

def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):
        return None
    n = len(z)
    if np.max(np.abs(z)) > 1e12:
        return None
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - np.sum(x)
    if r > 0:
        for i in range(n):
            if r <= 0: break
            a = min(r, 1.0 - x[i]); x[i] += a; r -= a
    elif r < 0:
        for i in range(n):
            if r >= 0: break
            a = min(-r, x[i]); x[i] -= a; r += a
    if abs(np.sum(x) - S) > 1e-10 or not np.all(np.isfinite(x)):
        return None
    return np.clip(x, 0.0, 1.0)

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-15) and np.all(h <= 1.0 + 1e-15)
            and abs(np.sum(h) - n / 2.0) <= 1e-10)

def compute_q(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def exact_score(h, n):
    try:
        q = compute_q(h, n)
        if not np.all(np.isfinite(q)): return None
        return float(np.max(q))
    except Exception:
        return None

def pair_line_search(h, q, score, p, r, n, dx):
    L = max(-h[p], h[r] - 1.0)
    U = min(1.0 - h[p], h[r])
    if U - L <= 1e-12: return None
    off = n - 1
    a = np.zeros(2 * n - 1)
    b = np.zeros(2 * n - 1)
    a[off] = -2.0 * dx
    dr = r - p
    if 0 < dr < n: a[off + dr] += dx
    if -n < dr < 0: a[off + dr] += dx
    for d in range(-(n - 1), n):
        b_d = 0.0
        if 0 <= p + d < n: b_d += (1.0 - h[p + d])
        if 0 <= r + d < n: b_d -= (1.0 - h[r + d])
        if 0 <= p - d < n: b_d -= h[p - d]
        if 0 <= r - d < n: b_d += h[r - d]
        b[d + off] = dx * b_d
    c = q
    best_t, best_f = 0.0, score
    t_arr = np.concatenate([np.linspace(L, U, 17), [0.0]])
    for t in t_arr:
        f = float(np.max(c + b * t + a * t * t))
        if f < best_f - 1e-14: best_f, best_t = f, t
        elif f < best_f + 1e-14 and abs(t) < abs(best_t): best_t = t
    radius = (U - L) / 16.0
    for _ in range(4):
        t_lo, t_hi = max(L, best_t - radius), min(U, best_t + radius)
        for t in np.linspace(t_lo, t_hi, 5):
            f = float(np.max(c + b * t + a * t * t))
            if f < best_f - 1e-14: best_f, best_t = f, t
        radius /= 4.0
    if best_f >= score - 1e-12: return None
    h_new = h.copy()
    h_new[p] += best_t; h_new[r] -= best_t
    if not (np.all(h_new >= 0) and np.all(h_new <= 1)):
        h_new = project_capped_simplex(h_new, n / 2.0)
        if h_new is None: return None
    try:
        nq = compute_q(h_new, n)
        ns = float(np.max(nq))
    except Exception:
        return None
    if ns < score - 1e-12:
        return (h_new, ns, nq)
    return None

def build_starts(n, S, parent, rng):
    starts = [np.full(n, 0.5)]
    if parent is not None:
        starts.append(parent.copy()); starts.append(parent[::-1].copy())
    M = min(4, (n - 1) // 2)
    idx = np.arange(n, dtype=np.float64)
    u = np.full(n, 0.5)
    for m in range(1, M + 1):
        u += rng.normal(0, 0.12 / math.sqrt(m)) * np.cos(2 * np.pi * m * idx / n + rng.uniform(0, 2 * np.pi))
    pu = project_capped_simplex(u, S)
    if pu is not None: starts.append(pu)
    perm = rng.permutation(n)
    binv = np.zeros(n); binv[perm[:n // 2]] = 1.0
    if n % 2 == 1: binv[perm[n // 2]] = 0.5
    starts.append(binv)
    pr = project_capped_simplex(rng.random(n), S)
    if pr is not None: starts.append(pr)
    deduped = []
    for s in starts:
        if not is_feasible(s, n): continue
        if not any(np.max(np.abs(s - d)) < 1e-10 for d in deduped):
            deduped.append(s)
    cap = 6 if n <= 128 else (4 if n <= 256 else (3 if n <= 512 else 2))
    return deduped[:cap]

def polish_sweep(h, q, score, n, dx, rng, max_sweeps):
    for _ in range(max_sweeps):
        accepted = 0
        P = min(n * (n - 1) // 2, max(24, 80000 // n))
        pairs, pset = [], set()
        n_adj = min(n - 1, max(1, P // 2))
        for i in np.linspace(0, n - 2, n_adj, dtype=int):
            pairs.append((i, i + 1)); pset.add((i, i + 1))
        attempts = 0
        while len(pairs) < P and attempts < P * 3:
            i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
            attempts += 1
            if i == j: continue
            if i > j: i, j = j, i
            if (i, j) not in pset: pset.add((i, j)); pairs.append((i, j))
        rng.shuffle(pairs)
        for (p, r) in pairs:
            res = pair_line_search(h, q, score, p, r, n, dx)
            if res is not None:
                h, score, q = res; accepted += 1
        if accepted == 0: break
    return h, score, q

def anneal_trajectory(h0, score0, n, dx, rng):
    h = h0.copy(); score = score0; q = compute_q(h, n)
    E = 5 if n <= 128 else (4 if n <= 256 else (3 if n <= 512 else 2))
    scale = 0.12; S = n / 2.0
    for e in range(E):
        T = max(5e-5, 3e-3 * (0.4 ** e))
        K = min(1000, max(80, 80000 // n))
        acc = 0
        for _ in range(K):
            p, r = int(rng.integers(0, n)), int(rng.integers(0, n))
            if p == r: continue
            Lo = max(-h[p], h[r] - 1.0); Hi = min(1.0 - h[p], h[r])
            if Hi - Lo <= 0: continue
            t = rng.uniform(-scale, scale)
            if t < Lo or t > Hi: continue
            h_new = h.copy(); h_new[p] += t; h_new[r] -= t
            if not (np.all(h_new >= 0) and np.all(h_new <= 1)):
                h_new = project_capped_simplex(h_new, S)
                if h_new is None: continue
            try:
                nq = compute_q(h_new, n); ns = float(np.max(nq))
            except Exception:
                continue
            delta = ns - score
            if delta <= 0:
                h, score, q = h_new, ns, nq; acc += 1
            elif -delta / T > -700 and rng.random() < math.exp(-delta / T):
                h, score, q = h_new, ns, nq; acc += 1
        ratio = acc / max(K, 1)
        if ratio < 0.12: scale = min(0.4, 1.25 * scale)
        elif ratio > 0.55: scale = max(0.01, 0.75 * scale)
        h, score, q = polish_sweep(h, q, score, n, dx, rng, 1)
    return h, score, q

def run(seed=42, **kwargs):
    if seed is None: seed = 42
    try: seed = int(seed)
    except (TypeError, ValueError): seed = 42
    rng = np.random.default_rng(seed % 2**32)
    n, parent = None, None
    iv = globals().get('initial_h_values', None)
    if iv is not None:

        except Exception:

    if n is None:

        if n < 8 or n > 999: n = 80
    S, dx = n / 2.0, 2.0 / n
    h_const = np.full(n, 0.5)
    best_h, best_score = h_const.copy(), exact_score(h_const, n)
    if best_score is None: best_score = 0.5
    if parent is not None:
        ps = exact_score(parent, n)
        if ps is not None and ps < best_score:
            best_score, best_h = ps, parent.copy()
    starts = build_starts(n, S, parent, rng)
    for s in starts:
        sc = exact_score(s, n)
        if sc is not None and sc < best_score:
            best_score, best_h = sc, s.copy()
    psweeps = 4 if n <= 128 else (3 if n <= 256 else (2 if n <= 512 else 1))
    for s in starts:
        try:
            sc = exact_score(s, n)
            if sc is None: continue
            h_a, sc_a, q_a = anneal_trajectory(s, sc, n, dx, rng)
            if sc_a < best_score: best_score, best_h = sc_a, h_a.copy()
            h_p, sc_p, _ = polish_sweep(h_a, q_a, sc_a, n, dx, rng, psweeps)
            if sc_p < best_score: best_score, best_h = sc_p, h_p.copy()
        except Exception:
            continue
    try:
        q_b = compute_q(best_h, n)
        h_f, sc_f, _ = polish_sweep(best_h, q_b, best_score, n, dx, rng, psweeps)
        if sc_f < best_score: best_score, best_h = sc_f, h_f
    except Exception:
        pass
    if not is_feasible(best_h, n):
        rep = project_capped_simplex(best_h, S)
        if rep is not None and is_feasible(rep, n): best_h = rep
        else: best_h = h_const.copy()
    fs = exact_score(best_h, n)
    if fs is None: fs = 0.5
    return best_h, fs, n
