        G[row, lo:hi] += dx * (1.0 - h[lo - k:hi - k])
        G[row, lo - k:hi - k] -= dx * h[lo:hi]
    return G

def project_simplex(z):
    n = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(n, 1.0 / n)
    s = np.sort(z)[::-1]
    css = np.cumsum(s)
    rho = np.where(s - (css - 1.0) / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        return np.full(n, 1.0 / n)

    theta = (css[r] - 1.0) / (r + 1)
    p = np.maximum(z - theta, 0.0)
    r2 = 1.0 - p.sum()
    if abs(r2) > 1e-15 and p.sum() > 0:
        p[np.argmax(p)] += r2
    return p

def project_capped(y, n):

        else:

    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        lo = max(0, k)
        hi = min(n, n + k)
        row = k + n - 1
        G[row, lo:hi] += dx * (1.0 - h[lo - k:hi - k])
        G[row, lo - k:hi - k] -= dx * h[lo:hi]
    return G

def project_simplex(z):
    n = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(n, 1.0 / n)
    s = np.sort(z)[::-1]
    css = np.cumsum(s)
    idx = np.arange(1, n + 1, dtype=np.float64)
    res = np.where(s - (css - 1.0) / idx > 0)[0]
    if len(res) == 0:
        return np.full(n, 1.0 / n)
    r = res[-1]
    theta = (css[r] - 1.0) / (r + 1)
    p = np.maximum(z - theta, 0.0)
    r2 = 1.0 - p.sum()
    if abs(r2) > 1e-14 and p.max() > 0:

    return p
