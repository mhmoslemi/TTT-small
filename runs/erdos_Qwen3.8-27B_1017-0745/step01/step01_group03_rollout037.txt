def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0 or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if resid > 1e-14:
        for i in range(n):
            add = min(resid, 1.0 - y[i])
            y[i] += add
            resid -= add
            if resid < 1e-14:
                break
    elif resid < -1e-14:
        for i in range(n):
            sub = min(-resid, y[i])
            y[i] -= sub
            resid += sub
            if -resid < 1e-14:
                break
    y = np.clip(y, 0.0, 1.0)

    return y

def build_G(x, n):
    L = 2 * n - 1
    G = np.zeros((L, n))
    for k in range(-(n - 1), n):
        idx = k + n - 1
        i_lo, i_hi = max(0, k), min(n - 1, n - 1 + k)
        if i_lo > i_hi:
            continue
        i = np.arange(i_lo, i_hi + 1)
        j = i - k
        G[idx, i] += (1.0 - x[j])
        G[idx, j] += (-x[i])
    return G

def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * (np.arange(1, n + 1, dtype=np.float64)) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n, dtype=np.float64) / n
    r = int(rho[-1])
    theta = (cssv[r] - 1.0) / (r + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s <= 0 or not np.all(np.isfinite(w)):
        return None
    w /= s
    return w

import numpy as np

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0 or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if resid > 1e-14:
        for i in range(n):
            add = min(resid, 1.0 - y[i])
            y[i] += add
            resid -= add
            if resid < 1e-14:
                break
    elif resid < -1e-14:
        for i in range(n):
            sub = min(-resid, y[i])
            y[i] -= sub
            resid += sub
            if -resid < 1e-14:
                break
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(np.sum(y) - S) > 5e-10 * max(1.0, float(n)):
        return None
    return y

def exact_score(h, n):
    if h is None or h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    if c.shape != (2 * n - 1,) or not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c) * 2.0 / n)

def build_G(x, n):
    L = 2 * n - 1
    G = np.zeros((L, n))
    for k in range(-(n - 1), n):
        idx = k + n - 1
        i_lo = max(0, k)
        i_hi = min(n - 1, n - 1 + k)
        if i_lo > i_hi:
            continue
        i = np.arange(i_lo, i_hi + 1)
        j = i - k
        G[idx, i] += (1.0 - x[j])
        G[idx, j] += (-x[i])
    return G

def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    cnt = len(v)
    if not np.all(np.isfinite(v)):
        return None
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idxs = np.arange(1, cnt + 1, dtype=np.float64)
    rho = np.where(u * idxs > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(cnt, dtype=np.float64) / cnt
    r = int(rho[-1])
    theta = (cssv[r] - 1.0) / (r + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s <= 0 or not np.all(np.isfinite(w)):
        return None
    w /= s
    return w

def model_1d(alpha, dd, Q, Gd):
    return alpha * alpha * dd + float(np.max(Q + alpha * Gd))

        w = project_simplex(warm)

    Imax = max(200, min(500, int(np.ceil(20000.0 / n))))
