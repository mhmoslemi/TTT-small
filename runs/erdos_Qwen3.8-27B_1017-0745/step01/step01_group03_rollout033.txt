import numpy as np
from scipy.signal import fftconvolve

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if resid > 1e-12:
        order = np.argsort(1.0 - y)
        for i in order:
            add = min(resid, 1.0 - y[i])
            y[i] += add
            resid -= add
            if resid < 1e-14:
                break
    elif resid < -1e-12:
        order = np.argsort(y)
        for i in order:
            sub = min(-resid, y[i])
            y[i] -= sub
            resid += sub
            if -resid < 1e-14:
                break
    y = np.clip(y, 0.0, 1.0)
    if abs(np.sum(y) - S) > 1e-10 * max(1.0, float(n)):
        # Final repair
        r = S - np.sum(y)
        if r > 0:
            idx = np.argmax(1.0 - y)
            y[idx] += min(r, 1.0 - y[idx])
        elif r < 0:
            idx = np.argmax(y)
            y[idx] += max(r, -y[idx])
    if not np.all(np.isfinite(y)):
        return None
    return y

def compute_Q(h, n):
    Q = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        idx = k + n - 1
        if k >= 0:
            Q[idx] = np.dot(h[k:], 1.0 - h[:n - k])
        else:
            Q[idx] = np.dot(h[:n + k], 1.0 - h[-k:])
    return Q

def compute_grad(h, w, n):
    g = np.zeros(n)
    for k in range(-(n - 1), n):
        wk = w[k + n - 1]
        if abs(wk) < 1e-16:
            continue
        if k >= 0:

            g[:n - k] -= wk * h[k:]
        else:
            nk = -k
            g[:n - nk] += wk * (1.0 - h[nk:])
            g[nk:] -= wk * h[:n - nk]
    return g
