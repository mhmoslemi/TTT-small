def check_feasible(h, n):

            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)

def project_capped(y, n):
    S = n / 2.0
    if y.shape != (n,) or not np.all(np.isfinite(y)) or np.max(np.abs(y)) > 1e12:
        return np.full(n, 0.5), False
    lo, hi = float(y.min()) - 1.0, float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
    c = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0).copy()
    for _ in range(5):
        r = S - c.sum()
        if abs(r) < 1e-14: break
        cap = (1.0 - c) if r > 0 else c.copy()
        if cap.sum() < 1e-15: break
        c += r * cap / cap.sum()
        np.clip(c, 0.0, 1.0, out=c)
    return (c, True) if check_feasible(c, n) else (np.full(n, 0.5), False)

def exact_score(h, n):
    if not check_feasible(h, n): return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / n) if np.all(np.isfinite(corr)) else float('inf')

def make_helmert(n):
    d = n - 1
    B = np.zeros((n, d))
    for j in range(d):
        s = math.sqrt((j + 1.0) * (j + 2.0))
        B[:j+1, j] = 1.0 / s
        B[j+1, j] = -(j + 1.0) / s
    return B

def lag_scores(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_val(h, tau):
    qs = lag_scores(h, len(h))
    m = qs.max()
    return float(m + tau * np.log(np.exp((qs - m) / tau).sum()))

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    qs = lag_scores(h, n)
    m = qs.max()
    w = np.exp((qs - m) / tau)
    p = w / w.sum()
    g = np.zeros(n)
    for l in range(2 * n - 1):
        if p[l] < 1e-15: continue
        k = l - (n - 1)
        lo1, hi1 = max(0, k), min(n, n + k)
        if hi1 > lo1: g[lo1:hi1] += p[l] * dx * (1.0 - h[lo1-k:hi1-k])
        lo2, hi2 = max(0, -k), min(n, n - k)
        if hi2 > lo2: g[lo2:hi2] -= p[l] * dx * h[lo2+k:hi2+k]
    return g

    best_h, best_s = h0.copy(), exact_score(h0, n)

            g = smooth_grad(h, tau)
            if not np.all(np.isfinite(g)): break

                if not ok:

                    alpha = min(1.2 * alpha, 5.0)
