    idx = np.arange(n, dtype=np.float64) + 0.5
    for j, f in enumerate(freqs):
        cols[:, j] = np.cos(np.pi * f * idx / n)
    cols -= cols.mean(axis=0)
    Q, R = np.linalg.qr(cols, mode='reduced')
    if Q.shape[1] < d or not np.all(np.isfinite(Q)):
        return None
    return Q[:, :d].copy()

def decode_candidate(anchor, B, sigma, u, n):
    scale = sigma * math.sqrt(3.0 * n / B.shape[1])
    raw = anchor + scale * (B @ u)
    if not np.all(np.isfinite(raw)):
        return None
    h, ok = project_capped(raw, n)
    return h if ok else None

def reflect1(x):
    t = (x + 1.0) % 4.0
    return -1.0 + t if t <= 2.0 else 3.0 - t

def evolve_stage(anchor, B, sigma, P, G, rng, n):
    d = B.shape[1]
    pop = np.zeros((P, d))
    pop[1:] = rng.uniform(-1.0, 1.0, (P - 1, d))
    scores = np.full(P, float('inf'))
    best_h = anchor.copy()
    best_s = exact_score(anchor)
    best_u = np.zeros(d)
    for i in range(P):
        h = decode_candidate(anchor, B, sigma, pop[i], n)
        if h is not None:
            s = exact_score(h)
            if s < scores[i]:
                scores[i] = s
            if s < best_s:
                best_s = s
                best_h = h.copy()
                best_u = pop[i].copy()
    stag = 0
    restarted = False
    for gen in range(G):
        if stag >= 4 and not restarted and gen < G - 1:
            for i in range(1, P):
                pop[i] = np.clip(best_u + 0.35 * rng.standard_normal(d), -1.0, 1.0)
            for i in range(P):
                h = decode_candidate(anchor, B, sigma, pop[i], n)
                if h is not None:
                    s = exact_score(h)
                    if s < scores[i]:
                        scores[i] = s
                    if s < best_s:
                        best_s = s
                        best_h = h.copy()
                        best_u = pop[i].copy()
            stag = 0
            restarted = True
            continue
        snap = pop.copy()
        snap_s = scores.copy()
        stag += 1
        for i in range(P):
            others = list(range(P))
            others.remove(i)
            r1, r2, r3 = rng.choice(others, 3, replace=False)
            F = rng.uniform(0.5, 0.9)
            v = snap[r1] + F * (snap[r2] - snap[r3])
            cmask = rng.random(d) < 0.8
            cmask[int(rng.integers(0, d))] = True
            t = np.where(cmask, v, snap[i]).copy()
            for j in range(d):
                t[j] = reflect1(t[j])
            h = decode_candidate(anchor, B, sigma, t, n)
            if h is not None:
                s = exact_score(h)
                if s < snap_s[i] - 1e-12:
                    pop[i] = t.copy()
                    scores[i] = s
                    if s < best_s - 1e-12:
                        best_s = s
                        best_h = h.copy()
                        best_u = t.copy()
    return best_h, best_s

def local_refine(h0, n, rng, n_iter):
    best_h = h0.copy()
    best_s = exact_score(h0)
    sig0 = 0.06
    for it in range(n_iter):
        sig = sig0 * (0.999 ** it)
        if sig < 0.0008:
            sig = 0.0008
        pert = sig * rng.standard_normal(n)
        h, ok = project_capped(best_h + pert, n)
        if not ok:
            continue
        s = exact_score(h)
        if s < best_s - 1e-13:
            best_s = s
            best_h = h.copy()
    return best_h, best_s

def pair_refine(h0, n, rng, n_iter):
    best_h = h0.copy()
    best_s = exact_score(h0)
    idx = np.arange(n)
    for it in range(n_iter):
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        eps = rng.uniform(0.002, 0.02)
        h = best_h.copy()
        a = min(eps, best_h[i], 1.0 - best_h[j])
        b = min(eps, best_h[j], 1.0 - best_h[i])
        for delta in (a, -a, b, -b):
            if abs(delta) < 1e-10:
                continue
            h2 = best_h.copy()
            h2[i] += delta
            h2[j] -= delta
            h2 = project_capped(h2, n)[0]
            s = exact_score(h2)
            if s < best_s - 1e-14:
                best_s = s
                best_h = h2.copy()
                break
    return best_h, best_s

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = globals().get('initial_h_values', None)
    n = 45
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.ndim == 1 and 2 <= len(p) <= 999 and np.all(np.isfinite(p)) and np.max(np.abs(p)) < 1e6:
            n = len(p)
            base, ok = project_capped(p.copy(), n)
            if not ok:
                base = np.full(n, 0.5)
        else:
            base = np.full(n, 0.5)
    else:
        np_kw = kwargs.get('n_points', None)
        if isinstance(np_kw, int) and not isinstance(np_kw, bool) and 2 <= np_kw <= 999:
            n = np_kw
        base = np.full(n, 0.5)

    best_h = base.copy()
    best_s = exact_score(base)

    B = make_basis(n)
    if B is not None:
        d = B.shape[1]
        if n <= 100:
            P = max(12, min(48, 2 * d + 8))
            G = 22
        elif n <= 400:
            P = max(10, min(32, d + 8))
            G = 8
        else:
            P = max(8, min(16, d))
            G = 4

        z = rng.standard_normal(n)
        z -= z.mean()
        nz = float(np.sqrt(np.mean(z * z)))
        alt = base.copy()
        if nz > 1e-10:
            alt, ok = project_capped(base + 0.12 * z / nz, n)
            if ok:
                s = exact_score(alt)
                if s < best_s:
                    best_s = s
                    best_h = alt.copy()

        try:
            bh, bs = evolve_stage(base, B, 0.10, P, G, rng, n)
            if bs < best_s:
                best_s = bs
                best_h = bh.copy()
        except Exception:
            pass
        try:
            bh, bs = evolve_stage(alt, B, 0.18, P, G, rng, n)
            if bs < best_s:
                best_s = bs
                best_h = bh.copy()
        except Exception:
            pass
        try:
            bh, bs = evolve_stage(best_h, B, 0.05, P, G, rng, n)
            if bs < best_s:
                best_s = bs
                best_h = bh.copy()
        except Exception:
            pass

    try:
        bh, bs = local_refine(best_h, n, rng, 4000)
        if bs < best_s:
            best_s = bs
            best_h = bh.copy()
    except Exception:
        pass
    try:
        bh, bs = pair_refine(best_h, n, rng, 8000)
        if bs < best_s:
            best_s = bs
            best_h = bh.copy()
    except Exception:
        pass
    try:
        bh, bs = local_refine(best_h, n, rng, 2000)
        if bs < best_s:
            best_s = bs
            best_h = bh.copy()
    except Exception:
        pass

    final_h, ok = project_capped(best_h, n)
    if not ok:
        final_h = base.copy()
    final_s = exact_score(final_h)
    fb_s = exact_score(base)
    if final_s > fb_s + 1e-12:
        final_h = base.copy()
        final_s = fb_s
    final_h, _ = project_capped(final_h, n)
    final_s = exact_score(final_h)
    return final_h.astype(np.float64), float(final_s), int(n)
