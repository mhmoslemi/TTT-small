            t = gen + 1
            p_sig = (1.0 - c_sigma) * p_sig + math.sqrt(c_sigma * (2.0 - c_sigma) * mu_eff) * (C_inv_sqrt @ y_w)
            nps = np.linalg.norm(p_sig)
            h_sig = 1.0 if nps / math.sqrt(1.0 - (1.0 - c_sigma) ** (2 * t)) < (1.4 + 2.0 / (d + 1)) * chi_d else 0.0
            p_c = (1.0 - c_c) * p_c + h_sig * math.sqrt(c_c * (2.0 - c_c) * mu_eff) * y_w
            C = (1.0 - c1 - c_mu) * C
            C = C + c1 * np.outer(p_c, p_c)
            if h_sig < 1.0:
                C = C + c1 * (1.0 - h_sig) * c_c * (2.0 - c_c) * C
            for i in range(mu_sel):
                C = C + c_mu * w[i] * np.outer(y_sel[i], y_sel[i])
            C = 0.5 * (C + C.T)
            expt = (c_sigma / d_sigma) * (nps / chi_d - 1.0)
            expt = max(-1.0, min(1.0, expt))
            sigma = max(1e-4, min(0.5, sigma * math.exp(expt)))
            mu = mu_new
            if abs(sigma - 1e-4) < 1e-8 or not np.all(np.isfinite(C)):
                break
            if scores[order[0]] >= best_in_run + 1e-12:
                stag += 1
            else:
                stag = 0
            if stag >= stag_limit:
                break
    best = project_box_sum(best, m)
    c5 = score_exact(best, N)
    b5 = score_exact(baseline, N)
    if not np.isfinite(c5) or c5 >= b5:
        best = baseline
        c5 = b5
    if not np.isfinite(c5):
        best = np.full(N, 0.5)
        c5 = score_exact(best, N)
    return best, float(c5), N
