        amp = 0.15 / np.sqrt(m) * rng.normal()
        u += amp * np.cos(2.0 * np.pi * m * (idx + 0.5) / n + phase)
    pu = project_capped_simplex(u, S)
    if pu is not None:
        starts.append(pu)
    pu2 = project_capped_simplex(rng.uniform(0.0, 1.0, n), S)
    if pu2 is not None:
        starts.append(pu2)
    deduped = []
    for s in starts:
        dup = False
        for d0 in deduped:
            if np.max(np.abs(s - d0)) < 1e-8:
                dup = True
                break
        if not dup and is_feasible(s, n):
            deduped.append(s)
    return deduped

def run(seed=42, **kwargs):
    if seed is None:
        seed = 42
    try:
        seed = int(seed)
    except (TypeError, ValueError):
        seed = 42
    rng = np.random.default_rng(seed)
    n = None
    parent = None
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        try:
            iv_arr = np.asarray(iv, dtype=np.float64)
            if iv_arr.ndim == 1 and 8 <= iv_arr.size <= 999:
                n = iv_arr.size
                if np.all(np.isfinite(iv_arr)) and np.max(np.abs(iv_arr)) <= 1e12:
                    S0 = n / 2.0
                    pr = project_capped_simplex(iv_arr, S0)
                    if pr is not None and is_feasible(pr, n):
                        parent = pr
        except Exception:
            pass
    if n is None:
        n = kwargs.get('n_points', 80)
        try:
            n = int(n)
        except (TypeError, ValueError):
            n = 80
        if n < 8 or n > 999:
            n = 80
    S = n / 2.0
    left, right, lag_id = build_pairs(n)
    const_fb = np.full(n, 0.5)
    best_h = const_fb.copy()
    best_score = exact_score(const_fb, n)
    if best_score is None:
        best_score = 0.5
    if parent is not None:
        ps = exact_score(parent, n)
        if ps is not None and np.isfinite(ps) and ps < best_score:
            best_score = ps
            best_h = parent.copy()
    starts = build_starts(n, S, parent, rng)
    for s in starts:
        sc = exact_score(s, n)
        if sc is not None and np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_h = s.copy()
    for s in starts:
        try:
            loc, loc_sc = project_descent_path(s, n, S, left, right, lag_id)
            if loc_sc is not None and np.isfinite(loc_sc) and loc_sc < best_score:
                best_score = loc_sc
                best_h = loc.copy()
        except Exception:
            pass
    nk = 3 if n <= 128 else (2 if n <= 256 else 1)
    sigma = 0.05
    for _ in range(nk):
        try:
            pert = best_h + sigma * rng.normal(0.0, 1.0, n)
            pr = project_capped_simplex(pert, S)
            if pr is None:
                continue
            old = best_score
            loc, loc_sc = project_descent_path(pr, n, S, left, right, lag_id)
            if loc_sc is not None and np.isfinite(loc_sc) and loc_sc < best_score:
                best_score = loc_sc
                best_h = loc.copy()
            sigma = max(0.02, 0.5 * sigma) if best_score < old - 1e-12 else min(0.25, 1.8 * sigma)
        except Exception:
            pass
    final_h = project_capped_simplex(best_h, S)
    if final_h is None or not is_feasible(final_h, n):
        final_h = parent.copy() if parent is not None else const_fb.copy()
    final_score = exact_score(final_h, n)
    if final_score is None or not np.isfinite(final_score):
        final_h = const_fb.copy()
        final_score = exact_score(final_h, n)
        if final_score is None:
            final_score = 0.5
    return final_h, final_score, n
