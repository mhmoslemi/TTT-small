def _smooth_value_grad(h, tau, N):
    one_minus = 1.0 - h
    corr = np.correlate(h, one_minus, mode='full')
    c = (2.0 / N) * corr
    cmax = float(np.max(c))
    e = np.exp((c - cmax) / tau)
    se = float(np.sum(e))
    val = cmax + tau * np.log(se)
    p = e / se
    grad = np.zeros(N)
    dx=2.0/N
    grad += dx * p[N-1] * (1.0 - 2.0*h)
    for d in range(1, N):
       L = N - d
       wx = dx * p[N-1 + d]
       if wx != 0.0:
           grad[:L] -= wx * h[d:]
           grad[d:] += wx * (1.0 - h[:L])
       wx = dx * p[N-1 - d]
       if wx != 0.0:
           grad[:L] += wx * (1.0 - h[d:])
           grad[d:] -= wx * h[:L]
    return val, grad
```

def project_box_sum(z, m):
    N = z.size
    if N == 0: return z
    if not np.all(np.isfinite(z)): return np.full(N,0.5)
    lo = float(z.min()) - 1.0
    hi = float(z.max())
    if not (lo < hi): lo-=1; hi+=1
    for _ in range(80):
       mid=0.5*(lo+hi)
       s=np.clip(z-mid,0,1).sum()
       if s > m: lo=mid
       else: hi=mid
    h=np.clip(z-0.5*(lo+hi),0,1)
    r=m - h.sum()
    for _ in range(3):
       if r > 1e-14:
          for i in range(N):
             cap=1.0-h[i]
             if cap>0:
                adj=min(r,cap); h[i]+=adj; r-=adj; 
                if r<=1e-14: break
       elif r < -1e-14:
          for i in range(N):
             if h[i]>0:
                adj=min(-r,h[i]); h[i]-=adj; r+=adj
                if r>=-1e-14: break
       else: break
    if not np.all(np.isfinite(h)) or h.min()< -1e-9 or h.max()>1+1e-9 or abs(h.sum()-m)>1e-7:
       return np.full(N,0.5)
    # remove tiny bound violations without changing sum much
    h = np.clip(h,0.0,1.0)
    r=m-h.sum()
    if abs(r)>1e-12:
       # distribute r to entries with slack or positive
       if r>0:
          for i in range(N):
             if h[i]<1.0:
                adj=min(r,1.0-h[i]); h[i]+=adj; r-=adj
                if r<=1e-15: break

                if r>=-1e-15: break

    return h

import numpy as np
from scipy.optimize import minimize

def project_box_sum(z, m):
    z = np.asarray(z, dtype=np.float64).ravel()
    N = z.size
    if N == 0:
        return z
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = float(z.min()) - 1.0
    hi = float(z.max())
    if hi <= lo:
        lo -= 1.0; hi += 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - float(h.sum())
    for _ in range(3):
        if r > 1e-14:
            for i in range(N):
                cap = 1.0 - h[i]
                if cap > 0.0:
                    adj = min(r, cap)
                    h[i] += adj
                    r -= adj
                    if r <= 1e-14:
                        break
        elif r < -1e-14:
            for i in range(N):
                if h[i] > 0.0:
                    adj = min(-r, h[i])
                    h[i] -= adj
                    r += adj
                    if r >= -1e-14:
                        break
        else:
            break
    if not np.all(np.isfinite(h)) or h.min() < -1e-9 or h.max() > 1.0 + 1e-9:
        return np.full(N, 0.5)
    if abs(float(h.sum()) - m) > 1e-7:
        return np.full(N, 0.5)
    return h

def smooth_value_grad(h, tau, N):
    h = np.asarray(h, dtype=np.float64)
    g = np.zeros(N, dtype=np.float64)
    if not np.all(np.isfinite(h)) or tau <= 0.0:
        return float('inf'), g
    if N <= 0:
        return 0.0, g
    dx = 2.0 / N
    cvals = np.empty(2 * N - 1, dtype=np.float64)
    h1 = 1.0 - h
    cvals[N-1] = dx * float(np.sum(h * h1))
    for d in range(1, N):
        L = N - d
        cvals[N-1-d] = dx * float(np.sum(h[d:] * h1[:L]))
        cvals[N-1+d] = dx * float(np.sum(h[:L] * h1[d:]))
    if not np.all(np.isfinite(cvals)):
        return float('inf'), g
    cmax = float(np.max(cvals))
    e = np.exp((cvals - cmax) / tau)
    se = float(np.sum(e))
    if not np.isfinite(se) or se <= 0.0:
        return float('inf'), g
    p = e / se
    val = cmax + tau * float(np.log(se))
    g += dx * p[N - 1] * (1.0 - 2.0 * h)
    for d in range(1, N):
        L = N - d
        ps_neg = dx * p[N - 1 - d]
        if ps_neg != 0.0:
            g[d:] += ps_neg * (1.0 - h[:L])
            g[:L] -= ps_neg * h[d:]
        ps_pos = dx * p[N - 1 + d]
        if ps_pos != 0.0:
            g[:L] += ps_pos * (1.0 - h[d:])
            g[d:] -= ps_pos * h[:L]
    below = h < 0.0
    above = h > 1.0
    if np.any(below):
        hb = h[below]
        val += 1.0e6 * float(np.dot(hb, hb))
        g[below] -= 2.0e6 * hb
    if np.any(above):
        ha = h[above] - 1.0
        val += 1.0e6 * float(np.dot(ha, ha))
        g[above] += 2.0e6 * ha
    if not np.isfinite(val):
        return float('inf'), g
    return float(val), g
```

    h = np.asarray(h, dtype=np.float64)

    cvals = np.empty(2 * N - 1, dtype=np.float64)
    cvals[N - 1] = dx * float(np.sum(h * h1))
    for d in range(1, N):

    p = e / se

import numpy as np
from scipy.optimize import minimize


def project_box_sum(z, m):
    z = np.asarray(z, dtype=np.float64).ravel()
    N = z.size
    if N == 0:
        return z
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)

    lo = float(z.min()) - 1.0
    hi = float(z.max())
    if hi <= lo:
        lo -= 1.0
        hi += 1.0

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid

    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - float(h.sum())

    for _ in range(3):
        if r > 1e-14:
            for i in range(N):
                cap = 1.0 - h[i]
                if cap > 0.0:
                    adj = min(r, cap)
                    h[i] += adj
                    r -= adj
                    if r <= 1e-14:
                        break
        elif r < -1e-14:
            for i in range(N):
                if h[i] > 0.0:
                    adj = min(-r, h[i])
                    h[i] -= adj
                    r += adj
                    if r >= -1e-14:
                        break
        else:
            break

    if not np.all(np.isfinite(h)) or h.min() < -1e-9 or h.max() > 1.0 + 1e-9:
        return np.full(N, 0.5)
    if abs(float(h.sum()) - m) > 1e-7:
        return np.full(N, 0.5)
    return h


def feasible_h(h, N, m):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return False
    if h.min() < -1e-10 or h.max() > 1.0 + 1e-10:
        return False
    return abs(float(h.sum()) - m) <= 1e-8 * max(1, N)


def exact_score(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return float("inf")
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / N))


def box_penalty(h):
    h = np.asarray(h, dtype=np.float64)
    val = 0.0
    g = np.zeros_like(h)
    below = h < 0.0
    if np.any(below):
        hb = h[below]
        val += 1.0e5 * float(np.dot(hb, hb))
        g[below] -= 2.0e5 * hb
    above = h > 1.0
    if np.any(above):
        ha = h[above] - 1.0
        val += 1.0e5 * float(np.dot(ha, ha))
        g[above] += 2.0e5 * ha
    return val, g


def cvals_p(h, tau, N):
    h = np.asarray(h, dtype=np.float64)
    if h.size != N or not np.all(np.isfinite(h)) or tau <= 0.0 or N <= 0:
        return None, None

    dx = 2.0 / float(N)
    h1 = 1.0 - h
    cvals = np.empty(2 * N - 1, dtype=np.float64)

    cvals[N - 1] = dx * float(np.dot(h, h1))
    for d in range(1, N):
        L = N - d
        cvals[N - 1 - d] = dx * float(np.dot(h[d:], h1[:L]))
        cvals[N - 1 + d] = dx * float(np.dot(h[:L], h1[d:]))

    if not np.all(np.isfinite(cvals)):
        return None, None

    cmax = float(np.max(cvals))
    e = np.exp((cvals - cmax) / tau)
    se = float(np.sum(e))
    if not np.isfinite(se) or se <= 0.0:
        return None, None

    p = e / se
    val = float(cmax + tau * np.log(se))
    return val, p


def smooth_value(h, tau, N):
    val, _ = cvals_p(h, tau, N)
    if val is None:
        return 1.0e100
    pen, _ = box_penalty(h)
    val += pen
    return float(val) if np.isfinite(val) else 1.0e100


def smooth_value_grad(h, tau, N):
    h = np.asarray(h, dtype=np.float64)
    g = np.zeros(N, dtype=np.float64)
    val, p = cvals_p(h, tau, N)
    if val is None:
        return 1.0e100, g

    dx = 2.0 / float(N)
    h1 = 1.0 - h
    g += dx * p[N - 1] * (1.0 - 2.0 * h)

    for d in range(1, N):
        L = N - d
        ps_neg = dx * p[N - 1 - d]
        if ps_neg != 0.0:
            g[d:] += ps_neg * h1[:L]
            g[:L] -= ps_neg * h[d:]
        ps_pos = dx * p[N - 1 + d]
        if ps_pos != 0.0:
            g[:L] += ps_pos * h1[d:]
            g[d:] -= ps_pos * h[:L]

    pen, pg = box_penalty(h)
    val += pen
    g += pg

    if not np.isfinite(val) or not np.all(np.isfinite(g)):
        return 1.0e100, g
    return float(val), g


def transform_h(x, m):
    x = np.asarray(x, dtype=np.float64).ravel()
    h = np.empty(x.size + 1, dtype=np.float64)
    h[:x.size] = x
    h[-1] = m - float(np.sum(x))
    return h


def fit_obj(x, tau, N, m):
    return smooth_value(transform_h(x, m), tau, N)


def fit_jac(x, tau, N, m):
    _, g = smooth_value_grad(transform_h(x, m), tau, N)
    if g.size != N or not np.all(np.isfinite(g)):
        return np.zeros(max(N - 1, 0), dtype=np.float64)
    return g[:N - 1] - g[N - 1]


def sum_low_fun(x, m):
    return np.array([float(np.sum(x) - (m - 1.0))])


def sum_low_jac(x, m):
    n = int(np.size(x))
    J = np.zeros((1, n), dtype=np.float64)
    if n:
        J[0, :] = 1.0
    return J


def sum_high_fun(x, m):
    return np.array([float(m - np.sum(x))])


def sum_high_jac(x, m):
    n = int(np.size(x))
    J = np.zeros((1, n), dtype=np.float64)
    if n:
        J[0, :] = -1.0
    return J


def add_start(cand, starts, m):
    cand = project_box_sum(cand, m)
    if not feasible_h(cand, cand.size, m):
        return starts
    for s in starts:
        if np.max(np.abs(cand - s)) <= 1e-9:
            return starts
    starts.append(cand)
    return starts


def fourier_start(N, rng, amp, n_modes):
    z = np.full(N, 0.5)
    idx = np.arange(N, dtype=np.float64)
    rmax = min(int(n_modes), (N - 1) // 2)
    for r in range(1, rmax + 1):
        A = rng.standard_normal()
        B = rng.standard_normal()
        t = 2.0 * np.pi * r * idx / float(N)
        z += (amp / float(r)) * (A * np.cos(t) + B * np.sin(t))
    return z


def pattern_start(N, kind, rng):
    idx = np.arange(N, dtype=np.float64)
    if kind == 0:
        phi = 2.0 * np.pi * rng.uniform()
        z = 0.5 + 0.5 * np.sin(2.0 * np.pi * idx * idx / float(N) + phi)
    elif kind == 1:
        phi = 2.0 * np.pi * rng.uniform()
        freq = 1.0 + rng.uniform()
        z = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * idx / float(N) + phi)
    else:
        freq = 0.25 + 0.75 * rng.uniform()
        z = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * idx / float(N))
    return z


def slsqp_tracks(x0, m, N, tau_list, maxiter):
    if N < 2 or maxiter <= 0 or len(tau_list) == 0:
        return [project_box_sum(transform_h(x0, m), m)]

    h = project_box_sum(transform_h(x0, m), m)
    x = h[:N - 1].copy()
    candidates = [h]

    bounds = [(0.0, 1.0)] * (N - 1)
    cons = [
        {"type": "ineq", "fun": sum_low_fun, "jac": sum_low_jac, "args": (m,)},
        {"type": "ineq", "fun": sum_high_fun, "jac": sum_high_jac, "args": (m,)},
    ]

    for tau in tau_list:
        try:
            res = minimize(
                fit_obj,
                x,
                jac=fit_jac,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                args=(tau, N, m),
                options={"maxiter": int(maxiter), "ftol": 1e-11, "maxls": 40, "disp": False},
            )
        except Exception:
            break

        if res is None or getattr(res, "x", None) is None or not np.all(np.isfinite(res.x)):
            break

        x = np.asarray(res.x, dtype=np.float64)
        h = project_box_sum(transform_h(x, m), m)
        x = h[:N - 1].copy()
        candidates.append(h)

    return candidates


def pg_optimize(h0, m, N, tau_list, max_iter, best_h, best_score):
    if N < 1:
        return h0, best_h, best_score

    h = project_box_sum(h0, m)
    s = exact_score(h, N)
    if s < best_score:
        best_h = h.copy()
        best_score = float(s)

    for tau in tau_list:
        val, grad = smooth_value_grad(h, tau, N)
        if not (np.isfinite(val) and np.all(np.isfinite(grad))):
            continue

        alpha = 1.0
        step = 1.0
        for _ in range(int(max_iter)):
            if step <= 1e-9:
                break
            accepted = False
            for _ in range(24):
                y = project_box_sum(h - alpha * grad, m)
                if not feasible_h(y, N, m):
                    alpha *= 0.5
                    continue

                sy = exact_score(y, N)
                if sy < best_score:
                    best_h = y.copy()
                    best_score = float(sy)

                Fy, _ = smooth_value_grad(y, tau, N)
                if np.isfinite(Fy):
                    dy = y - h
                    gd = float(np.dot(grad, dy))
                    if (gd < 0.0 and Fy <= val + 1e-4 * gd + 1e-12) or (Fy < val - 1e-12):
                        h = y.copy()
                        val = Fy
                        step = float(np.max(np.abs(dy)))
                        alpha = min(1.25 * alpha, 10.0)
                        accepted = True
                        break
                alpha *= 0.5
            if not accepted:
                break

    h = project_box_sum(h, m)
    s = exact_score(h, N)
    if s < best_score:
        best_h = h.copy()
        best_score = float(s)
    return h, best_h, best_score


def run(seed=42, **kwargs):
    raw = globals().get("initial_h_values", None)
    N = None
    base = None

    if raw is not None:
        arr = np.asarray(raw, dtype=np.float64).ravel()
        if arr.size >= 2 and np.all(np.isfinite(arr)):
            N = int(arr.size)
            base = arr

    if N is None:
        try:
            N = int(kwargs.get("n_points", 70))
        except Exception:
            N = 70
        if N < 2 or N > 999:
            N = 70
        base = np.full(N, 0.5)

    if N == 1:
        h = np.array([0.5], dtype=np.float64)
        return h, exact_score(h, 1), 1

    m = N / 2.0
    parent_h = project_box_sum(base, m)
    best_h = parent_h.copy()
    best_score = exact_score(best_h, N)
    initial_best_score = best_score

    if N <= 512:
        try:
            rng = np.random.default_rng(seed)
        except Exception:
            rng = np.random.default_rng(42)

        if N <= 128:
            max_starts = 5
            maxit_first = 90
            maxit_other = 60
            do_hops = True
            hop_scales = [0.04, 0.012, 0.004, 0.0015]
            hop_maxit = 50
            do_pg = True
            pg_taus = [0.0002, 0.00006, 0.00002]
            pg_iters = 25
        elif N <= 256:
            max_starts = 4
            maxit_first = 60
            maxit_other = 45
            do_hops = True
            hop_scales = [0.03]
            hop_maxit = 35
            do_pg = True
            pg_taus = [0.0002, 0.00006]
            pg_iters = 12
        else:
            max_starts = 3
            maxit_first = 40
            maxit_other = 30
            do_hops = False
            hop_scales = []
            hop_maxit = 0
            do_pg = False
            pg_taus = []
            pg_iters = 0

        uniform = float(np.max(np.abs(best_h - 0.5))) < 1e-14
        starts = [] if uniform else [best_h.copy()]

        cands = []
        cands.append(1.0 - best_h)
        cands.append(best_h + 0.10 * rng.standard_normal(N))
        cands.append(fourier_start(N, rng, 0.12, 6))
        cands.append(pattern_start(N, 0, rng))
        cands.append(rng.uniform(size=N))
        cands.append(best_h + 0.25 * rng.standard_normal(N))

        for cand in cands:
            if len(starts) >= max_starts:
                break
            starts = add_start(cand, starts, m)

        if not starts:
            starts = [best_h.copy()]

        full_taus = [0.02, 0.005, 0.0015, 0.0005, 0.00015, 0.00005]
        for si, h0 in enumerate(starts):
            maxit = maxit_first if si == 0 else maxit_other
            tracks = slsqp_tracks(h0[:N - 1], m, N, full_taus, maxit)
            for ht in tracks:
                st = exact_score(ht, N)
                if st < best_score:
                    best_h = ht.copy()
                    best_score = float(st)

        if best_score < initial_best_score - 1e-12:
            extra_taus = [0.001, 0.0003, 0.0001]
            tracks = slsqp_tracks(best_h[:N - 1], m, N, extra_taus, 50)
            for ht in tracks:
                st = exact_score(ht, N)
                if st < best_score:
                    best_h = ht.copy()
                    best_score = float(st)

        if do_hops:
            hop_taus = [0.0008, 0.00025, 0.00008]
            for sc in hop_scales:
                h0 = project_box_sum(best_h + sc * rng.standard_normal(N), m)
                sh = exact_score(h0, N)
                if sh < best_score:
                    best_h = h0.copy()
                    best_score = float(sh)

                tracks = slsqp_tracks(h0[:N - 1], m, N, hop_taus, hop_maxit)
                for ht in tracks:
                    st = exact_score(ht, N)
                    if st < best_score:
                        best_h = ht.copy()
                        best_score = float(st)

        if do_pg:
            _, best_h, best_score = pg_optimize(best_h, m, N, pg_taus, pg_iters, best_h, best_score)

    final_h = project_box_sum(best_h, m)
    c5 = exact_score(final_h, N)
    parent_score = exact_score(parent_h, N)

    if parent_score < c5:
        final_h = parent_h
        c5 = parent_score

    if not np.isfinite(c5):
        final_h = np.full(N, 0.5)
        c5 = exact_score(final_h, N)

    final_h = project_box_sum(final_h, m)
    c5 = exact_score(final_h, N)

    if not np.isfinite(c5):
        final_h = parent_h
        c5 = exact_score(final_h, N)

    return final_h, float(c5), int(N)
