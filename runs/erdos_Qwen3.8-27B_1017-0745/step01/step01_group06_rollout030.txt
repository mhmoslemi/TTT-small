        mx, nT, M, npol = 2, 6, 6, 8

    starts = [best_h.copy()]
    for sc, hv in scored:
        if len(starts) >= mx:
            break
        if not any(np.max(np.abs(hv - s)) <= 1e-7 for s in starts):
            starts.append(hv.copy())

    for s in starts:
        try:
            r = anneal_chain(s, n, S, rng, nT, M)
            if r is not None:
                rh, rf = r
                if feasible(rh, n, S):
                    rf = exact_score(rh, n)
                    if rf < best_s - 1e-12:
                        best_s, best_h = rf, rh.copy()
        except Exception:
            pass

    try:
        gh = grad_polish(best_h, n, S, 40)
        gs = exact_score(gh, n)
        if gs < best_s - 1e-12:
            best_s, best_h = gs, gh.copy()
    except Exception:
        pass

    try:
        p2h, p2f = zeroT_polish(best_h, n, S, rng, npol, [0.02, 0.006])
        if feasible(p2h, n, S):
            p2f = exact_score(p2h, n)
            if p2f < best_s - 1e-12:
                best_s, best_h = p2f, p2h.copy()
    except Exception:
        pass

    try:
        gh2 = grad_polish(best_h, n, S, 30)
        gs2 = exact_score(gh2, n)
        if gs2 < best_s - 1e-12:
            best_s, best_h = gs2, gh2.copy()
    except Exception:
        pass

    best_h = project_K(best_h, n, S)
    fs = exact_score(best_h, n)
    if not feasible(best_h, n, S) or not np.isfinite(fs) or fs > fs_fb + 1e-10:
        best_h, fs = fb.copy(), fs_fb
    best_h = np.clip(best_h, 0.0, 1.0)
    fs = exact_score(best_h, n)
    if not np.isfinite(fs):
        best_h, fs = fb.copy(), fs_fb
    return (best_h, float(fs), n)
