    if n > 1 and NP >= 4 and gens > 0:
        pop = [s.copy() for s in starts[:NP]]
        for _ in range(NP - len(pop)):
            base = pop[rng.integers(len(pop))] if pop else uniform
            sc = [0.015, 0.04, 0.10, 0.22][rng.integers(4)]
            cand = base + rng.normal(0, sc, n)
            hn = project_capped(cand, n)
            if hn is None:
                hn = project_capped(rng.uniform(0, 1, n), n)
            if hn is None:
                hn = uniform.copy()
            pop.append(hn)
        pop = pop[:NP]
        scores = [exact_score(p, n) for p in pop]
        for i in range(NP):
            if np.isfinite(scores[i]) and scores[i] < best_score - 1e-12:
                best, best_score = pop[i].copy(), scores[i]
        patience = max(12, min(40, gens // 4))
        stag, rst_cnt = 0, 0
        for gen in range(gens):
            xb = best_score
            pop, scores = de_step(pop, scores, NP, n, rng)
            for i in range(NP):
                if np.isfinite(scores[i]) and scores[i] < best_score - 1e-12:
                    best, best_score = pop[i].copy(), scores[i]
            if best_score < xb - 1e-12:
                stag = 0
            else:
                stag += 1
            if stag >= patience and rst_cnt < mrst:
                rst_cnt += 1
                order = np.argsort(scores)
                new_pop = [best.copy()]
                for k in order:
                    if len(new_pop) >= max(2, math.ceil(0.2 * NP)):
                        break
                    if float(np.max(np.abs(pop[k] - best))) > 1e-10:
                        new_pop.append(pop[k].copy())
                scs = [0.03, 0.07, 0.15][min(rst_cnt - 1, 2)] * rng.uniform(0.5, 1.5)
                while len(new_pop) < NP:
                    base = new_pop[rng.integers(len(new_pop))] if new_pop else uniform
                    hn = project_capped(base + rng.normal(0, scs, n), n)
                    if hn is None:
                        hn = project_capped(rng.uniform(0, 1, n), n)
                    if hn is None:
                        hn = uniform.copy()
                    new_pop.append(hn)
                    hs = exact_score(hn, n)
                    if np.isfinite(hs) and hs < best_score - 1e-12:
                        best, best_score = hn.copy(), hs
                pop = new_pop[:NP]
                scores = [exact_score(p, n) for p in pop]
                for i in range(NP):
                    if np.isfinite(scores[i]) and scores[i] < best_score - 1e-12:
                        best, best_score = pop[i].copy(), scores[i]
                stag = 0
            elif stag >= patience:
                break
    cands = [best, uniform]
    if parent_fb is not None:
        cands.append(parent_fb)
    cs = [exact_score(c, n) for c in cands]
    bi = int(np.argmin(cs))
    final = project_capped(cands[bi].copy(), n)
    if final is None:
        final = uniform.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final.copy(), float(fs), int(n)
