```
def epi_obj(z, n):
    return float(z[-1])

def epi_obj_jac(z, n):
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return g

def epi_ineq(z, n):
    h = z[:n]
    if not np.all(np.isfinite(h)):
        return np.full(2*n - 1, -1e100)
    q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / float(n))
    if not np.all(np.isfinite(q)):
        return np.full(2*n - 1, -1e100)
    return z[-1] - q

def epi_ineq_jac(z, n):
    h = z[:n]
    L = 2*n - 1
    G = np.zeros((L, n), dtype=np.float64)
    if h.size == n and np.all(np.isfinite(h)):
       dx = 2.0 / float(n)
       for d in range(-(n-1), n):
          row = d + n - 1
          p0 = d if d > 0 else 0
          q0 = -d if d < 0 else 0
          m = n - abs(d)
          G[row, p0:p0 + m] += dx * (1.0 - h[q0:q0 + m])
          G[row, q0:q0 + m] -= dx * h[p0:p0 + m]
    J = np.empty((L, n + 1), dtype=np.float64)
    J[:, :n] = -G
    J[:, n] = 1.0
    if not np.all(np.isfinite(J)):
       J.fill(0.0)
    return J

def epi_eq(z,n):
    return float(np.sum(z[:n]) - 0.5*float(n))

def epi_eq_jac(z,n):
    J=np.zeros((1,n+1)); J[0,:n]=1.0; return J
```

```
    if h is None or not is_feasible(h,n):
        return h0, exact_score(h0,n)
    sc = exact_score(h,n)
    for _ in range(2):
        q = np.correlate(h, 1.0-h, mode='full')*(2.0/float(n))
        if not np.all(np.isfinite(q)):
            break
        t0 = float(np.max(q)) + max(1e-8, 1e-7*max(1.0, abs(float(np.max(q)))))
        z0 = np.concatenate((h, np.array([t0], dtype=np.float64)))
        bounds = [(0.0,1.0)] * n

            {'type': 'eq', 'fun': epi_eq, 'jac': epi_eq_jac, 'args': (n,)}
        )
        try:
           res = minimize(epi_obj, z0, method='SLSQP', jac=epi_obj_jac, args=(n,), constraints=cons, bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-8})
```

```python
import numpy as np
from scipy.optimize import minimize


def is_feasible(h, n):
    if h is None or h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < 0.0) or np.any(h > 1.0):
        return False
    return abs(float(np.sum(h)) - 0.5 * float(n)) <= 1e-9 * max(1, n)


def project_capped(z, n):
    n = int(n)
    if n <= 0:
        return None
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n:
        return None
    z = np.where(np.isfinite(z), z, 0.0)
    z = np.clip(z, -1e12, 1e12)

    s = 0.5 * float(n)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0

    for _ in range(90):
        mid = 0.5 * (lo + hi)
        c = float(np.sum(np.clip(z - mid, 0.0, 1.0)))
        if c > s:
            lo = mid
        else:
            hi = mid

    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))

    for _ in range(8):
        if abs(r) <= 1e-13 * max(1.0, float(np.sqrt(n))):
            break
        if r > 0.0:
            room = 1.0 - h
            cap = float(np.sum(room))
            if cap <= 1e-15:
                break
            h = np.clip(h + (r * room) / cap, 0.0, 1.0)
        else:
            cap = float(np.sum(h))
            if cap <= 1e-15:
                break
            h = np.clip(h + (r * h) / cap, 0.0, 1.0)
        r = s - float(np.sum(h))

    if not np.all(np.isfinite(h)) or np.any(h < 0.0) or np.any(h > 1.0):
        return None
    if abs(float(np.sum(h)) - s) > 1e-8 * max(1, n):
        h = np.clip(h + (s - float(np.sum(h))) / float(n), 0.0, 1.0)
        if abs(float(np.sum(h)) - s) > 1e-7 * max(1, n):
            return None
    return h


def exact_score(h, n):
    if not is_feasible(h, n):
        return float("inf")
    q = np.correlate(h, 1.0 - h, mode="full")
    q = q * (2.0 / float(n))
    if q.size != 2 * n - 1 or not np.all(np.isfinite(q)):
        return float("inf")
    return float(np.max(q))


def add_candidate(lst, v, n):
    if v is None:
        return
    p = project_capped(v, n)
    if p is None or not is_feasible(p, n):
        return
    for old in lst:
        if np.max(np.abs(p - old)) < 1e-10:
            return
    lst.append(p)


def candidate_pool(base, n):
    pool = []
    add_candidate(pool, base, n)
    add_candidate(pool, np.full(n, 0.5), n)

    i = np.arange(n, dtype=np.float64)
    u = (i + 0.5) / float(n)
    bx = np.cos(np.pi * u)
    bx2 = np.cos(2.0 * np.pi * u)

    if n <= 80:
        amp_list = (1.0, 0.95, 0.90, 0.88, 0.85, 0.80)
        a_list = (1.0, 0.95, 0.90, 0.85)
        b_list = (-0.4, -0.2, -0.1, 0.0, 0.1, 0.2, 0.4)
    elif n <= 200:
        amp_list = (1.0, 0.90, 0.85)
        a_list = (0.95, 0.90)
        b_list = (-0.2, 0.0, 0.2)
    else:
        amp_list = (1.0, 0.90)
        a_list = (0.90,)
        b_list = (0.0,)

    for amp in amp_list:
        add_candidate(pool, 0.5 + 0.5 * amp * bx, n)
        add_candidate(pool, 0.5 - 0.5 * amp * bx, n)

    for a in a_list:
        for b in b_list:
            x = a * bx + b * bx2
            add_candidate(pool, 0.5 + 0.5 * x, n)
            x = -a * bx + b * bx2
            add_candidate(pool, 0.5 + 0.5 * x, n)

    add_candidate(pool, 1.0 - u, n)
    add_candidate(pool, u, n)
    return pool


def item_score(item):
    return item[0]


def ranked_candidates(pool, n, k):
    scored = []
    for p in pool:
        s = exact_score(p, n)
        if np.isfinite(s):
            scored.append((s, p))
    if not scored:
        return scored
    scored.sort(key=item_score)
    return scored[:k] if k > 0 else scored


def epi_obj(z, n):
    if z is None or z.size != n + 1 or not np.all(np.isfinite(z)):
        return 1e100
    return float(z[-1])


def epi_obj_jac(z, n):
    g = np.zeros(n + 1, dtype=np.float64)
    g[-1] = 1.0
    return g


def epi_ineq(z, n):
    L = 2 * n - 1
    if z is None or z.size != n + 1:
        return np.full(L, -1e100, dtype=np.float64)
    h = z[:n]
    if not np.all(np.isfinite(h)) or not np.isfinite(z[-1]):
        return np.full(L, -1e100, dtype=np.float64)
    q = np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))
    if not np.all(np.isfinite(q)):
        return np.full(L, -1e100, dtype=np.float64)
    return z[-1] - q


def epi_ineq_jac(z, n):
    L = 2 * n - 1
    if z is None or z.size != n + 1:
        return np.zeros((L, n + 1), dtype=np.float64)
    h = z[:n]
    G = np.zeros((L, n), dtype=np.float64)
    if np.all(np.isfinite(h)):
        dx = 2.0 / float(n)
        for d in range(-(n - 1), n):
            row = d + n - 1
            p0 = d if d > 0 else 0
            q0 = -d if d < 0 else 0
            m = n - abs(d)
            G[row, p0:p0 + m] += dx * (1.0 - h[q0:q0 + m])
            G[row, q0:q0 + m] -= dx * h[p0:p0 + m]
    J = np.empty((L, n + 1), dtype=np.float64)
    J[:, :n] = -G
    J[:, n] = 1.0
    if not np.all(np.isfinite(J)):
        J.fill(0.0)
    return J


def epi_eq(z, n):
    if z is None or z.size != n + 1 or not np.all(np.isfinite(z[:n])):
        return 0.0
    return float(np.sum(z[:n]) - 0.5 * float(n))


def epi_eq_jac(z, n):
    J = np.zeros((1, n + 1), dtype=np.float64)
    J[0, :n] = 1.0
    return J


def slsqp_polish(h0, n, maxiter=25):
    h = project_capped(h0, n)
    if h is None or not is_feasible(h, n):
        return h0, exact_score(h0, n)

    sc = exact_score(h, n)
    for _ in range(2):
        q = np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))
        if not np.all(np.isfinite(q)):
            break
        t0 = float(np.max(q)) + max(1e-8, 1e-7 * max(1.0, abs(float(np.max(q)))))
        z0 = np.concatenate((h, np.array([t0], dtype=np.float64)))

        bounds = [(0.0, 1.0)] * n
        bounds.append((None, None))
        cons = (
            {"type": "ineq", "fun": epi_ineq, "jac": epi_ineq_jac, "args": (n,)},
            {"type": "eq", "fun": epi_eq, "jac": epi_eq_jac, "args": (n,)}
        )

        try:
            res = minimize(
                epi_obj,
                z0,
                method="SLSQP",
                jac=epi_obj_jac,
                args=(n,),
                constraints=cons,
                bounds=bounds,
                options={"maxiter": maxiter, "ftol": 1e-8, "disp": False},
            )
        except Exception:
            break

        if res is None or getattr(res, "x", None) is None:
            break
        x = np.asarray(res.x, dtype=np.float64).ravel()
        if x.size != n + 1 or not np.all(np.isfinite(x)):
            break

        y = project_capped(x[:n], n)
        if y is None or not is_feasible(y, n):
            break
        sy = exact_score(y, n)
        if np.isfinite(sy) and sy < sc - 1e-12:
            h = y
            sc = sy
        else:
            break

    return h, float(sc)
```
