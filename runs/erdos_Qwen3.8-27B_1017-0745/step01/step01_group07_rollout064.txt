import numpy as np

def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):
        return None
    if np.max(np.abs(z)) > 1e12:
        return None
    n = len(z)
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z))
    for _ in range(80):
        mid = lo + 0.5*(hi - lo)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = lo + 0.5*(hi - lo)
    y = np.clip(z - lam, 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 0:
        for i in range(n):
            if r <= 0: break
            a = min(r, 1.0 - y[i])
            y[i] += a; r -= a
    elif r < 0:
        for i in range(n):
            if r >= 0: break
            a = min(-r, y[i])
            y[i] -= a; r += a
    if abs(float(np.sum(y)) - S) > 1e-10 or not np.all(np.isfinite(y)):
        return None
    return np.clip(y, 0.0, 1.0)

    b = np.zeros(2*n-1)
    b[n-1-p : 2*n-1-p] += (1.0 - h)
    b[n-1-r : 2*n-1-r] -= (1.0 - h)
    b[p : p+n] += -(h[::-1])
    b[r : r+n] += h[::-1]
    b *= dx
    a = np.zeros(2*n-1)
    a[n-1] = -2.0*dx
    a[n-1 + (r-p)] = dx
    a[n-1 + (p-r)] = dx
    return b, a

def line_search(h, p, r, q, score, n, dx):
    L = max(-h[p], h[r] - 1.0)
    U = min(1.0 - h[p], h[r])
    if U - L < 1e-12:
        return None
    b, a = compute_ba(h, p, r, n, dx)
    # Coarse grid: 17 points + t=0
    t_grid = np.linspace(L, U, 17)
    if abs(0.0 - L) > 1e-15 and abs(0.0 - U) > 1e-15:
        t_grid = np.insert(t_grid, len(t_grid)//2, 0.0)
    best_t = t_grid[0]
    best_f = q[0] + b[0]*t_grid[0] + a[0]*t_grid[0]**2
    for j in range(1, len(t_grid)-1):
        # Actually compute max over all lags
        vals = q + b*t_grid[j] + a*t_grid[j]**2
        f = float(np.max(vals))
        if f < best_f - 1e-15 or (abs(f - best_f) <= 1e-15 and abs(t_grid[j]) < abs(best_t)):
            best_f = f
            best_t = t_grid[j]
    # Also check t=0
    # Actually t=0 gives f = score, so we only want to accept if best_f < score
    if best_f >= score - 1e-12:
        return None
    # Refine 4 times
    radius = (U - L) / 16.0
    for _ in range(4):
        tlo = max(L, best_t - radius)
        thi = min(U, best_t + radius)
        t_ref = np.linspace(tlo, thi, 5)
        # Include best_t
        t_ref = np.concatenate([t_ref, [best_t]])
        for tj in t_ref:
            vals = q + b*tj + a*tj**2
            f = float(np.max(vals))
            if f < best_f - 1e-15 or (abs(f - best_f) <= 1e-15 and abs(tj) < abs(best_t)):
                best_f = f
                best_t = tj
        radius /= 4.0
    # Construct new h
    h_new = h.copy()
    h_new[p] += best_t
    h_new[r] -= best_t
    if h_new[p] < -1e-12 or h_new[p] > 1.0+1e-12 or h_new[r] < -1e-12 or h_new[r] > 1.0+1e-12:
        h_new = project_capped_simplex(h_new, n/2.0)
        if h_new is None:
            return None
    # Verify with exact score
    s_new = exact_score(h_new, n)
    if s_new is None or s_new >= score - 1e-12:
        return None
    q_new = exact_lags(h_new, n)
    if q_new is None:
        return None
    return h_new, s_new, q_new

def build_pair_list(n, rng, max_pairs):
    pairs = []
    # Adjacent pairs
    n_adj = min(n-1, max(1, max_pairs//2))
    if n_adj == n-1:
        adj_idx = list(range(n-1))
    else:
        adj_idx = [int(i*(n-1)/n_adj) for i in range(n_adj)]
    for i in adj_idx:
        pairs.append((i, i+1))
    # Random pairs
    while len(pairs) < max_pairs:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        if i > j:
            i, j = j, i
        if (i, j) not in pairs:
            pairs.append((i, j))
        if len(pairs) >= max_pairs:
            break
    # Fallback: enumerate
    if len(pairs) < max_pairs:
        for i in range(n):
            for j in range(i+1, n):
                if len(pairs) >= max_pairs:
                    break
                if (i, j) not in pairs:
                    pairs.append((i, j))
            if len(pairs) >= max_pairs:
                break
    rng.shuffle(pairs)
    return pairs[:max_pairs]

def polish_sweep(h, score, q, n, dx, rng, n_sweeps):
    max_pairs = min(n*(n-1)//2, 2*n, max(24, 80000//n))
    for sw in range(n_sweeps):
        pairs = build_pair_list(n, rng, max_pairs)
        improved = False
        for (p, r) in pairs:
            result = line_search(h, p, r, q, score, n, dx)
            if result is not None:
                h, score, q = result
                improved = True
        if not improved:
            break
    return h, score, q

def build_starts(n, S, parent, rng):
    starts = []
    if parent is not None:
        starts.append(parent.copy())
    # Low-frequency sinusoidal
    idx = np.arange(n, dtype=np.float64)
    M = min(4, (n-1)//2)
    u = np.full(n, 0.5)
    for m in range(1, M+1):
        A = 0.12/np.sqrt(m) * rng.normal()
        phi = rng.uniform(0, 2*np.pi)
        u += A * np.cos(2*np.pi*m*(idx+0.5)/n + phi)
    pu = project_capped_simplex(u, S)
    if pu is not None and is_feasible(pu, n):
        starts.append(pu)
    # Balanced binary
    perm = rng.permutation(n)
    bv = np.zeros(n)
    bv[perm[:n//2]] = 1.0
    if n % 2 == 1:
        bv[perm[n//2]] = 0.5
    starts.append(bv.copy())
    # Random
    rv = rng.random(n)
    pr = project_capped_simplex(rv, S)
    if pr is not None and is_feasible(pr, n):
        starts.append(pr)
    # Reversed parent
    if parent is not None:
        starts.append(parent[::-1].copy())
    # Complement
    if parent is not None:
        starts.append(1.0 - parent.copy())
    # Dedup
    deduped = []
    for s in starts:
        dup = False
        for d in deduped:
            if np.max(np.abs(s - d)) < 1e-10:
                dup = True; break
        if not dup and is_feasible(s, n):
            deduped.append(s)
    return deduped

def anneal_trajectory(h0, n, dx, rng, best_h, best_score):
    S = n/2.0
    h = h0.copy()
    score = exact_score(h, n)
    if score is None:
        score = best_score
    q = exact_lags(h, n)
    if q is None:
        q = np.zeros(2*n-1)
    E = 5 if n <= 128 else (4 if n <= 256 else (3 if n <= 512 else 2))
    K = min(1000, max(80, 80000//n))
    scale = 0.12
    for e in range(E):
        T = max(5e-5, 3e-3*(0.4**e))
        accepted = 0
        for k in range(K):
            p = int(rng.integers(0, n))
            r = int(rng.integers(0, n))
            if p == r:
                continue
            L = max(-h[p], h[r]-1.0)
            U = min(1.0-h[p], h[r])
            if U - L < 1e-12:
                continue
            t = rng.uniform(-scale, scale)
            if t < L or t > U:
                continue
            h_new = h.copy()
            h_new[p] += t
            h_new[r] -= t
            if h_new[p] < -1e-12 or h_new[p] > 1+1e-12 or h_new[r] < -1e-12 or h_new[r] > 1+1e-12:
                h_new = project_capped_simplex(h_new, S)
                if h_new is None:
                    continue
            s_new = exact_score(h_new, n)
            if s_new is None:
                continue
            if s_new < best_score - 1e-12:
                best_score = s_new
                best_h = h_new.copy()
            delta = s_new - score
            if delta <= 0:
                h = h_new
                score = s_new
                q = exact_lags(h, n)
                accepted += 1
            else:
                if delta < 700*T:  # exp(-delta/T) won't underflow
                    if rng.random() < np.exp(-delta/T):
                        h = h_new
                        score = s_new
                        q = exact_lags(h, n)
                        accepted += 1
        ratio = accepted / K
        if ratio < 0.12:
            scale = min(0.4, 1.25*scale)
        elif ratio > 0.55:
            scale = max(0.01, 0.75*scale)
        # One polishing sweep
        n_pol = 4 if n <= 128 else (3 if n <= 256 else (2 if n <= 512 else 1))
        h, score, q = polish_sweep(h, score, q, n, dx, rng, n_pol)
        if score < best_score - 1e-12:
            best_score = score
            best_h = h.copy()
    return h, score, best_h, best_score
