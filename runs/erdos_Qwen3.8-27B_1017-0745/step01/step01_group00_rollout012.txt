def proj_grad_continuation(start, n, tau_schedule, max_iter_per_stage):
    x = start.copy()
    dx = 2.0 / n
    for tau in tau_schedule:
        eta = max(1.0, 0.25 / dx)
        eta = min(eta, 2.0 / dx)
        for _ in range(max_iter_per_stage):
            F, g = smooth_value_gradient(x, tau, n)
            z = project_capped(x - eta * g, n)
            if z is None:
                break
            v = z - x
            if np.max(np.abs(v)) < 1e-9:
                break
            gv = np.dot(g, v)
            if gv >= 0:
                break
            # Armijo
            accepted = False
            eta_try = eta
            for _ in range(20):
                z2 = project_capped(x - eta_try * g, n)
                if z2 is None:
                    break
                F2, _ = smooth_value_gradient(z2, tau, n)
                if F2 <= F + 1e-4 * np.dot(g, z2 - x):
                    x = z2
                    accepted = True
                    break
                eta_try *= 0.5
            if not accepted:
                break
            eta = min(eta * 1.2, 2.0 / dx)
    return x

    add_start(pool, fallback.copy())
    add_start(pool, project_capped(fallback[::-1].copy(), n))
    add_start(pool, project_capped(1.0 - fallback[::-1].copy(), n))
    add_start(pool, np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    add_start(pool, project_capped(blk, n))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    add_start(pool, project_capped(alt, n))
    add_start(pool, project_capped(fallback + rng.normal(0, 0.10, n), n))
    add_start(pool, project_capped(fallback + rng.normal(0, 0.25, n), n))
    add_start(pool, project_capped(fallback + rng.normal(0, 0.40, n), n))
    add_start(pool, project_capped(rng.uniform(0, 1, n), n))
    return pool[:count]

def slsqp_continuation(start, n, tau_schedule, max_iter_per_stage):
    global _cur_n
    _cur_n = n
    x = start.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _mc_fun, 'jac': _mc_jac}]
    for tau in tau_schedule:
        try:
            res = minimize(_obj_slsqp, x, method='SLSQP', jac=True,
                          args=(tau, n), bounds=bounds, constraints=cons,
                          options={'maxiter': max_iter_per_stage, 'ftol': 1e-10, 'disp': False})
        except Exception:
            continue
        xr = None
        if res.x is not None and res.x.shape == (n,) and np.all(np.isfinite(res.x)):
            xr = project_capped(res.x, n)
        if xr is None:
            continue
        best_x, best_s = x, exact_score(x, n)
        for alpha in [1.0, 0.5, 0.25, 0.125]:
            z = project_capped(x + alpha * (xr - x), n)
            if z is not None and is_feasible(z, n):
                sz = exact_score(z, n)
                if sz < best_s:
                    best_s, best_x = sz, z
        x = best_x
    return x

def proj_grad_continuation(start, n, tau_schedule, max_iter_per_stage):
    x = start.copy()
    dx = 2.0 / n
    for tau in tau_schedule:
        eta = max(1.0, 0.25 / dx)
        eta = min(eta, 2.0 / dx)
        for _ in range(max_iter_per_stage):
            F, g = smooth_value_gradient(x, tau, n)
            z = project_capped(x - eta * g, n)
            if z is None:
                break
            v = z - x
            if np.max(np.abs(v)) < 1e-9:
                break
            gv = np.dot(g, v)
            if gv >= 0:
                break
            accepted = False
            eta_try = eta
            for _ in range(20):
                z2 = project_capped(x - eta_try * g, n)
                if z2 is None:
                    break
                F2, _ = smooth_value_gradient(z2, tau, n)
                if F2 <= F + 1e-4 * np.dot(g, z2 - x):
                    x = z2
                    accepted = True
                    break
                eta_try *= 0.5
            if not accepted:
                break
            eta = min(eta * 1.2, 2.0 / dx)
    return x

def run(seed=42, **kwargs):
    global _cur_n
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    fallback = project_capped(parent, n) if parent is not None else None
    if fallback is None:
        fallback = np.full(n, 0.5)
    try:
        rng_seed = int(seed)
    except Exception:
        rng_seed = 42
    rng = np.random.default_rng(rng_seed)

    if n <= 64:
        n_starts, max_iter, n_kicks = 9, 90, 3
    elif n <= 128:
        n_starts, max_iter, n_kicks = 5, 60, 2
    elif n <= 256:
        n_starts, max_iter, n_kicks = 2, 35, 0
    else:
        n_starts, max_iter, n_kicks = 1, 20, 0

    starts = build_starts(fallback, n, rng, n_starts)
    gb = fallback.copy()
    gs = exact_score(fallback, n)

    for st in starts:
        s = exact_score(st, n)
        if s < gs - 1e-12:
            gs, gb = s, st.copy()

    tau_schedule = [0.03, 0.01, 0.003, 0.001, 0.0003]

    for st in starts:
        if n <= 128:
            xo = slsqp_continuation(st, n, tau_schedule, max_iter)
        else:
            xo = proj_grad_continuation(st, n, tau_schedule, max_iter)
        if xo is not None:
            xo = project_capped(xo, n)
            if xo is not None and is_feasible(xo, n):
                so = exact_score(xo, n)
                if so < gs - 1e-12:
                    gs, gb = so, xo.copy()

    if n_kicks > 0 and n <= 128:
        sigma = 0.08
        for _ in range(n_kicks):
            kick = project_capped(gb + sigma * rng.normal(0, 1, n), n)
            if kick is None:
                sigma = min(0.35, 1.7 * sigma)
                continue
            if n <= 128:
                xo = slsqp_continuation(kick, n, tau_schedule, max_iter)
            else:
                xo = proj_grad_continuation(kick, n, tau_schedule, max_iter)
            improved = False
            if xo is not None:
                xo = project_capped(xo, n)
                if xo is not None and is_feasible(xo, n):
                    so = exact_score(xo, n)
                    if so < gs - 1e-12:
                        gs, gb = so, xo.copy()
                        sigma = 0.08
                        improved = True
            if not improved:
                sigma = min(0.35, 1.7 * sigma)

    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    if fs > exact_score(fallback, n):
        final = fallback
        fs = exact_score(fallback, n)
    final = np.clip(final, 0.0, 1.0)
    if not is_feasible(final, n):
        final = np.full(n, 0.5)
        fs = exact_score(final, n)
    return final, float(fs), int(n)
