            qt = float(np.max(q + b * t + a * t * t))
            if qt < best_qt - 1e-14:
                best_qt, best_t = qt, t
        span /= 3.0
    if best_qt > cur + 1e-12:
        return 0.0
    return float(best_t)

def make_pair(h, p, r, t):
    pt = h[p] + h[r]
    pn = np.clip(h[p] + t, max(0.0, pt - 1.0), min(1.0, pt))
    h2 = h.copy()
    h2[p] = pn
    h2[r] = pt - pn
    return h2

def pair_descent(h, N, m, rng, n_sweeps, n_pairs):
    q = lag_q(h, N)
    cs = float(np.max(q) * 2.0 / N)
    tot = N * (N - 1) // 2
    if n_pairs is None or n_pairs > tot:
        n_pairs = tot
    for _ in range(n_sweeps):
        acc = False
        if n_pairs >= tot:
            pairs = [(p, r) for p in range(N - 1) for r in range(p + 1, N)]
        else:
            pairs, seen = [], set()
            for _a in range(50 * n_pairs + 100):
                p = int(rng.integers(0, N - 1))
                r = int(rng.integers(p + 1, N))
                if (p, r) not in seen:
                    pairs.append((p, r))
                    seen.add((p, r))
                    if len(pairs) >= n_pairs:
                        break
            for p in range(N - 1):
                for r in range(p + 1, N):
                    if len(pairs) >= n_pairs:
                        break
                    if (p, r) not in seen:
                        pairs.append((p, r))
                        seen.add((p, r))
        pred = list(range(len(pairs)))
        rng.shuffle(pred)
        for idx in pred:
            p, r = pairs[idx]
            t = pair_line_search(h, q, p, r, N)
            if t is None or abs(t) < 1e-14:
                continue
            h2 = make_pair(h, p, r, t)
            s2 = exact_score(h2, N)
            if np.isfinite(s2) and s2 < cs - 1e-12:
                h, q, cs = h2, lag_q(h2, N), s2
                acc = True
        if not acc:
            break
    return h, cs

def annealing(h, N, m, rng, n_prop):
    if N < 2:
        return h
    best, bs = h.copy(), exact_score(h, N)
    q = lag_q(h, N)
    for j in range(n_prop):
        f = j / max(1, n_prop - 1)
        sigma = 0.12 * (0.025 / 0.12) ** f
        T = 0.002 * (0.0001 / 0.002) ** f
        p = int(rng.integers(0, N))
        r = int(rng.integers(0, N - 1))
        if r >= p:
            r += 1
        t_lo = max(-h[p], h[r] - 1.0)
        t_hi = min(1.0 - h[p], h[r])
        lo, hi = max(t_lo, -sigma), min(t_hi, sigma)
        if hi - lo <= 1e-12:
            continue
        t = float(rng.uniform(lo, hi))
        h2 = make_pair(h, p, r, t)
        if not np.all(np.isfinite(h2)):
            continue
        s2 = exact_score(h2, N)
        if not np.isfinite(s2):
            continue
        delta = s2 - float(np.max(q) * 2.0 / N)
        if delta <= 0 or (T > 1e-15 and rng.random() < np.exp(-min(delta / T, 700.0))):
            h, q = h2, lag_q(h2, N)
            if s2 < bs:
                best, bs = h2.copy(), s2
    return best

def fourier_start(N, rng, nm, amp):
    z = np.full(N, 0.5)
    ns = min(nm, (N - 1) // 2)
    idx = np.arange(N)
    for r in range(1, ns + 1):
        z += (amp / r) * (rng.standard_normal() * np.sin(2 * np.pi * r * idx / N) + rng.standard_normal() * np.cos(2 * np.pi * r * idx / N))
    return z

def run(seed=42, **kwargs):
    N, m = None, None
    raw = globals().get('initial_h_values', None)
    if raw is not None:
        raw = np.asarray(raw, dtype=np.float64)
        if raw.ndim == 1 and raw.size > 1 and np.all(np.isfinite(raw)):
            N, m = raw.size, raw.size / 2.0
    if N is None:
        N = kwargs.get('n_points', 70)
        if isinstance(N, bool) or not isinstance(N, (int, np.integer)) or N < 2:
            N = 70
        m = N / 2.0
    if N == 1:
        hh = np.array([0.5])
        return hh, exact_score(hh, 1), 1

    if raw is not None and raw.ndim == 1 and raw.size == N and np.all(np.isfinite(raw)):
        if raw.min() >= -1e-12 and raw.max() <= 1 + 1e-12 and abs(raw.sum() - m) < 1e-8:
            baseline = np.clip(raw.copy(), 0.0, 1.0)
        else:
            baseline = project_box_sum(raw.copy(), m)
    else:
        baseline = np.full(N, 0.5)
    baseline = project_box_sum(baseline, m)
    bs = exact_score(baseline, N)
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)

    starts = [baseline.copy(), project_box_sum(1.0 - baseline, m)]
    for sc in [0.08, 0.20, 0.35]:
        starts.append(project_box_sum(baseline + sc * rng.standard_normal(N), m))
    starts.append(project_box_sum(fourier_start(N, rng, 5, 0.12), m))
    starts.append(project_box_sum(fourier_start(N, rng, 4, 0.20), m))
    starts.append(project_box_sum(0.5 + 0.5 * rng.uniform(size=N), m))

    uniq = [starts[0]]
    for s in starts[1:]:
        if all(np.max(np.abs(s - u)) > 1e-10 for u in uniq):
            uniq.append(s)
    starts = uniq[:6]

    best, best_s = baseline.copy(), bs
    n_sw = 2 if N <= 96 else 1
    n_pr = N * (N - 1) // 2 if N <= 96 else min(N * (N - 1) // 2, 4 * N)
    n_an = 160 if N <= 96 else 64
    n_ch = 2 if N <= 96 else 1

    for h0 in starts:
        s0 = exact_score(h0, N)
        if np.isfinite(s0) and s0 < best_s - 1e-12:
            best, best_s = h0.copy(), s0
        ho, so = pair_descent(h0, N, m, rng, n_sw, n_pr)
        if np.isfinite(so) and so < best_s - 1e-12:
            best, best_s = ho.copy(), so

    for ch in range(n_ch):
        ha = best.copy() if ch == 0 else project_box_sum(best + 0.15 * rng.standard_normal(N), m)
        hb = annealing(ha, N, m, rng, n_an)
        sa = exact_score(hb, N)
        ho, so = pair_descent(hb, N, m, rng, 1, n_pr)
        if np.isfinite(so) and so < best_s - 1e-12:
            best, best_s = ho.copy(), so
        elif np.isfinite(sa) and sa < best_s - 1e-12:
            best, best_s = hb.copy(), sa
        # second pass on best after first chain
        if ch == 0 and n_ch > 1:
            hb2 = annealing(ho.copy(), N, m, rng, n_an // 2)
            sa2 = exact_score(hb2, N)
            ho2, so2 = pair_descent(hb2, N, m, rng, 1, n_pr)
            if np.isfinite(so2) and so2 < best_s - 1e-12:
                best, best_s = ho2.copy(), so2

    best = project_box_sum(best, m)
    c5 = exact_score(best, N)
    bf = project_box_sum(baseline, m)
    sbf = exact_score(bf, N)
    if np.isfinite(sbf) and sbf <= c5:
        best, c5 = bf, sbf
    if not np.isfinite(c5):
        best = np.full(N, 0.5)
        c5 = exact_score(best, N)
    return best, float(c5), int(N)
