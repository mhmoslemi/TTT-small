    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-14:
        if resid > 0:
            for i in range(n):
                add = min(resid, 1.0 - y[i])
                y[i] += add
                resid -= add
                if abs(resid) < 1e-15:
                    break
        else:
            for i in range(n):
                sub = min(-resid, y[i])
                y[i] -= sub
                resid += sub
                if abs(resid) < 1e-15:
                    break
    if not np.all(np.isfinite(y)):
        return None
    if np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    y = np.clip(y, 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 0:
        if resid > 0:
            y[0] = min(1.0, y[0] + resid)
        else:
            y[0] = max(0.0, y[0] + resid)
    if abs(np.sum(y) - S) > 1e-10:
        return None
    return y

def exact_score(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(c) * dx)
    if not np.isfinite(val):
        return np.inf
    return val

def is_feasible(h, n):
    if h is None or h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10:
        return False
    return True

def smooth_val_grad(h, n, theta):
    # Compute log-sum-exp smooth value and gradient
    Q = np.correlate(h, 1.0 - h, mode='full')
    tau = n * theta
    m = np.max(Q)
    ex = np.exp((Q - m) / tau)
    s = np.sum(ex)
    if s <= 0 or not np.isfinite(s):
        return None, None
    w = ex / s
    F = (m + tau * np.log(s))
    # Gradient via FFT
    conv1 = fftconvolve(w, 1.0 - h, mode='full')
    conv2 = fftconvolve(w[::-1], h, mode='full')
    grad = conv1[n-1:2*n-1] - conv2[n-1:2*n-1]
    if not np.all(np.isfinite(grad)) or not np.all(np.isfinite(Q)):
        return None, None
    return float(F), grad

def local_optimize(h0, n, S):
    # Continuation projected gradient descent
    h = project_box_sum(h0, S)
    if h is None:
        return None
    best = h.copy()
    best_sc = exact_score(h, n)
    
    thetas = [0.05, 0.015, 0.004, 0.001, 0.0002, 0.00005]
    B = max(25, min(100, 4500 // n))
    ckpt = max(1, n // 45)
    
    for theta in thetas:
        alpha = 0.5
        for it in range(B):
            F, grad = smooth_val_grad(h, n, theta)
            if F is None:
                break
            accepted = False
            for bt in range(20):
                p = project_box_sum(h - alpha * grad, S)
                if p is None:
                    alpha *= 0.5
                    continue
                d = p - h
                if np.max(np.abs(d)) / max(alpha, 1e-15) < 1e-7:
                    break
                Fp, _ = smooth_val_grad(p, n, theta)
                if Fp is None:
                    alpha *= 0.5
                    continue
                if Fp <= F + 1e-4 * np.dot(grad, d) + 1e-12 * max(1.0, abs(F)):
                    h = p
                    accepted = True
                    alpha = min(2.0, 1.25 * alpha)
                    break
                else:
                    alpha *= 0.5
            if not accepted:
                break
            if (it + 1) % ckpt == 0 or it == B - 1:
                sc = exact_score(h, n)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best = h.copy()
        # End of stage
        sc = exact_score(h, n)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best = h.copy()
    
    if is_feasible(best, n):
        return best
    return None

def make_harmonic_noise(n, rng):
    f1 = rng.integers(1, n // 2 + 1)
    f2 = rng.integers(1, n // 2 + 1)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    x = np.arange(n)
    v = np.cos(2 * np.pi * f1 * x / n + p1) + np.cos(2 * np.pi * f2 * x / n + p2)
    v -= np.mean(v)
    rms = np.sqrt(np.mean(v**2))
    if rms < 1e-10:
        v = rng.standard_normal(n)
        v -= np.mean(v)
        rms = np.sqrt(np.mean(v**2))
    return v / rms
