import numpy as np
from scipy.optimize import minimize

def project_box_sum(z, m):
    N = len(z)
    if N == 0:
        return z
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    if np.max(np.abs(z)) > 1e12:
        return np.full(N, 0.5)
    z = z - np.median(z)
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    if r > 0:
        for i in range(N):
            a = min(r, 1.0 - h[i])
            if a <= 0:
                continue
            h[i] += a
            r -= a
            if r <= 1e-14:
                break
    elif r < 0:
        for i in range(N):
            a = min(-r, h[i])
            if a <= 0:
                continue
            h[i] -= a
            r += a
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1 + 1e-10:
        return np.full(N, 0.5)
    if abs(h.sum() - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return h

def exact_score(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = float(np.max(corr) * (2.0 / N))
    if not np.isfinite(s):
        return np.inf
    return s

def lag_data(h, N):
    L = 2 * N - 1
    q = np.zeros(L)
    G = np.zeros((L, N))
    for s in range(-(N - 1), N):
        r = s + N - 1
        if s >= 0:
            ell = N - s
            a = h[:ell]
            b = h[s:s + ell]
            q[r] = np.dot(a, 1.0 - b)
            G[r, :ell] += (1.0 - b)
            G[r, s:N] -= a
        else:
            d = -s
            ell = N - d
            a = h[d:]
            b = h[:ell]
            q[r] = np.dot(a, 1.0 - b)
            G[r, d:] += (1.0 - b)
            G[r, :ell] -= a
    return q, G

def mm_qp_obj(z, x, N, m):
    y = z[:N]
    w = z[N]
    d = y - x
    return float(np.dot(d, d) + w)

def mm_qp_obj_jac(z, x, N, m):
    y = z[:N]
    g = np.zeros(N + 1)
    g[:N] = 2.0 * (y - x)
    g[N] = 1.0
    return g

def mm_qp_ineq(z, x, q, G, N, m):
    y = z[:N]
    w = z[N]
    d = y - x
    return w - q - G @ d

def mm_qp_ineq_jac(z, x, q, G, N, m):
    J = np.zeros((2 * N - 1, N + 1))
    J[:, :N] = -G
    J[:, N] = 1.0
    return J

def mm_qp_eq(z, x, N, m):
    return np.array([np.sum(z[:N]) - m])

def mm_qp_eq_jac(z, x, N, m):
    J = np.zeros((1, N + 1))
    J[0, :N] = 1.0
    return J

def mm_local(x, N, m, max_iter):
    h = x.copy()
    best = x.copy()
    best_s = exact_score(best, N)
    cur_s = best_s
    for it in range(max_iter):
        q, G = lag_data(h, N)
        if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
            break
        w0 = max(np.max(q), 0.0)
        z0 = np.concatenate([h, [w0]])
        bounds = [(0.0, 1.0)] * N + [(0.0, 1e6)]
        cons = [
            {'type': 'eq', 'fun': mm_qp_eq, 'jac': mm_qp_eq_jac, 'args': (x, N, m)},
            {'type': 'ineq', 'fun': mm_qp_ineq, 'jac': mm_qp_ineq_jac, 'args': (x, q, G, N, m)},
        ]
        try:
            res = minimize(mm_qp_obj, z0, jac=mm_qp_obj_jac, method='SLSQP',
                           bounds=bounds, constraints=cons, args=(x, N, m),
                           options={'maxiter': 120, 'ftol': 1e-10, 'disp': False})
        except Exception:
            break
        if res.x is None or len(res.x) != N + 1:
            break
        y_new = project_box_sum(res.x[:N], m)
        if y_new is None or not np.all(np.isfinite(y_new)):
            break
        # Line search
        alphas = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125, 0.00390625])
        best_cand = None
        best_cand_s = np.inf
        for a in alphas:
            cand = h + a * (y_new - h)
            cand = project_box_sum(cand, m)
            if not np.all(np.isfinite(cand)):
                continue
            s = exact_score(cand, N)
            if s < best_cand_s:
                best_cand_s = s
                best_cand = cand
        if best_cand is None:
            break
        if best_cand_s < cur_s - 1e-12:
            h = best_cand
            cur_s = best_cand_s
            if cur_s < best_s:
                best_s = cur_s
                best = h.copy()
        else:
            break
        if np.max(np.abs(h - best)) <= 1e-8:
            break
    return best, best_s

import numpy as np
from scipy.optimize import minimize

def project_box_sum(z, m):
    N = len(z)
    if N == 0:
        return z
    if not np.all(np.isfinite(z)) or np.max(np.abs(z)) > 1e12:
        return np.full(N, 0.5)
    z = z - np.median(z)
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    if r > 0:
        for i in range(N):
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
            if r <= 1e-14:
                break
    elif r < 0:
        for i in range(N):
            a = min(-r, h[i])
            h[i] -= a
            r += a
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1 + 1e-10 or abs(h.sum() - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return h

def exact_score(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = float(np.max(corr) * (2.0 / N))
    return s if np.isfinite(s) else np.inf

def lag_data(h, N):
    L = 2 * N - 1
    q = np.zeros(L)
    G = np.zeros((L, N))
    for s in range(-(N - 1), N):
        r = s + N - 1
        if s >= 0:
            ell = N - s
            a = h[:ell]
            b = h[s:s + ell]
            q[r] = np.dot(a, 1.0 - b)
            G[r, :ell] += (1.0 - b)
            G[r, s:] -= a
        else:
            d = -s
            ell = N - d
            a = h[d:]
            b = h[:ell]
            q[r] = np.dot(a, 1.0 - b)
            G[r, d:] += (1.0 - b)
            G[r, :ell] -= a
    return q, G

def mm_qp_obj(z, ctr, N, m):
    d = z[:N] - ctr
    return float(np.dot(d, d) + z[N])

def mm_qp_obj_jac(z, ctr, N, m):
    g = np.zeros(N + 1)
    g[:N] = 2.0 * (z[:N] - ctr)
    g[N] = 1.0
    return g

def mm_qp_ineq(z, ctr, q, G, N, m):
    d = z[:N] - ctr
    return z[N] - q - G @ d

def mm_qp_ineq_jac(z, ctr, q, G, N, m):
    J = np.zeros((2 * N - 1, N + 1))
    J[:, :N] = -G
    J[:, N] = 1.0
    return J

def mm_qp_eq(z, ctr, N, m):
    return np.array([np.sum(z[:N]) - m])

def mm_qp_eq_jac(z, ctr, N, m):
    J = np.zeros((1, N + 1))
    J[0, :N] = 1.0
    return J

def mm_local(h0, N, m, max_iter):

    best = h0.copy()
    best_s = exact_score(best, N)
    cur_s = best_s
    alphas = np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125, 0.00390625])
    for _ in range(max_iter):
        q, G = lag_data(h, N)
        if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
            break
        w0 = max(float(np.max(q)), 0.0)
        z0 = np.concatenate([h, [w0]])
        bounds = [(0.0, 1.0)] * N + [(0.0, 1e6)]
        cons = [

            {'type': 'ineq', 'fun': mm_qp_ineq, 'jac': mm_qp_ineq_jac, 'args': (h, q, G, N, m)},

            res = minimize(mm_qp_obj, z0, jac=mm_qp_obj_jac, method='SLSQP',
                           bounds=bounds, constraints=cons, args=(h, N, m),
                           options={'maxiter': 120, 'ftol': 1e-10, 'disp': False})

            break

            break

        if not np.all(np.isfinite(y_new)):

        for a in alphas:

            best_s, best = cur_s, h.copy()
        if np.max(np.abs(h - best)) <= 1e-8:
            break
    return best, best_s

def make_fourier(N, rng, R, A):
    idx = np.arange(N)
    z = np.full(N, 0.5)
    Rs = min(R, (N - 1) // 2)
    for r in range(1, Rs + 1):
        cr = rng.standard_normal()
        sr = rng.standard_normal()
        z += (A / r) * (cr * np.cos(2.0 * np.pi * r * idx / N) + sr * np.sin(2.0 * np.pi * r * idx / N))
    return z

def build_starts(base, N, m, rng):
    starts = [base.copy()]
    comp = project_box_sum(1.0 - base, m)
    starts.append(comp)
    starts.append(base[::-1].copy())
    sh = max(1, N // 3)
    starts.append(np.roll(base, sh).copy())
    for sc in [0.05, 0.15]:
        starts.append(project_box_sum(base + sc * rng.standard_normal(N), m))
    starts.append(project_box_sum(rng.uniform(size=N), m))
    starts.append(project_box_sum(make_fourier(N, rng, 6, 0.12), m))
    starts.append(project_box_sum(make_fourier(N, rng, 6, 0.24), m))
    dedup = []
    for s in starts:
        if not np.all(np.isfinite(s)):
            continue
        dup = False
        for d in dedup:
            if np.max(np.abs(s - d)) <= 1e-10:
                dup = True
                break
        if not dup:
            dedup.append(s)
        if len(dedup) >= 8:
            break
    return dedup

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    N = None
    raw = globals().get('initial_h_values', None)
    if raw is not None:
        try:
            raw = np.asarray(raw, dtype=np.float64)
            if raw.ndim == 1 and raw.size > 1 and np.all(np.isfinite(raw)):
                N = raw.size
        except Exception:
            N = None
    if N is None:
        N = kwargs.get('n_points', 70)
        if not isinstance(N, (int, np.integer)) or isinstance(N, (bool, np.bool_)) or N < 2 or N > 999:
            N = 70
        N = int(N)
    m = N / 2.0
    if N == 1:
        h = np.array([0.5])
        return h, exact_score(h, 1), 1
    if raw is not None:
        try:
            raw = np.asarray(raw, dtype=np.float64)
            if raw.ndim == 1 and raw.size == N and np.all(np.isfinite(raw)):
                base = project_box_sum(raw, m)
            else:
                base = np.full(N, 0.5)
        except Exception:
            base = np.full(N, 0.5)
    else:
        base = np.full(N, 0.5)
    base_s = exact_score(base, N)
    best, best_s = base.copy(), base_s

    max_iter_main = 40
    max_iter_restart = 24
    max_starts = 8
    if N > 160:
        max_iter_main, max_iter_restart, max_starts = 12, 8, 3
    elif N > 96:
        max_iter_main, max_iter_restart, max_starts = 25, 16, 5

    starts = build_starts(base, N, m, rng)[:max_starts]
    for h0 in starts:
        s0 = exact_score(h0, N)
        if s0 < best_s - 1e-12:
            best, best_s = h0.copy(), s0
        try:
            hc, sc = mm_local(h0, N, m, max_iter_main)
            if sc < best_s - 1e-12:
                best, best_s = hc.copy(), sc
        except Exception:
            pass

    for sigma in [0.05, 0.015]:
        try:
            hp = project_box_sum(best + sigma * rng.standard_normal(N), m)
            sp = exact_score(hp, N)
            if sp < best_s - 1e-12:
                best, best_s = hp.copy(), sp
            hc, sc = mm_local(hp, N, m, max_iter_restart)
            if sc < best_s - 1e-12:
                best, best_s = hc.copy(), sc
        except Exception:
            pass

    best = project_box_sum(best, m)
    c5 = exact_score(best, N)
    base = project_box_sum(base, m)
    base_s2 = exact_score(base, N)
    if not np.isfinite(c5) or (np.isfinite(base_s2) and base_s2 < c5):
        best, c5 = base, base_s2
    if not np.isfinite(c5):
        best = np.full(N, 0.5)
        c5 = exact_score(best, N)
    return best, float(c5), int(N)
