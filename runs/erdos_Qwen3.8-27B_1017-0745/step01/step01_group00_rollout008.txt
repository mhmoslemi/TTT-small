from scipy.optimize import minimize

_S = {}

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo, hi = z.min() - 1.0, z.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or h.min() < -1e-14 or h.max() > 1.0 + 1e-14:
        return None
    return h

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and h.min() >= -1e-12
            and h.max() <= 1.0 + 1e-12 and abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n))

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def lag_model(x, n):
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.zeros(L)
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        p0, q0 = max(d, 0), max(-d, 0)
        m = n - abs(d)
        q[row] = dx * np.dot(x[p0:p0 + m], 1.0 - x[q0:q0 + m])
        G[row, p0:p0 + m] += dx * (1.0 - x[q0:q0 + m])
        G[row, q0:q0 + m] -= dx * x[p0:p0 + m]
    return q, G

def _s_obj(z):
    y, t = z[:-1], z[-1]
    return t + _S['bd'] * np.dot(y - _S['x'], y - _S['x'])

def _s_obj_grad(z):
    y = z[:-1]
    n = len(y)
    g = np.empty(n + 1)
    g[:n] = 2.0 * _S['bd'] * (y - _S['x'])
    g[n] = 1.0
    return g

def _s_ineq(z):
    return z[-1] - _S['q'] - _S['G'].dot(z[:-1] - _S['x'])

def _s_ineq_jac(z):
    G = _S['G']
    n = G.shape[1]
    J = np.empty((n * 2 - 1, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def _s_eq(z):
    return np.sum(z[:-1]) - _S['mass']

def _s_eq_jac(z):
    n = _S['mass']
    J = np.zeros(int(n * 2) + 1)
    J[:int(n * 2)] = 1.0
    return J

def slsqp_step(x, n, beta, maxiter):
    dx = 2.0 / n
    q, G = lag_model(x, n)
    if G is None or not np.all(np.isfinite(G)):
        return None
    _S['x'], _S['q'], _S['G'] = x, q, G
    _S['bd'] = beta * dx
    _S['mass'] = n / 2.0
    z0 = np.empty(n + 1)
    z0[:n] = x
    z0[n] = q.max() + 1e-6
    cons = [{'type': 'ineq', 'fun': _s_ineq, 'jac': _s_ineq_jac},
            {'type': 'eq', 'fun': _s_eq, 'jac': _s_eq_jac}]
    bnds = [(0.0, 1.0)] * n + [(None, None)]
    try:
        res = minimize(_s_obj, z0, jac=_s_obj_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
    except Exception:
        return None
    if res.x is None or res.x.shape != (n + 1,) or not np.all(np.isfinite(res.x)):
        return None
    return res.x[:n]

def subgrad_step(x, n, beta, niters):
    dx = 2.0 / n
    y = x.copy()
    best_y, best_val = y.copy(), float('inf')
    for k in range(niters):
        Amax, Ad = float('-inf'), 0
        for d in range(-(n - 1), n):
            p0, q0, m = max(d, 0), max(-d, 0), n - abs(d)
            Ad = dx * (np.dot(x[p0:p0 + m], 1.0 - x[q0:q0 + m])
                       + np.dot(1.0 - x[q0:q0 + m], y[p0:p0 + m] - x[p0:p0 + m])
                       - np.dot(x[p0:p0 + m], y[q0:q0 + m] - x[q0:q0 + m]))
            if Ad > Amax:
                Amax, best_d = Ad, d
        u = y - x
        model = Amax + beta * dx * np.dot(u, u)
        if model < best_val:
            best_val, best_y = model, y.copy()
        p0, q0, m = max(best_d, 0), max(-best_d, 0), n - abs(best_d)
        g = np.zeros(n)
        g[p0:p0 + m] += dx * (1.0 - x[q0:q0 + m])
        g[q0:q0 + m] -= dx * x[p0:p0 + m]
        v = g + 2.0 * beta * dx * u
        nv = np.linalg.norm(v)
        eta = min(0.5 / (nv + 1e-12), 1.0 / (2.0 * beta * dx * np.sqrt(k + 1) + 1e-12))
        yn = project_capped(y - eta * v, n)
        if yn is None or not np.all(np.isfinite(yn)):
            break
        if np.max(np.abs(yn - y)) < 1e-9:
            break
        y = yn
    return best_y

def optimize_start(x, n, n_outer, siter):
    x = x.copy().astype(np.float64)
    score = exact_score(x, n)
    if not np.isfinite(score):
        return x, score
    beta, rejs = 1.0, 0
    for _ in range(n_outer):
        if rejs >= 5:
            break
        if n <= 128:
            prop = slsqp_step(x, n, beta, siter)
        else:
            prop = subgrad_step(x, n, beta, siter)
        if prop is None:
            beta = min(2.0 * beta, 64.0)
            rejs += 1
            continue
        prop = project_capped(prop, n)
        if prop is None or not is_feasible(prop, n):
            beta = min(2.0 * beta, 64.0)
            rejs += 1
            continue
        if np.max(np.abs(prop - x)) < 1e-9:
            break
        best_z, best_s = None, score
        for alpha in (1.0, 0.5, 0.25, 0.125, 0.0625):
            z = project_capped(x + alpha * (prop - x), n)
            if z is None or not is_feasible(z, n):
                continue
            sz = exact_score(z, n)
            if sz < best_s - 1e-12:
                best_s, best_z = sz, z
        if best_z is None:
            beta = min(2.0 * beta, 64.0)
            rejs += 1
            continue
        x, score = best_z, best_s
        rejs = 0
        beta = max(1.0, beta / 1.25)
    return x, exact_score(x, n)

def build_starts(fb, n, rng, cnt):
    pool = [fb.copy()]
    cands = [fb[::-1].copy(), 1.0 - fb[::-1].copy(), np.full(n, 0.5)]
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    cands.append(blk)
    cands.append(np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64))
    for sd in (0.08, 0.20):
        cands.append(fb + rng.normal(0, sd, n))
    cands.append(rng.uniform(0, 1, n))
    for c in cands:
        p = project_capped(c, n)
        if p is None or not is_feasible(p, n):
            continue
        if any(np.max(np.abs(p - q)) < 1e-8 for q in pool):
            continue
        pool.append(p)
    pool.sort(key=lambda v: exact_score(v, n))
    if fb not in pool:
        pool.insert(0, fb)
    return pool[:cnt]

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and p.min() >= -1e-8 and p.max() <= 1.0 + 1e-8:
                parent = p
    except Exception:
        pass
    if n is None:
        n = 47
        try:
            v = int(kwargs.get('n_points', 47))
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    fb = project_capped(parent, n) if parent is not None else None
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
        fb = project_capped(fb, n)
        if fb is None:
            fb = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 1:
        return fb, exact_score(fb, n), n
    if n <= 64:
        n_st, n_out, s_iter = 8, 10, 80
    elif n <= 128:
        n_st, n_out, s_iter = 5, 7, 60
    elif n <= 256:
        n_st, n_out, s_iter = 3, 5, 60
    elif n <= 512:
        n_st, n_out, s_iter = 2, 3, 40
    else:
        n_st, n_out, s_iter = 2, 2, 25
    starts = build_starts(fb, n, rng, n_st)
    gb, gs = fb.copy(), exact_score(fb, n)
    for st in starts:
        s = exact_score(st, n)
        if s < gs - 1e-12:
            gs, gb = s, st.copy()
    for st in starts:
        try:
            xo, so = optimize_start(st, n, n_out, s_iter)
            if so < gs - 1e-12:
                gs, gb = float(so), xo.copy()
        except Exception:
            pass
    gb = project_capped(gb, n)
    fb_score = exact_score(fb, n)
    if gb is None or not is_feasible(gb, n) or exact_score(gb, n) > fb_score + 1e-12:
        gb = fb
    fs = exact_score(gb, n)
    if not np.isfinite(fs):
        gb = np.full(n, 0.5)
        fs = exact_score(gb, n)
    return gb, float(fs), int(n)
