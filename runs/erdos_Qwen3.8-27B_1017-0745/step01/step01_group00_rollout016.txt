2. Find minimizer of H on [L,U] (golden section)
3. If H(tH) >= 0, G dominates everywhere, done
4. Else find roots of H, split into intervals, minimize G on outer intervals

def project_capped(z, n):
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)

            break

def is_feasible(h, n):

def exact_score(h, n):

def lag_grad(h, d, k, dx, n):

    a = max(d, 0)
    b = max(-d, 0)
    m = n - abs(d)
    g = 0.0
    if a <= k < a + m:
        r = k - a
        g += dx * (1.0 - h[b + r])
    if b <= k < b + m:
        r = k - b
        g -= dx * h[a + r]
    return g

def golden_min(f, lo, hi, iters=52):
    gr = (5.0 - sqrt(5.0)) / 2.0  # golden ratio conjugate
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc, fd = f(c), f(d)
    for _ in range(iters):
        if fc < fd:
            hi, d, fd = d, c, fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo, c, fc = c, d, fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    x = 0.5 * (lo + hi)
    fm = f(x)
    best = x
    for cand in (lo, hi, x):
        if f(cand) < fm:
            fm = f(cand)
            best = cand
    return best

def bisection_root(f, lo, hi, iters=52):
    flo, fhi = f(lo), f(hi)
    if flo * fhi > 0:
        return None
    for _ in range(iters):
        mid = 0.5 * (lo + hi)
        fm = f(mid)
        if flo * fm <= 0:
            hi, fhi = mid, fm
        else:
            lo, flo = mid, fm
    return 0.5 * (lo + hi)

def pair_line_min(h, i, j, n):
    dx = 2.0 / n
    L = max(-h[i], h[j] - 1.0)
    U = min(1.0 - h[i], h[j])
    if U - L < 1e-14:
        return None
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    r = abs(i - j)
    # A coefficients: A[0] for zero lag
    # zero lag index in q: n-1
    # For nonzero lags at offset ±r, A = dx; else A = 0
    # B_d = grad_d[i] - grad_d[j]
    B = np.zeros(2 * n - 1)
    for d in range(-(n - 1), n):
        B[d + n - 1] = lag_grad(h, d, i, dx, n) - lag_grad(h, d, j, dx, n)
    # D = q (current values)
    # P_d(t) = A_d*t^2 + B_d*t + D_d
    # A: index n-1 -> -2*dx, index n-1±r -> dx (if in range)
    A = np.zeros(2 * n - 1)
    A[n - 1] = -2.0 * dx
    if r > 0 and n - 1 + r < 2 * n - 1:
        A[n - 1 + r] = dx
    if r > 0 and n - 1 - r >= 0:
        A[n - 1 - r] = dx
    zero_idx = n - 1
    
    def eval_P(idx, t):
        return A[idx] * t * t + B[idx] * t + q[idx]
    
    def G(t):
        # max over nonzero lags
        vals = np.array([A[idx] * t * t + B[idx] * t + q[idx] 
                        for idx in range(2 * n - 1) if idx != zero_idx])
        return float(np.max(vals))
    
    def H(t):
        return G(t) - eval_P(zero_idx, t)
    
    candidates = [L, U, 0.0, golden_min(G, L, U), golden_min(H, L, U)]
    tH = golden_min(H, L, U)
    if H(tH) < 0.0:
        for (blo, bhi) in [(L, tH), (tH, U)]:
            if H(blo) * H(bhi) < 0:
                root = bisection_root(H, blo, bhi)
                if root is not None:
                    candidates.append(root)
        # On outer intervals where H>=0, minimize G
        # Check which parts of [L,U] have H>=0
        if H(L) > 0:
            # Left part [L, left_root] has H>0
            left_root = bisection_root(H, L, tH) if H(tH) < 0 < H(L) else None
            if left_root is not None:
                candidates.append(golden_min(G, L, left_root))
        if H(U) > 0:
            right_root = bisection_root(H, tH, U) if H(tH) < 0 < H(U) else None
            if right_root is not None:
                candidates.append(golden_min(G, right_root, U))
    
    best_t, best_val = None, float('inf')
    for t in candidates:
        if t is None or not np.isfinite(t) or t < L - 1e-14 or t > U + 1e-14:
            continue
        t = clip_t = max(L, min(U, t))
        # Evaluate exact max
        val = max(eval_P(idx, t) for idx in range(2 * n - 1))
        if val < best_val - 1e-15 or (abs(val - best_val) < 1e-15 and abs(t) < abs(best_t or 1e9)):
            best_val = val
            best_t = t
    if best_t is None:
        return None
    return best_t
