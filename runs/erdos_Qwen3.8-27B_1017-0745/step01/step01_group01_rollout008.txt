            if np.isfinite(ts) and ts < x_score - 1e-12:
                x, x_score = trial, ts
                improved = True
                if x_score < best_score - 1e-12:
                    best_h, best_score = x.copy(), x_score
        if not improved:
            break
    return x, best_h, best_score

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 0 < len(iv) <= 999:
            n = len(iv)
            if np.all(np.isfinite(iv)) and np.all(iv >= -1e-8) and np.all(iv <= 1.0 + 1e-8):
                parent = np.clip(iv.copy(), 0.0, 1.0)
    except Exception:
        pass
    if n is None:
        try:
            n = max(1, min(999, int(kwargs.get('n_points', 47))))
        except Exception:
            n = 47
    uniform = np.full(n, 0.5)
    if parent is not None and abs(float(np.sum(parent)) - n / 2.0) <= 2e-12 * max(1, n):
        h0 = parent.copy()
    elif parent is not None:
        h0 = project_capped(parent, n)
        h0 = h0 if h0 is not None else uniform.copy()
    else:
        h0 = uniform.copy()
    fallback = h0.copy()
    fb_score = exact_score(fallback, n)
    if not np.isfinite(fb_score):
        fb_score = exact_score(uniform, n)
        fallback = uniform.copy()
    try:
        sd = int(seed) % (2 ** 32)
    except Exception:
        sd = 42
    rng = np.random.default_rng(sd)
    starts = []
    add_start(starts, uniform, n)
    if parent is not None:
        add_start(starts, parent, n)
        add_start(starts, parent[::-1], n)
        add_start(starts, 1.0 - parent, n)
        add_start(starts, 1.0 - parent[::-1], n)
        add_start(starts, parent + rng.normal(0, 0.03, n), n)
        add_start(starts, parent + rng.normal(0, 0.08, n), n)
        add_start(starts, parent + rng.normal(0, 0.15, n), n)
        t = np.arange(n, dtype=np.float64)
        add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * t / n), n)
        add_start(starts, 0.5 + 0.35 * np.cos(4.0 * np.pi * t / n), n)
    else:
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        add_start(starts, blk, n)
        alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
        add_start(starts, alt, n)
        t = np.arange(n, dtype=np.float64)
        for q in (1, 2, 3):
            add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * q * t / n), n)
        for _ in range(3):
            add_start(starts, rng.uniform(0, 1, n), n)
    if n <= 64:
        ns_def, lv_def, it_def = 8, 8, 80
    elif n <= 128:
        ns_def, lv_def, it_def = 6, 7, 50
    elif n <= 256:
        ns_def, lv_def, it_def = 4, 6, 20
    else:
        ns_def, lv_def, it_def = 3, 0, 0
    try:
        n_starts = max(1, min(int(kwargs.get('starts', ns_def)), 12))
    except Exception:
        n_starts = ns_def
    try:
        levels = max(2, min(int(kwargs.get('levels', lv_def)), 10)) if lv_def > 0 else 0
    except Exception:
        levels = lv_def
    try:
        iters = max(0, min(int(kwargs.get('iters_per_stage', it_def)), 150))
    except Exception:
        iters = it_def
    starts = starts[:n_starts]
    best_h, best_score = fallback.copy(), fb_score
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_score - 1e-12:
            best_h, best_score = st.copy(), ss
    if n <= 256 and iters > 0 and levels > 0:
        tau_hi, tau_lo = 0.03, 0.0001
        for si, st in enumerate(starts):
            x_cur, _, best_h, best_score = optimize_start(
                st.copy(), n, levels, iters, tau_hi, tau_lo, rng, best_h, best_score)
            x_cur, best_h, best_score = polish(x_cur, n, best_h, best_score, rng, 3)
    final = project_capped(best_h, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > best_score + 1e-12:
        final, fs = best_h.copy(), best_score
    fs_fb = exact_score(fallback, n)
    if np.isfinite(fs_fb) and fs_fb < fs:
        final, fs = fallback.copy(), fs_fb
    if not np.isfinite(fs):
        final, fs = uniform.copy(), exact_score(uniform, n)
    return final, float(fs), int(n)
