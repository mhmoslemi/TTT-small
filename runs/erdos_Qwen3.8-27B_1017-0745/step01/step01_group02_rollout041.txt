import numpy as np
import math

def _conv(a, b):
    return np.convolve(a, b, mode='full')

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.ndim != 1 or len(v) == 0 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    S = float(S)
    if S < -1e-12 or S > n + 1e-12:
        return None
    S = max(0.0, min(float(n), S))
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(v - tau, 0.0, 1.0)
    r = S - float(np.sum(y))
    tol = 1e-13 * max(1, n)
    if r > tol:
        for i in np.argsort(1.0 - y)[::-1]:
            if r <= tol:
                break
            y[i] += min(r, 1.0 - y[i])
            r -= min(r, 1.0 - y[i] if r > 1.0 - y[i] else 1.0 - y[i])
    elif r < -tol:
        for i in np.argsort(y)[::-1]:
            if r >= -tol:
                break
            y[i] -= min(-r, y[i])
            r += min(-r, y[i] if -r > y[i] else y[i])
    if not np.all(np.isfinite(y)):
        return None
    if abs(float(np.sum(y)) - S) > 1e-9 * max(1, n):
        return None
    return np.clip(y, 0.0, 1.0)

def check_feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(float(np.sum(h)) - 0.5 * n) < 1e-10 * max(1, n))

def exact_score(h):
    n = len(h)
    if not check_feasible(h, n):
        return float('inf')
    val = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
    return val if np.isfinite(val) else float('inf')

def smooth_val_grad(h, n, tau):
    if tau <= 0 or not np.all(np.isfinite(h)):
        return None, None
    bh = 1.0 - h
    r = _conv(h[::-1].copy(), bh)
    if len(r) != 2 * n - 1:
        return None, None
    q = (2.0 / n) * r
    qmax = float(np.max(q))
    ex = np.exp((q - qmax) / tau)
    Z = float(np.sum(ex))
    if not np.isfinite(Z) or Z <= 0:
        return None, None
    p = ex / Z
    F = qmax + tau * math.log(Z)
    c1 = _conv(bh, p[::-1].copy())
    c2 = _conv(h.copy(), p)
    T1 = c1[n - 1:2 * n - 1]
    T2 = c2[n - 1:2 * n - 1]
    g = (2.0 / n) * (T1 - T2)
    if not np.all(np.isfinite(g)):
        return None, None
    return F, g

def proj_stage(h, n, tau, max_iter):
    h_cur = h.copy()
    F, g = smooth_val_grad(h_cur, n, tau)
    if F is None or not np.isfinite(F):
        s = exact_score(h_cur)
        return h_cur, h_cur, s
    best_h = h_cur.copy()
    best_s = exact_score(h_cur)
    alpha = 1.0
    stag = 0
    for it in range(max_iter):
        if float(np.max(np.abs(g))) < 1e-14:
            break
        accepted = False
        for bt in range(25):
            trial = project_box_sum(h_cur - alpha * g, 0.5 * n)
            if trial is None:
                alpha *= 0.5
                continue
            d = trial - h_cur
            if float(np.max(np.abs(d))) < 1e-11:
                break
            gd = float(np.dot(g, d))
            if gd >= -1e-14:
                break
            F_t, _ = smooth_val_grad(trial, n, tau)
            if F_t is None or not np.isfinite(F_t):
                alpha *= 0.5
                continue
            if F_t <= F + 1e-4 * gd + 1e-12:
                F_old = F
                g_old = g.copy()
                h_cur = trial
                F = F_t
                g, _ = smooth_val_grad(h_cur, n, tau)
                if g is None:
                    g = np.zeros(n)
                dy = float(np.dot(d, g - g_old))
                if dy > 1e-14:
                    alpha = max(1e-3, min(10.0, float(np.dot(d, d)) / dy))
                else:
                    alpha = min(10.0, 1.5 * alpha)
                if (F_old - F) < 1e-12:
                    stag += 1
                else:
                    stag = 0
                accepted = True
                break
            alpha *= 0.5
        if not accepted:
            break
        if stag >= 8:
            break
        if it % 10 == 0:
            s = exact_score(h_cur)
            if s < best_s:
                best_s = s
                best_h = h_cur.copy()
    s = exact_score(h_cur)
    if s < best_s:
        best_s = s
        best_h = h_cur.copy()
    return h_cur, best_h, best_s

def run(seed=42, **kwargs):
    try:

    except Exception:
        rng = np.random.default_rng(42)
    parent = None
    n = 45

    if parent is None:

    if parent is None:
        parent = np.full(n, 0.5)
    S = 0.5 * n
    best_h = parent.copy()
    best_c = exact_score(parent)
    scale = math.log(89.0) / math.log(2.0 * n - 1)
    taus = [0.04 * (0.3 ** j) * scale for j in range(7)]
    M = max(30, min(120, 5400 // n))

                break
        if not dup:

    starts = unique_starts

        c = exact_score(s)

            best_c = c
            best_h = s.copy()
    for s in starts:
        h_cur = s.copy()

        if h0 is None:
            continue
        h_cur = h0
        for tau in taus:
            h_cur, bh, bs = proj_stage(h_cur, n, tau, M_bh)
            if bs < best_c:
                best_c = bs
                best_h = bh.copy()
    final_h = project_box_sum(best_h, S)
    if final_h is not None:
        best_h = final_h
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c = exact_score(best_h)
    fb_c = exact_score(parent)
    if not np.isfinite(best_c) or best_c > fb_c:
