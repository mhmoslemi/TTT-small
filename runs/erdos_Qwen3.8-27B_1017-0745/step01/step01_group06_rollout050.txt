                           method='SLSQP', bounds=bounds, constraints=con,
                           options={'maxiter': maxslq, 'ftol': 1e-10, 'disp': False})
        except Exception:
            res = None
        if res is None or not np.all(np.isfinite(res.x)):
            delta *= 0.5; cfail += 1
            if cfail >= 4 or delta < 1e-5:
                break
            continue
        dsol = res.x[:-1]
        if np.any(dsol < dlo - 1e-5) or np.any(dsol > dhi + 1e-5) or abs(dsol.sum()) > 1e-4:
            delta *= 0.5; cfail += 1
            if cfail >= 4 or delta < 1e-5:
                break
            continue
        cfail = 0
        ht = project_K(h + dsol, n, S)
        if not feasible(ht, n, S):
            delta *= 0.5; cfail += 1
            if cfail >= 4 or delta < 1e-5:
                break
            continue
        d_act = ht - h
        if max(np.abs(d_act)) < 1e-8:
            break
        pred = max(q) - max(q + G @ d_act)
        act = best_e - exact_score(ht, dx)
        accept = False
        if act > 1e-12:
            if pred <= 1e-12:
                accept = True
            elif act / max(pred, 1e-15) >= 0.1:
                accept = True
        if accept:
            h, best_e = ht, max(lag_vec(ht, dx))
            crej = 0
            q = lag_vec(h, dx)
            if pred > 1e-12:
                ratio = act / max(pred, 1e-15)
                if ratio >= 0.75 and max(np.abs(d_act)) >= 0.8 * delta:
                    delta = min(0.5, 2.0 * delta)
                    eta = min(2.0, 1.5 * eta)
                elif ratio < 0.25:
                    delta *= 0.5
        else:
            crej += 1
            delta *= 0.5
            eta = max(1e-4, eta * 0.5)
            if delta < 1e-5 or crej >= 8:
                break
    return h

def transfer_pv(parent, n0, m):
    S = m / 2.0
    y = np.zeros(m)
    for i in range(n0):
        a0, a1 = i / n0, (i + 1) / n0
        for j in range(m):
            ov = max(0.0, min(a1, (j + 1) / m) - max(a0, j / m))
            if ov > 0:
                y[j] += parent[i] * ov
    return project_K(y * m, m, S)

def make_starts(base, n, S, rng, is_parent):
    St = [base.copy(), (1.0 - base).copy(), base[::-1].copy(),
          project_K(base + 0.08 * rng.standard_normal(n), n, S),
          project_K(base + 0.25 * rng.standard_normal(n), n, S),
          project_K(rng.uniform(0, 1, n), n, S),
          project_K(rng.beta(0.5, 0.5, n), n, S)]
    hb = np.zeros(n)
    k = int(np.floor(S))
    hb[:k] = 1.0
    if k < n and S > k:
        hb[k] = S - k
    St.append(project_K(hb, n, S))
    js = np.arange(n)
    St.append(project_K(0.5 + 0.45 * np.cos(2.0 * np.pi * js / n), n, S))
    if is_parent:
        St.append(project_K(base[rng.permutation(n)], n, S))
    dd, seen = [], []
    for s in St:
        if not feasible(s, n, S):
            continue
        dup = any(np.max(np.abs(s - sv)) <= 1e-6 for sv in seen)
        if not dup:
            dd.append(s)
            seen.append(s)
    return dd

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    init = globals().get('initial_h_values', None)
    n0, pv = 45, False
    if init is not None:
        init = np.array(init, dtype=np.float64).ravel()
        if (2 <= len(init) <= 999 and np.all(np.isfinite(init)) and
                min(init) >= -1e-6 and max(init) <= 1.0 + 1e-6 and
                abs(init.sum() - len(init) / 2.0) <= 1e-6):
            n0, pv = len(init), True
    if not pv:
        nd = kwargs.get('n_points', 45)
        if isinstance(nd, (int, np.integer)) and not isinstance(nd, bool) and 2 <= nd <= 999:
            n0 = int(nd)
        init = np.full(n0, 0.5)
    ders = sorted(set([n0, max(2, n0 - 1), min(999, n0 + 1)])) if pv else [n0]
    dx0, S0 = 2.0 / n0, n0 / 2.0
    fb_h = project_K(init.copy(), n0, S0) if pv else np.full(n0, 0.5)
    fb_s = exact_score(fb_h, dx0)
    best = (fb_h.copy(), n0, fb_s)
    for m in ders:
        dxm, Sm = 2.0 / m, m / 2.0
        if pv and m != n0:
            base_m = transfer_pv(init, n0, m)
        elif m == n0:
            base_m = project_K(init.copy(), m, Sm)
        else:
            base_m = np.full(m, 0.5)
        if m <= 96:
            ns, nr, mi, ms = 8, 1, 28, 50
        elif m <= 256:
            ns, nr, mi, ms = 3, 0, 12, 30
        else:
            ns, nr, mi, ms = 1, 0, 5, 15
        starts = make_starts(base_m, m, Sm, rng, pv and m == n0)[:ns]
        if not starts:
            starts = [base_m]
        rb_h, rb_s = base_m.copy(), exact_score(base_m, dxm)
        for s in starts:
            sc = exact_score(s, dxm)
            if sc < rb_s:
                rb_s, rb_h = sc, s.copy()
            if sc < best[2] - 1e-12:
                best = (s.copy(), m, sc)
        for s in starts:
            ho = local_solve(s, m, dxm, Sm, rng, mi, ms)
            ho = project_K(ho, m, Sm)
            if feasible(ho, m, Sm):
                sc = exact_score(ho, dxm)
                if sc < rb_s:
                    rb_s, rb_h = sc, ho.copy()
                if sc < best[2] - 1e-12:
                    best = (ho.copy(), m, sc)
        for _ in range(nr):
            hr = project_K(rb_h + 0.18 * rng.standard_normal(m), m, Sm)
            ho2 = local_solve(hr, m, dxm, Sm, rng, mi, ms)
            ho2 = project_K(ho2, m, Sm)
            if feasible(ho2, m, Sm):
                sc2 = exact_score(ho2, dxm)
                if sc2 < rb_s:
                    rb_s, rb_h = sc2, ho2.copy()
                if sc2 < best[2] - 1e-12:
                    best = (ho2.copy(), m, sc2)
    bh, bn, bs = best
    bh = project_K(bh, bn, bn / 2.0)
    if not feasible(bh, bn, bn / 2.0):
        bh, bn, bs = fb_h.copy(), n0, fb_s
    else:
        fs = exact_score(bh, 2.0 / bn)
        if fs > best[2] + 1e-10:
            bh, bn, bs = fb_h.copy(), n0, fb_s
        else:
            bs = fs
    bh = np.clip(project_K(bh, bn, bn / 2.0), 0.0, 1.0)
    return (bh, exact_score(bh, 2.0 / bn), int(bn))
