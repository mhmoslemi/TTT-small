```python
import numpy as np
import math

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    r = m - h.sum()
    if r > 0:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
            if r <= 1e-14:
                break
    elif r < 0:
        for i in range(N):
            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
            if r >= -1e-14:
                break
    if h.min() < -1e-10 or h.max() > 1.0+1e-10:
        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)

def exact_score(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / N)

def lag_vector(h, N):
    # Returns q_s in order s = -(N-1), ..., N-1 (matching correlate reversed)
    c = np.correlate(h, 1.0 - h, mode='full')
    return c[::-1].copy()

def grad_col(h, N, j):
    # g(s, j) for s = -(N-1)..N-1
    g = np.zeros(2*N - 1)
    idx = range(max(0, j - N + 1), min(N, j + N))
    # g(s,j) = (1-h[j+s]) if 0<=j+s<N minus (h[j-s]) if 0<=j-s<N
    for s in range(-(N-1), N):
        v = 0.0
        ip = j + s
        if 0 <= ip < N:
            v += 1.0 - h[ip]
        im = j - s
        if 0 <= im < N:
            v -= h[im]
        g[s + N - 1] = v
    return g
```

    s_min1, s_max1 = -j, N - 1 - j

def pair_line_search(h, q, p, r, N):
    # Feasible interval
    lo = max(-h[p], h[r] - 1.0)
    hi = min(1.0 - h[p], h[r])
    if hi - lo < 1e-12:
        return None
    # Gradient columns
    gp = grad_col(h, N, p)
    gr = grad_col(h, N, r)
    b = gp - gr
    # a vector
    a = np.zeros(2*N - 1)
    a[N-1] = -2.0  # s=0
    d = r - p
    a[N-1+d] = 1.0  # s=d
    a[N-1-d] = 1.0  # s=-d
    # Mesh search
    coarse = np.linspace(lo, hi, 9)
    if 0.0 >= lo and 0.0 <= hi:
        coarse = np.unique(np.concatenate([coarse, [0.0]]))
    best_t = 0.0
    best_val = np.inf
    for t in coarse:
        Q = np.max(q + b*t + a*t*t)
        if Q < best_val - 1e-15:
            best_val = Q
            best_t = t
        elif abs(Q - best_val) <= 1e-15 and abs(t) < abs(best_t):
            best_t = t
    # Refinement
    # Find the bracketing interval
    sorted_t = np.sort(coarse)
    best_idx = np.where(coarse == best_t)[0][0]
    if best_idx > 0 and best_idx < len(sorted_t) - 1:
        seg_lo, seg_hi = sorted_t[best_idx-1], sorted_t[best_idx+1]
    elif best_idx > 0:
        seg_lo, seg_hi = sorted_t[best_idx-1], sorted_t[best_idx]
    else:
        seg_lo, seg_hi = sorted_t[best_idx], sorted_t[best_idx+1]
    for _ in range(3):
        mesh = np.linspace(seg_lo, seg_hi, 5)
        local_best_t = best_t
        local_best_val = best_val
        for t in mesh:
            Q = np.max(q + b*t + a*t*t)
            if Q < local_best_val - 1e-15:
                local_best_val = Q
                local_best_t = t
            elif abs(Q - local_best_val) <= 1e-15 and abs(t) < abs(local_best_t):
                local_best_t = t
        if local_best_val < best_val:
            best_val = local_best_val
            best_t = local_best_t
        # Narrow bracket
        mesh_sorted = np.sort(mesh)
        bi = np.where(mesh == best_t)[0]
        if len(bi) > 0:
            bi = bi[0]
        else:
            bi = np.argmin(np.abs(mesh - best_t))
        if bi > 0 and bi < len(mesh_sorted)-1:
            seg_lo, seg_hi = mesh_sorted[bi-1], mesh_sorted[bi+1]
        elif bi > 0:
            seg_lo, seg_hi = mesh_sorted[bi-1], mesh_sorted[bi]
        else:
            seg_lo, seg_hi = mesh_sorted[bi], mesh_sorted[bi+1]
    if best_val * 2.0 / N > q.max() * 2.0 / N + 1e-12:
        return None
    return best_t

def pair_descent(h, N, m, rng, n_sweeps=2, pair_cap=None):
    best = h.copy()
    best_score = exact_score(best, N)
    q = lag_vector(best, N)
    n_pairs = N*(N-1)//2
    if pair_cap is not None:
        n_pairs = min(n_pairs, pair_cap)
    for sweep in range(n_sweeps):
        # Generate pair list
        if n_pairs == N*(N-1)//2:
            pairs = [(i,j) for i in range(N) for j in range(i+1, N)]
        else:
            pairs = set()
            attempts = 0
            while len(pairs) < n_pairs and attempts < 50*n_pairs + 100:
                i, j = rng.integers(0, N, size=2)
                if i != j:
                    pairs.add((min(i,j), max(i,j)))
                attempts += 1
            # Fill remaining
            if len(pairs) < n_pairs:
                for i in range(N):
                    for j in range(i+1, N):
                        if len(pairs) >= n_pairs:
                            break
                        if (i,j) not in pairs:
                            pairs.add((i,j))
            pairs = list(pairs)
        rng.shuffle(pairs)
        improved = False
        for p, r in pairs:
            t = pair_line_search(best, q, p, r, N)
            if t is None or abs(t) < 1e-14:
                continue
            # Apply
            total = best[p] + best[r]
            newp = min(max(best[p] + t, max(0.0, total - 1.0)), min(1.0, total))
            newr = total - newp
            cand = best.copy()
            cand[p] = newp
            cand[r] = newr
            if not np.all(np.isfinite(cand)) or cand.min() < -1e-10 or cand.max() > 1.0+1e-10:
                continue
            if abs(cand.sum() - m) > 1e-8:
                continue
            s_new = exact_score(cand, N)
            if s_new < best_score - 1e-12:
                best = cand
                best_score = s_new
                q = lag_vector(best, N)
                improved = True
        if not improved:
            break
    return best, best_score

def annealing_chain(h, N, m, rng, L=160):
    cur = h.copy()
    cur_score = exact_score(cur, N)
    best = cur.copy()
    best_score = cur_score
    for j in range(L):
        f = j / max(1, L-1)
        sigma = 0.12 * (0.025/0.12)**f
        T = 0.002 * (0.0001/0.002)**f
        p = rng.integers(0, N)
        r = rng.integers(0, N-1)
        if r >= p:
            r += 1
        lo = max(-cur[p], cur[r] - 1.0)
        hi = min(1.0 - cur[p], cur[r])
        lo = max(lo, -sigma)
        hi = min(hi, sigma)
        if hi - lo < 1e-12:
            continue
        t = rng.uniform(lo, hi)
        total = cur[p] + cur[r]
        newp = min(max(cur[p] + t, max(0.0, total-1.0)), min(1.0, total))
        newr = total - newp
        cand = cur.copy()
        cand[p] = newp
        cand[r] = newr
        s_new = exact_score(cand, N)
        delta = s_new - cur_score
        if delta <= 0 or (T > 0 and delta/T < 700 and rng.random() < math.exp(-delta/T)):
            cur = cand
            cur_score = s_new
            if s_new < best_score:
                best = cand.copy()
                best_score = s_new
    return best, best_score
