    lo, hi = z.min() - 1.0, z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    if r > 0:
        for i in range(N):
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
            if r <= 1e-14:
                break
    elif r < 0:
        for i in range(N):
            a = min(-r, h[i])
            h[i] -= a
            r += a
            if r >= -1e-14:
                break
    if h.min() < -1e-10 or h.max() > 1 + 1e-10:
        return np.full(N, 0.5)
    return h

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / N)

def smooth_obj_grad(h, tau, N):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = dx * corr
    cmax = np.max(c)
    e = np.exp((c - cmax) / tau)
    se = np.sum(e)
    p = e / se
    F = cmax + tau * np.log(se)
    grad = np.zeros(N)
    grad += dx * p[N - 1] * (1.0 - 2.0 * h)
    for s in range(1, N):
        r = N - 1 + s
        w = dx * p[r]
        grad[s:] += w * (1.0 - h[:N - s])
        grad[:N - s] -= w * h[s:]
    for d in range(1, N):
        r = N - 1 - d
        w = dx * p[r]
        grad[:N - d] += w * (1.0 - h[d:])
        grad[d:] -= w * h[:N - d]
    return F, grad, c

def optimize_start(h0, N, m, tau_list, max_iter):
    h = h0.copy()
    best_h = h.copy()
    best_score = score_exact(h, N)
    for tau in tau_list:
        F_cur, grad, c = smooth_obj_grad(h, tau, N)
        if not np.isfinite(F_cur) or not np.all(np.isfinite(grad)):
            break
        alpha = 1.0
        stall = 0
        for it in range(max_iter):
            y = project_box_sum(h - alpha * grad, m)
            d = y - h
            if not np.all(np.isfinite(d)) or np.max(np.abs(d)) < 1e-10:
                break
            gd = np.dot(grad, d)
            if gd >= 0:
                alpha *= 0.5
                if alpha < 1e-12:
                    break
                continue
            F_new, _, c_new = smooth_obj_grad(y, tau, N)
            s_new = float(np.max(c_new))
            if s_new < best_score:
                best_h = y.copy()
                best_score = s_new
            if F_new <= F_cur + 1e-4 * gd + 1e-12:
                dec = F_cur - F_new
                h = y
                F_cur = F_new
                if dec < 1e-12 * max(1.0, abs(F_cur)):
                    stall += 1
                    if stall >= 5:
                        break
                else:
                    stall = 0
                alpha = min(1.25 * alpha, 10.0)
            else:
                alpha *= 0.5
                if alpha < 1e-12:
                    break
    return h, best_h, best_score

def make_fourier_start(N, rng, n_modes, amp):
    z = np.full(N, 0.5)
    ns = min(n_modes, (N - 1) // 2)
    idx = np.arange(N)
    for r in range(1, ns + 1):
        A = rng.standard_normal()
        B = rng.standard_normal()
        z += (amp / r) * (A * np.sin(2.0 * np.pi * r * idx / N) + B * np.cos(2.0 * np.pi * r * idx / N))
    return z

def make_block_start(N):
    h = np.zeros(N)
    third = N // 3
    h[:third] = 1.0
    h[2 * third:] = 1.0
    return h

def run(seed=42, **kwargs):
    raw = globals().get('initial_h_values', None)
    N = None
    if raw is not None:
        raw = np.asarray(raw, dtype=np.float64)
        if raw.ndim == 1 and raw.size >= 2 and np.all(np.isfinite(raw)):
            N = raw.size
    if N is None:
        N = kwargs.get('n_points', 70)
        if not isinstance(N, (int, np.integer)) or N < 2 or N > 999:
            N = 70
    m = N / 2.0
    rng = np.random.default_rng(seed)

    if raw is not None and raw.size == N and np.all(np.isfinite(raw)):
        base = project_box_sum(raw.copy(), m)
    else:
        base = np.full(N, 0.5)
    base_score = score_exact(base, N)
    best_h = base.copy()
    best_score = base_score

    starts = [base.copy(), project_box_sum(1.0 - base, m)]
    for sc in [0.04, 0.12, 0.25]:
        starts.append(project_box_sum(base + sc * rng.standard_normal(N), m))
    starts.append(project_box_sum(rng.uniform(size=N), m))
    starts.append(project_box_sum(make_fourier_start(N, rng, 4, 0.12), m))
    starts.append(project_box_sum(make_fourier_start(N, rng, 3, 0.22), m))
    starts.append(project_box_sum(make_block_start(N), m))

    unique_starts = [starts[0]]
    for s in starts[1:]:
        if all(np.max(np.abs(s - u)) > 1e-10 for u in unique_starts):
            unique_starts.append(s)

    if N <= 128:
        tau_list = [0.02, 0.008, 0.003, 0.001, 0.0004, 0.00015, 5e-05, 1e-05]
        max_iter = 60
        n_starts = 8
        hop_scales = [0.04, 0.015]
    elif N <= 256:
        tau_list = [0.02, 0.008, 0.003, 0.001, 0.0004]
        max_iter = 30
        n_starts = 6
        hop_scales = [0.04]
    elif N <= 512:
        tau_list = [0.02, 0.008, 0.003, 0.001]
        max_iter = 16
        n_starts = 4
        hop_scales = []
    else:
        tau_list = [0.02, 0.008, 0.003]
        max_iter = 8
        n_starts = 3
        hop_scales = []

    for h0 in unique_starts[:n_starts]:
        _, bh, bs = optimize_start(h0, N, m, tau_list, max_iter)
        if bs < best_score:
            best_h = bh.copy()
            best_score = bs

    for sc in hop_scales:
        hop = project_box_sum(best_h + sc * rng.standard_normal(N), m)
        _, bh, bs = optimize_start(hop, N, m, tau_list, max_iter)
        if bs < best_score:
            best_h = bh.copy()
            best_score = bs

    best_h = project_box_sum(best_h, m)
    c5 = score_exact(best_h, N)
    if not np.isfinite(c5):
        best_h = base.copy()
        c5 = base_score
    return best_h, float(c5), int(N)
