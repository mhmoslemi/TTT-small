def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def lag_data(x, n):
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.zeros(L)
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        p0 = max(d, 0)
        q0 = max(-d, 0)
        m = n - abs(d)
        G[row, p0:p0 + m] += dx * (1.0 - x[q0:q0 + m])
        G[row, q0:q0 + m] -= dx * x[p0:p0 + m]
        q[row] = dx * np.dot(x[p0:p0 + m], 1.0 - x[q0:q0 + m])
    return q, G

def _slq_obj(z):
    y, t = z[:-1], z[-1]
    return t + _slq['bdx'] * np.dot(y - _slq['x'], y - _slq['x'])

def _slq_grad(z):
    gy = 2.0 * _slq['bdx'] * (z[:-1] - _slq['x'])
    return np.concatenate([gy, [1.0]])

def _slq_ineq_fun(z):
    return z[-1] - _slq['q'] - _slq['G'] @ (z[:-1] - _slq['x'])

def _slq_ineq_jac(z):
    return np.hstack([-_slq['G'], np.ones((_slq['L'], 1))])

def _slq_eq_fun(z):
    return np.sum(z[:-1]) - _slq['s']

def _slq_eq_jac(z):
    j = np.zeros(z.shape[0])
    j[:-1] = 1.0
    return j

def mm_step(x, n, beta, maxiter):
    dx = 2.0 / n
    q, G = lag_data(x, n)
    _slq['x'] = x
    _slq['q'] = q
    _slq['G'] = G
    _slq['bdx'] = beta * dx
    _slq['s'] = n / 2.0
    _slq['L'] = 2 * n - 1
    y0 = np.concatenate([x, [np.max(q) + 1e-6]])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    cons = [
        {'type': 'ineq', 'fun': _slq_ineq_fun, 'jac': _slq_ineq_jac},
        {'type': 'eq', 'fun': _slq_eq_fun, 'jac': _slq_eq_jac}
    ]
    try:
        res = minimize(_slq_obj, y0, jac=_slq_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        y = res.x[:n]
    except Exception:
        return None
    if y is None or y.shape != (n,) or not np.all(np.isfinite(y)):
        return None
    return project_capped(y, n)

def optimize_start(start, n, n_outer, slsqp_maxiter):
    x = start.copy().astype(np.float64)
    score = exact_score(x, n)
    if not np.isfinite(score):
        return x, score
    beta = 1.0
    rejs = 0
    for _ in range(n_outer):
        if rejs >= 5 or beta > 64.0:
            break
        y = mm_step(x, n, beta, slsqp_maxiter)
        if y is None or not is_feasible(y, n):
            beta = min(beta * 2.0, 64.0)
            rejs += 1
            continue
        if np.max(np.abs(y - x)) < 1e-9:
            break
        best_z, best_s = None, score
        for k in range(5):
            alpha = 1.0 / (2 ** k)
            z = project_capped(x + alpha * (y - x), n)
            if z is None:
                continue
            sz = exact_score(z, n)
            if np.isfinite(sz) and sz < best_s:
                best_s, best_z = sz, z
        if best_z is None or best_s >= score - 1e-12:
            beta = min(beta * 2.0, 64.0)
            rejs += 1
            continue
        x, score = best_z.copy(), best_s
        rejs = 0
        beta = max(1.0, beta / 1.5)
    return x, exact_score(x, n)

def coord_refine(x, n, n_pass):
    x = x.copy()
    for _ in range(n_pass):
        improved = False
        for i in range(n):
            old = x[i]
            s0 = exact_score(x, n)
            for delta in (-0.08, -0.03, 0.03, 0.08):
                x[i] = old + delta
                xp = project_capped(x, n)
                if xp is None:
                    continue
                s = exact_score(xp, n)
                if np.isfinite(s) and s < s0 - 1e-12:
                    x = xp.copy()
                    s0 = s
                    improved = True
                    break
        if not improved:
            break
    return x, exact_score(x, n)

def build_starts(fallback, n, rng, count):
    pool = []
    def _add(v):
        if v is None:
            return
        for s in pool:
            if np.max(np.abs(v - s)) < 1e-8:
                return
        pool.append(v)
    _add(fallback.copy())
    _add(project_capped(fallback[::-1], n))
    _add(project_capped(1.0 - fallback[::-1], n))
    _add(project_capped(1.0 - fallback, n))
    _add(np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    _add(project_capped(blk, n))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    _add(project_capped(alt, n))
    for sd in (0.06, 0.15, 0.30):
        _add(project_capped(fallback + rng.normal(0, sd, n), n))
    _add(project_capped(rng.uniform(0, 1, n), n))
    return pool[:count]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if parent is not None:
        fallback = project_capped(parent, n)
    else:
        fallback = np.full(n, 0.5)
    if fallback is None:
        fallback = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 1:
        return fallback, float(exact_score(fallback, n)), n
    if n <= 64:
        n_starts, n_outer, slq_max, n_ref = 8, 12, 80, 3
    elif n <= 128:
        n_starts, n_outer, slq_max, n_ref = 5, 8, 60, 2
    elif n <= 256:
        n_starts, n_outer, slq_max, n_ref = 3, 5, 60, 1
    else:
        n_starts, n_outer, slq_max, n_ref = 2, 3, 40, 1
    starts = build_starts(fallback, n, rng, n_starts)
    scored = []
    for st in starts:
        s = exact_score(st, n)
        scored.append((s, st))
    scored.sort(key=lambda t: t[0])
    promote = [scored[0][1].copy()]
    for s_val, st in scored:
        if len(promote) >= n_starts:
            break
        if s_val < float('inf'):
            promote.append(st.copy())
    gb, gs = fallback.copy(), exact_score(fallback, n)
    for st in promote:
        s = exact_score(st, n)
        if np.isfinite(s) and s < gs - 1e-12:
            gs, gb = s, st.copy()
    for st in promote:
        try:
            xo, so = optimize_start(st, n, n_outer, slq_max)
            if np.isfinite(so) and so < gs - 1e-12:
                gs, gb = so, xo.copy()
        except Exception:
            pass
    if n_ref > 0:
        gb, gs = coord_refine(gb, n, n_ref)
    gb_proj = project_capped(gb, n)
    if gb_proj is None or not is_feasible(gb_proj, n):
        gb = fallback
    fb_score = exact_score(fallback, n)
    gb_score = exact_score(gb, n)
    if not np.isfinite(gb_score) or gb_score >= fb_score:
        gb = fallback
        gb_score = fb_score
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = np.full(n, 0.5)
    fscore = exact_score(final, n)
    return final.astype(np.float64), float(fscore), int(n)
