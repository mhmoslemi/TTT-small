import numpy as np
from scipy.optimize import minimize

def project_K(y, n, S):
    if not np.all(np.isfinite(y)) or len(y) != n:
        return np.full(n, S / n)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    err = S - h.sum()
    if abs(err) > 1e-12:
        if err > 0:
            idx = np.where(1.0 - h > 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) > 0:
            h = h.copy()
            h[idx] = np.clip(h[idx] + err / len(idx), 0.0, 1.0)
    if abs(h.sum() - S) > 1e-8:
        return np.full(n, S / n)
    return h

def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lag_vec(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_grad_mat(h, n, dx):
    G = np.zeros((2 * n - 1, n))
    j = np.arange(n)
    for k in range(2 * n - 1):
        ell = k - (n - 1)
        if ell == 0:
            G[k, :] = dx * (1.0 - 2.0 * h)
        else:
            i1 = j - ell
            v1 = (i1 >= 0) & (i1 < n)
            i2 = j + ell
            v2 = (i2 >= 0) & (i2 < n)
            t1 = np.zeros(n)
            t2 = np.zeros(n)
            if np.any(v1):
                t1[v1] = 1.0 - h[i1[v1]]
            if np.any(v2):
                t2[v2] = h[i2[v2]]
            G[k, :] = dx * (t1 - t2)
    return G

def feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def tr_obj(z, n, eta):
    d = z[:n]
    return z[n] + np.dot(d, d) / (2.0 * eta)

def tr_obj_jac(z, n, eta):
    g = np.empty(len(z))
    g[:n] = z[:n] / eta
    g[n] = 1.0
    return g

def tr_ineq(z, n, q, G):
    return z[n] - q - G.dot(z[:n])

def tr_ineq_jac(z, n, q, G):
    J = np.empty((len(q), n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def tr_eq(z, n):
    return z[:n].sum()

def tr_eq_jac(z, n):
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def transfer_parent(hp, n0, m):
    y = np.zeros(m)
    for j in range(m):
        total = 0.0
        for i in range(n0):
            lo = max(i * 1.0 / n0, j * 1.0 / m)
            hi = min((i + 1) * 1.0 / n0, (j + 1) * 1.0 / m)
            if hi > lo:
                total += hp[i] * (hi - lo)
        y[j] = total * m
    return y

def trust_region_opt(h0, n, dx, max_iter=25, slp_maxiter=45):
    S = n / 2.0
    h = project_K(h0.copy(), n, S)
    if not feasible(h, n, S):
        return h
    score = exact_score(h, dx)
    delta = 0.12
    eta = 0.5
    cons_fail = 0
    cons_rej = 0
    
    for it in range(max_iter):
        q = lag_vec(h, dx)
        G = lag_grad_mat(h, n, dx)
        max_q = float(q.max())
        
        z0 = np.zeros(n + 1)
        z0[n] = max_q
        
        bounds = [(max(-delta, -h[j]), min(delta, 1.0 - h[j])) for j in range(n)]
        bounds.append((None, None))
        
        try:
            res = minimize(tr_obj, z0, jac=tr_obj_jac, args=(n, eta),
                           method='SLSQP', bounds=bounds,
                           constraints=[
                               {'type': 'ineq', 'fun': tr_ineq, 'jac': tr_ineq_jac,
                                'args': (n, q, G)},
                               {'type': 'eq', 'fun': tr_eq, 'jac': tr_eq_jac,
                                'args': (n,)}
                           ],
                           options={'maxiter': slp_maxiter, 'ftol': 1e-10})
        except Exception:
            cons_fail += 1
            delta *= 0.5
            if cons_fail >= 4 or delta < 1e-5:
                break
            continue
        
        if res is None or not hasattr(res, 'x') or not np.all(np.isfinite(res.x)):
            cons_fail += 1
            delta *= 0.5
            if cons_fail >= 4 or delta < 1e-5:
                break
            continue
        
        d = res.x[:n]
        # Check constraints
        if (abs(d.sum()) > 1e-4 or np.max(np.abs(d)) > delta + 1e-6):
            cons_fail += 1
            delta *= 0.5
            if cons_fail >= 4 or delta < 1e-5:
                break
            continue
        
        h_trial = project_K(h + d, n, S)
        if not feasible(h_trial, n, S):
            cons_fail += 1
            delta *= 0.5
            if cons_fail >= 4 or delta < 1e-5:
                break
            continue
        
        d_act = h_trial - h
        score_trial = exact_score(h_trial, dx)
        pred = max_q - float(np.max(q + G.dot(d_act)))
        act = score - score_trial
        
        if act > 1e-12 and (pred <= 1e-12 or act / pred >= 0.1):
            h = h_trial
            score = score_trial
            cons_fail = 0
            cons_rej = 0
            ratio = 1.0 if pred <= 1e-12 else act / pred
            if ratio < 0.25 or pred <= 1e-12:
                delta *= 0.5
            elif ratio > 0.75 and np.max(np.abs(d_act)) >= 0.8 * delta:
                delta = min(0.5, 2.0 * delta)
                eta = min(2.0, 1.5 * eta)
            if np.max(np.abs(d_act)) < 1e-9:
                break
        else:
            cons_rej += 1
            delta *= 0.5
            eta = max(1e-4, eta * 0.5)
            if cons_rej >= 8 or delta < 1e-5:
                break
    
    return h

import numpy as np
from scipy.optimize import minimize

def project_K(y, n, S):
    if not np.all(np.isfinite(y)) or len(y) != n:
        return np.full(n, S / n)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    err = S - h.sum()
    if abs(err) > 1e-13:
        if err > 0:
            idx = np.where(1.0 - h > 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) > 0:
            h = h.copy()
            h[idx] = np.clip(h[idx] + err / len(idx), 0.0, 1.0)
    if abs(h.sum() - S) > 1e-8:
        return np.full(n, S / n)
    return h

def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lag_vec(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_grad_mat(h, n, dx):
    G = np.zeros((2 * n - 1, n))
    j = np.arange(n)
    for k in range(2 * n - 1):
        ell = k - (n - 1)
        i1, v1 = j - ell, (j - ell >= 0) & (j - ell < n)
        i2, v2 = j + ell, (j + ell >= 0) & (j + ell < n)
        t1, t2 = np.zeros(n), np.zeros(n)
        if np.any(v1):
            t1[v1] = 1.0 - h[i1[v1]]
        if np.any(v2):
            t2[v2] = h[i2[v2]]
        G[k, :] = dx * (t1 - t2)
    return G

def feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def tr_obj(z, n, eta):
    d = z[:n]
    return z[n] + np.dot(d, d) / (2.0 * eta)

def tr_obj_jac(z, n, eta):
    g = np.empty(len(z))
    g[:n] = z[:n] / eta
    g[n] = 1.0
    return g

def tr_ineq(z, n, q, G):
    return z[n] - q - G.dot(z[:n])

def tr_ineq_jac(z, n, q, G):
    J = np.empty((len(q), n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def tr_eq(z, n):
    return z[:n].sum()

def tr_eq_jac(z, n):
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def transfer_parent(hp, n0, m):
    y = np.zeros(m)
    for j in range(m):
        tot = 0.0
        for i in range(n0):
            lo = max(i / n0, j / m)
            hi = min((i + 1) / n0, (j + 1) / m)
            if hi > lo:
                tot += hp[i] * (hi - lo)
        y[j] = tot * m
    return y
