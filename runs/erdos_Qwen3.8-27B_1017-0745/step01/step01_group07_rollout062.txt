import numpy as np

def proj_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):
        return None
    if np.max(np.abs(z)) > 1e12:
        return None
    n = len(z)
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    y = np.clip(z - lam, 0.0, 1.0)
    r = S - np.sum(y)
    if r > 0:
        for i in range(n):
            if r <= 0: break
            a = min(r, 1.0 - y[i])
            y[i] += a; r -= a
    elif r < 0:
        for i in range(n):
            if r >= 0: break
            a = min(-r, y[i])
            y[i] -= a; r += a
    if abs(np.sum(y) - S) > 1e-10 or not np.all(np.isfinite(y)):
        return None
    return y

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) - n / 2.0) <= 1e-10)

def exact_lags(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if corr.shape != (2 * n - 1,) or not np.all(np.isfinite(corr)):
        return None
    return corr

def exact_score(h, n):
    corr = exact_lags(h, n)
    if corr is None:
        return None
    return float(np.max(corr))

def pair_line_search(h, q, n, dx, p, r):
    L = max(-h[p], h[r] - 1.0)
    U = min(1.0 - h[p], h[r])
    if U - L <= 1e-12:
        return None
    # Compute b and a vectors (length 2n-1, indexed by correlate index)
    m = 2 * n - 1
    b = np.zeros(m)
    # b[k] for correlate index k, lag d = k - (n-1)
    # b_d = dx * [(1-h[p+d]) - (1-h[r+d]) - h[p-d] + h[r-d]] (valid indices only)
    for d in range(-(n-1), n):
        k = d + n - 1
        val = 0.0
        if 0 <= p + d < n:
            val += (1.0 - h[p + d])
        if 0 <= r + d < n:
            val -= (1.0 - h[r + d])
        if 0 <= p - d < n:
            val -= h[p - d]
        if 0 <= r - d < n:
            val += h[r - d]
        b[k] = dx * val
    # a vector
    a = np.zeros(m)
    a[n-1] = -2.0 * dx  # d=0
    dr = r - p
    if abs(dr) < n and dr != 0:
        a[n-1 + dr] = dx
        a[n-1 - dr] = dx
    # 1D minimax: f(t) = max_k(c_k + b_k*t + a_k*t^2)
    # Coarse grid
    n_grid = 17
    ts = np.linspace(L, U, n_grid)
    # Evaluate f at each t
    best_t = 0.0
    best_val = np.inf
    for t in ts:
        vals = q + b * t + a * t * t
        mv = float(np.max(vals))
        if mv < best_val or (abs(mv - best_val) < 1e-14 and abs(t) < abs(best_t)):
            best_val = mv
            best_t = t
    # Also check t=0
    cv = float(np.max(q))
    if cv < best_val:
        best_val = cv
        best_t = 0.0
    # Refine
    radius = (U - L) / 16.0
    for _ in range(4):
        lo_r = max(L, best_t - radius)
        hi_r = min(U, best_t + radius)
        ts2 = np.linspace(lo_r, hi_r, 5)
        for t in ts2:
            vals = q + b * t + a * t * t
            mv = float(np.max(vals))
            if mv < best_val or (abs(mv - best_val) < 1e-14 and abs(t) < abs(best_t)):
                best_val = mv
                best_t = t
        radius /= 4.0
    if best_val >= float(np.max(q)) - 1e-12:
        return None
    return best_t

def polish_sweep(h, score, q, n, dx, rng, n_sweeps, pair_list):
    for _ in range(n_sweeps):
        improved = False
        for (p, r) in pair_list:
            t = pair_line_search(h, q, n, dx, p, r)
            if t is not None and abs(t) > 1e-12:
                h_new = h.copy()
                h_new[p] += t
                h_new[r] -= t
                if not is_feasible(h_new, n):
                    h_new = proj_simplex(h_new, n / 2.0)
                    if h_new is None or not is_feasible(h_new, n):
                        continue
                q_new = exact_lags(h_new, n)
                if q_new is None:
                    continue
                sc_new = float(np.max(q_new))
                if sc_new < score - 1e-12:
                    h = h_new
                    score = sc_new
                    q = q_new
                    improved = True
        if not improved:
            break
    return h, score, q

def build_pair_list(n, rng):
    # Adjacent + random pairs
    pairs = []
    n_adj = min(n - 1, max(1, n // 2))
    if n_adj > 0:
        adj_idx = np.arange(n - 1)
        if n_adj < n - 1:
            adj_idx = np.linspace(0, n - 2, n_adj).astype(int)
        for i in adj_idx:
            pairs.append((int(i), int(i + 1)))
    n_tot = min(n * (n - 1) // 2, 2 * n, max(24, 80000 // n))
    while len(pairs) < n_tot:
        p = int(rng.integers(0, n))
        r = int(rng.integers(0, n))
        if p == r:
            continue
        if p > r:
            p, r = r, p
        if (p, r) not in pairs:
            pairs.append((p, r))
        if len(pairs) > n_tot * 2:
            break
    pairs = np.array(pairs, dtype=int)
    order = rng.permutation(len(pairs))
    return [(int(pairs[i, 0]), int(pairs[i, 1])) for i in order]

def anneal_trajectory(h0, n, dx, rng):
    h = h0.copy()
    S = n / 2.0
    q = exact_lags(h, n)
    if q is None:
        return h, exact_score(h0, n)
    score = float(np.max(q))
    epochs = 5 if n <= 128 else (4 if n <= 256 else 3)
    K = min(1000, max(80, 80000 // n))
    scale = 0.12
    pair_list = build_pair_list(n, rng)
    for e in range(epochs):
        T = max(5e-5, 3e-3 * (0.4 ** e))
        accepted = 0
        for _ in range(K):
            p = int(rng.integers(0, n))
            r = int(rng.integers(0, n))
            if p == r:
                continue
            L = max(-h[p], h[r] - 1.0)
            U = min(1.0 - h[p], h[r])
            if U <= L:
                continue
            t = float(rng.uniform(-scale, scale))
            if t < L or t > U:
                continue
            h_new = h.copy()
            h_new[p] += t
            h_new[r] -= t
            if not is_feasible(h_new, n):
                h_new = proj_simplex(h_new, S)
                if h_new is None or not is_feasible(h_new, n):
                    continue
            sc_new = exact_score(h_new, n)
            if sc_new is None:
                continue
            delta = sc_new - score
            if delta <= 0 or (delta < 700 * T and rng.random() < np.exp(-delta / T)):
                h = h_new
                score = sc_new
                q = exact_lags(h, n)
                if q is None:
                    q = np.full(2*n-1, score)
                accepted += 1
        ratio = accepted / K
        if ratio < 0.12:
            scale = min(0.4, 1.25 * scale)
        elif ratio > 0.55:
            scale = max(0.01, 0.75 * scale)
        h, score, q = polish_sweep(h, score, q, n, dx, rng, 1, pair_list)
    return h, score

            best_h, best_score, q_best = polish_sweep(best_h, best_score, q_best, n, dx, pair_list, ns)
    except Exception: pass
    if not is_feasible(best_h, n):
        best_h = proj_simplex(best_h, S)
        if best_h is None or not is_feasible(best_h, n):
            best_h = h_const.copy()
    final_score = exact_score(best_h, n)
    if final_score is None: final_score = best_score
    if parent is not None:
        ps2 = exact_score(parent, n)
        if ps2 is not None and final_score > ps2:
            best_h, final_score = parent.copy(), ps2
    return best_h, float(final_score), n
