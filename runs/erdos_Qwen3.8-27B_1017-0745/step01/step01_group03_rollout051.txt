                s = min(-resid, y[i]); y[i] -= s; resid += s
                if abs(resid) < 1e-15: break
    if np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12): return None
    if abs(np.sum(y) - S) > 1e-10 * max(1.0, float(n)): return None
    y = np.clip(y, 0.0, 1.0); resid = S - np.sum(y)
    if abs(resid) > 0:
        if resid > 0:
            for i in range(len(y)):
                a = min(resid, 1.0 - y[i]); y[i] += a; resid -= a
                if abs(resid) < 1e-15: break
        else:
            for i in range(len(y)):
                s = min(-resid, y[i]); y[i] -= s; resid += s
                if abs(resid) < 1e-15: break
    return y

def check_feasible(h, n):
    if h is None or h.shape != (n,) or not np.all(np.isfinite(h)): return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12): return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1.0, float(n)): return False
    return True

def exact_score(h, n):
    if not check_feasible(h, n): return float('inf')
    val = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
    return val if np.isfinite(val) else float('inf')

def lag_coefficients(x, d, n):
    n_lags = 2 * n - 1
    A = np.zeros(n_lags); B = np.zeros(n_lags); D = np.zeros(n_lags)
    for k in range(-(n - 1), n):
        idx = k + n - 1
        i_lo = max(0, k); i_hi = min(n - 1, n - 1 + k)
        if i_lo > i_hi: continue
        i = np.arange(i_lo, i_hi + 1); j = i - k
        D[idx] = np.sum(x[i] * (1.0 - x[j]))
        B[idx] = np.sum(d[i] * (1.0 - x[j]) - x[i] * d[j])
        A[idx] = -np.sum(d[i] * d[j])
    return A, B, D

def line_minimax(x, y, n, S):
    if not check_feasible(x, n) or not check_feasible(y, n): return x, exact_score(x, n)
    d = y - x
    if np.max(np.abs(d)) < 1e-14: return x, exact_score(x, n)
    A, B, D = lag_coefficients(x, d, n)
    n_lags = 2 * n - 1
    cands = [0.0, 1.0]
    for k in range(n_lags):
        if abs(A[k]) > 1e-14:
            a = -B[k] / (2.0 * A[k])
            if -1e-10 <= a <= 1.0 + 1e-10: cands.append(np.clip(a, 0.0, 1.0))
    for k in range(n_lags):
        for l in range(k + 1, n_lags):
            da, db, dd = A[k]-A[l], B[k]-B[l], D[k]-D[l]
            if abs(da) < 1e-14:
                if abs(db) > 1e-14:
                    a = -dd / db
                    if -1e-10 <= a <= 1.0 + 1e-10: cands.append(np.clip(a, 0.0, 1.0))
            else:
                disc = db * db - 4.0 * da * dd
                if disc >= -1e-10:
                    sq = np.sqrt(max(0.0, disc))
                    for a in [(-db+sq)/(2.0*da), (-db-sq)/(2.0*da)]:
                        if np.isfinite(a) and -1e-10 <= a <= 1.0 + 1e-10:
                            cands.append(np.clip(a, 0.0, 1.0))
    cands = np.unique(np.round(np.array(cands), 10))
    best_q = float('inf'); best_a = 0.0
    for alpha in cands:
        q = np.max(A * alpha * alpha + B * alpha + D)
        if q < best_q: best_q = q; best_a = alpha
    alphas = np.unique(np.concatenate([[0.0, 1.0, best_a,
        np.clip(best_a-1e-8, 0, 1), np.clip(best_a+1e-8, 0, 1)]]))
    best_h = x.copy(); best_s = exact_score(x, n)
    for alpha in alphas:
        pt = project_box_sum(x + alpha * d, S)
        if pt is None or not check_feasible(pt, n): continue
        s = exact_score(pt, n)
        if s < best_s: best_s = s; best_h = pt.copy()
    return best_h, best_s

def build_and_solve_lp(x, r, n, S):
    lb = np.maximum(0.0, x - r); ub = np.minimum(1.0, x + r)
    P = n * (n + 1) // 2; nvar = n + P + 1
    def pidx(i, j): return n + min(i, j) * n - min(i, j) * (min(i, j) - 1) // 2 + (max(i, j) - min(i, j))
    R, C, V, b = [], [], [], []
    for i in range(n):
        for j in range(i, n):
            pi = pidx(i, j)
            li, ui, lj, uj = lb[i], ub[i], lb[j], ub[j]
            for (sI, rI, cI, vI, bI) in [
                (0, pi, n+j, -1.0, -li*lj), (0, n+j, pi, li, 0.0), (0, n+i, pi, lj, 0.0),
                (1, pi, n+j, -1.0, -ui*uj), (1, n+j, pi, ui, 0.0), (1, n+i, pi, uj, 0.0),
                (2, pi, n+j, -1.0, ui*lj), (2, n+j, pi, ui, 0.0), (2, n+i, pi, lj, 0.0),
                (3, pi, n+j, -1.0, li*uj), (3, n+j, pi, li, 0.0), (3, n+i, pi, uj, 0.0)]:
                row = len(b); b.append(bI)
                R.append(row); C.append(cI); V.append(vI)
    n_lags = 2 * n - 1
    for k in range(-(n-1), n):
        idx = k + n - 1
        i_lo = max(0, k); i_hi = min(n-1, n-1+k)
        if i_lo > i_hi: continue
        row = len(b); b.append(0.0)
        for i in range(i_lo, i_hi + 1):
            j = i - k; pi = pidx(i, j)
            R.append(row); C.append(i); V.append(1.0)
            R.append(row); C.append(pi); V.append(-1.0)
        R.append(row); C.append(n + P); V.append(-1.0)
    nrow = len(b)
    A_ub = coo_matrix((V, (R, C)), shape=(nrow, nvar)).toarray()
    b_ub = np.array(b)
    A_eq = np.zeros((1, nvar)); b_eq = np.array([S])
    A_eq[0, :n] = 1.0
    c_obj = np.zeros(nvar); c_obj[n+P] = 1.0
    bounds = [(lb[i], ub[i]) for i in range(n)]
    for i in range(n):
        for j in range(i, n):
            bounds.append((lb[i]*lb[j], ub[i]*ub[j]))
    bounds.append((0.0, float(n)))
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'time_limit': 30})
    if res.status != 0 or res.x is None: return None
    v = res.x[:n]
    if not np.all(np.isfinite(v)): return None
    return np.clip(v, lb - 1e-7, ub + 1e-7)

def local_lp_search(start, n, S):
    curr = start.copy(); best = start.copy()
    best_s = exact_score(curr, n)
    if not np.isfinite(best_s): return best, best_s
    r = 0.20; max_att = 8 if n <= 64 else 5; fails = 0
    for _ in range(max_att):
        if r < 1e-3 and fails >= 2: break
        try:
            v = build_and_solve_lp(curr, r, n, S)
        except Exception:
            v = None
        if v is None:
            r *= 0.5; fails += 1; continue
        v_proj = project_box_sum(v, S)
        if v_proj is None or not check_feasible(v_proj, n):
            r *= 0.5; fails += 1; continue
        if np.max(np.abs(v_proj - curr)) < 1e-12:
            r *= 0.5; fails += 1; continue
        h_c, s_c = line_minimax(curr, v_proj, n, S)
        if s_c < best_s - 1e-12:
            best_s = s_c; best = h_c.copy(); curr = h_c.copy()
            fails = 0; r = min(0.25, 1.2 * r)
        else:
            r *= 0.5; fails += 1
    return best, best_s

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = None; n = 45
    try:
        pv = np.array(initial_h_values, dtype=np.float64)
        if pv.ndim == 1 and 2 <= len(pv) <= 999 and np.all(np.isfinite(pv)):
            if np.all(pv >= -1e-9) and np.all(pv <= 1.0 + 1e-9):
                n = len(pv); S = n / 2.0
                if abs(np.sum(pv) - S) < 1e-6 * max(1.0, float(n)):
                    parent = project_box_sum(np.clip(pv, 0.0, 1.0), S)
    except Exception:
        pass
    S = n / 2.0
    uniform = project_box_sum(np.full(n, 0.5), S)
    fallback = parent if parent is not None else uniform
    fallback_s = exact_score(fallback, n)

    starts = []
    if parent is not None:
        for s in [parent, parent[::-1], 1.0 - parent]:
            c = project_box_sum(s, S)
            if c is not None: starts.append(c)
        c = project_box_sum(np.roll(parent, n // 3), S)
        if c is not None: starts.append(c)
        for amp in [0.04, 0.12]:
            c = project_box_sum(parent + amp * rng.standard_normal(n), S)
            if c is not None: starts.append(c)
        f1 = rng.integers(1, max(2, n // 2)); f2 = rng.integers(1, max(2, n // 2))
        hv = 0.08 * (np.cos(2*np.pi*f1*np.arange(n)/n + rng.uniform(0,6.28)) +
                     np.cos(2*np.pi*f2*np.arange(n)/n + rng.uniform(0,6.28)))
        if np.sqrt(np.mean(hv**2)) > 1e-12:
            c = project_box_sum(parent + hv, S)
            if c is not None: starts.append(c)
        for _ in range(4):
            c = project_box_sum(rng.uniform(0,1,n), S)
            if c is not None: starts.append(c)
    else:
        starts.append(uniform)
        for _ in range(6):
            c = project_box_sum(rng.uniform(0,1,n), S)
            if c is not None: starts.append(c)
        for m in range(1, min(4, n // 2 + 1)):
            c = project_box_sum(0.5 + 0.35*np.cos(2*np.pi*m*np.arange(n)/n + rng.uniform(0,6.28)), S)
            if c is not None: starts.append(c)

    starts = [s for s in starts if s is not None]
    deduped = []
    for s in starts:
        if not any(np.max(np.abs(s - t)) < 1e-10 for t in deduped):
            deduped.append(s)
    starts = deduped

    best_h = fallback.copy(); best_s = fallback_s
    for s in starts:
        sc = exact_score(s, n)
        if sc < best_s - 1e-12: best_s = sc; best_h = s.copy()

    if n <= 64:
        for s in starts:
            h_opt, s_opt = local_lp_search(s, n, S)
            if s_opt < best_s - 1e-12: best_s = s_opt; best_h = h_opt.copy()

    amps = [0.005, 0.02, 0.06, 0.12]
    n_pert = min(8, max(4, int(np.ceil(800.0 / n))))

        amp = amps[i % 4]
        if i % 2 == 0:
            pert = best_h + amp * rng.standard_normal(n)
        else:
            f = rng.integers(1, max(2, n // 2))
            pert = best_h + amp * np.cos(2*np.pi*f*np.arange(n)/n + rng.uniform(0,6.28))
        c = project_box_sum(pert, S)
        if c is None: continue
        sc = exact_score(c, n)
        if sc < best_s - 1e-12: best_s = sc; best_h = c.copy()

    final = project_box_sum(best_h, S)
    if final is None or not check_feasible(final, n):

    final_s = exact_score(final, n)
    if not np.isfinite(final_s) or final_s > fallback_s + 1e-12:
        final = fallback; final_s = fallback_s
    return (np.array(final, dtype=np.float64), final_s, n)
