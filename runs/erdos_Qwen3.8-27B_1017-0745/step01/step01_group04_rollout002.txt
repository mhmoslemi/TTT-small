```python
def compute_b(p, r, h, N):
    b = np.zeros(2*N-1)
    for k in range(-(N-1), N):
        j = k + N - 1
        val = 0.0
        pk = p - k
        rk = r - k
        p_k = p + k
        r_k = r + k
        if 0 <= pk < N:
            val += (1.0 - h[pk])
        if 0 <= rk < N:
            val -= (1.0 - h[rk])
        if 0 <= p_k < N:
            val -= h[p_k]
        if 0 <= r_k < N:
            val += h[r_k]
        b[j] = val
    return b
```
