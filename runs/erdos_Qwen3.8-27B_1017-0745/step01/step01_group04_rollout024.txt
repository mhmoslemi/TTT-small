def project_box_sum(z, m):

    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    shifted = z - np.median(z)
    lo, hi = shifted.min() - 1.0, shifted.max()
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        w = np.clip(shifted - mid, 0.0, 1.0)
        if w.sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(shifted - 0.5 * (lo + hi), 0.0, 1.0)

            adj = min(r, 1.0 - h[i])

def grad_col(h, j, N):

def apply_pair_transfer(h, p, r_idx, t):
    h_new = h.copy()
    total = h_new[p] + h_new[r_idx]
    h_new[p] = np.clip(h_new[p] + t, 0.0, min(1.0, total if total <= 1.0 else 1.0))

def pair_descent(h0, N, m, n_sweeps, rng, max_pairs=None):

                    pairs.append(key)
                    if len(pairs) >= pairs_per_sweep:

            if len(pairs) < pairs_per_sweep:

                            seen.add((p,r))
                    if len(pairs) >= pairs_per_sweep:

                        q = q_cand

def sa_chain(h0, N, m, n_props, rng):

    score = np.max(q) * 2.0 / N
    best_h = h.copy()
    best_score = score
    
    for j in range(n_props):
        f = j / max(1, n_props - 1)
        sigma = 0.12 * (0.025 / 0.12) ** f
        T = 0.002 * (0.0001 / 0.002) ** f
        
        p = rng.integers(0, N)
        r = rng.integers(0, N-1)
        if r >= p:
            r += 1
        
        t_lo = max(-h[p], h[r] - 1.0)
        t_hi = min(1.0 - h[p], h[r])
        t_lo = max(t_lo, -sigma)
        t_hi = min(t_hi, sigma)
        if t_hi - t_lo <= 1e-12:
            continue
        
        t = rng.uniform(t_lo, t_hi)
        h_cand = h.copy()
        h_cand[p] = h[p] + t
        h_cand[r] = h[r] - t
        if h_cand.min() < -1e-12 or h_cand.max() > 1+1e-12:
            continue
        h_cand = np.clip(h_cand, 0.0, 1.0)
        if abs(h_cand.sum() - m) > 1e-8:
            continue
        
        q_cand = get_q(h_cand, N)
        score_cand = float(np.max(q_cand) * 2.0 / N)
        delta = score_cand - score
        
        if delta <= 0 or (T > 1e-15 and rng.random() < math.exp(-delta/T) if delta/T < 700 else True):
            h = h_cand
            q = q_cand
            score = score_cand
            if score < best_score:
                best_h = h.copy()
                best_score = score
    
    return best_h, best_score

def make_fourier_start(N, rng, n_modes, amp):
    z = np.full(N, 0.5)
    ns = min(n_modes, (N-1)//2)
    idx = np.arange(N)
    for rr in range(1, ns+1):
        A = rng.standard_normal()
        B = rng.standard_normal()
        z += (amp/rr) * (A*np.sin(2*np.pi*rr*idx/N) + B*np.cos(2*np.pi*rr*idx/N))
    return z

def project_box_sum(z, m):

        else:

            adj = min(r, 1.0-h[i]); h[i] += adj; r -= adj
            if r <= 1e-14: break
    elif r < 0:

def grad_col(h, j, N):
    out = np.zeros(2*N-1)
    lo1, hi1 = max(-(N-1), j-N+1), min(N-1, j)
    if hi1 >= lo1:
        ks = np.arange(lo1, hi1+1)
        out[ks+N-1] += 1.0 - h[j-ks]
    lo2, hi2 = max(-(N-1), -j), min(N-1, N-1-j)
    if hi2 >= lo2:
        ks = np.arange(lo2, hi2+1)
        out[ks+N-1] -= h[j+ks]
    return out

    c5, c5b = exact_score(best, N), exact_score(base, N)
    if not np.isfinite(c5) or c5 >= c5b:
        best, c5 = base, c5b
    return best, float(c5), int(N)
