import numpy as np

def proj_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):
        return None
    if np.max(np.abs(z)) > 1e12:
        return None
    n = len(z)
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - np.sum(x)
    if r > 0:
        for i in range(n):
            if r <= 0: break
            a = min(r, 1.0 - x[i])
            x[i] += a; r -= a
    elif r < 0:
        for i in range(n):
            if r >= 0: break
            a = min(-r, x[i])
            x[i] -= a; r += a
    if abs(np.sum(x) - S) > 1e-10 or not np.all(np.isfinite(x)):
        return None
    return np.clip(x, 0.0, 1.0)

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-9)

def exact_score(h, n):
    vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(vals)):
        return None
    return float(np.max(vals))

def exact_lags(h, n):
    vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(vals)) or len(vals) != 2 * n - 1:
        return None
    return vals  # index li corresponds to lag d = li - (n-1)

def pair_line_search(h, q, p, r, n, dx):

    U = min(1.0 - h[p], h[r])
    if U - L <= 1e-12:

    #             - h[p-d]*I(0<=p-d<n) + h[r-d]*I(0<=r-d<n)]
    # We iterate over d from -(n-1) to n-1
    # For efficiency, let's compute directly
    # lag index for d is d + (n-1)
    for d in range(-(n-1), n):
        li = d + n - 1
        val = 0.0
        pd = p + d
        rd = r + d
        pm = p - d
        rm = r - d
        if 0 <= pd < n:
            val += (1.0 - h[pd])
        if 0 <= rd < n:
            val -= (1.0 - h[rd])
        if 0 <= pm < n:
            val -= h[pm]
        if 0 <= rm < n:
            val += h[rm]
        b[li] = dx * val
    
    a = np.zeros(m)
    a[n-1] = -2.0 * dx  # d=0
    dr = r - p  # r > p or r < p
    if dr != 0:
        a[n-1 + dr] = dx  # d = r-p
        a[n-1 - dr] = dx  # d = p-r

    # If p=78, r=2, dr=-76, n-1+dr=79-76=3 >= 0 ✓

    t_best = 0.0
    f_best = float(np.max(c))  # t=0 gives current score

        if f < f_best or (abs(f - f_best) < 1e-14 and abs(t) < abs(t_best)):

            t_best = t

def build_starts(n, S, parent, rng):
    starts = []
    const = np.full(n, 0.5)
    if parent is not None:
        starts.append(parent.copy())
    # Low-frequency
    idx = np.arange(n, dtype=np.float64)
    u = np.full(n, 0.5)
    M = min(4, (n - 1) // 2)
    for m in range(1, M + 1):
        A = rng.normal(0.0, 0.12 / np.sqrt(m))
        phi = rng.uniform(0.0, 2.0 * np.pi)
        u += A * np.cos(2.0 * np.pi * m * (idx + 0.5) / n + phi)
    pp = proj_simplex(u, S)
    if pp is not None:
        starts.append(pp)
    # Binary
    perm = rng.permutation(n)
    binv = np.zeros(n)
    binv[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        binv[perm[n // 2]] = 0.5
    starts.append(binv.copy())
    # Random
    rv = rng.random(n)
    rp = proj_simplex(rv, S)
    if rp is not None:
        starts.append(rp)
    if parent is not None:
        starts.append(parent[::-1].copy())
        starts.append(1.0 - parent.copy())
    # Dedup
    deduped = []
    for s in starts:
        s2 = np.asarray(s, dtype=np.float64)
        if not is_feasible(s2, n):
            s2 = proj_simplex(s2, S)
            if s2 is None:
                continue
        dup = False
        for d in deduped:
            if np.max(np.abs(s2 - d)) < 1e-10:
                dup = True; break
        if not dup:
            deduped.append(s2)
    if parent is None:
        deduped.insert(0, const)
    return deduped

def polish_sweep(h, score, q, n, dx, rng, max_pairs):
    P = min(n * (n - 1) // 2, max_pairs)
    pairs = []
    # Adjacent pairs
    n_adj = min(n - 1, max(1, P // 2))
    adj_idx = np.linspace(0, n - 2, n_adj).astype(int)
    for i in adj_idx:
        pairs.append((i, i + 1))
    # Random pairs to fill
    while len(pairs) < P:
        p = int(rng.integers(0, n))
        r = int(rng.integers(0, n))
        if p == r:
            continue
        if p > r:
            p, r = r, p
        pairs.append((p, r))
    rng.shuffle(pairs)
    accepted = 0
    for (p, r) in pairs:
        if p == r:
            continue
        res = pair_line_search(h, q, p, r, n, dx)
        if res is not None:
            h, score, q = res
            accepted += 1
    return h, score, q, accepted

import numpy as np

def proj_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):
        return None
    if np.max(np.abs(z)) > 1e12:
        return None
    n = len(z)
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - np.sum(x)
    if r > 0:
        for i in range(n):
            if r <= 0: break
            a = min(r, 1.0 - x[i]); x[i] += a; r -= a
    elif r < 0:
        for i in range(n):
            if r >= 0: break
            a = min(-r, x[i]); x[i] -= a; r += a
    if abs(np.sum(x) - S) > 1e-10 or not np.all(np.isfinite(x)):
        return None
    return np.clip(x, 0.0, 1.0)

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-9)

def exact_score(h, n):
    vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(vals)):
        return None
    return float(np.max(vals))

def exact_lags(h, n):
    vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(vals)) or len(vals) != 2 * n - 1:
        return None
    return vals

def pair_line_search(h, q, p, r, n, dx):
    L = max(-h[p], h[r] - 1.0)
    U = min(1.0 - h[p], h[r])
    if U - L <= 1e-12:
        return None
    m = 2 * n - 1
    b = np.zeros(m)
    for d in range(-(n - 1), n):
        li = d + n - 1
        v = 0.0
        if 0 <= p + d < n: v += 1.0 - h[p + d]
        if 0 <= r + d < n: v -= 1.0 - h[r + d]
        if 0 <= p - d < n: v -= h[p - d]
        if 0 <= r - d < n: v += h[r - d]
        b[li] = dx * v
    a = np.zeros(m)
    a[n - 1] = -2.0 * dx
    dr = r - p
    a[n - 1 + dr] = dx
    a[n - 1 - dr] = dx
    cur_max = float(np.max(q))
    t_best = 0.0
    f_best = cur_max
    ts = np.linspace(L, U, 21)
    for t in ts:
        f = float(np.max(q + b * t + a * t * t))
        if f < f_best - 1e-15 or (abs(f - f_best) < 1e-15 and abs(t) < abs(t_best)):
            f_best = f; t_best = t
    rad = (U - L) / 20.0
    for _ in range(5):
        lo_r = max(L, t_best - rad)
        hi_r = min(U, t_best + rad)
        ts2 = np.linspace(lo_r, hi_r, 5)
        for t in ts2:
            f = float(np.max(q + b * t + a * t * t))
            if f < f_best - 1e-15 or (abs(f - f_best) < 1e-15 and abs(t) < abs(t_best)):
                f_best = f; t_best = t
        rad *= 0.25
    if f_best >= cur_max - 1e-12:
        return None
    h_new = h.copy()
    h_new[p] += t_best
    h_new[r] -= t_best
    if not is_feasible(h_new, n):
        h_new = proj_simplex(h_new, n / 2.0)
        if h_new is None:
            return None
    sc = exact_score(h_new, n)
    if sc is None or sc >= cur_max - 1e-12:
        return None
    q_new = exact_lags(h_new, n)
    if q_new is None:
        return None
    return h_new, sc, q_new

def build_starts(n, S, parent, rng):
    starts = []
    if parent is not None:
        starts.append(parent.copy())
    idx = np.arange(n, dtype=np.float64)
    u = np.full(n, 0.5)
    M = min(5, (n - 1) // 2)
    for mm in range(1, M + 1):
        A = rng.normal(0.0, 0.15 / np.sqrt(mm))
        phi = rng.uniform(0.0, 6.2832)
        u += A * np.cos(6.2832 * mm * (idx + 0.5) / n + phi)
    pp = proj_simplex(u, S)
    if pp is not None and is_feasible(pp, n):
        starts.append(pp)
    perm = rng.permutation(n)
    binv = np.zeros(n)
    binv[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        binv[perm[n // 2]] = 0.5
    starts.append(binv.copy())
    rv = proj_simplex(rng.random(n), S)
    if rv is not None and is_feasible(rv, n):
        starts.append(rv)
    if parent is not None:
        starts.append(parent[::-1].copy())
        starts.append(1.0 - parent.copy())
    ded = []
    for s in starts:
        s2 = np.asarray(s, dtype=np.float64)
        if not is_feasible(s2, n):
            s2 = proj_simplex(s2, S)
            if s2 is None:
                continue
        dup = False
        for dd in ded:
            if np.max(np.abs(s2 - dd)) < 1e-10:
                dup = True; break
        if not dup:
            ded.append(s2)
    return ded

def polish(h, sc, q, n, dx, rng, P):
    if q is None:
        return h, sc, None, 0
    pairs = []
    n_adj = min(n - 1, max(1, P // 2))
    for i in np.linspace(0, n - 2, n_adj).astype(int):
        pairs.append((int(i), int(i) + 1))
    while len(pairs) < P:
        p = int(rng.integers(0, n)); r = int(rng.integers(0, n))
        if p != r:
            pairs.append((p, r))
    rng.shuffle(pairs)
    acc = 0
    seen = set()
    for (p, r) in pairs:
        key = (min(p, r), max(p, r))
        if key in seen:
            continue
        seen.add(key)
        res = pair_line_search(h, q, p, r, n, dx)
        if res is not None:
            h, sc, q = res
            acc += 1
    return h, sc, q, acc

    if seed is None:

    if n <= 128:
