import numpy as np

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    if r > 1e-14:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
            if r <= 1e-14:
                break
    elif r < -1e-14:
        for i in range(N):
            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1 + 1e-10:
        return np.full(N, 0.5)
    if abs(h.sum() - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / N)

def smooth_grad(h, tau, dx):
    N = len(h)
    c = dx * np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    e = np.exp((c - cmax) / tau)
    se = np.sum(e)
    F = cmax + tau * np.log(se)
    p = e / se
    g = np.zeros(N)
    # lag 0: index N-1 in c
    p0 = p[N - 1]
    g += dx * p0 * (1.0 - 2.0 * h)
    # positive lags s = 1..N-1, index r = N-1+s
    for d in range(1, N):
        pd = p[N - 1 + d]
        if pd > 0:
            g[:N - d] += dx * pd * (1.0 - h[d:])
            g[d:] -= dx * pd * h[:N - d]
        pn = p[N - 1 - d]
        if pn > 0:
            g[d:] += dx * pn * (1.0 - h[:N - d])
            g[:N - d] -= dx * pn * h[d:]
    return F, g

def pg_optimize(h0, N, m, dx, tau_list, max_iter, rng):
    h = h0.copy()
    best = h.copy()
    best_s = score_exact(h, N)
    for tau in tau_list:
        F, g = smooth_grad(h, tau, dx)
        if not np.isfinite(F) or not np.all(np.isfinite(g)):
            break
        alpha = 1.0
        stalls = 0
        for it in range(max_iter):
            y = project_box_sum(h - alpha * g, m)
            d = y - h
            if not np.all(np.isfinite(d)) or np.max(np.abs(d)) <= 1e-9:
                break
            s_y = score_exact(y, N)
            if np.isfinite(s_y) and s_y < best_s:
                best = y.copy()
                best_s = s_y
            F_y, _ = smooth_grad(y, tau, dx)
            gd = np.dot(g, d)
            if np.isfinite(F_y) and gd < 0 and F_y <= F + 1e-4 * gd + 1e-12:
                F_before = F
                h = y
                F = F_y
                alpha = min(1.25 * alpha, 10.0)
                if abs(F_y - F_before) <= 1e-12 * max(1.0, abs(F_before)):
                    stalls += 1
                else:
                    stalls = 0
                if stalls >= 5:
                    break
            else:
                alpha *= 0.5
                if alpha < 1e-10:
                    break
    return best, best_s

def make_fourier(N, rng, n_modes, amp):
    z = np.full(N, 0.5)
    idx = np.arange(N)
    for r in range(1, n_modes + 1):
        A = rng.standard_normal()
        B = rng.standard_normal()
        z += (amp / r) * (A * np.sin(2.0 * np.pi * r * idx / N) + B * np.cos(2.0 * np.pi * r * idx / N))
    return z

def run(seed=42, **kwargs):
    N = None
    m = None
    raw = globals().get('initial_h_values', None)
    if raw is not None:
        raw = np.asarray(raw, dtype=np.float64)
        if raw.ndim == 1 and raw.size >= 2 and np.all(np.isfinite(raw)):
            N = raw.size
            m = N / 2.0
    if N is None:
        N = kwargs.get('n_points', 70)
        if not isinstance(N, (int, np.integer)) or N < 2:
            N = 70
        m = N / 2.0
    m = N / 2.0
    dx = 2.0 / N
    rng = np.random.default_rng(seed)

    if raw is not None and np.all(np.isfinite(raw)):
        base = project_box_sum(raw.copy(), m)
    else:
        base = np.full(N, 0.5)
    best_h = base.copy()
    best_score = score_exact(base, N)

    if N > 999:
        return best_h, best_score, int(N)

    if N <= 128:
        n_starts, max_iter, n_hops = 8, 50, 2
    elif N <= 256:
        n_starts, max_iter, n_hops = 6, 30, 1
    elif N <= 512:
        n_starts, max_iter, n_hops = 4, 16, 0
    else:
        n_starts, max_iter, n_hops = 3, 8, 0

    tau_list = [0.02, 0.006, 0.002, 0.0006, 0.0002, 0.00006, 0.000015]

    # Build starts
    starts = [base.copy()]
    starts.append(project_box_sum(1.0 - base, m))
    perm = base[rng.permutation(N)]
    starts.append(project_box_sum(perm, m))
    for sc in [0.04, 0.12]:
        starts.append(project_box_sum(base + sc * rng.standard_normal(N), m))
    starts.append(project_box_sum(0.5 + 0.5 * rng.uniform(size=N), m))
    starts.append(project_box_sum(make_fourier(N, rng, 5, 0.12), m))
    starts.append(project_box_sum(make_fourier(N, rng, 3, 0.22), m))

    # Deduplicate
    seen = []
    unique_starts = []
    for s in starts[:n_starts]:
        dup = False
        for t in seen:
            if np.max(np.abs(s - t)) <= 1e-10:
                dup = True
                break
        if not dup:
            seen.append(s)
            unique_starts.append(s)

    # Optimize each start
    for h0 in unique_starts:
        s0 = score_exact(h0, N)
        if np.isfinite(s0) and s0 < best_score:
            best_h = h0.copy()
            best_score = s0
        h_opt, s_opt = pg_optimize(h0, N, m, dx, tau_list, max_iter, rng)
        if np.isfinite(s_opt) and s_opt < best_score:
            best_h = h_opt.copy()
            best_score = s_opt

    # Basin hops
    hop_scales = [0.04, 0.015]
    for i in range(min(n_hops, len(hop_scales))):
        hop = project_box_sum(best_h + hop_scales[i] * rng.standard_normal(N), m)
        s_hop = score_exact(hop, N)
        if np.isfinite(s_hop) and s_hop < best_score:
            best_h = hop.copy()
            best_score = s_hop
        h_opt, s_opt = pg_optimize(hop, N, m, dx, tau_list, max_iter, rng)
        if np.isfinite(s_opt) and s_opt < best_score:
            best_h = h_opt.copy()
            best_score = s_opt

    # Final validation
    best_h = project_box_sum(best_h, m)
    c5 = score_exact(best_h, N)
    s_base = score_exact(base, N)
    if not np.isfinite(c5) or c5 > s_base:
        best_h = base
        c5 = s_base
    return best_h.astype(np.float64), float(c5), int(N)

import numpy as np

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    if r > 1e-14:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
            if r <= 1e-14:
                break
    elif r < -1e-14:
        for i in range(N):
            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1 + 1e-10:
        return np.full(N, 0.5)
    if abs(h.sum() - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / N)

def smooth_obj_grad(h, tau, dx):
    N = len(h)
    c = dx * np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    e = np.exp((c - cmax) / tau)
    se = np.sum(e)
    F = cmax + tau * np.log(se)
    p = e / se
    g = np.zeros(N)
    p0 = p[N - 1]
    g += dx * p0 * (1.0 - 2.0 * h)
    for d in range(1, N):
        pd = p[N - 1 + d]
        if pd > 1e-15:
            g[d:] += dx * pd * (1.0 - h[:N - d])
            g[:N - d] -= dx * pd * h[d:]
        pn = p[N - 1 - d]
        if pn > 1e-15:
            g[:N - d] += dx * pn * (1.0 - h[d:])
            g[d:] -= dx * pn * h[:N - d]
    return F, g

def pg_optimize(h0, N, m, dx, tau_list, max_iter, rng):
    h = h0.copy()
    best = h.copy()
    best_s = score_exact(h, N)
    for tau in tau_list:
        F, g = smooth_obj_grad(h, tau, dx)
        if not np.isfinite(F) or not np.all(np.isfinite(g)):
            break
        alpha = 1.0
        g_mom = g.copy()
        stalls = 0
        for it in range(max_iter):
            y = project_box_sum(h - alpha * g_mom, m)
            d = y - h
            if not np.all(np.isfinite(d)) or np.max(np.abs(d)) <= 1e-10:
                break
            s_y = score_exact(y, N)
            if np.isfinite(s_y) and s_y < best_s:
                best = y.copy()
                best_s = s_y
            F_y, g_y = smooth_obj_grad(y, tau, dx)
            gd = np.dot(g_mom, d)
            if np.isfinite(F_y) and gd < -1e-15 and F_y <= F + 1e-4 * gd + 1e-12:
                F_dec = F - F_y
                h = y
                F = F_y
                alpha = min(1.25 * alpha, 8.0)
                g_mom = 0.8 * g_mom + 0.2 * g_y
                if F_dec <= 1e-13 * max(1.0, abs(F)):
                    stalls += 1
                else:
                    stalls = 0
                if stalls >= 6:
                    break
            else:
                alpha *= 0.5
                if alpha < 1e-12:
                    break
    return best, best_s

def make_fourier(N, rng, n_modes, amp):
    z = np.full(N, 0.5)
    idx = np.arange(N)
    for r in range(1, n_modes + 1):
        A = rng.standard_normal()
        B = rng.standard_normal()
        z += (amp / r) * (A * np.sin(2.0 * np.pi * r * idx / N) + B * np.cos(2.0 * np.pi * r * idx / N))
    return z
