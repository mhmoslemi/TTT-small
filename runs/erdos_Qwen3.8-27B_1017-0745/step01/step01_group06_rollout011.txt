        ns_ml, nl, np_, nc, npol = 3, 8, 18, 12, 20
    else:
        ns_ml, nl, np_, nc, npol = 2, 6, 6, 8, 8

    scored = []
    for s in dedup:
        scored.append((exact_score(s, dx), s.copy()))
    scored.sort(key=score_key)
    sel = [x[1] for x in scored[:ns_ml]]

    best_h, best_f = fb_h.copy(), fb_f
    for s in sel:
        try:
            ch, cf = anneal_chain(s, n, S, dx, rng, nl, np_, nc)
            if ch is not None and cf < best_f - 1e-12:
                best_f, best_h = cf, ch.copy()
        except Exception:
            pass

    for sig in [0.02, 0.006]:
        h = best_h.copy()
        f = exact_score(h, dx)
        for _ in range(npol):
            trial = make_proposal(h, n, S, sig, rng)
            if trial is None:
                continue
            fn = exact_score(trial, dx)
            if fn < f - 1e-12:
                h, f = trial, fn
        if f < best_f - 1e-12:
            best_f, best_h = f, h.copy()

    best_h = project_K(best_h, n, S)
    if is_feasible(best_h, n, S):
        fs = exact_score(best_h, dx)
        if fs <= fb_f + 1e-12:
            best_f = fs
        else:
            best_h, best_f = fb_h.copy(), fb_f
    else:
        best_h, best_f = fb_h.copy(), fb_f
    if not is_feasible(fb_h, n, S):
        fb_h, fb_f = np.full(n, S / n), exact_score(np.full(n, S / n), dx)
    final = np.clip(best_h, 0.0, 1.0)
    if not is_feasible(final, n, S):
        final = np.clip(fb_h, 0.0, 1.0)
    return (final, float(exact_score(final, dx)), int(n))
