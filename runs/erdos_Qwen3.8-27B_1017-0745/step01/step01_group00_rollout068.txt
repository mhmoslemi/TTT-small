def project_capped(z,n):
    if z is None: return None
    try: z=np.asarray(z,dtype=np.float64).ravel()
    except: return None
    if z.shape!=(n,): return None
    if not np.all(np.isfinite(z)): return None
    z=np.clip(z,-1e12,1e12)
    s=0.5*n
    lo=np.min(z)-1.0; hi=np.max(z)+1.0
    for _ in range(80):
       mid=0.5*(lo+hi)
       if np.sum(np.clip(z-mid,0,1)) > s: lo=mid
       else: hi=mid
    h=np.clip(z-0.5*(lo+hi),0,1)
    r=s-h.sum()
    if r>0:
       for i in range(n):
          if r<=1e-14: break
          add=min(r,1-h[i]); h[i]+=add; r-=add
    elif r<0:
       for i in range(n):
          if r>=-1e-14: break
          sub=min(-r,h[i]); h[i]-=sub; r+=sub
    if not np.all(np.isfinite(h)) or np.any(h<0) or np.any(h>1): return None
    if abs(h.sum()-s)>1e-10*max(1,n): return None
    return h

def project_capped(z, n):
    if z is None: return None
    try:
        z = np.asarray(z, dtype=np.float64).ravel()
    except Exception:
        return None
    if z.shape != (n,): return None
    if not np.all(np.isfinite(z)): return None
    z = np.clip(z, -1e12, 1e12)
    s = 0.5 * n

    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5*(lo+hi)
        if np.sum(np.clip(z-mid,0.0,1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    r = s - np.sum(h)
    if r > 0:
        for i in range(n):
            if r <= 1e-14: break
            add = min(r, 1.0 - h[i])
            h[i] += add
            r -= add
    elif r < 0:

            h[i] -= sub
            r += sub
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12): return None
    if abs(np.sum(h)-s) > 1e-10*max(1,n): return None

def lag_gradient_matrix(h,n):
    h=np.asarray(h,dtype=np.float64).ravel()
    if h.shape != (n,) or not np.all(np.isfinite(h)): return None
    h=np.clip(h,0.0,1.0)
    L=2*n-1
    G=np.zeros((L,n), dtype=np.float64)
    dx=2.0/n
    start=-(n-1)
    for idx in range(L):
        k=idx+(start)
        m=n-abs(k)
        p0=max(k,0); q0=max(-k,0)
        G[idx,p0:p0+m] += dx*(1.0-h[q0:q0+m])
        G[idx,q0:q0+m] -= dx*h[p0:p0+m]
    if not np.all(np.isfinite(G)): return None
    return G
```

import numpy as np
from scipy.optimize import minimize, differential_evolution


def project_capped(z, n):
    if z is None:
        return None
    try:
        z = np.asarray(z, dtype=np.float64).ravel()
    except Exception:
        return None
    if z.shape != (n,):
        return None
    if not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    s = 0.5 * n
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    if r > 0:
        for i in range(n):
            if r <= 1e-14:
                break
            add = min(r, 1.0 - h[i])
            h[i] += add
            r -= add
    elif r < 0:
        for i in range(n):
            if r >= -1e-14:
                break
            sub = min(-r, h[i])
            h[i] -= sub
            r += sub
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(np.sum(h) - s) > 1e-10 * max(1, n):
        return None
    return np.clip(h, 0.0, 1.0)


def is_feasible(h, n):
    if h is None:
        return False
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - 0.5 * n) <= 1e-10 * max(1, n)


def q_values(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return None
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def exact_score(h, n):
    if not is_feasible(h, n):
        return float("inf")
    q = q_values(h, n)
    if q is None or q.shape != (2 * n - 1,):
        return float("inf")
    if not np.all(np.isfinite(q)):
        return float("inf")
    return float(np.max(q))


def lag_gradient_matrix(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return None
    h = np.clip(h, 0.0, 1.0)
    L = 2 * n - 1
    G = np.zeros((L, n), dtype=np.float64)
    dx = 2.0 / n
    start = -(n - 1)
    for idx in range(L):
        k = idx + start
        m = n - abs(k)
        p0 = max(k, 0)
        q0 = max(-k, 0)
        G[idx, p0:p0 + m] += dx * (1.0 - h[q0:q0 + m])
        G[idx, q0:q0 + m] -= dx * h[p0:p0 + m]
    if not np.all(np.isfinite(G)):
        return None
    return G


def line_best(current, proposal, n):
    current = np.asarray(current, dtype=np.float64).ravel()
    best_z = current.copy()
    best_s = exact_score(current, n)
    if proposal is None:
        return best_z, best_s
    p = np.asarray(proposal, dtype=np.float64).ravel()
    if p.shape != (n,) or not np.all(np.isfinite(p)):
        return best_z, best_s
    for alpha in (1.0, 0.5, 0.25, 0.125, 0.0625):
        v = current + alpha * (p - current)
        z = project_capped(v, n)
        if z is None:
            continue
        s = exact_score(z, n)
        if s < best_s - 1e-12:
            best_s = s
            best_z = z.copy()
    return best_z, best_s


def add_candidate(pool, cand, n):
    c = project_capped(cand, n)
    if c is None:
        return
    for x in pool:
        if np.max(np.abs(x - c)) < 1e-8:
            return
    pool.append(c.copy())


def quantize_beta(z, n):
    try:
        z = np.asarray(z, dtype=np.float64).ravel()
    except Exception:
        return None
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    order = np.argsort(-z)
    h = np.zeros(n, dtype=np.float64)
    m = n // 2
    if m > 0:
        h[order[:m]] = 1.0
    if n % 2 == 1:
        h[order[m]] = 0.5
    return h


def build_starts(base, n, rng, max_starts):
    pool = []
    add_candidate(pool, base, n)
    add_candidate(pool, base[::-1], n)
    add_candidate(pool, 1.0 - base[::-1], n)
    add_candidate(pool, np.full(n, 0.5), n)

    blk = np.zeros(n, dtype=np.float64)
    m = n // 2
    blk[:m] = 1.0
    if n % 2 == 1:
        blk[m] = 0.5
    add_candidate(pool, blk, n)

    alt = (np.arange(n) % 2 == 0).astype(np.float64)
    add_candidate(pool, quantize_beta(alt, n), n)

    for sd in (0.08, 0.25, 0.60):
        add_candidate(pool, base + rng.normal(0.0, sd, n), n)
    for sd in (0.30, 0.80):
        add_candidate(pool, quantize_beta(base + rng.normal(0.0, sd, n), n), n)

    add_candidate(pool, quantize_beta(rng.normal(0.0, 1.0, n), n), n)
    add_candidate(pool, quantize_beta(rng.uniform(0.0, 1.0, n), n), n)
    rh = quantize_beta(rng.normal(0.0, 1.0, n), n)
    if rh is not None:
        add_candidate(pool, quantize_beta(1.0 - rh, n), n)

    while len(pool) > max_starts:
        pool.pop()
    return pool


def smooth_fun(z, n, tau):
    h = np.asarray(z, dtype=np.float64).ravel()
    if h.shape != (n,):
        return 1e100
    h = np.clip(h, 0.0, 1.0)
    q = q_values(h, n)
    if q is None:
        return 1e100
    vals = q / tau
    if not np.all(np.isfinite(vals)):
        return 1e100
    mx = np.max(vals)
    e = np.exp(vals - mx)
    s = np.sum(e)
    if s <= 0.0 or not np.isfinite(s):
        return 1e100
    return float(tau * (mx + np.log(s)))


def smooth_grad(z, n, tau):
    h = np.asarray(z, dtype=np.float64).ravel()
    if h.shape != (n,):
        return np.zeros(n, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    q = q_values(h, n)
    if q is None:
        return np.zeros(n, dtype=np.float64)
    vals = q / tau
    if not np.all(np.isfinite(vals)):
        return np.zeros(n, dtype=np.float64)
    mx = np.max(vals)
    e = np.exp(vals - mx)
    s = np.sum(e)
    if s <= 0.0 or not np.all(np.isfinite(e)):
        return np.zeros(n, dtype=np.float64)
    w = e / s
    G = lag_gradient_matrix(h, n)
    if G is None:
        return np.zeros(n, dtype=np.float64)
    g = G.T @ w
    if not np.all(np.isfinite(g)):
        return np.zeros(n, dtype=np.float64)
    return g


def sum_eq_n(z, n):
    return float(np.sum(z) - 0.5 * n)


def sum_eq_n_jac(z, n):
    return np.ones((1, n), dtype=np.float64)


def smooth_refine(h, n, taus, maxiter):
    cur = project_capped(h, n)
    if cur is None:
        return None, float("inf")
    score = exact_score(cur, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_eq_n, "jac": sum_eq_n_jac, "args": (n,)}]
    for tau in taus:
        try:
            res = minimize(
                smooth_fun,
                cur,
                method="SLSQP",
                jac=smooth_grad,
                bounds=bounds,
                constraints=cons,
                args=(n, tau),
                options={"maxiter": maxiter, "ftol": 1e-10},
            )
            if res is None or res.x is None:
                continue
            rx = np.asarray(res.x, dtype=np.float64).ravel()
            if rx.size >= n:
                y = project_capped(rx[:n], n)
                z, s = line_best(cur, y, n)
                if s < score - 1e-12:
                    cur, score = z, s
        except Exception:
            pass
    cur = project_capped(cur, n)
    if cur is None:
        return None, float("inf")
    return cur, exact_score(cur, n)


def obj_last(z, n):
    return float(np.asarray(z, dtype=np.float64).ravel()[-1])


def obj_last_grad(z, n):
    g = np.zeros(max(1, len(z)), dtype=np.float64)
    g[-1] = 1.0
    return g


def sum_eq_full(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    return float(np.sum(z[:n]) - 0.5 * n)


def sum_eq_full_jac(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    J = np.zeros((1, max(1, len(z))), dtype=np.float64)
    J[0, :n] = 1.0
    return J


def epineq_fun(z, n, indices):
    z = np.asarray(z, dtype=np.float64).ravel()
    h = np.clip(z[:n], 0.0, 1.0)
    q = q_values(h, n)
    if q is None:
        return np.full(len(indices), 1e100, dtype=np.float64)
    return z[-1] - q[indices]


def epineq_jac(z, n, indices):
    z = np.asarray(z, dtype=np.float64).ravel()
    h = np.clip(z[:n], 0.0, 1.0)
    P = len(indices)
    Gs = np.zeros((P, n), dtype=np.float64)
    dx = 2.0 / n
    start = -(n - 1)
    for r, idx in enumerate(indices):
        k = int(idx) + start
        m = n - abs(k)
        p0 = max(k, 0)
        q0 = max(-k, 0)
        Gs[r, p0:p0 + m] += dx * (1.0 - h[q0:q0 + m])
        Gs[r, q0:q0 + m] -= dx * h[p0:p0 + m]
    J = np.empty((P, n + 1), dtype=np.float64)
    J[:, :n] = -Gs
    J[:, n] = 1.0
    if not np.all(np.isfinite(J)):
        return np.zeros((P, n + 1), dtype=np.float64)
    return J


def epigraph_subset(h, n, P, maxiter):
    cur = project_capped(h, n)
    if cur is None:
        return None, float("inf")
    score = exact_score(cur, n)
    q = q_values(cur, n)
    if q is None:
        return cur, score
    P = int(min(max(1, P), 2 * n - 1))
    idx = np.argsort(q)[-P:]
    z0 = np.concatenate((cur, [float(np.max(q)) + 1e-8]))
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    cons = [
        {"type": "ineq", "fun": epineq_fun, "jac": epineq_jac, "args": (n, idx)},
        {"type": "eq", "fun": sum_eq_full, "jac": sum_eq_full_jac, "args": (n,)},
    ]
    try:
        res = minimize(
            obj_last,
            z0,
            method="SLSQP",
            jac=obj_last_grad,
            bounds=bounds,
            constraints=cons,
            args=(n,),
            options={"maxiter": maxiter, "ftol": 1e-10},
        )
        if res is not None and res.x is not None:
            rx = np.asarray(res.x, dtype=np.float64).ravel()
            if rx.size >= n:
                y = project_capped(rx[:n], n)
                z, s = line_best(cur, y, n)
                if s < score - 1e-12:
                    cur, score = z, s
    except Exception:
        pass
    cur = project_capped(cur, n)
    if cur is None:
        return None, float("inf")
    return cur, exact_score(cur, n)


def pair_scan_refine(h, n, passes=1):
    cur = project_capped(h, n)
    if cur is None:
        return None, float("inf")
    best = exact_score(cur, n)
    for _ in range(passes):
        accepted = False
        for i in range(n - 1):
            if accepted:
                break
            for j in range(i + 1, n):
                y = cur.copy()
                y[i], y[j] = cur[j], cur[i]
                sy = exact_score(y, n)
                if sy < best - 1e-12:
                    cur = y.copy()
                    best = sy
                    accepted = True
                    break

                hi = i if cur[i] > cur[j] else j
                lo = j if hi == i else i
                diff = cur[hi] - cur[lo]
                if diff > 0.02:
                    y = cur.copy()
                    delta = 0.5 * diff
                    y[hi] -= delta
                    y[lo] += delta
                    sy = exact_score(y, n)
                    if sy < best - 1e-12:
                        cur = y.copy()
                        best = sy
                        accepted = True
                        break
        if not accepted:
            break
    cur = project_capped(cur, n)
    if cur is None:
        return None, float("inf")
    s = exact_score(cur, n)
    if s < best - 1e-12:
        best = s
    return cur, best


def de_objective(z, n):
    h = project_capped(z, n)
    if h is None:
        return 1e10
    return exact_score(h, n)


def run(seed=42, **kwargs):
    parent_arr = None
    n = 47
    p = None
    try:
        p = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0:
            n = int(p.size)
    except Exception:
        p = None

    if not (1 <= n <= 999):
        n = 47

    if p is not None and p.size == n:
        parent_arr = project_capped(p, n)

    if parent_arr is None:
        parent_arr = np.full(n, 0.5, dtype=np.float64)

    fallback = project_capped(parent_arr, n)
    if fallback is None:
        fallback = np.full(n, 0.5, dtype=np.float64)
    fallback_score = exact_score(fallback, n)

    best = fallback.copy()
    best_score = fallback_score

    try:
        sd = int(seed)
    except Exception:
        sd = 42
    try:
        rng = np.random.default_rng(sd)
    except Exception:
        rng = np.random.default_rng(42)

    de_h = None
    if 2 <= n <= 60:
        try:
            bounds = [(0.0, 1.0)] * n
            resd = differential_evolution(
                de_objective,
                bounds,
                args=(n,),
                seed=sd + 1,
                maxiter=12,
                popsize=10,
                tol=1e-8,
                atol=0.0,
                polish=False,
                init="latinhypercube",
                workers=1,
                x0=best,
            )
            if resd is not None and hasattr(resd, "x"):
                de_h = project_capped(resd.x, n)
                if de_h is not None:
                    ds = exact_score(de_h, n)
                    if ds < best_score - 1e-12:
                        best = de_h.copy()
                        best_score = ds
        except Exception:
            de_h = None

    max_starts = 8 if n <= 80 else 4
    starts = build_starts(fallback, n, rng, max_starts)
    if de_h is not None:
        add_candidate(starts, de_h, n)

    scored = []
    for st in starts:
        scored.append((exact_score(st, n), st))

    score_vals = np.empty(len(scored), dtype=np.float64)
    for j, (sv, _) in enumerate(scored):
        score_vals[j] = sv
    order = np.argsort(score_vals)

    opt_max = 3 if n <= 70 else 1
    opt = []
    for ii in order:
        if len(opt) >= opt_max:
            break
        st = scored[ii][1]
        dup = False
        for o in opt:
            if np.max(np.abs(o - st)) < 1e-8:
                dup = True
                break
        if not dup:
            opt.append(st.copy())

    for oi, st in enumerate(opt):
        h = st.copy()

        if n <= 100:
            hp, sp = pair_scan_refine(h, n, 1)
            if hp is not None and sp < best_score - 1e-12:
                best = hp.copy()
                best_score = sp
            if hp is not None:
                h = hp

        if n <= 60:
            if oi == 0:
                taus = (0.02, 0.005, 0.001)
                pe = 10
                mi = 60
            else:
                taus = (0.02, 0.005)
                pe = 8
                mi = 40

            hs, ss = smooth_refine(h, n, taus, mi)
            if hs is not None and ss < best_score - 1e-12:
                best = hs.copy()
                best_score = ss
            if hs is not None:
                h = hs

            he, se = epigraph_subset(h, n, pe, mi)
            if he is not None and se < best_score - 1e-12:
                best = he.copy()
                best_score = se
            if he is not None:
                h = he

    if n <= 60:
        h = best.copy()

        hf, sf = epigraph_subset(h, n, 16, 80)
        if hf is not None and sf < best_score - 1e-12:
            best = hf.copy()
            best_score = sf
        if hf is not None:
            h = hf

        hf2, sf2 = epigraph_subset(h, n, min(24, 2 * n - 1), 80)
        if hf2 is not None and sf2 < best_score - 1e-12:
            best = hf2.copy()
            best_score = sf2
        if hf2 is not None:
            h = hf2

        hs2, ss2 = smooth_refine(h, n, (0.0003,), 60)
        if hs2 is not None and ss2 < best_score - 1e-12:
            best = hs2.copy()
            best_score = ss2

        hp2, sp2 = pair_scan_refine(best, n, 1)
        if hp2 is not None and sp2 < best_score - 1e-12:
            best = hp2.copy()
            best_score = sp2

    final = project_capped(best, n)
    if final is None or not is_feasible(final, n):
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fallback_score + 1e-12:
        final = fallback.copy()
        fs = fallback_score
    if final is None:
        final = np.full(n, 0.5, dtype=np.float64)
        fs = exact_score(final, n)

    return final, float(fs), int(n)
