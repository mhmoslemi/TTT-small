Wait, I need to be careful about the indexing. Let me reconsider.

For lag d (where d can be negative):
f_d(h) = sum over valid i of h[i] * (1 - h[i+d])

The valid range of i: 0 <= i < n and 0 <= i+d < n
So: max(0, -d) <= i < min(n, n-d)

Gradient:
- d/dh[i] of h[i]*(1-h[i+d]) = (1-h[i+d]) for the i-coordinate
- d/dh[i+d] of h[i]*(1-h[i+d]) = -h[i] for the i+d-coordinate

For d=0: d/dh[i] of h[i]*(1-h[i]) = 1 - 2*h[i]

Now for SLSQP, I need:
- Objective: f(z) = z[-1] (the t variable)
- Objective gradient: [0, 0, ..., 0, 1]
- Equality constraint: sum(h) - n/2 = 0
- Inequality constraints: t - dx * f_d(h) >= 0 for all d

Actually, SLSQP handles inequality constraints as g(z) >= 0 or g(z) <= 0 depending on the type. Let me check: in scipy, for 'ineq' type, the constraint should be g(x) >= 0.

So I need: t - dx * f_d(h) >= 0 for all d.

The Jacobian of the inequality constraint with respect to z = (h, t):
- d/dt (t - dx*f_d) = 1 for all d
- d/dh[i] (t - dx*f_d) = -dx * G[d, i]

def lag_vals_grads(h, n):
    L = 2 * n - 1
    v = np.zeros(L)
    G = np.zeros((L, n))
    dx = 2.0 / n
    for d in range(-(n - 1), n):
        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        if d == 0:
            vals = h[lo:hi] * (1.0 - h[lo:hi])
            v[row] = float(np.sum(vals))
            G[row, lo:hi] = 1.0 - 2.0 * h[lo:hi]
        else:
            a = h[lo:hi]
            b = h[lo + d:hi + d]
            v[row] = float(np.sum(a * (1.0 - b)))
            G[row, lo:hi] += (1.0 - b)
            G[row, lo + d:hi + d] -= a
    return v, G

def obj_func(z, n):
    return z[-1]

def obj_grad(z, n):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def eq_constraint(z, n):
    return float(np.sum(z[:n]) - n / 2.0)

def eq_jac(z, n):
    J = np.zeros(n)
    J[:] = 1.0
    return J

def ineq_constraint(z, n):
    h = z[:n]
    t = z[-1]
    dx = 2.0 / n
    v, _ = lag_vals_grads(h, n)
    return t - dx * v

def ineq_jac(z, n):
    h = z[:n]
    dx = 2.0 / n
    _, G = lag_vals_grads(h, n)
    L = 2 * n - 1
    J = np.zeros((L, n + 1))
    J[:, :n] = -dx * G
    J[:, -1] = 1.0
    return J

def add_start(starts, v, n):
    h2 = project_capped(v, n)
    if h2 is None:
        return
    for ex in starts:
        if float(np.max(np.abs(h2 - ex))) < 1e-8:
            return
    starts.append(h2)

def polish(h, n, sweeps):
    dx = 2.0 / n
    deltas = [0.0, 0.001, -0.001, 0.003, -0.003, 0.01, -0.01, 0.03, -0.03, 0.1, -0.1, 0.3, -0.3]
    for _ in range(sweeps):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                lo = max(-h[i], h[j] - 1.0)
                hi = min(1.0 - h[i], h[j])
                best_sc = exact_score(h, n)
                best_d = 0.0
                trials = set()
                trials.add(lo)
                trials.add(hi)
                for dd in deltas:
                    t = dd
                    if lo < t < hi:
                        trials.add(t)
                for dd in sorted(trials):
                    if dd < lo - 1e-12 or dd > hi + 1e-12:

                    h[j] -= dd
                    sc = exact_score(h, h.shape[0])

                    h[j] += dd

                        best_d = dd
                if abs(best_d) > 1e-14:
                    h[i] += best_d
                    h[j] -= best_d
                    improved = True
        if not improved:
