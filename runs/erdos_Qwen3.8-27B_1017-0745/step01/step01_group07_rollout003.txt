        hi = min(n, n - d)
        for i in range(lo, hi):
            rows.append(row); cols.append(i); data.append(dx * (1.0 - h[i + d]))
            rows.append(row); cols.append(i + d); data.append(-dx * h[i])
    G = coo_matrix((data, (rows, cols)), shape=(m, n)).tocsr()
    if not np.all(np.isfinite(G.data)):
        return None, None
    return q, G

def solve_lp_step(x, q, G, rho, n, mu):
    m = 2 * n - 1
    var_n = 2 * n + 1
    c = np.zeros(var_n)
    c[n:2 * n] = mu
    c[2 * n] = 1.0
    aub_r, aub_c, aub_d = [], [], []
    Gc = G.tocoo()
    for r, cc, v in zip(Gc.row, Gc.col, Gc.data):
        aub_r.append(r); aub_c.append(cc); aub_d.append(v)
    for r in range(m):
        aub_r.append(r); aub_c.append(2 * n); aub_d.append(-1.0)
    for i in range(n):
        aub_r.append(m + i); aub_c.append(i); aub_d.append(1.0)
        aub_r.append(m + i); aub_c.append(n + i); aub_d.append(-1.0)
        aub_r.append(m + n + i); aub_c.append(i); aub_d.append(-1.0)
        aub_r.append(m + n + i); aub_c.append(n + i); aub_d.append(-1.0)
    A_ub = coo_matrix((aub_d, (aub_r, aub_c)), shape=(m + 2 * n, var_n)).tocsr()
    b_ub = np.concatenate([-q, np.zeros(n), np.zeros(n)])
    A_eq = np.zeros((1, var_n)); A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        bounds.append((max(-rho, -x[i]), min(rho, 1.0 - x[i])))
    for i in range(n):
        bounds.append((0.0, None))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 15.0})
        if res is not None and res.status == 0 and res.x is not None:
            xv = res.x
            if xv.shape == (var_n,) and np.all(np.isfinite(xv)):
                delta = xv[:n]
                if abs(np.sum(delta)) <= 1e-5:
                    return delta
    except Exception:
        pass
    return None

def optimize_trajectory(x0, n, rng, best_h, best_score, max_iter):
    x = x0.copy()
    f = exact_score(x, n)
    if f is None:
        return best_h, best_score
    if f < best_score - 1e-12:
        best_h = x.copy(); best_score = f
    rho, rho_max, rho_min = 0.08, 0.25, 2e-4
    mu = 0.002 * (2.0 / n)
    consec_fail, stall = 0, 0
    for _ in range(max_iter):
        if rho < rho_min or consec_fail >= 4:
            break
        q, G = build_lag_model(x, n)
        if q is None or G is None:
            break
        delta = solve_lp_step(x, q, G, rho, n, mu)
        if delta is None:
            rho = max(rho_min, rho * 0.5); consec_fail += 1
            continue
        if np.max(np.abs(delta)) < 1e-9:
            break
        best_local, best_local_f = None, f
        for j in range(9):
            beta = 2.0 ** (-j)
            cand = x + beta * delta
            pr = proj_simplex(cand, n / 2.0)
            if pr is None or not is_feasible(pr, n):
                continue
            sc = exact_score(pr, n)
            if sc is None:
                continue
            if sc < best_local_f - 1e-12:
                best_local_f = sc; best_local = pr.copy()
        if best_local is None:
            rho = max(rho_min, rho * 0.5); consec_fail += 1
            continue
        s = best_local - x
        pred = float(np.max(q + G.dot(s)))
        act = f - best_local_f
        if pred <= 1e-12 or (act / pred) < 0.25:
            rho = max(rho_min, rho * 0.5)
        elif (act / pred) > 0.75 and np.max(np.abs(s)) >= 0.8 * rho:
            rho = min(rho_max, 1.5 * rho)
        x = best_local; f = best_local_f; consec_fail = 0
        if f < best_score - 1e-12:
            best_h = x.copy(); best_score = f
        if np.max(np.abs(s)) < 1e-8 or (act <= 1e-10 and np.max(np.abs(s)) <= 1e-5):
            stall += 1
            if stall >= 3: break
        else:
            stall = 0
    return best_h, best_score

def build_starts(n, S, parent, rng):
    starts = [np.full(n, 0.5)]
    has_p = False
    if parent is not None:
        has_p = True
        starts.append(parent.copy())
        for sg in [0.04, 0.12]:
            p = proj_simplex(parent + rng.normal(0.0, sg, n), S)
            if p is not None: starts.append(p)
        starts.append(parent[::-1].copy())
    if n <= 256:
        M = min(5, (n - 1) // 2)
        idx = np.arange(n, dtype=np.float64)
        u = np.full(n, 0.5)
        for mm in range(1, M + 1):
            amp = rng.normal(0.0, 0.10 / math.sqrt(mm))
            phi = rng.uniform(0.0, 2.0 * np.pi)
            u += amp * np.cos(2.0 * np.pi * mm * (idx + 0.5) / n + phi)
        pp = proj_simplex(u, S)
        if pp is not None: starts.append(pp)
    if not has_p:
        pu = proj_simplex(rng.uniform(0.0, 1.0, n), S)
        if pu is not None: starts.append(pu)
        perm = rng.permutation(n)
        bv = np.zeros(n); bv[perm[:n // 2]] = 1.0
        if n % 2 == 1: bv[perm[n // 2]] = 0.5
        starts.append(bv)
    deduped = []
    for s in starts:
        dup = False
        for d in deduped:
            if np.max(np.abs(s - d)) < 1e-9: dup = True; break
        if not dup and is_feasible(s, n):
            deduped.append(s)
    return deduped

def run(seed=42, **kwargs):
    if seed is None: seed = 42
    try: seed = int(seed)
    except (TypeError, ValueError): seed = 42
    rng = np.random.default_rng(seed)
    n = None; parent = None
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        try:
            iv_arr = np.asarray(iv, dtype=np.float64)
            if iv_arr.ndim == 1 and 8 <= iv_arr.size <= 999:
                n = iv_arr.size
                if np.all(np.isfinite(iv_arr)) and np.max(np.abs(iv_arr)) <= 1e12:
                    pr = proj_simplex(iv_arr, n / 2.0)
                    if pr is not None and is_feasible(pr, n):
                        parent = pr
        except Exception:
            pass
    if n is None:
        try: n = int(kwargs.get('n_points', 80))
        except (TypeError, ValueError): n = 80
        if n < 8 or n > 999: n = 80
    S = n / 2.0
    const_fb = np.full(n, 0.5)
    best_h = const_fb.copy()
    best_score = exact_score(const_fb, n)
    if best_score is None or not np.isfinite(best_score):
        best_score = 0.5
    parent_valid = None
    if parent is not None:
        parent_valid = parent.copy()
        ps = exact_score(parent, n)
        if ps is not None and np.isfinite(ps) and ps < best_score:
            best_score = ps; best_h = parent.copy()
    starts = build_starts(n, S, parent, rng)
    for s in starts:
        sc = exact_score(s, n)
        if sc is not None and np.isfinite(sc) and sc < best_score:
            best_score = sc; best_h = s.copy()
    if n <= 128: max_iter, n_st, n_re = 30, 6, 2
    elif n <= 256: max_iter, n_st, n_re = 22, 4, 1
    elif n <= 512: max_iter, n_st, n_re = 14, 3, 0
    else: max_iter, n_st, n_re = 6, 2, 0
    for s in starts[:n_st]:
        best_h, best_score = optimize_trajectory(s, n, rng, best_h, best_score, max_iter)
    sigma = 0.05
    for _ in range(n_re):
        pert = best_h + sigma * rng.normal(0.0, 1.0, n)
        pr = proj_simplex(pert, S)
        if pr is None or not is_feasible(pr, n) or np.max(np.abs(pr - best_h)) < 1e-9:
            sigma = min(0.18, 1.8 * sigma); continue
        old = best_score
        best_h, best_score = optimize_trajectory(pr, n, rng, best_h, best_score, max_iter)
        sigma = max(0.02, 0.5 * sigma) if best_score < old - 1e-12 else min(0.18, 1.8 * sigma)
    if not is_feasible(best_h, n):
        rep = proj_simplex(best_h, S)
        if rep is not None and is_feasible(rep, n): best_h = rep
    fs = exact_score(best_h, n)
    if fs is None or not np.isfinite(fs):
        if parent_valid is not None:
            best_h = parent_valid.copy()
            fs = exact_score(best_h, n)
        if fs is None or not np.isfinite(fs):
            best_h = const_fb.copy()
            fs = exact_score(const_fb, n)
            if fs is None or not np.isfinite(fs): fs = 0.5
    return best_h, float(fs), int(n)
