            N += 1e-10 * I
            Lp = np.linalg.cholesky(P)
            Ln = np.linalg.cholesky(N)
        cp_vec = np.zeros(n)
        cm_vec = np.zeros(n)
        if d == 0:
            cp_vec[:] = dx
            cm_vec[:] = dx
        else:
            cp_vec[d:] = dx
            cm_vec[:n - d] = dx
        data.append((Lp, Ln, cp_vec, cm_vec))
    return data

    S_global = None

    init = globals().get('initial_h_values', None)
    ok = False
    if init is not None:
        try:
            init = np.array(init, dtype=np.float64).ravel()
            if 2 <= len(init) <= 999 and np.all(np.isfinite(init)):
                if init.min() >= -1e-8 and init.max() <= 1.0 + 1e-8:
                    n = len(init)
                    S = n / 2.0
                    if abs(init.sum() - S) <= 1e-6:
                        ok = True
        except Exception:
            pass
    if not ok:
        nd = kwargs.get('n_points', 45)
        if isinstance(nd, (int, np.integer)) and not isinstance(nd, bool) and 2 <= nd <= 999:
            n = int(nd)
        else:
            n = 45
        S = n / 2.0
        init = np.full(n, 0.5)
    dx = 2.0 / n
    base = project_capped_simplex(init.copy(), n, S)
    fb_h, fb_s = base.copy(), score_exact(base, n)
    best_h, best_s = base.copy(), fb_s
    
    # Generate pool
    pool = [base.copy()]
    pool.append((1.0 - base).copy())
    pool.append(base[::-1].copy())
    pool.append((1.0 - base[::-1]).copy())
    for sig in [0.04, 0.12, 0.25]:
        pool.append(project_capped_simplex(base + sig * rng.standard_normal(n), n, S))
    pool.append(project_capped_simplex(rng.uniform(0, 1, n), n, S))
    pool.append(project_capped_simplex(rng.beta(0.5, 0.5, n), n, S))

    blk = np.zeros(n)
    fk = int(np.floor(S))

    rem = S - fk
    if rem > 1e-9 and fk < n:
        blk[perm[fk]] = rem
    pool.append(project_capped_simplex(blk, n, S))
    # Sinusoid
    j = np.arange(n)
    phase = rng.uniform(0, 2 * math.pi)

                dup = True
                break
