    g = np.zeros(n)
    for idx, d in enumerate(range(-(n - 1), n)):
        ad = max(d, 0)
        bd = max(-d, 0)
        md = n - abs(d)
        wd = w[idx]
        if wd < 1e-16:
            continue
        g[ad:ad + md] += wd * dx * (1.0 - h[bd:bd + md])
        g[bd:bd + md] -= wd * dx * h[ad:ad + md]
    if not np.all(np.isfinite(g)):
        return 1e10, np.zeros(n)
    return F, g

def mass_fun(x, n):
    return np.sum(x) - n / 2.0

def mass_jac(x, n):
    return np.ones(n)

def build_starts(fb, n, rng, count):
    pool = [fb.copy()]
    cands = [fb[::-1].copy(), (1.0 - fb[::-1]).copy(), np.full(n, 0.5)]
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    cands.append(blk)
    cands.append(np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64))
    cands.append(fb + rng.normal(0, 0.10, n))
    cands.append(fb + rng.normal(0, 0.25, n))
    cands.append(rng.uniform(0, 1, n))
    for c in cands:
        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-8:
                dup = True
                break
        if not dup:
            pool.append(p)
    return pool[:count]

def continuation_opt(w0, n, taus, maxiter):
    h = w0.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': partial(mass_fun, n=n), 'jac': partial(mass_jac, n=n)}]
    best_local = h.copy()
    best_local_s = exact_score(h, n)
    for tau in taus:
        try:
            obj = partial(smooth_objgrad, tau=tau, n=n)
            res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                           jac=True, options={'maxiter': maxiter, 'ftol': 1e-10, 'disp': False})
            if res.x is not None and res.x.shape == (n,) and np.all(np.isfinite(res.x)):
                proj = project_capped(res.x, n)
                if proj is not None and is_feasible(proj, n):
                    s = exact_score(proj, n)
                    if s < best_local_s:
                        best_local_s, best_local = s, proj.copy()
                    f0 = smooth_objgrad(h, tau, n)[0]
                    best_c, best_fc = h, f0
                    for a in [1.0, 0.5, 0.25, 0.125]:
                        z = project_capped(h + a * (proj - h), n)
                        if z is not None and is_feasible(z, n):
                            sz = exact_score(z, n)
                            if sz < best_local_s:
                                best_local_s, best_local = sz, z.copy()
                            fz = smooth_objgrad(z, tau, n)[0]
                            if fz < best_fc:
                                best_c, best_fc = z, fz
                    if best_fc < f0 - 1e-12:
                        h = best_c
        except Exception:
            pass
    return best_local, best_local_s

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    fb = project_capped(parent, n) if parent is not None else None
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)

    if n <= 64:
        n_starts, n_iter, n_kicks = 8, 90, 4
    elif n <= 128:
        n_starts, n_iter, n_kicks = 5, 60, 2
    else:
        n_starts, n_iter, n_kicks = 1, 30, 0

    taus = [0.03, 0.01, 0.003, 0.001, 0.0003]
    starts = build_starts(fb, n, rng, n_starts)

    gb, gs = fb.copy(), exact_score(fb, n)
    for st in starts:
        s0 = exact_score(st, n)
        if s0 < gs - 1e-12:
            gs, gb = s0, st.copy()

    for st in starts:
        xo, so = continuation_opt(st, n, taus, n_iter)
        if so < gs - 1e-12:
            gs, gb = so, xo.copy()

    if n_kicks > 0:
        sigma = 0.08
        for _ in range(n_kicks):
            kick = project_capped(gb + sigma * rng.normal(0, 1, n), n)
            if kick is not None and is_feasible(kick, n):
                xo, so = continuation_opt(kick, n, taus, n_iter)
                if so < gs - 1e-12:
                    gs, gb = so, xo.copy()
                    sigma = 0.08
                else:
                    sigma = min(0.35, 1.7 * sigma)

    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb
    if exact_score(final, n) > exact_score(fb, n):
        final = fb
    return final.astype(np.float64), float(exact_score(final, n)), int(n)
