            sc = exact_score(proj, n)
            if not np.isfinite(sc): continue
            if sc < ss[i] - 1e-14 * max(1.0, abs(ss[i])):
                npop[i], nsco[i] = proj, sc
                archive.append((snap[i].copy(), ss[i]))
                if sc < best_c5:
                    best_c5, best_h = sc, proj.copy()
                sF.append(Fi); sCR.append(CRi)
        pop, scores = npop, nsco
        while len(archive) > P:
            archive.pop(rng.integers(len(archive)))
        if sF:
            mu_F = min(max(0.9 * mu_F + 0.1 * sum(f * f for f in sF) / sum(sF), 0.05), 1.0)
        if sCR:
            mu_CR = min(max(0.9 * mu_CR + 0.1 * (sum(sCR) / len(sCR)), 0.0), 1.0)
        saved = ss[order[0]]
        if best_c5 < saved - 1e-12 * max(1.0, abs(saved)):
            no_imp = 0
        else:
            no_imp += 1
        if no_imp >= stag:
            if restarts >= mxr: break
            restarts += 1; no_imp = 0; archive = []
            mu_F, mu_CR = 0.55, 0.70
            si = np.argsort(scores)
            elig, elites = min(2, P // 4), []
            for k in range(len(si)):
                if len(elites) >= elig: break
                c = pop[si[k]].copy()
                if all(is_distinct(c, e) for e in elites):
                    elites.append(c)
            sig = [0.04, 0.12, 0.30][restarts - 1]
            nnew = elites[:]
            for _ in range(int(0.7 * (P - len(nnew)))):
                c = project_box_sum(best_h + sig * rng.standard_normal(n), S)
                if c is not None and check_feasible(c, n) and all(is_distinct(c, e) for e in nnew):
                    nnew.append(c)
            while len(nnew) < P:
                c = project_box_sum(rng.random(n), S)
                nnew.append(c if c is not None and check_feasible(c, n) else best_h.copy())
            pop = nnew[:P]
            scores = [exact_score(p, n) for p in pop]
            for i in range(P):
                if not np.isfinite(scores[i]):
                    pop[i], scores[i] = best_h.copy(), exact_score(best_h, n)
    best_h, best_c5 = polish(best_h, n, rng)
    fh = project_box_sum(best_h, S)
    if fh is None or not check_feasible(fh, n):
        fh, n = parent.copy(), len(parent)
    fc5 = exact_score(fh, n)
    if fc5 > exact_score(parent, len(parent)) + 1e-12:
        fh, fc5, n = parent.copy(), exact_score(parent, len(parent)), len(parent)
    return (fh, fc5, int(n))
