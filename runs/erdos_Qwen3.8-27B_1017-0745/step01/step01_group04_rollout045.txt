    if r > 0:

def exact_score(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    c = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(c)):
        return np.inf
    return float(np.max(c) * 2.0 / N)

def lag_data(h, N):
    L = 2 * N - 1
    q = np.zeros(L)
    G = np.zeros((L, N))
    for s in range(-(N - 1), N):
        row = s + N - 1
        if s >= 0:
            ell = N - s
            q[row] = np.dot(h[:ell], 1.0 - h[s:s + ell])
            G[row, :ell] += 1.0 - h[s:s + ell]
            G[row, s:N] -= h[:ell]
        else:
            d = -s
            ell = N - d
            q[row] = np.dot(h[d:N], 1.0 - h[:ell])
            G[row, d:N] += 1.0 - h[:ell]
            G[row, :ell] -= h[d:N]
    return q, G

    return d @ d + z[N]

def make_fourier(N, rng, n_modes, amp):
    z = np.full(N, 0.5)
    ns = min(n_modes, (N - 1) // 2)
    idx = np.arange(N, dtype=np.float64)
    for r in range(1, ns + 1):

        z += (amp / r) * (A * np.sin(2.0 * np.pi * r * idx / N) + B * np.cos(2.0 * np.pi * r * idx / N))
    return z

    starts.append(project_box_sum(base[::-1].copy(), m))
    sh = max(1, N // 3)
    starts.append(project_box_sum(np.roll(base, sh), m))
    for sc in [0.05, 0.15, 0.30]:
        starts.append(project_box_sum(base + sc * rng.standard_normal(N), m))
    starts.append(project_box_sum(rng.uniform(size=N), m))
    starts.append(project_box_sum(make_fourier(N, rng, 5, 0.12), m))
    starts.append(project_box_sum(make_fourier(N, rng, 4, 0.20), m))
    deduped = []
    for s in starts:
        if not np.all(np.isfinite(s)):
            continue
        unique = True
        for d in deduped:
            if np.max(np.abs(s - d)) <= 1e-10:
                unique = False; break
        if unique:
            deduped.append(s)
        if len(deduped) >= cap:
            break
    return deduped

def run(seed=42, **kwargs):
    N = None; m = None
    raw = globals().get('initial_h_values', None)
    if raw is not None:
        try:
            raw = np.asarray(raw, dtype=np.float64)
            if raw.ndim == 1 and raw.size > 1 and np.all(np.isfinite(raw)):
                N = raw.size; m = N / 2.0
        except Exception:
            raw = None
    if N is None:
        N = kwargs.get('n_points', 70)
        if not isinstance(N, (int, np.integer)) or isinstance(N, bool) or N < 2 or N > 999:
            N = 70
        m = N / 2.0
    if N == 1:
        h = np.array([0.5], dtype=np.float64)
        return h, exact_score(h, 1), 1

    if raw is not None and N is not None and raw.size == N:
        base = np.clip(raw, 0.0, 1.0)
        base = project_box_sum(base, m)
    else:
        base = np.full(N, 0.5)
    base = project_box_sum(base, m)
    global_best = base.copy()
    global_sc = exact_score(global_best, N)
    if not np.isfinite(global_sc):
        global_best = np.full(N, 0.5)
        global_sc = exact_score(global_best, N)

    if N > 256:
        return global_best, float(global_sc), int(N)

    rng = np.random.default_rng(seed)

    if N <= 96:
        n_starts = 8; mm_iters = 40
        restarts = [(0.05, 24), (0.015, 24)]
    elif N <= 160:
        n_starts = 5; mm_iters = 25
        restarts = [(0.04, 16)]
    else:
        n_starts = 3; mm_iters = 12
        restarts = []

    starts = build_starts(global_best, N, m, rng, n_starts)

    for h0 in starts:
        sc0 = exact_score(h0, N)
        if np.isfinite(sc0) and sc0 < global_sc - 1e-12:
            global_best = h0.copy(); global_sc = sc0
        try:
            h_opt, sc_opt = mm_local(h0, N, m, mm_iters)
            if np.isfinite(sc_opt) and sc_opt < global_sc - 1e-12:
                global_best = h_opt copy if False else h_opt.copy()
                global_sc = sc_opt
        except Exception:
            pass

    for (sigma, iters) in restarts:
        try:
            hop = project_box_sum(global_best + sigma * rng.standard_normal(N), m)
            sc_hop = exact_score(hop, N)
            if np.isfinite(sc_hop) and sc_hop < global_sc - 1e-12:
                global_best = hop.copy(); global_sc = sc_hop
            h_opt, sc_opt = mm_local(hop, N, m, iters)
            if np.isfinite(sc_opt) and sc_opt < global_sc - 1e-12:
                global_best = h_opt.copy(); global_sc = sc_opt
        except Exception:
            pass

    global_best = project_box_sum(global_best, m)
    final_sc = exact_score(global_best, N)
    if not np.isfinite(final_sc) or final_sc > global_sc + 1e-12:
        global_best = project_box_sum(base, m)
        final_sc = exact_score(global_best, N)
    if not np.isfinite(final_sc):
        global_best = np.full(N, 0.5)
        final_sc = exact_score(global_best, N)
    return global_best.astype(np.float64), float(final_sc), int(N)
