    best_h = pop[bi].copy()
    best_score = float(scores[bi])
    if fallback_score < best_score:
        best_h = fallback.copy()
        best_score = fallback_score

    R = max(1, P // 4)
    E_budget = max(2 * P + 3 * R + 3, min(5000, int(math.ceil(8e7 / (n * n)))))
    G = max(1, (E_budget - P - 3 * R) // P)
    max_restarts = 3
    restarts = 0
    no_improve = 0
    prev_best = best_score
    P_div = 0.0
    n_pbest = max(2, math.ceil(0.2 * P))

    for gen in range(G):
        new_pop = pop.copy()
        new_scores = scores.copy()
        new_Fs = Fs.copy()
        new_CRs = CRs.copy()
        use_pbest = gen >= max(1, (2 * G) // 3)

        for i in range(P):
            fi = Fs[i]
            cri = CRs[i]
            if rng.random() < 0.1:
                fi = 0.1 + 0.9 * rng.random()
            if rng.random() < 0.1:
                cri = rng.random()
            new_Fs[i] = fi
            new_CRs[i] = cri

            if use_pbest and rng.random() < 0.75:
                idx_order = np.argsort(scores)[:n_pbest]
                pbest_idx = int(idx_order[rng.integers(n_pbest)])
                others = [j for j in range(P) if j != i and j != pbest_idx]
                r1 = others[int(rng.integers(len(others)))]
                others.remove(r1)
                r2 = others[int(rng.integers(len(others)))]
                v = pop[i] + fi * (pop[pbest_idx] - pop[i]) + fi * (pop[r1] - pop[r2])
            else:
                others = [j for j in range(P) if j != i]
                r1 = others[int(rng.integers(len(others)))]
                others.remove(r1)
                r2 = others[int(rng.integers(len(others)))]
                others.remove(r2)
                r3 = others[int(rng.integers(len(others)))]
                v = pop[r1] + fi * (pop[r2] - pop[r3])

            trial = pop[i].copy()
            jrand = int(rng.integers(n))
            for j in range(n):
                if j == jrand or rng.random() < cri:
                    trial[j] = v[j]

            trial = project_box_sum(trial, S)
            if trial is None or not check_feasible(trial, n):
                continue
            if np.max(np.abs(trial - pop[i])) < 1e-14:
                continue
            sc = exact_score(trial, n)
            if not np.isfinite(sc):
                continue
            if sc < new_scores[i] - 1e-14:
                new_pop[i] = trial
                new_scores[i] = sc
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = trial.copy()

        pop = new_pop
        scores = new_scores
        Fs = new_Fs
        CRs = new_CRs

        if P > 1:
            P_div = math.sqrt(float(np.mean(np.var(pop, axis=0))))
        else:
            P_div = 0.0

        if best_score < prev_best - 1e-12:
            no_improve = 0
        else:
            no_improve += 1
        prev_best = best_score

        if (no_improve >= 12 or P_div < 1e-3) and restarts < max_restarts:
            worst_idx = np.argsort(scores)[-R:]
            sigmas = [0.03, 0.10, 0.25]
            for k, wi in enumerate(worst_idx):
                if k % 4 == 3:
                    cand = rng.random(n)
                else:
                    cand = best_h + sigmas[k % 3] * rng.standard_normal(n)
                cand = project_box_sum(cand, S)
                if cand is not None and check_feasible(cand, n):
                    sc = exact_score(cand, n)
                    if np.isfinite(sc):
                        pop[wi] = cand
                        scores[wi] = sc
                        Fs[wi] = 0.5
                        CRs[wi] = 0.9
                        if sc < best_score - 1e-14:
                            best_score = sc
                            best_h = cand.copy()
            restarts += 1
            no_improve = 0
            continue

        if restarts >= max_restarts and no_improve >= 20 and P_div < 1e-5:
            break

    final_h = project_box_sum(best_h, S)
    if final_h is None or not check_feasible(final_h, n):
        final_h = fallback.copy()
    final_score = exact_score(final_h, n)
    if not np.isfinite(final_score) or final_score > fallback_score:
        final_h = fallback.copy()
        final_score = fallback_score
    return (np.array(final_h, dtype=np.float64), float(final_score), int(n))
