def build_edges(n):
    edges, seen = [], set()
    for i in range(n - 1):
        edges.append((i, i + 1))
    r = 2
    while r < n:
        for i in range(0, n - r, 2 * r):
            p, q = i, i + r
            if (p, q) not in seen:
                edges.append((p, q))
                seen.add((p, q))
        r *= 2
    if n > 2 and (0, n - 1) not in seen:
        edges.append((0, n - 1))
    return edges
