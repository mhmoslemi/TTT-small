    if not check_feasible(h, n): return float('inf')

    try: vals = np.correlate(hh, 1.0 - hh, mode='full')

def final_adjust_sum(h):

    if n == 0: return h
    S = 0.5 * n

        return None
    if np.any(trial < 0.0) or np.any(trial > 1.0) or not check_mass(trial, n, 0.5 * n):
        rep = project_box_sum(trial, 0.5 * n)
        if rep is None or float(np.max(np.abs(rep - h))) > 1e-7:
            return None
        trial = rep
    if not check_mass(trial, n, 0.5 * n):
        return None
    return trial
def try_add_start(cands, seen, vec, n):
    p = project_box_sum(vec, 0.5 * n)
    if p is None:
        return
    for q in seen:
        if float(np.max(np.abs(p - q))) < 1e-10:
            return
    seen.append(p)
    cands.append(p)
def build_starts(parent, n, rng):
    cands = []
    seen = []
    p = project_box_sum(parent, 0.5 * n)
    if p is not None:
        cands.append(p)
        seen.append(p)
    else:
        cands.append(parent.copy())
        seen.append(parent.copy())
    try_add_start(cands, seen, 1.0 - parent, n)
    try_add_start(cands, seen, parent[::-1].copy(), n)
    try_add_start(cands, seen, np.roll(parent, 1), n)
    for sd in (0.05, 0.12, 0.25):
        try_add_start(cands, seen, parent + sd * rng.standard_normal(n), n)
    try_add_start(cands, seen, rng.random(n), n)
    i = np.arange(n)
    ph = 2.0 * math.pi * float(rng.random())
    th = 2.0 * math.pi * float(rng.random())
    h1 = 0.5 + 0.28 * np.cos(2.0 * math.pi * i / n + ph) + 0.12 * np.sin(4.0 * math.pi * i / n + th)
    try_add_start(cands, seen, h1, n)
    ph = 2.0 * math.pi * float(rng.random())
    th = 2.0 * math.pi * float(rng.random())
    h2 = 0.5 + 0.32 * np.sin(2.0 * math.pi * 3.0 * i / n + ph) + 0.1 * np.cos(2.0 * math.pi * 5.0 * i / n + th)
    try_add_start(cands, seen, h2, n)
    v = parent.copy()
    b = n // 3
    v[:b] += 0.25
    v[b:2 * b] -= 0.25
    v[2 * b:] += 0.12
    try_add_start(cands, seen, v, n)
    return cands
def calibrate_temperature(start, n, rng, sigma0):
    cur = start.copy()
    cs = exact_score(cur, n)
    best = start.copy()
    bs = cs
    gains = []
    if np.isfinite(cs):
        for _ in range(24):
            trial = propose_feasible(cur, n, sigma0, rng)
            if trial is None:
                continue
            ts = exact_score(trial, n)
            if not np.isfinite(ts):
                continue
            if ts < bs - 1e-15:
                best = trial.copy()
                bs = ts
            dl = ts - cs
            if dl > 0.0:
                gains.append(dl)
            else:
                cur = trial.copy()
                cs = ts
    if len(gains) > 0:
        gains.sort()
        med = gains[len(gains) // 2]
        T = 1e-3 if med <= 1e-15 else -med / math.log(0.8)
        if T < 2e-5:
            T = 2e-5
        if T > 2e-2:
            T = 2e-2
    else:
        T = 1e-3
    return T, best, bs
def anneal_cycle(start, n, rng):
    if n <= 64:
        levels, props, sig0 = 10, 400, 0.18
    elif n <= 256:
        levels, props, sig0 = 8, 200, 0.13
    else:
        levels, props, sig0 = 5, 48, 0.08
    cur = start.copy()
    cs = exact_score(cur, n)
    best = start.copy()
    bs = cs
    T0, bcal, bcal_s = calibrate_temperature(start, n, rng, sig0)
    if np.isfinite(bcal_s) and bcal_s < bs - 1e-15:
        best = bcal.copy()
        bs = bcal_s
    sigma = sig0
    for lev in range(levels):
        T = T0 * (0.55 ** lev)
        if T < 1e-7:
            T = 1e-7
        acc = 0
        cnt = 0
        for _ in range(props):
            trial = propose_feasible(cur, n, sigma, rng)
            cnt += 1
            if trial is None:
                continue
            ts = exact_score(trial, n)
            if not np.isfinite(ts):
                continue
            if ts < bs - 1e-15:
                best = trial.copy()
                bs = ts
            dl = ts - cs
            if dl <= 0.0:
                acc += 1
                cur = trial.copy()
                cs = ts
            else:
                r = dl / T
                if r < 745.0 and rng.random() < math.exp(-r):
                    acc += 1
                    cur = trial.copy()
                    cs = ts
            if cnt >= 50:
                rate = float(acc) / float(cnt)
                sigma = float(np.clip(sigma * math.exp(0.5 * (rate - 0.30)), 0.003, 0.6))
                acc = 0
                cnt = 0
        sigma = float(np.clip(sigma * 0.75, 0.003, 0.6))
        rep = project_box_sum(cur, 0.5 * n)
        if rep is not None and check_feasible(rep, n) and float(np.max(np.abs(rep - cur))) <= 1e-7:
            cur = rep
            cs = exact_score(cur, n)
        if np.isfinite(cs) and cs < bs - 1e-15:
            best = cur.copy()
            bs = cs
    rep = project_box_sum(best, 0.5 * n)
    if rep is not None and check_feasible(rep, n):
        best = rep
    best = final_adjust_sum(best)
    return best, exact_score(best, n)
def pair_local_search(h, n):
    best = h.copy()
    bs = exact_score(best, n)
    if not np.isfinite(bs):
        return best, bs
    for i in range(n - 1):
        for j in range(i + 1, n):
            tlo = -float(best[i])
            thi = 1.0 - float(best[j])
            lo_s = -tlo
            hi_s = thi
            if lo_s + hi_s <= 1e-14:
                continue
            ts = (-0.02 * lo_s, 0.02 * hi_s, -0.1 * lo_s, 0.1 * hi_s, -0.35 * lo_s, 0.35 * hi_s, -0.98 * lo_s, 0.98 * hi_s)
            for t in ts:
                if t < tlo - 1e-12 or t > thi + 1e-12:
                    continue
                trial = best.copy()
                trial[i] += t
                trial[j] -= t
                s = exact_score(trial, n)
                if np.isfinite(s) and s < bs - 1e-14:
                    best = trial
                    bs = s
                    break
    rep = project_box_sum(best, 0.5 * n)
    if rep is not None and check_feasible(rep, n):
        best = rep
    best = final_adjust_sum(best)
    return best, exact_score(best, n)
def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent, n = get_parent()
    best = parent.copy()
    best_score = exact_score(best, n)
    if not np.isfinite(best_score):
        best = np.full(n, 0.5, dtype=np.float64)
        best_score = exact_score(best, n)
    fallback = best.copy()
    fallback_score = best_score
    starts = build_starts(parent, n, rng)
    if len(starts) == 0:
        starts = [parent.copy()]
    valid = []
    scores = []
    for st in starts:
        s = exact_score(st, n)
        if np.isfinite(s):
            valid.append(st.copy())
            scores.append(s)
            if s < best_score - 1e-15:
                best = st.copy()
                best_score = s
    if len(valid) == 0:
        valid = [fallback.copy()]
        scores = [fallback_score]
    order = np.argsort(np.asarray(scores, dtype=np.float64))
    if n <= 64:
        cycles = 6
    elif n <= 256:
        cycles = 3
    else:
        cycles = 2
    for cyc in range(cycles):
        if cyc == 0:
            h0 = valid[int(order[0])].copy()
        elif cyc == 1 and len(order) > 1:
            h0 = valid[int(order[1])].copy()
        else:
            if n <= 64:
                sdp = 0.15 if cyc == 2 else 0.30
            elif n <= 256:
                sdp = 0.22
            else:
                sdp = 0.16
            h0 = None
            for _ in range(6):
                cand = project_box_sum(best + sdp * rng.standard_normal(n), 0.5 * n)
                if cand is not None and check_feasible(cand, n):
                    h0 = cand
                    break
            if h0 is None:
                h0 = best.copy()
        s0 = exact_score(h0, n)
        if np.isfinite(s0) and s0 < best_score - 1e-15:
            best = h0.copy()
            best_score = s0
        try:
            lbest, ls = anneal_cycle(h0, n, rng)
            if np.isfinite(ls) and ls < best_score - 1e-15:
                best = lbest.copy()
                best_score = ls
        except Exception:
            pass
    if n <= 128:
        try:
            lbest, ls = pair_local_search(best, n)
            if np.isfinite(ls) and ls < best_score - 1e-15:
                best = lbest.copy()
                best_score = ls
        except Exception:
            pass
    chosen_h = fallback.copy()
    chosen_s = exact_score(fallback, n)
    if not np.isfinite(chosen_s):
        chosen_h = np.full(n, 0.5, dtype=np.float64)
        chosen_s = exact_score(chosen_h, n)
    for vec in (best, fallback):
        rep = project_box_sum(vec, 0.5 * n)
        if rep is not None and check_feasible(rep, n):
            rep = final_adjust_sum(rep)
            s = exact_score(rep, n)
            if np.isfinite(s) and s < chosen_s - 1e-15:
                chosen_h = rep
                chosen_s = s
    rep = project_box_sum(chosen_h, 0.5 * n)
    if rep is not None and check_feasible(rep, n):
        chosen_h = rep
    chosen_h = final_adjust_sum(chosen_h)
    chosen_h = np.clip(np.asarray(chosen_h, dtype=np.float64), 0.0, 1.0)
    chosen_s = exact_score(chosen_h, n)
    if not np.isfinite(chosen_s):
        chosen_h = np.full(n, 0.5, dtype=np.float64)
        chosen_s = exact_score(chosen_h, n)
    return (chosen_h, float(chosen_s), n)
