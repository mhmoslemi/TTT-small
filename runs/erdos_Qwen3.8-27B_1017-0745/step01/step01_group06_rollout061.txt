import numpy as np
from scipy.optimize import minimize

def project_K(y, n, S):
    if not np.all(np.isfinite(y)) or len(y) != n:
        return np.full(n, S / n)
    y = y.astype(np.float64)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    e = S - h.sum()
    for _ in range(2):
        if abs(e) < 1e-13:
            break
        if e > 0:
            idx = np.where(1.0 - h > 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) > 0:
            h[idx] = np.clip(h[idx] + e / len(idx), 0.0, 1.0)
            e = S - h.sum()
    if abs(h.sum() - S) > 1e-8:
        return np.full(n, S / n)
    return h

def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lag_vec(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_grad(h, dx, n):
    G = np.zeros((2 * n - 1, n))
    j = np.arange(n)
    for k in range(2 * n - 1):
        ell = k - (n - 1)
        row = np.zeros(n)
        v1 = (j >= ell) & (j - ell < n)
        row[v1] += (1.0 - h[j[v1] - ell])
        v2 = (j + ell >= 0) & (j + ell < n)
        row[v2] -= h[j[v2] + ell]
        G[k] = row * dx
    return G

def feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and
            abs(h.sum() - S) <= 1e-8)

def transfer(h_old, n_old, n_new):
    lo_e = np.arange(n_old, dtype=np.float64) / n_old
    ro = (np.arange(n_old) + 1).astype(np.float64) / n_old
    ln_e = np.arange(n_new, dtype=np.float64) / n_new
    rn = (np.arange(n_new) + 1).astype(np.float64) / n_new
    ov = np.maximum(np.minimum(ro[:, None], rn[None, :]) -
                    np.maximum(lo_e[:, None], ln_e[None, :]), 0.0)
    return (ov.T @ h_old) * n_new

    return z[n] + 0.5 / eta * np.dot(z[:n], z[:n])

def _tr_obj_jac(z, a):

    g = np.zeros(n + 1)
    g[:n] = z[:n] / eta
    g[n] = 1.0
    return g

    return z[n] - q - G @ z[:n]

    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def _tr_eq_jac(z, a):
