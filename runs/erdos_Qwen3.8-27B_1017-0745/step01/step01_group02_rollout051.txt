The task is to find a step function h: [0,2] → [0,1] that minimizes C₅ = max_k ∫ h(x)(1-h(x+k)) dx.

Key constraints:
- h(x) ∈ [0,1]
- ∫₀² h(x) dx = 1 (i.e., sum(h) * dx = 1, or sum(h) = n/2)
- C₅ = max(np.correlate(h, 1-h, mode='full')) * dx where dx = 2/n

The previous program achieved 0.381021. We need to beat 0.38085.

The strategy suggests using a sequential convex Majorization-Minimization (MM) approach where:
1. At each feasible point x, build a quadratic upper bound for each lag overlap
2. The Hessian of each lag score has max eigenvalue L = 2*dx = 4/n
3. So q_s(y) ≤ q_s(x) + ∇q_s(x)·(y-x) + (L/2)||y-x||²
4. Minimize the max of these tangent planes plus the quadratic term
5. This gives a convex subproblem

        g[i] += (1.0 - h[j])
        g[j] -= h[i]
    q *= dx
    g *= dx
    return q, g

    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(float(np.sum(h)) - 0.5 * n) < 1e-10 * max(1, n))

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    try:
        c = np.correlate(h, 1.0 - h, mode='full')
        if c.shape[0] != 2 * n - 1:
            return float('inf')
        val = float(np.max(c) * (2.0 / n))
        return val if np.isfinite(val) else float('inf')
    except Exception:
        return float('inf')

def lag_value_grad(h, s, n):
    dx = 2.0 / n
    lo = max(0, -s)
    hi = min(n, n - s)
    q = 0.0
    g = np.zeros(n)
    for i in range(lo, hi):
        j = i + s
        q += h[i] * (1.0 - h[j])
        g[i] += (1.0 - h[j])
        g[j] -= h[i]
    return q * dx, g * dx

def separation(x, d, n):
    dx = 2.0 / n
    best = -float('inf')
    best_idx = 0
    for idx, s in enumerate(range(-(n - 1), n)):
        lo = max(0, -s)
        hi = min(n, n - s)
        ell = 0.0
        for i in range(lo, hi):
            j = i + s
            ell += x[i] * (1.0 - x[j]) + (1.0 - x[j]) * d[i] - x[i] * d[j]
        ell *= dx
        if ell > best:
            best = ell
            best_idx = idx
    return best, best_idx

def mm_obj(z):
    n = _ctx['n']
    y = z[:n]
    d = y - _ctx['x']
    return 0.5 * _ctx['L'] * float(np.dot(d, d)) + z[n]

def mm_obj_grad(z):
    n = _ctx['n']
    d = z[:n] - _ctx['x']
    g = np.zeros(n + 1)
    g[:n] = _ctx['L'] * d
    g[n] = 1.0
    return g

def mm_ineq(z):
    y = z[:_ctx['n']]
    d = y - _ctx['x']
    return z[_ctx['n']] - _ctx['qJ'] - _ctx['GJ'] @ d

def mm_ineq_jac(z):
    n = _ctx['n']
    J = np.zeros((_ctx['GJ'].shape[0], n + 1))
    J[:, :n] = -_ctx['GJ']
    J[:, n] = 1.0
    return J

def mm_eq(z):
    return float(np.sum(z[:_ctx['n']])) - 0.5 * _ctx['n']

def mm_eq_jac(z):
    g = np.zeros(_ctx['n'] + 1)
    g[:_ctx['n']] = 1.0
    return g

def solve_mm_subproblem(x, n):
    dx = 2.0 / n
    L = 2.0 * dx
    nv = 2 * n - 1
    qv = np.array([lag_value_grad(x, s, n)[0] for s in range(-(n-1), n)])
    G = np.array([lag_value_grad(x, s, n)[1] for s in range(-(n-1), n)])
    max_solves = 1 if n <= 64 else 8
    cut_cap = min(nv, 96)
    if n <= 64:
        active = list(range(nv))
    else:
        active = list(np.argsort(qv)[::-1][:cut_cap])
    y_cur = x.copy()
    t_cur = float(np.max(qv)) + 1e-6
    _ctx['n'] = n
    _ctx['x'] = x
    _ctx['L'] = L
    bounds = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
    cons = [{'type': 'ineq', 'fun': mm_ineq, 'jac': mm_ineq_jac},
            {'type': 'eq', 'fun': mm_eq, 'jac': mm_eq_jac}]
    for it in range(max_solves):
        _ctx['GJ'] = G[active]
        _ctx['qJ'] = qv[active]
        z0 = np.concatenate([y_cur, [t_cur]])
        try:
            res = minimize(mm_obj, z0, jac=mm_obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 150, 'ftol': 1e-11, 'disp': False})
        except Exception:
            break
        if not np.all(np.isfinite(res.x)):
            break
        y_raw = np.clip(res.x[:n], 0.0, 1.0)
        y_proj = project_box_sum(y_raw, 0.5 * n)
        if y_proj is None:
            break
        d = y_proj - x
        if n > 64:
            full_best, full_idx = separation(x, d, n)
            EJ = float(np.max(qv[active] + G[active] @ d))
            if full_best > EJ + 1e-9 * max(1.0, abs(EJ)) and full_idx not in active and len(active) < cut_cap:
                active.append(full_idx)
                y_cur = y_proj.copy()
                t_cur = EJ + 1e-6
                continue
        model_val = 0.5 * L * float(np.dot(d, d)) + float(np.max(qv[active] + G[active] @ d))
        if model_val > float(np.max(qv)) + 1e-7:
            break
        y_cur = y_proj.copy()
        break
    return y_cur

def local_mm_descent(h0, n, max_steps):
    h_cur = h0.copy()
    c_cur = exact_score(h_cur, n)
    for _ in range(max_steps):
        y_prop = solve_mm_subproblem(h_cur, n)
        if y_prop is None or not np.all(np.isfinite(y_prop)):
            break
        if float(np.max(np.abs(y_prop - h_cur))) <= 1e-11:
            break
        best_f = c_cur
        best_p = h_cur.copy()
        ndak = 9 if n <= 128 else 6
        for k in range(ndak):
            alpha = 2.0 ** (-k)
            cand = h_cur + alpha * (y_prop - h_cur)
            proj = project_box_sum(cand, 0.5 * n)
            if proj is None:
                continue
            c_new = exact_score(proj, n)
            if c_new < best_f - 1e-14:
                best_f = c_new
                best_p = proj.copy()
        if float(np.max(np.abs(best_p - h_cur))) <= 1e-13:
            break
        h_cur = best_p
        c_cur = best_f
        if c_cur >= 0 and (c_cur - c_cur) < 1e-10:
            break
    return h_cur, c_cur

def _get_parent():
    n = 45
    parent = None
    if 'initial_h_values' in globals():
        ph = globals().get('initial_h_values', None)
        if ph is not None:
            ph = np.asarray(ph, dtype=np.float64)
            if ph.ndim == 1 and 2 <= len(ph) <= 999 and np.all(np.isfinite(ph)):
                if np.all(ph >= -1e-8) and np.all(ph <= 1.0 + 1e-8):
                    n = len(ph)
                    if check_feasible(ph, n):
                        parent = ph.copy()
                    else:
                        p = project_box_sum(np.clip(ph, 0.0, 1.0), 0.5 * n)
                        if p is not None and check_feasible(p, n):
                            parent = p
    if parent is None:
        parent = project_box_sum(np.full(45, 0.5), 22.5)
        n = 45
    return parent, n

def _build_starts(parent, n, rng):
    starts = [parent.copy()]
    comp = project_box_sum(1.0 - parent, 0.5 * n)
    if comp is not None:
        starts.append(comp)
    rev = project_box_sum(parent[::-1].copy(), 0.5 * n)
    if rev is not None:
        starts.append(rev)
    if n <= 64:
        for sd in [0.05, 0.15, 0.25]:
            p = project_box_sum(parent + sd * rng.standard_normal(n), 0.5 * n)
            if p is not None:
                starts.append(p)
        p = project_box_sum(rng.random(n), 0.5 * n)
        if p is not None:
            starts.append(p)
        ar = np.arange(n, dtype=np.float64)
        u = 0.5 + 0.32 * np.cos(2.0 * np.pi * ar / n + 0.7) + 0.10 * np.sin(4.0 * np.pi * ar / n + 1.4)
        p = project_box_sum(u, 0.5 * n)
        if p is not None:
            starts.append(p)
        v = 0.5 + 0.25 * np.cos(np.pi * ar / n)
        p = project_box_sum(v, 0.5 * n)
        if p is not None:
            starts.append(p)
    elif n <= 256:
        for sd in [0.12, 0.25]:
            p = project_box_sum(parent + sd * rng.standard_normal(n), 0.5 * n)
            if p is not None:
                starts.append(p)
        p = project_box_sum(rng.random(n), 0.5 * n)
        if p is not None:
            starts.append(p)
    else:
        p = project_box_sum(parent + 0.15 * rng.standard_normal(n), 0.5 * n)
        if p is not None:
            starts.append(p)
    unique = []
    for s in starts:
        dup = False
        for u in unique:
            if float(np.max(np.abs(s - u))) < 1e-10:
                dup = True
                break
        if not dup:
            unique.append(s)
    return unique

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent, n = _get_parent()
    best_h = parent.copy()
    best_c5 = exact_score(parent, n)
    M = max(8, min(40, 1800 // n))
    starts = _build_starts(parent, n, rng)
    for h0 in starts:
        try:
            h_res, c_res = local_mm_descent(h0, n, M)
            if c_res < best_c5:
                best_h = h_res.copy()
                best_c5 = c_res
        except Exception:
            pass
    M2 = max(4, M // 2)
    sds = [0.04, 0.10, 0.22, 0.35] if n <= 64 else [0.08, 0.20] if n <= 256 else [0.15]
    for sd in sds:
        try:
            h0 = project_box_sum(best_h + sd * rng.standard_normal(n), 0.5 * n)
            if h0 is None:
                continue
            h_res, c_res = local_mm_descent(h0, n, M2)
            if c_res < best_c5:
                best_h = h_res.copy()
                best_c5 = c_res
        except Exception:
            pass
    best_h = project_box_sum(best_h, 0.5 * n)
    if best_h is None or not check_feasible(best_h, n):
        best_h = parent.copy()
        best_c5 = exact_score(parent, n)
    else:
        c_new = exact_score(best_h, n)
        if c_new < best_c5:
            best_c5 = c_new
        c_par = exact_score(parent, n)
        if c_par < best_c5:
            best_h = parent.copy()
            best_c5 = c_par
    best_c5 = exact_score(best_h, n)
    return (best_h, best_c5, n)
