import numpy as np
import math

def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):
        return None
    if np.max(np.abs(z)) > 1e12:
        return None
    n = len(z)
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(80):
        mid = lo + 0.5 * (hi - lo)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - np.sum(y)
    for i in range(n):
        if r > 0:
            a = min(r, 1.0 - y[i]); y[i] += a; r -= a
        elif r < 0:
            a = min(-r, y[i]); y[i] -= a; r += a
        if abs(r) <= 1e-13:
            break
    if abs(np.sum(y) - S) > 1e-9 or not np.all(np.isfinite(y)):
        return None
    return np.clip(y, 0.0, 1.0)

def is_feas(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-8)

def exact_lags(h, n):
    vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if vals.shape[0] != 2 * n - 1 or not np.all(np.isfinite(vals)):
        return None
    return vals

def exact_score(h, n):
    v = exact_lags(h, n)
    return float(np.max(v)) if v is not None else None

def pair_line(h, q, n, dx, p, r):
    L = max(-h[p], h[r] - 1.0)
    U = min(1.0 - h[p], h[r])
    if U - L <= 1e-12:
        return None
    m = 2 * n - 1
    a = np.zeros(m)
    b = np.zeros(m)
    j0 = n - 1
    a[j0] = -2.0 * dx
    jp = n - 1 - (r - p)
    jr = n - 1 - (p - r)
    if 0 <= jp < m:
        a[jp] = dx
    if 0 <= jr < m:
        a[jr] = dx
    for j in range(m):
        d = (n - 1) - j
        s = 0.0
        if 0 <= p + d < n:
            s += 1.0 - h[p + d]
        if 0 <= r + d < n:
            s -= 1.0 - h[r + d]
        if 0 <= p - d < n:
            s -= h[p - d]
        if 0 <= r - d < n:
            s += h[r - d]
        b[j] = dx * s
    cur = float(np.max(q))
    pts = np.linspace(L, U, 17)
    best_t, best_f = 0.0, np.inf
    for t in pts:
        f = float(np.max(q + b * t + a * t * t))
        if f < best_f or (f == best_f and abs(t) < abs(best_t)):
            best_f, best_t = f, t
    ts0 = 0.0
    f0 = cur
    if f0 < best_f or (f0 == best_f and abs(ts0) < abs(best_t)):
        best_f, best_t = f0, ts0
    rad = (U - L) / 16.0
    for _ in range(4):
        lo_r, hi_r = max(L, best_t - rad), min(U, best_t + rad)
        pr = np.linspace(lo_r, hi_r, 5)
        for t in pr:
            f = float(np.max(q + b * t + a * t * t))
            if f < best_f or (f == best_f and abs(t) < abs(best_t)):
                best_f, best_t = f, t
        rad /= 4.0
    if best_f > cur + 1e-12:
        return None
    h_new = h.copy()
    h_new[p] += best_t
    h_new[r] -= best_t
    return h_new

def polish(h, q, sc, n, dx, rng):
    P = min(n * (n - 1) // 2, 2 * n, max(24, 80000 // max(n, 1)))
    adj = min(n - 1, max(1, P // 2))
    pairs = [(i, i + 1) for i in range(0, n - 1, max(1, (n - 1) // adj))][:adj]
    need = P - len(pairs)
    if need > 0:
        existing = set(pairs)
        attempts = 0
        while len(pairs) < P and attempts < need * 3:
            i, j = tuple(sorted(rng.integers(0, n, 2)))
            if i != j and (i, j) not in existing:
                pairs.append((i, j)); existing.add((i, j))
            attempts += 1
        i = 0
        while len(pairs) < P and i < n:
            j = i + 1
            while j < n and (i, j) in existing:
                j += 1
            if j < n:
                pairs.append((i, j)); existing.add((i, j))
            i += 1
    rng.shuffle(pairs)
    improved = False
    for (p, r) in pairs:
        if not is_feas(h, n):
            h = project_capped_simplex(h, n / 2.0)
            if h is None:
                break
        q = exact_lags(h, n)
        if q is None:
            break
        sc = float(np.max(q))
        res = pair_line(h, q, n, dx, p, r)
        if res is not None:
            h2 = res
            if not is_feas(h2, n):
                h2 = project_capped_simplex(h2, n / 2.0)
                if h2 is None:
                    continue
            q2 = exact_lags(h2, n)
            if q2 is not None:
                sc2 = float(np.max(q2))
                if sc2 < sc - 1e-13:
                    h, q, sc = h2, q2, sc2
                    improved = True
    return h, q, sc, improved

def anneal(h, q, sc, n, dx, rng):
    E = 5 if n <= 128 else 4 if n <= 256 else 3 if n <= 512 else 2
    K = min(1000, max(80, 80000 // max(n, 1)))
    scale = 0.12
    for e in range(E):
        T = max(5e-5, 3e-3 * (0.4 ** e))
        acc = 0
        for _ in range(K):
            p, r = tuple(sorted(rng.integers(0, n, 2)))
            if p == r:
                continue
            L = max(-h[p], h[r] - 1.0)
            U = min(1.0 - h[p], h[r])
            if U - L <= 1e-14:
                continue
            t = rng.uniform(-scale, scale)
            if t < L or t > U:
                continue
            h_new = h.copy()
            h_new[p] += t
            h_new[r] -= t
            if not is_feas(h_new, n):
                h_new = project_capped_simplex(h_new, n / 2.0)
                if h_new is None:
                    continue
            q_new = exact_lags(h_new, n)
            if q_new is None:
                continue
            sc_new = float(np.max(q_new))
            delta = sc_new - sc
            if delta <= 0 or rng.random() < math.exp(max(-700, -delta / T)):
                h, q, sc = h_new, q_new, sc_new
                acc += 1
        ratio = acc / max(K, 1)
        if ratio < 0.12:
            scale = min(0.4, 1.25 * scale)
        elif ratio > 0.55:
            scale = max(0.01, 0.75 * scale)
        h, q, sc, _ = polish(h, q, sc, n, dx, rng)
    return h, q, sc

def build_starts(n, S, parent, rng):
    starts = []
    if parent is not None:
        starts.append(parent.copy())
    idx = np.arange(n, dtype=np.float64) + 0.5
    M = min(4, (n - 1) // 2)
    u = np.full(n, 0.5)
    for mpm in range(1, M + 1):
        amp = rng.normal(0.0, 0.12 / math.sqrt(mpm))
        phi = rng.uniform(0.0, 2.0 * np.pi)
        u += amp * np.cos(2.0 * np.pi * mpm * idx / n + phi)
    p = project_capped_simplex(u, S)
    if p is not None and is_feas(p, n):
        starts.append(p)
    if parent is not None:
        perm = rng.permutation(n)
        bb = np.zeros(n)
        bb[perm[:n // 2]] = 1.0
        if n % 2 == 1:
            bb[perm[n // 2]] = 0.5
        starts.append(bb)
        rv = project_capped_simplex(rng.uniform(0.0, 1.0, n), S)
        if rv is not None and is_feas(rv, n):
            starts.append(rv)
        starts.append(parent[::-1].copy())
    else:
        starts.append(np.full(n, 0.5))
        perm = rng.permutation(n)
        bb = np.zeros(n)
        bb[perm[:n // 2]] = 1.0
        if n % 2 == 1:
            bb[perm[n // 2]] = 0.5
        starts.append(bb)
        rv = project_capped_simplex(rng.uniform(0.0, 1.0, n), S)
        if rv is not None and is_feas(rv, n):
            starts.append(rv)
    deduped = []
    for s in starts:
        dup = any(np.max(np.abs(s - d)) < 1e-10 for d in deduped)
        if not dup and is_feas(s, n):
            deduped.append(s)
    cap = 6 if n <= 128 else 4 if n <= 256 else 3 if n <= 512 else 2
    return deduped[:cap]

def run(seed=42, **kwargs):

        seed = int(seed) if seed is not None else 42
    except (TypeError, ValueError):
        seed = 42
    rng = np.random.default_rng(seed % (2 ** 32))

    if n is None:

    h_fin = best_h.copy()
    if not is_feas(h_fin, n):

        sc_fin = exact_score(h_fin, n) or 0.5
    if sc_fin > best_score + 1e-10:
        h_fin = best_h.copy()
        sc_fin = best_score
    return h_fin, sc_fin, n
