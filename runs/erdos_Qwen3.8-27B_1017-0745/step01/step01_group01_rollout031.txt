            g[idx] = dx * np.dot(h[:m], 1.0 - h[d:d + m])

def smooth_eval(h, tau, n):
    g = lag_values(h, n)
    if not np.all(np.isfinite(g)):
        return float('inf'), g
    gm = g - np.max(g)
    return tau * np.log(np.sum(np.exp(gm / tau))), g

def smooth_grad(h, tau, n):
    g = lag_values(h, n)
    if not np.all(np.isfinite(g)):
        return None, None
    gm = g - np.max(g)
    eg = np.exp(gm / tau)
    S = float(np.sum(eg))
    if S <= 0 or not np.isfinite(S):
        return None, None
    w = eg / S
    F = tau * np.log(S)
    dx = 2.0 / n
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        idx = d + (n - 1)
        wi = w[idx] * dx
        if wi < 1e-15:
            continue
        if d == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            grad[:m] += wi * (1.0 - h[d:d + m])
            grad[d:d + m] -= wi * h[:m]
        else:
            dp = -d
            m = n - dp
            grad[dp:dp + m] += wi * (1.0 - h[:m])
            grad[:m] -= wi * h[dp:dp + m]
    if not np.all(np.isfinite(grad)):
        return None, None
    return F, grad

def project_capped(v, n):
    s = n / 2.0
    v = np.asarray(v, dtype=np.float64).ravel()
    if v.shape != (n,) or not np.all(np.isfinite(v)):
        return None
    v = np.clip(v, -1e12, 1e12)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(100):
        if hi - lo < 1e-15:
            break
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(3):
        if abs(r) < 1e-14:
            break
        if r > 0:
            for i in range(n):
                if h[i] < 1.0:
                    add = min(r, 1.0 - h[i])
                    h[i] += add
                    r -= add
                    if abs(r) < 1e-14:
                        break
        else:
            for i in range(n):
                if h[i] > 0.0:
                    sub = min(-r, h[i])
                    h[i] -= sub
                    r += sub
                    if abs(r) < 1e-14:
                        break
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 5e-12 * max(1, n):
        return None
    return h

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return float('inf')
    if abs(float(np.sum(h)) - n / 2.0) > 1e-10 * max(1, n):
        return float('inf')
    try:
        return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
    except Exception:
        return float('inf')

def add_start(starts, v, n):

            return
    starts.append(h2)

def optimize_start(h0, n, tau_high, tau_low, n_levels, iters_per_stage):
    x = h0.copy()
    alpha = 0.1 * n
    a_min, a_max = 1e-6, 20.0 * n
    for l in range(n_levels):

        F, grad = smooth_grad(x, tau, n)
        if F is None:
            break

                    break

            if not accepted:

- Index n-1 corresponds to d = 0
- Index 2n-2 corresponds to d = n-1

For d = 0: result = sum_i h[i]*(1-h[i]) ✓ (same in both)

For d > 0 (result index = d + n - 1):
np.correlate: sum_{i=0}^{n-d-1} h[i+d]*(1-h[i])

For d < 0, let d' = -d (result index = n-1-d'):
np.correlate: sum_{i=d'}^{n-1} h[i-d']*(1-h[i])

So I need to fix my lag_values:

```python
def lag_values(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    g = np.empty(L)
    for d in range(-(n - 1), n):
        idx = d + (n - 1)
        if d == 0:
            g[idx] = dx * np.dot(h, 1.0 - h)
        elif d > 0:
            m = n - d
            g[idx] = dx * np.dot(h[d:d+m], 1.0 - h[:m])
        else:
            dp = -d
            m = n - dp
            g[idx] = dx * np.dot(h[:m], 1.0 - h[dp:dp+m])
    return g
```

np.correlate: sum_{i=0}^{m-1} h[i+d] * (1-h[i]) where m = n-d

For d < 0, dp = -d:

Let j = i - dp, so i = j + dp, j = 0 to n-1-dp = m-1:
= sum_{j=0}^{m-1} h[j] * (1-h[j+dp])
= dot(h[:m], 1-h[dp:dp+m])
