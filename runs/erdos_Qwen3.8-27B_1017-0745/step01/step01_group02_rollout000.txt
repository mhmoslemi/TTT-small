import numpy as np
from math import ceil, tan, pi

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.ndim != 1 or len(v) == 0 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    if S < -1e-12 or S > n + 1e-12:
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(v - tau, 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-14:
        for i in np.argsort(1.0 - y):
            d = min(r, 1.0 - y[i])
            y[i] += d
            r -= d
            if r <= 1e-14:
                break
    elif r < -1e-14:
        for i in np.argsort(y):
            d = min(-r, y[i])
            y[i] -= d
            r += d
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(y)):
        return None
    y = np.clip(y, 0.0, 1.0)
    if abs(float(np.sum(y)) - S) > 1e-10 * max(1, n):
        return None
    return y

def check_feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(float(np.sum(h)) - 0.5 * n) < 1e-10 * max(1, n))

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    try:
        c = np.correlate(h, 1.0 - h, mode='full')
        if not np.all(np.isfinite(c)) or len(c) != 2 * n - 1:
            return float('inf')
        return float(np.max(c) * (2.0 / n))
    except Exception:
        return float('inf')

def get_parent():
    n = 45
    ph = globals().get('initial_h_values', None)
    if ph is not None:
        p = np.asarray(ph, dtype=np.float64).copy()
        if p.ndim == 1 and 2 <= len(p) <= 999 and np.all(np.isfinite(p)):
            if np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                n = len(p)
                if check_feasible(p, n):
                    return p, n
                p2 = project_box_sum(np.clip(p, 0.0, 1.0), 0.5 * n)
                if p2 is not None and check_feasible(p2, n):
                    return p2, n
    return np.full(n, 0.5), n

def build_pop(parent, n, rng, P):
    pop = [parent.copy()]
    scores = [exact_score(parent, n)]
    def try_add(hc):
        y = project_box_sum(hc, 0.5 * n)
        if y is None or not check_feasible(y, n):
            return
        for m in pop:
            if np.max(np.abs(y - m)) < 1e-10:
                return
        pop.append(y)
        scores.append(exact_score(y, n))
        if len(pop) >= P:
            raise StopIteration
    try:
        try_add(1.0 - parent)
        try_add(parent[::-1].copy())
        try_add(np.roll(parent, 1))
        for sd in [0.03, 0.08, 0.18, 0.35]:
            try_add(parent + sd * rng.standard_normal(n))
        for _ in range(6):
            try_add(rng.random(n))
        idx = np.arange(n)
        for m in range(1, min(5, n // 2 + 1)):
            phi = rng.uniform(0, 2 * pi)
            theta = rng.uniform(0, 2 * pi)
            u = 0.5 + 0.28 * np.cos(2 * pi * m * idx / n + phi) + 0.12 * np.sin(2 * pi * (m + 1) * idx / n + theta)
            try_add(u)
    except StopIteration:
        pass
    attempts = 0
    while len(pop) < P and attempts < 30 * P:
        attempts += 1
        if attempts % 2 == 0:
            try_add(parent + rng.uniform(0.03, 0.35) * rng.standard_normal(n))
        else:
            try_add(rng.random(n))
    while len(pop) < P:
        c = parent.copy()
        pop.append(c)
        scores.append(exact_score(c, n))
    return pop, scores

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent, n = get_parent()
    S = 0.5 * n
    best_h = parent.copy()
    best_s = exact_score(parent, n)
    if n <= 64:
        P, G, stag_lim, max_restart = 96, 300, 50, 3
    elif n <= 128:
        P, G, stag_lim, max_restart = 56, 150, 30, 2
    else:
        P, G, stag_lim, max_restart = 20, 32, 10, 1
    p_frac = 0.20
    mu_F, mu_CR = 0.55, 0.70
    pop, scores = build_pop(parent, n, rng, P)
    for i in range(P):
        if scores[i] < best_s:
            best_s = scores[i]
            best_h = pop[i].copy()
    archive = []
    no_improve = 0
    restarts = 0
    try:
        for gen in range(G):
            snap = [x.copy() for x in pop]
            snap_s = list(scores)
            next_pop = [x.copy() for x in pop]
            next_s = list(scores)
            order = np.argsort(snap_s)
            top_cnt = max(2, ceil(p_frac * P))
            top_idx = list(order[:top_cnt])
            successes_F, successes_CR = [], []
            saved_best = best_s
            for i in range(P):
                tgt = snap[i]
                ti = i
                p_choices = [j for j in top_idx if j != ti]
                if not p_choices:
                    p_choices = top_idx
                pj = p_choices[rng.integers(len(p_choices))]
                r1_choices = [j for j in range(P) if j != ti and j != pj]
                r1j = r1_choices[rng.integers(len(r1_choices))]
                r2_choices = [j for j in range(P) if j != ti and j != pj and j != r1j]
                if archive:
                    r2_pool = list(range(len(archive)))
                else:
                    r2_pool = []
                if r2_choices:
                    use_arch = r2_pool and rng.random() < 0.3
                else:
                    use_arch = bool(r2_pool)
                if use_arch and r2_pool:
                    r2v = archive[rng.integers(len(archive))]
                elif r2_choices:
                    r2v = snap[r2_choices[rng.integers(len(r2_choices))]]
                else:
                    continue
                u = rng.uniform(0.02, 0.98)
                F_i = mu_F + 0.1 * tan(pi * (u - 0.5))
                if F_i <= 0 or not np.isfinite(F_i):
                    F_i = mu_F
                F_i = float(np.clip(F_i, 0.05, 1.0))
                CR_i = float(np.clip(rng.normal(mu_CR, 0.1), 0.0, 1.0))
                donor = tgt + F_i * (snap[pj] - tgt) + F_i * (snap[r1j] - r2v)
                jrand = rng.integers(n)
                mask = rng.random(n) < CR_i
                mask[jrand] = True
                trial_cross = np.where(mask, donor, tgt)

                    next_s[i] = ts
                    archive.append(tgt.copy())
                    successes_F.append(F_i)

            if successes_CR:
                mu_CR = 0.9 * mu_CR + 0.1 * (sum(successes_CR) / len(successes_CR))
                mu_CR = float(np.clip(mu_CR, 0.0, 1.0))
            if best_s < saved_best - 1e-12 * max(1.0, abs(saved_best)):
                no_improve = 0
            else:
                no_improve += 1
            if no_improve >= stag_lim:
                if restarts >= max_restart:
                    break
                restarts += 1
                no_improve = 0
                archive = []
                mu_F, mu_CR = 0.55, 0.70
                sort_idx = np.argsort(scores)
                elite_cnt = max(2, P // 4)
                new_pop, new_s = [best_h.copy()], [best_s]
                for j in sort_idx:
                    if len(new_pop) >= elite_cnt:
                        break
                    if np.max(np.abs(snap[j] - best_h)) >= 1e-10:

                        new_s.append(scores[j])

                    if hc is None:

                    new_pop.append(hc)
                    new_s.append(exact_score(hc, n))
                for k in range(fill_need - noise_cnt):

    except Exception:
        pass
    final_h = project_box_sum(best_h, S)

        final_h = parent.copy()

        final_h = parent.copy()
        final_s = parent_s
    final_h = np.clip(final_h, 0.0, 1.0).astype(np.float64)
    final_s = float(np.max(np.correlate(final_h, 1.0 - final_h, mode='full')) * (2.0 / n))
    return (final_h, final_s, n)
