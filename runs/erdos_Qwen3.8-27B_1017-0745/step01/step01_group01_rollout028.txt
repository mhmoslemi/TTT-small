            add_start(starts, (1.0 - pp[::-1]).copy(), n)
            add_start(starts, pp + rng.normal(0, 0.02, n), n)
            add_start(starts, pp + rng.normal(0, 0.08, n), n)
    add_start(starts, uniform, n)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    add_start(starts, blk, n)
    idx = np.arange(n, dtype=np.float64)
    add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * idx / n), n)
    add_start(starts, 0.5 + 0.35 * np.cos(4.0 * np.pi * idx / n), n)
    add_start(starts, rng.uniform(0, 1, n), n)

    ns_def = 8 if n <= 64 else (5 if n <= 128 else 3)
    n_starts = max(1, min(ns_def, len(starts)))
    starts = starts[:n_starts]

    best_h, best_s = None, float('inf')
    for st in starts:
        s = exact_score(st, n)
        if np.isfinite(s) and s < best_s:
            best_h, best_s = st.copy(), s
    if best_h is None:
        best_h = project_capped(uniform, n)
        best_s = exact_score(best_h, n)

    mi_def = 250 if n <= 64 else (160 if n <= 128 else 80)
    ftol = 1e-10
    if n <= 256 and mi_def > 0:
        bounds = [(0.0, 1.0)] * (n + 1)
        cons = [
            {'type': 'eq', 'fun': _eq, 'jac': _eq_jac, 'args': (n,)},
            {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac, 'args': (n,)}
        ]
        for st in starts:
            x0 = st.copy()
            s0 = exact_score(x0, n)
            if not np.isfinite(s0):
                continue
            z0 = np.concatenate([x0, [min(1.0, s0 + 1e-8)]])
            try:
                res = minimize(_obj, z0, method='SLSQP', jac=_obj_jac,
                               args=(n,), bounds=bounds, constraints=cons,
                               options={'ftol': ftol, 'maxiter': mi_def})
            except Exception:
                continue
            if res.x is not None and len(res.x) == n + 1 and np.all(np.isfinite(res.x)):
                hp = project_capped(res.x[:n], n)
                if hp is not None:
                    s_new = exact_score(hp, n)
                    if np.isfinite(s_new) and s_new < best_s - 1e-12:
                        best_h, best_s = hp, s_new
            r2 = z0.copy()
            r2[-1] = min(1.0, best_s + 1e-6)
            try:
                res2 = minimize(_obj, r2, method='SLSQP', jac=_obj_jac,
                                args=(n,), bounds=bounds, constraints=cons,
                                options={'ftol': ftol, 'maxiter': mi_def})
            except Exception:
                continue
            if res2.x is not None and len(res2.x) == n + 1 and np.all(np.isfinite(res2.x)):
                hp2 = project_capped(res2.x[:n], n)
                if hp2 is not None:
                    s2 = exact_score(hp2, n)
                    if np.isfinite(s2) and s2 < best_s - 1e-12:
                        best_h, best_s = hp2, s2

    if n <= 64:
        h_pol = polish_pairs(best_h.copy(), n, 1)
        hp = project_capped(h_pol, n)
        if hp is not None:
            sp = exact_score(hp, n)
            if np.isfinite(sp) and sp < best_s - 1e-12:
                best_h, best_s = hp, sp

    final = project_capped(best_h, n)
    if final is None:
        final = project_capped(uniform, n)
    s_final = exact_score(final, n)
    if not np.isfinite(s_final) or s_final > best_s:
        final = best_h.copy()
        s_final = best_s
    if not np.isfinite(s_final):
        final = np.full(n, 0.5)
        s_final = exact_score(final, n)
    return final, float(s_final), int(n)
