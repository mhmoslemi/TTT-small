            lam = 0.5 * (lo + hi)
    c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(3):
        diff = S - c.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = c < 1.0 - 1e-12
        else:
            mask = c > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        c[mask] += diff / cnt
        np.clip(c, 0.0, 1.0, out=c)
    return c

def lag_scores(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    qs = lag_scores(h, n)
    if len(qs) != 2 * n - 1 or not np.all(np.isfinite(qs)):
        return float('inf')
    return float(qs.max())

def build_gradient_matrix(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        row = k + n - 1
        lo1, hi1 = max(0, k), min(n, n + k)
        if hi1 > lo1:
            G[row, lo1:hi1] += dx * (1.0 - h[lo1 - k:hi1 - k])
        lo2, hi2 = max(0, -k), min(n, n - k)
        if hi2 > lo2:
            G[row, lo2:hi2] -= dx * h[lo2 + k:hi2 + k]
    if not np.all(np.isfinite(G)):
        return None
    return G

def project_simplex(z):
    n = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(n, 1.0 / n)
    sz = np.sort(z)[::-1]
    css = np.cumsum(sz)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = sz - (css - 1.0) / rho > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    r = int(np.max(np.where(cond)[0]))
    theta = (css[r] - 1.0) / (r + 1)
    p = np.maximum(z - theta, 0.0)
    s = p.sum()
    if s > 0 and abs(s - 1.0) > 1e-12:
        p /= s
    return p

def dual_eval(p, h, G, q, L, n):
    gmix = G.T @ p
    y = project_capped(h - gmix / L, n)
    d = y - h
    u = q + G @ d
    D = float(p @ u + 0.5 * L * (d @ d))
    gap = float(u.max() - p @ u)
    return d, u, D, gap

def solve_majorant(h, q, G, n, max_inner):
    dx = 2.0 / n
    L = 2.0 * dx
    m = len(q)
    Ldual = max(1e-12, 1.05 * float(np.sum(G * G)) / L)
    qmax = float(q.max())
    tol_q = max(1e-12, 1e-8 * max(1.0, abs(qmax)))
    mask = q >= qmax - tol_q
    if not np.any(mask):
        p = np.zeros(m)
        p[int(np.argmax(q))] = 1.0
    else:
        p = np.zeros(m)
        p[mask] = 1.0 / float(mask.sum())
    best_D = -1e18
    best_d = np.zeros(n)
    best_u = q.copy()
    x = p.copy()
    yv = p.copy()
    theta = 1.0
    consec_small = 0
    for it in range(max_inner):
        d_y, u_y, D_y, gap_y = dual_eval(yv, h, G, q, L, n)
        if not np.all(np.isfinite(u_y)):
            break
        x_trial = project_simplex(yv + u_y / Ldual)
        d_xt, u_xt, D_xt, gap_xt = dual_eval(x_trial, h, G, q, L, n)
        if not np.all(np.isfinite(u_xt)):
            break
        if D_xt < D_y - 1e-13:
            x_trial2 = project_simplex(x + u_y / Ldual)
            d_xt2, u_xt2, D_xt2, gap_xt2 = dual_eval(x_trial2, h, G, q, L, n)
            if np.isfinite(D_xt2) and D_xt2 >= D_y - 1e-13:
                x = x_trial2
                yv = x_trial2
                theta = 1.0
            else:
                break
        else:
            theta_new = (1.0 + math.sqrt(1.0 + 4.0 * theta * theta)) / 2.0
            yv = x_trial + ((theta - 1.0) / theta_new) * (x_trial - x)
            x = x_trial
            theta = theta_new
        if D_xt > best_D:
            best_D = D_xt
            best_d = d_xt.copy()
            best_u = u_xt.copy()
        gap_tol = max(1e-10, 1e-8 * max(1.0, abs(float(best_u.max()))))
        if gap_xt <= gap_tol:
            break
        if it > 0:
            step_n = float(np.abs(x_trial - x).max())
            if step_n <= 1e-10:
                consec_small += 1
                if consec_small >= 3:
                    break
            else:
                consec_small = 0
    return best_d, best_u

def optimize_start(h0, n, max_outer, max_inner):
    h = project_capped(h0.copy(), n)
    if not check_feasible(h, n):
        return h0.copy(), exact_score(h0, n)
    cur_s = exact_score(h, n)
    best_h = h.copy()
    best_s = cur_s
    stall = 0
    for _ in range(max_outer):
        q = lag_scores(h, n)
        G = build_gradient_matrix(h, n)
        if G is None:
            break
        d, u = solve_majorant(h, q, G, n, max_inner)
        if not np.all(np.isfinite(d)) or float(np.abs(d).max()) <= 1e-10:
            break
        v = G @ d
        norm2 = float(d @ d)
        L = 4.0 / n
        qmax = float(q.max())
        accepted = False
        for alpha in [1.0 / (2.0 ** j) for j in range(20)]:
            model_val = float(np.max(q + alpha * v) + 0.5 * L * alpha * alpha * norm2)
            if model_val >= qmax - 1e-14:
                break
            h_trial = h + alpha * d
            h_trial = project_capped(h_trial, n)
            if not check_feasible(h_trial, n):
                continue
            s_trial = exact_score(h_trial, n)
            if s_trial < cur_s - 1e-13:
                h = h_trial
                cur_s = s_trial
                if cur_s < best_s - 1e-12:
                    best_s = cur_s
                    best_h = h.copy()
                stall = 0
                accepted = True
                break
        if not accepted:
            break
        if cur_s - best_s < 1e-11:
            break
    return best_h, best_s

def local_swap_polish(h, n, max_passes=5, step=0.02):
    h = h.copy()
    cur_s = exact_score(h, n)
    for _ in range(max_passes):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                if h[i] < step or h[j] < step:
                    continue
                h2 = h.copy()
                h2[i] -= step
                h2[j] += step
                if h2[i] >= -1e-12 and h2[j] <= 1.0 + 1e-12:
                    h2 = project_capped(h2, n)
                    s2 = exact_score(h2, n)
                    if s2 < cur_s - 1e-12:
                        h = h2
                        cur_s = s2
                        improved = True
        if not improved:
            break
    return h, cur_s

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent = globals().get('initial_h_values', None)
    n = 45
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.ndim == 1 and 2 <= len(p) <= 999 and np.all(np.isfinite(p)) and float(np.max(np.abs(p))) < 1e6:
            n = len(p)
            fallback = project_capped(p.copy(), n)
            if not check_feasible(fallback, n):
                fallback = np.full(n, 0.5)
        else:
            fallback = np.full(n, 0.5)
    else:
        np_kw = kwargs.get('n_points', None)
        if isinstance(np_kw, int) and not isinstance(np_kw, bool) and 2 <= np_kw <= 999:
            n = np_kw
        fallback = np.full(n, 0.5)
    fallback_s = exact_score(fallback, n)
    best_h = fallback.copy()
    best_s = fallback_s
    if n <= 100:
        mo, mi = 80, 80
    elif n <= 400:
        mo, mi = 12, 20
    else:
        mo, mi = 5, 12
    starts = [fallback.copy(), 1.0 - fallback.copy(), fallback[::-1].copy(),
              1.0 - fallback[::-1].copy()]
    starts.append(project_capped(fallback + 0.06 * rng.standard_normal(n), n))
    starts.append(project_capped(fallback + 0.20 * rng.standard_normal(n), n))
    starts.append(project_capped(rng.uniform(0, 1, n), n))
    dedup = []
    for s in starts:
        if not np.all(np.isfinite(s)):
            continue
        dup = False
        for d in dedup:
            if float(np.abs(s - d).max()) < 1e-8:
                dup = True
                break
        if not dup:
            dedup.append(s)
    starts = dedup
    max_starts = 7 if n <= 100 else (4 if n <= 400 else 2)
    for st in starts[:max_starts]:
        try:
            bh, bs = optimize_start(st, n, mo, mi)
            if bs < best_s - 1e-12:
                best_s = bs
                best_h = bh.copy()
        except Exception:
            pass
    try:
        restart = project_capped(best_h + 0.025 * rng.standard_normal(n), n)
        bh, bs = optimize_start(restart, n, mo // 2, mi)
        if bs < best_s - 1e-12:
            best_s = bs
            best_h = bh.copy()
    except Exception:
        pass
    if n <= 100:
        try:
            ph, ps = local_swap_polish(best_h, n, max_passes=6, step=0.015)
            if ps < best_s - 1e-12:
                best_s = ps
                best_h = ph.copy()
        except Exception:
            pass
    final_h = project_capped(best_h, n)
    final_s = exact_score(final_h, n)
    if final_s > fallback_s + 1e-12 or not check_feasible(final_h, n):
        final_h = fallback.copy()
        final_s = fallback_s
    else:
        final_h = project_capped(final_h, n)
        final_s = exact_score(final_h, n)
    return final_h.astype(np.float64), float(final_s), int(n)
