def soft_obj(z,N,mu):

       return 1e100
    q=np.correlate(h, 1.0-h, mode='full')

    qm=float(np.max(q))

def project_box_sum(z, m):
    z = np.asarray(z, dtype=float)
    N = z.size
    if N == 0:
        return z.copy()
    if not np.all(np.isfinite(z)):
        return np.full(N, float(np.clip(m / N, 0, 1)) if N else 0)
    if m <= 0:
        return np.zeros(N)
    if m >= N:
        return np.ones(N)
    lo, hi = float(z.min()) - 1.0, float(z.max())
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    r = m - h.sum()
    if r > 0:
        for i in range(N):
            if r <= 1e-13: break
            d = min(1.0 - h[i], r)
            h[i] += d; r -= d
    elif r < 0:
        for i in range(N):
            if r >= -1e-13: break
            d = min(h[i], -r)
            h[i] -= d; r += d
    h = np.clip(h,0,1)
    if not np.all(np.isfinite(h)) or abs(h.sum()-m) > 1e-6:
        return np.full(N, float(np.clip(m / N, 0, 1)))
    return h

import numpy as np
from scipy.optimize import minimize, linprog


def project_box_sum(z, m):
    z = np.asarray(z, dtype=float)
    N = z.size
    if N == 0:
        return z.copy()
    if not np.all(np.isfinite(z)):
        return np.full(N, float(np.clip(m / N, 0.0, 1.0)))
    if m <= 0.0:
        return np.zeros(N)
    if m >= N:
        return np.ones(N)
    lo = float(z.min()) - 1.0
    hi = float(z.max())
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    if r > 0.0:
        for i in range(N):
            if r <= 1e-13:
                break
            d = min(1.0 - h[i], r)
            h[i] += d
            r -= d
    elif r < 0.0:
        for i in range(N):
            if r >= -1e-13:
                break
            d = min(h[i], -r)
            h[i] -= d
            r += d
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(h.sum() - m) > 1e-6:
        return np.full(N, float(np.clip(m / N, 0.0, 1.0)))
    return h


def check_feasible(h, N, m):
    return (
        h.shape == (N,)
        and np.all(np.isfinite(h))
        and h.min() >= -1e-8
        and h.max() <= 1.0 + 1e-8
        and abs(h.sum() - m) <= 1e-5
    )


def exact_score(h, N):
    try:
        if h.shape != (N,) or not np.all(np.isfinite(h)):
            return np.inf
        c = np.correlate(h, 1.0 - h, mode="full")
        if c.size == 0 or not np.all(np.isfinite(c)):
            return np.inf
        return float(np.max(c) * 2.0 / N)
    except Exception:
        return np.inf


def compute_q_grad(h, N):
    if N <= 0:
        return np.empty(0), np.empty(0)
    q = np.zeros(2 * N - 1)
    g = np.zeros(N)
    q[N - 1] = float(np.dot(h, 1.0 - h))
    g += 1.0 - 2.0 * h
    for s in range(1, N):
        L = N - s
        a = h[:L]
        b = h[s:s + L]
        q[N - 1 + s] = float(np.dot(a, 1.0 - b))
        q[N - 1 - s] = float(np.dot(b, 1.0 - a))
        g[:L] += 1.0 - b
        g[s:] -= a
        g[s:] += 1.0 - a
        g[:L] -= b
    return q, g


def soft_obj(z, N, mu):
    h = z[:N]
    if not np.all(np.isfinite(h)):
        return 1e100
    q = np.correlate(h, 1.0 - h, mode="full")
    if q.size == 0 or not np.all(np.isfinite(q)):
        return 1e100
    qm = float(np.max(q))
    if mu <= 1e-8:
        return qm
    w = np.exp((q - qm) / mu)
    S = float(w.sum())
    if S <= 0.0:
        return qm
    val = qm + mu * np.log(S)
    return float(val) if np.isfinite(val) else 1e100


def soft_jac(z, N, mu):
    h = z[:N]
    if not np.all(np.isfinite(h)):
        return np.zeros(N)
    q, g = compute_q_grad(h, N)
    if not (np.all(np.isfinite(q)) and np.all(np.isfinite(g))):
        return np.zeros(N)
    qm = float(np.max(q))
    if mu <= 1e-8:
        act = q >= qm - 1e-9
        return g[act].mean(axis=0)
    w = np.exp((q - qm) / mu)
    S = float(w.sum())
    if S <= 0.0:
        return np.zeros(N)
    out = (w / S) @ g
    return out if np.all(np.isfinite(out)) else np.zeros(N)


def eq_sum(z, N, m):
    return np.array([np.sum(z[:N]) - m])


def eq_sum_jac(z, N, m):
    J = np.zeros((1, N))
    J[0, :N] = 1.0
    return J


def lp_polish(h0, N, m, steps=20):
    h = project_box_sum(h0, m)
    s = exact_score(h, N)
    if not np.isfinite(s):
        return h, s
    stall = 0
    for _ in range(steps):
        if stall >= 2:
            break
        q, g = compute_q_grad(h, N)
        if not (np.all(np.isfinite(q)) and np.all(np.isfinite(g))):
            break
        qm = float(np.max(q))
        if stall == 0:
            tol = max(0.08, 0.04 * max(1.0, qm))
        else:
            tol = 0.5 * max(1.0, qm)
        mask = q >= qm - tol
        if not np.any(mask):
            mask = np.ones(q.shape, dtype=bool)
        K = int(np.count_nonzero(mask))
        A_ub = np.column_stack((g[mask], -np.ones(K)))
        b_ub = -q[mask]
        A_eq = np.zeros((1, N + 1))
        A_eq[0, :N] = 1.0
        b_eq = np.array([0.0])
        bounds = [(-float(h[i]), 1.0 - float(h[i])) for i in range(N)] + [(0.0, float(N) + 1.0)]
        cvec = np.zeros(N + 1)
        cvec[N] = 1.0
        try:
            res = linprog(cvec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method="highs")
        except Exception:
            break
        if res.status != 0 or res.x is None or not np.all(np.isfinite(res.x)):
            break
        delta = res.x[:N]
        if float(np.linalg.norm(delta)) < 1e-9:
            break
        best_h = h
        best_s = s
        found = False
        for a in (1.0, 0.66, 0.44, 0.29, 0.19, 0.13, 0.08, 0.05):
            cand = project_box_sum(h + a * delta, m)
            cs = exact_score(cand, N)
            if cs < best_s - 1e-12:
                best_h = cand
                best_s = cs
                found = True
        if found:
            h = best_h
            s = best_s
            stall = 0
        else:
            stall += 1
    return h, s


def soft_polish(h0, N, m):
    h = project_box_sum(h0, m)
    s = exact_score(h, N)
    if not np.isfinite(s):
        return h, s
    try:
        bounds = [(0.0, 1.0)] * N
        cons = [{"type": "eq", "fun": eq_sum, "jac": eq_sum_jac, "args": (N, m)}]
        if N <= 100:
            mus = (0.03, 0.012, 0.004)
        else:
            mus = (0.05, 0.02)
        for mu in mus:
            res = minimize(
                soft_obj, h.copy(), jac=soft_jac, method="SLSQP",
                bounds=bounds, constraints=cons, args=(N, mu),
                options={"maxiter": 150, "ftol": 1e-10}
            )
            if res.x is not None:
                cand = project_box_sum(res.x[:N], m)
                if check_feasible(cand, N, m):
                    cs = exact_score(cand, N)
                    if cs < s - 1e-12:
                        h = cand
                        s = cs
    except Exception:
        pass
    if N <= 80:
        steps = 20
    elif N <= 200:
        steps = 10
    else:
        steps = 5
    h, s = lp_polish(h, N, m, steps=steps)
    return h, s


def add_candidate(cands, h, N, m, tol=1e-8):
    h = np.asarray(h, dtype=float)
    if h.size != N or not np.all(np.isfinite(h)):
        return len(cands)
    hp = project_box_sum(h, m)
    if not check_feasible(hp, N, m):
        return len(cands)
    for old in cands:
        if float(np.max(np.abs(hp - old))) <= tol:
            return len(cands)
    cands.append(hp)
    return len(cands)


def build_population(base, N, m, rng, P):
    c = []
    add_candidate(c, base, N, m)
    add_candidate(c, 1.0 - base, N, m)
    for lam in (1.0, -1.0, 0.85, -0.85, 0.60, -0.60):
        add_candidate(c, 0.5 + lam * (base - 0.5), N, m)
    for sh in rng.integers(0, N, size=4):
        add_candidate(c, np.roll(base, int(sh)), N, m)
    for sd in (0.03, 0.08, 0.18, 0.32):
        add_candidate(c, base + sd * rng.normal(size=N), N, m)
    add_candidate(c, rng.uniform(size=N), N, m)
    add_candidate(c, rng.integers(0, 2, N).astype(float), N, m)
    idx = np.arange(N)
    for A in (0.12, 0.22, 0.32):
        z = 0.5 * np.ones(N)
        R = min(6, (N - 1) // 2)
        for r in range(1, R + 1):
            ang1 = 2.0 * np.pi * r * idx / N
            z += (A / r) * (rng.standard_normal() * np.sin(ang1)
                            + rng.standard_normal() * np.cos(ang1))
        add_candidate(c, z, N, m)
    tries = 0
    while len(c) < P and tries < 20 * P:
        sd = float(rng.uniform(0.04, 0.35))
        add_candidate(c, base + sd * rng.normal(size=N), N, m)
        add_candidate(c, rng.uniform(size=N), N, m)
        tries += 2
    while len(c) < P:
        c.append((c[-1] if c else np.full(N, 0.5)).copy())
    return np.array(c[:P], dtype=float)


def de_search(base, N, m, rng, P, gens, stag_limit, max_restart):
    pop = build_population(base, N, m, rng, P)
    scores = np.array([exact_score(pop[i], N) for i in range(P)])
    bi = int(np.argmin(scores))
    best = pop[bi].copy()
    best_s = float(scores[bi])
    F, CR, tol = 0.7, 0.85, 1e-12
    stag = 0
    rn = 0
    gen = 0
    while gen < gens:
        snap = pop.copy()
        scap = scores.copy()
        nturn = snap.copy()
        nsc = scap.copy()
        start_s = best_s
        for i in range(P):
            for _ in range(8):
                ch = rng.choice(P, 3, replace=False)
                a, b, c = int(ch[0]), int(ch[1]), int(ch[2])
                if a != i and b != i and c != i:
                    break
            else:
                continue
            v = snap[a] + F * (snap[b] - snap[c])
            mask = rng.random(N) < CR
            mask[int(rng.integers(N))] = True
            u = np.where(mask, v, snap[i])
            if not np.all(np.isfinite(u)):
                continue
            trial = project_box_sum(u, m)
            if not check_feasible(trial, N, m):
                continue
            ts = exact_score(trial, N)
            if ts < nsc[i] - tol:
                nturn[i] = trial
                nsc[i] = ts
                if ts < best_s - tol:
                    best = trial.copy()
                    best_s = ts
        pop = nturn
        scores = nsc
        gen += 1
        if best_s < start_s - tol:
            stag = 0
        else:
            stag += 1
            if stag >= stag_limit and rn < max_restart:
                keep = max(2, P // 4)
                order = np.argsort(scores)
                nc = []
                add_candidate(nc, best, N, m, 1e-8)
                for j in order[:keep]:
                    if len(nc) >= P:
                        break
                    add_candidate(nc, pop[j], N, m, 1e-8)
                sig = 0.14 if rn == 0 else 0.05
                while len(nc) < P:
                    if (len(nc) % 4) < 3:
                        add_candidate(nc, best + sig * rng.normal(size=N), N, m, 1e-8)
                    else:
                        add_candidate(nc, rng.uniform(size=N), N, m, 1e-8)
                if len(nc) >= P:
                    pop = np.array(nc[:P], dtype=float)
                    scores = np.array([exact_score(pop[i], N) for i in range(P)])
                    stag = 0
                    rn += 1
                else:
                    if stag >= 2 * stag_limit:
                        break
            elif stag >= 2 * stag_limit and rn >= max_restart:
                break
    return best, pop, scores


def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)

    base = None
    N = None
    m = None

    raw = globals().get("initial_h_values", None)
    if raw is not None:
        try:
            arr = np.asarray(raw, dtype=float)
        except Exception:
            arr = None
        if arr is not None and arr.ndim == 1 and arr.size > 1 and np.all(np.isfinite(arr)):
            N = int(arr.size)
            m = N / 2.0
            if arr.min() >= -1e-8 and arr.max() <= 1.0 + 1e-8 and abs(arr.sum() - m) <= 1e-5:
                base = np.clip(arr, 0.0, 1.0)
            else:
                base = project_box_sum(arr, m)
            if not check_feasible(base, N, m):
                base = np.full(N, 0.5)

    if N is None:
        nn = kwargs.get("n_points", 70)
        if isinstance(nn, (int, np.integer)) and not isinstance(nn, bool) and 2 <= int(nn) <= 999:
            N = int(nn)
        else:
            N = 70
        m = N / 2.0
        base = np.full(N, 0.5)

    if N == 1:
        h = np.array([0.5])
        return h, exact_score(h, 1), 1

    base = project_box_sum(base, m)
    base_s = exact_score(base, N)
    best = base.copy()
    best_s = base_s

    if N > 999:
        final = project_box_sum(best, m)
        return final, float(exact_score(final, N)), int(N)

    if N <= 96:
        P, gens, stag, rr = 48, 120, 18, 2
    elif N <= 192:
        P, gens, stag, rr = 32, 80, 16, 1
    elif N <= 384:
        P, gens, stag, rr = 20, 45, 14, 1
    else:
        P, gens, stag, rr = 12, 25, 10, 0

    try:
        de_best, pop, scores = de_search(base, N, m, rng, P, gens, stag, rr)
        ds = exact_score(de_best, N)
        if ds < best_s - 1e-12:
            best = de_best.copy()
            best_s = ds

        cands = []
        add_candidate(cands, best, N, m, 1e-6)
        add_candidate(cands, base, N, m, 1e-6)
        add_candidate(cands, 1.0 - best, N, m, 1e-6)
        order = np.argsort(scores)
        for j in order[:8]:
            add_candidate(cands, pop[j], N, m, 1e-5)
            if len(cands) >= 4:
                break
        add_candidate(cands, 0.5 + 0.7 * (best - 0.5), N, m, 1e-5)
        if len(cands) < 4:
            add_candidate(cands, rng.uniform(size=N), N, m, 1e-5)

        if N <= 80:
            max_polish = 3
        elif N <= 200:
            max_polish = 1
        else:
            max_polish = 0

        for k in range(min(max_polish, len(cands))):
            cp = cands[k].copy()
            hp, hs = soft_polish(cp, N, m)
            if hs < best_s - 1e-12 and check_feasible(hp, N, m):
                best = hp.copy()
                best_s = hs
    except Exception:
        pass

    final = project_box_sum(best, m)
    fs = exact_score(final, N)
    if not np.isfinite(fs) or not check_feasible(final, N, m):
        final = project_box_sum(base, m)
        fs = exact_score(final, N)

    bs = exact_score(project_box_sum(base, m), N)
    if bs < fs - 1e-12:
        final = project_box_sum(base, m)
        fs = bs

    return final, float(fs), int(N)
