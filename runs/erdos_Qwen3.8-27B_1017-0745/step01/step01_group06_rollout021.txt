import cvxpy as cp

def project_capped(y, n, S):
    if not np.all(np.isfinite(y)):
        return np.full(n, S / n)
    lo, hi = float(y.min()) - 1.0, float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0, 1).sum() > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    h = np.clip(y - 0.5 * (lo + hi), 0, 1)
    e = S - h.sum()
    if abs(e) > 1e-12:
        if e > 0:
            idx = np.where(h < 1 - 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) > 0:
            h[idx] = np.clip(h[idx] + e / len(idx), 0, 1)
    if abs(h.sum() - S) > 1e-8:
        return np.full(n, S / n)
    return h

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-10 and h.max() <= 1 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def build_dc_data(n, dx):
    gamma = 1.25 * dx
    Lps, Lns = [], []
    for d in range(n):
        B = np.zeros((n, n))
        if d == 0:
            B = dx * np.eye(n)
        else:
            ii = np.arange(n - d)
            B[ii, ii + d] = dx / 2
            B[ii + d, ii] = dx / 2
        P = 0.5 * (gamma * np.eye(n) + B)
        N = 0.5 * (gamma * np.eye(n) - B)
        Lps.append(np.linalg.cholesky(P))
        Lns.append(np.linalg.cholesky(N))
    cvecs = []
    for k in range(2 * n - 1):
        ell = k - (n - 1)
        c = np.zeros(n)
        if ell == 0:
            c[:] = dx
        elif ell > 0:
            c[:n - ell] = dx
        else:
            c[-ell:] = dx
        cvecs.append(c)
    return Lps, Lns, cvecs

def dc_solve(x, n, dx, S, Lps, Lns, cvecs, rho, solver_name):
    y = cp.Variable(n)
    t = cp.Variable()
    obj = cp.Minimize(t + (rho / 2.0) * cp.sum_squares(y) - rho * (x @ y))
    cons = [y >= 0, y <= 1, cp.sum(y) == S]
    for k in range(2 * n - 1):
        ell = k - (n - 1)
        d = abs(ell)
        v = Lps[d] @ (Lps[d].T @ x)
        cons.append(cvecs[k] @ y + cp.sum_squares(Lns[d].T @ y) - 2.0 * (v @ y) + (x @ v) <= t)
    prob = cp.Problem(obj, cons)
    prob.solve(solver=solver_name, verbose=False, max_iter=200, abstol=1e-8, reltol=1e-8)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    if y.value is None or not np.all(np.isfinite(y.value)):
        return None
    return y.value
