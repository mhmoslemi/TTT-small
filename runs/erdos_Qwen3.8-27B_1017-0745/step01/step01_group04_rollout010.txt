def project_box_sum(z, m):
    N = len(z)
    if N == 0 or not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    if np.max(np.abs(z)) > 1e6:
        return np.full(N, 0.5)
    lo, hi = z.min() - 1.0, z.max()
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid

    r = m - h.sum()

            adj = min(r, 1.0 - h[i])

            r -= adj
            if r <= 1e-14:

    elif r < 0:

            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
            if r >= -1e-14:

    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1+1e-10:
        return np.full(N, 0.5)
    if abs(h.sum() - m) > 1e-8 * max(1, N):

    return np.clip(h, 0.0, 1.0)

    t_lo = max(-h[p], h[r] - 1.0)
    t_hi = min(1.0 - h[p], h[r])
    if t_hi - t_lo <= 1e-12:

        Q[N-1] += -2.0 * t * t
        if d != 0:
            Q[N-1+d] += t * t
            Q[N-1-d] += t * t
        return float(np.max(Q))

    coarse = np.linspace(t_lo, t_hi, 9)
    if 0.0 >= t_lo and 0.0 <= t_hi:

    coarse = np.unique(np.sort(coarse))

    for t in coarse:

        if qv < best_q - 1e-15 or (abs(qv - best_q) < 1e-15 and abs(t) < abs(best_t)):

            best_t = t

            hi_ref = grid[min(4, gi+1)]
    
    if best_q * dx < cur_max * dx - 1e-12:
