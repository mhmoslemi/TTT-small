import numpy as np

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-14:
        if resid > 0:
            order = np.argsort(1.0 - y)
        else:
            order = np.argsort(y)
        for i in order:
            if resid > 0:
                add = min(resid, 1.0 - y[i])
                y[i] += add
                resid -= add
            else:
                sub = min(-resid, y[i])
                y[i] -= sub
                resid += sub
            if abs(resid) < 1e-15:
                break
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)):
        return None
    if np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    if abs(np.sum(y) - S) > 1e-10 * max(1.0, float(n)):
        return None
    return y

def feasible(h, n, S):
    if h is None or h.shape != (n,):

        return False

        return False
    if abs(np.sum(h) - S) > 1e-10 * max(1.0, float(n)):
        return False
    return True

def exact_score(h, n):
    S = n / 2.0
    if not feasible(h, n, S):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    if len(corr) != 2 * n - 1 or not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr) * (2.0 / n))
