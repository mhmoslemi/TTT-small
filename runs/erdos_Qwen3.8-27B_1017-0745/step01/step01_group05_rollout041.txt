    return p

import numpy as np

def project_capped(y, n):
    S = n / 2.0
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return np.full(n, 0.5), False
    lo = float(y.min()) - 1.0
    hi = float(y.max())
    lam = float(np.clip(y.mean() - 0.5, lo, hi))
    for _ in range(60):
        c = np.clip(y - lam, 0.0, 1.0)
        r = c.sum() - S
        if abs(r) <= 1e-13:
            break
        if r > 0:
            lo = lam
        else:
            hi = lam
        a = ((y - lam) > 0) & ((y - lam) < 1)
        cnt = int(a.sum())
        if cnt > 0:
            ln = lam + r / cnt
        else:
            ln = 0.5 * (lo + hi)
        if not np.isfinite(ln) or ln <= lo or ln >= hi:
            ln = 0.5 * (lo + hi)
        lam = ln
    c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(3):
        r = S - c.sum()
        if abs(r) < 1e-14:
            break
        m = (c < 1.0 - 1e-12) if r > 0 else (c > 1e-12)
        cnt = int(m.sum())
        if cnt == 0:
            break
        c[m] += r / cnt
        np.clip(c, 0.0, 1.0, out=c)
    ok = (np.all(c >= -1e-12) and np.all(c <= 1.0 + 1e-12)
          and abs(c.sum() - S) <= 1e-10)
    return c, ok

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(h.sum() - n / 2.0) <= 1e-10)

def lag_scores(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    return float(lag_scores(h, n).max())

def build_G(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for K in range(-(n - 1), n):
        row = K + n - 1
        f_lo, f_hi = max(0, -K), min(n, n - K)
        s_lo, s_hi = max(0, K), min(n, n + K)
        if f_hi > f_lo:
            G[row, f_lo:f_hi] += dx * (1.0 - h[f_lo + K:f_hi + K])
        if s_hi > s_lo:
            G[row, s_lo:s_hi] -= dx * h[s_lo - K:s_hi - K]
    return G

def project_simplex(z):
    nz = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(nz, 1.0 / nz)
    sz = np.sort(z)[::-1]
    cs = np.cumsum(sz)
    rho = np.arange(1, nz + 1)
    cond = sz - (cs - 1.0) / rho > 0
    r = int(np.where(cond)[0][0]) if cond.any() else nz - 1
    theta = (cs[r] - 1.0) / (r + 1)
    p = np.maximum(z - theta, 0.0)
    s = p.sum()
    if s > 1e-15 and abs(s - 1.0) < 1e-12:
        pm = p > 0
        p[pm] += (1.0 - s) / int(pm.sum())
    return p

def dual_eval(p, h, q, G, n):
    dx = 2.0 / n
    L = 2.0 * dx
    gmix = G.T @ p
    y, ok = project_capped(h - gmix / L, n)
    if not ok:
        return None
    d = y - h
    u = q + G @ d
    D = float(p @ u + 0.5 * L * (d @ d))
    gap = float(u.max() - p @ u)
    return d, u, D, gap

def solve_majorant(h, q, G, n, max_inner):
    dx = 2.0 / n
    L = 2.0 * dx
    m = 2 * n - 1
    Ldual = max(1e-12, 1.05 * float(np.sum(G * G)) / L)
    qmax = float(q.max())
    thresh = max(qmax * 1e-6, 1e-12)
    active = np.where(q >= qmax - thresh)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(q))])
    p = np.zeros(m)
    p[active] = 1.0 / len(active)
    p = project_simplex(p)
    x = p.copy()
    yv = p.copy()
    theta = 1.0
    best_D = -float('inf')
    best_d = np.zeros(n)
    small_steps = 0
    for _ in range(max_inner):
        res_y = dual_eval(yv, h, q, G, n)
        if res_y is None:
            res_y = dual_eval(x, h, q, G, n)
            if res_y is None:
                break
            yv = x.copy()
        dy, uy, Dy, gap_y = res_y
        if Dy > best_D:
            best_D = Dy
            best_d = dy.copy()
        grad = q + G @ dy
        x_trial = project_simplex(yv + grad / Ldual)
        step_norm = np.linalg.norm(x_trial - x)
        if step_norm < 1e-10:
            small_steps += 1
        else:
            small_steps = 0
        res_t = dual_eval(x_trial, h, q, G, n)
        if res_t is None or res_t[2] < Dy - 1e-13:
            xt2 = project_simplex(x + grad / Ldual)
            res2 = dual_eval(xt2, h, q, G, n)
            if res2 is not None and res2[2] >= Dy - 1e-13:
                x, (dy, uy, Dy, _) = xt2, res2
                yv = x.copy()
                theta = 1.0
            else:
                x = x_trial.copy()
                yv = x.copy()
                theta = 1.0
                if res_t is not None and res_t[2] > best_D:
                    best_D = res_t[2]
                    best_d = res_t[0].copy()
        else:
            th_new = (1.0 + np.sqrt(1.0 + 4.0 * theta * theta)) / 2.0
            yv = x_trial + ((theta - 1.0) / th_new) * (x_trial - x)
            x = x_trial.copy()
            if res_t[2] > best_D:
                best_D = res_t[2]
                best_d = res_t[0].copy()
            theta = th_new
        if gap_y <= max(1e-10, 1e-8 * max(1.0, abs(qmax))):
            break
        if small_steps >= 3:
            break
    return best_d

def optimize_start(h0, n, max_outer, max_inner):
    h, ok = project_capped(h0.copy(), n)
    if not ok:
        h = np.full(n, 0.5)
    s = exact_score(h, n)
    best_h, best_s = h.copy(), s
    stale = 0
    for _ in range(max_outer):
        q = lag_scores(h, n)
        G = build_G(h, n)
        if not np.all(np.isfinite(G)):
            break
        d = solve_majorant(h, q, G, n, max_inner)
        if d is None or np.max(np.abs(d)) < 1e-10:
            break
        dx = 2.0 / n
        L = 2.0 * dx
        v = G @ d
        norm2 = float(d @ d)
        qmax = float(q.max())
        accepted = False
        prev_s = s
        for j in range(19):
            alpha = 2.0 ** (-j)
            Mval = float((q + alpha * v).max()) + 0.5 * L * alpha * alpha * norm2
            if Mval > qmax - 1e-14:
                continue
            ht, ok2 = project_capped(h + alpha * d, n)
            if not ok2:
                continue
            st = exact_score(ht, n)
            if st < s - 1e-12:
                h = ht
                s = st
                if s < best_s:
                    best_s = s
                    best_h = h.copy()
                stale = 0 if (prev_s - s) > 1e-11 else stale + 1
                accepted = True
                break
        if not accepted:
            break
        if stale >= 2:
            break
    return best_h, best_s

def coord_refine(h0, n, n_iter=2):
    h, ok = project_capped(h0.copy(), n)
    if not ok:
        return h0, exact_score(h0, n)
    s = exact_score(h, n)
    for _ in range(n_iter):
        improved = False
        for j in range(n):
            for delta in [0.02, -0.02, 0.005, -0.005]:
                ht = h.copy()
                ht[j] += delta
                ht, ok2 = project_capped(ht, n)
                if not ok2:
                    continue
                st = exact_score(ht, n)
                if st < s - 1e-12:
                    h, s = ht, st
                    improved = True
                    break
        if not improved:
            break
    return h, s
