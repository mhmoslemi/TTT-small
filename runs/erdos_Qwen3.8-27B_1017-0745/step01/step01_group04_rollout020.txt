def de_search(pop, N, m, P, max_gen, stag_lim, max_rest, rng):

    best_h = pop[np.argmin(scores)].copy()
    best_s = float(np.min(scores))
    F = 0.7
    CR = 0.85
    tol = 1e-12
    stag = 0
    nrest = 0
    
    for gen in range(max_gen):
        gen_start_best = best_s
        snap = [p.copy() for p in pop]
        snap_s = scores.copy()
        next_pop = [p.copy() for p in pop]
        
        for i in range(P):
            cands = [j for j in range(P) if j != i]
            idx = rng.choice(len(cands), 3, replace=False)
            r1, r2, r3 = cands[idx[0]], cands[idx[1]], cands[idx[2]]
            mut = snap[r1] + F * (snap[r2] - snap[r3])
            j_r = rng.integers(0, N)
            mask = rng.random(N) < CR
            mask[j_r] = True
            trial = np.where(mask, mut, snap[i])
            if not np.all(np.isfinite(trial)):
                continue
            trial = project_box_sum(trial, N, m)
            if not is_feasible(trial, N, m):

            if not np.isfinite(ts):
                continue
            if ts < snap_s[i] - tol:
                next_pop[i] = trial
                scores[i] = ts
                if ts < best_s - tol:
                    best_h = trial.copy()
                    best_s = ts
        
        pop = next_pop
        
        if best_s < gen_start_best - tol:
            stag = 0
        else:
            stag += 1
        
        if stag >= stag_lim and nrest < max_rest:
            # Restart
            si = 0.14 if nrest == 0 else 0.05
            nkeep = max(2, P // 4)
            order = np.argsort(scores)
            new_pop = [best_h.copy()]
            for k in range(1, nkeep):
                new_pop.append(pop[order[k]].copy())
            n_noise = (P - len(new_pop)) * 3 // 4
            n_unif = P - len(new_pop) - n_noise

                new_pop.append(project_box_sum(rng.uniform(0.0, 1.0, N), N, m))

                new_pop.append(best_h + si * rng.standard_normal(N))
            while len(new_pop) > P:
                new_pop.pop()
            # Validate and fix
            for k in range(P):
                new_pop[k] = project_box_sum(new_pop[k], N, m)

                scores[0] = best_s

    lo, hi = z.min() - 1.0, z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    for _ in range(2):
        if r > 1e-14:
            for i in range(N):
                adj = min(r, 1.0 - h[i]); h[i] += adj; r -= adj
                if r <= 1e-14: break
        elif r < -1e-14:
            for i in range(N):
                adj = min(-r, h[i]); h[i] -= adj; r += adj
                if r >= -1e-14: break
        else:
            break
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1.0 + 1e-10:
        return np.full(N, 0.5)
    if abs(h.sum() - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return h

def is_feasible(h, N, m):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return False
    return (h.min() >= -1e-12 and h.max() <= 1.0 + 1e-12
            and abs(h.sum() - m) <= 1e-10 * max(1, N))

def exact_score(h, N):
    try:
        c = np.correlate(h, 1.0 - h, mode='full')
        if not np.all(np.isfinite(c)):
            return np.inf
        return float(np.max(c) * 2.0 / N)
    except Exception:
        return np.inf

def make_fourier(N, rng, A, R):
    idx = np.arange(N)
    z = np.full(N, 0.5)
    R = min(R, (N - 1) // 2)
    for r in range(1, R + 1):
        z += (A / r) * (rng.standard_normal() * np.cos(2*np.pi*r*idx/N)
                        + rng.standard_normal() * np.sin(2*np.pi*r*idx/N))
    return z

def build_pop(base, N, m, P, rng):
    pop = [base.copy()]
    seen = [base.copy()]
    props = [1.0 - base, base[rng.permutation(N)]]
    for sd in [0.02, 0.06, 0.15, 0.30]:
        props.append(base + sd * rng.standard_normal(N))
    props.append(rng.uniform(0.0, 1.0, N))
    props.append(rng.integers(0, 2, N).astype(np.float64))
    props.append(make_fourier(N, rng, 0.12, 6))
    props.append(make_fourier(N, rng, 0.24, 6))
    for c in props:
        c = project_box_sum(c, N, m)
        if not is_feasible(c, N, m):
            c = np.full(N, 0.5)
        dup = any(np.abs(c - s).max() <= 1e-10 for s in seen)
        if not dup and len(pop) < P:
            pop.append(c); seen.append(c)
    att = 0
    while len(pop) < P and att < 20 * P:
        if att % 2 == 0:
            sd = 10.0 ** rng.uniform(np.log10(0.03), np.log10(0.35))
            c = base + sd * rng.standard_normal(N)
        else:
            c = rng.uniform(0.0, 1.0, N)
        c = project_box_sum(c, N, m)
        if not is_feasible(c, N, m):
            c = np.full(N, 0.5)
        dup = any(np.abs(c - s).max() <= 1e-10 for s in seen)
        if not dup:
            pop.append(c); seen.append(c)
        if len(pop) < P:
            att += 1
    while len(pop) < P:
        pop.append(pop[0].copy())
    return pop

def de_search(pop, N, m, P, max_gen, stag_lim, max_rest, rng):
    scores = np.array([exact_score(p, N) for p in pop])
    best_h = pop[np.argmin(scores)].copy()
    best_s = float(np.min(scores))
    F, CR, tol = 0.7, 0.85, 1e-12
    stag, nrest = 0, 0
    for gen in range(max_gen):
        gs = best_s
        snap = [p.copy() for p in pop]
        snap_s = scores.copy()
        nxt = [p.copy() for p in pop]
        for i in range(P):
            cands = [j for j in range(P) if j != i]
            pi = rng.choice(len(cands), 3, replace=False)
            r1, r2, r3 = cands[pi[0]], cands[pi[1]], cands[pi[2]]
            mut = snap[r1] + F * (snap[r2] - snap[r3])
            jr = rng.integers(0, N)
            mask = rng.random(N) < CR
            mask[jr] = True
            trial = np.where(mask, mut, snap[i])
            if not np.all(np.isfinite(trial)):
                continue
            trial = project_box_sum(trial, N, m)
            if not is_feasible(trial, N, m):
                continue
            ts = exact_score(trial, N)
            if not np.isfinite(ts) or ts >= snap_s[i] - tol:
                continue
            nxt[i] = trial; scores[i] = ts
            if ts < best_s - tol:
                best_h = trial.copy(); best_s = ts
        pop = nxt
        stag = 0 if best_s < gs - tol else stag + 1
        if stag >= stag_lim and nrest < max_rest:
            si = 0.14 if nrest == 0 else 0.05
            nk = max(2, P // 4)
            o = np.argsort(scores)
            np_ = [best_h.copy()] + [pop[o[k]].copy() for k in range(1, nk)]
            nn = (P - len(np_)) * 3 // 4
            nu = P - len(np_) - nn
            for _ in range(nn):
                np_.append(project_box_sum(best_h + si * rng.standard_normal(N), N, m))
            for _ in range(nu):
                np_.append(project_box_sum(rng.uniform(0.0, 1.0, N), N, m))
            while len(np_) < P:
                np_.append(project_box_sum(best_h + si * rng.standard_normal(N), N, m))
            for k in range(P):
                np_[k] = project_box_sum(np_[k], N, m)
                if not is_feasible(np_[k], N, m):
                    np_[k] = best_h.copy()
            pop = np_[:P]
            scores = np.array([exact_score(p, N) for p in pop])
            bi = np.argmin(scores)
            if scores[bi] > best_s + tol:
                pop[0] = best_h.copy(); scores[0] = best_s
            stag, nrest = 0, nrest + 1
        elif nrest >= max_rest and stag >= 2 * stag_lim:
            break
    return best_h, best_s

def local_refine(h, N, m, rng, n_it=200):
    h = project_box_sum(h, N, m)
    best = exact_score(h, N)
    tol = 1e-12
    for it in range(n_it):
        sd = 0.04 * (0.92 ** (it // 50))
        trial = project_box_sum(h + sd * rng.standard_normal(N), N, m)
        ts = exact_score(trial, N)
        if np.isfinite(ts) and ts < best - tol:
            h = trial; best = ts
    return h, best

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    N, raw = None, globals().get('initial_h_values', None)
    if raw is not None:
        raw = np.asarray(raw, dtype=np.float64)
        if raw.ndim == 1 and raw.size > 1 and np.all(np.isfinite(raw)):
            N = raw.size
    if N is None:
        N = kwargs.get('n_points', 70)
        if (not isinstance(N, (int, np.integer)) or isinstance(N, bool)
                or N < 1 or N > 999):
            N = 70
    N = int(N)
    m = N / 2.0
    if N == 1:
        h = np.array([0.5])
        return h, exact_score(h, 1), 1
    if raw is not None and raw.ndim == 1 and raw.size == N and np.all(np.isfinite(raw)):
        if (raw.min() >= -1e-12 and raw.max() <= 1.0 + 1e-12
                and abs(raw.sum() - m) < 1e-8):
            base = np.clip(raw, 0.0, 1.0)
        else:
            base = project_box_sum(raw, N, m)
    else:
        base = np.full(N, 0.5)
    base = project_box_sum(base, N, m)
    base_s = exact_score(base, N)
    if N > 999:
        return base, base_s, N
    if N <= 96:
        P, mg, sl, mr = 48, 120, 24, 2
    elif N <= 192:
        P, mg, sl, mr = 32, 80, 20, 1
    elif N <= 384:
        P, mg, sl, mr = 20, 40, 16, 1
    else:
        P, mg, sl, mr = 12, 20, 10, 0
    best_h, best_s = base, base_s
    try:
        pop = build_pop(base, N, m, P, rng)
        best_h, best_s = de_search(pop, N, m, P, mg, sl, mr, rng)
        best_h, best_s = local_refine(best_h, N, m, rng, 200)
    except Exception:
        pass
    best_h = project_box_sum(best_h, N, m)
    fs = exact_score(best_h, N)
    if not np.isfinite(fs) or fs > base_s + 1e-12:
        best_h = project_box_sum(base, N, m)
        fs = exact_score(base, N)
    if not np.isfinite(fs):
        best_h = np.full(N, 0.5)
        fs = exact_score(best_h, N)
    return best_h, float(fs), N
