import numpy as np

def project_onto_k(y, n):
    S = n / 2.0
    if not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        c = np.clip(y - mid, 0.0, 1.0)
        if c.sum() > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(20):
        diff = S - c.sum()
        if abs(diff) < 1e-14:
            break
        mask = (c < 1.0 - 1e-12) if diff > 0 else (c > 1e-12)
        cnt = int(mask.sum())
        if cnt == 0:
            break
        c[mask] += diff / cnt
        np.clip(c, 0.0, 1.0, out=c)
    return c

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(h.sum() - n / 2.0) <= 1e-10)

def lag_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def pair_coeffs(h, i, j, n):
    dx = 2.0 / n
    a = np.zeros(2 * n - 1)
    a[n - 1] = -2.0 * dx
    if i != j:
        ki = (i - j) + (n - 1)
        kj = (j - i) + (n - 1)
        if 0 <= ki < 2 * n - 1:
            a[ki] += dx
        if 0 <= kj < 2 * n - 1:
            a[kj] += dx
    b = np.zeros(2 * n - 1)
    for k in range(-(n - 1), n):
        t1 = 0.0
        ix = i - k
        if 0 <= ix < n:
            t1 = 1.0 - h[ix]
        t2 = 0.0
        ix = j - k
        if 0 <= ix < n:
            t2 = 1.0 - h[ix]
        t3 = 0.0
        ix = i + k
        if 0 <= ix < n:
            t3 = h[ix]
        t4 = 0.0
        ix = j + k
        if 0 <= ix < n:
            t4 = h[ix]
        b[k + (n - 1)] = dx * (t1 - t2 - t3 + t4)
    return b, a

def pair_line_search(h, i, j, n, q):
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_hi - t_lo < 1e-13:
        return None
    b, a = pair_coeffs(h, i, j, n)
    ts = np.linspace(t_lo, t_hi, 15)
    if t_lo <= 0.0 <= t_hi:
        ts = np.sort(np.unique(np.append(ts, 0.0)))
    best_f = float('inf')
    best_t = 0.0
    for t in ts:
        f = float(np.max(q + b * t + a * t * t))
        if f < best_f:
            best_f = f
            best_t = t
    step0 = (t_hi - t_lo) / max(1, len(ts) - 1)
    for _ in range(4):
        tl = max(t_lo, best_t - step0)

def exchange_polish(h0, n, rng, n_sweeps=3, pair_budget=None):
    h = h0.copy()
    q = lag_scores(h)
    s0 = float(np.max(q))
    if pair_budget is None:
        pair_budget = min(n * (n - 1) // 2, 3 * n)
    for _swe in range(n_sweps):
        accepted = 0
        pairs = []
        attempts = 0
        while len(pairs) < pair_budget and attempts < 50 * pair_budget:
            x = rng.integers(0, n)
            y = rng.integers(0, n)
            if x != y:
                pairs.append((x, y))
            attempts += 1
        seen = set()
        uniq_pairs = []
        for p in pairs:
            key = (min(p[0], p[1]), max(p[0], p[1]))
            if key not in seen:
                seen.add(key)
                uniq_pairs.append(key)
        rng.shuffle(uniq_pairs)
        for (i, j) in uniq_pairs:
            res = pair_line_search(h, i, j, n, q)
            if res is None:
                continue
            t_best, f_pred = res
            if f_pred >= s0 - 1e-12:
                continue
            h_try = h.copy()
            h_try[i] += t_best
            h_try[j] -= t_best
            h_try = project_onto_k(h_try, n)
            if not check_feasible(h_try, n):
                continue
            s_try = exact_score(h_try)
            if s_try < s0 - 1e-12:
                h = h_try
                q = lag_scores(h)
                s0 = s_try
                accepted += 1
        if accepted == 0:
            break
    return h, s0

def basin_hopping(h_best, s_best, n, rng, n_props=18):
    h_cur = h_best.copy()
    s_cur = s_best
    sigma = 0.08
    temps = [0.0015, 0.0005, 0.00015]
    blocks = [6, 4, 3]
    best_h = h_best.copy()
    best_s = s_best
    for bi in range(len(temps)):
        T = temps[bi]
        nb = blocks[bi]
        nacc = 0
        for _ in range(nb):
            z = rng.standard_normal(n)
            z -= z.mean()
            h_prop = project_onto_k(h_cur + sigma * z, n)
            if not check_feasible(h_prop, n):
                continue
            h_pol, s_pol = exchange_polish(h_prop, n, rng, n_sweeps=1,
                                           pair_budget=min(n * (n - 1) // 2, max(8, 3 * n // 4)))
            if s_pol < best_s - 1e-12:
                best_s = s_pol
                best_h = h_pol.copy()
            if s_pol <= s_cur:
                h_cur, s_cur = h_pol, s_pol
                nacc += 1
            else:
                ratio = min(745.0, (s_pol - s_cur) / T)
                if rng.random() < math.exp(-ratio):
                    h_cur, s_cur = h_pol, s_pol
                    nacc += 1
        acc_frac = nacc / max(1, nb)
        if acc_frac >= 0.6:
            sigma = min(0.25, sigma * 1.2)
        elif acc_frac <= 0.15:
            sigma = max(0.01, sigma * 0.8)
    return best_h, best_s

def run(seed=42, **kwargs):

    parent = globals().get('initial_h_values', None)

    if parent is not None:

    n_sweps = 3 if n <= 100 else (2 if n <= 400 else 1)

            best_h = h_pol.copy()

        best_s = fb_s

        best_s = bh_s
        best_h = bh_h.copy()

    if check_feasible(h_pol, n) and s_pol < best_s - 1e-12:
        best_s = s_pol
        best_h = h_pol.copy()

    final_h = project_onto_k(best_h, n)
    if not check_feasible(final_h, n):
        final_h = fallback.copy()
    final_s = exact_score(final_h)
    if final_s > fb_s + 1e-12:
        final_h = project_onto_k(fallback, n)
        final_s = exact_score(final_h)
    return final_h.astype(np.float64), float(final_s), int(n)
