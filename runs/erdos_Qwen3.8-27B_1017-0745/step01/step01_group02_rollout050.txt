        g[j] -= h[i]
    q *= dx
    g *= dx
    return q, g

def sep_max(h, d, n):
    dx = 2.0 / n
    best_v = -1e30
    best_i = 0
    for idx, s in enumerate(range(-(n - 1), n)):
        lo_i = max(0, -s)
        hi_i = min(n, n - s)
        q = 0.0
        for i in range(lo_i, hi_i):
            j = i + s
            q += h[i] * (1.0 - h[j]) + (1.0 - h[j]) * d[i] - h[i] * d[j]
        q *= dx
        if q > best_v:
            best_v = q
            best_i = idx
    return best_v, best_i

def mm_obj(z, x, L):
    n = len(z) - 1
    d = z[:n] - x
    return 0.5 * L * np.dot(d, d) + z[n]

def mm_obj_grad(z, x, L):
    n = len(z) - 1
    d = z[:n] - x
    g = np.empty(n + 1)
    g[:n] = L * d
    g[n] = 1.0
    return g

def mm_ineq(z, x, qJ, GJ):
    n = len(z) - 1
    d = z[:n] - x
    return z[n] - qJ - GJ @ d

def mm_ineq_jac(z, x, qJ, GJ):
    n = len(z) - 1
    nc = len(qJ)
    J = np.zeros((nc, n + 1))
    J[:, :n] = -GJ
    J[:, n] = 1.0
    return J

def mm_eq(z, n):
    return float(np.sum(z[:n]) - 0.5 * n)

def mm_eq_jac(z, n):
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def solve_mm(x, n):
    dx = 2.0 / n
    L = 2.0 * dx
    q_all = np.zeros(2 * n - 1)
    G_all = np.zeros((2 * n - 1, n))
    for idx, s in enumerate(range(-(n - 1), n)):
        q, g = lag_val_grad(x, s, n)
        q_all[idx] = q
        G_all[idx, :] = g
    if n <= 64:
        active = list(range(2 * n - 1))
    else:
        active = list(np.argsort(-q_all)[:min(96, 2 * n - 1)])
    y_curr = x.copy()
    for _ in range(8):
        qJ = q_all[active]
        GJ = G_all[active]
        d = y_curr - x
        t0 = max(float(np.max(qJ + GJ @ d)), 1e-6) + L * np.dot(d, d) * 0.5
        z0 = np.append(y_curr.copy(), t0)
        bounds = [(0.0, 1.0)] * n + [(None, None)]
        cons = [{'type': 'ineq', 'fun': mm_ineq, 'jac': mm_ineq_jac, 'args': (x, qJ, GJ)},
                {'type': 'eq', 'fun': mm_eq, 'jac': mm_eq_jac, 'args': (n,)}]
        try:
            res = minimize(mm_obj, z0, jac=mm_obj_grad, args=(x, L),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 150, 'ftol': 1e-11, 'disp': False})
        except Exception:
            break
        if not np.all(np.isfinite(res.x)):
            break
        yr = res.x[:n]
        if np.any(yr < -1e-4) or np.any(yr > 1.0 + 1e-4):
            break
        y_new = project_box_sum(yr, 0.5 * n)
        if y_new is None:
            break
        d_new = y_new - x
        EJ = float(np.max(qJ + GJ @ d_new))
        fv, fi = sep_max(x, d_new, n)
        if fv <= EJ + 1e-8:
            mv = 0.5 * L * np.dot(d_new, d_new) + EJ
            if mv > float(np.max(q_all)) + 1e-6:
                break
            y_curr = y_new
            break
        else:
            if fi not in active and len(active) < min(2 * n - 1, 120):
                active.append(fi)
            y_curr = y_new
    if check_feasible(y_curr, n):
        return y_curr
    return x.copy()

def local_mm(h0, n, max_steps):
    h = h0.copy()
    c = exact_score(h, n)
    for _ in range(max_steps):
        prop = solve_mm(h, n)
        d = prop - h
        if np.max(np.abs(d)) < 1e-11:
            break
        best_c = c
        best_a = None
        na = 9 if n <= 128 else 6
        for k in range(na):
            a = 2.0 ** (-k)
            cand = project_box_sum(h + a * d, 0.5 * n)
            if cand is None:
                continue
            ct = exact_score(cand, n)
            if ct < best_c - 1e-13:
                best_c = ct
                best_a = a
        if best_a is None:
            break
        h = project_box_sum(h + best_a * d, 0.5 * n)
        if h is None:
            break
        c = exact_score(h, n)
        if best_c >= c:
            c = best_c
    return h, c

def get_parent():
    n_def = 45
    if 'initial_h_values' in globals():
        ph = np.asarray(globals()['initial_h_values'], dtype=np.float64)
        if ph.ndim == 1 and 2 <= len(ph) <= 999 and np.all(np.isfinite(ph)):
            if np.all(ph >= -1e-6) and np.all(ph <= 1.0 + 1e-6):
                n = len(ph)
                p = project_box_sum(np.clip(ph, 0.0, 1.0), 0.5 * n)
                if p is not None:
                    return p, n
    return project_box_sum(np.full(n_def, 0.5), 0.5 * n_def), n_def

def build_starts(parent, n, rng):
    starts = [parent.copy(), (1.0 - parent).copy(), parent[::-1].copy()]
    for sd in [0.05, 0.10, 0.20, 0.35]:
        p = project_box_sum(parent + sd * rng.standard_normal(n), 0.5 * n)
        if p is not None:
            starts.append(p)
    p = project_box_sum(rng.random(n), 0.5 * n)
    if p is not None:
        starts.append(p)
    for (A1, m1, p1, A2, m2, p2) in [
        (0.30, 1, 0.5, 0.12, 2, 1.3),
        (0.25, 1, 0.0, 0.15, 3, 0.7),
        (0.40, 2, 1.2, 0.10, 1, 0.3),
    ]:
        u = 0.5 + A1 * np.cos(2.0 * np.pi * m1 * np.arange(n) / n + p1) \
            + A2 * np.sin(2.0 * np.pi * m2 * np.arange(n) / n + p2)
        p = project_box_sum(u, 0.5 * n)
        if p is not None:
            starts.append(p)
    unique = []
    for s in starts:
        dup = False
        for u in unique:
            if np.max(np.abs(s - u)) < 1e-10:
                dup = True
                break
        if not dup:
            unique.append(s)
    return unique

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent, n = get_parent()
    best_h = parent.copy()
    best_c5 = exact_score(parent, n)
    m = max(8, min(40, int(1800 // max(1, n))))
    starts = build_starts(parent, n, rng)
    for h0 in starts:
        try:
            h_r, c_r = local_mm(h0, n, m)
            if c_r < best_c5:
                best_h = h_r.copy()
                best_c5 = c_r
        except Exception:
            pass
    restart_sds = [0.03, 0.08, 0.15, 0.28] if n <= 64 else [0.06, 0.18]
    for sd in restart_sds:
        try:
            h0 = project_box_sum(best_h + sd * rng.standard_normal(n), 0.5 * n)
            if h0 is None:
                continue
            h_r, c_r = local_mm(h0, n, max(4, m // 2))
            if c_r < best_c5:
                best_h = h_r.copy()
                best_c5 = c_r
        except Exception:
            pass
    best_h = project_box_sum(best_h, 0.5 * n)
    if best_h is not None and check_feasible(best_h, n):
        best_c5 = exact_score(best_h, n)
        pc = exact_score(parent, n)
        if pc < best_c5:
            return (parent, pc, n)
        return (best_h, best_c5, n)
    fb = np.full(n, 0.5)
    return (fb, exact_score(fb, n), n)
