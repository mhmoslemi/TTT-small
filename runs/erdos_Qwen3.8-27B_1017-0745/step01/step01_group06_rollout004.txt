    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    err = S - h.sum()
    if abs(err) > 1e-12:
        if err > 0:
            idx = np.where(1.0 - h > 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) > 0:
            h[idx] = np.clip(h[idx] + err / len(idx), 0.0, 1.0)
    if abs(h.sum() - S) > 1e-8:
        return np.full(n, S / n)
    return h

def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def lag_vec(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_grad(h, n, dx):
    nl = 2 * n - 1
    G = np.zeros((nl, n))
    j = np.arange(n)
    for k in range(nl):
        ell = k - (n - 1)
        i1, v1 = j - ell, (j - ell >= 0) & (j - ell < n)
        i2, v2 = j + ell, (j + ell >= 0) & (j + ell < n)
        t1, t2 = np.zeros(n), np.zeros(n)
        t1[v1] = 1.0 - h[i1[v1]]
        t2[v2] = h[i2[v2]]
        G[k] = dx * (t1 - t2)
    return G

def _sop_obj(z, eta):
    d = z[:-1]
    return z[-1] + np.dot(d, d) / (2.0 * eta)

def _sop_obj_jac(z, eta):
    g = np.zeros(len(z))
    g[:-1] = z[:-1] / eta
    g[-1] = 1.0
    return g

def _sop_ineq(z, q, G):
    return z[-1] - q - G @ z[:-1]

def _sop_ineq_jac(z, q, G):
    n = G.shape[1]
    J = np.zeros((G.shape[0], n + 1))
    J[:, :-1] = -G
    J[:, -1] = 1.0
    return J

def _sop_eq(z, n):
    return z[:-1].sum()

def _sop_eq_jac(z, n):
    g = np.zeros(len(z))
    g[:-1] = 1.0
    return g

def trust_step(h, n, dx, delta, eta, miter):
    q = lag_vec(h, dx)
    G = lag_grad(h, n, dx)
    z0 = np.zeros(n + 1)
    z0[-1] = q.max()
    bounds = []
    for j in range(n):
        lb = max(-delta, -h[j] + 1e-14)
        ub = min(delta, 1.0 - h[j] - 1e-14)
        bounds.append((lb, ub) if lb <= ub else (0.0, 0.0))
    bounds.append((None, None))
    con = [{'type': 'eq', 'fun': _sop_eq, 'jac': _sop_eq_jac, 'args': (n,)},
           {'type': 'ineq', 'fun': _sop_ineq, 'jac': _sop_ineq_jac, 'args': (q, G)}]
    try:
        res = minimize(_sop_obj, z0, jac=_sop_obj_jac, args=(eta,),
                       method='SLSQP', bounds=bounds, constraints=con,
                       options={'maxiter': miter, 'ftol': 1e-12})
    except Exception:
        return None
    if not hasattr(res, 'x') or not np.all(np.isfinite(res.x)):
        return None
    d = res.x[:-1]
    if abs(d.sum()) > 1e-4:
        return None
    if np.any(res.x[-1] - (q + G @ d) < -1e-4):
        return None
    return d

def local_opt(h0, n, dx, max_iter, mslq):
    S = n / 2.0
    h = project_K(h0.copy(), n, S)
    sc = exact_score(h, dx)
    delta, eta, rej = 0.12, 0.5, 0
    for _ in range(max_iter):
        d = trust_step(h, n, dx, delta, eta, mslq)
        if d is None:
            delta *= 0.5
            rej += 1
            if delta < 1e-5 or rej >= 8:
                break
            continue
        h_t = project_K(h + d, n, S)
        if not feasible(h_t, n, S):
            delta *= 0.5
            rej += 1
            continue
        d_a = h_t - h
        if np.max(np.abs(d_a)) < 1e-10:
            break
        sc_t = exact_score(h_t, dx)
        q = lag_vec(h, dx)
        G = lag_grad(h, n, dx)
        pred = q.max() - (q + G @ d_a).max()
        actual = sc - sc_t
        if actual > 1e-12:
            h, sc, rej = h_t, sc_t, 0
            if pred > 1e-12:
                r = actual / pred
                if r < 0.25:
                    delta *= 0.5
                elif r > 0.75 and np.max(np.abs(d_a)) >= 0.8 * delta:
                    delta = min(0.5, 2.0 * delta)
                    eta = min(2.0, 1.5 * eta)
            else:
                delta *= 0.5
        else:
            delta *= 0.5
            eta = max(1e-4, eta * 0.5)
            rej += 1
        if delta < 1e-5 or rej >= 8:
            break
    return h, sc

def transfer_par(hp, n0, m):
    y = np.zeros(m)
    for j in range(m):
        a, b = j / m, (j + 1) / m
        ov = 0.0
        for i in range(n0):
            a0, b0 = i / n0, (i + 1) / n0
            ov += hp[i] * max(0.0, min(b0, b) - max(a0, a))
        y[j] = m * ov
    return project_K(y, m, m / 2.0)

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    init = globals().get('initial_h_values', None)
    nd = kwargs.get('n_points', 45)
    if not isinstance(nd, (int, np.integer)):
        nd = 45
    nd = max(2, min(999, int(nd)))
    ok = False
    if init is not None:
        init = np.array(init, dtype=np.float64).ravel()
        if (2 <= len(init) <= 999 and np.all(np.isfinite(init)) and
                init.min() >= -0.01 and init.max() <= 1.01):
            ok = True
    n0 = len(init) if ok else nd
    parent = project_K(init.copy(), n0, n0 / 2.0) if ok else np.full(n0, 0.5)
    fb_h, fb_s, fb_n = parent.copy(), exact_score(parent, 2.0 / n0), n0
    best_h, best_n, best_s = parent.copy(), n0, fb_s
    for m in [n0]:
        Sm, dxm = m / 2.0, 2.0 / m
        base = parent.copy()
        starts = [base.copy(), (1.0 - base).copy(), base[::-1].copy(),
                  project_K(base + 0.08 * rng.standard_normal(m), m, Sm),
                  project_K(base + 0.25 * rng.standard_normal(m), m, Sm),
                  project_K(rng.uniform(0, 1, m), m, Sm),
                  project_K(0.5 + 0.4 * np.cos(2 * np.pi * np.arange(m) / m), m, Sm)]
        hb = np.zeros(m)
        hb[:int(Sm)] = 1.0
        if Sm - int(Sm) > 1e-12:
            hb[int(Sm)] = Sm - int(Sm)
        starts.append(project_K(hb, m, Sm))
        perm = rng.permutation(m)
        starts.append(project_K(base[perm], m, Sm))
        dd, seen = [], []
        for s in starts:
            s = project_K(s, m, Sm)
            dup = any(np.max(np.abs(s - p)) < 1e-6 for p in dd)
            if not dup:
                dd.append(s)
        for s in dd:
            sc = exact_score(s, dxm)
            if sc < best_s - 1e-12:
                best_s, best_h, best_n = sc, s.copy(), m
        for s in dd[:8]:
            hopt, sopt = local_opt(s, m, dxm, 30, 50)
            if feasible(hopt, m, Sm):
                sc = exact_score(hopt, dxm)
                if sc < best_s - 1e-12:
                    best_s, best_h, best_n = sc, hopt.copy(), m
        hr = project_K(best_h + 0.15 * rng.standard_normal(m), m, Sm)
        hopt, _ = local_opt(hr, m, dxm, 25, 50)
        if feasible(hopt, m, Sm):
            sc = exact_score(hopt, dxm)
            if sc < best_s - 1e-12:
                best_s, best_h, best_n = sc, hopt.copy(), m
        hr = project_K(best_h + 0.35 * rng.standard_normal(m), m, Sm)
        hopt, _ = local_opt(hr, m, dxm, 20, 50)
        if feasible(hopt, m, Sm):
            sc = exact_score(hopt, dxm)
            if sc < best_s - 1e-12:
                best_s, best_h, best_n = sc, hopt.copy(), m
    for m in [n0 - 1, n0 + 1]:
        if m < 2 or m > 999:
            continue
        Sm, dxm = m / 2.0, 2.0 / m
        base = transfer_par(parent, n0, m) if ok else np.full(m, 0.5)
        base = project_K(base, m, Sm)
        starts = [base.copy(), (1.0 - base).copy(), base[::-1].copy(),
                  project_K(base + 0.1 * rng.standard_normal(m), m, Sm),
                  project_K(base + 0.3 * rng.standard_normal(m), m, Sm)]
        dd = []
        for s in starts:
            s = project_K(s, m, Sm)
            if not any(np.max(np.abs(s - p)) < 1e-6 for p in dd):
                dd.append(s)
        for s in dd:
            sc = exact_score(s, dxm)
            if sc < best_s - 1e-12:
                best_s, best_h, best_n = sc, s.copy(), m
        for s in dd[:5]:
            hopt, _ = local_opt(s, m, dxm, 25, 45)
            if feasible(hopt, m, Sm):
                sc = exact_score(hopt, dxm)
                if sc < best_s - 1e-12:
                    best_s, best_h, best_n = sc, hopt.copy(), m
    best_h = project_K(best_h, best_n, best_n / 2.0)
    fs = exact_score(best_h, 2.0 / best_n)
    if not feasible(best_h, best_n, best_n / 2.0) or fs > fb_s + 1e-10:
        best_h, best_s, best_n = fb_h.copy(), fb_s, fb_n
        fs = exact_score(best_h, 2.0 / best_n)
    best_h = np.clip(best_h, 0.0, 1.0)
    return (best_h, float(fs), int(best_n))
