    init = globals().get('initial_h_values', None)
    got_parent = False
    n = 45
    if init is not None:
        try:
            init = np.array(init, dtype=np.float64).ravel()
            if 2 <= len(init) <= 999 and np.all(np.isfinite(init)):
                n = len(init)
                if (init.min() >= -1e-8 and init.max() <= 1.0 + 1e-8 and
                        abs(init.sum() - n / 2.0) <= 1e-6):
                    got_parent = True
        except Exception:
            pass
    if not got_parent:
        nd = kwargs.get('n_points', 45)
        if isinstance(nd, (int, np.integer)) and not isinstance(nd, bool) and 2 <= int(nd) <= 999:
            n = int(nd)
    S = n / 2.0
    if got_parent:
        base = project_K(init.copy(), n, S)
    else:
        base = np.full(n, 0.5)
    fb_h = base.copy()
    fb_s = exact_score(fb_h, n)
    best_h = base.copy()
    best_s = fb_s
    if n <= 64:
        P, gens, max_rs, st_all = 72, 120, 2, 18
    elif n <= 128:
        P, gens, max_rs, st_all = 48, 60, 1, 12
    elif n <= 256:
        P, gens, max_rs, st_all = 32, 30, 1, 8
    else:
        P, gens, max_rs, st_all = 20, 12, 0, 999
    try:
        pop = build_population(base, n, S, rng, P)
        scores = np.empty(P)
        for i in range(P):
            scores[i] = exact_score(pop[i], n)
            if np.isfinite(scores[i]) and scores[i] < best_s - 1e-12:
                best_s = scores[i]
                best_h = pop[i].copy()
        mu_F, mu_CR = 0.55, 0.75
        stall, nrs, prev_bs = 0, 0, best_s
        for g in range(gens):
            pop, scores, mu_F, mu_CR, best_h, best_s = de_one_gen(
                pop, scores, n, S, mu_F, mu_CR, rng, best_h, best_s)
            if best_s < prev_bs - 1e-12:
                stall = 0
            else:
                stall += 1
            prev_bs = best_s
            if stall >= st_all and nrs < max_rs and g < gens - 1:
                nrs += 1
                rk = np.argsort(scores, kind='stable')
                keep = max(2, math.ceil(0.2 * P))
                sigs = [0.08, 0.18, 0.35]
                for slot in range(keep, P):
                    mm = (slot - keep) % 4
                    if mm < 3:
                        c = best_h + sigs[mm] * rng.standard_normal(n)
                    else:
                        c = rng.beta(0.5, 0.5, n)
                    h = project_K(c, n, S)
                    if feasible_check(h, n, S):
                        sc = exact_score(h, n)
                        if np.isfinite(sc) and sc < best_s - 1e-12:
                            best_s = sc
                            best_h = h.copy()
                        pop[slot] = h
                        scores[slot] = sc
                    else:
                        pop[slot] = best_h.copy()
                        scores[slot] = best_s
                mu_F, mu_CR = 0.7, 0.8
                stall = 0
        best_h, best_s = random_refine(best_h, best_s, n, S, rng)
    except Exception:
        pass
    best_h = project_K(best_h, n, S)
    if not feasible_check(best_h, n, S):
        best_h, best_s = fb_h.copy(), fb_s
    else:
        fs = exact_score(best_h, n)
        if not np.isfinite(fs) or fs > fb_s + 1e-10:
            best_h, best_s = fb_h.copy(), fb_s
        else:
            best_s = fs
    best_h = np.clip(best_h, 0.0, 1.0)
    final_s = exact_score(best_h, n)
    return (best_h, final_s, n)
