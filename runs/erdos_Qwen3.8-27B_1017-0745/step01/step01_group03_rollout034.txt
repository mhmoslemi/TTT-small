def build_G(x, n):
    L = 2 * n - 1
    G = np.zeros((L, n))
    for idx in range(L):
        k = idx - (n - 1)
        if k >= 0:
            for r in range(k, n):
                G[idx, r] += (1.0 - x[r - k])
            for r in range(0, n - k):
                G[idx, r] -= x[r + k]
        else:
            kp = -k
            for r in range(0, n - kp):
                G[idx, r] += (1.0 - x[r + kp])
            for r in range(kp, n):
                G[idx, r] -= x[r - kp]
    return G

def solve_surrogate(x, Q, n, warm):
    L = 2 * n - 1
    S = n / 2.0
    G = build_G(x, n)
    Ldual = max(1e-8, 0.5001 * float(np.sum(G * G)))
    if warm is not None and np.all(np.isfinite(warm)) and len(warm) == L:
        w = project_simplex(warm)
        if w is None:
            w = np.zeros(L)
            w[int(np.argmax(Q))] = 1.0
    else:
        w = np.zeros(L)
        w[int(np.argmax(Q))] = 1.0
    v = w.copy()
    t = 1.0
    best_y = x.copy()
    best_model = float(np.max(Q))
    Imax = max(200, min(400, 14000 // max(1, n) + 1))
    gap_check = max(10, Imax // 25)
    for it in range(Imax):
        z = G.T @ v
        y = project_box_sum(x - z / 2.0, S)
        if y is None:
            break
        d = y - x
        vals = Q + G @ d
        model = float(np.dot(d, d)) + float(np.max(vals))
        if model < best_model:
            best_model = model
            best_y = y.copy()
        gphi = Q + G @ d
        w_new = project_simplex(v + gphi / Ldual)
        if w_new is None:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        v_new = w_new + ((t - 1.0) / t_new) * (w_new - w)
        w, v, t = w_new, v_new, t_new
        if (it + 1) % gap_check == 0:
            z2 = G.T @ w
            y2 = project_box_sum(x - z2 / 2.0, S)
            if y2 is not None:
                d2 = y2 - x
                vals2 = Q + G @ d2
                gap = float(np.max(vals2) - float(np.dot(w, vals2)))
                if gap <= 1e-7 * max(1.0, float(np.max(np.abs(Q)))):
                    break
    return best_y, w

def mm_local(start, n):
    S = n / 2.0
    h = project_box_sum(start, S)
    if h is None:
        return None
    score = exact_score(h)
    Omax = max(4, min(20, 700 // max(1, n) + 1))
    warm = None
    for _ in range(Omax):
        Q = np.correlate(h, 1.0 - h, mode='full')
        y_new, warm = solve_surrogate(h, Q, n, warm)
        if y_new is None or not np.all(np.isfinite(y_new)):
            break
        d = y_new - h
        if np.max(np.abs(d)) < 1e-10:
            break
        best_a = None
        best_sc = score
        for a in [1.0, 0.75, 0.5, 0.25, 0.125]:
            cand = h + a * d
            if np.all(cand >= -1e-12) and np.all(cand <= 1.0 + 1e-12):
                cand = np.clip(cand, 0.0, 1.0)
                sc = exact_score(cand)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    best_a = cand.copy()
        if best_a is None:
            break
        h = best_a
        score = best_sc
        if score < 0.38:
            break
    return h

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent = None
    try:
        pv = np.array(initial_h_values, dtype=np.float64)
        if pv.ndim == 1 and 2 <= len(pv) <= 999:
            n = len(pv)
            if np.all(np.isfinite(pv)) and np.all(pv >= -1e-9) and np.all(pv <= 1.0 + 1e-9):
                if abs(np.sum(pv) - n / 2.0) < 1e-6 * max(1.0, float(n)):
                    pv = np.clip(pv, 0.0, 1.0)
                    parent = project_box_sum(pv, n / 2.0)
    except Exception:
        pass
    if parent is None:
        n = 45
        parent = project_box_sum(np.full(n, 0.5), n / 2.0)
    else:
        n = len(parent)
    S = n / 2.0
    fallback = parent.copy()
    fallback_sc = exact_score(fallback)
    starts = [parent.copy()]
    if n <= 128:
        starts.append(parent[::-1].copy())
        starts.append(1.0 - parent)
        starts.append(np.roll(parent, n // 3))
        for sig in [0.03, 0.08, 0.15]:
            p = project_box_sum(parent + sig * rng.standard_normal(n), S)
            if p is not None:
                starts.append(p)
        x_idx = np.arange(n)
        for m in [1, 2, 3]:
            ph = rng.uniform(0, 2.0 * np.pi)
            harm = 0.12 * np.cos(2.0 * np.pi * m * x_idx / n + ph)
            p = project_box_sum(parent + harm, S)
            if p is not None:
                starts.append(p)
        for _ in range(3):
            p = project_box_sum(rng.uniform(0.0, 1.0, n), S)
            if p is not None:
                starts.append(p)
    else:
        starts.append(parent[::-1].copy())
        starts.append(1.0 - parent)
        p = project_box_sum(parent + 0.08 * rng.standard_normal(n), S)
        if p is not None:
            starts.append(p)
    dedup = [starts[0]]
    for s in starts[1:]:
        if s is None or not np.all(np.isfinite(s)):
            continue
        ok = True
        for d in dedup:
            if np.max(np.abs(s - d)) < 1e-10:
                ok = False
                break
        if ok:
            dedup.append(s)
    best_h = fallback.copy()
    best_score = fallback_sc
    for s in dedup:
        h_res = mm_local(s, n)
        if h_res is not None and is_feasible(h_res, n):
            sc = exact_score(h_res)
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_res.copy()
    final_h = project_box_sum(best_h, S)
    if final_h is None or not is_feasible(final_h, n):
        final_h = fallback
    final_score = exact_score(final_h)
    return (np.array(final_h, dtype=np.float64), final_score, int(n))
