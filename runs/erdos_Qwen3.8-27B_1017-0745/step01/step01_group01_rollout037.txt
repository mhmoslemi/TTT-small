        if sy > ref_score: dms.append(sy - ref_score)
    T_hot = 0.002 if not dms else max(1e-5, min(0.02, float(np.median(dms)) / 0.2231))
    T_cold = max(1e-6, 0.02 * T_hot)
    temps = np.array([T_cold * (T_hot/T_cold)**(j/max(1,M-1)) for j in range(M)])
    rhos = np.array([0.02 + 0.12*j/max(1,M-1) for j in range(M)])
    states, scores = [best.copy()], [best_score]
    for st in starts:
        if len(states) >= M: break
        if any(float(np.max(np.abs(st-s2))) < 1e-10 for s2 in states): continue
        states.append(st.copy()); scores.append(exact_score(st, n))
    cnt = 0
    while len(states) < M and cnt < 30:
        base = best if cnt % 2 == 0 else (starts[cnt % len(starts)] if starts else best)
        sd2 = [0.03, 0.08, 0.18][cnt % 3]
        cand = project_capped(base + rng.normal(0, sd2, n), n)
        if cand is None: cand = base.copy()
        if any(float(np.max(np.abs(cand-s2))) < 1e-10 for s2 in states):
            cnt += 1; continue
        states.append(cand); scores.append(exact_score(cand, n))
        cnt += 1
    while len(states) < M:
        states.append(states[0].copy()); scores.append(scores[0])
    stagnation, n_restarts = 0, 0
    for sweep in range(steps):
        improved = False
        for j in range(M):
            y = chord_proposal(states[j], rng, rhos[j], n)
            if y is None: continue
            if float(np.max(np.abs(y - states[j]))) < 1e-14: continue
            sy = exact_score(y, n)
            if not np.isfinite(sy): continue
            d = sy - scores[j]
            accept = False
            if d <= 0:
                accept = True
            elif d / temps[j] < 745:
                accept = rng.random() < math.exp(-d / temps[j])
            if accept:
                states[j], scores[j] = y, sy
                if sy < best_score - 1e-12:
                    best, best_score = y.copy(), sy
                    improved = True
        parity = sweep % 2
        for j in range(parity, M - 1, 2):
            i2 = j
            lr = (scores[i2] - scores[j+1]) * (1.0/temps[i2] - 1.0/temps[j+1])
            if lr >= 0 or math.log(max(rng.random(), 1e-300)) < lr:
                states[i2], states[j+1] = states[j+1].copy(), states[i2].copy()
                scores[i2], scores[j+1] = scores[j+1], scores[i2]
        stagnation = 0 if improved else stagnation + 1
        patience = max(50, steps // 5)
        if stagnation >= patience:
            if n_restarts < max_restarts:
                for j in range(max(1, M//2), M):
                    base = best if j % 2 == 0 else (starts[j % len(starts)] if starts else best)
                    sd2 = [0.03, 0.08, 0.18][j % 3]
                    cand = project_capped(base + rng.normal(0, sd2, n), n)
                    if cand is None: cand = base.copy()
                    states[j] = cand
                    scores[j] = exact_score(cand, n)
                    if scores[j] < best_score - 1e-12:
                        best, best_score = cand.copy(), scores[j]
                n_restarts += 1; stagnation = 0
            else:
                break
    cands = [best]
    if parent is not None: cands.append(parent)
    cands.append(uniform)
    best_sel, best_sel_score = best.copy(), best_score
    for c in cands:
        cs = exact_score(c, n)
        if np.isfinite(cs) and cs < best_sel_score - 1e-12:
            best_sel, best_sel_score = c.copy(), cs
    final = project_capped(best_sel, n)
    if final is not None:
        fs = exact_score(final, n)
        if np.isfinite(fs) and fs <= best_sel_score + 1e-12:
            best_sel, best_sel_score = final, fs
    if not np.isfinite(best_sel_score):
        best_sel, best_sel_score = uniform, exact_score(uniform, n)
    return best_sel, float(best_sel_score), int(n)
