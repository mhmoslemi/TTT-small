        return np.full(m, 1.0 / m)

    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full(m, 1.0 / m)
    return w

import math

def project_capped(y, n):
    S = n / 2.0
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return np.full(n, 0.5), False
    lo = float(y.min()) - 1.0
    hi = float(y.max())
    lam = float(np.clip(np.mean(y) - 0.5, lo, hi))
    for _ in range(60):
        c = np.clip(y - lam, 0.0, 1.0)
        r = c.sum() - S
        if abs(r) <= 1e-13:
            break
        if r > 0:
            lo = lam
        else:
            hi = lam
        active = (y - lam > 0.0) & (y - lam < 1.0)
        a = active.sum()
        if a > 0:
            lam_new = lam + r / a
        else:
            lam_new = 0.5 * (lo + hi)
        if np.isfinite(lam_new) and lo < lam_new < hi:
            lam = lam_new
        else:
            lam = 0.5 * (lo + hi)
    c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(3):
        r = S - c.sum()
        if abs(r) < 1e-14:
            break
        if r > 0:
            mask = c < 1.0 - 1e-12
        else:
            mask = c > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        c[mask] += r / cnt
        np.clip(c, 0.0, 1.0, out=c)
    ok = (c.shape == (n,) and np.all(np.isfinite(c)) and np.all(c >= -1e-12)
          and np.all(c <= 1.0 + 1e-12) and abs(c.sum() - S) <= 1e-10)

    return c, True

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(h.sum() - n / 2.0) <= 1e-10)

def lag_scores(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    qs = lag_scores(h, n)
    if qs.shape != (2 * n - 1,) or not np.all(np.isfinite(qs)):
        return float('inf')
    return float(qs.max())

def build_grad_matrix(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        lo = max(0, k)
        hi = min(n, n + k)
        row = k + n - 1
        if hi > lo:
            G[row, lo:hi] += dx * (1.0 - h[lo - k:hi - k])
        lo2 = max(0, lo - k)
        hi2 = min(n, hi - k)
        if hi2 > lo2:
            G[row, lo2:hi2] -= dx * h[lo2 + k:hi2 + k]
    if not np.all(np.isfinite(G)):
        return None
    return G

def project_simplex(z):
    m = len(z)
    if m == 0 or not np.all(np.isfinite(z)):
        return np.full(m, 1.0 / m)
    sz = np.sort(z)[::-1]
    css = np.cumsum(sz)
    idx = np.where(sz - (css - 1.0) / np.arange(1, m + 1) > 0)[0]
    if len(idx) == 0:
        return np.full(m, 1.0 / m)
    r = idx[-1]
    theta = (css[r] - 1.0) / (r + 1.0)
    w = np.maximum(z - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full(m, 1.0 / m)
    return w

def dual_eval(p, h, G, q, n, L):
    n_lags = 2 * n - 1
    gmix = G.T @ p
    y, ok = project_capped(h - gmix / L, n)
    if not ok:
        return None
    d = y - h
    u = q + G @ d
    if not np.all(np.isfinite(u)):
        return None
    val = p @ u + (L / 2.0) * (d @ d)
    gap = u.max() - p @ u
    return d, u, val, gap

def solve_majorant(h, q, G, n, max_inner):
    dx = 2.0 / n
    L = 2.0 * dx
    n_lags = 2 * n - 1
    qmax = q.max()
    tol = max(1e-12, 1e-8 * max(1.0, abs(qmax)))
    idx = np.where(q >= qmax - tol)[0]
    if len(idx) == 0:
        idx = np.array([int(np.argmax(q))])
    p_init = np.zeros(n_lags)
    p_init[idx] = 1.0 / len(idx)
    Ldual = max(1e-12, 1.05 * float(np.sum(G * G)) / L)
    best_d = None
    best_val = -np.inf
    x = p_init.copy()
    y = p_init.copy()
    t = 1.0
    for _ in range(max_inner):
        ds = dual_eval(y, h, G, q, n, L)
        if ds is None:
            break
        d_y, u_y, v_y, _ = ds
        if v_y > best_val:
            best_val = v_y
            best_d = d_y.copy()
        x_trial = project_simplex(y + u_y / Ldual)
        ds2 = dual_eval(x_trial, h, G, q, n, L)
        if ds2 is None:
            break
        d_t, u_t, v_t, gap_t = ds2
        if v_t < best_val - 1e-13:
            x_trial = project_simplex(x + u_y / Ldual)
            ds2 = dual_eval(x_trial, h, G, q, n, L)
            if ds2 is None:
                break
            d_t, u_t, v_t, gap_t = ds2
        if v_t > best_val:
            best_val = v_t
            best_d = d_t.copy()
        t_new = (1.0 + math.sqrt(1.0 + 4.0 * t * t)) / 2.0

        y = project_simplex(y)
        x = x_trial
        t = t_new
        if gap_t <= max(1e-10, 1e-8 * max(1.0, abs(u_t.max()))):
            break
    if best_d is None:
        p_f = np.zeros(n_lags)
        p_f[int(np.argmax(q))] = 1.0
        ds = dual_eval(p_f, h, G, q, n, L)
        if ds is None:
