Gradient of Q_d with respect to h_i:
- For the term h[a_d+r] * (1-h[b_d+r]):

g[i] += dx * (1 - h[i2]) where i = a_d+r and i2 = b_d+r
g[i] += -dx * h[i1] where i = b_d+r and i1 = a_d+r

import numpy as np
from scipy.optimize import minimize

def proj(z, n):
    z = np.clip(np.asarray(z, dtype=np.float64), -1e12, 1e12).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    s = n * 0.5
    lo, hi = z.min() - 1.0, z.max() + 1.0
    for _ in range(80):
        m = 0.5 * (lo + hi)
        if np.sum(np.clip(z - m, 0, 1)) > s: lo = m
        else: hi = m
    h = np.clip(z - 0.5*(lo+hi), 0, 1)
    r = s - h.sum()
    for i in range(n):
        if r > 0:
            d = min(r, 1-h[i]); h[i] += d; r -= d
        elif r < 0:
            d = min(-r, h[i]); h[i] -= d; r += d
        else: break
    if not np.all(np.isfinite(h)) or h.min() < -1e-14 or h.max() > 1+1e-14:
        return None
    return h

def feas(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and h.min() >= -1e-12 
            and h.max() <= 1+1e-12 and abs(h.sum()-n*0.5) <= 1e-10*max(1,n))

def score(h, n):
    if not feas(h, n): return float('inf')
    return float(np.max(np.correlate(h, 1-h, mode='full') * 2.0/n))

def svgrad(h, n, tau):
    dx = 2.0/n; L = 2*n-1; Q = np.empty(L)
    for d in range(-(n-1), n):
        a, b, m = max(d,0), max(-d,0), n-abs(d)
        Q[d+n-1] = dx*np.sum(h[a:a+m]*(1-h[b:b+m]))
    qm = Q.max(); E = np.exp((Q-qm)/tau); w = E/E.sum()
    F = qm + tau*np.log(E.sum())
    g = np.zeros(n)
    for d in range(-(n-1), n):
        wd = w[d+n-1]
        if wd < 1e-15: continue
        a, b, m = max(d,0), max(-d,0), n-abs(d)
        g[a:a+m] += wd*dx*(1-h[b:b+m])
        g[b:b+m] -= wd*dx*h[a:a+m]
    if not np.isfinite(F) or not np.all(np.isfinite(g)):
        return 1e10, np.zeros(n)
    return F, g

def sofg(x, n, tau):
    return svgrad(x, n, tau)

def eqf(x, n): return np.sum(x) - n*0.5
def eqj(x, n): return np.ones((1, n))

def cont_opt(h0, n, taus, mxit):

    for tau in taus:
        try:
            res = minimize(sofg, h, args=(n,tau), jac=True, method='SLSQP',
                bounds=[(0,1)]*n, constraints=[{'type':'eq','fun':eqf,'jac':eqj,'args':(n,)}],
                options={'maxiter':mxit,'ftol':1e-10,'disp':False})
        except Exception: continue
        cands = [h]
        if res.x is not None and len(res.x)==n and np.all(np.isfinite(res.x)):
            xr = proj(res.x, n)
            if xr is not None:
                cands.append(xr)
                for al in [0.5, 0.25, 0.125]:
                    c = proj(h + al*(xr-h), n)
                    if c is not None: cands.append(c)
        bF = svgrad(h, n, tau)[0]; bh = h
        for c in cands:
            Fc = svgrad(c, n, tau)[0]
            if Fc < bF: bF, bh = Fc, c
        h = bh
    return h

def pg_cont(h0, n, taus, mxit_pg):

def proj(z, n):
    z = np.clip(np.asarray(z, dtype=np.float64).ravel(), -1e12, 1e12)
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    s = n * 0.5
    lo, hi = z.min() - 1.0, z.max() + 1.0
    for _ in range(80):
        m = 0.5 * (lo + hi)
        if np.sum(np.clip(z - m, 0, 1)) > s:
            lo = m
        else:
            hi = m
    h = np.clip(z - 0.5 * (lo + hi), 0, 1)
    r = s - h.sum()
    for i in range(n):
        if r > 0:
            d = min(r, 1 - h[i])
            h[i] += d
            r -= d
        elif r < 0:
            d = min(-r, h[i])
            h[i] -= d
            r += d
        else:
            break
    if not np.all(np.isfinite(h)) or h.min() < -1e-14 or h.max() > 1 + 1e-14:
        return None
    return h

            h.min() >= -1e-12 and h.max() <= 1 + 1e-12 and

        return float('inf')

    L = 2 * n - 1
    Q = np.empty(L)

        Q[d + n - 1] = dx * np.sum(h[a:a+m] * (1 - h[b:b+m]))
    qm = Q.max()
    E = np.exp((Q - qm) / tau)
    tot = E.sum()
    w = E / tot
    F = qm + tau * np.log(tot)
    g = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[d + n - 1]

    if not np.isfinite(F) or not np.all(np.isfinite(g)):
        return 1e10, np.zeros(n)
    return F, g

import numpy as np
from scipy.optimize import minimize

def proj(z, n):
    z = np.clip(np.asarray(z, dtype=np.float64).ravel(), -1e12, 1e12)
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    s = n * 0.5
    lo, hi = z.min() - 1.0, z.max() + 1.0
    for _ in range(80):
        m = 0.5 * (lo + hi)
        if np.sum(np.clip(z - m, 0, 1)) > s:
            lo = m
        else:
            hi = m
    h = np.clip(z - 0.5 * (lo + hi), 0, 1)
    r = s - h.sum()
    for i in range(n):
        if r > 0:
            d = min(r, 1 - h[i]); h[i] += d; r -= d
        elif r < 0:
            d = min(-r, h[i]); h[i] -= d; r += d
        else:
            break
    if not np.all(np.isfinite(h)) or h.min() < -1e-14 or h.max() > 1 + 1e-14:
        return None
    return h

def feas(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-12 and h.max() <= 1 + 1e-12 and
            abs(h.sum() - n * 0.5) <= 1e-10 * max(1, n))

def sc(h, n):
    if not feas(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1 - h, mode='full') * 2.0 / n))

def svgrad(h, n, tau):
    dx = 2.0 / n
    L = 2 * n - 1
    Q = np.empty(L)
    for d in range(-(n - 1), n):
        a, b, m = max(d, 0), max(-d, 0), n - abs(d)
        Q[d + n - 1] = dx * np.sum(h[a:a + m] * (1 - h[b:b + m]))
    qm = Q.max()
    E = np.exp((Q - qm) / tau)
    tot = E.sum()
    w = E / tot
    F = qm + tau * np.log(tot)
    g = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[d + n - 1]
        if wd < 1e-15:
            continue
        a, b, m = max(d, 0), max(-d, 0), n - abs(d)
        g[a:a + m] += wd * dx * (1 - h[b:b + m])
        g[b:b + m] -= wd * dx * h[a:a + m]
    if not np.isfinite(F) or not np.all(np.isfinite(g)):
        return 1e10, np.zeros(n)
    return F, g

def sofg(x, n, tau):
    return svgrad(x, n, tau)

def eqf(x, n):
    return np.sum(x) - n * 0.5

def eqj(x, n):
    return np.ones((1, n))

def cons_dict(n):
    return {'type': 'eq', 'fun': eqf, 'jac': eqj, 'args': (n,)}

def cont_opt(h0, n, taus, mxit):
    h = h0.copy()
    cn = [cons_dict(n)]
    bd = [(0, 1)] * n
    for tau in taus:
        try:
            res = minimize(sofg, h, args=(n, tau), jac=True, method='SLSQP',
                           bounds=bd, constraints=cn,
                           options={'maxiter': mxit, 'ftol': 1e-10, 'disp': False})
        except Exception:
            continue
        cands = [h]
        if res.x is not None and len(res.x) == n and np.all(np.isfinite(res.x)):
            xr = proj(res.x, n)
            if xr is not None:
                cands.append(xr)
                for al in [0.5, 0.25, 0.125]:
                    c = proj(h + al * (xr - h), n)
                    if c is not None:
                        cands.append(c)
        bF, bh = svgrad(h, n, tau)[0], h
        for c in cands:
            Fc = svgrad(c, n, tau)[0]
            if Fc < bF:
                bF, bh = Fc, c
        h = bh
    return h

def pg_cont(h0, n, taus, mxi):
    dx = 2.0 / n
    h = h0.copy()
    for tau in taus:
        eta = min(max(1.0, 0.25 / dx), 2.0 / dx)
        for _ in range(mxi):
            F, g = svgrad(h, n, tau)
            z = proj(h - eta * g, n)
            if z is None:
                break
            v = z - h
            if np.max(np.abs(v)) < 1e-9 or np.dot(g, v) >= 0:
                break
            Fz = svgrad(z, n, tau)[0]
            if Fz <= F + 1e-4 * np.dot(g, v):
                h = z
                eta = min(1.2 * eta, 2.0 / dx)
            else:
                ok = False
                for _ in range(20):
                    eta *= 0.5
                    z2 = proj(h - eta * g, n)
                    if z2 is None:
                        break
                    if svgrad(z2, n, tau)[0] <= F + 1e-4 * np.dot(g, z2 - h):
                        h = z2; ok = True; break
                if not ok:
                    break
    return h

def coord_polish(h0, n, passes, dlt):
    h = h0.copy()
    best = sc(h, n)
    for _ in range(passes):
        imp = False
        for i in range(n):

                s2 = sc(h2, n)
                if s2 < best - 1e-12:

                    imp = True

    return h, best

def mk_starts(fb, n, rng, cnt):

        for o in out:
            if np.max(np.abs(c - o)) < 1e-8:

        if not dup:
            out.append(c)
    return out[:cnt]
