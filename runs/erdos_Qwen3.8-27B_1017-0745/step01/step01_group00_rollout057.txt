    q0 = np.correlate(h0, 1.0 - h0, mode='full') * (2.0 / n)
    t0 = np.max(q0) + max(1e-10, 1e-8 * max(1.0, abs(np.max(q0))))
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    cons = [
        {'type': 'ineq', 'fun': epi_ineq, 'jac': epi_ineq_grad},
        {'type': 'eq', 'fun': epi_eq, 'jac': epi_eq_grad},
    ]
    try:
        res = minimize(epi_obj, z0, jac=epi_obj_grad, bounds=bounds,
                       constraints=cons, method='SLSQP',
                       options={'maxiter': 250, 'ftol': 1e-12})
    except Exception:
        return None
    if res.x is None or res.x.shape[0] != n + 1:
        return None
    if not np.all(np.isfinite(res.x)):
        return None
    y = project_capped(res.x[:n], n)
    return y

def optimize_epigraph(h0, n, n_passes=3):
    current = h0.copy()
    score = exact_score(current, n)
    for _ in range(n_passes):
        y = slsqp_one_pass(current, n)
        if y is None or not is_feasible(y, n):
            break
        best_z, best_s = current.copy(), score
        for k in range(6):
            alpha = 1.0 / (2.0 ** k)
            if alpha >= 1.0 - 1e-15:
                z = y
            else:
                z = project_capped(current + alpha * (y - current), n)
            if z is None or not is_feasible(z, n):
                continue
            sz = exact_score(z, n)
            if sz < best_s - 1e-12:
                best_s, best_z = sz, z.copy()
        if best_s >= score - 1e-12:
            break
        current, score = best_z, best_s
    return current, score

def optimize_subgrad(h0, n, n_iter=50):
    current = h0.copy()
    score = exact_score(current, n)
    dx = 2.0 / n
    eta = min(0.3 / dx, 2.0 / dx)
    eta_max = 2.0 / dx
    failed = 0
    for _ in range(n_iter):
        if failed >= 10 or eta < 1e-9 / dx:
            break
        q = np.correlate(current, 1.0 - current, mode='full') * dx
        idx = int(np.argmax(q))
        d = idx - (n - 1)
        p0 = max(d, 0)
        q0 = max(-d, 0)
        m = n - abs(d)
        g = np.zeros(n)
        g[p0:p0 + m] += dx * (1.0 - current[q0:q0 + m])
        g[q0:q0 + m] -= dx * current[p0:p0 + m]
        z = project_capped(current - eta * g, n)
        if z is None:
            eta *= 0.5
            failed += 1
            continue
        v = z - current
        if np.max(np.abs(v)) < 1e-11:
            eta *= 0.5
            failed += 1
            continue
        best_z, best_s = None, score
        for k in range(6):
            alpha = 1.0 / (2.0 ** k)
            if alpha >= 1.0 - 1e-15:
                trial = z
            else:
                trial = project_capped(current + alpha * v, n)
            if trial is None or not is_feasible(trial, n):
                continue
            ts = exact_score(trial, n)
            if ts < best_s - 1e-12:
                best_s, best_z = ts, trial.copy()
        if best_z is None:
            eta *= 0.5
            failed += 1
        else:
            current, score = best_z, best_s
            failed = 0
            eta = min(1.15 * eta, eta_max)
    return current, score

def build_starts(fallback, n, rng, count):
    pool = []
    def _add(v):
        if v is None:
            return
        for s in pool:
            if np.max(np.abs(v - s)) < 1e-8:
                return
        pool.append(v)
    _add(fallback.copy())
    _add(project_capped(fallback[::-1], n))
    _add(project_capped(1.0 - fallback[::-1], n))
    _add(np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    _add(project_capped(blk, n))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    _add(project_capped(alt, n))
    for sd in (0.06, 0.15, 0.30):
        _add(project_capped(fallback + rng.normal(0, sd, n), n))
    _add(project_capped(rng.uniform(0, 1, n), n))
    t = np.linspace(0, 1, n)
    _add(project_capped(0.5 + 0.3 * np.cos(np.pi * t), n))
    return pool[:count]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if parent is not None:
        proj = project_capped(parent, n)
        fallback = proj if proj is not None else np.full(n, 0.5)
    else:
        fallback = np.full(n, 0.5)
    if not is_feasible(fallback, n):
        fallback = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        return fallback, float(exact_score(fallback, n)), n
    if n <= 64:
        n_starts, n_passes = 10, 3
    elif n <= 128:
        n_starts, n_passes = 4, 45
    elif n <= 256:
        n_starts, n_passes = 2, 30
    else:
        n_starts, n_passes = 1, 18
    starts = build_starts(fallback, n, rng, n_starts)
    gs, gb = exact_score(fallback, n), fallback.copy()
    for st in starts:
        s = exact_score(st, n)
        if s < gs - 1e-12:
            gs, gb = s, st.copy()
    for st in starts:
        try:
            if n <= 64:
                xo, so = optimize_epigraph(st, n, n_passes)
            else:
                xo, so = optimize_subgrad(st, n, n_passes)
            if so < gs - 1e-12:
                gs, gb = so, xo.copy()
        except Exception:
            pass
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    fb_score = exact_score(fallback, n)
    if fs > fb_score + 1e-12:
        final, fs = fallback, fb_score
    return final, float(fs), int(n)
