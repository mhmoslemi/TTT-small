        if h is not None and sc < best_sc - 1e-12:
            best_h, best_sc = h, sc
            fallback_h, fallback_sc = h.copy(), sc
    for sig in [0.015, 0.04, 0.09, 0.15]:
        p = project_box_sum(best_h + sig * np.random.randn(n), S)
        if p is None:
            continue
        h, sc = local_lp_search(p, n, max_iter=5)
        if h is not None and sc < best_sc - 1e-12:
            best_h, best_sc = h, sc
            fallback_h, fallback_sc = h.copy(), sc
    fh = project_box_sum(best_h, S)
    if fh is None or not feasible(fh, n):
        fh = fallback_h
    sc_final = exact_score(fh, n)
    if not np.isfinite(sc_final) or sc_final > fallback_sc + 1e-12:
        fh = fallback_h
        sc_final = fallback_sc
    fh = np.clip(fh, 0.0, 1.0)
    return (fh.astype(np.float64), sc_final, n)
