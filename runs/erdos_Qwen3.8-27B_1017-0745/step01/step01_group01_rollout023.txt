def lag_values_and_grads(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    vals = np.zeros(L)
    grads = np.zeros((L, n))
    for d in range(-(n-1), n):
        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        # i ranges from lo to hi-1
        idx = np.arange(lo, hi)
        vals[row] = np.sum(h[idx] * (1 - h[idx + d]))
        # Gradient contribution from h[i] * (1-h[i+d]):
        # d/dh[j] = (1-h[j+d]) if j in [lo, hi) [as the i index]
        #           - h[j-d] if j-d in [lo, hi), i.e., j in [lo+d, hi+d) [as the i+d index]
        if d >= 0:
            # i ranges [0, n-1-d], so i+d ranges [d, n-1]
            # As i-index: j in [0, n-1-d] → adds (1-h[j+d])
            # As i+d-index: j in [d, n-1] → subtracts h[j-d]
            j1 = np.arange(0, n - d)
            grads[row, j1] += (1 - h[j1 + d])
            j2 = np.arange(d, n)
            grads[row, j2] -= h[j2 - d]
        else:
            # d < 0, let dd = -d > 0
            # i ranges [dd, n-1], i+d = i-dd ranges [0, n-1-dd]
            # As i-index: j in [dd, n-1] → adds (1-h[j+d]) = (1-h[j-dd])
            # As i+d-index: j in [0, n-1-dd] → subtracts h[j + dd] = h[j - d]
            dd = -d
            j1 = np.arange(dd, n)
            grads[row, j1] += (1 - h[j1 - dd])
            j2 = np.arange(0, n - dd)
            grads[row, j2] -= h[j2 + dd]
    return vals, grads

def project_capped(z, n):
    s = n / 2.0
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if lo == hi:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(n + 2):
        if abs(r) < 1e-13 * max(1, n):
            break
        if r > 0:
            for idx in range(n):
                room = 1.0 - h[idx]
                if room > 1e-15:
                    add = min(r, room)
                    h[idx] += add
                    r -= add

def project_capped(z, n):

        mid = 0.5 * (lo + hi)

            break

            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))

            break
        if r > 0:

                    h[idx] += add
                    r -= add
                    if abs(r) < 1e-13 * max(1, n):

def lag_vals_grads(h, n):
    L = 2 * n - 1
    vals = np.zeros(L)
    grads = np.zeros((L, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        if d >= 0:
            m = n - d
            idx = np.arange(m)
            vals[row] = float(np.sum(h[idx] * (1.0 - h[idx + d])))
            if m > 0:
                grads[row, idx] += (1.0 - h[idx + d])
                grads[row, np.arange(d, n)] -= h[np.arange(d, n) - d]
        else:
            dd = -d
            m = n - dd
            idx = np.arange(dd, n)
            vals[row] = float(np.sum(h[idx] * (1.0 - h[idx - dd])))
            if m > 0:
                grads[row, idx] += (1.0 - h[idx - dd])
                grads[row, np.arange(m)] -= h[np.arange(m) + dd]
    return vals, grads
