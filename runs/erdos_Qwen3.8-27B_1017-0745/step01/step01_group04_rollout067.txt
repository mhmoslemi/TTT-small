        else:
            hi = mid

                if stag >= 28:
                    break
    s_end = score_exact(h, N)
    if np.isfinite(s_end) and s_end < cbest:
        cbest = s_end
        hbest = h.copy()
    return cbest, hbest

def make_fourier(N, rng, n_modes, amp):
    idx = np.arange(N)
    z = np.full(N, 0.5)
    ns = min(n_modes, (N - 1) // 2)
    for r_ in range(1, ns + 1):
        A = rng.standard_normal()
        B = rng.standard_normal()
        z += (amp / r_) * (A * np.sin(2.0 * np.pi * r_ * idx / N) +
                           B * np.cos(2.0 * np.pi * r_ * idx / N))
    return z

def make_block_start(N, rng, baseline):
    nblocks = min(7, N)
    per = N // nblocks
    z = np.zeros(N)
    for b in range(nblocks):
        val = 1.0 if rng.random() < 0.5 else 0.0
        start = b * per
        end = start + per if b < nblocks - 1 else N
        z[start:end] = val + 0.3 * rng.standard_normal(end - start)
    z += 0.5 * rng.standard_normal(N) * 0.1
    return project_box_sum(z, N / 2.0)

def run(seed=42, **kwargs):
    N = None
    m = None
    raw = globals().get('initial_h_values', None)
    if raw is not None:
        raw = np.asarray(raw, dtype=np.float64)
        if raw.ndim == 1 and raw.size > 1 and np.all(np.isfinite(raw)):
            N = int(raw.size)
            m = N / 2.0
    if N is None:
        N = int(kwargs.get('n_points', 70))
        if N < 2 or N > 999:
            N = 70
        m = N / 2.0

    if raw is not None and np.all(np.isfinite(raw)) and raw.ndim == 1 and raw.size == N:
        baseline = project_box_sum(raw.copy(), m)
    else:
        baseline = np.full(N, 0.5)

    cbest = score_exact(baseline, N)
    hbest = baseline.copy()
    rng = np.random.default_rng(seed)

    if N <= 128:
        max_iter = 70
        n_hops = 3
    elif N <= 256:
        max_iter = 40
        n_hops = 1
    else:
        max_iter = 15
        n_hops = 0

    tau_sched = [0.03, 0.01, 0.004, 0.0015, 0.0005, 0.00015, 0.00004]

    starts = [baseline.copy()]
    starts.append(project_box_sum(1.0 - baseline, m))
    starts.append(project_box_sum(baseline + 0.06 * rng.standard_normal(N), m))
    starts.append(project_box_sum(baseline + 0.18 * rng.standard_normal(N), m))
    starts.append(project_box_sum(0.5 + 0.45 * rng.standard_normal(N), m))
    starts.append(project_box_sum(make_fourier(N, rng, 5, 0.12), m))
    starts.append(project_box_sum(make_fourier(N, rng, 3, 0.22), m))
    starts.append(make_block_start(N, rng, baseline))
    starts.append(project_box_sum(make_fourier(N, rng, 7, 0.10), m))

    dedup = [starts[0]]
    for s in starts[1:]:
        dup = False
        for ex in dedup:
            if np.max(np.abs(s - ex)) < 1e-9:
                dup = True
                break
        if not dup:
            dedup.append(s)
    starts = dedup

    for h0 in starts:
        cbest, hbest = optimize_pg(h0, N, m, tau_sched, max_iter, cbest, hbest)

    hop_scales = [0.035, 0.015, 0.007]
    for i in range(min(n_hops, len(hop_scales))):
        hop = project_box_sum(hbest + hop_scales[i] * rng.standard_normal(N), m)
        cbest, hbest = optimize_pg(hop, N, m, tau_sched, max_iter, cbest, hbest)

    # Second round of basin hops with smaller tau
    tau_fine = [0.002, 0.0006, 0.00015, 0.00004]
    for i in range(min(n_hops, 2)):
        hop = project_box_sum(hbest + 0.02 * rng.standard_normal(N), m)
        cbest, hbest = optimize_pg(hop, N, m, tau_fine, max_iter, cbest, hbest)

    if score_exact(baseline, N) < cbest:
        hbest = baseline.copy()
        cbest = score_exact(baseline, N)

    hbest = project_box_sum(hbest, m)
    cfinal = score_exact(hbest, N)
    if not np.isfinite(cfinal):
        hbest = np.full(N, 0.5)
        cfinal = score_exact(hbest, N)
    return hbest, cfinal, int(N)
