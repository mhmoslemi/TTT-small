            delta *= 0.5
    return x, s

def in_pop(h, pop):
    for p in pop:
        if np.max(np.abs(h - p)) < 1e-10:
            return True
    return False

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 0 < p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None:
        n = 47
        try:
            v = int(kwargs.get('n_points', 47))
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if parent is not None:
        fb = project_capped(parent, n)
    else:
        fb = np.full(n, 0.5)
    if fb is None:
        fb = np.full(n, 0.5)
    fb_score = exact_score(fb, n)
    if n == 1:
        return fb.astype(np.float64), float(fb_score), 1
    if n <= 64:
        pop_sz, max_gen, stall_lim, max_re, p_rounds, p_dirs = 40, 120, 20, 3, 28, 10
    elif n <= 128:
        pop_sz, max_gen, stall_lim, max_re, p_rounds, p_dirs = 36, 90, 16, 2, 20, 8
    elif n <= 256:
        pop_sz, max_gen, stall_lim, max_re, p_rounds, p_dirs = 28, 55, 12, 1, 12, 6
    elif n <= 512:
        pop_sz, max_gen, stall_lim, max_re, p_rounds, p_dirs = 20, 30, 10, 1, 8, 4
    else:
        pop_sz, max_gen, stall_lim, max_re, p_rounds, p_dirs = 14, 18, 8, 1, 5, 4
    pop = make_starts(fb, n, rng, pop_sz)
    scores = [exact_score(p, n) for p in pop]
    gi = int(np.argmin(scores))
    gscore, gbest_h = scores[gi], pop[gi].copy()
    F = rng.uniform(0.45, 0.85, pop_sz)
    CR = rng.uniform(0.4, 0.9, pop_sz)
    stall, restarts = 0, 0
    for _ in range(max_gen):
        prev_min = min(scores)
        pop, scores = de_generation(pop, scores, n, rng, F, CR)
        for i in range(pop_sz):
            if scores[i] < gscore - 1e-12:
                gscore, gbest_h = scores[i], pop[i].copy()
        if not in_pop(gbest_h, pop):
            worst = int(np.argmax(scores))
            pop[worst] = gbest_h.copy()
            scores[worst] = gscore
        if min(scores) >= prev_min - 1e-12:
            stall += 1
        else:
            stall = 0
        if stall >= stall_lim and restarts < max_re:
            restarts += 1
            stall = 0
            order = np.argsort(scores)
            keep = max(2, pop_sz // 4)
            retained = [pop[order[i]].copy() for i in range(keep)]
            retained[0] = gbest_h.copy()
            while len(retained) < pop_sz:
                frac = rng.random()
                if frac < 0.5:
                    c = gbest_h + rng.normal(0, 0.16 * (1 + restarts), n)
                elif frac < 0.75:
                    c = retained[rng.integers(0, len(retained))] + rng.normal(0, 0.15, n)
                else:
                    c = rng.uniform(0, 1, n)
                p = project_capped(c, n)
                if p is not None:
                    retained.append(p)
            while len(retained) < pop_sz:
                retained.append(gbest_h.copy())
            pop = retained[:pop_sz]
            scores = [exact_score(p, n) for p in pop]
    gbest_h, gscore = poll_search(gbest_h, n, pop, rng, p_rounds, p_dirs)
    final = project_capped(gbest_h, n)
    if final is None or not is_feasible(final, n):
        final = fb
    fs = exact_score(final, n)
    if fs > fb_score - 1e-12:
        final, fs = fb, fb_score
    final = project_capped(final, n)
    if final is None:
        final = np.full(n, 0.5)
    fs = exact_score(final, n)
    return final.astype(np.float64), float(fs), int(n)
