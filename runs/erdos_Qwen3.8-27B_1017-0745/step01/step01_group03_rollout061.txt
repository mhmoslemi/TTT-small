def lp_local_search(h0, n, S):
    cur = h0.copy()
    cur_sc = exact_score(cur, n)
    r = 0.20; fails = 0
    max_att = 8 if n <= 64 else 5
    for _ in range(max_att):
        if r < 1e-3 and fails >= 2:
            break
        h_ep = solve_lp(cur, r, n, S)
        improved = False
        if h_ep is not None and np.abs(h_ep - cur).max() > 1e-12:
            h_new = line_search(cur, h_ep, n, S)
            if h_new is not None and feasible(h_new, n):
                sc = exact_score(h_new, n)
                if sc < cur_sc - 1e-12:
                    cur = h_new; cur_sc = sc; fails = 0
                    r = min(0.25, 1.2 * r)
                    improved = True
        if not improved:
            fails += 1
            r = max(1e-4, 0.5 * r)
    return cur, cur_sc

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = None; n = 45
    try:
        pv = np.array(initial_h_values, dtype=np.float64)
        if pv.ndim == 1 and 2 <= len(pv) <= 999 and np.all(np.isfinite(pv)):
            if pv.min() >= -1e-9 and pv.max() <= 1.0 + 1e-9:
                n = len(pv)
                if abs(pv.sum() - n / 2.0) < 1e-5 * max(1, n):
                    parent = project_box_sum(np.clip(pv, 0, 1), n / 2.0)
    except Exception:
        pass
    S = n / 2.0
    if parent is None:
        parent = project_box_sum(np.full(n, 0.5), S)
    starts = [parent.copy(), parent[::-1].copy(), (1.0 - parent).copy()]
    if n <= 64:
        starts.append(np.roll(parent, n // 3).copy())
        for sig in [0.04, 0.12]:
            p = project_box_sum(parent + sig * rng.standard_normal(n), S)
            if p is not None:
                starts.append(p)
        f1 = int(rng.integers(1, max(2, n // 2)))
        f2 = int(rng.integers(1, max(2, n // 2)))
        p1 = float(rng.uniform(0, 2 * np.pi)); p2 = float(rng.uniform(0, 2 * np.pi))
        hd = np.cos(2*np.pi*f1*np.arange(n)/n + p1) + np.cos(2*np.pi*f2*np.arange(n)/n + p2)
        hd -= hd.mean(); rms = np.sqrt(np.mean(hd**2))
        if rms > 1e-12:
            p = project_box_sum(parent + 0.08 * hd / rms, S)
            if p is not None:
                starts.append(p)
    else:
        p = project_box_sum(parent + 0.06 * rng.standard_normal(n), S)
        if p is not None:
            starts.append(p)
    dedup = [starts[0]]
    for s in starts[1:]:
        if not any(np.abs(s - d).max() < 1e-10 for d in dedup):
            dedup.append(s)
    best_h = parent.copy()
    best_sc = exact_score(best_h, n)
    for s in dedup:
        if feasible(s, n):
            sc = exact_score(s, n)
            if sc < best_sc:
                best_sc = sc; best_h = s.copy()
    if n <= 96:
        for s in dedup:
            h_loc, sc_loc = lp_local_search(s, n, S)
            if sc_loc < best_sc - 1e-12:
                best_sc = sc_loc; best_h = h_loc.copy()
    else:
        for _ in range(5):
            p = project_box_sum(best_h + 0.05 * rng.standard_normal(n), S)
            if p is not None:
                sc = exact_score(p, n)
                if sc < best_sc - 1e-12:
                    best_sc = sc; best_h = p
    fh = project_box_sum(best_h, S)
    if fh is None or not feasible(fh, n):
        fh = parent
    fs = exact_score(fh, n)
    if not np.isfinite(fs) or fs > best_sc + 1e-10:
        fh = project_box_sum(parent, S)
        if fh is None:
            fh = np.full(n, 0.5)
        fs = exact_score(fh, n)
    return (fh, fs, n)
