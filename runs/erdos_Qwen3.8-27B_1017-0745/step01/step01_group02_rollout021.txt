def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(v - tau, 0.0, 1.0)
    r = S - float(y.sum())
    if abs(r) > 1e-12 * max(1, n):
        idx = np.argsort(1.0 - y) if r > 0 else np.argsort(y)
        for i in idx:
            d = min(abs(r), (1.0 - y[i]) if r > 0 else y[i])
            y[i] += d if r > 0 else -d
            r += d if r < 0 else -d
            if abs(r) < 1e-12:
                break
    if abs(y.sum() - S) > 1e-10 * max(1, n):
        return None
    return y

def check_feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(float(h.sum()) - 0.5 * n) < 1e-10 * max(1, n))

    if not check_feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

    p = np.arange(2*n-1)
    s = n-1-p  # s values
    # grad_p(i): (1-h[i+s])*I(0<=i+s<n) - h[i-s]*I(0<=i-s<n)
    # grad_p(j): (1-h[j+s])*I(0<=j+s<n) - h[j-s]*I(0<=j-s<n)
    
    b = np.zeros(2*n-1)
    # For i:
    valid1 = (i+s >= 0) & (i+s < n)
    valid2 = (i-s >= 0) & (i-s < n)
    b[valid1] += (1.0 - h[i+s[valid1]])

    b[valid4] += h[j-s[valid4]]
    b *= dx
    return b

    # b: slopes, c: intercepts (current values)
    # Sort by slope, break ties by keeping highest intercept
    order = np.argsort(b[indices], kind='stable')
    sorted_idx = indices[order]
    sorted_b = b[sorted_idx]
    sorted_c = c[sorted_idx]
    
    hull = []  # list of (start, slope, intercept)
    for i in range(len(sorted_idx)):
        if hull and sorted_b[i] == hull[-1][1]:
            # Same slope, keep higher intercept
            if sorted_c[i] <= hull[-1][2]:
                continue
            hull.pop()
        if not hull:
            hull.append((-float('inf'), sorted_b[i], sorted_c[i]))
        break
    # Actually this is getting complex. Let me use a stack-based approach.
    
    # Standard upper hull construction:
    hull_b = []
    hull_c = []
    hull_start = []
    
    for i in range(len(sorted_idx)):
        m, cc = sorted_b[i], sorted_c[i]
        if hull_b and m == hull_b[-1]:
            if cc <= hull_c[-1]:
                continue
            hull_b.pop()
            hull_c.pop()
            hull_start.pop()
        if not hull_b:
            hull_b.append(m)
            hull_c.append(cc)
            hull_start.append(-float('inf'))
        else:
            # Intersection: m * x + cc = hull_b[-1] * x + hull_c[-1]
            x_cross = (hull_c[-1] - cc) / (m - hull_b[-1])
            while len(hull_b) > 1 and x_cross <= hull_start[-1]:
                hull_b.pop()
                hull_c.pop()
                hull_start.pop()
                x_cross = (hull_c[-1] - cc) / (m - hull_b[-1])
            hull_b.append(m)
            hull_c.append(cc)
            hull_start.append(x_cross)
    
    return hull_start, hull_b, hull_c
