    u = np.full(n, 0.5)
    idx = np.arange(n, dtype=np.float64)
    for m in range(1, M + 1):
        a_m = rng.normal(0.0, 0.12 / np.sqrt(m))
        phi_m = rng.uniform(0.0, 2.0 * np.pi)
        u += a_m * np.cos(2.0 * np.pi * m * (idx + 0.5) / n + phi_m)
    u = project_capped_simplex(u, S)
    if u is not None:
        starts.append(u)
    r = project_capped_simplex(rng.uniform(0.0, 1.0, n), S)
    if r is not None:
        starts.append(r)
    r2 = project_capped_simplex(0.5 + 0.3 * rng.standard_normal(n), S)
    if r2 is not None:
        starts.append(r2)
    deduped = []
    for s in starts:
        if s is None or not is_feasible(s, n):
            continue
        dup = False
        for d in deduped:
            if float(np.max(np.abs(s - d))) < 1e-8:
                dup = True; break
        if not dup:
            deduped.append(s)
    return deduped

def run(seed=42, **kwargs):
    try:
        seed = int(seed) if seed is not None else 42
    except (TypeError, ValueError):
        seed = 42
    rng = np.random.default_rng(seed)
    n = None
    parent = None
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        try:
            iv_arr = np.asarray(iv, dtype=np.float64)
            if iv_arr.ndim == 1 and 8 <= iv_arr.size <= 999:
                n = iv_arr.size
                S = n / 2.0
                proj = project_capped_simplex(iv_arr, S)
                if proj is not None and is_feasible(proj, n):
                    parent = proj
                elif is_feasible(iv_arr, n):
                    parent = iv_arr.copy()
        except Exception:
            pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 80))
        except (TypeError, ValueError):
            n = 80
        if n < 8 or n > 999:
            n = 80
    S = n / 2.0
    L, R, LI = build_pairs(n)
    const = np.full(n, 0.5)
    best_h, best_sc = const.copy(), exact_score(const, n)
    if best_sc is None:
        best_sc = 0.5
    if parent is not None:
        ps = exact_score(parent, n)
        if ps is not None and ps < best_sc:
            best_sc = ps; best_h = parent.copy()
    starts = build_starts(n, S, parent, rng)
    for s in starts:
        sc = exact_score(s, n)
        if sc is not None and sc < best_sc:
            best_sc = sc; best_h = s.copy()
    nrel = n
    if nrel <= 128:
        max_it = 120
    elif nrel <= 256:
        max_it = 80
    elif nrel <= 512:
        max_it = 40
    else:
        max_it = 20
    taus = [0.02, 0.005, 0.001, 0.0002, 0.00005]
    for s in starts:
        bh, bs = pg_descent(s, taus, L, R, LI, n, S, max_it)
        if bs < best_sc:
            best_sc = bs; best_h = bh
    nkick = 3 if nrel <= 128 else (2 if nrel <= 256 else 1)
    sigma = 0.04
    for _ in range(nkick):
        pert = best_h + sigma * rng.standard_normal(n)
        pk = project_capped_simplex(pert, S)
        if pk is not None and is_feasible(pk, n):
            bh, bs = pg_descent(pk, [0.01, 0.002, 0.0004, 0.0001], L, R, LI, n, S, max_it)
            if bs < best_sc:
                best_sc = bs; best_h = bh
                sigma = max(0.02, 0.5 * sigma)
            else:
                sigma = min(0.20, 1.8 * sigma)
    fh = project_capped_simplex(best_h, S)
    if fh is None or not is_feasible(fh, n):
        fh = best_h
        if not is_feasible(fh, n) and parent is not None:
            fh = parent
    fs = exact_score(fh, n)
    if fs is None or not np.isfinite(fs):
        fh = const.copy()
        fs = exact_score(fh, n)
        if fs is None:
            fs = 0.5
    return fh, fs, n
