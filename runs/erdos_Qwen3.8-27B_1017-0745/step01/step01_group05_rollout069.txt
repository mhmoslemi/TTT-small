    d = project_capped(h - G.T @ p / L, n) - h
    u = q + G @ d
    D = p @ u + (L / 2.0) * (d @ d)
    return d, u, D

def solve_majorant(h, q, G, n, max_inner):
    dx = 2.0 / n
    L = 2.0 * dx
    Ldual = max(1e-12, 1.05 * float(np.sum(G * G)) / L)
    m = 2 * n - 1
    qm = q.max()
    mask = q >= qm - max(1e-12, 1e-8 * abs(qm))
    if not mask.any():
        mask = np.zeros(m, bool)
        mask[int(q.argmax())] = True
    x = np.zeros(m)
    x[mask] = 1.0 / mask.sum()
    yv = x.copy()
    theta = 1.0
    best_d = None
    best_D = -1e99
    for _ in range(max_inner):
        d_p, u_p, D_p = dual_eval(h, q, G, L, x, n)
        if D_p > best_D:
            best_D, best_d = D_p, d_p.copy()
        d_y, u_y, D_y = dual_eval(h, q, G, L, yv, n)
        if D_y > best_D:
            best_D, best_d = D_y, d_y.copy()
        xt = project_simplex(yv + u_y / Ldual)
        _, _, D_t = dual_eval(h, q, G, L, xt, n)
        if D_t < D_p - 1e-13:
            xt2 = project_simplex(x + u_p / Ldual)
            _, _, D_t2 = dual_eval(h, q, G, L, xt2, n)
            if D_t2 < D_p - 1e-13:
                break
            x, yv, theta = xt2, xt2.copy(), 1.0
        else:
            tn = (1.0 + math.sqrt(1.0 + 4.0 * theta * theta)) / 2.0
            yv = project_simplex(xt + (theta - 1.0) / tn * (xt - x))
            x, theta = xt, tn
        gap = u_p.max() - x @ u_p
        if gap <= 1e-10:
            break
    return best_d

def optimize_start(h0, n, mo, mi):
    h = project_capped(h0.astype(np.float64), n)
    sc = exact_score(h, n)
    bh, bs = h.copy(), sc
    dx = 2.0 / n
    L = 2.0 * dx
    stall = 0
    for _ in range(mo):
        q = lag_scores(h, n)
        G = build_G(h, n)
        if not np.all(np.isfinite(G)):
            break
        d = solve_majorant(h, q, G, n, mi)
        if d is None or np.max(np.abs(d)) < 1e-11:
            break
        v = G @ d
        n2 = d @ d
        qm = q.max()
        acc = False
        for j in range(18):
            a = 1.0 / (2 ** j)
            if (q + a * v).max() + (L / 2.0) * a * a * n2 >= qm - 1e-14:
                continue
            ht = project_capped(h + a * d, n)
            s = exact_score(ht, n)
            if s < sc - 1e-13:
                h, sc = ht, s
                if s < bs - 1e-13:
                    bs, bh = s, h.copy()
                acc = True
                break
        if not acc:
            break
        stall = stall + 1 if sc > bs - 1e-11 else 0
        if stall >= 3:
            break
    return bh, bs

def _pol_obj(x):
    n = len(x)
    return float(np.max(np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)))

def _pol_eq(x):
    return x.sum() - len(x) / 2.0

def _pol_eq_jac(x):
    return np.ones(len(x))

def slsqp_polish(h0, n):
    s0 = exact_score(h0, n)
    try:
        from scipy.optimize import minimize
        x0 = h0.copy()
        diff = n / 2.0 - x0.sum()
        x0[int(np.argmax(x0 < 1.0))] += diff if diff > 0 else 0
        x0[int(np.argmin(x0 > 0.0))] += diff if diff < 0 else 0
        x0 = np.clip(x0, 0, 1)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _pol_eq, 'jac': _pol_eq_jac}]
        res = minimize(_pol_obj, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 400, 'ftol': 1e-14})
        hx = project_capped(res.x, n)
        s = exact_score(hx, n)
        if s < s0 - 1e-14:
            return hx, s
    except Exception:
        pass
    return h0, s0

def greedy_perturb(h, n, rng, n_iter):
    hs = h.copy()
    sc = exact_score(hs, n)
    for _ in range(n_iter):
        i = rng.integers(0, n)
        delta = 0.02 * rng.standard_normal()
        ht = hs.copy()
        ht[i] = np.clip(ht[i] + delta, 0, 1)
        ht = project_capped(ht, n)
        s = exact_score(ht, n)
        if s < sc - 1e-13:
            hs, sc = ht, s
    return hs, sc

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    n = 45
    parent = globals().get('initial_h_values', None)
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.ndim == 1 and 2 <= p.size <= 999 and np.all(np.isfinite(p)) and np.max(np.abs(p)) < 1e6:
            n = p.size
            fallback = project_capped(p.copy(), n)
        else:
            fallback = np.full(n, 0.5)
    else:
        nk = kwargs.get('n_points', None)
        if isinstance(nk, int) and not isinstance(nk, bool) and 2 <= nk <= 999:
            n = nk
        fallback = np.full(n, 0.5)
    fb_s = exact_score(fallback, n)
    best_h, best_s = fallback.copy(), fb_s

    half = n // 2
    blk = np.zeros(n)
    blk[:half] = 1.0
    if n % 2 == 1:
        blk[half] = 0.5

    starts = [fallback.copy(), (1.0 - fallback).copy(), fallback[::-1].copy(),
              (1.0 - fallback[::-1]).copy(), blk.copy(), (1.0 - blk[::-1]).copy(),
              project_capped(fallback + 0.05 * rng.standard_normal(n), n),
              project_capped(fallback + 0.2 * rng.standard_normal(n), n),
              project_capped(rng.uniform(0, 1, n), n)]
    filt = [starts[0]]
    for st in starts[1:]:
        if all(np.max(np.abs(st - s)) > 1e-8 for s in filt):
            filt.append(st)

    if n <= 100:
        mo, mi, ns = 90, 80, len(filt)
    elif n <= 400:
        mo, mi, ns = 12, 16, min(4, len(filt))
    else:
        mo, mi, ns = 5, 10, min(2, len(filt))

    for st in filt[:ns]:
        try:
            bh, bs = optimize_start(st, n, mo, mi)
            if check_feasible(bh, n) and bs < best_s - 1e-12:
                best_s, best_h = bs, bh.copy()
        except Exception:
            pass

    if n <= 100:
        try:
            ph, ps = slsqp_polish(best_h, n)
            if ps < best_s - 1e-13:
                best_s, best_h = ps, ph
        except Exception:
            pass
        try:
            pert = project_capped(best_h + 0.02 * rng.standard_normal(n), n)
            bh, bs = optimize_start(pert, n, mo // 2, mi)
            if check_feasible(bh, n) and bs < best_s - 1e-12:
                best_s, best_h = bs, bh.copy()
            ph, ps = slsqp_polish(best_h, n)
            if ps < best_s - 1e-13:
                best_s, best_h = ps, ph
        except Exception:
            pass
        try:
            gh, gs = greedy_perturb(best_h, n, rng, 300)
            if gs < best_s - 1e-13:
                best_s, best_h = gs, gh
        except Exception:
            pass

    fh = project_capped(best_h, n)
    fs = exact_score(fh, n)
    if fs > fb_s + 1e-12:
        fh, fs = fallback.copy(), fb_s
    return fh.astype(np.float64), float(fs), int(n)
