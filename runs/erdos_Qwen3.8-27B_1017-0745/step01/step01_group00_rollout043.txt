import numpy as np

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n/2.0) <= 1e-10 * max(1, n)

def project_capped(z, n):

    s = n / 2.0

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    r = s - np.sum(h)
    for i in range(n):

            break
        if r > 0:

def exact_score(h, n):

def lag_vals(h, n):

def lag_grad_k(h, d, k, n, dx):

    bd = max(-d, 0)
    m = n - abs(d)
    if ad <= k < ad + m:
        g += dx * (1.0 - h[k - d])
    if bd <= k < bd + m:

def golden_min(f, a, b, iters=52):
    gr = (np.sqrt(5.0) + 1.0) / 2.0
    c = b - (b - a) / gr
    d2 = a + (b - a) / gr
    fc, fd = f(c), f(d2)
    for _ in range(iters):
        if fc < fd:
            b, d2, fd = d2, c, fc
            c = b - (b - a) / gr
            fc = f(c)
        else:
            a, c, fc = c, d2, fd
            d2 = a + (b - a) / gr
            fd = f(d2)
    xa, xb, xm = a, b, 0.5*(a+b)
    fa, fb, fxm = f(xa), f(xb), f(xm)
    if fa < fb and fa < fxm:
        return xa
    if fb < fa and fb < fxm:
        return xb
    return xm

def bisect_root(f, a, b, iters=52):
    fa, fb = f(a), f(b)
    if fa * fb > 0:
        return None
    for _ in range(iters):
        m = 0.5*(a+b)
        fm = f(m)
        if fa * fm <= 0:
            b, fb = m, fm
        else:
            a, fa = m, fm
    return 0.5*(a+b)

def pair_line_min(h, qv, i, j, n):

    L = max(-h[i], h[j]-1.0)

def local_descent(x0, n, rng, max_moves, n_scr, n_rand):

        pairs = screen_and_pick(x, qv, n, rng, n_scr, n_rand)
        if not pairs:
            break
        best_trial, best_sc = None, score - 1e-12
        for (i, j) in pairs:
            t = pair_line_min(x, qv, i, j, n)
            if t is None:

            trial = x.copy()
            trial[i] += t
            trial[j] -= t
            proj = project_capped(trial, n)
            if proj is None or not is_feasible(proj, n):
                continue
            sc = exact_score(proj, n)
            if sc < best_sc:
                best_sc, best_trial = sc, proj
        if best_trial is None:
            break
        if np.max(np.abs(best_trial - x)) < 1e-10:
            break
        x, score = best_trial, best_sc
    return x, exact_score(x, n)

    # balanced binary

    bb[idx[:n//2]] = 1.0

    pool.append(project_capped(bb, n))
    for sd in (0.08, 0.20):

    pool.append(project_capped(rng.uniform(0, 1, n), n))

        dup = False
        for s in out:

                m = v
        return m
    tG = golden_min(G_fn, L, U); tH = golden_min(H_fn, L, U)
    candidates = [L, U, min(max(0.0, L), U), tG, tH]
    if H_fn(tH) < 0:
        hL, hU = H_fn(L), H_fn(U)
        if hL > 0:
            r1 = bisect_root(H_fn, L, tH)
            if r1 is not None:
                candidates.append(r1); candidates.append(golden_min(G_fn, L, r1))
        if hU > 0:
            r2 = bisect_root(H_fn, tH, U)
            if r2 is not None:
                candidates.append(r2); candidates.append(golden_min(G_fn, r2, U))
    best_t, best_v = 0.0, Pmax(0.0)
    for t in candidates:
        if not np.isfinite(t):
            continue
        t = min(max(t, L), U)
        v = Pmax(t)
        if v < best_v - 1e-15 or (abs(v - best_v) < 1e-15 and abs(t) < abs(best_t)):
            best_v, best_t = v, t
    return best_t

def screen_pairs(h, qv, n, rng, n_scr, n_rand):
    dx = 2.0 / n
    L = 2 * n - 1
    top = list(np.argsort(qv)[-8:]) if L > 8 else list(range(L))
    if n <= 128:
        ii, jj = np.triu_indices(n, 1)
    else:
        np_ = min(3000, 6 * n)
        ii = rng.integers(0, n, np_); jj = rng.integers(0, n, np_)
        msk = ii < jj; ii, jj = ii[msk], jj[msk]
    merits = np.full(len(ii), 1e10)
    for idx in range(len(ii)):
        i, j = int(ii[idx]), int(jj[idx])
        Lf = max(-h[i], h[j] - 1.0); Uf = min(1.0 - h[i], h[j])
        if Uf - Lf < 1e-12:
            continue
        mx_r, mx_l = -1e10, -1e10
        for k in top:
            d = k - (n - 1)
            bd = lag_grad_k(h, d, i, n, dx) - lag_grad_k(h, d, j, n, dx)
            if Uf > 0:
                mx_r = max(mx_r, bd)
            if Lf < 0:
                mx_l = max(mx_l, -bd)
        if Uf > 0 and Lf < 0:
            merits[idx] = min(mx_r, mx_l)
        elif Uf > 0:
            merits[idx] = mx_r
        else:
            merits[idx] = mx_l
    order = np.argsort(merits)
    sel = order[:n_scr]
    if n_rand > 0 and len(ii) > len(sel):
        extra = rng.choice(len(ii), min(n_rand, len(ii)), replace=False)
        sel = np.concatenate([sel, extra])
    pairs, seen = [], set()
    for s in sel:
        p = (int(ii[s]), int(jj[s]))
        if p not in seen:
            seen.add(p); pairs.append(p)
    return pairs

def local_descent(x0, n, rng, max_moves, n_scr, n_rnd):
    x = x0.copy().astype(np.float64)
    score = exact_score(x, n)
    if score >= float('inf'):
        return x, float('inf')
    for _ in range(max_moves):
        qv = lag_vals(x, n)
        pairs = screen_pairs(x, qv, n, rng, n_scr, n_rnd)
        if not pairs:
            break
        best_trial, best_sc = None, score - 1e-12
        for (i, j) in pairs:
            t = pair_line_min(x, qv, i, j, n)
            if t is None:
                continue
            trial = x.copy(); trial[i] += t; trial[j] -= t
            proj = project_capped(trial, n)
            if proj is None or not is_feasible(proj, n):
                continue
            sc = exact_score(proj, n)
            if sc < best_sc:
                best_sc, best_trial = sc, proj
        if best_trial is None or np.max(np.abs(best_trial - x)) < 1e-10:
            break
        x, score = best_trial, best_sc
    return x, exact_score(x, n)

def build_starts(fb, n, rng, count):
    pool = [fb.copy(), project_capped(fb[::-1], n), project_capped(1.0 - fb[::-1], n),
            np.full(n, 0.5)]
    blk = np.zeros(n); blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    pool.append(project_capped(blk, n))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    pool.append(project_capped(alt, n))
    for _ in range(3):
        idx = rng.permutation(n); bb = np.zeros(n)
        bb[idx[:n // 2]] = 1.0
        if n % 2:
            bb[idx[n // 2]] = 0.5
        pool.append(project_capped(bb, n))
    for sd in (0.06, 0.15, 0.30):
        pool.append(project_capped(fb + rng.normal(0, sd, n), n))
    pool.append(project_capped(rng.uniform(0, 1, n), n))
    out = []
    for v in pool:
        if v is None:
            continue
        if not any(np.max(np.abs(v - s)) < 1e-8 for s in out):
            out.append(v)
    return out[:count]

def run(seed=42, **kwargs):
    n = None; parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is None:
        try:
            v = int(kwargs.get('n_points', 0))
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    fb = project_capped(parent, n) if parent is not None else None
    if fb is None:
        fb = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 64:
        n_st, n_mv, n_sc, n_rn, n_hp, n_hm = 12, 30, 6, 3, 8, 12
    elif n <= 128:
        n_st, n_mv, n_sc, n_rn, n_hp, n_hm = 8, 20, 5, 3, 5, 8
    elif n <= 256:
        n_st, n_mv, n_sc, n_rn, n_hp, n_hm = 5, 10, 4, 2, 3, 5
    else:
        n_st, n_mv, n_sc, n_rn, n_hp, n_hm = 3, 6, 3, 1, 2, 3
    starts = build_starts(fb, n, rng, n_st)
    gb, gs = fb.copy(), exact_score(fb, n)
    scored = []
    for st in starts:
        s = exact_score(st, n)
        scored.append((s, st))
        if s < gs - 1e-12:
            gs, gb = s, st.copy()
    scored.sort(key=lambda x: x[0])
    for _, st in scored:
        xo, so = local_descent(st, n, rng, n_mv, n_sc, n_rn)
        if so < gs - 1e-12:
            gs, gb = so, xo.copy()
    # basin hopping
    w, ws = gb.copy(), gs
    scale = np.clip(np.sqrt(47.0 / n), 0.5, 1.5)
    hop_sds = [0.04, 0.08, 0.15, 0.25]
    for hop in range(n_hp):
        sd = hop_sds[hop % 4] * scale
        prop = project_capped(w + rng.normal(0, sd, n), n)
        if prop is None or not is_feasible(prop, n):
            continue
        xo, so = local_descent(prop, n, rng, n_hm, n_sc, n_rn)
        if so < gs - 1e-12:
            gs, gb = so, xo.copy()
        delta = so - ws
        if delta > 1e-12:
            T = 0.002 * (0.25 + 0.75 * (1.0 - hop / max(1, n_hp - 1)))
            if rng.random() < np.exp(-min(700.0, delta / T)):
                w, ws = xo, so
        else:
            w, ws = xo, so
    # final polish with more pairs
    _, _ = local_descent(gb, n, rng, 12, min(10, n_sc + 4), 2)
    polish = local_descent(gb, n, rng, 8, n_sc, n_rn)
    if polish[1] < gs - 1e-12:
        gs, gb = polish
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb
    fs = exact_score(final, n)
    if fs > exact_score(fb, n) + 1e-12:
        final = fb; fs = exact_score(fb, n)
    return final, float(fs), int(n)
