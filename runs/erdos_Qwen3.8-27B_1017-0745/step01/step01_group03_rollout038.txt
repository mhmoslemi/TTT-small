    resid = S - float(np.sum(y))
    if resid > 0:
        for i in range(n):
            if resid <= 0: break
            a = min(resid, 1.0 - y[i]); y[i] += a; resid -= a
    elif resid < 0:
        for i in range(n):
            if resid >= 0: break
            a = min(-resid, y[i]); y[i] -= a; resid += a
    if abs(float(np.sum(y)) - S) > 1e-9 * n: return None
    if y.min() < -1e-12 or y.max() > 1.0 + 1e-12: return None
    return np.clip(y, 0.0, 1.0)

def esc(h, n):
    if h is None or not np.all(np.isfinite(h)): return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def ok(h, n):
    return (h is not None and h.shape == (n,) and np.all(np.isfinite(h))
            and h.min() >= -1e-12 and h.max() <= 1.0 + 1e-12
            and abs(float(h.sum()) - 0.5 * n) < 1e-9 * n)

def line_mm(x, y, n, S):
    d = y - x
    nl = 2 * n - 1
    A = np.zeros(nl); B = np.zeros(nl); D = np.zeros(nl)
    for k in range(-(n-1), n):
        il, ih = max(0, k), min(n-1, n-1+k)
        idx = np.arange(il, ih+1); jj = idx - k
        A[k+n-1] = -np.sum(d[idx]*d[jj])
        B[k+n-1] = np.sum(d[idx]*(1.0-x[jj]) - x[idx]*d[jj])
        D[k+n-1] = np.sum(x[idx]*(1.0-x[jj]))
    cands = [0.0, 1.0]
    for k in range(nl):
        if abs(A[k]) > 1e-15:
            a = -B[k] / (2.0 * A[k])
            if -1e-8 < a < 1.0 + 1e-8: cands.append(a)
    for k in range(nl):
        for l in range(k+1, nl):
            da, db, dc = A[k]-A[l], B[k]-B[l], D[k]-D[l]
            if abs(da) > 1e-15:
                disc = db*db - 4.0*da*dc
                if disc >= -1e-10:
                    sd = np.sqrt(max(0.0, disc))
                    for a in [(-db+sd)/(2.0*da), (-db-sd)/(2.0*da)]:
                        if -1e-8 < a < 1.0 + 1e-8: cands.append(a)
            elif abs(db) > 1e-15:
                a = -dc / db
                if -1e-8 < a < 1.0 + 1e-8: cands.append(a)
    cands = sorted(set(round(min(1.0, max(0.0, a)), 12) for a in cands))
    best_a, best_v = 0.0, np.inf
    for a in cands:
        v = float(np.max(A*a*a + B*a + D))
        if v < best_v - 1e-14: best_v, best_a = v, a
    h_new = project_box_sum(x + best_a * d, S)
    if h_new is not None and ok(h_new, n): return h_new
    return x.copy()

def solve_lp(x, r, n, S):
    P = n * (n + 1) // 2
    nv = n + P + 1
    l = np.maximum(0.0, x - r); u = np.minimum(1.0, x + r)
    c = np.zeros(nv); c[nv-1] = 1.0
    lb = np.zeros(nv); ub = np.zeros(nv)
    lb[:n] = l; ub[:n] = u; ub[nv-1] = float(n)
    rb, cb, vb, rhs = [], [], [], []
    for i in range(n):
        for j in range(i, n):
            p = n + prod_idx(i, j, n)
            lb[p] = l[i]*l[j]; ub[p] = u[i]*u[j]
            for co_i, co_j, co_p, b in [
                (l[j], l[i], -1.0, l[i]*l[j]),
                (u[j], u[i], -1.0, u[i]*u[j]),
                (-l[j], -u[i], 1.0, -u[i]*l[j]),
                (-u[j], -l[i], 1.0, -l[i]*u[j])
            ]:
                rr = len(rhs)
                rb.extend([rr, rr, rr]); cb.extend([i, j, p]); vb.extend([co_i, co_j, co_p]); rhs.append(b)
    for k in range(-(n-1), n):
        il, ih = max(0, k), min(n-1, n-1+k)
        rr = len(rhs)
        for i in range(il, ih+1):
            j = i - k
            rb.append(rr); cb.append(i); vb.append(1.0)
            rb.append(rr); cb.append(n + prod_idx(i, j, n)); vb.append(-1.0)
        rb.append(rr); cb.append(nv-1); vb.append(-1.0); rhs.append(0.0)
    A_ub = coo_matrix((vb, (rb, cb)), shape=(len(rhs), nv)).tocsr()
    b_ub = np.array(rhs, dtype=np.float64)
    A_eq = np.zeros((1, nv)); A_eq[0, :n] = 1.0
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=np.array([S]),
                      bounds=list(zip(lb, ub)), method='highs',
                      options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None
    if res is None or res.x is None or not getattr(res, 'success', False): return None
    h_out = np.array(res.x[:n], dtype=np.float64)
    if not np.all(np.isfinite(h_out)): return None
    h_out = np.clip(h_out, l - 1e-6, u + 1e-6)
    h_out = project_box_sum(h_out, S)
    return h_out

def local_search(start, n, S):
    cur = project_box_sum(np.array(start, dtype=np.float64), S)
    if cur is None: return start
    cs = esc(cur, n)
    r = 0.20
    fails = 0
    mx = 8 if n <= 64 else 5
    for _ in range(mx):
        if r < 1e-3 and fails >= 2: break
        ep = solve_lp(cur, r, n, S)
        if ep is None or not ok(ep, n):
            r *= 0.5; fails += 1; continue
        if np.max(np.abs(ep - cur)) < 1e-12:
            r *= 0.5; fails += 1; continue
        h_new = line_mm(cur, ep, n, S)
        ns = esc(h_new, n)
        if np.isfinite(ns) and ns < cs - 1e-12:
            cur, cs = h_new, ns; r = min(0.30, 1.2 * r); fails = 0
        else:
            r *= 0.5; fails += 1
    return cur

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    n = 45
    parent = None
    try:
        pv = np.array(initial_h_values, dtype=np.float64).ravel()
        if 2 <= len(pv) <= 999 and np.all(np.isfinite(pv)) and pv.min() >= -1e-9 and pv.max() <= 1.0 + 1e-9:
            n = len(pv)
            if abs(float(pv.sum()) - 0.5 * n) < 1e-6 * n:
                parent = project_box_sum(np.clip(pv, 0.0, 1.0), 0.5 * n)
    except Exception:
        pass
    S = 0.5 * n
    if parent is None:
        parent = project_box_sum(np.full(n, 0.5), S)
    if parent is None:
        parent = np.full(n, 0.5)
    best_h, best_s = parent.copy(), esc(parent, n)
    starts = [parent, parent[::-1], 1.0 - parent]
    for sig in [0.04, 0.08, 0.12, 0.20]:
        p = project_box_sum(parent + sig * rng.standard_normal(n), S)
        if p is not None: starts.append(p)
    for m in range(1, min(4, n // 2 + 1)):
        ph = rng.uniform(0, 2 * np.pi)
        hh = 0.5 + 0.18 * np.cos(2 * np.pi * m * np.arange(n) / n + ph)
        p = project_box_sum(hh, S)
        if p is not None: starts.append(p)
    p = project_box_sum(rng.uniform(0, 1, n), S)
    if p is not None: starts.append(p)
    seen = [best_h.copy()]
    for s in starts:
        if not np.all(np.isfinite(s)): continue
        dup = any(np.max(np.abs(s - sv)) < 1e-10 for sv in seen)
        if not dup:
            seen.append(s.copy())
            ss = esc(s, n)
            if np.isfinite(ss) and ss < best_s:
                best_s, best_h = ss, s.copy()
    if n <= 96:
        for s in seen:
            h_new = local_search(s, n, S)
            ns = esc(h_new, n)
            if np.isfinite(ns) and ns < best_s - 1e-12:
                best_s, best_h = ns, h_new.copy()
    else:
        for _ in range(8):
            sig = [0.02, 0.06, 0.12, 0.20][_%4]
            p = project_box_sum(best_h + sig * rng.standard_normal(n), S)
            if p is not None:
                ps = esc(p, n)
                if np.isfinite(ps) and ps < best_s: best_s, best_h = ps, p
    for _ in range(4):
        sig = 0.015 * (1 + _ * 0.5)
        p = project_box_sum(best_h + sig * rng.standard_normal(n), S)
        if p is not None:
            ps = esc(p, n)
            if np.isfinite(ps) and ps < best_s - 1e-13: best_s, best_h = ps, p
    fh = project_box_sum(best_h, S)
    if fh is None or not ok(fh, n): fh = parent
    fs = esc(fh, n)
    if not np.isfinite(fs):
        fh = parent; fs = esc(fh, n)
    return (np.array(fh, dtype=np.float64), float(fs), n)
