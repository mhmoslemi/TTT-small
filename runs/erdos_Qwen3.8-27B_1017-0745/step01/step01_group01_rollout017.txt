def polish_pairwise(h, n, n_sweeps, deltas):
    best = h.copy()
    best_s = exact_score(best, n)
    for _ in range(n_sweeps):
        changed = False
        for i in range(n):
            for j in range(i+1, n):
                lo = max(-best[i], best[j] - 1.0)
                hi = min(1.0 - best[i], best[j])
                if lo >= hi:
                    continue
                cands = [lo, hi, 0.0]
                for d in deltas:
                    cands.append(max(lo, min(hi, d)))
                    cands.append(max(lo, min(hi, -d)))
                best_d, best_sc = 0.0, best_s
                for d in cands:
                    if d < lo - 1e-15 or d > hi + 1e-15:

                    trial = best.copy()
                    trial[i] += d
                    trial[j] -= d
                    sc = exact_score(trial, n)

                if best_d != 0.0:
                    best[i] += best_d
                    best[j] -= best_d
                    best_s = best_sc
                    changed = True
        if not changed:

    return best, best_s
