import numpy as np
from math import exp

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n < 1 or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if resid > 0:
        order = np.argsort(1.0 - y)
        for i in order:
            add = min(resid, 1.0 - y[i])
            y[i] += add
            resid -= add
            if resid < 1e-15:
                break
    elif resid < 0:
        order = np.argsort(y)
        for i in order:
            sub = min(-resid, y[i])
            y[i] -= sub
            resid += sub
            if -resid < 1e-15:
                break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):

    if abs(np.sum(y) - S) > 1e-9 * max(1.0, float(n)):
        return None
    y = np.clip(y, 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 0:
        if resid > 0:

                add = min(resid, 1.0 - y[i])

def compute_delta(h, p, q, a, n):

    L_range = np.arange(-(n-1), n)

    # L values: -(n-1), ..., n-1
    # Index: L + (n-1) so index 0..n_lags-1

    # Term 1a: a*(1-h[p+L]) for L in [-p, n-1-p]
    # L + (n-1) goes from (n-1-p) to (n-1-p) + (n-1) ... 
    # L range: [-p, n-1-p], corresponds to indices [n-1-p, n-1-p + (n-1)] = [n-1-p, 2n-2-p]
    # h indices: p+L ranges from p-p=0 to p+n-1-p=n-1

    if hi1 > lo1:

def compute_delta(h, p, q, a, n):
    n_lags = 2 * n - 1
    delta = np.zeros(n_lags)
    h_flip = h[::-1]
    inv_h = 1.0 - h
    delta[(n-1-p):(2*n-1-p)] += a * inv_h
    delta[(n-1-q):(2*n-1-q)] += -a * inv_h
    delta[p:(p+n)] += -a * h_flip
    delta[q:(q+n)] += a * h_flip

    if 0 <= idx_a < n_lags:
        delta[idx_a] += a * a
    idx_b = n - 1 + p - q
    if 0 <= idx_b < n_lags:
        delta[idx_b] += a * a
    delta[n - 1] -= 2.0 * a * a
    return delta
