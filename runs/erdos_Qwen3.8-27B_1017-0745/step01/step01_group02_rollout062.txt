import numpy as np
from scipy.optimize import minimize
import math

_MM = {}

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    if S < -1e-12 or S > n + 1e-12:
        return None
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-14:
        for i in np.argsort(1.0 - y):
            d = min(r, 1.0 - y[i])
            y[i] += d
            r -= d
            if r <= 1e-13:
                break
    elif r < -1e-14:
        for i in np.argsort(y):
            d = min(-r, y[i])
            y[i] -= d
            r += d
            if r >= -1e-13:
                break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-9) or np.any(y > 1.0 + 1e-9):
        return None
    if abs(float(np.sum(y)) - S) > 1e-10 * max(1, n):
        return None
    return np.clip(y, 0.0, 1.0)

def check_ok(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and np.all(h >= -1e-12) and
            np.all(h <= 1.0 + 1e-12) and abs(float(np.sum(h)) - 0.5 * n) < 1e-9 * max(1, n))

def exact_score(h, n):
    if not check_ok(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    if len(c) != 2 * n - 1 or not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c) * 2.0 / n)

def lag_qg(h, s, n):
    dx = 2.0 / n
    lo, hi = max(0, -s), min(n, n - s)
    q = 0.0
    g = np.zeros(n)
    for i in range(lo, hi):
        j = i + s
        q += h[i] * (1.0 - h[j])
        g[i] += 1.0 - h[j]
        g[j] -= h[i]
    return q * dx, g * dx

def get_parent():
    n = 45
    ph = globals().get('initial_h_values', None)
    if ph is not None:
        ph = np.asarray(ph, dtype=np.float64)
        if ph.ndim == 1 and 2 <= len(ph) <= 999 and np.all(np.isfinite(ph)) and \
           np.all(ph >= -1e-6) and np.all(ph <= 1.0 + 1e-6):
            n = len(ph)
            p = project_box_sum(np.clip(ph, 0.0, 1.0), 0.5 * n)
            if p is not None and check_ok(p, n):
                return p.copy(), n
    return np.full(45, 0.5), 45

def _mm_obj(z):
    x, L = _MM['x'], _MM['L']
    y = z[:len(x)]
    return 0.5 * L * float(np.dot(y - x, y - x)) + z[-1]

def _mm_obj_grad(z):
    x, L = _MM['x'], _MM['L']
    g = np.zeros(len(z))
    g[:len(x)] = L * (z[:len(x)] - x)
    g[-1] = 1.0
    return g

def _mm_ineq(z):
    x, qJ, GJ = _MM['x'], _MM['qJ'], _MM['GJ']
    d = z[:len(x)] - x
    return z[-1] - qJ - GJ.dot(d)

def _mm_ineq_jac(z):
    n, GJ, qJ = _MM['n'], _MM['GJ'], _MM['qJ']
    J = np.zeros((len(qJ), n + 1))
    J[:, :n] = -GJ
    J[:, n] = 1.0
    return J

def _mm_eq(z):
    return float(np.sum(z[:_MM['n']])) - 0.5 * _MM['n']

def _mm_eq_jac(z):
    n = _MM['n']
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def solve_mm(x, n):
    dx = 2.0 / n
    L = 2.0 * dx
    q_all = np.zeros(2 * n - 1)
    G_all = np.zeros((2 * n - 1, n))
    for idx, s in enumerate(range(-(n - 1), n)):
        q, g = lag_qg(x, s, n)
        q_all[idx] = q
        G_all[idx] = g
    _MM['x'] = x
    _MM['L'] = L
    _MM['n'] = n
    _MM['qJ'] = q_all
    _MM['GJ'] = G_all
    t0 = float(np.max(q_all)) + 1e-6
    z0 = np.append(x.copy(), t0)
    bounds = [(0.0, 1.0)] * n + [(-np.inf, np.inf)]
    cons = [
        {'type': 'ineq', 'fun': _mm_ineq, 'jac': _mm_ineq_jac},
        {'type': 'eq', 'fun': _mm_eq, 'jac': _mm_eq_jac},
    ]
    try:
        res = minimize(_mm_obj, z0, jac=_mm_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-12, 'disp': False})
        y = np.array(res.x[:n], dtype=np.float64)
        y = project_box_sum(y, 0.5 * n)
        if y is not None and check_ok(y, n):
            return y
    except Exception:
        pass
    return x.copy()

def mm_descent(h0, n, max_iter):
    h = h0.copy()
    s = exact_score(h, n)
    if not np.isfinite(s):
        return h, float('inf')
    for it in range(max_iter):
        prop = solve_mm(h, n)
        if prop is None or not np.all(np.isfinite(prop)):
            break
        d = prop - h
        if np.max(np.abs(d)) < 1e-11:
            break
        best_alpha, best_s = 0.0, s
        for k in range(10):
            alpha = 2.0 ** (-k)
            cand = project_box_sum(h + alpha * d, 0.5 * n)
            if cand is None:
                continue
            cs = exact_score(cand, n)
            if np.isfinite(cs) and cs < best_s - 1e-13:
                best_alpha, best_s = alpha, cs
        if best_alpha == 0.0:
            break
        h_new = project_box_sum(h + best_alpha * d, 0.5 * n)
        if h_new is None:
            break
        s_new = exact_score(h_new, n)
        if not np.isfinite(s_new):
            break
        h = h_new
        s = s_new
    return h, s

def build_starts(parent, n, rng):
    starts = [parent.copy(), (1.0 - parent).copy(), parent[::-1].copy()]
    for sd in [0.05, 0.15, 0.25]:
        p = project_box_sum(parent + sd * rng.standard_normal(n), 0.5 * n)
        if p is not None:
            starts.append(p)
    p = project_box_sum(rng.random(n) * 1.5, 0.5 * n)
    if p is not None:
        starts.append(p)
    ii = np.arange(n)
    u = 0.5 + 0.35 * np.cos(2.0 * np.pi * ii / n + 0.7) + 0.15 * np.sin(4.0 * np.pi * ii / n + 1.3)
    p = project_box_sum(u, 0.5 * n)
    if p is not None:
        starts.append(p)
    w = 0.5 + 0.4 * np.cos(6.0 * np.pi * ii / n)
    p = project_box_sum(w, 0.5 * n)
    if p is not None:
        starts.append(p)
    # Block-like start: high in first half, low in second
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    p = project_box_sum(blk, 0.5 * n)
    if p is not None:
        starts.append(p)
    unique = []
    for s in starts:
        dup = False
        for u2 in unique:
            if np.max(np.abs(s - u2)) < 1e-10:
                dup = True
                break
        if not dup:
            unique.append(s)
    return unique

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent, n = get_parent()

            if h0 is None:

            h_res, s_res = mm_descent(h0, n, max(6, M // 2))
            if np.isfinite(s_res) and s_res < best_c5:
                best_h, best_c5 = h_res.copy(), s_res
        except Exception:
            pass
    # Fine polish: small perturbations around best
    for _ in range(3):
        try:
            h0 = project_box_sum(best_h + 0.015 * rng.standard_normal(n), 0.5 * n)
            if h0 is None:

        except Exception:

        nn = 45
