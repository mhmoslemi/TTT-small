        return None

        return None

            return None
        return float(np.max(vals))

        return None

def build_lag_model(h, n):
    dx = 2.0 / n
    m = 2 * n - 1
    q = np.zeros(m)
    rows, cols, data = [], [], []
    for d in range(-(n-1), n):
        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        q[row] = dx * np.dot(h[lo:hi], 1.0 - h[lo+d:hi+d])
        for i in range(lo, hi):
            rows.append(row); cols.append(i); data.append(dx * (1.0 - h[i+d]))
            rows.append(row); cols.append(i+d); data.append(-dx * h[i])
    G = sp.coo_matrix((data, (rows, cols)), shape=(m, n)).tocsr()
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G.data)):
        return None, None
    return q, G

def solve_lp_step(x, q, G, rho, n):

    dx = 2.0 / n
    mu = 0.002 * dx
    # Variables: delta[0:n], u[0:n], t[n:2n+1]
    # dim = 2n+1
    I = sp.eye(n, format='csr')
    Z = sp.csr_matrix((m, n))
    Zt = sp.csr_matrix((m, 1))
    # Row 1: G@delta - t <= -q
    A1 = sp.hstack([G, Z, -sp.csr_matrix(np.ones((m, 1)))]).tocsr()
    # Row 2: delta - u <= 0
    A2 = sp.hstack([I, -I, sp.csr_matrix(np.zeros((n, 1)))]).tocsr()
    # Row 3: -delta - u <= 0
    A3 = sp.hstack([-I, -I, sp.csr_matrix(np.zeros((n, 1)))]).tocsr()
    A_ub = sp.vstack([A1, A2, A3]).tocsr()
    b_ub = np.concatenate([-q, np.zeros(n), np.zeros(n)])
    A_eq = sp.csr_matrix(np.array([[np.ones(n), np.zeros(n), 0.0]]))
    b_eq = np.array([0.0])
    c_obj = np.concatenate([np.zeros(n), mu * np.ones(n), [1.0]])
    bounds = []
    for i in range(n):
        lo_i = max(-rho, -x[i])
        hi_i = min(rho, 1.0 - x[i])
        bounds.append((lo_i, hi_i))
    for i in range(n):
        bounds.append((0.0, None))
    bounds.append((None, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

        return None
    if res is None or not res.success:
        return None
    delta = np.asarray(res.x[:n], dtype=np.float64)
    if not np.all(np.isfinite(delta)) or abs(np.sum(delta)) > 1e-5:
        return None
    return delta

    except Exception:
        return None

def build_lag_model(h, n):
    dx = 2.0 / n
    m = 2 * n - 1
    q = np.zeros(m)
    rows, cols, data = [], [], []
    for d in range(-(n - 1), n):
        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        q[row] = dx * np.dot(h[lo:hi], 1.0 - h[lo + d:hi + d])
        for i in range(lo, hi):
            rows.append(row); cols.append(i); data.append(dx * (1.0 - h[i + d]))
            rows.append(row); cols.append(i + d); data.append(-dx * h[i])
    G = sp.coo_matrix((data, (rows, cols)), shape=(m, n)).tocsr()
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G.data)):
        return None, None
    return q, G

def solve_lp_step(x, q, G, rho, n):
    m = 2 * n - 1
    mu = 0.002 * (2.0 / n)
    I = sp.eye(n, format='csr')
    Z = sp.csr_matrix((m, n))
    one_col = sp.csr_matrix(np.ones((m, 1)))
    zero_n1 = sp.csr_matrix(np.zeros((n, 1)))
    A1 = sp.hstack([G, Z, -one_col]).tocsr()
    A2 = sp.hstack([I, -I, zero_n1]).tocsr()
    A3 = sp.hstack([-I, -I, zero_n1]).tocsr()
    A_ub = sp.vstack([A1, A2, A3]).tocsr()
    b_ub = np.concatenate([-q, np.zeros(n), np.zeros(n)])
    A_eq = sp.csr_matrix(np.array([[np.concatenate([np.ones(n), np.zeros(n), [0.0]])]]))
    b_eq = np.array([0.0])
    c_obj = np.concatenate([np.zeros(n), mu * np.ones(n), [1.0]])
    bounds = []
    for i in range(n):
        bounds.append((max(-rho, -x[i]), min(rho, 1.0 - x[i])))
    bounds.extend([(0.0, None)] * n)
    bounds.append((None, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    if res is None or not res.success:
        return None
    delta = np.asarray(res.x[:n], dtype=np.float64)
    if not np.all(np.isfinite(delta)) or abs(np.sum(delta)) > 1e-5:
        return None
    return delta

def optimize_trajectory(x0, n, rng, best_h, best_score):
    S = n / 2.0
    x = x0.copy()
    f = exact_score(x, n)
    if f is None:
        return best_h, best_score
    if f < best_score - 1e-12:
        best_h, best_score = x.copy(), f
    rho, rho_max, rho_min = 0.08, 0.25, 2e-4
    max_iter = 30 if n <= 128 else (22 if n <= 256 else (14 if n <= 512 else 6))
    consec_fail, small_acc = 0, 0
    for it in range(max_iter):
        if rho < rho_min or consec_fail >= 4:
            break
        q, G = build_lag_model(x, n)
        if q is None:
            break
        delta = solve_lp_step(x, q, G, rho, n)
        if delta is None:
            rho = max(rho_min, rho * 0.5); consec_fail += 1; continue
        if np.max(np.abs(delta)) < 1e-9:
            break
        best_cand, best_cs = None, f
        for j in range(9):
            beta = 2.0 ** (-j)
            xn = project_capped_simplex(x + beta * delta, S)
            if xn is None or not is_feasible(xn, n):
                continue
            sc = exact_score(xn, n)
            if sc is not None and sc < best_cs - 1e-12:
                best_cand, best_cs = xn, sc
        if best_cand is None:
            rho = max(rho_min, rho * 0.5); consec_fail += 1; continue
        s = best_cand - x
        pred = f - np.max(q + G.dot(s))
        act = f - best_cs
        if pred <= 1e-12 or (pred > 0 and act / pred < 0.25):
            rho = max(rho_min, rho * 0.5)
        elif pred > 0 and act / pred > 0.75 and np.max(np.abs(s)) >= 0.8 * rho:
            rho = min(rho_max, 1.5 * rho)
        x, f = best_cand, best_cs
        consec_fail = 0
        if f < best_score - 1e-12:
            best_h, best_score = x.copy(), f
        if np.max(np.abs(s)) < 1e-8:
            small_acc += 1
            if small_acc >= 3 and act <= 1e-10:
                break
        else:
            small_acc = 0
    return best_h, best_score

def build_starts(n, S, parent, rng):
    starts = []
    const = np.full(n, 0.5)
    if parent is not None:
        starts.append(parent.copy())
        for sigma in [0.04, 0.12]:
            p = project_capped_simplex(parent + rng.normal(0, sigma, n), S)
            if p is not None:
                starts.append(p)
        starts.append(parent[::-1].copy())
        starts.append(1.0 - parent)
    else:
        starts.append(const)
    idx = np.arange(n, dtype=np.float64)
    M = min(5, (n - 1) // 2)
    u = np.full(n, 0.5)
    for m in range(1, M + 1):
        amp = rng.normal(0, 0.10 / np.sqrt(m))
        phi = rng.uniform(0, 2 * np.pi)
        u += amp * np.cos(2 * np.pi * m * (idx + 0.5) / n + phi)
    pu = project_capped_simplex(u, S)
    if pu is not None:
        starts.append(pu)
    perm = rng.permutation(n)
    binv = np.zeros(n)
    binv[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        binv[perm[n // 2]] = 0.5
    starts.append(binv)
    deduped = []
    for s in starts:
        dup = False
        for d in deduped:
            if np.max(np.abs(s - d)) < 1e-9:
                dup = True; break
        if not dup and is_feasible(s, n):
            deduped.append(s)
    if not any(np.allclose(d, const) for d in deduped):
        deduped.append(const)
    max_starts = 6 if n <= 128 else (4 if n <= 256 else 3)
    return deduped[:max_starts]

def run(seed=42, **kwargs):
    if seed is None:
        seed = 42
    try:
        seed = int(seed)
    except (TypeError, ValueError):
        seed = 42
    if seed < 0:
        seed = 42
    rng = np.random.default_rng(seed)
    n, parent = None, None
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        try:
            iv_arr = np.asarray(iv, dtype=np.float64)
            if iv_arr.ndim == 1 and 8 <= iv_arr.size <= 999:
                n = iv_arr.size
                if np.all(np.isfinite(iv_arr)) and np.max(np.abs(iv_arr)) < 1e12:
                    S = n / 2.0
                    if np.all(iv_arr >= -1e-12) and np.all(iv_arr <= 1.0 + 1e-12) and abs(np.sum(iv_arr) - S) <= 1e-12:
                        parent = iv_arr.copy()
                    else:
                        proj = project_capped_simplex(iv_arr, S)
                        if proj is not None and is_feasible(proj, n):
                            parent = proj
        except Exception:
            pass
    if n is None:
        n = kwargs.get('n_points', 80)
        try:
            n = int(n)
        except (TypeError, ValueError):
            n = 80
        if n < 8 or n > 999:
            n = 80
    S = n / 2.0
    const_fb = np.full(n, 0.5)
    best_h = const_fb.copy()
    best_score = exact_score(const_fb, n)
    if best_score is None:
        best_score = 0.5
    parent_fb = None
    if parent is not None:
        ps = exact_score(parent, n)
        if ps is not None and ps < best_score:
            best_score, best_h = ps, parent.copy()
        parent_fb = parent.copy()
    starts = build_starts(n, S, parent, rng)
    for s in starts:
        sc = exact_score(s, n)
        if sc is not None and sc < best_score - 1e-12:
            best_score, best_h = sc, s.copy()
    for s in starts:
        best_h, best_score = optimize_trajectory(s, n, rng, best_h, best_score)
    n_restarts = 2 if n <= 128 else (1 if n <= 256 else 0)
    sigma = 0.05
    for _ in range(n_restarts):
        pert = project_capped_simplex(best_h + sigma * rng.normal(0, 1, n), S)
        if pert is None or not is_feasible(pert, n):
            sigma = min(0.18, 1.8 * sigma); continue
        old_sc = best_score
        best_h, best_score = optimize_trajectory(pert, n, rng, best_h, best_score)
        sigma = max(0.02, 0.5 * sigma) if best_score < old_sc - 1e-12 else min(0.18, 1.8 * sigma)
    if not is_feasible(best_h, n):
        rep = project_capped_simplex(best_h, S)
        if rep is not None and is_feasible(rep, n):
            best_h = rep
    sc = exact_score(best_h, n)
    if sc is None:
        best_h = const_fb.copy()
        sc = exact_score(best_h, n)
        if sc is None:
            sc = 0.5
    if parent_fb is not None:
        ps2 = exact_score(parent_fb, n)
        if ps2 is not None and ps2 < sc:
            best_h, sc = parent_fb.copy(), ps2
    return best_h.astype(np.float64), float(sc), int(n)
