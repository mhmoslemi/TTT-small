def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) - n / 2.0) <= 1e-10)

def project_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):
        return None
    if np.max(np.abs(z)) > 1e12:
        return None
    n = len(z)
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z))
    for _ in range(80):
        mid = lo + 0.5 * (hi - lo)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = lo + 0.5 * (hi - lo)
    y = np.clip(z - lam, 0.0, 1.0)
    r = S - np.sum(y)
    for i in range(n):
        if r > 0:
            a = min(r, 1.0 - y[i])
            y[i] += a; r -= a
        elif r < 0:
            a = min(-r, y[i])
            y[i] -= a; r += a
        else:
            break
    if abs(np.sum(y) - S) > 1e-10 or not np.all(np.isfinite(y)):
        return None
    return np.clip(y, 0.0, 1.0)

def exact_score(h, n):
    vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(vals)):
        return None
    return float(np.max(vals))

correlate(h, 1-h)[k] = sum h[i] * (1-h)[i+k-(n-1)]
correlate(1-h, h)[k] = sum (1-h)[i] * h[i+k-(n-1)]

These are related by: correlate(1-h, h)[k] = correlate(h, 1-h)[2(n-1)-k]

So the multiset of values is the same, just in reverse order. The max is the same!

    n = len(z)
    if S < 0 or S > n or np.max(np.abs(z)) > 1e12:
        return None

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    r = S - np.sum(y)
    for i in range(n):
        if r == 0.0: break
        if r > 0:
