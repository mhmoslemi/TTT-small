    m = len(z)

        return z
    if not np.all(np.isfinite(z)):

import numpy as np

def project_capped(y, n):
    S = n / 2.0
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    lam = 0.5 * (lo + hi)
    for _ in range(100):
        c = np.clip(y - lam, 0.0, 1.0)
        r = c.sum() - S
        if abs(r) < 1e-14:
            break
        if r > 0:
            lo = lam
        else:
            hi = lam
        act = np.sum((y - lam > 0) & (y - lam < 1))
        if act > 0:
            nl = lam + r / act
            if lo < nl < hi:
                lam = nl
            else:
                lam = 0.5 * (lo + hi)
        else:
            lam = 0.5 * (lo + hi)
    c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(5):
        diff = S - c.sum()
        if abs(diff) < 1e-13:
            break
        mask = (c < 1.0 - 1e-12) if diff > 0 else (c > 1e-12)
        cnt = mask.sum()
        if cnt == 0:
            break
        c[mask] += diff / cnt
        np.clip(c, 0.0, 1.0, out=c)
    return c

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def build_G(h, n):
    dx = 2.0 / n
    m = 2 * n - 1
    G = np.zeros((m, n))
    for k in range(-(n-1), n):
        l = k + n - 1
        lo, hi = max(0, k), min(n, n + k)
        if hi > lo:
            G[l, lo:hi] += dx * (1.0 - h[lo-k:hi-k])
        lo2, hi2 = max(0, -k), min(n, n - k)
        if hi2 > lo2:
            G[l, lo2:hi2] -= dx * h[lo2+k:hi2+k]
    return G

def project_simplex(z):
    m = len(z)
    if m == 0 or not np.all(np.isfinite(z)):
        return np.full(max(m,1), 1.0/max(m,1))
    sz = np.sort(z)[::-1]
    cs = np.cumsum(sz)
    rho = np.arange(1, m + 1)
    cond = sz - (cs - 1.0) / rho > 0
    if not np.any(cond):
        return np.full(m, 1.0 / m)
    r = int(np.max(rho[cond]))
    theta = (cs[r - 1] - 1.0) / r
    p = np.maximum(z - theta, 0.0)
    p[np.argmax(p)] += 1.0 - p.sum()
    return p
