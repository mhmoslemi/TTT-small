import numpy as np

def project_onto_k(y, n):
    S = n / 2.0
    if not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        c = np.clip(y - mid, 0.0, 1.0)
        if c.sum() > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(30):
        diff = S - c.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = c < 1.0 - 1e-13
        else:
            mask = c > 1e-13
        cnt = mask.sum()
        if cnt == 0:
            break
        c[mask] += diff / cnt
        np.clip(c, 0.0, 1.0, out=c)
    return c

def exact_score(h):
    n = len(h)
    if not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

    a = np.zeros(2 * n - 1)

        aval = 0.0

        # b_k = dx * [(1-h_{i-k}) if valid - (1-h_{j-k}) if valid - h_{k+i} if valid + h_{k+j} if valid]

    # Grid search + local refinement
    npts = 17
    ts = np.linspace(lo, hi, npts)
    vals = np.array([np.max(q + b * t + a * t * t) for t in ts])
    best_idx = int(np.argmin(vals))
    # Refine around best
    for _ in range(4):
        i = best_idx
        if i == 0:
            new_ts = np.linspace(lo, ts[1], 5)
        elif i == npts - 1:
            new_ts = np.linspace(ts[-2], hi, 5)
        else:
            new_ts = np.linspace(ts[i-1], ts[i+1], 5)
        new_vals = np.array([np.max(q + b * t + a * t * t) for t in new_ts])
        vals = new_vals
        ts = new_ts
        best_idx = int(np.argmin(vals))
    best_t = float(ts[best_idx])
    best_v = float(vals[best_idx])
    return best_t, best_v

def basin_hop(h, score, rng, n):

    best_score = score

    temps = [0.0015, 0.0005, 0.00015]

def pair_polish(h, q, score, rng, n, pbudget, nsweeps):
    best_h = h.copy()
    best_score = score
    for _ in range(nsweeps):
        moved = False
        pairs = set()
        att = 0
        while len(pairs) < pbudget and att < 40 * pbudget:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                pairs.add((min(i,j), max(i,j)))
            att += 1
        if len(pairs) < pbudget:
            for i in range(n):
                for j in range(i+1, n):
                    if len(pairs) >= pbudget:
                        break
                    pairs.add((i, j))
                if len(pairs) >= pbudget:
                    break
        plist = list(pairs)
        rng.shuffle(plist)
        for (i, j) in plist:
            lo_t = max(-h[i], h[j] - 1.0)
            hi_t = min(1.0 - h[i], h[j])
            if hi_t - lo_t < 1e-12:
                continue
            b, a = pair_coeffs(h, i, j, n)
            bt, bv = line_search(q, b, a, lo_t, hi_t)
            if bv >= score - 1e-12:
                continue
            h2 = h.copy()
            h2[i] += bt
            h2[j] -= bt
            h2 = project_onto_k(h2, n)
            if not check_feasible(h2, n):
                continue
            s2 = exact_score(h2)
            if s2 < score - 1e-12:
                h = h2
                q = lag_scores(h)
                score = s2
                if s2 < best_score:
                    best_score = s2
                    best_h = h.copy()
                moved = True
        if not moved:
            break
    return best_h, best_score

def basin_hop(h, score, rng, n):
    best_h = h.copy()
    best_score = score
    sigma = 0.08
    temps = [0.0015, 0.0005, 0.00015]
    block_lens = [6, 4, 3]
    for ti in range(3):
        T = temps[ti]
        blen = block_lens[ti]
        acc = 0
        for _ in range(blen):
            z = rng.standard_normal(n)
            z -= z.mean()
            h2 = project_onto_k(h + sigma * z, n)
            if not check_feasible(h2, n):
                continue
            q2 = lag_scores(h2)
            s2 = exact_score(h2)
            pb = max(8, n)
            h2, s2 = pair_polish(h2, q2, s2, rng, n, pb, 1)
            if s2 < best_score - 1e-12:

                best_h = h2.copy()

                h = h2.copy()
                score = s2
                acc += 1

        if fr >= 0.6:

import numpy as np
import math

def project_onto_k(y, n):
    S = n / 2.0
    if not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        c = np.clip(y - mid, 0.0, 1.0)
        if c.sum() > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(30):
        diff = S - c.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = c < 1.0 - 1e-13
        else:
            mask = c > 1e-13
        cnt = mask.sum()
        if cnt == 0:
            break
        c[mask] += diff / cnt
        np.clip(c, 0.0, 1.0, out=c)
    return c

def exact_score(h):
    n = len(h)
    if not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lag_scores(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(h.sum() - n / 2.0) <= 1e-10)

def pair_coeffs(h, i, j, n):
    dx = 2.0 / n
    L = 2 * n - 1
    b = np.zeros(L)
    a = np.zeros(L)
    for k in range(-(n - 1), n):
        idx = k + n - 1
        bv = 0.0
        ik = i - k
        if 0 <= ik < n:
            bv += 1.0 - h[ik]
        jk = j - k
        if 0 <= jk < n:
            bv -= 1.0 - h[jk]
        ki = k + i
        if 0 <= ki < n:
            bv -= h[ki]
        kj = k + j
        if 0 <= kj < n:
            bv += h[kj]
        b[idx] = dx * bv
    a[n - 1] = -2.0 * dx
    a[i - j + n - 1] += dx
    a[j - i + n - 1] += dx
    return b, a

def line_search(q, b, a, lo_t, hi_t):
    npts = 21
    ts = np.linspace(lo_t, hi_t, npts)
    vals = np.empty(npts)
    for i in range(npts):
        t = ts[i]
        vals[i] = np.max(q + b * t + a * t * t)
    bi = int(np.argmin(vals))
    for _ in range(5):
        if bi == 0:
            t0, t1 = lo_t, ts[1]
        elif bi == npts - 1:
            t0, t1 = ts[-2], hi_t
        else:
            t0, t1 = ts[bi - 1], ts[bi + 1]
        ts = np.linspace(t0, t1, 5)
        for i in range(5):
            t = ts[i]
            vals[i] = np.max(q + b * t + a * t * t)
        npts = 5
        bi = int(np.argmin(vals))
    return float(ts[bi]), float(vals[bi])

def pair_polish(h, q, score, rng, n, pbudget, nsweeps):
    best_h = h.copy()
    best_score = score
    for _ in range(nsweeps):
        moved = False
        pairs = set()
        att = 0
        while len(pairs) < pbudget and att < 40 * pbudget:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                pairs.add((min(i, j), max(i, j)))
            att += 1
        ii = 0
        while len(pairs) < pbudget and ii < n - 1:
            for jj in range(ii + 1, n):
                if len(pairs) >= pbudget:
                    break
                pairs.add((ii, jj))
            ii += 1
        plist = list(pairs)
        rng.shuffle(plist)
        for (i, j) in plist:
            lo_t = max(-h[i], h[j] - 1.0)
            hi_t = min(1.0 - h[i], h[j])
            if hi_t - lo_t < 1e-12:
                continue
            b, a = pair_coeffs(h, i, j, n)
            bt, bv = line_search(q, b, a, lo_t, hi_t)
            if bv >= score - 1e-12:
                continue
            h2 = h.copy()
            h2[i] += bt
            h2[j] -= bt
            h2 = project_onto_k(h2, n)
            if not check_feasible(h2, n):

            s2 = exact_score(h2)
            if s2 < score - 1e-12:
                h = h2
                q = lag_scores(h)
                score = s2
                if s2 < best_score:
                    best_score = s2
                    best_h = h.copy()
                moved = True
        if not moved:
            break
    return best_h, best_score

def basin_hop(h, score, rng, n):
    best_h = h.copy()
    best_score = score
    sigma = 0.08
    temps = [0.0015, 0.0005, 0.00015]
    blens = [6, 4, 3]
    for ti in range(3):

        acc = 0

            s2 = exact_score(h2)
            h2, s2 = pair_polish(h2, q2, s2, rng, n, max(8, n // 2), 1)
            if s2 < best_score - 1e-12:
                best_score = s2
                best_h = h2.copy()

                score = s2
                acc += 1
            else:

                if math.exp(r) > rng.random():
                    h = h2.copy()
                    score = s2
                    acc += 1
        fr = acc / float(blens[ti])
        if fr >= 0.6:
            sigma = min(0.25, sigma * 1.2)
        elif fr <= 0.15:
            sigma = max(0.01, sigma * 0.8)
    return best_h, best_score

    parent = globals().get('initial_h_values', None)
    n = 45
    if parent is not None:

    else:

        fallback = np.full(n, 0.5)

    best_score = exact_score(best_h)
    starts = [fallback.copy(),
              1.0 - fallback,
              fallback[::-1].copy(),

    coefs = rng.standard_normal(4)

    for m in range(4):
        pert += coefs[m] * np.cos(freqs[m] * math.pi * xg + phases[m])

        dup = False
        for u in ustarts:
            if np.linalg.norm(s - u) < 1e-8:
                dup = True
                break
        if not dup:
            ustarts.append(s)
    idx_list = list(range(len(ustarts))) if n <= 100 else [i for i in [0,1,3,5,6] if i < len(ustarts)]

    try:

                s = project_onto_k(s, n)
            q = lag_scores(s)

                best_h = s.copy()
            ph, ps = pair_polish(s, q, sc, rng, n, pb, ns)
            if ps < best_score - 1e-12:
                best_score = ps
                best_h = ph.copy()
        bh, bs = basin_hop(best_h, best_score, rng, n)
        if bs < best_score - 1e-12:
            best_score = bs
            best_h = bh.copy()
        ph2, ps2 = pair_polish(best_h, lag_scores(best_h), best_score, rng, n, pb, 2)
        if ps2 < best_score - 1e-12:
            best_score = ps2
            best_h = ph2.copy()

        pass

    if not check_feasible(final_h, n):
        final_h = fallback.copy()
    final_score = exact_score(final_h)
    fb_score = exact_score(fallback)
    if final_score > fb_score + 1e-12:
        final_h = fallback.copy()
        final_score = fb_score
    final_h = project_onto_k(final_h, n)
    final_score = exact_score(final_h)
    return final_h.astype(np.float64), float(final_score), int(n)

        if ps2 < best_score - 1e-12:

```python
import numpy as np
import math

def project_onto_k(y, n):
    S = n / 2.0
    if not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        c = np.clip(y - mid, 0.0, 1.0)
        if c.sum() > S:
            lo = mid
        else:
            hi = mid
    c = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):
        diff = S - c.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = c < 1.0 - 1e-13
        else:
            mask = c > 1e-13
        cnt = mask.sum()
        if cnt == 0:
            break
        c[mask] += diff / cnt
        np.clip(c, 0.0, 1.0, out=c)
    return c

def exact_score(h):
    n = len(h)
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_scores(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def check_feasible(h, n):

            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(h.sum() - n / 2.0) <= 1e-10)
