    blk = np.zeros(n)
    blk[:int(S)] = 1.0
    if S != int(S):
        blk[int(S)] = S - int(S)
    blk = blk[rng.permutation(n)]
    pool.append(project_K(blk, n, S))
    phase = rng.uniform(0.0, 2.0 * math.pi)
    js = np.arange(n)
    pool.append(project_K(0.5 + 0.45 * np.cos(2.0 * math.pi * js / n + phase), n, S))
    pool = [p for p in pool if feasible(p, n, S)]
    pool = dedup_pool(pool)
    scored = []
    for p in pool:
        sc = exact_score(p, dx)
        scored.append((sc, p))
        if np.isfinite(sc) and sc < best_s - 1e-12:
            best_s, best_h = sc, p.copy()
    keys = [s[0] if np.isfinite(s[0]) else 999.0 for s in scored]
    order = list(np.argsort(keys))
    scored = [scored[i] for i in order]
    n_starts = min(8, len(scored)) if n <= 64 else min(5, len(scored))
    starts = [scored[i][1].copy() for i in range(n_starts)]
    if n <= 64:
        n_levels, ppl, n_calib, polish_n = 14, 80, 32, 80
    elif n <= 128:
        n_levels, ppl, n_calib, polish_n = 10, 40, 20, 40
    elif n <= 256:
        n_levels, ppl, n_calib, polish_n = 8, 18, 12, 20
    else:
        n_levels, ppl, n_calib, polish_n = 6, 6, 8, 8
    for st in starts:
        try:
            ch, cf = anneal_chain(st, n, S, dx, rng, 0.02, n_levels, ppl, n_calib)
            if feasible(ch, n, S):
                cs = exact_score(ch, dx)
                if np.isfinite(cs) and cs < best_s - 1e-12:
                    best_s, best_h = cs, ch.copy()
        except Exception:
            pass
    ph, pf = best_h.copy(), exact_score(best_h, dx)
    for sig in [0.02, 0.006]:
        for _ in range(polish_n):
            p = make_proposal(ph, n, S, sig, rng)
            if p is None:
                continue
            sp = exact_score(p, dx)
            if np.isfinite(sp) and sp < pf - 1e-12:
                ph, pf = p, sp
    if np.isfinite(pf) and pf < best_s - 1e-12:
        best_h, best_s = ph.copy(), pf
    fh = project_K(best_h, n, S)
    if not feasible(fh, n, S):
        fh = fb_h.copy()
    fs = exact_score(fh, dx)
    if not np.isfinite(fs) or fs > fb_s + 1e-10:
        fh, fs = fb_h.copy(), fb_s
    return (fh.astype(np.float64), float(fs), int(n))
