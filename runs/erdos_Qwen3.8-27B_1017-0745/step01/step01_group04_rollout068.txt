def _smooth_fg(h,N,tau):
    h=np.asarray(h,dtype=np.float64)
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return (float('inf'), np.zeros(N), float('inf'))
    dx = 2.0 / N
    corr = np.correlate(h, 1.0-h, mode='full')
    c = dx*corr
    if c.size == 0 or not np.all(np.isfinite(c)):
        return (float('inf'), np.zeros(N), float('inf'))
    cmax = float(np.max(c))
    if tau <= 0: tau=1e-12
    expv = np.exp((c - cmax) / tau)
    s = float(expv.sum())
    if not np.isfinite(s) or s <= 0:
        expv = np.zeros_like(c)
        expv[int(np.argmax(c))] = 1.0
        s = 1.0
    p = expv / s
    F = cmax + tau * float(np.log(s))
    g = np.zeros(N, dtype=np.float64)
    n1=N-1
    for k in range(2*N-1):

            g[e:] -= w * h[:L]

import numpy as np

try:
    from scipy.optimize import minimize
except Exception:
    minimize = None


def _uniform(N):
    return np.full(N, 0.5, dtype=np.float64)


def _feasible(h, N, m):
    if h.shape != (N,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if h.min() < -1e-9 or h.max() > 1.0 + 1e-9:
        return False
    return abs(float(h.sum()) - m) <= 1e-8 * max(1, N)


def _project(z, m, N):
    z = np.asarray(z, dtype=np.float64)
    if z.shape != (N,) or not np.all(np.isfinite(z)):
        return _uniform(N)

    lo = float(z.min()) - 1.0
    hi = float(z.max())
    if lo >= hi:
        lo -= 1.0
        hi += 1.0

    for _ in range(64):
        mid = 0.5 * (lo + hi)
        val = float(np.clip(z - mid, 0.0, 1.0).sum())
        if val > m:
            lo = mid
        else:
            hi = mid

    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(4):
        r = m - float(h.sum())
        if abs(r) <= 1e-14:
            break
        if r > 0.0:
            cap = 1.0 - h
            tot = float(cap.sum())
            if tot <= 1e-15:
                break
            if r > tot:
                h = h + cap
            else:
                h = h + (r / tot) * cap
        else:
            val = h
            tot = float(val.sum())
            if tot <= 1e-15:
                break
            if r < -tot:
                h = h - val
            else:
                h = h + (r / tot) * val
        h = np.clip(h, 0.0, 1.0)

    if (not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1.0 + 1e-10
            or abs(float(h.sum()) - m) > 1e-6):
        return _uniform(N)
    return h


def _score(h, N):
    h = np.asarray(h, dtype=np.float64)
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return float("inf")
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0 or not np.all(np.isfinite(corr)):
        return float("inf")
    return float(np.max(corr) * (2.0 / N))


def _taus():
    return (0.02, 0.006, 0.0018, 0.0005, 0.00015, 0.00005, 0.000015, 0.000005)


def _smooth_fg(h, N, tau):
    h = np.asarray(h, dtype=np.float64)
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return float("inf"), np.zeros(N, dtype=np.float64), float("inf")

    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = dx * corr
    if c.size == 0 or not np.all(np.isfinite(c)):
        return float("inf"), np.zeros(N, dtype=np.float64), float("inf")

    cmax = float(np.max(c))
    if tau <= 0.0:
        tau = 1e-12

    expv = np.exp((c - cmax) / tau)
    s = float(expv.sum())
    if not np.isfinite(s) or s <= 0.0:
        expv = np.zeros_like(c)
        expv[int(np.argmax(c))] = 1.0
        s = 1.0

    p = expv / s
    F = cmax + tau * float(np.log(s))
    g = np.zeros(N, dtype=np.float64)
    n1 = N - 1

    for k in range(2 * N - 1):
        w = dx * float(p[k])
        if w == 0.0:
            continue
        d = k - n1
        if d == 0:
            g += w * (1.0 - 2.0 * h)
        elif d > 0:
            e = d
            L = N - e
            if L <= 0:
                continue
            g[e:] += w * (1.0 - h[:L])
            g[:L] -= w * h[e:]
        else:
            e = -d
            L = N - e
            if L <= 0:
                continue
            g[:L] += w * (1.0 - h[e:])
            g[e:] -= w * h[:L]

    if not np.all(np.isfinite(g)):
        return float("inf"), np.zeros(N, dtype=np.float64), float("inf")
    return F, g, cmax


def _optimize(h0, N, m, best_h, best_score, max_iters):
    h = _project(h0, N, m)
    if not _feasible(h, N, m):
        return best_h, best_score

    for tau in _taus():
        F, g, sh = _smooth_fg(h, N, tau)
        if not np.isfinite(F) or not np.isfinite(sh):
            break

        if sh < best_score:
            best_h = h.copy()
            best_score = float(sh)

        gnorm = float(np.linalg.norm(g))
        if gnorm < 1e-13:
            continue

        alpha = 0.25 / (gnorm + 1e-8)
        alpha = min(max(alpha, 1e-6), 8.0)

        it = 0
        bt = 0
        stall = 0

        while it < max_iters:
            y = _project(h - alpha * g, N, m)
            if not _feasible(y, N, m):
                break

            Fy, gy, sy = _smooth_fg(y, N, tau)
            if not np.isfinite(Fy) or not np.isfinite(sy):
                break

            if sy < best_score:
                best_h = y.copy()
                best_score = float(sy)

            dy = y - h
            mov = float(np.max(np.abs(dy))) if dy.size else 0.0
            gdot = float(np.dot(g, dy))
            oldF = F
            oldsh = sh

            accept = False
            if Fy < F - 1e-14 and Fy <= F + 1e-4 * gdot + 1e-12:
                accept = True
            elif sy < oldsh - 1e-10 and Fy <= F + 1e-6:
                accept = True

            if accept:
                h = y
                F = Fy
                g = gy
                sh = sy
                it += 1

                if sy < oldsh - 1e-12:
                    stall = 0
                elif oldF - Fy > 1e-10 * max(1.0, abs(oldF)):
                    stall = 0
                else:
                    stall += 1

                if mov < 1e-11:
                    stall += 1

                alpha = min(alpha * 1.25, 8.0)
                if stall >= 6:
                    break
            else:
                alpha *= 0.5
                bt += 1
                if alpha < 1e-8 or bt > 30:
                    break

    h = _project(h, N, m)
    if _feasible(h, N, m):
        sh = _score(h, N)
        if sh < best_score:
            best_h = h.copy()
            best_score = float(sh)

    return best_h, best_score


def _add_start(starts, cand, N, m, limit):
    if len(starts) >= limit:
        return starts
    cand = _project(cand, N, m)
    if not _feasible(cand, N, m):
        return starts
    for s in starts:
        if np.linalg.norm(cand - s) <= 1e-10:
            return starts
    starts.append(cand)
    return starts


def _random_binary(N, rng):
    arr = np.zeros(N, dtype=np.float64)
    if N % 2 == 0:
        k = N // 2
        if k > 0:
            arr[rng.choice(N, k, replace=False)] = 1.0
    else:
        k = (N - 1) // 2
        if k > 0:
            arr[rng.choice(N, k, replace=False)] = 1.0
        rem = np.where(arr == 0.0)[0]
        if rem.size > 0:
            arr[rng.choice(rem, 1, replace=False)[0]] = 0.5
    rng.shuffle(arr)
    return arr


def _fourier_start(N, rng, amp, n_modes):
    z = np.full(N, 0.5, dtype=np.float64)
    idx = np.arange(N, dtype=np.float64)
    ns = min(n_modes, (N - 1) // 2)
    for r in range(1, ns + 1):
        A = float(rng.standard_normal())
        B = float(rng.standard_normal())
        phase = 2.0 * np.pi * float(r) * idx / float(N)
        z += (amp / float(r)) * (A * np.sin(phase) + B * np.cos(phase))
    return z


def _thue_start(N):
    vals = [0]
    while len(vals) < max(N, 8):
        extra = [1 - v for v in vals]
        vals.extend(extra)
    arr = np.asarray(vals[:N], dtype=np.float64)

    need = N / 2.0 - float(arr.sum())
    if need > 0.0:
        idx = np.where(arr == 0.0)[0][::-1]
        cnt = min(int(round(need)), idx.size)
        if cnt > 0:
            arr[idx[:cnt]] = 1.0
    elif need < 0.0:
        idx = np.where(arr == 1.0)[0][::-1]
        cnt = min(int(round(-need)), idx.size)
        if cnt > 0:
            arr[idx[:cnt]] = 0.0

    if abs(need - round(need)) > 0.0 and need > 0.0:
        idx = np.where(arr == 0.0)[0]
        if idx.size > 0:
            arr[idx[0]] = 0.5
    elif abs(need - round(need)) > 0.0 and need < 0.0:
        idx = np.where(arr == 1.0)[0]
        if idx.size > 0:
            arr[idx[0]] = 0.5

    arr = rng.shuffle(arr) if False else arr
    return arr


def _eq_sum(z, N, m):
    z = np.asarray(z, dtype=np.float64)
    return np.array([float(np.sum(z[:N])) - m], dtype=np.float64)


def _eq_sum_jac(z, N, m):
    J = np.zeros((1, N), dtype=np.float64)
    J[0, :] = 1.0
    return J


def _slsqp_obj(z, N, tau):
    F, _, _ = _smooth_fg(np.asarray(z, dtype=np.float64)[:N], N, tau)
    if not np.isfinite(F):
        return 1e6
    return float(F)


def _slsqp_jac(z, N, tau):
    F, g, _ = _smooth_fg(np.asarray(z, dtype=np.float64)[:N], N, tau)
    if not np.isfinite(F):
        return np.zeros(N, dtype=np.float64)
    return g


def _slsqp_solve(h0, N, m, tau, maxiter):
    if minimize is None:
        return _project(h0, N, m)

    h0 = _project(h0, N, m)
    if not _feasible(h0, N, m):
        return h0

    bounds = [(0.0, 1.0)] * N
    cons = [{"type": "eq", "fun": _eq_sum, "jac": _eq_sum_jac, "args": (N, m)}]

    try:
        res = minimize(
            _slsqp_obj,
            h0,
            jac=_slsqp_jac,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            args=(N, tau),
            options={"maxiter": maxiter, "ftol": 1e-10, "disp": False},
        )
    except Exception:
        return h0

    if res.x is not None:
        cand = np.asarray(res.x[:N], dtype=np.float64)
        cand = _project(cand, N, m)
        if _feasible(cand, N, m):
            return cand
    return h0


def run(seed=42, **kwargs):
    raw = globals().get("initial_h_values", None)
    N = None
    arr = None

    if raw is not None:
        a = np.asarray(raw, dtype=np.float64)
        if a.ndim == 1 and a.size >= 2 and np.all(np.isfinite(a)):
            N = int(a.size)
            arr = a

    if N is None:
        nv = kwargs.get("n_points", 70)
        try:
            N = int(nv)
        except Exception:
            N = 70
        if N < 2:
            N = 70

    if N == 1:
        h = np.array([0.5], dtype=np.float64)
        return h, float(_score(h, 1)), 1

    m = N / 2.0
    try:
        seed_int = int(seed)
    except Exception:
        seed_int = 42
    rng = np.random.default_rng(seed_int)

    if arr is not None and arr.shape == (N,):
        base = _project(arr, m, N)
    else:
        base = _uniform(N)
    if not _feasible(base, N, m):
        base = _uniform(N)

    base_score = float(_score(base, N))

    if N > 999:
        return base, base_score, int(N)

    if N <= 128:
        limit = 10
        max_iter = 26
        hop_scales = (0.05, 0.02, 0.008)
        slsqp_iters = 80
    elif N <= 256:
        limit = 7
        max_iter = 18
        hop_scales = (0.04,)
        slsqp_iters = 50
    elif N <= 512:
        limit = 5
        max_iter = 12
        hop_scales = ()
        slsqp_iters = 0
    else:
        limit = 4
        max_iter = 8
        hop_scales = ()
        slsqp_iters = 0

    best_h = base.copy()
    best_score = base_score

    starts = []
    _add_start(starts, base, N, m, limit)
    for sc in (0.025, 0.07, 0.15):
        _add_start(starts, base + sc * rng.standard_normal(N), N, m, limit)
    _add_start(starts, 1.0 - base, N, m, limit)
    _add_start(starts, _random_binary(N, rng), N, m, limit)
    _add_start(starts, _thue_start(N), N, m, limit)
    _add_start(starts, _random_binary(N, rng), N, m, limit)
    _add_start(starts, _fourier_start(N, rng, 0.18, 4), N, m, limit)
    _add_start(starts, _fourier_start(N, rng, 0.30, 2), N, m, limit)
    _add_start(starts, 0.5 + 0.42 * rng.random(N), N, m, limit)

    base_ref = base_score
    for start in starts:
        s0 = float(_score(start, N))
        if not np.isfinite(s0):
            continue
        if s0 > base_ref + 0.06:
            iters = max(12, max_iter // 2)
        elif s0 > base_ref + 0.015:
            iters = max(16, (3 * max_iter) // 4)
        else:
            iters = max_iter
        best_h, best_score = _optimize(start, N, m, best_h, best_score, iters)

    for sc in hop_scales:
        mut = best_h + sc * rng.standard_normal(N)
        best_h, best_score = _optimize(mut, N, m, best_h, best_score, max_iter)

    if slsqp_iters > 0 and minimize is not None:
        for tau in (5e-05, 1e-05, 3e-06):
            cand = _slsqp_solve(best_h, N, m, tau, slsqp_iters)
            s = float(_score(cand, N))
            if _feasible(cand, N, m) and s < best_score:
                best_h = cand.copy()
                best_score = s

    final = _project(best_h, N, m)
    s_final = float(_score(final, N))
    base_final = _project(base, N, m)
    s_base = float(_score(base_final, N))

    if _feasible(final, N, m) and s_final <= s_base:
        h = final
    else:
        h = base_final

    s = float(_score(h, N))
    if not np.isfinite(s):
        h = _uniform(N)
        s = float(_score(h, N))

    return h, s, int(N)
