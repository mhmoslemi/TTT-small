def project_capped(z,n):
    z=np.asarray(z,dtype=np.float64).ravel()
    if z.size != n: return None
    if not np.all(np.isfinite(z)): return None
    z=np.clip(z,-1.0e12,1.0e12)
    s=0.5*float(n)
    lo=float(np.min(z))-1.0
    hi=float(np.max(z))+1.0
    for _ in range(80):
        mid=0.5*(lo+hi)
        val=np.sum(np.clip(z-mid,0.0,1.0))
        if val > s: lo=mid
        else: hi=mid
    h=np.clip(z-0.5*(lo+hi),0.0,1.0)
    r=s-float(np.sum(h))
    for i in range(n):
        if r > 0.0:
            a=min(r,1.0-h[i]); h[i]+=a; r-=a
        elif r < 0.0:

import numpy as np
from scipy.optimize import minimize

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n:
        return None
    if not np.all(np.isfinite(z)):
        return None
    s = 0.5 * float(n)
    if np.all(z >= -1.0e-12) and np.all(z <= 1.0 + 1.0e-12):
        h = np.clip(z, 0.0, 1.0)
        if abs(float(np.sum(h)) - s) <= 1.0e-8:
            return h
    z = np.clip(z, -1.0e12, 1.0e12)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for i in range(n):
        if r > 0.0:
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
        elif r < 0.0:
            a = min(-r, h[i])
            h[i] -= a
            r += a
        if abs(r) < 1.0e-14:
            break
    for _ in range(n + 2):
        if abs(r) < 1.0e-14:
            break
        if r > 0.0:
            i = int(np.argmax(1.0 - h))
            h[i] = min(h[i] + r, 1.0)
        else:
            i = int(np.argmax(h))
            h[i] = max(h[i] + r, 0.0)
        r = s - float(np.sum(h))
    if not np.all(np.isfinite(h)) or np.any(h < 0.0) or np.any(h > 1.0) or abs(s - float(np.sum(h))) > 1.0e-8:
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1.0e-10) or np.any(h > 1.0 + 1.0e-10):
        return False
    return abs(float(np.sum(h)) - 0.5 * float(n)) <= 1.0e-8

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float("inf")
    q = np.correlate(h, 1.0 - h, mode="full")
    if q.size != 2 * n - 1 or not np.all(np.isfinite(q)):
        return float("inf")
    return float(np.max(q) * (2.0 / float(n)))

def q_own(h, n):
    dx = 2.0 / float(n)
    out = np.empty(2 * n - 1, dtype=np.float64)
    for idx, d in enumerate(range(-(n - 1), n)):
        m = n - abs(d)
        ph = d if d > 0 else 0
        po = -d if d < 0 else 0
        out[idx] = dx * np.dot(h[ph:ph + m], 1.0 - h[po:po + m])
    return out

def q_and_g(h, n):
    dx = 2.0 / float(n)
    L = 2 * n - 1
    q = np.empty(L, dtype=np.float64)
    G = np.zeros((L, n), dtype=np.float64)
    for idx, d in enumerate(range(-(n - 1), n)):
        m = n - abs(d)
        ph = d if d > 0 else 0
        po = -d if d < 0 else 0
        a = h[ph:ph + m]
        b = h[po:po + m]
        q[idx] = dx * np.dot(a, 1.0 - b)
        G[idx, ph:ph + m] += dx * (1.0 - b)
        G[idx, po:po + m] -= dx * a
    return q, G

def lag_grad_one(h, n, d):
    dx = 2.0 / float(n)
    g = np.zeros(n, dtype=np.float64)
    m = n - abs(d)
    ph = d if d > 0 else 0
    po = -d if d < 0 else 0
    g[ph:ph + m] += dx * (1.0 - h[po:po + m])
    g[po:po + m] -= dx * h[ph:ph + m]
    return g

def safe_h(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n:
        return np.full(n, 0.5, dtype=np.float64)
    if not np.all(np.isfinite(z)):
        z = np.where(np.isfinite(z), z, 0.5)
    return np.clip(z, 0.0, 1.0)

def smooth_eq_fun(z, n):
    return float(np.sum(z[:n]) - 0.5 * float(n))

def smooth_eq_jac(z, n):
    return np.ones(n, dtype=np.float64)

def soft_obj(z, n, eps):
    h = safe_h(z, n)
    q = q_own(h, n)
    if not np.all(np.isfinite(q)):
        return 1.0e12
    m = float(np.max(q))
    e = np.exp((q - m) / eps)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return m
    return float(m + eps * np.log(s))

def soft_jac(z, n, eps):
    h = safe_h(z, n)
    q, G = q_and_g(h, n)
    if not np.all(np.isfinite(q)):
        return np.zeros(n, dtype=np.float64)
    m = float(np.max(q))
    e = np.exp((q - m) / eps)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return np.zeros(n, dtype=np.float64)
    return G.T.dot(e / s)

def smooth_step(x, n, eps, maxiter):
    cur = project_capped(x, n)
    if cur is None:
        return x, exact_score(x, n)
    sc0 = exact_score(cur, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': smooth_eq_fun, 'jac': smooth_eq_jac, 'args': (n,)}]
    res = None
    try:
        res = minimize(soft_obj, cur.copy(), jac=soft_jac, args=(n, float(eps)),
                       bounds=bounds, constraints=cons, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1.0e-10, 'disp': False})
    except Exception:
        res = None
    cands = [cur]
    raw_y = None
    raw_s = None
    if res is not None and getattr(res, 'x', None) is not None:
        raw_y = project_capped(res.x, n)
        if raw_y is not None:
            raw_s = exact_score(raw_y, n)
            if np.isfinite(raw_s):
                cands.append(raw_y)
    y = raw_y
    if y is not None:
        delta = y - cur
        for alpha in (1.0, 0.5, 0.25, 0.125, 0.0625):
            zz = project_capped(cur + alpha * delta, n)
            if zz is not None:
                cands.append(zz)
    best = cur
    bestsc = sc0
    for c in cands:
        if is_feasible(c, n):
            s = exact_score(c, n)
            if s < bestsc - 1.0e-13:
                best, bestsc = c, s
    if raw_y is not None and raw_s is not None and np.isfinite(raw_s) and bestsc >= sc0 - 1.0e-13 and raw_s < sc0 + 1.0e-5:
        return raw_y, raw_s
    return best, bestsc

def smooth_chain(x, n, eps_list, maxiter):
    cur = project_capped(x, n)
    if cur is None:
        return x, exact_score(x, n)
    sc = exact_score(cur, n)
    best = cur.copy()
    bestsc = sc
    for eps in eps_list:
        y, scy = smooth_step(cur, n, float(eps), maxiter)
        cur, sc = y, scy
        if np.isfinite(scy) and scy < bestsc - 1.0e-13:
            best, bestsc = y.copy(), scy
    return best, bestsc

def epi_obj(z, n):

    if z.size != n + 1 or not np.all(np.isfinite(z)):
        return 1.0e12
    return float(z[-1])

def epi_obj_jac(z, n):

    v[-1] = 1.0
    return v

def epi_eq_fun(z, n):

    if z.size != n + 1 or not np.all(np.isfinite(z)):
        return 0.0
    return float(np.sum(z[:n]) - 0.5 * float(n))

def epi_eq_jac(z, n):

def epi_ineq(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    L = 2 * n - 1
    if z.size != n + 1 or not np.all(np.isfinite(z)):
        return np.full(L, 1.0e12)
    h = np.clip(z[:n], 0.0, 1.0)
    q = q_own(h, n)
    if not np.all(np.isfinite(q)):
        return np.full(L, 1.0e12)
    return float(z[-1]) - q

def epi_ineq_jac(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    L = 2 * n - 1
    if z.size != n + 1 or not np.all(np.isfinite(z)):
        return np.zeros((L, n + 1))
    h = np.clip(z[:n], 0.0, 1.0)
    _, G = q_and_g(h, n)
    if not np.all(np.isfinite(G)):
        return np.zeros((L, n + 1))
    J = np.empty((L, n + 1), dtype=np.float64)
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def epigraph_step(x, n, maxiter):
    cur = project_capped(x, n)
    if cur is None:
        return x, exact_score(x, n)
    cur_sc = exact_score(cur, n)
    best = cur.copy()
    bestsc = cur_sc
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'ineq', 'fun': epi_ineq, 'jac': epi_ineq_jac, 'args': (n,)},
        {'type': 'eq', 'fun': epi_eq_fun, 'jac': epi_eq_jac, 'args': (n,)}
    ]
    for _ in range(2):
        q = q_own(cur, n)
        if not np.all(np.isfinite(q)):
            break
        m = float(np.max(q))
        z0 = np.empty(n + 1, dtype=np.float64)
        z0[:n] = cur
        z0[n] = m + max(1.0e-9, 1.0e-7 * abs(m))
        res = None
        try:
            res = minimize(epi_obj, z0, jac=epi_obj_jac, args=(n,),
                           bounds=bounds, constraints=cons, method='SLSQP',
                           options={'maxiter': maxiter, 'ftol': 1.0e-10, 'disp': False})
        except Exception:
            res = None
        y = None
        if res is not None and getattr(res, 'x', None) is not None:
            y = project_capped(res.x[:n], n)
        cands = [cur]
        if y is not None:
            cands.append(y)
            delta = y - cur
            for alpha in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125):
                zz = project_capped(cur + alpha * delta, n)
                if zz is not None:
                    cands.append(zz)
        loc = cur
        locsc = cur_sc
        for c in cands:
            if is_feasible(c, n):
                s = exact_score(c, n)
                if s < locsc - 1.0e-13:
                    loc, locsc = c, s
        if locsc < cur_sc - 1.0e-13:
            cur, cur_sc = loc, locsc
            if cur_sc < bestsc - 1.0e-13:
                best, bestsc = cur.copy(), cur_sc
        else:
            break
    return best, bestsc

def active_polish(x, n, iters):
    cur = project_capped(x, n)
    if cur is None:
        return x, exact_score(x, n)
    sc = exact_score(cur, n)
    best = cur.copy()
    bestsc = sc
    if n < 2 or iters <= 0:
        return best, bestsc
    eta = min(20.0, 0.5 / (2.0 / float(n)))
    fail = 0
    use_soft = n <= 120
    eps_act = 0.004
    for _ in range(iters):
        if use_soft:
            q, G = q_and_g(cur, n)
            if not np.all(np.isfinite(q)):
                break
            m = float(np.max(q))
            e = np.exp((q - m) / eps_act)
            s = float(np.sum(e))
            if s <= 0.0 or not np.isfinite(s):
                break
            g = G.T.dot(e / s)
        else:
            q = np.correlate(cur, 1.0 - cur, mode="full")
            if q.size != 2 * n - 1 or not np.all(np.isfinite(q)):
                break
            idx = int(np.argmax(q))
            d = idx - (n - 1)
            g = lag_grad_one(cur, n, d)
        if not np.all(np.isfinite(g)):
            break
        z = project_capped(cur - eta * g, n)
        if z is None or float(np.max(np.abs(z - cur))) < 1.0e-12:
            fail += 1
            eta *= 0.5
            if fail >= 8 or eta < 1.0e-8:
                break
            continue
        bestloc = cur
        bestlocsc = sc
        for alpha in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625):
            y = project_capped(cur + alpha * (z - cur), n)
            if y is not None and is_feasible(y, n):
                sy = exact_score(y, n)
                if sy < bestlocsc - 1.0e-13:
                    bestloc, bestlocsc = y, sy
        if bestlocsc < sc - 1.0e-13:
            cur, sc = bestloc, bestlocsc
            if sc < bestsc - 1.0e-13:
                best, bestsc = cur.copy(), sc
            fail = 0
            eta = min(20.0, 1.1 * eta)
        else:
            fail += 1
            eta *= 0.5
            if fail >= 8 or eta < 1.0e-8:
                break
    return best, bestsc

def add_candidate(pool, v, thr):
    if v is None:
        return
    v = np.asarray(v, dtype=np.float64).ravel()
    if v.size < 1:
        return
    v = project_capped(v, v.size)
    if v is None:
        return
    for u in pool:
        if u.size == v.size and float(np.max(np.abs(v - u))) < thr:
            return
    pool.append(v)

def build_starts(base, n, rng, max_starts):
    base = np.asarray(base, dtype=np.float64).ravel()
    if base.size != n:
        base = np.full(n, 0.5, dtype=np.float64)
    base = project_capped(base, n)
    if base is None:
        base = np.full(n, 0.5, dtype=np.float64)
    pool = [base]
    add_candidate(pool, 0.5 * (base + base[::-1]), 1.0e-8)
    add_candidate(pool, base[::-1].copy(), 1.0e-8)
    add_candidate(pool, 0.5 * (base + 1.0 - base[::-1]), 1.0e-8)
    add_candidate(pool, 1.0 - base[::-1], 1.0e-8)
    add_candidate(pool, np.full(n, 0.5, dtype=np.float64), 1.0e-8)
    blk = np.zeros(n, dtype=np.float64)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    add_candidate(pool, blk, 1.0e-8)
    alt = np.zeros(n, dtype=np.float64)
    alt[0::2] = 1.0
    add_candidate(pool, alt, 1.0e-8)
    add_candidate(pool, rng.uniform(0.0, 1.0, n), 1.0e-8)
    ii = np.arange(n)
    for ph in (0.0, 0.5 * np.pi, np.pi):
        add_candidate(pool, 0.5 + 0.45 * np.sin(2.0 * np.pi * ii / float(n) + ph), 1.0e-8)
    for sd in (0.05, 0.12, 0.25, 0.40):
        add_candidate(pool, base + rng.normal(0.0, sd, n), 1.0e-8)
    k = 0
    while len(pool) < max_starts and k < 20:
        add_candidate(pool, base + rng.normal(0.0, 0.05 + 0.025 * k, n), 1.0e-8)
        k += 1
    return pool[:max_starts]

def make_base_for_n(parent, n):
    if parent is None or parent.size == 0:
        return np.full(n, 0.5, dtype=np.float64)
    if n == parent.size:
        p = project_capped(parent, n)
        return p if p is not None else np.full(n, 0.5, dtype=np.float64)
    xold = np.linspace(0.0, 2.0, parent.size)
    xnew = np.linspace(0.0, 2.0, n)
    v = np.interp(xnew, xold, parent)
    p = project_capped(v, n)
    if p is not None:
        return p
    idx = np.clip(np.searchsorted(xold, xnew), 0, parent.size - 1)
    p = project_capped(parent[idx], n)
    if p is not None:
        return p
    return np.full(n, 0.5, dtype=np.float64)

def better_best(h, n, s, bh, bn, bs, tol):
    if h is not None and np.isfinite(s) and s < bs - tol:
        return h.copy(), int(n), float(s)
    return bh, bn, bs

def run(seed=42, **kwargs):
    parent = None

    try:
        p = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:

    except Exception:
        pass
    if n_primary is None:
        try:
            if 'n_points' in kwargs:
                v = int(kwargs['n_points'])
                if 1 <= v <= 999:
                    n_primary = v
        except Exception:
            pass
    if n_primary is None or not (1 <= n_primary <= 999):
        n_primary = 47

        base0 = project_capped(parent, n_primary)

    if n_primary < 2:
        base0 = np.array([0.5], dtype=np.float64)
        return base0, float(exact_score(base0, 1)), 1
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    best_h = base0.copy()
    best_n = n_primary
    best_sc = exact_score(base0, n_primary)
    ns = [int(n_primary)]
    if 6 <= n_primary <= 60:
        aux = int(min(96, max(n_primary + 13, 2 * n_primary)))
        if aux > 1 and aux != n_primary and aux <= 999:

    elif 60 < n_primary <= 100:
        aux = int(min(128, n_primary + 24))
        if aux != n_primary and aux <= 999:
            ns.append(aux)
    for n in ns[:2]:
        if n < 2:

        if n == n_primary:
            base = base0
        else:
            base = make_base_for_n(parent, n)
        if base is None or not is_feasible(base, n):
            base = np.full(n, 0.5, dtype=np.float64)
        if n <= 64:
            max_starts = 8 if n == n_primary else 6

            epi_iter = 140 if n == n_primary else 0
            active_iter = 12
        elif n <= 96:
            max_starts = 6
            eps_list = (0.05, 0.02, 0.006)
            smooth_iter = 70
            epi_iter = 90 if n == n_primary else 70
            active_iter = 20
        elif n <= 128:
            max_starts = 4
            eps_list = (0.05, 0.02)

            epi_iter = 0

            max_starts = 3
            eps_list = ()
            smooth_iter = 0
            epi_iter = 0
            active_iter = 40
        starts = build_starts(base, n, rng, max_starts)
        cands = []
        for st in starts:
            st = project_capped(st, n)
            if st is None or not is_feasible(st, n):
                continue
            s = exact_score(st, n)
            cands.append((st, float(s)))
            best_h, best_n, best_sc = better_best(st, n, s, best_h, best_n, best_sc, 1.0e-12)
        if not cands:
            continue
        if smooth_iter > 0:
            opt = []
            for st, s in cands:
                cu, su = smooth_chain(st, n, eps_list, smooth_iter)
                if is_feasible(cu, n):
                    su = exact_score(cu, n)
                else:
                    cu, su = base, exact_score(base, n)
                best_h, best_n, best_sc = better_best(cu, n, su, best_h, best_n, best_sc, 1.0e-12)

                    cu, _ = opt[pos]
                    ca, sa = active_polish(cu, n, active_iter)

    if not np.isfinite(s_final):

        s_final = exact_score(final_h, best_n)
    return final_h, float(s_final), int(best_n)
