def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if resid > 1e-14:
        for i in np.argsort(1.0 - y):
            add = min(resid, 1.0 - y[i])
            y[i] += add
            resid -= add
            if resid < 1e-15: break
    elif resid < -1e-14:
        for i in np.argsort(y):
            sub = min(-resid, y[i])
            y[i] -= sub
            resid += sub
            if -resid < 1e-15: break
    y = np.clip(y, 0.0, 1.0)
    if np.any(~np.isfinite(y)): return None
    if abs(np.sum(y) - S) > 1e-9: return None
    return y
```

def project_simplex(v):
    n = len(v)
    sv = np.sort(v)[::-1]
    cv = np.cumsum(sv)
    rho = np.where(sv - (cv - 1.0) / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0: return np.zeros(n)
    r = rho[-1]
    theta = (cv[r] - 1.0) / (r + 1)
    w = np.maximum(v - theta, 0.0)
    w /= w.sum()  # fix roundoff
    return w
```

def compute_Q_G(h, n):

import numpy as np

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0 or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if resid > 1e-13:
        idx = np.argsort(1.0 - y)
        for ii in idx:
            a = min(resid, 1.0 - y[ii])
            y[ii] += a
            resid -= a
            if resid < 1e-14:
                break
    elif resid < -1e-13:
        idx = np.argsort(y)
        for ii in idx:
            a = min(-resid, y[ii])
            y[ii] -= a
            resid += a
            if -resid < 1e-14:
                break
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(np.sum(y) - S) > 1e-8:
        return None
    return y

def project_simplex(v):
    n = len(v)
    sv = np.sort(v)[::-1]
    cv = np.cumsum(sv)
    rn = np.arange(1, n + 1, dtype=np.float64)
    rho = np.where(sv - (cv - 1.0) / rn > 0)[0]
    if len(rho) == 0:
        return np.full(n, 1.0 / n)
    r = int(rho[-1])
    theta = (cv[r] - 1.0) / (r + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.full(n, 1.0 / n)
    w /= s
    return w

def exact_score(h, n):
    if h is None or len(h) != n or not np.all(np.isfinite(h)):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def compute_QG(h, n):
    nl = 2 * n - 1
    Q = np.zeros(nl)
    G = np.zeros((nl, n))
    for k in range(-(n - 1), n):
        row = k + n - 1
        ilo = max(0, k)
        ihi = min(n - 1, n - 1 - k)
        i = np.arange(ilo, ihi + 1)
        j = i - k
        Q[row] = np.sum(h[i] * (1.0 - h[j]))
        G[row, i] += (1.0 - h[j])
        G[row, j] += (-h[i])
    return Q, G

def solve_surrogate(x, Q, G, S, warm, n, Imax):
    nl = len(Q)
    L = max(1e-8, 0.5001 * float(np.sum(G * G)))
    if warm is not None and np.all(np.isfinite(warm)) and len(warm) == nl:
        w = project_simplex(warm.copy())
    else:
        w = np.zeros(nl)
        w[int(np.argmax(Q))] = 1.0
    v = w.copy()
    t = 1.0
    best_y = x.copy()
    best_model = float(np.max(Q))
    chi = max(5, Imax // 25)
    for it in range(Imax):
        z = (G.T @ v) * 0.5
        y = project_box_sum(x - z, S)
        if y is None:
            break
        d = y - x
        if not np.all(np.isfinite(d)):
            break
        vals = Q + G @ d
        model = float(np.dot(d, d) + np.max(vals))
        if model < best_model:
            best_model = model
            best_y = y.copy()
        if it % chi == chi - 1 or it == Imax - 1:
            gap = float(np.max(vals) - np.dot(w, vals))
            if gap <= 1e-7 * max(1.0, float(np.max(np.abs(Q)))):
                break
        grad = Q + G @ d
        w_new = project_simplex(v + grad / L)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        v_new = w_new + ((t - 1.0) / t_new) * (w_new - v)
        t, v, w = t_new, v_new, w_new
    z = (G.T @ w) * 0.5
    y_f = project_box_sum(x - z, S)
    if y_f is not None and np.all(np.isfinite(y_f)):
        d_f = y_f - x
        m_f = float(np.dot(d_f, d_f) + np.max(Q + G @ d_f))
        if m_f < best_model:
            best_y = y_f.copy()
    return best_y, w

def line_search(x, d, Q, G, n, S):
    dn = float(np.dot(d, d))
    if dn < 1e-20:
        return x.copy(), exact_score(x, n)
    p = G @ d
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c_ = b - gr * (b - a)
    d_ = a + gr * (b - a)
    fc = c_ * c_ * dn + float(np.max(Q + c_ * p))
    fd = d_ * d_ * dn + float(np.max(Q + d_ * p))
    for _ in range(40):
        if fc < fd:
            b, d_, fd = d_, c_, fc
            c_ = b - gr * (b - a)
            fc = c_ * c_ * dn + float(np.max(Q + c_ * p))
        else:
            a, c_, fc = c_, d_, fd
            d_ = a + gr * (b - a)
            fd = d_ * d_ * dn + float(np.max(Q + d_ * p))
    candidates = [1.0, 0.5, 0.25, 0.125, 0.5 * (a + b)]
    best_h = x.copy()
    best_sc = exact_score(x, n)
    for alpha in candidates:
        alpha = max(0.0, min(1.0, alpha))
        y = project_box_sum(x + alpha * d, S)
        if y is None:
            continue
        sc = exact_score(y, n)
        if sc < best_sc:
            best_sc = sc
            best_h = y
    return best_h, best_sc

def mm_local(start, n, S, Omax, Imax):
    x = project_box_sum(start, S)
    if x is None:
        return start, exact_score(start, n)
    sc = exact_score(x, n)
    best_x, best_sc = x.copy(), sc
    warm = None
    for _ in range(Omax):
        Q, G = compute_QG(x, n)
        y, warm = solve_surrogate(x, Q, G, S, warm, n, Imax)
        d = y - x
        if np.max(np.abs(d)) < 1e-10:
            break
        new_x, new_sc = line_search(x, d, Q, G, n, S)
        if new_sc < sc - 1e-13:
            if new_sc < best_sc - 1e-14:
                best_sc = new_sc
                best_x = new_x.copy()
            sc = new_sc
            x = new_x
        else:
            break
    return best_x, best_sc
