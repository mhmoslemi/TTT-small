import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-9)

def proj(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):
        return None
    if np.max(np.abs(z)) > 1e12:
        return None
    n = len(z)
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(z - mid, 0, 1)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(z - (lo + hi) * 0.5, 0, 1)
    r = S - y.sum()
    for i in range(n):
        if r <= 1e-14: break
        a = min(r, 1.0 - y[i]); y[i] += a; r -= a
    for i in range(n):
        if r >= -1e-14: break
        a = min(-r, y[i]); y[i] -= a; r += a
    return y if (abs(y.sum() - S) < 1e-9 and np.all(np.isfinite(y))) else None

def exact_score(h, n):
    try:
        v = np.correlate(h, 1.0 - h, 'full') * (2.0 / n)
        return float(v.max()) if v.shape == (2*n-1,) and np.all(np.isfinite(v)) else None
    except Exception:
        return None

def lags_grad(h, n):
    dx = 2.0 / n
    m = 2 * n - 1
    q = np.zeros(m)
    R, C, D = [], [], []
    for d in range(-(n-1), n):
        row = d + n - 1
        lo, hi = max(0, -d), min(n, n - d)
        s = 0.0
        for i in range(lo, hi):
            s += dx * h[i] * (1.0 - h[i+d])
            R += [row, row]; C += [i, i+d]
            D += [dx*(1.0-h[i+d]), -dx*h[i]]
        q[row] = s
    G = sp.coo_matrix((D, (R, C)), shape=(m, n)).tocsr()
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G.data)):
        return None, None
    return q, G

def lp_step(x, q, G, rho, n):
    dx = 2.0 / n
    m = 2 * n - 1
    c = np.zeros(2*n+1); c[n:2*n] = 0.002*dx; c[2*n] = 1.0
    I = sp.eye(n, format='csr')
    Zm = sp.csr_matrix((m, n)); Z1 = sp.csr_matrix((m, 1)); Z2 = sp.csr_matrix((n, 1))
    Em = sp.csr_matrix(np.eye(m))
    A1 = sp.bmat([[G, Zm, -Em]], 'csr')
    A2 = sp.bmat([[I, -I, Z2]], 'csr')
    A3 = sp.bmat([[-I, -I, Z2]], 'csr')
    A_ub = sp.vstack([A1, A2, A3], 'csr')
    b_ub = np.concatenate([-q, np.zeros(2*n)])
    A_eq = sp.csr_matrix(np.hstack([np.ones(n), np.zeros(n+1)]))
    lb = np.maximum(-rho, -x + 1e-12)
    ub = np.minimum(rho, 1.0 - x - 1e-12)
    lb = np.concatenate([lb, np.zeros(n), [-np.inf]])
    ub = np.concatenate([ub, np.full(n, np.inf), [np.inf]])
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                    bounds=list(zip(lb, ub)), method='highs',
                    options={'time_limit': 8, 'presolve': True})
        if r.success and np.all(np.isfinite(r.x[:n])) and abs(r.x[:n].sum()) < 1e-5:
            return r.x[:n]
    except Exception:
        pass
    return None

def optimize_traj(x, n, rng, max_iter):
    f = exact_score(x, n)
    if f is None: return x, f
    best_x, best_f = x.copy(), f
    rho = 0.08
    fails = 0
    for _ in range(max_iter):
        if fails >= 4 or rho < 2e-4: break
        q, G = lags_grad(x, n)
        if q is None: break
        d = lp_step(x, q, G, rho, n)
        if d is None:
            rho *= 0.5; fails += 1; continue
        accepted = False
        for j in range(9):
            beta = 2.0**(-j)
            xn = proj(x + beta * d, n / 2.0)
            if xn is None or not is_feasible(xn, n): continue
            fn = exact_score(xn, n)
            if fn is None: continue
            if fn < f - 1e-12:
                Gd = (G @ d).flatten()
                pred = f - (q + Gd).max()
                act = f - fn
                x = xn.copy(); f = fn
                if f < best_f: best_x, best_f = x.copy(), f
                if pred > 1e-12:
                    r = act / pred
                    rho = max(2e-4, min(0.25, rho * (0.5 if r < 0.25 else 1.5 if r > 0.75 else 1.0)))
                else: rho = max(2e-4, rho * 0.5)
                accepted = True
                break
        if not accepted:
            rho *= 0.5; fails += 1
        if np.max(np.abs(x - best_x)) < 1e-10: break
    return best_x, best_f

def polish(x, n):
    f = exact_score(x, n)
    if f is None: return x, f
    for j in range(n):
        for eps in [0.005, 0.002, 0.001]:
            for sign in [1, -1]:
                xn = x.copy()
                new_val = xn[j] + sign * eps
                if new_val < 0 or new_val > 1: continue
                xn[j] = new_val
                if abs(xn.sum() - n/2.0) > 1e-9: continue
                fn = exact_score(xn, n)
                if fn is not None and fn < f - 1e-12:
                    x = xn.copy(); f = fn
    return x, f

def run(seed=42, **kwargs):
    try: seed = int(seed)
    except (TypeError, ValueError): seed = 42
    if seed < 0: seed = 42
    rng = np.random.default_rng(seed)
    n = 80
    parent = None
    iv = globals().get('initial_h_values', None)

                    p = proj(arr, n / 2.0)
                    if p is not None and is_feasible(p, n): parent = p
        except Exception: pass
    S = n / 2.0
    h_const = np.full(n, 0.5)
    best_h, best_f = h_const.copy(), exact_score(h_const, n)
    if best_f is None: best_f = 0.5
    if parent is not None:
        pf = exact_score(parent, n)
        if pf is not None and pf < best_f: best_f, best_h = pf, parent.copy()
    starts = []
    if parent is not None:
        starts.append(parent.copy())
        for sig in [0.04, 0.10, 0.18]:
            p2 = proj(parent + rng.normal(0, sig, n), S)
            if p2 is not None and is_feasible(p2, n): starts.append(p2)
        p3 = parent[::-1].copy()
        if is_feasible(p3, n): starts.append(p3)
        p4 = 1.0 - parent.copy()
        if is_feasible(p4, n): starts.append(p4)
    idx = np.arange(n, dtype=np.float64)
    M = min(5, (n-1) // 2)
    lf = np.full(n, 0.5)
    for m in range(1, M+1):
        a = rng.normal(0, 0.10 / np.sqrt(m))
        phi = rng.uniform(0, 2.0 * np.pi)
        lf += a * np.cos(2.0 * np.pi * m * (idx + 0.5) / n + phi)
    lf_p = proj(lf, S)
    if lf_p is not None and is_feasible(lf_p, n): starts.append(lf_p)
    if parent is None:
        perm = rng.permutation(n)
        bv = np.zeros(n); bv[perm[:n//2]] = 1.0
        if n % 2: bv[perm[n//2]] = 0.5
        starts.append(bv)
    deduped = []
    for s in starts:
        if not any(np.max(np.abs(s - d)) < 1e-9 for d in deduped):
            deduped.append(s)
    max_starts = 6 if n <= 128 else 4 if n <= 256 else 3
    deduped = deduped[:max_starts]
    for s in deduped:
        sf = exact_score(s, n)
        if sf is not None and sf < best_f: best_f, best_h = sf, s.copy()
    max_iter = 35 if n <= 128 else 22 if n <= 256 else 14
    for s in deduped:
        ox, of = optimize_traj(s.copy(), n, rng, max_iter)
        if of is not None and of < best_f: best_f, best_h = of, ox.copy()
    for rst in range(2):

        rs = proj(best_h + rng.normal(0, sig, n), S)
        if rs is not None and is_feasible(rs, n) and np.max(np.abs(rs - best_h)) > 1e-6:
            ox, of = optimize_traj(rs.copy(), n, rng, max_iter)
            if of is not None and of < best_f: best_f, best_h = of, ox.copy()
    best_h, best_f = polish(best_h, n)
    if not is_feasible(best_h, n):
        best_h = proj(best_h, S)
        if best_h is None: best_h = h_const.copy()
    ff = exact_score(best_h, n)
    if ff is None or ff >= best_f:
        if parent is not None:
            pf2 = exact_score(parent, n)
            if pf2 is not None and pf2 < (ff if ff is not None else 1.0):
                best_h, ff = parent.copy(), pf2
        if ff is None or ff > 0.5:
            best_h, ff = h_const.copy(), 0.5
    return best_h, float(ff), n
