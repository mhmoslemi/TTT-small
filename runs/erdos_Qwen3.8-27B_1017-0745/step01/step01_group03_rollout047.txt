```
def compute_QG(h, lags):
    n = int(h.size)
    m = len(lags)
    Q = np.zeros(m, dtype=np.float64)
    G = np.zeros((m, n), dtype=np.float64)
    for idx, k in enumerate(lags):
        i0 = max(0, int(k)); i1 = min(n-1, n-1+int(k))
        i = np.arange(i0, i1+1)
        j = i - int(k)
        hi = h[i]; hj = h[j]
        Q[idx] = float(np.sum(hi * (1.0 - hj)))
        G[idx, i] += (1.0 - hj)
        G[idx, j] += (-hi)
    return Q, G
```

```
    n=int(v.size)

    y=np.clip(v-0.5*(lo+hi),0.0,1.0)
    resid=S-float(np.sum(y))
```

```python
import numpy as np
from scipy.optimize import linprog, minimize

ACTIVE_LAGS = []

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        m = 0.5 * (lo + hi)
        y = np.clip(v - m, 0.0, 1.0)
        if float(np.sum(y)) > S:
            lo = m
        else:
            hi = m
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    if abs(float(np.sum(y)) - S) > 1e-9 * max(1.0, float(v.size)):
        return None
    return y

def ok(h, n):
    return (h is not None and h.shape == (n,) and np.all(np.isfinite(h)) and
            np.all(h >= -1e-9) and np.all(h <= 1.0 + 1e-9) and
            abs(float(np.sum(h)) - n / 2.0) < 1e-7 * max(1.0, float(n)))

def score(h, n):
    if not ok(h, n):
        return np.inf
    c = np.correlate(h, 1.0 - h, mode="full")
    v = float(np.max(c)) * (2.0 / n)
    return v if np.isfinite(v) else np.inf

def add_candidate(v, cands, best, best_s, n, maxc):
    if v is None or len(cands) >= maxc:
        return cands, best, best_s
    hp = project_box_sum(v, n / 2.0)
    if not ok(hp, n):
        return cands, best, best_s
    s = score(hp, n)
    cands.append((s, len(cands), np.array(hp, dtype=np.float64)))
    if best is None or s < best_s:
        return cands, hp, s
    return cands, best, best_s

def run_mask(n, r, hs):
    if r < 2 or hs < r or hs > n or (n - hs) < (r - 1):
        return None
    hl = hs // r
    rem = hs % r
    hi = [hl + (1 if i < rem else 0) for i in range(r)]
    gs = n - hs
    gc = r - 1
    gl = gs // gc
    gr = gs % gc
    ga = [gl + (1 if i < gr else 0) for i in range(gc)]
    a = []
    for i in range(r):
        a.extend([1] * hi[i])
        if i < r - 1:
            a.extend([0] * ga[i])
    return np.array(a, dtype=np.float64) if len(a) == n else None

def line_minimax(x, y, n, S):
    if not ok(x, n) or not ok(y, n):
        return None, np.inf
    d = y - x
    m = 2 * n - 1
    A = np.empty(m, dtype=np.float64)
    B = np.empty(m, dtype=np.float64)
    D = np.empty(m, dtype=np.float64)
    for kk in range(-(n - 1), n):
        idx = kk + n - 1
        i0 = max(0, kk)
        i1 = min(n - 1, n - 1 + kk)
        i = np.arange(i0, i1 + 1)
        j = i - kk
        di = d[i]
        dj = d[j]
        xi = x[i]
        xj = x[j]
        A[idx] = -float(np.sum(di * dj))
        B[idx] = float(np.sum(di * (1.0 - xj) - xi * dj))
        D[idx] = float(np.sum(xi * (1.0 - xj)))

    cand = [0.0, 1.0]
    for q in range(m):
        if abs(A[q]) > 1e-12:
            a = -B[q] / (2.0 * A[q])
            if np.isfinite(a) and -1e-10 <= a <= 1.0 + 1e-10:
                cand.append(min(1.0, max(0.0, float(a))))

    for p in range(m):
        for q in range(p + 1, m):
            ca = A[p] - A[q]
            cb = B[p] - B[q]
            cc = D[p] - D[q]
            if abs(ca) > 1e-12:
                disc = cb * cb - 4.0 * ca * cc
                scale = max(1.0, abs(cb * cb), abs(4.0 * ca * cc))
                if disc < 0.0:
                    if disc > -1e-10 * scale:
                        disc = 0.0
                    else:
                        continue
                sq = np.sqrt(disc) if disc > 0.0 else 0.0
                den = 2.0 * ca
                for rr in ((-cb - sq) / den, (-cb + sq) / den):
                    if np.isfinite(rr) and -1e-10 <= rr <= 1.0 + 1e-10:
                        cand.append(min(1.0, max(0.0, float(rr))))
            elif abs(cb) > 1e-12:
                rr = -cc / cb
                if np.isfinite(rr) and -1e-10 <= rr <= 1.0 + 1e-10:
                    cand.append(min(1.0, max(0.0, float(rr))))

    cand.sort()
    uniq = []
    for a in cand:
        if a < 0.0 or a > 1.0:
            continue
        if not uniq or a - uniq[-1] > 1e-10:
            uniq.append(a)
    alphas = np.array(uniq, dtype=np.float64)
    if alphas.size == 0:
        alphas = np.array([0.0], dtype=np.float64)

    Qv = A[None, :] * alphas[:, None] ** 2 + B[None, :] * alphas[:, None] + D[None, :]
    pred = np.max(Qv, axis=1)
    order = np.argsort(pred)
    nsel = min(alphas.size, 22)
    sel = order[:nsel]
    sel_list = [0.0, 1.0]
    for ii in sel:
        sel_list.append(float(alphas[ii]))
    ba = float(alphas[order[0]])
    for da in (-1e-5, -1e-4, 1e-4, 1e-5):
        a = ba + da
        if 0.0 <= a <= 1.0:
            sel_list.append(a)
    sel_list.sort()
    final_sel = []
    for a in sel_list:
        if a < 0.0 or a > 1.0:
            continue
        if not final_sel or a - final_sel[-1] > 1e-10:
            final_sel.append(a)

    best_h = None
    best_s = np.inf
    for a in final_sel:
        hp = project_box_sum(x + a * d, S)
        if ok(hp, n):
            s = score(hp, n)
            if s < best_s - 1e-15:
                best_h = hp
                best_s = s
    return best_h, best_s

def build_lp(x, r, n, S):
    if n > 46:
        return None
    try:
        lo = np.clip(x - r, 0.0, 1.0)
        ui = np.clip(x + r, 0.0, 1.0)
        P = n * (n + 1) // 2
        N = n + P + 1
        t = N - 1
        nrow = 4 * P + (2 * n - 1)
        Ab = np.zeros((nrow, N), dtype=np.float64)
        bb = np.zeros(nrow, dtype=np.float64)
        row = 0
        lb = np.zeros(P, dtype=np.float64)
        ub = np.ones(P, dtype=np.float64)

        for i in range(n):
            li = lo[i]
            uii = ui[i]
            for j in range(i, n):
                p = i * n - i * (i - 1) // 2 + (j - i)
                lj = lo[j]
                uj = ui[j]
                lb[p] = li * lj
                ub[p] = uii * uj
                cp = n + p
                Ab[row, j] -= li
                Ab[row, i] -= lj
                Ab[row, cp] = -1.0
                bb[row] = -li * lj
                row += 1
                Ab[row, j] -= uii
                Ab[row, i] -= uj
                Ab[row, cp] = -1.0
                bb[row] = -uii * uj
                row += 1
                Ab[row, cp] = 1.0
                Ab[row, j] -= uii
                Ab[row, i] -= lj
                bb[row] = -uii * lj
                row += 1
                Ab[row, cp] = 1.0
                Ab[row, j] -= li
                Ab[row, i] -= uj
                bb[row] = -li * uj
                row += 1

        for k in range(-(n - 1), n):
            i0 = max(0, k)
            i1 = min(n - 1, n - 1 + k)
            rr = row
            Ab[rr, t] = -1.0
            bb[rr] = 0.0
            for i in range(i0, i1 + 1):
                j = i - k
                a, b = i, j
                if b < a:
                    a, b = b, a
                p = a * n - a * (a - 1) // 2 + (b - a)
                Ab[rr, i] += 1.0
                Ab[rr, n + p] -= 1.0
            row += 1

        Aeq = np.zeros((1, N), dtype=np.float64)
        Aeq[0, :n] = 1.0
        b_eq = np.array([S], dtype=np.float64)

        bounds = [(float(lo[i]), float(ui[i])) for i in range(n)]
        for p in range(P):
            lbelow = float(lb[p])
            ubove = float(ub[p])
            if lbelow > ubove:
                lbelow, ubove = ubove, lbelow
            bounds.append((lbelow, ubove))
        bounds.append((0.0, float(n)))

        c = np.zeros(N, dtype=np.float64)
        c[t] = 1.0
        res = linprog(c, A_ub=Ab, b_ub=bb, A_eq=Aeq, b_eq=b_eq,
                      bounds=bounds, method="highs", options={"presolve": True})
        if res is None or not getattr(res, "success", False):
            return None
        h = np.asarray(res.x[:n], dtype=np.float64)
        if h.shape != (n,) or not np.all(np.isfinite(h)):
            return None
        if np.any(h < lo - 1e-6) or np.any(h > ui + 1e-6):
            return None
        ep = project_box_sum(np.clip(h, lo, ui), S)
        return ep if ok(ep, n) else None
    except Exception:
        return None

def lp_search(h0, n, S):
    h = project_box_sum(h0, S)
    if not ok(h, n):
        return h0, score(h0, n)
    hs = score(h, n)
    if n > 46:
        return h, hs
    r = 0.20
    fail = 0
    for _ in range(5):
        if r < 1e-4 and fail >= 2:
            break
        ep = build_lp(h, r, n, S)
        moved = False
        if ep is not None:
            es = score(ep, n)
            if es < hs - 1e-12:
                h, hs, moved = ep, es, True
            else:
                if float(np.max(np.abs(ep - h))) > 1e-12:
                    ch, cs = line_minimax(h, ep, n, S)
                    if ch is not None and cs < hs - 1e-12:
                        h, hs, moved = ch, cs, True
        if moved:
            r = min(0.25, 1.15 * r)
            fail = 0
        else:
            r *= 0.5
            fail += 1
    return h, hs

def qg(h, lags):
    n = int(h.size)
    m = len(lags)
    Q = np.zeros(m, dtype=np.float64)
    G = np.zeros((m, n), dtype=np.float64)
    for idx, k in enumerate(lags):
        i0 = max(0, int(k))
        i1 = min(n - 1, n - 1 + int(k))
        i = np.arange(i0, i1 + 1)
        j = i - int(k)
        hi = h[i]
        hj = h[j]
        Q[idx] = float(np.sum(hi * (1.0 - hj)))
        G[idx, i] += (1.0 - hj)
        G[idx, j] += (-hi)
    return Q, G

def sl_eq(z):
    n = int(z.size) - 1
    return np.array([float(np.sum(z[:n])) / n - 0.5], dtype=np.float64)

def sl_eq_jac(z):
    n = int(z.size) - 1
    J = np.zeros((1, n + 1), dtype=np.float64)
    J[0, :n] = 1.0 / n
    return J

def sl_ineq(z):
    n = int(z.size) - 1
    Q, _ = qg(z[:n], ACTIVE_LAGS)
    return z[-1] - Q

def sl_ineq_jac(z):
    n = int(z.size) - 1
    _, G = qg(z[:n], ACTIVE_LAGS)
    J = np.zeros((len(ACTIVE_LAGS), n + 1), dtype=np.float64)
    J[:, :n] = -G
    J[:, -1] = 1.0
    return J

def sl_obj(z):
    return float(z[-1])

def sl_obj_jac(z):
    g = np.zeros(int(z.size), dtype=np.float64)
    g[-1] = 1.0
    return g

def slsqp_polish(h):
    if not ACTIVE_LAGS:
        return None
    n = int(h.size)
    S = n / 2.0
    Q, _ = qg(h, ACTIVE_LAGS)
    z0 = np.concatenate([np.asarray(h, dtype=np.float64),
                         np.array([float(np.max(Q)) + 1e-4], dtype=np.float64)])
    cons = [
        {"type": "eq", "fun": sl_eq, "jac": sl_eq_jac},
        {"type": "ineq", "fun": sl_ineq, "jac": sl_ineq_jac},
    ]
    try:
        res = minimize(sl_obj, z0, jac=sl_obj_jac, method="SLSQP",
                       bounds=[(0.0, 1.0)] * n + [(0.0, float(n))],
                       constraints=cons,
                       options={"maxiter": 60, "ftol": 1e-11, "disp": False})
    except Exception:
        return None
    if res is None or res.x is None or len(res.x) != n + 1:
        return None
    hp = project_box_sum(res.x[:n], S)
    return hp if ok(hp, n) else None

def set_active(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    Q = c * (n / 2.0)
    idx = np.argsort(Q)[-12:]
    ACTIVE_LAGS[:] = [int(o - (n - 1)) for o in idx]

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = None
    n = 45
    try:
        pv = np.asarray(initial_h_values, dtype=np.float64)
        if (pv.ndim == 1 and 2 <= pv.size <= 999 and np.all(np.isfinite(pv)) and
                np.all(pv >= -1e-9) and np.all(pv <= 1.0 + 1e-9)):
            target = pv.size / 2.0
            if abs(float(np.sum(pv)) - target) < 1e-6 * max(1.0, float(pv.size)):
                n = int(pv.size)
                parent = project_box_sum(np.clip(pv, 0.0, 1.0), target)
    except Exception:
        parent = None
    if parent is None:
        n = 45

    S = n / 2.0
    fallback_h = parent.copy() if parent is not None else np.full(n, 0.5, dtype=np.float64)

    cands = []
    best = None
    best_s = np.inf

    if parent is not None:
        cands, best, best_s = add_candidate(parent, cands, best, best_s, n, 14)
        for sig in (0.02, 0.05, 0.09, 0.16, 0.28):
            cands, best, best_s = add_candidate(
                parent + sig * rng.standard_normal(n), cands, best, best_s, n, 14)
        cands, best, best_s = add_candidate(parent[::-1], cands, best, best_s, n, 14)
        for rr in (8, 9, 10):
            for hs in (n // 2, n // 2 + 1):
                msk = run_mask(n, rr, hs)
                if msk is not None:
                    cands, best, best_s = add_candidate(
                        0.5 + 0.45 * (msk - 0.5), cands, best, best_s, n, 14)
    else:
        base = np.full(n, 0.5, dtype=np.float64)
        cands, best, best_s = add_candidate(base, cands, best, best_s, n, 14)
        for _ in range(5):
            cands, best, best_s = add_candidate(
                base + 0.25 * rng.standard_normal(n), cands, best, best_s, n, 14)
        x = np.arange(n, dtype=np.float64)
        for amp, fr in ((0.20, 1), (0.18, 2), (0.20, 3)):
            ph = 2.0 * np.pi * float(rng.random())
            cands, best, best_s = add_candidate(
                base + amp * np.cos(2.0 * np.pi * fr * x / n + ph),
                cands, best, best_s, n, 14)

    cands.sort()
    for i in range(min(3, len(cands))):
        h1, s1 = lp_search(cands[i][2], n, S)
        cands, best, best_s = add_candidate(h1, cands, best, best_s, n, 30)
        if best_s <= 0.3808:
            break

    if best_s > 0.3808:
        cands.sort()
        for i in range(min(3, len(cands))):
            h = cands[i][2]
            sc = cands[i][0]
            for _ in range(2):
                set_active(h, n)
                h1 = slsqp_polish(h)
                if h1 is None:
                    break
                s1 = score(h1, n)
                if s1 < sc - 1e-12:
                    h = h1
                    sc = s1
                    cands.append((sc, len(cands), np.array(h, dtype=np.float64)))
                    if sc < best_s:
                        best, best_s = h, sc
                else:
                    break
                if best_s <= 0.3808:
                    break
            if best_s <= 0.3808:
                break

    if best_s > 0.3808 and n <= 46:
        for sig in (0.03, 0.08):
            if best_s <= 0.3808:
                break
            v = best + sig * rng.standard_normal(n)
            h1, s1 = lp_search(v, n, S)
            cands, best, best_s = add_candidate(h1, cands, best, best_s, n, 30)

    final_h = project_box_sum(best, S)
    final_s = score(final_h, n) if final_h is not None else np.inf
    fb = project_box_sum(fallback_h, S)
    fb_s = score(fb, n) if fb is not None else np.inf
    if final_h is None or not np.isfinite(final_s) or final_s > fb_s:
        final_h = fb
        final_s = fb_s
    if final_h is None or not ok(final_h, n):
        final_h = np.full(n, 0.5, dtype=np.float64)
        final_h = project_box_sum(final_h, S)
        final_s = score(final_h, n)
    return np.array(final_h, dtype=np.float64), float(final_s), int(n)
```
