import numpy as np

def project_capped(z, S):
    n = len(z)
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(80):
        mid = 0.5*(lo+hi)
        if np.sum(np.clip(z-mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5*(lo+hi)
    x = np.clip(z-lam, 0.0, 1.0)
    r = S - np.sum(x)
    for i in range(n):
        if r > 0:
            a = min(r, 1.0-x[i]); x[i] += a; r -= a
            if r <= 0: break
        elif r < 0:
            a = min(-r, x[i]); x[i] -= a; r += a
            if r >= 0: break
    if abs(np.sum(x)-S) > 1e-10 or not np.all(np.isfinite(x)):
        return None
    return np.clip(x, 0.0, 1.0)

def is_feasible(h, n):
    return (h.ndim==1 and len(h)==n and np.all(np.isfinite(h))
            and np.all(h>=0) and np.all(h<=1) and abs(np.sum(h)-n/2.0)<=1e-10)

def exact_score(h, n):
    vals = np.correlate(h, 1.0-h, mode='full') * (2.0/n)
    if not np.all(np.isfinite(vals)):
        return None
    return float(np.max(vals))

def get_lags(h, n):
    return np.correlate(h, 1.0-h, mode='full') * (2.0/n)

    kp = n-1 + (p-r)

    if 0 <= kp <= 2*n-2: a[kp] = dx
    if 0 <= kr <= 2*n-2: a[kr] = dx

def build_polish_pairs(n, rng):
    P = min(n*(n-1)//2, 2*n, max(24, 80000//n))
    P = max(2, P)
    pairs = []
    n_adj = min(n-1, max(1, P//2))
    if n_adj > 0:
        if n_adj == n-1:
            for i in range(n-1):
                pairs.append((i, i+1))
        else:
            idxs = np.linspace(0, n-2, n_adj).astype(int)
            seen = set()
            for i in idxs:
                key = (min(i, i+1), max(i, i+1))
                if key not in seen:
                    pairs.append(key); seen.add(key)
    n_fill = P - len(pairs)
    if n_fill > 0:
        used = set(pairs)
        tries = 0
        while len(pairs) < P and tries < n_fill * 10 + 50:
            i = int(rng.integers(0, n-1))
            j = int(rng.integers(i+1, n))
            key = (i, j)
            if key not in used:
                pairs.append(key); used.add(key)
            tries += 1
        # Fallback: enumerate
        ii = 0
        while len(pairs) < P and ii < n-1:
            jj = ii + 1
            while jj < n:
                key = (ii, jj)
                if key not in used:
                    pairs.append(key); used.add(key)
                    if len(pairs) >= P: break
                jj += 1
            ii += 1
    rng.shuffle(pairs)
    return pairs

def polish(h, score, q, n, rng, max_sweeps=1):
    for _ in range(max_sweeps):
        pairs = build_polish_pairs(n, rng)
        any_accept = False
        for (p, r) in pairs:
            res = pair_line_search(h, q, p, r, n)
            if res is not None:
                h, score, q = res
                any_accept = True
        if not any_accept:
            break
    return h, score, q

def anneal_trajectory(h0, score0, q0, n, rng, n_epochs, n_prop):
    h = h0.copy()
    score = score0
    q = q0.copy()
    scale = 0.12
    T0 = 3e-3
    for e in range(n_epochs):
        T = max(5e-5, T0 * (0.4 ** e))
        accepted = 0
        for _ in range(n_prop):
            p = int(rng.integers(0, n))
            r = int(rng.integers(0, n))
            if p == r:
                continue
            L = max(-h[p], h[r]-1.0)
            U = min(1.0-h[p], h[r])
            if U - L <= 1e-12:
                continue
            t = float(rng.uniform(-scale, scale))
            if t < L or t > U:
                continue
            h_new = h.copy()
            h_new[p] += t
            h_new[r] -= t
            if not (np.all(h_new >= 0) and np.all(h_new <= 1)):
                h_new = project_capped(h_new, n/2.0)
                if h_new is None:
                    continue
            if abs(np.sum(h_new) - n/2.0) > 1e-10:
                h_new = project_capped(h_new, n/2.0)
                if h_new is None:
                    continue
            sc = exact_score(h_new, n)
            if sc is None:
                continue
            delta = sc - score
            if delta <= 0 or (T > 0 and -delta/T < 700 and rng.random() < np.exp(-delta/T)):
                h = h_new
                score = sc
                q = get_lags(h, n)
                accepted += 1
        ratio = accepted / max(1, n_prop)
        if ratio < 0.12:
            scale = min(0.4, 1.25 * scale)
        elif ratio > 0.55:
            scale = max(0.01, 0.75 * scale)
        h, score, q = polish(h, score, q, n, rng, max_sweeps=1)
    return h, score, q
