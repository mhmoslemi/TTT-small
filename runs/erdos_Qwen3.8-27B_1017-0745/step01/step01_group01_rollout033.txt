def project_capped(z, n):
    s = n / 2.0
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(100):
        mid = lo + (hi - lo) / 2
        if mid <= lo or mid >= hi:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(3):
        if abs(r) < 1e-13 * max(1, n):
            break
        if r > 0:
            for i in range(n):
                if r <= 0: break
                add = min(r, 1.0 - h[i])
                h[i] += add
                r -= add
        else:
            for i in range(n):
                if r >= 0: break
                sub = min(-r, h[i])
                h[i] -= sub
                r += sub
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(np.sum(h) - s) > 5e-12 * max(1, n):
        return None
    return h

def exact_score(h, n):

def lag_raw(h, n):

def quad_roots(A, B, C, lo, hi):
    tol = 1e-14 * max(1.0, abs(A), abs(B), abs(C))
    roots = []
    pad = 1e-12 * max(1.0, abs(lo), abs(hi))

        if abs(B) <= tol:
            return roots
        r = -C / B

            roots.append(max(lo, min(hi, r)))
        return roots

    if disc < 0:

            disc = 0.0
        else:
            return roots
    if disc == 0.0:

            roots.append(max(lo, min(hi, r)))

def exhull(slopes, intercepts):
    # slopes and intercepts are arrays of same length, sorted by slope
    # Returns list of (slope, intercept, start) for upper hull
    if len(slopes) == 0:
        return []
    # For equal slopes, keep max intercept
    unique = []
    i = 0
    while i < len(slopes):
        j = i
        while j < len(slopes) and slopes[j] == slopes[i]:
            j += 1
        best_k = i
        for k in range(i, j):
            if intercepts[k] > intercepts[best_k]:
                best_k = k
        unique.append((slopes[best_k], intercepts[best_k]))
        i = j
    if len(unique) == 1:
        return [(unique[0][0], unique[0][1], -1e18)]
    hull = []
    hull.append((unique[0][0], unique[0][1], -1e18))
    for m, c0 in unique[1:]:
        if m == hull[-1][0]:
            if c0 > hull[-1][1]:
                hull[-1] = (m, c0, hull[-1][2])
            continue
        cross = (hull[-1][1] - c0) / (m - hull[-1][0])
        while len(hull) >= 1 and cross <= hull[-1][2]:
            hull.pop()
            if len(hull) == 0:
                break
            cross = (hull[-1][1] - c0) / (m - hull[-1][0])
        if len(hull) == 0:
            hull.append((m, c0, -1e18))
        else:
            hull.append((m, c0, cross))
    return hull

def build_edges(n):

            edges.add((i, i+r))
            i += 2*r
        r <<= 1
    if n > 2:

def project_capped(z, n):

    if z.shape != (n,) or not np.all(np.isfinite(z)):

    hi = float(np.max(z)) + 1.0
    for _ in range(100):
        mid = lo + (hi - lo) / 2.0
        if mid <= lo or mid >= hi:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
