    if x is None:

    cur_score = exact_score(x, n)
    if not np.isfinite(cur_score):
        return None
    best_local = x.copy()
    best_local_score = cur_score
    
    thetas = [0.05, 0.015, 0.004, 0.001, 0.0002, 0.00005]
