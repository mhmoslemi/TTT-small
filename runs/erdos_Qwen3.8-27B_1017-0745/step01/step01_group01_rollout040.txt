        v[n - 1] = -v[:n - 1].sum()
    else:
        bsize = min(n, [4, 8, 16][int(rng.integers(0, 3))])
        if bsize < 2:
            return None
        start = int(rng.integers(0, n - bsize + 1))
        v = np.zeros(n)
        seg = rng.standard_normal(bsize)
        seg[bsize - 1] = -seg[:bsize - 1].sum()
        v[start:start + bsize] = seg
    norm = np.linalg.norm(v)
    if not np.isfinite(norm) or norm < 1e-14:
        return None
    v /= norm
    v[n - 1] -= v.sum()
    return v

def chord_bounds(x, v, n):
    a, b = -1e30, 1e30
    for i in range(n):
        vi = v[i]
        if abs(vi) < 1e-15:
            continue
        if vi > 0.0:
            a = max(a, -x[i] / vi)
            b = min(b, (1.0 - x[i]) / vi)
        else:
            a = max(a, (1.0 - x[i]) / vi)
            b = min(b, -x[i] / vi)
    if not (np.isfinite(a) and np.isfinite(b)) or b - a < 1e-14:
        return None
    return a, b

def roundoff_repair(y, n):
    s = n / 2.0
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return None
    if np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    y = np.clip(y, 0.0, 1.0)
    r = s - float(np.sum(y))
    if abs(r) > 5e-11 * max(1, n):
        return None
    idx = range(n) if r > 0 else range(n - 1, -1, -1)
    for i in idx:
        if r > 0 and r <= 0.0:
            break
        if r < 0 and r >= 0.0:
            break
        d = min(r, 1.0 - y[i]) if r > 0 else max(r, -y[i])
        y[i] += d
        r -= d
    y = np.clip(y, 0.0, 1.0)
    return y if is_feasible(y, n) else None

def add_start(starts, v, n):
    h2 = project_capped(v, n) if v.shape == (n,) else None
    if h2 is None or not is_feasible(h2, n):
        return
    for ex in starts:
        if float(np.max(np.abs(h2 - ex))) < 1e-10:
            return
    starts.append(h2)

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= len(iv) <= 999:
            n = len(iv)
            if np.all(np.isfinite(iv)) and np.all(iv >= -1e-8) and np.all(iv <= 1.0 + 1e-8):
                parent = project_capped(iv, n)
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
    n = max(1, min(999, n))
    rng = np.random.default_rng(int(seed) % (2 ** 32))
    uniform = np.full(n, 0.5)
    fallback = parent.copy() if parent is not None else uniform.copy()

    starts = []
    add_start(starts, uniform, n)
    if parent is not None:
        add_start(starts, parent, n)
        add_start(starts, parent[::-1], n)
        add_start(starts, 1.0 - parent, n)
        add_start(starts, 1.0 - parent[::-1], n)
        for sd in (0.025, 0.08, 0.20):
            add_start(starts, parent + rng.normal(0, sd, n), n)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    add_start(starts, blk, n)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    add_start(starts, alt, n)
    for q in (1, 2):
        add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * q * np.arange(n) / n), n)
    add_start(starts, rng.uniform(0, 1, n), n)
    add_start(starts, rng.beta(0.5, 0.5, n), n)

    best, best_score = fallback.copy(), exact_score(fallback, n)
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_score - 1e-12:
            best, best_score = st.copy(), ss

    if n == 1 or n > 256 or steps_parsed(kwargs, n) < 1:
        final = project_capped(best, n)
        if final is None:
            final = fallback.copy()
        fs = exact_score(final, n)
        if not np.isfinite(fs) or fs > best_score:
            final, fs = best.copy(), best_score
        if not np.isfinite(fs):
            final, fs = uniform.copy(), exact_score(uniform, n)
        return final, float(fs), int(n)

    M = 5 if n <= 64 else (4 if n <= 128 else 3)
    steps = steps_parsed(kwargs, n)
    rmax = 2 if n <= 64 else 1
    restarts_max = min(int(kwargs.get('restarts', rmax)), rmax)

    T_hot = 0.002
    pilot_deltas = []
    for _ in range(24):
        v = gen_direction(rng, n)
        if v is None:
            continue
        cb = chord_bounds(best, v, n)
        if cb is None:
            continue
        a, b = cb
        t = rng.uniform(a, b)
        y = best + t * v
        yr = roundoff_repair(y, n)
        if yr is None:
            continue
        sy = exact_score(yr, n)
        if np.isfinite(sy) and sy > best_score:
            pilot_deltas.append(sy - best_score)
        elif np.isfinite(sy) and sy < best_score - 1e-12:
            best, best_score = yr.copy(), sy
    if pilot_deltas:
        d_med = float(np.median(pilot_deltas))
        T_hot = max(1e-5, min(0.02, d_med / (-math.log(0.8))))
    T_cold = max(1e-6, 0.02 * T_hot)
    temps = [T_cold * (T_hot / T_cold) ** (j / max(1, M - 1)) for j in range(M)]
    rho = [0.02 + 0.12 * j / max(1, M - 1) for j in range(M)]

    states, scores = [best.copy()], [best_score]
    for st in starts:
        if len(states) >= M:
            break
        dup = False
        for j in range(len(states)):
            if float(np.max(np.abs(st - states[j]))) < 1e-10:
                dup = True
                break
        if not dup:
            states.append(st.copy())
            scores.append(exact_score(st, n))
    for _ in range(M - len(states)):
        sd = [0.025, 0.08, 0.20][len(states) % 3]
        h2 = project_capped(best + rng.normal(0, sd, n), n)
        if h2 is not None:
            states.append(h2)
            scores.append(exact_score(h2, n))
        elif states:
            states.append(states[0].copy())
            scores.append(scores[0])

    patience_limit = max(50, steps // 5)
    no_improve = 0
    restart_count = 0
    for sweep in range(steps):
        improved = False
        for j in range(M):
            v = gen_direction(rng, n)
            if v is None:
                continue
            cb = chord_bounds(states[j], v, n)
            if cb is None:
                continue
            a, b = cb
            if rng.random() < 0.20:
                t = rng.uniform(a, b)
            else:
                t = rng.uniform(-rho[j] * (b - a), rho[j] * (b - a))
                if t < a - 1e-14 or t > b + 1e-14:
                    continue
            y = states[j] + t * v
            if float(np.max(np.abs(y - states[j]))) <= 1e-14:
                continue
            yr = roundoff_repair(y, n)
            if yr is None:
                continue
            sy = exact_score(yr, n)
            if not np.isfinite(sy):
                continue
            delta = sy - scores[j]
            if delta <= 0.0:
                states[j], scores[j] = yr, sy
            else:
                thr = delta / temps[j]
                if thr > 745.0:
                    continue
                if math.log(rng.random()) < -thr:
                    states[j], scores[j] = yr, sy
            if sy < best_score - 1e-12:
                best, best_score = yr.copy(), sy
                improved = True
        if improved:
            no_improve = 0
        else:
            no_improve += 1
        parity = sweep % 2
        for j in range(parity, M - 1, 2):
            lr = (scores[j] - scores[j + 1]) * (1.0 / temps[j] - 1.0 / temps[j + 1])
            if lr >= 0.0 or (math.log(rng.random()) < lr):
                states[j], states[j + 1] = states[j + 1], states[j]
                scores[j], scores[j + 1] = scores[j + 1], scores[j]
        if no_improve >= patience_limit:
            if restart_count < restarts_max:
                for j in range(max(1, M // 2), M):
                    base = best if j % 2 == 0 else (starts[j % len(starts)] if starts else best)
                    sd = [0.03, 0.08, 0.18][j % 3]
                    h2 = project_capped(base + rng.normal(0, sd, n), n)
                    if h2 is None:
                        h2 = base.copy()
                    states[j] = h2
                    scores[j] = exact_score(h2, n)
                    if scores[j] < best_score - 1e-12:
                        best, best_score = h2.copy(), scores[j]
                restart_count += 1
                no_improve = 0
            else:
                break

    final = project_capped(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > best_score + 1e-12:
        final, fs = best.copy(), best_score
    if not np.isfinite(fs):
        final, fs = uniform.copy(), exact_score(uniform, n)
    return final, float(fs), int(n)

def steps_parsed(kwargs, n):
    default = 800 if n <= 64 else (400 if n <= 128 else 120)
    try:
        v = int(kwargs.get('steps', default))
    except Exception:
        v = default
    cap = 800 if n <= 64 else (800 if n <= 128 else 200)
    return max(0, min(v, cap))
