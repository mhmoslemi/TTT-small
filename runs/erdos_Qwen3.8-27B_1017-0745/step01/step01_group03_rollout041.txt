def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0 or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-14:
        if resid > 0:
            for i in range(n):
                add = min(resid, 1.0 - y[i])
                y[i] += add
                resid -= add
                if abs(resid) < 1e-15:
                    break
        else:
            for i in range(n):
                sub = min(-resid, y[i])
                y[i] -= sub
                resid += sub
                if abs(resid) < 1e-15:

    y = np.clip(y, 0.0, 1.0)

def project_box_sum(v, S):

    n = len(v)
    if n == 0 or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):

        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)

                y[i] += add
                resid -= add
                if abs(resid) < 1e-15:

def pair_delta_lags(h, p, q, a, n):
    n_lags = 2 * n - 1
    ks = np.arange(-(n - 1), n, dtype=np.int64)
    s1 = np.zeros(n_lags)
    m1 = (ks <= p) & (ks >= p - n + 1)
    if np.any(m1):
        s1[m1] += a * (1.0 - h[p - ks[m1]])
    m2 = (ks <= q) & (ks >= q - n + 1)
    if np.any(m2):
        s1[m2] -= a * (1.0 - h[q - ks[m2]])
    s2 = np.zeros(n_lags)
    m3 = (ks >= -p) & (ks <= n - 1 - p)
    if np.any(m3):
        s2[m3] -= a * h[p + ks[m3]]
    m4 = (ks >= -q) & (ks <= n - 1 - q)
    if np.any(m4):
        s2[m4] += a * h[q + ks[m4]]
    s3 = np.zeros(n_lags)
    s3[n - 1] = -2.0 * a * a
    if p != q:
        s3[p - q + n - 1] += a * a
        s3[q - p + n - 1] += a * a
    D = s1 + s2 + s3
    return D if np.all(np.isfinite(D)) else None
