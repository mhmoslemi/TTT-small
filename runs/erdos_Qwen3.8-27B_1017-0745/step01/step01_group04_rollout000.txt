    corr = np.correlate(h, 1.0 - h, mode='full')
    c = dx * corr  # shape (2N-1,)

    sum_e = np.sum(e)

    if len(distinct_starts) > 8:
        distinct_starts = distinct_starts[:8]

    if Nc <= 128:
        tau_sched = [0.02, 0.006, 0.002, 0.0006, 0.0002, 0.00006, 0.000015]
        iters = 60
    elif Nc <= 256:
        tau_sched = [0.02, 0.005, 0.0015, 0.0004, 0.0001]
        iters = 40
    elif Nc <= 512:
        tau_sched = [0.015, 0.004, 0.001, 0.0003]
        iters = 25
    else:
        tau_sched = [0.015, 0.004, 0.001]
        iters = 12

    for h0 in distinct_starts:
        s0 = exact_score(h0, Nc)
        if np.isfinite(s0) and s0 < best_s:
            best_s = s0
            best_h = h0.copy()
        if Nc > 999:
            continue
        h_opt, s_opt = optimize_start(h0, Nc, m, tau_sched, iters)
        if np.isfinite(s_opt) and s_opt < best_s:
            best_s = s_opt
            best_h = h_opt.copy()

    # Basin hops
    if Nc <= 128:
        hop_scales = [0.05, 0.02, 0.008]
    elif Nc <= 256:
        hop_scales = [0.04, 0.015]
    else:
        hop_scales = [0.03]
    for sc in hop_scales:
        hop = project_box_sum(best_h + sc * rng.standard_normal(Nc), m)
        s_hop = exact_score(hop, Nc)
        if np.isfinite(s_hop) and s_hop < best_s:
            best_s = s_hop
            best_h = hop.copy()
        if Nc <= 512:
            h_opt, s_opt = optimize_start(hop, Nc, m, tau_sched, iters)
            if np.isfinite(s_opt) and s_opt < best_s:
                best_s = s_opt
                best_h = h_opt.copy()

    # Final validation
    best_h = project_box_sum(best_h, m)
    final_s = exact_score(best_h, Nc)
    if not np.isfinite(final_s):
        best_h = base.copy()
        final_s = base_s
    if final_s > base_s:
        best_h = base.copy()
        final_s = base_s
    return best_h, float(final_s), int(Nc)
