import numpy as np
import math

def project_capped(y, n):
    S = n / 2.0
    y = np.asarray(y, dtype=np.float64)
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return np.full(n, 0.5), False
    lo = float(y.min()) - 1.0
    hi = float(y.max())
    lam = 0.0
    for _ in range(80):
        lam = 0.5 * (lo + hi)
        c = np.clip(y - lam, 0.0, 1.0)
        r = c.sum() - S
        if abs(r) < 1e-13:
            break
        if r > 0:
            lo = lam
        else:
            hi = lam
    c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(5):
        r = S - c.sum()
        if abs(r) < 1e-13:
            break
        if r > 0:
            mask = c < 1.0 - 1e-12
        else:
            mask = c > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        c[mask] += r / cnt
        np.clip(c, 0.0, 1.0, out=c)
    if not (np.all(c >= -1e-12) and np.all(c <= 1.0 + 1e-12) and abs(c.sum() - S) <= 1e-10):
        return np.full(n, 0.5), False
    np.clip(c, 0.0, 1.0, out=c)
    return c, True

def exact_score(h, n):
    if not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / n
    cs = np.correlate(h, 1.0 - h, mode='full')
    if cs.shape[0] != 2 * n - 1:
        return float('inf')
    return float(np.max(cs) * dx)

def lag_scores(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def build_gradient_matrix(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        row = k + n - 1
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if hi1 > lo1:
            G[row, lo1:hi1] += dx * (1.0 - h[lo1 - k:hi1 - k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if hi2 > lo2:
            G[row, lo2:hi2] -= dx * h[lo2 + k:hi2 + k]
    if not np.all(np.isfinite(G)):
        return None
    return G

def project_simplex(z):
    n = len(z)
    if n <= 0 or not np.all(np.isfinite(z)):
        return np.full(n, 1.0 / max(1, n))
    sz = np.sort(z)[::-1]
    cs = np.cumsum(sz)
    rho = np.arange(1, n + 1)
    cond = sz - (cs - 1.0) / rho > 0
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.full(n, 1.0 / n)
    r = idx[-1]
    theta = (cs[r] - 1.0) / (r + 1)
    p = np.maximum(z - theta, 0.0)
    s = p.sum()
    if s > 1e-15:
        p *= 1.0 / s
    else:
        p = np.full(n, 1.0 / n)
    return p

def dual_state(p, h, G, q, L, n):
    gmix = G.T @ p
    y, ok = project_capped(h - gmix / L, n)
    if not ok:
        return None
    d = y - h
    u = q + G @ d
    if not np.all(np.isfinite(u)):
        return None
    D = float(p @ u + 0.5 * L * d @ d)
    return (d, u, D)

def solve_majorant(h, q, G, n, max_inner):
    L = 4.0 / n
    m = 2 * n - 1
    maxq = float(np.max(q))
    tol_q = max(1e-12, 1e-8 * max(1.0, abs(maxq)))
    sel = q >= maxq - tol_q
    p = np.zeros(m)
    p[sel] = 1.0 / sel.sum()
    Ldual = max(1e-12, 1.05 * float(np.sum(G * G)) / L)
    st = dual_state(p, h, G, q, L, n)
    if st is None:
        return None
    best_d, best_u, best_D = st
    x = p.copy()
    yv = p.copy()
    theta = 1.0
    stall = 0
    for _ in range(max_inner):
        st_y = dual_state(yv, h, G, q, L, n)
        if st_y is None:
            break
        _, u_y, _ = st_y
        grad = q + G @ (st_y[0])
        x_new = project_simplex(yv + grad / Ldual)
        st_x = dual_state(x_new, h, G, q, L, n)
        if st_x is None:
            break
        _, _, D_new = st_x
        if D_new < best_D - 1e-13:
            st_r = dual_state(project_simplex(x + grad / Ldual), h, G, q, L, n)
            if st_r is not None:
                _, D_r = st_r[2], st_r[1]
                if D_r >= best_D - 1e-13:
                    x_new = project_simplex(x + grad / Ldual)
                    st_x = st_r
                    theta = 1.0
                    yv = x_new
                    D_new = st_r[2]
                else:
                    break
            else:
                break
        else:
            theta_new = (1.0 + math.sqrt(1.0 + 4.0 * theta * theta)) / 2.0
            yv = x_new + ((theta - 1.0) / theta_new) * (x_new - x)
            theta = theta_new
            if np.max(np.abs(x_new - x)) < 1e-10:
                stall += 1
            else:
                stall = 0
            if stall >= 3:
                break
        if D_new >= best_D - 1e-13:
            best_D = max(best_D, D_new)
            best_d = st_x[0].copy()
            best_u = st_x[1].copy()
        gap = float(np.max(best_u) - p @ best_u)
        if gap <= max(1e-10, 1e-8 * max(1.0, abs(float(np.max(best_u))))):
            break
    if best_d is None or not np.all(np.isfinite(best_d)):
        return None
    return best_d

def optimize_start(h0, n, max_outer, max_inner):
    h, ok = project_capped(h0.copy(), n)
    if not ok:
        return h0.copy(), exact_score(h0, n)
    best_h = h.copy()
    best_s = exact_score(h, n)
    for _ in range(max_outer):
        q = lag_scores(h, n)
        G = build_gradient_matrix(h, n)
        if G is None:
            break
        d = solve_majorant(h, q, G, n, max_inner)
        if d is None or np.max(np.abs(d)) <= 1e-12:
            break
        v = G @ d
        L = 4.0 / n
        nrm2 = float(d @ d)
        accepted = False
        prev_dec = 1e6
        for it in range(24):
            alpha = 0.5 ** it
            model_val = float(np.max(q + alpha * v)) + 0.5 * L * alpha * alpha * nrm2
            if model_val >= float(np.max(q)) - 1e-14:
                continue
            h_trial = h + alpha * d
            h_trial, ok2 = project_capped(h_trial, n)
            if not ok2:
                continue
            s_trial = exact_score(h_trial, n)
            if s_trial < best_s - 1e-12:
                h = h_trial
                best_h = h_trial.copy()
                best_s = s_trial
                accepted = True
                dec = prev_dec if it > 0 else best_s
                if dec < 1e-11:
                    break
                break
        if not accepted:
            break
        if it >= 1 and dec < 1e-11:
            break
    return best_h, best_s
