def lag_data(h, N):

            G[r, :ell] -= h[d:]
    return q, G

    J[:, N] = 1.0
    return J

        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)

def exact_score(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    c = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(c)):
        return np.inf
    return float(np.max(c) * 2.0 / N)

def lag_data(h, N):
    L = 2 * N - 1
    q = np.zeros(L)
    G = np.zeros((L, N))
    for s in range(-(N - 1), N):
        r = s + N - 1
        if s >= 0:
            ell = N - s
            q[r] = np.dot(h[:ell], 1.0 - h[s:s + ell])
            G[r, :ell] += 1.0 - h[s:s + ell]
            G[r, s:N] -= h[:ell]
        else:
            d = -s; ell = N - d
            q[r] = np.dot(h[d:], 1.0 - h[:ell])
            G[r, d:] += 1.0 - h[:ell]
            G[r, :ell] -= h[d:]
    return q, G

def _mm_obj(z, x, N):
    y = z[:N]
    return np.sum((y - x) ** 2) + z[N]

def _mm_obj_jac(z, x, N):
    g = np.zeros(N + 1)
    g[:N] = 2.0 * (z[:N] - x)
    g[N] = 1.0
    return g

def _mm_ineq(z, x, q, G, N):
    return z[N] - q - G @ (z[:N] - x)

def _mm_ineq_jac(z, x, q, G, N):
    J = np.zeros((G.shape[0], N + 1))
    J[:, :N] = -G
    J[:, N] = 1.0
    return J

def _mm_eq(z, x, m):
    return np.array([z[:N].sum() - m])

def _mm_eq_jac(z, x, m):
    J = np.zeros((1, N + 1))
    J[0, :N] = 1.0
    return J

def mm_step(x, N, m):
    q, G = lag_data(x, N)
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return x.copy()
    z0 = np.concatenate([x, [max(np.max(q), 0.0)]])
    bounds = [(0.0, 1.0)] * N + [(-1e6, 1e6)]
    cons = [
        {'type': 'eq', 'fun': _mm_eq, 'jac': _mm_eq_jac, 'args': (x, m)},
        {'type': 'ineq', 'fun': _mm_ineq, 'jac': _mm_ineq_jac, 'args': (x, q, G, N)},
    ]
    try:
        res = minimize(_mm_obj, z0, jac=_mm_obj_jac, args=(x, N),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 150, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return x.copy()
    if res.x is None or len(res.x) != N + 1 or not np.all(np.isfinite(res.x[:N])):
        return x.copy()
    return project_box_sum(res.x[:N], N, m)

def mm_run(x, N, m, max_iter):
    best = x.copy()
    best_sc = exact_score(best, N)
    cur_sc = best_sc
    for it in range(max_iter):
        y = mm_step(x, N, m)
        accepted = False
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
            cand = project_box_sum((1.0 - alpha) * x + alpha * y, N, m)
            sc = exact_score(cand, N)
            if np.isfinite(sc) and sc < cur_sc - 1e-13:
                x = cand; cur_sc = sc
                if sc < best_sc:
                    best = cand.copy(); best_sc = sc
                accepted = True
                break
        if not accepted:
            break
        if np.max(np.abs(x - best)) < 1e-10:
            break
    return best

def make_starts(baseline, N, m, rng):
    starts = [baseline.copy()]
    starts.append(project_box_sum(1.0 - baseline, N, m))
    starts.append(project_box_sum(baseline[::-1], N, m))
    starts.append(project_box_sum(np.roll(baseline, max(1, N // 3)), N, m))
    for sc in [0.05, 0.15]:
        starts.append(project_box_sum(baseline + sc * rng.standard_normal(N), N, m))
    starts.append(project_box_sum(rng.uniform(0.0, 1.0, N), N, m))
    idx = np.arange(N, dtype=np.float64)
    for A in [0.12, 0.24]:
        z = np.full(N, 0.5)
        for r in range(1, min(6, (N - 1) // 2) + 1):
            a = rng.standard_normal(); b = rng.standard_normal()
            z += (A / r) * (a * np.sin(2.0 * np.pi * r * idx / N) + b * np.cos(2.0 * np.pi * r * idx / N))
        starts.append(project_box_sum(z, N, m))
    kept = [starts[0]]
    for s in starts[1:]:
        if np.all(np.isfinite(s)) and np.max(np.abs(s - np.array(kept).mean(axis=0))) > 1e-10:
            dup = False
            for k in kept:
                if np.max(np.abs(s - k)) <= 1e-10:
                    dup = True; break
            if not dup:
                kept.append(s)
    return kept

def run(seed=42, **kwargs):
    N = None; m = None; raw = None
    r = globals().get('initial_h_values', None)
    if r is not None:
        try:
            raw = np.asarray(r, dtype=np.float64)
            if raw.ndim == 1 and raw.size > 1 and np.all(np.isfinite(raw)):
                N = raw.size; m = N / 2.0
            else:
                raw = None
        except Exception:
            raw = None
    if N is None:
        N = kwargs.get('n_points', 70)
        if not isinstance(N, (int, np.integer)) or isinstance(N, bool) or N < 2 or N > 999:
            N = 70
        m = N / 2.0
    if N <= 1:
        h = np.array([0.5]); return h, 0.0, 1
    if raw is not None:
        baseline = project_box_sum(raw, N, m) if (raw.min() < -1e-12 or raw.max() > 1 + 1e-12 or abs(raw.sum() - m) > 1e-6) else np.clip(raw.copy(), 0.0, 1.0)
    else:
        baseline = np.full(N, 0.5)
    baseline = project_box_sum(baseline, N, m)
    base_sc = exact_score(baseline, N)
    best = baseline.copy(); best_sc = base_sc
    if N > 256:
        return baseline, float(base_sc), int(N)
    rng = np.random.default_rng(seed)
    if N <= 96:
        nstarts, mm_I, restarts = 8, 40, [(0.05, 24), (0.015, 24)]
    elif N <= 160:
        nstarts, mm_I, restarts = 5, 25, [(0.04, 16)]
    else:
        nstarts, mm_I, restarts = 3, 12, []
    starts = make_starts(baseline, N, m, rng)[:nstarts]
    for h0 in starts:
        sc0 = exact_score(h0, N)
        if np.isfinite(sc0) and sc0 < best_sc - 1e-13:
            best = h0.copy(); best_sc = sc0
        h_o = mm_run(h0, N, m, mm_I)
        sc_o = exact_score(h_o, N)
        if np.isfinite(sc_o) and sc_o < best_sc - 1e-13:
            best = h_o.copy(); best_sc = sc_o
    for sigma, cap in restarts:
        hop = project_box_sum(best + sigma * rng.standard_normal(N), N, m)
        sc_h = exact_score(hop, N)
        if np.isfinite(sc_h) and sc_h < best_sc - 1e-13:
            best = hop.copy(); best_sc = sc_h
        h_o = mm_run(hop, N, m, cap)
        sc_o = exact_score(h_o, N)
        if np.isfinite(sc_o) and sc_o < best_sc - 1e-13:
            best = h_o.copy(); best_sc = sc_o
    best = project_box_sum(best, N, m)
    c5 = exact_score(best, N)
    if not np.isfinite(c5) or c5 >= base_sc:
        best = baseline; c5 = base_sc
    c5 = exact_score(best, N)
    if not np.isfinite(c5):
        best = np.full(N, 0.5); c5 = exact_score(best, N)
    return best, float(c5), int(N)
