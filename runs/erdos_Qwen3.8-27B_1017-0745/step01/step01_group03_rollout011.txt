def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-12 * max(1.0, float(n)):

    if resid > 0:

        for i in order:
            add = min(resid, 1.0 - y[i])

def basin_hop(walker, n, S, rng, global_best, global_score):
    hop_amps = [0.025, 0.06, 0.12, 0.20]
    T_base = 0.002
    
    for r, amp in enumerate(hop_amps):
        T = T_base * (0.65 ** r)
        # Generate noise
        if r % 2 == 0:
            noise = rng.standard_normal(n)
        else:
            # Harmonic noise
            noise = np.zeros(n)
            for _ in range(2):
                m = rng.integers(1, n // 2 + 1)
                ph = rng.uniform(0, 2 * np.pi)
                noise += np.cos(2 * np.pi * m * np.arange(n) / n + ph)
            noise -= np.mean(noise)
            rms = np.sqrt(np.mean(noise**2))
            if rms > 1e-10:
                noise /= rms
            else:
                noise = rng.standard_normal(n)
        
        proposal = walker + amp * noise
        p = project_box_sum(proposal, S)
        if p is None or not check_feasible(p, n):
            continue
        
        result = local_optimize(p, n, S, rng)
        if result is None:
            continue
        
        sc = exact_score(result, n)
        if sc < global_score - 1e-12:
            global_best = result.copy()
            global_score = sc
        
        walker_score = exact_score(walker, n)
        delta = sc - walker_score
        if delta <= 0:
            walker = result.copy()
        else:
            prob = np.exp(max(-700.0, -delta / T))
            if rng.random() < prob:
                walker = result.copy()
    
    return walker, global_best, global_score

import numpy as np
from scipy.signal import fftconvolve

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-10 * max(1.0, float(n)):
        return None
    if resid > 1e-15:
        order = np.argsort(1.0 - y)
        for i in order:
            add = min(resid, 1.0 - y[i])
            y[i] += add
            resid -= add
            if abs(resid) < 1e-15:
                break
    elif resid < -1e-15:
        order = np.argsort(y)
        for i in order:
            sub = min(-resid, y[i])
            y[i] -= sub
            resid += sub
            if abs(resid) < 1e-15:
                break
    if not np.all(np.isfinite(y)):
        return None
    if np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    return np.clip(y, 0.0, 1.0)

def check_feasible(h, n):
    if h is None or h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10:
        return False
    return True

def exact_score(h, n):
    if not check_feasible(h, n):
        return np.inf
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(c) * dx)
    return val if np.isfinite(val) else np.inf

    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_value(Q, tau):

    if not np.isfinite(m):
        return np.inf
    vals = np.exp((Q - m) / tau)
    s = float(np.sum(vals))
    if s <= 0 or not np.isfinite(s):
        return np.inf
    return m + tau * np.log(s)

    vals = np.exp((Q - m) / tau)
    s = float(np.sum(vals))

        return np.full(n, np.nan)

    conv1 = fftconvolve(1.0 - h, w_arr, mode='full')
    conv2 = fftconvolve(h, w_arr[::-1], mode='full')

    D = conv2[n - 1 : 2 * n - 1]
    grad = A - D
    if not np.all(np.isfinite(grad)):
        return np.full(n, np.nan)
    return grad
