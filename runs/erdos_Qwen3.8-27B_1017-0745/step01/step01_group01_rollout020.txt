        pop.append(best.copy()); psc.append(best_score)
    NP = len(pop)

    # DE evolution
    patience = max(12, min(40, generations // 4))
    stagn = 0; rstarts = 0; prev_b = best_score
    for gen in range(generations):
        opop = [p.copy() for p in pop]
        osc = list(psc)
        bidx = int(np.argmin(osc))
        for i in range(NP):
            F = rng.uniform(0.45, 0.95)
            CR = rng.uniform(0.75, 0.95)
            cands = [j for j in range(NP) if j != i]
            if rng.random() < 0.8:
                r1, r2, r3 = rng.choice(cands, size=3, replace=False)
                v = opop[r1] + F * (opop[r2] - opop[r3])
            else:
                r1, r2 = rng.choice(cands, size=2, replace=False)
                v = opop[i] + F * (opop[bidx] - opop[i] + opop[r1] - opop[r2])
            mask = rng.random(n) < CR
            mask[int(rng.integers(0, n))] = True
            trial = np.where(mask, v, opop[i])
            hp = project_capped(trial, n)
            if hp is None: continue
            if float(np.max(np.abs(hp - pop[i]))) <= 1e-12: continue
            sc = exact_score(hp, n)
            if np.isfinite(sc) and sc < osc[i] - 1e-12:
                pop[i] = hp; psc[i] = sc
                if sc < best_score - 1e-12:
                    best, best_score = hp.copy(), sc
        if best_score < prev_b - 1e-12:
            stagn = 0; prev_b = best_score
        else:
            stagn += 1
        if stagn >= patience and rstarts < max_restarts:
            rstarts += 1; stagn = 0
            el = max(2, math.ceil(0.2 * NP))
            order = np.argsort(psc)
            npop, nsc = [best.copy()], [best_score]
            for oi in order:
                if len(npop) >= el: break
                c = pop[oi]
                if any(float(np.max(np.abs(c - e))) < 1e-10 for e in npop): continue
                npop.append(c.copy()); nsc.append(psc[oi])
            sbase = [0.03, 0.07, 0.15][min(rstarts - 1, 2)]
            for j in range(len(npop), NP):
                b = npop[j % max(1, len(npop))]
                hp = project_capped(b + rng.normal(0, sbase * rng.uniform(0.5, 1.5), n), n)
                if hp is None: hp = project_capped(rng.uniform(0, 1, n), n)
                if hp is None: hp = uniform.copy()
                npop.append(hp); nsc.append(exact_score(hp, n))
                if nsc[-1] < best_score - 1e-12:
                    best, best_score = hp.copy(), nsc[-1]
            pop = npop[:NP]; psc = nsc[:NP]
            prev_b = best_score

    # Final selection
    cands = [(best, best_score), (uniform, exact_score(uniform, n))]
    if parent_fb is not None and np.isfinite(parent_sc):
        cands.append((parent_fb, parent_sc))
    bf, bfs = None, float('inf')
    for c, cs in cands:
        if np.isfinite(cs) and cs < bfs:
            bf, bfs = c, cs
    if bf is None: bf, bfs = uniform, exact_score(uniform, n)
    fin = project_capped(bf.copy(), n)
    if fin is not None:
        fs = exact_score(fin, n)
        if np.isfinite(fs) and fs <= bfs + 1e-12:
            bf, bfs = fin, fs
    else:
        bf, bfs = uniform, exact_score(uniform, n)
    return bf.copy(), float(bfs), int(n)
