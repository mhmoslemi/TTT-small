Let me analyze the problem and strategy carefully.

The task is to find a step function h: [0, 2] → [0, 1] that minimizes:
C₅ = max_k ∫ h(x)(1 - h(x+k)) dx

With constraints:
- 0 ≤ h[i] ≤ 1
- sum(h) * dx = 1, i.e., sum(h) = n/2

The discretization uses np.correlate(h, 1-h, mode="full") * dx where dx = 2.0/n_points.

The previous program achieved C₅ = 0.381021. Current record is C₅ ≤ 0.38085. Target is 0.3808.

The strategy calls for a Sequential Convex Majorization-Minimization (MM) approach:
1. At each feasible point x, build a quadratic upper bound for each lag overlap
2. Minimize the maximum of those tangent planes plus a common quadratic term
3. Use SLSQP for the convex subproblem
4. Accept only if exact score improves

Key mathematical facts:
- Each lag score q_s(h) = dx * sum_{i in I_s} h[i](1-h[i+s])
- The Hessian of q_s has spectral norm at most 2*dx = 4/n
- So q_s(y) ≤ q_s(x) + ∇q_s(x)·(y-x) + (L/2)||y-x||² where L = 2*dx

_mm = {}

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).flatten()
    n = len(v)
    if S < 0 or S > n or not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, 0.5, 1.0, 0.0)

        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - float(np.sum(y))
    if abs(r) > 1e-13:

        for i in idx:
            if r > 0:

            else:

        return None
    y = np.clip(y, 0.0, 1.0)

    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-11) and np.all(h <= 1.0 + 1e-11) and
            abs(float(np.sum(h)) - 0.5 * n) < 1e-9)

def score(h):
    n = len(h)
    if not feas(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    if len(c) != 2 * n - 1:
        return float('inf')
    return float(np.max(c)) * (2.0 / n)

def lag_qg(h, s, n):
    dx = 2.0 / n
    lo_i, hi_i = max(0, -s), min(n, n - s)
    q = 0.0
    g = np.zeros(n)
    for i in range(lo_i, hi_i):
        j = i + s
        q += h[i] * (1.0 - h[j])
        g[i] += dx * (1.0 - h[j])
        g[j] -= dx * h[i]
    return q * dx, g

    dx = 2.0 / n
    best, best_s = -1e30, 0
    for s in range(-(n - 1), n):
        lo_i, hi_i = max(0, -s), min(n, n - s)
        v = 0.0
        for i in range(lo_i, hi_i):
            j = i + s
            v += x[i] * (1.0 - x[j]) + (1.0 - x[j]) * d[i] - x[i] * d[j]
        v *= dx
        if v > best:
            best, best_s = v, s
    return best, best_s

    n, x, L = _mm['n'], _mm['x'], _mm['L']
    d = z[:n] - x
    return 0.5 * L * float(np.dot(d, d)) + z[n]

    n, x, L = _mm['n'], _mm['x'], _mm['L']
    g = np.empty(n + 1)
    g[:n] = L * (z[:n] - x)
    g[n] = 1.0
    return g

def _ineq(z):
    n, qJ, GJ, x = _mm['n'], _mm['qJ'], _mm['GJ'], _mm['x']
    d = z[:n] - x

    m = len(_mm['qJ'])
    J = np.zeros((m, n + 1))
    J[:, :n] = -GJ
    J[:, n] = 1.0
    return J

def _eq(z):
    return float(np.sum(z[:_mm['n']])) - 0.5 * _mm['n']

def _ejac(z):
    g = np.zeros(_mm['n'] + 1)
    g[:_mm['n']] = 1.0
    return g

def mm_solve(x, n):

    _mm['n'], _mm['x'], _mm['L'] = n, x, L
    all_q, all_g = [], []

        all_g.append(gv)

    all_g = np.array(all_g)

    y_p = x.copy()

        _mm['qJ'], _mm['GJ'] = qJ, GJ
        t0 = float(np.max(qJ))
        z0 = np.append(y_p, t0)
        bnds = [(0.0, 1.0)] * n + [(-1e6, 1e6)]

            break

            break

            break

        if pos not in act and len(act) < max_act:
            act.append(pos)
        y_p = y_new
    return y_p

def mm_descend(x, n, M):
    best = x.copy()
    bs = score(best)
    for _ in range(M):
        y = mm_solve(best, n)
        if y is None:
            y = best.copy()
        if np.max(np.abs(y - best)) < 1e-12:
            break
        # Line search
        improved = False
        for k in range(8):
            alpha = 2.0 ** (-k)
            z = best + alpha * (y - best)
            zp = project_box_sum(z, 0.5 * n)
            if zp is None:
                continue
            cs = score(zp)
            if cs < bs - 1e-13:
                best = zp
                bs = cs
                improved = True
                break
        if not improved:
            break
        if bs < 1e-6:
            break
    return best, bs

import numpy as np
from scipy.optimize import minimize

_mm = {}

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).flatten()
    n = len(v)
    if S < 0 or S > n:
        S = max(0.0, min(float(n), S))
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, 0.5, 1.0, 0.0)
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - float(np.sum(y))
    if abs(r) > 1e-13:
        idx = np.argsort(1.0 - y) if r > 0 else np.argsort(y)
        for i in idx:
            if r > 0:
                d = min(r, 1.0 - y[i]); y[i] += d; r -= d
            else:
                d = min(-r, y[i]); y[i] -= d; r += d
            if abs(r) < 1e-14:
                break
    y = np.clip(y, 0.0, 1.0)
    if abs(float(np.sum(y)) - S) > 1e-8 * max(1, n):
        return None
    return y

def feas(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-11) and np.all(h <= 1.0 + 1e-11) and
            abs(float(np.sum(h)) - 0.5 * n) < 1e-9)

def score(h):
    n = len(h)
    if not feas(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    if len(c) != 2 * n - 1 or not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c)) * (2.0 / n)

def lag_qg(h, s, n):
    dx = 2.0 / n
    lo_i, hi_i = max(0, -s), min(n, n - s)
    q, g = 0.0, np.zeros(n)
    for i in range(lo_i, hi_i):
        j = i + s
        q += h[i] * (1.0 - h[j])
        g[i] += dx * (1.0 - h[j])
        g[j] -= dx * h[i]
    return q * dx, g

def sep_max(x, d, n):
    dx = 2.0 / n
    best, best_s = -1e30, 0
    for s in range(-(n - 1), n):
        lo_i, hi_i = max(0, -s), min(n, n - s)
        v = 0.0
        for i in range(lo_i, hi_i):
            j = i + s
            v += x[i] * (1.0 - x[j]) + (1.0 - x[j]) * d[i] - x[i] * d[j]
        v *= dx
        if v > best:
            best, best_s = v, s
    return best, best_s

def _obj(z):
    n, x, L = _mm['n'], _mm['x'], _mm['L']
    d = z[:n] - x
    return 0.5 * L * float(np.dot(d, d)) + z[n]

def _og(z):
    n, x, L = _mm['n'], _mm['x'], _mm['L']
    g = np.empty(n + 1)
    g[:n] = L * (z[:n] - x)
    g[n] = 1.0
    return g

def _ineq(z):
    n, qJ, GJ, x = _mm['n'], _mm['qJ'], _mm['GJ'], _mm['x']
    d = z[:n] - x
    return z[n] - qJ - GJ @ d

def _ijac(z):
    n, GJ = _mm['n'], _mm['GJ']
    m = len(_mm['qJ'])
    J = np.zeros((m, n + 1))
    J[:, :n] = -GJ
    J[:, n] = 1.0
    return J

def _eq(z):
    n = _mm['n']
    return float(np.sum(z[:n])) - 0.5 * n

def _ejac(z):
    g = np.zeros(_mm['n'] + 1)
    g[:_mm['n']] = 1.0
    return g

def mm_solve(x, n):
    dx, L = 2.0 / n, 4.0 / n
    _mm['n'], _mm['x'], _mm['L'] = n, x, L
    all_lags = list(range(-(n - 1), n))
    all_q, all_g = [], []
    for s in all_lags:
        qv, gv = lag_qg(x, s, n)
        all_q.append(qv)
        all_g.append(gv)
    all_q = np.array(all_q)
    all_g = np.array(all_g)
    if n <= 64:

    max_act = min(2 * n - 1, 100)
    y_p = x.copy()

        z0 = np.append(y_p, t0)

            break

            cs = score(zp)
            if cs < bs - 1e-13:
                best, bs = zp, cs
                improved = True
                break

    return best, bs
