def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    if r > 0:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
            if r <= 1e-14:

    elif r < 0:

            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
            if r >= -1e-14:

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / N)

    c = dx * corr

    e = np.exp((c - cmax) / tau)
    se = e.sum()
    F = cmax + tau * np.log(se)
    p = e / se
    grad = np.zeros(N)

import numpy as np

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    if r > 0:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
            if r <= 1e-14:
                break
    elif r < 0:
        for i in range(N):
            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1 + 1e-10:
        return np.full(N, 0.5)
    if abs(h.sum() - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / N)

def smooth_obj_grad(h, tau, dx):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = dx * corr
    cmax = c.max()
    e = np.exp((c - cmax) / tau)
    se = e.sum()
    F = cmax + tau * np.log(se)
    p = e / se
    grad = np.zeros(N)
    for r in range(2 * N - 1):
        s = r - (N - 1)
        pw = dx * p[r]
        if pw <= 1e-16:
            continue
        if s == 0:
            grad += pw * (1.0 - 2.0 * h)
        elif s > 0:
            d = s
            L = N - d
            grad[d:] += pw * (1.0 - h[:L])
            grad[:L] -= pw * h[d:]
        else:
            d = -s
            L = N - d
            grad[:L] += pw * (1.0 - h[d:])
            grad[d:] -= pw * h[:L]
    return F, grad, c

def lag_gradient_only(h, r_star, dx, N):
    s = r_star - (N - 1)
    grad = np.zeros(N)
    pw = dx
    if s == 0:
        grad = pw * (1.0 - 2.0 * h)
    elif s > 0:
        d = s
        L = N - d
        grad[d:] += pw * (1.0 - h[:L])
        grad[:L] -= pw * h[d:]
    else:
        d = -s
        L = N - d
        grad[:L] += pw * (1.0 - h[d:])
        grad[d:] -= pw * h[:L]
    return grad

def optimize_smooth(h0, N, m, tau_schedule, max_iter):
    dx = 2.0 / N
    h = h0.copy()
    best_h = h.copy()
    best_score = score_exact(h, N)
    for tau in tau_schedule:
        F, grad, c = smooth_obj_grad(h, tau, dx)
        if not np.all(np.isfinite(grad)):
            break
        alpha = 1.0
        fails = 0
        for it in range(max_iter):
            F, grad, c = smooth_obj_grad(h, tau, dx)
            if not np.all(np.isfinite(grad)):
                break
            y = project_box_sum(h - alpha * grad, m)
            if not np.all(np.isfinite(y)):
                break
            dvec = y - h
            if np.max(np.abs(dvec)) <= 1e-11:
                break
            s_y = score_exact(y, N)
            if np.isfinite(s_y) and s_y < best_score:
                best_score = s_y
                best_h = y.copy()
            Fy, _, _ = smooth_obj_grad(y, tau, dx)
            gd = np.dot(grad, dvec)
            if gd < 0 and Fy <= F + 1e-4 * gd + 1e-12:
                h = y
                alpha = min(1.25 * alpha, 10.0)
                fails = 0
            else:
                alpha *= 0.5
                fails += 1
                if fails > 24:
                    break
            if F - Fy <= 1e-12 * max(1.0, abs(F)):
                fails += 1
                if fails >= 5:
                    break
    s_final = score_exact(h, N)
    if s_final < best_score:
        best_score = s_final
        best_h = h.copy()
    best_h = project_box_sum(best_h, m)
    return best_h, best_score

def refine_active_lag(h, N, m, n_iter=20):
    dx = 2.0 / N
    h = h.copy()
    best_h = h.copy()
    best_score = score_exact(h, N)
    for _ in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        r_star = int(np.argmax(corr))
        g = lag_gradient_only(h, r_star, dx, N)
        step = dx * 2.0 / (np.linalg.norm(g) + 1e-12)
        y = project_box_sum(h - step * g, m)
        s_y = score_exact(y, N)
        if np.isfinite(s_y) and s_y < best_score:
            best_score = s_y
            best_h = y.copy()
            h = y
    return best_h, best_score
