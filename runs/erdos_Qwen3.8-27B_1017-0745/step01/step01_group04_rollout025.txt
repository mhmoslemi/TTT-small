        best = project_box_sum(u, m)
    best = project_box_sum(best, m)
    best_score = score_exact(best, N)
    base_h = best.copy()
    base_score = best_score
    if N > 256:
        c5 = score_exact(best, N)
        return best, c5, int(N)
    B = build_helmert(N)
    BT = B.T
    d = N - 1
    sigmas = [0.12, 0.07, 0.035]
    max_generations = 100
    stag_limit = 24
    if N >= 97 and N <= 160:
        sigmas = [0.10, 0.05]
        max_generations = 60
        stag_limit = 18
    elif N >= 161:
        sigmas = [0.08]
        max_generations = 30
        stag_limit = 12
    for r_i in range(len(sigmas)):
        mu_x0 = BT @ (best - u)
        C0 = np.eye(d)
        try:
            best, best_score = cma_run(mu_x0, C0.copy(), sigmas[r_i], B, BT, N, m, u,
                                        rng, max_generations, stag_limit, best, best_score)
        except Exception:
            pass
    best = project_box_sum(best, m)
    c5 = score_exact(best, N)
    bs = score_exact(base_h, N)
    if not np.isfinite(c5) or c5 >= bs:
        best = project_box_sum(base_h, m)
        c5 = score_exact(best, N)
    return best, c5, int(N)
