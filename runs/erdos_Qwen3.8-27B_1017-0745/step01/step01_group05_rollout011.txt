            stall = 0
        if stall >= 5:
            break
    return best_h, best_s

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = globals().get('initial_h_values', None)
    n = 45
    fallback = None
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.ndim == 1 and 2 <= len(p) <= 999 and np.all(np.isfinite(p)) and np.max(np.abs(p)) <= 1e6:
            n = len(p)
            fb, ok = project_capped(p, n)
            fallback = fb if ok else np.full(n, 0.5)
    if fallback is None:
        nk = kwargs.get('n_points', None)
        if isinstance(nk, int) and not isinstance(nk, bool) and 2 <= nk <= 999:
            n = nk
        fallback = np.full(n, 0.5)
    best_h = fallback.copy()
    best_s = exact_score(best_h, n)
    d = n - 1 if n <= 64 else min(n - 1, 48)
    B = helmert_basis(n, d) if n <= 64 else random_basis(n, d, rng)
    starts = [fallback.copy()]
    for sym in (1.0 - fallback, fallback[::-1].copy(), 1.0 - fallback[::-1].copy()):
        sp, ok = project_capped(sym, n)
        if ok and all(np.max(np.abs(sp - s)) > 1e-10 for s in starts):
            starts.append(sp)
    n_gen = 36 if n <= 100 else (12 if n <= 400 else 6)
    configs = [(starts[0], 0.06)]
    if len(starts) > 1:
        configs.append((starts[1], 0.10))
    if n <= 100 and len(starts) > 2:
        configs.append((starts[2], 0.08))
    if n <= 100:
        z = rng.standard_normal(n)
        z -= z.mean()
        zn = math.sqrt(np.mean(z ** 2))
        if zn > 1e-12:
            pp, ok = project_capped(best_h + 0.08 * z / zn, n)
            if ok and all(np.max(np.abs(pp - c[0])) > 1e-10 for c in configs):
                configs.append((pp, 0.04))
    for anchor, sig in configs:
        try:
            Bh = B if n <= 64 else random_basis(n, d, rng)
            bh, bs = cma_restart(anchor, Bh, sig, n_gen, rng, n)
            if bs < best_s - 1e-12:
                best_s = bs
                best_h = bh.copy()
        except Exception:
            pass
    final_h, fok = project_capped(best_h, n)
    if fok:
        best_h = final_h
    fs = exact_score(best_h, n)
    fb_s = exact_score(fallback, n)
    if (not np.isfinite(fs)) or fs > fb_s + 1e-12:
        best_h = fallback.copy()
        fs = fb_s
    return best_h.astype(np.float64), float(fs), int(n)
