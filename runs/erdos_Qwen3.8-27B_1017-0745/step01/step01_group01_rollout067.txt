import numpy as np
from scipy.optimize import minimize

def project_capped(z, n):
    s = n / 2.0
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if mid == lo or mid == hi:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for i in range(n):
        if abs(r) < 1e-13:
            break
        if r > 0:
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
        else:
            a = min(-r, h[i])
            h[i] -= a
            r += a
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-9:
        return None
    return h

def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return float('inf')
    if abs(float(np.sum(h)) - n / 2.0) > 1e-8:
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_vals_grads(h, n):
    L = 2 * n - 1
    v = np.zeros(L)
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        ii = np.arange(lo, hi)
        jj = ii + d
        v[row] = float(np.dot(h[ii], 1.0 - h[jj]))
        G[row, lo:hi] += 1.0 - h[lo + d:hi + d]
        G[row, lo + d:hi + d] -= h[lo:hi]
    return v, G

def slsqp_obj(z, n):
    return float(z[n])

def slsqp_obj_jac(z, n):
    j = np.zeros(n + 1)
    j[n] = 1.0
    return j

def slsqp_eq(z, n):
    return float(np.sum(z[:n]) - n / 2.0)

def slsqp_eq_jac(z, n):
    j = np.zeros(n + 1)
    j[:n] = 1.0
    return j

def slsqp_ineq(z, n):
    h = z[:n]
    t = float(z[n])
    v, _ = lag_vals_grads(h, n)
    return t - (2.0 / n) * v

def slsqp_ineq_jac(z, n):
    h = z[:n]
    _, G = lag_vals_grads(h, n)
    L = 2 * n - 1
    jac = np.zeros((L, n + 1))
    jac[:, :n] = -(2.0 / n) * G
    jac[:, n] = 1.0
    return jac

def polish(h, n, sweeps):
    h = h.copy()
    deltas = np.array([0.001, 0.003, 0.01, 0.03, 0.05, 0.1, 0.2, 0.3])
    for _ in range(sweeps):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                lo_d = max(-h[i], h[j] - 1.0)
                hi_d = min(1.0 - h[i], h[j])
                if hi_d <= lo_d + 1e-12:
                    continue
                cur = exact_score(h, n)
                best_d, best_s = 0.0, cur
                for d_val in np.concatenate([deltas, -deltas, [lo_d, hi_d]]):
                    if d_val == 0.0:
                        continue
                    d_val = max(lo_d, min(hi_d, d_val))
                    h[i] += d_val
                    h[j] -= d_val
                    s_ev = exact_score(h, n)
                    h[i] -= d_val
                    h[j] += d_val
                    if np.isfinite(s_ev) and s_ev < best_s - 1e-12:

                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    improved = True
        if not improved:
            break
    return h
