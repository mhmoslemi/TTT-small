def smooth_grad(z, x, tau, n):
    dx = 2.0 / n
    L = 2 * n - 1
    vals = np.zeros(L)
    grads = np.zeros((L, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        p0, q0, m = max(d, 0), max(-d, 0), n - abs(d)
        p, qv = z[p0:p0 + m], z[q0:q0 + m]
        vals[row] = dx * np.dot(p, 1.0 - qv)
        grads[row, p0:p0 + m] = dx * (1.0 - qv)
        grads[row, q0:q0 + m] -= dx * p
    vmax = np.max(vals)
    shifted = tau * (vals - vmax)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    return grads.T @ weights

def smooth_eq(z, x, tau, n):
    return np.sum(z) - n / 2.0

def smooth_eq_jac(z, x, tau, n):
    return np.ones(len(z))

def smooth_step(x, n, tau, maxiter):
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, x.copy(), args=(x, tau, n),
                       jac=smooth_grad, method='L-BFGS-B', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': smooth_eq, 'jac': smooth_eq_jac, 'args': (x, tau, n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-8})
    except Exception:
        return None
    if res.x is None or res.x.shape != (n,) or not np.all(np.isfinite(res.x)):
        return None
    return res.x

def subgrad_step(x, n, beta, n_inner):
    dx = 2.0 / n
    y = x.copy()
    best_model = float('inf')
    best_y = x.copy()
    for k in range(n_inner):
        u = y - x
        Amax = -1e30
        d_star = 0
        g_active = np.zeros(n)
        for d in range(-(n - 1), n):
            p0, q0, m = max(d, 0), max(-d, 0), n - abs(d)
            p, qv = x[p0:p0 + m], x[q0:q0 + m]
            A_d = dx * (np.dot(p, 1.0 - qv) + np.dot(1.0 - qv, u[p0:p0+m]) - np.dot(p, u[q0:q0+m]))
            if A_d > Amax:
                Amax = A_d
                d_star = d
                g_active = np.zeros(n)
                g_active[p0:p0 + m] += dx * (1.0 - qv)
                g_active[q0:q0 + m] -= dx * p
        model_val = Amax + beta * dx * np.dot(u, u)
        if model_val < best_model:
            best_model = model_val
            best_y = y.copy()
        v = g_active + 2.0 * beta * dx * u
        nv = np.linalg.norm(v)
        if nv < 1e-15:
            break
        eta = min(0.5 / (nv + 1e-12), 1.0 / (2.0 * beta * dx * np.sqrt(k + 1)))
        y_new = project_capped(y - eta * v, n)

            break
        if np.max(np.abs(y_new - y)) < 1e-9:
            y = y_new
            break
        y = y_new
    return best_y

def build_starts(base, n, rng, count):
    pool = []
    def _add(v):

    _add(project_capped(1.0 - base[::-1], n))

    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    _add(project_capped(alt, n))
    _add(project_capped(base + rng.normal(0, 0.08, n), n))
    _add(project_capped(base + rng.normal(0, 0.20, n), n))
    _add(project_capped(rng.uniform(0, 1, n), n))
    return pool[:count]

def optimize_mm(x, n, n_outer, slsqp_maxiter):

        best_z, best_s = None, score

            rejs += 1
            continue
        x, score = best_z.copy(), best_s

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False

        return False
    return True

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)

        return float('inf')

    q = np.zeros(L)
    G = np.zeros((L, n))

        G[row, p0:p0 + m] += dx * (1.0 - qv)
        G[row, q0:q0 + m] -= dx * p
    return q, G

    n = len(x)
    u = z[:n] - x
    return z[n] + beta * dx * np.dot(u, u)

def mm_obj_jac(z, x, beta, dx):
    n = len(x)
    u = z[:n] - x
    return np.concatenate([2.0 * beta * dx * u, [1.0]])

    u = z[:n] - x
    return z[n] - q - G @ u

def mm_ineq_jac(z, x, dx, q, G):
    n = len(x)
    J = np.zeros((len(q), n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def mm_eq(z, x, dx, q, G):
    return np.sum(z[:len(x)]) - len(x) / 2.0

def mm_eq_jac(z, x, dx, q, G):
    J = np.zeros(len(x) + 1)
    J[:len(x)] = 1.0
    return J

def mm_step(x, n, beta, maxiter):

    q, G = lag_q_and_grad(x, n)
    z0 = np.concatenate([x, [np.max(q) + 1e-8]])

    shifted = tau * (vals - vmax)
    exp_v = np.exp(shifted)
    w = exp_v / np.sum(exp_v)
    return grads.T @ w

        if nv < 1e-15:
            break
        eta = min(0.5 / (nv + 1e-12), 1.0 / (2.0 * beta * dx * np.sqrt(k + 1)))
        yn = project_capped(y - eta * v, n)
        if yn is None:
            break
        if np.max(np.abs(yn - y)) < 1e-9:
            y = yn
            break
        y = yn
    return best_y

def build_starts(base, n, rng, count):
    pool = []
    def _add(v):
        if v is None:
            return
        v = np.asarray(v, dtype=np.float64)
        for s in pool:
            if np.max(np.abs(v - s)) < 1e-8:
                return
        pool.append(v)
    _add(base.copy())
    _add(project_capped(base[::-1], n))
    _add(project_capped(1.0 - base[::-1], n))
    _add(np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    _add(project_capped(blk, n))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    _add(project_capped(alt, n))
    _add(project_capped(base + rng.normal(0, 0.08, n), n))
    _add(project_capped(base + rng.normal(0, 0.20, n), n))
    _add(project_capped(rng.uniform(0, 1, n), n))
    return pool[:count]

def optimize_mm(x, n, n_outer, slsqp_mi):
    x = x.copy().astype(np.float64)
    score = exact_score(x, n)
    beta, rejs = 1.0, 0
    for _ in range(n_outer):
        if rejs >= 5 or beta > 64.0:
            break
        if n <= 128:
            y = mm_step(x, n, beta, slsqp_mi)
        else:
            y = subgrad_step(x, n, beta, min(50, 200 // n))
        if y is None:
            beta = min(2.0 * beta, 64.0)
            rejs += 1
            continue
        y = project_capped(y, n)
        if y is None or not is_feasible(y, n):
            beta = min(2.0 * beta, 64.0)
            rejs += 1
            continue
        best_z, best_s = None, score
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            z = project_capped(x + alpha * (y - x), n)
            if z is None:
                continue
            sz = exact_score(z, n)
            if sz < best_s:
                best_s, best_z = sz, z
        if best_z is None or best_s >= score - 1e-12:
            beta = min(2.0 * beta, 64.0)
            rejs += 1
            continue
        x, score, rejs = best_z.copy(), best_s, 0
        beta = max(1.0, beta / 1.25)
    return x, score

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None or n < 1 or n > 999:
        try:
            v = int(kwargs.get('n_points', 0))
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
        if n is None:
            n = 47
    fb = project_capped(parent, n) if parent is not None else None
    if fb is None:
        fb = np.full(n, 0.5)
    fb_score = exact_score(fb, n)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        one = project_capped(np.full(1, 0.5), 1)
        return one, exact_score(one, 1), 1
    if n <= 64:
        n_st, n_mm, smi = 8, 10, 80
    elif n <= 128:
        n_st, n_mm, smi = 5, 7, 60
    elif n <= 256:
        n_st, n_mm, smi = 3, 5, 0
    elif n <= 512:
        n_st, n_mm, smi = 2, 3, 0
    else:
        n_st, n_mm, smi = 2, 2, 0
    starts = build_starts(fb, n, rng, n_st)
    scored = []
    for st in starts:
        st = project_capped(st, n)
        if st is not None and is_feasible(st, n):
            scored.append((exact_score(st, n), st))
    scored.sort(key=lambda t: t[0])
    gb, gs = fb.copy(), fb_score
    for s, st in scored:
        if s < gs - 1e-12:
            gs, gb = s, st.copy()
    search_list = [fb]
    for _, st in scored:
        if not any(np.max(np.abs(st - s2)) < 1e-8 for s2 in search_list):
            search_list.append(st)
        if len(search_list) >= n_st:
            break
    for st in search_list:
        try:
            xo, so = optimize_mm(st, n, n_mm, smi)
            if so < gs - 1e-12:
                gs, gb = so, xo.copy()
        except Exception:
            pass
    try:
        xo2, so2 = smooth_pg(gb, n, n_mm)
        if so2 < gs - 1e-12:
            gs, gb = so2, xo2.copy()
    except Exception:
        pass
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n) or exact_score(final, n) >= fb_score:
        final = fb
    fs = exact_score(final, n)
    return final, float(fs), int(n)
