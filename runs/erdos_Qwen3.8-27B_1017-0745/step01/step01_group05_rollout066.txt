def project_capped(y, n):
    S = n / 2.0
    if not np.all(np.isfinite(y)) or y.shape != (n,):
        return np.full(n, 0.5)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    lam = min(hi, max(lo, float(np.mean(y)) - 0.5))
    for _ in range(60):
        c = np.clip(y - lam, 0.0, 1.0)
        r = c.sum() - S
        if abs(r) < 1e-14:
            break
        if r > 0:
            lo = lam
        else:
            hi = lam
        a = np.sum((y - lam > 1e-15) & (y - lam < 1.0 - 1e-15))
        if a > 0:
            nl = lam + r / a
            lam = nl if (np.isfinite(nl) and lo < nl < hi) else 0.5 * (lo + hi)
        else:
            lam = 0.5 * (lo + hi)
    c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(3):
        diff = S - c.sum()
        if abs(diff) < 1e-13:
            break
        mask = c < 1.0 - 1e-12 if diff > 0 else c > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        c[mask] += diff / cnt
        np.clip(c, 0.0, 1.0, out=c)
    return c

def exact_score(h, n):
    dx = 2.0 / n
    qs = np.correlate(h, 1.0 - h, mode='full') * dx
    if qs.shape[0] != 2 * n - 1 or not np.all(np.isfinite(qs)):
        return float('inf')
    return float(qs.max())

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(h.sum() - n / 2.0) <= 1e-10)

def build_G(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for l in range(2 * n - 1):
        k = l - (n - 1)
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if hi1 > lo1:
            G[l, lo1:hi1] += dx * (1.0 - h[lo1 - k:hi1 - k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if hi2 > lo2:
            G[l, lo2:hi2] -= dx * h[lo2 + k:hi2 + k]
    return G

def project_simplex(z):
    n = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(n, 1.0 / n)
    sv = np.sort(z)[::-1]
    cv = np.cumsum(sv)
    rho = np.arange(1, n + 1)
    cond = sv - (cv - 1.0) / rho > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    r = int(np.where(cond)[0][-1])
    theta = (cv[r] - 1.0) / (r + 1)
    res = np.clip(z - theta, 0.0, None)
    s = res.sum()
    if s > 1e-15 and abs(s - 1.0) < 1e-10:
        res[int(np.argmax(res))] += 1.0 - s
    return res

def dual_eval(p, h, G, q, L, n):

    y = project_capped(h - gmix / L, n)
    d = y - h
    u = q + G @ d
    Dv = p @ u + 0.5 * L * float(d @ d)
    gap = float(u.max()) - float(p @ u)
    return y, d, u, Dv, gap

def optimize_start(h0, n, max_outer, max_inner):

    if not check_feasible(h, n):
        h = np.full(n, 0.5)
    s = exact_score(h, n)
    best_h = h.copy()
    best_s = s
    L = 4.0 / n
    stall = 0
    for _ in range(max_outer):
        q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        G = build_G(h, n)
        if not np.all(np.isfinite(G)):
            break
        d = solve_majorant(h, q, G, n, max_inner)
        if not np.all(np.isfinite(d)) or np.max(np.abs(d)) < 1e-12:
            break
        v = G @ d
        norm2 = float(d @ d)
        accepted = False
        for ap in range(18):
            alpha = 2.0 ** (-ap)
            M_a = float(np.max(q + alpha * v)) + 0.5 * L * alpha * alpha * norm2
            if M_a > float(q.max()) - 1e-14:

            if not check_feasible(ht, n):

def slsqp_obj(x):
    return x[-1]

def slsqp_obj_jac(x):
    g = np.zeros(len(x))
    g[-1] = 1.0
    return g

def slsqp_ineq(x):
    n = len(x) - 1
    h = x[:n]
    return x[-1] - np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def slsqp_ineq_jac(x):
    n = len(x) - 1
    h = x[:n]
    dx = 2.0 / n
    jac = np.zeros((2 * n - 1, n + 1))
    jac[:, -1] = 1.0
    for l in range(2 * n - 1):
        k = l - (n - 1)
        lo1, hi1 = max(0, k), min(n, n + k)
        if hi1 > lo1:
            jac[l, lo1:hi1] -= dx * (1.0 - h[lo1 - k:hi1 - k])
        lo2, hi2 = max(0, -k), min(n, n - k)
        if hi2 > lo2:
            jac[l, lo2:hi2] += dx * h[lo2 + k:hi2 + k]
    return jac

def slsqp_eq(x):
    n = len(x) - 1
    return x[:n].sum() - n / 2.0

def slsqp_eq_jac(x):
    g = np.zeros(len(x))
    g[:len(x) - 1] = 1.0
    return g

def slsqp_refine(h0, n):
    s0 = exact_score(h0, n)
    x0 = np.concatenate([h0, [min(1.0, s0 + 1e-6)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    con = [{'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac},
           {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}]
    try:
        from scipy.optimize import minimize
        res = minimize(slsqp_obj, x0, method='SLSQP', jac=slsqp_obj_jac,
                       bounds=bounds, constraints=con,
                       options={'maxiter': 200, 'ftol': 1e-14, 'disp': False})
        hx = res.x[:n]
        if not np.all(np.isfinite(hx)):
            return h0, s0
        hx = project_capped(hx, n)
        if not check_feasible(hx, n):
            return h0, s0
        s = exact_score(hx, n)
        return (hx, s) if s < s0 - 1e-13 else (h0, s0)
    except Exception:
        return h0, s0
