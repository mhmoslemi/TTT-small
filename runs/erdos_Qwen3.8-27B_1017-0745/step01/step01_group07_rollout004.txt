    except Exception:

def optimize_trajectory(x0, n, iters):
    S = n / 2.0
    x = x0.copy()
    f = exact_score(x, n)
    if f is None:
        return x0, None
    rho = 0.08
    rho_max = 0.25
    rho_min = 2e-4
    fails = 0
    best_x, best_f = x.copy(), f
    small_acc = 0
    for _ in range(iters):
        if rho < rho_min:
            break
        q, G = build_lag_model(x, n)
        if q is None:
            break
        delta = solve_lp_step(x, q, G, rho, n)
        if delta is None or np.max(np.abs(delta)) < 1e-9:
            rho = max(rho_min, rho * 0.5)
            fails += 1
            if fails >= 4:
                break
            continue
        best_cand, best_cf = None, None
        for j in range(9):
            beta = 2.0 ** (-j)
            cand = project_capped_simplex(x + beta * delta, S)
            if cand is None or not is_feasible(cand, n):
                continue
            cf = exact_score(cand, n)
            if cf is not None and cf < f - 1e-12:
                if best_cf is None or cf < best_cf:
                    best_cf, best_cand = cf, cand
        if best_cand is None:
            rho = max(rho_min, rho * 0.5)
            fails += 1
            if fails >= 4:
                break
            continue
        s = best_cand - x
        pred = f - float(np.max(q + G.dot(s)))
        act = f - best_cf
        if pred <= 1e-12 or (pred > 0 and act / pred < 0.25):
            rho = max(rho_min, rho * 0.5)
        elif pred > 0 and act / pred > 0.75 and np.max(np.abs(s)) >= 0.8 * rho:
            rho = min(rho_max, 1.5 * rho)
        x, f = best_cand, best_cf
        fails, small_acc = 0, small_acc + 1
        if f < best_f:
            best_f, best_x = f, x.copy()
        if act <= 1e-10 and np.max(np.abs(s)) <= 1e-5:
            small_acc += 1
        else:
            small_acc = 0
        if small_acc >= 3 or np.max(np.abs(s)) < 1e-8:
            break
    return best_x, best_f

def build_starts(n, S, parent, rng):
    starts = []
    starts.append(np.full(n, 0.5))
    if parent is not None:
        starts.append(parent.copy())
        for sd in (0.04, 0.12):
            p = project_capped_simplex(parent + rng.normal(0, sd, n), S)
            if p is not None:
                starts.append(p)
        starts.append(parent[::-1].copy())
        p = project_capped_simplex(1.0 - parent, S)
        if p is not None:
            starts.append(p)
    idx = np.arange(n, dtype=np.float64)
    M = min(5, (n - 1) // 2)
    u = 0.5 * np.ones(n)
    for m in range(1, M + 1):
        a = rng.normal(0, 0.10 / np.sqrt(m))
        phi = rng.uniform(0, 2.0 * np.pi)
        u += a * np.cos(2.0 * np.pi * m * (idx + 0.5) / n + phi)
    pf = project_capped_simplex(u, S)
    if pf is not None:
        starts.append(pf)
    perm = rng.permutation(n)
    binv = np.zeros(n)
    binv[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        binv[perm[n // 2]] = 0.5
    starts.append(binv)
    deduped = []
    for s in starts:
        if not is_feasible(s, n):
            s = project_capped_simplex(s, S)
            if s is None or not is_feasible(s, n):
                continue
        dup = False
        for d in deduped:
            if np.max(np.abs(s - d)) < 1e-9:
                dup = True
                break
        if not dup:
            deduped.append(s)
    return deduped

def run(seed=42, **kwargs):
    if seed is None:
        seed = 42
    try:
        seed = max(0, int(seed))
    except (TypeError, ValueError):
        seed = 42
    rng = np.random.default_rng(seed)
    n, parent = None, None
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        try:
            iv_arr = np.asarray(iv, dtype=np.float64)
            if iv_arr.ndim == 1 and 8 <= iv_arr.size <= 999:
                n = iv_arr.size
                S = n / 2.0
                if np.all(np.isfinite(iv_arr)) and np.max(np.abs(iv_arr)) <= 1e12:
                    if np.all(iv_arr >= -1e-12) and np.all(iv_arr <= 1.0 + 1e-12) and abs(np.sum(iv_arr) - S) <= 1e-10:
                        parent = project_capped_simplex(np.clip(iv_arr, 0.0, 1.0), S)
                    else:
                        parent = project_capped_simplex(iv_arr, S)
        except Exception:
            pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 80))
        except (TypeError, ValueError):
            n = 80
        if n < 8 or n > 999:
            n = 80
    S = n / 2.0
    h_const = np.full(n, 0.5)
    best_h, best_score = h_const.copy(), exact_score(h_const, n)
    if best_score is None:
        best_score = 0.5
    parent_fb = None
    if parent is not None and is_feasible(parent, n):
        ps = exact_score(parent, n)
        if ps is not None and ps < best_score:
            best_score, best_h = ps, parent.copy()
        parent_fb = parent.copy()
    starts = build_starts(n, S, parent, rng)
    max_traj = 6 if n <= 128 else (4 if n <= 256 else (3 if n <= 512 else 2))
    max_iter = 30 if n <= 128 else (22 if n <= 256 else (14 if n <= 512 else 6))
    for idx in range(min(len(starts), max_traj)):
        ox, of = optimize_trajectory(starts[idx], n, max_iter)
        if of is not None and of < best_score - 1e-12:
            best_score, best_h = of, ox.copy()
    n_rest = 2 if n <= 128 else (1 if n <= 256 else 0)
    sigma = 0.05
    for _ in range(n_rest):
        rp = project_capped_simplex(best_h + sigma * rng.normal(0, 1, n), S)
        if rp is not None and is_feasible(rp, n):
            ox, of = optimize_trajectory(rp, n, max_iter)
            if of is not None and of < best_score - 1e-12:
                best_score, best_h = of, ox.copy()
                sigma = max(0.02, 0.5 * sigma)
            else:
                sigma = min(0.18, 1.8 * sigma)
    if not is_feasible(best_h, n):
        best_h = project_capped_simplex(best_h, S)
        if best_h is None or not is_feasible(best_h, n):
            best_h = h_const.copy()
    fs = exact_score(best_h, n)
    if fs is None:
        best_h = h_const.copy()
        fs = exact_score(best_h, n)
    if fs is None:
        fs = 0.5
    if parent_fb is not None:
        pfs = exact_score(parent_fb, n)
        if pfs is not None and pfs < fs:
            best_h, fs = parent_fb.copy(), pfs
    return best_h.astype(np.float64), float(fs), n
