def lags_QG(h):
    n = h.size
    N = 2 * n - 1
    Q = np.zeros(N)
    G = np.zeros((N, n))
    for k in range(-(n - 1), n):
        r = k + n - 1
        i0 = max(0, k)
        i1 = min(n - 1, n - 1 + k)
        if i0 > i1:
            continue
        i = np.arange(i0, i1 + 1)
        j = i - k
        hi = h[i]
        Q[r] = float(np.dot(hi, 1.0 - h[j]))
        G[r, i] += 1.0 - h[j]
        G[r, j] += -hi
    return Q, G

def power_obj(h):
    c = np.correlate(h, 1.0 - h, mode='full')
    Qp = np.maximum(c, 0.0)
    s = float(np.sum(Qp ** _P))
    if s <= 0.0:
        return 0.0
    return float(s ** (1.0 / _P))

def power_jac(h):
    Q, G = lags_QG(h)
    Qp = np.maximum(Q, 0.0)
    s = float(np.sum(Qp ** _P))
    if s <= 1e-18:
        return np.zeros(h.size)
    w = Qp ** (_P - 1.0)
    w *= s ** (1.0 / _P - 1.0)
    return np.dot(G.T, w)

def mass_fun(h):
    return float(np.sum(h) - _SUM)

def mass_jac(h):
    return np.ones(len(h))

def power_refine(h0, n, S, p_list):
    global _SUM, _P
    h0 = project_box_sum(np.asarray(h0, dtype=np.float64), S)
    if h0 is None:
        return None, np.inf
    cur = h0
    cur_s = exact_score(cur, n)
    best = cur
    best_s = cur_s
    _SUM = float(S)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': mass_fun, 'jac': mass_jac}]
    for p in p_list:
        _P = float(p)
        try:
            res = minimize(power_obj, cur, jac=power_jac, method='SLSQP',
                           bounds=bnds, constraints=cons,
                           options={'maxiter': 50, 'ftol': 1e-12, 'disp': False})
            if res.x is not None and len(res.x) == n:
                hp = project_box_sum(res.x, S)
                if hp is not None and check_feasible(hp, n):
                    s = exact_score(hp, n)
                    if np.isfinite(s):
                        if s < best_s - 1e-12:
                            best = hp
                            best_s = s
                        if s < cur_s + 1e-10:
                            cur = hp
                            cur_s = s
        except Exception:
            pass
    return best, best_s

import numpy as np
from scipy.optimize import linprog, minimize

_LP_LIMIT = 60
_P = 8.0
_SUM = 0.0

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    n = v.size
    S = float(S)
    if S < -1e-12 or S > float(n) + 1e-12:
        return None
    lo = float(v.min()) - 1.0
    hi = float(v.max())
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if y.sum() > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    err = S - y.sum()
    if abs(err) > 1e-13:
        idx = int(np.argmax(1.0 - y)) if err > 0 else int(np.argmax(y))
        y[idx] = float(np.clip(y[idx] + err, 0.0, 1.0))
    if y.min() < -1e-9 or y.max() > 1.0 + 1e-9 or abs(y.sum() - S) > 1e-7 * max(1.0, float(n)):
        return None
    return y

def check_feasible(h, n):
    if h is None or h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-8) or np.any(h > 1.0 + 1e-8):
        return False
    if abs(float(h.sum()) - 0.5 * float(n)) > 1e-6 * max(1.0, float(n)):
        return False
    return True

def exact_score(h, n):
    if h is None or h.shape != (n,) or not np.all(np.isfinite(h)):
        return np.inf
    if np.any(h < -1e-8) or np.any(h > 1.0 + 1e-8):
        return np.inf
    if abs(float(h.sum()) - 0.5 * float(n)) > 1e-5 * max(1.0, float(n)):
        return np.inf
    c = np.correlate(h, 1.0 - h, mode='full')
    if c.shape != (2 * n - 1,) or not np.all(np.isfinite(c)):
        return np.inf
    val = float(np.max(c) * 2.0 / float(n))
    return val if np.isfinite(val) else np.inf

def lags_QG(h):
    n = h.size
    N = 2 * n - 1
    Q = np.zeros(N)
    G = np.zeros((N, n))
    for k in range(-(n - 1), n):
        r = k + n - 1
        i0 = max(0, k)
        i1 = min(n - 1, n - 1 + k)
        if i0 > i1:
            continue
        i = np.arange(i0, i1 + 1)
        j = i - k
        hi = h[i]
        Q[r] = float(np.dot(hi, 1.0 - h[j]))
        G[r, i] += 1.0 - h[j]
        G[r, j] += -hi
    return Q, G

def power_obj(h):
    c = np.correlate(h, 1.0 - h, mode='full')
    Qp = np.maximum(c, 0.0)
    s = float(np.sum(Qp ** _P))
    if s <= 0.0:
        return 0.0
    return float(s ** (1.0 / _P))

def power_jac(h):
    Q, G = lags_QG(h)
    Qp = np.maximum(Q, 0.0)
    s = float(np.sum(Qp ** _P))
    if s <= 1e-18:
        return np.zeros(h.size)
    w = Qp ** (_P - 1.0)
    w *= s ** (1.0 / _P - 1.0)
    return np.dot(G.T, w)

def mass_fun(h):
    return float(np.sum(h) - _SUM)

def mass_jac(h):
    return np.ones(len(h))

def power_refine(h0, n, S, p_list):
    global _SUM, _P
    h0 = project_box_sum(np.asarray(h0, dtype=np.float64), S)
    if h0 is None:
        return None, np.inf
    cur = h0
    cur_s = exact_score(cur, n)
    best = cur
    best_s = cur_s
    _SUM = float(S)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': mass_fun, 'jac': mass_jac}]
    for p in p_list:
        _P = float(p)
        try:
            res = minimize(
                power_obj, cur, jac=power_jac, method='SLSQP',
                bounds=bnds, constraints=cons,
                options={'maxiter': 60, 'ftol': 1e-12, 'disp': False}
            )
            if res.x is not None and len(res.x) == n:
                hp = project_box_sum(res.x, S)
                if hp is not None and check_feasible(hp, n):
                    s = exact_score(hp, n)
                    if np.isfinite(s):
                        if s < best_s - 1e-12:
                            best = hp
                            best_s = s
                        if s < cur_s + 1e-10:
                            cur = hp
                            cur_s = s
        except Exception:
            pass
    return best, best_s

def solve_lp(x, n, S, l, u, qlim, bias):
    if n > _LP_LIMIT:
        return None
    P = n * (n + 1) // 2
    N = 2 * n - 1
    nv = n + P + 1
    t = n + P
    rc = 4 * P + N
    A = np.zeros((rc, nv))
    b = np.zeros(rc)
    ri = 0
    for i in range(n):
        li = float(l[i])
        ui = float(u[i])
        for j in range(i, n):
            lj = float(l[j])
            uj = float(u[j])
            w = n + i * n - i * (i - 1) // 2 + (j - i)
            A[ri, i] = lj
            A[ri, j] += li
            A[ri, w] -= 1.0
            b[ri] = li * lj
            ri += 1
            A[ri, i] = uj
            A[ri, j] += ui
            A[ri, w] -= 1.0
            b[ri] = ui * uj
            ri += 1
            A[ri, i] = -lj
            A[ri, j] = -ui
            A[ri, w] = 1.0
            b[ri] = -ui * lj
            ri += 1
            A[ri, i] = -uj
            A[ri, j] = -li
            A[ri, w] = 1.0
            b[ri] = -li * uj
            ri += 1
    for k in range(-(n - 1), n):
        i0 = max(0, k)
        i1 = min(n - 1, n - 1 + k)
        for ii in range(i0, i1 + 1):
            jj = ii - k
            p = ii if ii < jj else jj
            q = jj if ii < jj else ii
            w = n + p * n - p * (p - 1) // 2 + (q - p)
            A[ri, ii] = 1.0
            A[ri, w] -= 1.0
            A[ri, t] -= 1.0
        ri += 1
    Aeq = np.zeros((1, nv))
    Aeq[0, :n] = 1.0
    c = np.zeros(nv)
    if bias is not None:
        nb = float(np.linalg.norm(bias))
        if nb > 0.0:
            c[:n] = bias / nb
    bnds = [(0.0, 1.0)] * nv
    for i in range(n):
        bnds[i] = (float(l[i]), float(u[i]))
    bnds[t] = (0.0, float(qlim))
    try:
        sol = linprog(c, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=[float(S)], bounds=bnds, method='highs')
    except Exception:
        return None
    if sol.status == 0 and sol.x is not None:
        v = np.array(sol.x[:n], dtype=np.float64)
        if np.all(np.isfinite(v)):
            v = np.clip(v, l - 1e-7, u + 1e-7)
            vp = project_box_sum(v, S)
            if vp is not None and check_feasible(vp, n):
                return vp
    return None

def add_candidate(lst, h, n, S):
    if h is None:
        return
    p = project_box_sum(h, S)
    if p is not None and check_feasible(p, n):
        lst.append(p)

def harmonic_perturb(n, amp, m1, m2, ph1, ph2):
    x = np.arange(n)
    v = amp * (
        np.cos(2.0 * np.pi * m1 * x / float(n) + ph1) +
        np.cos(2.0 * np.pi * m2 * x / float(n) + ph2)
    )
    v = v - v.mean()
    sd = float(np.sqrt(np.mean(v * v)))
    if sd > 1e-12:
        v = amp * v / sd
    else:
        v = amp * np.random.randn(n)
        v = v - v.mean()
    return v

def dedup_starts(lst, n):
    out = []
    for h in lst:
        if h is None or h.shape != (n,):
            continue
        dup = False
        for g in out:
            if float(np.max(np.abs(h - g))) < 1e-10:
                dup = True
                break
        if not dup:
            out.append(h)
    return out

def score_pool(pool, n):
    pairs = []
    for h in pool:
        if h is None or h.shape != (n,):
            continue
        s = exact_score(h, n)
        if np.isfinite(s):
            pairs.append((float(s), h))
    if not pairs:
        return None, None, pairs
    sc = np.array([p[0] for p in pairs], dtype=float)
    order = np.argsort(sc)
    return pairs[order[0]][1], float(sc[order[0]]), pairs

def run(seed=42, **kwargs):
    np.random.seed(int(seed) % 2147483647)
    parent = None
    n = 45
    try:
        pv = np.asarray(initial_h_values, dtype=np.float64)
        if pv.ndim == 1 and 2 <= pv.size <= 999 and np.all(np.isfinite(pv)):
            if pv.min() >= -1e-9 and pv.max() <= 1.0 + 1e-9:
                n = int(pv.size)
                target = 0.5 * float(n)
                if abs(pv.sum() - target) < 1e-6 * max(1.0, float(n)):
                    pp = project_box_sum(pv, target)
                    if pp is not None and check_feasible(pp, n):
                        parent = pp
    except Exception:
        parent = None
        n = 45
    if parent is None:
        n = 45
    S = 0.5 * float(n)
    fallback = parent
    if fallback is None:
        fallback = project_box_sum(np.full(n, 0.5), S)
    if n > 120:
        fh = project_box_sum(fallback, S)
        if fh is None or not check_feasible(fh, n):
            fh = project_box_sum(np.full(n, 0.5), S)
        fs = exact_score(fh, n)
        if not np.isfinite(fs):
            fh = project_box_sum(np.full(n, 0.5), S)
            fs = exact_score(fh, n)
        return fh, float(fs), int(n)

    starts = []
    if parent is not None:
        add_candidate(starts, parent, n, S)
        add_candidate(starts, parent[::-1], n, S)
        add_candidate(starts, 1.0 - parent, n, S)
        add_candidate(starts, np.roll(parent, n // 3), n, S)
        add_candidate(starts, np.roll(parent, 1), n, S)
        for amp in (0.02, 0.05, 0.10, 0.18):
            g = amp * np.random.randn(n)
            g = g - g.mean()
            add_candidate(starts, parent + g, n, S)
        for m1, m2 in ((1, 2), (2, 3), (3, 5)):
            ph = float(np.random.uniform(0.0, 2.0 * np.pi))
            add_candidate(starts, parent + 0.07 * harmonic_perturb(n, 1.0, m1, m2, ph, 1.3 * ph), n, S)
        for _ in range(3):
            add_candidate(starts, np.random.rand(n), n, S)
        for _ in range(2):
            add_candidate(starts, 0.5 + 0.25 * np.random.randn(n), n, S)
    else:
        add_candidate(starts, np.full(n, 0.5), n, S)
        for _ in range(5):
            add_candidate(starts, np.random.rand(n), n, S)
        for _ in range(3):
            add_candidate(starts, 0.5 + 0.3 * np.random.randn(n), n, S)

    starts = dedup_starts(starts, n)
    if not starts:
        starts = [fallback]

    pool = starts[:]
    for h0 in starts[:5]:
        hh, ss = power_refine(h0, n, S, (8.0, 16.0))
        if hh is not None and check_feasible(hh, n):
            pool.append(hh)
    pool = dedup_starts(pool, n)
    best_h, best_s, pairs = score_pool(pool, n)
    best_n = n
    if best_h is None:
        best_h = project_box_sum(fallback, S)
        if best_h is None:
            best_h = project_box_sum(np.full(n, 0.5), S)
        best_s = exact_score(best_h, n)
        best_n = int(best_h.size)

    if n <= _LP_LIMIT and best_h is not None:
        sc = np.array([p[0] for p in pairs], dtype=float)
        order = np.argsort(sc)
        lp = []
        for idx in order[:2]:
            base = pairs[idx][1]
            qmax = float(exact_score(base, n) * float(n) / 2.0) + 1e-6
            for r in (0.20, 0.40):
                l = np.maximum(0.0, base - r)
                u = np.minimum(1.0, base + r)
                for _ in range(2):
                    bias = np.random.randn(n)
                    bias = bias - bias.mean()
                    v = solve_lp(base, n, S, l, u, qmax, bias)
                    add_candidate(lp, v, n, S)
                bias = np.sign(base - 0.5)
                add_candidate(lp, solve_lp(base, n, S, l, u, qmax, bias), n, S)
                add_candidate(lp, solve_lp(base, n, S, l, u, qmax, -bias), n, S)
        if lp:
            lp = dedup_starts(lp, n)
            for h0 in lp[:4]:
                hh, ss = power_refine(h0, n, S, (16.0, 32.0))
                if hh is not None and check_feasible(hh, n):
                    pool.append(hh)
            pool = dedup_starts(pool, n)
            bh, bs, pairs = score_pool(pool, n)
            if bh is not None:
                best_h = bh
                best_s = bs
                best_n = int(bh.size)

    bn = int(best_h.size)
    bS = 0.5 * float(bn)
    if best_h is not None and bn <= 120 and np.isfinite(best_s) and best_s > 0.38075:
        last = []
        add_candidate(last, best_h, bn, bS)
        g = 0.01 * np.random.randn(bn)
        g = g - g.mean()
        add_candidate(last, best_h + g, bn, bS)
        ph = float(np.random.uniform(0.0, 2.0 * np.pi))
        add_candidate(last, best_h + 0.03 * harmonic_perturb(bn, 1.0, 1, 2, ph, 1.7 * ph), bn, bS)
        for h0 in last:
            hh, ss = power_refine(h0, bn, bS, (16.0, 32.0))
            if hh is not None and check_feasible(hh, bn):
                s = exact_score(hh, bn)
                if np.isfinite(s) and s < best_s - 1e-14:
                    best_h = hh
                    best_s = s
                    bn = int(bn)
                    bS = 0.5 * float(bn)

    final = project_box_sum(best_h, bS)
    fn = int(best_h.size)
    if final is None or not check_feasible(final, fn) or not np.isfinite(exact_score(final, fn)):
        fbS = 0.5 * float(fallback.size)
        final = project_box_sum(fallback, fbS)
        fn = int(final.size)
        if final is None:
            final = project_box_sum(np.full(n, 0.5), 0.5 * float(n))
            fn = n
    fs = exact_score(final, fn)
    if not np.isfinite(fs):
        final = project_box_sum(np.full(n, 0.5), 0.5 * float(n))
        fn = n
        fs = exact_score(final, fn)
    return final, float(fs), int(fn)
