                order = np.argsort(np.where(np.isfinite(scores), scores, 1e30))
                n_el = max(2, math.ceil(0.2 * NP))
                for oi in order:
                    if len(new_pop) >= n_el:
                        break
                    if float(np.max(np.abs(pop[oi] - new_pop[0]))) > 1e-10:
                        new_pop.append(pop[oi].copy())
                for _ in range(NP - len(new_pop)):
                    base = new_pop[rng.integers(len(new_pop))]
                    h2 = project_capped(base + rng.normal(0, sc_base, n), n)
                    if h2 is None:
                        h2 = project_capped(rng.uniform(0, 1, n), n)
                    if h2 is None:
                        h2 = best.copy()
                    new_pop.append(h2)
                pop = new_pop[:NP]
                scores = np.array([exact_score(p, n) for p in pop])
                for i in range(NP):
                    if np.isfinite(scores[i]) and scores[i] < best_score - 1e-12:
                        best, best_score = pop[i].copy(), float(scores[i])
                stag = 0
            else:
                break
    final = project_capped(best, n)
    if final is None:
        final = best.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > best_score + 1e-12:
        final, fs = best.copy(), best_score
    if not np.isfinite(fs):
        final, fs = uniform.copy(), exact_score(uniform, n)
    return final, float(fs), int(n)
