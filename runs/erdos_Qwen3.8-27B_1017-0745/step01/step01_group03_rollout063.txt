def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-14:
        if resid > 0:
            for i in np.argsort(1.0 - y):
                add = min(resid, 1.0 - y[i])
                y[i] += add; resid -= add
                if abs(resid) < 1e-15: break
        else:
            for i in np.argsort(y):
                sub = min(-resid, y[i])
                y[i] -= sub; resid += sub
                if abs(resid) < 1e-15: break
    y = np.clip(y, 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-10:
        return None
    if resid > 0 and y[0] < 1.0: y[0] += min(resid, 1.0-y[0])
    elif resid < 0 and y[0] > 0.0: y[0] += max(resid, -y[0])
    return y

            best_sc = sc_s; best_h = s.copy()
    for s in starts:
        h_loc, sc_loc = local_lp_search(s, n, rng)
        if h_loc is not None and sc_loc < best_sc - 1e-12:
            best_sc = sc_loc; best_h = h_loc.copy()
    for _ in range(6):
        sig = 0.03 + 0.04 * rng.random()
        pert = best_h + sig * rng.standard_normal(n)
        c = project_box_sum(pert, S)
        if c is not None:
            h_loc, sc_loc = local_lp_search(c, n, rng)
            if h_loc is not None and sc_loc < best_sc - 1e-12:
                best_sc = sc_loc; best_h = h_loc.copy()
    final_h = project_box_sum(best_h, S)
    if final_h is None or not check_feasible(final_h, n):
        final_h = fallback
    final_sc = exact_score(final_h)
    if not np.isfinite(final_sc) or final_sc > fb_score:
        final_h = fallback; final_sc = fb_score
    return (np.array(final_h, dtype=np.float64), float(final_sc), n)
