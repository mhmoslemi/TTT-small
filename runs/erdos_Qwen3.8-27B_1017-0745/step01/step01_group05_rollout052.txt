        cnt = int(mask.sum())
        if cnt == 0: break
        c[mask] += diff / cnt
        np.clip(c, 0.0, 1.0, out=c)
    ok = bool(np.all(c >= -1e-12) and np.all(c <= 1.0 + 1e-12) and abs(c.sum() - S) < 1e-10)
    return c, ok

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and np.all(h >= -1e-12)
            and np.all(h <= 1.0 + 1e-12) and abs(h.sum() - n / 2.0) <= 1e-10)

def lag_scores(h, n):
    dx = 2.0 / n
    cs = np.correlate(h, 1.0 - h, mode='full')
    if len(cs) != 2 * n - 1 or not np.all(np.isfinite(cs)):
        return None
    return cs * dx

def exact_score(h, n):
    if not check_feasible(h, n): return float('inf')
    cs = lag_scores(h, n)
    return float(np.max(cs)) if cs is not None else float('inf')

def build_G(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        row = k + n - 1
        lo1, hi1 = max(0, k), min(n, n + k)
        if hi1 > lo1:
            G[row, lo1:hi1] += dx * (1.0 - h[lo1 - k:hi1 - k])
        lo2, hi2 = max(0, -k), min(n, n - k)
        if hi2 > lo2:
            G[row, lo2:hi2] -= dx * h[lo2 + k:hi2 + k]
    return G

def project_simplex(z):
    m = len(z)
    if not np.all(np.isfinite(z)): return np.full(m, 1.0 / m)
    s = np.sort(z)[::-1]
    cs = np.cumsum(s)
    idx = np.where(s - (cs - 1.0) / np.arange(1, m + 1) > 0)[0]
    r = idx[0] if len(idx) > 0 else m - 1
    p = np.maximum(z - (cs[r] - 1.0) / (r + 1), 0.0)
    d = 1.0 - p.sum()
    if abs(d) > 1e-14:
        p[np.argmax(p)] += d
        np.clip(p, 0.0, None, out=p)
    return p

def dual_state(h, p, G, q, L, n):
    gmix = G.T @ p
    y, ok = project_capped(gmix, n)
    if not ok: return None
    d = y - h
    u = q + G @ d
    if not np.all(np.isfinite(u)): return None
    Dv = float(p @ u) + 0.5 * L * float(d @ d)
    gap = float(np.max(u) - p @ u)
    return d, u, Dv, gap

def solve_majorant(h, q, G, n, max_inner):
    dx = 2.0 / n
    L = 2.0 * dx
    m = 2 * n - 1
    Ldual = max(1e-12, 1.05 * float(np.sum(G * G)) / L)
    qmax = float(np.max(q))
    thr = max(1e-12, 1e-8 * max(1.0, abs(qmax)))
    act = np.where(q >= qmax - thr)[0]
    if len(act) == 0: act = np.array([int(np.argmax(q))])
    x = np.zeros(m); x[act] = 1.0 / len(act)
    best_d, best_D, best_gap = np.zeros(n), float(x @ q), float(np.max(q) - x @ q)
    y_a = x.copy(); theta = 1.0; stall = 0
    for _ in range(max_inner):
        ds = dual_state(h, y_a, G, q, L, n)
        if ds is None: break
        u_a = ds[1]; D_a = ds[2]
        x_new = project_simplex(y_a + u_a / Ldual)
        ds2 = dual_state(h, x_new, G, q, L, n)
        if ds2 is None: break
        d_new, u_new, D_new, gap_new = ds2
        if D_new < best_D - 1e-13:
            x_restart = project_simplex(x + u_a / Ldual)
            dsr = dual_state(h, x_restart, G, q, L, n)
            if dsr is not None and dsr[2] >= best_D - 1e-13:
                x, y_a, theta = x_restart, x_restart.copy(), 1.0
            else:
                x, y_a, theta = x_new, x_new.copy(), 1.0
        else:
            tn = (1.0 + math.sqrt(1.0 + 4.0 * theta * theta)) / 2.0
            y_t = project_simplex(x_new + ((theta - 1.0) / tn) * (x_new - x))
            x, y_a, theta = x_new, y_t, tn
        if gap_new < best_gap:
            best_gap, best_D, best_d = gap_new, D_new, d_new.copy()
        sn = float(np.max(np.abs(x_new - x)))
        if sn < 1e-10: stall += 1
        else: stall = 0
        if stall >= 3 or best_gap <= max(1e-10, 1e-8 * max(1.0, abs(qmax))):
            break
    return best_d

def optimize_start(h0, n, max_outer, max_inner):
    h = h0.copy()
    h, ok = project_capped(h, n)
    if not ok: h = np.full(n, 0.5)
    s = exact_score(h, n)
    best_h, best_s = h.copy(), s
    for _ in range(max_outer):
        q = lag_scores(h, n)
        if q is None: break
        G = build_G(h, n)
        if not np.all(np.isfinite(G)): break
        d = solve_majorant(h, q, G, n, max_inner)
        if d is None or not np.all(np.isfinite(d)) or np.max(np.abs(d)) < 1e-12:
            break
        v = G @ d
        n2 = float(d @ d)
        dx = 2.0 / n
        L = 2.0 * dx
        accepted = False
        for i in range(20):
            alpha = 0.5 ** i
            m_alpha = float(np.max(q + alpha * v)) + 0.5 * L * alpha * alpha * n2
            if m_alpha >= float(np.max(q)) - 1e-14:
                if alpha < 0.01: break
                continue
            ht = h + alpha * d
            ht, ok2 = project_capped(ht, n)
            if not ok2: continue
            st = exact_score(ht, n)
            if st < s - 1e-12:
                h, s = ht, st
                if s < best_s: best_s, best_h = s, ht.copy()
                accepted = True
                break
        if not accepted: break
    return best_h, best_s
