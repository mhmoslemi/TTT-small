def grad_lag(h, n, l):
    dx = 2.0 / n
    k = l - (n - 1)
    g = np.zeros(n)
    # Terms where m = i (h[m] in 1-h[i] part): i=m valid 0<=m<n, need i+k=m+k in [0,n)
    lo1, hi1 = max(0, -k), min(n, n - k)  # i range where i+k is valid
    # Actually: i ranges over [max(0,-k), min(n, n-k)) for the sum
    # d/dh[i] includes -dx*h[i+k]
    for i in range(max(0, -k), min(n, n - k)):
        g[i] -= dx * h[i + k]
    # Terms where m = i+k (h[m] in h[i+k] part): i=m-k valid, need i=m-k in [0,n)
    # i ranges over [max(0,k), min(n, n+k)) -> m = i+k ranges over [max(0,k)+k, ...)
    # Actually: for the sum, i ranges over [max(0,-k), min(n, n-k))
    # m = i+k, so m ranges over [max(0,-k)+k, min(n, n-k)+k) = [max(k,0), min(n+k, n))
    for i in range(max(0, -k), min(n, n - k)):
        m = i + k
        g[m] += dx * (1.0 - h[i])
    return g

            mut = pop[r1] + F * (pop[r2] - pop[r3])
            msk = rng.random(d) < 0.8
            msk[int(rng.integers(d))] = True
            tri = np.where(msk, mut, pop[i])
            tri = reflect_unit(tri)
            h = decode(anchor, B, sigma, tri, n, d)
            if h is not None:
                s = exact_score(h, n)
                if s < scores[i] - 1e-12:
                    pop[i] = tri
                    scores[i] = s
                    if s < best_s - 1e-12:
                        best_s = s
                        best_h = h.copy()
                        best_u = tri.copy()
                        improved = True
        if improved:
            stag = 0
        else:
            stag += 1
            if stag >= 4 and not restarted and gen < G - 1:
                restarted = True
                for i in range(1, P):
                    pop[i] = reflect_unit(best_u + 0.35 * rng.standard_normal(d))
                scores = np.full(P, float('inf'))
                for i in range(P):
                    h = decode(anchor, B, sigma, pop[i], n, d)
                    if h is not None:
                        scores[i] = exact_score(h, n)
                stag = 0
    return best_h, best_s

def grad_lag(h, n, l):
    dx = 2.0 / n
    k = l - (n - 1)
    g = np.zeros(n)
    lo, hi = max(0, -k), min(n, n - k)
    for i in range(lo, hi):
        g[i] -= dx * h[i + k]
        g[i + k] += dx * (1.0 - h[i])
    return g

def pgd_polish(h0, n, steps=80, lr0=0.02):
    h = h0.copy()
    best_h = h0.copy()
    best_s = exact_score(h0, n)
    lr = lr0
    for _ in range(steps):
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        l = int(np.argmax(corr))
        g = grad_lag(h, n, l)
        y = h - lr * g
        hn, ok = project_capped(y, n)
        if not ok:
            lr *= 0.5
            continue
        s = exact_score(hn, n)
        if s < best_s - 1e-14:
            best_s = s
            best_h = h.copy() * 0 + hn
            best_h = hn.copy()
            lr = min(lr * 1.1, 0.1)
        else:
            lr *= 0.5
        h = hn
    return best_h, best_s

def swap_polish(h0, n, rng, iters=200):
    h = h0.copy()
    best_h = h0.copy()
    best_s = exact_score(h0, n)
    for _ in range(iters):
        i, j = rng.integers(n), rng.integers(n)
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        s = exact_score(h, n)
        if s < best_s - 1e-12:
            best_s = s
            best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_s

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = globals().get('initial_h_values', None)
    n = 45
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.ndim == 1 and 2 <= len(p) <= 999 and np.all(np.isfinite(p)) and np.max(np.abs(p)) < 1e6:
            n = len(p)
            base, ok = project_capped(p, n)
            if not ok:
                base = np.full(n, 0.5)
        else:
            base = np.full(n, 0.5)
    else:
        nk = kwargs.get('n_points', None)
        if isinstance(nk, int) and not isinstance(nk, bool) and 2 <= nk <= 999:
            n = nk
        base = np.full(n, 0.5)
    base, _ = project_capped(base, n)
    best_h = base.copy()
    best_s = exact_score(base, n)

    d_target = min(n - 1, 20)
    B, d = make_basis(n, d_target)
    if B is not None and d >= 2:
        z = rng.standard_normal(n)
        z -= z.mean()
        nm = math.sqrt(float(np.dot(z, z)))
        alt = base.copy()
        if nm > 1e-12:
            alt_raw = base + 0.12 * z / nm * math.sqrt(n) * 0.3
            alt, ok = project_capped(alt_raw, n)
            if ok:
                sa = exact_score(alt, n)
                if sa < best_s - 1e-12:
                    best_s = sa
                    best_h = alt.copy()
        else:
            alt = base.copy()

        if n <= 100:
            P, G = max(12, min(48, 2 * d + 8)), 18
        elif n <= 400:
            P, G = max(10, min(32, d + 8)), 6
        else:
            P, G = max(8, min(16, d)), 3

        stages = [(base, 0.10), (alt, 0.18), (best_h, 0.05)]
        for anchor, sigma in stages:
            try:
                sh, ss = evolve_stage(anchor, B, sigma, P, G, rng, n, d)
                sh, ok2 = project_capped(sh, n)
                if ok2:
                    ss = exact_score(sh, n)
                    if ss < best_s - 1e-12:
                        best_s = ss
                        best_h = sh.copy()
            except Exception:
                pass

    for _ in range(3):
        try:
            ph, ps = pgd_polish(best_h, n, steps=60, lr0=0.015)
            if ps < best_s - 1e-12:
                best_s = ps
                best_h = ph.copy()
            ph, ps = swap_polish(best_h, n, rng, iters=150)
            if ps < best_s - 1e-12:
                best_s = ps
                best_h = ph.copy()
        except Exception:
            break

    fh, ok = project_capped(best_h, n)
    if ok:
        fs = exact_score(fh, n)
        fb = exact_score(base, n)
        if fs <= fb + 1e-12:
            best_h, best_s = fh, fs
        else:
            best_h, best_s = base, fb
    else:
        best_h, best_s = base, exact_score(base, n)
    final_h = np.asarray(best_h, dtype=np.float64)
    final_s = exact_score(final_h, n)
    return final_h, float(final_s), int(n)
