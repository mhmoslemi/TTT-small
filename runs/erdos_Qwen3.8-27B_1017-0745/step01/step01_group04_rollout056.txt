            s_trial = exact_score(y, N)
            if s_trial < best_s:
                best_s = s_trial
                best_h = y.copy()
            F_y, _ = smooth_obj_grad(y, tau, N)
            if not np.isfinite(F_y):
                alpha *= 0.5
                if alpha < 1e-9:
                    break
                continue
            gd = np.dot(g, d)
            if gd < 0 and F_y <= F + 1e-4 * gd + 1e-12:
                h = y
                alpha = min(1.25 * alpha, 10.0)
            else:
                alpha *= 0.5
                if alpha < 1e-9:
                    break
        s_end = exact_score(h, N)
        if s_end < best_s:
            best_s = s_end
            best_h = h.copy()
    return best_h, best_s

def make_fourier_start(N, rng, n_modes, amp):
    z = np.full(N, 0.5)
    ns = min(n_modes, (N - 1) // 2)
    idx = np.arange(N)
    for r in range(1, ns + 1):
        A = rng.standard_normal()
        B = rng.standard_normal()
        z += (amp / r) * (A * np.sin(2.0 * np.pi * r * idx / N) + B * np.cos(2.0 * np.pi * r * idx / N))
    return z

def make_block_start(N, rng, nb):
    z = np.zeros(N)
    blk = N // nb
    for i in range(N):
        z[i] = 0.5 + 0.4 * np.sin(2.0 * np.pi * i / N)
    return z

def make_periodic_start(N, rng, period):
    z = np.zeros(N)
    for i in range(N):
        frac = (i % period) / period
        z[i] = 0.5 + 0.3 * np.cos(2.0 * np.pi * frac)
    z += 0.03 * rng.standard_normal(N)
    return z

def run(seed=42, **kwargs):
    N = None
    m = None
    raw = globals().get('initial_h_values', None)
    if raw is not None:
        raw = np.asarray(raw, dtype=np.float64)
        if raw.ndim == 1 and raw.size > 1 and np.all(np.isfinite(raw)):
            N = raw.size
            m = N / 2.0
    if N is None:
        N = kwargs.get('n_points', 70)
        if not isinstance(N, (int, np.integer)) or N < 2:
            N = 70
        m = N / 2.0

    rng = np.random.default_rng(seed)
    tau_sched = [0.02, 0.006, 0.002, 0.0006, 0.0002, 0.00006, 0.000015]
    max_iter = 60

    if raw is not None and np.all(np.isfinite(raw)):
        parent = project_box_sum(np.clip(raw, 0.0, 1.0), m)
    else:
        parent = np.full(N, 0.5)

    best_h = parent.copy()
    best_s = exact_score(parent, N)

    starts = [parent.copy(), np.full(N, 0.5)]
    comp = 1.0 - parent
    starts.append(project_box_sum(comp, m))
    for sc in [0.04, 0.12, 0.25]:
        starts.append(project_box_sum(parent + sc * rng.standard_normal(N), m))
    starts.append(project_box_sum(0.5 + 0.4 * rng.uniform(size=N), m))
    starts.append(project_box_sum(make_fourier_start(N, rng, 5, 0.12), m))
    starts.append(project_box_sum(make_fourier_start(N, rng, 3, 0.20), m))
    starts.append(project_box_sum(make_fourier_start(N, rng, 7, 0.08), m))
    starts.append(project_box_sum(make_periodic_start(N, rng, 5), m))
    starts.append(project_box_sum(make_periodic_start(N, rng, 10), m))

    # Deduplicate
    unique_starts = []
    for s in starts:
        dup = False
        for u in unique_starts:
            if np.max(np.abs(s - u)) <= 1e-10:
                dup = True
                break
        if not dup:
            unique_starts.append(s)
    starts = unique_starts[:10]

    for h0 in starts:
        s0 = exact_score(h0, N)
        if s0 < best_s:
            best_s = s0
            best_h = h0.copy()
        h_opt, s_opt = optimize_start(h0, N, m, tau_sched, max_iter)
        if s_opt < best_s:
            best_s = s_opt
            best_h = h_opt.copy()

    # Basin hops
    hop_scales = [0.05, 0.02]
    for sc in hop_scales:
        hop = project_box_sum(best_h + sc * rng.standard_normal(N), m)
        s_hop = exact_score(hop, N)
        if s_hop < best_s:
            best_s = s_hop
            best_h = hop.copy()
        h_opt, s_opt = optimize_start(hop, N, m, tau_sched, max_iter)
        if s_opt < best_s:
            best_s = s_opt
            best_h = h_opt.copy()

    # Final validation
    best_h = project_box_sum(best_h, m)
    c5 = exact_score(best_h, N)
    if not np.isfinite(c5):
        best_h = project_box_sum(parent, m)
        c5 = exact_score(best_h, N)
    return best_h, c5, int(N)
