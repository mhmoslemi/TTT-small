import numpy as np
from scipy.optimize import minimize

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if v.ndim != 1 or not np.all(np.isfinite(v)) or S < -1e-6 or S > n + 1e-6:
        return None
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-13:
        for i in np.argsort(1.0 - y)[::-1]:
            d = min(r, 1.0 - y[i]); y[i] += d; r -= d
            if r <= 1e-14: break
    elif r < -1e-13:
        for i in np.argsort(y):
            d = min(-r, y[i]); y[i] -= d; r += d
            if r >= -1e-14: break
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(float(np.sum(y)) - S) > 1e-8:
        return None
    return y

def feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and np.all(h >= -1e-12) and
            np.all(h <= 1.0 + 1e-12) and abs(float(np.sum(h)) - 0.5 * n) < 1e-10 * n)

def score(h, n):
    if not feasible(h, n): return float('inf')
    try:
        c = np.correlate(h, 1.0 - h, mode='full')
        if len(c) != 2 * n - 1 or not np.all(np.isfinite(c)): return float('inf')
        return float(np.max(c) * 2.0 / n)
    except Exception:
        return float('inf')

def lag_qg(h, s, n):
    dx = 2.0 / n
    lo, hi = max(0, -s), min(n, n - s)
    q = 0.0
    g = np.zeros(n)
    for i in range(lo, hi):
        j = i + s
        q += h[i] * (1.0 - h[j])
        g[i] += 1.0 - h[j]
        g[j] -= h[i]
    return q * dx, g * dx

def all_lags(x, n):
    q_all = np.zeros(2 * n - 1)
    G_all = np.zeros((2 * n - 1, n))
    for idx, s in enumerate(range(-(n - 1), n)):
        q_all[idx], G_all[idx] = lag_qg(x, s, n)
    return q_all, G_all

def mm_f(z, ctx):
    x, L = ctx
    n = len(x)
    d = z[:n] - x
    return 0.5 * L * float(np.dot(d, d)) + z[n]

def mm_fjac(z, ctx):
    x, L = ctx
    n = len(x)
    g = np.zeros(n + 1)
    g[:n] = L * (z[:n] - x)
    g[n] = 1.0
    return g

def mm_ineq_f(z, ctx):
    x, qJ, GJ = ctx
    n = len(x)
    d = z[:n] - x
    return z[n] - qJ - GJ @ d

def mm_ineq_jac(z, ctx):
    x, qJ, GJ = ctx
    n = len(x)
    m = GJ.shape[0]
    J = np.zeros((m, n + 1))
    J[:, :n] = -GJ
    J[:, n] = 1.0
    return J

def mm_eq_f(z, n):
    return float(np.sum(z[:n])) - 0.5 * n

def mm_eq_jac(z, n):
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def solve_mm(x, n):
    dx = 2.0 / n
    L = 2.0 * dx
    q_all, G_all = all_lags(x, n)
    active = list(range(2 * n - 1)) if n <= 64 else list(np.argsort(-q_all)[:12])
    qJ = q_all[active]
    GJ = G_all[active]
    y = x.copy()
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    ineq_args = (x, qJ, GJ)
    cons = [
        {'type': 'ineq', 'fun': mm_ineq_f, 'jac': mm_ineq_jac, 'args': ineq_args},
        {'type': 'eq', 'fun': mm_eq_f, 'jac': mm_eq_jac, 'args': (n,)}
    ]
    for _ in range(10):
        t_init = float(np.max(qJ + GJ @ (y - x))) + 1e-8
        z0 = np.append(y, t_init)
        try:
            res = minimize(mm_f, z0, jac=mm_fjac, args=(x, L), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-11, 'disp': False})
            zr = np.array(res.x, dtype=np.float64)
        except Exception:
            break
        y_raw = zr[:n]
        t_raw = zr[n]
        if not np.all(np.isfinite(zr)):
            break
        if np.any(y_raw < -1e-4) or np.any(y_raw > 1.0 + 1e-4):
            break
        if abs(float(np.sum(y_raw)) - 0.5 * n) > 1e-4 * n:
            break
        d_cand = y_raw - x
        slacks = t_raw - qJ - GJ @ d_cand
        if np.any(slacks < -1e-4):
            break
        y = project_box_sum(y_raw, 0.5 * n)
        if y is None:
            break
        d = y - x
        if n > 64 and len(active) < min(2 * n - 1, 96):
            max_viol = 0.0
            best_s = -1
            for idx in range(2 * n - 1):
                if idx in active: continue
                ell = q_all[idx] + G_all[idx] @ d
                viol = ell - t_raw
                if viol > max_viol:
                    max_viol = viol
                    best_s = idx
            if max_viol > 1e-8:
                active.append(best_s)
                qJ = np.append(qJ, q_all[best_s])
                GJ = np.vstack([GJ, G_all[best_s]])
                ineq_args = (x, qJ, GJ)
                cons[0]['args'] = ineq_args
                continue
        break
    return y

def mm_descent(x0, n, M):
    h = project_box_sum(x0, 0.5 * n)
    if h is None: return x0.copy()
    c = score(h, n)
    for _ in range(M):
        prop = solve_mm(h, n)
        if prop is None or not np.all(np.isfinite(prop)):
            break
        if np.max(np.abs(prop - h)) < 1e-11:
            break
        best_alpha = 1.0
        best_c = c
        n_evals = 9 if n <= 128 else 6
        for k in range(n_evals):
            alpha = 2.0 ** (-k)
            cand = project_box_sum(h + alpha * (prop - h), 0.5 * n)
            if cand is None: continue
            sc = score(cand, n)
            if sc < best_c - 1e-13:
                best_c = sc
                best_alpha = alpha
        if best_alpha >= 2.0 ** (-(n_evals - 1)) - 1e-12:
            h = prop.copy()
            c = score(h, n)
        elif best_c < c - 1e-13:
            h = project_box_sum(h + best_alpha * (prop - h), 0.5 * n)
            c = score(h, n)
        else:
            break
        if c >= c - 1e-10 * max(1.0, abs(c)):
            pass
    return h

def get_parent():
    n = 45
    if 'initial_h_values' in globals():
        ph = np.asarray(globals()['initial_h_values'], dtype=np.float64)
        if ph.ndim == 1 and 2 <= len(ph) <= 999 and np.all(np.isfinite(ph)):
            if np.all(ph >= -1e-6) and np.all(ph <= 1.0 + 1e-6):
                n = len(ph)
                p = project_box_sum(np.clip(ph, 0.0, 1.0), 0.5 * n)
                if p is not None and feasible(p, n):
                    return p, n
    return np.full(n, 0.5), n

def build_starts(parent, n, rng):
    starts = []
    starts.append(parent.copy())
    starts.append(1.0 - parent.copy())
    starts.append(parent[::-1].copy())
    for sd in [0.05, 0.12, 0.25]:
        p = project_box_sum(parent + sd * rng.standard_normal(n), 0.5 * n)
        if p is not None: starts.append(p)
    p = project_box_sum(rng.random(n), 0.5 * n)
    if p is not None: starts.append(p)
    u = 0.5 + 0.30 * np.cos(2.0 * np.pi * np.arange(n) / n + 0.7) + 0.10 * np.sin(4.0 * np.pi * np.arange(n) / n + 1.3)
    p = project_box_sum(u, 0.5 * n)
    if p is not None: starts.append(p)
    u2 = 0.5 + 0.35 * np.cos(2.0 * np.pi * 2.0 * np.arange(n) / n + 0.3)
    p = project_box_sum(u2, 0.5 * n)
    if p is not None: starts.append(p)
    unique = []
    for s in starts:
        if not feasible(s, n): continue
        dup = False
        for u in unique:
            if np.max(np.abs(s - u)) < 1e-10:
                dup = True; break
        if not dup: unique.append(s)
    return unique

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent, n = get_parent()
    M = max(8, min(40, 1800 // n))
    best_h = parent.copy()
    best_c = score(parent, n)
    starts = build_starts(parent, n, rng)
    for s in starts:
        h = mm_descent(s, n, M)
        c = score(h, n)
        if c < best_c - 1e-14:
            best_h = h.copy()
            best_c = c
    for sd in [0.04, 0.10, 0.22]:
        p = project_box_sum(best_h + sd * rng.standard_normal(n), 0.5 * n)
        if p is None: continue
        h = mm_descent(p, n, max(4, M // 2))
        c = score(h, n)
        if c < best_c - 1e-14:
            best_h = h.copy()
            best_c = c
    best_h = project_box_sum(best_h, 0.5 * n)
    if best_h is not None and feasible(best_h, n):
        best_c = score(best_h, n)
    parent_c = score(parent, n)
    if parent_c < best_c:
        best_h = parent.copy()
        best_c = parent_c
    if not np.all(np.isfinite(best_h)) or not feasible(best_h, n):
        best_h = np.full(n, 0.5)
        best_c = score(best_h, n)
    return (best_h, best_c, n)
