def smooth_value_grad(h, tau):
    N = len(h)
    dx = 2.0 / float(N)
    u = 1.0 - h
    corr = np.correlate(h, u, mode='full')
    c = dx * corr
    cmax = float(np.max(c))
    tau = max(float(tau), 1e-12)
    e = np.exp((c - cmax) / tau)
    s = float(np.sum(e))
    p = e / s
    F = cmax + tau * math.log(s)
    C = N - 1
    A = np.convolve(u, p, mode='full')[C:C+N]
    B = np.correlate(h, p, mode='full')[C:C+N]
    grad = dx * (A - B)
    if not (np.isfinite(F) and np.all(np.isfinite(grad))):
        return np.inf, np.zeros(N, dtype=np.float64), np.inf
    return F, grad, cmax

import numpy as np
import math
from scipy.optimize import minimize


def project_box_sum(z, N, m):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape[0] != N:
        return np.full(N, 0.5, dtype=np.float64)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5, dtype=np.float64)
    lo = float(z.min()) - 1.0
    hi = float(z.max())
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(z - mid, 0.0, 1.0)))
        if s > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = float(m - h.sum())
    if r > 0.0:
        for i in range(N):
            if r <= 1e-14:
                break
            adj = min(r, 1.0 - h[i])
            if adj > 0.0:
                h[i] += adj
                r -= adj
    elif r < 0.0:
        for i in range(N):
            if r >= -1e-14:
                break
            adj = min(-r, h[i])
            if adj > 0.0:
                h[i] -= adj
                r += adj
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or h.min() < -1e-9 or h.max() > 1.0 + 1e-9:
        return np.full(N, 0.5, dtype=np.float64)
    if abs(float(h.sum()) - m) > 1e-7 * max(1, N):
        return np.full(N, 0.5, dtype=np.float64)
    return h


def is_feasible(h, N, m):
    return (
        h.shape == (N,)
        and bool(np.all(np.isfinite(h)))
        and h.min() >= -1e-7
        and h.max() <= 1.0 + 1e-7
        and abs(float(h.sum()) - m) <= 1e-6 * max(1, N)
    )


def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / float(N)))


def smooth_value_grad(h, tau):
    N = len(h)
    dx = 2.0 / float(N)
    u = 1.0 - h
    corr = np.correlate(h, u, mode="full")
    c = dx * corr
    cmax = float(np.max(c))
    tau = max(float(tau), 1e-12)
    e = np.exp((c - cmax) / tau)
    s = float(np.sum(e))
    p = e / s
    F = cmax + tau * math.log(s)
    C = N - 1
    A = np.convolve(u, p, mode="full")[C:C + N]
    B = np.correlate(h, p, mode="full")[C:C + N]
    grad = dx * (A - B)
    if not (np.isfinite(F) and np.all(np.isfinite(grad))):
        return np.inf, np.zeros(N, dtype=np.float64), np.inf
    return F, grad, cmax


def _sfun(z, tau):
    F, _, _ = smooth_value_grad(z, tau)
    return float(F) if np.isfinite(F) else 1e6


def _sjac(z, tau):
    _, g, _ = smooth_value_grad(z, tau)
    if np.all(np.isfinite(g)):
        return g
    return np.zeros(len(z), dtype=np.float64)


def _eqsum(z, m):
    return np.array([float(np.sum(z)) - m])


def _eqsum_jac(z, m):
    return np.ones((1, len(z)), dtype=np.float64)


def polish_slsqp(h0, N, m, tau, maxiter):
    h0 = project_box_sum(h0, N, m)
    try:
        bounds = [(0.0, 1.0)] * N
        cons = [{"type": "eq", "fun": _eqsum, "jac": _eqsum_jac, "args": (m,)}]
        res = minimize(
            _sfun, h0, jac=_sjac, args=(tau,), method="SLSQP",
            bounds=bounds, constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-12, "disp": False}
        )
        x = res.x
        if x is not None and len(x) == N:
            y = project_box_sum(x, N, m)
            sc = score_exact(y, N)
            if np.isfinite(sc):
                return y, float(sc)
    except Exception:
        pass
    return h0, float(score_exact(h0, N))


def random_binary(N, m, rng):
    out = np.zeros(N, dtype=np.float64)
    fl = int(math.floor(m + 1e-12))
    fr = m - fl
    if fl < 0:
        fl = 0
    if fl > N:
        fl = N
    if fl > 0:
        out[rng.permutation(N)[:fl]] = 1.0
    if fr > 0.25:
        rem = np.flatnonzero(out == 0.0)
        if rem.size > 0:
            out[rem[rng.integers(0, rem.size)]] = 0.5
    return out


def fourier_start(N, rng, amp, R):
    idx = np.arange(N, dtype=np.float64)
    z = np.full(N, 0.5, dtype=np.float64)
    R = min(int(R), (N - 1) // 2)
    for r in range(1, R + 1):
        A = rng.standard_normal()
        B = rng.standard_normal()
        z += (amp / float(r)) * (
            A * np.sin(2.0 * np.pi * r * idx / N) + B * np.cos(2.0 * np.pi * r * idx / N)
        )
    return z


def block_start(N, rng, K, vals):
    K = max(1, min(int(K), N))
    vals = np.asarray(vals, dtype=np.float64).ravel()
    if vals.size == 0:
        return np.full(N, 0.5, dtype=np.float64)
    blocks = np.empty(K, dtype=np.float64)
    for i in range(K):
        blocks[i] = vals[rng.integers(0, vals.size)]
    idx = np.floor(np.linspace(0, K - 1, N)).astype(int)
    return blocks[idx]


def add_start(starts, cand, N, m, max_starts, tol=1e-10):
    if max_starts > 0 and len(starts) >= max_starts:
        return
    y = project_box_sum(cand, N, m)
    if not is_feasible(y, N, m):
        return
    for s in starts:
        if float(np.max(np.abs(s - y))) <= tol:
            return
    starts.append(y)


def build_starts(baseline, N, m, rng, max_starts):
    starts = []
    add_start(starts, baseline, N, m, max_starts)
    add_start(starts, 1.0 - baseline, N, m, max_starts)
    for sh in (1, -1, N // 3, N // 4, N // 2):
        add_start(starts, np.roll(baseline, sh), N, m, max_starts)
    for _ in range(3):
        add_start(starts, random_binary(N, m, rng), N, m, max_starts)
    for _ in range(2):
        add_start(starts, 0.5 + 0.45 * rng.standard_normal(N), N, m, max_starts)
    add_start(starts, fourier_start(N, rng, 0.10, 4), N, m, max_starts)
    add_start(starts, fourier_start(N, rng, 0.22, 6), N, m, max_starts)
    add_start(starts, block_start(N, rng, 7, np.array([0.0, 0.25, 0.5, 0.75, 1.0])), N, m, max_starts)
    add_start(starts, block_start(N, rng, 13, np.array([0.0, 0.5, 1.0])), N, m, max_starts)
    for _ in range(max(1, 2 * max_starts)):
        add_start(starts, 0.5 + 0.35 * rng.standard_normal(N), N, m, max_starts)
        if len(starts) >= max_starts:
            break
    return starts


def optimize_path(h0, N, m, tau_sched, max_iter):
    h = project_box_sum(h0, N, m)
    best_h = h.copy()
    best_score = score_exact(best_h, N)
    alpha_mem = 1.0

    for tau in tau_sched:
        F, grad, exact = smooth_value_grad(h, tau)
        if not (np.isfinite(F) and np.all(np.isfinite(grad))):
            break
        if exact < best_score:
            best_h = h.copy()
            best_score = float(exact)
        gmax = float(np.max(np.abs(grad)))
        if gmax > 1e-10:
            alpha = min(10.0, max(0.02, 0.5 / gmax))
        else:
            alpha = alpha_mem

        for _ in range(int(max_iter)):
            accepted = False
            for _bt in range(30):
                y = project_box_sum(h - alpha * grad, N, m)
                if not is_feasible(y, N, m):
                    alpha *= 0.5
                    continue
                Fy, gy, exacty = smooth_value_grad(y, tau)
                if not (np.isfinite(Fy) and np.all(np.isfinite(gy))):
                    alpha *= 0.5
                    continue
                if exacty < best_score:
                    best_h = y.copy()
                    best_score = float(exacty)
                d = y - h
                dn = float(np.max(np.abs(d)))
                if not np.all(np.isfinite(d)) or dn <= 1e-12:
                    alpha *= 0.5
                    continue
                gd = float(np.dot(grad, d))
                if gd < -1e-15 and Fy <= F + 1e-3 * gd + 1e-10:
                    h = y
                    grad = gy
                    F = Fy
                    exact = exacty
                    alpha = min(10.0, alpha * 1.25)
                    alpha_mem = alpha
                    accepted = True
                    break
                alpha *= 0.5
                if alpha < 1e-5:
                    break
            if not accepted:
                break

        if best_score < exact - 1e-9:
            h = best_h.copy()
            F, grad, exact = smooth_value_grad(h, tau)
            if not (np.isfinite(F) and np.all(np.isfinite(grad))):
                break

    best_h = project_box_sum(best_h, N, m)
    bs = score_exact(best_h, N)
    if np.isfinite(bs) and bs < best_score:
        best_score = float(bs)
    return best_h, float(best_score)


def tau_schedule(N):
    if N <= 128:
        return (0.03, 0.012, 0.005, 0.002, 0.0008, 0.0003, 0.00012, 0.000045, 0.000015)
    if N <= 256:
        return (0.03, 0.012, 0.005, 0.002, 0.0008, 0.0003, 0.00012, 0.00005)
    return (0.03, 0.012, 0.005, 0.002, 0.0008, 0.0004)
