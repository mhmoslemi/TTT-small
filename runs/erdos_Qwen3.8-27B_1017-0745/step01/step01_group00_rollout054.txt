                        ps = exact_score(proj, n)
                        cs = exact_score(h_cur, n)
                        if ps < cs - 1e-12:
                            h_cur = proj
            if exact_score(h_cur, n) < gs - 1e-12:
                gs, gb = exact_score(h_cur, n), h_cur.copy()
            # SLSQP epigraph phase (2 passes)
            for _ in range(2):
                h = h_cur.copy()
                q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
                t0 = float(np.max(q)) + 1e-8
                z0 = np.concatenate([h, [t0]])
                bounds = [(0.0, 1.0)] * n + [(None, None)]
                cons = [{'type': 'ineq', 'fun': ep_ineq, 'jac': ep_ineq_j, 'args': (n,)},
                        {'type': 'eq', 'fun': ep_eq, 'jac': ep_eq_j, 'args': (n,)}]
                try:
                    res = minimize(ep_obj, z0, jac=ep_obj_j, args=(n,),
                                   method='SLSQP', bounds=bounds, constraints=cons,
                                   options={'maxiter': 250, 'ftol': 1e-12})
                except Exception:
                    break
                if res.x is None or res.x.shape != (n + 1,) or not np.all(np.isfinite(res.x)):
                    break
                h_new = project_capped(res.x[:n], n)
                if h_new is None:
                    break
                if np.max(np.abs(h_new - h_cur)) < 1e-13:
                    break
                best_h, best_s = h_cur, exact_score(h_cur, n)
                for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625]:
                    trial = project_capped(h_cur + alpha * (h_new - h_cur), n)
                    if trial is None: continue
                    ts = exact_score(trial, n)
                    if ts < best_s - 1e-12:
                        best_s, best_h = ts, trial.copy()
                if best_s < exact_score(h_cur, n) - 1e-12:
                    h_cur = best_h
                else:
                    break
            s = exact_score(h_cur, n)
            if s < gs - 1e-12:
                gs, gb = s, h_cur.copy()
    else:
        for st in starts:
            h_cur = st.copy()
            cur_s = exact_score(h_cur, n)
            dx = 2.0 / n
            eta, eta_max = 0.2 / dx, 2.0 / dx
            fails = 0
            for _ in range(50 if n <= 128 else 25):
                if fails >= 8 or eta < 1e-8 / dx: break
                q = np.correlate(h_cur, 1.0 - h_cur, mode='full') * dx
                idx = int(np.argmax(q)); d = idx - (n - 1)
                p0, q0, m = max(d, 0), max(-d, 0), n - abs(d)
                g = np.zeros(n)
                g[p0:p0 + m] += dx * (1.0 - h_cur[q0:q0 + m])
                g[q0:q0 + m] -= dx * h_cur[p0:p0 + m]
                z = project_capped(h_cur - eta * g, n)
                if z is None:
                    eta *= 0.5; fails += 1; continue
                v = z - h_cur
                if np.max(np.abs(v)) < 1e-10:
                    eta *= 0.5; fails += 1; continue
                best_h, best_s = None, cur_s
                for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625]:
                    trial = project_capped(h_cur + alpha * v, n)
                    if trial is None: continue
                    ts = exact_score(trial, n)
                    if ts < best_s - 1e-12:
                        best_s, best_h = ts, trial.copy()
                if best_h is not None:
                    h_cur, cur_s = best_h, best_s
                    fails = 0; eta = min(1.2 * eta, eta_max)
                    if cur_s < gs - 1e-12:
                        gs, gb = cur_s, h_cur.copy()
                else:
                    eta *= 0.5; fails += 1
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    if fs > fb_score + 1e-12:
        final, fs = fallback, fb_score
    return final, float(fs), int(n)
