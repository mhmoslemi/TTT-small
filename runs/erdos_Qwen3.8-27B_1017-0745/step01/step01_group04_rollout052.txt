        if sh < best_score:
            best_h = hop.copy()
            best_score = sh
        ho, so = optimize_start(hop, N, m)
        if so < best_score:
            best_h = ho.copy()
            best_score = so

    best_h = project_box_sum(best_h, m)
    c5 = score_exact(best_h, N)
    base_score = score_exact(base, N)
    if not np.isfinite(c5) or (np.isfinite(base_score) and base_score < c5):
        best_h = base.copy()
        c5 = base_score
    if not np.isfinite(c5):
        best_h = np.full(N, 0.5)
        c5 = score_exact(best_h, N)
    return best_h.astype(np.float64), float(c5), int(N)
