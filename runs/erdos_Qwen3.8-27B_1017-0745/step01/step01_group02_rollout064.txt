import numpy as np
from scipy.optimize import minimize

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    if S < -1e-12 or S > len(v) + 1e-12:
        return None
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(v - tau, 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-15:
        for i in np.argsort(1.0 - y):
            d = min(r, 1.0 - y[i])
            y[i] += d
            r -= d
            if r <= 0:
                break
    elif r < -1e-15:
        for i in np.argsort(y):
            d = min(-r, y[i])
            y[i] -= d
            r += d
            if r >= 0:
                break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-9) or np.any(y > 1.0 + 1e-9):
        return None
    if abs(float(np.sum(y)) - S) > 1e-10 * max(1, len(y)):
        return None
    return y

def check_feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(float(np.sum(h)) - 0.5 * n) < 1e-10 * max(1, n))

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    try:
        corr = np.correlate(h, 1.0 - h, mode='full')
        if len(corr) != 2 * n - 1 or not np.all(np.isfinite(corr)):
            return float('inf')
        c = float(np.max(corr) * (2.0 / n))
        return c if np.isfinite(c) else float('inf')
    except Exception:
        return float('inf')

def all_lag_data(h, n):
    dx = 2.0 / n
    nv = 2 * n - 1
    Q = np.zeros(nv)
    G = np.zeros((nv, n))
    for idx, s in enumerate(range(-(n - 1), n)):
        lo_i = max(0, -s)
        hi_i = min(n, n - s)
        for i in range(lo_i, hi_i):
            j = i + s
            Q[idx] += h[i] * (1.0 - h[j])
            G[idx, i] += (1.0 - h[j])
            G[idx, j] -= h[i]
    Q *= dx
    G *= dx
    return Q, G

def mm_obj(z, x, L):
    d = z[:len(x)] - x
    return 0.5 * L * float(d @ d) + z[-1]

def mm_obj_grad(z, x, L):
    d = z[:len(x)] - x
    g = np.zeros(len(z))
    g[:len(x)] = L * d
    g[-1] = 1.0
    return g

def mm_ineq(z, cJ, GJ, n):
    return z[-1] - cJ - GJ @ z[:n]

def mm_ineq_jac(z, cJ, GJ, n):
    J = np.zeros((len(cJ), n + 1))
    J[:, :n] = -GJ
    J[:, n] = 1.0
    return J

def mm_eq(z, n):
    return float(np.sum(z[:n])) - 0.5 * n

def mm_eq_jac(z, n):
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def solve_mm(x, n):
    dx = 2.0 / n
    L = 2.0 * dx
    Q, G = all_lag_data(x, n)
    nv = len(Q)
    if nv <= 128:
        idx = list(range(nv))
    else:
        idx = list(np.argsort(Q)[-min(nv, 96):])
    Qs = Q[idx]
    Gs = G[idx]
    cJ = Qs - Gs @ x
    t0 = float(np.max(Qs)) + 1e-6
    z0 = np.append(x.copy(), t0)
    bounds = [(0.0, 1.0)] * n + [(-np.inf, np.inf)]
    cons = [
        {'type': 'ineq', 'fun': mm_ineq, 'jac': mm_ineq_jac, 'args': (cJ, Gs, n)},
        {'type': 'eq', 'fun': mm_eq, 'jac': mm_eq_jac, 'args': (n,)}
    ]
    opts = {'maxiter': 200, 'ftol': 1e-13, 'disp': False}
    try:
        res = minimize(mm_obj, z0, jac=mm_obj_grad, args=(x, L),
                       method='SLSQP', bounds=bounds, constraints=cons, options=opts)
        y = np.array(res.x[:n], dtype=np.float64)
        yp = project_box_sum(y, 0.5 * n)
        if yp is not None and check_feasible(yp, n):
            return yp
    except Exception:
        pass
    return x.copy()

def mm_descent(h0, n, max_iter=30):
    h = h0.copy()
    c = exact_score(h, n)
    if not np.isfinite(c):
        return h, c
    for _ in range(max_iter):
        y = solve_mm(h, n)
        if np.max(np.abs(y - h)) < 1e-13:
            break
        best_h = h.copy()
        best_c = c
        for k in range(10):
            alpha = 2.0 ** (-k)
            cand = h + alpha * (y - h)
            cp = project_box_sum(cand, 0.5 * n)
            if cp is not None:
                cc = exact_score(cp, n)
                if cc < best_c - 1e-14:
                    best_h = cp.copy()
                    best_c = cc
        if best_c < c - 1e-12:
            h = best_h
            c = best_c
        else:
            break
    return h, c

def _get_parent():
    if 'initial_h_values' in globals():
        ph = np.asarray(globals()['initial_h_values'], dtype=np.float64)
        if ph.ndim == 1 and 2 <= len(ph) <= 999 and np.all(np.isfinite(ph)):

                n = len(ph)
                p = project_box_sum(np.clip(ph, 0.0, 1.0), 0.5 * n)
                if p is not None:
                    return p, n
    n = 45
    return project_box_sum(np.full(n, 0.5), 0.5 * n), n

def build_starts(parent, n, rng):
    starts = [parent.copy(), parent[::-1].copy(), (1.0 - parent).copy()]
    for sd in [0.04, 0.10, 0.20, 0.35, 0.50]:
        p = project_box_sum(parent + sd * rng.standard_normal(n), 0.5 * n)
        if p is not None:
            starts.append(p)
    p = project_box_sum(rng.random(n), 0.5 * n)
    if p is not None:
        starts.append(p)
    for (A, m1, m2, ph) in [(0.3, 1, 2, 0.7), (0.25, 1, 3, 1.3),
                            (0.4, 2, 1, 0.3), (0.35, 1, 4, 2.0)]:
        u = 0.5 + A * np.cos(2.0 * np.pi * m1 * np.arange(n) / n + ph)
        u += 0.12 * np.sin(2.0 * np.pi * m2 * np.arange(n) / n + 2.0 * ph)
        p = project_box_sum(u, 0.5 * n)
        if p is not None:
            starts.append(p)
    for frac in [0.3, 0.4, 0.5, 0.7]:
        u = np.zeros(n)
        k = int(n * frac)
        u[:k] = 1.0 / (2.0 * frac)
        p = project_box_sum(u, 0.5 * n)
        if p is not None:
            starts.append(p)
    unique = []
    for s in starts:
        dup = False
        for u in unique:
            if np.max(np.abs(s - u)) < 1e-8:
                dup = True
                break
        if not dup:
            unique.append(s)
    return unique

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent, n = _get_parent()
    best_h = parent.copy()
    best_c = exact_score(parent, n)

    starts = build_starts(parent, n, rng)
    for h0 in starts:
        h_loc, c_loc = mm_descent(h0, n, max_iter=30)
        if c_loc < best_c:
            best_h = h_loc.copy()
            best_c = c_loc

        h0 = project_box_sum(best_h + sd * rng.standard_normal(n), 0.5 * n)
        if h0 is None:

            best_c = c_loc

        h0 = project_box_sum(best_h + sd * rng.standard_normal(n), 0.5 * n)

            best_c = final_c
    else:
        best_h = parent.copy()
        best_c = exact_score(parent, n)

    return (best_h, best_c, n)
