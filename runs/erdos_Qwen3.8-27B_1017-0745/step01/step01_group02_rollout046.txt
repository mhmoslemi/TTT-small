def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = len(v)
    if not np.all(np.isfinite(v)) or S < -1e-10 or S > n + 1e-10:
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(v - tau, 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-14:
        idx = np.argsort(1.0 - y, kind='stable')
        for i in idx:
            d = min(r, 1.0 - y[i])
            y[i] += d
            r -= d
            if r <= 1e-14:
                break
    elif r < -1e-14:
        idx = np.argsort(y, kind='stable')
        for i in idx:
            d = min(-r, y[i])
            y[i] -= d
            r += d
            if r >= -1e-14:
                break
    if abs(float(np.sum(y)) - S) > 1e-9 * max(1, n):
        return None
    if np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    return y

def exact_score(h):

def lag_q_g(h, s, n):
    dx = 2.0 / n
    lo = max(0, -s)
    hi = min(n, n - s)
    q = 0.0
    g = np.zeros(n)
    for i in range(lo, hi):
        j = i + s
        q += h[i] * (1.0 - h[j])
        g[i] += (1.0 - h[j])
        g[j] -= h[i]
    return dx * q, dx * g

def _mm_obj(z):
    y = z[:-1]
    t = z[-1]
    x = _mm_state['x']
    L = _mm_state['L']
    return 0.5 * L * np.dot(y - x, y - x) + t

def _mm_obj_grad(z):
    y = z[:-1]
    x = _mm_state['x']
    L = _mm_state['L']
    g = np.zeros(len(z))
    g[:-1] = L * (y - x)
    g[-1] = 1.0
    return g

def _mm_ineq(z):
    y = z[:-1]
    t = z[-1]
    d = y - _mm_state['x']
    return t - _mm_state['qJ'] - _mm_state['GJ'] @ d

def _mm_ineq_grad(z):
    n = _mm_state['n']
    J = np.zeros((_mm_state['GJ'].shape[0], n + 1))
    J[:, :-1] = -_mm_state['GJ']
    J[:, -1] = 1.0
    return J

def _mm_eq(z):

def _mm_eq_grad(z):
    g = np.zeros(_mm_state['n'] + 1)

def solve_mm_sub(x, n):
    dx = 2.0 / n
    L = 2.0 * dx
    sqs = list(range(-(n - 1), n))
    qJ = np.zeros(len(sqs))
    GJ = np.zeros((len(sqs), n))
    for k, s in enumerate(sqs):
        qJ[k], GJ[k] = lag_q_g(x, s, n)
    
    _mm_state['x'] = x
    _mm_state['L'] = L
    _mm_state['qJ'] = qJ
    _mm_state['GJ'] = GJ
    _mm_state['n'] = n
    
    t0 = float(np.max(qJ))
    z0 = np.append(x.copy(), t0)
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    cons = [{'type': 'ineq', 'fun': _mm_ineq, 'jac': _mm_ineq_grad},
            {'type': 'eq', 'fun': _mm_eq, 'jac': _mm_eq_grad}]
    opts = {'maxiter': 200, 'ftol': 1e-12, 'disp': False}
    
    try:
        res = minimize(_mm_obj, z0, jac=_mm_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons, options=opts)
        y = np.array(res.x[:n], dtype=np.float64)
        if not np.all(np.isfinite(y)):
            return x.copy()
        y = project_box_sum(y, 0.5 * n)
        if y is None:
            return x.copy()
        return y
    except Exception:
        return x.copy()

def local_mm_descent(h0, n, max_iter=30):
    h = h0.copy()
    c = exact_score(h)
    for _ in range(max_iter):
        y = solve_mm_sub(h, n)
        if np.max(np.abs(y - h)) < 1e-12:
            break
        best_alpha = None
        best_c = c
        for k in range(9):
            alpha = 2.0 ** (-k)
            z = h + alpha * (y - h)
            zp = project_box_sum(z, 0.5 * n)
            if zp is None:
                continue
            cz = exact_score(zp)
            if cz < best_c - 1e-12:
                best_c = cz
                best_alpha = alpha
                best_h = zp.copy()
        if best_alpha is None:
            break
        h = best_h
        c = best_c
    return h, c

import numpy as np
from scipy.optimize import minimize

_mm_st = {}

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = len(v)
    if not np.all(np.isfinite(v)) or S < -1e-10 or S > n + 1e-10:
        return None
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(v - tau, 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-13:
        for i in np.argsort(1.0 - y, kind='stable'):
            d = min(r, 1.0 - y[i]); y[i] += d; r -= d
            if r <= 1e-13: break
    elif r < -1e-13:
        for i in np.argsort(y, kind='stable'):
            d = min(-r, y[i]); y[i] -= d; r += d
            if r >= -1e-13: break
    if abs(float(np.sum(y)) - S) > 1e-9 * max(1, n):
        return None
    if np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    return y

def check_feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(float(np.sum(h)) - 0.5 * n) < 1e-9 * max(1, n))

def exact_score(h):
    n = len(h)
    if not check_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    if len(c) != 2 * n - 1 or not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c) * (2.0 / n))

def lag_q_g(h, s, n):
    dx = 2.0 / n
    lo, hi = max(0, -s), min(n, n - s)
    q, g = 0.0, np.zeros(n)
    for i in range(lo, hi):
        j = i + s
        q += h[i] * (1.0 - h[j])
        g[i] += (1.0 - h[j]); g[j] -= h[i]
    return dx * q, dx * g

def _mm_obj(z):
    y, t = z[:-1], z[-1]
    return 0.5 * _mm_st['L'] * np.dot(y - _mm_st['x'], y - _mm_st['x']) + t

def _mm_obj_grad(z):
    g = np.zeros(len(z))
    g[:-1] = _mm_st['L'] * (z[:-1] - _mm_st['x']); g[-1] = 1.0
    return g

def _mm_ineq(z):
    d = z[:-1] - _mm_st['x']
    return z[-1] - _mm_st['qJ'] - _mm_st['GJ'] @ d

def _mm_ineq_grad(z):
    J = np.zeros((_mm_st['GJ'].shape[0], _mm_st['n'] + 1))
    J[:, :-1] = -_mm_st['GJ']; J[:, -1] = 1.0
    return J

def _mm_eq(z):
    return float(np.sum(z[:-1])) - 0.5 * _mm_st['n']

def _mm_eq_grad(z):
    g = np.zeros(_mm_st['n'] + 1); g[:-1] = 1.0
    return g

def solve_mm_sub(x, n):
    dx = 2.0 / n
    L = 2.0 * dx
    sqs = list(range(-(n - 1), n))
    qJ = np.empty(len(sqs)); GJ = np.empty((len(sqs), n))
    for k, s in enumerate(sqs):
        qJ[k], GJ[k] = lag_q_g(x, s, n)
    _mm_st['x'] = x; _mm_st['L'] = L
    _mm_st['qJ'] = qJ; _mm_st['GJ'] = GJ; _mm_st['n'] = n
    z0 = np.append(x.copy(), float(np.max(qJ)))
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    cons = [{'type': 'ineq', 'fun': _mm_ineq, 'jac': _mm_ineq_grad},
            {'type': 'eq', 'fun': _mm_eq, 'jac': _mm_eq_grad}]
    try:
        res = minimize(_mm_obj, z0, jac=_mm_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-14, 'disp': False})
        y = np.array(res.x[:n], dtype=np.float64)
        if not np.all(np.isfinite(y)):
            return x.copy()
        y = project_box_sum(y, 0.5 * n)
        return y if y is not None else x.copy()
    except Exception:
        return x.copy()

def local_mm_descent(h0, n, max_iter):
    h, c = h0.copy(), exact_score(h0)
    for _ in range(max_iter):
        y = solve_mm_sub(h, n)
        if np.max(np.abs(y - h)) < 1e-12:
            break
        found = False
        for k in range(10):
            a = 2.0 ** (-k)
            zp = project_box_sum(h + a * (y - h), 0.5 * n)
            if zp is None:
                continue
            cz = exact_score(zp)
            if cz < c - 1e-12:
                h, c, found = zp, cz, True
                break
        if not found:
            break
    return h, c

def _get_parent():
    if 'initial_h_values' in globals():
        ph = np.asarray(globals()['initial_h_values'], dtype=np.float64)
        if ph.ndim == 1 and 2 <= len(ph) <= 999 and np.all(np.isfinite(ph)):
            if np.all(ph >= -1e-6) and np.all(ph <= 1.0 + 1e-6):
                n = len(ph)
                p = project_box_sum(np.clip(ph, 0.0, 1.0), 0.5 * n)
                if p is not None:
                    return p, n
    return project_box_sum(np.full(45, 0.5), 22.5), 45

def _build_starts(parent, n, rng):
    starts = [parent.copy(), 1.0 - parent.copy(), parent[::-1].copy()]
    for sd in [0.04, 0.10, 0.20, 0.35]:
        p = project_box_sum(parent + sd * rng.standard_normal(n), 0.5 * n)
        if p is not None:
            starts.append(p)
    p = project_box_sum(rng.random(n), 0.5 * n)
    if p is not None:
        starts.append(p)
    u = 0.5 + 0.35 * np.cos(2.0 * np.pi * np.arange(n) / n + 0.6) + 0.12 * np.sin(4.0 * np.pi * np.arange(n) / n + 1.4)
    p = project_box_sum(u, 0.5 * n)
    if p is not None:
        starts.append(p)
    unique = []
    for s in starts:
        if not any(np.max(np.abs(s - u2)) < 1e-10 for u2 in unique):
            unique.append(s)
    return unique

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent, n = _get_parent()
    best_h, best_c = parent.copy(), exact_score(parent)
    max_iter = max(10, min(35, 1500 // n))
    starts = _build_starts(parent, n, rng)
    for h0 in starts:
        h_new, c_new = local_mm_descent(h0, n, max_iter)
        if c_new < best_c:
            best_h, best_c = h_new.copy(), c_new
    for sd in [0.03, 0.08, 0.18, 0.30]:
        h0 = project_box_sum(best_h + sd * rng.standard_normal(n), 0.5 * n)
        if h0 is not None:
            h_new, c_new = local_mm_descent(h0, n, max(5, max_iter // 2))
            if c_new < best_c:
                best_h, best_c = h_new.copy(), c_new
    best_h = project_box_sum(best_h, 0.5 * n)
    if best_h is not None and check_feasible(best_h, n):
        best_c = exact_score(best_h)
    else:
        best_h, best_c = parent.copy(), exact_score(parent)
    return (best_h, best_c, n)
