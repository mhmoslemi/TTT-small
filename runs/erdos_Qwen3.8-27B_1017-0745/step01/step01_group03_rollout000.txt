def smooth_value_and_grad(h, tau):
    n = len(h)
    Q = np.correlate(h, 1.0-h, 'full')  # length 2n-1, index k corresponds to lag k-(n-1)
    Q_max = np.max(Q)
    exp_vals = np.exp((Q - Q_max) / tau)
    w = exp_vals / np.sum(exp_vals)
    F = tau * (Q_max / tau + np.log(np.sum(exp_vals)))  # = Q_max + tau*log(sum(exp_vals))
    # Gradient
    grad = np.zeros(n)
    for ki in range(2*n-1):
        lag = ki - (n-1)
        wi = w[ki]
        if wi < 1e-15:
            continue
        for r in range(n):
            term = 0.0
            if 0 <= r+lag < n:
                term += (1.0 - h[r+lag])
            if 0 <= r-lag < n:
                term += (-h[r-lag])
            grad[r] += wi * term
    return F, grad

import numpy as np

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if resid > 1e-13 * max(1.0, float(n)):
        order = np.argsort(1.0 - y)
        for i in order:
            add = min(resid, 1.0 - y[i])
            y[i] += add
            resid -= add
            if resid < 1e-15:
                break
    elif resid < -1e-13 * max(1.0, float(n)):
        order = np.argsort(y)
        for i in order:
            sub = min(-resid, y[i])
            y[i] -= sub
            resid += sub
            if -resid < 1e-15:
                break
    if abs(resid) > 1e-12 * max(1.0, float(n)):
        return None
    if np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    y = np.clip(y, 0.0, 1.0)
    r2 = S - np.sum(y)
    if abs(r2) > 1e-14:
        if r2 > 0:
            for i in range(n):
                a = min(r2, 1.0 - y[i])
                y[i] += a
                r2 -= a
                if r2 < 1e-15:
                    break
        else:
            for i in range(n):
                s = min(-r2, y[i])
                y[i] -= s
                r2 += s
                if -r2 < 1e-15:
                    break
    if np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    return y

def smooth_val_grad(h, tau):
    n = len(h)
    Q = np.correlate(h, 1.0 - h, mode='full')
    Qm = float(np.max(Q))
    ev = np.exp((Q - Qm) / tau)
    sv = float(np.sum(ev))
    if sv <= 0.0 or not np.isfinite(sv):
        return None, None
    w = ev / sv
    F = Qm + tau * np.log(sv)
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        wi = w[ki]
        if wi < 1e-15:
            continue
        lag = ki - (n - 1)
        for r in range(n):
            t = 0.0
            if 0 <= r + lag < n:
                t += (1.0 - h[r + lag])
            if 0 <= r - lag < n:
                t += (-h[r - lag])
            grad[r] += wi * t
    if not np.all(np.isfinite(grad)):
        return None, None
    return float(F), grad
