```python
        self.n = n

        t = z[self.n]
        u = y - self.x
        return t + self.beta * self.dx * np.dot(u, u)
```

```python
def compute_lags(x, n):
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.zeros(L)
    G = np.zeros((L, n))
    for d in range(-(n-1), n):
        p0 = max(d, 0)
        q0 = max(-d, 0)
        m = n - abs(d)
        row = d + n - 1
        p = x[p0:p0+m]
        qv = x[q0:q0+m]
        q[row] = dx * np.dot(p, 1.0 - qv)
        G[row, p0:p0+m] += dx * (1.0 - qv)
        G[row, q0:q0+m] -= dx * p
    return q, G
```

```python
def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def compute_lags(x, n):
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.zeros(L)
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        p0 = max(d, 0)
        q0 = max(-d, 0)
        m = n - abs(d)
        row = d + n - 1
        p = x[p0:p0 + m]
        qv = x[q0:q0 + m]
        q[row] = dx * np.dot(p, 1.0 - qv)
        G[row, p0:p0 + m] += dx * (1.0 - qv)
        G[row, q0:q0 + m] -= dx * p
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None, None
    return q, G

class _QP:
    def __init__(self, x, n, beta, q, G):
        self.x = x
        self.n = n
        self.beta = beta
        self.dx = 2.0 / n
        self.q = q
        self.G = G
        self.s = n / 2.0
    def obj(self, z):
        y = z[:self.n]
        t = z[self.n]
        u = y - self.x
        return t + self.beta * self.dx * np.dot(u, u)
    def obj_jac(self, z):
        y = z[:self.n]
        u = y - self.x
        g = np.empty(self.n + 1)
        g[:self.n] = 2.0 * self.beta * self.dx * u
        g[self.n] = 1.0
        return g
    def ineq(self, z):
        y = z[:self.n]
        t = z[self.n]
        return t - self.q - self.G @ (y - self.x)
    def ineq_jac(self, z):
        J = np.empty((2 * self.n - 1, self.n + 1))
        J[:, :self.n] = -self.G
        J[:, self.n] = 1.0
        return J
    def eq(self, z):
        return np.sum(z[:self.n]) - self.s
    def eq_jac(self, z):
        J = np.zeros(self.n + 1)
        J[:self.n] = 1.0
        return J

def mm_step_slsqp(x, n, beta, maxiter):
    q, G = compute_lags(x, n)
    if G is None:
        return None
    prob = _QP(x, n, beta, q, G)
    z0 = np.zeros(n + 1)
    z0[:n] = x
    z0[n] = np.max(q) + 1e-8
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    cons = [
        {'type': 'ineq', 'fun': prob.ineq, 'jac': prob.ineq_jac},
        {'type': 'eq', 'fun': prob.eq, 'jac': prob.eq_jac}
    ]
    try:
        res = minimize(prob.obj, z0, jac=prob.obj_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        z = res.x
        if z is None or z.shape != (n + 1,) or not np.all(np.isfinite(z)):
            return None
        y = z[:n]
        return y
    except Exception:
        return None

def mm_step_subgrad(x, n, beta, maxiter):
    dx = 2.0 / n
    y = x.copy()
    best_y = x.copy()
    best_model = exact_score(x, n)
    for k in range(maxiter):
        Amax = -1e18
        best_d = 0
        for d in range(-(n - 1), n):
            p0 = max(d, 0)
            q0 = max(-d, 0)
            m = n - abs(d)
            u = y - x
            p = x[p0:p0 + m]
            qv = x[q0:q0 + m]
            Au = dx * (np.dot(1.0 - qv, u[p0:p0 + m]) - np.dot(p, u[q0:q0 + m]))
            A0 = dx * np.dot(p, 1.0 - qv)
            val = A0 + Au
            if val > Amax:
                Amax = val
                best_d = d
        u = y - x
        model_val = Amax + beta * dx * np.dot(u, u)
        if model_val < best_model - 1e-14:
            best_model = model_val
            best_y = y.copy()
        p0 = max(best_d, 0)
        q0 = max(-best_d, 0)
        m = n - abs(best_d)
        g = np.zeros(n)
        g[p0:p0 + m] += dx * (1.0 - x[q0:q0 + m])
        g[q0:q0 + m] -= dx * x[p0:p0 + m]
        v = g + 2.0 * beta * dx * u
        nv = np.linalg.norm(v)
        eta = min(0.5 / (nv + 1e-12), 1.0 / (2.0 * beta * dx * np.sqrt(k + 1.0) + 1e-12))
        y_new = project_capped(y - eta * v, n)
        if y_new is None or not np.all(np.isfinite(y_new)):
            break
        if np.max(np.abs(y_new - y)) < 1e-9:
            y = y_new
            break
        y = y_new
    return best_y

def optimize_start(x0, n, n_outer, maxiter_qp, subgrad_iters):
    x = project_capped(x0, n)
    if x is None or not is_feasible(x, n):
        return None, float('inf')
    score = exact_score(x, n)
    beta = 1.0
    rejs = 0
    for it in range(n_outer):
        if rejs >= 5:
            break
        if n <= 128:
            y = mm_step_slsqp(x, n, beta, maxiter_qp)
        else:
            y = mm_step_subgrad(x, n, beta, subgrad_iters)
        if y is None:
            beta = min(beta * 2.0, 64.0)
            rejs += 1
            continue
        y = project_capped(y, n)
        if y is None or not is_feasible(y, n):
            beta = min(beta * 2.0, 64.0)
            rejs += 1
            continue
        if np.max(np.abs(y - x)) < 1e-10:
            break
        best_z, best_s = None, score
        for k in range(5):
            alpha = 1.0 / (2 ** k)
            z = project_capped(x + alpha * (y - x), n)
            if z is None or not is_feasible(z, n):
                continue
            sz = exact_score(z, n)
            if sz < best_s:
                best_s, best_z = sz, z
        if best_z is None or best_s >= score - 1e-12:
            beta = min(beta * 2.0, 64.0)
            rejs += 1
            continue
        x = best_z.copy()
        score = best_s
        rejs = 0
        beta = max(1.0, beta / 1.25)
    return x, exact_score(x, n)

def local_search(x, n, rng, n_trials, sigma0):
    x = x.copy()
    score = exact_score(x, n)
    sigma = sigma0
    for _ in range(n_trials):
        sigma *= 0.97
        if sigma < 1e-7:
            break
        z = x + rng.normal(0, sigma, n)
        zp = project_capped(z, n)
        if zp is None or not is_feasible(zp, n):
            continue
        sz = exact_score(zp, n)
        if sz < score - 1e-12:
            x = zp
            score = sz
    return x, score

def build_starts(fallback, n, rng, count):
    pool = [fallback.copy()]
    cands = [
        fallback[::-1].copy(),
        1.0 - fallback,
        1.0 - fallback[::-1],
        np.full(n, 0.5),
    ]
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    cands.append(blk)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    cands.append(alt)
    for sd in (0.06, 0.15, 0.30):
        cands.append(fallback + rng.normal(0, sd, n))
    cands.append(rng.uniform(0, 1, n))
    tentative = np.zeros((n, 4), dtype=np.float64)
    freqs = np.array([np.pi, 2.0 * np.pi, 0.5 * np.pi, 3.0 * np.pi])
    for j in range(4):
        tentative[:, j] = 0.5 + 0.3 * np.sin(freqs[j] * np.arange(n) / n * 2.0 * np.pi + rng.uniform(0, 6.28))
    for j in range(4):
        cands.append(tentative[:, j])
    for c in cands:
        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-8:
                dup = True
                break
        if not dup:
            pool.append(p)
        if len(pool) >= count:
            break
    return pool[:count]

def penalized_obj(z, n):
    h = project_capped(z, n)
    if h is None or not is_feasible(h, n):
        return 10.0
    s = exact_score(h, n)
    return s if np.isfinite(s) else 10.0

    n = None
    parent = None
    try:

        if p.size >= 1 and p.size <= 999:

            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        if 'n_points' in kwargs:
            try:
                v = int(kwargs['n_points'])
                if 1 <= v <= 999:
                    n = v
            except Exception:
                pass
        if n is None:
            n = 47
    fb = None
    if parent is not None:
        fb = project_capped(parent, n)
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    fb_score = exact_score(fb, n)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        h1 = np.array([0.5])
        return h1, exact_score(h1, 1), 1
    if n <= 64:
        n_starts, n_outer, qp_it, sg_it = 8, 12, 60, 40
    elif n <= 128:
        n_starts, n_outer, qp_it, sg_it = 6, 8, 50, 50
    elif n <= 256:
        n_starts, n_outer, qp_it, sg_it = 4, 6, 40, 60
    else:
        n_starts, n_outer, qp_it, sg_it = 3, 4, 30, 40
    starts = build_starts(fb, n, rng, n_starts)
    gb = fb.copy()
    gs = fb_score
    for st in starts:
        ss = exact_score(st, n)
        if ss < gs - 1e-12:
            gs, gb = ss, st.copy()
    for st in starts:
        try:
            xo, so = optimize_start(st, n, n_outer, qp_it, sg_it)
            if xo is not None and so < gs - 1e-12:
                gs, gb = so, xo.copy()
        except Exception:
            pass
    kb = gb.copy()
    ks = gs
    try:
        kb, ks = local_search(kb, n, rng, 200, 0.05)
    except Exception:
        pass
    if ks < gs - 1e-12:
        gs, gb = ks, kb.copy()
    try:
        res_nm = minimize(penalized_obj, gb, args=(n,), method='Nelder-Mead',
                          options={'maxiter': 300, 'xatol': 1e-10, 'fatol': 1e-12})
        if res_nm.x is not None:
            hn = project_capped(res_nm.x, n)
            if hn is not None and is_feasible(hn, n):
                sn = exact_score(hn, n)
                if sn < gs - 1e-12:
                    gs, gb = sn, hn.copy()
    except Exception:
        pass
    try:
        gb2, gs2 = local_search(gb, n, rng, 150, 0.02)
        if gs2 < gs - 1e-12:
            gs, gb = gs2, gb2.copy()
    except Exception:
        pass
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb.copy()
    fs = exact_score(final, n)
    if fs > fb_score + 1e-12:
        final = fb.copy()
        fs = fb_score
    return final.astype(np.float64), float(fs), int(n)
```
