            {'type': 'eq', 'fun': _eq, 'jac': _eq_j}]
    try:
        res = minimize(_obj, z0, args=(n,), jac=_obj_g, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
    except Exception:
        return None
    if res.x is None or res.x.shape != (n + 1,) or not np.all(np.isfinite(res.x)):
        return None
    y = project_capped(res.x[:n], n)
    return y

def swap_refine(h, n, n_sweeps=6):
    cur = h.copy()
    sc = exact_score(cur, n)
    dx = 2.0 / n
    for _ in range(n_sweeps):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                if cur[i] <= cur[j]:
                    continue
                eps = min(0.5 * (cur[i] - cur[j]), 0.05)
                trial = cur.copy()
                trial[i] -= eps
                trial[j] += eps
                st = exact_score(trial, n)
                if st < sc - 1e-13:
                    cur, sc, improved = trial, st, True
        if not improved:
            break
    return cur, sc

def build_starts(base, n, rng):
    pool = [base.copy()]
    cands = [base[::-1].copy(), 1.0 - base[::-1], np.full(n, 0.5)]
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    cands.append(blk)
    cands.append(np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64))
    cands.append(np.where(np.arange(n) % 2 == 1, 1.0, 0.0).astype(np.float64))
    for sd in (0.06, 0.18, 0.35, 0.60):
        cands.append(base + rng.normal(0, sd, n))
    cands.append(rng.uniform(0, 1, n))
    freq = 1.0 / (n + 1)
    cands.append(0.5 + 0.4 * np.sin(2 * np.pi * np.arange(n) * freq))
    cands.append(0.5 + 0.4 * np.cos(2 * np.pi * 2 * np.arange(n) * freq))
    for c in cands:
        p = project_capped(c, n)
        if p is None:
            continue
        dup = any(np.max(np.abs(p - s)) < 1e-8 for s in pool)
        if not dup:
            pool.append(p)
    return pool[:10]

def optimize_from(h0, n, n_passes=3, maxiter=250):
    cur = h0.copy()
    sc = exact_score(cur, n)
    for _ in range(n_passes):
        y = slsqp_step(cur, n, maxiter)
        if y is None:
            break
        best, bs = cur.copy(), sc
        for k in range(10):
            a = 1.0 / (2 ** k)
            tr = project_capped(cur + a * (y - cur), n)
            if tr is None:
                continue
            ts = exact_score(tr, n)
            if ts < bs:
                bs, best = ts, tr.copy()
        if bs >= sc - 1e-12:
            break
        cur, sc = best, bs
    cur, sc = swap_refine(cur, n, 4)
    return cur, sc

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if parent is not None:
        fallback = project_capped(parent, n)
        if fallback is None:
            fallback = np.full(n, 0.5)
    else:
        fallback = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        fb = np.array([0.5])
        return fb, float(exact_score(fb, 1)), 1
    single_mode = n > 64
    starts = build_starts(fallback, n, rng) if not single_mode else [fallback]
    if single_mode:
        xs, xe = n // 3, 2 * n // 3
        c1 = fallback.copy()
        c1[:xs] += 0.15
        c1[xs:xe] -= 0.10
        c1[xe:] += 0.05
        starts.append(project_capped(c1, n) if project_capped(c1, n) is not None else fallback.copy())
    gb, gs = fallback.copy(), exact_score(fallback, n)
    for st in starts:
        s = exact_score(st, n)
        if s < gs - 1e-12:
            gs, gb = s, st.copy()
    n_passes = 3 if not single_mode else 1
    miter = 250 if not single_mode else 150
    for st in starts:
        try:
            xo, so = optimize_from(st, n, n_passes, miter)
            if so < gs - 1e-12:
                gs, gb = so, xo.copy()
        except Exception:
            pass
    try:
        xo2, so2 = optimize_from(gb, n, 2, 250)
        if so2 < gs - 1e-12:
            gs, gb = so2, xo2.copy()
    except Exception:
        pass
    gb = project_capped(gb, n)
    if gb is None or not is_feasible(gb, n):
        gb = fallback.copy()
    if exact_score(gb, n) > exact_score(fallback, n) + 1e-12:
        gb = fallback.copy()
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = np.full(n, 0.5)
    return final, float(exact_score(final, n)), int(n)
