import numpy as np
import math

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if v.ndim != 1 or not np.all(np.isfinite(v)) or S < 0 or S > n:
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(v - tau, 0.0, 1.0)
    r = S - float(np.sum(y))
    tol = 5e-13 * max(1, n)
    if r > tol:
        for i in range(n):
            d = min(r, 1.0 - y[i])
            y[i] += d
            r -= d
            if r <= tol:
                break
    elif r < -tol:
        for i in range(n):
            d = min(-r, y[i])
            y[i] -= d
            r += d
            if r >= -tol:
                break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-9) or np.any(y > 1.0 + 1e-9):
        return None
    if abs(float(np.sum(y)) - S) > 1e-10 * max(1, n):
        return None
    return y

def check_feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(float(np.sum(h)) - 0.5 * n) < 1e-10 * max(1, n))

def exact_score(h):
    n = len(h)
    if not check_feasible(h, n):
        return float('inf')
    val = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
    return val if np.isfinite(val) else float('inf')

def score_and_corr(h, n):
    if not check_feasible(h, n):
        return (float('inf'), None)
    q = (2.0 / n) * np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(q)):
        return (float('inf'), None)
    return (float(np.max(q)), q)

def compute_affine_hull(lines, lo_t, hi_t):
    # lines: list of (m, c) where function is m*t + c
    # Returns: list of (x_start, m, c) for upper hull
    if not lines:
        return []
    # Sort by slope, then by intercept (descending for same slope)
    lines = sorted(lines, key=lambda lc: (lc[0], -lc[1]))
    # Remove duplicates by slope (keep max intercept)
    unique = []
    for m, c in lines:
        if not unique or m != unique[-1][0]:
            unique.append((m, c))
        # else skip (we already have the max intercept for this slope)
    # Build upper hull
    hull = []  # list of (x_start, m, c)
    INF = float('inf')
    for m, c in unique:
        if not hull:
            hull.append((INF, m, c))  # start at -inf... wait
            hull[0] = (-INF, m, c)
            continue
        # Compute intersection of (m, c) with last line in hull
        lm, lc = hull[-1][1], hull[-1][2]
        if m == lm:
            continue  # shouldn't happen after dedup
        x_cross = (lc - c) / (m - lm)
        # Pop lines that are no longer on the hull
        while len(hull) > 1 and x_cross <= hull[-1][0]:
            hull.pop()
            lm, lc = hull[-1][1], hull[-1][2]
            if m == lm:
                break
            x_cross = (lc - c) / (m - lm)
        if not hull:
            hull.append((-INF, m, c))
        else:
            hull.append((x_cross, m, c))
    # Restrict to [lo_t, hi_t]
    result = []
    for x_s, m, c in hull:
        if x_s < hi_t:  # segment extends to hi_t or next x_s
            result.append((max(x_s, lo_t), m, c))

import numpy as np
import math

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    r = S - float(y.sum())
    tol = 5e-13 * max(1, n)
    if r > tol:
        for i in range(n):
            d = min(r, 1.0 - y[i]); y[i] += d; r -= d
            if r <= tol: break
    elif r < -tol:
        for i in range(n):
            d = min(-r, y[i]); y[i] -= d; r += d
            if r >= -tol: break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-9) or np.any(y > 1.0+1e-9):
        return None
    if abs(float(y.sum()) - S) > 1e-10*max(1,n):
        return None
    return y

def check_feasible(h, n):
    return (len(h)==n and np.all(np.isfinite(h)) and
            np.all(h>=-1e-12) and np.all(h<=1.0+1e-12) and
            abs(float(h.sum())-0.5*n) < 1e-10*max(1,n))

def score_and_corr(h, n):
    if not check_feasible(h, n):
        return (float('inf'), None)
    q = (2.0/n) * np.correlate(h, 1.0-h, mode='full')
    if not np.all(np.isfinite(q)):
        return (float('inf'), None)
    return (float(np.max(q)), q)

def exact_score(h):
    s, _ = score_and_corr(h, len(h))
    return s

def quad_roots(A, B, C):
    if abs(A) < 1e-18:
        if abs(B) < 1e-18: return []
        return [-C/B]
    disc = B*B - 4.0*A*C
    tol = 1e-14 * max(1.0, B*B, abs(4.0*A*C))
    if disc < -tol: return []
    disc = max(0.0, disc)
    sq = math.sqrt(disc)
    q = -0.5*(B + math.copysign(sq, B))
    r = []
    if abs(q) > 1e-18:
        r.append(q/A)
        r.append(C/q)
    else:
        r.append(-B/(2.0*A))
    return [x for x in r if np.isfinite(x)]

def upper_hull(lines):
    if not lines: return []
    lines = sorted(set(lines), key=lambda x: (x[0], -x[1]))
    # dedup by slope
    uniq = []
    for m, c in lines:
        if not uniq or m != uniq[-1][0]:
            uniq.append((m, c))
    hull = []
    for m, c in uniq:
        while hull:
            lm, lc = hull[-1][1], hull[-1][2]
            if m == lm: break
            xc = (lc - c) / (m - lm)
            if xc > hull[-1][0] + 1e-15:
                break
            hull.pop()
        if not hull:
            hull.append([float('-inf'), m, c])
        else:
            hull.append([xc, m, c])
    return hull
