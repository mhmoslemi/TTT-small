    if final is not None:
        fs = exact_score(final, n)
        if np.isfinite(fs) and fs <= best_sc:
            best_h, best_sc = final, fs
    fsc = exact_score(fb_h, n)
    if np.isfinite(fsc) and fsc < best_sc:
        best_h, best_sc = fb_h, fsc
    if not np.isfinite(best_sc):
        best_h, best_sc = uniform.copy(), u_sc
    return best_h, float(best_sc), int(n)
