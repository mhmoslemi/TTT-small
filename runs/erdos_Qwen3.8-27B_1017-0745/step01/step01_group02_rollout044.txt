def project_box_sum(v, S):

    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0

            lo = mid
        else:
            hi = mid

    r = S - float(np.sum(y))

            d = min(r, 1.0 - y[i]); y[i] += d; r -= d
            if r <= 1e-13: break

1. Starting from feasible x
2. Solve LP → get y*
3. Line search: for α in {1, 0.5, 0.25, ..., 2^{-8}}, check h + α(y*-h)
4. Accept the best improving point
5. Repeat until no improvement

This is clean and guaranteed to decrease C at each step (if the LP is solved exactly).

import numpy as np
from scipy.optimize import linprog

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(v - tau, 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-12:
        for i in np.argsort(1.0 - y, kind='stable'):
            d = min(r, 1.0 - y[i]); y[i] += d; r -= d
            if r <= 1e-13: break
    elif r < -1e-12:
        for i in np.argsort(y, kind='stable'):
            d = min(-r, y[i]); y[i] -= d; r += d
            if r >= -1e-13: break
    if not np.all(np.isfinite(y)): return None
    y = np.clip(y, 0.0, 1.0)
    if abs(float(np.sum(y)) - S) > 1e-9*max(1, n): return None
    return y

def check_feas(h, n):
    return (len(h)==n and np.all(np.isfinite(h)) and
            np.all(h>=-1e-12) and np.all(h<=1.0+1e-12) and
            abs(float(np.sum(h))-0.5*n) < 1e-9*max(1,n))

def exact_score(h):
    n = len(h)
    if not check_feas(h, n): return float('inf')
    try:
        c = float(np.max(np.correlate(h, 1.0-h, mode='full'))*(2.0/n))
        return c if np.isfinite(c) else float('inf')
    except Exception:
        return float('inf')

def lag_vg(h, s, n):
    dx = 2.0/n
    lo, hi = max(0, -s), min(n, n-s)
    q = 0.0
    g = np.zeros(n)
    for i in range(lo, hi):
        j = i + s
        q += h[i]*(1.0-h[j])
        g[i] += (1.0-h[j])
        g[j] -= h[i]
    return dx*q, g*dx

def solve_lp(x, n):
    nl = 2*n - 1
    qv = np.zeros(nl)
    GR = np.zeros((nl, n))
    for idx, s in enumerate(range(-(n-1), n)):
        qv[idx], GR[idx] = lag_vg(x, s, n)
    c = np.zeros(n+1); c[n] = 1.0
    A_ub = np.hstack([GR, -1.0*np.ones((nl,1))])
    b_ub = -qv
    A_eq = np.zeros((1, n+1)); A_eq[0,:n] = 1.0
    b_eq = np.array([0.5*n])
    bounds = [(0.0,1.0)]*n + [(None,None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    if res is None or not hasattr(res, 'x') or res.x is None:
        return None
    if not np.all(np.isfinite(res.x)): return None
    y = np.asarray(res.x[:n], dtype=np.float64)
    if np.any(y < -0.05) or np.any(y > 1.05): return None
    return project_box_sum(y, 0.5*n)

def mm_descent(x, n, max_steps=40):
    h = x.copy()
    c = exact_score(h)
    for _ in range(max_steps):
        y = solve_lp(h, n)
        if y is None: break
        d = y - h
        if float(np.max(np.abs(d))) < 1e-11: break
        best_c, best_h = c, h
        for k in range(9):
            a = 2.0**(-k)
            cand = project_box_sum(h + a*d, 0.5*n)
            if cand is None: continue
            cc = exact_score(cand)
            if cc < best_c:
                best_c, best_h = cc, cand
        if best_c >= c - 1e-14: break
        h, c = best_h, best_c
    return h, c
