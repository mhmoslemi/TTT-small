    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-12 * max(1.0, float(n)):
        return None
    if resid > 0:
        for i in range(n):
            add = min(resid, 1.0 - y[i])
            if add <= 0:
                continue
            y[i] += add
            resid -= add

    return y

        starts = [project_box_sum(np.full(n, 0.5), S)]
    fallback_h = parent.copy() if parent is not None else starts[0].copy()
    best_h = starts[0].copy()
    best_score = exact_score(best_h)
    if parent is not None:
        ps = exact_score(parent)
        if ps < best_score:
            best_score = ps
            best_h = parent.copy()
    for h0 in starts:
        h_new = local_optimize(h0, n, S)
        if h_new is not None:
            sc = exact_score(h_new)
            if sc < best_score - 1e-12:
                best_score = sc
                best_h = h_new
    walker = best_h.copy()
    walker_sc = best_score
    for r, amp in enumerate([0.025, 0.06, 0.12, 0.20]):
        noise = rng.standard_normal(n) if r % 2 == 0 else harmonic_noise(n, rng)
        proj = project_box_sum(walker + amp * noise, S)
        if proj is None or float(np.max(np.abs(proj - walker))) < 1e-12:
            continue
        h_new = local_optimize(proj, n, S)
        if h_new is None:
            continue
        sc = exact_score(h_new)
        if not np.isfinite(sc):
            continue
        if sc < best_score - 1e-12:
            best_score = sc
            best_h = h_new
            walker = h_new.copy()
            walker_sc = sc
        else:
            delta = sc - walker_sc
            T = 0.002 * (0.65 ** r)
            if rng.random() < np.exp(max(-700.0, -delta / T)):
                walker = h_new.copy()
                walker_sc = sc
    final_h = project_box_sum(best_h, S)
    if final_h is None or not check_feasible(final_h, n):
        final_h = fallback_h
    final_score = exact_score(final_h)
    if not np.isfinite(final_score):
        final_h = project_box_sum(fallback_h, S)
        final_score = exact_score(final_h)
    return (np.array(final_h, dtype=np.float64), float(final_score), n)
