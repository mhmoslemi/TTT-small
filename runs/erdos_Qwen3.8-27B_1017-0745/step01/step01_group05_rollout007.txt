import numpy as np

def project_onto_k(y, n):
    S = n / 2.0
    if not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    lo = float(y.min()) - 1.0
    hi = float(y.max())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        c = np.clip(y - mid, 0.0, 1.0)
        if c.sum() > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(50):
        diff = S - c.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = c < 1.0 - 1e-12
        else:
            mask = c > 1e-12
        cnt = int(mask.sum())
        if cnt == 0:
            break
        c[mask] += diff / cnt
        np.clip(c, 0.0, 1.0, out=c)
    if abs(c.sum() - S) > 1e-10 or not np.all(np.isfinite(c)):
        return np.full(n, 0.5)
    return c

def exact_score(h):
    n = len(h)
    if not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lag_scores_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(h.sum() - n / 2.0) <= 1e-10)

def compute_pair_coeffs(h, i, j, n):
    dx = 2.0 / n
    b = np.zeros(2 * n - 1)
    a = np.zeros(2 * n - 1)
    nk1 = n - 1
    for k in range(-(n - 1), n):
        idx = k + nk1
        val = 0.0
        r0 = i - k
        if 0 <= r0 < n:
            val += (1.0 - h[r0])
        r0 = j - k
        if 0 <= r0 < n:
            val -= (1.0 - h[r0])
        r0 = i + k
        if 0 <= r0 < n:
            val -= h[r0]
        r0 = j + k
        if 0 <= r0 < n:
            val += h[r0]
        b[idx] = dx * val
        if k == 0:
            a[idx] = -2.0 * dx
        elif k == i - j:
            a[idx] = dx
        elif k == j - i:
            a[idx] = dx
    return b, a

    hi_t = min(1.0 - h[i], h[j])

        return None
    b, a = compute_pair_coeffs(h, i, j, n)
    ts = np.linspace(lo_t, hi_t, 21)
    if lo_t < 0.0 < hi_t:
        ts = np.sort(np.concatenate([ts, [0.0]]))
    scores = np.array([float(np.max(q + b * t + a * t * t)) for t in ts])
