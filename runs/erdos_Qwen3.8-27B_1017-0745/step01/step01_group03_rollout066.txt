            hi = mid

            y[i] += np.sign(resid) * add

def line_mm(x, y, n, S):
    d = y - x
    if np.max(np.abs(d)) < 1e-13: return x.copy()
    nl = 2 * n - 1
    A = np.zeros(nl); B = np.zeros(nl); D = np.zeros(nl)
    for k in range(-(n - 1), n):
        il, ih = max(0, k), min(n - 1, n - 1 + k)
        i = np.arange(il, ih + 1); j = i - k
        kk = k + n - 1
        A[kk] = -np.sum(d[i] * d[j])
        B[kk] = np.sum(d[i] * (1. - x[j]) - x[i] * d[j])
        D[kk] = np.sum(x[i] * (1. - x[j]))
    als = [0.0, 1.0]
    for kk in range(nl):
        if abs(A[kk]) > 1e-15:
            a = -B[kk] / (2. * A[kk])
            if -1e-9 <= a <= 1 + 1e-9: als.append(max(0., min(1., a)))
    for kk in range(nl):
        for ll in range(kk + 1, nl):
            da, db, dc = A[kk] - A[ll], B[kk] - B[ll], D[kk] - D[ll]
            if abs(da) < 1e-15:
                if abs(db) > 1e-15:
                    a = -dc / db
                    if -1e-9 <= a <= 1 + 1e-9: als.append(max(0., min(1., a)))
            else:
                disc = db * db - 4. * da * dc
                if disc >= -1e-12:
                    sq = np.sqrt(max(0., disc))
                    for a in [(-db + sq) / (2 * da), (-db - sq) / (2 * da)]:
                        if -1e-9 <= a <= 1 + 1e-9: als.append(max(0., min(1., a)))
    als = np.unique(np.round(np.array(als), 10))
    best_a, best_v = 0.0, float('inf')
    for s0 in range(0, len(als), 2048):
        ch = als[s0:s0 + 2048]
        vals = A[None, :] * ch[:, None] ** 2 + B[None, :] * ch[:, None] + D[None, :]
        mx = np.max(vals, axis=1)
        for i in range(len(ch)):
            if mx[i] < best_v: best_v, best_a = mx[i], ch[i]
    cands = sorted(set([0., 1., best_a, max(0., best_a - 1e-7), min(1., best_a + 1e-7)]))
    bh, bsc = None, float('inf')
    for a in cands:
        hc = project_box_sum(x + a * d, S)
        if hc is not None and feas(hc, n):
            s = sc(hc, n)
            if s < bsc: bsc, bh = s, hc
    return bh

def solve_lp(c0, rad, n, S):
    lo_b = np.clip(c0 - rad, 0., 1.)
    hi_b = np.clip(c0 + rad, 0., 1.)
    P = n * (n + 1) // 2
    nv = n + P + 1
    rr, cc, dd, rb = [], [], [], []
    for i in range(n):
        for j in range(i, n):
            p = pi(i, j, n)
            zi = n + p
            for ci, cj, cz, rv in [(lo_b[j], lo_b[i], -1., lo_b[i] * lo_b[j]),
                                   (hi_b[j], hi_b[i], -1., hi_b[i] * hi_b[j]),
                                   (-lo_b[j], -hi_b[i], 1., -hi_b[i] * lo_b[j]),
                                   (-hi_b[j], -lo_b[i], 1., -lo_b[i] * hi_b[j])]:
                row = len(rb)
                rr += [row, row, row]
                cc += [i, j, zi]
                dd += [ci, cj, cz]
                rb.append(rv)
    for k in range(-(n - 1), n):
        row = len(rb)
        for i in range(max(0, k), min(n - 1, n - 1 + k) + 1):
            j = i - k
            p = pi(min(i, j), max(i, j), n)
            rr += [row, row]
            cc += [i, n + p]
            dd += [1., -1.]
        rr.append(row); cc.append(nv - 1); dd.append(-1.)
        rb.append(0.)
    er, ec, ed = [], [], []
    for i in range(n):
        er.append(0); ec.append(i); ed.append(1.)
    c = np.zeros(nv); c[nv - 1] = 1.
    bnds = [(lo_b[i], hi_b[i]) for i in range(n)]
    for i in range(n):
        for j in range(i, n):
            bnds.append((lo_b[i] * lo_b[j], hi_b[i] * hi_b[j]))
    bnds.append((0., float(n)))
    try:
        res = linprog(c, A_ub=coo_matrix((dd, (rr, cc)), shape=(len(rb), nv)).tocsc(),
                      b_ub=np.array(rb), A_eq=coo_matrix((ed, (er, ec)), shape=(1, nv)).tocsc(),
                      b_eq=np.array([S]), bounds=bnds, method='highs')
        if res.status == 0 and res.x is not None and len(res.x) == nv:
            h = res.x[:n]
            if np.all(np.isfinite(h)) and np.all(h >= -1e-7) and np.all(h <= 1 + 1e-7):
                h = np.clip(h, 0., 1.)
                return project_box_sum(h, S)
    except Exception:
        pass
    return None

def lp_search(start, n, S):
    cur = project_box_sum(start, S)
    if cur is None or not feas(cur, n): return start
    cs = sc(cur, n)
    rad = 0.20
    fails = 0
    max_att = 8 if n <= 64 else 5
    for _ in range(max_att):
        end = solve_lp(cur, rad, n, S)
        if end is None or not feas(end, n):
            rad *= 0.5; fails += 1
            if rad < 1e-3 and fails >= 2: break
            continue
        if np.max(np.abs(end - cur)) < 1e-12:
            rad *= 0.5; fails += 1
            continue
        hnew = line_mm(cur, end, n, S)
        if hnew is not None and feas(hnew, n):
            hs = sc(hnew, n)
            if hs < cs - 1e-12:
                cur, cs, rad, fails = hnew, hs, min(0.25, 1.2 * rad), 0
                continue
        rad *= 0.5; fails += 1
        if rad < 1e-3 and fails >= 2: break
    return cur

def make_harm(rng, n, amp):
    xs = np.arange(n, dtype=np.float64)
    m1 = int(rng.integers(1, max(2, n // 2)))
    m2 = int(rng.integers(1, max(2, n // 2)))
    phi1, phi2 = rng.uniform(0, 2 * np.pi, 2)
    h = np.cos(2 * np.pi * m1 * xs / n + phi1) + 0.6 * np.cos(2 * np.pi * m2 * xs / n + phi2)
    h -= h.mean()
    rms = np.sqrt(np.mean(h ** 2))
    if rms < 1e-12: return amp * rng.standard_normal(n)
    return amp * h / rms

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    n = 45
    parent = None
    try:
        pv = np.array(initial_h_values, dtype=np.float64)
        if pv.ndim == 1 and 2 <= len(pv) <= 999 and np.all(np.isfinite(pv)):
            if pv.min() >= -1e-9 and pv.max() <= 1 + 1e-9:
                n = len(pv)
                tgt = n / 2.
                if abs(pv.sum() - tgt) < 1e-6 * max(1., float(n)):
                    parent = project_box_sum(np.clip(pv, 0., 1.), tgt)
    except Exception:
        pass
    S = n / 2.0
    if parent is None:
        parent = project_box_sum(np.full(n, 0.5), S)
    best_h = parent.copy()
    best_sc = sc(best_h, n)
    starts = [parent.copy(), parent[::-1].copy(), (1. - parent).copy()]
    starts.append(np.roll(parent, n // 3).copy())
    for amp in [0.04, 0.08, 0.12]:
        ppt = project_box_sum(parent + amp * rng.standard_normal(n), S)
        if ppt is not None: starts.append(ppt)
    for amp in [0.06, 0.10]:
        hnoise = make_harm(rng, n, amp)
        ppt = project_box_sum(parent + hnoise, S)
        if ppt is not None: starts.append(ppt)
    if n <= 4:
        for _ in range(4):
            ppt = project_box_sum(rng.random(n), S)
            if ppt is not None: starts.append(ppt)
    deduped = []
    for s in starts:
        ok = True
        for d in deduped:
            if np.max(np.abs(s - d)) < 1e-10: ok = False; break
        if ok: deduped.append(s)
    for s in deduped:
        ssc = sc(s, n)
        if ssc < best_sc: best_sc, best_h = ssc, s.copy()
    if n <= 96:
        for s in deduped:
            hnew = lp_search(s, n, S)
            if hnew is not None and feas(hnew, n):
                hsc = sc(hnew, n)
                if hsc < best_sc - 1e-12: best_sc, best_h = hsc, hnew.copy()
    amps = [0.005, 0.02, 0.06, 0.12]
    ntr = max(4, min(12, int(np.ceil(1200. / n))))
    for i in range(ntr):
        amp = amps[i % len(amps)]
        if i % 2 == 0:
            pert = best_h + amp * rng.standard_normal(n)
        else:
            pert = best_h + make_harm(rng, n, amp)
        ppt = project_box_sum(pert, S)
        if ppt is not None and feas(ppt, n):
            pps = sc(ppt, n)
            if pps < best_sc - 1e-12: best_sc, best_h = pps, ppt.copy()
            else:
                hnew2 = line_mm(best_h, ppt, n, S)
                if hnew2 is not None and feas(hnew2, n):
                    s2 = sc(hnew2, n)
                    if s2 < best_sc - 1e-12: best_sc, best_h = s2, hnew2.copy()
    final_h = project_box_sum(best_h, S)
    if final_h is None or not feas(final_h, n):
        final_h = parent
    final_sc = sc(final_h, n)
    if not np.isfinite(final_sc):
        final_h = np.full(n, 0.5)
        final_sc = sc(final_h, n)
    return (np.array(final_h, dtype=np.float64), final_sc, n)
