        return None

        m = float(np.median(deltas_up))
        T0 = max(2e-5, min(2e-2, -m / math.log(0.8)))
    else:
        T0 = 1e-3
    return T0, best_c, best_cs

def anneal_cycle(start, n, rng, n_levels, n_prop, sigma0):
    cur = start.copy()
    cur_s = exact_score(cur, n)
    lb = cur.copy()
    lb_s = cur_s
    T0, bc, bcs = calibrate_temp(start, n, rng, sigma0)
    if bcs < lb_s:
        lb_s = bcs
        lb = bc.copy()
    sigma = sigma0
    for j in range(n_levels):
        T = max(1e-7, T0 * (0.55 ** j))
        n_acc = 0
        n_tried = 0
        for _ in range(n_prop):
            prop = propose_feasible(cur, n, sigma, rng)
            n_tried += 1
            if prop is None:
                continue
            ps = exact_score(prop, n)
            if not np.isfinite(ps):
                continue
            if ps < lb_s:
                lb_s = ps
                lb = prop.copy()
            delta = ps - cur_s
            if delta <= 0:
                cur = prop.copy()
                cur_s = ps
                n_acc += 1
            else:
                if delta / T < 745.0:
                    if float(rng.random()) < math.exp(-delta / T):
                        cur = prop.copy()
                        cur_s = ps
                        n_acc += 1
        if n_tried > 0:
            r = n_acc / n_tried
            sigma = max(0.003, min(0.25, sigma * math.exp(0.5 * (r - 0.25))))
        sigma = max(0.003, sigma * 0.75)
        rep = project_box_sum(cur, 0.5 * n)
        if rep is not None and check_feasible(rep, n):
            if float(np.max(np.abs(rep - cur))) < 1e-8:
                cur = rep
                cur_s = exact_score(cur, n)
                if cur_s < lb_s:
                    lb_s = cur_s
                    lb = cur.copy()
    return lb, exact_score(lb, n)

def get_parent():
    n = 45
    ph = globals().get('initial_h_values', None)
    if ph is not None:
        ph = np.asarray(ph, dtype=np.float64)
        if ph.ndim == 1 and 2 <= len(ph) <= 999 and np.all(np.isfinite(ph)):
            if np.all(ph >= -1e-6) and np.all(ph <= 1.0 + 1e-6):
                n = len(ph)
                p = project_box_sum(np.clip(ph, 0.0, 1.0), 0.5 * n)
                if p is not None and check_feasible(p, n):
                    return p, n
    return project_box_sum(np.full(n, 0.5), 0.5 * n), n

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent, n = get_parent()
    best_h = parent.copy()
    best_c5 = exact_score(parent, n)
    starts = build_starts(parent, n, rng)
    for s in starts:
        sc = exact_score(s, n)
        if sc < best_c5:
            best_c5 = sc
            best_h = s.copy()
    scores_list = np.array([exact_score(s, n) for s in starts])
    order = np.argsort(scores_list)
    if n <= 64:
        n_cycles, n_levels, n_prop, sigma0 = 4, 6, 240, 0.08
    elif n <= 256:
        n_cycles, n_levels, n_prop, sigma0 = 3, 5, 90, 0.06
    else:
        n_cycles, n_levels, n_prop, sigma0 = 2, 4, 24, 0.04
    cycle_sd = [0.10, 0.25, 0.18, 0.12]
    for ci in range(n_cycles):
        try:
            if ci == 0:
                cst = starts[int(order[0])].copy()
            elif ci == 1 and len(starts) > 1:
                cst = starts[int(order[1])].copy()
            else:
                sd = cycle_sd[(ci - 2) % len(cycle_sd)]
                cst = None
                for _ in range(4):
                    cst = project_box_sum(best_h + sd * rng.standard_normal(n), 0.5 * n)
                    if cst is not None and check_feasible(cst, n):
                        break
                if cst is None:
                    cst = best_h.copy()
            cs = exact_score(cst, n)
            if cs < best_c5:
                best_c5 = cs
                best_h = cst.copy()
            lb, lbs = anneal_cycle(cst, n, rng, n_levels, n_prop, sigma0)
            if lbs < best_c5:
                best_c5 = lbs
                best_h = lb.copy()
        except Exception:
            pass
    final = project_box_sum(best_h, 0.5 * n)
    if final is not None and check_feasible(final, n):
        best_h = final
        best_c5 = exact_score(best_h, n)
        return (best_h, best_c5, n)
    fb = project_box_sum(parent, 0.5 * n)
    if fb is None:
        fb = np.full(n, 0.5)
    return (fb, exact_score(fb, n), n)
