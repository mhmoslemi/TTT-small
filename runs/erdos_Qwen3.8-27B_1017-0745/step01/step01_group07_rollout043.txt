    b, a = ab_coeffs(h, n, p, r)
    c = q
    ts = np.concatenate([np.linspace(L, U, 17), [0.0]])
    best_t, best_f = 0.0, score
    for t in ts:
        if t < L-1e-14 or t > U+1e-14: continue
        t = max(L, min(U, t))
        f = float(np.max(c + b*t + a*t*t))
        if f < best_f - 1e-14: best_f, best_t = f, t
    radius = (U-L)/16.0
    for _ in range(4):
        lo_r, hi_r = max(L, best_t-radius), min(U, best_t+radius)
        if hi_r - lo_r < 1e-14: break
        for t in np.linspace(lo_r, hi_r, 5):
            f = float(np.max(c + b*t + a*t*t))
            if f < best_f - 1e-14: best_f, best_t = f, t
        radius /= 4.0
    if best_f >= score - 1e-12: return None
    h2 = h.copy(); h2[p] += best_t; h2[r] -= best_t
    if not feas(h2, n):
        h2 = proj(h2, S)
        if h2 is None or not feas(h2, n): return None
    ns = escore(h2, n)
    if ns is None or ns >= score - 1e-12: return None
    nq = elags(h2, n)
    if nq[0] is None: return None
    return (h2, ns, nq[0])

def polish(h, score, q, n, dx, S, rng, budget):
    pairs = []
    for p in range(n-1): pairs.append((p, p+1))
    npairs = min(n*(n-1)//2, budget)
    while len(pairs) < npairs:
        p, r = int(rng.integers(0,n)), int(rng.integers(0,n))
        if p == r: continue
        if p > r: p, r = r, p
        if (p,r) not in pairs: pairs.append((p,r))
        if len(pairs) >= npair*20: break
    rng.shuffle(pairs)
    accepted = 0
    for (p,r) in pairs:
        res = line_search(h, q, score, p, r, n, dx, S)
        if res is not None:
            h, score, q = res; accepted += 1
    return h, score, q, accepted

def anneal_traj(h, score, q, n, dx, S, rng, E, K):
    scale = 0.12
    for e in range(E):
        T = max(5e-5, 3e-3*(0.4**e))
        acc = 0
        for _ in range(K):
            p, r = int(rng.integers(0,n)), int(rng.integers(0,n))
            if p == r: continue
            L = max(-h[p], h[r]-1.0); U = min(1.0-h[p], h[r])
            if U - L <= 1e-12: continue
            t = float(rng.uniform(-scale, scale))
            if t < L or t > U: continue
            h2 = h.copy(); h2[p]+=t; h2[r]-=t
            if not feas(h2, n):
                h2 = proj(h2, S)
                if h2 is None or not feas(h2, n): continue
            ns = escore(h2, n)
            if ns is None: continue
            if ns <= score or (ns - score) / T < 700 and rng.random() < np.exp(-(ns-score)/T):
                h, score = h2, ns
                q, _ = elags(h, n)
                if q is None: q = np.zeros(2*n-1)
                acc += 1
        ratio = acc / max(1, K)
        if ratio < 0.12: scale = min(0.4, 1.25*scale)
        elif ratio > 0.55: scale = max(0.01, 0.75*scale)
        h, score, q, _ = polish(h, score, q, n, dx, S, rng, min(2*n, 200//max(1,n//40)))
    return h, score, q

def run(seed=42, **kwargs):
    if seed is None: seed = 42
    try: seed = int(seed)
    except: seed = 42
    seed = seed % (2**32)
    rng = np.random.default_rng(seed)
    n = None; parent = None
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        try:
            ia = np.asarray(iv, dtype=np.float64)
            if ia.ndim==1 and 2 <= ia.size <= 999 and np.all(np.isfinite(ia)) and np.max(np.abs(ia))<=1e12:
                n = ia.size; S = n/2.0
                if feas(ia, n): parent = ia.copy()
                else:
                    pr = proj(ia, S)
                    if pr is not None and feas(pr, n): parent = pr
        except: pass
    if n is None:
        try: n = int(kwargs.get('n_points', 80))
        except: n = 80
        if n < 8 or n > 999: n = 80
    S = n/2.0; dx = 2.0/n
    h_const = np.full(n, 0.5)
    best_h, best_score = h_const.copy(), 0.5
    bs = escore(h_const, n)
    if bs is not None: best_score = bs
    if parent is not None:
        ps = escore(parent, n)
        if ps is not None and ps < best_score:
            best_score, best_h = ps, parent.copy()
    starts = []
    if parent is not None:
        starts.append(parent.copy())
    idx = np.arange(n, dtype=np.float64)
    u = np.full(n, 0.5)
    for m_ in range(1, min(4, (n-1)//2+1)):
        phi = float(rng.uniform(0, 2*np.pi)); amp = rng.normal(0, 0.12/np.sqrt(m_))
        u += amp*np.cos(2*np.pi*m_*idx/n + phi)
    up = proj(u, S)
    if up is not None and feas(up, n): starts.append(up)
    perm = rng.permutation(n); bv = np.zeros(n)
    bv[perm[:n//2]] = 1.0
    if n % 2 == 1: bv[perm[n//2]] = 0.5
    starts.append(bv.copy())
    rv = np.clip(rng.uniform(0, 1, n), 0, 1); rvp = proj(rv, S)
    if rvp is not None and feas(rvp, n): starts.append(rvp)
    if parent is not None:
        starts.append(parent[::-1].copy())
        starts.append(1.0 - parent.copy())
    cap = 6 if n<=128 else (4 if n<=256 else (3 if n<=512 else 2))
    deduped = []
    for s in starts[:cap+4]:
        if not feas(s, n): continue
        dup = False
        for d in deduped:
            if np.max(np.abs(np.asarray(d)-s)) < 1e-10: dup=True; break
        if not dup: deduped.append(s)
        if len(deduped) >= cap: break
    E = 5 if n<=128 else (4 if n<=256 else (3 if n<=512 else 2))
    K = min(1000, max(80, 80000//n))
    for s in deduped:
        sc = escore(s, n)
        if sc is not None and sc < best_score: best_score, best_h = sc, s.copy()
        try:
            q, _ = elags(s, n)
            if q is None: continue
            th, ts_, tq = anneal_traj(s.copy(), sc, q, n, dx, S, rng, E, K)
            if ts_ < best_score - 1e-12: best_score, best_h = ts_, th.copy()
        except: pass
    try:
        q, _ = elags(best_h, n)
        if q is not None:
            pb = 4 if n<=128 else (3 if n<=256 else (2 if n<=512 else 1))
            for _ in range(pb):
                bh, bs2, bq, acct = polish(best_h.copy(), best_score, q, n, dx, S, rng, min(2*n,300//max(1,n//40)))
                if bs2 < best_score - 1e-12:
                    best_score, best_h, q = bs2, bh, bq
                elif acct == 0: break
    except: pass
    if not feas(best_h, n):
        best_h = proj(best_h, S)
    if best_h is None or not feas(best_h, n):
        best_h = h_const.copy()
        best_score = escore(best_h, n)
        if best_score is None: best_score = 0.5
    fs = escore(best_h, n)
    if fs is not None: best_score = fs
    return best_h, best_score, n
