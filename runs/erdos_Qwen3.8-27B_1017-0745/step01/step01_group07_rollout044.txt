Let me think about the DC decomposition:

For lag d, with valid indices I_d = {i : 0 <= i < n, 0 <= i+d < n}:
q_d(h) = dx * sum_{i in I_d} h[i] * (1 - h[i+d])

Let u = h[i], v = h[i+d]. Then:
u*(1-v) = u - u*v

For the DC decomposition:
-u*v = ((u-v)^2 - (u+v)^2)/4

So:
q_d(y) = dx * sum [y[i] - y[i]*y[j]]
       = dx * sum [y[i] + (y[i]-y[j])^2/4 - (y[i]+y[j])^2/4]

Let A_d(y) = dx * sum [y[i] + (y[i]-y[j])^2/4] (convex - has convex quadratic)
Let B_d(y) = dx/4 * sum (y[i]+y[j])^2 (convex - quadratic form)

q_d(y) = A_d(y) - B_d(y)

Tangent to B_d at anchor a:
B_d(y) = dx/4 * sum (y[i]+y[j])^2
∇B_d(y) at a: dx/4 * sum 2(a[i]+a[j]) * (gradient of (y[i]+y[j]))
= dx/2 * sum (a[i]+a[j]) * (e_i + e_j)

T_d(y; a) = B_d(a) + ∇B_d(a)^T (y-a)
= dx/4 * sum (a[i]+a[j])^2 + dx/2 * sum (a[i]+a[j]) * ((y[i]+y[j]) - (a[i]+a[j]))
= dx/4 * sum (a[i]+a[j])^2 + dx/2 * sum (a[i]+a[j])*(y[i]+y[j]) - dx/2 * sum (a[i]+a[j])^2
= dx/2 * sum (a[i]+a[j])*(y[i]+y[j]) - dx/4 * sum (a[i]+a[j])^2

So M_d(y; a) = A_d(y) - T_d(y; a)
= dx * sum [y[i] + (y[i]-y[j])^2/4] - dx/2 * sum (a[i]+a[j])*(y[i]+y[j]) + dx/4 * sum (a[i]+a[j])^2
= dx * sum [y[i] + (y[i]-y[j])^2/4 - (a[i]+a[j])*(y[i]+y[j])/2 + (a[i]+a[j])^2/4]

This is convex because:
- y[i] is linear
- (y[i]-y[j])^2/4 is convex
- -(a[i]+a[j])*(y[i]+y[j])/2 is linear in y (a is fixed)
- (a[i]+a[j])^2/4 is constant

So M_d is convex. And q_d(y) <= M_d(y; a) for all y, with equality at y=a.

This is a convex quadratic constraint (left side is convex in y, right side is affine). CVXPY can handle this as an SOCP constraint.

```python
import numpy as np
import cvxpy as cp
from scipy.optimize import minimize

def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):
        return None
    if np.max(np.abs(z)) > 1e9:
        return None
    n = len(z)
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(80):
        mid = lo + 0.5 * (hi - lo)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = lo + 0.5 * (hi - lo)
    x = np.clip(z - lam, 0.0, 1.0)
    r = S - np.sum(x)
    if r > 0:
        for i in range(n):
            if r <= 0: break
            a = min(r, 1.0 - x[i])
            x[i] += a; r -= a
    elif r < 0:
        for i in range(n):
            if r >= 0: break
            a = min(-r, x[i])
            x[i] -= a; r += a
    if abs(np.sum(x) - S) > 1e-10 or not np.all(np.isfinite(x)):
        return None
    return np.clip(x, 0.0, 1.0)

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-10)

def exact_score(h, n):
    try:
        vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if not np.all(np.isfinite(vals)) or len(vals) != 2 * n - 1:
            return None
        return float(np.max(vals))
    except Exception:
        return None

def solve_dc_subproblem(anchor, n, rho, eta, dx):
    S = n / 2.0
    y = cp.Variable(n)
    t = cp.Variable()
    constraints = [y >= 0, y <= 1, cp.sum(y) == S,
                   y - anchor <= rho, anchor - y <= rho]
    for d in range(-(n-1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        y0 = y[lo:hi]
        y1 = y[lo+d:hi+d]
        s = anchor[lo:hi] + anchor[lo+d:hi+d]
        lhs = dx * (cp.sum(y0) + 0.25 * cp.sum_squares(y0 - y1)
                    - 0.5 * cp.sum(s * (y0 + y1)) + 0.25 * cp.sum_squares(s))
        constraints.append(lhs <= t)
    objective = cp.Minimize(t + 0.5 * eta * cp.sum_squares(y - anchor))
    prob = cp.Problem(objective, constraints)
    if not prob.is_dcp():
        return None
    for solver in [cp.ECOS, cp.CVXOPT]:
        try:
            prob.solve(solver=solver, verbose=False, max_iters=500,
                       abstol=1e-8, reltol=1e-8, feastol=1e-8)
            if prob.status in ('optimal', 'optimal_inaccurate') and y.value is not None:
                val = np.asarray(y.value, dtype=np.float64)
                if val.ndim == 1 and len(val) == n and np.all(np.isfinite(val)):
                    return val
        except Exception:
            continue
    return None
```

```python
def optimize_dc(start, n, dx, S, best_h, best_score):
    x = start.copy()
    f = exact_score(x, n)
    if f is None:
        return best_h, best_score
    rho = 0.20
    eta = 0.20 * dx
    max_attempts = 8 if n <= 128 else 5
    fail_count = 0
    no_imp_count = 0
    for _ in range(max_attempts):
        cand = solve_dc_subproblem(x, n, rho, eta, dx)
        if cand is None:
            rho = max(rho * 0.5, 0.005)
            fail_count += 1
            if fail_count >= 2 or rho < 0.006:
                break
            continue
        proj = project_capped_simplex(cand, S)
        if proj is None or not is_feasible(proj, n):
            rho = max(rho * 0.5, 0.005)
            fail_count += 1
            if fail_count >= 2:
                break
            continue
        # Line search
        best_trial = None
        best_trial_score = f
        for j in range(7):
            alpha = 0.5 ** j
            trial = x + alpha * (proj - x)
            if not is_feasible(trial, n):
                trial = project_capped_simplex(trial, S)
                if trial is None or not is_feasible(trial, n):
                    continue
            sc = exact_score(trial, n)
            if sc is not None and sc < best_trial_score - 1e-12:
                best_trial = trial
                best_trial_score = sc
            if sc is not None and sc < float(best_score) - 1e-12:
                best_h = trial.copy()
                best_score = sc
        if best_trial is not None:
            x = best_trial.copy()
            f = best_trial_score
            fail_count = 0
            no_imp_count = 0
        else:
            rho = max(rho * 0.5, 0.005)
            eta = max(eta * 0.5, 0.02 * dx)
            no_imp_count += 1
            if no_imp_count >= 3 or (rho < 0.006 and eta < 0.03 * dx):
                break
    return best_h, best_score
```

import numpy as np
import cvxpy as cp

def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):
        return None
    n = len(z)
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z))
    for _ in range(80):
        mid = lo + 0.5 * (hi - lo)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(z - (lo + 0.5*(hi-lo)), 0.0, 1.0)
    r = S - np.sum(x)
    for i in range(n):
        if abs(r) < 1e-14: break
        if r > 0:
            a = min(r, 1.0 - x[i]); x[i] += a; r -= a
        else:
            a = min(-r, x[i]); x[i] -= a; r += a
    if abs(np.sum(x) - S) > 1e-10 or not np.all(np.isfinite(x)):
        return None
    return np.clip(x, 0.0, 1.0)

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0+1e-12)
            and abs(np.sum(h) - n/2.0) <= 1e-10)

def exact_score(h, n):
    try:
        v = np.correlate(h, 1.0-h, mode='full') * (2.0/n)
        if len(v) != 2*n-1 or not np.all(np.isfinite(v)):
            return None
        return float(np.max(v))
    except Exception:
        return None

def solve_dc(anchor, n, rho, eta, dx):
    S = n / 2.0
    y = cp.Variable(n)
    t = cp.Variable()
    cons = [y >= 0, y <= 1, cp.sum(y) == S,
            y - anchor <= rho, anchor - y <= rho]
    for d in range(-(n-1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        y0 = y[lo:hi]
        y1 = y[lo+d:hi+d]
        s = anchor[lo:hi] + anchor[lo+d:hi+d]
        c = 0.25 * float(np.dot(s, s))
        lhs = dx * (cp.sum(y0) + 0.25*cp.sum_squares(y0 - y1)
                    - 0.5*cp.sum(s * (y0 + y1)) + c)
        cons.append(lhs <= t)
    prob = cp.Problem(cp.Minimize(t + 0.5*eta*cp.sum_squares(y - anchor)), cons)
    if not prob.is_dcp():
        return None
    for solver in [cp.ECOS, cp.CVXOPT]:
        try:
            prob.solve(solver=solver, verbose=False, max_iters=500,
                       abstol=1e-8, reltol=1e-8, feastol=1e-8)
            if prob.status in ('optimal', 'optimal_inaccurate') and y.value is not None:
                val = np.asarray(y.value, dtype=np.float64)
                if val.ndim == 1 and len(val) == n and np.all(np.isfinite(val)):
                    return val
        except Exception:
            continue
    return None

def optimize_dc(x0, n, dx, S):

    f = exact_score(x, n)
    if f is None:
        return x, f
    rho, eta = 0.20, 0.20*dx
    max_att = 8 if n <= 128 else 5
    fail, noimp = 0, 0
    for _ in range(max_att):
        sol = solve_dc(x, n, rho, eta, dx)
        if sol is None:
            rho = max(rho*0.5, 0.005); fail += 1
            if fail >= 2 or rho < 0.006: break
            continue
        pr = project_capped_simplex(sol, S)
        if pr is None or not is_feasible(pr, n):
            rho = max(rho*0.5, 0.005); fail += 1
            if fail >= 2: break
            continue
        best_trial, best_f = None, f
        for j in range(7):
            tr = x + (0.5**j) * (pr - x)
            if not is_feasible(tr, n):

        if best_trial is not None:
            dec = f - best_f
            x, f = best_trial, best_f
            fail = noimp = 0
            if dec > 1e-10:
                rho = min(0.40, max(rho*1.25, rho))
        else:
            rho = max(rho*0.5, 0.005)
            eta = max(eta*0.5, 0.02*dx)
            noimp += 1
            if noimp >= 3 or (rho < 0.006 and eta < 0.03*dx): break
    return x.copy(), f

    const = np.full(n, 0.5)

            if np.max(np.abs(s - d)) < 1e-9: ok = True; break
