        for j in order:
            d = helmert(n, j)
            h2, s2 = line_min(h, d, n)
            if s2 < s - 1e-13:
                h, s = h2, s2
                nimp += 1
        if nimp == 0:
            break
    return h, s

def solve_sdp(n, dx, S, fb_score):
    try:
        import cvxpy as cp
    except ImportError:
        return None
    if n > 50:
        return None
    try:
        hv = cp.Variable(n)
        X = cp.Variable((n, n), symmetric=True)
        t = cp.Variable()
        blk = cp.hstack([cp.Constant([[1.0]]), hv.T.reshape(1, n)])
        blk = cp.vstack([blk, cp.hstack([hv.reshape(n, 1), X])])
        cons = [blk >> 0, hv >= 0, hv <= 1, cp.sum(hv) == S]
        for i in range(n):
            cons.append(X[i, i] <= hv[i])
            cons.append(X[i, i] >= 2 * hv[i] - 1)
            for j in range(i + 1, n):
                cons.append(X[i, j] <= hv[i])
                cons.append(X[i, j] <= hv[j])
                cons.append(X[i, j] >= 0)
                cons.append(X[i, j] >= hv[i] + hv[j] - 1)
            cons.append(cp.sum(X[i, :]) == S * hv[i])
        for k in range(-(n - 1), n):
            li, hi = max(0, k), min(n, n + k)
            idx = np.arange(li, hi)
            cons.append(t >= dx * cp.sum(hv[idx] - X[idx, idx - k]))
        prob = cp.Problem(cp.Minimize(t), cons)
        for sn in ['MOSEK', 'CVXOPT']:
            try:
                kw = {}
                if sn == 'CVXOPT':
                    kw['options'] = {'abstol': 1e-6, 'reltol': 1e-6, 'ftol': 1e-6}
                prob.solve(solver=sn, **kw)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    hval = np.ravel(np.array(hv.value, dtype=np.float64))
                    Xval = np.array(X.value, dtype=np.float64)
                    tv = float(t.value)
                    if np.all(np.isfinite(hval)) and np.all(np.isfinite(Xval)) and tv < fb_score + 1.5:
                        return hval, Xval, tv
            except Exception:
                continue
    except Exception:
        pass
    return None

def sdp_candidates(hrel, Xrel, fb, n, rng):
    cands = []
    hp, _ = project_capped(hrel, n)
    cands.append(hp)
    for al in [0.25, 0.5, 0.75]:
        bl = fb + al * (hp - fb)
        cands.append(bl)
    V = Xrel - np.outer(hrel, hrel)
    V = 0.5 * (V + V.T)
    c1 = np.ones(n)
    V = V - np.outer(V @ c1, c1) / n
    V = V - np.outer(c1, V @ c1) / n
    V = 0.5 * (V + V.T)
    try:
        eigv, eigw = np.linalg.eigh(V)
        if eigv.min() > -1e-3 * max(1.0, np.max(np.abs(eigv))):
            eigv = np.clip(eigv, 0.0, None)
            maxdiag = np.max(np.abs(np.diag(V)))
            if maxdiag > 0.25:
                eigv *= 0.25 / maxdiag
            Lmask = eigv > 1e-12
            L = eigv[:, Lmask] * eigw[:, Lmask].T
            for _ in range(16):
                z = rng.standard_normal(L.shape[1])
                cands.append(hp + L @ z)
            modes = L.shape[1]
            for mi in range(min(4, modes)):
                cands.append(hp + 0.5 * L[:, mi] * eigw[:, mi])
                cands.append(hp - 0.5 * L[:, mi] * eigw[:, mi])
    except Exception:
        pass
    for _ in range(8):
        z = rng.standard_normal(n)
        z -= z.mean()
        rms = np.sqrt(np.mean(z ** 2))
        if rms > 1e-10:
            z /= rms
        cands.append(fb + 0.05 * z)
        cands.append(fb + 0.15 * z)
    return cands

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = globals().get('initial_h_values', None)
    n = 45
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.ndim == 1 and 2 <= len(p) <= 999 and np.all(np.isfinite(p)) and np.max(np.abs(p)) < 1e6:
            n = len(p)
            fb, _ = project_capped(p, n)
        else:
            fb = np.full(n, 0.5)
    else:
        npk = kwargs.get('n_points', None)
        if isinstance(npk, int) and 2 <= npk <= 999:
            n = npk
        fb = np.full(n, 0.5)
    if not check_feasible(fb, n):
        fb = np.full(n, 0.5)
    fb_s = exact_score(fb, n)
    best_h, best_s = fb.copy(), fb_s
    pool = []
    def record(hc):
        nonlocal best_h, best_s
        if hc is None:
            return
        hc, ok = project_capped(hc, n)
        if not ok:
            return
        s = exact_score(hc, n)
        if s < best_s - 1e-12:
            best_s, best_h = s, hc.copy()
        dup = False
        for (_, ps) in pool:
            if np.max(np.abs(hc - _)) < 1e-7:
                if ps > s:
                    _[:] = hc
                dup = True
                break
        if not dup and len(pool) < 16:
            pool.append((hc.copy(), s))
    record(fb)
    if n <= 50:
        dx = 2.0 / n
        S = n / 2.0
        sd = solve_sdp(n, dx, S, fb_s)
        if sd is not None:
            cands = sdp_candidates(sd[0], sd[1], fb, n, rng)
            for cc in cands:
                record(cc)
    pool.sort(key=lambda x: x[1])
    starts = [fb.copy(), (1.0 - fb).copy(), fb[::-1].copy()]
    for sc in [0.05, 0.15, 0.30]:
        starts.append(fb + sc * rng.standard_normal(n))
    starts.append(rng.uniform(0, 1, n))
    for st in starts[:6]:
        ph, ps = polish(st, n, rng, 2)
        record(ph)
    if best_h is not fb:
        bh, bs = polish(best_h, n, rng, 3)
        record(bh)
    for hc, _ in pool[:4]:
        ph, ps = polish(hc, n, rng, 3)
        record(ph)
    record(best_h)
    fh, ok = project_capped(best_h, n)
    if not ok or not check_feasible(fh, n):
        fh = fb.copy()
    fs = exact_score(fh, n)
    if fs > fb_s + 1e-12:
        fh, fs = fb.copy(), fb_s
    fh, _ = project_capped(fh, n)
    fs = exact_score(fh, n)
    if not check_feasible(fh, n):
        fh = np.full(n, 0.5)
        fs = exact_score(fh, n)
    return fh.astype(np.float64), float(fs), int(n)
