        if rejects >= 6:
            break
        if n <= 128:
            y = convex_qp_step(x, n, beta, qp_maxiter)
        else:
            y = matfree_step(x, n, beta, 50)
        if y is None:
            beta = min(beta * 2.0, 64.0)
            rejects += 1
            continue
        y = project_capped(y, n)
        if y is None or not is_feasible(y, n):
            beta = min(beta * 2.0, 64.0)
            rejects += 1
            continue
        disp = np.max(np.abs(y - x))
        if disp < 1e-11:
            break
        best_y, best_s = None, score
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            z = project_capped(x + alpha * (y - x), n)
            if z is None or not is_feasible(z, n):
                continue
            s = exact_score(z, n)
            if s < best_s:
                best_s, best_y = s, z
        if best_y is None or best_s >= score - 1e-13:
            beta = min(beta * 2.0, 64.0)
            rejects += 1
            continue
        x = best_y.copy()
        score = best_s
        rejects = 0
        beta = max(1.0, beta / 1.4)
        beta = min(beta, 64.0)
    return x, exact_score(x, n)

def build_starts(fallback, n, rng, count):
    pool = []
    def _add(v):
        if v is None:
            return
        for s in pool:
            if np.max(np.abs(v - s)) < 1e-8:
                return
        pool.append(v)
    _add(fallback.copy())
    _add(project_capped(fallback[::-1], n))
    _add(project_capped(1.0 - fallback, n))
    _add(project_capped(1.0 - fallback[::-1], n))
    _add(full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    _add(project_capped(blk, n))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    _add(project_capped(alt, n))
    for sd in [0.05, 0.15, 0.35, 0.6]:
        _add(project_capped(fallback + rng.normal(0, sd, n), n))
    _add(project_capped(rng.uniform(0, 1, n), n))
    return pool[:count]

def full(n, v):
    return np.full(n, v, dtype=np.float64)

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
        if not (1 <= n <= 999):
            n = 47
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if parent is not None:
        fallback = project_capped(parent, n)
    else:
        fallback = full(n, 0.5)
    if fallback is None:
        fallback = full(n, 0.5)
    if n == 1:
        fb = project_capped(np.array([0.5]), 1)
        s = exact_score(fb, 1)
        return fb, s, 1
    if n <= 64:
        n_starts, max_outer, qp_it = 8, 12, 100
    elif n <= 128:
        n_starts, max_outer, qp_it = 5, 8, 80
    elif n <= 256:
        n_starts, max_outer, qp_it = 3, 5, 0
    else:
        n_starts, max_outer, qp_it = 2, 3, 0
    starts = build_starts(fallback, n, rng, n_starts)
    fb_score = exact_score(fallback, n)
    gb, gs = fallback.copy(), fb_score
    scored = []
    for st in starts:
        s = exact_score(st, n)
        scored.append((s, st))
    scored.sort(key=lambda t: t[0])
    scored_starts = [scored[0][1]] + [t[1] for t in scored[1:] if t[0] < 1.0]
    scored_starts = scored_starts[:n_starts]
    if fallback not in scored_starts:
        scored_starts[0] = fallback
    else:
        scored_starts = [fallback] + [s2 for s2 in scored_starts if np.max(np.abs(s2 - fallback)) > 1e-8]
    for st in scored_starts:
        if not is_feasible(st, n):
            continue
        s = exact_score(st, n)
        if s < gs - 1e-13:
            gs, gb = s, st.copy()
        try:
            xo, so = optimize_start(st, n, max_outer, qp_it)
            so = exact_score(xo, n)
            if so < gs - 1e-13:
                gs, gb = so, xo.copy()
        except Exception:
            pass
    gb = project_capped(gb, n)
    if gb is None or not is_feasible(gb, n):
        gb = fallback
    final_score = exact_score(gb, n)
    if final_score > fb_score + 1e-12:
        gb, final_score = fallback.copy(), fb_score
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = full(n, 0.5)
        final_score = exact_score(final, n)
    return final, float(final_score), int(n)
