Actually numpy's correlate: `np.correlate(a, v)` where a has length M and v has length N (N<=M) computes:
c[k] = sum_n a[n+k] * conj(v[n])

For mode='full', k ranges from -(N-1) to M-1, giving M+N-1 values.

So `np.correlate(h, 1-h, mode='full')` where both have length n:
c[k] = sum_i h[i+k] * (1-h[i]) for valid i, k from -(n-1) to n-1

This is the same as sum_i h[i] * (1-h[i-k]) by reindexing.

The overlap integral for lag k is: sum_i h[i] * (1-h[i+k]) * dx (for the discrete approximation).

So the max over all lags of this sum times dx = 2/n gives C5.

Since np.correlate gives us all these values in the 'full' output, taking max and multiplying by 2/n is correct.

                next_sc[i] = sc
                g_arch.append(gen_pop[i].copy())
                sF.append(Fi)
                sCR.append(Cri)
                if sc < gb_s:
                    gb_h, gb_s = proj.copy(), sc
            else:
                next_pop[i] = gen_pop[i]
                next_sc[i] = gen_sc[i]
        pop, scores = next_pop, next_sc
        while len(g_arch) > P:
            del g_arch[int(rng.integers(len(g_arch)))]
        archive = g_arch
        if sF:
            aF = np.array(sF)
            mu_F = max(0.05, min(1.0, 0.9 * mu_F + 0.1 * float(np.sum(aF**2) / np.sum(aF))))
        if sCR:
            mu_CR = max(0.0, min(1.0, 0.9 * mu_CR + 0.1 * float(np.mean(sCR))))
        if gb_s < saved_gb - 1e-12 * max(1.0, abs(saved_gb)):
            no_imp = 0
        else:
            no_imp += 1
        if no_imp >= stag_lim and rs_cnt < max_rs:
            oR = np.argsort(scores)
            new_pop = [gb_h.copy()]
            for ri in oR:
                if len(new_pop) >= max(2, P // 4):
                    break
                if all(np.max(np.abs(pop[ri] - x)) >= 1e-10 for x in new_pop):
                    new_pop.append(pop[ri].copy())
            sds = [0.04, 0.12, 0.30]
            sd = sds[rs_cnt % 3]
            fill_t = int(0.7 * (P - len(new_pop)))
            att = 0
            while len(new_pop) < P and att < 30 * P:
                if len(new_pop) - 1 < fill_t:
                    c = gb_h + sd * rng.standard_normal(n)
                else:
                    c = rng.random(n)
                p = project_box_sum(c, S)
                if p is not None and check_feasible(p, n) and _diverse(p, new_pop):
                    sc = exact_score(p, n)
                    if np.isfinite(sc) and sc < gb_s:
                        gb_h, gb_s = p.copy(), sc
                    new_pop.append(p)
                att += 1
            while len(new_pop) < P:
                new_pop.append(gb_h.copy())
            pop, scores = new_pop[:P], np.array([exact_score(new_pop[i], n) for i in range(P)])
            archive, mu_F, mu_CR, no_imp, rs_cnt = [], 0.55, 0.70, 0, rs_cnt + 1
        elif no_imp >= stag_lim:
            break
    fh = project_box_sum(gb_h, S)
    if fh is not None and check_feasible(fh, n):
        fh = fh
        fs = exact_score(fh, n)
    else:
        fh, fs = fb_h, fb_s
    if not np.isfinite(fs) or fs > fb_s:
        fh, fs = fb_h.copy(), fb_s
    fh = np.clip(fh, 0.0, 1.0)
    return (fh, fs, n)
