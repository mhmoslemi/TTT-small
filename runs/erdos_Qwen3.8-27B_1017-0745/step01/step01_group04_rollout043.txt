def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)) or np.max(np.abs(z)) > 1e100:
        return np.full(N, 0.5)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - float(np.sum(h))
    if r > 1e-14:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
            if r <= 1e-14:
                break
    elif r < -1e-14:
        for i in range(N):
            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
            if r >= -1e-14:

def slsqp_polish(h0, N, m, rng):
    from scipy.optimize import minimize
    h0 = project_box_sum(h0, m)
    N2 = 2 * N - 1
    def _obj(z):
        return z[N]
    def _obj_j(z):
        g = np.zeros(N + 1)
        g[N] = 1.0
        return g
    def _eq(z):
        return np.array([np.sum(z[:N]) - m])
    def _eq_j(z):
        J = np.zeros((1, N + 1))
        J[0, :N] = 1.0
        return J
    def _ineq(z):
        h = z[:N]
        q = np.correlate(h, 1.0 - h, mode='full')
        return z[N] - q
    def _ineq_j(z):
        h = z[:N]
        J = np.zeros((N2, N + 1))
        for s in range(-(N - 1), N):
            ii = s + N - 1
            if s >= 0:
                L = N - s
                J[ii, :L] = 1.0 - h[s:s + L]
                J[ii, s:N] -= h[:L]
            else:
                d2 = -s
                L = N - d2
                J[ii, d2:] = 1.0 - h[:L]
                J[ii, :L] -= h[d2:]
        J[:, N] = 1.0
        return J
    q0 = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = max(float(np.max(q0)), 0.0)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * N + [(0.0, float(N))]
    cons = [
        {'type': 'eq', 'fun': _eq, 'jac': _eq_j},
        {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_j},
    ]
    try:
        res = minimize(_obj, z0, jac=_obj_j, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-12, 'disp': False})
        if res.x is not None and len(res.x) == N + 1 and np.all(np.isfinite(res.x[:N])):
            return project_box_sum(res.x[:N], m)
    except Exception:
        pass
    return h0
