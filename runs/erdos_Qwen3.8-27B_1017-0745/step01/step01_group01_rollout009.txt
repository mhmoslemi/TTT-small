def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= len(iv) <= 999 and np.all(np.isfinite(iv)) and np.all(iv >= -1e-8) and np.all(iv <= 1.0 + 1e-8):
            n, parent = len(iv), iv.copy()
    except Exception:
        pass
    if n is None:
        n = 47
        try:
            n = int(kwargs.get('n_points', 47))
            if not (1 <= n <= 999):
                n = 47
        except Exception:
            pass
    try:
        sd = int(seed)
        if sd < 0:
            sd = 42
    except Exception:
        sd = 42
    rng = np.random.default_rng(sd)

    uniform = np.full(n, 0.5)
    fallback = project_capped(uniform, n)
    if fallback is None:
        fallback = uniform.copy()
    fb_score = exact_score(fallback, n)

    if parent is not None:
        par = project_capped(parent, n)
        if par is not None:
            ps = exact_score(par, n)
            if np.isfinite(ps) and ps < fb_score:
                fallback, fb_score = par.copy(), ps

    starts = []
    add_start(starts, fallback, n)
    add_start(starts, fallback[::-1].copy(), n)
    add_start(starts, (1.0 - fallback).copy(), n)
    add_start(starts, (1.0 - fallback[::-1]).copy(), n)
    add_start(starts, uniform.copy(), n)
    if parent is not None:
        add_start(starts, parent + rng.normal(0, 0.025, n), n)
        add_start(starts, parent + rng.normal(0, 0.10, n), n)
    idx = np.arange(n)
    add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * idx / n), n)
    add_start(starts, 0.5 + 0.45 * np.cos(4.0 * np.pi * idx / n), n)
    add_start(starts, rng.uniform(0, 1, n), n)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    add_start(starts, blk, n)

    n_starts = 8
    max_iter = 250
    try:
        n_starts = max(1, min(int(kwargs.get('starts', n_starts)), 10))
    except Exception:
        pass
    try:
        max_iter = max(0, min(int(kwargs.get('max_iter', max_iter)), 300))
    except Exception:
        pass
    if n > 64 and n <= 128:
        n_starts = min(n_starts, 5)
        max_iter = min(max_iter, 160)
    elif n > 128 and n <= 256:
        n_starts = min(n_starts, 3)
        max_iter = min(max_iter, 80)
    elif n > 256:
        max_iter = 0

    starts = starts[:n_starts]

    best, best_score = fallback.copy(), fb_score
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_score - 1e-12:
            best, best_score = st.copy(), ss

    if n <= 256 and max_iter > 0:
        bounds = [(0.0, 1.0)] * (n + 1)
        cons = [
            {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac},
            {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac}
        ]
        for st in starts:
            st_score = exact_score(st, n)
            if not np.isfinite(st_score):
                continue
            z0 = np.concatenate([st, [min(1.0, st_score + 1e-8)]])
            try:
                res = minimize(slsqp_obj, z0, args=(n,), method='SLSQP',
                               jac=slsqp_obj_jac, bounds=bounds,
                               constraints=cons,
                               options={'ftol': 1e-10, 'maxiter': max_iter})
            except Exception:
                continue
            if res.x is None or len(res.x) != n + 1 or not np.all(np.isfinite(res.x)):
                continue
            h_cand = project_capped(res.x[:n], n)
            if h_cand is None:
                continue
            sc = exact_score(h_cand, n)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best, best_score = h_cand.copy(), sc

    if n <= 64:
        n_sweeps = 1
        try:
            n_sweeps = max(0, min(int(kwargs.get('polish_sweeps', n_sweeps)), 2))
        except Exception:
            pass
        x = best.copy()
        for _ in range(n_sweeps):
            if not polish_sweep(x, n):
                break
        ps = exact_score(x, n)
        if np.isfinite(ps) and ps < best_score - 1e-12:
            best, best_score = x.copy(), ps

    final = project_capped(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final, fs = fallback.copy(), fb_score
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final, float(fs), int(n)
