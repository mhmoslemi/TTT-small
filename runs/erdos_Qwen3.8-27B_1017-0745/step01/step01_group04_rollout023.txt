```python
import numpy as np

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    z = z - np.median(z)
    lo, hi = z.min() - 1.0, z.max()
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    if r > 0:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
            if r <= 1e-14:
                break
    elif r < 0:
        for i in range(N):
            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1 + 1e-10:
        return np.full(N, 0.5)
    if abs(h.sum() - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)

def exact_score(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / N)

def compute_lags(h, N):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.copy()
```

```python
            best_Q = Qm
            best_t = t_mid
    return best_t

def pair_descent(h, N, m, rng, n_sweeps):
    corr = np.correlate(h, 1.0 - h, mode='full')
    curr_score = float(np.max(corr) * 2.0 / N)
    n_pairs = N * (N - 1) // 2
    for sweep in range(n_sweeps):
        accepted = False
        idx = np.arange(N)
        pairs = []
        for i in range(N):
            for j in range(i + 1, N):
                pairs.append((i, j))
        rng.shuffle(pairs)
        for p, r in pairs:
            t = pair_line_search(h, corr, p, r, N)
            if t is None or abs(t) < 1e-10:
                continue
            pt = h[p] + h[r]
            hp = np.clip(h[p] + t, max(0.0, pt - 1.0), min(1.0, pt))
            hr = pt - hp
            if hr < -1e-10 or hr > 1 + 1e-10:
                continue
            h_new = h.copy()
            h_new[p] = hp
            h_new[r] = hr
            s_new = exact_score(h_new, N)
            if s_new < curr_score - 1e-12:
                h = h_new
                corr = np.correlate(h, 1.0 - h, mode='full')
                curr_score = s_new
                accepted = True
        if not accepted:
            break
    return h, curr_score

def make_fourier_start(N, rng, n_modes, amp):
    z = np.full(N, 0.5)
    idx = np.arange(N)
    for rr in range(1, n_modes + 1):
        A = rng.standard_normal()
        B = rng.standard_normal()
        z += (amp / rr) * (A * np.sin(2.0 * np.pi * rr * idx / N) + B * np.cos(2.0 * np.pi * rr * idx / N))
    return z

def anneal(h, N, m, rng, L):
    best_h = h.copy()
    best_score = exact_score(h, N)
    curr_h = h.copy()
    curr_score = best_score
    for j in range(L):
        f = j / max(1, L - 1)
        sigma = 0.12 * (0.025 / 0.12) ** f
        T = 0.002 * (0.0001 / 0.002) ** f
        if T < 1e-15:
            T = 1e-15
        p = int(rng.integers(0, N))
        r = int(rng.integers(0, N - 1))
        if r >= p:
            r += 1
        t_lo = max(-curr_h[p], curr_h[r] - 1.0, -sigma)
        t_hi = min(1.0 - curr_h[p], curr_h[r], sigma)
        if t_hi - t_lo <= 1e-12:
            continue
        t = float(rng.uniform(t_lo, t_hi))
        pt = curr_h[p] + curr_h[r]
        hp = np.clip(curr_h[p] + t, max(0.0, pt - 1.0), min(1.0, pt))
        hr = pt - hp
        if hr < -1e-10 or hr > 1 + 1e-10:
            continue
        h_new = curr_h.copy()
        h_new[p] = hp
        h_new[r] = hr
        s_new = exact_score(h_new, N)
        if not np.isfinite(s_new):
            continue
        delta = s_new - curr_score
        if delta <= 0 or (delta / T < 700 and rng.random() < np.exp(-delta / T)):
            curr_h = h_new
            curr_score = s_new
            if s_new < best_score - 1e-12:
                best_h = h_new.copy()
                best_score = s_new
    return best_h, best_score

def run(seed=42, **kwargs):
    N = None
    m = None
    raw = globals().get('initial_h_values', None)
    if raw is not None:
        raw = np.asarray(raw, dtype=np.float64)
        if raw.ndim == 1 and raw.size > 1 and np.all(np.isfinite(raw)):
            N = raw.size
            m = N / 2.0
    if N is None:
        N = kwargs.get('n_points', 70)
        if not isinstance(N, (int, np.integer)) or isinstance(N, bool) or N < 2:
            N = 70
        m = N / 2.0
    if N == 1:
        h = np.array([0.5])
        return h, exact_score(h, 1), 1
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    if raw is not None and raw.size == N:
        best = project_box_sum(raw, m)
    else:
        best = np.full(N, 0.5)
    best = project_box_sum(best, m)
    best_score = exact_score(best, N)
    baseline = best.copy()
    baseline_score = best_score
    starts = [best.copy(), np.full(N, 0.5)]
    comp = 1.0 - best
    starts.append(project_box_sum(comp, m))
    for sc in [0.08, 0.20]:
        starts.append(project_box_sum(best + sc * rng.standard_normal(N), m))
    starts.append(project_box_sum(make_fourier_start(N, rng, 4, 0.15), m))
    starts.append(project_box_sum(0.5 + 0.3 * rng.uniform(size=N), m))
    dedup = []
    for s in starts:
        if not is_feasible(s, N, m):
            continue
        dup = False
        for d in dedup:
            if np.max(np.abs(s - d)) <= 1e-10:
                dup = True
                break
        if not dup:
            dedup.append(s)
    starts = dedup[:6]
    for h0 in starts:
        s0 = exact_score(h0, N)
        if s0 < best_score - 1e-12:
            best = h0.copy()
            best_score = s0
        h_opt, s_opt = pair_descent(h0, N, m, rng, 2)
        if s_opt < best_score - 1e-12:
            best = h_opt.copy()
            best_score = s_opt
    h_a, s_a = anneal(best, N, m, rng, 160)
    h_a, s_a = pair_descent(h_a, N, m, rng, 1)
    if s_a < best_score - 1e-12:
        best = h_a.copy()
        best_score = s_a
    h_s2 = project_box_sum(best + 0.15 * rng.standard_normal(N), m)
    h_a2, s_a2 = anneal(h_s2, N, m, rng, 120)
    h_a2, s_a2 = pair_descent(h_a2, N, m, rng, 1)
    if s_a2 < best_score - 1e-12:
        best = h_a2.copy()
        best_score = s_a2
    for sc in [0.05, 0.02]:
        hop = project_box_sum(best + sc * rng.standard_normal(N), m)
        h_h, s_h = pair_descent(hop, N, m, rng, 1)
        if s_h < best_score - 1e-12:
            best = h_h.copy()
            best_score = s_h
    best = project_box_sum(best, m)
    c5 = exact_score(best, N)
    baseline = project_box_sum(baseline, m)
    c5_b = exact_score(baseline, N)
    if c5_b < c5:
        best = baseline
        c5 = c5_b
    if not np.isfinite(c5):
        best = np.full(N, 0.5)
        c5 = exact_score(best, N)
    return best, float(c5), int(N)
```
