import numpy as np
from scipy.optimize import minimize

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    if S < 0 or S > n:
        return None
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - float(np.sum(y))
    if abs(r) > 1e-14:
        if r > 0:
            for i in np.argsort(1.0 - y):
                d = min(r, 1.0 - y[i])
                y[i] += d; r -= d
                if r <= 1e-14: break
        else:
            for i in np.argsort(y):
                d = min(-r, y[i])
                y[i] -= d; r += d
                if r >= -1e-14: break
    if not np.all(np.isfinite(y)) or abs(float(np.sum(y)) - S) > 1e-9 * max(1, n):
        return None
    return np.clip(y, 0.0, 1.0)

def check_feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(float(np.sum(h)) - 0.5 * n) < 1e-10 * max(1, n))

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    try:
        c = np.correlate(h, 1.0 - h, mode='full')
        if len(c) != 2 * n - 1 or not np.all(np.isfinite(c)):
            return float('inf')
        return float(np.max(c)) * (2.0 / n)
    except Exception:
        return float('inf')

def lag_val_grad(h, s, n):
    dx = 2.0 / n
    lo_i = max(0, -s)
    hi_i = min(n, n - s)
    q = 0.0
    g = np.zeros(n)
    for i in range(lo_i, hi_i):
        j = i + s
        q += h[i] * (1.0 - h[j])
        g[i] += (1.0 - h[j])
        g[j] -= h[i]
    return dx * q, dx * g

_MM = {}

def mm_obj(z):
    x, L, n = _MM['x'], _MM['L'], _MM['n']
    return 0.5 * L * np.dot(z[:n] - x, z[:n] - x) + z[n]

def mm_obj_grad(z):
    x, L, n = _MM['x'], _MM['L'], _MM['n']
    g = np.zeros(n + 1)
    g[:n] = L * (z[:n] - x)
    g[n] = 1.0
    return g

def mm_ineq(z):
    n = _MM['n']
    return z[n] - _MM['qJ'] - _MM['GJ'] @ (z[:n] - _MM['x'])

def mm_ineq_jac(z):
    n = _MM['n']
    J = np.zeros((_MM['qJ'].shape[0], n + 1))
    J[:, :n] = -_MM['GJ']
    J[:, n] = 1.0
    return J

def mm_eq(z):
    return float(np.sum(z[:_MM['n']])) - 0.5 * _MM['n']

def mm_eq_jac(z):
    g = np.zeros(_MM['n'] + 1)
    g[:_MM['n']] = 1.0
    return g

def solve_mm(x, n):
    dx = 2.0 / n
    L = 2.0 * dx
    nv = 2 * n - 1
    q_all = np.zeros(nv)
    G_all = np.zeros((nv, n))
    for idx in range(nv):

        q, g = lag_val_grad(x, s, n)
        q_all[idx] = q

    else:

        _MM['qJ'], _MM['GJ'] = q_all[idx], G_all[idx]

    bounds = [(0.0, 1.0)] * n + [(-1e10, 1e10)]

    except Exception:

    return x.copy()
