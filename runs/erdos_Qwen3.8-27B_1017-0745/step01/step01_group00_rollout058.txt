    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def obj_fun(z, n):
    return z[n]

def obj_jac(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def ineq_fun(z, n):
    h = z[:n]
    t = z[n]
    q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return t - q

def ineq_jac(z, n):
    h = z[:n]
    L = 2 * n - 1
    dx = 2.0 / n
    G = np.zeros((L, n + 1))
    for d in range(-(n - 1), n):
        row = d + n - 1
        p0 = max(d, 0)
        q0 = max(-d, 0)
        m = n - abs(d)
        g = np.zeros(n)
        g[p0:p0 + m] += dx * (1.0 - h[q0:q0 + m])
        g[q0:q0 + m] -= dx * h[p0:p0 + m]
        G[row, :n] = -g
        G[row, n] = 1.0
    return G

def eq_fun(z, n):
    return np.array([np.sum(z[:n]) - n / 2.0])

def eq_jac(z, n):
    J = np.zeros((1, n + 1))
    J[0, :n] = 1.0
    return J

def epigraph_solve(start, n):
    cur = project_capped(start, n)
    if cur is None:
        return None
    score = exact_score(cur, n)
    if score == float('inf'):
        return None
    for _ in range(2):
        q = np.correlate(cur, 1.0 - cur, mode='full') * (2.0 / n)
        t0 = np.max(q) + max(1e-9, 1e-7 * max(1.0, abs(np.max(q))))
        z0 = np.append(cur.copy(), t0)
        ci = {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}
        ce = {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
        bounds = [(0.0, 1.0)] * n + [(None, None)]
        try:
            res = minimize(obj_fun, z0, method='SLSQP', jac=obj_jac,
                           constraints=[ci, ce], bounds=bounds,
                           args=(n,), options={'ftol': 1e-10, 'maxiter': 220})
        except Exception:
            break
        if res.x is None or res.x.shape != (n + 1,) or not np.all(np.isfinite(res.x)):
            break
        y = project_capped(res.x[:n], n)
        if y is None:
            break
        best_z, best_s = cur.copy(), score
        for alpha in (1.0, 0.5, 0.25, 0.125, 0.0625):
            zt = project_capped(cur + alpha * (y - cur), n)
            if zt is None:
                continue
            sz = exact_score(zt, n)
            if sz < best_s - 1e-12:
                best_s, best_z = sz, zt.copy()
        if best_s < score - 1e-12:
            cur, score = best_z, best_s
        else:
            break
    return cur

def active_grad(h, n):
    q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    idx = int(np.argmax(q))
    d = idx - (n - 1)
    p0, q0 = max(d, 0), max(-d, 0)
    m = n - abs(d)
    dx = 2.0 / n
    g = np.zeros(n)
    g[p0:p0 + m] += dx * (1.0 - h[q0:q0 + m])
    g[q0:q0 + m] -= dx * h[p0:p0 + m]
    return g

def matrix_free_search(start, n, n_iter):
    h = project_capped(start, n)
    if h is None:
        return None
    score = exact_score(h, n)
    if score == float('inf'):
        return None
    dx = 2.0 / n
    eta = min(0.2 / dx, 2.0 / dx)
    fails = 0
    for _ in range(n_iter):
        if fails >= 8 or eta < 1e-8 / dx:
            break
        g = active_grad(h, n)
        z = project_capped(h - eta * g, n)
        if z is None:
            eta *= 0.5
            fails += 1
            continue
        v = z - h
        if np.max(np.abs(v)) < 1e-10:
            eta *= 0.5
            fails += 1
            continue
        best_z, best_s = None, score
        for alpha in (1.0, 0.5, 0.25, 0.125, 0.0625):
            zt = project_capped(h + alpha * v, n)
            if zt is None:
                continue
            sz = exact_score(zt, n)
            if sz < best_s - 1e-12:
                best_s, best_z = sz, zt.copy()
        if best_z is None:
            eta *= 0.5
            fails += 1
            continue
        h, score = best_z, best_s
        fails = 0
        eta = min(1.2 * eta, 2.0 / dx)
    return h

def dedup_add(pool, v):
    if v is None:
        return
    for s in pool:
        if np.max(np.abs(v - s)) < 1e-8:
            return
    pool.append(v)

def build_starts(fall, n, rng):
    pool = []
    dedup_add(pool, fall.copy())
    dedup_add(pool, project_capped(fall[::-1], n))
    dedup_add(pool, project_capped(1.0 - fall[::-1], n))
    dedup_add(pool, np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    dedup_add(pool, project_capped(blk, n))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    dedup_add(pool, project_capped(alt, n))
    dedup_add(pool, project_capped(fall + 0.08 * rng.standard_normal(n), n))
    dedup_add(pool, project_capped(fall + 0.20 * rng.standard_normal(n), n))
    dedup_add(pool, project_capped(fall + 0.35 * rng.standard_normal(n), n))
    dedup_add(pool, project_capped(rng.uniform(0, 1, n), n))
    return pool

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 0 < p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None:
        try:
            v = int(kwargs.get('n_points', 47))
            n = v if 1 <= v <= 999 else 47
        except Exception:
            n = 47
    if not (1 <= n <= 999):
        n = 47
    if parent is not None:
        fall = project_capped(parent, n)
    else:
        fall = np.full(n, 0.5)
    if fall is None:
        fall = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        return np.array([0.5]), 0.25, 1
    starts = build_starts(fall, n, rng)
    if n <= 64:
        n_keep = min(len(starts), 8)
    elif n <= 128:
        n_keep = min(len(starts), 3)
    elif n <= 256:
        n_keep = min(len(starts), 2)
    else:
        n_keep = 1
    gb = fall.copy()
    gs = exact_score(fall, n)
    for st in starts:
        s = exact_score(st, n)
        if s < gs - 1e-12:
            gs, gb = s, st.copy()
    if n <= 64:
        for st in starts[:n_keep]:
            try:
                res = epigraph_solve(st, n)
            except Exception:
                res = None
            if res is not None:
                s = exact_score(res, n)
                if s < gs - 1e-12:
                    gs, gb = s, res.copy()
    else:
        n_iter = 45 if n <= 128 else (30 if n <= 256 else 18)
        for st in starts[:n_keep]:
            try:
                res = matrix_free_search(st, n, n_iter)
            except Exception:
                res = None
            if res is not None:
                s = exact_score(res, n)
                if s < gs - 1e-12:
                    gs, gb = s, res.copy()
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fall
    fs = exact_score(final, n)
    if fs > gs + 1e-12:
        final = fall
        fs = exact_score(fall, n)
    if fs == float('inf'):
        final = np.full(n, 0.5)
        fs = exact_score(final, n)
    return final, float(fs), int(n)
