        return float('inf')
    return float(np.max(q * (2.0 / n)))

def lag_grad(h, n, d):
    dx = 2.0 / n
    g = np.zeros(n)
    p0 = max(d, 0)
    q0 = max(-d, 0)
    m = n - abs(d)
    g[p0:p0 + m] += dx * (1.0 - h[q0:q0 + m])
    g[q0:q0 + m] -= dx * h[p0:p0 + m]
    return g

def obj_fn(z, n):
    return z[n], np.r_[np.zeros(n), 1.0]

def ineq_fn(z, n):
    h = z[:n]
    t = z[n]
    q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    val = t - q
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        G[d + n - 1, :] = lag_grad(h, n, d)
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return val, J

def eq_fn(z, n):
    return np.sum(z[:n]) - n / 2.0, np.r_[np.ones(n), 0.0]

def slsqp_pass(cur, n, maxiter):
    q = np.correlate(cur, 1.0 - cur, mode='full') * (2.0 / n)
    t0 = float(np.max(q)) + max(1e-10, 1e-8)
    z0 = np.append(cur.copy(), t0)
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    cons = [{'type': 'ineq', 'fun': ineq_fn, 'args': (n,)},
            {'type': 'eq', 'fun': eq_fn, 'args': (n,)}]
    try:
        res = minimize(obj_fn, z0, args=(n,), method='SLSQP',
                       jac=True, bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
    except Exception:
        return None
    if res.x is None or res.x.shape[0] != n + 1 or not np.all(np.isfinite(res.x)):
        return None
    return project_capped(res.x[:n], n)

def epigraph_optimize(h0, n, n_passes=2, maxiter=220):
    cur = project_capped(h0, n)
    if cur is None:
        return None
    best = cur.copy()
    best_s = exact_score(best, n)
    for _ in range(n_passes):
        y = slsqp_pass(cur, n, maxiter)
        if y is None or not is_feasible(y, n):
            break
        best_y, best_sy = None, best_s
        for k in range(6):
            alpha = 1.0 / (2 ** k)
            z = project_capped(cur + alpha * (y - cur), n)
            if z is None or not is_feasible(z, n):
                continue
            sz = exact_score(z, n)
            if sz < best_sy:
                best_sy, best_y = sz, z
        if best_y is None or best_sy >= best_s - 1e-12:
            break
        cur, best_s = best_y.copy(), best_sy
        if best_s < exact_score(best, n):
            best = cur.copy()
    return best

def coord_polish(h, n, n_iter=80):
    cur = h.copy()
    cur_s = exact_score(cur, n)
    for _ in range(n_iter):
        q = np.correlate(cur, 1.0 - cur, mode='full') * (2.0 / n)
        d = int(np.argmax(q)) - (n - 1)
        g = lag_grad(cur, n, d)
        idx = int(np.argmax(np.abs(g)))
        improved = False
        for delta in (-0.04, -0.015, -0.006, 0.006, 0.015, 0.04):
            trial = cur.copy()
            trial[idx] += delta
            trial = project_capped(trial, n)
            if trial is None:
                continue
            ts = exact_score(trial, n)
            if ts < cur_s - 1e-14:
                cur, cur_s = trial, ts
                improved = True
                break
        if not improved:
            break
    return cur, cur_s

def subgrad_search(h0, n, n_iter=45):
    cur = project_capped(h0, n)
    if cur is None:
        return None
    cur_s = exact_score(cur, n)
    dx = 2.0 / n
    eta = min(0.2 / dx, 2.0 / dx)
    fails = 0
    for _ in range(n_iter):
        q = np.correlate(cur, 1.0 - cur, mode='full') * dx
        d = int(np.argmax(q)) - (n - 1)
        g = lag_grad(cur, n, d)
        z = project_capped(cur - eta * g, n)
        if z is None:
            eta *= 0.5
            fails += 1
            if fails >= 8 or eta < 1e-8 / dx:
                break
            continue
        v = z - cur
        if np.max(np.abs(v)) < 1e-10:
            eta *= 0.5
            fails += 1
            if fails >= 8 or eta < 1e-8 / dx:
                break
            continue
        best_y, best_sy = None, cur_s
        for k in range(6):
            alpha = 1.0 / (2 ** k)
            t = project_capped(cur + alpha * v, n)
            if t is None or not is_feasible(t, n):
                continue
            ts = exact_score(t, n)
            if ts < best_sy:
                best_sy, best_y = ts, t
        if best_y is None or best_sy >= cur_s - 1e-12:
            eta *= 0.5
            fails += 1
            if fails >= 8 or eta < 1e-8 / dx:
                break
            continue
        cur, cur_s = best_y.copy(), best_sy
        fails = 0
        eta = min(1.2 * eta, 2.0 / dx)
    return cur

def build_starts(fallback, n, rng):
    pool = [fallback.copy()]
    for v in [project_capped(fallback[::-1], n),
              project_capped(1.0 - fallback[::-1], n),
              np.full(n, 0.5)]:
        if v is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(v - s)) < 1e-8:
                dup = True
                break
        if not dup:
            pool.append(v)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    pb = project_capped(blk, n)
    if pb is not None:
        pool.append(pb)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    pa = project_capped(alt, n)
    if pa is not None:
        pool.append(pa)
    pn1 = project_capped(fallback + 0.08 * rng.standard_normal(n), n)
    if pn1 is not None:
        pool.append(pn1)
    pn2 = project_capped(fallback + 0.20 * rng.standard_normal(n), n)
    if pn2 is not None:
        pool.append(pn2)
    pr = project_capped(rng.uniform(0, 1, n), n)
    if pr is not None:
        pool.append(pr)
    return pool[:8]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    fallback = project_capped(parent, n) if parent is not None else np.full(n, 0.5)
    if fallback is None or not is_feasible(fallback, n):
        fallback = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    fs = exact_score(fallback, n)
    gb, gs = fallback.copy(), fs
    if n == 1:
        return gb.astype(np.float64), float(gs), 1
    starts = build_starts(fallback, n, rng)
    for st in starts:
        s = exact_score(st, n)
        if s < gs - 1e-12:
            gs, gb = s, st.copy()
    if n <= 64:
        for st in starts:
            try:
                res = epigraph_optimize(st, n, 2, 220)
            except Exception:
                continue
            if res is None:
                continue
            s = exact_score(res, n)
            if s < gs - 1e-12:
                gs, gb = s, res.copy()
    elif n <= 128:
        for st in starts[:3]:
            try:
                res = subgrad_search(st, n, 45)
            except Exception:
                continue
            if res is not None:
                s = exact_score(res, n)
                if s < gs - 1e-12:
                    gs, gb = s, res.copy()
    else:
        try:
            res = subgrad_search(fallback, n, 30)
        except Exception:
            res = None
        if res is not None:
            s = exact_score(res, n)
            if s < gs - 1e-12:
                gs, gb = s, res.copy()
    # Coordinate polish on best
    try:
        pc, pc_s = coord_polish(gb, n, 100)
        if pc_s < gs - 1e-12:
            gs, gb = pc_s, pc
    except Exception:
        pass
    fb_s = exact_score(fallback, n)
    if gs > fb_s + 1e-12:
        gb, gs = fallback.copy(), fb_s
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = np.full(n, 0.5)
    gs = exact_score(final, n)
    return final.astype(np.float64), float(gs), int(n)
