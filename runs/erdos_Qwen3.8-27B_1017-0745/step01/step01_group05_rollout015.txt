            mask = c > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        c[mask] += diff / cnt
        np.clip(c, 0.0, 1.0, out=c)
    if abs(c.sum() - S) > 1e-10 or not np.all(np.isfinite(c)):

    return c

def exact_score(h):
    n = len(h)
    if not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(h.sum() - n/2.0) <= 1e-10)

def pair_coeffs(h, i, j, n):
    L = 2*n - 1
    b = np.zeros(L)
    b[:i+1] += 1.0 - h[:i+1][::-1]
    b[:j+1] -= 1.0 - h[:j+1][::-1]
    b[:n-i] -= h[i:]
    b[:n-j] += h[j:]
    a = np.zeros(L)
    a[0] = -2.0
    a[abs(i-j)] = 1.0
    return b, a

def line_search(h, i, j, q, n, rng):
    dx = 2.0 / n
    lo = max(-h[i], h[j]-1.0)
    hi = min(1.0-h[i], h[j])
    if hi - lo < 1e-12:
        return None
    b, a = pair_coeffs(h, i, j, n)
    # Grid points
    grid = np.linspace(lo, hi, 15)
    if abs(0.0 - lo) > 1e-12 and abs(0.0 - hi) > 1e-12:
        grid = np.sort(np.concatenate([grid, [0.0]]))
    best_t = 0.0
    best_f = float(np.max(q) * dx)
    for t in grid:
        val = np.max((q + b*t + a*t*t) * dx)
        if val < best_f - 1e-14:
            best_f = val
            best_t = t
    # Refine around best
    for _ in range(3):
        span = (hi - lo) / 15.0
        lo_r = max(lo, best_t - span)
        hi_r = min(hi, best_t + span)
        if hi_r - lo_r < 1e-14:
            break
        for t in np.linspace(lo_r, hi_r, 7):
            val = float(np.max((q + b*t + a*t*t) * dx))
            if val < best_f - 1e-14:
                best_f = val
                best_t = t
    if abs(best_t) < 1e-10 or best_f >= float(np.max(q)*dx) - 1e-12:
        return None
    # Apply and verify
    h2 = h.copy()
    h2[i] += best_t
    h2[j] -= best_t
    h2 = project_k(h2, n)
    s = exact_score(h2)
    s0 = float(np.max(q) * dx)
    if s < s0 - 1e-12:
        return h2, s
    return None

def exchange_polish(h, n, rng, n_pairs, n_sweeps):
    h = h.copy()
    q = np.correlate(h, 1.0-h, mode='full')
    s = float(np.max(q) * 2.0/n)
    best_h = h.copy()
    best_s = s
    for sweep in range(n_sweeps):
        accepted = False
        pairs = []
        attempts = 0
        max_att = 20 * n_pairs
        while len(pairs) < n_pairs and attempts < max_att:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                key = (min(i,j), max(i,j))
                if key not in pairs:
                    pairs.append(key)
            attempts += 1
        # Fill with sequential if needed
        if len(pairs) < n_pairs:
            used = set(pairs)
            for ii in range(n):
                for jj in range(ii+1, n):
                    if (ii,jj) not in used:
                        pairs.append((ii,jj))
                        used.add((ii,jj))
                        if len(pairs) >= n_pairs:
                            break
                if len(pairs) >= n_pairs:
                    break
        rng.shuffle(pairs)
        for (i,j) in pairs:
            res = line_search(h, i, j, q, n, rng)
            if res is not None:
                h, s = res
                q = np.correlate(h, 1.0-h, mode='full')
                accepted = True
                if s < best_s - 1e-12:
                    best_s = s
                    best_h = h.copy()
        if not accepted:
            break
    return best_h, best_s
