import numpy as np
from scipy.optimize import minimize

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n)

def project_capped(z, n):
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    s = n / 2.0
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    if r > 1e-14:
        for i in range(n):
            if r <= 1e-14:
                break
            h[i] = min(h[i] + r, 1.0)
            r = s - np.sum(h)
    elif r < -1e-14:
        for i in range(n):
            if -r <= 1e-14:
                break
            h[i] = max(h[i] + r, 0.0)
            r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    return h

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full')
    if q.shape != (2 * n - 1,) or not np.all(np.isfinite(q)):
        return float('inf')
    return float(np.max(q * (2.0 / n)))

def epiq_fun(z, n):
    h, t = z[:n], z[n]
    q = np.correlate(h, 1.0 - h, mode='full')
    return t - q

def epiq_jac(z, n):
    h = z[:n]
    dx = 2.0 / n
    L = 2 * n - 1
    J = np.zeros((L, n + 1))
    for idx in range(L):
        d = idx - (n - 1)
        p0, q0 = max(d, 0), max(-d, 0)
        m = n - abs(d)
        J[idx, p0:p0 + m] -= dx * (1.0 - h[q0:q0 + m])
        J[idx, q0:q0 + m] += dx * h[p0:p0 + m]
        J[idx, n] = 1.0
    return J

def eq_fun(z, n):
    return np.sum(z[:n]) - n / 2.0

def eq_jac(z, n):
    J = np.zeros(n + 1)
    J[:n] = 1.0
    return J

def obj_fun(z, n):
    return z[n]

def obj_jac(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def polish_start(h0, n):
    h = project_capped(h0.copy(), n)
    if h is None:
        return h0, exact_score(h0, n)
    score = exact_score(h, n)
    if score == float('inf'):
        return h, float('inf')
    for _ in range(2):
        q = np.correlate(h, 1.0 - h, mode='full')
        t0 = float(np.max(q * (2.0 / n))) + max(1e-10, 1e-8 * abs(float(np.max(q))))
        z0 = np.concatenate([h, [t0]])
        bounds = [(0.0, 1.0)] * n + [(None, None)]
        constraints = [
            {'type': 'ineq', 'fun': epiq_fun, 'jac': epiq_jac},
            {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        ]
        try:
            res = minimize(obj_fun, z0, jac=obj_jac, args=(n,), method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 200, 'ftol': 1e-12})
        except Exception:
            break
        if res.x is None or res.x.shape != (n + 1,) or not np.all(np.isfinite(res.x)):
            break
        yh = project_capped(res.x[:n], n)
        if yh is None:
            break
        best_h, best_s = h.copy(), score
        for alpha in (1.0, 0.5, 0.25, 0.125, 0.0625):
            z = project_capped(h + alpha * (yh - h), n)
            if z is None or not is_feasible(z, n):
                continue
            sz = exact_score(z, n)
            if sz < best_s - 1e-12:
                best_s, best_h = sz, z.copy()
        if best_s >= score - 1e-12:
            break
        h, score = best_h, best_s
    return h, score

def build_starts(fallback, n, rng, count):
    pool = []

    def _add(v):
        if v is None:
            return
        v = project_capped(v, n)
        if v is None or not is_feasible(v, n):
            return
        for s in pool:
            if np.max(np.abs(v - s)) < 1e-8:
                return
        pool.append(v)

    _add(fallback)
    _add(fallback[::-1])
    _add(1.0 - fallback[::-1])
    _add(np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    _add(blk)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    _add(alt)
    for sd in (0.06, 0.15, 0.35):
        _add(fallback + rng.normal(0, sd, n))
    _add(rng.uniform(0, 1, n))
    _add(fallback + 0.1 * np.sin(2.0 * np.pi * np.arange(n) / n))
    return pool[:count]
