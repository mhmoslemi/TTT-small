def optimize_trajectory(x, n, rng, best_h, best_score):
    dx = 2.0 / n
    f = exact_score(x, n)
    if f is None or not np.isfinite(f):
        return best_h, best_score
    if f < best_score - 1e-12:
        best_h, best_score = x.copy(), f
    rho = 0.08
    rho_max = 0.25
    rho_min = 2e-4
    max_iter = 30 if n <= 128 else (22 if n <= 256 else (14 if n <= 512 else 6))
    fails = 0
    for it in range(max_iter):
        if rho < rho_min or fails >= 4:
            break
        q, G = build_lag_model(x, n)
        if q is None or G is None:
            break
        delta = solve_lp_step(x, q, G, rho, n, dx)
        if delta is None or np.max(np.abs(delta)) < 1e-9:
            rho = max(rho_min, rho * 0.5)
            fails += 1
            continue
        best_step = None
        best_f_trial = f
        for j in range(9):
            beta = 2.0 ** (-j)
            cand = x + beta * delta
            pc = project_capped_simplex(cand, n / 2.0)
            if pc is None or not is_feasible(pc, n):
                continue
            sc = exact_score(pc, n)
            if sc is None or not np.isfinite(sc):
                continue
            if sc < best_f_trial - 1e-12:
                best_f_trial = sc
                best_step = pc
            if sc < best_score - 1e-12:
                best_h, best_score = pc.copy(), sc
        if best_step is None:
            rho = max(rho_min, rho * 0.5)
            fails += 1
            continue
        s = best_step - x
        pred = f - float(np.max(q + G.dot(s)))
        act = f - best_f_trial
        if pred <= 1e-12 or act / pred < 0.25:
            rho = max(rho_min, rho * 0.5)
        elif act / pred > 0.75 and np.max(np.abs(s)) >= 0.8 * rho:
            rho = min(rho_max, rho * 1.5)
        x = best_step
        f = best_f_trial
        fails = 0
        if np.max(np.abs(s)) < 1e-8:
            break
    return best_h, best_score

def run(seed=42, **kwargs):
    if seed is None:
        seed = 42
    try:
        seed = int(seed)
    except (TypeError, ValueError):
        seed = 42
    if seed < 0:
        seed = 42
    rng = np.random.default_rng(seed)
    n = None
    parent = None
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        try:
            iv_arr = np.asarray(iv, dtype=np.float64)
            if iv_arr.ndim == 1 and 8 <= iv_arr.size <= 999:
                n = iv_arr.size
                if np.all(np.isfinite(iv_arr)) and np.max(np.abs(iv_arr)) <= 1e3:
                    proj = project_capped_simplex(iv_arr, n / 2.0)
                    if proj is not None and is_feasible(proj, n):
                        parent = proj.copy()
        except Exception:
            pass
    if n is None:
        n = 80
        try:
            n = int(kwargs.get('n_points', 80))
        except (TypeError, ValueError):
            n = 80
        if n < 8 or n > 999:
            n = 80
    S = n / 2.0
    dx = 2.0 / n
    h_const = np.full(n, 0.5)
    best_h = h_const.copy()
    best_score = exact_score(h_const, n)
    if best_score is None or not np.isfinite(best_score):
        best_score = 0.5
    parent_fb = None
    if parent is not None:
        ps = exact_score(parent, n)
        if ps is not None and np.isfinite(ps):
            parent_fb = parent.copy()
            if ps < best_score - 1e-12:
                best_score = ps
                best_h = parent.copy()
    starts = []
    if parent is not None:
        starts.append(parent.copy())
        for sig in (0.04, 0.12):
            pert = parent + rng.normal(0, sig, n)
            pp = project_capped_simplex(pert, S)
            if pp is not None:
                starts.append(pp)
        starts.append(parent[::-1].copy())
    idx = np.arange(n, dtype=np.float64)
    M = min(5, (n - 1) // 2)
    u = np.full(n, 0.5)
    for mm in range(1, M + 1):
        amp = rng.normal(0, 0.10 / np.sqrt(mm))
        phi = rng.uniform(0, 2.0 * np.pi)
        u += amp * np.cos(2.0 * np.pi * mm * (idx + 0.5) / n + phi)
    up = project_capped_simplex(u, S)
    if up is not None:
        starts.append(up)
    perm = rng.permutation(n)
    binv = np.zeros(n)
    binv[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        binv[perm[n // 2]] = 0.5
    starts.append(binv.copy())
    if parent is None:
        starts.append(h_const.copy())
        u2 = project_capped_simplex(rng.uniform(0, 1, n), S)
        if u2 is not None:
            starts.append(u2)
    dd = []
    for s in starts:
        dup = False
        for d2 in dd:
            if np.max(np.abs(s - d2)) < 1e-9:
                dup = True
                break
        if not dup and is_feasible(s, n):
            dd.append(s)
    max_starts = 6 if n <= 128 else (4 if n <= 256 else (3 if n <= 512 else 2))
    dd = dd[:max_starts]
    for s in dd:
        sc = exact_score(s, n)
        if sc is not None and np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_h = s.copy()
    for s in dd:
        best_h, best_score = optimize_trajectory(s.copy(), n, rng, best_h, best_score)
    n_restarts = 2 if n <= 128 else (1 if n <= 256 else 0)
    sigma = 0.05
    for _ in range(n_restarts):
        pert = best_h + sigma * rng.normal(0, 1, n)
        pp = project_capped_simplex(pert, S)
        if pp is not None and is_feasible(pp, n):
            old_sc = best_score
            best_h, best_score = optimize_trajectory(pp, n, rng, best_h, best_score)
            if best_score < old_sc - 1e-12:
                sigma = max(0.02, 0.5 * sigma)
            else:
                sigma = min(0.18, 1.8 * sigma)
    if not is_feasible(best_h, n):
        best_h = project_capped_simplex(best_h, S)
    if best_h is None or not is_feasible(best_h, n):
        if parent_fb is not None:
            best_h = parent_fb.copy()
        else:
            best_h = h_const.copy()
    final_score = exact_score(best_h, n)
    if final_score is None or not np.isfinite(final_score):
        if parent_fb is not None:
            best_h = parent_fb.copy()
            final_score = exact_score(best_h, n)
        if final_score is None or not np.isfinite(final_score):
            best_h = h_const.copy()
            final_score = 0.5
    if not np.isfinite(final_score):
        final_score = 0.5
    return best_h.astype(np.float64), float(final_score), n
