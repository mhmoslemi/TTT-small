def project_K(y, n, S):
    if not np.all(np.isfinite(y)):
        return np.full(n, S / n)

    hi = y.max() + 1.0
    for _ in range(80):

        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid

            break

    for sc, aa, idx in cands:
        if any(abs(idx - u) < 2 for u in used):
            continue
        branches.append(idx)
        used.add(idx)
        if len(branches) >= nb:
            break
    if not branches:
        branches = [int(np.argmin(Fm))]
    best_alpha, best_sc = 0.0, float(q.max())
    for bi in branches:
        a_lo = float(alphas[max(0, bi - 1)])
        a_hi = float(alphas[min(len(alphas) - 1, bi + 1)])
        for _ in range(nr):
            pts = np.linspace(a_lo, a_hi, rp)
            Fv = (q[np.newaxis, :] + pts[:, np.newaxis] * b[np.newaxis, :] + pts[:, np.newaxis] ** 2 * a[np.newaxis, :]).max(axis=1)
            mi = int(np.argmin(Fv))
            if Fv[mi] < best_sc:
                best_sc = float(Fv[mi])
                best_alpha = float(pts[mi])
            a_lo = float(pts[max(0, mi - 1)])
            a_hi = float(pts[min(len(pts) - 1, mi + 1)])
    return best_alpha, best_sc

def local_descent(h0, n, dx, S, rng, mw, nsp, cp, nb, nr, rp, usa, mp):
    h = project_K(h0.copy(), n, S)
    q = lag_vec(h, dx)
    sc = exact_score(h, dx)
    if usa and n <= 64:
        pairs = [(p, r) for r in range(1, n) for p in range(r)]
    else:
        seen, pairs = set(), []
        att = 0
        while len(pairs) < mp and att < mp * 4:
            p, r = int(rng.integers(0, n)), int(rng.integers(0, n))
            if p < r and (p, r) not in seen:
                pairs.append((p, r))
                seen.add((p, r))
            att += 1
        if len(pairs) < mp:
            for r in range(1, n):
                for p in range(r):
                    if (p, r) not in seen:
                        pairs.append((p, r))
                        seen.add((p, r))
                        if len(pairs) >= mp:
                            break
                if len(pairs) >= mp:
                    break
    for _ in range(mw):
        changed = False
        rng.shuffle(pairs)
        for (p, r) in pairs:
            b = np.zeros(2 * n - 1)
            a = np.zeros(2 * n - 1)
            a[n - 1] = -2.0 * dx
            d1 = p - r
            if -(n - 1) <= d1 <= n - 1:
                li, hii = max(0, -d1), min(n, n - d1)
                if li <= r < hii:
                    a[d1 + n - 1] = dx
            d2 = r - p
            if 0 <= d2 <= n - 1:
                li, hii = max(0, -d2), min(n, n - d2)
                if li <= p < hii:
                    a[d2 + n - 1] = dx
            for ell in range(-(n - 1), n):
                val, li, hii = 0.0, max(0, -ell), min(n, n - ell)
                i1, i2 = p - ell, r - ell
                if li <= i1 < hii:
                    val += 1.0 - h[i1]
                if li <= i2 < hii:
                    val -= 1.0 - h[i2]
                if li <= p < hii and 0 <= p + ell < n:
                    val -= h[p + ell]
                if li <= r < hii and 0 <= r + ell < n:
                    val += h[r + ell]
                if val != 0.0:
                    b[ell + n - 1] = dx * val
            if not (np.all(np.isfinite(b)) and np.all(np.isfinite(a))):
                continue
            lo_a = max(-h[p], h[r] - 1.0)
            hi_a = min(1.0 - h[p], h[r])
            if hi_a - lo_a < 1e-14:
                continue
            ba, bs = do_line_search(q, b, a, lo_a, hi_a, cp, nb, nr, rp)
            if bs > sc:
                continue
            ht = h.copy()
            ht[p] += ba
            ht[r] -= ba
            if ht.min() < -1e-10 or ht.max() > 1.0 + 1e-10:
                ht = project_K(ht, n, S)
            if not feasible(ht, n, S):
                continue
            st = exact_score(ht, dx)
            if st < sc - 1e-12:
                h, q, sc = ht, lag_vec(ht, dx), st
                changed = True
        for _ in range(nsp):
            ssz = min(n, max(3, int(rng.integers(3, min(13, n + 1)))))
            sup = rng.choice(n, size=ssz, replace=False)
            vals = rng.standard_normal(ssz)
            vals -= vals.mean()
            if np.max(np.abs(vals)) < 1e-15:
                continue
            z = np.zeros(n)
            z[sup] = vals
            z[sup[-1]] = -z[sup[:-1]].sum()
            if np.max(np.abs(z)) < 1e-15:
                continue
            z /= np.max(np.abs(z))
            bv = dx * (np.correlate(z, 1.0 - h, mode='full') - np.correlate(h, z, mode='full'))
            av = -dx * np.correlate(z, z, mode='full')
            if not (np.all(np.isfinite(bv)) and np.all(np.isfinite(av))):
                continue
            lo_a, hi_a = 0.0, 0.0
            for jj in range(n):
                if z[jj] > 1e-15:
                    lo_a = max(lo_a, -h[jj] / z[jj])
                    hi_a = min(hi_a, (1.0 - h[jj]) / z[jj])
                elif z[jj] < -1e-15:
                    lo_a = max(lo_a, (1.0 - h[jj]) / z[jj])
                    hi_a = min(hi_a, -h[jj] / z[jj])
            if hi_a - lo_a < 1e-14:
                continue
            ba, bs = do_line_search(q, bv, av, lo_a, hi_a, cp, nb, nr, rp)
            if bs > sc:
                continue
            ht = h + ba * z
            if ht.min() < -1e-10 or ht.max() > 1.0 + 1e-10:
                ht = project_K(ht, n, S)
            if not feasible(ht, n, S):
                continue
            st = exact_score(ht, dx)
            if st < sc - 1e-12:
                h, q, sc = ht, lag_vec(ht, dx), st
                changed = True
        if not changed:
            break
    return h, sc
