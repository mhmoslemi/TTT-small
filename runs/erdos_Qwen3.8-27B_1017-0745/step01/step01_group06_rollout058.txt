            if sc < best_s - 1e-12:
                best_s, best_h, best_n = sc, s.copy(), m
            sel = project_K(local_optimize(s, m, dx, S, mx_it, sl_it), m, S)
            if feasible(sel, m, S):
                sc2 = exact_score(sel, dx)
                if sc2 < best_s - 1e-12:
                    best_s, best_h, best_n = sc2, sel.copy(), m
                sel2 = project_K(binary_polish(sel, m, dx, S), m, S)
                if feasible(sel2, m, S):
                    sc3 = exact_score(sel2, dx)
                    if sc3 < best_s - 1e-12:
                        best_s, best_h, best_n = sc3, sel2.copy(), m
        if m <= 96 and feasible(best_h if best_n == m else h0, m, S):
            base = best_h.copy() if best_n == m else h0.copy()
            hp = project_K(base + 0.18 * rng.standard_normal(m), m, S)
            sel = project_K(local_optimize(hp, m, dx, S, mx_it, sl_it), m, S)
            if feasible(sel, m, S):
                sc2 = exact_score(sel, dx)
                if sc2 < best_s - 1e-12:
                    best_s, best_h, best_n = sc2, sel.copy(), m
                sel2 = project_K(binary_polish(sel, m, dx, S), m, S)
                if feasible(sel2, m, S):
                    sc3 = exact_score(sel2, dx)
                    if sc3 < best_s - 1e-12:
                        best_s, best_h, best_n = sc3, sel2.copy(), m
    if best_h is None or not feasible(best_h, best_n, best_n / 2.0):
        if parent is not None:
            best_h, best_n = parent.copy(), len(parent)
        else:
            best_h, best_n = np.full(nd, 0.5), nd
        best_s = exact_score(best_h, 2.0 / best_n)
    else:
        best_h = project_K(best_h, best_n, best_n / 2.0)
        best_s = exact_score(best_h, 2.0 / best_n)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_s = exact_score(best_h, 2.0 / best_n)
    return (best_h, best_s, best_n)
