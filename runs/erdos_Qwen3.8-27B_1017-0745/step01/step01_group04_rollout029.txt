Let me analyze the problem and the strategy carefully.

The problem is to find a step function h: [0,2] → [0,1] that minimizes the overlap integral C₅ = max_k ∫ h(x)(1-h(x+k)) dx, subject to ∫₀² h(x) dx = 1 and 0 ≤ h(x) ≤ 1.

The previous program achieved C₅ = 0.381024 and the target is 0.3808.

The strategy proposes using a curvature-majorized majorization-minimization (MM) approach where:
1. At each iterate, construct a convex quadratic upper bound for every lag
2. Minimize the maximum of those bounds over the box-and-sum feasible set
3. Use SLSQP on the resulting convex QP subproblem
4. Use exact scoring for acceptance

Key mathematical insight: For each lag s, q_s(h) is quadratic with a Hessian whose largest eigenvalue is at most 2. So q_s(y) ≤ q_s(x) + g_s(x)ᵀ(y-x) + ‖y-x‖₂² for all y.

Let me implement this carefully following the strategy:

1. `project_box_sum(z, m)` - joint projection onto {0≤h≤1, sum(h)=m}
2. `lag_data(h, N)` - returns q and gradient matrix G
3. MM subproblem via SLSQP
4. Multiple starts, perturbation restarts
5. Exact scoring

import numpy as np
from scipy.optimize import minimize

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    if np.max(np.abs(z)) > 1e12:
        return np.full(N, 0.5)
    z = z - np.median(z)
    lo, hi = z.min() - 1.0, z.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    for _ in range(2):
        if r > 1e-14:
            for i in range(N):
                if r <= 1e-14:
                    break
                adj = min(r, 1.0 - h[i])
                h[i] += adj
                r -= adj
        elif r < -1e-14:
            for i in range(N):
                if r >= -1e-14:
                    break
                adj = min(-r, h[i])
                h[i] -= adj
                r += adj
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1 + 1e-10:
        return np.full(N, 0.5)
    if abs(h.sum() - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)

def exact_score(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    if h.min() < -1e-10 or h.max() > 1 + 1e-10:
        return np.inf
    corr = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(corr)):
        return np.inf
    return float(np.max(corr) * 2.0 / N)

def lag_data(h, N):
    L = 2 * N - 1
    q = np.zeros(L)
    G = np.zeros((L, N))
    for s in range(-(N - 1), N):
        r = s + N - 1
        if s >= 0:
            ell = N - s
            q[r] = np.dot(h[:ell], 1.0 - h[s:s + ell])
            G[r, :ell] += 1.0 - h[s:s + ell]
            G[r, s:s + ell] -= h[:ell]
        else:
            d = -s
            ell = N - d
            q[r] = np.dot(h[d:], 1.0 - h[:ell])
            G[r, d:] += 1.0 - h[:ell]
            G[r, :ell] -= h[d:]
    return q, G

def _mm_obj(z, x, N, m):
    d = z[:N] - x
    return np.dot(d, d) + z[N]

def _mm_obj_jac(z, x, N, m):
    g = np.zeros(N + 1)
    g[:N] = 2.0 * (z[:N] - x)
    g[N] = 1.0
    return g

def _mm_eq(z, x, N, m):
    return np.array([np.sum(z[:N]) - m])

def _mm_eq_jac(z, x, N, m):
    J = np.zeros((1, N + 1))
    J[0, :N] = 1.0
    return J

def _mm_ineq(z, x, N, m, q, G):
    d = z[:N] - x
    return z[N] - q - G.dot(d)

def _mm_ineq_jac(z, x, N, m, q, G):
    L = 2 * N - 1
    J = np.zeros((L, N + 1))
    J[:, :N] = -G
    J[:, N] = 1.0
    return J

def mm_solve(x, N, m, max_iter=40):
    h = x.copy()
    h_score = exact_score(h, N)
    best = h.copy()
    best_score = h_score
    for it in range(max_iter):
        q, G = lag_data(h, N)
        if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
            break
        t0 = max(np.max(q), 0.0)
        z0 = np.concatenate([h, [t0]])
        bounds = [(0.0, 1.0)] * N + [(0.0, float(N) + 1)]
        cons = [
            {'type': 'eq', 'fun': _mm_eq, 'jac': _mm_eq_jac, 'args': (x, N, m)},
            {'type': 'ineq', 'fun': _mm_ineq, 'jac': _mm_ineq_jac, 'args': (x, N, m, q, G)},
        ]
        try:
            res = minimize(_mm_obj, z0, jac=_mm_obj_jac, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           args=(x, N, m),
                           options={'maxiter': 150, 'ftol': 1e-10, 'disp': False})
        except Exception:
            break
        if res.x is None or len(res.x) != N + 1 or not np.all(np.isfinite(res.x[:N])):
            break
        y = project_box_sum(res.x[:N], m)
        # Line search
        improvements = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125, 0.00390625]
        best_cand = None
        best_cand_score = h_score
        for alpha in improvements:
            cand = project_box_sum((1.0 - alpha) * h + alpha * y, m)
            cs = exact_score(cand, N)
            if cs < best_cand_score - 1e-12:
                best_cand = cand
                best_cand_score = cs
        if best_cand is None:
            break
        h = best_cand
        h_score = best_cand_score
        if h_score < best_score - 1e-12:
            best = h.copy()
            best_score = h_score
        if np.max(np.abs(h - best)) <= 1e-8:
            break
    return best, best_score

def make_starts(baseline, N, m, rng, n_starts=8):
    starts = [baseline.copy()]
    # complement
    starts.append(project_box_sum(1.0 - baseline, m))
    # reversal
    starts.append(project_box_sum(baseline[::-1].copy(), m))
    # shift
    sh = max(1, N // 3)
    starts.append(project_box_sum(np.roll(baseline, sh), m))
    # noisy
    for sc in [0.05, 0.15]:
        starts.append(project_box_sum(baseline + sc * rng.standard_normal(N), m))
    # uniform random
    starts.append(project_box_sum(rng.uniform(size=N), m))
    # Fourier
    for amp in [0.12, 0.24]:
        z = np.full(N, 0.5)
        idx = np.arange(N)
        R = min(6, (N - 1) // 2)
        for r in range(1, R + 1):
            a = rng.standard_normal()
            b = rng.standard_normal()
            z += (amp / r) * (a * np.cos(2.0 * np.pi * r * idx / N) + b * np.sin(2.0 * np.pi * r * idx / N))
        starts.append(project_box_sum(z, m))
    # deduplicate
    unique = [starts[0]]
    for s in starts[1:]:
        if np.max(np.abs(s - unique[-1])) > 1e-10:
            dup = False
            for u in unique:
                if np.max(np.abs(s - u)) <= 1e-10:
                    dup = True
                    break
            if not dup:
                unique.append(s)
        if len(unique) >= n_starts:
            break
    return unique

from scipy.optimize import minimize

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)) or np.max(np.abs(z)) > 1e12:
        return np.full(N, 0.5)
    z = z - np.median(z)
    lo, hi = z.min() - 1.0, z.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    for _ in range(2):
        if r > 1e-14:
            for i in range(N):
                if r <= 1e-14: break
                a = min(r, 1.0 - h[i]); h[i] += a; r -= a
        elif r < -1e-14:
            for i in range(N):
                if r >= -1e-14: break
                a = min(-r, h[i]); h[i] -= a; r += a
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1 + 1e-10:
        return np.full(N, 0.5)
    if abs(h.sum() - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return h

def exact_score(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    if h.min() < -1e-10 or h.max() > 1 + 1e-10:
        return np.inf
    c = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(c)):
        return np.inf
    return float(np.max(c) * 2.0 / N)

def lag_data(h, N):
    L = 2 * N - 1
    q = np.zeros(L)
    G = np.zeros((L, N))
    for s in range(-(N - 1), N):
        r = s + N - 1
        if s >= 0:
            ell = N - s
            q[r] = np.dot(h[:ell], 1.0 - h[s:s+ell])
            G[r, :ell] += 1.0 - h[s:s+ell]
            G[r, s:s+ell] -= h[:ell]
        else:
            d = -s; ell = N - d
            q[r] = np.dot(h[d:], 1.0 - h[:ell])
            G[r, d:] += 1.0 - h[:ell]
            G[r, :ell] -= h[d:]
    return q, G

def _obj(z, x, N, m, q, G):
    d = z[:N] - x
    return float(np.dot(d, d) + z[N])

def _obj_g(z, x, N, m, q, G):
    g = np.zeros(N + 1)
    g[:N] = 2.0 * (z[:N] - x)
    g[N] = 1.0
    return g

def _eq(z, x, N, m, q, G):
    return np.array([np.sum(z[:N]) - m])

def _eq_j(z, x, N, m, q, G):
    J = np.zeros((1, N + 1)); J[0, :N] = 1.0
    return J

def _ine(z, x, N, m, q, G):
    d = z[:N] - x
    return z[N] - q - G.dot(d)

def _ine_j(z, x, N, m, q, G):
    L = 2 * N - 1
    J = np.zeros((L, N + 1))
    J[:, :N] = -G; J[:, N] = 1.0
    return J

    best, bs = h.copy(), hs

            {'type': 'ineq', 'fun': _ine, 'jac': _ine_j, 'args': args},

def make_fourier(N, rng, amp, R):
    z = np.full(N, 0.5)
    idx = np.arange(N)
    for r in range(1, R + 1):
        a, b = rng.standard_normal(), rng.standard_normal()
        z += (amp / r) * (a * np.cos(2.0*np.pi*r*idx/N) + b * np.sin(2.0*np.pi*r*idx/N))
    return z
