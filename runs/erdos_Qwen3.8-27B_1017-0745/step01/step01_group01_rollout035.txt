        add_unique(starts, p, n)

        add_unique(starts, 1.0 - p.copy(), n)
        add_unique(starts, 1.0 - p[::-1].copy(), n)
        for sh in (n // 3, n // 2):
            add_unique(starts, np.roll(p, sh).copy(), n)
        for sd in (0.02, 0.06, 0.14, 0.25):
            add_unique(starts, p + rng.normal(0, sd, n), n)
    else:
        add_unique(starts, rng.uniform(0, 1, n), n)
        add_unique(starts, 0.5 + rng.normal(0, 0.08, n), n)
        add_unique(starts, 0.5 + rng.normal(0, 0.2, n), n)
    return starts

def fill_pop(pop, n, NP, rng, centers, scales):
    try_ct = 0
    max_try = 30 * max(1, NP - len(pop))
    while len(pop) < NP and try_ct < max_try:
        try_ct += 1
        src = centers[rng.integers(len(centers))]
        sd = scales[rng.integers(len(scales))]
        if rng.random() < 0.5:
            v = src + rng.normal(0, sd, n)
        else:
            v = rng.beta(0.5, 0.5, n)
        h2 = project_capped(v, n)
        if h2 is None:
            continue
        dup = False
        for p in pop:
            if float(np.max(np.abs(p - h2))) < 1e-10:
                dup = True
                break
        if not dup:
            pop.append(h2)
        elif try_ct > max_try // 2 and rng.random() < 0.2:
            pop.append(h2)
    while len(pop) < NP:
        pop.append(np.full(n, 0.5))
    return pop

def local_polish(h, n, rng, n_iter=200, step0=0.02):
    best = h.copy()
    best_sc = exact_score(best, n)
    step = step0
    for _ in range(n_iter):
        v_dir = rng.normal(0, 1, n)
        v_dir -= float(np.sum(v_dir)) / n
        v = best + step * v_dir
        hp = project_capped(v, n)
        if hp is None:
            step *= 0.7
            continue
        sc = exact_score(hp, n)
        if np.isfinite(sc) and sc < best_sc - 1e-13:
            best = hp
            best_sc = sc
            step = min(step * 1.1, 0.15)
        else:
            step *= 0.75
            if step < 1e-5:
                step = step0 * 0.4
    return best, best_sc

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if len(iv) > 0:
            n = len(iv)
            if (np.all(np.isfinite(iv)) and np.all(iv >= -1e-8)
                    and np.all(iv <= 1.0 + 1e-8)):
                if (np.all(iv >= 0) and np.all(iv <= 1)
                        and abs(float(np.sum(iv)) - n / 2.0) <= 5e-12 * max(1, n)):
                    parent = iv.copy()
                else:
                    pp = project_capped(iv, n)
                    if pp is not None:
                        parent = pp
    except Exception:
        pass
    if n is None:
        n = parse_int(kwargs.get('n_points', 47), 47, 1, 999)
    if n < 1 or n > 999:
        n = 47
    try:
        sd = int(seed) & 0xFFFFFFFF
    except Exception:
        sd = 42
    rng = np.random.default_rng(sd)
    uniform = np.full(n, 0.5)
    if n == 1:
        return uniform, exact_score(uniform, 1), 1
    if n <= 64:
        NP_def, gen_def, rmax_def = 64, 160, 2
    elif n <= 128:
        NP_def, gen_def, rmax_def = 48, 80, 1
    elif n <= 256:
        NP_def, gen_def, rmax_def = 32, 30, 0
    else:
        NP_def, gen_def, rmax_def = 0, 0, 0
    NP = parse_int(kwargs.get('pop_size', NP_def), NP_def, 8, 96) if n <= 256 else 0
    gens = parse_int(kwargs.get('generations', gen_def), gen_def, 0, 250)
    cap_g = 250 if n <= 64 else (120 if n <= 128 else 50)
    gens = min(gens, cap_g)
    rmax = parse_int(kwargs.get('max_restarts', rmax_def), rmax_def, 0, 3)
    starts = build_starts(parent, n, rng)
    best_h, best_sc = uniform.copy(), exact_score(uniform, n)
    parent_fb = None
    if parent is not None:
        ps = exact_score(parent, n)
        if np.isfinite(ps) and ps < best_sc - 1e-12:
            best_h, best_sc = parent.copy(), ps
        parent_fb = parent.copy()
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_sc - 1e-12:
            best_h, best_sc = st.copy(), ss
    if NP < 4 or gens <= 0 or n > 256:
        fin = project_capped(best_h, n)
        if fin is not None:
            fs = exact_score(fin, n)
            if np.isfinite(fs) and fs <= best_sc + 1e-12:
                best_h, best_sc = fin, fs
        return best_h.astype(np.float64).copy(), float(best_sc), int(n)
    pop = [best_h.copy()]
    for st in starts:
        if len(pop) >= NP:
            break
        d = False
        for p in pop:
            if float(np.max(np.abs(p - st))) < 1e-10:
                d = True
                break
        if not d:
            pop.append(st)
    scales_init = [0.015, 0.04, 0.10, 0.22]
    centers_init = pop[:]
    pop = fill_pop(pop, n, NP, rng, centers_init, scales_init)
    pop = np.array(pop[:NP], dtype=np.float64)
    scores = np.empty(NP)
    for i in range(NP):
        scores[i] = exact_score(pop[i], n)
        if np.isfinite(scores[i]) and scores[i] < best_sc - 1e-12:
            best_h, best_sc = pop[i].copy(), scores[i]
    patience = max(12, min(40, gens // 4))
    stagnant, n_restarts = 0, 0
    all_idx = np.arange(NP)
    for gen in range(gens):
        prev_best = best_sc
        old_pop = pop.copy()
        old_sc = scores.copy()
        next_pop = pop.copy()
        next_sc = scores.copy()
        b_idx = int(np.argmin(old_sc))
        for i in range(NP):
            if rng.random() < 0.8:
                cands = [j for j in range(NP) if j != i]
                r1, r2, r3 = rng.choice(cands, size=3, replace=False)
                F = rng.uniform(0.45, 0.95)
                v = old_pop[r1] + F * (old_pop[r2] - old_pop[r3])
            else:
                cands = [j for j in range(NP) if j != i and j != b_idx]
                if len(cands) < 2:
                    cands = [j for j in range(NP) if j != i]
                if len(cands) < 2:
                    continue
                r1, r2 = rng.choice(cands, size=2, replace=False)
                F = rng.uniform(0.45, 0.95)
                v = (old_pop[i] + F * (old_pop[b_idx] - old_pop[i])
                     + F * (old_pop[r1] - old_pop[r2]))
            CR = rng.uniform(0.75, 0.95)
            mask = rng.random(n) < CR
            j0 = int(rng.integers(n))
            mask[j0] = True
            trial = np.where(mask, v, old_pop[i])
            trial = project_capped(trial, n)
            if trial is None:
                continue
            if float(np.max(np.abs(trial - old_pop[i]))) <= 1e-12:
                continue
            ts = exact_score(trial, n)
            if np.isfinite(ts) and ts < old_sc[i] - 1e-12:
                next_pop[i] = trial
                next_sc[i] = ts
                if ts < best_sc - 1e-12:
                    best_h, best_sc = trial.copy(), ts
        pop, scores = next_pop, next_sc
        if best_sc < prev_best - 1e-12:
            stagnant = 0
        else:
            stagnant += 1
        if stagnant >= patience:
            if n_restarts < rmax:
                n_restarts += 1
                stagnant = 0
                order = np.argsort(scores)
                n_el = max(2, int(np.ceil(0.2 * NP)))
                new_pop = [pop[order[0]].copy()]
                for ei in range(1, n_el):
                    new_pop.append(pop[order[ei]].copy())
                bases = [best_h.copy()] + new_pop
                rsc = [0.03, 0.07, 0.15][min(n_restarts - 1, 2)]
                rsc *= rng.uniform(0.5, 1.5)
                new_pop = fill_pop(new_pop, n, NP, rng, bases, [rsc, rsc * 2.0])
                pop = np.array(new_pop[:NP], dtype=np.float64)
                scores = np.empty(NP)
                for i in range(NP):
                    scores[i] = exact_score(pop[i], n)
                    if np.isfinite(scores[i]) and scores[i] < best_sc - 1e-12:
                        best_h, best_sc = pop[i].copy(), scores[i]
            else:
                break
    # Local polish
    polished, psc = local_polish(best_h, n, rng, n_iter=300, step0=0.015)
    if psc < best_sc - 1e-12:
        best_h, best_sc = polished, psc
    # Final selection
    cands = [(best_h, best_sc)]
    if parent_fb is not None:
        pfs = exact_score(parent_fb, n)
        if np.isfinite(pfs):
            cands.append((parent_fb, pfs))
    us = exact_score(uniform, n)
    cands.append((uniform, us))
    for i in range(NP):
        if np.isfinite(scores[i]):
            cands.append((pop[i].copy(), scores[i]))
    final_h, final_sc = None, float('inf')
    for ch, cs in cands:
        if np.isfinite(cs) and cs < final_sc:
            final_h, final_sc = ch.copy(), cs
    fin = project_capped(final_h, n)
    if fin is not None:
        fs = exact_score(fin, n)
        if np.isfinite(fs) and fs <= final_sc + 1e-12:
            final_h, final_sc = fin, fs
    if final_h is None or not np.isfinite(final_sc):
        final_h, final_sc = uniform.copy(), us
    return final_h.astype(np.float64).copy(), float(final_sc), int(n)
