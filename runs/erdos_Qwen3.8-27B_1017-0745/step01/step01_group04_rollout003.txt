    if abs(h.sum() - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)

def exact_score(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / N)

def smooth_obj_grad(h, tau, dx):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = dx * corr
    cmax = np.max(c)
    e = np.exp((c - cmax) / tau)
    se = e.sum()
    F = cmax + tau * np.log(se)
    p = e / se
    grad = np.zeros(N)
    grad += dx * p[N - 1] * (1.0 - 2.0 * h)
    for s in range(1, N):
        d = s
        pp = dx * p[N - 1 + s]
        pn = dx * p[N - 1 - s]
        grad[:N - d] += pp * (-h[d:])
        grad[d:] += pp * (1.0 - h[:N - d])
        grad[:N - d] += pn * (1.0 - h[d:])
        grad[d:] += pn * (-h[:N - d])
    return F, grad

def optimize_start(h0, N, m, rng, max_iter=50):
    dx = 2.0 / N
    taus = [0.02, 0.006, 0.002, 0.0006, 0.0002, 0.00006, 0.000015]
    h = h0.copy()
    best_h = h0.copy()
    best_sc = exact_score(h0, N)
    for tau in taus:
        F, grad = smooth_obj_grad(h, tau, dx)
        if not np.all(np.isfinite(grad)):
            break
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        alpha = 1.0
        for _ in range(max_iter):
            y = project_box_sum(h - alpha * grad, m)
            d = y - h
            mnd = np.max(np.abs(d))
            if mnd <= 1e-9:
                sc = exact_score(y, N)
                if sc < best_sc:
                    best_h = y.copy()
                    best_sc = sc
                break
            Fy, _ = smooth_obj_grad(y, tau, dx)
            sc = exact_score(y, N)
            if np.isfinite(sc) and sc < best_sc:
                best_h = y.copy()
                best_sc = sc
            gd = np.dot(grad, d)
            if gd < 0 and (not np.isfinite(Fy) or Fy <= F + 1e-4 * gd + 1e-12):
                h = y
                if np.isfinite(Fy):
                    F = Fy
                alpha = min(1.25 * alpha, 10.0)
            else:
                alpha *= 0.5
                if alpha < 1e-10:
                    break
    pert = project_box_sum(h + 0.02 * rng.standard_normal(N), m)
    sc = exact_score(pert, N)
    if sc < best_sc:
        best_h = pert.copy()
        best_sc = sc
    return best_h, best_sc

def make_fourier_start(N, rng, n_modes, amp):
    z = np.full(N, 0.5)
    ns = min(n_modes, (N - 1) // 2)
    idx = np.arange(N)
    for r in range(1, ns + 1):
        A = rng.standard_normal()
        B = rng.standard_normal()
        z += (amp / r) * (A * np.sin(2.0 * np.pi * r * idx / N) + B * np.cos(2.0 * np.pi * r * idx / N))
    return z

def cyclic_perturb(best_h, N, m, rng, n_tries=200):
    h.best_h = best_h.copy()
    best_sc = exact_score(best_h, N)
    for _ in range(n_tries):
        i = rng.integers(0, N)
        j = rng.integers(0, N)
        if i == j:
            continue
        trial = best_h.copy()
        delta = 0.01 * rng.standard_normal()
        trial[i] += delta
        trial[j] -= delta
        trial = project_box_sum(trial, m)
        sc = exact_score(trial, N)
        if sc < best_sc:
            best_h = trial
            best_sc = sc
    return best_h, best_sc

def run(seed=42, **kwargs):
    N = None
    m = None
    raw = globals().get('initial_h_values', None)
    if raw is not None:
        raw = np.asarray(raw, dtype=np.float64)
        if raw.ndim == 1 and raw.size > 1 and np.all(np.isfinite(raw)):
            N = raw.size
            m = N / 2.0
    if N is None:
        N = kwargs.get('n_points', 70)
        if not isinstance(N, (int, np.integer)) or N < 2:
            N = 70
        m = N / 2.0
    if N == 1:
        h = np.array([0.5])
        return h, exact_score(h, 1), 1

    if raw is not None and np.all(np.isfinite(raw)):
        baseline = project_box_sum(raw.copy(), m)
    else:
        baseline = np.full(N, 0.5)

    best_h = baseline.copy()
    best_score = exact_score(best_h, N)
    rng = np.random.default_rng(seed)

    starts = [best_h.copy()]
    starts.append(project_box_sum(1.0 - best_h, m))
    for sc in [0.04, 0.10, 0.20]:
        starts.append(project_box_sum(best_h + sc * rng.standard_normal(N), m))
    starts.append(project_box_sum(0.5 + 0.5 * rng.uniform(size=N), m))
    starts.append(project_box_sum(make_fourier_start(N, rng, 6, 0.12), m))
    starts.append(project_box_sum(make_fourier_start(N, rng, 4, 0.20), m))
    starts.append(project_box_sum(make_fourier_start(N, rng, 8, 0.08), m))

    dedup = [starts[0]]
    for s in starts[1:]:
        if all(np.linalg.norm(s - d) > 1e-10 for d in dedup):
            dedup.append(s)
    starts = dedup[:9]

    for h0 in starts:
        s0 = exact_score(h0, N)
        if s0 < best_score:
            best_h = h0.copy()
            best_score = s0
        h_opt, sc_opt = optimize_start(h0, N, m, rng, max_iter=50)
        if sc_opt < best_score:
            best_h = h_opt.copy()
            best_score = sc_opt

    # Basin hops
    for sc in [0.05, 0.02, 0.01]:
        hop = project_box_sum(best_h + sc * rng.standard_normal(N), m)
        h_hop, sc_hop = optimize_start(hop, N, m, rng, max_iter=40)
        if sc_hop < best_score:
            best_h = h_hop.copy()
            best_score = sc_hop

    # Cyclic perturbation polish
    best_h, best_score = cyclic_perturb(best_h, N, m, rng, n_tries=300)

    # Final projection and validation
    best_h = project_box_sum(best_h, m)
    c5 = exact_score(best_h, N)
    if not np.isfinite(c5):
        best_h = np.full(N, 0.5)
        c5 = exact_score(best_h, N)
    # Ensure parent is not worse
    if exact_score(baseline, N) < c5:
        best_h = baseline.copy()
        c5 = exact_score(baseline, N)
    return best_h, float(c5), int(N)
