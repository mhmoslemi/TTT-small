    except Exception:
        return float('inf')

def lag_val_grad(h, s, n):
    dx = 2.0 / n
    lo = max(0, -s)
    hi = min(n, n - s)
    q = 0.0
    g = np.zeros(n)
    for i in range(lo, hi):
        j = i + s
        q += h[i] * (1.0 - h[j])
        g[i] += dx * (1.0 - h[j])
        g[j] -= dx * h[i]
    return dx * q, g

def mm_obj(z):
    x, L = _mm['x'], _mm['L']
    d = z[:-1] - x
    return 0.5 * L * float(np.dot(d, d)) + z[-1]

def mm_obj_grad(z):
    x, L = _mm['x'], _mm['L']
    return np.append(L * (z[:-1] - x), 1.0)

def mm_ineq(z):
    return z[-1] - _mm['qJ'] - _mm['GJ'] @ (z[:-1] - _mm['x'])

def mm_ineq_jac(z):
    n = len(z) - 1
    J = np.zeros((_mm['GJ'].shape[0], n + 1))
    J[:, :n] = -_mm['GJ']
    J[:, n] = 1.0
    return J

def mm_eq(z):
    return float(np.sum(z[:-1])) - _mm['S']

def mm_eq_jac(z):
    n = len(z) - 1
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def separation(x, n, y):
    dx = 2.0 / n
    d = y - x
    bv, bk = -1e30, 0
    for k in range(2 * n - 1):
        s = k - (n - 1)
        lo = max(0, -s)
        hi = min(n, n - s)
        e = 0.0
        for i in range(lo, hi):
            j = i + s
            e += x[i] * (1.0 - x[j]) + (1.0 - x[j]) * d[i] - x[i] * d[j]
        e *= dx
        if e > bv:
            bv, bk = e, k
    return bv, bk

def solve_mm(x, n):
    dx = 2.0 / n
    L = 2.0 * dx
    nv = 2 * n - 1
    qa = np.zeros(nv)
    Ga = np.zeros((nv, n))
    for k in range(nv):
        qa[k], Ga[k] = lag_val_grad(x, k - (n - 1), n)
    if nv <= 96:
        active = list(range(nv))
        ms = 1
    else:
        active = list(np.argsort(-qa, kind='stable')[:min(nv, 96)])
        ms = 8
    _mm['x'] = x
    _mm['L'] = L
    _mm['S'] = 0.5 * n
    y_cur = x.copy()
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    cons = [{'type': 'ineq', 'fun': mm_ineq, 'jac': mm_ineq_jac},
            {'type': 'eq', 'fun': mm_eq, 'jac': mm_eq_jac}]
    for si in range(ms):
        qJ = qa[active]
        GJ = Ga[active].copy()
        _mm['qJ'] = qJ
        _mm['GJ'] = GJ
        d0 = y_cur - x
        t0 = max(float(np.max(qJ + GJ @ d0)), float(qa.max())) + 1e-8
        z0 = np.append(y_cur.copy(), t0)
        try:
            res = minimize(mm_obj, z0, jac=mm_obj_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 150, 'ftol': 1e-11, 'disp': False})
        except Exception:
            return x.copy()
        if res.x is None or not np.all(np.isfinite(res.x)):
            return x.copy()
        yp = project_box_sum(np.clip(res.x[:n], 0.0, 1.0), 0.5 * n)
        if yp is None:
            return x.copy()
        if nv > 96:
            EJ = float(np.max(qJ + GJ @ (yp - x)))
            fv, fk = separation(x, n, yp)
            if fv > EJ + 1e-9 * max(1.0, abs(EJ)):
                if fk not in active and len(active) < 96:
                    active.append(fk)
                    y_cur = yp
                    continue
                break
            mv = 0.5 * L * float(np.dot(yp - x, yp - x)) + EJ
            if mv > float(qa.max()) + 1e-7:
                return x.copy()
        y_cur = yp
        if nv > 96 and si < ms - 1:
            continue
        break
    return y_cur

def local_mm(x0, n, max_steps=None):
    if max_steps is None:
        max_steps = max(8, min(40, 1800 // max(n, 1)))
    h = x0.copy()
    c = exact_score(h, n)
    if not np.isfinite(c):
        return h, c
    for _ in range(max_steps):
        yp = solve_mm(h, n)
        if yp is None or not np.all(np.isfinite(yp)):
            break
        d = yp - h
        if np.max(np.abs(d)) < 1e-10:
            break
        improved = False
        for k in range(9 if n <= 128 else 6):
            a = 1.0 / (2.0 ** k)
            cand = project_box_sum(h + a * d, 0.5 * n)
            if cand is None:
                continue
            sc = exact_score(cand, n)
            if sc < c - 1e-12 * max(1.0, abs(c)):
                h, c = cand, sc
                improved = True
                break
        if not improved:
            break
    return h, c

def build_starts(parent, n, rng):
    starts = [parent.copy()]
    if n <= 64:
        starts.append(1.0 - parent.copy())
        starts.append(parent[::-1].copy())
        for sd in [0.06, 0.20]:
            p = project_box_sum(parent + sd * rng.standard_normal(n), 0.5 * n)
            if p is not None:
                starts.append(p)
        p = project_box_sum(rng.random(n), 0.5 * n)
        if p is not None:
            starts.append(p)
        arr = np.arange(n)
        u = 0.5 + 0.30 * np.cos(2.0 * np.pi * arr / n + 0.7) + 0.12 * np.sin(4.0 * np.pi * arr / n + 1.3)
        p = project_box_sum(u, 0.5 * n)
        if p is not None:
            starts.append(p)
    elif n <= 256:
        starts.append(1.0 - parent.copy())
        p = project_box_sum(parent + 0.12 * rng.standard_normal(n), 0.5 * n)
        if p is not None:
            starts.append(p)
        p = project_box_sum(rng.random(n), 0.5 * n)
        if p is not None:
            starts.append(p)
    else:
        p = project_box_sum(parent + 0.15 * rng.standard_normal(n), 0.5 * n)
        if p is not None:
            starts.append(p)
        p = project_box_sum(rng.random(n), 0.5 * n)
        if p is not None:
            starts.append(p)
    uniq = []
    for s in starts:
        if not any(np.max(np.abs(s - u)) < 1e-10 for u in uniq):
            uniq.append(s)
    return uniq

def get_parent():
    if 'initial_h_values' in globals():
        ph = np.asarray(globals()['initial_h_values'], dtype=np.float64)
        if ph.ndim == 1 and 2 <= len(ph) <= 999 and np.all(np.isfinite(ph)):
            if np.all(ph >= -1e-6) and np.all(ph <= 1.0 + 1e-6):
                n = len(ph)
                if check_feasible(ph, n):
                    return ph.copy(), n
                p = project_box_sum(np.clip(ph, 0.0, 1.0), 0.5 * n)
                if p is not None and check_feasible(p, n):
                    return p, n
    return np.full(45, 0.5), 45

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent, n = get_parent()
    best_h = parent.copy()
    best_c = exact_score(parent, n)
    starts = build_starts(parent, n, rng)
    for s in starts:
        if np.all(np.isfinite(s)):
            sc = exact_score(s, n)
            if np.isfinite(sc) and sc < best_c:
                best_h, best_c = s.copy(), sc
    for s in starts:
        try:
            hl, cl = local_mm(s, n)
            if np.isfinite(cl) and cl < best_c:
                best_h, best_c = hl.copy(), cl
        except Exception:
            pass
    if n <= 64:
        sds = [0.04, 0.12, 0.28]
    elif n <= 256:
        sds = [0.08, 0.22]
    else:
        sds = [0.15]
    for sd in sds:
        p = project_box_sum(best_h + sd * rng.standard_normal(n), 0.5 * n)
        if p is None:
            continue
        try:
            hr, cr = local_mm(p, n, max(4, 10 // max(n // 45, 1)))
            if np.isfinite(cr) and cr < best_c:
                best_h, best_c = hr.copy(), cr
        except Exception:
            pass
    fh = project_box_sum(best_h, 0.5 * n)
    if fh is not None and check_feasible(fh, n):
        fc = exact_score(fh, n)
        if np.isfinite(fc):
            return (fh, fc, n)
    fb = project_box_sum(parent, 0.5 * n)
    if fb is None:
        fb = np.full(n, 0.5)
    return (fb, exact_score(fb, n), n)
