    base = project_K(init.copy(), n, S) if ok else np.full(n, 0.5)
    if not is_feasible(base, n, S):
        base = np.full(n, 0.5)
    fb_h, fb_s = base.copy(), exact_score(base, dx)

    pool = build_pool(base, n, S, rng)
    best_h, best_s = base.copy(), fb_s
    for c in pool:
        sc = exact_score(c, dx)
        if sc < best_s - 1e-12:
            best_s, best_h = sc, c.copy()

    if n <= 64:
        nst, nlv, npr, nca, npl = 8, 14, 80, 32, 80
    elif n <= 128:
        nst, nlv, npr, nca, npl = 5, 10, 40, 20, 40
    elif n <= 256:
        nst, nlv, npr, nca, npl = 3, 8, 18, 12, 20
    else:
        nst, nlv, npr, nca, npl = 2, 6, 6, 8, 8

    scored = []
    for i in range(len(pool)):
        scored.append((exact_score(pool[i], dx), i))
    scored.sort()
    starts, seen = [], []
    for sc, i in scored:
        if len(starts) >= nst:
            break
        c = pool[i]
        dup = any(np.max(np.abs(c - sv)) <= 1e-7 for sv in seen)
        if not dup:
            starts.append(c)
            seen.append(c)
    if not starts:
        starts = [base.copy()]

    for s in starts:
        try:
            bh, bf = anneal_chain(s, n, S, dx, rng, nlv, npr, nca)
            if is_feasible(bh, n, S) and bf < best_s - 1e-12:
                best_s, best_h = bf, bh.copy()
        except Exception:
            pass

    try:
        bh, bf = zero_temp_polish(best_h, n, S, dx, rng, npl, [0.02, 0.006])
        if is_feasible(bh, n, S) and bf < best_s - 1e-12:
            best_s, best_h = bf, bh.copy()
    except Exception:
        pass

    best_h = project_K(best_h, n, S)
    fs = exact_score(best_h, dx)
    if not is_feasible(best_h, n, S) or not np.isfinite(fs) or fs > fb_s + 1e-10:
        best_h, fs = fb_h.copy(), fb_s
    else:
        best_s = fs
    best_h = np.clip(best_h, 0.0, 1.0)
    final_s = exact_score(best_h, dx)
    return (best_h, float(final_s), int(n))
