                    rx, rf, info = fmin_l_bfgs_b(slbfgs_fun, st.copy(),
                        fprime=slbfgs_grad, args=(n, eps_val, mu_val),
                        pgtol=1e-12, maxiter=400,
                        bounds=[(0.0, 1.0)] * n)
                    if rx is not None and rx.size == n and np.all(np.isfinite(rx)):
                        h2 = project_capped(rx, n)
                        if h2 is not None:
                            s2 = exact_score(h2, n)
                            if np.isfinite(s2) and s2 < best_score - 1e-12:
                                best, best_score = h2.copy(), s2
                except Exception:
                    pass

    if n <= 64:
        polished = polish(best.copy(), n, 2)
        if polished is not None:
            ps = exact_score(polished, n)
            if np.isfinite(ps) and ps < best_score - 1e-12:
                best, best_score = polished, ps

    final = project_capped(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final, fs = fallback.copy(), fb_score
    return final, float(fs), int(n)
