import numpy as np

def project_K(y, n, S):
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return np.full(n, S / n)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    e = S - h.sum()
    if abs(e) > 1e-12:
        for j in range(n):
            if abs(e) < 1e-12:
                break
            if e > 0:
                adj = min(e, 1.0 - h[j])
                h[j] += adj
                e -= adj
            else:
                adj = max(e, -h[j])
                h[j] += adj
                e += adj
    if abs(h.sum() - S) > 1e-8:
        return np.full(n, S / n)
    return h

    n = h.shape[0]
    if n == 0 or not np.all(np.isfinite(h)):
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    if q.size != 2 * n - 1 or not np.all(np.isfinite(q)):
        return float('inf')
    return float(np.max(q))
