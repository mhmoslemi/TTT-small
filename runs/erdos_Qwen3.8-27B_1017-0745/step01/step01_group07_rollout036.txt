    z = np.asarray(z, dtype=np.float64)

        return None
    if np.max(np.abs(z)) > 1e12:
        return None
    n = len(z)
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    y = np.clip(z - lam, 0.0, 1.0)
    r = S - np.sum(y)
    for i in range(n):

            break
        if r > 0:
            a = min(r, 1.0 - y[i])

            r -= a
