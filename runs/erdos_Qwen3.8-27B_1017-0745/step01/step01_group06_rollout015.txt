import numpy as np
import math

def project_K(y, n, S):
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return np.full(n, S / n)
    lo, hi = float(y.min() - 1), float(y.max() + 1)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0, 1).sum() > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    e = S - h.sum()
    if abs(e) > 1e-12:
        if e > 0:
            for j in range(n):
                t = min(e, 1.0 - h[j]); h[j] += t; e -= t
                if e <= 1e-12: break
        else:
            for j in range(n):
                t = min(-e, h[j]); h[j] -= t; e += t
                if e >= -1e-12: break
    if not np.all(np.isfinite(h)) or abs(h.sum() - S) > 1e-9:
        return np.full(n, S / n)
    return np.clip(h, 0.0, 1.0)

def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lag_vals(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def is_feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def grad_col(j, h, n, dx):
    res = np.zeros(2 * n - 1)
    nm = n - 1
    for k in range(2 * n - 1):
        ell = k - nm
        v = 0.0
        jp, jm = j - ell, j + ell
        if 0 <= jp < n:
            v += 1.0 - h[jp]
        if 0 <= jm < n:
            v -= h[jm]
        res[k] = dx * v
    return res

def ls_eval(q, b, a, alpha):
    return np.max(q + alpha * b + alpha * alpha * a)

def line_search(h, q, b, a, lo, hi, n, dx, z):
    if hi - lo < 1e-14:
        return h, q, float(np.max(q))
    nc = 25
    al = np.linspace(lo, hi, nc)
    F = np.array([ls_eval(q, b, a, al[i]) for i in range(nc)])
    mins = []
    for i in range(nc):
        if i == 0:
            if F[i] <= F[i + 1]: mins.append(i)
        elif i == nc - 1:
            if F[i] <= F[i - 1]: mins.append(i)
        else:
            if F[i] <= F[i - 1] and F[i] <= F[i + 1]: mins.append(i)
    mins.sort(key=lambda i: (F[i], abs(al[i])))
    branches = []
    for i in mins:
        if all(abs(i - j) >= 2 for j in branches):
            branches.append(i)
        if len(branches) >= 2: break
    if not branches:
        branches = [int(np.argmin(F))]
    best_a = float(al[int(np.argmin(F))])
    best_f = float(F.min())
    for bi in branches:
        clo = max(lo, float(al[max(0, bi - 1)]))
        chi = min(hi, float(al[min(nc - 1, bi + 1)]))
        for _ in range(4):
            pts = np.linspace(clo, chi, 9)
            vs = np.array([ls_eval(q, b, a, float(pts[i])) for i in range(9)])
            bi2 = int(np.argmin(vs))
            if vs[bi2] < best_f:
                best_f, best_a = float(vs[bi2]), float(pts[bi2])
            if bi2 == 0:
                chi = float(pts[1])
            elif bi2 == 8:
                clo = float(pts[7])
            else:
                clo, chi = float(pts[bi2 - 1]), float(pts[bi2 + 1])
    S = n / 2.0
    trial = h + best_a * z
    trial = project_K(trial, n, S)
    if not is_feasible(trial, n, S):
        return h, q, float(np.max(q))
    s = exact_score(trial, dx)
    cur_s = float(np.max(q))
    if np.isfinite(s) and s < cur_s - 1e-12:
        return trial, lag_vals(trial, dx), s
    return h, q, cur_s

def sparse_dir(rng, n, h):
    nz = min(12, n)
    supp = rng.choice(n, size=nz, replace=False)
    vals = rng.standard_normal(nz)
    vals -= vals.mean()
    vals[-1] = -vals[:-1].sum()
    m = np.abs(vals).max()
    if m < 1e-14:
        return None, None
    z = np.zeros(n)
    for i in range(nz):
        z[supp[i]] = vals[i] / m
    z.sum()  # should be ~0
    lo, hi = -1e30, 1e30
    pos = z > 1e-15
    neg = z < -1e-15
    if np.any(pos):
        lo = max(lo, float(np.max(-h[pos] / z[pos])))
        hi = min(hi, float(np.min((1.0 - h[pos]) / z[pos])))
    if np.any(neg):
        lo = max(lo, float(np.max((1.0 - h[neg]) / z[neg])))
        hi = min(hi, float(np.min(-h[neg] / z[neg])))
    if hi - lo < 1e-14:
        return None, None
    return z, (lo, hi)

def local_descent(h0, n, dx, rng, max_sweeps=3, n_sparse=8):
    h = h0.copy()
    S = n / 2.0
    q = lag_vals(h, dx)
    sc = float(np.max(q))
    pairs = [(p, r) for r in range(1, n) for p in range(r)]
    for sw in range(max_sweeps):
        changed = False
        rng.shuffle(pairs)
        for p, r in pairs:
            lo = max(-h[p], h[r] - 1.0)
            hi = min(1.0 - h[p], h[r])
            if hi - lo < 1e-14:
                continue
            b = grad_col(p, h, n, dx) - grad_col(r, h, n, dx)
            a = np.zeros(2 * n - 1)
            a[n - 1] = -2.0 * dx
            d = abs(p - r)
            if d > 0:
                a[n - 1 + d] += dx
                a[n - 1 - d] += dx
            z = np.zeros(n)
            z[p], z[r] = 1.0, -1.0
            new_h, new_q, new_s = line_search(h, q, b, a, lo, hi, n, dx, z)
            if new_s < sc - 1e-12:
                h, q, sc = new_h, new_q, new_s
                changed = True
        for _ in range(n_sparse):
            z, (lo, hi) = sparse_dir(rng, n, h)
            if z is None:
                continue
            b = dx * (np.correlate(z, 1.0 - h, mode='full') - np.correlate(h, z, mode='full'))
            a = -dx * np.correlate(z, z, mode='full')
            if not (np.all(np.isfinite(b)) and np.all(np.isfinite(a))):
                continue
            new_h, new_q, new_s = line_search(h, q, b, a, lo, hi, n, dx, z)
            if new_s < sc - 1e-12:
                h, q, sc = new_h, new_q, new_s
                changed = True
        if not changed and sw >= 1:
            break
    return h, sc

def build_starts(base, n, S, dx, rng):
    starts = [base.copy(), (1.0 - base).copy(), base[::-1].copy(), (1.0 - base[::-1]).copy()]
    starts.append(project_K(base + 0.05 * rng.standard_normal(n), n, S))
    starts.append(project_K(base + 0.15 * rng.standard_normal(n), n, S))
    starts.append(project_K(rng.uniform(0, 1, n), n, S))
    starts.append(project_K(rng.beta(0.5, 0.5, n), n, S))
    j = np.arange(n)
    starts.append(project_K(0.5 + 0.45 * np.cos(2.0 * math.pi * j / n), n, S))
    blk = np.zeros(n)
    k1 = int(math.floor(S))
    blk[:k1] = 1.0
    rem = S - k1
    if rem > 1e-12:
        blk[k1] = rem
    blk = blk[rng.permutation(n)]
    starts.append(project_K(blk, n, S))
    dedup = []
    for s in starts:
        if not is_feasible(s, n, S):
            continue
        close = False
        for ds in dedup:
            if np.max(np.abs(s - ds)) <= 1e-7:
                close = True
                break
        if not close:
            dedup.append(s)
    return dedup[:10] if n <= 64 else dedup[:6]

    fb_h = base.copy()
    fb_s = exact_score(fb_h, dx)
    best_h, best_s = base.copy(), fb_s
    starts = build_starts(base, n, S, dx, rng)
    ms = 3 if n <= 64 else 2
    ns = 8 if n <= 64 else 6
    for s in starts:
        sc = exact_score(s, dx)
        if sc < best_s - 1e-12:

        hopt, sc = local_descent(s, n, dx, rng, ms, ns)
        if sc < best_s - 1e-12:

    for sig in (0.04, 0.12):
        shake = project_K(best_h + sig * rng.standard_normal(n), n, S)
        if is_feasible(shake, n, S):

    best_h = project_K(best_h, n, S)
