                        v = project_capped(b2 + rng.normal(0, 0.12 * sc, n), n)
                    elif r < 0.875:
                        v = project_capped(rng.uniform(0, 1, n), n)
                    else:
                        v = project_capped(make_binary(rng, n), n)
                    pop[j] = v; scores[j] = exact_score(v, n)
                stall = 0
    return best_gv, best_gs

def pattern_poll(hb, sb, pop, scs, n, rng, rounds, ndirs):
    sc = max(0.5, min(1.5, math.sqrt(47.0 / n)))
    delta = min(0.25, 0.12 * sc)
    order = np.argsort(scs)
    top = [pop[j].copy() for j in order[:max(2, len(pop) // 4)]]
    if np.max(np.abs(top[0] - hb)) > 1e-10:
        top.append(hb.copy())
    for _ in range(rounds):
        if delta < 1e-4:
            break
        bt, bt_s = None, sb
        for _d in range(ndirs):
            if rng.random() < 0.5:
                v = rng.normal(0, 1, n)
            else:
                i1, i2 = int(rng.integers(len(top))), int(rng.integers(len(top)))
                v = top[i1] - top[i2]
            v = v - np.mean(v)
            mx = np.max(np.abs(v))
            if mx < 1e-14 or not np.isfinite(mx):
                continue
            v = v / mx
            p = v > 1e-14; q = v < -1e-14
            tlo, thi = -1e6, 1e6
            if np.any(p):
                tlo = max(tlo, float(np.max(-hb[p] / v[p])))
                thi = min(thi, float(np.max((1.0 - hb[p]) / v[p])))
            if np.any(q):
                tlo = max(tlo, float(np.min((1.0 - hb[q]) / v[q])))
                thi = min(thi, float(np.max(-hb[q] / v[q])))
            if tlo >= thi - 1e-12:
                continue
            tsts = []
            for tt in (-delta, -0.5 * delta, 0.5 * delta, delta):
                tt2 = max(tlo + 1e-14, min(thi - 1e-14, tt))
                if abs(tt2) < 1e-12:
                    continue
                if not any(abs(tt2 - w) < 1e-12 for w in tsts):
                    tsts.append(tt2)
            for tt in tsts:
                tr = project_capped(hb + tt * v, n)
                if tr is None:
                    continue
                ts = exact_score(tr, n)
                if ts < bt_s - 1e-12:
                    bt_s = ts; bt = tr
        if bt is not None:
            hb, sb = bt, bt_s
            delta = min(0.25, 1.2 * delta)
        else:
            delta *= 0.5
    return hb, sb

def run(seed=42, **kwargs):
    n = None; parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None:
        n = 47
        try:
            vk = int(kwargs.get('n_points', 47))
            if 1 <= vk <= 999: n = vk
        except Exception:
            pass
    if parent is not None:
        parent = np.clip(parent, 0.0, 1.0)
        fb = project_capped(parent, n)
    else:
        fb = np.full(n, 0.5)
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    fb = fb.copy()
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        h1 = np.array([0.5])
        return h1, float(exact_score(h1, 1)), 1
    if n <= 64:
        M, G, SL, NR, PR, PD = 40, 120, 20, 3, 28, 10
    elif n <= 128:
        M, G, SL, NR, PR, PD = 36, 90, 16, 2, 20, 8
    elif n <= 256:
        M, G, SL, NR, PR, PD = 28, 55, 12, 1, 12, 6
    elif n <= 512:
        M, G, SL, NR, PR, PD = 20, 30, 10, 1, 8, 4
    else:
        M, G, SL, NR, PR, PD = 14, 18, 8, 1, 5, 4
    pool = build_pool(fb, n, rng, M)
    scores = np.array([exact_score(pv, n) for pv in pool], dtype=np.float64)
    pop = [pv.copy() for pv in pool]
    bg, bs = de_evolve(pop, scores, n, rng, G, SL, NR)
    bg, bs = pattern_poll(bg, bs, pop, scores, n, rng, PR, PD)
    bg = project_capped(bg, n)
    fb_s = exact_score(fb, n)
    if bg is None or not is_feasible(bg, n):
        bg, bs = fb.copy(), fb_s
    else:
        bs2 = exact_score(bg, n)
        if bs2 < fb_s - 1e-12:
            bs = bs2
        else:
            bg, bs = fb.copy(), fb_s
    return bg.astype(np.float64), float(bs), int(n)
