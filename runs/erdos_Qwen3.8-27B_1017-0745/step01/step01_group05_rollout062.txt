def exact_score(h, n):

        return float('inf')
    qs = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(np.max(qs)) if np.all(np.isfinite(qs)) else float('inf')

def smooth_score(h, n, tau):
    qs = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = qs.max()
    return float(m + tau * np.log(np.sum(np.exp((qs - m) / tau))))

def smooth_grad(h, n, tau):
    dx = 2.0 / n
    qs = np.correlate(h, 1.0 - h, mode='full') * dx
    m = qs.max()
    w = np.exp((qs - m) / tau)
    p = w / w.sum()
    g = np.zeros(n)
    for l in range(2 * n - 1):
        if p[l] < 1e-15: continue

        lo1, hi1 = max(0, k), min(n, n + k)
        if hi1 > lo1:
            g[lo1:hi1] += p[l] * dx * (1.0 - h[lo1 - k:hi1 - k])
        lo2, hi2 = max(0, -k), min(n, n - k)
        if hi2 > lo2:
            g[lo2:hi2] -= p[l] * dx * h[lo2 + k:hi2 + k]
    return g

def build_G(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        lo = max(0, k)
        hi = min(n, n + k)
        if hi <= lo: continue
        row = k + n - 1
        G[row, lo:hi] += dx * (1.0 - h[lo - k:hi - k])
        lo2, hi2 = max(0, -k), min(n, n - k)
        if hi2 > lo2:
            G[row, lo2:hi2] -= dx * h[lo2 + k:hi2 + k]
    return G

def project_simplex(z):
    m = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(m, 1.0 / m)
    sz = np.sort(z)[::-1]
    cs = np.cumsum(sz)
    t = np.arange(1, m + 1, dtype=float)
    rho = np.where(sz - (cs - 1.0) / t > 0)[0]
    if len(rho) == 0:
        return np.full(m, 1.0 / m)
    r = rho[-1]
    theta = (cs[r] - 1.0) / (r + 1)
    p = np.maximum(z - theta, 0.0)
    s = p.sum()
    if s > 0: p /= s
    else: p = np.full(m, 1.0 / m)
    return p

def solve_dual(h, q, G, n, max_inner):

    L = 2.0 * dx
    nlag = len(q)
    Ld = max(1e-12, 1.5 * float(np.sum(G * G)) / L)
    qm = float(q.max())
    act = q >= qm - max(1e-12, 1e-8 * max(1.0, abs(qm)))
    if not act.any():
        act = np.zeros(nlag, dtype=bool)
        act[np.argmax(q)] = True
    p = np.zeros(nlag)
    p[act] = 1.0 / act.sum()
    x = p.copy()
    yv = p.copy()
    th = 1.0
    best_d = np.zeros(n)
    best_D = -1e100
