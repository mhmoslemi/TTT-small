- q_s(h) = dx * sum_{i in I_s} h[i](1 - h[i+s])
- ∂q_s/∂h[k] = dx * (sum over i where i=k: (1-h[k+s]) + sum over i where i+s=k: (1-h[i]) but with negative sign... )

Let me be more careful:
q_s = dx * sum_i h[i](1-h[i+s])

∂q_s/∂h[k] for a fixed k:
- If k appears as i (first factor): contributes dx * (1 - h[k+s]) for the term where i=k
- If k appears as i+s (second factor): contributes dx * (-h[i]) = dx * (-h[k-s]) for the term where i=k-s

So ∂q_s/∂h[k] = dx * [(1-h[k+s]) * 1_{k+s valid} - h[k-s] * 1_{k-s valid}]

For the Hessian:
∂²q_s/∂h[k]∂h[m]:
- From the (1-h[k+s]) term: -dx if m = k+s (for the i=k term)
- From the -h[k-s] term: -dx if m = k-s (for the i=k-s term)

So Hessian has -dx at positions (k, k+s) and (k+s, k) for valid pairs, plus -2dx on the diagonal for s=0.

For s≠0: the Hessian is a graph with edges between i and i+s. Each node has degree at most 2 (connected to i-s and i+s). So the spectral norm is at most 2*dx (by the maximum degree bound for adjacency-like matrices).

For s=0: Hessian = -2dx * I, spectral norm = 2dx.

So L = 2dx = 4/n is a valid upper bound on the spectral norm of all lag Hessians.

The majorization: q_s(y) ≤ q_s(x) + ∇q_s(x)·(y-x) + (L/2)||y-x||²

This is correct by Taylor's theorem with the Hessian bound.

        q += h[i] * (1.0 - h[j])
        g[i] += (1.0 - h[j])
        g[j] -= h[i]
    return dx * q, dx * g

class MMSub:
    def __init__(self, x, n, qv, G):
        self.x = x
        self.n = n
        self.L = 4.0 / n
        self.qv = qv
        self.G = G
    def obj(self, z):
        d = z[:self.n] - self.x
        return 0.5 * self.L * float(np.dot(d, d)) + z[-1]
    def obj_grad(self, z):
        g = np.zeros(self.n + 1)
        g[:self.n] = self.L * (z[:self.n] - self.x)
        g[-1] = 1.0
        return g
    def ineq(self, z):
        return z[-1] - self.qv - self.G @ (z[:self.n] - self.x)
    def ineq_jac(self, z):
        J = np.zeros((len(self.qv), self.n + 1))
        J[:, :self.n] = -self.G
        J[:, -1] = 1.0
        return J
    def eq(self, z):
        return float(np.sum(z[:self.n])) - 0.5 * self.n
    def eq_jac(self, z):
        g = np.zeros(self.n + 1)
        g[:self.n] = 1.0
        return g

def build_all_lags(h, n):
    qv = np.zeros(2 * n - 1)
    G = np.zeros((2 * n - 1, n))
    for idx, s in enumerate(range(-(n - 1), n)):
        qv[idx], G[idx] = lag_val_grad(h, s, n)
    return qv, G

def solve_mm_sub(x, n):
    qv, G = build_all_lags(x, n)
    sub = MMSub(x, n, qv, G)
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    cons = [{'type': 'ineq', 'fun': sub.ineq, 'jac': sub.ineq_jac},
            {'type': 'eq', 'fun': sub.eq, 'jac': sub.eq_jac}]
    z0 = np.zeros(n + 1)
    z0[:n] = x
    z0[-1] = float(np.max(qv)) + 0.005
    try:
        res = minimize(sub.obj, z0, jac=sub.obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-13, 'disp': False})
        if not np.all(np.isfinite(res.x)):
            return x.copy()
        y = project_box_sum(res.x[:n], 0.5 * n)
        if y is None:
            return x.copy()
        return y
    except Exception:
        return x.copy()

def local_mm(h0, n, mx):
    h = h0.copy()
    c = exact_score(h, n)
    for _ in range(mx):
        old_c = c
        y = solve_mm_sub(h, n)
        if not np.all(np.isfinite(y)) or float(np.max(np.abs(y - h))) < 1e-12:
            break
        found = False
        for k in range(9):
            a = 2.0 ** (-k)
            cand = project_box_sum(h + a * (y - h), 0.5 * n)
            if cand is None:
                continue
            sc = exact_score(cand, n)
            if sc < c - 1e-13:
                h = cand
                c = sc
                found = True
                break
        if not found:
            break
        if old_c - c < 1e-11 * max(1.0, abs(old_c)):
            break
    return h, c

def get_parent():
    n_default = 45
    if 'initial_h_values' in globals():
        ph = np.asarray(globals()['initial_h_values'], dtype=np.float64)
        if ph.ndim == 1 and 2 <= len(ph) <= 999 and np.all(np.isfinite(ph)):
            if np.all(ph >= -1e-6) and np.all(ph <= 1.0 + 1e-6):
                n = len(ph)
                p = project_box_sum(np.clip(ph, 0.0, 1.0), 0.5 * n)
                if p is not None:
                    return p, n
    return project_box_sum(np.full(n_default, 0.5), 0.5 * n_default), n_default

def build_starts(parent, n, rng):
    starts = [parent.copy(), parent[::-1].copy()]
    for sd in [0.05, 0.10, 0.18, 0.30]:
        p = project_box_sum(parent + sd * rng.standard_normal(n), 0.5 * n)
        if p is not None:
            starts.append(p)
    p = project_box_sum(rng.random(n), 0.5 * n)
    if p is not None:
        starts.append(p)
    i = np.arange(n)
    u = 0.5 + 0.25 * np.cos(2.0 * np.pi * i / n + 0.7) + 0.15 * np.sin(4.0 * np.pi * i / n + 1.3)
    p = project_box_sum(u, 0.5 * n)
    if p is not None:
        starts.append(p)
    i2 = np.arange(n)
    u2 = 0.5 + 0.20 * np.cos(6.0 * np.pi * i2 / n) + 0.10 * np.sin(2.0 * np.pi * i2 / n + 2.0)
    p2 = project_box_sum(u2, 0.5 * n)
    if p2 is not None:
        starts.append(p2)
    unique = [starts[0]]
    for s in starts[1:]:
        dup = False
        for u in unique:
            if float(np.max(np.abs(s - u))) < 1e-10:
                dup = True
                break
        if not dup:
            unique.append(s)
    return unique

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent, n = get_parent()
    best_h = parent.copy()
    best_c5 = exact_score(parent, n)
    mx = max(10, min(40, 1800 // n))
    starts = build_starts(parent, n, rng)
    for h0 in starts:
        h, c = local_mm(h0, n, mx)
        if c < best_c5:
            best_h = h.copy()
            best_c5 = c
    for sd in [0.04, 0.10, 0.22]:
        p = project_box_sum(best_h + sd * rng.standard_normal(n), 0.5 * n)
        if p is None:
            continue
        h, c = local_mm(p, n, max(5, mx // 2))
        if c < best_c5:
            best_h = h.copy()
            best_c5 = c
    for sd in [0.03, 0.08]:
        p = project_box_sum(best_h + sd * rng.standard_normal(n), 0.5 * n)
        if p is None:
            continue
        h, c = local_mm(p, n, max(4, mx // 3))
        if c < best_c5:
            best_h = h.copy()
            best_c5 = c
    best_h = project_box_sum(best_h, 0.5 * n)
    if best_h is not None and check_feasible(best_h, n):
        sc = exact_score(best_h, n)
        if sc < float('inf'):
            return (best_h, sc, n)
    fb = project_box_sum(parent, 0.5 * n)
    if fb is not None and check_feasible(fb, n):
        return (fb, exact_score(fb, n), n)
    fb2 = np.full(n, 0.5)
    return (fb2, exact_score(fb2, n), n)
