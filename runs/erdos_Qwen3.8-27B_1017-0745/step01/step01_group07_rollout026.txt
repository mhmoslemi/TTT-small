import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix, eye as sp_eye, hstack, vstack

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n * 0.5) <= 1e-10)

def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):
        return None
    if np.max(np.abs(z)) > 1e12:
        return None
    n = len(z)
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(80):
        mid = lo + 0.5 * (hi - lo)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)
    r = S - np.sum(y)
    if r > 1e-12:
        for i in range(n):
            if r <= 0: break
            a = min(r, 1.0 - y[i]); y[i] += a; r -= a
    elif r < -1e-12:
        for i in range(n):
            if r >= 0: break
            a = min(-r, y[i]); y[i] -= a; r += a
    if abs(np.sum(y) - S) > 1e-10 or not np.all(np.isfinite(y)):
        return None
    return y

def exact_lags(h, n):
    vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if len(vals) != 2 * n - 1 or not np.all(np.isfinite(vals)):
        return None
    return vals[::-1]

def exact_score(h, n):
    v = exact_lags(h, n)
    if v is None:
        return None
    return float(np.max(v))

def build_lag_model(h, n):
    q = exact_lags(h, n)
    if q is None:
        return None, None
    dx = 2.0 / n
    m = 2 * n - 1
    rows = []; cols = []; data = []
    for d in range(-(n - 1), n):
        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        for i in range(lo, hi):
            rows.append(row); cols.append(i); data.append(dx * (1.0 - h[i + d]))
            rows.append(row); cols.append(i + d); data.append(-dx * h[i])
    G = coo_matrix((data, (rows, cols)), shape=(m, n)).tocsr()
    return q, G

def solve_lp_step(x, q, G, rho, n):
    m = 2 * n - 1
    dx = 2.0 / n
    mu = 0.002 * dx
    I = sp_eye(n, format='csr')
    Zmn = csr_matrix((m, n))
    Zm1 = csr_matrix((m, 1))
    Zn1 = csr_matrix((n, 1))
    neg_ones_m = csr_matrix(-np.ones((m, 1)))
    A1 = hstack([G, Zmn, neg_ones_m], format='csr')
    A2 = hstack([I, -I, Zn1], format='csr')
    A3 = hstack([-I, -I, Zn1], format='csr')
    A_ub = vstack([A1, A2, A3], format='csr')
    b_ub = np.concatenate([-q, np.zeros(n), np.zeros(n)])
    ones_row = csr_matrix(np.ones((1, n)))
    Z1n = csr_matrix((1, n))
    Z11 = csr_matrix((1, 1))
    A_eq = hstack([ones_row, Z1n, Z11], format='csr')
    b_eq = np.array([0.0])
    c = np.zeros(2 * n + 1)
    c[n:2 * n] = mu
    c[2 * n] = 1.0
    lb = np.maximum(-rho, -x)
    ub = np.minimum(rho, 1.0 - x)
    bounds = list(zip(lb, ub)) + [(0.0, None)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res is not None and res.success:
            delta = np.asarray(res.x[:n], dtype=np.float64)
            if np.all(np.isfinite(delta)) and abs(np.sum(delta)) < 1e-5:
                return delta
    except Exception:
        pass
    return None

def optimize_trajectory(x0, n, rng, best_h, best_score):
    S = n * 0.5
    x = x0.copy()

    rho = 0.08

    max_iter = 30 if n <= 128 else (22 if n <= 256 else (14 if n <= 512 else 6))
    consec_fail = 0
    for _ in range(max_iter):
        if consec_fail >= 4 or rho < rho_min:
            break
        q, Gmat = build_lag_model(x, n)
        if q is None:
            break
        delta = solve_lp_step(x, q, Gmat, rho, n)
        if delta is None:
            rho = max(rho_min, rho * 0.5)
            consec_fail += 1
            continue
        if np.max(np.abs(delta)) < 1e-9:
            break
        best_x_new = None
        best_f_new = None
        for j in range(9):
            beta = 2.0 ** (-j)
            x_try = project_capped_simplex(x + beta * delta, S)
            if x_try is None or not is_feasible(x_try, n):
                continue
            s_try = exact_score(x_try, n)
            if s_try is not None and np.isfinite(s_try) and s_try < f - 1e-12:
                if best_f_new is None or s_try < best_f_new - 1e-14:
                    best_f_new = s_try
                    best_x_new = x_try
        if best_x_new is None:
            rho = max(rho_min, rho * 0.5)
            consec_fail += 1
            continue
        s_step = best_x_new - x
        pred_dec = f - float(np.max(q + Gmat.dot(s_step)))
        act_dec = f - best_f_new
        x = best_x_new
        f = best_f_new
        consec_fail = 0
        if f < best_score - 1e-12:
            best_score = f
            best_h = x.copy()
        if pred_dec > 1e-12:
            ratio = act_dec / pred_dec
            if ratio < 0.25:
                rho = max(rho_min, rho * 0.5)
            elif ratio > 0.75 and np.max(np.abs(s_step)) >= 0.8 * rho:
                rho = min(rho_max, rho * 1.5)
        else:
            rho = max(rho_min, rho * 0.5)
        if np.max(np.abs(s_step)) < 1e-8:
            break
    return best_h, best_score

def make_lowfreq(n, rng):
    idx = np.arange(n, dtype=np.float64)
    u = np.full(n, 0.5)
    M = min(5, (n - 1) // 2)
    for m in range(1, M + 1):
        amp = 0.10 / np.sqrt(m) * rng.normal()
        phi = rng.uniform(0.0, 2.0 * np.pi)
        u += amp * np.cos(2.0 * np.pi * m * (idx + 0.5) / n + phi)
    return u

def build_starts(n, S, parent, rng):
    starts = []
    if parent is not None:
        starts.append(parent.copy())
        starts.append(project_capped_simplex(parent + rng.normal(0, 0.04, n), S))
        starts.append(project_capped_simplex(parent + rng.normal(0, 0.12, n), S))
        lf = make_lowfreq(n, rng)
        pc = project_capped_simplex(lf, S)
        if pc is not None:
            starts.append(pc)
        starts.append(1.0 - parent)
        starts.append(project_capped_simplex(1.0 - parent + rng.normal(0, 0.06, n), S))
    else:
        starts.append(np.full(n, 0.5))
        lf = make_lowfreq(n, rng)
        pc = project_capped_simplex(lf, S)
        if pc is not None:
            starts.append(pc)
        starts.append(project_capped_simplex(rng.uniform(0, 1, n), S))
    deduped = []
    for s in starts:
        if s is None or not is_feasible(s, n):
            continue
        dup = False
        for d in deduped:
            if np.max(np.abs(s - d)) < 1e-9:
                dup = True
                break
        if not dup:
            deduped.append(s)
    max_starts = 6 if n <= 128 else (4 if n <= 256 else 3)
    return deduped[:max_starts]
