import numpy as np

def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):
        return None
    if np.max(np.abs(z)) > 1e12:
        return None
    n = len(z)
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    x = np.clip(z - lam, 0.0, 1.0)
    r = S - np.sum(x)
    if r > 0:
        for i in range(n):
            if r <= 0: break
            a = min(r, 1.0 - x[i])
            x[i] += a
            r -= a
    elif r < 0:
        for i in range(n):
            if r >= 0: break
            a = min(-r, x[i])
            x[i] -= a
            r += a
    if abs(np.sum(x) - S) > 1e-10 or not np.all(np.isfinite(x)):
        return None
    return x

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-15) and np.all(h <= 1.0 + 1e-15)
            and abs(np.sum(h) - n / 2.0) <= 1e-9)

def exact_score(h, n):
    vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(vals)):
        return None
    return float(np.max(vals))

def exact_lags(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def pair_line_search(h, q, p, r_idx, n, dx):

    U = min(1.0 - h[p], h[r_idx])
    if U - L < 1e-12:

    m = 2 * n - 1
    b = np.zeros(m)
    a = np.zeros(m)
    idx1 = np.arange(n) - p + n - 1
    b[idx1] += dx * (1.0 - h)
    idx2 = p - np.arange(n) + n - 1
    b[idx2] += -dx * h
    idx3 = np.arange(n) - r_idx + n - 1
    b[idx3] += -dx * (1.0 - h)
    idx4 = r_idx - np.arange(n) + n - 1
    b[idx4] += dx * h
    a[n - 1] = -2.0 * dx
    a[n - 1 + r_idx - p] = dx
    a[n - 1 + p - r_idx] = dx
    # Coarse mesh
    t_pts = np.linspace(L, U, 17)
    if 0.0 < L or 0.0 > U:
        pass
    t_pts = np.unique(np.concatenate([t_pts, [0.0, L, U]]))
    t_pts = t_pts[(t_pts >= L - 1e-15) & (t_pts <= U + 1e-15)]
    t_pts = np.clip(t_pts, L, U)
    # Evaluate f(t) = max(q + b*t + a*t^2)
    best_t = t_pts[0]
    best_f = np.max(q + b * t_pts[0] + a * t_pts[0] ** 2)

        proj = project_capped_simplex(best_h, S)
        if proj is not None:
            best_h = proj
        else:
            best_h = h_const.copy()
            best_score = 0.5
    fin_sc = exact_score(best_h, n)
    if fin_sc is None or not np.isfinite(fin_sc):
        if parent is not None:
            psc = exact_score(parent, n)
            if psc is not None and np.isfinite(psc):
                best_h, best_score = parent.copy(), psc
            else:
                best_h, best_score = h_const.copy(), 0.5
        else:
            best_h, best_score = h_const.copy(), 0.5
    else:
        best_score = fin_sc
    return best_h.astype(np.float64), float(best_score), n
