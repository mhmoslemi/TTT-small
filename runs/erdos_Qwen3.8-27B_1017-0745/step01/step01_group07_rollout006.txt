import numpy as np

def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)):
        return None
    n = len(z)
    if S < 0 or S > n or np.max(np.abs(z)) > 1e12:
        return None
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(80):
        mid = lo + 0.5 * (hi - lo)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = lo + 0.5 * (hi - lo)
    y = np.clip(z - lam, 0.0, 1.0)
    r = S - np.sum(y)
    if r > 1e-13:
        for i in range(n):
            if r <= 1e-13: break
            a = min(r, 1.0 - y[i]); y[i] += a; r -= a
    elif r < -1e-13:
        for i in range(n):
            if r >= -1e-13: break
            a = min(-r, y[i]); y[i] -= a; r += a
    y = np.clip(y, 0.0, 1.0)
    if abs(np.sum(y) - S) > 1e-9 or not np.all(np.isfinite(y)):
        return None
    return y

    v = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(v)):
        return None
    return float(np.max(v))

def sof_obj_grad(h, tau, dx, la, ra, li, n):
    nl = 2 * n - 1
    w = h[la] * (1.0 - h[ra])
    q = dx * np.bincount(li, weights=w, minlength=nl)
    m = np.max(q)
    e = np.exp((q - m) / tau)
    se = np.sum(e)
    if se <= 0 or not np.isfinite(se):
        return None, None
    F = m + tau * np.log(se)
    p = e / se
    gl = np.bincount(la, weights=p[li] * (1.0 - h[ra]), minlength=n)
    gr = np.bincount(ra, weights=-p[li] * h[la], minlength=n)
    g = dx * (gl + gr)
    if not np.isfinite(F) or not np.all(np.isfinite(g)):
        return None, None
    return F, g
