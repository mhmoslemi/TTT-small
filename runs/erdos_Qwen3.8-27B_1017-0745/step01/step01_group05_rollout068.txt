def project_capped(y, n):

import numpy as np
import math

def project_capped(y, n):
    S = 0.5 * n
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return np.full(n, 0.5), False
    lo = float(y.min()) - 1.0
    hi = float(y.max())
    lam = float(np.clip(np.mean(y) - 0.5, lo, hi))
    for _ in range(50):
        c = np.clip(y - lam, 0.0, 1.0)
        r = c.sum() - S
        if abs(r) <= 1e-13:
            break
        if r > 0:
            lo = lam
        else:
            hi = lam
        a = int(np.sum((y - lam > 1e-15) & (y - lam < 1.0 - 1e-15)))
        if a > 0:
            prop = lam + r / a
        else:
            prop = 0.5 * (lo + hi)
        if not (lo + 1e-15 < prop < hi - 1e-15):
            prop = 0.5 * (lo + hi)
        lam = prop
    c = np.clip(y - lam, 0.0, 1.0).astype(np.float64)
    for _ in range(5):
        r = S - c.sum()
        if abs(r) < 1e-14:
            break
        mask = (c < 1.0 - 1e-12) if r > 0 else (c > 1e-12)
        cnt = int(mask.sum())
        if cnt == 0:
            break
        c[mask] += r / cnt
        np.clip(c, 0.0, 1.0, out=c)
    ok = np.all(c >= 0.0) and np.all(c <= 1.0) and abs(c.sum() - S) <= 1e-10
    return c, bool(ok)

def lag_scores(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    qs = lag_scores(h, n)
    if qs.shape[0] != 2 * n - 1 or not np.all(np.isfinite(qs)):
        return float('inf')
    return float(qs.max())

def feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            np.all(h >= 0.0) and np.all(h <= 1.0) and abs(h.sum() - n / 2.0) <= 1e-10)

def build_G(h, n):
    dx = 2.0 / n
    M = 2 * n - 1
    G = np.zeros((M, n), dtype=np.float64)
    for m in range(M):
        k = m - (n - 1)
        lo1, hi1 = max(0, k), min(n, n + k)
        if hi1 > lo1:
            G[m, lo1:hi1] += dx * (1.0 - h[lo1 - k:hi1 - k])
        lo2, hi2 = max(0, -k), min(n, n - k)
        if hi2 > lo2:
            G[m, lo2:hi2] -= dx * h[lo2 + k:hi2 + k]
    return G

def proj_simplex(z):
    n = len(z)
    if n == 0:
        return np.array([], dtype=np.float64)
    if not np.all(np.isfinite(z)):
        return np.full(n, 1.0 / n)
    sz = np.sort(z)[::-1]
    cs = np.cumsum(sz)
    rho = np.arange(1, n + 1)
    cond = sz > (cs - 1.0) / rho
    if not cond.any():
        return np.full(n, 1.0 / n)
    r = int(np.max(np.where(cond)))
    theta = (cs[r - 1] - 1.0) / r
    p = np.maximum(z - theta, 0.0)
    s = p.sum()
    if s > 1e-15:
        p /= s
    else:
        p = np.full(n, 1.0 / n)
    return p

def dual_eval(p, h, G, q, n, L):
    gp = G.T @ p
    y, ok = project_capped(h - gp / L, n)
    if not ok:
        return None
    d = y - h
    u = q + G @ d
    val = float(p @ u) + 0.5 * L * float(d @ d)
    gap = float(u.max()) - float(p @ u)
    return d, u, val, gap
