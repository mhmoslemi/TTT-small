    if abs(float(np.sum(h)) - n / 2.0) > 1e-9 * max(1, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_raw(h, n):
    L = 2 * n - 1
    v = np.zeros(L)
    G = np.zeros((L, n))
    for dd in range(-(n - 1), n):
        row = dd + n - 1
        lo = max(0, -dd)
        hi = min(n, n - dd)
        idx = np.arange(lo, hi)
        jdx = idx + dd
        v[row] = float(np.sum(h[idx] * (1.0 - h[jdx])))
        G[row, idx] += (1.0 - h[jdx])
        G[row, jdx] -= h[idx]
    return v, G

def obj_fn(z, n):
    return z[-1]

def obj_grad(z, n):
    g = np.zeros(z.shape[0])
    g[-1] = 1.0
    return g

def eq_fn(z, n):
    return np.array([float(np.sum(z[:n])) - n / 2.0])

def eq_grad(z, n):
    g = np.zeros((1, z.shape[0]))
    g[0, :n] = 1.0
    return g

def ineq_fn(z, n):
    d = 2.0 / n
    v, _ = lag_raw(z[:n], n)
    return z[-1] - d * v

def ineq_grad(z, n):
    d = 2.0 / n
    L = 2 * n - 1
    _, G = lag_raw(z[:n], n)
    J = np.zeros((L, n + 1))
    J[:, :n] = -d * G
    J[:, n] = 1.0
    return J

def add_start(starts, v, n, seen):
    h2 = project_capped(v, n)
    if h2 is None:
        return
    for ex in seen:
        if float(np.max(np.abs(h2 - ex))) < 1e-8:
            return
    starts.append(h2)
    seen.append(h2)

def polish_pair(h, n, sweeps):
    h = h.copy()
    for _ in range(sweeps):
        changed = False
        for i in range(n):
            for j in range(i + 1, n):
                lo = max(-h[i], h[j] - 1.0)
                hi = min(1.0 - h[i], h[j])
                if lo >= hi:
                    continue
                cur = exact_score(h, n)
                best_d, best_sc = 0.0, cur
                trials = [lo, hi, 0.001, -0.001, 0.01, -0.01, 0.03, -0.03, 0.1, -0.1]
                for delta in trials:
                    if delta < lo or delta > hi:
                        continue
                    h[i] += delta
                    h[j] -= delta
                    sc = exact_score(h, n)
                    h[i] -= delta
                    h[j] += delta
                    if sc < best_sc - 1e-12:
                        best_sc, best_d = sc, delta
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    changed = True
        if not changed:
            break
    return h

def subgrad_refine(h, n, steps, lr0):
    h = h.copy()
    lr = lr0
    for _ in range(steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mx = float(np.max(corr))
        dg = np.zeros(n)
        for dd in range(-(n - 1), n):
            if corr[dd + n - 1] >= mx - 1e-10:
                lo = max(0, -dd)
                hi = min(n, n - dd)
                idx = np.arange(lo, hi)
                jdx = idx + dd
                dg[idx] += (1.0 - h[jdx])
                dg[jdx] -= h[idx]
        h_new = project_capped(h - lr * dg, n)
        if h_new is None:
            lr *= 0.5
            continue
        sc0 = exact_score(h, n)
        sc1 = exact_score(h_new, n)
        if sc1 < sc0 - 1e-14:
            h = h_new
        else:
            lr *= 0.5
            if lr < 1e-12:
                break
    return h

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 0 < len(iv) <= 999:
            if np.all(np.isfinite(iv)) and np.all(iv >= -1e-8) and np.all(iv <= 1.0 + 1e-8):
                n = len(iv)
                parent = iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
    n = max(1, min(n, 999))
    try:
        sd = int(seed)
    except Exception:
        sd = 42
    rng = np.random.default_rng(abs(sd) % (2**32))
    uniform = np.full(n, 0.5)
    if parent is not None:
        h0 = project_capped(parent, n)
        fallback = h0 if h0 is not None else uniform.copy()
    else:
        fallback = uniform.copy()
    fb_score = exact_score(fallback, n)
    if not np.isfinite(fb_score):
        fallback = uniform.copy()
        fb_score = exact_score(uniform, n)
    try:
        n_starts = max(1, min(int(kwargs.get('starts', 8)), 10))
    except Exception:
        n_starts = 8
    try:
        max_iter = max(0, min(int(kwargs.get('max_iter', 300)), 500))
    except Exception:
        max_iter = 300
    try:
        ftol = max(1e-12, min(1e-6, float(kwargs.get('ftol', 1e-10))))
    except Exception:
        ftol = 1e-10
    try:
        n_polish = max(0, min(int(kwargs.get('polish_sweeps', 2)), 3))
    except Exception:
        n_polish = 2
    if n > 256:
        max_iter = 0
        n_polish = 0
    starts, seen = [], []
    add_start(starts, fallback, n, seen)
    if parent is not None:
        add_start(starts, fallback[::-1], n, seen)
        add_start(starts, 1.0 - fallback, n, seen)
        add_start(starts, (1.0 - fallback)[::-1], n, seen)
        for ns_ in (0.03, 0.10, 0.20):
            add_start(starts, fallback + rng.normal(0.0, ns_, n), n, seen)
        add_start(starts, rng.uniform(0.0, 1.0, n), n, seen)
    else:
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        add_start(starts, blk, n, seen)
        alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
        add_start(starts, alt, n, seen)
        idx = np.arange(n, dtype=np.float64)
        add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * idx / n), n, seen)
        add_start(starts, 0.5 + 0.35 * np.cos(4.0 * np.pi * idx / n + 0.5), n, seen)
        add_start(starts, rng.uniform(0.0, 1.0, n), n, seen)
    add_start(starts, uniform, n, seen)
    starts = starts[:n_starts]
    best, best_score = fallback.copy(), fb_score
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_score - 1e-12:
            best, best_score = st.copy(), ss
    if n <= 256 and max_iter > 0:
        bounds = [(0.0, 1.0)] * (n + 1)
        constraints = [
            {'type': 'eq', 'fun': eq_fn, 'jac': eq_grad},
            {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_grad},
        ]
        for st in starts:
            sc0 = exact_score(st, n)
            if not np.isfinite(sc0):
                continue
            z0 = np.concatenate([st, [min(1.0, sc0 + 1e-8)]])
            try:
                res = minimize(obj_fn, z0, args=(n,), method='SLSQP',
                               jac=obj_grad, bounds=bounds,
                               constraints=constraints,
                               options={'ftol': ftol, 'maxiter': max_iter})
            except Exception:
                continue
            if res is not None and hasattr(res, 'x') and res.x is not None:
                if len(res.x) == n + 1 and np.all(np.isfinite(res.x[:n])):
                    hp = project_capped(res.x[:n], n)
                    if hp is not None:
                        sp = exact_score(hp, n)
                        if np.isfinite(sp) and sp < best_score - 1e-12:
                            best, best_score = hp.copy(), sp
        hp2 = project_capped(best, n)
        if hp2 is not None:
            sp2 = exact_score(hp2, n)
            if np.isfinite(sp2) and sp2 < best_score - 1e-12:
                best, best_score = hp2, sp2
            z0 = np.concatenate([best, [min(1.0, best_score + 1e-8)]])
            for attempt in range(3):
                try:
                    res = minimize(obj_fn, z0, args=(n,), method='SLSQP',
                                   jac=obj_grad, bounds=bounds,
                                   constraints=constraints,
                                   options={'ftol': ftol * 0.1, 'maxiter': max_iter // 2})
                except Exception:
                    break
                if res is not None and hasattr(res, 'x') and res.x is not None:
                    if len(res.x) == n + 1 and np.all(np.isfinite(res.x[:n])):
                        hp3 = project_capped(res.x[:n], n)
                        if hp3 is not None:
                            sp3 = exact_score(hp3, n)
                            if np.isfinite(sp3) and sp3 < best_score - 1e-12:
                                best, best_score = hp3.copy(), sp3
                                z0 = np.concatenate([best, [min(1.0, best_score + 1e-8)]])
                            else:
                                break
                    else:
                        break
                else:
                    break
    if n <= 64:
        hr = subgrad_refine(best, n, 200, 0.02)
        hr = project_capped(hr, n)
        if hr is not None:
            sr = exact_score(hr, n)
            if np.isfinite(sr) and sr < best_score - 1e-12:
                best, best_score = hr, sr
    if n_polish > 0 and n <= 64:
        hp = polish_pair(best, n, n_polish)
        hp = project_capped(hp, n)
        if hp is not None:
            sp = exact_score(hp, n)
            if np.isfinite(sp) and sp < best_score - 1e-12:
                best, best_score = hp, sp
    final = project_capped(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final, fs = fallback.copy(), fb_score
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(uniform, n)
    return final.copy(), float(fs), int(n)
