    if np.max(np.abs(d)) < 1e-10:
        return x.copy(), exact_score(x, n)
    dd = float(d @ d)
    p = G @ d
    a, b = 0.0, 1.0
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    for _ in range(48):
        c1 = b - gr * (b - a)
        c2 = a + gr * (b - a)
        m1 = c1 * c1 * dd + float(np.max(Q + c1 * p))
        m2 = c2 * c2 * dd + float(np.max(Q + c2 * p))
        if m1 < m2:
            b = c2
        else:
            a = c1
    alpha_star = 0.5 * (a + b)
    best_h = x.copy()
    best_sc = exact_score(x, n)
    for al in [alpha_star, 1.0, 0.5, 0.25, 0.125, 0.75, 0.375]:
        if al < -1e-12 or al > 1.0 + 1e-12:
            continue
        al = max(0.0, min(1.0, al))
        h_c = x + al * d
        h_p = project_box_sum(h_c, S)
        if h_p is None or not check_feasible(h_p, n, S):
            continue
        sc = exact_score(h_p, n)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_h = h_p
    return best_h, best_sc

def mm_local(start, n, S):
    h_cur = project_box_sum(start, S)
    if h_cur is None or not check_feasible(h_cur, n, S):
        return None
    sc_cur = exact_score(h_cur, n)
    h_best = h_cur.copy()
    sc_best = sc_cur
    warm = None
    Omax = max(6, min(20, int(math.ceil(800.0 / n))))
    for _ in range(Omax):
        Q = exact_lags(h_cur, n)
        if Q is None:
            break
        G = build_G(h_cur, n)
        y_dir, warm = solve_surrogate(h_cur, Q, S, warm, n, G)
        if y_dir is None:
            break
        d = y_dir - h_cur
        h_new, sc_new = line_search(h_cur, d, Q, n, S, G)
        if sc_new < sc_cur - 1e-13:
            h_cur = h_new
            sc_cur = sc_new
            if sc_new < sc_best - 1e-13:
                h_best = h_cur.copy()
                sc_best = sc_new
            if sc_cur - sc_best < 1e-10 or np.max(np.abs(d)) < 1e-8:
                break
        else:
            break
    return h_best, sc_best

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    n = 45
    parent = None
    try:
        pv = np.array(initial_h_values, dtype=np.float64)
        if pv.ndim == 1 and 2 <= len(pv) <= 999 and np.all(np.isfinite(pv)):
            if np.all(pv >= -1e-9) and np.all(pv <= 1.0 + 1e-9):
                n = len(pv)
                tgt = n / 2.0
                if abs(np.sum(pv) - tgt) < 1e-6 * max(1.0, float(n)):
                    parent = project_box_sum(np.clip(pv, 0.0, 1.0), tgt)
    except Exception:
        pass
    S = n / 2.0
    if parent is None:
        parent = project_box_sum(np.full(n, 0.5), S)
    fallback = parent.copy()
    best_h = fallback.copy()
    best_score = exact_score(fallback, n)
    starts = [fallback.copy(), fallback[::-1].copy(), 1.0 - fallback]
    for sig in [0.03, 0.06, 0.10, 0.15]:
        c = project_box_sum(parent + sig * rng.standard_normal(n), S)
        if c is not None:
            starts.append(c)
    x = np.arange(n)
    for m in [1, 2, 3]:
        if m < n:
            ph = float(rng.uniform(0, 2.0 * np.pi))
            c = project_box_sum(0.5 + 0.3 * np.cos(2.0 * np.pi * m * x / n + ph), S)
            if c is not None:
                starts.append(c)
    blk = n // 3
    p2 = parent.copy()
    p2[blk:2 * blk] += 0.1
    p2[:blk] -= 0.05
    p2[2 * blk:] -= 0.05
    c = project_box_sum(p2, S)
    if c is not None:
        starts.append(c)
    p3 = parent.copy()
    p3[:n // 5] += 0.12
    p3[-n // 5:] += 0.12
    c = project_box_sum(p3, S)
    if c is not None:
        starts.append(c)
    dedup = [starts[0]]
    for s in starts[1:]:
        if not any(np.max(np.abs(s - d)) < 1e-10 for d in dedup):
            dedup.append(s)
    for h0 in dedup:
        try:
            res = mm_local(h0, n, S)
            if res is not None:
                h_r, sc_r = res
                if check_feasible(h_r, n, S) and sc_r < best_score - 1e-14:
                    best_score = sc_r
                    best_h = h_r.copy()
        except Exception:
            continue
    final_h = project_box_sum(best_h, S)
    if final_h is None or not check_feasible(final_h, n, S):
        final_h = fallback
    final_h = np.clip(final_h, 0.0, 1.0)
    final_score = exact_score(final_h, n)
    if final_score > best_score + 1e-14:
        final_h = fallback
        final_score = exact_score(final_h, n)
    return (np.array(final_h, dtype=np.float64), final_score, n)
