def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):

        else:

        return None

def raw_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def line_search(h, n, p, r, out):
    L = max(-h[p], h[r] - 1.0)
    U = min(1.0 - h[p], h[r])
    if U - L <= 1e-14:
        return None
    N = 2 * n - 1
    b = np.zeros(N)
    a = np.zeros(N)
    for d in range(-p, n - p):
        b[d + n - 1] += (1.0 - h[p + d])
    for d in range(p - n + 1, p + 1):
        b[d + n - 1] -= h[p - d]
    for d in range(-r, n - r):
        b[d + n - 1] -= (1.0 - h[r + d])
    for d in range(r - n + 1, r + 1):
        b[d + n - 1] += h[r - d]
    a[n - 1] = -2.0
    diff = r - p
    if abs(diff) < n:
        a[n - 1 + diff] = 1.0
        a[n - 1 - diff] = 1.0
    ts = np.linspace(L, U, 17)
    ts = np.unique(np.concatenate([ts, [0.0]]))
    ts = ts[(ts >= L - 1e-15) & (ts <= U + 1e-15)]
    best_t, best_f = 0.0, np.inf
    for t in ts:
        f = np.max(out + b * t + a * t * t)
        if f < best_f - 1e-14 or (abs(f - best_f) <= 1e-14 and abs(t) < abs(best_t)):
            best_f, best_t = f, t
    radius = (U - L) / 16.0
    for _ in range(4):
        lo_r = max(L, best_t - radius)
        hi_r = min(U, best_t + radius)
        for t in np.linspace(lo_r, hi_r, 5):
            f = np.max(out + b * t + a * t * t)
            if f < best_f - 1e-14 or (abs(f - best_f) <= 1e-14 and abs(t) < abs(best_t)):
                best_f, best_t = f, t
        radius /= 4.0
    return best_t, best_f

def polish_sweep(h, n, rng):
    out = raw_corr(h)
    score = float(np.max(out)) * (2.0 / n)
    P = min(n * (n - 1) // 2, 2 * n, max(24, 80000 // n))
    pairs = []
    n_adj = min(n - 1, max(1, P // 2))
    if n > 1:
        adj_idx = np.linspace(0, n - 2, n_adj).astype(int)
        for i in adj_idx:
            pairs.append((int(i), int(i) + 1))
    tries = 0
    while len(pairs) < P and tries < P * 5:
        i = int(rng.integers(n)); j = int(rng.integers(n))
        if i != j:
            i2, j2 = min(i, j), max(i, j)
            if (i2, j2) not in pairs:
                pairs.append((i2, j2))
        tries += 1
    rng.shuffle(pairs)
    for (p, r) in pairs:
        res = line_search(h, n, p, r, out)
        if res is None:
            continue
        t_best, _ = res
        h_new = h.copy()
        h_new[p] += t_best
        h_new[r] -= t_best
        if not is_feasible(h_new, n):
            h_new = project_capped_simplex(h_new, n / 2.0)
            if h_new is None:
                continue
        sc = exact_score(h_new, n)
        if sc is not None and sc < score - 1e-12:
            h = h_new
            score = sc
            out = raw_corr(h)
    return h, score

def anneal_trajectory(h0, n, rng, E, K):
    h = h0.copy()
    score = exact_score(h, n)
    if score is None:
        score = 0.5
    scale = 0.12
    for e in range(E):
        T = max(5e-5, 3e-3 * (0.4 ** e))
        acc = 0
        for _ in range(K):
            p = int(rng.integers(n)); r = int(rng.integers(n))
            if p == r:
                continue
            t = float(rng.uniform(-scale, scale))
            if h[p] + t < -1e-12 or h[p] + t > 1.0 + 1e-12:
                continue
            if h[r] - t < -1e-12 or h[r] - t > 1.0 + 1e-12:
                continue
            h_new = h.copy()
            h_new[p] += t
            h_new[r] -= t
            if not is_feasible(h_new, n):
                h_new = project_capped_simplex(h_new, n / 2.0)
                if h_new is None:
                    continue
            sc = exact_score(h_new, n)
            if sc is None:
                continue
            delta = sc - score
            if delta <= 0.0:
                h = h_new; score = sc; acc += 1
            elif delta < 700.0 * T and rng.random() < np.exp(-delta / T):
                h = h_new; score = sc; acc += 1
        ratio = acc / max(1, K)
        if ratio < 0.12:
            scale = min(0.4, 1.25 * scale)
        elif ratio > 0.55:
            scale = max(0.01, 0.75 * scale)
        h, score = polish_sweep(h, n, rng)
    return h, score

def build_starts(n, S, parent, rng):
    starts = []
    if parent is not None:
        starts.append(parent.copy())
    idx = np.arange(n, dtype=np.float64)
    M = min(4, max(1, (n - 1) // 2))
    u = np.full(n, 0.5)
    for m in range(1, M + 1):
        A = float(rng.normal(0.0, 0.12 / np.sqrt(m)))
        phi = float(rng.uniform(0.0, 2.0 * np.pi))
        u += A * np.cos(2.0 * np.pi * m * (idx + 0.5) / n + phi)
    pu = project_capped_simplex(u, S)
    if pu is not None and is_feasible(pu, n):
        starts.append(pu)
    perm = rng.permutation(n)
    binv = np.zeros(n)
    binv[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        binv[perm[n // 2]] = 0.5
    starts.append(binv.copy())
    urand = rng.random(n)
    pup = project_capped_simplex(urand, S)
    if pup is not None and is_feasible(pup, n):
        starts.append(pup)
    if parent is not None:
        starts.append(parent[::-1].copy())
        starts.append(1.0 - parent.copy())
    deduped = []
    for s in starts:
        dup = False
        for d in deduped:
            if np.max(np.abs(s - d)) < 1e-10:
                dup = True; break
        if not dup and is_feasible(s, n):
            deduped.append(s)
    return deduped

def run(seed=42, **kwargs):
    if seed is None:
        seed = 42
    try:
        seed = int(seed) % (2 ** 32)
    except (TypeError, ValueError):
        seed = 42
    rng = np.random.default_rng(seed)
    n = None
    parent = None
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        try:
            iv_arr = np.asarray(iv, dtype=np.float64)
            if iv_arr.ndim == 1 and 2 <= iv_arr.size <= 999:
                n = iv_arr.size
                if np.all(np.isfinite(iv_arr)) and np.max(np.abs(iv_arr)) <= 1e12:
                    if is_feasible(iv_arr, n):
                        parent = iv_arr.copy()
                    else:
                        proj = project_capped_simplex(iv_arr, n / 2.0)
                        if proj is not None and is_feasible(proj, n):
                            parent = proj
        except Exception:
            pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 80))
        except (TypeError, ValueError):
            n = 80
        if n < 8 or n > 999:
            n = 80
    S = n / 2.0
    h_const = np.full(n, 0.5)
    best_h = h_const.copy()
    best_score = exact_score(h_const, n)
    if best_score is None:
        best_score = 0.5
    if parent is not None:
        ps = exact_score(parent, n)
        if ps is not None and ps < best_score:
            best_score = ps
            best_h = parent.copy()
    if n <= 128:
        max_starts, E = 6, 5
    elif n <= 256:
        max_starts, E = 4, 4
    elif n <= 512:
        max_starts, E = 3, 3
    else:
        max_starts, E = 2, 2
    K = min(1000, max(80, 80000 // n))
    starts = build_starts(n, S, parent, rng)[:max_starts]
    for s in starts:
        sc = exact_score(s, n)
        if sc is not None and sc < best_score - 1e-12:
            best_score = sc
            best_h = s.copy()
    for s in starts:
        try:
            h_ans, sc_ans = anneal_trajectory(s, n, rng, E, K)
            if sc_ans is not None and sc_ans < best_score - 1e-12:
                best_score = sc_ans
                best_h = h_ans.copy()
        except Exception:
            pass
    for _ in range(3):
        try:
            h_fin, sc_fin = polish_sweep(best_h, n, rng)
            if sc_fin < best_score - 1e-12:
                best_score = sc_fin
                best_h = h_fin
        except Exception:
            break
    for sym_h in [best_h[::-1].copy(), 1.0 - best_h.copy()]:
        if is_feasible(sym_h, n):
            ss = exact_score(sym_h, n)
            if ss is not None and ss < best_score - 1e-12:
                best_score = ss
                best_h = sym_h.copy()
    if not is_feasible(best_h, n):
        best_h = project_capped_simplex(best_h, S)
        if best_h is None:
            best_h = h_const.copy()
            best_score = 0.5
    rv = exact_score(best_h, n)
    if rv is not None:
        best_score = rv
    if not is_feasible(best_h, n):
        best_h = h_const.copy()
        best_score = 0.5
    return best_h, best_score, n
