import numpy as np

def proj_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    n = len(z)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > n:
        return None
    if np.max(np.abs(z)) > 1e12:
        return None
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    y = np.clip(z - lam, 0.0, 1.0)
    r = S - np.sum(y)
    if r > 0:
        for i in range(n):
            if r <= 0: break
            a = min(r, 1.0 - y[i])
            y[i] += a; r -= a
    elif r < 0:
        for i in range(n):
            if r >= 0: break
            a = min(-r, y[i])
            y[i] -= a; r += a
    if abs(np.sum(y) - S) > 1e-10 or not np.all(np.isfinite(y)):
        return None
    return np.clip(y, 0.0, 1.0)

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-15) and np.all(h <= 1.0 + 1e-15)
            and abs(np.sum(h) - n / 2.0) <= 1e-9)

def compute_q(h, n):
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    if vals.shape[0] != 2*n - 1 or not np.all(np.isfinite(vals)):
        return None, None
    return vals, float(np.max(vals))

def exact_score(h, n):
    _, s = compute_q(h, n)
    return s

def line_search_pair(h, q, score, p, r, dx, n):
    L = max(-h[p], h[r] - 1.0)
    U = min(1.0 - h[p], h[r])
    if U - L <= 1e-12:
        return None
    m = 2 * n - 1
    b = np.zeros(m)
    a = np.zeros(m)
    a[n-1] = -2.0 * dx
    if p != r:
        a[n-1 + p - r] += dx
        a[n-1 + r - p] += dx
    for k in range(m):
        d = k - (n - 1)
        bk = 0.0
        # contribution from p
        if 0 <= p - d < n:
            bk += (1.0 - h[p - d])
            if p - d == r:
                bk += h[p]
        # contribution from r
        if 0 <= r - d < n:
            bk -= (1.0 - h[r - d])
            if r - d == p:
                bk -= h[r]
        b[k] = dx * bk  # this is the coefficient of t
    # Now q_k(t) = q[k] + b[k]*t + a[k]*t^2
    # Line search: 17 points + 4 refinements
    pts = np.linspace(L, U, 17)
    pts = np.concatenate([pts, [0.0]])
    best_t, best_f = 0.0, score
    for t in pts:
        f = np.max(q + b * t + a * t * t)
        if f < best_f - 1e-12:
            best_f, best_t = f, t
    # 4 refinements
    radius = (U - L) / 16.0
    for _ in range(4):
        lo = max(L, best_t - radius)
        hi = min(U, best_t + radius)
        pts2 = np.linspace(lo, hi, 5)
        for t in pts2:
            f = np.max(q + b * t + a * t * t)
            if f < best_f - 1e-12:
                best_f, best_t = f, t
        radius /= 4.0
    if best_f >= score - 1e-12:
        return None
    # Apply the move
    h_new = h.copy()
    h_new[p] += best_t
    h_new[r] -= best_t
    if not is_feasible(h_new, n):
        h_new = proj_simplex(h_new, n / 2.0)
        if h_new is None:
            return None
    q_new, sc_new = compute_q(h_new, n)
    if q_new is None or sc_new >= score - 1e-12:
        return None
    return h_new, q_new, sc_new

    rad = (U - L) / 16.0

        lo2, hi2 = max(L, best_t - rad), min(U, best_t + rad)
        for t in np.linspace(lo2, hi2, 5):
            f = float(np.max(q + b * t + a * t * t))
            if f < best_f - 1e-13:
                best_f, best_t = f, t
        rad /= 4.0
    if best_f >= score - 1e-12:
        return None
    h2 = h.copy()
    h2[p] += best_t
    h2[r] -= best_t
    if not is_feasible(h2, n):
        h2 = proj_simplex(h2, n * 0.5)
        if h2 is None:
            return None
    q2, s2 = compute_q(h2, n)
    if q2 is None or s2 is None or s2 >= score - 1e-12:
        return None
    return h2, q2, s2

def polish(h, q, score, dx, n, rng, max_sweeps=3):
    P = min(n * (n - 1) // 2, max(32, 6 * n))
    pairs = []
    for i in range(min(n - 1, n // 2)):
        pairs.append((i, i + 1))
    for i in range(0, n - 1, max(1, n // P)):
        if (i, i + 1) not in pairs:
            pairs.append((i, i + 1))
    while len(pairs) < P:
        i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
        if i != j and (i, j) not in pairs and (j, i) not in pairs:
            pairs.append((i, j))
        if len(pairs) > int(P * 2.5):
            break
    pairs = pairs[:P]
    rng.shuffle(pairs)
    for _ in range(max_sweeps):
        moved = False
        for (p, r) in pairs:
            res = line_search(h, q, score, p, r, dx, n)
            if res is not None:
                h, q, score = res
                moved = True
        if not moved:
            break
    return h, q, score

def build_starts(n, S, parent, rng):
    starts = []
    if parent is not None:
        starts.append(parent.copy())
        starts.append(parent[::-1].copy())
    idx = np.arange(n, dtype=np.float64)
    M = min(4, (n - 1) // 2)
    u = np.full(n, 0.5)
    for m in range(1, M + 1):
        A = 0.15 / np.sqrt(m) * float(rng.normal())
        phi = float(rng.uniform(0, 2 * np.pi))
        u += A * np.cos(2 * np.pi * m * (idx + 0.5) / n + phi)
    pu = proj_simplex(u, S)
    if pu is not None:
        starts.append(pu)
    perm = rng.permutation(n)
    bnd = np.zeros(n)
    bnd[perm[:n // 2]] = 1.0
    if n % 2 == 1 and n > 1:
        bnd[perm[n // 2]] = 0.5
    starts.append(bnd.copy())
    rn = proj_simplex(rng.random(n), S)
    if rn is not None:
        starts.append(rn)
    out = []
    for s in starts:
        dup = False
        for d in out:
            if np.max(np.abs(s - d)) < 1e-10:
                dup = True; break
        if not dup and is_feasible(s, n):
            out.append(s)
    cap = 6 if n <= 128 else 4
    return out[:cap]

def anneal(h, dx, n, rng, epochs=5, K=1000):
    S = n * 0.5
    q, score = compute_q(h, n)
    if q is None:
        return h, 0.5
    scale = 0.15
    for e in range(epochs):
        T = max(5e-5, 3e-3 * (0.35 ** e))
        accepted = 0
        for _ in range(K):
            p = int(rng.integers(0, n))
            r = int(rng.integers(0, n))
            if p == r:
                continue
            t = float(rng.uniform(-scale, scale))
            if t < -h[p] - 1e-12 or t > h[r] + 1e-12:
                continue
            if h[p] + t > 1.0 + 1e-12 or h[r] - t < -1e-12:
                continue
            h2 = h.copy()
            h2[p] += t
            h2[r] -= t
            if not is_feasible(h2, n):
                h2 = proj_simplex(h2, S)
                if h2 is None:
                    continue
            q2, s2 = compute_q(h2, n)
            if s2 is None:
                continue
            delta = s2 - score
            if delta <= 0:
                h, q, score = h2, q2, s2
                accepted += 1
            else:
                expt = -delta / T
                if expt > -700 and float(rng.random()) < np.exp(expt):
                    h, q, score = h2, q2, s2
                    accepted += 1
        ratio = accepted / max(1, K)
        if ratio < 0.10:
            scale = min(0.45, 1.3 * scale)
        elif ratio > 0.55:
            scale = max(0.01, 0.7 * scale)
        h, q, score = polish(h, q, score, dx, n, rng, max_sweeps=1)
    h, q, score = polish(h, q, score, dx, n, rng, max_sweeps=4)
    return h, score

def run(seed=42, **kwargs):
    if seed is None:
        seed = 42
    try:
        seed = int(seed) & 0xFFFFFFFF
    except (TypeError, ValueError):
        seed = 42
    rng = np.random.default_rng(seed)
    n = None
    parent = None
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        try:
            iv = np.asarray(iv, dtype=np.float64)
            if iv.ndim == 1 and 4 <= iv.size <= 999:
                n = iv.size
                S = n * 0.5
                if np.all(np.isfinite(iv)) and np.max(np.abs(iv)) <= 1e12:
                    if is_feasible(iv, n):
                        parent = iv.copy()
                    else:
                        p = proj_simplex(iv, S)
                        if p is not None and is_feasible(p, n):
                            parent = p
        except Exception:
            pass
    if n is None:
        n = int(kwargs.get('n_points', 80))
        if n < 8 or n > 999:
            n = 80
    S = n * 0.5
    dx = 2.0 / n
    const = np.full(n, 0.5)
    q_c, sc_c = compute_q(const, n)
    if sc_c is None:
        sc_c = 0.5
        q_c = np.zeros(2 * n - 1)
    best_h, best_score, best_q = const.copy(), sc_c, q_c.copy()
    if parent is not None:
        q_p, sc_p = compute_q(parent, n)
        if sc_p is not None and sc_p < best_score - 1e-12:
            best_score, best_h, best_q = sc_p, parent.copy(), q_p.copy()
    starts = build_starts(n, S, parent, rng)
    epochs = 5 if n <= 128 else (4 if n <= 256 else 3)
    K = min(1000, max(100, 60000 // n))
    for s in starts:
        q_s, sc_s = compute_q(s, n)
        if sc_s is not None and sc_s < best_score - 1e-12:
            best_score, best_h, best_q = sc_s, s.copy(), q_s.copy()
        h_a, sc_a = anneal(s.copy(), dx, n, rng, epochs=epochs, K=K)
        if is_feasible(h_a, n) and sc_a < best_score - 1e-12:
            q_a, _ = compute_q(h_a, n)
            best_score, best_h, best_q = sc_a, h_a.copy(), q_a.copy()
    q_b, sc_b = compute_q(best_h, n)
    if q_b is not None and sc_b is not None:
        best_h, best_q, best_score = polish(best_h, q_b, sc_b, dx, n, rng, max_sweeps=5)
    if not is_feasible(best_h, n):
        best_h = proj_simplex(best_h, S)
        if best_h is None:
            best_h = const.copy()
    q_f, sc_f = compute_q(best_h, n)
    if sc_f is None:
        sc_f = 0.5
    return best_h, sc_f, n
