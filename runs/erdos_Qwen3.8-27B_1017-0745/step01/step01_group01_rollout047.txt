```
def smooth_value(h,n,tau):
    h=np.asarray(h,dtype=np.float64).ravel()
    if h.shape != (n,) or not np.all(np.isfinite(h)):
       return float('inf')
    h=np.clip(h,0,1)
    try:
       c=np.correlate(h,1.0-h,mode='full')*(2.0/float(n))
       if not np.all(np.isfinite(c)): return float('inf')
    except Exception:
       return float('inf')
    gm=float(np.max(c))
    if tau <= 0.0:
       return gm
    ex=np.exp((c-gm)/tau)
    se=float(np.sum(ex))
    if se <= 0.0 or not np.isfinite(se):
       return gm
    return float(gm + tau*np.log(se))
```

```
def smooth_gradient(h,n,tau):
    h=np.asarray(h,dtype=np.float64).ravel()
    grad=np.zeros(n,dtype=np.float64)
    if h.shape != (n,) or not np.all(np.isfinite(h)):
       return grad
    h=np.clip(h,0,1)
    try:
       c=np.correlate(h,1.0-h,mode='full')*(2.0/float(n))
       if not np.all(np.isfinite(c)): return grad
    except Exception:
       return grad
    gm=float(np.max(c))
    if tau <= 0.0:
       tol = max(1e-7, 1e-6*abs(gm))
       active = c >= gm - tol
       if not np.any(active):
           active[int(np.argmax(c))] = True
       pf = active.astype(np.float64) / float(np.sum(active))
    else:
       ex=np.exp((c-gm)/tau)
       se=float(np.sum(ex))
       if se <= 0.0 or not np.isfinite(se):
           pf=np.zeros(len(c)); pf[int(np.argmax(c))]=1.0
       else:
           pf=ex/se
    p=pf[::-1]
    dx=2.0/float(n)
    p0=dx*p[n-1]
    grad += p0*(1.0-2.0*h)
    for d in range(1,n):
        pd=dx*p[n-1+d]
        pn=dx*p[n-1-d]
        m=n-d
        grad[:m] += pd*(1.0-h[d:])
        grad[d:] -= pd*h[:m]
        grad[d:] += pn*(1.0-h[:-d])
        grad[:-d] -= pn*h[d:]
    if not np.all(np.isfinite(grad)):
        return np.zeros(n,dtype=np.float64)
    return grad
```

```
import numpy as np
from scipy.optimize import minimize

_OPT_N = 1
_OPT_TAU = 0.01

def parse_seed(seed):
    try: x = int(seed)
    except Exception: x = 42
    return x & 0xffffffff

def get_parent():
    try:
        a = np.asarray(initial_h_values, dtype=np.float64).ravel()
    except Exception:
        return None, 0
    n = int(a.size)
    if n < 1 or n > 512:
        return None, 0
    if not np.all(np.isfinite(a)) or np.any(a < -1e-8) or np.any(a > 1.0 + 1e-8):
        return None, n
    return a.copy(), n

def is_feasible(h, n):
    if h is None: return False
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,): return False
    if not np.all(np.isfinite(h)): return False
    if float(np.min(h)) < -1e-10 or float(np.max(h)) > 1.0 + 1e-10: return False
    return abs(float(np.sum(h)) - 0.5*float(n)) <= 1e-8*max(1.0, float(n))

def project_capped(v, n):
    v = np.asarray(v, dtype=np.float64).ravel()
    if v.shape != (n,) or not np.all(np.isfinite(v)): return None
    v = np.clip(v, -1e12, 1e12)
    s = 0.5*float(n)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5*(lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > s: lo = mid
        else: hi = mid
    h = np.clip(v - 0.5*(lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(3):
        if abs(r) < 1e-13: break
        if r > 0:
            order = np.argsort(1.0 - h, kind='stable')[::-1]
        else:
            order = np.argsort(h, kind='stable')[::-1]
        for idx in order:
            if r > 0:
                add = min(r, 1.0 - float(h[idx]))
                if add > 0: h[idx] += add; r -= add
            else:
                sub = min(-r, float(h[idx]))
                if sub > 0: h[idx] -= sub; r += sub
            if abs(r) < 1e-14: break
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)): return None
    if abs(float(np.sum(h)) - s) > 1e-8*max(1.0, float(n)): return None
    return h

def enforce_mass(h, n):
    if h is None: return None
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,): return None
    h = np.clip(h, 0.0, 1.0)
    s = 0.5*float(n)
    r = s - float(np.sum(h))
    for _ in range(n + 2):

            if cap <= 0.0: break
            h[idx] += min(r, cap); r -= min(r, cap)
        else:
            idx = int(np.argmax(h)); cap = float(h[idx])
            if cap <= 0.0: break

    h = np.clip(h, 0.0, 1.0)
    return h if is_feasible(h, n) else None

def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,) or not np.all(np.isfinite(h)): return float('inf')
    if float(np.min(h)) < -1e-8 or float(np.max(h)) > 1.0 + 1e-8: return float('inf')
    if abs(float(np.sum(h)) - 0.5*float(n)) > 1e-7*max(1.0, float(n)): return float('inf')
    try:
        c = np.correlate(h, 1.0 - h, mode='full')
        if not np.all(np.isfinite(c)): return float('inf')
        return float(np.max(c) * (2.0 / float(n)))
    except Exception:
        return float('inf')

def smooth_value(h, n, tau):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,) or not np.all(np.isfinite(h)): return float('inf')
    h = np.clip(h, 0.0, 1.0)
    try:
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / float(n))
        if not np.all(np.isfinite(c)): return float('inf')
    except Exception:
        return float('inf')
    gm = float(np.max(c))
    if tau <= 0.0: return gm
    ex = np.exp((c - gm) / tau)
    se = float(np.sum(ex))
    if se <= 0.0 or not np.isfinite(se): return gm
    return float(gm + tau * np.log(se))

def smooth_gradient(h, n, tau):
    grad = np.zeros(n, dtype=np.float64)
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,) or not np.all(np.isfinite(h)): return grad
    h = np.clip(h, 0.0, 1.0)
    try:
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / float(n))
        if not np.all(np.isfinite(c)): return grad
    except Exception:
        return grad
    gm = float(np.max(c))
    if tau <= 0.0:
        tol = max(1e-7, 1e-6 * abs(gm))
        active = c >= gm - tol
        if not np.any(active): active[int(np.argmax(c))] = True
        pf = active.astype(np.float64) / float(np.sum(active))
    else:
        ex = np.exp((c - gm) / tau)
        se = float(np.sum(ex))
        if se <= 0.0 or not np.isfinite(se):
            pf = np.zeros(len(c), dtype=np.float64)
            pf[int(np.argmax(c))] = 1.0
        else:
            pf = ex / se
    p = pf[::-1]
    dx = 2.0 / float(n)
    p0 = dx * p[n - 1]
    grad += p0 * (1.0 - 2.0 * h)
    for d in range(1, n):
        pd = dx * p[n - 1 + d]
        pn = dx * p[n - 1 - d]
        m = n - d
        grad[:m] += pd * (1.0 - h[d:])
        grad[d:] -= pd * h[:m]
        grad[d:] += pn * (1.0 - h[:-d])
        grad[:-d] -= pn * h[d:]
    if not np.all(np.isfinite(grad)): return np.zeros(n, dtype=np.float64)
    return grad
```

```python
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(3):
        if abs(r) < 1e-13:
            break
        if r > 0:
            order = np.argsort(1.0 - h, kind='stable')[::-1]
        else:
            order = np.argsort(h, kind='stable')[::-1]
        for idx in order:
            if r > 0:
                add = min(r, 1.0 - float(h[idx]))
                if add > 0:
                    h[idx] += add
                    r -= add
            else:
                sub = min(-r, float(h[idx]))
                if sub > 0:
                    h[idx] -= sub
                    r += sub
            if abs(r) < 1e-14:
                break
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)):
        return None
    if abs(float(np.sum(h)) - s) > 1e-8 * max(1.0, float(n)):
        return None
    return h


def enforce_mass(h, n):
    if h is None:
        return None
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,):
        return None
    h = np.clip(h, 0.0, 1.0)
    s = 0.5 * float(n)
    r = s - float(np.sum(h))
    for _ in range(n + 2):
        if abs(r) < 1e-12:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            cap = 1.0 - float(h[idx])
            if cap <= 0.0:
                break
            h[idx] += min(r, cap)
            r -= min(r, cap)
        else:
            idx = int(np.argmax(h))
            cap = float(h[idx])
            if cap <= 0.0:
                break
            h[idx] -= min(-r, cap)
            r += min(-r, cap)
    h = np.clip(h, 0.0, 1.0)
    return h if is_feasible(h, n) else None


def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if float(np.min(h)) < -1e-8 or float(np.max(h)) > 1.0 + 1e-8:
        return float('inf')
    if abs(float(np.sum(h)) - 0.5 * float(n)) > 1e-7 * max(1.0, float(n)):
        return float('inf')
    try:
        c = np.correlate(h, 1.0 - h, mode='full')
        if not np.all(np.isfinite(c)):
            return float('inf')
        return float(np.max(c) * (2.0 / float(n)))
    except Exception:
        return float('inf')


def smooth_value(h, n, tau):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    h = np.clip(h, 0.0, 1.0)
    try:
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / float(n))
        if not np.all(np.isfinite(c)):
            return float('inf')
    except Exception:
        return float('inf')
    gm = float(np.max(c))
    if tau <= 0.0:
        return gm
    ex = np.exp((c - gm) / tau)
    se = float(np.sum(ex))
    if se <= 0.0 or not np.isfinite(se):
        return gm
    return float(gm + tau * np.log(se))


def smooth_gradient(h, n, tau):
    grad = np.zeros(n, dtype=np.float64)
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return grad
    h = np.clip(h, 0.0, 1.0)
    try:
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / float(n))
        if not np.all(np.isfinite(c)):
            return grad
    except Exception:
        return grad
    gm = float(np.max(c))
    if tau <= 0.0:
        tol = max(1e-7, 1e-6 * abs(gm))
        active = c >= gm - tol
        if not np.any(active):
            active[int(np.argmax(c))] = True
        pf = active.astype(np.float64) / float(np.sum(active))
    else:
        ex = np.exp((c - gm) / tau)
        se = float(np.sum(ex))
        if se <= 0.0 or not np.isfinite(se):
            pf = np.zeros(len(c), dtype=np.float64)
            pf[int(np.argmax(c))] = 1.0
        else:
            pf = ex / se
    p = pf[::-1]
    dx = 2.0 / float(n)
    p0 = dx * p[n - 1]
    grad += p0 * (1.0 - 2.0 * h)
    for d in range(1, n):
        pd = dx * p[n - 1 + d]
        pn = dx * p[n - 1 - d]
        m = n - d
        grad[:m] += pd * (1.0 - h[d:])
        grad[d:] -= pd * h[:m]
        grad[d:] += pn * (1.0 - h[:-d])
        grad[:-d] -= pn * h[d:]
    if not np.all(np.isfinite(grad)):
        return np.zeros(n, dtype=np.float64)
    return grad


def smooth_value_global(h):
    return smooth_value(h, _OPT_N, _OPT_TAU)


def smooth_gradient_global(h):
    return smooth_gradient(h, _OPT_N, _OPT_TAU)


def sum_constraint_global(h):
    return np.array([np.sum(h) - 0.5 * float(_OPT_N)])


def add_start(starts, h, n):
    p = project_capped(h, n)
    if p is None:
        return
    p = enforce_mass(p, n)
    if p is None:
        return
    for q in starts:
        if float(np.max(np.abs(p - q))) < 1e-8:
            return
    starts.append(p)


def build_starts(base, n, rng, max_starts, has_parent):
    starts = []
    idx = np.arange(n, dtype=np.float64)
    inv_n = 1.0 / float(n)
    if has_parent:
        add_start(starts, base, n)
        add_start(starts, np.full(n, 0.5), n)
        add_start(starts, base[::-1], n)
        for sd in (0.05, 0.12, 0.25):
            add_start(starts, base + rng.normal(0.0, sd, n), n)
        add_start(starts, 1.0 - base[::-1], n)
        add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * idx * inv_n), n)
        add_start(starts, 0.5 + 0.45 * np.sin(2.0 * np.pi * idx * inv_n), n)
        add_start(starts, rng.uniform(0.0, 1.0, n), n)
    else:
        u = np.full(n, 0.5)
        add_start(starts, u, n)
        blk = np.zeros(n, dtype=np.float64)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        add_start(starts, blk, n)
        alt = (np.arange(n) % 2 == 0).astype(np.float64)
        add_start(starts, alt, n)
        add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * idx * inv_n), n)
        add_start(starts, 0.5 + 0.45 * np.sin(2.0 * np.pi * idx * inv_n), n)
        add_start(starts, base + rng.normal(0.0, 0.2, n), n)
        add_start(starts, rng.uniform(0.0, 1.0, n), n)
        add_start(starts, rng.uniform(0.0, 1.0, n), n)
    if len(starts) > max_starts:
        starts = starts[:max_starts]
    return starts


def optimize_slsqp(x0, n, levels, maxiter):
    global _OPT_N, _OPT_TAU
    _OPT_N = int(n)
    x = np.asarray(x0, dtype=np.float64).ravel().copy()
    p = project_capped(x, n)
    x = np.full(n, 0.5) if p is None else p
    x = enforce_mass(x, n)
    if x is None:
        x = np.full(n, 0.5)
    best = x.copy()
    bs = exact_score(x, n)
    if not np.isfinite(bs):
        return best, bs
    if n <= 64:
        high, low = 0.01, 0.0002
    elif n <= 128:
        high, low = 0.01, 0.001
    else:
        high, low = 0.005, 0.004
    if levels < 2:
        levels = 2
    stages = []
    for l in range(levels):
        f = float(l) / float(levels - 1)
        stages.append(high * (low / high) ** f)
    if n <= 64 and maxiter > 0:
        stages.append(0.0)
    bounds = [(0.0, 1.0)] * int(n)
    cons = ({'type': 'eq', 'fun': sum_constraint_global},)
    opts = {'maxiter': int(maxiter), 'ftol': 1e-10}
    for tau in stages:
        _OPT_TAU = float(tau)
        try:
            res = minimize(smooth_value_global, x, jac=smooth_gradient_global,
                           method='SLSQP', bounds=bounds, constraints=cons, options=opts)
            if res.x is not None and np.all(np.isfinite(res.x)):
                y = project_capped(res.x, n)
                if y is not None:
                    y = enforce_mass(y, n)
                if y is None:
                    y = best
                if is_feasible(y, n):
                    sy = exact_score(y, n)
                    if np.isfinite(sy):
                        if sy < bs - 1e-12:
                            best, bs = y.copy(), sy
                        if sy <= bs + 0.03:
                            x = y
                        else:
                            x = best
        except Exception:
            pass
    return best, bs


def projected_gradient(x0, n, tau, iters):
    x = np.asarray(x0, dtype=np.float64).ravel().copy()
    p = project_capped(x, n)
    x = np.full(n, 0.5) if p is None else p
    best = x.copy()
    bs = exact_score(x, n)
    if tau <= 0.0:
        tau = 1e-5
    alpha = 0.2 * float(n)
    for _ in range(int(max(1, iters))):
        grad = smooth_gradient(x, n, tau)
        F = smooth_value(x, n, tau)
        if not np.isfinite(F) or not np.all(np.isfinite(grad)):
            break
        accept = False
        for __ in range(25):
            y = project_capped(x - alpha * grad, n)
            if y is None:
                alpha *= 0.5
                continue
            d = y - x
            D = float(np.dot(grad, d))
            if D > -1e-14 or float(np.max(np.abs(d))) < 1e-12:
                alpha *= 0.5
                continue
            Fy = smooth_value(y, n, tau)
            if np.isfinite(Fy) and Fy <= F + 1e-4 * D + 1e-12:
                sy = exact_score(y, n)
                if np.isfinite(sy) and sy < bs - 1e-12:
                    best, bs = y.copy(), sy
                x, F = y, Fy
                accept = True
                break
            alpha *= 0.5
        if not accept:
            break
        alpha *= 1.15
        if alpha < 1e-4:
            alpha = 1e-4
        if alpha > 25.0 * float(n):
            alpha = 25.0 * float(n)
    best = enforce_mass(best, n)
    if best is not None:
        bs = exact_score(best, n)
    return best, bs


def run(seed=42, **kwargs):
    parent, pn = get_parent()
    if 1 <= pn <= 1000:
        n = pn
    else:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
        if n < 1 or n > 1000:
            n = 47
    uniform = np.full(n, 0.5)
    fallback = uniform.copy()
    fb = exact_score(uniform, n)
    parent_proj = None
    if parent is not None:
        pp = enforce_mass(project_capped(parent, n), n)
        parent_proj = pp
        if pp is not None:
            ps = exact_score(pp, n)
            if np.isfinite(ps) and ps < fb:
                fallback, fb = pp, ps
    rng = np.random.default_rng(parse_seed(seed))
    if n <= 64:
        max_starts, levels, maxiter, pg1, pg2 = 8, 4, 70, 180, 90
    elif n <= 128:
        max_starts, levels, maxiter, pg1, pg2 = 6, 3, 45, 90, 45
    elif n <= 256:
        max_starts, levels, maxiter, pg1, pg2 = 4, 3, 25, 50, 20
    else:
        max_starts, levels, maxiter, pg1, pg2 = 3, 2, 10, 20, 10
    base = parent_proj if parent_proj is not None else fallback
    starts = build_starts(base, n, rng, max_starts, parent_proj is not None)
    best = fallback.copy()
    best_score = fb
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_score - 1e-12:
            best, best_score = st.copy(), ss
    if n <= 128 and maxiter > 0:
        opt_count = min(max_starts, 6 if n <= 64 else 4)
        for st in starts[:opt_count]:
            ob, oscore = optimize_slsqp(st, n, levels, maxiter)
            ob = enforce_mass(ob, n)
            if ob is not None:
                oscore = exact_score(ob, n)
            if np.isfinite(oscore) and oscore < best_score - 1e-12:
                best, best_score = ob.copy(), oscore
    cands = []
    for cand in (best, fallback, uniform,
                 starts[0] if len(starts) > 0 else None,
                 starts[1] if len(starts) > 1 else None):
        if cand is None:
            continue
        cc = enforce_mass(cand, n)
        if cc is None:
            continue
        dup = False
        for q in cands:
            if float(np.max(np.abs(cc - q))) < 1e-10:
                dup = True
                break
        if not dup:
            cands.append(cc)
    for cand in cands[:3]:
        if n <= 256 and pg1 > 0:
            pb, _ = projected_gradient(cand, n, 0.0005, pg1)
            if pb is not None:
                ps = exact_score(pb, n)
                if np.isfinite(ps) and ps < best_score - 1e-12:
                    best, best_score = pb.copy(), ps
                if n <= 128 and pg2 > 0:
                    pb2, _ = projected_gradient(pb, n, 0.00005, pg2)
                    if pb2 is not None:
                        ps2 = exact_score(pb2, n)
                        if np.isfinite(ps2) and ps2 < best_score - 1e-12:
                            best, best_score = pb2.copy(), ps2
    final = best
    for cand in (best, fallback, uniform):
        cc = enforce_mass(cand, n)
        if cc is None:
            continue
        cs = exact_score(cc, n)
        if np.isfinite(cs) and cs < best_score - 1e-12:
            best_score, final = cs, cc
    final = enforce_mass(final, n)
    if final is None:
        final, fs = uniform.copy(), exact_score(uniform, n)
    else:
        fs = exact_score(final, n)
        if not np.isfinite(fs):
            final, fs = uniform.copy(), exact_score(uniform, n)
    return final.copy(), float(fs), int(n)
```
