def solve_mm_subproblem(x, n):
    dx = 2.0 / n
    L = 2.0 * dx
    _MM['x'] = x
    _MM['n'] = n
    _MM['L'] = L
    
    # Get all lag values and gradients
    lags = list(range(-(n-1), n))
    all_q = np.zeros(2*n-1)
    all_G = np.zeros((2*n-1, n))
    for idx, s in enumerate(lags):
        q, g = lag_value_grad(x, s, n)
        all_q[idx] = q
        all_G[idx] = g
    
    if n <= 64:
        # Full problem
        _MM['qJ'] = all_q
        _MM['GJ'] = all_G
        z0 = np.append(x.copy(), float(np.max(all_q)) + 1e-6)
        bounds = [(0.0, 1.0)] * n + [(None, None)]
        cons = [{'type': 'ineq', 'fun': mm_ineq, 'jac': mm_ineq_grad},
                {'type': 'eq', 'fun': mm_eq, 'jac': mm_eq_grad}]
        try:
            res = minimize(mm_obj, z0, jac=mm_obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-12, 'disp': False})
            y_raw = np.array(res.x[:n], dtype=np.float64)
            y = project_box_sum(y_raw, 0.5 * n)
            if y is not None and np.all(np.isfinite(y)):
                return y
        except Exception:
            pass
        return x
    else:
        # Cut generation
        # Start with top 8 lags
        order = np.argsort(-all_q)
        active = sorted(order[:min(8, 2*n-1)].tolist())
        max_iter = 8
        y = x.copy()
        for it in range(max_iter):
            qJ = all_q[active]

            _MM['GJ'] = GJ
            d = y - x
            t0 = float(np.max(qJ + GJ @ d))
            z0 = np.append(y.copy(), t0)
            bounds = [(0.0, 1.0)] * n + [(None, None)]

import numpy as np
from scipy.optimize import minimize

_MM = {}

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    if S < -1e-12 or S > n + 1e-12:
        return None
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(v - tau, 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-14:
        idx = np.argsort(1.0 - y)
        for i in idx:
            d = min(r, 1.0 - y[i])
            y[i] += d
            r -= d
            if r <= 0:
                break
    elif r < -1e-14:
        idx = np.argsort(y)
        for i in idx:
            d = min(-r, y[i])
            y[i] -= d
            r += d
            if r >= 0:
                break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    if abs(float(np.sum(y)) - S) > 1e-10 * max(1, n):
        return None
    return y

def check_feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(float(np.sum(h)) - 0.5 * n) < 1e-10 * max(1, n))

def exact_score(h):
    n = len(h)
    if not check_feasible(h, n):
        return float('inf')
    try:
        c = np.correlate(h, 1.0 - h, mode='full')
        if len(c) != 2 * n - 1 or not np.all(np.isfinite(c)):
            return float('inf')
        return float(np.max(c) * (2.0 / n))
    except Exception:
        return float('inf')

def lag_value_grad(h, s, n):
    dx = 2.0 / n
    lo = max(0, -s)
    hi = min(n, n - s)
    q = 0.0
    g = np.zeros(n)
    for i in range(lo, hi):
        j = i + s
        q += h[i] * (1.0 - h[j])
        g[i] += 1.0 - h[j]
        g[j] -= h[i]
    return q * dx, g * dx

def separation(x, d, n):
    dx = 2.0 / n
    best_val = -1e100
    best_lag = 0
    for s in range(-(n - 1), n):
        lo = max(0, -s)
        hi = min(n, n - s)
        ell = 0.0
        for i in range(lo, hi):
            j = i + s
            ell += x[i] * (1.0 - x[j]) + (1.0 - x[j]) * d[i] - x[i] * d[j]
        ell *= dx
        if ell > best_val:
            best_val = ell
            best_lag = s
    return best_val, best_lag

def mm_obj(z):
    x = _MM['x']
    L = _MM['L']
    y = z[:_MM['n']]
    return 0.5 * L * np.dot(y - x, y - x) + z[-1]

def mm_obj_grad(z):
    n = _MM['n']
    g = np.zeros(n + 1)
    g[:n] = _MM['L'] * (z[:n] - _MM['x'])
    g[n] = 1.0
    return g

def mm_ineq(z):
    n = _MM['n']
    d = z[:n] - _MM['x']
    return z[n] - _MM['qJ'] - _MM['GJ'] @ d

def mm_ineq_grad(z):
    n = _MM['n']
    J = np.zeros((len(_MM['qJ']), n + 1))
    J[:, :n] = -_MM['GJ']
    J[:, n] = 1.0
    return J

def mm_eq(z):
    return float(np.sum(z[:_MM['n']])) - 0.5 * _MM['n']

def mm_eq_grad(z):
    n = _MM['n']
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def solve_mm_subproblem(x, n):
    dx = 2.0 / n
    L = 2.0 * dx
    _MM['x'] = x
    _MM['n'] = n
    _MM['L'] = L
    lags = list(range(-(n - 1), n))
    all_q = np.zeros(2 * n - 1)
    all_G = np.zeros((2 * n - 1, n))
    for idx, s in enumerate(lags):
        q, g = lag_value_grad(x, s, n)
        all_q[idx] = q
        all_G[idx] = g
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    cons = [{'type': 'ineq', 'fun': mm_ineq, 'jac': mm_ineq_grad},
            {'type': 'eq', 'fun': mm_eq, 'jac': mm_eq_grad}]
    if n <= 64:
        _MM['qJ'] = all_q
        _MM['GJ'] = all_G
        z0 = np.append(x.copy(), float(np.max(all_q)) + 1e-6)
        try:
            res = minimize(mm_obj, z0, jac=mm_obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-12, 'disp': False})
            y_raw = np.array(res.x[:n], dtype=np.float64)
            y = project_box_sum(y_raw, 0.5 * n)
            if y is not None:
                return y
        except Exception:
            pass
        return x.copy()
    else:
        order = np.argsort(-all_q)
        active = sorted(order[:min(8, 2 * n - 1)].tolist())
        y = x.copy()
        for _ in range(8):
            qJ = all_q[active]
            GJ = all_G[active]
            _MM['qJ'] = qJ
            _MM['GJ'] = GJ
            d = y - x
            t0 = float(np.max(qJ + GJ @ d))
            z0 = np.append(y.copy(), t0)
            try:
                res = minimize(mm_obj, z0, jac=mm_obj_grad, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 150, 'ftol': 1e-11, 'disp': False})
                y_raw = np.array(res.x[:n], dtype=np.float64)
                y_new = project_box_sum(y_raw, 0.5 * n)
                if y_new is None:
                    break
                d = y_new - x
                full_val, full_lag = separation(x, d, n)
                EJ = float(np.max(qJ + GJ @ d))
                y = y_new
                if full_val <= EJ + 1e-9:
                    break
                lag_idx = full_lag + (n - 1)
                if lag_idx not in active and len(active) < min(2 * n - 1, 96):
                    active.append(lag_idx)
            except Exception:
                break
        return y

def local_mm_descent(x, n):
    x = x.copy()
    cur_score = exact_score(x)
    M = max(8, min(40, 1800 // n))
    best_z = x
    best_s = cur_score
    for _ in range(M):
        y = solve_mm_subproblem(x, n)
        if y is None or not np.all(np.isfinite(y)):
            break
        if np.max(np.abs(y - x)) <= 1e-10:
            break
        max_k = 8 if n <= 128 else 5
        found = False
        for k in range(max_k + 1):
            alpha = 2.0 ** (-k)
            z = project_box_sum(x + alpha * (y - x), 0.5 * n)
            if z is None:
                continue
            s = exact_score(z)
            if s < cur_score - 1e-13:
                cur_score = s
                x = z
                found = True
                break
        if not found:
            break
        if cur_score <= best_s - 1e-13:
            best_s = cur_score
            best_z = x.copy()
    return best_z, best_s

def build_starts(parent, n, rng):
    starts = []
    t = np.arange(n)
    cands = [parent.copy(), 1.0 - parent, parent[::-1].copy(),
             parent + 0.06 * rng.standard_normal(n),
             parent + 0.20 * rng.standard_normal(n),
             rng.random(n),
             0.5 + 0.30 * np.cos(2.0 * np.pi * t / n + 0.7) + 0.12 * np.sin(4.0 * np.pi * t / n + 1.3),
             parent + 0.12 * rng.standard_normal(n),
             0.5 + 0.25 * np.sin(4.0 * np.pi * t / n + 0.3) + 0.1 * np.cos(6.0 * np.pi * t / n + 2.0)]
    for c in cands:
        p = project_box_sum(c, 0.5 * n)
        if p is None:
            continue
        dup = False
        for ex in starts:
            if np.max(np.abs(p - ex)) < 1e-10:
                dup = True
                break
        if not dup:
            starts.append(p)
    return starts
