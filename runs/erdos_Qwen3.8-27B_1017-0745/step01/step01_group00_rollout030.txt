def lag_jacobian(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        i = d + n - 1
        p0, q0 = max(d, 0), max(-d, 0)
        m = n - abs(d)
        G[i, p0:p0 + m] += dx * (1.0 - h[q0:q0 + m])
        G[i, q0:q0 + m] -= dx * h[p0:p0 + m]
    return G
