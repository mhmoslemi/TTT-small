import numpy as np
import math

def project_capped(y, n):
    S = n / 2.0
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return np.full(n, 0.5), False
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    lam = float(np.clip(np.mean(y) - 0.5, lo, hi))
    for _ in range(60):
        c = np.clip(y - lam, 0.0, 1.0)
        r = c.sum() - S
        if abs(r) <= 1e-13:
            break
        if r > 0:
            lo = lam
        else:
            hi = lam
        a = np.sum((y > lam) & (y < lam + 1.0))
        if a > 0:
            nl = lam + r / a
        else:
            nl = 0.5 * (lo + hi)
        if not np.isfinite(nl):
            nl = 0.5 * (lo + hi)
        lam = nl if lo < nl < hi else 0.5 * (lo + hi)
    c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(3):
        diff = S - c.sum()
        if abs(diff) < 1e-14:
            break
        mask = (c < 1.0 - 1e-12) if diff > 0 else (c > 1e-12)
        cnt = mask.sum()
        if cnt == 0:
            break
        c[mask] += diff / cnt
        np.clip(c, 0.0, 1.0, out=c)
    ok = (c.shape == (n,) and np.all(np.isfinite(c)) and np.all(c >= -1e-12)
          and np.all(c <= 1.0 + 1e-12) and abs(c.sum() - S) <= 1e-10)
    return c, ok

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(h.sum() - n / 2.0) <= 1e-10)

def lag_scores(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def exact_score(h, n):
    if not np.all(np.isfinite(h)):
        return float('inf')
    qs = lag_scores(h, n)
    if qs.shape[0] != 2 * n - 1 or not np.all(np.isfinite(qs)):
        return float('inf')
    return float(qs.max())

def build_G(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        lo = max(0, k)
        hi = min(n, n + k)
        row = k + n - 1
        G[row, lo:hi] += dx * (1.0 - h[lo - k:hi - k])
        G[row, lo - k:hi - k] -= dx * h[lo:hi]
    return G

def project_simplex(z):
    m = len(z)
    sz = np.sort(z)[::-1]
    cs = np.cumsum(sz)
    idx = np.arange(1, m + 1, dtype=np.float64)
    rho = np.where(sz - (cs - 1.0) / idx > 0)[0]
    if len(rho) == 0:
        return np.full(m, 1.0 / m)
    r = int(rho[0])
    theta = (cs[r] - 1.0) / (r + 1)
    p = np.maximum(z - theta, 0.0)
    diff = 1.0 - p.sum()
    if abs(diff) > 1e-12:
        p[p.argmax()] += diff
        np.clip(p, 0.0, 1.0, out=p)
    return p

def eval_dual(h, q, G, n, p, L):
    gmix = G.T @ p
    y, ok = project_capped(h - gmix / L, n)
    if not ok:
        return None
    d = y - h
    u = q + G @ d
    if not np.all(np.isfinite(u)):
        return None
    D = float(p @ u + 0.5 * L * d @ d)
    gap = float(u.max() - p @ u)
    return d, u, D, gap

def solve_majorant(h, q, G, n, max_inner):
    dx = 2.0 / n
    L = 2.0 * dx
    Ldual = max(1e-12, 1.2 * float(np.sum(G * G)) / L)
    m = max(1e-12, 1e-8 * max(1.0, abs(q.max())))
    active = q >= q.max() - m
    if not active.any():
        active[q.argmax()] = True
    p = np.zeros(2 * n - 1)
    p[active] = 1.0 / active.sum()
    x = p.copy()
    yv = p.copy()
    th = 1.0
    best_st = eval_dual(h, q, G, n, x, L)
    if best_st is None:
        return None
    best_D = best_st[2]
    prev_x = x.copy()
    stall = 0
    for _ in range(max_inner):
        st_y = eval_dual(h, q, G, n, yv, L)
        if st_y is None:
            break
        grad_y = st_y[1]
        x_new = project_simplex(yv + grad_y / Ldual)
        st_x = eval_dual(h, q, G, n, x_new, L)
        if st_x is None:
            break
        if st_x[2] < best_D - 1e-13:
            x_r = project_simplex(x + best_st[1] / Ldual)
            st_r = eval_dual(h, q, G, n, x_r, L)
            if st_r is not None and st_r[2] >= best_D - 1e-13:
                x, best_st, best_D = x_r, st_r, st_r[2]
                yv = x.copy()
                th = 1.0
            else:
                break
        else:
            if st_x[2] > best_D:
                best_D = st_x[2]
                best_st = st_x
            th_new = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * th * th))
            yv = x_new + ((th - 1.0) / th_new) * (x_new - x)
            yv = project_simplex(yv)
            x = x_new
            th = th_new
        if np.max(np.abs(x - prev_x)) < 1e-11:
            stall += 1
        else:
            stall = 0
        prev_x = x.copy()
        st_c = eval_dual(h, q, G, n, x, L)
        if st_c is None:
            break
        if st_c[3] <= max(1e-10, 1e-8 * max(1.0, abs(st_c[1].max()))):
            break
        if stall >= 5:
            break
    return best_st[0] if best_st is not None else None

def optimize_start(h0, n, max_outer, max_inner):
    h, ok = project_capped(h0.copy(), n)
    if not ok:
        return h0.copy()
    s = exact_score(h, n)
    best_h = h.copy()
    best_s = s
    for _ in range(max_outer):
        q = lag_scores(h, n)
        G = build_G(h, n)
        if not np.all(np.isfinite(G)):
            break
        d = solve_majorant(h, q, G, n, max_inner)
        if d is None or not np.all(np.isfinite(d)) or np.max(np.abs(d)) <= 1e-10:
            break
        v = G @ d
        norm2 = float(d @ d)
        L = 4.0 / n
        qmax = float(q.max())
        accepted = False
        for i in range(24):
            alpha = 1.0 / (2.0 ** i)
            Mval = float((q + alpha * v).max()) + 0.5 * L * alpha * alpha * norm2
            if Mval > qmax - 1e-14:
                continue
            ht, ok2 = project_capped(h + alpha * d, n)
            if not ok2:
                continue
            st = exact_score(ht, n)
            if st < s - 1e-12:
                h = ht
                s = st
                if s < best_s:
                    best_s = s
                    best_h = h.copy()
                accepted = True
                break
        if not accepted:
            break
    return best_h

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = globals().get('initial_h_values', None)
    n = 45
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.ndim == 1 and 2 <= len(p) <= 999 and np.all(np.isfinite(p)) and np.max(np.abs(p)) < 1e6:
            n = len(p)
            fallback, fok = project_capped(p, n)
            if not fok:
                fallback = np.full(n, 0.5)
    else:
        nk = kwargs.get('n_points', None)
        if isinstance(nk, int) and not isinstance(nk, bool) and 2 <= nk <= 999:
            n = nk
        fallback = np.full(n, 0.5)
    fallback = project_capped(fallback, n)[0]
    fb_s = exact_score(fallback, n)

    if n <= 100:
        mo, mi, ns = 80, 60, 7
    elif n <= 400:
        mo, mi, ns = 10, 16, 4
    else:
        mo, mi, ns = 4, 10, 2

    starts = [
        fallback.copy(),
        1.0 - fallback,
        fallback[::-1].copy(),
        1.0 - fallback[::-1].copy(),
        project_capped(fallback + 0.08 * rng.standard_normal(n), n)[0],
        project_capped(fallback + 0.20 * rng.standard_normal(n), n)[0],
        project_capped(rng.uniform(0, 1, n), n)[0],
    ][:ns]
    seen = []
    ustarts = []
    for st in starts:
        dup = False
        for s2 in seen:
            if np.max(np.abs(st - s2)) < 1e-8:
                dup = True
                break
        if not dup:
            seen.append(st.copy())
            ustarts.append(st)

    best_h = fallback.copy()
    best_s = fb_s
    for st in ustarts:
        try:
            bh = optimize_start(st, n, mo, mi)
            bh, ok = project_capped(bh, n)
            if ok and check_feasible(bh, n):
                bs = exact_score(bh, n)
                if bs < best_s - 1e-12:
                    best_s = bs
                    best_h = bh.copy()
        except Exception:
            pass

    if n <= 100:
        try:
            pert = project_capped(best_h + 0.025 * rng.standard_normal(n), n)[0]
            bh = optimize_start(pert, n, mo // 2, mi)
            bh, ok = project_capped(bh, n)
            if ok and check_feasible(bh, n):
                bs = exact_score(bh, n)
                if bs < best_s - 1e-12:
                    best_s = bs
                    best_h = bh.copy()
        except Exception:
            pass

    final_h, fok = project_capped(best_h, n)
    if not fok or not check_feasible(final_h, n):
        final_h = fallback.copy()
    final_s = exact_score(final_h, n)
    if final_s > fb_s + 1e-12:
        final_h = fallback.copy()
        final_s = fb_s
    return final_h.astype(np.float64), float(final_s), int(n)
