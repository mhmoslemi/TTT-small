import numpy as np
import math

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    if np.max(np.abs(z)) > 1e100:
        return np.full(N, m / N)
    z = z - np.median(z)  # shift doesn't change projection
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    if r > 1e-14:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
            if r <= 1e-14:
                break
    elif r < -1e-14:
        for i in range(N):
            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1 + 1e-10:
        return np.full(N, m / N)
    if abs(h.sum() - m) > 1e-8 * max(1, N):
        return np.full(N, m / N)
    return np.clip(h, 0.0, 1.0)

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr) * 2.0 / N)

def build_helmert(N):
    d = N - 1
    B = np.zeros((N, d))
    for k in range(d):
        a = math.sqrt((k + 1) * (k + 2))
        B[:k + 1, k] = 1.0 / a
        B[k + 1, k] = -(k + 1) / a
    return B

def cma_solve(h0, N, m, sigma_init, max_gen, stag_limit, rng):
    d = N - 1
    if d < 1:
        return h0
    B = build_helmert(N)
    u = np.full(N, 0.5)
    mu = B.T @ (h0 - u)
    C = np.eye(d)
    p_sigma = np.zeros(d)
    p_c = np.zeros(d)
    sigma = sigma_init
    
    lambda_pop = max(4, 4 + int(3 * math.log(d)))
    mu_sel = max(2, lambda_pop // 2)
    
    # Weights

    w /= w.sum()
    mu_eff = 1.0 / np.sum(w ** 2)
    
    c_sigma = (mu_eff + 2.0) / (d + mu_eff + 5.0)
    d_sigma = 1.0 + 2.0 * max(0.0, math.sqrt((mu_eff - 1.0) / (d + 1.0)) - 1.0) + c_sigma
    c_c = (4.0 + mu_eff / d) / (d + 4.0 + 2.0 * mu_eff / d)
    c1 = 2.0 / ((d + 1.3) ** 2 + mu_eff)
    c_mu = min(1.0 - c1, 2.0 * (mu_eff - 2.0 + 1.0 / mu_eff) / ((d + 2.0) ** 2 + mu_eff))
    chi_d = math.sqrt(d) * (1.0 - 1.0 / (4.0 * d) + 1.0 / (21.0 * d * d))
    
    best_h = h0.copy()
    best_score = score_exact(h0, N)
    stag = 0
    local_gen = 0
    
    for gen in range(max_gen):
        local_gen += 1
        # Symmetrize C
        C = 0.5 * (C + C.T)
        
        # Eigen decomposition
        try:
            eigvals, eigvecs = np.linalg.eigh(C)
        except np.linalg.LinAlgError:
            C = np.eye(d)
            p_sigma = np.zeros(d)
            p_c = np.zeros(d)
            sigma = max(1e-4, sigma * 0.5)
            continue
        
        max_ev = max(np.max(eigvals), 1e-12)
        eigvals = np.clip(eigvals, max(1e-12, max_ev * 1e-10), 1e4)
        C = eigvecs @ np.diag(eigvals) @ eigvecs.T
        C = 0.5 * (C + C.T)
        
        sqrt_ev = np.sqrt(eigvals)
        inv_sqrt_ev = 1.0 / sqrt_ev
        C_half = eigvecs @ np.diag(inv_sqrt_ev) @ eigvecs.T
        C_half = 0.5 * (C_half + C_half.T)
        
        # Sample population
        xs = np.zeros((lambda_pop, d))
        hs = np.zeros((lambda_pop, N))
        scores = np.zeros(lambda_pop)
        
        for j in range(lambda_pop):
            z = rng.standard_normal(d)
            x_raw = mu + sigma * (eigvecs @ (sqrt_ev * z))
            
            # Decode
            if not np.all(np.isfinite(x_raw)):
                x_raw = mu
            z_dec = u + B @ x_raw
            h_dec = project_box_sum(z_dec, m)
            h_dec = np.clip(h_dec, 0.0, 1.0)
            hs[j] = h_dec
            scores[j] = score_exact(h_dec, N)
            xs[j] = B.T @ (h_dec - u)
        
        # Update global best
        gen_best_score = best_score
        for j in range(lambda_pop):
            if scores[j] < best_score - 1e-12:
                best_score = scores[j]
                best_h = hs[j].copy()
        
        # Rank and select
        order = np.argsort(scores, kind='stable')
        sel = order[:mu_sel]
        
        # y_i for selected
        y_sel = np.zeros((mu_sel, d))
        for i in range(mu_sel):
            y_sel[i] = (xs[sel[i]] - mu) / sigma
        
        # Weighted mean
        mu_new = mu.copy()
        for i in range(mu_sel):
            mu_new += w[i] * xs[sel[i]]
        
        # Project mean
        h_mean = u + B @ mu_new
        h_mean = project_box_sum(h_mean, m)
        h_mean = np.clip(h_mean, 0.0, 1.0)
        mu_new = B.T @ (h_mean - u)
        y_w = (mu_new - mu) / sigma
        
        s_mean = score_exact(h_mean, N)
        if s_mean < best_score - 1e-12:
            best_score = s_mean
            best_h = h_mean.copy()
        
        # Path updates
        p_sigma = (1 - c_sigma) * p_sigma + math.sqrt(c_sigma * (2 - c_sigma) * mu_eff) * (C_half @ y_w)
        t = local_gen
        h_sigma = 0.0
        if np.linalg.norm(p_sigma) / math.sqrt(1.0 - (1.0 - c_sigma) ** (2 * t)) < (1.4 + 2.0 / (d + 1)) * chi_d:
            h_sigma = 1.0
        p_c = (1 - c_c) * p_c + h_sigma * math.sqrt(c_c * (2 - c_c) * mu_eff) * y_w
        
        # Covariance update
        C_new = (1 - c1 - c_mu) * C
        C_new += c1 * (np.outer(p_c, p_c) + (1 - h_sigma) * c_c * (2 - c_c) * C)
        for i in range(mu_sel):
            C_new += c_mu * w[i] * np.outer(y_sel[i], y_sel[i])
        C_new = 0.5 * (C_new + C_new.T)
        
        if not np.all(np.isfinite(C_new)):
            C = np.eye(d)
            p_sigma = np.zeros(d)
            p_c = np.zeros(d)
            sigma = max(1e-4, sigma * 0.5)
            continue
        
        C = C_new
        
        # Step size update
        expo = c_sigma / d_sigma * (np.linalg.norm(p_sigma) / chi_d - 1.0)
        expo = np.clip(expo, -1.0, 1.0)
        sigma = sigma * math.exp(float(expo))
        sigma = np.clip(sigma, 1e-4, 0.5)
        
        mu = mu_new
        
        # Stagnation check
        if best_score < gen_best_score - 1e-12:
            stag = 0
        else:
            stag += 1
            if stag >= stag_limit or sigma <= 1e-4:
                break
    
    return best_h

import numpy as np
import math

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, m / N)
    if np.max(np.abs(z)) > 1e100:
        return np.full(N, m / N)
    z = z - np.median(z)
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    if r > 1e-14:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
            if r <= 1e-14:
                break
    elif r < -1e-14:
        for i in range(N):
            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1 + 1e-10:
        return np.full(N, m / N)
    if abs(h.sum() - m) > 1e-8 * max(1, N):
        return np.full(N, m / N)
    return np.clip(h, 0.0, 1.0)

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr) * 2.0 / N)

def build_helmert(N):
    d = N - 1
    B = np.zeros((N, d))
    for k in range(d):
        a = math.sqrt((k + 1) * (k + 2))
        B[:k + 1, k] = 1.0 / a
        B[k + 1, k] = -(k + 1) / a
    return B

def cma_solve(h0, N, m, sigma_init, max_gen, stag_limit, rng):
    d = N - 1
    if d < 1:
        return h0
    B = build_helmert(N)
    u = np.full(N, 0.5)
    mu = B.T @ (h0 - u)
    C = np.eye(d)
    p_sigma = np.zeros(d)
    p_c = np.zeros(d)
    sigma = sigma_init
    lambda_pop = max(4, 4 + int(3.0 * math.log(d)))
    mu_sel = max(2, lambda_pop // 2)
    w = np.array([math.log(mu_sel + 0.5) - math.log(float(i + 1)) for i in range(mu_sel)])
    w /= w.sum()
    mu_eff = 1.0 / np.sum(w * w)
    c_sigma = (mu_eff + 2.0) / (d + mu_eff + 5.0)
    d_sigma = 1.0 + 2.0 * max(0.0, math.sqrt((mu_eff - 1.0) / (d + 1.0)) - 1.0) + c_sigma
    c_c = (4.0 + mu_eff / d) / (d + 4.0 + 2.0 * mu_eff / d)
    c1 = 2.0 / ((d + 1.3) ** 2 + mu_eff)
    c_mu = min(1.0 - c1, 2.0 * (mu_eff - 2.0 + 1.0 / mu_eff) / ((d + 2.0) ** 2 + mu_eff))
    chi_d = math.sqrt(d) * (1.0 - 1.0 / (4.0 * d) + 1.0 / (21.0 * d * d))
    best_h = h0.copy()
    best_score = score_exact(h0, N)
    stag = 0
    for gen in range(max_gen):
        C = 0.5 * (C + C.T)
        try:
            eigvals, eigvecs = np.linalg.eigh(C)
        except np.linalg.LinAlgError:
            C = np.eye(d)
            p_sigma = np.zeros(d)
            p_c = np.zeros(d)
            sigma = max(1e-4, sigma * 0.5)
            continue
        max_ev = max(float(np.max(eigvals)), 1e-12)
        eigvals = np.clip(eigvals, max(1e-12, max_ev * 1e-10), 1e4)
        C = eigvecs @ (eigvals[:, None] * eigvecs.T)
        C = 0.5 * (C + C.T)
        sqrt_ev = np.sqrt(eigvals)
        inv_sqrt_ev = 1.0 / sqrt_ev
        C_half = eigvecs @ (inv_sqrt_ev[:, None] * eigvecs.T)
        C_half = 0.5 * (C_half + C_half.T)
        xs = np.empty((lambda_pop, d))
        hs = np.empty((lambda_pop, N))
        scores = np.empty(lambda_pop)
        for j in range(lambda_pop):
            z = rng.standard_normal(d)
            x_raw = mu + sigma * (eigvecs @ (sqrt_ev * z))
            if not np.all(np.isfinite(x_raw)):
                x_raw = mu
            z_dec = u + B @ x_raw
            h_dec = project_box_sum(z_dec, m)
            hs[j] = h_dec
            scores[j] = score_exact(h_dec, N)
            xs[j] = B.T @ (h_dec - u)
        gen_best = best_score
        for j in range(lambda_pop):
            if scores[j] < best_score - 1e-12:
                best_score = scores[j]
                best_h = hs[j].copy()
        order = np.argsort(scores, kind='stable')
        sel = order[:mu_sel]
        y_sel = np.empty((mu_sel, d))
        for i in range(mu_sel):
            y_sel[i] = (xs[sel[i]] - mu) / sigma
        mu_new = mu + np.sum(w[:, None] * xs[sel], axis=0)
        h_mean = project_box_sum(u + B @ mu_new, m)
        mu_new = B.T @ (h_mean - u)
        y_w = (mu_new - mu) / sigma
        s_mean = score_exact(h_mean, N)
        if s_mean < best_score - 1e-12:
            best_score = s_mean
            best_h = h_mean.copy()
        p_sigma = (1 - c_sigma) * p_sigma + math.sqrt(c_sigma * (2 - c_sigma) * mu_eff) * (C_half @ y_w)
        t = gen + 1
        denom = math.sqrt(1.0 - (1.0 - c_sigma) ** (2 * t))
        h_sigma = 1.0 if (np.linalg.norm(p_sigma) / denom < (1.4 + 2.0 / (d + 1)) * chi_d) else 0.0
        p_c = (1 - c_c) * p_c + h_sigma * math.sqrt(c_c * (2 - c_c) * mu_eff) * y_w
        C_new = (1 - c1 - c_mu) * C + c1 * (np.outer(p_c, p_c) + (1 - h_sigma) * c_c * (2 - c_c) * C)
        for i in range(mu_sel):
            C_new += c_mu * w[i] * np.outer(y_sel[i], y_sel[i])
        C_new = 0.5 * (C_new + C_new.T)
        if not np.all(np.isfinite(C_new)):
            C = np.eye(d)
            p_sigma = np.zeros(d)
            p_c = np.zeros(d)
            sigma = max(1e-4, sigma * 0.5)
            mu = mu_new
            continue
        C = C_new
        expo = np.clip(c_sigma / d_sigma * (np.linalg.norm(p_sigma) / chi_d - 1.0), -1.0, 1.0)
        sigma = float(np.clip(sigma * math.exp(float(expo)), 1e-4, 0.5))
        mu = mu_new
        if best_score < gen_best - 1e-12:
            stag = 0
        else:
            stag += 1
            if stag >= stag_limit or sigma <= 1e-4:
                break
    return best_h

def subgrad_polish(h, N, m, iters, rng):
    best = h.copy()
    best_s = score_exact(h, N)
    lr0 = 0.02
    for it in range(iters):
        h_curr = best.copy()
        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        s_star = int(np.argmax(corr))
        k = s_star - (N - 1)
        grad = np.zeros(N)
        for j in range(N):
            val = 0.0
            if 0 <= j + k < N:
                val += 1.0 - h_curr[j + k]
            if 0 <= j - k < N:
                val -= h_curr[j - k]
            grad[j] = val
        lr = lr0 / (1.0 + 0.01 * it)
        h_new = project_box_sum(h_curr - lr * grad, m)
        s_new = score_exact(h_new, N)
        if s_new < best_s - 1e-14:
            best = h_new
            best_s = s_new
        else:
            lr0 *= 0.7
            if lr0 < 1e-6:
                break
        if it % 50 == 49:
            h_pert = project_box_sum(best + 0.01 * rng.standard_normal(N), m)
            s_pert = score_exact(h_pert, N)
            if s_pert < best_s - 1e-14:
                best = h_pert
                best_s = s_pert
    return best, best_s
