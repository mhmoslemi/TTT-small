def smooth_obj_grad(h, tau, N):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2N-1
    c = dx * corr
    cmax = c.max()
    # Stable log-sum-exp
    e = np.exp((c - cmax) / tau)
    se = e.sum()
    p = e / se  # softmax weights
    F = cmax + tau * np.log(se)
    
    grad = np.zeros(N)
    for k in range(-(N-1), N):
        r = k + N - 1  # index in corr array
        w = dx * p[r]
        if w < 1e-15:
            continue
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            d = k
            L = N - d
            # h[n+d]*(1-h[n]) for n=0..L-1
            # d/dh_j: (1-h[j-d]) for d<=j<N, -(h[j+d]) for 0<=j<L
            grad[d:] += w * (1.0 - h[:L])
            grad[:L] -= w * h[d:]
        else:
            d = -k
            L = N - d
            # h[n-d]*(1-h[n]) for n=d..N-1
            # d/dh_j: (1-h[j+d]) for 0<=j<L, -(h[j-d]) for d<=j<N
            grad[:L] += w * (1.0 - h[d:])
            grad[d:] -= w * h[:L]
    
    return F, grad, c

import numpy as np

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    if r > 0:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
            if r <= 1e-14:
                break
    elif r < 0:
        for i in range(N):
            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1 + 1e-10:
        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)

def exact_score(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / N)

def is_feasible(h, N, m):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return False
    if h.min() < -1e-9 or h.max() > 1.0 + 1e-9:
        return False
    return abs(h.sum() - m) <= 1e-7 * max(1, N)

def smooth_obj_grad(h, tau, N):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = dx * corr
    cmax = c.max()
    e = np.exp((c - cmax) / tau)
    se = e.sum()
    p = e / se
    F = cmax + tau * np.log(se)
    grad = np.zeros(N)
    for k in range(-(N - 1), N):
        r = k + N - 1
        w = dx * p[r]
        if w < 1e-15:
            continue
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            d = k
            L = N - d
            grad[d:] += w * (1.0 - h[:L])
            grad[:L] -= w * h[d:]
        else:
            d = -k
            L = N - d
            grad[:L] += w * (1.0 - h[d:])
            grad[d:] -= w * h[:L]
    return F, grad, c

def optimize_start(h0, N, m, max_iter):
    taus = [0.02, 0.006, 0.002, 0.0006, 0.0002, 0.00006, 0.000015]
    h = h0.copy()
    best_h = h.copy()
    best_s = exact_score(h, N)
    for tau in taus:
        F, grad, _ = smooth_obj_grad(h, tau, N)
        if not np.all(np.isfinite(grad)):
            break
        alpha = 1.0
        for it in range(max_iter):
            trial = project_box_sum(h - alpha * grad, m)
            if not is_feasible(trial, N, m):
                alpha *= 0.5
                if alpha < 1e-10:
                    break
                continue
            d = trial - h
            if not np.all(np.isfinite(d)):
                break
            if np.max(np.abs(d)) <= 1e-9:
                break
            s_trial = exact_score(trial, N)
            if s_trial < best_s:
                best_s = s_trial
                best_h = trial.copy()
            F_y, grad_y, _ = smooth_obj_grad(trial, tau, N)
            gd = np.dot(grad, d)
            if gd < 0 and F_y <= F + 1e-4 * gd + 1e-12:
                h = trial
                F = F_y
                grad = grad_y
                alpha = min(1.25 * alpha, 10.0)
            else:
                alpha *= 0.5
                if alpha < 1e-10:
                    break
    s_end = exact_score(h, N)
    if s_end < best_s:
        best_s = s_end
        best_h = h.copy()
    return best_h, best_s

def make_fourier(N, rng, n_modes, amp):
    z = np.full(N, 0.5)
    idx = np.arange(N, dtype=np.float64)
    ns = min(n_modes, (N - 1) // 2)
    for r in range(1, ns + 1):
        A = rng.standard_normal()
        B = rng.standard_normal()
        z += (amp / r) * (A * np.sin(2.0 * np.pi * r * idx / N) + B * np.cos(2.0 * np.pi * r * idx / N))
    return z
