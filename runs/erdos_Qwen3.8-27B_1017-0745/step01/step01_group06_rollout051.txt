            cap = h[idx]
        if len(idx) > 0 and cap.sum() > 1e-14:
            h[idx] = h[idx] + err * cap / cap.sum()
            h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - S) > 1e-8:
        return np.full(n, S / n)
    return h

def exact_score(h, dx):
    if not (np.all(np.isfinite(h)) and len(h) > 0):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lag_vec(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_grad_mat(h, n, dx):
    nl = 2 * n - 1
    G = np.zeros((nl, n))
    j = np.arange(n)
    for k in range(nl):
        ell = k - (n - 1)
        i1 = j - ell
        v1 = (i1 >= 0) & (i1 < n)
        i2 = j + ell
        v2 = (i2 >= 0) & (i2 < n)
        t1 = np.zeros(n)
        t1[v1] = 1.0 - h[i1[v1]]
        t2 = np.zeros(n)
        t2[v2] = h[i2[v2]]
        G[k] = dx * (t1 - t2)
    return G

def feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def tr_obj(z, eta):
    d = z[:-1]
    return z[-1] + 0.5 / eta * np.dot(d, d)

def tr_obj_jac(z, eta):
    g = np.zeros(len(z))
    g[:-1] = z[:-1] / eta
    g[-1] = 1.0
    return g

def tr_ineq(z, q, G):
    return z[-1] - q - G @ z[:-1]

def tr_ineq_jac(z, q, G):
    J = np.empty((G.shape[0], G.shape[1] + 1))
    J[:, :-1] = -G
    J[:, -1] = 1.0
    return J

def tr_eq(z):
    return z[:-1].sum()

def tr_eq_jac(z):
    g = np.ones(len(z))
    g[-1] = 0.0
    return g

def transfer_h(hp, n0, m):
    S = m / 2.0
    y = np.zeros(m)
    for jj in range(m):
        tot = 0.0
        for ii in range(n0):
            ov = min((jj + 1) / m, (ii + 1) / n0) - max(jj / m, ii / n0)
            if ov > 0:
                tot += hp[ii] * ov
        y[jj] = m * tot
    return project_K(y, m, S)

def local_solve(h, n, dx, S, max_iter, slq_max):
    h = project_K(h.copy(), n, S)
    delta, eta, cfail, crej = 0.12, 0.5, 0, 0
    for _ in range(max_iter):
        q = lag_vec(h, dx)
        G = lag_grad_mat(h, n, dx)
        d_lo = np.maximum(-delta, -h)
        d_hi = np.minimum(delta, 1.0 - h)
        if np.any(d_hi - d_lo < -1e-14):
            break
        bounds = list(zip(d_lo, d_hi)) + [(None, None)]
        z0 = np.concatenate([np.zeros(n), [float(q.max())]])
        con = [
            {'type': 'ineq', 'fun': tr_ineq, 'jac': tr_ineq_jac, 'args': (q, G)},
            {'type': 'eq', 'fun': tr_eq, 'jac': tr_eq_jac}
        ]
        try:
            res = minimize(tr_obj, z0, jac=tr_obj_jac, args=(eta,),
                           method='SLSQP', bounds=bounds, constraints=con,
                           options={'maxiter': slq_max, 'ftol': 1e-10})
        except Exception:
            delta *= 0.5; cfail += 1
            if cfail >= 4 or delta < 1e-5:
                break
            continue
        if res is None or not np.all(np.isfinite(res.x)) or len(res.x) != n + 1:
            delta *= 0.5; cfail += 1
            if cfail >= 4 or delta < 1e-5:
                break
            continue
        d_sol = res.x[:-1]
        if abs(d_sol.sum()) > 1e-4 or np.max(np.abs(res.x)) < 0:
            delta *= 0.5; cfail += 1
            if cfail >= 4 or delta < 1e-5:
                break
            continue
        cfail = 0
        ht = project_K(h + d_sol, n, S)
        if not feasible(ht, n, S):
            delta *= 0.5
            if delta < 1e-5:
                break
            continue
        da = ht - h
        if np.max(np.abs(da)) < 1e-9:
            break
        F0 = float(q.max())
        F1 = exact_score(ht, dx)
        pred = F0 - float((q + G @ da).max())
        actual = F0 - F1
        if actual > 1e-12:
            h = ht
            crej = 0
            if pred > 1e-12:
                r = actual / pred
                if r >= 0.75 and np.max(np.abs(da)) >= 0.7 * delta:
                    delta = min(0.5, 2.0 * delta)
                    eta = min(2.0, 1.5 * eta)
                elif r < 0.25:
                    delta *= 0.5
                    eta = max(1e-4, 0.5 * eta)
        else:
            crej += 1; delta *= 0.5; eta = max(1e-4, 0.5 * eta)
            if crej >= 8 or delta < 1e-5:
                break
    return project_K(h, n, S)

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    init = globals().get('initial_h_values', None)
    ok = False
    if init is not None:
        init = np.array(init, dtype=np.float64).ravel()
        if 2 <= len(init) <= 999 and np.all(np.isfinite(init)) and init.min() >= -1e-6 and init.max() <= 1.0 + 1e-6:
            ok = True
    if ok:
        n0 = len(init)
        h_parent = project_K(init.copy(), n0, n0 / 2.0)
        s0 = exact_score(h_parent, 2.0 / n0)
        res_list = [n0]
        if n0 > 2:
            res_list.append(n0 - 1)
        if n0 < 999:
            res_list.append(n0 + 1)
        res_list = list(dict.fromkeys(res_list))
    else:
        nd = kwargs.get('n_points', 45)
        if not isinstance(nd, (int, np.integer)):
            nd = 45
        nd = max(2, min(999, int(nd)))
        n0 = nd
        h_parent = np.full(nd, 0.5)
        s0 = exact_score(h_parent, 2.0 / nd)
        res_list = [nd]
    best_h, best_s, best_n = h_parent.copy(), s0, n0
    for m in res_list:
        dx, S = 2.0 / m, m / 2.0
        if m == n0 and ok:
            base = h_parent.copy()
        elif ok:
            base = transfer_h(h_parent, n0, m)
        else:
            base = np.full(m, 0.5)
        starts = [base.copy(), (1.0 - base).copy(), base[::-1].copy(),
                  project_K(base + 0.08 * rng.standard_normal(m), m, S),
                  project_K(base + 0.20 * rng.standard_normal(m), m, S),
                  project_K(base + 0.40 * rng.standard_normal(m), m, S),
                  project_K(rng.uniform(0, 1, m), m, S),
                  project_K(rng.beta(0.5, 0.5, m), m, S),
                  project_K(0.5 + 0.45 * np.cos(2.0 * np.pi * np.arange(m) / m), m, S),
                  project_K(0.5 + 0.4 * np.sin(4.0 * np.pi * np.arange(m) / m), m, S)]
        hb = np.zeros(m)
        hb[:int(S)] = 1.0
        if S - int(S) > 0:
            hb[int(S)] = S - int(S)
        starts.append(project_K(hb, m, S))
        deduped = []
        for s in starts:
            dup = any(np.max(np.abs(s - d)) <= 1e-6 for d in deduped)
            if not dup:
                deduped.append(s)
        for s in deduped:
            sc = exact_score(s, dx)
            if sc < best_s - 1e-12:
                best_s, best_h, best_n = sc, s.copy(), m
        if m <= 96:
            ns, ni, slq, do_rst = min(len(deduped), 8), 30, 50, True
        elif m <= 256:
            ns, ni, slq, do_rst = min(len(deduped), 4), 12, 30, False
        else:
            ns, ni, slq, do_rst = min(len(deduped), 1), 5, 15, False
        for s in deduped[:ns]:
            if not feasible(s, m, S):
                continue
            ho = local_solve(s, m, dx, S, ni, slq)
            if feasible(ho, m, S):
                sc = exact_score(ho, dx)
                if sc < best_s - 1e-12:
                    best_s, best_h, best_n = sc, ho.copy(), m
        if do_rst:
            for sig in [0.12, 0.22]:
                hr = project_K(base + sig * rng.standard_normal(m), m, S)
                if feasible(hr, m, S):
                    ho = local_solve(hr, m, dx, S, ni, slq)
                    if feasible(ho, m, S):
                        sc = exact_score(ho, dx)
                        if sc < best_s - 1e-12:
                            best_s, best_h, best_n = sc, ho.copy(), m
    fn, fdx, fs = best_n, 2.0 / best_n, best_n / 2.0
    best_h = project_K(best_h, fn, fs)
    if not feasible(best_h, fn, fs):
        best_h = np.full(fn, fs / fn)
    final_s = exact_score(best_h, fdx)
    if ok and final_s > s0 + 1e-10:
        best_h, final_s, fn = h_parent.copy(), s0, n0
    best_h = np.clip(best_h, 0.0, 1.0)
    return (best_h, float(exact_score(best_h, 2.0 / fn)), int(fn))
