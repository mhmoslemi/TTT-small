def project_capped(z, n):
    s = n / 2.0
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None

    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if hi - lo < 1e-15:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for idx in range(n):
        if abs(r) < 1e-13 * max(1, n):
            break
        if r > 0:
            d = min(r, 1.0 - h[idx])
            h[idx] += d
            r -= d
        else:
            d = max(r, -h[idx])
            h[idx] += d
            r -= d
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(np.sum(h) - s) > 5e-12 * max(1, n):
        return None
    return h

    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return float('inf')

        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_vals_grads(h, n):
    L = 2 * n - 1
    v = np.zeros(L)
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        idx = np.arange(lo, hi)
        jdx = idx + d
        v[row] = float(np.sum(h[idx] * (1.0 - h[jdx])))
        G[row, idx] += (1.0 - h[jdx])
        G[row, jdx] -= h[idx]
    return v, G

def obj_fn(z, n):
    return z[-1]

def obj_grad(z, n):
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return g

def eq_fn(z, n):
    return float(np.sum(z[:n])) - n / 2.0

def eq_jac(z, n):
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def ineq_fn(z, n):
    h = z[:n]
    t = z[-1]
    dx = 2.0 / n
    v, _ = lag_vals_grads(h, n)
    return t - dx * v

def ineq_jac(z, n):
    h = z[:n]
    dx = 2.0 / n
    L = 2 * n - 1
    _, G = lag_vals_grads(h, n)
    J = np.zeros((L, n + 1))
    J[:, :n] = -dx * G
    J[:, n] = 1.0
    return J

def add_start(starts, raw, n, seen):

def polish_pairs(hb, n, sweeps):
    for _ in range(sweeps):
        improved = False
        for i in range(n - 1):
            for j in range(i + 1, n):
                lo = max(-hb[i], hb[j] - 1.0)
                hi = min(1.0 - hb[i], hb[j])
                if lo >= hi:
                    continue
                cur = exact_score(hb, n)
                deltas = [0.0, lo, hi]
                for delta in (0.001, 0.003, 0.01, 0.03, 0.1, 0.3):
                    if lo <= delta <= hi:
                        deltas.append(delta)
                    if lo <= -delta <= hi:
                        deltas.append(-delta)
                best_sc, best_d = cur, 0.0
                for d in deltas:
                    d = max(lo, min(hi, d))
                    try_h = hb.copy()
                    try_h[i] += d
                    try_h[j] -= d

                if best_d != 0.0:
                    hb[i] += best_d
                    hb[j] -= best_d
                    improved = True
        if not improved:
            break
    return hb
