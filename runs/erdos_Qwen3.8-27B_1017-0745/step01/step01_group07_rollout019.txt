import numpy as np
import math

def proj_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):
        return None
    n = len(z)
    if np.max(np.abs(z)) > 1e12:
        return None
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(80):
        mid = lo + 0.5 * (hi - lo)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = lo + 0.5 * (hi - lo)
    y = np.clip(z - lam, 0.0, 1.0)
    r = S - np.sum(y)
    if r > 0:
        for i in range(n):
            if r <= 0: break
            a = min(r, 1.0 - y[i])
            y[i] += a; r -= a
    elif r < 0:
        for i in range(n):
            if r >= 0: break
            a = min(-r, y[i])
            y[i] -= a; r += a
    if abs(np.sum(y) - S) > 1e-10 or not np.all(np.isfinite(y)):
        return None
    return np.clip(y, 0.0, 1.0)

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-8)

def exact_score(h, n):
    vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(vals)) or len(vals) != 2*n-1:
        return None
    return float(np.max(vals))

def solve_dc_subproblem(anchor, n, S, dx, rho, eta):
    try:
        import cvxpy as cp
    except ImportError:
        return None
    y = cp.Variable(n)
    t = cp.Variable()
    constraints = [y >= 0, y <= 1, cp.sum(y) == S,
                   y - anchor <= rho, anchor - y <= rho]
    for d in range(-(n-1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        y0 = y[lo:hi]
        y1 = y[lo+d:hi+d]
        a0 = anchor[lo:hi]
        a1 = anchor[lo+d:hi+d]
        s = a0 + a1
        expr = dx*(cp.sum(y0) + 0.25*cp.sum_squares(y0-y1)
                   - 0.5*s@cp.vec(y0+y1) + 0.25*cp.sum_squares(s))
        constraints.append(expr <= t)
    obj = cp.Minimize(t + 0.5*eta*cp.sum_squares(y - anchor))
    prob = cp.Problem(obj, constraints)
    for solver in [cp.ECOS, cp.CVXOPT]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                yv = np.asarray(y.value, dtype=np.float64).flatten()
                if yv.shape[0] == n and np.all(np.isfinite(yv)):
                    return yv
        except Exception:
            continue
    return None

def build_starts(n, S, parent, rng):

    if parent is not None:

        starts.append(1.0 - parent)
        starts.append(parent[::-1].copy())

            if np.max(np.abs(s - d)) < 1e-9:

        if not dup:

    return deduped

def optimize_dc(x, n, S, dx, best_h, best_score, max_attempts):

    rho_max = 0.40
    rho_min = 0.005
    eta = 0.20 * dx

    no_improve = 0
    no_solver = 0

        if cand is None:

            continue
        proj = proj_simplex(cand, S)

                best_trial = trial.copy()
            if tf < best_score - 1e-12:

                best_h = trial.copy()
        if best_trial is not None:

            f = best_trial_f
            no_improve = 0
            if np.max(np.abs(best_trial - x)) < 1e-7:

        else:
            no_improve += 1
            rho = max(rho_min, 0.5 * rho)
            eta = max(eta_min, 0.5 * eta)
            if no_improve >= 3:
