        DQ[k + n - 1] = d
    return DQ

def build_starts(parent, n, S, rng):
    raw = []
    if parent is not None:
        raw.append(parent.copy())
        raw.append(parent[::-1].copy())
        raw.append(1.0 - parent)
        raw.append(1.0 - parent[::-1])
        raw.append(parent[rng.permutation(n)])
        raw.append(rng.uniform(0, 1, n))
        raw.append(0.5 + 0.25 * rng.standard_normal(n))
    else:
        raw.append(np.full(n, 0.5))
        raw.append(rng.uniform(0, 1, n))
        raw.append(0.5 + 0.3 * rng.standard_normal(n))
        for m in range(1, min(4, n // 2 + 1)):
            x = np.arange(n)
            raw.append(0.5 + 0.35 * np.cos(2 * np.pi * m * x / n + rng.uniform(0, 2 * np.pi)))
    valid = []
    for s in raw:
        c = project_box_sum(s, S)
        if c is not None and feasible(c, n, S):
            dup = any(np.max(np.abs(c - v)) < 1e-10 for v in valid)
            if not dup:
                valid.append(c)
    cap = 8 if n <= 128 else 5
    return valid[:cap]

def anneal_start(start, n, S, rng, moves_per_start):
    h = project_box_sum(start, S)
    if h is None or not feasible(h, n, S):
        return None
    Q = np.correlate(h, 1.0 - h, mode='full')
    C_cur = float(np.max(Q)) * (2.0 / n)
    best = h.copy()
    best_sc = C_cur
    radii = [0.20, 0.12, 0.07, 0.04, 0.02, 0.008]
    stages = 6
    T_base = min(0.03, 0.003 * (45.0 / n))
    mps = max(1, moves_per_start // stages)
    for s in range(stages):
        T = T_base * (0.35 ** s)
        rho = radii[s]
        extra = 1 if s < moves_per_start % stages else 0
        accepted = 0
        for _ in range(mps + extra):
            p = int(rng.integers(0, n))
            q = int(rng.integers(0, n))
            if p == q:
                continue
            if p > q:
                p, q = q, p
            a = float(rng.uniform(-rho, rho))
            a_lo = max(-h[p], h[q] - 1.0)
            a_hi = min(1.0 - h[p], h[q])
            if a < a_lo or a > a_hi:
                continue
            DQ = pair_delta_lags(h, p, q, a, n)
            if not np.all(np.isfinite(DQ)):
                continue
            Qt = Q + DQ
            Ct = float(np.max(Qt)) * (2.0 / n)
            if not np.isfinite(Ct):
                continue
            if Ct <= C_cur or rng.random() < np.exp(max(-700.0, -(Ct - C_cur) / T)):
                h[p] += a
                h[q] -= a
                Q, C_cur = Qt, Ct
                accepted += 1
                if accepted % 64 == 0:
                    Q = np.correlate(h, 1.0 - h, mode='full')
                    C_cur = float(np.max(Q)) * (2.0 / n)
                    if C_cur < best_sc - 1e-14:
                        best_sc = C_cur
                        best = h.copy()
        Q = np.correlate(h, 1.0 - h, mode='full')
        C_cur = float(np.max(Q)) * (2.0 / n)
        if C_cur < best_sc - 1e-14:
            best_sc = C_cur
            best = h.copy()
    return best

def greedy_polish(h, n, S, rng):
    h = h.copy()
    Q = np.correlate(h, 1.0 - h, mode='full')
    C_cur = float(np.max(Q)) * (2.0 / n)
    best = h.copy()
    best_sc = C_cur
    for rho in [0.02, 0.006, 0.0015]:
        P = max(100, min(1500, int(60000.0 / n)))
        for _ in range(P):
            p = int(rng.integers(0, n))
            q = int(rng.integers(0, n))
            if p == q:
                continue
            if p > q:
                p, q = q, p
            a = float(rng.uniform(-rho, rho))
            a_lo = max(-h[p], h[q] - 1.0)
            a_hi = min(1.0 - h[p], h[q])
            if a < a_lo or a > a_hi:
                continue
            DQ = pair_delta_lags(h, p, q, a, n)
            if not np.all(np.isfinite(DQ)):
                continue
            Qt = Q + DQ
            Ct = float(np.max(Qt)) * (2.0 / n)
            if Ct < C_cur - 1e-13:
                h[p] += a
                h[q] -= a
                Q, C_cur = Qt, Ct
        Q = np.correlate(h, 1.0 - h, mode='full')
        C_cur = float(np.max(Q)) * (2.0 / n)
        if C_cur < best_sc - 1e-14:
            best_sc = C_cur
            best = h.copy()
        else:
            h = best.copy()
            Q = np.correlate(h, 1.0 - h, mode='full')
            C_cur = float(np.max(Q)) * (2.0 / n)
    return best, best_sc

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    n = 45
    parent = None
    try:
        pv = np.array(initial_h_values, dtype=np.float64)
        if pv.ndim == 1 and 2 <= len(pv) <= 999 and np.all(np.isfinite(pv)):
            if np.all(pv >= -1e-9) and np.all(pv <= 1.0 + 1e-9):
                n = len(pv)
                tgt = n / 2.0
                if abs(np.sum(pv) - tgt) < 1e-6 * max(1.0, float(n)):
                    parent = project_box_sum(np.clip(pv, 0.0, 1.0), tgt)
    except Exception:
        pass
    S = n / 2.0
    if parent is None:
        parent = project_box_sum(np.full(n, 0.5), S)
    fallback = parent.copy()
    fallback_sc = exact_score(fallback, n)
    starts = build_starts(parent, n, S, rng)
    if not starts:
        starts = [parent.copy()]
    B_total = max(600, min(12000, int(540000.0 / n)))
    moves_per_start = max(60, B_total // len(starts))
    best_h = parent.copy()
    best_sc = fallback_sc
    for st in starts:
        try:
            res = anneal_start(st, n, S, rng, moves_per_start)
            if res is not None:
                sc = exact_score(res, n)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_h = res
        except Exception:
            continue
    try:
        ph, psc = greedy_polish(best_h, n, S, rng)
        if psc < best_sc - 1e-14:
            best_sc = psc
            best_h = ph
    except Exception:
        pass
    final_h = project_box_sum(best_h, S)
    if final_h is None or not feasible(final_h, n, S):
        final_h = fallback
    final_sc = exact_score(final_h, n)
    if not np.isfinite(final_sc) or final_sc > fallback_sc + 1e-12:
        final_h, final_sc = fallback, fallback_sc
    return (np.array(final_h, dtype=np.float64), final_sc, n)
