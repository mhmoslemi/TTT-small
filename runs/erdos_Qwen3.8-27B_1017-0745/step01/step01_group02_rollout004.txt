import numpy as np
import math

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1 or len(v) == 0 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    if S < -1e-12 or S > n + 1e-12:
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - float(np.sum(y))
    if abs(r) > 1e-14:
        if r > 0:
            for i in np.argsort(1.0 - y):
                d = min(r, 1.0 - y[i])
                if d > 0:
                    y[i] += d
                    r -= d
                if r <= 1e-14:
                    break
        else:
            for i in np.argsort(y):
                d = min(-r, y[i])
                if d > 0:
                    y[i] -= d
                    r += d
                if r >= -1e-14:
                    break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-10) or np.any(y > 1.0 + 1e-10):
        return None
    if abs(float(np.sum(y)) - S) > 1e-9 * max(1, n):
        return None
    return y

def check_feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and np.all(h >= -1e-12) and
            np.all(h <= 1.0 + 1e-12) and abs(float(np.sum(h)) - 0.5 * n) < 1e-9 * max(1, n))

def exact_score(h):
    n = len(h)
    if not check_feasible(h, n):
        return float('inf')
    try:
        c = np.correlate(h, 1.0 - h, mode='full')
        if not np.all(np.isfinite(c)) or len(c) != 2 * n - 1:
            return float('inf')
        return float(np.max(c)) * (2.0 / n)
    except Exception:
        return float('inf')

def get_parent():
    if 'initial_h_values' in globals():
        ph = np.asarray(globals()['initial_h_values'], dtype=np.float64)
        if ph.ndim == 1 and 2 <= len(ph) <= 999 and np.all(np.isfinite(ph)):
            if np.all(ph >= -1e-6) and np.all(ph <= 1.0 + 1e-6):
                n = len(ph)
                if abs(float(np.sum(ph)) - 0.5 * n) < 1e-5 * max(1, n):
                    p = project_box_sum(np.clip(ph, 0.0, 1.0), 0.5 * n)
                    if p is not None and check_feasible(p, n):
                        return p, n
    return np.full(45, 0.5), 45

def vdist(a, b):
    return float(np.max(np.abs(a - b)))

def try_add_member(members, cand, n, rng):
    if len(members) >= 200:
        return
    p = project_box_sum(cand, 0.5 * n)
    if p is None or not check_feasible(p, n):
        return
    for m in members:
        if vdist(p, m) < 1e-10:
            return
    members.append(p)

def build_pop(parent, n, P, rng):
    members = [parent.copy()]
    for c in [1.0 - parent, parent[::-1].copy(), np.roll(parent, 1)]:
        try_add_member(members, c, n, rng)
    for sd in [0.03, 0.08, 0.18, 0.35]:
        try_add_member(members, parent + sd * rng.standard_normal(n), n, rng)
    for _ in range(6):
        try_add_member(members, rng.random(n), n, rng)
    idx = np.arange(n)
    for m in range(1, min(5, n // 2 + 1)):
        phi = rng.uniform(0, 2 * math.pi)
        th = rng.uniform(0, 2 * math.pi)
        u = 0.5 + 0.28 * np.cos(2 * math.pi * m * idx / n + phi) + 0.12 * np.sin(2 * math.pi * (m + 1) * idx / n + th)
        try_add_member(members, u, n, rng)
    flip, att = False, 0
    while len(members) < P and att < 30 * P:
        att += 1
        c = (parent + rng.uniform(0.03, 0.35) * rng.standard_normal(n)) if flip else rng.random(n)
        flip = not flip
        try_add_member(members, c, n, rng)
    while len(members) < P:
        members.append(parent.copy())
    return np.array(members[:P])

def sample_F(rng, muF):
    for _ in range(10):
        U = min(max(rng.random(), 1e-6), 1.0 - 1e-6)
        F = muF + 0.1 * math.tan(math.pi * (U - 0.5))
        if F > 0 and np.isfinite(F):
            return min(max(F, 0.05), 1.0)
    return muF

def run(seed=42, **kwargs):

        rng = np.random.default_rng(seed)

        oc = np.argsort(ss)

        ga = list(archive)
        for i in range(P):
            ot = [t for t in top if t != i]
            pb = ot[int(rng.integers(len(ot)))] if ot else top[int(rng.integers(len(top)))]

            else:

            Fi = sample_F(rng, muF)
            CRi = min(max(rng.normal(muCR, 0.1), 0.0), 1.0)

            donor = xi + Fi * (xp - xi) + Fi * (xr1 - r2v)
            mask = rng.random(n) < CRi
