def project_onto_k(y, n):
    S = n / 2.0
    if not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        c = np.clip(y - mid, 0.0, 1.0)
        if c.sum() > S:
            lo = mid
        else:
            hi = mid
    c = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(50):
        d = S - c.sum()
        if abs(d) < 1e-14:
            break
        mask = (c < 1.0 - 1e-12) if d > 0 else (c > 1e-12)
        cnt = mask.sum()
        if cnt == 0:
            break
        c[mask] += d / cnt
        np.clip(c, 0.0, 1.0, out=c)
    return c

def exact_score(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_scores(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

            and abs(h.sum() - n / 2.0) <= 1e-10)

    b = np.zeros(m)

        bk = 0.0
        if 0 <= i - k < n:

        return h, q, current_s
    hc = h.copy()
    hc[i] += best_t
