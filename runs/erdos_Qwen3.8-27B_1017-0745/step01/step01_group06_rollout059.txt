    if n % 2 == 1:
        hb[n // 2] = 1.0 / n
    cands.append(project_K(hb, n, S))
    cands.append(project_K(0.5 + 0.45 * np.cos(2.0 * np.pi * js / n), n, S))
    cands.append(project_K(0.5 + 0.3 * np.sin(2.0 * np.pi * js / n) + 0.15 * np.cos(4.0 * np.pi * js / n), n, S))
    if is_parent:
        cands.append(project_K(base[rng.permutation(n)], n, S))
    dx = 2.0 / n
    dedup = []
    for c in cands:
        c = project_K(c, n, S)
        if not feasible(c, n, S):
            continue
        dup = False
        for _, dc in dedup:
            if np.max(np.abs(c - dc)) <= 1e-5:
                dup = True
                break
        if not dup:
            dedup.append((exact_score(c, dx), c.copy()))
    return dedup

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    init = globals().get('initial_h_values', None)
    ok, n0 = False, 45
    if init is not None:
        try:
            init = np.array(init, dtype=np.float64).ravel()
            if 2 <= len(init) <= 999 and np.all(np.isfinite(init)) and init.min() >= -1e-6 and init.max() <= 1.0 + 1e-6:
                n0 = len(init)
                if abs(init.sum() - n0 / 2.0) <= 1e-6:
                    ok = True
        except Exception:
            pass
    if not ok:
        nd = kwargs.get('n_points', 45)
        if isinstance(nd, (int, np.integer)) and 2 <= nd <= 999:
            n0 = int(nd)
    parent = np.clip(np.array(init, dtype=np.float64).ravel(), 0, 1) if ok else np.full(n0, 0.5)
    parent = project_K(parent, n0, n0 / 2.0)
    best_h, best_n, best_s = parent.copy(), n0, exact_score(parent, 2.0 / n0)
    fb_h, fb_n, fb_s = parent.copy(), n0, best_s
    resolutions = [n0]
    if ok:
        for m in [n0 - 1, n0 + 1]:
            if 2 <= m <= 999 and m not in resolutions:
                resolutions.append(m)
    for m in resolutions:
        S_m, dx_m = m / 2.0, 2.0 / m
        base = parent.copy() if (m == n0 and ok) else (transfer(parent, n0, m) if ok else np.full(m, 0.5))
        base = project_K(base, m, S_m)
        starts = make_starts(base, m, S_m, rng, ok and m == n0)
        for sc, sv in starts:
            if sc < best_s - 1e-12:
                best_s, best_h, best_n = sc, sv.copy(), m
        if m <= 96:
            ns, mm, ms = min(8, len(starts)), 32, 55
        elif m <= 256:
            ns, mm, ms = min(3, len(starts)), 14, 35
        else:
            ns, mm, ms = 1, 5, 15
        starts.sort(key=lambda x: x[0])
        lbh, lbs = base.copy(), exact_score(base, dx_m)
        for idx in range(ns):
            _, sv = starts.pop(0)
            ho, so = local_search(sv, m, dx_m, S_m, mm, ms, rng)
            ho = project_K(ho, m, S_m)
            so = exact_score(ho, dx_m)
            if feasible(ho, m, S_m):
                if so < lbs:
                    lbs, lbh = so, ho.copy()
                if so < best_s - 1e-12:
                    best_s, best_h, best_n = so, ho.copy(), m
        if m <= 96:
            for bp in [0.15, 0.30]:
                hp = project_K(lbh + bp * rng.standard_normal(m), m, S_m)
                if feasible(hp, m, S_m):
                    ho, so = local_search(hp, m, dx_m, S_m, mm, ms, rng)
                    ho = project_K(ho, m, S_m)
                    so = exact_score(ho, dx_m)
                    if feasible(ho, m, S_m) and so < best_s - 1e-12:
                        best_s, best_h, best_n = so, ho.copy(), m
    best_h = project_K(best_h, best_n, best_n / 2.0)
    fs = exact_score(best_h, 2.0 / best_n)
    if not feasible(best_h, best_n, best_n / 2.0) or fs > fb_s + 1e-10:
        best_h, best_s, best_n = fb_h, fb_s, fb_n
    else:
        best_s = fs
    best_h = np.clip(best_h, 0.0, 1.0)
    return (best_h, float(best_s), int(best_n))
