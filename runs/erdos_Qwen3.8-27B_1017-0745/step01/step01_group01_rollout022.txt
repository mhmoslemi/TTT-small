            Dp -= h[p - d]
        Dq = 0.0
        if 0 <= q + d < n:
            Dq += 1.0 - h[q + d]
        if 0 <= q - d < n:
            Dq -= h[q - d]
        b[row] = Dp - Dq
    if not np.all(np.isfinite(a)) or not np.all(np.isfinite(b)):
        return None
    return a, b

def quad_roots(A, B, C, lo, hi):
    roots = []
    scale = max(1.0, abs(A), abs(B), abs(C))
    if abs(A) <= 1e-14 * scale:
        if abs(B) <= 1e-14 * scale:
            return roots
        r = -C / B
        pad = 1e-12 * max(1.0, abs(lo), abs(hi))
        if lo - pad <= r <= hi + pad and np.isfinite(r):
            roots.append(float(np.clip(r, lo, hi)))
        return roots
    disc = B * B - 4.0 * A * C
    if disc < -1e-14 * (B * B + 4.0 * abs(A * C) + 1.0):
        return roots
    disc = max(0.0, disc)
    sq = math.sqrt(disc)
    qv = -0.5 * (B + math.copysign(sq, B))
    if abs(qv) > 1e-14:
        r1, r2 = qv / A, C / qv
    else:
        r1 = r2 = -B / (2.0 * A)
    pad = 1e-12 * max(1.0, abs(lo), abs(hi))
    for r in (r1, r2):
        if np.isfinite(r) and lo - pad <= r <= hi + pad:
            roots.append(float(np.clip(r, lo, hi)))
    return roots

def build_hull(slopes, intercepts, rows):
    ns = slopes.shape[0]
    if ns == 0:
        return np.array([], dtype=np.float64), np.array([], dtype=np.int32)
    order = np.argsort(slopes, kind='stable')
    hull_m, hull_c, hull_r, hull_s = [], [], [], []
    for idx in order:
        m, cc, rr = slopes[idx], intercepts[idx], rows[idx]
        if hull_m and abs(m - hull_m[-1]) < 1e-15:
            continue
        if not hull_m:
            hull_m.append(m)
            hull_c.append(cc)
            hull_r.append(rr)
            hull_s.append(-1e18)
        else:
            if abs(m - hull_m[-1]) < 1e-15:
                continue
            x_int = (hull_c[-1] - cc) / (m - hull_m[-1])
            while len(hull_m) > 1 and x_int <= hull_s[-1] + 1e-15:
                hull_m.pop()
                hull_c.pop()
                hull_r.pop()
                hull_s.pop()
                if not hull_m:
                    break
                if abs(m - hull_m[-1]) < 1e-15:
                    break
                x_int = (hull_c[-1] - cc) / (m - hull_m[-1])
            if not hull_m or abs(m - hull_m[-1]) > 1e-15:
                hull_m.append(m)
                hull_c.append(cc)
                hull_r.append(rr)
                hull_s.append(x_int)
    return np.array(hull_s, dtype=np.float64), np.array(hull_r, dtype=np.int32)

def exact_line_minimizer(h, p, q, n, c, a, b):
    lo = max(-h[p], h[q] - 1.0)
    hi = min(1.0 - h[p], h[q])
    if hi - lo <= 1e-14:
        return 0.0
    D = q - p
    L = 2 * n - 1
    special_set = set()
    for sr in (n - 1, n - 1 + D, n - 1 - D):
        if 0 <= sr < L:
            special_set.add(sr)
    special = sorted(special_set)
    affine_mask = np.ones(L, dtype=bool)
    for sr in special:
        affine_mask[sr] = False
    aff_idx = np.where(affine_mask)[0]
    candidates = [lo, hi, 0.0]
    if len(aff_idx) > 0:
        sl = b[aff_idx]
        ic = c[aff_idx]
        rs = aff_idx
        hull_starts, hull_rows = build_hull(sl, ic, rs)
        nh = len(hull_starts)
        for si in range(nh):
            seg_lo = max(lo, hull_starts[si])
            seg_hi = hi
            if si + 1 < nh:
                seg_hi = min(seg_hi, hull_starts[si + 1])
            if seg_lo >= seg_hi:
                continue
            if lo < hull_starts[si] < hi:
                candidates.append(float(hull_starts[si]))
            m_act = b[hull_rows[si]]
            c_act = c[hull_rows[si]]
            for sr in special:
                roots = quad_roots(a[sr], b[sr] - m_act, c[sr] - c_act, seg_lo, seg_hi)
                candidates.extend(roots)
    for i in range(len(special)):
        for j in range(i + 1, len(special)):
            si, sj = special[i], special[j]
            roots = quad_roots(a[si] - a[sj], b[si] - b[sj], c[si] - c[sj], lo, hi)
            candidates.extend(roots)
    for sr in special:
        if a[sr] > 1e-14:
            v = -b[sr] / (2.0 * a[sr])
            if lo - 1e-14 <= v <= hi + 1e-14:
                candidates.append(float(np.clip(v, lo, hi)))
    valid_cands = []
    for cd in candidates:
        if np.isfinite(cd) and lo - 1e-12 <= cd <= hi + 1e-12:
            valid_cands.append(float(np.clip(cd, lo, hi)))
    if not valid_cands:
        return 0.0
    seen_set = set()
    unique_cands = []
    for vc in valid_cands:
        key = round(vc, 12)
        if key not in seen_set:
            seen_set.add(key)
            unique_cands.append(vc)
    best_delta, best_val = 0.0, float('inf')
    for cd in unique_cands:
        vals = c + b * cd + a * cd * cd
        mv = float(np.max(vals))
        if mv < best_val:
            best_val = mv
            best_delta = cd
    return best_delta

def build_edges(n):
    edges = set()
    for i in range(n - 1):
        edges.add((i, i + 1))
    r = 2
    while r < n:
        for i in range(0, n - r, 2 * r):
            edges.add((i, i + r))
        r *= 2
    if n > 2:
        edges.add((0, n - 1))
    return sorted(edges)

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if len(iv) > 0 and len(iv) <= 999:
            n = len(iv)
            if np.all(np.isfinite(iv)) and np.all(iv >= -1e-8) and np.all(iv <= 1.0 + 1e-8):
                parent = iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
    if n < 1 or n > 999:
        n = 47
    try:
        sd = int(seed)
    except Exception:
        sd = 42
    sd = sd % (2**32)
    rng = np.random.default_rng(sd)
    uniform = np.full(n, 0.5)
    parent_feasible = None
    if parent is not None:
        parent_feasible = project_capped(parent, n)
    starts = []
    seen_starts = []
    all_candidates = [uniform]
    if parent_feasible is not None:
        all_candidates.append(parent_feasible)
        all_candidates.append(parent_feasible[::-1].copy())
        all_candidates.append(1.0 - parent_feasible[::-1].copy())
        all_candidates.append(parent_feasible + rng.normal(0, 0.04, n))
        all_candidates.append(parent_feasible + rng.normal(0, 0.12, n))
        shift = n // 3
        all_candidates.append(np.roll(parent_feasible, shift).copy())
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    all_candidates.append(blk)
    all_candidates.append(0.5 + 0.4 * np.cos(2.0 * np.pi * np.arange(n) / n))
    all_candidates.append(0.5 + 0.3 * np.cos(4.0 * np.pi * np.arange(n) / n))
    all_candidates.append(rng.uniform(0, 1, n))
    for av in all_candidates:
        h2 = project_capped(av, n)
        if h2 is None:
            continue
        dup = False
        for ex in seen_starts:
            if float(np.max(np.abs(h2 - ex))) < 1e-9:
                dup = True
                break
        if not dup:
            seen_starts.append(h2)
            starts.append(h2)
    if n <= 64:
        n_starts, max_sweeps = 5, 4
    elif n <= 96:
        n_starts, max_sweeps = 3, 2
    else:
        n_starts, max_sweeps = 3, 0
    try:
        n_starts = max(1, min(int(kwargs.get('starts', n_starts)), 10))
    except Exception:
        pass
    try:
        max_sweeps = max(0, min(int(kwargs.get('max_sweeps', max_sweeps)), 5))
    except Exception:
        pass
    starts = starts[:max(n_starts, len(starts))]
    best, best_score = uniform.copy(), exact_score(uniform, n)
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_score - 1e-12:
            best, best_score = st.copy(), ss
    edges = build_edges(n)
    if len(edges) == 0:
        edges = [(0, 0)]
    for st in starts[:n_starts]:
        x = st.copy()
        x_score = exact_score(x, n)
        if not np.isfinite(x_score):
            x_score = best_score
        for sweep in range(max_sweeps):
            edge_order = np.array(edges, dtype=np.int32)
            rng.shuffle(edge_order)
            accepted = False
            for ei in range(len(edge_order)):
                p, q = int(edge_order[ei, 0]), int(edge_order[ei, 1])
                if p == q:
                    continue
                c = lag_raw_values(x, n)
                if c is None:
                    continue
                coeffs = line_coefficients(x, p, q, n)
                if coeffs is None:
                    continue
                a_vec, b_vec = coeffs
                delta = exact_line_minimizer(x, p, q, n, c, a_vec, b_vec)
                best_trial, best_trial_score = None, x_score
                for k in range(7):
                    d_t = delta * (0.5 ** k)
                    if abs(d_t) < 1e-15:
                        break
                    y = x.copy()
                    y[p] += d_t
                    y[q] -= d_t
                    y = project_capped(y, n)
                    if y is None:
                        continue
                    s_y = exact_score(y, n)
                    if np.isfinite(s_y) and s_y < best_trial_score - 1e-12:
                        best_trial, best_trial_score = y, s_y
                if best_trial is not None:
                    x = best_trial
                    x_score = best_trial_score
                    accepted = True
                    if x_score < best_score - 1e-12:
                        best, best_score = x.copy(), x_score
            if not accepted:
                break
    final = project_capped(best, n)
    if final is None:
        final = best.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > best_score + 1e-12:
        final = best.copy()
        fs = exact_score(best, n)
    if parent_feasible is not None:
        ps = exact_score(parent_feasible, n)
        if np.isfinite(ps) and ps < fs - 1e-12:
            final, fs = parent_feasible.copy(), ps
    us = exact_score(uniform, n)
    if not np.isfinite(fs) or fs > us:
        final, fs = uniform.copy(), us
    final = np.clip(final, 0.0, 1.0)
    if not np.all(np.isfinite(final)):
        final = uniform.copy()
        fs = us
    return final, float(fs), int(n)
