import numpy as np

def project_box_sum(z, m):
    # Bisection for lambda
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    r = m - h.sum()
    if r > 0:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
            if r <= 1e-14: break
    elif r < 0:
        for i in range(N):
            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
            if r >= -1e-14: break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1+1e-10:
        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0 / N)

def smooth_val_grad(h, tau, N):
    dx = 2.0 / N
    c = np.correlate(h, 1.0-h, mode='full') * dx
    cmax = c.max()
    exps = np.exp((c - cmax) / tau)
    se = exps.sum()
    p = exps / se
    F = cmax + tau * np.log(se)
    grad = np.zeros(N)
    for r in range(2*N-1):
        s = r - (N-1)  # lag
        pr = p[r]
        if pr < 1e-15:
            continue
        if s == 0:
            grad += pr * dx * (1.0 - 2.0*h)
        elif s > 0:
            d = s
            L = N - d
            # q = sum_{n=d}^{N-1} h[n](1-h[n-d])
            # d/dh_j: (1-h[j-d]) for j>=d, -h[j+d] for j<=N-1-d
            grad[d:] += pr * dx * (1.0 - h[:L])
            grad[:L] -= pr * dx * h[d:]
        else:
            d = -s
            L = N - d
            # q = sum_{n=0}^{N-1-d} h[n](1-h[n+d])
            # d/dh_j: (1-h[j+d]) for j<=N-1-d, -h[j-d] for j>=d
            grad[:L] += pr * dx * (1.0 - h[d:])
            grad[d:] -= pr * dx * h[:L]
    return F, grad, cmax

    h = h0.copy()
    best_h = h0.copy()
    best_sc = score_exact(h0, N)
    for tau in tau_sched:
        alpha = 1.0
        stagnant = 0
        for it in range(max_upd):
            F, g, _ = smooth_val_grad(h, tau, N)
            if not np.all(np.isfinite(g)):
                break
            y = None
            accepted = False
            for bt in range(24):
                trial = project_box_sum(h - alpha * g, m)

                d = trial - h

                Fy, _, _ = smooth_val_grad(trial, tau, N)

                gd = np.dot(g, d)

            # Check convergence

    sc_end = score_exact(h, N)
    if sc_end < best_sc:
        best_sc = sc_end
        best_h = h.copy()
    return best_h, best_sc

def make_fourier(N, rng, n_modes, amp):
    z = np.full(N, 0.5)
    idx = np.arange(N)
    for r in range(1, n_modes+1):
        A = rng.standard_normal()
        B = rng.standard_normal()
        z += (amp/r) * (A*np.cos(2*np.pi*r*idx/N) + B*np.sin(2*np.pi*r*idx/N))
    return z

import numpy as np

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    if r > 0:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
            if r <= 1e-14:
                break
    elif r < 0:
        for i in range(N):
            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1 + 1e-10:
