def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = len(v)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-14:
        if resid > 0:
            for i in range(n):
                add = min(resid, 1.0 - y[i])
                y[i] += add
                resid -= add
                if abs(resid) < 1e-15: break
        else:
            for i in range(n):
                sub = min(-resid, y[i])
                y[i] -= sub
                resid += sub
                if abs(resid) < 1e-15: break
    y = np.clip(y, 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 0:
        if resid > 0 and y[0] < 1.0:
            y[0] += min(resid, 1.0 - y[0])
        elif resid < 0 and y[0] > 0.0:
            y[0] += max(resid, -y[0])
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1.0+1e-12):
        return None
    if abs(np.sum(y) - S) > 1e-10 * max(1.0, float(n)):
        return None
    return y

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def check_feasible(h, n):

def compute_lags_coeffs(h, d, n):
    # Compute A_k, B_k, D_k for Q_k(a) = A_k*a^2 + B_k*a + D_k
    n_lags = 2 * n - 1
    A = np.zeros(n_lags)
    B = np.zeros(n_lags)
    D = np.zeros(n_lags)
    for k in range(-(n-1), n):
        idx = k + n - 1
        i_lo = max(0, k)
        i_hi = min(n-1, n-1+k)

        B[idx] = np.sum(d[i]*(1.0-h[j]) - h[i]*d[j])
        A[idx] = -np.sum(d[i]*d[j])
    return A, B, D

def build_lp(l, u, S, n):
    P = n*(n+1)//2
    nv = n + P + 1
    t_idx = n + P
    # Objective: min t
    c = np.zeros(nv); c[t_idx] = 1.0
    rows_ub = []; cols_ub = []; vals_ub = []; rhs_ub = []
    
    def add_row(coeff_dict, rhs):
        idx = len(rhs_ub)
        for var, coef in coeff_dict.items():
            rows_ub.append(idx); cols_ub.append(var); vals_ub.append(coef)
        rhs_ub.append(rhs)
    
    # McCormick constraints
    for i in range(n):
        for j in range(i, n):
            pidx = n + i*n - i*(i-1)//2 + (j - i)
            li, ui = l[i], u[i]
            lj, uj = l[j], u[j]
            # -z + li*vj + lj*vi <= li*lj
            add_row({pidx: -1.0, j: li, i: lj}, li*lj)
            # -z + ui*vj + uj*vi <= ui*uj
            add_row({pidx: -1.0, j: ui, i: uj}, ui*uj)
            # z - ui*vj - lj*vi <= -ui*lj
            add_row({pidx: 1.0, j: -ui, i: -lj}, -ui*lj)
            # z - li*vj - uj*vi <= -li*uj
            add_row({pidx: 1.0, j: -li, i: -uj}, -li*uj)
    
    # Lag constraints: sum_{i in I_k} v_i - sum_{i in I_k} z[min(i,j),max(i,j)] - t <= 0
    for k in range(-(n-1), n):
        row = {}
        i_lo = max(0, k)
        i_hi = min(n-1, n-1+k)
        for ii in range(i_lo, i_hi+1):
            jj = ii - k
            p = min(ii, jj)
            q = max(ii, jj)
            pidx = n + p*n - p*(p-1)//2 + (q - p)
            row[ii] = row.get(ii, 0.0) + 1.0
            row[pidx] = row.get(pidx, 0.0) - 1.0
        row[t_idx] = row.get(t_idx, 0.0) - 1.0
        add_row(row, 0.0)
    
    # Variable bounds
    lb = np.zeros(nv); ub = np.zeros(nv)
    for i in range(n):
        lb[i] = l[i]; ub[i] = u[i]
    for i in range(n):
        for j in range(i, n):
            pidx = n + i*n - i*(i-1)//2 + (j - i)
            lb[pidx] = l[i]*l[j]; ub[pidx] = u[i]*u[j]
    lb[t_idx] = 0.0; ub[t_idx] = float(n)
    
    A_ub = coo_matrix((vals_ub, (rows_ub, cols_ub)), shape=(len(rhs_ub), nv)).tocsr()
    A_eq_co = []; A_eq_c = []; A_eq_v = []; b_eq = [S]
    for i in range(n):
        A_eq_co.append(0); A_eq_c.append(i); A_eq_v.append(1.0)
    A_eq = coo_matrix((A_eq_v, (A_eq_co, A_eq_c)), shape=(1, nv)).tocsr()
    
    res = linprog(c, A_ub=A_ub, b_ub=np.array(rhs_ub), A_eq=A_eq, b_eq=np.array(b_eq),
                  bounds=list(zip(lb, ub)), method='highs',
                  options={'presolve': True, 'time_limit': 10})
    if res.status not in (0, 2):  # 0=optimal, 2=infeasible... actually 2 is infeasible
        return None
    if res.x is None or len(res.x) != nv or not np.all(np.isfinite(res.x)):
        return None
    return res.x[:n]

def local_lp_search(start, n, S):
    current = project_box_sum(start, S)
    if current is None or not check_feasible(current, n):
        return start, exact_score(start)
    cur_sc = exact_score(current)
    r = 0.20
    max_iter = 8 if n <= 64 else 5
    fail = 0
    for it in range(max_iter):
        l = np.maximum(0.0, current - r)
        u = np.minimum(1.0, current + r)
        y = build_lp(l, u, S, n)
        if y is None:
            r *= 0.5; fail += 1
            continue
        # Validate y is in trust box
        if np.any(y < l - 1e-7) or np.any(y > u + 1e-7):
            y = np.clip(y, l, u)
        y = project_box_sum(y, S)
        if y is None or not check_feasible(y, n):
            r *= 0.5; fail += 1
            continue
        if np.max(np.abs(y - current)) < 1e-12:
            r *= 0.5; fail += 1
            continue
        nh, nsc = line_minimax(current, y, n, S)
        if nsc < cur_sc - 1e-12:
            current = nh; cur_sc = nsc; fail = 0
            r = min(0.25, 1.2 * r)
        else:
            r *= 0.5; fail += 1
        if r < 1e-3 and fail >= 2:
            break
    return current, cur_sc

def gen_harmonic(n, rng):
    f1 = rng.integers(1, max(2, n//2+1))
    f2 = rng.integers(1, max(2, n//2+1))
    p1 = rng.uniform(0, 2*np.pi)
    p2 = rng.uniform(0, 2*np.pi)
    c1 = rng.standard_normal()
    c2 = rng.standard_normal()
    x = np.arange(n)
    v = c1*np.cos(2*np.pi*f1*x/n + p1) + c2*np.cos(2*np.pi*f2*x/n + p2)
    v -= np.mean(v)
    rms = np.sqrt(np.mean(v**2))
    if rms < 1e-12:
        v = rng.standard_normal(n); v -= np.mean(v)
    else:
        v /= rms
    return v

            rr.append(rix); cc.append(pi); vv.append(1.0)
            rr.append(rix); cc.append(j); vv.append(-ui)
            rr.append(rix); cc.append(i); vv.append(-lj)
            # row 4: z - li*vj - uj*vi <= -li*uj
            rix = len(bb); bb.append(-li*uj)
            rr.append(rix); cc.append(pi); vv.append(1.0)
            rr.append(rix); cc.append(j); vv.append(-li)
            rr.append(rix); cc.append(i); vv.append(-uj)
    # Lag constraints
    for k in range(-(n-1), n):
        rix = len(bb); bb.append(0.0)
        ilo = max(0,k); ihi = min(n-1, n-1+k)
        for ii in range(ilo, ihi+1):
            jj = ii - k
            p, q = min(ii,jj), max(ii,jj)
            pi = n + p*n - p*(p-1)//2 + (q-p)
            rr.append(rix); cc.append(ii); vv.append(1.0)
            rr.append(rix); cc.append(pi); vv.append(-1.0)
        rr.append(rix); cc.append(ti); vv.append(-1.0)
    # Bounds
    lb = np.zeros(nv); ub = np.zeros(nv)
    for i in range(n): lb[i], ub[i] = l[i], u[i]
    for i in range(n):
        for j in range(i, n):
            pi = n + i*n - i*(i-1)//2 + (j-i)
            lb[pi], ub[pi] = l[i]*l[j], u[i]*u[j]
    lb[ti], ub[ti] = 0.0, float(n)
    A_ub = coo_matrix((vv, (rr, cc)), shape=(len(bb), nv)).tocsr()
    A_eq = coo_matrix((np.ones(n), ([0]*n, np.arange(n))), shape=(1, nv)).tocsr()
    res = linprog(c, A_ub=A_ub, b_ub=np.array(bb), A_eq=A_eq, b_eq=np.array([S]),
                  bounds=list(zip(lb, ub)), method='highs', options={'time_limit': 8.0})
    if res is None or res.status != 0 or res.x is None or len(res.x) != nv:
        return None
    if not np.all(np.isfinite(res.x)): return None
    return res.x[:n]

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = len(v)
    if n == 0 or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S: lo = mid
        else: hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-14:
        if resid > 0:
            for i in range(n):
                a = min(resid, 1.0 - y[i]); y[i] += a; resid -= a
                if abs(resid) < 1e-15: break
        else:
            for i in range(n):
                a = min(-resid, y[i]); y[i] -= a; resid += a
                if abs(resid) < 1e-15: break
    y = np.clip(y, 0.0, 1.0)
    resid = S - np.sum(y)
    if resid > 0 and y[0] < 1.0: y[0] += min(resid, 1.0 - y[0])
    elif resid < 0 and y[0] > 0.0: y[0] += max(resid, -y[0])
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    if abs(np.sum(y) - S) > 1e-10 * max(1.0, float(n)):
        return None
    return y

def exact_score(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def check_feasible(h, n):
    return (h is not None and h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1.0, float(n)))

def line_minimax(x, y, n, S):
    d = y - x
    if np.max(np.abs(d)) < 1e-14:
        return x.copy(), exact_score(x)
    nl = 2 * n - 1
    A = np.zeros(nl); B = np.zeros(nl); D = np.zeros(nl)
    for k in range(-(n - 1), n):
        idx = k + n - 1
        ilo = max(0, k); ihi = min(n - 1, n - 1 + k)
        if ilo > ihi: continue
        i = np.arange(ilo, ihi + 1); j = i - k
        D[idx] = np.sum(x[i] * (1.0 - x[j]))
        B[idx] = np.sum(d[i] * (1.0 - x[j]) - x[i] * d[j])
        A[idx] = -np.sum(d[i] * d[j])
    act = np.abs(A) + np.abs(B) + np.abs(D) > 1e-15
    Aa, Ba, Da = A[act], B[act], D[act]
    na = len(Aa)
    cands = [0.0, 1.0]
    for i in range(na):
        if abs(Aa[i]) > 1e-14:
            a = -Ba[i] / (2.0 * Aa[i])
            if -1e-10 <= a <= 1.0 + 1e-10: cands.append(np.clip(a, 0.0, 1.0))
    for i in range(na):
        for j2 in range(i + 1, na):
            da, db, dd = Aa[i] - Aa[j2], Ba[i] - Ba[j2], Da[i] - Da[j2]
            if abs(da) < 1e-14:
                if abs(db) > 1e-14:
                    a = -dd / db
                    if -1e-10 <= a <= 1.0 + 1e-10: cands.append(np.clip(a, 0.0, 1.0))
            else:
                disc = db * db - 4.0 * da * dd
                if disc >= -1e-12:
                    sq = np.sqrt(max(disc, 0.0))
                    for s in (1.0, -1.0):
                        a = (-db + s * sq) / (2.0 * da)
                        if -1e-10 <= a <= 1.0 + 1e-10: cands.append(np.clip(a, 0.0, 1.0))
    cands = np.sort(np.unique(np.round(np.array(cands), 10)))
    cands = cands[(cands >= 0.0) & (cands <= 1.0)]
    if len(cands) > 400:
        vals = Aa[:, None] * cands**2 + Ba[:, None] * cands + Da[:, None]
        mx = np.max(vals, axis=0)
        idx = np.argsort(mx)[:100]
        cands = np.sort(np.unique(np.concatenate([cands[idx], [0.0, 1.0]])))
    vals = Aa[:, None] * cands**2 + Ba[:, None] * cands + Da[:, None]
    mx = np.max(vals, axis=0)
    best_i = int(np.argmin(mx))
    check = set([0, len(cands) - 1, best_i])
    for ci in range(len(cands)):
        if abs(cands[ci] - cands[best_i]) < 0.05: check.add(ci)
    best_h, best_sc = x.copy(), exact_score(x)
    for ci in sorted(check):
        h_t = project_box_sum(x + cands[ci] * d, S)
        if h_t is not None and check_feasible(h_t, n):
            sc = exact_score(h_t)
            if sc < best_sc: best_sc, best_h = sc, h_t
    return best_h, best_sc

def build_lp(l, u, S, n):
    P = n * (n + 1) // 2
    nv = n + P + 1
    ti = n + P
    c = np.zeros(nv); c[ti] = 1.0
    rr, cc, vv, bb = [], [], [], []
    for i in range(n):
        for j in range(i, n):
            pi = n + i * n - i * (i - 1) // 2 + (j - i)
            li, ui, lj, uj = l[i], u[i], l[j], u[j]
            rix = len(bb); bb.append(li * lj)
            rr.append(rix); cc.append(pi); vv.append(-1.0)
            rr.append(rix); cc.append(j); vv.append(li)
            rr.append(rix); cc.append(i); vv.append(lj)
            rix = len(bb); bb.append(ui * uj)
            rr.append(rix); cc.append(pi); vv.append(-1.0)
            rr.append(rix); cc.append(j); vv.append(ui)
            rr.append(rix); cc.append(i); vv.append(uj)
            rix = len(bb); bb.append(-ui * lj)
            rr.append(rix); cc.append(pi); vv.append(1.0)
            rr.append(rix); cc.append(j); vv.append(-ui)
            rr.append(rix); cc.append(i); vv.append(-lj)
            rix = len(bb); bb.append(-li * uj)
            rr.append(rix); cc.append(pi); vv.append(1.0)
            rr.append(rix); cc.append(j); vv.append(-li)
            rr.append(rix); cc.append(i); vv.append(-uj)
    for k in range(-(n - 1), n):
        rix = len(bb); bb.append(0.0)
        ilo = max(0, k); ihi = min(n - 1, n - 1 + k)
        for ii in range(ilo, ihi + 1):
            jj = ii - k
            p, q = (ii, jj) if ii <= jj else (jj, ii)
            pi = n + p * n - p * (p - 1) // 2 + (q - p)
            rr.append(rix); cc.append(ii); vv.append(1.0)
            rr.append(rix); cc.append(pi); vv.append(-1.0)
        rr.append(rix); cc.append(ti); vv.append(-1.0)
    lb = np.zeros(nv); ub = np.zeros(nv)
    for i in range(n): lb[i], ub[i] = l[i], u[i]
    for i in range(n):
        for j in range(i, n):
            pi = n + i * n - i * (i - 1) // 2 + (j - i)
            lb[pi], ub[pi] = l[i] * l[j], u[i] * u[j]
    lb[ti], ub[ti] = 0.0, float(n)
    A_ub = coo_matrix((vv, (rr, cc)), shape=(len(bb), nv)).tocsr()
    A_eq = coo_matrix((np.ones(n), ([0] * n, np.arange(n))), shape=(1, nv)).tocsr()
    res = linprog(c, A_ub=A_ub, b_ub=np.array(bb), A_eq=A_eq, b_eq=np.array([S]),
                  bounds=list(zip(lb, ub)), method='highs', options={'time_limit': 8.0})
    if res is None or res.status != 0 or res.x is None or len(res.x) != nv:
        return None
    if not np.all(np.isfinite(res.x)): return None
    return res.x[:n]

def local_lp(start, n, S):
    cur = project_box_sum(start, S)
    if cur is None or not check_feasible(cur, n):
        return start, exact_score(start)
    cs = exact_score(cur)
    r, mx_it, fail = 0.20, (6 if n <= 64 else 4), 0
    for _ in range(mx_it):
        l = np.maximum(0.0, cur - r); u = np.minimum(1.0, cur + r)
        y = build_lp(l, u, S, n)
        if y is None:
            r *= 0.5; fail += 1; continue
        if np.any(y < l - 1e-7) or np.any(y > u + 1e-7):
            y = np.clip(y, l, u)
        y = project_box_sum(y, S)
        if y is None or not check_feasible(y, n):
            r *= 0.5; fail += 1; continue
        if np.max(np.abs(y - cur)) < 1e-12:
            r *= 0.5; fail += 1; continue
        nh, nsc = line_minimax(cur, y, n, S)
        if nsc < cs - 1e-12:
            cur, cs, fail = nh, nsc, 0
            r = min(0.25, 1.2 * r)
        else:
            r *= 0.5; fail += 1
        if r < 1e-3 and fail >= 2: break
    return cur, cs

def gen_harm(n, rng):
    f1 = int(rng.integers(1, max(2, n // 2 + 1)))
    f2 = int(rng.integers(1, max(2, n // 2 + 1)))
    x = np.arange(n)
    v = (rng.standard_normal() * np.cos(2.0 * np.pi * f1 * x / n + rng.uniform(0, 6.28)) +
         rng.standard_normal() * np.cos(2.0 * np.pi * f2 * x / n + rng.uniform(0, 6.28)))
    v -= np.mean(v)
    rms = np.sqrt(np.mean(v ** 2))
    if rms < 1e-12:
        v = rng.standard_normal(n); v -= np.mean(v)
    else:
        v /= rms
    return v
