def harm(n, rng):
    f = rng.integers(1, max(2,n//2+1), 2)

    v -= v.mean()

    ub_n = np.minimum(1.0, x + r)

        for j in range(i, n):
            z = zvar(i, j)
            li, ui = lb_n[i], ub_n[i]
            lj, uj = lb_n[j], ub_n[j]
            # -z + li*vj + lj*vi <= li*lj

    # Lag constraints

        if i0 > i1: continue
        nr = len(bb)
