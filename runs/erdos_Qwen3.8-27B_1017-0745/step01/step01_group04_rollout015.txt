        else:
            try_add(pop, rng.uniform(0, 1, N), N, m)
    while len(pop) < P:
        pop.append(pop[len(pop) % len(pop)].copy())
    return pop[:P]

def de_search(pop, N, m, P, max_gen, stag_limit, max_restarts, F, CR, rng):
    scores = [exact_score(p, N) for p in pop]
    bi = int(np.argmin(scores))
    best_h, best_score = pop[bi].copy(), scores[bi]
    stag, restarts = 0, 0
    for gen in range(max_gen):
        prev_best = best_score
        snap = [p.copy() for p in pop]
        ss = list(scores)
        npop, ns = [p.copy() for p in pop], list(scores)
        for i in range(P):
            others = [j for j in range(P) if j != i]
            r1, r2, r3 = rng.choice(others, size=3, replace=False)
            v = snap[r1] + F * (snap[r2] - snap[r3])
            if not np.all(np.isfinite(v)):
                continue
            mask = rng.random(N) < CR
            mask[rng.integers(N)] = True
            u = np.where(mask, v, snap[i])
            pr = project_box_sum(u, N, m)
            if not is_feasible(pr, N, m):
                continue
            sc = exact_score(pr, N)
            if np.isfinite(sc) and sc < ss[i] - 1e-12:
                npop[i] = pr
                ns[i] = sc
                if sc < best_score - 1e-12:
                    best_h, best_score = pr.copy(), sc
        pop, scores = npop, ns
        if best_score < prev_best - 1e-12:
            stag = 0
        else:
            stag += 1
        if stag >= stag_limit and restarts < max_restarts:
            sigma = 0.14 if restarts == 0 else 0.05
            restarts += 1
            stag = 0
            keep_n = max(2, P // 4)
            ranked = np.argsort(scores)
            new_pop = [best_h.copy()]
            for k in range(1, min(keep_n, P)):
                new_pop.append(pop[ranked[k]].copy())
            fill = P - len(new_pop)
            n_g = (3 * fill) // 4
            n_r = fill - n_g
            for _ in range(n_g):
                new_pop.append(project_box_sum(best_h + sigma * rng.standard_normal(N), N, m))
            for _ in range(n_r):
                new_pop.append(project_box_sum(rng.uniform(0, 1, N), N, m))
            while len(new_pop) < P:
                new_pop.append(project_box_sum(best_h + 0.05 * rng.standard_normal(N), N, m))
            pop = new_pop[:P]
            scores = [exact_score(p, N) for p in pop]
            bi2 = int(np.argmin(scores))
            if scores[bi2] < best_score - 1e-12:
                best_h, best_score = pop[bi2].copy(), scores[bi2]
        if stag >= 2 * stag_limit and restarts >= max_restarts:
            break
    return best_h, best_score

def local_polish(h, N, m, rng, n_rounds=3):
    best_h = h.copy()
    best_sc = exact_score(h, N)
    for _ in range(n_rounds):
        improved = False
        for _ in range(8):
            pert = best_h + 0.01 * rng.standard_normal(N)
            pr = project_box_sum(pert, N, m)
            if not is_feasible(pr, N, m):
                continue
            sc = exact_score(pr, N)
            if np.isfinite(sc) and sc < best_sc - 1e-13:
                best_h, best_sc = pr.copy(), sc
                improved = True
        if not improved:
            break
    return best_h, best_sc

def run(seed=42, **kwargs):
    N = None
    m = None
    raw = globals().get('initial_h_values', None)
    if raw is not None:
        raw = np.asarray(raw, dtype=np.float64)
        if raw.ndim == 1 and raw.size > 1 and np.all(np.isfinite(raw)):
            N = raw.size
            m = N / 2.0
    if N is None:
        N = kwargs.get('n_points', 70)
        if not isinstance(N, (int, np.integer)) or isinstance(N, bool) or not (1 <= N <= 999):
            N = 70
        m = N / 2.0
    if N == 1:
        h = np.array([0.5])
        return h, exact_score(h, 1), 1
    if raw is not None and np.all(np.isfinite(raw)):
        if raw.min() >= -1e-12 and raw.max() <= 1.0 + 1e-12 and abs(raw.sum() - m) < 1e-8:
            baseline = np.clip(raw.copy(), 0.0, 1.0)
        else:
            baseline = project_box_sum(raw, N, m)
    else:
        baseline = np.full(N, 0.5)
    baseline = project_box_sum(baseline, N, m)
    base_score = exact_score(baseline, N)
    if not np.isfinite(base_score):
        baseline = np.full(N, 0.5)
        base_score = exact_score(baseline, N)
    if N > 999:
        return baseline, float(base_score), int(N)
    if N <= 96:
        P, max_gen, stag_limit, max_restarts = 48, 120, 24, 2
    elif N <= 192:
        P, max_gen, stag_limit, max_restarts = 32, 80, 20, 1
    elif N <= 384:
        P, max_gen, stag_limit, max_restarts = 20, 40, 16, 1
    else:
        P, max_gen, stag_limit, max_restarts = 12, 20, 10, 0
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    P = max(P, 4)
    best_h, best_score = baseline, base_score
    try:
        pop = build_population(baseline, N, m, P, rng)
        res_h, res_sc = de_search(pop, N, m, P, max_gen, stag_limit, max_restarts, 0.7, 0.85, rng)
        if is_feasible(res_h, N, m) and np.isfinite(res_sc) and res_sc < best_score - 1e-12:
            best_h, best_score = res_h, res_sc
        # Local polish
        pol_h, pol_sc = local_polish(best_h, N, m, rng, 3)
        if is_feasible(pol_h, N, m) and np.isfinite(pol_sc) and pol_sc < best_score - 1e-12:
            best_h, best_score = pol_h, pol_sc
        # Second DE pass from polished result
        pop2 = build_population(best_h, N, m, P, rng)
        res2_h, res2_sc = de_search(pop2, N, m, P, max_gen // 2, stag_limit, 1, 0.6, 0.8, rng)
        if is_feasible(res2_h, N, m) and np.isfinite(res2_sc) and res2_sc < best_score - 1e-12:
            best_h, best_score = res2_h, res2_sc
        pol2_h, pol2_sc = local_polish(best_h, N, m, rng, 3)
        if is_feasible(pol2_h, N, m) and np.isfinite(pol2_sc) and pol2_sc < best_score - 1e-12:
            best_h, best_score = pol2_h, pol2_sc
    except Exception:
        pass
    if not is_feasible(best_h, N, m) or not np.isfinite(best_score):
        best_h, best_score = baseline, base_score
    best_h = project_box_sum(best_h, N, m)
    c5 = exact_score(best_h, N)
    if not np.isfinite(c5) or c5 > base_score:
        best_h = baseline
        c5 = base_score
    return best_h, float(c5), int(N)
