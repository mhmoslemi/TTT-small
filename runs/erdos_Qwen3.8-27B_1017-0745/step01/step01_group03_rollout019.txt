def make_Q(h, n):
    return np.asarray(np.correlate(h, 1.0 - h, mode='full'), dtype=np.float64)

def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    nn = len(v)
    s = np.sort(v)[::-1]
    csum = np.cumsum(s)
    idx = np.arange(1, nn + 1)
    rho = np.where(s - (csum - 1.0) / idx > 0)[0]
    if len(rho) == 0:
        return np.ones(nn) / nn
    r = rho[-1]
    theta = (csum[r] - 1.0) / (r + 1)
    w = np.maximum(v - theta, 0.0)
    sw = w.sum()
    if sw < 1e-15:
        return np.ones(nn) / nn
    return w / sw

def dual_eval(v, x, Q, G, S):
    n = len(x)
    z = G.T @ v
    y = project_box_sum(x - z * 0.5, S)
    if y is None:
        return None
    d = y - x
    p = G @ d
    vals = Q + p
    return y, d, vals

def solve_surrogate(x, Q, G, S, w0, n):
    L = len(Q)
    Ldual = max(1e-8, 0.5001 * float(np.sum(G * G)))
    if w0 is not None and np.all(np.isfinite(w0)) and len(w0) == L:
        w = project_simplex(w0)
    else:
        w = np.zeros(L); w[int(np.argmax(Q))] = 1.0
    v = w.copy(); w_prev = w.copy(); t = 1.0
    Imax = max(100, min(320, int(np.ceil(14000.0 / n))))
    best_model = float(np.max(Q))
    best_y = x.copy()
    gc = max(5, Imax // 25)
    for it in range(Imax):
        res = dual_eval(v, x, Q, G, S)
        if res is None: break
        y_c, d_c, vals_c = res
        m_c = float(np.dot(d_c, d_c)) + float(np.max(vals_c))
        if m_c < best_model:
            best_model = m_c; best_y = y_c.copy()
        w_new = project_simplex(v + vals_c / Ldual)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) * 0.5
        v = w_new + ((t - 1.0) / t_new) * (w_new - w_prev)
        w_prev = w; w = w_new; t = t_new
        if (it + 1) % gc == 0:
            res2 = dual_eval(w, x, Q, G, S)
            if res2 is None: break
            _, d2, vals2 = res2
            gap = float(np.max(vals2)) - float(np.dot(w, vals2))
            if gap <= 1e-7 * max(1.0, float(np.max(np.abs(Q)))): break
    res_f = dual_eval(w, x, Q, G, S)
    if res_f is not None:
        y_f, d_f, vals_f = res_f
        m_f = float(np.dot(d_f, d_f)) + float(np.max(vals_f))
        if m_f < best_model:
            best_model = m_f; best_y = y_f.copy()
    return best_y, w

def do_line_search(x, d, Q, G, S, n):
    if float(np.max(np.abs(d))) < 1e-10:
        return x
    p = G @ d
    dd = float(np.dot(d, d))
    lo_a, hi_a = 0.0, 1.0
    for _ in range(36):
        m1 = lo_a + (hi_a - lo_a) * (3.0 - np.sqrt(5.0)) * 0.5
        m2 = lo_a + (hi_a - lo_a) * (np.sqrt(5.0) - 1.0) * 0.5
        f1 = m1 * m1 * dd + float(np.max(Q + m1 * p))
        f2 = m2 * m2 * dd + float(np.max(Q + m2 * p))
        if f1 < f2: hi_a = m2
        else: lo_a = m1
    alpha_m = (lo_a + hi_a) * 0.5
    cands = sorted(set(round(a, 10) for a in [alpha_m, 1.0, 0.5, 0.25, 0.125]))
    best_h = x.copy()
    best_sc = exact_score(x, n)
    for a in cands:
        h_c = project_box_sum(x + a * d, S)
        if h_c is None: continue
        sc = exact_score(h_c, n)
        if sc < best_sc - 1e-14:
            best_sc = sc; best_h = h_c
    return best_h

def mm_local(start, n, S):
    h = project_box_sum(start, S)
    if h is None: return None
    Omax = max(4, min(16, int(np.ceil(700.0 / n))))
    w_warm = None
    for _ in range(Omax):
        Q = make_Q(h, n)
        G = build_G(h, n)
        y_new, w_warm = solve_surrogate(h, Q, G, S, w_warm, n)
        d = y_new - h
        h2 = do_line_search(h, d, Q, G, S, n)
        sc2 = exact_score(h2, n)
        sc1 = exact_score(h, n)
        if sc2 < sc1 - 1e-13:
            h = h2
            if float(np.max(np.abs(d))) < 1e-8: break
        else:
            break
    return h

def make_starts(n, S, rng):
    starts = []
    try:
        pv = np.array(initial_h_values, dtype=np.float64)
        if pv.ndim == 1 and 2 <= len(pv) <= 999 and np.all(np.isfinite(pv)):
            if np.all(pv >= -1e-9) and np.all(pv <= 1.0 + 1e-9) and abs(np.sum(pv) - len(pv) / 2.0) < 1e-6 * max(1, len(pv)):
                pv = np.clip(pv, 0.0, 1.0)
                p = project_box_sum(pv, S)
                if p is not None:
                    starts.append(p)
                    starts.append(p[::-1].copy())
                    pc = project_box_sum(1.0 - p, S)
                    if pc is not None: starts.append(pc)
                    for sig in [0.03, 0.08, 0.15]:
                        c = project_box_sum(p + sig * rng.standard_normal(n), S)
                        if c is not None: starts.append(c)
                    for m in [1, 2]:
                        ph = rng.random() * 6.2832
                        wave = 0.05 * np.cos(2.0 * np.pi * m * np.arange(n) / n + ph)
                        c = project_box_sum(p + wave, S)
                        if c is not None: starts.append(c)
    except Exception:
        pass
    if not starts:
        c = project_box_sum(np.full(n, 0.5), S)
        if c is not None: starts.append(c)
        c = project_box_sum(rng.random(n), S)
        if c is not None: starts.append(c)
        for m in [1, 2, 3]:
            ph = rng.random() * 6.2832
            v = 0.5 + 0.35 * np.cos(2.0 * np.pi * m * np.arange(n) / n + ph)
            c = project_box_sum(v, S)
            if c is not None: starts.append(c)
    dedup = []
    for s in starts:
        if s is None: continue
        dup = False
        for dd in dedup:
            if np.max(np.abs(s - dd)) < 1e-10: dup = True; break
        if not dup: dedup.append(s)
    return dedup

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    n = 45
    S = n / 2.0
    fallback = project_box_sum(np.full(n, 0.5), S)
    try:
        pv = np.array(initial_h_values, dtype=np.float64)
        if pv.ndim == 1 and 2 <= len(pv) <= 999 and np.all(np.isfinite(pv)):
            if np.all(pv >= -1e-9) and np.all(pv <= 1.0 + 1e-9) and abs(np.sum(pv) - len(pv) / 2.0) < 1e-6 * max(1, len(pv)):
                n = len(pv); S = n / 2.0
                fb = project_box_sum(np.clip(pv, 0.0, 1.0), S)
                if fb is not None: fallback = fb
    except Exception:
        pass
    starts = make_starts(n, S, rng)
    best_h = fallback.copy()
    best_sc = exact_score(fallback, n)
    for s in starts:
        if s is None: continue
        try:
            result = mm_local(s, n, S)
        except Exception:
            result = None
        if result is not None:
            sc = exact_score(result, n)
            if sc < best_sc - 1e-13:
                best_sc = sc; best_h = result
    for _ in range(3):
        pert = project_box_sum(best_h + 0.06 * rng.standard_normal(n), S)
        if pert is None: continue
        try:
            result = mm_local(pert, n, S)
        except Exception:
            result = None
        if result is not None:
            sc = exact_score(result, n)
            if sc < best_sc - 1e-13:
                best_sc = sc; best_h = result
    final = project_box_sum(best_h, S)
    if final is None: final = fallback
    final_sc = exact_score(final, n)
    fb_sc = exact_score(fallback, n)
    if final_sc > fb_sc + 1e-12:
        final = fallback; final_sc = fb_sc
    return (np.array(final, dtype=np.float64), float(final_sc), int(n))
