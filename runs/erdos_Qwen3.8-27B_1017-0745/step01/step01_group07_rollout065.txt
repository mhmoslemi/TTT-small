```python
import numpy as np

def project_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    n = len(z)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > n:
        return None
    if np.max(np.abs(z)) > 1e12:
        return None
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z))
    for _ in range(80):
        mid = lo + 0.5 * (hi - lo)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = lo + 0.5 * (hi - lo)
    y = np.clip(z - lam, 0.0, 1.0)
    r = S - np.sum(y)
    for i in range(n):
        if r > 0:
            a = min(r, 1.0 - y[i])
            y[i] += a
            r -= a
        elif r < 0:
            a = min(-r, y[i])
            y[i] -= a
            r += a
        if abs(r) < 1e-14:
            break
    if abs(np.sum(y) - S) > 1e-10 or not np.all(np.isfinite(y)):
        return None
    return np.clip(y, 0.0, 1.0)

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-8)

def exact_score(h, n):
    try:
        vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if not np.all(np.isfinite(vals)):
            return None
        return float(np.max(vals))
    except Exception:
        return None

def line_search(h, n, dx, p, r, cur_score):
    L = max(-h[p], h[r] - 1.0)
    U = min(1.0 - h[p], h[r])
    if U - L < 1e-12:
        return None
    h2 = h.copy()
    best_t = 0.0
    best_val = cur_score
    # 17-point grid
    ts = np.linspace(L, U, 17)
    ts = np.insert(ts, len(ts)//2, 0.0)
    for t in ts:
        if t < L - 1e-14 or t > U + 1e-14:
            continue
        h2[p] = h[p] + t
        h2[r] = h[r] - t
        sc = exact_score(h2, n)
        if sc is not None and sc < best_val - 1e-13:
            best_val = sc
            best_t = t
    h2[p] = h[p]
    h2[r] = h[r]
    # Refine
    radius = (U - L) / 16.0
    for _ in range(4):
        lo_r = max(L, best_t - radius)
        hi_r = min(U, best_t + radius)
        ts = np.linspace(lo_r, hi_r, 5)
        ts = np.insert(ts, 2, best_t)
        for t in ts:
            if t < L - 1e-14 or t > U + 1e-14:
                continue
            h2[p] = h[p] + t
            h2[r] = h[r] - t
            sc = exact_score(h2, n)
            if sc is not None and sc < best_val - 1e-13:
                best_val = sc
                best_t = t
        h2[p] = h[p]
        h2[r] = h[r]
        radius /= 4.0
    if best_val >= cur_score - 1e-12:
        return None
    h_new = h.copy()
    h_new[p] += best_t
    h_new[r] -= best_t
    if not is_feasible(h_new, n):
        h_new = project_simplex(h_new, n / 2.0)
        if h_new is None or not is_feasible(h_new, n):
            return None
        sc = exact_score(h_new, n)
        if sc is None or sc >= cur_score - 1e-12:
            return None
        return h_new, sc
    return h_new, best_val
```

```python
        pool = picked if len(picked) >= 10 else extra[:P]
        rng.shuffle(pool)
        for (p, r) in pool[:P]:
            res = line_search(h, n, p, r, cur_sc)
            if res is not None:
                h, cur_sc = res
                improved = True
        if not improved:
            break
    return h, cur_sc

def run(seed=42, **kwargs):
    try:
        seed = int(seed) if seed is not None else 42
    except (TypeError, ValueError):
        seed = 42
    rng = np.random.default_rng(seed % (2 ** 32))
    n = None
    parent = None
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        try:
            arr = np.asarray(iv, dtype=np.float64)
            if arr.ndim == 1 and 2 <= arr.size <= 999:
                n = arr.size
                S = n * 0.5
                if np.all(np.isfinite(arr)) and np.max(np.abs(arr)) <= 1e12:
                    if is_feasible(arr, n):
                        parent = arr.copy()
                    else:
                        pr = project_simplex(arr, S)
                        if pr is not None and is_feasible(pr, n):
                            parent = pr
        except Exception:
            pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 80))
        except (TypeError, ValueError):
            n = 80
        if n < 8 or n > 999:
            n = 80
    S = n * 0.5
    best_h = np.full(n, 0.5)
    best_sc = exact_score(best_h, n)
    if best_sc is None:
        best_sc = 0.5
    if parent is not None:
        ps = exact_score(parent, n)
        if ps is not None and ps < best_sc:
            best_sc, best_h = ps, parent.copy()
    starts = build_starts(n, S, parent, rng)
    for s in starts:
        sc = exact_score(s, n)
        if sc is not None and sc < best_sc - 1e-12:
            best_sc, best_h = sc, s.copy()
    E = 5
    K = 500
    for s in starts:
        h = s.copy()
        cur_sc = exact_score(h, n)
        if cur_sc is None:
            continue
        for e in range(E):
            T = max(3e-5, 4e-3 * (0.35 ** e))
            acc = 0
            for _ in range(K):
                p = int(rng.integers(0, n))
                r = int(rng.integers(0, n))
                if p == r:
                    continue
                L = max(-h[p], h[r] - 1.0)
                U = min(1.0 - h[p], h[r])
                if U - L < 1e-11:
                    continue
                t = rng.uniform(L, U)
                hn = h.copy()
                hn[p] += t
                hn[r] -= t
                if not is_feasible(hn, n):
                    hn = project_simplex(hn, S)
                    if hn is None:
                        continue
                sc = exact_score(hn, n)
                if sc is None:
                    continue
                delta = sc - cur_sc
                if delta <= 0:
                    h, cur_sc = hn, sc
                    acc += 1
                elif delta / T < 700:
                    if np.log(rng.uniform() + 1e-300) > -delta / T:
                        h, cur_sc = hn, sc
                        acc += 1
            h, cur_sc = polish(h, n, rng, n_sweeps=1)
            if cur_sc < best_sc - 1e-12:
                best_sc, best_h = cur_sc, h.copy()
    best_h, best_sc = polish(best_h, n, rng, n_sweeps=4)
    if parent is not None:
        ps = exact_score(parent, n)
        if ps is not None and ps < best_sc:
            best_sc, best_h = ps, parent.copy()
    if not is_feasible(best_h, n):
        best_h = project_simplex(best_h, S)
        if best_h is None:
            best_h = np.full(n, 0.5)
        best_sc = exact_score(best_h, n)
        if best_sc is None:
            best_sc = 0.5
    return best_h, best_sc, n
```
