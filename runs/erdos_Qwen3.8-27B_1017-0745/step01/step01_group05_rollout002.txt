def project_capped(y, n):
    S = n / 2.0
    if not np.all(np.isfinite(y)) or y.shape != (n,):
        return np.full(n, 0.5)
    lo = float(y.min()) - 1.0
    hi = float(y.max())

        else:

def check_feasible(h, n):

            and h.min() >= -1e-12 and h.max() <= 1.0 + 1e-12
            and abs(h.sum() - n / 2.0) <= 1e-9)

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

                freqs.append(int(round(0.5 * (lo_f + hi_f))))
            else:

                    f = lo_f + idx * (hi_f - lo_f) / (r - 1)
                    j = int(round(f))

    freqs = freqs[:d]

        return None

    return None

def reflect_array(u):
