    p = np.exp(exponents)

        return None, None
    p = p / ps
    F = qmax + tau * math.log(ps)
    # Gradient
    # A[k] = sum_j p[j-k+n-1] * b[j]  (valid indices)
    # A = conv(p[::-1], b, 'full')[2n-2 : 3n-2][::-1]
