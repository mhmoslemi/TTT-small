import numpy as np
from math import exp, log

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0 or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-14:
        if resid > 0:
            order = np.argsort(1.0 - y)
            for i in order:
                add = min(resid, 1.0 - y[i])
                y[i] += add
                resid -= add
                if abs(resid) < 1e-15:
                    break
        else:
            order = np.argsort(y)
            for i in order:
                sub = min(-resid, y[i])
                y[i] -= sub
                resid += sub
                if abs(resid) < 1e-15:
                    break
    if not np.all(np.isfinite(y)):
        return None
    y = np.clip(y, 0.0, 1.0)
    if abs(np.sum(y) - S) > 1e-10:
        return None
    return y

def check_feasible(h, n):
    if h is None or h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10:
        return False
    return True

def exact_score(h):
    n = len(h)
    if not check_feasible(h, n):
        return float('inf')
    lags = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(lags) * 2.0 / n)

def pair_delta_lags(h, p, q, a, n):
    delta = np.zeros(2 * n - 1)
    pq = p - q
    qp = q - p
    for k in range(-(n - 1), n):
        d = 0.0
        if 0 <= p - k < n:
            d += a * (1.0 - h[p - k])
        if 0 <= q - k < n:
            d -= a * (1.0 - h[q - k])
        if 0 <= p + k < n:
            d -= a * h[p + k]
        if 0 <= q + k < n:
            d += a * h[q + k]
        if k == 0:
            d -= 2.0 * a * a
        elif k == pq:
            d += a * a
        elif k == qp:
            d += a * a
        delta[k + n - 1] = d
    return delta

    C = float(np.max(Q) * 2.0 / n)
    best_h = h.copy()
    best_C = C
    
    stages = 6
    temps = [min(0.03, 0.003 * (45.0 / n)) * (0.35 ** s) for s in range(stages)]
    radii = [0.20, 0.12, 0.07, 0.04, 0.02, 0.008]
    moves_per_stage = [moves_per_start // stages] * stages
    moves_per_stage[-1] += moves_per_start - sum(moves_per_stage)
    
    for s in range(stages):
        T = temps[s]
        rho = radii[s]
        n_accept = 0
        for m in range(moves_per_stage[s]):
            p = rng.integers(0, n)
            q = rng.integers(0, n)
            if p == q:

            a = rho * (2.0 * rng.random() - 1.0)
            a_lo = max(-h[p], h[q] - 1.0)
            a_hi = min(1.0 - h[p], h[q])
            if a < a_lo or a > a_hi:

                h[q] -= a
                Q = Q_trial
                C = C_trial
                n_accept += 1
                if n_accept % 64 == 0:

                    C = float(np.max(Q) * 2.0 / n)
                    if C < best_C - 1e-14:

                        best_h = h.copy()
        # End of stage
