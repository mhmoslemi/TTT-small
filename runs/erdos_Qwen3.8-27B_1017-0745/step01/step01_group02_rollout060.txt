    hi = min(n, n - s)
    q = 0.0
    g = np.zeros(n)

        q += h[i] * (1.0 - h[j])
        g[i] += dx * (1.0 - h[j])
        g[j] -= dx * h[i]

def separation(h, d, n):

    best_s = 0
    dx = 2.0 / n

            best_s = s
    return best, best_s

    t = z[-1]

    g[:-1] = L * (y - x)
    g[-1] = 1.0

        h = best_h.copy()
        c_curr = best_c
        if c_curr < c_curr - 1e-10:  # never true, just for clarity

    best_c5 = exact_score(parent, n)
    mx = max(8, min(40, 1800 // n))

    starts = _build_starts(parent, n, rng)
    seen = []
    for h0 in starts:
        if not check_feasible(h0, n):
            continue
        dup = False
        for s in seen:
            if float(np.max(np.abs(h0 - s))) < 1e-10:
                dup = True
                break
        if dup:
            continue
        seen.append(h0.copy())
        try:
            h_loc, c_loc = local_mm(h0, n, mx)
            if c_loc < best_c5:
                best_h = h_loc.copy()
                best_c5 = c_loc
        except Exception:
            pass

    for sd in [0.04, 0.12, 0.28]:
        h0 = project_box_sum(best_h + sd * rng.standard_normal(n), 0.5 * n)
        if h0 is None or not check_feasible(h0, n):
            continue
        dup = False
        for s in seen:
            if float(np.max(np.abs(h0 - s))) < 1e-10:
                dup = True
                break
        if dup:
            continue
        seen.append(h0.copy())
        try:
            h_loc, c_loc = local_mm(h0, n, max(4, mx // 2))
            if c_loc < best_c5:
                best_h = h_loc.copy()
                best_c5 = c_loc
        except Exception:
            pass

    for sd in [0.08, 0.20]:
        h0 = project_box_sum(best_h + sd * rng.standard_normal(n), 0.5 * n)
        if h0 is None:
            continue
        try:
            h_loc, c_loc = local_mm(h0, n, max(4, mx // 2))
            if c_loc < best_c5:
                best_h = h_loc.copy()
                best_c5 = c_loc
        except Exception:
            pass

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, 0.5 * n)
    if best_h is None or not check_feasible(best_h, n):
        best_h = parent.copy()
        best_c5 = exact_score(parent, n)
    else:
        c5 = exact_score(best_h, n)
        if c5 < best_c5:
            best_c5 = c5
    pc = exact_score(parent, n)
    if pc < best_c5:
        best_h = parent.copy()
        best_c5 = pc
    best_h = np.asarray(best_h, dtype=np.float64)
    best_c5 = exact_score(best_h, n)
    return (best_h, best_c5, n)
