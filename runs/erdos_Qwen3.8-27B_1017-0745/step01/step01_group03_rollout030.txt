import numpy as np
from math import sqrt, ceil

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0 or not np.all(np.isfinite(v)):
        return None
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v))
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - np.sum(y)
    if abs(r) > 1e-14:
        if r > 0:
            for i in range(n):
                a = min(r, 1.0 - y[i])
                y[i] += a; r -= a
                if abs(r) < 1e-15: break
        else:
            for i in range(n):
                s = min(-r, y[i])
                y[i] -= s; r += s
                if abs(r) < 1e-15: break
    np.clip(y, 0.0, 1.0, out=y)
    r2 = S - np.sum(y)
    if abs(r2) > 0:
        if r2 > 0 and y[0] < 1.0:
            y[0] += min(r2, 1.0 - y[0])
        elif r2 < 0 and y[0] > 0.0:
            y[0] += max(r2, -y[0])
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    if abs(np.sum(y) - S) > 5e-12 * max(1.0, float(n)):
        return None
    return y

def is_feasible(h, n, S):
    if h is None or h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - S) <= 5e-12 * max(1.0, float(n))

def exact_score(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    if len(c) != 2 * n - 1 or not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c)) * dx

def build_G(h, n):
    L = 2 * n - 1
    G = np.zeros((L, n), dtype=np.float64)
    for k in range(-(n - 1), n):
        row = k + n - 1
        for i in range(max(0, k), min(n, n + k)):
            j = i - k
            G[row, i] += (1.0 - h[j])
            G[row, j] += (-h[i])
    return G

def compute_Q(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_simplex(v):
    nv = len(v)
    if not np.all(np.isfinite(v)):
        return None
    sv = np.sort(v)[::-1]
    csum = np.cumsum(sv)
    idx = np.arange(1, nv + 1)
    valid = sv * idx > (csum - 1.0)
    if not np.any(valid):
        return np.ones(nv) / nv
    r = int(np.max(np.where(valid)))
    theta = (csum[r] - 1.0) / (r + 1)
    w = np.maximum(v - theta, 0.0)
    rsum = 1.0 - np.sum(w)
    if abs(rsum) > 1e-15:
        w[int(np.argmax(w))] += rsum
        np.clip(w, 0.0, None, out=w)
    if np.any(w < -1e-14) or abs(np.sum(w) - 1.0) > 1e-10:
        w = np.maximum(w, 0.0)
        if np.sum(w) > 1e-14:
            w /= np.sum(w)
        else:
            w = np.ones(nv) / nv
    return w

def solve_surrogate(x, Q, G, S, n, warm):

    if warm is not None and len(warm) == L and np.all(np.isfinite(warm)):
        w = project_simplex(warm)
        if w is None:
            w = np.zeros(L); w[int(np.argmax(Q))] = 1.0
    else:
        w = np.zeros(L); w[int(np.argmax(Q))] = 1.0
    Ldual = max(1e-8, 0.5001 * float(np.sum(G * G)))
    Imax = max(80, min(200, int(ceil(8000.0 / n))))
    v = w.copy()
    t = 1.0
    best_y = x.copy()
    best_model = float(np.max(Q))
    best_w = w.copy()
    for it in range(Imax):
        st = project_simplex(v)
        if st is None:
            st = best_w
        z = G.T @ st
        y_c = project_box_sum(x - 0.5 * z, S)
        if y_c is None:
            break
        d = y_c - x
        p = G @ d
        vals = Q + p
        if not np.all(np.isfinite(vals)):
            break
        model = float(d @ d) + float(np.max(vals))
        if model < best_model:
            best_model = model
            best_y = y_c.copy()
            best_w = st.copy()
        grad_phi = Q + p
        if not np.all(np.isfinite(grad_phi)):
            break
        w_new = project_simplex(v + grad_phi / Ldual)
        if w_new is None:
            w_new = best_w
        t_new = 0.5 * (1.0 + sqrt(1.0 + 4.0 * t * t))
        v = w_new + ((t - 1.0) / t_new) * (w_new - v)
        t = t_new

            if model_f < best_model:
                best_model = model_f
                best_y = y_f.copy()
    return best_y, best_w
