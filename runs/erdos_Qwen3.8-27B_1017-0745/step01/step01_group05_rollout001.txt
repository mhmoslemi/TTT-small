import numpy as np
import math

def project_capped(y, n):
    S = n / 2.0
    if not np.all(np.isfinite(y)) or len(y) != n:
        return np.full(n, 0.5)
    lo = float(y.min()) - 1.0
    hi = float(y.max())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        c = np.clip(y - mid, 0.0, 1.0)
        if c.sum() > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(5):
        r = S - h.sum()
        if abs(r) < 1e-14:
            break
        if r > 0:
            sl = 1.0 - h
            if sl.sum() < 1e-15:
                break
            h += r * sl / sl.sum()
        else:
            if h.sum() < 1e-15:
                break
            h += r * h / h.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(h.sum() - n / 2.0) <= 1e-9)

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(np.max(corr))

def make_basis(n):
    d = min(n - 1, 20)
    if d < 2:
        d = max(1, n - 1)
    q = min(d, 6)
    freqs = list(range(1, q + 1))
    r = d - q
    if r > 0:
        lo_f, hi_f = q + 1, n - 1
        if r == 1:
            freqs.append(int(round(0.5 * (lo_f + hi_f))))
        else:
            pts = np.linspace(lo_f, hi_f, r)
            for p in pts:
                freqs.append(int(round(p)))
            for i in range(len(freqs) - 1, -1, -1):
                if freqs[i] in freqs[:i] or freqs[i] < 1 or freqs[i] > n - 1:
                    freqs.pop(i)
    d = len(freqs)
    cols = np.zeros((n, d))
    for j in range(d):
        i_arr = np.arange(n, dtype=np.float64)
        cols[:, j] = np.cos(np.math.pi * freqs[j] * (i_arr + 0.5) / n)
    cols -= cols.mean(axis=0, keepdims=True)
    Q, _ = np.linalg.qr(cols)
    if not np.all(np.isfinite(Q)):
        return Q if Q.shape[1] > 0 else np.zeros((n, 1))
    return Q

def decode_candidate(anchor, B, sigma, u, n):
    d = B.shape[1]
    scale = sigma * math.sqrt(3.0 * n / d)
    raw = anchor + scale * (B @ u)
    if not np.all(np.isfinite(raw)):
        return None
    h = project_capped(raw, n)
    if check_feasible(h, n):
        return h
    return None

def reflect_unit(x):
    t = np.abs(x)
    return 2.0 * ((t + 1.0) % 2.0) - 1.0 if np.isscalar(t) else 2.0 * ((t + 1.0) % 2.0) - 1.0

def reflect_array(u):
    t = (u + 1.0) % 4.0
    out = np.where(t <= 2.0, t - 1.0, 3.0 - t)
    return out

def evolve_stage(anchor, B, sigma, P, G, rng, n):
    d = B.shape[1]

        h = decode_candidate(anchor, B, sigma, pop[i], n)
        if h is not None:
            s = exact_score(h, n)
            if s < float('inf'):
                scores[i] = s
                decoded[i] = h
    best_i = int(np.argmin(scores))
    best_h = decoded[best_i].copy() if decoded[best_i] is not None else anchor.copy()
    best_s = scores[best_i]

                best_h = decoded[bi2].copy()

            restarted = True
    return best_h, best_s

def run(seed=42, **kwargs):

    parent = globals().get('initial_h_values', None)
