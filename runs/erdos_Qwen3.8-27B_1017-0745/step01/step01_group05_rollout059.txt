def build_gradient_matrix(h, n):
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for k in range(-(n-1), n):
        row = k + n - 1
        # First term: (1-h[m-k]) for valid m
        lo1 = max(0, k)
        hi1 = min(n, n + k)

        hi2 = min(n, n - k)
        if hi2 > lo2:
            G[row, lo2:hi2] -= dx * h[lo2+k:hi2+k]
    return G

s.t. y ∈ P

≤ ||G||_2 ||p1-p2|| / L

So ||∇D(p1) - ∇D(p2)|| ≤ ||G||_2² / L

A safe upper bound is ||G||_F² / L.

import numpy as np
import math

def project_capped(y, n):
    S = n / 2.0
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return np.full(n, 0.5), False
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    lam = np.clip(np.mean(y) - 0.5, lo, hi)
    for _ in range(80):
        c = np.clip(y - lam, 0.0, 1.0)
        r = c.sum() - S
        if abs(r) <= 1e-14:
            break
        if r > 0:
            hi = lam
        else:
            lo = lam
        a = np.sum((y - lam > 0) & (y - lam < 1))
        if a > 0:
            lam_new = lam + r / a
        else:
            lam_new = 0.5 * (lo + hi)
        if lo < lam_new < hi and np.isfinite(lam_new):
            lam = lam_new
        else:
            lam = 0.5 * (lo + hi)
    c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(5):
        r = S - c.sum()
        if abs(r) < 1e-14:
            break
        if r > 0:
            mask = c < 1.0 - 1e-12
        else:
            mask = c > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        c[mask] += r / cnt
        np.clip(c, 0.0, 1.0, out=c)
    ok = (np.all(c >= -1e-12) and np.all(c <= 1.0 + 1e-12) and abs(c.sum() - S) <= 1e-10)
    if not ok:
        return np.full(n, 0.5), False
    np.clip(c, 0.0, 1.0, out=c)
    return c, True

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(h.sum() - n / 2.0) <= 1e-10)

def lag_scores(h, n):
    dx = 2.0 / n
    qs = np.correlate(h, 1.0 - h, mode='full') * dx
    return qs

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    qs = lag_scores(h, n)
    if not np.all(np.isfinite(qs)):
        return float('inf')
    return float(np.max(qs))

def build_G(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        row = k + n - 1
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if hi1 > lo1:
            G[row, lo1:hi1] += dx * (1.0 - h[lo1 - k:hi1 - k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if hi2 > lo2:
            G[row, lo2:hi2] -= dx * h[lo2 + k:hi2 + k]
    return G

def project_simplex(z):
    n_s = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(n_s, 1.0 / n_s)
    sv = np.sort(z)[::-1]
    sv = sv.copy()
    cs = np.cumsum(sv)
    rho = np.where(sv - (cs - 1.0) / np.arange(1, n_s + 1) > 0)[0]
    if len(rho) == 0:
        return np.full(n_s, 1.0 / n_s)
    r = rho[-1]
    theta = (cs[r] - 1.0) / (r + 1)
    out = np.maximum(z - theta, 0.0)
    # fix roundoff
    d = out.sum() - 1.0
    if abs(d) > 1e-15:
        pos = out > 0
        out[pos] -= d / pos.sum()
        np.clip(out, 0.0, None, out=out)
    return out

def dual_state(p, h, q, G, n, L):
    gmix = G.T @ p
    y, ok = project_capped(h - gmix / L, n)
    if not ok:
        return None
    d = y - h
    u = q + G @ d
    D = p.dot(u) + 0.5 * L * d.dot(d)
    gap = float(np.max(u) - p.dot(u))
    if not np.all(np.isfinite(u)) or not np.isfinite(D):
        return None
    return d, u, D, gap

def solve_majorant(h, q, G, n, max_inner):
    L = 4.0 / n  # 2*dx = 2*(2/n) = 4/n
    Ldual = max(1e-12, 1.5 * (G * G).sum() / L)
    m = len(q)
    qmax = float(np.max(q))
    tol = max(1e-12, 1e-8 * max(1.0, abs(qmax)))
    idx = np.where(q >= qmax - tol)[0]
    if len(idx) == 0:
        p = np.zeros(m)
        p[np.argmax(q)] = 1.0
    else:
        p = np.zeros(m)
        p[idx] = 1.0 / len(idx)
    best_D = -np.inf
    best_d = None
    best_u = None
    st = dual_state(p, h, q, G, n, L)
    if st is None:
        return None
    d_cur, u_cur, D_cur, gap_cur = st
    best_D = D_cur
    best_d = d_cur.copy()
    best_u = u_cur.copy()
    x = p.copy()
    yv = p.copy()
    theta = 1.0
    for it in range(max_inner):
        # Evaluate at y
        st_y = dual_state(yv, h, q, G, n, L)
        if st_y is None:
            break
        d_y, u_y, D_y, _ = st_y
        grad = u_y  # gradient of dual is q + G*d = u
        x_new = project_simplex(yv + grad / Ldual)
        st_x = dual_state(x_new, h, q, G, n, L)
        if st_x is None:
            break
        d_x, u_x, D_x, gap_x = st_x
        if D_x < best_D - 1e-13:
            # restart
            yv = x_new.copy()
            theta = 1.0
        else:
            theta_new = (1.0 + math.sqrt(1.0 + 4.0 * theta * theta)) / 2.0
            yv = x_new + ((theta - 1.0) / theta_new) * (x_new - yv)
            yv = project_simplex(yv)
            theta = theta_new
        x = x_new
        if D_x > best_D:
            best_D = D_x
            best_d = d_x.copy()
            best_u = u_x.copy()
        if D_y > best_D:
            best_D = D_y
            best_d = d_y.copy()
            best_u = u_y.copy()
        if gap_x <= max(1e-10, 1e-8 * max(1.0, abs(float(np.max(u_x))))):
            break
        if np.max(np.abs(x_new - x)) <= 1e-12:
            break
    return best_d

def optimize_start(h0, n, max_outer, max_inner):
    h, ok = project_capped(h0.copy(), n)
    if not ok:
        return h0.copy(), float('inf')
    s = exact_score(h, n)
    best_h = h.copy()
    best_s = s
    stall = 0
    for outer in range(max_outer):
        q = lag_scores(h, n)
        if not np.all(np.isfinite(q)):
            break
        G = build_G(h, n)
        if not np.all(np.isfinite(G)):
            break
        d = solve_majorant(h, q, G, n, max_inner)
        if d is None or not np.all(np.isfinite(d)) or np.max(np.abs(d)) <= 1e-10:
            break
        v = G @ d
        nrm2 = d.dot(d)
        L = 4.0 / n
        accepted = False
        for aw in range(24):
            alpha = 2.0 ** (-aw)
            Mval = float(np.max(q + alpha * v)) + 0.5 * L * alpha * alpha * nrm2
            if Mval > float(np.max(q)) - 1e-14:
                continue
            h_trial = h + alpha * d
            ht, ok2 = project_capped(h_trial, n)
            if not ok2:
                continue
            st = exact_score(ht, n)
            if st < s - 1e-12:
                h = ht
                s = st
                if s < best_s - 1e-12:
                    best_s = s
                    best_h = h.copy()
                accepted = True
                break
        if not accepted:
            break
        if s >= best_s - 1e-11:
            stall += 1
            if stall >= 2:
                break
        else:
            stall = 0
    return best_h, best_s

    fs_fb = exact_score(fb, n)
    if fs > fs_fb + 1e-12:
        fh = fb.copy()
        fs = fs_fb
    fh = project_capped(fh, n)[0]
    fs = exact_score(fh, n)
    return fh.astype(np.float64), float(fs), int(n)

import math

def project_capped(y, n):
    S = n / 2.0
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return np.full(n, 0.5), False
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    lam = float(np.clip(y, lo, hi).mean()) - 0.5
    lam = max(lo, min(hi, lam))
    for _ in range(80):

        if abs(r) <= 1e-14:
            break
        if r > 0:
            hi = lam
        else:
            lo = lam
        a = int(np.sum((y - lam > 0) & (y - lam < 1)))
        if a > 0:
            lam_new = lam + r / a
        else:
            lam_new = 0.5 * (lo + hi)
        if lo < lam_new < hi and np.isfinite(lam_new):
            lam = lam_new
        else:
            lam = 0.5 * (lo + hi)

    np.clip(c, 0.0, 1.0, out=c)
    return c, True

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(h.sum() - n / 2.0) <= 1e-10)

def lag_scores(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    qs = lag_scores(h, n)
    if not np.all(np.isfinite(qs)):
        return float('inf')
    return float(np.max(qs))

def build_G(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        row = k + n - 1
        lo1, hi1 = max(0, k), min(n, n + k)
        if hi1 > lo1:
            G[row, lo1:hi1] += dx * (1.0 - h[lo1 - k:hi1 - k])
        lo2, hi2 = max(0, -k), min(n, n - k)
        if hi2 > lo2:
            G[row, lo2:hi2] -= dx * h[lo2 + k:hi2 + k]
    return G

def project_simplex(z):
    ns = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(ns, 1.0 / ns)
    sv = np.sort(z)[::-1].copy()
    cs = np.cumsum(sv)
    idx = np.where(sv - (cs - 1.0) / np.arange(1, ns + 1) > 0)[0]
    if len(idx) == 0:
        return np.full(ns, 1.0 / ns)
    r = idx[-1]
    theta = (cs[r] - 1.0) / (r + 1.0)
    out = np.maximum(z - theta, 0.0)
    d = out.sum() - 1.0
    if abs(d) > 1e-15:
        pos = out > 0
        if pos.sum() > 0:
            out[pos] -= d / pos.sum()
        np.clip(out, 0.0, None, out=out)
    return out

def dual_eval(p, h, q, G, n, L):
    gmix = G.T.dot(p)
    y, ok = project_capped(h - gmix / L, n)
    if not ok:
        return None
    d = y - h
    u = q + G.dot(d)
    D = float(p.dot(u) + 0.5 * L * d.dot(d))
    if not np.all(np.isfinite(u)) or not np.isfinite(D):
        return None
    gap = float(np.max(u) - p.dot(u))
    return d, u, D, gap

def solve_majorant(h, q, G, n, max_inner):
    L = 4.0 / n
    Ldual = max(1e-12, 1.5 * float((G * G).sum()) / L)
    m = len(q)
    qmax = float(np.max(q))
    tol = max(1e-12, 1e-8 * max(1.0, abs(qmax)))
    sel = np.where(q >= qmax - tol)[0]
    p = np.zeros(m)
    if len(sel) > 0:
        p[sel] = 1.0 / len(sel)
    else:
        p[np.argmax(q)] = 1.0
    st = dual_eval(p, h, q, G, n, L)
    if st is None:
        return None
    d_cur, u_cur, D_cur, _ = st
    best_D = D_cur
    best_d = d_cur.copy()
    x = p.copy()
    yv = p.copy()
    theta = 1.0
    for _ in range(max_inner):
        st_y = dual_eval(yv, h, q, G, n, L)
        if st_y is None:
            break
        d_y, u_y, D_y, _ = st_y
        x_new = project_simplex(yv + u_y / Ldual)
        st_x = dual_eval(x_new, h, q, G, n, L)
        if st_x is None:
            break
        d_x, u_x, D_x, gap_x = st_x
        if D_x < best_D - 1e-13:
            yv = x_new.copy()
            theta = 1.0
        else:
            tn = (1.0 + math.sqrt(1.0 + 4.0 * theta * theta)) / 2.0
            yv = x_new + ((theta - 1.0) / tn) * (x_new - yv)
            yv = project_simplex(yv)
            theta = tn
        x = x_new
        if D_x > best_D:
            best_D, best_d = D_x, d_x.copy()
        if D_y > best_D:
            best_D, best_d = D_y, d_y.copy()
        if gap_x <= max(1e-10, 1e-8 * max(1.0, abs(float(np.max(u_x))))):
            break
        if float(np.max(np.abs(x_new - x))) <= 1e-13:
            break
    return best_d
