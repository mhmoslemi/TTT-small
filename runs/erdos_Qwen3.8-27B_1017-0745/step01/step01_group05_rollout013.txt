def project_onto_k(y, n):
    S = n / 2.0
    if not np.all(np.isfinite(y)):

    hi = float(np.max(y))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        c = np.clip(y - mid, 0.0, 1.0)
        if c.sum() > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(50):
        diff = S - c.sum()
        if abs(diff) < 1e-14:

        else:
