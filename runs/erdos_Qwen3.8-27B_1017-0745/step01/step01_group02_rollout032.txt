def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.ndim != 1 or v.size == 0 or not np.all(np.isfinite(v)):
        return None
    S = float(S)
    lo = float(np.max(v)) + 1.0
    hi = float(np.min(v)) - 1.0
    # We want sum(clip(v - mid, 0, 1)) = S
    # clip(v - mid, 0, 1) is non-increasing in mid
    for _ in range(80):

        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            hi = mid
        else:

        idx = np.argsort(1.0 - y)

            r -= d

                s_best, h_best = s_cur, h_cur.copy()
    return h_best, s_best

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent, n = get_parent()
    best_h, best_s = parent.copy(), exact_score(parent, n)
    starts = build_starts(parent, n, rng)
    for s in starts:
        sc = exact_score(s, n)
        if sc < best_s:
            best_s, best_h = sc, s.copy()
    scored = sorted(starts, key=start_score.__get__(None, type(starts[0])) if False else (lambda x: exact_score(x, n)))
    if n <= 64:
        ncyc, nlev, npr, sig0 = 4, 6, 240, 0.08
    elif n <= 256:
        ncyc, nlev, npr, sig0 = 3, 5, 90, 0.06
    else:
        ncyc, nlev, npr, sig0 = 2, 4, 24, 0.04
    for cyc in range(ncyc):
        try:
            if cyc == 0 and len(scored) > 0:
                cstart = scored[0].copy()
            elif cyc == 1 and len(scored) > 1:
                cstart = scored[1].copy()
            else:
                sd = 0.10 if cyc == 2 else 0.22
                cstart = None
                for _ in range(5):
                    cstart = project_box_sum(best_h + sd * rng.standard_normal(n), 0.5 * n)
                    if cstart is not None and check_feasible(cstart, n):
                        break
                if cstart is None or not check_feasible(cstart, n):
                    cstart = best_h.copy()
            sc0 = exact_score(cstart, n)
            if sc0 < best_s:
                best_s, best_h = sc0, cstart.copy()
            lb, ls = anneal_cycle(cstart, n, rng, nlev, npr, sig0)
            if ls < best_s:
                best_s, best_h = ls, lb.copy()
        except Exception:
            pass
    fh = project_box_sum(best_h, 0.5 * n)
    if fh is not None and check_feasible(fh, n):
        best_h = fh
    else:
        fb = project_box_sum(parent, 0.5 * n)
        best_h = fb if fb is not None and check_feasible(fb, n) else np.full(n, 0.5)
    best_s = float(np.max(np.correlate(best_h, 1.0 - best_h, mode='full'))) * (2.0 / n)
    if not np.isfinite(best_s):
        best_h = np.full(n, 0.5)
        best_s = float(np.max(np.correlate(best_h, 1.0 - best_h, mode='full'))) * (2.0 / n)
    return (best_h, best_s, int(len(best_h)))
