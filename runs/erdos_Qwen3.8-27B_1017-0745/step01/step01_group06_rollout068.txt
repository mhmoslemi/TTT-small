def transfer(parent, n0, m):
    y = np.zeros(m)
    for j in range(m):
        b1, b2 = j / m, (j + 1) / m
        ov = 0.0
        for i in range(n0):
            ov += parent[i] * max(0.0, min((i + 1) / n0, b2) - max(i / n0, b1))
        y[j] = m * ov
    return y
