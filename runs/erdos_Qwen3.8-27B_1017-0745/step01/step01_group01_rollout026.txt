                continue
            ts = exact_score(proj, n)
            if np.isfinite(ts) and ts < old_scores[i] - 1e-12:
                pop[i] = proj
                pscores[i] = ts
                if ts < best_score - 1e-12:
                    best, best_score = proj.copy(), ts
        if best_score < prev_best - 1e-12:
            stag = 0
        else:
            stag += 1
        if stag >= patience:
            if restarts_done >= nrestart:
                break
            restarts_done += 1
            stag = 0
            n_keep = max(2, math.ceil(0.2 * NP))
            order = np.argsort(pscores)
            new_pop = [best.copy()]
            new_scores = [best_score]
            for idx2 in order:
                if len(new_pop) >= n_keep:
                    break
                if float(np.max(np.abs(pop[idx2] - best))) < 1e-10:
                    continue
                new_pop.append(pop[idx2].copy())
                new_scores.append(pscores[idx2])
            base_scales = [0.03, 0.07, 0.15]
            for j in range(len(new_pop), NP):
                src = new_pop[rng.integers(0, len(new_pop))]
                sc = base_scales[(restarts_done - 1) % 3] * (0.5 + rng.random())
                proj = project_capped(src + rng.normal(0, sc, n), n)
                if proj is None:
                    proj = project_capped(np.clip(rng.uniform(0, 1, n) * 2.0, 0, 1), n)
                if proj is None:
                    proj = best.copy()
                new_pop.append(proj)
                new_scores.append(exact_score(proj, n))
            pop = np.array(new_pop[:NP])
            pscores = np.array(new_scores[:NP])
    candidates = [best, fallback, uniform]
    final, final_score = uniform.copy(), fb_score
    for c in candidates:
        if c is None:
            continue
        cs = exact_score(c, n)
        if np.isfinite(cs) and cs < final_score - 1e-14:
            final, final_score = c.copy(), cs
    rep = project_capped(final, n)
    if rep is not None:
        rs = exact_score(rep, n)
        if np.isfinite(rs) and rs <= final_score + 1e-12:
            final, final_score = rep, rs
    if not np.isfinite(final_score):
        final, final_score = uniform.copy(), fb_score
    return final.copy(), float(final_score), n
