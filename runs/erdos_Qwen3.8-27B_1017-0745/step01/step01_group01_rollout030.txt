    add_unique(starts, rng.uniform(0, 1, n), n)
    best = uniform.copy()
    best_score = exact_score(uniform, n)
    for st in starts:
        sc = exact_score(st, n)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best, best_score = st.copy(), sc
    if n <= 64:
        NP_d, ng_d, nr_d = min(64, max(16, 4 * n)), 160, 2
    elif n <= 128:
        NP_d, ng_d, nr_d = 48, 80, 1
    elif n <= 256:
        NP_d, ng_d, nr_d = 32, 30, 0
    else:
        NP_d, ng_d, nr_d = 1, 0, 0
    NP = parse_int(kwargs.get('pop_size', NP_d), 8, 96) or NP_d
    cap_g = 250 if n <= 64 else 120 if n <= 128 else 50
    ng = parse_int(kwargs.get('generations', ng_d), 0, cap_g)
    nrestart = parse_int(kwargs.get('max_restarts', nr_d), 0, 3) or 0
    if n > 256 or ng <= 0 or NP < 4:
        final = project_capped(best, n)
        if final is None:
            final = uniform.copy()
        fs = exact_score(final, n)
        if not np.isfinite(fs) or fs > best_score:
            final = uniform.copy()
            fs = exact_score(uniform, n)
        return final, float(fs), int(n)
    pop_list = [best.copy()]
    if parent_f is not None:
        pop_list.append(parent_f.copy())
    for st in starts:
        if len(pop_list) >= NP:
            break
        if any(float(np.max(np.abs(st - p))) < 1e-10 for p in pop_list):
            continue
        pop_list.append(st.copy())
    psc = [0.015, 0.04, 0.10, 0.22]
    pi = 0
    while len(pop_list) < NP and pi < 300:
        base = pop_list[pi % max(1, len(pop_list))]
        proj = project_capped(base + rng.normal(0, psc[pi % 4], n), n)
        if proj is None:
            proj = project_capped(rng.uniform(0, 1, n), n)
        if proj is None:
            proj = uniform.copy()
        pop_list.append(proj)
        pi += 1
    pop = np.array(pop_list[:NP], dtype=np.float64)
    scores = np.array([exact_score(pop[i], n) for i in range(NP)])
    for i in range(NP):
        if np.isfinite(scores[i]) and scores[i] < best_score - 1e-12:
            best, best_score = pop[i].copy(), scores[i]
    patience = max(12, min(40, ng // 4))
    stag = 0
    restarts_used = 0
    gen_best = best_score
    for gen in range(ng):
        old_pop = pop.copy()
        old_scores = scores.copy()
        b_idx = int(np.argmin(old_scores))
        for i in range(NP):
            r_idx = list(range(NP))
            r_idx.remove(i)
            if rng.random() < 0.8:
                sel = rng.choice(len(r_idx), size=3, replace=False)
                r1, r2, r3 = r_idx[sel[0]], r_idx[sel[1]], r_idx[sel[2]]
                F = 0.45 + 0.50 * rng.random()
                mut = old_pop[r1] + F * (old_pop[r2] - old_pop[r3])
            else:
                sel = rng.choice(len(r_idx), size=2, replace=False)
                r1, r2 = r_idx[sel[0]], r_idx[sel[1]]
                F = 0.45 + 0.50 * rng.random()
                mut = old_pop[i] + F * (old_pop[b_idx] - old_pop[i]) + F * (old_pop[r1] - old_pop[r2])
            CR = 0.75 + 0.20 * rng.random()
            mask = rng.random(n) < CR
            mask[int(rng.integers(0, n))] = True
            cross = old_pop[i].copy()
            cross[mask] = mut[mask]
            trial = project_capped(cross, n)
            if trial is None:
                continue
            if float(np.max(np.abs(trial - old_pop[i]))) < 1e-12:
                continue
            ts = exact_score(trial, n)
            if np.isfinite(ts) and ts < old_scores[i] - 1e-12:
                pop[i] = trial
                scores[i] = ts
                if ts < best_score - 1e-12:
                    best, best_score = trial.copy(), ts
        if best_score < gen_best - 1e-12:
            gen_best = best_score
            stag = 0
        else:
            stag += 1
            if stag >= patience:
                if restarts_used < nrestart:
                    restarts_used += 1
                    elite_n = max(2, int(np.ceil(0.2 * NP)))
                    order = np.argsort(scores)
                    new_pop = [best.copy()]
                    for ei in order:
                        if len(new_pop) >= elite_n:
                            break
                        if any(float(np.max(np.abs(pop[ei] - x))) < 1e-10 for x in new_pop):
                            continue
                        new_pop.append(pop[ei].copy())
                    bs = [0.03, 0.07, 0.15][min(restarts_used - 1, 2)] * (0.5 + rng.random())
                    while len(new_pop) < NP:
                        base = new_pop[int(rng.integers(0, len(new_pop)))]
                        proj = project_capped(base + rng.normal(0, bs, n), n)
                        if proj is None:
                            proj = project_capped(rng.uniform(0, 1, n), n)
                        if proj is None:
                            proj = uniform.copy()
                        new_pop.append(proj)
                    pop = np.array(new_pop[:NP], dtype=np.float64)
                    scores = np.array([exact_score(pop[i], n) for i in range(NP)])
                    for i in range(NP):
                        if np.isfinite(scores[i]) and scores[i] < best_score - 1e-12:
                            best, best_score = pop[i].copy(), scores[i]
                            gen_best = best_score
                    stag = 0
                else:
                    break
    cands = [best.copy()]
    if parent_f is not None:
        cands.append(parent_f.copy())
    cands.append(uniform.copy())
    best_h, best_sc = None, float('inf')
    for c in cands:
        pc = project_capped(c, n)
        if pc is None:
            continue
        sc = exact_score(pc, n)
        if np.isfinite(sc) and sc < best_sc:
            best_h, best_sc = pc, sc
    if best_h is None:
        best_h, best_sc = uniform.copy(), exact_score(uniform, n)
    final = project_capped(best_h, n)
    if final is not None:
        fs = exact_score(final, n)
        if np.isfinite(fs) and fs <= best_sc + 1e-12:
            best_h, best_sc = final, fs
    return best_h, float(best_sc), int(n)
