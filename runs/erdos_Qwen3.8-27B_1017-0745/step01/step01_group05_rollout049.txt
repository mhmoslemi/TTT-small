import numpy as np
import math

def project_onto_k(y, n):
    S = n / 2.0
    if y.shape[0] != n or not np.all(np.isfinite(y)) or np.max(np.abs(y)) > 1e12:
        return np.full(n, 0.5)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        c = np.clip(y - mid, 0.0, 1.0)
        if c.sum() > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(5):
        diff = S - c.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = c < 1.0 - 1e-12
        else:
            mask = c > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        c[mask] += diff / cnt
        np.clip(c, 0.0, 1.0, out=c)
    return c

def exact_score(h, n):
    if h.shape[0] != n or not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(h.sum() - n / 2.0) <= 1e-10)

    m = j + 1
    d[:m] = 1.0 / math.sqrt(m * (m + 1))
    d[m] = -m / math.sqrt(m * (m + 1))
    return d

def project_onto_k(y, n):
    S = n / 2.0
    if y.shape[0] != n or not np.all(np.isfinite(y)) or np.max(np.abs(y)) > 1e12:
        return np.full(n, 0.5)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        c = np.clip(y - mid, 0.0, 1.0)
        if c.sum() > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(5):
        diff = S - c.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = c < 1.0 - 1e-12
        else:
            mask = c > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        c[mask] += diff / cnt

def exact_score(h, n):
    if h.shape[0] != n or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(h.sum() - n / 2.0) <= 1e-10)

def helmert_dir(j, n):

    m = j + 1
    d[:m] = 1.0 / math.sqrt(m * (m + 1))
    d[m] = -m / math.sqrt(m * (m + 1))
    return d

def line_search(h, d, n):
    lo = -1e15
    hi = 1e15
    for i in range(n):
        if d[i] > 1e-15:
            lo = max(lo, -h[i] / d[i])
            hi = min(hi, (1.0 - h[i]) / d[i])
