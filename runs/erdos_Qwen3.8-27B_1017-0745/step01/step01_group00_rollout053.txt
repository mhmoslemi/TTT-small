    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0

def active_grad(h, d, n):
    dx = 2.0 / n
    g = np.zeros(n)
    p0 = max(d, 0)
    q0 = max(-d, 0)
    m = n - abs(d)
    g[p0:p0+m] += dx * (1.0 - h[q0:q0+m])
    g[q0:q0+m] -= dx * h[p0:p0+m]
    return g

def subgradient_search(start, n, n_iter=45):
    h = start.copy()
    score = exact_score(h, n)
    dx = 2.0 / n
    eta = 0.2 / dx
    eta_max = 2.0 / dx
    n_fail = 0
    for _ in range(n_iter):
        if n_fail >= 8 or eta < 1e-8 / dx:
            break
        q = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = int(np.argmax(q))
        d = idx - (n - 1)
        g_d = active_grad(h, d, n)
        z = project_capped(h - eta * g_d, n)
        if z is None or not is_feasible(z, n):
            eta *= 0.5
            n_fail += 1
            continue
        v = z - h
        if np.max(np.abs(v)) < 1e-10:
            eta *= 0.5
            n_fail += 1
            continue
        best_h, best_s = h, score
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            z2 = project_capped(h + alpha * v, n)
            if z2 is None or not is_feasible(z2, n):

        if best_s >= score - 1e-12:
            eta *= 0.5
            n_fail += 1
        else:
            h, score = best_h, best_s
            n_fail = 0
            eta = min(1.2 * eta, eta_max)
