            if d % 2 == 0 or len(pop) < 2:
                v = rng.normal(0, 1, n)
            else:
                i1 = int(rng.integers(len(pop)))
                i2 = int(rng.integers(len(pop)))
                if i1 == i2:
                    i2 = (i2 + 1) % len(pop)
                v = pop[i1] - pop[i2]
            v = v - np.mean(v)
            vm = np.max(np.abs(v))
            if vm < 1e-12 or not np.isfinite(vm):
                continue
            v = v / vm
            t_lo, t_hi = 0.0, 0.0
            for j in range(n):
                if v[j] > 1e-14:
                    t_lo = max(t_lo, -bs_arm0(h, j, v, j))
                    t_hi = min(t_hi, bs_arm1(h, j, v, j))
                elif v[j] < -1e-14:
                    t_lo = max(t_lo, bs_arm2(h, j, v, j))
                    t_hi = min(t_hi, bs_arm3(h, j, v, j))
            if t_lo > t_hi:
                continue
            for t in (-delta, -0.5 * delta, 0.5 * delta, delta):
                if abs(t) < 1e-12:
                    continue
                tc = min(max(t, t_lo), t_hi)
                if abs(tc) < 1e-12:
                    continue
                trial = h + tc * v
                p = project_capped(trial, n)
                if p is None:
                    continue
                sc = exact_score(p, n)
                if sc < bs - 1e-12:
                    bs, bh = sc, p
        if bs < gs - 1e-12:
            h, gs = bh, bs
            delta = min(0.25, 1.2 * delta)
        else:
            delta *= 0.5
    return h, gs

def bs_arm0(h, j, v, jj):
    return h[j] / v[j]

def bs_arm1(h, j, v, jj):
    return (1.0 - h[j]) / v[j]

def bs_arm2(h, j, v, jj):
    return (1.0 - h[j]) / v[j]

def bs_arm3(h, j, v, jj):
    return -h[j] / v[j]

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    fallback = project_capped(parent, n) if parent is not None else np.full(n, 0.5)
    if fallback is None:
        fallback = np.full(n, 0.5)
    fb_score = exact_score(fallback, n)
    rng = np.random.default_rng(int(seed))
    if n <= 1:
        return fallback, fb_score, 1
    pop_size, n_gen, stall_lim, n_rst, pol_rnd, pol_dr = de_config(n)
    cands = [fallback.copy()]
    cands.append(project_capped(fallback[::-1], n))
    cands.append(project_capped(1.0 - fallback[::-1], n))
    cands.append(np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    cands.append(project_capped(blk, n))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    cands.append(project_capped(alt, n))
    for sd in (0.08, 0.22):
        cands.append(project_capped(fallback + rng.normal(0, sd, n), n))
    cands.append(project_capped(rng.uniform(0, 1, n), n))
    for sh in (n // 3, n // 2):
        cands.append(project_capped(np.roll(fallback, sh), n))
    pm = rng.permutation(n)
    bb = np.zeros(n)
    bb[pm[:n // 2]] = 1.0
    if n % 2:
        bb[pm[n // 2]] = 0.5
    cands.append(project_capped(bb, n))
    pop, scores = [], []
    for c in cands:
        if c is None:
            continue
        c = project_capped(c, n)
        if c is None:
            continue
        dup = any(np.max(np.abs(c - pp)) < 1e-8 for pp in pop)
        if dup:
            continue
        pop.append(c)
        scores.append(exact_score(c, n))
    sn = np.clip(np.sqrt(47.0 / float(n)), 0.5, 1.5)
    tries = 0
    while len(pop) < pop_size and tries < 6 * pop_size:
        tries += 1
        r = rng.random()
        bi = int(np.argmin(scores))
        if r < 0.45:
            sig = float(rng.choice((0.03, 0.08, 0.16, 0.32))) * sn
            v = pop[bi] + rng.normal(0, sig, n)
        elif r < 0.70:
            v = pop[int(rng.integers(len(pop)))] + rng.normal(0, 0.12 * sn, n)
        elif r < 0.85:
            v = rng.uniform(0, 1, n)
        else:
            pm2 = rng.permutation(n)
            v = np.zeros(n)
            v[pm2[:n // 2]] = 1.0
            if n % 2:
                v[pm2[n // 2]] = 0.5
        p = project_capped(v, n)
        if p is None:
            p = pop[bi].copy()
        if any(np.max(np.abs(p - pp)) < 1e-8 for pp in pop):
            continue
        pop.append(p)
        scores.append(exact_score(p, n))
    while len(pop) < pop_size:
        bi = int(np.argmin(scores))
        pop.append(pop[bi].copy())
        scores.append(scores[bi])
    F_arr = rng.uniform(0.45, 0.85, pop_size)
    CR_arr = rng.uniform(0.4, 0.9, pop_size)
    gs = min(scores)
    gb = pop[int(np.argmin(scores))].copy()
    stall, rst_used = 0, 0
    for gen in range(n_gen):
        for i in range(pop_size):
            Fi = F_arr[i]
            if rng.random() < 0.1:
                Fi = 0.35 + 0.55 * rng.random()
            CRi = CR_arr[i]
            if rng.random() < 0.1:
                CRi = 0.10 + 0.85 * rng.random()
            bs_i = int(np.argmin(scores))
            oth = [j for j in range(pop_size) if j != i]
            if len(oth) < 3:
                continue
            r1, r2, r3 = rng.choice(oth, size=3, replace=False)
            if rng.random() < 0.5:
                mut = pop[r1] + Fi * (pop[r2] - pop[r3])
            else:
                mut = pop[i] + Fi * (pop[bs_i] - pop[i]) + Fi * (pop[r2] - pop[r3])
            trial = pop[i].copy()
            ci = int(rng.integers(n))
            for j in range(n):
                if rng.random() < CRi or j == ci:
                    trial[j] = mut[j]
            p = project_capped(trial, n)
            if p is None:
                continue
            sc = exact_score(p, n)
            if sc < scores[i] - 1e-12:
                pop[i] = p
                scores[i] = sc
                F_arr[i] = Fi
                CR_arr[i] = CRi
                if sc < gs - 1e-12:
                    gs, gb = sc, p.copy()
        if gs < min(scores) - 1e-12:
            wk = int(np.argmax(scores))
            pop[wk] = gb.copy()
            scores[wk] = gs
        if min(scores) < gs - 1e-12:
            gs = min(scores)
            gb = pop[int(np.argmin(scores))].copy()
            stall = 0
        else:
            stall += 1
        if stall >= stall_lim and rst_used < n_rst:
            pop, scores = restart_pop(pop, scores, n, rng, gb, gs)
            F_arr = rng.uniform(0.45, 0.85, pop_size)
            CR_arr = rng.uniform(0.4, 0.9, pop_size)
            stall, rst_used = 0, rst_used + 1
    gb, gs = pattern_poll(gb, n, rng, pop, pol_rnd, pol_dr)
    gb = project_capped(gb, n)
    if gb is None or not is_feasible(gb, n):
        gb = fallback.copy()
    fin_sc = exact_score(gb, n)
    if fin_sc > fb_score:
        gb, fin_sc = fallback.copy(), fb_score
    return gb.astype(np.float64), float(fin_sc), int(n)
