```python
def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0 or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-14:
        if resid > 0:
            order = np.argsort(1.0 - y)
            for i in order:
                add = min(resid, 1.0 - y[i])
                y[i] += add
                resid -= add
                if abs(resid) < 1e-15:
                    break
        else:
            order = np.argsort(y)
            for i in order:
                sub = min(-resid, y[i])
                y[i] -= sub
                resid += sub

                    break
    y = np.clip(y, 0.0, 1.0)
    if abs(np.sum(y) - S) > 1e-10 * max(1.0, float(n)):
        return None
    return y

def exact_score(h, n):

        return np.inf
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))
```

```python
def subgrad_polish(h, n, S, steps=200, lr0=0.01, rng=None):
    cur = h.copy(); sc = score(cur, n)
    for it in range(steps):
        lr = lr0/(1+0.01*it)
        # Compute active lag
        c = np.correlate(cur, 1-cur, 'full')
        k_star = np.argmax(c)
        # Gradient of Q_k_star with respect to h
        k = k_star - (n-1)
        i0, i1 = max(0,k), min(n-1,n-1+k)
        grad = np.zeros(n)
        ii = np.arange(i0, i1+1)
        jj = ii - k
        grad[ii] += (1-cur[jj])
        grad[jj] += (-cur[ii])
        # Projected subgradient step
        new = cur - lr*grad
        proj = project_box_sum(new, S)
        if proj is not None:
            new_sc = score(proj, n)
            if new_sc < sc - 1e-14:
                cur, sc = proj, new_sc
    return cur
```

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0 or not np.all(np.isfinite(v)):
        return None
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0, 1)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0, 1)
    r = S - np.sum(y)
    if abs(r) > 1e-14:
        order = np.argsort(1 - y) if r > 0 else np.argsort(y)
        for i in order:
            if r > 0:
                a = min(r, 1 - y[i]); y[i] += a; r -= a
            else:
                a = min(-r, y[i]); y[i] -= a; r += a
            if abs(r) < 1e-15:
                break
    y = np.clip(y, 0, 1)
    return y if abs(np.sum(y) - S) < 1e-10 * max(1.0, n) else None

def score(h, n):
    if h is None or h.shape != (n,) or not np.all(np.isfinite(h)):
        return np.inf
    if np.any(h < -1e-12) or np.any(h > 1 + 1e-12):
        return np.inf
    if abs(np.sum(h) - n / 2) > 1e-10 * max(1.0, n):
        return np.inf
    return float(np.max(np.correlate(h, 1 - h, 'full')) * 2.0 / n)

def is_feas(h, n):
    return score(h, n) < np.inf

def line_min(x, y, n, S):
    if not (is_feas(x, n) and is_feas(y, n)):
        return None
    d = y - x
    nl = 2 * n - 1
    A = np.zeros(nl); B = np.zeros(nl); D = np.zeros(nl)
    for k in range(-(n - 1), n):
        i0, i1 = max(0, k), min(n - 1, n - 1 + k)
        if i0 > i1:
            continue
        ii = np.arange(i0, i1 + 1); jj = ii - k; I = k + n - 1
        D[I] = np.sum(x[ii] * (1 - x[jj]))
        B[I] = np.sum(d[ii] * (1 - x[jj]) - x[ii] * d[jj])
        A[I] = -np.sum(d[ii] * d[jj])
    al = [0.0, 1.0]
    for k in range(nl):

            a = -B[k] / (2 * A[k])

                al.append(max(0.0, min(1.0, a)))
    for k in range(nl):
        for l in range(k + 1, nl):

            if abs(da) < 1e-15:
                if abs(db) > 1e-15:
                    a = -dc / db
                    if -1e-10 <= a <= 1 + 1e-10:
