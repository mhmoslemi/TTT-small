With discretization, dx = 2/n, and the integral becomes sum * dx.

Actually, np.correlate(a, v, mode='full') computes:

So np.correlate(h, 1-h, mode='full')[k] = sum_i h[i+k] * (1-h[i]) for valid i

The full result has 2n-1 elements indexed 0 to 2n-2.

At index k (0 ≤ k ≤ 2n-2):
result[k] = sum_{i=0}^{n-1} h[i+k-(n-1)] * (1-h[i]) where both indices are in [0,n-1]

Let me verify: for k = n-1 (middle), result[n-1] = sum_i h[i] * (1-h[i]) = sum h[i] - sum h[i]². This is the k=0 lag.

import numpy as np
from scipy.optimize import minimize

_md = {}

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    if S < -1e-6 or S > n + 1e-6:
        return None
    S = max(0.0, min(float(n), S))
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-14:
        for i in np.argsort(1.0 - y)[::-1]:
            d = min(r, 1.0 - y[i]); y[i] += d; r -= d
            if r <= 0: break
    elif r < -1e-14:
        for i in np.argsort(y)[::-1]:
            d = min(-r, y[i]); y[i] -= d; r += d
            if r >= 0: break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-9) or np.any(y > 1.0+1e-9):
        return None
    if abs(float(np.sum(y)) - S) > 1e-10 * max(1, n):
        return None
    return y

def chk(h, n):
    return (len(h)==n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0+1e-12) and
            abs(float(np.sum(h)) - 0.5*n) < 1e-10*max(1,n))

def escore(h):
    n = len(h)
    if not chk(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * (2.0/n))

def lag_vg(h, s, n):
    dx = 2.0 / n
    lo, hi = max(0, -s), min(n, n - s)
    q, g = 0.0, np.zeros(n)
    for i in range(lo, hi):
        j = i + s
        q += h[i] * (1.0 - h[j])
        g[i] += 1.0 - h[j]
        g[j] -= h[i]
    return dx * q, dx * g

def _obj(z):
    x, L = _md['x'], _md['L']
    d = z[:len(x)] - x
    return 0.5 * L * np.dot(d, d) + z[-1]

def _objg(z):
    x, L = _md['x'], _md['L']
    d = z[:len(x)] - x
    g = np.zeros(len(x) + 1)
    g[:len(x)] = L * d; g[-1] = 1.0
    return g

def _ineq(z):
    x, qJ, GJ = _md['x'], _md['qJ'], _md['GJ']
    d = z[:len(x)] - x
    return z[-1] - qJ - GJ @ d

def _ineqg(z):
    GJ = _md['GJ']
    n2, m = GJ.shape
    J = np.zeros((m, n2 + 1))
    J[:, :n2] = -GJ; J[:, n2] = 1.0
    return J

def _eq(z):
    return float(np.sum(z[:-1])) - 0.5 * (len(z) - 1)

def _eqg(z):
    n2 = len(z) - 1
    g = np.zeros(n2 + 1); g[:n2] = 1.0
    return g

def mm_step(x, n):
    L = 4.0 / n
    qv, G = [], []
    for s in range(-(n-1), n):
        q, g = lag_vg(x, s, n)
        qv.append(q); G.append(g)
    qv, G = np.array(qv), np.array(G)
    _md.update(x=x.copy(), L=L, qJ=qv, GJ=G)
    b = [(0.0, 1.0)] * n + [(None, None)]
    c = [{'type':'ineq','fun':_ineq,'jac':_ineqg},
         {'type':'eq','fun':_eq,'jac':_eqg}]
    t0 = float(np.max(qv)) + 1e-6
    z0 = np.append(x.copy(), t0)
    try:
        res = minimize(_obj, z0, jac=_objg, method='SLSQP',
                       bounds=b, constraints=c,
                       options={'maxiter':150,'ftol':1e-12,'disp':False})
        y = project_box_sum(np.array(res.x[:n], dtype=np.float64), 0.5*n)
        if y is not None and chk(y, n):
            return y
    except Exception:
        pass
    return x.copy()

def local_mm(h, n, M):
    h = h.copy(); sc = escore(h)
    for _ in range(M):
        y = mm_step(h, n)
        if not np.all(np.isfinite(y)) or np.max(np.abs(y - h)) < 1e-12:
            break
        bs, bp = sc, h
        for k in range(9):
            a = 0.5**k
            pt = project_box_sum(h + a*(y - h), 0.5*n)
            if pt is None: continue
            s = escore(pt)
            if s < bs - 1e-13: bs, bp = s, pt
        if bs >= sc - 1e-12: break
        if sc - bs < 1e-11 * max(1, sc): break
        h, sc = bp.copy(), bs
    return h, sc

def _get_parent():
    if 'initial_h_values' in globals():
        ph = np.asarray(globals()['initial_h_values'], dtype=np.float64)
        if ph.ndim==1 and 2<=len(ph)<=999 and np.all(np.isfinite(ph)):
            if np.all(ph >= -1e-6) and np.all(ph <= 1.0+1e-6):
                n = len(ph)
                if abs(float(np.sum(ph)) - 0.5*n) < 1e-4*n:
                    p = project_box_sum(np.clip(ph, 0, 1), 0.5*n)
                    if p is not None and chk(p, n):
                        return p, n
    return (project_box_sum(np.full(45, 0.5), 22.5), 45)

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent, n = _get_parent()
    best_h = parent.copy(); best_c5 = escore(parent)
    
    # Build diverse starts
    starts = [parent.copy(), (1.0-parent).copy(), parent[::-1].copy()]
    for sd in [0.06, 0.15, 0.30]:
        p = project_box_sum(parent + sd*rng.standard_normal(n), 0.5*n)
        if p is not None: starts.append(p)
    p = project_box_sum(rng.random(n), 0.5*n)
    if p is not None: starts.append(p)
    idx = np.arange(n)
    u = 0.5 + 0.3*np.cos(2*np.pi*idx/n + 0.7) + 0.12*np.sin(4*np.pi*idx/n + 1.3)
    p = project_box_sum(u, 0.5*n)
    if p is not None: starts.append(p)
    u = 0.5 + 0.25*np.cos(4*np.pi*idx/n)
    p = project_box_sum(u, 0.5*n)
    if p is not None: starts.append(p)
    
    # Deduplicate
    uniq = [starts[0]]
    for s in starts[1:]:
        if not any(np.max(np.abs(s - u)) < 1e-10 for u in uniq):
            uniq.append(s)
    
    M = max(10, min(30, 1500 // n))
    
    for h0 in uniq:
        try:
            hres, sc = local_mm(h0, n, M)
            if sc < best_c5:
                best_h, best_c5 = hres.copy(), sc
        except Exception:
            pass
    
    # Perturbation restarts
    for sd in [0.03, 0.10, 0.25, 0.15]:
        h0 = project_box_sum(best_h + sd*rng.standard_normal(n), 0.5*n)
        if h0 is None: continue
        try:
            hres, sc = local_mm(h0, n, max(5, M//2))
            if sc < best_c5:
                best_h, best_c5 = hres.copy(), sc
        except Exception:
            pass
    
    # Fine polish
    for _ in range(3):
        h0 = project_box_sum(best_h + 0.01*rng.standard_normal(n), 0.5*n)
        if h0 is None: break
        try:
            hres, sc = local_mm(h0, n, 8)
            if sc < best_c5:
                best_h, best_c5 = hres.copy(), sc
        except Exception:
            pass
    
    # Final validation
    best_h = project_box_sum(best_h, 0.5*n)
    if best_h is not None and chk(best_h, n):
        best_c5 = escore(best_h)
    pf = project_box_sum(np.clip(parent, 0, 1), 0.5*n)
    if pf is not None and chk(pf, n) and escore(pf) < best_c5:
        best_h, best_c5 = pf.copy(), escore(pf)
    
    return (best_h, best_c5, n)
