def project_capped(y, n):

    if not np.all(np.isfinite(y)):

        if c.sum() > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(10):
        diff = S - c.sum()

        else:
            mask = c > 1e-12

    if not check_feasible(h, n):
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

        d[:j+1] = 1.0 / s

import numpy as np
import math

def project_capped(y, n):
    S = n / 2.0
    if not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        c = np.clip(y - mid, 0.0, 1.0)
        if c.sum() > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(10):
        diff = S - c.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = c < 1.0 - 1e-12
        else:
            mask = c > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        c[mask] += diff / cnt
        np.clip(c, 0.0, 1.0, out=c)
    return c

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(h.sum() - n / 2.0) <= 1e-10)

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def helmert_dir(j, n):
    d = np.zeros(n)
    if j < n - 1:
        s = math.sqrt((j + 1) * (j + 2))
        d[:j+1] = 1.0 / s
        d[j+1] = -(j + 1) / s
    return d

def line_search(h, d, n):
    s0 = exact_score(h, n)
    if not np.all(np.isfinite(d)) or abs(d.sum()) > 1e-10:
        return h, s0
    lo_t, hi_t = 0.0, 0.0
    mp = d > 1e-15
    mn = d < -1e-15
    if mp.any():
        lo_t = max(lo_t, float(np.min(-h[mp] / d[mp])))
        hi_t = min(hi_t, float(np.min((1.0 - h[mp]) / d[mp])))
    if mn.any():
        lo_t = max(lo_t, float(np.min((1.0 - h[mn]) / d[mn])))
        hi_t = min(hi_t, float(np.min(-h[mn] / d[mn])))
    lo_t = min(lo_t, 0.0)
    hi_t = max(hi_t, 0.0)
    if hi_t - lo_t < 1e-14:
        return h, s0
    dx = 2.0 / n
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    b = (np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')) * dx
    a = -np.correlate(d, d, mode='full') * dx
    m0 = (lo_t + hi_t) / 2.0
    r0 = (hi_t - lo_t) / 2.0
    A = a * r0 * r0
    B = r0 * (b + 2.0 * a * m0)
    C = q + b * m0 + a * m0 * m0
    nl = len(q)
    sc_list = [-1.0, 1.0, -m0 / r0 if r0 > 0 else 0.0]
    for i in range(nl):
        if abs(A[i]) > 1e-15:
            sp = -B[i] / (2.0 * A[i])
            if -1.0 <= sp <= 1.0:
                sc_list.append(sp)
    for i in range(nl):
        for j in range(i + 1, nl):
            da = A[i] - A[j]
            db = B[i] - B[j]
            dc = C[i] - C[j]
            if abs(da) < 1e-15:
                if abs(db) > 1e-15:
                    rt = -dc / db
                    if -1.0 <= rt <= 1.0:
                        sc_list.append(rt)
            else:
                disc = db * db - 4.0 * da * dc
                if disc >= -1e-12:
                    disc = max(0.0, disc)
                    sq = math.sqrt(disc)
                    for rt in [(-db + sq) / (2.0 * da), (-db - sq) / (2.0 * da)]:
                        if -1.0 <= rt <= 1.0:
                            sc_list.append(rt)
    sc_list = [max(-1.0, min(1.0, s)) for s in sc_list]
    sc_list = sorted(set(round(s, 13) for s in sc_list))
    best_val = float('inf')
    best_sc = 0.0
    for sc in sc_list:
        vals = A * sc * sc + B * sc + C
        v = float(np.max(vals))
        if v < best_val:
            best_val = v
            best_sc = sc
    best_t = max(lo_t, min(hi_t, m0 + best_sc * r0))
    h_new = project_capped(h + best_t * d, n)
    s_new = exact_score(h_new, n)
    if s_new < s0 - 1e-14:
        return h_new, s_new
    return h, s0

def polish(h0, n, rng):
    h = h0.copy()
    s = exact_score(h, n)
    best_h, best_s = h.copy(), s
    n_dirs = n - 1
    for _sweep in range(2):
        perm = rng.permutation(n_dirs)
        improved = False
        for jj in perm:
            d = helmert_dir(int(jj), n)
            h, s = line_search(h, d, n)
            if s < best_s - 1e-14:
                best_s = s
                best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_s

def sdp_candidates(n, dx, fallback, rng):
    try:
        import cvxpy as cp
        S = n / 2.0
        hv = cp.Variable(n)
        Xv = cp.Variable((n, n), symmetric=True)
        tv = cp.Variable()
        blk = cp.bmat([[cp.Constant(1), hv], [hv.T, Xv]])
        cons = [blk >> 0, hv >= 0, hv <= 1, cp.sum(hv) == S]
        for i in range(n):
            for j in range(i + 1, n):
                cons += [Xv[i, j] >= 0, Xv[i, j] >= hv[i] + hv[j] - 1,
                         Xv[i, j] <= hv[i], Xv[i, j] <= hv[j]]
            cons += [Xv[i, i] >= 0, Xv[i, i] >= 2 * hv[i] - 1, Xv[i, i] <= hv[i]]
            cons += [cp.sum(Xv[i, :]) == S * hv[i]]
        for k in range(-(n - 1), n):
            lo_i = max(0, k)
            hi_i = min(n, n + k)
            if hi_i <= lo_i:
                continue
            terms = sum(hv[ri] - Xv[ri, ri - k] for ri in range(lo_i, hi_i))
            cons += [tv >= dx * terms]
        prob = cp.Problem(cp.Minimize(tv), cons)
        solved = False
        for solver in [cp.MOSEK, cp.CVXOPT]:
            try:
                if solver is cp.MOSEK:
                    prob.solve(solver=cp.MOSEK, iterations=100)
                else:
                    prob.solve(solver=cp.CVXOPT, abstol=1e-7, reltol=1e-7, feastol=1e-7, num_iters=100)
                if prob.status in ('optimal', 'optimal_inaccurate') and tv.value is not None and np.isfinite(tv.value):
                    solved = True
                    break
            except Exception:
                continue
        if not solved:
            return [], []
        h_rel = np.array(hv.value, dtype=np.float64)
        X_rel = np.array(Xv.value, dtype=np.float64)

        if not (np.all(np.isfinite(h_rel)) and np.all(np.isfinite(X_rel))):
            return [], []

        V = X_rel - np.outer(h_rel, h_rel)
        V = 0.5 * (V + V.T)
        Q = np.eye(n) - np.ones((n, n)) / n
        V = Q @ V @ Q
        V = 0.5 * (V + V.T)
        eigvals, eigvecs = np.linalg.eigh(V)
        max_ev = max(1.0, float(np.max(np.abs(eigvals))))
        if float(np.min(eigvals)) > -1e-3 * max_ev:
            ev = np.clip(eigvals, 0, None)
            dmax = float(np.max(np.diag(V)))

            top = np.argsort(ev)[-3:][::-1]

                    cands.append(project_capped(h_rel - 0.3 * v, n))
