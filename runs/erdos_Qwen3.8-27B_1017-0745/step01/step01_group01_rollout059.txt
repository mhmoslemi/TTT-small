    deltas = [0.002, 0.007, 0.02, 0.06, 0.15, 0.4, 0.8]
    for _ in range(sweeps):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                lo_d = max(-h[i], h[j] - 1.0)
                hi_d = min(1.0 - h[i], h[j])
                if hi_d < lo_d - 1e-14:
                    continue
                best_sc = sc
                best_d = 0.0
                for d in deltas:
                    for dd in (d, -d):
                        trial = min(max(dd, lo_d), hi_d)
                        if abs(trial) < 1e-14:
                            continue
                        h[i] += trial
                        h[j] -= trial
                        s_t = exact_score(h, n)
                        h[i] -= trial
                        h[j] += trial
                        if s_t < best_sc - 1e-12:
                            best_sc = s_t
                            best_d = trial
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    sc = best_sc
                    improved = True
        if not improved:
            break
    return h, sc

def add_start(starts, v, n):
    hp = project_capped(v, n)
    if hp is None:
        return
    for ex in starts:
        if float(np.max(np.abs(hp - ex))) < 1e-8:
            return
    starts.append(hp)

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if len(iv) > 0 and len(iv) <= 999:
            n = len(iv)
            if np.all(np.isfinite(iv)) and np.all(iv >= -1e-8) and np.all(iv <= 1.0 + 1e-8):
                parent = iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
    n = max(1, min(n, 999))
    try:
        sd = int(seed)
    except Exception:
        sd = 42
    if sd < 0:
        sd = 42
    rng = np.random.default_rng(sd)
    uniform = np.full(n, 0.5)
    if parent is not None:
        h0 = project_capped(parent, n)
        fallback = h0 if h0 is not None else uniform.copy()
    else:
        fallback = uniform.copy()
    fb_score = exact_score(fallback, n)
    if not np.isfinite(fb_score):
        fallback = uniform.copy()
        fb_score = exact_score(fallback, n)
    starts = []
    add_start(starts, fallback, n)
    add_start(starts, fallback[::-1], n)
    add_start(starts, 1.0 - fallback[::-1], n)
    add_start(starts, 1.0 - fallback, n)
    add_start(starts, uniform, n)
    for nsd in (0.02, 0.06, 0.15):
        add_start(starts, fallback + rng.normal(0, nsd, n), n)
    add_start(starts, rng.uniform(0, 1, n), n)
    idx = np.arange(n)
    add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * idx / n), n)
    if n > 1:
        add_start(starts, 0.5 + 0.45 * np.cos(4.0 * np.pi * idx / n), n)
    if len(starts) > 10:
        starts = starts[:10]
    best = fallback.copy()
    best_score = fb_score
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_score - 1e-12:
            best, best_score = st.copy(), ss
    if n <= 256:
        max_iter = kwargs.get('max_iter', 300)
        try:
            max_iter = max(1, min(int(max_iter), 500))
        except Exception:
            max_iter = 300
        ftol = kwargs.get('ftol', 1e-11)
        try:
            ftol = max(1e-12, min(float(ftol), 1e-7))
        except Exception:
            ftol = 1e-11
        for st in starts:
            res_h = do_slsqp(st, n, max_iter, ftol)
            if res_h is not None:
                sc = exact_score(res_h, n)
                if np.isfinite(sc) and sc < best_score - 1e-12:
                    best, best_score = res_h.copy(), sc
        res_h2 = do_slsqp(best, n, max_iter, ftol * 0.1)
        if res_h2 is not None:
            sc2 = exact_score(res_h2, n)
            if np.isfinite(sc2) and sc2 < best_score - 1e-12:
                best, best_score = res_h2.copy(), sc2
    if n <= 80:
        n_sweeps = kwargs.get('polish_sweeps', 3)
        try:
            n_sweeps = max(0, min(int(n_sweeps), 6))
        except Exception:
            n_sweeps = 3
        if n_sweeps > 0:
            hp, hp_sc = polish(best, n, n_sweeps)
            if np.isfinite(hp_sc) and hp_sc < best_score - 1e-12:
                best, best_score = hp, hp_sc
            hp2 = project_capped(hp, n)
            if hp2 is not None:
                sc_hp2 = exact_score(hp2, n)
                if np.isfinite(sc_hp2) and sc_hp2 < best_score - 1e-12:
                    best, best_score = hp2, sc_hp2
    final = project_capped(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final, fs = fallback.copy(), fb_score
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final, float(fs), int(n)
