            if iv_arr.ndim == 1 and 8 <= iv_arr.size <= 999:
                n = iv_arr.size
                if np.all(np.isfinite(iv_arr)) and np.max(np.abs(iv_arr)) <= 1e12:
                    S_tmp = n / 2.0
                    proj = project_capped_simplex(iv_arr, S_tmp)
                    if proj is not None and is_feasible(proj, n):
                        parent = proj
        except Exception:
            pass
    if n is None:
        n = kwargs.get('n_points', 80)
        try:
            n = int(n)
        except (TypeError, ValueError):
            n = 80
        if n < 8 or n > 999:
            n = 80
    S = n / 2.0
    h_const = np.full(n, 0.5)
    best_h = h_const.copy()
    best_score = exact_score(h_const, n)
    if best_score is None or not np.isfinite(best_score):
        best_score = 0.5
    if parent is not None:
        ps = exact_score(parent, n)
        if ps is not None and np.isfinite(ps) and ps < best_score:
            best_score = ps
            best_h = parent.copy()
    if n <= 128:
        D = min(n - 1, 40)
        gen_cap = 80
        max_anchors = 4
        n_restarts = 2
    elif n <= 256:
        D = min(n - 1, 32)
        gen_cap = 45
        max_anchors = 3
        n_restarts = 1
    elif n <= 512:
        D = min(n - 1, 20)
        gen_cap = 18
        max_anchors = 2
        n_restarts = 0
    else:
        D = min(n - 1, 12)
        gen_cap = 6
        max_anchors = 1
        n_restarts = 0
    patience = max(12, gen_cap // 4)
    basis = build_dct_basis(n, D)
    if basis is None:
        return best_h, best_score, n
    anchors = []
    if parent is not None:
        anchors.append(parent.copy())
        pert = parent + rng.normal(0, 0.08, n)
        pp = project_capped_simplex(pert, S)
        if pp is not None and is_feasible(pp, n):
            anchors.append(pp)
        anchors.append(parent[::-1].copy())
        comp = 1.0 - parent
        cp = project_capped_simplex(comp, S)
        if cp is not None and is_feasible(cp, n):
            anchors.append(cp)
    else:
        anchors.append(h_const.copy())
        z_rand = rng.normal(0, 0.12, D)
        lf = project_capped_simplex(h_const + basis @ z_rand, S)
        if lf is not None and is_feasible(lf, n):
            anchors.append(lf)
    deduped = []
    for a in anchors:
        if len(deduped) >= max_anchors:
            break
        dup = False
        for d in deduped:
            if np.max(np.abs(a - d)) < 1e-9:
                dup = True
                break
        if not dup and is_feasible(a, n):
            deduped.append(a)
    for a in deduped:
        sc = exact_score(a, n)
        if sc is not None and sc < best_score:
            best_score = sc
            best_h = a.copy()
    for a in deduped:
        res_h, res_s = cma_es_run(a, basis, n, S, rng, gen_cap, patience)
        if res_h is not None and res_s is not None and np.isfinite(res_s) and res_s < best_score:
            best_score = res_s
            best_h = res_h.copy()
    for _ in range(n_restarts):
        res_h, res_s = cma_es_run(best_h, basis, n, S, rng, gen_cap, patience)
        if res_h is not None and res_s is not None and np.isfinite(res_s) and res_s < best_score:
            best_score = res_s
            best_h = res_h.copy()
    final_h = best_h
    if not is_feasible(final_h, n):
        final_h = project_capped_simplex(final_h, S)
    if final_h is None or not is_feasible(final_h, n):
        final_h = parent if parent is not None else h_const
    final_score = exact_score(final_h, n)
    if final_score is None or not np.isfinite(final_score):
        final_h = h_const
        final_score = exact_score(h_const, n)
        if final_score is None:
            final_score = 0.5
    return final_h.astype(np.float64), float(final_score), n
