def project_capped(z, n):
    z = np.clip(np.asarray(z, dtype=np.float64), -1e12, 1e12)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for i in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
        else:
            a = min(-r, h[i])
            h[i] -= a
            r += a
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    if abs(np.sum(h) - s) > 1e-10 * max(1, n):
        return None
    return h

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n))

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')

    if q.shape[0] != 2 * n - 1 or not np.all(np.isfinite(q)):
        return float('inf')
    return float(np.max(q) * (2.0 / n))

def lag_grad(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        p0 = max(d, 0)
        q0 = max(-d, 0)
        m = n - abs(d)
        G[row, p0:p0 + m] += dx * (1.0 - h[q0:q0 + m])
        G[row, q0:q0 + m] -= dx * h[p0:p0 + m]
    return G

    return z[n]

def _obj_jac(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def _ineq(z, n):
    h = z[:n]
    q = np.correlate(h, 1.0 - h, mode='full')
    return z[n] - q * (2.0 / n)

def _ineq_jac(z, n):
    h = z[:n]
    G = lag_grad(h, n)
    jac = np.zeros((2 * n - 1, n + 1))
    jac[:, :n] = -G
    jac[:, n] = 1.0
    return jac

def _eq(z, n):
    return np.sum(z[:n]) - n / 2.0

def _eq_jac(z, n):

    g[n] = 0.0
    return g

def solve_epigraph(h0, n, maxiter=250):
    q0 = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = float(np.max(q0) * (2.0 / n)) + 1e-6
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    cons = [
        {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac, 'args': (n,)},
        {'type': 'eq', 'fun': _eq, 'jac': _eq_jac, 'args': (n,)},
