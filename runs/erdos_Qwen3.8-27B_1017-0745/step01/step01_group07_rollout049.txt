```
def exact_score(h, n):
    h = np.asarray(h,dtype=np.float64)
    if h.ndim !=1 or len(h)!=n or not finite: return None
    vals = np.correlate(h, 1.0-h, mode='full') * (2.0/n)
    if vals.size != 2*n-1 or not finite: return None
    return float(np.max(vals))
```

```
def cma_run(anchor, B, n, S, rng, gcap, pat, sig0, bh, bs):
    D = B.shape[1]
    if D <= 0: return bh, bs
    P = 4 + int(3.0 * math.log(D))
    mu = P // 2
    if mu < 1: mu = 1
    w = np.zeros(mu)
    for i in range(mu):
        w[i] = math.log(mu + 0.5) - math.log(i + 1)
    w /= np.sum(w)
    mueff = 1.0 / float(np.dot(w, w))
    c_sig = (mueff + 2.0) / (D + mueff + 5.0)
    d_sig = 1.0 + 2.0 * max(0.0, math.sqrt((mueff - 1.0) / (D + 1.0)) - 1.0) + c_sig
    c_c = (4.0 + mueff / D) / (D + 4.0 + 2.0 * mueff / D)
    c1 = 2.0 / ((D + 1.3) ** 2 + mueff)
    cmu = min(1.0 - c1, 2.0 * (mueff - 2.0 + 1.0 / mueff) / ((D + 2.0) ** 2 + mueff))
    chi = math.sqrt(D) * (1.0 - 1.0 / (4.0 * D) + 1.0 / (21.0 * D * D))
    m = np.zeros(D); C = np.eye(D); p_sigma = np.zeros(D); p_c = np.zeros(D)
    sigma = min(1.5, max(1e-3, sig0))
    rb_h = anchor.copy(); rb_s = exact_score(anchor, n)
    if rb_s is None or not np.isfinite(rb_s): rb_s = bs
    noimp = 0; lowst = 0
    for g in range(gcap):
        C = 0.5 * (C + C.T)
        try:
            ev, V = np.linalg.eigh(C)
        except Exception:
            break
        if not np.all(np.isfinite(ev)) or not np.all(np.isfinite(V)): break
        ev = np.clip(ev, 1e-8, 1e8)
        se = np.sqrt(ev); ie = 1.0 / se
        A = V * se  # rows? V shape DxD, se D. broadcasting V * se multiplies columns? In numpy, V*(se) where se shape (D,) broadcasts to last dimension, so columns scaled. good.
        Cih = (V * ie) @ V.T
        scores = np.full(P, np.inf)
        zs = [None]*P; stps=[None]*P
        gen_best_s = rb_s
        for p in range(P):
            r = rng.standard_normal(D)
            st = A @ r
            z = m + sigma * st
            y = anchor + B @ z
            h = project_capped_simplex(y, S)
            if h is None: continue
            sc = exact_score(h, n)
            if sc is None or not np.isfinite(sc): continue
            scores[p] = sc; zs[p] = z; stps[p] = st
            if sc + 1e-12 < bs:
                bs = sc; bh = h.copy()
            if sc + 1e-12 < rb_s:
                rb_s = sc; rb_h = h.copy()
        if rb_s <= gen_best_s + 1e-12:

        else:

        deco.sort()
        order = [i for _, i in deco[:mu]]
        Z = np.array([zs[i] for i in order])
        ST = np.array([stps[i] for i in order])
        m_new = np.sum(w[:, None] * Z, axis=0)

        ps_new = (1.0 - c_sig) * p_sigma + math.sqrt(c_sig * (2.0 - c_sig) * mueff) * (Cih @ y_w)
        den = math.sqrt(max(1e-12, 1.0 - (1.0 - c_sig) ** (2 * (g + 1))))
```

```
        else: props = [int(np.round(float(L + 1 + i * (n - 1 - (L + 1)) / (R - 1)))) for i in range(R)]
```

```
def build_dct_basis(n, D):
    D = int(D)
    if D <= 0 or D > n - 1:
        return None, 0
    L = (D + 1) // 2
    K = [int(k) for k in range(1, L + 1)]
    R = D - L
    if R > 0:
        if R == 1:
            props = [n - 1]
        else:
            a = float(L + 1)
            b = float(n - 1)
            props = []
            for i in range(R):
                props.append(int(np.round(a + (b - a) * i / (R - 1))))
        for k in props:
            if 1 <= k <= n - 1 and k not in K:
                K.append(k)
    for k in range(L + 1, n):
        if len(K) >= D:
            break
        if k not in K:
            K.append(k)
    K = sorted(K[:D])
    idx = np.arange(n, dtype=np.float64)
    B = np.empty((n, D), dtype=np.float64)
    for j, k in enumerate(K):
        B[:, j] = math.sqrt(2.0 / n) * np.cos(math.pi * k * (idx + 0.5) / n)
    B -= np.mean(B, axis=0)
    try:
        Q, _ = np.linalg.qr(B)
        Q, _ = np.linalg.qr(Q)
    except Exception:
        return None, 0
    if Q.shape != (n, D) or not np.all(np.isfinite(Q)):
        return None, 0
    return Q, D
```

```
def active_lags(h, n, dx, max_act=20):
    ds = []
    qs = []
    maxq = -1e100
    for d in range(-(n - 1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        if hi <= lo: continue
        q = dx * float(np.dot(h[lo:hi], 1.0 - h[lo + d:hi + d]))
        ds.append(d)
        qs.append(q)
        if q > maxq: maxq = q
    if not ds: return ds
    tol = max(1e-10, 1e-8 * maxq)
    sel = []
    for i in range(len(ds)):
        if qs[i] >= maxq - tol:
            sel.append((qs[i], i))
    sel.sort(reverse=True)
    m = max_act if max_act < len(sel) else len(sel)
    top = [sel[i][1] for i in range(m)]
    return [ds[i] for i in top]
```

```
def pair_gradient(h, n, ds, dx):
    g = np.zeros(n, dtype=np.float64)
    if not ds:
        return g
    for d in ds:
        lo = max(0, -d)
        hi = min(n, n - d)
        if hi <= lo: continue
        g[lo:hi] += 1.0 - h[lo + d:hi + d]
        g[lo + d:hi + d] -= h[lo:hi]
    g /= float(len(ds))
    g *= dx
    return g
```

```
def cma_run(anchor,B,n,S,rng,gcap,pat,sig0,bh,bs):
    D=B.shape[1]
    if D<=0: return bh,bs
    P=4+int(3*math.log(D)); mu=P//2; m_w=np.zeros(mu)
    for i in range(mu): m_w[i]=math.log(mu+0.5)-math.log(i+1)
    m_w/=m_w.sum(); me=1/np.dot(m_w,m_w)
    cs=(me+2)/(D+me+5); dsig=1+2*max(0,math.sqrt((me-1)/(D+1))-1)+cs
    cc=(4+me/D)/(D+4+2*me/D); c1v=2/((D+1.3)**2+me)
    cmu=max(0,min(1-c1v,2*(me-2+1/me)/((D+2)**2+me)))
    chi=math.sqrt(D)*(1-1/(4*D)+1/(21*D*D))
    m=np.zeros(D); C=np.eye(D); ps=np.zeros(D); pc=np.zeros(D)
    sigma=min(1.5,max(1e-3,sig0)); rbs=exact_score(anchor,n); rbh=anchor.copy()
    if rbs is None or not np.isfinite(rbs): rbs=bs
    no=0; low=0
    for g in range(gcap):
       C=.5*(C+C.T)
       try: ev,V=np.linalg.eigh(C)
       except Exception: break

       ev=np.clip(ev,1e-8,1e8); se=np.sqrt(ev); ie=1/se

       scores=np.full(P,np.inf); zs=[None]*P; stps=[None]*P; oldrbs=rbs

       den=math.sqrt(max(1e-12,1-(1-cs)**(2*(g+1))))
       hs=1.0 if np.linalg.norm(psn)/den < (1.4+2/(D+1))*chi else 0.0
       pcn=(1-cc)*pc+hs*math.sqrt(cc*(2-cc)*me)*yw
```

import numpy as np
import math

def project_capped_simplex(y, S):
    y = np.asarray(y, dtype=np.float64)
    if y.ndim != 1 or y.size < 1 or not np.all(np.isfinite(y)) or S < 0 or S > y.size or np.max(np.abs(y)) > 1e12:
        return None
    n = y.size
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(3):
        r = S - float(np.sum(x))
        if abs(r) <= 1e-13:
            break
        if r > 0.0:
            for i in range(n):
                if r <= 0.0:
                    break
                a = min(r, 1.0 - x[i])
                if a > 0.0:
                    x[i] += a
                    r -= a
        else:
            for i in range(n):
                if r >= 0.0:
                    break
                a = min(-r, x[i])
                if a > 0.0:
                    x[i] -= a
                    r += a
    x = np.clip(x, 0.0, 1.0)
    if abs(float(np.sum(x)) - S) > 1e-10 or not np.all(np.isfinite(x)):
        return None
    return x

def make_feasible(h, n):
    if h is None:
        return None
    a = np.asarray(h, dtype=np.float64)
    if a.ndim != 1 or a.size != n or not np.all(np.isfinite(a)):
        return None
    a = np.clip(a, 0.0, 1.0)
    S = n / 2.0
    if abs(float(np.sum(a)) - S) > 1e-12:
        a = project_capped_simplex(a, S)
    if a is None:
        return None
    a = np.clip(a, 0.0, 1.0)
    if abs(float(np.sum(a)) - S) > 1e-10:
        return None
    return a

def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64)
    if h.ndim != 1 or h.size != n or not np.all(np.isfinite(h)):
        return None
    v = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if v.size != 2 * n - 1 or not np.all(np.isfinite(v)):
        return None
    return float(np.max(v))

def build_dct_basis(n, D):
    D = int(D)
    if D <= 0 or D > n - 1:
        return None, 0
    L = (D + 1) // 2
    K = list(range(1, L + 1))
    R = D - L
    if R > 0:
        if R == 1:
            props = [n - 1]
        else:
            props = [int(round(L + 1 + i * (n - 1 - L - 1) / (R - 1))) for i in range(R)]
        for k in props:
            if 1 <= k <= n - 1 and k not in K:
                K.append(k)
    for k in range(L + 1, n):
        if len(K) >= D:
            break
        if k not in K:
            K.append(k)
    K = sorted(K[:D])
    idx = np.arange(n, dtype=np.float64)
    B = np.empty((n, D), dtype=np.float64)
    for j, k in enumerate(K):
        B[:, j] = math.sqrt(2.0 / n) * np.cos(math.pi * k * (idx + 0.5) / n)
    B -= np.mean(B, axis=0)
    try:
        Q, _ = np.linalg.qr(B)
        Q, _ = np.linalg.qr(Q)
    except Exception:
        return None, 0
    if Q.shape != (n, D) or not np.all(np.isfinite(Q)):
        return None, 0
    return Q, D

def add_anchor(anchors, h, n):
    h = make_feasible(h, n)
    if h is None:
        return
    for a in anchors:
        if np.max(np.abs(h - a)) < 1e-9:
            return
    anchors.append(h)

def active_lags(h, n, dx, ma=20):
    ds = []
    qs = []
    oh = 1.0 - h
    mq = -1e100
    for d in range(-(n - 1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        if hi <= lo:
            continue
        q = dx * float(np.dot(h[lo:hi], oh[lo + d:hi + d]))
        ds.append(d)
        qs.append(q)
        if q > mq:
            mq = q
    if not ds:
        return ds
    tol = max(1e-10, 1e-8 * mq)
    sel = [(qs[i], i) for i in range(len(qs)) if qs[i] >= mq - tol]
    sel.sort(reverse=True)
    m = ma if ma < len(sel) else len(sel)
    return [ds[sel[i][1]] for i in range(m)]

def pair_gradient(h, n, ds, dx):
    g = np.zeros(n, dtype=np.float64)
    if not ds:
        return g
    for d in ds:
        lo = max(0, -d)
        hi = min(n, n - d)
        if hi <= lo:
            continue
        g[lo:hi] += 1.0 - h[lo + d:hi + d]
        g[lo + d:hi + d] -= h[lo:hi]
    g /= float(len(ds))
    g *= dx
    return g

def subgradient_polish(h, n, S, dx, rng, iters=18):
    h0 = make_feasible(h, n)
    if h0 is None:
        return h, None
    cur = h0.copy()
    best = h0.copy()
    bs = exact_score(best, n)
    if bs is None or not np.isfinite(bs):
        return best, 0.5
    no = 0
    factors = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.12]
    for _ in range(iters):
        ds = active_lags(cur, n, dx, 20)
        sets = []
        if ds:
            sets.append(ds[:1])
        if len(ds) > 2:
            sets.append(ds[:5])
        if ds:
            sets.append(ds)
        imp = False
        for aset in sets:
            g = pair_gradient(cur, n, aset, dx)
            if not np.all(np.isfinite(g)):
                continue
            for f in factors:
                nh = project_capped_simplex(cur - f * g, S)
                if nh is None:
                    continue
                sc = exact_score(nh, n)
                if sc is not None and sc + 1e-12 < bs:
                    best = nh.copy()
                    bs = sc
                    cur = nh.copy()
                    imp = True
                    break
            if imp:
                break
        if not imp:
            nh = project_capped_simplex(cur + rng.normal(0.0, 0.05, n), S)
            if nh is not None:
                sc = exact_score(nh, n)
                if sc is not None and sc + 1e-12 < bs:
                    best = nh.copy()
                    bs = sc
                    cur = nh.copy()
                    imp = True
        no = 0 if imp else no + 1
        if no >= 4:
            break
    return best, bs

def cma_run(anchor, B, n, S, rng, gcap, pat, sig0, bh, bs):
    D = B.shape[1]
    if D <= 0:
        return bh, bs
    P = 4 + int(3 * math.log(D))
    mu = P // 2
    if mu < 1:
        mu = 1
    w = np.zeros(mu, dtype=np.float64)
    for i in range(mu):
        w[i] = math.log(mu + 0.5) - math.log(i + 1)
    w /= float(np.sum(w))
    me = 1.0 / float(np.dot(w, w))
    cs = (me + 2.0) / (D + me + 5.0)
    dsig = 1.0 + 2.0 * max(0.0, math.sqrt((me - 1.0) / (D + 1.0)) - 1.0) + cs
    cc = (4.0 + me / D) / (D + 4.0 + 2.0 * me / D)
    c1v = 2.0 / ((D + 1.3) ** 2 + me)
    cmu = max(0.0, min(1.0 - c1v, 2.0 * (me - 2.0 + 1.0 / me) / ((D + 2.0) ** 2 + me)))
    chi = math.sqrt(D) * (1.0 - 1.0 / (4.0 * D) + 1.0 / (21.0 * D * D))
    mean = np.zeros(D, dtype=np.float64)
    C = np.eye(D, dtype=np.float64)
    ps = np.zeros(D, dtype=np.float64)
    pc = np.zeros(D, dtype=np.float64)
    sigma = min(1.5, max(1e-3, float(sig0)))
    rbs = exact_score(anchor, n)
    if rbs is None or not np.isfinite(rbs):
        rbs = bs
    no = 0
    low = 0
    for gen in range(gcap):
        C = 0.5 * (C + C.T)
        try:
            ev, V = np.linalg.eigh(C)
        except Exception:
            break
        if not np.all(np.isfinite(ev)) or not np.all(np.isfinite(V)):
            break
        ev = np.clip(ev, 1e-8, 1e8)
        se = np.sqrt(ev)
        ie = 1.0 / se
        Cm = V * se
        Cih = (V * ie) @ V.T
        scores = np.full(P, np.inf)
        zs = [None] * P
        stps = [None] * P
        oldrbs = rbs
        for p in range(P):
            r = rng.standard_normal(D)
            st = Cm @ r
            z = mean + sigma * st
            hh = project_capped_simplex(anchor + B @ z, S)
            if hh is None:
                continue
            sc = exact_score(hh, n)
            if sc is None or not np.isfinite(sc):
                continue
            scores[p] = sc
            zs[p] = z
            stps[p] = st
            if sc + 1e-12 < bs:
                bs = sc
                bh = hh.copy()
            if sc + 1e-12 < rbs:
                rbs = sc
        if rbs <= oldrbs + 1e-12:
            no += 1
        else:
            no = 0
        valid = [i for i in range(P) if scores[i] < np.inf]
        if len(valid) < mu or no >= pat:
            break
        deco = [(scores[i], i) for i in valid]
        deco.sort()
        order = [i for _, i in deco[:mu]]
        Z = np.array([zs[i] for i in order], dtype=np.float64)
        ST = np.array([stps[i] for i in order], dtype=np.float64)
        new_mean = np.sum(w[:, None] * Z, axis=0)
        yw = (new_mean - mean) / sigma
        psn = (1.0 - cs) * ps + math.sqrt(cs * (2.0 - cs) * me) * (Cih @ yw)
        den = math.sqrt(max(1e-12, 1.0 - (1.0 - cs) ** (2 * (gen + 1))))
        hs = 1.0 if np.linalg.norm(psn) / den < (1.4 + 2.0 / (D + 1)) * chi else 0.0
        pcn = (1.0 - cc) * pc + hs * math.sqrt(cc * (2.0 - cc) * me) * yw
        Cn = (1.0 + c1v * (2.0 - hs - cc) * cc - cmu) * C
        Cn += c1v * np.outer(pcn, pcn)
        Cn += c1v * (1.0 - hs) * cc * (2.0 - cc) * C
        for i in range(mu):
            Cn += cmu * w[i] * np.outer(ST[i], ST[i])
        Cn = 0.5 * (Cn + Cn.T)
        if not np.all(np.isfinite(Cn)) or not np.all(np.isfinite(new_mean)):
            break
        expv = np.clip((cs / dsig) * (np.linalg.norm(psn) / chi - 1.0), -1.0, 1.0)
        sigma = min(1.5, max(1e-3, sigma * math.exp(float(expv))))
        mean = new_mean
        ps = psn
        pc = pcn
        C = Cn
        if sigma <= 1.05e-3:
            low += 1
            if low >= 8:
                break
        else:
            low = 0
    return bh, bs

def active_lags(h,n,dx,ma=20):
    ds=[]; qs=[]; oh=1.0-h; mq=-1e100
    for d in range(-(n-1),n):
       lo=max(0,-d); hi=min(n,n-d)
       if hi>lo:
          q=dx*float(np.dot(h[lo:hi],oh[lo+d:hi+d])); ds.append(d); qs.append(q)
          if q>mq: mq=q
    if not ds: return ds
    tol=max(1e-10,1e-8*mq)
    sel=[(qs[i],i) for i in range(len(qs)) if qs[i]>=mq-tol]
    sel.sort(reverse=True)
    m=ma if ma<len(sel) else len(sel)
    return [ds[sel[i][1]] for i in range(m)]

def pair_gradient(h,n,ds,dx):
    g=np.zeros(n)
    for d in ds:
       lo=max(0,-d); hi=min(n,n-d)
       if hi>lo:
          g[lo:hi]+=1.0-h[lo+d:hi+d]
          g[lo+d:hi+d]-=h[lo:hi]
    if ds: g*=dx/float(len(ds))
    return g

def build_dct_basis(n,D):
    D=int(D)
    if D<=0 or D>n-1: return None,0
    L=(D+1)//2; K=list(range(1,L+1)); R=D-L
    if R>0:
       props=[n-1] if R==1 else [int(round(L+1+i*(n-1-L-1)/(R-1))) for i in range(R)]
       for k in props:
          if 1<=k<=n-1 and k not in K: K.append(k)
    for k in range(L+1,n):
       if len(K)>=D: break
       if k not in K: K.append(k)
    K=sorted(K[:D]); idx=np.arange(n,dtype=np.float64); B=np.empty((n,D))
    for j,k in enumerate(K): B[:,j]=math.sqrt(2.0/n)*np.cos(math.pi*k*(idx+0.5)/n)
    B-=B.mean(axis=0)
    try: Q,_=np.linalg.qr(B); Q,_=np.linalg.qr(Q)
    except Exception: return None,0
    if Q.shape!==(n,D) or not np.all(np.isfinite(Q)): return None,0
    return Q,D

```python
import numpy as np
import math

def project_capped_simplex(y, S):
    y = np.asarray(y, dtype=np.float64)
    if y.ndim != 1 or y.size < 1 or not np.all(np.isfinite(y)) or S < 0 or S > y.size or np.max(np.abs(y)) > 1e12:
        return None
    n = y.size
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(8):
        r = S - x.sum()
        if abs(r) <= 1e-13:
            break
        x = np.clip(x + r / n, 0.0, 1.0)
    r = S - x.sum()
    if abs(r) <= 1e-7:
        if r > 0.0:
            i = int(np.argmax(1.0 - x))
            x[i] = min(1.0, x[i] + r)
        else:
            i = int(np.argmax(x))
            x[i] = max(0.0, x[i] + r)
    x = np.clip(x, 0.0, 1.0)
    return x if abs(float(x.sum()) - S) <= 1e-10 else None

def make_feasible(h, n):
    if h is None:
        return None
    a = np.asarray(h, dtype=np.float64)
    if a.ndim != 1 or a.size != n or not np.all(np.isfinite(a)):
        return None
    a = np.clip(a, 0.0, 1.0)
    S = n / 2.0
    if abs(float(np.sum(a)) - S) > 1e-12:
        a = project_capped_simplex(a, S)
    if a is None:
        return None
    a = np.clip(a, 0.0, 1.0)
    return a if abs(float(np.sum(a)) - S) <= 1e-10 else None

def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64)
    if h.ndim != 1 or h.size != n or not np.all(np.isfinite(h)):
        return None
    v = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if v.size != 2 * n - 1 or not np.all(np.isfinite(v)):
        return None
    return float(np.max(v))

def build_dct_basis(n, D):
    D = int(D)
    if D <= 0 or D > n - 1:
        return None, 0
    L = (D + 1) // 2
    K = list(range(1, L + 1))
    R = D - L
    if R > 0:
        props = [n - 1] if R == 1 else [int(round(L + 1 + i * (n - 1 - L - 1) / (R - 1))) for i in range(R)]
        for k in props:
            if 1 <= k <= n - 1 and k not in K:
                K.append(k)
    for k in range(L + 1, n):
        if len(K) >= D:
            break
        if k not in K:
            K.append(k)
    K = sorted(K[:D])
    idx = np.arange(n, dtype=np.float64)
    B = np.empty((n, D), dtype=np.float64)
    for j, k in enumerate(K):
        B[:, j] = math.sqrt(2.0 / n) * np.cos(math.pi * k * (idx + 0.5) / n)
    B -= np.mean(B, axis=0)
    try:
        Q, _ = np.linalg.qr(B)
        Q, _ = np.linalg.qr(Q)
    except Exception:
        return None, 0
    if Q.shape != (n, D) or not np.all(np.isfinite(Q)):
        return None, 0
    return Q, D

def add_anchor(anchors, h, n):
    h = make_feasible(h, n)
    if h is None:
        return
    for a in anchors:
        if np.max(np.abs(h - a)) < 1e-9:
            return
    anchors.append(h)

def active_lags(h, n, dx, ma=20):
    ds = []
    qs = []
    oh = 1.0 - h
    mq = -1e100
    for d in range(-(n - 1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        if hi > lo:
            q = dx * float(np.dot(h[lo:hi], oh[lo + d:hi + d]))
            ds.append(d)
            qs.append(q)
            if q > mq:
                mq = q
    if not ds:
        return ds
    tol = max(1e-10, 1e-8 * mq)
    sel = [(qs[i], i) for i in range(len(qs)) if qs[i] >= mq - tol]
    sel.sort(reverse=True)
    m = ma if ma < len(sel) else len(sel)
    return [ds[sel[i][1]] for i in range(m)]

def pair_gradient(h, n, ds, dx):
    g = np.zeros(n, dtype=np.float64)
    for d in ds:
        lo = max(0, -d)
        hi = min(n, n - d)
        if hi > lo:
            g[lo:hi] += 1.0 - h[lo + d:hi + d]
            g[lo + d:hi + d] -= h[lo:hi]
    if ds:
        g *= dx / float(len(ds))
    return g

def subgradient_polish(h, n, S, dx, rng, iters=18):
    h0 = make_feasible(h, n)
    if h0 is None:
        return h, None
    cur = h0.copy()
    best = h0.copy()
    bs = exact_score(best, n)
    if bs is None or not np.isfinite(bs):
        return best, 0.5
    no = 0
    factors = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.12]
    for _ in range(iters):
        ds = active_lags(cur, n, dx, 20)
        sets = []
        if ds:
            sets.append(ds[:1])
        if len(ds) > 2:
            sets.append(ds[:5])
        if ds:
            sets.append(ds)
        imp = False
        for aset in sets:
            g = pair_gradient(cur, n, aset, dx)
            if not np.all(np.isfinite(g)):
                continue
            for f in factors:
                nh = project_capped_simplex(cur - f * g, S)
                if nh is None:
                    continue
                sc = exact_score(nh, n)
                if sc is not None and sc + 1e-12 < bs:
                    best = nh.copy()
                    bs = sc
                    cur = nh.copy()
                    imp = True
                    break
            if imp:
                break
        if not imp:
            nh = project_capped_simplex(cur + rng.normal(0.0, 0.05, n), S)
            if nh is not None:
                sc = exact_score(nh, n)
                if sc is not None and sc + 1e-12 < bs:
                    best = nh.copy()
                    bs = sc
                    cur = nh.copy()
                    imp = True
        no = 0 if imp else no + 1
        if no >= 4:
            break
    return best, bs

def cma_run(anchor, B, n, S, rng, gcap, pat, sig0, bh, bs):
    D = B.shape[1]
    if D <= 0:
        return bh, bs
    P = 4 + int(3 * math.log(D))
    mu = max(1, P // 2)
    w = np.array([math.log(mu + 0.5) - math.log(i + 1) for i in range(mu)], dtype=np.float64)
    w /= float(np.sum(w))
    me = 1.0 / float(np.dot(w, w))
    cs = (me + 2.0) / (D + me + 5.0)
    dsig = 1.0 + 2.0 * max(0.0, math.sqrt((me - 1.0) / (D + 1.0)) - 1.0) + cs
    cc = (4.0 + me / D) / (D + 4.0 + 2.0 * me / D)
    c1v = 2.0 / ((D + 1.3) ** 2 + me)
    cmu = max(0.0, min(1.0 - c1v, 2.0 * (me - 2.0 + 1.0 / me) / ((D + 2.0) ** 2 + me)))
    chi = math.sqrt(D) * (1.0 - 1.0 / (4.0 * D) + 1.0 / (21.0 * D * D))
    mean = np.zeros(D, dtype=np.float64)
    C = np.eye(D, dtype=np.float64)
    ps = np.zeros(D, dtype=np.float64)
    pc = np.zeros(D, dtype=np.float64)
    sigma = min(1.5, max(1e-3, float(sig0)))
    rbs = exact_score(anchor, n)
    if rbs is None or not np.isfinite(rbs):
        rbs = bs
    no = 0
    low = 0
    for gen in range(gcap):
        C = 0.5 * (C + C.T)
        try:
            ev, V = np.linalg.eigh(C)
        except Exception:
            break
        if not np.all(np.isfinite(ev)) or not np.all(np.isfinite(V)):
            break
        ev = np.clip(ev, 1e-8, 1e8)
        se = np.sqrt(ev)
        ie = 1.0 / se
        Cm = V * se
        Cih = (V * ie) @ V.T
        scores = np.full(P, np.inf)
        zs = [None] * P
        stps = [None] * P
        oldrbs = rbs
        for p in range(P):
            r = rng.standard_normal(D)
            st = Cm @ r
            z = mean + sigma * st
            hh = project_capped_simplex(anchor + B @ z, S)
            if hh is None:
                continue
            sc = exact_score(hh, n)
            if sc is None or not np.isfinite(sc):
                continue
            scores[p] = sc
            zs[p] = z
            stps[p] = st
            if sc + 1e-12 < bs:
                bs = sc
                bh = hh.copy()
            if sc + 1e-12 < rbs:
                rbs = sc
        no = no + 1 if rbs <= oldrbs + 1e-12 else 0
        valid = [i for i in range(P) if scores[i] < np.inf]
        if len(valid) < mu or no >= pat:
            break
        deco = [(scores[i], i) for i in valid]
        deco.sort()
        order = [i for _, i in deco[:mu]]
        Z = np.array([zs[i] for i in order], dtype=np.float64)
        ST = np.array([stps[i] for i in order], dtype=np.float64)
        new_mean = np.sum(w[:, None] * Z, axis=0)
        yw = (new_mean - mean) / sigma
        psn = (1.0 - cs) * ps + math.sqrt(cs * (2.0 - cs) * me) * (Cih @ yw)
        den = math.sqrt(max(1e-12, 1.0 - (1.0 - cs) ** (2 * (gen + 1))))
        hs = 1.0 if np.linalg.norm(psn) / den < (1.4 + 2.0 / (D + 1)) * chi else 0.0
        pcn = (1.0 - cc) * pc + hs * math.sqrt(cc * (2.0 - cc) * me) * yw
        Cn = (1.0 + c1v * (2.0 - hs - cc) * cc - cmu) * C
        Cn += c1v * np.outer(pcn, pcn)
        Cn += c1v * (1.0 - hs) * cc * (2.0 - cc) * C
        for i in range(mu):
            Cn += cmu * w[i] * np.outer(ST[i], ST[i])
        Cn = 0.5 * (Cn + Cn.T)
        if not np.all(np.isfinite(Cn)) or not np.all(np.isfinite(new_mean)):
            break
        sigma = min(1.5, max(1e-3, sigma * math.exp(float(np.clip((cs / dsig) * (np.linalg.norm(psn) / chi - 1.0), -1.0, 1.0)))))
        mean = new_mean
        ps = psn
        pc = pcn
        C = Cn
        if sigma <= 1.05e-3:
            low += 1
            if low >= 8:
                break
        else:
            low = 0
    return bh, bs

        if asp is not None and asp + 1e-12 < bs:
            bs = asp
            bh = ap.copy()
    gc = 80 if n <= 128 else 45 if n <= 256 else 18 if n <= 512 else 6
    pat = max(12, gc // 4)
    if D > 0:
        sig = 0.10 * math.sqrt(n / D)
        sig = min(0.7, max(0.05, sig))
    else:
        sig = 0.1
    for a in anchors[:maxa]:
        if D == 0:
            break
        _, bs = cma_run(a, B, n, S, rng, gc, pat, sig, bh, bs)
        ph, pps = subgradient_polish(bh, n, S, dx, rng, p2)
        if pps is not None and pps + 1e-12 < bs:
            bs = pps
            bh = ph.copy()
    reps = 2 if n <= 128 else 1 if n <= 256 else 0
    fac = 1.0
    for _ in range(reps):
        if D == 0:
            break
        pre = bs
        s = min(1.5, max(0.03, sig * fac))
        _, bs = cma_run(bh, B, n, S, rng, gc, pat, s, bh, bs)
        ph, pps = subgradient_polish(bh, n, S, dx, rng, p2)
        if pps is not None and pps + 1e-12 < bs:
            bs = pps
            bh = ph.copy()
        if bs + 1e-12 < pre:
            fac = max(0.5, 0.65 * fac)
        else:
            fac = min(6.0, 1.6 * fac)
    fh, fs = subgradient_polish(bh, n, S, dx, rng, p3)
    if fs is not None and fs + 1e-12 < bs:
        bs = fs
        bh = fh.copy()
    fh = make_feasible(bh, n)
    if fh is None:
        fh = const
    fs = exact_score(fh, n)
    if fs is None or not np.isfinite(fs):
        if parent is not None:
            fh = make_feasible(parent, n)
            if fh is not None:
                fs = exact_score(fh, n)
        if fs is None or not np.isfinite(fs):
            fh = const
            fs = exact_score(fh, n)
    if fh is None or fs is None or not np.isfinite(fs):
        fh = const
        fs = 0.5
    return fh, float(fs), int(n)
```
