                v = old_pop[r1] + F * (old_pop[r2] - old_pop[r3])
            else:
                candidates = [j for j in range(NP) if j != i and j != b_idx]
                r1, r2 = rng.choice(candidates, size=2, replace=False)
                v = old_pop[i] + F * (old_pop[b_idx] - old_pop[i]) + F * (old_pop[r1] - old_pop[r2])
            mask = rng.random(n) < CR
            if not np.any(mask):
                mask[rng.integers(0, n)] = True
            trial = np.where(mask, v, old_pop[i])
            proj = project_capped(trial, n)
            if proj is None:
                continue
            if float(np.max(np.abs(proj - old_pop[i]))) <= 1e-12:
                continue
            ts = exact_score(proj, n)
            if np.isfinite(ts) and ts < old_scores[i] - 1e-12:
                pop[i] = proj
                scores[i] = ts
                if ts < novel_best - 1e-12:
                    novel_best = ts
        if novel_best < best_score - 1e-12:
            best_score = novel_best
            bi = int(np.argmin(scores))
            best = pop[bi].copy()
            stagnant = 0
        else:
            stagnant += 1
        if stagnant >= patience and n_restarts < max_restarts:
            n_restarts += 1
            stagnant = 0
            elite_n = max(2, int(np.ceil(0.2 * NP)))
            order = np.argsort(scores)
            new_pop = [best.copy()]
            for ei in range(min(elite_n - 1, NP - 1)):
                p = pop[order[ei]].copy()
                if float(np.max(np.abs(p - best))) > 1e-10:
                    new_pop.append(p)
            sc = [0.03, 0.07, 0.15][min(n_restarts - 1, 2)] * (0.5 + rng.random())
            while len(new_pop) < NP:
                if new_pop:
                    src = best if rng.random() < 0.6 else new_pop[rng.integers(0, len(new_pop))]
                else:
                    src = uniform
                cand = src + rng.normal(0, sc, n)
                proj = project_capped(cand, n)
                if proj is None:
                    proj = project_capped(rng.uniform(0, 1, n), n)
                if proj is None:
                    proj = uniform.copy()
                dup = False
                for q in new_pop:
                    if float(np.max(np.abs(proj - q))) < 1e-10:
                        dup = True
                        break
                new_pop.append(proj if not dup else proj.copy())
            pop = np.array(new_pop, dtype=np.float64)
            scores = np.array([exact_score(pop[i], n) for i in range(NP)])
            NP = len(pop)
            for i in range(NP):
                if np.isfinite(scores[i]) and scores[i] < best_score - 1e-12:
                    best, best_score = pop[i].copy(), float(scores[i])
    candidates = [best]
    if fb_parent is not None:
        candidates.append(fb_parent)
    candidates.append(uniform)
    best_final, best_fs = None, float('inf')
    for c in candidates:
        c2 = project_capped(c, n)
        if c2 is None:
            continue
        sc = exact_score(c2, n)
        if np.isfinite(sc) and sc < best_fs:
            best_final, best_fs = c2, sc
    if best_final is None:
        best_final, best_fs = uniform.copy(), exact_score(uniform, n)
    return best_final, float(best_fs), int(n)
