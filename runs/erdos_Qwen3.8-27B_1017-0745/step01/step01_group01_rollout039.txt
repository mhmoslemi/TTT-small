        if abs(B) <= tol * max(1.0, abs(B), abs(C)):

        return []

    if disc < -tol * (B*B + 4.0*abs(A*C) + 1.0):
        return []
    if disc < 0:
        disc = 0.0
    sqrt_disc = math.sqrt(disc)
    # Numerically stable: q = -0.5*(B + sign(B)*sqrt(disc))
    q = -0.5 * (B + math.copysign(sqrt_disc, B))

    disc = B * B - 4.0 * A * C
    if disc < -tol * (B * B + 4.0 * abs(A * C) + 1.0):
        return []
    disc = max(0.0, disc)
    sq = math.sqrt(disc)
    q = -0.5 * (B + math.copysign(sq, B))
    if abs(q) > tol:
        r1, r2 = q / A, C / q
    else:
        r1 = r2 = -B / (2.0 * A)
    res = []
    for r in (r1, r2):
        if np.isfinite(r) and lo - pad <= r <= hi + pad:
            res.append(max(lo, min(hi, r)))
    return res
