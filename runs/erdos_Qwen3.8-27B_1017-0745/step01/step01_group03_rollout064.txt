    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

            and abs(np.sum(h) - n / 2.0) <= 1e-9)

    if i > j: i, j = j, i

        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - np.sum(y)
    if abs(r) > 1e-14:
        if r > 0:
            for i in np.argsort(1.0 - y):
                a = min(r, 1.0 - y[i]); y[i] += a; r -= a
                if abs(r) < 1e-15: break
        else:
            for i in np.argsort(y):
                a = min(-r, y[i]); y[i] -= a; r += a
                if abs(r) < 1e-15: break
    if np.any(y < -1e-12) or np.any(y > 1 + 1e-12) or abs(np.sum(y) - S) > 1e-9:
        return None
    return np.clip(y, 0, 1)

def escore(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def feas(h, n):
    return (h is not None and h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-9)

def pidx(i, j, n):
    if i > j: i, j = j, i
    return i * (2 * n - i - 1) // 2 + (j - i)

def lag_c(x, d, n):
    nl = 2 * n - 1
    A = np.zeros(nl); B = np.zeros(nl); D = np.zeros(nl)
    for k in range(-(n - 1), n):
        i0 = max(0, k); i1 = min(n - 1, n - 1 + k)
        i = np.arange(i0, i1 + 1); j = i - k
        D[k + n - 1] = np.dot(x[i], 1.0 - x[j])
        B[k + n - 1] = np.dot(d[i], 1.0 - x[j]) - np.dot(x[i], d[j])
        A[k + n - 1] = -np.dot(d[i], d[j])
    return A, B, D
