    best_x = a; best_v = np.max(A*a*a+B*a+D) - (A0*a*a+B0*a+D0)
    for x in (b, m, c, d):
        v = np.max(A*x*x+B*x+D) - (A0*x*x+B0*x+D0)
        if v < best_v:
            best_x, best_v = x, v
    return best_x, best_v

def H_val(t, B, D, A, A0, B0, D0):
    return np.max(A*t*t + B*t + D) - (A0*t*t + B0*t + D0)

def bisect_H(a, b, B, D, A, A0, B0, D0, it=52):
    fa = H_val(a, B, D, A, A0, B0, D0)
    fb = H_val(b, B, D, A, A0, B0, D0)
    if fa * fb > 0:
        return None
    for _ in range(it):
        m = 0.5 * (a + b)
        fm = H_val(m, B, D, A, A0, B0, D0)
        if b - a < 1e-14:
            return m
        if fa * fm <= 0:
            b = m; fb = fm
        else:
            a = m; fa = fm
    return 0.5 * (a + b)

def pair_line_min(h, q, i, j, n):
    dx = 2.0 / n
    r = abs(i - j)
    L = max(-h[i], h[j] - 1.0)
    U = min(1.0 - h[i], h[j])
    if U - L < 1e-13:
        return None, None
    B = np.empty(2 * n - 1)
    for li in range(2 * n - 1):
        d = li - (n - 1)
        B[li] = lag_grad(h, d, i, n, dx) - lag_grad(h, d, j, n, dx)
    D = q.copy()
    A = np.zeros(2 * n - 1)
    A[n - 1] = -2.0 * dx
    if r > 0 and n - 1 + r < 2 * n and n - 1 - r >= 0:
        A[n - 1 + r] = dx; A[n - 1 - r] = dx
    A0, B0, D0 = -2.0 * dx, B[n - 1], D[n - 1]
    AG = A.copy(); BG = B.copy(); DG = D.copy()
    DG[n - 1] = -1e30
    tG, _ = gsec_convex(L, U, BG, DG, AG)
    tH, vH = gsec_H(L, U, BG, DG, AG, A0, B0, D0)
    cands = [L, U, min(max(0.0, L), U), tG, tH]
    if vH < 0:
        if H_val(L, B, D, A, A0, B0, D0) > 0:
            rl = bisect_H(L, tH, B, D, A, A0, B0, D0)
            if rl is not None:
                cands.append(rl)
        if H_val(U, B, D, A, A0, B0, D0) > 0:
            rr = bisect_H(tH, U, B, D, A, A0, B0, D0)
            if rr is not None:
                cands.append(rr)
    best_t, best_w = 0.0, float('inf')
    for t in cands:
        if t < L - 1e-14 or t > U + 1e-14:
            continue
        t = min(max(t, L), U)
        wG = np.max(AG * t * t + BG * t + DG)
        wQ = A0 * t * t + B0 * t + D0
        w = max(wG, wQ)
        if w < best_w - 1e-15 or (abs(w - best_w) <= 1e-15 and abs(t) < abs(best_t)):
            best_w, best_t = w, t
    return best_t, best_w

def sort_pairs_by_merit(merits):
    # merits is list of (value, ia, ib)
    # return sorted by value ascending
    n2 = len(merits)
    idx = list(range(n2))
    for i in range(n2):
        for j in range(i + 1, n2):
            if merits[idx[j]] < merits[idx[i]]:
                idx[i], idx[j] = idx[j], idx[i]
    return [merits[k] for k in idx]

def local_descent(start, n, rng, n_moves, n_screen, n_rand):
    h = project_capped(start.astype(np.float64).copy(), n)
    if h is None:
        return None, float('inf')
    score = exact_score(h, n)
    if not np.isfinite(score):
        return None, float('inf')
    dx = 2.0 / n
    stall = 0
    for _ in range(n_moves):
        q = np.correlate(h, 1.0 - h, mode='full') * dx
        n_lags = 2 * n - 1
        if n_lags > 8:
            top_lags = np.argsort(q)[-8:]
        else:
            top_lags = np.arange(n_lags)
        if n <= 128:
            ii, jj = np.triu_indices(n, k=1)
        else:
            np_ = min(2000, 8 * n)
            ii = rng.integers(0, n, np_); jj = rng.integers(0, n, np_)
            msk = ii < jj
            ii, jj = ii[msk], jj[msk]
        merits = []
        for (ia, ib) in zip(ii, jj):
            Lm = max(-h[ia], h[ib] - 1.0)
            Um = min(1.0 - h[ia], h[ib])
            if Um - Lm < 1e-12:
                continue
            ms = 0.0
            for li in top_lags:
                d = li - (n - 1)
                bd = lag_grad(h, d, ia, n, dx) - lag_grad(h, d, ib, n, dx)
                if Um > 0:
                    ms = max(ms, bd)
                if Lm < 0:
                    ms = max(ms, -bd)
            merits.append((ms, int(ia), int(ib)))
        if not merits:
            break
        merits = sort_pairs_by_merit(merits)
        sel = merits[:n_screen]
        extra = rng.integers(0, len(merits), min(n_rand, len(merits)))
        for e in extra:
            sel.append(merits[e])
        best_trial, best_s = None, score
        for (_, ia, ib) in sel:
            t, _ = pair_line_min(h, q, ia, ib, n)
            if t is None or abs(t) < 1e-13:
                continue
            h2 = h.copy()
            h2[ia] += t; h2[ib] -= t
            h2 = project_capped(h2, n)
            if h2 is None or not is_feasible(h2, n):
                continue
            s2 = exact_score(h2, n)
            if s2 < best_s - 1e-13:
                best_s, best_trial = s2, h2.copy()
        if best_trial is None:
            stall += 1
            if stall >= 3:
                break
        else:
            h, score, stall = best_trial, best_s, 0
    if not is_feasible(h, n):
        return None, float('inf')
    return h, exact_score(h, n)

def build_starts(base, n, rng, count):
    pool = []
    pool.append(base.copy())
    pool.append(project_capped(base[::-1], n))
    pool.append(project_capped(1.0 - base[::-1], n))
    pool.append(np.full(n, 0.5))
    blk = np.zeros(n); blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    pool.append(project_capped(blk, n))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    pool.append(project_capped(alt, n))
    for sd in (0.06, 0.15, 0.30):
        pool.append(project_capped(base + rng.normal(0, sd, n), n))
    pool.append(project_capped(rng.uniform(0, 1, n), n))
    # binary patterns
    for _ in range(3):
        b = np.zeros(n)
        perm = rng.permutation(n)
        b[perm[:n // 2]] = 1.0
        if n % 2:
            b[perm[n // 2]] = 0.5
        pool.append(project_capped(b, n))
    dedup = []
    for v in pool:
        if v is None:
            continue
        dup = False
        for s in dedup:
            if np.max(np.abs(v - s)) < 1e-8:
                dup = True; break
        if not dup:
            dedup.append(v)
        if len(dedup) >= count:
            break
    return dedup

def run(seed=42, **kwargs):
    n = None; parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None:
        try:
            v = int(kwargs.get('n_points', 47))
            n = v if 1 <= v <= 999 else 47
        except Exception:
            n = 47
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    fallback = project_capped(parent, n) if parent is not None else None
    if fallback is None:
        fallback = np.full(n, 0.5)
    if n <= 64:
        ns, nm, nsr, nrd, nh, nhm = 10, 28, 6, 3, 8, 12
    elif n <= 128:
        ns, nm, nsr, nrd, nh, nhm = 7, 18, 5, 2, 5, 8
    elif n <= 256:
        ns, nm, nsr, nrd, nh, nhm = 4, 10, 4, 2, 3, 5
    else:
        ns, nm, nsr, nrd, nh, nhm = 3, 6, 3, 1, 2, 3
    starts = build_starts(fallback, n, rng, ns)
    fb_score = exact_score(fallback, n)
    gb, gs = fallback.copy(), fb_score
    for st in starts:
        s = exact_score(st, n)
        if s < gs - 1e-13:
            gs, gb = s, st.copy()
    for st in starts:
        hx, sx = local_descent(st, n, rng, nm, nsr, nrd)
        if hx is not None and sx < gs - 1e-13:
            gs, gb = sx, hx.copy()
    # basin hopping
    walker, ws = gb.copy(), gs
    noise_sds = (0.04, 0.10, 0.20, 0.30)
    scale = min(1.5, max(0.5, np.sqrt(47.0 / max(1, n))))
    for hop in range(nh):
        sd = noise_sds[hop % 4] * scale
        prop = project_capped(walker + rng.normal(0, sd, n), n)
        if prop is None:
            continue
        hx, hx_s = local_descent(prop, n, rng, nhm, max(3, nsr - 1), max(1, nrd))
        if hx is None:
            continue
        if hx_s < gs - 1e-13:
            gs, gb = hx_s, hx.copy()
            walker, ws = gb.copy(), gs
        else:
            delta = hx_s - ws
            T = 0.002 * (0.25 + 0.75 * (1.0 - hop / max(1, nh)))
            if delta <= 1e-12 or rng.random() < np.exp(-min(700.0, delta / max(T, 1e-10))):
                walker, ws = hx.copy(), hx_s
    h_out = project_capped(gb, n)
    if h_out is None or not is_feasible(h_out, n):
        h_out = fallback
    sc = exact_score(h_out, n)
    if not np.isfinite(sc) or sc > fb_score + 1e-12:
        h_out, sc = fallback, fb_score
    return h_out, float(sc), int(n)
