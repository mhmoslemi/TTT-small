    if x < lo:
        return lo
    if x > hi:
        return hi
    return x

def lag_scores(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    qs = lag_scores(h, n)
    if not np.all(np.isfinite(qs)):
        return float('inf')
    return float(np.max(qs))

def build_grad_matrix(h, n):
    dx = 2.0 / n
    m = 2 * n - 1
    G = np.zeros((m, n), dtype=np.float64)
    for k in range(-(n - 1), n):
        row = k + n - 1
        lo = max(0, k)
        hi = min(n, n + k)
        if hi <= lo:
            continue
        G[row, lo:hi] += dx * (1.0 - h[lo - k:hi - k])
        G[row, lo - k:hi - k] -= dx * h[lo:hi]
    return G

def project_simplex(z):
    m = len(z)
    if m == 0:
        return z
    if not np.all(np.isfinite(z)):
        return np.ones(m) / m
    s = np.sort(z)[::-1]
    cs = np.cumsum(s)
    rho = m - 1
    for i in range(m - 1, -1, -1):
        if (s[i] - (cs[i] - 1.0) / (i + 1.0)) > 0:
            rho = i
            break
    theta = (cs[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(z - theta, 0.0)
    r = w.sum() - 1.0
    if abs(r) > 1e-15:
        pos = w > 0
        if pos.sum() > 0:
            w[pos] += r / pos.sum()
            np.clip(w, 0.0, None, out=w)
    return w

    y = project_capped(h - gmix / L, n)
    d = y - h
    u = q + G @ d
    D_val = float(p @ u) + 0.5 * L * float(d @ d)
    gap = float(np.max(u) - p @ u)
    return d, u, D_val, gap

    Ldual = max(1e-12, 1.1 * float(np.sum(G * G)) / L)

        G[row, lo - k:hi - k] -= dx * h[lo:hi]

            break
