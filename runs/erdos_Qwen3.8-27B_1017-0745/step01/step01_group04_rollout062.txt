    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / N)

def smooth_obj_grad(h, tau, N):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = dx * corr
    cmax = c.max()
    e = np.exp((c - cmax) / tau)
    se = e.sum()
    F = cmax + tau * np.log(se)
    p = e / se
    grad = np.zeros(N)
    for r in range(2 * N - 1):
        s = r - (N - 1)
        w = dx * p[r]
        if w < 1e-16:
            continue
        if s == 0:
            grad += w * (1.0 - 2.0 * h)
        elif s > 0:
            L = N - s
            grad[s:] += w * (1.0 - h[:L])
            grad[:L] -= w * h[s:]
        else:
            d = -s
            L = N - d
            grad[:L] += w * (1.0 - h[d:])
            grad[d:] -= w * h[:L]
    return F, grad

def pg_optimize(h0, N, m, max_iter=50):
    taus = [0.02, 0.006, 0.002, 0.0006, 0.0002, 0.00006, 0.000015]
    h = project_feasible(h0.copy(), m)
    best_h = h.copy()
    best_s = exact_score(h, N)
    for tau in taus:
        alpha = 1.0
        stall = 0
        for _ in range(max_iter):
            F, grad = smooth_obj_grad(h, tau, N)
            if not np.isfinite(F) or not np.all(np.isfinite(grad)):
                break
            y = project_feasible(h - alpha * grad, m)
            d = y - h
            if not np.all(np.isfinite(d)) or np.max(np.abs(d)) < 1e-11:
                break
            sy = exact_score(y, N)
            if sy < best_s:
                best_h = y.copy()
                best_s = sy
            Fy, _ = smooth_obj_grad(y, tau, N)
            gd = np.dot(grad, d)
            if gd < 0 and np.isfinite(Fy) and Fy <= F + 1e-4 * gd + 1e-12:
                h = y.copy()
                stall = 0
                alpha = min(1.25 * alpha, 10.0)
                if abs(Fy - F) < 1e-13 * max(1.0, abs(F)):
                    stall += 1
                    if stall >= 5:
                        break
            else:
                alpha *= 0.5
                if alpha < 1e-10:
                    break
    return best_h, best_s

def make_fourier_start(N, rng, n_modes, amp):
    z = np.full(N, 0.5)
    ns = min(n_modes, (N - 1) // 2)
    idx = np.arange(N)
    for r in range(1, ns + 1):
        A = rng.standard_normal()
        B = rng.standard_normal()
        z += (amp / r) * (A * np.cos(2.0 * np.pi * r * idx / N) + B * np.sin(2.0 * np.pi * r * idx / N))
    return z

def run(seed=42, **kwargs):
    N = None
    m = None
    parent_base = None
    raw = globals().get('initial_h_values', None)
    if raw is not None:
        raw = np.asarray(raw, dtype=np.float64)
        if raw.ndim == 1 and raw.size >= 2 and np.all(np.isfinite(raw)):
            N = raw.size
            m = N / 2.0
            parent_base = project_feasible(raw.copy(), m)
    if N is None:
        N = kwargs.get('n_points', 70)
        if not isinstance(N, (int, np.integer)) or N < 2:
            N = 70
        m = N / 2.0
    if N == 1:
        h = np.array([0.5])
        return h, exact_score(h, 1), 1
    if N > 999:
        h = parent_base if parent_base is not None else np.full(N, 0.5)
        return h, exact_score(h, N), int(N)

    rng = np.random.default_rng(seed)
    best = parent_base.copy() if parent_base is not None else np.full(N, 0.5)
    best = project_feasible(best, m)
    best_score = exact_score(best, N)

    starts = [best.copy(), np.full(N, 0.5), project_feasible(1.0 - best, m)]
    for sc in [0.04, 0.12]:
        starts.append(project_feasible(best + sc * rng.standard_normal(N), m))
    starts.append(project_feasible(0.5 + 0.5 * rng.uniform(size=N), m))
    starts.append(project_feasible(make_fourier_start(N, rng, 6, 0.12), m))
    starts.append(project_feasible(make_fourier_start(N, rng, 4, 0.22), m))

    dedup = [starts[0]]
    for s in starts[1:]:
        if min(np.max(np.abs(s - d)) for d in dedup) > 1e-10:
            dedup.append(s)
    starts = dedup

    if N <= 128:
        max_start, max_iter, n_hops = 8, 50, 2
    elif N <= 256:
        max_start, max_iter, n_hops = 6, 30, 1
    elif N <= 512:
        max_start, max_iter, n_hops = 4, 16, 0
    else:
        max_start, max_iter, n_hops = 3, 8, 0

    for h0 in starts[:max_start]:
        s0 = exact_score(h0, N)
        if s0 < best_score:
            best = h0.copy()
            best_score = s0
        h_opt, s_opt = pg_optimize(h0, N, m, max_iter)
        if s_opt < best_score:
            best = h_opt.copy()
            best_score = s_opt

    hop_scales = [0.04, 0.015]
    for i in range(min(n_hops, len(hop_scales))):
        hop = project_feasible(best + hop_scales[i] * rng.standard_normal(N), m)
        sh = exact_score(hop, N)
        if sh < best_score:
            best = hop.copy()
            best_score = sh
        h_opt, s_opt = pg_optimize(hop, N, m, max_iter)
        if s_opt < best_score:
            best = h_opt.copy()
            best_score = s_opt

    best = project_feasible(best, m)
    c5 = exact_score(best, N)
    if not np.isfinite(c5):
        best = parent_base if parent_base is not None else np.full(N, 0.5)
        c5 = exact_score(best, N)
    if parent_base is not None:
        ps = exact_score(parent_base, N)
        if ps < c5:
            best = parent_base.copy()
            c5 = ps
    return best, c5, int(N)
