def quad_roots(A, B, C, lo, hi):
    tol = 1e-14
    roots = []
    if abs(A) <= tol * max(1.0, abs(B), abs(C)):
        if abs(B) <= tol:
            return roots
        r = -C / B
        if lo - tol*max(1,abs(lo),abs(hi)) <= r <= hi + tol*max(1,abs(lo),abs(hi)):
            roots.append(np.clip(r, lo, hi))
        return roots
    disc = B*B - 4.0*A*C
    if disc < -tol * (B*B + 4.0*abs(A*C) + 1.0):
        return roots
    if disc < 0:
        disc = 0.0
    sd = sqrt(disc)
    # Stable formula
    if B >= 0:
        q = -0.5 * (B + sd)
    else:
        q = -0.5 * (B - sd)
    if abs(q) > tol:
        r1 = q / A
        r2 = C / q
    else:
        r1 = -B / (2.0 * A)
        r2 = r1
    for r in (r1, r2):
        if np.isfinite(r):
            if lo - tol*max(1,abs(lo),abs(hi)) <= r <= hi + tol*max(1,abs(lo),abs(hi)):
                roots.append(np.clip(r, lo, hi))

    return a, b

def solve_quad(A, B, C, lo, hi):
    # Returns list of roots in [lo, hi] for A*x^2 + B*x + C = 0
    roots = []
    tol = 1e-14
    lg = max(1.0, abs(lo), abs(hi))
    if abs(A) <= tol * max(1.0, abs(B), abs(C)):
        if abs(B) > tol:
            r = -C / B
            if lo - tol*lg <= r <= hi + tol*lg:
                roots.append(float(np.clip(r, lo, hi)))
        return roots
    disc = B*B - 4.0*A*C
    if disc < -tol*(B*B + 4.0*abs(A*C) + 1.0):
        return roots
    disc = max(0.0, disc)
    sd = sqrt(disc)
    if B >= 0:
        q = -0.5*(B + sd)
    else:
        q = -0.5*(B - sd)
    cands = []
    if abs(q) > tol:
        cands = [q/A, C/q]
    else:
        cands = [-B/(2.0*A)]
    for r in cands:
        if np.isfinite(r) and lo - tol*lg <= r <= hi + tol*lg:
            roots.append(float(np.clip(r, lo, hi)))
    return roots

def line_minimizer(h, p, q, n, c, a, b):
    lo = max(-h[p], h[q] - 1.0)
    hi = min(1.0 - h[p], h[q])
    if hi - lo <= 1e-14:
        return 0.0
    L = 2 * n - 1
    D = q - p
    # Special rows
    spec_rows = [n-1, n-1+D, n-1-D]
    spec_set = set(spec_rows)
    # Affine rows
    aff_mask = np.ones(L, dtype=bool)
    for r in spec_rows:
        if 0 <= r < L:
            aff_mask[r] = False
    aff_rows = np.where(aff_mask)[0]
    # Build candidate deltas
    candidates = [lo, hi, 0.0]
    # Hull switch points
    m_aff = b[aff_rows]
    c_aff = c[aff_rows]
    n_aff = len(m_aff)
    if n_aff > 0:
        # Sort by slope
        order = np.argsort(m_aff, kind='mergesort')
        sm = m_aff[order]
        sc = c_aff[order]
        # Remove same-slope duplicates (keep max intercept)
        us, uc, um = [], [], []
        for i in range(n_aff):
            if us and sm[i] == us[-1]:
                if sc[i] > uc[-1]:
                    uc[-1] = sc[i]
            else:
                us.append(float(sm[i]))
                uc.append(float(sc[i]))
        # Build upper hull
        hull = []  # (slope, intercept, start_x)
        for i in range(len(us)):
            m_i, c_i = us[i], uc[i]
            while True:
                if not hull:
                    hull.append([m_i, c_i, float('-inf')])
                    break
                m_prev, c_prev, s_prev = hull[-1]
                if m_i == m_prev:
                    break
                x_cross = (c_prev - c_i) / (m_i - m_prev)
                if x_cross <= s_prev + 1e-15:
                    hull.pop()
                else:
                    hull.append([m_i, c_i, x_cross])
                    break
        # Add hull switch points
        for seg in hull:
            if lo <= seg[2] <= hi:
                candidates.append(seg[2])
    # Intersections of special quadratics with affine lines
    for s_row in spec_rows:
        if not (0 <= s_row < L):
            continue
        a_s, b_s, c_s = a[s_row], b[s_row], c[s_row]
        for ar in aff_rows:
            A = a_s
            B = b_s - b[ar]
            C = c_s - c[ar]
            roots = solve_quad(A, B, C, lo, hi)
            candidates.extend(roots)
    # Intersections between special quadratics
    for i in range(3):
        for j in range(i+1, 3):
            r1, r2 = spec_rows[i], spec_rows[j]
            if not (0 <= r1 < L and 0 <= r2 < L):
                continue
            A = a[r1] - a[r2]
            B = b[r1] - b[r2]
            C = c[r1] - c[r2]
            roots = solve_quad(A, B, C, lo, hi)
            candidates.extend(roots)
    # Vertices of convex special quadratics
    for s_row in spec_rows:
        if 0 <= s_row < L and a[s_row] > 0:
            v = -b[s_row] / (2.0 * a[s_row])
            if lo <= v <= hi:
                candidates.append(v)
    # Evaluate envelope at candidates
    cands = np.array([cd for cd in candidates if np.isfinite(cd)], dtype=np.float64)
    cands = cands[(cands >= lo - 1e-14) & (cands <= hi + 1e-14)]
    cands = np.clip(cands, lo, hi)
    if len(cands) == 0:
        return 0.0
    # Vectorized evaluation: for each candidate delta, compute max over all lags
    # chunked to avoid huge temp arrays
    best_val = float('inf')
    best_delta = 0.0
    chunk = 512
    for start in range(0, len(cands), chunk):
        cdf = cands[start:start+chunk]
        # values[i,d] = c[d] + b[d]*cdf[i] + a[d]*cdf[i]^2
        vals = c[None,:] + b[None,:]*cdf[:,None] + a[None,:]*cdf[:,None]**2
        env = np.max(vals, axis=1)
        for i in range(len(cdf)):
            if env[i] < best_val - 1e-15:
                best_val = env[i]
                best_delta = float(cdf[i])
    return best_delta

def add_start(starts, v, n):
    h2 = project_capped(v, n)
    if h2 is None:
        return
    for ex in starts:
        if float(np.max(np.abs(h2 - ex))) < 1e-9:
            return
    starts.append(h2)

def build_edges(n):
    if n <= 1:
        return []
    edges = set()
    for i in range(n-1):
        edges.add((i, i+1))
    r = 2
    while r < n:
        for i in range(0, n, r):
            if i + r < n:
                edges.add((min(i, i+r), max(i, i+r)))
        r *= 2
    if n > 2:
        edges.add((0, n-1))
    return sorted(edges)
