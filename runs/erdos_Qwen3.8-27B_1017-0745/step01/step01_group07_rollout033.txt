Let me carefully analyze the problem and strategy to implement a DC (Difference of Convex) majorization-minimization approach for the Erdős minimum overlap problem.

Key points:
1. We need to find h: [0,2] → [0,1] with ∫h = 1 that minimizes C₅ = max_k ∫h(x)(1-h(x+k))dx
2. Discretized as n_points samples, sum(h)*dx = 1 where dx = 2/n_points
3. Score: C₅ = max(np.correlate(h, 1-h, mode="full") * dx)
4. Current best: 0.38085, previous achieved 0.381038, target 0.3808

The strategy uses DC decomposition:
- For each lag d, q_d(h) = dx * sum h[i]*(1-h[i+d]) over valid indices
- Decompose: q_d(y) = A_d(y) - B_d(y) where both are convex
- A_d(y) = dx * sum [y[i] + (y[i]-y[j])²/4]
- B_d(y) = dx/4 * sum (y[i]+y[j])²
- Majorize B_d by its tangent at anchor a: T_d(y;a) = dx/2 * sum (a[i]+a[j])*(y[i]+y[j]) - dx/4 * sum (a[i]+a[j])²
- The convex majorizer: M_d(y;a) = A_d(y) - T_d(y;a)

Then solve:
minimize t + eta/2 * ||y-a||²
subject to M_d(y;a) <= t for all d
           0 <= y[i] <= 1
           sum y[i] = S
           -rho <= y[i]-a[i] <= rho

This is a convex QP/SOCP that CVXPY can handle.

import numpy as np
import cvxpy as cp
from scipy.optimize import minimize

def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)):
        return None
    n = len(z)
    if S < 0 or S > n or np.max(np.abs(z)) > 1e9:
        return None
    lo_v = np.min(z) - 1.0
    hi_v = np.max(z)
    for _ in range(80):
        mid = lo_v + 0.5 * (hi_v - lo_v)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo_v = mid
        else:
            hi_v = mid
    lam = lo_v + 0.5 * (hi_v - lo_v)
    x = np.clip(z - lam, 0.0, 1.0)
    r = S - np.sum(x)
    if r > 0:
        for i in range(n):
            if r <= 0: break
            a = min(r, 1.0 - x[i])
            x[i] += a; r -= a
    elif r < 0:
        for i in range(n):
            if r >= 0: break
            a = min(-r, x[i])
            x[i] -= a; r += a
    if abs(np.sum(x) - S) > 1e-10 or not np.all(np.isfinite(x)):
        return None
    return np.clip(x, 0.0, 1.0)

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-9)

def exact_score(h, n):
    try:
        vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if not np.all(np.isfinite(vals)):
            return None
        return float(np.max(vals))
    except Exception:
        return None

def dc_subproblem(anchor, n, rho, eta):
    """Solve the convex DC subproblem at anchor point."""
    S = n / 2.0
    dx = 2.0 / n
    y = cp.Variable(n)
    t = cp.Variable()
    constraints = [y >= 0, y <= 1, cp.sum(y) == S,
                   y - anchor <= rho, anchor - y <= rho]
    for d in range(-(n-1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        y0 = y[lo:hi]
        y1 = y[lo+d:hi+d]
        s = anchor[lo:hi] + anchor[lo+d:hi+d]
        # M_d(y;a) = dx * (sum(y0) + 0.25*||(y0-y1)||² - 0.5*s·(y0+y1) + 0.25*||s||²)
        obj_d = dx * (cp.sum(y0) + 0.25 * cp.sum_squares(y0 - y1)
                      - 0.5 * cp.sum(s * (y0 + y1)) + 0.25 * np.dot(s, s))
        constraints.append(obj_d <= t)
    objective = cp.Minimize(t + 0.5 * eta * cp.sum_squares(y - anchor))
    prob = cp.Problem(objective, constraints)
    if not prob.is_dcp():
        return None
    for solver in [cp.ECOS, cp.CVXOPT]:
        try:
            prob.solve(solver=solver, verbose=False, max_iters=400)
            if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                if y.value is not None and np.all(np.isfinite(y.value)):
                    return np.asarray(y.value, dtype=np.float64)
        except Exception:
            pass
    return None

def dc_optimize(start, n, best_h, best_score):
    """Run DC-MM iterations from start."""
    dx = 2.0 / n
    S = n / 2.0
    x = start.copy()
    f = exact_score(x, n)
    if f is None:
        return best_h, best_score
    rho = 0.20
    rho_max = 0.40
    rho_min = 0.005
    eta = 0.20 * dx
    eta_min = 0.02 * dx
    max_attempts = 8 if n <= 128 else 5
    fail_count = 0
    noimp_count = 0
    small_decr = 0
    for _ in range(max_attempts):
        cand = dc_subproblem(x, n, rho, eta)
        if cand is None:
            rho = max(rho_min, rho * 0.5)
            fail_count += 1
            if fail_count >= 2:
                break
            continue
        fc = project_capped_simplex(cand, S)
        if fc is None or not is_feasible(fc, n):
            rho = max(rho_min, rho * 0.5)
            fail_count += 1
            if fail_count >= 2:
                break
            continue
        # Line search
        best_alpha = None
        best_trial_score = f
        for j in range(7):
            alpha = 2.0 ** (-j)
            trial = x + alpha * (fc - x)
            tc = project_capped_simplex(trial, S)
            if tc is None or not is_feasible(tc, n):
                continue
            ts = exact_score(tc, n)
            if ts is None:
                continue
            if ts < best_trial_score - 1e-12:
                best_trial_score = ts
                best_alpha = alpha
                best_trial = tc.copy()
        if best_alpha is not None:
            disp = np.max(np.abs(best_trial - x))
            decr = f - best_trial_score
            if decr > best_score - 1e-12:
                best_score = best_trial_score
                best_h = best_trial.copy()
            x = best_trial
            f = best_trial_score
            fail_count = 0
            noimp_count = 0
            if decr <= 1e-10:
                small_decr += 1
            else:
                small_decr = 0
            if small_decr >= 2:
                break
            if disp <= 1e-7:
                break
            if disp >= 0.8 * rho and decr > 1e-5:
                rho = min(rho_max, 1.25 * rho)
        else:
            rho = max(rho_min, rho * 0.5)
            eta = max(eta_min, eta * 0.5)
            noimp_count += 1
            if noimp_count >= 3:
                break
    return best_h, best_score

def build_starts(n, S, parent, rng):
    starts = []
    const = np.full(n, 0.5)
    starts.append(const)
    if parent is not None:
        starts.append(parent.copy())
        starts.append((1.0 - parent).copy())
        starts.append(parent[::-1].copy())
        pert = parent + rng.normal(0.0, 0.05, n)
        p = project_capped_simplex(pert, S)
        if p is not None:
            starts.append(p)
    # Low-frequency
    idx = np.arange(n, dtype=np.float64)
    M = min(6, (n - 1) // 2)
    wave = np.zeros(n)
    for m in range(1, M + 1):
        A = rng.normal(0.0, 0.10 / np.sqrt(m))
        phi = rng.uniform(0.0, 2.0 * np.pi)
        wave += A * np.cos(2.0 * np.pi * m * (idx + 0.5) / n + phi)
    lf = project_capped_simplex(0.5 + wave, S)
    if lf is not None:
        starts.append(lf)
    # Balance binary
    perm = rng.permutation(n)
    binv = np.zeros(n)
    binv[perm[:n // 2]] = 1.0
    if n % 2 == 1:
        binv[perm[n // 2]] = 0.5
    starts.append(binv.copy())
    # Deduplicate
    deduped = []
    for s in starts:
        dup = False
        for d in deduped:
            if np.max(np.abs(s - d)) < 1e-9:
                dup = True; break
        if not dup and is_feasible(s, n):
            deduped.append(s)
    return deduped

def run(seed=42, **kwargs):
    if seed is None:
        seed = 42
    try:
        seed = int(seed) & 0xFFFFFFFF
    except (TypeError, ValueError):
        seed = 42
    rng = np.random.default_rng(seed)
    n = None
    parent = None
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        try:
            iv_arr = np.asarray(iv, dtype=np.float64)
            if iv_arr.ndim == 1 and 8 <= iv_arr.size <= 999:

                    proj = project_capped_simplex(iv_arr, S)
                    if proj is not None and is_feasible(proj, n):
                        parent = proj
        except Exception:
            pass
    if n is None:
        n = kwargs.get('n_points', 80)
        try:
            n = int(n)
        except (TypeError, ValueError):
            n = 80
        if n < 8 or n > 999:
            n = 80
    S = n / 2.0
    dx = 2.0 / n
    const = np.full(n, 0.5)
    best_h = const.copy()
    best_score = exact_score(const, n)
    if best_score is None:
        best_score = 0.5
    if parent is not None:
        ps = exact_score(parent, n)
        if ps is not None and ps < best_score:
            best_score = ps
            best_h = parent.copy()
    starts = build_starts(n, S, parent, rng)
    for s in starts:
        sc = exact_score(s, n)
        if sc is not None and sc < best_score:
            best_score = sc
            best_h = s.copy()
    # DC optimization from each start
    if n <= 256:
        for s in starts:
            best_h, best_score = dc_optimize(s, n, best_h, best_score)
        # Additional restart from best
        if n <= 128:
            pert = best_h + rng.normal(0.0, 0.03, n)
            p = project_capped_simplex(pert, S)
            if p is not None and is_feasible(p, n):
                best_h, best_score = dc_optimize(p, n, best_h, best_score)
    # Final validation
    if not is_feasible(best_h, n):
        best_h = project_capped_simplex(best_h, S)
        if best_h is None:
            best_h = const.copy()
    final_score = exact_score(best_h, n)
    if final_score is None:
        final_score = 0.5
        best_h = const.copy()
    return best_h, final_score, n

import numpy as np
import cvxpy as cp

def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)):
        return None
    n = len(z)
    if S < 0 or S > n or np.max(np.abs(z)) > 1e9:
        return None
    lo_v, hi_v = np.min(z) - 1.0, np.max(z)
    for _ in range(80):
        mid = lo_v + 0.5 * (hi_v - lo_v)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo_v = mid
        else:
            hi_v = mid
    lam = lo_v + 0.5 * (hi_v - lo_v)
    x = np.clip(z - lam, 0.0, 1.0)
    r = S - np.sum(x)
    if r > 0:
        for i in range(n):
            if r <= 0: break
            a = min(r, 1.0 - x[i]); x[i] += a; r -= a
    elif r < 0:
        for i in range(n):
            if r >= 0: break
            a = min(-r, x[i]); x[i] -= a; r += a
    if abs(np.sum(x) - S) > 1e-10 or not np.all(np.isfinite(x)):
        return None
    return np.clip(x, 0.0, 1.0)

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-11) and np.all(h <= 1 + 1e-11)
            and abs(np.sum(h) - n / 2.0) <= 1e-9)

def exact_score(h, n):
    try:
        vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if not np.all(np.isfinite(vals)):
            return None
        return float(np.max(vals))
    except Exception:
        return None

def dc_solve(anchor, n, rho, eta):
    S, dx = n / 2.0, 2.0 / n
    y = cp.Variable(n)
    t = cp.Variable()
    cons = [y >= 0, y <= 1, cp.sum(y) == S,
            y - anchor <= rho, anchor - y <= rho]
    for j in range(2 * n - 1):
        d = j - (n - 1)
        lo = max(0, -d)
        hi = min(n, n - d)
        ya = y[lo + d:hi + d]
        yb = y[lo:hi]
        s = anchor[lo + d:hi + d] + anchor[lo:hi]
        cons.append(dx * (cp.sum(ya) + 0.25 * cp.sum_squares(ya - yb)
                         - 0.5 * cp.sum(s * (ya + yb))
                         + 0.25 * np.dot(s, s)) <= t)
    prob = cp.Problem(cp.Minimize(t + 0.5 * eta * cp.sum_squares(y - anchor)), cons)
    if not prob.is_dcp():
        return None
    for solver in [cp.ECOS, cp.CVXOPT]:
        try:
            prob.solve(solver=solver, verbose=False, max_iters=500)
            if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE) and y.value is not None:
                if np.all(np.isfinite(y.value)):
                    return np.asarray(y.value, dtype=np.float64)
        except Exception:
            pass
    return None

def dc_optimize(start, n, best_h, best_score):
    dx, S = 2.0 / n, n / 2.0
    x = start.copy()
    f = exact_score(x, n)
    if f is None:
        return best_h, best_score
    rho, rho_max, rho_min = 0.20, 0.40, 0.005
    eta, eta_min = 0.20 * dx, 0.02 * dx
    max_att = 8 if n <= 128 else 5
    fail_c, noimp_c, small_c = 0, 0, 0
    for _ in range(max_att):
        cand = dc_solve(x, n, rho, eta)
        if cand is None or not np.all(np.isfinite(cand)):
            rho = max(rho_min, rho * 0.5); fail_c += 1
            if fail_c >= 2: break
            continue
        fc = project_capped_simplex(cand, S)
        if fc is None or not is_feasible(fc, n):
            rho = max(rho_min, rho * 0.5); fail_c += 1
            if fail_c >= 2: break
            continue
        best_a, best_ts, best_tv = None, f, None
        for jj in range(7):
            alpha = 2.0 ** (-jj)
            trial = x + alpha * (fc - x)
            tc = project_capped_simplex(trial, S)
            if tc is None or not is_feasible(tc, n):
                continue
            ts = exact_score(tc, n)
            if ts is not None and ts < best_ts - 1e-12:
                best_ts, best_a, best_tv = ts, alpha, tc.copy()
        if best_a is not None:
            decr = f - best_ts
            if best_ts < best_score - 1e-12:
                best_score, best_h = best_ts, best_tv.copy()
            x, f = best_tv, best_ts
            fail_c, noimp_c = 0, 0
            disp = np.max(np.abs(x - (x - best_a * (fc - x))))
            if decr <= 1e-10: small_c += 1
            else: small_c = 0
            if small_c >= 2: break
            if disp >= 0.8 * rho and decr > 1e-5:
                rho = min(rho_max, 1.25 * rho)
        else:
            rho = max(rho_min, rho * 0.5)
            eta = max(eta_min, eta * 0.5)
            noimp_c += 1
            if noimp_c >= 3: break
    return best_h, best_score

def build_starts(n, S, parent, rng):
    starts = [np.full(n, 0.5)]
    if parent is not None:
        starts.append(parent.copy())
        starts.append((1.0 - parent).copy())
        starts.append(parent[::-1].copy())
        p = project_capped_simplex(parent + rng.normal(0.0, 0.05, n), S)
        if p is not None: starts.append(p)
    idx = np.arange(n, dtype=np.float64)
    M = min(6, (n - 1) // 2)
    wave = np.zeros(n)
    for m in range(1, M + 1):
        wave += rng.normal(0.0, 0.10 / np.sqrt(m)) * np.cos(2.0 * np.pi * m * (idx + 0.5) / n + rng.uniform(0, 2 * np.pi))
    lf = project_capped_simplex(0.5 + wave, S)
    if lf is not None: starts.append(lf)
    perm = rng.permutation(n)
    bv = np.zeros(n); bv[perm[:n // 2]] = 1.0
    if n % 2: bv[perm[n // 2]] = 0.5
    starts.append(bv.copy())
    deduped = []
    for s in starts:
        dup = any(np.max(np.abs(s - d)) < 1e-9 for d in deduped)
        if not dup and is_feasible(s, n):
            deduped.append(s)
    return deduped

def run(seed=42, **kwargs):
    if seed is None: seed = 42
    try: seed = int(seed) & 0xFFFFFFFF
    except (TypeError, ValueError): seed = 42
    rng = np.random.default_rng(seed)
    n, parent = None, None
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        try:
            iv_arr = np.asarray(iv, dtype=np.float64)
            if iv_arr.ndim == 1 and 8 <= iv_arr.size <= 999:
                n = iv_arr.size
                S = n / 2.0
                if np.all(np.isfinite(iv_arr)) and np.max(np.abs(iv_arr)) <= 1e9:
                    pr = project_capped_simplex(iv_arr, S)
                    if pr is not None and is_feasible(pr, n):
                        parent = pr
        except Exception:
            pass
    if n is None:
        try: n = int(kwargs.get('n_points', 80))
        except (TypeError, ValueError): n = 80
        if n < 8 or n > 999: n = 80
    S, dx = n / 2.0, 2.0 / n
    const = np.full(n, 0.5)
    best_h, best_score = const.copy(), exact_score(const, n)
    if best_score is None: best_score = 0.5
    if parent is not None:
        ps = exact_score(parent, n)
        if ps is not None and ps < best_score:
            best_score, best_h = ps, parent.copy()
    starts = build_starts(n, S, parent, rng)
    for s in starts:
        sc = exact_score(s, n)
        if sc is not None and sc < best_score:
            best_score, best_h = sc, s.copy()
    if n <= 256:
        for s in starts:
            best_h, best_score = dc_optimize(s, n, best_h, best_score)
        if n <= 128:
            p2 = project_capped_simplex(best_h + rng.normal(0.0, 0.03, n), S)
            if p2 is not None and is_feasible(p2, n):
                best_h, best_score = dc_optimize(p2, n, best_h, best_score)
    if not is_feasible(best_h, n):
        best_h = project_capped_simplex(best_h, S)
        if best_h is None: best_h = const.copy()
    fs = exact_score(best_h, n)
    if fs is None:
        best_h = const.copy(); fs = exact_score(best_h, n)
    if fs is None: fs = 0.5
    return best_h, fs, n
