            G[row, lo + d:hi + d] -= h[lo:hi]
    return v, G

    v = np.zeros(L)
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        lo = max(0, -d)
        hi_t = min(n, n - d)
        v[row] = float(np.sum(h[lo:hi_t] * (1.0 - h[lo + d:hi_t + d])))
        G[row, lo:hi_t] += (1.0 - h[lo + d:hi_t + d])
        G[row, lo + d:hi_t + d] -= h[lo:hi_t]
    return v, G

    return z[-1]

    j[-1] = 1.0
    return j

    j[:n] = 1.0
    return j

    t = z[-1]
    v, _ = lag_vals_grads(h, n)
    return t - (2.0 / n) * v

    L = 2 * n - 1
    J = np.zeros((L, n + 1))

    return J

def polish(h, n, sweeps):
    cur = exact_score(h, n)
    for _ in range(sweeps):
        imp = False
        for i in range(n - 1):
            for j in range(i + 1, n):
                lo = max(-h[i], h[j] - 1.0)
                hi_p = min(1.0 - h[i], h[j])
                if hi_p - lo < 1e-12:
                    continue
                cs = [lo, hi_p, 0.5 * (lo + hi_p)]
                for fr in (0.005, 0.02, 0.08, 0.3):
                    c = 0.5 * (lo + hi_p)
                    cs.append(c + fr)
                    cs.append(c - fr)
                best_d, best_s = 0.0, cur

                    h[j] -= best_d
                    cur = best_s

            lo = mid
        else:

    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))

        for i in range(n):
            if r <= 1e-13: break
            a = min(r, 1.0 - h[i]); h[i] += a; r -= a
    elif r < -1e-13:
        for i in range(n):
            if -r <= 1e-13: break

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)): return float('inf')

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_vals_grads(h, n):
    L = 2 * n - 1
    v = np.zeros(L); G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        lo, hp = max(0, -d), min(n, n - d)
        v[row] = float(np.sum(h[lo:hp] * (1.0 - h[lo + d:hp + d])))
        G[row, lo:hp] += 1.0 - h[lo + d:hp + d]
        G[row, lo + d:hp + d] -= h[lo:hp]
    return v, G

    j = np.zeros(n + 1); j[-1] = 1.0; return j

    j = np.zeros(n + 1); j[:n] = 1.0; return j
def _ineq(z, n):
    v, _ = lag_vals_grads(z[:n], n)
    return z[-1] - (2.0 / n) * v

    _, G = lag_vals_grads(z[:n], n)
    L = 2 * n - 1
    J = np.zeros((L, n + 1))
    J[:, :n] = -(2.0 / n) * G; J[:, -1] = 1.0
    return J

def polish(h, n, sweeps):
    cur = exact_score(h, n)
    for _ in range(sweeps):
        imp = False
        for i in range(n - 1):
            for j in range(i + 1, n):
                if h[i] < 1e-5 and h[j] < 1e-5: continue
                if h[i] > 1 - 1e-5 and h[j] > 1 - 1e-5: continue
                lo = max(-h[i], h[j] - 1.0)
                hip = min(1.0 - h[i], h[j])
                if hip - lo < 1e-12: continue
                c = 0.5 * (lo + hip)
                ds = [0.0, lo, hip, c, c + 0.01, c - 0.01, c + 0.05, c - 0.05]
                best_d, best_s = 0.0, cur
                for dd in ds:
                    dd = max(lo, min(hip, dd))
                    h[i] += dd; h[j] -= dd
                    sc = exact_score(h, n)
                    if sc < best_s - 1e-12: best_s, best_d = sc, dd
                    h[i] -= dd; h[j] += dd
                if best_d != 0.0:
                    h[i] += best_d; h[j] -= best_d
                    cur = best_s; imp = True
        if not imp: break
    return h, cur
