```python
import numpy as np
import math

def proj_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):
        return None
    if np.max(np.abs(z)) > 1e12:
        return None
    n = len(z)
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    y = np.clip(z - lam, 0.0, 1.0)
    r = S - np.sum(y)
    if r > 1e-14:
        for i in range(n):
            if r <= 0: break
            a = min(r, 1.0 - y[i])
            y[i] += a; r -= a
    elif r < -1e-14:
        for i in range(n):
            if r >= 0: break
            a = min(-r, y[i])
            y[i] -= a; r += a
    if abs(np.sum(y) - S) > 1e-10 or not np.all(np.isfinite(y)):
        return None
    return np.clip(y, 0.0, 1.0)

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-14) and np.all(h <= 1.0 + 1e-14)
            and abs(np.sum(h) - n / 2.0) <= 1e-10)

def exact_score(h, n):
    try:
        vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if len(vals) != 2 * n - 1 or not np.all(np.isfinite(vals)):
            return None
        return float(np.max(vals))
    except Exception:
        return None

def exact_lags(h, n):
    try:
        vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if len(vals) != 2 * n - 1 or not np.all(np.isfinite(vals)):
            return None
        return vals[::-1].copy()
    except Exception:
        return None

def pair_line_search(h, p, r, q, n, dx):
    L = max(-h[p], h[r] - 1.0)
    U = min(1.0 - h[p], h[r])
    if U - L <= 1e-12:
        return None
    c = q
    m = 2 * n - 1
    b = np.zeros(m)
    a = np.zeros(m)
    d0 = n - 1
    a[d0] = -2.0 * dx
    dr = r - p
    if dr != 0:
        a[d0 + dr] = dx
        a[d0 - dr] = dx
    # b_d = dx * [(1-h[p+d]) if valid - (1-h[r+d]) if valid - h[p-d] if valid + h[r-d] if valid]
    for d in range(-(n-1), n):
        bd = 0.0
        ip = p + d
        if 0 <= ip < n: bd += 1.0 - h[ip]
        ir = r + d
        if 0 <= ir < n: bd -= 1.0 - h[ir]
        ip2 = p - d
        if 0 <= ip2 < n: bd -= h[ip2]
        ir2 = r - d
        if 0 <= ir2 < n: bd += h[ir2]
        b[d + d0] = dx * bd
    # Line search
    ts = np.linspace(L, U, 17)
    if 0.0 > L and 0.0 < U:
        ts = np.concatenate([ts, [0.0]])
    best_t, best_f = 0.0, np.max(c)
    for t in ts:
        f = np.max(c + b * t + a * t * t)
        if f < best_f - 1e-14 or (abs(f - best_f) <= 1e-14 and abs(t) < abs(best_t)):
            best_f = f; best_t = t
    # Refine
    radius = (U - L) / 16.0
    for _ in range(4):
        lo_t = max(L, best_t - radius)
        hi_t = min(U, best_t + radius)
        ts2 = np.linspace(lo_t, hi_t, 5)
        for t in ts2:
            f = np.max(c + b * t + a * t * t)
            if f < best_f - 1e-14 or (abs(f - best_f) <= 1e-14 and abs(t) < abs(best_t)):
                best_f = f; best_t = t
        radius /= 4.0
    cur_score = float(np.max(c))
    if best_f >= cur_score - 1e-12:
        return None
    # Construct new h
    hn = h.copy()
    hn[p] += best_t
    hn[r] -= best_t
    if not is_feasible(hn, n):
        hn = proj_simplex(hn, n / 2.0)
        if hn is None:
            return None
    new_lags = exact_lags(hn, n)
    if new_lags is None:
        return None
    new_score = float(np.max(new_lags))
    if new_score >= cur_score - 1e-12:
        return None
    return hn, new_score, new_lags

def polish_sweep(h, score, q, n, dx, rng, max_sweeps):
    for sw in range(max_sweeps):
        P = min(n * (n - 1) // 2, 2 * n, max(24, 80000 // n))
        adj = min(n - 1, max(1, P // 2))
        pairs = []
        for i in range(adj):
            pairs.append((i, i + 1))
        while len(pairs) < P:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j: continue
            if i > j: i, j = j, i
            dup = False
            for a2, b2 in pairs:
                if i == a2 and j == b2: dup = True; break
            if not dup:
                pairs.append((i, j))
        np.random.default_rng(0)  # noop to avoid import issues
        # shuffle pairs
        perm = rng.permutation(len(pairs))
        pairs = [pairs[k] for k in perm]
        accepted = 0
        for p, r in pairs:
            res = pair_line_search(h, p, r, q, n, dx)
            if res is not None:
                h, score, q = res
                accepted += 1
        if accepted == 0:
            break
    return h, score, q

def anneal_trajectory(h, score, q, n, dx, rng):
    if n <= 128: E = 5
    elif n <= 256: E = 4
    elif n <= 512: E = 3
    else: E = 2
    K = min(1000, max(80, 80000 // n))
    scale = 0.12
    S = n / 2.0
    for e in range(E):
        T = max(5e-5, 3e-3 * (0.4 ** e))
        acc = 0
        for _ in range(K):
            p = int(rng.integers(0, n))
            r = int(rng.integers(0, n))
            if p == r: continue
            L = max(-h[p], h[r] - 1.0)
            U = min(1.0 - h[p], h[r])
            if U - L <= 1e-12: continue
            t = float(rng.uniform(-scale, scale))
            if t < L or t > U: continue
            hn = h.copy()
            hn[p] += t
            hn[r] -= t
            if not is_feasible(hn, n):
                hn = proj_simplex(hn, S)
                if hn is None: continue
            ns = exact_score(hn, n)
            if ns is None: continue
            delta = ns - score
            if delta <= 0 or (delta < 700 * T and rng.random() < math.exp(-delta / T)):
                h, score, q = hn, ns, exact_lags(hn, n)
                if q is None:
                    h, score = hn, ns
                    q = exact_lags(h, n)
                    if q is None: break
                acc += 1
        ratio = acc / K if K > 0 else 0
        if ratio < 0.12: scale = min(0.4, 1.25 * scale)
        elif ratio > 0.55: scale = max(0.01, 0.75 * scale)
        h, score, q = polish_sweep(h, score, q, n, dx, rng, 1)
    return h, score, q

def build_starts(n, S, parent, rng):
    starts = []
    if parent is not None:
        starts.append(parent.copy())
        starts.append(parent[::-1].copy())
        comp = 1.0 - parent[::-1].copy()
        starts.append(comp)
    M = min(4, (n - 1) // 2)
    idx = np.arange(n)
    u = np.full(n, 0.5)
    for m in range(1, M + 1):
        A = rng.normal(0, 0.12 / math.sqrt(m))
        phi = rng.uniform(0, 2 * np.pi)
        u += A * np.cos(2 * np.pi * m * (idx + 0.5) / n + phi)
    u = proj_simplex(u, S)
    if u is not None: starts.append(u)
    perm = rng.permutation(n)
    bv = np.zeros(n)
    bv[perm[:n // 2]] = 1.0
    if n % 2 == 1: bv[perm[n // 2]] = 0.5
    starts.append(bv)
    rv = rng.uniform(0, 1, n)
    rv = proj_simplex(rv, S)
    if rv is not None: starts.append(rv)
    const = np.full(n, 0.5)
    starts.append(const)
    deduped = []
    for s in starts:
        ok = True
        for d in deduped:
            if np.max(np.abs(s - d)) < 1e-10: ok = False; break
        if ok and is_feasible(s, n): deduped.append(s)
    return deduped
```

```python
import numpy as np
import math

def proj_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):
        return None
    if np.max(np.abs(z)) > 1e12:
        return None
    n = len(z)
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - np.sum(y)
    for i in range(n):
        if r > 1e-15 and y[i] < 1.0:
            a = min(r, 1.0 - y[i]); y[i] += a; r -= a
        elif r < -1e-15 and y[i] > 0.0:
            a = min(-r, y[i]); y[i] -= a; r += a
        if abs(r) < 1e-15: break
    if abs(np.sum(y) - S) > 1e-10 or not np.all(np.isfinite(y)) or np.any(y < -1e-14) or np.any(y > 1.0 + 1e-14):
        return None
    return np.clip(y, 0.0, 1.0)

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-14) and np.all(h <= 1.0 + 1e-14)
            and abs(np.sum(h) - n / 2.0) <= 1e-10)

def exact_corr(h, n):
    try:
        v = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if len(v) != 2 * n - 1 or not np.all(np.isfinite(v)):
            return None
        return v
    except Exception:
        return None

def pair_line_search(h, p, r, corr, n, dx):
    L = max(-h[p], h[r] - 1.0)
    U = min(1.0 - h[p], h[r])
    if U - L <= 1e-12:
        return None
    d0 = n - 1
    c = corr
    m = 2 * n - 1
    a = np.zeros(m)
    a[d0] = -2.0 * dx
    dr = r - p
    if dr != 0:
        a[d0 + dr] = dx
        a[d0 - dr] = dx
    b = np.zeros(m)
    for d in range(-(n - 1), n):
        v = 0.0
        ip = p + d
        if 0 <= ip < n: v += 1.0 - h[ip]
        ir = r + d
        if 0 <= ir < n: v -= 1.0 - h[ir]
        ip2 = p - d
        if 0 <= ip2 < n: v -= h[ip2]
        ir2 = r - d
        if 0 <= ir2 < n: v += h[ir2]
        b[d0 + d] = dx * v
    cur = float(np.max(c))
    ts = np.linspace(L, U, 17)
    if L < 0.0 < U:
        ts = np.append(ts, 0.0)
    best_t, best_f = 0.0, cur
    for t in ts:
        f = float(np.max(c + b * t + a * t * t))
        if f < best_f - 1e-14 or (abs(f - best_f) <= 1e-14 and abs(t) < abs(best_t)):
            best_f, best_t = f, t
    radius = (U - L) / 16.0
    for _ in range(4):
        lo_t = max(L, best_t - radius)
        hi_t = min(U, best_t + radius)
        if hi_t - lo_t < 1e-15: break
        ts2 = np.linspace(lo_t, hi_t, 5)
        for t in ts2:
            f = float(np.max(c + b * t + a * t * t))
            if f < best_f - 1e-14 or (abs(f - best_f) <= 1e-14 and abs(t) < abs(best_t)):
                best_f, best_t = f, t
        radius *= 0.25
    if best_f >= cur - 1e-12:
        return None
    hn = h.copy()
    hn[p] += best_t
    hn[r] -= best_t
    if not is_feasible(hn, n):
        hn = proj_simplex(hn, n / 2.0)
        if hn is None:
            return None
    nc = exact_corr(hn, n)
    if nc is None:
        return None
    ns = float(np.max(nc))
    if ns >= cur - 1e-12:
        return None
    return hn, ns, nc

def build_pair_list(n, rng):
    P = min(n * (n - 1) // 2, 2 * n, max(24, 80000 // n))
    adj = min(n - 1, max(1, P // 2))
    pairs = [(i, i + 1) for i in range(adj)]
    attempts = 0
    while len(pairs) < P and attempts < 2000:
        attempts += 1
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j: continue
        if i > j: i, j = j, i
        dup = any(i == a2 and j == b2 for a2, b2 in pairs)
        if not dup:
            pairs.append((i, j))
    while len(pairs) < P:
        pairs.append((0, 1))
    perm = rng.permutation(len(pairs))
    return [pairs[k] for k in perm]

def polish(h, score, corr, n, dx, rng, max_sweeps):
    for _ in range(max_sweeps):
        pairs = build_pair_list(n, rng)
        acc = 0
        for p, r in pairs:
            res = pair_line_search(h, p, r, corr, n, dx)
            if res is not None:
                h, score, corr = res
                acc += 1
        if acc == 0:
            break
    return h, score, corr

def anneal(h, score, corr, n, dx, rng):
    S = n / 2.0
    if n <= 128: E = 5
    elif n <= 256: E = 4
    elif n <= 512: E = 3
    else: E = 2
    K = min(1000, max(80, 80000 // n))
    scale = 0.12
    for e in range(E):
        T = max(5e-5, 3e-3 * (0.4 ** e))
        acc = 0
        for _ in range(K):
            p = int(rng.integers(0, n))
            r = int(rng.integers(0, n))
            if p == r: continue
            L = max(-h[p], h[r] - 1.0)
            U = min(1.0 - h[p], h[r])
            if U - L <= 1e-13: continue
            t = float(rng.uniform(-scale, scale))
            if t < L or t > U: continue
            hn = h.copy()
            hn[p] += t
            hn[r] -= t
            if not is_feasible(hn, n):
                hn = proj_simplex(hn, S)
                if hn is None: continue
            nc = exact_corr(hn, n)
            if nc is None: continue
            ns = float(np.max(nc))
            delta = ns - score
            if delta <= 0:
                h, score, corr = hn, ns, nc; acc += 1
            else:
                exp_arg = -delta / T
                if exp_arg > -700 and rng.random() < math.exp(exp_arg):
                    h, score, corr = hn, ns, nc; acc += 1
        ratio = acc / K if K > 0 else 0.0
        if ratio < 0.12: scale = min(0.4, 1.25 * scale)
        elif ratio > 0.55: scale = max(0.01, 0.75 * scale)
        h, score, corr = polish(h, score, corr, n, dx, rng, 1)
    return h, score, corr

def build_starts(n, S, parent, rng):
    starts = []
    if parent is not None:
        starts.append(parent.copy())
        starts.append(parent[::-1].copy())
    idx = np.arange(n, dtype=np.float64)
    M = min(4, (n - 1) // 2)
    u = np.full(n, 0.5)
    for m in range(1, M + 1):
        A = rng.normal(0, 0.12 / math.sqrt(m))
        phi = rng.uniform(0, 2.0 * np.pi)
        u += A * np.cos(2.0 * np.pi * m * (idx + 0.5) / n + phi)
    u = proj_simplex(u, S)
    if u is not None: starts.append(u)
    perm = rng.permutation(n)
    bv = np.zeros(n)
    bv[perm[:n // 2]] = 1.0
    if n % 2 == 1: bv[perm[n // 2]] = 0.5
    starts.append(bv)
    rv = proj_simplex(rng.uniform(0, 1, n), S)
    if rv is not None: starts.append(rv)
    starts.append(np.full(n, 0.5))
    deduped = []
    for s in starts:
        ok = True
        for d in deduped:
            if np.max(np.abs(s - d)) < 1e-10:
                ok = False; break
        if ok and is_feasible(s, n): deduped.append(s)
    return deduped

def run(seed=42, **kwargs):

    try: seed = int(seed)
    except: seed = 42

        except: pass
    if n is None:

    S = n / 2.0

    const_h = np.full(n, 0.5, dtype=np.float64)
    best_h = const_h.copy()

    best_score = float(np.max(cc)) if cc is not None else 0.5
    if parent is not None:
        pc = exact_corr(parent, n)
        if pc is not None:
            ps = float(np.max(pc))
            if ps < best_score:

    starts = build_starts(n, S, parent, rng)

    elif n <= 256: cap = 4
    elif n <= 512: cap = 3
    else: cap = 2
    starts = starts[:cap]

        elif n <= 256: nsw = 3
```
