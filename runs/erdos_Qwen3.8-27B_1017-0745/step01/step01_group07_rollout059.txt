def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    n = len(z)
    if n == 0 or not np.all(np.isfinite(z)) or S < 0 or S > n or np.max(np.abs(z)) > 1e12:
        return None
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
