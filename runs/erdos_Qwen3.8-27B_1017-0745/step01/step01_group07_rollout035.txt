    y = np.asarray(y, dtype=np.float64)

    if y.ndim != 1 or not np.all(np.isfinite(y)) or not (0 <= S <= n):
        return None
    if np.max(np.abs(y)) > 1e12:
        return None
    lo, hi = float(np.min(y)) - 1.0, float(np.max(y))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - np.sum(x)
    for _ in range(2):
        if r > 0:
            for i in range(n):
                if r <= 0: break
                a = min(r, 1.0 - x[i]); x[i] += a; r -= a
        elif r < 0:
            for i in range(n):
                if r >= 0: break
                a = min(-r, x[i]); x[i] -= a; r += a
    if abs(np.sum(x) - S) > 1e-10 or not np.all(np.isfinite(x)):
        return None
    return np.clip(x, 0.0, 1.0)

def is_feasible(h, n):
    return (h.ndim == 1 and h.shape[0] == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-14) and np.all(h <= 1.0 + 1e-14)
            and abs(np.sum(h) - n * 0.5) <= 1e-10)

def exact_score(h, n):
    try:
        v = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if v.size != 2 * n - 1 or not np.all(np.isfinite(v)):
            return None
        return float(np.max(v))
    except Exception:
        return None

def build_basis(n, D):
    L = (D + 1) // 2
    K = list(range(1, L + 1))
    R = D - L
    if R > 0:
        K.extend(int(round(float(s))) for s in np.linspace(L + 1, n - 1, R))
    seen, K2 = set(), []
    for k in K:
        if 1 <= k <= n - 1 and k not in seen:
            seen.add(k); K2.append(k)
    for k in range(1, n):
        if len(K2) >= D: break
        if k not in seen:
            seen.add(k); K2.append(k)
    K2 = sorted(K2[:D])
    idx = np.arange(n, dtype=np.float64) + 0.5
    B = np.zeros((n, D))
    for j, k in enumerate(K2):
        B[:, j] = np.sqrt(2.0 / n) * np.cos(np.pi * k * idx / n)
    B -= B.mean(axis=0, keepdims=True)
    try:
        Q, _ = np.linalg.qr(B)
    except np.linalg.LinAlgError:
        return None
    return Q if Q.shape == (n, D) and np.all(np.isfinite(Q)) else None

def cma_es(anchor, B, n, D, S, gen_cap, rng):
    P = max(5, 4 + int(3.0 * math.log(max(D, 2))))
    mu = max(1, P // 2)
    m, C = np.zeros(D), np.eye(D)
    p_s, p_c = np.zeros(D), np.zeros(D)
    w_r = np.array([math.log(mu + 0.5) - math.log(i + 1) for i in range(mu)])
    w = w_r / w_r.sum()
    mu_e = 1.0 / (w @ w)
    c_s = (mu_e + 2.0) / (D + mu_e + 5.0)
    d_s = 1.0 + 2.0 * max(0.0, math.sqrt(max(0.0, (mu_e - 1.0) / (D + 1.0))) - 1.0) + c_s
    c_c = (4.0 + mu_e / D) / (D + 4.0 + 2.0 * mu_e / D)
    c1 = 2.0 / ((D + 1.3) ** 2 + mu_e)
    c_m = min(1.0 - c1, max(0.0, 2.0 * (mu_e - 2.0 + 1.0 / mu_e) / ((D + 2.0) ** 2 + mu_e)))
    chi = math.sqrt(D) * (1.0 - 1.0 / (4.0 * D) + 1.0 / (21.0 * D * D))
    sigma = float(np.clip(0.10 * math.sqrt(n / D), 0.05, 0.70))
    pat = max(12, gen_cap // 4)
    run_best, no_imp, bh, bs = 1e18, 0, None, 1e18
    for g in range(gen_cap):
        C = 0.5 * (C + C.T)
        try:
            ev, evec = np.linalg.eigh(C)
        except Exception:
            break
        if not np.all(np.isfinite(ev)): break
        ev = np.clip(ev, 1e-10, 1e10)
        A_mat = evec @ np.diag(np.sqrt(ev))
        C_ih = evec @ np.diag(1.0 / np.sqrt(ev)) @ evec.T
        Zs, Sc = [], []
        for i in range(P):
            z = m + sigma * (A_mat @ rng.standard_normal(D))
            h_c = proj_simplex(anchor + B @ z, S)
            if h_c is not None and is_feasible(h_c, n):
                sc = exact_score(h_c, n)
                if sc is not None:
                    Zs.append(z); Sc.append(sc)
                    if sc < bs - 1e-12: bs, bh = sc, h_c.copy()
        if len(Sc) < mu: break
        ord_ = np.argsort(Sc, kind='stable')
        zs = np.array([Zs[i] for i in ord_[:mu]])
        m_old = m.copy()
        m = (w[:, None] * zs).sum(axis=0)
        yw = (m - m_old) / max(sigma, 1e-15)
        p_s = (1 - c_s) * p_s + math.sqrt(c_s * (2 - c_s) * mu_e) * (C_ih @ yw)
        nps = np.linalg.norm(p_s)
        denom = max(1e-15, math.sqrt(1.0 - (1 - c_s) ** (2 * (g + 1))))
        hs = 1.0 if nps / denom < (1.4 + 2.0 / (D + 1.0)) * chi else 0.0
        p_c = (1 - c_c) * p_c + hs * math.sqrt(c_c * (2 - c_c) * mu_e) * yw
        C = (1 - c1 - c_m) * C + c1 * (np.outer(p_c, p_c) + (1 - hs) * c_c * (2 - c_c) * C)
        for i in range(mu):
            yi = (zs[i] - m_old) / max(sigma, 1e-15)
            C += c_m * w[i] * np.outer(yi, yi)
        C = 0.5 * (C + C.T)
        expo = float(np.clip(c_s / d_s * (nps / max(chi, 1e-15) - 1.0), -1.0, 1.0))
        sigma = float(np.clip(sigma * math.exp(expo), 1e-3, 1.5))
        if bs < run_best - 1e-12: run_best, no_imp = bs, 0
        else: no_imp += 1
        if no_imp >= pat and sigma <= 1e-2: break
        if not (np.all(np.isfinite(m)) and np.all(np.isfinite(C)) and np.isfinite(sigma)): break
    return bh, bs

def dct_refine(h, n, S, D, B, rng, iters=400):
    best_h, best_sc = h.copy(), exact_score(h, n)
    if best_sc is None: return h
    step = 0.06
    for _ in range(iters):
        imp = False
        j = int(rng.integers(0, D))
        for d in [step, -step]:
            delta = np.zeros(D); delta[j] = d
            h2 = proj_simplex(best_h + B @ delta, S)
            if h2 is not None and is_feasible(h2, n):
                s2 = exact_score(h2, n)
                if s2 is not None and s2 < best_sc - 1e-12:
                    best_sc, best_h = s2, h2.copy(); imp = True; break
        if not imp:
            step *= 0.92
            if step < 1e-4: break
    return best_h

def run(seed=42, **kwargs):
    try: seed = int(seed) % (2 ** 32)
    except Exception: seed = 42
    rng = np.random.default_rng(seed)
    n, parent = None, None
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        try:
            iv = np.asarray(iv, dtype=np.float64)
            if iv.ndim == 1 and 8 <= iv.size <= 999:
                n = iv.size
                if np.all(np.isfinite(iv)) and np.max(np.abs(iv)) <= 1e10:
                    p = proj_simplex(iv, n * 0.5)
                    if p is not None and is_feasible(p, n): parent = p
        except Exception: pass
    if n is None:
        try: n = int(kwargs.get('n_points', 80))
        except Exception: n = 80
        if not (8 <= n <= 999): n = 80
    S, const = n * 0.5, np.full(n, 0.5)
    sc0 = exact_score(const, n)
    best_h, best_sc = const.copy(), sc0 if sc0 is not None else 0.5
    if parent is not None:
        ps = exact_score(parent, n)
        if ps is not None and ps < best_sc: best_sc, best_h = ps, parent.copy()
    Dcap = 40 if n <= 128 else (32 if n <= 256 else (20 if n <= 512 else 12))
    D = min(n - 1, Dcap)
    B = build_basis(n, D)
    if B is None: return best_h, best_sc, n
    gc = 80 if n <= 128 else (45 if n <= 256 else (18 if n <= 512 else 6))
    nA = 4 if n <= 128 else (3 if n <= 256 else (2 if n <= 512 else 1))
    anchors = []
    if parent is not None:
        anchors.append(parent.copy())
        pt = proj_simplex(parent + rng.normal(0, 0.08, n), S)
        if pt is not None and is_feasible(pt, n): anchors.append(pt)
        anchors.append(parent[::-1].copy())
        c2 = 1.0 - parent
        if is_feasible(c2, n): anchors.append(c2)
    else:
        anchors.append(const.copy())
    dedup = []
    for a in anchors[:nA]:
        dup = False
        for d in dedup:
            if np.max(np.abs(a - d)) < 1e-9: dup = True; break
        if not dup and is_feasible(a, n):
            dedup.append(a)
            sc = exact_score(a, n)
            if sc is not None and sc < best_sc: best_sc, best_h = sc, a.copy()
    if not dedup:
        dedup.append(const.copy())
    for a in dedup:
        bh, bs = cma_es(a, B, n, D, S, gc, rng)
        if bh is not None and bs < best_sc: best_sc, best_h = bs, bh.copy()
    if n <= 256:
        nr = 2 if n <= 128 else 1
        for _ in range(nr):
            bh, bs = cma_es(best_h.copy(), B, n, D, S, gc, rng)
            if bh is not None and bs < best_sc: best_sc, best_h = bs, bh.copy()
    best_h = dct_refine(best_h, n, S, D, B, rng)
    sc = exact_score(best_h, n)
    if sc is not None and sc < best_sc: best_sc = sc
    if not is_feasible(best_h, n):
        best_h = proj_simplex(best_h, S)
        if best_h is None: best_h = const.copy()
    sc = exact_score(best_h, n)
    if sc is None: sc = 0.5
    return best_h, sc, n
