    if abs(np.sum(h) - n / 2.0) > 1e-9 * max(1, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_values_grads(h, n):
    L = 2 * n - 1
    v = np.zeros(L)
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        m = hi - lo
        if m <= 0:
            continue
        i = lo + np.arange(m)
        j = i + d
        v[row] = float(np.sum(h[i] * (1.0 - h[j])))
        G[row, i] += (1.0 - h[j])
        G[row, j] -= h[i]
    return v, G

def obj_fun(x, n, dx):
    return x[-1]

def obj_jac(x, n, dx):
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return g

def constr_ineq_fun(x, n, dx):
    h = x[:-1]
    t = x[-1]
    v, _ = lag_values_grads(h, n)
    return t - dx * v

def constr_ineq_jac(x, n, dx):
    h = x[:-1]
    _, G = lag_values_grads(h, n)
    L = 2 * n - 1
    J = np.zeros((L, n + 1))
    J[:, :n] = -dx * G
    J[:, n] = 1.0
    return J

def constr_eq_fun(x, n, dx):
    return np.sum(x[:-1]) - n / 2.0

def constr_eq_jac(x, n, dx):
    J = np.zeros((1, n + 1))
    J[0, :n] = 1.0
    return J

def make_starts(n, parent, rng):
    raw = [np.full(n, 0.5), rng.uniform(0, 1, n)]
    idx = np.arange(n, dtype=np.float64)
    raw.append(0.5 + 0.45 * np.cos(2.0 * np.pi * idx / n))
    raw.append(0.5 + 0.45 * np.cos(4.0 * np.pi * idx / n))
    raw.append(0.5 + 0.35 * np.cos(6.0 * np.pi * idx / n) + 0.15 * np.cos(2.0 * np.pi * idx / n))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    raw.append(blk)
    if parent is not None:
        p = parent.copy()
        raw.extend([p, p[::-1], 1.0 - p, (1.0 - p)[::-1],
                    p + rng.normal(0, 0.025, n), p + rng.normal(0, 0.08, n),
                    p + rng.normal(0, 0.15, n)])
    proj = []
    for s in raw:
        pp = project_capped(s, n)
        if pp is None:
            continue
        dup = False
        for ex in proj:
            if float(np.max(np.abs(pp - ex))) < 1e-8:
                dup = True
                break
        if not dup:
            proj.append(pp)
    return proj

def polish_pairwise(h, n, max_sweeps):
    best_h = h.copy()
    best_s = exact_score(best_h, n)
    if not np.isfinite(best_s):
        return best_h
    deltas = np.array([0.001, 0.003, 0.01, 0.03, 0.1, 0.2, 0.3, 0.5])
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                hi, hj = best_h[i], best_h[j]
                lo = max(-hi, hj - 1.0)
                up = min(1.0 - hi, hj)
                if up - lo < 1e-12:
                    continue
                cands = np.array([lo, up, 0.0])
                for d in deltas:
                    for sgn in (1.0, -1.0):
                        dd = sgn * d
                        if lo - 1e-12 <= dd <= up + 1e-12:
                            cands = np.append(cands, np.clip(dd, lo, up))
                bd, bs = 0.0, best_s
                for dd in cands:
                    trial = best_h.copy()
                    trial[i] = hi + dd
                    trial[j] = hj - dd
                    ts = exact_score(trial, n)
                    if np.isfinite(ts) and ts < bs - 1e-13:
                        bs = ts
                        bd = dd
                if bd != 0.0:
                    best_h[i] += bd
                    best_h[j] -= bd
                    best_s = bs
                    improved = True
        if not improved:
            break
    return best_h

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 0 < len(iv) <= 999 and np.all(np.isfinite(iv)) and np.all(iv >= -1e-8) and np.all(iv <= 1.0 + 1e-8):
            n, parent = len(iv), iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
    n = max(1, min(999, n))
    try:
        sd = int(seed)
        if sd < 0:
            sd = 42
    except Exception:
        sd = 42
    rng = np.random.default_rng(sd)
    dx = 2.0 / n
    if n <= 64:
        ns_d, mi_d, ps_d = 8, 250, 2
    elif n <= 128:
        ns_d, mi_d, ps_d = 5, 160, 0
    elif n <= 256:
        ns_d, mi_d, ps_d = 3, 80, 0
    else:
        ns_d, mi_d, ps_d = 1, 0, 0
    n_starts = max(1, min(10, int(kwargs.get('starts', ns_d)))) if n <= 256 else 1
    max_iter = max(0, min(300, int(kwargs.get('max_iter', mi_d)))) if n <= 256 else 0
    polish_sw = max(0, min(2, int(kwargs.get('polish_sweeps', ps_d)))) if n <= 64 else 0
    if parent is not None:
        parent = project_capped(parent, n)
    starts = make_starts(n, parent, rng)[:n_starts]
    uniform = np.full(n, 0.5)
    best_h, best_s = uniform.copy(), exact_score(uniform, n)
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_s - 1e-12:
            best_h, best_s = st.copy(), ss
    if n <= 256 and max_iter > 0:
        args_opt = (n, dx)
        cons = [
            {'type': 'ineq', 'fun': constr_ineq_fun, 'jac': constr_ineq_jac},
            {'type': 'eq', 'fun': constr_eq_fun, 'jac': constr_eq_jac},
        ]
        bounds = [(0.0, 1.0)] * (n + 1)
        for st in starts:
            ss = exact_score(st, n)
            if not np.isfinite(ss):
                continue
            x0 = np.concatenate([st, [min(1.0, ss + 1e-8)]])
            try:
                res = minimize(obj_fun, x0, args=args_opt, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': max_iter, 'ftol': 1e-10})
                y = res.x[:n]
                if np.all(np.isfinite(y)):
                    yp = project_capped(y, n)
                    if yp is not None:
                        pys = exact_score(yp, n)
                        if np.isfinite(pys) and pys < best_s - 1e-12:
                            best_h, best_s = yp.copy(), pys
            except Exception:
                continue
    if n <= 64 and polish_sw > 0:
        pre_s = best_s
        best_h = polish_pairwise(best_h, n, polish_sw)
        post_s = exact_score(best_h, n)
        if not np.isfinite(post_s) or post_s > pre_s:
            best_h, best_s = best_h.copy(), pre_s
        else:
            best_s = post_s
    final = project_capped(best_h, n)
    if final is None:
        final = uniform.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > best_s + 1e-12:
        final = project_capped(best_h, n)
        fs = exact_score(final, n)
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final, float(fs), int(n)
