```python
def build_upper_hull(m_vec, c_vec, lo, hi):
    # m_vec, c_vec: slopes and intercepts of affine lines
    # Returns list of (m, c, start, end) for segments within [lo, hi]
    if len(m_vec) == 0:
        return []
    idx = np.argsort(m_vec, kind='stable')
    # Remove duplicates: for same slope, keep max intercept
    m_sorted = m_vec[idx]
    c_sorted = c_vec[idx]
    # Keep unique slopes with max intercept
    keep = np.ones(len(m_sorted), dtype=bool)
    for i in range(1, len(m_sorted)):
        if m_sorted[i] == m_sorted[i-1]:
            # Keep the one with max intercept
            if c_sorted[i] > c_sorted[i-1]:
                keep[i-1] = False
            else:
                keep[i] = False
    m_u = m_sorted[keep]
    c_u = c_sorted[keep]
    if len(m_u) == 0:
        return []
    if len(m_u) == 1:
        return [(m_u[0], c_u[0], lo, hi)]
    # Build upper hull using stack
    # Each element: (m, c, start_x)
    stack = [(m_u[0], c_u[0], -1e18)]
    for i in range(1, len(m_u)):
        m_new, c_new = m_u[i], c_u[i]
        # Crossing with top of stack
        m_top, c_top, st_top = stack[-1]
        x_cross = (c_top - c_new) / (m_new - m_top)
        while x_cross <= st_top and len(stack) > 1:
            stack.pop()
            m_top, c_top, st_top = stack[-1]
            x_cross = (c_top - c_new) / (m_new - m_top)
        stack.append((m_new, c_new, x_cross))
    # Build segments
    segments = []
    for i in range(len(stack)):
        m_s, c_s, st = stack[i]
        if i < len(stack) - 1:
            en = stack[i+1][2]
        else:
            en = 1e18
        # Clip to [lo, hi]
        st_c = max(st, lo)
        en_c = min(en, hi)
        if st_c < en_c:
            segments.append((m_s, c_s, st_c, en_c))
    return segments
```

```python
def exact_line_minimizer(h, p, q, n, c, a, b, lo, hi):
    if hi - lo <= 1e-14:
        return 0.0
    Delta = q - p
    s0, s1, s2 = n-1, n-1+Delta, n-1-Delta
    special = [s0, s1, s2]
    affine_mask = np.ones(2*n-1, dtype=bool)
    for s in special:
        affine_mask[s] = False
    m_aff = b[affine_mask]
    c_aff = c[affine_mask]
    
    candidates = [lo, hi, 0.0]
    
    segments = build_upper_hull(m_aff, c_aff, lo, hi)
    # Add hull switch points
    for i in range(1, len(segments)):
        candidates.append(segments[i][2])
    
    # For each segment, find intersections with special quadratics
    for (m_s, c_s, st, en) in segments:
        for s in special:
            # Solve a[s]*δ² + (b[s]-m_s)*δ + (c[s]-c_s) = 0
            A = a[s]
            B = b[s] - m_s
            C = c[s] - c_s
            roots = quad_roots(A, B, C, st, en)
            candidates.extend(roots)
    # Intersections between special quadratics
    for i in range(3):
        for j in range(i+1, 3):
            si, sj = special[i], special[j]
            A = a[si] - a[sj]
            B = b[si] - b[sj]
            C = c[si] - c[sj]
            roots = quad_roots(A, B, C, lo, hi)
            candidates.extend(roots)
    # Vertices of convex special quadratics
    for s in special:
        if a[s] > 0:
            v = -b[s] / (2.0 * a[s])
            if lo <= v <= hi:
                candidates.append(v)
    
    # Evaluate
    candidates = np.array(candidates)
    candidates = candidates[np.isfinite(candidates)]
    candidates = candidates[(candidates >= lo - 1e-12) & (candidates <= hi + 1e-12)]
    candidates = np.clip(candidates, lo, hi)
    
    if len(candidates) == 0:
        return 0.0
    # Remove near-duplicates
    candidates = np.unique(np.round(candidates, 12))
    
    best_val = np.inf
    best_delta = 0.0
    # Evaluate in chunks
    for i in range(0, len(candidates), 512):

        env = np.max(vals, axis=1)

            if env[k] < best_val:
                best_val = env[k]
                best_delta = chunk[k]
    return float(best_delta)
```

```python
def quad_roots(A, B, C, lo, hi):
    tol = 1e-14 * max(1.0, abs(A), abs(B), abs(C))
    roots = []
    if abs(A) <= tol:
        # Linear: B*δ + C = 0
        if abs(B) <= tol:
            return []
        r = -C / B
        if lo - tol <= r <= hi + tol:
            roots.append(min(max(r, lo), hi))
        return roots
    # Quadratic
    disc = B * B - 4.0 * A * C
    if disc < -tol * (B*B + 4*abs(A*C) + 1.0):
        return []
    if disc < 0:
        disc = 0.0
    sqrt_disc = math.sqrt(disc)
    # Stable formula
    if B >= 0:
        q_val = -0.5 * (B + sqrt_disc)
    else:
        q_val = -0.5 * (B - sqrt_disc)
    if abs(q_val) > tol:
        r1 = q_val / A
        r2 = C / q_val
    else:
        r1 = -B / (2.0 * A)
        r2 = r1
    for r in [r1, r2]:
        if np.isfinite(r) and lo - tol <= r <= hi + tol:
            roots.append(min(max(r, lo), hi))
    return roots
```
