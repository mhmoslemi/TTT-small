def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo, hi = z.min() - 1.0, z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)

        for i in range(N):
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
            if r <= 1e-14:

    elif r < -1e-14:

            h[i] -= adj
            r += adj
            if r >= -1e-14:
                break

        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / N)

    corr = np.correlate(h, 1.0 - h, mode='full')
    c = dx * corr
    cmax = np.max(c)
    e = np.exp((c - cmax) / tau)
    s = np.sum(e)
    F = cmax + tau * np.log(s)
    p = e / s
    g = np.zeros(N)
    # s=0: index N-1 in corr, p[N-1]
    g += dx * p[N-1] * (1.0 - 2.0 * h)

        w = dx * p[N-1-d]

        g[d:N] += w2 * (1.0 - h[:N-d])
        g[:N-d] -= w2 * h[d:N]
    return F, g, c

import numpy as np

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo, hi = z.min() - 1.0, z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    if r > 1e-14:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
            if r <= 1e-14:
                break
    elif r < -1e-14:
        for i in range(N):
            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
            if r >= -1e-14:
                break
    if h.min() < -1e-10 or h.max() > 1 + 1e-10 or not np.all(np.isfinite(h)):
        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)

        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / N)

def lse_val_grad(h, tau, dx):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = dx * corr
    cmax = np.max(c)
    e = np.exp((c - cmax) / tau)
    s = np.sum(e)
    F = cmax + tau * np.log(s)
    p = e / s
    g = np.zeros(N)
    g += dx * p[N - 1] * (1.0 - 2.0 * h)
    for d in range(1, N):
        w = dx * p[N - 1 + d]
        g[d:] += w * (1.0 - h[:N - d])
        g[:N - d] -= w * h[d:]
        w2 = dx * p[N - 1 - d]
        g[:N - d] += w2 * (1.0 - h[d:])
        g[d:] -= w2 * h[:N - d]
    return F, g

def optimize_start(h0, N, m, dx, max_iter):
    h = h0.copy()
    best_h = h0.copy()
    best_c = score_exact(h0, N)
    taus = [0.02, 0.006, 0.002, 0.0006, 0.0002, 0.00006, 0.000015]
    for tau in taus:
        alpha = 1.0
        stag = 0
        for it in range(max_iter):
            F, g = lse_val_grad(h, tau, dx)
            if not np.all(np.isfinite(g)) or not np.isfinite(F):
                break
            accepted = False
            for bt in range(24):
                y = project_box_sum(h - alpha * g, m)
                if not np.all(np.isfinite(y)):
                    break
                dv = y - h
                if np.max(np.abs(dv)) <= 1e-10:
                    accepted = True
                    break
                dy = np.dot(g, dv)
                if dy >= -1e-14:
                    alpha *= 0.5
                    continue
                F_y, _ = lse_val_grad(y, tau, dx)
                c_y = float(np.max(np.correlate(y, 1.0 - y, mode='full')))
                if c_y < best_c:
                    best_h = y.copy()
                    best_c = c_y
                if F_y <= F + 1e-4 * dy + 1e-12:
                    accepted = True
                    h = y.copy()
                    break
                alpha *= 0.5
            c_now = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
            if c_now < best_c:
                best_c = c_now
                best_h = h.copy()
            if not accepted:
                break
            alpha = min(1.25 * alpha, 10.0)
            if np.max(np.abs(dv)) < 1e-8:
                stag += 1
                if stag >= 5:
                    break
            else:
                stag = 0
    return best_h, best_c

def run(seed=42, **kwargs):
    N = None
    raw = globals().get('initial_h_values', None)
    if raw is not None:
        raw = np.asarray(raw, dtype=np.float64)
        if raw.ndim == 1 and raw.size > 1 and np.all(np.isfinite(raw)):
            N = int(raw.size)
    if N is None:
        N = int(kwargs.get('n_points', 70))
        if N < 2:
            N = 70
    m = N / 2.0
    dx = 2.0 / N
    rng = np.random.default_rng(seed)
    if raw is not None and raw.size == N and np.all(np.isfinite(raw)):
        baseline = project_box_sum(raw.copy(), m)
    else:
        baseline = np.full(N, 0.5)
    best_h = baseline.copy()
    best_c = score_exact(baseline, N)
    if N <= 128:
        n_starts, max_iter, hop_scales = 8, 50, [0.04, 0.015]
    elif N <= 256:
        n_starts, max_iter, hop_scales = 6, 30, [0.04]
    elif N <= 512:
        n_starts, max_iter, hop_scales = 4, 16, []
    else:
        n_starts, max_iter, hop_scales = 3, 8, []
    starts = [baseline.copy(), project_box_sum(1.0 - baseline, m)]
    for sc in [0.04, 0.12, 0.25]:
        starts.append(project_box_sum(baseline + sc * rng.standard_normal(N), m))
    starts.append(project_box_sum(0.5 + 0.5 * rng.uniform(size=N), m))
    idx = np.arange(N)
    for amp, nm in [(0.12, 4), (0.22, 3), (0.08, 6)]:
        z = np.full(N, 0.5)
        for r in range(1, nm + 1):
            z += (amp / r) * (rng.standard_normal() * np.sin(2 * np.pi * r * idx / N)
                              + rng.standard_normal() * np.cos(2 * np.pi * r * idx / N))
        starts.append(project_box_sum(z, m))
    perm = baseline.copy()
    perm[rng.permutation(N)] *= 1.0
    starts.append(project_box_sum(perm, m))
    unique_starts = [starts[0]]
    for s in starts[1:]:
        if all(np.max(np.abs(s - us)) > 1e-8 for us in unique_starts):
            unique_starts.append(s)
        if len(unique_starts) >= n_starts:
            break
    for h0 in unique_starts:
        c0 = score_exact(h0, N)
        if c0 < best_c:
            best_h = h0.copy()
            best_c = c0
        h_opt, c_opt = optimize_start(h0, N, m, dx, max_iter)
        if c_opt < best_c:
            best_h = h_opt.copy()
            best_c = c_opt
    for sc in hop_scales:
        hop = project_box_sum(best_h + sc * rng.standard_normal(N), m)
        c_hop = score_exact(hop, N)
        if c_hop < best_c:
            best_h = hop.copy()
            best_c = c_hop
        h_opt, c_opt = optimize_start(hop, N, m, dx, max_iter)
        if c_opt < best_c:
            best_h = h_opt.copy()
            best_c = c_opt
    best_h = project_box_sum(best_h, m)
    c_final = score_exact(best_h, N)
    c_base = score_exact(baseline, N)
    if c_base < c_final:
        best_h = baseline.copy()
        c_final = c_base
    if not np.isfinite(c_final):
        best_h = np.full(N, 0.5)
        c_final = score_exact(best_h, N)
    return best_h, float(c_final), int(N)
