import numpy as np

def project_K(y, n, S):
    if not np.all(np.isfinite(y)):
        return np.full(n, S / n)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    e = S - h.sum()
    if abs(e) > 1e-12:
        if e > 0:
            for j in range(n):
                a = min(1.0 - h[j], e)
                h[j] += a
                e -= a
                if e < 1e-14:
                    break
        else:
            for j in range(n):
                a = min(h[j], -e)
                h[j] -= a
                e += a
                if e > -1e-14:
                    break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1.0 + 1e-10 or abs(h.sum() - S) > 1e-8:
        return np.full(n, S / n)
    return h

def exact_score(h, dx):
    if h.ndim != 1 or not np.all(np.isfinite(h)):
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    if len(q) != 2 * len(h) - 1 or not np.all(np.isfinite(q)):
        return float('inf')
    return float(q.max())

def feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def make_direction(rng, n, mode):
    if mode > 0.75 or n < 4:
        z = rng.standard_normal(n)
        z -= z.mean()
        z[-1] = -z[:-1].sum()
    else:
        k = max(2, min(8, n))
        idx = rng.choice(n, size=k, replace=False)
        z = np.zeros(n)
        vals = rng.standard_normal(k)
        vals -= vals.mean()
        vals[-1] = -vals[:-1].sum()
        z[idx] = vals
    mx = np.max(np.abs(z))
    if mx < 1e-12:
        return None
    return z / mx

        else:

    fb_s = exact_score(fb_h, dx)
