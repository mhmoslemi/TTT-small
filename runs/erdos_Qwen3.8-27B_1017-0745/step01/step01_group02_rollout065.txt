def lag_value_gradient(h, s, n):
    dx = 2.0 / n
    lo_i = max(0, -s)
    hi_i = min(n, n - s)
    q = 0.0
    g = np.zeros(n)
    for i in range(lo_i, hi_i):
        j = i + s
        q += h[i] * (1.0 - h[j])
        g[i] += (1.0 - h[j])
        g[j] -= h[i]
    q *= dx
    g *= dx
    return q, g

def solve_mm_subproblem(x, n):

    L = 2.0 * dx
    n_lags = 2 * n - 1
    qJ = np.zeros(n_lags)
    GJ = np.zeros((n_lags, n))
    for idx, s in enumerate(range(-(n - 1), n)):
        qJ[idx], GJ[idx] = lag_value_gradient(x, s, n)
    
    _MM['x'] = x
    _MM['L'] = L
    _MM['n'] = n
    _MM['qJ'] = qJ
    _MM['GJ'] = GJ

    z0 = np.append(x.copy(), t0)
