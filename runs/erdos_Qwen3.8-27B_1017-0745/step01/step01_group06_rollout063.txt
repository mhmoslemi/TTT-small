from scipy.optimize import minimize

def project_K(y, n, S):
    if not np.all(np.isfinite(y)) or len(y) != n:
        return np.full(n, S / n)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    e = S - h.sum()
    if abs(e) > 1e-12:
        if e > 0:
            idx = np.where(1.0 - h > 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) > 0:
            h[idx] = np.clip(h[idx] + e / len(idx), 0.0, 1.0)
    if abs(h.sum() - S) > 1e-8:
        return np.full(n, S / n)
    return h

def exact_score(h, dx):
    if h is None or not np.all(np.isfinite(h)) or len(h) < 2:
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def feasible(h, n, S):
    return (isinstance(h, np.ndarray) and h.shape == (n,) and np.all(np.isfinite(h))
            and h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def lag_vec(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_mat(h, n, dx):
    G = np.zeros((2 * n - 1, n))
    j = np.arange(n)
    for k in range(2 * n - 1):
        ell = k - (n - 1)
        i1 = j - ell
        v1 = (i1 >= 0) & (i1 < n)
        i2 = j + ell
        v2 = (i2 >= 0) & (i2 < n)
        G[k, v1] += dx * (1.0 - h[i1[v1]])
        G[k, v2] -= dx * h[i2[v2]]
    return G

def tr_obj(z, eta, n):
    d = z[:n]
    return z[n] + 0.5 / eta * d @ d

def tr_obj_jac(z, eta, n):
    d = z[:n]
    g = np.zeros(n + 1)
    g[:n] = d / eta
    g[n] = 1.0
    return g

def tr_ineq(z, q, G, n):
    d = z[:n]
    return z[n] - q - G @ d

def tr_ineq_jac(z, q, G, n):
    nj = 2 * n - 1
    J = np.zeros((nj, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def tr_eq(z, n):
    return z[:n].sum()

def tr_eq_jac(z, n):
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def transfer(h_p, n0, m):
    y = np.zeros(m)
    for jj in range(m):
        lo = max(0, int(jj * n0 / m))
        hi = min(n0, int((jj + 1) * n0 / m) + 1)

            b = min((ii + 1) / n0, (jj + 1) / m)
            if b > a:
                y[jj] += h_p[ii] * (b - a)
        y[jj] *= m  # normalize by cell width 1/m (in u-space)
    return y
