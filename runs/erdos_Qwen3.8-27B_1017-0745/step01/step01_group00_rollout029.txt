def lag_grad(h, d, k, dx, n):

    b = max(-d, 0)

    g = 0.0

        q = b + r

    if b <= k < b + m:
        r = k - b
        p = a + r
        if 0 <= p < n:

def golden_min(f, a, b, iters=52):
    if b - a < 1e-15:
        return a
    gr = (1.0 + 5.0**0.5) / 2.0
    c = b - (b - a) / gr
    d2 = a + (b - a) / gr
    fc, fd = f(c), f(d2)
    for _ in range(iters):
        if fc < fd:
            b, d2, fd = d2, c, fc
            c = b - (b - a) / gr
            fc = f(c)
        else:
            a, c, fc = c, d2, fd
            d2 = a + (b - a) / gr
            fd = f(d2)
    t = 0.5 * (a + b)
    # check endpoints
    best_t, best_f = t, f(t)
    for tp in (a, b):
        fp = f(tp)
        if fp < best_f:
            best_t, best_f = tp, fp
    return best_t

import numpy as np

def project_capped(z, n):
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for i in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[i])
            h[i] += d
            r -= d
        else:
            d = min(-r, h[i])
            h[i] -= d
            r += d
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    if abs(np.sum(h) - s) > 1e-10 * max(1, n):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def lag_grad(h, d, k, dx, n):
    a = max(d, 0)
    b = max(-d, 0)
    m = n - abs(d)
    g = 0.0
    if a <= k < a + m:
        q = k - d
        if 0 <= q < n:
            g += dx * (1.0 - h[q])
    if b <= k < b + m:
        p = k + d
        if 0 <= p < n:
            g -= dx * h[p]
    return g

def golden_min(f, a, b, iters=52):
    if b - a < 1e-15:
        return a
    gr = (1.0 + 5.0 ** 0.5) / 2.0
    c = b - (b - a) / gr
    d2 = a + (b - a) / gr
    fc, fd = f(c), f(d2)
    for _ in range(iters):
        if fc < fd:
            b, d2, fd = d2, c, fc
            c = b - (b - a) / gr
            fc = f(c)
        else:
            a, c, fc = c, d2, fd
            d2 = a + (b - a) / gr
            fd = f(d2)
    best_t, best_f = 0.5 * (a + b), f(0.5 * (a + b))
    for tp in (a, b):
        fp = f(tp)
        if fp < best_f:
            best_t, best_f = tp, fp
    return best_t

def pair_line_min(h, lags, i, j, n):
    dx = 2.0 / n
    D = lags.copy()
    L = max(-h[i], h[j] - 1.0)
    U = min(1.0 - h[i], h[j])
    if U - L < 1e-14:
        return None
    A = np.zeros(2 * n - 1)
    A[n - 1] = -2.0 * dx
    r = abs(i - j)
    if r > 0:
        A[n - 1 + r] = dx
        A[n - 1 - r] = dx
    B = np.zeros(2 * n - 1)
    for d in range(-(n - 1), n):
        kd = d + n - 1
        B[kd] = lag_grad(h, d, i, dx, n) - lag_grad(h, d, j, dx, n)

    def eval_all(t):
        return A * t * t + B * t + D

    def G(t):
        v = eval_all(t)
        v2 = v.copy()
        v2[n - 1] = -1e30
        return float(np.max(v2))

    def Q0(t):
        return A[n - 1] * t * t + B[n - 1] * t + D[n - 1]

    def H(t):
        return G(t) - Q0(t)

    tG = golden_min(G, L, U)
    tH = golden_min(H, L, U)
    cands = [L, U, max(L, min(U, 0.0)), tG, tH]
    hH = H(tH)
    if hH < 0:
        hL, hU = H(L), H(U)
        if hL > 0:
            lo2, hi2 = L, tH
            for _ in range(52):
                mid = 0.5 * (lo2 + hi2)
                if H(mid) > 0:
                    lo2 = mid
                else:
                    hi2 = mid
            cands.append(0.5 * (lo2 + hi2))
            cands.append(golden_min(G, L, 0.5 * (lo2 + hi2)))
        if hU > 0:
            lo2, hi2 = tH, U
            for _ in range(52):
                mid = 0.5 * (lo2 + hi2)
                if H(mid) < 0:
                    lo2 = mid
                else:
                    hi2 = mid
            cands.append(0.5 * (lo2 + hi2))
            cands.append(golden_min(G, 0.5 * (lo2 + hi2), U))
    best_t, best_val = 0.0, float('inf')
    for t in cands:
        if not np.isfinite(t):
            continue
        t = max(L, min(U, t))
        v = eval_all(t)
        val = float(np.max(v))
        if val < best_val - 1e-15 or (abs(val - best_val) < 1e-15 and abs(t) < abs(best_t)):
            best_t, best_val = t, val
    return best_t

    if not np.isfinite(score):

    dx = 2.0 / n
    if n <= 128:
        all_ii, all_jj = np.triu_indices(n, 1)
        all_pairs = list(zip(all_ii.tolist(), all_jj.tolist()))
    else:
        all_pairs = None
    for _ in range(max_moves):
        lags = np.correlate(h, 1.0 - h, mode='full') * dx
        idx_sorted = np.argsort(-lags)
        screen_kds = idx_sorted[:min(8, 2 * n - 1)]
        merits = []
        candidates = all_pairs if all_pairs else []
        if all_pairs is None:
            seen = set()
            for _ in range(min(3000, 6 * n)):
                a, b = int(rng.integers(0, n)), int(rng.integers(0, n))
                if a == b:
                    continue
                if a > b:
                    a, b = b, a
                if (a, b) not in seen:
                    seen.add((a, b))
                    candidates.append((a, b))
        for (a, b2) in candidates:
            Lm = max(-h[a], h[b2] - 1.0)
            Um = min(1.0 - h[a], h[b2])
            if Um - Lm < 1e-12:
                continue
            merit = float('inf')
            for kd in screen_kds:
                d = kd - (n - 1)
                g = lag_grad(h, d, a, dx, n) - lag_grad(h, d, b2, dx, n)
                if Lm < -1e-14:
                    merit = min(merit, -g)
                if Um > 1e-14:
                    merit = min(merit, g)
            if np.isfinite(merit):
                merits.append((merit, a, b2))
        merits.sort(key=lambda x: x[0])
        for _ in range(n_rnd):
            a = int(rng.integers(0, n))
            b2 = int(rng.integers(0, n))
            if a != b2:
                if a > b2:
                    a, b2 = b2, a
                merits.append((0.0, a, b2))
        best_trial, best_sc = None, score
        tried = min(n_scr, len(merits))
        for idx in range(tried):
            a, b2 = merits[idx][1], merits[idx][2]
            t = pair_line_min(h, lags, a, b2, n)
            if t is None or abs(t) < 1e-12:
                continue
            h2 = h.copy()
            h2[a] += t
            h2[b2] -= t
            h2 = project_capped(h2, n)
            if h2 is None:
                continue
            sc = exact_score(h2, n)
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                best_sc, best_trial = sc, h2
        if best_trial is None:
            break
        h, score = best_trial, best_sc
    return h, score
