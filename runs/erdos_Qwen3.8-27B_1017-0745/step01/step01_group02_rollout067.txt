            if y.value is not None and np.all(np.isfinite(np.asarray(y.value, dtype=np.float64))):
                return np.asarray(y.value, dtype=np.float64)
        except Exception:
            continue
    return None

def local_mm(h0, n, niter):
    h = np.array(h0, dtype=np.float64).copy()
    c = exact_score(h, n)
    for _ in range(niter):
        q, G = lag_data(h, n)
        y = mm_subproblem(h, n, q, G)
        if y is None:
            break
        yp = project_box_sum(y, 0.5 * n)
        if yp is None or np.max(np.abs(yp - h)) < 1e-10:
            break
        best_c, best_a = c, -1.0
        for k in range(10):
            a = 2.0 ** (-k)
            cand = project_box_sum(h + a * (yp - h), 0.5 * n)
            if cand is None:
                continue
            cc = exact_score(cand, n)
            if cc < c - 1e-13:
                best_c = cc
                best_a = a
                break
        if best_a < 0:
            break
        h = project_box_sum(h + best_a * (yp - h), 0.5 * n)
        c = best_c
    return h, c

def get_parent():
    n = 45
    if 'initial_h_values' in globals():
        ph = np.asarray(globals()['initial_h_values'], dtype=np.float64)
        if ph.ndim == 1 and 2 <= len(ph) <= 999 and np.all(np.isfinite(ph)):
            if np.all(ph >= -1e-6) and np.all(ph <= 1.0 + 1e-6):
                n = len(ph)
                p = project_box_sum(np.clip(ph, 0.0, 1.0), 0.5 * n)
                if p is not None and check_feas(p, n):
                    return p.copy(), n
    return project_box_sum(np.full(n, 0.5), 0.5 * n), n

def build_starts(parent, n, rng):
    starts = [parent.copy(), (1.0 - parent).copy(), parent[::-1].copy()]
    for sd in [0.05, 0.12, 0.25]:
        p = project_box_sum(parent + sd * rng.standard_normal(n), 0.5 * n)
        if p is not None:
            starts.append(p)
    p = project_box_sum(rng.random(n), 0.5 * n)
    if p is None:
        p = project_box_sum(np.full(n, 0.5) + 0.1 * rng.standard_normal(n), 0.5 * n)
    if p is not None:
        starts.append(p)
    idx = np.arange(n)
    u = 0.5 + 0.30 * np.cos(2.0 * np.pi * idx / n + 0.7) + 0.15 * np.sin(4.0 * np.pi * idx / n + 1.4)
    p = project_box_sum(u, 0.5 * n)
    if p is not None:
        starts.append(p)
    unique = []
    for s in starts:
        if not any(np.max(np.abs(s - u2)) < 1e-8 for u2 in unique):
            unique.append(s)
    return unique

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent, n = get_parent()
    best_h = parent.copy()
    best_c = exact_score(parent, n)
    starts = build_starts(parent, n, rng)
    niter = max(8, min(25, 900 // n))
    for h0 in starts:
        try:
            h_new, c_new = local_mm(h0, n, niter)
            if c_new < best_c:
                best_h = np.array(h_new, dtype=np.float64).copy()
                best_c = c_new
        except Exception:
            pass
    for sd in [0.03, 0.10, 0.22]:
        h0 = project_box_sum(best_h + sd * rng.standard_normal(n), 0.5 * n)
        if h0 is None:
            continue
        try:
            h_new, c_new = local_mm(h0, n, max(5, niter // 2))
            if c_new < best_c:
                best_h = np.array(h_new, dtype=np.float64).copy()
                best_c = c_new
        except Exception:
            pass
    final = project_box_sum(best_h, 0.5 * n)
    if final is not None and check_feas(final, n):
        best_h = final
        best_c = exact_score(final, n)
    else:
        best_h = parent.copy()
        best_c = exact_score(parent, n)
    if not np.isfinite(best_c):
        fb = project_box_sum(np.full(n, 0.5), 0.5 * n)
        if fb is None:
            fb = np.full(n, 0.5)
        best_h = fb
        best_c = exact_score(fb, n)
    return (best_h, best_c, n)
