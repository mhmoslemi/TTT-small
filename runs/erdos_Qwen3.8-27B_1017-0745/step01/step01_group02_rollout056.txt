import numpy as np
from scipy.optimize import minimize

_mm = {}  # global state for MM subproblem

def mm_obj(z):
    y = z[:-1]
    x = _mm['x']
    L = _mm['L']
    d = y - x
    return float(0.5 * L * np.dot(d, d) + z[-1])

def mm_obj_grad(z):
    y = z[:-1]
    x = _mm['x']
    L = _mm['L']
    n = len(x)
    g = np.zeros(n + 1)
    g[:n] = L * (y - x)
    g[n] = 1.0
    return g

def mm_ineq(z):
    y = z[:-1]
    d = y - _mm['x']
    qj = _mm['qj']
    gj = _mm['gj']  # shape (m, n)
    return z[-1] - qj - gj @ d

def mm_ineq_grad(z):
    n = _mm['x'].shape[0]
    gj = _mm['gj']
    m = gj.shape[0]
    J = np.zeros((m, n + 1))
    J[:, :n] = -gj
    J[:, n] = 1.0
    return J

def mm_eq(z):
    n = _mm['x'].shape[0]
    return float(np.sum(z[:n])) - 0.5 * n

def mm_eq_grad(z):
    n = _mm['x'].shape[0]
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def lag_grad(h, s, n):

    lo = max(0, -s)
    hi = min(n, n - s)
    q = 0.0
    g = np.zeros(n)

        q += h[i] * (1.0 - h[j])
