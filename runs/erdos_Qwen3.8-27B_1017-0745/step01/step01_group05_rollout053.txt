The term min_d (c(p)·d + (L/2)||d||^2) is concave in p (minimum of affine functions is concave... wait, no. The minimum of linear functions is concave, but here the linear functions depend on p through c(p) = G^T p, and the quadratic term is fixed in d).

Actually, for fixed d, (G^T p)·d + (L/2)||d||^2 is affine in p. The minimum over d of affine functions of p is concave. Plus p·q is linear (both convex and concave). So D(p) is concave. Good.

So we're doing concave maximization on a simplex. Projected gradient ascent (with acceleration) is appropriate.

The gradient is ∇D(p) = q + G·d*(p) where d* is the argmin. This is what I have.

Now, for the step size in the dual: the Lipschitz constant of ∇D is ||G||_F / L (from the chain rule: the gradient of the projection is bounded by 1, and G^T p changes by ||G||_F * ||Δp||, divided by L gives the change in the projection input, and the projection is 1-Lipschitz, so the change in d is at most ||G||_F * ||Δp|| / L, and the change in G·d is at most ||G||_F * ||Δd|| ≤ ||G||_F^2 * ||Δp|| / L).

So L_dual = ||G||_F^2 / L. Step size = 1/L_dual.

def project_capped(y, n):

    hi = float(y.max())
    lam = float(np.clip(y.mean() - 0.5, lo, hi))

            lo = lam
        else:
            hi = lam
        a = np.count_nonzero((y - lam > 0) & (y - lam < 1))

        else:

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and np.all(h >= -1e-12)
            and np.all(h <= 1.0 + 1e-12) and abs(h.sum() - n / 2.0) <= 1e-10)

def lag_scores(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    qs = lag_scores(h, n)
    if not np.all(np.isfinite(qs)):
        return float('inf')
    return float(qs.max())

def build_G(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        row = k + n - 1
        lo = max(0, k)
        hi = min(n, n + k)
        if hi > lo:
            G[row, lo:hi] += dx * (1.0 - h[lo - k:hi - k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if hi2 > lo2:
            G[row, lo2:hi2] -= dx * h[lo2 + k:hi2 + k]
    return G

def project_simplex(z):
    m = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(m, 1.0 / m)
    s = np.sort(z)[::-1]
    css = np.cumsum(s)
    idx = np.arange(1, m + 1)
    rho = np.where(s - (css - 1.0) / idx > 0)[0]
    if len(rho) == 0:
        return np.full(m, 1.0 / m)
    r = rho[0]
    theta = (css[r] - 1.0) / (r + 1)
    out = np.maximum(z - theta, 0.0)
    r2 = 1.0 - out.sum()
    if abs(r2) > 1e-15:
        out[np.argmax(out)] += r2
    return out

def dual_eval(p, h, G, q, L):
    gmix = G.T @ p
    y, ok = project_capped(h - gmix / L, len(h))
    if not ok:
        return None
    d = y - h
    u = q + G @ d
    Dval = p @ u + 0.5 * L * (d @ d)
    gap = u.max() - p @ u
    return d, u, Dval, gap

def solve_majorant(h, q, G, n, max_inner):
    dx = 2.0 / n
    L = 2.0 * dx
    m = len(q)
    Ldual = max(1e-12, 1.05 * float(np.sum(G * G)) / L)
    qmax = float(q.max())
    tol = max(1e-12, 1e-8 * max(1.0, abs(qmax)))
    active = np.where(q >= qmax - tol)[0]
    p = np.zeros(m)
    if len(active) > 0:
        p[active] = 1.0 / len(active)
    else:
        p[int(np.argmax(q))] = 1.0
    x = p.copy()
    yv = p.copy()
    theta = 1.0
    best_D = -1e18
    best_d = None
    consec_small = 0
    for it in range(max_inner):
        ds = dual_eval(yv, h, G, q, L)
        if ds is None:
            break
        d_y, u_y, D_y, gap_y = ds
        if D_y > best_D:
            best_D = D_y
            best_d = d_y.copy()
        trial = project_simplex(yv + u_y / Ldual)
        dt = dual_eval(trial, h, G, q, L)
        if dt is None:
            break
        d_t, u_t, D_t, gap_t = dt
        restarted = False
        if D_t < D_y - 1e-12:
            trial2 = project_simplex(x + u_y / Ldual)
            dt2 = dual_eval(trial2, h, G, q, L)
            if dt2 is not None and dt2[2] >= D_y - 1e-12:
                trial = trial2
                d_t, u_t, D_t, gap_t = dt2
            else:
                theta = 1.0
                yv = project_simplex(x + u_y / Ldual)
                x = yv.copy()
                restarted = True
        if not restarted:
            if D_t > best_D:
                best_D = D_t
                best_d = d_t.copy()
            x_new = trial
            snorm = float(np.max(np.abs(x_new - x)))
            if snorm < 1e-10:
                consec_small += 1
            else:
                consec_small = 0
            theta_new = (1.0 + math.sqrt(1.0 + 4.0 * theta * theta)) / 2.0
            yv = x_new + ((theta - 1.0) / theta_new) * (x_new - x)
            x = x_new
            theta = theta_new
        if gap_t <= max(1e-10, 1e-8 * max(1.0, abs(float(u_t.max())))):
            break

            break
    if best_d is not None:

    return None

def optimize_start(h0, n, max_outer, max_inner):
    dx = 2.0 / n
    L = 2.0 * dx
    h, ok = project_capped(h0.copy(), n)
    if not ok:
        return h0.copy(), exact_score(h0, n)
    best_h = h.copy()
