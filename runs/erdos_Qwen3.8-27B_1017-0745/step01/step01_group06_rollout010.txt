    if not np.all(np.isfinite(y)):
        return np.full(n, S / n)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid

    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

        row[v1] = dx * (1.0 - h[i1[v1]])

    return G

def project_K(y, n, S):
    if not np.all(np.isfinite(y)):
        return np.full(n, S / n)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    err = S - h.sum()
    if abs(err) > 1e-12:
        if err > 0:
            idx = np.where(1.0 - h > 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) > 0:
            h[idx] = np.clip(h[idx] + err / len(idx), 0.0, 1.0)
    if abs(h.sum() - S) > 1e-9:
        return np.full(n, S / n)

def exact_score(h, dx):
    if not np.all(np.isfinite(h)):

def feasible(h, n, S):

            h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def lag_vec(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def build_G(h, n, dx):
    G = np.zeros((2 * n - 1, n))
    j = np.arange(n)
    for k in range(2 * n - 1):
        ell = k - (n - 1)
        i1 = j - ell
        v1 = (i1 >= 0) & (i1 < n)
        i2 = j + ell
        v2 = (i2 >= 0) & (i2 < n)
        G[k, v1] = dx * (1.0 - h[i1[v1]])
        G[k, v2] -= dx * h[i2[v2]]
    return G
