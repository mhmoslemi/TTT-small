    if not np.isfinite(m): return inf

    if s<=0: return inf

def fix_mass(h, n, S):
    h = np.clip(np.asarray(h, dtype=np.float64).copy(), 0.0, 1.0)
    for _ in range(8):
        e = S - float(np.sum(h))
        if abs(e) < 1e-12:
            break
        if e > 0.0:
            idx = np.where(h < 1.0 - 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if idx.size == 0:
            h.fill(np.clip(S / n, 0.0, 1.0))

    e = S - float(np.sum(h))

        i = int(np.argmax(1.0 - h)) if e > 0 else int(np.argmin(h))

def project_box(y, n, S):
    if y is None or np.asarray(y).shape != (n,):

    y = np.asarray(y, dtype=np.float64).copy()

        return fix_mass(y, n, S)
    lo, hi = float(np.min(y)) - 1.0, float(np.max(y)) + 1.0

        else:

def fix_mass(h,n,S):
    h=np.clip(np.asarray(h,dtype=np.float64).copy(),0.0,1.0)
    for _ in range(8):

        if idx.size==0:

def project_box(y,n,S):
