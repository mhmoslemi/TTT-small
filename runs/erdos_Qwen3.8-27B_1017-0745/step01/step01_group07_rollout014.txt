import numpy as np

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1+1e-12)
            and abs(np.sum(h) - n/2.0) <= 1e-10)

def proj(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)):
        return None
    if S < 0 or S > len(z) or np.max(np.abs(z)) > 1e12:
        return None
    n = len(z)
    lo, hi = np.min(z)-1.0, np.max(z)
    for _ in range(80):
        mid = 0.5*(lo+hi)
        if np.sum(np.clip(z-mid, 0, 1)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(z - 0.5*(lo+hi), 0, 1)
    r = S - np.sum(y)
    for _ in range(2):
        if abs(r) < 1e-15:
            break
        if r > 0:
            for i in range(n):
                a = min(r, 1.0-y[i]); y[i] += a; r -= a
                if r <= 0: break
        else:
            for i in range(n):
                a = min(-r, y[i]); y[i] -= a; r += a
                if r >= 0: break
    if abs(np.sum(y)-S) > 1e-10 or not np.all(np.isfinite(y)):
        return None
    return np.clip(y, 0, 1)

def do_score(h):
    n = len(h)
    v = np.correlate(h, 1-h, mode='full') * (2.0/n)
    return float(np.max(v)) if np.all(np.isfinite(v)) else None

def do_lags(h):
    return np.correlate(h, 1-h, mode='full') * (2.0/len(h))

def line_search(h, q, sc_now, p, r, n):
    dx = 2.0/n
    L = max(-h[p], h[r]-1.0)
    U = min(1.0-h[p], h[r])
    if U - L <= 1e-12:
        return None
    b = np.zeros(2*n-1)
    for di in range(2*n-1):
        d = di - (n-1)
        s1 = 0.0
        if 0 <= p+d < n: s1 += 1.0-h[p+d]
        if 0 <= r+d < n: s1 -= 1.0-h[r+d]
        s2 = 0.0
        if 0 <= p-d < n: s2 += h[p-d]
        if 0 <= r-d < n: s2 -= h[r-d]
        b[di] = dx*(s1-s2)
    a = np.zeros(2*n-1)
    a[n-1] = -2.0*dx
    k = abs(r-p)
    if k < n:
        a[n-1+k] = dx
        a[n-1-k] = dx
    ts = np.linspace(L, U, 17)
    if L <= 0.0 <= U:
        ts = np.unique(np.concatenate([ts, [0.0]]))
    t_best = ts[0]
    f_best = float(np.max(q + b*ts[0] + a*ts[0]*ts[0]))
    for t in ts[1:]:
        ft = float(np.max(q + b*t + a*t*t))
        if ft < f_best - 1e-14 or (ft <= f_best + 1e-14 and abs(t) < abs(t_best)):
            f_best = ft; t_best = t
    radius = (U-L)/16.0
    for _ in range(4):
        lo_r = max(L, t_best-radius)
        hi_r = min(U, t_best+radius)
        for t in np.linspace(lo_r, hi_r, 5):
            ft = float(np.max(q + b*t + a*t*t))
            if ft < f_best - 1e-14 or (ft <= f_best + 1e-14 and abs(t) < abs(t_best)):
                f_best = ft; t_best = t
        radius /= 4.0
    if f_best >= sc_now - 1e-12:
        return None
    h_new = h.copy()
    h_new[p] += t_best; h_new[r] -= t_best
    if not is_feasible(h_new, n):
        h_new = proj(h_new, n/2.0)
        if h_new is None:
            return None
    sc = do_score(h_new)
    if sc is None or sc >= sc_now - 1e-12:
        return None
    return h_new, sc, do_lags(h_new)

def build_starts(n, S, parent, rng):
    starts = []
    const = np.full(n, 0.5)
    starts.append(const)
    if parent is not None:
        starts.append(parent.copy())
        starts.append(parent[::-1].copy())
        c = 1.0 - parent.copy()
        starts.append(c)
    idx = np.arange(n, dtype=np.float64)
    M = min(4, (n-1)//2)
    freq_vec = np.zeros(n)
    for m in range(1, M+1):
        phi = float(rng.uniform(0, 2*np.pi))
        amp = float(rng.normal(0, 0.12/np.sqrt(m)))
        freq_vec += amp*np.cos(2*np.pi*m*(idx+0.5)/n + phi)
    u = 0.5 + freq_vec
    pu = proj(u, S)
    if pu is not None:
        starts.append(pu)
    perm = rng.permutation(n)
    binv = np.zeros(n)
    binv[perm[:n//2]] = 1.0
    if n % 2 == 1:
        binv[perm[n//2]] = 0.5
    starts.append(binv.copy())
    uinv = rng.uniform(0, 1, n)
    pu2 = proj(uinv, S)
    if pu2 is not None:
        starts.append(pu2)
    deduped = []
    for s in starts:
        dup = False
        for d in deduped:
            if np.max(np.abs(s-d)) < 1e-10:
                dup = True; break
        if not dup and is_feasible(s, n):
            deduped.append(s)
    cap = 6 if n <= 128 else (4 if n <= 256 else (3 if n <= 512 else 2))
    return deduped[:cap]

def anneal_traj(h, sc, q, n, rng, E, K):
    S = n/2.0
    scale = 0.12
    for e in range(E):
        T = max(5e-5, 3e-3*(0.4**e))
        accepted = 0
        for _ in range(K):
            p = int(rng.integers(0, n))
            r = int(rng.integers(0, n))
            if p == r:
                continue
            t = float(rng.uniform(-scale, scale))
            L = max(-h[p], h[r]-1.0)
            U = min(1.0-h[p], h[r])
            if t < L or t > U:
                continue
            h_new = h.copy()
            h_new[p] += t; h_new[r] -= t
            if not is_feasible(h_new, n):
                h_new = proj(h_new, S)
                if h_new is None:
                    continue
            sc_new = do_score(h_new)
            if sc_new is None:
                continue
            delta = sc_new - sc
            if delta <= 0:
                accept = True
            else:
                x = -delta/T
                accept = (x > -700 and rng.random() < np.exp(x))
            if accept:
                h = h_new; sc = sc_new; q = do_lags(h_new)
                accepted += 1
        ratio = accepted/max(K, 1)
        if ratio < 0.12:
            scale = min(0.4, 1.25*scale)
        elif ratio > 0.55:
            scale = max(0.01, 0.75*scale)
    return h, sc, q

def polish(h, sc, q, n, rng, n_sweeps):
    S = n/2.0
    P = min(n*(n-1)//2, 2*n, max(24, 80000//n))
    pairs = []
    n_adj = min(n-1, max(1, P//2))
    adj_indices = np.linspace(0, n-2, n_adj).astype(int)
    for i in adj_indices:
        pairs.append((int(i), int(i+1)))
    remaining = max(0, P - len(pairs))
    attempts = 0
    while len(pairs) < P and attempts < remaining*5:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            attempts += 1; continue
        if i > j: i, j = j, i
        if (i, j) not in pairs:
            pairs.append((i, j))
        attempts += 1
    if len(pairs) < P:
        for i in range(n):
            for j in range(i+1, n):
                if len(pairs) >= P: break
                if (i, j) not in pairs:
                    pairs.append((i, j))
            if len(pairs) >= P: break
    rng.shuffle(pairs)
    for sweep in range(n_sweeps):
        any_move = False
        for (p, r) in pairs:
            res = line_search(h, q, sc, p, r, n)
            if res is not None:
                h, sc, q = res
                any_move = True
        if not any_move:
            break
    return h, sc, q

def run(seed=42, **kwargs):

        seed = 42
    try:

    except (TypeError, ValueError):

                        parent = iv_arr.copy()
                    else:
                        p = proj(iv_arr, S)
                        if p is not None and is_feasible(p, n):
                            parent = p
        except Exception:
            pass
    if n is None:
        n = kwargs.get('n_points', 80)
        try:
            n = int(n)
        except (TypeError, ValueError):
            n = 80
        if n < 8 or n > 999:
            n = 80
    S = n/2.0
    h_const = np.full(n, 0.5)
    sc_const = do_score(h_const)
    if sc_const is None:
        sc_const = 0.5
    best_h = h_const.copy()
    best_sc = sc_const
    parent_sc = None
    if parent is not None:
        parent_sc = do_score(parent)

            best_sc = parent_sc; best_h = parent.copy()
    starts = build_starts(n, S, parent, rng)
    for s in starts:
        sc = do_score(s)
        if sc is not None and sc < best_sc:

    n_epochs = 5 if n <= 128 else (4 if n <= 256 else (3 if n <= 512 else 2))
    K = min(1000, max(80, 80000//n))

            if sc_a < best_sc:

    except Exception:

    if not is_feasible(best_h, n):

    final_sc = do_score(best_h)
