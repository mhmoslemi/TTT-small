            r *= 0.5; fails += 1; continue
        if np.max(np.abs(endpoint - cur)) < 1e-12:
            r *= 0.5; fails += 1; continue
        ht, st = line_minimax(cur, endpoint, n, S)
        if st < best_sc - 1e-12:
            best = ht.copy(); best_sc = st
            cur = ht.copy(); fails = 0
            r = min(0.25, 1.2*r)
        else:
            r *= 0.5; fails += 1
    return best, best_sc

def subgradient_refine(h, n, S, iters=200, lr0=0.01):
    best = h.copy(); best_sc = exact_score(h, n)
    cur = h.copy()
    for it in range(iters):
        corr = np.correlate(cur, 1.0-cur, mode='full')
        k_max = int(np.argmax(corr)) - (n-1)
        k = k_max
        i_lo = max(0,k); i_hi = min(n-1, n-1+k)
        if i_lo > i_hi: break
        i = np.arange(i_lo, i_hi+1); j = i-k
        grad = np.zeros(n)
        grad[i] += (1.0-cur[j])
        grad[j] -= cur[i]
        lr = lr0/(1.0+0.05*it)
        prop = cur - lr*grad
        prop -= np.mean(prop - S/n)
        prop = project_box_sum(prop, S)
        if prop is None or not check_feasible(prop, n): break
        sc = exact_score(prop, n)
        if sc < best_sc - 1e-13:
            best = prop.copy(); best_sc = sc
            cur = prop.copy()
    return best, best_sc

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    n = 45
    parent = None
    try:
        pv = np.array(initial_h_values, dtype=np.float64)
        if pv.ndim==1 and 2<=len(pv)<=999 and np.all(np.isfinite(pv)):
            if np.all(pv>=-1e-9) and np.all(pv<=1.0+1e-9):
                nn = len(pv); tgt = nn/2.0
                if abs(np.sum(pv)-tgt) < 1e-5*max(1.0,float(nn)):
                    pv = np.clip(pv, 0.0, 1.0)
                    parent = project_box_sum(pv, tgt)
                    if parent is not None: n = nn
    except Exception:
        pass
    if parent is None:
        n = 45; parent = project_box_sum(np.full(n, 0.5), n/2.0)
    S = n/2.0
    starts = [parent.copy(), parent[::-1].copy(), (1.0-parent).copy()]
    starts.append(np.roll(parent, n//3).copy())
    for sig in [0.04, 0.10, 0.18]:
        p = parent + sig*rng.standard_normal(n)
        c = project_box_sum(p, S)
        if c is not None: starts.append(c)
    for m, ph in [(1, 0.5), (2, 1.3), (3, 2.1)]:
        x = np.arange(n)
        u = 0.5 + 0.15*np.cos(2.0*np.pi*m*x/n + ph)
        c = project_box_sum(u, S)
        if c is not None: starts.append(c)
    deduped = [starts[0]]
    for s in starts[1:]:
        if not any(np.max(np.abs(s-d))<1e-10 for d in deduped):
            deduped.append(s)
    starts = deduped
    best_h = parent.copy(); best_sc = exact_score(best_h, n)
    use_lp = n <= 96
    for s in starts:
        if use_lp:
            hh, sc = local_lp_search(s, n, S, max_iters=8 if n<=64 else 5)
        else:
            hh, sc = subgradient_refine(s, n, S, iters=100)
        if sc < best_sc - 1e-13:
            best_sc = sc; best_h = hh.copy()
    best_h, best_sc = subgradient_refine(best_h, n, S, iters=300, lr0=0.008)
    for am in [0.005, 0.015, 0.04, 0.08]:
        for _ in range(3):
            p = best_h + am*rng.standard_normal(n)
            c = project_box_sum(p, S)
            if c is None: continue
            hh, sc = subgradient_refine(c, n, S, iters=150, lr0=0.005)
            if sc < best_sc - 1e-13:
                best_sc = sc; best_h = hh.copy()
            if use_lp:
                hh2, sc2 = local_lp_search(c, n, S, max_iters=4)
                if sc2 < best_sc - 1e-13:
                    best_sc = sc2; best_h = hh2.copy()
    fh = project_box_sum(best_h, S)
    if fh is None or not check_feasible(fh, n):
        fh = project_box_sum(parent, S)
    fs = exact_score(fh, n)
    if fs >= best_sc:
        fh = project_box_sum(best_h, S)
        fs = exact_score(fh, n)
    if not np.isfinite(fs) or fh is None:
        fh = parent; fs = exact_score(fh, n)
    return (np.array(fh, dtype=np.float64), float(fs), int(n))
