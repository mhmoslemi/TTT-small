import numpy as np
import cvxpy as cp

def project_capped_simplex(y, n, S):
    if y is None or y.shape != (n,) or not np.all(np.isfinite(y)):
        return np.full(n, S / n)
    y = np.clip(y, -10, 10)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    e = S - h.sum()
    # distribute residual
    for _ in range(2):
        if abs(e) < 1e-12:
            break
        j = np.arange(n)
        if e > 0:
            room = 1.0 - h[j]
            add = np.minimum(room, max(e, 0) / max(room.sum(), 1e-15))
            h += add * np.where(room > 0, 1.0, 0.0)
        else:
            room = h[j]
            sub = np.minimum(room, max(-e, 0) / max(room.sum(), 1e-15))
            h -= sub * np.where(room > 0, 1.0, 0.0)
        e = S - h.sum()
    if abs(h.sum() - S) > 1e-10:
        return np.full(n, S / n)
    return h

def score_exact(h, n):
    if h is None or h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def is_feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and
            abs(h.sum() - S) <= 1e-8)

def build_lag_data(n, dx):
    # Returns list of (c_ell, d) for ell = -(n-1), ..., 0, ..., (n-1)
    lags = []
    for d in range(n):
        c_pos = np.zeros(n)  # c for ell = +d
        c_neg = np.zeros(n)  # c for ell = -d
        if d == 0:
            c_pos[:] = dx
            c_neg[:] = dx
        else:
            c_pos[d:] = dx
            c_neg[:n-d] = dx
        lags.append((c_pos, d))
        if d > 0:
            lags.append((c_neg, d))
    # Reorder to match np.correlate output order
    # np.correlate(h, 1-h, 'full') has lags from -(n-1) to +(n-1)
    # Actually, let me just store them in order and handle indexing
    return lags

def build_dc_factors(n, dx, lags):
    gamma = 1.25 * dx
    I = np.eye(n)
    factors = {}  # d -> (Lp, Ln)
    for d in range(n):
        B = np.zeros((n, n))
        if d == 0:
            B *= dx  # B = dx * I
        else:
            for i in range(n - d):
                B[i, i+d] = dx / 2.0
                B[i+d, i] = dx / 2.0
        P = (gamma * I + B) / 2.0
        N = (gamma * I - B) / 2.0
        try:
            Lp = np.linalg.cholesky(P)
            Ln = np.linalg.cholesky(N)
        except np.linalg.LinAlgError:
            return None
        factors[d] = (Lp, Ln)
    return factors
