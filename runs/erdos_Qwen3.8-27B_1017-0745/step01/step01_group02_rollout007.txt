def project_box_sum(v, S):

    if v.ndim != 1 or not np.all(np.isfinite(v)):

    n = len(v)

        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0

        else:

            m2, c2 = hull[-1]
            # crossing of (m1,c1) and (m2,c2): t = (c1-c2)/(m2-m1)
            # crossing of (m2,c2) and (m,c): t = (c2-c)/(m-m2)
            tc12 = (c1 - c2) / (m2 - m1)
            tc_new = (c2 - c) / (m - m2)
            if tc_new <= tc12:
                hull.pop()
            else:
                break
        hull.append((m, c))
    # Compute start points
    starts = [-np.inf]
    for k in range(1, len(hull)):
        m1, c1 = hull[k-1]
        m2, c2 = hull[k]
        starts.append((c1 - c2) / (m2 - m1))
    return list(zip(hull, starts))

def pair_line_search(h, corr, cur_score, i, j, n):
    dx = 2.0 / n
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-14:
        return None
    
    s_vals = np.arange(2 * n - 1) - (n - 1)  # s_p values
    # Compute b_p for each p
    # b_p = [valid(i-s) ? (1-h[i-s]) : 0] - [valid(j-s) ? (1-h[j-s]) : 0]
    #       - [valid(i+s) ? h[i+s] : 0] + [valid(j-s) ? h[j-s] : 0]
    b = np.zeros(2 * n - 1)
    for p in range(2 * n - 1):
        s = s_vals[p]
        val = 0.0
        if 0 <= i - s < n:
            val += (1.0 - h[i - s])
        if 0 <= j - s < n:
            val -= (1.0 - h[j - s])
        if 0 <= i + s < n:
            val -= h[i + s]
        if 0 <= j + s < n:
            val += h[j - s]
        b[p] = val
    
    # a_p: nonzero only at p=n-1 (s=0), p=n-1+(j-i) (s=j-i), p=n-1+(i-j) (s=i-j)
    a = np.zeros(2 * n - 1)
    a[n - 1] = 2.0  # s=0
    a[n - 1 + (j - i)] = -1.0  # s=j-i
    a[n - 1 + (i - j)] = -1.0  # s=i-j
    
    c = corr.copy()
    
    # Separate affine and quadratic
    quad_mask = (a != 0.0)
    affine_mask = ~quad_mask
    
    # Build hull of affine lines
    if np.any(affine_mask):
        hull = build_hull(b[affine_mask], c[affine_mask])
    else:
        hull = []
    
    # Collect candidate t values
    candidates = [lo, hi, 0.0]
    
    # Add hull start points in [lo, hi]
    for (m, c_line), start in hull:
        if lo - 1e-12 <= start <= hi + 1e-12:
            candidates.append(np.clip(start, lo, hi))
    
    # For each quadratic, find intersections with hull segments and other quadratics
    quad_indices = np.where(quad_mask)[0]
    
    # Quadratic-quad intersections
    for ki in range(len(quad_indices)):
        for kj in range(ki + 1, len(quad_indices)):
            pi, pj = quad_indices[ki], quad_indices[kj]
            # (a_i - a_j)t^2 + (b_i - b_j)t + (c_i - c_j) = 0
            A = a[pi] - a[pj]
            B = b[pi] - b[pj]
            C = c[pi] - c[pj]
            if abs(A) < 1e-16:
                if abs(B) > 1e-16:
                    t = -C / B
                    if lo - 1e-12 <= t <= hi + 1e-12:
                        candidates.append(np.clip(t, lo, hi))
            else:
                disc = B * B - 4 * A * C
                if disc >= -1e-14:
                    disc = max(0.0, disc)
                    sq = np.sqrt(disc)
                    for t in [(-B + sq) / (2 * A), (-B - sq) / (2 * A)]:
                        if np.isfinite(t) and lo - 1e-12 <= t <= hi + 1e-12:
                            candidates.append(np.clip(t, lo, hi))
    
    # Quadratic-hull intersections and vertices
    for pi in quad_indices:
        # Vertex
        if abs(a[pi]) > 1e-16:
            tv = -b[pi] / (2.0 * a[pi])
            if np.isfinite(tv) and lo - 1e-12 <= tv <= hi + 1e-12:
                candidates.append(np.clip(tv, lo, hi))
        
        # Intersections with each hull segment
        for k in range(len(hull)):
            (m_h, c_h), start_k = hull[k]
            end_k = hull[k + 1][1] if k + 1 < len(hull) else np.inf
            seg_lo = max(lo, start_k)
            seg_hi = min(hi, end_k)
            if seg_lo >= seg_hi:
                continue
            # a[pi]*t^2 + (b[pi]-m_h)*t + (c[pi]-c_h) = 0
            A = a[pi]
            B = b[pi] - m_h
            C = c[pi] - c_h
            if abs(A) < 1e-16:
                if abs(B) > 1e-16:
                    t = -C / B
                    if seg_lo - 1e-12 <= t <= seg_hi + 1e-12:
                        candidates.append(np.clip(t, seg_lo, seg_hi))
            else:
                disc = B * B - 4 * A * C
                if disc >= -1e-14:
                    disc = max(0.0, disc)
                    sq = np.sqrt(disc)
                    for t in [(-B + sq) / (2 * A), (-B - sq) / (2 * A)]:
                        if np.isfinite(t) and seg_lo - 1e-12 <= t <= seg_hi + 1e-12:
                            candidates.append(np.clip(t, seg_lo, seg_hi))
    
    # Deduplicate candidates
    candidates = sorted(set(np.round(np.array(candidates), 12).tolist()))
    
    # Evaluate envelope at each candidate
    def eval_envelope(t):
        # Evaluate max of all f_p(t)
        vals = c + b * t + a * t * t
        # Also need to consider the affine envelope is already captured
        # since all lines are in c, b, a (with a=0 for affine)
        return float(np.max(vals))
    
    # Find best candidate
    best_t = 0.0
    best_val = eval_envelope(0.0)
    for t in candidates:
        v = eval_envelope(t)
        if v < best_val - 1e-14:
            best_val = v
            best_t = t
    
    if best_val >= cur_score - 1e-12:
        return None
    
    # Try the best few candidates with exact scoring
    # Sort candidates by predicted value
    scored = [(eval_envelope(t), t) for t in candidates]
    scored.sort()
    
    best_h = None
    best_c = cur_score
    for _, t in scored[:6]:
        h_new = h.copy()
        t_clamped = np.clip(t, lo, hi)
        h_new[i] += t_clamped
        h_new[j] -= t_clamped
        h_new = project_box_sum(h_new, 0.5 * n)
        if h_new is None:
            continue
        sc, _ = score_and_corr(h_new, n)
        if sc < best_c:
            best_c = sc
            best_h = h_new
    
    if best_h is not None and best_c < cur_score - 1e-12:
        return best_h, best_c
    return None

def local_descent(h, n, rng, max_sweeps):
    cur_h = h.copy()
    cur_corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
    cur_score = float(np.max(cur_corr)) * (2.0 / n)
    
    num_pairs = n * (n - 1) // 2
    all_pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    
    for sweep in range(max_sweeps):
        rng.shuffle(all_pairs)
        improved = False
        for i, j in all_pairs:
            try:
                result = pair_line_search(cur_h, cur_corr, cur_score, i, j, n)
                if result is not None:
                    cur_h, cur_score = result
                    cur_corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
                    improved = True
            except Exception:
                pass
        if not improved:
            break
    return cur_h, cur_score
