        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(n + 2):
        if abs(r) < 1e-13 * max(1, n):
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            if 1.0 - h[idx] <= 1e-15:
                break
            h[idx] += min(r, 1.0 - h[idx])
        else:
            idx = int(np.argmax(h))
            if h[idx] <= 1e-15:
                break
            h[idx] += max(r, -h[idx])
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(np.sum(h) - s) > 5e-11 * max(1, n):
        return None
    return h

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return float('inf')
    if abs(np.sum(h) - n / 2.0) > 1e-9 * max(1, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_vals_grads(h, n):
    L = 2 * n - 1
    v = np.zeros(L)
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        ii = np.arange(lo, hi)
        ij = ii + d
        v[row] = float(np.sum(h[ii] * (1.0 - h[ij])))
        G[row, ii] += (1.0 - h[ij])
        G[row, ij] -= h[ii]
    return v, G

def slsqp_obj(z, n):
    return z[n]

def slsqp_obj_jac(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def slsqp_eq(z, n):
    return float(np.sum(z[:n]) - n / 2.0)

def slsqp_eq_jac(z, n):
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def slsqp_ineq(z, n):
    dx = 2.0 / n
    v, _ = lag_vals_grads(z[:n], n)
    return z[n] - dx * v

def slsqp_ineq_jac(z, n):
    dx = 2.0 / n
    _, G = lag_vals_grads(z[:n], n)
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -dx * G
    J[:, n] = 1.0
    return J

def add_start(starts, v, n):
    h2 = project_capped(v, n)
    if h2 is None:
        return
    for ex in starts:
        if float(np.max(np.abs(h2 - ex))) < 1e-8:
            return
    starts.append(h2)

def pairwise_polish(h, n, sweeps):
    best_h = h.copy()
    best_s = exact_score(h, n)
    if not np.isfinite(best_s):
        return best_h, best_s
    deltas = (0.001, 0.003, 0.01, 0.03, 0.1, 0.3)
    for _ in range(sweeps):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                lo = max(-best_h[i], best_h[j] - 1.0)
                hi = min(1.0 - best_h[i], best_h[j])
                if lo >= hi - 1e-12:
                    continue
                trials = [lo, hi, 0.0]
                for d in deltas:
                    for sg in (1, -1):
                        t = sg * d
                        trials.append(max(lo, min(hi, t)))
                bt_s, bt_db = best_s, 0.0
                for db in trials:
                    h2 = best_h.copy()
                    h2[i] += db
                    h2[j] -= db
                    h2 = np.clip(h2, 0.0, 1.0)
                    if abs(float(np.sum(h2)) - n / 2.0) > 1e-9 * n:
                        continue
                    s2 = exact_score(h2, n)
                    if s2 < bt_s - 1e-12:
                        bt_s, bt_db = s2, db
                if bt_db != 0.0:
                    best_h[i] += bt_db
                    best_h[j] -= bt_db
                    best_h = np.clip(best_h, 0.0, 1.0)
                    best_s = bt_s
                    improved = True
        if not improved:
            break
    return best_h, best_s

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 0 < len(iv) <= 999:
            if np.all(np.isfinite(iv)) and np.all(iv >= -1e-8) and np.all(iv <= 1.0 + 1e-8):
                n = len(iv)
                parent = iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
    n = max(1, min(n, 999))
    try:
        sd = max(0, int(seed))
    except Exception:
        sd = 42
    rng = np.random.default_rng(sd)
    uniform = np.full(n, 0.5)
    fallback = project_capped(parent, n) if parent is not None else uniform.copy()
    if fallback is None:
        fallback = uniform.copy()
    fb_score = exact_score(fallback, n)
    if not np.isfinite(fb_score):
        fallback, fb_score = uniform.copy(), exact_score(uniform, n)
    nsd = 8 if n <= 64 else (5 if n <= 128 else 3)
    mid_ = 250 if n <= 64 else (160 if n <= 128 else 80)
    try:
        n_starts = max(1, min(int(kwargs.get('starts', nsd)), 10))
    except Exception:
        n_starts = nsd
    try:
        max_iter = max(0, min(int(kwargs.get('max_iter', mid_)), 300))
    except Exception:
        max_iter = mid_
    try:
        ftol_v = max(1e-12, min(float(kwargs.get('ftol', 1e-10)), 1e-6))
    except Exception:
        ftol_v = 1e-10
    try:
        psw = max(0, min(int(kwargs.get('polish_sweeps', 1)), 2))
    except Exception:
        psw = 1
    starts = []
    add_start(starts, fallback, n)
    add_start(starts, fallback[::-1].copy(), n)
    add_start(starts, 1.0 - fallback[::-1].copy(), n)
    add_start(starts, uniform, n)
    if parent is not None:
        for sdt in (0.025, 0.10):
            add_start(starts, fallback + rng.normal(0, sdt, n), n)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    add_start(starts, blk, n)
    add_start(starts, np.where(np.arange(n) % 2 == 0, 1.0, 0.0), n)
    add_start(starts, 0.5 + 0.45 * np.cos(2 * np.pi * np.arange(n) / n), n)
    add_start(starts, rng.uniform(0, 1, n), n)
    starts = starts[:n_starts]
    best, best_score = fallback.copy(), fb_score
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_score - 1e-12:
            best, best_score = st.copy(), ss
    if n <= 256 and max_iter > 0:
        bnds = [(0.0, 1.0)] * (n + 1)
        cons = [
            {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac, 'args': (n,)},
            {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac, 'args': (n,)}
        ]
        for st in starts:
            s0 = exact_score(st, n)
            if not np.isfinite(s0):
                continue
            z0 = np.append(st, min(1.0, s0 + 1e-8))
            try:
                res = minimize(slsqp_obj, z0, args=(n,), method='SLSQP',
                               jac=slsqp_obj_jac, bounds=bnds, constraints=cons,
                               options={'ftol': ftol_v, 'maxiter': max_iter})
            except Exception:
                continue
            if res.x is not None and len(res.x) == n + 1 and np.all(np.isfinite(res.x)):
                hp = project_capped(res.x[:n], n)
                if hp is not None:
                    sn = exact_score(hp, n)
                    if np.isfinite(sn) and sn < best_score - 1e-12:
                        best, best_score = hp.copy(), sn
        z0r = np.append(best, min(1.0, best_score + 1e-8))
        try:
            res2 = minimize(slsqp_obj, z0r, args=(n,), method='SLSQP',
                            jac=slsqp_obj_jac, bounds=bnds, constraints=cons,
                            options={'ftol': 1e-12, 'maxiter': max_iter})
        except Exception:
            res2 = None
        if res2 is not None and res2.x is not None and len(res2.x) == n + 1 and np.all(np.isfinite(res2.x)):
            hp2 = project_capped(res2.x[:n], n)
            if hp2 is not None:
                sn2 = exact_score(hp2, n)
                if np.isfinite(sn2) and sn2 < best_score - 1e-12:
                    best, best_score = hp2.copy(), sn2
    if n <= 64 and psw > 0:
        ph, ps = pairwise_polish(best, n, psw)
        if np.isfinite(ps) and ps < best_score - 1e-12:
            best, best_score = ph, ps
    final = project_capped(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > best_score + 1e-12:
        final = project_capped(fallback, n)
        if final is None:
            final = uniform.copy()
        fs = exact_score(final, n)
    if not np.isfinite(fs):
        final, fs = uniform.copy(), exact_score(uniform, n)
    return final, float(fs), int(n)
