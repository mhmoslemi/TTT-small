import numpy as np
import math

def project_capped(y, n):
    S = n / 2.0
    if y.shape != (n,) or not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        c = np.clip(y - mid, 0.0, 1.0)
        if c.sum() > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(5):
        diff = S - c.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = c < 1.0 - 1e-12
        else:
            mask = c > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        c[mask] += diff / cnt
        np.clip(c, 0.0, 1.0, out=c)
    return c

def exact_score(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def helmert_dir(j, n):
    d = np.zeros(n)
    sf = math.sqrt((j + 1) * (j + 2))
    d[:j+1] = 1.0 / sf
    d[j+1] = -(j + 1) / sf
    return d

def line_search(h, d, n):
    lo_t, hi_t = -1e6, 1e6
    for i in range(n):
        di = d[i]
        if abs(di) < 1e-15:
            continue
        if di > 0:
            lo_t = max(lo_t, -h[i] / di)
            hi_t = min(hi_t, (1.0 - h[i]) / di)
        else:
            lo_t = max(lo_t, (1.0 - h[i]) / di)
            hi_t = min(hi_t, -h[i] / di)
    if hi_t - lo_t < 1e-12:
        return None
    dx = 2.0 / n
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    b = (np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')) * dx
    a = -np.correlate(d, d, mode='full') * dx
    mh = 0.5 * (lo_t + hi_t)
    rh = 0.5 * (hi_t - lo_t)
    A = a * rh * rh

    C = q + b * mh + a * mh * mh
    cands = [-1.0, 1.0]

        cands.append(s0)

                cands.append(sp)

        for j in range(i + 1, nl):

            db = B[i] - B[j]
            dc = C[i] - C[j]
            if abs(da) < 1e-18:
                if abs(db) > 1e-18:
                    rt = -dc / db
                    if -1.0 <= rt <= 1.0:
                        cands.append(rt)

def polish(h0, n, rng, max_sweeps):
    h = h0.copy()
    s = exact_score(h, n)
    for _ in range(max_sweeps):
        improved = False
        order = list(range(n - 1))
        rng.shuffle(order)
        for j in order:
            d = helmert_dir(j, n)
            t = line_search(h, d, n)
            if t is not None:
                h_new = project_capped(h + t * d, n)
                s_new = exact_score(h_new, n)
                if s_new < s - 1e-13:
                    h = h_new
                    s = s_new
                    improved = True
        if not improved:
            break
    return h, s

def try_sdp(n, dx, S, fb_s):
    try:
        import cvxpy as cp
        h = cp.Variable(n)
        X = cp.Variable((n, n), symmetric=True)
        t = cp.Variable()
        block = cp.vstack([cp.hstack([cp.Constant(1), h.T]), cp.hstack([h, X])])
        con = [block >> 0, h >= 0, h <= 1, cp.sum(h) == S, t >= 0]
        for i in range(n):
            con.append(cp.sum(X[i, :]) == S * h[i])
            con.append(X[i, i] >= 2.0 * h[i] - 1)
            con.append(X[i, i] <= h[i])
        for i in range(n):
            for j in range(i + 1, n):
                con.append(X[i, j] >= 0)
                con.append(X[i, j] >= h[i] + h[j] - 1)
                con.append(X[i, j] <= h[i])
        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, n + k)
            rows = np.arange(lo, hi)
            cols = rows - k
            con.append(t >= dx * cp.sum(h[lo:hi] - X[rows, cols]))
        prob = cp.Problem(cp.Minimize(t), con)
        solved = False
        try:
            prob.solve(solver=cp.MOSEK)
            solved = True
        except Exception:
            pass
        if not solved:
            try:
                prob.solve(solver=cp.CVXOPT, abstol=1e-7, reltol=1e-7, feastol=1e-7)
            except Exception:
                pass
        if prob.status in ('optimal', 'optimal_inaccurate') and h.value is not None:
            hv = np.array(h.value, dtype=np.float64)
            Xv = np.array(X.value, dtype=np.float64)
            if (np.all(np.isfinite(hv)) and np.all(np.isfinite(Xv))
                    and t.value is not None and float(t.value) <= fb_s + 1e-2):
                return hv, Xv
    except Exception:
        pass
    return None

def covariance_candidates(hc, Xc, n, rng):
    V = Xc - np.outer(hc, hc)
    V = 0.5 * (V + V.T)
    Q = np.eye(n) - np.ones((n, n)) / n
    V = Q @ V @ Q
    V = 0.5 * (V + V.T)
    try:
        ev, evec = np.linalg.eigh(V)
    except Exception:
        return []
    max_ev = max(1.0, float(np.max(np.abs(ev))))
    if float(ev[0]) < -1e-3 * max_ev:
        return []
    ev = np.clip(ev, 0.0, None)
    md = float(np.max(np.abs(np.diag(V))))
    if md > 0.25:
        V = V * (0.25 / md)
        ev, evec = np.linalg.eigh(V)
        ev = np.clip(ev, 0.0, None)
    L = evec * np.sqrt(np.maximum(ev, 0.0))
    cands = []
    for scale in [0.5, 1.0]:
        for _ in range(4):
            z = rng.standard_normal(n)
            cands.append(hc + scale * (L @ z))
    oev = np.argsort(ev)[::-1]
    for idx in oev[:4]:
        if ev[idx] < 1e-12:
            break
        v = evec[:, idx] * math.sqrt(ev[idx])
        cands.append(hc + 0.5 * v)
        cands.append(hc - 0.5 * v)
    return cands
