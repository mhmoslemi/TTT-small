def proj_simplex(y, n, S):
    if not (isinstance(y, np.ndarray) and y.shape == (n,) and np.all(np.isfinite(y))):
        return np.full(n, S / n)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    e = S - x.sum()
    if abs(e) > 1e-14:
        if e > 0:
            for j in range(n):
                add = min(e, 1.0 - x[j])
                x[j] += add
                e -= add
                if abs(e) < 1e-14:
                    break
        else:
            for j in range(n):
                sub = min(-e, x[j])
                x[j] -= sub
                e += sub
                if abs(e) < 1e-14:
                    break

def is_feasible(h, n, S):

def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
