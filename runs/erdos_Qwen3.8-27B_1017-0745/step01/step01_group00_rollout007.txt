    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)

            while len(np2) < M:
                r = rng.random()
                if r < 0.5:
                    v = project_capped(gb + rng.normal(0, sg, n), n)
                elif r < 0.75:
                    v = project_capped(np2[int(rng.integers(len(np2)))] + rng.normal(0, sg, n), n)
                else:
                    v = project_capped(rng.uniform(0, 1, n), n)
                if v is not None:
                    np2.append(v)
                    ns2.append(exact_score(v, n))
                else:
                    np2.append(gb.copy())
                    ns2.append(gs)
            while len(np2) < M:
                np2.append(gb.copy())
                ns2.append(gs)
            pop = np2[:M]
            scores = np.array(ns2[:M])
    return gb, gs

def poll_refine(gb, gs, n, rng, n_rounds, n_dir, sc):
    delta = min(0.25, 0.12 * sc)
    for rd in range(n_rounds):
        if delta < 1e-4:
            break
        bt, bts = None, gs
        for d in range(n_dir):
            v = rng.standard_normal(n) - np.mean(rng.standard_normal(n))
            v = v - np.mean(v)
            vm = np.max(np.abs(v))
            if vm < 1e-12 or not np.isfinite(vm):
                continue
            v = v / vm
            pos = v > 1e-15
            neg = v < -1e-15
            tlo = -1e18
            thi = 1e18
            if np.any(pos):
                thi = min(thi, float(np.min((1.0 - gb[pos]) / v[pos])))
            if np.any(neg):
                tlo = max(tlo, float(np.min((1.0 - gb[neg]) / v[neg])))
            if tlo >= thi:
                continue
            tried = []
            for tv in (-delta, -0.5 * delta, 0.5 * delta, delta):
                tc = max(tlo + 1e-12, min(thi - 1e-12, tv))
                if abs(tc) < 1e-12:
                    continue
                if not any(abs(tc - x) < 1e-12 for x in tried):
                    tried.append(tc)
                    tr = project_capped(gb + tc * v, n)
                    if tr is None or not is_feasible(tr, n):
                        continue
                    tsc = exact_score(tr, n)
                    if tsc < bts:
                        bts = tsc
                        bt = tr
        if bt is not None and bts < gs - 1e-12:
            gb, gs = bt, bts
            delta = min(0.25, 1.2 * delta)
        else:
            delta *= 0.5
    return gb, gs

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if parent is not None:
        fb = project_capped(np.clip(parent, 0.0, 1.0), n)
    else:
        fb = np.full(n, 0.5)
    if fb is None:
        fb = np.full(n, 0.5)
    rng = np.random.default_rng(int(seed))
    sc = min(1.5, max(0.5, np.sqrt(47.0 / n)))
    if n == 1:
        return fb, exact_score(fb, n), 1
    if n <= 64:
        M, ng, sl, mr, nr, nd = 40, 120, 20, 3, 28, 10
    elif n <= 128:
        M, ng, sl, mr, nr, nd = 36, 90, 16, 2, 20, 8
    elif n <= 256:
        M, ng, sl, mr, nr, nd = 28, 55, 12, 1, 12, 6
    elif n <= 512:
        M, ng, sl, mr, nr, nd = 20, 30, 10, 1, 8, 4
    else:
        M, ng, sl, mr, nr, nd = 14, 18, 8, 1, 5, 4
    pool = build_pool(fb, n, rng, M)
    scores = np.array([exact_score(p, n) for p in pool])
    try:
        gb, gs = de_evolve(pool, scores, n, rng, ng, sl, mr)
    except Exception:
        gi = int(np.argmin(scores))
        gb, gs = pool[gi].copy(), float(scores[gi])
    try:
        gb, gs = poll_refine(gb, gs, n, rng, nr, nd, sc)
    except Exception:
        pass
    fb = project_capped(fb, n)
    if fb is None:
        fb = np.full(n, 0.5)
    gsf = exact_score(fb, n)
    gb = project_capped(gb, n)
    if gb is None or not is_feasible(gb, n) or exact_score(gb, n) > gsf:
        gb, gs = fb, gsf
    else:
        gs = exact_score(gb, n)
    return gb, float(gs), int(n)
