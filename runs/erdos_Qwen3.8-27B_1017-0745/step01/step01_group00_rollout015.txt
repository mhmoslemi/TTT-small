Key implementation details:
- Lag d ranges from -(n-1) to (n-1)
- Q_d(h) = dx * sum_{r=0}^{m_d-1} h[a_d+r] * (1-h[b_d+r])
  where a_d = max(d,0), b_d = max(-d,0), m_d = n-|d|
- Gradient: for each r, add dx*(1-h[b_d+r]) to g[a_d+r], subtract dx*h[a_d+r] from g[b_d+r]
- At d=0, both contributions hit the same index: g[i] += dx*(1-h[i]) - dx*h[i] = dx*(1-2*h[i])

For the log-sum-exp:
F_τ(h) = q_max + τ * log(sum(exp((Q_d - q_max)/τ)))
gradient = sum_d w_d * g_d where w_d = exp((Q_d - q_max)/τ) / sum(exp((Q_j - q_max)/τ))

def smooth_value_gradient(h, tau):
    n = h.size
    Q = lag_values(h, n)
    q_max = np.max(Q)
    vals = (Q - q_max) / tau
    exps = np.exp(vals)
    s_exp = np.sum(exps)
    F = q_max + tau * np.log(s_exp)
    w = exps / s_exp
    g = np.zeros(n)
    dx = 2.0 / n
    for d in range(-(n - 1), n):
        row = d + n - 1
        wt = w[row]
        if wt < 1e-15:
            continue
        a = max(d, 0)
        b = max(-d, 0)
        m = n - abs(d)
        g[a:a + m] += wt * dx * (1.0 - h[b:b + m])
        g[b:b + m] -= wt * dx * h[a:a + m]
    if not np.isfinite(F) or not np.all(np.isfinite(g)):
        return 1e10, np.zeros(n)
    return F, g

def _mass_con(x):
    return np.sum(x) - _N * 0.5

def _mass_jac(x):
    return np.ones(_N)

def build_starts(fb, n, rng):
    pool = []
    def try_add(v):
        if v is None:
            return
        for s in pool:
            if np.max(np.abs(v - s)) < 1e-8:
                return
        pool.append(v)
    try_add(fb.copy())
    try_add(project_capped(fb[::-1], n))
    try_add(project_capped(1.0 - fb[::-1], n))
    try_add(np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    try_add(project_capped(blk, n))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    try_add(project_capped(alt, n))
    try_add(project_capped(fb + 0.10 * rng.standard_normal(n), n))
    try_add(project_capped(fb + 0.25 * rng.standard_normal(n), n))
    try_add(project_capped(rng.uniform(0, 1, n), n))
    return pool

def slsqp_cont(h0, n, tau_sched, max_iter):
    global _N
    _N = n
    h = h0.copy().astype(np.float64)
    constraints = [{'type': 'eq', 'fun': _mass_con, 'jac': _mass_jac}]
    bounds = [(0.0, 1.0)] * n
    for tau in tau_sched:
        F0, _ = smooth_value_gradient(h, tau)
        res = minimize(smooth_value_gradient, h, method='SLSQP', jac=True,
                       args=(tau,), bounds=bounds, constraints=constraints,
                       ftol=1e-10, maxiter=max_iter, disp=False)
        if res.x is not None and np.all(np.isfinite(res.x)):
            proj = project_capped(res.x, n)
            if proj is not None:
                best = h
                best_fs = F0
                for alpha in (1.0, 0.5, 0.25, 0.125):
                    cand = project_capped(h + alpha * (proj - h), n)
                    if cand is not None:
                        fs, _ = smooth_value_gradient(cand, tau)
                        if fs < best_fs - 1e-13:
                            best_fs, best = fs, cand
                h = best
    return h

def proj_grad_cont(h0, n, tau_sched, max_iter):
    h = h0.copy().astype(np.float64)
    for tau in tau_sched:
        dx = 2.0 / n
        eta = min(max(1.0, 0.25 / dx), 2.0 / dx)
        for _ in range(max_iter):
            F, g = smooth_value_gradient(h, tau)
            if not np.isfinite(F):
                break
            z = project_capped(h - eta * g, n)
            if z is None:
                break
            v = z - h
            if np.max(np.abs(v)) < 1e-9:
                break
            gv = np.dot(g, v)
            if gv >= 0:
                break
            Fz, _ = smooth_value_gradient(z, tau)
            if Fz <= F + 1e-4 * gv:
                h = z
                eta = min(eta * 1.2, 2.0 / dx)
            else:
                accepted = False
                for _2 in range(5):
                    eta *= 0.5
                    z2 = project_capped(h - eta * g, n)
                    if z2 is None:
                        break
                    v2 = z2 - h
                    Fz2, _ = smooth_value_gradient(z2, tau)
                    if Fz2 <= F + 1e-4 * np.dot(g, v2):
                        h = z2
                        accepted = True
                        break
                if not accepted:
                    break
    return h

def run(seed=42, **kwargs):
    global _N
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    _N = n
    fb = project_capped(parent, n) if parent is not None else np.full(n, 0.5)
    if fb is None:
        fb = np.full(n, 0.5)
    tau_sched = [0.03, 0.01, 0.003, 0.001, 0.0003]
    if n <= 64:
        n_starts, max_iter, n_kicks = 8, 90, 3
    elif n <= 128:
        n_starts, max_iter, n_kicks = 5, 60, 2
    else:
        n_starts, max_iter, n_kicks = 2, 35, 0
    starts = build_starts(fb, n, rng)[:n_starts]
    gb = fb.copy()
    gs = exact_score(fb, n)
    for st in starts:
        s = exact_score(st, n)
        if s < gs - 1e-12:
            gs, gb = s, st.copy()
    for st in starts:
        try:
            if n <= 128:
                result = slsqp_cont(st, n, tau_sched, max_iter)
            else:
                result = proj_grad_cont(st, n, tau_sched, max_iter)
        except Exception:
            continue
        result = project_capped(result, n)
        if result is not None:
            s = exact_score(result, n)
            if s < gs - 1e-12:
                gs, gb = s, result.copy()
    if n_kicks > 0:
        sigma = 0.08
        for _ in range(n_kicks):
            try:
                kick = project_capped(gb + sigma * rng.standard_normal(n), n)
                if kick is None:
                    continue
                if n <= 128:
                    result = slsqp_cont(kick, n, tau_sched, max_iter)
                else:
                    result = proj_grad_cont(kick, n, tau_sched, max_iter)
            except Exception:
                continue
            result = project_capped(result, n)
            if result is not None:
                s = exact_score(result, n)
                if s < gs - 1e-12:
                    gs, gb = s, result.copy()
                    sigma = 0.08
                else:
                    sigma = min(0.35, 1.7 * sigma)
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb
    if exact_score(final, n) > exact_score(fb, n):
        final = fb
    final = project_capped(final, n)
    if final is None:
        final = np.full(n, 0.5)
    s = exact_score(final, n)
    if not np.isfinite(s):
        final = np.full(n, 0.5)
        s = exact_score(final, n)
    return final, float(s), int(n)
