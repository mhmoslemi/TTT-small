            h.min() >= -1e-10 and h.max() <= 1.0+1e-10 and abs(h.sum()-S) <= 1e-8)

    return np.correlate(h, 1.0-h, mode='full') * dx

    nl = 2*n - 1
    G = np.zeros((nl, n))
    j = np.arange(n)
    for k in range(nl):
        ell = k - (n-1)
        i1, v1 = j - ell, (j-ell >= 0) & (j-ell < n)
        i2, v2 = j + ell, (j+ell >= 0) & (j+ell < n)
        t1, t2 = np.zeros(n), np.zeros(n)
        t1[v1] = 1.0 - h[i1[v1]]
        t2[v2] = h[i2[v2]]
        G[k] = dx * (t1 - t2)
