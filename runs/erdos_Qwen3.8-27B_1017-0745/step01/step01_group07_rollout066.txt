    bv = np.zeros(n); bv[perm[:n // 2]] = 1.0
    if n % 2: bv[perm[n // 2]] = 0.5
    starts.append(bv)
    ur = project_capped_simplex(rng.random(n), S)
    if ur is not None: starts.append(ur)
    starts.append(np.full(n, 0.5))
    ded = []
    for s in starts:
        if any(np.max(np.abs(s - d)) < 1e-10 for d in ded): continue
        if is_feasible(s, n): ded.append(s)
    cap = 6 if n <= 128 else 4
    return ded[:cap]

def polish_sweep(h, q, score, n, dx, S, rng):
    P = min(n * (n - 1) // 2, max(24, 80000 // n))
    pairs = []
    adj = min(n - 1, max(1, P // 2))
    for i in (np.linspace(0, n - 2, adj, dtype=int) if adj > 1 else np.array([0])):
        pairs.append((int(i), int(i) + 1))
    att = 0
    while len(pairs) < P and att < P * 3:
        i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
        if i != j:
            if i > j: i, j = j, i
            if (i, j) not in pairs: pairs.append((i, j))
        att += 1
    rng.shuffle(pairs)
    acc = 0
    for i, j in pairs:
        res = pair_line_search(h, q, score, i, j, n, dx, S)
        if res is not None:
            nq, ns = exact_lags(res, n)
            if ns is not None and ns < score - 1e-12:
                h, q, score, acc = res, nq, ns, acc + 1
    return h, q, score, acc

def run(seed=42, **kwargs):
    if seed is None: seed = 42
    try: seed = int(seed) % (2 ** 32)
    except (TypeError, ValueError): seed = 42
    rng = np.random.default_rng(seed)
    n, parent = None, None
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        try:
            iv_arr = np.asarray(iv, dtype=np.float64)
            if iv_arr.ndim == 1 and 2 <= iv_arr.size <= 999:
                n = iv_arr.size
                if np.all(np.isfinite(iv_arr)) and np.max(np.abs(iv_arr)) <= 1e12:
                    S = n / 2.0
                    if is_feasible(iv_arr, n):
                        parent = iv_arr.copy()
                    else:
                        pr = project_capped_simplex(iv_arr, S)
                        if pr is not None: parent = pr
        except Exception: pass
    if n is None:
        try: n = int(kwargs.get('n_points', 80))
        except (TypeError, ValueError): n = 80
        if n < 8 or n > 999: n = 80
    S = n / 2.0
    dx = 2.0 / n
    h_const = np.full(n, 0.5)
    best_h, best_score = h_const.copy(), 0.5
    if parent is not None:
        _, ps = exact_lags(parent, n)
        if ps is not None and ps < best_score:
            best_score, best_h = ps, parent.copy()
    starts = build_starts(n, S, parent, rng)
    for s in starts:
        _, ss = exact_lags(s, n)
        if ss is not None and ss < best_score:
            best_score, best_h = ss, s.copy()
    E = 5 if n <= 128 else 4
    K = min(1000, max(80, 60000 // n))
    for s in starts:
        h = s.copy()
        q, score = exact_lags(h, n)
        if q is None: continue
        scale = 0.12
        for e in range(E):
            T = max(5e-5, 3e-3 * (0.4 ** e))
            acc = 0
            for _ in range(K):
                p, r = int(rng.integers(0, n)), int(rng.integers(0, n))
                if p == r: continue
                L = max(-h[p], h[r] - 1.0)
                U = min(1.0 - h[p], h[r])
                if U - L <= 1e-12: continue
                t = float(rng.uniform(-scale, scale))
                if t < L or t > U: continue
                h_new = h.copy()
                h_new[p] += t; h_new[r] -= t
                if not is_feasible(h_new, n):
                    h_new = project_capped_simplex(h_new, S)
                    if h_new is None: continue
                _, ns = exact_lags(h_new, n)
                if ns is None: continue
                if ns < best_score: best_score, best_h = ns, h_new.copy()
                delta = ns - score
                if delta <= 0 or math.exp(-min(delta / T, 700.0)) > rng.random():
                    h, score, acc = h_new, ns, acc + 1
                    q = exact_lags(h, n)[0]
            ratio = acc / max(1, K)
            if ratio < 0.12: scale = min(0.4, 1.25 * scale)
            elif ratio > 0.55: scale = max(0.01, 0.75 * scale)
            h, q, score, _ = polish_sweep(h, q, score, n, dx, S, rng)
            if score < best_score: best_score, best_h = score, h.copy()
    for _ in range(4):
        q0, s0 = exact_lags(best_h, n)
        if q0 is None: break
        h2, q2, sc2, _ = polish_sweep(best_h.copy(), q0, s0, n, dx, S, rng)
        if sc2 < best_score: best_score, best_h = sc2, h2
    if not is_feasible(best_h, n):
        best_h = project_capped_simplex(best_h, S)
        if best_h is None: best_h = h_const.copy()
    _, fs = exact_lags(best_h, n)
    if fs is None: fs = 0.5
    return best_h, fs, n
