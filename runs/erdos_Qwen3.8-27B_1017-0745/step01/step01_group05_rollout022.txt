            and abs(h.sum() - n / 2.0) <= 1e-10)

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def helmert_dir(j, n):
    d = np.zeros(n)
    s = math.sqrt((j + 1) * (j + 2))
    d[:j + 1] = 1.0 / s
    d[j + 1] = -(j + 1) / s
    return d

def exact_line_min(h, d, n):
    lo, hi = -1e15, 1e15
    for i in range(n):
        if d[i] > 1e-15:
            lo = max(lo, -h[i] / d[i])
            hi = min(hi, (1.0 - h[i]) / d[i])
        elif d[i] < -1e-15:
            lo = max(lo, (1.0 - h[i]) / d[i])
            hi = min(hi, -h[i] / d[i])
    if hi - lo < 1e-12:
        return None
    mid, rad = 0.5 * (lo + hi), 0.5 * (hi - lo)
    if rad < 1e-14:
        return None
    dx = 2.0 / n
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    b = (np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')) * dx
    a = -np.correlate(d, d, mode='full') * dx
    m = 2 * n - 1
    A = a * rad * rad
    B = rad * (b + 2.0 * a * mid)
    C = q + b * mid + a * mid * mid
    cands = [-1.0, 1.0, -mid / rad]
    for i in range(m):
        if abs(A[i]) > 1e-18:
            sp = -B[i] / (2.0 * A[i])
            if -1.05 <= sp <= 1.05:
                cands.append(sp)
    for i in range(m):
        for j in range(i + 1, m):
            da, db, dc = A[i] - A[j], B[i] - B[j], C[i] - C[j]
            if abs(da) < 1e-18:
                if abs(db) > 1e-18:
                    rt = -dc / db
                    if -1.05 <= rt <= 1.05:
                        cands.append(rt)
            else:
                disc = db * db - 4.0 * da * dc
                if disc >= -1e-12:
                    if disc < 0:
                        disc = 0.0
                    sq = math.sqrt(disc)
                    for rt in [(-db + sq) / (2.0 * da), (-db - sq) / (2.0 * da)]:
                        if -1.05 <= rt <= 1.05:
                            cands.append(rt)
    ca = np.clip(np.array(cands), -1.0, 1.0)
    ca = np.unique(np.round(ca, 10))
    best_val, best_s = float('inf'), 0.0
    for st in range(0, len(ca), 256):
        chunk = ca[st:st + 256]
        vals = A * chunk[None, :] ** 2 + B * chunk[None, :] + C[:, None]
        upper = np.max(vals, axis=0)
        idx = np.argmin(upper)
        if upper[idx] < best_val:
            best_val, best_s = upper[idx], chunk[idx]
    if best_val == float('inf'):
        return None
    t_best = mid + rad * best_s
    return max(lo, min(hi, t_best))

def solve_sdp(n, S, dx, fb_score):
    if n > 50:
        return None, None
    try:
        import cvxpy as cp
    except Exception:
        return None, None
    try:
        hv = cp.Variable(n)
        X = cp.Variable((n, n), symmetric=True)
        t = cp.Variable()
        M = cp.bmat([[cp.Constant(1.0), hv], [hv, X]])
        cons = [M >> 0, hv >= 0, hv <= 1, cp.sum(hv) == S]
        for i in range(n):
            cons.append(X[i, i] >= 2.0 * hv[i] - 1.0)
            cons.append(X[i, i] <= hv[i])
        for i in range(n):
            for j in range(i + 1, n):
                cons.append(X[i, j] >= 0)
                cons.append(X[i, j] >= hv[i] + hv[j] - 1.0)
                cons.append(X[i, j] <= hv[i])
                cons.append(X[i, j] <= hv[j])
        for i in range(n):
            cons.append(cp.sum(X[i, :]) == S * hv[i])
        for k in range(-(n - 1), n):
            li, hi2 = max(0, k), min(n, n + k)
            if hi2 > li:
                idx = np.arange(li, hi2)
                cons.append(t >= dx * cp.sum(hv[li:hi2] - X[idx, idx - k]))
        prob = cp.Problem(cp.Minimize(t), cons)
        for solver in [cp.MOSEK, cp.CVXOPT]:
            try:
                prob.solve(solver=solver, max_iters=100)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    if hv.value is not None and np.all(np.isfinite(hv.value)):
                        Xv = X.value
                        if Xv is not None and np.all(np.isfinite(Xv)):
                            return hv.value.copy(), Xv.copy()
            except Exception:
                continue
    except Exception:
        pass
    return None, None

def covariance_samples(hp, X, n, rng):
    cands = []
    V = X - np.outer(hp, hp)
    V = 0.5 * (V + V.T)
    Q = np.eye(n) - np.ones((n, n)) / n
    V = Q @ V @ Q
    V = 0.5 * (V + V.T)
    ev, evec = np.linalg.eigh(V)
    mx = max(1.0, float(np.max(np.abs(ev))))
    if ev.min() < -1e-3 * mx:
        return cands
    ev = np.maximum(ev, 0.0)
    md = float(np.max(np.abs(np.diag(V))))
    if md > 0.25:
        ev *= 0.25 / md
    pos = ev > 1e-12
    L = evec[:, pos] * np.sqrt(ev[pos])[None, :]
    for _ in range(12):
        cands.append(hp + 0.5 * (L @ rng.standard_normal(n)))
    for _ in range(12):
        cands.append(hp + (L @ rng.standard_normal(n)))
    tk = min(4, len(ev))
    for ki in np.argsort(ev)[::-1][:tk]:
        mode = evec[:, ki]
        sc = 0.5 * math.sqrt(max(ev[ki], 0))
        cands.append(hp + sc * mode)
        cands.append(hp - sc * mode)
    return cands

def polish_start(h0, n, rng):
    h = h0.copy()
    s = exact_score(h, n)
    if s == float('inf'):
        return h0, s
    nw = 2 if n <= 50 else 1
    dc = n - 1 if n <= 50 else min(16, n - 1)
    for _ in range(nw):
        improved = False
        for j in rng.permutation(dc):
            j = int(j)
            if n <= 50:
                d = helmert_dir(j, n)
            else:
                d = rng.standard_normal(n)
                d -= d.mean()
                nm = np.linalg.norm(d)
                if nm < 1e-12:
                    continue
                d /= nm
            tb = exact_line_min(h, d, n)
            if tb is not None and abs(tb) > 1e-14:
                hn = h + tb * d
                hn, ok = project_capped(hn, n)
                if ok:
                    sn = exact_score(hn, n)
                    if sn < s - 1e-12:
                        h, s, improved = hn, sn, True
    return h, s

def record(hc, n, best_h, best_s, pool):
    np.clip(hc, 0.0, 1.0, out=hc)
    hc, ok = project_capped(hc, n)
    if not ok:
        return best_h, best_s
    s = exact_score(hc, n)
    if s < float('inf') and s < best_s - 1e-12:
        best_h, best_s = hc.copy(), s
    if len(pool) < 16:
        dup = any(np.max(np.abs(hc - p)) < 1e-7 for p in pool)
        if not dup:
            pool.append(hc.copy())
    return best_h, best_s

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = globals().get('initial_h_values', None)
    n = 45
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.ndim == 1 and 2 <= len(p) <= 999 and np.all(np.isfinite(p)) and np.max(np.abs(p)) < 1e6:
            n = len(p)
            fallback, _ = project_capped(p, n)
        else:
            fallback = np.full(n, 0.5)
    else:
        nk = kwargs.get('n_points', None)
        if isinstance(nk, int) and 2 <= nk <= 999:
            n = nk
        fallback = np.full(n, 0.5)
    fallback, _ = project_capped(fallback, n)
    fb_score = exact_score(fallback, n)
    best_h, best_s, pool = fallback.copy(), fb_score, []
    S, dx = n / 2.0, 2.0 / n
    for sig in [0.03, 0.08, 0.20, 0.40]:
        best_h, best_s = record(fallback + sig * rng.standard_normal(n), n, best_h, best_s, pool)
    h_rel, X_rel = solve_sdp(n, S, dx, fb_score)
    if h_rel is not None:
        hp, ok = project_capped(h_rel, n)
        if ok:
            best_h, best_s = record(hp, n, best_h, best_s, pool)
            for al in [0.25, 0.5, 0.75]:
                best_h, best_s = record(fallback + al * (hp - fallback), n, best_h, best_s, pool)
        for c in covariance_samples(hp, X_rel, n, rng):
            best_h, best_s = record(c, n, best_h, best_s, pool)
    starts = [fallback.copy()]
    for ph in pool:
        if len(starts) >= 3:
            break
        if np.max(np.abs(ph - fallback)) > 1e-6:
            starts.append(ph.copy())
    for ps in starts:
        ph, _ = polish_start(ps, n, rng)
        best_h, best_s = record(ph, n, best_h, best_s, pool)
    for ph in pool[:3]:
        ph2, _ = polish_start(ph, n, rng)
        best_h, best_s = record(ph2, n, best_h, best_s, pool)
    final_h, ok = project_capped(best_h, n)
    if not ok:
        final_h = fallback.copy()
    final_s = exact_score(final_h, n)
    if final_s > fb_score + 1e-12 or final_s == float('inf'):
        final_h, final_s = fallback.copy(), fb_score
    return final_h.astype(np.float64), float(final_s), int(n)
