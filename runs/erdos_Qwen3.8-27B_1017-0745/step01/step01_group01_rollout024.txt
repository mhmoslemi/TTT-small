import numpy as np

def project_cap(v, n):
    s = n / 2.0
    if v.shape != (n,) or not np.all(np.isfinite(v)):
        return None
    v = np.clip(v, -1e10, 1e10)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    r = s - h.sum()
    for _ in range(3):
        if abs(r) < 1e-13:
            break
        if r > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) == 0:
                break
            i = idx[0]
            h[i] += min(r, 1.0 - h[i])
            r -= min(r, 1.0 - h[i])
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) == 0:
                break
            i = idx[0]
            h[i] += max(r, -h[i])
            r -= max(r, -h[i])
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(h.sum() - s) > 1e-10:
        return None
    return h

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if h.min() < -1e-12 or h.max() > 1.0 + 1e-12:
        return float('inf')

        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c) * (2.0/n))

def lag_values(h, n):
    dx = 2.0 / n
    L = 2*n - 1
    g = np.zeros(L)
    for d in range(-(n-1), n):
        row = d + n - 1
        i0 = max(0, -d)
        j0 = max(0, d)
        m = n - abs(d)
        g[row] = dx * np.sum(h[i0:i0+m] * (1.0 - h[j0:j0+m]))
    return g

def smooth_grad(h, n, tau):
    dx = 2.0 / n
    g = lag_values(h, n)
    L = 2*n - 1
    gmax = float(np.max(g))
    exps = np.exp((g - gmax) / tau)
    p = exps / exps.sum()
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        row = d + n - 1
        i0 = max(0, -d)
        j0 = max(0, d)
        m = n - abs(d)
        w = dx * p[row]
        grad[i0:i0+m] += w * (1.0 - h[j0:j0+m])
        grad[j0:j0+m] -= w * h[i0:i0+m]
    F = tau * (gmax + np.log(exps.sum()))
    return F, grad
