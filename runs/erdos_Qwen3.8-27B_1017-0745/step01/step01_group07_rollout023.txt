import numpy as np
from math import log, sqrt, ceil, pi

def proj_capped(y, S):
    y = np.asarray(y, dtype=np.float64)
    if y.ndim != 1 or not np.all(np.isfinite(y)) or S < 0 or S > len(y):
        return None
    if np.max(np.abs(y)) > 1e12:
        return None
    n = len(y)
    lo, hi = np.min(y) - 1.0, np.max(y)
    for _ in range(80):
        mid = lo + 0.5 * (hi - lo)
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    r = S - np.sum(x)
    for i in range(n):
        if r > 1e-14:
            a = min(r, 1.0 - x[i])
            x[i] += a
            r -= a
        elif r < -1e-14:
            a = min(-r, x[i])
            x[i] -= a
            r += a
    if abs(np.sum(x) - S) > 1e-10 or not np.all(np.isfinite(x)):
        return None
    return x

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-8)

def exact_score(h, n):
    try:
        vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    except Exception:
        return None
    if vals.size != 2 * n - 1 or not np.all(np.isfinite(vals)):
        return None
    return float(np.max(vals))

def build_basis(n, D):
    L = ceil(D / 2)
    K = list(range(1, L + 1))
    R = D - L
    if R == 1:
        K.append(n - 1)
    elif R > 1:
        for t in range(R):
            c = int(round(L + 1 + t * (n - 1 - L - 1) / (R - 1)))
            if c not in K and L + 1 <= c <= n - 1:
                K.append(c)
    for k in range(L + 1, n):
        if len(K) >= D:
            break
        if k not in K:
            K.append(k)
    K = sorted(K[:D])
    idx = np.arange(n, dtype=np.float64)
    B = np.zeros((n, D))
    for j in range(D):
        B[:, j] = sqrt(2.0 / n) * np.cos(pi * K[j] * (idx + 0.5) / n)
    B -= np.mean(B, axis=0, keepdims=True)
    try:
        Q, _ = np.linalg.qr(B)
        B = Q[:, :D]
    except Exception:
        return None
    if B.shape != (n, D) or not np.all(np.isfinite(B)):
        return None
    return B

    P = 4 + int(3 * log(D))
    mu = P // 2
    w = np.array([log(mu + 0.5) - log(i + 1) for i in range(mu)])
    w /= w.sum()
    mueff = 1.0 / np.sum(w ** 2)
    c_s = (mueff + 2.0) / (D + mueff + 5.0)
    d_s = 1.0 + 2.0 * max(0.0, sqrt((mueff - 1.0) / (D + 1.0)) - 1.0) + c_s
    c_c = (4.0 + mueff / D) / (D + 4.0 + 2.0 * mueff / D)
    c1 = 2.0 / ((D + 1.3) ** 2 + mueff)
    cmu = min(1.0 - c1, 2.0 * (mueff - 2.0 + 1.0 / mueff) / ((D + 2.0) ** 2 + mueff))
    chi_D = sqrt(D) * (1.0 - 1.0 / (4.0 * D) + 1.0 / (21.0 * D ** 2))
    m = np.zeros(D)
    C = np.eye(D)
    ps = np.zeros(D)
    pc = np.zeros(D)

    else:
        sigma = float(np.clip(sigma_init, 0.03, 1.5))
    best_rs = arch[1]

        except Exception:
            break

        nvalid = 0

def slsqp_polish(h0, n, rng):
    from scipy.optimize import minimize
    S = n / 2.0
    defs = {}
    def o_fun(z):
        return z[-1]
    def o_fun_g(z):
        g = np.zeros(z.size)
        g[-1] = 1.0
        return g
    def o_ineq(z):
        h = z[:n]
        B = np.correlate(h, 1.0 - h, mode='full')
        return z[-1] - B
    def o_ineq_j(z):
        h = z[:n]
        G = np.zeros((2 * n - 1, n))
        for row in range(2 * n - 1):
            d = row - (n - 1)
            lo = max(0, -d)
            hi = min(n, n - d)
            G[row, lo:hi] += (1.0 - h[lo + d:hi + d])
            G[row, lo + d:hi + d] -= h[lo:hi]
        J = np.column_stack([-G, np.ones(2 * n - 1)])
        return J
    def o_eq(z):
        return np.sum(z[:n]) - S
    def o_eq_j(z):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    try:
        B0 = np.correlate(h0, 1.0 - h0, mode='full')
        t0 = float(np.max(B0)) + 1e-8
        z0 = np.append(h0.copy(), t0)
        res = minimize(o_fun, z0, jac=o_fun_g, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n + [(None, None)],
                       constraints=[{'type': 'ineq', 'fun': o_ineq, 'jac': o_ineq_j},
                                     {'type': 'eq', 'fun': o_eq, 'jac': o_eq_j}],
