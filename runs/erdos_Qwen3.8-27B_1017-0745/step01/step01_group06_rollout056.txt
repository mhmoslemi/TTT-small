                delta *= 0.5
            elif r > 0.7 and np.max(np.abs(d_a)) >= 0.8 * delta:
                delta = min(0.5, 2.0 * delta)
                eta = min(2.0, 1.4 * eta)
            if delta < 1e-6:
                break
        else:
            rej += 1
            delta *= 0.5
            eta = max(1e-4, 0.5 * eta)
            if rej >= 10 or delta < 1e-5:
                break
    h = project_K(np.clip(h, 0.0, 1.0), n, S)
    return h, exact_score(h, dx)

def transfer_parent(hp, n0, m):
    if m == n0:
        return hp.copy()
    dx0, dxm = 2.0 / n0, 2.0 / m
    y = np.zeros(m)
    for j in range(m):
        a, b = j * dxm, (j + 1) * dxm
        tot, ws = 0.0, 0.0
        for i in range(n0):
            ov = min((i + 1) * dx0, b) - max(i * dx0, a)
            if ov > 0:
                tot += hp[i] * ov
                ws += ov
        y[j] = tot / ws if ws > 1e-15 else 0.5
    return y

def make_starts(base, n, S, rng):
    j = np.arange(n)
    blk = np.zeros(n)
    blk[:int(S)] = 1.0
    if S > int(S):
        blk[int(S)] = S - int(S)
    cands = [
        base.copy(), (1.0 - base).copy(), base[::-1].copy(),
        project_K(0.5 + 0.45 * np.cos(2.0 * np.pi * j / n), n, S),
        project_K(0.5 + 0.40 * np.sin(4.0 * np.pi * j / n), n, S),
        project_K(0.5 + 0.35 * np.cos(6.0 * np.pi * j / n + 0.5), n, S),
        project_K(blk, n, S),
        project_K(base + 0.12 * rng.standard_normal(n), n, S),
        project_K(base + 0.35 * rng.standard_normal(n), n, S),
        project_K(base + 0.60 * rng.standard_normal(n), n, S),
        project_K(rng.uniform(0, 1, n), n, S),
        project_K(rng.beta(0.4, 0.6, n), n, S),
    ]
    seen, out = [], []
    for c in cands:
        cp = project_K(c, n, S)
        if not feasible(cp, n, S):
            continue
        dup = any(np.max(np.abs(cp - s)) <= 1e-5 for s in seen)
        if not dup:
            seen.append(cp)
            out.append(cp)
    return out

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    init = globals().get('initial_h_values', None)
    ok_p = False
    if init is not None:
        try:
            ia = np.array(init, dtype=np.float64).ravel()
            n0 = len(ia)
            if (2 <= n0 <= 999 and np.all(np.isfinite(ia))
                    and ia.min() >= -1e-6 and ia.max() <= 1.0 + 1e-6
                    and abs(ia.sum() - n0 / 2.0) <= 1e-4):
                ok_p = True
        except Exception:
            pass
    if ok_p:
        hp = project_K(ia.copy(), n0, n0 / 2.0)
        res_list = [n0]
        for m in [n0 - 1, n0 + 1, 40, 50, 44, 46]:
            if 2 <= m <= 999 and m not in res_list:
                res_list.append(m)
    else:
        n0 = 45
        hp = np.full(n0, 0.5)
        res_list = [n0, n0 - 1, n0 + 1]

    gH, gN, gS = hp.copy(), n0, exact_score(hp, 2.0 / n0)
    fH, fS = hp.copy(), gS

    for m in res_list:
        Sm, dxm = m / 2.0, 2.0 / m
        if ok_p and m != n0:
            base = project_K(transfer_parent(ia, n0, m), m, Sm)
        else:
            base = project_K(hp.copy() if m == n0 else np.full(m, 0.5), m, Sm)
        ns = 10 if m <= 100 else 4
        mi = 35 if m <= 100 else 12
        starts = make_starts(base, m, Sm, rng)[:ns]
        for s in starts:
            ho, sc = local_opt(s, m, dxm, mi)
            if sc < gS - 1e-12:
                gH, gN, gS = ho.copy(), m, sc
        if m <= 100:
            for sig in [0.15, 0.30]:
                hs = project_K(base + sig * rng.standard_normal(m), m, Sm)
                ho, sc = local_opt(hs, m, dxm, mi)
                if sc < gS - 1e-12:
                    gH, gN, gS = ho.copy(), m, sc

    fm = gN
    fH2 = project_K(np.clip(gH, 0.0, 1.0), fm, fm / 2.0)
    fS2 = exact_score(fH2, 2.0 / fm)
    if not feasible(fH2, fm, fm / 2.0) or fS2 > fS + 1e-10:
        fH2, fS2, fm = np.clip(fH, 0.0, 1.0), fS, n0
    elif fS2 > gS - 1e-10 and gN > n0:
        fH2, fS2, fm = np.clip(fH, 0.0, 1.0), fS, n0
    return (fH2, fS2, fm)
