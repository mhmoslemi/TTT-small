import numpy as np

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    if S < 0 or S > n:
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(v - tau, 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-14:
        for i in np.argsort(1.0 - y)[::-1]:
            d = min(r, 1.0 - y[i])
            y[i] += d
            r -= d
            if r <= 1e-14:
                break
    elif r < -1e-14:
        for i in np.argsort(y)[::-1]:
            d = min(-r, y[i])
            y[i] -= d
            r += d
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-9) or np.any(y > 1.0 + 1e-9):
        return None
    if abs(float(np.sum(y)) - S) > 1e-9 * max(1, n):
        return None

        g_old = g.copy()
        g = g_new
        sc = sc + 1 if F_o - F < 1e-12 else 0
        if sc >= 8:
            break
        if (it + 1) % 10 == 0:
            c = exact_score(h, n)
            if c < sbc:
                sb = h.copy()
                sbc = c
    c = exact_score(h, n)
    if c < sbc:
        sb = h.copy()
        sbc = c
    return h, sb, sbc

def get_parent():
    n = 45
    ph = globals().get('initial_h_values', None)
    if ph is not None:
        ph = np.asarray(ph, dtype=np.float64)
        if ph.ndim == 1 and 2 <= len(ph) <= 999 and np.all(np.isfinite(ph)):
            if np.all(ph >= -1e-6) and np.all(ph <= 1.0 + 1e-6):
                n = len(ph)
                p = project_box_sum(np.clip(ph, 0.0, 1.0), 0.5 * n)
                if p is not None:
                    return p, n
    return project_box_sum(np.full(45, 0.5), 22.5), 45

def build_starts(parent, n, rng):
    S = 0.5 * n
    cands = [parent.copy(), 1.0 - parent.copy(), parent[::-1].copy()]
    cands.append(parent + 0.07 * rng.standard_normal(n))
    cands.append(parent + 0.20 * rng.standard_normal(n))
    cands.append(rng.random(n))
    idx = np.arange(n, dtype=np.float64)
    cands.append(0.5 + 0.32 * np.cos(2.0 * np.pi * idx / n + 0.7) +
                 0.12 * np.sin(4.0 * np.pi * idx / n + 1.3))
    starts = []
    for c in cands:
        p = project_box_sum(c, S)
        if p is not None and not any(np.max(np.abs(p - s)) < 1e-10 for s in starts):
            starts.append(p)
    return starts

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent, n = get_parent()
    S = 0.5 * n
    NT = 7
    taus = [0.04 * (0.3 ** j) for j in range(NT)]
    M = max(30, min(120, 5400 // n))
    best_h = parent.copy()
    best_c = exact_score(parent, n)
    starts = build_starts(parent, n, rng)
    for h0 in starts:
        hc = h0.copy()
        for tau in taus:
            hc, hsb, csb = pstage(hc, tau, M, n)
            if csb < best_c:
                best_h = hsb.copy()
                best_c = csb
        ce = exact_score(hc, n)
        if ce < best_c:
            best_h = hc.copy()
            best_c = ce
    for sig in [0.04, 0.12, 0.30]:
        h0 = project_box_sum(best_h + sig * rng.standard_normal(n), S)
        if h0 is None:
            continue
        hc = h0.copy()
        M2 = max(20, M // 2)
        for tau in taus:
            hc, hsb, csb = pstage(hc, tau, M2, n)
            if csb < best_c:
                best_h = hsb.copy()
                best_c = csb
        ce = exact_score(hc, n)
        if ce < best_c:
            best_h = hc.copy()
            best_c = ce
    best_h = project_box_sum(best_h, S)
    if best_h is None or not check_feasible(best_h, n):
        best_h = parent.copy()
    bf = exact_score(best_h, n)
    pf = exact_score(parent, n)
    if pf < bf:
        best_h = parent.copy()
        bf = pf
    if not np.isfinite(bf):
        best_h = project_box_sum(np.full(n, 0.5), S)
        bf = exact_score(best_h, n)
    return (best_h, bf, n)
