import numpy as np

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(z - lam, 0.0, 1.0)
    r = m - h.sum()
    if r > 1e-14:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            if adj <= 0:
                continue
            h[i] += adj
            r -= adj
            if r <= 1e-14:
                break
    elif r < -1e-14:
        for i in range(N):
            adj = min(-r, h[i])
            if adj <= 0:
                continue
            h[i] -= adj
            r += adj
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1 + 1e-10:
        return np.full(N, 0.5)
    if abs(h.sum() - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / N)

def smooth_val_grad(h, tau):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = dx * corr
    cmax = c.max()
    e = np.exp((c - cmax) / tau)
    s = e.sum()
    F = cmax + tau * np.log(s)
    p = e / s
    grad = np.zeros(N)
    for idx in range(2 * N - 1):
        w = dx * p[idx]
        if w < 1e-15:
            continue
        k = idx - (N - 1)
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            d = k
            L = N - d
            grad[:L] += w * (1.0 - h[d:L + d])  # (1-h[j+d]) for j in [0, L-1]
            grad[d:] -= w * h[:L]  # -h[j-d] for j in [d, N-1]
        else:
            d = -k
            L = N - d
            grad[:L] += w * (1.0 - h[d:L + d])  # (1-h[j+d]) for j in [0, L-1]
            grad[d:] -= w * h[:L]  # -h[j-d] for j in [d, N-1]
    return F, grad, c

def pg_optimize(h0, N, m, max_updates=50):
    dx = 2.0 / N
    taus = [0.02, 0.006, 0.002, 0.0006, 0.0002, 0.00006, 0.000015]
    best_h = h0.copy()
    best_score = score_exact(h0, N)
    h = h0.copy()
    alpha = 1.0
    for tau in taus:
        F, grad, c = smooth_val_grad(h, tau)
        if not np.all(np.isfinite(grad)):
            break
        updates = 0
        for it in range(max_updates):
            y = project_box_sum(h - alpha * grad, m)
            d = y - h
            if not np.all(np.isfinite(d)) or np.max(np.abs(d)) < 1e-10:
                break
            Fy, _, _ = smooth_val_grad(y, tau)
            # Check exact score for candidate
            sy = score_exact(y, N)
            if np.isfinite(sy) and sy < best_score:
                best_h = y.copy()
                best_score = sy
            gd = np.dot(grad, d)
            if gd < 0 and Fy <= F + 1e-4 * gd + 1e-12:
                h = y
                alpha = min(1.25 * alpha, 10.0)
                F = Fy
                updates += 1
            else:
                alpha *= 0.5
                if alpha < 1e-12:
                    break
    return best_h, best_score

def run(seed=42, **kwargs):

    m = None
    raw = globals().get('initial_h_values', None)
    if raw is not None:
        raw = np.asarray(raw, dtype=np.float64)

            N = raw.size
            m = N / 2.0
    if N is None:
        N = kwargs.get('n_points', 70)
        if not isinstance(N, (int, np.integer)) or N < 2:
            N = 70
        m = N / 2.0
    if N == 1:
        h = np.array([0.5])
        return h, score_exact(h, 1), 1
    
    rng = np.random.default_rng(seed)
    
    # Baseline
    if raw is not None and np.all(np.isfinite(raw)):
        best = project_box_sum(raw.copy(), m)
    else:
        best = np.full(N, 0.5)
    best_score = score_exact(best, N)
    
    # Tiered parameters
    if N <= 128:
        n_starts, max_upd, n_hops = 8, 50, 2
    elif N <= 256:
        n_starts, max_upd, n_hops = 6, 30, 1
    elif N <= 512:
        n_starts, max_upd, n_hops = 4, 16, 0
    else:
        n_starts, max_upd, n_hops = 3, 8, 0
    
    # Build starts
    starts = [best.copy()]
    # Complement
    starts.append(project_box_sum(1.0 - best, m))
    # Perturbations
    for sc in [0.04, 0.12]:
        starts.append(project_box_sum(best + sc * rng.standard_normal(N), m))
    # Random uniform
    starts.append(project_box_sum(rng.uniform(size=N), m))
    # Fourier profiles
    idx = np.arange(N)
    for amp, R in [(0.12, 4), (0.22, 3)]:
        z = np.full(N, 0.5)
        ns = min(R, (N - 1) // 2)
        for r in range(1, ns + 1):
            A = rng.standard_normal()
            B = rng.standard_normal()
            z += (amp / r) * (A * np.sin(2.0 * np.pi * r * idx / N) + B * np.cos(2.0 * np.pi * r * idx / N))
        starts.append(project_box_sum(z, m))
    
    # Deduplicate
    dedup = [starts[0]]
    for s in starts[1:]:
        if all(np.max(np.abs(s - d)) > 1e-10 for d in dedup):
            dedup.append(s)
        if len(dedup) >= n_starts:
            break
    starts = dedup
    
    # If no parent, add more random starts
    if raw is None:
        while len(starts) < min(n_starts, 8):
            cand = project_box_sum(0.3 + 0.4 * rng.uniform(size=N), m)
            if all(np.max(np.abs(cand - d)) > 1e-10 for d in starts):

            best = h_opt.copy()
            best_score = s_opt

    best = project_box_sum(best, m)
    c5 = score_exact(best, N)
    if not np.isfinite(c5):
        best = np.full(N, 0.5)
        c5 = score_exact(best, N)
    return best, float(c5), int(N)

import numpy as np

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(z - lam, 0.0, 1.0)
    r = m - h.sum()
    if r > 1e-14:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            if adj <= 0:
                continue
            h[i] += adj
            r -= adj
            if r <= 1e-14:
                break
    elif r < -1e-14:
        for i in range(N):
            adj = min(-r, h[i])
            if adj <= 0:
                continue
            h[i] -= adj
            r += adj
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1 + 1e-10:
        return np.full(N, 0.5)
    if abs(h.sum() - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / N)

def smooth_val_grad(h, tau):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = dx * corr
    cmax = c.max()
    e = np.exp((c - cmax) / tau)
    s = e.sum()
    F = cmax + tau * np.log(s)
    p = e / s
    grad = np.zeros(N)
    for idx in range(2 * N - 1):
        w = dx * p[idx]
        if w < 1e-15:
            continue
        k = idx - (N - 1)
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            L = N - k
            grad[k:] += w * (1.0 - h[:L])
            grad[:L] -= w * h[k:]
        else:
            d = -k
            L = N - d
            grad[:L] += w * (1.0 - h[d:])
            grad[d:] -= w * h[:L]
    return F, grad, c

    best_h = h0.copy()
    best_score = score_exact(h0, N)
    h = h0.copy()
    alpha = 1.0
    for tau in taus:
        F, grad, c = smooth_val_grad(h, tau)
        if not np.all(np.isfinite(grad)):
            break
        for it in range(max_updates):
            y = project_box_sum(h - alpha * grad, m)
            d = y - h
            if not np.all(np.isfinite(d)) or np.max(np.abs(d)) < 1e-10:
                break
            Fy, _, _ = smooth_val_grad(y, tau)
            sy = score_exact(y, N)
            if np.isfinite(sy) and sy < best_score:
                best_h = y.copy()
                best_score = sy
            if not np.isfinite(Fy):

            else:

                    break

def lag_target_step(h, N, m, rng, max_lags=3, step=0.005):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = dx * corr
    topk = np.argsort(c)[-max_lags:]
    grad = np.zeros(N)
    for idx in topk:
        w = c[idx]
        k = idx - (N - 1)
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            L = N - k
            grad[k:] += w * (1.0 - h[:L])
            grad[:L] -= w * h[k:]
        else:
            d = -k
            L = N - d
            grad[:L] += w * (1.0 - h[d:])
            grad[d:] -= w * h[:L]
    gn = np.linalg.norm(grad)
    if gn < 1e-12:
        return h
    direction = -grad / gn
    hop = project_box_sum(h + step * direction + 0.01 * rng.standard_normal(N), m)
    return hop
