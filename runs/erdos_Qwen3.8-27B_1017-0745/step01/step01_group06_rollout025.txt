import numpy as np
from scipy.optimize import minimize

def project_K(y, n, S):
    if not np.all(np.isfinite(y)):
        return np.full(n, S / n)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    err = S - h.sum()
    for _ in range(3):
        if abs(err) < 1e-14:
            break
        if err > 0:
            idx = np.where(1.0 - h > 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) == 0:
            break
        adj = err / len(idx)
        h[idx] = np.clip(h[idx] + adj, 0.0, 1.0)
        err = S - h.sum()
    if abs(h.sum() - S) > 1e-8:
        return np.full(n, S / n)
    return h

def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def lag_vec(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_grad(h, n, dx):
    G = np.zeros((2 * n - 1, n))
    j = np.arange(n)
    for k in range(2 * n - 1):
        ell = k - (n - 1)
        if ell == 0:
            G[k, :] = dx * (1.0 - 2.0 * h)
        else:
            i1 = j - ell
            v1 = (i1 >= 0) & (i1 < n)
            i2 = j + ell
            v2 = (i2 >= 0) & (i2 < n)
            row = np.zeros(n)
            row[v1] = dx * (1.0 - h[i1[v1]])
            row[v2] = -dx * h[i2[v2]]
            G[k, :] = row
    return G
```

def tr_obj(z, n, eta):
    d = z[:n]
    return z[-1] + 0.5 / eta * d @ d

def tr_obj_jac(z, n, eta):
    d = z[:n]
    g = np.zeros(n + 1)
    g[:n] = d / eta
    g[-1] = 1.0
    return g

def tr_ineq(z, n, q, G):
    d = z[:n]
    return z[-1] - q - G @ d

def tr_ineq_jac(z, n, q, G):
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def tr_eq(z, n):
    return z[:n].sum()

def tr_eq_jac(z, n):
    J = np.zeros((1, n + 1))
    J[0, :n] = 1.0
    return J
```

def local_solve(h0, n, dx, rng, max_iter=28, slsqp_maxiter=50):
    S = n / 2.0
    h = project_K(h0, n, S)
    score = exact_score(h, dx)
    delta = 0.12
    eta = 0.5
    consecutive_fail = 0
    consecutive_reject = 0
    
    for it in range(max_iter):
        if delta < 1e-5 or consecutive_fail >= 4 or consecutive_reject >= 8:
            break
        q = lag_vec(h, dx)
        G = lag_grad(h, n, dx)
        qmax = q.max()
        
        z0 = np.concatenate([np.zeros(n), [qmax]])
        bounds = []
        for jj in range(n):
            lo_b = -min(delta, h[jj])
            hi_b = min(delta, 1.0 - h[jj])
            bounds.append((lo_b, hi_b))
        bounds.append((None, None))
        
        con = [
            {'type': 'ineq', 'fun': tr_ineq, 'jac': tr_ineq_jac, 'args': (n, q, G)},
            {'type': 'eq', 'fun': tr_eq, 'jac': tr_eq_jac, 'args': (n,)}
        ]
        
        try:
            res = minimize(tr_obj, z0, jac=tr_obj_jac, args=(n, eta),
                          method='SLSQP', bounds=bounds, constraints=con,
                          options={'maxiter': slsqp_maxiter, 'ftol': 1e-10, 'disp': False})
        except Exception:
            consecutive_fail += 1
            delta *= 0.5
            continue
        
        if res.x is None or not np.all(np.isfinite(res.x)) or len(res.x) != n + 1:
            consecutive_fail += 1
            delta *= 0.5
            continue
        
        d_sol = res.x[:n]
        # Check bounds
        for jj in range(n):
            if d_sol[jj] < -min(delta, h[jj]) - 1e-5 or d_sol[jj] > min(delta, 1.0 - h[jj]) + 1e-5:
                consecutive_fail += 1
                delta *= 0.5
                break
        else:
            # Check equality
            if abs(d_sol.sum()) > 1e-5:
                consecutive_fail += 1
                delta *= 0.5
            else:
                h_trial = project_K(h + d_sol, n, S)
                if not feasible(h_trial, n, S):
                    consecutive_fail += 1
                    delta *= 0.5
                else:
                    d_actual = h_trial - h
                    pred = qmax - np.max(q + G @ d_actual)
                    actual = score - exact_score(h_trial, dx)
                    
                    if np.max(np.abs(d_actual)) < 1e-8:
                        break
                    consecutive_fail = 0
                    
                    if actual > 1e-12 and (pred <= 1e-12 or pred > 1e-12 and actual / pred >= 0.1):
                        h = h_trial
                        score = exact_score(h, dx)
                        consecutive_reject = 0
                        ratio = actual / pred if pred > 1e-12 else 1.0
                        if ratio > 0.75 and np.max(np.abs(d_actual)) >= 0.8 * delta:
                            delta = min(0.5, 2.0 * delta)
                            eta = min(2.0, 1.5 * eta)
                        elif pred <= 1e-12 or ratio < 0.25:
                            delta *= 0.5
                    else:
                        consecutive_reject += 1
                        delta *= 0.5
                        eta = max(1e-4, eta * 0.5)
    
    return h, exact_score(h, dx)

def transfer_parent(h_parent, n0, m):
    S = m / 2.0
    if n0 == m:
        return project_K(h_parent.copy(), m, S)
    y = np.zeros(m)
    for j in range(m):
        # New cell j is [j/m, (j+1)/m] in [0,1]
        # Old cell i is [i/n0, (i+1)/n0] in [0,1]
        # Overlap integral = integral of h_parent over intersection

import numpy as np
from scipy.optimize import minimize

def project_K(y, n, S):
    if not np.all(np.isfinite(y)):
        return np.full(n, S / n)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    err = S - h.sum()
    for _ in range(4):
        if abs(err) < 1e-14:
            break
        if err > 0:
            idx = np.where(1.0 - h > 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) == 0:
            break
        h[idx] = np.clip(h[idx] + err / len(idx), 0.0, 1.0)
        err = S - h.sum()
    if abs(h.sum() - S) > 1e-8:
        return np.full(n, S / n)
    return h

def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def lag_vec(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_grad(h, n, dx):
    G = np.zeros((2 * n - 1, n))
    j = np.arange(n)
    for k in range(2 * n - 1):
        ell = k - (n - 1)
        i1, v1 = j - ell, (j - ell >= 0) & (j - ell < n)
        i2, v2 = j + ell, (j + ell >= 0) & (j + ell < n)
        row = np.zeros(n)
        row[v1] = dx * (1.0 - h[i1[v1]])
        row[v2] = -dx * h[i2[v2]]
        G[k, :] = row
    return G

def tr_obj(z, n, eta):
    return z[-1] + 0.5 / eta * np.dot(z[:n], z[:n])

def tr_obj_jac(z, n, eta):
    g = np.zeros(n + 1)
    g[:n] = z[:n] / eta
    g[-1] = 1.0
    return g

def tr_ineq(z, n, q, G):
    return z[-1] - q - G @ z[:n]

def tr_ineq_jac(z, n, q, G):
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def tr_eq(z, n):
    return z[:n].sum()

def tr_eq_jac(z, n):
    J = np.zeros((1, n + 1))
    J[0, :n] = 1.0
    return J

def local_solve(h0, n, dx, max_iter, slsqp_it):
    S = n / 2.0
    h = project_K(h0, n, S)
    score = exact_score(h, dx)
    delta, eta = 0.12, 0.5
    cfail, crej = 0, 0
    for _ in range(max_iter):
        if delta < 1e-5 or cfail >= 4 or crej >= 8:
            break
        q = lag_vec(h, dx)
        G = lag_grad(h, n, dx)
        qmax = float(q.max())
        z0 = np.concatenate([np.zeros(n), [qmax]])
        bnds = [(-min(delta, h[jj]), min(delta, 1.0 - h[jj])) for jj in range(n)]
        bnds.append((None, None))
        con = [{'type': 'ineq', 'fun': tr_ineq, 'jac': tr_ineq_jac, 'args': (n, q, G)},
               {'type': 'eq', 'fun': tr_eq, 'jac': tr_eq_jac, 'args': (n,)}]
        try:
            res = minimize(tr_obj, z0, jac=tr_obj_jac, args=(n, eta),
                          method='SLSQP', bounds=bnds, constraints=con,
                          options={'maxiter': slsqp_it, 'ftol': 1e-10, 'disp': False})
        except Exception:
            cfail += 1; delta *= 0.5; continue
        if res.x is None or len(res.x) != n + 1 or not np.all(np.isfinite(res.x)):
            cfail += 1; delta *= 0.5; continue
        d_sol = res.x[:n]
        bad = abs(d_sol.sum()) > 1e-4
        if not bad:
            for jj in range(n):
                if d_sol[jj] < -min(delta, h[jj]) - 1e-4 or d_sol[jj] > min(delta, 1 - h[jj]) + 1e-4:
                    bad = True; break
        if bad:
            cfail += 1; delta *= 0.5; continue
        cfail = 0
        ht = project_K(h + d_sol, n, S)
        if not feasible(ht, n, S):
            crej += 1; delta *= 0.5; continue
        da = ht - h
        if np.max(np.abs(da)) < 1e-8:
            break
        pred = qmax - float(np.max(q + G @ da))
        actual = score - exact_score(ht, dx)
        if actual > 1e-12 and (pred <= 1e-12 or actual / max(pred, 1e-15) >= 0.1):
            h = ht; score = exact_score(h, dx); crej = 0
            r = actual / pred if pred > 1e-12 else 1.0

        else:
            crej += 1; delta *= 0.5; eta = max(1e-4, eta * 0.5)
    return h, exact_score(h, dx)

def transfer_parent(hp, n0, m):
    S = m / 2.0
    if n0 == m:
        return project_K(hp.copy(), m, S)
    y = np.zeros(m)
    for j in range(m):
        acc = 0.0
        for i in range(n0):
            lo = max(i / n0, j / m)
            hi = min((i + 1) / n0, (j + 1) / m)
            if hi > lo:
                acc += hp[i] * (hi - lo)
        y[j] = m * acc
    return project_K(y, m, S)

def get_params(m):
    if m <= 96:
        return 8, True, 28, 50
    elif m <= 256:
        return 3, False, 12, 30
    return 1, False, 5, 15

def run(seed=42, **kwargs):

    init = globals().get('initial_h_values', None)
    ok = False
    n_parent = 0
    if init is not None:
        init = np.array(init, dtype=np.float64).ravel()
        if (2 <= len(init) <= 999 and np.all(np.isfinite(init)) and

            abs(init.sum() - len(init) / 2.0) <= 1e-8):

            n_parent = len(init)
    nd = kwargs.get('n_points', 45)

    nd = max(2, min(999, int(nd)))

    fb_h = project_K(init.copy(), fb_n, fb_n / 2.0) if ok else np.full(fb_n, 0.5)
    fb_score = exact_score(fb_h, 2.0 / fb_n)

        dx = 2.0 / m
        n_starts, do_restart, mi, si = get_params(m)
        if ok and m == n_parent:

        elif ok:
            base = transfer_parent(init, n_parent, m)
        else:
            base = np.full(m, 0.5)
        jj = np.arange(m)
        starts = [base.copy(), (1.0 - base).copy(), base[::-1].copy(),

        hb = np.zeros(m)
