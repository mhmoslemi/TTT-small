def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)):
        return None
    if np.max(np.abs(z)) > 1e12:

    n = len(z)

        return None

        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    y = np.clip(z - lam, 0.0, 1.0)
    r = S - np.sum(y)
    if r > 0:
        for i in range(n):
            if r <= 0: break
            a = min(r, 1.0 - y[i])

import numpy as np

def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or np.max(np.abs(z)) > 1e12:
        return None
    n = len(z)
    if S < 0 or S > n:
        return None
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    y = np.clip(z - lam, 0.0, 1.0)
    r = S - np.sum(y)
    if r > 0:
        for i in range(n):
            if r <= 0: break
            a = min(r, 1.0 - y[i]); y[i] += a; r -= a
    elif r < 0:
        for i in range(n):
            if r >= 0: break
            a = min(-r, y[i]); y[i] -= a; r += a
    y = np.clip(y, 0.0, 1.0)
    if abs(np.sum(y) - S) > 1e-10 or not np.all(np.isfinite(y)):
        return None
    return y

def is_feasible(h, n):
    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-10)

def exact_score(h, n):
    try:
        vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if not np.all(np.isfinite(vals)):
            return None
        return float(np.max(vals))
    except Exception:
        return None

def build_pairs(n):
    left = np.repeat(np.arange(n, dtype=np.int32), n)
    right = np.tile(np.arange(n, dtype=np.int32), n)
    lag_id = (right - left + (n - 1)).astype(np.int32)
    return left, right, lag_id

def softmax_obj(h, tau, left, right, lag_id, n):
    dx = 2.0 / n
    B = np.bincount(lag_id, weights=h[left] * (1.0 - h[right]), minlength=2 * n - 1)
    q = dx * B
    m = np.max(q)
    e = np.exp((q - m) / tau)
    s = np.sum(e)
    if s <= 0 or not np.isfinite(s):
        return None, None
    return m + tau * np.log(s), e / s

def softmax_grad(h, tau, left, right, lag_id, n):
    dx = 2.0 / n
    B = np.bincount(lag_id, weights=h[left] * (1.0 - h[right]), minlength=2 * n - 1)
    q = dx * B
    m = np.max(q)
    e = np.exp((q - m) / tau)
    s = np.sum(e)
    if s <= 0 or not np.isfinite(s):
        return None
    p = e / s
    g = np.zeros(n, dtype=np.float64)
    g += np.bincount(left, weights=p[lag_id] * (1.0 - h[right]), minlength=n) * dx
    g += np.bincount(right, weights=-p[lag_id] * h[left], minlength=n) * dx
    if not np.all(np.isfinite(g)):
        return None
    return g

def project_gradient_path(x0, S, n, left, right, lag_id, rng, max_iter):
    sigmas = [0.02, 0.005, 0.001, 0.0002, 0.00005]
    best_h = x0.copy()
    best_sc = exact_score(x0, n)
    if best_sc is None:
        best_sc = 1e9
    x = x0.copy()
    for tau in sigmas:
        g = softmax_grad(x, tau, left, right, lag_id, n)
        if g is None:
            break
        alpha = 1.0
        prev_x, prev_g = x.copy(), g.copy()
        stall = 0
        for it in range(max_iter):
            Fx, _ = softmax_obj(x, tau, left, right, lag_id, n)
            if Fx is None:
                break
            accepted = False
            for bt in range(24):
                y = project_capped_simplex(x - alpha * g, S)
                if y is None:
                    alpha *= 0.5
                    continue
                if not np.all(np.isfinite(y)) or np.max(np.abs(y - x)) <= 1e-12:
                    break
                Fy, _ = softmax_obj(y, tau, left, right, lag_id, n)
                if Fy is None:
                    alpha *= 0.5
                    continue
                d = y - x
                if np.dot(g, d) >= 0:
                    break
                if Fy <= Fx + 0.1 * np.dot(g, d) + 1e-14:
                    accepted = True
                    break
                alpha *= 0.5
            if not accepted or alpha < 1e-9:
                break
            s = y - x
            gy = softmax_grad(y, tau, left, right, lag_id, n)
            sc = exact_score(y, n)
            if sc is not None and sc < best_sc - 1e-12:
                best_sc = sc
                best_h = y.copy()
            if gy is not None:
                yv = gy - g
                sy = np.dot(s, yv)
                if sy > 1e-18:
                    alpha = np.dot(s, s) / sy
                else:
                    alpha = min(10.0, 2.0 * alpha)
                alpha = max(1e-6, min(10.0, alpha))
            else:
                alpha = min(10.0, 1.5 * alpha)
            stall = stall + 1 if (Fy > Fx - 1e-12 and np.max(np.abs(s)) < 1e-6) else 0
            x = y
            g = gy if gy is not None else g
            if stall >= 8:
                break
    return best_h, best_sc

def build_starts(n, S, parent, rng):
    starts = [np.full(n, 0.5)]
    if parent is not None:
        starts.append(parent.copy())
        p1 = project_capped_simplex(parent + rng.normal(0, 0.04, n), S)
        if p1 is not None:
            starts.append(p1)
        p2 = project_capped_simplex(parent + rng.normal(0, 0.12, n), S)

            starts.append(p2)
    idx = np.arange(n, dtype=np.float64) + 0.5
    M = min(6, (n - 1) // 2)
    lf = 0.5 * np.ones(n)
    for m in range(1, M + 1):
        amp = rng.normal(0, 0.12 / np.sqrt(m))
        phi = rng.uniform(0, 2 * np.pi)
        lf += amp * np.cos(2 * np.pi * m * idx / n + phi)
    lf_p = project_capped_simplex(lf, S)
    if lf_p is not None:
        starts.append(lf_p)
    rv = project_capped_simplex(rng.uniform(0, 1, n), S)
    if rv is not None:
        starts.append(rv)
    rv2 = project_capped_simplex(0.5 + 0.25 * rng.normal(0, 1, n), S)
    if rv2 is not None:
        starts.append(rv2)
    deduped = []
    for s in starts:
        if s is None or not is_feasible(s, n):
            continue
        dup = False
        for d in deduped:
            if np.max(np.abs(s - d)) < 1e-8:
                dup = True
                break
        if not dup:
            deduped.append(s)
    return deduped

            final = h_const.copy()
    fs = exact_score(final, n)
