Let me be careful about the gradient formula. For q_k(h) = dx * sum_{r in I_k} h[r]*(1-h[r-k]):

The gradient with respect to h[j]:
- If j is in I_k (role as h[r]): contributes dx * (1 - h[j-k])
- If j+k is in I_k (role as h[r-k] where r = j+k): contributes -dx * h[j+k]

So g_k[j] = dx * (1_{j in I_k} * (1-h[j-k]) - 1_{j+k in I_k} * h[j+k])

Where I_k = {r : max(0,k) <= r <= min(n-1, n-1+k)}

For the dual: D(p) = p^T q + min_{y in P} (p^T G^T (y-h) + (L/2)||y-h||^2)

The minimizer is y = Pi_P(h - G^T p / L), and d = y - h.
D(p) = p^T q + p^T G^T d + (L/2)||d||^2 = p^T (q + G d) + (L/2)||d||^2

Grad D(p) = q + G d (where d depends on p)

Lipschitz constant: since y = Pi_P(h - G^T p / L), and Pi_P is 1-Lipschitz,
||d(p1) - d(p2)|| <= ||G^T(p1-p2)/L|| = ||G^T||/L * ||p1-p2||
||grad D(p1) - grad D(p2)|| = ||G(d1-d2)|| <= ||G||_2 * ||G^T||/L * ||p1-p2|| <= ||G||_F^2 / L * ||p1-p2||

So Ldual = ||G||_F^2 / L is a valid Lipschitz constant. I'll use a safety factor.

import numpy as np
import math

def project_capped(y, n):
    S = n / 2.0
    if not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    lo = float(y.min()) - 1.0
    hi = float(y.max())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        c = np.clip(y - mid, 0.0, 1.0)
        r = c.sum() - S
        if abs(r) < 1e-14:
            break
        if r > 0:
            lo = mid
        else:
            hi = mid
    c = np.clip(y - 0.5*(lo+hi), 0.0, 1.0)
    # Small correction
    for _ in range(5):
        r = S - c.sum()
        if abs(r) < 1e-13:
            break
        if r > 0:
            m = c < 1.0 - 1e-14
        else:
            m = c > 1e-14
        cnt = m.sum()
        if cnt == 0:
            break
        c[m] += r / cnt
        np.clip(c, 0.0, 1.0, out=c)
    return c

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(h.sum() - n/2.0) <= 1e-10)

def lag_scores(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    qs = lag_scores(h, n)
    if not np.all(np.isfinite(qs)):
        return float('inf')
    return float(qs.max())

def build_G(h, n):
    dx = 2.0 / n
    G = np.zeros((2*n - 1, n))
    for k in range(-(n-1), n):
        row = k + n - 1
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if hi1 > lo1:
            G[row, lo1:hi1] += dx * (1.0 - h[lo1-k:hi1-k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if hi2 > lo2:
            G[row, lo2-k:hi2-k] -= dx * h[lo2:hi2]
    return G

def project_simplex(z):
    n = len(z)
    if n == 0:
        return np.array([])
    if not np.all(np.isfinite(z)):
        return np.full(n, 1.0/n)
    sz = np.sort(z)[::-1]
    csum = np.cumsum(sz)
    rho = np.arange(1, n+1)
    cond = sz - (csum - 1.0) / rho > 0
    if not np.any(cond):
        return np.full(n, 1.0/n)
    r = int(np.max(np.where(cond)[0]))
    theta = (csum[r] - 1.0) / (r + 1)
    p = np.maximum(z - theta, 0.0)
    s = p.sum()
    if s > 0:
        p /= s
    return p

def dual_eval(p, G, q, h, n, L):
    n_lag = 2*n - 1
    gmix = G.T @ p
    y = project_capped(h - gmix / L, n)
    d = y - h
    u = q + G @ d
    D = p @ u + 0.5 * L * (d @ d)
    gap = float(u.max() - p @ u)
    return D, u, d, gap

    L = 2.0 * dx
    n_lag = 2*n - 1
    Gnorm2 = float(np.sum(G*G))

    tol = max(1e-12, 1e-8 * max(1.0, abs(qmax)))

        p[mask] = 1.0 / mask.sum()
    else:

        p[np.argmax(q)] = 1.0
    
    D, u, d, gap = dual_eval(p, G, q, h, n, L)
    if not np.isfinite(D):
        return d if np.all(np.isfinite(d)) else np.zeros(n)
    
    x = p.copy()
    yx = p.copy()

    best_u = u.copy()

        if not np.isfinite(D_new):

            x = x_new
            theta = theta_new
            D = D_new
            u = u_new
            d = d_new

            best_d = d.copy()
            best_u = u.copy()

            break
        if np.max(np.abs(step_size)) < 1e-12:
            stall += 1
            if stall >= 5:
                break
        else:
            stall = 0
    
    return best_d

    n = len(z)
    if n == 0:
        return np.array(0.0)
    if not np.all(np.isfinite(z)):
        return np.full(n, 1.0/n)
    sz = np.sort(z)[::-1]
    csum = np.cumsum(sz)
    rho = np.arange(1, n+1, dtype=np.float64)
    cond = sz - (csum - 1.0) / rho > 0
    if not np.any(cond):
        return np.full(n, 1.0/n)
    r = int(np.max(np.where(cond)[0]))
    theta = (csum[r] - 1.0) / (r + 1)

    s = p.sum()
    if s > 1e-300:
        p /= s
    else:
        p = np.full(n, 1.0/n)
    return p

    starts = []
    starts.append(fallback.copy())
    starts.append(1.0 - fallback)
    starts.append(fallback[::-1].copy())
    starts.append(1.0 - fallback[::-1].copy())
    starts.append(project_capped(fallback + 0.05*rng.standard_normal(n), n))
    starts.append(project_capped(fallback + 0.15*rng.standard_normal(n), n))
    starts.append(project_capped(rng.uniform(0, 1, n), n))
    starts = starts[:max_starts]
    unique_starts = [starts[0]]
    for st in starts[1:]:
        if all(float(np.max(np.abs(st - us))) > 1e-8 for us in unique_starts):
            unique_starts.append(st)
    for st in unique_starts:
        try:
            bh, bs = optimize_start(st, n, max_outer, max_inner)
            if check_feasible(bh, n) and bs < best_s - 1e-12:
                best_s = bs
                best_h = bh.copy()
        except Exception:
            pass
    if n <= 100:
        try:
            pert = project_capped(best_h + 0.025*rng.standard_normal(n), n)
            bh, bs = optimize_start(pert, n, max_outer//2, max_inner)
            if check_feasible(bh, n) and bs < best_s - 1e-12:
                best_s = bs
                best_h = bh.copy()
        except Exception:
            pass
    final_h = project_capped(best_h, n)
    if not check_feasible(final_h, n):
        final_h = fallback.copy()
    final_s = exact_score(final_h, n)
    if final_s > fb_s + 1e-12:
        final_h = fallback.copy()
        final_s = fb_s
    return final_h.astype(np.float64), float(final_s), int(n)
