import numpy as np
import math

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0 or not np.all(np.isfinite(v)):
        return None
    if S < -1e-1 or S > n + 1e-1:
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(v - tau, 0.0, 1.0)
    r = S - float(np.sum(y))
    tol = 5e-13 * max(1, n)
    if r > tol:
        for i in range(n):
            if r <= tol:
                break
            d = min(r, 1.0 - y[i])
            y[i] += d
            r -= d
    elif r < -tol:
        for i in range(n):
            if r >= -tol:
                break
            d = min(-r, y[i])
            y[i] -= d
            r += d
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    return y

def check_feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(float(np.sum(h)) - 0.5 * n) < 1e-9 * max(1, n))

def exact_score(h):
    n = len(h)
    if not check_feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

        return (float('inf'), None)

    dx = 2.0 / n
    q = corr * dx
    sc = float(np.max(q))
    if not np.isfinite(sc):
        return (float('inf'), None)
    return (sc, q)

def lag_grad_col(h, k, n):
    """Gradient of all (2n-1) lag scores w.r.t. h[k]"""
    dx = 2.0 / n
    p = np.arange(2 * n - 1)
    s = n - 1 - p  # shifts: n-1, n-2, ..., 0, ..., -(n-1)
    # term 1: (1-h[k+s]) * 1_{0 <= k+s < n}
    ks = k + s
    mask1 = (ks >= 0) & (ks < n)
    g = np.zeros(2 * n - 1)
    g[mask1] += (1.0 - h[ks[mask1]])
    # term 2: -h[k-s] * 1_{0 <= k-s < n}
    ks2 = k - s
    mask2 = (ks2 >= 0) & (ks2 < n)
    g[mask2] -= h[ks2[mask2]]
    return g * dx

def quad_roots(A, B, C):
    """Solve A*t^2 + B*t + C = 0, return real roots"""
    roots = []
    if abs(A) < 1e-18:
        if abs(B) < 1e-18:
            return roots
        t = -C / B
        if np.isfinite(t):
            roots.append(t)
        return roots

        return roots

        r2 = r1

import numpy as np
import math

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0 or not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(v - tau, 0.0, 1.0)
    r = S - float(np.sum(y))
    tol = 5e-13 * max(1, n)
    if r > tol:
        for i in range(n):
            if r <= tol:
                break
            d = min(r, 1.0 - y[i])
            y[i] += d
            r -= d
    elif r < -tol:
        for i in range(n):
            if r >= -tol:
                break
            d = min(-r, y[i])
            y[i] -= d
            r += d
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1.0 + 1e-12):
        return None
    return y

def check_feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(float(np.sum(h)) - 0.5 * n) < 1e-9 * max(1, n))

def score_and_corr(h, n):
    if not check_feasible(h, n):
        return (float('inf'), None)
    corr = np.correlate(h, 1.0 - h, mode='full')
    q = corr * (2.0 / n)
    sc = float(np.max(q))
    if not np.isfinite(sc):
        return (float('inf'), None)
    return (sc, q)

def lag_grad_col(h, k, n):
    dx = 2.0 / n
    L = 2 * n - 1
    s = n - 1 - np.arange(L)
    g = np.zeros(L)
    ks1 = k + s
    m1 = (ks1 >= 0) & (ks1 < n)
    g[m1] += (1.0 - h[ks1[m1]])
    ks2 = k - s
    m2 = (ks2 >= 0) & (ks2 < n)
    g[m2] -= h[ks2[m2]]
    return g * dx

def quad_roots(A, B, C):
    roots = []
    if abs(A) < 1e-18:
        if abs(B) > 1e-18:
            t = -C / B
            if np.isfinite(t):
                roots.append(t)
        return roots
    disc = B * B - 4.0 * A * C
    scale = max(1.0, abs(B * B), abs(4.0 * A * C))
    if disc < -1e-12 * scale:
        return roots
    if disc < 0.0:
        disc = 0.0
    sq = math.sqrt(disc)
    if B >= 0.0:
        q = -0.5 * (B + sq)
    else:
        q = -0.5 * (B - sq)
    if abs(q) > 1e-30:
        roots.append(C / q)
        roots.append(q / A)
    else:
        roots.append(-B / (2.0 * A))
    return [r for r in roots if np.isfinite(r)]

def build_affine_hull(lines):
    if not lines:
        return []
    best = {}
    for m, c in lines:
        if m not in best or c > best[m]:
            best[m] = c
    items = sorted(best.items())
    if len(items) == 1:
        return [(items[0][0], items[0][1], -np.inf)]
    hull = [list(items[0])]
    starts = [-np.inf]
    for m, c in items[1:]:
        if len(hull) == 0:
            hull.append([m, c])
            starts.append(-np.inf)
            continue
        x = (hull[-1][1] - c) / (m - hull[-1][0])
        while len(hull) > 1 and x <= starts[-1] + 1e-14:
            hull.pop()
            starts.pop()
            x = (hull[-1][1] - c) / (m - hull[-1][0])
        hull.append([m, c])
        starts.append(x)
    return [(h[0], h[1], st) for h, st in zip(hull, starts)]

def pair_line_min(h, q, cur_score, i, j, n):
    dx = 2.0 / n
    L = 2 * n - 1
