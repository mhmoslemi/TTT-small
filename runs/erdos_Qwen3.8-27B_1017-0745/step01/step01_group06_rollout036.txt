    d = z[:n]
    if abs(d.sum()) > 1e-4:
        return None, False
    for i in range(n):
        lo = max(-delta, -h[i])
        hi = min(delta, 1.0 - h[i])
        if d[i] < lo - 1e-4 or d[i] > hi + 1e-4:
            return None, False
    if np.min(tr_ineq(z, q, G)) < -1e-4:
        return None, False
    return d, True

def local_optimize(h0, n, dx, S, rng, max_iter, max_slqp):
    h = project_K(h0.copy(), n, S)
    delta = 0.12
    eta = 0.5
    cf, cr = 0, 0
    for _ in range(max_iter):
        q = lag_vals(h, dx)
        qm = q.max()
        if not np.isfinite(qm):
            break
        G = build_G(h, dx, n)
        d, ok = solve_tr(h, q, G, delta, eta, n, dx, S, max_slqp)
        if not ok or d is None:
            delta *= 0.5
            cf += 1
            if cf >= 4 or delta < 1e-5:
                break
            continue
        cf = 0
        ht = project_K(h + d, n, S)
        if not feasible(ht, n, S):
            delta *= 0.5
            cr += 1
            if cr >= 8 or delta < 1e-5:
                break
            continue
        da = ht - h
        if np.max(np.abs(da)) < 1e-8:
            break
        pred = qm - np.max(q + G @ da)
        act = qm - exact_score(ht, dx)
        if act > 1e-12 and (pred <= 1e-12 or (pred > 1e-12 and act / pred >= 0.1)):
            h = ht
            cr = 0
            if pred > 1e-12:
                r = act / pred
                if r < 0.25:
                    delta *= 0.5
                elif r > 0.75 and np.max(np.abs(da)) >= 0.8 * delta:
                    delta = min(0.5, 2.0 * delta)
                    eta = min(2.0, 1.5 * eta)
        else:
            delta *= 0.5
            eta = max(1e-4, eta * 0.5)
            cr += 1
            if cr >= 8 or delta < 1e-5:
                break
    return project_K(h, n, S)

def transfer_parent(hp, n0, n_new):
    S_new = n_new / 2.0
    y = np.zeros(n_new)
    for j in range(n_new):
        acc = 0.0
        for i in range(n0):
            lo = max(i / n0, j / n_new)
            hi = min((i + 1) / n0, (j + 1) / n_new)
            if hi > lo:
                acc += hp[i] * (hi - lo)
        y[j] = n_new * acc
    return project_K(y, n_new, S_new)

def make_starts(base, n, S, rng, has_parent):
    starts = [base.copy(), (1.0 - base).copy(), base[::-1].copy(), (1.0 - base[::-1]).copy()]
    starts.append(project_K(base + 0.08 * rng.standard_normal(n), n, S))
    starts.append(project_K(base + 0.25 * rng.standard_normal(n), n, S))
    starts.append(project_K(rng.uniform(0, 1, n), n, S))
    starts.append(project_K(rng.beta(0.5, 0.5, n), n, S))
    blk = np.zeros(n)
    k = int(np.floor(S))
    blk[:k] = 1.0
    if k < n:
        blk[k] = S - k
    starts.append(project_K(blk, n, S))
    j = np.arange(n)
    starts.append(project_K(0.5 + 0.45 * np.cos(2.0 * np.pi * j / n), n, S))
    chk = np.zeros(n)
    chk[0::2] = S / np.sum(0::2 < n) if n % 2 == 0 else 1.0
    chk[1::2] = 0.0
    starts.append(project_K(chk, n, S))
    if has_parent:
        starts.append(project_K(base[rng.permutation(n)], n, S))
    sel = []
    for s in starts:
        if not feasible(s, n, S):
            continue
        dup = False
        for x in sel:
            if np.max(np.abs(s - x)) <= 1e-6:
                dup = True
                break
        if not dup:
            sel.append(s)
    return sel

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    init = globals().get('initial_h_values', None)
    has_parent, hp, n0 = False, None, 0
    if init is not None:
        init = np.array(init, dtype=np.float64).ravel()
        if 2 <= len(init) <= 999 and np.all(np.isfinite(init)) and init.min() >= -1e-6 and init.max() <= 1.0 + 1e-6:
            n0 = len(init)
            if abs(init.sum() - n0 / 2.0) <= 1e-6:
                hp = np.clip(init.copy(), 0.0, 1.0)
                has_parent = True
    nd = kwargs.get('n_points', None)
    if not has_parent:
        n_res = [int(nd)] if isinstance(nd, (int, np.integer)) and 2 <= int(nd) <= 999 else [45]
    else:
        n_res = [n0]
        if n0 - 1 >= 2:
            n_res.append(n0 - 1)
        if n0 + 1 <= 999:
            n_res.append(n0 + 1)
        if isinstance(nd, (int, np.integer)) and 2 <= int(nd) <= 999 and int(nd) not in n_res:
            n_res.append(int(nd))
    seen, n_res2 = set(), []
    for x in n_res:
        if x not in seen:
            seen.add(x)
            n_res2.append(x)
    n_res = n_res2
    fb_h, fb_s, fb_n = None, float('inf'), None
    if has_parent:
        n_fb, dx_fb, S_fb = len(hp), 2.0 / len(hp), len(hp) / 2.0
        h_fb = project_K(hp.copy(), n_fb, S_fb)
        if feasible(h_fb, n_fb, S_fb):
            fb_h, fb_s, fb_n = h_fb.copy(), exact_score(h_fb, dx_fb), n_fb
    best_h, best_s, best_n = None, float('inf'), None
    for m in n_res:
        dx, S = 2.0 / m, m / 2.0
        if has_parent and m == len(hp):
            base = project_K(hp.copy(), m, S)
        elif has_parent:
            base = transfer_parent(hp, len(hp), m)
        else:
            base = np.full(m, 0.5)
        if not feasible(base, m, S):
            base = np.full(m, 0.5)
        sc = exact_score(base, dx)
        if sc < best_s:
            best_h, best_s, best_n = base.copy(), sc, m
        if m <= 96:
            n_st, mi, msl, rst = 8, 28, 50, True
        elif m <= 256:
            n_st, mi, msl, rst = 3, 12, 30, False
        else:
            n_st, mi, msl, rst = 1, 5, 15, False
        starts = make_starts(base, m, S, rng, has_parent)[:n_st]
        for s in starts:
            if not feasible(s, m, S):
                continue
            hopt = local_optimize(s, m, dx, S, rng, mi, msl)
            if feasible(hopt, m, S):
                sc2 = exact_score(hopt, dx)
                if sc2 < best_s - 1e-12:
                    best_h, best_s, best_n = hopt.copy(), sc2, m
        if rst:
            s2 = project_K(base + 0.18 * rng.standard_normal(m), m, S)
            if feasible(s2, m, S):
                hopt = local_optimize(s2, m, dx, S, rng, mi, msl)
                if feasible(hopt, m, S):
                    sc2 = exact_score(hopt, dx)
                    if sc2 < best_s - 1e-12:
                        best_h, best_s, best_n = hopt.copy(), sc2, m
    if best_h is not None:
        dx_f, S_f = 2.0 / best_n, best_n / 2.0
        best_h = project_K(best_h, best_n, S_f)
        if feasible(best_h, best_n, S_f):
            fs = exact_score(best_h, dx_f)
            if fb_h is not None and fb_s + 1e-12 < fs:
                return (fb_h, fb_s, fb_n)
            return (best_h, fs, best_n)
    if fb_h is not None:
        return (fb_h, fb_s, fb_n)
    n_f = 45
    h_f = np.full(n_f, 0.5)
    return (h_f, exact_score(h_f, 2.0 / n_f), n_f)
