    sigma_new = float(np.clip(sigma * math.exp(expo), 1e-4, 0.5))
    
    return mu_new, C_new, sigma_new, p_sigma, p_c, hs_arr, scores_arr, h_mean

import numpy as np
import math

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)) or np.max(np.abs(z)) > 1e100:
        return np.full(N, 0.5)
    zc = z - np.median(z)
    lo = zc.min() - 1.0
    hi = zc.max()
    for _ in range(100):
        mid = lo + (hi - lo) * 0.5
        if np.clip(zc - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(zc - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    for i in range(N):
        if r > 0:
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
        elif r < 0:
            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
        else:
            break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1 + 1e-10:
        return np.full(N, 0.5)
    if abs(h.sum() - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    corr = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(corr)):
        return np.inf
    return float(np.max(corr) * 2.0 / N)

def build_helmert(N):
    d = N - 1
    B = np.zeros((N, d))
    for k in range(d):
        a = math.sqrt(float((k + 1) * (k + 2)))
        B[:k + 1, k] = 1.0 / a
        B[k + 1, k] = -(k + 1) / a
    return B

def active_lag_grad(h, N):
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    s = k_star - (N - 1)
    grad = np.zeros(N)
    if s >= 0:
        L = N - s
        grad[:L] += (1.0 - h[s:s + L])
        grad[s:] -= h[:L]
    else:
        d = -s
        L = N - d
        grad[d:] += (1.0 - h[:L])
        grad[:L] -= h[d:]
    return grad

def polish_subgrad(h, N, m, n_iter=30):
    h = h.copy()
    best_sc = score_exact(h, N)
    alpha = 0.05
    for it in range(n_iter):
        g = active_lag_grad(h, N)
        g = g - g.mean()
        step = h - alpha * g
        h_new = project_box_sum(step, m)
        sc = score_exact(h_new, N)
        if sc < best_sc:
            h = h_new
            best_sc = sc
        else:
            alpha *= 0.8
            if alpha < 1e-5:
                break
    return h, best_sc

def cma_generate(mu, C, sigma, p_sigma, p_c, lam, rng, d, B, u, m, N,
                 mu_sel, weights, c_sigma, d_sigma, c_c, c1, c_mu, chi_d, gen):
    eigvals, eigvecs = np.linalg.eigh(C)
    ev_max = max(1.0, float(eigvals.max()))
    eigvals = np.clip(eigvals, 1e-12 * ev_max, 1e4)
    C_reg = (eigvecs * eigvals) @ eigvecs.T
    C_reg = 0.5 * (C_reg + C_reg.T)
    sqrt_D = np.sqrt(eigvals)
    A = eigvecs * sqrt_D
    C_inv_sqrt = (eigvecs * (1.0 / sqrt_D)) @ eigvecs.T
    hs_list = []
    xs_list = []
    sc_list = []
    for j in range(lam):
        z = rng.standard_normal(d)
        x_raw = mu + sigma * (A @ z)
        if not np.all(np.isfinite(x_raw)):
            h_cand = project_box_sum(u + B @ mu, m)
        else:
            h_cand = project_box_sum(u + B @ x_raw, m)
        x_rep = B.T @ (h_cand - u)
        hs_list.append(h_cand)
        xs_list.append(x_rep)
        sc_list.append(score_exact(h_cand, N))
    xs_arr = np.array(xs_list)
    hs_arr = np.stack(hs_list)
    scores_arr = np.array(sc_list)
    order = np.argsort(scores_arr, kind='stable')
    sel = order[:mu_sel]
    y_sel = (xs_arr[sel] - mu[None, :]) / sigma
    mu_new = np.sum(weights[:, None] * xs_arr[sel], axis=0)
    h_mean = project_box_sum(u + B @ mu_new, m)
    mu_new = B.T @ (h_mean - u)
    y_w = (mu_new - mu) / sigma
    mu_eff = 1.0 / np.sum(weights**2)
    p_sigma = (1.0 - c_sigma) * p_sigma + math.sqrt(c_sigma * (2.0 - c_sigma) * mu_eff) * (C_inv_sqrt @ y_w)
    t = gen + 1
    norm_ps = np.linalg.norm(p_sigma)
    bound = (1.4 + 2.0 / (d + 1)) * chi_d * math.sqrt(max(0.0, 1.0 - (1.0 - c_sigma)**(2 * t)))
    h_sig = 1.0 if norm_ps < bound else 0.0
    p_c = (1.0 - c_c) * p_c + h_sig * math.sqrt(c_c * (2.0 - c_c) * mu_eff) * y_w
    C_new = (1.0 - c1 - c_mu) * C_reg
    C_new += c1 * (np.outer(p_c, p_c) + (1.0 - h_sig) * c_c * (2.0 - c_c) * C_reg)
    for i in range(mu_sel):
        C_new += c_mu * weights[i] * np.outer(y_sel[i], y_sel[i])
    C_new = 0.5 * (C_new + C_new.T)
    expo = (c_sigma / d_sigma) * (norm_ps / chi_d - 1.0)
    expo = float(np.clip(expo, -1.0, 1.0))
    sigma_new = float(np.clip(sigma * math.exp(expo), 1e-4, 0.5))
    return mu_new, C_new, sigma_new, p_sigma, p_c, hs_arr, scores_arr, h_mean
