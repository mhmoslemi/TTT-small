            freqs.append(int(round((lo_f + hi_f) / 2.0)))
        else:
            pts = np.linspace(lo_f, hi_f, r).round().astype(int)
            for p in pts:
                if p not in freqs and p >= 1:
                    freqs.append(p)
    while len(freqs) < d:
        freqs.append(len(freqs) + 1)
    freqs = freqs[:d]
    idx = np.arange(n)
    cols = np.array([np.sqrt(2.0 / n) * np.cos(np.pi * j * (idx + 0.5) / n)
                     - (2.0 / n) * np.cos(np.pi * j * (idx + 0.5) / n).mean()
                     for j in freqs])
    Q, _ = np.linalg.qr(cols)
    for j in range(d):
        if Q[np.argmax(np.abs(Q[:, j])), j] < 0:
            Q[:, j] *= -1
    return Q[:, :d]

def decode_candidate(anchor, B, sigma, u, n):
    d = B.shape[1]
    raw = anchor + sigma * math.sqrt(3.0 * n / d) * (B @ u)
    if not np.all(np.isfinite(raw)):
        return None
    c, ok = project_capped(raw, n)
    return c if ok else None

def reflect_unit(x):
    t = (x + 1.0) % 4.0
    return np.where(t <= 2.0, -1.0 + t, 3.0 - t)

def evolve_stage(anchor, B, sigma, P, G, rng, n):
    d = B.shape[1]
    pop = np.zeros((P, d))
    for i in range(1, P):
        pop[i] = rng.uniform(-1.0, 1.0, d)
    scores = np.full(P, float('inf'))
    best_h = anchor.copy()
    best_s = exact_score(anchor, n)
    best_u = np.zeros(d)
    for i in range(P):
        hc = decode_candidate(anchor, B, sigma, pop[i], n)
        if hc is not None:
            s = exact_score(hc, n)
            scores[i] = s
            if s < best_s:
                best_s = s; best_h = hc.copy(); best_u = pop[i].copy()
    stag = 0
    restarted = False
    for gen in range(G):
        snap = pop.copy()
        ssnap = scores.copy()
        gen_best = best_s
        for i in range(P):
            others = list(range(P))
            others.pop(i)
            r1, r2, r3 = rng.choice(others, size=3, replace=False)
            F = rng.uniform(0.5, 0.9)
            v = snap[r1] + F * (snap[r2] - snap[r3])
            forced = int(rng.integers(0, d))
            mask = rng.random(d) < 0.8
            mask[forced] = True
            tu = np.where(mask, v, snap[i])
            tu = reflect_unit(tu)
            hc = decode_candidate(anchor, B, sigma, tu, n)
            if hc is not None:
                s = exact_score(hc, n)
                if s < ssnap[i] - 1e-12:
                    pop[i] = tu; scores[i] = s
                    if s < gen_best:
                        gen_best = s
                        best_s = s; best_h = hc.copy(); best_u = tu.copy()
                else:
                    pop[i] = snap[i]; scores[i] = ssnap[i]
            else:
                pop[i] = snap[i]; scores[i] = ssnap[i]
        if gen_best < best_s + 1e-12:
            stag = 0
        else:
            stag += 1
        if stag >= 3 and not restarted and gen < G - 1:
            restarted = True
            stag = 0
            pop[0] = best_u.copy(); scores[0] = best_s
            for i in range(1, P):
                pert = reflect_unit(best_u + 0.35 * rng.standard_normal(d))
                pop[i] = pert
                hc = decode_candidate(anchor, B, sigma, pert, n)
                scores[i] = exact_score(hc, n) if hc is not None else float('inf')
    return best_h, best_s

def local_refine(h, n, rng, iters=200):
    best_h = h.copy()
    best_s = exact_score(h, n)
    deltas = [0.01, 0.02, 0.04, 0.08]
    for _ in range(iters):
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        improved = False
        for dl in deltas:
            for sg in [1, -1]:
                d = sg * dl
                nh = best_h.copy()
                nh[i] += d; nh[j] -= d
                if nh[i] < -1e-12 or nh[i] > 1.0 + 1e-12:
                    continue
                if nh[j] < -1e-12 or nh[j] > 1.0 + 1e-12:
                    continue
                c, ok = project_capped(nh, n)
                if not ok:
                    continue
                s = exact_score(c, n)
                if s < best_s - 1e-10:
                    best_h = c; best_s = s; improved = True
                    break
            if improved:
                break
        if not improved and rng.random() < 0.1:
            i2 = int(rng.integers(0, n))
            j2 = int(rng.integers(0, n))
            if i2 != j2:
                for dl in [0.05, 0.1]:
                    for sg in [1, -1]:
                        d = sg * dl
                        nh = best_h.copy()
                        nh[i2] += d; nh[j2] -= d
                        if nh[i2] < -1e-12 or nh[i2] > 1.0 + 1e-12:
                            continue
                        if nh[j2] < -1e-12 or nh[j2] > 1.0 + 1e-12:
                            continue
                        c, ok = project_capped(nh, n)
                        if ok:
                            s = exact_score(c, n)
                            if s < best_s - 1e-10:
                                best_h = c; best_s = s
    return best_h, best_s

def heuristic_starts(n, rng):
    starts = []
    idx = np.arange(n)
    p5 = np.tile(np.array([1.0, 1.0, 0.5, 0.0, 0.0]), n // 5 + 1)[:n]
    starts.append(p5)
    starts.append(1.0 - p5)
    p3 = np.tile(np.array([1.0, 0.5, 0.0]), n // 3 + 1)[:n]
    starts.append(p3)
    starts.append(1.0 - p3)
    starts.append(rng.uniform(0, 1, n))
    starts.append(0.5 + 0.4 * np.sin(2 * np.pi * idx / n * 5))
    starts.append(0.5 + 0.3 * np.cos(2 * np.pi * idx / n * 2))
    return starts

def run(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = globals().get('initial_h_values', None)
    n = 45
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.ndim == 1 and 2 <= len(p) <= 999 and np.all(np.isfinite(p)) and np.max(np.abs(p)) < 1e6:
            n = len(p)
            c, ok = project_capped(p.copy(), n)
            fallback = c if ok else np.full(n, 0.5)
        else:
            fallback = np.full(n, 0.5)
    else:
        np_kw = kwargs.get('n_points', None)
        if isinstance(np_kw, int) and not isinstance(np_kw, bool) and 2 <= np_kw <= 999:
            n = np_kw
        fallback = np.full(n, 0.5)
    best_h = fallback.copy()
    best_s = exact_score(best_h, n)
    B = make_basis(n)
    d = B.shape[1]
    # Heuristic starts
    for hs in heuristic_starts(n, rng):
        c, ok = project_capped(hs, n)
        if ok:
            s = exact_score(c, n)
            if s < best_s:
                best_h, best_s = c, s
    # Alt anchor from full-dim perturbation
    alt_anchor = fallback.copy()
    try:
        z = rng.standard_normal(n)
        z -= z.mean()
        rz = np.sqrt(np.mean(z * z))
        if rz > 1e-10:
            c, ok = project_capped(fallback + 0.12 * z / rz, n)
            if ok:
                s = exact_score(c, n)
                if s < best_s:
                    best_h, best_s = c, s
                alt_anchor = c
    except Exception:
        pass
    # DE stages
    if n <= 100:
        P = max(14, min(52, 2 * d + 10)); G = 20
    elif n <= 400:
        P = max(10, min(32, d + 8)); G = 6
    else:
        P = max(8, min(16, d)); G = 3
    stages = [(fallback, 0.10), (alt_anchor, 0.18), (best_h, 0.05)]
    for anchor, sigma in stages:
        try:
            sh, ss = evolve_stage(anchor, B, sigma, P, G, rng, n)
            if check_feasible(sh, n):
                ss2 = exact_score(sh, n)
                if ss2 < best_s - 1e-12:
                    best_h, best_s = sh, ss2
        except Exception:
            pass
    # Local refinement
    try:
        rh, rs = local_refine(best_h, n, rng, iters=200)
        if rs < best_s - 1e-12:
            best_h, best_s = rh, rs
    except Exception:
        pass
    # Second DE stage from refined best with smaller sigma
    try:
        sh, ss = evolve_stage(best_h, B, 0.03, max(P // 2, 8), 10, rng, n)
        if check_feasible(sh, n):
            ss2 = exact_score(sh, n)
            if ss2 < best_s - 1e-12:
                best_h, best_s = sh, ss2
        rh, rs = local_refine(best_h, n, rng, iters=100)
        if rs < best_s - 1e-12:
            best_h, best_s = rh, rs
    except Exception:
        pass
    # Finalize
    final_h = best_h.copy()
    c, ok = project_capped(final_h, n)
    if ok:
        final_h = c
    if not check_feasible(final_h, n):
        final_h = fallback.copy()
    final_s = exact_score(final_h, n)
    fb_s = exact_score(fallback, n)
    if fb_s < final_s:
        final_h, final_s = fallback, fb_s
    final_h = np.clip(final_h, 0.0, 1.0).astype(np.float64)
    c2, ok2 = project_capped(final_h, n)
    if ok2:
        final_h = c2
        final_s = exact_score(final_h, n)
    if not check_feasible(final_h, n):
        final_h = fallback.astype(np.float64)
        final_s = exact_score(final_h, n)
    return final_h, float(final_s), int(n)
