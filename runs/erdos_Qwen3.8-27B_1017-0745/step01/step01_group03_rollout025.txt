        if np.max(np.abs(trial - pop[i])) < 1e-14:
            continue
        sc = exact_score(trial, n)
        if sc < scores[i] - 1e-14:
            new_pop[i] = trial
            scores[i] = sc
            new_F[i] = fi
            new_CR[i] = cri
    return new_pop, scores, new_F, new_CR

def restart_members(pop, scores, best_h, P, n, S, R, rng):
    worst = np.argsort(scores)[::-1]
    sigmas = [0.03, 0.10, 0.25]
    cnt = 0
    for idx in worst:
        if cnt >= R:
            break
        if idx == 0:
            continue
        if cnt % 4 == 3:
            u = rng.uniform(0, 1, n)
        else:
            u = best_h + sigmas[cnt % 3] * rng.standard_normal(n)
        c = project_box_sum(u, S)
        if c is not None and feasible(c, n, S):
            pop[idx] = c
            scores[idx] = exact_score(c, n)
        cnt += 1
    return pop, scores

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent, n = get_parent()
    S = n / 2.0
    if parent is None:
        parent = project_box_sum(np.full(n, 0.5), S)
    if parent is None:
        parent = np.full(n, 0.5)
        for i in range(n):
            if np.sum(parent) < S:
                parent[i] = min(1.0, parent[i] + (S - np.sum(parent)) / max(1, n - i))
        parent = project_box_sum(parent, S)
        if parent is None:
            parent = np.full(n, 0.5)
    fallback = parent.copy()
    fb_score = exact_score(fallback, n)

    pop = build_population(fallback, n, S, rng)
    P = len(pop)
    scores = np.array([exact_score(pop[i], n) for i in range(P)], dtype=np.float64)
    best_h = fallback.copy()
    best_score = fb_score
    for i in range(P):
        if scores[i] < best_score - 1e-14:
            best_score = scores[i]
            best_h = pop[i].copy()

    F = np.full(P, 0.5)
    CR = np.full(P, 0.9)
    R = max(1, P // 4)
    E = max(2 * P + 3 * R, min(5000, int(math.ceil(8e7 / (n * n)))))
    G = max(1, int((E - P - 3 * R) / P))

    restarts = 0
    no_improve = 0
    for g in range(G):
        prev_best = best_score
        pop, scores, F, CR = de_step(pop, scores, P, n, S, F, CR, g, G, rng)
        for i in range(P):
            if scores[i] < best_score - 1e-14:
                best_score = scores[i]
                best_h = pop[i].copy()
        D = np.sqrt(np.mean(np.var(pop, axis=0)))
        if best_score > prev_best - 1e-12:
            no_improve += 1
        else:
            no_improve = 0
        if (no_improve >= 12 or D < 1e-3) and restarts < 3:
            pop, scores = restart_members(pop, scores, best_h, P, n, S, R, rng)
            for i in range(P):
                if scores[i] < best_score - 1e-14:
                    best_score = scores[i]
                    best_h = pop[i].copy()
            restarts += 1
            no_improve = 0
        if restarts >= 3 and no_improve >= 20 and D < 1e-5:
            break

    # local polish on best via small random perturbations + re-score
    for _ in range(30):
        sig = 0.01 + 0.02 * _ / 30.0
        u = best_h + sig * rng.standard_normal(n)
        c = project_box_sum(u, S)
        if c is not None and feasible(c, n, S):
            sc = exact_score(c, n)
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = c

    final_h = project_box_sum(best_h, S)
    if final_h is not None and feasible(final_h, n, S):
        fs = exact_score(final_h, n)
        if fs < fb_score:
            return (np.array(final_h, dtype=np.float64), fs, n)
    return (np.array(fallback, dtype=np.float64), fb_score, n)
