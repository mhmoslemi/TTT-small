Also, I need to double-check the McCormick constraints. For a bilinear term z = v[i]*v[j] on the box [l_i, u_i] × [l_j, u_j]:

The four McCormick inequalities are:
1. z ≥ l_i * v_j + l_j * v_i - l_i * l_j
2. z ≥ u_i * v_j + u_j * v_i - u_i * u_j
3. z ≤ u_i * v_j + l_j * v_i - u_i * l_j
4. z ≤ l_i * v_j + u_j * v_i - l_i * u_j

For i = j, these become:
1. z ≥ 2*l_i * v_i - l_i^2
2. z ≥ 2*u_i * v_i - u_i^2
3. z ≤ (u_i + l_i) * v_i - u_i * l_i
4. z ≤ (l_i + u_i) * v_i - l_i * u_i

Note that 3 and 4 are actually the same when i=j! Both give z ≤ (u_i+l_i)*v_i - u_i*l_i. So for the diagonal, we only need 3 distinct constraints (not 4). But it's fine to include the duplicate; it won't hurt.

def pidx(i, j, n):
    return i * n - i * (i - 1) // 2 + (j - i)

def solve_lp(center, r, n, S):
    l = np.maximum(0.0, center - r)
    u = np.minimum(1.0, center + r)
    P = n * (n + 1) // 2
    nv = n + P + 1
    ti = n + P
    rl, cl, vl, rh = [], [], [], []
    for i in range(n):
        for j in range(i, n):
            zi = n + pidx(i, j, n)
            li, ui, lj, uj = l[i], u[i], l[j], u[j]
            if i == j:
                coefs = [(-1.0, -2.0*li, -li*li), (-1.0, -2.0*ui, -ui*ui),
                         (1.0, -(ui+li), -ui*li), (1.0, -(li+ui), -li*ui)]
            else:
                coefs = [(-1.0, -li, -lj, -li*lj), (-1.0, -ui, -uj, -ui*uj),
                         (1.0, -ui, -lj, -ui*lj), (1.0, -li, -uj, -li*uj)]
            for cf in coefs:
                ri = len(rh)
                if i == j:
                    _, vc, rhs = cf
                    rl.append(ri); cl.append(zi); vl.append(1.0)
                    rl.append(ri); cl.append(i); vl.append(vc)
                else:
                    _, vcj, vci, rhs = cf
                    rl.append(ri); cl.append(zi); vl.append(1.0)
                    rl.append(ri); cl.append(j); vl.append(vcj)
                    rl.append(ri); cl.append(i); vl.append(vci)
                rh.append(rhs)
    for k in range(-(n-1), n):
        ri = len(rh)
        i_lo, i_hi = max(0, k), min(n-1, n-1+k)
        for i in range(i_lo, i_hi + 1):
            j = i - k
            rl.append(ri); cl.append(i); vl.append(1.0)
            a, b = min(i, j), max(i, j)
            rl.append(ri); cl.append(n + pidx(a, b, n)); vl.append(-1.0)
        rl.append(ri); cl.append(ti); vl.append(-1.0)
        rh.append(0.0)
    n_ineq = len(rh)
    c = np.zeros(nv)
    c[ti] = 1.0
    bounds = [(l[i], u[i]) for i in range(n)]
    for i in range(n):
        for j in range(i, n):
            bounds.append((l[i]*l[j], u[i]*u[j]))
    bounds.append((0.0, float(n)))
    A_ub = coo_matrix((vl, (rl, cl)), shape=(n_ineq, nv)).tocsr()
    b_ub = np.array(rh)
    A_eq = coo_matrix((np.ones(n), (np.zeros(n, dtype=int), np.arange(n))), shape=(1, nv)).tocsr()
    b_eq = np.array([S])
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    if res.status != 0 or res.x is None or len(res.x) != nv:
        return None
    if not np.all(np.isfinite(res.x)):
        return None
    return np.clip(res.x[:n], 0.0, 1.0), float(res.x[ti])

def line_minimax(x, y, n, S):
    d = y - x
    n_lags = 2 * n - 1
    A = np.zeros(n_lags)
    B = np.zeros(n_lags)
    D = np.zeros(n_lags)
    for k in range(-(n-1), n):
        idx = k + n - 1
        i_lo, i_hi = max(0, k), min(n-1, n-1+k)
        for i in range(i_lo, i_hi + 1):
            j = i - k
            A[idx] += -d[i] * d[j]
            B[idx] += d[i] * (1.0 - x[j]) - x[i] * d[j]
            D[idx] += x[i] * (1.0 - x[j])
    cands = {0.0, 1.0}
    for idx in range(n_lags):
        if abs(A[idx]) > 1e-15:
            s = -B[idx] / (2.0 * A[idx])
            if np.isfinite(s) and -1e-8 <= s <= 1.0 + 1e-8:
                cands.add(round(float(np.clip(s, 0.0, 1.0)), 10))
    for p in range(n_lags):
        for q in range(p + 1, n_lags):
            da, db, dd = A[p]-A[q], B[p]-B[q], D[p]-D[q]
            if abs(da) < 1e-14:
                if abs(db) > 1e-14:
                    r = -dd / db
                    if np.isfinite(r) and -1e-8 <= r <= 1.0 + 1e-8:
                        cands.add(round(float(np.clip(r, 0.0, 1.0)), 10))
            else:
                disc = db*db - 4.0*da*dd
                if disc >= -1e-10:
                    sq = math.sqrt(max(0.0, disc))
                    for r in [(-db+sq)/(2.0*da), (-db-sq)/(2.0*da)]:
                        if np.isfinite(r) and -1e-8 <= r <= 1.0 + 1e-8:
                            cands.add(round(float(np.clip(r, 0.0, 1.0)), 10))
    alg = np.array(sorted(cands), dtype=np.float64)
    qs = np.zeros(len(alg))
    for ci in range(len(alg)):
        a = alg[ci]
        qs[ci] = float(np.max(A*a*a + B*a + D))
    order = np.argsort(qs)
    top = set(order[:25].tolist())
    for ei in range(len(alg)):
        if alg[ei] < 1e-6 or alg[ei] > 1.0 - 1e-6:
            top.add(ei)
    best_sc = float('inf')
    best_h = None
    for ti2 in sorted(top):
        h_try = project_box_sum(x + alg[ti2] * d, S)
        if h_try is None or not okhttp(h_try, n):
            continue
        sc = exact_score(h_try, n)
        if sc < best_sc:
            best_sc = sc
            best_h = h_try
    if best_h is None:
        best_h = project_box_sum(x, S)
        best_sc = exact_score(best_h, n) if best_h is not None else float('inf')
    return best_h, best_sc

def local_lp_search(start, n, S, max_iter=8):
    cur = project_box_sum(start, S)
    if cur is None:
        return start
    cur_sc = exact_score(cur, n)
    r = 0.20
    fails = 0
    for _ in range(max_iter):
        if r < 1e-3 and fails >= 2:
            break
        res = solve_lp(cur, r, n, S)
        if res is None:
            r *= 0.5
            fails += 1
            continue
        h_lp, _ = res
        h_proj = project_box_sum(h_lp, S)
        if h_proj is None or not okhttp(h_proj, n):
            r *= 0.5
            fails += 1
            continue
        if float(np.max(np.abs(h_proj - cur))) < 1e-12:
            r *= 0.5
            fails += 1
            continue
        h_new, sc_new = line_minimax(cur, h_proj, n, S)
        if h_new is not None and sc_new < cur_sc - 1e-12:
            cur = h_new
            cur_sc = sc_new
            fails = 0
            r = min(0.25, 1.2 * r)
        else:
            r *= 0.5
            fails += 1
    return cur

def make_starts(parent, n, S, rng):
    starts = []
    if parent is not None:
        starts.append(parent.copy())
        starts.append(parent[::-1].copy())
        starts.append(1.0 - parent)
        starts.append(np.roll(parent, n // 3).copy())
        for sig in [0.04, 0.12, 0.20]:
            p = parent + sig * rng.standard_normal(n)
            c = project_box_sum(p, S)
            if c is not None:
                starts.append(c)
        freq = int(rng.integers(1, max(2, n // 2 + 1)))
        ph = rng.uniform(0, 2 * np.pi)
        harm = 0.08 * np.cos(2.0 * np.pi * freq * np.arange(n) / n + ph)
        harm -= np.mean(harm)
        rms = np.sqrt(np.mean(harm**2))
        if rms > 1e-12:
            harm = harm / rms * 0.08
            c = project_box_sum(parent + harm, S)
            if c is not None:
                starts.append(c)
        for _ in range(3):
            c = project_box_sum(rng.random(n), S)
            if c is not None:
                starts.append(c)
    else:
        starts.append(np.full(n, 0.5))
        for _ in range(5):
            c = project_box_sum(rng.random(n), S)
            if c is not None:
                starts.append(c)
        for m in range(1, min(3, n // 2) + 1):
            ph = rng.uniform(0, 2 * np.pi)
            c = project_box_sum(0.5 + 0.35 * np.cos(2.0 * np.pi * m * np.arange(n) / n + ph), S)
            if c is not None:
                starts.append(c)
    dedup = []
    for s in starts:
        if s is None:
            continue
        dup = False
        for d in dedup:
            if float(np.max(np.abs(s - d))) < 1e-10:
                dup = True
                break
        if not dup:
            dedup.append(s)
    return dedup
