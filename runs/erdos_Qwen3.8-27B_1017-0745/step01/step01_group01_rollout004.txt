    add_start(starts, 0.5 + 0.35 * np.sin(2 * np.pi * idx / n) + 0.2 * np.cos(4 * np.pi * idx / n), n)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    add_start(starts, blk, n)
    if n <= 64:
        ns_d, mi_d, pw_d, ft_d = 8, 250, 1, 1e-10
    elif n <= 128:
        ns_d, mi_d, pw_d, ft_d = 5, 160, 1, 1e-10
    elif n <= 256:
        ns_d, mi_d, pw_d, ft_d = 3, 80, 0, 1e-10
    else:
        ns_d, mi_d, pw_d, ft_d = 3, 0, 0, 1e-10
    try:
        ns = max(1, min(int(kwargs.get('starts', ns_d)), 10))
    except Exception:
        ns = ns_d
    try:
        mi = max(0, min(int(kwargs.get('max_iter', mi_d)), 300))
    except Exception:
        mi = mi_d
    try:
        ft = max(1e-12, min(float(kwargs.get('ftol', ft_d)), 1e-6))
    except Exception:
        ft = ft_d
    try:
        pw = max(0, min(int(kwargs.get('polish_sweeps', pw_d)), 2))
    except Exception:
        pw = pw_d
    starts = starts[:ns]
    best = starts[0].copy()
    best_sc = exact_score(best, n)
    for st in starts[1:]:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_sc - 1e-12:
            best, best_sc = st.copy(), ss
    if n <= 256 and mi > 0:
        bnds = [(0.0, 1.0)] * (n + 1)
        cons = [
            {'type': 'eq', 'fun': slsqp_eq_fun, 'jac': slsqp_eq_jac},
            {'type': 'ineq', 'fun': slsqp_ineq_fun, 'jac': slsqp_ineq_jac},
        ]
        for st in starts:
            sc0 = exact_score(st, n)
            if not np.isfinite(sc0):
                continue
            t0 = min(1.0, sc0 + 1e-8)
            z0 = np.concatenate([st, np.array([t0])])
            try:
                res = minimize(slsqp_obj, z0, method='SLSQP', jac=slsqp_obj_jac,
                               bounds=bnds, constraints=cons, args=(n,),
                               options={'ftol': ft, 'maxiter': mi, 'disp': False})
            except Exception:
                continue
            if res is None or not hasattr(res, 'x') or res.x is None:
                continue
            y = np.asarray(res.x[:n], dtype=np.float64)
            if y.shape != (n,) or not np.all(np.isfinite(y)):
                continue
            yp = project_capped(y, n)
            if yp is None:
                continue
            sy = exact_score(yp, n)
            if np.isfinite(sy) and sy < best_sc - 1e-12:
                best, best_sc = yp.copy(), sy
    if n <= 64 and pw > 0:
        px, ps = polish_pair(best.copy(), n, pw)
        if np.isfinite(ps) and ps < best_sc - 1e-12:
            best, best_sc = px, ps
    fin = project_capped(best, n)
    if fin is None:
        fin = uniform.copy()
    fs = exact_score(fin, n)
    if not np.isfinite(fs) or fs > best_sc + 1e-12:
        fin2 = project_capped(best, n)
        if fin2 is not None:
            fs2 = exact_score(fin2, n)
            if np.isfinite(fs2):
                fin, fs = fin2, fs2
    if not np.isfinite(fs):
        fin = uniform.copy()
        fs = exact_score(fin, n)
    return fin.copy(), float(fs), int(n)
