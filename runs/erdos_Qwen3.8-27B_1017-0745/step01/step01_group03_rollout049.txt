def compute_qg(h):

            if r<=1e-15: break
            a=min(r,1.0-y[i]); y[i]+=a; r-=a
    elif r<0:
        for i in range(n):
            if r>=-1e-15: break
            a=min(-r,y[i]); y[i]-=a; r+=a
    if abs(S-y.sum())>1e-10*max(1.0,n) or np.any(y<-1e-12) or np.any(y>1.0+1e-12): return None
    y=np.clip(y,0.0,1.0); r=S-y.sum()
    if r>0:
        i=int(np.argmax(1.0-y)); y[i]+=min(r,1.0-y[i])
    elif r<0:
        i=int(np.argmin(y)); y[i]+=max(r,-y[i])
    if abs(S-y.sum())>1e-9*max(1.0,n): return None
    return y
def feas(h,n):
    if h is None or h.shape!=(n,) or not np.all(np.isfinite(h)): return False
    if np.any(h<-1e-12) or np.any(h>1.0+1e-12): return False
    return abs(float(h.sum())-0.5*float(n))<=1e-10*max(1.0,float(n))
def score(h,n):
    if not feas(h,n): return np.inf
    h=np.asarray(h,dtype=np.float64)
    c=np.correlate(h,1.0-h,mode='full')
    v=float(np.max(c)*2.0/float(n))
    return v if np.isfinite(v) else np.inf
def get_lags(n):
    if n not in _LAGS:
        L=[]
        for k in range(1-n,n):
            i=np.arange(max(0,k),min(n-1,n-1+k)+1)
            L.append((i,i-k))
        _LAGS[n]=L
    return _LAGS[n]
def qg(h):
    n=len(h); lags=get_lags(n); L=len(lags)
    Q=np.empty(L); G=np.zeros((L,n))
    for row,(i,j) in enumerate(lags):
        hi=h[i]; hj=h[j]; Q[row]=np.sum(hi*(1.0-hj))
        G[row,i]+=1.0-hj; G[row,j]-=hi
    return Q,G
def lcoeff(x,d):
    n=len(x); lags=get_lags(n); L=len(lags)
    A=np.zeros(L); B=np.zeros(L); D=np.zeros(L)
    for row,(i,j) in enumerate(lags):
        hi=x[i]; hj=x[j]; di=d[i]; dj=d[j]
        D[row]=np.sum(hi*(1.0-hj)); A[row]=-np.sum(di*dj); B[row]=np.sum(di*(1.0-hj)-hi*dj)
    return A,B,D
def ada(al,a):
    if np.isfinite(a) and -1e-9<=a<=1.0+1e-9:
        if a<0: a=0.0
        elif a>1: a=1.0
        al.append(float(a))
def line_best(x,y,n,S):
    if not feas(x,n) or not feas(y,n): return x, score(x,n)
    d=y-x
    if np.max(np.abs(d))<1e-12: return x, score(x,n)
    A,B,D=lcoeff(x,d); L=len(A); al=[0.0,1.0,0.5]; al.extend([0.25,0.75])
    for row in range(L):
        if abs(A[row])>1e-14: ada(al, -B[row]/(2.0*A[row]))
    for p in range(L):
        ap,bp,cp=A[p],B[p],D[p]
        for q in range(p+1,L):
            aa=ap-A[q]; bb=bp-B[q]; cc=cp-D[q]; sc=max(abs(aa),abs(bb),abs(cc),1e-300)
            if abs(aa)<=1e-13*sc:
                if abs(bb)>1e-14*sc: ada(al, -cc/bb)
            else:
                disc=bb*bb-4.0*aa*cc
                if disc<-1e-10*sc*sc: continue
                if disc<0: disc=0
                sq=np.sqrt(disc); den=2.0*aa
                ada(al, (-bb-sq)/den); ada(al, (-bb+sq)/den)
    an=np.asarray(al,float); an=an[(an>=-1e-9)&(an<=1.0+1e-9)]
    if an.size==0: return x, score(x,n)
    np.clip(an,0.0,1.0,an); an.sort()
    uu=[]; pv=None
    for a in an:
        if pv is None or abs(a-pv)>1e-9: uu.append(float(a)); pv=a
    qmax=np.empty(len(uu))
    for st in range(0,len(uu),2048):
        aa=np.asarray(uu[st:st+2048]); qv=D+B*aa[:,None]+A*(aa[:,None]*aa[:,None]); qmax[st:st+2048]=qv.max(axis=1)
    test=[0.0,0.5,1.0]
    for i in np.argsort(qmax)[:20]: test.append(uu[int(i)])
    best=x; bs=score(x,n); seen=[]
    for a in test:
        if a<0.0 or a>1.0: continue
        seen_dup=False
        for s in seen:
            if abs(a-s)<1e-9: seen_dup=True; break
        if seen_dup: continue
        seen.append(a); c=proj(x+a*d,S)
        if c is not None and feas(c,n):
            sc=score(c,n)
            if sc<bs: bs=sc; best=c
    return best,bs
def aco(A,r,c,v):
    if v!=0.0: A[r,c]=A[r,c]+v
def lp_dir(x,n,r,S):
    if n>64: return None
    l=np.maximum(0.0,x-r); u=np.minimum(1.0,x+r)
    P=n*(n+1)//2; L=2*n-1; m=4*P+L; nv=n+P+1; tv=n+P
    A=lil_matrix((m,nv)); b=np.zeros(m); zlb=np.empty(P); zub=np.empty(P); row=0
    for i in range(n):
        off=i*n-i*(i-1)//2
        for j in range(i,n):
            p=off+(j-i); li=l[i]; lj=l[j]; ui=u[i]; uj=u[j]; pc=n+p
            zlb[p]=li*lj; zub[p]=ui*uj
            A[row,pc]=-1.0; aco(A,row,i,-lj); aco(A,row,j,-li); b[row]=-li*lj; row+=1
            A[row,pc]=-1.0; aco(A,row,i,-uj); aco(A,row,j,-ui); b[row]=-ui*uj; row+=1
            A[row,pc]=1.0; aco(A,row,i,-lj); aco(A,row,j,-ui); b[row]=-ui*lj; row+=1
            A[row,pc]=1.0; aco(A,row,i,-uj); aco(A,row,j,-li); b[row]=-li*uj; row+=1
    for k in range(1-n,n):
        row=4*P+k+n-1; i_lo=max(0,k); i_hi=min(n-1,n-1+k)
        for ii in range(i_lo,i_hi+1):
            jj=ii-k; a=ii if ii<jj else jj; cidx=jj if ii<jj else ii
            p=a*n-a*(a-1)//2+(cidx-a)
            A[row,ii]=1.0; A[row,n+p]=-1.0
        A[row,tv]=-1.0
    c=np.zeros(nv); c[tv]=1.0
    bounds=list(zip(l,u))+list(zip(zlb,zub))+[(0.0,float(n))]
    Aeq=np.zeros((1,nv)); Aeq[0,:n]=1.0
    try:
        res=linprog(c,A_ub=A.tocsr(),b_ub=b,A_eq=Aeq,b_eq=np.array([S]),bounds=bounds,method='highs')
    except Exception:
        return None
    if res is None or getattr(res,'status',-1) not in (0,1) or getattr(res,'x',None) is None: return None
    v=res.x[:n]
    if not np.all(np.isfinite(v)): return None
    return np.clip(v,np.maximum(0.0,l-1e-6),np.minimum(1.0,u+1e-6))
def mm_obj(z): return float(z[-1])
def mm_jac(z):
    g=np.zeros(len(z)); g[-1]=1.0; return g
def mm_ineq(z):
    n=len(z)-1; Q,_=qg(z[:n]); return z[-1]-Q*(2.0/n)
def mm_ineq_jac(z):
    n=len(z)-1; _,G=qg(z[:n]); L=len(G); J=np.zeros((L,n+1)); J[:,:n]=-(2.0/n)*G; J[:,-1]=1.0; return J
def mm_eq(z):
    n=len(z)-1; return np.array([np.sum(z[:n])-0.5*float(n)])
def mm_eq_jac(z):
    n=len(z)-1; J=np.zeros((1,n+1)); J[0,:n]=1.0; return J
def mm_refine(h0,n,S,bounds,zcons,maxit):
    base=h0.copy(); best=score(h0,n)
    Q,_=qg(h0); t0=min(1.0,max(0.0,float(np.max(Q)*(2.0/n))+1e-5))
    z0=np.concatenate([h0,np.array([t0])])
    try:
        res=minimize(mm_obj,z0,jac=mm_jac,method='SLSQP',bounds=bounds+[(0.0,1.0)],constraints=zcons,options={'maxiter':int(maxit),'ftol':1e-12,'disp':0})
        if res.x is not None and len(res.x)>=n+1:
            c=proj(res.x[:n],S)
            if c is not None and feas(c,n):
                sc=score(c,n)
                if sc<best: best=sc; base=c.copy()
    except Exception:
        pass
    return base,best
def first(pair): return pair[0]
def scored_starts(starts,n):

def lp_search(h0,n,S,radii):
    h=proj(h0,S)
