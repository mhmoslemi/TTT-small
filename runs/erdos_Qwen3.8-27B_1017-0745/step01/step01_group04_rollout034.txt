import numpy as np

def project_box_sum(z, m):
    N = len(z)
    if N == 0 or not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo = float(z.min()) - 1.0
    hi = float(z.max())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    for _ in range(2):
        if r > 1e-14:
            for i in range(N):
                adj = min(r, 1.0 - h[i])
                h[i] += adj
                r -= adj
                if r <= 1e-14:
                    break
        elif r < -1e-14:
            for i in range(N):
                adj = min(-r, h[i])
                h[i] -= adj
                r += adj
                if r >= -1e-14:
                    break
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(h.sum() - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return h

def exact_score(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    try:
        corr = np.correlate(h, 1.0 - h, mode='full')
        if not np.all(np.isfinite(corr)):
            return np.inf
        return float(np.max(corr) * 2.0 / N)
    except Exception:
        return np.inf

def is_feasible(h, N, m):

        return False
    if abs(h.sum() - m) > 1e-10 * max(1, N):
        return False
    return True
