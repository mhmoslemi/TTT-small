import numpy as np
from scipy.optimize import minimize

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n)

def project_capped(z, n):
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for i in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
        else:
            a = min(-r, h[i])
            h[i] -= a
            r += a
    if abs(r) > 1e-10 * max(1, n) or not np.all(np.isfinite(h)):
        return None
    return np.clip(h, 0.0, 1.0)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def smooth_fg(h, tau):
    n = len(h)
    dx = 2.0 / n
    q = np.zeros(2 * n - 1)
    for d in range(-(n - 1), n):
        i = d + n - 1
        a = max(d, 0)
        b = max(-d, 0)
        m = n - abs(d)
        q[i] = dx * np.dot(h[a:a + m], 1.0 - h[b:b + m])
    if not np.all(np.isfinite(q)):
        return 1e10, np.zeros(n)
    qm = q.max()
    e = np.exp((q - qm) / tau)
    se = e.sum()
    w = e / se
    F = qm + tau * np.log(se)
    g = np.zeros(n)
    for d in range(-(n - 1), n):
        i = d + n - 1
        a = max(d, 0)
        b = max(-d, 0)
        m = n - abs(d)
        wd = w[i] * dx
        if wd > 1e-15:
            g[a:a + m] += wd * (1.0 - h[b:b + m])
            g[b:b + m] -= wd * h[a:a + m]
    if not np.all(np.isfinite(g)) or not np.isfinite(F):
        return 1e10, np.zeros(n)
    return F, g

def sobj(x, tau):
    return smooth_fg(x, tau)[0]

def sjac(x, tau):
    return smooth_fg(x, tau)[1]

def meq_fun(x, n):
    return np.array([np.sum(x) - n / 2.0])

def meq_jac(x, n):
    return np.ones((1, len(x)))

def build_starts(fb, n, rng):

    cands = [fb.copy(),
             project_capped(fb[::-1].copy(), n),
             project_capped(1.0 - fb[::-1].copy(), n),
