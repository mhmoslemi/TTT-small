            sc = exact_score(hb, n)
            if sc is not None: return hb, sc
    return None, None

def run(seed=42, **kwargs):
    if seed is None: seed = 42
    try: seed = int(seed) % (2**32)
    except (TypeError, ValueError): seed = 42
    rng = np.random.default_rng(seed)
    n = None; parent = None
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        try:
            iv_arr = np.asarray(iv, dtype=np.float64)
            if iv_arr.ndim == 1 and 8 <= iv_arr.size <= 999:
                n = iv_arr.size
                if np.all(np.isfinite(iv_arr)) and np.max(np.abs(iv_arr)) <= 1e12:
                    p = proj_cs(iv_arr, n / 2.0)
                    if p is not None: parent = p
        except Exception: pass
    if n is None:
        try: n = int(kwargs.get('n_points', 80))
        except (TypeError, ValueError): n = 80
        if n < 8 or n > 999: n = 80
    S = n / 2.0
    h_const = np.full(n, 0.5)
    best_h = h_const.copy()
    best_sc = exact_score(h_const, n)
    if best_sc is None: best_sc = 0.5
    if parent is not None:
        ps = exact_score(parent, n)
        if ps is not None and ps < best_sc:
            best_sc = ps; best_h = parent.copy()
    Dcap = 40 if n <= 128 else (32 if n <= 256 else (20 if n <= 512 else 12))
    D = min(n - 1, Dcap)
    B = build_dct(n, D)
    if B is None: return best_h, best_sc, n
    anchors = []
    if parent is not None:
        anchors.append(parent.copy())
        pp = proj_cs(parent + rng.normal(0, 0.08, n), S)
        if pp is not None: anchors.append(pp)
        anchors.append(parent[::-1].copy())
        pc = proj_cs(1.0 - parent, S)
        if pc is not None: anchors.append(pc)
    else:
        anchors.append(h_const.copy())
        hr = proj_cs(h_const + B @ rng.normal(0, 0.12, D), S)
        if hr is not None: anchors.append(hr)
        perm = rng.permutation(n); bv = np.zeros(n)
        bv[perm[:n // 2]] = 1.0
        if n % 2 == 1: bv[perm[n // 2]] = 0.5
        anchors.append(bv)
    ded = []
    for a in anchors:
        if all(np.max(np.abs(a - d)) >= 1e-9 for d in ded): ded.append(a)
    anchors = ded
    for a in anchors:
        sc = exact_score(a, n)
        if sc is not None and sc < best_sc: best_sc = sc; best_h = a.copy()
    gc = 80 if n <= 128 else (45 if n <= 256 else (18 if n <= 512 else 6))
    ma = 4 if n <= 128 else (3 if n <= 256 else (2 if n <= 512 else 1))
    nr = 2 if n <= 128 else (1 if n <= 256 else 0)
    pat = max(12, gc // 4)
    sb = float(np.clip(0.10 * np.sqrt(n / D), 0.05, 0.70))
    for ai in range(min(ma, len(anchors))):
        ho, sc_o = cma_run(anchors[ai], B, D, S, n, sb, gc, pat, rng)
        if ho is not None:
            s2 = exact_score(ho, n)
            if s2 is not None and s2 < best_sc: best_sc = s2; best_h = ho.copy()
    factor = 1.0
    for _ in range(nr):
        sb_r = float(np.clip(sb * factor, 0.03, 1.5))
        sc_before = best_sc
        ho, _ = cma_run(best_h, B, D, S, n, sb_r, gc, pat, rng)
        if ho is not None:
            s2 = exact_score(ho, n)
            if s2 is not None and s2 < best_sc: best_sc = s2; best_h = ho.copy()
        factor = max(0.5, 0.65 * factor) if best_sc < sc_before - 1e-12 else min(6.0, 1.6 * factor)
    if not is_feas(best_h, n):
        fp = proj_cs(best_h, S)
        best_h = fp if fp is not None else h_const
    fs = exact_score(best_h, n)
    if fs is None or not np.isfinite(fs):
        if parent is not None:
            best_h = parent; fs = exact_score(parent, n)
        if fs is None or not np.isfinite(fs):
            best_h = h_const; fs = exact_score(h_const, n)
    if fs is None: fs = 0.5
    return np.clip(best_h, 0.0, 1.0), float(fs), n
