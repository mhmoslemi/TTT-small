            except Exception:

                if stagnation >= 2:

import numpy as np
import cvxpy as cp
from scipy.optimize import minimize

def project_cs(y, n, S):
    if not np.all(np.isfinite(y)):
        return np.full(n, S / n)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    e = S - h.sum()
    if abs(e) > 1e-10:
        if e > 0:
            for j in range(n):
                d = min(e, 1.0 - h[j])
                h[j] += d
                e -= d
                if abs(e) < 1e-12:
                    break
        else:
            for j in range(n):
                d = min(-e, h[j])
                h[j] -= d
                e += d
                if abs(e) < 1e-12:
                    break
    if h.min() < -1e-10 or h.max() > 1 + 1e-10 or abs(h.sum() - S) > 1e-8:
        return np.full(n, S / n)
    return h

def score_exact(h, n):
    if not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def feasible(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-10 and h.max() <= 1.0 + 1e-10 and abs(h.sum() - S) <= 1e-8)

def build_dc(n, dx):
    gamma = 1.25 * dx
    Lp_list, Ln_list = [], []
    for d in range(n):
        B = np.zeros((n, n))
        if d == 0:
            B = dx * np.eye(n)
        else:
            idx = np.arange(n - d)
            B[idx, idx + d] = dx / 2
            B[idx + d, idx] = dx / 2
        P = 0.5 * (gamma * np.eye(n) + B)
        N = 0.5 * (gamma * np.eye(n) - B)
        try:
            Lp = np.linalg.cholesky(P)
            Ln = np.linalg.cholesky(N)
        except np.linalg.LinAlgError:
            return None
        Lp_list.append(Lp)
        Ln_list.append(Ln)
    c_list, d_list = [], []
    for ell in range(-(n - 1), n):
        d = abs(ell)
        c = np.zeros(n)
        if d == 0:
            c[:] = dx
        elif ell > 0:
            c[:n - d] = dx
        else:
            c[d:] = dx
        c_list.append(c)
        d_list.append(d)
    return Lp_list, Ln_list, c_list, d_list

    h = h0.copy()
    best_h, best_s = h.copy(), score_exact(h, n)
    solver = None
    try:
        for s in ['ECOS', 'CVXOPT']:
            if s in cp.installed_solvers():
                solver = s
                break
        if solver is None:
            return best_h
    except Exception:
        return best_h
    try:
        y_var = cp.Variable(n)
        t_var = cp.Variable()
        x_par = cp.Parameter(n)
        rho_par = cp.Parameter(nonneg=True)
        v_pars = [cp.Parameter(n) for _ in range(n)]
        a_pars = [cp.Parameter() for _ in range(n)]

def slsqp_polish(h0, n, S, dx, maxiter=300):
    h = h0.copy()

    def obj(z):
        return z[-1]

    def obj_jac(z):
        g = np.zeros(len(z))
        g[-1] = 1.0
        return g

    def eq_fun(z):
        return z[:n].sum() - S

    def eq_jac(z):
        g = np.ones(len(z))
        g[-1] = 0.0
        return g

    def ineq_fun(z):
        hv = z[:n]
        return z[-1] - np.correlate(hv, 1.0 - hv, mode='full') * dx

    def ineq_jac(z):
        hv = z[:n]
        nl = 2 * n - 1
        J = np.zeros((nl, n + 1))
        j = np.arange(n)
        for k in range(nl):
            ell = k - (n - 1)
            i1, v1 = j - ell, (j - ell >= 0) & (j - ell < n)
            i2, v2 = j + ell, (j + ell >= 0) & (j + ell < n)
            t1, t2 = np.zeros(n), np.zeros(n)
            t1[v1] = 1.0 - hv[i1[v1]]
            t2[v2] = hv[i2[v2]]
            J[k, :n] = -dx * (t1 - t2)
            J[k, n] = 1.0
        return J

    try:
        z0 = np.concatenate([h, [score_exact(h, n) + 1e-8]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
                {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}]
        res = minimize(obj, z0, jac=obj_jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
        if res is not None and hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            hs = project_cs(res.x[:n].copy(), n, S)
            if feasible(hs, n, S):
                return hs
    except Exception:
        pass
    return h

        return (fb_h, fb_s, n)
