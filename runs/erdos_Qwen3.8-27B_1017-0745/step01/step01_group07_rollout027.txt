        for k in range(5):
            t = lo2 + (hi2-lo2)*k/4.0
            f = np.max(c + b*t + a*t*t)
            if f < best_f - 1e-13:
                best_f = f; best_t = t
        rad /= 4.0
    if best_f >= score - 1e-12:
        return None
    h_new = h.copy()
    h_new[p] += best_t
    h_new[r] -= best_t
    h_new = np.clip(h_new, 0.0, 1.0)
    if not is_feasible(h_new, n):
        h_new = project_capped_simplex(h_new, n/2.0)
        if h_new is None:
            return None
    return h_new

def polish_all_pairs(h, n, S, max_sweeps=3):
    sc = exact_score(h, n)
    if sc is None:
        return h, 0.5
    for _ in range(max_sweeps):
        improved = False
        q = exact_lags(h, n)
        if q is None:
            break
        for p in range(n-1):
            for r in range(p+1, n):
                res = pair_line_search(h, q, sc, p, r, n, 1.0)
                if res is not None:
                    ns = exact_score(res, n)
                    if ns is not None and ns < sc - 1e-12:
                        h = res; sc = ns
                        q = exact_lags(h, n)
                        if q is None:
                            break
                        improved = True
            if q is None:
                break
        if not improved:
            break
    return h, sc

def exp_safe(x):
    if x < -700:
        return 0.0
    try:
        return math.exp(x)
    except (OverflowError, ValueError):
        return 0.0

def run(seed=42, **kwargs):
    if seed is None:
        seed = 42
    try:
        seed = int(seed) % (2**32)
    except (TypeError, ValueError):
        seed = 42
    rng = np.random.default_rng(seed)
    n, parent = None, None
    iv = globals().get('initial_h_values', None)
    if iv is not None:
        try:
            iv_arr = np.asarray(iv, dtype=np.float64)
            if iv_arr.ndim == 1 and 2 <= iv_arr.size <= 999:
                n = iv_arr.size
                S = n/2.0
                if np.all(np.isfinite(iv_arr)) and np.max(np.abs(iv_arr)) <= 1e12:
                    if np.all(iv_arr >= 0) and np.all(iv_arr <= 1) and abs(np.sum(iv_arr)-S) <= 1e-10:
                        parent = iv_arr.copy()
                    else:
                        parent = project_capped_simplex(iv_arr, S)
        except Exception:
            pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 80))
        except (TypeError, ValueError):
            n = 80
        if n < 8 or n > 999:
            n = 80
    S = n/2.0
    dx = 2.0/n
    h_const = np.full(n, 0.5, dtype=np.float64)
    best_h = h_const.copy()
    best_score = exact_score(h_const, n)
    if best_score is None:
        best_score = 0.5
    if parent is not None:
        ps = exact_score(parent, n)
        if ps is not None and ps < best_score:
            best_score, best_h = ps, parent.copy()
    starts = []
    if parent is not None:
        starts.append(parent.copy())
    idx = np.arange(n, dtype=np.float64)
    u = 0.5*np.ones(n)
    for m in range(1, min(4, (n-1)//2)+1):
        u += rng.normal(0, 0.12/np.sqrt(m))*np.cos(2*np.pi*m*(idx+0.5)/n + rng.uniform(0, 2*np.pi))
    up = project_capped_simplex(u, S)
    if up is not None:
        starts.append(up)
    perm = rng.permutation(n)
    bv = np.zeros(n); bv[perm[:n//2]] = 1.0
    if n % 2: bv[perm[n//2]] = 0.5
    starts.append(bv)
    rp = project_capped_simplex(rng.random(n), S)
    if rp is not None:
        starts.append(rp)
    if parent is not None:
        starts.append(parent[::-1].copy())
    starts.append(h_const.copy())
    deduped = []
    for s in starts:
        s = np.asarray(s, dtype=np.float64)
        if not is_feasible(s, n):
            s = project_capped_simplex(s, S)
            if s is None:
                continue
        if not any(np.max(np.abs(s-d)) < 1e-10 for d in deduped):
            deduped.append(s)
    cap = 6 if n <= 128 else 4 if n <= 256 else 3 if n <= 512 else 2
    deduped = deduped[:cap]
    for s in deduped:
        sc = exact_score(s, n)
        if sc is not None and sc < best_score:
            best_score, best_h = sc, s.copy()
    n_ep = 6 if n <= 128 else 4 if n <= 256 else 3 if n <= 512 else 2
    K = min(1500, max(100, 100000//n))
    for sh in deduped:
        h_cur = sh.copy()
        sc_cur = exact_score(sh, n)
        if sc_cur is None:
            continue
        scale = 0.15
        for ep in range(n_ep):
            T = max(5e-5, 4e-3*(0.4**ep))
            acc = 0
            for _ in range(K):
                p = int(rng.integers(0, n)); r = int(rng.integers(0, n))
                if p == r:
                    continue
                t = float(rng.uniform(-scale, scale))
                if t < max(-h_cur[p], h_cur[r]-1.0) or t > min(1.0-h_cur[p], h_cur[r]):
                    continue
                h_new = h_cur.copy()
                h_new[p] += t; h_new[r] -= t
                h_new = np.clip(h_new, 0.0, 1.0)
                if not is_feasible(h_new, n):
                    h_new = project_capped_simplex(h_new, S)
                    if h_new is None:
                        continue
                sc_new = exact_score(h_new, n)
                if sc_new is None:
                    continue
                if sc_new < best_score:
                    best_score, best_h = sc_new, h_new.copy()
                dlt = sc_new - sc_cur
                if dlt <= 0 or rng.random() < exp_safe(-dlt/T):
                    h_cur, sc_cur = h_new, sc_new
                    acc += 1
            ratio = acc/max(K, 1)
            if ratio < 0.10:
                scale = min(0.45, 1.3*scale)
            elif ratio > 0.60:
                scale = max(0.01, 0.7*scale)
            h_pol, sc_pol = polish_all_pairs(h_cur, n, S, max_sweeps=1)
            if sc_pol < best_score:
                best_score, best_h = sc_pol, h_pol.copy()
            h_cur = h_pol
    best_h, best_score = polish_all_pairs(best_h, n, S, max_sweeps=4)
    # Second round: perturb best and re-polish
    h_pert = best_h + rng.normal(0, 0.08, n)
    h_pert = project_capped_simplex(h_pert, S)
    if h_pert is not None and is_feasible(h_pert, n):
        hp2, sp2 = polish_all_pairs(h_pert, n, S, max_sweeps=2)
        if sp2 < best_score:
            best_score, best_h = sp2, hp2
    best_h, best_score = polish_all_pairs(best_h, n, S, max_sweeps=2)
    if not is_feasible(best_h, n):
        best_h = project_capped_simplex(best_h, S)
        if best_h is None:
            best_h = h_const.copy()
    fs = exact_score(best_h, n)
    if fs is None:
        fs = 0.5
    return best_h, fs, n
