        if isinstance(kwn, int) and not isinstance(kwn, bool) and 2 <= kwn <= 999:
            n = kwn
        fallback = np.full(n, 0.5)
    fb_s = exact_score(fallback, n)
    best_h, best_s = fallback.copy(), fb_s
    starts = [fallback.copy(), project_capped(1.0 - fallback, n)[0],
              project_capped(fallback[::-1].copy(), n)[0],
              project_capped(1.0 - fallback[::-1].copy(), n)[0]]
    uniq = []
    for s in starts:
        if all(np.max(np.abs(s - u)) > 1e-10 for u in uniq):
            uniq.append(s)
    n_gen = 36 if n <= 100 else (12 if n <= 400 else 6)
    for ri, sig in enumerate([0.06, 0.10, 0.04]):
        anchor = uniq[ri % len(uniq)]
        try:
            bh, bs = cma_restart(anchor, n, sig, n_gen, rng)
            if bs < best_s - 1e-12:
                best_s, best_h = bs, bh.copy()
        except Exception:
            pass
    if n <= 100:
        for _ in range(3):
            z = rng.standard_normal(n)
            z -= z.mean()
            nz = np.sqrt(float(np.sum(z * z)))
            if nz > 1e-12:
                pc, ok = project_capped(best_h + 0.08 * z / nz, n)
                if ok and np.max(np.abs(pc - best_h)) > 1e-5:
                    try:
                        bh, bs = cma_restart(pc, n, 0.03, 20, rng)
                        if bs < best_s - 1e-12:
                            best_s, best_h = bs, bh.copy()
                    except Exception:
                        pass
                    break
    fh, ok = project_capped(best_h, n)
    if not ok:
        fh = fallback.copy()
    fs = exact_score(fh, n)
    if fs > fb_s + 1e-12:
        fh, fs = fallback.copy(), fb_s
    return fh.astype(np.float64), float(fs), int(n)
