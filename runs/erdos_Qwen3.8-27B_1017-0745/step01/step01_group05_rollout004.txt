def project_onto_k(y, n):

        else:

    n = len(h)
    if not np.all(np.isfinite(h)):
        return float('inf')

def lag_scores(h):

        if 0 <= j - k < n:

        if 0 <= j + k < n:

import numpy as np
import math

def project_onto_k(y, n):
    S = n / 2.0
    if not np.all(np.isfinite(y)):
        return np.full(n, 0.5)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        c = np.clip(y - mid, 0.0, 1.0)
        if c.sum() > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(50):
        diff = S - c.sum()
        if abs(diff) < 1e-14:
            break
        mask = c < 1.0 - 1e-12 if diff > 0 else c > 1e-12
        cnt = int(mask.sum())
        if cnt == 0:
            break
        c[mask] += diff / cnt
        np.clip(c, 0.0, 1.0, out=c)
    return c

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(h.sum() - n / 2.0) <= 1e-10)

def exact_score(h):
    n = len(h)
    if not np.all(np.isfinite(h)):
        return float('inf')
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lag_scores(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def pair_coeffs(h, i, j, n):
    dx = 2.0 / n
    b = np.zeros(2 * n - 1)
    ks = np.arange(-(n - 1), n)
    ik = i - ks
    jk = j - ks
    ikp = i + ks
    jkp = j + ks
    m1 = (ik >= 0) & (ik < n)
    m2 = (jk >= 0) & (jk < n)
    m3 = (ikp >= 0) & (ikp < n)
    m4 = (jkp >= 0) & (jkp < n)
    b = np.zeros(2 * n - 1)
    b[m1] += (1.0 - h[ik[m1]])
    b[m2] -= (1.0 - h[jk[m2]])
    b[m3] -= h[ikp[m3]]
    b[m4] += h[jkp[m4]]
    b *= dx
    a = np.zeros(2 * n - 1)
    a[n - 1] = -2.0 * dx
    if i != j:
        k1 = i - j
        k2 = j - i
        if abs(k1) <= n - 1:
            a[k1 + n - 1] = dx
        if abs(k2) <= n - 1:
            a[k2 + n - 1] = dx
    return b, a

def pair_line_search(h, i, j, q, n):
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-12:
        return h, q, float(q.max())
    b, a = pair_coeffs(h, i, j, n)
    cur = float(q.max())
    ts = np.linspace(lo, hi, 13)
    idx0 = np.searchsorted(ts, 0.0)
    if idx0 < len(ts) and abs(ts[idx0]) > 1e-13:
        ts = np.insert(ts, idx0, 0.0)
    best_t = 0.0
    best_pred = cur
    for t in ts:
        p = float(np.max(q + b * t + a * t * t))
        if p < best_pred:
            best_pred = p
            best_t = t
    for _ in range(3):
        span = (hi - lo) / 13.0
        lo_b = max(lo, best_t - 2 * span)
        hi_b = min(hi, best_t + 2 * span)
        if hi_b - lo_b < 1e-13:
            break
        bracket = np.linspace(lo_b, hi_b, 5)
        lt = best_t
        lp = best_pred
        for t in bracket:
            p = float(np.max(q + b * t + a * t * t))
            if p < lp:
                lp = p
                lt = t
        if abs(lt - best_t) < 1e-13:
            break
        best_t = lt
        best_pred = lp
    cands = [0.0, best_t]
    for t in ts:

            p = float(np.max(q + b * t + a * t * t))
            if p < best_pred + 1e-9:
                cands.append(t)
                if len(cands) >= 5:

    best_s = cur
    best_q = q

            continue

        if s < best_s - 1e-13:

            best_h = hc

def exchange_polish(h0, n, rng, max_sweeps, pair_budget):
    h = h0.copy()
    q = lag_scores(h)
    s = float(q.max())
    total_pairs = n * (n - 1) // 2
    P = min(total_pairs, pair_budget)
    for sw in range(max_sweeps):
        pairs = []
        seen = set()
        attempts = 0
        while len(pairs) < P and attempts < 20 * P:
            attempts += 1
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            key = (min(i, j), max(i, j))
            if key in seen:
                continue
            seen.add(key)
            pairs.append(key)
        ii = 0
        while len(pairs) < P:
            done = True
            for i in range(n):
                for j2 in range(i + 1, n):
                    if (i, j2) not in seen:
                        pairs.append((i, j2))
                        seen.add((i, j2))
                        if len(pairs) >= P:
                            break
                if len(pairs) >= P:
                    break
            if not done and len(pairs) < P:
                break
        rng.shuffle(np.arange(len(pairs)))
        perm = rng.permutation(len(pairs))
        accepted = 0
        for idx in perm:
            i, j = pairs[idx]
            h, q, s2 = pair_line_search(h, i, j, q, n)
            if s2 < s - 1e-13:
                s = s2
                accepted += 1
        if accepted == 0:
            break
    return h, s
