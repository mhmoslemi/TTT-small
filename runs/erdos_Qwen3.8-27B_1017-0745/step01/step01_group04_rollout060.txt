import numpy as np

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo, hi = float(z.min()) - 1.0, float(z.max())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    if r > 0:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
            if r <= 1e-14:
                break
    elif r < 0:
        for i in range(N):
            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
            if r >= -1e-14:
                break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1 + 1e-10:
        return np.full(N, 0.5)
    if abs(h.sum() - m) > 1e-8 * max(1, N):
        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return np.inf
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / N)

def smooth_and_grad(h, tau, N):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = dx * corr
    cmax = c.max()
    e = np.exp((c - cmax) / tau)
    S = e.sum()
    F = cmax + tau * np.log(S)
    p = e / S
    grad = np.zeros(N)
    for r in range(2 * N - 1):
        w = dx * p[r]
        if w < 1e-16:
            continue
        k = r - (N - 1)
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:N] += w * (1.0 - h[:N - k])
            grad[:N - k] -= w * h[k:]
        else:
            d = -k
            grad[:N - d] += w * (1.0 - h[d:])
            grad[d:] -= w * h[:N - d]
    return F, grad, c

                break

            d = y - h

                alpha *= 0.5
                if alpha < 1e-12:

                continue
            s_new = float(c_new.max())

def subgrad_step(h, N, m):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    r_best = int(np.argmax(corr))
    k = r_best - (N - 1)
    g = np.zeros(N)
    if k == 0:
        g = 1.0 - 2.0 * h
    elif k > 0:
        g[k:N] += (1.0 - h[:N - k])
        g[:N - k] -= h[k:]
    else:
        d = -k
        g[:N - d] += (1.0 - h[d:])
        g[d:] -= h[:N - d]
    alpha = 2.0 * dx / (np.dot(g, g) + 1e-20)
    y = project_box_sum(h - alpha * g, m)
    return y

def make_fourier_start(N, rng, n_modes, amp):
    z = np.full(N, 0.5)
    idx = np.arange(N)
    R = min(n_modes, (N - 1) // 2)

        B = rng.standard_normal()
        z += (amp / r) * (A * np.sin(2.0 * np.pi * r * idx / N) + B * np.cos(2.0 * np.pi * r * idx / N))
    return z
