import numpy as np

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo, hi = z.min() - 1.0, z.max()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    if r > 0:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
            if r <= 1e-14:
                break
    elif r < 0:
        for i in range(N):
            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
            if r >= -1e-14:
                break
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)):
        return np.full(N, 0.5)
    return h

def exact_score(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / N)

def smooth_obj_and_grad(h, tau, N):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx  # length 2N-1, C[i] = dx * q_{i-(N-1)}
    cmax = np.max(C)
    shifted = (C - cmax) / tau
    exps = np.exp(shifted)
    exp_sum = np.sum(exps)
    p = exps / exp_sum  # softmax-like weights
    F = cmax + tau * np.log(exp_sum)
    
    grad = np.zeros(N)
    # For each lag s from -(N-1) to N-1, index in C is i = N-1+s
    for s in range(-(N-1), N):
        i = N - 1 + s
        if p[i] < 1e-15:
            continue
        w = dx * p[i]
        if s == 0:
            grad += w * (1.0 - 2.0 * h)
        elif s > 0:
            d = s
            # dC_d/dh[m] = (1-h[m-d]) for d<=m<=N-1 + (-h[m+d]) for 0<=m<=N-1-d
            if N - 1 - d >= 0:

def smooth_obj_grad(h, tau, N):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    cmax = float(np.max(C))
    shifted = (C - cmax) / tau
    exps = np.exp(shifted)
    exp_sum = float(np.sum(exps))
    p = exps / exp_sum
    F = cmax + tau * np.log(exp_sum)
    grad = np.zeros(N)
    for s in range(-(N - 1), N):
        i = N - 1 + s
        pi = p[i]
        if pi < 1e-15:
            continue
        w = dx * pi
        d = abs(s)
        if s == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d < N:
            if s > 0:
                grad[:N - d] -= w * h[d:N]
                grad[d:N] += w * (1.0 - h[:N - d])
            else:
                grad[:N - d] += w * (1.0 - h[d:N])
                grad[d:N] -= w * h[:N - d]
    return F, grad

import numpy as np

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(N, 0.5)
    lo, hi = float(z.min()) - 1.0, float(z.max())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(z - mid, 0.0, 1.0).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - float(h.sum())
    if r > 1e-14:
        for i in range(N):
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
            if r <= 1e-14:
                break
    elif r < -1e-14:
        for i in range(N):
            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
            if r >= -1e-14:
                break

        if s == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            d = abs(s)
            if d >= N:
                continue
            if s > 0:
                grad[:N - d] -= w * h[d:N]
                grad[d:N] += w * (1.0 - h[:N - d])
            else:
                grad[:N - d] += w * (1.0 - h[d:N])
                grad[d:N] -= w * h[:N - d]
    return F, grad
