def exact_score(h, n):

    c = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(c) * dx)
    return val if np.isfinite(val) else float('inf')

def product_idx(i, j, n):
    # index for pair (min(i,j), max(i,j))
    a, b = min(i, j), max(i, j)
    return a * n - a * (a - 1) // 2 + (b - a)

def compute_lag_coeffs(x, d, n):
    # Returns A, B, D arrays of length 2n-1 for h(a) = x + a*d
    n_lags = 2 * n - 1
    A = np.zeros(n_lags)
    B = np.zeros(n_lags)
    D = np.zeros(n_lags)
    for k in range(-(n - 1), n):
        idx = k + n - 1

        i_hi = min(n - 1, n - 1 + k)

            j = i - k
            D[idx] += x[i] * (1.0 - x[j])
            B[idx] += d[i] * (1.0 - x[j]) - x[i] * d[j]
            A[idx] += -d[i] * d[j]
    return A, B, D
