def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(v - tau, 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-14:
        for i in np.argsort(1.0 - y)[::-1]:
            d = min(r, 1.0 - y[i])
            y[i] += d
            r -= d
            if r <= 1e-15:
                break
    elif r < -1e-14:
        for i in np.argsort(y)[::-1]:
            d = min(-r, y[i])
            y[i] -= d
            r += d
            if r >= -1e-15:
                break

        return None

    return y

                    alpha = min(10.0, max(1e-3, float(np.dot(d_old, d_old)) / dy))
                else:
                    alpha = min(10.0, 1.5 * alpha)
                if abs(F_old - F_cur) < 1e-12:
                    small_count += 1
                else:
                    small_count = 0
                break
            alpha *= 0.5
        if not accepted:
            break
        # Checkpoint every 10 iterations
        if (it + 1) % 10 == 0:
            c = exact_score(cur)
            if c < best_stage_c:
                best_stage_c = c
                best_stage_h = cur.copy()
        if small_count >= 8:
            break
    c = exact_score(cur)
    if c < best_stage_c:
        best_stage_c = c
        best_stage_h = cur.copy()
    return cur, best_stage_h, best_stage_c

def run(seed=42, **kwargs):

    else:
        sds = [0.12]

    for sd in sds:
        pert = project_box_sum(best_h + sd * rng.standard_normal(n), 0.5 * n)
