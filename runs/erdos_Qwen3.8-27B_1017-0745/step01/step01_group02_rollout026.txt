def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1 or len(v) == 0 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    if S < 0 or S > n:
        return None
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-15:
        for i in np.argsort(1.0 - y):
            d = min(r, 1.0 - y[i]); y[i] += d; r -= d
            if r <= 0: break
    elif r < -1e-15:
        for i in np.argsort(y):
            d = min(-r, y[i]); y[i] -= d; r += d
            if r >= 0: break
    if abs(r) > 1e-12 * max(1, n) or not np.all(np.isfinite(y)):
        return None
    if abs(float(np.sum(y)) - S) > 1e-10 * max(1, n):
        return None
    return y

def ok(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and np.all(h >= -1e-12)
            and np.all(h <= 1.0 + 1e-12) and abs(float(np.sum(h)) - 0.5 * n) < 1e-10 * max(1, n))

def score(h, n):
    if not ok(h, n):
        return float('inf')
    try:
        c = np.correlate(h, 1.0 - h, mode='full')
        if len(c) != 2 * n - 1 or not np.all(np.isfinite(c)):
            return float('inf')
        s = float(np.max(c)) * (2.0 / n)
        return s if np.isfinite(s) else float('inf')
    except Exception:
        return float('inf')

def parent_vec():
    if 'initial_h_values' in globals():
        ph = np.asarray(globals()['initial_h_values'], dtype=np.float64).copy()
        if ph.ndim == 1 and 2 <= len(ph) <= 999 and np.all(np.isfinite(ph)):
            if np.all(ph >= -1e-8) and np.all(ph <= 1.0 + 1e-8):
                n = len(ph)
                if ok(ph, n):
                    return ph, n
                p = project_box_sum(np.clip(ph, 0, 1), 0.5 * n)
                if p is not None and ok(p, n):
                    return p, n
    return np.full(45, 0.5), 45

def uadd(pop, c, n):
    if c is None or not ok(c, n):
        return False
    for m in pop:
        if float(np.max(np.abs(c - m))) < 1e-10:
            return False
    pop.append(c.copy())
    return True

def sF(rng, mf):
    for _ in range(11):
        u = min(max(rng.random(), 1e-6), 1.0 - 1e-6)
        v = mf + 0.1 * math.tan(math.pi * (u - 0.5))
        if v > 0:
            return min(max(v, 0.05), 1.0)
    return 0.5

def sCR(rng, mc):
    return min(max(rng.normal(mc, 0.1), 0.0), 1.0)

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    par, n = parent_vec()
    S = 0.5 * n
    psc = score(par, n)
    if n <= 64:
        P, G, sl, mr = 96, 300, 50, 3
    elif n <= 128:
        P, G, sl, mr = 56, 150, 30, 2
    else:
        P, G, sl, mr = 20, 32, 10, 1
    pf, muF, muCR = 0.20, 0.55, 0.70
    pop = []
    uadd(pop, par.copy(), n)
    uadd(pop, project_box_sum(1.0 - par, S), n)
    uadd(pop, project_box_sum(par[::-1], S), n)
    uadd(pop, project_box_sum(np.roll(par, 1), S), n)
    for sd in [0.03, 0.08, 0.18, 0.35]:
        uadd(pop, project_box_sum(par + sd * rng.standard_normal(n), S), n)
    for _ in range(6):
        uadd(pop, project_box_sum(rng.random(n), S), n)
    idx = np.arange(n)
    for m in range(1, min(5, n // 2)):
        p1 = rng.random() * 2 * math.pi
        p2 = rng.random() * 2 * math.pi
        u = 0.5 + 0.28 * np.cos(2 * math.pi * m * idx / n + p1) + 0.12 * np.sin(2 * math.pi * (m + 1) * idx / n + p2)
        uadd(pop, project_box_sum(u, S), n)
    for _ in range(30 * P):
        if len(pop) >= P:
            break
        uadd(pop, project_box_sum(par + (0.03 + 0.32 * rng.random()) * rng.standard_normal(n), S), n)
        uadd(pop, project_box_sum(rng.random(n), S), n)
    while len(pop) < P:
        pop.append(par.copy())
    pop = pop[:P]
    sc = np.array([score(m, n) for m in pop])
    for i in range(P):
        if not np.isfinite(sc[i]):
            pop[i] = par.copy(); sc[i] = psc
    gb = pop[int(np.argmin(sc))].copy(); gbs = float(np.min(sc))
    arch = []; ni = 0; nr = 0
    for g in range(G):
        SH = [h.copy() for h in pop]
        SS = sc.copy()
        NP = [h.copy() for h in pop]
        NS = sc.copy()
        order = np.argsort(SS)
        tc = max(2, int(math.ceil(pf * P)))
        top = order[:tc]
        sFl = []; sCRl = []
        asp = list(arch)
        for i in range(P):
            pi = int(top[rng.integers(tc)])
            if pi == i:
                pi = int(top[(top != i)[0] if np.any(top != i) else 0])
            rem = [j for j in range(P) if j != i and j != pi]
            if not rem:
                continue
            r1 = int(rem[rng.integers(len(rem))])
            rem.remove(r1)
            if asp and rng.random() < 0.3:
                r2 = asp[rng.integers(len(asp))]
            else:
                r2 = SH[rem[rng.integers(len(rem))]]
            Fi = sF(rng, muF); CRi = sCR(rng, muCR)
            vi = SH[i] + Fi * (SH[pi] - SH[i]) + Fi * (SH[r1] - r2)
            mask = rng.random(n) > CRi
            mask[rng.integers(n)] = False
            trial = np.where(mask, SH[i], vi)
            trial = project_box_sum(trial, S)
            if trial is None:
                continue
            ts = score(trial, n)
            if not np.isfinite(ts):
                continue
            if ts < NS[i] - 1e-14 * max(1.0, abs(NS[i])):
                NP[i] = trial; NS[i] = ts
                arch.append(SH[i].copy())
                sFl.append(Fi); sCRl.append(CRi)
                if ts < gbs:
                    gb = trial.copy(); gbs = ts
        while len(arch) > P:
            arch.pop(int(rng.integers(len(arch))))
        pop = NP; sc = NS
        if sFl:
            a = np.array(sFl)
            muF = min(max(0.9 * muF + 0.1 * float(np.sum(a * a) / np.sum(a)), 0.05), 1.0)
        if sCRl:
            muCR = min(max(0.9 * muCR + 0.1 * float(np.mean(sCRl)), 0.0), 1.0)
        if gbs < float(np.min(SS)) - 1e-12:
            ni = 0
        else:
            ni += 1
        if ni >= sl:
            if nr >= mr:
                break
            nr += 1; ni = 0
            so = np.argsort(sc)
            npop = [pop[int(so[0])].copy()]
            for k in range(1, P):
                if len(npop) >= max(2, P // 4):
                    break
                cvec = pop[int(so[k])]
                dup = False
                for e in npop:
                    if float(np.max(np.abs(cvec - e))) < 1e-10:
                        dup = True; break
                if not dup:
                    npop.append(cvec.copy())
            nsd = [0.04, 0.12, 0.30][(nr - 1) % 3]
            fill = int(0.7 * (P - len(npop)))
            for _ in range(fill):
                uadd(npop, project_box_sum(gb + nsd * rng.standard_normal(n), S), n)
            for _ in range(30 * P):
                if len(npop) >= P:
                    break
                uadd(npop, project_box_sum(gb + (0.03 + 0.3 * rng.random()) * rng.standard_normal(n), S), n)
                uadd(npop, project_box_sum(rng.random(n), S), n)
            while len(npop) < P:
                npop.append(gb.copy())
            pop = npop[:P]
            sc = np.array([score(m, n) for m in pop])
            for i in range(P):
                if not np.isfinite(sc[i]):
                    pop[i] = par.copy(); sc[i] = psc
            arch = []; muF = 0.55; muCR = 0.70
    fh = project_box_sum(gb, S)
    if fh is not None and ok(fh, n):
        fs = score(fh, n)
        if np.isfinite(fs) and fs < psc:
            return (fh, fs, n)
    fh = project_box_sum(par, S)
    if fh is None:
        fh = np.full(n, 0.5)
    fs = score(fh, n)
    if not np.isfinite(fs):
        fs = 0.5
    return (fh, fs, n)
