            eta = min(1.2 * eta, eta_max)
        else:
            eta *= 0.5
            fails += 1
    return current

def build_starts(base, n, rng, count):
    pool = [base.copy()]
    candidates = []
    candidates.append(project_capped(base[::-1], n))
    candidates.append(project_capped(1.0 - base[::-1], n))
    candidates.append(np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    candidates.append(project_capped(blk, n))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    candidates.append(project_capped(alt, n))
    candidates.append(project_capped(base + rng.normal(0, 0.08, n), n))
    candidates.append(project_capped(base + rng.normal(0, 0.20, n), n))
    candidates.append(project_capped(base + rng.normal(0, 0.35, n), n))
    candidates.append(project_capped(rng.uniform(0, 1, n), n))
    for c in candidates:
        if c is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(c - s)) < 1e-8:
                dup = True
                break
        if not dup:
            pool.append(c)
    return pool[:count]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0:
            n = int(p.size)
            if n <= 999 and np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        if 'n_points' in kwargs:
            try:
                v = int(kwargs['n_points'])
                if 1 <= v <= 999:
                    n = v
            except Exception:
                pass
        if n is None or not (1 <= n <= 999):
            n = 47
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if parent is not None:
        pb = project_capped(parent, n)
        fallback = pb if (pb is not None and is_feasible(pb, n)) else np.full(n, 0.5)
    else:
        fallback = np.full(n, 0.5)
    if n == 1:
        return fallback, exact_score(fallback, 1), 1
    if n <= 64:
        n_starts, maxiter_slsqp, t_i = 8, 220, 0
    elif n <= 128:
        n_starts, maxiter_slsqp, t_i = 3, 0, 45
    elif n <= 256:
        n_starts, maxiter_slsqp, t_i = 2, 0, 30
    else:
        n_starts, maxiter_slsqp, t_i = 1, 0, 18
    starts = build_starts(fallback, n, rng, n_starts)
    gs = exact_score(fallback, n)
    gb = fallback.copy()
    for st in starts:
        s = exact_score(st, n)
        if s < gs - 1e-12:
            gs, gb = s, st.copy()
    if n <= 64:
        for st in starts:
            try:
                xo = slsqp_start(st, n, maxiter_slsqp)
                so = exact_score(xo, n)
                if so < gs - 1e-12:
                    gs, gb = so, xo.copy()
            except Exception:
                pass
    else:
        for st in starts:
            try:
                xo = subgrad_start(st, n, t_i)
                so = exact_score(xo, n)
                if so < gs - 1e-12:
                    gs, gb = so, xo.copy()
            except Exception:
                pass
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    fb_s = exact_score(fallback, n)
    if fs > fb_s + 1e-12:
        final, fs = fallback, fb_s
    return final, float(fs), int(n)
