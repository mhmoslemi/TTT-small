            lo = mid
        else:
            hi = mid
    theta = lo + 0.5 * (hi - lo)

    r = S - np.sum(x)
    for i in range(n):
        if r > 0:
            a = min(r, 1.0 - x[i])
            x[i] += a; r -= a
        elif r < 0:
            a = min(-r, x[i])
            x[i] -= a; r += a

            a = min(-r, x[i])

def exact_score(h, n):

        if len(vals) != 2 * n - 1 or not np.all(np.isfinite(vals)):

        return float(np.max(vals))
    except Exception:
        return None

def build_dct_basis(n, D):
    idx = np.arange(n, dtype=np.float64)
    modes = []
    L = (D + 1) // 2
    for k in range(1, L + 1):
        modes.append(k)
    R = D - L
    if R > 0:
        if R == 1:
            candidates = [n - 1]
        else:
            candidates = [int(round(L + 1 + (n - 1 - L) * j / (R - 1))) for j in range(R)]
        for c in candidates:

    for k in range(L + 1, n):

            modes.append(k)

        C = (1 - c1 - c_mu) * C + c1 * (np.outer(p_c, p_c) + (1 - h_sigma) * c_c * (2 - c_c) * C)
        for i in range(mu):

        anchors = anchors[:2]
    gen_cap = 80 if n <= 128 else (45 if n <= 256 else 18)
    for an in anchors:
        bh, bs = cmaes_optimize(an, B, S, n, rng, gen_cap, D)
        if bs < best_score:
            best_score = bs
            best_h = bh.copy()
    n_rest = 2 if n <= 128 else (1 if n <= 256 else 0)
    for _ in range(n_rest):
        anchor2 = project_capped_simplex(best_h + rng.normal(0, 0.05, n), S)
        if anchor2 is None:
            anchor2 = best_h.copy()
        g2 = max(20, gen_cap // 2)
        bh, bs = cmaes_optimize(anchor2, B, S, n, rng, g2, D)
        if bs < best_score:
            best_score = bs
            best_h = bh.copy()
    if not is_feasible(best_h, n):
        best_h = project_capped_simplex(best_h, S)
        if best_h is None:
            best_h = parent.copy() if parent is not None else h_const.copy()
    final_sc = exact_score(best_h, n)
    if final_sc is None or not np.isfinite(final_sc):
        if parent is not None:
            best_h = parent.copy()
            final_sc = exact_score(best_h, n)
        if final_sc is None or not np.isfinite(final_sc):
            best_h = h_const.copy()
            final_sc = 0.5
    return best_h, final_sc, n
