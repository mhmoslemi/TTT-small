            if rng.uniform() < 0.5:
                mutant = pop[idx[0]].copy() + F * (pop[idx[1]] - pop[idx[2]])
            else:
                mutant = pop[i].copy() + F * (pop[bj] - pop[i]) + F * (pop[idx[1]] - pop[idx[2]])
            mask = rng.random(n) < CR
            mask[int(rng.integers(n))] = True
            trial = np.where(mask, mutant, pop[i])
            trial = project_capped(trial, n)
            if trial is None or not is_feasible(trial, n):
                continue
            ts = exact_score(trial, n)
            if ts < scores[i] - 1e-12:
                pop[i] = trial
                scores[i] = ts
                Fa[i] = F
                CRa[i] = CR
                if ts < gs - 1e-12:
                    gs, gb = ts, trial.copy()
        if gs < sgs - 1e-12:
            stall = 0
        else:
            stall += 1
        if np.min(scores) > gs + 1e-12:
            wi = int(np.argmax(scores))
            pop[wi] = gb.copy()
            scores[wi] = gs
        if stall >= sl and restarts < mr:
            order = np.argsort(scores)
            keep = max(2, M // 4)
            kp = [int(k) for k in order[:keep]]
            npop = [pop[k].copy() for k in kp]
            ns = [float(scores[k]) for k in kp]
            if gs < min(ns) - 1e-12:
                npop.append(gb.copy())
                ns.append(gs)
            sigma = [0.06, 0.16, 0.32][restarts % 3] * sc
            while len(npop) < M:
                r = rng.uniform()
                if r < 0.5:
                    base = gb
                elif r < 0.75:
                    base = npop[int(rng.integers(len(npop)))]
                else:
                    base = rng.uniform(0, 1, n)
                p = project_capped(base + rng.normal(0, sigma, n), n)
                if p is None:
                    p = project_capped(rng.uniform(0, 1, n), n)
                if p is None:
                    npop.append(gb.copy())
                    ns.append(gs)
                else:
                    npop.append(p)
                    ns.append(exact_score(p, n))
            pop = npop
            scores = np.array(ns)
            restarts += 1
            stall = 0
    return gb, gs

def pattern_poll(gb, gs, pop, scores, n, rng, rounds, nd):
    sc = min(1.5, np.sqrt(47.0 / n))
    delta = min(0.25, 0.12 * sc)
    L = len(pop)
    for _ in range(rounds):
        if delta < 1e-4:
            break
        bt, bs = None, gs
        for _ in range(nd):
            if rng.uniform() < 0.5:
                v = rng.standard_normal(n)
            else:
                a, b = int(rng.integers(L)), int(rng.integers(L))
                v = pop[a].copy() - pop[b]
            v -= np.mean(v)
            vm = np.max(np.abs(v))
            if vm < 1e-12 or not np.isfinite(vm):
                continue
            v = v / vm
            tlo, thi = 0.0, 0.0
            pos = v > 1e-15
            neg = v < -1e-15
            if np.any(pos):
                tlo = max(tlo, float(np.max(-gb[pos] / v[pos])))
                thi = min(thi, float(np.min((1.0 - gb[pos]) / v[pos])))
            if np.any(neg):
                tlo = max(tlo, float(np.max((1.0 - gb[neg]) / v[neg])))
                thi = min(thi, float(np.min(-gb[neg] / v[neg])))
            tlo = max(tlo, -delta)
            thi = min(thi, delta)
            if thi - tlo < 1e-12:
                continue
            for t in (-delta, -0.5 * delta, 0.5 * delta, delta):
                tc = max(tlo, min(thi, t))
                if abs(tc) < 1e-12:
                    continue
                trial = project_capped(gb + tc * v, n)
                if trial is None or not is_feasible(trial, n):
                    continue
                ts = exact_score(trial, n)
                if ts < bs:
                    bs, bt = ts, trial
        if bt is not None and bs < gs - 1e-12:
            gb, gs = bt.copy(), bs
            delta = min(0.25, 1.2 * delta)
        else:
            delta *= 0.5
    return gb, gs

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None and 'n_points' in kwargs:
        try:
            n = int(kwargs['n_points'])
            if not (1 <= n <= 999):
                n = None
        except Exception:
            n = None
    if n is None:
        n = 47
    fb = project_capped(parent, n) if parent is not None else np.full(n, 0.5)
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    fb_s = exact_score(fb, n)
    if n == 1:
        return fb, fb_s, 1
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 64:
        M, gens, sl, mr, pr, nd = 40, 120, 20, 3, 28, 10
    elif n <= 128:
        M, gens, sl, mr, pr, nd = 36, 90, 16, 2, 20, 8
    elif n <= 256:
        M, gens, sl, mr, pr, nd = 28, 55, 12, 1, 12, 6
    elif n <= 512:
        M, gens, sl, mr, pr, nd = 20, 30, 10, 1, 8, 4
    else:
        M, gens, sl, mr, pr, nd = 14, 18, 8, 1, 5, 4
    gb, gs = fb.copy(), fb_s
    try:
        pop = make_starts(fb, n, rng, M)
        scores = np.array([exact_score(p, n) for p in pop])
        for i in range(M):
            if scores[i] < gs - 1e-12:
                gs, gb = scores[i], pop[i].copy()
        gb, gs = de_evolve([p.copy() for p in pop], scores.copy(), n, rng, gens, sl, mr)
        pop2 = [p for p in pop]
        sc2 = np.array([exact_score(p, n) for p in pop2])
        if gs < np.min(sc2) - 1e-12:
            wi = int(np.argmax(sc2))
            pop2[wi] = gb.copy()
            sc2[wi] = gs
        gb, gs = pattern_poll(gb, gs, pop2, sc2, n, rng, pr, nd)
    except Exception:
        pass
    fin = project_capped(gb, n)
    if fin is None or not is_feasible(fin, n):
        fin = fb
    fs = exact_score(fin, n)
    if fs > fb_s + 1e-12:
        fin, fs = fb, fb_s
    return fin, float(fs), int(n)
