import numpy as np
import math

def project_K(y, n, S):
    if y is None or len(y) != n or not np.all(np.isfinite(y)):
        return np.full(n, S / n)
    lo, hi = float(y.min()) - 1.0, float(y.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if float(np.clip(y - mid, 0.0, 1.0).sum()) > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    v = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    e = S - float(v.sum())
    if abs(e) > 1e-13:
        for j in range(n):
            if abs(e) < 1e-14:
                break
            if e > 0:
                d = min(e, 1.0 - v[j])
                v[j] += d
                e -= d
            else:
                d = min(-e, v[j])
                v[j] -= d
                e += d
    v = np.clip(v, 0.0, 1.0)
    if abs(float(v.sum()) - S) > 1e-9:
        return np.full(n, S / n)
    return v

def escore(h, dx):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def is_feas(h, n, S):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            float(h.min()) >= -1e-10 and float(h.max()) <= 1.0 + 1e-10 and
            abs(float(h.sum()) - S) <= 1e-8)

def is_dup(hv, pool, tol=1e-10):
    for ph in pool:
        if float(np.max(np.abs(hv - ph))) <= tol:
            return True
    return False
