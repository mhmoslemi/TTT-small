import numpy as np
from math import exp

def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):
        return None
    if np.max(np.abs(z)) > 1e12:
        return None
    n = len(z)
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z))
    for _ in range(80):
        mid = lo + 0.5 * (hi - lo)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = lo + 0.5 * (hi - lo)
    y = np.clip(z - lam, 0.0, 1.0)
    r = S - np.sum(y)
    for i in range(n):
        if r > 0:
            a = min(r, 1.0 - y[i])
            y[i] += a; r -= a
            if r <= 1e-12: break
        elif r < 0:
            a = min(-r, y[i])
            y[i] -= a; r += a
            if r >= -1e-12: break
    if abs(np.sum(y) - S) > 1e-10 or not np.all(np.isfinite(y)):
        return None
    return np.clip(y, 0.0, 1.0)

def is_feasible(h, n):
    return (isinstance(h, np.ndarray) and h.ndim == 1 and len(h) == n
            and np.all(np.isfinite(h)) and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-9)

def exact_lags(h, n):
    vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if vals.shape != (2 * n - 1,) or not np.all(np.isfinite(vals)):
        return None
    return vals

def exact_score(h, n):
    vals = exact_lags(h, n)
    if vals is None:
        return None
    return float(np.max(vals))

def pair_line_search(h, vals, p, r, n, dx):
    hp, hr = h[p], h[r]
    L = max(-hp, hr - 1.0)
    U = min(1.0 - hp, hr)
    if U - L <= 1e-12:
        return None
    c = vals  # (2n-1,)
    b = np.zeros(2 * n - 1)
    a = np.zeros(2 * n - 1)
    # a coefficients
    a[n - 1] = -2.0 * dx
    if r != p:
        a[n - 1 + (r - p)] = dx
        a[n - 1 + (p - r)] = dx
    # b coefficients
    for d_idx in range(2 * n - 1):
        d = d_idx - (n - 1)
        bd = 0.0
        # v[p]=1 term: (1-h[p+d]) if 0<=p+d<n
        ip = p + d
        if 0 <= ip < n:
            bd += (1.0 - h[ip])
        # v[r]=-1 term: -(1-h[r+d]) if 0<=r+d<n
        ir = r + d
        if 0 <= ir < n:
            bd -= (1.0 - h[ir])
        # -h[p-d]*v at i=p-d: -h[p-d] if 0<=p-d<n
        ip2 = p - d
        if 0 <= ip2 < n:
            bd -= h[ip2]
        # +h[r-d]*(-1) at i=r-d: ... wait
        # The second sum is -sum h[i]*v[i+d]
        # v[i+d] nonzero at i+d=r -> i=r-d, v[r]=−1
        # So -h[r-d]*(-1) = +h[r-d] if 0<=r-d<n
        ir2 = r - d
        if 0 <= ir2 < n:
            bd += h[ir2]
        b[d_idx] = dx * bd
    # Line search: minimize max(c + b*t + a*t²) over t in [L, U]
    n_pts = 17
    ts = np.linspace(L, U, n_pts)
    t_zero = 0.0
    ts = np.concatenate([ts, [t_zero]])
    ts = np.unique(ts)
    best_t = t_zero
    best_f = np.max(c)  # current score at t=0
    for t in ts:
        f = np.max(c + b * t + a * t * t)
        if f < best_f - 1e-14 or (abs(f - best_f) <= 1e-14 and abs(t) < abs(best_t)):
            best_f = f
            best_t = t
    # Refinement
    radius = (U - L) / 16.0
    for _ in range(4):
        lo_r = max(L, best_t - radius)
        hi_r = min(U, best_t + radius)
        if hi_r <= lo_r:
            radius /= 4.0
            continue
        ts2 = np.linspace(lo_r, hi_r, 5)
        for t in ts2:
            f = np.max(c + b * t + a * t * t)
            if f < best_f - 1e-14 or (abs(f - best_f) <= 1e-14 and abs(t) < abs(best_t)):
                best_f = f
                best_t = t
        radius /= 4.0
    if best_f >= np.max(c) - 1e-12:
        return None
    # Verify with exact score
    h_new = h.copy()
    h_new[p] += best_t
    h_new[r] -= best_t
    if h_new[p] < -1e-9 or h_new[p] > 1.0 + 1e-9 or h_new[r] < -1e-9 or h_new[r] > 1.0 + 1e-9:
        S = n / 2.0
        h_new = project_capped_simplex(h_new, S)
        if h_new is None or not is_feasible(h_new, n):
            return None
    sc = exact_score(h_new, n)
    if sc is None or sc >= exact_score(h, n) - 1e-12:
        return None
    vals_new = exact_lags(h_new, n)
    if vals_new is None:
        return None
    return h_new, sc, vals_new

def pairwise_polish(h, vals, score, n, dx, rng, max_sweeps=4):
    n_pairs = min(n * (n - 1) // 2, 2 * n, max(24, 80000 // max(n, 1)))
    pairs = []
    # Adjacent pairs
    n_adj = min(n - 1, max(1, n_pairs // 2))
    adj_idx = np.linspace(0, n - 2, n_adj).astype(int)
    for i in adj_idx:
        pairs.append((int(i), int(i + 1)))
    # Random pairs
    while len(pairs) < n_pairs:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        if i > j:
            i, j = j, i
        if (i, j) not in pairs:
            pairs.append((i, j))
    rng.shuffle(pairs)
    for sweep in range(max_sweeps):
        any_move = False
        cur_h, cur_vals, cur_score = h, vals, score
        for (p, r) in pairs:
            res = pair_line_search(cur_h, cur_vals, p, r, n, dx)
            if res is not None:
                cur_h, cur_score, cur_vals = res
                any_move = True
        h, vals, score = cur_h, cur_vals, cur_score
        if not any_move:
            break
    return h, vals, score

def anneal_trajectory(h0, n, dx, rng, epochs=5):
    S = n / 2.0
    h = h0.copy()
    score = exact_score(h, n)
    if score is None:
        return h0, 0.5
    vals = exact_lags(h, n)
    scale = 0.12
    K = min(1000, max(80, 80000 // max(n, 1)))
    for e in range(epochs):
        T = max(5e-5, 3e-3 * (0.4 ** e))
        accepted = 0
        for k in range(K):
            p = int(rng.integers(0, n))
            r = int(rng.integers(0, n))
            if p == r:
                continue
            t_target = float(rng.uniform(-scale, scale))
            hp, hr = h[p], h[r]
            L = max(-hp, hr - 1.0)
            U = min(1.0 - hp, hr)
            if t_target < L or t_target > U:
                continue
            h_new = h.copy()
            h_new[p] += t_target
            h_new[r] -= t_target
            if h_new[p] < -1e-9 or h_new[p] > 1.0 + 1e-9 or h_new[r] < -1e-9 or h_new[r] > 1.0 + 1e-9:
                h_new = project_capped_simplex(h_new, S)
                if h_new is None or not is_feasible(h_new, n):
                    continue
            sc = exact_score(h_new, n)
            if sc is None:
                continue
            delta = sc - score
            if delta <= 0:
                accept = True
            else:
                if delta < -700 * T:
                    accept = False
                else:
                    accept = rng.random() < exp(-delta / T) if T > 0 else False
            if accept:
                h = h_new
                score = sc
                vals = exact_lags(h, n)
                if vals is None:
                    vals = np.zeros(2 * n - 1)
                accepted += 1
        ratio = accepted / max(K, 1)
        if ratio < 0.12:
            scale = min(0.4, 1.25 * scale)
        elif ratio > 0.55:
            scale = max(0.01, 0.75 * scale)
        # One polish sweep per epoch
        h, vals, score = pairwise_polish(h, vals, score, n, dx, rng, max_sweeps=1)

import numpy as np
from math import exp

def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):
        return None
    if np.max(np.abs(z)) > 1e12:
        return None
    n = len(z)
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z))
    for _ in range(80):
        mid = lo + 0.5 * (hi - lo)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = lo + 0.5 * (hi - lo)
    y = np.clip(z - lam, 0.0, 1.0)
    r = S - float(np.sum(y))
    for i in range(n):
        if r > 1e-12:
            a = min(r, 1.0 - y[i]); y[i] += a; r -= a
        elif r < -1e-12:
            a = min(-r, y[i]); y[i] -= a; r += a
        else:
            break
    if abs(float(np.sum(y)) - S) > 1e-10 or not np.all(np.isfinite(y)):
        return None
    return np.clip(y, 0.0, 1.0)

def is_feasible(h, n):
    return (isinstance(h, np.ndarray) and h.ndim == 1 and len(h) == n
            and np.all(np.isfinite(h)) and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(float(np.sum(h)) - n / 2.0) <= 1e-9)

def exact_lags(h, n):
    vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if vals.shape != (2 * n - 1,) or not np.all(np.isfinite(vals)):
        return None
    return vals

def exact_score(h, n):
    vals = exact_lags(h, n)
    if vals is None:
        return None
    return float(np.max(vals))

def pair_line_search(h, vals, p, r, n, dx):
    hp, hr = h[p], h[r]
    L = max(-hp, hr - 1.0)
    U = min(1.0 - hp, hr)
    if U - L <= 1e-12:
        return None
    c = vals
    b = np.zeros(2 * n - 1)
    a = np.zeros(2 * n - 1)
    a[n - 1] = -2.0 * dx
    a[n - 1 + (r - p)] = dx
    a[n - 1 + (p - r)] = dx
    for d_idx in range(2 * n - 1):
        d = d_idx - (n - 1)
        bd = 0.0
        ip = p + d
        if 0 <= ip < n:
            bd += (1.0 - h[ip])
        ir = r + d
        if 0 <= ir < n:
            bd -= (1.0 - h[ir])
        ip2 = p - d
        if 0 <= ip2 < n:
            bd -= h[ip2]
        ir2 = r - d
        if 0 <= ir2 < n:
            bd += h[ir2]
        b[d_idx] = dx * bd
    cur_best = float(np.max(c))
    ts = np.linspace(L, U, 17)
    ts = np.unique(np.concatenate([ts, [0.0]]))
    best_t, best_f = 0.0, cur_best
    for t in ts:
        f = float(np.max(c + b * t + a * t * t))
        if f < best_f - 1e-14 or (abs(f - best_f) <= 1e-14 and abs(t) < abs(best_t)):
            best_f, best_t = f, t
    radius = (U - L) / 16.0
    for _ in range(4):
        lo_r = max(L, best_t - radius)
        hi_r = min(U, best_t + radius)
        if hi_r > lo_r:
            for t in np.linspace(lo_r, hi_r, 5):
                f = float(np.max(c + b * t + a * t * t))
                if f < best_f - 1e-14 or (abs(f - best_f) <= 1e-14 and abs(t) < abs(best_t)):
                    best_f, best_t = f, t
        radius /= 4.0
    if best_f >= cur_best - 1e-12:
        return None
    h_new = h.copy()
    h_new[p] += best_t
    h_new[r] -= best_t
    if h_new[p] < -1e-9 or h_new[p] > 1.0 + 1e-9 or h_new[r] < -1e-9 or h_new[r] > 1.0 + 1e-9:
        h_new = project_capped_simplex(h_new, n / 2.0)
        if h_new is None or not is_feasible(h_new, n):
            return None
    sc = exact_score(h_new, n)
    if sc is None or sc >= cur_best - 1e-12:
        return None
    vn = exact_lags(h_new, n)
    if vn is None:
        return None
    return h_new, sc, vn

def pairwise_polish(h, vals, score, n, dx, rng, max_sweeps):
    n_pairs = min(n * (n - 1) // 2, 2 * n, max(24, 80000 // max(n, 1)))
    pairs = []
    n_adj = min(n - 1, max(1, n_pairs // 2))
    for i in np.linspace(0, n - 2, n_adj).astype(int):
        pairs.append((int(i), int(i + 1)))
    tries = 0
    while len(pairs) < n_pairs and tries < n_pairs * 3:
        i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
        if i == j:
            tries += 1; continue
        if i > j: i, j = j, i
        if (i, j) not in pairs:
            pairs.append((i, j))
        tries += 1
    rng.shuffle(pairs)
    cur_h, cur_vals, cur_score = h, vals, score
    for _ in range(max_sweeps):
        any_move = False
        for (p, r) in pairs:
            res = pair_line_search(cur_h, cur_vals, p, r, n, dx)
            if res is not None:
                cur_h, cur_score, cur_vals = res
                any_move = True
        if not any_move:
            break
    return cur_h, cur_vals, cur_score

def anneal_trajectory(h0, n, dx, rng, epochs):
    S = n / 2.0
    h = h0.copy()
    score = exact_score(h, n)
    if score is None:
        return h0, 0.5
    scale = 0.12
    K = min(1000, max(80, 80000 // max(n, 1)))
    for e in range(epochs):
        T = max(5e-5, 3e-3 * (0.4 ** e))
        accepted = 0
        for _ in range(K):
            p = int(rng.integers(0, n))
            r = int(rng.integers(0, n))
            if p == r:
                continue
            t_t = float(rng.uniform(-scale, scale))
            hp, hr = h[p], h[r]
            L = max(-hp, hr - 1.0)
            U = min(1.0 - hp, hr)
            if t_t < L or t_t > U:
                continue
            h_new = h.copy()
            h_new[p] += t_t
            h_new[r] -= t_t
            if h_new[p] < -1e-9 or h_new[p] > 1.0 + 1e-9 or h_new[r] < -1e-9 or h_new[r] > 1.0 + 1e-9:
                h_new = project_capped_simplex(h_new, S)
                if h_new is None or not is_feasible(h_new, n):
                    continue
            sc = exact_score(h_new, n)
            if sc is None:
                continue
            delta = sc - score
            if delta <= 0:
