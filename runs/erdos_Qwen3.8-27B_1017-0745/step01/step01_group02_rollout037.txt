import numpy as np
import math

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.ndim != 1 or len(v) == 0 or not np.all(np.isfinite(v)):
        return None
    n = len(v)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(v - tau, 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-15:
        for i in np.argsort(1.0 - y)[::-1]:
            d = min(r, 1.0 - y[i])
            y[i] += d
            r -= d
            if r <= 0:
                break
    elif r < -1e-15:
        for i in np.argsort(y):
            d = min(-r, y[i])
            y[i] -= d
            r += d
            if r >= 0:
                break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-9) or np.any(y > 1.0 + 1e-9):
        return None
    if abs(float(np.sum(y)) - S) > 1e-10 * max(1, n):
        return None
    return y

def check_feasible(h, n):
    return (len(h) == n and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(float(np.sum(h)) - 0.5 * n) < 1e-10 * max(1, n))

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(corr)) or len(corr) != 2*n - 1:
        return float('inf')
    return float(np.max(corr) * (2.0 / n))

def anneal_cycle(start, n, rng):
    cur = start.copy()
    cur_sc = exact_score(cur, n)
    best_local = cur.copy()
    best_sc = cur_sc
    
    # Calibration
    sigma = 0.08
    cal_proposals = 24
    deltas = []
    cal_best = cur.copy()
    cal_best_sc = cur_sc
    for _ in range(cal_proposals):
        trial = propose_feasible(cur, n, sigma, rng)
        if trial is None:
            continue
        sc = exact_score(trial, n)
        if sc < cal_best_sc:
            cal_best = trial.copy()
            cal_best_sc = sc
        if sc > cur_sc:
            deltas.append(sc - cur_sc)
    if deltas:
        m = float(np.median(deltas))
        T0 = max(2e-5, min(2e-2, -m / math.log(0.8)))
    else:
        T0 = 1e-3
    
    # Annealing
    n_levels = 6
    props_per_level = 240
    t_mult = 0.55
    for lev in range(n_levels):
        T = max(1e-7, T0 * (t_mult ** lev))
        accepted = 0
        total = 0
        for p in range(props_per_level):
            trial = propose_feasible(cur, n, sigma, rng)
            total += 1
            if trial is None:
                continue
            sc = exact_score(trial, n)
            if not np.isfinite(sc):
                continue
            if sc < best_sc:
                best_local = trial.copy()
                best_sc = sc
            delta = sc - cur_sc
            if delta <= 0:
                cur = trial.copy()
                cur_sc = sc
                accepted += 1
            else:
                ratio = -delta / T
                if ratio < 745 and rng.random() < math.exp(-ratio):
                    cur = trial.copy()
                    cur_sc = sc

            if total % 50 == 0:
                r = accepted / total

        if rep is not None and check_feasible(rep, n) and float(np.max(np.abs(rep - cur))) < 1e-8:
            cur = rep
            cur_sc = exact_score(cur, n)
            if cur_sc < best_sc:
                best_local = cur.copy()
                best_sc = cur_sc
    best_sc = exact_score(best_local, n)
    return best_local, best_sc

def polish(h, n, rng):
    cur = h.copy()
    cur_sc = exact_score(cur, n)
    sigma = 0.015
    for lev in range(4):
        T = 5e-5 * (0.5 ** lev)
        for _ in range(200):
            trial = propose_feasible(cur, n, sigma, rng)
            if trial is None:
                continue
            sc = exact_score(trial, n)
            if not np.isfinite(sc):
                continue
            delta = sc - cur_sc
            if delta <= 0:
                cur = trial.copy()
                cur_sc = sc
            elif -delta / T < 745.0 and rng.random() < math.exp(-delta / T):
                cur = trial.copy()
                cur_sc = sc
    return cur, cur_sc

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    n = 45
    parent = None
    ph = globals().get('initial_h_values', None)
    if ph is not None:
        ph = np.asarray(ph, dtype=np.float64)
        if ph.ndim == 1 and 2 <= len(ph) <= 999 and np.all(np.isfinite(ph)):
            if np.all(ph >= -1e-8) and np.all(ph <= 1.0 + 1e-8):
                nn = len(ph)
                if abs(float(np.sum(ph)) - 0.5 * nn) < 1e-6 * max(1, nn):
                    n = nn
                    rep = project_box_sum(np.clip(ph, 0.0, 1.0), 0.5 * n)
                    if rep is not None and check_feasible(rep, n):
                        parent = rep
    if parent is None:
        n = 45
        parent = np.full(n, 0.5)
    best_h = parent.copy()
    best_c5 = exact_score(parent, n)
    starts = [parent.copy(), (1.0 - parent).copy(), parent[::-1].copy()]
    for sd in [0.05, 0.12, 0.20]:
        p = project_box_sum(parent + sd * rng.standard_normal(n), 0.5 * n)
        if p is not None and check_feasible(p, n):
            starts.append(p)
    p = project_box_sum(rng.random(n), 0.5 * n)
    if p is not None:
        starts.append(p)
    i_arr = np.arange(n)
    for _ in range(2):
        phi, theta = float(rng.uniform(0, 6.28318)), float(rng.uniform(0, 6.28318))
        u = 0.5 + 0.28 * np.cos(2.0 * np.pi * i_arr / n + phi) + 0.12 * np.sin(4.0 * np.pi * i_arr / n + theta)
        p = project_box_sum(u, 0.5 * n)
        if p is not None:
            starts.append(p)
    seen_sc = set()
    unique_starts = []
    for s in starts:
        if check_feasible(s, n):
            sc = exact_score(s, n)
            dup = False
            for us in unique_starts:
                if float(np.max(np.abs(s - us))) < 1e-10:
                    dup = True
                    break
            if not dup:
                unique_starts.append(s.copy())
                if sc < best_c5:
                    best_h = s.copy()
                    best_c5 = sc
    unique_starts.sort(key=lambda x: exact_score(x, n))
    n_cycles = 4
    restart_sds = [0.0, 0.0, 0.10, 0.22]
    for cyc in range(n_cycles):
        if cyc < len(unique_starts):
            start = unique_starts[cyc].copy()
        else:
            start = None
            for _ in range(4):
                start = project_box_sum(best_h + restart_sds[cyc] * rng.standard_normal(n), 0.5 * n)
                if start is not None and check_feasible(start, n):
                    break
            if start is None:
                start = best_h.copy()
        sc = exact_score(start, n)
        if sc < best_c5:
            best_h = start.copy()
            best_c5 = sc
        try:
            lb, lb_sc = anneal_cycle(start, n, rng)
            if lb_sc < best_c5:
                best_h = lb.copy()
                best_c5 = lb_sc
        except Exception:
            pass
    try:
        ph2, psc = polish(best_h, n, rng)
        if psc < best_c5:
            best_h, best_c5 = ph2, psc
    except Exception:
        pass
    best_h = project_box_sum(best_h, 0.5 * n)
    if best_h is not None and check_feasible(best_h, n):
        best_c5 = exact_score(best_h, n)
        if np.isfinite(best_c5):
            return (best_h, best_c5, n)
    fb = project_box_sum(parent, 0.5 * n)
    if fb is None:
        fb = np.full(n, 0.5)
    fb_sc = exact_score(fb, n)
    return (fb, fb_sc, n)
