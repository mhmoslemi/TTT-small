def exact_score(h, N):
    try:
        c = np.correlate(h, 1.0 - h, mode='full')
        if not np.all(np.isfinite(c)):
            return np.inf
        return float(np.max(c) * (2.0 / N))
    except Exception:
        return np.inf

def _fourier(N, rng, R, amp):
    idx = np.arange(N)
    z = np.full(N, 0.5)
    R = min(R, (N - 1) // 2)
    for r in range(1, R + 1):
        a, b = rng.standard_normal(2)
        z += (amp / r) * (a * np.sin(2*np.pi*r*idx/N) + b * np.cos(2*np.pi*r*idx/N))
    return z

def _novel(p, pop, m):
    if not is_feasible(p, len(p), m):
        return False
    return all(np.max(np.abs(p - q)) > 1e-10 for q in pop)

def build_pop(base, N, m, P, rng):
    pop = [base.copy()]
    cands = [project_box_sum(1.0 - base, m),
             project_box_sum(base[rng.permutation(N)], m)]
    for sd in (0.02, 0.06, 0.15, 0.30):
        cands.append(project_box_sum(base + sd * rng.standard_normal(N), m))
    cands.append(project_box_sum(rng.uniform(0, 1, N), m))
    cands.append(project_box_sum((rng.random(N) < 0.5).astype(float), m))
    cands.append(project_box_sum(_fourier(N, rng, 6, 0.12), m))
    cands.append(project_box_sum(_fourier(N, rng, 6, 0.24), m))
    for c in cands:
        if len(pop) < P and _novel(c, pop, m):
            pop.append(c)
    att = 0
    while len(pop) < P and att < 20*P:
        if att % 2 == 0:
            sd = 10 ** rng.uniform(np.log10(0.03), np.log10(0.35))
            c = project_box_sum(base + sd * rng.standard_normal(N), m)
        else:
            c = project_box_sum(rng.uniform(0, 1, N), m)
        att += 1
        if _novel(c, pop, m):
            pop.append(c)
    while len(pop) < P:
        pop.append(pop[int(rng.integers(len(pop)))].copy())
    return pop

def de_search(pop, N, m, P, max_gen, stag_lim, max_res, F, CR, rng):
    scores = np.array([exact_score(p, N) for p in pop])
    bi = int(np.argmin(scores))
    best_h, best_s = pop[bi].copy(), float(scores[bi])
    stag, nres = 0, 0
    for gen in range(max_gen):
        snap = [p.copy() for p in pop]
        ss = scores.copy()
        improved = False
        for i in range(P):
            others = np.delete(np.arange(P), i)
            idx3 = rng.choice(len(others), 3, replace=False)
            r1, r2, r3 = int(others[idx3[0]]), int(others[idx3[1]]), int(others[idx3[2]])
            v = snap[r1] + F * (snap[r2] - snap[r3])
            mask = rng.random(N) < CR
            mask[int(rng.integers(N))] = True
            u = np.where(mask, v, snap[i])
            if not np.all(np.isfinite(u)):
                continue
            u = project_box_sum(u, m)
            if not is_feasible(u, N, m):
                continue
            su = exact_score(u, N)
            if np.isfinite(su) and su < ss[i] - 1e-12:
                pop[i] = u
                scores[i] = su
                improved = True
                if su < best_s - 1e-12:
                    best_h, best_s = u.copy(), float(su)
        if improved:
            stag = 0
        else:
            stag += 1
        if stag >= stag_lim and nres < max_res:
            nres += 1
            sig = 0.14 if nres == 1 else 0.05
            k = max(1, P // 4)
            order = np.argsort(scores)
            npop = [best_h.copy()]
            for j in range(min(k-1, P)):
                npop.append(pop[int(order[j])].copy())
            rem = P - len(npop)
            nn = rem * 3 // 4
            for _ in range(nn):
                npop.append(project_box_sum(best_h + sig*rng.standard_normal(N), m))
            for _ in range(rem - nn):
                npop.append(project_box_sum(rng.uniform(0, 1, N), m))
            while len(npop) < P:
                npop.append(project_box_sum(best_h + sig*rng.standard_normal(N), m))
            pop = npop
            scores = np.array([exact_score(p, N) for p in pop])
            bi2 = int(np.argmin(scores))
            if scores[bi2] < best_s:
                best_h, best_s = pop[bi2].copy(), float(scores[bi2])
            stag = 0
        elif stag >= 2 * stag_lim and nres >= max_res:
            break
    return best_h, best_s

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    raw = globals().get('initial_h_values', None)
    N = m = None
    base = None
    if raw is not None:
        raw = np.asarray(raw, dtype=np.float64)
        if raw.ndim == 1 and raw.size > 1 and np.all(np.isfinite(raw)):
            N = raw.size
            m = N / 2.0
            base = project_box_sum(np.clip(raw, 0.0, 1.0), m)
    if N is None:
        N = kwargs.get('n_points', 70)
        if not isinstance(N, (int, np.integer)) or isinstance(N, bool) or N < 1 or N > 999:
            N = 70
        m = float(N) / 2.0
        base = np.full(N, 0.5)
    if N == 1:
        return base, exact_score(base, 1), 1
    bs = exact_score(base, N)
    if N > 999:
        return base, bs, int(N)
    if N <= 96:
        P, mg, sl, mr = 48, 120, 24, 2
    elif N <= 192:
        P, mg, sl, mr = 32, 80, 20, 1
    elif N <= 384:
        P, mg, sl, mr = 20, 40, 16, 1
    else:
        P, mg, sl, mr = 12, 20, 10, 0
    P = max(P, 4)
    try:
        pop = build_pop(base, N, m, P, rng)
        pop[0] = base
        sh, ss = de_search(pop, N, m, P, mg, sl, mr, 0.7, 0.85, rng)
