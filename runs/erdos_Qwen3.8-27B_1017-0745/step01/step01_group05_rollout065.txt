        lam = mid
    c = np.clip(y - lam, 0.0, 1.0)
    for _ in range(10):
        diff = S - c.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            m = (c < 1.0 - 1e-12)
        else:
            m = (c > 1e-12)
        cnt = m.sum()
        if cnt == 0:
            break
        c[m] += diff / cnt
        np.clip(c, 0.0, 1.0, out=c)
    return c

def check_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-10) and np.all(h <= 1.0 + 1e-10)
            and abs(h.sum() - n / 2.0) <= 1e-9)

def lag_scores(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def exact_score(h, n):
    if not check_feasible(h, n):
        return float('inf')
    qs = lag_scores(h, n)
    if not np.all(np.isfinite(qs)):
        return float('inf')
    return float(qs.max())

def build_G(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(-(n-1), n):
        lo = max(0, k)
        hi = min(n, n + k)
        if hi > lo:
            G[k + n - 1, lo:hi] += dx * (1.0 - h[lo - k:hi - k])
        lo2 = max(0, lo - k)
        hi2 = min(n, hi - k)
        if hi2 > lo2:
            G[k + n - 1, lo2:hi2] -= dx * h[lo2 + k:hi2 + k]
    return G

def project_simplex(z):
    m = len(z)
    if not np.all(np.isfinite(z)):
        return np.full(m, 1.0 / m)
    sz = np.sort(z)[::-1]
    cumsum = np.cumsum(sz)
    rho = np.max(np.where(sz - (cumsum - 1.0) / np.arange(1, m + 1) > 0)[0] + 1)
    theta = (cumsum[rho - 1] - 1.0) / rho
    return np.maximum(z - theta, 0.0)

def solve_majorant(h, q, G, n, max_inner):
    dx = 2.0 / n
    L = 2.0 * dx
    nm = 2 * n - 1
    G2 = float(np.sum(G * G))
    Ldual = max(1e-12, 1.5 * G2 / L)
    qmax = q.max()
    tol_q = max(1e-12, 1e-8 * max(1.0, abs(qmax)))
    near_max = q >= qmax - tol_q
    if near_max.sum() == 0:
        near_max = np.zeros(nm, dtype=bool)
        near_max[np.argmax(q)] = True
    p = np.zeros(nm)
    p[near_max] = 1.0 / near_max.sum()
    
    def dual_state(p_vec):
        gmix = G.T @ p_vec
        y = project_capped(h - gmix / L, n)
        d = y - h
        u = q + G @ d
        D_val = p_vec @ u + 0.5 * L * d @ d
        return d, u, D_val
    
    d, u, D_cur = dual_state(p)
    D_best = D_cur
    p_best = p.copy()
    x = p.copy()
    yv = p.copy()
    theta = 1.0
    stall = 0
    for it in range(max_inner):
        dy, uy, Dy = dual_state(yv)
        grad_y = uy  # q + G*d_y = uy
        x_trial = project_simplex(yv + grad_y / Ldual)
        dx_t, ux_t, Dx_t = dual_state(x_trial)
        if Dx_t < D_best - 1e-14:
            # restart
            x_trial = project_simplex(x + (q + G @ (project_capped(h - G.T @ x / L, n) - h)) / Ldual)
            dx_t2, ux_t2, Dx_t2 = dual_state(x_trial)
            if Dx_t2 < D_best - 1e-14:
                break
            x = x_trial
            yv = x.copy()
            theta = 1.0
        else:
            theta_new = (1.0 + math.sqrt(1.0 + 4.0 * theta * theta)) / 2.0
            yv = x_trial + ((theta - 1.0) / theta_new) * (x_trial - x)
            x = x_trial
            theta = theta_new
        d, u, D_cur = dual_state(x)
        if D_cur > D_best:
            D_best = D_cur
            p_best = x.copy()
        step = x_trial - x  # actually this is 0 since we already updated x
        # Check gap
        gap = u.max() - x @ u
        tol_gap = max(1e-10, 1e-8 * max(1.0, abs(u.max())))
        if gap <= tol_gap:
            break
        if np.max(np.abs(x_trial - p)) < 1e-12:
            stall += 1
            if stall >= 3:
                break
        else:
            stall = 0
    d_best, u_best, _ = dual_state(p_best)
    return d_best

def optimize_start(h0, n, max_outer, max_inner):
    h = project_capped(h0.copy(), n)
    s = exact_score(h, n)
    best_h = h.copy()
    best_s = s
    for outer in range(max_outer):
        q = lag_scores(h, n)
        G = build_G(h, n)
        if not np.all(np.isfinite(G)):
            break
