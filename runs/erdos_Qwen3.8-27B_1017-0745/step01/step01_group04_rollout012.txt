            C = np.eye(d)
            p_s = np.zeros(d)
            p_c = np.zeros(d)
            sigma = max(1e-4, sigma * 0.5)
    return best_h, best_score

def run(seed=42, **kwargs):
    raw = globals().get('initial_h_values', None)
    if raw is not None:
        try:
            raw = np.asarray(raw, dtype=np.float64)
        except Exception:
            raw = None
    if raw is not None and raw.ndim == 1 and raw.size > 1 and np.all(np.isfinite(raw)):
        N = int(raw.size)
        m = N / 2.0
    else:
        N = kwargs.get('n_points', 70)
        if not isinstance(N, (int, np.integer)) or isinstance(N, bool) or N < 1:
            N = 70
        N = min(int(N), 999)
        m = N / 2.0
    if N <= 1:
        h = np.array([0.5])
        return h, 0.0, 1
    if raw is not None and raw.ndim == 1 and raw.size == N and np.all(np.isfinite(raw)):
        baseline = project_box_sum(raw.copy(), m)
    else:
        baseline = np.full(N, 0.5)
    best_h = baseline.copy()
    best_score = score_exact(best_h, N)
    if N > 256:
        return best_h, best_score, N
    d = N - 1
    u = 0.5 * np.ones(N)
    B = helmert_matrix(N)
    if 2 <= N <= 96:
        n_runs, max_gen, stag_limit, sigmas = 3, 100, 24, [0.12, 0.07, 0.035]
    elif 97 <= N <= 160:
        n_runs, max_gen, stag_limit, sigmas = 2, 60, 18, [0.10, 0.05]
    else:
        n_runs, max_gen, stag_limit, sigmas = 1, 30, 12, [0.08]
    lambda_pop = max(4, 4 + int(3 * math.log(d)))
    mu_sel = max(1, lambda_pop // 2)
    w_raw = np.array([math.log(mu_sel + 0.5) - math.log(i + 1) for i in range(mu_sel)])
    w_raw = np.maximum(w_raw, 1e-10)
    weights = w_raw / w_raw.sum()
    mu_eff = 1.0 / np.sum(weights ** 2)
    c_sigma = (mu_eff + 2) / (d + mu_eff + 5)
    d_sigma = 1 + 2 * max(0.0, math.sqrt((mu_eff - 1) / (d + 1)) - 1) + c_sigma
    c_c = (4 + mu_eff / d) / (d + 4 + 2 * mu_eff / d)
    c1 = 2.0 / ((d + 1.3) ** 2 + mu_eff)
    c_mu = min(1 - c1, 2 * (mu_eff - 2 + 1.0 / mu_eff) / ((d + 2) ** 2 + mu_eff))
    chi_d = math.sqrt(d) * (1 - 1.0 / (4 * d) + 1.0 / (21 * d * d))
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    for r in range(n_runs):
        try:
            mu = B.T @ (best_h - u)
            C = np.eye(d)
            p_s = np.zeros(d)
            p_c = np.zeros(d)
            best_h, best_score = cma_single_run(
                mu, C, sigmas[r], p_s, p_c, B, u, m, rng,
                lambda_pop, mu_sel, weights, max_gen, stag_limit,
                c_sigma, d_sigma, c_c, c1, c_mu, chi_d, best_h, best_score, N)
        except Exception:
            pass
    final = project_box_sum(best_h.copy(), m)
    final_score = score_exact(final, N)
    base_score = score_exact(baseline, N)
    if np.isfinite(final_score) and final_score < base_score - 1e-12:
        return final, float(final_score), N
    if np.isfinite(base_score):
        return baseline, float(base_score), N
    h = np.full(N, 0.5)
    return h, float(score_exact(h, N)), N
