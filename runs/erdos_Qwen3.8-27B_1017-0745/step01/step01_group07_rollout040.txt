from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix, eye as speye, hstack, vstack

def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):
        return None
    if np.max(np.abs(z)) > 1e12:
        return None
    n = len(z)
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(80):
        mid = (lo + hi) / 2
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(z - (lo + hi) / 2, 0.0, 1.0)
    r = S - np.sum(y)
    step = 1 if r > 0 else -1
    r = abs(r)
    rng_idx = range(n) if step > 0 else range(n - 1, -1, -1)
    for i in rng_idx:
        if r <= 1e-15:
            break
        a = min(r, 1.0 - y[i]) if step > 0 else min(r, y[i])
        y[i] += step * a
        r -= a
    if abs(np.sum(y) - S) > 1e-10 or not np.all(np.isfinite(y)):
        return None
    return y

    return (h.ndim == 1 and len(h) == n and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-10)

def exact_score(h, n):
    try:
        vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if vals.size != 2 * n - 1 or not np.all(np.isfinite(vals)):
            return None
        return float(np.max(vals))
    except Exception:
        return None

def exact_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def build_lag_model(h, n):
    dx = 2.0 / n
    q = exact_lags(h, n)
    m = 2 * n - 1
    if not np.all(np.isfinite(q)):
        return None, None
    total = n * (n + 1)
    rows = np.zeros(total, dtype=np.int32)
    cols = np.zeros(total, dtype=np.int32)
    data = np.zeros(total, dtype=np.float64)
    idx = 0
    for d in range(-(n - 1), n):
        row = d + (n - 1)
        lo = max(0, -d)
        hi = min(n, n - d)
        for i in range(lo, hi):
            rows[idx] = row; cols[idx] = i; data[idx] = dx * (1.0 - h[i + d]); idx += 1
            rows[idx] = row; cols[idx] = i + d; data[idx] = -dx * h[i]; idx += 1
    G = coo_matrix((data, (rows, cols)), shape=(m, n)).tocsr()
    return q, G

def solve_lp_step(x, q, G, rho, n, mu):
    m = 2 * n - 1
    # Build A_ub sparsely
    # B1: [G | 0(m,n) | -1(m,1)] ≤ -q
    # B2: [I | -I | 0(n,1)] ≤ 0
    # B3: [-I | -I | 0(n,1)] ≤ 0
    I_n = speye(n, dtype=np.float64)
    zero_mn = csr_matrix((m, n))
    zero_m1 = csr_matrix((m, 1))
    neg_m1 = csr_matrix((m, 1))
    neg_m1[range(m), 0] = -1.0
    B1 = hstack([G, zero_mn, neg_m1])
    B2 = hstack([I_n, -I_n, zero_m1])
    B3 = hstack([-I_n, -I_n, zero_m1])
    A_ub = vstack([B1, B2, B3])
    b_ub = np.concatenate([-q, np.zeros(2 * n)])
    
    c = np.zeros(2 * n + 1)
    c[n:2*n] = mu
    c[2*n] = 1.0
    
    A_eq = np.zeros((1, 2 * n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    lb = np.empty(2 * n + 1)
    ub = np.empty(2 * n + 1)
    for i in range(n):
        lb[i] = max(-rho, -x[i] + 1e-12)
        ub[i] = min(rho, 1.0 - x[i] - 1e-12)
    lb[n:2*n] = 0.0
    ub[n:2*n] = np.inf
    lb[2*n] = -np.inf
    ub[2*n] = np.inf
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

                  options={'presolve': True, 'max_iter': 5000})

    delta = np.asarray(res.x[:n], dtype=np.float64)
    if not np.all(np.isfinite(delta)) or abs(np.sum(delta)) > 1e-5:
        return None
    return delta

def optimize_trajectory(x0, n, rng, best_h, best_score):

    S = n / 2.0
    mu = 0.002 * dx
    x = x0.copy()
    f = exact_score(x, n)
    if f is None:
        return best_h, best_score
    rho = 0.08
    rho_max = 0.25
    rho_min = 2e-4
    fails = 0
    max_iter = 30 if n <= 128 else (22 if n <= 256 else 14)
    accept_streak = 0
    small_act_streak = 0
    
    for it in range(max_iter):
        if fails >= 4 or rho < rho_min:
            break
        q, G = build_lag_model(x, n)
        if q is None:
            break
        delta = solve_lp_step(x, q, G, rho, n, mu)
        if delta is None:
            rho = max(rho_min, rho / 2)
            fails += 1
            continue
        if np.max(np.abs(delta)) < 1e-10:
            break
        # Line search: try 9 step lengths
        best_cand = None
        best_cand_score = f
        for j in range(9):
            beta = 2.0 ** (-j)
            x_trial = x + beta * delta
            x_p = project_capped_simplex(x_trial, S)
            if x_p is None or not is_feasible(x_p, n):
                continue
            s = exact_score(x_p, n)
            if s is not None and s < best_cand_score - 1e-12:

            break
        if best_cand_score < best_score - 1e-12:

            best_score = best_cand_score
