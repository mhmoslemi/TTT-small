    n = _mm_n; y = z[:n]; d = y - _mm_x
    return 0.5 * _mm_L * np.dot(d, d) + z[n]

def mm_obj_grad(z):
    n = _mm_n; d = z[:n] - _mm_x
    g = np.zeros(n + 1); g[:n] = _mm_L * d; g[n] = 1.0
    return g

def mm_ineq(z):
    n = _mm_n; d = z[:n] - _mm_x
    return z[n] - _mm_qJ - _mm_GJ @ d

def mm_ineq_grad(z):
    n = _mm_n; m = _mm_qJ.shape[0]
    J = np.zeros((m, n + 1)); J[:, :n] = -_mm_GJ; J[:, n] = 1.0
    return J

def mm_eq(z):
    return float(np.sum(z[:_mm_n])) - 0.5 * _mm_n

def mm_eq_grad(z):
    g = np.zeros(_mm_n + 1); g[:_mm_n] = 1.0
    return g

def solve_mm(x, n):
    global _mm_x, _mm_L, _mm_qJ, _mm_GJ, _mm_n
    dx = 2.0 / n; L = 2.0 * dx
    _mm_n = n; _mm_x = x.copy(); _mm_L = L
    q_all = all_lag_vals(x, n)
    lags = list(range(2 * n - 1)) if n <= 80 else list(np.argsort(-q_all)[:96])
    G = np.zeros((len(lags), n)); qJ = np.zeros(len(lags))
    for k, idx in enumerate(lags):
        s = -(n - 1) + idx
        q, g = lag_q_grad(x, s, n); qJ[k] = q; G[k] = g
    _mm_qJ = qJ; _mm_GJ = G
    bounds = [(0.0, 1.0)] * n + [(-1.0, 2.0)]
    cons = [{'type': 'ineq', 'fun': mm_ineq, 'jac': mm_ineq_grad},
            {'type': 'eq', 'fun': mm_eq, 'jac': mm_eq_grad}]
    z0 = np.append(x.copy(), float(np.max(qJ)) + 1e-6)
    y_prop = x.copy()
    max_solves = 12 if n <= 80 else 8
    max_cuts = 2 * n - 1 if n <= 80 else 96
    for it in range(max_solves):
        _mm_qJ = qJ; _mm_GJ = G
        try:
            res = minimize(mm_obj, z0, jac=mm_obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 150, 'ftol': 1e-10, 'disp': False})
            y_raw = res.x[:n]; t_val = res.x[n]
        except Exception:
            break
        if not np.all(np.isfinite(y_raw)):
            break
        y_proj = project_box_sum(y_raw, 0.5 * n)
        if y_proj is None:
            break
        d = y_proj - x
        active_val = qJ + G @ d
        if np.min(active_val) - t_val < -1e-4:
            break
        sep_val, sep_idx = separation(x, d, n)
        if sep_val <= float(np.max(active_val)) + 1e-8:
            y_prop = y_proj
            break
        if len(lags) < max_cuts:
            s_new = -(n - 1) + sep_idx
            q_new, g_new = lag_q_grad(x, s_new, n)
            lags.append(sep_idx); qJ = np.append(qJ, q_new); G = np.vstack([G, g_new])
        z0 = np.append(y_proj, max(float(np.max(qJ + G @ d)), t_val))
    return y_prop

def mm_descent(h0, n, max_steps=None):
    if max_steps is None:
        max_steps = max(8, min(40, 1800 // n))
    h = h0.copy(); c = exact_score(h, n)
    if not np.isfinite(c):
        return h, float('inf')
    for _ in range(max_steps):
        y_prop = solve_mm(h, n)
        if not np.all(np.isfinite(y_prop)):
            break
        d = y_prop - h
        if np.max(np.abs(d)) < 1e-10:
            break
        best_c = c; found = False
        n_alk = 10 if n <= 128 else 6
        for k in range(n_alk):
            alpha = 2.0 ** (-k)
            h_trial = project_box_sum(h + alpha * d, 0.5 * n)
            if h_trial is None:
                continue
            c_trial = exact_score(h_trial, n)
            if c_trial < best_c - 1e-12:
                best_c = c_trial; h = h_trial.copy(); found = True
            if alpha < 0.01:
                break
        if not found:
            break
        c = best_c
        if c < 0.3804:
            break
    return h, c

def get_parent():
    n = 45
    if 'initial_h_values' in globals():
        ph = np.asarray(globals()['initial_h_values'], dtype=np.float64)
        if ph.ndim == 1 and 2 <= len(ph) <= 999 and np.all(np.isfinite(ph)):
            if np.all(ph >= -1e-6) and np.all(ph <= 1.0 + 1e-6):
                n = len(ph)
                if check_feasible(ph, n):
                    return ph.copy(), n
                p = project_box_sum(np.clip(ph, 0.0, 1.0), 0.5 * n)
                if p is not None:
                    return p, n
    return np.full(n, 0.5), n

def build_starts(parent, n, rng):
    starts = [parent.copy(), parent[::-1].copy(), (1.0 - parent).copy()]
    for sd in [0.06, 0.20, 0.12]:
        p = project_box_sum(parent + sd * rng.standard_normal(n), 0.5 * n)
        if p is not None:
            starts.append(p)
    p = project_box_sum(rng.random(n), 0.5 * n)
    if p is not None:
        starts.append(p)
    u = 0.5 + 0.30 * np.cos(2.0 * np.pi * np.arange(n) / n + 0.7) + 0.12 * np.sin(4.0 * np.pi * np.arange(n) / n + 1.3)
    p = project_box_sum(u, 0.5 * n)
    if p is not None:
        starts.append(p)
    unique = [starts[0]]
    for s in starts[1:]:
        dup = False
        for u2 in unique:
            if np.max(np.abs(s - u2)) < 1e-10:
                dup = True; break
        if not dup:
            unique.append(s)
    return unique

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent, n = get_parent()
    best_h = parent.copy()
    best_c5 = exact_score(best_h, n)
    parent_fallback = parent.copy()
    parent_c5 = best_c5
    starts = build_starts(parent, n, rng)
    for h0 in starts:
        try:
            h_new, c_new = mm_descent(h0, n)
            if np.isfinite(c_new) and c_new < best_c5:
                best_h = h_new.copy(); best_c5 = c_new
        except Exception:
            pass
    # Perturbation restarts from best
    sd_list = [0.04, 0.12, 0.28] if n <= 80 else [0.08, 0.22]
    for sd in sd_list:
        try:
            h_pert = project_box_sum(best_h + sd * rng.standard_normal(n), 0.5 * n)
            if h_pert is None:
                continue
            h_new, c_new = mm_descent(h_pert, n, max_steps=max(4, 200 // n))
            if np.isfinite(c_new) and c_new < best_c5:
                best_h = h_new.copy(); best_c5 = c_new
        except Exception:
            pass
    # Final validation
    final_h = project_box_sum(best_h, 0.5 * n)
    final_c5 = exact_score(final_h, n) if final_h is not None else float('inf')
    pb = project_box_sum(parent_fallback, 0.5 * n)
    pc5 = exact_score(pb, n) if pb is not None else float('inf')
    if np.isfinite(pc5) and pc5 < final_c5:
        final_h, final_c5 = pb, pc5
    if not np.isfinite(final_c5):
        fb = np.full(n, 0.5)
        final_c5 = exact_score(fb, n)
        final_h = fb
    return (final_h, final_c5, int(len(final_h)))
