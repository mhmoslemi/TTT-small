import numpy as np
import math

def project_box_sum(z, m):
    N = len(z)
    if not np.all(np.isfinite(z)) or np.max(np.abs(z)) > 1e100:
        return np.full(N, 0.5)
    lo, hi = z.min() - 1.0, z.max()
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.clip(z - mid, 0, 1).sum() > m:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = m - h.sum()
    if r > 1e-14:
        for i in range(N):
            a = min(r, 1.0 - h[i]); h[i] += a; r -= a
            if r <= 1e-14: break
    elif r < -1e-14:
        for i in range(N):
            a = min(-r, h[i]); h[i] -= a; r += a
            if r >= -1e-14: break
    if not np.all(np.isfinite(h)) or h.min() < -1e-10 or h.max() > 1 + 1e-10:
        return np.full(N, 0.5)
    if abs(h.sum() - m) > 1e-7 * max(1, N):
        return np.full(N, 0.5)
    return np.clip(h, 0.0, 1.0)

def score_exact(h, N):
    if h.shape != (N,) or not np.all(np.isfinite(h)):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c) * 2.0 / N)

def helmert_basis(N):
    d = N - 1
    B = np.zeros((N, d))
    for k in range(d):
        a = math.sqrt((k + 1) * (k + 2))
        B[:k+1, k] = 1.0 / a
        B[k+1, k] = -(k + 1) / a
    return B

def cma_params(d, lam, mu_sel):
    w = np.array([math.log(mu_sel + 0.5) - math.log(i) for i in range(1, mu_sel + 1)])
    w = w / w.sum()
    mu_eff = 1.0 / np.dot(w, w)
    c_sig = (mu_eff + 2) / (d + mu_eff + 5)
    d_sig = 1 + 2 * max(0, math.sqrt((mu_eff - 1) / (d + 1)) - 1) + c_sig
    c_c = (4 + mu_eff / d) / (d + 4 + 2 * mu_eff / d)
    c1 = 2.0 / ((d + 1.3) ** 2 + mu_eff)
    c_mu = min(1 - c1, 2 * (mu_eff - 2 + 1 / mu_eff) / ((d + 2) ** 2 + mu_eff))
    chi_d = math.sqrt(d) * (1 - 1.0 / (4 * d) + 1.0 / (21 * d * d))
    return w, mu_eff, c_sig, d_sig, c_c, c1, c_mu, chi_d

def cma_one_run(mu0, sigma, B, u, N, m, rng, lam, mu_sel, w, chi_d,
                c_sig, d_sig, c_c, c1, c_mu, max_gen, stag_lim):
    d = N - 1
    C = np.eye(d)
    p_sig = np.zeros(d)
    p_c = np.zeros(d)
    mu = mu0.copy()
    best_x = mu0.copy()
    best_s = score_exact(u + B @ mu0, N)
    stag = 0
    for gen in range(max_gen):
        Cs = 0.5 * (C + C.T)
        try:
            ev, evec = np.linalg.eigh(Cs)
        except Exception:
            break
        ev = np.clip(ev, max(1e-12, ev.max() * 1e-10), 1e4)
        Q = evec
        D = ev
        sqrtD = np.sqrt(D)
        inv_sqrtD = 1.0 / sqrtD
        samples = np.zeros((lam, d))
        scores = np.zeros(lam)
        xs = np.zeros((lam, d))
        for j in range(lam):
            z = rng.standard_normal(d)
            x_raw = mu + sigma * (Q * sqrtD) @ z
            if not np.all(np.isfinite(x_raw)):
                x_raw = mu.copy()
            z_x = u + B @ x_raw
            h_cand = project_box_sum(z_x, m)
            s_cand = score_exact(h_cand, N)
            if not np.isfinite(s_cand) or h_cand.min() < -1e-12 or h_cand.max() > 1 + 1e-12:
                h_cand = u + B @ mu
                s_cand = best_s if np.isfinite(best_s) else 1.0
            xs[j] = B.T @ (h_cand - u)
            samples[j] = h_cand
            scores[j] = s_cand
            if s_cand < best_s - 1e-14:
                best_s = s_cand
                best_x = xs[j].copy()
        order = np.argsort(scores, kind='stable')
        sel = order[:mu_sel]
        y_sel = (xs[sel] - mu) / sigma
        mu_new = mu + sigma * np.dot(w, (xs[sel] - mu) / sigma)
        mu_new = B.T @ (project_box_sum(u + B @ mu_new, m) - u)
        y_w = (mu_new - mu) / sigma
        h_mean = project_box_sum(u + B @ mu_new, m)
        s_mean = score_exact(h_mean, N)
        if s_mean < best_s - 1e-14:
            best_s = s_mean
            best_x = B.T @ (h_mean - u)
        Cs_hinv = (Q * inv_sqrtD) @ Q.T
        p_sig = (1 - c_sig) * p_sig + math.sqrt(c_sig * (2 - c_sig) / np.dot(w, w)) * Cs_hinv @ y_w
        t = gen + 1
        h_sig = 1.0 if np.linalg.norm(p_sig) / math.sqrt(1 - (1 - c_sig) ** (2 * t)) < 1.5 * chi_d else 0.0
        p_c = (1 - c_c) * p_c + h_sig * math.sqrt(c_c * (2 - c_c) / np.dot(w, w)) * y_w
        C = (1 - c1 - c_mu) * Cs

        for ii in range(mu_sel):

        C = 0.5 * (C + C.T)
        ln_sig = math.exp(min(1.0, max(-1.0, c_sig / d_sig * (np.linalg.norm(p_sig) / chi_d - 1))))
        sigma = min(0.5, max(1e-4, sigma * ln_sig))
        mu = mu_new

            stag = 0
        else:
            stag += 1
        if stag >= stag_lim or sigma <= 1.01e-4:
            break
    return u + B @ best_x, best_s
