            sigma = max(1e-4, sigma * 0.5)
            continue
        emax = max(eigvals.max(), 1e-12)
        eigvals = np.clip(eigvals, max(1e-12, emax * 1e-10), 1e4)
        C_sym = (Q * eigvals) @ Q.T
        D_sqrt = np.sqrt(eigvals)
        C_inv_sqrt = (Q * (1.0 / D_sqrt)) @ Q.T
        xs = np.zeros((lam, d))
        scs = np.full(lam, np.inf)
        for j in range(lam):
            z = rng.standard_normal(d)
            x_raw = mu + sigma * ((Q * D_sqrt) @ z)
            if not np.all(np.isfinite(x_raw)):
                h_j = project_box_sum(u + B @ mu, m)
            else:
                h_j = project_box_sum(u + B @ x_raw, m)
            xs[j] = B.T @ (h_j - u)
            scs[j] = score_exact(h_j, N)
        for j in range(lam):
            if scs[j] < best_sc - 1e-12:
                best_sc = scs[j]
                best_h = project_box_sum(u + B @ xs[j], m)
        order = np.argsort(scs, kind='stable')
        sel = order[:mus]
        mu_new = np.sum(w[:, None] * xs[sel], axis=0)
        h_m = project_box_sum(u + B @ mu_new, m)
        mu_new = B.T @ (h_m - u)
        sc_m = score_exact(h_m, N)
        if sc_m < best_sc - 1e-12:
            best_sc = sc_m; best_h = h_m.copy()
        y_w = (mu_new - mu) / sigma if sigma > 1e-15 else np.zeros(d)
        ps = (1 - cs) * ps + math.sqrt(cs * (2 - cs) * mueff) * (C_inv_sqrt @ y_w)
        t = gen + 1
        hsig = 1.0 if np.linalg.norm(ps) / math.sqrt(max(1e-30, 1 - (1 - cs) ** (2 * t))) < (1.4 + 2 / (d + 1)) * chid else 0.0
        pc = (1 - cc) * pc + hsig * math.sqrt(cc * (2 - cc) * mueff) * y_w
        C_new = (1 - c1 - cmu) * C_sym
        C_new += c1 * (np.outer(pc, pc) + (1 - hsig) * cc * (2 - cc) * C_sym)
        for i in range(mus):
            yi = (xs[sel[i]] - mu) / sigma if sigma > 1e-15 else np.zeros(d)
            C_new += cmu * w[i] * np.outer(yi, yi)
        C_new = 0.5 * (C_new + C_new.T)
        if not np.all(np.isfinite(C_new)):
            C = np.eye(d); ps[:] = 0; pc[:] = 0
            sigma = max(1e-4, sigma * 0.5)
            continue
        C = C_new
        ev = max(-1.0, min(1.0, (cs / ds) * (np.linalg.norm(ps) / chid - 1)))
        sigma = max(1e-4, min(0.5, sigma * math.exp(ev)))
        mu = mu_new
        if best_sc < prev_best - 1e-12:
            stag = 0
        else:
            stag += 1
        if stag >= stag_lim or sigma <= 1e-4:
            break
    return best_h, best_sc

def run(seed=42, **kwargs):
    raw = globals().get('initial_h_values', None)
    N = None
    if raw is not None:
        raw = np.asarray(raw, dtype=np.float64)
        if raw.ndim == 1 and raw.size > 1 and np.all(np.isfinite(raw)):
            N = raw.size
    if N is None:
        N = kwargs.get('n_points', 70)
        if not isinstance(N, (int, np.integer)) or isinstance(N, bool) or N < 2:
            N = 70
    N = int(N)
    m = N / 2.0
    if N == 1:
        h = np.array([0.5])
        return h, score_exact(h, 1), 1
    if raw is not None and np.all(np.isfinite(raw)):
        best = project_box_sum(raw, m)
    else:
        best = np.full(N, 0.5)
    baseline = best.copy()
    best_score = score_exact(best, N)
    if N > 256:
        final = project_box_sum(best, m)
        return final, score_exact(final, N), N
    if N <= 96:
        sigmas = [0.12, 0.07, 0.035]
        max_gen, stag_lim = 100, 24
    elif N <= 160:
        sigmas = [0.10, 0.05]
        max_gen, stag_lim = 60, 18
    else:
        sigmas = [0.08]
        max_gen, stag_lim = 30, 12
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    B = build_helmert(N)
    u = 0.5 * np.ones(N)
    for sig0 in sigmas:
        try:
            mu0 = B.T @ (best - u)
            h_c, s_c = cma_run(mu0, sig0, B, N, m, rng, max_gen, stag_lim)
            if s_c < best_score - 1e-12:
                best = h_c.copy()
                best_score = s_c
        except Exception:
            pass
    # Final validation
    final = project_box_sum(best, m)
    c5_best = score_exact(final, N)
    final_b = project_box_sum(baseline, m)
    c5_base = score_exact(final_b, N)
    if c5_best < c5_base - 1e-12 and np.isfinite(c5_best):
        best = final
        c5 = c5_best
    else:
        best = final_b
        c5 = c5_base
    if not np.isfinite(c5):
        best = np.full(N, 0.5)
        c5 = score_exact(best, N)
    best = project_box_sum(best, m)
    c5 = score_exact(best, N)
    return best, c5, N
