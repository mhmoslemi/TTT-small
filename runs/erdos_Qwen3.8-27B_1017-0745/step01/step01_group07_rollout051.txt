import numpy as np

def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):
        return None
    if np.max(np.abs(z)) > 1e12:
        return None
    n = len(z)
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    y = np.clip(z - lam, 0.0, 1.0)
    r = S - np.sum(y)
    if r > 0:
        for i in range(n):
            if r <= 0:
                break
            y[i] += min(r, 1.0 - y[i]); r -= min(r, 1.0 - y[i])
    elif r < 0:
        for i in range(n):
            if r >= 0:
                break
            y[i] -= min(-r, y[i]); r += min(-r, y[i])
    return y if (abs(np.sum(y) - S) <= 1e-10 and np.all(np.isfinite(y))) else None

            and np.all(np.isfinite(h)) and np.all(h >= 0) and np.all(h <= 1)
            and abs(np.sum(h) - n * 0.5) <= 1e-9)

def exact_score(h, n):
    try:
        vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if vals.shape[0] != 2 * n - 1 or not np.all(np.isfinite(vals)):
            return None, None
        return float(np.max(vals)), vals
    except Exception:
        return None, None

def pair_line_search(h, q, p, r, n):

    L = max(-h[p], h[r] - 1.0)
    U = min(1.0 - h[p], h[r])
    if U - L < 1e-12:
        return None
    nc = 2 * n - 1
    c = q.copy()
    a = np.zeros(nc)
    mid = n - 1
    a[mid] = -2.0 * dx
    dr = r - p
    if abs(dr) > 0:
        a[mid + dr] = dx
        a[mid - dr] = dx
    b = np.zeros(nc)
    for d_idx in range(nc):
        d = d_idx - mid
        bv = 0.0
        if 0 <= p + d < n:
            bv += (1.0 - h[p + d])
        if 0 <= r + d < n:
            bv -= (1.0 - h[r + d])
        if 0 <= p - d < n:
            bv -= h[p - d]
        if 0 <= r - d < n:
            bv += h[r - d]
        b[d_idx] = dx * bv
    # Line search: minimize max_i(c_i + b_i*t + a_i*t^2) over t in [L,U]
    t_best, f_best = 0.0, np.max(c)
    # Coarse grid
    for k in range(17):
        t = L + k * (U - L) / 16.0
        f = float(np.max(c + b * t + a * t * t))
        if f < f_best or (f == f_best and abs(t) < abs(t_best)):
            f_best, t_best = f, t
    # Refine
    radius = (U - L) / 16.0
    for _ in range(4):
        lo_t = max(L, t_best - radius)
        hi_t = min(U, t_best + radius)
        for k in range(5):
            t = lo_t + k * (hi_t - lo_t) / 4.0
            f = float(np.max(c + b * t + a * t * t))
            if f < f_best or (f == f_best and abs(t) < abs(t_best)):
                f_best, t_best = f, t
        radius /= 4.0
    if f_best >= float(np.max(c)) - 1e-12:

    if not is_feasible(h_new, n):

        if h_new is None:

    sc, q_new = exact_score(h_new, n)
    if sc is None or sc >= float(np.max(c)) - 1e-12:

    return h_new, sc, q_new

def polish(h, q, score, n, rng, nsweeps):
    dx = 2.0 / n
    nmoves = min(n * (n - 1) // 2, 2 * n, max(24, 80000 // n))
    for sweep in range(nsweeps):
        accepts = 0
        pairs = []
        # Adjacent pairs
        nadj = min(n - 1, max(1, nmoves // 2))
        for i in range(nadj):
            pairs.append((i, i + 1))
        # Random pairs
        while len(pairs) < nmoves:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i < j:
                pairs.append((i, j))
        rng.shuffle(pairs)
        for (p, r) in pairs:
            res = pair_line_search(h, q, p, r, n)
            if res is not None:
                h, score, q = res
                accepts += 1
        if accepts == 0:

    return h, score, q

def anneal_trajectory(h, q, score, n, rng, n_epochs):
    K = min(1000, max(80, 80000 // n))
    scale = 0.12
    for e in range(n_epochs):
        T = max(5e-5, 3e-3 * (0.4 ** e))
        accepted = 0
        for _ in range(K):
            p = int(rng.integers(0, n))
            r = int(rng.integers(0, n))
            if p == r:

            L = max(-h[p], h[r] - 1.0)
            U = min(1.0 - h[p], h[r])
            if U - L < 1e-12:

            t = float(rng.uniform(-scale, scale))
            if t < L or t > U:

            h_new = h.copy()
            h_new[p] += t
            h_new[r] -= t
            if not is_feasible(h_new, n):
                h_new = project_capped_simplex(h_new, n * 0.5)
                if h_new is None:

            sc, _ = exact_score(h_new, n)
            if sc is None:

            delta = sc - score
            if delta <= 0:
                h, score, q = h_new, sc, None
                if q is None:
                    _, q = exact_score(h, n)

            else:
                x = -delta / T
                if x > -700 and rng.random() < np.exp(x):

        elif ratio > 0.55:

import numpy as np

def project_capped_simplex(z, S):
    z = np.asarray(z, dtype=np.float64)
    if z.ndim != 1 or not np.all(np.isfinite(z)) or S < 0 or S > len(z):
        return None
    if float(np.max(np.abs(z))) > 1e12:
        return None
    n = len(z)
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    y = np.clip(z - lam, 0.0, 1.0)
    r = S - np.sum(y)
    if r > 0:
        for i in range(n):
            if r <= 0: break
            y[i] += min(r, 1.0 - y[i]); r -= min(r, 1.0 - y[i])
    elif r < 0:
        for i in range(n):
            if r >= 0: break
            y[i] -= min(-r, y[i]); r += min(-r, y[i])
    return y if (abs(np.sum(y) - S) <= 1e-10 and np.all(np.isfinite(y))) else None

def is_feasible(h, n):
    return (isinstance(h, np.ndarray) and h.ndim == 1 and len(h) == n
            and np.all(np.isfinite(h)) and np.all(h >= 0) and np.all(h <= 1)
            and abs(np.sum(h) - n * 0.5) <= 1e-9)

def exact_score(h, n):
    try:
        vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if vals.shape[0] != 2 * n - 1 or not np.all(np.isfinite(vals)):
            return None, None
        return float(np.max(vals)), vals
    except Exception:
        return None, None

def pair_line_search(h, q, p, r, n):
    dx = 2.0 / n
    L = max(-h[p], h[r] - 1.0)
    U = min(1.0 - h[p], h[r])
    if U - L < 1e-12:
        return None
    nc = 2 * n - 1
    mid = n - 1
    a = np.zeros(nc)
    a[mid] = -2.0 * dx
    dr = r - p
    a[mid + dr] = dx
    a[mid - dr] = dx
    b = np.zeros(nc)
    for d_idx in range(nc):
        d = d_idx - mid
        bv = 0.0
        if 0 <= p + d < n: bv += 1.0 - h[p + d]
        if 0 <= r + d < n: bv -= 1.0 - h[r + d]
        if 0 <= p - d < n: bv -= h[p - d]
        if 0 <= r - d < n: bv += h[r - d]
        b[d_idx] = dx * bv
    t_best, f_best = 0.0, float(np.max(q))
    for k in range(17):
        t = L + k * (U - L) / 16.0
        f = float(np.max(q + b * t + a * t * t))
        if f < f_best: f_best, t_best = f, t
    radius = (U - L) / 16.0
    for _ in range(4):
        lo_t, hi_t = max(L, t_best - radius), min(U, t_best + radius)
        for k in range(5):
            t = lo_t + k * (hi_t - lo_t) / 4.0
            f = float(np.max(q + b * t + a * t * t))
            if f < f_best: f_best, t_best = f, t
        radius /= 4.0
    if f_best >= float(np.max(q)) - 1e-12:
        return None
    h_new = h.copy()
    h_new[p] += t_best
    h_new[r] -= t_best
    if not is_feasible(h_new, n):
        h_new = project_capped_simplex(h_new, n * 0.5)
        if h_new is None: return None
    sc, q_new = exact_score(h_new, n)
    if sc is None or sc >= float(np.max(q)) - 1e-12:
        return None
    return h_new, sc, q_new

def do_polish(h, q, score, n, rng, nsweeps):
    nmoves = min(n * (n - 1) // 2, max(24, min(2 * n, 80000 // max(n, 1))))
    for _ in range(nsweeps):
        accepted = 0
        pairs = list(range(n - 1))
        pairs = [(i, i + 1) for i in pairs]
        while len(pairs) < nmoves:
            i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
            if i < j: pairs.append((i, j))
        rng.shuffle(pairs)
        for (p, r) in pairs[:nmoves]:
            res = pair_line_search(h, q, p, r, n)
            if res is not None:
                h, score, q = res
                accepted += 1
        if accepted == 0: break
    return h, score, q

def anneal_traj(h, q, score, n, rng, n_epochs, best_h, best_s):
    K = min(1000, max(80, 80000 // max(n, 1)))
    scale = 0.12
    for e in range(n_epochs):
        T = max(5e-5, 3e-3 * (0.4 ** e))
        accepted = 0
        for _ in range(K):
            p, r = int(rng.integers(0, n)), int(rng.integers(0, n))
            if p == r: continue
            L = max(-h[p], h[r] - 1.0)
            U = min(1.0 - h[p], h[r])
            if U - L < 1e-12: continue
            t = float(rng.uniform(-scale, scale))
            if t < L or t > U: continue
            h_new = h.copy()
            h_new[p] += t
            h_new[r] -= t
            if not is_feasible(h_new, n):
                h_new = project_capped_simplex(h_new, n * 0.5)
                if h_new is None: continue
            sc, _ = exact_score(h_new, n)
            if sc is None: continue
            delta = sc - score
            if delta <= 0 or (-delta / T > -700 and rng.random() < np.exp(-delta / T)):
                h, score = h_new, sc
                _, q = exact_score(h, n)
                accepted += 1
                if sc < best_s[0] - 1e-12:
                    best_s[0] = sc; best_h[0] = h.copy()
        ratio = accepted / K
        if ratio < 0.12: scale = min(0.4, 1.25 * scale)
        elif ratio > 0.55: scale = max(0.01, 0.75 * scale)
        h, score, q = do_polish(h, q, score, n, rng, 1)
        if score < best_s[0] - 1e-12:
            best_s[0] = score; best_h[0] = h.copy()
    return h, score, q

def build_starts(n, S, parent, rng):
    starts = []
    if parent is not None:
        starts.append(parent.copy())
        starts.append(parent[::-1].copy())
        comp = 1.0 - parent.copy()
        p = project_capped_simplex(comp, S)
        if p is not None: starts.append(p)
    M = min(4, max(1, (n - 1) // 2))
    idx = np.arange(n, dtype=np.float64) + 0.5

        phi = float(rng.uniform(0.0, 2.0 * np.pi))
        A = float(rng.normal(0.0, 0.12 / np.sqrt(m)))
        u += A * np.cos(2.0 * np.pi * m * idx / n + phi)
    pu = project_capped_simplex(u, S)
    if pu is not None: starts.append(pu)
    perm = rng.permutation(n)
    binv = np.zeros(n)
    binv[perm[:n // 2]] = 1.0

    rand = rng.random(n)
    pr = project_capped_simplex(rand, S)

    deduped = []
    for s in starts:

        for d in deduped:
            if np.max(np.abs(s - d)) < 1e-10: dup = True; break
        if not dup and is_feasible(s, n): deduped.append(s)
