    h = np.clip(z[:_N], 0.0, 1.0)
    return z[_N] - lag_values(h, _N)

def _epi_ineq_jac(z):
    h = np.clip(z[:_N], 0.0, 1.0)
    G = lag_jacobian(h, _N)
    L = 2 * _N - 1
    J = np.zeros((L, _N + 1))
    J[:, :_N] = -G
    J[:, _N] = 1.0
    return J

def _epi_eq(z):
    return np.sum(z[:_N]) - _N / 2.0

def _epi_eq_jac(z):
    J = np.zeros(_N + 1)
    J[:_N] = 1.0
    return J

def slsqp_epigraph(h0, n, maxiter, ftol):
    global _N
    _N = n
    h0 = h0.copy().astype(np.float64)
    q = lag_values(h0, n)
    t0 = np.max(q) + max(1e-10, 1e-8 * max(1.0, abs(np.max(q))))
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    con = [
        {'type': 'ineq', 'fun': _epi_ineq, 'jac': _epi_ineq_jac},
        {'type': 'eq', 'fun': _epi_eq, 'jac': _epi_eq_jac},
    ]
    try:
        res = minimize(_epi_obj, z0, method='SLSQP', jac=_epi_obj_jac,
                       bounds=bounds, constraints=con,
                       options={'maxiter': maxiter, 'ftol': ftol})
    except Exception:
        return None
    if res.x is None or res.x.shape != (n + 1,) or not np.all(np.isfinite(res.x)):
        return None
    return res.x[:n]

def subgrad_search(h0, n, n_iter):
    h = h0.copy().astype(np.float64)
    dx = 2.0 / n
    eta = 0.2 / dx
    eta_max = 2.0 / dx
    cur_s = exact_score(h, n)
    fails = 0
    for _ in range(n_iter):
        q = lag_values(h, n)
        idx = int(np.argmax(q))
        d = idx - (n - 1)
        p0 = max(d, 0)
        q0 = max(-d, 0)
        m = n - abs(d)
        g = np.zeros(n)
        g[p0:p0 + m] += dx * (1.0 - h[q0:q0 + m])
        g[q0:q0 + m] -= dx * h[p0:p0 + m]
        z = project_capped(h - eta * g, n)
        if z is None:
            eta *= 0.5
            fails += 1
            if fails >= 8 or eta < 1e-8 / dx:
                break
            continue
        v = z - h
        if np.max(np.abs(v)) < 1e-12:
            eta *= 0.5
            fails += 1
            if fails >= 8 or eta < 1e-8 / dx:
                break
            continue
        best_z, best_sz = h, cur_s
        for alpha in (1.0, 0.5, 0.25, 0.125, 0.0625):
            trial = project_capped(h + alpha * v, n)
            if trial is None:
                continue
            sz = exact_score(trial, n)
            if sz < best_sz - 1e-12:
                best_sz = sz
                best_z = trial
        if best_z is h:
            eta *= 0.5
            fails += 1
            if fails >= 8 or eta < 1e-8 / dx:
                break
        else:
            h = best_z
            cur_s = best_sz
            fails = 0
            eta = min(1.2 * eta, eta_max)
    return h, cur_s

def build_starts(fb, n, rng, count):
    pool = [fb.copy()]
    cands = [fb[::-1].copy(), 1.0 - fb[::-1]]
    cands.append(np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    cands.append(blk)
    cands.append(np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64))
    for sd in (0.08, 0.20, 0.40):
        cands.append(fb + rng.normal(0, sd, n))
    cands.append(rng.uniform(0, 1, n))
    for c in cands:
        if len(pool) >= count:
            break
        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-8:
                dup = True
                break
        if not dup:
            pool.append(p)
    return pool[:count]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size >= 1:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    fb = project_capped(parent, n) if parent is not None else np.full(n, 0.5)
    if fb is None:
        fb = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    fs = exact_score(fb, n)
    gb, gs = fb.copy(), fs
    if n == 1:
        return fb, fs, 1
    if n <= 64:
        n_sc, n_mi, n_pass = 8, 250, 2
    elif n <= 128:
        n_sc, n_si, n_pass = 3, 45, 0
    elif n <= 256:
        n_sc, n_si, n_pass = 2, 30, 0
    else:
        n_sc, n_si, n_pass = 1, 18, 0
    starts = build_starts(fb, n, rng, n_sc)
    scored = []
    for st in starts:
        s = exact_score(st, n)
        if s < gs - 1e-12:
            gs, gb = s, st.copy()
        scored.append((s, st))
    scored.sort(key=lambda x: x[0])
    for _, st in scored:
        try:
            if n <= 64:
                cur, cur_s = st.copy(), exact_score(st, n)
                for _ in range(n_pass):
                    found = slsqp_epigraph(cur, n, n_mi, 1e-12)
                    if found is None:
                        break
                    y = project_capped(found, n)
                    if y is None:
                        break
                    bz, bs = cur, cur_s
                    for a in (1.0, 0.5, 0.25, 0.125, 0.0625):
                        tr = project_capped(cur + a * (y - cur), n)
                        if tr is None:
                            continue
                        sz = exact_score(tr, n)
                        if sz < bs - 1e-12:
                            bs, bz = sz, tr
                    if bs >= cur_s - 1e-12:
                        break
                    cur, cur_s = bz, bs
                    if cur_s < gs - 1e-12:
                        gs, gb = cur_s, cur.copy()
            else:
                xo, so = subgrad_search(st, n, n_si)
                if so < gs - 1e-12:
                    gs, gb = so, xo.copy()
        except Exception:
            continue
    # Perturbation restarts from best
    if n <= 64:
        for _ in range(3):
            try:
                pert = project_capped(gb + rng.normal(0, 0.03, n), n)
                if pert is None:
                    continue
                cur, cur_s = pert, exact_score(pert, n)
                found = slsqp_epigraph(cur, n, 120, 1e-12)
                if found is None:
                    continue
                y = project_capped(found, n)
                if y is None:
                    continue
                for a in (1.0, 0.5, 0.25):
                    tr = project_capped(cur + a * (y - cur), n)
                    if tr is None:
                        continue
                    sz = exact_score(tr, n)
                    if sz < gs - 1e-12:
                        gs, gb = sz, tr.copy()
            except Exception:
                continue
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb
    sf = exact_score(final, n)
    ss = exact_score(fb, n)
    if sf > ss:
        final, sf = fb, ss
    if not is_feasible(final, n):
        final = np.full(n, 0.5)
        sf = exact_score(final, n)
    return final, float(sf), int(n)
