def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    lo, hi = float(v.min()) - 1.0, float(v.max())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S: lo = mid
        else: hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - y.sum()
    if abs(r) > 1e-14:
        if r > 0:
            for idx in np.argsort(1.0 - y):
                a = min(r, 1.0 - y[idx]); y[idx] += a; r -= a
                if abs(r) < 1e-15: break
        else:
            for idx in np.argsort(y):
                a = min(-r, y[idx]); y[idx] -= a; r += a
                if abs(r) < 1e-15: break
    y = np.clip(y, 0.0, 1.0)
    r = S - y.sum()
    if abs(r) > 1e-10:
        if r > 0 and y[0] < 1: y[0] += min(r, 1 - y[0])
        elif r < 0 and y[0] > 0: y[0] += max(r, -y[0])
    if np.any(y < -1e-12) or np.any(y > 1 + 1e-12): return None
    return y

def exact_score(h):
    n = len(h)
    if np.any(h < -1e-12) or np.any(h > 1 + 1e-12): return 1e10
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * (2.0 / n))

def feasible(h, n):
    return (h is not None and h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1 + 1e-12)
            and abs(h.sum() - n / 2.0) < 1e-9)

def pidx(i, j, n):

        A[k + n - 1] = -np.sum(d[i] * d[j])
        B[k + n - 1] = np.sum(d[i] * (1 - x[j]) - x[i] * d[j])

            s = -B[kk] / (2 * A[kk])

    for kk in range(nl):
        for ll in range(kk + 1, nl):
            da, db, dd = A[kk] - A[ll], B[kk] - B[ll], D[kk] - D[ll]
            if abs(da) < 1e-14:

                    r_ = -dd / db
                    if -1e-10 <= r_ <= 1 + 1e-10: cands.append(np.clip(r_, 0, 1))
            else:
