    return lb, lb_sc

def build_starts(parent, n, rng):
    cands = [parent.copy(), (1.0 - parent).copy(), parent[::-1].copy()]
    for sd in [0.05, 0.15]:
        p = project_box_sum(parent + sd * rng.standard_normal(n), 0.5 * n)
        if p is not None and check_feasible(p, n):
            cands.append(p)
    p = project_box_sum(rng.random(n), 0.5 * n)
    if p is not None and check_feasible(p, n):
        cands.append(p)
    i = np.arange(n, dtype=np.float64)
    phi = float(rng.uniform(0, 2 * math.pi))
    th = float(rng.uniform(0, 2 * math.pi))
    u = 0.5 + 0.28 * np.cos(2 * math.pi * i / n + phi) + 0.12 * np.sin(4 * math.pi * i / n + th)
    p = project_box_sum(u, 0.5 * n)
    if p is not None and check_feasible(p, n):
        cands.append(p)
    u2 = 0.5 + 0.35 * np.cos(2 * math.pi * 2 * i / n + 0.5)
    p2 = project_box_sum(u2, 0.5 * n)
    if p2 is not None and check_feasible(p2, n):
        cands.append(p2)
    uniq = []
    for s in cands:
        d = [float(np.max(np.abs(s - u))) for u in uniq]
        if len(uniq) == 0 or min(d) > 1e-10:
            uniq.append(s)
    return uniq

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)
    parent, n = get_parent()
    fb = parent.copy()
    best_h, best_c5 = parent.copy(), exact_score(parent, n)
    starts = build_starts(parent, n, rng)
    scored = []
    for s in starts:
        sc = exact_score(s, n)
        if np.isfinite(sc):
            scored.append((sc, s))
            if sc < best_c5:
                best_c5, best_h = sc, s.copy()
    scored.sort()
    if n <= 64:
        n_cyc = 4
    elif n <= 256:
        n_cyc = 3
    else:
        n_cyc = 2
    rds = [0.10, 0.25, 0.20, 0.30] if n <= 64 else ([0.18, 0.22] if n <= 256 else [0.12, 0.18])
    for ci in range(n_cyc):
        if ci == 0:
            cs = scored[0][1].copy() if scored else parent.copy()
        elif ci == 1 and len(scored) > 1:
            cs = scored[1][1].copy()
        else:
            sd = rds[min(ci - 2, len(rds) - 1)] if ci >= 2 else 0.10
            cs = None
            for _ in range(5):
                cs = project_box_sum(best_h + sd * rng.standard_normal(n), 0.5 * n)
                if cs is not None and check_feasible(cs, n):
                    break
            if cs is None:
                cs = best_h.copy()
        sc = exact_score(cs, n)
        if np.isfinite(sc) and sc < best_c5:
            best_c5, best_h = sc, cs.copy()
        try:
            lb, lb_sc = anneal_cycle(cs, n, rng)
            if np.isfinite(lb_sc) and lb_sc < best_c5:
                best_c5, best_h = lb_sc, lb.copy()
        except Exception:
            pass
    final = project_box_sum(best_h, 0.5 * n)
    if final is not None and check_feasible(final, n):
        sc = exact_score(final, n)
        if np.isfinite(sc):
            return (final, sc, n)
    if check_feasible(fb, n):
        sc = exact_score(fb, n)
        if np.isfinite(sc):
            return (fb, sc, n)
    u = np.full(n, 0.5)
    return (u, exact_score(u, n), n)
