import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if v.ndim != 1 or not np.all(np.isfinite(v)):
        return None
    lo, hi = float(np.min(v)) - 1.0, float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S: lo = mid
        else: hi = mid
    y = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-14:
        order = np.argsort(1.0-y) if resid > 0 else np.argsort(y)
        for i in order:
            if resid > 0:
                add = min(resid, 1.0-y[i]); y[i] += add; resid -= add
            else:
                sub = min(-resid, y[i]); y[i] -= sub; resid += sub
            if abs(resid) < 1e-15: break
    if not np.all(np.isfinite(y)) or np.any(y < -1e-12) or np.any(y > 1.0+1e-12): return None
    if abs(np.sum(y)-S) > 1e-10*max(1.0, float(n)): return None
    y = np.clip(y, 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-15:
        if resid > 0 and y[0] < 1.0: y[0] += min(resid, 1.0-y[0])
        elif resid < 0 and y[0] > 0.0: y[0] += max(resid, -y[0])
    return y

def exact_score(h, n):
    if h is None or len(h) != n or not np.all(np.isfinite(h)): return float('inf')
    if np.any(h < -1e-12) or np.any(h > 1.0+1e-12): return float('inf')
    if abs(np.sum(h) - n/2.0) > 1e-10*max(1.0, float(n)): return float('inf')
    c = np.correlate(h, 1.0-h, mode='full')
    sc = float(np.max(c) * 2.0/n)
    return sc if np.isfinite(sc) else float('inf')

def check_feasible(h, n):
    return (h is not None and h.shape == (n,) and np.all(np.isfinite(h))
            and not np.any(h < -1e-12) and not np.any(h > 1.0+1e-12)
            and abs(np.sum(h)-n/2.0) <= 1e-10*max(1.0, float(n)))

def prod_idx(i, j, n):
    return n + i*n - i*(i-1)//2 + (j - i)

def solve_lp(x, r, n):
    S = n / 2.0
    P = n*(n+1)//2
    nv = n + P + 1
    l_b = np.maximum(0.0, x - r)
    u_b = np.minimum(1.0, x + r)
    c = np.zeros(nv); c[n+P] = 1.0
    rows, cols, vals, rhs = [], [], [], []
    for i in range(n):
        for j in range(i, n):
            zi = prod_idx(i, j, n)
            li, ui, lj, uj = l_b[i], u_b[i], l_b[j], u_b[j]
            for vi_c, vj_c, rhs_v in [(-lj, -li, -li*lj), (-ui, -uj, -ui*uj),
                                        (-lj, -ui, -ui*lj), (-uj, -li, -li*uj)]:
                rows.append(len(rhs)); cols.extend([i, j, zi]); vals.extend([vi_c, vj_c, 1.0]); rhs.append(rhs_v)
    for i in range(n):
        for j in range(i, n):
            zlo, zhi = l_b[i]*l_b[j], u_b[i]*u_b[j]
            if zlo > zhi: zlo, zhi = zhi, zlo
            rows.append(len(rhs)); cols.extend([prod_idx(i,j,n)]); vals.extend([1.0]); rhs.append(zhi)
            rows.append(len(rhs)); cols.extend([prod_idx(i,j,n)]); vals.extend([-1.0]); rhs.append(-zlo)
    for k in range(-(n-1), n):
        i_lo, i_hi = max(0, k), min(n-1, n-1+k)
        rr = len(rhs); rows.append(rr)
        for ii in range(i_lo, i_hi+1):
            jj = ii - k
            rows.extend([rr, rr]); cols.extend([ii, prod_idx(min(ii,jj), max(ii,jj), n)]); vals.extend([1.0, -1.0])
        rows.extend([rr]); cols.extend([n+P]); vals.extend([-1.0]); rhs.append(0.0)
    A_ub = coo_matrix((vals, (rows, cols)), shape=(len(rhs), nv)).tocsr()
    A_eq = coo_matrix((np.ones(n), (np.zeros(n, dtype=int), np.arange(n))), shape=(1, nv)).tocsr()
    vb = list(zip(l_b, u_b)) + [(l_b[i]*l_b[j], u_b[i]*u_b[j]) for i in range(n) for j in range(i, n)] + [(0.0, float(n))]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=np.array(rhs), A_eq=A_eq, b_eq=np.array([S]),
                      bounds=vb, method='highs', options={'time_limit': 30})
        if res.success and res.x is not None and np.all(np.isfinite(res.x)):
            vw = res.x[:n]
            if np.all(vw >= l_b - 1e-7) and np.all(vw <= u_b + 1e-7):
                vw = np.clip(vw, l_b, u_b)
                hp = project_box_sum(vw, S)
                if hp is not None and check_feasible(hp, n): return hp
    except Exception: pass
    return None

def line_minimax(x, y, n, S):
    d = y - x
    n_lags = 2*n - 1
    A = np.zeros(n_lags); B = np.zeros(n_lags); D = np.zeros(n_lags)
    for k in range(-(n-1), n):
        idx = k + n - 1
        i_lo, i_hi = max(0, k), min(n-1, n-1+k)
        iv = np.arange(i_lo, i_hi+1); jv = iv - k
        D[idx] = np.sum(x[iv]*(1.0-x[jv]))
        B[idx] = np.sum(d[iv]*(1.0-x[jv]) - x[iv]*d[jv])
        A[idx] = -np.sum(d[iv]*d[jv])
    alphas = {0.0, 1.0}
    for idx in range(n_lags):
        if abs(A[idx]) > 1e-15:
            ast = -B[idx]/(2.0*A[idx])
            if -1e-9 <= ast <= 1.0+1e-9: alphas.add(np.clip(ast, 0.0, 1.0))
    for kk in range(n_lags):
        for ll in range(kk+1, n_lags):
            da, db, dc = A[kk]-A[ll], B[kk]-B[ll], D[kk]-D[ll]
            if abs(da) < 1e-14:
                if abs(db) > 1e-14:
                    a = -dc/db
                    if -1e-9 <= a <= 1.0+1e-9: alphas.add(np.clip(a, 0.0, 1.0))
            else:
                disc = db*db - 4.0*da*dc
                if disc >= -1e-12:
                    sq = np.sqrt(max(0.0, disc))
                    for a in [(-db+sq)/(2.0*da), (-db-sq)/(2.0*da)]:
                        if -1e-9 <= a <= 1.0+1e-9: alphas.add(np.clip(a, 0.0, 1.0))
    alphas = np.array(sorted(alphas))
    best_sc = float('inf'); best_h = None
    for a in alphas:
        pred = float(np.max(A*a*a + B*a + D))
        if pred < best_sc + 0.01:
            hc = project_box_sum(x + a*d, S)
            if hc is not None and check_feasible(hc, n):
                sc = exact_score(hc, n)
                if sc < best_sc: best_sc = sc; best_h = hc
    return best_h, best_sc

def local_lp_search(start, n):
    S = n / 2.0
    cur = project_box_sum(start, S)
    if cur is None or not check_feasible(cur, n): return None
    cur_sc = exact_score(cur, n)
    r = 0.20; fails = 0
    max_iter = 8 if n <= 64 else 5
    for _ in range(max_iter):
        if r < 1e-3 and fails >= 2: break
        ep = solve_lp(cur, r, n)
        if ep is None or np.max(np.abs(ep - cur)) < 1e-12:
            r *= 0.5; fails += 1; continue
        nh, nsc = line_minimax(cur, ep, n, S)
        if nh is not None and nsc < cur_sc - 1e-12:
            cur = nh; cur_sc = nsc; fails = 0; r = min(0.25, 1.2*r)
        else:
            r *= 0.5; fails += 1
    return cur if cur_sc < float('inf') else None
