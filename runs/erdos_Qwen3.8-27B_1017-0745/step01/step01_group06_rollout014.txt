from scipy.optimize import minimize

def project_K(y, n, S):
    if y is None or len(y) != n or not np.all(np.isfinite(y)):
        return np.full(n, S / n)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    err = S - h.sum()
    for _ in range(2):
        if abs(err) < 1e-12:
            break
        if err > 0:
            idx = np.where(1.0 - h > 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) == 0:
            break
        adj = err / len(idx)
        h[idx] = np.clip(h[idx] + adj, 0.0, 1.0)
        err = S - h.sum()
    if abs(h.sum() - S) > 1e-8:
        return np.full(n, S / n)
    return h

def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lag_vec(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_grad(h, dx):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        ell = k - (n - 1)
        j = np.arange(n)
        m1 = (j >= max(0, ell)) & (j < min(n, n + ell))
        m2 = (j >= max(0, -ell)) & (j < min(n, n - ell))
        G[k] = dx * (np.where(m1, 1.0 - h[j - ell], 0.0) - np.where(m2, h[j + ell], 0.0))
    return G

# SLSQP callbacks for epigraph subproblem
def ep_graph_obj(z, eta):
    d, t = z[:-1], z[-1]
    return t + np.dot(d, d) / (2.0 * eta)

def ep_graph_obj_jac(z, eta):
    g = np.zeros(len(z))
    g[:-1] = z[:-1] / eta
    g[-1] = 1.0
    return g

def ep_graph_ineq(z, q, G):
    d, t = z[:-1], z[-1]
    return t - q - G @ d

def ep_graph_ineq_jac(z, q, G):
    n = len(z) - 1
    J = np.zeros((len(q), n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def ep_graph_eq(z, n):
    return z[:-1].sum()

def ep_graph_eq_jac(z, n):
    g = np.zeros(len(z))
    g[:-1] = 1.0
    return g

def trust_region_step(h, n, dx, delta, eta):
    q = lag_vec(h, dx)
    G = lag_grad(h, dx)
    m = q.max()
    z0 = np.zeros(n + 1)
    z0[-1] = m
    bounds = []
    for j in range(n):
        lo = max(-delta, -h[j])
        hi = min(delta, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((None, None))
    cons = [
        {'type': 'eq', 'fun': ep_graph_eq, 'jac': ep_graph_eq_jac, 'args': (n,)},
        {'type': 'ineq', 'fun': ep_graph_ineq, 'jac': ep_graph_ineq_jac, 'args': (q, G)}
    ]
    maxit = 50 if n <= 96 else (30 if n <= 256 else 15)
    res = minimize(ep_graph_obj, z0, jac=ep_graph_obj_jac, args=(eta,),
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxit, 'ftol': 1e-10, 'disp': False})
    if res is None or not hasattr(res, 'x') or not np.all(np.isfinite(res.x)):
        return None, q
    z = res.x
    d = z[:-1]
    # Check feasibility of the step
    for j in range(n):
        if d[j] < -h[j] - 1e-5 or d[j] > 1.0 - h[j] + 1e-5:
            return None, q
    if abs(d.sum()) > 1e-5 * n:

    return d, q

import numpy as np
from scipy.optimize import minimize

def project_K(y, n, S):
    if y is None or len(y) != n or not np.all(np.isfinite(y)):
        return np.full(n, S / n)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(y - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(3):
        err = S - h.sum()
        if abs(err) < 1e-13:
            break
        if err > 0:
            idx = np.where(1.0 - h > 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) == 0:
            break
        h[idx] = np.clip(h[idx] + err / len(idx), 0.0, 1.0)
    if abs(h.sum() - S) > 1e-8:
        return np.full(n, S / n)
    return h

def exact_score(h, dx):
    if not np.all(np.isfinite(h)):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lag_vec(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lag_grad(h, dx):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    j = np.arange(n)
    for k in range(2 * n - 1):
        ell = k - (n - 1)
        m1 = (j >= max(0, ell)) & (j < min(n, n + ell))
        m2 = (j >= max(0, -ell)) & (j < min(n, n - ell))
        G[k] = dx * (np.where(m1, 1.0 - h[j - ell], 0.0) - np.where(m2, h[j + ell], 0.0))
    return G

def _eo(z, eta):
    d, t = z[:-1], z[-1]
    return t + np.dot(d, d) / (2.0 * eta)

def _eoj(z, eta):
    g = np.zeros(len(z))
    g[:-1] = z[:-1] / eta
    g[-1] = 1.0
    return g

def _ei(z, q, G):
    d, t = z[:-1], z[-1]
    return t - q - G @ d

def _eij(z, q, G):
    n = len(z) - 1
    J = np.zeros((len(q), n + 1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def _ee(z, n):
    return z[:-1].sum()

def _eej(z, n):
    g = np.zeros(len(z))
    g[:-1] = 1.0
    return g

def tr_step(h, n, dx, delta, eta):
    q = lag_vec(h, dx)
    G = lag_grad(h, dx)
    z0 = np.zeros(n + 1)
    z0[-1] = float(q.max())
    bounds = [(max(-delta, -h[j]), min(delta, 1.0 - h[j])) for j in range(n)]
    bounds.append((None, None))
    cons = [{'type': 'eq', 'fun': _ee, 'jac': _eej, 'args': (n,)},
            {'type': 'ineq', 'fun': _ei, 'jac': _eij, 'args': (q, G)}]
    mi = 50 if n <= 96 else (30 if n <= 256 else 15)
    try:
        res = minimize(_eo, z0, jac=_eoj, args=(eta,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': mi, 'ftol': 1e-10, 'disp': False})
    except Exception:
        return None, q, G
    if res is None or not hasattr(res, 'x') or not np.all(np.isfinite(res.x)):
        return None, q, G
    d = res.x[:-1]
    for j in range(n):
        if d[j] < -h[j] - 1e-4 or d[j] > 1.0 - h[j] + 1e-4:
            return None, q, G
    if abs(d.sum()) > 1e-4 * n:
        return None, q, G
    return d, q, G

def local_opt(h0, n, dx, max_it):
    h = project_K(h0.copy(), n, n / 2.0)
    S = n / 2.0
    cur_s = exact_score(h, dx)
    delta, eta = 0.12, 0.5
    cf, cr = 0, 0
    for _ in range(max_it):
        d, q, G = tr_step(h, n, dx, delta, eta)
        if d is None:
            cf += 1
            delta *= 0.5
            if cf >= 4 or delta < 1e-5:
                break
            continue
        cf = 0
        ht = project_K(h + d, n, S)
        if not np.all(np.isfinite(ht)) or ht.shape != (n,):
            delta *= 0.5
            cr += 1
            if cr >= 8:
                break
            continue
        st = exact_score(ht, dx)
        if not np.isfinite(st):
            delta *= 0.5
            cr += 1
            if cr >= 8:
                break
            continue
        da = ht - h
        qn = q + G @ da
        pred = cur_s - float(np.max(qn))
        act = cur_s - st
        if np.max(np.abs(da)) < 1e-8:
            break
        if act > 1e-12 and (pred <= 1e-12 or (pred > 1e-12 and act / pred >= 0.1)):
            h, cur_s = ht, st
            cr = 0
            if pred > 1e-12:
                r = act / pred
                if r >= 0.75 and np.max(np.abs(da)) >= 0.8 * delta:
                    delta = min(0.5, 2.0 * delta)
                    eta = min(2.0, 1.5 * eta)
                elif r < 0.25:
                    delta *= 0.5
        else:
            cr += 1
            delta *= 0.5
            eta = max(1e-4, 0.5 * eta)
            if cr >= 8 or delta < 1e-5:
                break
    return h, cur_s

def build_starts(base, n, S, rng, hp):
    s = [base.copy(), (1.0 - base).copy(), base[::-1].copy()]
    s.append(project_K(base + 0.08 * rng.standard_normal(n), n, S))
    s.append(project_K(base + 0.25 * rng.standard_normal(n), n, S))
    s.append(project_K(rng.uniform(0, 1, n), n, S))
    s.append(project_K(rng.beta(0.5, 0.5, n), n, S))
    hb = np.zeros(n)
    hb[:int(S)] = 1.0
    if S != int(S):
        hb[int(S)] = S - int(S)
    s.append(project_K(hb, n, S))
    j = np.arange(n)
    s.append(project_K(0.5 + 0.45 * np.cos(2.0 * np.pi * j / n), n, S))
    if hp and n > 4:
        s.append(project_K(base[rng.permutation(n)], n, S))
    s.append(project_K(0.5 + 0.3 * rng.standard_normal(n), n, S))
    uniq = []
    for x in s:
        if not any(np.max(np.abs(x - u)) <= 1e-6 for u in uniq):
            uniq.append(x)
    return uniq

def res_transfer(par, n0, m):
    if m == n0:
        return par.copy()
    y = np.zeros(m)
    for j in range(m):
        xl, xh = 2.0 * j / m, 2.0 * (j + 1) / m
        v = 0.0
        for i in range(n0):
            a, b = max(xl, 2.0 * i / n0), min(xh, 2.0 * (i + 1) / n0)
            if b > a:
                v += par[i] * (b - a)
        y[j] = v * m / 2.0
    return y

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)

        rng = np.random.default_rng(42)

    fb_h, fb_s, fb_n = par.copy(), exact_score(par, 2.0 / n0), n0

        if n <= 96:

            sr = exact_score(hr, dx)
