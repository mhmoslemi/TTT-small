def project_box_sum(u,S):
    u=np.asarray(u,dtype=np.float64)
    if u.ndim !=1 or u.size==0 or not np.all(np.isfinite(u)): return None
    n=u.size
    if S < -1e-12 or S > n+1e-12: return None
    lo=float(np.min(u))-1.0
    hi=float(np.max(u))
    for _ in range(80):
        mid=0.5*(lo+hi)
        y=np.clip(u-mid,0.0,1.0)
        if np.sum(y)>S:
           lo=mid
        else:
           hi=mid
    y=np.clip(u-0.5*(lo+hi),0.0,1.0)
    resid=S-np.sum(y)
    if resid>0:
       order=np.argsort(1.0-y)
       for i in order:
          add=min(resid,1.0-y[i]); y[i]+=add; resid-=add
          if abs(resid)<1e-15: break

       for i in order:
          sub=min(-resid,y[i]); y[i]-=sub; resid+=sub
          if abs(resid)<1e-15: break
    y=np.clip(y,0.0,1.0)
    if not np.all(np.isfinite(y)): return None
    if abs(np.sum(y)-S)>1e-9*max(1,n): return None
    return y

def smooth_eval(x):
    corr=np.correlate(x,1.0-x,mode='full')
    s=corr*_DX
    m=float(np.max(s))
    e=np.exp((s-m)/_T)
    Z=float(np.sum(e))
    if Z <= 0.0 or not math.isfinite(Z):
        return 1e100
    return _T*(m + math.log(Z))

def smooth_jac(x):
    n=_N
    h=x
    corr=np.correlate(h,1.0-h,mode='full')
    s=corr*_DX
    m=float(np.max(s))
    e=np.exp((s-m)/_T)
    Z=float(np.sum(e))
    if Z <= 0.0 or not math.isfinite(Z):
        return np.zeros(n)
    w=e/Z
    grad=np.zeros(n,dtype=np.float64)
    tol=1e-12
    for idx in range(2*n-1):
        wl=w[idx]
        if wl <= tol: continue
        k=idx-(n-1)
        if k>=0:
            mlen=n-k
            grad[k:n] += wl
            grad[k:n] -= wl*h[0:mlen]
            grad[0:mlen] -= wl*h[k:n]
        else:
            r=-k
            mlen=n-r
            grad[0:mlen] += wl
            grad[0:mlen] -= wl*h[r:n]
            grad[r:n] -= wl*h[0:mlen]
    return _DX*grad

def smooth_eq(x): return float(np.sum(x)-_S)
def smooth_eq_jac(x): return np.ones(_N,dtype=np.float64)

import numpy as np

def pbs(u, S):
    u = np.asarray(u, dtype=np.float64)
    if u.ndim != 1 or u.size == 0 or not np.all(np.isfinite(u)):
        return None
    n = u.size
    if S < -1e-12 or S > n + 1e-12:
        return None
    lo = float(np.min(u)) - 1.0
    hi = float(np.max(u))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(u - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(u - 0.5 * (lo + hi), 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 1e-15:
        for i in np.argsort(1.0 - y):
            a = min(r, 1.0 - y[i])
            y[i] += a
            r -= a
            if r < 1e-15:
                break
    elif r < -1e-15:
        for i in np.argsort(y):
            a = min(-r, y[i])
            y[i] -= a
            r += a
            if r > -1e-15:
                break
    y = np.clip(y, 0.0, 1.0)
    r = S - float(np.sum(y))
    if r > 0:
        i = int(np.argmax(1.0 - y))
        y[i] += min(r, 1.0 - y[i])
    elif r < 0:
        i = int(np.argmax(y))
        y[i] += max(r, -y[i])
    y = np.clip(y, 0.0, 1.0)
    if not np.all(np.isfinite(y)) or abs(float(np.sum(y)) - S) > 1e-9 * max(1.0, float(n)):
        return None
    return y

def ok(h, n, S):
    return (h is not None and h.shape == (n,) and np.all(np.isfinite(h))
            and np.min(h) >= -1e-12 and np.max(h) <= 1.0 + 1e-12
            and abs(float(np.sum(h)) - S) <= 1e-10 * max(1.0, float(n)))

def esc(h):
    if h is None:
        return float('inf')
    h = np.asarray(h, dtype=np.float64)
    n = h.size
    if n < 2 or not np.all(np.isfinite(h)):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    if c.shape != (2 * n - 1,):
        return float('inf')
    v = float(np.max(c)) * (2.0 / n)
    return v if math.isfinite(v) else float('inf')

def addc(lst, h, n, S):
    c = pbs(h, S)
    if not ok(c, n, S):
        return
    for x in lst:
        if np.max(np.abs(x - c)) < 1e-10:
            return
    lst.append(c)

def build(parent, n, S, rng, P):
    st = []
    u = pbs(np.full(n, 0.5), S)
    st.append(u if u is not None else np.full(n, 0.5))
    if parent is not None:
        addc(st, parent, n, S)
        addc(st, parent[::-1], n, S)
        addc(st, 1.0 - parent, n, S)
        for sh in (1, 2, 3, n // 3, 2 * n // 3):
            addc(st, np.roll(parent, sh), n, S)
        for sig in (0.05, 0.15, 0.35):
            addc(st, parent + sig * rng.standard_normal(n), n, S)
    else:
        for A, m, p in ((0.3, 1, 0.0), (0.4, 2, 0.5), (0.5, 3, 1.0)):
            x = np.arange(n)
            addc(st, 0.5 + A * np.cos(2.0 * np.pi * m * x / n + p), n, S)
    cyc = 0
    while len(st) < P and cyc < 30 * P:

    if not st:
        u = pbs(np.full(n, 0.5), S)
        st.append(u if u is not None else np.full(n, 0.5))
    arr = np.empty((P, n), dtype=np.float64)
    for i in range(P):
        arr[i] = st[i % len(st)]
    return arr

def devolve(pop, sc, rng, P, n, S, G, bh, bs):
    pop = np.array(pop, dtype=np.float64)
    sc = np.array(sc, dtype=np.float64)
    for i in range(P):
        if not math.isfinite(sc[i]):
            sc[i] = 1e100
    G = max(1, int(G))
    sw = int(0.5 * G)
    for gen in range(G):
        order = np.argsort(sc)
        pb = max(2, int(math.ceil(0.2 * P)))
        pb = min(pb, P)
        pbest = order[:pb]
        npop = pop.copy()
        nsc = sc.copy()
        for i in range(P):
            F = 0.5 + 0.5 * rng.random()
            CR = 0.5 + 0.5 * rng.random()
            others = [j for j in range(P) if j != i]
            if gen >= sw and rng.random() < 0.75:
                p = int(pbest[rng.integers(pb)])
                if p == i:
                    ab = rng.choice(others, size=2, replace=False)
                else:
                    pool = [j for j in others if j != p]
                    ab = rng.choice(pool, size=2, replace=False)
                a = int(ab[0])
                b = int(ab[1])
                v = pop[i] + F * (pop[p] - pop[i]) + F * (pop[a] - pop[b])
            else:
                abc = rng.choice(others, size=3, replace=False)
                a = int(abc[0])
                b = int(abc[1])
                c = int(abc[2])
                v = pop[a] + F * (pop[b] - pop[c])
            mask = rng.random(n) < CR
            mask[int(rng.integers(n))] = True
            t = np.empty(n, dtype=np.float64)
            t[mask] = v[mask]
            t[~mask] = pop[i][~mask]
            pr = pbs(t, S)
            if not ok(pr, n, S):
                continue
            if float(np.max(np.abs(pr - pop[i]))) < 1e-13:
                continue
            e = esc(pr)
            if math.isfinite(e) and e < nsc[i] - 1e-14:
                npop[i] = pr
                nsc[i] = e
                if e < bs - 1e-14:
                    bs = e
                    bh = pr.copy()
        pop = npop
        sc = nsc
    return pop, sc, bh, bs

def seval(x):
    s = np.correlate(x, 1.0 - x, mode='full') * _DX
    m = float(np.max(s))
    Z = float(np.sum(np.exp((s - m) / _T)))
    if Z <= 0.0 or not math.isfinite(Z):
        return 1e100
    return _T * (m + math.log(Z))

def sscore(x):
    n = _N
    h = x
    s = np.correlate(h, 1.0 - h, mode='full') * _DX
    m = float(np.max(s))
    w = np.exp((s - m) / _T)
    Z = float(np.sum(w))
    if Z <= 0.0 or not math.isfinite(Z):
        return np.zeros(n, dtype=np.float64)
    w /= Z
    g = np.zeros(n, dtype=np.float64)
    for idx in range(2 * n - 1):
        wl = w[idx]
        if wl <= 1e-12:
            continue
        k = idx - (n - 1)
        if k >= 0:
            ml = n - k
            g[k:n] += wl
            g[k:n] -= wl * h[0:ml]
            g[0:ml] -= wl * h[k:n]
        else:
            r = -k
            ml = n - r
            g[0:ml] += wl
            g[0:ml] -= wl * h[r:n]
            g[r:n] -= wl * h[0:ml]
    return _DX * g

def seq(x):
    return float(np.sum(x) - _S)

def seqj(x):
    return np.ones(_N, dtype=np.float64)

def eobj(z):
    return float(z[-1])

def eobjj(z):
    g = np.zeros(z.size, dtype=np.float64)
    g[-1] = 1.0
    return g

def eeq(z):
    return float(np.sum(z[:-1]) - _S)

def eeqj(z):
    J = np.zeros((1, z.size), dtype=np.float64)
    J[0, :-1] = 1.0
    return J

def ein(z):
    return z[-1] - np.correlate(z[:-1], 1.0 - z[:-1], mode='full') * _DX

def eij(z):
    n = _N
    h = z[:-1]
    L = 2 * n - 1
    J = np.zeros((L, n + 1), dtype=np.float64)
    J[:, n] = 1.0
    for idx in range(L):
        g = np.zeros(n, dtype=np.float64)
        k = idx - (n - 1)
        if k >= 0:
            m = n - k
            g[k:n] += 1.0
            g[k:n] -= h[0:m]
            g[0:m] -= h[k:n]
        else:
            r = -k
            m = n - r
            g[0:m] += 1.0
            g[0:m] -= h[r:n]
            g[r:n] -= h[0:m]
        J[idx, :n] = -_DX * g
    return J

        return h0, esc(h0)
    best = h.copy()
    lbs = esc(h)
    if n <= 120:
        _N = n
        _S = float(S)
        _DX = 2.0 / n
        bnds = [(0.0, 1.0)] * n
        cons = ({'type': 'eq', 'fun': seq, 'jac': seqj},)
        cur = best.copy()
        for T in (0.02, 0.008, 0.003, 0.001, 0.0003):
            _T = T
            try:
                res = minimize(seval, cur, jac=sscore, method='SLSQP', bounds=bnds,
                               constraints=cons, options={'maxiter': 100, 'ftol': 1e-12})
                if res.x is not None and len(res.x) == n:
                    p = pbs(res.x, S)
                    if ok(p, n, S):
                        cur = p
                        e = esc(p)
                        if math.isfinite(e) and e < lbs - 1e-14:
                            lbs = e
                            best = p.copy()
            except Exception:
                pass
        if do_exact and n <= 60:
            h2 = best.copy()
            for it in (80, 160):
                t0 = esc(h2)
                if not math.isfinite(t0):
                    break
                zb = np.empty(n + 1, dtype=np.float64)
                zb[:n] = h2
                zb[-1] = min(1.0, t0 + 1e-5)
                ebnds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
                econs = ({'type': 'eq', 'fun': eeq, 'jac': eeqj},
                         {'type': 'ineq', 'fun': ein, 'jac': eij})
                try:
                    res = minimize(eobj, zb, jac=eobjj, method='SLSQP', bounds=ebnds,
                                   constraints=econs, options={'maxiter': it, 'ftol': 1e-13})
                    if res.x is None or len(res.x) != n + 1:
                        break
                    p = pbs(res.x[:n], S)
                    if not ok(p, n, S):
                        break
                    e = esc(p)
                    if not math.isfinite(e):
                        break
                    if e < lbs - 1e-14:
                        lbs = e
                        best = p.copy()
                        h2 = p.copy()
                    elif e < t0 + 1e-12:
                        h2 = p.copy()
                    else:
                        break
                except Exception:
                    break
    fin = pbs(best, S)
    if fin is None or not ok(fin, n, S):
        fin = best
    e = esc(fin)
    if math.isfinite(e) and e < lbs:
        lbs = e
        best = fin
    return best, lbs

def topcand(pop, sc, K):
    if K <= 0:
        return []
    order = np.argsort(sc)
    out = []
    for idx in order:
        h = pop[idx]
        dup = False
        for s in out:
            if np.max(np.abs(h - s)) < 1e-7:
                dup = True
                break
        if not dup:
            out.append(h.copy())
            if len(out) >= K:
                break
    return out

def run(seed=42, **kwargs):
    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)

    n = 45
    parent = None
    try:
        pv = np.array(initial_h_values, dtype=np.float64)
        if pv.ndim == 1:
            m = int(pv.size)
            if (2 <= m <= 999 and np.all(np.isfinite(pv))
                    and np.all(pv >= -1e-9) and np.all(pv <= 1.0 + 1e-9)
                    and abs(float(np.sum(pv)) - m / 2.0) <= 1e-6 * max(1.0, float(m))):
                pc = np.clip(pv, 0.0, 1.0)
                pr = pbs(pc, m / 2.0)
                if ok(pr, m, m / 2.0):
                    n = m
                    parent = pr
    except Exception:
        pass

    S = float(n) / 2.0
    if parent is None:
        parent = pbs(np.full(n, 0.5), S)
    if not ok(parent, n, S):
        parent = pbs(np.full(n, 0.5), S)
    if not ok(parent, n, S):
        parent = np.clip(np.full(n, 0.5), 0.0, 1.0)

    fallback = parent.copy()
    bh = fallback.copy()
    bs = esc(bh)

    P = int(min(64, max(16, int(64.0 * math.sqrt(45.0 / float(n)) + 0.5))))
    pop = build(parent, n, S, rng, P)
    sc = np.array([esc(pop[i]) for i in range(P)], dtype=np.float64)
    for i in range(P):
        if not math.isfinite(sc[i]):
            sc[i] = 1e100
        if sc[i] < bs - 1e-14:
            bs = sc[i]
            bh = pop[i].copy()

    if n <= 120:
        rh, rs = refine(fallback, n, S, n <= 60)
        if math.isfinite(rs) and rs < bs - 1e-14:
            bs = rs
            bh = rh.copy()
        if math.isfinite(rs):
            w = int(np.argmax(sc))
            if rs < sc[w] - 1e-14:
                pop[w] = rh
                sc[w] = rs

    G = int(min(120, max(12, int(6000.0 / max(1, P) * (45.0 / float(n)) + 0.5))))
    pop, sc, bh, bs = devolve(pop, sc, rng, P, n, S, G, bh, bs)

    if n <= 120:
        K = 6 if n <= 60 else 3
        cands = topcand(pop, sc, K)
        for extra in (bh, fallback):
            dup = False
            for s in cands:
                if np.max(np.abs(extra - s)) < 1e-7:
                    dup = True
                    break
            if not dup:
                cands.append(extra.copy())
        if cands:
            cs = np.array([esc(h) for h in cands], dtype=np.float64)
            co = np.argsort(cs)
            for r, idx in enumerate(co):
                if r >= K:
                    break
                rh, rs = refine(cands[idx], n, S, (r < 2 and n <= 60))
                if math.isfinite(rs) and rs < bs - 1e-14:
                    bs = rs
                    bh = rh.copy()

    fh = pbs(bh, S)
    if fh is None or not ok(fh, n, S):
        fh = fallback
    e = esc(fh)
    if not math.isfinite(e):
        e = esc(fallback)
    if not math.isfinite(e):
        e = 1e-1
    return (np.array(fh, dtype=np.float64), float(e), int(n))
