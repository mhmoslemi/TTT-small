    s = n / 2.0

        if np.sum(np.clip(z - mid, 0, 1)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0, 1)
    r = s - h.sum()
    for _ in range(4 * n):
        if abs(r) < 1e-14:
            break

            if abs(r) < 1e-14:

            h[i] += a
            r -= a

import numpy as np
from scipy.optimize import minimize

def project_capped(z, n):
    z = np.clip(np.asarray(z, dtype=np.float64).ravel(), -1e12, 1e12)
    s = n / 2.0
    lo, hi = z.min() - 1.0, z.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0, 1)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0, 1)
    r = s - h.sum()
    for _ in range(4 * n):
        if abs(r) < 1e-14:
            break
        for i in range(n):
            if abs(r) < 1e-14:
                break
            if r > 0:
                a = min(1.0 - h[i], r)
            else:
                a = max(-h[i], r)
            h[i] += a
            r -= a
    if h.min() < -1e-12 or h.max() > 1 + 1e-12:
        return None
    return np.clip(h, 0, 1)

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            h.min() >= -1e-12 and h.max() <= 1 + 1e-12 and
            abs(h.sum() - n / 2.0) < 1e-9 * max(1, n))

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, 'full') * (2.0 / n)))

def compute_lags(x, n):
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.zeros(L)
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        p0, q0 = max(d, 0), max(-d, 0)
        m = n - abs(d)
        a, b = x[p0:p0 + m], x[q0:q0 + m]
        i = d + n - 1
        q[i] = dx * np.dot(a, 1.0 - b)
        G[i, p0:p0 + m] += dx * (1.0 - b)
        G[i, q0:q0 + m] -= dx * a
    return q, G

def mm_func(z, x, dx, beta):
    n = len(x)
    return z[-1] + beta * dx * np.dot(z[:n] - x, z[:n] - x)

def mm_grad(z, x, dx, beta):
    n = len(x)
    g = np.empty(n + 1)
    g[:n] = 2.0 * beta * dx * (z[:n] - x)
    g[n] = 1.0
    return g

def mm_ineq(z, x, q, G):
    n = len(x)
    return z[-1] - q - G @ (z[:n] - x)

def mm_ineq_j(z, x, q, G):
    n = len(x)
    return np.column_stack((-G, np.ones(G.shape[0])))

def mm_eq(z, x, q, G):
    return z[:len(x)].sum() - len(x) / 2.0

def mm_eq_j(z, x, q, G):
    n = len(x)
    J = np.zeros(n + 1)
    J[:n] = 1.0
    return J

def sm_lages(z, n, mu):
    dx = 2.0 / n
    L = 2 * n - 1
    vals = np.zeros(L)
    Gm = np.zeros((L, n))
    for d in range(-(n - 1), n):
        p0, q0 = max(d, 0), max(-d, 0)
        m = n - abs(d)
        a, b = z[p0:p0 + m], z[q0:q0 + m]
        i = d + n - 1
        vals[i] = dx * np.dot(a, 1.0 - b)
        Gm[i, p0:p0 + m] = dx * (1.0 - b)
        Gm[i, q0:q0 + m] -= dx * a
    mval = mu * vals.max()
    ev = np.exp(np.clip(mu * vals - mval, -700, 700))
    ev_sum = ev.sum()
    obj = (mval + np.log(ev_sum)) / mu
    w = ev / ev_sum
    return obj, w @ Gm

def sm_obj(z, n, mu):
    return sm_lages(z, n, mu)[0]

def sm_grad(z, n, mu):
    return sm_lages(z, n, mu)[1]

def eq_fun(z, n, mu):
    return z.sum() - n / 2.0

def eq_jac(z, n, mu):
