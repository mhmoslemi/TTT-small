def lag_values_and_grads(h, n):
    L = 2 * n - 1
    v = np.zeros(L)
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        if lo >= hi:
            continue
        idx_i = np.arange(lo, hi)
        idx_j = idx_i + d
        v[row] = np.sum(h[idx_i] * (1.0 - h[idx_j]))
        G[row, idx_i] += (1.0 - h[idx_j])
        G[row, idx_j] -= h[idx_i]
    return v, G
