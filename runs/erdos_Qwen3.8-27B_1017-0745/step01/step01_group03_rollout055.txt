def project_box_sum(v, S):

    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0

        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-14:

            order = np.argsort(1.0 - y)
            for idx in order:

                y[idx] += add
                resid -= add
                if abs(resid) < 1e-15:

            for idx in order:

                resid += sub

                    break
    y = np.clip(y, 0.0, 1.0)

        return None
    return y

    n = len(h)

    c = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(c) * dx)

def solve_lp(x, r, n):

def lag_coeffs(x, d, n):
    """Compute A_k, B_k, D_k for each lag."""
    n_lags = 2 * n - 1
    A = np.zeros(n_lags)
    B = np.zeros(n_lags)
    D = np.zeros(n_lags)
    for k in range(-(n - 1), n):
        i_lo = max(0, k)
        i_hi = min(n - 1, n - 1 + k)
        idx = np.arange(i_lo, i_hi + 1)
        jj = idx - k
        a_k = -np.sum(d[idx] * d[jj])
        b_k = np.sum(d[idx] * (1.0 - x[jj]) - x[idx] * d[jj])
        d_k = np.sum(x[idx] * (1.0 - x[jj]))
        A[k + n - 1] = a_k
        B[k + n - 1] = b_k
        D[k + n - 1] = d_k
    return A, B, D

    eps = 1e-10

    for k in range(n_lags):

            a = -B[k] / (2.0 * A[k])

                candidates.append(np.clip(a, 0.0, 1.0))

            dc = D[k] - D[m]

                    a = -dc / db

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def project_box_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if not np.all(np.isfinite(v)):
        return None
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(v - mid, 0.0, 1.0)
        if np.sum(y) > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    resid = S - np.sum(y)
    if abs(resid) > 1e-14:
        if resid > 0:
            order = np.argsort(1.0 - y)
            for idx in order:
                add = min(resid, 1.0 - y[idx])
                y[idx] += add
                resid -= add
                if abs(resid) < 1e-15:
                    break
        else:
            order = np.argsort(y)
            for idx in order:
                sub = min(-resid, y[idx])
                y[idx] -= sub
                resid += sub
                if abs(resid) < 1e-15:
                    break
    y = np.clip(y, 0.0, 1.0)
    if abs(np.sum(y) - S) > 1e-10 * max(1.0, float(n)):
        return None
    return y

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(c) * dx)
    return val if np.isfinite(val) else float('inf')

def check_feasible(h, n):
    if h is None or h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-9 * max(1.0, float(n)):
        return False
    return True

def prod_idx(i, j, n):
    if i > j:
        i, j = j, i
    return i * n - i * (i - 1) // 2 + (j - i)

def solve_lp(x, r, n):
    S = n / 2.0
    l = np.maximum(0.0, x - r)
    u = np.minimum(1.0, x + r)
    P = n * (n + 1) // 2
    nv = n + P + 1
    rows_list = []
    cols_list = []
    vals_list = []
    b_list = []
    for i in range(n):
        for j in range(i, n):
            p = prod_idx(i, j, n)
            zp = n + p
            li, ui = l[i], u[i]
            lj, uj = l[j], u[j]
            rr = len(b_list)
            cols_list += [zp, i, j, zp, i, j, zp, i, j, zp, i, j]
            vals_list += [-1.0, -li, -lj, -1.0, -ui, -uj, 1.0, -ui, -lj, 1.0, -li, -uj]
            rows_list += [rr, rr, rr, rr, rr, rr, rr, rr, rr, rr, rr, rr]
            b_list += [-li * lj, -ui * uj, -ui * lj, -li * uj]
    for k in range(-(n - 1), n):
        i_lo = max(0, k)
        i_hi = min(n - 1, n - 1 + k)
        row = len(b_list)
        for idx in range(i_lo, i_hi + 1):
            jj = idx - k
            p = prod_idx(min(idx, jj), max(idx, jj), n)
            cols_list += [idx, n + p]
            vals_list += [1.0, -1.0]
            rows_list += [row, row]
        cols_list.append(n + P)
        vals_list.append(-1.0)
        rows_list.append(row)
        b_list.append(0.0)
    A_ub = coo_matrix((vals_list, (rows_list, cols_list)), shape=(len(b_list), nv)).tocsr()
    A_eq = coo_matrix((np.ones(n), (np.arange(n), np.arange(n))), shape=(1, nv)).tocsr()
    b_eq = np.array([S])
    c_obj = np.zeros(nv)
    c_obj[n + P] = 1.0
    bounds = []
    for i in range(n):
        bounds.append((l[i], u[i]))
    for i in range(n):
        for j in range(i, n):
            bounds.append((l[i] * l[j], u[i] * u[j]))
    bounds.append((0.0, float(n)))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=np.array(b_list), A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status == 0 and res.x is not None and len(res.x) == nv:
            return np.array(res.x[:n], dtype=np.float64)
    except Exception:
        pass
    return None

def lag_coeffs(x, d, n):
    n_lags = 2 * n - 1
    A = np.zeros(n_lags)
    B = np.zeros(n_lags)
    D = np.zeros(n_lags)
    for k in range(-(n - 1), n):
        i_lo = max(0, k)
        i_hi = min(n - 1, n - 1 + k)
        idx = np.arange(i_lo, i_hi + 1)
        jj = idx - k
        A[k + n - 1] = -np.sum(d[idx] * d[jj])
        B[k + n - 1] = np.sum(d[idx] * (1.0 - x[jj]) - x[idx] * d[jj])
        D[k + n - 1] = np.sum(x[idx] * (1.0 - x[jj]))
    return A, B, D

def line_minimax(x, y, n):
    d = y - x
    A, B, D = lag_coeffs(x, d, n)
    n_lags = 2 * n - 1
    candidates = [0.0, 1.0]
    eps = 1e-10
    for k in range(n_lags):
        if abs(A[k]) > 1e-15:
            a = -B[k] / (2.0 * A[k])
            if -eps <= a <= 1.0 + eps:
                candidates.append(np.clip(a, 0.0, 1.0))
    for k in range(n_lags):
        for m in range(k + 1, n_lags):
            da = A[k] - A[m]
            db = B[k] - B[m]
            dc = D[k] - D[m]
            if abs(da) < 1e-14:
                if abs(db) > 1e-14:
                    a = -dc / db
                    if -eps <= a <= 1.0 + eps:
                        candidates.append(np.clip(a, 0.0, 1.0))
            else:
                disc = db * db - 4.0 * da * dc
                if disc >= -1e-13:
                    if disc < 0:
                        disc = 0.0
                    sq = np.sqrt(disc)
                    for a in [(-db + sq) / (2.0 * da), (-db - sq) / (2.0 * da)]:
                        if -eps <= a <= 1.0 + eps:
                            candidates.append(np.clip(a, 0.0, 1.0))
    candidates = sorted(set(np.round(candidates, 10).tolist()))
    deduped = [candidates[0]]
    for a in candidates[1:]:
        if a - deduped[-1] > 1e-8:
            deduped.append(a)
    candidates = deduped
    best_a = 0.0
    best_val = float('inf')
    for st in range(0, len(candidates), 2048):
        alphas = np.array(candidates[st:st + 2048])
        Q = A[:, None] * alphas[None, :] ** 2 + B[:, None] * alphas[None, :] + D[:, None]
        maxQ = np.max(Q, axis=0)
        for ci in range(len(maxQ)):
            if maxQ[ci] < best_val:
                best_val = maxQ[ci]
                best_a = alphas[ci]
    test_alphas = [0.0, 1.0, best_a]
    if 0.0 < best_a < 1.0:
        test_alphas += [best_a - 1e-5, best_a + 1e-5]
    best_h = None
    best_sc = float('inf')
    for a in test_alphas:
        a = np.clip(a, 0.0, 1.0)
        h_try = x + a * d
        h_proj = project_box_sum(h_try, n / 2.0)
        if h_proj is not None and check_feasible(h_proj, n):
            sc = exact_score(h_proj)
            if sc < best_sc:
                best_sc = sc
                best_h = h_proj
    return best_h, best_sc

def local_lp_search(start, n, S, max_attempts=8):
    current = project_box_sum(start.copy(), S)
    if current is None or not check_feasible(current, n):
        return None
    cur_score = exact_score(current)
    r = 0.20
    fails = 0
    for _ in range(max_attempts):
        if r < 1e-3 and fails >= 2:
            break
        endpoint = solve_lp(current, r, n)
        accepted = False
        if endpoint is not None:
            lb = np.maximum(0.0, current - r)
            ub = np.minimum(1.0, current + r)
            if np.all(endpoint >= lb - 1e-7) and np.all(endpoint <= ub + 1e-7):
                endpoint = np.clip(endpoint, 0.0, 1.0)
                ep_proj = project_box_sum(endpoint, S)
                if ep_proj is not None and check_feasible(ep_proj, n):
                    if np.max(np.abs(ep_proj - current)) > 1e-12:
                        new_h, new_sc = line_minimax(current, ep_proj, n)
                        if new_h is not None and new_sc < cur_score - 1e-12:
                            current = new_h
                            cur_score = new_sc
                            fails = 0
                            r = min(0.25, 1.2 * r)
                            accepted = True
        if not accepted:
            r *= 0.5
            fails += 1
    return current

        final_h = parent
        final_score = exact_score(parent)
    return (np.array(final_h, dtype=np.float64), final_score, n)
