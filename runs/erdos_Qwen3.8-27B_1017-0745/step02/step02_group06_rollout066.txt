    z = np.clip(z, -1e6, 1e6)

    for _ in range(80):

    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx])
            h[idx] += d
            r -= d
        else:
            d = min(-r, h[idx])
            h[idx] -= d
            r += d
    if abs(r) > 1e-10 * max(1, n) or not np.all(np.isfinite(h)):
        return None
    return np.clip(h, 0.0, 1.0)

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return float('inf')
    if abs(np.sum(h) - n / 2.0) > 1e-8:
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def lag_vals(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def lag_jac(h, n):
    dx = 2.0 / n
    n_lags = 2 * n - 1
    J = np.zeros((n_lags, n))
    for d in range(-(n - 1), n):
        p = d + n - 1
        if d >= 0:
            idx = np.arange(n - d)
            J[p, idx + d] += dx * (1.0 - h[idx])
            J[p, idx] -= dx * h[idx + d]
        else:
            e = -d
            idx = np.arange(n - e)
            J[p, idx] += dx * (1.0 - h[idx + e])
            J[p, idx + e] -= dx * h[idx]
    return J

def obj_func(z, n):
    return z[n]

def obj_grad(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def eq_fun(z, n):
    return np.sum(z[:n]) - n / 2.0

def eq_grad(z, n):
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def ineq_fun(z, n):
    h = z[:n]
    t = z[n]
    return t - lag_vals(h, n)

def ineq_grad(z, n):
    h = z[:n]
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n + 1))
    G[:, :n] = -lag_jac(h, n)
    G[:, n] = 1.0
    return G

def solve_slsqp(h0, n, maxiter, ftol):
    t0 = min(0.5, exact_score(h0, n) + 1e-9)
    z0 = np.concatenate([h0, [max(0.0, t0)]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 0.5)]
    cons = [
        {'type': 'eq', 'fun': lambda z: eq_fun(z, n), 'jac': lambda z: eq_grad(z, n)},
        {'type': 'ineq', 'fun': lambda z: ineq_fun(z, n), 'jac': lambda z: ineq_grad(z, n)},
    ]
    try:
        res = minimize(obj_func, z0, jac=lambda z: obj_grad(z, n),
                       bounds=bounds, constraints=cons,
                       method='SLSQP', options={'ftol': ftol, 'maxiter': maxiter, 'disp': False})
    except Exception:
        return None, None
    if res.x is None or len(res.x) != n + 1 or not np.all(np.isfinite(res.x)):
        return None, None
    h_new = project_capped(res.x[:n], n)
    if h_new is None:
        return None, None
    return h_new, exact_score(h_new, n)

def build_starts(parent, n, rng):
    pool = []
    h_unif = np.full(n, 0.5)
    pool.append(h_unif)
    if parent is not None:
        pool.append(parent.copy())
        pool.append(parent[::-1].copy())
        pool.append((1.0 - parent).copy())
        pool.append((1.0 - parent[::-1]).copy())
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    pool.append(blk)
    pool.append(np.where(np.arange(n) % 2 == 0, 1.0, 0.0))
    for m in [1, 2, 3]:
        for ph in [0.0, np.pi / 2]:
            pool.append(0.5 + 0.45 * np.cos(2.0 * np.pi * m * np.arange(n) / n + ph))
    base = parent if parent is not None else h_unif
    for sd in [0.04, 0.10, 0.20]:
        pool.append(base + rng.normal(0, sd, n))
    pool.append(rng.uniform(0, 1, n))
    seen = []
    valid = []
    for c in pool:
        p = project_capped(np.asarray(c, dtype=np.float64), n)
        if p is None:
            continue
        dup = False
        for s in seen:
            if np.max(np.abs(p - s)) < 1e-6:
                dup = True
                break
        if not dup:
            seen.append(p)
            sc = exact_score(p, n)
            if np.isfinite(sc) and sc <= 0.5 + 1e-6:
                valid.append((sc, p))
    valid.sort(key=lambda x: x[0])
    result = [h_unif]
    for sc, p in valid:
        if p is result[0]:
            continue
        dup = any(np.max(np.abs(p - s)) < 1e-6 for s in result)
        if not dup:
            result.append(p)
        if len(result) >= 12:
            break
    return result

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                if abs(np.sum(p) - n / 2.0) <= 1e-6 * n:
                    parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None:
        n = 47
    if n == 1:
        return np.array([0.5]), 0.5, 1
    h_unif = np.full(n, 0.5)
    if parent is not None:
        parent = project_capped(parent, n)
    fallback = h_unif
    fb_score = exact_score(h_unif, n)
    if parent is not None:
        ps = exact_score(parent, n)
        if np.isfinite(ps) and ps < fb_score:
            fallback = parent
            fb_score = ps
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 64:
        n_starts, n_restarts, maxiter = 8, 4, 220
    elif n <= 128:
        n_starts, n_restarts, maxiter = 5, 2, 150
    else:
        n_starts, n_restarts, maxiter = 3, 1, 80
    ftol = 1e-10
    starts = build_starts(parent, n, rng)
    n_starts = min(n_starts, len(starts))
    inc_best = fallback.copy()
    inc_score = fb_score
    for h0 in starts[:n_starts]:
        h_new, sc = solve_slsqp(h0, n, maxiter, ftol)
        if h_new is not None and np.isfinite(sc) and sc < inc_score - 1e-12:
            for alpha in [0.25, 0.5, 0.75, 1.0]:
                blend = project_capped(alpha * h_new + (1 - alpha) * inc_best, n)
                if blend is not None:
                    bsc = exact_score(blend, n)
                    if np.isfinite(bsc) and bsc < inc_score - 1e-12:
                        inc_best, inc_score = blend, bsc
            if sc < inc_score - 1e-12:
                inc_best, inc_score = h_new.copy(), sc
    sigma = 0.04
    for _ in range(n_restarts):
        used = False
        for _att in range(3):
            pert = inc_best + rng.normal(0, sigma, n)
            hp = project_capped(pert, n)
            if hp is None:
                continue
            sp = exact_score(hp, n)
            if np.isfinite(sp) and sp <= 0.5 + 1e-6:
                if all(np.max(np.abs(hp - s)) > 1e-4 for s in [inc_best]):
                    used = True
                    break
        if not used:
            sigma = min(0.20, 1.5 * sigma)
            continue
        h_new, sc = solve_slsqp(hp, n, maxiter, ftol)
        improved = False
        if h_new is not None and np.isfinite(sc) and sc < inc_score - 1e-12:
            for alpha in [0.25, 0.5, 0.75, 1.0]:
                blend = project_capped(alpha * h_new + (1 - alpha) * inc_best, n)
                if blend is not None:
                    bsc = exact_score(blend, n)
                    if np.isfinite(bsc) and bsc < inc_score - 1e-12:
                        inc_best, inc_score = blend, bsc
            if sc < inc_score - 1e-12:
                inc_best, inc_score = h_new.copy(), sc
            improved = True
        if improved:
            sigma = max(0.01, 0.7 * sigma)
        else:
            sigma = min(0.20, 1.5 * sigma)
    final = project_capped(inc_best, n)
    if final is None or not np.all(np.isfinite(final)):
        final = fallback
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score + 1e-12:
        final = fallback
        fs = fb_score
    return final, float(fs), int(n)
