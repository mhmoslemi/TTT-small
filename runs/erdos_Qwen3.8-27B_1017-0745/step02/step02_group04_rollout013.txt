    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 64:
        n_starts, maxiter, n_hops, n_refine = 10, 300, 8, 3
    elif n <= 128:
        n_starts, maxiter, n_hops, n_refine = 7, 200, 5, 2
    else:
        n_starts, maxiter, n_hops, n_refine = 4, 120, 2, 1
    starts = build_starts(fallback, n, rng, n_starts)
    gb = fallback.copy()
    gs = float(exact_score(fallback, n))
    for st in starts:
        s = float(exact_score(st, n))
        if s < gs - 1e-12:
            gs, gb = s, st.copy()
    for st in starts:
        try:
            y = solve_exact(st, n, maxiter)
            if y is not None and is_feasible(y, n):
                sy = float(exact_score(y, n))
                if sy < gs - 1e-12:
                    gs, gb = sy, y.copy()
        except Exception:
            pass
    if n_hops > 0:
        current = gb.copy()
        current_s = gs
        T = 1.5e-3
        sigmas = [0.04, 0.10, 0.20]
        for r in range(n_hops):
            if r % 4 == 3:
                prop = project_capped(rng.uniform(0, 1, n), n)
            else:
                sig = sigmas[r % 3]
                prop = project_capped(current + sig * rng.standard_normal(n), n)
            if prop is None:
                continue
            try:
                y = solve_exact(prop, n, maxiter)
            except Exception:
                y = None
            if y is None or not is_feasible(y, n):
                continue
            sy = float(exact_score(y, n))
            if sy < gs - 1e-12:
                gs, gb = sy, y.copy()
            if sy < current_s - 1e-12:
                current, current_s = y.copy(), sy
            elif sy <= current_s + 1e-12:
                current, current_s = y.copy(), sy
            else:
                acc = np.exp(-(sy - current_s) / max(T, 1e-7))
                if rng.random() < acc:
                    current, current_s = y.copy(), sy
            T *= 0.8
    if n_refine > 0:
        gb2, gs2 = coord_refine(gb, n, n_refine)
        if gs2 < gs - 1e-14:
            gb, gs = gb2, gs2
    gb = project_capped(gb, n)
    if gb is None or not is_feasible(gb, n):
        gb = fallback.copy()
    fs = float(exact_score(gb, n))
    fb = float(exact_score(fallback, n))
    if fs > fb + 1e-12:
        gb, fs = fallback.copy(), fb
    if fs == float('inf'):
        gb = np.full(n, 0.5)
        fs = float(exact_score(gb, n))
    return gb, fs, int(n)
