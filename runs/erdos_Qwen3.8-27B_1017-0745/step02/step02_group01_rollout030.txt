    blk = np.zeros(n); blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    add_unique_start(pool, blk, n, 10)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    add_unique_start(pool, project_capped(alt, n), n, 10)
    t = np.arange(n, dtype=np.float64) / n
    add_unique_start(pool, project_capped(0.5 + 0.35 * np.cos(2.0 * np.pi * t), n), n, 10)
    add_unique_start(pool, project_capped(0.5 + 0.3 * np.cos(4.0 * np.pi * t), n), n, 10)
    add_unique_start(pool, project_capped(fallback + rng.normal(0, 0.05, n), n), n, 10)
    add_unique_start(pool, project_capped(fallback + rng.normal(0, 0.15, n), n), n, 10)
    return pool

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None:
        try:
            v = int(kwargs.get('n_points', 47))
            if 1 <= v <= 999:
                n = v
        except Exception:
            n = 47
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if parent is not None and parent.shape[0] == n:
        proj = project_capped(parent, n)
        fallback = proj if proj is not None else np.full(n, 0.5)
    else:
        fallback = np.full(n, 0.5)
    if not is_feasible(fallback, n):
        fallback = np.full(n, 0.5)
    fb_score = exact_score(fallback, n)
    if n <= 1:
        return fallback, fb_score, int(n)
    if n > 64:
        return fallback, fb_score, int(n)
    starts = build_starts(fallback, n, rng)
    scored = []
    for st in starts:
        s = exact_score(st, n)
        scored.append((s, st))
    scored.sort(key=lambda x: x[0])
    best_h, best_score = fallback.copy(), fb_score
    for s, st in scored:
        if s < best_score - 1e-12:
            best_h, best_score = st.copy(), s
    dirs = build_directions(n)
    B = min(276, max(n - 1, int(276.0 * (47.0 / n) ** 2)))
    nc = min(len(scored), 3, max(1, B // max(1, n - 1)))
    per = B // nc
    rem = B - per * nc
    for ci in range(nc):
        sel = scored[ci % len(scored)][1]
        bc = per + (1 if ci < rem else 0)
        best_h, best_score = refine_chain(sel, n, dirs, bc, best_h, best_score)
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    if fs > fb_score + 1e-12:
        final, fs = fallback, fb_score
    return final, float(fs), int(n)
