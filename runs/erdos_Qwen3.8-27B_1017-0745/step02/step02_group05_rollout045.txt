def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):

        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:

        else:

    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)

    add(blk[::-1].copy())
    for sd in (0.05, 0.15, 0.30):
        add(base + rng.normal(0, sd, n))
    for freq, ph in [(1, 0.5), (2, 1.5), (3, 2.5)]:
        t = 0.5 + 0.35 * np.sin(2.0 * np.pi * freq * np.arange(n) / n + ph)
        add(t)
    add(rng.uniform(0, 1, n))
    add(base + rng.uniform(-0.1, 0.1, n))
    return pool[:count]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        try:
            v = int(kwargs.get('n_points', 47))
            n = v if 1 <= v <= 999 else 47
        except Exception:
            n = 47
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        h1 = np.array([0.5])
        return h1, exact_score(h1, 1), 1
    fb = project_capped(np.full(n, 0.5), n)
    if fb is None:
        fb = np.full(n, 0.5)
    safe_h = fb.copy()
    safe_s = exact_score(fb, n)
    if parent is not None:
        pp = project_capped(parent, n)
        if pp is not None and is_feasible(pp, n):
            ps = exact_score(pp, n)
            if ps < safe_s:
                safe_h, safe_s = pp.copy(), ps
    base = safe_h.copy()
    if n <= 64:
        n_starts, n_sweeps, n_plim = 6, 12, n * (n - 1) // 2
    elif n <= 256:
        n_starts, n_sweeps, n_plim = 4, 8, min(n * (n - 1) // 2, 4 * n)
    else:
        n_starts, n_sweeps, n_plim = 2, 5, min(n * (n - 1) // 2, 2 * n)
    starts = build_starts(base, n, rng, n_starts)
    gb, gs = safe_h.copy(), safe_s
    for st in starts:
        ss = exact_score(st, n)
        if ss < gs - 1e-12:
            gs, gb = ss, st.copy()
    for st in starts:
        try:
            ho, so = coordinate_sweep(st, n, rng, n_sweeps, n_plim)
            if ho is not None and is_feasible(ho, n) and so < gs - 1e-12:
                gs, gb = so, ho.copy()
        except Exception:
            pass
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = safe_h.copy()
    fs = exact_score(final, n)
    if fs > safe_s + 1e-12:
        final, fs = safe_h.copy(), safe_s
    return final.astype(np.float64), float(fs), int(n)
