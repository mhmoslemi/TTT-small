    hi = np.max(z) + 1.0
    for _ in range(80):

        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(3):

def lag_coefficients(h, d, n):

    a = np.zeros(2 * n - 1)
    b = np.zeros(2 * n - 1)
    c = np.zeros(2 * n - 1)
    # Zero lag: Q0 = dx * sum(h*(1-h))
    a[0] = dx * np.sum(h * (1.0 - h))
    b[0] = dx * np.sum(d * (1.0 - 2.0 * h))

        idx_p = 2 * r - 1  # positive lag
        idx_n = 2 * r      # negative lag
        h0, h1 = h[:m], h[r:]
        d0, d1 = d[:m], d[r:]
        # Q_{r,+} = dx * sum(h1*(1-h0))
        a[idx_p] = dx * np.sum(h1 * (1.0 - h0))

        b[idx_n] = dx * np.sum(d0 * (1.0 - h1) - h0 * d1)
        c[idx_n] = -dx * np.sum(d0 * d1)

    grads = []

    g0 = dx * (1.0 - 2.0 * h)
