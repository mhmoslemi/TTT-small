            cands.append(0.5 + 0.45 * np.cos(2.0 * math.pi * m * np.arange(n) / n + ph))
    for sd in (0.06, 0.15, 0.30):
        cands.append(base + rng.normal(0, sd, n))
    cands.append(rng.uniform(0, 1, n))
    for c in cands:
        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-7:
                dup = True; break
        if not dup:
            pool.append(p)
        if len(pool) >= count + 6:
            break
    for _ in range(20):
        if len(pool) >= count:
            break
        p = project_capped(base + rng.normal(0, 0.15, n), n)
        if p is None:
            continue
        if all(np.max(np.abs(p - s)) >= 1e-7 for s in pool):
            pool.append(p)
    return pool[:count]

def pair_polish(h, n, rng, nrounds):
    h = h.copy()
    s = exact_score(h, n)
    for _ in range(nrounds):
        best_imp = 1e-13
        best_move = None
        npairs = min(3000, 6 * n)
        ii = rng.integers(0, n, npairs)
        jj = rng.integers(0, n, npairs)
        mask = ii < jj
        ii, jj = ii[mask], jj[mask]
        for pi in range(min(80, len(ii))):
            i, j = int(ii[pi]), int(jj[pi])
            L = max(-h[i], h[j] - 1.0)
            U = min(1.0 - h[i], h[j])
            if U - L < 1e-12:
                continue
            fr = U - L
            for fr2 in (0.2, 0.5, 0.8, -0.2, -0.5, -0.8):
                t = L + fr2 * fr
                if t < L - 1e-14 or t > U + 1e-14:
                    continue
                ht = h.copy()
                ht[i] += t; ht[j] -= t
                pp = project_capped(ht, n)
                if pp is None:
                    continue
                s2 = exact_score(pp, n)
                if s2 < s - best_imp:
                    best_imp = s - s2
                    best_move = (pp.copy(), s2)
        if best_move is None:
            break
        h, s = best_move
    return h, s

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 1 and p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0, 1)
    except Exception:
        pass
    if n is None:
        n = 47
    fallback = project_capped(parent, n) if parent is not None else np.full(n, 0.5)
    if fallback is None:
        fallback = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 1:
        return np.full(1, 0.5), 0.5, 1
    if n <= 64:
        n_starts, mxit, n_restarts, n_pol = 8, 120, 3, 5
    elif n <= 128:
        n_starts, mxit, n_restarts, n_pol = 6, 80, 2, 4
    else:
        n_starts, mxit, n_restarts, n_pol = 4, 50, 2, 3
    taus = [0.01, 0.003, 0.001, 0.0003, 0.0001, 0.00003]
    starts = build_starts(fallback, n, rng, n_starts)
    best_h = fallback.copy()
    best_s = exact_score(fallback, n)
    scored = []
    for st in starts:
        s = exact_score(st, n)
        scored.append((s, st))
    scored.sort(key=lambda x: x[0])
    for s, st in scored:
        if s < best_s - 1e-12:
            best_s, best_h = s, st.copy()
    for s, st in scored:
        _, best_h, best_s = proj_descend(st, n, taus, mxit, best_h, best_s)
    sigma = 0.10
    for _ in range(n_restarts):
        pert = best_h + rng.normal(0, sigma, n)
        pp = project_capped(pert, n)
        if pp is None or np.max(np.abs(pp - best_h)) < 1e-5:
            sigma = min(0.30, sigma * 1.5)
            continue
        prev_s = best_s
        _, best_h, best_s = proj_descend(pp, n, taus, mxit, best_h, best_s)
        if best_s < prev_s - 1e-12:
            sigma = max(0.01, sigma * 0.7)
        else:
            sigma = min(0.35, sigma * 1.5)
    ph, ps = pair_polish(best_h, n, rng, n_pol)
    if ps < best_s - 1e-12:
        best_h, best_s = ph, ps
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    if fs > best_s + 1e-6:
        final = fallback
        fs = exact_score(final, n)
    return final, float(fs), int(n)
