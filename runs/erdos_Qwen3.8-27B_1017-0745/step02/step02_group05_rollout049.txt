def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,): return None
    z = np.where(np.isfinite(z), z, 0.0)
    z = np.clip(z, -1e6, 1e6)
    s = 0.5 * n
    lo = float(np.min(z) - 1.0)
    hi = float(np.max(z) + 1.0)
    for _ in range(80):
       mid = 0.5*(lo+hi)
       val = np.sum(np.clip(z - mid, 0.0, 1.0))
       if val > s:
          lo = mid
       else:
          hi = mid
    lam = 0.5*(lo+hi)
    h = np.clip(z - lam, 0.0, 1.0)
    r = s - float(np.sum(h))
    tol = 2e-14 * max(1, n)
    if r > 0:
       for i in range(n):
           if r <= tol: break
           add = min(r, 1.0 - h[i])
           if add > 0:
              h[i] += add; r -= add
    else:
       rr = -r
       if rr > tol:
          for i in range(n):
             if rr <= tol: break
             sub = min(rr, h[i])
             if sub > 0:
                h[i] -= sub; rr -= sub
    if not np.all(np.isfinite(h)): return None
    if np.any(h < -1e-12) or np.any(h > 1+1e-12): return None
    if abs(float(np.sum(h)) - s) > 1e-9*max(1,n): return None
    return h
```

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,):
        return None
    if not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = 0.5 * float(n)
    lo = float(np.min(z) - 1.0)
    hi = float(np.max(z) + 1.0)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(z - lam, 0.0, 1.0)
    for _ in range(n + 2):
        r = s - float(np.sum(h))
        if abs(r) <= 1e-13 * max(1, n):
            break
        if r > 0.0:
            i = int(np.argmax(1.0 - h))
            add = min(r, 1.0 - h[i])
            if add <= 0.0:
                break
            h[i] += add
        else:
            i = int(np.argmax(h))
            sub = min(-r, h[i])
            if sub <= 0.0:
                break
            h[i] -= sub
    if not np.all(np.isfinite(h)):
        return None
    if np.min(h) < -1e-11 or np.max(h) > 1.0 + 1e-11:
        return None
    if abs(float(np.sum(h)) - s) > 1e-9 * max(1, n):
        return None

import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint

_PEN = 1.0e9


def _as_valid_box(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return None
    if h.min() < -1.0e-7 or h.max() > 1.0 + 1.0e-7:
        return None
    if h.min() < 0.0 or h.max() > 1.0:
        h = np.clip(h, 0.0, 1.0)
    return h


def balance_vector(c, n):
    c = np.asarray(c, dtype=np.float64).ravel().copy()
    if c.shape != (n,) or not np.all(np.isfinite(c)):
        return None
    c = np.clip(c, 0.0, 1.0)
    s = 0.5 * float(n)
    for _ in range(n + 2):
        r = s - float(np.sum(c))
        if abs(r) <= 1.0e-13 * max(1, n):
            break
        if r > 0.0:
            i = int(np.argmin(c))
            add = min(r, 1.0 - c[i])
            if add <= 1.0e-15:
                break
            c[i] += add
        else:
            i = int(np.argmax(c))
            sub = min(-r, c[i])
            if sub <= 1.0e-15:
                break
            c[i] -= sub
    if abs(float(np.sum(c)) - s) > 1.0e-9 * max(1, n):
        return None
    return c


def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1.0e6, 1.0e6)
    s = 0.5 * float(n)
    lo = float(np.min(z) - 1.0)
    hi = float(np.max(z) + 1.0)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(z - lam, 0.0, 1.0)
    for _ in range(n + 2):
        r = s - float(np.sum(h))
        if abs(r) <= 1.0e-13 * max(1, n):
            break
        if r > 0.0:
            i = int(np.argmax(1.0 - h))
            add = min(r, 1.0 - h[i])
            if add <= 0.0:
                break
            h[i] += add
        else:
            i = int(np.argmax(h))
            sub = min(-r, h[i])
            if sub <= 0.0:
                break
            h[i] -= sub
    if not np.all(np.isfinite(h)):
        return None
    if np.min(h) < -1.0e-11 or np.max(h) > 1.0 + 1.0e-11:
        return None
    if abs(float(np.sum(h)) - s) > 1.0e-9 * max(1, n):
        return None
    return h


def is_feasible(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if h.min() < -1.0e-10 or h.max() > 1.0 + 1.0e-10:
        return False
    if abs(float(np.sum(h)) - 0.5 * float(n)) > 1.0e-8 * max(1, n):
        return False
    return True


def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if h.min() < -1.0e-9 or h.max() > 1.0 + 1.0e-9:
        return float('inf')
    if abs(float(np.sum(h)) - 0.5 * float(n)) > 1.0e-7 * max(1, n):
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / float(n))
    if not np.all(np.isfinite(q)):
        return float('inf')
    return float(np.max(q))


def smooth_value(h, n, tau):
    h = _as_valid_box(h, n)
    if h is None or tau <= 1.0e-14:
        return _PEN
    dx = 2.0 / float(n)
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    if not np.all(np.isfinite(q)):
        return _PEN
    m = float(np.max(q))
    ex = np.exp(np.maximum((q - m) / tau, -740.0))
    s = float(np.sum(ex))
    if s <= 0.0 or not np.isfinite(s):
        return _PEN
    val = m + tau * float(np.log(s))
    return val if np.isfinite(val) else _PEN


def smooth_grad(h, n, tau):
    n = int(n)
    g = np.zeros(n, dtype=np.float64)
    h = _as_valid_box(h, n)
    if h is None or tau <= 1.0e-14:
        return g
    dx = 2.0 / float(n)
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    if not np.all(np.isfinite(q)):
        return g
    m = float(np.max(q))
    ex = np.exp(np.maximum((q - m) / tau, -740.0))
    s = float(np.sum(ex))
    if s <= 0.0 or not np.isfinite(s):
        return g
    w = ex / s
    for idx in range(w.size):
        wd = float(w[idx])
        if wd <= 1.0e-15:
            continue
        d = idx - (n - 1)
        ad = abs(d)
        cnt = n - ad
        if cnt <= 0:
            continue
        p0 = d if d > 0 else 0
        q0 = -d if d < 0 else 0
        scaling = wd * dx
        g[p0:p0 + cnt] += scaling * (1.0 - h[q0:q0 + cnt])
        g[q0:q0 + cnt] -= scaling * h[p0:p0 + cnt]
    if not np.all(np.isfinite(g)):
        return np.zeros(n, dtype=np.float64)
    return g


def try_add_start(pool, c, n):
    p = project_capped(c, n)
    if p is None or not is_feasible(p, n):
        return
    sc = exact_score(p, n)
    if not np.isfinite(sc):
        return
    for q in pool:
        if np.max(np.abs(p - q)) < 1.0e-7:
            return
    pool.append(p)


def build_starts(n, rng, fallback, parent, count):
    pool = []
    idx = np.arange(n, dtype=np.float64)

    if parent is not None:
        try_add_start(pool, parent, n)

    a = (5.0 ** 0.5 - 1.0) / 2.0
    c = (np.mod(idx * float(a), 1.0) < 0.5).astype(np.float64)
    cb = balance_vector(c, n)
    if cb is not None:
        c = cb
    try_add_start(pool, c, n)

    if n % 2 == 0:
        rb = np.zeros(n, dtype=np.float64)
        rb[rng.choice(n, n // 2, replace=False)] = 1.0
    else:
        k = n // 2
        pos = rng.choice(n, k + 1, replace=False)
        rb = np.zeros(n, dtype=np.float64)
        rb[pos] = 1.0
        rb[pos[int(rng.integers(k + 1))]] = 0.5
    try_add_start(pool, rb, n)

    blk = np.zeros(n, dtype=np.float64)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    try_add_start(pool, blk, n)

    if parent is not None:
        try_add_start(pool, parent + rng.normal(0.0, 0.08, n), n)

    try_add_start(pool, fallback, n)

    c = (np.mod(idx * 0.5, 1.0) < 0.5).astype(np.float64)
    cb = balance_vector(c, n)
    if cb is not None:
        c = cb
    try_add_start(pool, c, n)

    a = np.sqrt(2.0) - 1.0
    c = (np.mod(idx * float(a), 1.0) < 0.5).astype(np.float64)
    cb = balance_vector(c, n)
    if cb is not None:
        c = cb
    try_add_start(pool, c, n)

    a = 0.3819660112501051
    c = (np.mod(idx * float(a), 1.0) < 0.5).astype(np.float64)
    cb = balance_vector(c, n)
    if cb is not None:
        c = cb
    try_add_start(pool, c, n)

    try_add_start(pool, rng.uniform(0.0, 1.0, n), n)

    phase = float(rng.uniform(0.0, 2.0 * np.pi))
    c = 0.5 + 0.25 * np.sin(2.0 * np.pi * 1.0 * idx / float(n) + phase)
    try_add_start(pool, c, n)

    phase = float(rng.uniform(0.0, 2.0 * np.pi))
    c = 0.5 + 0.30 * np.sin(2.0 * np.pi * 2.0 * idx / float(n) + phase)
    try_add_start(pool, c, n)

    if len(pool) > count:
        kept = []
        for c in pool:
            if len(kept) >= count:
                break
            kept.append(c)
        pool = kept
    if len(pool) == 0:
        pool.append(fallback.copy())
    return pool


def optimize_smooth(x0, n, taus, maxiter, ftol=1.0e-10):
    n = int(n)
    x = project_capped(x0, n)
    if x is None or not is_feasible(x, n):
        x = project_capped(np.full(n, 0.5, dtype=np.float64), n)
    if x is None:
        return None, float('inf')

    best_s = exact_score(x, n)
    best_x = x.copy()
    cx = x.copy()
    mass = 0.5 * float(n)
    bounds = Bounds(np.zeros(n, dtype=np.float64), np.ones(n, dtype=np.float64))
    lin = LinearConstraint(np.ones((1, n), dtype=np.float64), np.array([mass]), np.array([mass]))
    step_alphas = (1.0, 0.5, 0.25, 0.125, 0.0625)

    for tau in taus:
        tau = float(tau)
        if tau <= 0.0:
            continue
        res_x = None
        try:
            res = minimize(smooth_value, cx, jac=smooth_grad, args=(n, tau),
                           method='SLSQP', bounds=bounds, constraints=lin,
                           options={'maxiter': int(maxiter), 'ftol': float(ftol), 'disp': False})
            if hasattr(res, 'x'):
                res_x = np.asarray(res.x, dtype=np.float64).ravel()
        except Exception:
            res_x = None

        cur_phi = smooth_value(cx, n, tau)
        cur_score = exact_score(cx, n)
        best_key = cur_phi + 1.0e-4 * cur_score
        chosen = cx
        be_z = None
        be_score = cur_score
        be_phi = cur_phi

        if res_x is not None and res_x.shape == (n,) and np.all(np.isfinite(res_x)):
            y = project_capped(res_x, n)
            if y is not None and is_feasible(y, n):
                for alpha in step_alphas:
                    if alpha >= 1.0:
                        z = y
                    else:
                        z = project_capped(cx + alpha * (y - cx), n)
                    if z is None or not is_feasible(z, n):
                        continue
                    zs = exact_score(z, n)
                    if np.isfinite(zs) and zs < best_s - 1.0e-12:
                        best_s = zs
                        best_x = z.copy()
                    ph = smooth_value(z, n, tau)
                    if np.isfinite(ph):
                        key = ph + 1.0e-4 * zs
                        if key < best_key - 1.0e-12 or (abs(key - best_key) <= 1.0e-12 and zs < cur_score):
                            best_key = key
                            chosen = z
                        if zs < be_score and ph <= cur_phi + 5.0e-3:
                            be_score = zs
                            be_phi = ph
                            be_z = z

        if be_z is not None and be_score < cur_score - 1.0e-10 and np.max(np.abs(be_z - cx)) > 1.0e-13:
            chosen = be_z

        if np.max(np.abs(chosen - cx)) > 1.0e-13:
            cx = chosen.copy()

    return best_x, float(best_s)


def local_random_project(h, n, rng, trials, sigma0):
    n = int(n)
    h = project_capped(h, n)
    if h is None:
        return None, float('inf')
    s = exact_score(h, n)
    sigma = float(sigma0)
    for _ in range(int(trials)):
        sigma *= 0.985
        if sigma < 1.0e-6:
            break
        z = h + rng.normal(0.0, sigma, n)
        zp = project_capped(z, n)
        if zp is None or not is_feasible(zp, n):
            continue
        sz = exact_score(zp, n)
        if sz < s - 1.0e-12:
            h = zp
            s = sz
    return h, float(s)


def local_pair_swap(h, n, rng, trials, step):
    n = int(n)
    h = project_capped(h, n)
    if h is None:
        return None, float('inf')
    s = exact_score(h, n)
    step = float(step)
    for _ in range(int(trials)):
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        delta = float(rng.uniform(-step, step))
        if 0.0 <= h[i] + delta and h[j] - delta <= 1.0:
            h2 = h.copy()
            h2[i] += delta
            h2[j] -= delta
            hz = balance_vector(h2, n)
            if hz is None:
                hz = project_capped(h2, n)
            if hz is None or not is_feasible(hz, n):
                continue
            s2 = exact_score(hz, n)
            if s2 < s - 1.0e-12:
                h = hz
                s = s2
    return h, float(s)


def subgrad_exact(h, n, iters, eta0):
    n = int(n)
    h = project_capped(h, n)
    if h is None:
        return None, float('inf')
    s = exact_score(h, n)
    eta = float(eta0)
    dx = 2.0 / float(n)
    for k in range(int(iters)):
        q = np.correlate(h, 1.0 - h, mode='full') * dx
        if not np.all(np.isfinite(q)):
            break
        qmax = float(np.max(q))
        active = np.where(q >= qmax - 1.0e-3)[0]
        if active.size == 0:
            active = np.array([0], dtype=int)
        g = np.zeros(n, dtype=np.float64)
        for idx in active:
            d = int(idx) - (n - 1)
            cnt = n - abs(d)
            if cnt <= 0:
                continue
            p0 = d if d > 0 else 0
            q0 = -d if d < 0 else 0
            g[p0:p0 + cnt] += dx * (1.0 - h[q0:q0 + cnt])
            g[q0:q0 + cnt] -= dx * h[p0:p0 + cnt]
        ng = float(np.linalg.norm(g))
        if ng <= 1.0e-14:
            break
        g = g / ng

        i = int(np.argmax(g))
        j = int(np.argmin(g))
        if g[i] > g[j] + 1.0e-10:
            delta = min(0.02, float(h[i]), float(1.0 - h[j]))
            if delta > 1.0e-12:
                h3 = h.copy()
                h3[i] -= delta
                h3[j] += delta
                hbb = balance_vector(h3, n)
                if hbb is None:
                    hbb = project_capped(h3, n)
                if hbb is not None and is_feasible(hbb, n):
                    ss = exact_score(hbb, n)
                    if ss < s - 1.0e-13:
                        h = hbb
                        s = ss

        step = eta / (1.0 + 0.25 * np.sqrt(float(k + 1)))
        h2 = project_capped(h - step * g, n)
        if h2 is None or not is_feasible(h2, n):
            h2 = project_capped(h - 0.1 * step * g, n)
        if h2 is None or not is_feasible(h2, n):
            continue
        s2 = exact_score(h2, n)
        if s2 < s - 1.0e-13:
            h = h2
            s = s2
    return h, float(s)

        h5, s5 = local_random_project(global_best, n, rng, lr2, 0.012)
        if h5 is not None and np.isfinite(s5) and s5 < global_best_s - 1.0e-12:
            global_best = h5.copy()
            global_best_s = s5
    except Exception:
        pass

    try:
        h6, s6 = local_pair_swap(global_best, n, rng, lp2, lp2_step)
        if h6 is not None and np.isfinite(s6) and s6 < global_best_s - 1.0e-12:
            global_best = h6.copy()
            global_best_s = s6
    except Exception:
        pass

    final = project_capped(global_best, n)
    if final is None or not is_feasible(final, n):
        final = fb.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_s + 1.0e-12:
        final = fb.copy()
        fs = framebuffer_score if 'framebuffer_score' in dir() else fb_s
        fs = fb_s
    if not np.isfinite(fs):
        final = fallback.copy()
        fs = exact_score(final, n)

    return final.astype(np.float64), float(fs), int(n)
