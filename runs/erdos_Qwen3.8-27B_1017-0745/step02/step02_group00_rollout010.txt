        final, fs = uniform.copy(), exact_score(uniform, n)
    else:
        fs = exact_score(final, n)
    # Also check parent
    if parent is not None:
        pp = project_capped(parent, n)
        if pp is not None:
            ps = exact_score(pp, n)
            if ps < fs:
                final, fs = pp, ps
    final = project_capped(final, n)
    fs = exact_score(final, n)
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final.astype(np.float64), float(fs), int(n)
