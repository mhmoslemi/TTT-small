            bh, bf = local_search(st, n, n_it, 0.25)
            if bf < gs - 1e-12:
                gs, gb = bf, bh.copy()
        except Exception:
            pass
    if gs < exact_score(fb, n) - 1e-5:
        sd_list = [0.04, 0.10, 0.18]
    else:
        sd_list = [0.12, 0.22, 0.08]
    for pt in range(n_pr):
        sd = sd_list[pt % len(sd_list)]
        try:
            pert = project_capped(gb + rng.normal(0, sd, n), n)
            if pert is None:
                continue
            bh, bf = local_search(pert, n, n_it, 0.25)
            if bf < gs - 1e-12:
                gs, gb = bf, bh.copy()
        except Exception:
            pass
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb
    sf = exact_score(final, n)
    ss = exact_score(fb, n)
    if sf > ss:
        final, sf = fb, ss
    if not is_feasible(final, n):
        final = np.full(n, 0.5)
        sf = exact_score(final, n)
    return final, float(sf), int(n)
