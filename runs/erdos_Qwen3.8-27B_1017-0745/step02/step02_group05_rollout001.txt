    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for i in range(n):
        if abs(r) < 2e-14 * max(1, n):
            break
        if r > 0:
            add = min(r, 1.0 - h[i])
            h[i] += add
            r -= add
        else:
            sub = min(-r, h[i])
            h[i] -= sub
            r += sub
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(np.sum(h) - s) > 1e-10 * max(1, n):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def smooth_value(h, n, tau):
    dx = 2.0 / n
    Q = dx * np.correlate(h, 1.0 - h, mode='full')
    M = np.max(Q)
    e = np.maximum((Q - M) / tau, -745.0)
    w = np.exp(e)
    s = np.sum(w)
    if s < 1e-300:
        return M
    return M + tau * np.log(s)

def smooth_grad(h, n, tau):
    dx = 2.0 / n
    Q = dx * np.correlate(h, 1.0 - h, mode='full')
    M = np.max(Q)
    e = np.maximum((Q - M) / tau, -745.0)
    w = np.exp(e)
    ws = np.sum(w)
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-15:
            continue
        d = idx - (n - 1)
        a = max(d, 0)
        b = max(-d, 0)
        m = n - abs(d)
        grad[a:a + m] += w[idx] * dx * (1.0 - h[b:b + m])
        grad[b:b + m] -= w[idx] * dx * h[a:a + m]
    return grad

def build_starts(fallback, n, rng, count):
    pool = [fallback.copy()]
    cands = [fallback[::-1].copy(), 1.0 - fallback,
             np.roll(fallback, max(1, n // 4)), np.roll(fallback, -max(1, n // 4)),
             np.full(n, 0.5)]
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    cands.append(blk)
    cands.append(np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64))
    for sd in (0.05, 0.12, 0.25):
        cands.append(fallback + rng.normal(0, sd, n))
    ph = rng.uniform(0, 2 * np.pi)
    sv = np.sin(2 * np.pi * np.arange(n) / n + ph)
    sv -= np.mean(sv)
    cands.append(fallback + 0.08 * sv)
    cands.append(fallback + 0.15 * np.sin(4 * np.pi * np.arange(n) / n + rng.uniform(0, 6.28)))
    cands.append(rng.uniform(0, 1, n))
    for c in cands:
        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-8:
                dup = True
                break
        if not dup:
            pool.append(p)
        if len(pool) >= count:
            break
    return pool[:count]

def optimize_start(x0, n):
    x = project_capped(x0, n)
    if x is None or not is_feasible(x, n):
        return None
    if n <= 96:
        taus = [1e-2, 3e-3, 8e-4, 2e-4, 5e-5, 1e-5]
        mx = 100
    elif n <= 256:
        taus = [5e-3, 5e-4, 5e-5, 1e-5]
        mx = 60
    else:
        taus = [2e-3, 1e-4, 1e-5]
        mx = 35
    s = n / 2.0
    bds = Bounds(np.zeros(n), np.ones(n))
    lc = LinearConstraint(np.ones((1, n)), s, s)
    for tau in taus:
        try:
            res = minimize(smooth_value, x.copy(), args=(n, tau), jac=smooth_grad,
                           method='SLSQP', bounds=bds, constraints=lc,
                           options={'maxiter': mx, 'ftol': 1e-10, 'disp': False})
            y = res.x
        except Exception:
            y = None
        if y is None or y.shape != (n,) or not np.all(np.isfinite(y)):
            continue
        y = project_capped(y, n)
        if y is None or not is_feasible(y, n):
            continue
        best_z, best_phi = x, smooth_value(x, n, tau)
        for k in range(5):
            alpha = 1.0 / (2 ** k)
            z = project_capped(x + alpha * (y - x), n)
            if z is None or not is_feasible(z, n):
                continue
            pz = smooth_value(z, n, tau)
            if pz < best_phi - 1e-14:
                best_phi, best_z = pz, z
        x = best_z
    return x

def coord_search(x, n, rng, n_taken, sigma0):
    x = x.copy()
    score = exact_score(x, n)
    sigma = sigma0
    for _ in range(n_taken):
        sigma *= 0.97
        if sigma < 1e-8:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        delta = float(rng.normal(0, sigma))
        ni, nj = x[i] + delta, x[j] - delta
        if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
            continue
        x[i], x[j] = ni, nj
        sn = exact_score(x, n)
        if sn < score - 1e-13:
            score = sn
        else:
            x[i], x[j] = ni - delta, nj + delta
    return project_capped(x, n), score

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size >= 1 and p.size <= 999:
            n = int(p.size)
            if (np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8)
                    and abs(np.sum(p) - n / 2.0) <= 1e-6 * max(1, n)):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        if 'n_points' in kwargs:
            try:
                v = int(kwargs['n_points'])
                if 1 <= v <= 999:
                    n = v
            except Exception:
                pass
        if n is None:
            n = 47
    fb = None
    if parent is not None:
        fb = project_capped(parent, n)
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    fb_score = exact_score(fb, n)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        h1 = np.array([0.5])
        return h1, exact_score(h1, 1), 1
    if n <= 96:
        n_starts = 7
    elif n <= 256:
        n_starts = 4
    else:
        n_starts = 2
    starts = build_starts(fb, n, rng, n_starts)
    gb, gs = fb.copy(), fb_score
    for st in starts:
        ss = exact_score(st, n)
        if ss < gs - 1e-12:
            gs, gb = ss, st.copy()
    for st in starts:
        try:
            xo = optimize_start(st, n)
            if xo is not None and is_feasible(xo, n):
                so = exact_score(xo, n)
                if so < gs - 1e-12:
                    gs, gb = so, xo.copy()
        except Exception:
            pass
    try:
        xc, sc = coord_search(gb, n, rng, 500, 0.04)
        if xc is not None and sc < gs - 1e-12:
            gs, gb = sc, xc.copy()
    except Exception:
        pass
    try:
        xc, sc = coord_search(gb, n, rng, 500, 0.015)
        if xc is not None and sc < gs - 1e-12:
            gs, gb = sc, xc.copy()
    except Exception:
        pass
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb.copy()
    fs = exact_score(final, n)
    if fs > fb_score + 1e-12:
        final = fb.copy()
        fs = fb_score
    return final.astype(np.float64), float(fs), int(n)
