            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if parent is not None:
        proj = project_capped(parent, n)
        fallback = proj if proj is not None else np.full(n, 0.5)
    else:
        fallback = np.full(n, 0.5)
    if not is_feasible(fallback, n):
        fallback = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        return fallback, float(exact_score(fallback, n)), n

    P = min(64, max(20, round(64.0 * math.sqrt(47.0 / n))))
    B = max(2 * P, min(6000, int(6000.0 * (47.0 / n) ** 2)))
    pop = build_pop(fallback, n, P, rng)
    P = len(pop)
    scores = np.array([exact_score(pop[i], n) for i in range(P)])
    best_idx = int(np.argmin(scores))
    best_h, best_s = pop[best_idx].copy(), scores[best_idx]
    fb_s = exact_score(fallback, n)
    if best_s > fb_s:
        best_h, best_s = fallback.copy(), fb_s
    f_arr, cr_arr = np.full(P, 0.6), np.full(P, 0.8)
    attempts, stagnant, restarts = P, 0, 0
    sig_cycle = (0.03, 0.10, 0.30)

    while attempts < B:
        g_improved = False
        for i in range(P):
            if attempts >= B:
                break
            attempts += 1
            f_new = f_arr[i] if rng.random() >= 0.1 else 0.1 + 0.9 * rng.random()
            cr_new = cr_arr[i] if rng.random() >= 0.1 else rng.random()
            if rng.random() < 0.8:
                cands = [j for j in range(P) if j != i]
                if len(cands) < 3:
                    continue
                r1, r2, r3 = rng.choice(cands, 3, replace=False)
                v = pop[r1] + f_new * (pop[r2] - pop[r3])
            else:
                cands = [j for j in range(P) if j != i and j != best_idx]
                if len(cands) < 2:
                    continue
                r1, r2 = rng.choice(cands, 2, replace=False)
                v = pop[i] + f_new * (best_h - pop[i]) + f_new * (pop[r1] - pop[r2])
            mask = rng.random(n) < cr_new
            if not np.any(mask):
                mask[int(rng.integers(n))] = True
            trial_raw = np.where(mask, v, pop[i]).astype(np.float64)
            trial = project_capped(trial_raw, n)
            if trial is None:
                continue
            ts = exact_score(trial, n)
            if ts < best_s - 1e-12:
                best_h, best_s = trial.copy(), ts
                g_improved = True
            if ts < scores[i] - 1e-12:
                pop[i] = trial.copy()
                scores[i] = ts
                f_arr[i] = f_new
                cr_arr[i] = cr_new
        if g_improved:
            stagnant = 0
        else:
            stagnant += 1
            if stagnant >= 12 and restarts < 2 and attempts < B:
                restarts += 1
                stagnant = 0
                order = np.argsort(scores)
                n_keep = max(2, P // 2)
                keep_set = set(int(order[j]) for j in range(n_keep))
                for slot in range(P - 1, -1, -1):
                    if attempts >= B:
                        break
                    pidx = int(order[slot])
                    if pidx in keep_set:
                        continue
                    attempts += 1
                    if slot % 4 == 0:
                        rc = project_capped(rng.random(n), n)
                    else:
                        sig = sig_cycle[slot % 3]
                        rc = project_capped(best_h + sig * rng.standard_normal(n), n)
                    if rc is not None:
                        pop[pidx] = rc.copy()
                        scores[pidx] = exact_score(rc, n)
                        f_arr[pidx] = 0.6
                        cr_arr[pidx] = 0.8
        best_idx = int(np.argmin(scores))
        if scores[best_idx] < best_s - 1e-12:
            best_h, best_s = pop[best_idx].copy(), scores[best_idx]

    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    if fs > fb_s + 1e-12:
        final, fs = fallback, fb_s
    return final, float(fs), int(n)
