    if abs(np.sum(h) - s) > 5e-12 * max(1, n):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lse_value_grad(h, tau, n):
    dx = 2.0 / n
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    L = len(q)
    qmax = np.max(q)
    shifted = (q - qmax) / tau
    exps = np.exp(shifted)
    Z = np.sum(exps)
    w = exps / Z
    F = qmax + tau * np.log(Z)
    grad = np.zeros(n)
    for k in range(L):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        p = max(s, 0)
        r = max(-s, 0)
        m = n - abs(s)
        grad[p:p + m] += dx * wk * (1.0 - h[r:r + m])
        grad[r:r + m] -= dx * wk * h[p:p + m]
    return F, grad

def lse_descent(h0, n, tau_sched, max_it, eta_init, eta_max):
    h = h0.copy().astype(np.float64)
    best_h = h0.copy()
    best_sc = exact_score(h0, n)
    for tau in tau_sched:
        eta = eta_init
        Fcur, g = lse_value_grad(h, tau, n)
        if not np.isfinite(Fcur) or not np.all(np.isfinite(g)):
            continue
        for _ in range(max_it):
            Fcur, g = lse_value_grad(h, tau, n)
            if not np.isfinite(Fcur) or not np.all(np.isfinite(g)):
                break
            z = project_capped(h - eta * g, n)
            if z is None:
                break
            d = z - h
            if np.max(np.abs(d)) < 1e-11:
                break
            gd = np.dot(g, d)
            if not np.isfinite(gd) or gd >= 0:
                break
            accepted = False
            for alpha in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625):
                tri = project_capped(h + alpha * d, n)
                if tri is None:
                    continue
                Ft, _ = lse_value_grad(tri, tau, n)
                if np.isfinite(Ft) and Ft <= Fcur + alpha * 1e-4 * gd + 1e-14:
                    h = tri
                    Fcur = Ft
                    accepted = True
                    break
            if not accepted:
                break
            eta = min(1.2 * eta, eta_max)
            if is_feasible(h, n):
                sc = exact_score(h, n)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_h = h.copy()
    hp = project_capped(h, n)
    if hp is not None and is_feasible(hp, n):
        sc = exact_score(hp, n)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_h = hp.copy()
    return best_h, best_sc

def coord_polish(h, n, n_pass=8):
    best = h.copy()
    bs = exact_score(h, n)
    for _ in range(n_pass):
        improved = False
        for i in range(n):
            for delta in (-0.04, -0.02, -0.01, 0.01, 0.02, 0.04):
                tri = best.copy()
                tri[i] += delta
                if tri[i] < 0.0:
                    tri[i] = 0.0
                if tri[i] > 1.0:
                    tri[i] = 1.0
                p = project_capped(tri, n)
                if p is None:
                    continue
                sc = exact_score(p, n)
                if sc < bs - 1e-12:
                    best = p
                    bs = sc
                    improved = True
                    break
        if not improved:
            break
    return best, bs

def build_starts(fb, n, rng, count):
    pool = [fb.copy()]
    cands = [fb[::-1].copy(), np.full(n, 0.5)]
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    cands.append(blk)
    j = np.arange(n, dtype=np.float64)
    cands.append(np.clip(0.5 + 0.45 * np.cos(2 * np.pi * (j + 0.5) / n), 0, 1))
    cands.append(np.clip(0.5 + 0.45 * np.cos(2 * np.pi * (j + 0.5) / n + np.pi / 2), 0, 1))
    cands.append(np.clip(0.5 + 0.35 * np.sin(4 * np.pi * (j + 0.5) / n), 0, 1))
    cands.append(rng.uniform(0, 1, n))
    cands.append(fb + rng.normal(0, 0.06, n))
    for c in cands:
        if len(pool) >= count:
            break
        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-8:
                dup = True
                break
        if not dup:
            pool.append(p)
    return pool[:count]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size >= 1:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if parent is not None and parent.size == n:
        fb = project_capped(parent, n)
    elif parent is not None and parent.size != n and parent.size >= 2:
        oj = (np.arange(parent.size) + 0.5) / parent.size
        nj = (np.arange(n) + 0.5) / n
        fb = project_capped(np.interp(nj, oj, parent, left=parent[0], right=parent[-1]), n)
    else:
        fb = project_capped(np.full(n, 0.5), n)
    if fb is None:
        fb = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    fs = exact_score(fb, n)
    gb, gs = fb.copy(), fs
    if n == 1:
        return fb, fs, 1
    tau_sched = [0.02, 0.005, 0.001, 0.0002, 4e-05, 1e-05]
    if n <= 64:
        n_starts, max_it, n_rest, n_pol = 5, 140, 2, 8
    elif n <= 128:
        n_starts, max_it, n_rest, n_pol = 4, 90, 1, 6
    elif n <= 256:
        n_starts, max_it, n_rest, n_pol = 3, 55, 1, 4
    else:
        n_starts, max_it, n_rest, n_pol = 2, 25, 0, 3
    dx = 2.0 / n
    eta_init = 0.25 / dx
    eta_max = 2.0 / dx
    starts = build_starts(fb, n, rng, n_starts)
    for st in starts:
        try:
            bh, bs = lse_descent(st, n, tau_sched, max_it, eta_init, eta_max)
            bh, bs = coord_polish(bh, n, n_pol)
            if bs < gs - 1e-12:
                gs, gb = bs, bh
        except Exception:
            continue
    if gs < fs - 1e-5:
        sds = [0.03, 0.08] if n <= 64 else [0.05]
    else:
        sds = [0.08, 0.18] if n <= 64 else [0.12]
    sds = sds[:n_rest]
    for sd in sds:
        try:
            pert = project_capped(gb + rng.normal(0, sd, n), n)
            if pert is None:
                continue
            bh, bs = lse_descent(pert, n, tau_sched, max_it, eta_init, eta_max)
            bh, bs = coord_polish(bh, n, n_pol)
            if bs < gs - 1e-12:
                gs, gb = bs, bh
        except Exception:
            continue
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb
    sf = exact_score(final, n)
    ss = exact_score(fb, n)
    if sf > ss:
        final, sf = fb, ss
    if not is_feasible(final, n):
        final = np.full(n, 0.5).astype(np.float64)
        sf = exact_score(final, n)
    return final, float(sf), int(n)
