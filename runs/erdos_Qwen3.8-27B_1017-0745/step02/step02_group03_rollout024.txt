        else:
            r = max(1e-5, 0.5 * r)
        if float(np.max(np.abs(delta))) < 1e-8:
            break
    return h, cs

def _add_start(starts, v, n):
    h2 = project_capped(v, n)
    if h2 is None:
        return
    for ex in starts:
        if float(np.max(np.abs(h2 - ex))) < 1e-8:
            return
    starts.append(h2)

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= len(iv) <= 999 and np.all(np.isfinite(iv)):
            n = len(iv)
            parent = iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
        n = max(1, min(999, n))
    try:
        sd = int(seed)
        if sd < 0:
            sd = 42
    except Exception:
        sd = 42
    rng = np.random.default_rng(sd)
    uniform = np.full(n, 0.5)
    h0 = project_capped(parent, n) if parent is not None else None
    fallback = h0 if h0 is not None else uniform.copy()
    fb_score = exact_score(fallback, n)
    if not np.isfinite(fb_score):
        fallback = uniform.copy()
        fb_score = exact_score(fallback, n)
    n_starts_def = 7 if n <= 64 else (5 if n <= 128 else 2)
    mi_def = 30 if n <= 64 else (20 if n <= 128 else 8)
    try:
        n_starts = max(1, min(int(kwargs.get('starts', n_starts_def)), 8))
    except Exception:
        n_starts = n_starts_def
    try:
        max_iter = max(0, min(int(kwargs.get('max_iter', mi_def)), 60))
    except Exception:
        max_iter = mi_def
    starts = []
    _add_start(starts, fallback, n)
    if parent is not None:
        _add_start(starts, fallback[::-1].copy(), n)
        _add_start(starts, 1.0 - fallback, n)
        _add_start(starts, np.roll(fallback, n // 3), n)
        _add_start(starts, np.roll(fallback, -(n // 3)), n)
        _add_start(starts, fallback + rng.normal(0, 0.04, n), n)
        _add_start(starts, fallback + rng.normal(0, 0.10, n), n)
    else:
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        _add_start(starts, blk, n)
        _add_start(starts, np.where(np.arange(n) % 2 == 0, 1.0, 0.0), n)
        tt = np.arange(n)
        _add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * tt / n), n)
        _add_start(starts, 0.5 + 0.45 * np.cos(4.0 * np.pi * tt / n), n)
        _add_start(starts, rng.uniform(0, 1, n), n)
    starts = starts[:n_starts]
    gb = fallback.copy()
    gs = np.array([fb_score])
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < gs[0] - 1e-12:
            gb[:] = st
            gs[0] = ss
    if n <= 256 and max_iter > 0:
        for st in starts:
            try:
                trust_search(st, n, max_iter, gb, gs)
            except Exception:
                pass
        n_hops = 3 if n <= 128 else 1
        hop_sds = [0.02, 0.05, 0.08][:n_hops] if n <= 128 else [0.05]
        for hs in hop_sds:
            pert = gb + rng.normal(0, hs, n)
            pp = project_capped(pert, n)
            if pp is None:
                continue
            try:
                trust_search(pp, n, max_iter, gb, gs)
            except Exception:
                pass
        # Final polish with small radius
        try:
            trust_search(gb, n, max(5, max_iter // 3), gb, gs)
        except Exception:
            pass
    final = project_capped(gb, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final = fallback.copy()
        fs = fb_score
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final, float(fs), int(n)
