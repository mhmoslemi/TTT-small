def project_capped(z, n):
    s = n / 2.0
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if mid == lo or mid == hi:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for i in range(n):
        if abs(r) < 1e-13:
            break
        if r > 0:
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
        else:
            a = min(-r, h[i])
            h[i] -= a
            r += a
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-8:
        return None

        return float('inf')

    dx = 2.0 / n

        hi = min(n, n - d)
        q[d + n - 1] = dx * float(np.dot(h[lo:hi], 1.0 - h[lo + d:hi + d]))
    return q

    q = lag_scores(h, n)
    m = float(np.max(q))
    sh = (q - m) / tau
    ex = np.exp(sh)
    se = float(np.sum(ex))
    F = m + tau * float(np.log(se))
    w = ex / se
    grad = np.zeros(n)
    dx = 2.0 / n
    for d in range(-(n - 1), n):
        wd = w[d + n - 1]
        if wd < 1e-14:
            continue
        lo = max(0, -d)
        hi = min(n, n - d)
        grad[lo:hi] += dx * wd * (1.0 - h[lo + d:hi + d])
        grad[lo + d:hi + d] -= dx * wd * h[lo:hi]
    return F, grad
