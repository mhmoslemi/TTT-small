def project_capped(z, n):

        return None

    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)

            break
        if r > 0:
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
        else:
            a = min(-r, h[i])
            h[i] -= a
            r += a
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):

        return None
    return h

    return float(np.max(q))
