def exact_score(h, n):

def lag_values(h, n):
    L = 2 * n - 1
    v = np.zeros(L)
    for d in range(-(n - 1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        v[d + n - 1] = float(np.dot(h[lo:hi], 1.0 - h[lo + d:hi + d]))
    return v

def surrogate_and_grad(h, n, theta):
    F = lag_values(h, n)
    m = float(np.max(F))
    exps = np.exp(np.clip((F - m) / theta, -700.0, 0.0))
    tot = float(np.sum(exps))
    if tot < 1e-300:
        return m, np.zeros(n)
    p = exps / tot
    phi = m + theta * math.log(tot)
    grad = np.zeros(n)
    for k in range(len(F)):
        d = k - (n - 1)
        w = p[k]
        if w < 1e-18:
            continue
        lo = max(0, -d)
        hi = min(n, n - d)
        indices = np.arange(lo, hi)
        grad[indices] += w * (1.0 - h[lo + d:hi + d])
        grad[lo + d:hi + d] -= w * h[lo:hi]
    return phi, grad
