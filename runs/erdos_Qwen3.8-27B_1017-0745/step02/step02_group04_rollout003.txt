                        gs, gb = sy, y.copy()
            except Exception:
                pass
    sigmas = [0.03, 0.08, 0.15, 0.25]
    T0 = 2.0e-3
    for r in range(n_hops):
        if r % 5 == 4:
            prop = project_capped(rng.uniform(0, 1, n), n)
        else:
            sigma = sigmas[r % 4]
            prop = project_capped(gb + sigma * rng.standard_normal(n), n)
        if prop is None:
            continue
        try:
            y = solve_exact(prop, n, maxiter)
            if y is not None and is_feasible(y, n):
                sy = exact_score(y, n)
                if sy < gs - 1e-12:
                    gs, gb = sy, y.copy()
            sp = exact_score(prop, n)
            if sp < gs:
                gb2 = prop.copy()
            else:
                T = max(1e-5, T0 * (0.75 ** r))
                if sp - gs < 2.0 * T and rng.random() < np.exp(-(sp - gs) / T):
                    gb2 = prop.copy()
                else:
                    continue
        except Exception:
            continue
    gb, gs = binary_polish(gb, n, rng, 60)
    gb2 = project_capped(gb, n)
    if gb2 is not None and is_feasible(gb2, n):
        s2 = exact_score(gb2, n)
        if s2 < gs:
            gb, gs = gb2, s2
        else:
            gb = gb2
            gs = exact_score(gb, n)
    fb = float(exact_score(fallback, n))
    if gs > fb + 1e-12:
        gb, gs = fallback.copy(), fb
    if gs == float('inf'):
        gb = np.full(n, 0.5)
        gs = float(exact_score(gb, n))
    return gb, gs, int(n)
