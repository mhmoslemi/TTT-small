def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(2):
        if abs(r) < 2e-14 * max(1, n):
            break
        if r > 0:
            headroom = 1.0 - h
            total = headroom.sum()
            if total <= 0: break
            h += r * headroom / total
        else:
            total = h.sum()
            if total <= 0: break
            h += r * h / total
        r = s - np.sum(h)
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-13) or np.any(h > 1.0 + 1e-13):
        return None
    if abs(np.sum(h) - s) > 1e-10 * max(1, n):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def lag_values(h, n):
    if not is_feasible(h, n):
        return None
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    if q.shape != (L,) or not np.all(np.isfinite(q)):
        return None
    return q

def exact_score(h, n):
    L = lag_values(h, n)
    if L is None:
        return float('inf')
    return float(np.max(L))

def build_envelope(A_aff, B_aff, lo, hi):
    # A_aff[k] = intercept, B_aff[k] = slope for affine lines
    # Returns list of (t_start, t_end, A, B) segments
    n_lines = len(B_aff)
    if n_lines == 0:
        return [(lo, hi, 0.0, 0.0)]  # dummy
    # Sort by slope, break ties by largest intercept
    order = np.argsort(B_aff)
    # Remove duplicates: for same slope, keep only max intercept
    filtered = []
    for idx in order:
        if filtered and B_aff[filtered[-1]] == B_aff[idx]:
            if A_aff[idx] > A_aff[filtered[-1]]:
                filtered[-1] = idx
        else:
            filtered.append(idx)
    # Build upper hull
    hull = []  # list of (idx, t_start)
    for idx in filtered:
        if not hull:
            hull.append((idx, -1e18))
        else:
            # Intersection of last hull line with new line
            # A_last + B_last * t = A_new + B_new * t
            # t = (A_last - A_new) / (B_new - B_last)
            last_idx, last_start = hull[-1]
            if B_aff[idx] == B_aff[last_idx]:
                continue  # same slope, should have been filtered
            t_cross = (A_aff[last_idx] - A_aff[idx]) / (B_aff[idx] - B_aff[last_idx])
            while hull and t_cross <= hull[-1][1]:
                hull.pop()
                if not hull:
                    t_cross = -1e18
                    break
                last_idx, last_start = hull[-1]
                t_cross = (A_aff[last_idx] - A_aff[idx]) / (B_aff[idx] - B_aff[last_idx])
            hull.append((idx, t_cross))
    # Convert to segments clipped to [lo, hi]
    segments = []
    for k, (idx, t_start) in enumerate(hull):
        t_end = hull[k+1][1] if k+1 < len(hull) else 1e18
        seg_lo = max(t_start, lo)
        seg_hi = min(t_end, hi)
        if seg_lo < seg_hi:
            segments.append((seg_lo, seg_hi, A_aff[idx], B_aff[idx]))
    if not segments:
        segments = [(lo, hi, 0.0, 0.0)]
    return segments

def lag_values(h, n):
    if not is_feasible(h, n): return None
    q = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
    if q.shape != (2*n-1,) or not np.all(np.isfinite(q)): return None
    return q

def exact_score(h, n):
    L = lag_values(h, n)
    if L is None: return float('inf')
    return float(np.max(L))

def solve_quad(a, b, c):
    roots = []
    if abs(a) < 1e-15:
        if abs(b) > 1e-15: roots.append(-c/b)
    else:
        disc = b*b - 4.0*a*c
        if disc < 0.0:
            if disc > -1e-12*(b*b + abs(4.0*a*c) + 1e-300): disc = 0.0
            else: return roots
        sd = sqrt(disc)
        if abs(b) > 1e-15:
            q = -0.5*(b + copysign(sd, b))
            roots.append(q/a)
            if abs(q) > 1e-15: roots.append(c/q)
            else: roots.append(-b/a)
        else:
            roots.extend([sd/(2.0*abs(a)), -sd/(2.0*abs(a))])
    return roots

def pair_line_min(h, A, n, i, j):
    dx = 2.0/n
    r0 = abs(i - j)
    lo = max(-h[i], h[j]-1.0)
    hi = min(1.0-h[i], h[j])
    if hi - lo < 1e-15: return 0.0, A, A
    # Compute B for all lags
    B = np.zeros(2*n-1)
    for d in range(-(n-1), n):
        a = max(d, 0); b = max(-d, 0); m = n - abs(d)
        gi = 0.0
        if a <= i < a+m: gi += dx*(1.0-h[b+i-a])
        if b <= i < b+m: gi -= dx*h[a+i-b]
        gj = 0.0
        if a <= j < a+m: gj += dx*(1.0-h[b+j-a])
        if b <= j < b+m: gj -= dx*h[a+j-b]
        B[d+n-1] = gi - gj
    # D array
    D = np.zeros(2*n-1)
    D[n-1] = -2.0*dx
    D[n-1+r0] = dx
    D[n-1-r0] = dx
    # Identify special and affine indices
    special = set([n-1, n-1+r0, n-1-r0])
    aff_idx = [k for k in range(2*n-1) if k not in special]
    # Build upper envelope of affine lines on [lo, hi]
    if not aff_idx:
        segments = [(lo, hi, 0.0, 0.0, True)]  # no active line
    else:
        Aa = A[aff_idx]; Ba = B[aff_idx]
        # Sort by slope, keep max intercept for ties
        order = np.argsort(Ba)
        filt = []
        for idx in order:
            if filt and Ba[filt[-1]] == Ba[idx]:
                if Aa[idx] > Aa[filt[-1]]: filt[-1] = idx
            else:
                filt.append(idx)
        hull = []

            if not hull:
                hull.append([idx, -1e18])
