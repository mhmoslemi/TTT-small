def polish_subgrad(h0,n,n_iter=30,eps=1e-4):

       if failed>10 or eta<1e-8/dx: break
       q=np.correlate(current,1.0-current,mode='full')*dx
       mx=float(np.max(q))

def lag_grad(h,n):
    dx=2.0/n
    G=np.zeros((2*n-1,n),dtype=np.float64)
    for r in range(2*n-1):
        d=r-(n-1)
        a=0 if d<0 else d
        b=0 if d>0 else -d
        m=n-abs(d)
        G[r,a:a+m]+=dx*(1.0-h[b:b+m])
        G[r,b:b+m]-=dx*h[a:a+m]
    return G
```

def polish_subgrad(h0,n,n_iter=25,eps=1e-3):
    current=project_capped(h0,n)
    if current is None: return None,None

    if not math.isfinite(score): return current,score

    eta=1.0/dx
    eta_max=6.0/dx
    failed=0

        if failed > 10 or eta < 1e-8/dx:
            break

import numpy as np
from scipy.optimize import minimize
import math

def is_feasible(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(float(np.sum(h)) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def project_capped(z, n):
    if z is None:
        return None
    z = np.asarray(z, dtype=np.float64)
    if z.size != n:
        return None
    z = z.ravel()
    if not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        vv = np.clip(z - mid, 0.0, 1.0)
        if float(np.sum(vv)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(2):
        for i in range(n):
            if r > 0.0:
                c = min(r, 1.0 - h[i])
                h[i] += c
                r -= c
            elif r < 0.0:
                c = min(-r, h[i])
                h[i] -= c
                r += c
            if abs(r) < 1e-14:
                break
        if abs(r) < 1e-14:
            break
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)):
        return None
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(float(np.sum(h)) - s) > 1e-10 * max(1, n):
        return None
    return h

def exact_score(h, n):
    if not is_feasible(h, n):
        return float("inf")
    q = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    if q.size != 2 * n - 1 or not np.all(np.isfinite(q)):
        return float("inf")
    m = float(np.max(q))
    if not math.isfinite(m):
        return float("inf")
    return m

def lag_grad(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n), dtype=np.float64)
    for r in range(2 * n - 1):
        d = r - (n - 1)
        a = 0 if d < 0 else d
        b = 0 if d > 0 else -d
        m = n - abs(d)
        G[r, a:a + m] += dx * (1.0 - h[b:b + m])
        G[r, b:b + m] -= dx * h[a:a + m]
    return G

def epi_obj(z):
    return float(z[-1])

def epi_obj_grad(z):
    g = np.zeros(z.shape[0], dtype=np.float64)
    g[-1] = 1.0
    return g

def epi_ineq(z):
    n = z.shape[0] - 1
    h = z[:n]
    dx = 2.0 / n
    q = np.correlate(h, 1.0 - h, mode="full") * dx
    return z[-1] - q

def epi_ineq_grad(z):
    n = z.shape[0] - 1
    h = z[:n]
    G = lag_grad(h, n)
    J = -G
    J[:, n] = 1.0
    return J

def epi_eq(z):
    n = z.shape[0] - 1
    return float(np.sum(z[:n]) - n / 2.0)

def epi_eq_grad(z):
    n = z.shape[0] - 1
    J = np.zeros(n + 1, dtype=np.float64)
    J[:n] = 1.0
    return J

def slsqp_one_pass(h0, n, maxiter=150, t_margin=1e-8):
    hp = project_capped(h0, n)
    if hp is None:
        return None
    q = np.correlate(hp, 1.0 - hp, mode="full") * (2.0 / n)
    if q.size != 2 * n - 1 or not np.all(np.isfinite(q)):
        return None
    t0 = float(np.max(q)) + t_margin
    upper = max(0.6, t0 + 0.1)
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))
    bounds.append((0.0, upper))
    cons = [
        {"type": "ineq", "fun": epi_ineq, "jac": epi_ineq_grad},
        {"type": "eq", "fun": epi_eq, "jac": epi_eq_grad},
    ]
    z0 = np.concatenate([hp, [t0]])
    try:
        res = minimize(epi_obj, z0, jac=epi_obj_grad, bounds=bounds,
                       constraints=cons, method="SLSQP",
                       options={"maxiter": maxiter, "ftol": 1e-12})
    except Exception:
        return None
    if res is None or not hasattr(res, "x") or res.x is None:
        return None
    x = np.asarray(res.x, dtype=np.float64).ravel()
    if x.size != n + 1 or not np.all(np.isfinite(x)):
        return None
    return project_capped(x[:n], n)

def polish_slsqp(h0, n, n_passes=2, maxiter=150):
    current = project_capped(h0, n)
    if current is None:
        return None, None
    score = exact_score(current, n)
    if not math.isfinite(score):
        return current, score
    for p in range(n_passes):
        margin = 0.02 if p == 0 else 1e-8
        y = slsqp_one_pass(current, n, maxiter, margin)
        if y is None or not is_feasible(y, n):
            break
        best_z = current.copy()
        best_s = score
        for k in range(7):
            alpha = 1.0 / (2.0 ** k)
            if alpha >= 1.0 - 1e-15:
                z = y
            else:
                z = project_capped(current + alpha * (y - current), n)
            if z is None:
                continue
            sz = exact_score(z, n)
            if sz < best_s - 1e-12:
                best_s = float(sz)
                best_z = z.copy()
        if best_s >= score - 1e-12:
            break
        current = best_z
        score = best_s
    return current, score

def polish_subgrad(h0, n, n_iter=25, eps=1e-3):
    current = project_capped(h0, n)
    if current is None:
        return None, None
    score = exact_score(current, n)
    if not math.isfinite(score):
        return current, score
    dx = 2.0 / n
    eta = 1.0 / dx
    eta_max = 6.0 / dx
    failed = 0
    for _ in range(n_iter):
        if failed > 10 or eta < 1e-8 / dx:
            break
        q = np.correlate(current, 1.0 - current, mode="full") * dx
        mx = float(np.max(q))
        tol = eps * max(1.0, mx)
        idx = np.where(q >= mx - tol)[0]
        if idx.size == 0:
            break
        G = lag_grad(current, n)
        g = G[idx].mean(axis=0)
        if not np.all(np.isfinite(g)) or not np.any(g != 0.0):
            break
        v = project_capped(current - eta * g, n)
        if v is None:
            eta *= 0.5
            failed += 1
            continue
        best_z = current.copy()
        best_s = score
        for k in range(7):
            alpha = 1.0 / (2.0 ** k)
            if alpha >= 1.0 - 1e-15:
                z = v
            else:
                z = project_capped(current + alpha * (v - current), n)
            if z is None:
                continue
            sz = exact_score(z, n)
            if sz < best_s - 1e-12:
                best_s = float(sz)
                best_z = z.copy()
        if best_s >= score - 1e-12:
            eta *= 0.5
            failed += 1
        else:
            current = best_z
            score = best_s
            failed = 0
            eta = min(1.1 * eta, eta_max)
    return current, score

def choose_distinct(rng, count, size, exclude):
    if count <= 0:
        return []
    excl = set(exclude)
    avail = []
    for i in range(size):
        if i not in excl:
            avail.append(i)
    if len(avail) <= count:
        return avail
    perm = rng.permutation(len(avail))[:count]
    res = []
    for x in perm:
        res.append(avail[int(x)])
    return res

def add_candidate(pool, z, n, limit, dup):
    p = project_capped(z, n)
    if p is None:
        p = np.full(n, 0.5, dtype=np.float64)
    if not is_feasible(p, n):
        p = np.full(n, 0.5, dtype=np.float64)
    if len(pool) >= limit:
        return
    if dup:
        for v in pool:
            if np.max(np.abs(v - p)) < 1e-8:
                return
    pool.append(p)

def random_symmetric(n, rng):
    v = np.empty(n, dtype=np.float64)
    m = n // 2
    if m > 0:
        a = rng.uniform(0.0, 1.0, m)
        v[:m] = a
    if n % 2 == 1:
        v[m] = 0.5
        if m > 0:
            v[m + 1:] = 1.0 - a[::-1]
    else:
        if m > 0:
            v[m:] = 1.0 - a[::-1]
    if n == 1:
        v[0] = 0.5
    return v

def build_population(fallback, n, rng, P):

    add_candidate(pool, fallback, n, P, True)
    add_candidate(pool, fallback[::-1].copy(), n, P, True)

    blk = np.zeros(n, dtype=np.float64)

    if n % 2 == 1:

    add_candidate(pool, blk, n, P, True)
    alt = np.zeros(n, dtype=np.float64)
    alt[0::2] = 1.0
    add_candidate(pool, alt, n, P, True)
    for _ in range(4):
        add_candidate(pool, random_symmetric(n, rng), n, P, True)
    t = np.linspace(0.0, 1.0, n)
    for sd in (0.02, 0.05, 0.12, 0.25, 0.45):
        add_candidate(pool, fallback + sd * rng.normal(0.0, 1.0, n), n, P, True)
    add_candidate(pool, rng.uniform(0.0, 1.0, n), n, P, True)
    for k in (1, 2, 3):
        c = np.cos(2.0 * math.pi * k * t)
        add_candidate(pool, 0.5 + 0.35 * c, n, P, True)
        add_candidate(pool, np.where(c > 0.0, 1.0, 0.0), n, P, True)
        s = np.sin(2.0 * math.pi * k * t)
        add_candidate(pool, 0.5 + 0.35 * s, n, P, True)
        add_candidate(pool, np.where(s > 0.0, 1.0, 0.0), n, P, True)
    sigmas = (0.03, 0.08, 0.18, 0.35)
    j = 0
    while len(pool) < P:
        if j % 4 == 0:
            z = rng.uniform(0.0, 1.0, n)
        elif j % 4 == 1:
            z = fallback + sigmas[(j // 4) % 4] * rng.normal(0.0, 1.0, n)

    if not pool:

        add_candidate(cands, pop[ii], n, 6, True)
    add_candidate(cands, 0.5 * (best_h + 1.0 - best_h[::-1]), n, 6, True)
    add_candidate(cands, 0.5 * (fb + 1.0 - fb[::-1]), n, 6, True)
    add_candidate(cands, np.where(best_h > 0.5, 1.0, 0.0), n, 6, True)
    final = fb.copy()
    final_score = exact_score(fb, n)
    cand_list = []
    for c in cands:
        try:
            zs, ss = polish_subgrad(c, n, sg_iter, sg_eps)
            if zs is not None and is_feasible(zs, n) and math.isfinite(ss):
                if ss < final_score - 1e-12:
                    final = zs.copy()
                    final_score = float(ss)
                add_candidate(cand_list, zs, n, 4, True)
            else:
                add_candidate(cand_list, c, n, 4, True)
        except Exception:
            add_candidate(cand_list, c, n, 4, True)
    if not cand_list:
        cand_list = [fb.copy()]
    best_idx = 0
    best_cs = exact_score(cand_list[0], n)
    for i in range(1, len(cand_list)):
        cs = exact_score(cand_list[i], n)
        if cs < best_cs - 1e-12:
            best_cs = float(cs)
            best_idx = i
    for i in range(len(cand_list)):
        if i == best_idx:
            npass, miter = 2, sl_best
        else:
            npass, miter = 1, sl_other
        try:
            zo, so = polish_slsqp(cand_list[i], n, npass, miter)
            if zo is not None and is_feasible(zo, n) and math.isfinite(so) and so < final_score - 1e-12:
                final = zo.copy()
                final_score = float(so)
        except Exception:
            pass
    fb_score = exact_score(fb, n)
    if final_score > fb_score + 1e-12:
        final = fb.copy()
        final_score = float(fb_score)
    fp = project_capped(final, n)
    if fp is not None:
        sp = exact_score(fp, n)
        if math.isfinite(sp) and sp <= final_score + 1e-12:
            final = fp
            final_score = float(sp)
    if not is_feasible(final, n):
        final = fb.copy()
        final_score = float(exact_score(fb, n))
    return final, float(final_score), int(n)
