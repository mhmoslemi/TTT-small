def optimize_chain(start,n,line_budget,pairs):
  current=np.array(start,float).copy()
  score=exact_score(current,n)
  if not np.isfinite(score): return current,score
  dirs=build_haar(n)
  if not dirs: return current,score
  used=0; stagnant=0
  max_sweeps=4
  for sweep in range(max_sweeps):
    improved=False
    if used < line_budget:
       v=active_direction(current,n)
       if v is not None:
          new,ns=line_search(current,score,v,n,pairs)
          used += 1
          if ns < score - 1e-12:
             current=new; score=ns; improved=True
    if used < line_budget:
       if sweep % 2 == 0:
          order=range(len(dirs))
       else:
          order=range(len(dirs)-1,-1,-1)
       for di in order:
          if used >= line_budget: break
          v=dirs[di]
          new,ns=line_search(current,score,v,n,pairs)
          used +=1
          if ns < score -1e-12:
             current=new; score=ns; improved=True
    if not improved:
       stagnant +=1
       if stagnant >=2: break
    else: stagnant=0
    if used >= line_budget: break
  return current,score

import numpy as np
import math


def is_feasible(h, n):
    h = np.asarray(h, dtype=np.float64)
    if h.shape != (n,):
        return False
    if not bool(np.all(np.isfinite(h))):
        return False
    if float(np.min(h)) < -1e-12 or float(np.max(h)) > 1.0 + 1e-12:
        return False
    return abs(float(np.sum(h)) - 0.5 * n) <= 1e-10 * max(1, n)


def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n or not bool(np.all(np.isfinite(z))):
        return None
    z = np.clip(z, -1e12, 1e12)
    target = 0.5 * n
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(3):
        r = target - float(np.sum(h))
        if abs(r) < 1e-14:
            break
        for i in range(n):
            if r > 1e-13:
                cap = min(r, 1.0 - h[i])
                if cap > 0.0:
                    h[i] += cap
                    r -= cap
            elif r < -1e-13:
                cap = min(-r, h[i])
                if cap > 0.0:
                    h[i] -= cap
                    r += cap
            else:
                break
    return h if is_feasible(h, n) else None


def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    q = np.correlate(h, 1.0 - h, 'full') * (2.0 / n)
    if q.size != 2 * n - 1 or not bool(np.all(np.isfinite(q))):
        return float('inf')
    return float(np.max(q))


def interpolate_source(src, n):
    src = np.asarray(src, dtype=np.float64).ravel()
    m = src.size
    if m == 1:
        return np.full(n, float(src[0]), dtype=np.float64)
    x0 = np.linspace(0.0, 2.0, m)
    x1 = np.linspace(0.0, 2.0, n)
    return np.interp(x1, x0, src)


def normalize_vector(v, n):
    v = np.asarray(v, dtype=np.float64).ravel().copy()
    if v.size != n or n <= 1 or not bool(np.all(np.isfinite(v))):
        return None
    v = v - float(np.mean(v))
    s = float(np.sum(v))
    if abs(s) > 1e-16:
        v[-1] -= s
    m = float(np.max(np.abs(v)))
    if m <= 1e-16:
        return None
    v = v / m
    s = float(np.sum(v))
    if abs(s) > 1e-12:
        v[-1] -= s
    return v if bool(np.all(np.isfinite(v))) else None


def build_haar(n):
    dirs = []
    current = [(0, n)]
    while current:
        nxt = []
        for a, b in current:
            L = b - a
            if L < 2:
                continue
            L0 = L // 2
            L1 = L - L0
            v = np.zeros(n, dtype=np.float64)
            v[a:a + L0] = float(L1)
            v[a + L0:b] = -float(L0)
            m = float(np.max(np.abs(v)))
            if m > 0.0:
                v = v / m
            s = float(np.sum(v))
            if abs(s) > 1e-12:
                idx = a if v[a] != 0.0 else a + L0
                v[idx] -= s
            dirs.append(v)
            nxt.append((a, a + L0))
            nxt.append((a + L0, b))
        current = nxt
    return dirs


def build_pair_indices(p):
    out = []
    for i in range(p):
        for j in range(i + 1, p):
            out.append((i, j))
    return out


def active_direction(h, n):
    dx = 2.0 / float(n)
    q = np.correlate(h, 1.0 - h, 'full') * dx
    if q.size != 2 * n - 1:
        return None
    idx = int(np.argmax(q))
    d = idx - (n - 1)
    p0 = max(d, 0)
    q0 = max(-d, 0)
    m = n - abs(d)
    if m <= 0:
        return None
    g = np.zeros(n, dtype=np.float64)
    g[p0:p0 + m] += dx * (1.0 - h[q0:q0 + m])
    g[q0:q0 + m] -= dx * h[p0:p0 + m]
    return normalize_vector(-g, n)


def line_interval(h, v, n):
    if not bool(np.all(np.isfinite(h))) or not bool(np.all(np.isfinite(v))):
        return None
    lo = -1e300
    hi = 1e300
    pos = v > 1e-14
    neg = v < -1e-14
    if bool(np.any(pos)):
        lo = max(lo, float(np.max(-h[pos] / v[pos])))
        hi = min(hi, float(np.min((1.0 - h[pos]) / v[pos])))
    if bool(np.any(neg)):
        lo = max(lo, float(np.max((1.0 - h[neg]) / v[neg])))
        hi = min(hi, float(np.min(-h[neg] / v[neg])))
    if not (math.isfinite(lo) and math.isfinite(hi)):
        return None
    if lo > 0.0:
        lo = 0.0
    if hi < 0.0:
        hi = 0.0
    if hi - lo < -1e-12:
        return None
    return (float(lo), float(hi))


def line_coefficients(h, v, n):
    dx = 2.0 / float(n)
    one = 1.0 - h
    cc = dx * np.correlate(h, one, 'full')
    b1 = dx * np.correlate(v, one, 'full')
    b2 = dx * np.correlate(h, v, 'full')
    bb = b1 - b2
    aa = -dx * np.correlate(v, v, 'full')
    p = 2 * n - 1
    if aa.size != p or bb.size != p or cc.size != p:
        return None
    if not (bool(np.all(np.isfinite(aa))) and bool(np.all(np.isfinite(bb))) and bool(np.all(np.isfinite(cc)))):
        return None
    return (aa, bb, cc)


def append_roots(a, b, c, lo, hi, out):
    scale = max(abs(a), abs(b), abs(c), 1e-300)
    tol = 1e-14 * scale
    if abs(a) <= tol:
        if abs(b) > tol:
            r = -c / b
            if math.isfinite(r):
                eps = 1e-12 * max(1.0, abs(lo), abs(hi))
                if lo - eps <= r <= hi + eps:
                    out.append(float(min(max(r, lo), hi)))
        return
    disc = b * b - 4.0 * a * c
    dtol = 1e-12 * (b * b + 4.0 * abs(a * c) + 1e-300)
    if disc < -dtol:
        return
    if disc < 0.0:
        disc = 0.0
    s = math.sqrt(disc)
    q = -0.5 * (b + s) if b >= 0.0 else -0.5 * (b - s)
    if abs(q) > 1e-300:
        r1 = q / a
        r2 = c / q
    else:
        r1 = -b / (2.0 * a)
        r2 = r1
    eps = 1e-12 * max(1.0, abs(lo), abs(hi))
    for r in (r1, r2):
        if math.isfinite(r) and lo - eps <= r <= hi + eps:
            out.append(float(min(max(r, lo), hi)))


def add_unique_t(lst, t, eps):
    t = float(t)
    for x in lst:
        if abs(x - t) <= eps * max(1.0, abs(t)):
            return
    lst.append(t)


def line_candidates(aa, bb, cc, lo, hi, pairs):
    out = [float(lo), float(hi), 0.0]
    maxa = float(np.max(np.abs(aa))) if aa.size else 0.0
    if maxa < 1e-300:
        maxa = 1.0
    atol = 1e-14 * maxa
    for k in range(aa.size):
        if abs(aa[k]) > atol:
            t = -bb[k] / (2.0 * aa[k])
            if math.isfinite(t):
                eps = 1e-12 * max(1.0, abs(lo), abs(hi))
                if lo - eps <= t <= hi + eps:
                    out.append(float(min(max(t, lo), hi)))
    for i, j in pairs:
        append_roots(aa[i] - aa[j], bb[i] - bb[j], cc[i] - cc[j], lo, hi, out)
    vals = np.asarray(out, dtype=np.float64)
    vals = vals[bool(np.isfinite(vals))]
    if vals.size == 0:
        return np.asarray([0.0], dtype=np.float64)
    vals.sort()
    vals = np.clip(vals, lo, hi)
    kept = []
    prev = None
    for x in vals:
        xf = float(x)
        if prev is None or abs(xf - prev) > 1e-9 * max(1.0, abs(xf)):
            kept.append(xf)
            prev = xf
    for t in (float(lo), float(hi), 0.0):
        add_unique_t(kept, t, 1e-9)
    return np.asarray(kept, dtype=np.float64)


def fast_point(h, v, t, n):
    z = h + t * v
    if not bool(np.all(np.isfinite(z))):
        return None
    if float(np.min(z)) < -1e-5 or float(np.max(z)) > 1.0 + 1e-5:
        return None
    z = np.clip(z, 0.0, 1.0)
    target = 0.5 * n
    r = target - float(np.sum(z))
    if abs(r) <= 1e-8 * max(1, n):
        adj = r / float(n)
        if float(np.min(z + adj)) >= -1e-12 and float(np.max(z + adj)) <= 1.0 + 1e-12:
            z = z + adj
        else:
            for i in range(n):
                if r > 1e-13:
                    cap = min(r, 1.0 - z[i])
                    if cap > 0.0:
                        z[i] += cap
                        r -= cap
                elif r < -1e-13:
                    cap = min(-r, z[i])
                    if cap > 0.0:
                        z[i] -= cap
                        r += cap
                else:
                    break
        z = np.clip(z, 0.0, 1.0)
    else:
        z = project_capped(z, n)
    if z is None or not is_feasible(z, n):
        z = project_capped(h + t * v, n)
    return z


def line_search(h, score, v, n, pairs, max_exact=12):
    h = np.asarray(h, dtype=np.float64)
    v = np.asarray(v, dtype=np.float64)
    if h.shape != (n,) or v.shape != (n,) or not bool(np.all(np.isfinite(v))):
        return h.copy(), score
    if float(np.max(np.abs(v))) <= 1e-16:
        return h.copy(), score
    inter = line_interval(h, v, n)
    if inter is None:
        return h.copy(), score
    lo, hi = inter
    if hi - lo <= 1e-14:
        return h.copy(), score
    coeff = line_coefficients(h, v, n)
    if coeff is None:
        return h.copy(), score
    aa, bb, cc = coeff
    cand = line_candidates(aa, bb, cc, lo, hi, pairs)
    if cand is None or cand.size == 0:
        return h.copy(), score
    pred = []
    p = cc.size
    chunk = 128
    for st in range(0, cand.size, chunk):
        t = cand[st:st + chunk]
        tt = t[:, None]
        vals = cc[None, :] + bb[None, :] * tt + aa[None, :] * tt * tt
        rm = np.max(vals, axis=1)
        for i in range(t.size):
            val = float(rm[i])
            if math.isfinite(val):
                pred.append((val, float(t[i])))
    if not pred:
        return h.copy(), score
    pred.sort()
    selected = []
    for _, t in pred[:max_exact]:
        add_unique_t(selected, t, 1e-8)
    for t in (lo, hi, 0.0):
        add_unique_t(selected, t, 1e-8)
    best_z = None
    best_s = score
    for t in selected:
        z = fast_point(h, v, t, n)
        if z is None:
            continue
        s = exact_score(z, n)
        if s < best_s - 1e-12:
            best_s = s
            best_z = z.copy()
    if best_z is None:
        return h.copy(), score
    return best_z, best_s


def add_start(pool, z, n, limit):
    if len(pool) >= limit:
        return
    p = project_capped(z, n)
    if p is None:
        return
    for q in pool:
        if float(np.max(np.abs(p - q))) < 1e-8:
            return
    pool.append(p.copy())


def build_starts(fallback, n, rng, limit=16):
    pool = []
    add_start(pool, fallback, n, limit)
    add_start(pool, fallback[::-1].copy(), n, limit)
    comp = 1.0 - fallback
    add_start(pool, comp, n, limit)
    add_start(pool, comp[::-1].copy(), n, limit)
    if n > 1:
        blk = np.zeros(n, dtype=np.float64)
        blk[:n // 2] = 1.0
        if n % 2:
            blk[n // 2] = 0.5
        add_start(pool, blk, n, limit)
        alt = np.zeros(n, dtype=np.float64)
        alt[0::2] = 1.0
        add_start(pool, alt, n, limit)
    x = np.linspace(0.0, 2.0, n)
    for amp in (0.7, 1.0, 1.5):
        for freq in (math.pi, 2.0 * math.pi):
            for ph in (0.0, 0.5 * math.pi):
                v = 0.5 + 0.5 * amp * np.sin(freq * x + ph)
                add_start(pool, v, n, limit)
                v = 0.5 + 0.5 * amp * np.cos(freq * x + ph)
                add_start(pool, v, n, limit)
    for sd in (0.05, 0.12, 0.25):
        add_start(pool, fallback + rng.normal(0.0, sd, n), n, limit)
    for _ in range(8):
        v = np.full(n, 0.5, dtype=np.float64)
        for k in range(1, 5):
            a = rng.normal(0.0, 0.15)
            ph = rng.uniform(0.0, 2.0 * math.pi)
            v = v + a * np.sin(k * math.pi * x + ph)
        add_start(pool, v, n, limit)
    add_start(pool, rng.uniform(0.0, 1.0, n), n, limit)
    return pool


def optimize_chain(start, n, line_budget, dirs, pairs, rng):
    current = np.asarray(start, dtype=np.float64).copy()
    score = exact_score(current, n)
    if not math.isfinite(score) or not dirs:
        return current, score
    used = 0
    stagnant = 0
    for sweep in range(4):
        improved = False
        if used < line_budget:
            v = active_direction(current, n)
            if v is not None:
                new, ns = line_search(current, score, v, n, pairs)
                used += 1
                if ns < score - 1e-12:
                    current, score = new, ns
                    improved = True
        if used < line_budget:
            if sweep % 2 == 0:
                order = range(len(dirs))
            else:
                order = range(len(dirs) - 1, -1, -1)
            for di in order:
                if used >= line_budget:
                    break
                new, ns = line_search(current, score, dirs[di], n, pairs)
                used += 1
                if ns < score - 1e-12:
                    current, score = new, ns
                    improved = True
        if not improved:
            stagnant += 1
            if stagnant >= 2:
                break
        else:
            stagnant = 0
        if used >= line_budget:
            break
    if rng is not None and n <= 47:
        for _ in range(min(line_budget - used, 8)):
            if used >= line_budget or rng is None:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            v = np.zeros(n, dtype=np.float64)
            v[i] = 1.0
            v[j] = -1.0
            new, ns = line_search(current, score, v, n, pairs)
            used += 1
            if ns < score - 1e-12:
                current, score = new, ns
    return current, score


def run(seed=42, **kwargs):
    source = None
    n = None
    try:
        p = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0 and bool(np.all(np.isfinite(p))):
            source = p.copy()
            n = int(p.size)
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
    if n < 1:
        n = 47
    if n > 64:
        n = 47

    if source is not None:
        if source.size == n:
            fallback = project_capped(np.clip(source, 0.0, 1.0), n)
        else:
            fallback = project_capped(interpolate_source(source, n), n)
    else:
        fallback = project_capped(np.full(n, 0.5, dtype=np.float64), n)
    if fallback is None or not is_feasible(fallback, n):
        fallback = project_capped(np.full(n, 0.5, dtype=np.float64), n)
    if fallback is None or not is_feasible(fallback, n):
        return np.full(n, 0.5, dtype=np.float64), 0.5, int(n)

    if n <= 1:
        return fallback, float(exact_score(fallback, n)), int(n)

    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)

    pairs = build_pair_indices(2 * n - 1)
    dirs = build_haar(n)
    best_h = fallback.copy()
    best_score = exact_score(best_h, n)
    safe_h = best_h.copy()
    safe_n = n
    safe_score = best_score

    starts = build_starts(fallback, n, rng, 16)
    scored = []
    for i, z in enumerate(starts):
        s = exact_score(z, n)
        if math.isfinite(s):
            scored.append((s, i))
    scored.sort()
    selected = []
    for _, i in scored:
        if i not in selected:
            selected.append(i)
            if len(selected) >= 3:
                break
    if 0 not in selected:
        selected.append(0)
    if not selected:
        selected = [0]
    selected = selected[:3]

    for i in selected:
        s = exact_score(starts[i], n)
        if s < best_score - 1e-12:
            best_score = s
            best_h = starts[i].copy()

    per_budget = 60 if n <= 47 else 40
    for i in selected:
        if i < len(starts):
            cur, sc = optimize_chain(starts[i], n, per_budget, dirs, pairs, rng)
            if sc < best_score - 1e-12:
                best_score = sc
                best_h = cur.copy()

    if n < 64 and best_score > 0.3800:
        n2 = 64
        fb2 = project_capped(interpolate_source(best_h, n2), n2)
        if fb2 is not None and is_feasible(fb2, n2):
            rng2 = np.random.default_rng(int(seed) + 101 if isinstance(seed, int) else int(seed == 42))
            starts2 = build_starts(fb2, n2, rng2, 10)
            scored2 = []
            for i, z in enumerate(starts2):
                s = exact_score(z, n2)
                if math.isfinite(s):
                    scored2.append((s, i))
            scored2.sort()
            sel2 = []
            for _, i in scored2:
                if i not in sel2:
                    sel2.append(i)
                    if len(sel2) >= 2:
                        break
            if 0 not in sel2:
                sel2.append(0)
            if not sel2:
                sel2 = [0]
            sel2 = sel2[:2]

            best2_h = fb2.copy()
            best2_score = exact_score(best2_h, n2)
            for i in sel2:
                s = exact_score(starts2[i], n2)
                if s < best2_score - 1e-12:
                    best2_score = s
                    best2_h = starts2[i].copy()

            pairs2 = build_pair_indices(2 * n2 - 1)
            dirs2 = build_haar(n2)
            for i in sel2:
                if i < len(starts2):
                    cur, sc = optimize_chain(starts2[i], n2, 40, dirs2, pairs2, rng2)
                    if sc < best2_score - 1e-12:
                        best2_score = sc
                        best2_h = cur.copy()

            fin2 = project_capped(best2_h, n2)
            if fin2 is not None:
                fs2 = exact_score(fin2, n2)
                if math.isfinite(fs2) and fs2 < best_score - 1e-12:
                    best_h = fin2.copy()
                    best_score = fs2
                    n = n2

    final = project_capped(best_h, n)
    if final is None:
        final = safe_h.copy()
        final_score = safe_score
        n = safe_n
    else:
        final_score = exact_score(final, n)
        if not math.isfinite(final_score) or final_score > safe_score + 1e-12:
            final = safe_h.copy()
            final_score = safe_score
            n = safe_n

    if not is_feasible(final, n):
        final = project_capped(np.full(n, 0.5, dtype=np.float64), n)
        final_score = exact_score(final, n)
        n = int(final.size)

    return final, float(final_score), int(n)
