import math

def project_capped(z, n):
    # Project z onto {0 <= h <= 1, sum(h) = n/2}
    s = n / 2.0
    z = np.clip(z, -1e12, 1e12)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if abs(hi - lo) < 1e-15:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    # Fix residual
    r = s - float(np.sum(h))
    for _ in range(n + 4):
        if abs(r) < 1e-12:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            add = min(r, 1.0 - h[idx])
            h[idx] += add
            r -= add
        else:
            idx = int(np.argmax(h))
            rem = max(r, -h[idx])
            h[idx] += rem
            r -= rem
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - s) > 1e-9 * max(1, n):
        return None
    return h

def exact_score(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

    corr = np.correlate(h, 1.0 - h, mode='full')
    F = corr[::-1]
    b = np.zeros(2 * n - 1)
    idx = np.array([p - d for d in range(n - 1, -1, -1)])
    pk = (idx >= 0) & (idx < n)
    b[pk] += np.where(pk, 1.0 - h[idx[pk]], 0.0)
    idx = np.array([p + d for d in range(-n + 1, 1)])
    pk = (idx >= 0) & (idx < n)
    b[pk] -= np.where(pk, h[idx[pk]], 0.0)
    idx = np.array([q - d for d in range(n - 1, -1, -1)])
    pk = (idx >= 0) & (idx < n)
    b[pk] -= np.where(pk, 1.0 - h[idx[pk]], 0.0)
    idx = np.array([q + d for d in range(-n + 1, 1)])
    pk = (idx >= 0) & (idx < n)
    b[pk] += np.where(pk, h[idx[pk]], 0.0)
    a = np.zeros(2 * n - 1)
    a[n - 1] = -2.0
    a[q - p + n - 1] = 1.0
    a[p - q + n - 1] = 1.0
    li = np.where(a == 0.0)[0]
    bs, hm, hc = build_hull(b[li], F[li])
    nl = len(hm)
    c0, b0 = F[n - 1], b[n - 1]
    cp, bp_ = F[q - p + n - 1], b[q - p + n - 1]
    cm, bm_ = F[p - q + n - 1], b[p - q + n - 1]
    cands = {L, U, 0.0}
    q_stat = []
    if -0.5 * bp_ > L + 1e-13 and -0.5 * bp_ < U - 1e-13:
        q_stat.append(-0.5 * bp_)
    if -0.5 * bm_ > L + 1e-13 and -0.5 * bm_ < U - 1e-13:
        q_stat.append(-0.5 * bm_)
    endpoints = [L]
    for v in bs[1:]:
        if v > L + 1e-13 and v < U - 1e-13:
            endpoints.append(v)
    endpoints.append(U)
    k = len(endpoints)
    for ri in range(k - 1):
        ra, rb = endpoints[ri], endpoints[ri + 1]
        mid = 0.5 * (ra + rb)
        j = int(np.clip(np.searchsorted(bs, mid, side='right') - 1, 0, nl - 1))
        lm, lc = hm[j], hc[j]
        cands.add(mid)
        cands.update(quad_roots_in(2.0, lm - b0, lc - c0, ra, rb))
        cands.update(quad_roots_in(1.0, bp_ - lm, cp - lc, ra, rb))
        cands.update(quad_roots_in(1.0, bm_ - lm, cm - lc, ra, rb))
        cands.update(quad_roots_in(3.0, bp_ - b0, cp - c0, ra, rb))
        cands.update(quad_roots_in(3.0, bm_ - b0, cm - c0, ra, rb))
        if abs(bp_ - bm_) > 1e-18:
            cands.update(quad_roots_in(0.0, bp_ - bm_, cp - cm, ra, rb))
        for vt in q_stat:
            if vt > ra + 1e-13 and vt < rb - 1e-13:
                cands.add(vt)
    for v in (-0.25 * b0, 0.5 * b0):
        if v > L + 1e-13 and v < U - 1e-13:
            cands.add(v)
    best_t, best_v = 0.0, None
    for t in cands:
        t = max(L, min(U, t))
        j = int(np.clip(np.searchsorted(bs, t, side='right') - 1, 0, nl - 1))
        v0 = hm[j] * t + hc[j]
        v1 = -2.0 * t * t + b0 * t + c0
        v2 = t * t + bp_ * t + cp
        v3 = t * t + bm_ * t + cm
        mv = max(v0, v1, v2, v3)
        if best_v is None or mv < best_v - 1e-15 or (abs(mv - best_v) <= 1e-15 and abs(t) < abs(best_t)):
            best_v, best_t = mv, t
    if abs(best_t) < 1e-13:
        return 0.0
    return float(best_t)

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= len(iv) <= 999:
            n, parent = len(iv), iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
        n = max(1, min(999, n))
    try:
        sd = int(seed)
        if sd < 0:
            sd = 42
    except Exception:
        sd = 42
    rng = np.random.default_rng(sd)
    uniform = np.full(n, 0.5)
    if parent is not None:
        fb = project_capped(parent, n)
        if fb is None:
            fb = uniform.copy()
    else:
        fb = uniform.copy()
    fb_sc = exact_score(fb, n)
    if not np.isfinite(fb_sc):
        fb, fb_sc = uniform.copy(), exact_score(uniform, n)
    if n <= 1:
        return fb, float(fb_sc), int(n)
    starts = [fb.copy()]
    if parent is not None:
        for std in (0.025, 0.08, 0.16):
            p = project_capped(fb + rng.normal(0, std, n), n)
            if p is not None and not any(np.max(np.abs(p - s)) < 1e-8 for s in starts):
                starts.append(p)
        for tr in (fb[::-1], 1.0 - fb, np.roll(fb, n // 3)):
            p = project_capped(tr, n)
            if p is not None and not any(np.max(np.abs(p - s)) < 1e-8 for s in starts):
                starts.append(p)
    if not any(np.max(np.abs(s - uniform)) < 1e-8 for s in starts):
        starts.append(uniform.copy())
    starts = starts[:min(len(starts), 6)]
    for s in starts:
        sc = exact_score(s, n)
        if np.isfinite(sc) and sc < fb_sc - 1e-12:
            fb, fb_sc = s.copy(), sc
    best, best_sc = fb.copy(), fb_sc
    if n <= 128:
        npairs = n * (n - 1) // 2
        allp = [(i, j) for i in range(n) for j in range(i + 1, n)]
        nsw = 3
        for sw in range(nsw):
            if sw > 0:
                best_p = project_capped(best + rng.normal(0, 0.01, n), n)
                if best_p is not None and exact_score(best_p, n) < best_sc:
                    best, best_sc = best_p, exact_score(best_p, n)
            perm = allp[:]
            rng.shuffle(perm)
            improved = False
            for (p, q) in perm:
                t = chord_min(best, p, q, n)
                if abs(t) < 1e-13:
                    continue
                best[p] = min(1.0, max(0.0, best[p] + t))
                best[q] = min(1.0, max(0.0, best[q] - t))
                sc = exact_score(best, n)
                if np.isfinite(sc) and sc < best_sc - 1e-12:
                    best_sc = sc
                    improved = True
                else:
                    best[p] -= t
                    best[q] += t
            if not improved:
                break
    final = project_capped(best, n)
    if final is None:
        final = fb.copy()
    fsc = exact_score(final, n)
    if not np.isfinite(fsc) or fsc > fb_sc + 1e-14:
        final, fsc = fb.copy(), fb_sc
    if not np.isfinite(fsc):
        final, fsc = uniform.copy(), exact_score(uniform, n)
    return final, float(fsc), int(n)
