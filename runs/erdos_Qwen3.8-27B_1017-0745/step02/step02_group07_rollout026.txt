            step = float(rng.choice([0.01, 0.02, 0.05, 0.10, 0.15, 0.25]))
            if rng.random() < 0.5:
                step = -step
            h2 = h.copy()
            h2[i] = np.clip(h2[i] + step, 0.0, 1.0)
            h2 = project_capped(h2, n)
            if h2 is None:
                continue
            s2 = exact_score(h2, n)
            if s2 < cur_s - 1e-13:
                h = h2
                cur_s = s2
                improved = True
                break
        if not improved:
            break
    return h, cur_s

def add_pool(pool, scores, v, n):
    if v is None:
        return
    v = project_capped(v, n)
    if v is None or not is_feasible(v, n):
        return
    for s in pool:
        if np.max(np.abs(v - s)) < 1e-9:
            return
    pool.append(v)
    scores.append(exact_score(v, n))
