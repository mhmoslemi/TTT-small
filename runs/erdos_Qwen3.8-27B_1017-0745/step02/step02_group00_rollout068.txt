def lag_grad_matrix(x,n):
  dx=2/n; m=2*n-1

import numpy as np

def project_capped(z,n):
    z=np.asarray(z,dtype=float).ravel()
    if z.size!=n or not np.all(np.isfinite(z)): return None
    z=np.clip(z,-1e12,1e12)
    s=n/2.0
    lo=float(np.min(z))-1.0; hi=float(np.max(z))+1.0
    for _ in range(80):
        mid=0.5*(lo+hi)
        if mid==lo or mid==hi: break
        if float(np.sum(np.clip(z-mid,0.0,1.0)))>s:
            lo=mid
        else:
            hi=mid
    h=np.clip(z-0.5*(lo+hi),0.0,1.0)
    r=s-float(np.sum(h))
    for i in range(n):
        if abs(r)<1e-13: break
        if r>0:
            a=min(r,1.0-h[i]); h[i]+=a; r-=a
        else:
            a=min(-r,h[i]); h[i]-=a; r+=a
    h=np.clip(h,0.0,1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h))-s)>1e-8:
        return None
    return h

def exact_score(h,n):
    h=np.asarray(h,dtype=float).ravel()
    if h.size!=n or not np.all(np.isfinite(h)): return float('inf')

    val=np.max(np.correlate(h,1.0-h,mode='full'))*(2.0/float(n))

def softmax_weights(q,beta):

import numpy as np
from scipy.optimize import minimize

SMOOTH_STEPS = (0.03, 0.1, 0.3, 1.0, 0.01, 0.2, 3.0, 8.0)
SMOOTH_BETAS = (8.0, 25.0, 80.0, 200.0)
SMOOTH_TOLS = (0.0, 1e-4, 5e-3, 2e-2)
RANDOM_STEPS = (0.02, 0.05, 0.1, 0.2, 0.5, 1.0)


def project_capped(z, n):
    z = np.asarray(z, dtype=float).ravel()
    if z.size != n or not np.all(np.isfinite(z)):
        return None
    s = n / 2.0
    if np.all(z >= -1e-12) and np.all(z <= 1.0 + 1e-12):
        z = np.clip(z, 0.0, 1.0)
        if abs(float(np.sum(z)) - s) <= 1e-10:
            return z
    z = np.clip(z, -1e12, 1e12)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if mid == lo or mid == hi:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for i in range(n):
        if abs(r) < 1e-13:
            break
        if r > 0:
            a = min(r, 1.0 - h[i])
            if a <= 0:
                continue
            h[i] += a
            r -= a
        else:
            a = min(-r, h[i])
            if a <= 0:
                continue
            h[i] -= a
            r += a
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-8:
        return None
    return h


def exact_score(h, n):
    h = np.asarray(h, dtype=float).ravel()
    if h.size != n or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-8) or np.any(h > 1.0 + 1e-8):
        return float('inf')
    if abs(float(np.sum(h)) - n / 2.0) > 1e-7:
        return float('inf')
    v = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / float(n)))
    return v if np.isfinite(v) else float('inf')


def lag_values(h, n):
    dx = 2.0 / float(n)
    m = 2 * n - 1
    q = np.empty(m, dtype=float)
    for d in range(-(n - 1), n):
        r = d + n - 1
        lo = 0 if d >= 0 else -d
        hi = n if d <= 0 else n - d
        q[r] = dx * float(np.dot(h[lo:hi], 1.0 - h[lo + d:hi + d]))
    return q


def weighted_lag_gradient(x, w, n):
    dx = 2.0 / float(n)
    m = 2 * n - 1
    g = np.zeros(n, dtype=float)
    for d in range(-(n - 1), n):
        wv = float(w[d + n - 1])
        if abs(wv) < 1e-15:
            continue
        lo = 0 if d >= 0 else -d
        hi = n if d <= 0 else n - d
        g[lo:hi] += dx * wv * (1.0 - x[lo + d:hi + d])
        g[lo + d:hi + d] -= dx * wv * x[lo:hi]
    return g


def softmax_weights(q, beta):
    q = np.asarray(q, float)
    if q.size == 0:
        return np.array([], dtype=float)
    if not np.all(np.isfinite(q)):
        return np.full(q.size, 1.0 / q.size)
    mx = float(np.max(q))
    a = np.clip(beta * (q - mx), -50.0, 0.0)
    w = np.exp(a)
    s = float(np.sum(w))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(q.size, 1.0 / q.size)
    return w / s


def top_weights(q, tol):
    q = np.asarray(q, float)
    m = q.size
    if m == 0 or not np.all(np.isfinite(q)):
        return np.full(m, 1.0 / m)
    mx = float(np.max(q))
    act = q >= (mx - tol)
    c = int(np.count_nonzero(act))
    if c <= 0:
        return np.full(m, 1.0 / m)
    return act.astype(float) / float(c)


def direction_from_gradient(g):
    if g is None:
        return None
    g = np.asarray(g, float)
    if g.size == 0 or not np.all(np.isfinite(g)):
        return None
    d = -g
    d = d - float(np.mean(d))
    nm = float(np.linalg.norm(d))
    if nm < 1e-10:
        return None
    return d / nm


def unique_dirs(dirs, limit=5):
    out = []
    for d in dirs:
        if d is None:
            continue
        d = np.asarray(d, float)
        if d.size == 0 or not np.all(np.isfinite(d)):
            continue
        nm = float(np.linalg.norm(d))
        if nm < 1e-10:
            continue
        d = d / nm
        dup = False
        for e in out:
            if float(np.dot(d, e)) > 0.999:
                dup = True
                break
        if not dup:
            out.append(d)
        if len(out) >= limit:
            break
    return out


def finalize(h, n):
    p = project_capped(h, n)
    if p is None:
        return None
    s = exact_score(p, n)
    if not np.isfinite(s):
        return None
    return p, s


def smooth_local(x, n, max_iter):
    if max_iter <= 0:
        return finalize(x, n)
    res0 = finalize(x, n)
    if res0 is None:
        return None
    x, best = res0
    q = lag_values(x, n)
    no = 0
    for _ in range(max_iter):
        dirs = []
        for beta in SMOOTH_BETAS:
            w = softmax_weights(q, beta)
            dirs.append(direction_from_gradient(weighted_lag_gradient(x, w, n)))
        for tol in SMOOTH_TOLS:
            w = top_weights(q, tol)
            dirs.append(direction_from_gradient(weighted_lag_gradient(x, w, n)))
        dirs = unique_dirs(dirs, 5)
        if not dirs:
            break
        accept = False
        for d in dirs:
            for s in SMOOTH_STEPS:
                p = project_capped(x + s * d, n)
                if p is None:
                    continue
                sc = exact_score(p, n)
                if np.isfinite(sc) and sc < best - 1e-12:
                    x = p
                    best = sc
                    q = lag_values(x, n)
                    no = 0
                    accept = True
                    break
            if accept:
                break
        if not accept:
            no += 1
            if no > 12:
                break
    return finalize(x, n)


def random_dir_local(x, n, rng, iters):
    if iters <= 0:
        return finalize(x, n)
    res = finalize(x, n)
    if res is None:
        return None
    x, best = res
    no = 0
    for _ in range(iters):
        d = rng.normal(size=n)
        d = d - float(np.mean(d))
        nm = float(np.linalg.norm(d))
        if nm < 1e-12:
            continue
        d = d / nm
        accept = False
        for s in RANDOM_STEPS:
            p = project_capped(x + s * d, n)
            if p is None:
                continue
            sc = exact_score(p, n)
            if np.isfinite(sc) and sc < best - 1e-12:
                x = p
                best = sc
                no = 0
                accept = True
                break
        if not accept:
            no += 1
            if no > 18:
                break
    return finalize(x, n)


def pair_polish(x, n, sweeps):
    res = finalize(x, n)
    if res is None or sweeps <= 0:
        return res
    h, best = res
    for _ in range(sweeps):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                lo = max(-h[i], h[j] - 1.0)
                hi = min(1.0 - h[i], h[j])
                if hi <= lo + 1e-12:
                    continue
                mid = 0.5 * (lo + hi)
                cands = (mid, -mid, 0.5 * mid, -0.5 * mid, lo, hi)
                cur = best
                bd = 0.0
                for dv in cands:
                    if abs(dv) < 1e-6:
                        continue
                    if dv < lo:
                        dv = lo
                    elif dv > hi:
                        dv = hi
                    h[i] += dv
                    h[j] -= dv
                    sc = exact_score(h, n)
                    h[i] -= dv
                    h[j] += dv
                    if np.isfinite(sc) and sc < cur - 1e-12:
                        cur = sc
                        bd = dv
                if bd != 0.0:
                    h[i] += bd
                    h[j] -= bd
                    best = cur
                    improved = True
        if not improved:
            break
    return finalize(h, n)


def random_binary(n, rng):
    s = n / 2.0
    arr = np.zeros(n, dtype=float)
    cnt = int(np.floor(s))
    rem = s - cnt
    idx = rng.permutation(n)
    if cnt > n:
        cnt = n
    arr[idx[:cnt]] = 1.0
    if rem > 1e-9 and cnt < n:
        arr[idx[cnt]] = rem
    return arr


def try_add(arr, n, cands, seen):
    if arr is None:
        return
    res = finalize(arr, n)
    if res is None:
        return
    p, s = res
    for ref, rs in seen:
        if float(np.max(np.abs(p - ref))) < 1e-7:
            return
    seen.append((p.copy(), s))
    cands.append((p.copy(), s))


def get_int_arg(kwargs, key, default, lo, hi):
    try:
        v = int(kwargs[key])
    except Exception:
        return default
    if v < lo:
        return lo
    if v > hi:
        return hi
    return v


def start_key(item):

def lag_grad_matrix(x, n):
    dx = 2.0 / float(n)
    m = 2 * n - 1
    G = np.zeros((m, n), dtype=float)
    for d in range(-(n - 1), n):
        r = d + n - 1
        lo = 0 if d >= 0 else -d
        hi = n if d <= 0 else n - d
        G[r, lo:hi] += dx * (1.0 - x[lo + d:hi + d])
        G[r, lo + d:hi + d] -= dx * x[lo:hi]
    return G


def slsqp_obj(z, n):
    return float(z[n])


def slsqp_obj_jac(z, n):
    j = np.zeros(n + 1, dtype=float)
    j[n] = 1.0
    return j


def slsqp_eq(z, n):
    return float(np.sum(z[:n]) - n / 2.0)


def slsqp_eq_jac(z, n):
    j = np.zeros(n + 1, dtype=float)
    j[:n] = 1.0
    return j


def slsqp_ineq(z, n):
    m = 2 * n - 1
    if z.size != n + 1 or not np.all(np.isfinite(z)):
        return np.full(m, -1e6)
    h = np.clip(z[:n], 0.0, 1.0)
    q = lag_values(h, n)
    if not np.all(np.isfinite(q)):
        return np.full(m, -1e6)
    return float(z[n]) - q


def slsqp_ineq_jac(z, n):
    m = 2 * n - 1
    J = np.zeros((m, n + 1), dtype=float)
    J[:, -1] = 1.0
    if z.size != n + 1 or not np.all(np.isfinite(z)):
        return J
    h = np.clip(z[:n], 0.0, 1.0)
    G = lag_grad_matrix(h, n)
    J[:, :n] = -G
    return J


def run_slsqp(x, n, max_iter):
    res0 = finalize(x, n)
    if res0 is None or max_iter <= 0:
        return res0
    h, best = res0
    t0 = min(1.0, max(1e-9, best + 1e-8))
    z0 = np.concatenate([h, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac, 'args': (n,)},
        {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac, 'args': (n,)},
    ]
    try:
        r = minimize(slsqp_obj, z0, args=(n,), jac=slsqp_obj_jac, method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'ftol': 1e-11, 'maxiter': int(max_iter)})
    except Exception:
        return res0
    if r.x is None or len(r.x) != n + 1 or not np.all(np.isfinite(r.x)):
        return res0
    res = finalize(r.x[:n], n)
    if res is None:
        return res0
    if res[1] < best:
        return res
    return res0
