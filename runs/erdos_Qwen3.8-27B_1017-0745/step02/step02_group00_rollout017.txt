                    if abs(dv) < 1e-12:
                        continue
                    h[i] += dv; h[j] -= dv
                    sv = exact_score(h, n)
                    h[i] -= dv; h[j] += dv
                    if np.isfinite(sv) and sv < best_s - 1e-12:
                        best_s, best_d = sv, dv
                if best_d != 0.0:
                    h[i] += best_d; h[j] -= best_d
                    impr = True
        if not impr:
            break
    return h
