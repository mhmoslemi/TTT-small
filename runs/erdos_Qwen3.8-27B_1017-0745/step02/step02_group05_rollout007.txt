import numpy as np

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    if r > 1e-14:
        for i in range(n):
            if r <= 1e-14:
                break
            add = min(r, 1.0 - h[i])
            h[i] += add
            r -= add
    elif r < -1e-14:
        for i in range(n):
            if r >= -1e-14:
                break
            rem = min(-r, h[i])
            h[i] -= rem
            r += rem
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(np.sum(h) - s) > 1e-10 * max(1, n):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr))

def pair_polish(h, n, rng, sweeps, max_pairs):
    h = h.copy()
    sc = exact_score(h, n)
    best_h = h.copy()
    best_sc = sc
    fracs = np.array([1.0/64, 1.0/16, 1.0/4, 1.0/2, 1.0])
    for sw in range(sweeps):
        any_improve = False
        if n <= 64:
            pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
        else:
            pairs = [(int(rng.integers(n)), int(rng.integers(n))) for _ in range(max_pairs)]
            pairs = [(min(a,b), max(a,b)) for a, b in pairs if a != b]
        seen = set()
        for i, j in pairs:
            if i == j or (i, j) in seen:
                continue
            seen.add((i, j))
            c1 = min(1.0 - h[i], h[j])
            c2 = min(1.0 - h[j], h[i])
            best_local = h
            best_local_s = sc
            if c1 > 1e-12:
                for f in fracs:
                    amt = c1 * f
                    h2 = h.copy()
                    h2[i] += amt
                    h2[j] -= amt
                    h2p = project_capped(h2, n)
                    if h2p is None:
                        continue
                    s2 = exact_score(h2p, n)
                    if np.isfinite(s2) and s2 < best_local_s - 1e-12:
                        best_local_s = s2
                        best_local = h2p
            if c2 > 1e-12:
                for f in fracs:
                    amt = c2 * f
                    h2 = h.copy()
                    h2[j] += amt
                    h2[i] -= amt
                    h2p = project_capped(h2, n)
                    if h2p is None:

    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(np.sum(h) - s) > 1e-10 * max(1, n):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)): return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12): return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n): return False
    return True

def exact_score(h, n):
    if not is_feasible(h, n): return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(np.max(c)) if np.all(np.isfinite(c)) else float('inf')

def build_pop(base, n, rng, P):
    pop, scs = [base.copy()], [exact_score(base, n)]
    cands = [base[::-1].copy(), 1.0 - base, (1.0 - base)[::-1].copy()]
    sh = max(1, n // 4)
    cands += [np.roll(base, sh), np.roll(base, -sh)]
    blk = np.zeros(n); blk[:n//2] = 1.0
    if n % 2 == 1: blk[n//2] = 0.5
    cands += [blk, blk[::-1].copy(), np.where(np.arange(n)%2==0, 1.0, 0.0)]
    for k in range(1, 4):
        cands.append(0.5 + 0.35*np.sin(2*math.pi*k*np.arange(n)/n + rng.uniform(0, 6.28)))
    for sd in (0.03, 0.08, 0.18):
        cands.append(base + rng.normal(0, sd, n))
    cands.append(rng.uniform(0, 1, n))
    for c in cands:
        if len(pop) >= P: break
        p = project_capped(c, n)
        if p is None: continue
        if any(np.max(np.abs(p - m)) < 1e-8 for m in pop): continue
        pop.append(p); scs.append(exact_score(p, n))
    while len(pop) < P:
        p = project_capped(base + rng.normal(0, 0.1, n), n)
        if p is None: p = np.full(n, 0.5)
        pop.append(p); scs.append(exact_score(p, n))
    return pop[:P], scs[:P]

def diff_evol(pop, scs, n, rng, G, CR, stall_lim):
    P = len(pop)
    bi = int(np.argmin(scs)); bv, bsc = pop[bi].copy(), scs[bi]
    stall = 0
    for gen in range(G):
        q = gen / max(1, G - 1)
        Fb = (0.5 + 0.4 * rng.uniform()) * (1.0 - 0.25 * q)
        T = 2e-3 * (1e-5 / 2e-3) ** q
        frozen = [x.copy() for x in pop]; fs = list(scs)
        g = bv.copy(); imp = False
        for i in range(P):
            oth = [j for j in range(P) if j != i]
            a, b, c = (int(x) for x in rng.choice(oth, 3, replace=False))
            if rng.uniform() < 0.25:
                v = frozen[i] + Fb*(g - frozen[i]) + Fb*(frozen[b] - frozen[c])
            else:
                v = frozen[a] + Fb*(frozen[b] - frozen[c])
            mk = rng.random(n) < CR
            if not mk.any(): mk[rng.integers(n)] = True
            tr = project_capped(np.where(mk, v, frozen[i]), n)
            if tr is None: continue
            ts = exact_score(tr, n)
            if not np.isfinite(ts): continue
            d = ts - fs[i]
            if d < -1e-12: acc = True
            elif abs(d) <= 1e-12: acc = rng.uniform() < 0.1
            else: acc = T > 1e-30 and rng.uniform() < math.exp(max(-d/T, -50.0))
            if acc: pop[i], scs[i] = tr, ts
            if ts < bsc - 1e-12: bsc, bv, imp = ts, tr.copy(), True
        pop[0], scs[0] = bv.copy(), bsc
        if imp: stall = 0
        else:
            stall += 1
            if stall >= stall_lim and gen < G - 1:
                kp = max(2, P // 3)
                np_, ns = [bv.copy()], [bsc]
                for j in range(1, kp):
                    idx = (j * 7) % P
                    np_.append(pop[idx].copy()); ns.append(scs[idx])
                while len(np_) < P:
                    if rng.uniform() < 0.5:
                        sg = 0.025 + 0.075*(1-q)
                        rc = project_capped(bv + sg*rng.normal(0, 1, n), n)
                    else:
                        rc = project_capped(rng.uniform(0, 1, n), n)
                    if rc is None: rc = np.full(n, 0.5)
                    rs = exact_score(rc, n)
                    np_.append(rc); ns.append(rs)
                    if rs < bsc - 1e-12: bsc, bv = rs, rc.copy()
                pop, scs = np_[:P], ns[:P]
                pop[0], scs[0] = bv.copy(), bsc
                stall = 0
    return bv, bsc

def pair_anneal(h, n, rng, L):
    h = h.copy(); sc = exact_score(h, n)
    bh, bsc = h.copy(), sc
    for t in range(L):
        q = t / max(1, L - 1)
        T = 5e-4 * (1e-7 / 5e-4) ** q
        rho = 1.0 - 0.98 * q
        ho = h.copy(); found = False
        for _ in range(20):
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j: continue
            c1, c2 = min(1-h[i], h[j]), min(1-h[j], h[i])
            cp = max(c1, c2)
            if cp <= 1e-12: continue
            amt = cp * rho * (0.05 + 0.95 * rng.uniform())
            if amt < 1e-14: continue
            if c1 >= c2: h[i] += amt; h[j] -= amt
            else: h[j] += amt; h[i] -= amt
            found = True; break
        if not found: break
        hp = project_capped(h, n)
        if hp is None: h = ho; continue
        hs = exact_score(hp, n)
        if not np.isfinite(hs): h = ho; continue
        d = hs - sc
        if d < -1e-12: acc = True
        elif d > 1e-12 and T > 1e-30: acc = rng.uniform() < math.exp(max(-d/T, -50.0))
        else: acc = True
        if acc:
            h, sc = hp, hs
            if sc < bsc - 1e-12: bsc, bh = sc, h.copy()
        else: h = ho
    return bh, bsc

def pair_polish(h, n, rng, sweeps, max_pairs):
    h = h.copy(); sc = exact_score(h, n)
    bh, bsc = h.copy(), sc
    fr = np.array([1.0/64, 1.0/16, 1.0/4, 1.0/2, 1.0])
    for sw in range(sweeps):
        any_imp = False
        if n <= 64:
            pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
        else:
            pairs = [(min(int(rng.integers(n)), int(rng.integers(n))),
                      max(int(rng.integers(n)), int(rng.integers(n)))) for _ in range(max_pairs)]
            pairs = [(a, b) for a, b in pairs if a != b]
        seen = set()
        for i, j in pairs:
            if (i, j) in seen: continue
            seen.add((i, j))
            bl, bls = h, sc
            for (pi, pj, cap) in ((i, j, min(1-h[i], h[j])), (j, i, min(1-h[j], h[i]))):
                if cap <= 1e-12: continue
                for f in fr:
                    h2 = h.copy(); h2[pi] += cap*f; h2[pj] -= cap*f
                    h2p = project_capped(h2, n)
                    if h2p is None: continue
                    s2 = exact_score(h2p, n)
                    if np.isfinite(s2) and s2 < bls - 1e-12: bls, bl = s2, h2p
            if bls < sc - 1e-12:
                h, sc = bl, bls; any_imp = True
                if sc < bsc - 1e-12: bsc, bh = sc, h.copy()
        if not any_imp: break
    return bh, bsc

def coord_descent(h, n):
    h = h.copy(); sc = exact_score(h, n)
    bh, bsc = h.copy(), sc
    for _ in range(3):
        imp = False
        for i in range(n):
            for j in range(n):
                if i == j: continue
                c = min(1 - h[i], h[j])
                if c <= 1e-12: continue
                for f in np.array([0.25, 0.5, 0.75, 1.0]):
                    h2 = h.copy(); h2[i] += c*f; h2[j] -= c*f
                    h2p = project_capped(h2, n)
                    if h2p is None: continue
                    s2 = exact_score(h2p, n)
                    if np.isfinite(s2) and s2 < sc - 1e-12:
                        h, sc = h2p, s2; imp = True
                        if sc < bsc - 1e-12: bsc, bh = sc, h.copy()
                        break
        if not imp: break
    return bh, bsc

def run(seed=42, **kwargs):
    n = None; parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0+1e-8):
                parent = p.copy()
    except Exception: pass
    if n is None:
        try:
            v = int(kwargs.get('n_points', 0))
            if 1 <= v <= 999: n = v
        except Exception: pass
    if n is None: n = 47
    if n == 1:
        h1 = np.array([0.5])
        return h1, float(np.max(np.correlate(h1, 1-h1, mode='full')*2.0)), 1
    try: rng = np.random.default_rng(int(seed))
    except Exception: rng = np.random.default_rng(42)
    const = np.full(n, 0.5); csc = exact_score(const, n)
    if parent is not None:
        base = project_capped(parent, n)
        if base is None or not is_feasible(base, n): base = const.copy()
    else: base = const.copy()
    bsc0 = exact_score(base, n)
    safe_h, safe_sc = (base.copy(), bsc0) if bsc0 < csc else (const.copy(), csc)
    if n <= 64: P, G, L, sl, nw, mp = 12, 120, 1600, 20, 2, 0
    elif n <= 256: P, G, L, sl, nw, mp = 8, 70, 500, 14, 2, 160
    else: P, G, L, sl, nw, mp = 6, 25, 80, 7, 1, 40
    P = max(4, P)
    pop, scs = build_pop(base, n, rng, P)
    bv, bsc = safe_h.copy(), safe_sc
    for i in range(len(pop)):
        if scs[i] < bsc - 1e-12: bsc, bv = scs[i], pop[i].copy()
    try:
        dv, ds = diff_evol(pop, scs, n, rng, G, 0.85, sl)
        if ds < bsc - 1e-12: bsc, bv = ds, dv.copy()
    except Exception: pass
    for i in range(len(pop)):
        if scs[i] < bsc - 1e-12: bsc, bv = scs[i], pop[i].copy()
    try:
        ah, as_ = pair_anneal(bv, n, rng, L)
        if as_ < bsc - 1e-12: bsc, bv = as_, ah.copy()
    except Exception: pass
    try:
        ph, ps = pair_polish(bv, n, rng, nw, mp)
        if ps < bsc - 1e-12: bsc, bv = ps, ph.copy()
    except Exception: pass
    try:
        cd, cds = coord_descent(bv, n)
        if cds < bsc - 1e-12: bsc, bv = cds, cd.copy()
    except Exception: pass
    try:
        ah2, as2 = pair_anneal(bv, n, rng, L // 2)
        if as2 < bsc - 1e-12: bsc, bv = as2, ah2.copy()
    except Exception: pass
    try:
        ph2, ps2 = pair_polish(bv, n, rng, nw, mp)
        if ps2 < bsc - 1e-12: bsc, bv = ps2, ph2.copy()
    except Exception: pass
    try:
        cd2, cds2 = coord_descent(bv, n)
        if cds2 < bsc - 1e-12: bsc, bv = cds2, cd2.copy()
    except Exception: pass
    final = project_capped(bv, n)
    if final is None or not is_feasible(final, n): final = safe_h.copy()
    fs = exact_score(final, n)
    if fs > safe_sc + 1e-12: final, fs = safe_h.copy(), safe_sc
    return final.astype(np.float64), float(fs), int(n)
