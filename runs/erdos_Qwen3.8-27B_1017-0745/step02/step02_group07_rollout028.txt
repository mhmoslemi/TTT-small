            hi = mid

def anneal_chain(h0, sc0, n, steps, rng, best_h, best_score, dx):
    h = h0.copy()
    sc = sc0
    T0 = 0.25 * dx
    Tf = 0.01 * dx
    for k in range(steps):
        rho = k / max(1, steps - 1)
        T = T0 * (Tf / T0) ** rho
        delta = 0.18 * (1.0 - 0.75 * rho)
        # Choose move type
        if rng.random() < 0.8:
            # Pair transfer
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            while j == i:
                j = int(rng.integers(n))
            tlo = max(-h[i], h[j] - 1.0)
            thi = min(1.0 - h[i], h[j])
            tlo = max(tlo, -delta)
            thi = min(thi, delta)
            if thi - tlo < 1e-12:
                continue
            t = rng.uniform(tlo, thi)
            cand = h.copy()
            cand[i] += t
            cand[j] -= t
            # No projection needed for pair transfer (mass preserved, bounds checked by interval)
            if not is_feasible(cand, n):
                cand = project_capped(cand, n)
                if cand is None:
                    continue
        else:
            # Collective move
            z = rng.normal(0, 1, n)
            z -= np.mean(z)
            mx = np.max(np.abs(z))
            if mx < 1e-14:
                continue
            z /= mx
            cand = h + delta * z
            cand = project_capped(cand, n)
            if cand is None:
                continue
        if np.max(np.abs(cand - h)) < 1e-10:
            continue
        new_sc = exact_score(cand, n)
        if not np.isfinite(new_sc):
            continue
        # Update global best
        if new_sc < best_score[0] - 1e-12:
            best_score[0] = new_sc
            best_h[:] = cand
        # Metropolis acceptance
        if new_sc <= sc + 1e-12:
            h = cand
            sc = new_sc
        else:
            log_acc = math.log(max(rng.random(), 1e-300))
            if log_acc < -(new_sc - sc) / T:
                h = cand
                sc = new_sc
    return h, sc

    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(70):

            lo = mid
        else:
            hi = mid
