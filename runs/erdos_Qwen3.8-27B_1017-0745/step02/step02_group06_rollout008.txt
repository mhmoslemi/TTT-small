    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-13:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx])
            h[idx] += d
            r -= d
        else:
            d = min(-r, h[idx])
            h[idx] -= d
            r += d
    if abs(r) > 1e-9 or not np.all(np.isfinite(h)):
        return None
    return np.clip(h, 0.0, 1.0)

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h))
            and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12)
            and abs(np.sum(h) - n / 2.0) <= 1e-9)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if c.shape[0] != 2 * n - 1:
        return float('inf')
    return float(np.max(c))

def sg(h, n, tau):
    """Smooth value and gradient via log-sum-exp."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    a = float(np.max(c))
    e = np.exp(np.clip((c - a) / tau, -745.0, None))
    se = float(np.sum(e))
    if se <= 0.0:
        return a, np.zeros(n)
    val = a + tau * (math.log(se) - math.log(2.0 * n - 1))
    w = e / se
    g = np.zeros(n)
    for d in range(-(n - 1), n):
        i0 = max(0, -d)
        i1 = min(n, n - d)
        if i1 <= i0:
            continue
        j0 = i0 + d
        wi = w[d + n - 1] * dx
        g[j0:j0 + (i1 - i0)] += wi * (1.0 - h[i0:i1])
        g[i0:i1] -= wi * h[j0:j0 + (i1 - i0)]
    if not np.all(np.isfinite(g)):
        return float('inf'), None
    return val, g

def smooth_descent(x0, n, taus, max_it, best_h, best_s):
    x = x0.copy()
    for tau in taus:
        alpha = 1.0
        for _ in range(max_it):
            fv, g = sg(x, n, tau)
            if g is None:
                break
            for _bt in range(20):
                y = project_capped(x - alpha * g, n)
                if y is None:
                    break
                d = y - x
                dn = float(np.dot(d, d))
                if dn < 1e-20:
                    return x, best_h, best_s
                fyt, _ = sg(y, n, tau)
                if fyt <= fv - 1e-3 * dn / (alpha + 1e-30) + 1e-14:
                    # Accept
                    break
                alpha *= 0.5
            else:
                break
            # Newton-like step size update
            _, gn = sg(y, n, tau)
            if gn is not None:
                s_dir = y - x
                v = gn - g
                sv = float(np.dot(s_dir, v))
                ss = float(np.dot(s_dir, s_dir))
                if sv > 1e-14:
                    alpha = min(10.0, max(1e-4, ss / sv))
            x = y
            s = exact_score(x, n)
            if s < best_s - 1e-12:
                best_h[:] = x
                best_s = s
    return x, best_h, best_s

def pair_polish(h, n, moves, top_k):
    h = h.copy()

    ii, jj = np.triu_indices(n, 1)

        best_t, best_s, best_p = 0.0, sc, None
        for pi in range(n_check):

            Lp = max(-h[i], h[j] - 1.0)
