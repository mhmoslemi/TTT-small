    fallback = np.asarray(fallback, dtype=np.float64)
    fb_score = float(exact_score(fallback, n))
    if n == 1:
        return fallback, fb_score, int(n)
    rng = np.random.default_rng(int(seed) if isinstance(seed, (int, float)) else 42)
    if n <= 64:
        t_starts, t_iter, t_hops = 10, 240, 8
    elif n <= 128:
        t_starts, t_iter, t_hops = 7, 180, 5
    elif n <= 256:
        t_starts, t_iter, t_hops = 4, 100, 2
    else:
        t_starts, t_iter, t_hops = 2, 50, 0
    starts = build_starts(fallback, n, rng, t_starts)
    gb = fallback.copy()
    gs = fb_score
    for st in starts:
        y = solve_epigraph(st, n, t_iter)
        if y is not None and is_feasible(y, n):
            sy = float(exact_score(y, n))
            if sy < gs - 1e-12:
                gs, gb = sy, y.copy()
    if t_hops > 0:
        cur = gb.copy()
        cur_s = gs
        sigmas = [0.04, 0.10, 0.20]
        for r in range(t_hops):
            T = max(1e-5, 1.5e-3 * (0.8 ** r))
            if r % 4 == 3:
                prop = project_capped(rng.uniform(0, 1, n), n)
            else:
                prop = project_capped(cur + sigmas[r % 3] * rng.standard_normal(n), n)
            if prop is None:
                continue
            y = solve_epigraph(prop, n, min(t_iter, 120))
            if y is None or not is_feasible(y, n):
                continue
            sy = float(exact_score(y, n))
            if sy < gs - 1e-12:
                gs, gb = sy, y.copy()
                cur, cur_s = y.copy(), sy
            elif sy <= cur_s:
                cur, cur_s = y.copy(), sy
            else:
                p_acc = np.exp(-(sy - cur_s) / max(T, 1e-15))
                if rng.uniform() < p_acc:
                    cur, cur_s = y.copy(), sy
    gb = project_capped(gb, n)
    if gb is None or not is_feasible(gb, n):
        gb, gs = fallback.copy(), fb_score
    else:
        gs = float(exact_score(gb, n))
        if gs > fb_score + 1e-12:
            gb, gs = fallback.copy(), fb_score
    if gs == float('inf'):
        gb = np.full(n, 0.5)
        gs = float(exact_score(gb, n))
    return gb, gs, int(n)
