        if not any(np.max(np.abs(rr - s)) < 1e-8 for s in starts):
            starts.append(rr)
        else:
            starts.append(uniform.copy())
    return starts[:6]

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= len(iv) <= 999 and np.all(np.isfinite(iv)):
            n = len(iv)
            parent = iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
        n = max(1, min(999, n))
    try:
        sd = int(seed)
        if sd < 0:
            sd = 42
    except Exception:
        sd = 42
    sd = sd % (2**32)
    rng = np.random.default_rng(sd)
    if n == 1:
        h1 = np.array([0.5])
        return h1, float(np.max(np.correlate(h1, 1.0 - h1, mode='full')) * 2.0), 1
    uniform = np.full(n, 0.5)
    if parent is not None:
        fb = project_capped(parent, n)
        if fb is None:
            fb = uniform.copy()
    else:
        fb = uniform.copy()
    fb_score = exact_score(fb, n)
    if not np.isfinite(fb_score):
        fb = uniform.copy()
        fb_score = exact_score(fb, n)
    B_cap = min(16000, max(120, int(30000000 / (n * n))))
    try:
        B_user = int(kwargs.get('eval_budget', 0))
    except Exception:
        B_user = 0
    B = max(120, min(B_user, B_cap, 16000)) if B_user > 0 else B_cap
    temps = [0.0005, 0.002, 0.008, 0.032, 0.128, 0.512]
    scales = [0.002, 0.006, 0.018, 0.054, 0.162, 0.486]
    n_rep = 6
    starts = build_starts(n, parent, rng)
    reps, rep_scores = [], []
    for st in starts[:n_rep]:
        h_int = 0.99 * st + 0.01 * 0.5
        pp = project_capped(h_int, n)
        if pp is None:
            pp = st.copy()
        sc = exact_score(pp, n)
        reps.append(pp)
        rep_scores.append(sc)
    best = fb.copy()
    best_score = fb_score
    for i in range(n_rep):
        if rep_scores[i] < best_score - 1e-12:
            best = reps[i].copy()
            best_score = rep_scores[i]
    budget = n_rep
    stagnation = 0
    restarts_done = 0
    max_stag = max(30, min(200, int(B // 72)))
    for sweep in range(B + 1):
        if budget >= B:
            break
        sb = best_score
        for i in range(n_rep):
            if budget >= B:
                break
            h_cur = reps[i]
            sc_cur = rep_scores[i]
            T = temps[i]
            scale = scales[i]
            u = float(rng.random())
            if u < 0.35:
                h_prop = hit_and_run(h_cur, n, rng)
            else:
                h_prop = pair_proposal(h_cur, scale, rng, n)
            budget += 1
            if h_prop is None:
                continue
            sc_prop = exact_score(h_prop, n)
            if not np.isfinite(sc_prop):
                continue
            delta = sc_prop - sc_cur
            if delta <= 0:
                pass
            else:
                v = float(rng.random())
                if v <= 0 or log(max(v, 1e-300)) > -delta / T:
                    continue
            reps[i] = h_prop
            rep_scores[i] = sc_prop
            if sc_prop < best_score - 1e-12:
                best = h_prop.copy()
                best_score = sc_prop
        if sweep % 2 == 0:
            pswap = [(0, 1), (2, 3), (4, 5)]
        else:
            pswap = [(1, 2), (3, 4)]
        for (a, b) in pswap:
            if a >= n_rep or b >= n_rep:
                continue
            Ca, Cb = rep_scores[a], rep_scores[b]
            if not np.isfinite(Ca) or not np.isfinite(Cb):
                continue
            lr = (1.0 / temps[a] - 1.0 / temps[b]) * (Ca - Cb)
            do_swap = True
            if lr < 0:
                v = float(rng.random())
                do_swap = v > 0 and log(max(v, 1e-300)) < lr
            if do_swap:
                reps[a], reps[b] = reps[b], reps[a]
                rep_scores[a], rep_scores[b] = rep_scores[b], rep_scores[a]
        if best_score < sb - 1e-12:
            stagnation = 0
        else:
            stagnation += 1
        if stagnation >= max_stag and restarts_done < 2:
            restarts_done += 1
            stagnation = 0
            if restarts_done == 1:
                rs = 0.5 * best + 0.5 * rng.uniform(0, 1, n)
            else:
                rs = 0.7 * rng.beta(0.7, 0.7, n) + 0.3 * rng.beta(0.7, 0.7, n)
            pp = project_capped(rs, n)
            if pp is None:
                pp = project_capped(rng.uniform(0, 1, n), n)
            if pp is not None:
                hi2 = 0.99 * pp + 0.01 * 0.5
                pp2 = project_capped(hi2, n)
                if pp2 is not None:
                    sc = exact_score(pp2, n)
                    budget += 1
                    reps[5] = pp2
                    rep_scores[5] = sc
                    if np.isfinite(sc) and sc < best_score - 1e-12:
                        best = pp2.copy()
                        best_score = sc
    final = project_capped(best, n)
    if final is None:
        final = fb.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final = fb.copy()
        fs = fb_score
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final, float(fs), int(n)
