def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(float(np.sum(h)) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def project_capped(z, n):
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    s = n / 2.0
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(3):
        r = s - float(np.sum(h))
        if abs(r) < 1e-14:
            break
        for i in range(n):
            if abs(r) < 1e-14:
                break
            if r > 0:
                c = min(r, 1.0 - h[i])
                h[i] += c
                r -= c
            else:
                c = min(-r, h[i])
                h[i] -= c
                r += c
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):

def exact_score(h, n):

        return float('inf')

    if q.shape[0] != 2 * n - 1 or not np.all(np.isfinite(q)):

def build_directions(n):
    dirs = []
    stack = [(0, n)]
    while stack:
        a, b = stack.pop()
        L = b - a
        if L < 2:
            continue
        L0 = L // 2
        L1 = L - L0
        v = np.zeros(n, dtype=np.float64)
        v[a:a + L0] = L1
        v[a + L0:b] = -L0
        dirs.append(v)
        if a + L0 < b:
            stack.append((a + L0, b))
        if a < a + L0:
            stack.append((a, a + L0))
    return dirs

def append_roots(A, B, C, lo, hi, out):
    scale = max(abs(A), abs(B), abs(C), 1e-300)
    tol = 1e-14 * scale
    if abs(A) <= tol:
        if abs(B) > tol:
            r = -C / B
            if lo - 1e-10 <= r <= hi + 1e-10:
                out.append(max(lo, min(hi, r)))
        return
    disc = B * B - 4.0 * A * C
    if disc < -1e-13 * (B * B + abs(4.0 * A * C) + 1e-300):
        return
    if disc < 0.0:
        disc = 0.0
    s = math.sqrt(disc)
    q = -0.5 * (B + math.copysign(s, B))
    roots = []
    if abs(q) > 1e-300:
        roots = [q / A, C / q]
    else:
        roots = [-B / (2.0 * A)]
    for r in roots:
        if np.isfinite(r) and lo - 1e-10 <= r <= hi + 1e-10:
            out.append(max(lo, min(hi, r)))

    cands = [lo, hi, 0.0]

            B = b[i] - b[j]
            C = c[i] - c[j]

    for v in cands[1:]:

    return dedup

        return None

    dx = 2.0 / n
