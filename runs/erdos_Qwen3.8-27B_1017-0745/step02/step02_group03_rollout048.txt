    Returns (starts, idx) where starts[0]=-inf, starts[-1]=+inf,

    if n == 0:
        return np.array([-np.inf, np.inf]), np.array([], dtype=int)
    if n == 1:
        return np.array([-np.inf, np.inf]), np.array([0])
    # Sort by slope, then by -intercept (for dedup)

        if abs(B) < 1e-15:
            return roots
        r = -C / B
        if L - 1e-12 <= r <= U + 1e-12:

For upper hull with increasing slopes:
- Start with the leftmost (smallest slope) line.
- For each new line (increasing slope):
  - Compute intersection t with current top line.
  - If t <= start of top line, pop top line and recompute.
  - Repeat until t > start of top line (or only 1 line remains).
  - Add new line with start = t.

Now the chord minimizer. Let me think about the interface.

Input: h (current values), p, q (pair indices), n.
Output: optimal t.

Steps:
1. Compute c = np.correlate(h, 1-h, 'full')
2. Compute b for each lag d: b[d] = G[d,p] - G[d,q] where G is the gradient
3. Compute a for each lag: a[0]=0(a_d=0 for most), a[n-1]=-2, a[n-1±(q-p)]=1
4. Build hulls for groups
5. Collect candidates
6. Evaluate and return best t

def _chord_minimize(h, n, p, q):
    L = max(-h[p], h[q] - 1.0)
    U = min(1.0 - h[p], h[q])
    if U - L < 1e-14:
        return 0.0
    omh = 1.0 - h
    c = np.correlate(h, omh, mode='full')  # length 2n-1, c[d+n-1] = F_d
    L2 = 2 * n - 1
    dp = q - p  # positive since p < q
    # b[d] = G[d,p] - G[d,q]
    b = np.zeros(L2)
    for d in range(-(n-1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        gp = 0.0
        if lo <= p < hi:
            gp += omh[p + d]
        if lo <= p - d < hi:
            gp -= h[p - d]
        gq = 0.0
        if lo <= q < hi:
            gq += omh[q + d]
        if lo <= q - d < hi:
            gq -= h[q - d]
        b[d + n - 1] = gp - gq
    a = np.zeros(L2)
    a[n - 1] = -2.0
    a[n - 1 + dp] = 1.0
    a[n - 1 - dp] = 1.0
    # Group indices
    idx_n2 = [n - 1]  # a=-2
    idx_p1 = [n - 1 + dp, n - 1 - dp]  # a=+1
    idx_0 = [i for i in range(L2) if i not in (n-1, n-1+dp, n-1-dp)]
    # Build hulls
    # Group 0: lines b[i]*t + c[i]
    m0 = np.array([b[i] for i in idx_0], dtype=np.float64)
    c0 = np.array([c[i] for i in idx_0], dtype=np.float64)
