            for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625]:
                trial = x + alpha * (y_sol - x)
                pt = project_capped(trial, n)
                if pt is None or not is_feasible(pt, n):
                    continue
                ts = exact_score(pt, n)
                if ts < best_ts - 1e-13:
                    best_ts, best_tx = ts, pt
            if best_tx is None:
                break
            step_m = float(np.max(np.abs(best_tx - x)))
            x, xscore = best_tx, best_ts
            if xscore < gs - 1e-12:
                gs, gb = xscore, x.copy()
            if step_m < 1e-9:
                break

    # Subgradient polish from best incumbent
    hp, sp = subgradient_polish(gb, n, steps=100)
    if sp < gs - 1e-12:
        gs, gb = sp, hp

    # Pair polish
    hp, sp = pair_polish(gb, n, rng, rounds=150)
    if sp < gs - 1e-12:
        gs, gb = sp, hp

    # Second round: re-run MM from polished point
    if len(solver_order) > 0:
        x = gb.copy()
        xscore = gs
        for it in range(8):
            L, G = lag_data(x, n)
            if L is None:
                break
            y_sol = solve_mm_qp(x, L, G, n, solver_order)
            if y_sol is None:
                break
            best_tx, best_ts = None, xscore
            for alpha in [1.0, 0.5, 0.25]:
                trial = x + alpha * (y_sol - x)
                pt = project_capped(trial, n)
                if pt is None:
                    continue
                ts = exact_score(pt, n)
                if ts < best_ts - 1e-13:
                    best_ts, best_tx = ts, pt
            if best_tx is None:
                break
            step_m = float(np.max(np.abs(best_tx - x)))
            x, xscore = best_tx, best_ts
            if xscore < gs - 1e-12:
                gs, gb = xscore, x.copy()
            if step_m < 1e-9:
                break
        hp, sp = pair_polish(gb, n, rng, rounds=80)
        if sp < gs - 1e-12:
            gs, gb = sp, hp

    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    fbs = exact_score(fallback, n)
    if fs > fbs + 1e-12:
        final, fs = fallback, fbs
    return final, float(fs), int(n)
