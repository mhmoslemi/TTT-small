def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)

        return None
    return h

def is_feasible(h, n):

    return abs(np.sum(h) - n / 2.0) <= 1e-9 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    try:
        v = np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)
        if not np.isfinite(v):
            return float('inf')
        return float(v)
    except Exception:
        return float('inf')

def lag_values(h, n, need_jac=False):
    dx = 2.0 / n
    Q = np.zeros(2 * n - 1, dtype=np.float64)
    Q[0] = np.sum(h * (1.0 - h))
    G = None
    if need_jac:
        G = np.zeros((2 * n - 1, n), dtype=np.float64)
        G[0] = 1.0 - 2.0 * h
    for r in range(1, n):
        m = n - r
        Q[2 * r - 1] = np.dot(h[:m], 1.0 - h[r:])
        Q[2 * r] = np.dot(h[r:], 1.0 - h[:m])
        if need_jac:
            G[2 * r - 1, :m] += 1.0 - h[r:]
            G[2 * r - 1, r:] -= h[:m]
            G[2 * r, :m] -= h[r:]
            G[2 * r, r:] += 1.0 - h[:m]
    Q *= dx
    if G is not None:
        G *= dx
    if need_jac:
        return Q, G
    return Q

def slsqp_ineq(z, n):
    h = z[:n]
    t = z[n]
    Q, _ = lag_values(h, n, need_jac=False)
    return t - Q

def slsqp_ineq_jac(z, n):
    h = z[:n]
    Q, G = lag_values(h, n, need_jac=True)
    J = np.zeros((2 * n - 1, n + 1), dtype=np.float64)
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def slsqp_eq(z, n):
    return np.sum(z[:n]) - n / 2.0

def slsqp_eq_jac(z, n):
    J = np.zeros(n + 1, dtype=np.float64)
    J[:n] = 1.0
    return J

def slsqp_obj(z, n):
    return z[n]

    return J

def slsqp_solve(h0, n):

        t0 = max(t0, 1e-3)

                       args=(n,), bounds=bnds, constraints=cons,
                       options={'ftol': 1e-10, 'maxiter': mx, 'disp': False})
        return res.x

    dx = 2.0 / n
    Q, _ = lag_values(h, n, need_jac=True)  # actually just need Q
    # recompute Q without jac for efficiency

    Qv[0] = np.sum(h * (1.0 - h))

        Qv[2 * r] = np.dot(h[r:], 1.0 - h[:m])
    Qv *= dx
    qmax = np.max(Qv)
    ex = np.exp((Qv - qmax) / tau)
    s = np.sum(ex)
    p = ex / s
    g = np.zeros(n, dtype=np.float64)
    g += p[0] * dx * (1.0 - 2.0 * h)
    for r in range(1, n):

        g[r:] += dx * p[2 * r] * (1.0 - h[:m])
    F = qmax + tau * np.log(s)
    return F, g
