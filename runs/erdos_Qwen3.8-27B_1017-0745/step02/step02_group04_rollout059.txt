    if np.any(v < -1e-6) or np.any(v > 1.0 + 1e-6):
        return
    for s in pool:
        if np.max(np.abs(v - s)) < 1e-8:
            return
    pool.append(v.copy())

def build_starts(fb, n, rng, count):
    pool = []
    _try_add(pool, fb, n)
    _try_add(pool, project_capped(fb[::-1].copy(), n), n)
    _try_add(pool, project_capped(1.0 - fb, n), n)
    _try_add(pool, project_capped((1.0 - fb)[::-1].copy(), n), n)
    _try_add(pool, np.full(n, 0.5), n)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    _try_add(pool, project_capped(blk, n), n)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    _try_add(pool, project_capped(alt, n), n)
    _try_add(pool, project_capped(fb + rng.normal(0, 0.05, n), n), n)
    _try_add(pool, project_capped(fb + rng.normal(0, 0.18, n), n), n)
    P = max(2, n // 3)
    cos_vec = 0.5 + 0.45 * np.cos(2.0 * np.pi * np.arange(n) / P)
    _try_add(pool, project_capped(cos_vec, n), n)
    _try_add(pool, project_capped(rng.uniform(0, 1, n), n), n)
    return pool[:count]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        if 'n_points' in kwargs:
            try:
                v = int(kwargs['n_points'])
                if 1 <= v <= 999:
                    n = v
            except Exception:
                pass
        if n is None:
            n = 47
    if parent is not None:
        fb = project_capped(parent, n)
    else:
        fb = np.full(n, 0.5)
    if fb is None:
        fb = np.full(n, 0.5)
    fb = np.asarray(fb, dtype=np.float64)
    if n == 1:
        return fb, float(exact_score(fb, n)), int(n)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 64:
        n_starts, maxiter, n_hops = 10, 240, 8
    elif n <= 128:
        n_starts, maxiter, n_hops = 7, 180, 5
    elif n <= 256:
        n_starts, maxiter, n_hops = 4, 100, 2
    else:
        n_starts, maxiter, n_hops = 2, 50, 0
    starts = build_starts(fb, n, rng, n_starts)
    best = fb.copy()
    best_s = float(exact_score(fb, n))
    for st in starts:
        s0 = float(exact_score(st, n))
        y = solve_epigraph(st, n, maxiter)
        if y is not None and is_feasible(y, n):
            sy = exact_score(y, n)
            if sy < s0:
                s0, st = sy, y.copy()
        if s0 < best_s - 1e-12:
            best_s, best = s0, st.copy()
    cur = best.copy()
    cur_s = best_s
    sigmas = [0.04, 0.10, 0.20]
    T0 = 1.5e-3
    for r in range(n_hops):
        if r % 4 == 3:
            prop = project_capped(rng.uniform(0, 1, n), n)
        else:
            sig = sigmas[r % 3]
            prop = project_capped(cur + sig * rng.standard_normal(n), n)
        if prop is None:
            continue
        y = solve_epigraph(prop, n, maxiter)
        if y is not None and is_feasible(y, n):
            cand_s = exact_score(y, n)
        else:
            y = prop
            cand_s = exact_score(y, n)
        if cand_s < best_s - 1e-12:
            best_s, best = cand_s, y.copy()
        T = T0 * (0.8 ** r)
        d = cand_s - cur_s
        if d <= 0:
            cur, cur_s = y.copy(), cand_s
        else:
            p_acc = np.exp(-d / max(1e-5, T))
            if rng.random() < p_acc:
                cur, cur_s = y.copy(), cand_s
    best = project_capped(best, n)
    if best is None or not is_feasible(best, n):
        best = fb.copy()
    fs = float(exact_score(best, n))
    fb_s = float(exact_score(fb, n))
    if fs > fb_s + 1e-12:
        best, fs = fb.copy(), fb_s
    if fs == float('inf'):
        best = np.full(n, 0.5)
        fs = float(exact_score(best, n))
    return best, fs, int(n)
