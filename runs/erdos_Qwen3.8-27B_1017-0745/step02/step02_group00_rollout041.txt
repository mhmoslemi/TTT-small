        if 1 <= iv.size <= 999 and np.all(np.isfinite(iv)) and np.all(iv >= -1e-8) and np.all(iv <= 1.0 + 1e-8):
            n = iv.size; parent = np.clip(iv, 0.0, 1.0)
    except Exception:
        pass
    try:
        n = max(1, min(int(kwargs.get('n_points', n)), 999))
    except Exception:
        pass
    sd = 42
    try:
        sd = int(seed) % (2**31)
    except Exception:
        pass
    rng = np.random.default_rng(sd)
    uniform = np.full(n, 0.5)
    fallback = uniform.copy(); fb_sc = exact_score(fallback, n)
    if parent is not None:
        pf = project_capped(parent, n)
        if pf is not None:
            ps = exact_score(pf, n)
            if np.isfinite(ps) and ps < fb_sc:
                fallback = pf.copy(); fb_sc = ps
    starts = [fallback.copy()]
    if parent is not None:
        for raw in [parent[::-1], 1.0 - parent, 1.0 - parent[::-1]]:
            p = project_capped(raw, n)
            if p is not None:
                starts.append(p)
        for s2 in [0.03, 0.08, 0.15]:
            p = project_capped(parent + s2 * rng.standard_normal(n), n)
            if p is not None:
                starts.append(p)
    idx = np.arange(n, dtype=np.float64)
    for freq in [1, 2, 3]:
        p = project_capped(0.5 + 0.45 * np.cos(2.0 * np.pi * freq * idx / n), n)
        if p is not None:
            starts.append(p)
    p = project_capped(rng.uniform(0, 1, n), n)
    if p is not None:
        starts.append(p)
    if parent is None:
        p = project_capped(uniform + 0.12 * rng.standard_normal(n), n)
        if p is not None:
            starts.append(p)
    dedup = []
    for s in starts:
        dup = False
        for d in dedup:
            if float(np.max(np.abs(s - d))) < 1e-8:
                dup = True
                break
        if not dup:
            dedup.append(s)
    starts = dedup
    max_pairs = n * (n - 1) // 2
    if n <= 64:
        ms, mw, pb, nk = 8, 10, max_pairs, 3
    elif n <= 128:
        ms, mw, pb, nk = 5, 8, max_pairs, 2
    elif n <= 256:
        ms, mw, pb, nk = 3, 5, 4 * n, 1
    else:
        ms, mw, pb, nk = 2, 4, 2 * n, 1
    try:
        ms = max(1, min(int(kwargs.get('starts', ms)), 12))
        mw = max(0, min(int(kwargs.get('max_sweeps', mw)), 20))
        pb = max(0, min(int(kwargs.get('pair_budget', pb)), max_pairs))
        nk = max(0, min(int(kwargs.get('kick_rounds', nk)), 5))
    except Exception:
        pass
    starts = starts[:ms]
    best = fallback.copy(); best_sc = fb_sc
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_sc - 1e-12:
            best, best_sc = st.copy(), ss
        if mw <= 0:
            continue
        res_h, res_sc = cd_optimize(st, n, rng, mw, pb, max_pairs)
        if np.isfinite(res_sc) and res_sc < best_sc - 1e-12:
            best, best_sc = res_h.copy(), res_sc
    for kr in range(nk):
        sigma = 0.04 + 0.02 * kr
        pert = project_capped(best + sigma * rng.standard_normal(n), n)
        if pert is None:
            continue
        res_h, res_sc = cd_optimize(pert, n, rng, min(mw, 6), pb, max_pairs)
        if np.isfinite(res_sc) and res_sc < best_sc - 1e-12:
            best, best_sc = res_h.copy(), res_sc
    final = project_capped(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > best_sc + 1e-10:
        final = np.clip(best, 0.0, 1.0)
        fs = exact_score(final, n)
    if not np.isfinite(fs):
        final = uniform.copy(); fs = exact_score(final, n)
    return final, float(fs), int(n)
