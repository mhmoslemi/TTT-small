- A point where two quadratics cross (and that point is on the upper envelope)

def project_capped(z, n):
    if len(z) != n or not np.all(np.isfinite(z)):
        return None

        else:
            hi = mid

    r = s - np.sum(h)
    for i in range(n):
        if abs(r) < 5e-13 * n:
            break
        if r > 0:
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
        else:
            a = min(-r, h[i])
            h[i] -= a
            r += a
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None

        return None
    return h

        return float('inf')

def line_coeffs(h, v, n):
    dx = 2.0 / n
    L = 2 * n - 1
    A = np.zeros(L)
    B = np.zeros(L)
    C = np.zeros(L)

        p0 = max(s, 0)
        r0 = max(-s, 0)

        B[row] = dx * (np.dot(vp, 1.0 - hr) - np.dot(hp, vr))

    return A, B, C

    for c in cands:
        if len(pool) >= count:
            break
        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-8:
                dup = True
                break
        if not dup:
            pool.append(p)
    return pool[:count]

def get_freqs(n, count):
    if count >= n - 1:
        return list(range(1, n))
    k = np.round(np.exp(np.linspace(0.0, math.log(n - 1), count))).astype(int)
    k = np.sort(np.unique(np.clip(k, 1, n - 1)))
    return list(k)

def do_sweep(cur, cur_s, n, freq_list, n_box, rng, gs_ref):
    improved = False
    order = freq_list[:]
    rng.shuffle(order)
    for k in order:
        v = dct_dir(k, n)
        if v is None:
            continue
        try:
            nh, ns = exact_line(cur, v, n, cur_s) if n <= 64 else grid_line(cur, v, n, cur_s)
            if nh is not None:
                cur, cur_s, improved = nh, ns, True
        except Exception:
            continue
    for _ in range(n_box):
        try:
            v = box_dir(cur, n, rng)
            if v is None:
                continue
            nh, ns = exact_line(cur, v, n, cur_s) if n <= 64 else grid_line(cur, v, n, cur_s)
            if nh is not None:
                cur, cur_s, improved = nh, ns, True
        except Exception:
            continue
    return cur, cur_s, improved

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999 and np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
            n, parent = int(p.size), p
    except Exception:
        pass
    if n is None:
        try:
            v = int(kwargs.get('n_points', 47))
            n = v if 1 <= v <= 999 else 47
        except Exception:
            n = 47
    rng = np.random.default_rng(int(seed))
    fb = project_capped(parent, n) if parent is not None else np.full(n, 0.5)
    if fb is None:
        fb = np.full(n, 0.5)
    if n == 1:
        return fb, exact_score(fb, 1), 1
    gs, gb = exact_score(fb, n), fb.copy()
    n_sc = 4 if n <= 256 else 2
    starts = build_starts(fb, n, rng, n_sc)
    for st in starts:
        s = exact_score(st, n)
        if s < gs - 1e-12:
            gs, gb = s, st.copy()
    if n <= 48:
        freq_list = list(range(1, n))
    elif n <= 64:
        freq_list = get_freqs(n, 24)
    elif n <= 256:
        freq_list = get_freqs(n, 12)
    else:
        freq_list = get_freqs(n, 8)
    n_box = 2 if n <= 256 else 1
    n_sweeps = 2 if n <= 64 else 1
    for st in starts:
        cur = st.copy()
        cur_s = exact_score(cur, n)
        if cur_s < gs - 1e-12:
            gs, gb = cur_s, cur.copy()
        for sw in range(n_sweeps + 1):
            cur, cur_s, improved = do_sweep(cur, cur_s, n, freq_list, n_box, rng, gs)
            if cur_s < gs - 1e-12:
                gs, gb = cur_s, cur.copy()
            if not improved:
                break
    for _ in range(n_sweeps):
        cur = gb.copy()
        cur_s = exact_score(cur, n)
        cur, cur_s, improved = do_sweep(cur, cur_s, n, freq_list, n_box, rng, gs)
        if cur_s < gs - 1e-12:
            gs, gb = cur_s, cur.copy()
        if not improved:
            break
    for _ in range(2):
        try:
            pert = project_capped(gb + rng.normal(0, 0.04, n), n)
            if pert is None:
                continue
            cur = pert.copy()
            cur_s = exact_score(cur, n)
            cur, cur_s, _ = do_sweep(cur, cur_s, n, freq_list, n_box, rng, gs)
            if cur_s < gs - 1e-12:
                gs, gb = cur_s, cur.copy()
        except Exception:
            continue
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb
    sf = exact_score(final, n)
    ss = exact_score(fb, n)
    if sf > ss:
        final, sf = fb, ss
    if not is_feasible(final, n):
        final = np.full(n, 0.5)
        sf = exact_score(final, n)
    return final, float(sf), int(n)
