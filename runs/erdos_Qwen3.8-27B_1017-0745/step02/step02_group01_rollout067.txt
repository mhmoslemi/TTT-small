        v = np.zeros(n)
        v[a:a + L0] = L1
        v[a + L0:b] = -L0
        dirs.append(v)
        if b - (a + L0) >= 2:
            stack.append((a + L0, b))
        if (a + L0) - a >= 2:
            stack.append((a, a + L0))
    return dirs

def line_interval(h, v):
    lo, hi = -1e15, 1e15
    for i in range(v.shape[0]):
        vi = v[i]
        if vi > 1e-14:
            lo = max(lo, -h[i] / vi)
            hi = min(hi, (1.0 - h[i]) / vi)
        elif vi < -1e-14:
            lo = max(lo, (1.0 - h[i]) / vi)
            hi = min(hi, -h[i] / vi)
    if not np.isfinite(lo) or not np.isfinite(hi) or hi - lo < 1e-14:
        return None
    return (lo, hi)

def line_coefficients(h, v, n):
    dx = 2.0 / n
    L = 2 * n - 1
    c = dx * np.correlate(h, 1.0 - h, mode='full')
    b = dx * (np.correlate(v, 1.0 - h, mode='full') - np.correlate(h, v, mode='full'))
    a = -dx * np.correlate(v, v, mode='full')
    for arr in (a, b, c):
        if arr.shape[0] != L or not np.all(np.isfinite(arr)):
            return None
    return a, b, c

def append_roots(A, B, C, lo, hi, out):
    scale = max(abs(A), abs(B), abs(C), 1e-300)
    tol = 1e-14 * scale
    if abs(A) <= tol:
        if abs(B) > tol:
            r = -C / B
            if lo - 1e-12 <= r <= hi + 1e-12:
                out.append(float(np.clip(r, lo, hi)))
        return
    disc = B * B - 4.0 * A * C
    if disc < -1e-13 * (B * B + abs(4.0 * A * C) + 1e-300):
        return
    disc = max(disc, 0.0)
    s = math.sqrt(disc)
    q = -0.5 * (B + math.copysign(s, B if B != 0 else 1.0))
    if abs(q) > 1e-300:
        roots = [q / A, C / q]
    elif abs(B) > 1e-300:
        roots = [-B / (2.0 * A)]
    else:
        roots = [0.0]
    for r in roots:
        if np.isfinite(r) and lo - 1e-12 <= r <= hi + 1e-12:
            out.append(float(np.clip(r, lo, hi)))

def line_candidates(a, b, c, lo, hi):
    cands = [lo, hi, 0.0]
    L = len(a)
    for k in range(L):
        if abs(a[k]) > 1e-300:
            t = -b[k] / (2.0 * a[k])
            if np.isfinite(t) and lo - 1e-12 <= t <= hi + 1e-12:
                cands.append(float(np.clip(t, lo, hi)))
    for i in range(L):
        for j in range(i + 1, L):
            DA, DB, DC = a[i] - a[j], b[i] - b[j], c[i] - c[j]
            if abs(DA) + abs(DB) + abs(DC) < 1e-300:
                continue
            append_roots(DA, DB, DC, lo, hi, cands)
    cands.sort()
    filtered = [cands[0]]
    for t in cands[1:]:
        if t - filtered[-1] > 1e-12 * max(1.0, abs(t)):
            filtered.append(t)
    return filtered

def line_search(h, cur_score, v, n):
    iv = line_interval(h, v)
    if iv is None:
        return h.copy(), cur_score
    lo, hi = iv
    coeffs = line_coefficients(h, v, n)
    if coeffs is None:
        return h.copy(), cur_score
    a, b, c = coeffs
    cands = line_candidates(a, b, c, lo, hi)
    if len(cands) < 2:
        return h.copy(), cur_score
    tc = np.array(cands, dtype=np.float64)
    env = np.max(a[None, :] * tc[:, None] ** 2 + b[None, :] * tc[:, None] + c[None, :], axis=1)
    order = np.argsort(env)
    top_idx = order[:10]
    forced = []
    for t in (lo, hi, 0.0):
        idx = int(np.argmin(np.abs(tc - t)))
        if idx not in top_idx:
            forced.append(idx)
    check_idx = list(top_idx) + forced
    best_h, best_s = h.copy(), cur_score
    for idx in check_idx:
        t = tc[idx]
        z = project_capped(h + t * v, n)
        if z is None or not is_feasible(z, n):
            continue
        s = exact_score(z, n)
        if s < best_s - 1e-12:
            best_s = s
            best_h = z
    return best_h, best_s

def refine_chain(start, n, directions, line_limit, best_h, best_score):
    current = start.copy()
    cur = exact_score(current, n)
    if not np.isfinite(cur):
        return best_h, best_score
    if cur < best_score - 1e-12:
        best_h, best_score = current.copy(), cur
    lines_used, stagnant, fwd = 0, 0, True
    nd = len(directions)
    while lines_used < line_limit and stagnant < 2:
        improved = False
        rng = range(nd) if fwd else range(nd - 1, -1, -1)
        for di in rng:
            if lines_used >= line_limit:
                stagnant += 1
                break
            lines_used += 1
            try:
                new_h, new_s = line_search(current, cur, directions[di], n)
            except Exception:
                continue
            if new_s < cur - 1e-12:
                current, cur = new_h, new_s
                improved = True
                if cur < best_score - 1e-12:
                    best_h, best_score = current.copy(), cur
        fwd = not fwd
        stagnant = 0 if improved else stagnant + 1
    return best_h, best_score

def pool_add(pool, z, n):
    pz = project_capped(z, n)
    if pz is None:
        return
    for s in pool:
        if np.max(np.abs(pz - s)) < 1e-8:
            return
    pool.append(pz)

def build_starts(fallback, n, rng):
    pool = [fallback.copy()]
    pool_add(pool, fallback[::-1], n)
    pool_add(pool, 1.0 - fallback, n)
    pool_add(pool, 1.0 - fallback[::-1], n)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    pool_add(pool, blk, n)
    pool_add(pool, np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64), n)
    t = np.linspace(0, 1, n)
    pool_add(pool, 0.5 + 0.35 * np.cos(2.0 * np.pi * t), n)
    pool_add(pool, 0.5 + 0.35 * np.cos(4.0 * np.pi * t), n)
    pool_add(pool, fallback + rng.normal(0, 0.05, n), n)
    pool_add(pool, fallback + rng.normal(0, 0.15, n), n)
    return pool[:10]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0 and p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None:
        n = 47
        try:
            v = int(kwargs.get('n_points', 47))
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if parent is not None and parent.shape[0] == n:
        proj = project_capped(parent, n)
        fallback = proj if proj is not None else np.full(n, 0.5)
    else:
        fallback = np.full(n, 0.5)
    if not is_feasible(fallback, n):
        fallback = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        return fallback, float(exact_score(fallback, n)), 1
    if n > 64:
        return fallback, float(exact_score(fallback, n)), n
    starts = build_starts(fallback, n, rng)
    scored = []
    for s in starts:
        scored.append((exact_score(s, n), s))
    scored.sort(key=lambda x: x[0])
    selected = [s for _, s in scored[:3]]
    found_fb = False
    for s in selected:
        if np.max(np.abs(s - fallback)) < 1e-8:
            found_fb = True
            break
    if not found_fb:
        selected[0] = fallback.copy()
    fb_score = exact_score(fallback, n)
    best_h, best_score = fallback.copy(), fb_score
    B = min(276, max(n - 1, int(276 * (47.0 / n) ** 2)))
    n_chains = min(len(selected), 3, max(1, B // (n - 1)))
    per_chain = B // n_chains
    directions = build_directions(n)
    for i in range(n_chains):
        budget = per_chain + (1 if i < B % n_chains else 0)
        best_h, best_score = refine_chain(selected[i], n, directions, budget, best_h, best_score)
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score + 1e-12:
        final, fs = fallback, fb_score
    return final.astype(np.float64), float(fs), int(n)
