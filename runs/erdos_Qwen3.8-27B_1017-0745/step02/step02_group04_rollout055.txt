            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(np.sum(h) - s) > 1e-9:
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-9:
        return False
    return True

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c * (2.0 / n)))

def lag_qj(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.zeros(L)
    J = np.zeros((L, n))
    for d in range(-(n - 1), n):
        p0, q0 = max(d, 0), max(-d, 0)
        m = n - abs(d)
        q[d + n - 1] = dx * np.dot(h[p0:p0+m], 1.0 - h[q0:q0+m])
        J[d + n - 1, p0:p0+m] += dx * (1.0 - h[q0:q0+m])
        J[d + n - 1, q0:q0+m] -= dx * h[p0:p0+m]
    return q, J

def _obj(z):
    return z[_ctx['n']]

def _obj_jac(z):
    J = np.zeros(z.shape)
    J[_ctx['n']] = 1.0
    return J

def _ineq(z):
    ny = _ctx['n']
    q, _ = lag_qj(z[:ny], ny)
    return z[ny] - q

def _ineq_jac(z):
    ny = _ctx['n']
    _, J = lag_qj(z[:ny], ny)
    out = np.zeros((2 * ny - 1, ny + 1))
    out[:, :ny] = -J
    out[:, ny] = 1.0
    return out

def _eq(z):
    return np.array([np.sum(z[:_ctx['n']]) - _ctx['s']])

def _eq_jac(z):
    ny = _ctx['n']
    J = np.zeros((1, ny + 1))
    J[0, :ny] = 1.0
    return J

def solve_epigraph(x, n, maxiter):
    _ctx['n'] = n
    _ctx['s'] = n / 2.0
    q, _ = lag_qj(x, n)
    z0 = np.empty(n + 1)
    z0[:n] = x
    z0[n] = np.max(q) + 1e-6
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constr = [
        {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac},
        {'type': 'eq', 'fun': _eq, 'jac': _eq_jac},
    ]
    try:
        res = minimize(_obj, z0, jac=_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=constr,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
    except Exception:
        return None
    y = res.x[:n] if res.x is not None else None
    if y is None or y.shape != (n,) or not np.all(np.isfinite(y)):
        return None
    return project_capped(y, n)

def _try_add(pool, v, n):
    if v is None or v.shape != (n,):
        return
    if not np.all(np.isfinite(v)) or np.any(v < -1e-6) or np.any(v > 1.0 + 1e-6):
        return
    for s in pool:
        if np.max(np.abs(v - s)) < 1e-8:
            return
    pool.append(v.copy())

def build_starts(fb, n, rng, count):
    pool = []
    _try_add(pool, fb, n)
    _try_add(pool, project_capped(fb[::-1].copy(), n), n)
    _try_add(pool, project_capped(1.0 - fb, n), n)
    _try_add(pool, project_capped((1.0 - fb)[::-1].copy(), n), n)
    _try_add(pool, np.full(n, 0.5), n)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    _try_add(pool, project_capped(blk, n), n)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    if n % 2:
        alt[n - 1] = 0.5
    _try_add(pool, project_capped(alt, n), n)
    P = max(2, n // 3)
    cos_v = 0.5 + 0.45 * np.cos(2.0 * np.pi * np.arange(n) / P)
    _try_add(pool, project_capped(cos_v, n), n)
    _try_add(pool, project_capped(fb + rng.normal(0, 0.05, n), n), n)
    _try_add(pool, project_capped(fb + rng.normal(0, 0.15, n), n), n)
    _try_add(pool, project_capped(fb + rng.normal(0, 0.25, n), n), n)
    _try_add(pool, project_capped(rng.uniform(0, 1, n), n), n)
    sm = 0.5 + 0.3 * np.sin(4.0 * np.pi * np.arange(n) / n)
    _try_add(pool, project_capped(sm, n), n)
    return pool[:count]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        n = 47
    if parent is not None:
        fb = project_capped(parent, n)
    else:
        fb = np.full(n, 0.5)
    if fb is None:
        fb = np.full(n, 0.5)
    fb = np.asarray(fb, dtype=np.float64)
    if n == 1:
        return fb, float(exact_score(fb, n)), n
    rng = np.random.default_rng(int(seed))
    if n <= 64:
        n_starts, maxiter, n_hops = 12, 250, 10
    elif n <= 128:
        n_starts, maxiter, n_hops = 8, 180, 5
    elif n <= 256:
        n_starts, maxiter, n_hops = 5, 100, 3
    else:
        n_starts, maxiter, n_hops = 3, 60, 1
    starts = build_starts(fb, n, rng, n_starts)
    gb = fb.copy()
    gs = float(exact_score(fb, n))
    for st in starts:
        s0 = float(exact_score(st, n))
        if s0 < gs - 1e-12:
            gs, gb = s0, st.copy()
        y = solve_epigraph(st, n, maxiter)
        if y is not None and is_feasible(y, n):
            sy = float(exact_score(y, n))
            if sy < gs - 1e-12:
                gs, gb = sy, y.copy()
            for k in range(5):
                alpha = 1.0 / (2.0 ** k)
                z = project_capped(st + alpha * (y - st), n)
                if z is None or not is_feasible(z, n):
                    continue
                sz = float(exact_score(z, n))
                if sz < gs - 1e-12:
                    gs, gb = sz, z.copy()
    curr = gb.copy()
    curr_s = gs
    sig_cyc = [0.04, 0.10, 0.20]
    for r in range(n_hops):
        if (r + 1) % 4 == 0:
            prop = project_capped(rng.uniform(0, 1, n), n)
        else:
            sig = sig_cyc[r % 3]
            prop = project_capped(curr + sig * rng.standard_normal(n), n)
        if prop is None:
            continue
        y = solve_epigraph(prop, n, maxiter)
        cand = y if (y is not None and is_feasible(y, n)) else prop
        if not is_feasible(cand, n):
            continue
        sc = float(exact_score(cand, n))
        if sc == float('inf'):
            continue
        if sc < gs - 1e-12:
            gs, gb = sc, cand.copy()
            curr, curr_s = cand.copy(), sc
            continue
        T = max(1e-5, 1.5e-3 * (0.8 ** r))
        if sc <= curr_s or rng.random() < np.exp(-(sc - curr_s) / T):
            curr, curr_s = cand.copy(), sc
    gb = project_capped(gb, n)
    if gb is None or not is_feasible(gb, n):
        gb = fb.copy()
    fs = float(exact_score(gb, n))
    fb_s = float(exact_score(fb, n))
    if fs > fb_s + 1e-12:
        gb, fs = fb.copy(), fb_s
    if fs == float('inf'):
        gb = np.full(n, 0.5)
        fs = float(exact_score(gb, n))
    return gb, fs, n
