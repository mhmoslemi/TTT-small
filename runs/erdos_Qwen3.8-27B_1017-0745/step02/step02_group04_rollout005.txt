    z0 = np.empty(n + 1)
    z0[:n] = x
    z0[n] = t0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    constr = [
        {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac},
        {'type': 'eq', 'fun': _eq, 'jac': _eq_jac},
    ]
    try:
        res = minimize(_obj, z0, jac=_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=constr,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
    except Exception:
        return None
    if res.x is None or res.x.shape != (n + 1,):
        return None
    y = res.x[:n]
    if not np.all(np.isfinite(y)):
        return None
    return project_capped(y, n)

def _add_start(pool, v, n):
    if v is None or v.shape != (n,):
        return
    if not np.all(np.isfinite(v)) or np.any(v < -1e-6) or np.any(v > 1.0 + 1e-6):
        return
    for s in pool:
        if np.max(np.abs(v - s)) < 1e-8:
            return
    pool.append(v.copy())

def build_starts(fallback, n, rng, count):
    pool = []
    _add_start(pool, fallback, n)
    _add_start(pool, project_capped(fallback[::-1].copy(), n), n)
    _add_start(pool, project_capped(1.0 - fallback, n), n)
    _add_start(pool, project_capped((1.0 - fallback)[::-1].copy(), n), n)
    _add_start(pool, np.full(n, 0.5), n)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    _add_start(pool, project_capped(blk, n), n)
    _add_start(pool, project_capped(fallback + 0.05 * rng.standard_normal(n), n), n)
    _add_start(pool, project_capped(fallback + 0.15 * rng.standard_normal(n), n), n)
    _add_start(pool, project_capped(fallback + 0.30 * rng.standard_normal(n), n), n)
    _add_start(pool, project_capped(rng.uniform(0, 1, n), n), n)
    P = max(2, n // 3)
    ii = np.arange(n, dtype=np.float64)
    _add_start(pool, project_capped(0.5 + 0.45 * np.cos(2.0 * np.pi * ii / P), n), n)
    tb = np.zeros(n)
    tb[:n // 4] = 1.0
    tb[3 * n // 4:] = 1.0
    _add_start(pool, project_capped(tb, n), n)
    tri = 1.0 - np.abs(np.arange(n, dtype=np.float64) - n / 2.0) / max(1.0, n / 2.0)
    _add_start(pool, project_capped(tri, n), n)
    _add_start(pool, project_capped(fallback + 0.08 * rng.standard_normal(n), n), n)
    return pool[:count]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None:
        if 'n_points' in kwargs:
            try:
                v = int(kwargs['n_points'])
                if 1 <= v <= 999:
                    n = v
            except Exception:
                pass
        if n is None:
            n = 47
    if n == 1:
        h = np.array([0.5])
        return h, float(exact_score(h, 1)), 1
    if parent is not None:
        fallback = project_capped(parent, n)
    else:
        fallback = np.full(n, 0.5)
    if fallback is None:
        fallback = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 64:
        n_starts, maxiter, n_hops = 12, 300, 10
    elif n <= 128:
        n_starts, maxiter, n_hops = 7, 180, 5
    elif n <= 256:
        n_starts, maxiter, n_hops = 4, 100, 2
    else:
        n_starts, maxiter, n_hops = 2, 50, 0
    starts = build_starts(fallback, n, rng, n_starts)
    gb = fallback.copy()
    gs = float(exact_score(fallback, n))
    for st in starts:
        s = float(exact_score(st, n))
        if s < gs - 1e-12:
            gs, gb = s, st.copy()
        y = solve_direct(st, n, maxiter)
        if y is not None:
            sy = float(exact_score(y, n))
            if sy < gs - 1e-12:
                gs, gb = sy, y.copy()
    sigmas = [0.04, 0.10, 0.20]
    current = gb.copy()
    cur_score = gs
    T0 = 1.5e-3
    for r in range(n_hops):
        if r % 4 == 3:
            prop = project_capped(rng.uniform(0, 1, n), n)
        else:
            sig = sigmas[r % 3]
            prop = project_capped(current + sig * rng.standard_normal(n), n)
        if prop is None:
            continue
        y = solve_direct(prop, n, maxiter)
        if y is None:
            y = prop
        sc = float(exact_score(y, n))
        if sc < gs - 1e-12:
            gs, gb = sc, y.copy()
        if sc <= cur_score:
            current = y.copy()
            cur_score = sc
        else:
            Tr = max(1e-5, T0 * (0.8 ** r))
            if rng.random() < np.exp(-(sc - cur_score) / Tr):
                current = y.copy()
                cur_score = sc
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fallback.copy()
    fs = float(exact_score(final, n))
    fb = float(exact_score(fallback, n))
    if fs > fb + 1e-12:
        final, fs = fallback.copy(), fb
    if fs == float('inf'):
        final = np.full(n, 0.5)
        fs = float(exact_score(final, n))
    return final, fs, int(n)
