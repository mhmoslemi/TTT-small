```python
import numpy as np

def project_capped(z, n):
    z = np.where(np.isfinite(z), z, 0.0).astype(np.float64)
    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx])
            h[idx] += d; r -= d
        else:
            d = min(-r, h[idx])
            h[idx] -= d; r += d
    if abs(r) > 1e-10 * max(1, n) or not np.all(np.isfinite(h)):
        return None
    return np.clip(h, 0.0, 1.0)

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def helmert_fwd(x, n):
    d = n - 1
    if d == 0:
        return np.zeros(n)
    denom = np.sqrt(np.arange(1, d + 1, dtype=np.float64) * np.arange(2, d + 2, dtype=np.float64))
    a = x / denom
    T = np.zeros(d + 1)
    T[:d] = np.cumsum(a[::-1])[::-1]
    out = np.empty(n, dtype=np.float64)
    out[0] = T[0]
    for i in range(1, n):
        ti = T[i] if i <= d else 0.0
        corr = i * a[i - 1] if i - 1 < d else 0.0
        out[i] = ti - corr
    return out

def helmert_bwd(v, n):
    d = n - 1
    if d == 0:
        return np.zeros(0)
    prefix = np.cumsum(v)
    denom = np.sqrt(np.arange(1, d + 1, dtype=np.float64) * np.arange(2, d + 2, dtype=np.float64))
    out = (prefix[:d] - (np.arange(1, d + 1, dtype=np.float64)) * v[1:]) / denom
    return out
```

```python
def cem_run(h0, n, rng, pop, gens, stall_lim, cov_mode, gb_v, gb_s):
    d = n - 1
    u = np.full(n, 0.5, dtype=np.float64)
    m = helmert_bwd(h0 - u, n)
    sigma0 = 0.15
    C = sigma0**2 * np.eye(d) if cov_mode == 'full' else sigma0**2 * np.ones(d)
    eta = 0.35
    smax, smin = 0.5, 0.02
    mu = pop // 2
    half_pop = pop // 2
    stall = 0
    run_best_s = gb_s
    for gen in range(gens):
        # Factor C
        if cov_mode == 'full':
            try:
                eigvals, eigvecs = np.linalg.eigh(C)
                eigvals = np.clip(eigvals, smin*smin, smax*smax)
                L = eigvecs @ np.diag(np.sqrt(np.abs(eigvals)))
            except Exception:
                C = sigma0**2 * np.eye(d)
                eigvals = np.full(d, sigma0**2)
                eigvecs = np.eye(d)
                L = sigma0 * np.eye(d)
        else:
            L_diag = np.sqrt(np.clip(C, smin*smin, smax*smax))
        
        # Sample
        cands_h = []
        cands_x = []
        cands_s = []
        for _ in range(half_pop):
            z = rng.standard_normal(d)
            for sign in (1, -1):
                if cov_mode == 'full':
                    x_prop = m + sign * (L @ z)
                else:
                    x_prop = m + sign * (L_diag * z)
                h_prop = project_capped(u + helmert_fwd(x_prop, n), n)
                if h_prop is None:
                    continue
                s = exact_score(h_prop, n)
                if np.isfinite(s):
                    x_proj = helmert_bwd(h_prop - u, n)
                    cands_h.append(h_prop)
                    cands_x.append(x_proj)
                    cands_s.append(s)
                    if s < gb_s - 1e-12:
                        gb_s = s
                        gb_v = h_prop.copy()
        
        if len(cands_h) == 0:
            break
        
        # Sort by score
        order = np.argsort(cands_s)
        ne = min(mu, len(cands_h))
        elite_idx = order[:ne]
        
        # Compute elite statistics
        w = np.zeros(ne)
        for r in range(ne):
            w[r] = np.log(ne + 0.5) - np.log(r + 1)
        w /= w.sum()
        
        m_elite = np.zeros(d)
        for r in range(ne):
            m_elite += w[r] * cands_x[elite_idx[r]]
        
        m_new = (1 - eta) * m + eta * m_elite
        
        # Project m_new back
        h_proj = project_capped(u + helmert_fwd(m_new, n), n)
        if h_proj is not None:
            m_new = helmert_bwd(h_proj - u, n)
        else:
            break
        
        # Update covariance
        if cov_mode == 'full':
            S = np.zeros((d, d))
            for r in range(ne):
                diff = cands_x[elite_idx[r]] - m_new
                S += w[r] * np.outer(diff, diff)
            C_new = (1 - eta) * C + eta * S
            C_new = 0.5 * (C_new + C_new.T)
            if not np.all(np.isfinite(C_new)):
                break
            C = C_new
        else:
            S_diag = np.zeros(d)
            for r in range(ne):
                diff = cands_x[elite_idx[r]] - m_new
                S_diag += w[r] * diff**2
            v_new = (1 - eta) * C + eta * S_diag
            C = np.clip(v_new, smin*smin, smax*smax)
        
        m = m_new
        
        # Stall check
        gen_best = min(cands_s)
        if gen_best < run_best_s - 1e-12:
            run_best_s = gen_best
            stall = 0
        else:
            stall += 1
            if stall >= stall_lim:
                break
    return gb_v, gb_s
```

```python
def make_starts(parent, n, rng, max_starts):
    pool = []
    def add(h, sigma):
        p = project_capped(h, n)
        if p is None:
            return
        for existing in pool:
            if np.max(np.abs(p - existing[0])) < 1e-7:
                return
        pool.append((p, sigma))
    
    add(np.full(n, 0.5), 0.18)
    if parent is not None:
        add(parent, 0.10)
        add(parent[::-1], 0.10)
        add(1.0 - parent, 0.10)
        add(1.0 - parent[::-1], 0.10)
    blk = np.zeros(n); blk[:n//2] = 1.0
    if n % 2: blk[n//2] = 0.5
    add(blk, 0.28)
    add(np.where(np.arange(n) % 2 == 0, 1.0, 0.0), 0.28)
    perm = rng.permutation(n)
    bb = np.zeros(n); bb[perm[:n//2]] = 1.0
    if n % 2: bb[perm[n//2]] = 0.5
    add(bb, 0.28)
    base = parent if parent is not None else np.full(n, 0.5)
    add(base + rng.normal(0, 0.08, n), 0.12)
    add(base + rng.normal(0, 0.2, n), 0.12)
    add(rng.uniform(0, 1, n), 0.20)
    
    # Score and select
    scored = []
    for p, sig in pool:
        s = exact_score(p, n)
        scored.append((s, p, sig))
    scored.sort()
    return scored[:max_starts]
```

```python
def project_capped(z, n):

    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0

            lo = mid
        else:

def helmert_fwd(x, n):
    d = n - 1
    if d == 0:
        return np.zeros(n)
    j = np.arange(1, d + 1, dtype=np.float64)
    denom = np.sqrt(j * (j + 1))
    a = x / denom
    T = np.zeros(d + 1)
    T[:d] = np.cumsum(a[::-1])[::-1]
    out = np.empty(n, dtype=np.float64)
    out[0] = T[0]
    for i in range(1, n):
        ti = T[i] if i <= d else 0.0
        corr = i * a[i - 1] if i - 1 < d else 0.0
        out[i] = ti - corr
    return out

def helmert_bwd(v, n):
    d = n - 1
    if d == 0:
        return np.zeros(0)
    j = np.arange(1, d + 1, dtype=np.float64)
    denom = np.sqrt(j * (j + 1))
    prefix = np.cumsum(v)
    return (prefix[:d] - j * v[1:]) / denom
```
