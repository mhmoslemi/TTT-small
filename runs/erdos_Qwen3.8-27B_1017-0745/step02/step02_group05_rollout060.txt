                    inc_s = rs
            pop = np_
            scores = ns_
            pop[0] = inc.copy()
            scores[0] = inc_s
            stall = 0
    return inc, inc_s

def pa_anneal(x, n, L, rng):
    cur = x.copy()
    cs = exact_score(cur, n)
    best = cur.copy()
    bs = cs
    for t in range(L):
        q = t / max(1, L - 1)
        T = 8e-4 * (1e-7 / 8e-4) ** q
        rho = 1.0 - 0.97 * q
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        cap1 = min(1.0 - cur[i], cur[j])
        cap2 = min(1.0 - cur[j], cur[i])
        if cap1 >= cap2:
            cap = cap1
        else:
            i, j = j, i
            cap = cap2
        if cap < 1e-14:
            continue
        amt = cap * rho * (0.03 + 0.97 * rng.random())
        new = cur.copy()
        new[i] += amt
        new[j] -= amt
        np_ = project_capped(new, n)
        if np_ is None:
            continue
        ns = exact_score(np_, n)
        if not np.isfinite(ns):
            continue
        d = ns - cs
        if d < -1e-12:
            acc = True
        elif abs(d) <= 1e-12:
            acc = rng.random() < 0.1
        else:
            acc = rng.random() < math.exp(-max(0.0, d) / max(T, 1e-15))
        if acc:
            cur = np_
            cs = ns
        if cs < bs - 1e-12:
            bs = cs
            best = cur.copy()
    return best, bs

def pair_polish(x, n, rng, nsw, npairs):
    x = x.copy()
    s = exact_score(x, n)
    fracs = (1.0 / 128, 1.0 / 64, 1.0 / 32, 1.0 / 16, 1.0 / 8, 1.0 / 4, 1.0 / 2, 1.0)
    idx = np.arange(n)
    for sw in range(nsw):
        imp = False
        if n <= 64:
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        else:
            pairs = []
            for _ in range(npairs):
                i, j = rng.choice(n, size=2, replace=False)
                if i > j:
                    i, j = j, i
                pairs.append((int(i), int(j)))
        for (i, j) in pairs:
            for (a, b) in ((i, j), (j, i)):
                cap = min(1.0 - x[a], x[b])
                if cap < 1e-14:
                    continue
                for fr in fracs:
                    t = cap * fr
                    new = x.copy()
                    new[a] += t
                    new[b] -= t
                    cp = project_capped(new, n)
                    if cp is None:
                        continue
                    ns = exact_score(cp, n)
                    if ns < s - 1e-12:
                        x = cp
                        s = ns
                        imp = True
        if not imp:
            break
    return x, s

def coord_anneal(x, n, L, rng):
    cur = x.copy()
    cs = exact_score(cur, n)
    best = cur.copy()
    bs = cs
    for t in range(L):
        q = t / max(1, L - 1)
        T = 5e-4 * (1e-7 / 5e-4) ** q
        i = int(rng.integers(n))
        delta = rng.normal(0, 0.02 + 0.04 * (1 - q))
        new = cur.copy()
        new[i] += delta
        np_ = project_capped(new, n)
        if np_ is None:
            continue
        ns = exact_score(np_, n)
        if not np.isfinite(ns):
            continue
        d = ns - cs
        if d < -1e-12:
            acc = True
        elif abs(d) <= 1e-12:
            acc = rng.random() < 0.1
        else:
            acc = rng.random() < math.exp(-max(0.0, d) / max(T, 1e-15))
        if acc:
            cur = np_
            cs = ns
        if cs < bs - 1e-12:
            bs = cs
            best = cur.copy()
    return best, bs

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        if 'n_points' in kwargs:
            try:
                n = int(kwargs['n_points'])
                if not (1 <= n <= 999):
                    n = 47
            except Exception:
                n = 47
        else:
            n = 47
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        h1 = np.array([0.5], dtype=np.float64)
        return h1, exact_score(h1, 1), 1
    if parent is not None:
        base = project_capped(parent, n)
        if base is None or not is_feasible(base, n):
            base = np.full(n, 0.5)
    else:
        base = np.full(n, 0.5)
    safe_h = base.copy()
    safe_s = exact_score(safe_h, n)
    if n <= 64:
        P, G, L, Ca, stall, nsw, npa = 14, 150, 2500, 800, 18, 4, 999
    elif n <= 256:
        P, G, L, Ca, stall, nsw, npa = 8, 80, 600, 200, 12, 2, 160
    else:
        P, G, L, Ca, stall, nsw, npa = 6, 30, 100, 50, 7, 1, 40
    pop = build_pop(base, n, rng, P)
    scs = [exact_score(p, n) for p in pop]
    for i in range(len(pop)):
        if scs[i] < safe_s - 1e-12:
            safe_s = scs[i]
            safe_h = pop[i].copy()
    try:
        inc, is_ = de_search(pop, scs, n, G, rng, stall)
        if is_ < safe_s - 1e-12:
            safe_s = is_
            safe_h = inc.copy()
    except Exception:
        pass
    try:
        ah, as_ = pa_anneal(safe_h, n, L, rng)
        if as_ < safe_s - 1e-12:
            safe_s = as_
            safe_h = ah.copy()
    except Exception:
        pass
    try:
        ch, cs_ = coord_anneal(safe_h, n, Ca, rng)
        if cs_ < safe_s - 1e-12:
            safe_s = cs_
            safe_h = ch.copy()
    except Exception:
        pass
    try:
        ph, ps = pair_polish(safe_h, n, rng, nsw, max(npa, n * (n - 1) // 2 if npa > 900 else npa))
        if ps < safe_s - 1e-12:
            safe_s = ps
            safe_h = ph.copy()
    except Exception:
        pass
    try:
        pop2 = build_pop(safe_h, n, rng, P)
        scs2 = [exact_score(p, n) for p in pop2]
        inc2, is2 = de_search(pop2, scs2, n, G // 2, rng, stall)
        if is2 < safe_s - 1e-12:
            safe_s = is2
            safe_h = inc2.copy()
        ah2, as2 = pa_anneal(safe_h, n, L // 2, rng)
        if as2 < safe_s - 1e-12:
            safe_s = as2
            safe_h = ah2.copy()
        ph2, ps2 = pair_polish(safe_h, n, rng, nsw, max(npa, n * (n - 1) // 2 if npa > 900 else npa))
        if ps2 < safe_s - 1e-12:
            safe_s = ps2
            safe_h = ph2.copy()
    except Exception:
        pass
    final = project_capped(safe_h, n)
    if final is None or not is_feasible(final, n):
        final = project_capped(base, n)
        if final is None:
            final = np.full(n, 0.5)
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > safe_s + 1e-9:
        final = project_capped(base, n)
        if final is None:
            final = np.full(n, 0.5)
        fs = exact_score(final, n)
    return final.astype(np.float64), float(fs), int(n)
