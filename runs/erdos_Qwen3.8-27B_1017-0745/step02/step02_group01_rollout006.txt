                sv, sg = soft_fg(current, n, tau)
                sy, sg2 = soft_fg(y, n, tau)
                if np.isfinite(sy) and sy <= sv + 1e-9:
                    current = y
                else:
                    current = project_capped(0.5 * (current + y), n)
                    if current is None:
                        current = y
            except Exception:
                current = y
        polished, ps = subgrad_polish(current, n, 25)
        best_h, best_s = consider(polished, n, best_h, best_s)
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    if fs > fs_score + 1e-12:
        final, fs = fallback, fs_score
    return final, float(fs), int(n)
