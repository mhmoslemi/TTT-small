import numpy as np

def project_capped(z, n):
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) < 1e-13:
            break
        if r > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - s) > 5e-10:
        return None
    return h

        return False
    if abs(np.sum(h) - n / 2.0) > 1e-9:
        return False
    return True

    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

    c = np.correlate(h, 1.0 - h, mode='full') * dx
    rmax = float(np.max(c))
    exponents = np.clip((c - rmax) / tau, -500.0, 500.0)
    w = np.exp(exponents)
    sw = float(np.sum(w))

        return float('inf'), None, rmax
    F = rmax + tau * np.log(sw)
    w = w / sw
    g = np.zeros(n)

        idx = d + n - 1
        w_d = w[idx]
        if w_d < 1e-14:
            continue
        p0, q0 = max(d, 0), max(-d, 0)
        m = n - abs(d)
        g[p0:p0 + m] += w_d * dx * (1.0 - h[q0:q0 + m])
        g[q0:q0 + m] -= w_d * dx * h[p0:p0 + m]
    if not np.all(np.isfinite(g)):

    return F, g, rmax

        seen = set(pairs)
        while len(pairs) < cap:
            a, b = int(rng.integers(0, n)), int(rng.integers(0, n))

            if pr not in seen:
                pairs.append(pr)
                seen.add(pr)
    if n <= 64:
        meshes = [0.04, 0.01, 0.0025, 0.0006, 0.00015]
    elif n <= 256:
        meshes = [0.04, 0.01, 0.002, 0.0004]
    else:
        meshes = [0.04, 0.008, 0.0016]

        best_s = s
        best_x = None

            cap1 = min(dm, 1.0 - x[i], x[j])

                t = x.copy()

                t[j] -= cap1

                        best_x = t

                        best_x = t

            x = best_x
            s = best_s
