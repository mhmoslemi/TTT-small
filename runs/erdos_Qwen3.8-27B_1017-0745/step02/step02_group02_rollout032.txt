def project_capped(z, n):
    if z is None or len(z) != n:
        return None
    z = np.asarray(z, dtype=np.float64)
    if not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s_t = n / 2.0
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s_t:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s_t - float(np.sum(h))
    for i in range(n):
        if abs(r) < 5e-13 * max(1, n):
            break
        if r > 0:

        else:
            a = min(-r, h[i]); h[i] -= a; r += a
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    return h

def is_feasible(h, n):
    if h is None or len(h) != n or not np.all(np.isfinite(h)):
        return False

        return False
    return abs(float(np.sum(h)) - n / 2.0) <= 5e-10 * max(1, n)

def exact_score(h, n):
