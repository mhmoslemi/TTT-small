def refine_subgrad(h, n, n_iter):
    h = h.copy().astype(np.float64)
    best = h.copy()
    best_s = exact_score(h, n)
    dx = 2.0 / n
    eta = 0.15 / dx
    for _ in range(n_iter):
        q = lag_vals(h, n)
        idx = int(np.argmax(q))
        d = idx - (n - 1)
        p0 = max(d, 0)
        q0 = max(-d, 0)
        m = n - abs(d)
        g = np.zeros(n)
        g[p0:p0 + m] += dx * (1.0 - h[q0:q0 + m])
        g[q0:q0 + m] -= dx * h[p0:p0 + m]
        z = project_capped(h - eta * g, n)
        if z is None:
            eta *= 0.5
            if eta < 1e-8 / dx:
                break
            continue
        v = z - h
        if np.max(np.abs(v)) < 1e-13:
            eta *= 0.5
            if eta < 1e-8 / dx:
                break
            continue
        improved = False
        for alpha in (1.0, 0.5, 0.25, 0.125, 0.0625):
            trial = project_capped(h + alpha * v, n)
            if trial is None:
                continue
            sz = exact_score(trial, n)
            if sz < best_s - 1e-13:
                best_s = sz
                best = trial.copy()
                h = trial
                improved = True
                break
        if not improved:
            eta *= 0.5
            if eta < 1e-8 / dx:
                break
        else:
            eta = min(1.1 * eta, 2.0 / dx)
    return best, best_s

def build_starts(fb, n, rng, count):
    pool = [fb.copy()]
    j = np.arange(n)
    cands = [fb[::-1].copy(), 1.0 - fb[::-1].copy(), np.full(n, 0.5)]
    cands.append(np.clip(0.5 + 0.45 * np.cos(2.0 * np.pi * (j + 0.5) / n), 0, 1))
    cands.append(np.clip(0.5 + 0.45 * np.cos(2.0 * np.pi * (j + 0.5) / n + np.pi / 2), 0, 1))
    cands.append(np.clip(0.5 + 0.35 * np.cos(4.0 * np.pi * (j + 0.5) / n + 0.3), 0, 1))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    cands.append(blk)
    cands.append(rng.uniform(0, 1, n))
    cands.append(fb + rng.normal(0, 0.06, n))
    cands.append(fb + rng.normal(0, 0.12, n))
    for c in cands:
        if len(pool) >= count:
            break
        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for s2 in pool:
            if np.max(np.abs(p - s2)) < 1e-8:
                dup = True
                break
        if not dup:
            pool.append(p)
    return pool[:count]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size >= 1 and p.size <= 999:
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
                n = int(p.size)
    except Exception:
        pass
    if n is None:
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if not (1 <= n <= 999):
        n = 47
    if n == 1:
        return np.array([0.5], dtype=np.float64), 0.0, 1
    if parent is not None and parent.shape[0] == n:
        fb = project_capped(parent, n)
    else:
        fb = None
    if fb is None:
        fb = np.full(n, 0.5)
    fs = exact_score(fb, n)
    gb, gs = fb.copy(), fs
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 64:
        tau_list = [0.02, 0.005, 0.001, 0.0003, 0.0001, 0.00003]
        max_it = 150
        n_st = 6
        n_re = 3
        sg_it = 60
    elif n <= 128:
        tau_list = [0.02, 0.005, 0.001, 0.0003, 0.0001]
        max_it = 100
        n_st = 4
        n_re = 2
        sg_it = 40
    elif n <= 256:
        tau_list = [0.015, 0.004, 0.001, 0.0003]
        max_it = 60
        n_st = 3
        n_re = 1
        sg_it = 25
    else:
        tau_list = [0.01, 0.003, 0.001]
        max_it = 30
        n_st = 2
        n_re = 0
        sg_it = 15
    starts = build_starts(fb, n, rng, n_st)
    for st in starts:
        try:
            rh, rs = refine_smooth(st, n, tau_list, max_it)
            if rs < gs - 1e-12:
                gs = rs
                gb = rh.copy()
            rh2, rs2 = refine_subgrad(gb, n, sg_it)
            if rs2 < gs - 1e-12:
                gs = rs2
                gb = rh2.copy()
        except Exception:
            continue
    if n_re > 0:
        improved = (gs < fs - 1e-5)
        if n <= 64:
            sds = [0.03, 0.08, 0.15] if improved else [0.08, 0.18, 0.30]
        else:
            sds = [0.05, 0.12] if improved else [0.12, 0.25]
        for ri in range(n_re):
            try:
                sd = sds[ri] if ri < len(sds) else sds[-1]
                pert = project_capped(gb + rng.normal(0, sd, n), n)
                if pert is None:
                    continue
                rh, rs = refine_smooth(pert, n, tau_list, max_it)
                if rs < gs - 1e-12:
                    gs = rs
                    gb = rh.copy()
                rh2, rs2 = refine_subgrad(gb, n, sg_it)
                if rs2 < gs - 1e-12:
                    gs = rs2
                    gb = rh2.copy()
            except Exception:
                continue
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb
    sf = exact_score(final, n)
    if sf > fs:
        final, sf = fb, fs
    if not is_feasible(final, n):
        final = np.full(n, 0.5)
        sf = exact_score(final, n)
    return final, float(sf), int(n)
