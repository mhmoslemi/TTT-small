import numpy as np

def project_capped(z, n):
    z = np.array(z, dtype=np.float64).ravel()
    if z.shape != (n,):
        return None
    if not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    target = 0.5 * n
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        v = np.clip(z - mid, 0.0, 1.0)
        if float(np.sum(v)) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - float(np.sum(h))
    ip = 0
    while abs(r) > 1e-14 and ip < n:
        if r > 0.0:
            cap = 1.0 - h[ip]
            if cap > 1e-15:
                add = min(r, cap)
                h[ip] += add
                r -= add
        else:
            cap = h[ip]
            if cap > 1e-15:
                sub = min(-r, cap)
                h[ip] -= sub
                r += sub
        ip += 1
    if not np.all(np.isfinite(h)):
        return None
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(float(np.sum(h)) - target) > 1e-9 * max(1, n):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(float(np.sum(h)) - 0.5 * n) > 1e-9 * max(1, n):
        return False
    return True

def fast_score(h, n):
    q = np.correlate(h, 1.0 - h, mode='full')
    if q.shape[0] != 2 * n - 1 or not np.all(np.isfinite(q)):
        return float('inf')
    return float(np.max(q) * (2.0 / n))

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    return fast_score(h, n)

def balanced_binary(n, rng):
    v = np.zeros(n, dtype=np.float64)
    k = int(math.floor(n / 2.0))
    perm = rng.permutation(n)
    if k > 0:
        v[perm[:k]] = 1.0
    if n % 2 == 1 and k < n:
        v[perm[k]] = 0.5
    return v

def add_feasible(pool, cand, n):
    if cand is None:
        return
    h = project_capped(cand, n)
    if h is None:
        return
    for p in pool:
        if np.max(np.abs(h - p)) < 1e-7:
            return
    pool.append(h)

def pair_coefficients(h, n, i, j):
    L = 2 * n - 1
    a = np.zeros(L, dtype=np.float64)
    b = np.zeros(L, dtype=np.float64)
    mid = n - 1
    si = mid - i
    sj = mid - j
    a[si:si + n] += 1.0 - h
    a[sj:sj + n] += -1.0 + h
    hi = float(h[i])
    hj = float(h[j])
    a[mid] += -hi + hj
    b[mid] += -2.0
    idx = mid + (j - i)
    a[idx] += hi
    b[idx] += 1.0
    idx2 = mid + (i - j)
    a[idx2] -= hj
    b[idx2] += 1.0
    return a, b

def attempt_pair_update(h, qbase, score, n, i, j, base_topo, grid):
    hi = float(h[i])
    hj = float(h[j])
    lo = max(-hi, hj - 1.0)
    hb = min(1.0 - hi, hj)
    if hb - lo <= 1e-14:
        return h, qbase, score, False
    a, b = pair_coefficients(h, n, i, j)
    tx = lo + (hb - lo) * grid
    t2 = tx * tx
    topo = np.max(qbase[None, :] + a[None, :] * tx[:, None] + b[None, :] * t2[:, None], axis=1)
    idx = int(np.argmin(topo))
    best_t = float(tx[idx])
    best_topo = float(topo[idx])
    if 0 < idx < len(grid) - 1:
        tl = float(tx[idx - 1])
        th = float(tx[idx + 1])
        if th - tl > 1e-14:
            txr = tl + (th - tl) * grid
            t2r = txr * txr
            topor = np.max(qbase[None, :] + a[None, :] * txr[:, None] + b[None, :] * t2r[:, None], axis=1)
            idxr = int(np.argmin(topor))
            best_t = float(txr[idxr])
            best_topo = float(topor[idxr])
    if best_topo < base_topo - 1e-10:
        vi = hi + best_t
        vj = hj - best_t
        if vi < 0.0:
            vi = 0.0
        elif vi > 1.0:
            vi = 1.0
        if vj < 0.0:
            vj = 0.0
        elif vj > 1.0:
            vj = 1.0
        cand = h.copy()
        cand[i] = vi
        cand[j] = vj
        r = 0.5 * n - float(np.sum(cand))
        if abs(r) > 1e-10:
            cand = project_capped(cand, n)
            if cand is None:
                return h, qbase, score, False
        cs = fast_score(cand, n)
        if cs < score - 1e-12:
            qbase = np.correlate(cand, 1.0 - cand, mode='full')
            return cand, qbase, float(cs), True
    return h, qbase, score, False

def pair_sweep(h, qbase, score, n, rng, pair_i, pair_j, use_all, grid):
    dx = 2.0 / n
    base_topo = float(score / dx)
    imps = 0
    if use_all:
        order = rng.permutation(pair_i.shape[0])
        for pos in order:
            i = int(pair_i[pos])
            j = int(pair_j[pos])
            h, qbase, score, ok = attempt_pair_update(h, qbase, score, n, i, j, base_topo, grid)
            if ok:
                imps += 1
                base_topo = float(np.max(qbase))
    else:
        cnt = min(1200, n * (n - 1) // 2)
        for _ in range(cnt):
            i = int(rng.integers(0, n - 1))
            j = int(rng.integers(i + 1, n))
            h, qbase, score, ok = attempt_pair_update(h, qbase, score, n, i, j, base_topo, grid)
            if ok:
                imps += 1
                base_topo = float(np.max(qbase))
    return h, qbase, float(dx * base_topo), imps

def local_polish(x, n, rng, sweeps, pair_i, pair_j, use_all):
    h = project_capped(x, n)
    if h is None:
        return None, float('inf')
    qbase = np.correlate(h, 1.0 - h, mode='full')
    score = float(np.max(qbase) * (2.0 / n))
    for _ in range(sweeps):
        h, qbase, score, imps = pair_sweep(h, qbase, score, n, rng, pair_i, pair_j, use_all, PAIR_GRID)
        if imps == 0:
            break
    return h, float(score)

def basin_round(best_h, best_score, fallback, extra_bases, n, rng, pair_i, pair_j, use_all,
                n_restarts, sweeps, sigma_lo, sigma_hi):
    eb = extra_bases if len(extra_bases) > 0 else [fallback]
    for it in range(n_restarts):
        if it % 4 == 0:
            x = balanced_binary(n, rng)
        elif it % 4 == 1:
            x = project_capped(rng.normal(0.0, 1.0, n), n)
        elif it % 4 == 2:
            base = eb[it % len(eb)]
            sigma = sigma_lo + (sigma_hi - sigma_lo) * float(rng.random())
            x = project_capped(base + sigma * rng.normal(0.0, 1.0, n), n)
        else:
            sigma = sigma_lo + (sigma_hi - sigma_lo) * float(rng.random())
            x = project_capped(best_h + sigma * rng.normal(0.0, 1.0, n), n)
        if x is None:
            continue
        h, s = local_polish(x, n, rng, sweeps, pair_i, pair_j, use_all)
        if h is not None and s < best_score - 1e-12:
            best_h, best_score = h, float(s)
    return best_h, float(best_score)

def epi_ineq(z):
    n = z.shape[0] - 1
    return z[-1] - np.correlate(z[:n], 1.0 - z[:n], mode='full') * (2.0 / n)

def epi_ineq_grad(z):
    n = z.shape[0] - 1
    h = z[:n]
    dx = 2.0 / n
    L = 2 * n - 1
    G = np.zeros((L, n), dtype=np.float64)
    for idx, d in enumerate(range(-(n - 1), n)):
        if d >= 0:
            s1 = 0
            e1 = n - d
        else:
            s1 = -d
            e1 = n + d
        G[idx, s1:e1] += 1.0 - h[s1 + d:e1 + d]
        if d <= 0:
            s2 = 0
            e2 = n + d
        else:
            s2 = d
            e2 = n
        G[idx, s2:e2] -= h[s2 - d:e2 - d]
    J = np.empty((L, n + 1), dtype=np.float64)
    J[:, :n] = -dx * G
    J[:, n] = 1.0
    return J

def epi_eq(z):
    n = z.shape[0] - 1
    return float(np.sum(z[:n])) - 0.5 * n

def epi_eq_grad(z):
    n = z.shape[0] - 1
    g = np.zeros(n + 1, dtype=np.float64)
    g[:n] = 1.0
    return g

def epi_obj(z):
    return float(z[-1])

def epi_obj_grad(z):
    g = np.zeros(z.shape[0], dtype=np.float64)
    g[-1] = 1.0
    return g

def slsqp_polish(h0, n, maxiter):
    h = project_capped(h0, n)
    if h is None:
        return None
    q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    t0 = float(np.max(q)) + max(1e-8, 1e-6 * max(1.0, abs(float(np.max(q)))))
    z0 = np.concatenate([h, np.array([t0])])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    cons = [
        {'type': 'ineq', 'fun': epi_ineq, 'jac': epi_ineq_grad},
        {'type': 'eq', 'fun': epi_eq, 'jac': epi_eq_grad},
    ]
    try:
        res = minimize(epi_obj, z0, jac=epi_obj_grad, bounds=bounds,
                       constraints=cons, method='SLSQP',
                       options={'maxiter': int(maxiter), 'ftol': 1e-12})
    except Exception:
        return None
    if res.x is None or res.x.shape[0] != n + 1:
        return None
    if not np.all(np.isfinite(res.x)):
        return None
    return project_capped(res.x[:n], n)

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass

    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is not None and parent.size != n:
        parent = None
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass

    if parent is not None:
        proj = project_capped(parent, n)
        fallback = proj if proj is not None else np.full(n, 0.5, dtype=np.float64)
    else:
        fallback = np.full(n, 0.5, dtype=np.float64)
    if not is_feasible(fallback, n):
        fallback = np.full(n, 0.5, dtype=np.float64)

    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)

    if n == 1:
        h = np.array([0.5], dtype=np.float64)
        return h, float(exact_score(h, 1)), 1

    if n <= 80:
        pair_i, pair_j = np.triu_indices(n, 1)
        use_all = True
    else:
        pair_i = None
        pair_j = None
        use_all = False

    pool = []
    add_feasible(pool, fallback, n)
    if parent is not None:
        add_feasible(pool, parent, n)
        add_feasible(pool, parent[::-1], n)
        add_feasible(pool, 1.0 - parent, n)
        add_feasible(pool, (1.0 - parent)[::-1], n)

    add_feasible(pool, np.full(n, 0.5, dtype=np.float64), n)

    blk = np.zeros(n, dtype=np.float64)
    k = n // 2
    blk[:k] = 1.0
    if n % 2 == 1 and k < n:
        blk[k] = 0.5
    add_feasible(pool, blk, n)

    alt = (np.arange(n) % 2 == 0).astype(np.float64)
    add_feasible(pool, alt, n)

    n_bin = 4 if n <= 80 else 2
    for _ in range(n_bin):
        add_feasible(pool, balanced_binary(n, rng), n)

    for sd in (0.05, 0.15, 0.30):
        add_feasible(pool, fallback + sd * rng.normal(0.0, 1.0, n), n)

    add_feasible(pool, rng.uniform(0.0, 1.0, n), n)

    t = np.linspace(0.0, 1.0, n)
    for kk in (1, 2):
        for ph in (0.0, 1.5):
            add_feasible(pool, 0.5 + 0.3 * np.cos(2.0 * np.pi * kk * t + ph), n)

    if n <= 80:
        limit = 14
    elif n <= 200:
        limit = 8
    else:
        limit = 5
    if len(pool) > limit:
        pool = pool[:limit]
    if len(pool) == 0:
        pool = [fallback]

    scores = np.array([fast_score(s, n) for s in pool], dtype=np.float64)
    order = np.argsort(scores)
    top = []
    for ii in order[:min(4, len(pool))]:
        top.append(pool[int(ii)])

    best_h = top[0].copy()
    best_score = float(scores[int(order[0])])

    if n <= 80:
        init_sw, top_sw, final_sw, slqp_iter = 4, 1, 5, 140
    elif n <= 200:
        init_sw, top_sw, final_sw, slqp_iter = 3, 1, 3, 0
    else:
        init_sw, top_sw, final_sw, slqp_iter = 2, 0, 2, 0

    for idx, st in enumerate(top):
        sw = init_sw if idx == 0 else top_sw
        if sw <= 0:
            continue
        h, s = local_polish(st, n, rng, sw, pair_i, pair_j, use_all)
        if h is not None and s < best_score - 1e-12:
            best_h, best_score = h, float(s)

    extra_bases = top[:min(4, len(top))]

    if n <= 80:
        if best_score > 0.38085:
            best_h, best_score = basin_round(best_h, best_score, fallback, extra_bases,
                                             n, rng, pair_i, pair_j, use_all, 8, 2, 0.05, 0.35)
        if best_score > 0.38080:
            best_h, best_score = basin_round(best_h, best_score, fallback, extra_bases,
                                             n, rng, pair_i, pair_j, use_all, 10, 1, 0.15, 0.60)
        if best_score > 0.38075:
            best_h, best_score = basin_round(best_h, best_score, fallback, extra_bases,
                                             n, rng, pair_i, pair_j, use_all, 8, 1, 0.30, 0.90)
    elif n <= 200:
        if best_score > 0.38085:
            best_h, best_score = basin_round(best_h, best_score, fallback, extra_bases,
                                             n, rng, pair_i, pair_j, use_all, 6, 1, 0.10, 0.45)
    else:
        if best_score > 0.38085:
            best_h, best_score = basin_round(best_h, best_score, fallback, extra_bases,
                                             n, rng, pair_i, pair_j, use_all, 3, 1, 0.10, 0.45)

    h, s = local_polish(best_h, n, rng, final_sw, pair_i, pair_j, use_all)
    if h is not None and s < best_score - 1e-12:
        best_h, best_score = h, float(s)

    if slqp_iter > 0:
        for _ in range(2):
            cand = slsqp_polish(best_h, n, slqp_iter)
            if cand is None:
                break
            cs = exact_score(cand, n)
            if cs < best_score - 1e-12:
                best_h, best_score = cand, float(cs)
            else:
                break

    final = project_capped(best_h, n)
    if final is None:
        final = fallback
    fs = exact_score(final, n)
    fb = exact_score(fallback, n)
    if fs > fb + 1e-12:
        final, fs = fallback, fb

    return final, float(fs), int(n)
