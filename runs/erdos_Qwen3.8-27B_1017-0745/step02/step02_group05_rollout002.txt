def pair_polish(h0, n, rng, sweeps, max_pairs):

    for sw in range(sweeps):

            pairs = [(min(a, b), max(a, b)) for a, b in pairs if a != b]
        for (i, j) in pairs:
            best_h = h.copy()
            best_s = score
            for direction in (0, 1):
                if direction == 0:

                    rec, don = j, i
