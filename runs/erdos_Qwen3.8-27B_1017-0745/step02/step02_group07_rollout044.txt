import numpy as np
import math

def project_capped(z, n):
    z = np.clip(np.asarray(z, dtype=np.float64).ravel(), -1e6, 1e6)
    s = n / 2.0
    lo, hi = z.min() - 1.0, z.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - h.sum()
    for _ in range(4):
        if abs(r) < 1e-13:
            break
        if r > 0:
            room = 1.0 - h
            mx = room.argmax()
            h[mx] = min(h[mx] + r, 1.0)
        else:
            mx = h.argmax()
            h[mx] = max(h[mx] + r, 0.0)
        r = s - h.sum()
    if not np.all(np.isfinite(h)) or h.min() < -1e-12 or h.max() > 1.0 + 1e-12:
        return None
    return h

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and h.min() >= -1e-12
            and h.max() <= 1.0 + 1e-12 and abs(h.sum() - n/2.0) <= 1e-8)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def make_binary(rng, n):
    p = rng.permutation(n)
    v = np.zeros(n)
    v[p[:n // 2]] = 1.0
    if n % 2:
        v[p[n // 2]] = 0.5
    return v

def build_pool(fb, n, rng):
    pool, scores = [], []
    def add(v):
        if v is None or not is_feasible(v, n):
            return
        for s in pool:
            if np.max(np.abs(v - s)) < 1e-7:
                return
        pool.append(v.copy())
        scores.append(exact_score(v, n))
    add(fb)
    add(project_capped(fb[::-1], n))
    add(project_capped(1.0 - fb, n))
    add(project_capped(np.roll(fb, n // 3), n))
    add(project_capped(np.roll(fb, n // 2), n))
    add(np.full(n, 0.5))
    blk = np.zeros(n); blk[:n // 2] = 1.0
    if n % 2: blk[n // 2] = 0.5
    add(project_capped(blk, n))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    add(project_capped(alt, n))
    add(project_capped(make_binary(rng, n), n))
    for sd in (0.04, 0.10, 0.22):
        add(project_capped(fb + rng.normal(0, sd, n), n))
    for _ in range(5):
        add(project_capped(make_binary(rng, n), n))
        add(project_capped(rng.uniform(0, 1, n), n))
    return pool, scores

def solve_sdp(n, rng):
    if n < 3 or n > 60:
        return None, None
    try:
        import cvxpy as cp
    except ImportError:
        return None, None
    s = n / 2.0
    v = cp.Variable(n)
    W = cp.SymbolicVariable((n, n), symmetric=True) if hasattr(cp, 'SymbolicVariable') else cp.Variable((n, n), symmetric=True)
    W = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    ones = np.ones(n)
    Z = cp.concatenate([cp.hstack([cp.Constant(1.0), cp.reshape(v, (1, n))]),
                        cp.hstack([cp.reshape(v, (n, 1)), W])], axis=1)
    cons = [cp.psd(Z), v >= 0, v <= 1, cp.sum(v) == s, t >= 0, t <= 1, W @ ones == s * v]
    dx = 2.0 / n
    cons.append(dx * (cp.sum(v) - cp.trace(W)) <= t)
    for r in range(1, n):
        m = n - r
        qp, qm = 0, 0
        for i in range(m):
            qp = qp + v[i + r] - W[i, i + r]
            qm = qm + v[i] - W[i, i + r]
        cons.append(dx * qp <= t)
        cons.append(dx * qm <= t)
    prob = cp.Problem(cp.Minimize(t), cons)
    for solver in ('MOSEK', 'CVXOPT'):
        try:
            prob.solve(solver=solver, verbose=False, max_iters=100)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                vv = np.array(v.value, dtype=np.float64).ravel()
                Wv = np.array(W.value, dtype=np.float64)
                if vv.shape == (n,) and Wv.shape == (n, n) and np.all(np.isfinite(vv)) and np.all(np.isfinite(Wv)):
                    return vv, Wv
        except Exception:
            continue
    return None, None

def anneal_chain(h0, sc0, n, rng, steps, best_h, best_s):
    h = h0.copy()
    sc = sc0
    T0, Tf = 0.12 * (2.0 / n), 0.004 * (2.0 / n)
    for k in range(steps):
        rho = k / max(1, steps - 1)
        T = T0 * (Tf / T0) ** rho
        delta = 0.18 * (1.0 - 0.75 * rho)
        cand = None
        if rng.random() < 0.85:
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j:
                continue
            lo_t = max(-h[i], h[j] - 1.0, -delta)
            hi_t = min(1.0 - h[i], h[j], delta)
            if hi_t - lo_t < 1e-12:
                continue
            tv = rng.uniform(lo_t, hi_t)
            cand = h.copy()
            cand[i] += tv
            cand[j] -= tv
        else:
            d = rng.standard_normal(n)
            d -= d.mean()
            mx = np.max(np.abs(d))
            if mx < 1e-14:
                continue
            d /= mx
            lo_t, hi_t = -delta, delta
            for ii in range(n):
                if d[ii] > 1e-14:
                    lo_t = max(lo_t, -h[ii] / d[ii])
                    hi_t = min(hi_t, (1.0 - h[ii]) / d[ii])
                elif d[ii] < -1e-14:
                    lo_t = max(lo_t, (1.0 - h[ii]) / d[ii])
                    hi_t = min(hi_t, -h[ii] / d[ii])
            if hi_t - lo_t < 1e-12:
                continue
            tv = rng.uniform(lo_t, hi_t)
            cand = project_capped(h + tv * d, n)
        if cand is None:
            cand = project_capped(cand if cand is not None else h, n)
        if cand is None or not is_feasible(cand, n):
            continue
        if np.max(np.abs(cand - h)) < 1e-12:
            continue
        ns = exact_score(cand, n)
        if ns < best_s - 1e-12:
            best_s = ns
            best_h = cand.copy()
        if ns <= sc + 1e-15:
            h, sc = cand, ns
        elif T > 1e-15:
            if math.log(max(rng.random(), 1e-300)) < -(ns - sc) / T:
                h, sc = cand, ns
    return best_h, best_s
