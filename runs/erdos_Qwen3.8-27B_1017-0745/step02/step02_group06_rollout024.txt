    while len(pool) < 2 * P and att < 12 * P:
        ctr = fallback if att % 2 == 0 else h_uni
        ns = [0.025, 0.06, 0.14, 0.30][att % 4]
        p = project_capped(ctr + rng.normal(0, ns, n), n); att += 1
        if p is None or not is_feasible(p, n):
            continue
        sc = exact_score(p, n)
        if not np.isfinite(sc) or any(np.max(np.abs(p - sv)) < 1e-8 for sv in seen):
            continue
        pool.append((p.copy(), sc)); seen.append(p.copy())
    return pool

def select_pop(pool, n, P):
    if len(pool) <= P:
        return [pool[i][0].copy() for i in range(len(pool))] + [pool[0][0].copy()] * max(0, P - len(pool))
    order = sorted(range(len(pool)), key=lambda i: (pool[i][1], i))
    sel = [order[0]]
    for _ in range(P - 1):
        best_d, best_i = -1.0, sel[0]
        for idx in range(len(pool)):
            if idx in sel:
                continue
            md = min(math.sqrt(float(np.mean((pool[idx][0] - pool[s][0]) ** 2))) for s in sel)
            if md > best_d:
                best_d, best_i = md, idx
        sel.append(best_i)
    return [pool[i][0].copy() for i in sel]

def de_search(pop, scores, n, rng, max_gen, stall_lim, max_rest, P):
    mu_F, mu_CR = 0.55, 0.85
    archive = []
    stall, restarts, gen = 0, 0, 0
    n_rand1 = max(1, (max_gen + 3) // 4)
    inc_vec = pop[int(np.argmin(scores))].copy()
    inc_s = float(scores.min())
    scales = [0.04, 0.12, 0.25]
    while gen < max_gen:
        old_pop = [pop[i].copy() for i in range(P)]
        old_scores = scores.copy()
        order = list(np.argsort(old_scores))
        p_cnt = max(2, P // 5)
        F_ok, CR_ok = [], []
        prev_inc = inc_s
        for i in range(P):
            r1 = int(rng.integers(0, P))
            while r1 == i:
                r1 = int(rng.integers(0, P))
            r2 = int(rng.integers(0, P))
            while r2 == i or r2 == r1:
                r2 = int(rng.integers(0, P))
            if gen < n_rand1:
                r3 = int(rng.integers(0, P))
                while r3 == i or r3 == r1 or r3 == r2:
                    r3 = int(rng.integers(0, P))
                F = _cauchy(rng, mu_F, 0.1)
                v = old_pop[r1] + F * (old_pop[r2] - old_pop[r3])
            else:
                pb = int(rng.integers(0, p_cnt))
                pb_i = order[pb]
                if pb_i == i:
                    pb_i = order[(pb + 1) % p_cnt]
                F = _cauchy(rng, mu_F, 0.1)
                v = old_pop[i] + F * (old_pop[pb_i] - old_pop[i]) + F * (old_pop[r1] - old_pop[r2])
            CR = min(1.0, max(0.0, mu_CR + 0.1 * rng.standard_normal()))
            mask = rng.random(n) < CR
            if not mask.any():
                mask[int(rng.integers(0, n))] = True
            trial_raw = np.where(mask, v, old_pop[i])
            trial = project_capped(trial_raw, n)
            if trial is None or not is_feasible(trial, n):
                continue
            ts = exact_score(trial, n)
            if not np.isfinite(ts):
                continue
            if ts < old_scores[i] - 1e-12:
                archive.append(old_pop[i].copy())
                pop[i] = trial.copy()
                scores[i] = ts
                F_ok.append(F); CR_ok.append(CR)
                if ts < inc_s - 1e-12:
                    inc_vec, inc_s = trial.copy(), ts
        if len(archive) > P:
            keep = rng.choice(len(archive), P, replace=False)
            archive = [archive[k] for k in keep]
        if F_ok:
            c = 0.1
            mu_F = (1 - c) * mu_F + c * float(np.mean(np.array(F_ok) ** 2) / np.mean(np.array(F_ok)))
            mu_CR = (1 - c) * mu_CR + c * float(np.mean(CR_ok))
            mu_F = min(1.0, max(0.05, mu_F))
            mu_CR = min(1.0, max(0.0, mu_CR))
        gen += 1
        if inc_s < prev_inc - 1e-12:
            stall = 0
        else:
            stall += 1
        if stall >= stall_lim:
            if restarts < max_rest and gen < max_gen:
                idxs = list(np.argsort(scores))
                keep_n = max(2, P // 2)
                pop2 = [pop[idxs[k]].copy() for k in range(keep_n)]
                scores2 = [scores[idxs[k]] for k in range(keep_n)]
                if inc_s < scores2[-1]:
                    pop2[-1] = inc_vec.copy(); scores2[-1] = inc_s
                for k in range(keep_n, P):
                    sd = scales[restarts % 3]
                    got = False
                    for tt in range(5):
                        p = project_capped(inc_vec + rng.normal(0, sd * (1.5 ** tt), n), n)
                        if p is not None and is_feasible(p, n):
                            ps = exact_score(p, n)
                            if np.isfinite(ps):
                                pop2.append(p.copy()); scores2.append(ps); got = True; break
                    if not got:
                        pop2.append(pop2[0].copy()); scores2.append(scores2[0])
                pop = pop2[:P]; scores = np.array(scores2[:P])
                archive = []; mu_F, mu_CR = 0.55, 0.85
                stall = 0; restarts += 1
            else:
                break
    return pop, scores, inc_vec, inc_s

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None:
        n = 47
    if n == 1:
        h = np.array([0.5])
        return h, float(exact_score(h, 1)), 1
    if parent is not None:
        fp = project_capped(parent, n)
        parent = fp if (fp is not None and is_feasible(fp, n)) else None
    h_uni = np.full(n, 0.5)
    fallback = parent.copy() if parent is not None else h_uni.copy()
    fb_s = exact_score(fallback, n)
    rng = np.random.default_rng(int(seed))
    if n <= 64:
        P, max_gen, stall_lim, max_rest = 48, 80, 15, 2
    elif n <= 128:
        P, max_gen, stall_lim, max_rest = 36, 55, 12, 2
    elif n <= 256:
        P, max_gen, stall_lim, max_rest = 24, 35, 8, 1
    else:
        P, max_gen, stall_lim, max_rest = 16, 18, 5, 1
    pool = build_pool(fallback, n, rng, P)
    pop = select_pop(pool, n, P)
    while len(pop) < P:
        pop.append(fallback.copy())
    scores = np.array([exact_score(pop[i], n) for i in range(P)])
    inc_vec, inc_s = fallback.copy(), fb_s
    for pv, ps in pool:
        if ps < inc_s - 1e-12:
            inc_vec, inc_s = pv.copy(), ps
    pop, scores, inc_vec, inc_s = de_search(pop, scores, n, rng, max_gen, stall_lim, max_rest, P)
    final = project_capped(inc_vec, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    if fs > fb_s + 1e-12:
        final = fallback
        fs = fb_s
    return np.ascontiguousarray(final, dtype=np.float64), float(fs), int(n)
