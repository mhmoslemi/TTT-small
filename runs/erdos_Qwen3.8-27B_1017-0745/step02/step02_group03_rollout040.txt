    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if abs(hi - lo) < 1e-15:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(50):
        r = s - float(np.sum(h))
        if abs(r) < 1e-13 * max(1, n):
            break
        if r > 0:
            cap = 1.0 - h
            tot = float(np.sum(cap))
            if tot < 1e-15:
                break
            h = h + r * cap / tot
        else:
            cap = h.copy()
            tot = float(np.sum(cap))
            if tot < 1e-15:
                break
            h = h + r * cap / tot
        h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-9 * max(1, n):
        return None
    return h

def exact_score(h, n):
    try:
        if h.shape != (n,) or not np.all(np.isfinite(h)):
            return float('inf')
        if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
            return float('inf')
        if abs(float(np.sum(h)) - n / 2.0) > 1e-8 * max(1, n):
            return float('inf')
        return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
    except Exception:
        return float('inf')

def feasible_check(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    return abs(float(np.sum(h)) - n / 2.0) <= 1e-8 * max(1, n)

def pair_proposal(h, scale, rng, full):
    n = len(h)
    if n < 2:
        return None
    p = int(rng.integers(n))
    q = int(rng.integers(n - 1))
    if q >= p:
        q += 1
    L = max(-h[p], h[q] - 1.0)
    U = min(1.0 - h[p], h[q])
    if U - L <= 1e-14:
        return None
    if full:
        t = L + (U - L) * float(rng.random())
    else:
        t = -scale + 2.0 * scale * float(rng.random())
        if t < L or t > U:
            return None
    hp, hq = h[p] + t, h[q] - t
    if hp < -1e-10 or hp > 1 + 1e-10 or hq < -1e-10 or hq > 1 + 1e-10:
        return None
    out = h.copy()
    out[p], out[q] = hp, hq
    return out

def hit_run_proposal(h, rng):
    n = len(h)
    for _ in range(5):
        g = rng.standard_normal(n)
        d = g - float(np.mean(g))
        nm = float(np.linalg.norm(d))
        if nm < 1e-14:
            continue
        d = d / nm
        L, U = -1e18, 1e18
        for i in range(n):
            if d[i] > 1e-15:
                L, U = max(L, -h[i] / d[i]), min(U, (1.0 - h[i]) / d[i])
            elif d[i] < -1e-15:
                L, U = max(L, (1.0 - h[i]) / d[i]), min(U, -h[i] / d[i])
        if U - L <= 1e-14:
            continue
        t = L + (U - L) * float(rng.random())
        out = h + t * d
        if np.all(np.isfinite(out)) and feasible_check(out, n):
            return out
    return None

def add_start(starts, v, n):
    h2 = project_capped(v, n)
    if h2 is None:
        return
    for ex in starts:
        if float(np.max(np.abs(h2 - ex))) < 1e-8:
            return
    starts.append(h2)

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= len(iv) <= 999 and np.all(np.isfinite(iv)):
            n, parent = len(iv), iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
        n = max(1, min(999, n))
    sd = int(seed) % (2 ** 32) if isinstance(seed, (int, np.integer)) else 42
    rng = np.random.default_rng(sd)
    if n == 1:
        return np.array([0.5]), 0.0, 1

    if parent is not None:
        fb = project_capped(parent, n)
        if fb is None:
            fb = np.full(n, 0.5)
    else:
        fb = np.full(n, 0.5)
    fb_score = exact_score(fb, n)
    if not np.isfinite(fb_score):
        fb, fb_score = np.full(n, 0.5), exact_score(np.full(n, 0.5), n)

    starts = []
    add_start(starts, fb, n)
    if parent is not None:
        add_start(starts, fb + rng.normal(0, 0.02, n), n)
        add_start(starts, fb + rng.normal(0, 0.08, n), n)
        add_start(starts, fb[::-1], n)
        add_start(starts, 1.0 - fb, n)
        add_start(starts, rng.uniform(0, 1, n), n)
        add_start(starts, fb + rng.normal(0, 0.15, n), n)
    else:
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        add_start(starts, blk, n)
        tt = np.arange(n)
        add_start(starts, 0.5 + 0.45 * np.cos(2 * np.pi * tt / n), n)
        add_start(starts, 0.5 + 0.45 * np.cos(4 * np.pi * tt / n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
    while len(starts) < 6:
        add_start(starts, rng.uniform(0, 1, n), n)
    starts = starts[:6]
    nrep = len(starts)

    B_def = min(16000, max(120, 30000000 // (n * n)))
    B = B_def
    try:
        B_ov = int(kwargs.get('eval_budget', 0))
        if B_ov > 0:
            B = max(120, min(B_ov, B_def, 16000))
    except Exception:
        pass

    temps = np.array([0.0005, 0.002, 0.008, 0.032, 0.128, 0.512] + [0.512] * max(0, 6 - 6))
    scales = np.array([0.002, 0.006, 0.018, 0.054, 0.162, 0.486])
    temps, scales = temps[:nrep], scales[:nrep]

    states, scores = [], []
    for st in starts:
        hi = project_capped(0.99 * st + 0.01 * 0.5, n)
        if hi is None:
            hi = st.copy()
        states.append(hi)
        scores.append(exact_score(hi, n))

    best, best_sc = fb.copy(), fb_score
    for i in range(nrep):
        if np.isfinite(scores[i]) and scores[i] < best_sc - 1e-12:
            best, best_sc = states[i].copy(), scores[i]

    evals, attempts = nrep, 0
    max_att = 30 * B
    no_imp, rst = 0, 0
    no_imp_max = max(25, min(180, B // max(1, nrep * 10)))

    while evals < B and attempts < max_att:
        sweep_imp = False
        for i in range(nrep):
            if evals >= B or attempts >= max_att:
                break
            attempts += 1
            r = float(rng.random())
            if r < 0.35:
                prop = hit_run_proposal(states[i], rng)
            elif r < 0.65:
                prop = pair_proposal(states[i], scales[i], rng, True)
            else:
                prop = pair_proposal(states[i], scales[i], rng, False)
            if prop is None:
                continue
            sc2 = exact_score(prop, n)
            evals += 1
            if not np.isfinite(sc2):
                continue
            d = sc2 - scores[i]
            acc = d <= 0.0
            if not acc and temps[i] > 0:
                acc = math.log(max(float(rng.random()), 1e-300)) < -d / temps[i]
            if acc:
                states[i], scores[i] = prop, sc2
                if sc2 < best_sc - 1e-12:
                    best, best_sc, sweep_imp = prop.copy(), sc2, True

        pairs = [(0, 1), (2, 3), (4, 5)] if evals % 2 == 0 else [(1, 2), (3, 4)]
        for a, b in pairs:
            if a >= nrep or b >= nrep:
                continue
            lr = (1.0 / temps[a] - 1.0 / temps[b]) * (scores[a] - scores[b])
            if lr >= 0 or math.log(max(float(rng.random()), 1e-300)) < lr:
                states[a], states[b] = states[b], states[a]
                scores[a], scores[b] = scores[b], scores[a]

        if sweep_imp:
            no_imp = 0
        else:
            no_imp += 1
        if no_imp >= no_imp_max and rst < 2:
            hrep = nrep - 1
            if rst == 0:
                ns = project_capped(0.5 * rng.uniform(0, 1, n) + 0.5 * best, n)
            else:
                ns = project_capped(rng.uniform(0, 1, n) ** 0.7, n)
            if ns is None:
                ns = project_capped(rng.uniform(0, 1, n), n)
            if ns is not None:
                ns2 = project_capped(0.99 * ns + 0.01 * 0.5, n)
                if ns2 is not None:
                    states[hrep] = ns2
                    scores[hrep] = exact_score(ns2, n)
                    evals += 1
                    if np.isfinite(scores[hrep]) and scores[hrep] < best_sc - 1e-12:
                        best, best_sc = ns2.copy(), scores[hrep]
            no_imp, rst = 0, rst + 1

    # Lightweight final local polish
    for _ in range(3):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                L = max(-best[i], best[j] - 1.0)
                U = min(1.0 - best[i], best[j])
                if U - L <= 1e-14:
                    continue
                for frac in (0.1, 0.3, 0.5, 0.7, 0.9):
                    t = L + frac * (U - L)
                    trial = best.copy()
                    trial[i], trial[j] = best[i] + t, best[j] - t
                    if not feasible_check(trial, n):
                        continue
                    sc = exact_score(trial, n)
                    if np.isfinite(sc) and sc < best_sc - 1e-12:
                        best, best_sc, improved = trial, sc, True
                        break
        if not improved:
            break

    final = project_capped(best, n)
    if final is None:
        final = fb.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final, fs = fb.copy(), fb_score
    if not np.isfinite(fs):
        final = np.full(n, 0.5)
        fs = exact_score(final, n)
    return final.astype(np.float64), float(fs), int(n)
