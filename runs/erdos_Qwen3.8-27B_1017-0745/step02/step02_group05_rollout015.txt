def exact_score(h, n):

def build_starts(base, n, rng, P):
    pool = [base.copy()]
    cands = [base[::-1].copy(), 1.0 - base, 1.0 - base[::-1], np.full(n, 0.5)]
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:

    cands.append(np.roll(base, 1))
    cands.append(np.roll(base, -1))

def pair_anneal(h, n, rng, L):

    cur_s = exact_score(h, n)
    best_h = h.copy()
    best_s = cur_s
    T_init, T_end = 5e-4, 1e-7

        q = t / max(1, L - 1)
        T = T_init * (T_end / T_init) ** q
        rho = 1.0 - 0.98 * q
        accepted = False
        for _ in range(20):
            i, j = rng.choice(n, size=2, replace=False)
            ci = min(1.0 - h[i], h[j])
            cj = min(1.0 - h[j], h[i])
            if ci > 1e-12 and rng.random() < 0.5:
                amt = ci * rho * (0.05 + 0.95 * rng.random())
                h[i] += amt
                h[j] -= amt
                accepted = True
                break
            elif cj > 1e-12:
                amt = cj * rho * (0.05 + 0.95 * rng.random())
                h[j] += amt
                h[i] -= amt
                accepted = True
                break
        if not accepted:
            continue
        h = project_capped(h, n)
        if h is None or not is_feasible(h, n):
            continue
        ns = exact_score(h, n)
        if not np.isfinite(ns):
            continue
        delta = ns - cur_s
        if delta < 0:
            cur_s = ns
            if ns < best_s - 1e-12:
                best_s = ns
                best_h = h.copy()
        else:
            if T > 1e-10 and rng.random() < np.exp(-delta / T) if -delta / T > -700 else False:
                cur_s = ns
    return best_h, best_s

def pair_polish(h, n, rng, nsweeps, max_pairs):
    h = h.copy()
    cur_s = exact_score(h, n)
    best_h = h.copy()
    best_s = cur_s
    fracs = np.array([1.0/64, 1.0/16, 1.0/4, 0.5, 1.0])
    if n <= 64:
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    else:
        pairs = [(int(a), int(b)) for a, b in rng.integers(0, n, size=(max_pairs * nsweeps, 2)).T
                 if a != b]
    for sweep in range(nsweeps):
        imp = False
        for i, j in pairs:
            if sweep >= 1 and i * (n-1-i) // (n - 1) >= max_pairs:
                continue
            for direction in (0, 1):
                if direction == 0:
                    cap = min(1.0 - h[i], h[j])
                    di, dj = i, j
                else:
                    cap = min(1.0 - h[j], h[i])
                    di, dj = j, i
                if cap <= 1e-14:
                    continue
                for f in fracs:
                    amt = cap * f
                    h[di] += amt
                    h[dj] -= amt
                    tr = project_capped(h, n)
                    h[di] -= amt
                    h[dj] += amt
                    if tr is None:
                        continue

                    if np.isfinite(ts) and ts < cur_s - 1e-12:
                        h = tr
                        cur_s = ts
                        imp = True
                        if cur_s < best_s - 1e-12:
                            best_s = cur_s
                            best_h = h.copy()
        if not imp:
            break
    return best_h, best_s
