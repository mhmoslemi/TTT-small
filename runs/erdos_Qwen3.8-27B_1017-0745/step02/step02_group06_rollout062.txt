import numpy as np
from scipy.optimize import minimize

def project_capped(z, n):
    z = np.where(np.isfinite(z), z, 0.0).astype(np.float64)
    z = np.clip(z, -1e6, 1e6)
    s_t = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s_t:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(z - lam, 0.0, 1.0)
    r = s_t - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx])
            h[idx] += d; r -= d
        else:
            d = min(-r, h[idx])
            h[idx] -= d; r += d
    if abs(r) > 1e-10 * max(1, n) or not np.all(np.isfinite(h)):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)) or c.size != 2 * n - 1:
        return float('inf')
    return float(np.max(c))

def lag_values(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def lag_jacobian(h, n):
    dx = 2.0 / n
    J = np.zeros((2 * n - 1, n))
    for s in range(-(n - 1), n):
        a = max(0, -s)
        b = min(n - 1, n - 1 - s)
        if a > b:
            continue
        row = s + n - 1
        q1lo = max(0, a + s)
        q1hi = min(n - 1, b + s)
        if q1lo <= q1hi:
            qs = np.arange(q1lo, q1hi + 1)
            J[row, qs] += dx * (1.0 - h[qs - s])
        q2lo = max(a, -s)
        q2hi = min(b, n - 1 - s)
        if q2lo <= q2hi:
            qs = np.arange(q2lo, q2hi + 1)
            J[row, qs] -= dx * h[qs + s]
    return J

def _obj(z, n):
    return z[-1]

def _obj_grad(z, n):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def _ineq(z, n):
    return z[-1] - lag_values(z[:n], n)

def _ineq_jac(z, n):
    J = -lag_jacobian(z[:n], n)
    G = np.zeros((2 * n - 1, 2 * n))
    G[:, :n] = J
    G[:, n] = 1.0
    return G

def _eq(z, n):
    return np.sum(z[:n]) - n / 2.0

def _eq_jac(z, n):
    g = np.zeros(len(z))
    g[:n] = 1.0
    return g

def solve_slsqp(h0, n, maxiter):
    t0 = min(0.5, exact_score(h0, n) + 1e-9)
    t0 = max(0.0, t0)
    z0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 0.5)]
    cons = [
        {'type': 'eq', 'fun': _eq, 'jac': _eq_jac, 'args': (n,)},
        {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac, 'args': (n,)}
    ]
    try:
        res = minimize(_obj, z0, jac=_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
    except Exception:
        return None
    if res is None or not hasattr(res, 'x') or res.x is None:
        return None
    if len(res.x) != n + 1 or not np.all(np.isfinite(res.x)):
        return None
    return project_capped(res.x[:n], n)
