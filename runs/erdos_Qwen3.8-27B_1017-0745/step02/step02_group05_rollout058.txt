                    c = g_best + sig * rng.normal(0, 1, n)
                else:
                    c = rng.uniform(0, 1, n)
                p2 = project_capped(c, n)
                if p2 is not None and np.isfinite(exact_score(p2, n)):
                    new_pop.append(p2)
                else:
                    new_pop.append(g_best.copy())
            pop = new_pop[:P]
            scores = [exact_score(p, n) for p in pop]
            bi = int(np.argmin(scores))
            g_best, g_score = pop[bi].copy(), scores[bi]
            pop[0] = g_best.copy()
            scores[0] = g_score
            stalls = 0
    ax, as_ = pair_anneal(g_best.copy(), n, rng, L)
    if as_ < g_score - 1e-12:
        g_best, g_score = ax, as_
    px, ps = pair_polish(g_best.copy(), n, rng, nsw, full_pairs)
    if ps < g_score - 1e-12:
        g_best, g_score = px, ps
    ax2, as2 = pair_anneal(g_best.copy(), n, rng, L // 3)
    if as2 < g_score - 1e-12:
        g_best, g_score = ax2, as2
    px2, ps2 = pair_polish(g_best.copy(), n, rng, 1, full_pairs)
    if ps2 < g_score - 1e-12:
        g_best, g_score = px2, ps2
    final = project_capped(g_best, n)
    if final is None or not is_feasible(final, n):
        final = safe_h.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > safe_s + 1e-12:
        final, fs = safe_h.copy(), safe_s
    return final.astype(np.float64), float(fs), int(n)
