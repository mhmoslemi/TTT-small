    best_h = h.copy()
    best_s = exact_score(h, n)
    dx = 2.0 / n
    for tau in taus:
        eta = 0.25 / dx
        eta_max = 2.0 / dx
        for _ in range(maxits):
            val, g = smooth_grad(h, n, tau)
            if not np.isfinite(val) or not np.all(np.isfinite(g)):
                break
            accepted = False
            te = eta
            for _bt in range(25):
                z = project_capped(h - te * g, n)
                if z is None:
                    break
                d = z - h
                if np.max(np.abs(d)) <= 1e-11:
                    break
                dot = float(np.dot(g, d))
                if not np.isfinite(dot) or dot >= 0:
                    break
                vtr, _ = smooth_grad(z, n, tau)
                if np.isfinite(vtr) and vtr <= val + 1e-4 * dot + 1e-14:
                    h = z
                    val = vtr
                    accepted = True
                    break
                te *= 0.5
            if not accepted:
                break
            eta = min(1.2 * eta, eta_max)
            sz = exact_score(h, n)
            if sz < best_s - 1e-12:
                best_s = sz
                best_h = h.copy()
    return best_h, best_s

def symmetrize(h, n):
    sym = 0.5 * (h + h[::-1])
    return project_capped(sym, n)

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size >= 1 and np.all(np.isfinite(p)):
            n = int(p.size)
            if np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if parent is not None and parent.size != n:
        x_old = (np.arange(parent.size) + 0.5) / parent.size
        x_new = (np.arange(n) + 0.5) / n
        parent = np.interp(x_new, x_old, parent, left=parent[0], right=parent[-1])
    if n == 1:
        h1 = np.array([0.5], dtype=np.float64)
        return h1, 0.0, 1
    fb = project_capped(parent, n) if parent is not None else None
    if fb is None:
        fb = np.full(n, 0.5, dtype=np.float64)
    fs = exact_score(fb, n)
    gb, gs = fb.copy(), fs
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 64:
        n_sc, maxits, n_re = 5, 140, 2
    elif n <= 128:
        n_sc, maxits, n_re = 4, 90, 1
    elif n <= 256:
        n_sc, maxits, n_re = 3, 55, 1
    else:
        n_sc, maxits, n_re = 2, 25, 0
    taus = [0.02, 0.005, 0.001, 0.0002, 0.00004, 0.00001]
    starts = build_starts(fb, n, rng, n_sc)
    for st in starts:
        try:
            bh, bs = continuation_run(st, n, taus, maxits)
            if bs < gs - 1e-12:
                gs, gb = bs, bh
        except Exception:
            continue
    if n_re > 0:
        if n <= 64:
            sds = [0.08, 0.18] if gs < fs - 1e-5 else [0.03, 0.08]
        else:
            sds = [0.12] if gs < fs - 1e-5 else [0.05]
        sds = sds[:n_re]
        for sd in sds:
            try:
                pert = project_capped(gb + rng.normal(0, sd, n), n)
                if pert is None:
                    continue
                bh, bs = continuation_run(pert, n, taus, maxits)
                if bs < gs - 1e-12:
                    gs, gb = bs, bh
            except Exception:
                continue
    sym = symmetrize(gb, n)
    if sym is not None:
        ss = exact_score(sym, n)
        if ss < gs - 1e-12:
            gs, gb = ss, sym
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb
    sf = exact_score(final, n)
    if sf > fs:
        final, sf = fb, fs
    if not is_feasible(final, n):
        final = np.full(n, 0.5, dtype=np.float64)
        sf = exact_score(final, n)
    return final.astype(np.float64), float(sf), int(n)
