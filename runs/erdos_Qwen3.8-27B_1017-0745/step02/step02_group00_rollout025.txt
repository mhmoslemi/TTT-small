def project_capped(z, n):
    s = n / 2.0
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(80):
        mid = lo + 0.5 * (hi - lo)
        if mid == lo or mid == hi:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for i in range(n):
        if abs(r) < 1e-13:
            break
        if r > 0:
            a = min(r, 1.0 - h[i]); h[i] += a; r -= a
        else:
            a = min(-r, h[i]); h[i] -= a; r += a
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-8:
        return None
    return h

def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return float('inf')
    if abs(float(np.sum(h)) - n / 2.0) > 1e-8:
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_values(x, n):
    dx = 2.0 / n
    m = 2 * n - 1
    q = np.zeros(m)
    for d in range(-(n - 1), n):
        lo, hi = max(0, -d), min(n, n - d)
        q[d + n - 1] = dx * float(np.dot(x[lo:hi], 1.0 - x[lo + d:hi + d]))
    return q

def weighted_lag_grad(x, w, n):
    dx = 2.0 / n
    gbar = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[d + n - 1]
        if abs(wd) < 1e-15:
            continue
        lo, hi = max(0, -d), min(n, n - d)
        gbar[lo:hi] += dx * wd * (1.0 - x[lo + d:hi + d])
        gbar[lo + d:hi + d] -= dx * wd * x[lo:hi]
    return gbar

def directional_products(x, p, n):
    dx = 2.0 / n
    m = 2 * n - 1
    r = np.zeros(m)
    for d in range(-(n - 1), n):
        lo, hi = max(0, -d), min(n, n - d)
        r[d + n - 1] = dx * (float(np.dot(p[lo:hi], 1.0 - x[lo + d:hi + d]))
                          - float(np.dot(x[lo:hi], p[lo + d:hi + d])))
    return r

def project_simplex(v):
    m = len(v)
    v = np.asarray(v, dtype=np.float64).ravel()
    if not np.all(np.isfinite(v)):
        return np.full(m, 1.0 / m)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.nonzero(u * np.arange(1, m + 1) > (css - 1.0))[0]
    if len(idx) == 0:
        return np.full(m, 1.0 / m)
    rho = int(idx[-1]) + 1
    theta = (css[rho - 1] - 1.0) / float(rho)
    w = np.maximum(v - theta, 0.0)
    ss = float(np.sum(w))
    if ss < 1e-15:
        return np.full(m, 1.0 / m)
    return w / ss

def dual_solve(x, a, mu, w_init, n, K):
    m = 2 * n - 1
    L_D = 8.0 / mu
    w = project_simplex(w_init)

    best_cand = x.copy()

    prev_w = None

        y_z = project_capped(x - gbar_z / mu, n)

        r_z = directional_products(x, p_z, n)
        gradD = a + r_z

        gbar_n = weighted_lag_grad(x, w_new, n)
        y_new = project_capped(x - gbar_n / mu, n)
        if y_new is None:

        p_new = y_new - x
        r_new = directional_products(x, p_new, n)
        D_new = float(np.dot(w_new, a)) + float(np.dot(gbar_n, p_new)) + 0.5 * mu * float(np.dot(p_new, p_new))
        P_new = float(np.max(a + r_new)) + 0.5 * mu * float(np.dot(p_new, p_new))
        if not np.isfinite(P_new):
            P_new = float('inf')
        if P_new < best_P:
            best_P = P_new
            best_cand = y_new.copy()
        if prev_w is not None and float(np.max(np.abs(w_new - prev_w))) <= 1e-10:
            stall += 1
        else:

            break

            break

        w = w_new
        prev_w = w_new.copy()
        t = t_new
    return best_cand, w

def local_mmm(x, n, max_outer, dual_iter, mu_retries):
    dx = 2.0 / n
    m = 2 * n - 1
    mu_min = 2.0 * dx
    x = x.copy()

            for j in range(9):
