import numpy as np
import math

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(2 * n + 4):
        if abs(r) < 1e-14 * max(1, n):
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(np.sum(h) - s) > 1e-10 * max(1, n):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def helmert_fwd(x, n):
    d = n - 1
    y = np.zeros(n)
    if d <= 0:
        return y
    u = np.empty(d)
    for k in range(d):
        u[k] = x[k] / math.sqrt((k + 1) * (k + 2))
    suffix = np.zeros(d + 1)
    for k in range(d - 1, -1, -1):
        suffix[k] = suffix[k + 1] + u[k]
    y[0] = suffix[0]
    for i in range(1, n):
        y[i] = suffix[i] - i * u[i - 1]
    return y

def helmert_transpose(y, n):
    d = n - 1
    x = np.zeros(d)
    pref = np.zeros(d + 1)
    for i in range(1, d + 1):
        pref[i] = pref[i - 1] + y[i - 1]
    for k in range(d):
        x[k] = (pref[k + 1] - (k + 1) * y[k + 1]) / math.sqrt((k + 1) * (k + 2))
    return x

def latent_to_h(x, n):
    if not np.all(np.isfinite(x)):
        return None
    y = helmert_fwd(x, n)
    if not np.all(np.isfinite(y)):
        return None
    return project_capped(0.5 + y, n)

    mu = max(1, P // 2)

def helmert_fwd(x, n):
    d = n - 1
    y = np.zeros(n)
    if d <= 0:
        return y
    u = np.empty(d)
    for k in range(d):
        u[k] = x[k] / math.sqrt((k + 1) * (k + 2))
    suf = np.zeros(d + 1)
    for k in range(d - 1, -1, -1):
        suf[k] = suf[k + 1] + u[k]
    y[0] = suf[0]
    for i in range(1, n):
        y[i] = suf[i] - i * u[i - 1]
    return y

def helmert_trans(y, n):
    d = n - 1
    x = np.zeros(d)
    pref = 0.0
    for k in range(d):
        pref += y[k]
        x[k] = (pref - (k + 1) * y[k + 1]) / math.sqrt((k + 1) * (k + 2))
    return x
