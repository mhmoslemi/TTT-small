        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if parent is not None:
        proj = project_capped(parent, n)
        fallback = proj if proj is not None else np.full(n, 0.5)
    else:
        fallback = np.full(n, 0.5)
    if not is_feasible(fallback, n):
        fallback = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        return fallback, float(exact_score(fallback, n)), n
    if n <= 64:
        n_starts, n_passes = 6, 45
    elif n <= 256:
        n_starts, n_passes = 3, 30
    else:
        n_starts, n_passes = 2, 18
    pool = build_starts(fallback, n, rng, 12)
    scores = [exact_score(s, n) for s in pool]
    best_h, best_s = fallback.copy(), exact_score(fallback, n)
    for i in range(len(pool)):
        if scores[i] < best_s - 1e-12:
            best_h, best_s = pool[i].copy(), scores[i]
    order = np.argsort(scores)
    sel = [0]
    for idx in order:
        if len(sel) >= n_starts:
            break
        if int(idx) not in sel:
            sel.append(int(idx))
    taus = [0.01, 0.002, 0.0004, 0.00008, 0.000016, 0.00001]
    for si in sel:
        h0 = pool[si].copy()
        h_cur = h0.copy()
        try:
            h_cur = optimize_chain(h0, n, taus)
        except Exception:
            pass
        s_cur = exact_score(h_cur, n)
        if s_cur < best_s - 1e-12:
            best_h, best_s = h_cur.copy(), s_cur
        try:
            h2, s2 = subgrad_refine(h_cur, n, n_iter=n_passes)
        except Exception:
            h2, s2 = h_cur, s_cur
        if s2 < best_s - 1e-12:
            best_h, best_s = h2.copy(), s2
    try:
        h3, s3 = subgrad_refine(best_h, n, n_iter=n_passes)
        if s3 < best_s - 1e-12:
            best_h, best_s = h3.copy(), s3
    except Exception:
        pass
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    fb_s = exact_score(fallback, n)
    if fs > fb_s + 1e-12:
        final, fs = fallback, fb_s
    return final, float(fs), int(n)
