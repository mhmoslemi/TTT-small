def project_capped(z, n):
    if z.shape[0] != n:
        return None
    z = np.where(np.isfinite(z), z, 0.0).astype(np.float64)
    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx]); h[idx] += d; r -= d
        else:
            d = min(-r, h[idx]); h[idx] -= d; r += d

    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)):

    return float(np.max(c))

def helmert_fwd(x):
    d = len(x); n = d + 1
    if d == 0:
        return np.zeros(1)
    idx = np.arange(1, d + 1, dtype=float)
    a = x / np.sqrt(idx * (idx + 1.0))
    T = np.cumsum(a[::-1])[::-1]
    out = np.zeros(n); out[0] = T[0]
    ii = np.arange(1, n, dtype=float)
    tp = np.zeros(d + 1); tp[:d] = T
    out[1:] = tp[1:] - np.sqrt(ii / (ii + 1.0)) * x
    return out

def helmert_inv(v):
