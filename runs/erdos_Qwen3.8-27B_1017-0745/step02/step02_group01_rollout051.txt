import numpy as np
from scipy.optimize import linprog
import math

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-9:
        return False
    return True

def project_capped(z, n):
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    target = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - np.sum(h)
    for _ in range(3):
        if abs(r) < 1e-14:
            break
        for i in range(n):
            if abs(r) < 1e-14:
                break
            if r > 0:
                c = min(r, 1.0 - h[i])
                h[i] += c
                r -= c
            else:
                c = min(-r, h[i])
                h[i] -= c
                r += c
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    if abs(np.sum(h) - target) > 1e-10:
        return None
    return h

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    raw = np.correlate(h, 1.0 - h, mode='full')
    if raw.shape[0] != 2 * n - 1 or not np.all(np.isfinite(raw)):
        return float('inf')
    return float(np.max(raw) * (2.0 / n))

def build_lag_model(h, n):
    dx = 2.0 / n
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    L = 2 * n - 1
    A = np.zeros((L, n + 1))
    for k in range(L):
        d = k - (n - 1)
        p = max(d, 0)
        r = max(-d, 0)
        m = n - abs(d)
        A[k, p:p + m] += dx * (1.0 - h[r:r + m])
        A[k, r:r + m] -= dx * h[p:p + m]
        A[k, n] = -1.0
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(A)):
        return None, None
    return q, A

def lp_step(h, n, rho, q, A):
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    b_ub = -q
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = []
    for i in range(n):
        lo_b = max(-rho, -h[i])
        hi_b = min(rho, 1.0 - h[i])
        bounds.append((lo_b, hi_b))
    bounds.append((None, None))
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'maxiter': 2000,
                           'primal_feasibility_tolerance': 1e-10,
                           'dual_feasibility_tolerance': 1e-10})
    if res.status != 0 or res.x is None or res.x.shape[0] != n + 1:
        return None
    if not np.all(np.isfinite(res.x)):
        return None
    u = res.x[:n]
    if abs(np.sum(u)) > 1e-7:
        return None
    for i in range(n):
        if u[i] < -rho - 1e-9 or u[i] > rho + 1e-9:
            return None
        if h[i] + u[i] < -1e-9 or h[i] + u[i] > 1.0 + 1e-9:
            return None
    return u

def add_start(pool, z, n, limit):
    if len(pool) >= limit:
        return
    p = project_capped(z, n)
    if p is None:
        return
    for s in pool:
        if np.max(np.abs(p - s)) < 1e-8:
            return
    pool.append(p.copy())

    pool = []

    add_start(pool, fallback[::-1], n, 8)

    add_start(pool, np.full(n, 0.5), n, 8)

    add_start(pool, fallback + rng.normal(0, 0.05, n), n, 8)

    if len(pool) < 2:
        add_start(pool, rng.uniform(0, 1, n), n, 8)
