        hi = min(n, n-d)
        cnt = hi - lo
        # g_d has non-zeros in [lo,hi) and [lo+d, hi+d)
        # Each entry bounded by 2*dx in the overlap, dx elsewhere
        # Simple: 2*cnt entries, each <= 2*dx

import numpy as np

def project_capped(z, n):
    s = n / 2.0
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(80):
        mid = lo + 0.5 * (hi - lo)
        if mid == lo or mid == hi:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for i in range(n):
        if abs(r) < 1e-13:
            break
        if r > 0:
            a = min(r, 1.0 - h[i])
            h[i] += a; r -= a
        else:
            a = min(-r, h[i])
            h[i] -= a; r += a
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-8:
        return None
    return h

def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()

        return float('inf')
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return float('inf')
    if abs(float(np.sum(h)) - n / 2.0) > 1e-8:
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_values(x, n):
    dx = 2.0 / n
    m = 2 * n - 1
    q = np.zeros(m)
    for d in range(-(n - 1), n):
        lo, hi = max(0, -d), min(n, n - d)
        q[d + n - 1] = dx * float(np.dot(x[lo:hi], 1.0 - x[lo + d:hi + d]))
    return q

def dir_products(x, p, n):
    dx = 2.0 / n
    m = 2 * n - 1
    r = np.zeros(m)
    for d in range(-(n - 1), n):
        lo, hi = max(0, -d), min(n, n - d)
        r[d + n - 1] = dx * (float(np.dot(p[lo:hi], 1.0 - x[lo + d:hi + d])) - float(np.dot(x[lo:hi], p[lo + d:hi + d])))
    return r

def weighted_grad(x, w, n):
    dx = 2.0 / n
    g = np.zeros(n)
    for d in range(-(n - 1), n):
        wi = w[d + n - 1]
        if abs(wi) < 1e-16:
            continue
        lo, hi = max(0, -d), min(n, n - d)
        g[lo:hi] += dx * wi * (1.0 - x[lo + d:hi + d])
        g[lo + d:hi + d] -= dx * wi * x[lo:hi]
    return g

def proj_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    m = v.size
    if m == 0 or not np.all(np.isfinite(v)):
        return np.ones(m) / max(m, 1)
    u = np.sort(v)[::-1]
    cs = np.cumsum(u)
    idx = np.arange(1, m + 1, dtype=np.float64)
    mask = (u - (cs - 1.0) / idx) > 0
    j = max(1, int(np.max(np.where(mask)[0])) + 1) if np.any(mask) else 1
    lam = (cs[j - 1] - 1.0) / j
    w = np.maximum(v - lam, 0.0)
    r = 1.0 - float(np.sum(w))
    if abs(r) > 1e-12:
        ip = int(np.argmax(w))
        w[ip] = max(0.0, w[ip] + r)
    return w

def frob_sq(n):
    dx = 2.0 / n
    total = 0.0
    for d in range(-(n - 1), n):
        cnt = min(n, n - abs(d))
        total += 8.0 * cnt * dx * dx
    return total

def dual_solve(x, a, mu, w_init, n, K):
    m = 2 * n - 1
    L_D = max(1.05 * frob_sq(n) / mu, 1e-8)
    w = proj_simplex(w_init)
    z = w.copy()
    t = 1.0
    best_y, best_P = None, float('inf')
    for it in range(K):
        gbar = weighted_grad(x, z, n)
        y = project_capped(x - gbar / mu, n)
        if y is None:
            break
        p = y - x
        r = dir_products(x, p, n)
        gradD = a + r
        p2 = 0.5 * mu * float(np.dot(p, p))
        D_val = float(np.dot(z, a)) + float(np.dot(gbar, p)) + p2
        P_val = float(np.max(a + r)) + p2
        if np.isfinite(P_val) and P_val < best_P:
            best_P, best_y = P_val, y.copy()
        if np.isfinite(D_val) and np.isfinite(P_val):
            if max(0.0, P_val - D_val) <= 1e-9 * max(1.0, abs(P_val)):
                break
        w_new = proj_simplex(z + gradD / L_D)
        if it >= 2 and float(np.max(np.abs(w_new - w))) <= 1e-11:
            break
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = w_new + ((t - 1.0) / t_new) * (w_new - w)
        t = t_new
        w = w_new
    if best_y is None:
        gbar = weighted_grad(x, w, n)
        best_y = project_capped(x - gbar / mu, n)
    return best_y, w

            if y is None:
