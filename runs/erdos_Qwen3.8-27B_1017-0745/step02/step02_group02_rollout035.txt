    try_add(pool, np.clip(0.5 + 0.45 * np.cos(2.0 * np.pi * x), 0, 1), n, P)
    try_add(pool, np.clip(0.5 + 0.45 * np.cos(2.0 * np.pi * x + np.pi / 2), 0, 1), n, P)
    ns = np.array([0.04, 0.12, 0.30])
    for _ in range(20 * P):
        if len(pool) >= P:
            break
        if rng.random() < 0.25:
            c = rng.uniform(0, 1, n)
        else:
            base = pool[int(rng.integers(len(pool)))]
            c = base + rng.normal(0, ns[int(rng.integers(3))], n)
        try_add(pool, c, n, P)
    while len(pool) < P:
        pool.append(pool[int(rng.integers(len(pool)))].copy())
    return pool, P, G

def run(seed=42, **kwargs):
    n = 47
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999 and np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
            n = int(p.size)
            parent = p
    except Exception:
        pass
    if 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    fb = None
    if parent is not None:
        if parent.size == n:
            fb = project_capped(parent, n)
        else:
            old_n = parent.size
            old_x = (np.arange(old_n) + 0.5) / old_n
            new_x = (np.arange(n) + 0.5) / n
            fb = project_capped(np.interp(new_x, old_x, parent), n)
    const_fb = np.full(n, 0.5)
    if n == 1:
        return const_fb, exact_score(const_fb, 1), 1
    pp = parent if (parent is not None and parent.size == n) else None
    pool, P, G = build_pop(n, pp, rng)
    scores = [exact_score(pool[i], n) for i in range(P)]
    gi = int(np.argmin(scores))
    gb, gs = pool[gi].copy(), scores[gi]
    if fb is not None:
        sf = exact_score(fb, n)
        if sf < gs:
            gs, gb = sf, fb.copy()
    sc = exact_score(const_fb, n)
    if sc < gs:
        gs, gb = sc, const_fb.copy()
    if n <= 64:
        stag_len, max_restarts = 12, 2
    elif n <= 128:
        stag_len, max_restarts = 8, 1
    elif n <= 256:
        stag_len, max_restarts = 5, 1
    else:
        stag_len, max_restarts = 3, 0
    restart_scales = np.array([0.08, 0.20]) if n <= 64 else np.array([0.12])
    stag_count = 0
    restart_used = 0
    for gen in range(G):
        if stag_count >= stag_len and restart_used < max_restarts:
            n_imm = max(2, P // 4)
            n_imm = min(n_imm, P - 1)
            gi2 = int(np.argmin(scores))
            order = np.argsort(scores)[::-1]
            worst = [int(j) for j in order if int(j) != gi2][:n_imm]
            sigma = restart_scales[min(restart_used, len(restart_scales) - 1)]
            for slot in worst:
                if rng.random() < 0.667:
                    c = gb + sigma * rng.standard_normal(n)
                else:
                    c = rng.uniform(0, 1, n)
                pm = project_capped(c, n)
                if pm is not None:
                    pool[slot] = pm
                    scores[slot] = exact_score(pm, n)
            stag_count = 0
            restart_used += 1
            found = False
            for i in range(P):
                if np.max(np.abs(pool[i] - gb)) < 1e-14:
                    found = True
                    break
            if not found:
                wi = int(np.argmax(scores))
                if wi != gi2:
                    pool[wi] = gb.copy()
                    scores[wi] = gs
            continue
        if stag_count >= stag_len and restart_used >= max_restarts:
            break
        gen_gs = gs
        X = [pool[i].copy() for i in range(P)]
        f = scores[:]
        for i in range(P):
            donors = np.array([j for j in range(P) if j != i])
            ridx = rng.choice(P - 1, 3, replace=False)
            r1, r2, r3 = int(donors[ridx[0]]), int(donors[ridx[1]]), int(donors[ridx[2]])
            Fv = 0.45 + 0.45 * rng.random()
            v = X[r1] + Fv * (X[r2] - X[r3])
            mask = rng.random(n) < 0.85
            if not np.any(mask):
                mask[int(rng.integers(n))] = True
            trial = np.where(mask, v, X[i])
            pm = project_capped(trial, n)
            if pm is not None:
                s = exact_score(pm, n)
                if s < f[i] - 1e-12:
                    pool[i] = pm
                    scores[i] = s
                    if s < gs - 1e-12:
                        gs, gb = s, pm.copy()
                else:
                    pool[i] = X[i]
                    scores[i] = f[i]
            else:
                pool[i] = X[i]
                scores[i] = f[i]
        if gs < gen_gs - 1e-12:
            stag_count = 0
        else:
            stag_count += 1
    found_gi = False
    for i in range(P):
        if np.max(np.abs(pool[i] - gb)) < 1e-14:
            found_gi = True
            break
    if not found_gi:
        wi = int(np.argmax(scores))
        pool[wi] = gb.copy()
        scores[wi] = gs
    final = project_capped(gb, n)
    if final is None:
        final = gb
    sf = exact_score(final, n)
    cands = [(sf, final)]
    if fb is not None:
        cands.append((exact_score(fb, n), fb))
    cands.append((exact_score(const_fb, n), const_fb))
    best_i = 0
    for i in range(1, len(cands)):
        if cands[i][0] < cands[best_i][0]:
            best_i = i
    bh, bs = cands[best_i]
    if not is_feasible(bh, n):
        bh, bs = const_fb, exact_score(const_fb, n)
    return bh, float(bs), int(n)
