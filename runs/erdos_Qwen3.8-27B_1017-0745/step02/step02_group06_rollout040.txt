import numpy as np
import math

def project_capped(z, n):
    z = np.where(np.isfinite(z), z, 0.0).astype(np.float64)
    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for i in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[i])
            h[i] += d; r -= d
        else:
            d = min(-r, h[i])
            h[i] -= d; r += d
    if abs(r) > 1e-10 * max(1, n) or not np.all(np.isfinite(h)):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if c.shape != (2 * n - 1,) or not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def helmert_fwd(x):
    n = len(x) + 1
    d = len(x)
    a = x / np.sqrt(np.arange(1, d + 1) * np.arange(2, d + 2))
    T = np.zeros(d + 1)
    T[:d] = np.cumsum(a[::-1])[::-1]
    out = np.zeros(n)
    out[0] = T[0]
    for i in range(1, n):
        out[i] = T[i] - i * x[i - 1] / math.sqrt(i * (i + 1))
    return out

def helmert_bwd(v):
    n = len(v)
    d = n - 1
    P = np.cumsum(v)
    out = np.zeros(d)
    for j in range(d):
        out[j] = (P[j] - (j + 1) * v[j + 1]) / math.sqrt((j + 1) * (j + 2))
    return out

import numpy as np
import math

def project_capped(z, n):
    z = np.where(np.isfinite(z), z, 0.0).astype(np.float64)
    if z.shape != (n,):
        return None
    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for i in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[i])
            h[i] += d
            r -= d
        else:
            d = min(-r, h[i])
            h[i] -= d
            r += d
    if abs(r) > 1e-10 * max(1, n) or not np.all(np.isfinite(h)):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if c.shape[0] != 2 * n - 1 or not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def helmert_fwd(x):
    d = len(x)
    n = d + 1
    if d == 0:
        return np.zeros(1)
    a = x / np.sqrt(np.arange(1, d + 1, dtype=np.float64) * np.arange(2, d + 2, dtype=np.float64))
    T = np.zeros(d + 1)
    T[:d] = np.cumsum(a[::-1])[::-1]
    out = np.zeros(n)
    out[0] = T[0]
    for i in range(1, n):
        out[i] = T[i] - i * x[i - 1] / math.sqrt(i * (i + 1))
    return out

def helmert_bwd(v):
    n = len(v)
    d = n - 1
    if d == 0:
        return np.zeros(0)
    P = np.cumsum(v)
    out = np.zeros(d)
    for j in range(d):
        out[j] = (P[j] - (j + 1) * v[j + 1]) / math.sqrt((j + 1) * (j + 2))
    return out

def cem_run(h_start, n, lam, gens, stall_lim, full_cov, rng, gbest, gscore):
    d = n - 1
    if d == 0:
        return gbest, gscore
    u = np.full(n, 0.5)
    m = helmert_bwd(h_start - u)
    sigma0 = 0.15
    eta = 0.35
    smin, smax = 0.02, 0.5
    mu = lam // 2
    # Initialize covariance
    if full_cov:
        C = sigma0 * sigma0 * np.eye(d)
    else:
        vdiag = np.full(d, sigma0 * sigma0)
    run_best = gscore
    stall = 0
    for gen in range(gens):
        # Factor covariance
        if full_cov:
            try:
                Cc = 0.5 * (C + C.T)
                eigvals, eigvecs = np.linalg.eigh(Cc)
                eigvals = np.clip(eigvals, smin * smin, smax * smax)
                if not np.all(np.isfinite(eigvals)) or not np.all(np.isfinite(eigvecs)):
                    C = sigma0 * sigma0 * np.eye(d)
                    continue
                L = eigvecs @ np.diag(np.sqrt(eigvals))
            except Exception:
                C = sigma0 * sigma0 * np.eye(d)
                continue
        else:
            vdiag = np.clip(vdiag, smin * smin, smax * smax)
        half = lam // 2
        h_list = []
        x_list = []
        scores = []
        for _ in range(half):
            z = rng.standard_normal(d)
            for sign in (1.0, -1.0):
                if full_cov:
                    x = m + sign * (L @ z)
                else:
                    x = m + sign * (np.sqrt(vdiag) * z)
                h_p = u + helmert_fwd(x)
                h_proj = project_capped(h_p, n)
                if h_proj is None:
                    continue
                xp = helmert_bwd(h_proj - u)
                sc = exact_score(h_proj, n)
                if sc == float('inf'):
                    continue
                h_list.append(h_proj)
                x_list.append(xp)
                scores.append(sc)
                if sc < run_best - 1e-12:
                    run_best = sc
                    stall = 0
                if sc < gscore - 1e-12:
                    gscore = sc
                    gbest = h_proj.copy()
        if len(h_list) == 0:
            break
        scores_arr = np.array(scores)
        order = np.argsort(scores_arr)
        mu_eff = min(mu, len(order))
        # Elite weights
        a = np.array([math.log(mu_eff + 0.5) - math.log(r + 1) for r in range(mu_eff)])
        w = a / a.sum()
        # Elite mean
        X_elite = np.array([x_list[order[r]] for r in range(mu_eff)])
        m_elite = (w[:, None] * X_elite).sum(axis=0)
        m_new = (1 - eta) * m + eta * m_elite
        # Project mean back
        h_m = u + helmert_fwd(m_new)
        h_mp = project_capped(h_m, n)
        if h_mp is not None and np.all(np.isfinite(h_mp)):
            m_new = helmert_bwd(h_mp - u)
        else:
            break
        # Update covariance
        if full_cov:
            D = X_elite - m_new
            S = (w[:, None] * D * w[:, None]).T @ D
            C = (1 - eta) * C + eta * S
            C = 0.5 * (C + C.T)
        else:
            D = X_elite - m_new
            S = (w[:, None] * D ** 2).sum(axis=0)
            vdiag = np.clip((1 - eta) * vdiag + eta * S, smin * smin, smax * smax)
        m = m_new
        if stall >= stall_lim:
            break
        if not np.any(np.isfinite(m)):
            break
        stall += 1
    return gbest, gscore

def build_starts(parent, n, rng):
    u = np.full(n, 0.5)
    pool = []
    def add_entry(h_val, sigma):
        if h_val is None or not np.all(np.isfinite(h_val)):
            return
        for p in pool:
            if np.max(np.abs(h_val - p[0])) < 1e-7:
                return
        pool.append((h_val, sigma))
    # Parent and symmetries
    if parent is not None:
        add_entry(parent, 0.10)
        add_entry(parent[::-1].copy(), 0.10)
        add_entry(1.0 - parent[::-1].copy(), 0.10)
        add_entry(1.0 - parent.copy(), 0.10)
    add_entry(u.copy(), 0.18)
    # Block
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    add_entry(blk, 0.28)
    # Alternating
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    add_entry(alt, 0.28)
    # Random balanced
    perm = rng.permutation(n)
    rb = np.zeros(n)
    rb[perm[:n // 2]] = 1.0
    if n % 2:
        rb[perm[n // 2]] = 0.5
    add_entry(rb, 0.28)
    # Gaussian perturbations
    base = parent if parent is not None else u
    for sd in (0.06, 0.15, 0.30):
        p = project_capped(base + rng.normal(0, sd, n), n)
        if p is not None:
            add_entry(p, 0.12)
    # Uniform random
    p = project_capped(rng.uniform(0, 1, n), n)
    if p is not None:
        add_entry(p, 0.20)
    return pool

def project_capped(z, n):

        else:

            break

            d = min(r, 1.0 - h[i]); h[i] += d; r -= d

        return np.zeros(1)
    a = x / np.sqrt(np.arange(1, d + 1, dtype=np.float64) * np.arange(2, d + 2, dtype=np.float64))
    T = np.zeros(d + 1)
    T[:d] = np.cumsum(a[::-1])[::-1]

    out[0] = T[0]

        out[i] = T[i] - i * x[i - 1] / math.sqrt(i * (i + 1))
    return out
