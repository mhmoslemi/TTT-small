        _try_start(starts, 1.0 - fb, n)
        _try_start(starts, np.roll(fb, n // 3), n)
        _try_start(starts, np.roll(fb, -n // 3), n)
        _try_start(starts, fb + rng.normal(0, 0.04, n), n)
        _try_start(starts, fb + rng.normal(0, 0.10, n), n)
        _try_start(starts, fb + rng.normal(0, 0.07, n), n)
        _try_start(starts, rng.uniform(0, 1, n), n)
    else:
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        _try_start(starts, blk, n)
        _try_start(starts, np.where(np.arange(n) % 2 == 0, 1.0, 0.0), n)
        tt = np.arange(n)
        _try_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * tt / n), n)
        _try_start(starts, 0.5 + 0.45 * np.cos(4.0 * np.pi * tt / n), n)
        _try_start(starts, rng.uniform(0, 1, n), n)
        _try_start(starts, fb + rng.normal(0, 0.05, n), n)
    starts = starts[:n_starts]
    best, best_score = fb.copy(), fb_score
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_score - 1e-12:
            best, best_score = st.copy(), ss
    if max_iter > 0:
        for st in starts:
            try:
                hin, hs = trust_region_search(st, n, max_iter)
                if np.isfinite(hs) and hs < best_score - 1e-12:
                    best, best_score = hin.copy(), hs
            except Exception:
                pass
        n_hops = 3 if n <= 64 else (2 if n <= 128 else 1)
        hop_sds = [0.025, 0.06, 0.10] if n <= 128 else [0.05]
        for hop_i in range(min(n_hops, len(hop_sds))):
            try:
                pert = best + rng.normal(0, hop_sds[hop_i], n)
                pv = project_capped(pert, n)
                if pv is None:
                    continue
                ps = exact_score(pv, n)
                if np.isfinite(ps) and ps < best_score - 1e-12:
                    best, best_score = pv.copy(), ps
                hin, hs = trust_region_search(pv, n, max_iter)
                if np.isfinite(hs) and hs < best_score - 1e-12:
                    best, best_score = hin.copy(), hs
            except Exception:
                pass
    final = project_capped(best, n)
    if final is None:
        final = fb.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final, fs = fb.copy(), fb_score
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final, float(fs), int(n)
