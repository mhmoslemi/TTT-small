            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None: n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999: n = v
        except Exception:
            pass
    if parent is not None:
        pr = project_capped(parent, n)
        fall = pr if pr is not None else np.full(n, 0.5)
    else:
        fall = np.full(n, 0.5)
    if not is_feasible(fall, n):
        fall = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        return fall, float(exact_score(fall, n)), n
    P = min(64, max(20, int(round(64.0 * math.sqrt(47.0 / n)))))
    B = max(2 * P, min(6000, int(6000 * (47.0 / n) ** 2)))
    pop = build_pop(fall, n, P, rng)
    bh, bs = de_evolve(pop, n, B, rng)
    h1, s1 = subgrad_polish(bh, n, 40)
    if s1 < bs - 1e-12: bh, bs = h1, s1
    h2, s2 = slsq_polish(bh, n)
    if s2 < bs - 1e-12: bh, bs = h2, s2
    h3, s3 = pairwise_swap(bh, n, 3)
    if s3 < bs - 1e-12: bh, bs = h3, s3
    h4, s4 = subgrad_polish(bh, n, 20)
    if s4 < bs - 1e-12: bh, bs = h4, s4
    fin = project_capped(bh, n)
    if fin is None or not is_feasible(fin, n):
        fin = fall
    fs = exact_score(fin, n)
    fb = exact_score(fall, n)
    if fs > fb + 1e-12:
        fin, fs = fall, fb
    return fin, float(fs), int(n)
