import numpy as np
from scipy.optimize import minimize

def project_capped(z, n):
    z = np.where(np.isfinite(z), z, 0.0).astype(np.float64)
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-15:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx]); h[idx] += d; r -= d
        else:
            d = min(-r, h[idx]); h[idx] -= d; r += d
    if abs(r) > 1e-10 * max(1, n) or not np.all(np.isfinite(h)):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def lag_jac(h, n):
    dx = 2.0 / n
    J = np.zeros((2 * n - 1, n))
    for d in range(n - 1, -(n), -1):
        row = d + n - 1
        if d >= 0:
            i_s = np.arange(0, n - d)
        else:
            i_s = np.arange(-d, n)
        if len(i_s) == 0:
            continue
        j_s = i_s + d
        np.add.at(J[row], i_s, dx * (1.0 - h[j_s]))
        np.add.at(J[row], j_s, -dx * h[i_s])
    return J
