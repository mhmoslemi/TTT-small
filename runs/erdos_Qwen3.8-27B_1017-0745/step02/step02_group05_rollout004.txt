        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size >= 1 and p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        if 'n_points' in kwargs:
            try:
                v = int(kwargs['n_points'])
                if 1 <= v <= 999:
                    n = v
            except Exception:
                pass
        if n is None:
            n = 47
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        return np.array([0.5]), 0.5, 1
    fb = None
    if parent is not None:
        fb = project_capped(parent, n)
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    fb_score = exact_score(fb, n)
    if n <= 96:
        n_starts, taus, maxiter = 6, [1e-2, 2e-3, 4e-4, 8e-5, 1e-5, 2e-6], 100
    elif n <= 256:
        n_starts, taus, maxiter = 4, [5e-3, 5e-4, 5e-5, 1e-5], 70
    else:
        n_starts, taus, maxiter = 2, [2e-3, 1e-4, 1e-5], 40
    starts = build_starts(fb, n, rng, n_starts)
    gb = fb.copy()
    gs = fb_score
    for st in starts:
        ss = exact_score(st, n)
        if ss < gs - 1e-12:
            gs, gb = ss, st.copy()
    for st in starts:
        try:
            xo = optimize_start(st, n, taus, maxiter)
            if xo is not None and is_feasible(xo, n):
                so = exact_score(xo, n)
                if so < gs - 1e-12:
                    gs, gb = so, xo.copy()
        except Exception:
            pass
    try:
        xo2 = optimize_start(gb, n, [2e-6, 5e-7], 60)
        if xo2 is not None and is_feasible(xo2, n):
            so2 = exact_score(xo2, n)
            if so2 < gs - 1e-12:
                gs, gb = so2, xo2.copy()
    except Exception:
        pass
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb.copy()
    fs = exact_score(final, n)
    if fs > fb_score + 1e-12:
        final = fb.copy()
        fs = fb_score
    return final.astype(np.float64), float(fs), int(n)
