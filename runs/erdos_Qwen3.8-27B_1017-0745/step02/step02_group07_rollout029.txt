import numpy as np
import math

def project_capped(z, n):
    z = np.clip(np.asarray(z, dtype=np.float64), -1e6, 1e6)
    s = n / 2.0
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - h.sum()
    for _ in range(n + 4):
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + r, 0.0)
        r = s - h.sum()
    if not np.all(np.isfinite(h)) or h.min() < -1e-14 or h.max() > 1.0 + 1e-14:
        return None
    if abs(h.sum() - s) > 1e-9:
        return None
    return h

    dx = 2.0 / n
    L = 2 * n - 1
    b = np.zeros(L)
    # zero lag: index n-1
    b[n-1] = dx * np.sum(d * (1.0 - 2.0 * h))
    for r in range(1, n):

        h0, h1 = h[:m], h[r:]

        # positive lag r: index (n-1) + r
        b[n-1+r] = dx * np.sum(d1 * (1.0 - h0) - h1 * d0)

        b[n-1-r] = dx * np.sum(d0 * (1.0 - h1) - h0 * d1)
    return b

import numpy as np
import math

def project_capped(z, n):
    z = np.clip(np.asarray(z, dtype=np.float64).ravel(), -1e6, 1e6)
    s = n / 2.0
    lo = float(z.min()) - 1.0
    hi = float(z.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - h.sum()
    for _ in range(n + 4):
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + r, 0.0)
        r = s - h.sum()
    if not np.all(np.isfinite(h)) or h.min() < -1e-14 or h.max() > 1.0 + 1e-14:
        return None
    if abs(h.sum() - s) > 1e-9:
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if h.min() < -1e-12 or h.max() > 1.0 + 1e-12:
        return False
    return abs(h.sum() - n / 2.0) <= 1e-9

def exact_score(h, n):
    try:
        if not is_feasible(h, n):
            return float('inf')
        v = float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))
        return v if np.isfinite(v) else float('inf')
    except Exception:
        return float('inf')

def lag_coeffs(h, d, n):
    dx = 2.0 / n
    L = 2 * n - 1
    a = np.zeros(L); b = np.zeros(L); c = np.zeros(L)
    a[n-1] = np.sum(h * (1.0 - h))
    b[n-1] = np.sum(d * (1.0 - 2.0 * h))
    c[n-1] = -np.sum(d * d)
    for s in range(1, n):
        m = n - s
        h0, h1 = h[:m], h[s:]
        d0, d1 = d[:m], d[s:]
        a[n-1+s] = np.sum(h1 * (1.0 - h0))
        b[n-1+s] = np.sum(d1 * (1.0 - h0) - h1 * d0)
        c[n-1+s] = -np.sum(d1 * d0)
        a[n-1-s] = np.sum(h0 * (1.0 - h1))
        b[n-1-s] = np.sum(d0 * (1.0 - h1) - h0 * d1)
        c[n-1-s] = -np.sum(d0 * d1)
    return dx * a, dx * b, dx * c

def feasible_interval(h, d):
    nz = np.nonzero(np.abs(d) > 1e-15)[0]
    if len(nz) == 0:
        return None
    lo = 0.0; hi = 0.0
    for i in nz:
        if d[i] > 0:
            lo = max(lo, -h[i] / d[i])
            hi = min(hi, (1.0 - h[i]) / d[i])
        else:
            lo = max(lo, (1.0 - h[i]) / d[i])
            hi = min(hi, -h[i] / d[i])
    if not np.isfinite(lo) or not np.isfinite(hi) or hi - lo < 1e-12:
        return None
    return (lo, hi)

def line_search(a, b, c, lo, hi):
    L = len(a)
    cands = [lo, hi, 0.0]
    for j in range(L):
        if abs(c[j]) > 1e-15:
            t = -b[j] / (2.0 * c[j])
            if lo - 1e-10 <= t <= hi + 1e-10:
                cands.append(np.clip(t, lo, hi))
    for i in range(L):
        for j in range(i + 1, L):
            A = c[i] - c[j]; B = b[i] - b[j]; C = a[i] - a[j]
            sc = max(1.0, abs(A), abs(B), abs(C))
            if abs(A) < 1e-10 * sc:
                if abs(B) > 1e-10 * sc:
                    t = -C / B
                    if lo - 1e-10 <= t <= hi + 1e-10:
                        cands.append(np.clip(t, lo, hi))
            else:
                disc = B * B - 4.0 * A * C
                if disc < 0:
                    if disc > -1e-12 * (1.0 + B * B + abs(4.0 * A * C)):
                        disc = 0.0
                    else:
                        continue
                sq = math.sqrt(max(0.0, disc))
                if abs(B) > 1e-15:
                    q = -0.5 * (B + math.copysign(sq, B))
                else:
                    q = -0.5 * sq
                if abs(q) > 1e-15:
                    t1 = q / A; t2 = C / q
                else:
                    t1 = t2 = -B / (2.0 * A)
                for t in (t1, t2):
                    if np.isfinite(t) and lo - 1e-10 <= t <= hi + 1e-10:
                        cands.append(np.clip(t, lo, hi))
    cands = np.array(cands, dtype=np.float64)
    best_t = float(cands[0]); best_v = float('inf')
    for cs in range(0, len(cands), 512):
        ch = cands[cs:cs + 512]
        if len(ch) == 0:
            break
        vals = a[None, :] + b[None, :] * ch[:, None] + c[None, :] * ch[:, None] * ch[:, None]
        env = vals.max(axis=1)
        il = int(np.argmin(env))
        if float(env[il]) < best_v:
            best_v = float(env[il]); best_t = float(ch[il])
    return best_t, best_v

def compute_grads(h, n, active_idx):
    L = 2 * n - 1
    dx = 2.0 / n
    G = np.zeros((len(active_idx), n))
    for k, r_idx in enumerate(active_idx):
        s = r_idx - (n - 1)
        g = np.zeros(n)
        if s == 0:
            g = dx * (1.0 - 2.0 * h)
        elif s > 0:
            m = n - s
            g[:m] += dx * (-h[s:])
            g[s:] += dx * (1.0 - h[:m])
        else:
            r = -s
            m = n - r
            g[:m] += dx * (1.0 - h[r:])
            g[r:] += dx * (-h[:m])
        G[k] = g
    return G

def make_start_pool(fb, n, rng):
    pool = [fb.copy()]
    def ta(v):
        if v is None:
            return
        for s in pool:
            if np.max(np.abs(v - s)) < 1e-9:
                return
        pool.append(v)
    ta(project_capped(fb[::-1], n))
    ta(project_capped(1.0 - fb, n))
    ta(project_capped(1.0 - fb[::-1], n))
    ta(project_capped(np.roll(fb, n // 3), n))
    ta(project_capped(np.roll(fb, n // 2), n))
    for sd in (0.06, 0.18):
        ta(project_capped(fb + sd * rng.standard_normal(n), n))
    idx = np.arange(n, dtype=np.float64)
    for k in (1, 2, 3):
        for ph in (0.0, math.pi / 2):
            ta(project_capped(0.5 + 0.4 * np.cos(2.0 * math.pi * k * idx / n + ph), n))
    return pool[:16]

def local_search(h, n, rng, sweeps, active_k, n_pairs):
    h = h.copy()
    sc = exact_score(h, n)
    for sw in range(sweeps):
        lags = np.asarray(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n))
        L = len(lags)
        k_act = min(active_k, L)
        ai = np.argsort(lags)[-k_act:]
        G = compute_grads(h, n, ai)
        psc = []
        for i in range(n - 1):
            gi = G[:, i]
            for j in range(i + 1, n):
                gj = G[:, j]
                ds = float(np.max(gi - gj))
                lo_v = max(-h[i], h[j] - 1.0)
                hi_v = min(1.0 - h[i], h[j])
                if hi_v - lo_v < 1e-12:
                    continue
                psc.append(ds)
                psc.append(i)
                psc.append(j)
        pp = np.array(psc, dtype=np.float64).reshape(-1, 3)
        if len(pp) > 0:
            order = np.argsort(pp[:, 0])
            top_pp = pp[order[:n_pairs]]
        else:
            top_pp = np.zeros((0, 3))
        rnd_pp = []
        for _ in range(4):
            i = int(rng.integers(n)); j = int(rng.integers(n))
            if i == j:
                continue
            if i > j:
                i, j = j, i
            if min(1.0 - h[i], h[j]) - max(-h[i], h[j] - 1.0) > 1e-12:
                rnd_pp.append([0.0, float(i), float(j)])
        if len(rnd_pp) > 0:
            top_pp = np.vstack([top_pp, np.array(rnd_pp)])
        best_new = None; best_sc = sc
        for row in top_pp:
            i = int(row[1]); j = int(row[2])
            d = np.zeros(n); d[i] = 1.0; d[j] = -1.0
            iv = feasible_interval(h, d)
            if iv is None:
                continue
            lo, hi = iv
            aa, bb, cc = lag_coeffs(h, d, n)
            t, pred = line_search(aa, bb, cc, lo, hi)
            if abs(t) < 1e-12:
                continue
            trial = project_capped(h + t * d, n)
            if trial is None:
                continue
            ts = exact_score(trial, n)
            if ts < best_sc - 1e-12:
                best_sc = ts; best_new = trial
        if best_new is not None:
            h = best_new; sc = best_sc
        else:
            sv = 0.08 + 0.08 * sw
            kick = project_capped(h + sv * rng.standard_normal(n), n)
            if kick is not None:
                ks = exact_score(kick, n)
                if ks < sc - 1e-12:
                    h = kick; sc = ks
    return h, sc
