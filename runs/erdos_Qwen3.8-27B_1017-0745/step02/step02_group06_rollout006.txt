    z = np.where(np.isfinite(z), z, 0.0).astype(np.float64)
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0

    for _ in range(60):

        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx]); h[idx] += d; r -= d
        else:
            d = min(-r, h[idx]); h[idx] -= d; r += d
    if abs(r) > 1e-10 * max(1, n) or not np.all(np.isfinite(h)):
        return None
    return np.clip(h, 0.0, 1.0)

    return float(np.max(c))

            j = i + k
            J[k + n - 1, j] += dx * (1.0 - h[i])
            J[k + n - 1, i] -= dx * h[j]
    return J

            pert = project_capped(gb + rng.normal(0, sigma, n), n)
            if pert is None or exact_score(pert, n) > 0.5 + 1e-12:
                continue
            if np.max(np.abs(pert - gb)) < 1e-6:
                continue
            hp = try_solve(pert, n, maxiter)
            if hp is not None:
                cands2 = [(exact_score(hp, n), hp)]
                for a in [0.5, 0.75]:
                    blend = project_capped(a * hp + (1.0 - a) * gb, n)
                    if blend is not None:
                        cands2.append((exact_score(blend, n), blend))
                for sc, hc in cands2:
                    if sc < gs - 1e-12:
                        gs, gb = sc, hc.copy()
                        improved = True
            break
        if improved:
            sigma = max(0.01, 0.7 * sigma)
        else:
            sigma = min(0.20, 1.5 * sigma)

    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = parent
    fs = exact_score(final, n)
    ps = exact_score(parent, n)
    if ps < fs - 1e-12:
        final, fs = parent, ps
    uf = np.full(n, 0.5)
    us = exact_score(uf, n)
    if us < fs - 1e-12:
        final, fs = uf, us
    fs = exact_score(final, n)
    return final, float(fs), int(n)
