    if n <= 2:
        return fallback, fb_score, int(n)
    E = min(50000, max(48, int(150000000 / (n * n))))
    P = min(192, max(24, min(4 * n, E // 2)))
    if n <= 64:
        gen_cap = 160
    elif n <= 128:
        gen_cap = 60
    elif n <= 256:
        gen_cap = 18
    else:
        gen_cap = 4
    gen_limit = min(gen_cap, max(0, E // P - 1))
    try:
        P = max(24, min(int(kwargs.get('population', P)), 256, E // 2))
    except Exception:
        pass
    try:
        gen_limit = max(0, min(int(kwargs.get('generations', gen_limit)), gen_cap, E // P - 1))
    except Exception:
        pass
    pop = np.zeros((P, n))
    scores = np.full(P, float('inf'))
    pop[0] = fallback
    scores[0] = fb_score
    starts = [fallback]
    if parent is not None:
        for sfx in (fallback[::-1], 1.0 - fallback, np.roll(fallback, max(1, n // 3))):
            pr = project_capped(sfx.astype(np.float64), n)
            if pr is not None:
                starts.append(pr)
    else:
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        starts.append(project_capped(blk, n) if project_capped(blk, n) is not None else fallback)
        alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
        ap = project_capped(alt, n)
        if ap is not None:
            starts.append(ap)
        tt = np.arange(n)
        for k in (1, 2):
            cp = project_capped(0.5 + 0.45 * np.cos(2.0 * np.pi * k * tt / n), n)
            if cp is not None:
                starts.append(cp)
    eval_count = 0
    for idx in range(1, P):
        h_c = None
        if idx < len(starts) and starts[idx] is not None:
            h_c = starts[idx]
        else:
            u = rng.random()
            if parent is not None and u < 0.6:
                sig = 10.0 ** rng.uniform(math.log10(0.005), math.log10(0.25))
                g = rng.standard_normal(n)
                g -= g.mean()
                h_c = project_capped(parent + sig * g, n)
            elif parent is not None and u < 0.8:
                h_c = project_capped(rng.random(n), n)
            else:
                h_c = project_capped(rng.beta(0.7, 0.7, n), n)
        if h_c is None:
            h_c = fallback.copy()
        dup = False
        for j in range(idx):
            if float(np.max(np.abs(h_c - pop[j]))) < 1e-9:
                dup = True
                break
        if dup:
            h_c = project_capped(rng.random(n) * 0.5 + fallback * 0.5, n)
            if h_c is None:
                h_c = fallback.copy()
        pop[idx] = h_c
        scores[idx] = exact_score(h_c, n)
        eval_count += 1
    P_act = int(np.sum(np.isfinite(scores)))
    if P_act < 3:
        for idx in range(3 - P_act):
            hc = project_capped(rng.random(n), n)
            if hc is None:
                hc = uniform.copy()
            pop[P_act + idx] = hc
            scores[P_act + idx] = exact_score(hc, n)
            eval_count += 1
        P_act = 3
    P_act = min(P, P_act)
    pop, scores = pop[:P_act], scores[:P_act]
    P_act = len(scores)
    best = pop[int(np.argmin(scores))].copy()
    best_score = float(np.min(scores))
    mu_F, mu_CR = 0.55, 0.85
    stagnation = 0
    n_restarts = 0
    for gen in range(gen_limit):
        if eval_count >= E:
            break
        ranked = np.argsort(scores)
        k_pb = max(1, int(math.ceil(0.2 * P_act)))
        F_succ, CR_succ = [], []
        stall_this_gen = 0
        for i in range(P_act):
            if eval_count >= E:
                break
            ppool = [int(x) for x in ranked[:k_pb] if x != i]
            if not ppool:
                ppool = [int(ranked[0])]
            pb = ppool[int(rng.integers(len(ppool)))]
            excl = {i, pb}
            r1p = [j for j in range(P_act) if j not in excl]
            if not r1p:
                continue
            r1 = r1p[int(rng.integers(len(r1p)))]
            excl.add(r1)
            r2p = [j for j in range(P_act) if j not in excl]
            r2 = r2p[int(rng.integers(len(r2p)))] if r2p else int(rng.integers(P_act))
            uu = rng.random()
            F = mu_F + 0.1 * math.tan(math.pi * (uu - 0.5))
            F = max(0.05, min(1.0, F))
            CR = mu_CR + 0.1 * rng.standard_normal()
            CR = max(0.0, min(1.0, CR))
            donor = pop[i] + F * (pop[pb] - pop[i]) + F * (pop[r1] - pop[r2])
            mask = rng.random(n) < CR
            mask[int(rng.integers(n))] = True
            trial = pop[i].copy()
            trial[mask] = donor[mask]
            th = project_capped(trial, n)
            if th is None:
                continue
            sc = exact_score(th, n)
            eval_count += 1
            if sc < scores[i] - 1e-12:
                F_succ.append(F)
                CR_succ.append(CR)
                pop[i] = th
                scores[i] = sc
                if sc < best_score - 1e-12:
                    best = th.copy()
                    best_score = sc
                    stall_this_gen = 0
                else:
                    stall_this_gen += 1
            else:
                stall_this_gen += 1
        if F_succ:
            mu_CR = 0.9 * mu_CR + 0.1 * float(np.mean(CR_succ))
            fs_sum = float(np.sum(F_succ))
            if fs_sum > 1e-15:
                mu_F = 0.9 * mu_F + 0.1 * float(np.sum(np.array(F_succ) ** 2)) / fs_sum
        stagnation = stall_this_gen
        if stagnation >= 20 and n_restarts < 2 and eval_count < E:
            n_restarts += 1
            best_row = int(np.argmin(scores))
            n_rep = max(1, P_act // 4)
            to_rep = [j for j in range(P_act) if j != best_row][-n_rep:]
            sig_r = 0.12 if n_restarts == 1 else 0.04
            for j in to_rep:
                if eval_count >= E:
                    break
                g = rng.standard_normal(n)
                g -= g.mean()
                hc = project_capped(best + sig_r * g, n)
                if hc is None:
                    hc = project_capped(rng.random(n), n)
                if hc is None:
                    hc = uniform.copy()
                pop[j] = hc
                scores[j] = exact_score(hc, n)
                eval_count += 1
            mu_F = 0.65 if n_restarts == 1 else 0.50
            mu_CR = 0.85
            stagnation = 0
    # Local refinement
    if n <= 80 and gen_limit > 0:
        polished, poly_sc = _pairwise_hill(best, n, 2)
        fp = project_capped(polished, n)
        if fp is not None:
            fs2 = exact_score(fp, n)
            if np.isfinite(fs2) and fs2 < best_score - 1e-13:
                best, best_score = fp, fs2
    final = project_capped(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final, fs = fallback.copy(), fb_score
    return final, float(fs), int(n)
