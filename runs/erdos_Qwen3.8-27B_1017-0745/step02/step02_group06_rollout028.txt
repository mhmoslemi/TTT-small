import numpy as np
import math

def project_capped(z, n):
    z = np.where(np.isfinite(z), z, 0.0).astype(float)
    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx])
            h[idx] += d; r -= d
        else:
            d = min(-r, h[idx])
            h[idx] -= d; r += d
    if abs(r) > 1e-9 * n or not np.all(np.isfinite(h)):
        return None
    return h

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(np.sum(h) - n / 2.0) <= 1e-9 * n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if c.shape[0] != 2 * n - 1 or not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def helmert_fwd(x, n):
    d = n - 1
    a = x / np.sqrt(np.arange(1, d + 1).astype(float) * (np.arange(2, d + 2).astype(float)))
    T = np.zeros(d + 1)
    for i in range(d - 1, -1, -1):
        T[i] = T[i + 1] + a[i]
    out = np.empty(n)
    out[0] = T[0]
    for i in range(1, n):
        if i < d:
            out[i] = T[i] - i * x[i - 1] / math.sqrt(i * (i + 1))
        else:
            out[i] = -x[d - 1] * math.sqrt(d / (d + 1))
    return out

def helmert_bwd(v, n):
    d = n - 1
    ps = np.zeros(n)
    for i in range(1, n):
        ps[i] = ps[i - 1] + v[i]
    out = np.empty(d)
    for j in range(d):
        out[j] = (ps[j] - (j + 1) * v[j + 1]) / math.sqrt((j + 1) * (j + 2))
    return out

def cem(h0, n, lam, gens, stall_lim, rng, gbest, gscore):
    d = n - 1
    if d < 1:
        return gbest, gscore
    u = np.full(n, 0.5)
    m = helmert_bwd(h0 - u, n)
    full_cov = d <= 64
    if full_cov:
        C = 0.01 * np.eye(d)  # sigma=0.1
    else:
        C = np.full(d, 0.01)
    eta = 0.35
    smax = 0.5
    smin = 0.02
    mu = lam // 2
    run_best_s = gscore
    stall = 0
    for gen in range(gens):
        # Sample
        zs = rng.standard_normal((lam // 2, d))
        if full_cov:
            try:
                w, V = np.linalg.eigh(0.5 * (C + C.T))
                w = np.clip(w, smin**2, smax**2)
                L = V @ np.diag(np.sqrt(np.maximum(w, 0)))
            except:
                C = 0.01 * np.eye(d)
                w, V = np.linalg.eigh(C)
                w = np.clip(w, smin**2, smax**2)
                L = V @ np.diag(np.sqrt(np.maximum(w, 0)))
        else:
            sv = np.sqrt(np.clip(C, smin**2, smax**2))
        cands = []  # (score, x_proj, h)
        for ii in range(lam // 2):
            for sign in (1, -1):
                x = m + sign * (L @ zs[ii])
                h = project_capped(u + helmert_fwd(x, n), n)
                if h is None:
                    continue
                xp = helmert_bwd(h - u, n)
                sc = exact_score(h, n)
                if sc < float('inf'):
                    cands.append((sc, xp, h))
                    if sc < gscore - 1e-12:
                        gbest, gscore = h.copy(), sc
        if not cands:
            break
        cands.sort(key=lambda t: t[0])
        mu_eff = min(mu, len(cands))
        # Weights
        ws = np.zeros(mu_eff)
        sw = 0.0
        for r in range(mu_eff):
            ws[r] = math.log(mu_eff + 0.5) - math.log(r + 1)
            sw += ws[r]
        ws /= sw
        # Elite mean
        m_ela = np.zeros(d)
        for r in range(mu_eff):
            m_ela += ws[r] * cands[r][1]
        m_new = (1 - eta) * m + eta * m_ela
        # Project m_new back
        hp = project_capped(u + helmert_fwd(m_new, n), n)
        if hp is not None:
            m_new = helmert_bwd(hp - u, n)
        else:
            break
        # Covariance update
        if full_cov:
            S = np.zeros((d, d))
            for r in range(mu_eff):
                dv = cands[r][1] - m_new
                S += ws[r] * np.outer(dv, dv)
            C = (1 - eta) * C + eta * S
            C = 0.5 * (C + C.T)
        else:
            sv_new = np.empty(d)
            for j in range(d):
                sv = 0.0
                for r in range(mu_eff):
                    dv = cands[r][1][j] - m_new[j]
                    sv += ws[r] * dv * dv
                sv_new[j] = np.clip((1 - eta) * C[j] + eta * sv, smin**2, smax**2)
            C = sv_new
        m = m_new
        # Stall check
        best_this = cands[0][0]
        if best_this < run_best_s - 1e-12:
            run_best_s = best_this
            stall = 0
        else:
            stall += 1
            if stall >= stall_lim:
                break
    return gbest, gscore

import numpy as np
import math

def project_capped(z, n):
    z = np.where(np.isfinite(z), z, 0.0).astype(float)
    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for i in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[i]); h[i] += d; r -= d
        else:
            d = min(-r, h[i]); h[i] -= d; r += d
    if abs(r) > 1e-9 * n or not np.all(np.isfinite(h)):
        return None
    return h

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(np.sum(h) - n / 2.0) <= 1e-9 * n)

def exact_c5(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if c.shape[0] != 2 * n - 1 or not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def pair_eval(h, i, j, t, n):
    ht = h.copy()
    ht[i] += t
    ht[j] -= t
    if ht[i] < -1e-12 or ht[i] > 1.0 + 1e-12 or ht[j] < -1e-12 or ht[j] > 1.0 + 1e-12:
        return float('inf')
    np.clip(ht, 0.0, 1.0, out=ht)
    c = np.correlate(ht, 1.0 - ht, mode='full') * (2.0 / n)
    return float(np.max(c))

def cem(h0, n, lam, gens, stall_lim, rng, gbest, gscore, sigma0):
    m = h0.copy()
    C = (sigma0 * sigma0) * np.eye(n)
    eta, smax, smin = 0.35, 0.5, 0.02
    mu = lam // 2
    run_best, stall = gscore, 0
    for gen in range(gens):
        try:
            w, V = np.linalg.eigh(0.5 * (C + C.T))
            w = np.clip(w, smin * smin, smax * smax)
            L = V @ np.diag(np.sqrt(np.maximum(w, 0.0)))
        except Exception:
            break
        zs = rng.standard_normal((lam // 2, n))
        cands = []
        for ii in range(lam // 2):
            for sg in (1, -1):
                x = m + sg * (L @ zs[ii])
                h = project_capped(x, n)
                if h is None:
                    continue
                sc = exact_c5(h, n)
                if sc < float('inf'):
                    cands.append((sc, h))
                    if sc < gscore - 1e-12:
                        gbest, gscore = h.copy(), sc
        if not cands:
            break
        cands_scores = [c[0] for c in cands]
        order = np.argsort(cands_scores)
        cands = [cands[k] for k in order]
        me = min(mu, len(cands))
        ws = np.zeros(me)
        for r in range(me):
            ws[r] = math.log(me + 0.5) - math.log(r + 1)
        ws /= ws.sum()
        mn = (1.0 - eta) * m
        for r in range(me):
            mn += ws[r] * cands[r][1]
        hp = project_capped(mn, n)
        if hp is None:
            break
        mn = hp
        S = np.zeros((n, n))
        for r in range(me):
            dv = cands[r][1] - mn
            S += ws[r] * np.outer(dv, dv)
        C = 0.5 * ((1.0 - eta) * C + eta * S + ((1.0 - eta) * C + eta * S).T)
        m = mn
        if cands[0][0] < run_best - 1e-12:
            run_best = cands[0][0]
            stall = 0
        else:
            stall += 1
            if stall >= stall_lim:
                break
    return gbest, gscore

def local_polish(h, n, rng, ntrials):
    h = h.copy()
    sc = exact_c5(h, n)
    for _ in range(ntrials):
        i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
        if i == j:
            continue
        L = max(-h[i], h[j] - 1.0)
        U = min(1.0 - h[i], h[j])
        if U - L < 1e-12:
            continue
        best_s, best_t = sc, 0.0
        for _ in range(10):
            t = L + rng.uniform(0.0, 1.0) * (U - L)
            s2 = pair_eval(h, i, j, t, n)
            if s2 < best_s - 1e-13:
                best_s, best_t = s2, t
        if best_t == 0.0:
            continue
        ht = h.copy()
        ht[i] += best_t
        ht[j] -= best_t
        ht = project_capped(ht, n)
        if ht is None:
            continue
        s2 = exact_c5(ht, n)
        if s2 < sc - 1e-13:
            h, sc = ht, s2
    return h, sc
