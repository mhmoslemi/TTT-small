            vs = 0.5 + 0.4 * np.cos(2.0 * np.pi * k * np.arange(n) / n + ph)
            try_add(st, project_capped(vs, n))
    for sd in (0.08, 0.22):
        try_add(st, project_capped(fb + rng.normal(0, sd, n), n))
    try_add(st, project_capped(np.roll(fb, n // 3), n))
    try_add(st, project_capped(np.roll(fb, n // 2), n))
    try_add(st, project_capped(rng.uniform(0, 1, n), n))
    return st[:mx]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        n = 47
        try:
            vk = int(kwargs.get('n_points', 47))
            if 1 <= vk <= 999:
                n = vk
        except Exception:
            pass
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if parent is not None:
        fb = project_capped(parent, n)
    else:
        fb = np.full(n, 0.5)
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    if n == 1:
        return np.array([0.5], dtype=np.float64), 0.5, 1
    fb_s = exact_score(fb, n)
    best_h = fb.copy()
    best_s = fb_s
    mx = 10 if n <= 64 else (7 if n <= 128 else 4)
    starts = build_starts(fb, n, rng, mx)
    if n <= 128:
        mit = 220 if n <= 64 else 140
        bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_j, 'args': (n,)},
            {'type': 'eq', 'fun': _eq, 'jac': _eq_j, 'args': (n,)}
        ]
        for h0 in starts:
            if h0 is None or not is_feasible(h0, n):
                continue
            z0 = np.zeros(n + 1)
            z0[:n] = h0
            z0[n] = exact_score(h0, n) + 1e-8
            try:
                res = minimize(_obj, z0, args=(n,), jac=_obj_j, method='SLSQP',
                               bounds=bnds, constraints=cons,
                               options={'ftol': 1e-10, 'maxiter': mit, 'disp': False})
                if res.x is not None and len(res.x) == n + 1 and np.all(np.isfinite(res.x)):
                    for a in (1.0, 0.75, 0.5, 0.25):
                        hv = project_capped((1.0 - a) * h0 + a * res.x[:n], n)
                        if hv is not None and is_feasible(hv, n):
                            s = exact_score(hv, n)
                            if s < best_s - 1e-12:
                                best_h = hv.copy()
                                best_s = s
            except Exception:
                pass
    taus = [0.02, 0.005, 0.001, 0.0001]
    itp = 12 if n <= 128 else (30 if n <= 256 else 20)
    rstarts = [best_h]
    for h0 in starts:
        if h0 is not None and is_feasible(h0, n) and len(rstarts) < 3:
            if np.max(np.abs(h0 - best_h)) > 1e-8:
                rstarts.append(h0.copy())
    for h0 in rstarts:
        hr, sr = smooth_refine(h0.copy(), n, taus, itp)
        if hr is not None and is_feasible(hr, n):
            s = exact_score(hr, n)
            if s < best_s - 1e-12:
                best_h = hr.copy()
                best_s = s
    if best_h is None or not is_feasible(best_h, n):
        best_h, best_s = fb.copy(), fb_s
    else:
        fs = exact_score(best_h, n)
        if fs < best_s:
            best_s = fs
    return best_h.astype(np.float64), float(best_s), int(n)
