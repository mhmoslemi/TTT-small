Let me also make sure the gradient is correct. For F_d = sum_{i in I_d} h[i]*(1-h[i+d]):
- dF_d/dh[r] = sum over contributions where r appears
- If r = i (first factor): contribution is (1-h[i+d]) for r=i where i+d is in range
- If r = i+d (second factor): contribution is -h[i] for r=i+d where i is in range

So for lag d, the indices in I_d are lo to hi-1.
- For the first factor h[i]: i ranges over [lo, hi), so for each such i, dF_d/dh[i] += (1-h[i+d])
- For the second factor (1-h[i+d]): i+d ranges over [lo+d, hi+d), so for each such i, dF_d/dh[i+d] -= h[i]

        add_start(starts, fallback[::-1], n)
        add_start(starts, 1.0 - fallback, n)
        add_start(starts, np.roll(fallback, max(1, n // 3)), n)
        add_start(starts, fallback + rng.normal(0, 0.03, n), n)
        add_start(starts, fallback + rng.normal(0, 0.10, n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
        add_start(starts, fallback + rng.normal(0, 0.05, n), n)
    else:
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        add_start(starts, blk, n)
        add_start(starts, np.where(np.arange(n) % 2 == 0, 1.0, 0.0), n)
        tt = np.arange(n)
        add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * tt / n), n)
        add_start(starts, 0.5 + 0.45 * np.cos(4.0 * np.pi * tt / n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
        add_start(starts, fallback + rng.normal(0, 0.05, n), n)
    max_starts = 6 if n <= 128 else (4 if n <= 512 else 2)
    starts = starts[:max_starts]
    best_h = fallback.copy().astype(np.float64)
    best_sc = fb_score
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_sc - 1e-12:
            best_h[:] = st
            best_sc = ss
    for st in starts:
        best_h, best_sc = cont_pgcd(st.copy(), n, best_h.copy(), best_sc, rng)
    hop_sds = [0.025, 0.08] if n <= 128 else [0.05]
    for std in hop_sds:
        pert = best_h + rng.normal(0, std, n)
        pc = project_capped(pert, n)
        if pc is None:
            continue
        best_h, best_sc = cont_pgcd(pc, n, best_h.copy(), best_sc, rng)
    final = project_capped(best_h, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score + 1e-12:
        final, fs = fallback.copy(), fb_score
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final.astype(np.float64), float(fs), int(n)
