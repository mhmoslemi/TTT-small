import numpy as np

def project_capped(z, n):
    if z is None:
        return None
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(n + 4):
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + r, 0.0)
        r = s - float(np.sum(h))
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(float(np.sum(h)) - s) > 1e-9 * max(1, n):
        return None
    return h

def is_feasible(h, n):
    if h is None or h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    return abs(float(np.sum(h)) - n / 2.0) <= 1e-9 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    try:
        return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))
    except Exception:
        return float('inf')

def lag_grads(h, n):
    dx = 2.0 / n
    nlag = 2 * n - 1
    q = np.zeros(nlag, dtype=np.float64)
    G = np.zeros((nlag, n), dtype=np.float64)
    q[0] = dx * float(np.sum(h * (1.0 - h)))
    G[0, :] = dx * (1.0 - 2.0 * h)
    for r in range(1, n):
        m = n - r
        ip = 2 * r - 1
        im = 2 * r
        q[ip] = dx * float(np.sum(h[:m] * (1.0 - h[r:r + m])))
        q[im] = dx * float(np.sum(h[r:r + m] * (1.0 - h[:m])))
        G[ip, :m] += dx * (1.0 - h[r:r + m])
        G[ip, r:] -= dx * h[:m]
        G[im, r:] += dx * (1.0 - h[:m])
        G[im, :m] -= dx * h[r:r + m]
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None, None
    return q, G

def solve_qp(q, G, h, n, L):
    try:
        import cvxpy as cp
    except Exception:
        return None
    nlag = 2 * n - 1
    d = cp.Variable(n)
    u = cp.Variable()
    obj = cp.Minimize(u + 0.5 * L * cp.sum_squares(d))
    one = np.ones(nlag)
    cons = [G @ d + q <= u * one,
            cp.sum(d) == 0,
            d >= -h,
            d <= 1.0 - h]
    prob = cp.Problem(obj, cons)
    for solver, kw in ((cp.ECOS, {
        'abs_tol': 1e-8, 'rel_tol': 1e-8, 'max_iter': 500}),
        (cp.CVXOPT, {
        'abs_tol': 1e-8, 'rel_tol': 1e-8}),
        (cp.MOSEK, {
        'abs_tol': 1e-8, 'rel_tol': 1e-8, 'max_iters': 500}),
        (cp.SCS, {
        'eps': 1e-8, 'max_iters': 2000})):
        try:
            prob.solve(solver=solver, **kw)
        except Exception:
            continue
        if d.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
            dv = np.asarray(d.value, dtype=np.float64).ravel()
            if dv.shape == (n,) and np.all(np.isfinite(dv)):
                return dv
    return None

def subgrad_step(h, n, alpha):
    q, G = lag_grads(h, n)
    if q is None:
        return h, exact_score(h, n)
    j = int(np.argmax(q))
    h2 = project_capped(h - alpha * G[j], n)
    if h2 is None:
        return h, exact_score(h, n)
    s2 = exact_score(h2, n)
    if s2 < exact_score(h, n) - 1e-14:
        return h2, s2
    return h, exact_score(h, n)

def mm_optimize(h0, n, max_iter, use_subgrad):

def try_add(pool, v, n, th=1e-8):
    if v is None or not is_feasible(v, n):
        return
    for s in pool:
        if float(np.max(np.abs(v - s))) < th:
            return
    pool.append(v)

def build_pool(fb, n, rng):
    pool = []
    try_add(pool, fb.copy(), n)
    try_add(pool, project_capped(fb[::-1], n), n)
    try_add(pool, project_capped(1.0 - fb, n), n)
    try_add(pool, project_capped(1.0 - fb[::-1], n), n)
    try_add(pool, np.full(n, 0.5), n)
    blk = np.zeros(n)

    try_add(pool, project_capped(blk, n), n)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    try_add(pool, project_capped(alt, n), n)
    p = rng.permutation(n)
    bv = np.zeros(n)
    bv[p[:n // 2]] = 1.0
    if n % 2:
        bv[p[n // 2]] = 0.5
    try_add(pool, project_capped(bv, n), n)

    try_add(pool, project_capped(rng.uniform(0, 1, n), n), n)
    try_add(pool, project_capped(np.roll(fb, n // 3), n), n)
    try_add(pool, project_capped(np.roll(fb, n // 2), n), n)
    for k in (1, 2, 3):

            try_add(pool, project_capped(cv, n), n)
    while len(pool) < 4:
        pool.append(fb.copy())
    return pool[:16]
