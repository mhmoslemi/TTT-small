    for k in range(L):
        d = k - (n - 1)
        lo = max(0, -d)
        hi = min(n, n - d)
        v[k] = float(np.dot(h[lo:hi], 1.0 - h[lo + d:hi + d]))
    return v

def lag_grad(h, n, p):
    L = 2 * n - 1
    g = np.zeros(n)
    for k in range(L):
        pk = p[k]
        if pk < 1e-16:
            continue
        d = k - (n - 1)
        lo = max(0, -d)
        hi = min(n, n - d)
        g[lo:hi] += pk * (1.0 - h[lo + d:hi + d])
        g[lo + d:hi + d] -= pk * h[lo:hi]
    return g

def soft_surrogate(h, n, theta):
    v = lag_values(h, n)
    m = float(np.max(v))
    args = (v - m) / theta
    args = np.clip(args, -700.0, 700.0)
    ex = np.exp(args)
    s = float(np.sum(ex))
    if s <= 0 or not np.isfinite(s):
        return float('inf'), None
    phi = m + theta * math.log(s)
    p = ex / s
    return phi, p

def continuation_optimize(h0, n, init_best, init_score):
    tau_sch = [0.012, 0.004, 0.0012, 0.0004, 0.00012, 0.00004, 0.00001]
    if n <= 128:
        iters_per = 100
    elif n <= 512:
        iters_per = 35
    else:
        iters_per = 14
    h = h0.copy()
    best = init_best.copy() if init_best is not None else h0.copy()
    best_s = float(init_score)
    for tau in tau_sch:
        theta = (n / 2.0) * tau
        alpha = 0.12
        for it in range(iters_per):
            phi, p = soft_surrogate(h, n, theta)
            if not np.isfinite(phi) or p is None:
                break
            g = lag_grad(h, n, p)
            if not np.all(np.isfinite(g)):
                break
            accepted = False
            d_step = np.zeros(n)
            for bt in range(26):
                y = project_capped(h - alpha * g, n)
                if y is None:
                    alpha *= 0.5
                    continue
                d_step = y - h
                gd = float(np.dot(g, d_step))
                if gd > -1e-14:
                    alpha *= 0.5
                    continue
                phi_y, _ = soft_surrogate(y, n, theta)
                allow = 1e-12 * max(1.0, abs(phi))
                if phi_y <= phi + 1e-4 * gd + allow:
                    h = y
                    accepted = True
                    break
                alpha *= 0.5
            if not accepted:
                break
            if float(np.max(np.abs(d_step))) < 1e-10:
                break
            alpha = min(0.5, 1.25 * alpha)
            sc = exact_score(h, n)
            if np.isfinite(sc) and sc < best_s - 1e-13:
                best = h.copy()
                best_s = sc
    sc = exact_score(h, n)
    if np.isfinite(sc) and sc < best_s - 1e-13:
        best = h.copy()
        best_s = sc
    return best, best_s

def active_lag_polish(h, n, iters=40):
    best = h.copy()
    best_s = exact_score(best, n)
    theta = (n / 2.0) * 0.000005
    alpha = 0.05
    for _ in range(iters):
        phi, p = soft_surrogate(best, n, theta)
        if not np.isfinite(phi) or p is None:
            break
        g = lag_grad(best, n, p)
        if not np.all(np.isfinite(g)):
            break
        y = project_capped(best - alpha * g, n)
        if y is None:
            alpha *= 0.5
            continue
        sc = exact_score(y, n)
        if np.isfinite(sc) and sc < best_s - 1e-14:
            best = y
            best_s = sc
            alpha = min(0.3, alpha * 1.1)
        else:
            alpha *= 0.7
    return best, best_s

def add_start(starts, v, n, dedup_tol=1e-8):
    h2 = project_capped(v, n)
    if h2 is None:
        return
    for ex in starts:
        if float(np.max(np.abs(h2 - ex))) < dedup_tol:
            return
    starts.append(h2)

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= len(iv) <= 999 and np.all(np.isfinite(iv)):
            n = len(iv)
            parent = iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
        n = max(1, min(999, n))
    try:
        sd = int(seed)
        if sd < 0:
            sd = 42
    except Exception:
        sd = 42
    rng = np.random.default_rng(sd)

    uniform = np.full(n, 0.5)
    if parent is not None:
        h0 = project_capped(parent, n)
        fallback = h0 if h0 is not None else uniform.copy()
    else:
        fallback = uniform.copy()
    fb_score = exact_score(fallback, n)
    if not np.isfinite(fb_score):
        fallback = uniform.copy()
        fb_score = exact_score(fallback, n)

    starts = []
    add_start(starts, fallback, n)
    if parent is not None:
        add_start(starts, fallback[::-1], n)
        add_start(starts, 1.0 - fallback, n)
        add_start(starts, np.roll(fallback, n // 7), n)
        add_start(starts, np.roll(fallback, 3), n)
        add_start(starts, parent + rng.normal(0, 0.02, n), n)
        add_start(starts, parent + rng.normal(0, 0.06, n), n)
        add_start(starts, parent + rng.normal(0, 0.15, n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
    else:
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        add_start(starts, blk, n)
        alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
        add_start(starts, alt, n)
        tt = np.arange(n)
        add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * tt / n), n)
        add_start(starts, 0.5 + 0.45 * np.cos(4.0 * np.pi * tt / n), n)
        add_start(starts, rng.uniform(0, 1, n), n)

    if n <= 128:
        max_starts = 7
    elif n <= 512:
        max_starts = 4
    else:
        max_starts = 2
    starts = starts[:max_starts]

    best = fallback.copy()
    best_score = fb_score
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_score - 1e-12:
            best = st.copy()
            best_score = ss

    for st in starts:
        try:
            b2, s2 = continuation_optimize(st, n, best.copy(), best_score)
            if s2 < best_score - 1e-13:
                best = b2
                best_score = s2
        except Exception:
            continue

    if n <= 128:
        hop_sds = [0.015, 0.04, 0.09]
    else:
        hop_sds = [0.03, 0.07]
    for sdp in hop_sds:
        try:
            pert = best + rng.normal(0, sdp, n)
            hp = project_capped(pert, n)
            if hp is None:
                continue
            b2, s2 = continuation_optimize(hp, n, best.copy(), best_score)
            if s2 < best_score - 1e-13:
                best = b2
                best_score = s2
        except Exception:
            continue

    if n <= 256:
        try:
            bp, sp = active_lag_polish(best, n, 60)
            if sp < best_score - 1e-13:
                best = bp
                best_score = sp
        except Exception:
            pass

    final = project_capped(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score + 1e-12:
        final = fallback.copy()
        fs = exact_score(final, n)
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final, float(fs), int(n)
