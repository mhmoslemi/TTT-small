            deltas.append(d)
        if ps < best_s - 1e-12:
            best_s = ps
            best_u = pu.copy()
    if deltas:
        T0 = float(np.median(deltas)) / math.log(2.0)
    else:
        T0 = 0.04 * dx
    T0 = max(0.002 * dx, min(T0, 0.25 * dx))
    fracs = [0.08, 0.04, 0.02, 0.01, 0.005, 0.0025]
    for l in range(6):
        qmax = max(1, int(round(fracs[l] * M)))
        T = T0 * (0.4 ** l)
        for _ in range(steps):
            nu = propose_exchange(cur, n, M, qmax, rng)
            if nu is None:
                continue
            ns = exact_score(units_to_h(nu, M), n)
            if math.isinf(ns):
                continue
            if ns < best_s - 1e-12:
                best_s = ns
                best_u = nu.copy()
            delta = ns - cur_s
            if delta <= 0:
                cur = nu
                cur_s = ns
            else:
                ratio = delta / T if T > 1e-15 else float('inf')
                if ratio > 745:
                    prob = 0.0
                else:
                    prob = math.exp(-ratio)
                if rng.random() < prob:
                    cur = nu
                    cur_s = ns
    return best_u, best_s

def local_refine(u, n, M, iters):
    best_u = u.copy()
    best_s = exact_score(units_to_h(u, M), n)
    for _ in range(iters):
        improved = False
        best_move = None
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                nu = u.copy()
                nu[i] -= 1
                nu[j] += 1
                if nu[i] < 0 or nu[j] > M:
                    continue
                ns = exact_score(units_to_h(nu, M), n)
                if ns < best_s - 1e-15:
                    best_s = ns
                    best_u = nu.copy()
                    best_move = (i, j)
                    improved = True
        u = best_u.copy()
        if not improved:
            break
    return best_u, best_s
