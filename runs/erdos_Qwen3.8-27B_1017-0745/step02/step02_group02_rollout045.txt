def box_direction(h, n, rng):
    donors = [i for i in range(n) if h[i] > 1e-8]

        return None
    rng.shuffle(donors)
    rng.shuffle(recv)
    grp = min(4, len(donors), len(recv))
    d_idx = donors[:grp]
    r_idx = recv[:grp]
    v = np.zeros(n)
    w1 = np.array([1.0] * grp)
    w2 = np.array([1.0] * grp)
    s1 = np.sum(w1)
    s2 = np.sum(w2)
    sc = 1.0 / max(s1, s2)
    for k in range(grp):
        v[d_idx[k]] = -w1[k] * sc
        v[r_idx[k]] = w2[k] * sc
    v -= np.mean(v)
    mx = np.max(np.abs(v))

        return None
    return v / mx

    fs = exact_score(fb, n)
    gb, gs = fb.copy(), fs
    if n == 1:
        return fb, fs, 1
    n_sc = 5 if n <= 64 else 2
    n_mi = n - 1 if n <= 48 else min(24, n - 1)
    n_pass = 3 if n <= 64 else 1
    starts = [fb.copy()]
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    pb = project_capped(blk, n)
    if pb is not None and np.max(np.abs(pb - fb)) > 1e-8:
        starts.append(pb)
    i = np.arange(n)
    wav = np.clip(0.5 + 0.45 * np.cos(2.0 * np.pi * (i + 0.5) / n), 0, 1)
    pw = project_capped(wav, n)
    if pw is not None and np.max(np.abs(pw - fb)) > 1e-8:
        starts.append(pw)
    pr = project_capped(fb + rng.normal(0, 0.06, n), n)
    if pr is not None and np.max(np.abs(pr - fb)) > 1e-8:
        starts.append(pr)
    pu = project_capped(rng.uniform(0, 1, n), n)
    if pu is not None:
        starts.append(pu)
    starts = starts[:n_sc]
    for st in starts:
        ss = exact_score(st, n)
        if ss < gs - 1e-12:
            gs, gb = ss, st.copy()
    for st in starts:
        try:
            res_h, res_s = sweep(st, n, rng, n_mi, n_pass)
            if res_s < gs - 1e-12:
                gs, gb = res_s, res_h.copy()
        except Exception:
            continue
    for _ in range(2):
        try:
            res_h, res_s = sweep(gb, n, rng, n_mi, n_pass)
            if res_s < gs - 1e-12:
                gs, gb = res_s, res_h.copy()
            else:
                break
        except Exception:
            break
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb.copy()
    sf = exact_score(final, n)
    ss = exact_score(fb, n)
    if sf > ss + 1e-14:
        final, sf = fb.copy(), ss
    if not is_feasible(final, n):
        final = np.full(n, 0.5)
        sf = exact_score(final, n)
    return final, float(sf), int(n)
