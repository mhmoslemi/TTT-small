        return False
    if abs(float(np.sum(h)) - n / 2.0) > 1e-9:
        return False
    return True

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    try:
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if not np.all(np.isfinite(c)):
            return float('inf')
        return float(np.max(c))
    except Exception:
        return float('inf')

def try_add_pool(pool, v, n):
    if v is None or v.shape != (n,) or not np.all(np.isfinite(v)):
        return
    for old in pool:
        if np.max(np.abs(v - old)) < 1e-8:
            return
    pool.append(v.copy())

def build_pop(fallback, n, rng, P):
    pool = []
    n2 = n // 2
    try_add_pool(pool, fallback, n)
    try_add_pool(pool, project_capped(fallback[::-1].copy(), n), n)
    try_add_pool(pool, project_capped(1.0 - fallback, n), n)
    try_add_pool(pool, project_capped((1.0 - fallback)[::-1].copy(), n), n)
    try_add_pool(pool, project_capped(np.full(n, 0.5), n), n)
    blk = np.zeros(n); blk[:n2] = 1.0
    if n % 2: blk[n2] = 0.5
    try_add_pool(pool, project_capped(blk, n), n)
    try_add_pool(pool, project_capped(np.where(np.arange(n) % 2 == 0, 1.0, 0.0), n), n)
    Q = max(2, n // 3)
    try_add_pool(pool, project_capped(0.5 + 0.45 * np.cos(2.0 * np.pi * np.arange(n) / Q), n), n)
    for sc in [0.04, 0.12, 0.24]:
        try_add_pool(pool, project_capped(fallback + rng.normal(0, sc, n), n), n)
    try_add_pool(pool, project_capped(rng.uniform(0, 1, n), n), n)
    att = 0
    while len(pool) < P and att < 20 * P:
        att += 1
        if att % 3 == 0:
            nv = project_capped(rng.uniform(0, 1, n), n)
        else:
            scs = [0.05, 0.12, 0.25]
            nv = project_capped(fallback + rng.normal(0, scs[(att // 3) % 3], n), n)
        try_add_pool(pool, nv, n)
    while len(pool) < P:
        pool.append(pool[len(pool) % max(1, len(pool))].copy())
    return pool[:P]

def de_search(pop, scores, n, rng, n_gen, patience, max_restarts):
    P = len(pop)
    X = np.array([p.copy() for p in pop], dtype=np.float64)
    sc = np.array(scores, dtype=np.float64)
    bi = int(np.argmin(sc))
    bv = float(sc[bi])
    bvec = X[bi].copy()
    stag = 0
    rests = 0
    for gen in range(n_gen):
        Xold = X.copy()
        scold = sc.copy()
        imp = False
        for i in range(P):
            others = np.delete(np.arange(P), i)
            r1, r2, r3 = int(others[rng.integers(0, P - 1)]), int(others[rng.integers(0, P - 1)]), int(others[rng.integers(0, P - 1)])
            while r2 == r1: r2 = int(others[rng.integers(0, P - 1)])
            while r3 == r1 or r3 == r2: r3 = int(others[rng.integers(0, P - 1)])
            F = rng.uniform(0.40, 0.90)
            CR = rng.uniform(0.70, 0.95)
            donor = Xold[r1] + F * (Xold[r2] - Xold[r3])
            dp = project_capped(donor, n)
            if dp is None:
                continue
            mask = rng.random(n) < CR
            if not mask.any():
                mask[rng.integers(0, n)] = True
            trial = np.where(mask, dp, Xold[i])
            tp = project_capped(trial, n)
            if tp is None or np.max(np.abs(tp - Xold[i])) < 1e-12:
                continue
            ts = exact_score(tp, n)
            if np.isfinite(ts) and ts < scold[i] - 1e-12:
                X[i] = tp; sc[i] = ts; imp = True
            if np.isfinite(ts) and ts < bv - 1e-12:
                bv = ts; bvec = tp.copy(); imp = True
        if not imp:
            stag += 1
        else:
            stag = 0
        if stag >= patience:
            if rests >= max_restarts:
                break
            rests += 1
            stag = 0
            newpool = [bvec.copy()]
            order = np.argsort(sc)
            for j in order:
                if len(newpool) >= max(2, P // 2) + 1:
                    break
                try_add_pool(newpool, X[j], n)
            att2 = 0
            while len(newpool) < P and att2 < 8 * (P - len(newpool)):
                att2 += 1
                if att2 % 3 == 0:
                    nv = project_capped(rng.uniform(0, 1, n), n)
                else:
                    scs2 = [0.05, 0.12, 0.25]
                    nv = project_capped(bvec + rng.normal(0, scs2[(att2 // 3) % 3], n), n)
                if nv is not None:
                    try_add_pool(newpool, nv, n)
                else:
                    newpool.append(bvec.copy())
            while len(newpool) < P:
                newpool.append(newpool[len(newpool) % max(1, len(newpool))].copy())
            X = np.array([b.copy() for b in newpool[:P]], dtype=np.float64)
            sc = np.array([exact_score(b, n) for b in newpool[:P]], dtype=np.float64)
            mi = int(np.argmin(sc))
            if sc[mi] < bv - 1e-12:
                bv = float(sc[mi]); bvec = X[mi].copy()
    return bvec, bv

def coord_polish(x, n, n_iter=60):
    best = x.copy()
    bs = exact_score(x, n)
    if not np.isfinite(bs):
        return best, bs
    for _ in range(n_iter):
        imp = False
        for i in range(n):
            for d in [-0.03, 0.03, -0.015, 0.015, -0.007, 0.007]:
                cand = best.copy()
                cand[i] += d
                cp = project_capped(cand, n)
                if cp is None:
                    continue
                cs = exact_score(cp, n)
                if np.isfinite(cs) and cs < bs - 1e-12:
                    best = cp; bs = cs; imp = True; break
        if not imp:
            break
    return best, bs

def binary_snap(x, n):
    best = x.copy()
    bs = exact_score(x, n)
    if not np.isfinite(bs):
        return best, bs
    for i in range(n):
        for val in [0.0, 1.0]:
            if abs(best[i] - val) < 1e-6:
                continue
            cand = best.copy()
            cand[i] = val
            cp = project_capped(cand, n)
            if cp is None:
                continue
            cs = exact_score(cp, n)
            if np.isfinite(cs) and cs < bs - 1e-12:
                best = cp; bs = cs
    return best, bs

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        if 'n_points' in kwargs:
            try:
                v = int(kwargs['n_points'])
                if 1 <= v <= 999:
                    n = v
            except Exception:
                pass
        if n is None:
            n = 47
    if parent is not None:
        fallback = project_capped(parent, n)
    else:
        fallback = project_capped(np.full(n, 0.5), n)
    if fallback is None:
        fallback = np.full(n, 0.5)
    fallback = np.asarray(fallback, dtype=np.float64)
    if n == 1:
        return fallback, float(exact_score(fallback, n)), int(n)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 64:
        P, ng, pat, mr = 48, 220, 30, 2
    elif n <= 128:
        P, ng, pat, mr = 32, 150, 25, 2
    elif n <= 256:
        P, ng, pat, mr = 24, 100, 20, 1
    else:
        P, ng, pat, mr = 16, 60, 15, 1
    fb_s = float(exact_score(fallback, n))
    best_vec = fallback.copy()
    best_sc = fb_s
    pop = build_pop(fallback, n, rng, P)
    scs = [exact_score(p, n) for p in pop]
    try:
        bv, bs = de_search(pop, scs, n, rng, ng, pat, mr)
        if bs < best_sc - 1e-12:
            best_vec, best_sc = bv, bs
    except Exception:
        pass
    try:
        pv, ps = coord_polish(best_vec, n, 80)
        if ps < best_sc - 1e-12:
            best_vec, best_sc = pv, ps
    except Exception:
        pass
    try:
        bv2, bs2 = binary_snap(best_vec, n)
        if bs2 < best_sc - 1e-12:
            best_vec, best_sc = bv2, bs2
    except Exception:
        pass
    try:
        pv2, ps2 = coord_polish(best_vec, n, 40)
        if ps2 < best_sc - 1e-12:
            best_vec, best_sc = pv2, ps2
    except Exception:
        pass
    P2 = max(16, P // 2)
    pop2 = build_pop(best_vec, n, rng, P2)
    scs2 = [exact_score(p, n) for p in pop2]
    try:
        bv3, bs3 = de_search(pop2, scs2, n, rng, ng // 2, pat // 2, 1)
        if bs3 < best_sc - 1e-12:
            best_vec, best_sc = bv3, bs3
    except Exception:
        pass
    try:
        pv3, ps3 = coord_polish(best_vec, n, 60)
        if ps3 < best_sc - 1e-12:
            best_vec, best_sc = pv3, ps3
    except Exception:
        pass
    try:
        bv4, bs4 = binary_snap(best_vec, n)
        if bs4 < best_sc - 1e-12:
            best_vec, best_sc = bv4, bs4
    except Exception:
        pass
    final = project_capped(best_vec, n)
    if final is None or not is_feasible(final, n):
        final = fallback.copy()
    fs = float(exact_score(final, n))
    if not np.isfinite(fs) or fs > fb_s + 1e-12:
        final = fallback.copy()
        fs = fb_s
    if not np.isfinite(fs):
        final = project_capped(np.full(n, 0.5), n)
        if final is None:
            final = np.full(n, 0.5)
        fs = float(exact_score(final, n))
    return final.astype(np.float64), fs, int(n)
