def pair_refine(h,n,sweeps):

  for _ in range(sweeps):

      for j in range(i+1,n):

        hi = min(1.0-h[i], h[j])

def pair_refine(h,n,sweeps):
    h=h.copy()
    deltas=[0.001,0.004,0.015,0.06,0.18]
    for _ in range(sweeps):
        improved=False
        for i in range(n):
            for j in range(i+1,n):
                lo = max(-h[i], h[j]-1.0)
                hi = min(1.0-h[i], h[j])
                if hi <= lo + 1e-12: continue
                s0 = exact_score(h,n)
                if not np.isfinite(s0): continue
                best_s=s0; best_d=0.0
                # raw list
                raw=[lo,hi,0.5*(lo+hi), 0.25*(lo+hi), 0.75*(lo+hi)]
                for dv in deltas:
                    raw.append(dv); raw.append(-dv)
                seen=set()
                for d in raw:
                    if d < lo -1e-12 or d > hi +1e-12: continue
                    if d < lo: d=lo
                    if d > hi: d=hi
                    r=round(float(d),10)
                    if r in seen: continue
                    seen.add(r)
                    if abs(d) < 1e-13: continue
                    h[i]+=d; h[j]-=d
                    s=exact_score(h,n)
                    if np.isfinite(s) and s < best_s - 1e-13:
                        best_s=s; best_d=d
                    h[i]-=d; h[j]+=d
                if best_d != 0.0:
                    h[i]+=best_d; h[j]-=best_d; improved=True
        if not improved: break
    return h

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = 0.5 * n
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if mid == lo or mid == hi: break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for i in range(n):
        if abs(r) <= 1e-13: break
        if r > 0:
            room = 1.0 - h[i]

                h[i] += add
                r -= add
        else:
            room = h[i]
            if room > 0:
                sub = min(-r, room)
                h[i] -= sub
                r += sub
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-8:
        return None
    return h

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if mid == lo or mid == hi:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)

            break

def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()

        return float('inf')

def smooth_value_grad(h, tau, n, infos):
    h = np.asarray(h, dtype=np.float64).ravel()

        return float('inf'), np.zeros(n, dtype=np.float64)
    ih = 1.0 - h
    L = len(infos)
    dx = 2.0 / n
    q = np.empty(L, dtype=np.float64)

        lo, hi, l2, h2 = infos[k]
        q[k] = dx * float(np.dot(h[lo:hi], ih[l2:h2]))
    m = float(np.max(q))
    if not np.isfinite(m):
        return float('inf'), np.zeros(n, dtype=np.float64)

    es = float(np.sum(e))

            best, best_s = projected_continue(trial, n, infos, taus_hop, iters_hop, best, best_s)
            if best_s < old_s - 1e-13:
                scale = max(0.01, 0.75 * scale)
            else:
                scale = min(0.25, 1.5 * scale)

    if pair_ok and n <= 80:
        polished = pair_refine(best, n)
        polished = project_capped(polished, n)
        if polished is not None:
            ps = exact_score(polished, n)
            if np.isfinite(ps) and ps < best_s - 1e-13:
                best = polished
                best_s = ps

    final = project_capped(best, n)
    if final is None:
        final = uniform.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > best_s + 1e-12:
        cands = [uniform]
        if parent is not None:
            cands.append(parent)
        for c in cands:
            c = project_capped(c, n)
            if c is None:
                continue
            sc = exact_score(c, n)
            if np.isfinite(sc) and (not np.isfinite(fs) or sc < fs):
                final = c
                fs = sc
        if not np.isfinite(fs):
            final = uniform.copy()
            fs = exact_score(final, n)
    final = project_capped(final, n)
    if final is None:
        final = uniform.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = 0.5
    return final, float(fs), int(n)
