def lag_model(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    A = np.zeros((L, n + 1))
    for k in range(L):
        d = k - (n - 1)
        p = max(d, 0)
        r = max(-d, 0)
        m = n - abs(d)
        A[k, p:p+m] += dx * (1.0 - h[r:r+m])
        A[k, r:r+m] -= dx * h[p:p+m]
        A[k, n] = -1.0
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(A)):
        return None, None
    return q, A

def lp_step(h, n, rho, q, A):
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = A
    b_ub = -q
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        bounds.append((max(-rho, -h[i]), min(rho, 1.0 - h[i])))
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'max_iter': 5000})
    except Exception:
        return None
    if res.status != 0 or res.x is None or res.x.shape[0] != n + 1:
        return None
    if not np.all(np.isfinite(res.x)):
        return None
    u = res.x[:n]
    if not np.all(u >= -rho - 1e-7) or not np.all(u <= rho + 1e-7):
        return None
    if abs(np.sum(u)) > 1e-6:
        return None
    return u

import numpy as np
from scipy.optimize import linprog
import math

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-9 * max(1, n):
        return False
    return True

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64)
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    target = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - np.sum(h)
    for _ in range(3):
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        h[idx] += r / len(idx)
        h = np.clip(h, 0.0, 1.0)
        r = target - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    if abs(np.sum(h) - target) > 1e-9:
        return None
    return h

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full')
    if q.shape[0] != 2 * n - 1 or not np.all(np.isfinite(q)):
        return float('inf')
    return float(np.max(q) * (2.0 / n))

def lag_model(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.correlate(h, 1.0 - h, mode='full') * dx
    A = np.zeros((L, n + 1))
    for k in range(L):
        d = k - (n - 1)
        p = max(d, 0)
        r = max(-d, 0)
        m = n - abs(d)
        A[k, p:p+m] += dx * (1.0 - h[r:r+m])
        A[k, r:r+m] -= dx * h[p:p+m]
        A[k, n] = -1.0
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(A)):
        return None, None
    return q, A

def lp_step(h, n, rho, q, A):
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub, b_ub = A, -q
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(max(-rho, -h[i]), min(rho, 1.0 - h[i])) for i in range(n)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'max_iter': 5000})
    except Exception:
        return None
    if res.status != 0 or res.x is None or res.x.shape[0] != n + 1:
        return None
    if not np.all(np.isfinite(res.x)):
        return None
    u = res.x[:n]
    if np.any(u < -rho - 1e-6) or np.any(u > rho + 1e-6) or abs(np.sum(u)) > 1e-5:
        return None
    return u

def refine_chain(start, n, max_iter, rho0, best_h, best_score):
    start = project_capped(start, n)
    if start is None or not is_feasible(start, n):
        return best_h, best_score
    current = start.copy()
    cur_score = exact_score(current, n)
    if cur_score < best_score - 1e-12:
        best_score, best_h = cur_score, current.copy()
    rho = rho0
    no_imp, no_solve = 0, 0
    for _ in range(max_iter):
        if rho < 1e-5 or no_imp >= 6 or no_solve >= 3:
            break
        q, A = lag_model(current, n)
        if q is None:
            rho *= 0.5
            no_solve += 1
            continue
        u = lp_step(current, n, rho, q, A)
        if u is None:
            rho *= 0.5
            no_solve += 1
            no_imp += 1
            continue
        no_solve = 0
        best_trial, best_s = None, cur_score
        for alpha in (1.0, 0.5, 0.25, 0.125, 0.0625):
            h_new = project_capped(current + alpha * u, n)
            if h_new is None:
                continue
            s = exact_score(h_new, n)
            if s < best_s - 1e-12:
                best_s, best_trial = s, h_new
        if best_trial is None:
            rho *= 0.5
            no_imp += 1
            continue
        old_score = cur_score
        delta = best_trial - current
        model_max = np.max(q + A[:, :n] @ delta)
        predicted = old_score - model_max
        current = best_trial
        cur_score = best_s
        no_imp = 0
        if cur_score < best_score - 1e-12:
            best_score, best_h = cur_score, current.copy()
        if predicted > 1e-12 and np.isfinite(cur_score):
            ratio = (old_score - cur_score) / predicted
            if ratio < 0.1:
                rho *= 0.5
            elif ratio > 0.75 and np.max(np.abs(delta)) >= 0.8 * rho:
                rho = min(1.5 * rho, 0.32)
        else:
            rho *= 0.5
    return best_h, best_score

def subgrad_polish(h, n, n_iter=12):
    dx = 2.0 / n
    current = h.copy()
    score = exact_score(current, n)
    eta = 0.1
    for _ in range(n_iter):
        q = np.correlate(current, 1.0 - current, mode='full') * dx
        idx = int(np.argmax(q))
        d = idx - (n - 1)
        p0, q0, m = max(d, 0), max(-d, 0), n - abs(d)
        g = np.zeros(n)
        g[p0:p0+m] += dx * (1.0 - current[q0:q0+m])
        g[q0:q0+m] -= dx * current[p0:p0+m]
        z = project_capped(current - eta * g, n)
        if z is None:
            eta *= 0.5
            continue
        best_z, best_s = None, score
        for alpha in (1.0, 0.5, 0.25):

                best_s, best_z = ts, trial
        if best_z is None:
            eta *= 0.5
        else:
            current, score = best_z, best_s
    return current, score

def try_add(pool, v, n):
    p = project_capped(v, n)
    if p is None:
        return
    for s in pool:
        if np.max(np.abs(p - s)) < 1e-8:
