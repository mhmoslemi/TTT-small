import numpy as np
from math import sqrt, copysign, isfinite

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(2):
        if abs(r) < 2e-14 * max(1, n):
            break
        if r > 0:
            for k in range(n):
                if r <= 0:
                    break
                add = min(r, 1.0 - h[k])
                h[k] += add
                r -= add
        else:
            for k in range(n):
                if r >= 0:
                    break
                rem = min(-r, h[k])
                h[k] -= rem
                r += rem
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    if abs(np.sum(h) - s) > 1e-10 * max(1, n):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def lag_values(h, n):
    L = 2 * n - 1
    out = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if out.shape != (L,) or not np.all(np.isfinite(out)):
        return None
    return out

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    A = lag_values(h, n)
    if A is None:
        return float('inf')
    return float(np.max(A))

def pair_b_d(h, n, d, i, j):
    """Compute B_d for pair (i,j)."""
    dx = 2.0 / n
    a = max(d, 0)
    b = max(-d, 0)
    m = n - abs(d)
    B = 0.0
    # i in a-range
    if 0 <= i - a < m:
        B += dx * (1.0 - h[b + i - a])
    # j in a-range
    if 0 <= j - a < m:
        B += dx * (1.0 - h[b + j - a])
    # i in b-range
    if 0 <= i - b < m:
        B -= dx * h[a + i - b]
    # j in b-range
    if 0 <= j - b < m:
        B -= dx * h[a + j - b]
    return B

def pair_all_B(h, n, i, j):
    """Compute B for all lags."""
    dx = 2.0 / n
    L = 2 * n - 1
    B = np.zeros(L)
    for d in range(-(n - 1), n):
        B[d + n - 1] = pair_b_d(h, n, d, i, j)
    return B

def pair_line_min(h, A, n, i, j):

    dx = 2.0 / n
    L = 2 * n - 1
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-14:
        return 0.0, float(np.max(A))
    B = pair_all_B(h, n, i, j)

        # Let me redo: sort by slope ascending, for same slope keep max intercept

        # For same slope, keep only max intercept
        # Build hull

        hull_start = []  # t where this line becomes active
        hull_s = []  # slopes

        for k in range(len(order)):
            s_k = sorted_s[k]
            a_k = sorted_i[k]
            if not hull_idx:
                hull_idx.append(k)
                hull_start.append(-1e18)
                hull_s.append(s_k)
                hull_a.append(a_k)
            else:

                top = len(hull_idx) - 1
                if abs(s_k - hull_s[top]) < 1e-15:

                    continue

                    hull_start.pop()
                    hull_s.pop()

                    # Only add if crossing is valid

                    hull_start.append(t_cross)
                    hull_s.append(s_k)
                    hull_a.append(a_k)
                else:
                    hull_idx.append(k)
                    hull_start.append(t_cross)
                    hull_s.append(s_k)
                    hull_a.append(a_k)
    # Now evaluate candidates
    candidates = [lo, hi, 0.0]
    # Add segment boundaries from hull
    if len(aff_idx) > 0:
        for k in range(len(hull_start)):
            if hull_start[k] > lo and hull_start[k] < hi:
                candidates.append(hull_start[k])

    for k in sorted(spec):

            tv = -B[k] / (2.0 * D[k])

def _quad_roots(a, b, c):
    """Solve a*t^2 + b*t + c = 0. Return list of real roots."""
    roots = []
    if abs(a) < 1e-15:
        if abs(b) < 1e-15:
            return roots
        t = -c / b
        if isfinite(t):
            roots.append(t)
        return roots
    disc = b * b - 4.0 * a * c
    if disc < -1e-14 * (b * b + abs(4.0 * a * c) + 1e-300):
        return roots
    if disc < 0:
        disc = 0.0
    sq = sqrt(disc)
    q = -0.5 * (b + copysign(sq, b))
    t1 = q / a
    t2 = c / q if abs(q) > 1e-300 else -b / a
    for t in (t1, t2):
        if isfinite(t):
            roots.append(t)
    return roots

def coord_sweep(h, n, rng, n_sweeps, stall_limit):
    h = h.copy()
    A = lag_values(h, n)
    if A is None:
        return h, exact_score(h, n)
    cur_score = float(np.max(A))
    total_pairs = n * (n - 1) // 2
    K = total_pairs if n <= 64 else min(total_pairs, 4 * n)
    stall = 0
    for sw in range(n_sweeps):
        start_score = cur_score
        pairs = []
        if K >= total_pairs:
            for i in range(n):
                for j in range(i + 1, n):
                    pairs.append((i, j))
        else:
            # cycle-neighbor pairs
            for i in range(n):
                j = (i + 1) % n
                if j > i:
                    pairs.append((i, j))
                else:
                    pairs.append((j, i))
            # fill with random
            while len(pairs) < K:
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i != j:
                    pairs.append((min(i, j), max(i, j)))
        rng.shuffle(pairs)
        n_accept = 0
        for (i, j) in pairs:
            t_star, pred = pair_line_min(h, A, n, i, j)
            if not isfinite(t_star) or not isfinite(pred):
                continue
            if pred >= cur_score - 1e-12:
                continue
            h_new = h.copy()
            h_new[i] += t_star
            h_new[j] -= t_star
            if not np.all(np.isfinite(h_new)) or np.any(h_new < -1e-12) or np.any(h_new > 1.0 + 1e-12):
                h_new = project_capped(h_new, n)
                if h_new is None:
                    continue
            A_new = lag_values(h_new, n)
            if A_new is None:
                continue
            s_new = float(np.max(A_new))
            if s_new < cur_score - 1e-12:
                h = h_new
                A = A_new
                cur_score = s_new
                n_accept += 1
        # Refresh A periodically
        if n_accept > 0 and n_accept % max(1, n // 2) < 1:
            A = lag_values(h, n)
            if A is None:
                break
            cur_score = float(np.max(A))
        # End of sweep: refresh
        A = lag_values(h, n)
        if A is None:
            break
        end_score = float(np.max(A))
        cur_score = end_score
        if end_score > start_score + 1e-12:
            # revert? Actually we only accept improvements, so this shouldn't happen
            pass
        if end_score < start_score - 1e-12:
            stall = 0
        else:
            stall += 1
        if stall >= stall_limit:
            break
    return h, cur_score

def build_starts(parent, n, rng):
    pool = []
    def add(candidate):
        p = project_capped(candidate, n)
        if p is None:
            return
        for existing in pool:
            if np.max(np.abs(p - existing)) < 1e-8:
                return
        pool.append(p)
    if parent is not None:
        add(parent)
    base = parent if parent is not None else np.full(n, 0.5)
    add(base[::-1].copy())
    add(1.0 - base)
    add(np.roll(base, n // 4))
    add(np.roll(base, -n // 4))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    add(blk)
    add(blk[::-1].copy())
    add(base + 0.05 * rng.standard_normal(n))
    add(base + 0.15 * rng.standard_normal(n))
    for k in (1, 2):
        ph = rng.uniform(0, 2 * np.pi)
        add(0.5 + 0.35 * np.sin(2 * np.pi * k * np.arange(n) / n + ph))
    add(rng.uniform(0, 1, n))
    return pool[:8]
