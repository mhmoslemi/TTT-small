        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def smooth_val_grad(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    qmax = np.max(corr)
    shifted = (corr - qmax) / tau
    exps = np.exp(shifted)
    Z = np.sum(exps)
    w = exps / Z
    val = qmax + tau * np.log(Z)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        p0 = max(s, 0)
        q0 = max(-s, 0)
        m = n - abs(s)
        g[p0:p0 + m] += dx * wk * (1.0 - h[q0:q0 + m])
        g[q0:q0 + m] -= dx * wk * h[p0:p0 + m]
    return val, g

                if zt is None:
                    continue
                dt = zt - h
                if np.max(np.abs(dt)) < 1e-11:
                    continue
                gd = np.dot(g, dt)
                if not np.isfinite(gd) or gd >= 0:
                    continue
                vht, _ = smooth_val_grad(zt, n, tau)
                if not np.isfinite(vht):
                    continue
                if vht <= val + 1e-4 * gd + 1e-14:
                    h = zt
                    accepted = True
                    break
            if not accepted:
                break
            eta = min(1.2 * eta, eta_max)
            s = exact_score(h, n)
            if s < best_s - 1e-12:
                best_s = s
                best_h = h.copy()
    return best_h, best_s

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999 and np.all(np.isfinite(p)):
            n = int(p.size)
            parent = p
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if parent is not None and parent.size != n:
        old_n = parent.size
        old_pos = (np.arange(old_n) + 0.5) / old_n
        new_pos = (np.arange(n) + 0.5) / n
        parent = np.interp(new_pos, old_pos, parent)
    fb = project_capped(parent, n) if parent is not None else np.full(n, 0.5)
    if fb is None:
        fb = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        return fb, exact_score(fb, 1), 1
    fs = exact_score(fb, n)
    gb, gs = fb.copy(), fs
    if n <= 64:
        n_sc, n_mi = 5, 140
    elif n <= 128:
        n_sc, n_mi = 4, 90
    elif n <= 256:
        n_sc, n_mi = 3, 55
    else:
        n_sc, n_mi = 2, 25
    dx = 2.0 / n
    taur = [0.02, 0.005, 0.001, 4e-5, 1e-5]
    eta0 = 0.25 / dx
    etamax = 2.0 / dx
    starts = build_starts(fb, n, rng, n_sc)
    for st in starts:
        try:
            bh, bs = cont_descent(st, n, taur, n_mi, eta0, etamax)
            if bs < gs - 1e-12:
                gs, gb = bs, bh
        except Exception:
            continue
    if n <= 64:
        perts = [0.03, 0.08] if gs < fs - 1e-5 else [0.08, 0.18]
    elif n <= 256:
        perts = [0.05] if gs < fs - 1e-5 else [0.12]
    else:
        perts = []
    for sd in perts:
        try:
            pert = project_capped(gb + rng.normal(0, sd, n), n)
            if pert is None:
                continue
            bh, bs = cont_descent(pert, n, taur, n_mi, eta0, etamax)
            if bs < gs - 1e-12:
                gs, gb = bs, bh
        except Exception:
            continue
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb
    sf = exact_score(final, n)
    if sf > fs:
        final, sf = fb, fs
    if not is_feasible(final, n):
        final = np.full(n, 0.5)
        sf = exact_score(final, n)
    return final, float(sf), int(n)
