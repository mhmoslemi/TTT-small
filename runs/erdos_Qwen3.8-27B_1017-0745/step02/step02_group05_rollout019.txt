            h[i] += min(r, 1.0 - h[i])
        else:
            h[i] -= min(-r, h[i])
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    if abs(np.sum(h) - s) > 1e-10 * max(1, n):
        return None
    return h

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n))

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(np.max(c)) if np.all(np.isfinite(c)) else float('inf')

def build_fourier_basis(n):
    F = (n - 1) // 2
    Phi = np.zeros((n, n - 1), dtype=np.float64)
    labels = []
    col = 0
    idx = np.arange(n)
    for k in range(1, F + 1):
        t = 2.0 * np.pi * k * idx / n
        Phi[:, col] = np.sqrt(2.0 / n) * np.cos(t)
        Phi[:, col + 1] = np.sqrt(2.0 / n) * np.sin(t)
        labels.extend([k, k])
        col += 2
    if n % 2 == 0:
        Phi[:, col] = np.array([(-1.0) ** i for i in range(n)], dtype=np.float64) / np.sqrt(n)
        labels.append(n // 2)
    return Phi, labels

def build_starts(base, n, rng, count):
    pool = [base.copy()]
    cands = [base[::-1].copy(), 1.0 - base, np.full(n, 0.5)]
    blk = np.zeros(n); blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    cands.append(blk)
    for sd in (0.06, 0.16):
        cands.append(base + rng.normal(0, sd, n))
    for sh in (1, max(1, n // 4)):
        cands.extend([np.roll(base, sh), np.roll(base, -sh)])
    for k in (1, 2):
        cands.append(0.5 + 0.3 * np.sin(2 * np.pi * k * np.arange(n) / n + rng.uniform(0, 6.28)))
    cands.append(np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64))
    cands.append(rng.uniform(0, 1, n))
    for c in cands:
        if len(pool) >= count:
            break
        p = project_capped(c, n)
        if p is None:
            continue
        if all(np.max(np.abs(p - s)) < 1e-8 for s in pool):
            continue
        pool.append(p)
    return pool[:count]

def scalar_scan(z, uj, col, n, ng, nref, nrounds):
    R = np.sqrt(n) / 2.0
    grid = np.linspace(-R, R, ng)
    scores = np.full(ng, float('inf'))
    for gi, t in enumerate(grid):
        ht = project_capped(z + (t - uj) * col, n)
        if ht is not None:
            scores[gi] = exact_score(ht, n)
    fc = [gi for gi in range(ng) if np.isfinite(scores[gi])]
    if not fc:
        hcur = project_capped(z, n)
        return uj, hcur, exact_score(hcur, n) if hcur is not None else float('inf')
    minima = []
    for gi in fc:
        if gi == 0:
            minima.append(gi) if scores[0] <= scores[1] else None
        elif gi == ng - 1:
            minima.append(gi) if scores[gi] <= scores[gi - 1] else None
        else:
            if scores[gi] <= scores[gi - 1] and scores[gi] <= scores[gi + 1]:
                minima.append(gi)
    if not minima:
        minima = [min(fc, key=lambda i: scores[i])]
    minima.sort(key=lambda i: scores[i])
    best_s = min(scores[fc], initial=float('inf'))
    best_t = grid[fc[scores[fc] == best_s][0]]
    best_h = project_capped(z + (best_t - uj) * col, n)
    for gi in minima[:nref]:
        lo_b, hi_b = grid[max(gi - 1, 0)], grid[min(gi + 1, ng - 1)]
        if uj < lo_b: lo_b = min(uj, lo_b)
        if uj > hi_b: hi_b = max(uj, hi_b)
        for _ in range(nrounds):
            inner = np.linspace(lo_b, hi_b, 5)
            lb, lti = float('inf'), 0
            for ii, t in enumerate(inner):
                ht = project_capped(z + (t - uj) * col, n)
                st = exact_score(ht, n) if ht is not None else float('inf')
                if st < lb:
                    lb, lti = st, ii
            if not np.isfinite(lb):
                break
            if lb < best_s:
                best_s, best_t = lb, inner[lti]
                best_h = project_capped(z + (best_t - uj) * col, n)
            if lti == 0: hi_b = inner[1]
            elif lti == 4: lo_b = inner[3]
            else:
                lo_b, hi_b = inner[lti - 1], inner[lti + 1]
            if hi_b - lo_b <= 1e-12 * max(1.0, R):
                break
    hcur = project_capped(z, n)
    if hcur is not None:
        sc = exact_score(hcur, n)
        if sc < best_s - 1e-12:
            best_s, best_t, best_h = sc, uj, hcur
    return best_t, best_h, best_s

def optimize_start(h0, Phi, labels, n, rng, nsweeps, ng, nref, nrounds):
    h = project_capped(h0, n)
    if h is None:
        return h0, exact_score(h0, n)
    score = exact_score(h, n)
    h_inc, s_inc = h.copy(), score
    u = Phi.T @ (h - 0.5)
    z = 0.5 + Phi @ u
    F = (n - 1) // 2
    if n <= 64:
        cuts = [1, 2, 4, 8, 16, F]
    elif n <= 256:
        cuts = [1, 2, 4, 8, min(F, 12)]
    else:
        cuts = [1, 2, 4, 8, min(F, 8)]
    cuts = sorted(set(c for c in cuts if 1 <= c <= F))
    ncol = Phi.shape[1]
    for ci, cut in enumerate(cuts):
        active = [j for j in range(ncol) if labels[j] <= cut]
        order = list(active)
        rng.shuffle(order)
        for sw in range(nsweeps):
            u_sv, z_sv, h_sv, s_sv = u.copy(), z.copy(), h.copy(), score
            accepted = False
            for j in order:
                t_new, h_new, s_new = scalar_scan(z, u[j], Phi[:, j], n, ng, nref, nrounds)
                if s_new < score - 1e-12:
                    z = z + (t_new - u[j]) * Phi[:, j]
                    u[j] = t_new
                    h = project_capped(z, n)
                    if h is None:
                        h = project_capped(0.5 + Phi @ u, n)
                    score = s_new
                    accepted = True
            sc = exact_score(h, n) if h is not None else float('inf')
            if sc > s_sv + 1e-12:
                u, z, h, score = u_sv, z_sv, h_sv, s_sv
            if not accepted:
                break
        if score < s_inc - 1e-12:
            s_inc, h_inc = score, h.copy()
    return h_inc, s_inc

def polish(h, n, rng, n_trials):
    h = h.copy()
    score = exact_score(h, n)
    for _ in range(n_trials):
        i = rng.integers(0, n)
        d = rng.choice([-1, 1]) * rng.uniform(0.005, 0.04)
        ht = h.copy(); ht[i] += d
        hp = project_capped(ht, n)
        if hp is None:
            continue
        sp = exact_score(hp, n)
        if sp < score - 1e-12:
            h, score = hp, sp
    return h, score

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        try:
            v = int(kwargs.get('n_points', 47)); n = v if 1 <= v <= 999 else 47
        except Exception:
            n = 47
    if n == 1:
        return np.array([0.5]), exact_score(np.array([0.5]), 1), 1
    fb = project_capped(parent, n) if parent is not None else None
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    safe_h = fb.copy()
    safe_s = exact_score(fb, n)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    Phi, labels = build_fourier_basis(n)
    if n <= 64:
        ns, nsw, ng, nr, nrnd = 6, 2, 9, 2, 3
    elif n <= 256:
        ns, nsw, ng, nr, nrnd = 3, 2, 7, 2, 2
    else:
        ns, nsw, ng, nr, nrnd = 2, 1, 5, 1, 2
    starts = build_starts(fb, n, rng, ns)
    gb, gs = safe_h.copy(), safe_s
    for st in starts:
        ss = exact_score(st, n)
        if ss < gs - 1e-12:
            gs, gb = ss, st.copy()
    for st in starts:
        try:
            xo, so = optimize_start(st, Phi, labels, n, rng, nsw, ng, nr, nrnd)
            if xo is not None and is_feasible(xo, n):
                xs = exact_score(xo, n)
                if xs < gs - 1e-12:
                    gs, gb = xs, xo.copy()
        except Exception:
            pass
    try:
        ph, ps = polish(gb, n, rng, 300)
        if ps < gs - 1e-12:
            gs, gb = ps, ph.copy()
    except Exception:
        pass
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = safe_h.copy()
    fs = exact_score(final, n)
    if fs > safe_s + 1e-12:
        final, fs = safe_h.copy(), safe_s
    return final.astype(np.float64), float(fs), int(n)
