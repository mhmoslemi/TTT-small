Let me think about the gradient more carefully:
F_d(h) = sum_{i in I_d} h[i] * (1 - h[i+d])

dF_d/dh[r] = sum over terms where r appears:
- If r is in the "first" slot (i = r, so r+d is in range): contributes (1 - h[r+d])
- If r is in the "second" slot (i+d = r, so i = r-d is in range): contributes -h[r-d]

So grad[r] = p_d * [(1-h[r+d]) if r in "first" positions] + p_d * [-h[r-d]] if r in "second" positions]

For lag d:
- First positions: i from max(0,-d) to min(n, n-d)-1, i.e., h[i] appears
- Second positions: i+d from max(0,-d)+d to min(n,n-d)-1+d

Let me code this up:

```python
def lag_values(h, n):
    L = 2*n - 1
    vals = np.zeros(L)
    for d in range(-(n-1), n):
        lo = max(0, -d)
        hi = min(n, n-d)
        idx = np.arange(lo, hi)
        vals[d + n - 1] = np.sum(h[idx] * (1.0 - h[idx+d]))
    return vals
```

```python
import numpy as np

def project_capped(z, n):
    s = n / 2.0
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    if abs(r) > 1e-13:
        if r > 0:
            cap = 1.0 - h
            tc = float(np.sum(cap))
            if tc > 1e-14:
                h += (r / tc) * cap
        else:
            th = float(np.sum(h))
            if th > 1e-14:
                h += (r / th) * h
    h = np.clip(h, 0.0, 1.0)
    r2 = s - float(np.sum(h))
    if abs(r2) > 1e-14:
        for i in range(n):
            if r2 > 0 and h[i] < 1.0 - abs(r2):
                h[i] += r2
                break
            elif r2 < 0 and h[i] > abs(r2):
                h[i] += r2
                break
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-9 * max(1, n):
        return None
    return h

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return float('inf')
    if abs(float(np.sum(h)) - n / 2.0) > 1e-8 * max(1, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_values(h, n):
    L = 2 * n - 1
    vals = np.zeros(L)
    for d in range(-(n - 1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        vals[d + n - 1] = float(np.sum(h[lo:hi] * (1.0 - h[lo + d:hi + d])))
    return vals

def smooth_obj_grad(h, n, theta):
    L = 2 * n - 1
    F = lag_values(h, n)
    m = float(np.max(F))
    exponents = np.clip((F - m) / theta, -700.0, 700.0)
    e = np.exp(exponents)
    se = float(np.sum(e))
    if se <= 0.0 or not np.isfinite(se):
        return float('inf'), None
    p = e / se
    phi = (2.0 / n) * (m + theta * np.log(se))
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        pd = p[d + n - 1]
        if abs(pd) < 1e-16:
            continue
        lo = max(0, -d)
        hi = min(n, n - d)
        grad[lo:hi] += pd * (1.0 - h[lo + d:hi + d])
        grad[lo + d:hi + d] -= pd * h[lo:hi]
    grad *= (2.0 / n)
    return phi, grad

def continuation_optimize(h0, n, iters_per_stage, best_h, best_score):
    taus = [0.02, 0.006, 0.0018, 0.0005, 0.00015, 0.00004, 0.00001]
    h = h0.copy()
    for tau in taus:
        theta = max(1e-10, (n / 2.0) * tau)
        alpha = 0.2
        for _ in range(iters_per_stage):
            phi_h, g = smooth_obj_grad(h, n, theta)
            if not np.isfinite(phi_h) or g is None:
                break
            for bt in range(24):
                y = project_capped(h - alpha * g, n)
                if y is None:
                    alpha *= 0.5
                    continue
                d = y - h
                if np.max(np.abs(d)) < 1e-10:
                    alpha = 0.0
                    break
                phi_y, _ = smooth_obj_grad(y, n, theta)
                if not np.isfinite(phi_y):
                    alpha *= 0.5
                    continue
                gd = float(np.dot(g, d))
                tol = 1e-12 * max(1.0, abs(phi_h))
                if phi_y <= phi_h + 1e-4 * gd + tol:
                    h = y
                    if gd < -1e-12:
                        alpha = min(0.5, 1.15 * alpha)
                    break
                else:
                    alpha *= 0.5
            if alpha == 0.0 or np.max(np.abs(d)) < 1e-10:
                break
            sc = exact_score(h, n)
            if np.isfinite(sc) and sc < best_score - 1e-13:
                best_score = sc
                best_h = h.copy()
    return best_h, best_score

def _add_start(starts, v, n, tol=1e-8):
    h2 = project_capped(v, n)
    if h2 is None:
        return
    for ex in starts:
        if float(np.max(np.abs(h2 - ex))) < tol:
            return
    starts.append(h2)

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= len(iv) <= 999 and np.all(np.isfinite(iv)):
            n = len(iv)
            parent = iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
        n = max(1, min(999, n))
    try:
        sd = int(seed)
        if sd < 0:
            sd = 42
    except Exception:
        sd = 42
    rng = np.random.default_rng(sd)
    uniform = np.full(n, 0.5)
    if parent is not None:
        h0 = project_capped(parent, n)
        fallback = h0 if h0 is not None else uniform.copy()
    else:
        fallback = uniform.copy()
    fb_score = exact_score(fallback, n)
    if not np.isfinite(fb_score):
        fallback = uniform.copy()
        fb_score = exact_score(fallback, n)
    if n <= 64:
        n_starts_def, iters_def = 7, 60
    elif n <= 256:
        n_starts_def, iters_def = 5, 30
    else:
        n_starts_def, iters_def = 3, 12
    try:
        n_starts = max(1, min(int(kwargs.get('starts', n_starts_def)), 10))
    except Exception:
        n_starts = n_starts_def
    try:
        max_iter = max(1, min(int(kwargs.get('max_iter', iters_def)), 150))
    except Exception:
        max_iter = iters_def
    starts = []
    _add_start(starts, fallback, n)
    if parent is not None:
        _add_start(starts, fallback[::-1], n)
        _add_start(starts, 1.0 - fallback, n)
        _add_start(starts, 1.0 - fallback[::-1], n)
        _add_start(starts, np.roll(fallback, n // 4), n)
        _add_start(starts, np.roll(fallback, n // 3), n)
        _add_start(starts, fallback + rng.normal(0, 0.03, n), n)
        _add_start(starts, fallback + rng.normal(0, 0.10, n), n)
        _add_start(starts, fallback + rng.normal(0, 0.06, n), n)
    else:
        _add_start(starts, uniform, n)
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        _add_start(starts, blk, n)
        alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
        _add_start(starts, alt, n)
        tt = np.arange(n)
        _add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * tt / n), n)
        _add_start(starts, 0.5 + 0.45 * np.cos(4.0 * np.pi * tt / n), n)
        _add_start(starts, rng.uniform(0, 1, n), n)
        _add_start(starts, 0.5 + 0.3 * np.sin(2.0 * np.pi * tt / n) + 0.2 * np.cos(6.0 * np.pi * tt / n), n)
    starts = starts[:n_starts]
    best_h, best_score = fallback.copy(), fb_score
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_score - 1e-13:
            best_h, best_score = st.copy(), ss
    for st in starts:
        try:
            best_h, best_score = continuation_optimize(st, n, max_iter, best_h, best_score)
        except Exception:
            pass
    n_hops = 4 if n <= 128 else 2
    for _ in range(n_hops):
        try:
            sv = 0.02 + 0.03 * (_ % 3)
            pert = best_h + rng.normal(0, sv, n)
            hp = project_capped(pert, n)
            if hp is None:
                continue
            sc_p = exact_score(hp, n)
            if np.isfinite(sc_p) and sc_p < best_score - 1e-13:
                best_score = sc_p
                best_h = hp.copy()
            best_h, best_score = continuation_optimize(hp, n, max_iter, best_h, best_score)
        except Exception:
            pass
    final = project_capped(best_h, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score + 1e-12:
        final = fallback.copy()
        fs = fb_score
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final.astype(np.float64), float(fs), int(n)
```
