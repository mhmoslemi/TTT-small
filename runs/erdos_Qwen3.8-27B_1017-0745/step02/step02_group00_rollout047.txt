def project_capped(z,n):

def lag_values(h,n):
   q=np.empty(2*n-1)
   dx=2.0/n
   for idx,d in enumerate(range(-n+1,n)):
      if d>=0:
         q[idx]=float(np.dot(h[:n-d], 1.0-h[d:])*dx)
      else:
         m=-d
         q[idx]=float(np.dot(h[m:], 1.0-h[:-m])*dx)
   return q

def active_gradient(h,n,top_k):
   q=lag_values(h,n)
   L=q.size
   k=min(max(1,int(top_k)),L)
   if k==L: idxs=np.arange(L); w=1.0/L
   else: idxs=np.argpartition(q, L-k)[-k:]; w=1.0/k
   g=np.zeros(n)
   dx=2.0/n
   for idx in idxs:
      d=idx-(n-1)
      if d>=0:
         m=n-d
         a=h[:m]; b=h[d:]
         g[:m]+= w*dx*(1.0-b)
         g[d:]-= w*dx*a
      else:
         m=-d
         a=h[m:]; b=h[:-m]
         g[m:]+= w*dx*(1.0-b)
         g[:-m]-= w*dx*a
   return g, float(q.max())

def pair_sweep(h,n,score,top_k,max_pairs,fracs):
   g,_=active_gradient(h,n,top_k)
   # derivative matrix D rows dest increase, cols source decrease
   D = g[:,None] - g[None,:]
   D[np.diag_indices(n)] = 1e18
   flat = np.argsort(D.ravel())
   improved=False
   for flat_idx in flat[:max_pairs]:
       i=int(flat_idx//n); j=int(flat_idx%n)
       md=min(1.0-h[i], h[j])
       if md <= 1e-12: continue
       cur=score
       best=cur; best_d=0.0
       for f in fracs:

       if best_d !=0:
          h[i]+=best_d; h[j]-=best_d
          score=best; improved=True
   return h, score, improved

def active_gradient(h,n,k):
    q=lag_values(h,n)
    L=q.size; k=int(min(max(1,k),L))
    if k>=L:
       idx=np.arange(L); w=1.0/L

import numpy as np

def clamp_int(v, d, lo, hi):
    try:
        x = int(v)
    except Exception:
        return d
    if x < lo:
        return lo
    if x > hi:
        return hi

def pick_distinct(p, t, c, rg):

    while len(out) < c:

        ok = True
        if v == t:
            ok = False

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,):
        return None
    if not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = 0.5 * n
    lo = float(z.min()) - 1.0
    hi = float(z.max()) + 1.0
    for _ in range(70):
        m = 0.5 * (lo + hi)
        if m == lo or m == hi:
            break
        if float(np.sum(np.clip(z - m, 0.0, 1.0))) > s:
            lo = m
        else:
            hi = m
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for i in range(n):
        if abs(r) < 1e-13:
            break
        if r > 0.0:
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
        else:
            a = min(-r, h[i])
            h[i] -= a
            r += a

    return h


def feasible(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    return (h.shape == (n,) and np.all(np.isfinite(h)) and np.min(h) >= -1e-9 and
            np.max(h) <= 1.0 + 1e-9 and abs(float(np.sum(h)) - 0.5 * n) <= 1e-8)


def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if not feasible(h, n):
        return float('inf')
    h = np.clip(h, 0.0, 1.0)
    try:
        v = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
    except Exception:
        v = float(np.max(lag_values(h, n)))
    return v if np.isfinite(v) else float('inf')


def lag_values(h, n):
    q = np.empty(2 * n - 1, dtype=np.float64)
    dx = 2.0 / float(n)
    for idx, d in enumerate(range(-n + 1, n)):
        if d >= 0:
            q[idx] = float(np.dot(h[:n - d], 1.0 - h[d:])) * dx
        else:
            m = -d
            q[idx] = float(np.dot(h[m:], 1.0 - h[:-m])) * dx
    return q


def active_gradient(h, n, k):
    q = lag_values(h, n)
    L = q.size
    k = int(min(max(1, k), L))
    if k >= L:
        idx = np.arange(L)
        w = 1.0 / float(L)
    else:
        idx = np.argpartition(q, L - k)[-k:]
        w = 1.0 / float(k)
    g = np.zeros(n, dtype=np.float64)
    dx = 2.0 / float(n)
    for t in idx:
        d = int(t) - (n - 1)
        if d >= 0:
            m = n - d
            a = h[:m]
            b = h[d:]
            g[:m] += w * dx * (1.0 - b)
            g[d:] -= w * dx * a
        else:
            m = -d
            a = h[m:]
            b = h[:-m]
            g[m:] += w * dx * (1.0 - b)
            g[:-m] -= w * dx * a
    return g, float(q.max())


def proj_step(h, n, s, k, al):
    bh = h.copy()
    bs = float(s)
    try:
        g, _ = active_gradient(h, n, k)
    except Exception:
        return bh, bs
    for a in al:
        try:
            c = project_capped(h - a * g, n)
        except Exception:
            continue
        if c is None:
            continue
        sc = exact_score(c, n)
        if np.isfinite(sc) and sc < bs - 1e-12:
            bh = c
            bs = float(sc)
    return bh, bs


def pair_sweep(h, n, s, k, mp, fr):
    old = h.copy()
    olds = float(s)
    imp = False
    try:
        g, _ = active_gradient(h, n, k)
    except Exception:
        return old, olds, imp
    D = np.abs(g[:, None] - g[None, :])
    D[np.diag_indices(n)] = -1.0
    flat = np.argsort(D.ravel())[::-1][:mp]
    for f in flat:
        fi = int(f)
        i = fi // n
        j = fi % n
        if i == j:
            continue
        if g[i] >= g[j]:
            a, b = i, j
        else:
            a, b = j, i
        md = min(1.0 - h[a], h[b])
        if md <= 1e-13:
            continue
        best = float(s)
        bd = 0.0
        for frac in fr:
            d = frac * md
            if d <= 1e-14:
                continue
            h[a] += d
            h[b] -= d
            sc = exact_score(h, n)
            h[a] -= d
            h[b] += d
            if np.isfinite(sc) and sc < best - 1e-12:
                best = float(sc)
                bd = d
        if bd > 0.0:
            h[a] += bd
            h[b] -= bd
            s = best
            imp = True
    return h, float(s), imp


def local_refine(h, n, s, ro, ps, mp):
    al = (0.002, 0.008, 0.02, 0.05, 0.1, 0.3, 1.0, 3.0)
    tk = (1, 1, 2, 3)
    fr = (0.002, 0.008, 0.02, 0.05, 0.1, 0.2, 0.4, 0.7)
    h = h.copy()
    best = float(s)
    for _ in range(ro):
        for k in tk:
            h, bs = proj_step(h, n, best, k, al)
            if np.isfinite(bs) and bs < best:
                best = float(bs)
        for sw in range(ps):
            h, bs, _ = pair_sweep(h, n, best, tk[sw % 4], mp, fr)
            if np.isfinite(bs) and bs < best:
                best = float(bs)
    c = project_capped(h, n)
    if c is None:
        c = h
    sc = exact_score(c, n)
    if not np.isfinite(sc) or sc > best + 1e-9:
        c, sc = h, best
    return c, float(sc)


def try_add(hl, sl, raw, n):
    c = project_capped(raw, n)
    if c is None:
        return
    sc = exact_score(c, n)
    if not np.isfinite(sc):
        return
    for i in range(len(hl)):
        if float(np.max(np.abs(hl[i] - c))) < 1e-9:
            if sc < sl[i] - 1e-12:
                hl[i] = c
                sl[i] = sc
            return
    hl.append(c)
    sl.append(sc)


def nm_obj(z, n):
    c = project_capped(z, n)
    if c is None:
        return 2.0
    sc = exact_score(c, n)
    return sc if np.isfinite(sc) else 2.0


def nm_finish(h, n, s, it):
    bh = h.copy()
    bs = float(s)
    if minimize is None:
        return bh, bs
    try:
        res = minimize(nm_obj, h, method='Nelder-Mead', args=(n,),
                       options={'maxiter': max(1, it), 'fatol': 1e-12, 'xatol': 1e-12, 'adaptive': True})
    except Exception:
        return bh, bs
    if res is not None and getattr(res, 'x', None) is not None:
        c = project_capped(res.x, n)
        if c is not None:
            sc = exact_score(c, n)
            if np.isfinite(sc) and sc < bs - 1e-12:
                bh = c
                bs = float(sc)
    return bh, bs


def de_search(X, S, n, rg, gen, p, st, rs):
    F = 0.45 + 0.4 * rg.random(p)
    CR = 0.35 + 0.55 * rg.random(p)
    bi = int(np.argmin(S))
    no = 0
    rl = rs
    gd = 0
    while gd < gen and rl >= 0:
        X0 = X.copy()
        S0 = S.copy()
        F0 = F.copy()
        CR0 = CR.copy()
        b0 = int(np.argmin(S0))
        imp = False
        order = rg.permutation(p)
        for i in order:
            fi = float(F0[i])
            cri = float(CR0[i])
            if rg.random() < 0.1:
                fi = float(0.1 + 0.8 * rg.random())
            if rg.random() < 0.1:
                cri = float(0.05 + 0.9 * rg.random())
            F0[i] = fi
            CR0[i] = cri
            if rg.random() < 0.2 and b0 != i:
                v = pick_distinct(p, i, 2, rg)
                don = X0[i] + fi * (X0[b0] - X0[i]) + fi * (X0[v[0]] - X0[v[1]])
            else:
                v = pick_distinct(p, i, 3, rg)
                don = X0[v[0]] + fi * (X0[v[1]] - X0[v[2]])
            mask = rg.random(n) < cri
            mask[int(rg.integers(n))] = True
            trial = X0[i] + np.where(mask, don - X0[i], 0.0)
            c = project_capped(trial, n)
            if c is None:
                continue
            sc = exact_score(c, n)
            if not np.isfinite(sc):
                continue
            if sc < S0[i] - 1e-12:
                X[i] = c
                S[i] = sc
            if sc < S[bi] - 1e-12:
                bi = i
                imp = True
        F, CR = F0, CR0
        gd += 1
        if imp:
            no = 0
        else:
            no += 1
            if no >= st and rl > 0:
                rl -= 1
                no = 0
                ord_s = np.argsort(S)
                keep = max(2, p // 8)
                ki = ord_s[:keep].tolist()
                if bi not in ki:
                    ki[-1] = bi
                Xn = X.copy()
                Sn = S.copy()
                for pos, ii in enumerate(ki):
                    Xn[pos] = X[ii]
                    Sn[pos] = S[ii]
                rnum = rs - rl
                sig = min(0.25, 0.04 * (2.0 ** rnum))
                for pos in range(keep, p):
                    if rg.random() < 0.7:
                        raw = X[bi] + sig * rg.normal(0.0, 1.0, n)
                    else:
                        raw = 0.5 + [0.15, 0.4, 0.8][int(rg.integers(3))] * rg.normal(0.0, 1.0, n)
                    c = project_capped(raw, n)
                    if c is None:
                        c = X[ord_s[-1]].copy()
                    sc = exact_score(c, n)
                    Xn[pos] = c
                    Sn[pos] = sc
                X, S = Xn, Sn
                bi = int(np.argmin(S))
                F = 0.45 + 0.4 * rg.random(p)
                CR = 0.35 + 0.55 * rg.random(p)
            elif no >= st:
                break
    return X[bi].copy(), float(S[bi]), X.copy(), S.copy()


def run(seed=42, **kwargs):
    n = None
    par = None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= iv.size <= 999:
            n = iv.size
            if np.all(np.isfinite(iv)) and np.all(iv >= -1e-8) and np.all(iv <= 1.0 + 1e-8):
                par = iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
    n = max(1, min(n, 999))

    try:
        sd = int(seed)
    except Exception:
        sd = 42
    if sd == 0:
        sd = 42
    rg = np.random.default_rng(abs(sd) % (2 ** 32) or 42)

    if n == 1:
        return np.array([0.5], dtype=np.float64), 0.5, 1

    if n <= 64:
        P, gen, st, rs, lo, ps, mp, pmax, gmax = 96, 120, 18, 2, 2, 8, 120, 96, 160
    elif n <= 128:
        P, gen, st, rs, lo, ps, mp, pmax, gmax = 56, 60, 12, 2, 2, 6, 90, 64, 80
    elif n <= 256:
        P, gen, st, rs, lo, ps, mp, pmax, gmax = 36, 30, 7, 1, 1, 4, 50, 40, 35
    elif n <= 512:
        P, gen, st, rs, lo, ps, mp, pmax, gmax = 24, 12, 4, 1, 1, 2, 30, 24, 15
    else:
        P, gen, st, rs, lo, ps, mp, pmax, gmax = 14, 5, 2, 1, 1, 1, 18, 16, 6

    P = clamp_int(kwargs.get('pop_size', P), P, max(4, min(12, pmax)), pmax)
    gen = clamp_int(kwargs.get('generations', gen), gen, 0, gmax)
    st = clamp_int(kwargs.get('stall_generations', st), st, 1, max(1, gen))
    rs = clamp_int(kwargs.get('restarts', rs), rs, 0, 3 if n <= 64 else (2 if n <= 128 else 1))

    hl = []
    sl = []
    try_add(hl, sl, np.full(n, 0.5), n)

    pp = None
    if par is not None:
        pp = project_capped(par, n)

    if pp is not None:
        for vec in (pp, pp[::-1], 1.0 - pp, 1.0 - pp[::-1]):
            try_add(hl, sl, vec, n)
        for sc in (0.01, 0.02, 0.05, 0.1, 0.2, 0.35):
            try_add(hl, sl, pp + sc * rg.standard_normal(n), n)

    idx = np.arange(n, dtype=np.float64)
    for f in (1, 2, 3, 5):
        for ph in (0.0, 0.5 * math.pi):
            try_add(hl, sl, 0.5 + 0.5 * np.cos((2.0 * math.pi * f / n) * idx + ph), n)

    xx = idx / max(1, n - 1)
    for amp in (0.3, 0.45, 0.6):
        for base in (np.sin(math.pi * xx), np.cos(math.pi * xx), np.cos(2.0 * math.pi * xx)):
            try_add(hl, sl, 0.5 + amp * base, n)
            try_add(hl, sl, 0.5 - amp * base, n)

    if n > 3 and n % 2 == 1:
        res = np.zeros(n, dtype=bool)
        for i in range(1, n):
            res[(i * i) % n] = True
        base = res.astype(np.float64)
        base[0] = 0.5
        try_add(hl, sl, base, n)
        for sft in range(min(12, n - 1)):
            try_add(hl, sl, np.roll(base, sft), n)

    for b in (2, 3, 4, 5, 7):
        per = 2 * b
        mask = (np.arange(n) % per) < b
        raw = np.zeros(n)
        raw[mask] = 1.0
        try_add(hl, sl, raw, n)
        try_add(hl, sl, 1.0 - raw, n)

    if len(hl) > P:
        hl = hl[:P]
        sl = sl[:P]

    att = 0
    while len(hl) < P and att < 60 * P:
        att += 1
        if att % 3 == 1:
            scale = (0.02, 0.05, 0.1, 0.2, 0.4, 0.8)[att % 6]
            raw = 0.5 + scale * rg.standard_normal(n)
        elif att % 3 == 2:
            nf = 3
            c = rg.normal(0.0, 0.25, 2 * nf)
            raw = np.full(n, 0.5)
            for k in range(nf):
                om = 2.0 * math.pi * (k + 1) / n
                raw += c[2 * k] * np.cos(om * idx) + c[2 * k + 1] * np.sin(om * idx)
            raw += 0.2 * rg.standard_normal(n)
        else:
            raw = 0.5 * rg.random(n) + 0.5 * np.roll(pp if pp is not None else hl[0], att % max(1, n - 1))
        try_add(hl, sl, raw, n)

    while len(hl) < P:
        hl.append(hl[0].copy())
        sl.append(float(sl[0]))

    X = np.vstack(hl)
    S = np.array(sl, dtype=np.float64)
    bi = int(np.argmin(S))
    best_h = X[bi].copy()
    best_s = float(S[bi])

    Xd, Sd = X.copy(), S.copy()
    de_h = None
    de_s = None
    if gen > 0 and P >= 4:
        de_h, de_s, Xd, Sd = de_search(X, S, n, rg, gen, P, st, rs)
        if np.isfinite(de_s) and de_s < best_s - 1e-12:
            best_h = de_h
            best_s = float(de_s)

    ll = []
    ls = []
    try_add(ll, ls, best_h, n)
    if de_h is not None:
        try_add(ll, ls, de_h, n)
    order = np.argsort(Sd)
    for oi in order[:3]:
        try_add(ll, ls, Xd[oi], n)
    if pp is not None:
        try_add(ll, ls, pp, n)
    if len(ll) > 4:
        ll = ll[:4]
        ls = ls[:4]

    for ii in range(len(ll)):
        h2, s2 = local_refine(ll[ii], n, ls[ii], lo, ps, mp)
        if np.isfinite(s2) and s2 < best_s - 1e-12:
            best_h = h2
            best_s = float(s2)

    if n <= 128:
        best_h, best_s = nm_finish(best_h, n, best_s, 350 if n <= 64 else 120)

    fin = project_capped(best_h, n)
    if fin is None:
        fin = best_h
    fs = exact_score(fin, n)
    if not np.isfinite(fs) or fs > best_s + 1e-9:
        fin = np.asarray(best_h, dtype=np.float64).ravel()
        fs = exact_score(fin, n)
        if not np.isfinite(fs):
            fin = np.full(n, 0.5, dtype=np.float64)
            fs = exact_score(fin, n)

    return np.asarray(fin, dtype=np.float64).ravel(), float(fs), int(n)
