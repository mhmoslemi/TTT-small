def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):

    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)

    return h

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12) or abs(np.sum(h) - n / 2.0) > 1e-9:
        return float('inf')
    v = float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))
    return v if np.isfinite(v) else float('inf')

def lags_and_jac(h, n):
    dx = 2.0 / n
    Q = np.empty(2 * n - 1)
    J = np.zeros((2 * n - 1, n), dtype=np.float64)
    Q[0] = float(np.dot(h, 1.0 - h)) * dx
    J[0, :] = (1.0 - 2.0 * h) * dx
    for r in range(1, n):
        m = n - r
        Q[r] = float(np.dot(h[:m], 1.0 - h[r:])) * dx
        J[r, :m] += (1.0 - h[r:]) * dx
        J[r, r:] += (-h[:m]) * dx
        Q[n - 1 + r] = float(np.dot(h[r:], 1.0 - h[:m])) * dx
        J[n - 1 + r, r:] += (1.0 - h[:m]) * dx
        J[n - 1 + r, :m] += (-h[r:]) * dx
    return Q, J

def slsqp_obj(z, n):
    return z[-1]

def slsqp_obj_jac(z, n):
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return g

def slsqp_ineq(z, n):

    Q, _ = lags_and_jac(h, n)
    return z[-1] - Q

    h = z[:n]
    _, J = lags_and_jac(h, n)
    A = np.zeros(J.shape + (1,))
    A[:, :, 0] = -J
    A[:, :, 1] = 1.0

def solve_epigraph(h0, n):
    t0 = exact_score(h0, n) + 1e-6
    if not np.isfinite(t0) or t0 > 2.0:
        t0 = 1.0
    z0 = np.concatenate([h0, [t0]])

    cons = [
        {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac, 'args': (n,)},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac, 'args': (n,)},
    ]
    try:

                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'ftol': 1e-10, 'maxiter': 300, 'disp': False})

def smooth_grad(h, n, tau):
    dx = 2.0 / n
    Q, _ = lags_and_jac(h, n)
    qmax = np.max(Q)
    sh = Q - qmax
    p = np.exp(sh)
    p /= np.sum(p)
    g = np.zeros(n)
    g += p[0] * dx * (1.0 - 2.0 * h)
    for r in range(1, n):
        m = n - r
        g[:m] += p[r] * dx * (1.0 - h[r:])
        g[r:] += p[r] * (-dx) * h[:m]
        g[r:] += p[n - 1 + r] * dx * (1.0 - h[:m])
        g[:m] += p[n - 1 + r] * (-dx) * h[r:]
    return g

    best = h.copy()
    for tau in [0.02, 0.005, 0.001, 0.0004]:

            alpha = min(1.5 * alpha, 8.0)
    return best, bs

            if np.max(np.abs(v - s)) < 1e-9:
                dup = True
                break
        if not dup:
            pool.append(v)
    return pool

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999 and np.all(np.isfinite(p)):
            if np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                n = int(p.size)
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
        if n < 1 or n > 999:
            n = 47
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if parent is not None:
        fb = project_capped(np.clip(parent, 0.0, 1.0), n)
    else:
        fb = np.full(n, 0.5)
    if fb is None or not np.all(np.isfinite(fb)):
        fb = np.full(n, 0.5)
    if n == 1:
        h1 = np.array([0.5])
        return h1, float(exact_score(h1, 1)), 1
    fb_s = exact_score(fb, n)
    best_h = fb.copy()
    best_s = fb_s
    starts = build_starts(fb, n, rng)
    max_starts = 12 if n <= 64 else (8 if n <= 128 else 5)
    starts = starts[:max_starts]
    for s0 in starts:
        ss = exact_score(s0, n)
        if ss < best_s - 1e-12:
            best_s = ss
            best_h = s0.copy()
    for s0 in starts:
        h_sol = solve_epigraph(s0, n)
        if h_sol is not None:
            hs = exact_score(h_sol, n)
            if hs < best_s - 1e-12:
                best_s = hs
                best_h = h_sol.copy()
        for a in [0.25, 0.5, 0.75]:
            hk = project_capped((1 - a) * s0 + a * (h_sol if h_sol is not None else s0), n)
            if hk is not None:
                hs = exact_score(hk, n)
                if hs < best_s - 1e-12:
                    best_s = hs
                    best_h = hk.copy()
    me = 15 if n <= 128 else 20
    for s0 in sorted(starts, key=lambda x: exact_score(x, n))[:3]:
        hr, sr = smooth_refine(s0, n, me, rng)
        if sr < best_s - 1e-12:
            best_s = sr
            best_h = hr.copy()
    hr, sr = smooth_refine(best_h, n, me, rng)
    if sr < best_s - 1e-12:
        best_s = sr
        best_h = hr.copy()
    final = project_capped(best_h, n)
    if final is None:
        final = fb.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs):
        final = fb.copy()
        fs = exact_score(final, n)
    return final.astype(np.float64), float(fs), int(n)
