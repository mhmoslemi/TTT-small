def is_duplicate(v, pool, tol=1e-5):
   if not pool: return False

def project_capped(z, n):
    if z is None:
        return None
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    target = 0.5 * float(n)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(z - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - float(np.sum(h))
    idxs = np.arange(n)
    if r >= 0.0:
        for q in idxs:
            if abs(r) < 1e-13:

            d = min(r, 1.0 - h[q])

                break
            d = min(-r, h[q])
            if d > 0.0:
                h[q] -= d
                r += d
    if not np.all(np.isfinite(h)) or abs(r) > 1e-8 * max(1, n):
        return None
    return np.clip(h, 0.0, 1.0)

def exact_score(h,n):
    h=np.asarray(h,dtype=np.float64).ravel()
    if h.size!=n or not np.all(np.isfinite(h)): return float('inf')
    if np.any(h<-1e-8) or np.any(h>1+1e-8): return float('inf')
    h=np.clip(h,0,1)
    if abs(np.sum(h)-0.5*n)>1e-7*max(1,n): return float('inf')
    c=np.correlate(h,1-h,mode='full')*(2.0/n)
    if c.size!=2*n-1 or not np.all(np.isfinite(c)): return float('inf')
    return float(np.max(c))

def lag_jac(h,n):
    h=np.asarray(h,dtype=np.float64).ravel()
    if h.size!=n or not np.all(np.isfinite(h)): return np.zeros((2*n-1,n))
    J=np.zeros((2*n-1,n)); dx=2.0/n
    for idx,k in enumerate(range(-(n-1),n)):
        i=np.arange(max(0,-k), min(n,n-k))
        a=i+k
        np.add.at(J[idx,:],a,1-h[i]); np.add.at(J[idx,:],i,-h[a])
    return J*dx

def solve_epigraph(h0,n,maxiter,ftol):
    h0=project_capped(h0,n)
    if h0 is None: return []
    s0=exact_score(h0,n)
    if not np.isfinite(s0): return []
    z0=np.empty(n+1); z0[:-1]=h0; z0[-1]=min(0.5,max(0.0,s0+1e-6))
    c=[z0[:-1].copy()]
    b=[(0.0,1.0)]*n+[(0.0,0.5)]
    cons=({'type':'eq','fun':eq_func,'jac':eq_jac},{'type':'ineq','fun':ineq_func,'jac':ineq_jac})
    try:
        r=minimize(obj_func,z0,jac=obj_jac,method='SLSQP',bounds=b,constraints=cons,options={'maxiter':int(maxiter),'ftol':ftol,'disp':False})

        if x.size==n+1 and np.all(np.isfinite(x)): c.append(x[:-1].copy())

def is_duplicate(v,pool,tol=1e-5):
    v=np.asarray(v,dtype=np.float64).ravel()
    if v.size==0 or not np.all(np.isfinite(v)): return True
    for p in pool:
        if np.max(np.abs(v-p))<tol: return True

import numpy as np
from scipy.optimize import minimize

def project_capped(z, n):
    if z is None:
        return None
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    t = 0.5 * float(n)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        m = 0.5 * (lo + hi)
        if float(np.sum(np.clip(z - m, 0.0, 1.0))) > t:
            lo = m
        else:
            hi = m
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = t - float(np.sum(h))
    for q in range(n):
        if abs(r) < 1e-13:
            break
        if r > 0.0:
            d = min(r, 1.0 - h[q])
            if d > 0.0:
                h[q] += d
                r -= d
        else:
            d = min(-r, h[q])
            if d > 0.0:
                h[q] -= d
                r += d
    if not np.all(np.isfinite(h)) or abs(r) > 1e-7 * max(1, n):
        return None
    return np.clip(h, 0.0, 1.0)

def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.size != n or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-8) or np.any(h > 1.0 + 1e-8):
        return float('inf')
    h = np.clip(h, 0.0, 1.0)
    if abs(float(np.sum(h)) - 0.5 * float(n)) > 1e-7 * max(1, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / float(n))
    if c.size != 2 * n - 1 or not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def lag_jac(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.size != n or not np.all(np.isfinite(h)):
        return np.zeros((2 * n - 1, n), dtype=np.float64)
    J = np.zeros((2 * n - 1, n), dtype=np.float64)
    dx = 2.0 / float(n)
    for idx, k in enumerate(range(-(n - 1), n)):
        i = np.arange(max(0, -k), min(n, n - k))
        a = i + k
        np.add.at(J[idx, :], a, 1.0 - h[i])
        np.add.at(J[idx, :], i, -h[a])
    return J * dx

def obj_func(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0 or not np.isfinite(z[-1]):
        return 0.0
    return float(np.clip(z[-1], -1e6, 1e6))

def obj_jac(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    J = np.zeros(len(z), dtype=np.float64)
    J[-1] = 1.0
    return J

def eq_func(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size <= 1:
        return 0.0
    return float(np.sum(z[:-1]) - 0.5 * float(len(z) - 1))

def eq_jac(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    J = np.ones(len(z), dtype=np.float64)
    if len(z) > 0:
        J[-1] = 0.0
    return J

def ineq_func(z):

    if z.size <= 1:
        return np.empty(0, dtype=np.float64)
    n = len(z) - 1

def ineq_jac(z):
    z = np.asarray(z, dtype=np.float64).ravel()

        return np.empty((0, z.size), dtype=np.float64)
    n = len(z) - 1

    out = np.empty((2 * n - 1, n + 1), dtype=np.float64)
    out[:, :n] = -J
    out[:, n] = 1.0
    return out

def solve_epigraph(h0, n, maxiter, ftol):

    if h0 is None:
        return []
    s0 = exact_score(h0, n)
    if not np.isfinite(s0):
        return []

    z0[:-1] = h0
    z0[-1] = min(0.5, max(0.0, s0 + 1e-6))
    cands = [z0[:-1].copy()]
    bounds = [(0.0, 1.0)] * n + [(0.0, 0.5)]
    cons = ({'type': 'eq', 'fun': eq_func, 'jac': eq_jac},
            {'type': 'ineq', 'fun': ineq_func, 'jac': ineq_jac})

                     bounds=bounds, constraints=cons,

        x = np.asarray(r.x, dtype=np.float64).ravel()
        if x.size == n + 1 and np.all(np.isfinite(x)):
            cands.append(x[:-1].copy())
    except Exception:
        pass
    return cands

def is_duplicate(v, pool, tol=1e-5):
    v = np.asarray(v, dtype=np.float64).ravel()
    if v.size == 0 or not np.all(np.isfinite(v)):
        return True
    for p in pool:
        if np.max(np.abs(v - p)) < tol:
            return True
    return False

        for cand in solve_epigraph(best_h, n, maxit, ftol):
            best_h, best_s = consider_candidate(best_h, best_s, cand, n)
    fh = project_capped(best_h, n)
    fs = exact_score(fh, n) if fh is not None else float('inf')
    if not np.isfinite(fs) or fs > best_s + 1e-7:
        fh = fb
        fs = exact_score(fh, n)
    if not np.isfinite(fs):
        fh = uni
        fs = exact_score(uni, n)
    return fh, float(fs), int(n)
