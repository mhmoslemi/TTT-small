            return []
        x = -c / b
        return [max(lo, min(hi, x))] if lo - 1e-10 <= x <= hi + 1e-10 else []
    disc = b * b - 4.0 * a * c
    if disc < -1e-12:
        return []
    if disc < 0:
        disc = 0.0
    sq = np.sqrt(disc)
    q = -0.5 * (b + sq) if b >= 0 else -0.5 * (b - sq)
    r1 = q / a
    r2 = c / q if abs(q) > 1e-30 else -b / a
    out = []
    for r in (r1, r2):
        if np.isfinite(r) and lo - 1e-10 <= r <= hi + 1e-10:
            out.append(max(lo, min(hi, r)))
    return out

def line_search(h, v, n, cur_score):
    iv = feasible_interval(h, v, n)
    if iv is None:
        return h, cur_score
    lo, hi = iv
    if hi - lo < 1e-13:
        return h, cur_score
    A, B, C = line_coeffs(h, v, n)
    L = len(A)
    if not (np.all(np.isfinite(A)) & np.all(np.isfinite(B)) & np.all(np.isfinite(C))).all():
        return h, cur_score
    alphas = [lo, hi, 0.0]
    for s in range(L):
        if abs(C[s]) > 1e-16:
            x = -B[s] / (2.0 * C[s])
            if np.isfinite(x) and lo - 1e-10 <= x <= hi + 1e-10:
                alphas.append(max(lo, min(hi, x)))
    if n <= 64:
        for s in range(L):
            for t in range(s + 1, L):
                alphas.extend(quad_roots_in(C[s]-C[t], B[s]-B[t], A[s]-A[t], lo, hi))
    alphas = [a for a in alphas if np.isfinite(a)]
    alphas.sort()
    merged = [alphas[0]]
    tol = 1e-10 * max(1.0, abs(lo), abs(hi))
    for a in alphas[1:]:
        if a - merged[-1] > tol:
            merged.append(a)
    best_a, best_f = 0.0, cur_score
    for i in range(0, len(merged), 512):
        batch = np.array(merged[i:i+512])
        vals = A[None,:] + B[None,:]*batch[:,None] + C[None,:]*batch[:,None]**2
        fs = np.max(vals, axis=1)
        for j in range(len(batch)):
            if fs[j] < best_f - 1e-13:
                best_f = fs[j]; best_a = batch[j]
    if best_a == 0.0:
        return h, cur_score
    h2 = project_capped(h + best_a * v, n)
    if h2 is None:
        return h, cur_score
    s2 = exact_score(h2, n)
    return (h2, s2) if s2 < cur_score - 1e-12 else (h, cur_score)

def dct_mode(k, n):
    i = np.arange(n, dtype=np.float64)
    v = cos(pi * k * (i + 0.5) / n)
    v -= np.mean(v)
    v[-1] -= np.sum(v)
    m = np.max(np.abs(v))
    return v / m if m > 1e-15 else None

def box_direction(h, n, rng):
    d_idx = np.where(h > 1e-6)[0]
    r_idx = np.where(h < 1.0 - 1e-6)[0]
    if len(d_idx) < 1 or len(r_idx) < 1:
        return None
    ds = set(d_idx.tolist())
    d_only = np.array([x for x in d_idx if x not in set(r_idx.tolist())])
    r_only = np.array([x for x in r_idx if x not in ds])
    if len(d_only) < 1 or len(r_only) < 1:
        return None
    rng.shuffle(d_only); rng.shuffle(r_only)
    nd = min(4, len(d_only)); nr = min(4, len(r_only))
    v = np.zeros(n)
    dw = rng.random(nd); rw = rng.random(nr)
    dw /= dw.sum(); rw /= rw.sum()
    v[d_only[:nd]] = -dw; v[r_only[:nr]] = rw
    m = np.max(np.abs(v))
    return v / m if m > 1e-15 else None

def get_freqs(n, count):
    if count >= n - 1:
        return list(range(1, n))
    f = np.unique(np.round(np.logspace(0, np.log10(n-1), count)).astype(int))
    f = f[(f >= 1) & (f <= n - 1)]
    return [int(x) for x in f] if len(f) else [1]

def sweep(h, n, rng, n_pass):
    cur = h.copy(); cs = exact_score(cur, n)
    nm = n - 1 if n <= 48 else (24 if n <= 64 else (12 if n <= 256 else 8))
    freqs = get_freqs(n, nm)
    for _ in range(n_pass):
        rng.shuffle(freqs)
        imp = False
        for k in freqs:
            v = dct_mode(k, n)
            if v is None:
                continue
            h2, s2 = line_search(cur, v, n, cs)
            if s2 < cs - 1e-12:
                cur, cs = h2, s2; imp = True
        for _ in range(2):
            v = box_direction(cur, n, rng)
            if v is None:
                continue
            h2, s2 = line_search(cur, v, n, cs)
            if s2 < cs - 1e-12:
                cur, cs = h2, s2; imp = True
        if not imp:
            break
    return cur, cs

def build_starts(fb, n, rng, cnt):
    pool = [fb.copy()]
    jc = np.arange(n, dtype=np.float64)
    cands = [np.full(n, 0.5),
             np.concatenate([np.ones(n//2), ([0.5] if n%2 else []), np.zeros(n - n//2 - (n%2))]),
             np.clip(0.5 + 0.45*cos(2*pi*(jc+0.5)/n), 0, 1),
             rng.uniform(0, 1, n),
             np.clip(0.5 + 0.35*cos(pi*(jc+0.5)/n), 0, 1)]
    for c in cands:
        if len(pool) >= cnt:
            break
        p = project_capped(c, n)
        if p is None:
            continue
        if not any(np.max(np.abs(p - s)) < 1e-8 for s in pool):
            pool.append(p)
    return pool[:cnt]

def run(seed=42, **kwargs):
    n = None; parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size >= 1 and np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
            n = int(p.size); parent = p
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999: n = v
        except Exception:
            pass
    if parent is not None and len(parent) != n:
        old_n = len(parent)
        roi = np.clip((np.arange(n) + 0.5) / n * old_n - 0.5, 0, old_n - 1)
        i0 = np.floor(roi).astype(int); i1 = np.minimum(i0 + 1, old_n - 1)
        fr = roi - i0
        parent = parent[i0]*(1-fr) + parent[i1]*fr
    fb = project_capped(parent, n) if parent is not None else np.full(n, 0.5)
    if fb is None:
        fb = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    fs = exact_score(fb, n)
    gb, gs = fb.copy(), fs
    if n == 1:
        return (fb, fs, 1)
    ns = 3 if n <= 256 else 2
    np_ = 2 if n <= 64 else 1
    starts = build_starts(fb, n, rng, ns)
    for st in starts:
        s = exact_score(st, n)
        if s < gs - 1e-12:
            gs, gb = s, st.copy()
    starts.sort(key=lambda x: exact_score(x, n))
    for st in starts:
        try:
            xo, so = sweep(st, n, rng, 1)
            if so < gs - 1e-12:
                gs, gb = so, xo.copy()
        except Exception:
            continue
    for _ in range(np_):
        try:
            hn, sn = sweep(gb, n, rng, 1)
            if sn >= gs - 1e-12:
                break
            gs, gb = sn, hn
        except Exception:
            break
    if n <= 64:
        for _ in range(3):
            try:
                pt = project_capped(gb + rng.normal(0, 0.03, n), n)
                if pt is None:
                    continue
                hn, sn = sweep(pt, n, rng, 1)
                if sn < gs - 1e-12:
                    gs, gb = sn, hn
            except Exception:
                continue
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb
    sf = exact_score(final, n)
    ss = exact_score(fb, n)
    if sf > ss:
        final, sf = fb, ss
    if not is_feasible(final, n):
        final = np.full(n, 0.5); sf = exact_score(final, n)
    return (final, float(sf), int(n))
