            if d < -1e-12 or (abs(d) <= 1e-12 and rng.random() < 0.5) or (d > 1e-12 and np.exp(-d / T) > rng.random()):
                pop[i], scs[i] = p2, sp
            if sp < gs - 1e-12:
                gb, gs = p2.copy(), sp
        pop[0], scs[0] = gb.copy(), gs
        if gs < pgs - 1e-12:
            stall = 0
        else:
            stall += 1
            if stall >= 25 and g < G - 1:
                for k in range(4, P):
                    np2 = gb + (0.025 + 0.075 * (1 - q)) * rng.normal(0, 1, n) if rng.random() < 0.6 else rng.uniform(0, 1, n)
                    pp = project_capped(np2, n)
                    if pp is None:
                        pp = gb.copy()
                    pop[k], scs[k] = pp, sc(pp, n)
                    if scs[k] < gs - 1e-12:
                        gb, gs = pop[k].copy(), scs[k]
                pop[0], scs[0] = gb.copy(), gs
                stall = 0
    x, sv = gb.copy(), gs
    for t in range(L):
        q = t / max(1, L - 1)
        T = 5e-4 * (1e-7 / 5e-4) ** q
        ri, rj = int(rng.integers(n)), int(rng.integers(n))
        if ri == rj:
            continue
        ci = min(1.0 - x[ri], x[rj])
        cj = min(1.0 - x[rj], x[ri])
        xn = None
        if ci > 1e-12 and (cj <= 1e-12 or rng.random() < 0.5):
            am = ci * (1 - 0.98 * q) * (0.05 + 0.95 * rng.random())
            xn = x.copy(); xn[ri] += am; xn[rj] -= am
        elif cj > 1e-12:
            am = cj * (1 - 0.98 * q) * (0.05 + 0.95 * rng.random())
            xn = x.copy(); xn[rj] += am; xn[ri] -= am
        if xn is None:
            continue
        pn = project_capped(xn, n)
        if pn is None:
            continue
        sn = sc(pn, n)
        d = sn - sv
        if d < -1e-12 or (d >= -1e-12 and np.exp(-max(d / T, 0.0)) > rng.random()):
            x, sv = pn, sn
            if sn < gs - 1e-12:
                gb, gs = x.copy(), sn
    x = gb.copy()
    fx = sc(x, n)
    fr = np.array([1.0 / 64, 1.0 / 32, 1.0 / 16, 1.0 / 8, 0.25, 0.5, 0.75, 1.0])
    for _ in range(NSW):
        imp = False
        for i in range(n):
            for j in range(i + 1, n):
                bs, bx = fx, None
                ci = min(1.0 - x[i], x[j])
                if ci > 1e-12:
                    for f in fr:
                        ni, nj = x[i] + f * ci, x[j] - f * ci
                        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                            xn = x.copy(); xn[i], xn[j] = ni, nj
                            sv2 = sc(xn, n)
                            if sv2 < bs - 1e-12:
                                bs, bx = sv2, xn
                cj = min(1.0 - x[j], x[i])
                if cj > 1e-12:
                    for f in fr:
                        ni, nj = x[i] - f * cj, x[j] + f * cj
                        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                            xn = x.copy(); xn[i], xn[j] = ni, nj
                            sv2 = sc(xn, n)
                            if sv2 < bs - 1e-12:
                                bs, bx = sv2, xn
                if bx is not None:
                    x, fx, imp = bx, bs, True
                    if fx < gs - 1e-12:
                        gb, gs = x.copy(), fx
        if not imp:
            break
    final = project_capped(gb, n)
    if final is None or not ok(final, n) or sc(final, n) > safe_s + 1e-12:
        final = safe_h.copy()
    fs = sc(final, n)
    return final.astype(np.float64), float(fs), int(n)
