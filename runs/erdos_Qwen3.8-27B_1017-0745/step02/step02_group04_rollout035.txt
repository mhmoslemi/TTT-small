import numpy as np
import math

ML = 65536

def project_capped(z, n):
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(3):
        if abs(r) < 1e-12:
            break
        if r > 0:
            for i in np.argsort(-(1.0 - h)):
                if r <= 1e-13:
                    break
                a = min(1.0 - h[i], r)
                h[i] += a
                r -= a
        else:
            for i in np.argsort(h):
                if -r <= 1e-13:
                    break
                a = min(h[i], -r)
                h[i] -= a
                r += a
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - s) > 1e-9:
        return None
    return h

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(np.sum(h) - n / 2.0) <= 1e-9)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def to_units(h, n):
    if not is_feasible(h, n):
        return None
    K = ML * n // 2
    z = np.clip(h, 0.0, 1.0) * ML
    base = np.floor(z).astype(np.int64)
    frac = z - base.astype(np.float64)
    R = K - int(np.sum(base))
    if R > 0:
        order = np.argsort(-frac)
        cnt = 0
        for i in order:
            if cnt >= R:
                break
            if base[i] < ML:
                base[i] += 1
                cnt += 1
        if cnt < R:
            return None
    elif R < 0:
        order = np.argsort(frac)
        cnt = 0
        for i in order:
            if cnt >= -R:
                break
            if base[i] > 0:
                base[i] -= 1
                cnt += 1
        if cnt < -R:
            return None
    if np.any(base < 0) or np.any(base > ML) or int(np.sum(base)) != K:
        return None
    return base
