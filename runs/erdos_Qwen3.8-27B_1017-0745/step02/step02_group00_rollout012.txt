                    delta = max(0.002, delta * 0.75)

    candidates = [(best, best_score)]
    for fb, fs in fallbacks:
        candidates.append((fb, fs))
    candidates.append((uniform, us))
    final, final_score = None, float('inf')
    for cand, cs in candidates:
        cand = np.asarray(cand, dtype=np.float64).ravel()
        if cand.shape != (n,) or not np.all(np.isfinite(cand)):
            continue
        if np.any(cand < -1e-8) or np.any(cand > 1.0 + 1e-8):
            cand = project_capped(cand, n)
            if cand is None:
                continue
        cs = exact_score(cand, n)
        if np.isfinite(cs) and cs < final_score:
            final, final_score = cand.copy(), cs
    if final is None:
        final, final_score = uniform.copy(), us
    final_score = exact_score(final, n)
    if not np.isfinite(final_score):
        final, final_score = uniform.copy(), exact_score(uniform, n)
    return final, float(final_score), int(n)
