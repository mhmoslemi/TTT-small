def _quad_roots(a, b, c):

    roots = []
    a = float(a); b = float(b); c = float(c)
    if abs(a) < 1e-15:
        if abs(b) > 1e-15:
            roots.append(-c / b)
        return roots
    disc = b * b - 4.0 * a * c
    tol = 1e-14 * (b * b + abs(4.0 * a * c) + 1e-300)
    if disc < -tol:
        return roots
    if disc < 0:
        disc = 0.0
    sq = sqrt(disc)
    q = -0.5 * (b + copysign(sq, b))
    if abs(q) > 1e-300:
        roots.append(q / a)
        roots.append(c / q)
    else:
        roots.append(-b / (2.0 * a))
    return [r for r in roots if isfinite(r)]
