def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,):
        return None
    if not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx])
            h[idx] += d; r -= d
        else:
            d = min(-r, h[idx])
            h[idx] -= d; r += d
    if abs(r) > 1e-10 * max(1, n) or not np.all(np.isfinite(h)):
        return None
    return h

        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
