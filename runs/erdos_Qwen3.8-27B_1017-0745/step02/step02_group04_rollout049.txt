def project_capped(v,n):
    if v is None:
        return None
    v = np.asarray(v, dtype=np.float64)
    if v.shape != (n,):
        return None
    if not np.all(np.isfinite(v)):
        return None
    v = v.copy()
    v = np.clip(v, -1e6, 1e6)
    s = 0.5 * n
    if np.all(v >= -1e-10) and np.all(v <= 1.0 + 1e-10):
        h = np.clip(v, 0.0, 1.0)
        if abs(float(np.sum(h)) - s) <= 1e-7 * max(1, n):
            return h
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(v - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(n + 3):
        r = s - float(np.sum(h))
        if abs(r) < 1e-13:
            break
        if r > 0.0:
            idx = int(np.argmax(1.0 - h))
            delta = min(r, 1.0 - h[idx])
            if delta <= 0.0:
                break
            h[idx] += delta
        else:
            idx = int(np.argmax(h))
            delta = min(-r, h[idx])
            if delta <= 0.0:
                break
            h[idx] -= delta
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-7 * max(1, n):
        return None
    return h

def de_search(pop, scores, n, rng, gens, patience, max_restart, best_v, best_s):

        improved=False

def ctx_n(n):
    _ctx['n']=n; _ctx['dx']=2.0/n; _ctx['s']=0.5*n; _ctx['lags']=lag_indices(n)
def sm_obj(z):
    n=_ctx['n']; dx=_ctx['dx']; lam=_ctx['lam']; beta=_ctx['beta']; ref=_ctx['ref']
    if not np.all(np.isfinite(z)): return 1e100
    q=np.correlate(z,1-z,mode='full')*dx; mx=float(q.max()); e=np.exp((q-mx)/lam); ss=float(e.sum())
    if ss<=0 or not np.isfinite(ss): return 1e100
    val=mx+lam*np.log(ss)+beta*float((z-ref)@(z-ref)); return 1e100 if not np.isfinite(val) else float(val)
def sm_jac(z):
    n=_ctx['n']; dx=_ctx['dx']; lam=_ctx['lam']; beta=_ctx['beta']; ref=_ctx['ref']; lags=_ctx['lags']
    if not np.all(np.isfinite(z)): return np.zeros(n)
    g=2*beta*(z-ref); q=np.correlate(z,1-z,mode='full')*dx; mx=float(q.max()); e=np.exp((q-mx)/lam); ss=float(e.sum())

        if wd>1e-16:

    return g
def sm_eq(z): return np.array([float(z.sum()-_ctx['s'])])
def sm_eqj(z):

def lag_indices(n):
    return [(max(d, 0), max(-d, 0), n - abs(d)) for d in range(-(n - 1), n)]

def project_capped(v, n):

def sm_obj(z):

    dx = _ctx['dx']
    lam = _ctx['lam']
    beta = _ctx['beta']
    ref = _ctx['ref']
    if not np.all(np.isfinite(z)):
        return 1e100
    q = np.correlate(z, 1 - z, mode='full') * dx
    mx = float(q.max())
    e = np.exp((q - mx) / lam)
    ss = float(e.sum())
    if ss <= 0 or not np.isfinite(ss):
        return 1e100
    val = mx + lam * np.log(ss) + beta * float((z - ref) @ (z - ref))
    return 1e100 if not np.isfinite(val) else float(val)

def sm_jac(z):
    n = _ctx['n']
    dx = _ctx['dx']
    lam = _ctx['lam']
    beta = _ctx['beta']
    ref = _ctx['ref']
    lags = _ctx['lags']
    if not np.all(np.isfinite(z)):
        return np.zeros(n)

    q = np.correlate(z, 1 - z, mode='full') * dx
    mx = float(q.max())
    e = np.exp((q - mx) / lam)
    ss = float(e.sum())

    return g
