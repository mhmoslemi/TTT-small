    best_s = exact_score(h, n)
    dx = 2.0 / n
    for tau in taus:
        eta = 0.25 / dx
        eta_max = 2.0 / dx
        F, g = smoothed_val_grad(h, n, tau)
        if not np.isfinite(F) or not np.all(np.isfinite(g)):
            break
        for _ in range(max_iters):
            accepted = False
            for _bt in range(25):
                z = project_capped(h - eta * g, n)
                if z is None:
                    break
                d = z - h
                if np.max(np.abs(d)) < 1e-12:
                    break
                dot_gd = float(np.dot(g, d))
                if not np.isfinite(dot_gd) or dot_gd >= 0:
                    break
                Fz, gz = smoothed_val_grad(z, n, tau)
                if np.isfinite(Fz) and Fz <= F + 1e-4 * dot_gd + 1e-14:
                    h = z
                    F = Fz
                    g = gz
                    cur_s = exact_score(h, n)
                    if cur_s < best_s - 1e-12:
                        best_s = cur_s
                        best_h = h.copy()
                    eta = min(1.2 * eta, eta_max)
                    accepted = True
                    break
                eta *= 0.5
            if not accepted:
                break
    return best_h, best_s

def build_starts(fb, n, rng, count):
    pool = [fb.copy()]
    cands = []
    cands.append(fb[::-1].copy())
    cands.append(1.0 - fb[::-1])
    cands.append(np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    cands.append(blk)
    cands.append(np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64))
    j = np.arange(n) + 0.5
    cands.append(np.clip(0.5 + 0.45 * np.cos(2.0 * np.pi * j / n), 0, 1))
    cands.append(np.clip(0.5 + 0.45 * np.cos(2.0 * np.pi * j / n + np.pi / 2), 0, 1))
    cands.append(fb + rng.normal(0, 0.06, n))
    cands.append(rng.uniform(0, 1, n))
    for c in cands:
        if len(pool) >= count:
            break
        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-8:
                dup = True
                break
        if not dup:
            pool.append(p)
    return pool[:count]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size >= 1 and p.size <= 999:
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                n = int(p.size)
                parent = p.copy()
    except Exception:
        pass
    if n is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if n is None:
        n = 47
    if n == 1:
        h1 = np.array([0.5])
        return h1, exact_score(h1, 1), 1
    fb = project_capped(parent, n) if parent is not None else np.full(n, 0.5)
    if fb is None:
        fb = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    taus = [0.02, 0.005, 0.001, 0.0002, 0.00004, 0.00001]
    if n <= 64:
        n_sc, n_mi, n_rest = 5, 140, 2
    elif n <= 128:
        n_sc, n_mi, n_rest = 4, 90, 1
    elif n <= 256:
        n_sc, n_mi, n_rest = 3, 55, 1
    else:
        n_sc, n_mi, n_rest = 2, 25, 0
    starts = build_starts(fb, n, rng, n_sc)
    gs = exact_score(fb, n)
    gb = fb.copy()
    for st in starts:
        try:
            yo, so = continuation_search(st, n, taus, n_mi)
            if so < gs - 1e-12:
                gs, gb = so, yo.copy()
        except Exception:
            continue
    if n_rest > 0 and n <= 64:
        sds = [0.03, 0.08] if gs > exact_score(fb, n) - 1e-5 else [0.08, 0.18]
    elif n_rest > 0:
        sds = [0.05, 0.12] if gs > exact_score(fb, n) - 1e-5 else [0.12, 0.20]
    else:
        sds = []
    for sd in sds:
        try:
            pert = project_capped(gb + rng.normal(0, sd, n), n)
            if pert is None:
                continue
            yo, so = continuation_search(pert, n, taus, n_mi)
            if so < gs - 1e-12:
                gs, gb = so, yo.copy()
        except Exception:
            continue
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = project_capped(fb, n)
    if final is None or not is_feasible(final, n):
        final = np.full(n, 0.5)
    sf = exact_score(final, n)
    ss = exact_score(fb, n)
    if sf > ss:
        final, sf = fb, ss
    return final, float(sf), int(n)
