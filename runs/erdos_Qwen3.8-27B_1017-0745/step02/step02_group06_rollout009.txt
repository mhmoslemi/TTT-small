def project_capped(z, n):
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e15, 1e15)
    s = n / 2.0
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx])
            h[idx] += d; r -= d
        else:
            d = min(-r, h[idx])
            h[idx] -= d; r += d
    if abs(r) > 1e-9 * max(1, n) or not np.all(np.isfinite(h)):
        return None
    return np.clip(h, 0.0, 1.0)

        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9 * max(1, n)

def lag_values(h, n):

    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if c.shape[0] != 2 * n - 1 or not np.all(np.isfinite(c)):
        return None
    return c

    c = lag_values(h, n)
    if c is None or not is_feasible(h, n):
        return float('inf')
    return float(np.max(c))

        return None, None
    a = float(np.max(L))
    e = np.exp(np.clip((L - a) / tau, -745.0, 0.0))

        val, g = smooth_val_grad(x, n, tau)
        if val is None:
            break
        alpha = 1.0
        for _ in range(iters_per_temp):
            y = project_capped(x - alpha * g, n)
            if y is None:
                break
            d = y - x
            if float(np.max(np.abs(d))) / alpha <= max(1e-7, 1e-3 * tau):
                break
            accepted = False
            a_t = alpha
            for _ in range(25):
                y_t = project_capped(x - a_t * g, n)
                if y_t is None:
                    break
                d_t = y_t - x
                v_t, _ = smooth_val_grad(y_t, n, tau)
                if v_t is None:
                    break
                if v_t <= val - 1e-4 * float(np.dot(d_t, d_t)) / a_t + 1e-14 * max(1.0, abs(val)):
                    accepted = True
                    y = y_t
                    break
                a_t *= 0.5
            if not accepted:
                break
            val_new, g_new = smooth_val_grad(y, n, tau)
            if val_new is None:
                x = y
                break
            sv = float(np.dot(d, g_new - g))
            if sv > 1e-14:
                ratio = float(np.dot(d, d)) / sv
                alpha = float(np.clip(ratio, 1e-4, 10.0))
            else:
                alpha = float(np.clip(alpha, 1e-4, 10.0))
            x = y
            val = val_new
            g = g_new
            sx = exact_score(x, n)
            if sx < best_s - 1e-12:
                best_h = x.copy()
                best_s = sx
    st = exact_score(x, n)
    if st < best_s - 1e-12:
        best_h = x.copy()
        best_s = st
    return best_h, best_s

def build_starts(parent, n, rng, count):
    pool = []
    cands = [np.full(n, 0.5)]
    if parent is not None:
        cands.append(parent.copy())
        cands.append(parent[::-1].copy())
        cands.append((1.0 - parent)[::-1].copy())
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2:
            blk[n // 2] = 0.5
        cands.append(blk)
        cands.append(np.where(np.arange(n) % 2 == 0, 1.0, 0.0))
        for m in range(1, 4):
            for ph in [0.0, np.pi / 2]:
                cands.append(0.5 + 0.45 * np.cos(2.0 * np.pi * m * np.arange(n) / n + ph))
        for sd in [0.04, 0.12, 0.25]:
            cands.append(parent + rng.normal(0, sd, n))
    cands.append(rng.uniform(0, 1, n))
    for c in cands:
        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-6:
                dup = True
                break
        if not dup:
            pool.append(p)
    while len(pool) < count:
        base = parent if parent is not None else np.full(n, 0.5)
        p = project_capped(base + rng.normal(0, 0.1, n), n)
        if p is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-6:
                dup = True
                break
        if not dup:
            pool.append(p)
        if len(pool) > count * 4:
            break
    return pool[:count]

def run(seed=42, **kwargs):
    n = 47
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 < p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                if abs(np.sum(p) - n / 2.0) <= 1e-6 * max(1, n):
                    parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n <= 1:
        return np.array([0.5]), 0.0, 1
    rng = np.random.default_rng(int(seed))
    fallback = np.full(n, 0.5)
    fb_score = exact_score(fallback, n)
    best_h, best_s = fallback.copy(), fb_score
    if parent is not None:
        pr = project_capped(parent, n)
        if pr is not None:
            ps = exact_score(pr, n)
            if ps < best_s - 1e-12:
                best_h, best_s = pr.copy(), ps
    if n <= 64:
        n_starts, n_restarts, iters_pt = 8, 3, 120
    elif n <= 128:
        n_starts, n_restarts, iters_pt = 6, 2, 80
    elif n <= 256:
        n_starts, n_restarts, iters_pt = 4, 1, 50
    else:
        n_starts, n_restarts, iters_pt = 3, 1, 30
    starts = build_starts(parent, n, rng, n_starts)
    for st in starts:
        s = exact_score(st, n)
        if s < best_s - 1e-12:
            best_h, best_s = st.copy(), s
        best_h, best_s = projected_descent(st, n, iters_pt, best_h, best_s)
    sigma = 0.08
    for _ in range(n_restarts):
        saved_s = best_s
        prop = None
        for _ in range(3):
            c = project_capped(best_h + sigma * rng.normal(0, 1, n), n)
            if c is not None and np.max(np.abs(c - best_h)) > 1e-5:
                prop = c
                break
        if prop is None:
            continue
        sp = exact_score(prop, n)
        if sp < best_s - 1e-12:
            best_h, best_s = prop.copy(), sp
        best_h, best_s = projected_descent(prop, n, iters_pt, best_h, best_s)
        if best_s < saved_s - 1e-12:
            sigma = max(0.01, 0.7 * sigma)
        else:
            sigma = min(0.25, 1.5 * sigma)
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fallback.copy()
    fs = exact_score(final, n)
    if fs > fb_score + 1e-12:
        final = fallback.copy()
        fs = fb_score
    if parent is not None:
        pr2 = project_capped(parent, n)
        if pr2 is not None:
            ps2 = exact_score(pr2, n)
            if ps2 < fs:
                final, fs = pr2, ps2
    return final, float(fs), int(n)
