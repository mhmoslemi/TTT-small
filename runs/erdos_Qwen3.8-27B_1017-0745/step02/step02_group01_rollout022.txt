import numpy as np
import math

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):
        return False
    return True

def project_capped(z, n):
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(3):
        if abs(r) < 1e-14:
            break
        for i in range(n):
            if abs(r) < 1e-14:
                break
            if r > 0:
                c = min(r, 1.0 - h[i])
                h[i] += c
                r -= c
            else:
                c = min(-r, h[i])
                h[i] -= c
                r += c
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    if abs(np.sum(h) - s) > 1e-10 * max(1, n):
        return None
    return h

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full')
    if q.shape[0] != 2 * n - 1 or not np.all(np.isfinite(q)):
        return float('inf')
    return float(np.max(q) * (2.0 / n))

def build_directions(n):
    dirs = []
    stack = [(0, n)]
    while stack:
        a, b = stack.pop()
        L = b - a
        if L < 2:
            continue
        L0 = L // 2
        L1 = L - L0
        v = np.zeros(n)
        v[a:a+L0] = L1
        v[a+L0:b] = -L0
        dirs.append(v)
        stack.append((a+L0, b))
        stack.append((a, a+L0))
    return dirs

def line_interval(h, v):
    if abs(np.sum(v)) > 1e-12:
        return None
    lo = -1e15
    hi = 1e15
    pos = v > 1e-14
    neg = v < -1e-14
    if np.any(pos):
        lo = max(lo, np.max(-h[pos] / v[pos]))
        hi = min(hi, np.max((1.0 - h[pos]) / v[pos]))
    if np.any(neg):
        lo = max(lo, np.max((1.0 - h[neg]) / v[neg]))
        hi = min(hi, np.max(-h[neg] / v[neg]))
    if not np.isfinite(lo) or not np.isfinite(hi) or lo > hi + 1e-14:
        return None
    lo = max(lo, -1.0)
    hi = min(hi, 1.0)
    if hi - lo < 1e-14:
        return None
    return lo, hi

def line_coefficients(h, v, n):
    dx = 2.0 / n
    c = dx * np.correlate(h, 1.0 - h, mode='full')
    b = dx * (np.correlate(v, 1.0 - h, mode='full') - np.correlate(h, v, mode='full'))
    a = -dx * np.correlate(v, v, mode='full')
    L = 2 * n - 1
    if a.shape[0] != L or b.shape[0] != L or c.shape[0] != L:
        return None
    if not (np.all(np.isfinite(a)) and np.all(np.isfinite(b)) and np.all(np.isfinite(c))):
        return None
    return a, b, c

def append_roots(A, B, C, lo, hi, out):
    scale = max(abs(A), abs(B), abs(C), 1e-300)
    tol = 1e-14 * scale
    if abs(A) <= tol:
        if abs(B) > tol:
            r = -C / B
            if lo - 1e-12 <= r <= hi + 1e-12:
                out.append(min(max(r, lo), hi))
        return
    disc = B * B - 4.0 * A * C
    disc_tol = 1e-13 * (B * B + abs(4.0 * A * C) + 1e-300)
    if disc < -disc_tol:
        return
    if disc < 0:
        disc = 0.0
    s = math.sqrt(disc)
    q_val = -0.5 * (B + (s if B >= 0 else -s))
    if abs(q_val) > 1e-300:
        r1 = q_val / A
        r2 = C / q_val
    else:
        r1 = -B / (2.0 * A)
        r2 = r1
    span = max(1.0, abs(lo), abs(hi))
    for r in (r1, r2):
        if np.isfinite(r) and lo - 1e-12 * span <= r <= hi + 1e-12 * span:
            out.append(min(max(r, lo), hi))

def line_candidates(a, b, c, lo, hi):
    L = a.shape[0]
    cands = [lo, hi, 0.0]
    for k in range(L):
        if abs(a[k]) > 1e-16:
            t = -b[k] / (2.0 * a[k])
            span = max(1.0, abs(lo), abs(hi))
            if np.isfinite(t) and lo - 1e-12 * span <= t <= hi + 1e-12 * span:
                cands.append(min(max(t, lo), hi))
    for i in range(L):
        for j in range(i + 1, L):
            A = a[i] - a[j]
            B = b[i] - b[j]
            C = c[i] - c[j]
            append_roots(A, B, C, lo, hi, cands)
    cands.sort()
    deduped = [cands[0]]
    for t in cands[1:]:
        if abs(t - deduped[-1]) > 1e-12 * max(1.0, abs(t)):
            deduped.append(t)
    return deduped

def line_search(h, cur_score, v, n):
    tv = line_interval(h, v)
    if tv is None:
        return h.copy(), cur_score
    lo, hi = tv
    coefs = line_coefficients(h, v, n)
    if coefs is None:
        return h.copy(), cur_score
    a, b, c = coefs
    cands = line_candidates(a, b, c, lo, hi)
    # Evaluate envelope at candidates, pick best
    best_s = cur_score
    best_t = 0.0
    # Chunk evaluation
    M = a.shape[0]
    K = len(cands)
    chunk = 512
    best_chunk_s = cur_score
    for start in range(0, K, chunk):
        end = min(start + chunk, K)
        t_arr = np.array(cands[start:end], dtype=np.float64)
        # Build (chunk_size, M) matrix
        A_m = a[np.newaxis, :]  # (1, M)
        B_m = b[np.newaxis, :]
        C_m = c[np.newaxis, :]
        T = t_arr[:, np.newaxis]  # (chunk_size, 1)
        vals = A_m * T * T + B_m * T + C_m  # (chunk_size, M)
        env = np.max(vals, axis=1)  # (chunk_size,)
        for idx in range(end - start):
            if not np.isfinite(env[idx]):
                continue
            pred = env[idx]
            if pred >= best_chunk_s:
                continue
            t_val = cands[start + idx]
            z = h + t_val * v
            proj = project_capped(z, n)
            if proj is None or not is_feasible(proj, n):
                continue
            sc = exact_score(proj, n)
            if sc < best_s - 1e-12:
                best_s = sc
                best_t = t_val
                best_chunk_s = pred
    if best_t == 0.0 and best_s >= cur_score:
        return h.copy(), cur_score
    z = h + best_t * v
    proj = project_capped(z, n)
    if proj is None:
        return h.copy(), cur_score
    return proj, best_s
