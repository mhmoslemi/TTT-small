        fallback = np.full(n, 0.5)
    fallback = np.clip(np.asarray(fallback, dtype=np.float64), 0.0, 1.0)
    if n == 1:
        return fallback, float(exact_score(fallback, n)), int(n)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 64:
        n_starts, iters = 8, 60
    elif n <= 128:
        n_starts, iters = 7, 40
    elif n <= 256:
        n_starts, iters = 5, 25
    else:
        n_starts, iters = 3, 12
    starts = build_starts(fallback, n, rng, n_starts)
    gb = fallback.copy()
    gs = float(exact_score(fallback, n))
    for st in starts:
        try:
            res, res_s = smooth_optimize(st, n, iters)
            if res_s < gs - 1e-12:
                gs, gb = res_s, res.copy()
        except Exception:
            pass
    try:
        pf, pf_s = pairwise_refine(fallback, n, rng)
        if pf_s < gs - 1e-12:
            gs, gb = pf_s, pf.copy()
    except Exception:
        pass
    if np.max(np.abs(gb - fallback)) > 1e-12:
        try:
            pg, pg_s = pairwise_refine(gb, n, rng)
            if pg_s < gs - 1e-12:
                gs, gb = pg_s, pg.copy()
        except Exception:
            pass
    gb = project_capped(gb, n)
    if gb is None or not is_feasible(gb, n):
        gb = fallback.copy()
    fs = float(exact_score(gb, n))
    fb = float(exact_score(fallback, n))
    if fs > fb + 1e-12:
        gb, fs = fallback.copy(), fb
    if fs == float('inf'):
        gb = np.full(n, 0.5)
        fs = float(exact_score(gb, n))
    return gb, fs, int(n)
