                    t = lo_t + frac * (hi_t - lo_t)
                    if t == 0.0:
                        continue
                    h[i] += t
                    h[j] -= t
                    s_ev = exact_score(h, n)
                    h[i] -= t
                    h[j] += t
                    if np.isfinite(s_ev) and s_ev < best_s - 1e-12:
                        best_s, best_t = s_ev, t
                if best_t != 0.0:
                    h[i] += best_t
                    h[j] -= best_t
                    improved = True
        if not improved:
            break
    return h

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= iv.size <= 999:
            n = iv.size
            if np.all(np.isfinite(iv)) and np.all(iv >= -1e-8) and np.all(iv <= 1.0 + 1e-8):
                parent = np.clip(iv, 0.0, 1.0).copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
    n = max(1, min(n, 999))
    uniform = np.full(n, 0.5)
    fb_score = exact_score(uniform, n)
    best, best_score = uniform.copy(), fb_score
    if parent is not None:
        pp = project_capped(parent, n)
        if pp is not None:
            ps = exact_score(pp, n)
            if np.isfinite(ps) and ps < best_score:
                best, best_score = pp.copy(), ps
            parent = pp

    if n <= 64:
        n_chains, n_epochs, moves_e, delta0, n_pilot = 5, 28, 180, 0.12, 20
    elif n <= 128:
        n_chains, n_epochs, moves_e, delta0, n_pilot = 3, 20, 100, 0.12, 14
    elif n <= 256:
        n_chains, n_epochs, moves_e, delta0, n_pilot = 2, 14, 45, 0.10, 10
    else:
        n_chains, n_epochs, moves_e, delta0, n_pilot = 1, 8, 12, 0.08, 6
    try:
        n_chains = max(1, min(6, int(kwargs.get('chains', n_chains))))
    except Exception:
        pass
    try:
        n_epochs = max(1, min(40, int(kwargs.get('epochs', n_epochs))))
    except Exception:
        pass
    try:
        moves_e = max(1, min(300, int(kwargs.get('moves_per_epoch', moves_e))))
    except Exception:
        pass
    try:
        delta0 = max(0.01, min(0.30, float(kwargs.get('proposal_scale', delta0))))
    except Exception:
        pass
    total_default = n_chains * n_epochs * moves_e
    cap = max(total_default, int(80000000 // max(1, n * n)))
    total_default = min(total_default, cap)

    sd = 42
    try:
        sd = int(seed)
    except Exception:
        pass
    rng = np.random.default_rng(abs(sd) % (2 ** 32) if sd != 0 else 42)

    starts = [uniform.copy()]
    if parent is not None:
        starts.append(parent.copy())
        for ns in [0.03, 0.08, 0.16]:
            p = project_capped(parent + rng.normal(0, ns, n), n)
            if p is not None:
                starts.append(p)
        for raw in [parent[::-1].copy(), 1.0 - parent, (1.0 - parent[::-1]).copy()]:
            p = project_capped(raw, n)
            if p is not None:
                starts.append(p)
    idx = np.arange(n, dtype=np.float64)
    for freq in [1, 2]:
        for ph in [0.0, math.pi / 2]:
            p = project_capped(0.5 + 0.45 * np.cos(2.0 * np.pi * freq * idx / n + ph), n)
            if p is not None:
                starts.append(p)
    p = project_capped(rng.uniform(0, 1, n), n)
    if p is not None:
        starts.append(p)
    dedup = []
    for s in starts:
        dup = False
        for d in dedup:
            if float(np.max(np.abs(s - d))) < 1e-8:
                dup = True
                break
        if not dup:
            dedup.append(s)
    starts = dedup

    proposals_left = total_default
    chain_noise = 0.04
    si = 0
    for c in range(n_chains):
        if n < 2 or proposals_left <= 0:
            break
        if si < len(starts):
            cur_h = starts[si].copy()
            si += 1
        else:
            cur_h = project_capped(best + rng.normal(0, chain_noise, n))
            if cur_h is None:
                cur_h = uniform.copy()
            chain_noise = min(0.20, chain_noise * 1.5)
        cur_s = exact_score(cur_h, n)
        if not np.isfinite(cur_s):
            cur_h, cur_s = uniform.copy(), fb_score
        if np.isfinite(cur_s) and cur_s < best_score - 1e-12:
            best, best_score = cur_h.copy(), cur_s
        T0 = calibrate_temp(cur_h, n, rng, delta0, n_pilot)
        proposals_left -= n_pilot
        if proposals_left <= 0:
            break
        Tmin = max(1e-7, 1e-3 * T0)
        delta = delta0
        for e in range(n_epochs):
            if proposals_left <= 0:
                break
            T_e = T0 * (Tmin / T0) ** (float(e) / max(1, n_epochs - 1)) if n_epochs > 1 else T0
            n_moves = min(moves_e, proposals_left)
            accepted = 0
            attempted = 0
            for _ in range(n_moves):
                prop = make_proposal(cur_h, n, rng, delta)
                attempted += 1
                if prop is None:
                    continue
                s_prop = exact_score(prop, n)
                if not np.isfinite(s_prop):
                    continue
                if s_prop < best_score - 1e-12:
                    best, best_score = prop.copy(), s_prop
                d = s_prop - cur_s
                if d <= 1e-15:
                    cur_h, cur_s = prop, s_prop
                    accepted += 1
                elif T_e > 1e-15:
                    ratio = d / T_e
                    if ratio < 700.0 and math.log(rng.random()) < -ratio:
                        cur_h, cur_s = prop, s_prop
                        accepted += 1
            proposals_left -= attempted
            if attempted > 0:
                rate = accepted / attempted
                if rate > 0.40:
                    delta = min(0.35, delta * 1.25)
                elif rate < 0.15:
                    delta = max(0.002, delta * 0.75)

    if n <= 128:
        pol = greedy_polish(best, n, 3)
        pol = project_capped(pol, n)
        if pol is not None:
            ps = exact_score(pol, n)
            if np.isfinite(ps) and ps < best_score - 1e-12:
                best, best_score = pol, ps

    finalists = [best.copy(), uniform.copy()]
    if parent is not None:
        finalists.append(parent.copy())
    final, final_score = uniform.copy(), fb_score
    for f in finalists:
        if not is_feasible(f, n):
            f = project_capped(f, n)
            if f is None:
                continue
        fs = exact_score(f, n)
        if np.isfinite(fs) and fs < final_score:
            final, final_score = f.copy(), fs
    final = np.clip(final, 0.0, 1.0)
    r = n / 2.0 - float(np.sum(final))
    for i in range(n):
        if abs(r) < 1e-13:
            break
        if r > 0:
            a = min(r, 1.0 - final[i])
            final[i] += a
            r -= a
        else:
            a = min(-r, final[i])
            final[i] -= a
            r += a
    final_score = exact_score(final, n)
    if not np.isfinite(final_score):
        final, final_score = uniform.copy(), fb_score
    return final.astype(np.float64), float(final_score), int(n)
