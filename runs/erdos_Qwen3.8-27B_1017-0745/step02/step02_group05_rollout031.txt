import numpy as np
import math

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(2):
        if abs(r) < 2e-14 * max(1, n):
            break
        if r > 0:
            for i in range(n):
                if r <= 2e-14 * max(1, n):
                    break
                add = min(r, 1.0 - h[i])
                h[i] += add
                r -= add
        else:
            for i in range(n):
                if r >= -2e-14 * max(1, n):
                    break
                sub = min(-r, h[i])
                h[i] -= sub
                r += sub
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(np.sum(h) - s) > 1e-10 * max(1, n):
        return None
    return np.clip(h, 0.0, 1.0)

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n)

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(np.max(c)) if np.all(np.isfinite(c)) else float('inf')

def helmert_fwd(x, n):
    d = n - 1
    y = np.zeros(n)
    for k in range(d - 1, -1, -1):
        pass
    u = np.array([x[k] / math.sqrt((k + 1) * (k + 2)) for k in range(d)])
    suffix = np.zeros(n)
    for i in range(n - 2, -1, -1):
        suffix[i] = suffix[i + 1] + u[i] if i < d else 0.0
    y[0] = suffix[0]
    for i in range(1, n):
        y[i] = suffix[i] - (i * x[i - 1] / math.sqrt(i * (i + 1))) if i - 1 < d else 0.0
    return y

def helmert_bwd(y, n):
    d = n - 1
    x = np.zeros(d)
    prefix = 0.0
    for i in range(n):
        prefix += y[i]
        if i < d:
            x[i] = (prefix - (i + 1) * y[i + 1]) / math.sqrt((i + 1) * (i + 2))
    return x

def latent_to_h(x, n):
    if not np.all(np.isfinite(x)):
        return None
    y = helmert_fwd(x, n)
    if not np.all(np.isfinite(y)):
        return None
    return project_capped(0.5 + y, n)

def cma_es(h0, n, sigma0, P, G, rng, full_cov):
    d = n - 1
    mu = max(1, d - 1) if d < 4 else d  # standard: mu ~ d/2 but use P/2
    mu = max(1, P // 2)
    w = np.array([math.log(mu + 0.5) - math.log(i) for i in range(1, mu + 1)])
    w /= w.sum()
    mu_eff = 1.0 / np.sum(w * w)
    c_sig = (mu_eff + 2.0) / (d + mu_eff + 5.0)
    d_sig = 1.0 + 2.0 * max(0.0, math.sqrt((mu_eff - 1.0) / (d + 1.0)) - 1.0) + c_sig
    c_c = (4.0 + mu_eff / d) / (d + 4.0 + 2.0 * mu_eff / d)
    c1 = 2.0 / ((d + 1.3) ** 2 + mu_eff)
    c_mu = max(0.0, min(1.0 - c1, 2.0 * (mu_eff - 2.0 + 1.0 / mu_eff) / ((d + 2.0) ** 2 + mu_eff)))
    chi_d = math.sqrt(d) * (1.0 - 1.0 / (4.0 * d) + 1.0 / (21.0 * d * d))
    m = helmert_bwd(h0 - 0.5, n)
    if not np.all(np.isfinite(m)):
        return h0.copy(), exact_score(h0, n)
    sigma = sigma0
    p_sig = np.zeros(d)
    p_c = np.zeros(d)
    if full_cov:
        C = np.eye(d)
    else:
        C = np.ones(d)
    best_h, best_s = h0.copy(), exact_score(h0, n)
    min_sigma_count = 0
    for t in range(1, G + 1):
        if full_cov:
            C = 0.5 * (C + C.T)
            try:
                eig, U = np.linalg.eigh(C)
            except Exception:
                break
            eig = np.clip(eig, 1e-10, 1e4)
            A = U @ np.diag(np.sqrt(eig))
            Cinv_sqrt = U @ np.diag(1.0 / np.sqrt(eig)) @ U.T
        else:
            C = np.clip(C, 1e-10, 1e4)
            A = np.sqrt(C)
        X = np.zeros((P, d))
        H = np.zeros((P, n))
        S = np.full(P, float('inf'))
        for i in range(P):
            z = rng.standard_normal(d)
            if full_cov:
                Y = A @ z
            else:
                Y = A * z
            xk = m + sigma * Y
            hk = latent_to_h(xk, n)
            if hk is not None:
                X[i] = xk
                H[i] = hk
                S[i] = exact_score(hk, n)
            else:
                X[i] = m + sigma * z  # placeholder
        idx = np.argsort(S, kind='stable')
        finite_mask = np.isfinite(S[idx[:mu]])
        if np.sum(finite_mask) < mu:
            break
        sel = idx[:mu]
        m_new = np.sum(w[:, None] * X[sel], axis=0)
        y_w = (m_new - m) / sigma
        if full_cov:
            p_sig_new = (1.0 - c_sig) * p_sig + math.sqrt(c_sig * (2.0 - c_sig) * mu_eff) * (Cinv_sqrt @ y_w)
        else:
            p_sig_new = (1.0 - c_sig) * p_sig + math.sqrt(c_sig * (2.0 - c_sig) * mu_eff) * (y_w / np.sqrt(C))
        h_sig = 1.0 if np.linalg.norm(p_sig_new) / math.sqrt(max(1e-15, 1.0 - (1.0 - c_sig) ** (2.0 * t))) / chi_d < 1.4 + 2.0 / (d + 1.0) else 0.0
        p_c_new = (1.0 - c_c) * p_c + h_sig * math.sqrt(c_c * (2.0 - c_c) * mu_eff) * y_w
        if full_cov:
            rank_mu_mat = np.zeros((d, d))
            for i in range(mu):
                yi = (X[sel[i]] - m) / sigma
                rank_mu_mat += w[i] * np.outer(yi, yi)
            C_new = (1.0 - c1 - c_mu) * C + c1 * (np.outer(p_c_new, p_c_new) + (1.0 - h_sig) * c_c * (2.0 - c_c) * C) + c_mu * rank_mu_mat
        else:
            rank_mu_diag = np.zeros(d)
            for i in range(mu):
                yi = (X[sel[i]] - m) / sigma
                rank_mu_diag += w[i] * yi * yi
            C_new = (1.0 - c1 - c_mu) * C + c1 * (p_c_new * p_c_new + (1.0 - h_sig) * c_c * (2.0 - c_c) * C) + c_mu * rank_mu_diag
        expo = np.clip((c_sig / d_sig) * (np.linalg.norm(p_sig_new) / chi_d - 1.0), -50, 50)
        sigma_new = np.clip(sigma * np.exp(expo), 1e-4, 0.5)
        if not np.all(np.isfinite(C_new)) or not np.all(np.isfinite(p_sig_new)) or not np.all(np.isfinite(p_c_new)):
            break
        m, p_sig, p_c, C, sigma = m_new, p_sig_new, p_c_new, C_new, sigma_new
        for i in range(P):
            if np.isfinite(S[i]) and S[i] < best_s - 1e-12:
                best_s, best_h = S[i], H[i].copy()
        if sigma <= 1e-4 + 1e-12:
            min_sigma_count += 1
            if min_sigma_count >= 5:
                break
        else:
            min_sigma_count = 0
    hm = latent_to_h(m, n)
    if hm is not None:
        sm = exact_score(hm, n)
        if sm < best_s - 1e-12:
            best_s, best_h = sm, hm.copy()
    return best_h, best_s

    for k in (1, 2):
        ph = rng.uniform(0, 2 * np.pi)
        cands.append(0.5 + 0.3 * np.sin(k * 2 * np.pi * np.arange(n) / n + ph))
    cands.append(rng.uniform(0, 1, n))
    for c in cands:

        if p is None:

        dup = False

                dup = True
                break
        if not dup:

        if len(pool) >= count:

        sig0_first, sig0_other = 0.08, 0.14
    elif n <= 256:
        n_starts, G = 4, 40
        P = max(8, min(20, 4 + int(3 * math.log(max(2, n - 1)))))
        full_cov = False
        sig0_first, sig0_other = 0.06, 0.10
    else:
        n_starts, G = 3, 12
        P = max(8, min(12, 4 + int(3 * math.log(max(2, n - 1)))))
        full_cov = False
        sig0_first, sig0_other = 0.05, 0.08
    starts = build_starts(fb, n, rng, n_starts)
    gb, gs = safe_h.copy(), safe_s
    for st in starts:
        ss = exact_score(st, n)
        if ss < gs - 1e-12:
            gb, gs = st.copy(), ss
    for i, st in enumerate(starts):
        sig = sig0_first if i == 0 else sig0_other
        try:
            ho, so = cma_es(st, n, sig, P, G, rng, full_cov)
            if so < gs - 1e-12:
                gb, gs = ho.copy(), so
        except Exception:
            pass
        if gs <= 0.38085:
            break
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = safe_h.copy()
    fs = exact_score(final, n)
    if fs > safe_s + 1e-12:
        final = safe_h.copy()
        fs = safe_s
    return final.astype(np.float64), float(fs), int(n)
