def pairwise_polish(x, n, sweeps):
    best = x.copy()
    best_s = exact_score(best, n)
    for _ in range(sweeps):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                lo = max(-best[i], best[j] - 1.0)
                hi = min(1.0 - best[i], best[j])
                if lo >= hi - 1e-14:
                    continue
                cands = [0.0]
                for step in (0.01, 0.03, 0.1, 0.3):
                    for sgn in (1, -1):
                        dd = sgn * step
                        if lo <= dd <= hi:
                            cands.append(dd)
                cands.append(max(lo + 1e-12, 0.0))
                cands.append(min(hi - 1e-12, 0.0))
                best_d, best_sc = 0.0, best_s
                for dd in cands:
                    dd = max(lo, min(hi, dd))
                    trial = best.copy()
                    trial[i] += dd
                    trial[j] -= dd

                    best[i] += best_d
                    best[j] -= best_d
                    best_s = best_sc
