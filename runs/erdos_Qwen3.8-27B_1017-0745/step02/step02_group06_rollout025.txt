def is_feasible(h, n):

        return float('inf')

    if not np.all(np.isfinite(c)):

    return float(np.max(c))

def lag_jacobian(h, n):
    dx = 2.0 / n
    J = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        lo1, hi1 = max(0, -d), min(n, n - d)
        for j in range(lo1, hi1):
            J[row, j] += dx * (1.0 - h[j + d])
        lo2, hi2 = max(0, d), min(n, n + d)
        for j in range(lo2, hi2):
            J[row, j] -= dx * h[j - d]
    return J

    return z[-1]

def slsqp_obj_jac(z, n):
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return g

def slsqp_ineq(z, n):

    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return z[-1] - corr

    h = z[:n]
    JL = lag_jacobian(h, n)
    J = np.zeros((2 * n - 1, n + 1))
    J[:, :n] = -JL
    J[:, n] = 1.0
    return J
