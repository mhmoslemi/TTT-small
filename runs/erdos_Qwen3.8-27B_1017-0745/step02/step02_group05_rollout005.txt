            break
    return x, sc

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
        if not (1 <= n <= 999):
            n = 47
    fb = project_capped(parent, n) if parent is not None else None
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    fb_score = exact_score(fb, n)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        return np.array([0.5]), 0.5, 1
    if n <= 96:
        n_starts, taus, maxit = 6, [1e-2, 2e-3, 4e-4, 8e-5, 1e-5], 100
    elif n <= 256:
        n_starts, taus, maxit = 4, [5e-3, 5e-4, 5e-5, 1e-5], 60
    else:
        n_starts, taus, maxit = 2, [2e-3, 1e-4, 1e-5], 35
    starts = build_starts(fb, n, rng, n_starts)
    gb = fb.copy()
    gs = fb_score
    for st in starts:
        ss = exact_score(st, n)
        if ss < gs - 1e-12:
            gs, gb = ss, st.copy()
    sv = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    eqc = LinearConstraint(np.ones((1, n)), sv, sv)
    for st in starts:
        x = st.copy()
        for tau in taus:
            try:
                res = minimize(smooth_value, x, args=(n, tau), jac=smooth_grad,
                               method='SLSQP', bounds=bounds, constraints=[eqc],
                               options={'maxiter': maxit, 'ftol': 1e-10, 'disp': False})
                y = res.x if hasattr(res, 'x') and res.x is not None else None
            except Exception:
                y = None
            if y is None or y.shape != (n,) or not np.all(np.isfinite(y)):
                continue
            y = project_capped(y, n)
            if y is None or not is_feasible(y, n):
                continue
            best_z, best_phi, best_sz = x, smooth_value(x, n, tau), exact_score(x, n)
            for al in (1.0, 0.5, 0.25, 0.125, 0.0625):
                z = project_capped(x + al * (y - x), n)
                if z is None or not is_feasible(z, n):
                    continue
                sz = exact_score(z, n)
                if sz < best_sz - 1e-12:
                    best_sz = sz
                zp = smooth_value(z, n, tau)
                if zp < best_phi - 1e-12:
                    best_phi, best_z, best_sz = zp, z, exact_score(z, n)
            if best_sz < gs - 1e-12:
                gs, gb = best_sz, best_z.copy()
            x = best_z.copy()
    gb = project_capped(gb, n)
    if gb is None or not is_feasible(gb, n):
        gb, gs = fb.copy(), fb_score
    else:
        gs = exact_score(gb, n)
    gb, gs = coordinate_polish(gb, n, 4)
    if gs < fb_score - 1e-12:
        fb, fb_score = gb.copy(), gs
    for st in starts[:3]:
        pc, ps = coordinate_polish(st, n, 3)
        if ps < gs - 1e-12:
            gs, gb = ps, pc
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb.copy()
    fs = exact_score(final, n)
    if fs > fb_score + 1e-12:
        final, fs = fb.copy(), fb_score
    return final.astype(np.float64), float(fs), int(n)
