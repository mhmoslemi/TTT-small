    z = np.clip(np.asarray(z, dtype=np.float64).ravel(), -1e6, 1e6)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(3):
        if abs(r) < 1e-13:
            break
        if r > 0:
            for i in range(n):
                if r <= 0: break
                add = min(r, 1.0 - h[i])
                h[i] += add
                r -= add
        else:
            for i in range(n):
                if r >= 0: break
                sub = min(-r, h[i])
                h[i] -= sub
                r += sub

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9

    if not is_feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def solve_moment_sdp(n):
    try:
        import cvxpy as cp
        s = n / 2.0
        dx = 2.0 / n
        v = cp.Variable(n)
        W = cp.Variable((n, n), symmetric=True)
        t = cp.Variable()
        cons = [v >= 0, v <= 1, cp.sum(v) == s]
        ones = np.ones(n)
        cons.append(W @ ones == s * v)
        v_col = v[:, cp.newaxis]
        v_row = v[cp.newaxis, :]
        cons.append(W >= 0)
        cons.append(W <= v_col)
        cons.append(W <= v_row)
        cons.append(W >= v_col + v_row - 1)
        Z = cp.bmat([[cp.Constant(1), v[cp.newaxis, :]], [v[:, cp.newaxis], W]])
        cons.append(Z >> 0)
        cons.append(t >= 0)
        cons.append(t <= 1)
        # Q0
        cons.append(dx * (s - cp.trace(W)) <= t)
        # Lag r
        for r in range(1, n):
            m = n - r
            w_sum = 0
            for i in range(m):
                w_sum += W[i, i + r]
            cons.append(dx * (cp.sum(v[r]) - w_sum) <= t)
            cons.append(dx * (cp.sum(v[:m]) - w_sum) <= t)
        prob = cp.Problem(cp.Minimize(t), cons)
        for solver, kws in [('MOSEK', {}), ('CVXOPT', {'max_iters': 200})]:
            try:
                if hasattr(cp, 'CONFIG') or solver in cp.installed_solvers():
                    if solver == 'MOSEK':
                        prob.solve(solver=solver, verbose=False, ms_max_iters=200)
                    else:
                        prob.solve(solver=solver, verbose=False, max_iters=200)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        vv = np.asarray(v.value, dtype=np.float64)
                        WW = np.asarray(W.value, dtype=np.float64)
                        if vv.shape == (n,) and WW.shape == (n, n) and np.all(np.isfinite(vv)) and np.all(np.isfinite(WW)):

            except Exception:
                continue
    except Exception:
        pass
    return None, None
