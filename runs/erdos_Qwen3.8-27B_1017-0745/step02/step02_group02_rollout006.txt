    if fb is None:
        fb = np.full(n, 0.5)
    if n <= 64:
        P, G, sl, mr, rds = 96, 64, 12, 2, [0.08, 0.20]
    elif n <= 128:
        P, G, sl, mr, rds = 64, 30, 8, 1, [0.12]
    elif n <= 256:
        P, G, sl, mr, rds = 40, 12, 5, 1, [0.12]
    else:
        P, G, sl, mr, rds = 12, 3, 3, 0, []
    pop = build_population(fb, n, rng, P)
    best, best_s = de_search(pop, n, G, sl, mr, rds, rng)
    best_p, best_ps = coord_polish(best, n, rng, 2)
    if best_ps < best_s:
        best, best_s = best_p, best_ps
    fs = exact_score(fb, n)
    consts = exact_score(np.full(n, 0.5), n)
    cands = [(best_s, best), (fs, fb), (consts, np.full(n, 0.5))]
    cands.sort(key=lambda x: x[0])
    final, sf = cands[0]
    if not is_feasible(final, n):
        proj = project_capped(final, n)
        if proj is not None:
            final, sf = proj, exact_score(proj, n)
        else:
            final, sf = fb, fs
    return final, float(sf), int(n)
