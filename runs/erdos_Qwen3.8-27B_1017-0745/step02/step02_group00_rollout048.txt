    if Z.ndim == 1:

    m, n = Z.shape
    s = 0.5 * n
    bad = ~np.all(np.isfinite(Z), axis=1)
    if np.any(bad):
        Z = Z.copy()
        Z[bad, :] = 0.5
    Z = np.clip(Z, -1e6, 1e6)
    lo = Z.min(axis=1) - 1.0
    hi = Z.max(axis=1)
    for _ in range(80):
        mid = 0.5*(lo+hi)
        vals = np.clip(Z - mid[:,None], 0.0, 1.0)
        su = vals.sum(axis=1)
        lo = np.where(su > s, mid, lo)
        hi = np.where(su <= s, mid, hi)
    lam = 0.5*(lo+hi)

    # correct residual
    res = s - H.sum(axis=1)

    m,n=H.shape; dx=2.0/n; out=np.empty(m)
    for i in range(m):
       h=H[i]
       v=np.correlate(h, 1.0-h, mode='full')
       val = float(np.max(v))*dx
       out[i] = val if np.isfinite(val) else 1e100
    return out

            for j in range(i+1,n):

                    continue

import numpy as np


def project_rows(Z):
    Z = np.asarray(Z, dtype=np.float64)
    if Z.ndim == 1:
        Z = Z.reshape(1, -1)
    m, n = Z.shape
    s = 0.5 * n
    bad = ~np.all(np.isfinite(Z), axis=1)
    if np.any(bad):
        Z = Z.copy()
        Z[bad] = 0.5
    Z = np.clip(Z, -1e6, 1e6)
    lo = Z.min(axis=1) - 1.0
    hi = Z.max(axis=1)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        su = np.clip(Z - mid[:, None], 0.0, 1.0).sum(axis=1)
        lo = np.where(su > s, mid, lo)
        hi = np.where(su <= s, mid, hi)
    H = np.clip(Z - 0.5 * (lo + hi)[:, None], 0.0, 1.0)
    res = s - H.sum(axis=1)
    for r in range(m):
        rr = float(res[r])
        if abs(rr) <= 1e-12:
            continue
        if rr > 0:
            for c in range(n):
                if abs(rr) < 1e-13:
                    break
                a = min(rr, 1.0 - H[r, c])
                H[r, c] += a
                rr -= a
        else:
            for c in range(n):
                if abs(rr) < 1e-13:
                    break
                a = min(-rr, H[r, c])
                H[r, c] -= a
                rr += a
    H = np.clip(H, 0.0, 1.0)
    bad2 = (~np.all(np.isfinite(H), axis=1)) | (H.min(axis=1) < -1e-9) | (
        H.max(axis=1) > 1.0 + 1e-9
    ) | (np.abs(H.sum(axis=1) - s) > 1e-7)
    if np.any(bad2):
        H = H.copy()
        H[bad2] = 0.5
    return H


def score_rows(H):
    H = np.asarray(H, dtype=np.float64)
    if H.ndim == 1:
        H = H.reshape(1, -1)
    m, n = H.shape
    dx = 2.0 / n
    out = np.empty(m)
    for i in range(m):
        v = np.correlate(H[i], 1.0 - H[i], mode="full")
        val = float(np.max(v)) * dx
        out[i] = val if np.isfinite(val) else 1e100
    return out


def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float("inf")
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return float("inf")
    if abs(float(np.sum(h)) - 0.5 * n) > 1e-7:
        return float("inf")
    v = np.correlate(h, 1.0 - h, mode="full")
    val = float(np.max(v)) * (2.0 / n)
    return val if np.isfinite(val) else float("inf")


def add_candidate(cands, vec, n, tol=1e-5):
    if vec is None:
        return
    vec = np.asarray(vec, dtype=np.float64).ravel()
    if vec.shape != (n,) or not np.all(np.isfinite(vec)):
        return
    for old in cands:
        if np.max(np.abs(vec - old)) < tol:
            return
    cands.append(vec)


def build_population(n, parent, fallback, rng, P):
    raw = [fallback]
    if parent is not None:
        raw.append(parent)
        raw.append(parent[::-1])
        raw.append(1.0 - parent)
        raw.append(1.0 - parent[::-1])

    idx = np.arange(n, dtype=np.float64)
    for f in (1, 2, 3):
        for ph in (0.0, 1.5707963267948966):
            raw.append(0.5 + 0.45 * np.cos(2.0 * np.pi * f * idx / n + ph))
    sig = (0.05, 0.2, 0.5, 1.0)
    k = 0
    while len(raw) < P:
        sg = sig[(k // 2) % 4]
        if k % 2 == 0:
            raw.append(0.5 + sg * rng.normal(0.0, 1.0, n))

    raw = np.vstack(raw[:P])

    kept = []
    for r in range(H.shape[0]):

                same = True
                break
        if not same:
            kept.append(r)
    while len(kept) < P:
        kept.append(0)
    pop = np.vstack([H[i] for i in kept[:P]])

def de_step(pop, scores, F, CR, rng, n):
    P = pop.shape[0]
    snap = pop.copy()
    s_snap = scores.copy()
    f_snap = F.copy()
    cr_snap = CR.copy()
    bi = int(np.argmin(s_snap))
    xbest = snap[bi]
    trials = snap.copy()
    newF = f_snap.copy()
    newCR = cr_snap.copy()
    all_idx = np.arange(P)
    perm = rng.permutation(P)
    for target in perm:
        if rng.random() < 0.1:
            nf = 0.1 + 0.8 * rng.random()
        else:
            nf = f_snap[target]
        if rng.random() < 0.1:
            ncr = 0.05 + 0.9 * rng.random()
        else:
            ncr = cr_snap[target]
        nf = min(0.9, max(0.1, nf))
        ncr = min(0.95, max(0.05, ncr))
        if rng.random() < 0.75:
            avail = np.delete(all_idx, target)
            r = rng.choice(avail, 3, replace=False)
            donor = snap[int(r[0])] + nf * (snap[int(r[1])] - snap[int(r[2])])
        else:
            avail = np.delete(all_idx, target)
            r = rng.choice(avail, 2, replace=False)
            donor = snap[target] + nf * (xbest - snap[target]) + nf * (
                snap[int(r[0])] - snap[int(r[1])]
            )
        mask = rng.random(n) < ncr
        mask[int(rng.integers(0, n))] = True
        trials[target] = np.where(mask, donor, snap[target])
        newF[target] = nf
        newCR[target] = ncr
    trials = project_rows(trials)
    tsc = score_rows(trials)
    for i in range(P):
        if tsc[i] < s_snap[i] - 1e-12:
            pop[i] = trials[i]
            scores[i] = tsc[i]
            F[i] = newF[i]
            CR[i] = newCR[i]
    return pop, scores


def pair_polish(h, n, sweeps, grid):
    h = np.asarray(h, dtype=np.float64).copy()
    L = 2 * n - 1
    dx = 2.0 / n
    alphas = np.linspace(0.0, 1.0, grid)
    fine = np.array(
        [-0.08, -0.03, -0.01, -0.003, -0.001, 0.0, 0.001, 0.003, 0.01, 0.03, 0.08]
    )
    for _ in range(max(0, sweeps)):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                lo = max(-h[i], h[j] - 1.0)
                hi = min(1.0 - h[i], h[j])
                if hi - lo < 1e-11:
                    continue
                base = np.correlate(h, 1.0 - h, mode="full")
                curv = float(np.max(base))
                A = np.zeros(L)
                hrev = h[::-1]
                A[n - 1 - i : n - 1 - i + n] += 1.0 - h
                A[i : i + n] -= hrev
                A[n - 1 - j : n - 1 - j + n] += h - 1.0
                A[j : j + n] += hrev
                B = np.zeros(L)
                B[n - 1] = -2.0
                B[n - 1 + (j - i)] += 1.0
                B[n - 1 + (i - j)] += 1.0
                dts = lo + (hi - lo) * alphas
                ok = (fine >= lo) & (fine <= hi)
                if np.any(ok):
                    dts = np.append(dts, fine[ok])
                dts = np.append(dts, 0.0)
                d2 = dts * dts
                vals = base[None, :] + A[:, None] * dts[None, :] + B[:, None] * d2[None, :]
                rm = np.max(vals, axis=0)
                bd = int(np.argmin(rm))
                bestv = float(rm[bd])
                if bestv < curv - 1e-12:
                    step = (hi - lo) / max(1, grid - 1)
                    if step > 1e-13:
                        dref = dts[bd] + np.linspace(-2.0 * step, 2.0 * step, 21)
                        dref = np.clip(dref, lo, hi)
                        rref = np.max(
                            base[None, :]
                            + A[:, None] * dref[None, :]
                            + B[:, None] * (dref * dref)[None, :],
                            axis=0,
                        )
                        br = int(np.argmin(rref))
                        if float(rref[br]) < bestv:
                            bestv = float(rref[br])
                            dts[bd] = dref[br]
                    h[i] += dts[bd]
                    h[j] -= dts[bd]
                    improved = True
        if not improved:
            break
    h = np.clip(h, 0.0, 1.0)
    sc0 = exact_score(h, n)
    h2 = project_rows(h.reshape(1, -1))[0]
    sc1 = exact_score(h2, n)
    if sc1 < sc0:
        return h2, float(sc1)
    return h, float(sc0)

    rng = np.random.default_rng(abs(sd) % (1 << 32))

    pop, scores = build_population(n, parent, fallback, rng, P)
    bi = int(np.argmin(scores))
    best_de = pop[bi].copy()
    best_de_sc = float(scores[bi])
    if best_de_sc < best_sc:
        best = best_de.copy()
        best_sc = best_de_sc

    F = rng.uniform(0.45, 0.85, P)
    CR = rng.uniform(0.35, 0.9, P)
    restarts_left = maxre
    rest_num = 1
    no = 0

    for _ in range(gens):
        pop, scores = de_step(pop, scores, F, CR, rng, n)
        bi = int(np.argmin(scores))
        gs = float(scores[bi])
        if gs < best_de_sc - 1e-12:
            best_de_sc = gs
            best_de = pop[bi].copy()
            no = 0
            if gs < best_sc - 1e-12:
                best = best_de.copy()
                best_sc = gs
        else:
            no += 1
            if no >= stall:
                if restarts_left <= 0:
                    break
                restarts_left -= 1
                order = np.argsort(scores)
                kc = max(2, P // 8)
                base_rows = [pop[order[r]] for r in range(kc)]
                sig = min(0.25, 0.04 * (2.0 ** rest_num))
                sgr = (0.15, 0.40, 0.80)[rest_num % 3]
                raw = []
                for _ in range(P - kc):
                    if rng.random() < 0.7:
                        raw.append(pop[bi] + sig * rng.normal(0.0, 1.0, n))
                    else:
                        raw.append(fallback + sgr * rng.normal(0.0, 1.0, n))
                Hnew = project_rows(np.vstack(raw))
                rows = base_rows + [Hnew[k] for k in range(Hnew.shape[0])]
                pop = np.vstack(rows)
                scores = score_rows(pop)
                F = rng.uniform(0.45, 0.85, P)
                CR = rng.uniform(0.35, 0.9, P)
                bi = int(np.argmin(scores))
                gs = float(scores[bi])
                if gs < best_de_sc - 1e-12:
                    best_de_sc = gs
                    best_de = pop[bi].copy()
                    if gs < best_sc - 1e-12:
                        best = best_de.copy()
                        best_sc = gs
                no = 0
                rest_num += 1

    if do_poly and maxc > 0:
        cands = []
        add_candidate(cands, parent, n)
        add_candidate(cands, best, n)
        add_candidate(cands, best_de, n)
        ordered = np.argsort(scores)
        for idx in ordered[:12]:
            add_candidate(cands, pop[idx], n)
            if len(cands) >= maxc:
                break
        if len(cands) < maxc:
            add_candidate(cands, fallback, n)

        scored = []
        for c in cands[:maxc]:
            cs = exact_score(c, n)
            if np.isfinite(cs):
                scored.append((cs, c))
        if scored:
            scs = np.empty(len(scored))
            for k, item in enumerate(scored):
                scs[k] = item[0]
            for rank, oidx in enumerate(np.argsort(scs)):
                vec = scored[oidx][1]
                sw = polysw if rank == 0 else max(1, polysw - 1)
                if sw <= 0:
                    continue
                hh, ss = pair_polish(vec, n, sw, grid)
                if ss < best_sc:
                    best = hh.copy()
                    best_sc = ss

    final = project_rows(best.reshape(1, -1))[0]
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > best_sc + 1e-9:
        final = best.copy()
        fs = best_sc
    if not np.isfinite(fs):
        final = fallback.copy()
        fs = exact_score(final, n)
    if not np.isfinite(fs):
        final = fallback.copy()
        fs = 0.5
    return final, float(fs), int(n)
