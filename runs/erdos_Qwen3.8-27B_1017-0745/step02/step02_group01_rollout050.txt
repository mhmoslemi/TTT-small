import numpy as np
import math

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(float(np.sum(h)) - n / 2.0) > 1e-9 * max(1, n):
        return False
    return True

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape[0] != n or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    target = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - np.sum(h)
    step_dir = 1
    for i in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            c = min(r, 1.0 - h[i])
            h[i] += c
            r -= c
        else:
            c = min(-r, h[i])
            h[i] -= c
            r += c
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    if abs(float(np.sum(h)) - target) > 1e-9 * max(1, n):
        return None
    return h

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full')
    if q.shape[0] != 2 * n - 1 or not np.all(np.isfinite(q)):
        return float('inf')
    return float(np.max(q) * (2.0 / n))

def pair_bounds(h, i, j):
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    return lo, hi

def make_proposal(h, n, rng, step):
    if rng.random() < 0.8:
        for _ in range(4):
            idx = rng.choice(n, size=2, replace=False)
            i, j = int(idx[0]), int(idx[1])
            lo, hi = pair_bounds(h, i, j)
            if hi - lo < 1e-12:
                continue
            if rng.random() < 0.75:
                p_lo = max(lo, -step)
                p_hi = min(hi, step)
                if p_hi - p_lo < 1e-12:
                    continue
                t = rng.uniform(p_lo, p_hi)
            else:
                ep = [lo, hi]
                t = ep[int(rng.integers(2))]
                if abs(t) < 1e-12:
                    t = ep[1 - int(rng.integers(2))]
                    if abs(t) < 1e-12:
                        continue
            cand = h.copy()
            cand[i] += t
            cand[j] -= t
            proj = project_capped(cand, n)
            if proj is not None:
                return proj
    z = h + step * rng.standard_normal(n)
    return project_capped(z, n)

def build_starts(fallback, n, rng, count):
    pool = []
    def add(cand):
        if cand is None:
            return
        for s in pool:
            if np.max(np.abs(cand - s)) < 1e-8:
                return
        if len(pool) < count:
            pool.append(cand.copy())
    add(fallback)
    add(project_capped(fallback[::-1].copy(), n))
    add(project_capped(1.0 - fallback[::-1].copy(), n))
    add(np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    add(project_capped(blk, n))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    add(project_capped(alt, n))
    for sd in (0.04, 0.12, 0.25):
        add(project_capped(fallback + sd * rng.standard_normal(n), n))
    add(project_capped(rng.uniform(0, 1, n), n))
    t = np.linspace(0, 1, n)
    for k in (1, 2):
        add(project_capped(0.5 + 0.3 * np.cos(2.0 * np.pi * k * t), n))
        add(project_capped(0.5 + 0.3 * np.cos(2.0 * np.pi * k * t + 1.0), n))
    return pool[:count]

def anneal_chain(start, n, rng, steps, best_h, best_score):
    current = start.copy()
    cur_score = exact_score(current, n)
    if cur_score < best_score - 1e-12:
        best_score = cur_score
        best_h = current.copy()
    step_size = 0.20
    T0 = 0.002
    Tmin = 1e-6
    epoch_len = max(1, steps // 3)
    epoch = 0
    accepted = 0
    attempted = 0
    for t_iter in range(steps):
        if t_iter > 0 and t_iter % epoch_len == 0 and epoch < 2:
            epoch += 1
            sigma = 0.08 + 0.12 * rng.random()
            restart = project_capped(best_h + sigma * rng.standard_normal(n), n)
            if restart is not None:
                current = restart
                cur_score = exact_score(current, n)
            step_size = 0.20
            accepted = 0
            attempted = 0
        phase = t_iter % epoch_len
        frac = phase / max(1, epoch_len - 1)
        T = T0 * (Tmin / T0) ** frac
        prop = make_proposal(current, n, rng, step_size)
        if prop is None:
            continue
        attempted += 1
        s_new = exact_score(prop, n)
        if not np.isfinite(s_new):
            continue
        if s_new <= cur_score or rng.random() < math.exp(max(-745, -(s_new - cur_score) / T)):
            current = prop
            cur_score = s_new
            accepted += 1
        if s_new < best_score - 1e-12:
            best_score = s_new
            best_h = prop.copy()
        if attempted > 0 and attempted % 100 == 0:
            ratio = accepted / max(1, attempted)
            if ratio > 0.45:
                step_size = min(0.6, step_size * 1.20)
            elif ratio < 0.15:
                step_size = max(1e-4, step_size * 0.70)
            accepted = 0
            attempted = 0
    return best_h, best_score, current

def polish_pairwise(h, n, rng):
    current = h.copy()
    cur_score = exact_score(current, n)
    if not np.isfinite(cur_score):
        return current, cur_score
    meshes = [0.04, 0.01, 0.0025, 0.0005, 0.0001]
    budget = min(2500, max(300, int(2500 * (47.0 / n) ** 2)))
    used = 0
    for mesh in meshes:
        improved = True
        while improved and used < budget:
            improved = False
            n_pairs = min(100, max(20, 2 * n))
            indices = np.arange(n)
            rng.shuffle(indices)
            for pi in range(n_pairs):
                if used >= budget:
                    break
                i = int(indices[pi % n])
                j = int(indices[(pi + 1 + int(pi * 0.618 * n)) % n])
                if i == j:
                    continue
                lo, hi = pair_bounds(current, i, j)
                for mult in (-1.0, -0.5, 0.5, 1.0):
                    if used >= budget:
                        break
                    raw = mult * mesh
                    t = max(lo, min(hi, raw))
                    if abs(t) < 1e-14:
                        continue
                    cand = current.copy()
                    cand[i] += t
                    cand[j] -= t
                    proj = project_capped(cand, n)
                    if proj is None:
                        continue
                    s = exact_score(proj, n)
                    used += 1
                    if s < cur_score - 1e-12:
                        current = proj
                        cur_score = s
                        improved = True
    return current, cur_score
