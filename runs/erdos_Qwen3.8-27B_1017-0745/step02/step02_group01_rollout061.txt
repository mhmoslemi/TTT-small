                v = fallback + rng.normal(0, 0.1, n)
            trial = pool[i].copy()
            mask = rng.random(n) < cri
            forced = int(rng.integers(n))
            mask[forced] = True
            trial[mask] = v[mask]
            proj = project_capped(trial, n)
            if proj is None:
                continue
            ts = exact_score(proj, n)
            if ts < best_score - 1e-12:
                best_score = ts
                best_h = proj.copy()
                improved = True
            if ts < scores[i] - 1e-12:
                scores[i] = ts
                pool[i] = proj
                best_idx = i
        if not improved:
            stagnant += 1
        else:
            stagnant = 0
        if stagnant >= 12 and restarts < 2 and used < B:
            stagnant = 0
            restarts += 1
            n_retain = max(2, P // 2)
            n_replace = P - n_retain
            avail = list(range(P))
            replace_idx = []
            for _ in range(n_replace):
                wi = avail[0]
                ws = scores[wi]
                for j in avail:
                    if scores[j] > ws:
                        ws = scores[j]
                        wi = j
                replace_idx.append(wi)
                avail.remove(wi)
            for cnt, idx in enumerate(replace_idx):
                if used >= B:
                    break
                used += 1
                if cnt % 4 == 0:
                    prop = rng.uniform(0, 1, n)
                else:
                    sig = sig_list[cnt % 3]
                    prop = best_h + sig * rng.normal(0, 1, n)
                np2 = project_capped(prop, n)
                if np2 is not None:
                    scores[idx] = exact_score(np2, n)
                    pool[idx] = np2
                    f_vals[idx] = 0.6
                    cr_vals[idx] = 0.8
                    if scores[idx] < best_score - 1e-12:
                        best_score = scores[idx]
                        best_h = np2.copy()
                        best_idx = idx
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    fb = exact_score(fallback, n)
    if fs > fb + 1e-12:
        final, fs = fallback, fb
    if not is_feasible(final, n):
        final = np.full(n, 0.5)
        fs = exact_score(final, n)
    return final, float(fs), int(n)
