def pairwise_polish(x, n, sweeps):
    best = x.copy()
    best_s = exact_score(best, n)
    for _ in range(sweeps):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                diff = best[i] - best[j]
                if abs(diff) < 1e-14:
                    continue
                if diff > 0:
                    lo, hi = 0.0, diff
                else:
                    lo, hi = diff, 0.0
                cands = np.linspace(lo, hi, 20)
                c0 = best_s
                for dd in cands:
                    if dd == 0:

                    trial = best.copy()
                    trial[i] += dd
                    trial[j] -= dd
                    if trial[i] < -1e-12 or trial[i] > 1 + 1e-12:

                    if trial[j] < -1e-12 or trial[j] > 1 + 1e-12:

                    if sc < c0 - 1e-12:
                        c0 = sc
                        best_dd = dd

                    best[j] -= best_dd
                    best_s = c0
                    improved = True
        if not improved:

    starts = starts[:n_starts]
    best, best_sc = fallback.copy(), fb_sc
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_sc - 1e-12:
            best, best_sc = st.copy(), ss
    for st in starts:
        try:
            best, best_sc = pg_optimize(st, n, iters, best, best_sc)
        except Exception:
            pass
    hop_sds = [0.025, 0.08] if n <= 128 else [0.05]
    for hs in hop_sds:
        try:
            pert = best + rng.normal(0, hs, n)
            pert_p = project_capped(pert, n)
            if pert_p is None:
                continue
            ps = exact_score(pert_p, n)
            if np.isfinite(ps) and ps < best_sc - 1e-12:
                best, best_sc = pert_p.copy(), ps
            best, best_sc = pg_optimize(pert_p, n, iters, best, best_sc)
        except Exception:
            pass
    if n <= 128:
        try:
            polished, poly_sc = pairwise_polish(best, n, 3)
            fp = project_capped(polished, n)
            if fp is not None:
                fs = exact_score(fp, n)
                if np.isfinite(fs) and fs < best_sc - 1e-12:
                    best, best_sc = fp, fs
        except Exception:
            pass
    final = project_capped(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_sc:
        final, fs = fallback.copy(), fb_sc
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final, float(fs), int(n)
