def smooth_val_grad(h, n, tau):
    Q = lag_vals(h, n)
    qmax = np.max(Q)
    ex = np.exp((Q - qmax) / tau)
    Z = np.sum(ex)
    p = ex / Z
    F = qmax + tau * np.log(Z)
    dx = 2.0 / n
    g = p[0] * dx * (1.0 - 2.0 * h)
    idx = 1
    for r in range(1, n):
        m = n - r
        g[:m] += dx * p[idx] * (1.0 - h[r:]); g[r:] -= dx * p[idx] * h[:m]
        idx += 1
        g[:m] -= dx * p[idx] * h[r:]; g[r:] += dx * p[idx] * (1.0 - h[:m])
        idx += 1
    return F, g

def smooth_refine(h, n):
    h = project_capped(h, n)
    if h is None:
        return None, float('inf')
    best, best_s = h.copy(), exact_score(h, n)
    for tau in [0.02, 0.005, 0.001, 0.0002]:
        alpha = 1.0
        for _it in range(15):
            F, g = smooth_val_grad(h, n, tau)
            if not np.all(np.isfinite(g)):
                break
            y = project_capped(h - alpha * g, n)
            if y is None:
                break
            d = y - h
            if np.max(np.abs(d)) < 1e-10:
                break
            accepted = False
            for _b in range(20):
                Fy, _ = smooth_val_grad(y, n, tau)
                if Fy <= F + 1e-4 * np.dot(g, d) + 1e-14:
                    accepted = True
                    break
                alpha *= 0.5
                y = project_capped(h - alpha * g, n)
                if y is None:
                    break
                d = y - h
            if not accepted:
                break
            h = y
            alpha = min(1.5 * alpha, 5.0)
            s = exact_score(h, n)
            if s < best_s - 1e-12:
                best_s = s
                best = h.copy()
    return best, best_s

def build_starts(fb, n, rng):
    starts = []
    def add(v):
        if v is None:
            return
        v = project_capped(v, n)
        if v is None:
            return
        for s in starts:
            if np.max(np.abs(v - s)) < 1e-8:
                return
        starts.append(v)
    add(fb.copy())
    add(fb[::-1].copy())
    add(1.0 - fb)
    add(1.0 - fb[::-1])
    add(np.full(n, 0.5))
    blk = np.zeros(n); blk[:n // 2] = 1.0
    if n % 2: blk[n // 2] = 0.5
    add(blk)
    add(fb + rng.normal(0, 0.08, n))
    add(fb + rng.normal(0, 0.2, n))
    add(rng.uniform(0, 1, n))
    for mode in [1, 2, 3]:
        t = np.arange(n)
        add(0.5 + 0.35 * np.cos(2.0 * np.pi * mode * t / n))
        add(0.5 + 0.35 * np.sin(2.0 * np.pi * mode * t / n))
    return starts

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None:
        n = 47
        try:
            vk = int(kwargs.get('n_points', 47))
            if 1 <= vk <= 999:
                n = vk
        except Exception:
            pass
    if parent is not None:
        fb = project_capped(np.clip(parent, 0.0, 1.0), n)
    else:
        fb = None
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    fb = fb.copy()
    rng = np.random.default_rng(int(seed) if seed is not None else 42)
    if n == 1:
        h1 = np.array([0.5])
        return h1, exact_score(h1, 1), 1
    bs = exact_score(fb, n)
    bh = fb.copy()
    starts = build_starts(fb, n, rng)
    max_starts = 10 if n <= 64 else (7 if n <= 128 else 4)
    starts = starts[:max_starts]
    for st in starts:
        s = exact_score(st, n)
        if s < bs - 1e-12:
            bs = s
            bh = st.copy()
    if n <= 128:
        for st in starts:
            res = slsqp_solve(st, n)
            for candidate in [res, project_capped(0.5 * st + 0.5 * (res if res is not None else st), n), project_capped(0.25 * st + 0.75 * (res if res is not None else st), n), project_capped(0.75 * st + 0.25 * (res if res is not None else st), n)]:
                if candidate is not None and is_feasible(candidate, n):
                    cs = exact_score(candidate, n)
                    if cs < bs - 1e-12:
                        bs = cs
                        bh = candidate.copy()
    bh2, bs2 = smooth_refine(bh, n)
    if bh2 is not None and bs2 < bs - 1e-12:
        bh, bs = bh2, bs2
    for st in starts[:3]:
        h3, s3 = smooth_refine(st, n)
        if h3 is not None and s3 < bs - 1e-12:
            bh, bs = h3, s3
    bh = project_capped(bh, n)
    if bh is None or not is_feasible(bh, n):
        bh = fb.copy()
        bs = exact_score(fb, n)
    else:
        bs_check = exact_score(bh, n)
        if bs_check < bs - 1e-12:
            bs = bs_check
        fb_s = exact_score(fb, n)
        if fb_s < bs - 1e-12:
            bh, bs = fb.copy(), fb_s
    return bh.astype(np.float64), float(bs), int(n)
