def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n or n <= 0 or not np.all(np.isfinite(z)): return None
    z = np.clip(z, -1e6, 1e6)
    s = 0.5*n
    lo, hi = float(np.min(z))-1.0, float(np.max(z))+1.0
    for _ in range(80):
        mid = 0.5*(lo+hi)
        if np.sum(np.clip(z-mid, 0.0, 1.0)) > s: lo = mid
        else: hi = mid
    h = np.clip(z-0.5*(lo+hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(3):
        if abs(r) < 1e-13: break
        if r > 0:
            order = np.argsort(1.0-h)
        else:
            order = np.argsort(-h)
        for idx in order:
            if r > 0:
                add = min(r, 1.0-h[idx])
                if add > 0:
                    h[idx] += add; r -= add
                    if abs(r) < 1e-14: break
            else:
                sub = min(-r, h[idx])
                if sub > 0:

                    if abs(r) < 1e-14: break

def active_direction(h,n):
    try:
        c = np.correlate(h, 1.0-h, mode='full')*(2.0/n)
    except Exception:
        return None
    m = float(np.max(c))
    active = c >= m - 1e-5
    d = np.zeros(n)
    for idx in np.nonzero(active)[0]:
        r = int(idx - (n - 1))
        g = np.zeros(n)
        if r == 0:
            g = 1.0 - 2.0*h
        elif r > 0:
            g[r:] = 1.0 - h[:n-r]
            g[:n-r] += -h[r:]
        else:
            s = -r
            g[:n-s] = 1.0 - h[s:]
            g[s:] += -h[:n-s]
        d += g
    d -= np.mean(d)
    mx = float(np.max(np.abs(d)))
    if mx < 1e-14 or not np.isfinite(mx): return None
    return d / mx

def cma_run(base, U, sigma0, gens, rng, bh, bs, n):
    p = U.shape[1]
    if p <= 0 or gens <= 0: return bh, bs
    lam = max(6, int(4 + 3*math.log(p)))
    mu = max(2, lam//2)
    raw = np.array([math.log(mu+0.5)-math.log(j) for j in range(1,mu+1)], dtype=np.float64)
    w = raw/raw.sum()
    meff = 1.0/float(np.sum(w*w))
    cs = (meff+2.0)/(p+meff+5.0)
    ds = 1.0 + 2.0*max(0.0, math.sqrt((meff-1.0)/(p+1.0)) - 1.0) + cs
    cc = (4.0 + meff/p)/(p + 4.0 + 2.0*meff/p)
    c1 = 2.0/((p+1.3)**2 + meff)
    cmu = max(0.0, min(1.0-c1, 2.0*(meff-2.0+1.0/meff)/((p+2.0)**2 + meff)))
    chi = math.sqrt(p)*(1.0 - 1.0/(4.0*p) + 1.0/(21.0*p*p))
    m = np.zeros(p); C = np.eye(p); ps = np.zeros(p); pc = np.zeros(p)
    sigma = max(1e-5, min(1.0, float(sigma0)))
    for g in range(1, gens+1):
        ok = True
        try:
            vals, V = np.linalg.eigh(C)
            if not (np.all(np.isfinite(vals)) and np.all(np.isfinite(V))): ok=False
        except Exception:
            ok = False
        if not ok:
            vals, V = np.ones(p), np.eye(p)
            C = np.eye(p); ps = np.zeros(p); pc = np.zeros(p)
            sigma = max(1e-5, 0.5*sigma)
        vals = np.clip(vals, 1e-10, 1e4)
        rt = np.sqrt(vals)
        X = np.empty((lam,p)); Y = np.empty((lam,p)); S = np.empty(lam)
        for i in range(lam):
            z = rng.standard_normal(p)
            y = V @ (rt*z)
            x = m + sigma*y
            X[i] = x; Y[i] = y
            h = project_capped(base + U @ x, n)
            sc = exact_score(h,n) if h is not None else float('inf')
            S[i] = sc
            if h is not None and math.isfinite(sc) and sc < bs - 1e-12:
                bs = sc; bh = h.copy()
        order = np.argsort(S, kind='mergesort')
        finite = [int(i) for i in order if math.isfinite(S[i])]
        if len(finite) < mu: break
        sel = finite[:mu]
        yw = np.zeros(p)
        for j, ii in enumerate(sel): yw += w[j]*Y[ii]
        mn = m + sigma*yw
        ciw = V @ ((V.T @ yw) / rt)
        ps = (1.0-cs)*ps + math.sqrt(cs*(2.0-cs)*meff)*ciw
        nps = float(np.linalg.norm(ps))
        hsig = 1 if nps / math.sqrt(max(1e-15, 1.0-(1.0-cs)**(2*g))) < (1.4 + 2.0/(p+1.0))*chi else 0
        pc = (1.0-cc)*pc + hsig*math.sqrt(cc*(2.0-cc)*meff)*yw
        rm = np.zeros((p,p))
        for j, ii in enumerate(sel): rm += w[j]*np.outer(Y[ii], Y[ii])
        corr = (1.0-float(hsig))*cc*(2.0-cc-c1)
        C = (1.0-c1-cmu)*C + c1*(np.outer(pc,pc)+corr*C) + cmu*rm
        C = 0.5*(C+C.T)
        step = (cs/ds)*(nps/chi - 1.0)
        step = max(-1.0, min(1.0, step))
        sigma = max(1e-5, min(1.0, sigma*math.exp(step)))
        if not (np.all(np.isfinite(mn)) and np.all(np.isfinite(C))): return bh, bs
        m = mn
        if sigma <= 1.000001e-5: break
    return bh, bs

import numpy as np
import math


def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n or n <= 0 or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = 0.5 * float(n)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(3):
        if abs(r) < 1e-13:
            break
        if r > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(-h)
        for idx in order:
            if r > 0:
                add = min(r, 1.0 - h[idx])
                if add > 0:
                    h[idx] += add
                    r -= add
                    if abs(r) < 1e-14:
                        break
            else:
                sub = min(-r, h[idx])
                if sub > 0:
                    h[idx] -= sub
                    r += sub
                    if abs(r) < 1e-14:
                        break
    for _ in range(2):
        r = s - float(np.sum(h))
        if abs(r) <= 1e-12:
            break
        idx = int(np.argmax(1.0 - h)) if r > 0 else int(np.argmax(h))
        h[idx] = min(1.0, max(0.0, h[idx] + r))
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(float(np.sum(h)) - s) > 1e-9:
        return None
    return h


def is_feasible(h, n):
    try:
        h = np.asarray(h, dtype=np.float64)
    except Exception:
        return False
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(float(np.sum(h)) - 0.5 * float(n)) <= 1e-9


def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    try:
        v = float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))
    except Exception:
        return float('inf')
    return v if math.isfinite(v) else float('inf')


def near_any(a, x, eps=1e-12):
    for y in a:
        if abs(x - y) < eps:
            return True
    return False


def add_candidate(c, t, lo, hi):
    eps = 1e-12
    if t <= lo + eps or t >= hi - eps:
        return
    t = min(hi - eps, max(lo + eps, t))
    if abs(t) < 1e-13 or near_any(c, t, eps):
        return
    if len(c) < 10:
        c.append(t)


def try_add(pool, v, n):
    if v is None or v.shape != (n,) or not np.all(np.isfinite(v)):
        return
    if np.any(v < -1e-12) or np.any(v > 1.0 + 1e-12):
        return
    if abs(float(np.sum(v)) - 0.5 * float(n)) > 1e-9:
        return
    for s in pool:
        if np.max(np.abs(v - s)) < 1e-8:
            return
    pool.append(v)


def add_proj(pool, v, n):
    try_add(pool, project_capped(v, n), n)


def make_binary(rng, n):
    v = np.zeros(n)
    p = rng.permutation(n)
    v[p[:n // 2]] = 1.0
    if n % 2:
        v[p[n // 2]] = 0.5
    return v


def build_basis(n):
    p = min(n - 1, 48)
    if p <= 0:
        return np.empty((n, 0), dtype=np.float64)
    ii = np.arange(n)[:, None]
    kk = np.arange(1, p + 1)[None, :]
    return (math.sqrt(2.0 / n) * np.cos(np.pi * (ii + 0.5) * kk / n)).astype(np.float64)


def build_pool(fb, n, rng, cap=16):
    pool = []
    try_add(pool, fb, n)
    if not pool:
        add_proj(pool, np.full(n, 0.5), n)
    add_proj(pool, np.roll(fb, n // 3), n)
    add_proj(pool, np.roll(fb, n // 2), n)
    add_proj(pool, fb[::-1], n)
    add_proj(pool, 1.0 - fb, n)
    add_proj(pool, (1.0 - fb)[::-1], n)
    add_proj(pool, np.full(n, 0.5), n)

    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    add_proj(pool, blk, n)

    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    add_proj(pool, alt, n)

    sc = max(0.5, min(1.5, math.sqrt(47.0 / n)))
    add_proj(pool, fb + rng.normal(0.0, 0.06 * sc, n), n)
    add_proj(pool, fb + rng.normal(0.0, 0.18 * sc, n), n)
    add_proj(pool, make_binary(rng, n), n)

    ts = np.arange(n, dtype=np.float64)
    for k in (1, 2):
        a = 2.0 * math.pi * k * ts / float(n)
        add_proj(pool, 0.5 + 0.35 * np.cos(a), n)
        add_proj(pool, 0.5 + 0.35 * np.cos(a + 0.5 * math.pi), n)

    attempts = 0
    while len(pool) < cap and attempts < 220:
        attempts += 1
        r = rng.random()
        if len(pool) and r < 0.45:
            base = pool[int(rng.integers(len(pool)))]
            sig = float(rng.choice([0.04, 0.10, 0.20])) * sc
            add_proj(pool, base + rng.normal(0.0, sig, n), n)
        elif r < 0.65:
            add_proj(pool, rng.uniform(0.0, 1.0, n), n)
        elif r < 0.85:
            add_proj(pool, make_binary(rng, n), n)
        else:
            base = pool[int(rng.integers(len(pool)))] if pool else np.full(n, 0.5)
            add_proj(pool, base + rng.normal(0.0, 0.35 * sc, n), n)

    while len(pool) < cap:
        pool.append(pool[0].copy())
    return [p.copy() for p in pool]


def interval_dir(h, v):
    p = v > 1e-14
    q = v < -1e-14
    lo, hi = -1e9, 1e9
    if np.any(p):
        lo = max(lo, float(np.max(-h[p] / v[p])))
        hi = min(hi, float(np.max((1.0 - h[p]) / v[p])))
    if np.any(q):
        lo = max(lo, float(np.min((1.0 - h[q]) / v[q])))
        hi = min(hi, float(np.max(-h[q] / v[q])))
    if not (np.isfinite(lo) and np.isfinite(hi)) or lo >= hi - 1e-12:
        return None
    return lo, hi


def active_direction(h, n):
    try:
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    except Exception:
        return None
    m = float(np.max(c))
    active = c >= m - 1e-5
    d = np.zeros(n)
    for idx in np.nonzero(active)[0]:
        r = int(idx - (n - 1))
        g = np.zeros(n)
        if r == 0:
            g = 1.0 - 2.0 * h
        elif r > 0:
            g[r:] = 1.0 - h[:n - r]
            g[:n - r] += -h[r:]
        else:
            s = -r
            g[:n - s] = 1.0 - h[s:]
            g[s:] += -h[:n - s]
        wt = max(0.0, float(c[idx]) - (m - 1e-5)) / 1e-5
        d += wt * g
    d -= np.mean(d)
    mx = float(np.max(np.abs(d)))
    if mx < 1e-14 or not np.isfinite(mx):
        return None
    return d / mx


def local_polish(hb, sb, pool, scores, n, rng, rounds, ndirs):
    if not is_feasible(hb, n):
        return None, float('inf')
    order = np.argsort(scores, kind='mergesort')
    top = [hb.copy()]
    for j in order[:3]:
        j = int(j)
        if 0 <= j < len(pool):
            top.append(pool[j].copy())
    sc = max(0.5, min(1.5, math.sqrt(47.0 / n)))
    delta = min(0.35, 0.22 * sc)
    for _ in range(rounds):
        if delta < 1e-4:
            break
        improved = False
        for _d in range(ndirs):
            r = rng.random()
            if r < 0.35:
                ad = active_direction(hb, n)
                v = ad if ad is not None else rng.normal(0.0, 1.0, n)
            elif r < 0.65:
                i1 = int(rng.integers(len(top)))
                i2 = int(rng.integers(len(top)))
                v = top[i1] - top[i2]
            elif r < 0.85:
                v = rng.normal(0.0, 1.0, n)
            else:
                v = np.zeros(n)
                if n == 1:
                    continue
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    j = (j + 1) % n
                v[i] = 1.0
                v[j] = -1.0
            v = v - np.mean(v)
            mx = float(np.max(np.abs(v)))
            if mx < 1e-14 or not np.isfinite(mx):
                continue
            v = v / mx
            iv = interval_dir(hb, v)
            if iv is None:
                continue
            lo, hi = iv
            if hi - lo < 1e-10:
                continue
            cands = []
            for mf in (-1.0, -0.5, -0.25, 0.25, 0.5, 1.0):
                add_candidate(cands, mf * delta, lo, hi)
            add_candidate(cands, 0.5 * (lo + hi), lo, hi)
            for t in cands:
                tr = project_capped(hb + t * v, n)
                if tr is None:
                    continue
                ts = exact_score(tr, n)
                if math.isfinite(ts) and ts < sb - 1e-12:
                    hb, sb = tr, ts
                    improved = True
                    break
            if improved:
                delta = 0.75 * delta if delta > 0.08 else min(0.35, 1.25 * delta)
                break
        if not improved:
            delta *= 0.5
    if not is_feasible(hb, n):
        return None, float('inf')
    return hb, sb


def cma_run(base, U, sigma0, gens, rng, bh, bs, n):
    p = U.shape[1]
    if p <= 0 or gens <= 0:
        return bh, bs
    lam = max(6, int(4 + 3 * math.log(p)))
    mu = max(2, lam // 2)
    raw = np.array([math.log(mu + 0.5) - math.log(j) for j in range(1, mu + 1)], dtype=np.float64)
    w = raw / raw.sum()
    meff = 1.0 / float(np.sum(w * w))
    cs = (meff + 2.0) / (p + meff + 5.0)
    ds = 1.0 + 2.0 * max(0.0, math.sqrt((meff - 1.0) / (p + 1.0)) - 1.0) + cs
    cc = (4.0 + meff / p) / (p + 4.0 + 2.0 * meff / p)
    c1 = 2.0 / ((p + 1.3) ** 2 + meff)
    cmu = max(0.0, min(1.0 - c1, 2.0 * (meff - 2.0 + 1.0 / meff) / ((p + 2.0) ** 2 + meff)))
    chi = math.sqrt(p) * (1.0 - 1.0 / (4.0 * p) + 1.0 / (21.0 * p * p))

    m = np.zeros(p)
    C = np.eye(p)
    ps = np.zeros(p)
    pc = np.zeros(p)
    sigma = max(1e-5, min(1.0, float(sigma0)))

    for g in range(1, gens + 1):
        ok = True
        try:
            vals, V = np.linalg.eigh(C)
            if not (np.all(np.isfinite(vals)) and np.all(np.isfinite(V))):
                ok = False
        except Exception:
            ok = False
        if not ok:
            vals = np.ones(p)
            V = np.eye(p)
            C = np.eye(p)
            ps = np.zeros(p)
            pc = np.zeros(p)
            sigma = max(1e-5, 0.5 * sigma)

        vals = np.clip(vals, 1e-10, 1e4)
        rt = np.sqrt(vals)
        X = np.empty((lam, p), dtype=np.float64)
        Y = np.empty((lam, p), dtype=np.float64)
        S = np.empty(lam, dtype=np.float64)

        for i in range(lam):
            z = rng.standard_normal(p)
            y = V @ (rt * z)
            x = m + sigma * y
            x = np.clip(x, -1e6, 1e6)
            X[i] = x
            Y[i] = y
            h = project_capped(base + U @ x, n)
            sc = exact_score(h, n) if h is not None else float('inf')
            S[i] = sc
            if h is not None and math.isfinite(sc) and sc < bs - 1e-12:
                bs = sc
                bh = h.copy()

        order = np.argsort(S, kind='mergesort')
        finite = [int(i) for i in order if math.isfinite(S[i])]
        if len(finite) < mu:
            break
        sel = finite[:mu]

        yw = np.zeros(p)
        for j, ii in enumerate(sel):
            yw += w[j] * Y[ii]
        mn = m + sigma * yw
        ciw = V @ ((V.T @ yw) / rt)
        ps = (1.0 - cs) * ps + math.sqrt(cs * (2.0 - cs) * meff) * ciw
        nps = float(np.linalg.norm(ps))
        hsig = 1 if nps / math.sqrt(max(1e-15, 1.0 - (1.0 - cs) ** (2 * g))) < (1.4 + 2.0 / (p + 1.0)) * chi else 0
        pc = (1.0 - cc) * pc + hsig * math.sqrt(cc * (2.0 - cc) * meff) * yw

        rm = np.zeros((p, p), dtype=np.float64)
        for j, ii in enumerate(sel):
            rm += w[j] * np.outer(Y[ii], Y[ii])

        corr = (1.0 - float(hsig)) * cc * (2.0 - cc - c1)
        C = (1.0 - c1 - cmu) * C + c1 * (np.outer(pc, pc) + corr * C) + cmu * rm
        C = 0.5 * (C + C.T)

        step = (cs / ds) * (nps / chi - 1.0)
        step = max(-1.0, min(1.0, step))
        sigma = max(1e-5, min(1.0, sigma * math.exp(step)))

        if not (np.all(np.isfinite(mn)) and np.all(np.isfinite(C))):
            return bh, bs
        m = mn
        if sigma <= 1.000001e-5:
            break
    return bh, bs


def choose_bases(pool, scores, R):
    order = np.argsort(scores, kind='mergesort')
    sel = [0]
    for idx in order:
        idx = int(idx)
        if idx in sel:
            continue
        if len(sel) >= R:
            break
        ok = True
        for si in sel:
            if np.max(np.abs(pool[idx] - pool[si])) <= 0.05:
                ok = False
                break
        if ok:
            sel.append(idx)
    for idx in order:
        idx = int(idx)
        if idx in sel:
            continue
        sel.append(idx)
        if len(sel) >= R:
            break
    return sel


def run(seed=42, **kwargs):
    n = None
    fb = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                fb = project_capped(p, n)
    except Exception:
        pass

    if fb is None:
        if n is None:
            try:
                vk = int(kwargs.get('n_points', 47))
                if 1 <= vk <= 999:
                    n = vk
            except Exception:
                pass
            if n is None:
                n = 47
        fb = np.full(n, 0.5)

    if n is None or not (1 <= n <= 999):
        n = 47
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    fb = fb.copy()

    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)

    if n == 1:
        h1 = np.array([0.5], dtype=np.float64)
        return h1, float(exact_score(h1, 1)), 1

    orig_fb = fb.copy()
    bh = fb.copy()
    bs = float(exact_score(fb, n))

    lp0 = local_polish(fb.copy(), bs, [fb], np.array([bs], dtype=np.float64), n, rng, 8, 6)
    if lp0[0] is not None and math.isfinite(lp0[1]) and lp0[1] < bs - 1e-12:
        bh, bs = lp0[0], float(lp0[1])
        fb = bh.copy()

    U = build_basis(n)
    pool = build_pool(fb, n, rng, 16)
    ps = np.array([exact_score(pv, n) for pv in pool], dtype=np.float64)
    bi = int(np.argmin(ps))
    if math.isfinite(ps[bi]) and ps[bi] < bs - 1e-12:
        bh = pool[bi].copy()
        bs = float(ps[bi])

    if n <= 64:
        R, G, GP = 4, 120, 35
    elif n <= 128:
        R, G, GP = 3, 80, 25
    elif n <= 512:
        R, G, GP = 2, 40, 15
    else:
        R, G, GP = 2, 25, 10

    sigma0 = max(0.04, min(0.35, 0.10 * math.sqrt(float(n) / max(1, U.shape[1]))))
    bases = choose_bases(pool, ps, R)
    for b in bases:
        bh, bs = cma_run(pool[b].copy(), U, sigma0, G, rng, bh, bs, n)

    bh, bs = cma_run(bh.copy(), U, 0.5 * sigma0, GP, rng, bh, bs, n)

    lp1 = local_polish(bh, bs, pool, ps, n, rng, 18, 10)
    if lp1[0] is not None and math.isfinite(lp1[1]) and lp1[1] < bs - 1e-12:
        bh, bs = lp1[0], float(lp1[1])

    fbase = float(exact_score(orig_fb, n))
    if math.isfinite(bs) and bs < fbase - 1e-12:
        bh, bs = cma_run(bh.copy(), U, 0.4 * sigma0, max(10, GP // 2), rng, bh, bs, n)
        lp2 = local_polish(bh, bs, pool, ps, n, rng, 10, 8)
        if lp2[0] is not None and math.isfinite(lp2[1]) and lp2[1] < bs - 1e-12:
            bh, bs = lp2[0], float(lp2[1])

    final = project_capped(bh, n)
    fs = exact_score(final, n) if final is not None else float('inf')
    fbase = float(exact_score(orig_fb, n))
    if final is None or not is_feasible(final, n) or not math.isfinite(fs) or fs > fbase + 1e-12:
        final = orig_fb.copy()
        fs = fbase

    final = final.astype(np.float64)
    fs = float(exact_score(final, n))
    return final, fs, int(n)
