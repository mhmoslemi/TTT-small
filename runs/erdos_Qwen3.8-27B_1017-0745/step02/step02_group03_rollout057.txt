        _dedup_add(np.roll(fallback, n // 4))
        _dedup_add(fallback + rng.normal(0, 0.025, n))
        _dedup_add(fallback + rng.normal(0, 0.08, n))
        _dedup_add(fallback + rng.normal(0, 0.15, n))
        _dedup_add(rng.uniform(0, 1, n))
    else:
        _dedup_add(1.0 - fallback)
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        _dedup_add(blk)
        tt = np.arange(n)
        _dedup_add(0.5 + 0.45 * np.cos(2.0 * np.pi * tt / n))
        _dedup_add(rng.uniform(0, 1, n))
    starts = starts[:n_starts]
    best, best_score = fallback.copy(), fb_score
    for st in starts:
        sc = exact_score(st, n)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best, best_score = st.copy(), sc
    for st in starts:
        try:
            res = cont_opt(st, n, mi)
            if res is not None:
                sc = exact_score(res, n)
                if np.isfinite(sc) and sc < best_score - 1e-12:
                    best, best_score = res.copy(), sc
        except Exception:
            pass
    n_hops = 3 if n <= 64 else (2 if n <= 128 else 1)
    hop_sds = [0.02, 0.06, 0.12]
    for i in range(n_hops):
        pert = best + rng.normal(0, hop_sds[i % 3], n)
        pert = project_capped(pert, n)
        if pert is None:
            continue
        sc = exact_score(pert, n)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best, best_score = pert, sc
        try:
            res = cont_opt(pert, n, mi)
            if res is not None:
                sc2 = exact_score(res, n)
                if np.isfinite(sc2) and sc2 < best_score - 1e-12:
                    best, best_score = res.copy(), sc2
        except Exception:
            pass
    if n <= 80:
        pol, ps = pairwise_polish(best, n, 1)
        fp = project_capped(pol, n)
        if fp is not None:
            fs = exact_score(fp, n)
            if np.isfinite(fs) and fs < best_score - 1e-12:
                best, best_score = fp, fs
    final = project_capped(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final, fs = fallback.copy(), fb_score
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final, float(fs), int(n)
