def lse_value_grad(h, n, tau):
    dx=2/n
    corr = np.correlate(h, 1-h, 'full') * dx
    qmax = np.max(corr)
    u = (corr-qmax)/tau
    expu = np.exp(np.minimum(u,0)) # u<=0 anyway
    wsum = np.sum(expu)
    w = expu / wsum
    val = qmax + tau*np.log(wsum)
    grad = np.zeros(n)
    for k in range(len(corr)):
      if w[k] <= 1e-15: continue
      s = k-(n-1)
      p = max(s,0); r=max(-s,0); m=n-abs(s)
      wk = w[k]
      grad[p:p+m] += dx*wk*(1.0-h[r:r+m])
      grad[r:r+m] -= dx*wk*h[p:p+m]
    return val, grad
```

def smooth_value_grad(h,n,tau):
   n=int(n)
   h=np.asarray(h,dtype=np.float64).ravel()
   if h.size != n or n <=0:
       return float('inf'), np.zeros(max(n,1))
   h=np.clip(h,0.0,1.0)
   if tau <= 0.0 or not np.isfinite(tau): tau=1e-12
   dx=2.0/float(n)
   corr=np.correlate(h,1.0-h,mode='full')*dx
   if not np.all(np.isfinite(corr)):
       return float('inf'), np.zeros(n)
   qmax=float(np.max(corr))
   u=(corr-qmax)/tau
   u=np.minimum(u,0.0)
   e=np.exp(u)
   wsum=float(np.sum(e))
   if not np.isfinite(wsum) or wsum <= 0.0:
       w=np.zeros(len(corr)); w[int(np.argmax(corr))]=1.0; wsum=1.0
   else:
       w=e/wsum
   val=qmax + tau*np.log(wsum)
   grad=np.zeros(n,dtype=np.float64)
   n1=n-1
   for k in range(len(corr)):
      wk=float(w[k])
      if wk < 1e-17: continue
      s=k-n1
      if s >= 0:
         m=n-s
         grad[s:s+m] += dx*wk*(1.0-h[:m])
         grad[:m] -= dx*wk*h[s:s+m]
      else:
         m=n+s
         idx=-s
         grad[:m] += dx*wk*(1.0-h[idx:idx+m])
         grad[idx:idx+m] -= dx*wk*h[:m]
   if not np.all(np.isfinite(grad)):
       return float('inf'), np.zeros(n)
   return float(val), grad

def project_capped(z,n):
    n=int(n)

    for _ in range(55):

        else: hi=m
    h=np.clip(z-0.5*(lo+hi),0.0,1.0)

        for i in range(n):
            if abs(r)<1e-13: break

def is_feasible(h,n):

def exact_score(h,n):

def smooth_value_grad(h,n,tau):

import numpy as np
from scipy.optimize import minimize

_SLS_N = 0
_SLS_TAU = 1.0

def project_capped(z, n):
    n = int(n)
    if n <= 0:
        return None
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    t = 0.5 * n
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(55):
        m = 0.5 * (lo + hi)
        if np.sum(np.clip(z - m, 0.0, 1.0)) > t:
            lo = m
        else:
            hi = m
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = t - float(np.sum(h))
    if r > 1e-13:
        for i in range(n):
            if abs(r) < 1e-13:
                break
            a = 1.0 - h[i]
            if a > 0.0:
                b = r if r < a else a
                h[i] += b
                r -= b
    elif r < -1e-13:
        for i in range(n):
            if abs(r) < 1e-13:
                break
            if h[i] > 0.0:
                b = h[i] if -r < h[i] else -r
                h[i] -= b
                r += b
    if not np.all(np.isfinite(h)) or np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return None
    if abs(float(np.sum(h)) - t) > 1e-9 * max(1, n):
        return None
    return h

def is_feasible(h, n):
    n = int(n)
    if h is None or h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return False
    return abs(float(np.sum(h)) - 0.5 * n) <= 1e-8 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / float(n))))

def resample_h(p, m):
    m = int(m)
    p = np.asarray(p, dtype=np.float64).ravel()
    if m <= 0:
        return np.array([], dtype=np.float64)
    if p.size == 0:
        return np.full(m, 0.5)
    if m == p.size:
        return p.copy()
    if m == 1:
        return np.array([0.5], dtype=np.float64)
    op = (np.arange(p.size) + 0.5) / float(p.size)
    np_ = (np.arange(m) + 0.5) / float(m)
    return np.interp(np_, op, p, left=float(p[0]), right=float(p[-1]))

def smooth_value_grad(h, n, tau):
    n = int(n)
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.size != n or n <= 0:
        return float('inf'), np.zeros(max(n, 1))
    h = np.clip(h, 0.0, 1.0)
    if tau <= 0 or not np.isfinite(tau):
        tau = 1e-12
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if not np.all(np.isfinite(c)):
        return float('inf'), np.zeros(n)
    qm = float(np.max(c))
    u = np.minimum((c - qm) / tau, 0.0)
    e = np.exp(u)
    ws = float(np.sum(e))
    if ws <= 0 or not np.isfinite(ws):
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        ws = 1.0
    else:
        w = e / ws
    val = qm + tau * np.log(ws)
    g = np.zeros(n, dtype=np.float64)
    n1 = n - 1
    for k in range(len(c)):
        wk = float(w[k])
        if wk < 1e-17:
            continue
        s = k - n1
        if s >= 0:
            m = n - s
            g[s:s + m] += dx * wk * (1.0 - h[:m])
            g[:m] -= dx * wk * h[s:s + m]
        else:
            m = n + s
            r = -s
            g[:m] += dx * wk * (1.0 - h[r:r + m])
            g[r:r + m] -= dx * wk * h[:m]
    if not np.all(np.isfinite(g)):
        return float('inf'), np.zeros(n)
    return float(val), g

                fails += 1
                if fails >= 10 or eta < 1e-8 / dx:
                    break
            else:
                fails = 0
                eta = min(1.1 * eta, emax)
    fin = project_capped(h, n)
    if fin is not None and is_feasible(fin, n):
        fs = exact_score(fin, n)
        if np.isfinite(fs) and fs <= bs + 1e-10:
            return fin, fs
    return h, bs

def sls_f(z):
    n = int(_SLS_N)
    if n <= 0 or len(z) != n:
        return 1e30
    h = np.clip(z, 0.0, 1.0)
    v, _ = smooth_value_grad(h, n, float(_SLS_TAU))
    return v if np.isfinite(v) else 1e30

def sls_df(z):
    n = int(_SLS_N)
    if n <= 0 or len(z) != n:
        return np.zeros(len(z))
    h = np.clip(z, 0.0, 1.0)
    _, g = smooth_value_grad(h, n, float(_SLS_TAU))
    if not np.all(np.isfinite(g)):
        g = np.zeros(n)
    return np.where((z > 0.0) & (z < 1.0), g, 0.0)

def sls_eq(z):
    return float(np.sum(z) - 0.5 * int(_SLS_N))

def sls_eq_jac(z):
    return np.ones(int(_SLS_N), dtype=np.float64)

def sls_polish(h0, n):
    global _SLS_N, _SLS_TAU
    h = project_capped(h0, n)
    if h is None:
        return h0, exact_score(h0, n)
    bs = exact_score(h, n)
    if not np.isfinite(bs):
        return h, bs
    for tau in (1e-4, 1e-6):
        _SLS_N = int(n)
        _SLS_TAU = float(tau)
        try:
            cons = [{'type': 'eq', 'fun': sls_eq, 'jac': sls_eq_jac}]
            res = minimize(sls_f, h.copy(), method='SLSQP', jac=sls_df,
                           bounds=[(0.0, 1.0)] * int(n), constraints=cons,
                           options={'maxiter': 120, 'ftol': 1e-14})
        except Exception:
            continue
        if res is None or getattr(res, 'x', None) is None:
            continue
        y = project_capped(res.x, n)
        if y is None or not is_feasible(y, n):
            continue
        ey = exact_score(y, n)
        if np.isfinite(ey) and ey < bs - 1e-12:
            h = y
            bs = ey
            continue
        Fh, _ = smooth_value_grad(h, n, tau)
        Fy, _ = smooth_value_grad(y, n, tau)
        if np.isfinite(Fy) and np.isfinite(Fh) and Fy < Fh - 1e-12:
            yh, ys = exact_polish(y, n, 40)
            if np.isfinite(ys) and ys < bs - 1e-12:
                h = yh
                bs = ys
    h, b = exact_polish(h, n, 60)
    return h, b

def build_starts(n, parent, rng, cap):
    pool = []
    cands = [np.full(n, 0.5)]
    if parent is not None:
        pr = project_capped(resample_h(parent, n), n)
        if pr is not None:
            cands.insert(0, pr)
            cands.append(pr[::-1].copy())
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    cands.append(blk)
    pos = (np.arange(n) + 0.5) / float(n)
    cands.append(np.clip(0.5 + 0.45 * np.cos(2.0 * np.pi * pos), 0.0, 1.0))
    cands.append(np.clip(0.5 + 0.45 * np.cos(2.0 * np.pi * pos + np.pi / 2), 0.0, 1.0))
    cands.append(rng.uniform(0.0, 1.0, n))
    for k, u in ((5, 2), (7, 3)):
        cands.append(np.where(np.arange(n) % k < u, 1.0, 0.0).astype(np.float64))
    if parent is not None:
        cands.append(resample_h(parent, n) + rng.normal(0.0, 0.08, n))
    for c in cands:
        if len(pool) >= cap:
            break
        h = project_capped(c, n)
        if h is None:
            continue
        dup = False
        for p in pool:
            if float(np.max(np.abs(h - p))) < 1e-7:
                dup = True
                break
        if not dup:
            pool.append(h)
    if not pool:
        pool.append(np.full(n, 0.5))
    return pool

def schedule_for(n):
    if n <= 64:
        return ((0.04, 0.01, 0.0025, 0.0006, 0.00015, 0.00004),
                (35, 55, 70, 70, 80, 100), 8, True, 150, 4)
    if n <= 128:
        return ((0.04, 0.01, 0.0025, 0.0006, 0.00015),
                (25, 40, 45, 45, 55), 5, False, 100, 1)
    return ((0.05, 0.012, 0.0035, 0.001),
            (15, 25, 30, 40), 4, False, 80, 0)

def search_one_n(n, parent, seed):
    taus, its, cap, do_sls, empi, rst = schedule_for(n)
    try:
        fb = project_capped(resample_h(parent, n), n) if parent is not None else None
    except Exception:
        fb = None
    if fb is None:
        fb = np.full(n, 0.5)
    fs = exact_score(fb, n)
    if not np.isfinite(fs):
        fs = 0.5
    bs = fs
    bh = fb.copy()
    try:
        rng = np.random.default_rng((int(seed) % 1000000) * 7919 + int(n))
    except Exception:
        rng = np.random.default_rng(42)
    pool = build_starts(n, parent, rng, cap)
    order = []
    for i in range(len(pool)):
        sc = exact_score(pool[i], n)
        order.append((sc, i))
        if np.isfinite(sc) and sc < bs - 1e-12:
            bs = sc
            bh = pool[i].copy()
    order.sort()
    for _, i in order:
        h, s = continuation_liu(pool[i], n, taus, its)
        if h is not None and np.isfinite(s) and s < bs - 1e-12:
            bh = h.copy()
            bs = s
    h, s = exact_polish(bh, n, empi)
    if np.isfinite(s) and s < bs - 1e-12:
        bh = h.copy()
        bs = s
    if do_sls:
        h, s = sls_polish(bh, n)
        if np.isfinite(s) and s < bs - 1e-12:
            bh = h.copy()
            bs = s
    bg = bs
    if rst > 0:
        if bs < fs - 1e-5:
            sig = (0.03, 0.06, 0.12, 0.25)
        else:
            sig = (0.06, 0.12, 0.25, 0.40)
        for sm in sig[:rst]:
            p = project_capped(bh + rng.normal(0.0, float(sm), n), n)
            if p is None:
                continue
            h, s = continuation_liu(p, n, taus, its)
            if h is not None and np.isfinite(s) and s < bs - 1e-12:
                bh = h.copy()
                bs = s
            h, s = exact_polish(bh, n, 80)
            if np.isfinite(s) and s < bs - 1e-12:
                bh = h.copy()
                bs = s
    if do_sls and bs < bg - 1e-12:
        h, s = sls_polish(bh, n)
        if np.isfinite(s) and s < bs - 1e-12:
            bh = h.copy()
            bs = s
    return bh, bs

def run(seed=42, **kwargs):
    parent = None
    n = 47
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if (p.size >= 1 and p.size <= 999 and np.all(np.isfinite(p))
                and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8)):
            parent = p
            n = int(p.size)
    except Exception:
        parent = None
    try:
        if 'n_points' in kwargs:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
    except Exception:
        pass
    if n < 1:
        n = 47
    try:
        seedv = int(seed)
    except Exception:
        seedv = 42
    try:
        h, s = search_one_n(n, parent, seedv)
    except Exception:
        h = None
        s = float('inf')
    if h is None or not np.isfinite(s):
        if parent is not None:
            h = project_capped(resample_h(parent, n), n)
        if h is None or not is_feasible(h, n):
            h = np.full(n, 0.5)
        s = exact_score(h, n)
    fin = project_capped(h, n)
    if fin is not None and is_feasible(fin, n):
        sf = exact_score(fin, n)
        if np.isfinite(sf) and sf <= s + 1e-10:
            h = fin
            s = sf
    if not is_feasible(h, n):
        h = np.full(n, 0.5)
        s = exact_score(h, n)
    return h, float(s), int(n)
