def project_capped(z, n):

    z = np.clip(z, -1e12, 1e12)

    r = s - np.sum(h)

            h[idx] += d; r -= d
        else:
            d = min(-r, h[idx])

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def compute_lag_J(h, n):
    dx = 2.0 / n
    L = np.correlate(h, 1.0 - h, mode='full') * dx
    nlag = 2 * n - 1
    G = np.zeros((nlag, n), dtype=np.float64)
    for d in range(-(n-1), n):
        row = d + n - 1
        a = max(0, -d)
        b = min(n, n - d)
        i = np.arange(a, b)
        j = i + d
        G[row, j] += dx * (1.0 - h[i])
        G[row, i] += -dx * h[j]

    return L, G

def project_capped(z, n):
    z = np.where(np.isfinite(z), z, 0.0).astype(np.float64)
    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx])
            h[idx] += d; r -= d
        else:
            d = min(-r, h[idx])
            h[idx] -= d; r += d
    if abs(float(r)) > 1e-10 * max(1, n) or not np.all(np.isfinite(h)):

    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(float(np.sum(h)) - n / 2.0) <= 1e-10 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def lag_jacobian(h, n):
    dx = 2.0 / n
    L = np.correlate(h, 1.0 - h, mode='full') * dx
    nlag = 2 * n - 1
    G = np.zeros((nlag, n), dtype=np.float64)
    for d in range(-(n - 1), n):
        row = d + n - 1
        a = max(0, -d)
        b = min(n, n - d)
        idx = np.arange(a, b)
        jidx = idx + d
        G[row, jidx] += dx * (1.0 - h[idx])
        G[row, idx] += -dx * h[jidx]
    if not np.all(np.isfinite(L)) or not np.all(np.isfinite(G)):
        return None
    return L, G

def solve_qp(x, L, G, n, solvers):
    dx = 2.0 / n
    rho = dx * math.cos(math.pi / (n + 1))
    if rho < 1e-12:
        rho = dx
    for si in range(len(solvers) - 1, -1, -1):
        solver = solvers[si]
        try:
            y = cp.Variable(n)
            u = cp.Variable()
            wy = y - x
            cons = [y >= 0, y <= 1, cp.sum(y) == n / 2.0]
            for k in range(2 * n - 1):
                cons.append(u >= L[k] + G[k, :] @ wy)
            prob = cp.Problem(cp.Minimize(u + rho * cp.sum_squares(wy)), cons)
            kw = {'verbose': False}
            prob.solve(solver=solver, **kw)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                yv = np.asarray(y.value, dtype=np.float64).ravel()
                if yv.shape == (n,) and np.all(np.isfinite(yv)):
                    return yv
            del solvers[si]
        except Exception:
            del solvers[si]
    return None
