def eval_envelope(a, b, c, ts):

        best = np.maximum(best, vals)
    return best

            B = b[i] - b[j]

                if lo - 1e-10 <= rt <= hi + 1e-10:

        return np.array([0.0])

        env = eval_envelope(a, b, c, chunk)

def lag_gradient(h, idx, n):
    dx = 2.0 / n
    g = np.zeros(n)
    if idx == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    else:
        r = (idx - 1) // 2 + 1
        m = n - r
        h0, h1 = h[:m], h[r:]
        if idx % 2 == 1:
            # positive lag: Q_{r,+} = dx * sum(h1*(1-h0))
            g[r:] += dx * (1.0 - h0)
            g[:m] -= dx * h1
        else:
            # negative lag: Q_{r,-} = dx * sum(h0*(1-h1))
            g[:m] += dx * (1.0 - h1)
            g[r:] -= dx * h0
    return g

    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(np.sum(h) - s) > 1e-9:
        return None
    return h

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and np.all(h >= -1e-12)
            and np.all(h <= 1.0 + 1e-12) and abs(np.sum(h) - n / 2.0) <= 1e-9)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    v = np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n))
    return float(v) if np.isfinite(v) else float('inf')

def line_coeffs(h, d, n):
    dx = 2.0 / n
    a = np.zeros(2 * n - 1)
    b = np.zeros(2 * n - 1)
    c = np.zeros(2 * n - 1)
    a[0] = dx * np.dot(h, 1.0 - h)
    b[0] = dx * np.dot(d, 1.0 - 2.0 * h)
    c[0] = -dx * np.dot(d, d)
    for r in range(1, n):
        m = n - r
        idx = 1 + 2 * (r - 1)
        a[idx] = dx * np.dot(h[r:], 1.0 - h[:m])
        b[idx] = dx * (np.dot(d[r:], 1.0 - h[:m]) - np.dot(h[r:], d[:m]))
        c[idx] = -dx * np.dot(d[r:], d[:m])
        a[idx+1] = dx * np.dot(h[:m], 1.0 - h[r:])
        b[idx+1] = dx * (np.dot(d[:m], 1.0 - h[r:]) - np.dot(h[:m], d[r:]))
        c[idx+1] = -dx * np.dot(d[:m], d[r:])
    return a, b, c

def feasible_interval(h, d, n):
    lo, hi = 0.0, 0.0
    for i in range(n):
        di = d[i]
        if di > 1e-15:
            lo = max(lo, -h[i] / di)
            hi = min(hi, (1.0 - h[i]) / di)
        elif di < -1e-15:
            lo = max(lo, (1.0 - h[i]) / di)
            hi = min(hi, -h[i] / di)
    if not np.isfinite(lo) or not np.isfinite(hi) or (hi - lo) <= 1e-12:
        return None
    return lo, hi

def eval_env(a, b, c, ts):
    k, nl = len(ts), len(a)
    best = np.full(k, -1e30)
    for j in range(nl):
        best = np.maximum(best, a[j] + b[j] * ts + c[j] * ts * ts)
    return best

def line_search_exact(a, b, c, lo, hi):
    nl = len(a)
    cands = {lo, hi, 0.0}
    for j in range(nl):
        if abs(c[j]) > 1e-15:
            tp = -b[j] / (2.0 * c[j])
            if lo - 1e-9 <= tp <= hi + 1e-9:
                cands.add(max(lo, min(hi, tp)))
    for i in range(nl):
        for j in range(i + 1, nl):
            A, B, C = c[i]-c[j], b[i]-b[j], a[i]-a[j]
            sc = max(1.0, abs(A), abs(B), abs(C))
            if abs(A) > 1e-12 * sc:
                disc = B*B - 4.0*A*C
                if disc < 0.0:
                    if disc > -1e-13*(1.0+B*B+abs(4.0*A*C)):
                        disc = 0.0
                    else:
                        continue
                sq = math.sqrt(disc)
                q = -0.5*(B + (sq if B >= 0 else -sq))
                if abs(q) > 1e-15:
                    for rt in (q/A, C/q):
                        if lo-1e-9 <= rt <= hi+1e-9:
                            cands.add(max(lo, min(hi, rt)))
                else:
                    rt = -B/(2.0*A)
                    if lo-1e-9 <= rt <= hi+1e-9:
                        cands.add(max(lo, min(hi, rt)))
            elif abs(B) > 1e-12 * sc:
                rt = -C/B
                if lo-1e-9 <= rt <= hi+1e-9:
                    cands.add(max(lo, min(hi, rt)))
    tarr = np.array(sorted(cands), dtype=np.float64)
    best_t, best_v = tarr[0], np.inf
    for s in range(0, len(tarr), 512):
        ch = tarr[s:s+512]
        env = eval_env(a, b, c, ch)
        for idx in range(len(ch)):
            if env[idx] < best_v - 1e-15:
                best_v = env[idx]
                best_t = ch[idx]
    return np.array([best_t])

def line_search_grid(a, b, c, lo, hi):
    ts = np.linspace(lo, hi, 17)
    if ((0.0 < ts[0]) or (0.0 > ts[-1])) and lo < 0.0 < hi:
        ts = np.insert(ts, len(ts)//2, 0.0)
    env = eval_env(a, b, c, ts)
    gm = int(np.argmin(env))
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    cands = [float(lo), float(hi), float(ts[gm])]
    for mi in range(len(ts)):
        if env[mi] < env[gm] + 1e-4 * max(1.0, env[gm]) and 0 < mi < len(ts)-1:
            tl, th = float(ts[mi-1]), float(ts[mi+1])
            if th - tl < 1e-14:
                continue
            ax, bx = tl+gr*(th-tl), th-gr*(th-tl)
            fa = float(np.max(a+b*ax+c*ax*ax))
            fb = float(np.max(a+b*bx+c*bx*bx))
            for _ in range(10):
                if fa < fb:
                    th, bx, fb = bx, ax, fa
                    ax = tl+(th-tl)*(1.0-gr)
                    fa = float(np.max(a+b*ax+c*ax*ax))
                else:
                    tl, ax, fa = ax, bx, fb
                    bx = tl+(th-tl)*gr
                    fb = float(np.max(a+b*bx+c*bx*bx))
            cands.append(0.5*(tl+th))
            if len(cands) >= 5:
                break
    tarr = np.array(sorted(set(cands)), dtype=np.float64)
    env2 = eval_env(a, b, c, tarr)
    return tarr[np.argsort(env2)[:3]]

def build_directions(h, n, rng, other_starts):
    dx = 2.0 / n
    lv = np.correlate(h, 1.0 - h, mode='full') * dx
    n_act = min(8, 2*n-1)
    act_ci = np.argsort(lv)[::-1][:n_act]
    grads = []
    for ci in act_ci:
        g = np.zeros(n)
        if ci == n - 1:
            g[:] = dx * (1.0 - 2.0 * h)
        elif ci < n - 1:
            r = n - 1 - ci
            m = n - r
            g[r:] += dx * (1.0 - h[:m])
            g[:m] -= dx * h[r:]
        else:
            r = ci - (n - 1)
            m = n - r
            g[:m] += dx * (1.0 - h[r:])
            g[r:] -= dx * h[:m]
        grads.append(g)
    dirs = []
    if n <= 64:
        pscores = []
        for i in range(n):
            for j in range(i + 1, n):
                if h[i] >= 1.0 - 1e-12 or h[j] <= 1e-12:
                    continue
                bp, bn = -1e30, -1e30
                for g in grads:
                    bp = max(bp, g[i] - g[j])
                    bn = max(bn, g[j] - g[i])
                pscores.append((min(bp, bn), i, j))
        pscores.sort(key=lambda x: -x[0])
        for _, i, j in pscores[:10]:
            d = np.zeros(n); d[i] = 1.0; d[j] = -1.0
            dirs.append(d)
    else:
        seen, att = set(), 0
        np2 = 5 if n <= 256 else 3
        while len(dirs) < np2 and att < 40:
            att += 1
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i == j or (min(i,j),max(i,j)) in seen:
                continue
            seen.add((min(i,j),max(i,j)))
            if h[i] < 1.0 - 1e-12 and h[j] > 1e-12:
                d = np.zeros(n); d[i] = 1.0; d[j] = -1.0
                dirs.append(d)
    ne = 4 if n <= 64 else (3 if n <= 256 else 2)
    for _ in range(ne):
        i, j = int(rng.integers(n)), int(rng.integers(n))
        while i == j:
            j = int(rng.integers(n))
        if h[i] < 1.0 - 1e-12 and h[j] > 1e-12:
            d = np.zeros(n); d[i] = 1.0; d[j] = -1.0
            dirs.append(d)
    if grads:
        v = -np.mean(grads, axis=0)
        v -= np.mean(v)
        mx = np.max(np.abs(v))
        if mx > 1e-14:
            v /= mx; v[-1] = -np.sum(v[:-1])
            dirs.append(v)
    for _ in range(2 if n <= 64 else 1):
        v = rng.normal(0, 1, n)
        v -= np.mean(v)
        mx = np.max(np.abs(v))
        if mx > 1e-14:
            v /= mx; v[-1] = -np.sum(v[:-1])
            dirs.append(v)
    for os_ in other_starts:
        if os_ is not None and np.max(np.abs(os_ - h)) > 1e-6:
            v = os_ - h
            v -= np.mean(v)
            mx = np.max(np.abs(v))
            if mx > 1e-14:
                v /= mx; v[-1] = -np.sum(v[:-1])
                dirs.append(v)
                break
    return dirs

def local_search(h0, n, rng, gb_h, gb_s, other_starts):
    h = h0.copy()
    hs = exact_score(h, n)
    kicks_used = 0
    n_sweeps = 5 if n <= 64 else (4 if n <= 256 else 3)
    for sweep in range(n_sweeps):
        dirs = build_directions(h, n, rng, other_starts)
        best_cand, best_sc = None, hs
        for d in dirs:
            if not np.all(np.isfinite(d)) or np.max(np.abs(d)) < 1e-14:
                continue
            fi = feasible_interval(h, d, n)
            if fi is None:
                continue
            lo, hi = fi
            a, b, c = line_coeffs(h, d, n)
            if n <= 64:
                ts = line_search_exact(a, b, c, lo, hi)
            else:
                ts = line_search_grid(a, b, c, lo, hi)
            for t in ts:
                if abs(t) < 1e-10:
                    continue
                cand = project_capped(h + t * d, n)
                if cand is None:
                    continue
                sc = exact_score(cand, n)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_cand = cand
                if sc < gb_s - 1e-12:
                    gb_h[:] = cand
                    gb_s = sc
        if best_cand is not None:
            h[:] = best_cand
            hs = best_sc
            if hs < gb_s - 1e-12:
                gb_h[:] = h
                gb_s = hs
        else:
            if kicks_used < 2:
                sig = 0.08 if kicks_used == 0 else 0.20
                kicks_used += 1
                z = h + sig * rng.normal(0, 1, n)
                proj = project_capped(z, n)
                if proj is not None:
                    h[:] = proj
                    hs = exact_score(h, n)
                    if hs < gb_s - 1e-12:
                        gb_h[:] = h
                        gb_s = hs
            else:
                break
    if hs < gb_s - 1e-12:
        gb_h[:] = h
        gb_s = hs
    return gb_h, gb_s

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0+1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None:
        n = 47
        try:
            vk = int(kwargs.get('n_points', 47))
            if 1 <= vk <= 999:
                n = vk
        except Exception:
            pass
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        return np.array([0.5]), 0.5, 1
    if parent is not None:
        fb = project_capped(parent, n)
    else:
        fb = project_capped(np.full(n, 0.5), n)
    if fb is None:
        fb = np.full(n, 0.5)
    fb_s = exact_score(fb, n)
    pool = [fb.copy()]
    def add_pool(z):
        p2 = project_capped(z, n)
        if p2 is not None:
            for s in pool:
                if np.max(np.abs(p2 - s)) < 1e-9:
                    return
            pool.append(p2)
            if len(pool) >= 16:
                return
    add_pool(fb[::-1].copy())
    add_pool(1.0 - fb)
    add_pool((1.0 - fb[::-1]).copy())
    add_pool(np.roll(fb, n//3))
    add_pool(np.roll(fb, n//2))
    blk = np.zeros(n); blk[:n//2] = 1.0
    if n % 2:
        blk[n//2] = 0.5
    add_pool(blk)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    add_pool(alt)
    add_pool(fb + 0.06 * rng.normal(0, 1, n))
    add_pool(fb + 0.18 * rng.normal(0, 1, n))
    for k in (1, 2, 3):
        for ph in (0.0, math.pi / 2):
            add_pool(0.5 + 0.4 * np.cos(2.0 * math.pi * k * np.arange(n) / n + ph))
    pool_s = np.array([exact_score(sv, n) for sv in pool])
    starts = [pool[0].copy()]
    for idx in np.argsort(pool_s):
        idx = int(idx)
        if idx == 0:
            continue
        if all(np.max(np.abs(pool[idx] - st)) > 0.05 for st in starts):
            starts.append(pool[idx].copy())
        if len(starts) >= (4 if n <= 64 else (3 if n <= 256 else 2)):
            break
    while len(starts) < (4 if n <= 64 else (3 if n <= 256 else 2)):
        for idx in np.argsort(pool_s):
            idx = int(idx)
            if all(np.max(np.abs(pool[idx] - st)) > 1e-9 for st in starts):
                starts.append(pool[idx].copy())
                break
        else:
            break
    gb_h, gb_s = fb.copy(), fb_s
    for si, st in enumerate(starts):
        others = [s2 for s2 in starts if s2 is not st]
        gb_h, gb_s = local_search(st, n, rng, gb_h, gb_s, others[:2])
    final_h = project_capped(gb_h.copy(), n)
    if final_h is None or not is_feasible(final_h, n):
        final_h = fb.copy()
    fs = exact_score(final_h, n)
    if not np.isfinite(fs) or fs > fb_s + 1e-12:
        final_h = fb.copy()
        fs = fb_s
    final_h = np.clip(final_h, 0.0, 1.0)
    return final_h.astype(np.float64), float(fs), int(n)
