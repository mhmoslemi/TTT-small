        return True
    return math.log(max(float(rng.uniform()), 1e-300)) < lr

def add_start(starts, v, n):
    h2 = project_capped(v, n)
    if h2 is None:
        return
    for ex in starts:
        if float(np.max(np.abs(h2 - ex))) < 1e-8:
            return
    starts.append(h2)

def local_polish(h, n, sweeps):
    best = h.copy().astype(np.float64)
    bs = exact_score(best, n)
    for _ in range(sweeps):
        imp = False
        for i in range(n):
            for j in range(i + 1, n):
                L = max(-best[i], best[j] - 1.0)
                U = min(1.0 - best[i], best[j])
                if U - L < 1e-13:
                    continue
                cands = [0.0, L + 1e-12, U - 1e-12]
                for st in (0.001, 0.005, 0.02, 0.08, 0.2):
                    for sg in (1, -1):
                        dd = sg * st
                        if L <= dd <= U:
                            cands.append(dd)
                bd, bsc = 0.0, bs
                for dd in cands:
                    dd = max(L, min(U, dd))
                    trial = best.copy()
                    trial[i] += dd
                    trial[j] -= dd
                    sc = exact_score(trial, n)
                    if sc < bsc - 1e-13:
                        bd, bsc = dd, sc
                if bd != 0.0:
                    best[i] += bd
                    best[j] -= bd
                    bs = bsc
                    imp = True
        if not imp:
            break
    return best, bs

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= len(iv) <= 999 and np.all(np.isfinite(iv)):
            n = len(iv)
            parent = iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
        n = max(2, min(999, n))
    try:
        sd = int(seed) % (2**32)
    except Exception:
        sd = 42
    rng = np.random.default_rng(sd)

    if parent is not None:
        h0 = project_capped(parent, n)
        fallback = h0 if h0 is not None else np.full(n, 0.5)
    else:
        fallback = np.full(n, 0.5)
    fb_score = exact_score(fallback, n)
    if not np.isfinite(fb_score):
        fallback = np.full(n, 0.5)
        fb_score = exact_score(fallback, n)

    starts = []
    add_start(starts, fallback, n)
    if parent is not None:
        add_start(starts, fallback + rng.normal(0, 0.015, n), n)
        add_start(starts, fallback + rng.normal(0, 0.06, n), n)
        add_start(starts, fallback[::-1], n)
        add_start(starts, 1.0 - fallback, n)
        add_start(starts, 1.0 - fallback[::-1], n)
        add_start(starts, rng.uniform(0, 1, n), n)
    else:
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        add_start(starts, blk, n)
        add_start(starts, np.where(np.arange(n) % 2 == 0, 1.0, 0.0), n)
        tt = np.arange(n)
        add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * tt / n), n)
        add_start(starts, 0.5 + 0.45 * np.cos(4.0 * np.pi * tt / n), n)
        add_start(starts, rng.uniform(0, 1, n), n)

    try:
        B = int(kwargs.get('eval_budget', 0))
    except Exception:
        B = 0
    B_cap = min(16000, max(120, 30000000 // (n * n)))
    B = B_cap if B <= 0 else max(120, min(B, B_cap, 16000))

    temps = [0.0005, 0.002, 0.008, 0.032, 0.128, 0.512]
    scales = [0.002, 0.006, 0.018, 0.054, 0.162, 0.486]
    n_rep = 6

    rep_h, rep_s = [], []
    for i in range(n_rep):
        st = starts[i % len(starts)].copy() if starts else fallback.copy()
        st = project_capped(0.99 * st + 0.005, n)
        if st is None:
            st = fallback.copy()
        sc = exact_score(st, n)
        if not np.isfinite(sc):
            st, sc = fallback.copy(), fb_score
        rep_h.append(st)
        rep_s.append(sc)

    best, best_score = fallback.copy(), fb_score
    for i in range(n_rep):
        if rep_s[i] < best_score - 1e-12:
            best, best_score = rep_h[i].copy(), rep_s[i]

    eval_ct = 0
    stag = 0
    restarts = 0
    stag_lim = max(30, min(200, B // (n_rep * 20)))
    sweep = 0
    max_sweeps = B + n_rep + 1

    while eval_ct < B and sweep < max_sweeps:
        sweep += 1
        prev_bs = best_score
        for i in range(n_rep):
            if eval_ct >= B:
                break
            r = rng.random()
            T = temps[i]
            if r < 0.35:
                cand = hit_and_run(rep_h[i], n, rng)
            elif r < 0.65:
                cand = pair_move(rep_h[i], n, rng, 0.0, True)
            else:
                cand = pair_move(rep_h[i], n, rng, scales[i], False)
            eval_ct += 1
            if cand is None or not valid_cand(cand, n):
                continue
            sc = exact_score(cand, n)
            if not np.isfinite(sc):
                continue
            if meta_accept(sc - rep_s[i], T, rng):
                rep_h[i] = cand
                rep_s[i] = sc
            if sc < best_score - 1e-12:
                best, best_score = cand.copy(), sc
        pairs = [(0, 1), (2, 3), (4, 5)] if sweep % 2 == 0 else [(1, 2), (3, 4)]
        for (a, b) in pairs:
            if b < n_rep and exchange_accept(temps[a], temps[b], rep_s[a], rep_s[b], rng):
                rep_h[a], rep_h[b] = rep_h[b], rep_h[a]
                rep_s[a], rep_s[b] = rep_s[b], rep_s[a]
        if best_score < prev_bs - 1e-12:
            stag = 0
        else:
            stag += 1
        if stag >= stag_lim and restarts < 2:
            restarts += 1
            if restarts == 1:
                ns = 0.5 * best + 0.5 * fallback
            else:
                ns = np.clip(best + rng.beta(0.7, 0.7, n) - 0.5, 0, 1)
            ns = project_capped(0.99 * ns + 0.005, n)
            if ns is not None:
                rep_h[n_rep - 1] = ns
                rep_s[n_rep - 1] = exact_score(ns, n)
                eval_ct += 1
                if rep_s[n_rep - 1] < best_score - 1e-12:
                    best, best_score = ns.copy(), rep_s[n_rep - 1]
            stag = 0

    fin = project_capped(best, n)
    if fin is not None:
        fsc = exact_score(fin, n)
        if np.isfinite(fsc) and fsc < best_score + 1e-12:
            best, best_score = fin, fsc
    if not np.isfinite(best_score) or best_score > fb_score:
        best, best_score = fallback.copy(), fb_score
    if not np.isfinite(best_score):
        best = np.full(n, 0.5)
        best_score = exact_score(best, n)
    return best.astype(np.float64), float(best_score), int(n)
