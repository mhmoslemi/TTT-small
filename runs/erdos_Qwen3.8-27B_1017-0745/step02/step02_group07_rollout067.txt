import numpy as np
import math
from scipy.optimize import minimize


def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = 0.5 * n
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(n + 4):
        r = s - float(np.sum(h))
        if abs(r) < 1e-13:
            break
        if r > 0:
            i = int(np.argmax(1.0 - h))
            h[i] = min(1.0, h[i] + r)
        else:
            i = int(np.argmax(h))
            h[i] = max(0.0, h[i] + r)
    if not np.all(np.isfinite(h)) or abs(s - float(np.sum(h))) > 1e-8:
        return None
    return h


def is_feasible(h, n):
    try:
        return (h.shape == (n,) and bool(np.all(np.isfinite(h)))
                and float(np.min(h)) >= -1e-12 and float(np.max(h)) <= 1.0 + 1e-12
                and abs(float(np.sum(h)) - 0.5 * n) <= 1e-8 * max(1, n))
    except Exception:
        return False


def exact_score(h, n):
    try:
        if not is_feasible(h, n):
            return float('inf')
        c = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
        return c if np.isfinite(c) else float('inf')
    except Exception:
        return float('inf')


def lag_values(h, n):
    dx = 2.0 / n
    Q = np.empty(2 * n - 1)
    Q[0] = dx * float(np.dot(h, 1.0 - h))
    idx = 1
    for r in range(1, n):
        m = n - r
        a = h[:m]
        b = h[r:]
        Q[idx] = dx * float(np.dot(a, 1.0 - b))
        idx += 1
        Q[idx] = dx * float(np.dot(b, 1.0 - a))
        idx += 1
    return Q


def lag_values_jac(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    Q = np.empty(L)
    G = np.zeros((L, n))
    Q[0] = dx * float(np.dot(h, 1.0 - h))
    G[0] = dx * (1.0 - 2.0 * h)
    idx = 1
    for r in range(1, n):
        m = n - r
        a = h[:m]
        b = h[r:]
        gp = G[idx]
        gp[:m] = 1.0 - b
        gp[r:] = -a
        Q[idx] = dx * float(np.dot(a, 1.0 - b))
        idx += 1
        gm = G[idx]
        gm[r:] = 1.0 - a
        gm[:m] = -b
        Q[idx] = dx * float(np.dot(b, 1.0 - a))
        idx += 1
    return Q, G


def smooth_fg(h, n, tau):
    dx = 2.0 / n
    Q = lag_values(h, n)
    qmax = float(np.max(Q))
    e = np.exp(np.clip((Q - qmax) / tau, -745.0, 0.0))
    ss = float(np.sum(e))
    if ss <= 0.0 or not np.isfinite(ss):
        return float('inf'), np.zeros(n)
    F = qmax + tau * math.log(ss)
    p = e / ss
    g = np.zeros(n)
    g += dx * p[0] * (1.0 - 2.0 * h)
    idx = 1
    for r in range(1, n):
        m = n - r
        a = h[:m]
        b = h[r:]
        pp = float(p[idx])
        pm = float(p[idx + 1])
        if pp > 1e-16:
            g[:m] += pp * dx * (1.0 - b)
            g[r:] -= pp * dx * a
        if pm > 1e-16:
            g[r:] += pm * dx * (1.0 - a)
            g[:m] -= pm * dx * b
        idx += 2
    if not np.all(np.isfinite(g)):
        return float('inf'), np.zeros(n)
    return F, g


def ep_obj(x):
    return float(x[-1])


def ep_obj_jac(x):
    g = np.ones(len(x))
    g[:-1] = 0.0
    return g


def ep_eq(x, n):
    return float(np.sum(x[:-1]) - 0.5 * n)


def ep_eq_jac(x, n):
    g = np.ones(len(x))
    g[-1] = 0.0
    return g


def ep_ineq(x, n, idx):
    if x.size != n + 1:
        return np.full(len(idx), 1e6)
    try:
        Q, _ = lag_values_jac(x[:-1], n)
        v = float(x[-1]) - Q[idx]
    except Exception:
        v = np.full(len(idx), 1e6)
    if not np.all(np.isfinite(v)):
        v = np.full(len(idx), 1e6)
    return v


def ep_ineq_jac(x, n, idx):
    J = np.zeros((len(idx), len(x)))
    J[:, -1] = 1.0
    if x.size != n + 1:
        return J
    try:
        _, G = lag_values_jac(x[:-1], n)
        if np.all(np.isfinite(G)):
            J[:, :-1] = -G[idx]
    except Exception:
        pass
    return J


def choose_active(h, n, K):
    Q = lag_values(h, n)
    qmax = float(np.max(Q))
    top = np.argsort(-Q)[:K]
    near = np.flatnonzero(Q >= qmax - 1e-6 * max(1.0, qmax))
    if near.size > K:
        near = near[np.argsort(-Q[near])[:K]]
    return np.unique(np.concatenate((top, near)))


def run_slq_start(h0, n, maxit, Ks, reps, best, best_s):
    h = project_capped(h0, n)
    if h is None:
        return best, best_s
    s = exact_score(h, n)
    if s < best_s:
        best = h.copy()
        best_s = s
    L = 2 * n - 1
    bounds = [(0.0, 1.0)] * n + [(0.0, float(max(1.0, s + 0.5)))]
    for K in Ks:
        Kc = min(int(K), L)
        for rep in range(reps):
            idx = choose_active(h, n, Kc)
            z0 = np.empty(n + 1)
            z0[:-1] = h
            z0[-1] = s + 1e-4
            cons = [
                {'type': 'ineq', 'fun': ep_ineq, 'jac': ep_ineq_jac, 'args': (n, idx)},
                {'type': 'eq', 'fun': ep_eq, 'jac': ep_eq_jac, 'args': (n,)}
            ]
            try:
                res = minimize(ep_obj, z0, jac=ep_obj_jac, bounds=bounds,
                               constraints=cons, method='SLSQP',
                               options={'maxiter': maxit, 'ftol': 1e-10, 'disp': False})
            except Exception:
                res = None
            imp = False
            if res is not None and hasattr(res, 'x') and len(res.x) == n + 1 and np.all(np.isfinite(res.x)):
                for mix in (1.0, 0.75, 0.5, 0.25):
                    if mix == 1.0:
                        hr = res.x[:-1]

                        hr = (1.0 - mix) * h + mix * res.x[:-1]

                        h = h2
                        s = sc
                        if s < best_s:
                            best = h.copy()
                            best_s = s
                        imp = True
                        break
            if not imp:
                break
    return best, best_s


def eq_h(x, n):
    return float(np.sum(x) - 0.5 * n)


def eq_h_jac(x, n):
    return np.ones(len(x), dtype=np.float64)


def smooth_obj_h(x, n, tau):

    except Exception:

    except Exception:
        return np.zeros(len(x), dtype=np.float64)

def make_binary(rng, n):
    p = rng.permutation(n)
    v = np.zeros(n)
    v[p[:n // 2]] = 1.0
    if n % 2:

    return v
