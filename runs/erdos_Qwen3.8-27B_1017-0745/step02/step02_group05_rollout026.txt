            break
        if r > 0:
            d = min(r, 1.0 - h[i]); h[i] += d; r -= d
        else:
            d = min(-r, h[i]); h[i] -= d; r += d
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    return h

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def smooth_obj(h, n, tau):
    if not np.all(np.isfinite(h)):
        return 10.0
    Q = (2.0 / n) * np.correlate(h, 1.0 - h, mode='full')
    M = np.max(Q)
    e = np.exp(np.clip((Q - M) / tau, -745, 0))
    return float(M + tau * np.log(np.sum(e)))

def smooth_grad(h, n, tau):
    Q = (2.0 / n) * np.correlate(h, 1.0 - h, mode='full')
    M = np.max(Q)
    e = np.exp(np.clip((Q - M) / tau, -745, 0))
    w = e / np.sum(e)
    dx = 2.0 / n
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        wd = w[idx]
        if wd < 1e-16:
            continue
        d = idx - (n - 1)
        a = max(d, 0)
        b = max(-d, 0)
        m = n - abs(d)
        g[a:a + m] += wd * dx * (1.0 - h[b:b + m])
        g[b:b + m] -= wd * dx * h[a:a + m]
    return g

def build_starts(base, n, rng, count):
    pool = [base.copy()]
    cands = [base[::-1].copy(), 1.0 - base,
             np.roll(base, max(1, n // 4)), np.roll(base, -max(1, n // 4)),
             np.full(n, 0.5)]
    blk = np.zeros(n); blk[:n // 2] = 1.0
    if n % 2 == 1:
        blk[n // 2] = 0.5
    cands.append(blk)
    cands.append(np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64))
    for sd in (0.05, 0.12):
        cands.append(base + rng.normal(0, sd, n))
    ph = rng.uniform(0, 6.283)
    cands.append(base + 0.08 * np.sin(2.0 * np.pi * np.arange(n) / n + ph))
    cands.append(rng.uniform(0, 1, n))
    for c in cands:
        p = project_capped(c, n)
        if p is None:
            continue
        if any(np.max(np.abs(p - s)) < 1e-8 for s in pool):
            continue
        pool.append(p)
        if len(pool) >= count:
            break
    return pool[:count]

def optimize_start(x0, n, taus, maxiter):
    x = project_capped(x0, n)
    if x is None:
        return None, float('inf')
    score = exact_score(x, n)
    s = n / 2.0
    bnds = Bounds(np.zeros(n), np.ones(n))
    cons = [LinearConstraint(np.ones((1, n)), s, s)]
    for tau in taus:
        try:
            res = minimize(smooth_obj, x, args=(n, tau), jac=smooth_grad,
                           method='SLSQP', bounds=bnds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-10})
        except Exception:
            res = None
        if res is None:
            continue
        y = res.x
        if y is None or y.shape != (n,) or not np.all(np.isfinite(y)):
            continue
        yp = project_capped(y, n)
        if yp is None:
            continue
        best_z, best_phi = x, smooth_obj(x, n, tau)
        for k in range(5):
            alpha = 1.0 / (2.0 ** k)
            z = project_capped(x + alpha * (yp - x), n)
            if z is None:
                continue
            sz = exact_score(z, n)
            if sz < score - 1e-12:
                score = sz
            zp = smooth_obj(z, n, tau)
            if zp < best_phi - 1e-12:
                best_phi, best_z = zp, z
        x = best_z
    return x, exact_score(x, n)

def final_polish(x, n, rng, n_trials):
    x = project_capped(x, n)
    if x is None:
        return x, float('inf')
    score = exact_score(x, n)
    sigma = 0.03
    for _ in range(n_trials):
        sigma *= 0.95
        if sigma < 1e-7:
            break
        z = x + rng.normal(0, sigma, n)
        zp = project_capped(z, n)
        if zp is None:
            continue
        sz = exact_score(zp, n)
        if sz < score - 1e-13:
            x, score = zp, sz
    return x, score

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p.copy()
    except Exception:
        pass
    if n is None:
        try:
            v = int(kwargs.get('n_points', 47))
            n = v if 1 <= v <= 999 else 47
        except Exception:
            n = 47
    if n == 1:
        h1 = np.array([0.5])
        return h1, 0.25, 1
    fb = project_capped(parent, n) if parent is not None else None
    if fb is None or not np.all(np.isfinite(fb)):
        fb = np.full(n, 0.5)
    gb, gs = fb.copy(), exact_score(fb, n)
    fb_score = gs
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 96:
        n_starts, taus, maxiter = 6, [1e-2, 2e-3, 4e-4, 8e-5, 1e-5], 80
    elif n <= 256:
        n_starts, taus, maxiter = 4, [5e-3, 5e-4, 5e-5, 1e-5], 60
    else:
        n_starts, taus, maxiter = 2, [2e-3, 1e-4, 1e-5], 35
    starts = build_starts(fb, n, rng, n_starts)
    for st in starts:
        ss = exact_score(st, n)
        if ss < gs - 1e-12:
            gs, gb = ss, st.copy()
    for st in starts:
        try:
            xo, so = optimize_start(st, n, taus, maxiter)
            if xo is not None and so < gs - 1e-12:
                gs, gb = so, xo.copy()
        except Exception:
            pass
    try:
        xp, sp = optimize_start(gb, n, [5e-6, 1e-6], max(40, maxiter))
        if xp is not None and sp < gs - 1e-12:
            gs, gb = sp, xp.copy()
    except Exception:
        pass
    try:
        xp, sp = final_polish(gb, n, rng, 150)
        if xp is not None and sp < gs - 1e-12:
            gs, gb = sp, xp.copy()
    except Exception:
        pass
    final = project_capped(gb, n)
    if final is None or not np.all(np.isfinite(final)):
        final = fb.copy()
    fs = exact_score(final, n)
    if fs > fb_score + 1e-12:
        final, fs = fb.copy(), fb_score
    return final.astype(np.float64), float(fs), int(n)
