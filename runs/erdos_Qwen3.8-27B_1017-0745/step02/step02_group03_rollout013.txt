        idx = np.arange(lo, hi)

        grad[idx + d] -= pd * h[idx]

import numpy as np
import math

def project_capped(z, n):
    s = n / 2.0
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if abs(hi - lo) < 1e-15:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(n + 4):
        if abs(r) < 1e-13:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            amt = min(r, 1.0 - h[idx])
            if amt <= 0:
                break
            h[idx] += amt
            r -= amt
        else:
            idx = int(np.argmax(h))
            amt = max(r, -h[idx])
            if amt >= 0:
                break
            h[idx] += amt
            r -= amt
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(np.sum(h) - s) > 1e-9 * max(1, n):
        return None
    return h

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return float('inf')
    if abs(np.sum(h) - n / 2.0) > 1e-8 * max(1, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_values(h, n):
    L = 2 * n - 1
    v = np.zeros(L)
    for d in range(-(n - 1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        idx = np.arange(lo, hi)
        v[d + n - 1] = float(np.dot(h[idx], 1.0 - h[idx + d]))
    return v

def smooth_phi_grad(h, n, theta):
    F = lag_values(h, n)
    fm = float(np.max(F))
    X = np.clip((F - fm) / theta, -700.0, 0.0)
    E = np.exp(X)
    SE = float(np.sum(E))
    if SE < 1e-300 or not np.isfinite(SE):
        return fm, None
    p = E / SE
    phi = fm + theta * math.log(SE)
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        pd = p[d + n - 1]
        if pd < 1e-16:
            continue
        lo = max(0, -d)
        hi = min(n, n - d)
        idx = np.arange(lo, hi)
        grad[lo:hi] += pd * (1.0 - h[idx + d])
        grad[idx + d] -= pd * h[idx]
    return phi, grad

def continuation_opt(h0, n, iters, taus):
    h = h0.copy()
    best_h = h.copy()
    best_s = exact_score(h, n)
    for tau in taus:
        theta = (n / 2.0) * tau
        alpha = 0.2
        for _ in range(iters):
            phi_c, grad = smooth_phi_grad(h, n, theta)
            if grad is None or not np.all(np.isfinite(grad)):
                break
            accepted = False
            for bt in range(24):
                y = project_capped(h - alpha * grad, n)
                if y is None:
                    alpha *= 0.5
                    continue
                d = y - h
                if float(np.max(np.abs(d))) < 1e-10:
                    accepted = True
                    break
                gd = float(np.dot(grad, d))
                if gd >= -1e-14:
                    alpha *= 0.5
                    continue
                phi_y, _ = smooth_phi_grad(y, n, theta)
                if phi_y > phi_c + 1e-4 * gd + 1e-12 * max(1.0, abs(phi_c)):
                    alpha *= 0.5
                    continue
                h = y
                accepted = True
                break
            if not accepted:
                break

            sc = exact_score(h, n)
            if sc < best_s:
                best_s = sc
                best_h = h.copy()
    return best_h, best_s

def _add_start(starts, v, n, tol=1e-8):
