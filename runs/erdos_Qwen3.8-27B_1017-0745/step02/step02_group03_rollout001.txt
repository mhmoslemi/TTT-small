                d = y - h
                if float(np.max(np.abs(d))) < 1e-10:
                    accepted = True
                    break
                gd = float(np.dot(g, d))
                if gd >= -1e-14:
                    alpha *= 0.5
                    continue
                phi_y, _ = lse_phi_grad(y, n, theta)
                if phi_y is None:
                    alpha *= 0.5
                    continue
                if phi_y <= phi + 1e-4 * gd + 1e-12 * max(1.0, abs(phi)):
                    h = y
                    alpha = min(0.5, 1.15 * alpha)
                    sc = exact_score(h, n)
                    if np.isfinite(sc) and sc < best_sc - 1e-13:
                        best_sc = sc
                        best_h = h.copy()
                    accepted = True
                    break
                alpha *= 0.5
            if not accepted:
                break
            if alpha < 1e-13:
                break
    sc = exact_score(h, n)
    if np.isfinite(sc) and sc < best_sc - 1e-13:
        best_sc = sc
        best_h = h.copy()
    return best_h, best_sc

def _add_start(starts, v, n):
    h2 = project_capped(v, n)
    if h2 is None:
        return
    for ex in starts:
        if float(np.max(np.abs(h2 - ex))) < 1e-8:
            return
    starts.append(h2)

def _pairwise_polish(x, n, sweeps):
    best = x.copy()
    bs = exact_score(best, n)
    for _ in range(sweeps):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                lo = max(-best[i], best[j] - 1.0)
                hi = min(1.0 - best[i], best[j])
                if lo >= hi - 1e-14:
                    continue
                for frac in (0.1, 0.3, 0.5, 0.7, 0.9):
                    dd = lo + frac * (hi - lo)
                    trial = best.copy()
                    trial[i] += dd
                    trial[j] -= dd
                    sc = exact_score(trial, n)
                    if sc < bs - 1e-13:
                        best = trial
                        bs = sc
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best, bs

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= len(iv) <= 999 and np.all(np.isfinite(iv)):
            n = len(iv)
            parent = iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
        n = max(1, min(999, n))
    try:
        sd = int(seed)
        if sd < 0:
            sd = 42
    except Exception:
        sd = 42
    rng = np.random.default_rng(sd)
    uniform = np.full(n, 0.5)
    if parent is not None:
        h0 = project_capped(parent, n)
        fallback = h0 if h0 is not None else uniform.copy()
    else:
        fallback = uniform.copy()
    fb_score = exact_score(fallback, n)
    if n <= 128:
        n_s_def, iters_def = 6, 60
    elif n <= 512:
        n_s_def, iters_def = 4, 25
    else:
        n_s_def, iters_def = 2, 10
    try:
        n_starts = max(1, min(int(kwargs.get('starts', n_s_def)), 8))
    except Exception:
        n_starts = n_s_def
    try:
        ips = max(1, min(int(kwargs.get('max_iter', iters_def)), 120))
    except Exception:
        ips = iters_def
    starts = []
    _add_start(starts, fallback, n)
    if parent is not None:
        _add_start(starts, fallback[::-1], n)
        _add_start(starts, 1.0 - fallback, n)
        _add_start(starts, 1.0 - fallback[::-1], n)
        _add_start(starts, np.roll(fallback, n // 3), n)
        _add_start(starts, fallback + rng.normal(0, 0.03, n), n)
        _add_start(starts, fallback + rng.normal(0, 0.10, n), n)
        _add_start(starts, fallback + rng.normal(0, 0.06, n), n)
    else:
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        _add_start(starts, blk, n)
        tt = np.arange(n)
        _add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * tt / n), n)
        _add_start(starts, 0.5 + 0.45 * np.cos(4.0 * np.pi * tt / n), n)
        _add_start(starts, rng.uniform(0, 1, n), n)
    starts = starts[:n_starts]
    best, best_score = fallback.copy(), fb_score
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_score - 1e-12:
            best, best_score = st.copy(), ss
    for st in starts:
        try:
            bh, bs = continuation_opt(st, n, ips)
            if np.isfinite(bs) and bs < best_score - 1e-12:
                best, best_score = bh, bs
        except Exception:
            pass
    hop_sds = [0.02, 0.05, 0.08] if n <= 128 else [0.04]
    for sd_hop in hop_sds:
        for _ in range(2):
            try:
                pert = best + rng.normal(0, sd_hop, n)
                hp = project_capped(pert, n)
                if hp is None:
                    continue
                bh, bs = continuation_opt(hp, n, ips)
                if np.isfinite(bs) and bs < best_score - 1e-12:
                    best, best_score = bh, bs
            except Exception:
                pass
    polished, polys = _pairwise_polish(best, n, 3)
    fp = project_capped(polished, n)
    if fp is not None:
        fs = exact_score(fp, n)
        if np.isfinite(fs) and fs < best_score - 1e-13:
            best, best_score = fp, fs
    final = project_capped(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final, fs = fallback.copy(), fb_score
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final, float(fs), int(n)
