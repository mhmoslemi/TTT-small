from scipy.optimize import minimize

def project_capped(z, n):
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    # Fix residual
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - s) > 1e-9:
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-9:
        return False
    return True

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c * (2.0 / n)))

def lag_q_and_jac(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.zeros(L)
    J = np.zeros((L, n))
    for d in range(-(n-1), n):

        if d >= 0:
            m = n - d
            a, b = d, 0  # h[a:a+m] * (1-h[b:b+m])

_ctx = {}

def project_capped(z, n):
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - s) > 1e-9:
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-9:
        return False
    return True

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c * (2.0 / n)))

def lag_q_and_jac(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.zeros(L)
    J = np.zeros((L, n))
    for d in range(-(n - 1), n):
        ki = d + n - 1
        if d >= 0:
            m = n - d
            a, b = d, 0
        else:
            m = n + d
            a, b = 0, -d
        xp = h[a:a + m]
        xq = h[b:b + m]
        q[ki] = dx * np.dot(xp, 1.0 - xq)
        J[ki, a:a + m] += dx * (1.0 - xq)
        J[ki, b:b + m] -= dx * xp
    return q, J

def _obj_jac(z):

def _ineq_jac(z):

    n = _ctx['n']
    L = 2 * n - 1
    out = np.empty((L, n + 1))
    out[:, :n] = -J
    out[:, n] = 1.0
    return out

def solve_slsqp(x, n, maxiter):

    _ctx['s'] = n / 2.0
    q0, _ = lag_q_and_jac(x, n)
    t0 = np.max(q0) + 1e-6
    z0 = np.empty(n + 1)
    z0[:n] = x
    z0[n] = t0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

        {'type': 'eq', 'fun': _eq, 'jac': _eq_jac},
    ]
    try:
        res = minimize(_obj, z0, jac=_obj_jac, method='SLSQP',
                       bounds=bounds, constraints=constr,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
    except Exception:
        return None
    y = res.x[:n] if res.x is not None else None
    if y is None or y.shape != (n,) or not np.all(np.isfinite(y)):
        return None
    proj = project_capped(y, n)
    if proj is None:
        return None
    return proj
