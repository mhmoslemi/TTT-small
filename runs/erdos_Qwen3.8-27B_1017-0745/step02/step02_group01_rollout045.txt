   x=np.asarray(x,dtype=np.float64)
   if x.shape!=(n,) or not np.all(np.isfinite(x)): return (1e30,np.zeros(n))

def soft_obj(x, n, tau):
    x = np.asarray(x, dtype=np.float64)
    if x.shape != (n,) or not np.all(np.isfinite(x)):
        return (1e30, np.zeros(n))
    if float(np.min(x)) < -1e-6 or float(np.max(x)) > 1.0 + 1e-6:
        return (1e30, np.zeros(n))
    xc = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    q = np.correlate(xc, 1.0 - xc, mode='full') * dx
    if not np.all(np.isfinite(q)):
        return (1e30, np.zeros(n))
    qmax = float(np.max(q))
    e = np.exp((q - qmax) / tau)
    s = float(np.sum(e))
    if s <= 0.0 or not math.isfinite(qmax) or not math.isfinite(s):
        return (1e30, np.zeros(n))
    val = qmax + tau * math.log(s)
    w = e / s
    g = np.zeros(n)
    L = 2 * n - 1
    for k in range(L):
        wk = float(w[k])
        if wk == 0.0:
            continue
        d = k - (n - 1)
        p = max(d, 0)
        r = max(-d, 0)
        m = n - abs(d)
        c = wk * dx
        g[p:p + m] += c * (1.0 - xc[r:r + m])
        g[r:r + m] -= c * xc[p:p + m]
    if float(np.max(np.abs(x - xc))) > 1e-9:
        mask = (x > -1e-12) & (x < 1.0 + 1e-12)
        g = g * mask
    if not np.all(np.isfinite(g)):
        return (1e30, np.zeros(n))
    return float(val), g

from scipy.optimize import minimize
import math

TAU_STAGES = (0.02, 0.005, 0.00125, 0.0003, 0.000075, 0.00002)
ALPHA_STEPS = (0.5, 0.25, 0.125, 0.0625)

def is_feasible(h, n):
    if h is None or h.shape != (n,):
        return False
    if not np.all(np.isfinite(h)):
        return False
    if float(np.min(h)) < -1e-9 or float(np.max(h)) > 1.0 + 1e-9:
        return False
    return abs(float(np.sum(h)) - 0.5 * n) <= 1e-8 * max(1, n)

def project_capped(z, n):
    try:
        z = np.asarray(z, dtype=np.float64).reshape(-1)
    except Exception:
        return None
    if z.size != n or not np.all(np.isfinite(z)):
        return None
    target = 0.5 * n
    if float(np.min(z)) >= -1e-14 and float(np.max(z)) <= 1.0 + 1e-14:

    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(3):

    return h

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if q.shape[0] != 2 * n - 1 or not np.all(np.isfinite(q)):
        return float('inf')
    return float(np.max(q))

def soft_obj(x, n, tau):
    x = np.asarray(x, dtype=np.float64)
    if x.shape != (n,) or not np.all(np.isfinite(x)):
        return (1e30, np.zeros(n))
    if float(np.min(x)) < -1e-5 or float(np.max(x)) > 1.0 + 1e-5:
        return (1e30, np.zeros(n))
    xc = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    q = np.correlate(xc, 1.0 - xc, mode='full') * dx
    if not np.all(np.isfinite(q)):
        return (1e30, np.zeros(n))
    qmax = float(np.max(q))
    e = np.exp((q - qmax) / tau)
    s = float(np.sum(e))
    if s <= 0.0 or not math.isfinite(qmax) or not math.isfinite(s):
        return (1e30, np.zeros(n))
    val = qmax + tau * math.log(s)
    w = e / s
    g = np.zeros(n)
    L = 2 * n - 1
    for k in range(L):
        wk = float(w[k])
        if wk == 0.0:
            continue
        d = k - (n - 1)
        p = max(d, 0)
        r = max(-d, 0)
        m = n - abs(d)
        c = wk * dx
        g[p:p + m] += c * (1.0 - xc[r:r + m])
        g[r:r + m] -= c * xc[p:p + m]
    if float(np.max(np.abs(x - xc))) > 1e-9:
        mask = (x > -1e-12) & (x < 1.0 + 1e-12)
        g = g * mask
    if not np.all(np.isfinite(g)):
        return (1e30, np.zeros(n))
    return float(val), g

def mass_jac(x, n):
    return np.ones(n)

def consider_candidate(z, n, best_h, best_score):

            )

def epi_ineq(z):
    n = z.shape[0] - 1
    h = z[:n]
    q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return z[-1] - q

def epi_ineq_grad(z):
    n = z.shape[0] - 1
    h = z[:n]
    dx = 2.0 / n
    L = 2 * n - 1
    J = np.zeros((L, n + 1))
    J[:, n] = 1.0
    for k in range(L):
        d = k - (n - 1)
        p = max(d, 0)
        r = max(-d, 0)
        m = n - abs(d)
        J[k, p:p + m] -= dx * (1.0 - h[r:r + m])
        J[k, r:r + m] += dx * h[p:p + m]
    return J

def epi_eq(z):
    n = z.shape[0] - 1
    return float(np.sum(z[:n]) - 0.5 * n)

        final_h = fallback
        final_score = exact_score(fallback, n)
    return final_h, float(final_score), int(n)
