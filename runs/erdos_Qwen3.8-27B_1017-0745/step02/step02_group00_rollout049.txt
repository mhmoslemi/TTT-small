def lag_grad_active(h,n,active):

  off=n-1

     d = idx - off

     G[a, lo:hi] += 1.0 - h[lo+d:hi+d]

```
import numpy as np
from scipy.optimize import minimize

POLISH_FRACTIONS = (0.0, 1.0, 0.5, 0.25, 0.75, 0.125, 0.875)

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    s = 0.5 * float(n)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if mid == lo or mid == hi:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for i in range(n):
        if abs(r) < 1e-13:
            break
        if r > 0.0:
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
        else:
            a = min(-r, h[i])
            h[i] -= a
            r += a
    if not np.all(np.isfinite(h)) or abs(s - float(np.sum(h))) > 1e-8:
        return None
    return h
```

```
def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.size != n or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return float('inf')
    if abs(float(np.sum(h)) - 0.5 * n) > 1e-8:
        return float('inf')
    val = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)
    return val if np.isfinite(val) else float('inf')
```

```
def select_active(h, n, frac, mx):
    q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    C = float(np.max(q))
    thr = C - max(1e-5, frac * C)
    active = np.where(q > thr)[0]
    if active.size == 0:
        active = np.array([int(np.argmax(q))])
    mx = max(1, min(mx, 2 * n - 1))
    if active.size > mx:
        order = np.argsort(-q[active])
        active = active[order[:mx]]
    return active
```

```
def lag_grad_active(h, n, active):
    mx = len(active)
    G = np.zeros((mx, n), dtype=np.float64)
    dx = 2.0 / n
    off = n - 1
    for a, idx in enumerate(active):
        d = off - int(idx)
        lo = max(0, -d)
        hi = min(n, n - d)
        G[a, lo:hi] += 1.0 - h[lo + d:hi + d]
        G[a, lo + d:hi + d] -= h[lo:hi]
    return G * dx
```

```
def sl_obj(z, n):
    return float(z[n])

def sl_obj_jac(z, n):
    j = np.zeros(n + 1)
    j[n] = 1.0
    return j

def sl_eq(z, n):
    return float(np.sum(z[:n]) - 0.5 * n)

def sl_eq_jac(z, n):
    j = np.zeros(n + 1)
    j[:n] = 1.0
    return j

def sl_ineq(z, n, active):
    h = z[:n]
    q = np.correlate(h, 1.0 - h, mode='full')[active] * (2.0 / n)
    return float(z[n]) - q

def sl_ineq_jac(z, n, active):
    h = z[:n]
    G = lag_grad_active(h, n, active)
    jc = np.zeros((len(active), n + 1))
    if len(active):
        jc[:, :n] = -G
        jc[:, n] = 1.0
    return jc
```

```
def try_slsqp(hstart, n, outer, iters, mcnt, frac):
    h = project_capped(hstart, n)
    if h is None:
        return None
    s = exact_score(h, n)
    if not np.isfinite(s):
        return None
    for _ in range(outer):
        active = select_active(h, n, frac, mcnt)
        z0 = np.empty(n + 1)
        z0[:n] = h
        z0[n] = min(1.0, s + 0.05)
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        cons = [
            {'type': 'eq', 'fun': sl_eq, 'jac': sl_eq_jac, 'args': (n,)},
            {'type': 'ineq', 'fun': sl_ineq, 'jac': sl_ineq_jac, 'args': (n, active)},
        ]
        try:
            res = minimize(sl_obj, z0, args=(n,), jac=sl_obj_jac, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'ftol': 1e-10, 'maxiter': iters})
        except Exception:
            break
        if res is None or res.x is None or len(res.x) != n + 1 or not np.all(np.isfinite(res.x)):
            break
        h2 = project_capped(res.x[:n], n)
        if h2 is None:
            break
        s2 = exact_score(h2, n)
        if np.isfinite(s2) and s2 < s - 1e-12:
            h, s = h2, s2
        else:
            break
    return h, s
```

```
def line_search_pair(h, i, j, n, cur):
    a = float(h[i]); b = float(h[j])
    lo = max(-a, b - 1.0)
    hi = min(1.0 - a, b)
    if hi - lo < 1e-12:
        return 0.0, cur
    base = hi - lo
    best = cur
    bestd = 0.0
    for f in POLISH_FRACTIONS:
        d = lo + f * base
        if abs(d) < 1e-15:
            continue
        h[i] = a + d
        h[j] = b - d
        s = exact_score(h, n)
        if np.isfinite(s) and s < best - 1e-12:
            best = s
            bestd = d
    h[i] = a
    h[j] = b
    return bestd, best
```

```
def get_pair_list(h, n, mode):
    if n < 2:
        return []
    total = n * (n - 1) // 2
    if mode == 'full':
        return [(i, j) for i in range(n) for j in range(i + 1, n)]
    active = select_active(h, n, 0.02, 24)
    off = n - 1
    ps = set()
    for idx in active:
        d = off - int(idx)
        ad = abs(d)
        if ad == 0 or ad >= n:
            continue
        for i in range(n - ad):
            a = i
            b = i + ad
            if a < b:
                ps.add((a, b))
            else:
                ps.add((b, a))
    if total and len(ps) < max(6, total // 4):
        return list(ps)
    return [(i, j) for i in range(n) for j in range(i + 1, n)]
```

```
def polish(hstart, n, s, rng, act_sw, full_sw):
    h = project_capped(hstart, n)
    if h is None:
        return None, s
    cur = exact_score(h, n)
    if not np.isfinite(cur):
        return None, s
    for mode, sw in (('active', act_sw), ('full', full_sw)):
        for _ in range(sw):
            pairs = get_pair_list(h, n, mode)
            if not pairs:
                break
            rng.shuffle(pairs)
            improved = False
            for i, j in pairs:
                d, best = line_search_pair(h, i, j, n, cur)
                if np.isfinite(best) and best < cur - 1e-12:
                    h[i] += d
                    h[j] -= d
                    cur = best
                    improved = True
            if not improved:
                break
            if abs(float(np.sum(h)) - 0.5 * n) > 1e-8:
                hp = project_capped(h, n)
                if hp is not None:
                    h = hp
                    cur = exact_score(h, n)
    hp = project_capped(h, n)
    if hp is not None:
        s2 = exact_score(hp, n)
        if np.isfinite(s2) and s2 <= cur + 1e-9:
            return hp, s2
    return h, cur
```

```
def balance_binary(n, v):
    t = n // 2
    h = (np.asarray(v, dtype=np.float64).ravel() > 0.5).astype(float)
    c = int(np.sum(h))
    if c > t:
        idx = np.where(h > 0.5)[0]
        rem = idx[np.argsort(v[idx])[:c - t]]
        h[rem] = 0.0
    elif c < t:
        idx = np.where(h <= 0.5)[0]
        add = idx[np.argsort(v[idx])[::-1][:t - c]]
        h[add] = 1.0
    return h

def random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def interp_to_length(hsrc, nsrc, ndst):
    if nsrc == ndst:
        return np.asarray(hsrc, dtype=np.float64).ravel().copy()
    if nsrc == 1:
        return np.full(ndst, float(hsrc.ravel()[0]))
    x1 = np.linspace(0.0, 2.0, nsrc)
    x2 = np.linspace(0.0, 2.0, ndst)
    return np.interp(x2, x1, np.asarray(hsrc, dtype=np.float64).ravel())

def binary_local(hstart, n, passes):
    h = np.asarray(hstart, dtype=np.float64).ravel().copy()
    if abs(float(np.sum(h)) - 0.5 * n) > 1e-8:
        h = balance_binary(n, h)
    cur = exact_score(h, n)
    if not np.isfinite(cur):
        return None, cur
    for _ in range(passes):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        best_s = cur + 1e-12
        best_ij = None
        for i in ones:
            for j in zeros:
                if h[i] < 0.5 or h[j] > 0.5:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                s = exact_score(h, n)
                h[i] = 1.0
                h[j] = 0.0
                if np.isfinite(s) and s < best_s:
                    best_s = s
                    best_ij = (int(i), int(j))
        if best_ij is not None:
            i, j = best_ij
            h[i] = 0.0
            h[j] = 1.0
            cur = best_s
        else:
            break
    return h, cur
```

```python
import numpy as np
from scipy.optimize import minimize

POLISH_FRACTIONS = (0.0, 1.0, 0.5, 0.25, 0.75, 0.125, 0.875)


def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    s = 0.5 * float(n)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if mid == lo or mid == hi:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for i in range(n):
        if abs(r) < 1e-13:
            break
        if r > 0.0:
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
        else:
            a = min(-r, h[i])
            h[i] -= a
            r += a
    if (not np.all(np.isfinite(h)) or np.any(h < 0.0) or np.any(h > 1.0)
            or abs(float(np.sum(h)) - s) > 1e-8):
        return None
    return h


def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.size != n or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return float('inf')
    if abs(float(np.sum(h)) - 0.5 * n) > 1e-8:
        return float('inf')
    val = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)
    return val if np.isfinite(val) else float('inf')


def add_start(lst, h, n, mx):
    if len(lst) >= mx:
        return
    p = project_capped(h, n)
    if p is None:
        return
    s = exact_score(p, n)
    if np.isfinite(s):
        lst.append((s, p, n))


def select_active(h, n, frac, mx):
    q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    c = float(np.max(q))
    thr = c - max(1e-5, frac * c)
    active = np.where(q > thr)[0]
    if active.size == 0:
        active = np.array([int(np.argmax(q))])
    mx = max(1, min(mx, 2 * n - 1))
    if active.size > mx:
        order = np.argsort(-q[active])
        active = active[order[:mx]]
    return active


def lag_grad_active(h, n, active):
    mx = len(active)
    g = np.zeros((mx, n), dtype=np.float64)
    dx = 2.0 / n
    off = n - 1
    for a, idx in enumerate(active):
        d = off - int(idx)
        lo = max(0, -d)
        hi = min(n, n - d)
        g[a, lo:hi] += 1.0 - h[lo + d:hi + d]
        g[a, lo + d:hi + d] -= h[lo:hi]
    return g * dx


def sl_obj(z, n):
    return float(z[n])


def sl_obj_jac(z, n):
    j = np.zeros(n + 1)
    j[n] = 1.0
    return j


def sl_eq(z, n):
    return float(np.sum(z[:n]) - 0.5 * n)


def sl_eq_jac(z, n):
    j = np.zeros(n + 1)
    j[:n] = 1.0
    return j


def sl_ineq(z, n, active):
    h = z[:n]
    q = np.correlate(h, 1.0 - h, mode='full')[active] * (2.0 / n)
    return float(z[n]) - q


def sl_ineq_jac(z, n, active):
    h = z[:n]
    g = lag_grad_active(h, n, active)
    j = np.zeros((len(active), n + 1))
    if len(active):
        j[:, :n] = -g
        j[:, n] = 1.0
    return j


def try_slsqp(hstart, n, outer, iters, mcnt, frac):
    h = project_capped(hstart, n)
    if h is None:
        return None
    s = exact_score(h, n)
    if not np.isfinite(s):
        return None
    for _ in range(outer):
        active = select_active(h, n, frac, mcnt)
        z0 = np.empty(n + 1)
        z0[:n] = h
        z0[n] = min(1.0, s + 0.05)
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        cons = [
            {'type': 'eq', 'fun': sl_eq, 'jac': sl_eq_jac, 'args': (n,)},
            {'type': 'ineq', 'fun': sl_ineq, 'jac': sl_ineq_jac, 'args': (n, active)},
        ]
        try:
            res = minimize(sl_obj, z0, args=(n,), jac=sl_obj_jac, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'ftol': 1e-10, 'maxiter': iters})
        except Exception:
            break
        if res is None or res.x is None or len(res.x) != n + 1 or not np.all(np.isfinite(res.x)):
            break
        h2 = project_capped(res.x[:n], n)
        if h2 is None:
            break
        s2 = exact_score(h2, n)
        if np.isfinite(s2) and s2 <= s + 1e-12:
            h, s = h2, s2
        else:
            break
    return h, s


def line_search_pair(h, i, j, n, cur):
    a = float(h[i])
    b = float(h[j])
    lo = max(-a, b - 1.0)
    hi = min(1.0 - a, b)
    if hi - lo < 1e-12:
        return 0.0, cur
    base = hi - lo
    best = cur
    bestd = 0.0
    for f in POLISH_FRACTIONS:
        d = lo + f * base
        if abs(d) < 1e-15:
            continue
        h[i] = a + d
        h[j] = b - d
        s = exact_score(h, n)
        if np.isfinite(s) and s < best - 1e-12:
            best = s
            bestd = d
    h[i] = a
    h[j] = b
    return bestd, best


def get_pair_list(h, n, mode):
    if n < 2:
        return []
    total = n * (n - 1) // 2
    if mode == 'full':
        return [(i, j) for i in range(n) for j in range(i + 1, n)]
    active = select_active(h, n, 0.02, 24)
    off = n - 1
    ps = set()
    for idx in active:
        d = off - int(idx)
        ad = abs(d)
        if ad == 0 or ad >= n:
            continue
        for i in range(n - ad):
            a = i
            b = i + ad
            if a < b:
                ps.add((a, b))
            else:
                ps.add((b, a))
    if total and len(ps) < max(6, total // 4):
        return list(ps)
    return [(i, j) for i in range(n) for j in range(i + 1, n)]


def polish(hstart, n, s, rng, act_sw, full_sw):
    h = project_capped(hstart, n)
    if h is None:
        return None, s
    cur = exact_score(h, n)
    if not np.isfinite(cur):
        return None, s
    for mode, sw in (('active', act_sw), ('full', full_sw)):
        for _ in range(sw):
            pairs = get_pair_list(h, n, mode)
            if not pairs:
                break
            rng.shuffle(pairs)
            improved = False
            for i, j in pairs:
                d, best = line_search_pair(h, i, j, n, cur)
                if np.isfinite(best) and best < cur - 1e-12:
                    h[i] += d
                    h[j] -= d
                    cur = best
                    improved = True
            if not improved:
                break
            if abs(float(np.sum(h)) - 0.5 * n) > 1e-8:
                hp = project_capped(h, n)
                if hp is not None:
                    h = hp
                    cur = exact_score(h, n)
    hp = project_capped(h, n)
    if hp is not None:
        s2 = exact_score(hp, n)
        if np.isfinite(s2) and s2 <= cur + 1e-9:
            return hp, s2
    return h, cur


def balance_binary(n, v):
    t = n // 2
    v = np.asarray(v, dtype=np.float64).ravel()
    h = (v > 0.5).astype(float)
    c = int(np.sum(h))
    if c > t:
        idx = np.where(h > 0.5)[0]
        rem = idx[np.argsort(v[idx])[:c - t]]
        h[rem] = 0.0
    elif c < t:
        idx = np.where(h <= 0.5)[0]
        add = idx[np.argsort(v[idx])[::-1][:t - c]]
        h[add] = 1.0
    return h


def random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def interp_to_length(hsrc, nsrc, ndst):
    if nsrc == ndst:
        return np.asarray(hsrc, dtype=np.float64).ravel().copy()
    if nsrc == 1:
        return np.full(ndst, float(np.asarray(hsrc).ravel()[0]))
    x1 = np.linspace(0.0, 2.0, nsrc)
    x2 = np.linspace(0.0, 2.0, ndst)
    return np.interp(x2, x1, np.asarray(hsrc, dtype=np.float64).ravel())


def binary_local(hstart, n, passes):
    h = np.asarray(hstart, dtype=np.float64).ravel().copy()
    if abs(float(np.sum(h)) - 0.5 * n) > 1e-8:
        h = balance_binary(n, h)
    cur = exact_score(h, n)
    if not np.isfinite(cur):
        return None, cur
    for _ in range(passes):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        best_s = cur + 1e-12
        best_ij = None
        for i in ones:
            for j in zeros:
                if h[i] < 0.5 or h[j] > 0.5:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                s = exact_score(h, n)
                h[i] = 1.0
                h[j] = 0.0
                if np.isfinite(s) and s < best_s:
                    best_s = s
                    best_ij = (int(i), int(j))
        if best_ij is not None:
            i, j = best_ij
            h[i] = 0.0
            h[j] = 1.0
            cur = best_s
        else:
            break
    return h, cur


def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        pv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= pv.size <= 999:
            n = int(pv.size)
            if np.all(np.isfinite(pv)) and np.all(pv >= -1e-8) and np.all(pv <= 1.0 + 1e-8):
                parent = pv.copy()
    except Exception:
        n = None
        parent = None

    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
    n = max(1, min(int(n), 999))

    try:
        sd = int(seed)
    except Exception:
        sd = 42
    rng = np.random.default_rng(abs(sd) % 1000000007 if sd != 0 else 42)

    if n <= 96:
        mx, osl, outer, iters, mcnt, act, fs, bs, bp = 14, 4, 4, 60, 28, 6, 2, 2, 25
        binok = True
    else:
        mx, osl, outer, iters, mcnt, act, fs, bs, bp = 8, 2, 2, 30, 16, 1, 0, 0, 0
        binok = False

    uniform = np.full(n, 0.5)
    starts = []
    add_start(starts, uniform, n, mx)

    if parent is not None:
        p = np.clip(parent, 0.0, 1.0)
        add_start(starts, p, n, mx)
        for raw in (p[::-1], 1.0 - p, 1.0 - p[::-1]):
            add_start(starts, raw, n, mx)
        for sdv in (0.03, 0.07, 0.12):
            add_start(starts, p + rng.normal(0.0, sdv, n), n, mx)
    else:
        add_start(starts, 0.5 + rng.normal(0.0, 0.1, n), n, mx)

    idx = np.arange(n, dtype=np.float64)
    for f in (1, 2, 3, 5):
        ang = 2.0 * np.pi * f * idx / float(n)
        add_start(starts, 0.5 + 0.45 * np.cos(ang), n, mx)
        add_start(starts, 0.5 + 0.45 * np.sin(ang), n, mx)

    for bl in (0.25, 0.5, 0.75):
        add_start(starts, np.where(idx < 2.0 * bl, 1.0, 0.0), n, mx)

    for aa in (0.2, 0.4, 0.5):
        x = idx * (2.0 / n)
        w = 2.0 * (1.0 - aa)
        hr = np.zeros(n)
        hr[x <= aa] = 1.0
        m = (x > aa) & (x < 2.0 - aa)
        if w > 1e-12:
            hr[m] = (2.0 - aa - x[m]) / w
        add_start(starts, hr, n, mx)

    add_start(starts, rng.uniform(0.0, 1.0, n), n, mx)

    if not starts:
        starts = [(exact_score(uniform, n), uniform.copy(), n)]

    items = [(s, k, arr, n) for k, (s, arr, nn) in enumerate(starts) if nn == n]
    items.sort()

    best_h = items[0][2].copy()
    best_s = items[0][0]
    best_n = items[0][3]

    for itm in items:
        if itm[0] < best_s:
            best_h = itm[2].copy()
            best_s = itm[0]
            best_n = itm[3]

    cand = list(items)
    cid = len(items)

    for k in range(min(osl, len(items))):
        arr = items[k][2]
        m = min(2 * n - 1, 120) if k == 0 else mcnt
        res = try_slsqp(arr, n, outer, iters, m, 0.02)
        if res is not None:
            h, s = res
            if np.isfinite(s):
                cand.append((s, cid, h, n))
                cid += 1
                if s < best_s:
                    best_h = h.copy()
                    best_s = s
                    best_n = n

    cand.sort()
    pcount = 3 if n <= 96 else 1
    last = None
    for k in range(min(pcount, len(cand))):
        s, _, arr, nn = cand[k]
        if last is not None and np.allclose(arr, last, atol=1e-7):
            continue
        last = arr
        hp, ps = polish(arr, nn, s, rng, act, fs)
        if hp is not None and np.isfinite(ps) and ps < best_s:
            best_h = hp.copy()
            best_s = ps
            best_n = nn

    if binok and bs > 0:
        nb = n if n % 2 == 0 else n + 1
        if nb > 160:
            nb = n if n % 2 == 0 else n - 1
        if 8 <= nb <= 160:
            for bk in range(bs):
                if bk == 0:
                    v = interp_to_length(best_h, best_n, nb)
                    h0 = balance_binary(nb, v)
                else:
                    h0 = random_binary(nb, rng)
                hb, sb = binary_local(h0, nb, bp)
                if hb is None:
                    continue
                if np.isfinite(sb) and sb < best_s:
                    best_h = hb.copy()
                    best_s = sb
                    best_n = nb
                if nb <= 128 and np.isfinite(sb) and sb < best_s + 0.02:
                    hp, ps = polish(hb, nb, sb, rng, 2, 1)
                    if hp is not None and np.isfinite(ps) and ps < best_s:
                        best_h = hp.copy()
                        best_s = ps
                        best_n = nb

    final = project_capped(best_h, best_n)
    if final is None:
        final = np.full(best_n, 0.5)
    fs = exact_score(final, best_n)
    if not np.isfinite(fs) or fs > best_s + 1e-8:
        final = best_h.copy()
        fs = best_s
    if not np.isfinite(fs):
        final = np.full(best_n, 0.5)
        fs = exact_score(final, best_n)
    return final, float(fs), int(best_n)
```
