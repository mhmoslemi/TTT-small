def corr_lags(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def pair_transfer(h, i, j, t, n):
    c = min(1.0 - h[i], h[j])
    if c <= 1e-15:
        return None
    t = min(t, c)
    trial = h.copy()
    trial[i] += t
    trial[j] -= t
    return project_capped(trial, n)

def local_refine(h, n, rng, max_iter=500, sigma0=0.01):
    best = h.copy()
    best_s = exact_score(h, n)
    sigma = sigma0
    for it in range(max_iter):
        z = best + sigma * rng.normal(0, 1, n)
        p = project_capped(z, n)
        if p is None:
            continue
        s = exact_score(p, n)
        if s < best_s - 1e-13:
            best = p
            best_s = s
        sigma *= 0.995
        if sigma < 1e-8:
            break
    return best, best_s

def pair_polish(h, n, rng, n_sweeps, max_pairs):
    best = h.copy()
    best_s = exact_score(h, n)
    pairs = []
    if n <= 64:
        for i in range(n):
            for j in range(i + 1, n):
                pairs.append((i, j))
    else:
        for _ in range(max_pairs):
            i, j = int(rng.integers(n)), int(rng.integers(n))
            if i != j:
                pairs.append((i, j))
    fractions = [1.0/32, 1.0/8, 1.0/4, 1.0/2, 0.75, 1.0]
    for sweep in range(n_sweeps):
        imp = False
        for (i, j) in pairs:
            cur = best
            cur_s = best_s
            for f in fractions:
                for (src, dst) in [(i, j), (j, i)]:
                    t = f * min(1.0 - cur[dst], cur[src])
                    if t < 1e-14:
                        continue
                    trial = cur.copy()
                    trial[dst] += t
                    trial[src] -= t
                    p = project_capped(trial, n)
                    if p is None:
                        continue
                    s = exact_score(p, n)
                    if s < cur_s - 1e-13:
                        cur = p
                        cur_s = s
            if cur_s < best_s - 1e-13:
                best = cur
                best_s = cur_s
                imp = True
        if not imp:
            break
    return best, best_s

def anneal_pairs(h, n, rng, L, T0=3e-4, T1=1e-8):
    best = h.copy()
    best_s = exact_score(h, n)
    cur = h.copy()
    cur_s = best_s
    for t in range(L):
        q = t / max(1, L - 1)
        T = T0 * (T1 / T0) ** q
        i, j = int(rng.integers(n)), int(rng.integers(n))
        if i == j:
            continue
        c = min(1.0 - cur[i], cur[j])
        if c <= 1e-14:
            continue
        rho = 1.0 - 0.99 * q
        amt = c * rho * (0.02 + 0.98 * rng.random())
        trial = cur.copy()
        trial[i] += amt
        trial[j] -= amt
        p = project_capped(trial, n)
        if p is None:
            continue
        s = exact_score(p, n)
        if s < cur_s - 1e-13:
            cur, cur_s = p, s
        elif rng.random() < np.exp(-max(s - cur_s, 0) / max(T, 1e-18)):
            cur, cur_s = p, s
        if cur_s < best_s - 1e-13:
            best, best_s = cur.copy(), cur_s
    return best, best_s
