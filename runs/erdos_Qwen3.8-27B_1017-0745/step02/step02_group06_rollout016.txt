            z = rng.standard_normal(d)
            if full_cov:
                ev, evec = np.linalg.eigh(0.5 * (C + C.T))
                ev = np.clip(ev, smin2, smax2)
                Lz = (evec * np.sqrt(ev)) @ z
            else:
                Lz = np.sqrt(np.clip(v, smin2, smax2)) * z
            for sgn in (1.0, -1.0):
                x = m + sgn * Lz
                h = project_capped(u + helmert_fwd(x, n), n)
                if h is None:
                    continue
                sc = exact_score(h, n)
                if np.isfinite(sc):
                    xs.append(helmert_t(h - u, n))
                    hs.append(h)
                    scs.append(sc)
                    if sc < gb_score - 1e-12:
                        gb_score = sc
                        gb_vec = h.copy()
        if not xs:
            break
        scs_arr = np.array(scs)
        idx = np.argsort(scs_arr)
        mu_eff = min(mu, len(xs))
        sel = idx[:mu_eff]
        wv = np.array([math.log(mu_eff + 0.5) - math.log(float(r + 1)) for r in range(mu_eff)])
        wv /= wv.sum()
        m_e = np.zeros(d)
        for r in range(mu_eff):
            m_e += wv[r] * xs[sel[r]]
        m_new = (1.0 - eta) * m + eta * m_e
        hp = project_capped(u + helmert_fwd(m_new, n), n)
        if hp is not None and np.all(np.isfinite(hp)):
            m_new = helmert_t(hp - u, n)
        else:
            break
        if full_cov:
            S = np.zeros((d, d))
            for r in range(mu_eff):
                diff = xs[sel[r]] - m_new
                S += wv[r] * np.outer(diff, diff)
            C = (1.0 - eta) * C + eta * S
            C = 0.5 * (C + C.T)
            if not np.all(np.isfinite(C)):
                break
        else:
            Sv = np.zeros(d)
            for r in range(mu_eff):
                diff = xs[sel[r]] - m_new
                Sv += wv[r] * diff * diff
            v = np.clip((1.0 - eta) * v + eta * Sv, smin2, smax2)
        best_gen = float(scs_arr[idx[0]])
        if best_gen < gb_score + 1e-12:
            stall = 0
        else:
            stall += 1
        if stall >= stall_lim:
            break
        m = m_new
    return gb_vec, gb_score

def build_starts(parent, n, rng):
    pool = []
    u = np.full(n, 0.5)
    cands = [u]
    if parent is not None:
        cands.append(parent.copy())
        cands.append(parent[::-1].copy())
        cands.append((1.0 - parent[::-1]).copy())
        cands.append((1.0 - parent).copy())
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    cands.append(blk)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0).astype(np.float64)
    cands.append(alt)
    perm = rng.permutation(n)
    bb = np.zeros(n)
    bb[perm[:n // 2]] = 1.0
    if n % 2:
        bb[perm[n // 2]] = 0.5
    cands.append(bb)
    base = parent if parent is not None else u
    for sd in (0.05, 0.12, 0.22):
        cands.append(base + rng.normal(0, sd, n))
    cands.append(rng.uniform(0, 1, n))
    for c in cands:
        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-7:
                dup = True
                break
        if not dup:
            pool.append(p)
    return pool

def refine_local(h, n, rng, max_rounds):
    h = h.copy()
    sc = exact_score(h, n)
    for _ in range(max_rounds):
        improved = False
        order = rng.permutation(n)
        for i in order:
            for di in (-0.02, -0.01, 0.01, 0.02):
                ht = h.copy()
                ht[i] += di
                for j in order:
                    if j == i:
                        continue
                    ht[j] -= di * (1.0 - ht[j]) / max(1e-12, 1.0 - 0.5)
                    ht = project_capped(ht, n)
                    if ht is None or not is_feasible(ht, n):
                        continue
                    s2 = exact_score(ht, n)
                    if s2 < sc - 1e-13:
                        h = ht
                        sc = s2
                        improved = True
                    break
        if not improved:
            break
    return h, sc

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 1 and p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                if abs(np.sum(p) - n / 2.0) <= 1e-5 * max(1, n):
                    parent = project_capped(np.clip(p, 0, 1), n)
    except Exception:
        pass
    if n is None:
        n = 47
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    u = np.full(n, 0.5)
    if n == 1:
        return u, 0.5, 1
    gb_vec = u.copy()
    gb_score = exact_score(u, n)
    if parent is not None and is_feasible(parent, n):
        ps = exact_score(parent, n)
        if ps < gb_score - 1e-12:
            gb_score = ps
            gb_vec = parent.copy()
    starts = build_starts(parent, n, rng)
    scored = []
    for st in starts:
        s = exact_score(st, n)
        if np.isfinite(s):
            scored.append((s, st))
            if s < gb_score - 1e-12:
                gb_score = s
                gb_vec = st.copy()
    scored_arr = np.array([s for s, _ in scored])
    order = np.argsort(scored_arr)
    if n <= 64:
        n_st, pop, ngen, stall, fc = min(5, len(scored)), 20, 60, 18, True
    elif n <= 128:
        n_st, pop, ngen, stall, fc = min(4, len(scored)), 16, 35, 14, True
    elif n <= 256:
        n_st, pop, ngen, stall, fc = min(2, len(scored)), 12, 20, 10, False
    else:
        n_st, pop, ngen, stall, fc = min(1, len(scored)), 8, 10, 6, False
    for oi in range(min(n_st, len(order))):
        st = scored[order[oi]][1]
        gb_vec, gb_score = cem_run(st, n, rng, pop, ngen, stall, fc, gb_vec, gb_score)
    if n <= 64:
        gb_vec, gb_score = refine_local(gb_vec, n, rng, 3)
    final = project_capped(gb_vec, n)
    if final is None or not is_feasible(final, n):
        final = u
    fs = exact_score(final, n)
    fb = exact_score(u, n)
    if fs > fb + 1e-12:
        final = u
        fs = fb
    if parent is not None and is_feasible(parent, n):
        pf = exact_score(parent, n)
        if pf < fs - 1e-12:
            final = parent
            fs = pf
    return final, float(fs), int(n)
