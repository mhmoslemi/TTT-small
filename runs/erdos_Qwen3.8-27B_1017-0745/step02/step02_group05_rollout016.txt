def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    tol = 2e-14 * max(1, n)
    for _ in range(n + 2):
        if abs(r) < tol:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    return h

def is_feasible(h, n):

        return False
    if abs(np.sum(h) - n / 2.0) > 1e-10 * max(1, n):

    return True

        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

    v = M + tau * np.log(np.sum(e))

def smooth_grad(h, n, tau):
    dx = 2.0 / n
    Q = dx * np.correlate(h, 1.0 - h, mode='full')
    if not np.all(np.isfinite(Q)):
        return np.zeros(n)
    M = float(np.max(Q))
    e = np.exp(np.clip((Q - M) / tau, -745.0, 0.0))
    sw = float(np.sum(e))
    if sw < 1e-300:
        return np.zeros(n)
    w = e / sw
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-15:
            continue
        d = idx - (n - 1)
        a = d if d > 0 else 0
        b = -d if d < 0 else 0
        m = n - abs(d)
        g[a:a + m] += wi * dx * (1.0 - h[b:b + m])
        g[b:b + m] -= wi * dx * h[a:a + m]
    return g if np.all(np.isfinite(g)) else np.zeros(n)
