import numpy as np
import math

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    s = n / 2.0
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if mid == lo or mid == hi:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for i in range(n):
        if abs(r) < 1e-13:
            break
        if r > 0:
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
        else:
            a = min(-r, h[i])
            h[i] -= a
            r += a
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-10:
        return None
    return h

def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return float('inf')
    if abs(float(np.sum(h)) - n / 2.0) > 1e-8:
        return float('inf')
    try:
        return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
    except Exception:
        dx = 2.0 / n
        best = 0.0
        for d in range(-(n - 1), n):
            lo = max(0, -d)
            hi = min(n, n - d)
            v = float(np.dot(h[lo:hi], 1.0 - h[lo + d:hi + d]))
            if v > best:
                best = v
        return best * dx

def is_feasible(h, n):
    h = np.asarray(h, dtype=np.float64)
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            np.all(h >= -1e-9) and np.all(h <= 1.0 + 1e-9) and
            abs(float(np.sum(h)) - n / 2.0) <= 1e-8)

def make_proposal(h, n, rng, delta):
    if n < 2:
        return None
    m = 1
    if rng.random() < 0.25 and n // 2 >= 2:
        m = int(rng.integers(2, min(6, n // 2) + 1))
    idx = rng.permutation(n)
    pairs = [(int(idx[2 * r]), int(idx[2 * r + 1])) for r in range(m)]
    new_h = h.copy()
    t_scale = delta / math.sqrt(float(m))
    for (i, j) in pairs:
        S = new_h[i] + new_h[j]
        t = rng.uniform(-t_scale, t_scale)
        lo_t = max(-new_h[i], new_h[j] - 1.0)
        hi_t = min(1.0 - new_h[i], new_h[j])
        if lo_t >= hi_t:
            return None
        if t < lo_t or t > hi_t:
            return None
        new_h[i] += t
        new_h[j] -= t
    if not is_feasible(new_h, n):
        return None
    if float(np.max(np.abs(new_h - h))) < 1e-15:
        return None
    return new_h

def build_starts(n, rng, parent):
    starts = []
    uniform = np.full(n, 0.5)
    starts.append(uniform)
    if parent is not None:
        starts.append(parent.copy())
        for sd in [0.03, 0.08, 0.16]:
            p = project_capped(parent + rng.normal(0, sd, n), n)
            if p is not None:
                starts.append(p)
        starts.append(1.0 - parent.copy())
        rev = parent[::-1].copy()
        starts.append(rev)
        p = project_capped(1.0 - rev, n)
        if p is not None:
            starts.append(p)
    else:
        for sd in [0.15, 0.35, 0.7]:
            p = project_capped(uniform + rng.normal(0, sd, n), n)
            if p is not None:
                starts.append(p)
    idx = np.arange(n, dtype=np.float64)
    for freq in [1, 2]:
        for phase in [0.0, math.pi / 2]:
            p = project_capped(0.5 + 0.45 * np.cos(2.0 * np.pi * freq * idx / n + phase), n)
            if p is not None:
                starts.append(p)
    p = project_capped(rng.uniform(0, 1, n), n)
    if p is not None:
        starts.append(p)
    dedup = []
    for s in starts:
        dup = False
        for d in dedup:
            if float(np.max(np.abs(np.asarray(s) - np.asarray(d)))) < 1e-8:
                dup = True
                break
        if not dup:
            dedup.append(s)
    return dedup

    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if iv.size > 0 and iv.size <= 999:
            if np.all(np.isfinite(iv)) and np.all(iv >= -1e-8) and np.all(iv <= 1.0 + 1e-8):
                n = iv.size
                parent = np.clip(iv, 0.0, 1.0).copy()
                pp = project_capped(parent, n)
                if pp is not None:
                    parent = pp
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
    n = max(2, min(n, 999))
    try:
        sd = int(seed)
    except Exception:
        sd = 42
    rng = np.random.default_rng(abs(sd) % (2**31) if sd != 0 else 42)
    uniform = np.full(n, 0.5)
    uniform_score = exact_score(uniform, n)
    best, best_score = uniform.copy(), uniform_score
    parent_fallback = None
    if parent is not None:
        ps = exact_score(parent, n)
        if np.isfinite(ps):
            parent_fallback = parent.copy()
            if ps < best_score:
                best, best_score = parent.copy(), ps
    starts = build_starts(n, rng, parent)
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_score - 1e-12:
            best, best_score = st.copy(), ss
    if n <= 64:
        n_chains, n_epochs, moves_ep = 4, 24, 160
    elif n <= 128:
        n_chains, n_epochs, moves_ep = 3, 18, 100
    elif n <= 256:
        n_chains, n_epochs, moves_ep = 2, 14, 45
    else:
        n_chains, n_epochs, moves_ep = 2, 10, 18
    try:
        n_chains = max(1, min(int(kwargs.get('chains', n_chains)), 6))

    total_budget = min(n_chains * n_epochs * moves_ep, cap)
    pilot_n, delta0 = 16, 0.12
    budget_left = total_budget
    for ci in range(n_chains):
        if budget_left <= 0:
            break
        if ci < len(starts):
            cur = starts[ci].copy()
            cur_s = exact_score(cur, n)
        else:
            sd_noise = min(0.04 * (1.5 ** (ci - len(starts))), 0.20)
            p = project_capped(best + rng.normal(0, sd_noise, n), n)

            cur_s = exact_score(cur, n)
        if not np.isfinite(cur_s):

        else:
            T0 = 0.002

                break
            T = T0 * (Tmin / T0) ** (float(ep) / max(1, n_epochs - 1))
            n_moves = min(moves_ep, budget_left)
            accepted = 0
            for _ in range(n_moves):
                p = make_proposal(cur, n, rng, delta)

                    continue
                ns = exact_score(p, n)

                    cur, cur_s = p, ns
                    accepted += 1
            budget_left -= n_moves

                ar = accepted / float(n_moves)
                if ar > 0.40:
                    delta = min(0.35, delta * 1.25)
                elif ar < 0.15:
                    delta = max(0.002, delta * 0.75)
    final = project_capped(best, n)
    if final is None:
        final = best.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs):
        if parent_fallback is not None:
            ps = exact_score(parent_fallback, n)
            if np.isfinite(ps):
                final, fs = parent_fallback.copy(), ps
        else:
            final, fs = uniform.copy(), uniform_score
    if parent_fallback is not None:
        ps = exact_score(parent_fallback, n)
        if np.isfinite(ps) and ps < fs - 1e-12:
            final, fs = parent_fallback.copy(), ps
    uf = exact_score(uniform, n)
    if not np.isfinite(fs) and np.isfinite(uf):
        final, fs = uniform.copy(), uf
    return final, float(fs), int(n)
