import numpy as np
import math

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64)
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    s_target = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s_target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s_target - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx])
            h[idx] += d; r -= d
        else:
            d = min(-r, h[idx])
            h[idx] -= d; r += d
    if abs(r) > 1e-10 or not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    return np.clip(h, 0.0, 1.0)

def is_feasible(h, n):

        return False

        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9 * max(1, n)

def smooth_grad(h, n, tau):
    dx = 2.0 / n
    L = np.correlate(h, 1.0 - h, mode='full') * dx
    M = 2 * n - 1
    a = float(np.max(L))
    e = np.exp(np.maximum((L - a) / tau, -745.0))
    S = float(np.sum(e))
    if S <= 0 or not math.isfinite(S):
        return None, None
    w = e / S
    val = a + tau * (math.log(S) - math.log(M))
    g = np.zeros(n, dtype=np.float64)
    for k in range(M):
        d = k - (n - 1)
        wk = w[k]
        if wk < 1e-14:
            continue
        i0 = max(0, -d)
        i1 = min(n, n - d)
        if i1 <= i0:
            continue
        fa = i0 + d
        fb = i1 + d
        g[fa:fb] += dx * wk * (1.0 - h[i0:i1])
        g[i0:i1] -= dx * wk * h[fa:fb]
    if not np.all(np.isfinite(g)):

    return val, g
