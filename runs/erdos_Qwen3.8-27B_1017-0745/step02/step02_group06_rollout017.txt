import numpy as np
import math

def project_capped(z, n):
    z = np.where(np.isfinite(z), z, 0.0).astype(np.float64)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx])
            h[idx] += d; r -= d
        else:
            d = min(-r, h[idx])
            h[idx] -= d; r += d
    if not np.all(np.isfinite(h)) or abs(np.sum(h) - s) > 1e-9 * max(1, n):
        return None
    return h

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(np.sum(h) - n / 2.0) <= 1e-9 * max(1, n))

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if corr.shape != (2 * n - 1,) or not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr))

def helmert_fwd(x):
    # x: (d,) where d = n-1, returns Q^T @ x of shape (n,)
    n = len(x) + 1
    d = n - 1
    if d == 0:
        return np.zeros(1)
    b = np.zeros(d)
    for j in range(d):
        b[j] = x[j] / math.sqrt((j + 1) * (j + 2))
    T = np.zeros(n)
    T[:d] = np.cumsum(b[::-1])[::-1]
    # T[n-1] = 0 already
    out = np.zeros(n)
    out[0] = T[0]
    for i in range(1, n):
        out[i] = T[i] - i * b[i - 1]
    return out

def helmert_bwd(v):

    out = np.zeros(d)

    return out

import numpy as np

def project_capped(z, n):
    z = np.where(np.isfinite(z), z, 0.0).astype(np.float64)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx])
            h[idx] += d
            r -= d
        else:
            d = min(-r, h[idx])
            h[idx] -= d
            r += d
    if not np.all(np.isfinite(h)) or abs(np.sum(h) - s) > 1e-9 * max(1, n):
        return None
    return h

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(np.sum(h) - n / 2.0) <= 1e-9 * max(1, n))

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if corr.shape != (2 * n - 1,) or not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr))

def helmert_fwd(x):
    d = len(x)
    n = d + 1
    if d == 0:
        return np.zeros(1)
    b = np.empty(d)
    for j in range(d):
        b[j] = x[j] / math.sqrt((j + 1) * (j + 2))
    T = np.zeros(n)
    T[:d] = np.cumsum(b[::-1])[::-1]
    out = np.zeros(n)
    out[0] = T[0]
    for i in range(1, n):
        out[i] = T[i] - i * b[i - 1]
    return out

def helmert_bwd(v):
    n = len(v)
    d = n - 1
    if d == 0:
        return np.zeros(0)
    ps = np.cumsum(v)
    out = np.empty(d)
    for j in range(d):
        out[j] = (ps[j] - v[j + 1]) * math.sqrt((j + 1) / (j + 2))
    return out
