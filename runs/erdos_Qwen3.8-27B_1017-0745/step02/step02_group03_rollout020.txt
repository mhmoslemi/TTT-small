    if not np.all(np.isfinite(z)):

        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)

            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            if 1.0 - h[idx] <= 1e-15:
                break
            h[idx] += min(r, 1.0 - h[idx])
        else:
            idx = int(np.argmax(h))
            if h[idx] <= 1e-15:
                break
            h[idx] += max(r, -h[idx])
    h = np.clip(h, 0.0, 1.0)

    return h

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return float('inf')
    if abs(np.sum(h) - n / 2.0) > 1e-8 * max(1, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_vals_grads(h, n):
    L = 2 * n - 1
    F = np.zeros(L)
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        i_lo = max(0, -d)
        i_hi = min(n, n - d)
        idx = np.arange(i_lo, i_hi)
        jdx = idx + d
        F[row] = float(np.dot(h[idx], 1.0 - h[jdx]))
        G[row, idx] += (1.0 - h[jdx])
        G[row, jdx] -= h[idx]
    return F, G

def repair_step(h, delta, r, n):
    # Clip delta to bounds
    d_lo = np.maximum(-r, -h)
    d_hi = np.minimum(r, 1.0 - h)
    delta = np.clip(delta, d_lo, d_hi)
    # Fix sum
    resid = -float(np.sum(delta))
    if resid > 0:
        cap = d_hi - delta
        total = float(np.sum(cap))
        if total < 1e-14:
            return None
        delta += cap * (resid / total)
    else:
        cap = delta - d_lo
        total = float(np.sum(cap))
        if total < 1e-14:
            return None
        delta -= cap * ((-resid) / total)
    # Final clip and residual fix
    delta = np.clip(delta, d_lo, d_hi)
    resid2 = -float(np.sum(delta))
    if abs(resid2) > 1e-12:
        if resid2 > 0:
            idx = int(np.argmax(d_hi - delta))
            delta[idx] += min(resid2, d_hi[idx] - delta[idx])
        else:
            idx = int(np.argmax(delta - d_lo))
            delta[idx] += max(resid2, d_lo[idx] - delta[idx])
    if not np.all(np.isfinite(delta)):
        return None
    return delta
