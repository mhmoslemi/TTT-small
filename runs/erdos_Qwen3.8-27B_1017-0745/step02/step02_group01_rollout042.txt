    for i in range(L):
        for j in range(i + 1, L):
            A = a[i] - a[j]
            B = b[i] - b[j]
            C = c[i] - c[j]
            append_roots(A, B, C, lo, hi, cands)
    cands.sort()
    dedup = [cands[0]]
    for t in cands[1:]:
        if t - dedup[-1] > 1e-13 * max(1.0, abs(t)):
            dedup.append(t)
    M = len(dedup)
    if M == 0:
        return h.copy(), cur_score
    tv = np.array(dedup, dtype=np.float64)
    t2 = tv * tv
    vals = a[None, :] * t2[:, None] + b[None, :] * tv[:, None] + c[None, :]
    if not np.all(np.isfinite(vals)):
        return h.copy(), cur_score
    pred = vals.max(axis=1)
    order = np.argsort(pred)[:8]
    trial_idx = set(order.tolist())
    for idx in range(min(3, M)):
        trial_idx.add(idx)
    trial_idx.add(M // 2)
    trial_idx.add(M - 1)
    best_s, best_z = cur_score, h.copy()
    for ti in sorted(trial_idx):
        if ti >= M:
            continue
        t = dedup[ti]
        z = project_capped(h + t * v, n)
        if z is None:
            continue
        sz = exact_score(z, n)
        if sz < best_s - 1e-12:
            best_s, best_z = sz, z.copy()
    return best_z, best_s

def refine_chain(start, n, directions, budget, gb_h, gb_s):
    cur = project_capped(start, n)
    if cur is None:
        return gb_h, gb_s
    cs = exact_score(cur, n)
    if not np.isfinite(cs):
        return gb_h, gb_s
    if cs < gb_s - 1e-12:
        gb_s, gb_h = cs, cur.copy()
    nd = len(directions)
    used = 0
    stagnant = 0
    sweep = 0
    while used < budget:
        fwd = (sweep % 2 == 0)
        improved = False
        for i in range(nd):
            if used >= budget:
                break
            di = i if fwd else (nd - 1 - i)
            used += 1
            try:
                nz, ns = line_search(cur, cs, directions[di], n)
            except Exception:
                continue
            if ns < cs - 1e-12:
                cur, cs = nz, ns
                improved = True
                if cs < gb_s - 1e-12:
                    gb_s, gb_h = cs, cur.copy()
        sweep += 1
        if improved:
            stagnant = 0
        else:
            stagnant += 1
            if stagnant >= 2:
                break
    return gb_h, gb_s

def build_starts(fallback, n, rng, count):
    pool = []
    def add(z):
        if len(pool) >= count:
            return
        p = project_capped(z, n)
        if p is None:
            return
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-8:
                return
        pool.append(p)
    add(fallback)
    add(fallback[::-1])
    add(1.0 - fallback[::-1])
    add(1.0 - fallback)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    add(blk)
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    add(alt)
    for k in (1, 2):
        add(0.5 + 0.35 * np.cos(2.0 * math.pi * k * np.arange(n) / n))
    add(fallback + rng.normal(0, 0.03, n))
    add(fallback + rng.normal(0, 0.10, n))
    add(rng.uniform(0, 1, n))
    return pool

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 0 and p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None or n < 1:
        n = 47
        try:
            v = int(kwargs.get('n_points', 47))
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if parent is not None:
        proj = project_capped(parent, n)
        fallback = proj if proj is not None else np.full(n, 0.5)
    else:
        fallback = np.full(n, 0.5)
    if not is_feasible(fallback, n):
        fallback = np.full(n, 0.5)
    fb_score = exact_score(fallback, n)
    if n == 1:
        return fallback, fb_score, 1
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    gb_h, gb_s = fallback.copy(), fb_score
    if n > 64:
        return fallback, fb_score, n
    directions = build_directions(n)
    starts = build_starts(fallback, n, rng, 10)
    scored = []
    for i, st in enumerate(starts):
        s = exact_score(st, n)
        if s < gb_s - 1e-12:
            gb_s, gb_h = s, st.copy()
        scored.append((s, i, st))
    scored.sort(key=lambda x: x[0])
    selected = [x[2] for x in scored[:3]]
    if not any(np.allclose(st, fallback, atol=1e-10) for st in selected):
        selected.append(fallback)
    B = min(276, max(n - 1, int(276 * (47.0 / n) ** 2)))
    nc = min(len(selected), 3, max(1, B // max(1, n - 1)))
    per = B // nc
    rem = B % nc
    for ci, st in enumerate(selected[:nc]):
        bud = per + (1 if ci < rem else 0)
        if bud <= 0:
            continue
        try:
            gb_h, gb_s = refine_chain(st, n, directions, bud, gb_h, gb_s)
        except Exception:
            pass
    final = project_capped(gb_h, n)
    if final is None or not is_feasible(final, n):
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score + 1e-12:
        final, fs = fallback, fb_score
    return final, float(fs), int(n)
