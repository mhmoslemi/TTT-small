def project_capped(z, n):
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0

            r -= d

        if len(archive) > P:
            keep = rng.choice(len(archive), size=P, replace=False)
            archive = [archive[int(k)] for k in keep]
        if F_succ:
            mu_F = float(np.clip(0.9 * mu_F + 0.1 * sum(f * f for f in F_succ) / sum(F_succ), 0.05, 1.0))
            mu_CR = float(np.clip(0.9 * mu_CR + 0.1 * float(np.mean(CR_succ)), 0.0, 1.0))
        if gen_improved:
            stall = 0
        else:
            stall += 1
        if stall >= stall_lim:
            if restarts >= max_restart:
                break
            si = np.argsort(scores)
            half = max(1, int(math.ceil(P / 2)))
            keep_idx = [int(x) for x in si[:half]]
            found = any(np.max(np.abs(pop[k] - best_h)) < 1e-10 for k in keep_idx)
            if not found:
                keep_idx[half - 1] = int(np.argmax(scores))
            new_pop = [pop[k].copy() for k in keep_idx]
            new_sc = [float(scores[k]) for k in keep_idx]
            rs = (0.04, 0.12, 0.25)
            fidx = 0
            while len(new_pop) < P and fidx < 5 * P:
                sc = rs[fidx % 3]
                tr = best_h + rng.normal(0, sc, n)
                tp = project_capped(tr, n)
                ok = tp is not None and is_feasible(tp, n)
                if ok:
                    dup = any(np.max(np.abs(tp - x)) < 1e-8 for x in new_pop)
                    if not dup:
                        new_pop.append(tp.copy())
                        new_sc.append(exact_score(tp, n))
                        fidx += 1
                        continue
                fidx += 1
            while len(new_pop) < P:
                pi = int(rng.integers(len(new_pop)))
                new_pop.append(new_pop[pi].copy())
                new_sc.append(new_sc[pi])
            pop = new_pop
            scores = np.array(new_sc, dtype=np.float64)
            archive = []
            mu_F, mu_CR = 0.55, 0.85
            stall = 0
            restarts += 1
        gen += 1
    return best_h, best_s

def use_explore_check(gen, n_explore):
    return gen < n_explore

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 1 and p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None:
        n = 47
    uni = np.full(n, 0.5)
    if parent is not None:
        parent = project_capped(parent, n)
        if parent is None or not is_feasible(parent, n):
            parent = None
    if n == 1:
        return uni, exact_score(uni, 1), 1
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 64:
        P_t, max_gen, stall_lim, max_restart = 64, 100, 18, 2
    elif n <= 128:
        P_t, max_gen, stall_lim, max_restart = 48, 70, 14, 2
    elif n <= 256:
        P_t, max_gen, stall_lim, max_restart = 32, 40, 10, 1
    else:
        P_t, max_gen, stall_lim, max_restart = 20, 20, 7, 1
    pool = build_pool(parent, n, rng)
    pool.sort(key=pool_key)
    pop, seen = [], set()
    for v, s in pool:
        key = tuple(np.round(v, 3).astype(np.float32))
        if key not in seen and len(pop) < P_t:
            pop.append(v.copy())
            seen.add(key)
    while len(pop) < P_t:
        pi = len(pop) % max(1, len(pop))
        tr = pop[pi].copy() + rng.normal(0, 0.02, n)
        tp = project_capped(tr, n)
        pop.append(tp if tp is not None else np.full(n, 0.5))
    scores = np.array([exact_score(h, n) for h in pop], dtype=np.float64)
    bi = int(np.argmin(scores))
    best_h, best_s = pop[bi].copy(), float(scores[bi])
    if parent is not None:
        ps = exact_score(parent, n)
        if ps < best_s:
            best_s, best_h = ps, parent.copy()
    us = exact_score(uni, n)
    if us < best_s:
        best_s, best_h = us, uni.copy()
    bh, bs = de_search(pop, scores, n, rng, max_gen, stall_lim, max_restart)
    if bs < best_s:
        best_s, best_h = bs, bh
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = np.clip(best_h, 0.0, 1.0)
        final = project_capped(final, n)
        if final is None:
            final = uni
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > best_s + 1e-10:
        final = uni
        fs = exact_score(final, n)
    return final, float(fs), int(n)
