import numpy as np
import math

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    if abs(np.sum(h) - n / 2.0) > 1e-9 * max(1, n):
        return False
    return True

def project_capped(z, n):
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    target = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = target - np.sum(h)
    for i in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            c = min(r, 1.0 - h[i])
            h[i] += c
            r -= c
        else:
            c = min(-r, h[i])
            h[i] -= c
            r += c
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(np.sum(h) - target) > 1e-10 * max(1, n):
        return None
    return h

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    q = np.correlate(h, 1.0 - h, mode='full')
    if q.shape[0] != 2 * n - 1 or not np.all(np.isfinite(q)):
        return float('inf')
    return float(np.max(q) * (2.0 / n))

def pair_bounds(h, i, j):
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    return lo, hi

def make_proposal(h, n, rng, step):
    p_pair = 0.8
    if rng.random() < p_pair:
        for _ in range(5):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            lo, hi = pair_bounds(h, i, j)
            if hi <= lo:
                continue
            if rng.random() < 0.75:
                a = max(lo, -step)
                b = min(hi, step)
                if b <= a:
                    continue
                t = rng.uniform(a, b)
            else:
                t = lo if rng.random() < 0.5 else hi
            if abs(t) < 1e-12:
                continue
            z = h.copy()
            z[i] += t
            z[j] -= t
            y = project_capped(z, n)
            if y is not None and np.max(np.abs(y - h)) > 1e-13:
                return y
        z = h + step * rng.standard_normal(n)
        return project_capped(z, n)
    else:
        z = h + step * rng.standard_normal(n)
        return project_capped(z, n)

def anneal_chain(start, n, rng, steps, best_h, best_score):
    current = start.copy()
    cur_score = exact_score(current, n)
    step = 0.20
    T0 = 0.002
    Tmin = 1e-6
    epochs = 3
    epoch_len = max(1, steps // epochs)
    accepted = 0
    attempted = 0
    for t in range(steps):
        phase = t % epoch_len
        if t > 0 and phase == 0:
            sigma = 0.08 + 0.12 * rng.random()
            z = project_capped(best_h + sigma * rng.standard_normal(n), n)

                current = z

            step = 0.20
            accepted = 0
            attempted = 0

            T = T0 * (Tmin / T0) ** (phase / epoch_len)
        else:
            T = T0 * (Tmin / T0) ** (phase / max(1, epoch_len - 1))
        prop = make_proposal(current, n, rng, step)
        if prop is None:
            continue
        s_new = exact_score(prop, n)
        if not np.isfinite(s_new):
            continue
        attempted += 1
        accept = False

            accept = True

            attempted = 0

                        base_s = s

                cur_s = base_s

def build_starts(fallback, n, rng, count):
    pool = [fallback.copy()]
    def try_add(cand):
        if len(pool) >= count:
            return
        for s in pool:
            if np.max(np.abs(cand - s)) < 1e-8:
                return
        proj = project_capped(cand, n)
        if proj is not None:
            pool.append(proj)
    try_add(fallback[::-1].copy())
    try_add(1.0 - fallback[::-1].copy())
    try_add(np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    try_add(blk)

    try_add(alt)
    for sd in (0.04, 0.12, 0.25):
        try_add(fallback + sd * rng.standard_normal(n))

        try_add(0.5 + 0.3 * np.cos(2.0 * np.pi * k * idx / n + ph))
    return pool[:count]
