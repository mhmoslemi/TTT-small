def project_capped(z, n):

    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None

    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for i in range(n):
        if abs(r) <= 2e-14 * max(1, n):
            break
        if r > 0:
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
        else:
            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
    if not np.all(np.isfinite(h)) or np.any(h < -1e-13) or np.any(h > 1.0 + 1e-13):

    if not np.all(np.isfinite(corr)):

    return float(np.max(corr))

import numpy as np
from math import sqrt, pi

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z)
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for i in range(n):
        if abs(r) <= 2e-14 * max(1, n):
            break
        if r > 0:
            adj = min(r, 1.0 - h[i])
            h[i] += adj
            r -= adj
        else:
            adj = min(-r, h[i])
            h[i] -= adj
            r += adj
    if not np.all(np.isfinite(h)) or np.any(h < -1e-13) or np.any(h > 1.0 + 1e-13):
        return None
    if abs(np.sum(h) - s) > 1e-10 * max(1, n):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(corr)):
        return float('inf')
    return float(np.max(corr))

def build_fourier(n):
    if n == 1:
        return np.zeros((1, 0)), [0]
    ncols = n - 1
    Phi = np.zeros((n, ncols), dtype=np.float64)
    labels = []
    col = 0
    idx = np.arange(n, dtype=np.float64)
    max_k = n // 2
    for k in range(1, max_k + 1):
        if n % 2 == 0 and k == n // 2:
            s = np.zeros(n)
            s[::2] = 1.0 / sqrt(n)
            s[1::2] = -1.0 / sqrt(n)
            Phi[:, col] = s
            labels.append(k)
            col += 1
            break
        Phi[:, col] = sqrt(2.0 / n) * np.cos(2.0 * pi * k * idx / n)
        labels.append(k)
        col += 1
        Phi[:, col] = sqrt(2.0 / n) * np.sin(2.0 * pi * k * idx / n)
        labels.append(k)
        col += 1
    return Phi, labels

def get_stages(n):
    F = (n - 1) // 2
    if n <= 64:
        cutoffs = [1, 2, 4, 8, 16, F]
    elif n <= 256:
        cutoffs = [1, 2, 4, 8, min(F, 12)]
    else:
        cutoffs = [1, 2, 4, 8, min(F, 8)]
    valid = []
    for c in sorted(set(cutoffs)):
        if 1 <= c <= F and c not in valid:

def scan_mode(z, j, u_j, Phi, n, R, grid_n, ref_n, ref_r):

    h_cur = project_capped(z, n)
    if h_cur is None:
        return None
