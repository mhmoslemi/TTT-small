        if iv.size > 0 and iv.size <= 999:
            if np.all(np.isfinite(iv)) and np.all(iv >= -1e-8) and np.all(iv <= 1.0 + 1e-8):
                n = iv.size
                parent = np.clip(iv, 0.0, 1.0).copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
    n = max(1, min(n, 999))
    try:
        sd = int(seed)
    except Exception:
        sd = 42
    rng = np.random.default_rng(abs(sd) % (2**32) if sd != 0 else 42)

    uniform = np.full(n, 0.5)
    best_h = uniform.copy()
    best_score = exact_score(uniform, n)
    if parent is not None:
        fp = project_capped(parent, n)
        if fp is not None:
            ps = exact_score(fp, n)
            if np.isfinite(ps) and ps < best_score:
                best_h, best_score = fp, ps

    starts = [best_h.copy()]
    if parent is not None:
        fp = project_capped(parent, n)
        if fp is not None:
            starts.append(fp)
        for sc in [0.025, 0.08, 0.15]:
            p = project_capped(parent + sc * rng.normal(0, 1, n), n)
            if p is not None:
                starts.append(p)
    idx = np.arange(n, dtype=np.float64)
    for freq in [1, 2, 3]:
        p = project_capped(0.5 + 0.45 * np.cos(2.0 * np.pi * freq * idx / n), n)
        if p is not None:
            starts.append(p)
    p = project_capped(rng.uniform(0, 1, n), n)
    if p is not None:
        starts.append(p)
    if parent is None:
        p = project_capped(0.5 + 0.12 * rng.normal(0, 1, n), n)
        if p is not None:
            starts.append(p)
    starts = dedup_pool(starts)

    if n <= 64:
        n_starts, iters, n_hops = 5, 60, 3
    elif n <= 128:
        n_starts, iters, n_hops = 4, 45, 2
    elif n <= 512:
        n_starts, iters, n_hops = 3, 20, 1
    else:
        n_starts, iters, n_hops = 2, 10, 1
    try:
        n_starts = max(1, min(int(kwargs.get('starts', n_starts)), 8))
    except Exception:
        pass
    try:
        iters = max(0, min(int(kwargs.get('max_iter', iters)), 100))
    except Exception:
        pass
    try:
        n_hops = max(0, min(int(kwargs.get('shake_rounds', n_hops)), 5))
    except Exception:
        pass
    starts = starts[:n_starts]

    for st in starts:
        try:
            best_h, best_score = local_optimize(st, n, iters, best_h, best_score)
        except Exception:
            pass

    sigma = 0.04
    for _ in range(n_hops):
        try:
            pert = project_capped(best_h + sigma * rng.normal(0, 1, n), n)
            if pert is None:
                continue
            bh, bs = local_optimize(pert, n, iters, best_h, best_score)
            if bs < best_score - 1e-12:
                best_h, best_score = bh, bs
                sigma = max(0.01, 0.75 * sigma)
            else:
                sigma = min(0.25, 1.5 * sigma)
        except Exception:
            pass

    final = project_capped(best_h, n)
    if final is None:
        final = uniform.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > best_score + 1e-12:
        final = best_h.copy()
        fs = exact_score(final, n)
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final, float(fs), int(n)
