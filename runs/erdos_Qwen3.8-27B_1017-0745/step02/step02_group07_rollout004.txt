    elif n <= 512:
        n_r, G, G_poly = 2, 35, 15
    else:
        n_r, G, G_poly = 2, 20, 10
    pool_size = max(16, n_r * 3)
    pool = build_pool(fb, n, rng, pool_size)
    scores = np.array([exact_score(pv, n) for pv in pool])
    bi = int(np.argmin(scores))
    best_h = pool[bi].copy()
    best_s = scores[bi]
    init_sig = max(0.04, min(0.35, 0.10 * math.sqrt(n / max(1, p_dim))))
    bases = select_bases(pool, scores, n_r, fb)
    for bv in bases:
        best_h, best_s = cma_es(bv, U, init_sig, G, rng, best_h, best_s, n)
    best_h, best_s = cma_es(best_h.copy(), U, max(0.02, init_sig * 0.5), G_poly, rng, best_h, best_s, n)
    fb_s = exact_score(fb, n)
    if best_s > fb_s + 1e-12:
        best_h, best_s = fb.copy(), fb_s
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fb
        best_s = fb_s
    else:
        fs = exact_score(final, n)
        if fs < best_s:
            best_s = fs
        if fs > fb_s + 1e-12:
            final = fb
            best_s = fb_s
    final = project_capped(final, n)
    if final is None:
        final = fb
    final_s = exact_score(final, n)
    return final.astype(np.float64), float(final_s), int(n)
