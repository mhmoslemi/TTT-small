                cur_score = float(np.max(q))
                improved = True
        hp = project_capped(h, n)
        if hp is not None:
            qp = lag_values(hp, n)
            sp = float(np.max(qp))
            if sp <= cur_score + 1e-10:
                h, q, cur_score = hp, qp, sp
        if not improved:
            break
    return h, cur_score

def run(seed=42, **kwargs):
    n = None; parent = None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= iv.size <= 999 and np.all(np.isfinite(iv)) and np.all(iv >= -1e-8) and np.all(iv <= 1.0 + 1e-8):
            n = iv.size; parent = np.clip(iv, 0.0, 1.0).copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
    n = max(2, min(n, 999))
    try:
        sd = int(seed)
    except Exception:
        sd = 42
    rng = np.random.default_rng(abs(sd) % (2**32))
    uniform = np.full(n, 0.5)
    fb = uniform.copy(); fb_s = exact_score(fb, n)
    if parent is not None:
        pp = project_capped(parent, n)
        if pp is not None:
            ps = exact_score(pp, n)
            if np.isfinite(ps) and ps < fb_s:
                fb, fb_s = pp, ps
    ij = np.arange(n, dtype=np.float64)
    starts = [fb.copy()]
    if parent is not None:
        for raw in [parent[::-1], 1.0 - parent, 1.0 - parent[::-1]]:
            p = project_capped(raw, n)
            if p is not None:
                starts.append(p)
        for st in [0.03, 0.08, 0.15]:
            p = project_capped(parent + rng.normal(0, st, n), n)
            if p is not None:
                starts.append(p)
    for freq in [1, 2, 3, 5]:
        p = project_capped(0.5 + 0.45 * np.cos(2.0 * np.pi * freq * ij / n), n)
        if p is not None:
            starts.append(p)
    p = project_capped(rng.uniform(0, 1, n), n)
    if p is not None:
        starts.append(p)
    dedup = []
    for s in starts:
        dup = False
        for d in dedup:
            if float(np.max(np.abs(s - d))) < 1e-8:
                dup = True; break
        if not dup:
            dedup.append(s)
    starts = dedup
    ms = 8 if n <= 64 else 4 if n <= 128 else 3
    ms = max(1, min(int(kwargs.get('starts', ms) if 'starts' in kwargs else ms), 10))
    msweeps = 8 if n <= 64 else 6 if n <= 128 else 4
    msweeps = max(0, min(int(kwargs.get('max_sweeps', msweeps) if 'max_sweeps' in kwargs else msweeps), 20))
    budget = n * (n - 1) // 2 if n <= 64 else min(4 * n, n * (n - 1) // 2)
    budget = max(0, min(int(kwargs.get('pair_budget', budget) if 'pair_budget' in kwargs else budget), n * (n - 1) // 2))
    kicks = 2
    kicks = max(0, min(int(kwargs.get('kick_rounds', kicks) if 'kick_rounds' in kwargs else kicks), 5))
    starts = starts[:ms]
    best, best_s = fb.copy(), fb_s
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_s - 1e-12:
            best, best_s = st.copy(), ss
        if msweeps <= 0:
            continue
        h_out, sc_out = cd_optimize(st, n, rng, msweeps, budget)
        sc_out = exact_score(h_out, n)
        if np.isfinite(sc_out) and sc_out < best_s - 1e-12:
            best, best_s = h_out.copy(), sc_out
    sigma = 0.04
    for _ in range(kicks):
        kd = project_capped(best + sigma * rng.standard_normal(n), n)
        if kd is None:
            sigma = min(0.25, 1.5 * sigma); continue
        h_out, sc_out = cd_optimize(kd, n, rng, msweeps, budget)
        sc_out = exact_score(h_out, n)
        if np.isfinite(sc_out) and sc_out < best_s - 1e-12:
            best, best_s = h_out.copy(), sc_out
            sigma = max(0.01, 0.75 * sigma)
        else:
            sigma = min(0.25, 1.5 * sigma)
    final = project_capped(best, n)
    if final is None:
        final = fb.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > best_s + 1e-10:
        final = best.copy(); fs = best_s
    if not np.isfinite(fs):
        final = uniform.copy(); fs = exact_score(uniform, n)
    return final, float(fs), int(n)
