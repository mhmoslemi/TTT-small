        return hp
    return None

def metropolis_accept(cur_sc, prop_sc, T, rng):
    delta = prop_sc - cur_sc
    if delta <= 0:
        return True
    if T < 1e-15:
        return False
    return math.log(max(rng.random(), 1e-300)) < -delta / T

def swap_accept(si, sj, Ti, Tj, rng):
    ratio = (1.0 / Ti - 1.0 / Tj) * (si - sj)
    if ratio >= 0:
        return True
    return math.log(max(rng.random(), 1e-300)) < ratio

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= len(iv) <= 999 and np.all(np.isfinite(iv)):
            n, parent = len(iv), iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
        n = max(1, min(999, n))
    try:
        sd = int(seed) % (2 ** 32)
    except Exception:
        sd = 42
    if n <= 1:
        h1 = np.array([0.5])
        return h1, float(exact_score(h1, 1)), 1
    rng = np.random.default_rng(sd)
    uniform = np.full(n, 0.5)
    if parent is not None:
        fb = project_capped(parent, n)
        if fb is None:
            fb = uniform.copy()
    else:
        fb = uniform.copy()
    fb_score = exact_score(fb, n)
    if not np.isfinite(fb_score):
        fb = uniform.copy()
        fb_score = exact_score(fb, n)
    B_def = min(16000, max(120, 30000000 // (n * n)))
    try:
        B = max(120, min(20000, int(kwargs.get('eval_budget', B_def))))
        B = min(B, max(120, 30000000 // (n * n)), 16000)
    except Exception:
        B = B_def
    starts = []
    add_start(starts, fb, n)
    if parent is not None:
        add_start(starts, parent + rng.normal(0, 0.02, n), n)
        add_start(starts, parent + rng.normal(0, 0.08, n), n)
        add_start(starts, fb[::-1].copy(), n)
        add_start(starts, 1.0 - fb, n)
        add_start(starts, rng.uniform(0, 1, n), n)
        add_start(starts, fb + rng.normal(0, 0.05, n), n)
    else:
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        add_start(starts, blk, n)
        add_start(starts, np.where(np.arange(n) % 2 == 0, 1.0, 0.0), n)
        tt = np.arange(n)
        add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * tt / n), n)
        add_start(starts, 0.5 + 0.45 * np.cos(4.0 * np.pi * tt / n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
    while len(starts) < 6:
        add_start(starts, 0.25 + 0.5 * rng.uniform(0, 1, n), n)
    NREP = min(6, len(starts))
    temps_all = [0.0005, 0.002, 0.008, 0.032, 0.128, 0.512]
    loc_all = [0.002, 0.006, 0.018, 0.054, 0.162, 0.486]
    temps = temps_all[:NREP]
    loc_s = loc_all[:NREP]
    reps = []
    rep_sc = []
    for i in range(NREP):
        hi_int = 0.99 * starts[i] + 0.005
        hi_int = project_capped(hi_int, n)
        if hi_int is None:
            hi_int = starts[i].copy()
        reps.append(hi_int)
        rep_sc.append(exact_score(hi_int, n))
    best = fb.copy()
    best_score = fb_score
    for i in range(NREP):
        if np.isfinite(rep_sc[i]) and rep_sc[i] < best_score - 1e-12:
            best, best_score = reps[i].copy(), rep_sc[i]
    eval_count = 0
    sweep = 0
    stagnation = 0
    n_restarts = 0
    max_stag = max(30, min(200, B // max(1, NREP * 12)))
    max_sweeps = B + 1
    while eval_count < B and sweep < max_sweeps:
        prev_best = best_score
        for i in range(NREP):
            if eval_count >= B:
                break
            h_cur = reps[i]
            sc_cur = rep_sc[i]
            if not np.isfinite(sc_cur):
                sc_cur = exact_score(h_cur, n)
                rep_sc[i] = sc_cur
                if not np.isfinite(sc_cur):
                    continue
            eval_count += 1
            r = rng.random()
            if r < 0.35:
                hp = hit_and_run(h_cur, n, rng)
            else:
                hp = pair_proposal(h_cur, n, rng, loc_s[i])
            if hp is None:
                continue
            sc_p = exact_score(hp, n)
            if not np.isfinite(sc_p):
                continue
            if metropolis_accept(sc_cur, sc_p, temps[i], rng):
                reps[i] = hp
                rep_sc[i] = sc_p
                if sc_p < best_score - 1e-12:
                    best, best_score = hp.copy(), sc_p
        if sweep % 2 == 0:
            pairs = [(0, 1), (2, 3), (4, 5)]
        else:
            pairs = [(1, 2), (3, 4)]
        for (i, j) in pairs:
            if i < NREP and j < NREP:
                if swap_accept(rep_sc[i], rep_sc[j], temps[i], temps[j], rng):
                    reps[i], reps[j] = reps[j], reps[i]
                    rep_sc[i], rep_sc[j] = rep_sc[j], rep_sc[i]
        if best_score > prev_best - 1e-12:
            stagnation += 1
        else:
            stagnation = 0
        if stagnation >= max_stag and n_restarts < 2:
            n_restarts += 1
            if n_restarts == 1:
                cand = 0.5 * best + 0.5 * rng.uniform(0, 1, n)
            else:
                cand = rng.beta(0.7, 0.7, n)
            c2 = project_capped(cand, n)
            if c2 is not None:
                c2 = 0.99 * c2 + 0.005
                c2 = project_capped(c2, n)
                if c2 is not None:
                    reps[NREP - 1] = c2
                    eval_count += 1
                    rep_sc[NREP - 1] = exact_score(c2, n)
                    if np.isfinite(rep_sc[NREP - 1]) and rep_sc[NREP - 1] < best_score - 1e-12:
                        best, best_score = c2.copy(), rep_sc[NREP - 1]
            stagnation = 0
        sweep += 1
    final = project_capped(best, n)
    if final is None:
        final = fb.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final, fs = fb.copy(), fb_score
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final, float(fs), int(n)
