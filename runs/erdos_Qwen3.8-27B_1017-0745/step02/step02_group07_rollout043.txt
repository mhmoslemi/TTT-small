import numpy as np

def project_capped(z, n):
    z = np.clip(np.asarray(z, dtype=np.float64).ravel(), -1e6, 1e6)
    if z.shape != (n,):
        return None
    s = n / 2.0
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(4):
        r = s - np.sum(h)
        if abs(r) < 1e-14:
            break
        if r > 0:
            h[int(np.argmax(1.0 - h))] = min(h[int(np.argmax(1.0 - h))] + r, 1.0)
        else:
            h[int(np.argmax(h))] = max(h[int(np.argmax(h))] + r, 0.0)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    if abs(np.sum(h) - s) > 1e-9:
        return None
    return h

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(np.sum(h) - n / 2.0) <= 1e-9 * max(1, n))

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    try:
        v = float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))
        return v if np.isfinite(v) else float('inf')
    except Exception:
        return float('inf')

def line_coefs(h, d, n):
    dx = 2.0 / n
    nl = 2 * n - 1
    a = np.zeros(nl); b = np.zeros(nl); c = np.zeros(nl)
    a[0] = dx * np.sum(h * (1.0 - h))
    b[0] = dx * np.sum(d * (1.0 - 2.0 * h))
    c[0] = -dx * np.sum(d * d)
    for r in range(1, n):
        m = n - r
        h0, h1 = h[:m], h[r:]
        d0, d1 = d[:m], d[r:]
        a[r] = dx * np.sum(h1 * (1.0 - h0))
        b[r] = dx * np.sum(d1 * (1.0 - h0) - h1 * d0)
        c[r] = -dx * np.sum(d1 * d0)
        a[n + r - 1] = dx * np.sum(h0 * (1.0 - h1))
        b[n + r - 1] = dx * np.sum(d0 * (1.0 - h1) - h0 * d1)
        c[n + r - 1] = -dx * np.sum(d0 * d1)
    return a, b, c

def feasible_interval(h, d, n):
    lo_v, hi_v = -1e18, 1e18
    for i in range(n):
        di = d[i]
        if di > 1e-15:
            lo_v = max(lo_v, -h[i] / di)
            hi_v = min(hi_v, (1.0 - h[i]) / di)
        elif di < -1e-15:
            lo_v = max(lo_v, (1.0 - h[i]) / di)
            hi_v = min(hi_v, -h[i] / di)
    if not (np.isfinite(lo_v) and np.isfinite(hi_v)) or hi_v - lo_v < 1e-12:
        return None
    return lo_v, hi_v

def eval_env(a, b, c, t_arr):
    Q = a[None, :] + b[None, :] * t_arr[:, None] + c[None, :] * (t_arr[:, None] ** 2)
    return np.max(Q, axis=1)

def line_candidates(h, d, n, lo, hi):
    a, b, c = line_coefs(h, d, n)
    nl = 2 * n - 1
    cands = [lo, hi, 0.0]
    for j in range(nl):
        if abs(c[j]) > 1e-15:
            t = -b[j] / (2.0 * c[j])
            if lo - 1e-10 <= t <= hi + 1e-10:
                cands.append(max(lo, min(hi, t)))
    for i in range(nl):
        for j in range(i + 1, nl):
            A, B, C = c[i] - c[j], b[i] - b[j], a[i] - a[j]
            sc = max(1.0, abs(A), abs(B), abs(C))
            if abs(A) < 1e-12 * sc:
                if abs(B) > 1e-12 * sc:
                    t = -C / B
                    if lo - 1e-10 <= t <= hi + 1e-10:
                        cands.append(max(lo, min(hi, t)))
            else:
                disc = B * B - 4.0 * A * C
                if disc < 0 and disc > -1e-14 * (1.0 + B * B + abs(4.0 * A * C)):
                    disc = 0.0
                if disc >= 0:
                    sq = math.sqrt(disc)
                    q = -0.5 * (B + math.copysign(sq, B))
                    if abs(q) > 1e-15:
                        t1, t2 = q / A, C / q
                    else:
                        t1 = t2 = -B / (2.0 * A)
                    for t in (t1, t2):
                        if np.isfinite(t) and lo - 1e-10 <= t <= hi + 1e-10:
                            cands.append(max(lo, min(hi, t)))
    cands = np.array(cands)
    cands = cands[np.isfinite(cands)]
    if len(cands) == 0:
        return np.array([0.5 * (lo + hi)])
    cands = np.sort(cands)
    uq = [cands[0]]
    tol = 1e-9 * max(1.0, hi - lo)
    for v in cands[1:]:
        if v - uq[-1] > tol:
            uq.append(v)
    cands = np.array(uq)
    vals = np.empty(len(cands))
    for st in range(0, len(cands), 512):
        vals[st:st + 512] = eval_env(a, b, c, cands[st:st + 512])
    order = np.argsort(vals)
    sel = []
    for k in order:
        if len(sel) >= 4:
            break
        if all(abs(cands[k] - s) > tol for s in sel):
            sel.append(cands[k])
    return np.array(sel) if sel else np.array([0.5 * (lo + hi)])

def lag_grads(h, n, top_idx):
    dx = 2.0 / n
    grads = np.zeros((len(top_idx), n))
    for ii, ti in enumerate(top_idx):
        g = np.zeros(n)
        if ti == 0:
            g[:] += dx * (1.0 - 2.0 * h)
        else:
            if ti < n:
                r = ti
            else:
                r = ti - n + 1
            m = n - r
            g[r:] += dx * (1.0 - h[:m])
            g[:m] -= dx * h[r:]
            if ti >= n:
                g[:m] += dx * (1.0 - h[r:])
                g[r:] -= dx * h[:m]
        grads[ii] = g
    return grads

    dirs = []
    seen = set()

    for (i, j) in pairs:

        for ii in range(len(grads)):

    nkeep = 10 if n <= 64 else (5 if n <= 256 else 3)

        d = np.zeros(n)

def local_search(h0, s0, n, rng, nsweeps, kicks_max):
    h = h0.copy()
    s = s0
    kicks = 0
    for sw in range(nsweeps):
        improved = False
        dirs = make_dirs(h, n, rng)
        best_s, best_h = s, h
        for d in dirs:
            try:
                if not np.all(np.isfinite(d)):
                    continue
                if np.max(np.abs(d)) < 1e-15:
                    continue
                fi = feasible_interval(h, d, n)
                if fi is None:
                    continue
                lo, hi = fi
                t_cands = line_candidates(h, d, n, lo, hi)
                for t in t_cands:
                    if not np.isfinite(t):
                        continue
                    if abs(t) < 1e-14:
                        continue
                    h_new = h + t * d
                    h_new = project_capped(h_new, n)
                    if h_new is None or not is_feasible(h_new, n):
                        continue
                    if np.max(np.abs(h_new - h)) < 1e-10:
                        continue
                    sc = exact_score(h_new, n)
                    if sc < best_s - 1e-12:
                        best_s, best_h = sc, h_new.copy()
            except Exception:
                continue
        if best_s < s - 1e-12:
            h, s = best_h, best_s
            improved = True
        elif kicks < kicks_max:
            sig = 0.08 if kicks == 0 else 0.20
            kicks += 1
            z = h + sig * rng.normal(0, 1, n)
            h_new = project_capped(z, n)
            if h_new is not None and is_feasible(h_new, n):
                h = h_new
                s = exact_score(h, n)
    return h, s

def build_pool(fb, n, rng):
    pool = [fb.copy()]
    def add(v):
        if v is not None and is_feasible(v, n):
            if all(np.max(np.abs(v - p)) > 1e-9 for p in pool):
                pool.append(v.copy())
    add(project_capped(fb[::-1], n))
    add(project_capped(1.0 - fb, n))
    add(project_capped(1.0 - fb[::-1], n))
    add(project_capped(np.full(n, 0.5), n))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    add(project_capped(blk, n))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    add(project_capped(alt, n))
    add(project_capped(fb + 0.06 * rng.normal(0, 1, n), n))
    add(project_capped(fb + 0.18 * rng.normal(0, 1, n), n))
    for k in range(1, 4):
        for ph in (0.0, math.pi / 2):
            v = 0.5 + 0.4 * np.cos(2.0 * math.pi * k * np.arange(n) / n + ph)
            add(project_capped(v, n))
    add(project_capped(np.roll(fb, n // 3), n))
    add(project_capped(rng.uniform(0, 1, n), n))
    return pool
