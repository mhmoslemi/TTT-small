        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-8:
                dup = True
                break
        if not dup:
            pool.append(p)
    return pool[:count]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
                n = int(p.size)
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    fb = np.full(n, 0.5)
    if parent is not None:
        if len(parent) == n:
            pf = project_capped(parent, n)
            if pf is not None:
                fb = pf
        else:
            old_n = parent.size
            oc = (np.arange(old_n) + 0.5) / old_n
            nc = (np.arange(n) + 0.5) / n
            interp = np.interp(nc, oc, parent)
            pf = project_capped(interp, n)
            if pf is not None:
                fb = pf
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        return fb, exact_score(fb, n), 1
    if n <= 64:
        n_iter, n_starts, n_restarts = 80, 7, 3
    elif n <= 128:
        n_iter, n_starts, n_restarts = 45, 4, 2
    elif n <= 256:
        n_iter, n_starts, n_restarts = 25, 3, 1
    elif n <= 512:
        n_iter, n_starts, n_restarts = 12, 2, 1
    else:
        n_iter, n_starts, n_restarts = 6, 2, 0
    delta0 = 0.25
    gs = exact_score(fb, n)
    gb = fb.copy()
    starts = build_starts(fb, n, rng, n_starts)
    scored = []
    for st in starts:
        s = exact_score(st, n)
        if s < gs - 1e-12:
            gs, gb = s, st.copy()
        scored.append((s, st))
    scored.sort(key=_sort_key)
    for _, st in scored:
        try:
            xo, so = local_search(st, n, n_iter, delta0)
            if so < gs - 1e-12:
                gs, gb = so, xo.copy()
        except Exception:
            continue
    fb_score = exact_score(fb, n)
    if n <= 64:
        sds = [0.12, 0.22, 0.08] if gs > fb_score - 1e-5 else [0.04, 0.10, 0.18]
    elif n <= 256:
        sds = [0.10, 0.18] if gs > fb_score - 1e-5 else [0.04, 0.08]
    else:
        sds = [0.08]
    sds = sds[:n_restarts]
    for sd in sds:
        try:
            pert = project_capped(gb + rng.normal(0.0, sd, n), n)
            if pert is None:
                continue
            xo, so = local_search(pert, n, n_iter, delta0)
            if so < gs - 1e-12:
                gs, gb = so, xo.copy()
        except Exception:
            continue
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb
    sf = exact_score(final, n)
    ss = exact_score(fb, n)
    if sf > ss:
        final, sf = fb, ss
    if not is_feasible(final, n):
        final = np.full(n, 0.5)
        sf = exact_score(final, n)
    return final, float(sf), int(n)
