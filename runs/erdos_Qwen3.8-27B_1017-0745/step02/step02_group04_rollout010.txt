        kw = {'verbose': False}
        if solver == 'CVXOPT':
            kw.update(abstol=1e-7, reltol=1e-7, feastol=1e-7, max_iters=200)
        prob.solve(solver=solver, **kw)
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None, None
        uv = np.asarray(u.value, dtype=np.float64).ravel()
        Yv = np.asarray(Y.value, dtype=np.float64)
        if not (np.all(np.isfinite(uv)) and np.all(np.isfinite(Yv))):
            return None, None
        S = 0.5 * (Yv + Yv.T) - np.outer(uv, uv)
        Pm = np.eye(n) - np.ones((n, n)) / n
        S = Pm @ S @ Pm
        S = 0.5 * (S + S.T)
        eig = np.linalg.eigvalsh(S)
        if not np.all(np.isfinite(eig)):
            return uv, None
        return uv, S
    except Exception:
        return None, None

def cma_es(center, n, rng, A, sigma, n_gen, lam):
    mu = project_capped(center, n)
    if mu is None:
        mu = np.full(n, 0.5)
    mu_s = exact_score(mu, n)
    best, best_s = mu.copy(), mu_s
    Pm = np.eye(n) - np.ones((n, n)) / n
    A = 0.5 * (A + A.T)
    Av, Ae = np.linalg.eigh(A)
    Av = np.clip(Av, 0.0, None)
    A = Ae @ np.diag(Av) @ Ae.T
    tr = np.trace(A)
    A = (n - 1.0) / tr * A if tr > 1e-10 else Pm
    stale = 0
    for g in range(n_gen):
        offspring, scores = [], []
        for _ in range(lam):
            _, Beig = np.linalg.eigh(A)
            B = Beig @ np.diag(np.sqrt(np.clip(np.eigvalsh(A), 0, None)))
            xi = rng.standard_normal(n)
            z = mu + sigma * (B @ xi)
            x = project_capped(z, n)
            if x is None or not is_feasible(x, n):
                x = mu + sigma * (rng.standard_normal(n) - np.mean(rng.standard_normal(n)))
                x = project_capped(x, n)
            if x is None:
                x, sc = mu.copy(), mu_s
            else:
                sc = exact_score(x, n)
            offspring.append(x)
            scores.append(sc)
        all_pts = [mu] + offspring
        all_scores = [mu_s] + scores
        order = np.argsort(all_scores)
        q = max(2, lam // 4)
        q = min(q, lam + 1)
        elite_idx = [order[j] for j in range(q)]
        weights = np.array([q - j for j in range(q)], dtype=float)
        weights /= weights.sum()
        new_mu = np.zeros(n)
        for w, idx in zip(weights, elite_idx):
            new_mu += w * all_pts[idx]
        new_mu = project_capped(new_mu, n)
        if new_mu is None:
            new_mu = mu.copy()
        nm_s = exact_score(new_mu, n)
        if nm_s < best_s - 1e-12:
            best, best_s = new_mu.copy(), nm_s
            stale = 0
        for idx in range(1, lam + 1):
            if all_scores[idx] < best_s - 1e-12:
                best, best_s = all_pts[idx].copy(), all_scores[idx]
                stale = 0
        if best_s < mu_s - 1e-12:
            stale = 0
        else:
            stale += 1
        # Covariance update
        E = np.zeros((n, n))
        for w, idx in zip(weights, elite_idx):
            d = all_pts[idx] - new_mu
            E += w * np.outer(d, d)
        E /= max(sigma * sigma, 1e-8)
        E = Pm @ E @ Pm
        E = 0.5 * (E + E.T)
        etr = np.trace(E)
        if etr > 1e-10:
            E = (n - 1.0) / etr * E
            A = Pm @ (0.8 * A + 0.2 * E + 1e-4 * Pm) @ Pm
            A = 0.5 * (A + A.T)
            Av2, Ae2 = np.linalg.eigh(A)
            Av2 = np.clip(Av2, 0.0, None)
            A = Ae2 @ np.diag(Av2) @ Ae2.T
            tr2 = np.trace(A)
            A = (n - 1.0) / tr2 * A if tr2 > 1e-10 else Pm
        # Sigma adaptation
        n_success = sum(1 for sc in scores if sc < mu_s - 1e-12)
        p_suc = n_success / lam if lam > 0 else 0
        if p_suc > 0.25:
            sigma = min(0.25, 1.12 * sigma)
        elif p_suc < 0.08:
            sigma = max(5e-4, 0.85 * sigma)
        mu = new_mu
        mu_s = nm_s
        if stale >= 12 and sigma <= 0.002:
            break
    return best, best_s
