    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0

def lag_data(h, n):
    dx = 2.0 / n
    L = np.correlate(h, 1.0 - h, mode='full') * dx
    nl = 2 * n - 1

    return float(np.max(c))

def lag_data(h, n):
    dx = 2.0 / n
    L = np.correlate(h, 1.0 - h, mode='full') * dx
    nl = 2 * n - 1
    G = np.zeros((nl, n), dtype=np.float64)
    for k in range(-(n - 1), n):
        row = k + n - 1
        start = max(0, -k)
        stop = min(n, n - k)
        i = np.arange(start, stop)
        j = i + k
        G[row, j] += dx * (1.0 - h[i])
        G[row, i] -= dx * h[j]
    if not np.all(np.isfinite(L)) or not np.all(np.isfinite(G)):
        return None, None
    return L, G
