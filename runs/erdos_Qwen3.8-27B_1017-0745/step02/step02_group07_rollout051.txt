import numpy as np
import math
import cvxpy as cp

def project_capped(z, n):
    z = np.clip(np.asarray(z, dtype=np.float64).ravel(), -1e6, 1e6)
    if z.shape != (n,):
        return None
    s = n / 2.0
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(3):
        if abs(r) < 1e-14:
            break
        if r > 0:
            for i in range(n):
                if r <= 1e-14:
                    break
                cap = 1.0 - h[i]
                add = min(r, cap)
                h[i] += add
                r -= add
        else:
            for i in range(n):
                if r >= -1e-14:
                    break
                cap = h[i]
                sub = min(-r, cap)
                h[i] -= sub
                r += sub
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    val = np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n))
    if not np.isfinite(val):
        return float('inf')
    return float(val)

def lag_data(h, n):
    dx = 2.0 / n
    nlag = 2 * n - 1
    q = np.zeros(nlag)
    G = np.zeros((nlag, n))
    m0 = int(nlag // 2)
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0] = dx * (1.0 - 2.0 * h)
    for r in range(1, n):
        m = n - r
        idx_p = m0 + r - 1
        idx_m = m0 + n - r
        q[idx_p] = dx * np.sum(h[:m] * (1.0 - h[r:]))
        q[idx_m] = dx * np.sum(h[r:] * (1.0 - h[:m]))
        G[idx_p, :m] += dx * (1.0 - h[r:r + m])
        G[idx_p, r:] += -dx * h[:m]
        G[idx_m, :m] += -dx * h[r:r + m]
        G[idx_m, r:] += dx * (1.0 - h[:m])
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None, None
    return q, G

def try_add(pool, v, th=1e-8):
    if v is None:
        return
    for s in pool:
        if np.max(np.abs(v - s)) < th:
            return
    pool.append(v)

    pool = []
    try_add(pool, fb.copy())
    try_add(pool, project_capped(fb[::-1], n))
    try_add(pool, project_capped(1.0 - fb, n))

    try_add(pool, np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:

    try_add(pool, project_capped(blk, n))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    try_add(pool, project_capped(alt, n))
    p = rng.permutation(n)
    rv = np.zeros(n)
    rv[p[:n // 2]] = 1.0
    if n % 2:
        rv[p[n // 2]] = 0.5
    try_add(pool, project_capped(rv, n))
    for sd in (0.06, 0.18):
        try_add(pool, project_capped(fb + rng.normal(0, sd, n), n))
    try_add(pool, project_capped(rng.uniform(0, 1, n), n))
    for k in range(1, 4):

            try_add(pool, project_capped(cv, n))
    try_add(pool, project_capped(np.roll(fb, n // 3), n))
    try_add(pool, project_capped(np.roll(fb, n // 2), n))
    while len(pool) < 4:
        pool.append(pool[0].copy())
    return pool[:16]

def build_qp(n):
    dx = 2.0 / n
    L = 2.0 * dx
    d = cp.Variable(n)
    u = cp.Variable()
    nlag = 2 * n - 1
    Qp = cp.Parameter(nlag)
    Gp = cp.Parameter((nlag, n))
    hp = cp.Parameter(n)
    obj = cp.Minimize(u + (L / 2.0) * cp.sum_squares(d))
    cons = [Gp @ d + Qp <= u * np.ones(nlag), cp.sum(d) == 0, d >= -hp, d <= 1.0 - hp]
    prob = cp.Problem(obj, cons)
    return d, Qp, Gp, hp, prob

    for s in ('ECOS', 'MOSEK', 'GUROBI', 'CVXOPT'):
        if cp.installed_solvers() and s in cp.installed_solvers():
            return s
    return None
