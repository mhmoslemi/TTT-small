    if h.shape != (n,) or not np.all(np.isfinite(h)): return float('inf')

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_derivs(h, n):
    L = 2 * n - 1
    F = np.zeros(L)
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        lo_d = max(0, -d)
        hi_d = min(n, n - d)
        idx = np.arange(lo_d, hi_d)
        jdx = idx + d
        F[row] = float(np.dot(h[idx], 1.0 - h[jdx]))
        G[row, idx] += (1.0 - h[jdx])
        G[row, jdx] -= h[idx]
    return F, G

    n = len(h)

    ub = np.minimum(r, 1.0 - h)
    delta = np.clip(delta, lb, ub)

    slack_dn = delta - lb
    if resid > 0:
        tot = slack_up.sum()
        if tot <= 1e-15: return None
        delta += (resid / tot) * slack_up
    elif resid < 0:
        tot = slack_dn.sum()
        if tot <= 1e-15: return None
        delta += (resid / tot) * slack_dn

                i = int(np.argmax(ub - delta))
                add = min(resid, ub[i] - delta[i])
                delta[i] += add; resid -= add
            else:
                i = int(np.argmax(delta - lb))
                rem = min(-resid, delta[i] - lb[i])
                delta[i] -= rem; resid += rem
            if abs(resid) < 1e-13: break

    proj = project_capped(trial, n)
    if proj is None: return None
    if float(np.max(np.abs(proj - trial))) > r + 1e-8: return None

    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if abs(hi - lo) < 1e-15: break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s: lo = mid
        else: hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - h.sum()
    for _ in range(n + 2):
        if abs(r) < 1e-13 * max(1, n): break
        if r > 0:
            cap = 1.0 - h; tot = cap.sum()
            if tot <= 1e-15: break
            h += (r / tot) * cap
        else:
            tot = h.sum()
            if tot <= 1e-15: break
            h += (r / tot) * h
        h = np.clip(h, 0.0, 1.0)
        r = s - h.sum()
    if not np.all(np.isfinite(h)) or abs(h.sum() - s) > 1e-9 * max(1, n):
        return None
    return h

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)): return float('inf')
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10): return float('inf')
    if abs(h.sum() - n / 2.0) > 1e-8 * max(1, n): return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_derivs(h, n):
    L = 2 * n - 1
    F = np.zeros(L); G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        lo_d, hi_d = max(0, -d), min(n, n - d)
        idx = np.arange(lo_d, hi_d); jdx = idx + d
        F[row] = float(np.dot(h[idx], 1.0 - h[jdx]))
        G[row, idx] += 1.0 - h[jdx]
        G[row, jdx] -= h[idx]
    return F, G

def repair_step(h, delta, r, n):
    lb = np.maximum(-r, -h); ub = np.minimum(r, 1.0 - h)
    delta = np.clip(delta, lb, ub)
    resid = -delta.sum()
    if resid > 1e-14:
        slack = ub - delta; tot = slack.sum()
        if tot <= 1e-15: return None
        delta += (resid / tot) * slack
    elif resid < -1e-14:
        slack = delta - lb; tot = slack.sum()
        if tot <= 1e-15: return None
        delta += (resid / tot) * slack
    delta = np.clip(delta, lb, ub)
    resid = -delta.sum()
    for _ in range(n):
        if abs(resid) < 1e-13: break
        if resid > 0:
            i = int(np.argmax(ub - delta)); a = min(resid, ub[i] - delta[i])
            delta[i] += a; resid -= a
        else:
            i = int(np.argmax(delta - lb)); a = min(-resid, delta[i] - lb[i])
            delta[i] -= a; resid += a
    if not np.all(np.isfinite(delta)): return None
    trial = h + delta
    proj = project_capped(trial, n)
    if proj is None: return None
    if float(np.max(np.abs(proj - trial))) > r + 1e-8: return None
    return proj

def lp_local_search(h0, n, max_iter, gb, gs):
    h = h0.copy(); cs = exact_score(h, n)
    if np.isfinite(cs) and cs < gs - 1e-12: gb[:] = h; gs = cs
    r = 0.08
    for _ in range(max_iter):
        F, G = lag_derivs(h, n)
        if not (np.all(np.isfinite(F)) and np.all(np.isfinite(G))): break
        L = 2 * n - 1
        c = np.zeros(n + 1); c[-1] = 1.0
        A_ub = np.zeros((L, n + 1)); A_ub[:, :n] = G; A_ub[:, n] = -1.0
        A_eq = np.zeros((1, n + 1)); A_eq[0, :n] = 1.0
        bounds = [(max(-r, -h[i]), min(r, 1.0 - h[i])) for i in range(n)]
        bounds.append((-np.inf, np.inf))
        try:
            res = linprog(c, A_ub=A_ub, b_ub=-F, A_eq=A_eq, b_eq=np.zeros(1),
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 5})
        except Exception:
            r *= 0.5
            if r < 1e-5: break
            continue
        if res is None or not res.success or res.x is None or len(res.x) != n + 1:
            r *= 0.5
            if r < 1e-5: break
            continue
        if not np.all(np.isfinite(res.x[:n])):
            r *= 0.5; continue
        trial = repair_step(h, res.x[:n], r, n)
        if trial is None:
            r *= 0.5
            if r < 1e-5: break
            continue
        ts = exact_score(trial, n)
        if not np.isfinite(ts):
            r *= 0.5; continue
        mx = float(np.max(np.abs(res.x[:n])))
        if ts < cs - 1e-12:
            mlin = float(np.max(F + G @ res.x[:n]))
            p = max(float(F.max()) - mlin, 0.0)
            rho = (cs - ts) / ((2.0 / n) * p) if p > 1e-14 else 0.0
            h = trial; cs = ts
            if cs < gs - 1e-12: gb[:] = h; gs = cs
            if rho >= 0.75 and mx >= 0.8 * r: r = min(0.25, 1.5 * r)
            elif rho < 0.25: r = max(1e-5, 0.5 * r)
            if mx < 1e-8: break
        else:
            r *= 0.5
        if r < 1e-5: break
    return gb, gs

def add_start(starts, vec, n, tol=1e-8):
    h2 = project_capped(vec, n)
    if h2 is None: return
    for ex in starts:
        if float(np.max(np.abs(h2 - ex))) < tol: return
    starts.append(h2)

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= len(iv) <= 999 and np.all(np.isfinite(iv)):
            n = len(iv); parent = iv.copy()
    except Exception: pass
    if n is None:
        try: n = int(kwargs.get('n_points', 47))
        except Exception: n = 47
        n = max(1, min(999, n))
    try: sd = abs(int(seed)) % (2 ** 32)
    except Exception: sd = 42
    rng = np.random.default_rng(sd)
    uniform = np.full(n, 0.5)
    fallback = project_capped(parent, n) if parent is not None else uniform.copy()
    if fallback is None: fallback = uniform.copy()
    fb_score = exact_score(fallback, n)
    if not np.isfinite(fb_score):
        fallback = uniform.copy(); fb_score = exact_score(fallback, n)
    if n <= 64: ns_def, mi_def = 7, 35
    elif n <= 128: ns_def, mi_def = 5, 20
    elif n <= 256: ns_def, mi_def = 2, 8
    else: ns_def, mi_def = 2, 0
    try: ns = max(1, min(int(kwargs.get('starts', ns_def)), 8))
    except Exception: ns = ns_def
    try: mi = max(0, min(int(kwargs.get('max_iter', mi_def)), 60))
    except Exception: mi = mi_def
    if n > 256: mi = 0
    starts = []
    add_start(starts, fallback, n)
    if parent is not None:
        add_start(starts, fallback[::-1], n)
        add_start(starts, 1.0 - fallback, n)
        add_start(starts, np.roll(fallback, max(1, n // 3)), n)
        add_start(starts, fallback + rng.normal(0, 0.035, n), n)
        add_start(starts, fallback + rng.normal(0, 0.09, n), n)
        add_start(starts, fallback + rng.normal(0, 0.15, n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
    else:
        blk = np.zeros(n); blk[:n // 2] = 1.0
        if n % 2 == 1: blk[n // 2] = 0.5
        add_start(starts, blk, n)
        add_start(starts, np.where(np.arange(n) % 2 == 0, 1.0, 0.0), n)
        tt = np.arange(n)
        add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * tt / n), n)
        add_start(starts, 0.5 + 0.45 * np.cos(4.0 * np.pi * tt / n), n)
        add_start(starts, rng.uniform(0, 1, n), n)
        add_start(starts, fallback + rng.normal(0, 0.06, n), n)
    starts = starts[:ns]
    gb = fallback.copy(); gs = fb_score
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < gs - 1e-12: gb = st.copy(); gs = ss
    if n <= 256 and mi > 0:
        for st in starts:
            try: gb, gs = lp_local_search(st, n, mi, gb.copy(), gs)
            except Exception: pass
        sigmas = [0.025, 0.07, 0.12] if n <= 128 else [0.05]
        for hh in range(min(3, len(sigmas))):
            try:
                pert = project_capped(gb + rng.normal(0, sigmas[hh], n), n)
                if pert is None: continue
                gb, gs = lp_local_search(pert, n, mi, gb.copy(), gs)
            except Exception: pass
    final = project_capped(gb, n)
    if final is None: final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_score:
        final = fallback.copy(); fs = fb_score
    if not np.isfinite(fs):
        final = uniform.copy(); fs = exact_score(final, n)
    return final, float(fs), int(n)
