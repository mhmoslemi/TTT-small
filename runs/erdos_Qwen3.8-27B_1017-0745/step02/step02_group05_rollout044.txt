def build_starts(base, n, rng, count):
    pool = [base.copy()]
    cands = [
        base[::-1].copy(),
        np.full(n, 0.5),
    ]

    i = np.arange(n, dtype=np.float64)

                break

            pool.append(p)

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z)
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for i in range(n):
        if abs(r) <= 2e-14:
            break
        if r > 0:
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
        else:
            a = min(-r, h[i])
            h[i] -= a
            r += a
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14):
        return None
    return h

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and
            abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n))

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(np.max(c)) if np.all(np.isfinite(c)) else float('inf')

def build_fourier(n):
    nm = n - 1
    F = (n - 1) // 2
    Phi = np.zeros((n, nm))
    labels = np.zeros(nm)
    idx = np.arange(n, dtype=np.float64)
    col = 0
    for k in range(1, F + 1):
        f = 2.0 * math.pi * k / n
        Phi[:, col] = math.sqrt(2.0 / n) * np.cos(f * idx)
        col += 1
        Phi[:, col] = math.sqrt(2.0 / n) * np.sin(f * idx)
        col += 1
        labels[col - 2] = k
        labels[col - 1] = k
    if n % 2 == 0:
        Phi[:, col] = np.where(np.arange(n) % 2 == 0, 1.0, -1.0) / math.sqrt(n)
        labels[col] = n / 2.0
        col += 1
    return Phi, labels

def scan_coeff(h, u_j, col_vec, n, R, gpts, nbas, nrounds):

            gs[i] = st

    mns = []

        ok = True
        if i > 0 and np.isfinite(gs[i - 1]) and gs[i] > gs[i - 1]:
            ok = False
        if i < gpts - 1 and np.isfinite(gs[i + 1]) and gs[i] > gs[i + 1]:
            ok = False
        if ok:
            mns.append((gs[i], i))
    if not mns:
        fin = [(gs[i], i) for i in range(gpts) if np.isfinite(gs[i])]
        if fin:
            mns.append(min(fin, key=lambda x: x[0]))
    mns.sort(key=lambda x: x[0])
    for _, gi in mns[:nbas]:
        if 0 < gi < gpts - 1:

        elif gi == 0:
