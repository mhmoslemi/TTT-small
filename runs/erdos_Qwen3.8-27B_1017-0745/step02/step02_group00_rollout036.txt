import numpy as np

def project_capped(z, n):
    s = n / 2.0
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if mid == lo or mid == hi:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for i in range(n):
        if abs(r) < 1e-13:
            break
        if r > 0:
            a = min(r, 1.0 - h[i]); h[i] += a; r -= a
        else:
            a = min(-r, h[i]); h[i] -= a; r += a
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-8:
        return None
    return h

def project_simplex(v):
    m = len(v)
    v = np.asarray(v, dtype=np.float64).ravel()
    if not np.all(np.isfinite(v)):
        return np.ones(m) / m
    u = np.sort(v)[::-1]
    cs = np.cumsum(u)
    rn = np.arange(1, m + 1, dtype=np.float64)
    cond = u - (cs - 1.0) / rn > 0
    if not np.any(cond):
        return np.ones(m) / m
    j = int(np.max(np.where(cond)[0]))
    lam = (cs[j] - 1.0) / (j + 1)
    w = np.maximum(v - lam, 0.0)
    rr = 1.0 - float(np.sum(w))
    if w.sum() > 0:
        w[int(np.argmax(w))] += rr
        w = np.maximum(w, 0.0)
    return w

def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return float('inf')
    if abs(float(np.sum(h)) - n / 2.0) > 1e-8:
        return float('inf')
    v = np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)
    return float(v) if np.isfinite(v) else float('inf')

def lag_vals(x, n):
    dx = 2.0 / n
    m = 2 * n - 1
    vals = np.zeros(m)
    for d in range(-(n - 1), n):
        lo, hi = max(0, -d), min(n, n - d)
        vals[d + n - 1] = dx * float(np.dot(x[lo:hi], 1.0 - x[lo + d:hi + d]))
    return vals

def dir_prod(x, p, n):
    dx = 2.0 / n
    m = 2 * n - 1
    vals = np.zeros(m)
    for d in range(-(n - 1), n):
        lo, hi = max(0, -d), min(n, n - d)
        vals[d + n - 1] = dx * (float(np.dot(p[lo:hi], 1.0 - x[lo + d:hi + d])) - float(np.dot(x[lo:hi], p[lo + d:hi + d])))
    return vals

def wgrad(x, w, n):
    dx = 2.0 / n
    g = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[d + n - 1]
        if abs(wd) < 1e-15:
            continue
        lo, hi = max(0, -d), min(n, n - d)
        g[lo:hi] += dx * wd * (1.0 - x[lo + d:hi + d])
        g[lo + d:hi + d] -= dx * wd * x[lo:hi]
    return g

def dual_solve(x, a, mu, w_init, n, iters):
    dx = 2.0 / n
    m = 2 * n - 1
    mu = max(mu, 2.0 * dx)
    L_D = max((2 * n - 1) * n * (4.0 / n) ** 2 / mu, 1e-8)
    w = project_simplex(w_init)
    z = w.copy()
    t = 1.0
    best_y, best_P = None, 1e18
    stall = 0
    for _ in range(iters):
        gb = wgrad(x, z, n)
        y = project_capped(x - gb / mu, n)
        if y is None:
            break
        p = y - x
        r = dir_prod(x, p, n)
        gradD = a + r
        if not np.all(np.isfinite(gradD)):
            break
        D_val = float(np.dot(z, a)) + float(np.dot(gb, p)) + 0.5 * mu * float(np.dot(p, p))
        P_val = float(np.max(a + r)) + 0.5 * mu * float(np.dot(p, p))
        if P_val < best_P:
            best_P = P_val
            best_y = y.copy()
        gap = max(0.0, P_val - D_val)
        if gap <= 1e-8 * max(1.0, abs(P_val)):
            break
        w_new = project_simplex(z + gradD / L_D)
        if w_new is None or not np.all(np.isfinite(w_new)):
            break
        if float(np.max(np.abs(w_new - w))) < 1e-10:
            stall += 1
            if stall >= 3:
                break
        else:
            stall = 0
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        z = w_new + ((t - 1.0) / t_new) * (w_new - w)
        w = w_new
        t = t_new
    if best_y is not None:
        return best_y, w
    return None, w

def mm_local(x, n, best, best_score, max_outer, dual_iter, mu_retries):
    dx = 2.0 / n
    x = x.copy()
    cur_score = exact_score(x, n)
    if not np.isfinite(cur_score):

    q = lag_vals(x, n)
    m = 2 * n - 1
    w = np.zeros(m)

    mu_cur = 2.0 * dx
    for _ in range(max_outer):
        accepted = False
        for r in range(mu_retries):
            mu_try = mu_cur * (2.0 ** r)
            y, w = dual_solve(x, q, mu_try, w, n, dual_iter)
            if y is None:
                continue
            if float(np.linalg.norm(y - x)) <= 1e-10 * max(1.0, np.sqrt(n)):
                break
            best_ls, best_th = cur_score, 0.0
            for j in range(9):

                zt = project_capped((1.0 - th) * x + th * y, n)
                if zt is None:

                s = exact_score(zt, n)
                if np.isfinite(s) and s < best_ls:

                    if s < best_score - 1e-12:

            if best_th > 0 and best_ls < cur_score - 1e-12:

                if x is None:

                    break

                q = lag_vals(x, n)
                mu_cur = max(2.0 * dx, 0.75 * mu_try) if best_th >= 0.99 else mu_try
                accepted = True
                break
        if not accepted:
            break
    return best, best_score

def polish(h, n, sweeps, best_score_ref):

    cur = exact_score(h, n)

    for _ in range(sweeps):

        for i in range(n):

        if not improved:
