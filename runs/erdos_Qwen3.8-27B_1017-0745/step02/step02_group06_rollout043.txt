def helmert_backward(v, n):
    """Compute Q^T v where Q is n x (n-1) Helmert basis."""
    d = n - 1
    if d <= 0:
        return np.zeros(0)
    P = np.cumsum(v)
    result = np.zeros(d)
    for j in range(d):
        denom = math.sqrt((j + 1) * (j + 2))
        result[j] = (P[j] - (j + 1) * v[j + 1]) / denom
    return result

    u = np.full(n, 0.5)
    m = helmert_backward(h_start - u, n)
    sigma0 = 0.15
    if full_cov:
        C = sigma0 * sigma0 * np.eye(d)
    else:
        v = np.full(d, sigma0 * sigma0)
    eta = 0.35
    smin2 = 0.0004  # 0.02^2
    smax2 = 0.25    # 0.5^2
    mu = lam // 2
    gbest2 = gbest.copy()
    gscore2 = gscore
    run_best = gscore2
    stall = 0
    eig_fail_count = 0

        zs = rng.standard_normal((lam // 2, d))
        # Apply L
        if full_cov:
            try:

                w, V = np.linalg.eigh(Cs)
                w = np.clip(w, smin2, smax2)

                C = sigma0 * sigma0 * np.eye(d)

                L = V @ np.diag(np.sqrt(w))
        else:
            v = np.clip(v, smin2, smax2)
            Lz = zs * np.sqrt(v)
        
        # Generate candidates

        for sc, xh, h in cands:

    if not is_feasible(h, n): return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if c.size != 2 * n - 1 or not np.all(np.isfinite(c)): return float('inf')
    return float(c.max())

def helmert_fwd(x, n):
    d = n - 1
    if d <= 0: return np.zeros(n)
    denom = np.sqrt(np.arange(2, n + 1, dtype=np.float64) * np.arange(1, n, dtype=np.float64))
    a = x / denom
    T = np.zeros(n)
    T[d-1] = a[d-1]
    for i in range(d-2, -1, -1): T[i] = T[i+1] + a[i]
    qx = np.zeros(n)
    qx[0] = T[0]
    for i in range(1, d+1): qx[i] = T[i] - math.sqrt(i / (i+1.0)) * x[i-1]
    return qx

    for j in range(d): res[j] = (P[j] - (j+1)*v[j+1]) / math.sqrt((j+1)*(j+2))
    return res

def project_capped(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    tgt = n / 2.0
    lo, hi = z.min() - 1.0, z.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > tgt:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = tgt - h.sum()
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            dd = min(r, 1.0 - h[idx]); h[idx] += dd; r -= dd
        else:
            dd = min(-r, h[idx]); h[idx] -= dd; r += dd
    if abs(r) > 1e-10 * max(1, n) or not np.all(np.isfinite(h)):

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if h.min() < -1e-12 or h.max() > 1.0 + 1e-12:
        return False
    return abs(h.sum() - n / 2.0) <= 1e-9 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if c.size != 2 * n - 1 or not np.all(np.isfinite(c)):
        return float('inf')
    return float(c.max())

def helmert_fwd(x, n):
    d = n - 1
    if d <= 0:
        return np.zeros(n)
    dn = np.sqrt(np.arange(2, n + 1, dtype=np.float64) * np.arange(1, n, dtype=np.float64))
    a = x / dn
    T = np.zeros(n)
    T[d - 1] = a[d - 1]
    for i in range(d - 2, -1, -1):
        T[i] = T[i + 1] + a[i]
    q = np.zeros(n)
    q[0] = T[0]
    for i in range(1, d + 1):
        q[i] = T[i] - math.sqrt(i / (i + 1.0)) * x[i - 1]
    return q

def helmert_bwd(v, n):
    d = n - 1
    if d <= 0:
        return np.zeros(0)
    P = np.cumsum(v)
    res = np.zeros(d)
    for j in range(d):
        res[j] = (P[j] - (j + 1) * v[j + 1]) / math.sqrt((j + 1) * (j + 2))
    return res
