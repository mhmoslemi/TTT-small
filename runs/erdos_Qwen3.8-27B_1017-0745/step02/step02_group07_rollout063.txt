    if z is None:
        return None
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def lag_values(h, n):
    dx = 2.0 / n
    q = np.empty(2 * n - 1)
    q[0] = dx * np.dot(h, 1.0 - h)
    for r in range(1, n):
        m = n - r
        q[2*r-1] = dx * np.dot(h[:m], 1.0 - h[r:])
        q[2*r] = dx * np.dot(h[r:], 1.0 - h[:m])
    return q

def lag_jac(h, n):
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    G[0] = dx * (1.0 - 2.0 * h)
    for r in range(1, n):
        m = n - r
        G[2*r-1, :m] += dx * (1.0 - h[r:])
        G[2*r-1, r:] -= dx * h[:m]
        G[2*r, :m] -= dx * h[r:]
        G[2*r, r:] += dx * (1.0 - h[:m])
    return G

def slsqp_obj(z, n):
    return z[n]

def slsqp_obj_jac(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def slsqp_ineq(z, n):
    h = z[:n]
    t = z[n]
    q = lag_values(h, n)
    return t - q

def slsqp_ineq_jac(z, n):
    h = z[:n]
    G = lag_jac(h, n)
    J = np.zeros((2*n-1, n+1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def slsqp_eq(z, n):
    return np.sum(z[:n]) - n / 2.0

def slsqp_eq_jac(z, n):
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def solve_slsqp(h0, n):
    sc = exact_score(h0, n)
    z0 = np.concatenate([h0, [sc + 1e-8]])
    mi = 220 if n <= 64 else 140
    bounds = [(0, 1)] * n + [(0, 2)]
    cons = [
        {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac, 'args': (n,)},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac, 'args': (n,)}
    ]
    try:
        res = minimize(slsqp_obj, z0, method='SLSQP', jac=slsqp_obj_jac,
                       args=(n,), bounds=bounds, constraints=cons,
                       options={'ftol': 1e-10, 'maxiter': mi, 'disp': False})
        if res is None or not hasattr(res, 'x') or res.x is None:
            return None
        z = np.asarray(res.x, dtype=np.float64)
        if z.shape != (n+1,) or not np.all(np.isfinite(z)):
            return None
        return project_capped(z[:n], n)
    except Exception:
        return None

def smooth_val(h, n, tau):
    q = lag_values(h, n)
    qm = np.max(q)
    e = np.exp((q - qm) / tau)
    return qm + tau * np.log(np.sum(e)), e / np.sum(e)

def smooth_grad(h, n, tau):
    q = lag_values(h, n)
    qm = np.max(q)
    e = np.exp((q - qm) / tau)
    p = e / np.sum(e)
    dx = 2.0 / n
    g = dx * p[0] * (1.0 - 2.0 * h)
    for r in range(1, n):
        m = n - r
        g[:m] += dx * p[2*r-1] * (1.0 - h[r:])
        g[r:] -= dx * p[2*r-1] * h[:m]
        g[:m] -= dx * p[2*r] * h[r:]
        g[r:] += dx * p[2*r] * (1.0 - h[:m])
    return g

    best_h = h0.copy()

import numpy as np
from scipy.optimize import minimize

def project_capped(z, n):
    if z is None:
        return None
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(n + 2):
        if abs(r) < 1e-14:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def lag_values(h, n):
    dx = 2.0 / n
    q = np.empty(2 * n - 1)
    q[0] = dx * np.dot(h, 1.0 - h)
    for r in range(1, n):
        m = n - r
        q[2*r-1] = dx * np.dot(h[:m], 1.0 - h[r:])
        q[2*r] = dx * np.dot(h[r:], 1.0 - h[:m])
    return q

def lag_jac(h, n):
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    G[0] = dx * (1.0 - 2.0 * h)
    for r in range(1, n):
        m = n - r
        G[2*r-1, :m] += dx * (1.0 - h[r:])
        G[2*r-1, r:] -= dx * h[:m]
        G[2*r, :m] -= dx * h[r:]
        G[2*r, r:] += dx * (1.0 - h[:m])
    return G

def slsqp_obj(z, n):
    return z[n]

def slsqp_obj_jac(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def slsqp_ineq(z, n):
    h = z[:n]
    t = z[n]
    return t - lag_values(h, n)

def slsqp_ineq_jac(z, n):
    h = z[:n]
    G = lag_jac(h, n)
    J = np.zeros((2*n-1, n+1))
    J[:, :n] = -G
    J[:, n] = 1.0
    return J

def slsqp_eq(z, n):
    return np.sum(z[:n]) - n / 2.0

def slsqp_eq_jac(z, n):
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

def solve_slsqp(h0, n):
    sc = exact_score(h0, n)
    z0 = np.concatenate([h0, [sc + 1e-8]])
    mi = 220 if n <= 64 else 140
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': slsqp_ineq, 'jac': slsqp_ineq_jac, 'args': (n,)},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac, 'args': (n,)}
    ]
    try:
        res = minimize(slsqp_obj, z0, method='SLSQP', jac=slsqp_obj_jac,
                       args=(n,), bounds=bounds, constraints=cons,
                       options={'ftol': 1e-10, 'maxiter': mi, 'disp': False})
        if res is None or not hasattr(res, 'x') or res.x is None:
            return None
        z = np.asarray(res.x, dtype=np.float64)
        if z.shape != (n+1,) or not np.all(np.isfinite(z)):
            return None
        return project_capped(z[:n], n)
    except Exception:
        return None

def smooth_val(h, n, tau):
    q = lag_values(h, n)
    qm = np.max(q)
    e = np.exp((q - qm) / tau)
    return qm + tau * np.log(np.sum(e))

def smooth_grad(h, n, tau):
    q = lag_values(h, n)
    qm = np.max(q)
    e = np.exp((q - qm) / tau)
    p = e / np.sum(e)
    dx = 2.0 / n
    g = dx * p[0] * (1.0 - 2.0 * h)
    for r in range(1, n):
        m = n - r
        g[:m] += dx * p[2*r-1] * (1.0 - h[r:])
        g[r:] -= dx * p[2*r-1] * h[:m]
        g[:m] -= dx * p[2*r] * h[r:]
        g[r:] += dx * p[2*r] * (1.0 - h[:m])
    return g
