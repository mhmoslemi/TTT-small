        gv = np.zeros(n)
        gv[0:m] = -dx * h[r:r + m]
        gv[r:n] += dx * (1.0 - h[0:m])
        G[row] = gv
        row += 1
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None, None
    return q, G

def solve_qp(n, q, G, h):
    import cvxpy as cp
    dx = 2.0 / n
    L = 2.0 * dx
    d = cp.Variable(n)
    u = cp.Variable()
    obj = cp.Minimize(u + (L / 2.0) * cp.sum_squares(d))
    cons = [
        G @ d + q <= u,
        cp.sum(d) == 0,
        d >= -h,
        d <= 1.0 - h,
    ]
    prob = cp.Problem(obj, cons)
    solvers = ['ECOS', 'CVXOPT', 'MOSEK', 'SCS']
    sol = None
    for sname in solvers:
        try:
            prob.solve(solver=sname, max_iters=300, tol_gap_abs=1e-8, tol_gap_rel=1e-8)
            if prob.status in ('optimal', 'optimal_inaccurate') and d.value is not None:
                dv = np.asarray(d.value, dtype=np.float64).ravel()
                if dv.shape == (n,) and np.all(np.isfinite(dv)):
                    sol = dv
                    break
        except Exception:
            continue
    return sol

def mm_optimize(h0, n, rng, max_iter):
    h = h0.copy()
    sc = exact_score(h, n)
    small_cnt = 0
    for it in range(max_iter):
        q, G = lag_data(h, n)
        if q is None:
            break
        d = solve_qp(n, q, G, h)
        if d is None:
            break
        if np.max(np.abs(d)) < 1e-9:
            break
        best_h, best_s = h.copy(), sc
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
            cand = project_capped(h + alpha * d, n)
            if cand is None:
                continue
            if np.max(np.abs(cand - h)) < 1e-10:
                continue
            cs = exact_score(cand, n)
            if cs < best_s - 1e-13:
                best_s = cs
                best_h = cand
        if best_h is not h and best_s < sc - 1e-13:
            red = sc - best_s
            h, sc = best_h, best_s
            if red < 1e-11:
                small_cnt += 1
            else:
                small_cnt = 0
            if small_cnt >= 4:
                break
        else:
            break
    return h, sc

def polish(h, n, rng, rounds=30, ndirs=8):
    sc = exact_score(h, n)
    delta = 0.15
    for _ in range(rounds):
        if delta < 1e-5:
            break
        found = False
        for _ in range(ndirs):
            v = rng.normal(0, 1, n)
            v -= np.mean(v)
            mx = np.max(np.abs(v))
            if mx < 1e-14:
                continue
            v /= mx
            for sgn in [1.0, -1.0]:
                t = sgn * delta
                cand = project_capped(h + t * v, n)
                if cand is None:
                    continue
                cs = exact_score(cand, n)
                if cs < sc - 1e-13:
                    h, sc = cand, cs
                    found = True
        if not found:
            delta *= 0.5
    return h, sc

def build_pool(fallback, n, rng, cap=16):
    pool = [fallback.copy()]

    def try_add(v):
        if v is None:
            return
        for s in pool:
            if np.max(np.abs(v - s)) < 1e-8:
                return
        pool.append(v)

    try_add(project_capped(fallback[::-1], n))
    try_add(project_capped(1.0 - fallback, n))
    try_add(project_capped(1.0 - fallback[::-1], n))
    try_add(project_capped(np.full(n, 0.5), n))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    try_add(project_capped(blk, n))
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
    try_add(project_capped(alt, n))
    p = rng.permutation(n)
    bv = np.zeros(n)
    bv[p[:n // 2]] = 1.0
    if n % 2:
        bv[p[n // 2]] = 0.5
    try_add(project_capped(bv, n))
    for sd in [0.06, 0.12, 0.18]:
        try_add(project_capped(fallback + rng.normal(0, sd, n), n))
    try_add(project_capped(rng.uniform(0, 1, n), n))
    for k in [1, 2, 3]:
        for ph in [0.0, math.pi / 2]:
            cv = 0.5 + 0.3 * np.cos(2.0 * math.pi * k * np.arange(n) / n + ph)
            try_add(project_capped(cv, n))
    try_add(project_capped(np.roll(fallback, n // 3), n))
    try_add(project_capped(np.roll(fallback, n // 2), n))
    while len(pool) < cap:
        pool.append(fallback.copy())
    return pool[:cap]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0.0, 1.0)
    except Exception:
        pass
    if n is None:
        n = 47
        try:
            vk = int(kwargs.get('n_points', 47))
            if 1 <= vk <= 999:
                n = vk
        except Exception:
            pass
    if n == 1:
        h1 = np.array([0.5])
        return h1, float(exact_score(h1, 1)), 1
    fb = project_capped(parent, n) if parent is not None else np.full(n, 0.5).copy()
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5).copy()
    fb_s = exact_score(fb, n)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n <= 64:
        n_starts, max_iter, n_pr = 5, 50, 40
    elif n <= 128:
        n_starts, max_iter, n_pr = 4, 35, 30
    elif n <= 256:
        n_starts, max_iter, n_pr = 3, 22, 20
    elif n <= 512:
        n_starts, max_iter, n_pr = 2, 14, 12
    else:
        n_starts, max_iter, n_pr = 1, 8, 8
    pool = build_pool(fb, n, rng)
    scores = np.array([exact_score(pv, n) for pv in pool])
    bi = int(np.argmin(scores))
    best_h = pool[bi].copy()
    best_s = float(scores[bi])
    order = np.argsort(scores)
    start_idx = [0]
    for j in order[1:]:
        if len(start_idx) >= n_starts:
            break
        ok = True
        for si in start_idx:
            if np.max(np.abs(pool[j] - pool[si])) < 0.04:
                ok = False
                break
        if ok:
            start_idx.append(int(j))
    qp_available = True
    try:
        import cvxpy as cp
        _ = cp.Variable(1)
    except Exception:
        qp_available = False
    if qp_available:
        for si in start_idx:
            h0 = pool[si].copy()
            try:
                hr, sr = mm_optimize(h0, n, rng, max_iter)
                if sr < best_s - 1e-12:
                    best_s = sr
                    best_h = hr.copy()
            except Exception:
                pass
    hr, sr = polish(best_h, n, rng, rounds=n_pr)
    if sr < best_s - 1e-12:
        best_s = sr
        best_h = hr.copy()
    best_h = project_capped(best_h, n)
    if best_h is None or not is_feasible(best_h, n):
        best_h, best_s = fb.copy(), fb_s
    else:
        best_s2 = exact_score(best_h, n)
        if best_s2 < fb_s - 1e-12:
            best_s = best_s2
        else:
            best_h, best_s = fb.copy(), fb_s
    final_sc = exact_score(best_h, n)
    if not np.isfinite(final_sc):
        final_sc = fb_s
        best_h = fb.copy()
    return best_h.astype(np.float64), float(final_sc), int(n)
