def project_capped(z, n):

    lo = float(np.min(z)) - 1.0

            lo = mid
        else:
            hi = mid

        wk = dx * w[k]
        grad[lo:hi] += wk * (1.0 - h[lo + d:hi + d])
        grad[lo + d:hi + d] -= wk * h[lo:hi]
    return float(F), grad
