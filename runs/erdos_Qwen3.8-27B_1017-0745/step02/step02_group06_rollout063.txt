def lag_jacobian(h, n):

    n_lags = 2 * n - 1
    J = np.zeros((n_lags, n))
    for k in range(n_lags):
        s = k - (n - 1)
        if s >= 0:
            # c = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
