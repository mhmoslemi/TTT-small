def project_capped(z, n):
    s = n / 2.0
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z.astype(np.float64), -1e12, 1e12)
    lo_t = float(np.min(z)) - 1.0
    hi_t = float(np.max(z))
    for _ in range(100):
        mid = 0.5 * (lo_t + hi_t)
        if abs(hi_t - lo_t) < 1e-15:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo_t = mid
        else:
            hi_t = mid
    lam = 0.5 * (lo_t + hi_t)
    h = np.clip(z - lam, 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(n + 2):
        if abs(r) < 1e-12 * max(1, n):
            break
        if r > 0:
            cap = 1.0 - h
            if cap.sum() <= 1e-15:
                break
            h += r * cap / cap.sum()
        else:
            cap = h.copy()
            if cap.sum() <= 1e-15:
                break
            h += r * cap / cap.sum()
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(np.sum(h) - s) > 1e-10 * max(1, n):
        return None
    return h

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return float('inf')
    if abs(np.sum(h) - n / 2.0) > 1e-9 * max(1, n):
        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_vals_grads(h, n):
    L = 2 * n - 1
    F = np.zeros(L)
    G = np.zeros((L, n))
    for d in range(-(n - 1), n):
        row = d + n - 1
        lo = max(0, -d)
        hi = min(n, n - d)
        F[row] = float(np.dot(h[lo:hi], 1.0 - h[lo + d:hi + d]))
        G[row, lo:hi] += 1.0 - h[lo + d:hi + d]
        G[row, lo + d:hi + d] -= h[lo:hi]
    if not np.all(np.isfinite(F)) or not np.all(np.isfinite(G)):
        return None, None
    return F, G

def repair_step(h, delta, r, n):
    lo_b = np.maximum(-r, -h)
    hi_b = np.minimum(r, 1.0 - h)
    delta = np.clip(delta, lo_b, hi_b)
    resid = -float(np.sum(delta))
    if abs(resid) > 1e-14:
        if resid > 0:
            slack = hi_b - delta
        else:
            slack = delta - lo_b
        tot = float(np.sum(slack))
        if tot > 1e-15:
            delta += resid * slack / tot
    delta = np.clip(delta, lo_b, hi_b)
    if not np.all(np.isfinite(delta)):
        return None
    h_new = np.clip(h + delta, 0.0, 1.0)
    proj = project_capped(h_new, n)
    if proj is None:
        return None
    if float(np.max(np.abs(proj - (h + delta)))) > r + 1e-8:
        return None
    return proj
