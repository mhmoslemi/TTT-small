def project_capped(z, n):

        else:
            hi = mid

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0+1e-12):
        return False
    if abs(np.sum(h) - n/2.0) > 1e-10*max(1, n):
        return False
    return True

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-12) or np.any(h > 1.0+1e-12):
        return float('inf')
    c = np.correlate(h, 1.0-h, mode='full') * (2.0/n)
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

        return False

    seen.append(h.copy())
    return True
