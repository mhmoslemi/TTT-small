import numpy as np

def project_capped(z, n):
    s = n / 2.0
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if mid == lo or mid == hi:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for i in range(n):
        if abs(r) < 1e-13:
            break
        if r > 0:
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
        else:
            a = min(-r, h[i])
            h[i] -= a
            r += a
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-8:
        return None
    return h

def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return float('inf')
    if abs(float(np.sum(h)) - n / 2.0) > 1e-8:
        return float('inf')
    val = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
    return val if np.isfinite(val) else float('inf')

def lag_values(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.zeros(L, dtype=np.float64)
    for d in range(-(n - 1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        q[d + n - 1] = dx * float(np.dot(h[lo:hi], 1.0 - h[lo + d:hi + d]))
    return q

def quadratic_roots(a, b, c):
    """Return real roots of a*t^2 + b*t + c = 0"""
    roots = []
    if abs(a) < 1e-14:
        if abs(b) > 1e-14:
            roots.append(-c / b)
        return roots
    disc = b * b - 4.0 * a * c
    if disc < -1e-12:
        return roots
    if disc < 0:
        disc = 0.0
    sq = math.sqrt(disc)
    # Cancellation-resistant
    if b >= 0:
        q_val = -0.5 * (b + sq)
    else:
        q_val = -0.5 * (b - sq)
    roots.append(q_val / a)
    if abs(c) > 1e-14:
        roots.append(c / q_val)
    return roots

def poly_roots_diff(a1, b1, c1, a2, b2, c2):
    """Roots of (a1-a2)t^2 + (b1-b2)t + (c1-c2) = 0"""
    return quadratic_roots(a1 - a2, b1 - b2, c1 - c2)

def build_upper_hull(lines):
    """lines: list of (intercept, slope). Returns list of (intercept, slope, start_t)."""
    if not lines:
        return []
    # Sort by slope
    lines_sorted = sorted(lines, key=lambda x: x[1])
    # Remove dominated lines (same slope, keep max intercept)
    unique = []
    for a, m in lines_sorted:
        if unique and unique[-1][1] == m:
            if a > unique[-1][0]:
                unique[-1] = (a, m)
        else:
            unique.append((a, m))
    lines_sorted = unique
    if len(lines_sorted) == 1:
        return [(lines_sorted[0][0], lines_sorted[0][1], float('-inf'))]
    hull = []
    for a, m in lines_sorted:
        if not hull:
            hull.append([a, m, float('-inf')])
            continue
        # Intersection with last hull line
        last = hull[-1]
        if m == last[1]:
            if a > last[0]:
                hull[-1] = [a, m, last[2]]
            continue
        t_cross = (last[0] - a) / (m - last[1])
        while len(hull) > 1 and t_cross <= hull[-1][2] + 1e-15:
            hull.pop()
            last = hull[-1]
            if m == last[1]:
                break
            t_cross = (last[0] - a) / (m - last[1])
        if len(hull) == 1 and m != hull[-1][1]:
            hull.append([a, m, t_cross])
        elif len(hull) > 1:
            hull.append([a, m, t_cross])
        elif m == hull[-1][1]:
            if a > hull[-1][0]:
                hull[-1] = [a, m, hull[-1][2]]
    return hull

def pair_line_search(h, q, n, i, j):
    """Find best t for transferring mass between h[i] and h[j]. Returns t or 0."""
    dx = 2.0 / n
    r = j - i
    L = max(-h[i], h[j] - 1.0)
    U = min(1.0 - h[i], h[j])
    if U - L < 1e-12:
        return 0.0
    
    N = 2 * n - 1
    special = {0, r, -r}
    
    # Compute b_d for all lags
    b = np.zeros(N, dtype=np.float64)
    for d in range(-(n - 1), n):
        gd_i = 0.0
        if 0 <= i + d < n:
            gd_i += (1.0 - h[i + d])
        if 0 <= i - d < n:
            gd_i -= h[i - d]
        gd_j = 0.0
        if 0 <= j + d < n:
            gd_j += (1.0 - h[j + d])
        if 0 <= j - d < n:
            gd_j -= h[j - d]
        b[d + n - 1] = dx * (gd_i - gd_j)
    
    # c_d
    c = np.zeros(N, dtype=np.float64)
    c[0 + n - 1] = -2.0 * dx
    if r >= -(n-1) and r <= n-1:
        c[r + n - 1] = dx
    if -r >= -(n-1) and -r <= n-1:
        c[-r + n - 1] = dx
    
    # Affine lines (all non-special lags)
    aff_lines = []
    for d in range(-(n - 1), n):
        idx = d + n - 1
        if idx in special:
            continue
        aff_lines.append((q[idx], b[idx]))
    
    # Build upper hull
    hull = build_upper_hull(aff_lines)
    
    # Special quadratics: (a, b_coeff, c_coeff) where value = a + b*t + c*t²
    quads = []
    for d in sorted(special):
        if d < -(n-1) or d > n-1:
            continue
        idx = d + n - 1
        quads.append((q[idx], b[idx], c[idx]))
    
    # Find intervals from hull
    if not hull:
        intervals = [(L, U, None)]
    else:

        for k in range(len(hull)):

            end = hull[k + 1][2] if k + 1 < len(hull) else float('inf')

            if te - ts > 1e-13:

        return 0.0
    
    best_t = 0.0

        # Polynomials on this interval: the line + quads
        # Each polynomial: (a, b_coeff, c_coeff)

        for qp in quads:
            polys.append(qp)

        cands = [ts, te]
        
        # Stationary points
        for (a, bc, cc) in polys:
            if abs(cc) > 1e-14:
                t_stat = -bc / (2.0 * cc)
                if ts - 1e-12 <= t_stat <= te + 1e-12:
                    cands.append(max(ts, min(te, t_stat)))
        
        # Pairwise intersections
        for pi in range(len(polys)):
            for pj in range(pi + 1, len(polys)):

                for rt in roots:

            val = 0.0
            for (a, bc, cc) in polys:

                if va > val:
                    val = va
            if val < best_val - 1e-12:

def project_capped(z, n):
    s = n / 2.0
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e6, 1e6)
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if mid == lo or mid == hi:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for i in range(n):
        if abs(r) < 1e-13:
            break
        if r > 0:
            a = min(r, 1.0 - h[i])
            h[i] += a
            r -= a
        else:
            a = min(-r, h[i])
            h[i] -= a
            r += a
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-8:
        return None
    return h

def exact_score(h, n):
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')
    if np.any(h < -1e-9) or np.any(h > 1.0 + 1e-9):
        return float('inf')
    if abs(float(np.sum(h)) - n / 2.0) > 1e-8:
        return float('inf')
    val = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
    return val if np.isfinite(val) else float('inf')

def lag_values(h, n):
    dx = 2.0 / n
    L = 2 * n - 1
    q = np.zeros(L, dtype=np.float64)
    for d in range(-(n - 1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        q[d + n - 1] = dx * float(np.dot(h[lo:hi], 1.0 - h[lo + d:hi + d]))
    return q

def quad_roots(a, b, c):
    if abs(a) < 1e-14:
        if abs(b) > 1e-14:
            return [-c / b]
        return []
    disc = b * b - 4.0 * a * c
    if disc < -1e-12:
        return []
    if disc < 0:
        disc = 0.0
    sq = math.sqrt(disc)
    if b >= 0:
        qv = -0.5 * (b + sq)
    else:
        qv = -0.5 * (b - sq)
    r1 = qv / a
    r2 = c / qv if abs(qv) > 1e-14 else float('inf')
    out = [r1]
    if np.isfinite(r2) and abs(r2 - r1) > 1e-12:
        out.append(r2)
    return out

def build_hull(lines):
    if not lines:
        return []
    sl = sorted(lines, key=lambda x: x[1])
    uniq = []
    for a, m in sl:
        if uniq and uniq[-1][1] == m:
            if a > uniq[-1][0]:
                uniq[-1] = (a, m)
        else:
            uniq.append((a, m))
    sl = uniq
    if len(sl) == 1:
        return [(sl[0][0], sl[0][1], float('-inf'))]
    hull = [[sl[0][0], sl[0][1], float('-inf')]]
    for a, m in sl[1:]:
        if m == hull[-1][1]:
            if a > hull[-1][0]:
                hull[-1][0] = a
            continue
        tc = (hull[-1][0] - a) / (m - hull[-1][1])
        while len(hull) > 1 and tc <= hull[-1][2] + 1e-14:
            hull.pop()
            tc = (hull[-1][0] - a) / (m - hull[-1][1])
        hull.append([a, m, tc])
    return hull

def pair_search(h, q, n, i, j):

    r = j - i
    Lo = max(-h[i], h[j] - 1.0)

        gi = 0.0

        if 0 <= i - d < n:

        if 0 <= j + d < n:
            gj += 1.0 - h[j + d]
        if 0 <= j - d < n:
            gj -= h[j - d]

    c = np.zeros(N, dtype=np.float64)
    c[n - 1] = -2.0 * dx
