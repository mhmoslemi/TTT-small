        for it in range(max_iter_per_t):
            res = lse_phi(h, n, theta)
            if res is None:
                break
            phi_h, p = res
            g = lag_grad(h, n, p)
            gm = float(np.max(np.abs(g)))
            if gm < 1e-12:
                break
            step_mx = 1.0
            for bt in range(24):
                y = project_capped(h - alpha * g, n)
                if y is None:
                    alpha *= 0.5
                    continue
                step = y - h
                step_mx = float(np.max(np.abs(step)))
                if step_mx < 1e-10:
                    break
                gd = float(np.dot(g, step))
                if gd > -1e-14:
                    alpha *= 0.5
                    continue
                res2 = lse_phi(y, n, theta)
                if res2 is None:
                    alpha *= 0.5
                    continue
                phi_y = res2[0]
                tol = 1e-12 * max(1.0, abs(phi_h))
                if phi_y <= phi_h + 1e-4 * gd + tol:
                    h = y
                    alpha = min(0.5, 1.15 * alpha)
                    break
                alpha *= 0.5
            sc = exact_score(h, n)
            if np.isfinite(sc) and sc < best_s - 1e-12:
                best_h[:] = h
                best_s = sc
            if step_mx < 1e-10:
                break
    return h, best_h.copy(), best_s

def _add_start(starts, v, n, tol=1e-8):
    h2 = project_capped(v, n)
    if h2 is None:
        return
    for ex in starts:
        if float(np.max(np.abs(h2 - ex))) < tol:
            return
    starts.append(h2)

def run(seed=42, **kwargs):
    n, parent = None, None
    try:
        iv = np.asarray(initial_h_values, dtype=np.float64).ravel()
        if 1 <= len(iv) <= 999 and np.all(np.isfinite(iv)):
            n = len(iv)
            parent = iv.copy()
    except Exception:
        pass
    if n is None:
        try:
            n = int(kwargs.get('n_points', 47))
        except Exception:
            n = 47
        n = max(1, min(999, n))
    try:
        sd = int(seed)
        if sd < 0:
            sd = 42
    except Exception:
        sd = 42
    rng = np.random.default_rng(sd)
    uniform = np.full(n, 0.5)
    if parent is not None:
        h0 = project_capped(parent, n)
        fallback = h0 if h0 is not None else uniform.copy()
    else:
        fallback = uniform.copy()
    fb_sc = exact_score(fallback, n)
    if not np.isfinite(fb_sc):
        fallback = uniform.copy()
        fb_sc = exact_score(fallback, n)
    if n <= 128:
        n_st, m_iter = 6, 60
    elif n <= 512:
        n_st, m_iter = 4, 25
    else:
        n_st, m_iter = 2, 10
    try:
        n_st = max(1, min(int(kwargs.get('starts', n_st)), 8))
    except Exception:
        pass
    try:
        m_iter = max(1, min(int(kwargs.get('max_iter', m_iter)), 120))
    except Exception:
        pass
    starts = []
    _add_start(starts, fallback, n)
    if parent is not None:
        _add_start(starts, fallback[::-1].copy(), n)
        _add_start(starts, 1.0 - fallback, n)
        _add_start(starts, np.roll(fallback, max(1, n // 3)), n)
        _add_start(starts, fallback + rng.normal(0, 0.03, n), n)
        _add_start(starts, fallback + rng.normal(0, 0.10, n), n)
        _add_start(starts, rng.uniform(0, 1, n), n)
    else:
        _add_start(starts, uniform, n)
        blk = np.zeros(n)
        blk[:n // 2] = 1.0
        if n % 2 == 1:
            blk[n // 2] = 0.5
        _add_start(starts, blk, n)
        alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
        _add_start(starts, alt, n)
        tt = np.arange(n)
        _add_start(starts, 0.5 + 0.45 * np.cos(2.0 * np.pi * tt / n), n)
        _add_start(starts, 0.5 + 0.45 * np.cos(4.0 * np.pi * tt / n), n)
        _add_start(starts, rng.uniform(0, 1, n), n)
    starts = starts[:n_st]
    best = fallback.copy()
    best_s = fb_sc
    for st in starts:
        ss = exact_score(st, n)
        if np.isfinite(ss) and ss < best_s - 1e-12:
            best, best_s = st.copy(), ss
    for st in starts:
        try:
            _, bh, bs = continuation_opt(st, n, best, best_s, m_iter)
            best, best_s = bh, bs
        except Exception:
            pass
    # Basin hops from current best
    hops = [0.025, 0.08] if n <= 128 else [0.05]
    for std in hops:
        pert = best + rng.normal(0, std, n)
        pp = project_capped(pert, n)
        if pp is None:
            continue
        ps = exact_score(pp, n)
        if np.isfinite(ps) and ps < best_s - 1e-12:
            best, best_s = pp.copy(), ps
        try:
            _, bh, bs = continuation_opt(pp, n, best, best_s, m_iter)
            best, best_s = bh, bs
        except Exception:
            pass
    # Fine refinement around best
    for fstd in [0.005, 0.002]:
        pert = best + rng.normal(0, fstd, n)
        pp = project_capped(pert, n)
        if pp is None:
            continue
        ps = exact_score(pp, n)
        if np.isfinite(ps) and ps < best_s - 1e-12:
            best, best_s = pp.copy(), ps
        try:
            _, bh, bs = continuation_opt(pp, n, best, best_s, m_iter)
            best, best_s = bh, bs
        except Exception:
            pass
    final = project_capped(best, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not np.isfinite(fs) or fs > fb_sc:
        final, fs = fallback.copy(), fb_sc
    if not np.isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final, float(fs), int(n)
