        oj = (np.arange(old_n) + 0.5) / old_n
        nj = (np.arange(n) + 0.5) / n
        parent = np.interp(nj, oj, parent, left=parent[0], right=parent[-1])
    fb = project_capped(parent, n) if parent is not None else np.full(n, 0.5).astype(np.float64)
    if fb is None:
        fb = np.full(n, 0.5).astype(np.float64)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    fs = exact_score(fb, n)
    gs, gb = fs, fb.copy()
    if n == 1:
        return fb, fs, 1
    taus = [0.03, 0.01, 0.004, 0.001, 0.0003, 0.00008, 0.00002]
    if n <= 64:
        n_starts, n_iters, n_polish, n_restarts = 6, 150, 60, 3
    elif n <= 128:
        n_starts, n_iters, n_polish, n_restarts = 4, 90, 40, 1
    elif n <= 256:
        n_starts, n_iters, n_polish, n_restarts = 3, 55, 30, 1
    else:
        n_starts, n_iters, n_polish, n_restarts = 2, 25, 20, 0
    starts = build_starts(fb, n, rng, n_starts)
    for st in starts:
        try:
            bh, bs = continuation_search(st, n, taus, n_iters)
            bh, bs = polish_subgrad(bh, n, n_polish)
            if bs < gs - 1e-12:
                gs, gb = bs, bh.copy()
        except Exception:
            continue
    if n_restarts > 0:
        improved = fs - gs
        if n <= 64:
            sds = (0.05, 0.10, 0.20) if improved < 2e-5 else (0.02, 0.04, 0.08)
        else:
            sds = (0.12, 0.05)
        for i in range(min(n_restarts, len(sds))):
            try:
                pert = project_capped(gb + rng.normal(0, sds[i], n), n)
                if pert is None:
                    continue
                bh, bs = continuation_search(pert, n, taus, n_iters)
                bh, bs = polish_subgrad(bh, n, n_polish)
                if bs < gs - 1e-12:
                    gs, gb = bs, bh.copy()
            except Exception:
                continue
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb
    sf = exact_score(final, n)
    ss = exact_score(fb, n)
    if sf > ss:
        final, sf = fb, ss
    if not is_feasible(final, n):
        final = np.full(n, 0.5).astype(np.float64)
        sf = exact_score(final, n)
    return final, float(sf), int(n)
