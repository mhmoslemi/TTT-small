def line_scan(gb, n, rng, budget, n_pm):
    gb = gb.copy()
    ds = 2.0 / n
    cur_s = exact_score(gb, n)
    seen = set()
    done = 0
    for _batch in range(2):
        for _ in range(n_pm):
            if done >= n_pm * (_batch + 1):
                break
            if budget.count >= budget.limit:
                break
            p = int(rng.integers(0, n))
            q = int(rng.integers(0, n))
            key = (min(p, q), max(p, q))
            if p == q or key in seen:
                continue
            seen.add(key)
            lo = max(-gb[p], gb[q] - 1.0)
            hi = min(1.0 - gb[p], gb[q])
            if hi - lo < 1e-12:
                continue
            # 7-point grid
            pts = [lo + rk * (hi - lo) / 6.0 for rk in range(7)]
            best_d, best_s = 0.0, cur_s
            sc_pts = {0.0: cur_s}
            exhausted = False
            for rk in range(7):
                d = pts[rk]
                if abs(d) < 1e-14:
                    continue
                trial = gb.copy()
                trial[p] += d
                trial[q] -= d
                sc = exact_score(trial, n)
                if not np.isfinite(sc):
                    continue
                budget.count += 1
                if budget.count >= budget.limit:
                    exhausted = True
                    break
                sc_pts[d] = sc
                if sc < best_s - 1e-14:
                    best_s, best_d = sc, d
            if exhausted:
                break
            # Refine
            for _ref in range(8):
                if budget.count >= budget.limit:
                    break
                # Find neighbors of best_d in sc_pts
                keys = sorted(sc_pts.keys())
                bd_idx = -1
                for ki in range(len(keys)):
                    if abs(keys[ki] - best_d) < 1e-14:
                        bd_idx = ki
                        break
                if bd_idx < 0:
                    break
                if bd_idx == 0:
                    if len(keys) < 2:
                        break
                    iv_lo, iv_hi = keys[0], keys[1]
                elif bd_idx == len(keys) - 1:
                    iv_lo, iv_hi = keys[-2], keys[-1]
                else:
                    iv_lo, iv_hi = keys[bd_idx - 1], keys[bd_idx + 1]
                improved = False
                for frac in (0.25, 0.5, 0.75):
                    d_new = iv_lo + frac * (iv_hi - iv_lo)

                    trial[q] -= d_new
                    sc = exact_score(trial, n)

                        continue

                        break

                gb[q] -= best_d
                cur_s = best_s

def chain_search(cur, cur_s, n, per_chain, thresholds, radii, rng, budget, gb, gs, targets):
    n_stages = len(thresholds)
    prop_per_stage = per_chain // n_stages
    rem = per_chain - prop_per_stage * n_stages
    for si in range(n_stages):
        n_prop = prop_per_stage + (1 if si < rem else 0)
        thresh = thresholds[si]
        radius = radii[si]
        for _ in range(n_prop):
            if budget.count >= budget.limit:
                return gb, gs
            if rng.random() < 0.12 and len(targets) > 0:
                target = targets[int(rng.integers(0, len(targets)))]
            else:
                target = gb
            prop = make_proposal(cur, n, rng, radius, target, 0.12 if len(targets) > 0 else 0.0)
            if prop is None:
                continue
            sc = exact_score(prop, n)
            if not np.isfinite(sc):
                continue
            budget.count += 1
            if budget.count >= budget.limit:
                return gb, gs

                cur = prop
                cur_s = sc
                if sc < gs - 1e-12:
                    gs, gb = sc, prop.copy()
    return gb, gs
