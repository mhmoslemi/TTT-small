    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

    dx = 2.0 / n
    J = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        r = k + n - 1
        q1a, q1b = max(0, k), min(n, k + n)
        if q1a < q1b:
