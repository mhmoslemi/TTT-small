def lag_vals(h, n):
    L = 2 * n - 1
    v = np.zeros(L)
    for d in range(-(n - 1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        v[d + n - 1] = float(np.dot(h[lo:hi], 1.0 - h[lo + d:hi + d]))
    return v

def softmax_grad(h, n, theta):
    F = lag_vals(h, n)
    L = len(F)
    m = float(np.max(F))
    sh = (F - m) / theta
    sh = np.clip(sh, -700.0, 700.0)
    eh = np.exp(sh)
    se = float(np.sum(eh))
    if not isfinite(se) or se < 1e-300:
        return m + theta * log(max(L, 1)), np.zeros(n)
    p = eh / se
    phi = m + theta * log(se)
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        w = p[d + n - 1]
        if w < 1e-16:
            continue
        lo = max(0, -d)
        hi = min(n, n - d)
        grad[lo:hi] += w * (1.0 - h[lo + d:hi + d])
        grad[lo + d:hi + d] -= w * h[lo:hi]
    return phi, grad

        if isfinite(ps) and ps < best_sc:
            best_h, best_sc = pert.copy(), ps
        best_h, best_sc = continuation_opt(pert, n, iters_per_tau, [taus[-1], taus[-2]], best_h, best_sc)
    final = project_capped(best_h, n)
    if final is None:
        final = fallback.copy()
    fs = exact_score(final, n)
    if not isfinite(fs) or fs > fb_score:
        final, fs = fallback.copy(), fb_score
    if not isfinite(fs):
        final = uniform.copy()
        fs = exact_score(final, n)
    return final, float(fs), int(n)
