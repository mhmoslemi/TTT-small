def solve_sdp(n, rng):
    try:
        import cvxpy as cp
    except Exception:
        return None, None
    if n < 2 or n > 64:
        return None, None
    s = n / 2.0
    dx = 2.0 / n
    h = cp.Variable(n)
    Y = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    ones_col = np.ones((n, 1))
    ones_row = np.ones((1, n))
    M = cp.bmat([[cp.Constant(1.0), h.reshape(1, n)],
                 [h.reshape(n, 1), Y]])
    cons = [M >> 0, h >= 0, h <= 1, cp.sum(h) == s, t >= 0]
    cons += [Y >= 0,
             Y >= h.reshape(n,1) * ones_row + ones_col * h.reshape(1,n) - 1.0,
             Y <= h.reshape(n,1) * ones_row,
             Y <= ones_col * h.reshape(1,n)]
    cons.append(Y @ cp.Constant(np.ones(n)) == s * h)
    for d in range(-(n-1), n):
        p0, q0 = max(d, 0), max(-d, 0)
        m = n - abs(d)
        cons.append(dx * (cp.sum(h[p0:p0+m]) - cp.trace(Y[p0:p0+m, q0:q0+m])) <= t)
    prob = cp.Problem(cp.Minimize(t), cons)
    # Try solvers
    for solver in [cp.MOSEK, cp.CVXOPT]:
        try:
            if solver not in cp.installed_solvers():
                continue
            kw = {}
            if solver == cp.CVXOPT:
                kw = {'max_iters': 100, 'abstol': 1e-7, 'reltol': 1e-7, 'feastol': 1e-7}
            prob.solve(solver=solver, **kw)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                if h.value is not None and Y.value is not None:
                    u = np.array(h.value).ravel()
                    Yv = np.array(Y.value)
                    if u.shape == (n,) and Yv.shape == (n, n) and np.all(np.isfinite(u)) and np.all(np.isfinite(Yv)):
                        S = Yv - np.outer(u, u)
                        P = np.eye(n) - np.ones((n,n))/n
                        S = P @ S @ P
                        S = 0.5*(S + S.T)
                        ev, evv = np.linalg.eigh(S)
                        ev = np.maximum(ev, 0)
                        S = evv @ np.diag(ev) @ evv.T
                        max_ev = np.max(ev)
                        if max_ev > n:
                            S = S * (n / max_ev)
                        return u, S
            break
        except Exception:
            continue
    return None, None
