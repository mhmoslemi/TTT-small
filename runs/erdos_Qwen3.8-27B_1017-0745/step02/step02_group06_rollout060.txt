        np.add.at(J[idx], ns, -h[ns + d] * dx)
    return J

def obj_fn(z):
    return z[-1]

def obj_jac(z):
    g = np.zeros(len(z))
    g[-1] = 1.0
    return g

def ineq_fn(z, n):
    return z[-1] - lag_values(z[:n], n)

def ineq_jac(z, n):
    h = z[:n]
    J = lag_jacobian(h, n)
    G = np.zeros((2 * n - 1, n + 1))
    G[:, :n] = -J
    G[:, n] = 1.0
    return G

def eq_fn(z, n):
    return np.sum(z[:n]) - n / 2.0

def eq_jac(z, n):
    g = np.zeros(len(z))
    g[:n] = 1.0
    return g

def build_pool(n, parent, rng):
    pool = [np.full(n, 0.5)]
    if parent is not None:
        p = project_capped(parent, n)
        if p is not None and is_feasible(p, n):
            pool.append(p)
            for v in [p[::-1].copy(), (1.0 - p).copy(), (1.0 - p[::-1]).copy()]:
                pp = project_capped(v, n)
                if pp is not None:
                    pool.append(pp)
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    pool.append(blk)
    for m in [1, 2, 3]:
        for ph in [0.0, np.pi / 2]:
            cv = 0.5 + 0.45 * np.cos(2.0 * np.pi * m * np.arange(n) / n + ph)
            pp = project_capped(cv, n)
            if pp is not None:
                pool.append(pp)
    base = parent if parent is not None else pool[0]
    for sd in [0.04, 0.10, 0.20]:
        nv = base + rng.normal(0, sd, n)
        pp = project_capped(nv, n)
        if pp is not None:
            pool.append(pp)
    pp = project_capped(rng.uniform(0, 1, n), n)
    if pp is not None:
        pool.append(pp)
    unique = []
    for v in pool:
        dup = False
        for u in unique:
            if np.max(np.abs(v - u)) < 1e-6:
                dup = True; break
        if not dup:
            unique.append(v)
    return unique

def solve_epigraph(z0, n, maxiter):
    bounds_h = [(0.0, 1.0)] * n + [(0.0, 0.5)]
    cons = [
        {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac, 'args': (n,)},
        {'type': 'ineq', 'fun': ineq_fn, 'jac': ineq_jac, 'args': (n,)}
    ]
    res = minimize(obj_fn, z0, jac=obj_jac, method='SLSQP',
                   bounds=bounds_h, constraints=cons,
                   options={'ftol': 1e-10, 'maxiter': maxiter, 'disp': False})
    if res.x is not None and len(res.x) == n + 1 and np.all(np.isfinite(res.x)):
        return res.x[:n].copy()
    return None

def try_blends(hc, gb, n):
    best_h, best_s = gb.copy(), exact_score(gb, n)
    for alpha in [0.25, 0.5, 0.75, 1.0]:
        blend = project_capped(alpha * hc + (1.0 - alpha) * gb, n)
        if blend is not None and is_feasible(blend, n):
            bs = exact_score(blend, n)
            if bs < best_s - 1e-12:
                best_h, best_s = blend.copy(), bs
    return best_h, best_s

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = np.clip(p, 0, 1)
    except Exception:
        pass
    if n is None:
        n = 47
    if parent is not None:
        pj = project_capped(parent, n)
        if pj is not None and is_feasible(pj, n):
            parent = pj
        else:
            parent = None
    fallback = np.full(n, 0.5)
    if n == 1:
        return fallback, exact_score(fallback, n), 1
    rng = np.random.default_rng(int(seed))
    if n <= 64:
        n_starts, n_restart, maxiter = 8, 4, 220
    elif n <= 128:
        n_starts, n_restart, maxiter = 5, 2, 150
    else:
        n_starts, n_restart, maxiter = 3, 1, 80
    gs = exact_score(fallback, n)
    gb = fallback.copy()
    if parent is not None:
        ps = exact_score(parent, n)
        if ps < gs - 1e-12:
            gb, gs = parent.copy(), ps
    pool = build_pool(n, parent, rng)
    scored = []
    for v in pool:
        s = exact_score(v, n)
        if np.isfinite(s) and s <= 0.5 + 1e-12:
            scored.append((s, v))
    scored.sort(key=lambda x: x[0])
    starts = [v for _, v in scored[:n_starts]]
    if len(starts) == 0:
        starts = [fallback]
    for h0 in starts:
        t0 = max(0.0, min(0.5, exact_score(h0, n) + 1e-9))
        z0 = np.append(h0, t0)
        try:
            hc = solve_epigraph(z0, n, maxiter)
            if hc is not None:
                hc = project_capped(hc, n)
                if hc is not None and is_feasible(hc, n):
                    sc = exact_score(hc, n)
                    if sc < gs - 1e-12:
                        bh, bs = try_blends(hc, gb, n)
                        if bs < gs - 1e-12:
                            gb, gs = bh, bs
                    elif sc < 0.5:
                        bh, bs = try_blends(hc, gb, n)
                        if bs < gs - 1e-12:
                            gb, gs = bh, bs
        except Exception:
            pass
    sigma = 0.04
    for _ in range(n_restart):
        improved = False
        start_h = None
        for _ in range(3):
            pert = gb + rng.normal(0, sigma, n)
            pp = project_capped(pert, n)
            if pp is not None and is_feasible(pp, n):
                sp = exact_score(pp, n)
                if sp <= 0.5 + 1e-12 and np.max(np.abs(pp - gb)) > 1e-6:
                    start_h = pp; break
        if start_h is None:
            sigma = min(0.20, 1.5 * sigma)
            continue
        t0 = max(0.0, min(0.5, exact_score(start_h, n) + 1e-9))
        z0 = np.append(start_h, t0)
        try:
            hc = solve_epigraph(z0, n, maxiter)
            if hc is not None:
                hc = project_capped(hc, n)
                if hc is not None and is_feasible(hc, n):
                    sc = exact_score(hc, n)
                    if sc < gs - 1e-12:
                        bh, bs = try_blends(hc, gb, n)
                        if bs < gs - 1e-12:
                            gb, gs = bh, bs
                            improved = True
                    elif sc < 0.5:
                        bh, bs = try_blends(hc, gb, n)
                        if bs < gs - 1e-12:
                            gb, gs = bh, bs
                            improved = True
        except Exception:
            pass
        if improved:
            sigma = max(0.01, 0.7 * sigma)
        else:
            sigma = min(0.20, 1.5 * sigma)
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    if parent is not None:
        ps = exact_score(parent, n)
        if ps < fs - 1e-12:
            final, fs = parent.copy(), ps
    fb_s = exact_score(fallback, n)
    if fb_s < fs - 1e-12:
        final, fs = fallback, fb_s
    return final.astype(np.float64), float(fs), int(n)
