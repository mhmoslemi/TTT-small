    uni = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        return uni, exact_score(uni, n), 1
    if n <= 64:
        n_starts, n_restarts, max_iter = 8, 3, 120
    elif n <= 128:
        n_starts, n_restarts, max_iter = 6, 2, 80
    elif n <= 256:
        n_starts, n_restarts, max_iter = 4, 1, 50
    else:
        n_starts, n_restarts, max_iter = 3, 1, 30
    fb_h = project_capped(parent, n) if parent is not None else uni.copy()
    if fb_h is None:
        fb_h = uni.copy()
    fb_s = exact_score(fb_h, n)
    best_h, best_s = fb_h.copy(), fb_s
    s_uni = exact_score(uni, n)
    if s_uni < best_s - 1e-12:
        best_s, best_h = s_uni, uni.copy()
    starts = build_starts(parent, n, rng, n_starts)
    scored = []
    for st in starts:
        s = exact_score(st, n)
        if s < float('inf') and s < best_s - 1e-12:
            best_s, best_h = s, st.copy()
        scored.append((s, st))
    scored.sort(key=lambda x: x[0])
    for s, st in scored:
        _, best_h, best_s = continuation_search(st, n, best_h, best_s, max_iter)
    sig = 0.08
    for _ in range(n_restarts):
        saved_s = best_s
        proposal = None
        for _a in range(3):
            prop = project_capped(best_h + sig * rng.normal(0, 1, n), n)
            if prop is not None and np.max(np.abs(prop - best_h)) >= 1e-5:
                proposal = prop
                break
        if proposal is not None:
            ps = exact_score(proposal, n)
            if ps < best_s - 1e-12:
                best_s, best_h = ps, proposal.copy()
            _, best_h, best_s = continuation_search(proposal, n, best_h, best_s, max_iter)
        if best_s < saved_s - 1e-12:
            sig = max(0.01, 0.7 * sig)
        else:
            sig = min(0.25, 1.5 * sig)
    final = project_capped(best_h, n)
    if final is None or not is_feasible(final, n):
        final = fb_h
    fs = exact_score(final, n)
    if fs > fb_s + 1e-12:
        final = fb_h
        fs = fb_s
    if fs > s_uni + 1e-12:
        final = uni
        fs = s_uni
    return final, float(fs), int(n)
