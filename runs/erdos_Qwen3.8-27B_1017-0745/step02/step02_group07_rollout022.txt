import numpy as np
import math

def project_capped(z, n):
    z = np.clip(np.asarray(z, dtype=np.float64).ravel(), -1e6, 1e6)
    s = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(4):
        if abs(r) < 1e-14:
            break
        if r > 0:
            for i in range(n):
                if r <= 1e-14:
                    break
                add = min(r, 1.0 - h[i])
                h[i] += add
                r -= add
        else:
            for i in range(n - 1, -1, -1):
                if r >= -1e-14:
                    break
                sub = min(-r, h[i])
                h[i] -= sub
                r += sub
    s2 = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-14) or np.any(h > 1.0 + 1e-14) or abs(s2) > 1e-9:
        return None
    return h

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-9

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    sc = float(np.max(c))
    return sc if np.isfinite(sc) else float('inf')

def solve_moment_sdp(n, rng):
    if n > 64:
        return None, None
    try:
        import cvxpy as cp
    except ImportError:
        return None, None
    s = n / 2.0
    dx = 2.0 / n
    v = cp.Variable(n)
    W = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    ones = np.ones(n)
    Z = cp.hstack([cp.hstack([1, v.reshape(1, n)]), cp.vstack([v.reshape(n, 1), W])])
    constraints = [
        Z >> 0,
        v >= 0, v <= 1,
        cp.sum(v) == s,
        W @ ones == s * v,
        t >= 0, t <= 1,
    ]
    # McCormick
    VI = v[:, None]
    VJ = v[None, :]
    constraints.append(W >= 0)
    constraints.append(W <= VI)
    constraints.append(W <= VJ)
    constraints.append(W >= VI + VJ - 1)
    # Lag constraints
    for r in range(0, n):
        m = n - r
        if m <= 0:
            continue
        if r == 0:
            expr = dx * (cp.sum(v) - cp.trace(W))
        else:
            idx = np.arange(m)
            if r % 2 == 1:  # Qplus
                expr = dx * (cp.sum(v[idx + r]) - cp.sum(W[tuple(idx, idx + r)]))
            else:  # Qminus
                expr = dx * (cp.sum(v[idx]) - cp.sum(W[tuple(idx, idx + r)]))
        constraints.append(expr <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    for solver in ['MOSEK', 'CVXOPT']:
        try:
            if not hasattr(cp, 'installed_solvers') or solver not in cp.installed_solvers:
                continue
            kwargs = {}
            if solver == 'MOSEK':
                kwargs = {'MSK_IPAR_OPTIMIZATION_THREADS': 1, 'MSK_DPAR_INFEAS_TOL': 1e-7, 'MSK_DPAR_DUAL_INFEAS_TOL': 1e-7, 'MSK_DPAR_REL_GAP_TOL': 1e-7}
            elif solver == 'CVXOPT':
                kwargs = {'max_iters': 200, 'abstol': 1e-7, 'reltol': 1e-7, 'feastol': 1e-7}
            prob.solve(solver=solver, **kwargs)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                vv = v.value
                WW = W.value
                if vv is not None and WW is not None and vv.shape == (n,) and WW.shape == (n, n):
                    if np.all(np.isfinite(vv)) and np.all(np.isfinite(WW)):
                        return vv.squeeze(), np.asarray(WW, dtype=np.float64)
        except Exception:
            continue
    return None, None

def build_pool(fb, n, rng, sdp_v=None, sdp_W=None):
    pool = []
    def try_add(vv):
        if vv is None:
            return
        for s in pool:
            if np.max(np.abs(vv - s)) < 1e-8:
                return
        pool.append(vv.copy())
    try_add(fb.copy())
    try_add(project_capped(fb[::-1], n))
    try_add(project_capped(1.0 - fb, n))
    try_add(project_capped(1.0 - fb[::-1], n))
    try_add(project_capped(np.full(n, 0.5), n))
    blk = np.zeros(n); blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    try_add(project_capped(blk, n))
    alt = np.zeros(n)
    for i in range(n):
        if i % 2 == 0:
            alt[i] = 1.0
    # Fix mass
    alt_proj = project_capped(alt, n)
    try_add(alt_proj)
    p = rng.permutation(n)
    bv = np.zeros(n)
    bv[p[:n // 2]] = 1.0
    if n % 2:
        bv[p[n // 2]] = 0.5
    try_add(project_capped(bv, n))
    for sd in (0.05, 0.12, 0.25):
        try_add(project_capped(fb + rng.normal(0, sd, n), n))
    for _ in range(8):
        try_add(project_capped(rng.uniform(0, 1, n), n))
    try_add(project_capped(np.roll(fb, n // 3), n))
    try_add(project_capped(np.roll(fb, n // 2), n))
    # SDP center
    if sdp_v is not None:
        sv = project_capped(sdp_v, n)
        try_add(sv)
        # Covariance sampling
        if sdp_W is not None:
            Sigma = sdp_W - np.outer(sdp_v, sdp_v)
            Sigma = 0.5 * (Sigma + Sigma.T)
            rm = Sigma.mean(axis=1)
            gm = Sigma.mean()
            Sigma = Sigma - rm[:, None] - rm[None, :] + gm
            try:
                eigs, evecs = np.linalg.eigh(Sigma)
                if np.all(np.isfinite(eigs)):
                    mx = max(1.0, np.max(np.abs(eigs)))
                    if eigs.min() > -1e-5 * mx:
                        eigs = np.maximum(eigs, 0.0)
                        eigs = np.minimum(eigs, n / 4.0)
                        keep = eigs > 1e-10 * max(1.0, eigs.max())
                        if np.any(keep):
                            F = evecs[:, keep] * np.sqrt(eigs[keep])
                            for gamma in (0.25, 0.5, 1.0, 1.5):
                                for _ in range(16):
                                    z = rng.standard_normal(F.shape[1])
                                    noise = F @ z
                                    try_add(project_capped(sv + gamma * noise, n))
                                    try_add(project_capped(sv - gamma * noise, n))
            except Exception:
                pass
    # Fallback Gaussian if no covariance
    if sdp_v is None:
        for _ in range(24):
            d = rng.standard_normal(n)
            d -= d.mean()
            try_add(project_capped(fb + 0.15 * d, n))
    # Cap pool size
    while len(pool) > 192:
        pool.pop()
    return pool

def anneal_chain(h0, score0, F_mat, n, steps, rng, best_h, best_score):
    h = h0.copy()
    sc = score0
    dx = 2.0 / n
    T0 = 0.25 * dx
    Tf = 0.01 * dx
    for k in range(steps):
        rho = k / max(1, steps - 1)
        T = T0 * (Tf / T0) ** rho
        delta = 0.18 * (1.0 - 0.75 * rho)
        if k % 5 != 4:
            # Pair transfer
            i = int(rng.integers(n))
            j = int(rng.integers(n - 1))
            if j >= i:
                j += 1
            tlo = max(-h[i], h[j] - 1.0)
            thi = min(1.0 - h[i], h[j])
            tlo = max(tlo, -delta)
            thi = min(thi, delta)
            if thi - tlo <= 1e-12:
                continue
            t = rng.uniform(tlo, thi)
            cand = h.copy()
            cand[i] += t
            cand[j] -= t
        else:
            # Collective move
            if F_mat is not None and F_mat.shape[1] > 0:
                z = rng.standard_normal(F_mat.shape[1])
                d = F_mat @ z
            else:
                d = rng.standard_normal(n)
            d -= d.mean()
            d[-1] -= d.sum()  # ensure zero sum
            mx = np.max(np.abs(d))
            if mx < 1e-14 or not np.isfinite(mx):
                continue
            d = d / mx
            # Feasible interval
            tlo, thi = -delta, delta
            for ii in range(n):
                if d[ii] > 1e-14:
                    tlo = max(tlo, -h[ii] / d[ii])
                    thi = min(thi, (1.0 - h[ii]) / d[ii])
                elif d[ii] < -1e-14:
                    tlo = max(tlo, (1.0 - h[ii]) / d[ii])
                    thi = min(thi, -h[ii] / d[ii])
            if thi - tlo <= 1e-12:
                continue
            t = rng.uniform(tlo, thi)
            cand = h + t * d
        cand = project_capped(cand, n)
        if cand is None:
            continue
        if np.max(np.abs(cand - h)) < 1e-10:
            continue
        cs = exact_score(cand, n)
        if cs < best_score - 1e-12:
            best_score[0] = cs
            best_h[0] = cand.copy()
        if cs <= sc + 1e-15:
            h = cand
            sc = cs
        else:
            log_p = math.log(max(rng.random(), 1e-300))
            if log_p < -(cs - sc) / T:
                h = cand
                sc = cs
    return best_h[0], best_score[0]
