    L = 2 * n - 1
    for k in range(L):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        p = max(s, 0)
        r = max(-s, 0)
        m = n - abs(s)
        grad[p:p + m] += dx * wk * (1.0 - h[r:r + m])
        grad[r:r + m] -= dx * wk * h[p:p + m]
    return F, grad

def continuation_run(h0, n, tau_sched, max_iters):
    h = h0.copy().astype(np.float64)
    best_h = h.copy()
    best_s = exact_score(h, n)
    for tau in tau_sched:
        F_cur, g = smooth_val_grad(h, n, tau)
        if not np.isfinite(F_cur):
            continue
        eta = 0.5 * n / 2.0
        eta_max = 2.0 * n
        for it in range(max_iters):
            F, g = smooth_val_grad(h, n, tau)
            if not np.isfinite(F) or not np.all(np.isfinite(g)):
                break
            F_cur = F
            accepted = False
            for _bt in range(25):
                z = project_capped(h - eta * g, n)
                if z is None:
                    eta *= 0.5
                    if eta < 1e-12:
                        break
                    continue
                d = z - h
                sn = np.max(np.abs(d))
                if sn <= 1e-12:
                    break
                gd = np.dot(g, d)
                if not np.isfinite(gd) or gd >= 0:
                    eta *= 0.5
                    if eta < 1e-12:
                        break
                    continue
                F_tr, _ = smooth_val_grad(z, n, tau)
                if np.isfinite(F_tr) and F_tr <= F_cur + 1e-4 * gd + 1e-14:
                    s_try = exact_score(z, n)
                    if s_try < best_s - 1e-12:
                        best_s = s_try
                        best_h = z.copy()
                    h = z
                    F_cur = F_tr
                    eta = min(1.2 * eta, eta_max)
                    accepted = True
                    break
                else:
                    eta *= 0.5
                    if eta < 1e-12:
                        break
            if not accepted:
                break
    s_final = exact_score(h, n)
    if s_final < best_s:
        best_s = s_final
        best_h = h.copy()
    return best_h, best_s

def coordinate_polish(h, n, n_passes):
    h = h.copy().astype(np.float64)
    best_s = exact_score(h, n)
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for _ in range(n_passes):
        improved = True
        while improved:
            improved = False
            for i in range(n):
                for dv in deltas:
                    for sign in (1.0, -1.0):
                        trial = h.copy()
                        trial[i] += sign * dv
                        z = project_capped(trial, n)
                        if z is None:
                            continue
                        s = exact_score(z, n)
                        if s < best_s - 1e-13:
                            best_s = s
                            h = z
                            improved = True
    return h, best_s

def subgrad_polish(h, n, n_iter):
    h = h.copy().astype(np.float64)
    best_h = h.copy()
    best_s = exact_score(h, n)
    dx = 2.0 / n
    eta = 0.15 / dx
    for _ in range(n_iter):
        q = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = int(np.argmax(q))
        s = idx - (n - 1)
        p = max(s, 0)
        r = max(-s, 0)
        m = n - abs(s)
        g = np.zeros(n)
        g[p:p + m] += dx * (1.0 - h[r:r + m])
        g[r:r + m] -= dx * h[p:p + m]
        for alpha in (1.0, 0.5, 0.25, 0.125):
            z = project_capped(h - alpha * eta * g, n)
            if z is None:
                continue
            sz = exact_score(z, n)
            if sz < best_s - 1e-13:
                best_s = sz
                best_h = z.copy()
                h = z
                break
    return best_h, best_s

def build_starts(fb, n, rng, count):
    pool = [fb.copy()]
    cands = [np.full(n, 0.5)]
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    cands.append(blk)
    j = np.arange(n).astype(np.float64)
    cands.append(np.clip(0.5 + 0.45 * np.cos(2.0 * np.pi * (j + 0.5) / n), 0, 1))
    cands.append(np.clip(0.5 + 0.45 * np.cos(2.0 * np.pi * (j + 0.5) / n + np.pi / 2), 0, 1))
    cands.append(rng.uniform(0, 1, n))
    cands.append(fb + rng.normal(0, 0.06, n))
    cands.append(fb + rng.normal(0, 0.15, n))
    cands.append(fb[::-1].copy())
    for c in cands:
        if len(pool) >= count:
            break
        p = project_capped(c, n)
        if p is None:
            continue
        dup = False
        for s in pool:
            if np.max(np.abs(p - s)) < 1e-8:
                dup = True
                break
        if not dup:
            pool.append(p)
    return pool[:count]

def slsqp_polish(h0, n, maxiter):
    def _obj(z):
        return z[n]
    def _obj_jac(z):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    def _ineq(z):
        hh = np.clip(z[:n], 0.0, 1.0)
        return z[n] - np.correlate(hh, 1.0 - hh, mode='full') * (2.0 / n)
    def _ineq_jac(z):
        hh = np.clip(z[:n], 0.0, 1.0)
        dx = 2.0 / n
        L = 2 * n - 1
        G = np.zeros((L, n + 1))
        for dt in range(-(n - 1), n):
            row = dt + n - 1
            p0 = max(dt, 0)
            q0 = max(-dt, 0)
            m = n - abs(dt)
            G[row, p0:p0 + m] += dx * (1.0 - hh[q0:q0 + m])
            G[row, q0:q0 + m] -= dx * hh[p0:p0 + m]
            G[row, n] = 1.0
        out = np.zeros((L, n + 1))
        out[:, :n] = -G[:, :n]
        out[:, n] = G[:, n]
        return out
    def _eq(z):
        return np.sum(z[:n]) - n / 2.0
    def _eq_jac(z):
        J = np.zeros(n + 1)
        J[:n] = 1.0
        return J
    q0v = np.max(np.correlate(h0, 1.0 - h0, mode='full') * (2.0 / n))
    z0 = np.concatenate([h0, [q0v + 1e-8]])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    con = [{'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac},
           {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}]
    try:
        res = minimize(_obj, z0, method='SLSQP', jac=_obj_jac,
                       bounds=bounds, constraints=con,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
    except Exception:
        return None
    if res.x is None or not np.all(np.isfinite(res.x)):
        return None
    return res.x[:n]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size >= 1:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None or not (1 <= n <= 999):
        n = 47
    if kwargs.get('n_points') is not None:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    if parent is not None and parent.size != n:
        old_n = parent.size
        old_pos = (np.arange(old_n) + 0.5) / old_n
        new_pos = (np.arange(n) + 0.5) / n
        parent_new = np.interp(new_pos, old_pos, parent, left=parent[0], right=parent[-1])
        parent = parent_new
    fb = project_capped(parent, n) if parent is not None else np.full(n, 0.5)
    if fb is None:
        fb = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        return fb, exact_score(fb, n), 1
    fs = exact_score(fb, n)
    gb, gs = fb.copy(), fs
    if n <= 64:
        n_sc, n_mi, n_restart = 6, 140, 3
    elif n <= 128:
        n_sc, n_mi, n_restart = 4, 90, 1
    elif n <= 256:
        n_sc, n_mi, n_restart = 3, 55, 1
    else:
        n_sc, n_mi, n_restart = 2, 25, 0
    tau_sched = [0.02, 0.005, 0.001, 0.0002, 0.00004, 0.00001]
    starts = build_starts(fb, n, rng, n_sc)
    for st in starts:
        try:
            cur, cur_s = continuation_run(st, n, tau_sched, n_mi)
            if cur_s < gs - 1e-13:
                gs, gb = cur_s, cur.copy()
        except Exception:
            continue
    # Coordinate polish on best
    try:
        gp, sp = coordinate_polish(gb, n, 2)
        if sp < gs - 1e-13:
            gs, gb = sp, gp.copy()
    except Exception:
        pass
    # Subgradient polish on best
    try:
        gp, sp = subgrad_polish(gb, n, 30)
        if sp < gs - 1e-13:
            gs, gb = sp, gp.copy()
    except Exception:
        pass
    # SLSQP polish for small n
    if n <= 64:
        for _ in range(2):
            try:
                sf = slsqp_polish(gb, n, 200)
                if sf is not None:
                    pf = project_capped(sf, n)
                    if pf is not None:
                        ps = exact_score(pf, n)
                        if ps < gs - 1e-13:
                            gs, gb = ps, pf.copy()
            except Exception:
                continue
    # Perturbation restarts
    if n_restart > 0:
        perts = [0.03, 0.08, 0.15][:n_restart]
        for pd in perts:
            try:
                pert = project_capped(gb + rng.normal(0, pd, n), n)
                if pert is None:
                    continue
                cur, cur_s = continuation_run(pert, n, tau_sched, n_mi)
                if cur_s < gs - 1e-13:
                    gs, gb = cur_s, cur.copy()
                cp, cs2 = coordinate_polish(gb, n, 1)
                if cs2 < gs - 1e-13:
                    gs, gb = cs2, cp.copy()
            except Exception:
                continue
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb
    sf = exact_score(final, n)
    ss = exact_score(fb, n)
    if sf > ss:
        final, sf = fb, ss
    if not is_feasible(final, n):
        final = np.full(n, 0.5)
        sf = exact_score(final, n)
    return final, float(sf), int(n)
