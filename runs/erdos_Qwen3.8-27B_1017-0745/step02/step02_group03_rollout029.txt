Actually, I realize I should double-check my understanding of the problem. The score is:
C₅ = max over all k of ∫ h(x)(1-h(x+k)) dx

In the discretization with dx = 2/n:
- h has n points
- The correlation np.correlate(h, 1-h, mode='full') gives 2n-1 values
- The score is max of these values times dx = 2/n

The lag d in my notation corresponds to x+k where k ranges over all real values. In the discrete case, k = d*dx for integer d from -(n-1) to (n-1).

So F_d = sum over valid i of h[i] * (1-h[i+d]), and the integral is F_d * dx = F_d * 2/n.

The score is max_d (F_d * 2/n) = (2/n) * max_d F_d.

This is exactly what exact_score computes. Good.

Now, one subtlety: np.correlate(h, 1-h, mode='full') computes:
out[m] = sum_k h[k] * (1-h)[k + (m - (n-1))]  for m = 0, ..., 2n-2

Let me verify: for full correlation of two sequences a (length n) and b (length n):
out[m] = sum_k a[k] * conj(b[k + m - (n-1)])

For m = n-1 (the center), out[n-1] = sum_k a[k] * b[k] = sum h[i]*(1-h[i]) -- this is the d=0 lag.

import numpy as np

def project_k(z, n):
    s = n / 2.0
    z = np.asarray(z, dtype=np.float64)
    if not np.all(np.isfinite(z)):
        return None
    z = np.clip(z, -1e12, 1e12)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if abs(hi - lo) < 1e-15:
            break
        if float(np.sum(np.clip(z - mid, 0.0, 1.0))) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - float(np.sum(h))
    for _ in range(n + 4):
        if abs(r) < 1e-13 * max(1, n):
            break
        if r > 0:
            cap = 1.0 - h
            idx = int(np.argmax(cap))
            if cap[idx] <= 1e-15:
                break
            h[idx] += min(r, cap[idx])
        else:
            idx = int(np.argmax(h))
            if h[idx] <= 1e-15:
                break
            h[idx] += max(r, -h[idx])
    h = np.clip(h, 0.0, 1.0)
    if not np.all(np.isfinite(h)) or abs(float(np.sum(h)) - s) > 1e-9 * max(1, n):
        return None
    return h

def exact_score(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return float('inf')

        return float('inf')

        return float('inf')
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def lag_vals(h, n):
    L = 2 * n - 1
    v = np.zeros(L)
    for d in range(-(n - 1), n):
        lo = max(0, -d)
        hi = min(n, n - d)
        v[d + n - 1] = float(np.dot(h[lo:hi], 1.0 - h[lo + d:hi + d]))
    return v

def lag_grad(h, n, p):
    G = np.zeros(n)
    for d in range(-(n - 1), n):
        pd = p[d + n - 1]
        if abs(pd) < 1e-15:
            continue
        lo = max(0, -d)
        hi = min(n, n - d)
        G[lo:hi] += pd * (1.0 - h[lo + d:hi + d])
        G[lo + d:hi + d] -= pd * h[lo:hi]
    return G

def smooth_obj(h, n, theta):
    F = lag_vals(h, n)
    m = float(np.max(F))
    e = np.clip((F - m) / theta, -700.0, 700.0)
    se = float(np.sum(np.exp(e)))
    if se < 1e-300:
        return m, None
    p = np.exp(e) / se
    return m + theta * math.log(se), p

def pairwise_polish(h, n, sweeps):
    best = h.copy()
    bs = exact_score(best, n)
    for _ in range(sweeps):
        imp = False
        for i in range(n):
            for j in range(i + 1, n):
                lo = max(-best[i], best[j] - 1.0)
                hi = min(1.0 - best[i], best[j])
                if lo >= hi - 1e-14:
                    continue
                cands = [0.0, lo + 1e-12, hi - 1e-12]
                for step in (0.001, 0.005, 0.02, 0.08, 0.25):
                    for sgn in (1, -1):
                        dd = sgn * step
                        if lo <= dd <= hi:
                            cands.append(dd)
                bd, bsc = 0.0, bs
                for dd in cands:
                    dd = max(lo, min(hi, dd))
                    t = best.copy()
                    t[i] += dd
                    t[j] -= dd
                    sc = exact_score(t, n)
                    if sc < bsc - 1e-13:
                        bd, bsc = dd, sc
                if bd != 0.0:
                    best[i] += bd
                    best[j] -= bd
                    bs = bsc
                    imp = True
        if not imp:
            break
    return best, bs
