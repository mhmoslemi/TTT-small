        return float('inf')
    try:
        v = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
        return v if np.isfinite(v) else float('inf')
    except Exception:
        return float('inf')

def lag_jacobian(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    G[n - 1] = dx * (1.0 - 2.0 * h)
    for r in range(1, n):
        m = n - r
        G[n - 1 + r, r:] += dx * (1.0 - h[:m])
        G[n - 1 + r, :m] += dx * (-h[r:])
        G[n - 1 - r, :m] += dx * (1.0 - h[r:])
        G[n - 1 - r, r:] += dx * (-h[:m])
    return G

def _obj(z, n):
    return z[n]

def _obj_jac(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def _ineq(z, n):
    h = z[:n]
    Q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return z[n] - Q

def _ineq_jac(z, n):
    h = z[:n]
    G = lag_jacobian(h, n)
    return np.hstack([-G, np.ones((2 * n - 1, 1))])

def _eq(z, n):
    return np.array([np.sum(z[:n]) - n / 2.0])

def _eq_jac(z, n):
    J = np.zeros(n + 1)
    J[:n] = 1.0
    return J

def smooth_grad(h, n, tau):
    Q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    qmax = np.max(Q)
    ex = np.exp((Q - qmax) / tau)
    p = ex / np.sum(ex)
    dx = 2.0 / n
    g = dx * p[n - 1] * (1.0 - 2.0 * h)
    for r in range(1, n):
        m = n - r
        g[r:] += dx * p[n - 1 + r] * (1.0 - h[:m])
        g[:m] += dx * p[n - 1 + r] * (-h[r:])
        g[:m] += dx * p[n - 1 - r] * (1.0 - h[r:])
        g[r:] += dx * p[n - 1 - r] * (-h[:m])
    return g

def smooth_val(h, n, tau):
    Q = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    qmax = np.max(Q)
    return qmax + tau * np.log(np.mean(np.exp((Q - qmax) / tau)))

def smooth_refine(h, n, tau_list, iters_per_tau, best_h, best_s):
    for tau in tau_list:
        for _ in range(iters_per_tau):
            g = smooth_grad(h, n, tau)
            if not np.all(np.isfinite(g)):
                break
            alpha = 1.0
            accepted = False
            for _bt in range(20):
                y = project_capped(h - alpha * g, n)
                if y is None:
                    alpha *= 0.5
                    continue
                fy = smooth_val(y, n, tau)
                fh = smooth_val(h, n, tau)
                if fy <= fh + 0.5 * np.dot(g, y - h) + 1e-14:
                    ts = exact_score(y, n)
                    if np.isfinite(ts) and ts < best_s - 1e-12:
                        best_s = ts
                        best_h = y.copy()
                    h = y
                    accepted = True
                    break
                alpha *= 0.5
            if not accepted:
                break
    return best_h, best_s

def try_add(pool, v, th=1e-8):
    if v is None:
        return
    for s in pool:
        if np.max(np.abs(v - s)) < th:
            return
    pool.append(v)

def build_starts(fb, n, rng):
    pool = []
    try_add(pool, fb.copy())
    try_add(pool, project_capped(fb[::-1], n))
    try_add(pool, project_capped(1.0 - fb, n))
    try_add(pool, project_capped(1.0 - fb[::-1], n))
    try_add(pool, np.full(n, 0.5))
    blk = np.zeros(n)
    blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    try_add(pool, project_capped(blk, n))
    for sd in (0.08, 0.22, 0.35):
        try_add(pool, project_capped(fb + rng.normal(0, sd, n), n))
    for k in (1, 2, 3):
        ph = 0 if rng.random() < 0.5 else np.pi / 2
        w = 0.5 + 0.4 * np.cos(2.0 * np.pi * k * np.arange(n) / n + ph)
        try_add(pool, project_capped(w, n))
    p = rng.permutation(n)
    bv = np.zeros(n)
    bv[p[:n // 2]] = 1.0
    if n % 2:
        bv[p[n // 2]] = 0.5
    try_add(pool, project_capped(bv, n))
    try_add(pool, project_capped(rng.uniform(0, 1, n), n))
    for sh in (n // 3, n // 2):
        try_add(pool, project_capped(np.roll(fb, sh), n))
    limit = 10 if n <= 64 else (7 if n <= 128 else 4)
    return pool[:limit]

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if 1 <= p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                parent = p
    except Exception:
        pass
    if n is None:
        n = 47
        try:
            vk = int(kwargs.get('n_points', 47))
            if 1 <= vk <= 999:
                n = vk
        except Exception:
            pass
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        h1 = np.array([0.5])
        return h1, float(exact_score(h1, 1)), 1
    if parent is not None:
        fb = project_capped(np.clip(parent, 0.0, 1.0), n)
    else:
        fb = np.full(n, 0.5)
    if fb is None or not is_feasible(fb, n):
        fb = np.full(n, 0.5)
    fb = fb.copy()
    fb_s = exact_score(fb, n)
    best_h, best_s = fb.copy(), fb_s
    starts = build_starts(fb, n, rng)
    # SLSQP for n <= 128
    if n <= 128:
        maxit = 220 if n <= 64 else 140
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        con_ineq = {'type': 'ineq', 'fun': _ineq, 'jac': _ineq_jac, 'args': (n,)}
        con_eq = {'type': 'eq', 'fun': _eq, 'jac': _eq_jac, 'args': (n,)}
        for h0 in starts:
            t0 = exact_score(h0, n) + 1e-8
            z0 = np.concatenate([h0, [t0]])
            try:
                res = minimize(_obj, z0, jac=_obj_jac, args=(n,), method='SLSQP',
                               bounds=bounds, constraints=[con_ineq, con_eq],
                               options={'ftol': 1e-10, 'maxiter': maxit, 'disp': False})
                if res.x is not None and res.x.shape == (n + 1,):
                    for a in (0.0, 0.25, 0.5, 0.75, 1.0):
                        mix = (1.0 - a) * h0 + a * res.x[:n]
                        pm = project_capped(mix, n)
                        if pm is not None:
                            sm = exact_score(pm, n)
                            if sm < best_s - 1e-12:
                                best_s = sm
                                best_h = pm.copy()
            except Exception:
                pass
    # Smooth continuation polish
    tau_list = [0.02, 0.005, 0.001, 0.0002]
    ipt = 12 if n <= 128 else (30 if n <= 256 else 12)
    for h0 in starts[:3]:
        bh, bs = smooth_refine(h0.copy(), n, tau_list, ipt, best_h.copy(), best_s)
        if bs < best_s - 1e-12:
            best_h, best_s = bh, bs
    bh2, bs2 = smooth_refine(best_h.copy(), n, tau_list, ipt, best_h.copy(), best_s)
    if bs2 < best_s - 1e-12:
        best_h, best_s = bh2, bs2
    # Final validity check
    final_h = project_capped(best_h, n)
    if final_h is None or not is_feasible(final_h, n):
        final_h = fb
    final_s = exact_score(final_h, n)
    if not np.isfinite(final_s) or final_s > fb_s + 1e-12:
        final_h, final_s = fb, fb_s
    return final_h.astype(np.float64), float(final_s), int(n)
