Key components:
1. Parse `initial_h_values` for parent (n=47)
2. Build feasible pool of starting points
3. For each starting point, run MM iterations:
   - Compute lag values q and gradients G
   - Solve convex QP: min u + (L/2)||d||² subject to linear constraints
   - Damped line search with exact score evaluation
4. Return best feasible solution

Key mathematical details:
- n samples, dx = 2/n, sum(h) = n/2
- Lag 0: Q_0 = dx * sum(h_i * (1 - h_i))
- Lag r, +: Q_{r,+} = dx * sum_{i=0}^{n-1-r} h_i * (1 - h_{i+r})
- Lag r, -: Q_{r,-} = dx * sum_{i=0}^{n-1-r} h_{i+r} * (1 - h_i)
- Objective: max of all these values

Gradients:
- g_0 = dx * (1 - 2h)
- For Q_{r,+}: entries 0..n-1-r get dx*(1-h[r+i]), entries r..n-1 get -dx*h[i]
  More precisely: g[r+i] -= dx*h[i] for i in 0..n-1-r, and g[i] += dx*(1-h[r+i]) for i in 0..n-1-r
- For Q_{r,-}: g[i+r] += dx*(1-h[i]) for i in 0..n-1-r, and g[i] -= dx*h[r+i] for i in 0..n-1-r

Hessian bound: L = 2*dx (each Hessian has eigenvalues bounded by 2dx)

QP formulation:
- Variables: d (n,), u (scalar)
- Objective: u + (L/2) * ||d||²
- Constraints:
  - q + G*d <= u (for each of 2n-1 lags)
  - sum(d) = 0
  - -h_i <= d_i <= 1 - h_i

Solver: Use CVXPY with ECOS, MOSEK, GUROBI, or CVXOPT.

    return float(val) if np.isfinite(val) else float('inf')

def lag_data(h, n):
    dx = 2.0 / n
    q = np.zeros(2 * n - 1)
    G = np.zeros((2 * n - 1, n))
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0, :] = dx * (1.0 - 2.0 * h)
    for r in range(1, n):
        m = n - r
        ip = 2 * r - 1
        im = 2 * r
        q[ip] = dx * np.dot(h[:m], 1.0 - h[r:r+m])
        G[ip, :m] += dx * (1.0 - h[r:r+m])
        G[ip, r:r+m] -= dx * h[:m]
        q[im] = dx * np.dot(h[r:r+m], 1.0 - h[:m])
        G[im, r:r+m] += dx * (1.0 - h[:m])
        G[im, :m] -= dx * h[r:r+m]
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None, None
    return q, G

def build_qp(n):
    try:
        import cvxpy as cp
    except Exception:
        return None
    try:
        d = cp.Variable(n)
        u = cp.Variable()
        qp = cp.Parameter(2 * n - 1)
        Gp = cp.Parameter((2 * n - 1, n))
        hp = cp.Parameter(n)
        L = 4.0 / n
        obj = cp.Minimize(u + 0.5 * L * cp.sum_squares(d))
        cons = [Gp @ d + qp <= u, cp.sum(d) == 0, d >= -hp, d <= 1.0 - hp]
        prob = cp.Problem(obj, cons)
        return prob, d, u, qp, Gp, hp
    except Exception:
        return None

def solve_qp(qp_obj, d, u, qp_p, Gp_p, hp_p, q, G, h):
    try:
        qp_p.value = q
        Gp_p.value = G
        hp_p.value = h
    except Exception:
        return None
    solvers = ['ECOS', 'MOSEK', 'GUROBI', 'CVXOPT']
    for solver in solvers:
        try:
            qp_obj.solve(solver=solver)
            if qp_obj.status in ('optimal', 'optimal_inaccurate'):

                    return dv
                # broadcast

                    return dv.ravel()

            continue
    return None

def try_add(pool, v, n, th=1e-8):
    if v is None:
        return

        hs = best_cs
        if red < 1e-10:

import numpy as np
import math

def project_capped(z, n):
    z = np.clip(np.asarray(z, dtype=np.float64).ravel(), -1e6, 1e6)
    if z.shape != (n,) or not np.all(np.isfinite(z)):
        return None
    s = n / 2.0
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for _ in range(6):
        if abs(r) < 1e-13:
            break
        if r > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + r, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + r, 0.0)
        r = s - np.sum(h)
    if not np.all(np.isfinite(h)) or np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return None
    if abs(np.sum(h) - s) > 1e-9:
        return None
    return h

def is_feasible(h, n):
    return (h.shape == (n,) and np.all(np.isfinite(h)) and
            np.all(h >= -1e-10) and np.all(h <= 1.0 + 1e-10) and
            abs(np.sum(h) - n / 2.0) <= 1e-9 * max(1, n))

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    val = np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n))
    return float(val) if np.isfinite(val) else float('inf')

def lag_data(h, n):
    dx = 2.0 / n
    q = np.zeros(2 * n - 1)
    G = np.zeros((2 * n - 1, n))
    q[0] = dx * np.sum(h * (1.0 - h))
    G[0, :] = dx * (1.0 - 2.0 * h)
    for r in range(1, n):
        m = n - r
        ip = 2 * r - 1
        im = 2 * r
        q[ip] = dx * np.dot(h[:m], 1.0 - h[r:r + m])
        G[ip, :m] += dx * (1.0 - h[r:r + m])
        G[ip, r:r + m] -= dx * h[:m]
        q[im] = dx * np.dot(h[r:r + m], 1.0 - h[:m])
        G[im, r:r + m] += dx * (1.0 - h[:m])
        G[im, :m] -= dx * h[r:r + m]
    if not np.all(np.isfinite(q)) or not np.all(np.isfinite(G)):
        return None, None
    return q, G

def build_qp(n):
    try:
        import cvxpy as cp
    except Exception:
        return None
    try:
        d = cp.Variable(n)
        u = cp.Variable()
        qp = cp.Parameter(2 * n - 1)
        Gp = cp.Parameter((2 * n - 1, n))
        hp = cp.Parameter(n)
        L = 4.0 / n
        obj = cp.Minimize(u + 0.5 * L * cp.sum_squares(d))
        cons = [Gp @ d + qp <= u, cp.sum(d) == 0, d >= -hp, d <= 1.0 - hp]
        prob = cp.Problem(obj, cons)
        return (prob, d, u, qp, Gp, hp)
    except Exception:
        return None

def solve_qp(qpdata, q, G, h):
    prob, dvar, uvar, qp_p, Gp_p, hp_p = qpdata
    try:
        qp_p.value = q
        Gp_p.value = G
        hp_p.value = h
    except Exception:
        return None
    for solver in ('ECOS', 'MOSEK', 'GUROBI', 'CVXOPT'):
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                dv = dvar.value
                if dv is not None:
                    dv = np.asarray(dv, dtype=np.float64).ravel()
                    if dv.shape == (len(q),) and np.all(np.isfinite(dv)):
                        return dv
        except Exception:
            continue
    return None

def try_add(pool, v, n, th=1e-8):
    if v is None:
        return
    for s in pool:
        if np.max(np.abs(v - s)) < th:

def mm_search(h0, n, qpdata, rng, max_iter=50):
    h = h0.copy()
    hs = exact_score(h, n)
    if hs == float('inf'):
        return h, hs
    small = 0
    for _ in range(max_iter):
        q, G = lag_data(h, n)
        if q is None:
            break
        d = solve_qp(qpdata, q, G, h)
        if d is None:
            break
        if np.max(np.abs(d)) < 1e-8:
            break
        best_c, best_s = None, hs
        for alpha in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125):
            cand = project_capped(h + alpha * d, n)
            if cand is None:
                continue
            if np.max(np.abs(cand - h)) < 1e-10:
                continue
            cs = exact_score(cand, n)
            if cs < best_s - 1e-12:
                best_s = cs
                best_c = cand
        if best_c is None:
            break
        red = hs - best_s
        h = best_c
        hs = best_s
        if red < 1e-10:
            small += 1
            if small >= 3:
                break
        else:
            small = 0
    return h, hs
