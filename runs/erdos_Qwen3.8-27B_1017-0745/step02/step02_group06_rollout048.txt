def try_add_pool(pool, v, n):
    p = project_capped(v, n)
    if p is None:
        return
    sc = exact_score(p, n)
    if math.isinf(sc):
        return
    for e in pool:
        if float(np.max(np.abs(p - e[0]))) < 1e-8:
            return
    pool.append((p.copy(), sc))
