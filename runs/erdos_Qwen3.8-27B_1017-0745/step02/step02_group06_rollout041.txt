    d = n - 1

        return np.zeros(n)

    out = np.zeros(n)
    out[0] = T[0]
    for i in range(1, n):
        out[i] = T[i] - i * a[i - 1]
    return out

    if d <= 0:
        return np.zeros(0)
    j = np.arange(1, n, dtype=np.float64)
    ps = np.concatenate([[0.0], np.cumsum(v)])
    return (ps[1:n] - j * v[1:n]) / np.sqrt(j * (j + 1.0))

def cem_search(h_start, n, rng, pop_size, max_gen, stall_limit, cov_mode, sigma0, gb_vec, gb_score):
    d = n - 1
    if d <= 0:
        return gb_vec, gb_score
    eta, sig_min, sig_max = 0.35, 0.02, 0.5
    u = np.full(n, 0.5)
    m = helmert_qtv(h_start - u, n).copy()
    mu = max(2, pop_size // 4)
    use_full = (cov_mode == 'full' and d <= 64)
    if use_full:
        C = sigma0 * sigma0 * np.eye(d)
    else:
        v_diag = np.full(d, sigma0 * sigma0)
    half_pop = max(1, pop_size // 2)
    stall = 0
    run_best = exact_score(h_start, n)
    eig_reset = 0
    for gen in range(max_gen):
        if use_full:
            C_sy = 0.5 * (C + C.T)
            try:
                ev, evec = np.linalg.eigh(C_sy)
            except np.linalg.LinAlgError:
                eig_reset += 1
                if eig_reset > 1:
                    break
                C = sigma0 * sigma0 * np.eye(d); ev = np.full(d, sigma0 * sigma0); evec = np.eye(d)
            ev = np.clip(ev, sig_min * sig_min, sig_max * sig_max)
            if not (np.all(np.isfinite(ev)) and np.all(np.isfinite(evec))):
                eig_reset += 1
                if eig_reset > 1:
                    break
                C = sigma0 * sigma0 * np.eye(d); ev = np.full(d, sigma0 * sigma0); evec = np.eye(d)
            L = evec @ np.diag(np.sqrt(np.maximum(ev, 0.0)))
        else:
            sv = np.sqrt(np.clip(v_diag, sig_min * sig_min, sig_max * sig_max))
        cx, ch, cs = [], [], []
        for k in range(half_pop):
            z = rng.standard_normal(d)
            for sgn in (1, -1):
                if use_full:
                    xp = m + sgn * (L @ z)
                else:
                    xp = m + sgn * sv * z
                hp = project_capped(u + helmert_qx(xp, n), n)
                if hp is None:
                    continue
                xh = helmert_qtv(hp - u, n)
                sc = exact_score(hp, n)
                if np.isfinite(sc):
                    cx.append(xh); ch.append(hp); cs.append(sc)
                    if sc < gb_score - 1e-12:
                        gb_score = sc; gb_vec = hp.copy()
        if not cx:
            break
        order = np.argsort(cs, kind='stable')
        mu_eff = min(mu, len(cx))
        sel = order[:mu_eff]
        w = np.array([math.log(mu_eff + 0.5) - math.log(r + 1) for r in range(mu_eff)])
        w /= w.sum()
        Xs = np.array([cx[i] for i in sel])
        m_e = (w[:, None] * Xs).sum(axis=0)
        m_new = (1.0 - eta) * m + eta * m_e
        h_m = project_capped(u + helmert_qx(m_new, n), n)
        if h_m is None:
            break
        m_new = helmert_qtv(h_m - u, n)
        if use_full:
            D = Xs - m_new
            S = (D.T * w) @ D
            C = 0.5 * ((1.0 - eta) * C + eta * S + ((1.0 - eta) * C + eta * S).T)
        else:
            D = Xs - m_new
            S = (w[:, None] * D * D).sum(axis=0)
            v_diag = np.clip((1.0 - eta) * v_diag + eta * S, sig_min * sig_min, sig_max * sig_max)
        m = m_new
        gen_best = float(min(cs))
        if gen_best < run_best - 1e-12:
            run_best = gen_best; stall = 0
        else:
            stall += 1
        if stall >= stall_limit:
            break
    return gb_vec, gb_score

def add_cand(h, n, seen, starts, sigmas, sig):
    p = project_capped(h, n)
    if p is None:
        return
    for s in seen:
        if np.max(np.abs(p - s)) < 1e-7:
            return
    seen.append(p.copy()); starts.append(p); sigmas.append(sig)

def run(seed=42, **kwargs):
    n = None
    parent = None
    try:
        p = np.array(initial_h_values, dtype=np.float64).ravel()
        if p.size > 1 and p.size <= 999:
            n = int(p.size)
            if np.all(np.isfinite(p)) and np.all(p >= -1e-8) and np.all(p <= 1.0 + 1e-8):
                if abs(np.sum(p) - n / 2.0) <= 1e-6 * max(1, n):
                    parent = np.clip(p, 0, 1)
    except Exception:
        pass
    if n is None:
        n = 47
    fallback = np.full(n, 0.5)
    parent_proj = None
    if parent is not None:
        parent_proj = project_capped(parent, n)
        if parent_proj is None:
            parent = None
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    if n == 1:
        h1 = np.array([0.5])
        return h1, float(exact_score(h1, 1)), 1
    seen, starts, sigmas = [], [], []
    if parent_proj is not None:
        add_cand(parent_proj, n, seen, starts, sigmas, 0.10)
    add_cand(fallback, n, seen, starts, sigmas, 0.18)
    if parent_proj is not None:
        add_cand(parent_proj[::-1].copy(), n, seen, starts, sigmas, 0.10)
        add_cand(1.0 - parent_proj.copy(), n, seen, starts, sigmas, 0.10)
        add_cand(1.0 - parent_proj[::-1].copy(), n, seen, starts, sigmas, 0.10)
    blk = np.zeros(n); blk[:n // 2] = 1.0
    if n % 2:
        blk[n // 2] = 0.5
    add_cand(blk, n, seen, starts, sigmas, 0.28)
    add_cand(np.where(np.arange(n) % 2 == 0, 1.0, 0.0), n, seen, starts, sigmas, 0.28)
    perm = rng.permutation(n)
    bb = np.zeros(n); bb[perm[:n // 2]] = 1.0
    if n % 2:
        bb[perm[n // 2]] = 0.5
    add_cand(bb, n, seen, starts, sigmas, 0.28)
    base = parent_proj if parent_proj is not None else fallback
    for sd in (0.04, 0.12, 0.25):
        add_cand(base + rng.normal(0, sd, n), n, seen, starts, sigmas, 0.12)
    add_cand(rng.uniform(0, 1, n), n, seen, starts, sigmas, 0.20)
    if n <= 64:
        max_starts, pop_sz, max_gens, stall_lim = 5, 20, 60, 18
    elif n <= 128:
        max_starts, pop_sz, max_gens, stall_lim = 4, 16, 35, 14
    elif n <= 256:
        max_starts, pop_sz, max_gens, stall_lim = 2, 12, 20, 10
    else:
        max_starts, pop_sz, max_gens, stall_lim = 1, 8, 10, 6
    starts = starts[:max_starts]
    sigmas = sigmas[:max_starts]
    scores = [exact_score(s, n) for s in starts]
    fb_score = exact_score(fallback, n)
    all_sc = scores + [fb_score]
    gi = int(np.argmin(all_sc))
    if gi < len(scores):
        gb_vec, gb_score = starts[gi].copy(), all_sc[gi]
    else:
        gb_vec, gb_score = fallback.copy(), fb_score
    if parent_proj is not None:
        ps = exact_score(parent_proj, n)
        if np.isfinite(ps) and ps < gb_score - 1e-12:
            gb_score, gb_vec = ps, parent_proj.copy()
    cov_mode = 'full' if (n - 1) <= 64 else 'diag'
    order = np.argsort(scores, kind='stable')
    for si in order:
        gb_vec, gb_score = cem_search(starts[si], n, rng, pop_sz, max_gens, stall_lim, cov_mode, sigmas[si], gb_vec, gb_score)
    final = project_capped(gb_vec, n)
    if final is None or not is_feasible(final, n):
        final = fallback
    fs = exact_score(final, n)
    if not np.isfinite(fs):
        final = fallback; fs = fb_score
    if parent_proj is not None:
        ps = exact_score(parent_proj, n)
        if np.isfinite(ps) and ps < fs:
            final, fs = parent_proj, ps
    uf = exact_score(fallback, n)
    if uf < fs:
        final, fs = fallback, uf
    return np.asarray(final, dtype=np.float64), float(fs), int(n)
