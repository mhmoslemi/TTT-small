import numpy as np

def project_capped(z, n):
    z = np.where(np.isfinite(z), z, 0.0).astype(np.float64)
    z = np.clip(z, -1e6, 1e6)
    s = n / 2.0
    lo, hi = np.min(z) - 1.0, np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > s:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    r = s - np.sum(h)
    for idx in range(n):
        if abs(r) < 1e-14:
            break
        if r > 0:
            d = min(r, 1.0 - h[idx])
            h[idx] += d
            r -= d
        else:
            d = min(-r, h[idx])
            h[idx] -= d
            r += d
    if abs(r) > 1e-10 * max(1, n) or not np.all(np.isfinite(h)):
        return None
    return np.clip(h, 0.0, 1.0)

def is_feasible(h, n):
    if h.shape != (n,) or not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-12) or np.any(h > 1.0 + 1e-12):
        return False
    return abs(np.sum(h) - n / 2.0) <= 1e-10 * max(1, n)

def exact_score(h, n):
    if not is_feasible(h, n):
        return float('inf')
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if not np.all(np.isfinite(c)):
        return float('inf')
    return float(np.max(c))

def lag_values(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def lag_jac(h, n):
    dx = 2.0 / n
    n_lags = 2 * n - 1
    J = np.zeros((n_lags, n))
    for k in range(n_lags):
        d = k - (n - 1)
        a = max(0, -d)
        b = min(n - 1 - d, n - 1)
        m = b - a + 1
        if m <= 0:
            continue
        i_arr = np.arange(a, b + 1)
        # Term 1: i+d = q, i.e., q = i+d, valid q in range [a+d, b+d]
        q1 = i_arr + d  # these are the q values for term 1
        # contribution to J[k, q1] += dx * (1 - h[i_arr])
        # But q1 = i+d, and i ranges from a to b
        # q1 ranges from a+d to b+d
        # Since a = max(0,-d), a+d = max(d, 0)
        # Since b = min(n-1-d, n-1), b+d = min(n-1, n-1+d)
        # For d>=0: q1 from d to n-1, h[q1] = h[i+d], 1-h[i] = 1-h[q1-d]
        # For d<0: q1 from 0 to n-1+d, h[q1] = h[i+d], 1-h[i] = 1-h[q1-d]
        # In both cases: contribution to J[k, q1[t]] += dx*(1-h[i_arr[t]])
        J[k, q1] += dx * (1.0 - h[i_arr])
        # Term 2: i = q, contribution to J[k, i_arr] -= dx * h[i_arr + d]
        J[k, i_arr] -= dx * h[i_arr + d]
    return J

    return g

def slsqp_ineq_jac(z, n):

    J = np.zeros((n_lags, n+1))
