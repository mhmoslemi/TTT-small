            ox = (np.arange(old_n) + 0.5) / old_n
            nx = (np.arange(n) + 0.5) / n
            parent = project_capped(np.interp(nx, ox, old), n)
        else:
            parent = None
    if parent is None and 'n_points' in kwargs:
        try:
            v = int(kwargs['n_points'])
            if 1 <= v <= 999:
                n = v
        except Exception:
            pass
    fb = project_capped(parent, n) if parent is not None else None
    if fb is None:
        fb = np.full(n, 0.5)
    try:
        rng = np.random.default_rng(int(seed))
    except Exception:
        rng = np.random.default_rng(42)
    fs = exact_score(fb, n)
    gs, gb = fs, fb.copy()
    if n == 1:
        return fb, fs, 1
    if n <= 64:
        nr, nmi, nst = 3, 80, 7
    elif n <= 128:
        nr, nmi, nst = 2, 45, 4
    elif n <= 256:
        nr, nmi, nst = 1, 25, 3
    elif n <= 512:
        nr, nmi, nst = 1, 12, 2
    else:
        nr, nmi, nst = 0, 6, 2
    starts = build_starts(fb, n, rng, nst)
    for st in starts:
        s = exact_score(st, n)
        if s < gs - 1e-12:
            gs, gb = s, st.copy()
    scored = sorted([(exact_score(st, n), st) for st in starts])
    for _, st in scored:
        try:
            xo, so = local_lp_search(st, n, nmi)
            if so < gs - 1e-12:
                gs, gb = so, xo.copy()
        except Exception:
            continue
    if nr > 0:
        if gs < fs - 1e-5:
            stds = (0.04, 0.10, 0.18) if n <= 64 else (0.04, 0.08) if n <= 256 else (0.08,)
        else:
            stds = (0.12, 0.22, 0.08) if n <= 64 else (0.10, 0.18) if n <= 256 else (0.08,)
        for sd in stds[:nr]:
            try:
                pert = project_capped(gb + rng.normal(0, sd, n), n)
                if pert is None:
                    continue
                xo, so = local_lp_search(pert, n, nmi)
                if so < gs - 1e-12:
                    gs, gb = so, xo.copy()
            except Exception:
                continue
    final = project_capped(gb, n)
    if final is None or not is_feasible(final, n):
        final = fb.copy()
    sf = exact_score(final, n)
    ss = exact_score(fb, n)
    if sf > ss:
        final, sf = fb.copy(), ss
    if not is_feasible(final, n):
        final = np.full(n, 0.5)
        sf = exact_score(final, n)
    return final, float(sf), int(n)
